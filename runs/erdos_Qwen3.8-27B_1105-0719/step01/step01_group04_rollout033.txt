import cvxpy as cp

def trust_region_qp(h_cur, grad_matrix, values, n, lam, t_upper=None):
    dx = 2.0 / n
    h_new = cp.Variable(n)
    t = cp.Variable()
    
    # Objective: t + lam * ||h_new - h_cur||^2
    obj = cp.Minimize(t + lam * cp.sum_squares(h_new - h_cur))
    
    # Constraints
    constraints = [
        sum(h_new) == n / 2.0,
        h_new >= 0,
        h_new <= 1
    ]
    
    # Lag constraints: values[m] + grad[m] @ (h_new - h_cur) <= t
    # i.e., grad[m] @ h_new <= t - values[m] + grad[m] @ h_cur
    for m in range(len(values)):
        rhs = t - values[m] + grad_matrix[m] @ h_cur
        constraints.append(grad_matrix[m] @ h_new <= rhs)
    
    # Optional: upper bound on t
    if t_upper is not None:
        constraints.append(t <= t_upper)
    
    prob = cp.Problem(obj, constraints)
    prob.solve()
    
    if h_new.value is not None:
        return h_new.value, t.value
    return None, None

        return None, None
    if h_new.value is not None:
        return np.asarray(h_new.value, dtype=np.float64), float(t.value)
    return None, None


def trust_region_solve(h_start, max_time, lam_init=0.05, n_lags_active=None, solver=None):
    t0 = time.time()
    h = h_start.copy()
    n = len(h)
    best_h = h.copy()
    best_score = exact_score(h)
    lam = lam_init
    lam_min = 1e-5
    lam_max = 50.0
    consec_fails = 0
    max_fails = 10
    n_iters = 0

    while time.time() - t0 < max_time and consec_fails < max_fails:
        values, grads = compute_lag_data(h)
        cur_score = float(np.max(values))
        n_iters += 1

        if n_lags_active is not None and n_lags_active < len(values):
            top_idx = np.argsort(values)[::-1][:n_lags_active]
            vals_use = values[top_idx]
            grads_use = grads[top_idx, :]
        else:
            vals_use = values
            grads_use = grads

        h_prop, t_val = trust_region_qp(h, vals_use, grads_use, n, lam, solver=solver)
        if h_prop is None:
            consec_fails += 1
            lam = min(lam * 2.0, lam_max)
            continue

        h_prop = repair_feasible(h_prop)
        actual_score = exact_score(h_prop)

        if actual_score < best_score - 1e-11:
            h = h_prop
            best_h = h.copy()
            best_score = actual_score
            consec_fails = 0
            lam = max(lam * 0.7, lam_min)
        elif actual_score < cur_score - 1e-11:
            h = h_prop
            consec_fails = 0
            lam = max(lam * 0.85, lam_min)
        else:
            lam = min(lam * 1.5, lam_max)
            consec_fails += 1

    return best_h, best_score


def logsumexp_obj_and_grad(x, beta):
    h = np.exp(x) / (1.0 + np.exp(x))
    h = np.clip(h, 1e-10, 1.0 - 1e-10)
    values, grads = compute_lag_data(h)
    v_max = np.max(values)
    shifted = beta * (values - v_max)
    exp_terms = np.exp(shifted)
    log_sum = np.log(np.sum(exp_terms)) + v_max
    obj = log_sum / len(values)
    weights = exp_terms / np.sum(exp_terms)
    grad_h = np.sum(weights[:, None] * grads, axis=0)
    sig = h * (1.0 - h)
    grad_x = grad_h / sig
    return obj, grad_x


def smooth_solve(h_start, max_time, beta_start=200.0, beta_end=2000.0, solver_method='L-BFGS-B'):
    t0 = time.time()
    h = h_start.copy()
    n = len(h)
    best_h = h.copy()
    best_score = exact_score(h)
    n_lags = 2 * n - 1

    def obj_and_grad(x):
        obj, g = logsumexp_obj_and_grad(x, beta)
        h_sig = 1.0 / (1.0 + np.exp(-x))
        h_sig = np.clip(h_sig, 1e-10, 1.0 - 1e-10)
        penalty = 1000.0 * (np.sum(h_sig) - n / 2.0) ** 2
        pen_grad = 2000.0 * (np.sum(h_sig) - n / 2.0) * h_sig * (1.0 - h_sig)
        total = obj + penalty / 1.0
        total_g = g + pen_grad
        return total, total_g

    beta = beta_start
    x0 = np.log(h / (1.0 - h + 1e-10))
    x0 = np.clip(x0, -10, 10)

    while time.time() - t0 < max_time:
        result = minimize(obj_and_grad, x0, jac=True, method=solver_method,
                          options={'maxiter': 200, 'ftol': 1e-14, 'gtol': 1e-10})
        if result.x is not None:
            h_trial = 1.0 / (1.0 + np.exp(-result.x))
            h_trial = repair_feasible(h_trial)
            s_trial = exact_score(h_trial)
            if s_trial < best_score - 1e-11:
                best_h = h_trial.copy()
                best_score = s_trial
                h = h_trial
                x0 = np.log(h / (1.0 - h + 1e-10))
                x0 = np.clip(x0, -10, 10)
        beta *= 3.0
        if beta > beta_end:
            break

    return best_h, best_score


def active_descent(h, max_steps=50, time_limit=30.0):
    t0 = time.time()
    h = repair_feasible(h)
    best = h.copy()
    best_score = exact_score(best)
    for step in range(max_steps):
        if time.time() - t0 > time_limit:
            break
        values, grads = compute_lag_data(best)
        c = np.max(values)
        n_active = min(20, len(values))
        active_idx = np.argsort(values)[::-1][:n_active]
        gaps = np.maximum(c - values[active_idx], 0.0)
        temp = 0.003 * c + 1e-10
        weights = np.exp(-gaps / temp)
        weights /= np.sum(weights) + 1e-15
        direction = -np.sum(weights[:, None] * grads[active_idx, :], axis=0)
        direction -= np.mean(direction)
        improved = False
        for alpha in [0.03, 0.015, 0.008, 0.004, 0.002, 0.001, 0.0005]:
            h_trial = project_onto_constraint(best + alpha * direction, len(best))
            score = exact_score(h_trial)
            if score < best_score - 1e-13:
                best = h_trial
                best_score = score
                improved = True
                break
        if not improved:
            break
    return best, best_score


def multi_start_trust(h_base, n_starts, max_time_each, rng):
    best_h = h_base.copy()
    best_score = exact_score(h_base)
    n = len(h_base)
    for i in range(n_starts):
        elapsed = time.time() - _mt_start[0]
        if elapsed + max_time_each > _mt_budget[0]:
            break
        eps = rng.uniform(0.003, 0.015)
        z = rng.randn(n)
        z -= z.mean()
        z /= (np.std(z) + 1e-15)
        h_pert = repair_feasible(h_base + eps * z)
        s_pert = exact_score(h_pert)
        if s_pert < best_score + 0.005:
            h_sol, s_sol = trust_region_solve(h_pert, max_time=max_time_each, lam_init=0.03)
            if s_sol < best_score - 1e-11:
                best_h = h_sol
                best_score = s_sol
    return best_h, best_score


_mt_start = [0.0]
_mt_budget = [0.0]


def symmetrize(h):
    return (h + h[::-1]) / 2.0


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)

    def tl():
        return budget_s - (time.time() - t_start)

    try:
        parent = np.asarray(initial_h_values, dtype=np.float64)
    except Exception:
        parent = None

    if parent is not None and len(parent) > 0:
        best_h = repair_feasible(parent)
    else:
        best_h = np.full(140, 0.5)

    h_sym = repair_feasible(symmetrize(best_h))
    s_sym = exact_score(h_sym)
    if s_sym < exact_score(best_h) - 1e-12:
        best_h = h_sym

    best_score = exact_score(best_h)
    n_cur = len(best_h)

    # Phase 1: Trust region QP with all lags
    if HAVE_CP and tl() > 80:
        h_sol, s_sol = trust_region_solve(best_h, max_time=min(180, tl() * 0.35), lam_init=0.05)
        if s_sol < best_score - 1e-12:
            best_h = h_sol
            best_score = s_sol
            n_cur = len(h_sol)

    # Phase 2: Trust region with top active lags only (faster convergence)
    if HAVE_CP and tl() > 60:
        values, _ = compute_lag_data(best_h)
        n_top = min(60, 2 * n_cur - 1)
        h_sol, s_sol = trust_region_solve(best_h, max_time=min(120, tl() * 0.3),
                                          lam_init=0.03, n_lags_active=n_top)
        if s_sol < best_score - 1e-12:
            best_h = h_sol
            best_score = s_sol
            n_cur = len(h_sol)

    # Phase 3: Log-sum-exp smooth approach
    if tl() > 40:
        h_sol, s_sol = smooth_solve(best_h, max_time=min(60, tl() * 0.3),
                                    beta_start=100.0, beta_end=3000.0)
        if s_sol < best_score - 1e-12:
            best_h = h_sol
            best_score = s_sol
            n_cur = len(h_sol)

    # Phase 4: Active descent polish
    if tl() > 15:
        h_pol, s_pol = active_descent(best_h, max_steps=60, time_limit=min(25, tl() * 0.25))
        if s_pol < best_score - 1e-12:
            best_h = h_pol
            best_score = s_pol
            n_cur = len(h_pol)

    # Phase 5: Another trust region pass with different lambda
    if HAVE_CP and tl() > 50:
        h_sol, s_sol = trust_region_solve(best_h, max_time=min(100, tl() * 0.35), lam_init=0.15)
        if s_sol < best_score - 1e-12:
            best_h = h_sol
            best_score = s_sol
            n_cur = len(h_sol)

    # Phase 6: Multi-start perturbations
    if tl() > 40:
        _mt_start[0] = time.time()
        _mt_budget[0] = tl() - 5
        n_starts = max(2, int(_mt_budget[0] / (40 + 5)))
        max_each = max(10, min(50, _mt_budget[0] / (n_starts + 1) - 5))
        h_ms, s_ms = multi_start_trust(best_h, n_starts, max_each, rng)
        if s_ms < best_score - 1e-12:
            best_h = h_ms
            best_score = s_ms
            n_cur = len(h_ms)

    # Phase 7: Dimension exploration
    if tl() > 50:
        dims_to_try = []
        base_n = len(best_h)
        for dn in [-20, -10, 10, 20, -40, 40]:
            nd = base_n + dn
            if 60 <= nd <= 200:
                dims_to_try.append(nd)
        # Also try specific good dimensions
        for nd in [100, 120, 130, 160, 180]:
            if nd not in dims_to_try and 60 <= nd <= 200:
                dims_to_try.append(nd)
        dims_to_try = sorted(set(dims_to_try))
        for nd in dims_to_try:
            if tl() < 35:
                break
            h_interp = repair_feasible(interpolate_to_dimension(best_h, nd))
            s_interp = exact_score(h_interp)
            if s_interp < best_score + 0.008:
                time_alloc = min(50, tl() * 0.25)
                if HAVE_CP:
                    h_sol, s_sol = trust_region_solve(h_interp, max_time=time_alloc, lam_init=0.04)
                else:
                    h_sol, s_sol = active_descent(h_interp, max_steps=40, time_limit=time_alloc)
                if s_sol < best_score - 1e-12:
                    best_h = h_sol
                    best_score = s_sol
                    n_cur = len(h_sol)

    # Phase 8: Final active descent + trust region polish
    if tl() > 15:
        h_pol, s_pol = active_descent(best_h, max_steps=40, time_limit=min(20, tl() * 0.3))
        if s_pol < best_score - 1e-12:
            best_h = h_pol
            best_score = s_pol

    if HAVE_CP and tl() > 25:
        h_sol, s_sol = trust_region_solve(best_h, max_time=min(30, tl() * 0.4), lam_init=0.02)
        if s_sol < best_score - 1e-12:
            best_h = h_sol
            best_score = s_sol

    # Final validation
    best_h = repair_feasible(best_h)
    best_score = exact_score(best_h)

    return best_h, best_score, len(best_h)
