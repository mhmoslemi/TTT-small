from scipy.optimize import minimize, linprog

def exact_score(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))

def get_profile(h):
    dx = 2.0 / len(h)
    return np.correlate(h, 1.0 - h, mode="full") * dx

def get_lag_gradients(h):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    G = np.zeros((M, n))
    for k in range(M):
        delta = k - (n - 1)
        if delta >= 0:
            if delta < n:
                L = n - delta
                G[k, delta:n] += dx * (1.0 - h[0:L])
                G[k, 0:L] -= dx * h[delta:n]
        else:
            d = -delta
            if d < n:
                L = n - d
                G[k, 0:L] += dx * (1.0 - h[d:n])
                G[k, d:n] -= dx * h[0:L]
    return G

def repair_feasible(h, target=None):
    h = np.array(h, dtype=np.float64).copy()
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)

    if diff > 0:

        active_all = np.argsort(p)[::-1][:n_constraints]
    
    def obj(x):
        return x[n]
    
    def obj_jac(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g

        t = x[n]
        eq = h.sum() - n / 2.0
        prof = get_profile(h)

        return np.concatenate([[eq], ineq])

        J[1:, n] = 1.0
        return J

def pairwise_swap(h, i, j):
    """Find optimal swap between coordinates i and j."""
    n = len(h)
    M = 2 * n - 1
    p = get_profile(h)
    G = get_lag_gradients(h)
    
    gamma = G[:, i] - G[:, j]  # M-vector
    # f_k changes as p[k] + gamma[k] * epsilon
    
    # Feasible range for epsilon
    eps_min = max(-h[i], -(n/2 - h.sum() + h[i] - h[j]))  # h[i] + eps >= 0
    # Actually: h[i] + eps >= 0 => eps >= -h[i]
    #           h[j] - eps >= 0 => eps <= h[j]
    #           h[i] + eps <= 1 => eps <= 1 - h[i]
    #           h[j] - eps <= 1 => eps >= h[j] - 1
    eps_min = max(-h[i], h[j] - 1.0)
    eps_max = min(h[j], 1.0 - h[i])
    
    if eps_min >= eps_max:
        return 0.0, p.max()
    
    # Find min of max_k (p[k] + gamma[k] * eps) over [eps_min, eps_max]
    # This is the minimum of a convex piecewise-linear function
    
    # The minimum occurs at:
    # - A breakpoint where the active set changes
    # - Or at a boundary
    
    # Find all breakpoints (intersections of pairs of lines)
    # For lines a and b: p[a] + gamma[a]*eps = p[b] + gamma[b]*eps
    # => eps = (p[b] - p[a]) / (gamma[a] - gamma[b])
    
    candidates = [eps_min, eps_max, 0.0]
    
    # Add breakpoints
    for a in range(M):
        for b in range(a+1, M):
            if abs(gamma[a] - gamma[b]) > 1e-15:
                eps_AB = (p[b] - p[a]) / (gamma[a] - gamma[b])
                if eps_min <= eps_AB <= eps_max:
                    candidates.append(eps_AB)
    
    # Evaluate at all candidates
    best_eps = 0.0
    best_val = float('inf')
    for eps_c in candidates:
        if eps_min <= eps_c <= eps_max:
            val = np.max(p + gamma * eps_c)
            if val < best_val:
                best_val = val
                best_eps = eps_c
    
    return best_eps, best_val
