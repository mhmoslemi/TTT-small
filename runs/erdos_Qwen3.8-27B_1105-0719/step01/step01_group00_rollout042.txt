def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-10:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)


def make_sym_h(q, n):
    m = n // 2
    h = np.empty(n)
    h[:m] = q
    h[m:] = 1.0 - q[::-1]
    return h


def make_q_from_h(h, n):
    m = n // 2
    return 0.5 * (h[:m] + 1.0 - h[n - 1::-1])


def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def exact_score(h):
    n = len(h)
    return compute_corrs(h).max() * (2.0 / n)


def full_grad_lag(h, r, n):
    d = r - (n - 1)
    G = np.zeros(n)
    if d >= 0:
        G[:n - d] += (1.0 - h[d:])
        G[d:] -= h[:n - d]
    else:
        ad = -d
        G[ad:] += (1.0 - h[:n - ad])
        G[:n - ad] -= h[ad:]
    return G


def q_grad_lag(h, r, n):
    G = full_grad_lag(h, r, n)
    m = n // 2
    return G[:m] - G[n - 1::-1][:m]


class SmoothQ:
    def __init__(self, n, beta):
        self.n = n
        self.m = n // 2
        self.dx = 2.0 / n
        self.beta = beta
        self.max_lag = 2 * n - 1

    def fun(self, q):
        h = make_sym_h(np.clip(q, 0, 1), self.n)
        corrs = compute_corrs(h)
        vals = corrs * self.dx
        return logsumexp(self.beta * vals) / self.beta

    def jac(self, q):

def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-10:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)


def make_sym_h(q, n):
    m = n // 2
    h = np.empty(n)
    h[:m] = q
    h[m:] = 1.0 - q[::-1]
    return h

def full_grad_lag(h, r, n):
    d = r - (n - 1)
    G = np.zeros(n)
    if d >= 0:
        G[:n - d] += (1.0 - h[d:])
        G[d:] -= h[:n - d]
    else:
        ad = -d
        G[ad:] += (1.0 - h[:n - ad])
        G[:n - ad] -= h[ad:]
    return G

        h = make_sym_h(q, n)
        corrs = compute_corrs(h)
        c5_val = corrs.max() * dx
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_q = q.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 3000:
            break

        active = np.argsort(corrs)[-12:]
        grad = np.zeros(m)
        for r in active:
            grad += q_grad_lag(h, r, n)
        grad /= len(active)
        grad *= dx
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        q = np.clip(q - alpha * grad, 0, 1)
        if it % 500 == 0:
            alpha *= 0.7
    return best_q, best_c5


def interp_to_n(h_src, n_new):
    x_src = np.linspace(0, 2, len(h_src))
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_src, h_src)
    return project_feasible(h_new, n_new / 2.0)


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')

    def rem():
        return budget_s - (time.time() - t_start)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    h_init = project_feasible(h_init, n0 / 2.0)
    c5_init = exact_score(h_init)
    best_h = h_init.copy()
    best_c5 = c5_init

    n = n0
    if n % 2 != 0:
        n = n + 1
        h_init = interp_to_n(h_init, n)
        best_c5 = exact_score(h_init)
        best_h = h_init.copy()

    m = n // 2
    q0 = make_q_from_h(best_h, n)
    q0 = np.clip(q0, 0, 1)

    # Phase 1: Smooth q-space optimization
    for beta in [40, 80, 160, 320, 500]:
        if rem() < 25:
            break
        q_opt = slsqp_smooth(q0, n, beta, maxiter=350)
        h_opt = make_sym_h(q_opt, n)
        c5_opt = exact_score(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
            q0 = q_opt

    # Phase 2: Epigraph with active set
    if rem() > 50:
        h_cur = make_sym_h(q0, n)
        corrs_cur = compute_corrs(h_cur)
        active0 = get_active_set(corrs_cur, 2.0 / n, top_k=40, thresh_frac=0.90)
        q_epi, c5_epi = epi_with_expansion(q0, n, t_start, budget_s, max_rounds=5)
        h_epi = make_sym_h(q_epi, n)
        if c5_epi < best_c5:
            best_c5 = c5_epi
            best_h = h_epi
            q0 = q_epi

    # Phase 3: Subgradient
    if rem() > 30:
        q_sg, c5_sg = subgrad_q(q0, n, max_iter=4000, t_start=t_start, budget_s=budget_s)
        h_sg = make_sym_h(q_sg, n)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg
            q0 = q_sg

    # Phase 4: Another epigraph pass
    if rem() > 40:
        q_e2, c5_e2 = epi_with_expansion(q0, n, t_start, budget_s, max_rounds=3)
        h_e2 = make_sym_h(q_e2, n)
        if c5_e2 < best_c5:
            best_c5 = c5_e2
            best_h = h_e2
            q0 = q_e2

    # Phase 5: Random restarts
    for trial in range(25):
        if rem() < 12:
            break
        scale = 0.08 * max(0.3, 1.0 - trial / 25.0)
        q_rand = np.clip(q0 + rng.randn(m) * scale, 0, 1)
        q_opt = slsqp_smooth(q_rand, n, 200, maxiter=200)
        h_opt = make_sym_h(q_opt, n)
        c5_opt = exact_score(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
            q0 = q_opt
            if rem() > 20:
                q_e, c5_e = epi_with_expansion(q0, n, t_start, budget_s, max_rounds=2)
                h_e = make_sym_h(q_e, n)
                if c5_e < best_c5:
                    best_c5 = c5_e
                    best_h = h_e
                    q0 = q_e

    # Phase 6: Try structured starts (sinusoidal patterns)
    if rem() > 30:
        for freq in [1.0, 1.5, 2.0, 3.0]:
            if rem() < 15:
                break
            x = np.arange(m, dtype=float)
            q_struct = 0.5 + 0.3 * np.sin(2.0 * np.pi * freq * x / m)
            q_struct = np.clip(q_struct, 0, 1)
            q_opt = slsqp_smooth(q_struct, n, 200, maxiter=250)
            h_opt = make_sym_h(q_opt, n)
            c5_opt = exact_score(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
                q0 = q_opt

    # Phase 7: Different n values
    for n_try in [n - 2, n + 2, n - 4, n + 4, 128, 124]:
        if rem() < 35:
            break
        if n_try < 40 or n_try > 200 or n_try % 2 != 0:
            continue
        h_interp = interp_to_n(best_h, n_try)
        q_try = make_q_from_h(h_interp, n_try)
        q_try = np.clip(q_try, 0, 1)
        q_opt = slsqp_smooth(q_try, n_try, 200, maxiter=250)
        h_opt = make_sym_h(q_opt, n_try)
        c5_opt = exact_score(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
            if rem() > 25:
                q_e, c5_e = epi_with_expansion(q_opt, n_try, t_start, budget_s, max_rounds=3)
                h_e = make_sym_h(q_e, n_try)
                if c5_e < best_c5:
                    best_c5 = c5_e
                    best_h = h_e

    # Phase 8: Full-dimensional SLSQP from best symmetric solution
    if rem() > 50:
        n_b = len(best_h)
        dx_b = 2.0 / n_b
        target_b = n_b / 2.0
        h_b = project_feasible(best_h, target_b)

        class FullSmooth:
            def __init__(self, nn, bb):
                self.nn = nn
                self.bb = bb
                self.dxb = 2.0 / nn
                self.maxl = 2 * nn - 1
                self.act = get_active_set(compute_corrs(h_b), self.dxb, 60, 0.90)

            def fun(self, x):
                hx = np.clip(x, 0, 1)
                corrs = compute_corrs(hx)
                vals = corrs[self.act] * self.dxb
                return logsumexp(self.bb * vals) / self.bb

            def jac(self, x):
                hx = np.clip(x, 0, 1)
                corrs = compute_corrs(hx)
                vals = corrs[self.act] * self.dxb * self.bb
                mx = vals.max()
                w = np.exp(vals - mx)
                ws = w.sum()
                if ws < 1e-30:
                    w = np.ones(len(w)) / len(w)
                else:
                    w /= ws
                grad = np.zeros(self.nn)
                for i, r in enumerate(self.act):
                    if w[i] < 1e-13:
                        continue
                    grad += w[i] * full_grad_lag(hx, r, self.nn)
                return grad * self.dxb

        for beta in [400, 800]:
            if rem() < 20:
                break
            fs = FullSmooth(n_b, beta)
            bounds_f = [(0.0, 1.0)] * n_b
            try:
                res_f = minimize(fs.fun, h_b, jac=fs.jac, method='SLSQP',
                                 bounds=bounds_f,
                                 options={'maxiter': 400, 'ftol': 1e-14})
                h_f = project_feasible(res_f.x, target_b)
                c5_f = exact_score(h_f)
                if c5_f < best_c5:
                    best_c5 = c5_f
                    best_h = h_f
            except Exception:
                pass

    # Phase 9: Final epigraph polish
    if rem() > 15 and len(best_h) % 2 == 0:
        n_f = len(best_h)
        m_f = n_f // 2
        q_f = make_q_from_h(best_h, n_f)
        q_f = np.clip(q_f, 0, 1)
        q_fin, c5_fin = epi_with_expansion(q_f, n_f, t_start, budget_s, max_rounds=3)
        h_fin = make_sym_h(q_fin, n_f)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin

    # Final subgradient if time
    if rem() > 10 and len(best_h) % 2 == 0:
        n_s = len(best_h)
        q_s = make_q_from_h(best_h, n_s)
        q_s = np.clip(q_s, 0, 1)
        q_sg2, c5_sg2 = subgrad_q(q_s, n_s, max_iter=2000, t_start=t_start, budget_s=budget_s)
        h_sg2 = make_sym_h(q_sg2, n_s)
        if c5_sg2 < best_c5:
            best_c5 = c5_sg2
            best_h = h_sg2

    # Final validation
    if best_h is None:
        best_h = h_init
        best_c5 = c5_init
    n_final = len(best_h)
    target_final = n_final / 2.0
    best_h = project_feasible(best_h, target_final)
    best_h = np.clip(best_h, 0, 1)
    best_c5 = exact_score(best_h)

    return best_h, best_c5, n_final
