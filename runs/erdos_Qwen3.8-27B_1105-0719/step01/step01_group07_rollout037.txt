                val -= h[m2]

                    best_improve = improve
                    best_swap = (i, j)
        if best_swap is not None:
            i, j = best_swap
            b[i] = 0.0
            b[j] = 1.0
            profile = compute_profile(b)
            curr_max = float(np.max(profile))
            ones = np.where(b == 1.0)[0].tolist()
            zeros = np.where(b == 0.0)[0].tolist()
        else:
            break
    return b, curr_max

def sa_binary(b, deadline, t_start=1.0, t_end=0.005, max_rounds=20000, seed=42):
    n = len(b)
    s_arr = np.arange(-(n - 1), n)
    profile = compute_profile(b).copy()
    curr_max = float(np.max(profile))
    best_b = b.copy()
    best_max = curr_max
    rng = np.random.RandomState(seed)
    ones = np.where(b == 1.0)[0]
    zeros = np.where(b == 0.0)[0]
    for rnd in range(max_rounds):
        if time.time() > deadline - 5:
            break
        T = t_start * (t_end / t_start) ** (rnd / max_rounds)
        i = int(rng.choice(ones))
        j = int(rng.choice(zeros))
        delta = fast_swap_delta(profile, b, i, j, n, s_arr)
        new_max = float(np.max(profile + delta))
        dC = new_max - curr_max
        if dC < 0 or rng.uniform() < np.exp(-dC / max(T, 1e-12)):
            b[i] = 0.0
            b[j] = 1.0
            profile += delta
            curr_max = new_max
            ones = np.where(b == 1.0)[0]
            zeros = np.where(b == 0.0)[0]
            if curr_max < best_max:
                best_max = curr_max
                best_b = b.copy()
    return best_b, best_max

def ils_binary(b, deadline, max_iters=300, seed=777):
    n = len(b)
    best_b = b.copy()
    best_max = exact_objective(b)
    rng = np.random.RandomState(seed)
    for it in range(max_iters):
        if time.time() > deadline - 8:
            break
        perturbed = best_b.copy()
        k = rng.randint(2, 6)
        for _ in range(k):
            ones_idx = np.where(perturbed == 1.0)[0]
            zeros_idx = np.where(perturbed == 0.0)[0]
            if len(ones_idx) == 0 or len(zeros_idx) == 0:
                break
            i = int(rng.choice(ones_idx))
            j = int(rng.choice(zeros_idx))
            perturbed[i] = 0.0
            perturbed[j] = 1.0
        perturbed, local_max = greedy_swap_binary(perturbed, deadline, max_rounds=500)
        if local_max < best_max - 1e-12:
            best_max = local_max
            best_b = perturbed.copy()
    return best_b, best_max

def make_random_binary(n, rng):
    b = np.zeros(n, dtype=np.float64)
    m = n // 2
    idx = rng.choice(n, size=m, replace=False)
    b[idx] = 1.0
    return b

def make_binary_pattern(n, rng, mode):
    b = np.zeros(n, dtype=np.float64)
    m = n // 2
    if mode == 0:
        idx = rng.choice(n, size=m, replace=False)
        b[idx] = 1.0
    elif mode == 1:
        b[:m] = 1.0
        b = b[rng.permutation(n)]
    elif mode == 2:
        b[::2] = 1.0
        if n % 4 == 2:
            b = b[rng.permutation(n)]
    elif mode == 3:
        x = np.linspace(0, 1, n)
        freq = rng.choice([2, 3, 4, 5, 6])
        phase = rng.uniform(0, 2 * np.pi)
        vals = 0.5 + 0.49 * np.sin(2 * np.pi * freq * x + phase)
        top_m = np.argsort(vals)[::-1][:m]
        b[top_m] = 1.0
    elif mode == 4:
        block = max(1, n // 8)
        pos = rng.choice(n - block, size=m // block + 1, replace=False)
        for p in pos:
            end = min(p + block, n)
            b[p:end] = 1.0
        total = int(b.sum())
        if total > m:
            one_idx = np.where(b == 1.0)[0]
            extra = one_idx[m:]
            b[extra[:total - m]] = 0.0
        elif total < m:
            zero_idx = np.where(b == 0.0)[0]
            need = m - total
            fill = zero_idx[:need]
            b[fill] = 1.0
    else:
        b[:m] = 1.0
    return b

def lse_obj_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    m = float(np.max(profile))
    shifted = np.minimum((profile - m) / tau, 0.0)
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    weights = exp_vals / sum_exp
    obj = m + tau * np.log(sum_exp)
    grad = np.zeros(n)
    for j in range(2 * n - 1):
        w = weights[j]
        if w < 1e-13:
            continue
        s = j - (n - 1)
        if s >= 0:
            k1 = np.arange(s, n)
            grad[k1] += w * dx * (1.0 - h[k1 - s])
            k2 = np.arange(0, n - s)
            grad[k2] -= w * dx * h[k2 + s]
        else:
            a = -s
            k1 = np.arange(0, n - a)
            grad[k1] += w * dx * (1.0 - h[k1 + a])
            k2 = np.arange(a, n)
            grad[k2] -= w * dx * h[k2 - a]
    return obj, grad

def slsqp_lse(h0, tau, maxiter=300):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    target = n / 2.0
    def obj_fun(x):
        return lse_obj_grad(x, tau)[0]
    def obj_grad(x):
        return lse_obj_grad(x, tau)[1]
    def con_fun(x):
        return np.sum(x) - target
    def con_jac(x):
        return np.ones(len(x))
    constraints = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        res = minimize(obj_fun, h0.copy(), method='SLSQP', jac=obj_grad,
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        h_new = project_box_sum(np.array(res.x, dtype=np.float64))
        return h_new
    except Exception:
        return h0.copy()

def epigraph_solve(h0, deadline):
    n = len(h0)
    dx = 2.0 / n
    s_arr = np.arange(-(n - 1), n)
    profile_init = compute_profile(h0)
    t0 = float(np.max(profile_init)) * 1.01
    x0 = np.concatenate([h0, [t0]])
    full_profile = np.correlate(h0, 1.0 - h0, mode='full')
    
    def efun(x):
        return x[-1]
    def egrad(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    
    cons = []
    cons.append({'type': 'eq',
                 'fun': lambda x: np.sum(x[:n]) - n / 2.0,
                 'jac': lambda x: np.concatenate([np.ones(n), [0.0]])})
    
    active_k = np.where(full_profile > full_profile.max() - 0.15)[0]
    for k in active_k:
        s = k - (n - 1)
        a = max(0, -s)
        b_val = min(n - 1, n - 1 - s)
        def ineq_fun(x, k=k, s=s, a=a, b_val=b_val):
            h = x[:n]
            prof_k = float(np.dot(h[a:b_val+1], 1.0 - h[a+s:b_val+s+1])) * dx
            return x[-1] - prof_k
        def ineq_jac(x, k=k, s=s, a=a, b_val=b_val):
            h = x[:n]
            g = np.zeros(n + 1)
            g[-1] = 1.0
            for i in range(a, b_val + 1):
                d = dx * (1.0 - h[i + s])
                m2 = i - s
                if 0 <= m2 < n and a <= m2 <= b_val:
                    d -= dx * h[m2]
                g[i] = -d
            return g
        cons.append({'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac})
    
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    try:
        res = minimize(efun, x0, method='SLSQP', jac=egrad,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': min(200, n * 3), 'ftol': 1e-10})
        h_new = project_box_sum(np.array(res.x[:n], dtype=np.float64))
        return h_new
    except Exception:
        return h0.copy()

def transfer_to_size(h_parent, n_target):
    n_src = len(h_parent)
    if n_src == n_target:
        return project_box_sum(h_parent.copy())
    x_src = (np.arange(n_src) + 0.5) / n_src
    x_tgt = (np.arange(n_target) + 0.5) / n_tgt_tmp(n_target)
    h_new = np.interp(x_tgt, x_src, h_parent)
    return project_box_sum(h_new)

def n_tgt_tmp(n):
    return n / n * 1.0 + 0.0

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    deadline = t_start + budget_s
    rng = np.random.RandomState(seed)
    
    best_h = None
    best_c = float('inf')
    best_n = 0
    
    parent_h = None
    try:
        parent_h = np.array(globals()['initial_h_values'], dtype=np.float64).copy()
        if len(parent_h) > 0:
            parent_h = project_box_sum(parent_h)
    except Exception:
        pass
    
    if parent_h is None:
        parent_h = make_random_binary(48, rng)
    
    def update_best(h, n):
        nonlocal best_h, best_c, best_n
        c = exact_objective(h)
        if c < best_c - 1e-12:
            best_c = c
            best_h = h.copy()
            best_n = n
        return c
    
    # Phase 1: Binary search at multiple sizes
    sizes_to_try = [48, 56, 64, 80, 96]
    if parent_h is not None and len(parent_h) not in sizes_to_try:
        sizes_to_try.insert(0, len(parent_h))
    
    for n in sizes_to_try:
        if time.time() > deadline - 30:
            break
        if n % 2 != 0:
            continue
        if n < 8 or n > 128:
            continue
        
        phase_deadline = deadline - 20
        # Generate diverse binary starting points
        candidates = []
        for mode in range(6):
            b = make_binary_pattern(n, rng, mode)
            candidates.append(b)
        # Also from parent if available
        if parent_h is not None and len(parent_h) == n:
            bt = binary_threshold(parent_h)
            if bt is not None:
                candidates.append(bt)
        elif parent_h is not None:
            # Transfer parent to this size
            x_src = (np.arange(len(parent_h)) + 0.5) / len(parent_h)
            x_tgt = (np.arange(n) + 0.5) / n
            h_trans = np.interp(x_tgt, x_src, parent_h)
            h_trans = project_box_sum(h_trans)
            bt = binary_threshold(h_trans)
            if bt is not None:
                candidates.append(bt)
        
        # Refine each candidate
        local_best_b = None
        local_best_c = float('inf')
        for c in candidates:
            if time.time() > phase_deadline:
                break
            c_init = exact_objective(c)
            b_ref, c_ref = greedy_swap_binary(c, phase_deadline, max_rounds=500)
            if c_ref < local_best_c:
                local_best_c = c_ref
                local_best_b = b_ref.copy()
            if c_ref < best_c - 1e-12:
                best_c = c_ref
                best_h = b_ref.copy()
                best_n = n
        
        if local_best_b is not None and time.time() < phase_deadline:
            # ILS
            b_ils, c_ils = ils_binary(local_best_b, phase_deadline, max_iters=100, seed=seed + n)
            if c_ils < local_best_c:
                local_best_c = c_ils
                local_best_b = b_ils.copy()
            if local_best_c < best_c - 1e-12:
                best_c = local_best_c
                best_h = local_best_b.copy()
                best_n = n
        
        if local_best_b is not None and time.time() < deadline - 15:
            # SA
            b_sa, c_sa = sa_binary(local_best_b, deadline - 5, t_start=0.5, t_end=0.002,
                                   max_rounds=10000, seed=seed + 100 + n)
            if c_sa < best_c - 1e-12:
                best_c = c_sa
                best_h = b_sa.copy()
                best_n = n
            # Greedy after SA
            if c_sa < local_best_c:
                local_best_b = b_sa.copy()
                local_best_c = c_sa
            b_g, c_g = greedy_swap_binary(b_sa, deadline - 3, max_rounds=300)
            if c_g < best_c - 1e-12:
                best_c = c_g
                best_h = b_g.copy()
                best_n = n
    
    # Phase 2: Continuous optimization from parent
    if time.time() < deadline - 40:
        n_cont = len(parent_h) if parent_h is not None else 48
        h = project_box_sum(parent_h.copy())
        curr_c = exact_objective(h)
        
        taus = [0.02, 0.006, 0.002, 0.0006, 0.0002]
        for tau in taus:
            if time.time() > deadline - 25:
                break
            h_new = slsqp_lse(h, tau, maxiter=min(200, n_cont * 4))
            c_new = exact_objective(h_new)
            if c_new < curr_c - 1e-12:
                h = h_new
                curr_c = c_new
        
        if curr_c < best_c - 1e-12:
            best_c = curr_c
            best_h = h.copy()
            best_n = n_cont
        
        # Epigraph polish
        if time.time() < deadline - 20:
            h_epi = epigraph_solve(h, deadline)
            c_epi = exact_objective(h_epi)
            if c_epi < best_c - 1e-12:
                best_c = c_epi
                best_h = h_epi.copy()
                best_n = n_cont
        
        # Threshold continuous to binary and refine
        if n_cont % 2 == 0 and time.time() < deadline - 10:
            b_t = binary_threshold(h)
            if b_t is not None:
                b_g, c_g = greedy_swap_binary(b_t, deadline - 3, max_rounds=500)
                if c_g < best_c - 1e-12:
                    best_c = c_g
                    best_h = b_g.copy()
                    best_n = n_cont
    
    # Phase 3: Try more sizes with focused binary search if time permits
    if time.time() < deadline - 15:
        extra_sizes = [40, 52, 60, 72, 88, 112]
        for n in extra_sizes:
            if time.time() > deadline - 8:
                break
            if n % 2 != 0 or n < 8 or n > 128:
                continue
            b = make_random_binary(n, rng)
            b, c = greedy_swap_binary(b, deadline - 5, max_rounds=300)
            if c < best_c - 1e-12:
                best_c = c
                best_h = b.copy()
                best_n = n
            # One more ILS round
            if time.time() < deadline - 5:
                b2, c2 = ils_binary(b, deadline - 3, max_iters=50, seed=seed + 500 + n)
                if c2 < best_c - 1e-12:
                    best_c = c2
                    best_h = b2.copy()
                    best_n = n
    
    # Final validation
    if best_h is not None:
        best_h = project_box_sum(best_h)
        best_c = exact_objective(best_h)
        best_n = len(best_h)
        if abs(np.sum(best_h) - best_n / 2.0) > 1e-8:
            best_h = project_box_sum(best_h)
            best_c = exact_objective(best_h)
    
    if best_h is None:
        best_h = make_random_binary(48, rng)
        best_c = exact_objective(best_h)
        best_n = 48
    
    return best_h, best_c, best_n
