import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def get_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")


def lse_obj(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = get_corr(h)
    m = np.max(corr)
    shifted = beta * (corr - m)
    shifted = np.clip(shifted, -60, 60)
    exps = np.exp(shifted)
    s = np.sum(exps)
    if s < 1e-30:
        return m * dx
    val = m + np.log(s) / beta
    return val * dx


def lse_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = get_corr(h)
    m = np.max(corr)
    shifted = beta * (corr - m)
    shifted = np.clip(shifted, -60, 60)
    exps = np.exp(shifted)
    s = np.sum(exps)
    if s < 1e-30:
        return np.zeros(n)
    weights = exps / s

    grad = np.zeros(n)
    for i in range(2 * n - 1):
        k = i - (n - 1)
        w = weights[i]
        if w < 1e-14:
            continue
        lo1 = max(0, k)
        hi1 = min(n - 1, n - 1 + k)
        if lo1 <= hi1:
            grad[lo1:hi1 + 1] += w * (1.0 - h[lo1 - k:hi1 - k + 1])
        lo2 = max(0, -k)
        hi2 = min(n - 1, n - 1 - k)
        if lo2 <= hi2:
            grad[lo2:hi2 + 1] -= w * h[lo2 + k:hi2 + k + 1]
    return grad * dx


def project_feasible(y, n):
    target = n / 2.0
    y = np.clip(y, -2.0, 3.0)
    lo = np.min(y) - 1.0
    hi = np.max(y) + 1.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    h = np.clip(y - t, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-10:
        eligible = np.where((h > 1e-8) & (h < 1.0 - 1e-8))[0]
        if len(eligible) > 0:
            h[eligible] += residual / len(eligible)
            h = np.clip(h, 0.0, 1.0)
    return h


def eq_con_fun(h):
    return np.sum(h) - len(h) / 2.0


def eq_con_jac(h):
    return np.ones(len(h))


def run_slsqp(h0, beta, max_iter=100):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_con_fun, 'jac': eq_con_jac}]
    res = minimize(
        lse_obj, h0, args=(beta,),
        jac=lse_grad,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': max_iter, 'ftol': 1e-13}
    )
    if res.x is not None:
        h_out = np.clip(res.x, 0.0, 1.0)
        if abs(np.sum(h_out) - n / 2.0) > 1e-8:
            h_out = project_feasible(h_out, n)
        return h_out
    return h0


def projected_gradient_descent(h0, beta, n_iter=500, alpha_init=0.005):
    h = h0.copy()
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    alpha = alpha_init

    for it in range(n_iter):
        obj_val = lse_obj(h, beta)
        grad_val = lse_grad(h, beta)

        h_new = h - alpha * grad_val
        h_new = project_feasible(h_new, n)

        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

        obj_new = lse_obj(h_new, beta)
        if obj_new > obj_val * (1 + 1e-6) + 1e-12:
            alpha *= 0.5
        elif it % 100 == 0 and alpha < 0.05:
            alpha *= 1.05

        h = h_new

    return best_h, best_c5


def simulated_anneal(h0, n, n_iter, rng, t_start=0.025, t_end=0.0005):
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    curr_h = h0.copy()
    curr_c5 = compute_c5(curr_h)

    for it in range(n_iter):
        t = t_start * (t_end / t_start) ** (float(it) / max(n_iter - 1, 1))

        move_type = rng.randint(5)
        h_new = curr_h.copy()

        if move_type == 0:
            i, j = rng.choice(n, size=2, replace=False)
            if curr_h[i] >= curr_h[j]:
                amt = min(curr_h[i], 0.12) * rng.uniform(0.5, 1.0)
                h_new[i] -= amt
                h_new[j] += amt
            else:
                amt = min(curr_h[j], 0.12) * rng.uniform(0.5, 1.0)
                h_new[j] -= amt
                h_new[i] += amt
        elif move_type == 1:
            i, j = rng.choice(n, size=2, replace=False)
            h_new[i], h_new[j] = curr_h[j], curr_h[i]
        elif move_type == 2:
            i, j = rng.choice(n, size=2, replace=False)
            h_new[i] += rng.uniform(-0.06, 0.06)
            h_new[j] += rng.uniform(-0.06, 0.06)
        elif move_type == 3:
            i = rng.randint(n)
            h_new[i] = 1.0 - h_new[i]
        else:
            # Block transfer
            blk = rng.randint(2, max(3, n // 8))
            i = rng.randint(0, n - blk)
            j = rng.randint(0, n - blk)
            if i != j:
                amt = rng.uniform(0.02, 0.08)
                h_new[i:i + blk] -= amt
                h_new[j:j + blk] += amt

        h_new = project_feasible(h_new, n)
        c5_new = compute_c5(h_new)
        delta = c5_new - curr_c5

        if delta < -1e-12 or (t > 1e-9 and rng.random() < np.exp(-max(delta, 0) / t)):
            curr_h = h_new
            curr_c5 = c5_new
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new

    return best_h, best_c5


def quick_hill_climb(h0, n, n_iter, rng):
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    for _ in range(n_iter):
        i, j = rng.choice(n, size=2, replace=False)
        if abs(best_h[i] - best_h[j]) < 0.03:
            continue
        h_new = best_h.copy()
        if best_h[i] > best_h[j]:
            h_new[j] = min(1.0, best_h[j] + best_h[i] * 0.7)
            h_new[i] = best_h[i] * 0.3
        else:
            h_new[i] = min(1.0, best_h[i] + best_h[j] * 0.7)
            h_new[j] = best_h[j] * 0.3
        h_new = project_feasible(h_new, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-13:
            best_h = h_new
            best_c5 = c5_new
    return best_h, best_c5
