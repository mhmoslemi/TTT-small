import numpy as np
from scipy.optimize import minimize
import time

def exact_objective(h):
    n = len(h)
    if n == 0:
        return float('inf')
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(profile))

def project_box_sum(h):
    h = np.array(h, dtype=np.float64).copy()
    n = len(h)
    target = n / 2.0
    h[~np.isfinite(h)] = 0.0
    for _ in range(200):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            h[mask] = np.minimum(1.0, h[mask] + diff / cnt)
        else:
            mask = h > 1e-14
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            h[mask] = np.maximum(0.0, h[mask] + diff / cnt)
    return np.clip(h, 0.0, 1.0)

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

# Epigraph: minimize t s.t. C_k(h) <= t, sum(h)=n/2, 0<=h<=1
class EpigraphProblem:
    def __init__(self, n):
        self.n = n
        self.dx = 2.0 / n
        self.ncons = 2 * n - 1

    def fun(self, x):
        return x[self.n]

    def grad(self, x):
        g = np.zeros(self.n + 1)
        g[self.n] = 1.0
        return g

    def constr_fun(self, x):
        h = x[:self.n]
        t = x[self.n]
        profile = compute_profile(h)
        c_ineq = t - profile  # t - C_k >= 0
        c_eq = self.n / 2.0 - np.sum(h)  # sum(h) - n/2 = 0, as >= 0 constraint pair
        return np.concatenate([c_ineq, [c_eq, -c_eq]])

    def constr_grad(self, x):
        h = x[:self.n]
        n = self.n
        dx = self.dx
        ncons = self.ncons
        jac = np.zeros((ncons + 2, n + 1))
        t_vals = np.arange(n)
        for s in range(-(n - 1), n):
            row = s + (n - 1)
            valid1 = (t_vals - s >= 0) & (t_vals - s < n)
            valid2 = (t_vals + s >= 0) & (t_vals + s < n)
            grad_row = np.zeros(n)
            grad_row[valid1] += (1.0 - h[t_vals[valid1] - s])
            grad_row[valid2] -= h[t_vals[valid2] + s]
            jac[row, :n] = dx * grad_row
            jac[row, n] = -1.0  # d(t - C_k)/dt = 1, but constraint is t - C_k >= 0, so d/dc = -1 for C_k part... 
            # Actually constraint is: t - C_k >= 0. d/dt = 1, d/dh = -dC_k/dh
            # Let me redo: constraint g(x) = t - C_k(h) >= 0
            # dg/dh = -dC_k/dh, dg/dt = 1
            jac[row, :n] = -dx * grad_row
            jac[row, n] = 1.0
        # Equality: sum(h) - n/2 = 0 expressed as two inequalities
        jac[ncons, :n] = 1.0   # sum(h) - n/2 >= 0 -> but we need equality
        jac[ncons + 1, :n] = -1.0  # -(sum(h) - n/2) >= 0
        return jac

def run_epigraph(h0, n, maxiter=300):
    """Run epigraph optimization starting from h0."""
    h0 = project_box_sum(h0)
    profile = compute_profile(h0)
    t0 = float(np.max(profile)) + 0.01
    x0 = np.concatenate([h0, [t0]])
    
    ep = EpigraphProblem(n)
    bounds = [(0.0, 1.0)] * n + [(-0.5, 2.0)]
    constraints = [
        {'type': 'ineq', 'fun': ep.constr_fun, 'jac': ep.constr_grad},
    ]
    
    try:
        res = minimize(ep.fun, x0, jac=ep.grad, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-10})
        if res.status == 0 or (res.fun < t0):
            h_new = project_box_sum(res.x[:n])
            c_new = exact_objective(h_new)
            if c_new < t0:
                return h_new, c_new
    except Exception:
        pass
    return h0, exact_objective(h0)

def binary_threshold(h):
    n = len(h)
    if n % 2 != 0:
        return None
    m = n // 2
    idx = np.argsort(h)[::-1][:m]
    b = np.zeros(n, dtype=np.float64)
    b[idx] = 1.0
    return b

def binary_swap_sa(b, deadline, seed, max_iters=20000):
    """Simulated annealing with single-swap moves on binary sequence."""
    rng_sa = np.random.RandomState(seed)
    n = len(b)
    b = b.copy()
    curr_c = exact_objective(b)
    best_b = b.copy()
    best_c = curr_c
    
    ones = list(np.where(b == 1.0)[0])
    zeros = list(np.where(b == 0.0)[0])
    
    T_start = 0.5
    T_end = 0.001
    iters = min(max_iters, max(5000, int((deadline - time.time()) * 3000)))
    
    for it in range(iters):
        if time.time() > deadline - 2:
            break
        T = T_start * (T_end / T_start) ** (it / max(1, iters - 1))
        
        i = ones[rng_sa.randint(len(ones))]
        j = zeros[rng_sa.randint(len(zeros))]
        
        b[i] = 0.0
        b[j] = 1.0
        new_c = exact_objective(b)
        
        delta = new_c - curr_c
        if delta < 0 or rng_sa.random() < np.exp(-delta / max(T, 1e-10)):
            curr_c = new_c
            ones[ones.index(i)] = j
            zeros[zeros.index(j)] = i
            if curr_c < best_c - 1e-14:
                best_c = curr_c
                best_b = b.copy()
        else:
            b[i] = 1.0
            b[j] = 0.0
    
    return best_b, best_c

def binary_greedy_swap(b, deadline, max_rounds=200):
    """Greedy single-swap local search."""
    n = len(b)
    b = b.copy()
    curr_c = exact_objective(b)
    ones = list(np.where(b == 1.0)[0])
    zeros = list(np.where(b == 0.0)[0])
    
    for rnd in range(max_rounds):
        if time.time() > deadline - 2:
            break
        best_improve = 1e-14
        best_swap = None
        for i in ones:
            for j in zeros:
                b[i] = 0.0
                b[j] = 1.0
                c = exact_objective(b)
                b[i] = 1.0
                b[j] = 0.0
                improve = curr_c - c
                if improve > best_improve:
                    best_improve = improve
                    best_swap = (i, j)
        if best_swap is not None:
            i, j = best_swap
            b[i] = 0.0
            b[j] = 1.0
            curr_c = exact_objective(b)
            ones = list(np.where(b == 1.0)[0])
            zeros = list(np.where(b == 0.0)[0])
        else:
            break
    return b, curr_c

def binary_double_swap(b, deadline, max_rounds=100):
    """Try swapping two 1s with two 0s simultaneously."""
    n = len(b)
    b = b.copy()
    curr_c = exact_objective(b)
    ones = list(np.where(b == 1.0)[0])
    zeros = list(np.where(b == 0.0)[0])
    
    for rnd in range(max_rounds):
        if time.time() > deadline - 2:
            break
        best_improve = 1e-14
        best_move = None
        for i1 in range(len(ones)):
            for i2 in range(i1 + 1, len(ones)):
                a, c_ = ones[i1], ones[i2]
                for j1 in range(len(zeros)):
                    for j2 in range(j1 + 1, len(zeros)):
                        b_val, d_val = zeros[j1], zeros[j2]
                        b[a] = 0.0; b[c_] = 0.0
                        b[b_val] = 1.0; b[d_val] = 1.0
                        new_c = exact_objective(b)
                        b[a] = 1.0; b[c_] = 1.0
                        b[b_val] = 0.0; b[d_val] = 0.0
                        improve = curr_c - new_c
                        if improve > best_improve:
                            best_improve = improve
                            best_move = (a, c_, b_val, d_val)
        if best_move is not None:
            a, c_, b_val, d_val = best_move
            b[a] = 0.0; b[c_] = 0.0
            b[b_val] = 1.0; b[d_val] = 1.0
            curr_c = exact_objective(b)
            ones = list(np.where(b == 1.0)[0])
            zeros = list(np.where(b == 0.0)[0])
        else:
            break
    return b, curr_c

def transfer_to_size(h_parent, n_target):
    n_src = len(h_parent)
    if n_src == n_target:
        return project_box_sum(h_parent.copy())
    x_src = (np.arange(n_src) + 0.5) / n_src
    x_tgt = (np.arange(n_target) + 0.5) / n_target
    h_new = np.interp(x_tgt, x_src, h_parent)
    return project_box_sum(h_new)

def subgradient_step(h, step_size, deadline):
    """One subgradient step for min max_k C_k."""
    n = len(h)
    profile = compute_profile(h)
    m = float(np.max(profile))
    # Find near-active lags
    active_thresh = m - 0.001
    active = np.where(profile >= active_thresh)[0]
    if len(active) == 0:
        active = np.where(profile >= m - 1e-6)[0]
    
    # Compute subgradient
    dx = 2.0 / n
    grad = np.zeros(n)
    t_vals = np.arange(n)
    for idx in active:
        s = idx - (n - 1)
        valid1 = (t_vals - s >= 0) & (t_vals - s < n)
        valid2 = (t_vals + s >= 0) & (t_vals + s < n)
        g_s = np.zeros(n)
        g_s[valid1] += (1.0 - h[t_vals[valid1] - s])
        g_s[valid2] -= h[t_vals[valid2] + s]
        grad += g_s
    grad *= dx / len(active)
    
    # Take step
    h_new = h - step_size * grad
    h_new = project_box_sum(h_new)
    return h_new

def subgradient_optimize(h0, deadline, max_iters=5000):
    """Subgradient method with decreasing step size."""
    h = h0.copy()
    n = len(h)
    curr_c = exact_objective(h)
    best_h = h.copy()
    best_c = curr_c
    
    for it in range(max_iters):
        if time.time() > deadline - 3:
            break
        step = 0.02 / np.sqrt(it + 1)
        h = subgradient_step(h, step, deadline)
        c = exact_objective(h)
        if c < best_c - 1e-14:
            best_c = c
            best_h = h.copy()
    return best_h, best_c

def lse_obj_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    m = float(np.max(profile))
    shifted = np.minimum((profile - m) / tau, 0.0)
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    weights = exp_vals / sum_exp
    obj = m + tau * np.log(sum_exp)
    grad = np.zeros(n)
    for j in range(2 * n - 1):
        w = weights[j]
        if w < 1e-13:
            continue
        s = j - (n - 1)
        if s >= 0:
            k1 = np.arange(s, n)
            grad[k1] += w * dx * (1.0 - h[k1 - s])
            k2 = np.arange(0, n - s)
            grad[k2] -= w * dx * h[k2 + s]
        else:
            a = -s
            k1 = np.arange(0, n - a)
            grad[k1] += w * dx * (1.0 - h[k1 + a])
            k2 = np.arange(a, n)
            grad[k2] -= w * dx * h[k2 - a]
    return obj, grad

class LSEFunc:
    def __init__(self, tau):
        self.tau = tau
    def value(self, x):
        return lse_obj_grad(x, self.tau)[0]
    def gradient(self, x):
        return lse_obj_grad(x, self.tau)[1]

class SumConstraint:
    def __init__(self, target):
        self.target = target
    def fun(self, x):
        return np.sum(x) - self.target
    def jac(self, x):
        return np.ones(len(x))

def slsqp_lse(h0, tau, maxiter=200):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    obj = LSEFunc(tau)
    con = SumConstraint(n / 2.0)
    constraints = [{'type': 'eq', 'fun': con.fun, 'jac': con.jac}]
    try:
        res = minimize(obj.value, h0.copy(), method='SLSQP', jac=obj.gradient,
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        h_new = project_box_sum(np.array(res.x, dtype=np.float64))
        return h_new
    except Exception:
        return h0.copy()

def refine_lse(h0, deadline):
    h = project_box_sum(h0.copy())
    n = len(h)
    curr_c = exact_objective(h)
    taus = [0.01, 0.004, 0.0015, 0.0006, 0.0002, 0.00008]
    for tau in taus:
        if time.time() > deadline - 5:
            break
        h_new = slsqp_lse(h, tau, maxiter=min(200, n * 3 + 30))
        c_new = exact_objective(h_new)
        if c_new < curr_c - 1e-13:
            h = h_new
            curr_c = c_new
    return h, curr_c

def make_start(n, rng, mode):
    if mode == 0:
        h = np.zeros(n)
        h[:n // 2] = 1.0
        h = h[rng.permutation(n)]
    elif mode == 1:
        x = np.linspace(0, 1, n)
        freq = rng.randint(2, 8)
        phase = rng.uniform(0, 2 * np.pi)
        h = 0.5 + 0.45 * np.sin(2 * np.pi * freq * x + phase)
        h = np.clip(h, 0, 1)
    elif mode == 2:
        h = np.zeros(n)
        h[::2] = 1.0
        if n % 2 == 1:
            h[-1] = 0.0
        if np.sum(h) != n // 2:
            h = np.zeros(n)
            h[:n // 2] = 1.0
    elif mode == 3:
        h = rng.uniform(0, 1, n)
    elif mode == 4:
        h = np.zeros(n)
        block = max(2, n // 8)
        for i in range(0, n, 2 * block):
            h[i:i + block] = 1.0
        h = h[:n]
    elif mode == 5:
        h = np.zeros(n)
        h[:n // 4] = 1.0
        h[3 * n // 4:] = 1.0
    elif mode == 6:
        # Golden ratio based
        h = np.zeros(n)
        golden = 0.6180339887
        positions = np.mod(np.arange(n) * golden * n, n).astype(int)
        h[positions[:n // 2]] = 1.0
    elif mode == 7:
        # Quadratic residue based (for n where applicable)
        h = np.zeros(n)
        m = n // 2
        # Use modular arithmetic pattern
        skip = max(1, m // 3)
        idx = np.mod(np.arange(m) * (skip + 1), n)
        h[idx] = 1.0
        # Adjust count
        while np.sum(h) < m:
            h[rng.randint(n)] = 1.0
        h = np.minimum(h, 1.0)
    else:
        h = np.full(n, 0.5)
    return project_box_sum(h)

def random_perturb(h, scale, rng):
    h_new = h + rng.normal(0, scale, len(h))
    return project_box_sum(h_new)

        # More SA with different seeds
        for s2 in range(3):
            if time.time() > phase3_deadline:
                break
            b_sa2, c_sa2 = binary_swap_sa(best_h, min(phase3_deadline, time.time() + 30), seed=seed + 10000 + s2 * 777, max_iters=5000)
            if c_sa2 < best_c - 1e-12:
                best_c = c_sa2
                best_h = b_sa2.copy()
                best_n = len(best_h)
    
    # Phase 4: Final polish
    if best_h is not None and time.time() < deadline - 10:
        n_final = len(best_h)
        
        # Try epigraph one more time on best
        if time.time() < deadline - 8:
            h_ep2, c_ep2 = run_epigraph(best_h, n_final, maxiter=100)
            if c_ep2 < best_c - 1e-12:
                best_c = c_ep2
                best_h = h_ep2.copy()
        
        # Binary threshold from continuous best
        if n_final % 2 == 0:
            b_final = binary_threshold(best_h)
            if b_final is not None:
                c_bf = exact_objective(b_final)
                if c_bf < best_c - 1e-12:
                    best_c = c_bf
                    best_h = b_final.copy()
                if time.time() < deadline - 5:
                    b_f, c_f = binary_greedy_swap(b_final, deadline - 3, max_rounds=50)
                    if c_f < best_c - 1e-12:
                        best_c = c_f
                        best_h = b_f.copy()
        
        # LSE one more round
        if not np.all((best_h == 0.0) | (best_h == 1.0)):
            if time.time() < deadline - 5:
                h_lse2, c_lse2 = refine_lse(best_h, deadline - 3)
                if c_lse2 < best_c - 1e-12:
                    best_c = c_lse2
                    best_h = h_lse2.copy()
    
    # Final validation
    best_h = project_box_sum(best_h)
    best_c = exact_objective(best_h)
    best_n = len(best_h)
    
    if abs(np.sum(best_h) - best_n / 2.0) > 1e-8:
        best_h = project_box_sum(best_h)
        best_c = exact_objective(best_h)
    
    # Safety: ensure values are in [0,1]
    best_h = np.clip(best_h, 0.0, 1.0)
    
    return best_h, best_c, best_n
