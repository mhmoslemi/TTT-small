These are different! The first counts h=1, h(x+k)=0. The second counts h=0, h(x+k)=1.

But by substituting x → x-k in the second:
∫ (1-h(x-k))·h(x) dx = ∫ h(x)·(1-h(x-k)) dx

Which is the profile at lag -k (i.e., the profile is symmetric: P_k = P_{2(n-1)-k}).

Wait, is the profile symmetric? Let's check:
P_k = dx * sum_i h[i](1-h[i+delta]) where delta = k-(n-1)
P_{2(n-1)-k} = dx * sum_i h[i](1-h[i-delta]) (with appropriate index changes)

By substitution j = i-delta: P_{2(n-1)-k} = dx * sum_j h[j+delta](1-h[j])

And P_k = dx * sum_i h[i](1-h[i+delta]) = dx * sum_j h[j-delta](1-h[j]) (substituting j=i+delta)

These are NOT the same in general. So the profile is not symmetric.

However, if we replace h by 1-h:
New P_k = dx * sum_i (1-h[i])·h[i+delta] = dx * sum_i h[i+delta](1-h[i])
= P_{2(n-1)-k} (original)

So complementing h reflects the profile. The max stays the same. Good.

Also, if we reverse h: h'[i] = h[n-1-i]:
New P_k = dx * sum_i h[n-1-i]·(1-h[n-1-(i+delta)])
Let j = n-1-i: = dx * sum_j h[j]·(1-h[j-delta]) = original P_{2(n-1)-k}

So reversing also reflects the profile. Max stays the same.

These symmetries mean we should try both h and 1-h, and h and reversed h, and take the best.

Actually, since the max is the same under these transformations, it doesn't help for finding a better value. But it does mean that if we find a good h, we can also report 1-h or reversed h.

            contrib = h[delta:n] * (1.0 - h[:n-delta])
            top_idx = np.argsort(contrib)[::-1][:5]
            for idx in top_idx[:3]:
                d = np.zeros(n)
                d[idx + delta] -= 1.0
                d[idx] += 1.0
                d -= d.mean()
                nm = np.linalg.norm(d)
                if nm > 1e-12:
                    dirs.append(d / nm)
        elif delta < 0:
            sep = -delta
            contrib = h[sep:n] * (1.0 - h[:n-sep])
            top_idx = np.argsort(contrib)[::-1][:5]
            for idx in top_idx[:3]:
                d = np.zeros(n)
                d[idx + sep] -= 1.0
                d[idx] += 1.0
                d -= d.mean()
                nm = np.linalg.norm(d)
                if nm > 1e-12:
                    dirs.append(d / nm)
    return dirs[:n_dirs]

def line_search_phase(h, budget_s, t0, n_dirs_per_iter=30):
    n = len(h)
    rng = np.random.RandomState(42)
    best_h = h.copy()
    best_score = exact_score(h)
    h_cur = h.copy()
    no_improve = 0
    it = 0
    while True:
        if time.perf_counter() - t0 > budget_s:
            break
        dirs = generate_directions(h_cur, rng, n_dirs)
        improved = False
        for d in dirs:
            if time.perf_counter() - t0 > budget_s:
                break
            tau, s = line_search(h_cur, d)
            if tau != 0.0 and s < best_score - 1e-14:
                h_new = h_cur + tau * d
                h_new = repair_feasible(h_new)
                s2 = exact_score(h_new)
                if s2 < best_score - 1e-14:
                    best_h = h_new.copy()
                    best_score = s2
                    h_cur = h_new
                    improved = True
                    no_improve = 0
        if not improved:
            no_improve += 1
            if no_improve >= 5:
                break
            pert = 0.005 * rng.randn(n)
            pert -= pert.mean()
            h_cur = repair_feasible(h_cur + pert)
        it += 1
    return best_h, best_score

def solve_epigraph(h_start, t_start, beta=0.0, maxiter=100, n_constraints=0):
    n = len(h_start)
    M = 2 * n - 1
    nc = n_constraints if n_constraints > 0 else M
    if nc < M:
        prof = get_profile(h_start)
        active_lags = np.argsort(prof)[::-1][:nc]
    else:
        active_lags = np.arange(M)

    def obj(x):
        h = x[:n]
        t = x[n]
        return t + beta * np.sum(h * (1.0 - h))

    def obj_jac(x):
        h = x[:n]
        g = np.zeros(n + 1)
        g[:n] = beta * (1.0 - 2.0 * h)
        g[n] = 1.0
        return g

    def cons_fun(x):
        h = x[:n]
        t = x[n]
        eq = h.sum() - n / 2.0
        prof = np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)
        ineq = t - prof[active_lags]
        return np.concatenate([[eq], ineq])

    def cons_jac(x):
        h = x[:n]
        G = get_lag_gradients(h)
        Ja = G[active_lags, :]
        J = np.zeros((1 + len(active_lags), n + 1))
        J[0, :n] = 1.0
        J[1:, :n] = -Ja
        J[1:, n] = 1.0
        return J

    bounds = [(0.0, 1.0)] * n + [(0.0, t_start + 2.0)]
    x0 = np.concatenate([h_start, [t_start]])
    try:
        result = minimize(
            obj, x0, jac=obj_jac,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'ineq', 'fun': cons_fun, 'jac': cons_jac}],
            options={'maxiter': maxiter, 'ftol': 1e-12}
        )
        if result.x is not None and np.all(np.isfinite(result.x)):
            h_sol = repair_feasible(result.x[:n])
            return h_sol
    except Exception:
        pass
    return None

def interpolate_to_dim(vec, new_len):
    n = len(vec)
    x_src = (np.arange(n) + 0.5) * 2.0 / n
    x_tgt = (np.arange(new_len) + 0.5) * 2.0 / new_len
    x_ext = np.concatenate([[0.0], x_src, [2.0]])
    f_ext = np.concatenate([[vec[0]], vec, [vec[-1]]])
    h_new = np.interp(x_tgt, x_ext, f_ext)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.perf_counter()

    def tleft():
        return budget_s - (time.perf_counter() - t0)

    try:
        parent = np.array(initial_h_values, dtype=np.float64).copy()
    except Exception:
        parent = np.full(140, 0.5)
    parent = repair_feasible(parent)

    best_h = parent.copy()
    best_score = exact_score(parent)

    # Phase 1: SLSQP coarse optimization
    if tleft() > 100:
        maxiter1 = min(150, max(40, int(tleft() * 0.15)))
        for beta in [0.0, 1e-6]:
            if tleft() < 60:
                break
            h_sol = solve_epigraph(best_h, best_score + 1e-6, beta=beta, maxiter=maxiter1)
            if h_sol is not None:
                s = exact_score(h_sol)
                if s < best_score - 1e-13:
                    best_h = h_sol
                    best_score = s

    # Phase 2: Heavy line search from current best
    if tleft() > 200:
        tl = min(300, tleft() * 0.35)
        h_ls, s_ls = line_search_phase(best_h, t0 + tl, t0, n_dirs_per_iter=35)
        if s_ls < best_score - 1e-13:
            best_h = h_ls
            best_score = s_ls

    # Phase 3: SLSQP with fewer constraints (faster per iteration)
    if tleft() > 80:
        n = len(best_h)
        for nc in [40, 60, 80]:
            if tleft() < 40:
                break
            if nc >= 2 * n - 1:
                nc = 2 * n - 1
            mi = min(80, max(20, int(tleft() * 0.1)))
            h_sol = solve_epigraph(best_h, best_score + 1e-6, beta=0.0, maxiter=mi, n_constraints=nc)
            if h_sol is not None:
                s = exact_score(h_sol)
                if s < best_score - 1e-13:
                    best_h = h_sol
                    best_score = s

    # Phase 4: More line search after SLSQP refinement
    if tleft() > 150:
        tl = min(200, tleft() * 0.3)
        h_ls, s_ls = line_search_phase(best_h, t0 + tl, t0, n_dirs_per_iter=40)
        if s_ls < best_score - 1e-13:
            best_h = h_ls
            best_score = s_ls

    # Phase 5: Multi-start perturbations with line search
    if tleft() > 100:
        n = len(best_h)
        rng = np.random.RandomState(seed + 1)
        for p in range(8):
            if tleft() < 60:
                break
            eps = 0.004 + 0.003 * p
            z = rng.randn(n)
            z -= z.mean()
            h_pert = repair_feasible(best_h + eps * z)
            tl = min(60, tleft() * 0.2)
            h_ls, s_ls = line_search_phase(h_pert, t0 + tl, t0, n_dirs_per_iter=25)
            if s_ls < best_score - 1e-13:
                best_h = h_ls
                best_score = s_ls

    # Phase 6: Try different dimensions
    if tleft() > 80:
        pn = len(best_h)
        dims = sorted(set([d for d in [pn-6, pn-4, pn+4, pn+8, pn+16, pn+32] if 60 <= d <= 220 and d != pn]))
        for new_n in dims:
            if tleft() < 40:
                break
            h_interp = interpolate_to_dim(best_h, new_n)
            s_interp = exact_score(h_interp)
            if tleft() > 30:
                h_sol = solve_epigraph(h_interp, s_interp + 1e-6, beta=0.0, maxiter=40)
                if h_sol is not None:
                    s = exact_score(h_sol)
                    if s < s_interp:
                        h_interp = h_sol
                        s_interp = s
            if tleft() > 20:
                tl = min(30, tleft() * 0.3)
                h_ls, s_ls = line_search_phase(h_interp, t0 + tl, t0, n_dirs_per_iter=20)
                if s_ls < s_interp:
                    h_interp = h_ls
                    s_interp = s_ls
            if s_interp < best_score - 1e-13:
                best_h = h_interp
                best_score = s_interp

    # Phase 7: Final SLSQP polish
    if tleft() > 50:
        n = len(best_h)
        h_sol = solve_epigraph(best_h, best_score + 1e-6, beta=0.0, maxiter=60)
        if h_sol is not None:
            s = exact_score(h_sol)
            if s < best_score - 1e-13:
                best_h = h_sol
                best_score = s

    # Phase 8: Final line search polish
    if tleft() > 30:
        tl = min(25, tleft() * 0.5)
        h_ls, s_ls = line_search_phase(best_h, t0 + tl, t0, n_dirs_per_iter=30)
        if s_ls < best_score - 1e-13:
            best_h = h_ls
            best_score = s_ls

    best_h = repair_feasible(best_h)
    final_score = exact_score(best_h)
    return (best_h, final_score, len(best_h))
