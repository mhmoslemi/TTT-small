def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr * dx))

def repair_feasible(vec):
    h = np.asarray(vec, dtype=np.float64)
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h.copy()
    if diff > 0:
        order = np.argsort(h)
        rem = diff
        for i in order:
            if rem <= 1e-15:
                break
            add = min(rem, 1.0 - h[i])
            h[i] += add
            rem -= add
    else:
        order = np.argsort(-h)
        rem = -diff
        for i in order:
            if rem <= 1e-15:
                break
            remv = min(rem, h[i])
            h[i] -= remv
            rem -= remv
    # Final precision fix
    diff2 = target - h.sum()
    if abs(diff2) > 1e-14:
        for i in range(len(h)):
            if diff2 > 0 and h[i] < 1.0:
                h[i] = min(1.0, h[i] + diff2)
                break
            elif diff2 < 0 and h[i] > 0.0:
                h[i] = max(0.0, h[i] + diff2)
                break
    return h

def smooth_min_and_grad(values, beta):
    """Compute smooth min of values and gradient weights.
    smooth_min = -(1/beta) * log(sum(exp(-beta * v)))
    Returns (smooth_min_value, weights) where weights are the softmax-like weights."""
    # For numerical stability
    v_min = np.min(values)
    shifted = values - v_min
    exp_vals = np.exp(-beta * shifted)
    Z = np.sum(exp_vals)
    weights = exp_vals / Z
    smooth_val = v_min - np.log(Z) / beta
    return smooth_val, weights

def smooth_objective_and_grad(h, beta_outer, beta_inner):

import numpy as np
from scipy.optimize import minimize
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr * dx))

def repair_feasible(vec):
    h = np.asarray(vec, dtype=np.float64).copy()
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(h)
        rem = diff
        for i in order:
            if rem <= 1e-15:
                break
            add = min(rem, 1.0 - h[i])
            h[i] += add
            rem -= add
    else:
        order = np.argsort(-h)
        rem = -diff
        for i in order:
            if rem <= 1e-15:
                break
            remv = min(rem, h[i])
            h[i] -= remv
            rem -= remv
    diff2 = target - h.sum()
    if abs(diff2) > 1e-13:
        for i in range(n):
            if diff2 > 0 and h[i] < 1.0:
                h[i] += diff2
                break
            elif diff2 < 0 and h[i] > 0.0:
                h[i] += diff2
                break
    return h

def compute_F(h):
    """Compute F_m = dx*(A_m + min(P_m, S_m)) for m=0..n-1."""
    n = len(h)
    dx = 2.0 / n
    total = h.sum()
    # Autocorrelation A_m for m=0..n-1
    auto = np.correlate(h, h, mode='full')
    A = auto[n-1:]  # A[m] = sum_{i=0}^{n-m-1} h[i]*h[i+m]
    # Prefix and suffix sums
    cumsum = np.concatenate([[0.0], np.cumsum(h)])
    P = cumsum[:n]  # P[m] = sum(h[0:m])
    S = np.zeros(n)
    for m in range(n):
        S[m] = total - cumsum[n-m]
    B = np.minimum(P, S)
    F = dx * (A + B)
    return F

def compute_F_grads(h):
    """Compute gradients dF_m/dh_k for all m, k. Returns array of shape (n, n)."""
    n = len(h)
    dx = 2.0 / n
    total = h.sum()
    grads = np.zeros((n, n))
    
    cumsum = np.concatenate([[0.0], np.cumsum(h)])
    
    for m in range(n):
        # Gradient of A_m
        if m == 0:
            g_auto = 2.0 * h
        else:
            g_auto = np.zeros(n)
            g_auto[:n-m] = h[m:]
            g_auto[m:] += h[:n-m]
        
        # Gradient of B_m = min(P_m, S_m)
        P_m = cumsum[m]
        S_m = total - cumsum[n-m]
        if m == 0:
            g_B = np.zeros(n)
        elif P_m < S_m:
            g_B = np.zeros(n)
            g_B[:m] = 1.0
        elif S_m < P_m:
            g_B = np.zeros(n)
            g_B[n-m:] = 1.0
        else:
            g_B = np.zeros(n)
            g_B[:m] = 0.5
            g_B[n-m:] = 0.5
        
        grads[m] = dx * (g_auto + g_B)
    
    return grads

def smooth_min_obj(h, beta):
    """Objective: -softmin(F_0, ..., F_{n-1}) and its gradient."""
    F = compute_F(h)
    # Soft min
    F_min = np.min(F)
    shifted = F - F_min
    exp_vals = np.exp(-beta * shifted)
    Z = np.sum(exp_vals)
    weights = exp_vals / Z
    softmin_val = F_min - np.log(Z) / beta
    obj_val = -softmin_val
    
    F_grads = compute_F_grads(h)
    grad = -np.sum(weights[:, np.newaxis] * F_grads, axis=0)
    
    return obj_val, grad

def solve_smooth(h_start, betas, maxiter=50):
    """Solve smooth optimization with annealed betas."""
    h = repair_feasible(h_start.copy())
    n = len(h)
    constraints = [
        {'type': 'eq', 'fun': lambda x: x.sum() - n/2.0,
         'jac': lambda x: np.ones(n)}
    ]
    bounds = [(0.0, 1.0)] * n
    
    for beta in betas:
        def obj(x):
            v, _ = smooth_min_obj(x, beta)
            return v
        def obj_jac(x):
            _, g = smooth_min_obj(x, beta)
            return g
        try:
            res = minimize(obj, h, jac=obj_jac, method='SLSQP',
                         bounds=bounds, constraints=constraints,
                         options={'maxiter': maxiter, 'ftol': 1e-12})
            if res.x is not None:
                h_new = repair_feasible(res.x)
                if exact_score(h_new) <= exact_score(h) + 0.005:
                    h = h_new
        except Exception:
            pass
    return h

def active_descent(h, max_steps=30):
    """Active descent using exact F values."""
    h = repair_feasible(h.copy())
    best_score = exact_score(h)

import numpy as np
from scipy.optimize import minimize
import time


def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr * dx))


def repair_feasible(vec):
    h = np.asarray(vec, dtype=np.float64).copy()
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(h)
        rem = diff
        for i in order:
            if rem <= 1e-15:
                break
            add = min(rem, 1.0 - h[i])
            h[i] += add
            rem -= add
    else:
        order = np.argsort(-h)
        rem = -diff
        for i in order:
            if rem <= 1e-15:
                break
            remv = min(rem, h[i])
            h[i] -= remv
            rem -= remv
    diff2 = target - h.sum()
    if abs(diff2) > 1e-13:
        for i in range(n):
            if diff2 > 0 and h[i] < 1.0:
                h[i] += diff2
                break
            elif diff2 < 0 and h[i] > 0.0:
                h[i] += diff2
                break
    return h


def compute_F(h):
    n = len(h)
    dx = 2.0 / n
    total = h.sum()
    auto = np.correlate(h, h, mode='full')
    A = auto[n - 1:]
    cumsum = np.concatenate([[0.0], np.cumsum(h)])
    P = np.zeros(n)
    S = np.zeros(n)
    for m in range(n):
        P[m] = cumsum[m]
        S[m] = total - cumsum[n - m]
    B = np.minimum(P, S)
    return dx * (A + B)


def compute_F_grads(h):
    n = len(h)
    dx = 2.0 / n
    total = h.sum()
    grads = np.zeros((n, n))
    cumsum = np.concatenate([[0.0], np.cumsum(h)])
    for m in range(n):
        if m == 0:
            g_auto = 2.0 * h
            P_m = 0.0
            S_m = 0.0
            g_B = np.zeros(n)
        else:
            g_auto = np.zeros(n)
            g_auto[:n - m] = h[m:]
            g_auto[m:] += h[:n - m]
            P_m = cumsum[m]
            S_m = total - cumsum[n - m]
            g_B = np.zeros(n)
            if P_m < S_m:
                g_B[:m] = 1.0
            elif S_m < P_m:
                g_B[n - m:] = 1.0
            else:
                g_B[:m] = 0.5
                g_B[n - m:] = 0.5
        grads[m] = dx * (g_auto + g_B)
    return grads


def smooth_min_obj(h, beta):
    F = compute_F(h)
    F_min = np.min(F)
    shifted = F - F_min
    exp_vals = np.exp(-beta * shifted)
    Z = np.sum(exp_vals)
    weights = exp_vals / Z
    softmin_val = F_min - np.log(Z) / beta
    obj_val = -softmin_val
    F_grads = compute_F_grads(h)
    grad = -np.sum(weights[:, np.newaxis] * F_grads, axis=0)
    return obj_val, grad

    n = len(h)
    current_score = exact_score(h)
    constraints = [
        {'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
         'jac': lambda x: np.ones(n)}
    ]
    bounds = [(0.0, 1.0)] * n
    for beta in betas:
        def obj(x):
            v, _ = smooth_min_obj(x, beta)
            return v

            _, g = smooth_min_obj(x, beta)
            return g

                s_new = exact_score(h_new)

                    current_score = s_new
        except Exception:
            pass
    return h

def epigraph_solve(h_start, maxiter=150, n_active=None):
    h = repair_feasible(h_start.copy())
    n = len(h)
    F_vals = compute_F(h)
    F_min = np.min(F_vals)

    def obj(x):
        return -x[n]

    def obj_jac(x):
        g = np.zeros(n + 1)
        g[n] = -1.0
        return g

    def eq_fun(x):
        return x[:n].sum() - n / 2.0

    def eq_jac(x):
        g = np.ones(n + 1)
        g[n] = 0.0
        return g

    if n_active is not None and n_active < n:
        F_cur = compute_F(x0[:n])
        active_idx = np.argsort(F_cur)[:n_active]
        active_idx = np.sort(active_idx)
    else:
        active_idx = np.arange(n)

    def ineq_fun(x):
        F = compute_F(x[:n])
        return x[n] - F[active_idx]

    def ineq_jac(x):
        grads = compute_F_grads(x[:n])
        g = np.zeros((len(active_idx), n + 1))
        g[:, :n] = -grads[active_idx, :]
        g[:, n] = 1.0
        return g

    bounds = [(0.0, 1.0)] * n + [(-1.0, 2.0)]
    constraints = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}

            score = exact_score(h_sol)
            if np.isfinite(score):

    return None, None

        [50, 150, 500],
    ]
    for sched in schedules:
        if tl() < 100:
            break
        h_sol = solve_smooth(best_h, sched, maxiter=40)
        s = exact_score(h_sol)
        if s < best_score - 1e-12:
            best_h = h_sol
            best_score = s
            best_n = len(h_sol)

    # Phase 2: Active descent
    if tl() > 30:
        h_ref, s_ref = active_descent(best_h, max_steps=50)
        if s_ref < best_score - 1e-12:
            best_h = h_ref
            best_score = s_ref
            best_n = len(h_ref)

    # Phase 3: Epigraph with small active sets
    if tl() > 80:
        for na in [3, 5, 8, 12]:
            if tl() < 40:
                break
            h_sol, score = epigraph_solve(best_h, maxiter=100, n_active=min(na, len(best_h)))
            if h_sol is not None and score < best_score - 1e-12:
                best_h = h_sol
                best_score = score
                best_n = len(h_sol)

    # Phase 4: Coordinate polish
    if tl() > 20:
        h_ref, s_ref = coordinate_polish(best_h, max_rounds=4)
        if s_ref < best_score - 1e-12:
            best_h = h_ref
            best_score = s_ref
            best_n = len(h_ref)

    # Phase 5: Dimension transfer (try coarser and finer)
    if tl() > 100:
        n_cur = len(best_h)
        targets = []
        for d in [-30, -20, -10, 10, 20, 30]:
            nd = n_cur + d
            if 40 <= nd <= 200:
                targets.append(nd)
        for nd in targets:
            if tl() < 50:
                break
            h_interp = repair_feasible(interpolate_dim(best_h, nd))
            s_interp = exact_score(h_interp)
            if s_interp < best_score + 0.008:
                h_sol = solve_smooth(h_interp, [30, 100, 300], maxiter=30)
                s_sol = exact_score(h_sol)
                if s_sol < best_score - 1e-12:
                    best_h = h_sol
                    best_score = s_sol
                    best_n = len(h_sol)
                if tl() > 20:
                    h_ref, s_ref = active_descent(h_interp, max_steps=25)
                    if s_ref < best_score - 1e-12:
                        best_h = h_ref
                        best_score = s_ref
                        best_n = len(h_ref)

    # Phase 6: Multi-start with diverse perturbations
    if tl() > 80:
        n_cur = len(best_h)
        for trial in range(15):
            if tl() < 35:
                break
            eps = rng.uniform(0.003, 0.012)
            if trial % 4 == 0:
                z = rng.randn(n_cur)
            elif trial % 4 == 1:
                freq = rng.randint(2, 8)
                z = np.sin(2 * np.pi * freq * np.arange(n_cur) / n_cur + rng.uniform(0, np.pi))
            elif trial % 4 == 2:
                z = np.zeros(n_cur)
                block = max(2, n_cur // 6)
                z[:block] = 1.0
                z[n_cur - block:] = -1.0
            else:
                # Mirror perturbation
                z = rng.randn(n_cur // 2)
                z_full = np.concatenate([z, -z[::-1]])[:n_cur]
            z -= z.mean()
            z /= (np.std(z) + 1e-10)
            h_pert = repair_feasible(best_h + eps * z)
            h_sol = solve_smooth(h_pert, [50, 200], maxiter=30)
            s_sol = exact_score(h_sol)
            if s_sol < best_score - 1e-12:
                best_h = h_sol
                best_score = s_sol
                best_n = len(h_sol)
            if tl() > 20:
                h_ref, s_ref = active_descent(h_pert, max_steps=15)
                if s_ref < best_score - 1e-12:
                    best_h = h_ref
                    best_score = s_ref
                    best_n = len(h_ref)

    # Phase 7: Final aggressive refinement
    if tl() > 25:
        h_sol = solve_smooth(best_h, [100, 300, 800], maxiter=30)
        s_sol = exact_score(h_sol)
        if s_sol < best_score - 1e-12:
            best_h = h_sol
            best_score = s_sol
            best_n = len(h_sol)

    if tl() > 20:
        h_ref, s_ref = active_descent(best_h, max_steps=40)
        if s_ref < best_score - 1e-12:
            best_h = h_ref
            best_score = s_ref
            best_n = len(h_ref)

    if tl() > 15:
        h_ref, s_ref = coordinate_polish(best_h, max_rounds=3)
        if s_ref < best_score - 1e-12:
            best_h = h_ref
            best_score = s_ref
            best_n = len(h_ref)

    # Final: one more epigraph pass
    if tl() > 20:
        h_sol, score = epigraph_solve(best_h, maxiter=200)
        if h_sol is not None and score < best_score - 1e-12:
            best_h = h_sol
            best_score = score
            best_n = len(h_sol)

    best_h = repair_feasible(best_h)
    best_score = exact_score(best_h)
    return best_h, best_score, len(best_h)
