import numpy as np
import time
from scipy.optimize import minimize, linprog

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))

def get_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def get_lag_gradients(h):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    G = np.zeros((M, n))
    for k in range(M):
        delta = k - (n - 1)
        if delta >= 0:
            if delta < n:
                L = n - delta
                small_idx = np.arange(0, L)
                big_idx = np.arange(delta, n)
                G[k, big_idx] += dx * (1.0 - h[small_idx])
                G[k, small_idx] -= dx * h[big_idx]
        else:
            d = -delta
            if d < n:
                L = n - d
                small_idx = np.arange(0, L)
                big_idx = np.arange(d, n)
                G[k, small_idx] += dx * (1.0 - h[big_idx])
                G[k, big_idx] -= dx * h[small_idx]
    return G

def repair_feasible(h, target=None):
    h = np.array(h, dtype=np.float64).copy()
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    if target is None:
        target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        order = np.argsort(-room)
        remaining = diff
        for idx in order:
            if remaining <= 1e-15:
                break
            add = min(remaining, room[idx])
            h[idx] += add
            remaining -= add
    else:
        order = np.argsort(-h)
        remaining = -diff
        for idx in order:
            if remaining <= 1e-15:
                break
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
    return h

    h_new = np.interp(x_tgt, x_ext, f_ext)

def solve_linearized_chebyshev(h, radius, max_time=5.0):
    """Solve the linearized Chebyshev LP at point h with trust region radius."""
    n = len(h)
    M = 2 * n - 1
    dx = 2.0 / n
    
    p = get_profile(h)  # current profile values
    G = get_lag_gradients(h)  # gradient matrix, shape (M, n)
    
    # LP: minimize theta
    # subject to: G @ h - p + G @ (x - h) <= theta  => G @ x - theta <= p - G @ h
    # Wait: p_i(h) + G_i · (x - h) <= theta
    # => G_i · x - theta <= p_i - G_i · h
    # But p_i - G_i · h is just the constant term... let me redo.
    # p_i(h) + sum_j G[i,j] * (x_j - h_j) <= theta
    # sum_j G[i,j] * x_j - theta <= p_i(h) - sum_j G[i,j] * h_j
    # = p_i(h) - (G @ h)_i
    
    # Note: (G @ h) is not the same as p! G is the gradient, not the function value.
    # So the RHS is p_i(h) - (G_i · h)
    
    rhs_const = p - G @ h  # this is the constant in the linearization
    # Wait no. The linearization is:
    # p_i(x) ≈ p_i(h) + G_i · (x - h) = G_i · x + (p_i(h) - G_i · h)
    # So: G_i · x + (p_i(h) - G_i · h) <= theta
    # => G_i · x - theta <= -(p_i(h) - G_i · h) = G_i · h - p_i(h)
    
    # In standard form for linprog: A_ub @ var <= b_ub
    # var = [x_0, ..., x_{n-1}, theta]
    # A_ub row i: [G_i, -1]
    # b_ub row i: G_i · h - p_i(h)
    
    A_ub = np.hstack([G, -np.ones((M, 1))])
    b_ub = G @ h - p  # = sum_j G[i,j]*h_j - p_i = -(p_i - G_i·h)
    
    # Equality: sum(x) = n/2
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    # Bounds: 0 <= x_i <= 1, theta free (but bounded)
    bounds = []
    for i in range(n):
        lo = max(0.0, h[i] - radius)
        hi = min(1.0, h[i] + radius)
        bounds.append((lo, hi))
    bounds.append((None, None))  # theta unbounded
    
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0  # minimize theta
    
    t0 = time.perf_counter()
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': max_time, 'presolve': True})
    except Exception:
        return None
    
    if time.perf_counter() - t0 > max_time:
        return None
    
    if res.status != 0:
        return None
    
    h_sol = res.x[:n]
    if not np.all(np.isfinite(h_sol)):
        return None
    
    return repair_feasible(h_sol)

def solve_slsqp_epigraph(h_start, t_start, beta=0.0, maxiter=80, n_constraints=0):
    n = len(h_start)
    M = 2 * n - 1
    nc = n_constraints if n_constraints > 0 else M
    
    if nc < M:
        prof = get_profile(h_start)
        active_lags = np.argsort(prof)[::-1][:nc]
    else:
        active_lags = np.arange(M)
    
    def obj(x):
        h = x[:n]
        t = x[n]
        return t + beta * np.sum(h * (1.0 - h))
    
    def obj_jac(x):
        h = x[:n]
        g = np.zeros(n + 1)
        g[:n] = beta * (1.0 - 2.0 * h)
        g[n] = 1.0
        return g
    
    def cons_fun(x):
        h = x[:n]
        t = x[n]
        eq = h.sum() - n / 2.0
        prof = np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)
        ineq = t - prof[active_lags]
        return np.concatenate([[eq], ineq])
    
    def cons_jac(x):
        h = x[:n]
        G = get_lag_gradients(h)
        Ja = G[active_lags, :]
        J = np.zeros((1 + len(active_lags), n + 1))
        J[0, :n] = 1.0
        J[1:, :n] = -Ja
        J[1:, n] = 1.0
        return J
    
    bounds = [(0.0, 1.0)] * n + [(0.0, t_start + 2.0)]
    x0 = np.concatenate([h_start, [t_start]])
    
    try:
        result = minimize(
            obj, x0, jac=obj_jac,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'ineq', 'fun': cons_fun, 'jac': cons_jac}],
            options={'maxiter': maxiter, 'ftol': 1e-12}
        )
        if result.x is not None and np.all(np.isfinite(result.x)):
            return repair_feasible(result.x[:n])
    except Exception:
        pass
    return None

        if wsum < 1e-15:

        h_sol = solve_linearized_chebyshev(h, r, max_time=max_lp_time)

            if s_trial < best_score - 1e-12:

import numpy as np
import time
from scipy.optimize import minimize, linprog

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))

def get_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def get_lag_gradients(h):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    G = np.zeros((M, n))
    for k in range(M):
        delta = k - (n - 1)
        if delta >= 0:
            if delta < n:
                L = n - delta
                small_idx = np.arange(0, L)
                big_idx = np.arange(delta, n)
                G[k, big_idx] += dx * (1.0 - h[small_idx])
                G[k, small_idx] -= dx * h[big_idx]
        else:
            d = -delta
            if d < n:
                L = n - d
                small_idx = np.arange(0, L)
                big_idx = np.arange(d, n)
                G[k, small_idx] += dx * (1.0 - h[big_idx])
                G[k, big_idx] -= dx * h[small_idx]
    return G

def repair_feasible(h, target=None):
    h = np.array(h, dtype=np.float64).copy()
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    if target is None:
        target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        order = np.argsort(-room)
        remaining = diff
        for idx in order:
            if remaining <= 1e-15:
                break
            add = min(remaining, room[idx])
            h[idx] += add
            remaining -= add
    else:
        order = np.argsort(-h)
        remaining = -diff
        for idx in order:
            if remaining <= 1e-15:
                break
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
    return h

def interpolate_to_dim(vec, new_len):
    n = len(vec)
    x_src = (np.arange(n) + 0.5) * 2.0 / n
    x_tgt = (np.arange(new_len) + 0.5) * 2.0 / new_len
    x_ext = np.concatenate([[0.0], x_src, [2.0]])
    f_ext = np.concatenate([[vec[0]], vec, [vec[-1]]])
    h_new = np.interp(x_tgt, x_ext, f_ext)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

def solve_linearized_chebyshev(h, radius, max_time=5.0, top_k=0):
    n = len(h)
    M = 2 * n - 1
    dx = 2.0 / n
    
    p = get_profile(h)
    G = get_lag_gradients(h)
    
    if top_k > 0 and top_k < M:
        active = np.argsort(p)[::-1][:top_k]
        G_use = G[active, :]
        p_use = p[active]
    else:
        G_use = G
        p_use = p
    
    A_ub = np.hstack([G_use, -np.ones((G_use.shape[0], 1))])
    b_ub = G_use @ h - p_use
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    bounds = []
    for i in range(n):
        lo = max(0.0, h[i] - radius)
        hi = min(1.0, h[i] + radius)
        bounds.append((lo, hi))
    bounds.append((None, None))
    
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    t0 = time.perf_counter()
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': max_time, 'presolve': True})
    except Exception:
        return None
    
    if time.perf_counter() - t0 > max_time * 1.5:
        return None
    
    if res.status != 0:
        return None
    
    h_sol = res.x[:n]
    if not np.all(np.isfinite(h_sol)):
        return None
    
    return repair_feasible(h_sol)

def chebyshev_step(h, radii, max_lp_time=4.0, top_k=0):
    n = len(h)
    best_h = h.copy()
    best_score = exact_score(h)
    
    for r in radii:
        h_sol = solve_linearized_chebyshev(h, r, max_time=max_lp_time, top_k=top_k)
        if h_sol is None:
            continue
        
        for alpha in [1.0, 0.8, 0.6, 0.4, 0.3, 0.2, 0.1, 0.05]:
            h_trial = repair_feasible((1.0 - alpha) * h + alpha * h_sol)
            s_trial = exact_score(h_trial)
            if s_trial < best_score - 1e-12:
                best_h = h_trial
                best_score = s_trial
                break
    
    return best_h, best_score
