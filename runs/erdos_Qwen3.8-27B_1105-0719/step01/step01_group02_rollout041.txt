    grads = []
    for t in A:
        d = (n-1) - t

        if d >= 0:

        best_i, best_j = -1, -1

        zeros = np.where(cur == 0.0)[0]

                    best_i, best_j = i, j

            cur[best_i] = 0.0
            cur[best_j] = 1.0
            cur_c -= best_delta
        else:
            break
    return cur, cur_c

def transfer_to_size(parent, target_n):

    new_h = np.interp(x_new, x_old, parent)
