def project_box_sum(h, target=None):

        grad = np.zeros(n)
        for i in range(2 * n - 1):
            k = i - (n - 1)
            w = weights[i]
            if w < 1e-14:
                continue
            if k == 0:
                grad += w * (1.0 - 2.0 * x)
            elif k > 0:
                grad[k:n] += w * (1.0 - x[0:n-k])
                grad[0:n-k] -= w * x[k:n]
            else:
                a = -k
                grad[0:n-a] += w * (1.0 - x[a:n])
                grad[a:n] -= w * x[0:n-a]
        grad *= dx
        return value, grad
    
    tau_stages = [0.02, 0.008, 0.003, 0.001, 0.0003, 0.0001]
    h_cur = h0.copy()
    
    for tau in tau_stages:
        if time.time() - start_time > time_limit - 5:
            break
        
        def _fun(x):
            v, g = obj_grad(x, tau)
            sv = np.sum(x) - n / 2.0
            return v + 1e4 * sv * sv, g + 2e4 * sv * np.ones(n)
        
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0,
                 'jac': lambda x: np.ones(n)}]
        try:
            res = minimize(_fun, h_cur, method='SLSQP', jac=True, bounds=bounds,
                          constraints=cons, options={'maxiter': 80, 'ftol': 1e-12})
            h_new = project_box_sum(res.x)
            c_new = exact_objective(h_new)
            c_cur = exact_objective(h_cur)
            if c_new < c_cur - 1e-12:
                h_cur = h_new
        except Exception:
            pass
    
    return h_cur, exact_objective(h_cur)

def run(seed=42, budget_s=1100, **kwargs):
    deadline = time.time() + budget_s
    rng = np.random.RandomState(seed)
    
    best_h = None
    best_c = float('inf')
    best_n = 0
    
    try:
        parent = np.array(initial_h_values, dtype=np.float64)
        parent = project_box_sum(parent)
    except Exception:
        parent = None
    
    if parent is not None:
        pc = exact_objective(parent)
        best_h = parent.copy()
        best_c = pc
        best_n = len(parent)
    else:
        n_init = 140
        best_h = np.ones(n_init) * 0.5
        best_c = exact_objective(best_h)
        best_n = n_init
    
    # Determine sizes to try
    sizes = []
    parent_n = best_n if parent is not None else 140
    for s in [parent_n, parent_n-2, parent_n+2, parent_n-4, parent_n+4,
              parent_n-6, parent_n+6, parent_n-8, parent_n+8, 128, 150, 132]:
        if s not in sizes and 20 <= s <= 300 and s % 2 == 0:
            sizes.append(s)
    
    # Phase 1: Binary local search on parent size (most important)
    if parent is not None:
        b = binary_threshold(parent)
        if b is not None:
            t_alloc = min(180.0, (deadline - time.time()) * 0.25)
            h_sol, c_sol = binary_local_search_fast(b, time_limit=t_alloc, rng_seed=seed)
            if c_sol < best_c:
                best_c = c_sol
                best_h = h_sol.copy()
                best_n = len(h_sol)
    
    # Phase 2: Binary local search on multiple sizes and restarts
    remaining = deadline - time.time()
    n_sizes = min(len(sizes), 5)
    time_per_size = max(60.0, (remaining - 60) / n_sizes)
    
    for n in sizes[:n_sizes]:
        if time.time() > deadline - 90:
            break
        
        t_alloc = min(time_per_size, deadline - time.time() - 60)
        if t_alloc < 30:
            break
        
        # Get starting binary vectors
        starts = []
        if parent is not None:
            tr = transfer_to_size(parent, n)
            b_tr = binary_threshold(tr)
            if b_tr is not None:
                starts.append(b_tr)
        
        # Add a periodic starting point
        alt = np.zeros(n)
        for i in range(n):
            if (i * 7 + n//3) % n < n//2:
                alt[i] = 1.0
        if np.sum(alt) != n//2:
            alt = np.zeros(n)
            alt[::2] = 1.0
        if np.sum(alt) == n//2:
            starts.append(alt)
        
        # Add from best_h if different size
        if len(best_h) != n:
            tr2 = transfer_to_size(best_h, n)
            b2 = binary_threshold(tr2)
            if b2 is not None and (not starts or np.sum(b2 != starts[0]) > 5):
                starts.append(b2)
        
        # Random binary
        rb = np.zeros(n)
        idx = rng.choice(n, n//2, replace=False)
        rb[idx] = 1.0
        starts.append(rb)
        
        for si, st in enumerate(starts[:4]):
            if time.time() > deadline - 90:
                break
            t_alloc2 = max(30.0, t_alloc / len(starts[:4]) - 5)
            if time.time() + t_alloc2 > deadline - 90:
                t_alloc2 = deadline - time.time() - 90
            if t_alloc2 < 15:
                break
            
            h_sol, c_sol = binary_local_search_fast(st, time_limit=t_alloc2, rng_seed=seed + si * 1000 + n)
            if c_sol < best_c:
                best_c = c_sol
                best_h = h_sol.copy()
                best_n = n
    
    # Phase 3: Continuous optimization on best solution
    if time.time() < deadline - 60:
        t_cont = min(120.0, deadline - time.time() - 40)
        if t_cont > 30:
            h_cont, c_cont = continuous_optimize(best_h, time_limit=t_cont)
            if c_cont < best_c - 1e-12:
                best_c = c_cont
                best_h = h_cont.copy()
                best_n = len(h_cont)
            # Also try binary after continuous
            b_cont = binary_threshold(h_cont)
            if b_cont is not None and time.time() < deadline - 40:
                t_bin2 = min(120.0, deadline - time.time() - 20)
                if t_bin2 > 20:
                    h_sol2, c_sol2 = binary_local_search_fast(b_cont, time_limit=t_bin2, rng_seed=seed + 9999)
                    if c_sol2 < best_c:
                        best_c = c_sol2
                        best_h = h_sol2.copy()
                        best_n = len(h_sol2)
    
    # Phase 4: Final polish - quick binary search on best
    if time.time() < deadline - 20:
        b_final = binary_threshold(best_h)
        if b_final is not None:
            t_fin = min(60.0, deadline - time.time() - 10)
            if t_fin > 15:
                h_fin, c_fin = binary_local_search_fast(b_final, time_limit=t_fin, rng_seed=seed + 7777)
                if c_fin < best_c:
                    best_c = c_fin
                    best_h = h_fin.copy()
                    best_n = len(h_fin)
    
    best_h = project_box_sum(best_h, best_n / 2.0)
    best_c = exact_objective(best_h)
    return (best_h, best_c, len(best_h))
