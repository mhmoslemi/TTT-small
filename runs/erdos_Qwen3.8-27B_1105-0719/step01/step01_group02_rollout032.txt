```python
def binary_local_search(h, max_rounds=100):
    n = len(h)
    h = h.copy()
    dx = 2.0 / n
    profile = np.correlate(h, 1-h, mode='full')
    cur_max = np.max(profile)
    
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    
    for round in range(max_rounds):
        best_delta = 0
        best_i, best_j = -1, -1
        # Try all swaps
        for i in ones:
            for j in zeros:
                h[i] = 0; h[j] = 1
                p = np.correlate(h, 1-h, mode='full')
                new_max = np.max(p)
                h[i] = 1; h[j] = 0
                delta = cur_max - new_max
                if delta > best_delta:
                    best_delta = delta
                    best_i, best_j = i, j
        if best_i >= 0:
            h[best_i] = 0; h[best_j] = 1
            profile = np.correlate(h, 1-h, mode='full')
            cur_max = np.max(profile)
        else:
            break
    return h, cur_max * dx
```
