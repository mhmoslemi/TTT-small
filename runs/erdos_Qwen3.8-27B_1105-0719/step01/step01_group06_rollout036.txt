def successive_linear_programming(h0, n_iters=100, time_limit=60):
    h = h0.copy()
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    best_h = h.copy()
    best_score = exact_score(h)
    t0 = time.perf_counter()
    
    for it in range(n_iters):
        if time.perf_counter() - t0 > time_limit:
            break
        
        profile = get_profile(h)
        M = len(profile)
        
        # Select active set (top lags)
        threshold = np.max(profile) - 1e-8
        active = np.where(profile >= threshold)[0]
        if len(active) < 3:
            active = np.argsort(profile)[::-1][:max(3, M//5)]
        active = active[:min(len(active), max(10, M//3))]
        
        # Compute gradient matrix for active lags
        G = compute_gradient_matrix(h)
        G_act = G[active]  # shape (|A|, n)
        prof_act = profile[active]
        
        # Solve LP:
        # minimize t
        # subject to: prof_act[k] + G_act[k] · (h' - h) ≤ t for all k in A
        #            sum(h') = n/2
        #            0 ≤ h' ≤ 1
        #
        # Variables: h' (n vars) + t (1 var)
        # Constraints:
        #   G_act @ h' - t ≤ -prof_act + G_act @ h  (|A| constraints)
        #   sum(h') = n/2  (1 constraint)
        #   0 ≤ h' ≤ 1  (2n bound constraints)
        
        import cvxpy
        h_var = cvxpy.Variable(n)
        t_var = cvxpy.Variable()
        
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cvxpy.sum(h_var) == target
        ]
        for i, k in enumerate(active):
            linearized = prof_act[i] + G_act[i] @ (h_var - h)
            constraints.append(linearized <= t_var)
        
        objective = cvxpy.Minimize(t_var)
        problem = cvxpy.Problem(objective, constraints)
        
        try:
            problem.solve(solver=cvxpy.ECOS, max_iters=500, abs_acc=1e-10)
            if h_var.value is not None:
                h_new = np.array(h_var.value).flatten()
                h_new = np.clip(h_new, 0, 1)
                h_new = repair_feasible(h_new)
                s = exact_score(h_new)
                if s < best_score - 1e-12:
                    best_score = s
                    best_h = h_new.copy()
                h = h_new
        except Exception:
            pass
        
        # Check convergence
        if abs(exact_score(h) - best_score) < 1e-10:
            break
    
    return best_h, best_score

def exact_score(h):

    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))

def get_profile(h):

def repair_feasible(h, target=None):
    h = np.array(h, dtype=np.float64).copy()
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    if target is None:
        target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        order = np.argsort(-room)
        remaining = diff
        for idx in order:
            if remaining <= 1e-15:
                break
            add = min(remaining, room[idx])
            h[idx] += add
            remaining -= add
    else:
        order = np.argsort(-h)
        remaining = -diff
        for idx in order:
            if remaining <= 1e-15:
                break
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
    return h

def compute_gradient_matrix(h):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    G = np.zeros((M, n))
    for k in range(M):
        delta = k - (n - 1)
        if delta >= 0:
            if delta < n:
                G[k, :n-delta] += dx * (1.0 - h[delta:n])
                G[k, delta:] -= dx * h[:n-delta]
        else:
            d = -delta
            if d < n:
                G[k, d:] += dx * (1.0 - h[:n-d])
                G[k, :n-d] -= dx * h[d:]
    return G

    dx = 2.0 / n
    profile = get_profile(h)
    
    # Active lags (top n_active)
    top_idx = np.argsort(profile)[::-1][:n_active]
    
    # Also include lags near the max
    thresh = profile[top_idx[0]] - 0.01
    near = np.where(profile >= thresh)[0]
    active = np.unique(np.concatenate([top_idx, near]))
    active = active[:n_active]
    
    # d_m = G[m,j] - G[m,k]
    G = compute_gradient_matrix(h)
    d = G[active, j] - G[active, k]
    p = profile[active]
    
    # Boundaries for δ
    delta_lo = max(-h[j], h[k] - 1)
    delta_hi = min(1 - h[j], h[k])

        return h.copy(), 0.0

    candidates = [0.0, delta_lo, delta_hi]

def exact_score(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))

def get_profile(h):
    dx = 2.0 / len(h)
    return np.correlate(h, 1.0 - h, mode="full") * dx

def repair_feasible(h, target=None):
    h = np.array(h, dtype=np.float64).copy()
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    if target is None:
        target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        order = np.argsort(-room)
        remaining = diff
        for idx in order:
            if remaining <= 1e-15:
                break
            add = min(remaining, room[idx])
            h[idx] += add
            remaining -= add
    else:
        order = np.argsort(-h)
        remaining = -diff
        for idx in order:
            if remaining <= 1e-15:
                break
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
    return h

def compute_gradient_matrix(h):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    G = np.zeros((M, n))
    for k in range(M):
        delta = k - (n - 1)
        if delta >= 0:
            if delta < n:
                G[k, :n-delta] += dx * (1.0 - h[delta:n])
                if delta > 0:
                    G[k, delta:] -= dx * h[:n-delta]
        else:
            d = -delta
            if d < n:
                G[k, d:] += dx * (1.0 - h[:n-d])
                if d > 0:
                    G[k, :n-d] -= dx * h[d:]
    return G

    # Stage 1: Paired CD on best start (longest time allocation)
    if tleft() > 60:
        tl = min(120, tleft() * 0.3)
        h_cd, s_cd = paired_cd(best_h, n_pairs_per_sweep=400, n_sweeps=100, time_limit=tl, seed=seed)
        if s_cd < best_score - 1e-13:
            best_h = h_cd
            best_score = s_cd
    
    # Stage 2: SLP on best
    if tleft() > 50:
        tl = min(80, tleft() * 0.25)
        h_slp, s_slp = solve_slp(best_h, n_active=35, max_iters=30, time_limit=tl)
        if s_slp < best_score - 1e-13:
            best_h = h_slp
            best_score = s_slp
    
    # Stage 3: Multi-active local descent
    if tleft() > 30:
        tl = min(40, tleft() * 0.2)
        h_ld, s_ld = local_multi_active_descent(best_h, max_iters=100, time_limit=tl)
        if s_ld < best_score - 1e-13:
            best_h = h_ld
            best_score = s_ld
    
    # Stage 4: Try other starts
    for si in range(1, len(starts)):
        if tleft() < 40:
            break
        s_start = exact_score(starts[si])
        if s_start > best_score + 0.02:
            continue
        
        tl_cd = min(40, tleft() * 0.15)
        h_cd, s_cd = paired_cd(starts[si], n_pairs_per_sweep=200, n_sweeps=40, time_limit=tl_cd, seed=seed + si)
        if s_cd < best_score - 1e-13:
            best_h = h_cd
            best_score = s_cd
        
        if tleft() > 30:
            tl_slp = min(30, tleft() * 0.1)
            h_slp, s_slp = solve_slp(best_h, n_active=30, max_iters=20, time_limit=tl_slp)
            if s_slp < best_score - 1e-13:
                best_h = h_slp
                best_score = s_slp
    
    # Stage 5: SLP with more active constraints on best
    if tleft() > 40:
        for na in [50, 80]:
            if tleft() < 25:
                break
            tl = min(40, tleft() * 0.2)
            h_slp, s_slp = solve_slp(best_h, n_active=na, max_iters=25, time_limit=tl)
            if s_slp < best_score - 1e-13:
                best_h = h_slp
                best_score = s_slp
    
    # Stage 6: Dimension exploration
    if tleft() > 60:
        n_best = len(best_h)
        dims = set()
        for d in [n_best - 20, n_best - 10, n_best - 4, n_best + 4, n_best + 10, n_best + 20, n_best + 40]:
            if 40 <= d <= 250:
                dims.add(d)
        if n_best < 200:
            dims.add(n_best + 44)
        if n_best > 80:
            dims.add(n_best - 30)
        dims = sorted(dims)
        
        for new_n in dims:
            if tleft() < 35:
                break
            h_interp = interpolate_to_dim(best_h, new_n)
            s_interp = exact_score(h_interp)
            
            if tleft() > 30:
                tl = min(25, tleft() * 0.2)
                h_cd, s_cd = paired_cd(h_interp, n_pairs_per_sweep=200, n_sweeps=30, time_limit=tl, seed=seed + new_n)
                if s_cd < best_score - 1e-13:
                    best_h = h_cd
                    best_score = s_cd
            
            if tleft() > 20:
                tl = min(15, tleft() * 0.15)
                h_slp, s_slp = solve_slp(best_h, n_active=25, max_iters=15, time_limit=tl)
                if s_slp < best_score - 1e-13:
                    best_h = h_slp
                    best_score = s_slp
    
    # Stage 7: Soft-max gradient (global smoothing)
    if tleft() > 25:
        tl = min(20, tleft() * 0.2)
        h_sm, s_sm = soft_max_grad_descent(best_h, temperature=0.003, n_steps=2000, time_limit=tl)
        if s_sm < best_score - 1e-13:
            best_h = h_sm
            best_score = s_sm
    
    # Stage 8: Final paired CD refinement
    if tleft() > 20:
        tl = min(25, tleft() * 0.3)
        h_cd, s_cd = paired_cd(best_h, n_pairs_per_sweep=500, n_sweeps=50, time_limit=tl, seed=seed + 999)
        if s_cd < best_score - 1e-13:
            best_h = h_cd
            best_score = s_cd
    
    # Stage 9: Final local descent
    if tleft() > 10:
        tl = min(10, tleft() * 0.4)
        h_ld, s_ld = local_multi_active_descent(best_h, max_iters=50, time_limit=tl)
        if s_ld < best_score - 1e-13:
            best_h = h_ld
            best_score = s_ld
    
    # Final repair and validation
    best_h = repair_feasible(best_h)
    final_score = exact_score(best_h)
    
    # If final repair made things worse, try symmetrization
    h_final_sym = symmetrize(best_h)
    s_final_sym = exact_score(h_final_sym)
    if s_final_sym < final_score - 1e-14:
        best_h = h_final_sym
        final_score = s_final_sym
    
    return (best_h, final_score, len(best_h))
