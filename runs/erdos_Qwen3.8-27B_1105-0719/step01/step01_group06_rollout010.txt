def repair_feasible(h, target=None):
    h = np.array(h, dtype=np.float64).copy()
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    if target is None:
        target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        order = np.argsort(-room)
        remaining = diff
        for idx in order:
            if remaining <= 1e-15:
                break
            add = min(remaining, room[idx])
            h[idx] += add
            remaining -= add
    else:
        order = np.argsort(-h)
        remaining = -diff
        for idx in order:
            if remaining <= 1e-15:
                break
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
    # Final check
    diff = target - h.sum()
    if abs(diff) > 1e-10:

        idx = np.argmax(1.0 - h) if diff > 0 else np.argmax(h)
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)

import numpy as np
import time

def exact_score(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))

def get_profile(h):
    dx = 2.0 / len(h)
    return np.correlate(h, 1.0 - h, mode="full") * dx

def repair_feasible(h, target=None):
    h = np.array(h, dtype=np.float64).copy()
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    if target is None:
        target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        order = np.argsort(-room)
        remaining = diff
        for idx in order:
            if remaining <= 1e-15:
                break
            add = min(remaining, room[idx])
            h[idx] += add
            remaining -= add
    else:
        order = np.argsort(-h)
        remaining = -diff
        for idx in order:
            if remaining <= 1e-15:
                break
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
    return h

def smooth_grad(h, t=50.0):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    prof = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(prof)
    shifted = prof - m
    denom = np.exp(t * shifted).sum()
    if denom < 1e-300:
        return np.zeros(n)
    w = t * np.exp(t * shifted) / denom
    G = np.zeros(n)
    for k in range(M):
        wk = w[k]
        if wk < 1e-14:
            continue
        d = k - (n - 1)
        if d >= 0:
            m2 = n - d
            G[:m2] += wk * dx * (1.0 - h[d:])
            G[d:] -= wk * dx * h[:m2]
        else:
            d2 = -d
            m2 = n - d2
            G[d2:] += wk * dx * (1.0 - h[:m2])
            G[:m2] -= wk * dx * h[d2:]
    return G

def project_constraint(h, target=None):
    n = len(h)
    if target is None:
        target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        room = 1.0 - h
        order = np.argsort(-room)
        rem = diff
        for idx in order:
            if rem <= 1e-15:
                break
            a = min(rem, room[idx])
            h[idx] += a
            rem -= a
    else:
        order = np.argsort(-h)
        rem = -diff
        for idx in order:
            if rem <= 1e-15:
                break
            a = min(rem, h[idx])
            h[idx] -= a
            rem -= a
    return h

def lse_descent(h0, t_start=30.0, t_end=100.0, max_iters=500, time_limit=25):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_score = exact_score(h)
    t0 = time.perf_counter()
    momentum = np.zeros(n)
    beta_m = 0.9
    step = 0.01
    
    for it in range(max_iters):
        if time.perf_counter() - t0 > time_limit:
            break
        
        frac = it / max(1, max_iters)
        t_cur = t_start + (t_end - t_start) * frac
        G = smooth_grad(h, t_cur)
        G -= G.mean()
        momentum = beta_m * momentum + (1 - beta_m) * G
        direction = -momentum
        
        norm = np.linalg.norm(direction)
        if norm < 1e-15:
            break
        direction /= norm
        
        improved = False
        for alpha in [step, step * 0.5, step * 0.25, step * 0.1, step * 0.05]:
            h_trial = h + alpha * direction
            h_trial = project_constraint(h_trial)
            s = exact_score(h_trial)
            if s < best_score - 1e-13:
                h = h_trial
                best_h = h.copy()
                best_score = s
                step = min(0.05, step * 1.1)
                improved = True
                break
        if not improved:
            step *= 0.6
            if step < 0.0005:
                break
    return best_h, best_score

def pair_local_search(h0, time_limit=15, n_pairs=80):

                best_s = s
                best_a = a

def symmetrize(h):
    hs = 0.5 * (h + h[::-1].copy())
    return repair_feasible(hs)

def interpolate_to_dim(vec, new_len):
    n = len(vec)
    x_src = (np.arange(n) + 0.5) * 2.0 / n
    x_tgt = (np.arange(new_len) + 0.5) * 2.0 / new_len
    x_ext = np.concatenate([[0.0], x_src, [2.0]])
    f_ext = np.concatenate([[vec[0]], vec, [vec[-1]]])
    h_new = np.interp(x_tgt, x_ext, f_ext)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)
