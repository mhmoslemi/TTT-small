```python
def repair_feasible(h, target=None):
    h = np.array(h, dtype=np.float64).copy()
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    if target is None:
        target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        order = np.argsort(-room)
        remaining = diff
        for idx in order:
            if remaining <= 1e-15:
                break
            add = min(remaining, room[idx])
            h[idx] += add
            remaining -= add
    else:
        order = np.argsort(-h)
        remaining = -diff
        for idx in order:
            if remaining <= 1e-15:
                break
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
    return h
```

```python
    # d < 0: c ranges from 1 to n-1, corresponding to k = n-1-c
    d_neg = np.arange(1, n)  # c values
    k_neg = n - 1 - d_neg  # = n-2, n-3, ..., 0

    term1 = np.zeros(n-1)
    idx1 = i + d_neg[mask1]
    term1[mask1] = 1.0 - h[idx1]
    
    # Term 2: -h[i-c] if c <= i and i-c <= n-c-1
    # i-c <= n-c-1 means i-1 <= n-1, always true
    # So condition is c <= i
    mask2 = d_neg <= i
    term2 = np.zeros(n-1)
    idx2 = i - d_neg[mask2]
    term2[mask2] = h[idx2]
    
    S[k_neg] = dx * (term1 - term2)
    
    # d = 0: k = n-1
    S[n-1] = dx * (1.0 - 2.0 * v)
    
    return S
```

```python
    t_vals = np.linspace(t_min, t_max, n_grid)

    P_lin = P[:, None].T  # Hmm, let me think about this differently
    
    # Actually: for each t, the linear values are P + (t-vi)*D (all M components)
    # Then replace the d0 component with the quadratic
    linear_vals = P[None, :] + (t_vals[:, None] - vi) * D[None, :]  # shape (n_grid, M)

    quad_val = P0 + 2*dx*(t_vals*(c-t_vals) - vi*(c-vi))  # shape (n_grid,)

    t_fine = np.linspace(max(t_min, t_opt - 3*margin), min(t_max, t_opt + 3*margin), 200)

    quad_val_fine = P0 + 2*dx*(t_fine*(c-t_fine) - vi*(c-vi))
    linear_vals_fine[:, d0_idx] = quad_val_fine
    f_fine = linear_vals_fine.max(axis=1)
    t_opt = t_fine[f_fine.argmin()]
    
    h_new = h.copy()
```

```python
def compute_lag_coeffs_vec(h, i):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    S = np.zeros(M)
    
    # d > 0: d = 1, 2, ..., n-1, k = n, n+1, ..., 2n-2
    d_pos = np.arange(1, n)
    k_pos = n - 1 + d_pos  # = n, n+1, ..., 2n-2
    
    # Term 1: (1-h[i-d]) if d <= i
    mask1 = d_pos <= i
    term1 = np.zeros(n - 1)
    if mask1.any():
        term1[mask1] = 1.0 - h[i - d_pos[mask1]]
    
    # Term 2: h[i+d] if d <= n-1-i (i.e., i+d <= n-1)
    mask2 = d_pos <= (n - 1 - i)
    term2 = np.zeros(n - 1)
    if mask2.any():
        term2[mask2] = h[i + d_pos[mask2]]
    
    S[k_pos] = dx * (term1 - term2)
    
    # d < 0: c = 1, 2, ..., n-1, k = n-2, n-3, ..., 0
    d_neg = np.arange(1, n)
    k_neg = n - 1 - d_neg  # = n-2, n-3, ..., 0
    
    # Term 1: (1-h[i+c]) if c <= n-1-i (i.e., i+c <= n-1)
    mask1 = d_neg <= (n - 1 - i)
    term1 = np.zeros(n - 1)
    if mask1.any():
        term1[mask1] = 1.0 - h[i + d_neg[mask1]]
    
    # Term 2: h[i-c] if c <= i (i.e., i-c >= 0)
    mask2 = d_neg <= i
    term2 = np.zeros(n - 1)
    if mask2.any():
        term2[mask2] = h[i - d_neg[mask2]]
    
    S[k_neg] = dx * (term1 - term2)
    
    # d = 0: k = n-1
    S[n - 1] = dx * (1.0 - 2.0 * h[i])
    
    return S
```
