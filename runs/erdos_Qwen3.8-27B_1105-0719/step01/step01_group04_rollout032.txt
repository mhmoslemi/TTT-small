import time

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")

def repair_feasible(vec):

    h = np.clip(h, 0.0, 1.0)

    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(h)
        rem = diff
        for i in order:
            if rem <= 1e-15:
                break
            add = min(rem, 1.0 - h[i])
            h[i] += add
            rem -= add
    else:

        rem = -diff
        for i in order:
            if rem <= 1e-15:
                break
            sub = min(rem, h[i])
            h[i] -= sub
            rem -= sub

                h[i] += diff2
                break

    """Return all correlation values C_l = dx * np.correlate(h, 1-h, 'full')[l]"""

    weights has length 2n-1."""
    n = len(h)
    dx = 2.0 / n

                grad[m:] += w * dx * (1.0 - h[:n-m])

    """Given unconstrained z, find c such that sum(sigma(z_i + c)) = n/2.
    Returns (h, c, sigma_primes)"""
    n = len(z)
    target = n / 2.0
    
    def phi(c):
        return np.sum(1.0 / (1.0 + np.exp(-(z + c))))
    
    def phi_prime(c):
        s = 1.0 / (1.0 + np.exp(-(z + c)))
        return np.sum(s * (1.0 - s))

    Returns (value, grad_z)"""

    # Log-sum-exp for numerical stability
    max_C = np.max(C)
    log_sum = max_C + np.log(np.sum(np.exp(beta * (C - max_C))))
    value = log_sum / beta

    w /= np.sum(w)

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr * dx))

def repair_feasible(vec):
    h = np.asarray(vec, dtype=np.float64).copy()
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(h)
        rem = diff
        for i in order:
            if rem <= 1e-15:
                break
            add = min(rem, 1.0 - h[i])
            h[i] += add
            rem -= add
    else:
        order = np.argsort(-h)
        rem = -diff
        for i in order:
            if rem <= 1e-15:
                break
            sub = min(rem, h[i])
            h[i] -= sub
            rem -= sub
    diff2 = target - h.sum()
    if abs(diff2) > 1e-14:
        for i in range(n):
            if diff2 > 0 and h[i] < 1.0:
                h[i] += diff2
                break
            elif diff2 < 0 and h[i] > 0.0:
                h[i] += diff2
                break
    return h

def corr_values(h):
    n = len(h)

def grad_weighted(h, weights):
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if abs(w) < 1e-15:
            continue
        delta = k - (n - 1)
        if delta == 0:
            grad += w * dx * (1.0 - 2.0 * h)
        elif delta > 0:
            grad[:n - delta] += w * dx * (1.0 - h[delta:])
            grad[delta:] += -w * dx * h[:n - delta]
        else:
            m = -delta
            grad[m:] += w * dx * (1.0 - h[:n - m])
            grad[:n - m] += -w * dx * h[m:]
    return grad

def logit_project(z):
    n = len(z)

    C = corr_values(h)
