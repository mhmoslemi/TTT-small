from scipy.optimize import minimize
from scipy.special import logsumexp

def build_h_from_q(q):
    m = len(q)
    h = np.empty(2 * m)
    h[:m] = q
    h[m:] = 1.0 - q[::-1]
    return h


def extract_q_from_h(h):
    n = len(h)
    m = n // 2
    q = 0.5 * (h[:m] + 1.0 - h[m:][::-1])
    return np.clip(q, 0.0, 1.0)


def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def exact_score(h):
    n = len(h)
    return compute_corrs(h).max() * (2.0 / n)


def grad_corr_full(h, r):
    n = len(h)
    d = r - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -d)
    i_hi = min(n - 1, n - 1 - d)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    jd = idx + d
    vd = (jd >= 0) & (jd <= n - 1)
    grad[idx[vd]] += (1.0 - h[jd[vd]])
    jm = idx - d
    vm = (jm >= 0) & (jm <= n - 1)
    grad[idx[vm]] -= h[jm[vm]]
    return grad


def grad_q_single(q, r):
    m = len(q)
    n = 2 * m
    h = build_h_from_q(q)
    g = grad_corr_full(h, r)

    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(100):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)

            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:

        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

            h_work = build_h_from_q(q_opt)
            corrs = compute_corrs(h_work)
            t_init = corrs.max()
            active = select_active(corrs, top_k=70)
            q_final, t_val = solve_epigraph_q(q_opt, active, t_init, 400)
            h_final = build_h_from_q(q_final)
            c5_final = exact_score(h_final)
            if c5_final < best_c5:
                best_c5 = c5_final
                best_h = h_final.copy()
                n_cur = nc

    # Phase 6: Final polish with full-space
    if remaining_time(t0, budget_s) > 20:
        n_final = len(best_h)
        target_sum = n_final / 2.0
        h_work = best_h.copy()
        corrs = compute_corrs(h_work)
        t_init = corrs.max()
        active = select_active(corrs, top_k=100)
        h_new, t_val = solve_full_epigraph(h_work, active, t_init, target_sum, 600)
        c5_new = exact_score(h_new)
        if validate_h(h_new) and c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Final validation
    if not validate_h(best_h):
        best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_c5 = exact_score(best_h)

    return best_h, best_c5, len(best_h)
