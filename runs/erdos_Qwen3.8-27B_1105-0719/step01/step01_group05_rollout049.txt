def project_box_sum(h, target=None):

    h = np.clip(h, 0.0, 1.0)

        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
            count = np.sum(mask)
            if count == 0:
                break
            step = min(diff / count, 1e-6)
            h[mask] = np.minimum(h[mask] + diff / count, 1.0)
        else:
            mask = h > 0.0
            count = np.sum(mask)
            if count == 0:

def correlation_grad(h, lag):
    """Compute gradient of q_r with respect to h, where lag = r - (n-1)."""
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    if lag >= 0:
        if lag < n:
            # Pairs (i, i+lag) for i = 0..n-lag-1
            i_idx = np.arange(n - lag)
            j_idx = i_idx + lag
            grad[i_idx] += (1.0 - h[j_idx])
            grad[j_idx] -= h[i_idx]
    else:
        a = -lag
        if a < n:
            # Pairs (i, i-a) for i = a..n-1
            i_idx = np.arange(a, n)
            j_idx = i_idx - a
            grad[i_idx] += (1.0 - h[j_idx])
            grad[j_idx] -= h[i_idx]
    return grad * dx

def correlation_grad(h, lag):
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    if lag >= 0:

            s = n - lag
            grad[:s] += (1.0 - h[lag:])
            grad[lag:] -= h[:s]
    else:
        a = -lag
        if a < n:
            s = n - a
            grad[a:] += (1.0 - h[:s])
            grad[:s] -= h[a:]
    return grad * dx

    for alpha in np.linspace(0.1, 0.9, n_steps):
        h = (1.0 - alpha) * ha + alpha * hb
        h = project_box_sum(h, 0.5 * n)
        c, _ = exact_objective(h)
        if c < best_c - 1e-12:
            best_c = c
            best_h = h
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    deadline = start_time + budget_s - 3
    rng = np.random.RandomState(seed)
    
    best_h = None
    best_c = float('inf')
    
    try:
        init = np.asarray(initial_h_values, dtype=np.float64)
        if init.size > 0 and np.all(np.isfinite(init)):
            n0 = len(init)
            init = project_box_sum(init, 0.5 * n0)
            c0, _ = exact_objective(init)
            best_h = init.copy()
            best_c = c0
    except (NameError, Exception):
        pass
    
    if best_h is None:
        n0 = 70
        best_h = make_start(n0, rng, mode=0)
        best_c, _ = exact_objective(best_h)
    
    n0 = len(best_h)
    
    # Phase 1: Smooth refinement on parent
    tau_list = [0.01, 0.005, 0.003, 0.001, 0.0005, 0.0003, 0.0001]
    if time.time() < deadline - 80:
        h_opt, c_opt = smooth_refine(best_h, deadline, tau_list)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    # Phase 2: Active set SLSQP
    for n_active in [6, 10, 15]:
        if time.time() > deadline - 80:
            break
        h_opt, c_opt = active_set_slsqp(best_h, best_c, deadline, max_active=n_active, max_rounds=6)
        if c_opt < best_c - 1e-11:
            best_c = c_opt
            best_h = h_opt
    
    # Phase 3: Local moves
    if time.time() < deadline - 40:
        h_opt, c_opt = local_moves(best_h, max_rounds=50, rng=rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    # Phase 4: Binary + refine
    if len(best_h) % 2 == 0 and time.time() < deadline - 30:
        b = binary_threshold(best_h)
        if b is not None:
            b, c_b = binary_swap_improve(b, max_rounds=50, rng=rng)
            if c_b < best_c:
                best_c = c_b
                best_h = b
            if time.time() < deadline - 20:
                h_opt, c_opt = smooth_refine(best_h, deadline, [0.001, 0.0005, 0.0002, 0.0001])
                if c_opt < best_c:
                    best_c = c_opt
                    best_h = h_opt
                if time.time() < deadline - 10:
                    h_opt, c_opt = active_set_slsqp(best_h, best_c, deadline, max_active=8, max_rounds=5)
                    if c_opt < best_c:
                        best_c = c_opt
                        best_h = h_opt
    
    # Phase 5: Explore other sizes
    n_best = len(best_h)
    candidate_sizes = []
    for s in range(16, 130):
        if s % 2 == 0:
            candidate_sizes.append(s)
    candidate_sizes.sort(key=lambda s: abs(s - n_best))
    
    for n_try in candidate_sizes[:30]:
        if time.time() > deadline - 15:
            break
        if n_try == n_best:
            continue
        
        best_for_size = None
        best_c_size = float('inf')
        
        starts = [downsample(best_h, n_try)]
        for mode in range(4):
            starts.append(make_start(n_try, rng, mode=mode))
        
        for h_start in starts:
            if time.time() > deadline - 12:
                break
            h_opt, c_opt = smooth_refine(h_start, deadline, [0.005, 0.002, 0.001])
            if c_opt < best_c_size:
                best_c_size = c_opt
                best_for_size = h_opt
        
        if best_for_size is not None and best_c_size < best_c - 1e-9:
            best_c = best_c_size
            best_h = best_for_size
            n_best = n_try
        
        if best_for_size is not None and best_c_size < best_c + 0.005 and time.time() < deadline - 10:
            h_opt, c_opt = active_set_slsqp(best_for_size, best_c_size, deadline, max_active=8, max_rounds=4)
            if c_opt < best_c - 1e-9:
                best_c = c_opt
                best_h = h_opt
                n_best = n_try
        
        if n_try % 2 == 0 and best_for_size is not None and best_c_size < best_c + 0.01 and time.time() < deadline - 8:
            b = binary_threshold(best_for_size)
            if b is not None:
                b, c_b = binary_swap_improve(b, max_rounds=30, rng=rng)
                if c_b < best_c - 1e-9:
                    best_c = c_b
                    best_h = b
                    n_best = n_try
    
    # Phase 6: Final intensive refinement
    if time.time() < deadline - 8:
        h_opt, c_opt = smooth_refine(best_h, deadline, [0.0005, 0.0003, 0.0001])
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    if time.time() < deadline - 5:
        h_opt, c_opt = active_set_slsqp(best_h, best_c, deadline, max_active=10, max_rounds=6)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    if time.time() < deadline - 3:
        h_opt, c_opt = local_moves(best_h, max_rounds=20, rng=rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    
    if len(best_h) % 2 == 0 and time.time() < deadline - 2:
        b = binary_threshold(best_h)
        if b is not None:
            b, c_b = binary_swap_improve(b, max_rounds=20, rng=rng)
            if c_b < best_c:
                best_c = c_b
                best_h = b
    
    # Final validation
    best_h = project_box_sum(best_h, 0.5 * len(best_h))
    final_c, _ = exact_objective(best_h)
    best_c = final_c
    best_n = len(best_h)
    
    return (best_h, best_c, best_n)
