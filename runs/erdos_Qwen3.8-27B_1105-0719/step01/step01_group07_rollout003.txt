            if not np.any(mask):
                break
            h[mask] = np.maximum(0.0, h[mask] + diff / np.sum(mask))
    return np.clip(h, 0.0, 1.0)

def random_binary(n, rng):
    h = np.zeros(n)
    positions = rng.permutation(n)
    h[positions[:n//2]] = 1.0
    return h

def binary_threshold(h):
    n = len(h)
    if n % 2 != 0:
        return None
    m = n // 2
    idx = np.argsort(h)[::-1][:m]
    b = np.zeros(n)
    b[idx] = 1.0
    return b

def sa_binary(h0, t_start, t_end, n_iter, rng, deadline):
    """Simulated annealing on binary sequences"""
    n = len(h0)
    m = n // 2
    h = h0.copy()
    curr_c = exact_objective(h)
    best_h = h.copy()
    best_c = curr_c
    
    ones_list = list(np.where(h == 1.0)[0])
    zeros_list = list(np.where(h == 0.0)[0])
    
    no_improve = 0
    patience = 5000  # iterations without improvement before early stop
    
    for step in range(n_iter):
        if time.time() > deadline:
            break
        
        T = t_start * (t_end / t_start) ** (step / n_iter)
        
        # Choose move type
        r = rng.random()
        if r < 0.7:
            # Single swap
            i = ones_list[rng.randint(len(ones_list))]
            j = zeros_list[rng.randint(len(zeros_list))]
            h[i] = 0.0
            h[j] = 1.0
            new_c = exact_objective(h)
            delta = new_c - curr_c
            if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
                curr_c = new_c
                ones_list.remove(i)
                ones_list.append(j)
                zeros_list.remove(j)
                zeros_list.append(i)
                if curr_c < best_c - 1e-12:
                    best_c = curr_c
                    best_h = h.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                h[i] = 1.0
                h[j] = 0.0
                no_improve += 1
        else:
            # Double swap
            if len(ones_list) >= 2 and len(zeros_list) >= 2:
                idx1 = rng.randint(len(ones_list))
                idx2 = rng.randint(len(ones_list) - 1)
                if idx1 > idx2:
                    idx1, idx2 = idx2, idx1
                    idx1 += 1
                i1 = ones_list[idx1]
                i2 = ones_list[idx2]
                idx3 = rng.randint(len(zeros_list))
                idx4 = rng.randint(len(zeros_list) - 1)
                if idx3 > idx4:
                    idx3, idx4 = idx4, idx3
                    idx3 += 1
                j1 = zeros_list[idx3]
                j2 = zeros_list[idx4]
                
                h[i1] = 0.0; h[i2] = 0.0
                h[j1] = 1.0; h[j2] = 1.0
                new_c = exact_objective(h)
                delta = new_c - curr_c
                if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
                    curr_c = new_c
                    # Update lists
                    ones_list.remove(i1); ones_list.remove(i2)
                    zeros_list.remove(j1); zeros_list.remove(j2)
                    ones_list.extend([j1, j2])
                    zeros_list.extend([i1, i2])
                    if curr_c < best_c - 1e-12:
                        best_c = curr_c
                        best_h = h.copy()
                        no_improve = 0
                    else:
                        no_improve += 1
                else:
                    h[i1] = 1.0; h[i2] = 1.0
                    h[j1] = 0.0; h[j2] = 0.0
                    no_improve += 1
        
        if no_improve > patience:
            break
    
    return best_h, best_c

def local_search_binary(h, deadline):
    """Exhaustive single-swap local search"""
    n = len(h)
    curr_c = exact_objective(h)
    ones = list(np.where(h == 1.0)[0])
    zeros = list(np.where(h == 0.0)[0])
    
    improved = True
    while improved and time.time() < deadline:
        improved = False
        best_delta = 1e-14
        best_move = None
        for i in ones:
            for j in zeros:
                h[i] = 0.0
                h[j] = 1.0
                c = exact_objective(h)
                h[i] = 1.0
                h[j] = 0.0
                delta = curr_c - c
                if delta > best_delta:
                    best_delta = delta
                    best_move = (i, j)
        if best_move is not None:
            i, j = best_move
            h[i] = 0.0
            h[j] = 1.0
            curr_c = exact_objective(h)
            ones = list(np.where(h == 1.0)[0])
            zeros = list(np.where(h == 0.0)[0])
            improved = True
    
    return h, curr_c

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    deadline = t_start + budget_s - 10  # safety margin
    rng = np.random.RandomState(seed)
    
    best_h = None
    best_c = float('inf')
    best_n = 0
    
    # Load parent

        if len(parent_h) > 0:

                best_h = parent_h.copy()
                best_n = len(parent_h)
    except Exception:
        pass
    
    # Sizes to try
    sizes = [24, 26, 28, 30, 32, 34, 36, 38, 40, 42, 44, 46, 48]
    if parent_h is not None:
        pn = len(parent_h)
        if pn not in sizes:
            sizes.append(pn)
        sizes = [s for s in sizes if 16 <= s <= 60]
    sizes = sorted(set(sizes))
    
    # For each size, try multiple approaches
    for n in sizes:
        if time.time() > deadline - 30:
            break
        if n % 2 != 0:
            continue
        
        # Generate initial candidates
        cands = []
        
        # From parent (if available and different size)
        if parent_h is not None and len(parent_h) != n:
            x_src = (np.arange(len(parent_h)) + 0.5) / len(parent_h)
            x_tgt = (np.arange(n) + 0.5) / n
            h_interp = np.interp(x_tgt, x_src, parent_h)
            cands.append(project_box_sum(h_interp))
        
        # From parent (if same size)
        if parent_h is not None and len(parent_h) == n:
            cands.append(parent_h.copy())
        
        # Random binary
        for k in range(5):
            cands.append(random_binary(n, rng))
        
        # Structured: low-discrepancy-like
        h_struct = np.zeros(n)
        m = n // 2
        # Place ones as evenly as possible
        step = n / m
        for k in range(m):
            pos = int(round(k * step))
            h_struct[pos] = 1.0
        cands.append(h_struct)
        
        # Alternating
        h_alt = np.zeros(n)
        h_alt[::2] = 1.0
        if np.sum(h_alt) != m:
            h_alt = np.zeros(n)
            h_alt[1::2] = 1.0
            if np.sum(h_alt) != m:
                h_alt[m-1] = 0
                h_alt[m] = 1
        cands.append(h_alt)
        
        # Score and select top candidates
        scored = []
        for c in cands:
            # Make binary by thresholding
            if np.all((c == 0) | (c == 1)):
                b = c
            else:
                b = binary_threshold(c)
            if b is not None:
                sc = exact_objective(b)
                scored.append((sc, b))
        scored.sort(key=lambda x: x[0])
        
        # Run SA on top candidates
        n_top = min(3, len(scored))
        local_time = min(60, (deadline - time.time()) / max(1, len(sizes)))
        
        for idx in range(n_top):
            if time.time() > deadline - 15:
                break
            h0 = scored[idx][1].copy()
            
            # SA with different temperature schedules
            for trial in range(2):
                if time.time() > deadline - 15:
                    break
                t0 = 0.15 + rng.uniform(0, 0.1)
                t1 = 0.001
                n_iter = min(200000, int(local_time * 20000))  # rough
                h_sa, c_sa = sa_binary(h0, t0, t1, n_iter, rng, deadline)
                if c_sa < best_c - 1e-12:
                    best_c = c_sa
                    best_h = h_sa.copy()
                    best_n = n
                
                # Local search polish
                h_ls, c_ls = local_search_binary(h_sa.copy(), deadline - 5)
                if c_ls < best_c - 1e-12:
                    best_c = c_ls
                    best_h = h_ls.copy()
                    best_n = n
    
    # Final: local search on best
    if best_h is not None and time.time() < deadline - 5:
        if np.all((best_h == 0.0) | (best_h == 1.0)):
            h_ls, c_ls = local_search_binary(best_h.copy(), deadline)
            if c_ls < best_c - 1e-12:
                best_c = c_ls
                best_h = h_ls.copy()
    
    # Final validation
    best_h = project_box_sum(best_h)
    best_c = exact_objective(best_h)
    best_n = len(best_h)
    
    if abs(np.sum(best_h) - best_n / 2.0) > 1e-8:
        best_h = project_box_sum(best_h)
        best_c = exact_objective(best_h)
    
    return best_h, best_c, best_n
