def smooth_objective(h, tau, penalty=1000):
    n = len(h)
    dx = 2.0 / n
    profile = np.correlate(h, 1 - h, mode="full") * dx
    # Log-sum-exp
    max_r = np.max(profile)
    lse = max_r + tau * np.log(np.sum(np.exp((profile - max_r) / tau)))
    # Penalty for sum constraint
    sum_penalty = penalty * (h.sum() - n/2.0)**2
    return lse + sum_penalty

def smooth_gradient(h, tau, penalty=1000):

import numpy as np

import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def repair_feasible(vec):
    """Project vec to [0,1]^n with sum = n/2 using water-filling."""
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    # Bisection to find shift a such that sum(clip(h+a, 0, 1)) = target
    lo, hi = -2.0, 2.0
    # Expand bracket
    while np.sum(np.clip(h + lo, 0, 1)) > target:
        lo *= 2
    while np.sum(np.clip(h + hi, 0, 1)) < target:
        hi *= 2
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h + mid, 0, 1))
        if s > target:
            hi = mid
        else:
            lo = mid
    a = (lo + hi) / 2
    h_new = np.clip(h + a, 0, 1)
    # Fix residual
    diff = target - h_new.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            order = np.argsort(1.0 - h_new)
        else:
            order = np.argsort(h_new)
        for idx in order:
            adjust = min(abs(diff), max(0, (1.0 - h_new[idx]) if diff > 0 else h_new[idx]))
            h_new[idx] += np.sign(diff) * adjust
            diff -= np.sign(diff) * adjust
            if abs(diff) < 1e-15:
                break
    return h_new

def coordinate_descent_1d(h0, n_sweeps=5, grid_size=200):
    """Coordinate descent with exact 1D minimization (via grid)."""
    n = len(h0)
    target = n / 2.0
    h = h0.copy()
    best_c = exact_score(h)
    e = n - 1  # Eliminated variable
    
    for sweep in range(n_sweeps):
        improved = False
        order = list(range(n-1))  # All variables except e
        rng_order = np.random.permutation(n-1)
        for i in rng_order:
            # Current value of eliminated variable
            h_e = target - h[:n-1].sum()
            # Feasible range for delta = h_i change
            delta_lo = max(-h[i], h_e - 1.0)
            delta_hi = min(1.0 - h[i], h_e)
            if delta_hi - delta_lo < 1e-10:
                continue
            # Grid search
            deltas = np.linspace(delta_lo, delta_hi, grid_size)
            best_delta = 0
            best_val = best_c
            h_saved = h[i]
            for d in deltas:
                h[i] = h_saved + d
                h[e] = target - h[:n-1].sum()
                if h[e] < -1e-10 or h[e] > 1 + 1e-10:
                    continue
                c_try = exact_score(h)
                if c_try < best_val:
                    best_val = c_try
                    best_delta = d
            h[i] = h_saved + best_delta
            h[e] = target - h[:n-1].sum()
            h[e] = np.clip(h[e], 0, 1)
            # Re-sum to fix
            h = repair_feasible(h)
            if best_val < best_c:
                best_c = best_val
                improved = True
        if not improved:
            break
    
    return h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    lo, hi = -2.0, 2.0
    while np.sum(np.clip(h + lo, 0.0, 1.0)) > target and lo > -100:
        lo *= 2.0
    while np.sum(np.clip(h + hi, 0.0, 1.0)) < target and hi < 100:
        hi *= 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h + mid, 0.0, 1.0)) > target:
            hi = mid
        else:
            lo = mid
    h_new = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - h_new.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            mask = h_new < 1.0 - 1e-12
        else:
            mask = h_new > 1e-12
        indices = np.where(mask)[0]
        if len(indices) > 0:
            per = diff / len(indices)
            h_new[indices] += per
            h_new = np.clip(h_new, 0.0, 1.0)
    diff2 = target - h_new.sum()
    if abs(diff2) > 1e-15:
        idx = int(np.argmax(1.0 - h_new)) if diff2 > 0 else int(np.argmax(h_new))
        h_new[idx] += diff2
        h_new = np.clip(h_new, 0.0, 1.0)
    return h_new

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def build_gradient(h):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    G = np.zeros((M, n))
    G[n - 1, :] = dx * (1.0 - 2.0 * h)
    for k in range(1, n):
        G[n - 1 + k, k:] += dx * (1.0 - h[:n - k])
        G[n - 1 + k, :n - k] += -dx * h[k:]
        G[n - 1 - k, :n - k] += dx * (1.0 - h[k:])
        G[n - 1 - k, k:] += -dx * h[:n - k]
    return G

def slqp_solve(h0, maxiter=400):
    n = len(h0)
    target = n / 2.0
    M = 2 * n - 1
    c0 = exact_score(h0)
    G_cache = [None]

    def _obj(x):
        return x[-1]

    def _obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g

    def _con_eq(x):
        return x[:-1].sum() - target

    def _con_eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g

    def _con_ineq(x):
        return x[-1] - compute_profile(x[:-1])

    def _con_ineq_jac(x):
        G = build_gradient(x[:-1])
        J = np.zeros((M, n + 1))
        J[:, :-1] = -G
        J[:, -1] = 1.0
        return J

    x0 = np.concatenate([h0, [c0 + 5e-4]])
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 1.0)]
    cons = [{'type': 'eq', 'fun': _con_eq, 'jac': _con_eq_jac},
            {'type': 'ineq', 'fun': _con_ineq, 'jac': _con_ineq_jac}]
    try:
        res = minimize(_obj, x0, method='SLSQP', jac=_obj_jac,
                       bounds=bounds, constraints=cons,

        if res.x is not None and np.isfinite(res.x).all():

    except Exception:

    return None

def lbfgs_smooth(h0, tau_start=0.004, tau_factor=0.6, n_tau=14, maxiter=200):
    n = len(h0)
    target = n / 2.0
    h = h0.copy()
    best_c = exact_score(h)
    best_h = h.copy()
    for itau in range(n_tau):
        tau = tau_start * (tau_factor ** itau)
        penalty = 200.0 + 300.0 * itau
        def _obj(x):
            prof = compute_profile(x)
            mx = np.max(prof)
            lse = mx + tau * np.log(np.sum(np.exp((prof - mx) / tau)))
            return lse + penalty * (x.sum() - target) ** 2
        def _grad(x):
            prof = compute_profile(x)
            mx = np.max(prof)
            w = np.exp((prof - mx) / tau)
            w /= w.sum()
            G = build_gradient(x)
            g = G.T @ w
            g += 2.0 * penalty * (x.sum() - target)
            return g
        try:
            res = minimize(_obj, h, jac=_grad, method='L-BFGS-B',
                           bounds=[(0.0, 1.0)] * n,
                           options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-9})
            h_new = repair_feasible(res.x)
            c_new = exact_score(h_new)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
                h = h_new.copy()
        except Exception:
            pass
    return best_h, best_c

def subgradient_descent(h0, n_iters=200, init_alpha=0.05):
    n = len(h0)
    h = h0.copy()
    best_c = exact_score(h)
    best_h = h.copy()
    alpha = init_alpha
    for t in range(n_iters):
        prof = compute_profile(h)
        c = np.max(prof)
        tau_s = max(c * 0.08, 1e-5)
        w = np.exp((prof - c) / tau_s)
        w /= w.sum()
        G = build_gradient(h)
        g = G.T @ w
        g -= g.mean()
        ng = np.linalg.norm(g)
        if ng < 1e-15:
            break
        h_try = repair_feasible(h - alpha * g)
        c_try = exact_score(h_try)
        steps = 0
        while c_try >= c - 1e-13 and steps < 25:
            alpha *= 0.5
            h_try = repair_feasible(h - alpha * g)
            c_try = exact_score(h_try)
            steps += 1
        if c_try < c - 1e-13:
            h = h_try
            c = c_try
            if c < best_c:
                best_c = c
                best_h = h.copy()
            alpha = min(alpha * 1.1, init_alpha)
        else:
            alpha *= 0.8
    return best_h, best_c

def interp_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)

    fh = np.concatenate([[h[0]], h, [h[-1]]])

def coordinate_descent(h0, n_sweeps=3, grid_pts=80):
    n = len(h0)
    target = n / 2.0
    h = h0.copy()
    best_c = exact_score(h)
    e = n - 1
    rng = np.random.RandomState(123)
    for sweep in range(n_sweeps):
        order = rng.permutation(n - 1)
        improved = False
        for i in order:
            h_e = target - h[:n - 1].sum()
            h_e = np.clip(h_e, 0.0, 1.0)
            d_lo = max(-h[i], h_e - 1.0)
            d_hi = min(1.0 - h[i], h_e)
            if d_hi - d_lo < 1e-8:
                continue
            deltas = np.linspace(d_lo, d_hi, grid_pts)
            best_d = 0.0
            best_v = exact_score(h)
            hv = h[i]
            for d in deltas:
                h[i] = hv + d
                h[e] = target - h[:n - 1].sum()
                h[e] = np.clip(h[e], 0.0, 1.0)
                cv = exact_score(h)
                if cv < best_v:
                    best_v = cv
                    best_d = d
            h[i] = hv + best_d
            h[e] = target - h[:n - 1].sum()
            h = repair_feasible(h)
            if best_v < best_c - 1e-13:
                best_c = best_v
                improved = True
        if not improved:
            break
    return h, best_c

        if isinstance(eval_result, (float, int)):
            final_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 2:
            final_c = float(eval_result[0])
    except Exception:
        pass

    return best_h, final_c, len(best_h)
