```python
def optimal_transfer_delta(h, p, q, cap):
    """Find optimal δ to transfer from q to p, minimizing max_k R_k."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")  # unscaled
    max_corr = np.max(corr)
    
    # Compute d_k for each lag
    # d_k = g_k[p] - g_k[q] where g_k[i] = (1-h[i+k]) - h[i-k]
    # (with appropriate bounds)
    
    d = np.zeros(2*n - 1)
    for lag_idx in range(2*n - 1):
        k = lag_idx - (n - 1)
        # g_k[p] = (1 - h[p+k]) - h[p-k] with bounds
        gp = 0.0
        if 0 <= p + k < n:
            gp += (1.0 - h[p + k])
        if 0 <= p - k < n:
            gp -= h[p - k]
        
        gq = 0.0
        if 0 <= q + k < n:
            gq += (1.0 - h[q + k])
        if 0 <= q - k < n:
            gq -= h[q - k]
        
        d[lag_idx] = gp - gq
    
    # Now minimize f(δ) = max_k (corr[k] + δ * d[k]) for δ in [0, cap]
    # f is convex piecewise linear
    
    # Bisection to find minimum
    lo, hi = 0.0, cap
    for _ in range(40):
        mid = (lo + hi) / 2.0
        # f'(mid) = max of d[k] for k that achieve the max at δ=mid
        vals = corr + mid * d
        max_val = np.max(vals)
        active = np.where(np.abs(vals - max_val) < 1e-10)[0]
        f_prime = np.max(d[active])
        if f_prime > 1e-10:
            lo = mid
        elif f_prime < -1e-10:
            hi = mid
        else:
            break
    
    delta_opt = (lo + hi) / 2.0
    
    # Evaluate at delta_opt and boundaries
    best_delta = 0.0
    best_val = max_corr
    for test_d in [0.0, delta_opt, cap, delta_opt * 0.5, delta_opt * 0.75]:
        if test_d < 0 or test_d > cap:
            continue
        val = np.max(corr + test_d * d)
        if val < best_val:
            best_val = val
            best_delta = test_d
    
    return best_delta, best_val
```
