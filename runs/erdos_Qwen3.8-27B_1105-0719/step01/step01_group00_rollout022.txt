def project_feasible(x, target_sum):
    n = len(x)
    if n == 0:
        return x
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        s = h.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
        else:
            # If all are at boundary, adjust the one closest to 0.5
            idx = np.argmin(np.abs(h - 0.5))
            h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return np.clip(h, 0.0, 1.0)

def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def c5_of_h(h, dx):
    return compute_corrs(h).max() * dx

def grad_corr_k(h, k, n):
    offset = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -offset)
    i_hi = min(n - 1, n - 1 - offset)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx + offset] += (1.0 - h[idx])
    grad[idx] += (-h[idx + offset])
    return grad

def smooth_max_grad(h, dx, beta):
    n = len(h)
    corrs = compute_corrs(h)
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones_like(corrs) / len(corrs)
    else:
        w /= w_sum
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        grad += w[k] * grad_corr_k(h, k, n)
    return grad * dx

def smooth_max_value(corrs, beta, dx):
    return logsumexp(beta * corrs) / beta * dx

def fista(h0, dx, target_sum, beta, max_iter, tol=1e-12):
    n = len(h0)
    x = h0.copy()
    y = h0.copy()
    t_k = 1.0
    best_h = h0.copy()
    best_c5 = c5_of_h(h0, dx)
    
    # Estimate Lipschitz constant
    # For smooth max of bilinear functions, L ≈ beta * (2*dx * max stuff)
    # Be conservative
    L = beta * 4.0 * dx + 1e-8
    step = 1.0 / L
    
    for it in range(max_iter):
        grad = smooth_max_grad(y, dx, beta)
        x_new = y - step * grad
        x_new = project_feasible(x_new, target_sum)
        
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = x_new + ((t_k - 1.0) / t_next) * (x_new - x)
        x = x_new
        t_k = t_next
        
        c5_val = c5_of_h(x, dx)
        if c5_val < best_c5 - tol:
            best_c5 = c5_val
            best_h = x.copy()
    
    return best_h, best_c5

def slsqp_optimize(h0, dx, target_sum, beta, maxiter):
    n = len(h0)
    def obj(x):
        return smooth_max_value(compute_corrs(x), beta, dx)
    
    def grad(x):
        return smooth_max_grad(x, dx, beta)
    
    def cons_fun(x):
        return x.sum() - target_sum
    
    def cons_jac(x):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    bounds = [(0.0, 1.0)] * n
    
    try:
        res = minimize(obj, h0, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        h_out = project_feasible(res.x, target_sum)
        return h_out, c5_of_h(h_out, dx)
    except Exception:
        return project_feasible(h0, target_sum), c5_of_h(h0, dx)

**Key idea: Use the structure of the problem more directly.**

The problem is: minimize max_k c_k(h) subject to 0 ≤ h ≤ 1, sum(h) = n/2.

This is equivalent to: find the smallest t such that there exists h with c_k(h) ≤ t for all k, 0 ≤ h ≤ 1, sum(h) = n/2.

For a fixed t, the feasibility problem is:
c_k(h) ≤ t for all k
0 ≤ h[i] ≤ 1
sum(h) = n/2

This is a system of quadratic inequalities, which is non-convex. However, we can use it in a bisection framework:
- Binary search on t
- At each t, try to find a feasible h (using local optimization)
- If we find one, we know C₅ ≤ t

But this doesn't directly help with the optimization.

Let me try yet another angle: **What if I formulate this as a sequence of LPs?**

At each iteration:
1. Start with current h
2. Compute active set A = {k : c_k(h) ≈ max}
3. Solve: minimize t subject to c_k(h') ≤ t for k ∈ A, 0 ≤ h' ≤ 1, sum(h') = n/2
   - This is still non-convex due to the quadratic constraints
   - But we can linearize: c_k(h') ≈ c_k(h) + ∇c_k(h) · (h' - h)
   - This gives an LP!
4. Solve the LP to get h'
5. Update active set
6. Repeat

This is essentially a **cutting-plane method** or **outer approximation**.

The LP at each step is:
minimize t
subject to c_k(h) + ∇c_k(h) · (h' - h) ≤ t for k ∈ A
           0 ≤ h'[i] ≤ 1
           sum(h') = n/2

This is a linear program with n+1 variables (h'[0], ..., h'[n-1], t) and |A| + 2n + 1 constraints. We can solve it with scipy's linprog or cvxpy.

This might be more effective than the smooth max approach because it directly attacks the max without smoothing.

    h = h0.copy()
    best_h = h0.copy()
    best_c5 = c5_of_h(h0, dx)
    
    for it in range(max_iter):
        corrs = compute_corrs(h) * dx
        max_c = corrs.max()
        
        if max_c < best_c5:
            best_c5 = max_c
            best_h = h.copy()
        
        # Active set: top few correlations
        sorted_idx = np.argsort(corrs)[::-1]
        n_active = min(20, len(corrs))
        active = sorted_idx[:n_active]
        
        # Build LP: minimize t
        # Variables: h'[0], ..., h'[n-1], t (total n+1 variables)
        # Constraints:
        #   for each k in active: sum_i (∂c_k/∂h[i]) * h'[i] ≤ t - c_k(h) + sum_i (∂c_k/∂h[i]) * h[i]
        #   0 ≤ h'[i] ≤ 1

        # For each active k:
        #   sum_i grad_k[i] * dx * h'[i] - t ≤ c_k(h)*dx - sum_i grad_k[i] * dx * h[i]
        #   i.e., sum_i (grad_k[i] * dx) * h'[i] - 1*t ≤ corrs[k] - sum_i (grad_k[i] * dx) * h[i]

        b_rhs = []

            b_ub = np.array(b_rhs)

            A_eq = np.zeros((1, n + 1))
            A_eq[0, :n] = 1.0
            b_eq = np.array([target_sum])
            
            # Bounds: 0 ≤ h'[i] ≤ 1, t is free (but we know t ≥ 0)
            bounds = [(0, 1)] * n + [(0, None)]

            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,

            if res.success:
                h_new = project_feasible(res.x[:n], target_sum)
                c5_new = c5_of_h(h_new, dx)

def c5_of_h(h, dx):
    return compute_corrs(h).max() * dx

def grad_corr_k(h, k, n):
    offset = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -offset)
    i_hi = min(n - 1, n - 1 - offset)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx + offset] += (1.0 - h[idx])
    grad[idx] += (-h[idx + offset])
    return grad

def smooth_max_grad(h, dx, beta):
    n = len(h)
    corrs = compute_corrs(h)
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones_like(corrs) / len(corrs)
    else:
        w /= w_sum
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        grad += w[k] * grad_corr_k(h, k, n)
    return grad * dx

def smooth_max_value(corrs, beta, dx):
    return logsumexp(beta * corrs) / beta * dx

def slsqp_optimize(h0, dx, target_sum, beta, maxiter):
    n = len(h0)
    def obj(x):
        return smooth_max_value(compute_corrs(x), beta, dx)
    def grad(x):
        return smooth_max_grad(x, dx, beta)
    def cons_fun(x):
        return x.sum() - target_sum
    def cons_jac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        h_out = project_feasible(res.x, target_sum)
        return h_out, c5_of_h(h_out, dx)
    except Exception:
        return project_feasible(h0, target_sum), c5_of_h(h0, dx)

def fista(h0, dx, target_sum, beta, max_iter):
    n = len(h0)
    x = h0.copy()
    y = h0.copy()
    t_k = 1.0
    best_h = h0.copy()
    best_c5 = c5_of_h(h0, dx)
    step = 1.0 / (beta * 3.0 * dx + 1e-10)
    
    for it in range(max_iter):
        grad = smooth_max_grad(y, dx, beta)
        x_new = y - step * grad
        x_new = project_feasible(x_new, target_sum)
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = x_new + ((t_k - 1.0) / t_next) * (x_new - x)
        x = x_new
        t_k = t_next
        c5_val = c5_of_h(x, dx)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = x.copy()
    return best_h, best_c5

def cutting_plane(h0, dx, target_sum, max_iter, n_active_max=30):
    n = len(h0)
    h = h0.copy()
    best_h = h0.copy()
    best_c5 = c5_of_h(h0, dx)
    damping = 0.5
    
    for it in range(max_iter):
        corrs_raw = compute_corrs(h)
        corrs = corrs_raw * dx
        max_c = corrs.max()
        
        if max_c < best_c5:
            best_c5 = max_c
            best_h = h.copy()
        
        # Active set
        sorted_idx = np.argsort(corrs)[::-1]
        n_active = min(n_active_max, len(corrs))
        active = sorted_idx[:n_active]
        
        # Build LP
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0
        
        A_rows = []
        b_rhs = []
        
        for k in active:
            gk = grad_corr_k(h, k, n) * dx
            A_rows.append(np.concatenate([gk, [-1.0]]))
            b_rhs.append(corrs[k] - np.dot(gk, h))
        
        A_ub = np.array(A_rows)
        b_ub = np.array(b_rhs)
        
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([target_sum])
        
        bounds = [(0, 1)] * n + [(0, None)]
        
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs')
        except Exception:
            break
        
        if not res.success:
            # Try with fewer active constraints
            if n_active > 5:
                continue
            break
        
        h_lp = project_feasible(res.x[:n], target_sum)
        c5_lp = c5_of_h(h_lp, dx)
        
        # Damped step
        h_new = (1 - damping) * h + damping * h_lp
        h_new = project_feasible(h_new, target_sum)
        c5_new = c5_of_h(h_new, dx)
        
        if c5_new > max_c * 1.01:
            # Bad step, reduce damping
            damping *= 0.7
            h_new = (1 - damping) * h + damping * h_lp
            h_new = project_feasible(h_new, target_sum)
            c5_new = c5_of_h(h_new, dx)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            damping = min(damping * 1.1, 1.0)
        else:
            damping = max(damping * 0.9, 0.1)
        
        h = h_new
        
        # Check convergence
        if it > 100 and (best_c5 - max_c) < 1e-10:
            break
    
    return best_h, best_c5

def make_golden_start(n, target_sum, alpha):
    phases = (np.arange(n) * alpha) % 1.0
    sorted_idx = np.argsort(phases)
    n_ones = int(target_sum)
    if n_ones >= n:
        return np.ones(n)
    h = np.zeros(n)
    h[sorted_idx[:n_ones]] = 1.0
    frac = target_sum - n_ones
    if frac > 1e-10 and n_ones < n:
        h[sorted_idx[n_ones]] = frac
    return h

def make_binary_start(n, target_sum, rng):
    h = np.zeros(n)
    idx = rng.permutation(n)
    n_ones = int(target_sum)
    h[idx[:n_ones]] = 1.0
    frac = target_sum - n_ones
    if frac > 1e-10 and n_ones < n:
        h[idx[n_ones]] = frac
    return h
