import time
from scipy.optimize import linprog

    dx = 2.0 / n

def lp_trust_step(h, G, profile, radius):
    """Solve LP trust region to find step direction.
    
    Variables: delta (n), t (1)
    Objective: minimize t
    Constraints:
      G @ delta - t <= -profile  (i.e., profile + G@delta - t <= 0)
      sum(delta) = 0
      -radius <= delta_i <= radius
    """
    n = len(h)
    M = 2 * n - 1
    
    # Variables: [delta_0, ..., delta_{n-1}, t]
    # Objective: minimize t
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize t
    
    # Inequality constraints: A_ub @ x <= b_ub
    # G @ delta - t <= -profile
    # [G, -1] @ [delta; t] <= -profile
    A_ub = np.zeros((M, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -profile
    
    # Equality: sum(delta) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    # Bounds: -radius <= delta_i <= radius, t >= 0 (unbounded above)
    bounds = [(-radius, radius)] * n + [(0, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                     bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], res.x[n]
    except Exception:
        pass
    return None, None

        if t_val >= score - 1e-10:

            radius *= 0.7

            if radius < 0.001:

import numpy as np
import time
from scipy.optimize import linprog

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))

def get_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def get_lag_gradients(h):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    G = np.zeros((M, n))
    for k in range(M):
        delta = k - (n - 1)
        if delta >= 0:
            if delta < n:
                L = n - delta
                G[k, delta:] += dx * (1.0 - h[:L])
                G[k, :L] -= dx * h[delta:]
        else:
            d = -delta
            if d < n:
                L = n - d
                G[k, :L] += dx * (1.0 - h[d:])
                G[k, d:] -= dx * h[:L]
    return G

def repair_feasible(h, target=None):
    h = np.array(h, dtype=np.float64).copy()
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    if target is None:
        target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        order = np.argsort(-room)
        remaining = diff
        for idx in order:
            if remaining <= 1e-15:
                break
            add = min(remaining, room[idx])
            h[idx] += add
            remaining -= add
    else:
        order = np.argsort(-h)
        remaining = -diff
        for idx in order:
            if remaining <= 1e-15:
                break
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
    return h

def interpolate_to_dim(vec, new_len):
    n = len(vec)
    x_src = (np.arange(n) + 0.5) * 2.0 / n
    x_tgt = (np.arange(new_len) + 0.5) * 2.0 / new_len
    x_ext = np.concatenate([[0.0], x_src, [2.0]])
    f_ext = np.concatenate([[vec[0]], vec, [vec[-1]]])
    h_new = np.interp(x_tgt, x_ext, f_ext)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

def lp_trust_step(h, G, profile, radius, top_k=None):
    n = len(h)
    M = 2 * n - 1
    c = np.zeros(n + 1)
    c[n] = 1.0
    if top_k is not None and top_k < M:
        idx = np.argsort(profile)[::-1][:top_k]
        G_sub = G[idx, :]
        p_sub = profile[idx]
    else:
        G_sub = G
        p_sub = profile
    ncon = G_sub.shape[0]
    A_ub = np.zeros((ncon, n + 1))
    A_ub[:, :n] = G_sub
    A_ub[:, n] = -1.0
    b_ub = -p_sub
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    bounds = [(-radius, radius)] * n + [(0.0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], res.x[n]
    except Exception:
        pass
    return None, None

def trust_region_optimize(h0, max_time=30.0, init_radius=0.12, top_k=None):

    h = h0.copy()

    t0 = time.perf_counter()
    radius = init_radius
    stagnation = 0

        profile = get_profile(h)
        score = float(np.max(profile))

def gradient_descent(h0, max_iters=50, time_limit=15.0):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_score = exact_score(h)
    t0 = time.perf_counter()
    
    for it in range(max_iters):
        if time.perf_counter() - t0 > time_limit:
            break
        profile = get_profile(h)
        score = float(np.max(profile))
        if score < best_score - 1e-12:
            best_score = score
            best_h = h.copy()
        A = max(3, min(15, n // 5))
        active_idx = np.argsort(profile)[::-1][:A]
        G = get_lag_gradients(h)
        w = profile[active_idx].copy()
        w = np.maximum(w - (w.min() - 0.01), 0.01)
        w /= w.sum()
        direction = np.sum(w[:, None] * G[active_idx, :], axis=0)
        direction -= direction.mean()
        norm = np.linalg.norm(direction)
        if norm < 1e-14:
            break
        direction /= norm
        improved = False
        for alpha in [0.04, 0.02, 0.01, 0.005, 0.002, 0.001]:
            h_trial = repair_feasible(h - alpha * direction)
            s_trial = exact_score(h_trial)
            if s_trial < best_score - 1e-12:
                h = h_trial
                best_h = h.copy()
                best_score = s_trial
                improved = True
                break
        if not improved:
            break
    return best_h, best_score

def make_structured_init(n, seed, style=0):
    rng = np.random.RandomState(seed)
    if style == 0:
        n_regions = rng.randint(3, 7)
        h = np.zeros(n)
        boundaries = np.sort(rng.choice(np.arange(1, n - 1), size=n_regions - 1, replace=False))
        val = rng.choice([0, 1])
        for i in range(n_regions):
            start = boundaries[i - 1] if i > 0 else 0
            end = boundaries[i] if i < n_regions - 1 else n
            h[start:end] = val
            val = 1 - val
    elif style == 1:
        center = rng.uniform(0.2, 0.8) * n
        width = rng.uniform(0.15, 0.35) * n
        x = np.arange(n)
        h = np.exp(-0.5 * ((x - center) / width) ** 2)
        h = np.clip(h, 0, 1)
    elif style == 2:
        freq = rng.randint(1, 4)
        phase = rng.uniform(0, 2 * np.pi)
        x = np.arange(n)
        h = 0.5 + 0.3 * np.sin(2 * np.pi * freq * x / n + phase)
        h = np.clip(h, 0, 1)
    else:
        h = np.full(n, 0.5)
        h[:n // 4] = 0.8
        h[3 * n // 4:] = 0.2
        h = h + rng.normal(0, 0.05, n)
        h = np.clip(h, 0, 1)
    return repair_feasible(h)

    top_k = min(10, len(profile))
    top_vals = np.sort(profile)[::-1][:top_k]
    return top_vals[0] - top_vals[-1]
