```python
    M = len(A)
    if M == 0:
        return 0.5, 0.0

    order = np.argsort(A)

    # A line is on the upper envelope if it's not dominated
    # For lines sorted by slope, the upper envelope is found by
    # maintaining a stack: a new line replaces the top of the stack
    # if it intersects the top at a point to the left of where
    # the top intersects the one below it

    # For M up to ~200, this is M^2/2 = 20000 checks per coordinate,
    # times n=70 coordinates = 1.4M checks per sweep. Still fast enough.
    
    # But actually, let me use the envelope approach which is cleaner:
    
    # Sort by slope, break ties by intercept (higher intercept first)
    idx = np.argsort(A, kind='stable')
    # For same slope, keep only the one with highest intercept
    # (it dominates the others)
    
    # Build envelope using stack
    env_lines = [idx[0]]
    for k in range(1, M):
        i = idx[k]
        # If same slope as top, keep the one with higher intercept
        while env_lines:
            j = env_lines[-1]
            if A[i] == A[j]:
                if B[i] >= B[j]:
                    env_lines.pop()
                else:
                    break
            else:
                # Compute intersection of line i and line j
                # A[i]*x + B[i] = A[j]*x + B[j]
                # x = (B[j] - B[i]) / (A[i] - A[j])
                x_int = (B[j] - B[i]) / (A[i] - A[j])
                if len(env_lines) >= 2:

                    prev = env_lines[-2]
                    x_prev = (B[prev] - B[j]) / (A[j] - A[prev])
                    if x_int >= x_prev:

                    break

            env_lines.append(i)

    best_x = lo

    # The minimum of the envelope is at a vertex (intersection of two adjacent envelope lines)

    candidates = [lo, hi]
```

```python
def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx] + 1e-15)
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx] + 1e-15)
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
```

```python
def solve_1d_minimax(A, B):
    """Find x in [0,1] minimizing max_m(A[m]*x + B[m])."""
    M = len(A)
    if M == 1:
        return 0.5  # doesn't matter, objective is constant
    
    # Sort by slope, break ties by intercept (descending)
    order = np.lexsort((-B, A))  # primary: A ascending, secondary: B descending
    
    # Build upper envelope (convex hull of lines viewed from above)
    # Stack of line indices
    env = []
    for k in range(M):
        idx = order[k]
        # Remove lines from stack that are no longer on the envelope
        while len(env) >= 2:
            i1 = env[-1]
            i2 = env[-2]
            # Intersection of new line with i1
            if A[idx] == A[i1]:
                if B[idx] > B[i1]:
                    env.pop()
                    continue
                else:
                    break
            x_new_i1 = (B[i1] - B[idx]) / (A[idx] - A[i1])
            # Intersection of i1 with i2
            if A[i1] == A[i2]:
                env.pop()
                continue
            x_i1_i2 = (B[i2] - B[i1]) / (A[i1] - A[i2])
            if x_new_i1 <= x_i1_i2:
                env.pop()
            else:
                break
        if not env or A[idx] != A[env[-1]]:
            env.append(idx)
    
    # Find minimum of upper envelope on [0,1]
    # Candidates: x=0, x=1, and intersections of adjacent envelope lines
    candidates = [0.0, 1.0]
    for k in range(len(env) - 1):
        i1 = env[k]
        i2 = env[k + 1]
        if A[i1] != A[i2]:
            x_int = (B[i2] - B[i1]) / (A[i2] - A[i1])
            if 0.0 <= x_int <= 1.0:
                candidates.append(x_int)
    
    best_x = 0.0
    best_val = float('inf')
    for x in candidates:
        vals = A * x + B
        v = vals.max()
        if v < best_val:
            best_val = v
            best_x = x
    
    return best_x
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx] + 1e-15)
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx] + 1e-15)
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

def solve_1d_minimax_bisect(A, B):
    M = len(A)
    lo, hi = 0.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        vals = A * mid + B
        amax = vals.max()
        active = vals >= (amax - 1e-14)
        smin = np.min(A[active])
        smax = np.max(A[active])
        if smin >= -1e-14:
            hi = mid
        elif smax <= 1e-14:
            lo = mid
        else:
            return mid
    return (lo + hi) / 2.0

def coord_descent(h, max_sweeps=300, tol=1e-13):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    h = h.copy()
    profile = compute_profile(h).copy()
    best_c = float(profile.max())
    d_all = np.arange(-(n - 1), n)
    for sweep in range(max_sweeps):
        max_change = 0.0
        for j in range(n):
            vp = (j + d_all >= 0) & (j + d_all < n)
            vm = (j - d_all >= 0) & (j - d_all < n)
            A = np.zeros(M)
            if np.any(vp):
                A[vp] += (1.0 - h[j + d_all[vp]])
            if np.any(vm):
                A[vm] -= h[j - d_all[vm]]
            A *= dx
            B = profile - h[j] * A
            x_opt = solve_1d_minimax_bisect(A, B)
            delta = x_opt - h[j]
            if abs(delta) > 1e-15:
                max_change = max(max_change, abs(delta))
                h[j] = x_opt
                profile += delta * A
        h = repair_feasible(h)
        profile = compute_profile(h)
        new_c = float(profile.max())
        if new_c < best_c:
            best_c = new_c
        else:
            profile -= profile  # reset
            profile = compute_profile(h)
        if max_change < tol:
            break
    return h, best_c

def build_gradient_matrix(h, n, dx):
    M = 2 * n - 1
    G = np.zeros((M, n))
    for m in range(M):
        d = m - (n - 1)
        if d == 0:
            G[m, :] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            G[m, d:n] += dx * (1.0 - h[0:n-d])
            G[m, 0:n-d] += -dx * h[d:n]
        else:
            dp = -d
            G[m, 0:n-dp] += dx * (1.0 - h[dp:n])
            G[m, dp:n] += -dx * h[0:n-dp]
    return G

class NDP:
    def __init__(self, n):
        self.n = n
        self.dx = 2.0 / n
        self.M = 2 * n - 1
    def obj(self, x):
        return x[-1]
    def obj_jac(self, x):
        g = np.zeros(self.n + 1)
        g[-1] = 1.0
        return g
    def eq_fun(self, x):
        return x[:-1].sum() - self.n / 2.0
    def eq_jac(self, x):
        g = np.zeros(self.n + 1)
        g[:-1] = 1.0
        return g
    def ineq_fun(self, x):
        h = x[:-1]
        t = x[-1]
        profile = np.correlate(h, 1 - h, mode="full") * self.dx
        return t - profile
    def ineq_jac(self, x):
        h = x[:-1]
        G = build_gradient_matrix(h, self.n, self.dx)
        J = np.zeros((self.M, self.n + 1))
        J[:, :-1] = -G
        J[:, -1] = 1.0
        return J

def solve_slsqp(h_start, maxiter=200, tol=1e-10):
    n = len(h_start)
    c0 = exact_score(h_start)
    t0 = c0 + 1e-5
    prob = NDP(n)
    x0 = np.concatenate([h_start, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 2.0)]
    cons = [
        {'type': 'eq', 'fun': prob.eq_fun, 'jac': prob.eq_jac},
        {'type': 'ineq', 'fun': prob.ineq_fun, 'jac': prob.ineq_jac}
    ]
    try:
        res = minimize(prob.obj, x0, method='SLSQP', jac=prob.obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': tol})
        if res.x is not None and np.isfinite(res.x).all():
            h_new = repair_feasible(res.x[:-1])
            if h_new.sum() > 0:
                return h_new
    except Exception:
        pass
    return None

def symmetrize(h):
    return (h + h[::-1]) / 2.0
```
