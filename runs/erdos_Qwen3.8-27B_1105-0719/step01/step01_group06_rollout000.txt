def exact_score(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))

def profile(h):
    dx = 2.0 / len(h)
    return np.correlate(h, 1.0 - h, mode="full") * dx

def lag_gradient(h):
    """Compute the gradient matrix G of shape (2n-1, n).
    Row k corresponds to lag d = k - (n-1).
    """
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    G = np.zeros((M, n))
    for k in range(M):
        d = k - (n - 1)
        if d > 0:
            L = n - d
            # c_d = dx * sum_{j=0}^{L-1} h[j] * (1-h[j+d])
            # ∂/∂h[a] = dx*(1-h[a+d]) * [a < L] - dx*h[a-d] * [a >= d]
            G[k, :L] += dx * (1.0 - h[d:d+L])
            G[k, d:] -= dx * h[:L]
        elif d < 0:
            c = -d
            L = n - c
            # c_{-c} = dx * sum_{i=0}^{L-1} h[i+c] * (1-h[i])
            # ∂/∂h[a] = dx*(1-h[a-c]) * [a >= c] - dx*h[a+c] * [a < L]
            G[k, c:] += dx * (1.0 - h[:L])
            G[k, :L] -= dx * h[c:c+L]
        else:
            G[k, :] = dx * (1.0 - 2.0 * h)
    return G

def repair(h, target=None):

def solve_lp_trust(h_center, rho, t_init, time_limit=5.0, n_active=0):

    We want: c_k(h_center) + G[k,:] @ d <= t for all k

    With constraints:
      |d[i]| <= rho (trust region)
      sum(d) = 0 (preserve integral)

def solve_epigraph_slsqp(h_start, t_start, beta=0.0, maxiter=50, n_constraints=0):
    n = len(h_start)
    M = 2 * n - 1
    nc = n_constraints if n_constraints > 0 else M
    
    if nc < M:
        p = profile(h_start)
        active_lags = np.argsort(p)[::-1][:nc]
    else:
        active_lags = np.arange(M)
    
    def obj(x):
        h = x[:n]
        t = x[n]
        return t + beta * np.sum(h * (1.0 - h))
    
    def obj_jac(x):
        h = x[:n]
        g = np.zeros(n + 1)
        g[:n] = beta * (1.0 - 2.0 * h)
        g[n] = 1.0
        return g
    
    def cons_fun(x):
        h = x[:n]
        t = x[n]
        prof = profile(h)
        ineq = t - prof[active_lags]
        eq = h.sum() - n / 2.0
        return np.concatenate([[eq], ineq])
    
    def cons_jac(x):
        h = x[:n]
        G = lag_gradient(h)
        Ja = G[active_lags, :]
        J = np.zeros((1 + len(active_lags), n + 1))
        J[0, :n] = 1.0
        J[1:, :n] = -Ja
        J[1:, n] = 1.0
        return J
    
    bounds = [(0.0, 1.0)] * n + [(0.0, t_start + 2.0)]
    x0 = np.concatenate([h_start, [t_start]])
    
    try:
        result = minimize(obj, x0, jac=obj_jac, method='SLSQP',
                          bounds=bounds,
                          constraints=[{'type': 'ineq', 'fun': cons_fun, 'jac': cons_jac}],
                          options={'maxiter': maxiter, 'ftol': 1e-12})
        if result.x is not None and np.all(np.isfinite(result.x)):
            return repair(result.x[:n])
    except Exception:
        pass
    return None

def local_descent(h_current, max_iters=40, time_limit=20):
    n = len(h_current)
    h = h_current.copy()
    best_h = h.copy()
    best_score = exact_score(h)
    t0 = time.perf_counter()
    
    for it in range(max_iters):
        if time.perf_counter() - t0 > time_limit:
            break
        p = profile(h)
        score = float(np.max(p))
        if score < best_score - 1e-12:
            best_score = score
            best_h = h.copy()
        
        A = max(5, min(30, n // 4))
        active_idx = np.argsort(p)[::-1][:A]
        G = lag_gradient(h)
        w = p[active_idx].copy()
        w = np.maximum(w, 0)
        wsum = w.sum()
        if wsum < 1e-15:
            break
        w /= wsum
        
        direction = np.sum(w[:, None] * G[active_idx, :], axis=0)
        direction -= direction.mean()
        norm = np.linalg.norm(direction)
        if norm < 1e-14:
            break
        direction /= norm
        
        improved = False
        for alpha in [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]:
            h_trial = repair(h - alpha * direction)
            s_trial = exact_score(h_trial)
            if s_trial < best_score - 1e-11:
                h = h_trial
                best_h = h.copy()
                best_score = s_trial
                improved = True
                break
        if not improved:
            break
    return best_h, best_score

def lp_trust_region(h_current, time_limit=60, start_rho=0.15):

    best_h = h.copy()
    best_score = exact_score(h)
    rho = start_rho
    t0 = time.perf_counter()
    no_improve = 0
    
    while time.perf_counter() - t0 < time_limit and rho > 0.005:
        # Determine active constraints
        p = profile(h)
        M = len(p)
        # Use top constraints
        n_active = min(M, max(40, n))
        top_idx = np.argsort(p)[::-1][:n_active]
        
        h_sol = solve_lp_trust(h, rho, best_score, time_limit=8.0, n_active=n_active)
        if h_sol is not None:
            s = exact_score(h_sol)
            if s < best_score - 1e-11:
                best_score = s
                best_h = h_sol.copy()
                h = h_sol.copy()
                rho = min(rho * 1.3, 0.4)
                no_improve = 0
            else:
                rho *= 0.6
                no_improve += 1
                if no_improve > 5:
                    break
        else:
            rho *= 0.6
            no_improve += 1
            if no_improve > 5:
                break
    
    # Follow up with local descent
    if time.perf_counter() - t0 < time_limit:
        h_fin, s_fin = local_descent(best_h, max_iters=20, 
                                     time_limit=max(1, time_limit - (time.perf_counter() - t0)))
        if s_fin < best_score - 1e-12:
            best_h = h_fin
            best_score = s_fin
    
    return best_h, best_score

def make_periodic_structure(n, period=4):
    """Create a periodic structure as a diverse starting point."""
    h = np.zeros(n)
    base = np.array([1.0, 1.0, 0.0, 0.0])[:period]
    for i in range(n):
        h[i] = base[i % period]
    return repair(h)

def make_cosine_structure(n, k=3, phase=0.0):
    """Create a cosine-based structure."""
    x = (np.arange(n) + 0.5) * 2.0 / n
    h = 0.5 + 0.4 * np.sin(2 * np.pi * k * x / 2.0 + phase)
    return repair(np.clip(h, 0.0, 1.0))
