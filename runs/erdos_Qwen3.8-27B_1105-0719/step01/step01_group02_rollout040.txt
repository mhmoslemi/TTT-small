from scipy.optimize import minimize

    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(profile))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_box_sum(h, target=None):
    n = len(h)
    h = np.array(h, dtype=np.float64).copy()
    h = np.nan_to_num(h, nan=0.0, posinf=1.0, neginf=0.0)
    if target is None:
        target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):

        diff = s - target
        if abs(diff) < 1e-14:

        if diff > 0:
            mask = h > 1e-15
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            min_val = np.min(h[mask])
            step = min(diff / cnt, min_val - 1e-15) if min_val > 1e-15 else diff / cnt
            h[mask] -= step
        else:
            mask = h < 1.0 - 1e-15
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            max_val = np.max(h[mask])
            step = min(-diff / cnt, 1.0 - max_val - 1e-15) if 1.0 - max_val > 1e-15 else -diff / cnt
            h[mask] += step
    return h

def smooth_obj_and_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    raw = np.correlate(h, 1.0 - h, mode='full')
    r_max = np.max(raw)
    shifted = np.clip((raw - r_max) / tau, -500.0, 500.0)
    exp_s = np.exp(shifted)
    log_sum = np.log(np.sum(exp_s))
    value = dx * (r_max + tau * log_sum)
    weights = exp_s / np.sum(exp_s)
    
    # Gradient computation
    grad = np.zeros(n)
    for i in range(2 * n - 1):

        w = weights[i]
        if w < 1e-14:

        elif k > 0:

For the binary local search, here's my vectorized approach:

Given binary b (n/2 ones, n/2 zeros):
- Compute profile = correlate(b, 1-b) (not scaled by dx yet)
- For each one-position i, compute rem[i][k] = b[i+k] + b[i-k] - 1 (with OOB=0)
  - This is the change in profile at shift k when we remove position i
- For each zero-position j, compute add[j][k] = 1 - b[j+k] - b[j-k] (with OOB=0)
  - This is the change in profile at shift k when we add position j
- For swap (i,j): new_profile[k] = profile[k] + rem[i][k] + add[j][k]
- Find (i,j) that minimizes max_k new_profile[k]

To vectorize:
- rem is a matrix of shape (n_ones, 2n-1)
- add is a matrix of shape (n_zeros, 2n-1)
- For each (i,j) pair, new_profile = profile + rem[i] + add[j]
- We want min over (i,j) of max_k (profile + rem[i] + add[j])

This is still O(n_ones * n_zeros * (2n-1)) which for n=140 is 70*70*279 ≈ 1.4M. That's fine for a single round.

                if 0 <= p-k < n:

import numpy as np
from scipy.optimize import minimize
import time

def exact_objective(h):
    n = len(h)
    if n == 0:
        return float('inf')
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(profile))

def project_box_sum(h, target=None):
    n = len(h)
    h = np.array(h, dtype=np.float64).copy()
    np.nan_to_num(h, copy=False, nan=0.0, posinf=1.0, neginf=0.0)
    if target is None:
        target = n / 2.0
    np.clip(h, 0.0, 1.0, out=h)
    for _ in range(200):
        s = np.sum(h)
        diff = s - target
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h > 0
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            step = min(diff / cnt, np.min(h[mask]))
            h[mask] -= step
        else:
            mask = h < 1.0
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            step = min(-diff / cnt, 1.0 - np.max(h[mask]))
            h[mask] += step
    return h

def smooth_obj_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    raw = np.correlate(h, 1.0 - h, mode='full')
    r_max = np.max(raw)
    shifted = np.clip((raw - r_max) / tau, -500.0, 500.0)
    exp_s = np.exp(shifted)
    s_exp = np.sum(exp_s)
    log_sum = np.log(s_exp)
    value = dx * (r_max + tau * log_sum)
    weights = exp_s / s_exp
    
    grad = np.zeros(n)
    n_prof = 2 * n - 1
    for i in range(n_prof):
        k = i - (n - 1)
        w = weights[i]
        if w < 1e-15:
            continue
        if k == 0:
            grad += w * (1.0 - 2.0 * h)
        elif k > 0:
            grad[k:n] += w * (1.0 - h[0:n-k])
            grad[0:n-k] -= w * h[k:n]
        else:
            a = -k
            grad[0:n-a] += w * (1.0 - h[a:n])
            grad[a:n] -= w * h[0:n-a]
    grad *= dx
    return value, grad

    sv = np.sum(x) - _g_n / 2.0
    return v + 1e4 * sv * sv, g + 2e4 * sv

    return np.sum(x) - _g_n / 2.0

    return np.ones(_g_n)

    global _g_n, _g_tau
    _g_n = len(h0)
    _g_tau = tau
    bounds = [(0.0, 1.0)] * _g_n
    cons = [{'type': 'eq', 'fun': _slsqp_eq, 'jac': _slsqp_eq_jac}]
    try:
        res = minimize(_slsqp_fun, h0, method='SLSQP', jac=True,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-10})

    except:

import numpy as np
from scipy.optimize import minimize
import time

def exact_objective(h):
    n = len(h)
    if n == 0:
        return float('inf')
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(profile))

def project_box_sum(h, target=None):
    n = len(h)
    h = np.array(h, dtype=np.float64).copy()
    np.nan_to_num(h, copy=False, nan=0.0, posinf=1.0, neginf=0.0)
    if target is None:
        target = n / 2.0
    np.clip(h, 0.0, 1.0, out=h)
    for _ in range(300):
        s = np.sum(h)
        diff = s - target
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h > 0
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            step = min(diff / cnt, np.min(h[mask]))
            h[mask] -= step
        else:
            mask = h < 1.0
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            step = min(-diff / cnt, 1.0 - np.max(h[mask]))
            h[mask] += step
    return h

_g_n = 0
_g_tau = 0.01

def smooth_obj_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    raw = np.correlate(h, 1.0 - h, mode='full')
    r_max = np.max(raw)
    shifted = np.clip((raw - r_max) / tau, -500.0, 500.0)
    exp_s = np.exp(shifted)
    s_exp = np.sum(exp_s)
    log_sum = np.log(s_exp)
    value = dx * (r_max + tau * log_sum)
    weights = exp_s / s_exp
    grad = np.zeros(n)
    n_prof = 2 * n - 1
    for i in range(n_prof):
        k = i - (n - 1)
        w = weights[i]
        if w < 1e-15:
            continue
        if k == 0:
            grad += w * (1.0 - 2.0 * h)
        elif k > 0:
            grad[k:n] += w * (1.0 - h[0:n-k])
            grad[0:n-k] -= w * h[k:n]
        else:
            a = -k
            grad[0:n-a] += w * (1.0 - h[a:n])
            grad[a:n] -= w * h[0:n-a]
    grad *= dx
    return value, grad

def _slsqp_fun(x):
    v, g = smooth_obj_grad(x, _g_tau)
    sv = np.sum(x) - _g_n / 2.0
    return v + 1e4 * sv * sv, g + 2e4 * sv

def _slsqp_eq(x):
    return np.sum(x) - _g_n / 2.0

def _slsqp_eq_jac(x):
    return np.ones(_g_n)

def optimize_slsqp(h0, tau, maxiter=100):
    global _g_n, _g_tau
    _g_n = len(h0)
    _g_tau = tau
    bounds = [(0.0, 1.0)] * _g_n
    cons = [{'type': 'eq', 'fun': _slsqp_eq, 'jac': _slsqp_eq_jac}]
    try:
        res = minimize(_slsqp_fun, h0, method='SLSQP', jac=True,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-11})
        return project_box_sum(res.x)
    except Exception:
        return None

def binary_local_search(b, max_rounds=80, deadline=None):
    n = len(b)
    dx = 2.0 / n
    cur = b.copy()
    cur_val = exact_objective(cur)
    for r in range(max_rounds):
        if deadline is not None and time.time() > deadline - 15:
            break
        profile = np.correlate(cur, 1.0 - cur, mode='full')
        ones = np.where(cur == 1.0)[0]
        zeros = np.where(cur == 0.0)[0]
        np_len = 2 * n - 1
        n_ones = len(ones)
        n_zeros = len(zeros)
        if n_ones == 0 or n_zeros == 0:
            break
        rem = np.zeros((n_ones, np_len))
        for ii, p in enumerate(ones):
            pk_plus = p + (np.arange(np_len) - (n - 1))
            pk_minus = p - (np.arange(np_len) - (n - 1))
            rem[ii, :] = (np.where((pk_plus >= 0) & (pk_plus < n), cur[pk_plus], 0.0) +
                          np.where((pk_minus >= 0) & (pk_minus < n), cur[pk_minus], 0.0) - 1.0)
        add = np.zeros((n_zeros, np_len))
        for jj, p in enumerate(zeros):
            pk_plus = p + (np.arange(np_len) - (n - 1))
            pk_minus = p - (np.arange(np_len) - (n - 1))
            add[jj, :] = (1.0 -
                          np.where((pk_plus >= 0) & (pk_plus < n), cur[pk_plus], 0.0) -
                          np.where((pk_minus >= 0) & (pk_minus < n), cur[pk_minus], 0.0))
        P = profile[None, :] + rem
        combined = P[:, :, None] + add[None, :, :]
        new_maxes = combined.max(axis=2) * dx
        deltas = cur_val - new_maxes
        best_idx = np.unravel_index(np.argmax(deltas), deltas.shape)
        best_delta = deltas[best_idx]
        if best_delta > 1e-12:
            cur[ones[best_idx[0]]] = 0.0
            cur[zeros[best_idx[1]]] = 1.0
            cur_val = cur_val - best_delta
        else:
            break
    return cur, cur_val

            break
        frac = step / float(n_steps)
        T = T0 * (T1 / T0) ** frac
        i_idx = rng.randint(len(ones))
        j_idx = rng.randint(len(zeros))
        p = ones[i_idx]
        q = zeros[j_idx]
        cur[p] = 0.0
        cur[q] = 1.0
        new_val = exact_objective(cur)
        delta = new_val - cur_val
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best = cur.copy()
            ones = np.where(cur == 1.0)[0]
            zeros = np.where(cur == 0.0)[0]
        else:
            cur[p] = 1.0
            cur[q] = 0.0
    return best, best_val

def transfer_to_size(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(target_n) + 0.5) / target_n
    new_h = np.interp(x_new, x_old, h)
    return project_box_sum(new_h)

def coordinate_descent(h, n_iter=200, deadline=None):
    n = len(h)
    cur = h.copy()
    cur_c = exact_objective(cur)
    best = cur.copy()
    best_c = cur_c
    for it in range(n_iter):
        if deadline is not None and time.time() > deadline - 10:
            break
        improved = False
        for i in range(n):
            old = cur[i]
            for trial in [0.0, 1.0, 0.5]:
                if trial == old:
                    continue
                cur[i] = trial
                new_h = project_box_sum(cur)
                nc = exact_objective(new_h)
                if nc < cur_c - 1e-14:
                    cur = new_h
                    cur_c = nc
                    improved = True
                    if cur_c < best_c:
                        best_c = cur_c
                        best = cur.copy()
                    break
            else:
                cur[i] = old
        if not improved:
            break
    return best, best_c

def run(seed=42, budget_s=1100, **kwargs):
    deadline = time.time() + budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c = float('inf')

    try:
        parent = np.array(initial_h_values, dtype=np.float64)
        parent = project_box_sum(parent)
        best_h = parent.copy()
        best_c = exact_objective(parent)
    except Exception:
        best_h = np.ones(140) * 0.5
        best_c = exact_objective(best_h)

    n_parent = len(best_h)

    sizes = []
    for s in [n_parent, 120, 100, 88, 80, 72, 64, 96, 112, 128]:
        if s % 2 == 0 and 48 <= s <= 200 and s not in sizes:
            sizes.append(s)

    for n in sizes:
        if time.time() > deadline - 50:
            break

        candidates = []
        if n_parent != n:
            tr = transfer_to_size(best_h, n)
            candidates.append(tr)
        else:
            candidates.append(best_h.copy())

        for sc in [0.03, 0.06, 0.1, 0.15, 0.25]:
            base = transfer_to_size(best_h, n)
            pert = base + sc * rng.randn(n)
            candidates.append(project_box_sum(pert))

        alt = np.zeros(n)
        alt[::2] = 1.0
        candidates.append(alt.copy())

        x = np.linspace(0, 4 * np.pi, n, endpoint=False)
        candidates.append(project_box_sum(0.5 + 0.3 * np.sin(x) + 0.1 * np.sin(3 * x)))

        x2 = np.linspace(0, 6 * np.pi, n, endpoint=False)
        candidates.append(project_box_sum(0.5 + 0.3 * np.sin(x2 + 0.7)))

        rb = np.zeros(n)
        idx = rng.choice(n, n // 2, replace=False)
        rb[idx] = 1.0
        candidates.append(rb)

        if n >= 64:
            pat_len = max(4, n // 16)
            pat = np.zeros(pat_len)
            pat[:pat_len // 2] = 1.0
            ms = np.tile(pat, n // pat_len + 1)[:n]
            candidates.append(project_box_sum(ms))

        scored = []
        seen = set()
        for c in candidates:
            c = project_box_sum(c)
            key = tuple(np.round(c, 3))
            if key in seen:
                continue
            seen.add(key)
            cc = exact_objective(c)
            scored.append((c, cc))
            if cc < best_c:
                best_c = cc
                best_h = c.copy()

        scored.sort(key=lambda z: z[1])
        top_k = min(3, len(scored))

        for ci in range(top_k):
            if time.time() > deadline - 45:
                break
            h_start, c_start = scored[ci]
            if c_start > best_c + 0.05:
                continue

            h_cur = h_start.copy()
            c_cur = c_start

            tau_stages = [0.05, 0.02, 0.008, 0.003, 0.001, 0.0004]
            for tau in tau_stages:
                if time.time() > deadline - 45:
                    break
                h_new = optimize_slsqp(h_cur, tau, maxiter=80)
                if h_new is not None:
                    c_new = exact_objective(h_new)
                    if c_new < c_cur - 1e-12:
                        h_cur = h_new
                        c_cur = c_new

            if n % 2 == 0:
                b = (h_cur >= 0.5).astype(float)
                if int(np.sum(b)) != n // 2:
                    idx = np.argsort(-h_cur)
                    b = np.zeros(n)
                    b[idx[:n // 2]] = 1.0
                b_c = exact_objective(b)
                if b_c < c_cur:
                    h_cur = b.copy()
                    c_cur = b_c

                b, b_c = binary_local_search(b, max_rounds=60, deadline=deadline)
                if b_c < c_cur:
                    h_cur = b.copy()
                    c_cur = b_c

            if time.time() > deadline - 40:
                break

            h_new = optimize_slsqp(h_cur, 0.0005, maxiter=60)
            if h_new is not None:
                c_new = exact_objective(h_new)
                if c_new < c_cur - 1e-12:
                    h_cur = h_new
                    c_cur = c_new

            if n % 2 == 0 and time.time() < deadline - 30:
                b = (h_cur >= 0.5).astype(float)
                if int(np.sum(b)) != n // 2:
                    idx = np.argsort(-h_cur)
                    b = np.zeros(n)
                    b[idx[:n // 2]] = 1.0
                b, b_c = binary_sa(b, n_steps=3000, T0=0.004,
                                   T1=0.0005, deadline=deadline, rng=rng)
                if b_c < c_cur:
                    h_cur = b.copy()
                    c_cur = b_c

            if c_cur < best_c:
                best_c = c_cur
                best_h = h_cur.copy()

    if time.time() < deadline - 20:
        best_h = project_box_sum(best_h)
        best_c = exact_objective(best_h)

    best_h = project_box_sum(best_h)
    best_c = exact_objective(best_h)
    return (best_h, best_c, len(best_h))
