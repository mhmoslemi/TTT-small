import numpy as np
from scipy.optimize import minimize
import time

def exact_objective(h):
    n = len(h)
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(profile))

def project_feasible(h, target_sum=None):
    n = len(h)
    if target_sum is None:
        target_sum = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = np.sum(h)
        diff = s - target_sum
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h > 0
            if not np.any(mask):
                break
            step = min(diff / np.sum(mask), np.min(h[mask]))
            h[mask] -= step
        else:
            mask = h < 1.0
            if not np.any(mask):
                break
            step = min(-diff / np.sum(mask), 1.0 - np.max(h[mask]))
            h[mask] += step
    h = np.clip(h, 0.0, 1.0)
    # Final fix
    diff = np.sum(h) - target_sum
    if abs(diff) > 1e-14:
        mask = (h > 0) if diff > 0 else (h < 1.0)
        if np.any(mask):
            h[mask] -= diff / np.sum(mask)
            h = np.clip(h, 0.0, 1.0)
    return h

def autocorr(h):
    """Compute R(k) = sum h[i]h[i+k] * dx for all k."""
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, h, mode='full') * dx

def subgradient_step(h, lr):
    """One subgradient step to maximize min_k R(k)."""
    n = len(h)
    dx = 2.0 / n
    r = autocorr(h)
    # Exclude k=0 (which is always the max)
    r_no0 = r.copy()
    r_no0[n-1] = -1e10  # k=0 position
    k_min = np.argmin(r_no0)
    k = k_min - (n - 1)  # actual shift
    
    # Gradient of R(k) w.r.t. h
    grad = np.zeros(n)
    if k == 0:
        grad[:] = 2.0 * h
    elif k > 0:
        grad[k:n] += h[0:n-k]
        grad[0:n-k] += h[k:n]
    else:
        a = -k
        grad[0:n-a] += h[a:n]
        grad[a:n] += h[0:n-a]
    grad *= dx
    
    h_new = h + lr * grad
    return project_feasible(h_new), k, r[k_min]

def smoothed_maxmin_obj(h, tau):
    """Smoothed version of C5 = max_k (1 - R(k)) = 1 - min_k R(k)."""
    n = len(h)
    dx = 2.0 / n
    r = autocorr(h)
    r_no0 = r.copy()
    r_no0[n-1] = -1e10  # exclude k=0

    shifted = -r_no0 / tau
    shifted = np.clip(shifted, -500, 500)
    log_sum = np.log(np.sum(np.exp(shifted)))
    soft_min = -tau * log_sum
    c5 = 1.0 - soft_min

import numpy as np
from scipy.optimize import minimize
import time

def exact_objective(h):
    n = len(h)
    if n == 0:
        return float('inf')
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(profile))

def project_feasible(h, target_sum=None):
    n = len(h)
    if target_sum is None:
        target_sum = n / 2.0
    h = np.array(h, dtype=np.float64)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = np.sum(h)
        diff = s - target_sum
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h > 0
            if not np.any(mask):
                break
            step = min(diff / np.sum(mask), np.min(h[mask]))
            h[mask] -= step
        else:
            mask = h < 1.0
            if not np.any(mask):
                break
            step = min(-diff / np.sum(mask), 1.0 - np.max(h[mask]))
            h[mask] += step
    h = np.clip(h, 0.0, 1.0)
    diff = np.sum(h) - target_sum
    if abs(diff) > 1e-14:
        mask = (h > 0) if diff > 0 else (h < 1.0)
        if np.any(mask):
            h[mask] -= diff / np.sum(mask)
            h = np.clip(h, 0.0, 1.0)
    return h

def autocorr_full(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, h, mode='full') * dx

def subgrad_direction(h, top_k=1):
    """Compute gradient direction to increase min R(k), focusing on worst shifts."""
    n = len(h)
    dx = 2.0 / n
    r = autocorr_full(h)
    r_work = r.copy()
    r_work[n - 1] = -1e10
    # Get the worst 'top_k' shifts
    k_indices = np.argsort(r_work)[:top_k]
    grad = np.zeros(n)
    for ki in k_indices:
        k = ki - (n - 1)
        if k == 0:
            grad[:] += 2.0 * h
        elif k > 0:
            grad[k:n] += h[0:n-k]
            grad[0:n-k] += h[k:n]
        else:
            a = -k
            grad[0:n-a] += h[a:n]
            grad[a:n] += h[0:n-a]
    grad *= dx
    r_min = float(np.min(r_work))
    return grad, r_min

def nesterov_subgrad(h0, n_steps, lr0, momentum=0.9, top_k=1):
    """Nesterov-accelerated subgradient method to maximize min R(k)."""
    n = len(h0)
    h = h0.copy()
    h_prev = h0.copy()
    velocity = np.zeros(n)
    best_h = h0.copy()
    best_c = exact_objective(h0)
    
    for step in range(n_steps):
        lr = lr0 / (1.0 + step * 0.02)
        y = h + momentum * (h - h_prev)
        y = np.clip(y, 0.0, 1.0)
        grad, r_min = subgrad_direction(y, top_k)
        h_new = y + momentum * (h - y) + lr * grad
        h_new = project_feasible(h_new)
        h_prev = h
        h = h_new
        c = exact_objective(h)
        if c < best_c - 1e-13:
            best_c = c
            best_h = h.copy()
    return best_h, best_c

def binary_sa(b, max_iters, rng_seed, temp_start, temp_end):

def slsqp_polish(h0, maxiter=50):

        r_w = r.copy()
        r_w[n-1] = -1e10
        r_min = np.min(r_w)
        shifted = -r_w / tau
        shifted = np.clip(shifted, -500, 500)
        log_sum = np.log(np.sum(np.exp(shifted)))
        val = 1.0 + tau * log_sum
        sv = np.sum(x) - n / 2.0
        return val + 1e5 * sv * sv
    
    def grad(x):
        r = np.correlate(x, x, mode='full') * dx
        r_w = r.copy()
        r_w[n-1] = -1e10
        shifted = -r_w / tau
        shifted = np.clip(shifted, -500, 500)
        exp_s = np.exp(shifted)
        weights = exp_s / np.sum(exp_s)
        r_min = np.min(r_w)
        val = 1.0 + tau * np.log(np.sum(exp_s))
        grad_r = np.zeros(n)
        for i in range(2 * n - 1):
            k = i - (n - 1)
            w = weights[i]
            if w < 1e-14 or i == n - 1:
                continue
            if k == 0:
                grad_r[:] += 2.0 * w * x
            elif k > 0:
                grad_r[k:n] += w * x[0:n-k]
                grad_r[0:n-k] += w * x[k:n]
            else:
                a = -k
                grad_r[0:n-a] += w * x[a:n]
                grad_r[a:n] += w * x[0:n-a]
        grad_r *= -tau * dx
        sv = np.sum(x) - n / 2.0
        grad_r += 2e5 * sv * np.ones(n)
        return grad_r
    
    bounds = [(0.0, 1.0)] * n

            if c_cur < best_c - 1e-12:
                best_c = c_cur
                best_h = h_cur.copy()
                best_n = n
        
        # Update parent if we improved
        if best_h is not None and best_c < pc:
            parent = best_h.copy()
            pc = best_c
    
    # Final polish
    if time.time() < deadline - 20:
        h_final = slsqp_polish(best_h, maxiter=30)
        if h_final is not None:
            c_final = exact_objective(h_final)
            if c_final < best_c:
                best_c = c_final
                best_h = h_final
    
    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_c = exact_objective(best_h)
    return (best_h, best_c, len(best_h))
