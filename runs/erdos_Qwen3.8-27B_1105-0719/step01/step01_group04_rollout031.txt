```python
def repair_feasible(vec):
    h = np.asarray(vec, dtype=np.float64)
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-12:
        return h.copy()
    if diff > 0:
        order = np.argsort(h)
        rem = diff
        for i in order:
            if rem <= 1e-15:
                break
            add = min(rem, 1.0 - h[i])
            h[i] += add
            rem -= add
    else:
        order = np.argsort(-h)
        rem = -diff
        for i in order:
            if rem <= 1e-15:
                break
            remv = min(rem, h[i])
            h[i] -= remv
            rem -= remv
    # Final fixup for roundoff
    diff2 = target - h.sum()
    if abs(diff2) > 1e-14:
        # Distribute evenly among non-bound entries
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.any():
            h[mask] += diff2 / mask.sum()
        else:
            h[0] += diff2
    return h

def compute_correlation_values(h):
    """Compute C_r * dx for all lags r."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx

def compute_grad_matrix(h):
    """Compute (2n-1) x n gradient matrix where row r is grad of C_r*dx wrt h."""
    n = len(h)
    dx = 2.0 / n
    n_lags = 2 * n - 1
    grads = np.zeros((n_lags, n))
    
    for delta in range(n):
        if delta == 0:
            grads[n-1, :] = dx * (1.0 - 2.0 * h)
        else:
            # Positive lag delta: C = dx * sum_{i=0}^{n-1-delta} h[i]*(1-h[i+delta])
            # For h[i] where i < n-delta: partial = dx*(1-h[i+delta])
            # For h[i] where i >= delta: partial += -dx*h[i-delta]
            idx_pos = np.arange(n - delta)
            grads[n-1+delta, idx_pos] += dx * (1.0 - h[idx_pos + delta])
            idx_neg = np.arange(delta, n)
            grads[n-1+delta, idx_neg] += -dx * h[idx_neg - delta]
            
            # Negative lag -delta: C = dx * sum_{i=delta}^{n-1} h[i]*(1-h[i-delta])
            idx_pos2 = np.arange(delta, n)
            grads[n-1-delta, idx_pos2] += dx * (1.0 - h[idx_pos2 - delta])
            idx_neg2 = np.arange(0, n - delta)
            grads[n-1-delta, idx_neg2] += -dx * h[idx_neg2 + delta]
    
    return grads
```

```python
def cdf_transfer(h, new_len):
    n = len(h)
    dx_old = 2.0 / n
    dx_new = 2.0 / new_len
    
    # Compute cumulative sum
    cum = np.zeros(n + 1)
    cum[1:] = np.cumsum(h)
    
    h_new = np.zeros(new_len)
    for j in range(new_len):
        x_start = j * dx_new
        x_end = (j + 1) * dx_new
        
        # Find which old cells overlap
        i_start = int(np.floor(x_start / dx_old))
        i_end = int(np.ceil(x_end / dx_old))
        i_start = max(0, min(n - 1, i_start))
        i_end = max(0, min(n - 1, i_end))
        
        mass = 0.0
        for i in range(i_start, i_end + 1):
            cell_start = i * dx_old
            cell_end = (i + 1) * dx_old
            overlap_start = max(x_start, cell_start)
            overlap_end = min(x_end, cell_end)
            if overlap_end > overlap_start:
                mass += h[i] * (overlap_end - overlap_start)
        
        h_new[j] = mass / dx_new
    
    return np.clip(h_new, 0.0, 1.0)
```

```python
def cdf_transfer(h, new_len):
    n = len(h)
    dx_old = 2.0 / n
    
    # CDF at old cell boundaries
    boundaries = np.linspace(0, 2, n + 1)
    cdf = np.zeros(n + 1)
    cdf[1:] = np.cumsum(h) * dx_old
    
    # New cell centers (actually boundaries)
    new_boundaries = np.linspace(0, 2, new_len + 1)
    
    # Evaluate CDF at new boundaries via interpolation
    cdf_new = np.interp(new_boundaries, boundaries, cdf)
    
    # h_new[j] = (cdf_new[j+1] - cdf_new[j]) / dx_new
    dx_new = 2.0 / new_len
    h_new = (cdf_new[1:] - cdf_new[:-1]) / dx_new
    
    return np.clip(h_new, 0.0, 1.0)
```

This is much cleaner and O(n + m). The interpolation handles the partial cell overlaps automatically.
