  - If 0 <= j <= n-1-k: contribution from (1-h[i]) term: dx * d(h[j+k]*(1-h[j]))/dh[j] = dx * (-h[j+k])
  
  Combined:
  dV_k/dh[j] = dx * [I(k<=j<=n-1)*(1-h[j-k]) + I(0<=j<=n-1-k)*(-h[j+k])]
  
  For j in [0, k-1]: only second term, dV = -dx*h[j+k]
  For j in [k, n-1-k]: both terms, dV = dx*(1-h[j-k]) - dx*h[j+k]
  For j in [n-k, n-1]: only first term, dV = dx*(1-h[j-k])
  
  (assuming k <= n-1-k, i.e., k <= n/2; otherwise the middle range is empty)

For values[n-1-k] (k>0): W_k = dx * sum_{i=0}^{n-1-k} h[i]*(1-h[i+k])
  For fixed j:
  - h[i] = h[j] when i=j, valid when 0<=j<=n-1-k: contribution dx*(1-h[j+k])
  - (1-h[i+k]) involves h[j] when i+k=j, i.e., i=j-k, valid when 0<=j-k<=n-1-k, i.e., k<=j<=n-1: contribution dx*(-h[j-k])
  
  dW_k/dh[j] = dx * [I(0<=j<=n-1-k)*(1-h[j+k]) + I(k<=j<=n-1)*(-h[j-k])]

def compute_profile_and_grads(h):
    n = len(h)
    dx = 2.0 / n
    n_lags = 2 * n - 1
    values = np.zeros(n_lags)
    grads = np.zeros((n_lags, n))
    
    for k in range(n):
        # Positive lag k: values[n-1+k]
        # V_k = dx * sum_{i=0}^{n-1-k} h[i+k]*(1-h[i])
        if k == 0:
            values[n-1] = dx * np.sum(h * (1.0 - h))
            grads[n-1, :] = dx * (1.0 - 2.0 * h)
        else:
            idx = np.arange(n - k)
            values[n-1+k] = dx * np.sum(h[idx+k] * (1.0 - h[idx]))
            # Gradient for positive lag
            # For j in [0, k-1]: -dx * h[j+k]
            # For j in [k, n-1-k]: dx*(1-h[j-k]) - dx*h[j+k]
            # For j in [n-k, n-1]: dx*(1-h[j-k])
            # Simplified: dx*(1-h[j-k]) for j in [k, n-1]
            #           - dx*h[j+k] for j in [0, n-1-k]
            j_range1 = np.arange(k, n)  # j where j-k is in [0, n-1-k]
            grads[n-1+k, j_range1] += dx * (1.0 - h[j_range1 - k])
            j_range2 = np.arange(0, n-k)  # j where j+k is in [k, n-1]
            grads[n-1+k, j_range2] += -dx * h[j_range2 + k]
            
            # Negative lag k: values[n-1-k]
            # W_k = dx * sum_{i=0}^{n-1-k} h[i]*(1-h[i+k])
            values[n-1-k] = dx * np.sum(h[idx] * (1.0 - h[idx+k]))
            # Gradient
            # For j in [0, n-1-k]: dx*(1-h[j+k])
            # For j in [k, n-1]: -dx*h[j-k]
            j_range3 = np.arange(0, n-k)
            grads[n-1-k, j_range3] += dx * (1.0 - h[j_range3 + k])
            j_range4 = np.arange(k, n)
            grads[n-1-k, j_range4] += -dx * h[j_range4 - k]
    
    return values, grads

import numpy as np
from scipy.optimize import minimize
import time


def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr * dx))


def repair_feasible(vec):
    h = np.asarray(vec, dtype=np.float64).copy()
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(h)
        rem = diff
        for i in order:
            if rem <= 1e-15:
                break
            add = min(rem, 1.0 - h[i])
            h[i] += add
            rem -= add
    else:
        order = np.argsort(-h)
        rem = -diff
        for i in order:
            if rem <= 1e-15:
                break
            remv = min(rem, h[i])
            h[i] -= remv
            rem -= remv
    # Final float fix
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        for i in range(n):
            if diff2 > 0 and h[i] < 1.0:
                h[i] += diff2
                break
            elif diff2 < 0 and h[i] > 0.0:
                h[i] += diff2
                break
    return h


def compute_profile_and_grads(h):
    n = len(h)
    dx = 2.0 / n
    n_lags = 2 * n - 1
    values = np.zeros(n_lags)
    grads = np.zeros((n_lags, n))
    
    for k in range(n):
        if k == 0:
            values[n - 1] = dx * np.sum(h * (1.0 - h))
            grads[n - 1, :] = dx * (1.0 - 2.0 * h)
        else:
            idx = np.arange(n - k)
            # Positive lag
            values[n - 1 + k] = dx * np.sum(h[idx + k] * (1.0 - h[idx]))
            j1 = np.arange(k, n)
            grads[n - 1 + k, j1] += dx * (1.0 - h[j1 - k])
            j2 = np.arange(0, n - k)
            grads[n - 1 + k, j2] += -dx * h[j2 + k]
            # Negative lag
            values[n - 1 - k] = dx * np.sum(h[idx] * (1.0 - h[idx + k]))
            j3 = np.arange(0, n - k)
            grads[n - 1 - k, j3] += dx * (1.0 - h[j3 + k])
            j4 = np.arange(k, n)
            grads[n - 1 - k, j4] += -dx * h[j4 - k]
    
    return values, grads


def soft_max_obj_and_grad(h, beta):
    values, grads = compute_profile_and_grads(h)
    # Stable log-sum-exp
    v_max = np.max(values)
    shifted = beta * (values - v_max)
    exp_vals = np.exp(shifted)
    log_sum = np.log(np.sum(exp_vals))
    J = (v_max + log_sum / beta)
    weights = exp_vals / np.sum(exp_vals)
    grad_J = weights @ grads
    return J, grad_J


def solve_soft_max(h_start, beta, maxiter=200):
    n = len(h_start)
    h_start = repair_feasible(h_start)
    
    def obj(x):
        val, _ = soft_max_obj_and_grad(x, beta)
        return val
    
    def jac(x):
        _, g = soft_max_obj_and_grad(x, beta)
        return g
    
    def eq_fun(x):
        return np.sum(x) - n / 2.0
    
    def eq_jac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    
    try:
        result = minimize(
            obj, h_start, jac=jac,
            method='SLSQP', bounds=bounds, constraints=constraints,
            options={'maxiter': maxiter, 'ftol': 1e-16}
        )
    except Exception:
        return None, None
    
    if result.x is not None and np.all(np.isfinite(result.x)):
        h_sol = repair_feasible(result.x)
        score = exact_score(h_sol)
        if np.isfinite(score):
            return h_sol, score
    return None, None


def active_descent(h, max_steps=50, sigma=0.003):
    n = len(h)
    h = repair_feasible(h)
    best = h.copy()
    best_score = exact_score(best)
    
    for step in range(max_steps):
        values, grads = compute_profile_and_grads(best)
        c = np.max(values)
        gaps = c - values
        # Soft active set weights
        w = np.exp(-gaps / sigma)
        w_sum = np.sum(w)
        if w_sum < 1e-15:
            break
        w = w / w_sum
        # Direction: negative weighted gradient, projected to zero-mean (preserve sum)
        direction = -(w @ grads)
        direction -= np.mean(direction)
        
        improved = False
        for alpha in [0.05, 0.03, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]:
            h_trial = repair_feasible(best + alpha * direction)
            score = exact_score(h_trial)
            if score < best_score - 1e-14:
                best = h_trial.copy()
                best_score = score
                improved = True
                break
        
        if not improved:
            # Try with smaller sigma for next iteration
            sigma *= 0.7
            if sigma < 1e-6:
                break
    
    return best, best_score


def cdf_transfer(h, new_n):
    """Transfer h to new dimension using CDF interpolation."""
    n = len(h)
    dx = 2.0 / n
    # Compute CDF at cell boundaries
    cdf = np.cumsum(h)
    x_src = np.linspace(0, 2.0, n + 1)
    cdf_src = np.concatenate([[0.0], cdf]) * dx
    
    # New grid
    x_tgt = np.linspace(0, 2.0, new_n + 1)
    cdf_tgt = np.interp(x_tgt, x_src, cdf_src)
    
    # Invert: h_new[i] = (cdf_tgt[i+1] - cdf_tgt[i]) / (2.0/new_n)
    dx_new = 2.0 / new_n
    h_new = np.diff(cdf_tgt) / dx_new
    return np.clip(h_new, 0.0, 1.0)


def symmetrize(h):
    return (h + h[::-1]) / 2.0

def soft_max_grad_direct(h, beta):

    dx = 2.0 / n
    values, _ = compute_profile_and_grads(h)  # Still need values
    # Actually, let me compute values separately for efficiency
    
    # Compute values
    vals = np.zeros(2*n-1)
    for k in range(n):
        if k == 0:
            vals[n-1] = dx * np.sum(h*(1-h))
        else:
            idx = np.arange(n-k)
            vals[n-1+k] = dx * np.sum(h[idx+k]*(1-h[idx]))
            vals[n-1-k] = dx * np.sum(h[idx]*(1-h[idx+k]))
    
    # Soft-max weights
    v_max = np.max(vals)
    shifted = beta * (vals - v_max)
    exp_vals = np.exp(shifted)
    weights = exp_vals / np.sum(exp_vals)
    
    # Compute gradient directly
    grad = np.zeros(n)
    for j in range(n):
        # From positive lags
        for k in range(1, n):
            if k <= j < n:
                grad[j] += weights[n-1+k] * dx * (1.0 - h[j-k])
            if j < n-k:

            if k <= j < n:

```python
def active_descent(h, max_steps=50, sigma=0.003):
    n = len(h)
    h = repair_feasible(h)
    best = h.copy()
    best_score = exact_score(best)
    
    for step in range(max_steps):
        values, grads = compute_profile_and_grads(best)
        c = np.max(values)
        gaps = c - values
        w = np.exp(-gaps / sigma)
        w_sum = np.sum(w)
        if w_sum < 1e-15:
            break
        w = w / w_sum
        direction = -(w @ grads)
        direction -= np.mean(direction)
        dir_norm = np.linalg.norm(direction)
        if dir_norm < 1e-15:
            break
        
        improved = False
        alpha = 0.1  # Initial step size
        for _ in range(20):
            h_trial = repair_feasible(best + alpha * direction)
            score = exact_score(h_trial)
            if score < best_score - 1e-14:
                best = h_trial.copy()
                best_score = score
                improved = True
                break
            alpha *= 0.5
            if alpha < 1e-8:
                break
        
        if not improved:
            sigma *= 0.7
            if sigma < 1e-6:
                break
    
    return best, best_score
```

This backtracking approach is more robust than the fixed step sizes.
