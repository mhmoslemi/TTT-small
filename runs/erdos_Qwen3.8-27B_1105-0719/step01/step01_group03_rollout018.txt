import numpy as np
from scipy.optimize import minimize
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))

def profile_of(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def repair_feasible(vec):
    n = len(vec)
    target = n / 2.0
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        # Need to increase sum by diff
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for idx in order:
            if remaining <= 1e-14:
                break
            add = min(remaining, capacity[idx])
            h[idx] += add
            remaining -= add
    else:
        # Need to decrease sum by -diff
        order = np.argsort(h)
        remaining = -diff
        for idx in order:
            if remaining <= 1e-14:
                break
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
    # Final adjustment for numerical precision
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        # Find a coordinate that can absorb the tiny adjustment
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_value_and_grad(h, tau):
    """Compute F_tau(h) = tau * log(sum exp(p_m/tau)) and its gradient."""
    n = len(h)
    dx = 2.0 / n
    p = profile_of(h)
    M = len(p)
    # Numerically stable log-sum-exp
    mu = np.max(p)
    exp_terms = np.exp((p - mu) / tau)
    s = np.sum(exp_terms)
    val = mu + tau * np.log(s)
    
    # Weights
    w = exp_terms / s
    
    # Gradient: sum_m w_m * grad(p_m)
    grad = np.zeros(n)
    for m in range(M):
        d = m - (n - 1)
        wm = w[m]
        if wm < 1e-12:
            continue
        if d == 0:
            grad += wm * dx * (1.0 - 2.0 * h)
        elif d > 0:
            grad[d:] += wm * dx * (1.0 - h[:n-d])
            grad[:n-d] -= wm * dx * h[d:]
        else:
            ds = -d
            grad[:n-ds] += wm * dx * (1.0 - h[ds:])
            grad[ds:] -= wm * dx * h[:n-ds]
    
    return val, grad

def fw_vertex(g, n):
    """Find v = argmin_{v in P} g^T v where P = {v: sum v = n/2, 0<=v<=1}."""
    target = n / 2.0
    order = np.argsort(g)
    v = np.zeros(n)
    remaining = target
    for idx in order:
        if remaining >= 1.0 - 1e-12:
            v[idx] = 1.0
            remaining -= 1.0
        elif remaining > 1e-12:
            v[idx] = remaining
            remaining = 0.0
        else:
            break
    return v

def fw_step(h, tau, alpha_candidates=None):
    """One Frank-Wolfe step on F_tau."""
    if alpha_candidates is None:
        alpha_candidates = [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]
    
    n = len(h)
    val_curr, grad = smooth_value_and_grad(h, tau)
    v = fw_vertex(grad, n)
    
    # Check if vertex is same as current
    if np.allclose(v, h, atol=1e-10):
        return h, val_curr
    
    # Line search: try different alphas
    best_h = h
    best_val = val_curr
    for alpha in alpha_candidates:
        h_new = (1 - alpha) * h + alpha * v
        val_new, _ = smooth_value_and_grad(h_new, tau)
        if val_new < best_val - 1e-14:
            best_val = val_new
            best_h = h_new
    return best_h, best_val

def fw_anneal(h_start, tau_start, tau_end, max_iters, deadline):
    """Run Frank-Wolfe with annealing from tau_start to tau_end."""
    h = h_start.copy()
    current_score = exact_score(h)
    
    n_iters = max_iters
    for it in range(n_iters):
        if time.perf_counter() >= deadline:
            break
        # Anneal tau
        frac = it / max(1, n_iters - 1)
        tau = tau_start * (tau_end / tau_start) ** frac
        
        # Do one FW step
        h, _ = fw_step(h, tau)
        
        # Check exact score
        s = exact_score(h)
        if s < current_score - 1e-12:
            current_score = s
    
    return h, current_score

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

def build_gradient_row(h, m, n):
    """Build gradient of profile[m] with respect to h."""
    dx = 2.0 / n
    d = m - (n - 1)
    g = np.zeros(n)
    if d == 0:
        g = dx * (1.0 - 2.0 * h)
    elif d > 0:
        g[d:] = dx * (1.0 - h[:n-d])
        g[:n-d] = -dx * h[d:]
    else:
        ds = -d
        g[:n-ds] = dx * (1.0 - h[ds:])
        g[ds:] = -dx * h[:n-ds]
    return g

        return x[-1]

        g = np.zeros(self.n + 1)
        g[-1] = 1.0
        return g

        return x[:-1].sum() - self.n / 2.0
