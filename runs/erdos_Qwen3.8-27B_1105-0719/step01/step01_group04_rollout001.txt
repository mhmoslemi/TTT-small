def compute_lag_data(h):
    """Compute all lag values and gradients."""
    n = len(h)
    dx = 2.0 / n
    n_lags = 2 * n - 1
    values = np.zeros(n_lags)
    grads = np.zeros((n_lags, n))
    
    for delta in range(n):
        if delta == 0:
            values[n-1] = dx * np.sum(h * (1.0 - h))
            grads[n-1, :] = dx * (1.0 - 2.0 * h)
        else:
            idx = np.arange(n - delta)
            h_a = h[idx]
            h_b = h[idx + delta]
            vals = h_a * (1.0 - h_b)
            values[n-1+delta] = dx * np.sum(vals)
            grads[n-1+delta, idx] += dx * (1.0 - h_b)
            grads[n-1+delta, idx+delta] += -dx * h_a
            
            idx2 = np.arange(delta, n)

            vals2 = h_a2 * (1.0 - h_b2)

            grads[n-1-delta, idx2] += dx * (1.0 - h_b2)
            grads[n-1-delta, idx2-delta] += -dx * h_a2

def smooth_max_descent(h0, sigma_init=5.0, sigma_final=200.0, sigma_growth=1.5, 
                       max_iter=200, tol=1e-12, time_budget=60.0):
    """Optimize using smooth max with increasing sigma."""
    t_start = time.perf_counter()
    h = project_feasible(h0.copy())
    best_h = h.copy()
    best_score = exact_score(h)
    sigma = sigma_init
    
    for it in range(max_iter):
        if time.perf_counter() - t_start > time_budget:
            break
        
        # Ensure sigma isn't too large relative to current max
        values = compute_lag_data(h)[0]
        c_max = np.max(values)
        if sigma > 0:
            # Cap sigma to avoid overflow
            sigma_max_allowed = 50.0 / max(c_max, 1e-10)
            sigma = min(sigma, sigma_max_allowed)
        
        # Compute smooth max and gradient
        values, grads = compute_lag_data(h)
        max_v = np.max(values)
        shifted = sigma * (values - max_v)
        # Clip to avoid overflow
        shifted = np.clip(shifted, -500, 500)
        exp_shifted = np.exp(shifted)
        weights = exp_shifted / np.sum(exp_shifted)
        
        # Gradient weighted by softmax
        grad = np.sum(weights[:, None] * grads, axis=0)
        
        # Line search
        best_alpha = 0
        best_f = best_score
        for alpha in [0.05, 0.04, 0.03, 0.02, 0.01, 0.005, 0.002, 0.001]:
            h_trial = project_feasible(h - alpha * grad)
            score = exact_score(h_trial)
            if score < best_f:
                best_alpha = alpha
                best_f = score
        
        if best_alpha == 0:
            # Increase sigma and try again
            sigma *= sigma_growth
            if sigma > sigma_final:
                break
            continue
        
        h = project_feasible(h - best_alpha * grad)
        score = exact_score(h)
        
        if score < best_score - 1e-13:
            best_h = h.copy()
            best_score = score
        
        # Check convergence
        if abs(grad).max() < tol:
            break
        
        # Gradually increase sigma
        if it % 20 == 19:
            sigma = min(sigma * sigma_growth, sigma_final)
    
    return best_h, best_score

    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr * dx))


def project_feasible(h):
    n = len(h)
    target = n / 2.0
    h = np.asarray(h, dtype=np.float64).copy()
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    lo = -3.0
    hi = 3.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    result = np.clip(h - lam, 0.0, 1.0)
    diff = target - result.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(result)
            for i in order:
                add = min(diff, 1.0 - result[i])
                result[i] += add
                diff -= add
                if diff <= 1e-14:
                    break
        else:
            order = np.argsort(-result)
            for i in order:
                rem = min(-diff, result[i])
                result[i] -= rem
                diff += rem
                if diff >= -1e-14:
                    break
    return result


def interpolate_to_dim(h, new_n):
    n = len(h)
    x_src = (np.arange(n) + 0.5) * 2.0 / n
    x_tgt = (np.arange(new_n) + 0.5) * 2.0 / new_n
    xs_ext = np.concatenate([[0.0], x_src, [2.0]])
    ys_ext = np.concatenate([[h[0]], h, [h[-1]]])
    return project_feasible(np.interp(x_tgt, xs_ext, ys_ext))


def smooth_max_optimize(h0, time_budget=120.0, sigma_start=3.0, sigma_end=300.0,
                        sigma_growth=1.4, stagnant_limit=15):
    t_start = time.perf_counter()
    h = project_feasible(h0.copy())
    best_h = h.copy()
    best_score = exact_score(h)
    sigma = sigma_start
    stagnant = 0

    for it in range(500):
        if time.perf_counter() - t_start > time_budget:
            break
        values, grads = compute_lag_data(h)
        c_max = np.max(values)
        sigma = min(sigma, 40.0 / max(c_max, 1e-10))

        shifted = sigma * (values - c_max)
        shifted = np.clip(shifted, -400.0, 400.0)
        exp_s = np.exp(shifted)
        w = exp_s / np.sum(exp_s)
        grad = np.sum(w[:, None] * grads, axis=0)
        gnorm = np.sqrt(np.sum(grad * grad))
        if gnorm < 1e-14:
            break

        grad = grad / gnorm
        best_alpha = 0.0
        best_f = best_score
        for alpha in [0.08, 0.05, 0.03, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]:
            h_t = project_feasible(h - alpha * grad * gnorm)
            s_t = exact_score(h_t)
            if s_t < best_f - 1e-14:
                best_alpha = alpha
                best_f = s_t

        if best_alpha == 0.0:
            stagnant += 1
            sigma = min(sigma * 1.6, sigma_end)
            if stagnant > stagnant_limit:
                break
            continue

        h = project_feasible(h - best_alpha * grad * gnorm)
        score = exact_score(h)
        if score < best_score - 1e-13:
            best_h = h.copy()
            best_score = score
            stagnant = 0
        else:
            stagnant += 1
            if stagnant > 10:
                break

        if it % 10 == 9:
            sigma = min(sigma * sigma_growth, sigma_end)

    return best_h, best_score


def cvx_trust_region_step(h, mu=0.05, top_k=20, time_limit=30.0):
    if not HAS_CP:
        return None
    t0 = time.perf_counter()
    if time.perf_counter() - t0 > time_limit:
        return None
    n = len(h)
    target = n / 2.0
    values, grads = compute_lag_data(h)
    c_max = np.max(values)
    top_idx = np.argsort(values)[::-1][:top_k]
    top_idx = np.sort(top_idx)
    G = grads[top_idx, :]
    v = values[top_idx]

    h_var = cp.Variable(n)
    t_var = cp.Variable()
    objective = cp.Minimize(t_var + (mu / 2.0) * cp.sum_squares(h_var - h))
    constraints = [t_var - v - G @ (h_var - h) >= 0,
                   sum(h_var) == target,
                   h_var >= 0, h_var <= 1]
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.SCS, max_iters=5000, eps=1e-8)
    except Exception:
        try:
            prob.solve(solver=cp.CLARABEL)
        except Exception:
            return None
    if time.perf_counter() - t0 > time_limit:
        return None
    if h_var.value is None:
        return None
    return np.array(h_var.value).flatten()


def cvx_trust_region_loop(h0, max_iters=80, time_budget=300.0, top_k=25, mu_start=0.1):
    t_start = time.perf_counter()
    h = project_feasible(h0.copy())
    best_h = h.copy()
    best_score = exact_score(h)
    mu = mu_start
    stagnant = 0

    for it in range(max_iters):
        if time.perf_counter() - t_start > time_budget - 10:
            break
        h_qp = cvx_trust_region_step(h, mu=mu, top_k=top_k)
        if h_qp is None:
            stale

            stagnant += 2
            if stagnant > 8:
                break
            continue
        h_qp = project_feasible(h_qp)
        direction = h_qp - h
        best_alpha = 0.0
        best_f = best_score
        for alpha in [1.0, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05, 0.02]:
            h_t = project_feasible(h + alpha * direction)
            s_t = exact_score(h_t)
            if s_t < best_f - 1e-14:
                best_alpha = alpha
                best_f = s_t

        if best_alpha == 0.0:
            stagnant += 1
            mu *= 0.5
            top_k = min(top_k + 5, 2 * len(h) - 1)
            if mu < 0.001 or stagnant > 10:
                break
            continue

        h = project_feasible(h + best_alpha * direction)
        score = exact_score(h)
        if score < best_score - 1e-13:
            best_h = h.copy()
            best_score = score
            stagnant = 0
            mu = min(mu * 1.5, 0.5)
        else:
            stagnant += 1
            mu *= 0.5
            if stagnant > 10:
                break

    return best_h, best_score


def active_descent_refined(h0, max_steps=100, time_budget=60.0):
    t_start = time.perf_counter()
    h = project_feasible(h0.copy())
    best_h = h.copy()
    best_score = exact_score(h)
    n = len(h)

    for it in range(max_steps):
        if time.perf_counter() - t_start > time_budget:
            break
        values, grads = compute_lag_data(h)
        c_val = np.max(values)
        top_k = min(20, 2 * n - 1)
        active = np.argsort(values)[::-1][:top_k]
        gaps = np.maximum(c_val - values[active], 0.0)
        sigma_w = 0.003 * c_val + 1e-10
        weights = np.exp(-gaps / sigma_w)
        weights /= np.sum(weights) + 1e-15
        direction = -np.sum(weights[:, None] * grads[active, :], axis=0)
        direction -= np.mean(direction)
        gnorm = np.sqrt(np.sum(direction * direction))
        if gnorm < 1e-15:
            break

        best_alpha = 0.0
        best_f = best_score
        for alpha in [0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]:
            h_t = project_feasible(h + alpha * direction)
            s_t = exact_score(h_t)
            if s_t < best_f - 1e-14:
                best_alpha = alpha
                best_f = s_t

        if best_alpha == 0.0:
            break

        h = project_feasible(h + best_alpha * direction)
        score = exact_score(h)
        if score < best_score - 1e-13:
            best_h = h.copy()
            best_score = score

    return best_h, best_score


def symmetrize(h):
    return project_feasible((h + h[::-1]) / 2.0)


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()

    def time_left():
        return budget_s - (time.perf_counter() - t_start)

    rng = np.random.RandomState(seed)

    try:
        parent = np.asarray(initial_h_values, dtype=np.float64)
    except Exception:
        parent = None

    if parent is not None and len(parent) > 0:
        best_h = project_feasible(parent.copy())
    else:
        best_h = project_feasible(np.full(140, 0.5))

    best_score = exact_score(best_h)
    best_n = len(best_h)

    h_sym = symmetrize(best_h)
    s_sym = exact_score(h_sym)
    if s_sym < best_score - 1e-13:
        best_h = h_sym
        best_score = s_sym

    # Phase 1: Smooth max descent
    if time_left() > 200:
        h_s, s_s = smooth_max_optimize(best_h, time_budget=min(150, time_left() - 20),
                                       sigma_start=3.0, sigma_end=200.0)
        if s_s < best_score - 1e-13:
            best_h = h_s
            best_score = s_s

    # Phase 2: CVXPY trust region
    if time_left() > 100 and HAS_CP:
        h_c, s_c = cvx_trust_region_loop(best_h, max_iters=60,
                                         time_budget=min(250, time_left() - 30),
                                         top_k=25, mu_start=0.1)
        if s_c < best_score - 1e-13:
            best_h = h_c
            best_score = s_c
            best_n = len(h_c)

    # Phase 3: Active descent
    if time_left() > 60:
        h_a, s_a = active_descent_refined(best_h, max_steps=80,
                                          time_budget=min(80, time_left() - 10))
        if s_a < best_score - 1e-13:
            best_h = h_a
            best_score = s_a

    # Phase 4: Second round - smooth max with higher sigma
    if time_left() > 80:
        h_s2, s_s2 = smooth_max_optimize(best_h, time_budget=min(60, time_left() - 20),
                                         sigma_start=50.0, sigma_end=500.0,
                                         sigma_growth=1.3)
        if s_s2 < best_score - 1e-13:
            best_h = h_s2
            best_score = s_s2

    # Phase 5: CVXPY with tighter parameters
    if time_left() > 60 and HAS_CP:
        h_c2, s_c2 = cvx_trust_region_loop(best_h, max_iters=40,
                                           time_budget=min(80, time_left() - 15),
                                           top_k=15, mu_start=0.03)
        if s_c2 < best_score - 1e-13:
            best_h = h_c2
            best_score = s_c2
            best_n = len(h_c2)

    # Phase 6: Multi-start perturbation
    if time_left() > 50:
        for trial in range(6):
            if time_left() < 25:
                break
            eps = rng.uniform(0.003, 0.015)
            z = rng.uniform(-1.0, 1.0, size=len(best_h))
            z -= z.mean()
            h_p = project_feasible(best_h + eps * z)
            h_try, s_try = active_descent_refined(h_p, max_steps=30,
                                                   time_budget=min(20, time_left() - 10))
            if s_try < best_score - 1e-13:
                best_h = h_try
                best_score = s_try
                best_n = len(h_try)

    # Phase 7: Dimension exploration
    if time_left() > 60:
        n_cur = len(best_h)
        dims_to_try = []
        for dn in [10, 20, 40]:
            for sign in [1, -1]:
                nd = n_cur + sign * dn
                if 60 <= nd <= 200 and nd != n_cur:
                    dims_to_try.append(nd)
        for nd in dims_to_try:
            if time_left() < 30:
                break
            h_int = interpolate_to_dim(best_h, nd)
            s_int = exact_score(h_int)
            if s_int < best_score + 0.01:
                h_d, s_d = active_descent_refined(h_int, max_steps=50,
                                                   time_budget=min(30, time_left() - 10))
                if s_d < best_score - 1e-13:
                    best_h = h_d
                    best_score = s_d
                    best_n = len(h_d)
                if HAS_CP and time_left() > 40:
                    h_c3, s_c3 = cvx_trust_region_loop(h_d, max_iters=20,
                                                       time_budget=min(40, time_left() - 10),
                                                       top_k=15, mu_start=0.05)
                    if s_c3 < best_score - 1e-13:
                        best_h = h_c3
                        best_score = s_c3
                        best_n = len(h_c3)

    # Phase 8: Final polish - alternating active descent and trust region
    if time_left() > 30:
        h_f, s_f = active_descent_refined(best_h, max_steps=40,
                                          time_budget=min(20, time_left() - 5))
        if s_f < best_score - 1e-13:
            best_h = h_f
            best_score = s_f

    if time_left() > 15 and HAS_CP:
        h_c4, s_c4 = cvx_trust_region_loop(best_h, max_iters=15,
                                           time_budget=min(20, time_left() - 5),
                                           top_k=12, mu_start=0.02)
        if s_c4 < best_score - 1e-13:
            best_h = h_c4
            best_score = s_c4
            best_n = len(h_c4)

    # Final validation
    best_h = project_feasible(best_h)
    best_score = exact_score(best_h)
    return best_h, best_score, len(best_h)
