def slp_step(h, R, rho, n_active=0):
    """One SLP step: solve trust-region QP subproblem.
    Returns new candidate h or None if solver fails.
    """
    n = len(h)
    corr = compute_correlations(h)
    grad = compute_gradients(h, corr)
    dx = 2.0 / n
    
    # Select active lags
    if n_active > 0 and n_active < 2 * n - 1:
        # Top n_active lags
        active_idx = np.argsort(corr)[::-1][:n_active]
    else:
        active_idx = np.arange(2 * n - 1)
    
    F_active = corr[active_idx]
    G_active = grad[active_idx]  # shape (len(active_idx), n)

    u = cp.Variable(n)

    constraints = [
        F_active + G_active @ u <= gamma,
        -h <= u,
        u <= 1 - h,
        cp.sum(u) == 0,

                    return h_new
                break

import numpy as np

def score_exact(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_feasible(y, n):
    target = n / 2.0
    lo = np.min(y) - 1.0
    hi = np.max(y) + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    h = np.clip(y - t, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        eligible = np.where((h > 1e-8) & (h < 1.0 - 1e-8))[0]
        if len(eligible) > 0:
            h[eligible] += residual / len(eligible)
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_all_gradients(h):

def slp_step(h, R, rho, n_active=None):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    grad = compute_all_gradients(h)
    dx = 2.0 / n
    total_lags = 2 * n - 1
    
    if n_active is not None and n_active < total_lags:
        active_idx = np.argsort(corr)[::-1][:n_active]
    else:
        active_idx = np.arange(total_lags)
    
    F = corr[active_idx]
    G = grad[active_idx]
    
    u = cp.Variable(n)
    gamma = cp.Variable()
    constraints = [
        F + G @ u <= gamma,
        -h <= u,
        u <= 1.0 - h,
        cp.sum(u) == 0,
        cp.norm(u, 2) <= R
    ]
    objective = cp.Minimize(gamma + 0.5 * rho * cp.sum_squares(u))
    prob = cp.Problem(objective, constraints)
    
    for solver in [cp.OSQP, cp.ECOS, cp.SCS, cp.CLARABEL]:
        try:
            kw = {'verbose': False}
            if solver == cp.OSQP:
                kw['eps_abs'] = 1e-8
                kw['eps_rel'] = 1e-8
                kw['max_iter'] = 5000
            prob.solve(solver=solver, **kw)
            if prob.status in ["optimal", "optimal_inaccurate"]:
                u_val = u.value
                if u_val is not None and np.all(np.isfinite(u_val)):
                    h_new = project_feasible(h + u_val, n)
                    return h_new
                break
            elif prob.status == "infeasible":
                break
        except Exception:
            continue
    return None

def resample_h(h, n_new):

    x_old = (np.arange(n) + 0.5) * (2.0 / n)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)

def slp_polish(h, max_time=60, rho=30.0, R_init=0.05, n_active=None):
    h = h.copy()
    best_c5 = score_exact(h)
    R = R_init
    fails = 0
    t_start = time.time()
    
    while time.time() - t_start < max_time and fails < 40:
        h_new = slp_step(h, R, rho, n_active=n_active)
        if h_new is not None:
            c5_new = score_exact(h_new)
            if c5_new < best_c5 - 1e-13:
                h = h_new
                best_c5 = c5_new
                fails = 0
                R = min(R * 1.2, 0.4)
            else:
                fails += 1
                R = max(R * 0.6, 0.005)
        else:
            fails += 1
            R = max(R * 0.5, 0.003)
        if R < 0.005:
            break
    
    return h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()
    
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float)
    else:
        h_init = np.full(60, 0.5)
    
    n = len(h_init)
    best_h = h_init.copy()
    best_c5 = score_exact(best_h)
    best_backup_h = best_h.copy()
    best_backup_c5 = best_c5
    
    deadline = t0 + budget_s - 8
    rem = lambda: deadline - time.time()
    
    # Phase 1: SLP polish from initial
    if rem() > 300:
        h_opt, c5_opt = slp_polish(best_h, max_time=min(120, rem() - 200), rho=20.0, R_init=0.06, n_active=15)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
    
    # Phase 2: Targeted local search
    if rem() > 150:
        n_iters = min(40000, int(rem() * 100))
        h_opt, c5_opt = fast_targeted_local_search(best_h, n_iters, rng, n_top=5)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
    
    # Phase 3: SLP again with tighter params
    if rem() > 200:
        h_opt, c5_opt = slp_polish(best_h, max_time=min(80, rem() - 150), rho=50.0, R_init=0.03, n_active=10)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
    
    # Phase 4: Simulated annealing
    if rem() > 150:
        n_iters = min(60000, int(rem() * 80))
        h_opt, c5_opt = sa_local_search(best_h, n_iters, rng, T0=0.008, cool=0.998)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
    
    # Phase 5: SLP after SA
    if rem() > 100:
        h_opt, c5_opt = slp_polish(best_h, max_time=min(60, rem() - 80), rho=30.0, R_init=0.04, n_active=12)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
    
    # Phase 6: Random local search
    if rem() > 80:
        n_iters = min(50000, int(rem() * 150))
        h_opt, c5_opt = random_local_search(best_h, n_iters, rng)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
    
    # Phase 7: Resolution exploration
    if rem() > 60:
        for n_try in [30, 36, 40, 45, 48, 50, 54, 72, 80, 90, 100, 120, 150]:
            if rem() < 50:
                break
            h_try = resample_h(best_h, n_try)
            c5_try = score_exact(h_try)
            if c5_try < best_c5 - 1e-7:
                if rem() > 40:
                    h_opt, c5_opt = slp_polish(h_try, max_time=min(40, rem() - 35), rho=25.0, R_init=0.04, n_active=12)
                    if c5_opt < c5_try:
                        h_try = h_opt
                        c5_try = c5_opt
                if c5_try < best_c5:
                    best_c5 = c5_try
                    best_h = h_try
    
    # Phase 8: Re-explore resolutions from the new best
    if rem() > 50:
        for n_try in [30, 36, 45, 50, 54, 72, 80, 90, 100, 120, 150, 180]:
            if n_try == len(best_h) or rem() < 35:
                break
            h_try = resample_h(best_h, n_try)
            c5_try = score_exact(h_try)
            if c5_try < best_c5 - 1e-7:
                if rem() > 30:
                    h_opt, c5_opt = slp_polish(h_try, max_time=min(25, rem() - 25), rho=40.0, R_init=0.03, n_active=10)
                    if c5_opt < c5_try:
                        h_try = h_opt
                        c5_try = c5_opt
                if c5_try < best_c5:
                    best_c5 = c5_try
                    best_h = h_try
    
    # Phase 9: Final targeted local search
    if rem() > 20:
        n_iters = min(30000, int(rem() * 200))
        h_opt, c5_opt = fast_targeted_local_search(best_h, n_iters, rng, n_top=3)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
    
    # Phase 10: Final SLP
    if rem() > 15:
        h_opt, c5_opt = slp_polish(best_h, max_time=min(12, rem() - 5), rho=80.0, R_init=0.02, n_active=5)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
    
    # Final verification
    best_h = project_feasible(best_h, len(best_h))
    final_c5 = score_exact(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    # Safety: ensure we have a valid answer
    if not np.all(np.isfinite(best_h)) or np.max(best_h) > 1.0 + 1e-8 or np.min(best_h) < -1e-8:
        best_h = best_backup_h
        best_c5 = best_backup_c5
    
    return best_h, best_c5, len(best_h)
