def exact_score(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")

def repair_feasible(vec):
    h = np.asarray(vec, dtype=np.float64)
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-12:
        return h.copy()
    if diff > 0:
        order = np.argsort(h)
        rem = diff
        for i in order:
            if rem <= 1e-15:
                break
            add = min(rem, 1.0 - h[i])
            if add > 0:
                h[i] += add
                rem -= add
    else:
        order = np.argsort(-h)
        rem = -diff
        for i in order:
            if rem <= 1e-15:
                break
            remv = min(rem, h[i])
            if remv > 0:
                h[i] -= remv
                rem -= remv
    # Final precision fix
    diff2 = target - h.sum()
    if abs(diff2) > 0:
        if diff2 > 0:
            for i in range(n):
                if h[i] < 1.0:
                    h[i] = min(1.0, h[i] + diff2)
                    break
        else:
            for i in range(n):
                if h[i] > 0.0:
                    h[i] = max(0.0, h[i] + diff2)
                    break
    return h

def compute_lag_values(h):

    n = len(h)
    dx = 2.0 / n

    for m in range(n):
        if m == 0:
            a_m = np.sum(h * h)
        else:
            a_m = np.sum(h[:n-m] * h[m:])
        p_m = h[:m].sum() if m > 0 else 0.0
        s_m = h[n-m:].sum() if m > 0 else 0.0
        L[m] = dx * (a_m + min(p_m, s_m))
    return L

import numpy as np

def repair_feasible(vec):
    h = np.asarray(vec, dtype=np.float64)
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-12:
        return h.copy()
    if diff > 0:
        order = np.argsort(h)
        rem = diff
        for i in order:
            if rem <= 1e-15:
                break
            add = min(rem, 1.0 - h[i])
            if add > 0:
                h[i] += add
                rem -= add
    else:
        order = np.argsort(-h)
        rem = -diff
        for i in order:
            if rem <= 1e-15:
                break
            remv = min(rem, h[i])
            if remv > 0:
                h[i] -= remv
                rem -= remv
    diff2 = target - h.sum()
    if abs(diff2) > 1e-14:
        if diff2 > 0:
            for i in range(n):
                if h[i] < 1.0:
                    h[i] = min(1.0, h[i] + diff2)
                    break
        else:
            for i in range(n):
                if h[i] > 0.0:
                    h[i] = max(0.0, h[i] + diff2)
                    break
    return h

    w = exp_vals / exp_vals.sum()

        return val

        _, g = smooth_objective_and_grad(h, beta, gamma)

        return np.sum(x) - n / 2.0

        return np.ones(n)

        if m == 0:
            a_m = np.dot(h, h)
            L[m] = dx * a_m
        else:
            a_m = np.dot(h[:n - m], h[m:])
            pm = prefix[m]
            sm = total - prefix[n - m]
            L[m] = dx * (a_m + min(pm, sm))

        acc[:m] += wm * wP
        acc[n - m:] += wm * wS

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr * dx))


def repair_feasible(vec):
    h = np.asarray(vec, dtype=np.float64)
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-13:
        return h.copy()
    if diff > 0:
        order = np.argsort(h)
        rem = diff
        for i in order:
            if rem <= 1e-15:
                break
            add = min(rem, 1.0 - h[i])
            if add > 0:
                h[i] += add
                rem -= add
    else:
        order = np.argsort(-h)
        rem = -diff
        for i in order:
            if rem <= 1e-15:
                break
            remv = min(rem, h[i])
            if remv > 0:
                h[i] -= remv
                rem -= remv
    diff2 = target - h.sum()
    if abs(diff2) > 1e-14:
        if diff2 > 0:
            for i in range(n):
                if h[i] < 1.0:
                    h[i] = min(1.0, h[i] + diff2)
                    break
        else:
            for i in range(n):
                if h[i] > 0.0:
                    h[i] = max(0.0, h[i] + diff2)
                    break
    return h


def compute_L(h):
    n = len(h)
    dx = 2.0 / n
    prefix = np.zeros(n + 1)
    for i in range(n):
        prefix[i + 1] = prefix[i] + h[i]
    total = prefix[n]
    L = np.zeros(n)
    L[0] = dx * np.dot(h, h)
    for m in range(1, n):
        a_m = np.dot(h[:n - m], h[m:])
        pm = prefix[m]
        sm = total - prefix[n - m]
        L[m] = dx * (a_m + min(pm, sm))
    return L


class SmoothProblem:
    def __init__(self, beta, gamma):
        self.beta = beta
        self.gamma = gamma
        self.n = 0

    def set_n(self, n):
        self.n = n

    def obj(self, x):
        hc = np.clip(x, 0.0, 1.0)
        v, _ = _smooth_obj_grad(hc, self.beta, self.gamma)
        return v

    def grad(self, x):
        hc = np.clip(x, 0.0, 1.0)
        _, g = _smooth_obj_grad(hc, self.beta, self.gamma)
        return g

    def eq_fun(self, x):
        return np.sum(x) - self.n / 2.0

    def eq_jac(self, x):
        return np.ones(self.n)


def _smooth_obj_grad(h, beta, gamma):
    n = len(h)
    dx = 2.0 / n
    L = compute_L(h)
    L_min = L.min()
    exp_vals = np.exp(-beta * (L - L_min))
    w = exp_vals / exp_vals.sum()
    obj = -L_min + np.log(exp_vals.sum()) / beta

    prefix = np.zeros(n + 1)
    for i in range(n):
        prefix[i + 1] = prefix[i] + h[i]
    total = prefix[n]

    acc = np.zeros(n)
    acc += w[0] * 2.0 * h
    for m in range(1, n):
        wm = w[m]
        if wm < 1e-14:
            continue
        acc[:n - m] += wm * h[m:]
        acc[m:] += wm * h[:n - m]
        pm = dx * prefix[m]
        sm = dx * (total - prefix[n - m])
        mp = min(pm, sm)
        ep = np.exp(-gamma * (pm - mp))
        es = np.exp(-gamma * (sm - mp))
        z = ep + es
        wP = ep / z
        wS = es / z
        acc[:m] += wm * wP
        acc[n - m:] += wm * wS
    grad = -dx * acc
    return obj, grad


def solve_smooth(h_start, beta, gamma=100.0, maxiter=300):
    n = len(h_start)
    x0 = repair_feasible(h_start).copy()
    prob = SmoothProblem(beta, gamma)
    prob.set_n(n)
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': prob.eq_fun, 'jac': prob.eq_jac}]
    try:
        result = minimize(
            prob.obj, x0, jac=prob.grad,
            method='SLSQP', bounds=bounds, constraints=constraints,
            options={'maxiter': maxiter, 'ftol': 1e-14}
        )
    except Exception:
        return None, None
    if result.x is not None:
        h_sol = repair_feasible(result.x)
        score = exact_score(h_sol)
        if np.isfinite(score):
            return h_sol, score
    return None, None


def active_descent(h, max_steps=100, tol_factor=0.004):
    n = len(h)
    h = repair_feasible(h)
    best = h.copy()
    best_score = exact_score(best)

    for _ in range(max_steps):
        L = compute_L(best)
        L_min = L.min()
        tol = tol_factor
        active = np.where(L <= L_min + tol)[0]
        if len(active) == 0:
            active = np.array([np.argmin(L)])
        dx = 2.0 / n
        prefix = np.zeros(n + 1)
        for i in range(n):
            prefix[i + 1] = prefix[i] + best[i]
        total = prefix[n]

        gaps = (L_min + tol) - L[active]
        weights = np.exp(gaps / 0.003 + 1e-10)
        weights /= weights.sum()

        direction = np.zeros(n)
        for idx, m in enumerate(active):
            wm = weights[idx]
            if m == 0:
                direction += wm * 2.0 * best
            else:
                direction[:n - m] += wm * best[m:]
                direction[m:] += wm * best[:n - m]
                pm = prefix[m]
                sm = total - prefix[n - m]
                if pm < sm:
                    direction[:m] += wm
                elif sm < pm:
                    direction[n - m:] += wm
                else:
                    direction[:m] += 0.5 * wm
                    direction[n - m:] += 0.5 * wm
        direction *= dx
        direction -= np.mean(direction)
        norm = np.linalg.norm(direction)
        if norm < 1e-15:
            break
        direction /= norm

        improved = False
        for alpha in [0.08, 0.05, 0.03, 0.02, 0.01, 0.005, 0.002, 0.001]:
            h_trial = repair_feasible(best + alpha * direction)
            score = exact_score(h_trial)
            if score < best_score - 1e-14:
                best = h_trial
                best_score = score
                improved = True
                break
        if not improved:
            break
    return best, best_score


def interpolate_to_dimension(vec, new_len):
    n = len(vec)
    x_src = (np.arange(n) + 0.5) * 2.0 / n
    x_tgt = (np.arange(new_len) + 0.5) * 2.0 / new_len
    xs_ext = np.concatenate([[0.0], x_src, [2.0]])
    ys_ext = np.concatenate([[vec[0]], vec, [vec[-1]]])
    return np.interp(x_tgt, xs_ext, ys_ext)


def gradient_descent_smooth(h_start, beta, gamma=100.0, n_iter=500, lr_init=0.02, momentum=0.9):
    n = len(h_start)
    h = repair_feasible(h_start).copy()
    v = np.zeros(n)
    best = h.copy()
    best_score = exact_score(best)
    for i in range(n_iter):
        lr = lr_init / (1.0 + 0.01 * i)
        _, g = _smooth_obj_grad(np.clip(h, 0, 1), beta, gamma)
        v = momentum * v - lr * g
        h_new = h + v
        h_new = repair_feasible(h_new)
        score = exact_score(h_new)
        if score < best_score - 1e-15:
            best = h_new.copy()
            best_score = score
        h = h_new
    return best, best_score


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()
    rng = np.random.RandomState(seed)

    def tl():
        return budget_s - (time.perf_counter() - t_start)

    try:
        parent = np.asarray(initial_h_values, dtype=np.float64)
    except Exception:
        parent = None

    if parent is not None and len(parent) > 0:
        best_h = repair_feasible(parent)
    else:
        best_h = np.full(70, 0.5)

    # Symmetrize
    h_sym = repair_feasible((best_h + best_h[::-1]) / 2.0)
    s_sym = exact_score(h_sym)
    if s_sym < exact_score(best_h) - 1e-13:
        best_h = h_sym
    best_score = exact_score(best_h)

    # Phase 1: Smooth SLSQP annealing
    betas = [15, 30, 60, 100, 150, 250]
    for beta in betas:
        if tl() < 30:
            break
        h_sol, score = solve_smooth(best_h, beta=beta, maxiter=200)
        if h_sol is not None and score < best_score - 1e-14:
            best_h = h_sol
            best_score = score
        # Quick active descent after each SLSQP
        if tl() > 20:
            h_ref, score_ref = active_descent(best_h, max_steps=40)
            if score_ref < best_score - 1e-14:
                best_h = h_ref
                best_score = score_ref

    # Phase 2: Gradient descent with multiple betas
    for beta in [50, 100, 200]:
        if tl() < 25:
            break
        h_gd, score_gd = gradient_descent_smooth(best_h, beta, n_iter=300, lr_init=0.01)
        if score_gd < best_score - 1e-14:
            best_h = h_gd
            best_score = score_gd
        if tl() > 15:
            h_ref, score_ref = active_descent(best_h, max_steps=50)
            if score_ref < best_score - 1e-14:
                best_h = h_ref
                best_score = score_ref

    # Phase 3: More SLSQP with higher beta
    for beta in [300, 500]:
        if tl() < 20:
            break
        h_sol, score = solve_smooth(best_h, beta=beta, maxiter=150)
        if h_sol is not None and score < best_score - 1e-14:
            best_h = h_sol
            best_score = score
        if tl() > 10:
            h_ref, score_ref = active_descent(best_h, max_steps=60, tol_factor=0.002)
            if score_ref < best_score - 1e-14:
                best_h = h_ref
                best_score = score_ref

    # Phase 4: Basin hopping (perturb + reoptimize)
    n_cur = len(best_h)
    for trial in range(20):
        if tl() < 30:
            break
        eps = rng.uniform(0.005, 0.03) * (1.0 + 0.5 * np.sin(trial * 0.7))
        z = rng.randn(n_cur)
        z -= z.mean()
        z /= np.linalg.norm(z)
        h_pert = repair_feasible(best_h + eps * z)
        h_sol, score = solve_smooth(h_pert, beta=100, maxiter=100)
        if h_sol is not None:
            h_ref, score_ref = active_descent(h_sol, max_steps=40)
            if score_ref < best_score - 1e-14:
                best_h = h_ref
                best_score = score_ref

    # Phase 5: Dimension reduction
    if tl() > 50:
        n_cur = len(best_h)
        coarse_dims = []
        for factor in [2, 3]:
            nd = n_cur // factor
            if nd >= 14 and nd % 2 == 0:
                coarse_dims.append(nd)
        for dn in [10, 20, 30, 40]:
            nd = n_cur - dn
            if 14 <= nd < n_cur and nd % 2 == 0:
                coarse_dims.append(nd)
        coarse_dims = sorted(set(coarse_dims))

        for nd in coarse_dims:
            if tl() < 25:
                break
            h_coarse = repair_feasible(interpolate_to_dimension(best_h, nd))
            s_coarse = exact_score(h_coarse)
            if s_coarse < best_score + 0.015:
                h_sol, score = solve_smooth(h_coarse, beta=80, maxiter=150)
                if h_sol is not None and score < best_score - 1e-14:
                    best_h = h_sol
                    best_score = score
                h_ref, score_ref = active_descent(best_h, max_steps=60)
                if score_ref < best_score - 1e-14:
                    best_h = h_ref
                    best_score = score_ref

    # Phase 6: Upsample and reoptimize (try slightly larger dims)
    if tl() > 40:
        n_cur = len(best_h)
        for nd in [n_cur + 10, n_cur + 20, n_cur + 40]:
            if tl() < 20 or nd > 250:
                break
            h_up = repair_feasible(interpolate_to_dimension(best_h, nd))
            s_up = exact_score(h_up)
            if s_up < best_score + 0.01:
                h_sol, score = solve_smooth(h_up, beta=100, maxiter=150)
                if h_sol is not None and score < best_score - 1e-14:
                    best_h = h_sol
                    best_score = score
                h_ref, score_ref = active_descent(best_h, max_steps=50)
                if score_ref < best_score - 1e-14:
                    best_h = h_ref
                    best_score = score_ref

    # Phase 7: Final active descent with tighter tolerance
    if tl() > 10:
        h_ref, score_ref = active_descent(best_h, max_steps=150, tol_factor=0.001)
        if score_ref < best_score - 1e-14:
            best_h = h_ref
            best_score = score_ref

    # Phase 8: Final SLSQP polish
    if tl() > 15:
        h_sol, score = solve_smooth(best_h, beta=400, maxiter=200)
        if h_sol is not None and score < best_score - 1e-14:
            best_h = h_sol
            best_score = score
        if tl() > 10:
            h_ref, score_ref = active_descent(best_h, max_steps=80, tol_factor=0.002)
            if score_ref < best_score - 1e-14:
                best_h = h_ref
                best_score = score_ref

    # Final symmetrization check
    h_sym2 = repair_feasible((best_h + best_h[::-1]) / 2.0)
    s_sym2 = exact_score(h_sym2)
    if s_sym2 < best_score - 1e-14:
        best_h = h_sym2
        best_score = s_sym2

    best_h = repair_feasible(best_h)
    best_score = exact_score(best_h)
    return best_h, best_score, len(best_h)
