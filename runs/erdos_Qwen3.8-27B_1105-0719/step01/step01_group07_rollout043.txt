import numpy as np
from scipy.optimize import minimize
import time


def exact_objective(h):
    n = len(h)
    if n == 0:
        return float('inf')
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(profile))


def exact_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx


def project_box_sum(h):
    h = np.array(h, dtype=np.float64).copy()
    n = len(h)
    target = n / 2.0
    h[~np.isfinite(h)] = 0.0
    for _ in range(200):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            h[mask] = np.minimum(1.0, h[mask] + diff / cnt)
        else:
            mask = h > 0.0
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            h[mask] = np.maximum(0.0, h[mask] + diff / cnt)
    return np.clip(h, 0.0, 1.0)


def binary_threshold(h):
    n = len(h)
    if n % 2 != 0:
        return None
    m = n // 2
    idx = np.argsort(-h, kind='stable')[:m]
    b = np.zeros(n, dtype=np.float64)
    b[idx] = 1.0
    return b


def transfer_resolution(h_src, n_tgt):
    n_src = len(h_src)
    if n_src == n_tgt:
        return project_box_sum(h_src.copy())
    x_src = (np.arange(n_src) + 0.5) / n_src
    x_tgt = (np.arange(n_tgt) + 0.5) / n_tgt
    h_new = np.interp(x_tgt, x_src, h_src)
    return project_box_sum(h_new)


def compute_1swap_delta(b, i, j, n):
    t_arr = np.arange(-(n - 1), n)
    delta = np.zeros(2 * n - 1)
    idx_i = i - t_arr
    idx_j = j - t_arr
    val_i = np.zeros(2 * n - 1)
    valid_i = (idx_i >= 0) & (idx_i < n)
    val_i[valid_i] = b[idx_i[valid_i]] - 1.0
    delta += val_i
    val_j = np.zeros(2 * n - 1)
    valid_j = (idx_j >= 0) & (idx_j < n)
    val_j[valid_j] = 1.0 - b[idx_j[valid_j]]
    delta += val_j
    shift_i = i + t_arr
    valid_si = (shift_i >= 0) & (shift_i < n) & (shift_i != i) & (shift_i != j)
    delta[valid_si] += b[shift_i[valid_si]]
    shift_j = j + t_arr
    valid_sj = (shift_j >= 0) & (shift_j < n) & (shift_j != i) & (shift_j != j)
    delta[valid_sj] -= b[shift_j[valid_sj]]
    return delta


def binary_1swap_search(b, profile_deadline, max_rounds=500):
    n = len(b)
    b = b.copy()
    curr_c = exact_objective(b)
    P = exact_profile(b) / (2.0 / n)
    ones = list(np.where(b == 1.0)[0])
    zeros = list(np.where(b == 0.0)[0])
    for rnd in range(max_rounds):
        if time.time() > profile_deadline - 2:
            break
        best_improve = 1e-14
        best_swap = None
        dx = 2.0 / n
        for ii in ones:
            for jj in zeros:
                d = compute_1swap_delta(b, ii, jj, n)
                new_P = P + d
                new_max = float(np.max(new_P)) * dx
                improve = curr_c - new_max
                if improve > best_improve:
                    best_improve = improve
                    best_swap = (ii, jj)
                    best_new_P = new_P.copy()
                    best_new_c = new_max
        if best_swap is not None:
            ii, jj = best_swap
            b[ii] = 0.0
            b[jj] = 1.0
            P = best_new_P
            curr_c = best_new_c
            ones = [ii for ii in ones if ii != ii]
            ones = [x for x in ones if x != ii]
            ones.append(jj)
            zeros.remove(jj)
            zeros.append(ii)
        else:
            break
    return b, exact_objective(b)


def binary_2swap_search(b, profile_deadline, rng, max_sample=3000, max_rounds=30):
    n = len(b)
    b = b.copy()
    curr_c = exact_objective(b)
    dx = 2.0 / n
    ones = list(np.where(b == 1.0)[0])
    zeros = list(np.where(b == 0.0)[0])
    for rnd in range(max_rounds):
        if time.time() > profile_deadline - 2:
            break
        best_improve = 1e-14
        best_move = None
        n_samples = min(max_sample, 5000)
        for _ in range(n_samples):
            i1, i2 = rng.choice(len(ones), 2, replace=False)
            j1, j2 = rng.choice(len(zeros), 2, replace=False)
            a1, a2 = ones[i1], ones[i2]
            b1, b2 = zeros[j1], zeros[j2]
            b[a1] = 0.0
            b[a2] = 0.0
            b[b1] = 1.0
            b[b2] = 1.0
            c = float(np.max(np.correlate(b, 1.0 - b, mode='full'))) * dx
            improve = curr_c - c
            if improve > best_improve:
                best_improve = improve
                best_move = (a1, a2, b1, b2)
            b[a1] = 1.0
            b[a2] = 1.0
            b[b1] = 0.0
            b[b2] = 0.0
        if best_move is not None:
            a1, a2, b1, b2 = best_move
            b[a1] = 0.0
            b[a2] = 0.0
            b[b1] = 1.0
            b[b2] = 1.0
            curr_c = exact_objective(b)
            ones = [x for x in ones if x not in (a1, a2)]
            ones.extend([b1, b2])
            zeros = [x for x in zeros if x not in (b1, b2)]
            zeros.extend([a1, a2])
        else:
            break
    return b, exact_objective(b)


def binary_sa(b, n, deadline, rng, max_iters=50000):
    b = b.copy()
    curr_c = exact_objective(b)
    best_c = curr_c
    best_b = b.copy()
    dx = 2.0 / n
    T = 0.01
    T_min = 0.0001
    cool = 0.995
    no_improve = 0
    for it in range(max_iters):
        if time.time() > deadline - 3:
            break
        if it % 200 == 0:
            no_improve = 0
        ones = list(np.where(b == 1.0)[0])
        zeros = list(np.where(b == 0.0)[0])
        if rng.random() < 0.7:
            i = rng.choice(len(ones))
            j = rng.choice(len(zeros))
            a, c_pos = ones[i], zeros[j]
            b[a] = 0.0
            b[c_pos] = 1.0
            new_c = float(np.max(np.correlate(b, 1.0 - b, mode='full'))) * dx
            delta_c = new_c - curr_c
            if delta_c < 0 or rng.random() < np.exp(-delta_c / T):
                curr_c = new_c
                if curr_c < best_c - 1e-12:
                    best_c = curr_c
                    best_b = b.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                b[a] = 1.0
                b[c_pos] = 0.0
                no_improve += 1
        else:
            if len(ones) >= 2 and len(zeros) >= 2:
                i1, i2 = rng.choice(len(ones), 2, replace=False)
                j1, j2 = rng.choice(len(zeros), 2, replace=False)
                a1, a2 = ones[i1], ones[i2]
                b1, b2 = zeros[j1], zeros[j2]
                b[a1] = 0.0
                b[a2] = 0.0
                b[b1] = 1.0
                b[b2] = 1.0
                new_c = float(np.max(np.correlate(b, 1.0 - b, mode='full'))) * dx
                delta_c = new_c - curr_c
                if delta_c < 0 or rng.random() < np.exp(-delta_c / T):
                    curr_c = new_c
                    if curr_c < best_c - 1e-12:
                        best_c = curr_c
                        best_b = b.copy()
                        no_improve = 0
                    else:
                        no_improve += 1
                else:
                    b[a1] = 1.0
                    b[a2] = 1.0
                    b[b1] = 0.0
                    b[b2] = 0.0
                    no_improve += 1
        T = max(T * cool, T_min)
        if no_improve > 2000:
            T = max(T * 3.0, 0.005)
            no_improve = 0
    return best_b, best_c


def lse_obj_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    m = float(np.max(profile))
    shifted = np.minimum((profile - m) / tau, 0.0)
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    weights = exp_vals / sum_exp
    obj = m + tau * float(np.log(sum_exp))
    grad = np.zeros(n)
    for j in range(2 * n - 1):
        w = weights[j]
        if w < 1e-12:
            continue
        s = j - (n - 1)
        if s >= 0:
            k1 = np.arange(s, n)
            grad[k1] += w * dx * (1.0 - h[k1 - s])
            k2 = np.arange(0, n - s)
            grad[k2] -= w * dx * h[k2 + s]
        else:
            a = -s
            k1 = np.arange(0, n - a)
            grad[k1] += w * dx * (1.0 - h[k1 + a])
            k2 = np.arange(a, n)
            grad[k2] -= w * dx * h[k2 - a]
    return obj, grad


class LSEFunc:
    def __init__(self, tau):
        self.tau = tau

    def value(self, x):
        return lse_obj_grad(x, self.tau)[0]

    def gradient(self, x):
        return lse_obj_grad(x, self.tau)[1]


class SumConstraint:
    def __init__(self, target):
        self.target = target

    def fun(self, x):
        return np.sum(x) - self.target

    def jac(self, x):
        return np.ones(len(x))


def slsqp_optimize(h0, tau, maxiter=200):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    obj = LSEFunc(tau)
    con = SumConstraint(n / 2.0)
    constraints = [{'type': 'eq', 'fun': con.fun, 'jac': con.jac}]
    try:
        res = minimize(obj.value, h0.copy(), method='SLSQP', jac=obj.gradient,
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-13})
        h_new = project_box_sum(np.array(res.x, dtype=np.float64))
        return h_new
    except Exception:
        return h0.copy()


def coord_descent_continuous(h, deadline, max_sweeps=50):
    n = len(h)
    h = project_box_sum(h.copy())
    curr_c = exact_objective(h)
    for sweep in range(max_sweeps):
        if time.time() > deadline - 3:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline - 3:
                break
            hi = h[i]
            best_h_i = hi
            best_c = curr_c
            for trial in np.clip(np.linspace(0, 1, 13), 0, 1):
                if abs(trial - hi) < 1e-10:
                    continue
                h[i] = trial
                target = n / 2.0
                adj = target - np.sum(h)
                if abs(adj) > 1e-14:
                    others = np.array([j for j in range(n) if j != i])
                    ovals = h[others]
                    mask_up = ovals < 1.0
                    mask_dn = ovals > 0.0
                    if adj > 0:
                        cnt = int(np.sum(mask_up))
                        if cnt == 0:
                            h[i] = hi
                            continue
                        h[others[mask_up]] = np.minimum(1.0, ovals[mask_up] + adj / cnt)
                    else:
                        cnt = int(np.sum(mask_dn))
                        if cnt == 0:
                            h[i] = hi
                            continue
                        h[others[mask_dn]] = np.maximum(0.0, ovals[mask_dn] + adj / cnt)
                    c = exact_objective(h)
                    if c < best_c - 1e-13:
                        best_c = c
                        best_h_i = trial.copy()
                        best_h_i = h.copy()
                    h[i] = hi
            if best_h_i != hi or (isinstance(best_h_i, np.ndarray)):
                h = best_h_i if isinstance(best_h_i, np.ndarray) else h
                if isinstance(best_h_i, np.ndarray):
                    h = best_h_i
                    curr_c = best_c
                    improved = True
        if not improved:
            break
    h = project_box_sum(h)
    return h, exact_objective(h)


def make_seed(n, rng, mode):
    if mode == 0:
        h = np.zeros(n)
        h[:n // 2] = 1.0
        h = h[rng.permutation(n)]
    elif mode == 1:
        x = np.linspace(0, 1, n)
        freq = rng.choice([2, 3, 4, 5, 6, 7, 8])
        phase = rng.uniform(0, 2 * np.pi)
        h = 0.5 + 0.4 * np.sin(2 * np.pi * freq * x + phase)
        h = np.clip(h, 0, 1)
    elif mode == 2:
        h = np.zeros(n)
        h[::2] = 1.0
    elif mode == 3:
        h = rng.uniform(0, 1, n)
    elif mode == 4:
        h = np.zeros(n)
        block = max(2, n // 8)
        for i in range(0, n, 2 * block):
            h[i:i + block] = 1.0
    elif mode == 5:
        h = np.zeros(n)
        h[:n // 4] = 1.0
        h[3 * n // 4:] = 1.0
    elif mode == 6:
        h = np.zeros(n)
        m = n // 2
        idx = rng.choice(n, m, replace=False)
        h[idx] = 1.0
    else:
        h = np.full(n, 0.5)
    return project_box_sum(h)


def refine_continuous(h0, deadline):
    h = project_box_sum(h0.copy())
    n = len(h)
    curr_c = exact_objective(h)
    taus = [0.005, 0.002, 0.0008, 0.0003, 0.0001]
    for tau in taus:
        if time.time() > deadline - 5:
            break
        h_new = slsqp_optimize(h, tau, maxiter=min(200, n * 3 + 20))
        c_new = exact_objective(h_new)
        if c_new < curr_c - 1e-13:
            h = h_new
            curr_c = c_new
    return h, curr_c
