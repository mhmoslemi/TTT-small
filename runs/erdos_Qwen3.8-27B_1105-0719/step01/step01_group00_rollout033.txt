    grad[idx] -= h[idx + d]

def time_remaining(t0, budget_s):

def project_feasible_full(h, target_sum):
    n = len(h)
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        hc = np.clip(h - mid, 0.0, 1.0)
        if hc.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    hc = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - hc.sum()
    if abs(diff) > 1e-12:
        mask = (hc > 1e-10) & (hc < 1.0 - 1e-10)
        if mask.sum() > 0:
            hc[mask] += diff / mask.sum()
    return np.clip(hc, 0.0, 1.0)


class LSEObj:
    def __init__(self, n, beta):
        self.n = n
        self.beta = beta
        self.dx = 2.0 / n

    def fun(self, q):
        h = make_sym_h(q, self.n)
        c = corr_vals(h)
        return logsumexp(self.beta * c) / self.beta * self.dx

    def jac(self, q):
        h = make_sym_h(q, self.n)
        c = corr_vals(h)
        sh = c - c.max()
        w = np.exp(self.beta * sh)
        ws = w.sum()
        if ws < 1e-30:
            w = np.ones_like(c) / len(c)
        else:
            w /= ws
        gf = np.zeros(self.n)
        active = np.where(w > 1e-13)[0]
        for k in active:
            gf += w[k] * grad_corr_lag(h, k, self.n)
        return q_grad_from_full(gf, self.n) * self.dx


class ConstrObj:
    def __init__(self, n, k):
        self.n = n
        self.k = k
        self.dx = 2.0 / n
        self.m = n // 2
        self.d = k - (n - 1)

    def __call__(self, x):
        q = x[:self.m]
        t = x[self.m]
        h = make_sym_h(q, self.n)
        d = self.d
        i_lo = max(0, -d)
        i_hi = self.n - 1 - abs(d)
        if i_lo > i_hi:
            return t
        idx = np.arange(i_lo, i_hi + 1)
        fk = np.sum(h[idx] * (1.0 - h[idx + d])) * self.dx
        return t - fk

    def jac(self, x):
        q = x[:self.m]
        h = make_sym_h(q, self.n)
        d = self.d
        i_lo = max(0, -d)
        i_hi = self.n - 1 - abs(d)
        gf = np.zeros(self.n)
        if i_lo <= i_hi:
            idx = np.arange(i_lo, i_hi + 1)
            gf[idx + d] += (1.0 - h[idx])
            gf[idx] -= h[idx + d]
        gq = q_grad_from_full(gf, self.n) * self.dx
        j = np.zeros(self.m + 1)
        j[:self.m] = -gq
        j[self.m] = 1.0
        return j


def epi_obj_fun(x):
    return x[-1]


def epi_obj_jac(x):
    j = np.zeros(len(x))
    j[-1] = 1.0
    return j


def slsqp_lse(q0, n, beta, maxiter=500):
    m = len(q0)
    obj = LSEObj(n, beta)
    try:
        res = minimize(obj.fun, q0, jac=obj.jac, method='SLSQP',
                       bounds=[(0.0, 1.0)] * m,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return np.clip(res.x, 0.0, 1.0)
    except Exception:
        return q0.copy()


def slsqp_epigraph(q0, n, active_lags, maxiter=300):
    m = n // 2
    h = make_sym_h(q0, n)
    c = corr_vals(h)
    dx = 2.0 / n
    t0 = float(c.max()) * dx + 1e-5
    x0 = np.concatenate([q0, [t0]])
    cons = []
    for k in active_lags:
        co = ConstrObj(n, k)
        cons.append({'type': 'ineq', 'fun': co, 'jac': co.jac})
    bounds = [(0.0, 1.0)] * m + [(0.0, 2.0)]
    try:
        res = minimize(epi_obj_fun, x0, jac=epi_obj_jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return np.clip(res.x[:m], 0.0, 1.0)
    except Exception:
        return q0.copy()


def subgrad_q(q0, n, max_iter=8000, alpha0=0.15):
    dx = 2.0 / n
    q = q0.copy()
    best_q = q.copy()
    best_c5 = c5_score(make_sym_h(q, n))
    alpha = alpha0
    no_imp = 0
    for _ in range(max_iter):
        h = make_sym_h(q, n)
        c = corr_vals(h)
        mc = float(c.max())
        c5v = mc * dx
        if c5v < best_c5 - 1e-15:
            best_c5 = c5v
            best_q = q.copy()
            no_imp = 0
        else:
            no_imp += 1
        if no_imp > 3000:
            break
        thresh = mc * 0.95
        active = np.where(c >= thresh)[0]
        gf = np.zeros(n)
        for k in active:
            gf += grad_corr_lag(h, k, n)
        gf /= len(active)
        gq = q_grad_from_full(gf, n) * dx
        q_new = np.clip(q - alpha * gq, 0.0, 1.0)
        nc5 = c5_score(make_sym_h(q_new, n))
        if nc5 < c5v:
            q = q_new
            alpha = min(alpha * 1.015, 0.12)
        else:
            alpha *= 0.94
        if alpha < 1e-13:
            break
    return best_q, best_c5


def active_set_refine(q0, n, max_rounds=5, max_lags=40):
    q = q0.copy()
    best_q = q.copy()
    best_c5 = c5_score(make_sym_h(q, n))
    for r in range(max_rounds):
        h = make_sym_h(q, n)
        c = corr_vals(h)
        order = np.argsort(c)[::-1]
        active = order[:max_lags]
        q_new = slsqp_epigraph(q, n, active, maxiter=200)
        c5_new = c5_score(make_sym_h(q_new, n))
        if c5_new < best_c5:
            best_c5 = c5_new
            best_q = q_new.copy()
            q = q_new
        else:
            q = q_new
            c5_new = c5_score(make_sym_h(q, n))
            if c5_new < best_c5:
                best_c5 = c5_new
                best_q = q.copy()
        c_all = corr_vals(make_sym_h(q, n))
        viol = np.where(c_all > c_all[active].max() + 1e-10)[0]
        if len(viol) == 0:
            break
        max_active = max(max_lags, len(active) + len(viol))
        if max_active > 2 * n - 1:
            max_active = 2 * n - 1
    return best_q, best_c5


def full_space_slsqp(h0, maxiter=400):
    n = len(h0)
    dx = 2.0 / n
    target = n / 2.0
    beta = 800.0

    class FullLSE:
        def fun(self, h):
            c = corr_vals(h)
            return logsumexp(beta * c) / beta * dx

        def jac(self, h):
            c = corr_vals(h)
            sh = c - c.max()
            w = np.exp(beta * sh)
            ws = w.sum()
            if ws < 1e-30:
                w = np.ones_like(c) / len(c)
            else:
                w /= ws
            gf = np.zeros(n)
            active = np.where(w > 1e-13)[0]
            for k in active:
                gf += w[k] * grad_corr_lag(h, k, n)
            return gf * dx

    obj = FullLSE()
    try:
        res = minimize(obj.fun, h0, jac=obj.jac, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        h_out = project_feasible_full(res.x, target)
        return h_out
    except Exception:
        return h0.copy()


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Get parent
    h_parent = np.array(initial_h_values, dtype=float)
    n_par = len(h_parent)
    dx_par = 2.0 / n_par
    target_par = n_par / 2.0
    h_parent = project_feasible_full(h_parent, target_par)
    c5_par = c5_score(h_parent)
    best_h = h_parent.copy()
    best_c5 = c5_par
    best_n = n_par

    # Symmetric projection
    n_sym = n_par
    if n_sym % 2 != 0:
        n_sym = n_par - 1
        x_old = np.linspace(0.0, 2.0, n_par)
        x_new = np.linspace(0.0, 2.0, n_sym)
        h_parent = project_feasible_full(np.interp(x_new, x_old, h_parent), n_sym / 2.0)
        c5_par = c5_score(h_parent)
        if c5_par < best_c5:
            best_c5 = c5_par
            best_h = h_parent.copy()
            best_n = n_sym

    m = n_sym // 2
    q0 = make_q_from_h(h_parent)
    h_sym = make_sym_h(q0, n_sym)
    h_sym = project_feasible_full(h_sym, n_sym / 2.0)
    c5_sym = c5_score(h_sym)

    if c5_sym < best_c5:
        best_c5 = c5_sym
        best_h = h_sym.copy()
        best_n = n_sym

    # Phase 2: LSE optimization in q-space with increasing beta
    betas = [80, 150, 300, 500, 800, 1200, 2000]
    for beta in betas:
        if time_remaining(t0, budget_s) < 30:
            break
        iters = int(min(600, max(200, time_remaining(t0, budget_s) * 2)))
        q_new = slsqp_lse(q0, n_sym, beta, maxiter=iters)
        c5_new = c5_score(make_sym_h(q_new, n_sym))
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            h_tmp = make_sym_h(q_new, n_sym)
            h_tmp = project_feasible_full(h_tmp, n_sym / 2.0)
            c5_new = c5_score(h_tmp)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_tmp.copy()
                best_n = n_sym
            q0 = q_new
        elif c5_new < best_c5 + 0.001:
            q0 = np.clip((q0 + q_new) / 2.0, 0.0, 1.0)

    # Phase 3: Active-set epigraph refinement
    if time_remaining(t0, budget_s) > 60:
        q0 = make_q_from_h(best_h) if len(best_h) == n_sym else make_q_from_h(
            project_feasible_full(np.interp(np.linspace(0, 2, n_sym),
                                            np.linspace(0, 2, len(best_h)), best_h), n_sym / 2.0))
        q_ref, c5_ref = active_set_refine(q0, n_sym, max_rounds=4, max_lags=35)
        h_ref = make_sym_h(q_ref, n_sym)
        h_ref = project_feasible_full(h_ref, n_sym / 2.0)
        c5_ref = c5_score(h_ref)
        if c5_ref < best_c5:
            best_c5 = c5_ref
            best_h = h_ref.copy()
            best_n = n_sym
            q0 = q_ref

    # Phase 4: Subgradient polishing in q-space
    if time_remaining(t0, budget_s) > 80:
        q0 = make_q_from_h(best_h) if len(best_h) == n_sym else make_q_from_h(
            project_feasible_full(np.interp(np.linspace(0, 2, n_sym),
                                            np.linspace(0, 2, len(best_h)), best_h), n_sym / 2.0))
        q_sg, c5_sg = subgrad_q(q0, n_sym, max_iter=10000, alpha0=0.12)
        h_sg = make_sym_h(q_sg, n_sym)
        h_sg = project_feasible_full(h_sg, n_sym / 2.0)
        c5_sg = c5_score(h_sg)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()
            best_n = n_sym
            q0 = q_sg

    # Phase 5: Another LSE + epigraph pass
    if time_remaining(t0, budget_s) > 50:
        q0 = make_q_from_h(best_h) if len(best_h) == n_sym else make_q_from_h(
            project_feasible_full(np.interp(np.linspace(0, 2, n_sym),
                                            np.linspace(0, 2, len(best_h)), best_h), n_sym / 2.0))
        q_new = slsqp_lse(q0, n_sym, 2000, maxiter=400)
        c5_new = c5_score(make_sym_h(q_new, n_sym))
        if c5_new < best_c5:
            best_c5 = c5_new
            h_tmp = make_sym_h(q_new, n_sym)
            h_tmp = project_feasible_full(h_tmp, n_sym / 2.0)
            best_c5 = c5_score(h_tmp)
            best_h = h_tmp.copy()
            best_n = n_sym
            q0 = q_new
        if time_remaining(t0, budget_s) > 40:
            q_ref, c5_ref = active_set_refine(q0, n_sym, max_rounds=3, max_lags=50)
            h_ref = make_sym_h(q_ref, n_sym)
            h_ref = project_feasible_full(h_ref, n_sym / 2.0)
            c5_ref = c5_score(h_ref)
            if c5_ref < best_c5:
                best_c5 = c5_ref
                best_h = h_ref.copy()
                best_n = n_sym

    # Phase 6: Full-space optimization from best symmetric result
    if time_remaining(t0, budget_s) > 100:
        h_fs = full_space_slsqp(best_h, maxiter=500)
        c5_fs = c5_score(h_fs)
        if c5_fs < best_c5:
            best_c5 = c5_fs
            best_h = h_fs.copy()
            best_n = len(h_fs)

    # Phase 7: Try different n values
    n_candidates = [124, 125, 126, 127, 128, 129, 130, 120, 132]
    n_candidates = [x for x in n_candidates if x != best_n and x >= 100 and x <= 200]
    for nc in n_candidates:
        if time_remaining(t0, budget_s) < 40:
            break
        if nc % 2 != 0:
            continue
        x_old = np.linspace(0.0, 2.0, len(best_h))
        x_new = np.linspace(0.0, 2.0, nc)
        h_new = project_feasible_full(np.interp(x_new, x_old, best_h), nc / 2.0)
        q_new = make_q_from_h(h_new)
        q_opt = slsqp_lse(q_new, nc, 400, maxiter=300)
        h_opt = make_sym_h(q_opt, nc)
        h_opt = project_feasible_full(h_opt, nc / 2.0)
        c5_opt = c5_score(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = nc
            if time_remaining(t0, budget_s) > 50:
                q2, c5_2 = active_set_refine(q_opt, nc, max_rounds=3, max_lags=40)
                h2 = make_sym_h(q2, nc)
                h2 = project_feasible_full(h2, nc / 2.0)
                c5_2 = c5_score(h2)
                if c5_2 < best_c5:
                    best_c5 = c5_2
                    best_h = h2.copy()
                    best_n = nc

    # Phase 8: Random restarts in q-space
    if time_remaining(t0, budget_s) > 60:
        for trial in range(20):
            if time_remaining(t0, budget_s) < 20:
                break
            h_cur = best_h
            if len(h_cur) != n_sym:
                x_old = np.linspace(0.0, 2.0, len(h_cur))
                x_new = np.linspace(0.0, 2.0, n_sym)
                h_cur = project_feasible_full(np.interp(x_new, x_old, h_cur), n_sym / 2.0)
            q_cur = make_q_from_h(h_cur)
            noise_scale = 0.05 + 0.1 * trial / 20.0
            q_rand = np.clip(q_cur + noise_scale * rng.randn(len(q_cur)), 0.0, 1.0)
            q_opt = slsqp_lse(q_rand, n_sym, 300, maxiter=200)
            h_opt = make_sym_h(q_opt, n_sym)
            h_opt = project_feasible_full(h_opt, n_sym / 2.0)
            c5_opt = c5_score(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = n_sym

    # Phase 9: Final polish
    if time_remaining(t0, budget_s) > 20 and best_h is not None:
        n_fin = len(best_h)
        if n_fin % 2 == 0:
            q_fin = make_q_from_h(best_h)
            q_fin = slsqp_lse(q_fin, n_fin, 3000, maxiter=300)
            h_fin = make_sym_h(q_fin, n_fin)
            h_fin = project_feasible_full(h_fin, n_fin / 2.0)
            c5_fin = c5_score(h_fin)
            if c5_fin < best_c5:
                best_c5 = c5_fin
                best_h = h_fin.copy()
                best_n = n_fin

    # Final validation
    if best_h is None:
        best_h = h_parent
        best_c5 = c5_par
        best_n = n_par

    best_h = np.clip(best_h, 0.0, 1.0)
    n_final = len(best_h)
    best_h = project_feasible_full(best_h, n_final / 2.0)
    final_c5 = c5_score(best_h)
    return best_h, final_c5, n_final
