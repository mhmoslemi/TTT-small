def two_coord_local_search(h, max_rounds=100, tol=1e-10):
    n = len(h)
    dx = 2.0 / n
    target = 0.5 * n
    h = h.copy()
    profile = np.correlate(h, 1.0 - h, mode='full')
    current_max = np.max(profile)
    
    for round in range(max_rounds):
        improved = False
        # Try all pairs in some order (or random order)
        pairs = [(i, j) for i in range(n) for j in range(i+1, n)]
        np.random.shuffle(pairs)  # or fixed order
        
        for i, j in pairs:
            # Feasible lambda range
            L = max(-h[i], h[j] - 1.0)
            U = min(1.0 - h[i], h[j])
            if U - L < 1e-12:
                continue
            
            # Compute move coefficients
            a = compute_move_coeffs(h, i, j, n)
            
            # Find best lambda
            best_lam, best_t = find_best_lambda(profile, a, L, U)
            
            if best_t < current_max - tol:
                # Accept the move
                h[i] += best_lam
                h[j] -= best_lam
                h = np.clip(h, 0.0, 1.0)
                # Re-normalize
                h = project_box_sum(h, target)
                profile = np.correlate(h, 1.0 - h, mode='full')
                current_max = np.max(profile)
                improved = True
        
        if not improved:
            break
    
    return h, np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def project_box_sum(h, target):
    n = len(h)
    h = np.array(h, dtype=np.float64).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
