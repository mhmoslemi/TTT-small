def exact_score(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))

def get_profile(h):

    return np.correlate(h, 1.0 - h, mode="full") * dx

def repair_feasible(h, target=None):
    h = np.array(h, dtype=np.float64).copy()
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    if target is None:
        target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        order = np.argsort(-room)
        remaining = diff
        for idx in order:
            if remaining <= 1e-15:
                break
            add = min(remaining, room[idx])
            h[idx] += add
            remaining -= add
    else:
        order = np.argsort(-h)
        remaining = -diff
        for idx in order:
            if remaining <= 1e-15:
                break
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
    return h

def get_lag_gradients(h):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    G = np.zeros((M, n))
    for k in range(M):
        delta = k - (n - 1)
        if delta >= 0:
            if delta < n:
                G[k, delta:n] += dx * (1.0 - h[0:n - delta])
                G[k, 0:n - delta] -= dx * h[delta:n]
        else:
            d = -delta
            if d < n:
                G[k, 0:n - d] += dx * (1.0 - h[d:n])
                G[k, d:n] -= dx * h[0:n - d]
    return G

            break
        h_interp = interpolate_to_dim(best_h, new_n)
        s_interp = exact_score(h_interp)
        if time.perf_counter() - t_start >= budget_s - 20:
            if s_interp < best_score - 1e-12:
                best_h, best_score = h_interp, s_interp
            continue
        h_new, s_new = pair_descent(h_interp, 30000, rng, elapsed, mode='random')
        if s_new < best_score - 1e-12:
            best_h, best_score = h_new, s_new

    # Phase 5: Greedy + random alternating on best
    if time.perf_counter() - t_start < budget_s - 20:
        for mode in ['greedy', 'random', 'greedy', 'random']:
            if time.perf_counter() - t_start >= budget_s - 15:
                break
            max_upd = 3000 if mode == 'greedy' else 50000
            h_new, s_new = pair_descent(best_h, max_upd, rng, elapsed, mode=mode)
            if s_new < best_score - 1e-12:
                best_h, best_score = h_new, s_new

    # Phase 6: Small perturbations to escape local optima
    if time.perf_counter() - t_start < budget_s - 20:
        n = len(best_h)
        for p in range(8):
            if time.perf_counter() - t_start >= budget_s - 15:
                break
            eps = 0.001 + 0.001 * p
            z = rng.randn(n)
            z -= z.mean()
            h_pert = repair_feasible(best_h + eps * z)
            s_pert = exact_score(h_pert)
            if s_pert < best_score - 1e-12:
                h_new, s_new = pair_descent(h_pert, 20000, rng, elapsed, mode='random')
                if s_new < best_score - 1e-12:
                    best_h, best_score = h_new, s_new
            elif time.perf_counter() - t_start < budget_s - 1

5:
                h_new, s_new = pair_descent(h_pert, 3000, rng, elapsed, mode='greedy')
                if s_new < best_score - 1e-12:
                    best_h, best_score = h_new, s_new

    # Phase 7: Final polishing with greedy
    if time.perf_counter() - t_start < budget_s - 5:
        h_fin, s_fin = pair_descent(best_h, 1000, rng, elapsed, mode='greedy')
        if s_fin < best_score - 1e-12:
            best_h, best_score = h_fin, s_fin

    # Final
    best_h = repair_feasible(best_h)
    final_score = exact_score(best_h)

    return (best_h, final_score, len(best_h))
