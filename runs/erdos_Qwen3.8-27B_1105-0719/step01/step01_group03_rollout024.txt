def smooth_objective(h, mu):
    profile = compute_profile(h)
    # Log-sum-exp for numerical stability
    max_p = np.max(profile)
    return (max_p + np.log(np.sum(np.exp(mu * (profile - max_p)))) / mu)

def smooth_gradient(h, mu):
    n = len(h)
    dx = 2.0 / n
    profile = compute_profile(h)
    max_p = np.max(profile)
    weights = np.exp(mu * (profile - max_p))
    weights /= np.sum(weights)
    # Gradient is weighted sum of individual gradients
    # Instead of computing all M gradients, use the formula directly
    # grad = ∑_k w_k * grad_k
    # For lag d, grad_k[i] = dx * [(1-h[i-d]) if valid - h[i+d] if valid]
    # Summing with weights:
    g = np.zeros(n)
    for k in range(2*n-1):
        d = k - (n-1)
        wk = weights[k]
        if wk < 1e-15:
            continue
        if d == 0:
            g += wk * dx * (1 - 2*h)
        elif d > 0:
            for i in range(n):
                if 0 <= i - d <= n - 1 - d:
                    g[i] += wk * dx * (1.0 - h[i-d])
                if 0 <= i <= n - 1 - d:
                    g[i] -= wk * dx * h[i+d]

def smooth_obj_and_grad(h, mu):
    """Compute log-sum-exp smoothed objective and gradient."""
    n = len(h)
    dx = 2.0 / n
    profile = compute_profile(h)
    # Log-sum-exp
    max_p = np.max(profile)
    shifted = mu * (profile - max_p)
    log_sum = np.log(np.sum(np.exp(shifted)))
    obj = max_p + log_sum / mu
    # Weights
    weights = np.exp(shifted) / np.sum(np.exp(shifted))
    # Gradient: weighted sum of profile gradients
    g = np.zeros(n)
    for k in range(2*n-1):
        wk = weights[k]
        if wk < 1e-14:
            continue
        d = k - (n-1)
        if d == 0:
            g += wk * dx * (1 - 2*h)
        elif d > 0:
            # (1-h[i-d]) for i in [d, n-1]
            g[d:] += wk * dx * (1 - h[:n-d])
            # -h[i+d] for i in [0, n-1-d]
            g[:n-d] -= wk * dx * h[d:]
        else:
            dp = -d
            # (1-h[i+dp]) for i in [0, n-1-dp]
            g[:n-dp] += wk * dx * (1 - h[dp:])
            # -h[i-dp] for i in [dp, n-1]
            g[dp:] -= wk * dx * h[:n-dp]
    # Project gradient onto sum=0 subspace
    g -= np.mean(g)
    return obj, g

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx] + 1e-15)
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx] + 1e-15)
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break

    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

    """Project onto [0,1]^n ∩ {sum = n/2}"""
    return repair_feasible(vec)

        h = h_new
        c = exact_score(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    
    return best_h, best_c

One key insight: at the optimum, the constraint sum(h) = n/2 is active, and h is bounded by [0,1]. The KKT conditions tell us that at the optimum:

def project_box_sum(v, target):
    """Project v onto [0,1]^n ∩ {sum = target}."""
    n = len(v)
    # Bisection on lambda
    lo = v.min() - 1  # ensures clip(v - lo, 0, 1) = 1 for all
    hi = v.max() + 1  # ensures clip(v - hi, 0, 1) = 0 for all
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(v - mid, 0, 1))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2
    return np.clip(v - lam, 0, 1)

def interpolate_to_n(h, new_n):

                break
        if not found:
            # Try smaller active set (only the top 1-3)
            top_k = np.argsort(profile)[-min(3, len(active)):]
            g2 = np.zeros(n)
            for k in top_k:
                d = k - (n - 1)
                if d == 0:
                    g2 += dx * (1.0 - 2.0 * h)
                elif d > 0:
                    g2[d:] += dx * (1.0 - h[:n - d])
                    g2[:n - d] -= dx * h[d:]
                else:
                    dp = -d
                    g2[:n - dp] += dx * (1.0 - h[dp:])
                    g2[dp:] -= dx * h[:n - dp]
            g2 -= np.mean(g2)
            g2_norm = np.linalg.norm(g2)
            if g2_norm > 1e-14:
                g2 /= g2_norm
                for alpha in [0.01, 0.005, 0.002, 0.001, 0.0005]:
                    h_try = project_box_sum(h - alpha * g2, target)
                    s = exact_score(h_try)
                    if s < best_c - 1e-13:
                        h = h_try
                        best_h = h.copy()
                        best_c = s
                        found = True
                        break
        if not found:
            break
    return best_h, best_c

def symmetrize(h):
    return (h + h[::-1]) / 2.0

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    return project_box_sum(h, target)

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()
    time_budget = budget_s
    rng = np.random.RandomState(seed)
    
    def time_left():
        return time_budget + t_start - time.perf_counter()
    
    parent = None
    try:
        parent = np.array(initial_h_values, dtype=np.float64)
    except NameError:
        pass
    except Exception:
        pass
    
    if parent is None or len(parent) < 10:
        parent = np.full(66, 0.5)
    
    parent = project_box_sum(parent, len(parent) / 2.0)
    best_h = parent.copy()
    best_c = exact_score(parent)
    best_n = len(parent)
    
    # Symmetrize
    h_sym = project_box_sum(symmetrize(parent), best_n / 2.0)
    c_sym = exact_score(h_sym)
    if c_sym < best_c - 1e-12:
        best_h = h_sym
        best_c = c_sym
    
    # Also try 1-h symmetry (same C5 by symmetry)
    h_inv = project_box_sum(1.0 - parent, best_n / 2.0)
    c_inv = exact_score(h_inv)
    if c_inv < best_c - 1e-12:
        best_h = h_inv
        best_c = c_inv
    
    # Multi-dimension optimization
    dim_list = []
    pn = len(parent)
    for d in [30, 40, 48, 50, 52, 55, 58, 60, 62, 64, 66, 68, 70, 72, 75, 80]:
        if 20 <= d <= 120:
            dim_list.append(d)
    # Sort: prefer closer to parent, then smaller
    dim_list.sort(key=lambda d: (abs(d - pn), d))
    # Limit to ~8 dims to fit budget
    dim_list = dim_list[:8]
    
    for dim in dim_list:
        if time_left() < 80:
            break
        h_dim = interpolate_to_n(best_h, dim)
        c_dim = exact_score(h_dim)
        
        # PGD continuation
        h_pgd, c_pgd = pgd_continuation(h_dim, t_start + time_left() * 0.5, max_total_iters=150)
        if c_pgd < c_dim - 1e-10:
            h_dim = h_pgd
            c_dim = c_pgd
        
        # FISTA
        h_fista, c_fista = fista_continuation(h_dim, t_start + time_left() * 0.3, max_total_iters=100)
        if c_fista < c_dim - 1e-10:
            h_dim = h_fista
            c_dim = c_fista
        
        # Active set descent
        h_as, c_as = active_set_descent(h_dim, t_start + time_left() * 0.3, max_iters=60)
        if c_as < c_dim - 1e-10:
            h_dim = h_as
            c_dim = c_as
        
        # SLSQP polish
        if time_left() > 50:
            h_sl = slsqp_solve(h_dim, maxiter=250, tol=1e-10)
            if h_sl is not None:
                c_sl = exact_score(h_sl)
                if c_sl < c_dim - 1e-10:
                    h_dim = h_sl
                    c_dim = c_sl
        
        if c_dim < best_c - 1e-12:
            best_h = h_dim.copy()
            best_c = c_dim
            best_n = dim
    
    # Additional perturbations from best
    if time_left() > 100:
        for eps in [0.01, 0.02, 0.03, 0.05]:
            if time_left() < 60:
                break
            z = rng.randn(best_n)
            z -= z.mean()
            h_pert = project_box_sum(best_h + eps * z, best_n / 2.0)
            h_pgd, c_pgd = pgd_continuation(h_pert, t_start + time_left() * 0.4, max_total_iters=80)
            if c_pgd < best_c - 1e-10:
                h_as, c_as = active_set_descent(h_pgd, t_start + time_left() * 0.3, max_iters=40)
                if c_as < c_pgd:
                    best_h = h_as
                    best_c = c_as
                else:
                    best_h = h_pgd
                    best_c = c_pgd
    
    # Symmetrize the best and check
    h_sym2 = project_box_sum(symmetrize(best_h), best_n / 2.0)
    c_sym2 = exact_score(h_sym2)
    if c_sym2 < best_c - 1e-12:
        best_h = h_sym2
        best_c = c_sym2
    
    # Final active set descent
    if time_left() > 30:
        h_as, c_as = active_set_descent(best_h, t_start + time_left() * 0.8, max_iters=80)
        if c_as < best_c - 1e-13:
            best_h = h_as
            best_c = c_as
    
    # Final SLSQP polish
    if time_left() > 50:
        h_sl = slsqp_solve(best_h, maxiter=400, tol=1e-11)
        if h_sl is not None:
            c_sl = exact_score(h_sl)
            if c_sl < best_c - 1e-13:
                best_h = h_sl
                best_c = c_sl
    
    # Final validation
    best_h = project_box_sum(best_h, best_n / 2.0)
    final_c = exact_score(best_h)
    
    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            final_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 2:
            final_c = float(eval_result[0])
    except Exception:
        pass
    
    return best_h, final_c, len(best_h)
