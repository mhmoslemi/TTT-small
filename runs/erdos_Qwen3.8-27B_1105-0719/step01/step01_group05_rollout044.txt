```python
def exact_objective(h):

def project_box_sum(h, target=None):
    n = len(h)
    if target is None:
        target = 0.5 * n
    h = np.array(h, dtype=np.float64).copy()
    h = np.nan_to_num(h, nan=0.0, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            count = np.sum(mask)
            if count == 0:
                break
            step = min(diff / count, 1.0 - h[mask].min())
            h[mask] = np.minimum(h[mask] + step, 1.0)
        else:
            mask = h > 0.0
            count = np.sum(mask)
            if count == 0:
                break
            step = min(-diff / count, h[mask].min())
            h[mask] = np.maximum(h[mask] - step, 0.0)
    # Final fix
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        # Try to fix on a non-boundary element
        for i in range(n):
            if h[i] > 1e-12 and h[i] < 1.0 - 1e-12:
                adj = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += adj
                diff -= adj
                if abs(diff) < 1e-15:
                    break
    return h

def compute_lag_gradient(h, lag, dx):
    """Compute gradient of q_lag w.r.t. h."""
    n = len(h)
    grad = np.zeros(n)
    if lag >= 0:
        for i in range(n - lag):
            grad[i + lag] += dx * (1.0 - h[i])
            grad[i] -= dx * h[i + lag]
    else:
        s = -lag
        for i in range(s, n):
            grad[i - s] += dx * (1.0 - h[i])
            grad[i] -= dx * h[i - s]
    return grad
```

```python
def compute_lag_gradient(h, lag, dx):
    n = len(h)
    grad = np.zeros(n)
    if lag >= 0:
        if lag == 0:
            # q = dx * sum h_i(1-h_i), grad_i = dx*(1-2h_i)
            grad[:] = dx * (1.0 - 2.0 * h)
        else:
            grad[lag:] += dx * (1.0 - h[:n-lag])
            grad[:n-lag] -= dx * h[lag:]
    else:
        s = -lag
        grad[:n-s] += dx * (1.0 - h[s:])
        grad[s:] -= dx * h[:n-s]
    return grad
```

```python
def slsqp_active(h0, n, active_lags, target, theta, maxiter=100):
    """
    SLSQP minimizing c subject to q_r(h) - c <= 0 for r in active_lags,
    sum(h) = target, 0 <= h <= 1.
    Variables: z = [h(0), ..., h(n-1), c]
    """
    z0 = np.concatenate([h0, [theta]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    
    def obj(z):
        return z[-1]
    
    def obj_grad(z):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    
    # Equality: sum(h) = target
    cons = [{'type': 'eq',
             'fun': lambda z: np.sum(z[:n]) - target,
             'jac': lambda z: np.concatenate([np.ones(n), [0.0]])}]
    
    # Inequality: q_r(h) - c <= 0, i.e., -q_r(h) + c >= 0
    for lag in active_lags:
        r = lag + (n - 1)  # Convert lag to correlation index
        def cons_fun(z, r=r):
            h = z[:n]
            c = z[-1]
            q = np.correlate(h, 1.0 - h, mode='full')[r] * (2.0/n)
            return c - q  # Must be >= 0
        def cons_jac(z, r=r):
            h = z[:n]
            dx = 2.0 / n
            grad_h = compute_lag_gradient(h, r - (n-1), dx)
            g = np.concatenate([grad_h, [-1.0]])  # d(c-q)/dz = (-grad_h, 1)
            return g
        cons.append({'type': 'ineq', 'fun': cons_fun, 'jac': cons_jac})
    
    res = minimize(obj, z0, jac=obj_grad, method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-12})
    
    if res.x is not None and np.all(np.isfinite(res.x)):
        h_new = project_box_sum(res.x[:n], target)
        c_new, _ = exact_objective(h_new)
        return h_new, c_new
    return h0, exact_objective(h0)[0]
```
