def project_box_sum(h, target=None):
    n = len(h)
    h = np.array(h, dtype=np.float64)
    h = np.nan_to_num(h, nan=0.0, posinf=1.0, neginf=0.0)
    if target is None:
        target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = np.sum(h)
        diff = s - target
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h > 0
            if not np.any(mask):
                break
            step = min(diff / np.sum(mask), np.min(h[mask]) - 1e-15)
            if step <= 0:
                break
            h[mask] -= step
        else:
            mask = h < 1.0
            if not np.any(mask):
                break
            step = min(-diff / np.sum(mask), 1e-15 + (1.0 - np.max(h[mask])))

            h[mask] -= diff / np.sum(mask)
        else:
            mask = h < 1
            h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

For a given pair (i, j) and a candidate alpha:
1. Create h_new = h with h_new[i] += alpha, h_new[j] -= alpha
2. Compute the score

But this requires O(n²) for the full correlate each time, which is expensive.

Better: use the fact that only O(n) profiles change. Let me precompute the baseline profile, and for a given (i,j,alpha), compute which profiles change and by how much.

Actually, the simplest efficient approach:
- Precompute base_profile = np.correlate(h, 1-h, 'full') * dx  (O(n²) once)
- For a pair (i,j), the changes to the profile are:
  - np.correlate(delta_h, 1-h, 'full') * dx + np.correlate(h, delta_g, 'full') * dx + np.correlate(delta_h, delta_g, 'full') * dx
  where delta_h has +alpha at i, -alpha at j, 0 elsewhere
  and delta_g = -delta_h (since g = 1-h)
  
  But np.correlate with a sparse vector is still O(n²) in numpy.

Hmm, let me think differently. The profile at lag d is:
c[d] = sum_j h[j+d]*(1-h[j])

If I change h[i] += alpha and h[j] -= alpha, the change in c[d] is:
Δc[d] = (change in terms involving h[i]) + (change in terms involving h[j])

Terms involving h[i]:
- As h[a] (first factor): term is h[i]*(1-h[i+d]) if 0<=i+d<n
  Change: alpha*(1-h[i+d]) [if i+d != j]
  or alpha*(1-h[j]+alpha) - ... [if i+d == j, more complex]

  Change: -alpha*h[i-d] [if i-d != j]

  Change: -alpha*(1-h[j+d]) [if j+d != i]

The cross terms (where both h[i] and h[j] appear in the same product) occur when:
- i+d = j (d = j-i): term h[i]*(1-h[j])
- i-d = j (d = i-j): term h[j]*(1-h[i])
- j+d = i (d = i-j): term h[i]*(1-h[j])  [same as second bullet]

So my algorithm:
1. Start from parent
2. In each iteration, try a random pair (i,j)
3. For that pair, do a 1D optimization over alpha (golden section or grid)
4. Each evaluation of the objective for a given alpha: create new h (O(n) copy + 2 modifications), compute correlate (O(n²) but fast in numpy)
5. Accept if improvement
6. Repeat until convergence or time budget

With this approach, I can do maybe 100,000-200,000 evaluations in the time budget, which should be enough to find good improvements.

def golden_section_minimize(f, a, b, tol=1e-8, max_iter=100):

    # Filter to feasible
    alphas = alphas[(alphas >= lo + 1e-12) & (alphas <= hi - 1e-12)]

    # d = t-N+1: corr[t] = sum_m x[m+d] * (1-x[m])
    
    # d/dx[i] corr[t] = (1-x[i+d]) [if i+d valid, from x[i+d] term] 
    #                   - x[i-d] [if i-d valid, from (1-x[m]) term where m=i-d, so m+d=i]
    # Wait: corr[t] = sum_m x[m+d]*(1-x[m])
    # d/dx[i] = (1-x[i+d]) * [i = m+d, m = i-d] - x[i-d] * [i = m, m+d = i, m = i-d]
    # Hmm, let me be more careful:
    # corr[t] = sum_m x[m+d] * (1-x[m])
    # Term m: x[m+d]*(1-x[m])
    # d/dx[i] of term m: 
    #   if m+d = i (m = i-d): (1-x[m]) = (1-x[i-d])  [x[i] is the x[m+d] factor]
    #   if m = i: -x[m+d] = -x[i+d]  [x[i] is in (1-x[m]) = (1-x[i])]
    # So d/dx[i] corr[t] = (1-x[i-d]) * [valid: 0<=i-d<n and 0<=i<n] 
    #                       - x[i+d] * [valid: 0<=i<n and 0<=i+d<n]
    
    # Actually wait, I need to double check the correlate convention.
    # Let me verify with a small example:
    # x = [a, b, c], 1-x = [1-a, 1-b, 1-c], n=3
    # np.correlate([a,b,c], [1-a,1-b,1-c], 'full') should give length 5
    # By numpy docs: np.correlate(a, v, 'full')[k] = sum_n a[n+k] * conj(v[n])
    # Wait no, the actual definition is:
    # c_k = sum_n a_{n+k} \bar{b_n}  for the 'full' mode
    # Or is it: c_k = sum_n a_n \bar{b_{n+k}}?
    
    # Let me just check: np.correlate([1,2,3], [0,1,0.5], 'full')
    # According to numpy: "The cross-correlation of a and v"
    # c_k = sum_n a[n] * v[n+k]  ... or is it shifted?
    
    # Actually from numpy source, correlate(a, v) in 'full' mode:
    # result has length len(a) + len(v) - 1
    # result[k] = sum_n a[n] * v[n-k+(len(v)-1)]  ... I'm getting confused
    
    # Let me just use the definition that works: for a=[a0,a1,a2], b=[b0,b1,b2]:
    # full = [a0*b2, a0*b1+a1*b2, a0*b0+a1*b1+a2*b2, a1*b0+a2*b1, a2*b0]
    # So full[k] for k=0..4:
    # full[0] = a[0]*b[2]
    # full[1] = a[0]*b[1] + a[1]*b[2]
    # full[2] = a[0]*b[0] + a[1]*b[1] + a[2]*b[2]
    # full[3] = a[1]*b[0] + a[2]*b[1]
    # full[4] = a[2]*b[0]
    
    # So full[k] = sum_n a[n] * b[n - (k - (len(b)-1))]  where valid indices
    # = sum_n a[n] * b[n-k+2]  (for n=3, len(b)-1=2)
    
    # In our case b = 1-a, so:
    # full[k] = sum_n a[n] * (1-a[n-k+2])
    # Let d = k - 2 = k - (n-1) where n=3: d = k-2
    # full[k] = sum_n a[n] * (1-a[n+d])  where 0<=n<n and 0<=n+d<n
    
    # So for general n: full[k] = sum_j h[j] * (1-h[j+d]) where d = k-(n-1), 0<=j<n, 0<=j+d<n
    
    # d/dx[i] full[k]:
    # Term j: h[j]*(1-h[j+d])
    # If j = i: contributes (1-h[i+d]) * [1 if 0<=i+d<n, 0 otherwise]
    # If j+d = i (j = i-d): contributes -h[i-d] * [1 if 0<=i-d<n, 0 otherwise]
    
    # So d/dx[i] full[k] = (1-h[i+d]) * I(0<=i+d<n) - h[i-d] * I(0<=i-d<n)
    # where d = k-(n-1)
    
    # This matches what I had before (with the convention d = k-(n-1)).
    
    for k in range(2 * n - 1):
        w_k = w[k]
        if w_k < 1e-14:
            continue
        d = k - (n - 1)
        for i in range(n):
            contrib = 0.0
            # (1-h[i+d]) term
            if 0 <= i + d < n:
                contrib += (1.0 - x[i + d])
            # -h[i-d] term
            if 0 <= i - d < n:
                contrib -= x[i - d]
            grad[i] += w_k * contrib * dx
    
    return m + tau * np.log(np.sum(exp_s)), grad

def slsqp_optimize(h0, tau, maxiter=100, ftol=1e-12):
    n = len(h0)
    dx = 2.0 / n
    
    def obj(x):
        return smooth_logsumexp_obj(x, n, tau)
    
    def obj_grad(x):
        return smooth_logsumexp_grad(x, n, tau)
    
    # Add penalty for sum constraint
    def penal_obj(x):
        v = smooth_logsumexp_obj(x, n, tau)
        sv = np.sum(x) - n / 2.0
        return v + 1e6 * sv * sv
    
    def penal_grad(x):
        _, g = smooth_logsumexp_grad(x, n, tau)
        sv = np.sum(x) - n / 2.0
        return g + 2e6 * sv * np.ones(n)
    
    from scipy.optimize import minimize
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(penal_obj, h0, method='SLSQP', jac=penal_grad,
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': ftol})
        return project_box_sum(res.x)
    except Exception:
        return None

def transfer_to_size(parent, target_n):
    n = len(parent)
    if n == target_n:
        return parent.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(target_n) + 0.5) / target_n
    new_h = np.interp(x_new, x_old, parent)
    return project_box_sum(new_h)

    def time_left():
        return deadline - time.time()
    
    # Phase 1: Pairwise transfer optimization on parent size
    if time_left() > 120:
        h = best_h.copy()

        n = len(h)

        n_passes = 0
        while time_left() > 120:

            n_trials = 500 + 50 * n_passes
            for t in range(n_trials):

                alphas = alphas[(alphas >= lo + 1e-10) & (alphas <= hi - 1e-10)]
                if len(alphas) == 0:
