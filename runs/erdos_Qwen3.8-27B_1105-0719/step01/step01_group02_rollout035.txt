def compute_delta_profile(b, i, j, n):
    """Compute the change in the correlation profile when swapping b[i]:1->0 and b[j]:0->1."""
    delta = np.zeros(2*n-1)
    nm1 = n - 1

    ks = np.arange(max(0, i), min(2*n-1, i+n))
    if len(ks) > 0:
        idx = i - ks + nm1
        valid = (idx >= 0) & (idx < n)
        delta[ks[valid]] += b[idx[valid]]
    
    # Term 2: -b[j-k+nm1]
    ks = np.arange(max(0, j), min(2*n-1, j+n))

        valid = (idx >= 0) & (idx < n)
        delta[ks[valid]] -= b[idx[valid]]

    ks = np.arange(max(0, nm1-j), min(2*n-1, nm1+j+1))

        delta[ks[valid]] += (1.0 - b[idx[valid]])

        valid = (idx >= 0) & (idx < n)

    km = nm1 - d
    if 0 <= kp <= 2*n-2:
        delta[kp] += 1
    if 0 <= km <= 2*n-2:
        delta[km] += 1
    delta[nm1] -= 2
    
    return delta

def project_box_sum(h, target=None):

def to_binary(h):
    n = len(h)
    if n % 2 != 0:
        n = n - 1
        h = h[:n]
    target = n // 2
    b = np.zeros(n)
    # Use a threshold based on rank
    idx = np.argsort(-h[:n])[:target]
    b[idx] = 1.0
    return b

def transfer_to_size(parent, target_n):
    n = len(parent)
    if n == target_n:
        return parent.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(target_n) + 0.5) / target_n
    new_h = np.interp(x_new, x_old, parent)
    return project_box_sum(new_h)

def compute_delta(b, i, j, n):
    """Fast delta computation for 2-exchange (i: 1->0, j: 0->1)."""
    nm1 = n - 1
    size = 2 * n - 1
    delta = np.zeros(size)
    
    # Term 1: +b[i-k+nm1] for valid k
    k_lo = max(0, i)
    k_hi = min(size - 1, i + n - 1)
    if k_lo <= k_hi:
        ks = np.arange(k_lo, k_hi + 1)
        idx = i - ks + nm1
        delta[ks] += b[idx]
    
    # Term 2: -b[j-k+nm1] for valid k
    k_lo = max(0, j)
    k_hi = min(size - 1, j + n - 1)
    if k_lo <= k_hi:
        ks = np.arange(k_lo, k_hi + 1)
        idx = j - ks + nm1
        delta[ks] -= b[idx]
    
    # Term 3: +(1-b[j+k-nm1]) for valid k  [j+k-(n-1) in [0,n-1]]
    k_lo = max(0, nm1 - j)
    k_hi = min(size - 1, nm1 + j)
    if k_lo <= k_hi:
        ks = np.arange(k_lo, k_hi + 1)
        idx = j + ks - nm1
        delta[ks] += (1.0 - b[idx])
    
    # Term 4: -(1-b[i+k-nm1]) for valid k
    k_lo = max(0, nm1 - i)
    k_hi = min(size - 1, nm1 + i)
    if k_lo <= k_hi:
        ks = np.arange(k_lo, k_hi + 1)
        idx = i + ks - nm1
        delta[ks] -= (1.0 - b[idx])
    
    # Spikes from correlate(da, db)
    d = i - j
    kp = nm1 + d
    km = nm1 - d
    if 0 <= kp <= size - 1:
        delta[kp] += 1.0
    if 0 <= km <= size - 1:
        delta[km] += 1.0
    delta[nm1] -= 2.0
    
    return delta

def binary_local_search(b, max_rounds=2000, rng_seed=42, strategy='best'):
    """Intensive binary local search using 2-exchange moves."""
    n = len(b)
    b = b.astype(np.float64).copy()
    P = np.correlate(b, 1.0 - b, mode='full').astype(np.float64)
    cur_max = np.max(P)
    ones = list(np.where(b == 1.0)[0].tolist())
    zeros = list(np.where(b == 0.0)[0].tolist())
    rg = np.random.RandomState(rng_seed)
    
    for rnd in range(max_rounds):
        best_improve = 0.0
        best_pair = None
        
        if strategy == 'best':
            # Check all pairs
            for i in ones:
                for j in zeros:
                    delta = compute_delta(b, i, j, n)
                    new_max = np.max(P + delta)
                    improve = cur_max - new_max
                    if improve > best_improve + 1e-15:
                        best_improve = improve
                        best_pair = (i, j)
        else:
            # First improvement with random order
            idx_i = rg.permutation(len(ones))
            idx_j = rg.permutation(len(zeros))
            found = False
            for ii in idx_i:
                if found:
                    break
                i = ones[ii]
                for jj in idx_j:
                    j = zeros[jj]
                    delta = compute_delta(b, i, j, n)
                    new_max = np.max(P + delta)
                    if cur_max - new_max > 1e-15:
                        best_pair = (i, j)
                        best_improve = cur_max - new_max
                        found = True
                        break
        
        if best_pair is None:

        i, j = best_pair
        b[i] = 0.0
        b[j] = 1.0
        ones.remove(i)
        zeros.remove(j)
        ones.append(j)
        zeros.append(i)
        P = np.correlate(b, 1.0 - b, mode='full').astype(np.float64)
        cur_max = np.max(P)
    
    return b, cur_max * (2.0 / n)

        b[i1] = 0; b[i2] = 0
