```python
def active_set_optimize(h_init, n):
    """Solve max_k f_k(h) = t, minimize t, using active set."""
    h = h_init.copy()
    dx = 2.0 / n
    
    for iteration in range(100):
        # Compute all f_k
        profile = np.correlate(h, 1-h, mode='full') * dx
        t = np.max(profile)
        
        # Active set: lags where f_k is within tolerance of max
        active = np.where(profile >= t - 1e-6)[0]
        
        # For each active lag, compute gradient
        # f_k(h) = sum_i h[i](1-h[i+k]) (with appropriate index bounds)
        # ∂f_k/∂h[j] = (1-h[j+k]) if j+k is valid, minus h[j-k] if j-k is valid
        
        m = len(active)
        # Build the gradient matrix for active lags
        G = np.zeros((m, n))
        for idx, lag in enumerate(active):
            d = lag - (n-1)
            if d == 0:
                G[idx, :] = dx * (1 - 2*h)
            elif d > 0:
                G[idx, d:n] += dx * (1 - h[:n-d])
                G[idx, :n-d] += -dx * h[d:]
            else:
                dp = -d
                G[idx, :n-dp] += dx * (1 - h[dp:])
                G[idx, dp:] += -dx * h[:n-dp]
        
        # We want to find the direction that decreases t
        # The subgradient of max is in conv{∇f_k : k in active}
        # At the optimum, 0 should be in the convex hull of {∇f_k} + normal cone
        
        # Simple approach: minimize the max by moving in the direction
        # that reduces all active f_k's simultaneously
        # Use the direction: -G^T w where w >= 0, sum w = 1
        # Choose w to maximize the minimum decrease
        
        # For now, use uniform weights
        w = np.ones(m) / m
        grad = G.T @ w
        # Project out the null space of the sum constraint
        grad -= grad.mean()
        
        # Line search
        best_alpha = None
        best_t = t
        for alpha in np.geomspace(0.1, 0.001, 20):
            h_try = repair_feasible(h - alpha * grad)
            t_try = exact_score(h_try)
            if t_try < best_t:
                best_t = t_try
                best_alpha = alpha
                h = h_try
        
        if best_alpha is None:
            break
    
    return h, exact_score(h)
```

from scipy.optimize import minimize
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full")) * dx)

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx] + 1e-15)
            h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx] + 1e-15)
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def build_gradient_matrix(h, n, dx):
    M = 2 * n - 1
    G = np.zeros((M, n))
    for m in range(M):
        d = m - (n - 1)
        if d == 0:
            G[m, :] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            G[m, d:n] += dx * (1.0 - h[0:n-d])
            G[m, 0:n-d] += -dx * h[d:n]
        else:
            dp = -d
            G[m, 0:n-dp] += dx * (1.0 - h[dp:n])
            G[m, dp:n] += -dx * h[0:n-dp]
    return G

def solve_epigraph(h_start, maxiter=500, tol=1e-12):
    n = len(h_start)
    dx = 2.0 / n
    M = 2 * n - 1
    c0 = exact_score(h_start)
    t0 = c0 + 1e-5
    
    def obj(x):
        return x[-1]
    
    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    
    def eq_fun(x):
        return x[:-1].sum() - n / 2.0
    
    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g
    
    def ineq_fun(x):
        h = x[:-1]
        t = x[-1]
        profile = np.correlate(h, 1 - h, mode="full") * dx
        return t - profile
    
    def ineq_jac(x):
        h = x[:-1]
        G = build_gradient_matrix(h, n, dx)
        J = np.zeros((M, n + 1))
        J[:, :-1] = -G
        J[:, -1] = 1.0
        return J
    
    x0 = np.concatenate([h_start, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 2.0)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': tol})
        if res.x is not None and np.isfinite(res.x).all():
            h_new = repair_feasible(res.x[:-1])
            if h_new.sum() > 0:
                return h_new
    except Exception:
        pass
    return None

def active_weighted_descent(h, max_iters=100, tol=1e-10):
    """Active-set weighted descent with Newton-like steps."""
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best = exact_score(h)
    
    for it in range(max_iters):
        profile = compute_profile(h)
        c = profile.max()
        if c < best - 1e-13:
            best = c
        
        # Active set with soft threshold
        thresh = c - max(1e-5, 0.01 * c)
        active = np.where(profile >= thresh)[0]
        if len(active) == 0:
            active = np.array([np.argmax(profile)])
        
        # Weight by proximity to max
        w = (profile[active] - thresh)
        w = w / (w.sum() + 1e-15)
        
        G = build_gradient_matrix(h, n, dx)
        grad = G[active, :].T @ w
        grad -= grad.mean()  # project onto sum=0 subspace
        
        norm = np.linalg.norm(grad)
        if norm < 1e-14:
            break
        
        grad /= norm
        
        # Backtracking line search
        improved = False
        for alpha in [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]:
            h_try = repair_feasible(h - alpha * grad)
            s = exact_score(h_try)
            if s < best - 1e-13:
                h = h_try
                best = s
                improved = True
                break
        
        if not improved:
            break
    
    return h, best

import numpy as np
from scipy.optimize import minimize
import time


def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full")) * dx)


def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx] + 1e-15)
            h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx] + 1e-15)
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h


def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx


def build_gradient_matrix(h, n, dx):
    M = 2 * n - 1
    G = np.zeros((M, n))
    for m in range(M):
        d = m - (n - 1)
        if d == 0:
            G[m, :] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            G[m, d:n] += dx * (1.0 - h[0:n - d])
            G[m, 0:n - d] += -dx * h[d:n]
        else:
            dp = -d
            G[m, 0:n - dp] += dx * (1.0 - h[dp:n])
            G[m, dp:n] += -dx * h[0:n - dp]
    return G


def solve_epigraph(h_start, maxiter=500, tol=1e-13):
    n = len(h_start)
    dx = 2.0 / n
    M = 2 * n - 1
    c0 = exact_score(h_start)
    t0 = c0 + 1e-5

    def obj(x):
        return x[-1]

    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g

    def eq_fun(x):
        return x[:-1].sum() - n / 2.0

    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g

    def ineq_fun(x):
        h = x[:-1]
        t = x[-1]
        profile = np.correlate(h, 1 - h, mode="full") * dx
        return t - profile

    def ineq_jac(x):
        h = x[:-1]
        G = build_gradient_matrix(h, n, dx)
        J = np.zeros((M, n + 1))
        J[:, :-1] = -G
        J[:, -1] = 1.0
        return J

    x0 = np.concatenate([h_start, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 2.0)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': tol})
        if res.x is not None and np.isfinite(res.x).all():
            h_new = repair_feasible(res.x[:-1])
            if h_new.sum() > 1e-10:
                return h_new
    except Exception:
        pass
    return None


def smooth_max_solve(h_start, betas=None, maxiter=250, mu=800.0):
    n = len(h_start)
    dx = 2.0 / n
    h = h_start.copy()
    if betas is None:
        betas = [8.0, 15.0, 30.0, 60.0, 120.0, 250.0, 500.0]

    for beta in betas:
        def obj(x):
            x = np.clip(x, 0.0, 1.0)
            profile = compute_profile(x)
            shifted = profile - profile.max()
            exp_b = np.exp(beta * shifted)
            val = profile.max() + np.log(exp_b.sum()) / beta
            val += mu * (x.sum() - n / 2.0) ** 2
            return val

        def grad(x):
            x = np.clip(x, 0.0, 1.0)
            profile = compute_profile(x)
            shifted = profile - profile.max()
            exp_b = np.exp(beta * shifted)
            w = exp_b / exp_b.sum()
            G = build_gradient_matrix(x, n, dx)
            g = G.T @ w
            g += 2.0 * mu * (x.sum() - n / 2.0)
            return g

        try:
            res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                           bounds=[(0.0, 1.0)] * n,
                           options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-12})
            if res.x is not None and np.isfinite(res.x).all():

                c_try = exact_score(h_try)
                c_curr = exact_score(h)
                if c_try < c_curr - 1e-14:
                    h = h_try
        except Exception:
            pass
    return h


def active_weighted_descent(h, max_iters=150):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best = exact_score(h)
    for it in range(max_iters):
        profile = compute_profile(h)
        c = profile.max()
        if c < best - 1e-13:
            best = c
        thresh = c - max(1e-5, 0.005 * c)
        active = np.where(profile >= thresh)[0]
        if len(active) == 0:
            active = np.array([np.argmax(profile)])
        w = profile[active] - thresh
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.ones(len(active)) / len(active)
        else:
            w = w / w_sum
        G = build_gradient_matrix(h, n, dx)
        grad = G[active, :].T @ w
        grad -= grad.mean()
        norm = np.linalg.norm(grad)
        if norm < 1e-14:
            break
        grad /= norm
        improved = False
        for alpha in [0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]:
            h_try = repair_feasible(h - alpha * grad)
            s = exact_score(h_try)
            if s < best - 1e-13:
                h = h_try
                best = s
                improved = True
                break
        if not improved:
            break
    return h, best


def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)


def symmetrize(h):
    return (h + h[::-1]) / 2.0


def fourier_perturb(h, rng, scale=0.015, max_mode=None):
    n = len(h)
    if max_mode is None:
        max_mode = min(n // 5, 10)
    delta = np.zeros(n)
    for k in range(1, max_mode + 1):
        amp = rng.randn() * scale / np.sqrt(k)
        phase = rng.uniform(0, 2 * np.pi)
        delta += amp * np.cos(2 * np.pi * k * np.arange(n) / n + phase)
    return repair_feasible(h + delta)


def random_perturb(h, rng, scale=0.02):
    z = rng.randn(len(h))
    z -= z.mean()
    z *= scale / (np.std(z) + 1e-15)
    return repair_feasible(h + z)
