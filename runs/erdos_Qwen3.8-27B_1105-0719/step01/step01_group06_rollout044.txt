        return best_h, best_score

    bounds = [(0.0, 1.0)] * n + [(0.0, best_score + 2.0)]
    x0 = np.concatenate([h, [best_score + 1e-6]])
    mi = min(60, max(15, int((time_limit - elapsed) * 0.7)))

    try:
        result = minimize(
            lambda x: x[n], x0,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'ineq', 'fun': cons_fun, 'jac': cons_jac}],
            options={'maxiter': mi, 'ftol': 1e-11}
        )
        if result.x is not None and np.all(np.isfinite(result.x)):
            h_sol = repair_feasible(result.x[:n])
            s_sol = exact_score(h_sol)
            if s_sol < best_score - 1e-12:
                best_h = h_sol
                best_score = s_sol
    except Exception:
        pass

    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t_start = time.perf_counter()

    def tleft():
        return budget_s - (time.perf_counter() - t_start)

    # Get parent
    try:
        parent = np.array(initial_h_values, dtype=np.float64).copy()
    except Exception:
        parent = np.full(64, 0.5)

    parent = repair_feasible(parent)
    npn = len(parent)
    best_h = parent.copy()
    best_score = exact_score(parent)

    # Try symmetry transforms
    for transform_name in ['sym', 'comp']:
        if tleft() < 10:
            break
        if transform_name == 'sym':
            h_t = symmetrize(best_h)
        else:
            h_t = complement_symmetry(best_h)
        s_t = exact_score(h_t)
        if s_t < best_score - 1e-12:
            best_h = h_t
            best_score = s_t

    # Verify gradient matrix
    grad_ok = True
    if npn < 300:
        h_test = repair_feasible(np.random.rand(npn) * 0.5 + 0.25)
        G_test = get_lag_gradients(h_test)
        p_test = get_profile(h_test)
        eps = 1e-6
        j_test = np.random.randint(0, npn)
        e_j = np.zeros(npn)
        e_j[j_test] = eps
        h_plus = repair_feasible(h_test + e_j)
        p_plus = get_profile(h_plus)
        numerical = (p_plus - p_test) / eps
        if np.max(np.abs(numerical - G_test[:, j_test])) > 1e-3:
            grad_ok = False

    # Phase 1: LP Chebyshev on parent dimension (main)
    if grad_ok and tleft() > 40:
        alloc = min(tleft() * 0.35, 380.0)
        h_lp, s_lp = solve_lp_chebyshev(best_h, radius=0.25, active_count=None,
                                         max_iter=30, time_limit=alloc)
        if s_lp < best_score - 1e-12:
            best_h = h_lp
            best_score = s_lp

    # Phase 2: Multi-start perturbations with LP
    if grad_ok and tleft() > 60:
        n = len(best_h)
        n_starts = min(12, max(3, int(tleft() * 0.03)))
        for p in range(n_starts):
            if tleft() < 30:
                break
            eps_val = 0.005 + 0.004 * p
            z = np.random.randn(n)
            z -= z.mean()
            z /= (np.std(z) + 1e-10)
            h_pert = repair_feasible(best_h + eps_val * z)
            s_pert = exact_score(h_pert)
            alloc = min(40.0, tleft() * 0.12)
            h_sol, s_sol = solve_lp_chebyshev(h_pert, radius=0.15,
                                               active_count=min(80, 2 * n - 1),
                                               max_iter=15, time_limit=alloc)
            if s_sol < best_score - 1e-12:
                best_h = h_sol
                best_score = s_sol

    # Phase 3: Dimension exploration
    if tleft() > 80:
        pn = len(best_h)
        dims = []
        for scale in [0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 2.0]:
            d = int(round(pn * scale))
            if 32 <= d <= 240:
                d = max(32, d)
                if d % 2 == 1:
                    d += 1
                dims.append(d)
        dims = sorted(set(dims))

        for new_n in dims:
            if tleft() < 40:
                break
            h_interp = interpolate_to_dim(best_h, new_n)
            s_interp = exact_score(h_interp)
            if s_interp < best_score - 1e-12:
                best_h = h_interp
                best_score = s_interp
            if not grad_ok or new_n > 200:
                continue
            alloc = min(35.0, tleft() * 0.12)
            h_sol, s_sol = solve_lp_chebyshev(h_interp, radius=0.2,
                                               active_count=min(100, 2 * new_n - 1),
                                               max_iter=12, time_limit=alloc)
            if s_sol < best_score - 1e-12:
                best_h = h_sol
                best_score = s_sol

    # Phase 4: LP with smaller active set (faster per iteration)
    if grad_ok and tleft() > 30:
        n = len(best_h)
        for nc in [40, 60, 100]:
            if tleft() < 20:
                break
            if nc >= 2 * n - 1:
                continue
            alloc = min(25.0, tleft() * 0.15)
            h_sol, s_sol = solve_lp_chebyshev(best_h, radius=0.12,
                                               active_count=nc, max_iter=10,
                                               time_limit=alloc)
            if s_sol < best_score - 1e-12:
                best_h = h_sol
                best_score = s_sol

    # Phase 5: SLSQP polish
    if tleft() > 25:
        alloc = min(30.0, tleft() * 0.25)
        h_pol, s_pol = slsqp_polish(best_h, time_limit=alloc)
        if s_pol < best_score - 1e-12:
            best_h = h_pol
            best_score = s_pol

    # Phase 6: Final local descent
    if tleft() > 10:
        alloc = min(12.0, tleft() * 0.4)
        h_loc, s_loc = local_descent(best_h, max_iters=25, time_limit=alloc)
        if s_loc < best_score - 1e-12:
            best_h = h_loc
            best_score = s_loc

    # Final repair and score
    best_h = repair_feasible(best_h)
    final_score = exact_score(best_h)

    return (best_h, final_score, len(best_h))
