Now, for the coordinate descent:

At each step:
1. Compute profile and identify max lags
2. For the max lags, compute the gradient
3. Find the pair (i,j) that most reduces the max: we want to minimize max_m (profile[m] + λ*(grad_m[i] - grad_m[j]))
4. For the best pair, find optimal λ by 1D search
5. Apply the move

For step 3, the optimal λ for pair (i,j) in the direction that reduces the most active lags:
- We want d = grad_m[i] - grad_m[j] < 0 for active m (so increasing λ helps)
- Or d > 0 (so decreasing λ helps, i.e., we move in the -λ direction)

The optimal λ (if the quadratic terms are negligible) is where the active profiles cross zero improvement. But since we're doing exact 1D search anyway, we just need to identify the best (i,j) pair.

A simple heuristic: for each active lag m, the "best" pair is one where grad_m[i] - grad_m[j] is most negative (or most positive, depending on direction). We can rank pairs by their "improvement potential" = -max_active grad_m[i] + max_active grad_m[j] (roughly).

Actually, let me simplify. For a given direction (increasing h[i], decreasing h[j]), the first-order change in the objective is approximately:
ΔC ≈ λ * max_m (grad_m[i] - grad_m[j])  (for λ > 0, small)

We want this to be negative. So we want max_m (grad_m[i] - grad_m[j]) < 0.

Equivalently: we want i to be a position where the gradient is low and j where it's high (for the active lags).

Let me define for each position k: g[k] = max_m grad_m[k] (over active lags m).

        h[i] += t
        h[j] -= t
        vc = float(np.max(compute_profile(h)))
        if vc < best_c:
            best_c = vc
            best_lambda = t
        h[i] -= t
        h[j] += t
    return best_lambda, best_c

def coordinate_descent(h_in, deadline, max_rounds=300, rng=None, tol=1e-8):
    if rng is None:
        rng = np.random.RandomState(0)
    n = len(h_in)
    target = 0.5 * n
    h = project_box_sum(h_in, target).copy()
    c, profile = exact_objective(h)
    no_improve = 0
    for rnd in range(max_rounds):
        if time.time() > deadline - 3:
            break
        max_c = float(np.max(profile))
        active_thresh = max_c - 0.002
        active = np.where(profile >= active_thresh)[0]
        g = np.zeros(n)
        for m in active:
            r = m - (n - 1)
            g = np.maximum(g, lag_gradient(h, r))
        order = np.argsort(g)
        best_c = c
        best_lambda = 0.0
        best_i = -1
        best_j = -1
        n_cand = min(30, n // 2)
        tried = set()
        for a in range(n_cand):
            i = int(order[a])
            for b in range(n - n_cand, n):
                j = int(order[b])
                if i == j:
                    continue
                key = i * n + j
                if key in tried:
                    continue
                tried.add(key)
                lam, vc = pair_line_search(h, i, j, best_c, n_coarse=20, n_fine=35)
                if vc < best_c - 1e-12:
                    best_c = vc
                    best_lambda = lam
                    best_i, best_j = i, j
        for _ in range(40):
            i = int(rng.randint(n))
            j = int(rng.randint(n))
            if i == j:
                continue
            key = i * n + j
            if key in tried:
                continue
            tried.add(key)
            lam, vc = pair_line_search(h, i, j, best_c, n_coarse=15, n_fine=25)
            if vc < best_c - 1e-12:
                best_c = vc
                best_lambda = lam
                best_i, best_j = i, j
        if best_i == -1:
            no_improve += 1
            if no_improve > 5:
                k = max(3, n // 8)
                idx = rng.choice(n, size=k, replace=False)
                h[idx] += rng.randn(k) * (0.02 + 0.01 * no_improve)
                h = project_box_sum(h, target)
                new_c, new_p = exact_objective(h)
                if new_c > c + 0.005:
                    break
                c, profile = new_c, new_p
                no_improve = 0
            elif no_improve > 2:
                k = max(2, n // 12)
                idx = rng.choice(n, size=k, replace=False)
                h[idx] += rng.randn(k) * 0.015
                h = project_box_sum(h, target)
                c, profile = exact_objective(h)
                no_improve = 0
            continue
        h[best_i] += best_lambda
        h[best_j] -= best_lambda
        h = project_box_sum(h, target)
        c, profile = exact_objective(h)
        if c < best_c - 1e-10:
            best_c = c
        no_improve = 0
        if rnd > 0 and abs(c - profile.max()) > 1e-10:
            c = float(profile.max())
    return h, c

def downsample(h, new_n):
    n = len(h)
    if new_n == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(new_n) + 0.5) / new_n
    vals = np.interp(x_new, x_old, h)
    return project_box_sum(vals, 0.5 * new_n)

def binary_threshold(h):
    n = len(h)
    if n % 2 != 0:
        return None
    target = n // 2
    idx = np.argsort(-h)
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b

def binary_local_search(b, max_rounds=100, rng=None):
    if rng is None:
        rng = np.random.RandomState(0)
    n = len(b)
    c, _ = exact_objective(b)
    for rnd in range(max_rounds):
        if time.time() > deadline_global - 5:
            break
        improved = False
        ones = list(np.where(b > 0.5)[0])
        zeros = list(np.where(b < 0.5)[0])
        n_trial = min(200, len(ones) * len(zeros))
        for _ in range(n_trial):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            b[i], b[j] = 0.0, 1.0
            nc, _ = exact_objective(b)
            if nc < c - 1e-12:
                c = nc
                improved = True
                ones = list(np.where(b > 0.5)[0])
                zeros = list(np.where(b < 0.5)[0])
            else:
                b[i], b[j] = 1.0, 0.0
        if not improved:
            break
    return b, c

def make_structured_start(n, mode, rng):
    target = 0.5 * n
    if mode == 0:
        h = np.zeros(n)
        h[::2] = 1.0
    elif mode == 1:
        h = np.zeros(n)
        h[:int(target)] = 1.0
    elif mode == 2:
        x = np.arange(n, dtype=np.float64)
        f = 1.0 + rng.randint(1, 5)
        ph = rng.uniform(0, 2 * np.pi)
        h = 0.5 + 0.5 * np.cos(2 * np.pi * f * x / n + ph)
    elif mode == 3:
        x = np.arange(n, dtype=np.float64)
        f1 = 1.0 + rng.randint(1, 4)
        f2 = 2.0 + rng.randint(1, 5)
        p1 = rng.uniform(0, 2 * np.pi)
        p2 = rng.uniform(0, 2 * np.pi)
        a1 = rng.uniform(0.3, 0.5)
        a2 = rng.uniform(0.1, 0.3)
        h = 0.5 + a1 * np.cos(2 * np.pi * f1 * x / n + p1) + a2 * np.cos(2 * np.pi * f2 * x / n + p2)
    elif mode == 4:
        h = np.zeros(n)
        q = n // 4
        h[:q] = 1.0
        h[(n // 2):(n // 2 + q)] = 1.0
    else:
        h = np.full(n, 0.5)
        h += rng.randn(n) * 0.2
    return project_box_sum(h, target)

deadline_global = 0

def run(seed=42, budget_s=1100, **kwargs):
    global deadline_global
    start_time = time.time()
    deadline_global = start_time + budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c = float('inf')

    # Load initial
    try:
        init = np.asarray(initial_h_values, dtype=np.float64)
        if init.size > 0 and np.all(np.isfinite(init)):
            n0 = len(init)
            init = project_box_sum(init, 0.5 * n0)
            c0, _ = exact_objective(init)
            best_h = init.copy()
            best_c = c0
    except (NameError, Exception):
        pass

    if best_h is None:
        n0 = 70
        best_h = make_structured_start(n0, 0, rng)
        best_c, _ = exact_objective(best_h)

    n0 = len(best_h)
    time_phase1 = budget_s * 0.45

    # Phase 1: Coordinate descent on initial size
    h1, c1 = coordinate_descent(best_h, start_time + time_phase1, max_rounds=400, rng=rng)
    if c1 < best_c:
        best_c, best_h = c1, h1

    # Phase 2: Binary projection + local search
    if len(best_h) % 2 == 0 and time.time() < deadline_global - 200:
        b = binary_threshold(best_h)
        if b is not None:
            b, c_b = binary_local_search(b, max_rounds=80, rng=rng)
            if c_b < best_c:
                best_c, best_h = c_b, b
                # Refine continuous from binary
                h_ref = b.astype(np.float64)
                h_ref, c_ref = coordinate_descent(h_ref, start_time + time_phase1 + 80, max_rounds=150, rng=rng)
                if c_ref < best_c:
                    best_c, best_h = c_ref, h_ref

    # Phase 3: Try different n values
    time_phase3 = budget_s * 0.65
    n_candidates = [n for n in [56, 60, 64, 66, 68, 70, 72, 74, 76, 80, 84] if n != n0]
    n_candidates.sort(key=lambda x: abs(x - n0))

    for n_try in n_candidates:
        if time.time() > start_time + time_phase3:
            break
        h_ds = downsample(best_h, n_try)
        h_opt, c_opt = coordinate_descent(h_ds, start_time + time_phase3, max_rounds=200, rng=rng)
        if c_opt < best_c:
            best_c, best_h = c_opt, h_opt
        # Also try structured start
        for mode in [2, 3]:
            if time.time() > start_time + time_phase3:
                break
            h_s = make_structured_start(n_try, mode, rng)
            h_opt2, c_opt2 = coordinate_descent(h_s, start_time + time_phase3, max_rounds=100, rng=rng)
            if c_opt2 < best_c:
                best_c, best_h = c_opt2, h_opt2

    # Phase 4: Intensive coordinate descent on best
    h4, c4 = coordinate_descent(best_h, deadline_global - 40, max_rounds=400, rng=rng)
    if c4 < best_c:
        best_c, best_h = c4, h4

    # Phase 5: Multiple small perturbations + refinement
    for attempt in range(8):
        if time.time() > deadline_global - 15:
            break
        scale = 0.005 + 0.005 * attempt
        h_pert = best_h + rng.randn(len(best_h)) * scale
        h_pert = project_box_sum(h_pert)
        h_opt, c_opt = coordinate_descent(h_pert, deadline_global - 10, max_rounds=100, rng=rng)
        if c_opt < best_c:
            best_c, best_h = c_opt, h_opt

    # Phase 6: Final polish with very fine line search
    if time.time() < deadline_global - 10:
        n = len(best_h)
        target = 0.5 * n
        h_final = project_box_sum(best_h, target)
        c_final, profile = exact_objective(h_final)
        for _ in range(50):
            if time.time() > deadline_global - 5:
                break
            max_c = float(np.max(profile))
            active = np.where(profile >= max_c - 0.001)[0]
            g = np.zeros(n)
            for m in active:
                r = m - (n - 1)
                g = np.maximum(g, lag_gradient(h_final, r))
            order = np.argsort(g)
            improved = False
            for a in range(min(20, n // 2)):
                i = int(order[a])
                for b in range(n - min(20, n // 2), n):
                    j = int(order[b])
                    if i == j:
                        continue
                    lam, vc = pair_line_search(h_final, i, j, c_final, n_coarse=30, n_fine=50)
                    if vc < c_final - 1e-12:
                        h_final[i] += lam
                        h_final[j] -= lam
                        h_final = project_box_sum(h_final, target)
                        c_final, profile = exact_objective(h_final)
                        improved = True
                        break
                if improved:
                    break
            if not improved:
                break
        if c_final < best_c:
            best_c, best_h = c_final, h_final

    # Ensure validity
    best_h = project_box_sum(best_h, 0.5 * len(best_h))
    final_c, _ = exact_objective(best_h)
    best_c = final_c
    best_n = len(best_h)
    return (best_h, best_c, best_n)
