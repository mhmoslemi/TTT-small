```python
    # B[j][s] = change in raw at lag s from flipping h[j] from 0 to 1
    # s ranges from -(n-1) to (n-1), stored as index s + (n-1) in [0, 2n-2]
    
    A = np.zeros((n, 2*n-1))  # A[i] for all i (only used for i in ones)
    B = np.zeros((n, 2*n-1))  # B[j] for all j (only used for j in zeros)
    
    for i in range(n):
        if h[i] == 1:
            for s in range(1, n):
                # s > 0
                t = s + (n-1)
                if i >= s:
                    A[i, t] -= (1 - h[i-s])
                if i <= n-1-s:
                    A[i, t] += h[i+s]
                # s < 0
                t2 = -s + (n-1)
                if i <= n-1-s:
                    A[i, t2] -= (1 - h[i+s])
                if i >= s:
                    A[i, t2] += h[i-s]
    
    for j in range(n):
        if h[j] == 0:
            for s in range(1, n):
                t = s + (n-1)
                if j >= s:
                    B[j, t] += (1 - h[j-s])
                if j <= n-1-s:
                    B[j, t] -= h[j+s]
                t2 = -s + (n-1)
                if j <= n-1-s:
                    B[j, t2] += (1 - h[j+s])
                if j >= s:
                    B[j, t2] -= h[j-s]
    
    return A, B
```

This is O(n²) which for n=140 is 19600 inner iterations, each doing O(1) work. Fast.
