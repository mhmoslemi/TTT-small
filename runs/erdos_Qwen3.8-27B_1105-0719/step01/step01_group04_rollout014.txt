def weighted_gradient_weights(h, weights):
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w == 0.0:
            continue
        delta = k - (n - 1)
        if delta >= 0:
            j1 = np.arange(0, n - delta)
            grad[j1] += w * dx * (1.0 - h[j1 + delta])
            if delta > 0:

                grad[j2] += -w * dx * h[j2 - delta]

            j1 = np.arange(m, n)
            grad[j1] += w * dx * (1.0 - h[j1 - m])
            j2 = np.arange(0, n - m)
            grad[j2] += -w * dx * h[j2 + m]

    proj = np.sum(sigma_prime * G_h) / T

        if score_ref < best_score - 1e-13:
            best_h = h_ref
            best_score = score_ref

    # Final validation
    best_h = repair_feasible(best_h)
    best_score = exact_score(best_h)
    return best_h, best_score, len(best_h)
