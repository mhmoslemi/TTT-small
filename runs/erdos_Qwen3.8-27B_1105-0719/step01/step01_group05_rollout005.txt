def project_box_sum(h, target=None):
    n = len(h)
    if target is None:
        target = 0.5 * n
    h = np.array(h, dtype=np.float64).copy()
    h = np.nan_to_num(h, nan=0.0, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            count = np.sum(mask)
            if count == 0:
                break
            step = diff / count
            h[mask] = np.minimum(h[mask] + step, 1.0)
        else:
            mask = h > 0.0
            count = np.sum(mask)
            if count == 0:
                break
            step = diff / count
            h[mask] = np.maximum(h[mask] + step, 0.0)
    # Final correction
    diff = target - np.sum(h)

        for i in range(n):

                break
            elif diff < 0 and h[i] > 0.0:
                h[i] = max(0.0, h[i] + diff)
                break

def compute_swap_delta(profile, b, a, b_idx, n):
    """
    Compute the new profile after swapping positions a (1→0) and b_idx (0→1).
    profile: current raw profile (length 2n-1)
    a: position where b goes from 1 to 0
    b_idx: position where b goes from 0 to 1
    """
    new_profile = profile.copy()
    # For each shift t (lag = m - (n-1)), compute the change
    
    # Position a was 1, now 0:
    # As "first" element: h[a]*(1-h[a+t]) was 1*(1-h[a+t]), now 0. Change: -(1-h[a+t])
    # As "second" element: h[a-t]*(1-h[a]) was h[a-t]*0=0, now h[a-t]*1. Change: +h[a-t]
    
    # Position b_idx was 0, now 1:
    # As "first" element: h[b_idx]*(1-h[b_idx+t]) was 0, now 1*(1-h[b_idx+t]). Change: +(1-h[b_idx+t])
    # As "second" element: h[b_idx-t]*(1-h[b_idx]) was h[b_idx-t]*1, now 0. Change: -h[b_idx-t]
    
    # For lag t (t = m - (n-1), m = 0..2n-2):
    # prof[m] = sum_i h[i] * (1-h[i+t]) where valid indices
    
    # Change from position a (1→0):
    #   As first: if 0 <= a+t < n, change = -(1 - b[a+t])
    #   As second: if 0 <= a-t < n, change = +b[a-t] (since 1-h[a] goes from 0 to 1)
    
    # Change from position b_idx (0→1):
    #   As first: if 0 <= b_idx+t < n, change = +(1 - b[b_idx+t])
    #   As second: if 0 <= b_idx-t < n, change = -b[b_idx-t] (since 1-h[b_idx] goes from 1 to 0)
    
    # t ranges from -(n-1) to (n-1)
    # m = t + (n-1) ranges from 0 to 2n-2
    
    for t in range(-(n-1), n):
        m = t + (n - 1)
        delta = 0.0
        # Position a: first element contribution
        if 0 <= a + t < n:
            delta -= (1.0 - b[a + t])
        # Position a: second element contribution (h[a-t] * (1-h[a]), 1-h[a] changes from 0 to 1)
        if 0 <= a - t < n:

        # Position b_idx: first element contribution (h[b_idx] changes from 0 to 1)
        if 0 <= b_idx + t < n:
            delta += (1.0 - b[b_idx + t])
        # Position b_idx: second element contribution (1-h[b_idx] changes from 1 to 0)
        if 0 <= b_idx - t < n:
            delta -= b[b_idx - t]
        
        new_profile[m] += delta
    
    return new_profile
