def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def compute_all_lags(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr  # length 2n-1

def grad_of_lag(h, d):
    """Gradient of f_d(h) where d is the lag offset (d = m - (n-1))"""
    n = len(h)
    grad = np.zeros(n)
    # f_d = sum_k h[k] * (1-h[k-d]) over valid k
    # ∂f_d/∂h[j] = (1-h[j-d]) * [valid] - h[j+d] * [valid]
    for j in range(n):
        val = 0.0
        if 0 <= j - d < n:
            val += (1.0 - h[j - d])
        if 0 <= j + d < n:
            val -= h[j + d]
        grad[j] = val
    return grad

def grad_of_lag_vec(h, d):
    """Vectorized gradient of f_d(h)"""
    n = len(h)
    grad = np.zeros(n)
    # Term 1: (1-h[j-d]) for valid j-d
    lo1 = max(0, d)
    hi1 = min(n, n + d)
    if lo1 < hi1:
        grad[lo1:hi1] += (1.0 - h[lo1-d:hi1-d])
    # Term 2: -h[j+d] for valid j+d
    lo2 = max(0, -d)
    hi2 = min(n, n - d)
    if lo2 < hi2:
        grad[lo2:hi2] -= h[lo2+d:hi2+d]
    return grad

    """Project y onto {h: 0<=h<=1, sum(h)=n/2}"""
    target = n / 2.0
    lo = np.min(y) - 2.0
    hi = np.max(y) + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    h = np.clip(y - t, 0.0, 1.0)
    # Fix residual
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        # Distribute among interior points
        mask = (h > 1e-8) & (h < 1.0 - 1e-8)
        if np.sum(mask) > 0:
            h[mask] += residual / np.sum(mask)
            h = np.clip(h, 0.0, 1.0)
        else:
            # All at boundary, find one to adjust
            if residual > 0:
                idx = np.argmin(h)
                h[idx] = min(1.0, h[idx] + residual)
            else:
                idx = np.argmax(h)
                h[idx] = max(0.0, h[idx] + residual)
    return h

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def grad_lag(h, d, n):
    grad = np.zeros(n)
    if d == 0:
        grad[:] = 1.0 - 2.0 * h
    elif d > 0:
        if d < n:
            grad[d:] += (1.0 - h[:n - d])
            grad[:n - d] -= h[d:]
    else:
        dd = -d
        if dd < n:
            grad[:n - dd] += (1.0 - h[dd:])
            grad[dd:] -= h[:n - dd]
    return grad

def project_feasible(y, n):
    target = n / 2.0
    y = np.asarray(y, dtype=float)
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y)) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    h = np.clip(y - t, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-10:
        mask = (h > 1e-8) & (h < 1.0 - 1e-8)
        count = int(np.sum(mask))
        if count > 0:
            h[mask] += residual / count
            h = np.clip(h, 0.0, 1.0)
    return h

def make_binary(h, n):
    h_bin = (h >= 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    target = n // 2
    if n_ones > target:
        ones = np.where(h_bin == 1.0)[0]
        order = ones[np.argsort(h[ones])]
        for i in range(n_ones - target):
            h_bin[order[i]] = 0.0
    elif n_ones < target:
        zeros = np.where(h_bin == 0.0)[0]
        order = zeros[np.argsort(-h[zeros])]
        for i in range(target - n_ones):
            h_bin[order[i]] = 1.0
    return h_bin

def binary_ls(h_bin, n, budget, rng, tabu_window=30):
    best = h_bin.copy()
    best_c5 = compute_c5(best)
    t0 = time.time()
    tabu = {}
    tick = 0
    while time.time() - t0 < budget:
        ones = np.where(best == 1.0)[0]
        zeros = np.where(best == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        key = int(min(i, j)) * n + int(max(i, j))
        if key in tabu and tabu[key] > tick - tabu_window:
            continue
        tabu[key] = tick
        tick += 1
        if tick % 5000 == 0 and len(tabu) > 20000:
            tabu = {k: v for k, v in tabu.items() if v > tick - tabu_window}
        new_h = best.copy()
        new_h[i] = 0.0
        new_h[j] = 1.0
        new_c5 = compute_c5(new_h)
        if new_c5 < best_c5 - 1e-15:
            best = new_h
            best_c5 = new_c5
    return best, best_c5

def pgd_optimize(h0, budget, seed=42):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t0 = time.time()
    dx = 2.0 / n
    alpha = 0.05
    momentum = np.zeros(n)
    beta_m = 0.9
    stall = 0

    for it in range(200000):
        if time.time() - t0 > budget:
            break
        corr = compute_corr(h)
        c5 = float(np.max(corr)) * dx
        if c5 < best_c5 - 1e-15:
            best_c5 = c5
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 5000 and alpha < 1e-5:
            break
        max_val = float(np.max(corr))
        active = np.where(corr >= max_val - 1e-9)[0]
        grad = np.zeros(n)
        for m in active:
            d = int(m - (n - 1))
            grad += grad_lag(h, d, n)
        grad /= len(active)
        momentum = beta_m * momentum - alpha * grad
        improved = False
        for _ in range(30):
            h_new = project_feasible(h + momentum, n)
            c5_new = compute_c5(h_new)
            if c5_new <= c5 + 1e-14:
                h = h_new
                improved = True
                break
            alpha *= 0.5
        if not improved:
            alpha *= 0.5
            if alpha < 1e-8:
                break
        alpha *= 0.998
    return best_h, best_c5

def pair_transfer_optimize(h0, budget, rng):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t0 = time.time()
    eps = 1e-4

    for it in range(500000):
        if time.time() - t0 > budget:
            break
        corr = compute_corr(h)
        c5 = float(np.max(corr))
        dx = 2.0 / n
        c5_scaled = c5 * dx
        if c5_scaled < best_c5 - 1e-15:
            best_c5 = c5_scaled
            best_h = h.copy()
        max_val = float(np.max(corr))
        active = np.where(corr >= max_val - 1e-9)[0]
        grad = np.zeros(n)
        for m in active:
            d = int(m - (n - 1))
            grad += grad_lag(h, d, n)
        grad /= len(active)
        i_candidates = np.argsort(grad)[-min(20, n):][::-1]
        j_candidates = np.argsort(grad)[min(20, n):]
        best_move = None
        best_score = c5
        for idx_i in i_candidates[:8]:
            for idx_j in j_candidates[:8]:
                if idx_i == idx_j:
                    continue
                for frac in [0.02, 0.05, 0.1, 0.2, 0.5, 1.0]:
                    tau = frac * min(h[idx_j], 1.0 - h[idx_i])
                    if tau < 1e-8:
                        continue
                    h_new = h.copy()
                    h_new[idx_i] += tau
                    h_new[idx_j] -= tau
                    h_new = np.clip(h_new, 0.0, 1.0)
                    res = n / 2.0 - np.sum(h_new)
                    if abs(res) > 1e-10:
                        h_new = project_feasible(h_new, n)
                    s = compute_c5(h_new)
                    if s < best_score:
                        best_score = s
                        best_move = h_new
        if best_move is not None:
            h = best_move
        else:
            if it % 100 == 0:
                i = rng.randint(n)
                j = rng.randint(n)
                if i != j:
                    tau = 0.01 * rng.uniform(0.1, 1.0)
                    h_new = h.copy()
                    h_new[i] += tau
                    h_new[j] -= tau
                    h_new = np.clip(h_new, 0.0, 1.0)
                    h_new = project_feasible(h_new, n)
                    s = compute_c5(h_new)
                    if s <= c5_scaled + 0.01 * max(0.01, c5_scaled) * (0.5 + 0.5 * rng.random()):
                        h = h_new
    return best_h, best_c5

def interpolate_resolution(h_old, n_new):
    n_old = len(h_old)
    x_old = np.arange(n_old)
    x_new = np.linspace(0, n_old - 1, n_new)
    h_interp = np.interp(x_new, x_old, h_old)
    return project_feasible(h_interp, n_new)

def smooth_max_obj(h, beta):
    corr = compute_corr(h)
    m = np.max(corr)
    shifted = beta * (corr - m)
    exps = np.exp(shifted)
    s = np.sum(exps)
    return (m + np.log(s) / beta) * (2.0 / len(h))

def smooth_max_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    m = np.max(corr)
    shifted = beta * (corr - m)
    exps = np.exp(shifted)
    s = np.sum(exps)
    weights = exps / s
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        k = m_idx - (n - 1)
        w = weights[m_idx]
        if w < 1e-14:
            continue
        lo1 = max(0, k)
        hi1 = min(n, n + k)
        if lo1 < hi1:
            grad[lo1:hi1] += w * (1.0 - h[lo1 - k:hi1 - k])
        lo2 = max(0, -k)
        hi2 = min(n, n - k)
        if lo2 < hi2:
            grad[lo2:hi2] -= w * h[lo2 + k:hi2 + k]
    return grad * dx

def slsqp_refine(h0, betas, budget, max_iter=300):
    n = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    t0 = time.time()
    bounds = [(0.0, 1.0)] * n

    def eq_fun(x):
        return np.sum(x) - n / 2.0

    def eq_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    for beta in betas:
        if time.time() - t0 > budget - 5:
            break
        res = minimize(
            smooth_max_obj, best_h, args=(beta,),
            jac=smooth_max_grad,
            method='SLSQP', bounds=bounds, constraints=cons,
            options={'maxiter': max_iter, 'ftol': 1e-15}
        )
        if res.x is not None:
            res_x = project_feasible(res.x, n)
            c5_new = compute_c5(res_x)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = res_x
    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t_start = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float)
        n_parent = len(h_parent)
    else:
        h_parent = np.full(60, 0.5)
        n_parent = 60

    h_parent = project_feasible(h_parent, n_parent)
    best_h = h_parent.copy()
    best_c5 = compute_c5(best_h)
    best_n = n_parent

    def elapsed():
        return time.time() - t_start

    def remaining():
        return budget_s - elapsed()

    # Phase 1: PGD on parent resolution
    if remaining() > 60:
        h_opt, c5_opt = pgd_optimize(best_h, min(80, remaining() - 20), seed)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 2: Binary local search on parent
    if remaining() > 40:
        h_bin = make_binary(best_h, best_n)
        h_b, c5_b = binary_ls(h_bin, best_n, min(40, remaining() - 10), rng)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b

    # Phase 3: SLSQP refine on parent
    if remaining() > 30:
        h_opt, c5_opt = slsqp_refine(best_h, [50, 200, 1000], min(30, remaining() - 10))
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 4: Pair transfer optimization
    if remaining() > 60:
        h_opt, c5_opt = pair_transfer_optimize(best_h, min(80, remaining() - 20), rng)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 5: Try different resolutions
    resolutions = [20, 24, 28, 30, 32, 36, 40, 44, 48, 52, 56, 60, 64, 72, 80]
    for n_try in resolutions:
        if remaining() < 40:
            break
        if n_try == best_n:
            continue
        h_interp = interpolate_resolution(best_h, n_try)
        budget_here = min(25, remaining() - 15)
        h_opt, c5_opt = pgd_optimize(h_interp, budget_here // 2, seed + n_try)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
            best_n = n_try
        if remaining() > 30:
            h_bin = make_binary(h_opt, n_try)
            h_b, c5_b = binary_ls(h_bin, n_try, min(15, remaining() - 15), rng)
            if c5_b < best_c5:
                best_c5 = c5_b
                best_h = h_b
                best_n = n_try
        if remaining() > 25:
            h_opt2, c5_opt2 = slsqp_refine(best_h, [100, 500], min(15, remaining() - 10))
            if c5_opt2 < best_c5:
                best_c5 = c5_opt2
                best_h = h_opt2
                best_n = len(h_opt2)

    # Phase 6: Random restarts with diverse strategies
    n_current = len(best_h)
    max_restarts = min(20, int(remaining() / 25))
    for r in range(max_restarts):
        if remaining() < 20:
            break
        strat = r % 6
        if strat == 0:
            h_pert = best_h + 0.05 * rng.randn(n_current)
        elif strat == 1:
            h_pert = best_h.copy()
            mask = rng.random(n_current) < 0.15
            h_pert[mask] = 1.0 - h_pert[mask]
        elif strat == 2:
            h_pert = np.full(n_current, 0.5)
            h_pert += 0.3 * rng.randn(n_current)
        elif strat == 3:
            h_pert = (best_h > 0.5).astype(float)
            idx1 = rng.choice(n_current, size=2, replace=False)
            h_pert[idx1[0]] = 1.0
            h_pert[idx1[1]] = 0.0
        elif strat == 4:
            shift = rng.randint(1, n_current)
            h_pert = np.roll(best_h, shift)
        else:
            k = rng.choice([1, 2, 3, 4])
            h_pert = 0.5 + 0.5 * np.sin(2.0 * np.pi * k * np.arange(n_current) / n_current + rng.uniform(0, 2 * np.pi))

        h_pert = project_feasible(h_pert, n_current)
        budget_here = min(20, remaining() - 10)
        h_opt, c5_opt = pgd_optimize(h_pert, budget_here // 2, seed + 1000 + r)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
            best_n = n_current
        if remaining() > 15:
            h_opt2, c5_opt2 = pair_transfer_optimize(h_opt, min(10, remaining() - 10), rng)
            if c5_opt2 < best_c5:
                best_c5 = c5_opt2
                best_h = h_opt2
                best_n = n_current

    # Phase 7: Final intensive local search
    if remaining() > 20:
        n_f = len(best_h)
        h_bin = make_binary(best_h, n_f)
        h_b, c5_b = binary_ls(h_bin, n_f, min(20, remaining() - 5), rng)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b

    # Phase 8: Final PGD polish
    if remaining() > 10:
        h_opt, c5_opt = pgd_optimize(best_h, min(10, remaining() - 2), seed + 99999)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Final verification
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
