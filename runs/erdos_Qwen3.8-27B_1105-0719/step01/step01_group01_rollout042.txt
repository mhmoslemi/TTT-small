def grad_shift(h, k):
    n = len(h)
    g = np.zeros(n)
    if k > 0:
        g[:n-k] += 1.0 - h[k:]
        g[k:] -= h[:n-k]
    elif k < 0:
        jk = -k
        g[jk:] += 1.0 - h[:n-jk]
        g[:n-jk] -= h[jk:]
    else:
        g[:] = 1.0 - 2.0 * h

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta) * dx

def smooth_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    e = np.exp(beta * (corr - m))
    w = e / np.sum(e)
    g = np.zeros(n)

        wi = w[i]
        if wi < 1e-14:
            continue
        k = i - (n - 1)
        g += wi * grad_shift(h, k)
    return g * dx

def eq_fun(h):
    return np.sum(h) - len(h) / 2.0

def eq_jac(h):
    return np.ones(len(h))

def slsqp_opt(h0, betas, max_iter=200):

            if c < best_c5:

        # The gradient of C5 = max(corr * dx) wrt h is (dx) * grad of argmax corr
        # So g should be multiplied by dx
        g *= dx
        
        h_new = project_feasible(h - alpha * g, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best = h_new.copy()
            no_improve = 0
            alpha = min(alpha * 1.02, 1.0)
        else:
            no_improve += 1
            if no_improve > 100:
                h = best.copy()
                alpha *= 0.7
                no_improve = 0
        alpha *= 0.99995
        h = h_new
    
    return best, best_c5

def anneal_opt(h0, n_iters=20000, T0=0.01, T1=0.0001, seed=42):

    n = len(h0)
    h = h0.copy()
    best = h.copy()
    best_c5 = compute_c5(h)
    cs = corr_values(h)
    
    for it in range(n_iters):
        frac = it / n_iters
        T = T0 * (T1 / T0) ** frac
        
        # Smart move: target worst shift
        k_star = int(np.argmax(cs)) - (n - 1)
        
        # Try to find a pair that helps the worst shift
        # We want to move mass from high-h to low-h at positions
        # that reduce overlap at k_star
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        
        h_try = h.copy()
        # Swap
        h_try[i], h_try[j] = h[j], h[i]
        h_try = project_feasible(h_try, n)
        
        c5_try = compute_c5(h_try)
        delta = c5_try - best_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            h = h_try
            if c5_try < best_c5:
                best_c5 = c5_try
                best = h_try.copy()
    
    return best, best_c5

import numpy as np
from scipy.optimize import minimize, dual_annealing
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def corr_arr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx


def project_feasible(y, n):
    target = n / 2.0
    lo, hi = -3.0, 3.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(y - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    h = np.clip(y - t, 0.0, 1.0)
    r = target - np.sum(h)
    if abs(r) > 1e-12:
        idx = np.where((h > 1e-12) & (h < 1.0 - 1e-12))[0]
        if len(idx) > 0:
            h[idx] = np.clip(h[idx] + r / len(idx), 0.0, 1.0)
    return h


def grad_shift(h, k):
    n = len(h)
    g = np.zeros(n)
    if k > 0:
        g[:n - k] += 1.0 - h[k:]
        g[k:] -= h[:n - k]
    elif k < 0:
        jk = -k
        g[jk:] += 1.0 - h[:n - jk]
        g[:n - jk] -= h[jk:]
    else:
        g[:] = 1.0 - 2.0 * h
    return g


def smooth_obj(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta) * dx


def smooth_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    e = np.exp(beta * (corr - m))
    w = e / np.sum(e)
    g = np.zeros(n)
    for i in range(2 * n - 1):
        wi = w[i]
        if wi < 1e-14:
            continue
        k = i - (n - 1)
        g += wi * grad_shift(h, k)
    return g * dx


def eq_fun(h):
    return np.sum(h) - len(h) / 2.0


def eq_jac(h):
    return np.ones(len(h))


def slsqp_opt(h0, betas, max_iter=250):
    n = len(h0)
    best = h0.copy()
    best_c5 = compute_c5(best)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    for b in betas:
        res = minimize(smooth_obj, best, args=(b,), jac=smooth_grad,
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-15})
        if res.x is not None:
            c = compute_c5(res.x)
            if c < best_c5:
                best_c5 = c
                best = res.x.copy()
    return best, best_c5

        h_try = project_feasible(h_try, n)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5 - 1e-15:
            h = h_try
            best = h_try.copy()
            best_c5 = c5_try
    return best, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float)
    else:
        h_init = np.full(45, 0.5)

    n = len(h_init)
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Subgradient optimization (fast)
    rem = budget_s - (time.time() - t0)
    if rem > 20:
        h1, c1 = subgrad_opt(best_h, n_iters=8000, eta0=0.4, top_k=3, seed=seed)
        if c1 < best_c5:
            best_h, best_c5 = h1, c1

    # Phase 2: SLSQP polish
    rem = budget_s - (time.time() - t0)
    if rem > 60:
        h2, c2 = slsqp_opt(best_h, [50, 200, 1000, 5000], max_iter=300)
        if c2 < best_c5:
            best_h, best_c5 = h2, c2

    # Phase 3: Dual annealing (global)
    rem = budget_s - (time.time() - t0)
    if rem > 60:
        h3, c3 = da_opt(best_h, maxiter=300, seed=seed + 10)
        if c3 < best_c5:
            best_h, best_c5 = h3, c3
        rem = budget_s - (time.time() - t0)
        if rem > 40:
            h3b, c3b = slsqp_opt(best_h, [500, 2000, 10000], max_iter=300)
            if c3b < best_c5:
                best_h, best_c5 = h3b, c3b

    # Phase 4: Targeted local search
    rem = budget_s - (time.time() - t0)
    if rem > 10:
        n_ls = min(40000, int(rem * 35))
        h4, c4 = targeted_ls(best_h, n_iters=n_ls, seed=seed + 20)
        if c4 < best_c5:
            best_h, best_c5 = h4, c4

    # Phase 5: SLSQP high beta
    rem = budget_s - (time.time() - t0)
    if rem > 50:
        h5, c5v = slsqp_opt(best_h, [2000, 10000, 50000], max_iter=400)
        if c5v < best_c5:
            best_h, best_c5 = h5, c5v

    # Phase 6: Random restarts with diverse in
    rem = budget_s - (time.time() - t0)
    n_restarts = min(15, int(rem / 25))
    for ri in range(n_restarts):
        if time.time() - t0 > budget_s - 20:
            break
        strategy = ri % 5
        if strategy == 0:
            h_pert = best_h + 0.12 * rng.randn(n)
        elif strategy == 1:
            h_pert = best_h.copy()
            mask = rng.random(n) < 0.15
            h_pert[mask] = 1.0 - h_pert[mask]
        elif strategy == 2:
            idx = rng.choice(n, n // 2, replace=False)
            h_pert = np.zeros(n)
            h_pert[idx] = 1.0
        elif strategy == 3:
            shift = rng.randint(1, n)
            h_pert = np.roll(best_h, shift)
        else:
            h_pert = best_h + 0.25 * rng.randn(n)

        h_pert = project_feasible(h_pert, n)
        hr, cr = subgrad_opt(h_pert, n_iters=3000, eta0=0.3, top_k=2, seed=seed + 100 + ri)
        hr, cr = slsqp_opt(hr, [100, 1000, 5000], max_iter=200)
        if cr < best_c5:
            best_h, best_c5 = hr, cr
        rem = budget_s - (time.time() - t0)
        if rem > 8:
            hr2, cr2 = targeted_ls(hr, n_iters=min(10000, int(rem * 40)), seed=seed + 200 + ri)
            if cr2 < best_c5:
                best_h, best_c5 = hr2, cr2

    # Phase 7: Try different resolutions
    rem = budget_s - (time.time() - t0)
    if rem > 80:
        n_list = [30, 36, 42, 48, 54, 60, 66, 72, 84, 96, 120]
        for nt in n_list:
            if time.time() - t0 > budget_s - 45:
                break
            if nt == n:
                continue
            x_old = np.linspace(0, 2, n, endpoint=False)
            x_new = np.linspace(0, 2, nt, endpoint=False)
            h_int = np.interp(x_new, x_old, best_h)
            h_int = project_feasible(h_int, nt)
            ht, ct = subgrad_opt(h_int, n_iters=4000, eta0=0.3, top_k=3, seed=seed + 300 + nt)
            ht, ct = slsqp_opt(ht, [100, 1000, 5000], max_iter=250)
            if ct < best_c5:
                best_h, best_c5 = ht, ct
                n = nt
            rem = budget_s - (time.time() - t0)
            if rem > 6:
                ht2, ct2 = targeted_ls(ht, n_iters=min(10000, int(rem * 50)), seed=seed + 400 + nt)
                if ct2 < best_c5:
                    best_h, best_c5 = ht2, ct2
                    n = nt

    # Phase 8: Final SLSQP + local search
    rem = budget_s - (time.time() - t0)
    if rem > 30:
        hf, cf = slsqp_opt(best_h, [5000, 20000, 100000], max_iter=500)
        if cf < best_c5:
            best_h, best_c5 = hf, cf
    rem = budget_s - (time.time() - t0)
    if rem > 5:
        n_il = min(50000, int(rem * 60))
        hf2, cf2 = targeted_ls(best_h, n_iters=n_il, seed=seed + 999)
        if cf2 < best_c5:
            best_h, best_c5 = hf2, cf2

    # Final feasibility check
    best_h = project_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
