def exact_objective(h):

def project_box_sum(h, target=None):
    n = len(h)
    if target is None:
        target = 0.5 * n
    h = np.array(h, dtype=np.float64).copy()
    h = np.clip(h, 0.0, 1.0)

        diff = target - np.sum(h)
        if abs(diff) < 1e-14:

def profile_delta_all_k(h, i, j, lam):
    """Compute the change in the full profile (all 2n-1 shifts) when h[i]+=lam, h[j]-=lam"""
    n = len(h)
    n_profile = 2*n - 1
    delta = np.zeros(n_profile)
    
    for m in range(n_profile):

        i_start = max(0, -k)
        i_end = min(n-1, n-1-k)

        rounds += 1

            if h[i] >= 1.0:
                continue
            for j in range(n):
                if i == j or h[j] <= 0.0:
                    continue
                
                # Feasible interval for lambda
                L = max(0.0, -h[i])  # h[i] + lam >= 0 (always true if h[i]>=0 and lam>=0)
                # Actually: h[i]+lam <= 1 => lam <= 1-h[i]
                # h[j]-lam >= 0 => lam <= h[j]
                # h[i]+lam >= 0 => lam >= -h[i] (always true since h[i]>=0, lam>=0)
                # h[j]-lam <= 1 => lam >= h[j]-1 (always true since h[j]<=1, lam>=0)
                U = min(1.0 - h[i], h[j])
                if U < 1e-12:
                    continue

                # f(0) = 0, f(1) = profile_delta at lam=1, f(2) = profile_delta at lam=2
                d1 = profile_delta_all_k(h, i, j, 1.0)
                d2 = profile_delta_all_k(h, i, j, 2.0)
                A = (d2 - 2.0*d1) / 2.0  # quadratic coefficient
                B = d1 - A  # linear coefficient
                C = profile  # constant term (current profile)
                
                # Find best lambda in [0, U]
                best_lam = 0.0
                best_max_val = current_c
                
                # Sample lambda values
                lams = np.linspace(0.0, U, lam_samples + 1)[1:]  # exclude 0
                for lam in lams:
                    vals = A * lam * lam + B * lam + C
                    m = np.max(vals)
                    if m < best_max_val - 1e-12:
                        best_max_val = m
                        best_lam = lam
                
                if best_lam > 1e-12:
                    # Verify with exact evaluation
                    h[i] += best_lam
                    h[j] -= best_lam
                    new_profile = np.correlate(h, 1.0 - h, mode='full') * (2.0/n)
                    new_c = np.max(new_profile)
                    if new_c < current_c - 1e-12:
                        current_c = new_c
                        profile = new_profile
                        improved = True
                    else:
                        # Revert
                        h[i] -= best_lam
                        h[j] += best_lam

                    profile = np.correlate(h, 1.0 - h, mode='full') * (2.0/n)
                    new_c = float(np.max(profile))
                    if new_c < cur_c - 1e-11:
                        cur_c = new_c
                        imp = True
                    else:
                        h[i] -= best_lam
                        h[j] += best_lam
        if not imp:
            break
    
    return h, cur_c

def _profile_change_vec(h, i, j, lam):
    """Compute change in full profile vector when h[i]+=lam, h[j]-=lam"""
    n = len(h)
    np_ = 2 * n - 1
    out = np.zeros(np_)
    
    for m in range(np_):
        k = m - (n - 1)
        lo = max(0, -k)
        hi = min(n - 1, n - 1 - k)
        
        # Affected first indices
        cands = ()
        if lo <= i <= hi:
            cands += (i,)
        if lo <= j <= hi and j != i:
            cands += (j,)
        ik = i - k
        if lo <= ik <= hi and ik != i and ik != j:
            cands += (ik,)
        jk = j - k
        if lo <= jk <= hi and jk != i and jk != j and jk != ik:
            cands += (jk,)
        
        for ip in cands:
            sec = ip + k
            if not (0 <= sec < n):
                continue
            dhf = lam if ip == i else (-lam if ip == j else 0.0)
            dhs = lam if sec == i else (-lam if sec == j else 0.0)
            if dhf != 0.0 and dhs != 0.0:
                out[m] += -h[ip] * dhs
