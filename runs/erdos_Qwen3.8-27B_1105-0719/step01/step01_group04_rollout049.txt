import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr * dx))

def repair_feasible(vec):
    h = np.asarray(vec, dtype=np.float64).copy()
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(h)
        rem = diff
        for i in order:
            if rem <= 1e-15:
                break
            add = min(rem, 1.0 - h[i])
            h[i] += add
            rem -= add
    else:
        order = np.argsort(-h)
        rem = -diff
        for i in order:
            if rem <= 1e-15:
                break
            remv = min(rem, h[i])
            h[i] -= remv
            rem -= remv
    # Final fix for floating point
    diff2 = target - h.sum()
    if abs(diff2) > 1e-14:
        for i in range(n):
            if diff2 > 0 and h[i] < 1.0:
                h[i] += diff2
                break
            elif diff2 < 0 and h[i] > 0.0:
                h[i] += diff2
                break
    return h

def compute_lags_and_grads(h):
    """Compute all lag values and their gradients w.r.t. h."""
    n = len(h)
    dx = 2.0 / n
    n_lags = 2 * n - 1
    values = np.zeros(n_lags)
    grads = np.zeros((n_lags, n))
    
    # Lag 0 (index n-1)
    values[n-1] = dx * np.sum(h * (1.0 - h))
    grads[n-1, :] = dx * (1.0 - 2.0 * h)
    
    # Positive lags k = 1, ..., n-1 (index n-1+k)
    for k in range(1, n):
        # Q_k = dx * sum_{j=k}^{n-1} h[j] * (1-h[j-k])
        j = np.arange(k, n)
        values[n-1+k] = dx * np.sum(h[j] * (1.0 - h[j-k]))
        # Gradient
        grads[n-1+k, j] += dx * (1.0 - h[j-k])  # d/dh[j]
        grads[n-1+k, j-k] += -dx * h[j]  # d/dh[j-k]
    
    # Negative lags k = -1, ..., -(n-1) (index n-1+k = n-1-|k|)
    for k in range(1, n):
        # Q_{-k} = dx * sum_{j=0}^{n-1-k} h[j] * (1-h[j+k])
        j = np.arange(0, n-k)
        values[n-1-k] = dx * np.sum(h[j] * (1.0 - h[j+k]))
        # Gradient
        grads[n-1-k, j] += dx * (1.0 - h[j+k])  # d/dh[j]
        grads[n-1-k, j+k] += -dx * h[j]  # d/dh[j+k]
    
    return values, grads

def trust_region_step(h_cur, values, grads, lambda_tr=0.1, n_active=50, max_lag_margin=0.0):
    """Solve the trust-region QP subproblem."""
    n = len(h_cur)
    n_lags = 2 * n - 1
    c_cur = np.max(values)
    
    # Select active lags: those within margin of max, or top n_active
    if max_lag_margin > 0:
        active_idx = np.where(values >= c_cur - max_lag_margin)[0]
    else:
        active_idx = np.argsort(values)[::-1][:min(n_active, n_lags)]
    
    active_idx = np.sort(active_idx)
    v_act = values[active_idx]
    g_act = grads[active_idx, :]  # shape (len(active_idx), n)
    
    h_new = cp.Variable(n)
    t = cp.Variable()
    
    # Objective: minimize t + lambda * ||h_new - h_cur||^2
    obj = cp.Minimize(t + lambda_tr * cp.sum_squares(h_new - h_cur))
    
    # Constraints
    constraints = [
        cp.sum(h_new) == n / 2.0,
        h_new >= 0,
        h_new <= 1,
    ]
    
    # Linearized lag constraints: Q_r + G_r . (h_new - h_cur) <= t
    # v_act[i] + g_act[i] . (h_new - h_cur) <= t
    delta = h_new - h_cur
    for i in range(len(active_idx)):
        constraints.append(v_act[i] + g_act[i] @ delta <= t)
    
    # Lower bound on t (to keep it positive)
    constraints.append(t >= 0)
    
    prob = cp.Problem(obj, constraints)
    
    try:
        prob.solve(solver=cp.CLARABEL)
    except Exception:
        try:
            prob.solve(solver=cp.SCS)
        except Exception:
            return None
    
    if prob.status in ['optimal', 'optimal_inaccurate'] and h_new.value is not None:
        return np.array(h_new.value, dtype=np.float64)
    return None

For k > 0: j goes from 0 to n-1-k
Q_k = dx * sum_{j=0}^{n-1-k} h[j] * (1 - h[j+k])

dQ_k/dh[m]:
- If m is in [0, n-1-k]: contributes dx * (1 - h[m+k])
- If m is in [k, n-1]: contributes dx * (-h[m-k])
- Otherwise: 0

So each h[m] appears in Q_k at most twice (once as the "first" factor and once as the "second" factor).

For k = 0:
Q_0 = dx * sum_j h[j](1-h[j])
dQ_0/dh[m] = dx * (1 - 2h[m])

import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize


def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr * dx))


def repair_feasible(vec):
    h = np.asarray(vec, dtype=np.float64).copy()
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(h)
        rem = diff
        for i in order:
            if rem <= 1e-15:
                break
            add = min(rem, 1.0 - h[i])
            h[i] += add
            rem -= add
    else:
        order = np.argsort(-h)
        rem = -diff
        for i in order:
            if rem <= 1e-15:
                break
            remv = min(rem, h[i])
            h[i] -= remv
            rem -= remv
    diff2 = target - h.sum()
    if abs(diff2) > 1e-14:
        for i in range(n):
            if diff2 > 0 and h[i] < 1.0:
                h[i] += diff2
                break
            elif diff2 < 0 and h[i] > 0.0:
                h[i] += diff2
                break
    return h


def compute_lags_and_grads(h):
    n = len(h)
    dx = 2.0 / n
    n_lags = 2 * n - 1
    values = np.zeros(n_lags)
    grads = np.zeros((n_lags, n))
    values[n - 1] = dx * np.sum(h * (1.0 - h))
    grads[n - 1, :] = dx * (1.0 - 2.0 * h)
    for k in range(1, n):
        j = np.arange(k, n)
        values[n - 1 + k] = dx * np.sum(h[j] * (1.0 - h[j - k]))
        grads[n - 1 + k, j] += dx * (1.0 - h[j - k])
        grads[n - 1 + k, j - k] += -dx * h[j]
        j2 = np.arange(0, n - k)
        values[n - 1 - k] = dx * np.sum(h[j2] * (1.0 - h[j2 + k]))
        grads[n - 1 - k, j2] += dx * (1.0 - h[j2 + k])
        grads[n - 1 - k, j2 + k] += -dx * h[j2]
    return values, grads


def trust_region_qp(h_cur, values, grads, lam, active_idx, solver_pref="CLARABEL"):
    n = len(h_cur)
    v_act = values[active_idx]
    g_act = grads[active_idx, :]
    h_new = cp.Variable(n)
    t = cp.Variable()
    delta = h_new - h_cur
    obj = cp.Minimize(t + lam * cp.sum_squares(delta))
    cons = [
        cp.sum(h_new) == n / 2.0,
        h_new >= 0,
        h_new <= 1,
        t >= 0,
    ]
    for i in range(len(active_idx)):
        cons.append(v_act[i] + g_act[i] @ delta <= t)
    prob = cp.Problem(obj, cons)
    solved = False
    for solver in [solver_pref, "SCS", "ECOS"]:
        try:
            prob.solve(solver=solver, max_iters=5000, verbose=False)
            solved = True
            break
        except Exception:
            continue
    if not solved:
        return None
    if prob.status in ["optimal", "optimal_inaccurate"] and h_new.value is not None:
        return np.array(h_new.value, dtype=np.float64)
    return None


def trust_region_optimize(h_start, time_limit, max_iter=200):
    h = repair_feasible(h_start.copy())
    best_h = h.copy()
    best_score = exact_score(h)
    t_start = time.perf_counter()
    lam = 0.1
    consec_rejects = 0
    consec_accepts = 0
    for it in range(max_iter):
        if time.perf_counter() - t_start > time_limit:
            break
        values, grads = compute_lags_and_grads(h)
        c_cur = np.max(values)
        n_lags = len(values)
        # Active set: lags within 30% of max, or top 60
        margin = 0.3 * c_cur
        active = np.where(values >= c_cur - margin)[0]
        if len(active) > 80:
            active = np.argsort(values)[::-1][:80]
        active = np.sort(active)
        h_new = trust_region_qp(h, values, grads, lam, active)
        if h_new is None:
            lam *= 2.0
            consecutive_rejects += 1
            if consec_rejects > 5:
                break
            continue
        h_new = repair_feasible(h_new)
        score_new = exact_score(h_new)
        if score_new < best_score - 1e-14:
            h = h_new.copy()
            best_h = h_new.copy()
            best_score = score_new
            consec_accepts += 1
            consec_rejects = 0
            if consec_accepts >= 3:
                lam = max(0.01, lam * 0.7)
                consec_accepts = 0
        else:
            # Try a fractional step
            improved = False
            for frac in [0.5, 0.25, 0.1]:
                h_frac = repair_feasible(h + frac * (h_new - h))
                s_frac = exact_score(h_frac)
                if s_frac < best_score - 1e-14:
                    h = h_frac
                    best_h = h_frac.copy()
                    best_score = s_frac
                    improved = True
                    break
            if not improved:
                lam = min(10.0, lam * 1.5)
                consec_rejects += 1
                if consec_rejects > 8:
                    break
            else:
                consec_rejects = 0
    return best_h, best_score


def active_descent(h_start, max_steps=80, time_limit=60.0):
    h = repair_feasible(h_start.copy())
    best_h = h.copy()
    best_score = exact_score(h)
    t_start = time.perf_counter()
    for step in range(max_steps):
        if time.perf_counter() - t_start > time_limit:
            break
        values, grads = compute_lags_and_grads(h)
        c = np.max(values)
        # Weighted combination of top gradients
        top_k = min(25, len(values))
        top_idx = np.argsort(values)[::-1][:top_k]
        gaps = np.maximum(c - values[top_idx], 0.0)
        sigma = max(0.003 * c, 1e-6)
        weights = np.exp(-gaps / sigma)
        weights /= (np.sum(weights) + 1e-15)
        direction = -np.sum(weights[:, None] * grads[top_idx, :], axis=0)
        direction -= np.mean(direction)
        dir_norm = np.linalg.norm(direction)
        if dir_norm < 1e-12:
            break
        direction /= dir_norm
        improved = False
        for alpha in [0.05, 0.03, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]:
            h_trial = repair_feasible(h + alpha * direction)
            s = exact_score(h_trial)
            if s < best_score - 1e-14:
                h = h_trial.copy()
                best_h = h_trial.copy()
                best_score = s
                improved = True
                break
        if not improved:
            # Try random perturbation
            rng_local = np.random.RandomState(step * 7 + 3)
            z = rng_local.randn(len(h))
            z -= z.mean()
            z /= (np.linalg.norm(z) + 1e-15)
            for alpha in [0.01, 0.005, 0.002]:
                h_trial = repair_feasible(h + alpha * z)
                s = exact_score(h_trial)
                if s < best_score - 1e-14:
                    h = h_trial.copy()
                    best_h = h_trial.copy()
                    best_score = s
                    improved = True
                    break
            if not improved:
                break
    return best_h, best_score


def interpolate_to_dimension(vec, new_len):
    n = len(vec)
    if n == new_len:
        return vec.copy()
    x_src = (np.arange(n) + 0.5) * 2.0 / n
    x_tgt = (np.arange(new_len) + 0.5) * 2.0 / new_len
    xs_ext = np.concatenate([[0.0], x_src, [2.0]])
    ys_ext = np.concatenate([[vec[0]], vec, [vec[-1]]])
    return np.interp(x_tgt, xs_ext, ys_ext)


def symmetrize(h):
    return (h + h[::-1]) / 2.0


def slsqp_polish(h_start, maxiter=100, time_limit=60.0):
    h = repair_feasible(h_start.copy())
    best_h = h.copy()
    best_score = exact_score(h)
    t_start = time.perf_counter()

    def objective(x):
        if time.perf_counter() - t_start > time_limit:
            return best_score
        h_x = np.clip(x, 0.0, 1.0)
        return exact_score(h_x)

    def gradient(x):
        h_x = np.clip(x, 0.0, 1.0)
        values, grads = compute_lags_and_grads(h_x)
        c = np.max(values)
        top_idx = np.where(values >= c - 0.02 * c)[0]
        if len(top_idx) == 0:
            top_idx = [np.argmax(values)]
        return np.mean(grads[top_idx, :], axis=0)

    bounds = [(0.0, 1.0)] * len(h)
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - len(h) / 2.0,
             'jac': lambda x: np.ones(len(h))}]
    try:
        res = minimize(objective, h, jac=gradient, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        if res.x is not None:
            h_sol = repair_feasible(res.x)
            s = exact_score(h_sol)
            if s < best_score - 1e-14:
                best_h = h_sol
                best_score = s
    except Exception:
        pass
    return best_h, best_score
