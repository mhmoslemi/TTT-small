                break
            step = min(-diff / cnt, 1.0 - np.max(h[mask]))
            h[mask] += step
    return h

def to_binary(h):
    n = len(h)
    if n % 2 != 0:
        return None
    target = n // 2
    idx = np.argsort(-h)
    b = np.zeros(n, dtype=np.float64)
    b[idx[:target]] = 1.0
    return b

def compute_sig(b, k_vals):
    """Compute signature matrix for all positions."""
    n = len(b)
    positions = np.arange(n)
    pos_plus = positions[:, None] + k_vals[None, :]
    pos_minus = positions[:, None] - k_vals[None, :]
    valid_plus = (pos_plus >= 0) & (pos_plus < n)
    valid_minus = (pos_minus >= 0) & (pos_minus < n)
    pp = np.clip(pos_plus, 0, n-1)
    pm = np.clip(pos_minus, 0, n-1)
    b_pp = b[pp]
    b_pm = b[pm]
    term1 = np.where(valid_plus, -np.where(b_pp == 0, 1.0, 0.0), 0.0)
    term2 = np.where(valid_minus, np.where(b_pm == 1, 1.0, 0.0), 0.0)
    return term1 + term2

def binary_ls(b, max_rounds=500, rng_seed=42, kick_size=3, max_kicks=5):
    """Iterated local search with kicks for binary sequence."""
    n = len(b)
    b = b.astype(np.float64).copy()
    k_vals = np.arange(-(n-1), n)
    dx = 2.0 / n
    
    overlap = np.correlate(b, 1.0-b, mode='full').astype(np.float64)
    cur_max = float(np.max(overlap))
    best_b = b.copy()
    best_max = cur_max
    
    sig = compute_sig(b, k_vals)
    rng = np.random.RandomState(rng_seed)
    
    for kick in range(max_kicks + 1):
        # Local search to convergence
        for rnd in range(max_rounds):
            ones = np.where(b == 1.0)[0]
            zeros = np.where(b == 0.0)[0]
            n_ones = len(ones)
            n_zeros = len(zeros)
            
            if n_ones == 0 or n_zeros == 0:
                break
            
            best_delta = 0.0
            best_i = -1
            best_j = -1
            best_new_overlap = None
            
            # Vectorized over j for each i
            sig_zeros = sig[zeros]  # shape (n_zeros, 2n-1)
            
            for ii in range(n_ones):
                i = int(ones[ii])
                # new_ov_all[j_idx] = overlap + sig[i] - sig[zeros[j_idx]]
                new_ov_all = overlap[None, :] + sig[i][None, :] - sig_zeros
                new_maxs = np.max(new_ov_all, axis=1)
                deltas = cur_max - new_maxs
                best_idx = int(np.argmax(deltas))
                d = float(deltas[best_idx])
                if d > best_delta + 1e-12:
                    best_delta = d
                    best_i = i
                    best_j = int(zeros[best_idx])
                    best_new_overlap = new_ov_all[best_idx].copy()
            
            if best_i >= 0:
                b[best_i] = 0.0
                b[best_j] = 1.0
                overlap = best_new_overlap
                cur_max = cur_max - best_delta
                sig = compute_sig(b, k_vals)
            
            if cur_max < best_max - 1e-12:
                best_max = cur_max
                best_b = b.copy()
            
            # Early stopping if we've been stuck for a while
            # (best_delta == 0 means no improvement found)
            if best_delta < 1e-12:
                break
        
        # If not at best, apply a kick
        if kick < max_kicks:
            # Apply kick_size random swaps
            for _ in range(kick_size):
                ones = np.where(b == 1.0)[0]
                zeros = np.where(b == 0.0)[0]
                if len(ones) == 0 or len(zeros) == 0:
                    break
                i = int(rng.choice(ones))
                j = int(rng.choice(zeros))
                b[i] = 0.0
                b[j] = 1.0
                overlap = np.correlate(b, 1.0-b, mode='full').astype(np.float64)
                cur_max = float(np.max(overlap))
                sig = compute_sig(b, k_vals)
    
    return best_b, best_max * dx
