                    fc = np.max(b + c * a)
                else:
                    lo = c
                    c = d
                    fc = fd
                    d = lo + gr * (hi - lo)
                    fd = np.max(b + d * a)
            
            v_opt = (lo + hi) / 2
            h[i] = v_opt
            improved = True
        
        # Project to constraint
        h = repair_feasible(h)
        s = exact_score(h)
        if s < best_score - 1e-13:
            best_score = s
            best_h = h.copy()
        # Early stopping
        if s >= best_score - 1e-10:
            # Check if we're making progress
            if p > 5 and not improved:
                break
    
    return best_h, best_score

And since p_k is linear in h[i] (for fixed other h[j]), we have:
f(v) = max_k [a_k * v + b_k]

where a_k and b_k depend on the current values of all other h[j].

By evaluating at v=0 and v=1:

- At v=1: p_k(1) = a_k + b_k

But actually, there's an even better approach for 1D convex PL minimization: we can find the minimum by binary search on the subgradient. At the minimum, 0 is in the subgradient. The subgradient at v is:
∂f(v) = {a_k : b_k + a_k*v = f(v)} (the set of slopes of active lines at v)

The minimum is where the subgradient contains 0, i.e., where there's a line with slope < 0 and a line with slope > 0 that both achieve the max (or a single line with slope 0).

import numpy as np
import time
from scipy.optimize import minimize

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))

def get_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def repair_feasible(h, target=None):
    h = np.array(h, dtype=np.float64).copy()
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    if target is None:
        target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        order = np.argsort(-room)
        remaining = diff
        for idx in order:
            if remaining <= 1e-15:
                break
            add = min(remaining, room[idx])
            h[idx] += add
            remaining -= add
    else:
        order = np.argsort(-h)
        remaining = -diff
        for idx in order:
            if remaining <= 1e-15:
                break
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
    return h

def get_lag_gradients(h):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    G = np.zeros((M, n))
    for k in range(M):
        delta = k - (n - 1)
        if delta >= 0:
            if delta < n:
                G[k, delta:n] += dx * (1.0 - h[0:n - delta])
                G[k, 0:n - delta] -= dx * h[delta:n]
        else:
            d = -delta
            if d < n:
                G[k, 0:n - d] += dx * (1.0 - h[d:n])
                G[k, d:n] -= dx * h[0:n - d]
    return G

def coord_descent_1d(h, i, max_iter=50, time_check=None, t0=None, t_limit=None):
    """Solve the 1D subproblem for coordinate i exactly via golden section."""
    n = len(h)
    M = 2 * n - 1
    orig = h[i]
    h[i] = 0.0
    prof0 = get_profile(h)
    h[i] = 1.0
    prof1 = get_profile(h)
    h[i] = orig
    a = prof1 - prof0
    b = prof0
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, 1.0
    c = lo + (1.0 - gr) * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = np.max(b + c * a)
    fd = np.max(b + d * a)
    for _ in range(max_iter):
        if time_check is not None and time.perf_counter() - t0 > t_limit:
            break
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = lo + (1.0 - gr) * (hi - lo)
            fc = np.max(b + c * a)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = np.max(b + d * a)
    return (lo + hi) / 2.0

def coord_descent(h_start, max_passes=100, time_limit=30):
    """Exact coordinate descent for the minimax overlap problem."""
    n = len(h_start)
    h = h_start.copy()
    best_h = h_start.copy()
    best_score = exact_score(h_start)
    t0 = time.perf_counter()
    prev_score = best_score

    for p in range(max_passes):
        if time.perf_counter() - t0 > time_limit:
            break
        for i in range(n):
            if time.perf_counter() - t0 > time_limit:
                break
            h[i] = coord_descent_1d(h, i, max_iter=45,
                                    time_check=True, t0=t0, t_limit=time_limit)
        h = repair_feasible(h)
        s = exact_score(h)
        if s < best_score - 1e-14:
            best_score = s
            best_h = h.copy()
        if abs(prev_score - s) < 1e-14 and p > 3:
            break
        prev_score = s

    return best_h, best_score

def solve_epigraph(h_start, t_start, beta=0.0, maxiter=80, n_constraints=0):
    n = len(h_start)
    M = 2 * n - 1
    nc = n_constraints if n_constraints > 0 else M
    if nc < M:
        prof = get_profile(h_start)
        active_lags = np.argsort(prof)[::-1][:nc]
    else:
        active_lags = np.arange(M)

    def obj(x):
        h = x[:n]
        t = x[n]
        return t + beta * np.sum(h * (1.0 - h))

    def obj_jac(x):
        h = x[:n]
        g = np.zeros(n + 1)
        g[:n] = beta * (1.0 - 2.0 * h)
        g[n] = 1.0
        return g

    def cons_fun(x):
        h = x[:n]
        t = x[n]
        eq = h.sum() - n / 2.0
        prof = np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)
        ineq = t - prof[active_lags]
        return np.concatenate([[eq], ineq])

    def cons_jac(x):
        h = x[:n]
        G = get_lag_gradients(h)
        Ja = G[active_lags, :]
        J = np.zeros((1 + len(active_lags), n + 1))
        J[0, :n] = 1.0
        J[1:, :n] = -Ja
        J[1:, n] = 1.0
        return J

    bounds = [(0.0, 1.0)] * n + [(0.0, t_start + 2.0)]
    x0 = np.concatenate([h_start, [t_start]])

    try:
        result = minimize(
            obj, x0, jac=obj_jac,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'ineq', 'fun': cons_fun, 'jac': cons_jac}],
            options={'maxiter': maxiter, 'ftol': 1e-12}
        )
        if result.x is not None and np.all(np.isfinite(result.x)):
            h_sol = repair_feasible(result.x[:n])
            return h_sol
    except Exception:
        pass
    return None

def interpolate_to_dim(vec, new_len):
    n = len(vec)
    x_src = (np.arange(n) + 0.5) * 2.0 / n
    x_tgt = (np.arange(new_len) + 0.5) * 2.0 / new_len
    x_ext = np.concatenate([[0.0], x_src, [2.0]])
    f_ext = np.concatenate([[vec[0]], vec, [vec[-1]]])
    h_new = np.interp(x_tgt, x_ext, f_ext)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

def smooth_h(h, sigma=1.0):
    """Apply a simple box blur for smoothing."""
    if sigma <= 0:
        return h.copy()
    k = max(1, int(round(sigma)))
    kernel = np.ones(2 * k + 1) / (2 * k + 1)
    h_pad = np.pad(h, k, mode='edge')
    h_smooth = np.convolve(h_pad, kernel, mode='valid')
    return repair_feasible(h_smooth)

def local_gradient_descent(h_current, max_iters=80, time_limit=25):
    """Weighted active-lag gradient descent with backtracking."""
    n = len(h_current)
    h = h_current.copy()
    best_h = h.copy()
    best_score = exact_score(h)
    t0 = time.perf_counter()
    step = 0.02

    for it in range(max_iters):
        if time.perf_counter() - t0 > time_limit:
            break
        prof = get_profile(h)
        score = float(np.max(prof))
        if score < best_score - 1e-14:
            best_score = score
            best_h = h.copy()

        A = max(8, min(40, n // 2))
        active_idx = np.argsort(prof)[::-1][:A]
        G = get_lag_gradients(h)
        w = prof[active_idx].copy()
        w = np.maximum(w, 0.0)
        wsum = w.sum()
        if wsum < 1e-15:
            break
        w /= wsum
        direction = np.sum(w[:, None] * G[active_idx, :], axis=0)
        direction -= direction.mean()
        norm = np.linalg.norm(direction)
        if norm < 1e-14:
            break
        direction /= norm

        improved = False
        for alpha in [step, step * 0.5, step * 0.25, step * 0.125]:
            h_trial = repair_feasible(h + alpha * direction)
            s_trial = exact_score(h_trial)
            if s_trial < best_score - 1e-13:
                h = h_trial
                best_h = h.copy()
                best_score = s_trial
                step = min(step * 1.3, 0.05)
                improved = True
                break
        if not improved:
            step *= 0.5
            if step < 1e-7:
                break
    return best_h, best_score

def multi_start_cd(h_best, n_starts=8, time_per_start=5, t0_global=None, t_left_fn=None):
    """Try random perturbations followed by coordinate descent."""
    n = len(h_best)

    best_score = exact_score(h_best)
    rng = np.random.RandomState(12345)

    for s in range(n_starts):

        eps = 0.02 + 0.01 * s
        z = rng.randn(n)
        z -= z.mean()
        h_pert = repair_feasible(best_h + eps * z)
        s_pert = exact_score(h_pert)
        tl = min(time_per_start, (t_left_fn() - 5) if t_left_fn else time_per_start)
        h_cd, s_cd = coord_descent(h_pert, max_passes=30, time_limit=tl)
        if s_cd < best_score - 1e-13:
            best_score = s_cd
            best_h = h_cd.copy()
    return best_h, best_score
