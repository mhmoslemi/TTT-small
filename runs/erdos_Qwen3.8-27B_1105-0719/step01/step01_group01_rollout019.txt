def slsqp_smooth(h0, n, betas, max_iter=200):
    dx = 2.0 / n
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    bounds = [(0.0, 1.0)] * n

    def obj(h):
        corr = np.correlate(h, 1.0 - h, mode="full")
        M = np.max(corr)
        return (M + np.log(np.sum(np.exp(betas[0] * (corr - M)))) / betas[0]) * dx

    def obj_grad(h):
        corr = np.correlate(h, 1.0 - h, mode="full")
        M = np.max(corr)
        exps = np.exp(betas[0] * (corr - M))
        w = exps / np.sum(exps)
        G = build_grad_matrix(h, n)
        return (G * w[:, None]).sum(axis=0) * dx

    def con_fun(h):
        return np.sum(h) - n / 2.0

    def con_jac(h):
        return np.ones(n)

    for beta in betas:
        def obj_b(h, b=beta):
            corr = np.correlate(h, 1.0 - h, mode="full")
            M = np.max(corr)
            return (M + np.log(np.sum(np.exp(b * (corr - M)))) / b) * dx

        def obj_grad_b(h, b=beta):
            corr = np.correlate(h, 1.0 - h, mode="full")
            M = np.max(corr)
            exps = np.exp(b * (corr - M))
            w = exps / np.sum(exps)
            G = build_grad_matrix(h, n)
            return (G * w[:, None]).sum(axis=0) * dx

        res = minimize(
            obj_b, best_h,
            jac=obj_grad_b,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': con_fun, 'jac': con_jac}],
            options={'maxiter': max_iter, 'ftol': 1e-14}
        )
        if res.x is not None:
            h_cand = project_feasible(res.x, n)
            c5_cand = compute_c5(h_cand)
            if c5_cand < best_c5:
                best_c5 = c5_cand
                best_h = h_cand
    return best_h, best_c5

        elif s % 3 == 1:
            h_new = h_best.copy()
            mask = rng.random(n) < 0.15
            h_new[mask] = 1.0 - h_new[mask]
        else:
            h_new = project_feasible(rng.uniform(0, 1, n), n)
        h_new = project_feasible(h_new, n)
        population.append(h_new)
        scores.append(compute_c5(h_new))

    best_h = population[np.argmin(scores)].copy()
    best_c5 = min(scores)

    t0 = time.time()
    generation = 0
    while time.time() - t0 < time_limit - 5:
        generation += 1
        idx_max = np.argmax(scores)
        idx_min = np.argmin(scores)
        p1 = population[idx_min].copy()
        p2 = population[idx_min + 1 + rng.randint(max(0, pop_size - 3)) if idx_min + 1 + rng.randint(max(0, pop_size - 3)) < pop_size else idx_min].copy()
        alpha = rng.uniform(0.2, 0.8)
        child = alpha * p1 + (1.0 - alpha) * p2
        if rng.random() < 0.3:
            child += 0.05 * rng.randn(n)
        child = project_feasible(child, n)
        c5_child = compute_c5(child)
        if c5_child < scores[idx_max]:
            population[idx_max] = child
            scores[idx_max] = c5_child
            if c5_child < best_c5:
                best_c5 = c5_child
                best_h = child.copy()
        if generation % 15 == 0:
            for pi in range(3):
                hpol, c5pol = coord_descent_fast(population[pi], n, n_sweeps=5)
                if c5pol < scores[pi]:
                    population[pi] = hpol
                    scores[pi] = c5pol
                    if c5pol < best_c5:
                        best_c5 = c5pol
                        best_h = hpol.copy()
    return best_h, best_c5


def tries_different_n(h_best, n_best, time_limit, rng):
    best_h = h_best.copy()
    best_c5 = compute_c5(best_h)
    t0 = time.time()
    for n_try in [30, 36, 42, 48, 54, 66, 72, 84]:
        if time.time() - t0 > time_limit - 15:
            break
        x_old = np.linspace(0, 2, n_best, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, n_try)
        h_opt, c5_opt = active_set_optimize(h_interp, n_try, max_iter=50, tol_init=0.02)
        if c5_opt < best_c5:
            h_opt2, c5_opt2 = coord_descent_fast(h_opt, n_try, n_sweeps=10)
            if c5_opt2 < best_c5:
                best_c5 = c5_opt2
                best_h = h_opt2
                n_best = n_try
            elif c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
                n_best = n_try
    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(45, 0.5)

    n = len(h_init)
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)

    rem = budget_s - (time.time() - t0)
    if rem < 30:
        return best_h, best_c5, n

    # Phase 1: Active-set optimization
    rem = budget_s - (time.time() - t0)
    if rem > 60:
        h_opt, c5_opt = active_set_optimize(best_h, n, max_iter=100, tol_init=0.01)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 2: Coordinate descent
    rem = budget_s - (time.time() - t0)
    if rem > 30:
        h_opt, c5_opt = coord_descent_fast(best_h, n, n_sweeps=20)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 3: Binary simulated annealing
    rem = budget_s - (time.time() - t0)
    if rem > 120:
        sa_time = min(150, rem - 30)
        h_sa, c5_sa, _ = binary_sa(best_h, n, sa_time, rng)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa
            # Chain: active set from binary
            h_opt, c5_opt = active_set_optimize(best_h, n, max_iter=50, tol_init=0.005)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
            h_opt, c5_opt = coord_descent_fast(best_h, n, n_sweeps=15)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt

    # Phase 4: More SA with different starting points
    rem = budget_s - (time.time() - t0)
    if rem > 80:
        for restart in range(3):
            if time.time() - t0 > budget_s - 40:
                break
            if restart == 0:
                h_start = best_h.copy()
            elif restart == 1:
                h_start = project_feasible(0.5 + 0.3 * rng.randn(n), n)
            else:
                h_start = project_feasible(rng.uniform(0, 1, n), n)
            sa_time = min(50, (budget_s - (time.time() - t0)) // 4 - 10)
            if sa_time < 15:
                break
            h_sa, c5_sa, _ = binary_sa(h_start, n, sa_time, rng)
            if c5_sa < best_c5:
                best_c5 = c5_sa
                best_h = h_sa
                h_opt, c5_opt = coord_descent_fast(best_h, n, n_sweeps=10)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt

    # Phase 5: Population evolution
    rem = budget_s - (time.time() - t0)
    if rem > 100:
        pop_time = min(250, rem - 30)
        h_pop, c5_pop = population_evolve(best_h, n, pop_time, rng)
        if c5_pop < best_c5:
            best_c5 = c5_pop
            best_h = h_pop

    # Phase 6: SLSQP polish
    rem = budget_s - (time.time() - t0)
    if rem > 60:
        h_opt, c5_opt = slsqp_smooth(best_h, n, [30, 100, 500, 2000], max_iter=150)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
        h_opt, c5_opt = coord_descent_fast(best_h, n, n_sweeps=15)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 7: Try different resolutions
    rem = budget_s - (time.time() - t0)
    if rem > 60:
        h_try, c5_try = tries_different_n(best_h, n, rem - 20, rng)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try
            n = len(best_h)

    # Phase 8: Final polish cycle
    rem = budget_s - (time.time() - t0)
    if rem > 30:
        h_opt, c5_opt = active_set_optimize(best_h, n, max_iter=50, tol_init=0.003)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
        h_opt, c5_opt = coord_descent_fast(best_h, n, n_sweeps=10)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Final exact evaluation
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    # Ensure feasibility
    best_h = np.clip(best_h, 0.0, 1.0)
    residual = n / 2.0 - np.sum(best_h)
    if abs(residual) > 1e-10:
        interior = np.where((best_h > 1e-8) & (best_h < 1.0 - 1e-8))[0]
        if len(interior) > 0:
            best_h[interior] += residual / len(interior)
            best_h = np.clip(best_h, 0.0, 1.0)

    final_c5 = compute_c5(best_h)
    return best_h.copy(), final_c5, len(best_h)
