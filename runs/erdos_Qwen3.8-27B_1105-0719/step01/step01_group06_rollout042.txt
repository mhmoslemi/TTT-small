def project_u(u):
    """Project u onto {v: sum(v)=0, -1<=v<=1}."""
    # Find shift s such that sum(clip(u - s, -1, 1)) = 0
    lo, hi = -1.0 - u.max(), 1.0 - u.min()
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        v = np.clip(u - mid, -1.0, 1.0)
        s = v.sum()
        if s > 0:
            lo = mid
        else:
            hi = mid
    v = np.clip(u - 0.5*(lo+hi), -1.0, 1.0)
    return v

    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    grad = np.zeros(n)

            grad[d:n] -= w * dx * h[0:n-d]
    
    return grad

def epigraph_polish(h_start, maxiter=50, nc=0):
    n = len(h_start)
    M = 2 * n - 1
    if nc == 0:
        nc = M
    prof = get_profile(h_start)
    active = np.argsort(prof)[::-1][:nc]

    def obj(x):
        return x[n]
    def obj_jac(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g
    def cons_fun(x):
        h = x[:n]
        t = x[n]
        p = get_profile(h)
        return np.concatenate([[h.sum() - n/2.0], t - p[active]])
    def cons_jac(x):
        h = x[:n]
        G = get_lag_gradients(h)
        Ja = G[active, :]
        J = np.zeros((1 + len(active), n + 1))
        J[0, :n] = 1.0
        J[1:, :n] = -Ja
        J[1:, n] = 1.0
        return J

    bounds = [(0.0, 1.0)] * n + [(0.0, 5.0)]
    x0 = np.concatenate([h_start, [exact_score(h_start) + 0.05]])
    try:
        res = minimize(obj, x0, jac=obj_jac, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'ineq', 'fun': cons_fun, 'jac': cons_jac}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return repair_feasible(res.x[:n])
    except Exception:
        pass
    return None

        if s_di < best_score - 1e-12:
            best_h = h_di
            best_score = s_di
        if tleft() > 20:
            h_db, s_db = binary_steepest_descent(h_di, time_limit=min(8, tleft() - 15))
            if s_db < best_score - 1e-12:
                best_h = h_db
                best_score = s_db

    # Phase 8: Final refinement
    if tleft() > 18:
        h_f, s_f = binary_steepest_descent(best_h, time_limit=min(10, tleft() - 8))
        if s_f < best_score - 1e-12:
            best_h = h_f
            best_score = s_f
    if tleft() > 12:
        h_ff, s_ff = lse_optimize(best_h, tau_start=40.0, tau_end=200.0,
                                   n_iters=2000, time_limit=min(8, tleft() - 5), lr_start=0.03)
        if s_ff < best_score - 1e-12:
            best_h = h_ff
            best_score = s_ff
    if tleft() > 10:
        h_epf = epigraph_polish(best_h, maxiter=60, nc=0)
        if h_epf is not None:
            s_epf = exact_score(h_epf)
            if s_epf < best_score - 1e-12:
                best_h = h_epf
                best_score = s_epf

    best_h = repair_feasible(best_h)
    final_score = exact_score(best_h)
    return (best_h, final_score, len(best_h))
