def exact_profile(h):

    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def exact_c5(h):
    p = exact_profile(h)
    return float(np.max(p))

def repair_feasible(vec):

    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0

    if abs(diff) < 1e-14:
        return h
    # Binary search for shift
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s < target:
            lo = mid
        else:
            hi = mid

    residual = target - h.sum()
    if abs(residual) > 1e-10:
        if residual > 0:
            idx = int(np.argmax(1.0 - h))

        else:

            h[idx] = max(0.0, h[idx] + residual)
    return h

    p = exact_profile(h)
    mu = np.max(p)
    w = np.exp((p - mu) / tau)
    w = w / w.sum()
    return w

    n = len(h)

    M = 2 * n - 1
    p = exact_profile(h)
    mu = np.max(p)
    w = np.exp((p - mu) / tau)
    w = w / w.sum()

    P_n = {v: 0<=v_i<=1, sum(v)=n/2}
    Optimal: v=1 where g is most negative (n/2 of them), v=0 elsewhere.
    """
    k = n // 2  # must be integer
    idx = np.argsort(g)
    v = np.zeros(n)
    v[idx[:k]] = 1.0
    return v

def project_onto_polytope(z, n):
    """Project z onto {h: 0<=h<=1, sum(h)=n/2}."""
    target = n / 2.0
    # Bisection for lambda
    lo, hi = -max(np.abs(z).max(), 1.0) - 1, max(np.abs(z).max(), 1.0) + 1
    for _ in range(60):

        s = np.clip(z + mid, 0.0, 1.0).sum()
        if s < target:

        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    # Fix residual

    if abs(residual) > 1e-12:

                add = min(residual, 1.0 - h[idx])
                h[idx] += add

import numpy as np
from scipy.optimize import minimize
import time

def exact_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def exact_c5(h):
    p = exact_profile(h)
    return float(np.max(p))

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(64):
        mid = (lo + hi) / 2.0
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        if residual > 0:
            idx = int(np.argmax(1.0 - h))
            h[idx] = min(1.0, h[idx] + residual)
        else:
            idx = int(np.argmax(h))
            h[idx] = max(0.0, h[idx] + residual)
    return h

def smooth_gradient(h, tau):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    p = exact_profile(h)
    mu = np.max(p)
    w = np.exp((p - mu) / tau)
    w = w / w.sum()
    grad = np.zeros(n)
    center = n - 1
    for idx in range(M):
        d = idx - center
        wm = w[idx]
        if wm < 1e-12:
            continue
        if d == 0:
            grad += wm * dx * (1.0 - 2.0 * h)
        elif d > 0:
            grad[d:n] += wm * dx * (1.0 - h[0:n - d])
            grad[0:n - d] -= wm * dx * h[d:n]
        else:
            s = -d
            grad[0:n - s] += wm * dx * (1.0 - h[s:n])
            grad[s:n] -= wm * dx * h[0:n - s]
    return grad

def fw_vertex(g, n):
    k = n // 2
    idx = np.argsort(g)
    v = np.zeros(n)
    v[idx[:k]] = 1.0
    return v

def fw_line_search_exact(h, v, current_c):
    alphas = [1.0, 0.8, 0.6, 0.5, 0.4, 0.3, 0.2, 0.15, 0.1, 0.07, 0.05, 0.03, 0.02]
    best_alpha = 0.0
    best_val = current_c
    for a in alphas:
        h_try = (1.0 - a) * h + a * v
        c = exact_c5(h_try)
        if c < best_val - 1e-13:
            best_val = c
            best_alpha = a
    if best_alpha == 0.0:
        return h, current_c
    return (1.0 - best_alpha) * h + best_alpha * v, best_val

def project_onto_polytope(z, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(64):
        mid = (lo + hi) / 2.0
        s = np.clip(z + mid, 0.0, 1.0).sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        if residual > 0:
            cand = np.where(h < 1.0)[0]
            if len(cand) > 0:
                idx = cand[np.argmax(1.0 - h[cand])]
                h[idx] = min(1.0, h[idx] + residual)
        else:
            cand = np.where(h > 0.0)[0]
            if len(cand) > 0:
                idx = cand[np.argmin(h[cand])]
                h[idx] = max(0.0, h[idx] + residual)
    return h

def fw_anneal(h_start, best_c, deadline, max_iters=250):
    n = len(h_start)
    h = h_start.copy()
    best_h = h_start.copy()
    best_c = min(best_c, exact_c5(h))
    
    tau_schedule = [0.25, 0.15, 0.10, 0.06, 0.04, 0.025, 0.015, 0.01, 0.007, 0.005]
    iters_per_tau = max(20, max_iters // len(tau_schedule))
    
    total_iter = 0
    for tau in tau_schedule:
        if time.perf_counter() > deadline - 5:
            break
        stall = 0
        for _ in range(iters_per_tau):
            if time.perf_counter() > deadline - 5 or total_iter >= max_iters:
                break
            total_iter += 1
            g = smooth_gradient(h, tau)
            v = fw_vertex(g, n)
            direction = v - h
            if np.allclose(direction, 0, atol=1e-8):
                stall += 1
                if stall > 4:
                    break
                continue

                h = h_new
                best_c = c_new

                stall = 0
            else:
                stall += 1
                if stall > 8:
                    break
    return best_h, best_c

def projected_grad(h, tau, best_c, deadline, n_iters=150):

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

def epigraph_solve(h_start, maxiter=200, deadline=None):

    dx = 2.0 / n
    c0 = exact_c5(h_start)

    center = n - 1

        return x[:-1].sum() - n / 2.0
    
    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g
    
    def ineq_fun(x):
        h = x[:-1]
        t = x[-1]
        return t - profile_vec(h)
    
    def ineq_jac(x):
        h = x[:-1]
        J = np.zeros((M, n + 1))
        J[:, -1] = 1.0
        for m in range(M):
            d = m - center
            if d == 0:
                J[m, :-1] = -dx * (1.0 - 2.0 * h)
            elif d > 0:
                J[m, d:n] = -dx * (1.0 - h[0:n - d])
                J[m, 0:n - d] = dx * h[d:n]
            else:
                s = -d
                J[m, 0:n - s] = -dx * (1.0 - h[s:n])
                J[m, s:n] = dx * h[0:n - s]
        return J
    
    x0 = np.concatenate([h_start, [c0 + 1e-3]])
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 1.0)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None and np.isfinite(res.x).all():
            h_new = repair_feasible(res.x[:-1])
            c_new = exact_c5(h_new)
            if c_new < c0 - 1e-10:
                return h_new
    except Exception:
        pass
    return None
