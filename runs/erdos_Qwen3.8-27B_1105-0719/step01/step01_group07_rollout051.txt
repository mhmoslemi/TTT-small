            grad[s:] -= w * dx * h[:n - s]
        else:
            a = -s
            grad[a:] += w * dx * (1.0 - h[:n - a])
            grad[:n - a] -= w * dx * h[a:]
    return obj, grad


class LSEObj:
    def __init__(self, tau):
        self.tau = tau
    def f(self, x):
        return lse_value_and_grad(x, self.tau)[0]
    def g(self, x):
        return lse_value_and_grad(x, self.tau)[1]


class SumEq:
    def __init__(self, target):
        self.target = target
    def f(self, x):
        return x.sum() - self.target
    def g(self, x):
        return np.ones(len(x))


def slsqp_lse(h0, tau, maxiter=400):
    n = len(h0)
    obj = LSEObj(tau)
    con = SumEq(n / 2.0)
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': con.f, 'jac': con.g}]
    try:
        res = minimize(obj.f, h0.copy(), jac=obj.g, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_box_sum(np.asarray(res.x, dtype=np.float64))
    except Exception:
        return h0.copy()


def refine_lse(h0, deadline):
    h = project_box_sum(h0.copy())
    best_c = exact_objective(h)
    taus = [0.01, 0.004, 0.0015, 0.0005, 0.00015, 0.00005, 0.00002, 0.000008]
    for tau in taus:
        if time.time() > deadline - 10:
            break
        h_new = slsqp_lse(h, tau, maxiter=min(500, len(h) * 8))
        c_new = exact_objective(h_new)
        if c_new < best_c - 1e-13:
            h = h_new
            best_c = c_new
    return h, best_c


def profile_jac_row(h, j, n):
    dx = 2.0 / n
    s = j - (n - 1)
    row = np.zeros(n)
    if s >= 0:
        if n - s > 0:
            row[0:n - s] += dx * (1.0 - h[s:s + (n - s)])
        if s < n and n - s > 0:
            row[s:n] -= dx * h[0:n - s]
    else:
        a = -s
        if n - a > 0:
            row[a:n] += dx * (1.0 - h[0:n - a])
        if n - a > 0:
            row[0:n - a] -= dx * h[a:a + (n - a)]
    return row


class EpigraphObj:
    def __init__(self, n):
        self.n = n
    def fun(self, x):
        return x[-1]
    def jac(self, x):
        g = np.zeros(self.n + 1)
        g[-1] = 1.0
        return g


class EpigraphIneq:
    def __init__(self, n):
        self.n = n
    def fun(self, x):
        h = x[:self.n]
        t = x[-1]
        profile = compute_profile(h)
        return t - profile
    def jac(self, x):
        h = x[:self.n]
        n = self.n
        nprof = 2 * n - 1
        J = np.ones((nprof, n + 1))
        J[:, :-1] = 0.0
        for j in range(nprof):
            J[j, :n] = -profile_jac_row(h, j, n)
        return J


class EpigraphEq:
    def __init__(self, n):
        self.n = n
    def fun(self, x):
        return x[:self.n].sum() - self.n / 2.0
    def jac(self, x):
        g = np.zeros(self.n + 1)
        g[:self.n] = 1.0
        return g


def slsqp_epigraph(h0, deadline, maxiter=300):
    n = len(h0)
    t0 = exact_objective(h0) * 1.01 + 0.001
    x0 = np.zeros(n + 1)
    x0[:n] = h0
    x0[n] = t0
    obj = EpigraphObj(n)
    ineq = EpigraphIneq(n)
    eq = EpigraphEq(n)
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constraints = [
        {'type': 'ineq', 'fun': ineq.fun, 'jac': ineq.jac},
        {'type': 'eq', 'fun': eq.fun, 'jac': eq.jac}
    ]
    try:
        res = minimize(obj.fun, x0, jac=obj.jac, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        if hasattr(res, 'x') and res.x is not None:
            h_new = project_box_sum(np.asarray(res.x[:n], dtype=np.float64))
            return h_new
    except Exception:
        pass
    return h0.copy()


def refine_epigraph(h0, deadline):
    h = project_box_sum(h0.copy())
    best_c = exact_objective(h)
    for trial in range(3):
        if time.time() > deadline - 10:
            break
        h_new = slsqp_epigraph(h, deadline, maxiter=200)
        c_new = exact_objective(h_new)
        if c_new < best_c - 1e-13:
            h = h_new
            best_c = c_new
        else:
            break
    return h, best_c


def nm_penalty_objective(x, n, lam):
    h = project_box_sum(x)
    return exact_objective(h)


def nm_optimize(h0, deadline, maxiter=3000):
    n = len(h0)
    h = project_box_sum(h0.copy())
    best_c = exact_objective(h)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(exact_objective, h.copy(), method='Nelder-Mead',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'xatol': 1e-10, 'fatol': 1e-12})
        h_new = project_box_sum(np.asarray(res.x, dtype=np.float64))
        c_new = exact_objective(h_new)
        if c_new < best_c:
            h = h_new
            best_c = c_new
    except Exception:
        pass
    return h, best_c


def binary_threshold(h):
    n = len(h)
    if n % 2 != 0:
        return None
    m = n // 2
    idx = np.argsort(h)[::-1][:m]
    b = np.zeros(n, dtype=np.float64)
    b[idx] = 1.0
    return b


def swap_delta(b, i, j, n):
    """Compute change in each profile entry when b[i]:1->0 and b[j]:0->1."""
    dx = 2.0 / n
    nprof = 2 * n - 1
    deltas = np.zeros(nprof)
    for jj in range(nprof):
        s = jj - (n - 1)
        if s >= 0:
            d = 0.0
            # j entering (0->1)
            if j + s < n:
                d += (1.0 - b[j + s])
            if j >= s:
                d -= b[j - s]
            # i leaving (1->0)
            if i + s < n:
                d -= (1.0 - b[i + s])
            if i >= s:
                d += b[i - s]
        else:
            a = -s
            d = 0.0
            if j >= a:
                d += (1.0 - b[j - a])
            if j + a < n:
                d -= b[j + a]
            if i >= a:
                d -= (1.0 - b[i - a])
            if i + a < n:
                d += b[i + a]
        deltas[jj] = d * dx
    return deltas


def binary_swap_search(b, deadline, max_rounds=500, use_2swap=False, rng=None):
    n = len(b)
    b = b.copy()
    curr_c = exact_objective(b)
    curr_profile = compute_profile(b)
    ones = list(np.where(b == 1.0)[0])
    zeros = list(np.where(b == 0.0)[0])
    
    for rnd in range(max_rounds):
        if time.time() > deadline - 5:
            break
        best_improve = 1e-14
        best_swap = None
        
        for i in ones:
            for j in zeros:
                if j == i:
                    continue
                deltas = swap_delta(b, i, j, n)
                new_profile = curr_profile + deltas
                new_max = new_profile.max()
                improve = curr_c - new_max
                if improve > best_improve:
                    best_improve = improve
                    best_swap = (i, j, new_profile)
        
        if best_swap is not None:
            i, j, new_profile = best_swap
            b[i] = 0.0
            b[j] = 1.0
            curr_c = float(new_profile.max())
            curr_profile = new_profile
            ones = list(np.where(b == 1.0)[0])
            zeros = list(np.where(b == 0.0)[0])
        else:
            break
        
        # Occasional 2-swap perturbation to escape local minima
        if use_2swap and rng is not None and len(ones) >= 2 and len(zeros) >= 2:
            if rnd % 10 == 9:
                i1, i2 = rng.choice(len(ones), 2, replace=False)
                j1, j2 = rng.choice(len(zeros), 2, replace=False)
                i1, i2 = ones[i1], ones[i2]
                j1, j2 = zeros[j1], zeros[j2]
                b[i1] = 0.0
                b[i2] = 0.0
                b[j1] = 1.0
                b[j2] = 1.0
                new_c = exact_objective(b)
                if new_c > curr_c + 0.005:
                    b[i1] = 1.0
                    b[i2] = 1.0
                    b[j1] = 0.0
                    b[j2] = 0.0
                else:
                    curr_c = new_c
                    curr_profile = compute_profile(b)
                    ones = list(np.where(b == 1.0)[0])
                    zeros = list(np.where(b == 0.0)[0])
    
    return b, curr_c


def transfer_to_size(h_parent, n_target):
    n_src = len(h_parent)
    if n_src == n_target:
        return project_box_sum(h_parent.copy())
    x_src = (np.arange(n_src) + 0.5) / n_src
    x_tgt = (np.arange(n_target) + 0.5) / n_tgt
    h_new = np.interp(x_tgt, x_src, h_parent)
    return project_box_sum(h_new)


def make_start(n, rng, mode):
    if mode == 0:
        h = np.zeros(n)
        h[:n // 2] = 1.0
        h = h[rng.permutation(n)]
    elif mode == 1:
        x = np.linspace(0, 1, n)
        freq = rng.choice([2, 3, 4, 5, 6, 7, 8])
        phase = rng.uniform(0, 2 * np.pi)
        h = 0.5 + 0.45 * np.sin(2 * np.pi * freq * x + phase)
        h = np.clip(h, 0, 1)
    elif mode == 2:
        h = np.zeros(n)
        h[::2] = 1.0
    elif mode == 3:
        h = rng.uniform(0, 1, n)
    elif mode == 4:
        h = np.zeros(n)
        block = max(2, n // 8)
        for i in range(0, n, 2 * block):
            h[i:i + block] = 1.0
    elif mode == 5:
        h = np.zeros(n)
        h[:n // 4] = 1.0
        h[3 * n // 4:] = 1.0
    elif mode == 6:
        h = np.zeros(n)
        h[::3] = 1.0
        h[1::3] = 0.5
    elif mode == 7:
        x = np.linspace(0, 1, n)
        h = 0.5 + 0.35 * np.sin(2 * np.pi * 3 * x) + 0.25 * np.sin(2 * np.pi * 7 * x + 1.0)
        h = np.clip(h, 0, 1)
    else:
        h = np.full(n, 0.5)
    return project_box_sum(h)


def shuffle_with_constraint(h, rng, n_swaps=5):
    n = len(h)
    h = h.copy()
    ones = list(np.where(h == 1.0)[0])
    zeros = list(np.where(h == 0.0)[0])
    if len(ones) < 2 or len(zeros) < 2:
        return h
    for _ in range(min(n_swaps, len(ones) // 2)):
        i = rng.randint(len(ones))
        j = rng.randint(len(zeros))
        if h[ones[i]] == 1.0 and h[zeros[j]] == 0.0:
            h[ones[i]] = 0.0
            h[zeros[j]] = 1.0
            i += 1
            j += 1
            if i < len(ones):
                ones[i-1], ones[-1] = ones[-1], ones[i-1]
                ones.pop()
            if j < len(zeros):
                zeros[j-1], zeros[-1] = zeros[-1], zeros[j-1]
                zeros.pop()
    return project_box_sum(h)


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    deadline = t_start + budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c = float('inf')

    # Load parent
    parent_h = None
    try:
        parent_h = np.array(globals()['initial_h_values'], dtype=np.float64).copy()
        if len(parent_h) > 0:
            parent_h = project_box_sum(parent_h)
            pc = exact_objective(parent_h)
            if pc < best_c:
                best_c = pc
                best_h = parent_h.copy()
    except Exception:
        pass

    if best_h is None:
        n = 48
        best_h = make_start(n, rng, 0)
        best_h = project_box_sum(best_h)
        best_c = exact_objective(best_h)

    parent_n = len(parent_h) if parent_h is not None else 48

    # Sizes to explore (prefer smaller, focus around parent)
    sizes = [parent_n]
    for s in [parent_n - 4, parent_n + 4, parent_n - 8, parent_n + 8,
              52, 56, 60, 64, 44, 40, 50, 54]:
        if s >= 20 and s <= 80 and s % 2 == 0 and s not in sizes:
            sizes.append(s)
    sizes = sorted(sizes)

    # Phase 1: For each size, try multiple methods
    for n in sizes:
        if time.time() > deadline - 40:
            break

        cands = []
        if parent_h is not None:
            cands.append(transfer_to_size(parent_h, n))
            cands.append(project_box_sum(transfer_to_size(parent_h, n) + rng.normal(0, 0.02, n)))
        for mode in range(8):
            cands.append(make_start(n, rng, mode))

        scored = []
        for c in cands:
            c_score = exact_objective(c)
            scored.append((c_score, c))
        scored.sort(key=lambda x: x[0])

        # Try top candidates with LSE
        n_top = min(4, len(scored))
        for idx in range(n_top):
            if time.time() > deadline - 35:
                break
            h0 = scored[idx][1]
            h_ref, c_ref = refine_lse(h0, deadline)
            if c_ref < best_c - 1e-12:
                best_c = c_ref
                best_h = h_ref.copy()

        # Try epigraph on best continuous
        if time.time() > deadline - 30:
            break
        h0 = scored[0][1]
        h_epi, c_epi = refine_epigraph(h0, deadline)
        if c_epi < best_c - 1e-12:
            best_c = c_epi
            best_h = h_epi.copy()

        # Binary optimization
        if n % 2 == 0 and time.time() < deadline - 20:
            b = binary_threshold(scored[0][1])
            if b is not None:
                b_sw, c_sw = binary_swap_search(b, deadline, max_rounds=300,
                                                 use_2swap=True, rng=rng)
                if c_sw < best_c - 1e-12:
                    best_c = c_sw
                    best_h = b_sw.copy()

            # Also try binary from second best
            if len(scored) > 1 and scored[1][0] < 0.39 and time.time() < deadline - 15:
                b2 = binary_threshold(scored[1][1])
                if b2 is not None:
                    b2_sw, c2_sw = binary_swap_search(b2, deadline, max_rounds=200,
                                                      use_2swap=True, rng=rng)
                    if c2_sw < best_c - 1e-12:
                        best_c = c2_sw
                        best_h = b2_sw.copy()

            # Binary from best continuous refinement (if we got one)
            if time.time() < deadline - 12:
                h_cont = project_box_sum(transfer_to_size(parent_h, n)) if parent_h is not None else scored[0][1]
                h_lr, _ = refine_lse(h_cont, deadline)
                b3 = binary_threshold(h_lr)
                if b3 is not None:
                    b3_sw, c3_sw = binary_swap_search(b3, deadline, max_rounds=200,
                                                      use_2swap=True, rng=rng)
                    if c3_sw < best_c - 1e-12:
                        best_c = c3_sw
                        best_h = b3_sw.copy()

    # Phase 2: Nelder-Mead on best solution
    if best_h is not None and time.time() < deadline - 20:
        n_b = len(best_h)
        h_nm, c_nm = nm_optimize(best_h.copy(), deadline, maxiter=2000)
        if c_nm < best_c - 1e-12:
            best_c = c_nm
            best_h = h_nm.copy()

    # Phase 3: Final intensive polish on best
    if best_h is not None and time.time() < deadline - 10:
        n_b = len(best_h)
        is_binary = np.all((best_h == 0.0) | (best_h == 1.0))
        if is_binary:
            b, c_b = binary_swap_search(best_h.copy(), deadline, max_rounds=500,
                                        use_2swap=True, rng=rng)
            if c_b < best_c - 1e-12:
                best_c = c_b
                best_h = b.copy()
        else:
            # LSE with very small tau
            h_ref, c_ref = refine_lse(best_h.copy(), deadline)
            if c_ref < best_c - 1e-12:
                best_c = c_ref
                best_h = h_ref.copy()
            # Try epigraph
            h_epi, c_epi = refine_epigraph(best_h.copy(), deadline)
            if c_epi < best_c - 1e-12:
                best_c = c_epi
                best_h = h_epi.copy()
            # Binary from best continuous
            if n_b % 2 == 0:
                b = binary_threshold(best_h)
                if b is not None:
                    b, c_b = binary_swap_search(b, deadline, max_rounds=300,
                                                use_2swap=True, rng=rng)
                    if c_b < best_c - 1e-12:
                        best_c = c_b
                        best_h = b.copy()

    # Final: one more LSE refinement on the best
    if best_h is not None and time.time() < deadline - 4:
        try:
            h_final = slsqp_lse(best_h, 0.00003, maxiter=300)
            c_final = exact_objective(h_final)
            if c_final < best_c - 1e-12:
                best_c = c_final
                best_h = h_final.copy()
        except Exception:
            pass

    # Final validation
    best_h = project_box_sum(best_h)
    final_c = exact_objective(best_h)
    if final_c < best_c + 1e-10:
        best_c = final_c
    best_n = len(best_h)

    return best_h, best_c, best_n
