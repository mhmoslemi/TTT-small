def exact_score(h):
    n = len(h)
    if n == 0:
        return float('inf')
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(profile))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n

def profile_grad(h, s):
    """Gradient of profile at offset s (where s = t - (n-1))"""
    n = len(h)
    g = np.zeros(n)
    if s == 0:
        g[:] = 1.0 - 2.0 * h
    elif s > 0:
        # h[i] * (1 - h[i+s]) for i = 0..n-s-1
        g[0:n-s] += (1.0 - h[s:n])
        g[s:n] -= h[0:n-s]
    else:
        a = -s
        # h[i] * (1 - h[i-a]) for i = a..n-1
        g[a:n] += (1.0 - h[0:n-a])
        g[0:n-a] -= h[a:n]
    return g

        ci = 0.0

            if i < n - s:
                ci += (1.0 - h[i+s])
            if i >= s:
                ci += -h[i-s]
