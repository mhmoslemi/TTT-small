```python
def projected_subgradient_step(h, step_size, n_active=10):
    """One step of projected subgradient method on C(h) = dx * max_k profile_k"""
    n = len(h)
    dx = 2.0 / n
    profile = np.correlate(h, 1-h, mode='full')
    # Find top lags
    k_max = np.argsort(profile)[-n_active:]
    # Average gradient over active lags
    grad = np.zeros(n)
    for t in k_max:
        s = t - (n-1)
        g = np.zeros(n)
        if s == 0:
            g[:] = 1 - 2*h
        elif s > 0:
            g[:n-s] += 1 - h[s:]  # d/dh[m] of h[m](1-h[m+s]) = 1-h[m+s]
            g[s:] -= h[:n-s]      # d/dh[m+s] of h[m](1-h[m+s]) = -h[m]
        else:
            a = -s
            g[a:] += 1 - h[:n-a]  # d/dh[m] of h[m](1-h[m-a]) where m>=a
            g[:n-a] -= h[a:]       # d/dh[m-a] of h[m](1-h[m-a])
        grad += g
    grad /= n_active
    grad *= dx
    # Take step
    h_new = h - step_size * grad
    # Project onto {0<=h<=1, sum=h_target}
    h_new = project_box_sum(h_new)
    return h_new
```

Actually, for the binary case, I can do something even simpler: **hill climbing with simulated annealing**. Start from a binary sequence, and at each step:
- Pick a random pair (i, j) with h[i]=1, h[j]=0
- Swap them
- Accept if it improves, or accept with probability exp(-Δ/T) if it worsens
- Decrease T over time

This is the standard approach for combinatorial optimization and what the previous program's `continuous_pair_swap` does but for binary variables.

def exact_objective(h):
    n = len(h)
    if n == 0:
        return float('inf')
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(profile))

def project_box_sum(h, target=None):
    # Water-filling projection
    n = len(h)
    h = np.array(h, dtype=np.float64)
    h = np.nan_to_num(h, nan=0.0, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    if target is None:
        target = n / 2.0
    for _ in range(100):
        s = np.sum(h)
        diff = s - target
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h > 0
            cnt = np.sum(mask)
            if cnt == 0: break
            step = min(diff / cnt, np.min(h[mask]))
            h[mask] -= step
        else:
            mask = h < 1.0
            cnt = np.sum(mask)
            if cnt == 0: break
            step = min(-diff / cnt, 1.0 - np.max(h[mask]))

def compute_subgradient(h, active_indices):
    """Compute subgradient of max_k profile_k at h, using given active indices."""

        else:
            a = -s

            grad[:n-a] -= h[a:]
    return grad / len(active_indices)
