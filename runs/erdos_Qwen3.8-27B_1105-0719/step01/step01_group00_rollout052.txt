import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time


def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
        else:
            # Distribute among all
            h += diff / n
    return np.clip(h, 0.0, 1.0)


def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def c5_of_h(h, dx):
    return compute_corrs(h).max() * dx


def grad_corr_k(h, k, n):
    """Gradient of correlation at index k with respect to h."""
    offset = k - (n - 1)
    grad = np.zeros(n)
    # h[i+m] * (1-h[i]) where m = offset
    # d/dh[j] of h[j]*(1-h[i]) for j = i+m: (1-h[i])
    # d/dh[j] of h[i+m]*(1-h[j]) for j = i: -h[i+m]
    i_lo = max(0, -offset)
    i_hi = min(n - 1, n - 1 - offset)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    # First term: h[idx+offset] * (1-h[idx]), derivative wrt h[idx+offset] is (1-h[idx])
    grad[idx + offset] += (1.0 - h[idx])
    # Second term: same product, derivative wrt h[idx] is -h[idx+offset]
    grad[idx] += (-h[idx + offset])
    return grad


def smooth_max_obj(h, dx, beta):
    corrs = compute_corrs(h)
    m = corrs.max()
    return (m + np.log(np.sum(np.exp(beta * (corrs - m)))) / beta) * dx


def smooth_max_grad(h, dx, beta):
    n = len(h)
    corrs = compute_corrs(h)
    m = corrs.max()
    shifted = corrs - m
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    w /= w_sum
    grad = np.zeros(n)
    for k in range(len(corrs)):
        if w[k] < 1e-14:
            continue
        grad += w[k] * grad_corr_k(h, k, n)
    return grad * dx


def local_slsqp(h0, dx, target_sum, beta, maxiter):
    n = len(h0)
    
    def obj(x):
        return smooth_max_obj(x, dx, beta)
    
    def obj_grad(x):
        return smooth_max_grad(x, dx, beta)
    
    def cons_fun(x):
        return x.sum() - target_sum
    
    def cons_jac(x):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    bounds = [(0.0, 1.0)] * n
    
    try:
        res = minimize(obj, h0, jac=obj_grad,
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        h_out = project_feasible(res.x, target_sum)
    except Exception:
        h_out = project_feasible(h0, target_sum)
    return h_out, c5_of_h(h_out, dx)


def binary_local_search(h0, dx, rng, max_iters=100000, patience=5000):
    """Local search on binary sequences by swapping 1s and 0s."""
    n = len(h0)
    target_sum = int(round(n / 2))
    
    # Snap to binary
    h = np.zeros(n)
    sorted_idx = np.argsort(-h0)
    h[sorted_idx[:target_sum]] = 1.0
    
    c5 = c5_of_h(h, dx)
    best_c5 = c5
    best_h = h.copy()
    corrs = compute_corrs(h)
    
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    
    no_improve = 0
    temp = 1.0  # Temperature for simulated annealing
    
    for it in range(max_iters):
        # Pick random 1 and 0 to swap
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        h[i], h[j] = 0.0, 1.0
        new_corrs = compute_corrs(h)
        new_c5 = new_corrs.max() * dx
        
        if new_c5 <= best_c5 + 1e-15:
            best_c5 = new_c5
            best_h = h.copy()
            corrs = new_corrs
            no_improve = 0
        else:
            # Metropolis acceptance
            delta = new_c5 - best_c5
            if rng.random() < np.exp(-delta / (temp * dx)):
                corrs = new_corrs
                no_improve = 0
            else:
                h[i], h[j] = 1.0, 0.0
        
        if no_improve > patience:
            break
        
        temp *= 0.999
        
        # Update ones and zeros sets
        if h[i] == 0.0 and h[j] == 1.0:
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
        elif h[i] == 1.0 and h[j] == 0.0:
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
    
    return best_h, best_c5


def coordinate_descent(h0, dx, target_sum, max_rounds=200, tol=1e-12):
    """Coordinate descent: optimize one variable at a time."""
    n = len(h0)
    h = h0.copy()
    best_c5 = c5_of_h(h, dx)
    
    for rd in range(max_rounds):
        improved = False

    return np.clip(h, 0.0, 1.0)


def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def c5_of_h(h, dx):
    return compute_corrs(h).max() * dx


def grad_corr_k(h, k, n):
    offset = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -offset)
    i_hi = min(n - 1, n - 1 - offset)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx + offset] += (1.0 - h[idx])
    grad[idx] += (-h[idx + offset])
    return grad


def smooth_max_obj(h, dx, beta):
    corrs = compute_corrs(h)
    m = corrs.max()
    return (m + np.log(np.sum(np.exp(beta * (corrs - m)))) / beta) * dx


def smooth_max_grad(h, dx, beta):
    n = len(h)
    corrs = compute_corrs(h)
    m = corrs.max()
    shifted = corrs - m
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    w /= w_sum
    grad = np.zeros(n)
    for k in range(len(corrs)):
        if w[k] < 1e-14:
            continue
        grad += w[k] * grad_corr_k(h, k, n)
    return grad * dx


def local_slsqp(h0, dx, target_sum, beta, maxiter):
    n = len(h0)

    def obj(x):
        return smooth_max_obj(x, dx, beta)

    def obj_grad(x):
        return smooth_max_grad(x, dx, beta)

    def cons_fun(x):
        return x.sum() - target_sum

    def cons_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    bounds = [(0.0, 1.0)] * n

    try:
        res = minimize(obj, h0, jac=obj_grad,
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        h_out = project_feasible(res.x, target_sum)
    except Exception:
        h_out = project_feasible(h0, target_sum)
    return h_out, c5_of_h(h_out, dx)


def compute_delta_binary(h, i, j, n):
    """Compute change in all correlations when h[i]:1->0 and h[j]:0->1."""
    ms = np.arange(-(n - 1), n)
    valid_i1 = (0 <= i - ms) & (i - ms <= n - 1)
    valid_i2 = (0 <= i + ms) & (i + ms <= n - 1)
    valid_j1 = (0 <= j - ms) & (j - ms <= n - 1)
    valid_j2 = (0 <= j + ms) & (j + ms <= n - 1)

    delta = np.zeros(2 * n - 1)
    idx_i1 = i - ms[valid_i1]
    delta[valid_i1] += -(1.0 - h[idx_i1])
    idx_i2 = i + ms[valid_i2]
    delta[valid_i2] += h[idx_i2]
    idx_j1 = j - ms[valid_j1]
    delta[valid_j1] += (1.0 - h[idx_j1])
    idx_j2 = j + ms[valid_j2]
    delta[valid_j2] += -h[idx_j2]
    return delta


def binary_local_search(h0, dx, rng, max_iters=300000, patience=8000):
    n = len(h0)
    target_sum = int(round(n / 2))

    h = np.zeros(n)
    sorted_idx = np.argsort(-h0)
    h[sorted_idx[:target_sum]] = 1.0

    corrs = compute_corrs(h)
    best_c5 = corrs.max() * dx
    best_h = h.copy()

    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]

    no_improve = 0
    temp = 0.05
    cooling = 0.9997

    for it in range(max_iters):
        if it % 5000 == 0:
            temp *= cooling

        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]

        delta = compute_delta_binary(h, i, j, n)
        new_corrs = corrs + delta
        new_max = new_corrs.max()
        new_c5 = new_max * dx

        if new_c5 <= best_c5 + 1e-15:
            h[i], h[j] = 0.0, 1.0
            corrs = new_corrs
            best_c5 = new_c5
            best_h = h.copy()
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            no_improve = 0
        else:
            delta_val = new_c5 - best_c5
            if temp > 1e-8 and rng.random() < np.exp(-delta_val / temp):
                h[i], h[j] = 0.0, 1.0
                corrs = new_corrs
                ones = np.where(h == 1.0)[0]
                zeros = np.where(h == 0.0)[0]
                no_improve = 0
            else:
                no_improve += 1

        if no_improve > patience:
            break

    return best_h, best_c5


def binary_steepest_descent(h0, dx, rng, max_sweeps=200):
    """Try all swaps per sweep, apply the best one."""
    n = len(h0)
    target_sum = int(round(n / 2))

    h = np.zeros(n)
    sorted_idx = np.argsort(-h0)
    h[sorted_idx[:target_sum]] = 1.0

    corrs = compute_corrs(h)
    best_c5 = corrs.max() * dx
    best_h = h.copy()

    for sweep in range(max_sweeps):
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        best_swap = None
        best_new_max = corrs.max()

        for i in ones:
            for j in zeros:
                delta = compute_delta_binary(h, i, j, n)
                new_max = (corrs + delta).max()
                if new_max < best_new_max - 1e-14:
                    best_new_max = new_max
                    best_swap = (i, j)

        if best_swap is None:
            break

        i, j = best_swap
        h[i], h[j] = 0.0, 1.0
        corrs = compute_corrs(h)
        new_c5 = corrs.max() * dx
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()

    return best_h, best_c5


def make_structured_start(n, target_sum, rng, mode=0):
    h = np.zeros(n)
    if mode == 0:
        phi = (1.0 + np.sqrt(5.0)) / 2.0
        for k in range(int(target_sum)):
            pos = int((k * phi * n) % n)
            h[pos] = 1.0
    elif mode == 1:
        for k in range(int(target_sum)):
            pos = int((k * 0.6180339887 * n) % n)
            h[pos] = 1.0
    elif mode == 2:
        # Sparse comb
        period = max(2, n // int(target_sum))
        for k in range(int(target_sum)):
            pos = (k * period) % n
            h[pos] = 1.0
    else:
        idx = rng.permutation(n)
        h[idx[:int(target_sum)]] = 1.0
    return project_feasible(h, target_sum)


def remaining_time(t0, budget_s):
    return budget_s - (time.time() - t0)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    best_h = np.array(initial_h_values, dtype=float)
    n = len(best_h)
    dx = 2.0 / n
    target_sum = n / 2.0
    best_h = project_feasible(best_h, target_sum)
    best_c5 = c5_of_h(best_h, dx)

    # Phase 1: SLSQP with increasing beta
    for beta in [50, 100, 200, 500, 1000]:
        if remaining_time(t0, budget_s) < 30:
            break
        h_opt, c5_val = local_slsqp(best_h, dx, target_sum, beta, 800)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_opt

    # Phase 2: Binary steepest descent (try all swaps, greedy)
    if remaining_time(t0, budget_s) > 150:
        h_bd, c5_bd = binary_steepest_descent(best_h, dx, rng, max_sweeps=50)
        if c5_bd < best_c5:
            best_c5 = c5_bd
            best_h = h_bd

    # Phase 3: Binary local search with simulated annealing
    if remaining_time(t0, budget_s) > 300:
        h_sa, c5_sa = binary_local_search(best_h, dx, rng, max_iters=300000, patience=8000)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa

    # Phase 4: SLSQP polish from binary
    if remaining_time(t0, budget_s) > 60:
        h_opt, c5_val = local_slsqp(best_h, dx, target_sum, 2000, 1000)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_opt

    # Phase 5: More binary local search (finer SA)
    if remaining_time(t0, budget_s) > 200:
        h_sa2, c5_sa2 = binary_local_search(best_h, dx, rng, max_iters=200000, patience=5000)
        if c5_sa2 < best_c5:
            best_c5 = c5_sa2
            best_h = h_sa2

    # Phase 6: Try structured initializations
    for mode in range(3):
        if remaining_time(t0, budget_s) < 80:
            break
        h_init = make_structured_start(n, target_sum, rng, mode)
        h_opt, c5_val = local_slsqp(h_init, dx, target_sum, 200, 400)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_opt
            if remaining_time(t0, budget_s) > 100:
                h_bd2, c5_bd2 = binary_steepest_descent(h_opt, dx, rng, max_sweeps=30)
                if c5_bd2 < best_c5:
                    best_c5 = c5_bd2
                    best_h = h_bd2

    # Phase 7: Final SLSQP + binary polish
    if remaining_time(t0, budget_s) > 50:
        h_opt, c5_val = local_slsqp(best_h, dx, target_sum, 5000, 1500)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_opt

    if remaining_time(t0, budget_s) > 100:
        h_final_bin, c5_final_bin = binary_local_search(best_h, dx, rng, max_iters=150000, patience=4000)
        if c5_final_bin < best_c5:
            best_c5 = c5_final_bin
            best_h = h_final_bin

    # Phase 8: Try n=128
    if remaining_time(t0, budget_s) > 150:
        n2 = 128
        dx2 = 2.0 / n2
        ts2 = n2 / 2.0
        x_orig = np.linspace(0.0, 2.0, len(best_h))
        x_new = np.linspace(0.0, 2.0, n2)
        h_interp = np.interp(x_new, x_orig, best_h)
        h_interp = project_feasible(h_interp, ts2)

        h_opt2, c5_val2 = local_slsqp(h_interp, dx2, ts2, 200, 600)
        if c5_val2 < best_c5:
            best_c5 = c5_val2
            best_h = h_opt2
            if remaining_time(t0, budget_s) > 80:
                h_bd3, c5_bd3 = binary_steepest_descent(h_opt2, dx2, rng, max_sweeps=40)
                if c5_bd3 < best_c5:
                    best_c5 = c5_bd3
                    best_h = h_bd3
                if remaining_time(t0, budget_s) > 60:
                    h_sa3, c5_sa3 = binary_local_search(best_h, dx2, rng, max_iters=100000, patience=5000)
                    if c5_sa3 < best_c5:
                        best_c5 = c5_sa3
                        best_h = h_sa3

    # Final cleanup
    n_b = len(best_h)
    dx_b = 2.0 / n_b
    ts_b = n_b / 2.0
    best_h = project_feasible(best_h, ts_b)
    best_c5 = c5_of_h(best_h, dx_b)

    return best_h, best_c5, len(best_h)
