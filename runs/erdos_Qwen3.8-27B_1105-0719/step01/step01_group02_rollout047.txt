    n = len(h)
    n_profiles = 2 * n - 1
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    
    # Find active profiles
    active = np.where(profile >= c - active_tol)[0]
    if len(active) == 0:
        active = np.where(profile >= c - 0.001)[0]
    
    # Compute gradients for active profiles
    grads = compute_all_grads(h, dx)
    active_grads = grads[active, :]  # shape (k, n)
    active_slacks = c - profile[active]  # shape (k,)
    
    # LP: maximize eps
    # x = [d_0, ..., d_{n-1}, eps]
    # minimize -eps
    # subject to:
    #   grad_α · d + eps ≤ slack_α  for each active α
    #   sum(d) = 0
    #   -h_i ≤ d_i ≤ 1-h_i
    #   eps ≥ 0
    #   eps ≤ 0.01 (upper bound to keep step small)
    
    k = len(active)
    c_obj = np.zeros(n + 1)
    c_obj[-1] = -1.0  # minimize -eps
    
    # Inequality constraints: A @ x ≤ b
    A = np.zeros((k, n + 1))
    A[:, :n] = active_grads
    A[:, n] = 1.0
    b = active_slacks.copy()
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    # Bounds
    bounds = []
    for i in range(n):
        bounds.append((-h[i], 1.0 - h[i]))
    bounds.append((0.0, 0.01))  # eps upper bound
    
    from scipy.optimize import linprog
    res = linprog(c_obj, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=b_eq, 
                  bounds=bounds, method='highs')
    
    if not res.success:
        return None, 0.0
    
    d = res.x[:n]
    eps = res.x[n]
    
    if eps < 1e-8:
        return None, 0.0
    
    # Line search
    best_h = h
    best_c = c
    for alpha in [1.0, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05]:

import numpy as np
from scipy.optimize import minimize, linprog
import time

def exact_objective(h):
    n = len(h)
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(profile) * dx)

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_box_sum(h, target=None):
    n = len(h)
    h = np.array(h, dtype=np.float64)
    h = np.nan_to_num(h, nan=0.0, posinf=1.0, neginf=0.0)
    if target is None:
        target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        s = np.sum(h)
        diff = s - target
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h > 0
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            step = min(diff / cnt, np.min(h[mask]))
            h[mask] -= step
        else:
            mask = h < 1.0
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            step = min(-diff / cnt, 1.0 - np.max(h[mask]))
            h[mask] += step
    diff = np.sum(h) - target
    if abs(diff) > 1e-12:
        mask = (h > 0) if diff > 0 else (h < 1.0)
        if np.any(mask):
            h[mask] -= diff / np.sum(mask)
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_grads_for_active(h, active_idx, dx):
    n = len(h)
    grads = np.zeros((len(active_idx), n))
    for idx, t in enumerate(active_idx):
        s = (n - 1) - t
        if s >= 0:
            if s < n:
                grads[idx, 0:n - s] += (1.0 - h[s:n])
                grads[idx, s:n] -= h[0:n - s]
        else:
            a = -s
            if a < n:
                grads[idx, a:n] += (1.0 - h[0:n - a])
                grads[idx, 0:n - a] -= h[a:n]
    return grads * dx

def lp_descent(h, c, max_steps=80, active_tol=0.004):
    n = len(h)
    dx = 2.0 / n
    cur = h.copy()
    cur_c = exact_objective(cur)
    for step_i in range(max_steps):
        profile = compute_profile(cur)
        c = float(np.max(profile))
        active = np.where(profile >= c - active_tol)[0]
        if len(active) == 0:
            break
        if len(active) > 80:
            active = np.where(profile >= c - active_tol * 2)[0]
        if len(active) == 0:
            break
        grads = compute_grads_for_active(cur, active, dx)
        slacks = c - profile[active]
        k = len(active)
        c_obj = np.zeros(n + 1)
        c_obj[-1] = -1.0
        A = np.zeros((k, n + 1))
        A[:, :n] = grads
        A[:, n] = 1.0
        b = slacks.copy()
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = [0.0]
        bounds = [(-cur[i] + 1e-12, 1.0 - cur[i] - 1e-12) for i in range(n)]
        bounds.append((0.0, 0.08))
        try:
            res = linprog(c_obj, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs',
                          options={'presolve': True, 'dual_feasibility_tolerance': 1e-8})
        except Exception:
            break
        if not res.success:
            break
        eps_val = res.x[n]
        if eps_val < 1e-10:
            break
        d = res.x[:n]
        improved = False
        for alpha in [1.0, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05, 0.02]:
            h_new = cur + alpha * d
            h_new = project_box_sum(h_new)
            c_new = exact_objective(h_new)
            if c_new < cur_c - 1e-13:
                cur = h_new
                cur_c = c_new
                improved = True
                break
        if not improved:
            if eps_val < 5e-6:
                break
    return cur, cur_c

    if n % 2 != 0:
        return None
    target = n // 2
    idx = np.argsort(-h)
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b

    return cur, cur_c

def binary_double_swap(b, n_trials=30000, rng_seed=42):

    cur = b.copy().astype(np.float64)

def binary_segment_reversal(b, max_len=10):
    n = len(b)
    cur = b.copy().astype(np.float64)
    cur_c = exact_objective(cur)
    for _ in range(50):
        improved = False
        best_delta = 1e-13
        best_seg = None
        for start in range(n):
            maxl = min(max_len, n - start)
            for length in range(2, maxl + 1):
                end = start + length - 1
                seg = cur[start:end + 1].copy()
                if np.array_equal(seg, seg[::-1]):
                    continue
                cur[start:end + 1] = seg[::-1]
                c = exact_objective(cur)
                cur[start:end + 1] = seg
                delta = cur_c - c

                    best_seg = (start, end)
        if best_seg is not None:

def smooth_obj_grad(x, tau, n):
    dx = 2.0 / n
    r = np.correlate(x, 1.0 - x, mode='full') * dx
    r_max = np.max(r)
    shifted = np.clip((r - r_max) / tau, -500.0, 500.0)
    exp_s = np.exp(shifted)
    log_sum = np.log(np.sum(exp_s))
    value = r_max + tau * log_sum
    weights = exp_s / np.sum(exp_s)
    grad = np.zeros(n)
    for i in range(2 * n - 1):
        k = i - (n - 1)
        w = weights[i]
        if w < 1e-14:
            continue
        if k == 0:
            grad += w * (1.0 - 2.0 * x)
        elif k > 0:
            grad[k:n] += w * (1.0 - x[0:n - k])
            grad[0:n - k] -= w * x[k:n]
        else:
            a = -k
            grad[0:n - a] += w * (1.0 - x[a:n])
            grad[a:n] -= w * x[0:n - a]
    grad *= dx
    return value, grad

def slsqp_refine(h0, n, max_tau_stages=5):
    cur = h0.copy()
    cur_c = exact_objective(cur)
    tau_stages = [0.01, 0.004, 0.0015, 0.0005, 0.0002]
    for tau in tau_stages[:max_tau_stages]:
        target = n / 2.0

        def obj_fun(x):
            v, g = smooth_obj_grad(x, tau, n)
            sv = np.sum(x) - target
            v = v + 1e5 * sv * sv
            g = g + 2e5 * sv * np.ones(n)
            return v, g

        def eq_fun(x):
            return np.sum(x) - target

        def eq_jac(x):
            return np.ones(n)

        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
        try:
            res = minimize(obj_fun, cur, method='SLSQP', jac=True, bounds=bounds,
                           constraints=cons, options={'maxiter': 100, 'ftol': 1e-11})
            h_new = project_box_sum(res.x)
            c_new = exact_objective(h_new)
            if c_new < cur_c - 1e-12:
                cur = h_new
                cur_c = c_new
        except Exception:
            pass
    return cur, cur_c
