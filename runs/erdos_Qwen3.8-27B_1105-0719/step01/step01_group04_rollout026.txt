import numpy as np
from scipy.optimize import minimize
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr * dx))

def repair_feasible(vec):
    h = np.asarray(vec, dtype=np.float64).copy()
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(h)
        rem = diff
        for i in order:
            if rem <= 1e-15:
                break
            add = min(rem, 1.0 - h[i])
            h[i] += add
            rem -= add
    else:
        order = np.argsort(-h)
        rem = -diff
        for i in order:
            if rem <= 1e-15:
                break
            remv = min(rem, h[i])
            h[i] -= remv
            rem -= remv
    # Final precision fix
    diff2 = target - h.sum()
    if abs(diff2) > 0:
        if diff2 > 0:
            for i in range(n):
                if h[i] < 1.0:
                    h[i] += min(diff2, 1.0 - h[i])
                    diff2 -= min(diff2, 1.0 - h[i])
                    if abs(diff2) < 1e-15:
                        break
        else:
            for i in range(n-1, -1, -1):
                if h[i] > 0.0:
                    h[i] += max(diff2, -h[i])
                    diff2 -= max(diff2, -h[i])
                    if abs(diff2) < 1e-15:
                        break
    return h

def compute_L_values(h, beta_e=100.0):
    """Compute L_m = dx * (A_m + smooth_min(P_m, S_m)) for m = 0..n-1"""
    n = len(h)
    dx = 2.0 / n
    L = np.zeros(n)
    
    # Prefix sums
    # P_m = sum_{i=0}^{m-1} h[i]
    # S_m = sum_{i=n-m}^{n-1} h[i]
    
    cumsum = np.concatenate([[0.0], np.cumsum(h)])
    
    for m in range(n):
        # A_m = sum_{i=0}^{n-m-1} h[i] * h[i+m]
        if m == 0:
            A_m = np.dot(h, h)
        else:
            A_m = np.dot(h[:n-m], h[m:])
        
        # P_m and S_m
        P_m = cumsum[m]  # sum h[0..m-1]
        S_m = cumsum[n] - cumsum[n-m]  # sum h[n-m..n-1]
        
        # Smooth min of P_m and S_m
        if m == 0:
            min_ps = 0.0
        else:
            # smooth_min(P_m, S_m) = -(1/beta_e) * log(exp(-beta_e*P_m) + exp(-beta_e*S_m))
            max_ps = max(P_m, S_m)
            if P_m == S_m:
                min_ps = P_m
            else:
                # For numerical stability
                diff_val = P_m - S_m
                if diff_val > 0:
                    min_ps = S_m - (1.0/beta_e) * np.log1p(np.exp(beta_e * diff_val))
                else:
                    min_ps = P_m - (1.0/beta_e) * np.log1p(np.exp(-beta_e * diff_val))
        
        L[m] = dx * (A_m + min_ps)
    
    return L

def compute_L_values_and_grad(h, beta_e=100.0):
    """Compute L values and their gradients w.r.t. h"""
    n = len(h)
    dx = 2.0 / n
    L = np.zeros(n)
    gradL = np.zeros((n, n))  # gradL[m, k] = dL_m/dh_k
    
    cumsum = np.concatenate([[0.0], np.cumsum(h)])
    total = h.sum()
    
    for m in range(n):
        # A_m gradient
        if m == 0:
            A_m = np.dot(h, h)
            gradA = 2.0 * h
        else:
            A_m = np.dot(h[:n-m], h[m:])
            gradA = np.zeros(n)
            gradA[:n-m] += h[m:]
            gradA[m:] += h[:n-m]
        
        if m == 0:
            min_ps = 0.0
            grad_min = np.zeros(n)
        else:
            P_m = cumsum[m]
            S_m = total - cumsum[n-m]
            
            # Smooth min weights
            diff_val = P_m - S_m
            if abs(diff_val) < 1e-10:
                wP = 0.5
                min_ps = P_m
            else:
                if diff_val > 0:
                    # P > S, min is closer to S
                    wP = 1.0 / (1.0 + np.exp(beta_e * diff_val))  # weight on P
                    min_ps = S_m - (1.0/beta_e) * np.log1p(np.exp(beta_e * diff_val))
                else:
                    # S > P, min is closer to P
                    wP = 1.0 / (1.0 + np.exp(beta_e * diff_val))  # weight on P
                    min_ps = P_m - (1.0/beta_e) * np.log1p(np.exp(-beta_e * diff_val))
            
            wS = 1.0 - wP
            
            # Gradient of min_ps w.r.t. h
            grad_min = np.zeros(n)
            if m > 0:
                grad_min[:m] += wP  # dP_m/dh_k = 1 for k < m
                grad_min[n-m:] += wS  # dS_m/dh_k = 1 for k >= n-m
        
        L[m] = dx * (A_m + min_ps)
        gradL[m, :] = dx * (gradA + grad_min)
    
    return L, gradL

def smooth_objective(h, beta_lag=50.0, beta_e=200.0):
    """Compute -smooth_min(L) and its gradient"""
    n = len(h)
    L, gradL = compute_L_values_and_grad(h, beta_e)
    
    # smooth_min(L) = -(1/beta) * log(sum exp(-beta * L_m))
    # For numerical stability
    L_min = np.min(L)
    shifted = -beta_lag * (L - L_min)
    max_shift = np.max(shifted)
    log_sum = max_shift + np.log(np.sum(np.exp(shifted - max_shift)))
    smooth_min_val = -L_min - (1.0/beta_lag) * log_sum
    
    # Weights
    exp_shifted = np.exp(shifted - max_shift)
    weights = exp_shifted / np.sum(exp_shifted)
    
    # Gradient: d smooth_min / dh_k = sum_m weights_m * dL_m/dh_k
    grad = np.sum(weights[:, None] * gradL, axis=0)
    
    # We want to minimize -smooth_min, so negate
    return -smooth_min_val, -grad

def solve_smooth_slsqp(h_start, beta_lag=50.0, beta_e=200.0, maxiter=200):
    n = len(h_start)
    h_start = repair_feasible(h_start)
    
    def obj(x):
        val, _ = smooth_objective(x, beta_lag, beta_e)
        return val
    
    def obj_grad(x):
        _, grad = smooth_objective(x, beta_lag, beta_e)
        return grad
    
    def eq_fun(x):
        return np.sum(x) - n / 2.0
    
    def eq_jac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    
    try:
        result = minimize(
            obj, h_start, jac=obj_grad,
            method='SLSQP', bounds=bounds, constraints=constraints,
            options={'maxiter': maxiter, 'ftol': 1e-14}
        )
    except Exception:
        return None, None
    
    if result.x is not None:
        h_sol = repair_feasible(result.x)
        score = exact_score(h_sol)
        if np.isfinite(score):
            return h_sol, score
    return None, None
