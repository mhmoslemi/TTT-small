```python
from scipy.optimize import minimize
import time

def exact_objective(h):
    n = len(h)
    if n == 0:
        return float('inf'), np.array([])
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(profile)), profile

def project_box_sum(h, target=None):
    n = len(h)
    if target is None:
        target = 0.5 * n
    h = np.array(h, dtype=np.float64).copy()
    h = np.nan_to_num(h, nan=0.0, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            count = np.sum(mask)
            if count == 0:
                break
            step = diff / count
            h[mask] = np.minimum(h[mask] + step, 1.0)
        else:
            mask = h > 0.0
            count = np.sum(mask)
            if count == 0:
                break
            step = diff / count
            h[mask] = np.maximum(h[mask] + step, 0.0)
    return h
```

```python
def project_box_sum(h, target=None):
    n = len(h)
    if target is None:
        target = 0.5 * n
    h = np.array(h, dtype=np.float64).copy()
    h = np.nan_to_num(h, nan=0.0, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            count = np.sum(mask)
            if count == 0:
                break
            step = diff / count
            h[mask] = np.minimum(h[mask] + step, 1.0)
        else:
            mask = h > 0.0
            count = np.sum(mask)
            if count == 0:
                break
            step = diff / count
            h[mask] = np.maximum(h[mask] + step, 0.0)
    return h

            old_val = h[i]

            # For lag offset e:
            #   If e > 0: h_i appears as first index in q_e if i < n-e
            #             h_i appears as second index in q_{-e} if i >= e
            #   If e < 0 (i.e., offset = -|e|): h_i appears as first index if i >= |e|
            #             h_i appears as second index in q_{|e|} if i < n-|e|
            
            # Actually let me compute the base and slope for each lag
            # q_e = dx * sum_{valid pairs} h_a * (1 - h_b)
            # When h_i = t:
            #   For pairs where i is 'a': contribution is t * (1 - h_b) = t * c1
            #   For pairs where i is 'b': contribution is h_a * (1 - t) = h_a - t * h_a
            # So q_e(t) = base_e + t * slope_e (for e != 0)
            # For e = 0: q_0 contains h_i*(1-h_i) = t - t^2, so q_0(t) = base_0 + t - t^2 (times dx)
            
            # Let me compute by evaluating q at h_i = 0 and h_i = 1
            h[i] = 0.0
            profile_0 = np.correlate(h, 1.0 - h, mode='full') * dx
            h[i] = 1.0
            profile_1 = np.correlate(h, 1.0 - h, mode='full') * dx
            h[i] = old_val
            
            # For e != 0: q_e(t) = profile_0[e] + t * (profile_1[e] - profile_0[e])
            # For e = 0: q_0(t) = profile_0[0] + t * (1 - t) * dx... 
            # Actually, let me just interpolate:
            # q_e(t) = profile_0[e] + t * (profile_1[e] - profile_0[e]) for all e
            # But this is only exact for e != 0. For e = 0, it's quadratic.
            # However, since we're taking the max, and the quadratic is concave,
            # the linear interpolation gives an upper bound on the quadratic for e=0.
            # So using linear interpolation is conservative (gives upper bound).
            # This is fine for minimization.
            
            # Actually wait: for e=0, q_0(t) = profile_0[n-1] + t*(profile_1[n-1] - profile_0[n-1])
            # is the LINEAR interpolation between the two endpoints.
            # The true q_0(t) is quadratic: it's profile_0[n-1] + dx*(t - t^2)
            # The linear interpolation is profile_0[n-1] + t * dx (since profile_1[n-1] - profile_0[n-1] = dx)
            # The true value is profile_0[n-1] + dx*t - dx*t^2
            # The difference is dx*t - dx*t^2 vs dx*t, so the linear interpolation overestimates q_0 by dx*t^2.
            # This means our "max" is slightly overestimated, which is conservative.
            # For finding the minimum, this might shift the optimum slightly but not by much.
            
            # Compute max over lags for each t in the grid
            slopes = profile_1 - profile_0  # shape (2n-1,)
            # q_e(t) ≈ profile_0[e] + t * slopes[e]
            # For each t, max_e (profile_0[e] + t * slopes[e])
            
            # This is the max of linear functions of t, which is convex.
            # Evaluate on grid:
            t_vals = t_grid
            # max over e of (profile_0[e] + t * slopes[e]) for each t
            # shape: (n_grid, 2n-1) -> max over axis 1
            values = profile_0[None, :] + t_vals[:, None] * slopes[None, :]
            max_values = np.max(values, axis=1)
            
            best_t_idx = np.argmin(max_values)
            best_t = t_vals[best_t_idx]
            best_val_at_t = max_values[best_t_idx]
            
            if best_val_at_t < best_c - 1e-12:
                h[i] = best_t
                improved = True
                # Update best_c
                # (We'll recompute at the end of the sweep for accuracy)
        
        # Project and recompute
        h = project_box_sum(h, 0.5 * n)
        c, _ = exact_objective(h)
        if c < best_c:
            best_c = c
        else:
            # Restore if projection made things worse
            h = project_box_sum(h, 0.5 * n)
        
        if not improved and sweep > 3:
            break
    
    return h, best_c
```

Actually, let me simplify: instead of a grid, I can analytically find the minimum of the upper envelope of linear functions. The upper envelope of linear functions is a convex piecewise-linear function. Its minimum is at a breakpoint (where two lines cross) or at the boundary.

For m linear functions f_j(t) = a_j + b_j * t, the upper envelope changes at breakpoints t* where f_i(t*) = f_j(t*) for some i,j. There are at most O(m^2) breakpoints. Actually, there's an even simpler approach: since we're just looking at the max of linear functions, we can use the fact that the minimum of the max is the smallest c such that a_j + b_j * t ≤ c for all j, for some t ∈ [0,1]. This is equivalent to:

For a given c, the feasible t values satisfy:
- For b_j > 0: t ≤ (c - a_j) / b_j
- For b_j < 0: t ≥ (c - a_j) / b_j
- For b_j = 0: c ≥ a_j

So the feasible interval for t is [max_{b_j<0} (c-a_j)/b_j, min_{b_j>0} (c-a_j)/b_j] ∩ [0,1].

The minimum c is the smallest c such that this interval is non-empty. We can binary search on c!

For a given c:
- lower_bound = max(0, max over j with b_j<0 of (c - a_j)/b_j)
- upper_bound = min(1, min over j with b_j>0 of (c - a_j)/b_j)
- Also need c >= max over j with b_j=0 of a_j
- Feasible if lower_bound ≤ upper_bound

Binary search on c: O(m) per iteration, about 50 iterations for good precision. Total per coordinate: O(m * 50) = O(139 * 50) ≈ 7000. For 70 coordinates: 490k per sweep. Very fast.

```python
def optimize_coordinate(profile_0, profile_1, dx):
    """Find optimal value of h_i given profiles at h_i=0 and h_i=1.
    Returns the optimal t in [0,1] that minimizes max_e (profile_0[e] + t*(profile_1[e]-profile_0[e])).
    """
    a = profile_0  # values at t=0
    b = profile_1 - profile_0  # slopes (values at t=1 minus values at t=0)
    
    # Binary search on c
    # For feasibility: need t ∈ [0,1] such that a_e + b_e * t ≤ c for all e
    # This means:
    #   For b_e > 0: t ≤ (c - a_e) / b_e
    #   For b_e < 0: t ≥ (c - a_e) / b_e  (note: dividing by negative flips inequality)
    #   For b_e = 0: need a_e ≤ c
    
    c_low = np.max(a)  # at t=0, max is max(a)
    c_high = np.max(profile_1)  # at t=1, max is max(profile_1)
    
    # Actually c_low should be min of max(a) and max(profile_1)
    # and c_high should be max of max(a) and max(profile_1)
    # But we want the minimum c, so:
    c_low = 0.0
    c_high = max(np.max(a), np.max(a + b))  # upper bound
    
    # Binary search
    for _ in range(60):
        c_mid = 0.5 * (c_low + c_high)
        # Check feasibility: is there t ∈ [0,1] with a + b*t ≤ c_mid for all?
        # Lower bound on t: max(0, max over b<0 of (c_mid - a)/b)
        # Upper bound on t: min(1, min over b>0 of (c_mid - a)/b)
        
        mask_pos = b > 1e-15
        mask_neg = b < -1e-15
        mask_zero = ~mask_pos & ~mask_neg
        
        lb = 0.0
        if np.any(mask_neg):
            lb = max(lb, np.max((c_mid - a[mask_neg]) / b[mask_neg]))
        
        ub = 1.0
        if np.any(mask_pos):
            ub = min(ub, np.min((c_mid - a[mask_pos]) / b[mask_pos]))
        
        if np.any(mask_zero):
            if np.max(a[mask_zero]) > c_mid + 1e-15:
                c_low = c_mid
                continue
        
        if lb <= ub + 1e-15:
            c_high = c_mid
        else:
            c_low = c_mid
    
    c_opt = c_high
    # Now find the optimal t
    mask_pos = b > 1e-15
    mask_neg = b < -1e-15
    
    lb = 0.0
    if np.any(mask_neg):
        lb = max(lb, np.max((c_opt - a[mask_neg]) / b[mask_neg]))
    
    ub = 1.0
    if np.any(mask_pos):
        ub = min(ub, np.min((c_opt - a[mask_pos]) / b[mask_pos]))
    
    lb = max(0.0, min(1.0, lb))
    ub = max(0.0, min(1.0, ub))
    
    if lb > ub:
        # Numerical issue, use midpoint
        t_opt = 0.5 * (lb + ub)
    else:
        t_opt = 0.5 * (lb + ub)  # Any value in [lb, ub] works; use midpoint
    
    return np.clip(t_opt, 0.0, 1.0), c_opt
```

import numpy as np
from scipy.optimize import minimize
import time

def exact_objective(h):
    n = len(h)
    if n == 0:
        return float('inf'), np.array([])
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(profile)), profile

def project_box_sum(h, target=None):
    n = len(h)
    if target is None:
        target = 0.5 * n
    h = np.array(h, dtype=np.float64).copy()
    h = np.nan_to_num(h, nan=0.0, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            count = np.sum(mask)
            if count == 0:
                break
            step = diff / count
            h[mask] = np.minimum(h[mask] + step, 1.0)
        else:
            mask = h > 0.0
            count = np.sum(mask)
            if count == 0:
                break
            step = diff / count
            h[mask] = np.maximum(h[mask] + step, 0.0)
    return h

def find_optimal_t(profile_0, profile_1):
    """Find t in [0,1] minimizing max_r (profile_0[r] + t*(profile_1[r]-profile_0[r]))."""
    a = profile_0
    b = profile_1 - profile_0
    
    c_lo = 0.0
    c_hi = max(np.max(a), np.max(a + b)) + 1e-12
    
    for _ in range(50):
        c_mid = 0.5 * (c_lo + c_hi)
        mask_pos = b > 1e-14
        mask_neg = b < -1e-14
        mask_zero = ~(mask_pos | mask_neg)
        
        feasible = True
        if np.any(mask_zero) and np.max(a[mask_zero]) > c_mid + 1e-14:
            feasible = False
        else:
            lb = 0.0
            if np.any(mask_neg):
                lb = max(lb, np.max((c_mid - a[mask_neg]) / b[mask_neg]))
            ub = 1.0
            if np.any(mask_pos):
                ub = min(ub, np.min((c_mid - a[mask_pos]) / b[mask_pos]))
            if lb > ub + 1e-14:
                feasible = False
        
        if feasible:
            c_hi = c_mid
        else:
            c_lo = c_mid
    
    c_opt = c_hi
    mask_pos = b > 1e-14
    mask_neg = b < -1e-14
    
    lb = 0.0
    if np.any(mask_neg):
        lb = max(lb, np.max((c_opt - a[mask_neg]) / b[mask_neg]))
    ub = 1.0
    if np.any(mask_pos):
        ub = min(ub, np.min((c_opt - a[mask_pos]) / b[mask_pos]))
    
    t = 0.5 * (max(0.0, lb) + min(1.0, ub))
    return float(np.clip(t, 0.0, 1.0))

def coord_descent(h, max_sweeps=50, deadline=None, direction='forward'):
    n = len(h)
    dx = 2.0 / n
    h = project_box_sum(h, 0.5 * n).copy()
    best_c, _ = exact_objective(h)
    
    indices = range(n) if direction == 'forward' else range(n-1, -1, -1)
    
    for sweep in range(max_sweeps):
        if deadline is not None and time.time() > deadline - 5:
            break
        improved = False
        for i in indices:
            old = h[i]
            h[i] = 0.0
            p0 = np.correlate(h, 1.0 - h, mode='full') * dx
            h[i] = 1.0
            p1 = np.correlate(h, 1.0 - h, mode='full') * dx

            t_opt = find_optimal_t(p0, p1)

                improved = True

        c, _ = exact_objective(h)
        if c < best_c:
            best_c = c
        if not improved and sweep >= 2:
            break
    
    return h, best_c

def make_alternating(n, rng=None):
    h = np.zeros(n)
    h[::2] = 1.0
    # Adjust to get exact sum
    n_ones = np.sum(h > 0.5)
    target = n // 2
    if n_ones > target:
        # Remove some ones
        excess = n_ones - target
        ones = np.where(h > 0.5)[0]
        h[ones[:excess]] = 0.0
    elif n_ones < target:
        deficit = target - n_ones
        zeros = np.where(h == 0.0)[0]
        h[zeros[:deficit]] = 1.0
    return project_box_sum(h, 0.5 * n)

def make_sinusoidal(n, rng, freq=1.0, amp=0.5):
    x = np.arange(n) / n
    phase = rng.uniform(0, 2 * np.pi)
    h = 0.5 + amp * np.cos(2 * np.pi * freq * x + phase)
    return project_box_sum(h, 0.5 * n)

def make_random_binary(n, rng):
    h = np.zeros(n)
    idx = rng.choice(n, size=n//2, replace=False)
    h[idx] = 1.0
    return project_box_sum(h, 0.5 * n)

def downsample(h, new_n):
    n = len(h)
    if new_n == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(new_n) + 0.5) / new_n
    vals = np.interp(x_new, x_old, h)
    return project_box_sum(vals, 0.5 * new_n)

    idx = np.argsort(-h)
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b
