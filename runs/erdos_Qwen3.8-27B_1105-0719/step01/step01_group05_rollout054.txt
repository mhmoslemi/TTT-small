import numpy as np
from scipy.optimize import minimize
import time

def exact_objective(h):
    n = len(h)
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(profile)), profile

def project_box_sum(h, target=None):
    n = len(h)
    if target is None:
        target = 0.5 * n
    h = np.array(h, dtype=np.float64).copy()
    h = np.nan_to_num(h)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-16 and n > 0:
        # spread among non-boundary elements
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
            h = np.clip(h, 0.0, 1.0)
        else:
            h[0] = np.clip(h[0] + diff, 0.0, 1.0)
    return h

def gradient_at_lag(h, lag, dx):
    """Gradient of the profile element at the given lag offset."""
    n = len(h)
    grad = np.zeros(n)
    if lag == 0:
        grad = (1.0 - 2.0 * h) * dx
    elif lag > 0:
        grad[lag:] += (1.0 - h[:n-lag]) * dx
        grad[:n-lag] += (-h[lag:]) * dx
    else:
        s = -lag
        grad[:n-s] += (1.0 - h[s:]) * dx
        grad[s:] += (-h[:n-s]) * dx
    return grad

def slsqp_smooth(h_start, n, tau_list, deadline, max_iter=100):
    """SLSQP with log-sum-exp smoothing."""
    best_h = project_box_sum(h_start, 0.5 * n)
    best_c, _ = exact_objective(best_h)
    target = 0.5 * n
    bounds = [(0.0, 1.0)] * n
    
    def eq_fun(x):
        return np.sum(x) - target
    
    def eq_jac(x):
        return np.ones(n)
    
    constraint = {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}
    
    for tau in tau_list:
        if time.time() > deadline - 3:
            break
        
        def obj_grad(x, _tau=tau):
            n_x = len(x)
            dx = 2.0 / n_x
            profile = np.correlate(x, 1.0 - x, mode='full') * dx
            pmax = np.max(profile)
            if _tau < 1e-12:
                m_idx = int(np.argmax(profile))
                lag = m_idx - (n_x - 1)
                grad = gradient_at_lag(x, lag, dx)
                se = np.sum(x) - target
                val = pmax + 1e4 * se * se
                grad = grad + 2e4 * se * np.ones(n_x)
                below = np.maximum(0.0, -x)
                above = np.maximum(0.0, x - 1.0)
                val += 1e4 * (np.sum(below*below) + np.sum(above*above))
                grad = grad - 2e4 * below + 2e4 * above
                return val, grad
            shifted = (profile - pmax) / _tau
            shifted = np.clip(shifted, -80, 80)
            weights = np.exp(shifted)
            wsum = np.sum(weights)
            value = pmax + _tau * np.log(wsum)
            wnorm = weights / wsum
            grad = np.zeros(n_x)
            active = wnorm > 1e-6
            for m_idx in np.where(active)[0]:
                lag = m_idx - (n_x - 1)
                grad += wnorm[m_idx] * gradient_at_lag(x, lag, dx)
            # Add penalty for sum and box violations
            se = np.sum(x) - target
            value += 1e4 * se * se
            grad += 2e4 * se * np.ones(n_x)
            below = np.maximum(0.0, -x)
            above = np.maximum(0.0, x - 1.0)
            value += 1e4 * (np.sum(below*below) + np.sum(above*above))
            grad -= 2e4 * below
            grad += 2e4 * above
            return value, grad
        
        try:
            res = minimize(obj_grad, best_h, method='SLSQP',
                           bounds=bounds, constraints=constraint, jac=True,
                           options={'maxiter': max_iter, 'ftol': 1e-14})
            if res.x is not None and np.all(np.isfinite(res.x)):
                cand = project_box_sum(res.x, target)
                c, _ = exact_objective(cand)
                if c < best_c - 1e-12:
                    best_c = c
                    best_h = cand
        except Exception:
            pass
    
    return best_h, best_c

def slsqp_direct(h_start, n, deadline, max_iter=50):
    """Direct constrained NLP: minimize c s.t. q_r(h) - c <= 0."""
    best_h = project_box_sum(h_start, 0.5 * n)
    best_c, profile = exact_objective(best_h)
    target = 0.5 * n
    # Only use constraints that are near-active (within 0.01 of max)
    active_mask = profile > best_c - 0.01
    active_indices = np.where(active_mask)[0]
    
    # Variable: z = [h_0, ..., h_{n-1}, c]
    n_total = n + 1
    
    def obj(z):
        return z[-1]  # minimize c
    
    def obj_grad(z):
        g = np.zeros(n_total)
        g[-1] = 1.0
        return g
    
    # Equality: sum(h) = target
    def eq_fun(z):
        return np.sum(z[:n]) - target
    
    def eq_jac(z):
        j = np.zeros(n_total)
        j[:n] = 1.0
        return j
    
    eq_constraint = {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}
    
    # Inequality constraints: c - q_r(h) >= 0 for active r
    def ineq_fun(z):
        h = z[:n]
        c = z[-1]
        prof = np.correlate(h, 1.0 - h, mode='full') * (2.0/n)
        return c - prof[active_indices]
    
    def ineq_jac(z):
        h = z[:n]
        dx = 2.0 / n
        m = len(active_indices)
        J = np.zeros((m, n_total))
        for i, r in enumerate(active_indices):
            lag = int(r) - (n - 1)
            J[i, :n] = -gradient_at_lag(h, lag, dx)  # -grad(q_r)
            J[i, -1] = 1.0  # +1 for c
        return J
    
    ineq_constraint = {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    
    bounds = [(0.0, 1.0)] * n + [(best_c - 0.05, best_c + 0.01)]
    
    z0 = np.concatenate([best_h, [best_c + 1e-6]])
    
    try:
        res = minimize(obj, z0, method='SLSQP',
                       bounds=bounds,
                       constraints=[eq_constraint, ineq_constraint],
                       jac=obj_grad,
                       options={'maxiter': max_iter, 'ftol': 1e-14})
        if res.x is not None and np.all(np.isfinite(res.x)):
            cand = project_box_sum(res.x[:n], target)
            c, _ = exact_objective(cand)
            if c < best_c - 1e-12:
                best_c = c
                best_h = cand
    except Exception:
        pass
    
    return best_h, best_c

def gradient_descent_active(h, n, steps=50, deadline=None):
    """Gradient descent targeting the maximum profile element."""
    target = 0.5 * n
    dx = 2.0 / n
    best_h = project_box_sum(h, target)
    best_c, profile = exact_objective(best_h)
    
    for step in range(steps):
        if deadline is not None and time.time() > deadline:
            break
        # Find top active lags
        top_k = min(5, 2*n-1)
        top_idx = np.argsort(-profile)[:top_k]
        
        # Compute weighted gradient
        weights = np.exp((profile[top_idx] - profile[top_idx][0]) / 0.01)
        weights /= np.sum(weights)
        
        grad = np.zeros(n)
        for w, r in zip(weights, top_idx):
            lag = int(r) - (n - 1)
            grad += w * gradient_at_lag(best_h, lag, dx)
        
        # Try several step sizes
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-12:
            break
        grad_unit = grad / grad_norm
        
        found_improvement = False
        for alpha in [0.001, 0.003, 0.005, 0.01, 0.02, 0.03]:
            cand = best_h - alpha * grad_unit
            cand = project_box_sum(cand, target)
            c, profile_new = exact_objective(cand)
            if c < best_c - 1e-13:
                best_c = c
                best_h = cand
                profile = profile_new
                found_improvement = True
                break
        
        if not found_improvement:
            break
    
    return best_h, best_c

def binary_threshold(h):

    # Integrate over each new cell

        else:

import numpy as np
from scipy.optimize import minimize
import time

def exact_objective(h):
    n = len(h)
    if n == 0:
        return float('inf'), np.array([])
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(profile)), profile

def project_box_sum(h, target=None):
    n = len(h)
    if target is None:
        target = 0.5 * n
    h = np.array(h, dtype=np.float64).copy()
    h = np.nan_to_num(h, nan=0.0, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-16 and n > 0:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
            h = np.clip(h, 0.0, 1.0)
        else:
            h[0] = np.clip(h[0] + diff, 0.0, 1.0)
    return h

def gradient_at_lag(h, lag, dx):
    n = len(h)
    grad = np.zeros(n)
    if lag == 0:
        grad = (1.0 - 2.0 * h) * dx
    elif lag > 0:
        grad[lag:] += (1.0 - h[:n - lag]) * dx
        grad[:n - lag] += (-h[lag:]) * dx
    else:
        s = -lag
        grad[:n - s] += (1.0 - h[s:]) * dx
        grad[s:] += (-h[:n - s]) * dx
    return grad

def slsqp_smooth(h_start, n, tau_list, deadline, max_iter=100):
    best_h = project_box_sum(h_start, 0.5 * n)
    best_c, _ = exact_objective(best_h)
    target = 0.5 * n
    bounds = [(0.0, 1.0)] * n

    def eq_fun(x):
        return np.sum(x) - target

    def eq_jac(x):
        return np.ones(n)

    constraint = {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}

    for tau in tau_list:
        if time.time() > deadline - 3:
            break

        def obj_grad(x, _tau=tau):
            n_x = len(x)
            dx = 2.0 / n_x
            profile = np.correlate(x, 1.0 - x, mode='full') * dx
            pmax = np.max(profile)
            if _tau < 1e-12:
                m_idx = int(np.argmax(profile))
                lag = m_idx - (n_x - 1)
                grad = gradient_at_lag(x, lag, dx)
                se = np.sum(x) - target
                val = pmax + 1e5 * se * se
                grad = grad + 2e5 * se * np.ones(n_x)
                below = np.maximum(0.0, -x)
                above = np.maximum(0.0, x - 1.0)
                val += 1e5 * (np.sum(below * below) + np.sum(above * above))
                grad = grad - 2e5 * below + 2e5 * above
                return val, grad
            shifted = (profile - pmax) / _tau
            shifted = np.clip(shifted, -80, 80)
            weights = np.exp(shifted)
            wsum = np.sum(weights)
            value = pmax + _tau * np.log(wsum)
            wnorm = weights / wsum
            grad = np.zeros(n_x)
            active = wnorm > 1e-6
            for m_idx in np.where(active)[0]:
                lag = int(m_idx) - (n_x - 1)
                grad += wnorm[m_idx] * gradient_at_lag(x, lag, dx)
            se = np.sum(x) - target
            value += 1e5 * se * se
            grad = grad + 2e5 * se * np.ones(n_x)
            below = np.maximum(0.0, -x)
            above = np.maximum(0.0, x - 1.0)
            value += 1e5 * (np.sum(below * below) + np.sum(above * above))
            grad = grad - 2e5 * below + 2e5 * above
            return value, grad

        try:
            mi = max(30, min(max_iter, n * 2 + 20))
            res = minimize(obj_grad, best_h, method='SLSQP',
                           bounds=bounds, constraints=constraint, jac=True,
                           options={'maxiter': mi, 'ftol': 1e-15})
            if res.x is not None and np.all(np.isfinite(res.x)):
                cand = project_box_sum(res.x, target)
                c, _ = exact_objective(cand)
                if c < best_c - 1e-13:
                    best_c = c
                    best_h = cand
        except Exception:
            pass
    return best_h, best_c

def slsqp_direct(h_start, n, deadline, max_iter=50, margin=0.005):
    best_h = project_box_sum(h_start, 0.5 * n)
    best_c, profile = exact_objective(best_h)
    target = 0.5 * n
    n_profile = 2 * n - 1
    # Select near-active constraints
    active_thresh = best_c - margin
    active_mask = profile > active_thresh
    active_indices = np.where(active_mask)[0]
    if len(active_indices) == 0:
        active_indices = np.array([int(np.argmax(profile))])

    n_total = n + 1

    def obj(z):
        return z[-1]

    def obj_grad(z):
        g = np.zeros(n_total)
        g[-1] = 1.0
        return g

    def eq_fun(z):
        return np.sum(z[:n]) - target

    def eq_jac(z):
        j = np.zeros(n_total)
        j[:n] = 1.0
        return j

    eq_constraint = {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}

    def ineq_fun(z):
        h = z[:n]
        c = z[-1]
        dx = 2.0 / n
        prof = np.correlate(h, 1.0 - h, mode='full') * dx
        return c - prof[active_indices]

    def ineq_jac(z):
        h = z[:n]
        dx = 2.0 / n
        m = len(active_indices)
        J = np.zeros((m, n_total))
        for i, r in enumerate(active_indices):
            lag = int(r) - (n - 1)
            J[i, :n] = -gradient_at_lag(h, lag, dx)
            J[i, -1] = 1.0
        return J

    ineq_constraint = {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}

    c_lo = max(0.0, best_c - 0.02)
    c_hi = best_c + 0.01
    bounds = [(0.0, 1.0)] * n + [(c_lo, c_hi)]
    z0 = np.concatenate([best_h, [best_c + 1e-5]])

    try:
        res = minimize(obj, z0, method='SLSQP',
                       bounds=bounds,
                       constraints=[eq_constraint, ineq_constraint],
                       jac=obj_grad,
                       options={'maxiter': max_iter, 'ftol': 1e-15})
        if res.x is not None and np.all(np.isfinite(res.x)):
            cand = project_box_sum(res.x[:n], target)
            c, _ = exact_objective(cand)
            if c < best_c - 1e-13:
                best_c = c
                best_h = cand
    except Exception:
        pass
    return best_h, best_c

def gradient_descent_active(h, n, steps=100, deadline=None):
    target = 0.5 * n
    dx = 2.0 / n
    best_h = project_box_sum(h, target)
    best_c, profile = exact_objective(best_h)

    for _ in range(steps):
        if deadline is not None and time.time() > deadline:
            break
        top_k = min(7, 2 * n - 1)
        top_idx = np.argsort(-profile)[:top_k]
        temps = [0.005, 0.01, 0.02]
        found = False
        for temp in temps:
            weights = np.exp((profile[top_idx] - profile[top_idx].max()) / temp)
            weights /= np.sum(weights)
            grad = np.zeros(n)
            for w, r in zip(weights, top_idx):
                lag = int(r) - (n - 1)
                grad += w * gradient_at_lag(best_h, lag, dx)
            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-12:
                break
            grad_unit = grad / grad_norm
            for alpha in [0.002, 0.005, 0.01, 0.02, 0.04, 0.06]:
                cand = best_h - alpha * grad_unit
                cand = project_box_sum(cand, target)
                c, prof_new = exact_objective(cand)
                if c < best_c - 1e-13:
                    best_c = c
                    best_h = cand
                    profile = prof_new
                    found = True
                    break
            if found:
                break
        if not found:
            break
    return best_h, best_c

def local_perturbation(h, n, n_trials=200, rng=None, deadline=None):
    target = 0.5 * n
    best_h = project_box_sum(h, target)
    best_c, _ = exact_objective(best_h)
    if rng is None:
        rng = np.random.RandomState(0)

    for _ in range(n_trials):
        if deadline is not None and time.time() > deadline:
            break
        eps = rng.uniform(0.01, 0.06)
        direction = rng.randn(n)
        direction -= np.mean(direction)  # zero mean to preserve sum
        cand = best_h + eps * direction
        cand = project_box_sum(cand, target)
        c, _ = exact_objective(cand)
        if c < best_c - 1e-13:
            best_c = c
            best_h = cand
    return best_h, best_c

def binary_threshold(h):
    n = len(h)
    if n % 2 != 0:
        return None
    target = n // 2
    idx = np.argsort(-h)
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b

def binary_swap_improve(b, max_rounds=50, rng=None, deadline=None):
    n = len(b)
    if n % 2 != 0:
        return b, exact_objective(b)[0]
    if rng is None:
        rng = np.random.RandomState(0)
    c5, _ = exact_objective(b)

    for rnd in range(max_rounds):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        ones = np.where(b > 0.5)[0]
        zeros = np.where(b < 0.5)[0]
        n_pairs = len(ones) * len(zeros)

        if n_pairs <= 5000:
            for i in ones:
                found = False
                for j in zeros:
                    b[i] = 0.0
                    b[j] = 1.0
                    nc, _ = exact_objective(b)
                    if nc < c5 - 1e-13:
                        c5 = nc
                        improved = True
                        found = True
                        ones = np.where(b > 0.5)[0]
                        zeros = np.where(b < 0.5)[0]
                        break
                    b[i] = 1.0
                    b[j] = 0.0
                if found:
                    break
        else:
            n_tries = min(800, n_pairs)
            for _ in range(n_tries):
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                b[i], b[j] = 0.0, 1.0
                nc, _ = exact_objective(b)
                if nc < c5 - 1e-13:
                    c5 = nc
                    improved = True
                    ones = np.where(b > 0.5)[0]
                    zeros = np.where(b < 0.5)[0]
                else:
                    b[i], b[j] = 1.0, 0.0
        if not improved:
            break
    return b, c5

def make_start(n, rng, mode=0):
    target = 0.5 * n
    if mode == 0:
        h = np.zeros(n)
        h[::2] = 1.0
    elif mode == 1:
        h = np.zeros(n)
        h[:int(target)] = 1.0
    elif mode == 2:
        h = np.zeros(n)
        idx = rng.choice(n, size=min(int(target), n), replace=False)
        h[idx] = 1.0
    elif mode == 3:
        x = np.arange(n, dtype=np.float64)
        freq = 1.0 + rng.randint(1, 5)
        phase = rng.uniform(0, 2 * np.pi)
        h = 0.5 + 0.35 * np.cos(2 * np.pi * freq * x / n + phase)
    elif mode == 4:
        x = np.arange(n, dtype=np.float64)
        h = 0.5 + 0.3 * np.sin(2 * np.pi * 2 * x / n) + 0.15 * np.cos(2 * np.pi * 3 * x / n + 0.5)
    else:
        h = np.full(n, 0.5) + rng.randn(n) * 0.1
    return project_box_sum(h, target)

def downsample(h, new_n):
    n = len(h)
    if new_n == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    result = np.zeros(new_n)
    for k in range(new_n):
        lo = k / new_n
        hi = (k + 1) / new_n
        mask = (x_old >= lo) & (x_old < hi)
        if np.any(mask):
            result[k] = np.mean(h[mask])
        else:
            result[k] = np.interp((lo + hi) / 2, x_old, h)
    return project_box_sum(result, 0.5 * new_n)
