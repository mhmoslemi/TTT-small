def evaluate_exact(h):
    n = len(h)
    dx = 2.0 / n
    return max(np.correlate(h, 1.0 - h, mode='full')) * dx


def project_to_target(x, target_sum):
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.clip(x + mid, 0.0, 1.0).sum()
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    h = np.clip(x + (lo + hi) / 2, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)


def build_sym_h(q, n=None):
    m = len(q)
    if n is None:
        n = 2 * m
    h = np.empty(n)
    if n == 2 * m:
        h[:m] = q
        h[m:] = 1.0 - q[::-1]
    else:
        h[:m] = q
        h[m] = 0.5
        h[m+1:] = 1.0 - q[::-1]
    return h


def q_from_h(h):
    n = len(h)
    m = n // 2
    if n % 2 == 0:
        q = 0.5 * (h[:m] + (1.0 - h[m:][::-1]))
    else:
        q = 0.5 * (h[:m] + (1.0 - h[m+1:][::-1]))
    return np.clip(q, 0.0, 1.0)


def corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')


def grad_corr_k_h(h, k, n):
    d = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -d)
    i_hi = min(n, n - d)
    if i_lo >= i_hi:
        return grad
    idx = np.arange(i_lo, i_hi)
    grad[idx] += (1.0 - h[idx + d])
    grad[idx + d] += (-h[idx])
    return grad


def grad_corr_k_q(q, k, n):
    m = len(q)
    h = build_sym_h(q, n)
    dh = grad_corr_k_h(h, k, n)
    return dh[:m] - dh[n-1::-1][:m]

        corr = corr_full(h)
        max_c = corr.max()
        shifted = corr - max_c
        w = np.exp(self.beta * shifted)
        ws = w.sum()
        if ws < 1e-300:
            return max_c * dx
        return (max_c + np.log(ws) / self.beta) * dx
