import numpy as np
from scipy.optimize import minimize, linprog
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    profile = np.correlate(h, 1 - h, mode="full") * dx
    return float(np.max(profile))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64).ravel(), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx] + 1e-15)
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx] + 1e-15)
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def build_gradient_matrix(h, n, dx):
    M = 2 * n - 1
    G = np.zeros((M, n))
    for m in range(M):
        d = m - (n - 1)
        if d == 0:
            G[m, :] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            G[m, 0:n-d] += dx * (1.0 - h[d:n])
            G[m, d:n] += -dx * h[0:n-d]
        else:
            dp = -d
            G[m, dp:n] += dx * (1.0 - h[0:n-dp])
            G[m, 0:n-dp] += -dx * h[dp:n]
    return G

def interp_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_x = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_x, full_h)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

def apply_rev_symmetry(h):
    n = len(h)
    h_sym = (h + (1.0 - h[::-1])) / 2.0
    return repair_feasible(h_sym)

def sca_step(h_cur, radius):
    n = len(h_cur)
    dx = 2.0 / n
    M = 2 * n - 1
    profile = compute_profile(h_cur)
    G = build_gradient_matrix(h_cur, n, dx)
    
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    A_ub = np.zeros((M, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = G @ h_cur - profile
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    lo = np.maximum(0.0, h_cur - radius)
    hi = np.minimum(1.0, h_cur + radius)
    bounds = list(zip(lo, hi))
    bounds.append((0.0, None))
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x is not None and np.all(np.isfinite(res.x)):
            return res.x[:n], float(res.x[n])
    except Exception:
        pass
    return None, None

def do_line_search(h_cur, h_prop):
    best_candidate = h_cur.copy()
    best_score = exact_score(h_cur)
    alphas = [1.0, 0.95, 0.9, 0.8, 0.7, 0.6, 0.5, 0.4, 0.3, 0.2, 0.15, 0.1, 0.06, 0.03, 0.01]
    for alpha in alphas:
        cand = h_cur + alpha * (h_prop - h_cur)
        cand = repair_feasible(cand)
        score = exact_score(cand)
        if score < best_score - 1e-13:
            best_score = score
            best_candidate = cand.copy()
    return best_candidate, best_score

def sca_refine(h_init, max_iters=40, initial_radius=0.12, min_radius=0.0008, time_fn=None):

    c = exact_score(h)
    radius = initial_radius
    stagnant = 0
    best_h = h.copy()
    best_c = c
    
    for it in range(max_iters):
        if time_fn is not None and time_fn() < 12:
            break
        z_prop, t_prop = sca_step(h, radius)
        if z_prop is None:
            radius *= 0.5
            stagnant += 1
            if stagnant > 6 or radius < min_radius:
                break
            continue
        
        h_new, c_new = do_line_search(h, z_prop)
        
        if c_new < c - 1e-13:
            h = h_new
            c = c_new
            stagnant = 0
            if c < best_c:
                best_c = c
                best_h = h.copy()
            radius = min(radius * 1.25, 0.4)
        else:
            stagnant += 1
            radius *= 0.55
            if radius < min_radius:
                break
            if stagnant > 8:
                break
    
    return best_h, best_c

def weighted_sca_refine(h_init, max_iters=30, initial_radius=0.08, min_radius=0.001, time_fn=None, weight_power=1.0):
    """SCA with weighted objective: minimize t + eps * sum(w_m * (t - tangent_m)) to flatten profile."""
    h = h_init.copy()
    c = exact_score(h)
    radius = initial_radius
    stagnant = 0
    best_h = h.copy()
    best_c = c
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    
    for it in range(max_iters):
        if time_fn is not None and time_fn() < 12:
            break
        profile = compute_profile(h)
        G = build_gradient_matrix(h, n, dx)
        
        # Compute weights: give more weight to near-active lags
        c_max = np.max(profile)
        gap = c_max - profile
        gap = np.maximum(gap, 1e-10)
        w = np.power(gap / c_max + 1e-6, -weight_power)
        w = w / w.sum()
        
        # LP: minimize t + eps * sum_m w_m * (t - (G[m]@z + profile[m] - G[m]@h))
        # = minimize t*(1 + eps*sum(w)) - eps * sum_m w_m*(G[m]@z + profile[m] - G[m]@h)
        # = minimize t*(1+eps) - eps*sum_m w_m*G[m]@z - eps*sum_m w_m*(profile[m]-G[m]@h)
        # The last term is constant, so minimize: t*(1+eps) - eps*sum_m w_m*G[m]@z
        # Constraints: G[m]@z - t <= G[m]@h - profile[m]
        
        eps = 0.1
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0 + eps
        c_obj[:n] = -eps * (w @ G)
        
        A_ub = np.zeros((M, n + 1))
        A_ub[:, :n] = G
        A_ub[:, n] = -1.0
        b_ub = G @ h - profile
        
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([n / 2.0])
        
        lo = np.maximum(0.0, h - radius)
        hi = np.minimum(1.0, h + radius)
        bounds = list(zip(lo, hi))
        bounds.append((0.0, None))
        
        z_prop = None
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if res.success and res.x is not None and np.all(np.isfinite(res.x)):
                z_prop = res.x[:n]
        except Exception:
            pass
        
        if z_prop is None:
            radius *= 0.5
            stagnant += 1
            if stagnant > 6 or radius < min_radius:
                break
            continue
        
        h_new, c_new = do_line_search(h, z_prop)
        
        if c_new < c - 1e-13:
            h = h_new
            c = c_new
            stagnant = 0
            if c < best_c:
                best_c = c
                best_h = h.copy()
            radius = min(radius * 1.2, 0.35)
        else:
            stagnant += 1
            radius *= 0.6
            if radius < min_radius:
                break
            if stagnant > 7:
                break
    
    return best_h, best_c

def slsqp_polish(h_init, maxiter=300):
    n = len(h_init)
    dx = 2.0 / n
    M = 2 * n - 1
    
    def obj(x):
        return x[-1]
    
    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    
    def eq_fun(x):
        return x[:-1].sum() - n / 2.0
    
    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g
    
    def ineq_fun(x):
        h = x[:-1]
        t = x[-1]
        prof = np.correlate(h, 1 - h, mode="full") * dx
        return t - prof
    
    def ineq_jac(x):
        h = x[:-1]
        G = build_gradient_matrix(h, n, dx)
        J = np.zeros((M, n + 1))
        J[:, :-1] = -G
        J[:, -1] = 1.0
        return J
    
    c0 = exact_score(h_init)
    x0 = np.concatenate([h_init, [c0 + 1e-5]])
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 1.0)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_sol = repair_feasible(res.x[:-1])
            c_sol = exact_score(h_sol)
            if np.isfinite(c_sol):
                return h_sol, c_sol
    except Exception:
        pass
    return None, None

def gen_diverse_starts(h_base, n, rng, num_starts=8):

    # Symmetric
    starts.append(apply_rev_symmetry(h_base))
    # Random perturbations
    for i in range(num_starts - 2):
        eps = 0.03 + 0.04 * i / max(1, num_starts - 2)

            vals = level + 0.3 * np.cos(2 * np.pi * x / period)
            starts.append(repair_feasible(np.clip(vals, 0, 1)))
    return starts

        h7, c7 = weighted_sca_refine(best_h, max_iters=25, initial_radius=0.03, min_radius=0.0005, time_fn=time_left, weight_power=0.5)
        if c7 < best_c - 1e-14:
            best_h, best_c = h7, c7
    
    # Stage 8: SLSQP polish
    if time_left() > 60:
        h_sol, c_sol = slsqp_polish(best_h, maxiter=400)
        if h_sol is not None and c_sol < best_c - 1e-14:
            best_h, best_c = h_sol, c_sol
    
    # Stage 9: More SCA after SLSQP (different basin)
    if time_left() > 40:
        h9, c9 = sca_refine(best_h, max_iters=30, initial_radius=0.05, time_fn=time_left)
        if c9 < best_c - 1e-14:
            best_h, best_c = h9, c9
    
    # Stage 10: Final SLSQP
    if time_left() > 30:
        h_sol, c_sol = slsqp_polish(best_h, maxiter=200)
        if h_sol is not None and c_sol < best_c - 1e-14:
            best_h, best_c = h_sol, c_sol
    
    # Final validation
    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)
    
    # Try pre-imported evaluator
    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            final_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 1:
            final_c = float(eval_result[0])
    except Exception:
        pass
    
    return best_h, final_c, len(best_h)
