    b = b.copy()
    curr_c = exact_objective(b)
    ones = np.where(b == 1.0)[0].tolist()

            break
        best_improve = 0
        best_move = None
        for i in ones:
            for j in zeros:
                b[i] = 0.0
                b[j] = 1.0
                c = exact_objective(b)

            curr_c = exact_objective(b)
            ones.remove(i)

            zeros.append(i)

        else:

def exact_objective(h):
    return float(np.max(exact_profile(h)))


def project_box_sum(h):
    h = np.array(h, dtype=np.float64).copy()
    n = len(h)
    target = n / 2.0
    h[~np.isfinite(h)] = 0.0
    lo = -np.max(h) - 2.0
    hi = 1.0 - np.min(h) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    res = np.clip(h + lam, 0.0, 1.0)
    diff = target - np.sum(res)
    if abs(diff) > 1e-13:
        if diff > 0:
            mask = res < 1.0 - 1e-14
            cnt = int(np.sum(mask))
            if cnt > 0:
                res[mask] = np.minimum(1.0, res[mask] + diff / cnt)
        else:
            mask = res > 1e-14
            cnt = int(np.sum(mask))
            if cnt > 0:
                res[mask] = np.maximum(0.0, res[mask] + diff / cnt)
    return np.clip(res, 0.0, 1.0)


def transfer_to_size(h, n_new):
    n = len(h)
    if n == n_new:
        return project_box_sum(h.copy())
    x_src = (np.arange(n) + 0.5) * (2.0 / n)
    x_tgt = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_new = np.interp(x_tgt, x_src, h)
    return project_box_sum(h_new)


def binary_threshold(h):
    n = len(h)
    m = n // 2
    idx = np.argsort(h)[::-1][:m]
    b = np.zeros(n, dtype=np.float64)
    b[idx] = 1.0
    return b


def profile_gradients(h):
    n = len(h)
    dx = 2.0 / n
    p = 2 * n - 1
    G = np.zeros((p, n), dtype=np.float64)
    for ell in range(-(n - 1), n):
        j = ell + n - 1
        lo_i = max(0, -ell)
        hi_i = min(n - 1, n - 1 - ell)
        for i in range(lo_i, hi_i + 1):
            q = i + ell
            if i == q:
                G[j, i] += dx * (1.0 - 2.0 * h[i])
            else:
                G[j, i] += dx * (1.0 - h[q])
                G[j, q] -= dx * h[i]
    return G


def binary_hill_climb(b, deadline, max_sweeps=2000):
    n = len(b)
    b = b.copy()
    curr_c = exact_objective(b)
    ones = np.where(b == 1.0)[0].tolist()
    zeros = np.where(b == 0.0)[0].tolist()
    for _ in range(max_sweeps):
        if time.time() > deadline - 3:
            break
        best_improve = 1e-15
        best_move = None
        for i in ones:
            for j in zeros:
                b[i] = 0.0
                b[j] = 1.0
                c = exact_objective(b)
                imp = curr_c - c
                if imp > best_improve:
                    best_improve = imp
                    best_move = (i, j)
                b[i] = 1.0
                b[j] = 0.0
        if best_move is not None:
            i, j = best_move
            b[i] = 0.0
            b[j] = 1.0
            curr_c = exact_objective(b)
            ones.remove(i)
            zeros.remove(j)
            zeros.append(i)
            ones.append(j)
        else:
            break
    return b, curr_c


def binary_sa(b, deadline, t0=0.12, t1=0.001, n_iter=50000):
    n = len(b)
    b = b.copy()
    curr_c = exact_objective(b)
    best_c = curr_c
    best_b = b.copy()
    ones_set = set(np.where(b == 1.0)[0].tolist())
    zeros_set = set(np.where(b == 0.0)[0].tolist())
    ones_list = list(ones_set)
    zeros_list = list(zeros_set)
    rng = np.random.RandomState(12345)
    for it in range(n_iter):
        if time.time() > deadline - 3:
            break
        T = t0 * (t1 / t0) ** (it / float(n_iter))
        i = ones_list[rng.randint(len(ones_list))]
        j = zeros_list[rng.randint(len(zeros_list))]
        b[i] = 0.0
        b[j] = 1.0
        c = exact_objective(b)
        delta = c - curr_c
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            curr_c = c
            ones_set.discard(i)
            ones_set.add(j)
            zeros_set.discard(j)
            zeros_set.add(i)
            ones_list = list(ones_set)
            zeros_list = list(zeros_set)
            if c < best_c:
                best_c = c
                best_b = b.copy()
        else:
            b[i] = 1.0
            b[j] = 0.0
    return best_b, best_c


def cvxpy_active_step(h, R, K, deadline):
    if not HAS_CVXPY:
        return None
    n = len(h)
    dx = 2.0 / n
    profile = exact_profile(h)
    C_curr = float(np.max(profile))
    G = profile_gradients(h)
    active_idx = np.argsort(profile)[::-1][:K]
    active_idx = np.sort(active_idx)
    A_mat = G[active_idx, :].copy()
    b_vec = profile[active_idx].copy() - A_mat @ h
    z = cp.Variable(n)
    t = cp.Variable()
    constraints = [
        cp.sum(z) == n / 2.0,
        z >= 0,
        z <= 1,
        cp.norm(z - h, 2) <= R,
        A_mat @ z <= t - b_vec,
    ]
    try:
        prob = cp.Problem(cp.Minimize(t), constraints)
        prob.solve(solver=cp.CLARABEL)
        if z.value is None:
            return None
        z_val = np.array(z.value, dtype=np.float64)
        return project_box_sum(z_val)
    except Exception:
        return None


def active_set_refine(h, deadline, n_steps=50):
    n = len(h)
    h = project_box_sum(h.copy())
    curr_c = exact_objective(h)
    R_sizes = [0.5, 0.3, 0.15, 0.08, 0.04, 0.02]
    rw = 0
    for step in range(n_steps):
        if time.time() > deadline - 5:
            break
        R = R_sizes[min(rw, len(R_sizes) - 1)]
        K = min(2 * n - 1, max(5, n // 3))
        z_new = cvxpy_active_step(h, R, K, deadline)
        if z_new is None:
            rw = min(rw + 1, len(R_sizes) - 1)
            continue
        accepted = False
        for alpha in [1.0, 0.7, 0.5, 0.35, 0.2, 0.1, 0.05]:
            y = h + alpha * (z_new - h)
            y = project_box_sum(y)
            c_y = exact_objective(y)
            if c_y < curr_c - 1e-12:
                h = y
                curr_c = c_y
                rw = 0
                accepted = True
                break
        if not accepted:
            rw = min(rw + 1, len(R_sizes) - 1)
            if rw >= len(R_sizes) - 1:
                break
    return h, curr_c


def subgradient_refine(h, deadline, max_iter=200):
    n = len(h)
    dx = 2.0 / n
    h = project_box_sum(h.copy())
    curr_c = exact_objective(h)
    tau = 0.05
    alpha = 0.1
    for it in range(max_iter):
        if time.time() > deadline - 5:
            break
        profile = exact_profile(h)
        C_curr = float(np.max(profile))
        G = profile_gradients(h)
        shifted = np.minimum((profile - C_curr) / tau, 0.0)
        weights = np.exp(shifted)
        weights /= weights.sum()
        g = weights @ G
        g -= np.mean(g)
        z = h - alpha * g
        h_new = project_box_sum(z)
        c_new = exact_objective(h_new)
        if c_new < curr_c - 1e-14:
            h = h_new
            curr_c = c_new
            alpha = min(alpha * 1.2, 0.5)
        else:
            alpha *= 0.7
        tau *= 0.95
    return h, curr_c


def make_start(n, rng, mode):
    if mode == 0:
        h = np.zeros(n)
        h[:n // 2] = 1.0
        h = h[rng.permutation(n)]
    elif mode == 1:
        x = np.linspace(0, 1, n)
        freq = rng.randint(2, 7)
        phase = rng.uniform(0, 2 * np.pi)
        h = 0.5 + 0.4 * np.sin(2 * np.pi * freq * x + phase)
        h = np.clip(h, 0, 1)
    elif mode == 2:
        h = np.zeros(n)
        h[::2] = 1.0
        if n % 2 == 1:
            h[-1] = 0.0
    elif mode == 3:
        h = rng.uniform(0, 1, n)
    elif mode == 4:
        h = np.zeros(n)
        block = max(2, n // 8)
        for i in range(0, n, 2 * block):
            h[i:i + block] = 1.0
    elif mode == 5:
        h = np.zeros(n)
        h[:n // 4] = 1.0
        h[3 * n // 4:] = 1.0
    elif mode == 6:
        h = np.zeros(n)
        idx = np.sort(rng.choice(n, n // 2, replace=False))
        h[idx] = 1.0
    else:
        h = np.full(n, 0.5)
    return project_box_sum(h)


def random_perturb(h, scale, rng):
    h_new = h + rng.normal(0, scale, len(h))
    return project_box_sum(h_new)


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    deadline = t_start + budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c = float('inf')
    best_n = 0

    parent_h = None
    try:
        parent_h = np.array(globals()['initial_h_values'], dtype=np.float64).copy()
        if len(parent_h) > 0:
            parent_h = project_box_sum(parent_h)
            pc = exact_objective(parent_h)
            best_c = pc
            best_h = parent_h.copy()
            best_n = len(parent_h)
    except Exception:
        pass

    if best_h is None:
        best_h = make_start(48, rng, 6)
        best_c = exact_objective(best_h)
        best_n = 48

    parent_n = len(parent_h) if parent_h is not None else 48

    sizes = []
    for s in [parent_n, 44, 46, 48, 50, 52, 54, 56, 58, 60, 64, 72, 80]:
        if s not in sizes and 16 <= s <= 100:
            sizes.append(s)

    n_tried = {best_n}

    for n in sizes:
        if time.time() > deadline - 30:
            break
        if n in n_tried:
            continue
        n_tried.add(n)

        cands = []
        if parent_h is not None:
            hp = transfer_to_size(parent_h, n)
            cands.append(hp)
            cands.append(random_perturb(hp, 0.02, rng))
            cands.append(random_perturb(hp, 0.06, rng))
        for mode in range(7):
            cands.append(make_start(n, rng, mode))

        scored = []
        for c in cands:
            scored.append((exact_objective(c), c))
        scored.sort(key=lambda x: x[0])

        local_best_h = scored[0][1]
        local_best_c = scored[0][0]

        n_top = min(3, len(scored))
        for idx in range(n_top):
            if time.time() > deadline - 25:
                break
            h0 = scored[idx][1]

            h_ref, c_ref = subgradient_refine(h0, deadline, max_iter=80)
            if c_ref < local_best_c:
                local_best_h, local_best_c = h_ref, c_ref

            if HAS_CVXPY and time.time() < deadline - 15:
                h_ref2, c_ref2 = active_set_refine(local_best_h, deadline, n_steps=40)
                if c_ref2 < local_best_c:
                    local_best_h, local_best_c = h_ref2, c_ref2

            if c_ref < best_c - 1e-12:
                best_c = c_ref
                best_h = h_ref.copy()
                best_n = n

        if n % 2 == 0 and time.time() < deadline - 20:
            b = binary_threshold(local_best_h)
            c_b = exact_objective(b)
            if c_b < local_best_c:
                local_best_h, local_best_c = b.copy(), c_b
            b, c_b = binary_hill_climb(b, deadline, max_sweeps=800)
            if c_b < local_best_c:
                local_best_h, local_best_c = b, c_b
            if time.time() < deadline - 12:
                b_sa, c_sa = binary_sa(b, deadline, t0=0.1, t1=0.0005, n_iter=30000)
                if c_sa < local_best_c:
                    local_best_h, local_best_c = b_sa, c_sa
            if c_b < best_c - 1e-12:
                best_c = c_b
                best_h = b.copy()
                best_n = n

    # Refine best with both methods
    if best_h is not None and time.time() < deadline - 15:
        nb = len(best_h)
        if np.all((best_h == 0.0) | (best_h == 1.0)):
            b, c_b = binary_hill_climb(best_h.copy(), deadline, max_sweeps=1000)
            if c_b < best_c - 1e-12:
                best_c = c_b
                best_h = b.copy()
            if time.time() < deadline - 10:
                b_sa, c_sa = binary_sa(b, deadline, t0=0.08, t1=0.0003, n_iter=20000)
                if c_sa < best_c - 1e-12:
                    best_c = c_sa
                    best_h = b_sa.copy()
        else:
            if HAS_CVXPY:
                h_ref, c_ref = active_set_refine(best_h.copy(), deadline, n_steps=60)
                if c_ref < best_c - 1e-12:
                    best_c = c_ref
                    best_h = h_ref.copy()
            h_ref2, c_ref2 = subgradient_refine(best_h.copy(), deadline, max_iter=100)
            if c_ref2 < best_c - 1e-12:
                best_c = c_ref2
                best_h = h_ref2.copy()
            if nb % 2 == 0 and time.time() < deadline - 8:
                b = binary_threshold(best_h)
                b, c_b = binary_hill_climb(b, deadline, max_sweeps=500)
                if c_b < best_c - 1e-12:
                    best_c = c_b
                    best_h = b.copy()

    # Try transferring best to nearby sizes
    if best_h is not None and time.time() < deadline - 20:
        nb = len(best_h)
        for ns in [nb - 2, nb + 2, nb - 4, nb + 4]:
            if ns < 16 or time.time() > deadline - 12:
                continue
            h_t = transfer_to_size(best_h, ns)
            c_t = exact_objective(h_t)
            if c_t < best_c - 1e-12:
                best_c = c_t
                best_h = h_t.copy()
                best_n = ns
            sub_res, c_sub = subgradient_refine(h_t, deadline, max_iter=50)
            if c_sub < best_c - 1e-12:
                best_c = c_sub
                best_h = sub_res.copy()
                best_n = ns
            if ns % 2 == 0 and time.time() < deadline - 8:
                b = binary_threshold(sub_res)
                b, c_b = binary_hill_climb(b, deadline, max_sweeps=400)
                if c_b < best_c - 1e-12:
                    best_c = c_b
                    best_h = b.copy()
                    best_n = ns

    # Final validation
    best_h = project_box_sum(best_h)
    best_c = exact_objective(best_h)
    best_n = len(best_h)

    if abs(np.sum(best_h) - best_n / 2.0) > 1e-8:
        best_h = project_box_sum(best_h)
        best_c = exact_objective(best_h)

    return best_h, best_c, best_n
