    n = len(h)

        delta = k - (n - 1)

def exact_score(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))

def get_profile(h):
    dx = 2.0 / len(h)
    return np.correlate(h, 1.0 - h, mode="full") * dx

def get_grad_profile(h):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    G = np.zeros((M, n))
    for k in range(M):
        delta = k - (n - 1)
        if delta >= 0:
            if delta < n:
                G[k, delta:n] += dx * (1.0 - h[0:n - delta])
                G[k, 0:n - delta] -= dx * h[delta:n]
        else:
            d = -delta
            if d < n:
                G[k, 0:n - d] += dx * (1.0 - h[d:n])
                G[k, d:n] -= dx * h[0:n - d]
    return G

def compute_quadratic_coeffs(h, d):

        elif delta > 0:

            b[k] = dx * np.sum(d[delta:n] * (1.0 - h[0:n - delta]) - d[0:n - delta] * h[delta:n])
        else:

def line_search_quad(h, d, c_profile, a_vec, b_vec, n_grid=60):

def repair_feasible(h, target=None):
    h = np.array(h, dtype=np.float64).copy()

def project_zero_sum(d):
    return d - d.mean()

def interpolate_to_dim(vec, new_len):
    n = len(vec)
    x_src = (np.arange(n) + 0.5) * 2.0 / n
    x_tgt = (np.arange(new_len) + 0.5) * 2.0 / new_len

    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

def solve_slsqp(h_start, t_start, maxiter=60, n_constraints=0):
    n = len(h_start)
    M = 2 * n - 1
    nc = n_constraints if n_constraints > 0 else M
    prof = get_profile(h_start)
    if nc < M:
        active_lags = np.argsort(prof)[::-1][:nc]
    else:
        active_lags = np.arange(M)

    def obj(x):
        return x[n]

    def cons_fun(x):
        h = x[:n]
        t = x[n]
        eq = h.sum() - n / 2.0
        cprof = np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)
        ineq = t - cprof[active_lags]
        return np.concatenate([[eq], ineq])

    def cons_jac(x):
        h = x[:n]
        G = get_grad_profile(h)
        Ja = G[active_lags, :]
        J = np.zeros((1 + len(active_lags), n + 1))
        J[0, :n] = 1.0
        J[1:, :n] = -Ja
        J[1:, n] = 1.0
        return J

    bounds = [(0.0, 1.0)] * n + [(0.0, t_start + 1.0)]
    x0 = np.concatenate([h_start, [t_start]])
    try:
        result = minimize(obj, x0, method='SLSQP', bounds=bounds,
                          constraints=[{'type': 'ineq', 'fun': cons_fun, 'jac': cons_jac}],
                          options={'maxiter': maxiter, 'ftol': 1e-12})
        if result.x is not None and np.all(np.isfinite(result.x)):
            return repair_feasible(result.x[:n])
    except Exception:
        pass
    return None
