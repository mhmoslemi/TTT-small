def repair_feasible(h, target=None):
    h = np.array(h, dtype=np.float64).copy()
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)

        target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:

    if diff > 0:
        room = 1.0 - h
        order = np.argsort(-room)
        remaining = diff
        for idx in order:
            if remaining <= 1e-15:
                break
            add = min(remaining, room[idx])
            h[idx] += add
            remaining -= add
    else:

            if remaining <= 1e-15:

Better approach: **importance sampling based on the current profile**.

For the current profile P, the "most constraining" lags are the ones at the maximum. For each such lag k with delta:
- The pairs that can decrease P_k are those where we move mass in the direction of negative gradient
- Specifically, for lag k with delta > 0: P_k = dx * Σ h[u+delta](1-h[u])
  - To decrease: move mass from (u+delta) to u for pairs where h[u+delta] is high and h[u] is low
  - Or: move mass from u to u+delta for pairs where h[u] is high and 1-h[u+delta] is high (i.e., h[u+delta] is low)

The "gain" from moving mass from position a to position b for lag k is:
ΔP_k = -t * G[k,a] + t * G[k,b] = t * (G[k,b] - G[k,a])
We want this negative: G[k,b] < G[k,a].

So the best pair for lag k is: a = argmax G[k,:], b = argmin G[k,:].

For multiple active lags, the combined direction is the sum of individual gradients, and the best pair is determined by the combined gradient.

1. Compute G
2. For top K active lags, compute combined gradient D = Σ G[k,:]
3. Generate candidate pairs: (argmax D, argmin D), plus random perturbation
4. Also generate pairs from individual active lags
5. Try all candidates, accept the best

One more optimization: instead of computing the full gradient matrix G (which is O(n²) per call), I can compute individual gradient rows on-demand. For a single lag k, computing G[k,:] is O(n). So for K active lags, it's O(K*n) which is much less than O(n²).

def get_lag_gradient(h, k):
    """Compute dP_k/dh for a specific lag k."""
    n = len(h)
    dx = 2.0 / n
    delta = k - (n - 1)
    g = np.zeros(n)
    if delta == 0:
        g[:] = dx * (1.0 - 2.0 * h)
    elif delta > 0:
        g[delta:] += dx * (1.0 - h[:n-delta])
        g[:n-delta] -= dx * h[delta:]
    else:
        d = -delta
        g[:n-d] += dx * (1.0 - h[d:])
        g[d:] -= dx * h[:n-d]
    return g

import numpy as np
import time

def exact_score(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))

def get_profile(h):
    dx = 2.0 / len(h)
    return np.correlate(h, 1.0 - h, mode="full") * dx

def repair_feasible(h, target=None):
    h = np.array(h, dtype=np.float64).copy()
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    if target is None:
        target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        order = np.argsort(-room)
        remaining = diff
        for idx in order:
            if remaining <= 1e-15:
                break
            add = min(remaining, room[idx])
            h[idx] += add
            remaining -= add
    else:
        order = np.argsort(-h)
        remaining = -diff
        for idx in order:
            if remaining <= 1e-15:
                break
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
    return h

def get_lag_gradient(h, k):
    n = len(h)
    dx = 2.0 / n
    delta = k - (n - 1)
    g = np.zeros(n)
    if delta == 0:
        g[:] = dx * (1.0 - 2.0 * h)
    elif delta > 0:
        g[delta:] += dx * (1.0 - h[:n - delta])
        g[:n - delta] -= dx * h[delta:]
    else:
        d = -delta
        g[:n - d] += dx * (1.0 - h[d:])
        g[d:] -= dx * h[:n - d]
    return g

def pair_line_search(h, i, j):
    n = len(h)
    dx = 2.0 / n
    t_lo = max(h[i] - 1.0, -h[j])
    t_hi = min(h[i], 1.0 - h[j])
    span = t_hi - t_lo
    if span < 1e-11:
        return None
    t_0 = 0.0
    t_1 = t_lo + 0.35 * span
    t_2 = t_lo + 0.75 * span
    pts = np.array([t_0, t_1, t_2])
    if np.any(np.abs(np.diff(np.sort(pts))) < 1e-12):
        pts = np.array([t_lo + 0.1 * span, t_lo + 0.5 * span, t_lo + 0.9 * span])
    profiles = np.zeros((3, 2 * n - 1))
    hi_save = h[i]
    hj_save = h[j]
    for k_idx, t in enumerate(pts):
        h[i] = hi_save - t
        h[j] = hj_save + t
        profiles[k_idx] = np.correlate(h, 1.0 - h, mode="full") * dx
    h[i] = hi_save
    h[j] = hj_save
    T = np.column_stack([pts**2, pts, np.ones(3)])
    try:
        T_inv = np.linalg.inv(T)
    except np.linalg.LinAlgError:
        return None
    coeffs = T_inv @ profiles
    a_vals = coeffs[0]
    b_vals = coeffs[1]
    c_vals = coeffs[2]
    n_samp = 250
    taus = np.linspace(t_lo, t_hi, n_samp)
    F = np.max(a_vals[None, :] * taus[:, None]**2 + b_vals[None, :] * taus[:, None] + c_vals[None, :], axis=1)
    best_idx = np.argmin(F)
    best_t = taus[best_idx]
    if best_idx > 0 and best_idx < n_samp - 1:
        nb = 3.0 * (t_hi - t_lo) / n_samp
        lt = np.linspace(best_t - nb, best_t + nb, 300)
        lt = lt[(lt >= t_lo) & (lt <= t_hi)]
        if len(lt) > 10:
            FL = np.max(a_vals[None, :] * lt[:, None]**2 + b_vals[None, :] * lt[:, None] + c_vals[None, :], axis=1)
            li = np.argmin(FL)
            best_t = lt[li]
    h_new = h.copy()
    h_new[i] -= best_t
    h_new[j] += best_t
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = repair_feasible(h_new)
    current_score = exact_score(h)
    new_score = exact_score(h_new)
    if new_score < current_score - 1e-13:
        return h_new, new_score
    return None

def block_line_search(h, sources, targets):
    n = len(h)
    dx = 2.0 / n
    d = np.zeros(n)
    for s in sources:
        d[s] -= 1.0
    for t in targets:
        d[t] += 1.0
    t_lo = 0.0
    t_hi = np.inf
    for idx in range(n):
        if d[idx] > 1e-15:
            t_hi = min(t_hi, (1.0 - h[idx]) / d[idx])
            t_lo = max(t_lo, -h[idx] / d[idx])
        elif d[idx] < -1e-15:
            t_hi = min(t_hi, -h[idx] / d[idx])
            t_lo = max(t_lo, (1.0 - h[idx]) / d[idx])
    t_lo = max(t_lo, 0.0)
    span = t_hi - t_lo
    if span < 1e-11:
        return None
    eps = 0.4 * span
    h_p = h + eps * d
    h_m = h - min(eps, t_lo + 0.01) * d if t_lo + 0.01 > 0 else h
    p_c = get_profile(h)
    p_p = get_profile(np.clip(h_p, 0, 1)) if t_lo >= -0.01 else get_profile(h - 0.1 * d)
    t_neg = max(t_lo, -0.1)
    if t_neg == 0 and t_lo < 0:
        h_m2 = h + t_neg * d
        p_m = get_profile(h_m2)
    else:
        t_m2 = max(t_lo + 1e-9, -0.3 * span)
        h_m2 = h + t_m2 * d
        p_m = get_profile(h_m2)
        eps = 0.4 * span
    # Use 3 points: 0, eps, 2*eps (clipped)
    tp = min(2.0 * eps, t_hi - 1e-9)
    tm = max(0.0, -0.5 * eps) if t_lo < 0 else 1e-10
    pts = np.array([tm, 0.0, tp])
    if np.any(np.abs(np.diff(np.sort(pts))) < 1e-12):
        pts = np.array([t_lo + 0.1 * span, t_lo + 0.5 * span, t_lo + 0.9 * span])
    profiles = np.zeros((3, 2 * n - 1))
    for k_idx, t in enumerate(pts):
        h_t = h + t * d
        profiles[k_idx] = np.correlate(h_t, 1.0 - h_t, mode="full") * dx
    T = np.column_stack([pts**2, pts, np.ones(3)])
    try:
        T_inv = np.linalg.inv(T)
    except np.linalg.LinAlgError:
        return None
    coeffs = T_inv @ profiles
    a_vals = coeffs[0]
    b_vals = coeffs[1]
    c_vals = coeffs[2]
    n_samp = 200
    taus = np.linspace(t_lo, t_hi, n_samp)
    F = np.max(a_vals[None, :] * taus[:, None]**2 + b_vals[None, :] * taus[:, None] + c_vals[None, :], axis=1)
    best_idx = np.argmin(F)
    best_t = taus[best_idx]
    h_new = h + best_t * d
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = repair_feasible(h_new)
    current_score = exact_score(h)
    new_score = exact_score(h_new)
    if new_score < current_score - 1e-13:
        return h_new, new_score
    return None

def generate_directed_pairs(h, profile, K=5):
    n = len(h)
    active_idx = np.argsort(profile)[::-1][:K]
    D = np.zeros(n)
    for k in active_idx:
        D += get_lag_gradient(h, int(k))
    pairs = []
    p_max = np.argmax(D)
    p_min = np.argmin(D)
    if p_max != p_min:
        pairs.append((int(p_max), int(p_min)))
    for k in active_idx:
        g = get_lag_gradient(h, int(k))
        a = np.argmax(g)
        b = np.argmin(g)
        if a != b and (int(a), int(b)) not in pairs:
            pairs.append((int(a), int(b)))
    return pairs

def symmetrize(h):
    h_sym = 0.5 * (h + h[::-1].copy())
    return repair_feasible(h_sym)

def complement_move(h):
    return repair_feasible(1.0 - h)

def interpolate_to_dim(vec, new_len):
    n = len(vec)
    x_src = (np.arange(n) + 0.5) * 2.0 / n
    x_tgt = (np.arange(new_len) + 0.5) * 2.0 / new_len

    h_new = np.interp(x_tgt, x_ext, f_ext)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)
