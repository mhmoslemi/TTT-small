def get_lag_gradients(h):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    G = np.zeros((M, n))
    for k in range(M):
        delta = k - (n - 1)
        if delta > 0:
            L = n - delta
            small_idx = np.arange(0, L)
            big_idx = np.arange(delta, n)
            G[k, big_idx] += dx * (1.0 - h[small_idx])
            G[k, small_idx] -= dx * h[big_idx]
        elif delta < 0:
            c = -delta
            L = n - c
            small_idx = np.arange(0, L)
            big_idx = np.arange(c, n)
            G[k, small_idx] += dx * (1.0 - h[big_idx])
            G[k, big_idx] -= dx * h[small_idx]
        else:
            G[k, :] += dx * (1.0 - 2.0 * h)
    return G

import numpy as np
import time
from scipy.optimize import minimize, linprog


def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))


def get_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx


def get_lag_gradients(h):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    G = np.zeros((M, n))
    for k in range(M):
        delta = k - (n - 1)
        if delta > 0:
            L = n - delta
            si = np.arange(0, L)
            bi = np.arange(delta, n)
            G[k, bi] += dx * (1.0 - h[si])
            G[k, si] -= dx * h[bi]
        elif delta < 0:
            c = -delta
            L = n - c
            si = np.arange(0, L)
            bi = np.arange(c, n)
            G[k, si] += dx * (1.0 - h[bi])
            G[k, bi] -= dx * h[si]
        else:
            G[k, :] += dx * (1.0 - 2.0 * h)
    return G


def repair(h, target=None):
    h = np.array(h, dtype=np.float64).copy()
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    if target is None:
        target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        order = np.argsort(-room)
        remaining = diff
        for idx in order:
            if remaining <= 1e-15:
                break
            add = min(remaining, room[idx])
            h[idx] += add
            remaining -= add
    else:
        order = np.argsort(-h)
        remaining = -diff
        for idx in order:
            if remaining <= 1e-15:
                break
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
    return h


def lse_objective(x, alpha, n):
    h = x
    p = get_profile(h)
    m = np.max(p)
    shifted = p - m
    return m + (1.0 / alpha) * np.log(np.sum(np.exp(alpha * shifted)))


def lse_gradient(h, alpha):
    n = len(h)
    p = get_profile(h)
    m = np.max(p)
    shifted = p - m
    w = np.exp(alpha * shifted)
    wsum = w.sum()
    if wsum < 1e-300:
        w = np.zeros_like(p)
        w[np.argmax(p)] = 1.0
    else:
        w /= wsum
    G = get_lag_gradients(h)
    return w @ G


def solve_lse(h_start, alpha, maxiter=80):
    n = len(h_start)

    def obj(h):
        return lse_objective(h, alpha, n)

    def grad(h):
        return lse_gradient(h, alpha)

    def eq_fun(h):
        return h.sum() - n / 2.0

    def eq_jac(h):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(
            obj, h_start, jac=grad,
            method='SLSQP', bounds=bounds,
            constraints=[{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}],
            options={'maxiter': maxiter, 'ftol': 1e-12}
        )
        if res.x is not None and np.all(np.isfinite(res.x)):
            return repair(res.x)
    except Exception:
        pass
    return None


def solve_lp_trust(h_center, rho):
    n = len(h_center)
    M = 2 * n - 1
    p = get_profile(h_center)
    G = get_lag_gradients(h_center)
    rhs = G @ h_center - p
    n_vars = n + 1
    A_prof = np.hstack([G, -np.ones((M, 1))])
    b_prof = rhs
    A_trust = np.zeros((2 * n, n_vars))
    b_trust = np.zeros(2 * n)
    for i in range(n):
        A_trust[i, i] = 1.0
        b_trust[i] = h_center[i] + rho
        A_trust[n + i, i] = -1.0
        b_trust[n + i] = -(h_center[i] - rho)
    A_ub = np.vstack([A_prof, A_trust])
    b_ub = np.concatenate([b_prof, b_trust])
    A_eq = np.zeros((1, n_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    c_obj = np.zeros(n_vars)
    c_obj[n] = 1.0
    bounds = [(0.0, 1.0)] * n + [(-1.0, 5.0)]

    except Exception:
        pass
    return None

def interpolate_dim(vec, new_len):
    n = len(vec)
    if n == new_len:
        return vec.copy()
    x_src = (np.arange(n) + 0.5) * 2.0 / n
    x_tgt = (np.arange(new_len) + 0.5) * 2.0 / new_len
    x_ext = np.concatenate([[0.0], x_src, [2.0]])
    f_ext = np.concatenate([[vec[0]], vec, [vec[-1]]])
    h_new = np.interp(x_tgt, x_ext, f_ext)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair(h_new)


def symmetrize(h):
