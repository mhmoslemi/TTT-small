            return jac
        
        constraints = [
            {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
            {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
        ]
        
        bounds = [(0.0, 1.0)] * n + [(max(0.0, best_c - 0.05), 1.0)]
        z0 = np.concatenate([best_h, [best_c + 1e-5]])
        
        try:
            res = minimize(obj, z0, jac=obj_grad, method='SLSQP',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': 300, 'ftol': 1e-12})
            if res.x is not None and np.all(np.isfinite(res.x)):
                cand_h = project_box_sum(res.x[:n], target)
                cand_c, _ = exact_objective(cand_h)
                if cand_c < best_c - 1e-10:
                    best_c = cand_c
                    best_h = cand_h
        except Exception:
            pass
    
    return best_h, best_c

def exhaustive_binary_vec(n, deadline):
    if n % 2 != 0 or n > 22:
        return None, float('inf')
    k = n // 2
    dx = 2.0 / n
    all_pos = list(combinations(range(n), k))
    m = len(all_pos)
    if m > 600000:
        return None, float('inf')
    
    B = np.zeros((m, n), dtype=np.float64)
    for i, pos in enumerate(all_pos):
        B[i, list(pos)] = 1.0
    
    profile = np.zeros((m, 2 * n - 1), dtype=np.float64)
    for r in range(2 * n - 1):
        e = r - (n - 1)
        if e >= 0:
            profile[:, r] = (B[:, e:] * (1.0 - B[:, :n - e])).sum(axis=1) * dx
        else:
            a = -e
            profile[:, r] = (B[:, :n - a] * (1.0 - B[:, a:])).sum(axis=1) * dx
    
    c_vals = profile.max(axis=1)
    best_idx = int(c_vals.argmin())
    best_c = c_vals[best_idx]
    best_b = B[best_idx].copy()
    
    # Also get top few for diversity
    top_k = min(10, m)
    top_idx = np.argsort(c_vals)[:top_k]
    
    return best_b, best_c, top_idx, B[top_idx]
