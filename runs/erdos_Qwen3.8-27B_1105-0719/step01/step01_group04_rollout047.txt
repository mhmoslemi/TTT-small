            idx2 = np.arange(delta, n)
            vals_bwd = h[idx2] * (1.0 - h[idx2 - delta])
            values[n - 1 - delta] = dx * np.sum(vals_bwd)
            grads[n - 1 - delta, idx2] += dx * (1.0 - h[idx2 - delta])
            grads[n - 1 - delta, idx2 - delta] += -dx * h[idx2]
    return values, grads

def interpolate_to_dimension(vec, new_len):
    n = len(vec)
    if n == new_len:
        return vec.copy()
    x_src = (np.arange(n) + 0.5) * 2.0 / n
    x_tgt = (np.arange(new_len) + 0.5) * 2.0 / new_len
    xs_ext = np.concatenate([[0.0], x_src, [2.0]])
    ys_ext = np.concatenate([[vec[0]], vec, [vec[-1]]])
    return np.interp(x_tgt, xs_ext, ys_ext)

def solve_trust_region(h_cur, values, grads, lam, active_idx):
    n = len(h_cur)
    n_lags = 2 * n - 1
    vals_act = values[active_idx]
    grads_act = grads[active_idx, :]
    num_con = len(active_idx)

    h_var = cp.Variable(n)
    t_var = cp.Variable()

    obj = cp.Minimize(t_var + lam * cp.sum_squares(h_var - h_cur))

    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n / 2.0]

    # Linearized: vals_act[i] + grads_act[i] . (h_var - h_cur) <= t_var
    # grads_act[i] . h_var <= t_var - vals_act[i] + grads_act[i] . h_cur
    for i in range(num_con):
        lhs = grads_act[i, :] @ h_var
        rhs_const = -vals_act[i] + float(np.dot(grads_act[i, :], h_cur))
        constraints.append(lhs <= t_var + rhs_const)

    prob = cp.Problem(obj, constraints)
    solvers_to_try = [cp.CLARABEL, cp.ECOS, cp.SCS]
    for solver in solvers_to_try:
        try:
            prob.solve(solver=solver)
            if prob.status in ['optimal', 'optimal_inaccurate']:
                h_model = np.asarray(h_var.value, dtype=np.float64)
                if np.all(np.isfinite(h_model)):
                    return h_model
            break
        except Exception:
            continue
    return None

def active_descent(h, max_steps=50):
    n = len(h)
    h = repair_feasible(h.copy())
    best = h.copy()
    best_score = exact_score(best)
    for step in range(max_steps):
        values, grads = compute_lag_values_and_grads(best)
        c = np.max(values)
        n_active = min(20, 2 * n - 1)
        active_idx = np.argsort(values)[::-1][:n_active]
        gaps = np.maximum(c - values[active_idx], 0.0)
        weights = np.exp(-gaps / (0.005 * c + 1e-10))
        weights /= np.sum(weights) + 1e-15
        direction = -np.sum(weights[:, None] * grads[active_idx, :], axis=0)
        direction -= np.mean(direction)
        improved = False
        for alpha in [0.03, 0.015, 0.008, 0.004, 0.002, 0.001, 0.0005]:
            h_trial = repair_feasible(best + alpha * direction)
            score = exact_score(h_trial)
            if score < best_score - 1e-14:
                best = h_trial.copy()
                best_score = score
                improved = True
                break
        if not improved:
            break
    return best, best_score

def trust_region_loop(h_start, max_iters, time_check_fn, min_time_left):
    h_cur = h_start.copy()
    s_cur = exact_score(h_cur)
    n = len(h_cur)
    n_lags = 2 * n - 1
    lam = 0.05
    stalled = 0

    for iter_count in range(max_iters):
        if time_check_fn() < min_time_left:
            break
        values, grads = compute_lag_values_and_grads(h_cur)
        c_cur = np.max(values)
        # Adaptive active set: lags within 0.015 of max, capped at 60
        threshold = c_cur - 0.015
        active_mask = values >= threshold
        n_active_count = np.sum(active_mask)
        if n_active_count < 5:
            top_idx = np.argsort(values)[::-1][:5]
            active_mask = np.zeros(n_lags, dtype=bool)
            active_mask[top_idx] = True
        elif n_active_count > 60:
            active_idx = np.argsort(values)[::-1][:60]
        else:
            active_idx = np.where(active_mask)[0]

        h_model = solve_trust_region(h_cur, values, grads, lam, active_idx)

        if h_model is None:
            lam *= 1.5
            stalled += 1
            if stalled > 6:
                break
            continue

        improved = False
        for alpha in [1.0, 0.8, 0.6, 0.4, 0.3, 0.2, 0.1, 0.05]:
            h_trial = repair_feasible(alpha * h_model + (1.0 - alpha) * h_cur)
            s_trial = exact_score(h_trial)
            if s_trial < s_cur - 1e-13:
                h_cur = h_trial
                s_cur = s_trial
                improved = True
                break

        if improved:
            stalled = 0
            if lam > 0.01:
                lam = max(0.01, lam * 0.92)
        else:
            stalled += 1
            lam *= 1.3
            if stalled > 12:
                break

    return h_cur, s_cur

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()
    rng = np.random.RandomState(seed)

    def tl():
        return budget_s - (time.perf_counter() - t_start)

    # Get parent
    try:
        parent = np.asarray(initial_h_values, dtype=np.float64)
    except Exception:
        parent = None

    if parent is not None and len(parent) > 0:
        best_h = repair_feasible(parent.copy())
    else:
        best_h = np.full(100, 0.5)

    # Try symmetrization
    h_sym = repair_feasible((best_h + best_h[::-1]) / 2.0)
    s_sym = exact_score(h_sym)
    s_orig = exact_score(best_h)
    if s_sym < s_orig - 1e-13:
        best_h = h_sym
        s_orig = s_sym

    best_score = s_orig
    best_n = len(best_h)

    # Phase 1: Trust region on parent dimension
    if tl() > 120:
        h_tr, s_tr = trust_region_loop(best_h, max_iters=200, time_check_fn=tl, min_time_left=80)
        if s_tr < best_score - 1e-13:
            best_h = h_tr
            best_score = s_tr
            best_n = len(h_tr)

    # Phase 2: Active descent polish
    if tl() > 30:
        h_ref, s_ref = active_descent(best_h, max_steps=60)
        if s_ref < best_score - 1e-13:
            best_h = h_ref
            best_score = s_ref
            best_n = len(h_ref)

    # Phase 3: Dimension exploration with trust region
    if tl() > 150:
        n_cur = len(best_h)
        dims_to_try = set()
        # Coarsening
        for dn in [10, 20, 30, 40, 50]:
            nd = n_cur - dn
            if nd >= 50:
                dims_to_try.add(nd)
        # Refinement
        for dn in [10, 20, 30, 40]:
            nd = n_cur + dn
            if nd <= 200:
                dims_to_try.add(nd)
        # Special values
        for nd in [70, 84, 100, 112, 120, 140, 160, 168]:
            if abs(nd - n_cur) <= 50:
                dims_to_try.add(nd)

        # Sort by score of interpolation (best first)
        dim_scores = []
        for nd in sorted(dims_to_try):
            if tl() < 60:
                break
            h_interp = repair_feasible(interpolate_to_dimension(best_h, nd))
            s_interp = exact_score(h_interp)
            if s_interp < best_score + 0.008:
                dim_scores.append((s_interp, nd))

        dim_scores.sort()
        for s_interp, nd in dim_scores[:4]:
            if tl() < 50:
                break
            h_interp = repair_feasible(interpolate_to_dimension(best_h, nd))
            h_tr, s_tr = trust_region_loop(h_interp, max_iters=80, time_check_fn=tl, min_time_left=30)
            if s_tr < best_score - 1e-13:
                best_h = h_tr
                best_score = s_tr
                best_n = len(h_tr)
                # Active descent on new best
                if tl() > 20:
                    h_ref, s_ref = active_descent(best_h, max_steps=30)
                    if s_ref < best_score - 1e-13:
                        best_h = h_ref
                        best_score = s_ref
                        best_n = len(h_ref)

    # Phase 4: Perturbation multi-start
    if tl() > 80:
        for trial in range(6):
            if tl() < 35:
                break
            eps = rng.uniform(0.003, 0.015)
            z = rng.randn(len(best_h))
            z -= z.mean()
            h_pert = repair_feasible(best_h + eps * z)
            s_pert = exact_score(h_pert)
            if s_pert < best_score + 0.005:
                h_ref, s_ref = active_descent(h_pert, max_steps=40)
                if s_ref < best_score - 1e-13:
                    best_h = h_ref
                    best_score = s_ref
                    best_n = len(h_ref)
                else:
                    # Try trust region from perturbation
                    if tl() > 40:
                        h_tr, s_tr = trust_region_loop(h_pert, max_iters=50, time_check_fn=tl, min_time_left=20)
                        if s_tr < best_score - 1e-13:
                            best_h = h_tr
                            best_score = s_tr
                            best_n = len(h_tr)

    # Phase 5: Final trust region with tighter active set
    if tl() > 50:
        h_cur = best_h.copy()
        s_cur = best_score
        n = len(h_cur)
        n_lags = 2 * n - 1
        lam = 0.02
        stalled = 0
        for iter_count in range(100):
            if tl() < 20:
                break
            values, grads = compute_lag_values_and_grads(h_cur)
            c_cur = np.max(values)
            threshold = c_cur - 0.008
            active_mask = values >= threshold
            if np.sum(active_mask) < 10:
                top_idx = np.argsort(values)[::-1][:10]
                active_mask = np.zeros(n_lags, dtype=bool)
                active_mask[top_idx] = True
            elif np.sum(active_mask) > 50:
                active_idx = np.argsort(values)[::-1][:50]
            else:
                active_idx = np.where(active_mask)[0]
            h_model = solve_trust_region(h_cur, values, grads, lam, active_idx)
            if h_model is None:
                lam *= 1.4
                stalled += 1
                if stalled > 8:
                    break
                continue
            improved = False
            for alpha in [1.0, 0.7, 0.5, 0.3, 0.15, 0.1]:
                h_trial = repair_feasible(alpha * h_model + (1.0 - alpha) * h_cur)
                s_trial = exact_score(h_trial)
                if s_trial < s_cur - 1e-13:
                    h_cur = h_trial
                    s_cur = s_trial
                    improved = True
                    break
            if improved:
                stalled = 0
                if lam > 0.005:
                    lam = max(0.005, lam * 0.93)
                if s_cur < best_score - 1e-13:
                    best_h = h_cur.copy()
                    best_score = s_cur
            else:
                stalled += 1
                lam *= 1.2
                if stalled > 12:
                    break

    # Phase 6: Final active descent polish
    if tl() > 15:
        h_ref, s_ref = active_descent(best_h, max_steps=40)
        if s_ref < best_score - 1e-13:
            best_h = h_ref
            best_score = s_ref

    # Final validation
    best_h = repair_feasible(best_h)
    best_score = exact_score(best_h)
    return best_h, best_score, len(best_h)
