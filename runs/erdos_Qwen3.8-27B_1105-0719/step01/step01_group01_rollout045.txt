    """Random pair search, accept first improvement found."""
    n = len(h)
    best_h = h.copy()
    best_c5 = current_c5
    t_start = time.time()
    if max_iter is None:
        max_iter = 100000000

        if time.time() - t_start > time_limit:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        # Try both directions with several fractions
        found = False
        for inc, dec in [(j, i), (i, j)]:
            cap = min(1.0 - best_h[inc], best_h[dec])
            if cap <= 1e-10:
                continue
            # Try a random fraction
            f = rng.choice([0.25, 0.5, 0.75, 1.0])
            tau = cap * f
            y = best_h.copy()
            y[inc] += tau
            y[dec] -= tau
            c = compute_c5(y)
            if c < best_c5 - 1e-14:
                best_h = y
                best_c5 = c
                found = True
                break
        if not found and rng.random() < 0.0001:
            # Occasionally try a different fraction set
            pass
    
    return best_h, best_c5
