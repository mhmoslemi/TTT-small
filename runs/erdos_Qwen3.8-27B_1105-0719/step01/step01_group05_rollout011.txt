```python
import numpy as np
from scipy.optimize import minimize
import time

def exact_objective(h):
    n = len(h)
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(profile)), profile

def project_box_sum(h, target=None):
    n = len(h)
    if target is None:
        target = 0.5 * n
    h = np.array(h, dtype=np.float64).copy()
    h = np.nan_to_num(h, nan=0.0, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
            count = np.sum(mask)
            if count == 0:
                break
            step = diff / count
            h[mask] = np.minimum(h[mask] + step, 1.0)
        else:
            mask = h > 0.0
            count = np.sum(mask)
            if count == 0:
                break
            step = diff / count
            h[mask] = np.maximum(h[mask] + step, 0.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-16 and n > 0:
        h[0] = np.clip(h[0] + diff, 0.0, 1.0)
    return h

def correlated_jacobian(h, dx):
    n = len(h)
    J = np.zeros((2*n-1, n))
    for r in range(2*n-1):
        k = r - (n-1)
        if k >= 0:
            s = n - k
            if s > 0:
                J[r, :s] = -h[k:k+s] * dx
                J[r, k:] += (1.0 - h[:s]) * dx
        else:
            a = -k
            s = n - a
            if s > 0:
                J[r, :s] = (1.0 - h[a:a+s]) * dx
                J[r, a:] += -h[:s] * dx
    return J

    n = len(h_start)
    target = n / 2.0
    dx = 2.0 / n
    h0 = project_box_sum(h_start, target)
    c0, _ = exact_objective(h0)
    
    z0 = np.append(h0, c0 + 1e-8)
    bounds = [(0.0, 1.0)] * n + [(1e-8, 1.0)]
    n_con = 2*n - 1
    
    def obj(z):
        return z[n]
    
    def obj_grad(z):
        g = np.zeros(n+1)
        g[n] = 1.0
        return g
    
    def eq_fun(z):
        return np.sum(z[:n]) - target
    
    def eq_jac(z):
        j = np.zeros(n+1)
        j[:n] = 1.0
        return j
    
    def ineq_fun(z):
        h_z = z[:n]
        c_z = z[n]
        prof = np.correlate(h_z, 1.0-h_z, mode='full') * dx
        return c_z - prof
    
    def ineq_jac(z):
        h_z = z[:n]
        Jq = correlated_jacobian(h_z, dx)
        J = np.empty((n_con, n+1))
        J[:, :n] = -Jq
        J[:, n] = 1.0
        return J
    
    constraints = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},

def downsample(h, new_n):
    n = len(h)
    if new_n == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n

def gradient_step(h, profile, step_sizes, target):
    """Try gradient descent on the max profile element."""
    n = len(h)
    dx = 2.0 / n
    best_h = h
    best_c = float(np.max(profile))
    r_max = int(np.argmax(profile))
    k = r_max - (n-1)
    
    # Gradient of q_{r_max} w.r.t. h
    grad = np.zeros(n)
    if k >= 0:
        s = n - k
        if s > 0:
            grad[:s] = -h[k:k+s] * dx
            grad[k:] += (1.0 - h[:s]) * dx
    else:
        a = -k
        s = n - a
        if s > 0:
            grad[:s] = (1.0 - h[a:a+s]) * dx
            grad[a:] = -h[:s] * dx
    
    # Also add gradient of all near-active lags
    for r in range(2*n-1):
        if profile[r] > best_c - 0.01:
            k_r = r - (n-1)
            if k_r >= 0:

                if s > 0:

                    grad[k_r:] += (1.0 - h[:s]) * dx * 0.5
            else:

                s = n - a
                if s > 0:
                    grad[:s] += (1.0 - h[a:a+s]) * dx * 0.5
                    grad[a:] -= h[:s] * dx * 0.5

        if c_new < best_c - 1e-14:
```

```python
    z0 = np.append(h0, c0 + 1e-8)
    bounds = [(0.0, 1.0)] * n + [(1e-8, 1.0)]
    
    def obj(z):
        return z[n]
    
    def obj_grad(z):
        g = np.zeros(n+1)
        g[n] = 1.0
        return g
    
    def eq_fun(z):
        return np.sum(z[:n]) - target
    
    def eq_jac(z):
        j = np.zeros(n+1)
        j[:n] = 1.0
        return j
    
    def ineq_fun(z):
        h_z = z[:n]
        c_z = z[n]
        prof = np.correlate(h_z, 1.0-h_z, mode='full') * dx
        return np.array([c_z - prof[i] for i in top_indices])
    
    def ineq_jac(z):
        h_z = z[:n]
        Jq = correlated_jacobian(h_z, dx)
        rows = [-Jq[i] for i in top_indices]
        J = np.array(rows)
        J = np.hstack([J, np.ones((len(top_indices), 1))])
        return J
    
    constraints = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    
    best_h = h0
    best_c = c0
    
    try:
        res = minimize(obj, z0, method='SLSQP', jac=obj_grad,
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': max_iter, 'ftol': ftol})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = project_box_sum(res.x[:n], target)
            c_new, prof_new = exact_objective(h_new)
            if c_new < best_c - 1e-14:
                best_h = h_new
                best_c = c_new
    except Exception:
        pass
    
    return best_h, best_c
```

Using top-K (K=20) reduces the constraint count from 139 to 20, which should make SLSQP much faster.
