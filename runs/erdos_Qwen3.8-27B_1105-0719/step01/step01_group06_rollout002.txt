                if st < best_score - 1e-12:
                    h = ht
                    best_h = h.copy()
                    best_score = st
                    improved = True
                    break
            if not improved:
                break
    return best_h, best_score

def interpolate_to_dim(vec, new_len):
    n = len(vec)
    x_src = (np.arange(n) + 0.5) * 2.0 / n
    x_tgt = (np.arange(new_len) + 0.5) * 2.0 / new_len
    x_ext = np.concatenate([[0.0], x_src, [2.0]])
    f_ext = np.concatenate([[vec[0]], vec, [vec[-1]]])
    h_new = np.interp(x_tgt, x_ext, f_ext)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

def symmetrize(h):
    return repair_feasible(0.5 * (h + h[::-1].copy()))

def lag_target_descent(h, max_iters=30, time_limit=20):
    """Directly target reducing the top lags while preserving others."""
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_score = exact_score(h)
    t0 = time.perf_counter()
    for it in range(max_iters):
        if time.perf_counter() - t0 > time_limit:
            break
        prof = get_profile(h)
        score = prof.max()
        if score < best_score - 1e-12:
            best_score = score
            best_h = h.copy()
        # Find lags within 5% of max
        threshold = score - 0.05 * score
        active = np.where(prof > threshold)[0]
        if len(active) == 0:
            break
        G = get_lag_gradients(h)
        # Equal-weight average of active lag gradients
        direction = G[active, :].mean(axis=0)
        direction -= direction.mean()
        dn = np.linalg.norm(direction)
        if dn < 1e-14:
            break
        direction /= dn
        improved = False
        for alpha in [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]:
            ht = repair_feasible(h - alpha * direction)
            st = exact_score(ht)
            if st < best_score - 1e-12:
                h = ht
                best_h = h.copy()
                best_score = st
                improved = True
                break
        if not improved:
            break
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0_global = time.perf_counter()

    def tleft():
        return budget_s - (time.perf_counter() - t0_global)

    try:
        parent = np.array(initial_h_values, dtype=np.float64).copy()
    except Exception:
        parent = np.full(140, 0.5)

    parent = repair_feasible(parent)
    best_h = parent.copy()
    best_score = exact_score(parent)

    # Symmetrize
    hs = symmetrize(best_h)
    ss = exact_score(hs)
    if ss < best_score - 1e-13:
        best_h = hs
        best_score = ss

    # Stage 1: LSE temperature sweep
    for beta in [3.0, 6.0, 10.0, 15.0, 25.0, 40.0]:
        if tleft() < 15:
            break
        h_sol = solve_lse(best_h, beta, maxiter=70)
        if h_sol is not None:
            s = exact_score(h_sol)
            if s < best_score - 1e-12:
                best_h = h_sol
                best_score = s

    # Stage 2: Lag-target descent
    if tleft() > 20:
        tl = min(20, tleft() * 0.2)
        h_ld, s_ld = lag_target_descent(best_h, max_iters=30, time_limit=tl)
        if s_ld < best_score - 1e-12:
            best_h = h_ld
            best_score = s_ld

    # Stage 3: Adaptive epigraph
    for nc in [25, 50, 100]:
        if tleft() < 15:
            break
        h_sol, s_sol = solve_epigraph(best_h, best_score + 1e-6, n_cons=nc, maxiter=55)
        if s_sol < best_score - 1e-12:
            best_h = h_sol
            best_score = s_sol

    # Stage 4: LSE local descent
    if tleft() > 25:
        tl = min(25, tleft() * 0.25)
        h_loc, s_loc = local_descent_lse(best_h, max_iters=50, time_limit=tl)
        if s_loc < best_score - 1e-12:
            best_h = h_loc
            best_score = s_loc

    # Stage 5: Multi-start perturbations
    if tleft() > 60:
        n = len(best_h)
        for trial in range(10):
            if tleft() < 20:
                break
            if trial < 4:
                mode = 2 + trial * 3
                idx = np.arange(n)
                freq = 2.0 * np.pi * mode / n
                phase = np.random.uniform(0, 2 * np.pi)
                pert = np.sin(idx * freq + phase)
            elif trial < 7:
                pert = np.random.randn(n)
                # Smooth
                from scipy.ndimage import uniform_filter1d
                pert = uniform_filter1d(pert, size=7)
            else:
                # Block perturbation
                pert = np.random.uniform(-0.5, 0.5, n)
                from scipy.ndimage import uniform_filter1d
                pert = uniform_filter1d(pert, size=3)
            pert -= pert.mean()
            eps = 0.03 * (1.0 / (1.0 + 0.5 * trial))
            hp = repair_feasible(best_h + eps * pert)
            sp = exact_score(hp)
            # Quick LSE
            h_sol = solve_lse(hp, beta=10.0, maxiter=40)
            if h_sol is not None:
                s = exact_score(h_sol)
                if s < best_score - 1e-12:
                    best_h = h_sol
                    best_score = s
                    # Follow up with epigraph
                    h_sol2, s_sol2 = solve_epigraph(best_h, best_score + 1e-6, n_cons=30, maxiter=30)
                    if s_sol2 < best_score - 1e-12:
                        best_h = h_sol2
                        best_score = s_sol2

    # Stage 6: Dimension search
    if tleft() > 50:
        pn = len(best_h)
        dims = []
        for d in [pn - 8, pn - 4, pn + 4, pn + 8, pn + 16, pn + 32, pn - 16]:
            if 40 <= d <= 300 and d != pn:
                dims.append(d)
        for p2 in [64, 96, 128, 160, 192, 224, 256]:
            if p2 != pn and 40 <= p2 <= 300:
                dims.append(p2)
        dims = sorted(set(dims))
        for new_n in dims:
            if tleft() < 20:
                break
            hi = interpolate_to_dim(best_h, new_n)
            si = exact_score(hi)
            if si < best_score + 0.005:
                h_sol = solve_lse(hi, beta=5.0, maxiter=40)
                if h_sol is not None:
                    s = exact_score(h_sol)
                    if s < best_score - 1e-12:
                        best_h = h_sol
                        best_score = s
                h_sol2, s_sol2 = solve_epigraph(best_h, best_score + 1e-6, n_cons=30, maxiter=35)
                if s_sol2 < best_score - 1e-12:
                    best_h = h_sol2
                    best_score = s_sol2

    # Stage 7: Final intensive polishing cycle
    for rnd in range(4):
        if tleft() < 10:
            break
        h_sol = solve_lse(best_h, beta=min(80.0, 10.0 * (rnd + 1)), maxiter=50)
        if h_sol is not None:
            s = exact_score(h_sol)
            if s < best_score - 1e-12:
                best_h = h_sol
                best_score = s
        h_sol2, s_sol2 = solve_epigraph(best_h, best_score + 1e-6, n_cons=60, maxiter=40)
        if s_sol2 < best_score - 1e-12:
            best_h = h_sol2
            best_score = s_sol2
        if tleft() > 8:
            h_loc, s_loc = local_descent_lse(best_h, max_iters=20, time_limit=min(8, tleft() * 0.3))
            if s_loc < best_score - 1e-12:
                best_h = h_loc
                best_score = s_loc
        h_ld, s_ld = lag_target_descent(best_h, max_iters=20, time_limit=min(6, tleft() * 0.2))
        if s_ld < best_score - 1e-12:
            best_h = h_ld
            best_score = s_ld

    best_h = repair_feasible(best_h)
    final_score = exact_score(best_h)
    return (best_h, final_score, len(best_h))
