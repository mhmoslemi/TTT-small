import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time


def evaluate_exact(h):
    n = len(h)
    dx = 2.0 / n
    return max(np.correlate(h, 1.0 - h, mode='full')) * dx


def corr_array(h):
    return np.correlate(h, 1.0 - h, mode='full')


def c5_of_h(h):
    return corr_array(h).max() * (2.0 / len(h))


def project_to_target(x, target_sum):
    n = len(x)
    lo = -max(x.max() + 1.0, 1.0)
    hi = max(1.0 - x.min(), 1.0)
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(x + mid, 0.0, 1.0).sum()
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    h = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h = h.copy()
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)


def build_symh(q):
    m = len(q)
    h = np.empty(2 * m)
    h[:m] = q
    h[m:] = 1.0 - q[::-1]
    return h


def extract_q(h):
    m = len(h) // 2
    q = (h[:m] + (1.0 - h[m:][::-1])) / 2.0
    return np.clip(q, 0.0, 1.0)


def grad_corr_k(h, k, n):
    offset = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -offset)
    i_hi = min(n - 1, n - 1 - offset)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    j = idx + offset
    grad[idx] += (1.0 - h[j])
    grad[j] -= h[idx]
    return grad


def sym_grad_from_full(gh, m):
    return gh[:m] - gh[m:][::-1]


class SymEpiCon:
    def __init__(self, k, n, dx):
        self.k = k
        self.n = n
        self.dx = dx

    def fun(self, z):
        m = self.n // 2
        q = z[:m]
        t = z[-1]
        h = build_symh(q)
        return t - self.dx * corr_array(h)[self.k]

    def jac(self, z):
        m = self.n // 2
        q = z[:m]
        h = build_symh(q)
        gh = grad_corr_k(h, self.k, self.n)
        gq = sym_grad_from_full(gh, m)
        g = np.zeros(m + 1)
        g[:m] = -self.dx * gq
        g[-1] = 1.0
        return g


class FullEpiCon:
    def __init__(self, k, n, dx):
        self.k = k
        self.n = n
        self.dx = dx

    def fun(self, z):
        h = z[:self.n]
        t = z[-1]
        return t - self.dx * corr_array(h)[self.k]

    def jac(self, z):
        h = z[:self.n]
        gh = -self.dx * grad_corr_k(h, self.k, self.n)
        g = np.zeros(self.n + 1)
        g[:self.n] = gh
        g[-1] = 1.0
        return g


class SumEqCon:
    def __init__(self, n):
        self.n = n

    def fun(self, z):
        return z[:self.n].sum() - self.n / 2.0

    def jac(self, z):
        g = np.zeros(self.n + 1)
        g[:self.n] = 1.0
        return g


def epi_obj_full(z):
    return z[-1]


def epi_obj_grad_full(z):
    dim = len(z)
    g = np.zeros(dim)
    g[-1] = 1.0
    return g


def smooth_sym_obj(z, n, beta):
    h = build_symh(z)
    dx = 2.0 / n
    corrs = corr_array(h)
    return logsumexp(beta * corrs) / beta * dx


def smooth_sym_grad(z, n, beta):
    m = n // 2
    h = build_symh(z)
    dx = 2.0 / n
    corrs = corr_array(h)
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones_like(corrs) / len(corrs)
    else:
        w /= w_sum
    gh = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-13:
            continue
        gh += w[k] * grad_corr_k(h, k, n)
    gh *= dx
    return sym_grad_from_full(gh, m)

        return np.clip(res.x, 0.0, 1.0)
    except Exception:
        return q0

        return q0

        return g

                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-10})
        return np.clip(res.x[:m], 0.0, 1.0)
    except Exception:
        return q0

    if len(active_list) > max_count:

    return active_list
