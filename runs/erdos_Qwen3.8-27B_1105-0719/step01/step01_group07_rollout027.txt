from scipy.optimize import minimize
import time

def exact_objective(h):
    n = len(h)
    if n == 0:
        return float('inf')
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(profile))

def exact_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_box_sum(h):
    h = np.array(h, dtype=np.float64).copy()
    n = len(h)
    target = n / 2.0
    h[~np.isfinite(h)] = 0.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        h[mask] = np.clip(h[mask] + diff / cnt, 0.0, 1.0)
    return h

def anti_sym_press(h, n):
    """Create anti-symmetric sequence from first half."""
    m = n // 2
    first = h[:m]
    second = 1.0 - first[::-1]
    return np.concatenate([first, second])

def make_anti_from_full(h):
    """Extract anti-symmetric component from full sequence."""
    n = len(h)
    m = n // 2
    first = (h[:m] + (1.0 - h[m-1::-1])) / 2.0
    first = np.clip(first, 0.0, 1.0)
    return np.concatenate([first, 1.0 - first[::-1]])

def compute_c_all(h):
    """Compute c[k] for all k (unscaled, just counts for binary)."""
    n = len(h)
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    # profile[k + n - 1] = c[k] for k = -(n-1), ..., n-1
    return profile

def local_search_anti_binary(first_bits, n, max_rounds=1000):
    """
    Local search on anti-symmetric binary sequence.
    first_bits: array of m = n/2 binary values (first half)
    Returns optimized first_bits and objective value.
    """
    m = n // 2
    h = np.concatenate([first_bits, 1.0 - first_bits[::-1]]).astype(float)
    curr_profile = compute_c_all(h)
    curr_max = float(np.max(curr_profile))
    
    for rnd in range(max_rounds):
        best_improve = 1e-14
        best_bit = -1
        for i in range(m):
            # Try flipping bit i
            # h[i] and h[n-1-i] both flip
            a = i
            b = n - 1 - i
            # Compute delta
            new_h = h.copy()
            new_h[a] = 1.0 - h[a]
            new_h[b] = 1.0 - h[b]
            new_profile = np.correlate(new_h, 1.0 - new_h, mode='full') * (2.0/n)
            new_max = float(np.max(new_profile))
            improve = curr_max - new_max
            if improve > best_improve:
                best_improve = improve
                best_bit = i
        if best_bit >= 0:
            a = best_bit
            b = n - 1 - best_bit
            h[a] = 1.0 - h[a]
            h[b] = 1.0 - h[b]
            first_bits = h[:m].copy()
            curr_profile = compute_c_all(h)
            curr_max = float(np.max(curr_profile))
        else:
            break
    
    return first_bits, curr_max

def simulated_anneal_anti_binary(first_bits_init, n, t_start, t_end, max_iters, rng):
    """Simulated annealing on anti-symmetric binary sequence."""
    m = n // 2
    h = np.concatenate([first_bits_init, 1.0 - first_bits_init[::-1]]).astype(float)
    curr_profile = compute_c_all(h)
    curr_max = float(np.max(curr_profile))
    best_h = h.copy()
    best_max = curr_max
    dx = 2.0 / n
    
    for it in range(max_iters):
        # Cool
        frac = it / max_iters
        T = t_start * (t_end / t_start) ** frac

        i = rng.randint(m)
        a = i
        b = n - 1 - i

        new_h[a] = 1.0 - h[a]
        new_h[b] = 1.0 - h[b]

        new_max = float(np.max(new_profile))

        if delta < 0 or (T > 0 and rng.random() < np.exp(-delta / T)):
            h = new_h
            curr_max = new_max
            if curr_max < best_max:
                best_max = curr_max
                best_h = h.copy()

def local_search_general_binary(h, max_rounds=500, deadline=None):
    """General binary local search (not restricted to anti-symmetric)."""
    n = len(h)
    h = np.array(h, dtype=np.float64).copy()

    curr_max = float(np.max(curr_profile))

    zeros = set(np.where(h == 0.0)[0].tolist())

            break
        best_improve = 1e-14
        best_swap = None
        for i in ones:
            for j in zeros:
                h[i] = 0.0
                h[j] = 1.0
                new_profile = np.correlate(h, 1.0 - h, mode='full') * dx
                new_max = float(np.max(new_profile))

            ones.add(j)
            zeros.add(i)
            zeros.remove(j)
            curr_profile = np.correlate(h, 1.0 - h, mode='full') * dx
            curr_max = float(np.max(curr_profile))
        else:
            break
    
    return h, curr_max

def make_binary_seed(n, rng, mode):
    """Create various binary seed patterns with exactly n/2 ones."""
    m = n // 2
    if mode == 0:
        # Random
        idx = rng.choice(n, size=m, replace=False)
        h = np.zeros(n)
        h[idx] = 1.0
    elif mode == 1:
        # Block
        h = np.zeros(n)
        h[:m] = 1.0
    elif mode == 2:
        # Alternating
        h = np.zeros(n)
        h[::2] = 1.0
        if m > n // 2:
            h[2*m-1] = 1.0  # shouldn't happen for even n
        # Actually for even n, n[::2] gives exactly n/2 elements
    elif mode == 3:
        # Anti-symmetric random
        first = np.zeros(m)
        first[rng.choice(m, size=m//2, replace=False)] = 1.0
        h = np.concatenate([first, 1.0 - first[::-1]])
    elif mode == 4:
        # Based on modular pattern
        h = np.zeros(n)
        # Use a low-discrepancy-like pattern
        for i in range(n):
            if (i * 7 + 3) % n < m:  # This won't give exactly m ones
                h[i] = 1.0
        # Fix to get exactly m ones
        count = int(np.sum(h))
        if count > m:
            ones = np.where(h == 1.0)[0]
            excess = count - m
            to_remove = rng.choice(ones, size=excess, replace=False)
            h[to_remove] = 0.0
        elif count < m:
            zeros = np.where(h == 0.0)[0]
            needed = m - count
            to_add = rng.choice(zeros, size=needed, replace=False)
            h[to_add] = 1.0
    else:
        # First m positions shuffled
        h = np.zeros(n)
        h[rng.permutation(n)[:m]] = 1.0
    return h

def transfer_to_size(h_src, n_target):
    n_src = len(h_src)

    h_new = np.interp(x_tgt, x_src, h_src)
    return project_box_sum(h_new)

For anti-symmetric binary with h determined by first_bits[0..m-1]:
Flipping first_bits[i] changes h[i] and h[n-1-i].

The delta to c[k] for each k:
- Position a = i goes from old_val to 1-old_val
- Position b = n-1-i goes from old_val_b to 1-old_val_b (where old_val_b = 1-old_val due to anti-sym)

So if old_val = 0: a goes 0→1, b goes 1→0.
If old_val = 1: a goes 1→0, b goes 0→1.

Case old_val = 0 (a: 0→1, b: 1→0):
delta_c[k] = (contribution from a going 0→1) + (contribution from b going 1→0)

From a (0→1):
- As j+k position: for k where a-k is valid (0 ≤ a-k ≤ n-1): c[k] += (1-h[a-k])
  Valid k: a-(n-1) ≤ k ≤ a, i.e., -(n-1-a) ≤ k ≤ a
- As j position: for k where a+k is valid (0 ≤ a+k ≤ n-1): c[k] -= h[a+k]
  Valid k: 0 ≤ k ≤ n-1-a

From b (1→0):
- As j+k position: for k where b-k is valid: c[k] -= (1-h[b-k])
  Valid k: b-(n-1) ≤ k ≤ b
- As j position: for k where b+k is valid: c[k] += h[b+k]
  Valid k: 0 ≤ k ≤ n-1-b

Combined:
delta_c[k] = [-(n-1-a) ≤ k ≤ a] * (1-h[a-k]) - [0 ≤ k ≤ n-1-a] * h[a+k] - [b-(n-1) ≤ k ≤ b] * (1-h[b-k]) + [0 ≤ k ≤ n-1-b] * h[b+k]

Since a + b = n-1:
- a-(n-1) = a-b = a-(n-1-a) = 2a-(n-1)
- b-(n-1) = b-a = n-1-2a
- n-1-a = b
- n-1-b = a

So:
delta_c[k] = [2a-(n-1) ≤ k ≤ a] * (1-h[a-k]) - [0 ≤ k ≤ b] * h[a+k] - [n-1-2a ≤ k ≤ b] * (1-h[b-k]) + [0 ≤ k ≤ a] * h[b+k]

This can be computed in O(n) for all k simultaneously using vectorized numpy operations.

One more consideration: the "2-swap" or "move" operation. Instead of just flipping one anti-symmetric pair, we could:
- Move a 1 from position i to position j (within the first half), which changes two positions
- This is equivalent to flipping bit i (1→0) and bit j (0→1) in the first half

This gives a richer search than single-bit flips.

For the general (non-anti) case, a "move" from i to j means: h[i] 1→0, h[j] 0→1. This is different from a simple swap in the anti-symmetric case.

Let me include both single-bit flips and 2-bit moves in the local search.

Wait, one more important consideration. Let me re-examine what np.correlate(h, 1-h, mode='full') actually computes.

From numpy docs: correlate(a, v, mode='full') computes the cross-correlation:
out[n] = sum_m a[m+n] * v[m] for valid m

Wait, let me check more carefully. The numpy docs say:
c_k = sum_n a_{n+k} * conj(v_n)

So c_k = sum_n a[n+k] * conj(v[n]) where the sum is over all n such that both n and n+k are valid indices.

With a = h and v = 1-h:
c_k = sum_n h[n+k] * (1-h[n])

The output has length 2*n-1, with the zero-lag at index n-1.
So output[j] = c_{j-(n-1)} for j = 0, 1, ..., 2*n-2.

So output[n-1] = c_0 = sum_n h[n]*(1-h[n])
output[n-1+k] = c_k for k > 0
output[n-1-k] = c_{-k} for k > 0

Great, this confirms my earlier analysis.

Now, for the objective: C₅ = max_k c_k * dx = max(output) * dx.

Note that c_0 = sum h[n]*(1-h[n]). For binary h, this is 0 (since h*(1-h) = 0 for h in {0,1}). For continuous h, this is sum h*(1-h) which can be positive.

So for binary sequences, the maximum is always achieved at some k ≠ 0. Good.

import numpy as np
from scipy.optimize import minimize
import time

def exact_objective(h):
    n = len(h)
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(profile))

def project_box_sum(h):
    h = np.array(h, dtype=np.float64).copy()
    n = len(h)
    target = n / 2.0
    h[~np.isfinite(h)] = 0.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        h[mask] = np.clip(h[mask] + diff / float(cnt), 0.0, 1.0)
    return h

def make_anti_full(first_bits):
    m = len(first_bits)
    return np.concatenate([first_bits, 1.0 - first_bits[::-1]])

def ls_anti(first_bits, n, max_rounds, deadline):
    m = n // 2
    h = make_anti_full(first_bits).astype(float)
    dx = 2.0 / n
    curr_prof = np.correlate(h, 1.0 - h, mode='full') * dx
    curr_max = float(np.max(curr_prof))
    
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        best_d = 1e-14
        best_i = -1
        for i in range(m):
            h[i] = 1.0 - h[i]
            h[n-1-i] = 1.0 - h[n-1-i]
            nm = float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)
            h[i] = 1.0 - h[i]
            h[n-1-i] = 1.0 - h[n-1-i]
            d = curr_max - nm
            if d > best_d:
                best_d = d
                best_i = i
        if best_i >= 0:
            h[best_i] = 1.0 - h[best_i]
            h[n-1-best_i] = 1.0 - h[n-1-best_i]
            curr_max = curr_max - best_d
            curr_prof = np.correlate(h, 1.0 - h, mode='full') * dx
        else:
            break
    
    return h[:m].copy(), curr_max

def ls_anti_2flip(first_bits, n, max_rounds, deadline, rng):
    """Local search with 2-bit flips (move operation)."""
    m = n // 2
    h = make_anti_full(first_bits).astype(float)
    dx = 2.0 / n
    curr_max = float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)
    
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        best_d = 1e-14
        best_move = None
        # Try all pairs of bits to swap (flip both)
        ones = [i for i in range(m) if h[i] == 1.0]
        zeros = [i for i in range(m) if h[i] == 0.0]
        for i in ones:
            for j in zeros:
                h[i] = 0.0
                h[j] = 1.0
                h[n-1-i] = 1.0
                h[n-1-j] = 0.0
                nm = float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)
                h[i] = 1.0
                h[j] = 0.0
                h[n-1-i] = 0.0
                h[n-1-j] = 1.0
                d = curr_max - nm
                if d > best_d:
                    best_d = d
                    best_move = (i, j)
        if best_move is not None:
            i, j = best_move
            h[i] = 0.0
            h[j] = 1.0
            h[n-1-i] = 1.0
            h[n-1-j] = 0.0
            curr_max = curr_max - best_d
        else:
            # Try single flips as fallback
            for i in range(m):
                h[i] = 1.0 - h[i]
                h[n-1-i] = 1.0 - h[n-1-i]
                nm = float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)
                h[i] = 1.0 - h[i]
                h[n-1-i] = 1.0 - h[n-1-i]
                d = curr_max - nm
                if d > best_d:
                    best_d = d
                    best_move = (i, -1)  # single flip marker
            if best_move is not None and best_move[1] == -1:
                i = best_move[0]
                h[i] = 1.0 - h[i]
                h[n-1-i] = 1.0 - h[n-1-i]
                curr_max = curr_max - best_d
            else:
                break
    
    return h[:m].copy(), curr_max

def sa_anti(first_bits, n, t0, t1, n_iters, deadline, rng):
    m = n // 2
    h = make_anti_full(first_bits).astype(float)
    dx = 2.0 / n
    curr_max = float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)
    best_h = h.copy()
    best_max = curr_max
    
    for it in range(n_iters):
        if time.time() > deadline:
            break
        frac = float(it) / float(n_iters)
        T = t0 * (t1 / t0) ** frac
        if T < 1e-8:
            T = 1e-8
        
        i = rng.randint(m)
        h[i] = 1.0 - h[i]
        h[n-1-i] = 1.0 - h[n-1-i]
        new_max = float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)
        delta = new_max - curr_max
        if delta < 0 or (T > 0 and rng.random() < np.exp(-delta / T)):
            curr_max = new_max
            if curr_max < best_max:
                best_max = curr_max
                best_h = h.copy()
        else:
            h[i] = 1.0 - h[i]
            h[n-1-i] = 1.0 - h[n-1-i]
    
    return best_h[:m].copy(), best_max

def ls_general(h, max_rounds, deadline):
    n = len(h)
    h = np.array(h, dtype=np.float64).copy()
    dx = 2.0 / n
    curr_max = float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)
    ones = set(np.where(h == 1.0)[0].tolist())
    zeros = set(np.where(h == 0.0)[0].tolist())
    
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        best_d = 1e-14
        best_swap = None
        for i in ones:
            for j in zeros:
                h[i] = 0.0
                h[j] = 1.0
                nm = float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)
                h[i] = 1.0
                h[j] = 0.0
                d = curr_max - nm
                if d > best_d:
                    best_d = d
                    best_swap = (i, j)
        if best_swap is not None:
            i, j = best_swap
            h[i] = 0.0
            h[j] = 1.0
            ones.discard(i)
            ones.add(j)
            zeros.add(i)
            zeros.discard(j)
            curr_max = curr_max - best_d
        else:
            break
    
    return h, curr_max
