def cdf_transfer(h, new_n):
    n = len(h)
    dx = 2.0 / n
    # Cumulative integral at cell boundaries
    # H(0) = 0, H(i*dx) = dx * sum(h[0:i])
    cumsum = np.concatenate(([0.0], np.cumsum(h))) * dx
    # New grid cell boundaries
    new_dx = 2.0 / new_n
    new_boundaries = np.arange(new_n + 1) * new_dx
    # Interpolate H at new boundaries
    old_boundaries = np.arange(n + 1) * dx
    new_H = np.interp(new_boundaries, old_boundaries, cumsum)
    # h_new[j] = (H((j+1)*new_dx) - H(j*new_dx)) / new_dx
    h_new = np.diff(new_H) / new_dx
    return h_new

        h_pert = repair_feasible(parent + eps * z)
        candidates.append(('pert_{}'.format(eps), h_pert))

    # Sort candidates by score
    scored_cands = []
    for name, h_cand in candidates:
        s = exact_score(h_cand)
        scored_cands.append((s, name, h_cand))
    scored_cands.sort(key=lambda x: x[0])

    # If perturbed ones are better, update best
    for s, name, h_cand in scored_cands:
        if s < best_score - 1e-12:
            best_h = h_cand
            best_score = s

    # Define dimensions to explore
    dims_to_try = set()
    for frac in [0.5, 0.6, 0.7, 0.75, 0.85, 1.0, 1.1, 1.25, 1.5]:
        d = int(round(base_n * frac))
        d = max(40, min(d, 220))
        dims_to_try.add(d)
    # Also add some specific values
    for d in [70, 80, 90, 100, 110, 120, base_n, base_n + 20, base_n + 40]:
        if 40 <= d <= 220:
            dims_to_try.add(d)
    dims_to_try = sorted(dims_to_try)

    # Beta schedule for annealing
    beta_schedule = [15, 30, 60, 120, 250]

    # Main optimization loop over dimensions
    for dim in dims_to_try:
        if tl() < 40:
            break
        # Get starting point for this dimension
        h_start = repair_feasible(cdf_transfer(best_h, dim))
        s_start = exact_score(h_start)

        # If starting score is much worse, skip
        if s_start > best_score + 0.02:
            continue

        # Run soft-max SLSQP with beta annealing
        h_cur = h_start.copy()
        for beta in beta_schedule:
            if tl() < 25:
                break
            # Scale maxiter based on dimension
            if dim > 150:
                maxiter = 100
            elif dim > 100:
                maxiter = 150
            else:
                maxiter = 200
            h_sol = solve_softmax(h_cur, beta, maxiter=maxiter)
            if h_sol is not None:
                s_sol = exact_score(h_sol)
                if s_sol < exact_score(h_cur) - 1e-13:
                    h_cur = h_sol
                else:
                    h_cur = h_sol  # Accept anyway, continue annealing

        s_dim = exact_score(h_cur)
        if s_dim < best_score - 1e-12:
            best_h = h_cur.copy()
            best_score = s_dim

        # Active descent polish on this dimension's solution
        if tl() > 20:
            h_ad, s_ad = active_descent(h_cur, max_steps=40)
            if s_ad < best_score - 1e-12:
                best_h = h_ad.copy()
                best_score = s_ad

    # Additional passes: re-run with best solution transferred to each dim
    for dim in dims_to_try:
        if tl() < 35:
            break
        h_start = repair_feasible(cdf_transfer(best_h, dim))
        s_start = exact_score(h_start)
        if s_start > best_score + 0.015:
            continue

        h_cur = h_start.copy()
        for beta in [50, 100, 200, 400]:
            if tl() < 20:
                break
            maxiter = 150 if dim <= 120 else 100
            h_sol = solve_softmax(h_cur, beta, maxiter=maxiter)
            if h_sol is not None:
                h_cur = h_sol

        s_dim = exact_score(h_cur)
        if s_dim < best_score - 1e-12:
            best_h = h_cur.copy()
            best_score = s_dim

        if tl() > 15:
            h_ad, s_ad = active_descent(h_cur, max_steps=50)
            if s_ad < best_score - 1e-12:
                best_h = h_ad.copy()
                best_score = s_ad

    # Multi-start with random perturbations at best dimension
    if tl() > 60:
        n_best = len(best_h)
        for trial in range(12):
            if tl() < 30:
                break
            eps = rng.uniform(0.002, 0.01)
            z = rng.uniform(-1, 1, size=n_best)
            z -= z.mean()
            h_pert = repair_feasible(best_h + eps * z)
            h_cur = h_pert
            for beta in [40, 80, 160]:
                if tl() < 20:
                    break
                h_sol = solve_softmax(h_cur, beta, maxiter=120)
                if h_sol is not None:
                    h_cur = h_sol
            s_r = exact_score(h_cur)
            if s_r < best_score - 1e-12:
                best_h = h_cur.copy()
                best_score = s_r
            if tl() > 10:
                h_ad, s_ad = active_descent(h_cur, max_steps=30)
                if s_ad < best_score - 1e-12:
                    best_h = h_ad.copy()
                    best_score = s_ad

    # Final intensive polishing at best dimension
    if tl() > 30:
        n_best = len(best_h)
        h_cur = best_h.copy()
        for beta in [100, 200, 500]:
            if tl() < 15:
                break
            h_sol = solve_softmax(h_cur, beta, maxiter=200)
            if h_sol is not None:
                h_cur = h_sol
        s_final = exact_score(h_cur)
        if s_final < best_score - 1e-12:
            best_h = h_cur.copy()
            best_score = s_final

    # Active descent final
    if tl() > 15:
        h_ad, s_ad = active_descent(best_h, max_steps=80)
        if s_ad < best_score - 1e-12:
            best_h = h_ad.copy()
            best_score = s_ad

    # Try symmetrization of final best
    h_sym_final = repair_feasible(symmetrize(best_h))
    s_sym = exact_score(h_sym_final)
    if s_sym < best_score - 1e-12:
        best_h = h_sym_final
        best_score = s_sym
        if tl() > 15:
            h_ad, s_ad = active_descent(best_h, max_steps=40)
            if s_ad < best_score - 1e-12:
                best_h = h_ad.copy()
                best_score = s_ad

    # Final one more SLSQP + active descent
    if tl() > 20:
        h_sol = solve_softmax(best_h, 200, maxiter=200)
        if h_sol is not None:
            s_sol = exact_score(h_sol)
            if s_sol < best_score - 1e-12:
                best_h = h_sol
                best_score = s_sol
        h_ad, s_ad = active_descent(best_h, max_steps=50)
        if s_ad < best_score - 1e-12:
            best_h = h_ad.copy()
            best_score = s_ad

    # Final validation
    best_h = repair_feasible(best_h)
    best_score = exact_score(best_h)
    n_final = len(best_h)
    return best_h, best_score, n_final
