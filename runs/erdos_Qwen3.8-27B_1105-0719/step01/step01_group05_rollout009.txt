def exact_c5(h):
    n = len(h)
    if n == 0:
        return float('inf')
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(profile))

def exact_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_box_sum(h, target=None):
    if target is None:
        target = 0.5 * len(h)
    n = len(h)
    h = np.array(h, dtype=np.float64).copy()
    h = np.nan_to_num(h, nan=0.0, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = np.sum(mask)
            if cnt == 0:
                break
            step = diff / cnt
            h[mask] = np.minimum(h[mask] + step, 1.0)
        else:
            mask = h > 0.0
            cnt = np.sum(mask)
            if cnt == 0:
                break
            step = diff / cnt
            h[mask] = np.maximum(h[mask] + step, 0.0)
    # Final correction
    diff = target - np.sum(h)
    if abs(diff) > 1e-15 and n > 0:
        h[0] = np.clip(h[0] + diff, 0.0, 1.0)
    return h

def lag_gradient(h, lag, dx):
    n = len(h)
    g = np.zeros(n)
    if lag >= 0:
        s = n - lag
        if s > 0:
            g[:s] -= h[lag:]
            g[lag:] += 1.0 - h[:s]
    else:
        a = -lag
        s = n - a
        if s > 0:
            g[:s] += 1.0 - h[a:]
            g[a:] -= h[:s]
    return g * dx

    dx = 2.0 / n

        G = np.array(grads)  # shape (len(active), n)

        # This is: find d such that G_i · d < 0 for all i

        # Better: solve LP minimize t s.t. G_i · d <= t, sum(d)=0, ||d||_inf <= 1
        # We want t < 0
        
        # Use scipy linprog
        from scipy.optimize import linprog
        # Variables: d (n) + t (1) = n+1 variables
        # Minimize t

        #                sum(d) = 0
        #                -1 <= d_i <= 1

        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0  # minimize t

        A_ub = np.zeros((m, n + 1))
        A_ub[:, :n] = G
        A_ub[:, n] = -1.0
        b_ub = np.zeros(m)

            direction = res.x[:n].copy()
            direction -= np.mean(direction)  # ensure sum = 0

import numpy as np
from scipy.optimize import linprog
import time


def exact_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx


def exact_c5(h):
    return float(np.max(exact_profile(h)))


def project_box_sum(h, target=None):
    if target is None:
        target = 0.5 * len(h)
    n = len(h)
    h = np.array(h, dtype=np.float64).copy()
    h = np.nan_to_num(h, nan=0.0, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = np.sum(mask)
            if cnt == 0:
                break
            step = diff / cnt
            h[mask] = np.minimum(h[mask] + step, 1.0)
        else:
            mask = h > 0.0
            cnt = np.sum(mask)
            if cnt == 0:
                break
            step = diff / cnt
            h[mask] = np.maximum(h[mask] + step, 0.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-15 and n > 0:
        h[0] = np.clip(h[0] + diff, 0.0, 1.0)
    return h


def lag_gradient(h, lag, dx):
    n = len(h)
    g = np.zeros(n)
    if lag >= 0:
        s = n - lag
        if s > 0:
            g[:s] -= h[lag:]
            g[lag:] += 1.0 - h[:s]
    else:
        a = -lag
        s = n - a
        if s > 0:
            g[:s] += 1.0 - h[a:]
            g[a:] -= h[:s]
    return g * dx


def find_direction(h, active_lags, dx, n):
    """Find descent direction via LP. Returns (direction, t_value)."""
    G = np.array([lag_gradient(h, lag, dx) for lag in active_lags])
    m = len(G)
    if m == 0:
        return None, 0.0

    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0

    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = np.zeros(m)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = [0.0]

    bounds = [(-1.0, 1.0)] * n + [(None, None)]

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            d = res.x[:n].copy()
            d -= np.mean(d)
            t_val = float(res.fun)
            if np.max(np.abs(d)) < 1e-14:
                return None, t_val
            return d, t_val
    except Exception:
        pass
    return None, 0.0


def active_set_descent(h, deadline, max_iter=400, trust_init=0.08):
    n = len(h)
    target = 0.5 * n
    dx = 2.0 / n
    best_h = h.copy()
    best_c = exact_c5(best_h)
    trust = trust_init
    stall = 0

    for it in range(max_iter):
        if time.time() > deadline - 5:
            break
        if best_c <= 0.38083:
            break

        profile = exact_profile(best_h)
        c = float(np.max(profile))

        act_tol = max(1e-5, 0.04 * c)
        active_idx = np.where(profile >= c - act_tol)[0]
        if len(active_idx) > 35:
            top = np.argsort(-profile)
            active_idx = top[:35]

        active_lags = [int(p - (n - 1)) for p in active_idx]

        direction, t_val = find_direction(best_h, active_lags, dx, n)

        if direction is None or t_val >= -1e-10:
            G = np.array([lag_gradient(best_h, lag, dx) for lag in active_lags])
            direction = -np.mean(G, axis=0)
            direction -= np.mean(direction)
            if np.max(np.abs(direction)) < 1e-14:
                stall += 1
                if stall > 40:
                    break
                continue

        direction -= np.mean(direction)
        dmax = np.max(np.abs(direction))
        if dmax > 0:
            direction /= dmax

        best_alpha = 0.0
        best_alpha_c = c
        alphas = [trust, trust * 0.7, trust * 0.5, trust * 0.3,
                  trust * 0.15, trust * 0.07, trust * 0.02, trust * 0.005]

        for alpha in alphas:
            h_try = best_h + alpha * direction
            h_try = project_box_sum(h_try, target)
            c_try = exact_c5(h_try)
            if c_try < best_alpha_c - 1e-14:
                best_alpha = alpha
                best_alpha_c = c_try
                best_h = h_try.copy()

        if best_alpha > 0 and best_alpha_c < best_c - 1e-13:
            best_c = best_alpha_c
            if best_alpha >= trust * 0.5:
                trust = min(trust * 1.25, 0.35)
            else:
                trust *= 0.85
            stall = 0
        else:
            trust = max(trust * 0.55, 0.003)
            stall += 1
            if stall > 60:
                break

    return best_h, best_c


def binary_optimize(b, deadline, max_rounds=100, seed=777):

    c5 = exact_c5(b)
    rng = np.random.RandomState(seed)
    profile = exact_profile(b)

    for rnd in range(max_rounds):
        if time.time() > deadline - 5:
            break
        if c5 <= 0.38083:
            break
        improved = False
        ones = np.where(b > 0.5)[0]
        zeros = np.where(b < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        n_pairs = len(ones) * len(zeros)
        if n_pairs <= 6000:
            for i in ones:
                new_imp = False
                for j in zeros:
                    b[i], b[j] = 0.0, 1.0
                    nc = exact_c5(b)
                    if nc < c5 - 1e-14:
                        c5 = nc
                        improved = True
                        new_imp = True
                        profile = exact_profile(b)
                        ones = np.where(b > 0.5)[0]
                        zeros = np.where(b < 0.5)[0]
                        break
                    b[i], b[j] = 1.0, 0.0
                if new_imp:
                    break
        else:
            n_tries = 3000
            for _ in range(n_tries):
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                b[i], b[j] = 0.0, 1.0
                nc = exact_c5(b)
                if nc < c5 - 1e-14:
                    c5 = nc
                    improved = True
                    ones = np.where(b > 0.5)[0]
                    zeros = np.where(b < 0.5)[0]
                else:
                    b[i], b[j] = 1.0, 0.0
        if not improved:
            break
    return b, c5


def sa_escape(h, deadline, max_iter=4000, T_init=0.004, seed=42):
    n = len(h)
    rng = np.random.RandomState(seed)
    current = h.copy()
    c_cur = exact_c5(current)
    best = current.copy()
    c_best = c_cur
    T = T_init

    for it in range(max_iter):
        if time.time() > deadline - 5:
            break
        if c_best <= 0.38083:
            break
        a = rng.randint(n)
        b = rng.randint(n)
        if a == b:
            continue
        if current[a] < 1e-6 and current[b] > 1.0 - 1e-6:
            continue
        if current[a] > current[b]:
            src, dst = a, b
        else:
            src, dst = b, a
        max_d = min(current[src], 1.0 - current[dst])
        if max_d < 1e-5:
            continue
        delta = rng.uniform(0.002, min(max_d, 0.25))
        current[src] -= delta
        current[dst] += delta
        current[src] = max(0.0, current[src])
        current[dst] = min(1.0, current[dst])
        c_new = exact_c5(current)
        dc = c_new - c_cur
        if dc < 0 or (dc < 0.05 and rng.random() < np.exp(-dc / max(T, 1e-8))):
            c_cur = c_new
            if c_new < c_best - 1e-14:
                c_best = c_new
                best = current.copy()
        if (it + 1) % 80 == 0:
            T = max(T * 0.94, 0.0003)
        if (it + 1) % 800 == 0 and c_cur > c_best + 0.004:
            current = best.copy()
            c_cur = c_best
    return best, c_best


def make_starts(n, rng):
    target = 0.5 * n
    starts = []
    h = np.zeros(n)
    h[::2] = 1.0
    starts.append(project_box_sum(h, target))
    h = np.zeros(n)
    h[:int(target)] = 1.0
    starts.append(project_box_sum(h, target))
    h = np.zeros(n)
    idx = rng.choice(n, size=int(round(target)), replace=False)
    h[idx] = 1.0
    starts.append(project_box_sum(h, target))
    x = np.arange(n, dtype=np.float64)
    h = 0.5 + 0.4 * np.cos(2 * np.pi * x / n) + 0.1 * np.cos(4 * np.pi * x / n)
    starts.append(project_box_sum(h, target))
    h = np.full(n, 0.5) + rng.randn(n) * 0.12
    starts.append(project_box_sum(h, target))
    return starts


def downsample(h, new_n):
    n = len(h)
    if new_n == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(new_n) + 0.5) / new_n
    vals = np.interp(x_new, x_old, h)
    return project_box_sum(vals, 0.5 * new_n)


def coordinate_polish(h, deadline, max_rounds=200):
    """Try adjusting pairs of coordinates to reduce C5."""
    n = len(h)
    target = 0.5 * n
    best_h = h.copy()
    best_c = exact_c5(best_h)
    rng = np.random.RandomState(31337)

    for rnd in range(max_rounds):
        if time.time() > deadline - 5:
            break
        if best_c <= 0.38083:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        profile = exact_profile(best_h)
        c = float(np.max(profile))
        active = np.where(profile >= c - 0.03 * c)[0]
        active_lags = [int(p - (n - 1)) for p in active[:20]]

        # For each active lag, compute partial derivative w.r.t. h[i] and h[j]
        # Then find optimal (delta_i, delta_j) with delta_i = -delta_j
        best_delta = 0.0
        best_score = c
        for lag in active_lags:
            g = lag_gradient(best_h, lag, dx=2.0 / n)
            # We want g[i]*d_i + g[j]*d_j < 0, with d_j = -d_i
            # So (g[i] - g[j]) * d_i < 0
            coeff = g[i] - g[j]
            if abs(coeff) < 1e-12:
                continue
            # Try a few deltas
            for delta_sign in [1.0, -1.0]:
                d = delta_sign * np.sign(coeff) * 0.0
                for mag in [0.02, 0.05, 0.1, 0.15, 0.2]:
                    d = delta_sign * np.sign(coeff) * (-1.0) * mag
                    # Ensure box constraint
                    hi_val = best_h[i] + d
                    hj_val = best_h[j] - d
                    if hi_val < 0 or hi_val > 1 or hj_val < 0 or hj_val > 1:
                        # Clip
                        if d > 0:
                            d = min(d, 1.0 - best_h[i], best_h[j])
                        else:
                            d = max(d, best_h[i] - 1.0, -(1.0 - best_h[j]))
                        if abs(d) < 1e-8:
                            continue
                        hi_val = best_h[i] + d
                        hj_val = best_h[j] - d
                        if hi_val < 0 or hi_val > 1 or hj_val < 0 or hj_val > 1:
                            continue
                    test = best_h.copy()
                    test[i] = hi_val
                    test[j] = hj_val
                    test = project_box_sum(test, target)
                    sc = exact_c5(test)
                    if sc < best_score - 1e-14:
                        best_score = sc
                        best_delta = d
                        best_candidate = test

        if best_delta != 0.0 and best_score < best_c - 1e-14:
            best_c = best_score
            best_h = best_candidate.copy()

    return best_h, best_c
