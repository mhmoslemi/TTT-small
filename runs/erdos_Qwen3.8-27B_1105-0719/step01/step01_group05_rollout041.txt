def exact_objective(h):
    n = len(h)
    if n == 0:
        return float('inf'), np.array([])
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(profile)), profile

def project_box_sum(h, target=None):
    n = len(h)
    if target is None:
        target = 0.5 * n
    h = np.array(h, dtype=np.float64).copy()
    h = np.nan_to_num(h, nan=0.0, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
            count = np.sum(mask)
            if count == 0:
                break
            step = diff / count
            h[mask] = np.minimum(h[mask] + step, 1.0)
        else:
            mask = h > 0.0
            count = np.sum(mask)
            if count == 0:
                break
            step = diff / count
            h[mask] = np.maximum(h[mask] + step, 0.0)

            h[idx] = min(1.0, h[idx] + diff)
        else:
            idx = np.argmin(h > 0.0)
            h[idx] = max(0.0, h[idx] + diff)
    return h

def gradient_step(h, step_sizes, max_active=5):
    """Take a gradient step on the active (max) lags."""
    n = len(h)
    dx = 2.0 / n
    profile = profile_only(h)
    # Find top active lags
    top_k = max_active
    top_indices = np.argsort(-profile)[:top_k]
    
    best_h = h.copy()
    best_c = np.max(profile)
    
    for step in step_sizes:
        # Compute gradient of max (weighted average of top active gradients)
        grad = np.zeros(n)
        for r in top_indices:
            k = r - (n - 1)
            s = abs(k)
            if s == 0:
                g = dx * (1.0 - 2.0 * h)
            else:
                g = np.zeros(n)
                g[0:n-s] = dx * (1.0 - h[s:])
                g[s:n] += -dx * h[0:n-s]
            grad += g  # We want to DECREASE profile, so go against gradient
        
        # Project gradient to maintain sum constraint
        grad -= np.mean(grad)  # zero mean to preserve sum
        
        # Try step

    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time


def exact_objective(h):
    n = len(h)
    if n == 0:
        return float('inf'), np.array([])
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(profile)), profile


def project_box_sum(h, target=None):
    n = len(h)
    if target is None:
        target = 0.5 * n
    h = np.array(h, dtype=np.float64).copy()
    h = np.nan_to_num(h, nan=0.0, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
            count = np.sum(mask)
            if count == 0:
                break
            step = diff / count
            h[mask] = np.minimum(h[mask] + step, 1.0)
        else:
            mask = h > 0.0
            count = np.sum(mask)
            if count == 0:
                break
            step = diff / count
            h[mask] = np.maximum(h[mask] + step, 0.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-16 and n > 0:
        if diff > 0:
            idx = np.argmax(h < 1.0)
            h[idx] = min(1.0, h[idx] + diff)
        else:
            idx = np.argmin(h > 0.0)
            h[idx] = max(0.0, h[idx] + diff)
    return h

    n = len(h_init)
    dx = 2.0 / n
    target = 0.5 * n
    n_lags = 2 * n - 1

    profile = profile_only(h_init)
    c0 = float(np.max(profile))

    x0 = np.concatenate([h_init, [c0 + 1e-9]])
    bounds = [(0.0, 1.0)] * (n + 1)

        return x[n]

    def obj_jac(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g

    def eq_func(x):
        return np.sum(x[:n]) - target

    def eq_jac(x):
        j = np.zeros(n + 1)
        j[:n] = 1.0
        return j

    def ineq_func(x):
        h = x[:n]
        c = x[n]
        prof = np.correlate(h, 1.0 - h, mode='full') * dx
        return c - prof

    def ineq_jac(x):
        h = x[:n]

            else:
                J[r, 0:n-s] += -dx * (1.0 - h[s:n])
                J[r, s:n] += dx * h[0:n-s]
        return J

    constraints = [
        {'type': 'eq', 'fun': eq_func, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_func, 'jac': ineq_jac}
    ]

    try:

        grad -= np.mean(grad)
        h_new = project_box_sum(h - step * grad, 0.5 * n)
        c_new = exact_objective(h_new)[0]
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            profile = profile_only(h_new)
            top_idx = np.argsort(-profile)[:max_active]

    return best_h, best_c


def binary_threshold(h):
    n = len(h)
    if n % 2 != 0:
        return None
    target = n // 2
    idx = np.argsort(-h)
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b


def binary_swap_improve(b, max_rounds=50, rng=None):
    n = len(b)
    if n % 2 != 0 or b is None:
        return b, exact_objective(b)[0]
    if rng is None:
        rng = np.random.RandomState(0)
    c5 = exact_objective(b)[0]
    for rnd in range(max_rounds):
        improved = False
        ones = list(np.where(b > 0.5)[0])
        zeros = list(np.where(b < 0.5)[0])
        n_pairs = len(ones) * len(zeros)
        if n_pairs <= 1500:
            for i in ones:
                broke = False
                for j in zeros:
                    b[i], b[j] = 0.0, 1.0
                    nc = exact_objective(b)[0]
                    if nc < c5 - 1e-13:
                        c5 = nc
                        improved = True
                        broke = True
                        ones = list(np.where(b > 0.5)[0])
                        zeros = list(np.where(b < 0.5)[0])
                        break
                    else:
                        b[i], b[j] = 1.0, 0.0
                if broke:
                    break
        else:
            n_tries = 300
            for _ in range(n_tries):
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                b[i], b[j] = 0.0, 1.0
                nc = exact_objective(b)[0]
                if nc < c5 - 1e-13:
                    c5 = nc
                    improved = True
                    ones = list(np.where(b > 0.5)[0])
                    zeros = list(np.where(b < 0.5)[0])
                else:
                    b[i], b[j] = 1.0, 0.0
        if not improved:
            break
    return b, c5


def make_start(n, rng, mode=0):
    target = 0.5 * n
    if mode == 0:
        h = np.zeros(n)
        h[::2] = 1.0
    elif mode == 1:
        h = np.zeros(n)
        h[:int(target)] = 1.0
    elif mode == 2:
        h = np.zeros(n)
        idx = rng.choice(n, size=int(target), replace=False)
        h[idx] = 1.0
    elif mode == 3:
        x = np.arange(n, dtype=np.float64)
        freq = float(2 + rng.randint(3))
        phase = rng.uniform(0, 2 * np.pi)
        h = 0.5 + 0.4 * np.cos(2.0 * np.pi * freq * x / n + phase)
    elif mode == 4:
        h = np.full(n, 0.5)
        h += rng.randn(n) * 0.2
    elif mode == 5:
        h = np.zeros(n)
        q = n // 3
        h[:q] = 1.0
        idx = rng.choice(n, size=int(target) - q, replace=False)
        h[idx] = 1.0
        h = np.clip(h, 0, 1)
    else:
        h = np.full(n, 0.5)
    return project_box_sum(h, target)


def transfer(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_src = (np.arange(n_src) + 0.5) / n_src * 2.0
    x_dst = (np.arange(n_dst) + 0.5) / n_dst * 2.0
    vals = np.interp(x_dst, x_src, h_src)
    return project_box_sum(vals, 0.5 * n_dst)


def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    deadline = start_time + budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c = float('inf')

    try:
        init = np.asarray(initial_h_values, dtype=np.float64)
        if init.size > 0 and np.all(np.isfinite(init)):
            n0 = len(init)
            init = project_box_sum(init, 0.5 * n0)
            c0 = exact_objective(init)[0]
            best_h = init.copy()
            best_c = c0
    except (NameError, Exception):
        pass

    if best_h is None:
        n0 = 70
        best_h = make_start(n0, rng, mode=0)
        best_c = exact_objective(best_h)[0]

    n0 = len(best_h)
    results = []

    # Phase 1: Direct SLSQP on parent
    if time.time() < deadline - 120:
        t_limit = min(deadline - 60, start_time + 180)
        h_opt, c_opt = slsqp_direct(best_h, t_limit, max_iter=300, ftol=1e-12)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
        # Reduced version for further refinement
        if time.time() < deadline - 90:
            h_opt2, c_opt2 = slsqp_reduced(best_h, min(deadline - 30, time.time() + 120),
                                           max_iter=200, K=15, n_inner=8)
            if c_opt2 < best_c:
                best_c = c_opt2
                best_h = h_opt2

    results.append((best_h.copy(), best_c, n0))

    # Phase 2: Gradient refinement
    if time.time() < deadline - 80:
        steps = [0.5, 0.3, 0.15, 0.08, 0.04, 0.02]
        for _ in range(4):
            h_opt, c_opt = gradient_step(best_h, steps, max_active=5)
            if c_opt >= best_c - 1e-15:
                break
            best_c = c_opt
            best_h = h_opt

    results.append((best_h.copy(), best_c, n0))

    # Phase 3: Binary threshold + swap
    if n0 % 2 == 0 and time.time() < deadline - 60:
        b = binary_threshold(best_h)
        if b is not None:
            b, c_b = binary_swap_improve(b, max_rounds=50, rng=rng)
            if c_b < best_c:
                best_c = c_b
                best_h = b.copy()
                results.append((best_h.copy(), best_c, n0))
            # SLSQP from binary
            if time.time() < deadline - 40:
                h_opt, c_opt = slsqp_direct(b, min(deadline - 20, time.time() + 90),
                                            max_iter=200, ftol=1e-11)
                if c_opt < best_c:
                    best_c = c_opt
                    best_h = h_opt
                    results.append((best_h.copy(), best_c, n0))

    # Phase 4: Size exploration
    n_best = len(best_h)
    sizes = []
    for s in list(range(18, 130, 2)):
        if s != n_best:
            sizes.append(s)
    sizes.sort(key=lambda x: abs(x - n_best))

    # Ensure we include some key sizes
    for key_s in [20, 30, 40, 44, 48, 52, 56, 60, 64, 68, 72, 76, 80, 84, 88, 92, 96, 100, 104, 108, 112, 116, 120]:
        if key_s not in sizes and key_s != n_best:
            sizes.insert(len(sizes) // 2, key_s)

    tried_count = 0
    for n_try in sizes:
        if time.time() > deadline - 25:
            break
        if tried_count >= 12:
            break

        h_trans = transfer(best_h, n_try)
        c_trans = exact_objective(h_trans)[0]

        h_fresh = make_start(n_try, rng, mode=rng.randint(6))
        c_fresh = exact_objective(h_fresh)[0]

        if c_fresh < c_trans:
            h_start = h_fresh
        else:
            h_start = h_trans

        remaining = deadline - time.time() - 10
        if remaining < 15:
            break

        t_lim = time.time() + min(remaining * 0.6, 50)
        h_opt, c_opt = slsqp_direct(h_start, t_lim, max_iter=150, ftol=1e-10)
        tried_count += 1

        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
            n_best = len(best_h)
            results.append((best_h.copy(), best_c, n_best))

        if n_try % 2 == 0 and c_opt < best_c + 0.01 and time.time() < deadline - 15:
            b = binary_threshold(h_opt)
            if b is not None:
                b, c_b = binary_swap_improve(b, max_rounds=30, rng=rng)
                if c_b < best_c:
                    best_c = c_b
                    best_h = b.copy()
                    n_best = len(best_h)
                    results.append((best_h.copy(), best_c, n_best))

    # Phase 5: Final tight refinement
    if time.time() < deadline - 15:
        t_lim = min(deadline - 5, time.time() + 60)
        h_opt, c_opt = slsqp_direct(best_h, t_lim, max_iter=250, ftol=1e-12)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    if time.time() < deadline - 8:
        h_opt, c_opt = gradient_step(best_h, [0.3, 0.1, 0.05, 0.02, 0.01], max_active=5)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    if len(best_h) % 2 == 0 and time.time() < deadline - 5:
        b = binary_thread = binary_threshold(best_h)
        if b is not None:
            b, c_b = binary_swap_improve(b, max_rounds=25, rng=rng)
            if c_b < best_c:
                best_c = c_b
                best_h = b

    # Choose best result (prefer lower C, then smaller n)
    if results:
        cand_min_c = min(results, key=lambda x: (x[1] + 1e-14, len(x[0])))
        if cand_min_c[1] <= best_c + 1e-14:
            best_h = cand_min_c[0]
            best_c = cand_min_c[1]

    # Final validation
    best_h = project_box_sum(best_h, 0.5 * len(best_h))
    final_c = exact_objective(best_h)[0]

    return (best_h, final_c, len(best_h))
