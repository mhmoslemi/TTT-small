def grad_corr_k(h, k, n):
    offset = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -offset)
    i_hi = min(n - 1, n - 1 - offset)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx] += (1.0 - h[idx + offset])
    grad[idx + offset] += (-h[idx])
    return grad

def smooth_max_val(h, dx, beta):
    corrs = compute_corrs(h)
    return logsumexp(beta * corrs) / beta * dx

def smooth_max_grad(h, dx, beta):
    n = len(h)
    corrs = compute_corrs(h)
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones_like(corrs) / len(corrs)
    else:
        w /= w_sum
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        gk = grad_corr_k(h, k, n)
        grad += w[k] * gk
    return grad * dx

def slsqp_run(h0, dx, target_sum, beta, maxiter):
    n = len(h0)
    def obj(x):
        return smooth_max_val(x, dx, beta)
    def grad(x):
        return smooth_max_grad(x, dx, beta)
    def cfun(x):
        return x.sum() - target_sum
    def cjac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0, jac=grad, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        h_out = project_feasible(res.x, target_sum)
    except Exception:
        h_out = project_feasible(h0, target_sum)
    return h_out, c5_of_h(h_out, dx)

def subgrad_refine(h0, dx, target_sum, max_iter, beta):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    alpha = 1.0
    no_improve = 0
    
    for it in range(max_iter):
        h = project_feasible(h, target_sum)
        corrs = compute_corrs(h)
        c5_cur = corrs.max() * dx
        
        if c5_cur < best_c5 - 1e-15:
            best_c5 = c5_cur
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 2000:
            break
        
        # Compute gradient using smooth max
        shifted = corrs - corrs.max()
        w = np.exp(beta * shifted)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones_like(corrs) / len(corrs)
        else:
            w /= w_sum
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] < 1e-14:
                continue
            grad += w[k] * grad_corr_k(h, k, n)
        grad *= dx
        
        h = h - alpha * grad
        alpha *= 0.999
        if alpha < 1e-6:
            break
    
    return best_h, best_c5

def make_sinusoidal(n, target_sum, freq, phase):
    x = np.arange(n)
    h = 0.5 + 0.5 * np.sin(2 * np.pi * freq * x / n + phase)
    return project_feasible(h, target_sum)

def make_block(n, target_sum, block_size):
    h = np.zeros(n)
    for i in range(0, n, 2 * block_size):
        end = min(i + block_size, n)
        h[i:end] = 1.0
    return project_feasible(h, target_sum)

def make_thue_morse(n, target_sum):
    # Thue-Morse sequence
    tm = np.zeros(n, dtype=int)
    for i in range(1, n):
        tm[i] = tm[i >> 1] ^ (i & 1)
    h = 1.0 - tm  # Invert so that 1s are where tm=0
    return project_feasible(h, target_sum)

def project_feasible(x, target_sum):

    n = len(x)
    lo, hi = -1.0, 1.0

        else:

def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def grad_corr_k(h, k, n):
    offset = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -offset)
    i_hi = min(n - 1, n - 1 - offset)

        return grad

    grad[idx] += (1.0 - h[idx + offset])
    grad[idx + offset] += (-h[idx])
    return grad
