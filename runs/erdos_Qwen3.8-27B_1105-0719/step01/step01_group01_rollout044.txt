import numpy as np
from scipy.optimize import minimize
import time
import math

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    """Project y to [0,1]^n with sum = n/2"""
    target = n / 2.0
    y = np.array(y, dtype=float)
    y[~np.isfinite(y)] = 0.5
    lo = np.min(y) - 2.0
    hi = np.max(y) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    h = np.clip(y - t, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-10:
        eligible = np.where((h > 1e-8) & (h < 1.0 - 1e-8))[0]
        if len(eligible) > 0:
            h[eligible] += residual / len(eligible)
            h = np.clip(h, 0.0, 1.0)
    # Final fix
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        # Find index with most room
        if diff > 0:
            idx = np.argmax(1.0 - h)
            h[idx] = min(1.0, h[idx] + diff)
        else:
            idx = np.argmin(h)
            h[idx] = max(0.0, h[idx] + diff)
    return h

def pair_transfer_line_search(h, i, j, n_candidates=8):
    """Try transferring from j to i (h[i] increases, h[j] decreases).
    Returns (best_h, best_c5, improved)."""
    cap = min(1.0 - h[i], h[j])
    if cap <= 1e-12:
        return h, compute_c5(h), False
    current_c5 = compute_c5(h)
    # Try several tau values
    taus = np.linspace(0.05, 0.95, n_candidates) * cap
    best_tau = 0.0
    best_c5 = current_c5
    for tau in taus:
        h_new = h.copy()
        h_new[i] += tau
        h_new[j] -= tau
        if h_new[i] < -1e-10 or h_new[i] > 1.0 + 1e-10 or h_new[j] < -1e-10 or h_new[j] > 1.0 + 1e-10:
            continue
        c5 = compute_c5(h_new)
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_tau = tau
    if best_tau > 0:
        h_new = h.copy()
        h_new[i] += best_tau
        h_new[j] -= best_tau
        return h_new, best_c5, True
    return h, current_c5, False

def exact_local_search(h, time_limit, rng, n_pairs_per_iter=200):
    """Greedy pair transfer local search."""
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    t_end = time.time() + time_limit
    stall_count = 0
    max_stall = 50
    
    while time.time() < t_end:
        improved = False
        for _ in range(n_pairs_per_iter):
            if time.time() > t_end:
                break
            i, j = rng.choice(n, size=2, replace=False)
            h_new, c5_new, did_improve = pair_transfer_line_search(best_h, i, j, n_candidates=6)
            if did_improve:
                best_h = h_new
                best_c5 = c5_new
                improved = True
                stall_count = 0
                break  # Restart pair sampling after improvement
        if not improved:
            stall_count += 1
            if stall_count >= max_stall:
                break
    return best_h, best_c5

    """Simulated annealing with pair transfer moves."""
    n = len(h)
    current_h = h.copy()
    current_c5 = compute_c5(current_h)
