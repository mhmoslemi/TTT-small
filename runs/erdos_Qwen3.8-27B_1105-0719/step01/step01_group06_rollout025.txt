import numpy as np

def exact_score(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))

def get_profile(h):
    dx = 2.0 / len(h)
    return np.correlate(h, 1.0 - h, mode="full") * dx

def get_grad_h(h):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    G = np.zeros((M, n))
    for k in range(M):
        delta = k - (n - 1)
        if delta == 0:
            G[k, :] = dx * (1.0 - 2.0 * h)
        elif delta > 0:
            if delta < n:
                G[k, delta:n] += dx * (1.0 - h[0:n - delta])
                G[k, 0:n - delta] -= dx * h[delta:n]
        else:
            d = -delta
            if d < n:
                G[k, 0:n-d] += dx * (1.0 - h[d:n])
                G[k, d:n] -= dx * h[0:n-d]
    return G

def repair_feasible(h, target=None):
    h = np.array(h, dtype=np.float64).copy()

    n = len(h)

        target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        order = np.argsort(-room)
        remaining = diff
        for idx in order:
            if remaining <= 1e-15:

            add = min(remaining, room[idx])
            h[idx] += add
            remaining -= add

            if remaining <= 1e-15:

    n = len(h)
    lo = -np.inf
    hi = np.inf
    pos = d > 1e-14
    neg = d < -1e-14
    if np.any(pos):

        hi = min(hi, float(np.max((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        lo = max(lo, float(np.max((1.0 - h[neg]) / d[neg])))
        hi = min(hi, float(np.max(-h[neg] / d[neg])))

    span = hi - lo

    hi -= marg

    def obj(x):
        h = x[:n]
        t = x[n]
        return t + 1e-7 * np.sum(h * (1.0 - h))
    
    def obj_jac(x):
        h = x[:n]
        g = np.zeros(n + 1)
        g[:n] = 1e-7 * (1.0 - 2.0 * np.clip(h, 0, 1))
        g[n] = 1.0
        return g
    
    def cons_fun(x):
        h = np.clip(x[:n], 0, 1)
        t = x[n]
        prof = get_profile(h)
        ineq = t - prof[active_lags]
        return np.concatenate([[h.sum() - n / 2.0], ineq])
    
    def cons_jac(x):
        h = np.clip(x[:n], 0, 1)
        G = get_grad_h(h)
        Ja = G[active_lags, :]
        J = np.zeros((1 + len(active_lags), n + 1))
        J[0, :n] = 1.0
        J[1:, :n] = -Ja
        J[1:, n] = 1.0
        return J
    
    bounds = [(0.0, 1.0)] * n + [(0.0, t_start + 2.0)]
    x0 = np.concatenate([h_start, [t_start]])
    
    try:
        result = minimize(obj, x0, jac=obj_jac, method='SLSQP',
                          bounds=bounds,
                          constraints=[{'type': 'ineq', 'fun': cons_fun, 'jac': cons_jac}],
                          options={'maxiter': maxiter, 'ftol': 1e-13})
        if result.x is not None and np.all(np.isfinite(result.x)):
            return repair_feasible(result.x[:n])
    except Exception:
        pass
    return None

def binary_start(n, rng):
    ones = rng.choice(n, size=n // 2, replace=False)
    h = np.zeros(n)
    h[ones] = 1.0
    return h

def smooth_v2(h, alpha=0.3):
    n = len(h)
    kernel = np.array([0.25, 0.5, 0.25])
    from scipy.ndimage import uniform_filter1d
    hs = uniform_filter1d(h, size=3, mode='wrap')
    h_blend = (1 - alpha) * h + alpha * hs
    return repair_feasible(h_blend)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t_start = time.perf_counter()
    
    def tleft():
        return budget_s - (time.perf_counter() - t_start)
    
    try:
        parent = np.array(initial_h_values, dtype=np.float64).copy()
    except Exception:
        parent = np.full(140, 0.5)
    
    parent = repair_feasible(parent)
    n = len(parent)
    best_h = parent.copy()
    best_score = exact_score(parent)
    
    # Phase 1: Quick SLSQP
    if tleft() > 50:
        h_sol = solve_slsqp(best_h, best_score + 1e-5, n_constraints=0, maxiter=100)
        if h_sol is not None:
            s = exact_score(h_sol)
            if s < best_score - 1e-13:
                best_h = h_sol
                best_score = s
    
    # Phase 2: LP cutting plane (main workhorse)
    if tleft() > 80:
        tl = min(120, tleft() - 20)
        h2, s2 = lp_cutting_plane(best_h, time_limit=tl, t_start_ref=t_start + (budget_s - tleft()))
        if s2 < best_score - 1e-13:
            best_h = h2
            best_score = s2
    
    # Phase 3: Subgradient refinement
    if tleft() > 60:
        tl = min(80, tleft() - 10)
        h3, s3 = local_optimize_subgrad(best_h, time_limit=tl, t_start_ref=t_start + (budget_s - tleft()))
        if s3 < best_score - 1e-13:
            best_h = h3
            best_score = s3
    
    # Phase 4: More LP cutting plane
    if tleft() > 60:
        tl = min(80, tleft() - 10)
        h4, s4 = lp_cutting_plane(best_h, time_limit=tl, t_start_ref=t_start + (budget_s - tleft()))
        if s4 < best_score - 1e-13:
            best_h = h4
            best_score = s4
    
    # Phase 5: SLSQP with top-K constraints
    if tleft() > 40:
        for nc in [30, 60, 0]:
            if tleft() < 25:
                break
            h_sol = solve_slsqp(best_h, best_score + 1e-6, n_constraints=nc, maxiter=120)
            if h_sol is not None:
                s = exact_score(h_sol)
                if s < best_score - 1e-13:
                    best_h = h_sol
                    best_score = s
    
    # Phase 6: Subgradient again
    if tleft() > 30:
        tl = min(40, tleft() - 5)
        h6, s6 = local_optimize_subgrad(best_h, time_limit=tl, t_start_ref=t_start + (budget_s - tleft()))
        if s6 < best_score - 1e-13:
            best_h = h6
            best_score = s6
    
    # Phase 7: Try different dimensions
    if tleft() > 50:
        pn = len(best_h)
        dims = sorted(set([d for d in [pn-8, pn-4, pn+4, pn+8, pn+16, pn*2] if 50 <= d <= 300 and d != pn]))
        for new_n in dims:
            if tleft() < 25:
                break
            h_interp = interpolate_to_dim(best_h, new_n)
            tl = min(15, tleft() - 8)
            h_new, s_new = lp_cutting_plane(h_interp, time_limit=tl, t_start_ref=t_start + (budget_s - tleft()))
            if s_new < best_score - 1e-13:
                best_h = h_new
                best_score = s_new
    
    # Phase 8: Random perturbations + quick optimize
    if tleft() > 30:
        for p in range(8):
            if tleft() < 15:
                break
            eps = 0.002 + 0.002 * p
            z = rng.standard_normal(len(best_h))
            z -= z.mean()
            h_pert = repair_feasible(best_h + eps * z)
            tl = min(8, tleft() - 5)
            h_r, s_r = local_optimize_subgrad(h_pert, time_limit=tl, t_start_ref=t_start + (budget_s - tleft()))
            if s_r < best_score - 1e-13:
                best_h = h_r
                best_score = s_r
    
    # Phase 9: Final LP + SLSQP polish
    if tleft() > 20:
        tl = min(15, tleft() - 5)
        h9, s9 = lp_cutting_plane(best_h, time_limit=tl, t_start_ref=t_start + (budget_s - tleft()))
        if s9 < best_score - 1e-13:
            best_h = h9
            best_score = s9
    
    if tleft() > 15:
        h_sol = solve_slsqp(best_h, best_score + 1e-6, n_constraints=40, maxiter=80)
        if h_sol is not None:
            s = exact_score(h_sol)
            if s < best_score - 1e-13:
                best_h = h_sol
                best_score = s
    
    # Phase 10: Final subgradient
    if tleft() > 8:
        tl = min(6, tleft() - 2)
        h10, s10 = local_optimize_subgrad(best_h, time_limit=tl, t_start_ref=t_start + (budget_s - tleft()))
        if s10 < best_score - 1e-13:
            best_h = h10
            best_score = s10
    
    best_h = repair_feasible(best_h)
    final_score = exact_score(best_h)
    return (best_h, final_score, len(best_h))
