def discrete_greedy_search(h, dx, max_iter=200, time_budget=60):
    n = len(h)
    h = h.copy()
    corrs = compute_corrs(h)
    c5 = corrs.max() * dx
    best_c5 = c5
    best_h = h.copy()
    t_start = time.time()
    
    for it in range(max_iter):
        if time.time() - t_start > time_budget:
            break
        
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Precompute delta for each one position
        n_corr = 2 * n - 1
        d = np.arange(-(n-1), n)  # shift values
        
        # For each a in ones: delta_a_general
        # delta_a[d] = -(1-h[a+d]) * I(0<=a+d<n) + h[a-d] * I(0<=a-d<n)
        # Vectorize: for each a, compute the two shifted arrays
        
        # Actually, let me compute it differently.
        # For a given a, the valid d values for term 1 are: d in [max(-(n-1), -a), min(n-1, n-1-a)]
        # The values are: -(1-h[a+d]) for those d
        
        # Let me just loop over ones (at most 63 iterations) and compute delta_a
        deltas_a = np.zeros((len(ones), n_corr))
        for i, a in enumerate(ones):
            # Term 1: -(1-h[a+d]) for valid a+d
            idx1 = a + d
            valid1 = (idx1 >= 0) & (idx1 < n)
            deltas_a[i, valid1] -= (1 - h[idx1[valid1]])
            # Term 2: h[a-d] for valid a-d
            idx2 = a - d
            valid2 = (idx2 >= 0) & (idx2 < n)
            deltas_a[i, valid2] += h[idx2[valid2]]
        
        deltas_b = np.zeros((len(zeros), n_corr))
        for i, b in enumerate(zeros):
            # Term 3: (1-h[b+d]) for valid b+d
            idx3 = b + d
            valid3 = (idx3 >= 0) & (idx3 < n)
            deltas_b[i, valid3] += (1 - h[idx3[valid3]])
            # Term 4: -h[b-d] for valid b-d
            idx4 = b - d
            valid4 = (idx4 >= 0) & (idx4 < n)
            deltas_b[i, valid4] -= h[idx4[valid4]]
        
        # Now for each pair (a_i, b_j), compute total delta and new max
        # total_delta[i,j] = deltas_a[i] + deltas_b[j]
        # Plus corrections
        
        # Vectorized: new_corrs = corrs[None, None, :] + deltas_a[:, None, :] + deltas_b[None, :, :]
        new_corrs = corrs[None, None, :] + deltas_a[:, None, :] + deltas_b[None, :, :]
        # new_corrs has shape (len(ones), len(zeros), n_corr)
        
        # Add corrections:
        # For pair (a_i, b_j):
        #   +1 at position (b_j - a_i) + (n-1) if valid
        #   +1 at position (a_i - b_j) + (n-1) if valid
        ai = ones[:, None]  # (len(ones), 1)
        bj = zeros[None, :]  # (1, len(zeros))
        
        diff1 = bj - ai  # (len(ones), len(zeros)) = b_j - a_i
        pos1 = diff1 + (n-1)  # position in corr vector
        valid_pos1 = (pos1 >= 0) & (pos1 < n_corr)
        
        # Add +1 at those positions
        # new_corrs[i, j, pos1[i,j]] += 1
        rows = np.arange(len(ones))[:, None]
        cols = np.arange(len(zeros))[None, :]
        rows = np.broadcast_to(rows, (len(ones), len(zeros)))
        cols = np.broadcast_to(cols, (len(ones), len(zeros)))
        new_corrs[rows[valid_pos1], cols[valid_pos1], pos1[valid_pos1]] += 1.0
        
        diff2 = ai - bj  # a_i - b_j
        pos2 = diff2 + (n-1)
        valid_pos2 = (pos2 >= 0) & (pos2 < n_corr)
        new_corrs[rows[valid_pos2], cols[valid_pos2], pos2[valid_pos2]] += 1.0
        
        # Find the best swap
        new_maxes = new_corrs.max(axis=2) * dx  # (len(ones), len(zeros))
        min_val = new_maxes.min()
        min_idx = np.unravel_index(new_maxes.argmin(), new_maxes.shape)
        
        if min_val >= c5 - 1e-12:
            break  # No improvement
        
        # Apply the swap
        a = ones[min_idx[0]]
        b = zeros[min_idx[1]]
        h[a] = 0.0
        h[b] = 1.0
        corrs = compute_corrs(h)
        c5 = corrs.max() * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5
