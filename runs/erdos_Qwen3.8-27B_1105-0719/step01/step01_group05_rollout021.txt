import numpy as np
from scipy.optimize import linprog
import time

def exact_objective(h):
    n = len(h)
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(profile)), profile

def project_box_sum(h, target=None):
    n = len(h)
    if target is None:
        target = 0.5 * n
    h = np.array(h, dtype=np.float64).copy()
    h = np.nan_to_num(h, nan=0.0, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            count = np.sum(mask)
            if count == 0:
                break
            step = diff / count
            h[mask] = np.minimum(h[mask] + step, 1.0)
        else:
            mask = h > 0.0
            count = np.sum(mask)
            if count == 0:
                break
            step = diff / count
            h[mask] = np.maximum(h[mask] + step, 0.0)
    # Final adjustment
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        # Find a coordinate with room
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if np.any(mask):
            idx = np.where(mask)[0]
            room = np.minimum(1.0 - h[idx], h[idx]) if diff > 0 else h[idx]
            # Distribute
            h[idx[0]] = np.clip(h[idx[0]] + diff, 0.0, 1.0)
            # If still off, try others
            remaining = target - np.sum(h)
            if abs(remaining) > 1e-13 and len(idx) > 1:
                h[idx[1]] = np.clip(h[idx[1]] + remaining, 0.0, 1.0)
    return h

def lag_gradient(h, lag, dx):
    n = len(h)
    grad = np.zeros(n)
    if lag >= 0:
        s = n - lag
        if s > 0:
            grad[:s] -= h[lag:]
            grad[lag:] += 1.0 - h[:s]
    else:
        a = -lag
        s = n - a
        if s > 0:
            grad[:s] += 1.0 - h[a:]
            grad[a:] -= h[:s]
    return grad * dx

def active_set_lp_direction(h, profile, c_max, active_tol, rho, max_active=30):
    """Find descent direction using active-set LP."""
    n = len(h)
    dx = 2.0 / n
    n_prof = 2 * n - 1
    
    # Find active lags
    active_idx = np.where(profile >= c_max - active_tol)[0]
    if len(active_idx) == 0:
        active_idx = np.array([np.argmax(profile)])
    
    # Take top max_active by profile value
    if len(active_idx) > max_active:
        active_vals = profile[active_idx]
        top_k = np.argsort(-active_vals)[:max_active]
        active_idx = active_idx[top_k]
    
    # Build constraint matrix
    # Variables: d[0..n-1], t
    # Minimize -t (maximize t)
    # Constraints: G_r^T d + t <= 0 for each active r
    m = len(active_idx)
    A_ub = np.zeros((m + n, n + 1))
    b_ub = np.zeros(m + n)
    
    for k, p in enumerate(active_idx):
        lag = p - (n - 1)
        g = lag_gradient(h, lag, dx)
        A_ub[k, :n] = g
        A_ub[k, n] = 1.0
        b_ub[k] = 0.0
    
    # Trust region: -rho <= d[i] <= rho
    for i in range(n):
        A_ub[m + i, i] = 1.0
        b_ub[m + i] = rho
        A_ub[m + i + n, i] = -1.0
        b_ub[m + i + n] = rho
    
    # Boundary constraints
    # If h[i] <= 1e-8: d[i] >= 0 → -d[i] <= 0
    # If h[i] >= 1-1e-8: d[i] <= 0 → d[i] <= 0
    # (These are already implied by trust region + line search, but add for LP)
    # Actually, with trust region, we can skip these in the LP and handle in line search.
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Objective: minimize -t
    c_obj = np.zeros(n + 1)
    c_obj[n] = -1.0
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=[(None, None)] * n + [(None, None)],
                  method='highs')
    
    if res.status != 0 or res.x is None:
        return None, 0.0
    
    t_val = float(res.x[n])
    d = res.x[:n].copy()
    
    # Ensure sum(d) = 0 (numerical)
    d -= np.mean(d)
    
    if t_val <= 1e-12:
        return None, t_val
    
    return d, t_val

def line_search(h, d, c_current, rho):
    """Exact line search along direction d."""
    n = len(h)
    target = 0.5 * n
    
    # Compute max safe alpha
    alpha_max = 1.0
    neg_mask = d < -1e-15
    pos_mask = d > 1e-15
    if np.any(neg_mask):
        alpha_max = min(alpha_max, np.min(h[neg_mask] / (-d[neg_mask])))
    if np.any(pos_mask):
        alpha_max = min(alpha_max, np.min((1.0 - h[pos_mask]) / d[pos_mask]))
    alpha_max = max(alpha_max, 1e-8)
    alpha_max = min(alpha_max, rho / max(np.max(np.abs(d)), 1e-15))
    
    best_h = h.copy()
    best_c = c_current
    
    # Try decreasing alphas
    alphas = [alpha_max * f for f in [1.0, 0.7, 0.5, 0.35, 0.25, 0.15, 0.08, 0.04]]
    for alpha in alphas:
        if alpha < 1e-10:
            continue
        h_try = h + alpha * d
        h_try = project_box_sum(h_try, target)
        c_try, _ = exact_objective(h_try)
        if c_try < best_c - 1e-13:
            best_c = c_try
            best_h = h_try.copy()
    
    return best_h, best_c

def active_set_optimize(h, max_iters=500, active_tol_init=0.005, rho_init=0.05, deadline=None):
    """Active-set LP optimization."""
    n = len(h)
    target = 0.5 * n
    h = project_box_sum(h, target).copy()
    c, profile = exact_objective(h)
    
    rho = rho_init
    active_tol = active_tol_init
    stall_count = 0
    
    for it in range(max_iters):
        if deadline is not None and time.time() > deadline:
            break
        
        # Determine active set tolerance (adaptive)
        active_tol = max(1e-5, 0.003 * c)
        
        # Get descent direction
        d, t_val = active_set_lp_direction(h, profile, c, active_tol, rho)
        
        if d is None:
            # No descent direction found
            stall_count += 1
            # Shrink trust region
            rho *= 0.7
            if rho < 1e-6:
                break
            # Also try with larger active set
            d, t_val = active_set_lp_direction(h, profile, c, active_tol * 3, rho)
            if d is None:
                if stall_count > 20:
                    break
                continue
        
        # Line search
        h_new, c_new = line_search(h, d, c, rho)
        
        if c_new < c - 1e-13:
            h = h_new
            c = c_new
            profile, _ = exact_objective(h)
            # Actually need to recompute
            c, profile = exact_objective(h)
            stall_count = 0
            rho = min(rho * 1.15, 0.3)
        else:
            rho *= 0.7
            stall_count += 1
            if rho < 1e-6:
                break
    
    return h, c

def binary_threshold(h):
    n = len(h)
    if n % 2 != 0:
        return None
    target = n // 2
    idx = np.argsort(-h)
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b

def binary_swap_improve(b, max_rounds=100, rng=None):
    n = len(b)
    if n % 2 != 0 or b is None:
        return b, exact_objective(b)[0]
    if rng is None:
        rng = np.random.RandomState(0)
    c5, _ = exact_objective(b)
    for rnd in range(max_rounds):
        improved = False
        ones = np.where(b > 0.5)[0]
        zeros = np.where(b < 0.5)[0]
        n_ones = len(ones)
        n_zeros = len(zeros)
        
        if n_ones * n_zeros <= 5000:
            for i in ones:
                for j in zeros:
                    b[i] = 0.0
                    b[j] = 1.0
                    nc, _ = exact_objective(b)
                    if nc < c5 - 1e-13:
                        c5 = nc
                        improved = True
                        ones = np.where(b > 0.5)[0]
                        zeros = np.where(b < 0.5)[0]
                        break
                    b[i] = 1.0
                    b[j] = 0.0
                if improved:
                    break
        else:
            n_tries = 800
            for _ in range(n_tries):
                i = ones[rng.randint(n_ones)]
                j = zeros[rng.randint(n_zeros)]
                b[i], b[j] = 0.0, 1.0
                nc, _ = exact_objective(b)
                if nc < c5 - 1e-13:
                    c5 = nc
                    improved = True
                    ones = np.where(b > 0.5)[0]
                    zeros = np.where(b < 0.5)[0]
                else:
                    b[i], b[j] = 1.0, 0.0
        if not improved:
            break
    return b, c5

def simulated_annealing(h, n_iter=5000, T_init=0.003, T_min=1e-6, deadline=None):
    """Simulated annealing with pair moves."""
    n = len(h)
    target = 0.5 * n
    h = h.copy()
    c, _ = exact_objective(h)
    best_h = h.copy()
    best_c = c
    T = T_init
    rng = np.random.RandomState(12345)
    
    for it in range(n_iter):
        if deadline is not None and time.time() > deadline:
            break
        if T < T_min:
            break
        
        # Propose: move mass from one coordinate to another
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        delta = rng.uniform(0.001, 0.05)
        
        # Try moving mass from i to j
        if h[i] > delta and h[j] < 1.0 - delta:
            h[i] -= delta
            h[j] += delta
            hc = h.copy()

def project_box_sum(h, target=None):

def lag_gradient(h, lag, dx):
    n = len(h)
    grad = np.zeros(n)
    if lag >= 0:
        s = n - lag
        if s > 0:
            grad[:s] -= h[lag:]
            grad[lag:] += 1.0 - h[:s]
    else:
        a = -lag
        s = n - a
        if s > 0:
            grad[:s] += 1.0 - h[a:]
            grad[a:] -= h[:s]
    return grad * dx

        if accept:
            h_cur = h_try
            c_cur = c_try
            if c_try < best_c:
                best_c = c_try
                best_h = h_try.copy()
        T *= 0.997
        if it % 400 == 399 and c_cur > best_c + 0.0005:
            h_cur = best_h.copy()
            c_cur = best_c
            T = max(T, T_init * 0.4)
    return best_h, best_c

def downsample(h, new_n):
    n = len(h)
    if new_n == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(new_n) + 0.5) / new_n
    vals = np.interp(x_new, x_old, h)
    return project_box_sum(vals, 0.5 * new_n)

def make_diverse_starts(n, rng, count=4):
    target = 0.5 * n
    starts = []
    h = np.zeros(n)
    h[::2] = 1.0
    starts.append(project_box_sum(h, target))
    h = np.zeros(n)
    h[:int(target)] = 1.0
    starts.append(project_box_sum(h, target))
    x = np.arange(n, dtype=np.float64)
    h = 0.5 + 0.35 * np.cos(2 * np.pi * x / n + rng.uniform(0, 2 * np.pi))
    starts.append(project_box_sum(h, target))
    h = rng.uniform(0, 1, n)
    starts.append(project_box_sum(h, target))
    return starts[:count]

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    deadline = start_time + budget_s - 15
    rng = np.random.RandomState(seed)

    best_h = None
    best_c = float('inf')

    try:
        init = np.asarray(initial_h_values, dtype=np.float64)
        if init.size > 0 and np.all(np.isfinite(init)):
            n_init = len(init)
            if n_init > 160:
                init = downsample(init, 140)
            init = project_box_sum(init, 0.5 * len(init))
            c0, _ = exact_objective(init)
            best_h = init.copy()
            best_c = c0
    except (NameError, Exception):
        pass

    if best_h is None:
        n0 = 70
        best_h = make_diverse_starts(n0, rng, 1)[0]
        best_c, _ = exact_objective(best_h)

    # Phase 1: Active-set LP on parent
    t_rem = deadline - time.time()
    if t_rem > 100:
        dl = min(deadline - 80, start_time + min(250, t_rem * 0.4))
        h_opt, c_opt = active_set_optimize(best_h, max_iters=400, deadline=dl)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt.copy()

    # Phase 2: Targeted 2-swap refinement (multiple rounds)
    for _ in range(3):
        if time.time() > deadline - 100:
            break
        h2, c2 = targeted_2swap(best_h, best_c, n_trials=300, rng=rng)
        if c2 < best_c - 1e-14:
            best_c = c2
            best_h = h2
        else:
            break

    # Phase 3: Binary threshold + swap
    n_cur = len(best_h)
    if n_cur % 2 == 0 and time.time() < deadline - 60:
        b = binary_threshold(best_h)
        if b is not None:
            b, c_b = binary_swap_improve(b, max_rounds=100, rng=rng)
            if c_b < best_c:
                best_c = c_b
                best_h = b.copy()
            if time.time() < deadline - 40:
                h_opt, c_opt = active_set_optimize(best_h, max_iters=200, deadline=deadline - 35)
                if c_opt < best_c:
                    best_c = c_opt
                    best_h = h_opt.copy()

    # Phase 4: SA escape + re-optimize
    if time.time() < deadline - 80:
        h_sa, c_sa = sa_escape(best_h, n_iter=2500, deadline=deadline - 70)
        if c_sa < best_c:
            best_c = c_sa
            best_h = h_sa.copy()
        if time.time() < deadline - 30:
            h_opt, c_opt = active_set_optimize(best_h, max_iters=250, deadline=deadline - 20)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt.copy()
            h2, c2 = targeted_2swap(best_h, best_c, n_trials=400, rng=rng)
            if c2 < best_c:
                best_c = c2
                best_h = h2

    # Phase 5: Try different sizes
    sizes = [48, 50, 52, 54, 56, 58, 60, 62, 64, 66, 68, 72, 74, 76, 80]
    sizes = [s for s in sizes if s != n_cur and 30 <= s <= 120]
    sizes.sort(key=lambda x: abs(x - n_cur))

    for n_try in sizes[:5]:
        if time.time() > deadline - 25:
            break
        h_ds = downsample(best_h, n_try)
        c_ds, _ = exact_objective(h_ds)
        starts = make_diverse_starts(n_try, rng, 3)
        h_start = h_ds
        c_start = c_ds
        for s in starts:
            cs, _ = exact_objective(s)
            if cs < c_start:
                c_start = cs
                h_start = s
        h_opt, c_opt = active_set_optimize(h_start, max_iters=150, deadline=deadline - 25)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt.copy()
        if n_try % 2 == 0 and time.time() < deadline - 15:
            b = binary_threshold(h_opt)
            if b is not None:
                b, c_b = binary_swap_improve(b, max_rounds=40, rng=rng)
                if c_b < best_c:
                    best_c = c_b
                    best_h = b.copy()

    # Phase 6: Final refinement on best
    if time.time() < deadline - 8:
        h_opt, c_opt = active_set_optimize(best_h, max_iters=300, deadline=deadline - 5)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt.copy()
        h2, c2 = targeted_2swap(best_h, best_c, n_trials=500, rng=rng)
        if c2 < best_c:
            best_c = c2
            best_h = h2
        n_fin = len(best_h)
        if n_fin % 2 == 0 and time.time() < deadline - 4:
            b = binary_threshold(best_h)
            if b is not None:
                b, c_b = binary_swap_improve(b, max_rounds=50, rng=rng)
                if c_b < best_c:
                    best_c = c_b
                    best_h = b.copy()

    # Final validation
    best_h = project_box_sum(best_h, 0.5 * len(best_h))
    final_c, _ = exact_objective(best_h)
    best_c = final_c
    best_n = len(best_h)
    return (best_h, best_c, best_n)
