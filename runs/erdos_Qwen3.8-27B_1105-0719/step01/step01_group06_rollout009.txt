                L = n - d
                G[k, 0:L] += dx * (1.0 - h[d:n])
                G[k, d:n] -= dx * h[0:L]
    return G

def repair_feasible(h, target=None):
    h = np.array(h, dtype=np.float64).copy()
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    if target is None:
        target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        room = 1.0 - h
        order = np.argsort(-room)
        remaining = diff
        for idx in order:
            if remaining <= 1e-14:
                break
            add = min(remaining, room[idx])
            h[idx] += add
            remaining -= add
    else:
        order = np.argsort(-h)
        remaining = -diff
        for idx in order:
            if remaining <= 1e-14:
                break
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
    return h

def interpolate_to_dim(vec, new_len):
    n = len(vec)
    x_src = (np.arange(n) + 0.5) * 2.0 / n
    x_tgt = (np.arange(new_len) + 0.5) * 2.0 / new_len
    x_ext = np.concatenate([[0.0], x_src, [2.0]])
    f_ext = np.concatenate([[vec[0]], vec, [vec[-1]]])
    h_new = np.interp(x_tgt, x_ext, f_ext)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

def lp_trust_step(h, radius, max_time=8.0):
    n = len(h)
    M = 2 * n - 1
    prof = get_profile(h)
    G = get_lag_gradients(h)
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_ub = np.hstack([G, -np.ones((M, 1))])
    b_ub = G @ h - prof
    lb = np.maximum(0.0, h - radius)
    ub = np.minimum(1.0, h + radius)
    bounds = list(zip(lb, ub))
    bounds.append((0.0, None))
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    t0 = time.perf_counter()
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
    except Exception:
        return None
    if time.perf_counter() - t0 > max_time:
        return None
    if res.status != 0:
        return None
    return res.x[:n]

def lp_trust_descent(h_start, max_iters=40, initial_radius=0.12, time_limit=120.0):
    h = h_start.copy()
    best_h = h.copy()
    best_score = exact_score(h)
    radius = initial_radius
    t0 = time.perf_counter()
    stall = 0
    for it in range(max_iters):
        if time.perf_counter() - t0 > time_limit:
            break
        x_new = lp_trust_step(h, radius, max_time=6.0)
        if x_new is None:
            radius *= 0.6
            stall += 1
            if stall > 5:
                break
            continue
        x_new = repair_feasible(x_new)
        step = x_new - h
        step_norm = np.linalg.norm(step)
        if step_norm < 1e-10:
            radius *= 0.6
            stall += 1
            if stall > 5:
                break
            continue
        improved = False
        for alpha in [1.0, 0.75, 0.5, 0.35, 0.2, 0.12, 0.06, 0.03]:
            h_trial = repair_feasible(h + alpha * step)
            s_trial = exact_score(h_trial)
            if s_trial < best_score - 1e-12:
                h = h_trial
                best_h = h.copy()
                best_score = s_trial
                radius = min(0.35, radius * 1.25)
                improved = True
                stall = 0
                break
        if not improved:
            radius *= 0.65
            stall += 1
            if stall > 6 or radius < 0.0008:
                break
    return best_h, best_score

def solve_epigraph_active(h_start, t_start, active_count=None, maxiter=80, beta=0.0):
    n = len(h_start)
    M = 2 * n - 1
    if active_count is None or active_count >= M:
        active_count = M

    def obj(x):
        h = x[:n]
        t = x[n]
        return t + beta * np.sum(h * (1.0 - h))

    def obj_jac(x):
        h = x[:n]
        g = np.zeros(n + 1)
        g[:n] = beta * (1.0 - 2.0 * h)
        g[n] = 1.0
        return g

    prof_init = get_profile(h_start)
    if active_count < M:
        active = np.argsort(prof_init)[::-1][:active_count]
    else:
        active = np.arange(M)

    def cons_fun(x):
        h = x[:n]
        t = x[n]
        eq = h.sum() - n / 2.0
        prof_x = get_profile(h)
        ineq = t - prof_x[active]
        return np.concatenate([[eq], ineq])

    def cons_jac(x):
        h = x[:n]
        G = get_lag_gradients(h)
        J = np.zeros((1 + len(active), n + 1))
        J[0, :n] = 1.0
        J[1:, :n] = -G[active, :]
        J[1:, n] = 1.0
        return J

    bounds = [(0.0, 1.0)] * n + [(0.0, t_start + 2.0)]
    x0 = np.concatenate([h_start, [t_start]])
    try:
        result = minimize(obj, x0, jac=obj_jac, method='SLSQP',
                          bounds=bounds,
                          constraints=[{'type': 'ineq', 'fun': cons_fun, 'jac': cons_jac}],
                          options={'maxiter': maxiter, 'ftol': 1e-12})
        if result.x is not None and np.all(np.isfinite(result.x)):
            return repair_feasible(result.x[:n])
    except Exception:
        pass
    return None

def directed_descent(h_current, max_iters=50, time_limit=30.0, temp=80.0):
    n = len(h_current)
    h = h_current.copy()
    best_h = h.copy()
    best_score = exact_score(h)
    t0 = time.perf_counter()
    for it in range(max_iters):
        if time.perf_counter() - t0 > time_limit:
            break
        prof = get_profile(h)
        score = float(np.max(prof))
        if score < best_score - 1e-13:
            best_score = score
            best_h = h.copy()
        A = max(10, min(50, n // 2))
        active_idx = np.argsort(prof)[::-1][:A]
        vals = prof[active_idx]
        vals_shifted = vals - vals.max()
        w = np.exp(-temp * vals_shifted)
        wsum = w.sum()
        if wsum < 1e-15:
            break
        w /= wsum
        G = get_lag_gradients(h)
        direction = np.sum(w[:, None] * G[active_idx, :], axis=0)
        direction -= direction.mean()
        norm = np.linalg.norm(direction)
        if norm < 1e-14:
            break
        direction /= norm
        improved = False
        for alpha in [0.04, 0.025, 0.015, 0.008, 0.004, 0.002, 0.001]:
            h_trial = repair_feasible(h + alpha * direction)
            s_trial = exact_score(h_trial)
            if s_trial < best_score - 1e-11:
                h = h_trial
                best_h = h.copy()
                best_score = s_trial
                improved = True
                break
        if not improved:
            break
    return best_h, best_score

def multi_start_lp(h_best, best_score, seed, n_starts, time_per_start, budget_fn):
    n = len(h_best)
    rng = np.random.RandomState(seed)
    for s in range(n_starts):
        if budget_fn() < time_per_start + 5:
            break
        eps = 0.002 + 0.003 * s
        z = rng.randn(n)
        z -= z.mean()
        h_pert = repair_feasible(h_best + eps * z)
        tl = min(time_per_start, budget_fn() - 5)
        if tl < 10:
            break
        h_lp, s_lp = lp_trust_descent(h_pert, max_iters=15, initial_radius=0.1, time_limit=tl)
        if s_lp < best_score - 1e-12:
            best_score = s_lp
            h_best = h_lp.copy()
    return h_best, best_score

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t_start_global = time.perf_counter()

    def tleft():
        return budget_s - (time.perf_counter() - t_start_global)

    try:
        parent = np.array(initial_h_values, dtype=np.float64).copy()
    except Exception:
        parent = np.full(64, 0.5)

    parent = repair_feasible(parent)
    best_h = parent.copy()
    best_score = exact_score(parent)

    # Stage 1: LP trust-region on parent
    if tleft() > 80:
        tl = min(150, tleft() * 0.25)
        h_lp, s_lp = lp_trust_descent(best_h, max_iters=50, initial_radius=0.15, time_limit=tl)
        if s_lp < best_score - 1e-12:
            best_h = h_lp
            best_score = s_lp

    # Stage 2: SLSQP polishing with active constraints
    if tleft() > 50:
        n = len(best_h)
        for ac in [40, 70, 100]:
            if tleft() < 30:
                break
            if ac >= 2 * n - 1:
                ac = 2 * n - 1
            mi = min(100, max(40, int(tleft() * 0.15)))
            h_sol = solve_epigraph_active(best_h, best_score + 1e-6, active_count=ac, maxiter=mi)
            if h_sol is not None:
                s = exact_score(h_sol)
                if s < best_score - 1e-12:
                    best_h = h_sol
                    best_score = s

    # Stage 3: Another LP pass with smaller radius
    if tleft() > 50:
        tl = min(80, tleft() * 0.18)
        h_lp2, s_lp2 = lp_trust_descent(best_h, max_iters=35, initial_radius=0.06, time_limit=tl)
        if s_lp2 < best_score - 1e-12:
            best_h = h_lp2
            best_score = s_lp2

    # Stage 4: Directed descent
    if tleft() > 30:
        tl = min(20, tleft() * 0.12)
        h_dd, s_dd = directed_descent(best_h, max_iters=50, time_limit=tl, temp=60.0)
        if s_dd < best_score - 1e-12:
            best_h = h_dd
            best_score = s_dd

    # Stage 5: Dimension exploration
    if tleft() > 80:
        pn = len(parent)
        dims = set()
        for scale in [0.5, 0.6, 0.7, 0.8, 0.85, 0.9, 1.1, 1.2, 1.4, 1.6]:
            d = int(round(pn * scale / 2)) * 2
            if 32 <= d <= 260:
                dims.add(d)
        for d in [60, 80, 100, 120, 140, 160, 180, 200, 220]:
            if 32 <= d <= 260:
                dims.add(d)
        dims.discard(pn)
        dims = sorted(dims)

        for new_n in dims:
            if tleft() < 35:
                break
            h_interp = interpolate_to_dim(best_h, new_n)
            s_interp = exact_score(h_interp)
            if s_interp < best_score - 1e-12:
                best_h = h_interp
                best_score = s_interp
            tl = min(50, tleft() * 0.12)
            if tl < 15:
                continue
            h_lp, s_lp = lp_trust_descent(h_interp, max_iters=25, initial_radius=0.12, time_limit=tl)
            if s_lp < best_score - 1e-12:
                best_h = h_lp
                best_score = s_lp
            if tleft() > 25:
                mi = min(60, max(20, int(tleft() * 0.08)))
                h_sol = solve_epigraph_active(h_lp, s_lp + 1e-6, active_count=50, maxiter=mi)
                if h_sol is not None:
                    s = exact_score(h_sol)
                    if s < best_score - 1e-12:
                        best_h = h_sol
                        best_score = s

    # Stage 6: Multi-start perturbations around best
    if tleft() > 60:
        h_best2, s_best2 = multi_start_lp(best_h, best_score, seed + 100,
                                          n_starts=5, time_per_start=20, budget_fn=tleft)
        if s_best2 < best_score - 1e-12:
            best_h = h_best2
            best_score = s_best2

    # Stage 7: Final directed descent
    if tleft() > 20:
        tl = min(15, tleft() * 0.3)
        h_dd2, s_dd2 = directed_descent(best_h, max_iters=40, time_limit=tl, temp=100.0)
        if s_dd2 < best_score - 1e-12:
            best_h = h_dd2
            best_score = s_dd2

    # Stage 8: Final tiny LP
    if tleft() > 12:
        tl = min(8, tleft() * 0.5)
        h_lp3, s_lp3 = lp_trust_descent(best_h, max_iters=10, initial_radius=0.03, time_limit=tl)
        if s_lp3 < best_score - 1e-12:
            best_h = h_lp3
            best_score = s_lp3

    best_h = repair_feasible(best_h)
    final_score = exact_score(best_h)
    return (best_h, final_score, len(best_h))
