import time

def exact_objective(h):
    n = len(h)
    if n == 0:
        return float('inf')
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(profile))

def profile_raw(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    """Project h to [0,1]^n with sum = n/2"""
    h = np.array(h, dtype=np.float64).copy()
    target = n / 2.0
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = np.sum(h)
        diff = s - target
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h > 0
            cnt = np.sum(mask)
            if cnt == 0:
                break
            step = min(diff / cnt, np.min(h[mask]))
            h[mask] -= step
        else:
            mask = h < 1.0
            cnt = np.sum(mask)
            if cnt == 0:
                break
            step = min(-diff / cnt, 1.0 - np.max(h[mask]))
            h[mask] += step
    # Final correction

    if abs(diff) > 1e-14:

            mask = h > 0
        else:
            mask = h < 1.0
        if np.any(mask):

    return h

    n = len(h)

    active = raw >= (raw_max - tol * raw_max)
    if np.sum(active) < 1:
        active[np.argmax(raw)] = True
    # Limit to top_k
    if np.sum(active) > top_k:
        idx = np.where(active)[0]
        sorted_idx = idx[np.argsort(-raw[idx])][:top_k]
        active = np.zeros_like(raw, dtype=bool)
        active[sorted_idx] = True
    
    # Compute gradient
    grad = np.zeros(n)
    for t in np.where(active)[0]:
        k = t - (n - 1)  # signed lag
        if k == 0:

            grad[0:n-a] -= h[a:n]

    return grad

def active_gradient_descent(h0, max_iter=5000, tol=0.01, top_k=15):
    """Projected subgradient descent on the max profile"""
    n = len(h0)
    h = h0.copy()
    c_cur = exact_objective(h)
    step_size = 0.01
    patience = 0
    
    for it in range(max_iter):
        grad = gradient_of_max(h, top_k=top_k, tol=tol)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-10:
            break
        
        # Backtracking line search
        for alpha in step_size:
            h_new = project_feasible(h - alpha * grad, n)
            c_new = exact_objective(h_new)
            if c_new < c_cur - 1e-12:
                h = h_new
                c_cur = c_new
                step_size = min(step_size * 1.5, 0.1)
                patience = 0
                break
            else:
                alpha *= 0.5
        
        patience += 1
        if patience > 200:
            step_size *= 0.5
            if step_size < 1e-6:
                break
            patience = 0
    
    return h, c_cur

def binary_local_search(b, max_rounds=5000):
    """Local search on binary sequences using directed swaps"""
    n = len(b)
    cur = b.copy()
    cur_c = exact_objective(cur)
    raw_cur = profile_raw(cur)
    
    for _ in range(max_rounds):
        best_delta = 0.0
        best_pair = None
        ones = np.where(cur == 1.0)[0]
        zeros = np.where(cur == 0.0)[0]
        
        # Focus on swaps that might reduce the max
        raw_max = np.max(raw_cur)
        
        for i in ones:
            for j in zeros:
                cur[i] = 0.0
                cur[j] = 1.0
                raw_new = profile_raw(cur)
                new_max = np.max(raw_new)
                cur[i] = 1.0
                cur[j] = 0.0
                
                delta = raw_max - new_max
                if delta > best_delta + 1e-12:
                    best_delta = delta
                    best_pair = (i, j)

            cur[i] = 0.0
            cur[j] = 1.0
            cur_c -= (best_delta) * (2.0 / n)
            raw_cur = profile_raw(cur)
        else:
            break
    
    return cur, cur_c

def compute_contrib_matrix(h):
    """Precompute partial derivatives of all raw[t] w.r.t. h[i] for all i.
    Returns matrix of shape (n, 2n-1)."""
    n = len(h)
    contrib = np.zeros((n, 2 * n - 1))
    for i in range(n):
        for t in range(2 * n - 1):
            s = t - (n - 1)
            val = 0.0
            if 0 <= i + s < n:
                val += (1.0 - h[i + s])
            if 0 <= i - s < n:
                val -= h[i - s]
            contrib[i, t] = val
    return contrib

def binary_local_search_fast(b, max_rounds=2000, deadline=None):
    """Fast binary local search using precomputed contribution matrix."""
    n = len(b)
    cur = b.copy()
    raw_cur = profile_raw(cur)
    cur_c = np.max(raw_cur) * (2.0 / n)
    
    for rnd in range(max_rounds):
        if deadline is not None and time.time() > deadline - 10:
            break
        contrib = compute_contrib_matrix(cur)
        ones = np.where(cur == 1.0)[0]
        zeros = np.where(cur == 0.0)[0]
        
        # Vectorized: compute delta for all pairs at once
        # delta for swap (a in ones, b in zeros): contrib[b] - contrib[a]
        best_delta = 0.0
        best_pair = None
        best_idx_a, best_idx_b = -1, -1
        
        # Batch process to speed up
        batch_size = 500
        for start in range(0, len(ones), batch_size):
            if deadline is not None and time.time() > deadline - 10:
                return cur, cur_c
            o_batch = ones[start:start+batch_size]
            for j in zeros:
                deltas = contrib[j] - contrib[o_batch[None, :].T]
                new_maxes = np.max(raw_cur[None, :] + deltas.T, axis=1)
                imps = cur_c * n - new_maxes  # in raw units
                best_local = np.argmax(imps)
                if imps[best_local] > best_delta:
                    best_delta = imps[best_local]
                    best_pair = (o_batch[best_local], j)
        
        if best_pair is not None and best_delta > 1e-10:
            a, b = best_pair
            cur[a] = 0.0
            cur[b] = 1.0
            raw_cur = profile_raw(cur)
            cur_c = np.max(raw_cur) * (2.0 / n)
        else:
            break
    
    return cur, cur_c

def active_subgradient_descent(h0, max_iter=3000, deadline=None, top_k=20, tol_frac=0.03):
    """Projected subgradient descent focusing on active lags."""
    n = len(h0)
    h = h0.copy()
    c_cur = exact_objective(h)
    step_sizes = [0.005, 0.003, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-5, 2e-5, 1e-5]
    si = 0
    stall = 0
    
    for it in range(max_iter):
        if deadline is not None and time.time() > deadline - 15:
            break
        
        raw = profile_raw(h)
        raw_max = np.max(raw)
        dx = 2.0 / n
        
        # Find active lags
        thresh = raw_max - tol_frac * raw_max
        active_mask = raw >= thresh
        if np.sum(active_mask) == 0:
            active_mask[np.argmax(raw)] = True
        if np.sum(active_mask) > top_k:
            idx = np.where(active_mask)[0]
            sorted_idx = idx[np.argsort(-raw[idx])][:top_k]
            active_mask = np.zeros(2*n-1, dtype=bool)
            active_mask[sorted_idx] = True
        
        # Compute subgradient (average over active)
        grad = np.zeros(n)
        for t in np.where(active_mask)[0]:
            s = t - (n - 1)
            if s == 0:
                grad += (1.0 - 2.0 * h)
            elif s > 0:
                grad[0:n-s] += (1.0 - h[s:n])
                grad[s:n] -= h[0:n-s]
            else:
                a = -s
                grad[a:n] += (1.0 - h[0:n-a])
                grad[0:n-a] -= h[a:n]
        grad *= dx / np.sum(active_mask)
        
        # Line search
        improved = False
        for alpha in step_sizes[si]:
            h_new = project_feasible(h - alpha * grad, n)
            c_new = exact_objective(h_new)
            if c_new < c_cur - 1e-13:
                h = h_new
                c_cur = c_new
                improved = True
                stall = 0
                break
            alpha = alpha * 0.5
        
        if not improved:
            stall += 1
            if stall >= 10 and si < len(step_sizes) - 1:
                si += 1
                stall = 0
            elif stall >= 50:
                break
    
    return h, c_cur

def smooth_refine(h0, tau=0.001, maxiter=100, deadline=None):
    """SLSQP with smooth log-sum-exp for final polish."""
    n = len(h0)
    dx = 2.0 / n
    target = n / 2.0
    
    def obj(x):
        r = profile_raw(x)
        r_max = np.max(r)
        shifted = np.clip((r - r_max) / tau, -400, 400)
        exp_s = np.exp(shifted)
        s = np.sum(exp_s)
        val = dx * (r_max + tau * np.log(s))
        # Add sum constraint penalty
        sv = np.sum(x) - target
        val += 1e6 * sv * sv
        return val
    
    def obj_grad(x):
        r = profile_raw(x)
        r_max = np.max(r)
        shifted = np.clip((r - r_max) / tau, -400, 400)
        exp_s = np.exp(shifted)
        s = np.sum(exp_s)
        weights = exp_s / s
        grad = np.zeros(n)
        for t in range(2*n-1):
            w = weights[t]
            if w < 1e-12:
                continue
            sg = t - (n-1)
            if sg == 0:
                grad += w * (1.0 - 2.0 * x)
            elif sg > 0:
                grad[0:n-sg] += w * (1.0 - x[sg:n])
                grad[sg:n] -= w * x[0:n-sg]
            else:
                a = -sg
                grad[a:n] += w * (1.0 - x[0:n-a])
                grad[0:n-a] -= w * x[a:n]
        grad *= dx
        sv = np.sum(x) - target
        grad += 2e6 * sv * np.ones(n)
        return grad
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0, jac=obj_grad, method='SLSQP', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        h_new = project_feasible(res.x, n)
        return h_new
    except Exception:
        return h0

def transfer_to_size(parent, target_n):
    n = len(parent)
    if n == target_n:
        return parent.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(target_n) + 0.5) / target_n
    new_h = np.interp(x_new, x_old, parent)
    return project_feasible(new_h, target_n)

def binary_threshold(h):
    n = len(h)
    if n % 2 != 0:
        return None
    target = n // 2
    idx = np.argsort(-h)
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b

def random_binary(n, rng):
    b = np.zeros(n)
    idx = rng.choice(n, n//2, replace=False)
    b[idx] = 1.0
    return b

def structured_initial(n, rng):
    """Generate structured initial binary patterns."""
    candidates = []
    # Alternating
    b1 = np.zeros(n)
    b1[::2] = 1.0
    if np.sum(b1) != n//2:
        b1[n-1] = 1.0 - b1[n-1]
    candidates.append(project_feasible(b1, n))
    
    # Block pattern: first half 1, second half 0
    b2 = np.zeros(n)
    b2[:n//2] = 1.0
    candidates.append(project_feasible(b2, n))
    
    # Sinusoidal threshold
    x = np.linspace(0, 2*np.pi, n, endpoint=False)
    b3 = (0.5 + 0.3*np.sin(x)) 
    b3[b3 > 0.5] = 1.0
    b3[b3 <= 0.5] = 0.0
    s = np.sum(b3)
    if s != n//2:
        idx = np.argsort(-b3)[:max(0, n//2-int(s))]
        b3[idx] = 1.0
    candidates.append(project_feasible(b3, n))
    
    # Multiple frequency
    x2 = np.linspace(0, 4*np.pi, n, endpoint=False)
    b4 = 0.5 + 0.25*np.sin(x2) + 0.15*np.sin(3*x2)
    b4[b4 > 0.5] = 1.0
    b4[b4 <= 0.5] = 0.0
    s4 = np.sum(b4)
    if s4 != n//2:
        if s4 < n//2:
            idx = np.where(b4 == 0)[0]
            picks = rng.choice(len(idx), min(n//2-int(s4), len(idx)), replace=False)
            b4[idx[picks]] = 1.0
        else:
            idx = np.where(b4 == 1)[0]
            picks = rng.choice(len(idx), min(int(s4)-n//2, len(idx)), replace=False)
            b4[idx[picks]] = 0.0
    candidates.append(project_feasible(b4, n))
    
    return candidates

def run(seed=42, budget_s=1100, **kwargs):
    deadline = time.time() + budget_s - 10
    rng = np.random.RandomState(seed)
    
    best_h = None
    best_c = float('inf')
    best_n = 0
    
    # Load parent
    try:
        parent = np.array(initial_h_values, dtype=np.float64)
        parent = project_feasible(parent, len(parent))
        pc = exact_objective(parent)
        best_h = parent.copy()
        best_c = pc
        best_n = len(parent)
    except Exception:
        best_h = np.ones(70) * 0.5
        best_h = project_feasible(best_h, 70)
        best_c = exact_objective(best_h)
        best_n = 70
    
    # Size pool: parent size + neighbors + smaller sizes
    sizes = [best_n]
    for s in [best_n-10, best_n-20, best_n+10, 100, 80, 70, 60, 50, 40, 120, 90, 160]:
        if s not in sizes and 20 <= s <= 400 and s % 2 == 0:
            sizes.append(s)
    
    # Sort by priority: parent first, then by size (smaller preferred for speed)
    sizes_sorted = [best_n] + sorted([s for s in sizes if s != best_n], key=lambda x: abs(x - best_n))
    
    for n in sizes_sorted:
        if time.time() > deadline - 80:
            break
        if n % 2 != 0 or n < 20:
            continue
        
        n_deadline = min(deadline, time.time() + 200)  # time per size
        
        # Generate candidates
        candidates = []
        
        # Transferred parent
        if len(best_h) != n:
            tr = transfer_to_size(best_h, n)
            candidates.append(tr)
        else:
            candidates.append(best_h.copy())
        
        # Perturbed parent
        tr = transfer_to_size(best_h, n)
        for scale in [0.05, 0.1, 0.2]:
            pert = project_feasible(tr + scale * rng.randn(n), n)
            candidates.append(pert)
        
        # Structured
        for sc in structured_initial(n, rng):
            candidates.append(sc)
        
        # Random binary
        for _ in range(2):
            rb = random_binary(n, rng)
            candidates.append(rb)
        
        # Evaluate and sort
        scored = []
        seen = set()
        for c in candidates:
            c = project_feasible(c, n)
            key = tuple(np.round(c, 3))
            if key in seen:
                continue
            seen.add(key)
            cc = exact_objective(c)
            scored.append((c, cc))
        
        scored.sort(key=lambda z: z[1])
        
        # Update best
        for c, cc in scored[:3]:
            if cc < best_c - 1e-12:
                best_c = cc
                best_h = c.copy()
                best_n = n
        
        # Optimize top candidates
        top_k = min(3, len(scored))
        for ci in range(top_k):
            if time.time() > n_deadline:
                break
            
            h_cur, c_cur = scored[ci][0].copy(), scored[ci][1]
            
            # Stage 1: Active subgradient descent (coarse)
            if time.time() < n_deadline - 30:
                h_sg, c_sg = active_subgradient_descent(h_cur, max_iter=2000, deadline=n_deadline, top_k=15, tol_frac=0.05)
                if c_sg < c_cur:
                    h_cur, c_cur = h_sg, c_sg
            
            # Stage 2: Active subgradient descent (fine)
            if time.time() < n_deadline - 20:
                h_sg2, c_sg2 = active_subgradient_descent(h_cur, max_iter=3000, deadline=n_deadline, top_k=25, tol_frac=0.02)
                if c_sg2 < c_cur:
                    h_cur, c_cur = h_sg2, c_sg2
            
            # Stage 3: Smooth SLSQP polish
            if time.time() < n_deadline - 20:
                for tau in [0.005, 0.002, 0.001, 0.0005]:
                    if time.time() > n_deadline - 15:
                        break
                    h_sm = smooth_refine(h_cur, tau=tau, maxiter=80, deadline=n_deadline)
                    c_sm = exact_objective(h_sm)
                    if c_sm < c_cur - 1e-12:
                        h_cur, c_cur = h_sm, c_sm
            
            # Stage 4: Binary threshold + local search
            if n % 2 == 0 and time.time() < n_deadline - 10:
                b = binary_threshold(h_cur)
                if b is not None:
                    b_c = exact_objective(b)
                    if b_c < c_cur:
                        h_cur, c_cur = b, b_c
                    b2, b2_c = binary_local_search_fast(h_cur, max_rounds=1000, deadline=n_deadline)
                    if b2_c < c_cur:
                        h_cur, c_cur = b2, b2_c
            
            # Stage 5: Another round of subgradient from binary
            if time.time() < n_deadline - 15 and c_cur < best_c:
                h_sg3, c_sg3 = active_subgradient_descent(h_cur, max_iter=2000, deadline=n_deadline, top_k=20, tol_frac=0.03)
                if c_sg3 < c_cur:
                    h_cur, c_cur = h_sg3, c_sg3
                if time.time() < n_deadline - 10:
                    h_sm2 = smooth_refine(h_cur, tau=0.001, maxiter=60, deadline=n_deadline)
                    c_sm2 = exact_objective(h_sm2)
                    if c_sm2 < c_cur:
                        h_cur, c_cur = h_sm2, c_sm2
            
            # Update global best
            if c_cur < best_c - 1e-12:
                best_c = c_cur
                best_h = h_cur.copy()
                best_n = n
    
    # Final: try binary local search on the best solution
    if time.time() < deadline - 30 and best_n % 2 == 0:
        b_final = binary_threshold(best_h)
        if b_final is not None:
            b_final, bc = binary_local_search_fast(b_final, max_rounds=2000, deadline=deadline)
            if bc < best_c - 1e-12:
                best_c = bc
                best_h = b_final.copy()
                best_n = len(b_final)
    
    # Final projection and evaluation
    best_h = project_feasible(best_h, best_n)
    best_c = exact_objective(best_h)
    
    return (best_h, best_c, len(best_h))
