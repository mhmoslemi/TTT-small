    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def get_lag_gradients(h):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    G = np.zeros((M, n))
    for k in range(M):
        delta = k - (n - 1)
        if delta >= 0:
            if delta < n:
                L = n - delta
                G[k, delta:n] += dx * (1.0 - h[0:L])
                G[k, 0:L] -= dx * h[delta:n]
        else:
            d = -delta
            if d < n:
                L = n - d
                G[k, 0:L] += dx * (1.0 - h[d:n])
                G[k, d:n] -= dx * h[0:L]
    return G

def repair_feasible(h, target=None):
