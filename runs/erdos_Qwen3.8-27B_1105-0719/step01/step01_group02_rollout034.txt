def project_box_sum(h, target=None):

    h = np.array(h, dtype=np.float64)

    # Final correction
    diff = np.sum(h) - target
    if abs(diff) > 1e-13:
        mask = (h > 1e-15) if diff > 0 else (h < 1.0 - 1e-15)
        if mask.any():
            h[mask] -= diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_gradients(h, lag):
    """Compute gradient of f_lag with respect to h.
    lag is the actual lag (k), where correlate index = lag + n - 1.
    """
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    if lag == 0:
        grad[:] = 1.0 - 2.0 * h
    elif lag > 0:
        # grad[i] for i in [0, lag): (1 - h[i+lag])
        grad[:n-lag] += (1.0 - h[lag:n])
        # grad[i] for i in [lag, n): -(h[i-lag])
        grad[lag:] -= h[:n-lag]
    else:
        a = -lag
        # grad[i] for i in [0, n-a): (1 - h[i+a])
        grad[:n-a] += (1.0 - h[a:n])
        # grad[i] for i in [a, n): -(h[i-a])
        grad[a:] -= h[:n-a]
    return grad * dx

def active_set_step(h, active_indices, delta=0.03):
    """Solve LP to find best step direction.
    active_indices: indices into the correlate output (0 to 2n-2)
    Returns (direction, t_value) or None.
    """
    n = len(h)
    nv = 1 + 2 * n  # t, d_pos, d_neg
    
    # Objective: minimize t
    c = np.zeros(nv)
    c[0] = 1.0
    
    # Build constraints
    A_rows = []
    b_vals = []
    
    # For each active lag: -t - sum g_k[i]*d_pos[i] + sum g_k[i]*d_neg[i] <= 0
    for idx in active_indices:
        lag = idx - (n - 1)
        g = compute_gradients(h, lag)
        row = np.zeros(nv)
        row[0] = -1.0  # -t
        row[1:1+n] = -g  # -sum g[i]*d_pos[i]
        row[1+n:] = g    # +sum g[i]*d_neg[i]
        A_rows.append(row)
        b_vals.append(0.0)
    
    # Trust region: d_pos[i] - d_neg[i] <= delta
    for i in range(n):
        row = np.zeros(nv)
        row[1+i] = 1.0
        row[1+n+i] = -1.0
        A_rows.append(row)
        b_vals.append(delta)
    
    # Trust region: -(d_pos[i] - d_neg[i]) <= delta → d_neg[i] - d_pos[i] <= delta
    for i in range(n):
        row = np.zeros(nv)
        row[1+i] = -1.0
        row[1+n+i] = 1.0
        A_rows.append(row)
        b_vals.append(delta)
    
    # Box: d_pos[i] <= 1 - h[i] (handled by bounds)
    # Box: d_neg[i] <= h[i] (handled by bounds)
    
    A_ub = np.array(A_rows)
    b_ub = np.array(b_vals)
    
    # Equality: sum d_pos - sum d_neg = 0
    A_eq = np.zeros((1, nv))
    A_eq[0, 1:1+n] = 1.0
    A_eq[0, 1+n:] = -1.0
    b_eq = np.zeros(1)
    
    # Bounds
    bounds = [(-np.inf, np.inf)]  # t unbounded
    for i in range(n):
        bounds.append((0, max(0, 1.0 - h[i])))
        bounds.append((0, max(0, h[i])))
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            t_val = res.x[0]
            d_pos = res.x[1:1+n]
            d_neg = res.x[1+n:]
            direction = d_pos - d_neg
            return direction, t_val
    except Exception:
        pass
    return None, None

def active_set_optimize(h, max_iter=200, delta_init=0.05, n_active=5):
    """Iterative active-set LP optimization."""
    n = len(h)
    h = h.copy()
    c_cur = exact_objective(h)
    
    for it in range(max_iter):
        profile = np.correlate(h, 1.0 - h, mode='full') * (2.0/n)
        c_max = profile.max()
        # Find active lags (within tolerance of max)
        tol = 0.01 * c_max + 0.001
        active_mask = profile >= c_max - tol
        active_indices = np.where(active_mask)[0]
        # Sort by value, take top n_active
        active_indices = active_indices[np.argsort(-profile[active_indices])][:n_active]
        
        if len(active_indices) == 0:
            break
        
        # Try step
        direction, t_val = active_set_step(h, active_indices, delta_init)
        if direction is None:
            break
        
        if t_val >= -1e-10:
            # No improvement found with this active set
            # Try smaller delta or break
            delta_init *= 0.5
            if delta_init < 0.001:
                break
            continue
        
        # Take step with line search
        best_alpha = 0.0
        best_c = c_cur
        for alpha in [1.0, 0.5, 0.25, 0.1]:
            h_try = project_box_sum(h + alpha * direction)
            c_try = exact_objective(h_try)
            if c_try < best_c:

                best_alpha = alpha

            if delta_init < 0.001:
                break
            continue
        
        h = project_box_sum(h + best_alpha * direction)
        c_cur = exact_objective(h)
        # Slightly increase delta for exploration
        delta_init = min(0.1, delta_init * 1.1)
    
    return h, exact_objective(h)

import numpy as np
from scipy.optimize import linprog
import time

def exact_objective(h):
    n = len(h)
    if n == 0:
        return float('inf')
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_box_sum(h, target=None):
    n = len(h)
    h = np.array(h, dtype=np.float64).copy()
    if target is None:
        target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = h.sum() - target
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            m = h > 0
            if not m.any(): break
            s = min(diff / m.sum(), h[m].min())
            h[m] -= s
        else:
            m = h < 1.0
            if not m.any(): break
            s = min(-diff / m.sum(), (1.0 - h[m]).max())
            h[m] += s
    diff = h.sum() - target
    if abs(diff) > 1e-13:
        m = (h > 0) if diff > 0 else (h < 1.0)
        if m.any():
            h[m] -= diff / m.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def grad_profile(h, t_idx):
    """Gradient of profile[t_idx] w.r.t. h.
    t_idx is the index into the full correlation output (0 to 2n-2).
    """
    n = len(h)
    dx = 2.0 / n
    k = t_idx - (n - 1)  # actual lag
    g = np.zeros(n)
    if k >= 0:
        # f_k = dx * sum_{j=0}^{n-k-1} h[j]*(1-h[j+k])
        # df_k/dh[i] = dx*((1-h[i+k])*1_{i<n-k} - h[i-k]*1_{i>=k})
        g[:n-k] += (1.0 - h[k:n])      # (1-h[i+k]) for i in [0, n-k)
        g[k:] -= h[:n-k]               # -h[i-k] for i in [k, n)
    else:
        a = -k
        g[:n-a] += (1.0 - h[a:n])      # (1-h[i+a]) for i in [0, n-a)
        g[a:] -= h[:n-a]               # -h[i-a] for i in [a, n)
    return g * dx

def active_set_lp(h, active_t, delta=0.03):
    """
    Solve LP: maximize t subject to
      grad_profile(h, t_i) @ d <= -(profile[t_i] - C) - t  for each active t_i
      sum(d) = 0
      |d[i]| <= delta (via bounds on d_pos, d_neg)
      0 <= h[i] + d[i] <= 1
    """
    n = len(h)
    profile = np.correlate(h, 1.0 - h, mode='full') * (2.0/n)
    C = profile.max()
    nv = 1 + 2*n
    
    c_obj = np.zeros(nv)
    c_obj[0] = -1.0  # minimize -t
    
    A_rows = []
    b_vals = []
    
    for t_i in active_t:
        g = grad_profile(h, t_i)  # already includes dx
        row = np.zeros(nv)
        row[0] = -1.0
        row[1:1+n] = g
        row[1+n:] = -g
        A_rows.append(row)
        b_vals.append(-(profile[t_i] - C))  # = C - profile[t_i]
    
    A_ub = np.array(A_rows) if A_rows else None
    b_ub = np.array(b_vals) if b_vals else None
    
    A_eq = np.zeros((1, nv))
    A_eq[0, 1:1+n] = 1.0
    A_eq[0, 1+n:] = -1.0
    
    bounds = [(0, None)]  # t >= 0
    for i in range(n):
        bounds.append((0, min(delta, 1.0 - h[i])))
        bounds.append((0, min(delta, h[i])))
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.zeros(1),
                      bounds=bounds, method='highs')
        if res.success and res.x[0] > 1e-10:
            d = res.x[1:1+n] - res.x[1+n:]
            return d, res.x[0]
    except:
        pass
    return None, 0.0

def active_set_optimize(h, max_iter=100, delta=0.03, n_active=5, tol=0.01):
    n = len(h)
    h = h.copy()
    c_cur = exact_objective(h)
    no_improve = 0
    
    for it in range(max_iter):
        profile = np.correlate(h, 1.0 - h, mode='full') * (2.0/n)
        C = profile.max()
        
        # Active set: top n_active lags
        sorted_idx = np.argsort(-profile)
        active_t = sorted_idx[:n_active]
        
        # Only include lags that are close to C
        active_t = active_t[profile[active_t] >= C - tol * C]
        if len(active_t) == 0:
            active_t = sorted_idx[:min(n_active, len(sorted_idx))]
        
        d, t_val = active_set_lp(h, active_t, delta)
        if d is None:
            no_improve += 1
            if no_improve >= 5:
                break
            delta *= 0.7
            continue
        
        # Line search
        best_alpha = 0.0
        best_c = c_cur
        for alpha in [1.0, 0.7, 0.5, 0.3, 0.1]:
            h_try = project_box_sum(h + alpha * d)
            c_try = exact_objective(h_try)
            if c_try < best_c - 1e-12:
                best_c = c_try
                best_alpha = alpha
        
        if best_alpha == 0:
            no_improve += 1
            delta *= 0.7
            if no_improve >= 10:
                break
            continue
        
        h = project_box_sum(h + best_alpha * d)
        c_cur = exact_objective(h)
        no_improve = 0
        delta = min(0.08, delta * 1.05)
    
    return h, exact_objective(h)

def transfer_to_size(parent, target_n):
    n = len(parent)
    if n == target_n:
        return parent.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(target_n) + 0.5) / target_n
    new_h = np.interp(x_new, x_old, parent)
    return project_box_sum(new_h)

def binary_threshold(h):
    n = len(h)
    if n % 2 != 0:
        return None
    target = n // 2
    idx = np.argsort(-h)
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b

def binary_SA(b, n_rounds=2000, rng_seed=42):
    """Simulated annealing for binary swaps."""
    n = len(b)
    dx = 2.0 / n
    b = b.copy()
    c_cur = exact_objective(b)
    rg = np.random.RandomState(rng_seed)
    T = 0.005
    best_c = c_cur
    best_b = b.copy()
    
    for it in range(n_rounds):
        T = 0.005 * (0.999 ** it)
        ones = np.where(b == 1.0)[0]
        zeros = np.where(b == 0.0)[0]
        i = ones[rg.randint(len(ones))]
        j = zeros[rg.randint(len(zeros))]
        b[i], b[j] = 0.0, 1.0
        c_new = exact_objective(b)
        if c_new < c_cur or rg.random() < np.exp(-(c_new - c_cur) / max(T, 1e-10)):
            c_cur = c_new
            if c_cur < best_c:
                best_c = c_cur
                best_b = b.copy()
        else:
            b[i], b[j] = 1.0, 0.0
    
    return best_b, best_c

Hmm wait, I realize there might be an issue with the LP constraint. Let me re-derive.

We want to find direction d such that for each active lag t_i:
profile[t_i] + grad[t_i] @ d <= C - t

where C = max(profile) and t is the improvement we want to achieve.

Rearranging:
grad[t_i] @ d <= (C - t) - profile[t_i]
grad[t_i] @ d <= C - profile[t_i] - t

Let me substitute d = d_pos - d_neg:
grad[t_i] @ d_pos - grad[t_i] @ d_neg <= C - profile[t_i] - t

Row for constraint t_i:
[t coefficient: 1, d_pos coefficients: grad[t_i], d_neg coefficients: -grad[t_i]] <= C - profile[t_i]

Wait: t + grad[t_i] @ d_pos - grad[t_i] @ d_neg <= C - profile[t_i]

So the row is: [1, grad[t_i], -grad[t_i]] with RHS = C - profile[t_i]

And we minimize -t (to maximize t).

Let me double-check: if profile[t_i] = C (fully active), then RHS = 0, and we need:
t + grad[t_i] @ d <= 0
→ t <= -grad[t_i] @ d

This means t is bounded above by the rate at which we're decreasing the active profile. Makes sense.

If profile[t_i] = C - ε (nearly active), then RHS = ε, and we need:
t + grad[t_i] @ d <= ε
→ t <= ε - grad[t_i] @ d

This gives more room for t since the profile is already below C.

OK so the LP formulation is:
minimize -t
subject to: for each active t_i: t + g_i @ (d_pos - d_neg) <= C - profile[t_i]
            sum(d_pos) - sum(d_neg) = 0
            0 <= d_pos[i] <= min(delta, 1-h[i])
            0 <= d_neg[i] <= min(delta, h[i])
            t >= 0

Variables: [t, d_pos(0), ..., d_pos(n-1), d_neg(0), ..., d_neg(n-1)]
Count: 1 + 2n

import numpy as np

def exact_objective(h):
    n = len(h)
    if n == 0:
        return float('inf')
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_box_sum(h, target=None):
    n = len(h)
    h = np.array(h, dtype=np.float64).copy()
    if target is None:
        target = n / 2.0
    h = np.nan_to_num(h, nan=0.0, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = h.sum() - target
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            m = h > 1e-15
            if not m.any():
                break
            s = min(diff / m.sum(), h[m].min() * 0.999)
            h[m] -= s
        else:
            m = h < 1.0 - 1e-15
            if not m.any():
                break
            s = min(-diff / m.sum(), (1.0 - h[m]).max() * 0.999)
            h[m] += s
    diff = h.sum() - target
    if abs(diff) > 1e-13:
        m = (h > 1e-15) if diff > 0 else (h < 1.0 - 1e-15)
        if m.any():
            h[m] -= diff / m.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def grad_profile(h, t_idx):
    n = len(h)
    dx = 2.0 / n
    k = t_idx - (n - 1)
    g = np.zeros(n)
    if k >= 0:
        g[:n - k] += (1.0 - h[k:n])
        g[k:] -= h[:n - k]
    else:
        a = -k
        g[:n - a] += (1.0 - h[a:n])
        g[a:] -= h[:n - a]
    return g * dx

def active_set_lp(h, active_t, delta=0.03):
    n = len(h)
    profile = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    C = profile.max()
    nv = 1 + 2 * n
    c_obj = np.zeros(nv)
    c_obj[0] = -1.0
    A_rows = []
    b_vals = []
    for t_i in active_t:
        g = grad_profile(h, t_i)
        row = np.zeros(nv)
        row[0] = 1.0
        row[1:1 + n] = g
        row[1 + n:] = -g
        A_rows.append(row)
        b_vals.append(C - profile[t_i])
    if not A_rows:
        return None, 0.0
    A_ub = np.array(A_rows)
    b_ub = np.array(b_vals)
    A_eq = np.zeros((1, nv))
    A_eq[0, 1:1 + n] = 1.0
    A_eq[0, 1 + n:] = -1.0
    bounds = [(0, None)]
    for i in range(n):
        bounds.append((0, min(delta, 1.0 - h[i])))
        bounds.append((0, min(delta, h[i])))
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq,
                      b_eq=np.zeros(1), bounds=bounds, method='highs')
        if res.success and res.x[0] > 1e-9:
            d = res.x[1:1 + n] - res.x[1 + n:]
            return d, res.x[0]
    except Exception:
        pass
    return None, 0.0

def transfer_to_size(parent, target_n):
