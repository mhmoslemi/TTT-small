def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def project_feasible(z):
    """Project z to {h: 0<=h_i<=1, sum(h)=n/2} using water-filling."""
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=np.float64).copy()
    
    def F(a):
        return np.sum(np.clip(z + a, 0.0, 1.0))
    
    # Find bracket
    a_low = -z.max() - 2.0
    a_high = 1.0 - z.min() + 2.0
    for _ in range(50):
        if F(a_low) > target:
            a_low -= 1.0
        else:
            break
    for _ in range(50):
        if F(a_high) < target:
            a_high += 1.0
        else:
            break
    
    # Bisection
    for _ in range(80):
        a_mid = 0.5 * (a_low + a_high)
        if F(a_mid) < target:
            a_low = a_mid
        else:
            a_high = a_mid
        if a_high - a_low < 1e-15:
            break
    
    a = 0.5 * (a_low + a_high)
    h = np.clip(z + a, 0.0, 1.0)
    
    # Fix residual
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            # Add to indices with most capacity
            order = np.argsort(1.0 - h)
            for i in order:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add

def build_gradient(h):
    """Build gradient matrix G where G[m, j] = d r_m / d h_j."""
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    G = np.zeros((M, n))
    mid = n - 1
    
    for m in range(M):
        d = m - mid
        if d == 0:
            G[m, :] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            G[m, d:] = dx * (1.0 - h[:n-d])
            G[m, :n-d] += -dx * h[d:]
        else:
            s = -d
            G[m, :n-s] = dx * (1.0 - h[s:])
            G[m, s:] += -dx * h[:n-s]
    
    return G

    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)

    """Smooth log-sum-exp gradient descent with temperature annealing."""
    n = len(h)
    h = h.copy()
    best_h = h.copy()

    taus = np.geomspace(tau0, tau_min, num_taus)
    
    for tau in taus:
        if time_limit is not None:
            elapsed = time.perf_counter() - t_start
            if elapsed > time_limit:
                break
        for step in range(n_steps_per_tau):
            if time_limit is not None:
                elapsed = time.perf_counter() - t_start
                if elapsed > time_limit:
                    break
            
            profile = compute_profile(h)
            max_r = profile.max()
            # Soft weights (numerically stable)
            shifted = (profile - max_r) / tau
            exp_shifted = np.exp(shifted)
            w = exp_shifted / exp_shifted.sum()

            G = build_gradient(h)
            grad = G.T @ w
            grad -= grad.mean()  # maintain sum constraint
            
            norm = np.linalg.norm(grad)
            if norm < 1e-14:
                break
            grad /= norm

            f_current = tau * np.log(exp_shifted.sum())  # smooth objective
            accepted = False

def active_set_descent(h, max_iters=200, time_limit=None, t_start=None):
    """Active-set subgradient descent with adaptive weights."""
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best_c = exact_score(h)
    
    for it in range(max_iters):
        if time_limit is not None:

            if elapsed > time_limit:
                break
        
        profile = compute_profile(h)
        c = profile.max()
        
        # Active set: all indices within epsilon of max
        eps = 1e-7
        active = np.where(profile >= c - eps)[0]
        
        # Also include "near-active" for smoother gradient
        near_active = np.where(profile >= c - 1e-3)[0]
        
        G = build_gradient(h)
        
        # Try different weight schemes
        best_step = None
        best_score = c
        
        for active_set, name in [(active, 'tight'), (near_active, 'wide')]:
            if len(active_set) == 0:
                active_set = np.array([np.argmax(profile)])
            G_sub = G[active_set, :]
            
            # Weight 1: uniform
            w1 = np.ones(len(active_set)) / len(active_set)
            g1 = G_sub.T @ w1
            g1 -= g1.mean()
            
            # Weight 2: proportional to (c - profile[active]) + small
            diff = c - profile[active_set]
            w2 = (diff + 1e-6)
            w2 /= w2.sum()
            g2 = G_sub.T @ w2
            g2 -= g2.mean()
            
            for g in [g1, g2]:
                norm = np.linalg.norm(g)
                if norm < 1e-14:
                    continue

                break

def solve_trust_constr(h0, maxiter=500, tol=1e-10):
    """Solve using trust-constr method."""
    n = len(h0)
    dx = 2.0 / n
    M = 2 * n - 1
    c0 = exact_score(h0)
    t0 = c0 + 1e-4
    
    x0 = np.concatenate([h0, [t0]])
    
    def obj(x):
        return x[-1]
    
    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    
    def cons_fun(x):
        h = x[:-1]
        t = x[-1]
        profile = compute_profile(h)
        return t - profile
    
    def cons_jac(x):
        h = x[:-1]
        G = build_gradient(h)
        J = np.zeros((M, n + 1))
        J[:, :-1] = -G
        J[:, -1] = 1.0
        return J
    
    constraints = [
        {'type': 'eq', 'fun': lambda x: x[:-1].sum() - n/2.0,
         'jac': lambda x: np.concatenate([np.ones(n), [0.0]])},
        {'type': 'ineq', 'fun': cons_fun, 'jac': cons_jac}
    ]
    
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 1.0)]
    
    try:

import numpy as np
from scipy.optimize import minimize
import time


def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))


def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx


def project_feasible(z):
    n = len(z)
    target = n / 2.0
    z = np.asarray(z, dtype=np.float64).copy()

    def F(a):
        return np.sum(np.clip(z + a, 0.0, 1.0))

    a_low = -z.max() - 3.0
    a_high = 1.0 - z.min() + 3.0
    for _ in range(60):
        if F(a_low) > target:
            a_low *= 1.5
        else:
            break
    for _ in range(60):
        if F(a_high) < target:
            a_high = 1.0 + a_high * 0.5
        else:
            break

    for _ in range(80):
        a_mid = 0.5 * (a_low + a_high)
        if F(a_mid) < target:
            a_low = a_mid
        else:
            a_high = a_mid
        if a_high - a_low < 1e-15:
            break

    a = 0.5 * (a_low + a_high)
    h = np.clip(z + a, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            order = np.argsort(1.0 - h)
            for i in order:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
                if diff < 1e-15:
                    break
        else:
            order = np.argsort(h)
            for i in order:
                rem = min(-diff, h[i])
                h[i] -= rem
                diff += rem
                if diff > -1e-15:
                    break
    return np.clip(h, 0.0, 1.0)


def build_gradient(h):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    mid = n - 1
    G = np.zeros((M, n))
    for m in range(M):
        d = m - mid
        if d == 0:
            G[m, :] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            G[m, d:n] += dx * (1.0 - h[0:n - d])
            G[m, 0:n - d] += -dx * h[d:n]
        else:
            s = -d
            G[m, 0:n - s] += dx * (1.0 - h[s:n])
            G[m, s:n] += -dx * h[0:n - s]
    return G


def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    return project_feasible(h_new)


def smooth_descent(h, tau0=0.05, tau_min=0.002, n_steps=30, num_taus=10,
                   t_start=0.0, time_limit=1000.0):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c = exact_score(h)
    taus = np.geomspace(tau0, tau_min, num_taus)
    for tau in taus:
        if time.perf_counter() - t_start > time_limit:
            break
        for _ in range(n_steps):
            if time.perf_counter() - t_start > time_limit:
                break
            profile = compute_profile(h)
            max_r = profile.max()
            shifted = (profile - max_r) / tau
            exp_s = np.exp(shifted)
            w = exp_s / exp_s.sum()
            G = build_gradient(h)
            grad = G.T @ w
            grad -= grad.mean()
            norm = np.linalg.norm(grad)
            if norm < 1e-14:
                break
            grad /= norm
            accepted = False
            for alpha in [0.05, 0.03, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]:
                h_try = project_feasible(h - alpha * grad)
                c_try = exact_score(h_try)
                if c_try < best_c - 1e-13:
                    h = h_try
                    best_h = h_try.copy()
                    best_c = c_try
                    accepted = True
                    break
                c_cur = exact_score(h)
                if c_try < c_cur - 1e-12:
                    h = h_try
                    if c_try < best_c:
                        best_h = h_try.copy()
                        best_c = c_try
                    accepted = True
                    break
            if not accepted:
                break
    return best_h, best_c


def active_set_descent(h, max_iters=150, t_start=0.0, time_limit=1000.0):
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)

        profile = compute_profile(h)

        G = build_gradient(h)
        best_step = None
        best_score = c

            G_sub = G[active, :]

                g = G_sub.T @ wset

def solve_nlp(h0, method='slsqp', maxiter=300, tol=1e-10):
    n = len(h0)
    c0 = exact_score(h0)
    t0 = c0 + 1e-4
    x0 = np.concatenate([h0, [t0]])
    prob = MinimaxNLP(n)
    constraints = [
        {'type': 'eq', 'fun': prob.eq_fun, 'jac': prob.eq_jac},
        {'type': 'ineq', 'fun': prob.ineq_fun, 'jac': prob.ineq_jac}
    ]
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 1.0)]
    try:
        if method == 'trust-constr':
            res = minimize(prob.obj, x0, method='trust-constr', jac=prob.obj_jac,
                           bounds=bounds, constraints=constraints,
                           options={'maxiter': maxiter, 'gtol': tol, 'xtol': tol})
        else:
            res = minimize(prob.obj, x0, method='SLSQP', jac=prob.obj_jac,
                           bounds=bounds, constraints=constraints,
                           options={'maxiter': maxiter, 'ftol': tol})
        if res.x is not None and np.isfinite(res.x).all():
            h_new = project_feasible(res.x[:-1])
            c_new = exact_score(h_new)
            if c_new < c0 - 1e-12:
                return h_new
    except Exception:
        pass
    return None

            [d for d in range(max(40, n_cur - 12), min(201, n_cur + 16), 2)] +
            [d for d in [48, 56, 60, 64, 66, 70, 72, 76, 80, 88, 96, 100,
                         108, 120, 128, 144, 160, 180, 192, 200] if d % 2 == 0]
        ), key=lambda d: (abs(d - n_cur), d))

        for new_n in dim_candidates[:10]:
            if tl() < 40:
                break
            if new_n == n_cur:
                continue
            h_interp = interpolate_to_n(best_h, new_n)
            h_sm, c_sm = smooth_descent(h_interp, tau0=0.04, tau_min=0.003,
                                        n_steps=25, num_taus=8, t_start=t_start,
                                        time_limit=budget_s)
            if c_sm < best_c - 1e-12:
                best_h = h_sm
                best_c = c_sm
                n_cur = new_n

    # Stage 6: SLSQP polish
    if tl() > 50:
        h_sq = solve_nlp(best_h, method='slsqp', maxiter=300, tol=1e-10)
        if h_sq is not None:
            c_sq = exact_score(h_sq)
            if c_sq < best_c - 1e-13:
                best_h = h_sq
                best_c = c_sq

    # Stage 7: trust-constr polish
    if tl() > 60:
        h_tc = solve_nlp(best_h, method='trust-constr', maxiter=200, tol=1e-10)
        if h_tc is not None:
            c_tc = exact_score(h_tc)
            if c_tc < best_c - 1e-13:
                best_h = h_tc
                best_c = c_tc

    # Stage 8: Final active-set
    if tl() > 20:
        h_as, c_as = active_set_descent(best_h, max_iters=80, t_start=t_start,
                                        time_limit=budget_s)
        if c_as < best_c - 1e-13:
            best_h = h_as
            best_c = c_as

    # Stage 9: Final fine smooth descent
    if tl() > 30:
        h_sm, c_sm = smooth_descent(best_h, tau0=0.01, tau_min=0.001,
                                    n_steps=30, num_taus=6, t_start=t_start,
                                    time_limit=budget_s)
        if c_sm < best_c - 1e-13:
            best_h = h_sm
            best_c = c_sm

    # Stage 10: One more SLSQP if time permits
    if tl() > 40:
        h_sq = solve_nlp(best_h, method='slsqp', maxiter=400, tol=1e-11)
        if h_sq is not None:
            c_sq = exact_score(h_sq)
            if c_sq < best_c - 1e-13:
                best_h = h_sq
                best_c = c_sq

    best_h = project_feasible(best_h)
    final_c = exact_score(best_h)

    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            final_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 2:
            final_c = float(eval_result[0])
    except Exception:
        pass

    return best_h, final_c, len(best_h)
