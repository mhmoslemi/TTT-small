def exact_objective(h):
    n = len(h)
    if n == 0:
        return float('inf')
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(profile))

def project_box_sum(h):
    h = np.array(h, dtype=np.float64).copy()
    n = len(h)
    target = n / 2.0
    h[~np.isfinite(h)] = 0.0
    for _ in range(200):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            h[mask] = np.minimum(1.0, h[mask] + diff / cnt)
        else:
            mask = h > 1e-15
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            h[mask] = np.maximum(0.0, h[mask] + diff / cnt)
    return np.clip(h, 0.0, 1.0)

def build_anti(params):
    """Build full h from anti-symmetric parameters."""
    m = len(params)
    n = 2 * m
    h = np.empty(n)
    for i in range(m):
        h[i] = params[i]
        h[n-1-i] = 1.0 - params[i]
    return h

    h = build_anti(params)

    """Cyclic coordinate descent on anti-symmetric parameters."""
    m = len(params)
    n = 2 * m
    dx = 2.0 / n
    params = np.array(params, dtype=np.float64).copy()
    curr_c = anti_objective(params)

                params[j] = x
                h = build_anti(params)

def exact_objective(h):
    n = len(h)
    if n == 0:
        return float('inf')
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(profile))

def project_box_sum(h):
    h = np.array(h, dtype=np.float64).copy()
    n = len(h)
    target = n / 2.0
    h[~np.isfinite(h)] = 0.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            h[mask] = np.minimum(1.0, h[mask] + diff / cnt)
        else:
            mask = h > 1e-15
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            h[mask] = np.maximum(0.0, h[mask] + diff / cnt)
    return np.clip(h, 0.0, 1.0)

def build_anti(params):
    m = len(params)
    n = 2 * m
    h = np.empty(n)
    for i in range(m):
        h[i] = params[i]
        h[n - 1 - i] = 1.0 - params[i]
    return h

def anti_obj(params):

    return exact_objective(h)

def anti_profile(params):

            new_val = float(np.clip(res.x, 0.0, 1.0))

        else:

def golden_section_1d(eval_func, a, b, tol=1e-9, maxiter=50):
    """Golden section minimization. eval_func takes a scalar x and returns a scalar."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = eval_func(c)
    fd = eval_func(d)
    for _ in range(maxiter):
        if b - a < tol:
            break
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = eval_func(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = eval_func(d)
    x_opt = (a + b) / 2.0
    return x_opt
