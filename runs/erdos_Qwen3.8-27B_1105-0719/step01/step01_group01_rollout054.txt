def project_feasible(y, n):
    target = n / 2.0
    lo = np.min(y) - 2.0
    hi = np.max(y) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    h = np.clip(y - t, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-10:
        eligible = np.where((h > 1e-6) & (h < 1.0 - 1e-6))[0]
        if len(eligible) > 0:

    return h

def corr_values(h):
    return np.correlate(h, 1.0 - h, mode="full")

def exchange_line_search(h, i, j, cap, current_c5):
    """Find best tau for exchange: h[i] += tau, h[j] -= tau"""
    n = len(h)
    dx = 2.0 / n
    # Fine grid search
    grid = np.linspace(0, cap, 21)
    best_tau = 0
    best_score = current_c5
    for tau in grid:
        if tau < 1e-12:
            continue
        h[i] += tau
        h[j] -= tau
        score = np.max(corr_values(h)) * dx
        if score < best_score:
            best_score = score
            best_tau = tau
        h[i] -= tau
        h[j] += tau
    # Refine around best with golden section
    if best_tau > 0:
        lo_t = max(0, best_tau - cap/20)
        hi_t = min(cap, best_tau + cap/20)
        gr = 0.618
        for _ in range(30):
            m1 = hi_t - gr * (hi_t - lo_t)
            m2 = lo_t + gr * (hi_t - lo_t)
            h[i] += m1; h[j] -= m1
            s1 = np.max(corr_values(h)) * dx
            h[i] -= m1; h[j] += m1
            h[i] += m2; h[j] -= m2
            s2 = np.max(corr_values(h)) * dx
            h[i] -= m2; h[j] += m2
            if s1 < s2:
                hi_t = m2
            else:
                lo_t = m1
        best_tau = (lo_t + hi_t) / 2
        h[i] += best_tau; h[j] -= best_tau
        best_score = np.max(corr_values(h)) * dx

def compute_exchange_update(h, i, j):
    """Precompute coefficients for the exchange h[i]+=tau, h[j]-=tau"""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    # corr is indexed from 0 to 2n-2, corresponding to lag from -(n-1) to n-1
    # lag = idx - (n-1)
    d = j - i
    # linear coefficients for each lag
    # linear[lag] = (1-h[i+lag])*[0<=i+lag<n] - (1-h[j+lag])*[0<=j+lag<n] 
    #              - h[i-lag]*[0<=i-lag<n] + h[j-lag]*[0<=i-lag<n]
    n_lags = 2*n - 1
    lags = np.arange(-(n-1), n)
    linear = np.zeros(n_lags)
    # (1-h[i+lag]) where 0<=i+lag<n
    valid1 = (i + lags >= 0) & (i + lags < n)
    linear[valid1] += (1.0 - h[i + lags[valid1]])
    # -(1-h[j+lag]) where 0<=j+lag<n
    valid2 = (j + lags >= 0) & (j + lags < n)
    linear[valid2] -= (1.0 - h[j + lags[valid2]])
    # -h[i-lag] where 0<=i-lag<n
    valid3 = (i - lags >= 0) & (i - lags < n)
    linear[valid3] -= h[i - lags[valid3]]
    # +h[j-lag] where 0<=j-lag<n
    valid4 = (j - lags >= 0) & (j - lags < n)
    linear[valid4] += h[j - lags[valid4]]
    # Quadratic correction: -1 for lags == d and lags == -d
    quad = np.zeros(n_lags)
    idx_d = d + (n-1)  # index in corr array for lag=d
    idx_neg_d = -d + (n-1)  # index in corr array for lag=-d
    quad[idx_d] = -1.0
    quad[idx_neg_d] = -1.0
    if d == 0:
        quad[idx_d] = -1.0  # same index, don't double count (but d!=0 since i!=j)
    return corr, linear, quad

def simulated_annealing(h, c5, n_moves, t0, time_limit, rng):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c5 = c5
    current_c5 = c5
    
    T_start = 0.01
    T_end = 0.0001
    log_ratio = np.log(T_end / T_start)
    
    for move in range(n_moves):
        if time.time() - t0 > time_limit:
            break
        progress = move / n_moves
        T = T_start * np.exp(log_ratio * progress)
        
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        
        # Random direction and magnitude
        if rng.random() < 0.5:
            cap = min(1.0 - h[i], h[j])
        else:
            cap = min(1.0 - h[j], h[i])
            i, j = j, i  # swap to always have h[i] increase
            cap = min(1.0 - h[i], h[j])
        
        if cap < 1e-10:
            continue
        
        # Random tau (biased towards larger moves at high T)
        tau = cap * rng.power(0.5 + 2.0 * (1.0 - progress))
        if tau < 1e-10:
            continue
        
        h[i] += tau
        h[j] -= tau
        new_c5 = compute_c5(h)
        
        delta = new_c5 - current_c5
        if delta < 0 or (T > 0 and rng.random() < np.exp(-delta / T)):
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[i] -= tau
            h[j] += tau
    
    return best_h, best_c5

def resample(h_old, n_new):
    """Resample h from n_old points to n_new points preserving integral."""
    n_old = len(h_old)
    dx_old = 2.0 / n_old
    dx_new = 2.0 / n_new
    
    # Build cumulative integral of h_old
    # h_old[i] represents the value on [i*dx_old, (i+1)*dx_old)
    # Integral up to position x: sum of h_old[i]*dx_old for i < x/dx_old
    
    # For new point j, it represents [j*dx_new, (j+1)*dx_new)
    # We want the average value of h on this interval
    
    # Build the step function as a dense array (for simplicity, if n is small)
    # Or use the overlap formula directly
    
    h_new = np.zeros(n_new)
    for j in range(n_new):
        x_start = j * dx_new
        x_end = (j + 1) * dx_new
        total = 0.0
        length = 0.0
        for i in range(n_old):
            x0 = i * dx_old
            x1 = (i + 1) * dx_old
            # Overlap of [x0, x1) and [x_start, x_end)
            lo = max(x0, x_start)
            hi = min(x1, x_end)
            if hi > lo:
                total += h_old[i] * (hi - lo)
                length += (hi - lo)
        if length > 0:
            h_new[j] = total / length
    
    # Project to feasible
    h_new = project_feasible(h_new)

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y):
    n = len(y)
    target = n / 2.0
    y = np.array(y, dtype=float)
    if n == 0:
        return y
    lo = np.min(y) - 2.0
    hi = np.max(y) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    h = np.clip(y - t, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        eligible = np.where((h > 1e-8) & (h < 1.0 - 1e-8))[0]
        if len(eligible) > 0:
            h[eligible] += residual / len(eligible)
            h = np.clip(h, 0.0, 1.0)
        else:
            idx = int(np.argmax(h))
            h[idx] -= residual
            h = np.clip(h, 0.0, 1.0)
    return h

def pairwise_scores_c(corr, h, i, j, cap, n, dx, n_grid=40):
    d = j - i
    n_lags = 2 * n - 1
    lags = np.arange(-(n - 1), n)
    linear = np.zeros(n_lags)
    v1 = (i + lags >= 0) & (i + lags < n)
    linear[v1] += 1.0 - h[i + lags[v1]]
    v2 = (j + lags >= 0) & (j + lags < n)
    linear[v2] -= 1.0 - h[j + lags[v2]]
    v3 = (i - lags >= 0) & (i - lags < n)
    linear[v3] -= h[i - lags[v3]]
    v4 = (j - lags >= 0) & (j - lags < n)
    linear[v4] += h[j - lags[v4]]
    quad = np.zeros(n_lags)
    quad[d + (n - 1)] = -1.0
    quad[-d + (n - 1)] = -1.0
    taus = np.linspace(0.0, cap, n_grid)
    new_corrs = corr[None, :] + taus[:, None] * linear[None, :] + (taus ** 2)[:, None] * quad[None, :]
    scores = np.max(new_corrs, axis=1) * dx
    best_idx = int(np.argmin(scores))
    return scores[best_idx], taus[best_idx]

def sweep_exchange(h, max_sweeps=300, tol=1e-14, t0=None, time_limit=None):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best_c5 = compute_c5(h)
    for sweep in range(max_sweeps):
        if t0 is not None:
            elapsed = time.time() - t0
            if elapsed > time_limit:
                break
        corr = np.correlate(h, 1.0 - h, mode="full")
        best_score = best_c5
        best_move = None
        for i in range(n):
            for j in range(i + 1, n):
                cap = min(1.0 - h[i], h[j])
                if cap > 1e-10:
                    sc, tau = pairwise_scores_c(corr, h, i, j, cap, n, dx)
                    if sc < best_score - tol:
                        best_score = sc
                        best_move = (i, j, tau)
                cap = min(1.0 - h[j], h[i])
                if cap > 1e-10:
                    sc, tau = pairwise_scores_c(corr, h, j, i, cap, n, dx)
                    if sc < best_score - tol:
                        best_score = sc
                        best_move = (j, i, tau)
        if best_move is None:
            break
        ii, jj, tau = best_move
        h[ii] += tau
        h[jj] -= tau
        h = np.clip(h, 0.0, 1.0)
        best_c5 = compute_c5(h)
    return h, best_c5

def refined_exchange(h, t0, time_limit, n_refine=15):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best_c5 = compute_c5(h)
    while time.time() - t0 < time_limit:
        corr = np.correlate(h, 1.0 - h, mode="full")
        improved = False
        for i in range(n):
            for j in range(i + 1, n):
                for (ia, ib) in [(i, j), (j, i)]:
                    cap = min(1.0 - h[ia], h[ib])
                    if cap < 1e-10:
                        continue
                    sc, tau0 = pairwise_scores_c(corr, h, ia, ib, cap, n, dx, n_grid=60)
                    if sc >= best_c5:
                        continue
                    lo_t = max(0.0, tau0 - cap / 60.0)
                    hi_t = min(cap, tau0 + cap / 60.0)
                    best_tau = tau0
                    best_sc = sc
                    for _ in range(40):
                        m1 = hi_t - 0.618 * (hi_t - lo_t)
                        m2 = lo_t + 0.618 * (hi_t - lo_t)
                        s1, _ = pairwise_scores_c(corr, h, ia, ib, min(m1, cap), n, dx, n_grid=1)
                        s2, _ = pairwise_scores_c(corr, h, ia, ib, min(m2, cap), n, dx, n_grid=1)
                        # Need to evaluate at exact tau
                        d = ib - ia
                        n_lags = 2 * n - 1
                        lags = np.arange(-(n - 1), n)
                        lin = np.zeros(n_lags)
                        v1 = (ia + lags >= 0) & (ia + lags < n)
                        lin[v1] += 1.0 - h[ia + lags[v1]]
                        v2 = (ib + lags >= 0) & (ib + lags < n)
                        lin[v2] -= 1.0 - h[ib + lags[v2]]
                        v3 = (ia - lags >= 0) & (ia - lags < n)
                        lin[v3] -= h[ia - lags[v3]]
                        v4 = (ib - lags >= 0) & (ib - lags < n)
                        lin[v4] += h[ib - lags[v4]]
                        qu = np.zeros(n_lags)
                        qu[d + (n - 1)] = -1.0
                        qu[-d + (n - 1)] = -1.0
                        for m_val in [m1, m2]:
                            nc = corr + m_val * lin + m_val * m_val * qu
                            sv = float(np.max(nc) * dx)
                            if m_val == m1:
                                s1 = sv
                            else:
                                s2 = sv
                        if s1 < s2:
                            hi_t = m2
                        else:
                            lo_t = m1
                    best_tau = (lo_t + hi_t) / 2.0
                    h[ia] += best_tau
                    h[ib] -= best_tau
                    h = np.clip(h, 0.0, 1.0)
                    new_c5 = compute_c5(h)
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        improved = True
                        corr = np.correlate(h, 1.0 - h, mode="full")
                    else:
                        h[ia] -= best_tau
                        h[ib] += best_tau
                        h = np.clip(h, 0.0, 1.0)
        if not improved:
            break
    return h, best_c5

def simulated_annealing(h, c5, t0, time_limit, rng):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c5 = c5
    current_c5 = c5
    n_moves = int(max(1000, (time_limit - (time.time() - t0)) * 8000))
    T_start = 0.008
    T_end = 0.00005
    log_ratio = np.log(T_end / T_start)
    for move in range(n_moves):
        if time.time() - t0 > time_limit:
            break
        progress = min(1.0, move / max(1, n_moves))
        T = T_start * np.exp(log_ratio * progress)
        i = int(rng.randint(n))
        j = int(rng.randint(n))
        if i == j:
            continue
        if rng.random() < 0.5:
            cap = min(1.0 - h[i], h[j])
        else:
            cap = min(1.0 - h[j], h[i])
            i, j = j, i
            cap = min(1.0 - h[i], h[j])
        if cap < 1e-10:
            continue
        tau = cap * float(rng.power(0.5 + 1.5 * (1.0 - progress) * 0.5))
        if tau < 1e-10:
            continue
        h[i] += tau
        h[j] -= tau
        new_c5 = compute_c5(h)
        delta = new_c5 - current_c5
        if delta < 0 or (T > 1e-15 and rng.random() < np.exp(-delta / T)):
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[i] -= tau
            h[j] += tau
    return best_h, best_c5

def resample(h_old, n_new):
    n_old = len(h_old)
    dx_old = 2.0 / n_old
    dx_new = 2.0 / n_new
    h_new = np.zeros(n_new)
    for j in range(n_new):
        xs = j * dx_new
        xe = (j + 1) * dx_new
        total = 0.0
        length = 0.0
        for i in range(n_old):
            x0 = i * dx_old
            x1 = (i + 1) * dx_old
            lo = max(x0, xs)
            hi = min(x1, xe)
            if hi > lo:
                total += h_old[i] * (hi - lo)
                length += (hi - lo)
        if length > 0:
            h_new[j] = total / length
    return project_feasible(h_new)

def make_binary_seed(n, rng):
    h = np.zeros(n)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    residual = n / 2.0 - np.sum(h)
    if abs(residual) > 1e-10:
        h[0] += residual
        h = np.clip(h, 0.0, 1.0)
    return h

def make_sine_seed(n, k, phase):
    i = np.arange(n)
    h = 0.5 + 0.5 * np.sin(2.0 * np.pi * k * i / n + phase)
    return project_feasible(h)

        # Follow up with steepest descent
        if time.time() < deadline - 15:
