20. **For the binary local search**, the key move is a "swap": change h[j] from 1 to 0 and h[j'] from 0 to 1.
    
    The change in overlap at shift s is:
    Δoverlap(s) = [h[j+s] + h[j-s] - h[j'+s] - h[j'-s]]
    (with boundary handling: terms are 0 if index out of range)
    
    We want to find swaps that reduce the maximum overlap.
    
    A simple greedy approach:
    1. Find the shift s* with maximum overlap
    2. Try all possible swaps and find the one that most reduces overlap(s*)
    3. Apply it, repeat
    
    But trying all swaps is O(n²) per iteration, and we might need many iterations.
    
    A faster approach:
    1. Find s* with max overlap
    2. The overlap at s* is: #{i: h[i]=1, h[i+s*]=0}
    3. To reduce this, we want to either:
       a. Change some h[i]=1 (where h[i+s*]=0) to 0 (and compensate elsewhere)
       b. Change some h[i+s*]=0 (where h[i]=1) to 1 (and compensate elsewhere)
    4. But we need to maintain the sum constraint, so we always do pairs (swaps).

    Let me think about a more efficient local search:
    
    For a given "bad" position i (where h[i]=1 and h[i+s*]=0):
    - If we swap h[i] with h[j] (where h[j]=0), the effect on overlap(s*) is:
      - We remove the contribution at position i: overlap decreases by 1 (at shift s*)

                h[i], h[j] = 0.0, 1.0
                corrs = compute_corrs(h)
                max_c = corrs.max()
                if max_c < best_max:
                    best_max = max_c
                    best_h = h.copy()
                    no_improve = 0
                else:
                    h[i], h[j] = 1.0, 0.0  # undo
            continue
        
        i = ones[best_i_idx]
        j = zeros[best_j_idx]
        h[i], h[j] = 0.0, 1.0
        
        # Recompute correlations
        corrs = compute_corrs(h)
        max_c = corrs.max()
        
        if max_c < best_max:
            best_max = max_c
            best_h = h.copy()
            no_improve = 0
        elif max_c == best_max:
            no_improve += 1
        else:
            # Undo
            h[i], h[j] = 1.0, 0.0
            no_improve += 1
            # Try random swap
            if rng.random() < 0.3:
                ii = rng.randint(len(ones))
                jj = rng.randint(len(zeros))
                ii2 = ones[ii]
                jj2 = zeros[jj]
                h[ii2], h[jj2] = 0.0, 1.0
                corrs_new = compute_corrs(h)
                max_new = corrs_new.max()
                if max_new <= best_max:
                    best_max = max_new
                    best_h = h.copy()
                    corrs = corrs_new
                    max_c = max_new
                    no_improve = 0
                else:
                    h[ii2], h[jj2] = 1.0, 0.0
    
    return best_h, best_max * dx

import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time


def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)


def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def c5_of_h(h, dx):
    return compute_corrs(h).max() * dx


def binary_local_search(h, dx, target, max_iters, rng, record_iters=50000):
    n = len(h)
    h = h.copy()
    corrs = compute_corrs(h)
    max_c = corrs.max()
    best_h = h.copy()
    best_max = max_c
    no_improve = 0
    best_c5 = max_c * dx

    for it in range(max_iters):
        if no_improve > record_iters:
            break
        if it % 20000 == 0 and it > 0:
            pass

        worst_idx = corrs.argmax()
        s = worst_idx - (n - 1)

        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        A = np.zeros(len(ones))
        for idx in range(len(ones)):
            i = ones[idx]
            ip = i + s
            im = i - s
            if 0 <= ip < n:
                A[idx] -= (1.0 - h[ip])
            if 0 <= im < n:
                A[idx] += h[im]

        B = np.zeros(len(zeros))
        for idx in range(len(zeros)):
            j = zeros[idx]
            jp = j + s
            jm = j - s
            if 0 <= jp < n:
                B[idx] += (1.0 - h[jp])
            if 0 <= jm < n:
                B[idx] -= h[jm]

        best_i = ones[A.argmin()]
        best_j = zeros[B.argmin()]
        best_delta = A.min() + B.min()

        if best_delta >= 0:
            no_improve += 1
            if no_improve % 1000 == 0:
                for _ in range(20):
                    ii = rng.randint(len(ones))
                    jj = rng.randint(len(zeros))
                    h[ones[ii]], h[zeros[jj]] = 0.0, 1.0
                    c2 = compute_corrs(h)
                    m2 = c2.max()
                    if m2 < best_max:
                        best_max = m2
                        best_h = h.copy()
                        corrs = c2
                        no_improve = 0
                        break
                    h[ones[ii]], h[zeros[jj]] = 1.0, 0.0
            if no_improve > record_iters:
                break
            continue

        h[best_i] = 0.0
        h[best_j] = 1.0
        corrs = compute_corrs(h)
        max_c = corrs.max()

        if max_c < best_max:
            best_max = max_c
            best_h = h.copy()
            no_improve = 0
            best_c5 = max_c * dx
        else:
            h[best_i] = 1.0
            h[best_j] = 0.0
            no_improve += 1

            if best_delta <= -1 and rng.random() < 0.2:
                for _ in range(30):
                    order = np.argsort(A)
                    ii = order[rng.randint(min(5, len(order)))]
                    order_b = np.argsort(B)
                    jj = order_b[rng.randint(min(5, len(order_b)))]
                    i2 = ones[ii]
                    j2 = zeros[jj]
                    if i2 == best_i and j2 == best_j:
                        continue
                    h[i2] = 0.0
                    h[j2] = 1.0
                    c2 = compute_corrs(h)
                    m2 = c2.max()
                    if m2 < best_max:
                        best_max = m2
                        best_h = h.copy()
                        corrs = c2
                        no_improve = 0
                        best_c5 = m2 * dx
                        break
                    h[i2] = 1.0
                    h[j2] = 0.0

    return best_h, best_c5


def smooth_max_obj(h, dx, beta):
    corrs = compute_corrs(h)
    return logsumexp(beta * corrs) / beta * dx


def smooth_max_grad(h, dx, beta):
    n = len(h)
    corrs = compute_corrs(h)
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones_like(corrs) / len(corrs)
    else:
        w /= w_sum
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        offset = k - (n - 1)
        i_lo = max(0, -offset)
        i_hi = min(n - 1, n - 1 - offset)
        if i_lo > i_hi:
            continue
        idx = np.arange(i_lo, i_hi + 1)
        grad[idx + offset] += (1.0 - h[idx])
        grad[idx] += (-h[idx + offset])
        grad *= 0
    # Recompute properly
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        offset = k - (n - 1)
        i_lo = max(0, -offset)
        i_hi = min(n - 1, n - 1 - offset)
        if i_lo > i_hi:
            continue
        idx = np.arange(i_lo, i_hi + 1)
        grad[idx] += w[k] * (1.0 - h[idx + offset])
        grad[idx + offset] += w[k] * (-h[idx])
    return grad * dx


def slsqp_polish(h0, dx, target, beta, maxiter):
    n = len(h0)

    def obj_func(x):
        return smooth_max_obj(x, dx, beta)

    def obj_grad(x):
        return smooth_max_grad(x, dx, beta)

    def constr_fun(x):
        return x.sum() - target

    def constr_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constr_fun, 'jac': constr_jac}]
    try:
        res = minimize(obj_func, h0, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        h_out = project_feasible(res.x, target)
    except Exception:
        h_out = project_feasible(h0, target)
    return h_out, c5_of_h(h_out, dx)


def subgradient_descent(h0, dx, target, max_iter, tol=1e-10):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    alpha = 0.3
    no_improve = 0

    for it in range(max_iter):
        corrs = compute_corrs(h)
        max_c = corrs.max()
        c5_val = max_c * dx

        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 3000:
            break

        threshold = max_c * 0.95
        active = np.where(corrs >= threshold)[0]
        grad = np.zeros(n)
        for k in active:
            offset = k - (n - 1)
            i_lo = max(0, -offset)
            i_hi = min(n - 1, n - 1 - offset)
            if i_lo > i_hi:
                continue
            idx = np.arange(i_lo, i_hi + 1)
            grad[idx] += (1.0 - h[idx + offset])
            grad[idx + offset] += (-h[idx])
        grad /= len(active)
        grad *= dx

        h_new = project_feasible(h - alpha * grad, target)
        new_c5 = c5_of_h(h_new, dx)

        if new_c5 < c5_val:
            h = h_new
            alpha = min(alpha * 1.01, 0.2)
        else:
            alpha *= 0.97

        if alpha < tol:
            break

    return best_h, best_c5


def make_binary_start(n, target, rng):
    h = np.zeros(n)
    idx = rng.permutation(n)
    n_ones = int(target)
    if n_ones > 0 and n_ones <= n:
        h[idx[:n_ones]] = 1.0
    frac = target - n_ones
    if frac > 1e-10 and n_ones < n:
        h[idx[n_ones]] = frac
    return h


def make_structured_binary(n, target, mode=0):
    h = np.zeros(n)
    n_ones = int(target)
    if mode == 0:
        # Alternating with perturbation
        h[0::2] = 1.0
        if target != n // 2:
            h[1::2][:int(target - n // 2)] = 1.0
            h[::2][int(target - n // 2):] *= (1.0 if int(target - n//2) < n//2 - (target - n//2) else 1.0)
        # Just ensure correct count
        ones = np.where(h == 1)[0]
        if len(ones) > n_ones:
            h[ones[n_ones:]] = 0.0
        elif len(ones) < n_ones:
            zeros = np.where(h == 0)[0]
            h[zeros[:n_ones - len(ones)]] = 1.0
    elif mode == 1:
        # Block structure
        block = 4
        for i in range(0, n, block):
            h[i:i+2] = 1.0
        ones = np.where(h == 1)[0]
        if len(ones) > n_ones:
            h[ones[n_ones:]] = 0.0
        elif len(ones) < n_ones:
            zeros = np.where(h == 0)[0]
            h[zeros[:n_ones - len(ones)]] = 1.0
    elif mode == 2:
        # Quadratic residue-like
        for i in range(n):
            if (i * i) % 4 < 2:
                h[i] = 1.0
        ones = np.where(h == 1)[0]
        if len(ones) > n_ones:
            h[ones[n_ones:]] = 0.0
        elif len(ones) < n_ones:
            zeros = np.where(h == 0)[0]
            h[zeros[:n_ones - len(ones)]] = 1.0
    return h


def remaining_time(t0, budget_s):
    return budget_s - (time.time() - t0)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    def update_best(h, c5, n):
        nonlocal best_h, best_c5, best_n
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_n = n

    # Phase 1: Binary local search on promising n values
    n_candidates = [128, 130, 126, 132, 124, 120, 136, 140]

        dx_try = 2.0 / n_try
        target_try = n_try / 2.0

            h_init = project_feasible(np.array(initial_h_values, dtype=float), target_try)
            c5_init = c5_of_h(h_init, dx_try)
            if c5_init < best_c5:
                update_best(h_init, c5_init, n_try)

        # Multiple random restarts of binary local search
        n_restarts = 8 if n_try in [128, 130, 126] else 4
        if remaining_time(t0, budget_s) < 15:
            n_restarts = 1

        for r in range(n_restarts):
            if remaining_time(t0, budget_s) < 10:
                break

            if r == 0:
                h_start = make_structured_binary(n_try, target_try, mode=r % 3)
            elif r == 1:
                h_start = make_structured_binary(n_try, target_try, mode=(r + 1) % 3)
            else:
                h_start = make_binary_start(n_try, target_try, rng)

            h_bin, c5_bin = binary_local_search(h_start, dx_try, target_try,

        target_b = n_best / 2.0
