import time

def exact_score(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))

def get_profile(h):
    dx = 2.0 / len(h)
    return np.correlate(h, 1.0 - h, mode="full") * dx

def repair_feasible(h, target=None):
    h = np.array(h, dtype=np.float64).copy()
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    if target is None:
        target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        order = np.argsort(-room)
        remaining = diff
        for idx in order:
            if remaining <= 1e-15:
                break
            add = min(remaining, room[idx])
            h[idx] += add
            remaining -= add
    else:
        order = np.argsort(-h)
        remaining = -diff
        for idx in order:
            if remaining <= 1e-15:
                break
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
    return h

def feasible_interval(h, d):
    """Find the interval [t_lo, t_hi] such that h + t*d is in [0,1]^n."""
    n = len(h)
    t_lo = -np.inf
    t_hi = np.inf
    for i in range(n):
        if d[i] > 1e-15:
            t_lo = max(t_lo, -h[i] / d[i])
            t_hi = min(t_hi, (1.0 - h[i]) / d[i])
        elif d[i] < -1e-15:
            t_lo = max(t_lo, (1.0 - h[i]) / d[i])
            t_hi = min(t_hi, -h[i] / d[i])
    if t_lo >= t_hi:
        return None
    return t_lo, t_hi

    n = len(h)
    interval = feasible_interval(h, d)
    if interval is None:
        return None
    t_lo, t_hi = interval
    
    if t_hi - t_lo < 1e-10:
        return None
    
    if eps is None:
        eps = 0.01 * (t_hi - t_lo) + 1e-6

    t_lo = min(t_lo, -1e-6)
    t_hi = max(t_hi, 1e-6)
    
    # Fit quadratics using 3 evaluations
    h_c = h  # center
    h_p = h + eps * d  # plus (may be infeasible, but that's OK for fitting)
    h_m = h - eps * d  # minus (may be infeasible)

    prof_p = get_profile(h_p)
    prof_m = get_profile(h_m)

    # c_k = prof_c[k]
    # a_k = (prof_p[k] - 2*prof_c[k] + prof_m[k]) / (2*eps^2)
    # b_k = (prof_p[k] - prof_m[k]) / (2*eps)

    candidates.append(max(t_lo + 1e-6, 0.0))

                s = exact_score(h_trial)
                if s < best_s - 1e-12:

                    best_h = h_trial.copy()

        d = np.sin(2*np.pi*f*np.arange(n)/n)
        d -= d.mean()
        if np.linalg.norm(d) > 1e-10:
            dirs.append(d / np.linalg.norm(d))
    
    return dirs

def symmetrize(h):
    return repair_feasible(0.5*(h + h[::-1]))

def make_binary(rng, n):
    idx = rng.choice(n, n//2, replace=False)
    b = np.zeros(n)
    b[idx] = 1.0
    return b
