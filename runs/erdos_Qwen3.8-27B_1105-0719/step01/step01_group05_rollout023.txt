    m = len(lin_a)
    for a_idx in range(m):
        for b_idx in range(a_idx + 1, m):
            da = lin_a[a_idx] - lin_a[b_idx]
            db = lin_b[b_idx] - lin_b[a_idx]
            if abs(db) > 1e-15:
                t = da / db
                if -1e-9 < t < 1.0 + 1e-9:
                    candidates.append(t)
    
    # Evaluate at all candidates
    best_t = 0.5
    best_val = float('inf')
    
    for t in candidates:
        t = max(0.0, min(1.0, t))
        # Compute max profile value
        val = 0.0
        if quad_a is not None:
            val = max(val, quad_a + t - t*t)
        for idx in range(m):
            val = max(val, lin_a[idx] + lin_b[idx] * t)
        if val < best_val:
            best_val = val
            best_t = t
    
    return best_t

def coordinate_descent(h, max_passes=50, tol=1e-10):
    """Run coordinate descent to minimize max_k p_k."""
    n = len(h)
    target = 0.5 * n
    h = project_box_sum(h, target)
    best_c = exact_objective(h)[0]
    
    for p in range(max_passes):
        improved = False
        for i in range(n):
            t_new = coord_update(h, i)
            if abs(t_new - h[i]) > 1e-12:
                h[i] = t_new
                # Project to maintain constraint
                h = project_box_sum(h, target)
                c = exact_objective(h)[0]
                if c < best_c:
                    best_c = c
                    improved = True
        if not improved:
            break
    
    return h, best_c

def binary_threshold(h):
    n = len(h)
    if n % 2 != 0:
        return None
    target = n // 2
    idx = np.argsort(-h)
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b

def binary_local_search(b, max_rounds=100, rng=None):
    """Local search for binary solutions using swaps."""
    n = len(b)
    if n % 2 != 0:
        return b, exact_objective(b)[0]
    if rng is None:
        rng = np.random.RandomState(42)
    
    c5, _ = exact_objective(b)
    
    for rnd in range(max_rounds):
        improved = False
        ones = list(np.where(b > 0.5)[0])
        zeros = list(np.where(b < 0.5)[0])
        
        # Try all pairs if small, random sample if large
        n_pairs = len(ones) * len(zeros)
        if n_pairs <= 5000:
            for i in ones:
                if improved:
                    break
                for j in zeros:
                    b[i], b[j] = 0.0, 1.0
                    nc, _ = exact_objective(b)
                    if nc < c5 - 1e-13:
                        c5 = nc
                        improved = True
                        ones = list(np.where(b > 0.5)[0])
                        zeros = list(np.where(b < 0.5)[0])
                        break
                    else:
                        b[i], b[j] = 1.0, 0.0
        else:
            n_tries = 1000
            for _ in range(n_tries):
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                b[i], b[j] = 0.0, 1.0
                nc, _ = exact_objective(b)
                if nc < c5 - 1e-13:
                    c5 = nc
                    improved = True
                    ones = list(np.where(b > 0.5)[0])
                    zeros = list(np.where(b < 0.5)[0])
                else:
                    b[i], b[j] = 1.0, 0.0
        
        if not improved:
            break
    
    return b, c5

def make_thue_morse(n):
    """Generate Thue-Morse sequence of length n."""
    t = np.zeros(n, dtype=np.float64)
    for i in range(1, n):
        # t[i] = t[i // 2] XOR (i % 2)
        if i % 2 == 0:
            t[i] = t[i // 2]
        else:
            t[i] = 1.0 - t[i // 2]
    return t

def make_start(n, rng, mode=0):
    target = 0.5 * n
    if mode == 0:
        h = np.zeros(n)
        h[::2] = 1.0
    elif mode == 1:
        h = np.zeros(n)
        h[:int(target)] = 1.0
    elif mode == 2:
        h = np.zeros(n)
        idx = rng.choice(n, size=int(target), replace=False)
        h[idx] = 1.0
    elif mode == 3:
        # Thue-Morse
        h = make_thue_morse(n)
        # Adjust to have exactly n/2 ones
        if n % 2 == 0:
            target = n // 2
            ones = np.sum(h > 0.5)
            if ones < target:
                # Add more ones
                zeros_idx = np.where(h < 0.5)[0]
                extra = int(target - ones)
                h[zeros_idx[:extra]] = 1.0
            elif ones > target:
                ones_idx = np.where(h > 0.5)[0]
                extra = int(ones - target)
                h[ones_idx[:extra]] = 0.0
    elif mode == 4:
        # Sine-based
        x = np.arange(n, dtype=np.float64)
        freq = 1.0 + rng.randint(1, 4)
        phase = rng.uniform(0, 2 * np.pi)
        h = (np.sin(2 * np.pi * freq * x / n + phase) > 0).astype(float)
        # Adjust to exactly n/2 ones
        if n % 2 == 0:
            target = n // 2
            ones = np.sum(h > 0.5)
            if ones < target:
                zeros_idx = np.where(h < 0.5)[0]
                h[zeros_idx[:int(target - ones)]] = 1.0
            elif ones > target:
                ones_idx = np.where(h > 0.5)[0]
                h[ones_idx[:int(ones - target)]] = 0.0
    elif mode == 5:
        h = np.full(n, 0.5) + rng.randn(n) * 0.3
        h = np.clip(h, 0, 1)
    else:
        h = np.full(n, 0.5)
    return project_box_sum(h, target)
