import numpy as np
from scipy.optimize import minimize
import time


def repair_feasible(vec):
    h = np.asarray(vec, dtype=np.float64).copy()
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) > 1e-14:
        if diff > 0:
            order = np.argsort(h)
            rem = diff
            for i in order:
                if rem <= 1e-15:
                    break
                add = min(rem, 1.0 - h[i])
                h[i] += add
                rem -= add
        else:
            order = np.argsort(-h)
            rem = -diff
            for i in order:
                if rem <= 1e-15:
                    break
                remv = min(rem, h[i])
                h[i] -= remv
                rem -= remv
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        for i in range(n):
            if diff2 > 0 and h[i] < 1.0:
                h[i] += diff2
                break
            elif diff2 < 0 and h[i] > 0.0:
                h[i] += diff2
                break
    return h


def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr * dx))


def compute_all_V(h):
    n = len(h)
    V = np.zeros(n)
    P = np.zeros(n)
    S = np.zeros(n)
    cumsum = np.cumsum(h)
    for m in range(1, n):
        P[m] = cumsum[m - 1]
        S[m] = cumsum[n - 1] - cumsum[n - m - 1]
    for m in range(n):
        if m == 0:
            V[m] = np.dot(h, h)
        else:
            A_m = np.dot(h[:n - m], h[m:])
            V[m] = A_m + min(P[m], S[m])
    return V, P, S


def grad_V_exact(h, m, P_m, S_m):
    n = len(h)
    g = np.zeros(n)
    if m == 0:
        g += 2.0 * h
    else:
        g[:n - m] += h[m:]
        g[m:] += h[:n - m]
        if P_m < S_m:
            g[:m] += 1.0
        elif S_m < P_m:
            g[n - m:] += 1.0
        else:
            g[:m] += 0.5
            g[n - m:] += 0.5
    return g


def smooth_obj_and_grad(h, gamma, beta_e):
    n = len(h)
    H = np.asarray(h, dtype=np.float64)
    V_raw = np.zeros(n)
    P = np.zeros(n)
    S = np.zeros(n)
    cumsum = np.cumsum(H)
    for m in range(1, n):
        P[m] = cumsum[m - 1]
        S[m] = cumsum[n - 1] - cumsum[n - m - 1]
    V = np.zeros(n)
    for m in range(n):
        if m == 0:
            V[m] = np.dot(H, H)
        else:
            A_m = np.dot(H[:n - m], H[m:])
            mu = min(P[m], S[m])
            eP = np.exp(-beta_e * max(0.0, P[m] - mu))
            eS = np.exp(-beta_e * max(0.0, S[m] - mu))
            b_smooth = mu - np.log(eP + eS) / beta_e
            V[m] = A_m + b_smooth
    V_min = np.min(V)
    exps = np.exp(-gamma * (V - V_min))
    Z = np.sum(exps)
    w = exps / Z
    sm_val = V_min - np.log(Z) / gamma
    grad = np.zeros(n)
    for m in range(n):
        if w[m] < 1e-12:
            continue
        if m == 0:
            gm = 2.0 * H
        else:
            gm = np.zeros(n)
            gm[:n - m] += H[m:]
            gm[m:] += H[:n - m]
            mu = min(P[m], S[m])
            eP = np.exp(-beta_e * max(0.0, P[m] - mu))
            eS = np.exp(-beta_e * max(0.0, S[m] - mu))
            Zp = eP + eS
            wP = eP / Zp
            wS = eS / Zp
            gm[:m] += wP
            gm[n - m:] += wS
        grad -= w[m] * gm
    return -sm_val, grad


def solve_smooth(h0, gamma, beta_e, maxiter=150):
    n = len(h0)
    h0 = repair_feasible(h0)

    def obj(x):
        val, _ = smooth_obj_and_grad(x, gamma, beta_e)
        return val

    def jac(x):
        _, g = smooth_obj_and_grad(x, gamma, beta_e)
        return g

    def eq_fun(x):
        return np.sum(x) - n / 2.0

    def eq_jac(x):
        return np.ones(n)

    constraints = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        result = minimize(obj, h0, jac=jac, method='SLSQP', bounds=bounds,
                          constraints=constraints, options={'maxiter': maxiter, 'ftol': 1e-18})
        if result.x is not None:
            h_sol = repair_feasible(result.x)
            s = exact_score(h_sol)
            if np.isfinite(s):
                return h_sol, s
    except Exception:
        pass
    return None, None


def active_descent(h, max_steps=50, n_active=15):
    n = len(h)
    best = repair_feasible(h)
    best_score = exact_score(best)
    for _ in range(max_steps):
        V, P, S = compute_all_V(best)
        V_min = np.min(V)
        idx = np.argsort(V)
        n_act = min(n_active, n)
        active = idx[:n_act]
        gaps = V[active] - V_min
        tau = 0.005 * max(abs(V_min), 0.01) + 1e-12
        w = np.exp(-gaps / tau)
        w /= np.sum(w)
        direction = np.zeros(n)
        for i, m in enumerate(active):
            direction += w[i] * grad_V_exact(best, m, P[m], S[m])
        direction -= np.mean(direction)
        improved = False
        for alpha in [0.03, 0.015, 0.008, 0.004, 0.002, 0.001, 0.0005]:
            h_trial = repair_feasible(best + alpha * direction)
            score = exact_score(h_trial)
            if score < best_score - 1e-15:
                best = h_trial.copy()
                best_score = score
                improved = True
                break
        if not improved:
            break
    return best, best_score


def interpolate_to_dimension(vec, new_len):
    n = len(vec)
    x_src = (np.arange(n) + 0.5) * 2.0 / n
    x_tgt = (np.arange(new_len) + 0.5) * 2.0 / new_len
    xs_ext = np.concatenate([[0.0], x_src, [2.0]])
    ys_ext = np.concatenate([[vec[0]], vec, [vec[-1]]])
    return np.interp(x_tgt, xs_ext, ys_ext)
