def repair_feasible(vec):

        G /= len(active_idx)

        G -= G.mean()

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx] + 1e-15)
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff

        return t - profile

    def ineq_jac(x):
        h = x[:-1]
        M = 2 * n - 1
        G = np.zeros((M, n))
        for m in range(M):
            d = m - (n - 1)
            G[m, :] = build_gradient(h, d, dx)
        J = np.zeros((M, n + 1))
        J[:, :-1] = -G
        J[:, -1] = 1.0
        return J

    x0 = np.concatenate([h_start, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 2.0)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': tol})
        if res.x is not None and np.isfinite(res.x).all():
            h_new = repair_feasible(res.x[:-1])
            c_new = exact_score(h_new)
            if c_new < c0 + 0.05:
                return h_new
    except Exception:
        pass
    return None

def paired_coord_descent(h, deadline, rng):
    """Paired coordinate descent maintaining sum constraint."""
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best_c = exact_score(h)
    it = 0
    while time.perf_counter() < deadline:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        s = h[i] + h[j]
        lo = max(0.0, s - 1.0)
        hi = min(1.0, s)
        if hi - lo < 1e-10:
            continue
        # Compute slope and intercept for each shift
        h[i] = 0.0
        h[j] = s
        prof0 = np.correlate(h, 1 - h, mode="full") * dx
        h[i] = 1.0
        h[j] = max(0.0, s - 1.0)
        prof1 = np.correlate(h, 1 - h, mode="full") * dx
        slope = prof1 - prof0
        intercept = prof0
        # Minimize max_k(slope[k]*t + intercept[k]) over t in [lo, hi]
        # Use ternary search (convex piecewise linear)
        a_lo, a_hi = lo, hi
        for _ in range(40):
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            t1 = a_lo + (1.0 - gr) * (a_hi - a_lo)
            t2 = a_lo + gr * (a_hi - a_lo)
            v1 = np.max(slope * t1 + intercept)
            v2 = np.max(slope * t2 + intercept)
            if v1 < v2:
                a_hi = t2
            else:
                a_lo = t1
        t_opt = (a_lo + a_hi) / 2.0
        h[i] = t_opt
        h[j] = s - t_opt
        c_new = exact_score(h)
        if c_new < best_c:
            best_c = c_new
        else:
            h[i] = s - t_opt
            h[j] = t_opt
        it += 1
    return h, best_c

def sim_anneal(h_init, deadline, rng):
    """Simulated annealing with various move types."""
    n = len(h_init)
    T = 0.03
    T_min = 0.00005
    cool = 0.9997
    h = h_init.copy()
    cur_c = exact_score(h)
    best_h = h.copy()
    best_c = cur_c
    polish_counter = 0
    while time.perf_counter() < deadline:
        r = rng.random()
        if r < 0.35:
            np_| = rng.randint(1, max(2, n // 3))
            idx = rng.choice(n, np_|, replace=False)
            delta = rng.randn(np_|) * (0.01 + T * 0.5)
            new_h = h.copy()
            new_h[idx] += delta
        elif r < 0.55:
            idx = rng.randint(n)
            step = (0.02 + T) * (1.0 if rng.random() < 0.5 else -1.0)
            new_h = h.copy()
            new_h[idx] += step
        elif r < 0.75:
            np_| = rng.randint(n // 5, max(3, n // 2))
            idx = rng.choice(n, np_|, replace=False)
            delta = rng.randn(np_|) * T
            new_h = h.copy()
            new_h[idx] += delta
        else:
            idx = rng.randint(n)
            if rng.random() < 0.5:
                new_h = h.copy()
                new_h[idx] = 0.0
            else:
                new_h = h.copy()
                new_h[idx] = 1.0
        new_h = repair_feasible(new_h)
        new_c = exact_score(new_h)
        delta_c = new_c - cur_c
        if delta_c < 0 or rng.random() < np.exp(-delta_c / max(T, 1e-10)):
            h = new_h
            cur_c = new_c
            if new_c < best_c:
                best_h = h.copy()
                best_c = new_c
                polish_counter = 0
        T = max(T_min, T * cool)
        polish_counter += 1
        if polish_counter >= 500 and time.perf_counter() < deadline - 30:
            polished = slsqp_solve(best_h, maxiter=80, tol=1e-10)
            if polished is not None:
                c_pol = exact_score(polished)
                if c_pol < best_c:
                    best_h = polished
                    best_c = c_pol
                    h = polished.copy()
                    cur_c = best_c
            polish_counter = 0
    return best_h, best_c

def logsumexp_descent(h_init, deadline, rng):
    """Optimize smoothed minimax using log-sum-exp approximation."""
    n = len(h_init)
    dx = 2.0 / n
    h = h_init.copy()
    best_c = exact_score(h)
    etas = [50, 100, 200, 500]
    for eta in etas:
        if time.perf_counter() > deadline:
            break
        def lse_obj(x):
            h_x = np.clip(x, 0.0, 1.0)
            profile = np.correlate(h_x, 1 - h_x, mode="full") * dx
            m = np.max(profile)
            return m + np.log(np.sum(np.exp((profile - m) * eta))) / eta

        def lse_jac(x):
            h_x = np.clip(x, 0.0, 1.0)
            profile = np.correlate(h_x, 1 - h_x, mode="full") * dx
            m = np.max(profile)
            w = np.exp((profile - m) * eta)
            w /= w.sum()
            M = len(profile)
            g = np.zeros(n)
            for k in range(M):
                d = k - (n - 1)
                g += w[k] * build_gradient(h_x, d, dx)
            return g

        h = h.copy()
        for step in range(30):
            if time.perf_counter() > deadline:
                break
            g = lse_jac(h)
            g -= g.mean()
            norm = np.linalg.norm(g)
            if norm < 1e-14:
                break
            for alpha in [0.005, 0.002, 0.001, 0.0005, 0.0001]:
                h_try = repair_feasible(h - alpha * g)
                c_try = exact_score(h_try)
                if c_try < best_c - 1e-13:
                    h = h_try
                    best_c = c_try
                    break
                elif c_try < lse_obj(h) + 0.01:
                    h = h_try
                    break
    return h, best_c

def multi_dim_optimize(h_best, best_c, deadline, rng):
    """Try optimizing at nearby dimensions."""
    n_best = len(h_best)
    best_result = (h_best, best_c)
    for dn in [-6, -4, -2, 2, 4, 6, 10, 14]:
        new_n = n_best + dn
        if new_n < 30 or new_n > 150:
            continue
        if time.perf_counter() > deadline:
            break
        xs = (np.arange(n_best) + 0.5) * (2.0 / n_best)
        xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
        full_xs = np.concatenate([[0.0], xs, [2.0]])
        full_h = np.concatenate([[h_best[0]], h_best, [h_best[-1]]])
        h_interp = np.interp(xt, full_xs, full_h)
        h_interp = repair_feasible(h_interp)
        polished = slsqp_solve(h_interp, maxiter=150, tol=1e-10)
        if polished is not None:
            c = exact_score(polished)
            if c < best_result[1]:
                best_result = (polished, c)
    return best_result

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.perf_counter()
    rng = np.random.RandomState(seed)

    parent = None
    try:
        parent = np.array(initial_h_values, dtype=np.float64)
    except NameError:
        pass
    except Exception:
        pass
    if parent is None or len(parent) < 4:
        parent = np.full(66, 0.5)
    parent = np.clip(parent, 0.0, 1.0)
    parent = repair_feasible(parent)

    best_h = parent.copy()
    best_c = exact_score(parent)
    n_best = len(best_h)

    def time_left():
        return budget_s - (time.perf_counter() - t0)

    # Stage 1: SLSQP from parent
    if time_left() > 80:
        res = slsqp_solve(best_h, maxiter=250, tol=1e-11)
        if res is not None:
            c = exact_score(res)
            if c < best_c:
                best_h, best_c = res, c

    # Stage 2: Paired coordinate descent
    if time_left() > 150:
        deadline = t0 + time_left() - 80
        h_cd, c_cd = paired_coord_descent(best_h, deadline, rng)
        if c_cd < best_c:
            best_h, best_c = h_cd, c_cd
            res = slsqp_solve(best_h, maxiter=150, tol=1e-11)
            if res is not None:
                c = exact_score(res)
                if c < best_c:
                    best_h, best_c = res, c

    # Stage 3: Simulated annealing
    if time_left() > 200:
        deadline = t0 + time_left() - 150
        h_sa, c_sa = sim_anneal(best_h, deadline, rng)
        if c_sa < best_c:
            best_h, best_c = h_sa, c_sa

    # Stage 4: SLSQP polish after SA
    if time_left() > 80:
        res = slsqp_solve(best_h, maxiter=300, tol=1e-12)
        if res is not None:
            c = exact_score(res)
            if c < best_c:
                best_h, best_c = res, c
                n_best = len(best_h)

    # Stage 5: Log-sum-exp descent
    if time_left() > 120:
        deadline = t0 + time_left() - 80
        h_lse, c_lse = logsumexp_descent(best_h, deadline, rng)
        if c_lse < best_c:
            best_h, best_c = h_lse, c_lse
            res = slsqp_solve(best_h, maxiter=200, tol=1e-11)
            if res is not None:
                c = exact_score(res)
                if c < best_c:
                    best_h, best_c = res, c

    # Stage 6: More coordinated descent
    if time_left() > 100:
        deadline = t0 + time_left() - 60
        h_cd2, c_cd2 = paired_coord_descent(best_h, deadline, np.random.RandomState(seed + 77))
        if c_cd2 < best_c:
            best_h, best_c = h_cd2, c_cd2

    # Stage 7: Multi-dimension exploration
    if time_left() > 100:
        deadline = t0 + time_left() - 40
        h_md, c_md = multi_dim_optimize(best_h, best_c, deadline, rng)
        if c_md < best_c:
            best_h, best_c = h_md, c_md
            n_best = len(best_h)

    # Stage 8: Final SA burst
    if time_left() > 100:
        deadline = t0 + time_left() - 50
        h_sa2, c_sa2 = sim_anneal(best_h, deadline, np.random.RandomState(seed + 200))
        if c_sa2 < best_c:
            best_h, best_c = h_sa2, c_sa2

    # Stage 9: Final SLSQP polish
    if time_left() > 40:
        res = slsqp_solve(best_h, maxiter=400, tol=1e-12)
        if res is not None:
            c = exact_score(res)
            if c < best_c:
                best_h, best_c = res, c

    # Stage 10: Coordinate descent final
    if time_left() > 30:
        deadline = t0 + time_left() - 5
        h_final, c_final = paired_coord_descent(best_h, deadline, np.random.RandomState(seed + 500))
        if c_final < best_c:
            best_h, best_c = h_final, c_final

    # Final validation
    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)
    n_best = len(best_h)

    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            final_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 2:
            final_c = float(eval_result[0])
    except Exception:
        pass

    return best_h, final_c, n_best
