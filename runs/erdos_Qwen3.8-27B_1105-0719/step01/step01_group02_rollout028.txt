import numpy as np
from scipy.optimize import minimize
import time

def exact_objective(h):
    n = len(h)
    dx = 2.0 / n
    raw = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(raw) * dx)

def project_box_sum(h, target=None):
    """Project h onto {x: 0<=x_i<=1, sum(x)=target}"""
    n = len(h)
    h = np.array(h, dtype=np.float64)
    h = np.clip(np.nan_to_num(h), 0.0, 1.0)
    if target is None:
        target = n / 2.0
    # Bisection on theta: find theta s.t. sum(clip(h+theta, 0, 1)) = target
    lo, hi = -1.0, 1.0  # h is in [0,1], so theta in [-1,1] suffices
    for _ in range(200):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    # Final fix for numerical precision
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        # Distribute among non-boundary elements
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_raw(h):
    return np.correlate(h, 1.0 - h, mode='full')

def grad_at_lag(h, k):
    """Gradient of raw at lag k (signed) with respect to h"""
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        # raw = sum_{i=0}^{n-1-k} h[i+k]*(1-h[i])
        for i in range(n - k):
            # h[i+k] contributes (1-h[i]) to its own derivative
            g[i+k] += (1.0 - h[i])
            # h[i] contributes -h[i+k]
            g[i] -= h[i+k]
    else:
        a = -k
        # raw = sum_{i=0}^{n-1-a} h[i]*(1-h[i+a])
        for i in range(n - a):
            g[i] += (1.0 - h[i+a])
            g[i+a] -= h[i]
    return g

Let me think about it as: we want to minimize the maximum "cross-correlation" between the indicator function of the 1-positions and the indicator of the 0-positions, over all shifts.

This is related to the concept of "balanced sequences" in coding theory. A sequence is "balanced" at shift k if the number of 1s in any window of length k is as close to k/2 as possible.

For a "perfectly balanced" sequence (which doesn't exist for all n), the raw at each lag k would be approximately (n-k)/4 (the "random" value). The maximum would then be at k=1, giving raw ≈ (n-1)/4.

C ≈ (2/n) * (n-1)/4 = (n-1)/(2n) ≈ 1/2.
