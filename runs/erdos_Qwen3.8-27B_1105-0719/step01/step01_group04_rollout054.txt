def exact_score(h):
    n = len(h)

    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr * dx))

def repair_feasible(vec):
    h = np.asarray(vec, dtype=np.float64)
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-12:
        return h.copy()
    if diff > 0:
        order = np.argsort(h)
        rem = diff
        for i in order:
            if rem <= 1e-15:
                break
            add = min(rem, 1.0 - h[i])
            h[i] += add
            rem -= add
    else:
        order = np.argsort(-h)
        rem = -diff
        for i in order:
            if rem <= 1e-15:
                break
            remv = min(rem, h[i])
            h[i] -= remv
            rem -= remv
    # Final fix
    diff2 = target - h.sum()
    if abs(diff2) > 0:
        for i in range(n):
            if diff2 > 0 and h[i] < 1.0:
                h[i] = min(1.0, h[i] + diff2)
                break
            elif diff2 < 0 and h[i] > 0.0:
                h[i] = max(0.0, h[i] + diff2)
                break
    return h

def compute_grads(h):
    """Compute values and gradient matrix for all lags.
    Returns: values (2n-1,), G (2n-1, n) where G[r,j] = d(values[r])/d(h[j])
    """
    n = len(h)
    dx = 2.0 / n
    n_lags = 2 * n - 1
    values = np.zeros(n_lags)
    G = np.zeros((n_lags, n))
    
    for k in range(n):  # k = 0, 1, ..., n-1 (shift amount)
        if k == 0:
            # g_0 = dx * sum h[i](1-h[i])
            values[n-1] = dx * np.sum(h * (1.0 - h))
            G[n-1, :] = dx * (1.0 - 2.0 * h)
        else:

            idx = np.arange(n - k)

            # Gradient: for j in [0, n-1-k]: d/dh[j] gives (1-h[j+k])

            # for j in [0, n-1-k]: d/dh[j] gives -h[j+k]

    n = len(vec)

import numpy as np

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr * dx))

def repair_feasible(vec):
    h = np.asarray(vec, dtype=np.float64)
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-12:
        return h.copy()
    if diff > 0:
        order = np.argsort(h)
        rem = diff
        for i in order:
            if rem <= 1e-15:
                break
            add = min(rem, 1.0 - h[i])
            h[i] += add
            rem -= add
    else:
        order = np.argsort(-h)
        rem = -diff
        for i in order:
            if rem <= 1e-15:
                break
            remv = min(rem, h[i])
            h[i] -= remv
            rem -= remv
    diff2 = target - h.sum()
    if abs(diff2) > 0:
        for i in range(n):
            if diff2 > 0 and h[i] < 1.0:
                h[i] = min(1.0, h[i] + diff2)
                break
            elif diff2 < 0 and h[i] > 0.0:
                h[i] = max(0.0, h[i] + diff2)
                break
    return h

def compute_grads(h):
    n = len(h)
    dx = 2.0 / n
    n_lags = 2 * n - 1
    values = np.zeros(n_lags)
    G = np.zeros((n_lags, n))
    for k in range(n):
        if k == 0:
            values[n-1] = dx * np.sum(h * (1.0 - h))
            G[n-1, :] = dx * (1.0 - 2.0 * h)
        else:
            idx = np.arange(n - k)
            values[n-1+k] = dx * np.sum(h[idx] * (1.0 - h[idx+k]))
            G[n-1+k, :n-k] += dx * (1.0 - h[k:])
            G[n-1+k, k:] += -dx * h[:n-k]
            idx2 = np.arange(k, n)
            values[n-1-k] = dx * np.sum(h[idx2] * (1.0 - h[idx2-k]))
            G[n-1-k, k:] += dx * (1.0 - h[:n-k])
            G[n-1-k, :n-k] += -dx * h[k:]
    return values, G

def project_box_sum(vec, target_sum):
    """Euclidean projection onto {h: 0<=h<=1, sum(h)=target_sum}"""
    n = len(vec)
    lo = vec.min() - 1.0
    hi = vec.max() + 0.1

        h_p = np.clip(vec - mid, 0.0, 1.0)
        s = h_p.sum()
        if abs(s - target_sum) < 1e-15:
            break
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(vec - (lo+hi)/2.0, 0.0, 1.0)

def smoothed_grad_descent(h_start, betas, inner_iters=500, verbose=False):
    h = repair_feasible(h_start.copy())
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_score = exact_score(h)
    
    for beta in betas:
        converged = False
        for it in range(inner_iters):
            values, G = compute_grads(h)
            c = values.max()
            # Numerically stable softmax
            shifted = beta * (values - c)
            exp_v = np.exp(shifted)
            w = exp_v / exp_v.sum()
            
            grad = w @ G  # (n,)
            
            # Lipschitz-based step size
            alpha_max = n / (beta * 8.0 + 1.0)
            
            improved = False
            for frac in [1.0, 0.5, 0.25, 0.1, 0.05]:
                alpha = alpha_max * frac
                h_new = project_box_sum(h - alpha * grad, target)
                score = exact_score(h_new)
                if score < best_score - 1e-14:
                    h = h_new
                    best_score = score
                    best_h = h.copy()
                    improved = True
                    break
            if not improved:
                # Try smaller steps
                for frac in [0.01, 0.005, 0.001]:
                    alpha = alpha_max * frac
                    h_new = project_box_sum(h - alpha * grad, target)
                    score = exact_score(h_new)
                    if score < best_score - 1e-15:
                        h = h_new
                        best_score = score
                        best_h = h.copy()
                        improved = True
                        break
                if not improved:
                    converged = True
                    break
        if verbose:
            print(f"beta={beta}, score={best_score:.8f}")
    
    return best_h, best_score

def active_descent(h, max_steps=100):
    """Gradient descent on active constraints only."""
    n = len(h)
    target = n / 2.0
    h = repair_feasible(h)
    best_h = h.copy()
    best_score = exact_score(h)
    
    for step in range(max_steps):
        values, G = compute_grads(h)
        c = values.max()
        # Select active lags (within tolerance of max)
        tol = 0.002 * c
        active = np.where(values > c - tol)[0]
        
        if len(active) == 0:
            active = np.where(values == c)[0]
        
        # Weighted gradient from active set
        gaps = np.maximum(c - values[active], 0)

        w /= w.sum() + 1e-15
        
        grad = w @ G[active, :]

        grad = grad - grad.mean()
        
        improved = False
        for alpha in [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]:
            h_new = project_box_sum(h - alpha * grad, target)

                h = h_new
                best_score = score
                best_h = h.copy()
                improved = True
                break
        if not improved:
            break
    
    return best_h, best_score

import numpy as np
from scipy.optimize import minimize
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr * dx))

def repair_feasible(vec):
    h = np.asarray(vec, dtype=np.float64).copy()
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(h)
        rem = diff
        for i in order:
            if rem <= 1e-15:
                break
            add = min(rem, 1.0 - h[i])
            h[i] += add
            rem -= add
    else:
        order = np.argsort(-h)
        rem = -diff
        for i in order:
            if rem <= 1e-15:
                break
            remv = min(rem, h[i])
            h[i] -= remv
            rem -= remv
    diff2 = target - h.sum()
    if abs(diff2) > 1e-10:
        for i in range(n):
            if diff2 > 0 and h[i] < 1.0:
                h[i] = min(1.0, h[i] + diff2)
            elif diff2 < 0 and h[i] > 0.0:
                h[i] = max(0.0, h[i] + diff2)
            break
    return h

def compute_all(h):
    """Compute all lag values and gradient matrix."""
    n = len(h)
    dx = 2.0 / n
    n_lags = 2 * n - 1
    values = np.zeros(n_lags)
    G = np.zeros((n_lags, n))
    for k in range(n):
        if k == 0:
            values[n-1] = dx * np.sum(h * (1.0 - h))
            G[n-1, :] = dx * (1.0 - 2.0 * h)
        else:
            # Lag +k: c[n-1+k] = sum_{i=0}^{n-1-k} h[i](1-h[i+k])
            values[n-1+k] = dx * np.sum(h[:n-k] * (1.0 - h[k:]))
            G[n-1+k, :n-k] = dx * (1.0 - h[k:])
            G[n-1+k, k:] = -dx * h[:n-k]
            # Lag -k: c[n-1-k] = sum_{i=k}^{n-1} h[i](1-h[i-k])
            values[n-1-k] = dx * np.sum(h[k:] * (1.0 - h[:n-k]))
            G[n-1-k, k:] = dx * (1.0 - h[:n-k])
            G[n-1-k, :n-k] = -dx * h[k:]
    return values, G

def project_box_sum(vec, target):
    n = len(vec)
    lo = vec.min() - 1.5
    hi = vec.max() + 1.5
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = np.clip(vec - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-14:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(vec - 0.5*(lo+hi), 0.0, 1.0)

def smooth_gd(h_init, betas, iters_per_beta=400):
    h = repair_feasible(h_init.copy())
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_score = exact_score(h)
    for beta in betas:
        stall = 0
        for it in range(iters_per_beta):
            values, G = compute_all(h)
            c = values[-1] if n == 1 else np.max(values)
            # softmax weights
            shift = beta * (values - c)
            ev = np.exp(shift)
            w = ev / ev.sum()
            grad = w @ G
            # step
            alpha0 = min(1.0, n / (2.0 * beta + 1.0))
            improved = False
            for f in [1.0, 0.5, 0.25, 0.125, 0.0625, 0.03]:
                h2 = project_box_sum(h - f*alpha0*grad, target)
                sc = exact_score(h2)
                if sc < best_score - 1e-14:
                    h = h2
                    best_score = sc
                    best_h = h.copy()
                    stall = 0
                    improved = True
                    break
            if not improved:
                stall += 1
                if stall > 30:
                    break
    return best_h, best_score

def active_refine_slsqp(h_init, n_active=10, maxiter=200):
    """SLSQP on a subset of active lags."""
    n = len(h_init)

    c = np.max(values)
    # Select top n_active lags
    top = np.argsort(values)[::-1][:n_active]
    top = np.sort(top)
    
    def obj(x):
        return x[n]
    
    def obj_grad(x):
        g = np.zeros(n+1)
        g[n] = 1.0
        return g
    
    def con_eq_fun(x):
        return x[:n].sum() - n/2.0
    
    def con_eq_jac(x):
        g = np.zeros(n+1)
        g[:n] = 1.0
        return g
    
    def con_ineq_fun(x):
        h = x[:n]
        t = x[n]
        vals, _ = compute_all(h)
        return t - vals[top]
    
    def con_ineq_jac(x):
        _, Gm = compute_all(x[:n])
        g = np.zeros((len(top), n+1))
        g[:, n] = 1.0
        g[:, :n] = -Gm[top, :]
        return g
    
    h0 = h_init.copy()
    t0 = exact_score(h0) + 0.05
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)]*n + [(0.0, t0+0.5)]
    constraints = [
        {'type': 'eq', 'fun': con_eq_fun, 'jac': con_eq_jac},
        {'type': 'ineq', 'fun': con_ineq_fun, 'jac': con_ineq_jac}
    ]
    try:
        res = minimize(obj, x0, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.x is not None:
            h_sol = repair_feasible(res.x[:n])
            sc = exact_score(h_sol)
            if np.isfinite(sc):
                return h_sol, sc
    except Exception:
        pass
    return None, None
