```python
def run_slsqp_direct(h_init, n, deadline, max_iter=50):
    """Run SLSQP with direct inequality constraints."""
    target = 0.5 * n
    dx = 2.0 / n
    c_init = exact_objective(h_init)
    
    # Variable: z = [h_0, ..., h_{n-1}, c]
    z0 = np.concatenate([h_init, [c_init + 1e-6]])
    
    bounds = [(0, 1)] * n + [(1e-6, 1.0)]
    
    def objective(z):
        return z[n]  # c
    
    def obj_grad(z):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g
    
    # Equality constraint: sum(h) - target = 0
    def eq_fun(z):
        return np.sum(z[:n]) - target
    
    def eq_jac(z):
        j = np.zeros(n + 1)
        j[:n] = 1.0
        return j
    
    constraints = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    
    # Inequality constraints: c - q_e(h) >= 0 for all e
    lags = np.arange(-(n-1), n)  # 2n-1 lags
    
    def ineq_fun(z):
        h = z[:n]
        c = z[n]
        profile = np.correlate(h, 1.0 - h, mode='full') * dx
        return c - profile  # shape (2n-1,)
    
    def ineq_jac(z):
        h = z[:n]
        # Build Jacobian matrix of shape (2n-1, n+1)
        jac = -np.column_stack([grad_q_e(h, e, dx) for e in lags])
        jac[:, n] = 1.0  # dc/dc = 1
        return jac
    
    constraints.append({'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac})
    
    try:
        res = minimize(objective, z0, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': max_iter, 'ftol': 1e-10})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = project_box_sum(res.x[:n], target)
            c_new = exact_objective(h_new)
            if c_new < c_init - 1e-12:
                return h_new, c_new
    except Exception:
        pass
    return h_init, c_init
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def exact_objective(h):
    n = len(h)
    if n == 0:
        return float('inf')
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(profile))

def project_box_sum(h, target=None):
    n = len(h)
    if target is None:
        target = 0.5 * n
    h = np.array(h, dtype=np.float64).copy()
    h = np.nan_to_num(h, nan=0.0, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] = np.minimum(h[mask] + diff / cnt, 1.0)
        else:
            mask = h > 0.0
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] = np.maximum(h[mask] + diff / cnt, 0.0)
    return h

def binary_threshold(h):
    n = len(h)
    if n % 2 != 0:
        return None
    target = n // 2
    idx = np.argsort(-h)
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b

def binary_local_search(b, deadline, max_rounds=200, rng=None):

    if n % 2 != 0:
        return b, exact_objective(b)
    if rng is None:

    best_b = b.copy()
    best_c5 = c5

                else:

def grad_q_all(h, dx):
    """Compute gradients for all lags. Returns array of shape (2n-1, n)."""
    n = len(h)
    jac = np.zeros((2*n-1, n))
    for idx, e in enumerate(range(-(n-1), n)):
        if e >= 0:
            # c_e = dx * sum_{m=0}^{n-e-1} h[m+e] * (1-h[m])
            for m in range(n - e):
                jac[idx, m+e] += dx * (1 - h[m])
                jac[idx, m] -= dx * h[m+e]
        else:
            s = -e
            # c_{-s} = dx * sum_{m=0}^{n-s-1} h[m] * (1-h[m+s])
            for m in range(n - s):
                jac[idx, m] += dx * (1 - h[m+s])
                jac[idx, m+s] -= dx * h[m]
    return jac

def run_slsqp_direct(h_init, deadline, max_iter=80):
    """SLSQP with direct inequality constraints on all lags."""
    n = len(h_init)
    target = 0.5 * n
    dx = 2.0 / n
    c_init = exact_objective(h_init)
    
    z0 = np.concatenate([h_init, [c_init + 1e-6]])
    bounds = [(0, 1)] * n + [(1e-8, 1.0)]
    
    lags = list(range(-(n-1), n))
    n_lags = len(lags)
    
    def objective(z):
        return z[n]
    
    def obj_grad(z):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g
    
    def eq_fun(z):
        return np.sum(z[:n]) - target
    
    def eq_jac(z):
        j = np.zeros(n + 1)
        j[:n] = 1.0
        return j
    
    def ineq_fun(z):
        h = z[:n]
        c = z[n]
        profile = np.correlate(h, 1.0 - h, mode='full') * dx
        return c - profile
    
    def ineq_jac(z):
        h = z[:n]
        jit = grad_q_all(h, dx)  # shape (2n-1, n)
        jac = np.hstack([-jit, np.ones((n_lags, 1))])
        return jac
    
    constraints = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    
    best_h = h_init.copy()
    best_c = c_init
    
    try:
        res = minimize(objective, z0, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': max_iter, 'ftol': 1e-12})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = project_box_sum(res.x[:n], target)
            c_new = exact_objective(h_new)
            if c_new < best_c - 1e-13:
                best_c = c_new
                best_h = h_new
    except Exception:
        pass
    
    return best_h, best_c

def make_binary_start(n, rng):
    """Create a random binary sequence with n/2 ones."""
    if n % 2 != 0:
        n += 1
    idx = rng.permutation(n)
    b = np.zeros(n)
    b[idx[:n//2]] = 1.0
    return b

def make_smooth_start(n, rng):
    """Create a smooth starting point."""
    x = np.arange(n, dtype=np.float64)
    freq = 1 + rng.randint(1, 4)
    phase = rng.uniform(0, 2*np.pi)
    h = 0.5 + 0.3 * np.cos(2*np.pi*freq*x/n + phase)
    return project_box_sum(h, 0.5*n)

def downsample(h, new_n):
    n = len(h)
    if new_n == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(new_n) + 0.5) / new_n
    vals = np.interp(x_new, x_old, h)
    return project_box_sum(vals, 0.5*new_n)
```

```python
            h = z[:n]
            c = z[n]
            profile_local = np.correlate(h, 1.0 - h, mode='full') * dx
            return c - profile_local[active_idx]
        
        def ineq_jac(z):
            h = z[:n]
            # Only compute gradients for active lags
            jac = np.zeros((len(active_idx), n + 1))
            for row, lag in enumerate(active_lags):
                if lag >= 0:
                    for m in range(n - lag):
                        jac[row, m+lag] += dx * (1 - h[m])
                        jac[row, m] -= dx * h[m+lag]
                else:
                    s = -lag
                    for m in range(n - s):
                        jac[row, m] += dx * (1 - h[m+s])
                        jac[row, m+s] -= dx * h[m]
                jac[row, n] = 1.0
            return jac
        
        constraints = [
            {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
            {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
        ]
        
        try:
            res = minimize(objective, z0, jac=obj_grad, method='SLSQP',
                           bounds=bounds, constraints=constraints,
                           options={'maxiter': 50, 'ftol': 1e-10})
            if res.x is not None and np.all(np.isfinite(res.x)):
                h_new = project_box_sum(res.x[:n], target)
                c_new = exact_objective(h_new)
                if c_new < best_c - 1e-13:
                    best_c = c_new
                    best_h = h_new.copy()
                # Check if new solution better even if not strictly lower
                # (could be that c variable is lower but profile max is same)
        except Exception:
            pass
    
    return best_h, best_c
```

This is better because it only uses top_k constraints per SLSQP call, making each call faster.
