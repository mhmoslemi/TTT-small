Now let me think about the L-BFGS-B optimization. The objective is:
F(h) = (1/beta) * log(sum_k exp(beta * corr_k(h))) * dx

Wait, I should be careful. The actual objective is max_k corr_k * dx. The log-sum-exp approximation is:
(1/beta) * log(sum_k exp(beta * corr_k * dx)) ≈ max_k (corr_k * dx) for large beta.

But for L-BFGS-B, I can use this smoothed objective with a finite beta, then do a final evaluation with the true max.

The constraint is: sum h = n/2, 0 ≤ h ≤ 1.
L-BFGS-B handles box constraints natively but not equality constraints.

To handle the equality constraint, I can either:
1. Use SLSQP (which handles both)
2. Parameterize: fix h[n-1] = target_sum - sum(h[0:n-1]), and only optimize h[0:n-1]
3. Use a penalty method

Option 2 is cleanest. Let me use it.

Actually, option 2 has a problem: h[n-1] might go outside [0,1]. I need to also constrain 0 ≤ target_sum - sum(h[0:n-1]) ≤ 1, which means target_sum - 1 ≤ sum(h[0:n-1]) ≤ target_sum. This is hard to express as box constraints.

Let me just use SLSQP. It's slower but handles the constraint properly.

Or better: use the **projection** approach. Optimize without the sum constraint using L-BFGS-B (with box constraints only), then project. This is a penalty-like approach.

Actually, the cleanest approach for L-BFGS-B with the equality constraint is to use a **dual variable** (Lagrange multiplier) for the sum constraint:

minimize F(h) + lambda * (sum h - target_sum)
subject to 0 ≤ h ≤ 1

Then alternate:
1. Given lambda, optimize h using L-BFGS-B (box constraints only)
2. Update lambda using bisection to satisfy sum h = target_sum

This is essentially the Uzawa algorithm or a variant of the Newton method for the saddle point problem.

One more thought: the `grad_corr_k` function has a loop-free implementation if I use numpy carefully. Let me vectorize it:

For a given offset (k - (n-1)):
- The first term: grad[idx + offset] += (1 - h[idx]) where idx ranges from max(0, -offset) to min(n-1, n-1-offset)
- The second term: grad[idx] += (-h[idx + offset]) for the same idx range

```python
def gradient_smooth_max(h, dx, beta):
    n = len(h)
    corrs = compute_corrs(h)  # shape (2n-1,)
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(2*n-1) / (2*n-1)
    else:
        w /= w_sum
    
    grad = np.zeros(n)
    for k in range(2*n-1):
        if w[k] < 1e-14:
            continue
        offset = k - (n-1)
        if offset >= 0:
            # offset >= 0: i ranges from 0 to n-1-offset
            i_max = n - 1 - offset
            idx = np.arange(0, i_max+1)
            grad[idx + offset] += (1.0 - h[idx]) * w[k]
            grad[idx] += (-h[idx + offset]) * w[k]
        else:
            # offset < 0: i ranges from -offset to n-1
            i_min = -offset
            idx = np.arange(i_min, n)
            grad[idx + offset] += (1.0 - h[idx]) * w[k]
            grad[idx] += (-h[idx + offset]) * w[k]
    
    return grad * dx
```

import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time


def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-13:
        mask = (h > 5e-5) & (h < 1.0 - 5e-5)
        if mask.sum() > 0:
            adj = diff / mask.sum()
            adj = np.clip(adj, -h[mask] + 1e-15, (1.0 - h[mask]) - 1e-15)
            h[mask] += adj
    return np.clip(h, 0.0, 1.0)


def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def c5_of_h(h, dx):
    return compute_corrs(h).max() * dx


def gradient_smooth_max(h, dx, beta):
    n = len(h)
    corrs = compute_corrs(h)
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        w /= w_sum
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        offset = k - (n - 1)
        if offset >= 0:
            i_max = n - 1 - offset
            if i_max < 0:
                continue
            idx = np.arange(0, i_max + 1)
            grad[idx + offset] += (1.0 - h[idx]) * w[k]
            grad[idx] += (-h[idx + offset]) * w[k]
        else:
            i_min = -offset
            idx = np.arange(i_min, n)
            grad[idx + offset] += (1.0 - h[idx]) * w[k]
            grad[idx] += (-h[idx + offset]) * w[k]
    return grad * dx


def fista_optimize(h0, dx, target_sum, beta, max_iter=3000, restart_period=200):
    n = len(h0)
    L = 4.0 / n
    alpha = 1.0 / L
    x = h0.copy()
    y = h0.copy()
    t = 1.0
    best_h = x.copy()
    best_c5 = c5_of_h(x, dx)
    no_improve = 0

    for it in range(max_iter):
        if it > 0 and it % restart_period == 0:
            y = x.copy()
            t = 1.0

        grad = gradient_smooth_max(y, dx, beta)
        x_new = project_feasible(y - alpha * grad, target_sum)

        c5_val = c5_of_h(x_new, dx)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = x_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = x_new + ((t - 1.0) / t_new) * (x_new - x)
        y = np.clip(y, 0.0, 1.0)
        x = x_new
        t = t_new

        if no_improve > restart_period * 3:
            y = x.copy()
            t = 1.0
            no_improve = 0

    return best_h, best_c5


def slsqp_optimize(h0, dx, target_sum, beta, maxiter=500):
    n = len(h0)

    def obj(x):
        corrs = compute_corrs(x)
        return logsumexp(beta * corrs) / beta * dx

    def grad_obj(x):
        return gradient_smooth_max(x, dx, beta)

    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n

    try:
        res = minimize(obj, h0, jac=grad_obj, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        h_out = project_feasible(res.x, target_sum)
    except Exception:
        h_out = project_feasible(h0, target_sum)
    return h_out, c5_of_h(h_out, dx)


def lbfgsb_with_project(h0, dx, target_sum, beta, max_iter=200):
    n = len(h0)

    def obj(x):
        corrs = compute_corrs(x)
        return logsumexp(beta * corrs) / beta * dx + 1e4 * (x.sum() - target_sum) ** 2

    def grad_obj(x):
        g = gradient_smooth_max(x, dx, beta)
        g += 2e4 * (x.sum() - target_sum) * np.ones(n)
        return g

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0, jac=grad_obj, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': max_iter, 'gtol': 1e-10})
        h_out = project_feasible(res.x, target_sum)
    except Exception:
        h_out = project_feasible(h0, target_sum)
    return h_out, c5_of_h(h_out, dx)


def make_perturbed_start(h_center, target_sum, rng, sigma=0.1):
    n = len(h_center)
    h_pert = h_center + rng.normal(0, sigma, n)
    return project_feasible(h_pert, target_sum)


def interpolate_h(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    x_old = np.linspace(0.0, 2.0, n_old)
    x_new = np.linspace(0.0, 2.0, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    target = n_new / 2.0
    return project_feasible(h_new, target)


def remaining_time(t0, budget_s):
    return budget_s - (time.time() - t0)
