        values[center + delta] = dx * np.sum(vals_pos)
        # d/dh[i] for i in [0, n-1-delta]: contribution from h[i]*(1-h[i+delta])
        grads[center + delta, idx] += dx * (1.0 - h[idx + delta])
        # d/dh[i+delta]: contribution from h[i]*(1-h[i+delta]) = -h[i]
        grads[center + delta, idx + delta] += -dx * h[idx]

        # d/dh[i]: -h[i+delta]

import numpy as np
from scipy.optimize import minimize
import time


def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr * dx))


def repair_feasible(vec):
    h = np.asarray(vec, dtype=np.float64)
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-12:
        return h.copy()
    if diff > 0:
        order = np.argsort(h)
        rem = diff
        for i in order:
            if rem <= 1e-15:
                break
            add = min(rem, 1.0 - h[i])
            h[i] += add
            rem -= add
    else:
        order = np.argsort(-h)
        rem = -diff
        for i in order:
            if rem <= 1e-15:
                break
            remv = min(rem, h[i])
            h[i] -= remv
            rem -= remv
    diff2 = target - h.sum()
    if abs(diff2) > 1e-14:
        for i in range(n):
            if diff2 > 0 and h[i] < 1.0:
                h[i] += diff2
                break
            elif diff2 < 0 and h[i] > 0.0:
                h[i] += diff2
                break
    return h


def compute_profile_grads(h):
    n = len(h)
    dx = 2.0 / n
    n_lags = 2 * n - 1
    values = np.zeros(n_lags)
    grads = np.zeros((n_lags, n))
    center = n - 1
    values[center] = dx * np.sum(h * (1.0 - h))
    grads[center, :] = dx * (1.0 - 2.0 * h)
    for delta in range(1, n):
        idx = np.arange(n - delta)
        values[center + delta] = dx * np.sum(h[idx] * (1.0 - h[idx + delta]))
        grads[center + delta, idx] += dx * (1.0 - h[idx + delta])
        grads[center + delta, idx + delta] += -dx * h[idx]
        values[center - delta] = dx * np.sum(h[idx + delta] * (1.0 - h[idx]))
        grads[center - delta, idx + delta] += dx * (1.0 - h[idx])
        grads[center - delta, idx] += -dx * h[idx + delta]
    return values, grads


def soft_max_obj(h, beta):
    values, _ = compute_profile_grads(h)
    max_val = np.max(values)
    shifted = beta * (values - max_val)
    shifted = np.clip(shifted, -500, 500)
    log_sum = beta * max_val + np.log(np.sum(np.exp(shifted)))
    return log_sum / beta


def soft_max_grad(h, beta):
    values, grads = compute_profile_grads(h)
    max_val = np.max(values)
    shifted = beta * (values - max_val)
    shifted = np.clip(shifted, -500, 500)
    weights = np.exp(shifted)
    weights /= np.sum(weights)
    return weights @ grads


def solve_slsqp_softmax(h_start, beta, maxiter=150):
    n = len(h_start)
    h0 = repair_feasible(h_start)

    def obj(x):
        return soft_max_obj(x, beta)

    def grad(x):
        return soft_max_grad(x, beta)

    def eq_fun(x):
        return np.sum(x) - n / 2.0

    def eq_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    try:
        result = minimize(
            obj, h0, jac=grad,
            method='SLSQP', bounds=bounds, constraints=constraints,
            options={'maxiter': maxiter, 'ftol': 1e-14, 'disp': False}
        )
        if result.x is not None:
            h_sol = repair_feasible(result.x)
            score = exact_score(h_sol)
            if np.isfinite(score):
                return h_sol, score
    except Exception:
        pass
    return None, None


def active_descent(h, max_steps=100, sigma=0.003):
    n = len(h)
    best = repair_feasible(h)
    best_score = exact_score(best)
    cur_sigma = sigma
    for _ in range(max_steps):
        values, grads = compute_profile_grads(best)
        c = np.max(values)
        gaps = c - values
        weights = np.exp(-gaps / (cur_sigma * c + 1e-10))
        total_w = np.sum(weights)
        if total_w < 1e-15:
            break
        weights /= total_w
        direction = -(weights @ grads)
        direction -= np.mean(direction)
        norm_d = np.linalg.norm(direction)
        if norm_d < 1e-14:
            break
        direction /= norm_d
        improved = False

            if cur_sigma < 5e-5:

                cur_sigma *= 1.5
                if cur_sigma > sigma * 2.0:

    return best, best_score


def cdf_transfer(h, new_n):
    n = len(h)
    dx = 2.0 / n
    cum_bound = np.concatenate([[0.0], np.cumsum(h * dx)])
    x_src = np.linspace(0.0, 2.0, n + 1)
    x_new = np.linspace(0.0, 2.0, new_n + 1)
    cum_new = np.interp(x_new, x_src, cum_bound)
    h_new = np.diff(cum_new) / (2.0 / new_n)
    return h_new


def project_feasible(x, n):

        rem = -diff
        for i in order:
            if rem <= 0:
                break
            sub = min(rem, x[i])
            x[i] -= sub
            rem -= sub
    return x
