Wait, the coordinate descent with projection is not quite right because the projection changes all coordinates, not just the one we're optimizing. This means we're not doing a true 1D optimization.

A better approach: fix all coordinates except h[j], and vary h[j]. The sum constraint becomes: h[j] = target - sum_{i≠j} h[i]. So h[j] is determined! We can't freely vary it.

Unless we allow a small perturbation: vary h[j] and h[k] simultaneously (swap a small amount between them). This is a 1D problem (the amount swapped) with the sum constraint automatically satisfied.

Let me reformulate: for a pair (j, k), consider h[j] + δ, h[k] - δ for δ in [-(h[j]), min(h[k], 1-h[j])]. The sum is preserved. This is a true 1D optimization.

        # h[j] + δ, h[k] - δ
        # δ range: [-h[j], min(1-h[j], h[k])]

        dmax = min(1.0 - h[j], h[k])

            return c_best, h

            h_temp = h.copy()
            h_temp[j] += delta
            h_temp[k] -= delta
            c_tmp = exact_c5(h_temp)
            if c_tmp < best_c:
                best_c = c_tmp
                best_delta = delta
        if best_delta != 0:

            h[k] -= best_delta

    for rnd in range(max_rounds):
        if time.time() > deadline - 5:
            break
        improved = False
        ones = list(np.where(b > 0.5)[0])
        zeros = list(np.where(b < 0.5)[0])
        for i in ones:
            if time.time() > deadline - 3:
                break
            for j in zeros:
                b[i] = 0.0
                b[j] = 1.0
                nc = exact_c5(b)
                if nc < c5 - 1e-13:
                    c5 = nc
                    improved = True
                    ones = list(np.where(b > 0.5)[0])
                    zeros = list(np.where(b < 0.5)[0])
                    break
                else:
                    b[i] = 1.0
                    b[j] = 0.0
            if improved:
                break
        if not improved:
            break
    return b, c5

def subgradient_descent(h_init, deadline, max_iter=200):
    h = h_init.copy()
    n = len(h)
    dx = 2.0 / n
    target = 0.5 * n
    c_best = exact_c5(h)
    for it in range(max_iter):
        if time.time() > deadline - 5:
            break
        profile = np.correlate(h, 1.0 - h, mode='full') * dx
        pmax = profile.max()
        k = min(10, len(profile))
        top_idx = np.argsort(-profile)[:k]
        top_lags = top_idx - (n - 1)
        w = profile[top_idx] - profile[top_idx].min()
        w = np.maximum(w, 0.0)
        if w.sum() < 1e-12:
            w = np.ones(k)
        w = w / w.sum()
        direction = np.zeros(n)
        for wi, lag in zip(w, top_lags):
            direction -= wi * _lag_grad(h, lag, dx)
        dnorm = np.linalg.norm(direction)
        if dnorm < 1e-14:
            break
        direction /= dnorm
        improved = False
        for alpha in [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0001]:
            h_t = project_box_sum(h + alpha * direction, target)
            c_t = exact_c5(h_t)
            if c_t < c_best - 1e-14:
                h = h_t
                c_best = c_t
                improved = True
                break
        if not improved:
            break
    return h, c_best

def downsample_h(h, new_n):
    n = len(h)
    if new_n == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(new_n) + 0.5) / new_n
    vals = np.interp(x_new, x_old, h)
    return project_box_sum(vals, 0.5 * new_n)

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    deadline = start_time + budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c = float('inf')

    try:
        init = np.asarray(initial_h_values, dtype=np.float64)
        if init.size > 0 and np.all(np.isfinite(init)):
            n0 = len(init)
            init = project_box_sum(init, 0.5 * n0)
            c0 = exact_c5(init)
            best_h = init.copy()
            best_c = c0
    except (NameError, Exception):
        pass

    if best_h is None:
        n0 = 70
        best_h = np.zeros(n0)
        best_h[::2] = 1.0
        best_h = project_box_sum(best_h, 0.5 * n0)
        best_c = exact_c5(best_h)

    n0 = len(best_h)

    # Phase 1: SLSQP on parent with progressively tighter tau
    tau_lists = [
        [0.01, 0.005, 0.002],
        [0.001, 0.0005, 0.0002, 0.0001],
        [0.00005, 0.00002, 0.00001],
    ]
    for tl in tau_lists:
        if time.time() > deadline - 60:
            break
        h_opt, c_opt = slsqp_refine(best_h, deadline, tl, max_iter_per_tau=200)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 2: Subgradient descent
    if time.time() < deadline - 80:
        h_sg, c_sg = subgradient_descent(best_h, deadline, max_iter=300)
        if c_sg < best_c:
            best_c = c_sg
            best_h = h_sg

    # Phase 3: Pair swap optimization
    if time.time() < deadline - 60:
        h_ps, c_ps = pair_swap_optimize(best_h, deadline, max_rounds=8, n_trials=10)
        if c_ps < best_c:
            best_c = c_ps
            best_h = h_ps

    # Phase 4: SLSQP again after local search
    if time.time() < deadline - 50:
        h_opt, c_opt = slsqp_refine(best_h, deadline, [0.0001, 0.00003, 0.00001], max_iter_per_tau=250)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 5: Binary refinement
    if n0 % 2 == 0 and time.time() < deadline - 40:
        b = binary_threshold(best_h)
        if b is not None:
            b, c_b = binary_swap_improve(b, deadline, max_rounds=80)
            if c_b < best_c:
                best_c = c_b
                best_h = b.copy()
                # SLSQP from binary
                if time.time() < deadline - 30:
                    h_opt, c_opt = slsqp_refine(b, deadline, [0.001, 0.0003, 0.0001, 0.00003], max_iter_per_tau=200)
                    if c_opt < best_c:
                        best_c = c_opt
                        best_h = h_opt

    # Phase 6: Pair swap again
    if time.time() < deadline - 30:
        h_ps, c_ps = pair_swap_optimize(best_h, deadline, max_rounds=6, n_trials=10)
        if c_ps < best_c:
            best_c = c_ps
            best_h = h_ps

    # Phase 7: Try different sizes
    sizes_to_try = sorted(set([s for s in [50, 54, 56, 58, 60, 62, 64, 66, 68, 70, 72, 74, 76, 80]
                               if 20 <= s <= 100]), key=lambda x: abs(x - n0))

    for n_try in sizes_to_try:
        if n_try == len(best_h):
            continue
        if time.time() > deadline - 25:
            break
        # Start from downsampled parent
        if best_h is not None and len(best_h) != n_try:
            h_start = downsample_h(best_h, n_try)
        else:
            h_start = np.zeros(n_try)
            h_start[::2] = 1.0
            h_start = project_box_sum(h_start, 0.5 * n_try)

        h_opt, c_opt = slsqp_refine(h_start, deadline, [0.005, 0.002, 0.0005], max_iter_per_tau=150)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

        # Quick pair swap
        if time.time() < deadline - 15:
            h_ps, c_ps = pair_swap_optimize(h_opt, deadline, max_rounds=4, n_trials=8)
            if c_ps < best_c:
                best_c = c_ps
                best_h = h_ps

    # Phase 8: Final tightening SLSQP + pair swap
    if time.time() < deadline - 15:
        h_opt, c_opt = slsqp_refine(best_h, deadline, [0.00003, 0.00001], max_iter_per_tau=300)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    if time.time() < deadline - 10:
        h_ps, c_ps = pair_swap_optimize(best_h, deadline, max_rounds=4, n_trials=10)
        if c_ps < best_c:
            best_c = c_ps
            best_h = h_ps

    # Final validation
    best_h = project_box_sum(best_h, 0.5 * len(best_h))
    final_c = exact_c5(best_h)
    best_c = final_c
    best_n = len(best_h)

    return (best_h, best_c, best_n)
