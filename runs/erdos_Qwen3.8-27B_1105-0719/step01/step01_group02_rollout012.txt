import numpy as np
from scipy.optimize import linprog, minimize

def compute_all_profiles(h):
    """Compute all shifted overlap values."""
    n = len(h)
    dx_local = 2.0 / n
    # correlate gives 2n-1 values
    prof = np.correlate(h, 1.0 - h, mode='full') * dx_local
    return prof

def compute_gradient_k(h, k):
    """Compute gradient of f_k wrt h, where k is the lag (-n+1 to n-1)."""
    n = len(h)
    dx_local = 2.0 / n
    a = abs(k)
    if a >= n:
        return np.zeros(n)
    grad = np.zeros(n)
    # For indices m where m + a < n: contribution from (1-h[m+a])
    if a == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        # m from 0 to n-1-a: +(1-h[m+a])
        grad[0:n-a] += (1.0 - h[a:n])
        # m from a to n-1: -h[m-a]
        grad[a:n] -= h[0:n-a]
    grad *= dx_local
    return grad

def project_box_sum(h, target=None):
    """Project h onto {x : 0<=x<=1, sum(x)=target}."""
    n = len(h)
    h = np.array(h, dtype=np.float64)
    h = np.nan_to_num(h, nan=0.0, posinf=1.0, neginf=0.0)
    if target is None:
        target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        s = np.sum(h)
        diff = s - target
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h > 1e-15
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            step = min(diff / cnt, np.min(h[mask]))
            h[mask] -= step
        else:
            mask = h < 1.0 - 1e-15
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            step = min(-diff / cnt, 1.0 - np.max(h[mask]))
            h[mask] += step
    # Final correction
    diff = np.sum(h) - target
    if abs(diff) > 1e-13:
        if diff > 0:
            mask = h > 1e-10
        else:
            mask = h < 1.0 - 1e-10
        if np.any(mask):
            h[mask] -= diff / np.sum(mask)
            h = np.clip(h, 0.0, 1.0)
    return h

def exact_objective(h):
    """Compute C5 = max of correlate(h, 1-h, mode='full') * dx."""
    n = len(h)
    dx_local = 2.0 / n
    prof = np.correlate(h, 1.0 - h, mode='full') * dx_local
    return float(np.max(prof))

def active_set_lp(h, tol_frac=0.05, t_max=0.5):
    """
    Active-set LP: find direction d that reduces all active profiles.
    Returns (d, M_new) or None.
    """
    n = len(h)
    prof = compute_all_profiles(h)
    M = prof.max()
    
    # Active set: all lags where prof > M - tol_frac * M (or absolute tolerance)
    tol = max(0.001, tol_frac * M)
    active_idx = np.where(prof > M - tol)[0]
    
    if len(active_idx) == 0:
        return None
    
    # Build LP:
    # Variables: d[0..n-1], M_new  (total n+1 variables)
    # Minimize: M_new
    # Subject to:
    #   For each active k: grad_k · d - M_new <= -f_k
    #   sum(d) = 0
    #   -h[i]/t_max <= d[i] <= (1-h[i])/t_max
    
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0  # minimize M_new
    
    # Inequality constraints: A_ub @ [d; M_new] <= b_ub
    # For each active lag: grad_k · d - M_new <= -f_k
    # i.e., [grad_k, -1] · [d; M_new] <= -f_k
    A_rows = []
    b_rows = []
    
    for idx in active_idx:
        k = int(idx - (n - 1))  # convert to lag
        gk = compute_gradient_k(h, k)
        row = np.zeros(n + 1)
        row[0:n] = gk
        row[n] = -1.0
        A_rows.append(row)
        b_rows.append(-prof[idx])
    
    A_ub = np.array(A_rows)
    b_ub = np.array(b_rows)
    
    # Bounds on d: -h[i]/t_max <= d[i] <= (1-h[i])/t_max
    bounds = []
    for i in range(n):
        bounds.append((-h[i] / t_max, (1.0 - h[i]) / t_max))
    bounds.append((0.0, M + 1.0))  # M_new >= 0, upper bound loose
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, 0:n] = 1.0
    b_eq = np.array([0.0])
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
    except Exception:
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='revised_simplex')
        except Exception:
            return None
    
    if res.status != 0:
        return None
    
    d = res.x[0:n]
    M_new = res.x[n]
    
    # Check if this is actually an improving direction
    if M_new >= M - 1e-10:
        return None
    
    return d, M_new

def line_search(h, d, c_current, t_max=0.5, n_steps=50):

        return h, c_current
    
    # Bisection / backtracking
    best_t = 0.0

    for i in range(n_steps):

def polish_active_set(h, max_iters=200, t_max=0.3, tol_frac=0.03):

    n = len(h)
    c_cur = exact_objective(h)

            c_cur = c_new

def binary_threshold(h):
    n = len(h)
    if n % 2 != 0:
        return h
    target = n // 2
    idx = np.argsort(-h)
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b

def transfer_to_size(parent, target_n):
    n = len(parent)
    if n == target_n:

    new_h = np.interp(x_new, x_old, parent)

def project_box_sum(h, target=None):
    n = len(h)
    h = np.array(h, dtype=np.float64).copy()
    h = np.nan_to_num(h, nan=0.0, posinf=1.0, neginf=0.0)
    if target is None:
        target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        s = np.sum(h)
        diff = s - target
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h > 1e-15
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            step = min(diff / cnt, np.min(h[mask]))
            h[mask] -= step
        else:
            mask = h < 1.0 - 1e-15
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            step = min(-diff / cnt, 1.0 - np.max(h[mask]))
            h[mask] += step
    diff = np.sum(h) - target
    if abs(diff) > 1e-13:
        if diff > 0:
            mask = h > 1e-10
        else:
            mask = h < 1.0 - 1e-10
        if np.any(mask):
            h[mask] -= diff / np.sum(mask)
            h = np.clip(h, 0.0, 1.0)
    return h

def exact_objective(h):
    n = len(h)
    dx = 2.0 / n
    prof = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(prof))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def gradient_lag(h, a):
    """Gradient of the overlap for lag a (a >= 0) with respect to h."""
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    if a == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        if n - a > 0:
            grad[0:n-a] += (1.0 - h[a:n])
            grad[a:n] -= h[0:n-a]
    grad *= dx
    return grad

def active_set_direction(h, tol_frac=0.05, t_max=0.5):
    """Find a direction via active-set LP."""
    n = len(h)
    prof = compute_profile(h)
    M = prof.max()
    tol = max(0.002, tol_frac * M)
    active_idx = np.where(prof > M - tol)[0]
    n_active = len(active_idx)
    if n_active == 0:
        return None
    
    # LP variables: d[0..n-1], M_new  (n+1 total)
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    A_rows = []
    b_rows = []
    for idx in active_idx:
        lag = int(idx - (n - 1))
        a = abs(lag)
        gk = gradient_lag(h, a)
        row = np.zeros(n + 1)
        row[0:n] = gk
        row[n] = -1.0
        A_rows.append(row)
        b_rows.append(-prof[idx])
    
    A_ub = np.array(A_rows)
    b_ub = np.array(b_rows)
    
    bounds = []
    for i in range(n):
        bounds.append((-h[i] / t_max, (1.0 - h[i]) / t_max))
    bounds.append((0.0, M + 2.0))
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, 0:n] = 1.0
    b_eq = np.array([0.0])
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
    except Exception:
        return None
    if res.status != 0:
        return None
    
    d = res.x[0:n]
    M_new = res.x[n]
    if M_new >= M - 1e-10:
        return None
    if np.max(np.abs(d)) < 1e-10:
        return None
    return d

def line_search_step(h, d, c_cur, t_max=0.4):
    """Find good step along direction d."""
    n = len(h)
    t_box = t_max
    for i in range(n):
        if d[i] > 1e-14:
            t_box = min(t_box, (1.0 - h[i]) / d[i])
        elif d[i] < -1e-14:
            t_box = min(t_box, -h[i] / d[i])
    t_box = min(t_box, t_max)
    if t_box < 1e-10:
        return h, c_cur
    best_t = 0.0
    best_c = c_cur
    for i in range(25):
        t = t_box * (0.65 ** i)
        if t < 1e-10:

        h_try = project_box_sum(h_try, n / 2.0)
        c_try = exact_objective(h_try)

            best_t = t
    if best_t > 0:

def polish_lp(h, max_iters=150, t_max=0.35, tol_frac=0.04):
    """Iterate active-set LP polishing."""
    n = len(h)
    c_cur = exact_objective(h)
    stuck = 0
    for it in range(max_iters):
        d = active_set_direction(h, tol_frac=tol_frac, t_max=t_max)
        if d is None:
            break
        h_new, c_new = line_search_step(h, d, c_cur, t_max=t_max)
        if c_new < c_cur - 1e-11:
            h = h_new
            c_cur = c_new
            stuck = 0
        else:
            stuck += 1
            if stuck > 5:
                break
    return h, c_cur

def subgradient_step(h, tol_frac=0.05, step_size=0.05):
    """Subgradient descent on the max of active profiles."""
    n = len(h)
    prof = compute_profile(h)
    M = prof.max()
    tol = max(0.002, tol_frac * M)
    active_idx = np.where(prof > M - tol)[0]
    if len(active_idx) == 0:
        return h, exact_objective(h)
    grad_sum = np.zeros(n)
    for idx in active_idx:
        lag = int(idx - (n - 1))
        a = abs(lag)
        gk = gradient_lag(h, a)
        w = (prof[idx] - (M - tol)) / (tol + 1e-15)
        grad_sum += w * gk
    if np.max(np.abs(grad_sum)) < 1e-14:
        return h, exact_objective(h)
    h_new = h - step_size * grad_sum
    h_new = project_box_sum(h_new, n / 2.0)
    return h_new, exact_objective(h_new)

    target = n // 2
    idx = np.argsort(-h)
    b = np.zeros(n)

        best_delta = 0.0

        total_pairs = len(ones) * len(zeros)

                    cur[i] = 0.0
                    cur[j] = 1.0
                    c = exact_objective(cur)
                    delta = cur_c - c
                    cur[i] = 1.0
                    cur[j] = 0.0
                    if delta > best_delta:

                        best_pair = (i, j)
        else:

def pair_swap_polish(h, n_trials=3000, scale=0.1, rng_seed=77):

    cur = h.copy()

    rg = np.random.RandomState(rng_seed)
    temp = max(0.001, 0.01 * cur_c)
    for t in range(n_trials):
        frac = t / float(n_trials)

        i = int(rg.randint(n))

        amt = scale * (0.3 + 0.7 * rg.random())
        if amt > cur[j] * 0.9:
            amt = cur[j] * 0.85
        if amt > (1.0 - cur[i]) * 0.9:
            amt = max(0.01, (1.0 - cur[i]) * 0.85)
        if amt < 0.003:
            continue
        cur[i] += amt
        cur[j] -= amt
        c = exact_objective(cur)

        else:

                cur_c = c
            else:
                cur[i] -= amt

    return cur, cur_c

def run(seed=42, budget_s=1100, **kwargs):
    deadline = time.time() + budget_s
    rng = np.random.RandomState(seed)
    
    best_h = None
    best_c = float('inf')
    best_n = 0
    parent = None
    
    try:
        parent = np.array(initial_h_values, dtype=np.float64)
        parent = project_box_sum(parent, len(parent) / 2.0)
        pc = exact_objective(parent)
        best_h = parent.copy()
        best_c = pc
        best_n = len(parent)
    except (NameError, Exception):
        pass
    
    if best_h is None:
        n_init = 120
        best_h = np.ones(n_init) * 0.5
        best_h = project_box_sum(best_h, n_init / 2.0)
        best_c = exact_objective(best_h)
        best_n = n_init
    
    sizes = [best_n, 140, 120, 100, 80, 160, 96, 112, 128, 64, 200, 80]
    seen = set()
    unique_sizes = []
    for s in sizes:
        if s not in seen and 20 <= s <= 300:
            seen.add(s)
            unique_sizes.append(s)
    
    candidate_count = 0
    for n in unique_sizes:
        if time.time() > deadline - 80:
            break
        cands = generate_candidates(n, parent, rng)
        scored = []
        for c in cands:
            c = project_box_sum(c, n / 2.0)
            cc = exact_objective(c)
            scored.append((cc, c))
        scored.sort(key=lambda z: z[0])
        
        n_top = min(4, len(scored))
        for ci in range(n_top):
            if time.time() > deadline - 50:
                break
            c_start, h_start = scored[ci]
            candidate_count += 1
            
            # Full LP + binary polish
            h_pol, c_pol = full_polish(h_start.copy(), deadline, seed_offset=candidate_count * 100 + ci)
            
            # Pair swap escape
            if time.time() < deadline - 30 and c_pol > best_c:
                h_swap, c_swap = pair_swap_polish(h_pol, n_trials=2000, scale=0.08, rng_seed=candidate_count * 7 + ci)
                if c_swap < c_pol:
                    h_pol = h_swap
                    c_pol = c_swap
                    if time.time() < deadline - 20:
                        h_pol, c_pol = full_polish(h_pol, deadline, seed_offset=candidate_count * 200 + ci)
            
            if c_pol < best_c - 1e-12:
                best_c = c_pol
                best_h = h_pol.copy()
                best_n = n
                parent = h_pol.copy()
    
    # Final deep polish on best
    if time.time() < deadline - 20 and best_h is not None:
        h_f, c_f = full_polish(best_h.copy(), deadline, seed_offset=9999)
        if c_f < best_c - 1e-12:
            best_c = c_f
            best_h = h_f
    
    if time.time() < deadline - 10 and best_h is not None and best_n % 2 == 0:
        b = binary_threshold(best_h)
        if b is not None:
            b_c = exact_objective(b)
            if b_c < best_c:
                best_h = b
                best_c = b_c
            h_bl, c_bl = binary_local_search(best_h, max_rounds=400, rng_seed=8888)
            if c_bl < best_c:
                best_h = h_bl
                best_c = c_bl
    
    best_h = project_box_sum(best_h, best_n / 2.0)
    best_c = exact_objective(best_h)
    return (best_h, best_c, len(best_h))
