The key innovation is using `scipy.optimize.linprog` with the linearized profile constraints. At each point h, the profile p_k = Σ h[i]·(1-h[i+k])·dx is differentiable, so we can linearize:
p_k(h + δ) ≈ p_k(h) + G_k · δ

where G_k is the gradient of the k-th profile component.

The LP is:
minimize θ
subject to: G_k · x - θ ≤ G_k · h - p_k(h) for all k
           sum(x) = n/2
           0 ≤ x_i ≤ 1

This gives a direction that decreases the maximum possible profile.

import numpy as np
import time
from scipy.optimize import minimize, linprog

def exact_score(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))

def get_profile(h):
    dx = 2.0 / len(h)
    return np.correlate(h, 1.0 - h, mode="full") * dx

def lag_gradient_matrix(h):
    """Build the (2n-1, n) matrix of gradients for each lag."""
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    G = np.zeros((M, n))
    for k in range(M):
        delta = k - (n - 1)
        if delta > 0:
            length = n - delta
            if length > 0:
                # P = dx * sum_{a=0}^{length-1} h[a+delta] * (1 - h[a])
                # dP/dh[a+delta] += dx * (1 - h[a])
                # dP/dh[a] += -dx * h[a+delta]
                G[k, delta:delta+length] += dx * (1.0 - h[0:length])
                G[k, 0:length] -= dx * h[delta:delta+length]
        elif delta < 0:
            c = -delta
            length = n - c
            if length > 0:
                # P = dx * sum_{a=0}^{length-1} h[a] * (1 - h[a+c])
                # dP/dh[a] += dx * (1 - h[a+c])
                # dP/dh[a+c] += -dx * h[a]
                G[k, 0:length] += dx * (1.0 - h[c:c+length])
                G[k, c:c+length] -= dx * h[0:length]
        else:
            # P = dx * sum h[i]*(1-h[i])
            G[k, :] = dx * (1.0 - 2.0 * h)
    return G

def repair_feasible(h, target=None):
    h = np.array(h, dtype=np.float64).copy()
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    if target is None:
        target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        order = np.argsort(-room)
        remaining = diff
        for idx in order:
            if remaining <= 1e-15:
                break
            add = min(remaining, room[idx])
            h[idx] += add
            remaining -= add
    else:
        order = np.argsort(-h)
        remaining = -diff
        for idx in order:
            if remaining <= 1e-15:
                break
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
    return h

def interpolate_to_dim(vec, new_len):
    n = len(vec)
    if n == new_len:
        return vec.copy()
    x_src = (np.arange(n) + 0.5) * 2.0 / n
    x_tgt = (np.arange(new_len) + 0.5) * 2.0 / new_len
    x_ext = np.concatenate([[0.0], x_src, [2.0]])
    f_ext = np.concatenate([[vec[0]], vec, [vec[-1]]])
    h_new = np.interp(x_tgt, x_ext, f_ext)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

def symmetrize(h):
    h_sym = 0.5 * (h + h[::-1].copy())
    return repair_feasible(h_sym)

def complement_symmetrize(h):
    """h_sym[i] = 0.5 * (h[i] + (1 - h[n-1-i]))"""
    h_c = 0.5 * (h + (1.0 - h[::-1].copy()))
    return repair_feasible(h_c)

def trust_lp(h, radius, max_time=None):
    """Solve the linearized Chebyshev LP within a trust region."""
    n = len(h)
    M = 2 * n - 1
    dx = 2.0 / n
    
    prof = get_profile(h)
    G = lag_gradient_matrix(h)
    
    # Linearization: p_k(h + delta) ≈ p_k(h) + G[k]·delta
    # We want: G[k]·x - theta <= G[k]·h - p_k(h) for all k
    # i.e., G[k]·(x - h) <= -p_k(h)  ... wait let me re-derive
    # p_k(x) ≈ p_k(h) + G[k]·(x-h)
    # We want p_k(x) <= theta for all k
    # So: p_k(h) + G[k]·(x-h) <= theta
    # G[k]·x - theta <= G[k]·h - p_k(h)
    # Actually: p_k(h) + G[k]·(x-h) <= theta
    # => G[k]·x + p_k(h) - G[k]·h <= theta
    # => G[k]·x - theta <= G[k]·h - p_k(h)
    
    # Variables: x[0..n-1], theta
    # Objective: minimize theta
    # Constraints:
    #   G[k,0:n]·x - theta <= G[k,0:n]·h - p_k(h)  for k=0..M-1
    #   sum(x) = n/2
    #   h[i] - radius <= x[i] <= h[i] + radius
    #   0 <= x[i] <= 1
    
    # In standard form for linprog: A_ub @ v <= b_ub
    # v = [x, theta]
    
    # G[k]·x - theta <= G[k]·h - p_k(h)
    # [G[k,0:n], -1] @ [x; theta] <= G[k,0:n]@h - p_k(h)
    
    A_ub = np.zeros((M, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    rhs = G @ h - prof
    
    b_ub = rhs
    
    # Equality: sum(x) = n/2
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    bounds = []
    for i in range(n):
        lo = max(0.0, h[i] - radius)
        hi = min(1.0, h[i] + radius)
        bounds.append((lo, hi))
    bounds.append((0.0, None))  # theta >= 0
    
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize theta
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x is not None:
            h_sol = res.x[:n]
            return repair_feasible(h_sol), res.x[n]
    except Exception:
        pass
    return None, None

def line_search(h_center, direction, best_score, max_evals=20):
    """Line search along direction from h_center."""
    n = len(h_center)
    best_h = h_center.copy()
    best_s = best_score

def local_gradient_descent(h, max_iters=40, time_limit=20):
    """Weighted gradient descent on active lags."""
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_score = exact_score(h)
    t0 = time.perf_counter()
    
    for it in range(max_iters):
        if time.perf_counter() - t0 > time_limit:
            break
        prof = get_profile(h)
        score = float(np.max(prof))
        if score < best_score - 1e-12:
            best_score = score
            best_h = h.copy()
        
        # Select top active lags
        A = max(5, min(20, n // 4))
        active_idx = np.argsort(prof)[::-1][:A]
        G = lag_gradient_matrix(h)
        
        # Weighted gradient
        w = np.maximum(prof[active_idx] - (prof[active_idx].min() + 0.01), 0)
        wsum = w.sum()
        if wsum < 1e-15:
            break
        w /= wsum
        
        direction = np.sum(w[:, None] * G[active_idx, :], axis=0)
        direction -= direction.mean()
        norm = np.linalg.norm(direction)
        if norm < 1e-14:
            break
        direction /= norm
        
        improved = False
        for alpha in [0.05, 0.03, 0.015, 0.008, 0.004, 0.002, 0.001, 0.0005]:
            h_trial = repair_feasible(h + alpha * direction)
            s_trial = exact_score(h_trial)
            if s_trial < best_score - 1e-12:
                h = h_trial
                best_h = h.copy()
                best_score = s_trial
                improved = True
                break
        if not improved:
            break
    return best_h, best_score

def slsqp_solve(h_start, t_start, beta=0.0, maxiter=60, n_constraints=0):
    n = len(h_start)
    M = 2 * n - 1
    nc = n_constraints if n_constraints > 0 else M
    
    if nc < M:
        prof = get_profile(h_start)
        active_lags = np.argsort(prof)[::-1][:nc]
    else:
        active_lags = np.arange(M)
    
    def obj(x):
        h = x[:n]
        t = x[n]
        return t + beta * np.sum(h * (1.0 - h))
    
    def obj_jac(x):
        h = x[:n]
        g = np.zeros(n + 1)
        g[:n] = beta * (1.0 - 2.0 * h)
        g[n] = 1.0
        return g
    
    def cons_fun(x):
        h = x[:n]
        t = x[n]
        eq = h.sum() - n / 2.0
        prof = np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)
        ineq = t - prof[active_lags]
        return np.concatenate([[eq], ineq])
    
    def cons_jac(x):
        h = x[:n]
        G = lag_gradient_matrix(h)
        Ja = G[active_lags, :]
        J = np.zeros((1 + len(active_lags), n + 1))
        J[0, :n] = 1.0
        J[1:, :n] = -Ja
        J[1:, n] = 1.0
        return J
    
    bounds = [(0.0, 1.0)] * n + [(0.0, t_start + 2.0)]
    x0 = np.concatenate([h_start, [t_start]])
    
    try:
        result = minimize(
            obj, x0, jac=obj_jac,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'ineq', 'fun': cons_fun, 'jac': cons_jac}],
            options={'maxiter': maxiter, 'ftol': 1e-11}
        )
        if result.x is not None and np.all(np.isfinite(result.x)):
            h_sol = repair_feasible(result.x[:n])
            return h_sol
    except Exception:
        pass
    return None
