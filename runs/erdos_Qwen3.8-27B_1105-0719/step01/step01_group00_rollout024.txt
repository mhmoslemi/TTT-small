Let me think about the optimization more carefully. The SLSQP with smooth max is the main workhorse. The issue is that with high beta, the smooth max becomes very peaked, and SLSQP struggles because the Hessian becomes ill-conditioned.

A potential fix: **use a two-level optimization**. Inner loop: SLSQP with moderate beta. Outer loop: update the weights based on the current corrs.

This is essentially the **outer approximation** method:
1. Start with uniform weights w.
2. Solve: min_h ∑_k w_k f_k(h) subject to constraints.
3. Update w: w_k = exp(β f_k(h)) / ∑ exp(β f_j(h)).
4. Repeat.

The key difference from the smooth max SLSQP is that at each outer iteration, we solve a *weighted sum* of the quadratics (not a smooth max). The weighted sum is a simpler objective that SLSQP can handle better.

Let me implement this:

```python
def _outer_approx(h0, dx, ts, max_outer, beta, deadline):
    n = len(h0)
    h = h0.copy()
    
    for outer in range(max_outer):
        if time.time() > deadline:
            break
        
        # Compute weights
        c = compute_corrs(h)
        w = np.exp(beta * (c - c.max()))
        w /= w.sum()
        
        # Solve: min_h ∑ w_k f_k(h)
        def obj(x):
            cx = compute_corrs(x)
            return (w * cx).sum() * dx
        
        def jac(x):
            g = np.zeros(n)
            for k in range(2*n-1):
                if w[k] < 1e-14:
                    continue
                g += w[k] * grad_corr_k(x, k, n)
            return g * dx
        
        def cf(x):
            return x.sum() - ts
        def cj(x):
            return np.ones(n)
        
        cons = [{'type': 'eq', 'fun': cf, 'jac': cj}]
        bnds = [(0.0, 1.0)] * n
        try:
            res = minimize(obj, h, jac=jac, method='SLSQP',
                           bounds=bnds, constraints=cons,
                           options={'maxiter': 500, 'ftol': 1e-15})
            h_new = project_feasible(res.x, ts)
        except Exception:
            h_new = h
        
        c5_old = c5_of_h(h, dx)
        c5_new = c5_of_h(h_new, dx)
        
        if c5_new < c5_old:
            h = h_new
        else:
            # Try smaller step
            h_half = (h + h_new) / 2
            h_half = project_feasible(h_half, ts)
            c5_half = c5_of_h(h_half, dx)
            if c5_half < c5_old:
                h = h_half
            # Else keep h
    
    return h
```

def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)


def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def c5_of_h(h, dx):
    return compute_corrs(h).max() * dx


def grad_corr_k(h, k, n):
    offset = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -offset)
    i_hi = min(n - 1, n - 1 - offset)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx + offset] += (1.0 - h[idx])
    grad[idx] += (-h[idx + offset])
    return grad
