def repair_feasible(vec):

        G = build_gradient_matrix(h, n, dx)

        grad = G_active.mean(axis=0)
        grad -= grad.mean()

                break
    return h, best

def repair_feasible(vec):

            G[m, 0:n-dp] += dx * (1.0 - h[dp:n])

def local_descent(h, max_iters=200, time_limit=None, t_start=None):
    """Active-set subgradient descent with top-k."""
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best = exact_score(h)
    for _ in range(max_iters):
        if time_limit is not None and time.perf_counter() - t_start > time_limit:
            break
        profile = compute_profile(h)
        c = profile.max()
        if c < best - 1e-12:
            best = c
        k_top = min(8, 2 * n - 1)
        active = np.argsort(profile)[-k_top:]
        G = build_gradient_matrix(h, n, dx)
        G_act = G[active, :]
        # Weight by how close each is to max
        weights = np.exp(50.0 * (profile[active] - c))
        weights /= weights.sum()
        grad = G_act.T @ weights
        grad -= grad.mean()
        norm = np.linalg.norm(grad)
        if norm < 1e-14:
            break
        grad /= norm
        for alpha in [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]:
            h_try = h - alpha * grad
            h_try = project_feasible(h_try)
            s = exact_score(h_try)
            if s < best - 1e-12:
                h = h_try
                best = s
                break
    return h, best

    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_feasible(h_new)

def smooth_lbfgs(h0, maxiter=800, beta=300.0, time_limit=None, t_start=None):
    """L-BFGS-B with log-sum-exp smoothing and penalty for equality."""
    n = len(h0)
    
    def obj_grad(x):
        h = np.clip(x, 0.0, 1.0)
        profile = compute_profile(h)
        m = profile.max()
        shifted = np.clip(beta * (profile - m), -400, 400)
        exp_s = np.exp(shifted)
        f_val = m + np.log(exp_s.sum()) / beta
        weights = exp_s / exp_s.sum()
        G = build_gradient_matrix(h, n, 2.0 / n)
        grad = G.T @ weights
        # Penalty for equality constraint
        viol = h.sum() - n / 2.0
        f_val += 50.0 * viol * viol
        grad += 100.0 * viol
        return f_val, grad
    
    try:
        res = minimize(obj_grad, h0.copy(), jac=True, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10})
        if res.x is not None and np.isfinite(res.x).all():
            return project_feasible(res.x)
    except Exception:
        pass
    return None

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()
    rng = np.random.RandomState(seed)
    
    def time_left():
        return budget_s - (time.perf_counter() - t_start)
    
    # Load parent
    parent = None
    try:
        parent = np.array(initial_h_values, dtype=np.float64)
    except NameError:
        pass
    except Exception:
        pass
    
    if parent is None or len(parent) < 4:
        parent = np.full(66, 0.5)
    
    parent = np.clip(parent, 0.0, 1.0)
    parent = repair_feasible(parent)
    
    best_h = parent.copy()
    best_c = exact_score(best_h)
    
    # Symmetrize
    h_sym = project_feasible(symmetrize(best_h))
    c_sym = exact_score(h_sym)
    if c_sym < best_c - 1e-12:
        best_h = h_sym
        best_c = c_sym
    
    # Stage 1: Coordinate descent (main workhorse)
    if time_left() > 60:
        h_cd, c_cd = coordinate_descent(best_h, max_sweeps=300,
                                        time_limit=50, t_start=t_start)
        if c_cd < best_c - 1e-12:
            best_h = h_cd
            best_c = c_cd
    
    # Stage 2: L-BFGS-B with smooth penalty
    if time_left() > 60:
        for beta_val in [200.0, 500.0]:
            if time_left() < 30:
                break
            h_lb = smooth_lbfgs(best_h, maxiter=1000, beta=beta_val,
                                time_limit=25, t_start=t_start)
            if h_lb is not None:
                c_lb = exact_score(h_lb)
                if c_lb < best_c - 1e-10:
                    best_h = h_lb
                    best_c = c_lb
    
    # Stage 3: Coordinate descent refinement
    if time_left() > 40:
        h_cd, c_cd = coordinate_descent(best_h, max_sweeps=500,
                                        time_limit=40, t_start=t_start)
        if c_cd < best_c - 1e-12:
            best_h = h_cd
            best_c = c_cd
    
    # Stage 4: Local active-set descent
    if time_left() > 30:
        h_ld, c_ld = local_descent(best_h, max_iters=200,
                                   time_limit=30, t_start=t_start)
        if c_ld < best_c - 1e-12:
            best_h = h_ld
            best_c = c_ld
    
    # Stage 5: Multi-restart with coordinate descent
    if time_left() > 100:
        n_restarts = 12
        for r in range(n_restarts):
            if time_left() < 20:
                break
            eps = 0.01 + 0.015 * r
            z = rng.randn(len(best_h))
            z -= z.mean()
            z /= (np.linalg.norm(z) + 1e-15)
            h_pert = project_feasible(best_h + eps * z)
            
            # Symmetrize perturbation too
            h_pert_sym = project_feasible(symmetrize(h_pert))
            h_cd, c_cd = coordinate_descent(h_pert_sym, max_sweeps=200,
                                            time_limit=8, t_start=t_start)
            if c_cd < best_c - 1e-10:
                best_h = h_cd
                best_c = c_cd
            
            h_cd2, c_cd2 = coordinate_descent(h_pert, max_sweeps=200,
                                              time_limit=8, t_start=t_start)
            if c_cd2 < best_c - 1e-10:
                best_h = h_cd2
                best_c = c_cd2
    
    # Stage 6: L-BFGS-B polish from refined best
    if time_left() > 40:
        h_lb = smooth_lbfgs(best_h, maxiter=1500, beta=1000.0,
                            time_limit=30, t_start=t_start)
        if h_lb is not None:
            c_lb = exact_score(h_lb)
            if c_lb < best_c - 1e-10:
                best_h = h_lb
                best_c = c_lb
        h_cd, c_cd = coordinate_descent(best_h, max_sweeps=300,
                                        time_limit=20, t_start=t_start)
        if c_cd < best_c - 1e-12:
            best_h = h_cd
            best_c = c_cd
    
    # Stage 7: Try higher resolution
    if time_left() > 80:
        for exp_n in [len(best_h) + 34, len(best_h) + 68]:
            if time_left() < 40 or exp_n > 250:
                break
            h_exp = interpolate_to_n(best_h, exp_n)
            h_cd, c_cd = coordinate_descent(h_exp, max_sweeps=300,
                                            time_limit=30, t_start=t_start)
            if c_cd < best_c - 1e-10:
                best_h = h_cd
                best_c = c_cd
            h_ld, c_ld = local_descent(best_h, max_iters=100,
                                       time_limit=15, t_start=t_start)
            if c_ld < best_c - 1e-12:
                best_h = h_ld
                best_c = c_ld
    
    # Stage 8: Final coordinate descent + local descent
    if time_left() > 20:
        h_cd, c_cd = coordinate_descent(best_h, max_sweeps=400,
                                        time_limit=15, t_start=t_start)
        if c_cd < best_c - 1e-12:
            best_h = h_cd
            best_c = c_cd
    if time_left() > 10:
        h_ld, c_ld = local_descent(best_h, max_iters=100,
                                   time_limit=10, t_start=t_start)
        if c_ld < best_c - 1e-12:
            best_h = h_ld
            best_c = c_cd  # bug fix: should be c_ld
            best_c = c_ld
    
    # Final validation
    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)
    
    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            final_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 2:
            final_c = float(eval_result[0])
    except Exception:
        pass
    
    return best_h, final_c, len(best_h)
