```python
    """Find optimal delta to transfer from j to i, return new h and improvement"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    c5 = np.max(corr) * dx
    
    # Bounds on delta
    delta_max_pos = 1.0 - h[i]  # h[i] + delta <= 1
    delta_max_neg = h[j]        # h[j] - delta >= 0, so delta >= -h[j]
    
    # For shift s, the change in corr at lag (n-1+s) is:
    # delta * [(1-h[i+s])*[0<=i+s<n] - h[i-s]*[0<=i-s<n] - (1-h[j+s])*[j+s valid] + h[j-s]*[j-s valid]]
    
    # Compute g for all lags
    g = np.zeros(2*n-1)
    for s in range(-(n-1), n):
        # h[i] contribution: (1-h[i+s]) if 0<=i+s<n, 0 otherwise
        # -h[i-s] if 0<=i-s<n, 0 otherwise
        gi = 0.0
        if 0 <= i+s < n:
            gi += 1.0 - h[i+s]
        if 0 <= i-s < n:
            gi -= h[i-s]
        
        gj = 0.0
        if 0 <= j+s < n:
            gj += 1.0 - h[j+s]
        if 0 <= j-s < n:
            gj -= h[j-s]
        
        g[n-1+s] = gi - gj
    
    # Now minimize max(corr + delta*g) over delta in [-delta_max_neg, delta_max_pos]
    # This is a 1D convex problem
    # Use scipy or manual bisection
    
    from scipy.optimize import minimize_scalar
    def obj(delta):
        return np.max(corr + delta * g)
    
    res = minimize_scalar(obj, bounds=(-delta_max_neg, delta_max_pos), method='bounded', 
                          options={'xatol': 1e-12})
    best_delta = res.x
    new_max = res.fun
    
    if new_max < c5/dx - 1e-14:  # Actually c5/dx is the max of corr
        h_new = h.copy()
        h_new[i] += best_delta
        h_new[j] -= best_delta
        h_new = np.clip(h_new, 0, 1)
        return h_new, new_max * dx
    return h, c5
```

```python
        if len(interior) > 0:
            h[interior] += residual / len(interior)
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = beta * (corr - m)
    shifted = np.clip(shifted, -500, 0)
    exps = np.exp(shifted)
    return (m + np.log(np.sum(exps)) / beta) * dx

def smooth_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = beta * (corr - m)
    shifted = np.clip(shifted, -500, 0)
    exps = np.exp(shifted)
    weights = exps / np.sum(exps)
    
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        s = idx - (n - 1)
        w = weights[idx]
        if w < 1e-14:
            continue
        # For shift s: d(c_s)/d(h[j]) = (1-h[j+s])*[0<=j+s<n] - h[j-s]*[0<=j-s<n]
        # Sum over j where j+s is valid: j in [max(0,-s), min(n-1, n-1-s)]
        lo1 = max(0, -s)
        hi1 = min(n - 1, n - 1 - s)
        if lo1 <= hi1:
            grad[lo1:hi1+1] += w * (1.0 - h[lo1+s:hi1+1+s])
        # Sum over j where j-s is valid: j in [max(0,s), min(n-1, n-1+s)]
        lo2 = max(0, s)
        hi2 = min(n - 1, n - 1 + s)
        if lo2 <= hi2:
            grad[lo2:hi2+1] -= w * h[lo2-s:hi2+1-s]
    
    return grad * dx

def slsqp_refine(h, betas, max_iter=200):
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0, 
             'jac': lambda x: np.ones(n)}]
    for beta in betas:
        try:
            res = minimize(smooth_obj, best_h, args=(beta,),
                          jac=smooth_grad, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': max_iter, 'ftol': 1e-16})
            if res.x is not None:
                h_new = project_feasible(res.x, n)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5 - 1e-15:
                    best_c5 = c5_new
                    best_h = h_new
        except Exception:
            pass
    return best_h, best_c5

def pairwise_descent(h, n_steps, rng, cooling=None):
    """Random pairwise coordinate descent"""
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    corr = np.correlate(best_h, 1.0 - best_h, mode="full")
    dx = 2.0 / n
    
    T = 1.0  # Temperature for simulated annealing acceptance
    
    for step in range(n_steps):
        if cooling and step % 100 == 0:
            T *= 0.999
        
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        
        delta_max_pos = 1.0 - best_h[i]
        delta_max_neg = best_h[j]
        if delta_max_pos < 1e-10 and delta_max_neg < 1e-10:
            continue
        
        # Compute g vector
        g = np.zeros(2*n-1)
        for s in range(-(n-1), n):
            gi = 0.0
            if 0 <= i+s < n:
                gi += 1.0 - best_h[i+s]
            if 0 <= i-s < n:
                gi -= best_h[i-s]
            gj = 0.0
            if 0 <= j+s < n:
                gj += 1.0 - best_h[j+s]
            if 0 <= j-s < n:
                gj -= best_h[j-s]
            g[n-1+s] = gi - gj
        
        # Minimize max(corr + delta*g) over delta in [-delta_max_neg, delta_max_pos]
        def obj(delta):
            return np.max(corr + delta * g)
        
        try:
            res = minimize_scalar(obj, bounds=(-delta_max_neg, delta_max_pos), 
                                  method='bounded', options={'xatol': 1e-10})
            best_delta = res.x
            new_max_val = res.fun
        except Exception:
            continue
        
        new_c5 = new_max_val * dx
        old_c5 = np.max(corr) * dx
        
        if new_c5 < best_c5 - 1e-14:
            best_h[i] += best_delta
            best_h[j] -= best_delta
            best_h = np.clip(best_h, 0, 1)
            corr = np.correlate(best_h, 1.0 - best_h, mode="full")
            best_c5 = compute_c5(best_h)
        elif T > 0.01 and new_c5 < old_c5 + T * 0.001:
            # Accept worse move for exploration
            best_h[i] += best_delta
            best_h[j] -= best_delta
            best_h = np.clip(best_h, 0, 1)
            corr = np.correlate(best_h, 1.0 - best_h, mode="full")
    
    return best_h, best_c5
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_feasible(y, n):
    target = n / 2.0
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y)) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    h = np.clip(y - t, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-10:
        interior = np.where((h > 1e-8) & (h < 1.0 - 1e-8))[0]
        if len(interior) > 0:
            h[interior] += residual / len(interior)
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = beta * (corr - m)
    shifted = np.clip(shifted, -500, 0)
    exps = np.exp(shifted)
    return (m + np.log(np.sum(exps)) / beta) * dx

def smooth_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = beta * (corr - m)
    shifted = np.clip(shifted, -500, 0)
    exps = np.exp(shifted)
    weights = exps / np.sum(exps)
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        s = idx - (n - 1)
        w = weights[idx]
        if w < 1e-14:
            continue
        lo1 = max(0, -s)
        hi1 = min(n - 1, n - 1 - s)
        if lo1 <= hi1:
            grad[lo1:hi1+1] += w * (1.0 - h[lo1+s:hi1+1+s])
        lo2 = max(0, s)
        hi2 = min(n - 1, n - 1 + s)
        if lo2 <= hi2:
            grad[lo2:hi2+1] -= w * h[lo2-s:hi2+1-s]
    return grad * dx

def slsqp_refine(h, betas, max_iter=200):
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    bounds = [(0.0, 1.0)] * n
    def eq_fun(x):
        return np.sum(x) - n / 2.0
    def eq_jac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    for beta in betas:
        try:
            res = minimize(smooth_obj, best_h, args=(beta,),
                          jac=smooth_grad, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': max_iter, 'ftol': 1e-16})
            if res.x is not None:
                h_new = project_feasible(res.x, n)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5 - 1e-15:
                    best_c5 = c5_new
                    best_h = h_new
        except Exception:
            pass
    return best_h, best_c5

def min_convex_line(corr, g, lo, hi):
    """Minimize max(corr + delta*g) for delta in [lo, hi] using bisection on subgradient"""
    for _ in range(60):
        mid = (lo + hi) / 2.0
        vals = corr + mid * g
        max_val = np.max(vals)
        active = vals >= max_val - 1e-12
        active_g = g[active]
        if np.min(active_g) <= 0 and np.max(active_g) >= 0:
            return mid, max_val
        elif np.min(active_g) > 0:
            hi = mid
        else:
            lo = mid
    delta = (lo + hi) / 2.0
    return delta, np.max(corr + delta * g)

def pairwise_descent(h, n_steps, rng):
    """Random pairwise coordinate descent with exact 1D optimization"""
    n = len(h)
    best_h = h.copy()
    dx = 2.0 / n
    corr = np.correlate(best_h, 1.0 - best_h, mode="full")
    best_c5 = np.max(corr) * dx
    s_arr = np.arange(-(n-1), n)
    
    for step in range(n_steps):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        
        delta_hi = 1.0 - best_h[i]
        delta_lo = -best_h[j]
        if delta_hi <= delta_lo:
            continue
        
        # Compute g = d(corr)/d(h[i]) - d(corr)/d(h[j])
        # d(c_s)/d(h[k]) = (1-h[k+s])*[0<=k+s<n] - h[k-s]*[0<=k-s<n]
        gi = np.zeros(2*n-1)
        mask_i_pos = (i + s_arr >= 0) & (i + s_arr < n)
        gi[mask_i_pos] += 1.0 - best_h[i + s_arr[mask_i_pos]]
        mask_i_neg = (i - s_arr >= 0) & (i - s_arr < n)
        gi[mask_i_neg] -= best_h[i - s_arr[mask_i_neg]]
        
        gj = np.zeros(2*n-1)
        mask_j_pos = (j + s_arr >= 0) & (j + s_arr < n)
        gj[mask_j_pos] += 1.0 - best_h[j + s_arr[mask_j_pos]]
        mask_j_neg = (j - s_arr >= 0) & (j - s_arr < n)
        gj[mask_j_neg] -= best_h[j - s_arr[mask_j_neg]]
        
        g = gi - gj
        
        # Check if g is all zero (no effect)
        if np.max(np.abs(g)) < 1e-15:
            continue
        
        delta, new_max = min_convex_line(corr, g, delta_lo, delta_hi)
        new_c5 = new_max * dx
        
        if new_c5 < best_c5 - 1e-14:
            best_h[i] += delta
            best_h[j] -= delta
            best_h = np.clip(best_h, 0.0, 1.0)
            corr = np.correlate(best_h, 1.0 - best_h, mode="full")
            best_c5 = np.max(corr) * dx
    
    return best_h, best_c5

def full_sweep_descent(h, n_sweeps, rng):
    """Full sweep: iterate through all pairs in random order"""
    n = len(h)
    best_h = h.copy()
    dx = 2.0 / n
    corr = np.correlate(best_h, 1.0 - best_h, mode="full")
    best_c5 = np.max(corr) * dx
    s_arr = np.arange(-(n-1), n)
    
    for sweep in range(n_sweeps):
        order = rng.permutation(n)
        for idx_i in range(n):
            for idx_j in range(n):
                if idx_i == idx_j:
                    continue
                i = order[idx_i]
                j = order[idx_j]
                # Actually this is O(n²) per sweep, too slow
                # Just do random instead
                pass
    
    # This is too slow, let's just do random
    return best_h, best_c5
```

```python
def targeted_pairwise(h, n_steps, rng):
    """Targeted pairwise: pick pairs based on gradient of active lags"""
    n = len(h)
    best_h = h.copy()
    dx = 2.0 / n
    corr = np.correlate(best_h, 1.0 - best_h, mode="full")
    best_c5 = np.max(corr) * dx
    s_arr = np.arange(-(n-1), n)
    
    for step in range(n_steps):
        # Find active lags
        max_corr = np.max(corr)
        active = np.where(corr >= max_corr - 0.01)[0]
        
        # Compute gradient of the "average" active correlation
        grad = np.zeros(n)
        for idx in active:
            s = idx - (n-1)
            lo1 = max(0, -s)
            hi1 = min(n-1, n-1-s)
            if lo1 <= hi1:
                grad[lo1:hi1+1] += 1.0 - best_h[lo1+s:hi1+1+s]
            lo2 = max(0, s)
            hi2 = min(n-1, n-1+s)
            if lo2 <= hi2:
                grad[lo2:hi2+1] -= best_h[lo2-s:hi2+1-s]
        
        # Pick i from indices with most negative grad, j from most positive
        if np.min(grad) < -1e-6:
            i = np.argmin(grad)
        else:
            i = rng.randint(n)
        if np.max(grad) > 1e-6:
            j = np.argmax(grad)
        else:
            j = rng.randint(n)
        if i == j:
            j = (j + 1) % n
            if i == j:
                continue
        
        # Same as before: compute g, find optimal delta
        delta_hi = 1.0 - best_h[i]
        delta_lo = -best_h[j]
        if delta_hi <= delta_lo:
            continue
        
        gi = np.zeros(2*n-1)
        mask_i_pos = (i + s_arr >= 0) & (i + s_arr < n)
        gi[mask_i_pos] += 1.0 - best_h[i + s_arr[mask_i_pos]]
        mask_i_neg = (i - s_arr >= 0) & (i - s_arr < n)
        gi[mask_i_neg] -= best_h[i - s_arr[mask_i_neg]]
        
        gj = np.zeros(2*n-1)
        mask_j_pos = (j + s_arr >= 0) & (j + s_arr < n)
        gj[mask_j_pos] += 1.0 - best_h[j + s_arr[mask_j_pos]]
        mask_j_neg = (j - s_arr >= 0) & (j - s_arr < n)
        gj[mask_j_neg] -= best_h[j - s_arr[mask_j_neg]]
        
        g = gi - gj
        if np.max(np.abs(g)) < 1e-15:
            continue
        
        delta, new_max = min_convex_line(corr, g, delta_lo, delta_hi)
        new_c5 = new_max * dx
        
        if new_c5 < best_c5 - 1e-14:
            best_h[i] += delta
            best_h[j] -= delta

            corr = np.correlate(best_h, 1.0 - best_h, mode="full")
            best_c5 = np.max(corr) * dx
    
    return best_h, best_c5
```
