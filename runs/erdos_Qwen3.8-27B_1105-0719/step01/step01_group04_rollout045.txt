import numpy as np
from scipy.optimize import minimize
import time

try:
    import cvxpy as cp
    HAS_CP = True
except Exception:
    HAS_CP = False


def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr * dx))


def repair_feasible(vec):
    h = np.asarray(vec, dtype=np.float64)
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h.copy()
    if diff > 0:
        order = np.argsort(h)
        rem = diff
        for i in order:
            if rem <= 1e-15:
                break
            add = min(rem, 1.0 - h[i])
            h[i] += add
            rem -= add
    else:
        order = np.argsort(-h)
        rem = -diff
        for i in order:
            if rem <= 1e-15:
                break
            remv = min(rem, h[i])
            h[i] -= remv
            rem -= remv
    diff2 = target - h.sum()
    if abs(diff2) > 1e-12:
        for i in range(n):
            if diff2 > 0 and h[i] < 1.0:
                h[i] += diff2
                break
            elif diff2 < 0 and h[i] > 0.0:
                h[i] += diff2
                break
    return h


def compute_lag_data(h):
    n = len(h)
    dx = 2.0 / n
    n_lags = 2 * n - 1
    values = np.zeros(n_lags)
    grads = np.zeros((n_lags, n))
    for delta in range(0, n):
        if delta == 0:
            values[n - 1] = dx * np.sum(h * (1.0 - h))
            grads[n - 1, :] = dx * (1.0 - 2.0 * h)
        else:
            idx = np.arange(n - delta)
            values[n - 1 + delta] = dx * np.sum(h[idx] * (1.0 - h[idx + delta]))
            grads[n - 1 + delta, idx] += dx * (1.0 - h[idx + delta])
            grads[n - 1 + delta, idx + delta] += -dx * h[idx]
            idx2 = np.arange(delta, n)
            values[n - 1 - delta] = dx * np.sum(h[idx2] * (1.0 - h[idx2 - delta]))
            grads[n - 1 - delta, idx2] += dx * (1.0 - h[idx2 - delta])
            grads[n - 1 - delta, idx2 - delta] += -dx * h[idx2]
    return values, grads


def interpolate_to_dimension(vec, new_len):
    n = len(vec)
    if n == new_len:
        return vec.copy()
    x_src = (np.arange(n) + 0.5) * 2.0 / n
    x_tgt = (np.arange(new_len) + 0.5) * 2.0 / new_len
    xs_ext = np.concatenate([[0.0], x_src, [2.0]])
    ys_ext = np.concatenate([[vec[0]], vec, [vec[-1]]])
    return np.interp(x_tgt, xs_ext, ys_ext)


def trust_region_qp(h_cur, h_prod, lambda_val, max_active=None, timeout_s=5.0):
    n = len(h_cur)
    h_prod = h_prod + h_cur
    values, grads = compute_lag_data(h_prod)
    n_lags = len(values)
    if max_active is not None and max_active < n_lags:
        top_idx = np.argsort(values)[::-1][:max_active]
    else:
        top_idx = np.arange(n_lags)
    top_idx = np.sort(top_idx)
    v_act = values[top_idx]
    g_act = grads[top_idx, :]
    if HAS_CP:
        try:
            h_var = cp.Variable(n)
            t_var = cp.Variable()
            lin_terms = v_act + g_act @ (h_var - h_prod)
            obj = cp.Minimize(t_var + lambda_val * cp.sum_squares(h_var - h_cur))
            constraints = [
                lin_terms <= t_var,
                cp.sum(h_var) == n / 2.0,
                h_var >= 0,
                h_var <= 1
            ]
            prob = cp.Problem(obj, constraints)
            solver_list = ['ECOS', 'SCS', 'OSQP', 'CLARABEL']
            solved = False
            for slv in solver_list:
                try:
                    prob.solve(solver=slv, max_time=timeout_s)
                    if prob.status in ('optimal', 'optimal_inaccurate') and h_var.value is not None:
                        solved = True
                        break
                except Exception:
                    continue
            if solved:
                return np.array(h_var.value, dtype=np.float64)
        except Exception:
            pass
    return None


def active_descent(h, max_steps=50, sigma_scale=0.004, n_active=20):
    h = repair_feasible(h)
    best = h.copy()
    best_score = exact_score(best)
    for step in range(max_steps):
        values, grads = compute_profile_and_gradients(best)
        c = np.max(values)
        na = min(n_active, 2 * len(best) - 1)
        active_idx = np.argsort(values)[::-1][:na]
        gaps = np.maximum(c - values[active_idx], 0.0)
        sigma = sigma_scale * max(c, 1e-10)
        weights = np.exp(-gaps / (sigma + 1e-15))
        weights /= np.sum(weights) + 1e-15
        direction = -np.sum(weights[:, None] * grads[active_idx, :], axis=0)
        direction -= np.mean(direction)
        norm_d = np.linalg.norm(direction)
        if norm_d < 1e-14:
            break
        direction /= norm_d
        improved = False
        for alpha in [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]:
            h_trial = repair_feasible(best + alpha * direction)
            score = exact_score(h_trial)
            if score < best_score - 1e-14:
                best = h_trial.copy()
                best_score = score
                improved = True
                break
        if not improved:
            break
    return best, best_score

def slsqp_epigraph(h_start, beta=0.0, maxiter=150, active_mask=None):
    n = len(h_start)
    h_start = repair_feasible(h_start)
    c_start = exact_score(h_start)
    t0 = c_start + 0.05
    x0 = np.concatenate([h_start, [t0]])
    dx = 2.0 / n
    n_lags = 2 * n - 1

    def obj(x):
        return x[n] + beta * np.sum(x[:n] * (1.0 - x[:n]))

    def obj_jac(x):
        g = np.zeros(n + 1)
        if beta > 0:
            g[:n] = beta * (1.0 - 2.0 * x[:n])
        g[n] = 1.0
        return g

    def eq_fun(x):
        return np.sum(x[:n]) - n / 2.0

    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:n] = 1.0
        return g

    def ineq_fun(x):
        h = x[:n]
        t = x[n]
        vals, _ = compute_profile_and_gradients(h)
        if active_mask is not None:
            return t - vals[active_mask]
        return t - vals

    def ineq_jac(x):
        h = x[:n]
        _, grads = compute_profile_and_gradients(h)
        if active_mask is not None:
            g = np.zeros((len(active_mask), n + 1))
            g[:, n] = 1.0
            g[:, :n] = -grads[active_mask, :]
            return g
        g = np.zeros((n_lags, n + 1))
        g[:, n] = 1.0
        g[:, :n] = -grads
        return g

    bounds = [(0.0, 1.0)] * n + [(0.0, t0 + 1.0)]
    constraints = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    try:
        result = minimize(
            obj, x0, jac=obj_jac,
            method='SLSQP', bounds=bounds, constraints=constraints,
            options={'maxiter': maxiter, 'ftol': 1e-14}
        )
    except Exception:
        return None, None
    if result.x is not None:
        h_sol = repair_feasible(result.x[:n])
        score = exact_score(h_sol)
        if np.isfinite(score):
            return h_sol, score
    return None, None

        for nd in dim_candidates:
            if time_left() < 40:
                break
            h_interp = repair_feasible(interpolate_to_dimension(best_h, nd))
            s_interp = exact_score(h_interp)
            if s_interp < best_score + 0.01:
                h_new, s_new = trust_region_loop(h_interp, max_iters=60, lambda_init=0.03, deadline=deadline)
                if s_new < best_score - 1e-14:
                    best_h = h_new
                    best_score = s_new
                    best_n = len(h_new)
                h_new2, s_new2 = active_descent(h_interp, max_steps=30, deadline=deadline)
                if s_new2 < best_score - 1e-14:
                    best_h = h_new2
                    best_score = s_new2
                    best_n = len(h_new2)

    # Phase 6: Subgradient + active descent combined
    if time_left() > 50:
        h_new, s_new = subgradient_descent(best_h, max_iters=200, deadline=deadline)
        if s_new < best_score - 1e-14:
            best_h = h_new
            best_score = s_new
            best_n = len(h_new)

    # Phase 7: Random perturbations + trust region
    if time_left() > 60:
        for trial in range(6):
            if time_left() < 30:
                break
            eps = rng.uniform(0.003, 0.015)
            z = rng.randn(len(best_h))
            z -= z.mean()
            h_pert = repair_feasible(best_h + eps * z)
            h_new, s_new = trust_region_loop(h_pert, max_iters=50, lambda_init=0.02, deadline=deadline)
            if s_new < best_score - 1e-14:
                best_h = h_new
                best_score = s_new
                best_n = len(h_new)

    # Phase 8: Symmetrize
    if time_left() > 20:
        h_sym = repair_feasible((best_h + best_h[::-1]) / 2.0)
        s_sym = exact_score(h_sym)
        if s_sym < best_score - 1e-14:
            best_h = h_sym
            best_score = s_sym
            best_n = len(h_sym)

    # Phase 9: Final fine polish - alternating active descent and SLSQP
    if time_left() > 40:
        for round_i in range(3):
            if time_left() < 20:
                break
            h_new, s_new = active_descent(best_h, max_steps=25, sigma_scale=0.002, n_active=12)
            if s_new < best_score - 1e-14:
                best_h = h_new
                best_score = s_new
                best_n = len(h_new)
            if time_left() < 15:
                break
            values, _ = compute_profile_and_gradients(best_h)
            top_idx = np.argsort(values)[::-1][:25]
            top_idx = np.sort(top_idx)
            h_new2, s_new2 = slsqp_epigraph(best_h, beta=1e-6, maxiter=80, active_mask=top_idx)
            if h_new2 is not None and s_new2 < best_score - 1e-14:
                best_h = h_new2
                best_score = s_new2
                best_n = len(h_new2)

    # Phase 10: One final trust region sweep
    if time_left() > 30:
        h_new, s_new = trust_region_loop(best_h, max_iters=100, lambda_init=0.01, deadline=deadline)
        if s_new < best_score - 1e-14:
            best_h = h_new
            best_score = s_new
            best_n = len(h_new)

    # Final validation
    best_h = repair_feasible(best_h)
    best_score = exact_score(best_h)
    return best_h, best_score, len(best_h)
