def repair_feasible(h, target=None):

def lse_grad(h, tau):
    """Compute gradient of the log-sum-exp of the profile."""
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    prof = get_profile(h)
    
    # Numerically stable softmax
    max_p = np.max(prof)
    exps = np.exp(tau * (prof - max_p))
    s = np.sum(exps)
    w = exps / s
    
    grad = np.zeros(n)
    for k in range(M):
        wk = w[k]
        if wk < 1e-14:
            continue
        delta = k - (n - 1)
        if delta >= 0:
            if delta < n:
                grad[delta:n] += wk * dx * (1.0 - h[:n-delta])
                grad[:n-delta] -= wk * dx * h[delta:]
        else:
            d = -delta
            if d < n:
                grad[:n-d] += wk * dx * (1.0 - h[d:])
                grad[d:] -= wk * dx * h[:n-d]

    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    count = 0

        if delta >= 0:

            if d < n:
                grad[:n-d] += dx * (1.0 - h[d:])
                grad[d:] -= dx * h[:n-d]
                count += 1
    if count > 0:

    return grad

def solve_lse(h_start, tau, maxiter=50):
    """Solve using log-sum-exp smooth objective with SLSQP."""
    n = len(h_start)
    
    def obj(x):
        prof = get_profile(x)
        max_p = np.max(prof)
        return float(np.log(np.sum(np.exp(tau * (prof - max_p)))) + tau * max_p) / tau
    
    def obj_jac(x):
        return lse_grad(x, tau)
    
    def cons_eq(x):
        return x.sum() - n / 2.0
    
    bounds = [(0.0, 1.0)] * n
    
    try:
        result = minimize(
            obj, h_start, jac=obj_jac,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': cons_eq, 'jac': lambda x: np.ones(n)}],
            options={'maxiter': maxiter, 'ftol': 1e-12}

def solve_epigraph_full(h_start, t_start, maxiter=100, n_constraints=None):

    else:

        return x[n]

        g = np.zeros(n + 1)
        g[n] = 1.0
        return g

        t = x[n]
        eq = h.sum() - n / 2.0
        prof = get_profile(h)
        ineq = t - prof[active_lags]
        return np.concatenate([[eq], ineq])

def exact_score(h):

def get_profile(h):

def repair_feasible(h, target=None):
    h = np.array(h, dtype=np.float64).copy()

    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        order = np.argsort(-room)
        remaining = diff
        for idx in order:
            if remaining <= 1e-15:
                break
            add = min(remaining, room[idx])
            h[idx] += add
            remaining -= add
    else:

        remaining = -diff
        for idx in order:
            if remaining <= 1e-15:

            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
    return h

def lse_compute(h, tau):

    dx = 2.0 / n

    exps = np.exp(tau * (prof - max_p))
    s = np.sum(exps)
    w = exps / s
    f_val = (np.log(s) + tau * max_p) / tau
    
    grad = np.zeros(n)
    for k in range(M):
        wk = w[k]
        if wk < 1e-14:
            continue
        delta = k - (n - 1)
        if delta >= 0:

        else:

            if d < n:

def solve_lse(h_start, tau, maxiter=50):
    """SLSQP on log-sum-exp smooth surrogate."""
    n = len(h_start)
    
    def obj(x):
        prof = get_profile(x)
        max_p = np.max(prof)
        return float(np.log(np.sum(np.exp(tau * (prof - max_p)))) + tau * max_p) / tau

        _, g = lse_compute(x, tau)
        return g
    
    bounds = [(0.0, 1.0)] * n
    
    try:
        result = minimize(
            obj, h_start, jac=obj_jac,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0,
                          'jac': lambda x: np.ones(n)}],
            options={'maxiter': maxiter, 'ftol': 1e-11}
        )
        if result.x is not None and np.all(np.isfinite(result.x)):

    except Exception:
        pass
    return None

def solve_epigraph(h_start, t_start, maxiter=100, n_constraints=None):

    n = len(h_start)
    M = 2 * n - 1
    nc = n_constraints if n_constraints is not None else M
    
    prof = get_profile(h_start)
    if nc < M:
        active_lags = np.argsort(prof)[::-1][:nc]

        active_lags = np.arange(M)

        return x[n]

        g = np.zeros(n + 1)
        g[n] = 1.0
        return g

        h_loc, s_loc = local_gradient_descent(best_h, max_iters=80, time_limit=tl)
        if s_loc < best_score - 1e-12:
            best_h = h_loc
            best_score = s_loc
    
    # Phase 4: High-temperature LSE for refinement
    if tleft() > 50:
        for tau in [1500, 5000]:
            if tleft() < 20:
                break
            mi = min(60, max(20, int(tleft() * 0.07)))
            h_sol = solve_lse(best_h, tau, maxiter=mi)
            if h_sol is not None:
                s = exact_score(h_sol)
                if s < best_score - 1e-12:
                    best_h = h_sol
                    best_score = s
    
    # Phase 5: Diversified restarts with LSE + epigraph
    if tleft() > 100:
        n = len(best_h)
        rng = np.random.RandomState(seed)
        for p in range(20):
            if tleft() < 45:
                break
            eps = 0.002 + 0.0015 * p
            z = rng.randn(n)
            z -= z.mean()
            nz = np.linalg.norm(z)
            if nz > 1e-10:
                z /= nz
            h_pert = repair_feasible(best_h + eps * z)
            s_pert = exact_score(h_pert)
            
            # LSE on perturbed
            h_sol = solve_lse(h_pert, 30, maxiter=35)
            if h_sol is not None:
                s = exact_score(h_sol)
                if s < best_score - 1e-12:
                    best_h = h_sol
                    best_score = s
                    s_pert = s
            
            # Epigraph on perturbed
            if tleft() > 55:
                nc = min(2*n-1, 80)
                h_sol2 = solve_epigraph(h_pert, s_pert + 1e-5, maxiter=45, n_constraints=nc)
                if h_sol2 is not None:
                    s2 = exact_score(h_sol2)
                    if s2 < best_score - 1e-12:
                        best_h = h_sol2
                        best_score = s2
    
    # Phase 6: Try different dimensions
    if tleft() > 70:
        pn = len(parent)
        dims = set()
        for d in [pn-12, pn-6, pn+6, pn+12, pn+24]:
            if 48 <= d <= 192 and d != pn:
                dims.add(d)
        if pn * 2 <= 192:
            dims.add(pn * 2)
        
        for new_n in sorted(dims):
            if tleft() < 35:
                break
            h_interp = interpolate_to_dim(best_h, new_n)
            s_interp = exact_score(h_interp)
            
            h_sol = solve_lse(h_interp, 15, maxiter=40)
            if h_sol is not None:
                s = exact_score(h_sol)
                if s < best_score - 1e-12:
                    best_h = h_sol
                    best_score = s
            
            if tleft() > 35:
                nc = min(2*new_n-1, 100)
                h_sol2 = solve_epigraph(best_h, best_score + 1e-5, maxiter=50, n_constraints=nc)
                if h_sol2 is not None:
                    s2 = exact_score(h_sol2)
                    if s2 < best_score - 1e-12:
                        best_h = h_sol2
                        best_score = s2
    
    # Phase 7: Final epigraph with all constraints
    if tleft() > 25:
        n = len(best_h)
        M = 2 * n - 1
        mi = min(200, max(50, int(tleft() * 0.3)))
        h_sol = solve_epigraph(best_h, best_score + 1e-7, maxiter=mi, n_constraints=M)
        if h_sol is not None:
            s = exact_score(h_sol)
            if s < best_score - 1e-12:
                best_h = h_sol
                best_score = s
    
    # Phase 8: Final local refinement
    if tleft() > 12:
        tl = min(12, tleft() * 0.7)
        h_fin, s_fin = local_gradient_descent(best_h, max_iters=50, time_limit=tl)
        if s_fin < best_score - 1e-12:
            best_h = h_fin
            best_score = s_fin
    
    # Phase 9: One more epigraph pass if time
    if tleft() > 20:
        h_sol = solve_epigraph(best_h, best_score + 1e-8, maxiter=80, n_constraints=60)
        if h_sol is not None:
            s = exact_score(h_sol)
            if s < best_score - 1e-12:
                best_h = h_sol
                best_score = s
    
    # Final repair and score
    best_h = repair_feasible(best_h)
    final_score = exact_score(best_h)
    
    return (best_h, final_score, len(best_h))
