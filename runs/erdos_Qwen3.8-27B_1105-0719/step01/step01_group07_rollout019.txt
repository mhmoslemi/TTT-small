def sa_binary(b0, n, deadline, rng, max_rounds=50000):
    """Simulated annealing on binary sequences with exactly n/2 ones."""
    b = b0.copy()
    n = len(b)
    m = n // 2
    dx = 2.0 / n
    
    curr_c = float(np.max(np.correlate(b, 1-b, mode='full'))) * dx
    best_c = curr_c
    best_b = b.copy()
    
    ones = list(np.where(b == 1)[0])
    zeros = list(np.where(b == 0)[0])
    
    T = 0.02
    T_min = 0.0001
    cooling = 0.9997
    
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        
        # Choose move type
        if rng.random() < 0.85:
            # Single swap
            i = rng.choice(ones)
            j = rng.choice(zeros)
            b[i], b[j] = b[j], b[i]
            new_c = float(np.max(np.correlate(b, 1-b, mode='full'))) * dx
            delta = new_c - curr_c
            if delta < 0 or rng.random() < np.exp(-delta / T):
                curr_c = new_c
                if curr_c < best_c:
                    best_c = curr_c
                    best_b = b.copy()
                # Update ones and zeros
                ones.remove(i)
                ones.append(j)
                zeros.remove(j)
                zeros.append(i)
            else:
                b[i], b[j] = b[j], b[i]  # revert
        else:
            # Double swap (2-for-2)
            if len(ones) >= 2 and len(zeros) >= 2:
                i1 = rng.choice(ones)
                i2 = rng.choice(ones)
                while i2 == i1:
                    i2 = rng.choice(ones)
                j1 = rng.choice(zeros)
                j2 = rng.choice(zeros)
                while j2 == j1:
                    j2 = rng.choice(zeros)
                
                b[i1] = b[j1] = 0
                b[i2] = b[j2] = 1
                new_c = float(np.max(np.correlate(b, 1-b, mode='full'))) * dx
                delta = new_c - curr_c
                if delta < 0 or rng.random() < np.exp(-delta / T):
                    curr_c = new_c
                    if curr_c < best_c:
                        best_c = curr_c
                        best_b = b.copy()
                    ones.remove(i1)
                    ones.remove(i2)
                    ones.extend([j1, j2])
                    zeros.remove(j1)
                    zeros.remove(j2)
                    zeros.extend([i1, i2])

                    b[i2] = b[j2] = 0
        
        T *= cooling
        if T < T_min:
            T = T_min
