    for freq in [1, 2, 3, 4]:
        h = 0.5 + 0.3 * np.sin(2 * np.pi * freq * x / 2.0)
        starts.append(repair_feasible(np.clip(h, 0, 1)))
    
    # Block patterns
    for block_size in [2, 3, 4, 5]:
        n_blocks = n // block_size
        for phase in range(block_size):
            h = np.zeros(n)
            for i in range(n):
                if (i // block_size + phase) % 2 == 0:
                    h[i] = 1.0
            starts.append(repair_feasible(h))
    
    # Random
    for _ in range(3):
        h = rng.rand(n)
        starts.append(repair_feasible(h))
    
    return starts

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()

    rng = np.random.RandomState(seed)

        return time_budget + t_start - time.perf_counter()

    parent = None
    try:
        parent = np.array(initial_h_values, dtype=np.float64)

        pass
    except Exception:

        parent = np.full(66, 0.5)

    parent = repair_feasible(parent)
    c_parent = exact_score(parent)
    
    best_h = parent.copy()
    best_c = c_parent
    best_n = len(parent)
    
    # Quick symmetrization
    h_sym = repair_feasible(symmetrize(parent))
    c_sym = exact_score(h_sym)
    if c_sym < best_c - 1e-12:
        best_h = h_sym.copy()
        best_c = c_sym

    beta_schedules = [
        [5, 10, 20, 50, 100, 200, 500],
        [10, 20, 50, 100, 300, 1000],
        [3, 7, 15, 30, 60, 150, 400],
    ]
    
    for schedule in beta_schedules:
        if time_left() < 40:
            break
        h_lse, c_lse = lse_projected_descent(best_h, schedule, max_iter=150)
        if c_lse < best_c - 1e-12:
            best_h = h_lse.copy()
            best_c = c_lse

    if time_left() > 60:

                    best_c = c_lse
