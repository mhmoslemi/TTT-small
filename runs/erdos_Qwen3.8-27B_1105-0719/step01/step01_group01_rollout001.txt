def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    """Project y to [0,1]^n with sum = n/2"""
    target = n / 2.0
    y = np.asarray(y, dtype=float)
    if y.shape != (n,):
        y = y[:n] if len(y) >= n else np.pad(y, (0, n - len(y)))
    y = np.nan_to_num(y, nan=0.5, posinf=1.0, neginf=0.0)
    lo = np.min(y) - 2.0
    hi = np.max(y) + 1.0

        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    h = np.clip(y - t, 0.0, 1.0)
    residual = target - np.sum(h)

def smooth_max_objective(h, beta):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = beta * (corr - m)
    exps = np.exp(shifted)
    s = np.sum(exps)
    return (m + np.log(s) / beta) * (2.0 / len(h))

def smooth_max_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = beta * (corr - m)
    exps = np.exp(shifted)
    s = np.sum(exps)
    weights = exps / s
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        k = idx - (n - 1)
        w = weights[idx]
        if w < 1e-14:
            continue
        m_lo1 = max(0, k)
        m_hi1 = min(n - 1, n - 1 + k)
        if m_lo1 <= m_hi1:
            grad[m_lo1:m_hi1 + 1] += w * (1.0 - h[m_lo1 - k:m_hi1 + 1 - k])
        m_lo2 = max(0, -k)
        m_hi2 = min(n - 1, n - 1 - k)
        if m_lo2 <= m_hi2:
            grad[m_lo2:m_hi2 + 1] -= w * h[m_lo2 + k:m_hi2 + 1 + k]
    return grad * dx

def slsqp_step(h, beta, max_iter=200):
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0,
             'jac': lambda x: np.ones(n)}]
    res = minimize(smooth_max_objective, h, args=(beta,),
                   jac=smooth_max_grad, method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-15})
    if res.success or res.x is not None:
        return res.x
    return h

def map_resolution(h_old, n_old, n_new):
    """Map from n_old resolution to n_new resolution using exact overlap"""
    dx_old = 2.0 / n_old
    dx_new = 2.0 / n_new
    
    h_new = np.zeros(n_new)
    # For each new interval, compute the average height from old intervals
    for j in range(n_new):
        # New interval: [j*dx_new, (j+1)*dx_new]
        a = j * dx_new
        b = (j + 1) * dx_new
        total_weighted = 0.0
        total_weight = 0.0
        for i in range(n_old):
            # Old interval: [i*dx_old, (i+1)*dx_old]
            oa = i * dx_old
            ob = (i + 1) * dx_old
            # Overlap
            lo = max(a, oa)
            hi = min(b, ob)
            if hi > lo:
                overlap = hi - lo
                total_weighted += h_old[i] * overlap
                total_weight += overlap
        if total_weight > 1e-12:
            h_new[j] = total_weighted / total_weight
        else:
            h_new[j] = 0.5
    
    return project_feasible(h_new, n_new)

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_feasible(y, n):
    target = n / 2.0
    y = np.asarray(y, dtype=float).copy()
    if y.shape != (n,):
        if len(y) >= n:
            y = y[:n]
        else:
            y = np.concatenate([y, np.full(n - len(y), 0.5)])
    y = np.nan_to_num(y, nan=0.5, posinf=1.0, neginf=0.0)
    lo = np.min(y) - 2.0
    hi = np.max(y) + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    h = np.clip(y - t, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-10:
        eligible = np.where((h > 1e-8) & (h < 1.0 - 1e-8))[0]
        if len(eligible) > 0:
            h[eligible] += residual / len(eligible)
            h = np.clip(h, 0.0, 1.0)
        else:
            h += residual / n
            h = np.clip(h, 0.0, 1.0)
    return h


def pair_exchange_descent(h, n_rounds=300, rng=None):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    n = len(best_h)
    if rng is None:
        rng = np.random.RandomState(0)
    fracs = np.array([0.15, 0.3, 0.5, 0.7, 0.85, 1.0, 0.08, 0.42, 0.65, 0.92])
    for _ in range(n_rounds):
        changed = False
        for _ in range(150):
            i, j = rng.choice(n, size=2, replace=False)
            if abs(best_h[i] - best_h[j]) < 1e-8:
                continue
            cap = min(1.0 - best_h[i], best_h[j])
            if cap < 1e-8:
                continue
            for frac in fracs:
                tau = cap * frac
                h_new = best_h.copy()
                h_new[i] += tau
                h_new[j] -= tau
                if h_new[i] < -1e-10 or h_new[i] > 1.0 + 1e-10:
                    continue
                if h_new[j] < -1e-10 or h_new[j] > 1.0 + 1e-10:
                    continue
                c5_new = compute_c5(h_new)
                if c5_new < best_c5 - 1e-14:
                    best_h = h_new
                    best_c5 = c5_new
                    changed = True
                    break
        if not changed:
            # Try reverse direction
            for _ in range(100):
                i, j = rng.choice(n, size=2, replace=False)
                if abs(best_h[i] - best_h[j]) < 1e-8:
                    continue
                cap = min(1.0 - best_h[j], best_h[i])
                if cap < 1e-8:
                    continue
                for frac in fracs:
                    tau = cap * frac
                    h_new = best_h.copy()
                    h_new[j] += tau
                    h_new[i] -= tau
                    if h_new[j] < -1e-10 or h_new[j] > 1.0 + 1e-10:
                        continue
                    if h_new[i] < -1e-10 or h_new[i] > 1.0 + 1e-10:
                        continue
                    c5_new = compute_c5(h_new)
                    if c5_new < best_c5 - 1e-14:
                        best_h = h_new
                        best_c5 = c5_new
                        changed = True
                        break
            if not changed:
                break
    return best_h, best_c5


def smooth_max_obj(h, beta):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = beta * (corr - m)
    exps = np.exp(shifted)
    s = np.sum(exps)
    return (m + np.log(s) / beta) * (2.0 / len(h))


def smooth_max_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = beta * (corr - m)
    exps = np.exp(shifted)
    s = np.sum(exps)
    weights = exps / s
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        k = idx - (n - 1)
        w = weights[idx]
        if w < 1e-14:
            continue
        m_lo1 = max(0, k)
        m_hi1 = min(n - 1, n - 1 + k)
        if m_lo1 <= m_hi1:
            grad[m_lo1:m_hi1 + 1] += w * (1.0 - h[m_lo1 - k:m_hi1 + 1 - k])
        m_lo2 = max(0, -k)
        m_hi2 = min(n - 1, n - 1 - k)
        if m_lo2 <= m_hi2:
            grad[m_lo2:m_hi2 + 1] -= w * h[m_lo2 + k:m_hi2 + 1 + k]
    return grad * dx


def eq_con_fun(x):
    return np.sum(x) - len(x) / 2.0


def eq_con_jac(x):
    return np.ones(len(x))


def slsqp_step(h, beta, max_iter=200):
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_con_fun, 'jac': eq_con_jac}]
    try:
        res = minimize(smooth_max_obj, h, args=(beta,),
                       jac=smooth_max_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-15})
        if res.x is not None:
            return res.x
    except Exception:
        pass
    return h


def simulated_annealing(h, t0, t_final, n_moves, rng):
    cur_h = h.copy()
    cur_c5 = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c5 = cur_c5
    n = len(cur_h)
    for move_idx in range(n_moves):
        progress = move_idx / max(n_moves - 1, 1)
        T = t0 * (t_final / t0) ** progress
        i, j = rng.choice(n, size=2, replace=False)
        if rng.random() < 0.5:
            inc, dec = i, j
        else:
            inc, dec = j, i
        cap = min(1.0 - cur_h[inc], cur_h[dec])
        if cap < 1e-8:
            continue
        tau = cap * (rng.random() ** 0.4)
        if tau < 1e-6:
            tau = cap * 0.5
        h_new = cur_h.copy()
        h_new[inc] = np.clip(cur_h[inc] + tau, 0.0, 1.0)
        h_new[dec] = np.clip(cur_h[dec] - tau, 0.0, 1.0)
        c5_new = compute_c5(h_new)
        delta = c5_new - cur_c5
        if delta < 0 or (T > 1e-15 and rng.random() < np.exp(-delta / T)):
            cur_h = h_new
            cur_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = cur_h.copy()
    return best_h, best_c5


def make_seeds(parent_h, rng):
    seeds = []
    n = len(parent_h)
    seeds.append(parent_h.copy())
    comp = project_feasible(1.0 - parent_h, n)
    seeds.append(comp)
    seeds.append(parent_h[::-1].copy())
    for s in [1, 2, 5, 10]:
        seeds.append(np.roll(parent_h, s))
    for _ in range(5):
        n_ones = n // 2
        idx = rng.choice(n, size=n_ones, replace=False)
        b = np.zeros(n)
        b[idx] = 1.0
        seeds.append(b)
    for k in [3, 5, 7]:
        kernel = np.ones(k) / k
        sm = np.convolve(parent_h, kernel, mode='same')
        seeds.append(project_feasible(sm, n))
    for freq in [1, 2, 3, 5]:
        x = np.arange(n)
        s = 0.5 + 0.5 * np.sin(2.0 * np.pi * freq * x / n + rng.uniform(0, 2 * np.pi))
        seeds.append(project_feasible(s, n))
    for _ in range(4):
        pert = parent_h + 0.12 * rng.randn(n)
        seeds.append(project_feasible(pert, n))
    for thresh in [0.3, 0.45, 0.55, 0.7]:
        t = (parent_h > thresh).astype(float)
        ones = int(np.sum(t))
        target_ones = n // 2
        if ones > target_ones:
            one_idx = np.where(t == 1)[0]
            to_off = rng.choice(one_idx, size=ones - target_ones, replace=False)
            t[to_off] = 0.0
        elif ones < target_ones:
            zero_idx = np.where(t == 0)[0]
            to_on = rng.choice(zero_idx, size=target_ones - ones, replace=False)
            t[to_on] = 1.0
        seeds.append(t)
    # Block patterns
    for block_len in [3, 4, 5]:
        pattern = []
        for i in range(n // (2 * block_len)):
            pattern.extend([1.0] * block_len)
            pattern.extend([0.0] * block_len)
        while len(pattern) < n:
            pattern.append(0.0)
        pattern = np.array(pattern[:n])
        ones = int(np.sum(pattern))
        if ones > n // 2:
            one_idx = np.where(pattern == 1)[0]
            to_off = rng.choice(one_idx, size=ones - n // 2, replace=False)
            pattern[to_off] = 0.0
        elif ones < n // 2:
            zero_idx = np.where(pattern == 0)[0]
            to_on = rng.choice(zero_idx, size=n // 2 - ones, replace=False)
            pattern[to_on] = 1.0
        seeds.append(pattern)
    return seeds


def map_resolution(h_old, n_old, n_new):
    dx_old = 2.0 / n_old
    dx_new = 2.0 / n_new
    h_new = np.zeros(n_new)
    for j in range(n_new):
        a = j * dx_new
        b = (j + 1) * dx_new
        total_weighted = 0.0
        total_weight = 0.0
        for i in range(n_old):
            oa = i * dx_old
            ob = (i + 1) * dx_old
            lo = max(a, oa)
            hi = min(b, ob)
            if hi > lo:
                overlap = hi - lo
                total_weighted += h_old[i] * overlap
                total_weight += overlap
        if total_weight > 1e-12:
            h_new[j] = total_weighted / total_weight
        else:
            h_new[j] = 0.5
    return project_feasible(h_new, n_new)


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t_start = time.time()

    def time_left():
        return budget_s - (time.time() - t_start)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        parent_h = np.array(initial_h_values, dtype=float).copy()
    else:
        parent_h = np.full(60, 0.5)

    n = len(parent_h)
    parent_c5 = compute_c5(parent_h)
    best_h = parent_h.copy()
    best_c5 = parent_c5
    best_n = n

    # Phase 1: Quick pair-exchange on parent
    if time_left() > 30:
        h_opt, c5_opt = pair_exchange_descent(best_h, n_rounds=150, rng=rng)
        if c5_opt < best_c5:
            best_h, best_c5, best_n = h_opt, c5_opt, len(h_opt)

    # Phase 2: SLSQP refinement
    if time_left() > 40:
        for beta in [50, 200, 1000, 5000, 20000]:
            if time_left() < 20:
                break
            h_new = slsqp_step(best_h, beta, max_iter=200)
            h_new = project_feasible(h_new, len(h_new))
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h, best_c5, best_n = h_new, c5_new, len(h_new)
            if time_left() > 15:
                h_opt, c5_opt = pair_exchange_descent(best_h, n_rounds=80, rng=rng)
                if c5_opt < best_c5:
                    best_h, best_c5, best_n = h_opt, c5_opt, len(h_opt)

    # Phase 3: Multi-seed exploration
    if time_left() > 50:
        seeds = make_seeds(best_h, rng)
        scored = []
        for s in seeds:
            s = project_feasible(s, len(s))
            sc = compute_c5(s)
            scored.append((sc, s))
        scored.sort(key=lambda x: x[0])
        for sc, s in scored:
            if time_left() < 35:
                break
            if sc < best_c5:
                best_h, best_c5, best_n = s.copy(), sc, len(s)
            for beta in [200, 5000]:
                if time_left() < 20:
                    break
                h_new = slsqp_step(s, beta, max_iter=150)
                h_new = project_feasible(h_new, len(h_new))
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_h, best_c5, best_n = h_new, c5_new, len(h_new)
            if time_left() > 15:
                h_opt, c5_opt = pair_exchange_descent(best_h, n_rounds=100, rng=rng)
                if c5_opt < best_c5:
                    best_h, best_c5, best_n = h_opt, c5_opt, len(h_opt)

    # Phase 4: Simulated annealing from best
    if time_left() > 50:
        n_sa = min(60000, int(time_left() * 600))
        t0 = max(3e-3, 0.04 * best_c5)
        h_opt, c5_opt = simulated_annealing(best_h, t0, 1e-7, n_sa, rng)
        if c5_opt < best_c5:
            best_h, best_c5, best_n = h_opt, c5_opt, len(h_opt)
        # Follow with local descent
        if time_left() > 15:
            h_opt, c5_opt = pair_exchange_descent(best_h, n_rounds=100, rng=rng)
            if c5_opt < best_c5:
                best_h, best_c5, best_n = h_opt, c5_opt, len(h_opt)

    # Phase 5: Resolution sweep
    if time_left() > 40:
        n_list = [30, 36, 40, 45, 50, 54, 60, 66, 72, 80, 90, 100, 120]
        for n_try in n_list:
            if time_left() < 30:
                break
            if n_try == best_n:
                continue
            h_mapped = map_resolution(best_h, best_n, n_try)
            c5_mapped = compute_c5(h_mapped)
            for beta in [200, 5000]:
                if time_left() < 18:
                    break
                h_new = slsqp_step(h_mapped, beta, max_iter=120)
                h_new = project_feasible(h_new, n_try)
                c5_new = compute_c5(h_new)
                if c5_new < c5_mapped:
                    h_mapped, c5_mapped = h_new, c5_new
            if time_left() > 12:
                h_opt, c5_opt = pair_exchange_descent(h_mapped, n_rounds=80, rng=rng)
                if c5_opt < c5_mapped:
                    h_mapped, c5_mapped = h_opt, c5_opt
            if c5_mapped < best_c5:
                best_h, best_c5, best_n = h_mapped, c5_mapped, n_try

    # Phase 6: Second round of annealing + local search
    if time_left() > 40:
        n_sa = min(40000, int(time_left() * 500))
        t0 = max(2e-3, 0.03 * best_c5)
        h_opt, c5_opt = simulated_annealing(best_h, t0, 5e-8, n_sa, rng)
        if c5_opt < best_c5:
            best_h, best_c5, best_n = h_opt, c5_opt, len(h_opt)
        if time_left() > 15:
            for beta in [1000, 10000]:
                if time_left() < 12:
                    break
                h_new = slsqp_step(best_h, beta, max_iter=150)
                h_new = project_feasible(h_new, len(h_new))
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_h, best_c5, best_n = h_new, c5_new, len(h_new)
            if time_left() > 10:
                h_opt, c5_opt = pair_exchange_descent(best_h, n_rounds=100, rng=rng)
                if c5_opt < best_c5:
                    best_h, best_c5, best_n = h_opt, c5_opt, len(h_opt)

    # Phase 7: Final intensive local search
    if time_left() > 15:
        n_iter = min(400, int(time_left() * 15))
        h_opt, c5_opt = pair_exchange_descent(best_h, n_rounds=n_iter, rng=rng)
        if c5_opt < best_c5:
            best_h, best_c5, best_n = h_opt, c5_opt, len(h_opt)

    # Phase 8: Final SLSQP polish
    if time_left() > 12:
        for beta in [5000, 20000, 50000]:
            if time_left() < 10:
                break
            h_new = slsqp_step(best_h, beta, max_iter=300)
            h_new = project_feasible(h_new, len(h_new))
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h, best_c5, best_n = h_new, c5_new, len(h_new)
        if time_left() > 8:
            h_opt, c5_opt = pair_exchange_descent(best_h, n_rounds=100, rng=rng)
            if c5_opt < best_c5:
                best_h, best_c5, best_n = h_opt, c5_opt, len(h_opt)

    # Final check
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
