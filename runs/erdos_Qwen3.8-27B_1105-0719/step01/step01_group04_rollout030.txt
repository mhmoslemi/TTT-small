def repair_feasible(vec):
    h = np.asarray(vec, dtype=np.float64).copy()
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(h)
        rem = diff
        for i in order:
            if rem <= 0:
                break
            add = min(rem, 1.0 - h[i])
            h[i] += add
            rem -= add
    else:
        order = np.argsort(-h)
        rem = -diff
        for i in order:
            if rem <= 0:
                break
            remv = min(rem, h[i])
            h[i] -= remv
            rem -= remv
    # Final fix
    diff2 = target - h.sum()
    if abs(diff2) > 1e-14:
        for i in range(n):
            if diff2 > 0 and h[i] < 1.0:
                h[i] += diff2
                break
            elif diff2 < 0 and h[i] > 0.0:
                h[i] += diff2
                break
    return h

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr * dx))

def sigmoid(x):
    # Numerically stable sigmoid
    result = np.empty_like(x)
    pos = x >= 0
    result[pos] = 1.0 / (1.0 + np.exp(-x[pos]))
    exp_x = np.exp(x[~pos])
    result[~pos] = exp_x / (1.0 + exp_x)
    return result

def logit_project(z):
    """Project unconstrained z to feasible h via logit + normalization shift."""
    n = len(z)
    target = n / 2.0
    
    def phi(c):
        h = sigmoid(z + c)
        return h.sum() - target
    
    # Find bracket
    lo, hi = -50.0, 50.0
    if phi(lo) > 0:
        # Even at lo, sum > target, need to go lower
        lo -= 50.0
    if phi(hi) < 0:
        hi += 50.0
    
    c = brentq(phi, lo, hi, xtol=1e-14)
    h = sigmoid(z + c)
    return h, c

def profile_values(h):
    """Compute all lag values C_l."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")

def profile_gradient(h):
    """Compute gradient of each lag value with respect to h.
    Returns array of shape (2n-1, n)."""
    n = len(h)
    dx = 2.0 / n
    L = 2 * n - 1
    grads = np.zeros((L, n))

            # First factor: h_i for i = 0..n-1-m

def transfer_dimension(h_src, n_tgt):
    """Transfer h from its current dimension to n_tgt using piecewise-constant averaging."""
    n_src = len(h_src)
    dx_src = 2.0 / n_src
    dx_tgt = 2.0 / n_tgt
    
    # Cumulative area
    cum = np.zeros(n_src + 1)
    for i in range(n_src):
        cum[i+1] = cum[i] + h_src[i] * dx_src
    
    # For each target cell, compute the area
    h_tgt = np.zeros(n_tgt)
    for j in range(n_tgt):
        x_left = j * dx_tgt
        x_right = (j + 1) * dx_tgt
        # Find which source cells these edges fall in
        i_left = min(int(x_left / dx_src), n_src - 1)
        i_right = min(int(x_right / dx_src), n_src)
        # Fractional parts
        frac_left = x_left / dx_src - i_left
        frac_right = x_right / dx_src - min(i_right, n_src - 1) if i_right < n_src else 0.0
        
        # Actually, let me use a cleaner approach
        # Area from x_left to x_right
        a_left = x_left / dx_src  # position in source units
        a_right = x_right / dx_src
        
        def area_up_to(x):

            i = int(np.floor(x))
            frac = x - i
            if i >= n_src:
                return cum[n_src]

                return 0.0

                    diff2 -= min(diff2, room)
                    if abs(diff2) < 1e-15:
                        break
            else:
                if h[i] > 0.0:
                    room = h[i]
                    h[i] -= min(-diff2, room)
                    diff2 += min(-diff2, room)
                    if abs(diff2) < 1e-15:
                        break
    return h

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr * dx))

def sigmoid(x):
    out = np.empty_like(x, dtype=np.float64)
    pos = x >= 0
    out[pos] = 1.0 / (1.0 + np.exp(-x[pos]))
    ex = np.exp(x[~pos])
    out[~pos] = ex / (1.0 + ex)
    return out

def logit_project(z):
    n = len(z)
    target = n / 2.0
    def phi(c):
        return sigmoid(z + c).sum() - target
    lo, hi = -80.0, 80.0
    try:
        c = brentq(phi, lo, hi, xtol=1e-13, maxiter=200)
    except:
        # Fallback: bisect
        c = (lo + hi) / 2.0
        for _ in range(100):
            c = (lo + hi) / 2.0
            if phi(c) > 0:
                hi = c
            else:
                lo = c
    h = sigmoid(z + c)
    return h

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx

def compute_profile_grads(h):
    n = len(h)
    dx = 2.0 / n
    L = 2 * n - 1
    grads = np.zeros((L, n))
    for k in range(L):
        delta = k - (n - 1)
        if delta == 0:
            grads[k] = dx * (1.0 - 2.0 * h)
        elif delta > 0:
            m = delta
            grads[k, :n-m] += dx * (1.0 - h[m:])
            grads[k, m:] += -dx * h[:n-m]
        else:
            m = -delta
            grads[k, m:] += dx * (1.0 - h[:n-m])
            grads[k, :n-m] += -dx * h[m:]
    return grads

def soft_max_obj_and_grad(z, beta):
    h = logit_project(z)
    n = len(h)
    values = compute_profile(h)
    L = len(values)
    bv = beta * values
    mx = bv.max()
    ev = np.exp(bv - mx)
    se = ev.sum()
    J = (mx + np.log(se)) / beta
    w = ev / se
    grads = compute_profile_grads(h)
    G_h = w @ grads
    sp = h * (1.0 - h)
    T = sp.sum()
    if T < 1e-12:
        return J, np.zeros_like(z)
    dot = np.dot(sp, G_h)
    G_z = sp * (G_h - dot / T)
    return J, G_z

def h_to_z(h):
    hc = np.clip(h, 1e-8, 1.0 - 1e-8)
    return np.log(hc / (1.0 - hc))

def active_descent(h, max_steps=60):
    h = repair_feasible(h.copy())
    best_score = exact_score(h)
    n = len(h)
    for step in range(max_steps):
        values = compute_profile(h)
        M = values.max()
        L = len(values)
        # Select top-K lags
        K = min(20, L)
        sort_idx = np.argsort(values)[::-1]
        top_idx = sort_idx[:K]
        # Weights
        gaps = M - values[top_idx]
        sigma = max(0.003, 0.1 * (values[sort_idx[0]] - values[sort_idx[K-1]]))
        w = np.exp(-gaps / sigma)
        w /= w.sum()
        # Weighted gradient
        grads = compute_profile_grads(h)
        G = -(w @ grads[top_idx])
        G -= G.mean()
        # Line search
        improved = False
        for alpha in [0.05, 0.03, 0.02, 0.01, 0.005, 0.003, 0.001]:
            h_try = repair_feasible(h + alpha * G)
            s = exact_score(h_try)
            if s < best_score - 1e-14:
                h = h_try
                best_score = s
                improved = True
                break
        if not improved:
            break
    return h, best_score

def transfer_dim(h_src, n_tgt):
    n_src = len(h_src)
    if n_src == n_tgt:
        return h_src.copy()
    dx_s = 2.0 / n_src
    dx_t = 2.0 / n_tgt
    cum = np.zeros(n_src + 1)
    cum[1:] = np.cumsum(h_src) * dx_s
    x_left = np.arange(n_tgt) * dx_t
    x_right = (np.arange(n_tgt) + 1) * dx_t
    a_left = x_left / dx_s
    a_right = x_right / dx_s
    def area_at(x):
        # x in source units [0, n_src]
        i = np.minimum(np.floor(x).astype(int), n_src - 1)
        i = np.maximum(i, 0)
        frac = x - np.floor(x)
        result = cum[i] + frac * h_src[i] * dx_s
        result[x >= n_src] = cum[n_src]
        return result
    areas = (area_at(a_right) - area_at(a_left))
    h_tgt = areas / dx_t
    return np.clip(h_tgt, 0.0, 1.0)

def symmetrize(h):
    return (h + h[::-1]) / 2.0
