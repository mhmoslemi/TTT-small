import numpy as np
from scipy.optimize import minimize
import time

def exact_objective(h):
    n = len(h)
    if n == 0:
        return float('inf')
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(profile))

def project_box_sum(h):
    h = np.array(h, dtype=np.float64).copy()
    n = len(h)
    target = n / 2.0
    h[~np.isfinite(h)] = 0.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            h[mask] = np.minimum(1.0, h[mask] + diff / cnt)
        else:
            mask = h > 1e-15
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            h[mask] = np.maximum(0.0, h[mask] + diff / cnt)
    return np.clip(h, 0.0, 1.0)

def binary_from_continuous(h):
    n = len(h)
    if n % 2 != 0:
        return None
    m = n // 2
    idx = np.argsort(h)[::-1][:m]
    b = np.zeros(n, dtype=np.float64)
    b[idx] = 1.0
    return b

import numpy as np
from scipy.optimize import minimize
import time

def exact_objective(h):
    n = len(h)
    if n == 0:
        return float('inf')
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(profile))

def project_box_sum(h):
    """Project to [0,1]^n with sum = n/2 using bisection on water level."""
    h = np.array(h, dtype=np.float64).copy()
    n = len(h)
    target = n / 2.0
    h[~np.isfinite(h)] = 0.0
    h = np.clip(h, 0.0, 1.0)
    # Bisection for water level
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        adjusted = np.clip(h + mid, 0.0, 1.0)
        s = adjusted.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    h_out = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    # Fix any remaining discrepancy
    diff = target - h_out.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h_out < 1.0 - 1e-10
        else:
            mask = h_out > 1e-10
        if mask.sum() > 0:
            h_out[mask] += diff / mask.sum()
            h_out = np.clip(h_out, 0.0, 1.0)
    return h_out

def binary_from_continuous(h):
    n = len(h)
    if n % 2 != 0:
        return None
    m = n // 2
    idx = np.argsort(h)[::-1][:m]
    b = np.zeros(n, dtype=np.float64)
    b[idx] = 1.0
    return b

def transfer_to_size(h_src, target_len):
    n_src = len(h_src)
    if n_src == target_len:
        return project_box_sum(h_src.copy())
    x_src = (np.arange(n_src) + 0.5) / n_src * 2.0
    x_tgt = (np.arange(target_len) + 0.5) / target_len * 2.0
    h_new = np.interp(x_tgt, x_src, h_src)
    return project_box_sum(h_new)

import numpy as np
from scipy.optimize import minimize
import time

def exact_objective(h):
    n = len(h)
    if n == 0:
        return float('inf')
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(profile))

def project_box_sum(h):
    h = np.array(h, dtype=np.float64).copy()
    n = len(h)
    target = n / 2.0
    h[~np.isfinite(h)] = 0.0
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    h_out = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - h_out.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = h_out < 1.0 - 1e-8
        else:
            mask = h_out > 1e-8
        if mask.sum() > 0:
            h_out[mask] = np.clip(h_out[mask] + diff / mask.sum(), 0, 1)
    return h_out

def binary_from_continuous(h):
    n = len(h)
    if n % 2 != 0:
        return None
    m = n // 2
    idx = np.argsort(h)[::-1][:m]
    b = np.zeros(n, dtype=np.float64)
    b[idx] = 1.0
    return b

def transfer_to_size(h_src, target_len):
    n_src = len(h_src)
    if n_src == target_len:
        return project_box_sum(h_src.copy())
    x_src = (np.arange(n_src) + 0.5) / n_src * 2.0
    x_tgt = (np.arange(target_len) + 0.5) / target_len * 2.0
    h_new = np.interp(x_tgt, x_src, h_src)
    return project_box_sum(h_new)

def make_binary_random(n, rng):
    """Random binary with exactly n/2 ones."""
    m = n // 2
    idx = rng.choice(n, size=m, replace=False)
    b = np.zeros(n, dtype=np.float64)
    b[idx] = 1.0
    return b

def make_binary_structured(n, rng, mode):
    """Structured binary initializations."""
    m = n // 2
    b = np.zeros(n, dtype=np.float64)
    if mode == 0:
        # First half
        b[:m] = 1.0
    elif mode == 1:
        # Alternating blocks
        block = max(1, n // 8)
        for i in range(0, n, 2 * block):
            b[i:min(i+block, n)] = 1.0
    elif mode == 2:
        # Every other position
        b[::2] = 1.0[:m]  # hmm this is tricky
        # Actually just set first m of every-other
        idx = np.arange(0, n, 2)[:m]
        b[idx] = 1.0
    elif mode == 3:
        # Two blocks
        b[:m//2] = 1.0
        b[3*m//2:] = 1.0
    else:
        idx = rng.choice(n, size=m, replace=False)
        b[idx] = 1.0
    # Ensure exactly m ones
    actual = int(np.sum(b))
    if actual > m:
        # Remove excess
        ones = np.where(b == 1)[0]
        excess = actual - m
        to_remove = ones[:excess]
        b[to_remove] = 0.0
    elif actual < m:
        zeros = np.where(b == 0)[0]
        needed = m - actual
        b[zeros[:needed]] = 1.0
    return b

def binary_greedy_search(b, max_rounds=200, deadline=None):
    """Greedy best-improvement: try all single swaps, pick the best."""
    n = len(b)
    curr_c = exact_objective(b)
    ones = set(np.where(b == 1.0)[0].tolist())
    zeros = set(np.where(b == 0.0)[0].tolist())
    
    for rnd in range(max_rounds):
        if deadline is not None and time.time() > deadline - 3:
            break
        best_improve = 1e-15
        best_swap = None
        for i in ones:
            for j in zeros:
                b[i] = 0.0
                b[j] = 1.0
                c = exact_objective(b)
                b[i] = 1.0
                b[j] = 0.0
                improve = curr_c - c
                if improve > best_improve:
                    best_improve = improve
                    best_swap = (i, j)
        if best_swap is not None:
            i, j = best_swap
            b[i] = 0.0
            b[j] = 1.0
            curr_c = exact_objective(b)
            ones.discard(i)
            ones.add(j)
            zeros.discard(j)
            zeros.add(i)
        else:
            break
    return b, curr_c

def binary_sa(b, time_limit=5.0, deadline=None):
    """Simulated annealing on binary."""
    n = len(b)
    curr_c = exact_objective(b)
    best_c = curr_c
    best_b = b.copy()
    
    ones = list(np.where(b == 1.0)[0])
    zeros = list(np.where(b == 0.0)[0])
    
    T = 0.01
    T_min = 0.0001
    T_factor = 0.995
    rng = np.random.RandomState(42)
    
    start = time.time()
    while time.time() - start < time_limit:
        if deadline is not None and time.time() > deadline - 3:
            break
        # Pick random swap
        i = rng.randint(len(ones))
        j = rng.randint(len(zeros))
        oi, oj = ones[i], zeros[j]
        
        b[oi] = 0.0
        b[oj] = 1.0
        new_c = exact_objective(b)
        diff = new_c - curr_c
        
        if diff < 0 or rng.random() < np.exp(-diff / T):
            curr_c = new_c
            ones[i] = oj
            zeros[j] = oi
            if curr_c < best_c:
                best_c = curr_c
                best_b = b.copy()
        else:
            b[oi] = 1.0
            b[oj] = 0.0
        
        T *= T_factor
        if T < T_min:
            T = T_min
    
    return best_b, best_c
