def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr * dx))


def repair_feasible(vec):
    h = np.asarray(vec, dtype=np.float64)
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h.copy()
    if diff > 0:
        order = np.argsort(h)
        rem = diff
        for i in order:
            if rem <= 1e-15:
                break
            add = min(rem, 1.0 - h[i])
            h[i] += add
            rem -= add
    else:
        order = np.argsort(-h)
        rem = -diff
        for i in order:
            if rem <= 1e-15:
                break
            remv = min(rem, h[i])
            h[i] -= remv
            rem -= remv

                break

def compute_F_and_grads(h):

    F = np.zeros(n)
    grads = np.zeros((n, n))
    
    # Precompute prefix and suffix sums

    A_0 = float(np.dot(h, h))
    F[0] = dx * A_0
    grads[0, :] = dx * 2.0 * h
    
    # For m >= 1
    for m in range(1, n):
        A_m = float(np.dot(h[:n-m], h[m:]))
        P_m = float(h[:m].sum())
        S_m = float(h[n-m:].sum())
        B_m = min(P_m, S_m)
        F[m] = dx * (A_m + B_m)

        g[m:] += h[:n-m]

        elif S_m < P_m:
            grads[m, n-m:] += dx
        else:
            grads[m, :m] += dx * 0.5
            grads[m, n-m:] += dx * 0.5
    
    return F, grads


def smooth_obj_and_grad(h, beta):
    """g(h) = (1/beta)*log(sum exp(-beta*F_m)), minimize to maximize min F_m."""
    F, grads = compute_F_and_grads(h)

    F_shift = F - F_min
    exp_vals = np.exp(-beta * F_shift)
    sum_exp = float(np.sum(exp_vals))
    g = -F_min + (1.0 / beta) * np.log(sum_exp)
    weights = exp_vals / sum_exp
    grad_g = -np.sum(weights[:, None] * grads, axis=0)
    return g, grad_g


def solve_smooth_slsqp(h_start, beta, maxiter=200):
    n = len(h_start)
    h_start = repair_feasible(h_start)
    
    def obj(x):
        val, _ = smooth_obj_and_grad(x, beta)
        return val

        _, g = smooth_obj_and_grad(x, beta)
        return g
    
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0,
                    'jac': lambda x: np.ones(n)}]

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")

def repair_feasible(vec):
    h = np.asarray(vec, dtype=np.float64).copy()
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(h)
        rem = diff
        for i in order:
            if rem <= 1e-15:
                break
            add = min(rem, 1.0 - h[i])
            h[i] += add
            rem -= add
    else:
        order = np.argsort(-h)
        rem = -diff
        for i in order:
            if rem <= 1e-15:
                break
            remv = min(rem, h[i])
            h[i] -= remv
            rem -= remv

                break
            elif diff2 < 0 and h[i] > 0.0:
                h[i] += diff2
                break

        S[1:] = rcs[n-1:0:-1]

    F = dx * (A + B)

import numpy as np
from scipy.optimize import minimize

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr * dx))


def repair_feasible(vec):
    h = np.asarray(vec, dtype=np.float64).copy()
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(h)
        rem = diff
        for i in order:
            if rem <= 1e-15:
                break
            add = min(rem, 1.0 - h[i])
            h[i] += add
            rem -= add
    else:
        order = np.argsort(-h)
        rem = -diff
        for i in order:
            if rem <= 1e-15:
                break
            remv = min(rem, h[i])
            h[i] -= remv
            rem -= remv
    diff2 = target - h.sum()
    if abs(diff2) > 1e-14:
        for i in range(n):
            if diff2 > 0 and h[i] < 1.0:
                h[i] += diff2
                break
            elif diff2 < 0 and h[i] > 0.0:
                h[i] += diff2
                break
    return h


def compute_F(h):
    n = len(h)
    dx = 2.0 / n
    autocorr = np.correlate(h, h, mode='full')
    A = autocorr[n - 1:]

    S = np.zeros(n)

        S[1:] = rcs[n - 1:0:-1]
    B = np.minimum(P, S)
    F = dx * (A + B)
    return F, P, S

def smooth_obj_and_grad(h, beta):

    F_min = float(np.min(F))
    F_shift = F - F_min
    exp_vals = np.exp(-beta * F_shift)
    sum_exp = float(np.sum(exp_vals))

def solve_smooth_slsqp(h_start, beta, maxiter=200):
    n = len(h_start)
    h_start = repair_feasible(h_start)

    def obj(x):
        val, _ = smooth_obj_and_grad(x, beta)
        return val

    def jac(x):
        _, g = smooth_obj_and_grad(x, beta)
        return g

    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
                    'jac': lambda x: np.ones(n)}]

    try:
        result = minimize(obj, h_start, jac=jac, method='SLSQP',
                          bounds=bounds, constraints=constraints,
