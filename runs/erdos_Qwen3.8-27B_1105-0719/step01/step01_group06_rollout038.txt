def lse_score(h, beta):
    """Smooth approximation to max using log-sum-exp."""
    p = profile(h)
    # LSE: (1/beta) * log(sum(exp(beta * p)))
    # For numerical stability: (1/beta) * (max(p) + log(sum(exp(beta * (p - max(p))))))

    return m + np.log(np.sum(np.exp(beta * (p - m)))) / beta

def lse_gradient(h, beta):
    """Compute gradient of LSE objective with respect to h."""
    n = len(h)
    dx = 2.0 / n
    P = profile_full(h)
    M = len(P)
    
    # Softmax weights (numerically stable)
    P_shifted = P - P.max()
    w = np.exp(beta * P_shifted)
    w /= w.sum()
    
    # Gradient: sum_k w[k] * G_k
    # G_k is the gradient of P[k] with respect to h
    grad = np.zeros(n)
    for k in range(M):
        if w[k] < 1e-15:
            continue
        delta = k - (n - 1)
        if delta >= 0:
            grad[delta:n] += w[k] * dx * (1.0 - h[0:n-delta])

        else:
            d = -delta
            grad[0:n-d] += w[k] * dx * (1.0 - h[d:n])
            grad[d:n] -= w[k] * dx * h[0:n-d]

def exact_score(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))

def profile_full(h):
    dx = 2.0 / len(h)
    return np.correlate(h, 1.0 - h, mode="full") * dx

def repair_feasible(h, target=None):
    h = np.array(h, dtype=np.float64).copy()
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    if target is None:
        target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        order = np.argsort(-room)
        remaining = diff
        for idx in order:
            if remaining <= 1e-15:
                break
            add = min(remaining, room[idx])
            h[idx] += add
            remaining -= add
    else:
        order = np.argsort(-h)
        remaining = -diff
        for idx in order:
            if remaining <= 1e-15:
                break
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
    return h

def project_feasible(y, S):
    lo = y.min() - 1.5
    hi = y.max() + 1.5
    for _ in range(100):
        tau = 0.5 * (lo + hi)
        s = np.clip(y - tau, 0.0, 1.0).sum()
        if s > S:
            lo = tau
        else:
            hi = tau
    x = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    diff = S - x.sum()
    if abs(diff) > 1e-13:
        free = (x > 1e-10) & (x < 1.0 - 1e-10)
        if free.sum() > 0:
            x[free] += diff / free.sum()
            x = np.clip(x, 0.0, 1.0)
    return x

def lse_objective(h, beta):
    P = profile_full(h)
    m = P.max()
    return m + np.log(np.sum(np.exp(beta * (P - m)))) / beta

def lse_gradient(h, beta):
    n = len(h)
    dx = 2.0 / n
    P = profile_full(h)
    M = len(P)
    P_shifted = P - P.max()
    w = np.exp(beta * P_shifted)
    w /= w.sum()
    grad = np.zeros(n)
    for k in range(M):
        wk = w[k]
        if wk < 1e-16:
            continue
        delta = k - (n - 1)
        if delta >= 0:
            grad[delta:n] += wk * dx * (1.0 - h[0:n - delta])
            grad[0:n - delta] -= wk * dx * h[delta:n]
        else:
            d = -delta
            grad[0:n - d] += wk * dx * (1.0 - h[d:n])
            grad[d:n] -= wk * dx * h[0:n - d]
    return grad

def interpolate_dim(h, new_n):
    n = len(h)

    x_tgt = (np.arange(new_n) + 0.5) * 2.0 / new_n

    h_new = np.interp(x_tgt, x_ext, f_ext)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

        J[1:, :n_local] = -Ja
        J[1:, n_local] = 1.0
        return J
