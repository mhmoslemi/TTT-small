    return float(np.max(corr) * dx)

def repair_feasible(vec):

    diff = target - s

        return h.copy()
    if diff > 0:

        for i in order:
            if rem <= 1e-15:
                break
            add = min(rem, 1.0 - h[i])
            h[i] += add
            rem -= add
    else:

        rem = -diff

                break

    return values

    values = compute_profile(h)
    # Stable log-sum-exp

    n = len(h)
    dx = 2.0 / n

def cdf_transfer(h, new_len):
    """Transfer h to new dimension using cumulative integral (mass-preserving)."""
    n = len(h)
    x_src = np.arange(n + 1) * 2.0 / n  # cumulative x positions
    # Build cumulative integral
    cum = np.zeros(n + 1)
    cum[1:] = np.cumsum(h) * (2.0 / n)
    
    # Resample at new positions
    x_tgt = np.linspace(0.0, 2.0, new_len + 1)
    cum_tgt = np.interp(x_tgt, x_src, cum)
    h_new = np.diff(cum_tgt) / (2.0 / new_len)
    return h_new

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def repair_feasible(vec):
    h = np.asarray(vec, dtype=np.float64)
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-12:
        return h.copy()
    if diff > 0:
        order = np.argsort(h)
        rem = diff
        for i in order:
            if rem <= 1e-15:
                break
            add = min(rem, 1.0 - h[i])
            h[i] += add
            rem -= add
    else:
        order = np.argsort(-h)
        rem = -diff
        for i in order:
            if rem <= 1e-15:
                break
                if rem <= 1e-15:
                    break
            remv = min(rem, h[i])
            h[i] -= remv
            rem -= remv
    diff2 = target - h.sum()
    if abs(diff2) > 1e-14:
        for i in range(n):
            if diff2 > 0 and h[i] < 1.0:
                h[i] += diff2
                break
            elif diff2 < 0 and h[i] > 0.0:
                h[i] += diff2
                break
    return h


def compute_profile(h):

    dx = 2.0 / n

            values[n - 1 - delta] = dx * np.sum(h[:n - delta] * (1.0 - h[delta:]))
    return values

    n = len(h)
    dx = 2.0 / n
    values = np.zeros(2 * n - 1)
    for delta in range(n):
        if delta == 0:

def cdf_transfer(h, new_len):
    n = len(h)
    x_src = np.arange(n + 1) * 2.0 / n
    cum = np.zeros(n + 1)
    cum[1:] = np.cumsum(h) * (2.0 / n)
    x_tgt = np.linspace(0.0, 2.0, new_len + 1)
    cum_tgt = np.interp(x_tgt, x_src, cum)
    h_new = np.diff(cum_tgt) / (2.0 / new_len)
    return h_new

    return (h + h[::-1]) / 2.0


def soft_max_solve(h_start, beta, maxiter=200):
    n = len(h_start)

    def obj(x):
        v, _ = compute_softmax_obj_grad(x, beta)
        return v

    def obj_grad(x):
        _, g = compute_softmax_obj_grad(x, beta)
        return g

    def eq_fun(x):
        return np.sum(x) - n / 2.0

    def eq_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]

    try:
        result = minimize(
            obj, h_start, jac=obj_grad,
            method='SLSQP', bounds=bounds, constraints=constraints,
            options={'maxiter': maxiter, 'ftol': 1e-14}

        if result.x is not None:
            h_sol = repair_feasible(result.x)
            score = exact_score(h_sol)
            if np.isfinite(score):

    except Exception:
        pass
    return None, None

        for nd in [n_cur + 20, n_cur + 40, min(250, n_cur * 2)]:
            if remaining() < 60:
                break
            if nd <= n_cur or nd > 300:
                continue
            h_big = repair_feasible(cdf_transfer(best_h, nd))
            s_big = exact_score(h_big)
            if s_big < best_score + 0.02:
                for beta in [50, 150, 400]:
                    if remaining() < 50:
                        break
                    mi = min(200, max(40, int(remaining() * 0.03)))
                    h_sol, score = soft_max_solve(h_big, beta, mi)
                    if h_sol is not None and score < best_score - 1e-13:
                        best_h = h_sol
                        best_score = score
                        best_n = len(h_sol)
                        h_big = h_sol.copy()

    # Phase 4: Dimension reduction
    if remaining() > 80:
        n_cur = len(best_h)
        for nd in [max(60, n_cur - 30), max(50, n_cur - 60)]:
            if remaining() < 50:
                break
            if nd >= n_cur:
                continue
            h_small = repair_feasible(cdf_transfer(best_h, nd))
            s_small = exact_score(h_small)
            if s_small < best_score + 0.01:
                for beta in [50, 150]:
                    if remaining() < 40:
                        break
                    mi = min(150, max(30, int(remaining() * 0.03)))
                    h_sol, score = soft_max_solve(h_small, beta, mi)
                    if h_sol is not None and score < best_score - 1e-13:
                        best_h = h_sol
                        best_score = score
                        best_n = len(h_sol)
                        h_small = h_sol.copy()

    # Phase 5: Multi-start perturbations
    if remaining() > 80:
        n_trials = min(15, int(remaining() / 35))
        for trial in range(n_trials):
            if remaining() < 45:
                break
            eps = rng.uniform(0.003, 0.025)
            z = rng.standard_normal(len(best_h))
            z -= z.mean()
            h_pert = repair_feasible(best_h + eps * z)
            s_pert = exact_score(h_pert)
            if s_pert < best_score + 0.04:
                mi = min(120, max(30, int(remaining() * 0.025)))
                h_sol, score = soft_max_solve(h_pert, 100, mi)
                if h_sol is not None and score < best_score - 1e-13:
                    best_h = h_sol
                    best_score = score
                    best_n = len(h_sol)

    # Phase 6: Final high-beta soft-max
    if remaining() > 30:
        h_sol, score = soft_max_solve(best_h, 1000, min(200, int(remaining() * 0.4)))
        if h_sol is not None and score < best_score - 1e-13:
            best_h = h_sol
            best_score = score

    # Phase 7: Final active descent
    if remaining() > 20:
        h_ref, score_ref = active_descent(best_h, max_steps=100)
        if score_ref < best_score - 1e-13:
            best_h = h_ref
            best_score = score_ref

    # Phase 8: One more soft-max at very high beta
    if remaining() > 25:
        h_sol, score = soft_max_solve(best_h, 2000, min(100, int(remaining() * 0.3)))
        if h_sol is not None and score < best_score - 1e-13:
            best_h = h_sol
            best_score = score

    # Final validation
    best_h = repair_feasible(best_h)
    best_score = exact_score(best_h)
    return best_h, best_score, len(best_h)
