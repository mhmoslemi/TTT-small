import numpy as np
import time

def exact_score(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))

def get_profile(h):
    dx = 2.0 / len(h)
    return np.correlate(h, 1.0 - h, mode="full") * dx

def get_lag_gradients(h):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    G = np.zeros((M, n))
    for k in range(M):
        delta = k - (n - 1)
        if delta >= 0:
            if delta < n:
                G[k, delta:n] += dx * (1.0 - h[0:n - delta])
                G[k, 0:n - delta] -= dx * h[delta:n]
        else:
            d = -delta
            if d < n:
                G[k, 0:n - d] += dx * (1.0 - h[d:n])
                G[k, d:n] -= dx * h[0:n - d]
    return G

def repair_feasible(h, target=None):
    h = np.array(h, dtype=np.float64).copy()
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    if target is None:
        target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        order = np.argsort(-room)
        remaining = diff
        for idx in order:
            if remaining <= 1e-15:
                break
            add = min(remaining, room[idx])
            h[idx] += add
            remaining -= add
    else:
        order = np.argsort(-h)
        remaining = -diff
        for idx in order:
            if remaining <= 1e-15:
                break
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
    return h

def lse_score(h, beta):
    """Log-sum-exp approximation of max of profile."""
    prof = get_profile(h)
    # Shift to avoid overflow
    pm = prof.max()
    if beta * pm > 700:
        return pm + np.log(np.sum(np.exp(beta * (prof - pm)))) / beta
    return np.log(np.sum(np.exp(beta * prof))) / beta

def lse_gradient(h, beta):
    """Gradient of LSE score w.r.t. h."""
    prof = get_profile(h)
    pm = prof.max()
    log_vals = beta * (prof - pm)
    log_vals = np.clip(log_vals, -700, 700)
    weights = np.exp(log_vals)
    weights /= weights.sum()
    G = get_lag_gradients(h)
    grad = weights @ G
    return grad

def symmetrize(h):
    h_sym = 0.5 * (h + h[::-1].copy())
    return repair_feasible(h_sym)

def interpolate_to_dim(vec, new_len):
    n = len(vec)
    x_src = (np.arange(n) + 0.5) * 2.0 / n
    x_tgt = (np.arange(new_len) + 0.5) * 2.0 / new_len
    x_ext = np.concatenate([[0.0], x_src, [2.0]])
    f_ext = np.concatenate([[vec[0]], vec, [vec[-1]]])
    h_new = np.interp(x_tgt, x_ext, f_ext)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

def lse_descent(h_start, beta_start=50.0, beta_end=500.0, n_steps=20, time_limit=30):
    """Descent on LSE objective with decreasing temperature."""
    t0 = time.perf_counter()
    h = h_start.copy()
    best_h = h.copy()
    best_score = exact_score(h)
    
    # Log-spaced temperature schedule
    betas = np.geomspace(beta_start, beta_end, n_steps)
    
    for beta in betas:
        if time.perf_counter() - t0 > time_limit:
            break
        
        # Inner loop: a few gradient steps at this temperature
        for it in range(5):
            if time.perf_counter() - t0 > time_limit:
                break
            
            grad = lse_gradient(h, beta)
            # Project to satisfy mean constraint
            grad -= grad.mean()
            norm = np.linalg.norm(grad)
            if norm < 1e-14:
                break
            
            # Backtracking line search
            improved = False
            for alpha in [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]:
                h_trial = repair_feasible(h + alpha * (-grad))
                s = exact_score(h_trial)
                if s < best_score - 1e-12:
                    h = h_trial
                    best_h = h.copy()
                    best_score = s
                    improved = True
                    break
            if not improved:
                break
    
    return best_h, best_score

def active_set_descent(h_start, max_iters=40, time_limit=20):

        w = np.maximum(excess, 0.01)

                h = h_trial
                best_h = h.copy()
                best_score = s

            break

def slsqp_epigraph(h_start, t_start, maxiter=60, n_constraints=0, beta=0.0):
    """SLSQP epigraph with selected active constraints."""
    from scipy.optimize import minimize
    
    n = len(h_start)
    M = 2 * n - 1
    nc = n_constraints if n_constraints > 0 else M
    
    if nc < M:
        prof = get_profile(h_start)
        active_lags = np.argsort(prof)[::-1][:nc]
    else:
        active_lags = np.arange(M)
    
    n_active = len(active_lags)
    
    def obj(x):
        t = x[n]
        if beta > 0:
            h = x[:n]
            return t + beta * np.sum((1 - h) ** 2) + beta * np.sum(h ** 2)
        return t
    
    def obj_jac(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        if beta > 0:
            h = x[:n]
            g[:n] = 2 * beta * (h - 0.5)
        return g
    
    def cons_fun(x):
        h = x[:n]
        t = x[n]
        eq = h.sum() - n / 2.0
        prof = np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)
        ineq = t - prof[active_lags]
        return np.concatenate([[eq], ineq])
    
    def cons_jac(x):
        h = x[:n]
        G = get_lag_gradients(h)
        J = np.zeros((1 + n_active, n + 1))
        J[0, :n] = 1.0
        J[1:, :n] = -G[active_lags, :]
        J[1:, n] = 1.0
        return J
    
    bounds = [(0.0, 1.0)] * n + [(t_start - 0.1, t_start + 0.5)]
    x0 = np.concatenate([h_start, [t_start]])
    
    try:
        result = minimize(
            obj, x0, jac=obj_jac,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'ineq', 'fun': cons_fun, 'jac': cons_jac}],
            options={'maxiter': maxiter, 'ftol': 1e-12}
        )
        if result.x is not None and np.all(np.isfinite(result.x)):
            h_sol = repair_feasible(result.x[:n])
            return h_sol
    except Exception:
        pass
    return None

        active_lags = np.arange(M)
    na = len(active_lags)

    def obj(x):
        t = x[n]
        if beta_reg > 0:
            h = x[:n]
            return t + beta_reg * np.sum((h - 0.5) ** 2)
        return t

        J[1:, n] = 1.0
        return J

            # Quick active descent from perturbation
            tl_pert = min(8, tleft() * 0.04)
            hp, sp = active_descent(h_pert, max_iters=20, time_limit=tl_pert)
            # SLSQP
            if tleft() > 15:
                mi = min(30, max(8, int(tleft() * 0.06)))
                hs = slsqp_solve(hp, sp + 5e-5, maxiter=mi, n_constr=20)
                if hs is not None:
                    s = exact_score(hs)
                    if s < sp:
                        hp, sp = hs, s
            if sp < best_score - 1e-12:
                best_h, best_score = hp, sp

    # Phase 7: Dimension exploration
    if tleft() > 50:
        pn = len(best_h)
        candidate_dims = []
        for d in range(max(40, pn - 20), min(250, pn + 40) + 1, 4):
            if d != pn and 40 <= d <= 250:
                candidate_dims.append(d)
        for d in [60, 80, 100, 120, 160, 180, 200, 220]:
            if d not in candidate_dims and d != pn and 40 <= d <= 250:
                candidate_dims.append(d)
        candidate_dims = sorted(set(candidate_dims))

        for new_n in candidate_dims[:12]:
            if tleft() < 20:
                break
            h_interp = interpolate_to_dim(best_h, new_n)
            s_interp = exact_score(h_interp)
            # LSE at new dim
            tl_d = min(10, tleft() * 0.06)
            hd, sd = lse_phase(h_interp, beta_lo=20.0, beta_hi=200.0, n_temps=6, time_limit=tl_d)
            if tleft() > 12:
                mi = min(30, max(8, int(tleft() * 0.05)))
                hs = slsqp_solve(hd, sd + 5e-5, maxiter=mi, n_constr=20)
                if hs is not None:
                    s = exact_score(hs)
                    if s < sd:
                        hd, sd = hs, s
            if sd < best_score - 1e-12:
                best_h, best_score = hd, sd

    # Phase 8: Final active descent
    if tleft() > 12:
        tl = min(10, tleft() * 0.35)
        h8, s8 = active_descent(best_h, max_iters=30, time_limit=tl)
        if s8 < best_score - 1e-12:
            best_h, best_score = h8, s8

    # Phase 9: Final SLSQP
    if tleft() > 8:
        mi = min(30, max(8, int(tleft() * 0.5)))
        h9 = slsqp_solve(best_h, best_score + 2e-5, maxiter=mi, n_constr=15)
        if h9 is not None:
            s9 = exact_score(h9)
            if s9 < best_score - 1e-12:
                best_h, best_score = h9, s9

    # Final
    best_h = repair_feasible(best_h)
    final_score = exact_score(best_h)
    return (best_h, final_score, len(best_h))
