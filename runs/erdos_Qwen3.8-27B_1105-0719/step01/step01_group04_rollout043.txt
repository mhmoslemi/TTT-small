```
def repair_feasible(vec):
    h = np.asarray(vec, dtype=np.float64).ravel()
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    n = h.size
    target = 0.5 * n
    target = float(target)
    s = float(h.sum())
    diff = target - s
    if abs(diff) <= 1e-13:
        return h
    if diff > 0:
        order = np.argsort(h, kind='mergesort')
        rem = diff
        for idx in order:
            if rem <= 1e-15: break
            space = 1.0 - h[idx]
            add = min(space, rem)
            if add > 0:
                h[idx] += add
                rem -= add
    else:
        order = np.argsort(-h, kind='mergesort')
        rem = -diff
        for idx in order:
            if rem <= 1e-15: break
            sub = min(h[idx], rem)
            if sub > 0:
                h[idx] -= sub
                rem -= sub
    diff2 = target - float(h.sum())
    if abs(diff2) > 1e-13:
        # distribute to any with capacity
        if diff2 > 0:
            for idx in range(n):
                if h[idx] < 1.0 - 1e-15:
                    h[idx] = min(1.0, h[idx] + diff2)
                    break
        else:
            for idx in range(n):
                if h[idx] > 1e-15:
                    h[idx] = max(0.0, h[idx] + diff2)
                    break
    return h
```

```
import time
import numpy as np
from scipy.optimize import minimize
```

```python
def repair_feasible(vec):
    h = np.asarray(vec, dtype=np.float64).ravel()
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    n = h.size
    target = 0.5 * float(n)
    s = float(h.sum())
    diff = target - s
    if abs(diff) <= 1e-13:
        return h

    if diff > 0.0:
        order = np.argsort(h, kind="mergesort")
        rem = diff
        for i in order:
            if rem <= 1e-15:
                break
            space = 1.0 - h[i]
            add = min(space, rem)
            if add > 0.0:
                h[i] += add
                rem -= add
    else:
        order = np.argsort(-h, kind="mergesort")
        rem = -diff
        for i in order:
            if rem <= 1e-15:
                break
            sub = min(h[i], rem)
            if sub > 0.0:
                h[i] -= sub
                rem -= sub

    diff2 = target - float(h.sum())
    if abs(diff2) > 1e-13:
        if diff2 > 0.0:
            for i in range(n):
                if h[i] < 1.0 - 1e-15:
                    h[i] = min(1.0, h[i] + diff2)
                    break
        else:
            for i in range(n):
                if h[i] > 1e-15:
                    h[i] = max(0.0, h[i] + diff2)
                    break
    return h


def exact_score(h):
    h = np.asarray(h, dtype=np.float64).ravel()
    n = h.size
    if n == 0:
        return 1.0
    dx = 2.0 / float(n)
    corr = np.correlate(h, 1.0 - h, mode="full")
    val = float(np.max(corr) * dx)
    if not np.isfinite(val):
        return 1.0
    return max(0.0, val)


def compute_B(h):
    h = np.asarray(h, dtype=np.float64).ravel()
    n = h.size
    if n == 1:
        return np.array([float(h[0] * h[0])])
    A = np.correlate(h, h, mode="full")[n - 1:]
    pref = np.empty(n + 1, dtype=np.float64)
    pref[0] = 0.0
    np.cumsum(h, out=pref[1:])
    total = pref[n]
    pos = A[1:] + (total - pref[1:n])
    neg = A[1:] + pref[1:n]
    B = np.empty(2 * n - 1, dtype=np.float64)
    B[0] = A[0]
    B[1:n] = pos
    B[n:] = neg
    return B


def score_B(h):
    h = np.asarray(h, dtype=np.float64).ravel()
    n = h.size
    if n == 0:
        return 1.0
    B = compute_B(h)
    val = 1.0 - 2.0 * float(B.min()) / float(n)
    if not np.isfinite(val):
        val = exact_score(h)
    return max(0.0, val)


def profile_weighted_grad(h, weights):
    h = np.asarray(h, dtype=np.float64).ravel()
    weights = np.asarray(weights, dtype=np.float64).ravel()
    n = h.size
    g = np.zeros(n, dtype=np.float64)
    w0 = float(weights[0])
    if w0 != 0.0:
        g += 2.0 * w0 * h
    for m in range(1, n):
        wp = float(weights[m])
        wn = float(weights[n + m - 1])
        wa = wp + wn
        if wa != 0.0:
            g[:n - m] += wa * h[m:]
            g[m:] += wa * h[:n - m]
        if wp != 0.0:
            g[n - m:] += wp
        if wn != 0.0:
            g[:m] += wn
    return g


def single_lag_grad(h, idx):
    h = np.asarray(h, dtype=np.float64).ravel()
    n = h.size
    idx = int(idx)
    g = np.zeros(n, dtype=np.float64)
    if idx == 0:
        return 2.0 * h
    if idx < n:
        m = idx
        pos_end = True
    else:
        m = idx - n + 1
        pos_end = False
    if 1 <= m < n:
        g[:n - m] += h[m:]
        g[m:] += h[:n - m]
        if pos_end:
            g[n - m:] += 1.0
        else:
            g[:m] += 1.0
    return g


def softmin_objective(h, beta, reg=0.0):
    h = np.asarray(h, dtype=np.float64).ravel()
    B = compute_B(h)
    bmin = float(B.min())
    x = -float(beta) * (B - bmin)
    ex = np.exp(np.clip(x, -745.0, 0.0))
    s = float(ex.sum())
    if s <= 0.0 or not np.isfinite(s):
        log_s = 0.0
    else:
        log_s = float(np.log(s))
    val = -bmin + log_s / float(beta)
    if reg != 0.0:
        val += float(reg) * float(np.sum(h * (1.0 - h)))
    if not np.isfinite(val):
        val = 1.0
    return val


def softmin_gradient(h, beta, reg=0.0):
    h = np.asarray(h, dtype=np.float64).ravel()
    B = compute_B(h)
    bmin = float(B.min())
    x = -float(beta) * (B - bmin)
    ex = np.exp(np.clip(x, -745.0, 0.0))
    s = float(ex.sum())
    if s <= 0.0 or not np.isfinite(s):
        w = np.zeros_like(B, dtype=np.float64)
        w[int(np.argmin(B))] = 1.0
    else:
        w = ex / s
    g = -profile_weighted_grad(h, w)
    if reg != 0.0:
        g += float(reg) * (1.0 - 2.0 * h)
    return g


class SoftMinProblem:
    def __init__(self, n, beta, reg=0.0):
        self.n = int(n)
        self.beta = float(beta)
        self.reg = float(reg)

    def fun(self, x):
        h = np.asarray(x, dtype=np.float64).ravel()
        h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
        h = np.clip(h, 0.0, 1.0)
        return softmin_objective(h, self.beta, self.reg)

    def jac(self, x):
        h = np.asarray(x, dtype=np.float64).ravel()
        h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
        h = np.clip(h, 0.0, 1.0)
        return softmin_gradient(h, self.beta, self.reg)


class SymSoftMinProblem:
    def __init__(self, n_full, beta, reg=0.0):
        self.n_full = int(n_full)
        self.m = self.n_full // 2
        self.beta = float(beta)
        self.reg = float(reg)

    def full(self, x):
        s = np.asarray(x, dtype=np.float64).ravel()
        if s.size != self.m:
            s = np.resize(s, self.m)
        s = np.nan_to_num(s, nan=0.5, posinf=1.0, neginf=0.0)
        s = np.clip(s, 0.0, 1.0)
        return np.concatenate((s, s[::-1]))

    def fun(self, x):
        h = self.full(x)
        return softmin_objective(h, self.beta, self.reg)

    def jac(self, x):
        s = np.asarray(x, dtype=np.float64).ravel()
        s = np.clip(np.nan_to_num(s, nan=0.5, posinf=1.0, neginf=0.0), 0.0, 1.0)
        h = self.full(s)
        g = softmin_gradient(h, self.beta, self.reg)
        return g[:self.m] + g[self.m:][::-1]


class SumConstraint:
    def __init__(self, n, target=None):
        self.n = int(n)
        if target is None:
            self.target = 0.5 * float(n)
        else:
            self.target = float(target)

    def fun(self, x):
        return float(np.sum(x) - self.target)

    def jac(self, x):
        return np.ones(len(x), dtype=np.float64)


class TimeStopper:
    def __init__(self, start_time, stop_after):
        self.start_time = float(start_time)
        self.stop_after = max(0.0, float(stop_after))

    def __call__(self, *args, **kwargs):
        return (time.perf_counter() - self.start_time) >= self.stop_after


def solve_softmin(h0, beta, maxiter, reg=0.0, time_left=60.0):
    if time_left < 14.0:
        return None, None, None
    h0 = repair_feasible(h0)
    n = h0.size
    t_loc = time.perf_counter()
    stop_after = float(np.clip(time_left * 0.85, 6.0, 60.0))
    ts = TimeStopper(t_loc, stop_after)
    problem = SoftMinProblem(n, beta, reg)
    cons = SumConstraint(n)
    cons_list = [{'type': 'eq', 'fun': cons.fun, 'jac': cons.jac}]
    bounds = [(0.0, 1.0)] * n
    options = {'maxiter': int(max(maxiter, 1)), 'ftol': 1e-12, 'disp': False}

    res = None
    try:
        res = minimize(problem.fun, h0, jac=problem.jac, method="SLSQP",
                       bounds=bounds, constraints=cons_list, options=options,
                       callback=ts)
    except Exception:
        try:
            options2 = {'maxiter': int(min(maxiter, 35)), 'ftol': 1e-12, 'disp': False}
            res = minimize(problem.fun, h0, jac=problem.jac, method="SLSQP",
                           bounds=bounds, constraints=cons_list, options=options2)
        except Exception:
            return None, None, None

    if res is not None and getattr(res, "x", None) is not None:
        x = np.asarray(res.x, dtype=np.float64)
        if x.size == n and np.all(np.isfinite(x)):
            h = repair_feasible(x)
            sc = exact_score(h)
            if np.isfinite(sc):
                f = softmin_objective(h, beta, reg)
                return h, sc, f
    return None, None, None

def anneal_softmin(start, betas, maxiter, reg=0.0, time_budget=300.0):
    best_h = repair_feasible(start)
    best_score = exact_score(best_h)
    work = best_h.copy()
    t0 = time.perf_counter()

    for beta in betas:
        el = time.perf_counter() - t0
        if el >= max(0.0, time_budget) - 6.0:
            break
        remain = time_budget - el
        if remain < 25.0:
            break
        beta = float(beta)
        f_before = softmin_objective(work, beta, reg)
        h_sol, sc, f_after = solve_softmin(work, beta, maxiter, reg,
                                           time_left=min(55.0, remain * 0.85))
        if h_sol is None:
            continue
        sc = exact_score(h_sol)
        if np.isfinite(sc) and sc < best_score:
            best_h = h_sol
            best_score = sc
        work_score = exact_score(work)
        if (f_after is not None and f_after < f_before - 1e-10) or (sc < work_score - 1e-12):
            work = h_sol
    return best_h, best_score, repair_feasible(work)


def symmetrize(h):
    h = np.asarray(h, dtype=np.float64).ravel()

def transfer_dimension(h, target_n):
    h = np.asarray(h, dtype=np.float64).ravel()
    m = h.size
    target_n = int(target_n)
    if target_n < 8:
        target_n = 8
    dx_s = 2.0 / float(m)
    dx_t = 2.0 / float(target_n)
    edges = np.arange(target_n + 1, dtype=np.float64) * dx_t
    y = edges / dx_s
    idx = np.floor(y).astype(np.int64)
    frac = y - idx.astype(np.float64)
    cum = np.empty(m + 1, dtype=np.float64)
    cum[0] = 0.0
    np.cumsum(h, out=cum[1:])
    total = cum[m]
    area = np.zeros(target_n + 1, dtype=np.float64)
    valid = idx < m
    idx_cl = np.clip(idx, 0, m)
    area[valid] = cum[idx_cl[valid]] + frac[valid] * h[idx_cl[valid]]
    area[~valid] = total
    area = np.maximum.accumulate(area)
    vals = (area[1:] - area[:-1]) / dx_t
    return repair_feasible(np.clip(vals, 0.0, 1.0))


def balanced_box_direction(g):
    g = np.asarray(g, dtype=np.float64).ravel()
    n = g.size
    d = np.zeros(n, dtype=np.float64)
    if n == 0:
        return d
    order = np.argsort(g, kind="mergesort")[::-1]
    if n % 2 == 0:
        p = n // 2
        d[order[:p]] = 1.0
        d[order[p:]] = -1.0
    else:
        p = (n - 1) // 2
        d[order[:p]] = 1.0
        d[order[p + 1:]] = -1.0
    return d


def lp_direction(h, K, B=None):
    if linprog is None:
        return None
    h = np.asarray(h, dtype=np.float64).ravel()
    n = h.size
    if n < 12 or n > 380:
        return None
    if B is None:
        B = compute_B(h)
    K = int(max(1, min(K, len(B))))
    idxs = np.argsort(B, kind="mergesort")[:K]
    G = np.zeros((K, n), dtype=np.float64)
    for r in range(K):
        G[r] = single_lag_grad(h, int(idxs[r]))
    norms = np.linalg.norm(G, axis=1)
    keep = norms > 1e-12
    if not np.any(keep):
        return None
    G = G[keep]
    G = G / np.linalg.norm(G, axis=1)[:, None]
    K2 = G.shape[0]
    A_ub = np.zeros((K2, n + 1), dtype=np.float64)
    A_ub[:, :n] = -G
    A_ub[:, n] = 1.0
    b_ub = np.zeros(K2, dtype=np.float64)
    A_eq = np.zeros((1, n + 1), dtype=np.float64)
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1, dtype=np.float64)
    c = np.zeros(n + 1, dtype=np.float64)
    c[n] = -1.0
    bounds = [(-1.0, 1.0)] * n + [(None, None)]
    res = None
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method="highs")
    except Exception:
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds)
        except Exception:
            return None
    if res is None or getattr(res, "x", None) is None:
        return None
    x = np.asarray(res.x, dtype=np.float64)
    if x.size != n + 1 or not np.all(np.isfinite(x[:n])):
        return None
    if x[n] <= 1e-9:
        return None
    d = x[:n]
    d -= np.mean(d)
    return d

            d_lp = lp_direction(best, Kuse, B=B)
            if d_lp is not None:
                directions.append(d_lp)
        d_box = balanced_box_direction(g)
        if np.max(np.abs(d_box)) > 1e-14:
            directions.append(d_box)
        g_norm = float(np.max(np.abs(g)))
        if g_norm > 1e-12:
            directions.append(g / g_norm)

        improved = False
        amps = (0.08, 0.03, 0.012, 0.005, 0.002, 0.001, 0.0004, 0.00015)
        for d in directions:
            d = np.asarray(d, dtype=np.float64).ravel()
            if d.size != best.size:
                continue
            d = d - np.mean(d)
            amax = float(np.max(np.abs(d)))
            if amax < 1e-12:
                continue
            d = d / amax
            for amp in amps:
                trial = repair_feasible(best + amp * d)
                sc = exact_score(trial)
                if np.isfinite(sc) and sc < best_score - 5e-13:
                    best = trial
                    best_score = sc
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break

    return best, best_score


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()
    deadline = t_start + float(budget_s)
    rng = np.random.RandomState(int(seed))

    try:
        parent = np.asarray(initial_h_values, dtype=np.float64)
    except Exception:
        parent = None

    if parent is None or parent.size == 0:
        parent = rotation_start(140, 0.405)

    best_h = repair_feasible(parent)
    best_score = exact_score(best_h)

    init_list = []
    init_list.append(best_h)
    init_list.append(symmetrize(best_h))
    init_list.append(threshold_binary(best_h))
    for alpha in (0.381966, 0.414214, 0.618034):
        init_list.append(rotation_start(best_h.size, alpha))
    for h_c in init_list:
        h_c = repair_feasible(h_c)
        sc = exact_score(h_c)
        if np.isfinite(sc) and sc < best_score:
            best_h = h_c
            best_score = sc

    rem = deadline - time.perf_counter()
    if rem > 130.0:
        n_best = best_h.size
        maxit = 120 if n_best <= 180 else 85 if n_best <= 260 else 55
        time_alloc = min(370.0, 0.36 * float(budget_s), rem - 12.0)
        best_h, best_score, work = anneal_softmin(
            best_h,
            [5.0, 10.0, 20.0, 40.0, 80.0, 160.0, 320.0, 640.0],
            maxit,
            0.0,
            time_alloc
        )
        rem = deadline - time.perf_counter()
        if rem > 45.0:
            h2, sc2 = active_descent(work, 30, 35, True, min(45.0, rem * 0.45))
            if sc2 < best_score:
                best_h = h2
                best_score = sc2

    rem = deadline - time.perf_counter()
    if rem > 65.0:
        for K in (16, 45):
            rem = deadline - time.perf_counter()
            if rem < 30.0:
                break
            h1, sc1 = active_descent(best_h, 35, K, True, min(38.0, rem * 0.45))
            if sc1 < best_score:
                best_h = h1
                best_score = sc1

    rem = deadline - time.perf_counter()
    if rem > 75.0 and best_h.size % 2 == 0:
        work = best_h
        work_score = exact_score(work)
        for beta in (10.0, 30.0, 100.0, 300.0):
            rem = deadline - time.perf_counter()
            if rem < 38.0:
                break
            h_sol, sc, f = solve_softmin_symmetric(
                work, beta, 110, 1e-6, min(50.0, rem * 0.55)
            )
            if h_sol is not None and np.isfinite(sc):
                if sc < best_score:
                    best_h = h_sol
                    best_score = sc
                if sc < work_score - 1e-12:
                    work = h_sol
                    work_score = sc

    rem = deadline - time.perf_counter()
    if rem > 120.0:
        nb = int(best_h.size)
        targets = []
        for t in (nb + 40, 2 * nb, nb + 80):
            t = int(t)
            t = 2 * (t // 2)
            if 48 <= t <= 420 and t not in targets:
                targets.append(t)
        targets.sort()
        for target in targets[:2]:
            rem = deadline - time.perf_counter()
            if rem < 95.0:
                break
            h_t = transfer_dimension(best_h, target)
            st = exact_score(h_t)
            work_t = h_t
            work_score = st
            for beta in (12.0, 50.0, 160.0):
                rem = deadline - time.perf_counter()
                if rem < 45.0:
                    break
                maxit = 75 if target <= 220 else 45
                h_sol, sc, f = solve_softmin(
                    work_t, beta, maxit, 1e-6, min(42.0, rem * 0.45)
                )
                if h_sol is not None and np.isfinite(sc):
                    if sc < best_score:
                        best_h = h_sol
                        best_score = sc
                    if sc < work_score - 1e-12:
                        work_t = h_sol
                        work_score = sc
            rem = deadline - time.perf_counter()
            if rem > 32.0:
                h_a, sc_a = active_descent(work_t, 22, min(35, target // 6), True, min(28.0, rem * 0.45))
                if sc_a < best_score:
                    best_h = h_a
                    best_score = sc_a

    rem = deadline - time.perf_counter()
    if rem > 160.0:
        starts = []
        nb = best_h.size
        for scale in (0.004, 0.015, 0.045):
            z = rng.normal(0.0, float(scale), size=nb)
            z -= np.mean(z)
            starts.append(repair_feasible(best_h + z))
        for alpha in (0.379, 0.618):
            starts.append(rotation_start(nb, alpha))
        starts.append(random_binary_start(nb, rng))
        for hstart in starts[:5]:
            rem = deadline - time.perf_counter()
            if rem < 62.0:

            work = repair_feasible(hstart)
            work_score = exact_score(work)
            for beta in (25.0, 100.0):
                rem = deadline - time.perf_counter()
                if rem < 42.0:
                    break
                h_sol, sc, f = solve_softmin(work, beta, 80, 1e-6, min(34.0, rem * 0.5))
                if h_sol is not None and np.isfinite(sc):
                    if sc < best_score:
                        best_h = h_sol
                        best_score = sc
                    if sc < work_score - 1e-12:

                        work_score = sc
            rem = deadline - time.perf_counter()
            if rem > 25.0:
                h_a, sc_a = active_descent(work, 18, 35, True, min(22.0, rem * 0.5))
                if sc_a < best_score:
                    best_h = h_a
                    best_score = sc_a

    rem = deadline - time.perf_counter()
    if rem > 55.0:
        h_sol, sc, f = solve_softmin(best_h, 500.0, 180, 1e-5, min(55.0, rem * 0.7))
        if h_sol is not None and sc < best_score:
            best_h = h_sol
            best_score = sc

    rem = deadline - time.perf_counter()
    if rem > 40.0:
        h_a, sc_a = active_descent(best_h, 50, 60, True, min(40.0, rem * 0.75))
        if sc_a < best_score:
            best_h = h_a
            best_score = sc_a

    rem = deadline - time.perf_counter()
    if rem > 26.0:
        h_sol, sc, f = solve_softmin(best_h, 1500.0, 140, 1e-5, min(26.0, rem * 0.8))
        if h_sol is not None and sc < best_score:
            best_h = h_sol
            best_score = sc

    best_h = repair_feasible(best_h)
    best_score = exact_score(best_h)
    if not np.isfinite(best_score):
        best_score = 1.0
    return best_h, float(best_score), int(best_h.size)
```
