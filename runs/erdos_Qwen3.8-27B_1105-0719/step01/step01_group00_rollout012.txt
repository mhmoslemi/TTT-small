import numpy as np
from scipy.optimize import minimize, linprog
from scipy.special import logsumexp
import time

def project_feasible(x, target_sum):
    """Project x to [0,1]^n with sum = target_sum."""
    n = len(x)
    lo, hi = -1.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def c5_of_h(h, dx):
    return np.correlate(h, 1.0 - h, mode='full').max() * dx

def grad_corr_k(h, k, n):
    """Gradient of the k-th correlation (index k in full mode) w.r.t. h."""
    offset = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -offset)
    i_hi = min(n - 1, n - 1 - offset)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx + offset] += (1.0 - h[idx])
    grad[idx] += (-h[idx + offset])
    return grad

def gradient_smooth_max(h, dx, beta):
    n = len(h)
    corrs = np.correlate(h, 1.0 - h, mode='full')
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones_like(corrs) / len(corrs)
    else:
        w /= w_sum
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-13:
            continue
        grad += w[k] * grad_corr_k(h, k, n)
    return grad * dx

def smooth_max_value(h, dx, beta):
    corrs = np.correlate(h, 1.0 - h, mode='full')
    return logsumexp(beta * corrs) / beta * dx

def smooth_max_grad(h, dx, beta):
    return gradient_smooth_max(h, dx, beta)

def local_slsqp(h0, dx, target_sum, beta, maxiter):
    n = len(h0)
    def obj(x):
        return smooth_max_value(x, dx, beta)
    def jac(x):
        return smooth_max_grad(x, dx, beta)
    constraints = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum,
                   'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0, jac=jac, method='SLSQP', bounds=bounds,
                       constraints=constraints,

        h_out = project_feasible(res.x, target_sum)
    except Exception:
        h_out = project_feasible(h0, target_sum)
    return h_out, c5_of_h(h_out, dx)
