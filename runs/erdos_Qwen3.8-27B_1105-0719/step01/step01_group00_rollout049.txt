def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def grad_corr_k(h, k, n):

    grad = np.zeros(n)

    i_hi = min(n - 1, n - 1 - offset)

    idx = np.arange(i_lo, i_hi + 1)
    grad[idx + offset] += (1.0 - h[idx])
    grad[idx] += (-h[idx + offset])
    return grad

Actually, let me think about a completely different approach: **alternating direction method of multipliers (ADMM)** or **proximal gradient**.

The problem is: minimize max_k f_k(h) s.t. sum(h) = n/2, 0 ≤ h ≤ 1.

We can write this as: minimize t s.t. f_k(h) ≤ t for all k, sum(h) = n/2, 0 ≤ h ≤ 1.

Since f_k is bilinear (h_i * h_{i+k}), this is a non-convex QCQP.

One approach: **Alternating minimization** - fix h, the constraint is t ≥ max_k f_k(h) (trivially satisfied). Fix t, the constraint is f_k(h) ≤ t for all k (non-convex).

Another approach: **Lagrangian relaxation** - for multipliers λ_k ≥ 0 on each constraint f_k(h) ≤ t:
L = t + Σ_k λ_k (f_k(h) - t) = t(1 - Σλ_k) + Σ_k λ_k f_k(h)

If we normalize Σλ_k = 1, this becomes: L = Σ_k λ_k f_k(h).
Then we minimize over h: min_h Σ_k λ_k f_k(h) s.t. sum(h) = n/2, 0≤h≤1.

But f_k(h) = Σ_i h_i(1-h_{i+k}) * dx is bilinear, so Σ λ_k f_k(h) is also bilinear in h, hence non-convex.

However, if we linearize around the current h, we get a quadratic program that we can solve exactly. This is essentially a sequential quadratic programming (SQP) approach.

Let me try a **linearized approach**:
1. At current h, linearize: f_k(h + δ) ≈ f_k(h) + ∇f_k(h)^T δ
2. Solve: min_t t s.t. f_k(h) + ∇f_k(h)^T δ ≤ t for all k, sum(h+δ) = n/2, 0 ≤ h+δ ≤ 1
3. This is a linear program (LP)! We can solve it with scipy or cvxpy.
4. Take a step in the δ direction (with damping if needed).

This is essentially a **cutting plane / linearized** method. It should work well because:
- Each subproblem is an LP (easy to solve)
- The linearization gets better as we converge
- It naturally handles the max of many constraints

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.special import logsumexp
import time


def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)


def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def c5_of_h(h, dx):
    return compute_corrs(h).max() * dx


def grad_corr_k(h, k, n):
    offset = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -offset)
    i_hi = min(n - 1, n - 1 - offset)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx + offset] += (1.0 - h[idx])
    grad[idx] += (-h[idx + offset])
    return grad


def gradient_smooth_max(h, dx, beta):
    n = len(h)
    corrs = compute_corrs(h)
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones_like(corrs) / len(corrs)
    else:
        w /= w_sum
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] > 1e-13:
            grad += w[k] * grad_corr_k(h, k, n)
    return grad * dx


def sigmoid_2z(z):
    return 1.0 / (1.0 + np.exp(-np.clip(2.0 * z, -500, 500)))


def h_to_z(h):
    h = np.clip(h, 1e-12, 1.0 - 1e-12)
    return -0.5 * np.log((1.0 - h) / h)


def slsqp_direct(h0, dx, target_sum, beta, maxiter):
    n = len(h0)

    def obj(x):
        corrs = compute_corrs(x)
        return logsumexp(beta * corrs) / beta * dx

    def grad_obj(x):
        return gradient_smooth_max(x, dx, beta)

    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0, jac=grad_obj, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        h_out = project_feasible(res.x, target_sum)
    except Exception:
        h_out = project_feasible(h0, target_sum)
    return h_out, c5_of_h(h_out, dx)


def slsqp_z_space(z0, dx, target_sum, beta, maxiter):
    n = len(z0)

    def obj(z):
        h = sigmoid_2z(z)
        corrs = compute_corrs(h)
        return logsumexp(beta * corrs) / beta * dx

    def grad_obj(z):
        h = sigmoid_2z(z)
        grad_h = gradient_smooth_max(h, dx, beta)
        return 2.0 * h * (1.0 - h) * grad_h

    def c_fun(z):
        return sigmoid_2z(z).sum() - target_sum

    def c_jac(z):
        h = sigmoid_2z(z)
        return 2.0 * h * (1.0 - h)

    cons = [{'type': 'eq', 'fun': c_fun, 'jac': c_jac}]
    try:
        res = minimize(obj, z0, jac=grad_obj, method='SLSQP',
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        h_out = project_feasible(sigmoid_2z(res.x), target_sum)
    except Exception:
        h_out = project_feasible(sigmoid_2z(z0), target_sum)
    return h_out, c5_of_h(h_out, dx)


def lp_polish(h0, dx, target_sum, n_steps, lam):
    """Linearized LP polishing steps."""
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)

    for s in range(n_steps):
        corrs = compute_corrs(h)
        max_c = corrs.max()
        c5_val = max_c * dx

        # Select active constraints (top fraction)
        n_active = max(10, int(0.3 * (2 * n - 1)))
        top_idx = np.argsort(corrs)[-n_active:]

        # Build LP
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0

        A_rows = []
        b_rows = []
        for k in top_idx:
            row = np.zeros(n + 1)
            row[:n] = dx * grad_corr_k(h, k, n)
            row[n] = -1.0
            A_rows.append(row)
            b_rows.append(-corrs[k] * dx - 1e-12)

        b_ub = np.array(b_rows)

            hi_b = (1.0 - h[i]) * lam

def make_binary_start(n, target_sum, rng):

    if frac > 1e-10 and n_ones < n:

    return h


def make_periodic_start(n, target_sum, rng, period):

        if remaining_time(t0, budget_s) > 15:
            h_lp, c5_lp = lp_polish(best_h, dx, target_sum, 30, 0.2)
            if c5_lp < best_c5:
                best_c5 = c5_lp
                best_h = h_lp

    # Phase 2: Diverse starts
    for trial in range(20):
        if remaining_time(t0, budget_s) < 12:
            break

        if trial % 5 == 0:
            h_init = make_binary_start(n, target_sum, rng)
        elif trial % 5 == 1:
            h_init = make_periodic_start(n, target_sum, rng, rng.choice([3, 4, 5, 6, 7]))
        elif trial % 5 == 2:
            h_init = make_two_block_start(n, target_sum, rng)
        elif trial % 5 == 3:
            h_init = make_binary_start(n, target_sum, rng)
        else:
            h_init = project_feasible(best_h + 0.2 * (rng.random(n) - 0.5), target_sum)

        h_opt, c5_val = slsqp_direct(h_init, dx, target_sum, 200, 500)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_opt
            if remaining_time(t0, budget_s) > 20:
                h_opt2, c5_val2 = slsqp_direct(h_opt, dx, target_sum, 800, 600)
                if c5_val2 < best_c5:
                    best_c5 = c5_val2
                    best_h = h_opt2
                if remaining_time(t0, budget_s) > 12:
                    h_lp, c5_lp = lp_polish(h_opt, dx, target_sum, 30, 0.25)
                    if c5_lp < best_c5:
                        best_c5 = c5_lp
                        best_h = h_lp

    # Phase 3: Try different n values
    n_candidates = [62, 64, 66, 90, 92, 100, 110, 126, 128, 130, 132, 136, 140]
    for n_try in n_candidates:
        if remaining_time(t0, budget_s) < 25:
            break
        if n_try == len(best_h):
            continue
        dx_try = 2.0 / n_try
        ts_try = n_try / 2.0
        x_orig = np.linspace(0.0, 2.0, len(best_h))
        x_new = np.linspace(0.0, 2.0, n_try)
        h_interp = np.interp(x_new, x_orig, best_h)
        h_interp = project_feasible(h_interp, ts_try)

        h_opt, c5_val = slsqp_direct(h_interp, dx_try, ts_try, 300, 800)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_opt
            if remaining_time(t0, budget_s) > 20:
                z_opt = h_to_z(best_h)
                h_z, c5_z = slsqp_z_space(z_opt, dx_try, ts_try, 800, 800)
                if c5_z < best_c5:
                    best_c5 = c5_z
                    best_h = h_z
            if remaining_time(t0, budget_s) > 15:
                h_lp, c5_lp = lp_polish(best_h, 2.0 / len(best_h), len(best_h) / 2.0, 30, 0.2)
                if c5_lp < best_c5:
                    best_c5 = c5_lp
                    best_h = h_lp

    # Phase 4: Final intensive polish
    if remaining_time(t0, budget_s) > 15 and best_h is not None:
        n_b = len(best_h)
        dx_b = 2.0 / n_b
        ts_b = n_b / 2.0

        h_final, c5_final = slsqp_direct(best_h, dx_b, ts_b, 3000, 1500)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final

        if remaining_time(t0, budget_s) > 12:
            z_final = h_to_z(best_h)
            h_zf, c5_zf = slsqp_z_space(z_final, dx_b, ts_b, 2000, 1200)
            if c5_zf < best_c5:
                best_c5 = c5_zf
                best_h = h_zf

        if remaining_time(t0, budget_s) > 10:
            h_lp, c5_lp = lp_polish(best_h, dx_b, ts_b, 40, 0.15)
            if c5_lp < best_c5:
                best_c5 = c5_lp
                best_h = h_lp

        if remaining_time(t0, budget_s) > 8:
            h_sg, c5_sg = subgradient_v3(best_h, dx_b, ts_b, 8000)
            if c5_sg < best_c5:
                best_c5 = c5_sg
                best_h = h_sg

        if remaining_time(t0, budget_s) > 5:
            h_final2, c5_f2 = slsqp_direct(best_h, dx_b, ts_b, 4000, 1000)
            if c5_f2 < best_c5:
                best_c5 = c5_f2
                best_h = h_final2

    return best_h, best_c5, len(best_h)
