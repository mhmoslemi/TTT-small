def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def c5_of_h(h, dx):
    return compute_corrs(h).max() * dx


def grad_corr_k(h, k, n):
    offset = k - (n - 1)
    grad = np.zeros(n)
    if offset >= 0:
        valid = n - offset
        if valid > 0:
            grad[:valid] += (1.0 - h[offset:])
            grad[offset:] += (-h[:valid])
    else:
        o = -offset
        valid = n - o
        if valid > 0:
            grad[o:] += (1.0 - h[:valid])
            grad[:valid] += (-h[o:])
    return grad


class SmoothMaxProblem:
    def __init__(self, dx, beta, target_sum, n):
        self.dx = dx
        self.beta = beta
        self.target_sum = target_sum
        self.n = n

    def objective(self, x):
        return logsumexp(self.beta * compute_corrs(x)) / self.beta * self.dx

    def gradient(self, x):
        n = self.n
        corrs = compute_corrs(x)
        shifted = corrs - corrs.max()
        w = np.exp(self.beta * shifted)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones_like(corrs) / len(corrs)
        else:
            w /= w_sum
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] < 1e-14:
                continue
            grad += w[k] * grad_corr_k(x, k, n)
        return grad * self.dx

    def cons_fun(self, x):
        return x.sum() - self.target_sum

    def cons_jac(self, x):
        return np.ones(self.n)

    prob = SmoothMaxProblem(dx, beta, target_sum, n)
    cons = [{'type': 'eq', 'fun': prob.cons_fun, 'jac': prob.cons_jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(prob.objective, h0, jac=prob.gradient,
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})

    # Phase 4: Perturbation search
    if remaining() > 80:
        h_pert, c_pert = perturbation_search(best_h, dx, target_sum, rng,
                                             n_trials=30, sigma=0.04)
        if c_pert < best_c5:
            best_c5 = c_pert
            best_h = h_pert

    # Phase 5: Try nearby n values
    n_candidates = [n - 2, n - 1, n + 1, n + 2, n + 4]
    for n_try in n_candidates:
        if n_try < 20:
            continue
        if remaining() < 45:
            break
        dx_try = 2.0 / n_try
        ts_try = n_try / 2.0
        x_orig = np.linspace(0.0, 2.0, len(best_h))
        x_new = np.linspace(0.0, 2.0, n_try)
        h_interp = np.interp(x_new, x_orig, best_h)
        h_interp = project_feasible(h_interp, ts_try)
        h_opt = slsqp_optimize(h_interp, dx_try, ts_try, 800, 500)
        c = c5_of_h(h_opt, dx_try)
        if c < best_c5:
            best_c5 = c
            best_h = h_opt
            if remaining() > 30:
                h2 = slsqp_optimize(best_h, 2.0 / len(best_h), len(best_h) / 2.0,
                                    2000, 500)
                c2 = c5_of_h(h2, 2.0 / len(best_h))
                if c2 < best_c5:
                    best_c5 = c2
                    best_h = h2

    # Phase 6: Final high-precision polish
    if remaining() > 20:
        n_b = len(best_h)
        dx_b = 2.0 / n_b
        ts_b = n_b / 2.0
        h_final = slsqp_optimize(best_h, dx_b, ts_b, 3000, 1000)
        c = c5_of_h(h_final, dx_b)
        if c < best_c5:
            best_c5 = c
            best_h = h_final

    if remaining() > 15:
        n_b = len(best_h)
        dx_b = 2.0 / n_b
        ts_b = n_b / 2.0
        h_sl2, c_sl2 = single_lag_optimize(best_h, dx_b, ts_b, max_rounds=60)
        if c_sl2 < best_c5:
            best_c5 = c_sl2
            best_h = h_sl2

    # Final check
    if remaining() > 8:
        n_b = len(best_h)
        dx_b = 2.0 / n_b
        ts_b = n_b / 2.0
        h_fin = slsqp_optimize(best_h, dx_b, ts_b, 5000, 800)
        c = c5_of_h(h_fin, dx_b)
        if c < best_c5:
            best_c5 = c
            best_h = h_fin

    return best_h, best_c5, len(best_h)
