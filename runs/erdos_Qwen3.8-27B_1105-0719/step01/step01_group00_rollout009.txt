def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full').max() * dx

def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def grad_corr_k(h, k, n):
    """Gradient of the k-th correlation value with respect to h."""
    offset = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -offset)
    i_hi = min(n - 1, n - 1 - offset)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    # c[k] = sum_i h[i+offset] * (1-h[i])
    # d/dc h[i+offset] = (1-h[i]) -> grad[i+offset] += (1-h[i])
    # d/dc (1-h[i]) = -h[i+offset] -> grad[i] += -h[i+offset]
    grad[idx + offset] += (1.0 - h[idx])
    grad[idx] += (-h[idx + offset])
    return grad

def make_sym_h(q, full_n):

        return np.concatenate([q, 1.0 - q[::-1]])
    else:
        return np.concatenate([q, [0.5], 1.0 - q[::-1]])

import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time


def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full').max() * dx


def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def grad_corr_k(h, k, n):
    offset = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -offset)
    i_hi = min(n - 1, n - 1 - offset)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx + offset] += (1.0 - h[idx])
    grad[idx] += (-h[idx + offset])
    return grad


def make_sym_h(q, full_n):
    m = full_n // 2
    if full_n % 2 == 0:
        return np.concatenate([q, 1.0 - q[::-1]])
    else:
        return np.concatenate([q, [0.5], 1.0 - q[::-1]])


def make_q_from_h(h, full_n):
    n = full_n
    m = n // 2
    if n % 2 == 0:
        q = 0.5 * (h[:m] + 1.0 - h[m:][::-1])
    else:
        tail = h[m + 1:]
        tail_rev = tail[::-1]
        q = 0.5 * (h[:m] + 1.0 - tail_rev)
    return np.clip(q, 0.0, 1.0)


def q_to_full_grad(full_grad, n):
    m = n // 2
    if n % 2 == 0:
        return full_grad[:m] - full_grad[m:][::-1]
    else:
        return full_grad[:m] - full_grad[m + 1:][::-1]


def smooth_obj_q(q, full_n, beta):
    h = make_sym_h(q, full_n)
    n = full_n
    dx = 2.0 / n
    corrs = compute_corrs(h)
    return logsumexp(beta * corrs) / beta * dx


def smooth_grad_q(q, full_n, beta):
    h = make_sym_h(q, full_n)
    n = full_n
    dx = 2.0 / n
    corrs = compute_corrs(h)
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones_like(corrs) / len(corrs)
    else:
        w /= w_sum
    full_grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        full_grad += w[k] * grad_corr_k(h, k, n)
    full_grad *= dx
    return q_to_full_grad(full_grad, n)


def smooth_optimize_q(q0, full_n, beta, maxiter):
    m = full_n // 2
    bounds = [(0.0, 1.0)] * m

    def obj(q):
        return smooth_obj_q(q, full_n, beta)

    def grad(q):
        return smooth_grad_q(q, full_n, beta)

    try:
        res = minimize(obj, q0, jac=grad, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10,
                                'maxcor': 20, 'maxls': 40})
        q_out = np.clip(res.x, 0.0, 1.0)
        h_out = make_sym_h(q_out, full_n)
        c5 = exact_score(h_out)
        return q_out, c5
    except Exception:
        h0 = make_sym_h(q0, full_n)
        return q0, exact_score(h0)


def epigraph_q(q0, full_n, t0, active_lags, maxiter):
    n = full_n
    m = n // 2
    dx = 2.0 / n

    def obj(y):
        return y[-1]

    def obj_grad(y):
        g = np.zeros(len(y))
        g[-1] = 1.0
        return g

    def cons_fun(y):
        q = y[:-1]
        t = y[-1]
        h = make_sym_h(q, full_n)
        corrs = compute_corrs(h) * dx
        return t - corrs[active_lags]

    def cons_jac(y):
        q = y[:-1]
        h = make_sym_h(q, full_n)
        L = len(active_lags)
        J = np.zeros((L, len(y)))
        J[:, -1] = 1.0
        for li, r in enumerate(active_lags):
            g_full = grad_corr_k(h, r, n) * dx
            g_q = q_to_full_grad(g_full, n)
            J[li, :m] = -g_q
        return J

    bounds = [(0.0, 1.0)] * m + [(t0 - 0.01, t0 + 0.5)]
    x0 = np.concatenate([q0, [t0]])
    cons = [{'type': 'ineq', 'fun': cons_fun, 'jac': cons_jac}]

    try:
        res = minimize(obj, x0, jac=obj_grad, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-13})
        q_out = np.clip(res.x[:-1], 0.0, 1.0)
        h_out = make_sym_h(q_out, full_n)
        c5 = exact_score(h_out)
        return q_out, c5
    except Exception:
        h0 = make_sym_h(q0, full_n)
        return q0, exact_score(h0)


def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)


def subgradient_q(q0, full_n, max_iter, tol):
    n = full_n
    m = n // 2
    q = q0.copy()
    best_q = q.copy()
    best_c5 = exact_score(make_sym_h(q, full_n))
    alpha = 0.1
    no_improve = 0

    for it in range(max_iter):
        h = make_sym_h(q, full_n)
        corrs = compute_corrs(h)
        max_c = corrs.max()
        c5 = max_c * (2.0 / n)

        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_q = q.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 2000:
            break

        threshold = max_c * 0.95
        active = np.where(corrs >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(corrs)])

        full_grad = np.zeros(n)
        for k in active:
            full_grad += grad_corr_k(h, k, n)
        full_grad /= len(active)
        full_grad *= 2.0 / n
        g_q = q_to_full_grad(full_grad, n)

        q_new = np.clip(q - alpha * g_q, 0.0, 1.0)
        new_c5 = exact_score(make_sym_h(q_new, full_n))
        if new_c5 < c5:
            q = q_new
            alpha = min(alpha * 1.03, 0.3)
        else:
            alpha *= 0.93
        if alpha < tol:
            break

    return best_q, best_c5


def make_random_q(m, rng, mode):
    if mode == 0:
        return rng.uniform(0.2, 0.8, size=m)
    elif mode == 1:
        return rng.uniform(0, 1, size=m)
    elif mode == 2:
        x = np.arange(m, dtype=float)
        f1 = rng.uniform(1, 4)
        f2 = rng.uniform(4, 10)
        p1 = rng.uniform(0, 2 * np.pi)
        p2 = rng.uniform(0, 2 * np.pi)
        a1 = rng.uniform(0.15, 0.35)
        a2 = rng.uniform(0.05, 0.15)
        q = 0.5 + a1 * np.sin(2 * np.pi * f1 * x / m + p1) + a2 * np.sin(2 * np.pi * f2 * x / m + p2)
        return np.clip(q, 0.0, 1.0)
    elif mode == 3:
        q = np.zeros(m)
        n_block = rng.randint(3, 8)
        for b in range(0, m, n_block):
            val = rng.uniform(0.0, 1.0)
            q[b:min(b + n_block, m)] = val
        return q
    elif mode == 4:
        q = rng.uniform(0.3, 0.7, size=m)
        q[::2] += 0.15
        q[1::2] -= 0.15
        return np.clip(q, 0.0, 1.0)
    else:
        return 0.5 * np.ones(m)
