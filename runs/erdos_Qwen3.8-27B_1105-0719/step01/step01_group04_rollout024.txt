def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr * dx))


def repair_feasible(vec):
    h = np.asarray(vec, dtype=np.float64)
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)

    diff = target - s

        return h.copy()
    if diff > 0:
        order = np.argsort(h)
        rem = diff
        for i in order:
            if rem <= 1e-15:
                break
            add = min(rem, 1.0 - h[i])
            h[i] += add
            rem -= add
    else:
        order = np.argsort(-h)
        rem = -diff
        for i in order:
            if rem <= 1e-15:
                break
            remv = min(rem, h[i])
            h[i] -= remv
            rem -= remv
    diff2 = target - h.sum()
    if abs(diff2) > 1e-14:
        if diff2 > 0:

                    h[i] = min(1.0, h[i] + diff2)
                    break
        else:

                    h[i] = max(0.0, h[i] + diff2)
                    break
    return h


def compute_lag_data(h):
    n = len(h)

        else:

            grads[n - 1 + delta, idx] += dx * (1.0 - h[idx - delta])

            idx2 = np.arange(0, n - delta)

            grads[n - 1 - delta, idx2] += dx * (1.0 - h[idx2 + delta])
            grads[n - 1 - delta, idx2 + delta] += -dx * h[idx2]
    return values, grads


def solve_trust_region(h_cur, values, grads, lam, active_idx):
    n = len(h_cur)
    h = cp.Variable(n)
    t = cp.Variable()
    base = values[active_idx] - grads[active_idx] @ h_cur
    constraints = [
        base + grads[active_idx] @ h <= t,
        cp.sum(h) == n / 2.0,
        h >= 0,
        h <= 1
    ]
    obj = cp.Minimize(t + lam * cp.sum_squares(h - h_cur))
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=cp.ECOS, max_iters=5000, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, verbose=False)
        except Exception:
            return None
    if h.value is None:
        return None
    h_sol = np.array(h.value).flatten()
    return repair_feasible(h_sol)

                if s_tr < best_score - 1e-13:
                    best_h = h_tr
                    best_score = s_tr

    # Phase 6: Final trust region + active descent
    if remaining() > 40:
        h_tr, s_tr = run_trust_region_loop(
            best_h, best_score, max_iters=20,
            t_start=t_start, budget_s=budget_s
        )
        if s_tr < best_score - 1e-13:
            best_h = h_tr
            best_score = s_tr

    if remaining() > 15:
        h_ad, s_ad = active_descent(best_h, max_steps=50, t_start=t_start, time_limit=remaining() - 5)
        if s_ad < best_score - 1e-13:
            best_h = h_ad
            best_score = s_ad

    # Final validation
    best_h = repair_feasible(best_h)
    best_score = exact_score(best_h)
    return best_h, best_score, len(best_h)
