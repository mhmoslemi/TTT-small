def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")

def repair_feasible(vec):
    h = np.asarray(vec, dtype=np.float64).copy()
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(h)  # most room to add first
        rem = diff
        for i in order:
            if rem <= 1e-15:
                break
            add = min(rem, 1.0 - h[i])
            h[i] += add
            rem -= add
    else:

        for i in order:
            if rem <= 1e-15:
                break
            remv = min(rem, h[i])
            h[i] -= remv
            rem -= remv
    # Final fix

                    break
        else:

                if h[i] > 0.0:
                    h[i] = max(0.0, h[i] + diff2)
                    break
    return h

def compute_values_and_grads(h):
    """Compute exact values and gradients for all lags.
    
    Returns:
        values: array of shape (2n-1,), values[i] corresponds to lag i-(n-1)
        grads: array of shape (2n-1, n), grads[i, j] = d(values[i])/dh[j]
    """
    n = len(h)
    dx = 2.0 / n
    n_lags = 2 * n - 1
    values = np.zeros(n_lags)
    grads = np.zeros((n_lags, n))
    
    # Lag 0
    values[n-1] = dx * np.sum(h * (1.0 - h))
    grads[n-1, :] = dx * (1.0 - 2.0 * h)

        #   j = m-k, contribution = (1-h[j]) = (1-h[m-k])
        m_pos = np.arange(k, n)  # m values where h[m] appears as h[j+k]
        grads[n-1+k, m_pos] += (1.0 - h[m_pos - k])
        
        # For m in [0, n-1-k]: contribution from h[m] being in (1-h[j]) term
        #   j = m, contribution = -h[j+k] = -h[m+k]
        m_neg = np.arange(0, n-k)  # m values where h[m] appears in (1-h[j])
        grads[n-1+k, m_neg] += -h[m_neg + k]
        
        # Negative lag: values[n-1-k] = dx * sum_{j=k}^{n-1} h[j-k] * (1-h[j])
        idx2 = np.arange(k, n)  # j indices
        values[n-1-k] = dx * np.sum(h[idx2-k] * (1.0 - h[idx2]))
        
        # Gradient: dV_{-k}/dh[m] = (1-h[m+k]) for 0<=m<=n-1-k, plus -h[m-k] for k<=m<=n-1
        # For m in [0, n-1-k]: h[m] appears as h[j-k] where j=m+k
        #   contribution = (1-h[j]) = (1-h[m+k])
        m_pos2 = np.arange(0, n-k)
        grads[n-1-k, m_pos2] += (1.0 - h[m_pos2 + k])
        
        # For m in [k, n-1]: h[m] appears in (1-h[j]) where j=m
        #   contribution = -h[j-k] = -h[m-k]
        m_neg2 = np.arange(k, n)
        grads[n-1-k, m_neg2] += -h[m_neg2 - k]
    
    return values, grads

def trust_region_step(h_cur, values, grads, lam=0.1, max_active=50):
    """Solve the trust-region QP.
    
    minimize t + lam * ||h_new - h_cur||^2
    subject to:
        values[k] + grads[k,:] @ (h_new - h_cur) <= t
        sum(h_new) = n/2
        0 <= h_new <= 1
    """
    n = len(h_cur)
    n_lags = 2 * n - 1
    
    # Select active constraints (top max_active by value)
    if n_lags > max_active:
        active_idx = np.argsort(values)[::-1][:max_active]
    else:
        active_idx = np.arange(n_lags)
    
    active_idx = np.sort(active_idx)
    v_act = values[active_idx]
    G_act = grads[active_idx, :]  # shape (len(active_idx), n)
    
    h_new = cp.Variable(n)
    t = cp.Variable()
    
    # Linearized constraints: v_act + G_act @ (h_new - h_cur) <= t
    obj = t + lam * cp.sum_squares(h_new - h_cur)
    constraints = [
        G_act @ (h_new - h_cur) - t <= -v_act,  # v_act + G_act@(h_new-h_cur) <= t
        cp.sum(h_new) == n / 2.0,
        h_new >= 0,
        h_new <= 1,
    ]
    
    prob = cp.Problem(cp.Minimize(obj), constraints)
    
    try:
        prob.solve(solver=cp.CLARABEL)
    except Exception:
        try:
            prob.solve(solver=cp.SCS)
        except Exception:
            try:
                prob.solve(solver=cp.ECOS)
            except Exception:
                return None
    
    if h_new.value is None:
        return None
    
    return np.asarray(h_new.value, dtype=np.float64)

def active_descent(h, max_steps=50, n_active=20):
    """Active-set weighted gradient descent."""
    n = len(h)
    h = repair_feasible(h)
    best = h.copy()
    best_score = exact_score(best)
    
    for step in range(max_steps):
        values, grads = compute_values_and_grads(best)
        c = np.max(values)
        
        # Select active lags
        k_act = min(n_active, 2*n-1)
        active_idx = np.argsort(values)[::-1][:k_act]
        
        # Softmax weights based on gap from max
        gaps = c - values[active_idx]
        sigma = 0.005 * c + 1e-10
        weights = np.exp(-gaps / sigma)
        weights /= np.sum(weights) + 1e-15
        
        # Direction: negative weighted sum of gradients
        direction = -np.sum(weights[:, None] * grads[active_idx, :], axis=0)
        direction -= np.mean(direction)  # preserve sum constraint
        
        improved = False
        for alpha in [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]:
            h_trial = repair_feasible(best + alpha * direction)
            score = exact_score(h_trial)
            if score < best_score - 1e-12:
                best = h_trial.copy()
                best_score = score
                improved = True
                break
        
        if not improved:
            break
    
    return best, best_score

But the actual value we want is B = dx * sum_{i=0}^{n-1-delta}
