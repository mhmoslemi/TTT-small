    h = np.array(h, dtype=np.float64).copy()

def lse_refine(h0, tau, deadline):
    """LSE-based continuous refinement."""
    n = len(h0)
    dx = 2.0 / n
    target = n / 2.0
    
    def obj_grad(x):
        profile = np.correlate(x, 1.0 - x, mode='full') * dx
        m = float(np.max(profile))
        shifted = np.minimum((profile - m) / tau, 0.0)
        exp_vals = np.exp(shifted)
        sum_exp = float(np.sum(exp_vals))
        weights = exp_vals / sum_exp
        obj_val = m + tau * np.log(sum_exp)
        grad = np.zeros(n)
        for j in range(2 * n - 1):
            w = weights[j]
            if w < 1e-13:
                continue
            s = j - (n - 1)
            if s >= 0:
                k1 = np.arange(s, n)
                grad[k1] += w * dx * (1.0 - x[k1 - s])
                k2 = np.arange(0, n - s)
                grad[k2] -= w * dx * x[k2 + s]
            else:
                a = -s
                k1 = np.arange(0, n - a)
                grad[k1] += w * dx * (1.0 - x[k1 + a])
                k2 = np.arange(a, n)
                grad[k2] -= w * dx * x[k2 - a]
        return obj_val, grad
    
    class Obj:
        def value(self, x):
            return obj_grad(x)[0]
        def gradient(self, x):
            return obj_grad(x)[1]
    
    class SumCon:
        def fun(self, x):
            return np.sum(x) - target
        def jac(self, x):
            return np.ones(len(x))
    
    h = project_box_sum(h0.copy())
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(Obj().value, h, method='SLSQP', jac=Obj().gradient,
                      bounds=bounds,
                      constraints=[{'type': 'eq', 'fun': SumCon().fun, 'jac': SumCon().jac}],
                      options={'maxiter': 200, 'ftol': 1e-12})
        h_new = project_box_sum(np.array(res.x, dtype=np.float64))
        if exact_objective(h_new) <= exact_objective(h) + 1e-10:
            return h_new
    except Exception:
        pass
    return h
