def compute_c5(h):
    """Exact C5 evaluation."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    """Project y onto {h: 0<=h<=1, sum(h)=n/2}."""
    target = n / 2.0
    y = np.array(y, dtype=float)
    # Bisection for the threshold
    lo = np.min(y) - 2.0
    hi = np.max(y) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    h = np.clip(y - t, 0.0, 1.0)

    if abs(residual) > 1e-10:
        eligible = np.where((h > 1e-8) & (h < 1.0 - 1e-8))[0]
        if len(eligible) > 0:
            h[eligible] += residual / len(eligible)
            h = np.clip(h, 0.0, 1.0)
    return h

def full_corr(h):
    """Compute full correlation array."""
    return np.correlate(h, 1.0 - h, mode="full")

def pair_exchange_c5(h, i, j, delta):
    """Compute C5 after exchanging delta from j to i."""
    n = len(h)
    dx = 2.0 / n
    h_new = h.copy()
    h_new[i] += delta
    h_new[j] -= delta
    if h_new[i] < -1e-12 or h_new[i] > 1+1e-12 or h_new[j] < -1e-12 or h_new[j] > 1+1e-12:
        return float('inf')
    h_new = np.clip(h_new, 0, 1)
    corr = full_corr(h_new)
    return float(np.max(corr) * dx)

def best_delta_for_pair(h, i, j, n_samples=20):
    """Find best delta for pair exchange (i gains, j loses)."""
    max_delta = min(1.0 - h[i], h[j])
    if max_delta < 1e-12:
        return 0.0, compute_c5(h)
    deltas = np.linspace(0, max_delta, n_samples)
    best_d = 0.0
    best_c = compute_c5(h)
    for d in deltas:
        c = pair_exchange_c5(h, i, j, d)
        if c < best_c:
            best_c = c
            best_d = d
    return best_d, best_c

def greedy_pair_descent(h, n_pairs, rng, time_limit=None, t0=None):
    """Greedy descent using pair exchanges."""
    n = len(h)
    best_h = h.copy()
    best_c = compute_c5(best_h)
    improved = True

        for _ in range(n_pairs):

            pairs.append((i, j))
        for i, j in pairs:

                return best_h, best_c
            # Try both directions

                    h_try[inc] += d

            cur_c = c_try

    m = float(np.max(corr))
    shifted = beta * (corr - m)
    exps = np.exp(shifted)
    s = float(np.sum(exps))
    return m + math.log(s) / beta

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max_val(corr, beta) * dx

def slsqp_smooth_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = float(np.max(corr))
    shifted = beta * (corr - m)
    exps = np.exp(shifted)
    weights = exps / float(np.sum(exps))
    grad = np.zeros(n)
    nn = 2 * n - 1
    for idx in range(nn):
        w = weights[idx]

        k = idx - (n - 1)

            if 0 <= i - k < n:
                d += (1.0 - h[i - k])
            if 0 <= i + k < n:
                d -= h[i + k]
            grad[i] += w * d
    return grad * dx
