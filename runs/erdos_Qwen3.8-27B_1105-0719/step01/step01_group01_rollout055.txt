    corr = np.correlate(h, 1.0 - h, mode="full")

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_feasible(y, n):
    target = n / 2.0
    y = np.asarray(y, dtype=float)
    lo, hi = float(np.min(y)) - 2.0, float(np.max(y)) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(y - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - h.sum()
    if abs(residual) > 1e-10:
        mask = (h > 1e-8) & (h < 1.0 - 1e-8)
        if mask.any():
            h[mask] += residual / float(mask.sum())
            h = np.clip(h, 0.0, 1.0)
    return h

def subgradient_direction(h, n, top_k=5):
    """Compute negative subgradient of max correlation at active lags"""
    corr = get_corr(h)
    top_idx = np.argsort(corr)[::-1][:top_k]
    grad = np.zeros(n)
    for idx in top_idx:
        d = idx - (n - 1)
        for i in range(n):
            g = 0.0
            if 0 <= i + d < n:
                g += (1.0 - h[i + d])
            if 0 <= i - d < n:
                g -= h[i - d]
            grad[i] += g
    # Normalize
    norm = np.linalg.norm(grad)
    if norm < 1e-12:
        return None
    return -grad / norm

def line_search(h, n, direction, alpha0=0.1, max_iter=30):
    """Backtracking line search along direction, maintaining feasibility"""
    best_h = h.copy()
    best_c = compute_c5(h)
    alpha = alpha0
    for _ in range(max_iter):
        new_h = project_feasible(h + alpha * direction, n)
        new_c = compute_c5(new_h)
        if new_c < best_c - 1e-14:
            best_h = new_h
            best_c = new_c
        alpha *= 0.5
        if alpha < 1e-6:
            break
    return best_h, best_c

def random_direction(n, rng):
    """Generate a random feasible direction"""
    d = rng.randn(n)
    d -= d.mean()  # make roughly sum-zero
    return d

To minimize this, we want to maximize autocorr_s(d) for all d. But the sum of autocorrelations over all d is 0 (for a balanced sequence), so we can't maximize all of them.

The key is to make the autocorrelation as "flat" as possible (close to 0 for all non-zero lags). This is the property of a "perfect" autocorrelation sequence, which is related to Barker sequences (which only exist for certain lengths) or m-sequences.

For length 60, there's no perfect autocorrelation sequence, but there are good constructions. This suggests that the optimal binary solution for n=60 should have relatively flat autocorrelation.

However, the continuous (non-binary) solution might do slightly better by allowing "soft" values that smooth out the correlation peaks.

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_feasible(y, n):
    target = n / 2.0
    y = np.asarray(y, dtype=float)
    if not np.all(np.isfinite(y)):
        y = np.nan_to_num(y, nan=0.5, posinf=1.0, neginf=0.0)
    lo, hi = float(np.min(y)) - 2.0, float(np.max(y)) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(y - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - float(np.sum(h))
    if abs(residual) > 1e-10:
        mask = (h > 1e-8) & (h < 1.0 - 1e-8)
        if mask.any():
            h[mask] += residual / float(np.sum(mask))
            h = np.clip(h, 0.0, 1.0)
    return h

def subgradient(h, n, top_k=5):
    corr = get_corr(h)
    top_idx = np.argsort(corr)[::-1][:top_k]
    grad = np.zeros(n)
    for idx in top_idx:
        d = idx - (n - 1)
        for i in range(n):
            g = 0.0
            j1 = i + d
            if 0 <= j1 < n:
                g += 1.0 - h[j1]
            j2 = i - d
            if 0 <= j2 < n:
                g -= h[j2]
            grad[i] += g
    return grad / float(top_k)

def line_search(h, n, direction, alpha0=0.05, max_ls=25):
    best_h = h.copy()
    best_c = compute_c5(h)
    alpha = alpha0
    for _ in range(max_ls):
        if alpha < 1e-7:
            break
        new_h = project_feasible(h + alpha * direction, n)
        new_c = compute_c5(new_h)
        if new_c < best_c - 1e-15:
            best_h = new_h
            best_c = new_c
        alpha *= 0.6
    return best_h, best_c

def pair_transfer_search(h, n, i, j):
    best_h = h.copy()
    best_c = compute_c5(h)
    max_tau = min(h[i], 1.0 - h[j])
    if max_tau > 1e-6:
        for frac in (0.15, 0.35, 0.55, 0.75, 0.95, 1.0):
            tau = frac * max_tau
            new_h = h.copy()
            new_h[i] -= tau
            new_h[j] += tau
            c = compute_c5(new_h)
            if c < best_c - 1e-15:
                best_h = new_h
                best_c = c
    max_tau_r = min(h[j], 1.0 - h[i])
    if max_tau_r > 1e-6:
        for frac in (0.15, 0.35, 0.55, 0.75, 0.95, 1.0):
            tau = frac * max_tau_r
            new_h = h.copy()
            new_h[j] -= tau
            new_h[i] += tau
            c = compute_c5(new_h)
            if c < best_c - 1e-15:
                best_h = new_h
                best_c = c
    return best_h, best_c

def swap_move(h, n, i, j):
    new_h = h.copy()
    new_h[i], new_h[j] = h[j], h[i]
    return new_h

            norm = np.linalg.norm(direction)
            if norm < 1e-12:
                direction = rng.randn(n)
                direction -= direction.mean()
                norm = np.linalg.norm(direction)
            if norm > 1e-12:
