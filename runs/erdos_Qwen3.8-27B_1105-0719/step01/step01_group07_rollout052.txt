        return None

        to_remove = idx_ones[:add_count]

    b[idx] = 1.0
    return b


def thue_morse(n):
    if n % 2 != 0:
        return None
    m = n // 2
    tm = np.array([bin(i).count('1') & 1 for i in range(n)], dtype=np.float64)
    num_ones = int(np.sum(tm))
    if num_ones != m:
        one_idx = list(np.where(tm == 1)[0])
        zero_idx = list(np.where(tm == 0)[0])
        if num_ones > m:
            for k in range(num_ones - m):
                tm[one_idx[k]] = 0.0
                tm[zero_idx[k]] = 1.0
        else:
            for k in range(m - num_ones):
                tm[zero_idx[k]] = 1.0
                tm[one_idx[k]] = 0.0
    return tm


def sa_binary(b, deadline, max_rounds=100000, t0=0.004, t_min=0.000005, alpha=0.99998, seed=0):
    rng = np.random.RandomState(seed)
    n = len(b)
    b = b.copy()
    curr_c = exact_objective(b)
    best_c = curr_c
    best_b = b.copy()
    ones_pos = list(np.where(b == 1.0)[0])
    zeros_pos = list(np.where(b == 0.0)[0])
    t = t0
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        if t < t_min:
            break
        pi = rng.randint(len(ones_pos))
        pj = rng.randint(len(zeros_pos))
        i = ones_pos[pi]
        j = zeros_pos[pj]
        b[i] = 0.0
        b[j] = 1.0
        new_c = exact_objective(b)
        delta = new_c - curr_c
        if delta <= 1e-15:
            curr_c = new_c
            ones_pos[pi] = j
            zeros_pos[pj] = i
            if new_c < best_c - 1e-14:
                best_c = new_c
                best_b = b.copy()
        elif t > 1e-15 and rng.random() < np.exp(-delta / t):
            curr_c = new_c
            ones_pos[pi] = j
            zeros_pos[pj] = i
            if new_c < best_c - 1e-14:
                best_c = new_c
                best_b = b.copy()
        else:
            b[i] = 1.0
            b[j] = 0.0
        t *= alpha
    return best_b, best_c


def sa_binary_2swap(b, deadline, max_rounds=50000, t0=0.003, t_min=0.000005, alpha=0.99995, seed=0):
    rng = np.random.RandomState(seed)
    n = len(b)
    b = b.copy()
    curr_c = exact_objective(b)
    best_c = curr_c
    best_b = b.copy()
    t = t0
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        if t < t_min:
            break
        r = rng.random()
        if r < 0.5:
            ones = list(np.where(b == 1.0)[0])
            zeros = list(np.where(b == 0.0)[0])
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            b[i] = 0.0
            b[j] = 1.0
            new_c = exact_objective(b)
            delta = new_c - curr_c
            if delta <= 1e-15 or (t > 1e-15 and rng.random() < np.exp(-delta / t)):
                curr_c = new_c
                if new_c < best_c - 1e-14:
                    best_c = new_c
                    best_b = b.copy()
            else:
                b[i] = 1.0
                b[j] = 0.0
        else:
            ones = list(np.where(b == 1.0)[0])
            zeros = list(np.where(b == 0.0)[0])
            if len(ones) < 2 or len(zeros) < 2:
                t *= alpha
                continue
            i1 = ones[rng.randint(len(ones))]
            i2 = ones[rng.randint(len(ones))]
            j1 = zeros[rng.randint(len(zeros))]
            j2 = zeros[rng.randint(len(zeros))]
            if i1 == i2 or j1 == j2:
                t *= alpha
                continue
            b[i1] = 0.0
            b[i2] = 0.0
            b[j1] = 1.0
            b[j2] = 1.0
            new_c = exact_objective(b)
            delta = new_c - curr_c
            if delta <= 1e-15 or (t > 1e-15 and rng.random() < np.exp(-delta / t)):
                curr_c = new_c
                if new_c < best_c - 1e-14:
                    best_c = new_c
                    best_b = b.copy()
            else:
                b[i1] = 1.0
                b[i2] = 1.0
                b[j1] = 0.0
                b[j2] = 0.0
        t *= alpha
    return best_b, best_c


def greedy_swap(b, deadline, max_rounds=200):
    n = len(b)
    b = b.copy()
    curr_c = exact_objective(b)
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        best_improve = 1e-14
        best_move = None
        ones = np.where(b == 1.0)[0]
        zeros = np.where(b == 0.0)[0]
        for i in ones:
            for j in zeros:
                b[i] = 0.0
                b[j] = 1.0
                c = exact_objective(b)
                b[i] = 1.0
                b[j] = 0.0
                imp = curr_c - c
                if imp > best_improve:
                    best_improve = imp
                    best_move = (i, j)
        if best_move is not None:
            i, j = best_move
            b[i] = 0.0
            b[j] = 1.0
            curr_c = exact_objective(b)
        else:
            break
    return b, curr_c


def lse_obj_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    m = float(np.max(profile))
    shifted = np.minimum((profile - m) / tau, 0.0)
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    weights = exp_vals / sum_exp
    obj = m + tau * np.log(sum_exp)
    grad = np.zeros(n)
    for j in range(2 * n - 1):
        w = weights[j]
        if w < 1e-13:
            continue
        s = j - (n - 1)
        if s >= 0:
            k1 = np.arange(s, n)
            grad[k1] += w * dx * (1.0 - h[k1 - s])
            k2 = np.arange(0, n - s)
            grad[k2] -= w * dx * h[k2 + s]
        else:
            a = -s
            k1 = np.arange(0, n - a)
            grad[k1] += w * dx * (1.0 - h[k1 + a])
            k2 = np.arange(a, n)
            grad[k2] -= w * dx * h[k2 - a]
    return obj, grad


class LSEFunc:
    def __init__(self, tau):
        self.tau = tau

    def value(self, x):
        return lse_obj_grad(x, self.tau)[0]

    def gradient(self, x):
        return lse_obj_grad(x, self.tau)[1]


class SumConstraint:
    def __init__(self, target):
        self.target = target

    def fun(self, x):
        return np.sum(x) - self.target

    def jac(self, x):
        return np.ones(len(x))


def slsqp_optimize(h0, tau, maxiter=200):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    obj = LSEFunc(tau)
    con = SumConstraint(n / 2.0)
    constraints = [{'type': 'eq', 'fun': con.fun, 'jac': con.jac}]
    try:
        res = minimize(obj.value, h0.copy(), method='SLSQP', jac=obj.gradient,
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        h_new = project_box_sum(np.array(res.x, dtype=np.float64))
        return h_new
    except Exception:
        return h0.copy()


def transfer_to_size(h_parent, n_target):
    n_src = len(h_parent)
    if n_src == n_target:
        return project_box_sum(h_parent.copy())
    x_src = (np.arange(n_src) + 0.5) / n_src
    x_tgt = (np.arange(n_target) + 0.5) / n_target
    h_new = np.interp(x_tgt, x_src, h_parent)
    return project_box_sum(h_new)


def make_random_binary(n, rng):
    b = np.zeros(n)
    idx = rng.choice(n, n // 2, replace=False)
    b[idx] = 1.0
    return b


def make_block_binary(n, block_len, rng):
    b = np.zeros(n)
    period = 2 * block_len
    for i in range(0, n, period):
        end = min(i + block_len, n)
        b[i:end] = 1.0
    # Adjust if needed
    num_ones = int(np.sum(b))
    m = n // 2
    if num_ones != m:
        ones_idx = list(np.where(b == 1.0)[0])
        zeros_idx = list(np.where(b == 0.0)[0])
        if num_ones > m:
            for k in range(num_ones - m):
                b[ones_idx[k]] = 0.0
                b[zeros_idx[k]] = 1.0
        else:
            for k in range(m - num_ones):
                b[zeros_idx[k]] = 1.0
                b[ones_idx[k]] = 0.0
    return b


def refine_continuous(h0, deadline):
    h = project_box_sum(h0.copy())
    n = len(h)
    curr_c = exact_objective(h)
    taus = [0.008, 0.003, 0.001, 0.0004, 0.00015, 0.00005]
    for tau in taus:
        if time.time() > deadline - 3:
            break
        h_new = slsqp_optimize(h, tau, maxiter=min(300, n * 4))
        c_new = exact_objective(h_new)
        if c_new < curr_c - 1e-13:
            h = h_new
            curr_c = c_new
    return h, curr_c


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    deadline = t_start + budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c = float('inf')

    # Load parent
    parent_h = None
    try:
        pv = globals()['initial_h_values']
        parent_h = np.array(pv, dtype=np.float64).copy()
        if len(parent_h) > 0:
            parent_h = project_box_sum(parent_h)
            pc = exact_objective(parent_h)
            best_c = pc
            best_h = parent_h.copy()
    except Exception:
        pass

    parent_n = len(parent_h) if parent_h is not None else 48

    # Sizes to explore (even, around parent)
    sizes = []
    for d in [0, -2, 2, -4, 4, -6, 6, 8, -8, 10, 12, 16]:
        s = parent_n + d
        if s % 2 == 0 and 20 <= s <= 80 and s not in sizes:
            sizes.append(s)
    sizes.sort()

    # Time allocation
    reserve = 50
    time_per_size = max(20, (budget_s - reserve) / max(1, len(sizes)))

    all_best = {}

    for n in sizes:
        if time.time() > deadline - reserve:
            break
        size_deadline = min(deadline - reserve, t_start + time_per_size * (sizes.index(n) + 1))
        if time.time() > size_deadline:
            continue

        # Generate diverse binary candidates
        candidates = []

        # Thue-Morse
        tm = thue_morse(n)
        if tm is not None:
            candidates.append(tm)
            for shift in [1, 2, 3, 4]:
                candidates.append(np.roll(tm, shift))

        # Parent-based
        if parent_h is not None:
            h_int = transfer_to_size(parent_h, n)
            b_parent = binary_threshold(h_int)
            if b_parent is not None:
                candidates.append(b_parent)
            # Perturbed parent
            for scale in [0.05, 0.1, 0.2]:
                h_pert = project_box_sum(h_int + rng.normal(0, scale, n))
                b_pert = binary_threshold(h_pert)
                if b_pert is not None:
                    candidates.append(b_pert)

        # Random
        for _ in range(4):
            candidates.append(make_random_binary(n, rng))

        # Block patterns
        for bl in [2, 3, 4, 5, 6, 7, 8]:
            candidates.append(make_block_binary(n, bl, rng))

        # Alternating
        b_alt = np.zeros(n)
        b_alt[::2] = 1.0
        if int(np.sum(b_alt)) != n // 2:
            b_alt = np.roll(b_alt, 1)
        candidates.append(b_alt.copy())

        # Score and sort
        scored = []
        seen_patterns = set()
        for c in candidates:
            c_c = exact_objective(c)
            key = tuple((c > 0.5).astype(int).tolist())
            if key not in seen_patterns:
                seen_patterns.add(key)
                scored.append((c_c, c))
        scored.sort(key=lambda x: x[0])

        # SA on top candidates with multiple restarts
        n_sa = min(5, len(scored))
        sa_budget = max(5, min(40, size_deadline - time.time() - 15))

        for idx in range(n_sa):
            remaining = size_deadline - time.time()
            if remaining < 3:
                break
            b_init = scored[idx][1].copy()
            # Multiple SA restarts
            n_restarts = max(1, min(4, int(remaining / 8)))
            for rst in range(n_restarts):
                rst_deadline = time.time() + (remaining / n_restarts) * 0.9
                if time.time() > rst_deadline:
                    break
                b_sa, c_sa = sa_binary(b_init, rst_deadline, max_rounds=80000,
                                       t0=0.005, t_min=0.000005, alpha=0.99998,
                                       seed=seed + idx * 100 + rst)
                if c_sa < best_c - 1e-12:
                    best_c = c_sa
                    best_h = b_sa.copy()
                if c_sa < best_c:
                    b_init = b_sa.copy()

        # 2-swap SA on best so far for this size
        if time.time() < size_deadline - 5 and best_h is not None:
            n_best = len(best_h)
            if n_best == n:
                b_2s, c_2s = sa_binary_2swap(best_h, min(size_deadline - 2, time.time() + 15),
                                             max_rounds=30000, t0=0.003, t_min=0.00001,
                                             alpha=0.99995, seed=seed + 500)
                if c_2s < best_c - 1e-12:
                    best_c = c_2s
                    best_h = b_2s.copy()

        # Greedy polish
        if time.time() < size_deadline - 3 and best_h is not None:
            if len(best_h) == n:
                b_gr, c_gr = greedy_swap(best_h, size_deadline - 1, max_rounds=100)
                if c_gr < best_c - 1e-12:
                    best_c = c_gr
                    best_h = b_gr.copy()

        # LSE continuous refinement (secondary)
        if time.time() < size_deadline - 8 and parent_h is not None:
            h_cont = transfer_to_size(parent_h, n)
            if int(np.sum(h_cont >= 0.5)) == n // 2 or True:
                h_lse, c_lse = refine_continuous(h_cont, size_deadline - 2)
                if c_lse < best_c - 1e-12:
                    best_c = c_lse
                    best_h = h_lse.copy()
                # Threshold and SA
                b_thr = binary_threshold(h_lse)
                if b_thr is not None:
                    b_sa2, c_sa2 = sa_binary(b_thr, min(size_deadline - 1, time.time() + 10),
                                             max_rounds=40000, t0=0.004,
                                             t_min=0.00001, alpha=0.99998, seed=seed + 900)
                    if c_sa2 < best_c - 1e-12:
                        best_c = c_sa2
                        best_h = b_sa2.copy()
                        b_gr2, c_gr2 = greedy_swap(b_sa2, size_deadline - 1, max_rounds=80)
                        if c_gr2 < best_c - 1e-12:
                            best_c = c_gr2
                            best_h = b_gr2.copy()

    # Final intensive polish on best solution
    if best_h is not None and time.time() < deadline - 20:
        remaining = deadline - time.time()
        if np.all((best_h <= 1e-8) | (best_h >= 1.0 - 1e-8)):
            # Already binary - do extended SA + greedy
            b_ext, c_ext = sa_binary(best_h, time.time() + remaining * 0.4,
                                     max_rounds=200000, t0=0.003, t_min=0.000003,
                                     alpha=0.99999, seed=seed + 2000)
            if c_ext < best_c - 1e-12:
                best_c = c_ext
                best_h = b_ext.copy()
            if time.time() < deadline - 5:
                b_2s, c_2s = sa_binary_2swap(best_h, time.time() + remaining * 0.3,
                                             max_rounds=50000, t0=0.002,
                                             t_min=0.00001, alpha=0.99997,
                                             seed=seed + 3000)
                if c_2s < best_c - 1e-12:
                    best_c = c_2s
                    best_h = b_2s.copy()
            if time.time() < deadline - 3:
                b_gr, c_gr = greedy_swap(best_h, deadline - 1, max_rounds=200)
                if c_gr < best_c - 1e-12:
                    best_c = c_gr
                    best_h = b_gr.copy()
        else:
            # Continuous - try LSE then binary
            h_ref, c_ref = refine_continuous(best_h, deadline - 10)
            if c_ref < best_c - 1e-12:
                best_c = c_ref
                best_h = h_ref.copy()
            b_thr = binary_threshold(best_h)
            if b_thr is not None and time.time() < deadline - 5:
                b_sa, c_sa = sa_binary(b_thr, deadline - 1, max_rounds=100000,
                                       t0=0.003, t_min=0.000005, alpha=0.99998,
                                       seed=seed + 4000)
                if c_sa < best_c - 1e-12:
                    best_c = c_sa
                    best_h = b_sa.copy()
                b_gr, c_gr = greedy_swap(b_sa, deadline - 1, max_rounds=100)
                if c_gr < best_c - 1e-12:
                    best_c = c_gr
                    best_h = b_gr.copy()

    # Final validation
    if best_h is None:
        # Fallback
        n_fb = 48
        best_h = thue_morse(n_fb)
        if best_h is None:
            best_h = make_random_binary(n_fb, rng)
    best_h = project_box_sum(best_h)
    best_c = exact_objective(best_h)
    return best_h, best_c, len(best_h)
