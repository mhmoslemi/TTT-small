    """Gradient of c[k] with respect to h, where k is the shift (can be negative)."""
    n = len(h)

        grad[:] = 1.0 - 2.0 * h

        # Case 1: j = n+k, j in [k, n-1], contribution (1-h[j-k])
        grad[k:] += (1.0 - h[:-k])
