import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

def project_feasible(x, target_sum):
    """Project x onto {0<=h<=1, sum(h)=target_sum}"""
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        h = np.clip(x + mid, 0.0, 1.0)
        if h.sum() > target_sum:
            hi = mid
        else:
            lo = mid
    h = np.clip(x + (lo + hi) / 2, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def eval_exact(h):
    """Compute the exact C5 score."""
    dx = 2.0 / len(h)
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def build_sym_h(q, n):
    """Build symmetric h from q variables. n must be even, len(q)=n/2."""
    m = len(q)
    h = np.empty(2 * m)
    h[:m] = q
    h[m:] = 1.0 - q[::-1]
    return h

def q_from_h(h, n):
    """Extract q from symmetric h. n must be even."""
    m = n // 2
    q = 0.5 * (h[:m] + 1.0 - h[m:][::-1])
    return q

def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def grad_corr_p(h, p, n):
    """Gradient of correlate(h, 1-h, 'full')[p] with respect to h."""
    d = p - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -d)
    i_hi = min(n - 1, n - 1 - d)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    # For each valid i: h[i]*(1-h[i+d])
    # d/dh[i] += (1 - h[i+d])
    # d/dh[i+d] += -h[i]
    j1 = idx  # i
    j2 = idx + d  # i + d
    grad[j1] += (1.0 - h[j2])
    grad[j2] += (-h[j1])
    return grad

def smooth_max_value(h, beta):
    corrs = compute_corrs(h)
    return logsumexp(beta * corrs) / beta

def smooth_max_grad(h, beta, dx):
    corrs = compute_corrs(h)
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(len(corrs)) / len(corrs)
    else:
        w /= w_sum
    grad = np.zeros(len(h))
    n = len(h)
    for p in range(len(corrs)):
        if w[p] < 1e-14:
            continue
        grad += w[p] * grad_corr_p(h, p, n)
    return grad * dx

def smooth_sym_value(q, n, beta):
    h = build_sym_h(q, n)
    return smooth_max_value(h, beta)

def smooth_sym_grad(q, n, beta):
    h = build_sym_h(q, n)
    dx = 2.0 / n
    g_h = smooth_max_grad(h, beta, dx)
    m = n // 2
    g_q = g_h[:m] - g_h[m:][::-1]
    return g_q

def solve_smooth_sym(q0, n, beta_list, maxiter_per_beta, time_check=None):
    """Solve smooth max problem in symmetric subspace."""

    Solve: minimize t

               0 <= q[i] <= 1

        return x[-1]  # t is the last variable

        g[-1] = 1.0
        return g

            q = x[:m]
            h = build_sym_h(q, n)

            q = x[:m]
            h = build_sym_h(q, n)
            g_h = grad_corr_p(h, p, n)
            g_q = g_h[:m] - g_h[m:][::-1]
            j = np.zeros(m + 1)
            j[:m] = -dx * g_q
            j[-1] = 1.0

def smooth_val_full(h, n, beta):
    c = corr_array(h)
    return logsumexp(beta * c) / beta


def smooth_grad_full(h, n, beta):
    dx = 2.0 / n
    c = corr_array(h)
    s = c - c.max()
    w = np.exp(beta * s)
    ws = w.sum()
    if ws < 1e-30:
        w = np.ones_like(c) / len(c)
    else:
        w /= ws
    g = np.zeros(n)
    for p in range(len(c)):
        if w[p] > 1e-14:
            g += w[p] * grad_one_corr(h, p, n)
    return g * dx


def sum_constraint_full(h, n, beta):
    return h.sum() - n / 2.0


def sum_jac_full(h, n, beta):
    return np.ones(n)


def select_active(h, n, top_k=25, margin_frac=0.03, neighbor_pad=3):
    c = corr_array(h)
    mx = c.max()
    margin = max(1e-10, margin_frac * mx)
    idx = np.where(c >= mx - margin)[0]
    s = set()
    for i in idx:
        for dp in range(-neighbor_pad, neighbor_pad + 1):
            if 0 <= i + dp < len(c):
                s.add(i + dp)
    top_idx = np.argsort(c)[-top_k:]
    s.update(top_idx.tolist())
    active = sorted(s)
    cap = 80
    if len(active) > cap:
        vals = c[active]
        order = np.argsort(-vals)
        active = [active[i] for i in order[:cap]]
    return active


def epi_sym_obj(x, n, active, dx):
    return x[-1]


def epi_sym_obj_grad(x, n, active, dx):
    m = n // 2
    g = np.zeros(m + 1)
    g[-1] = 1.0
    return g


def epi_sym_cfun(x, n, active, dx):
    m = n // 2
    q = x[:m]
    h = build_sym_h(q)
    c = corr_array(h)
    return x[-1] - dx * c[active]


def epi_sym_cjac(x, n, active, dx):
    m = n // 2
    q = x[:m]
    h = build_sym_h(q)
    na = len(active)
    j = np.zeros((na, m + 1))
    for k in range(na):
        p = active[k]
        gh = grad_one_corr(h, p, n)
        gq = gh[:m] - gq_tmp(gh, m)
        j[k, :m] = -dx * gq
        j[k, m] = 1.0
    return j


def gq_tmp(gh, m):
    return gh[m:][::-1]


def epi_full_obj(x, n, active, dx):
    return x[-1]


def epi_full_obj_grad(x, n, active, dx):
    g = np.zeros(n + 1)
    g[-1] = 1.0
    return g


def epi_full_cfun(x, n, active, dx):
    h = x[:n]
    c = corr_array(h)
    return x[-1] - dx * c[active]


def epi_full_cjac(x, n, active, dx):
    h = x[:n]
    na = len(active)
    j = np.zeros((na, n + 1))
    for k in range(na):
        p = active[k]
        gh = grad_one_corr(h, p, n)
        j[k, :n] = -dx * gh
        j[k, n] = 1.0
    return j


def epi_full_eq_fun(x, n, active, dx):
    return x[:n].sum() - n / 2.0


def epi_full_eq_jac(x, n, active, dx):
    j = np.zeros(n + 1)
    j[:n] = 1.0 / n
    return j


def do_smooth_sym(q0, n, betas, maxiter, deadline):
    q = q0.copy()
    best_q = q0.copy()
    best_s = eval_score(build_sym_h(q0))
    for beta in betas:
        if time.time() >= deadline:
            break
        res = minimize(smooth_val_q, q, jac=smooth_grad_q,
                       args=(n, beta), method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * (n // 2),
                       options={'maxiter': maxiter, 'gtol': 1e-9})
        if res.x is not None:
            q_new = np.clip(res.x, 0.0, 1.0)
            s = eval_score(build_sym_h(q_new))
            if s < best_s:
                best_s = s
                best_q = q_new.copy()
                q = q_new.copy()
    return best_q, best_s


def do_epigraph_sym(q0, n, active, maxiter=50, ftol=1e-10):
    m = n // 2
    dx = 2.0 / n
    h0 = build_sym_h(q0)
    t_init = eval_score(h0) + 1e-7
    x0 = np.concatenate([q0, [t_init]])
    bounds = [(0.0, 1.0)] * m + [(0.0, t_init + 0.5)]
    cons = [{'type': 'ineq', 'fun': epi_sym_cfun, 'jac': epi_sym_cjac,
             'args': (n, active, dx)}]
    try:
        res = minimize(epi_sym_obj, x0, jac=epi_sym_obj_grad,
                       args=(n, active, dx), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        if res.x is not None:
            q_out = np.clip(res.x[:m], 0.0, 1.0)
            h_out = build_sym_h(q_out)
            return q_out, eval_score(h_out)
    except Exception:
        pass
    return q0, eval_score(h0)


def do_epigraph_full(h0, n, active, maxiter=50, ftol=1e-10):
    dx = 2.0 / n
    t_init = eval_score(h0) + 1e-7
    x0 = np.concatenate([h0, [t_init]])
    bounds = [(0.0, 1.0)] * n + [(0.0, t_init + 0.5)]
    cons = [
        {'type': 'ineq', 'fun': epi_full_cfun, 'jac': epi_full_cjac,
         'args': (n, active, dx)},
        {'type': 'eq', 'fun': epi_full_eq_fun, 'jac': epi_full_eq_jac,
         'args': (n, active, dx)}
    ]
    try:
        res = minimize(epi_full_obj, x0, jac=epi_full_obj_grad,
                       args=(n, active, dx), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        if res.x is not None:
            h_out = project_feasible(res.x[:n], n / 2.0)
            return h_out, eval_score(h_out)
    except Exception:
        pass
    return h0, eval_score(h0)


def epigraph_expand(q0, n, max_rounds=4, deadline=0, top_k_start=25):
    q = q0.copy()
    best_q = q0.copy()
    best_s = eval_score(build_sym_h(q0))
    for rnd in range(max_rounds):
        if deadline > 0 and time.time() >= deadline:
            break
        h = build_sym_h(q)
        tk = top_k_start + 10 * rnd
        mg = 0.02 + 0.015 * rnd
        npad = 2 + rnd
        active = select_active(h, n, top_k=tk, margin_frac=mg, neighbor_pad=npad)
        mi = max(30, 60 - 10 * rnd)
        q_new, s_new = do_epigraph_sym(q, n, active, maxiter=mi, ftol=1e-10)
        if s_new < best_s - 1e-12:
            best_s = s_new
            best_q = q_new.copy()
            q = q_new.copy()
        else:
            h_chk = build_sym_h(q_new)
            c = corr_array(h_chk)
            mx = c.max()
            hidden = np.where(c * (2.0 / n) > best_s + 5e-9)[0]
            if len(hidden) == 0:
                break
            active2 = list(set(active + hidden.tolist()))
            if len(active2) > 100:
                active2 = active2[:100]
            q_new2, s_new2 = do_epigraph_sym(q, n, active2, maxiter=mi, ftol=1e-10)
            if s_new2 < best_s - 1e-12:
                best_s = s_new2
                best_q = q_new2.copy()
                q = q_new2.copy()
            else:
                break
    return best_q, best_s


def binary_local_search(q0, n, max_flips=2000, deadline=0, rng=None):
    m = n // 2
    q = q0.copy()
    best_q = q0.copy()
    best_s = eval_score(build_sym_h(q0))
    if rng is None:
        rng = np.random.RandomState(0)
    for _ in range(max_flips):
        if deadline > 0 and time.time() >= deadline:
            break
        j = rng.randint(m)
        q_flip = q.copy()
        q_flip[j] = 1.0 - q_flip[j]
        h_flip = build_sym_h(q_flip)
        s_flip = eval_score(h_flip)
        if s_flip < best_s - 1e-14:
            q = q_flip
            best_q = q_flip.copy()
            best_s = s_flip
    return best_q, best_s


def soft_binary_search(q0, n, max_iter=500, deadline=0, rng=None):
    m = n // 2
    q = q0.copy()
    best_q = q0.copy()
    best_s = eval_score(build_sym_h(q0))
    alpha = 0.1
    if rng is None:
        rng = np.random.RandomState(1)
    for it in range(max_iter):
        if deadline > 0 and time.time() >= deadline:
            break
        j = rng.randint(m)
        direction = 1.0 if q[j] < 0.5 else -1.0
        q_trial = q.copy()
        q_trial[j] = np.clip(q[j] + direction * alpha, 0.0, 1.0)
        h_trial = build_sym_h(q_trial)
        s_trial = eval_score(h_trial)
        if s_trial < best_s - 1e-15:
            q = q_trial
            best_q = q_trial.copy()
            best_s = s_trial
            alpha = min(alpha * 1.05, 0.4)
        else:
            alpha *= 0.9
        if alpha < 1e-6:
            break
    return best_q, best_s


def interpolate_h(h_old, n_new, target_sum):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    x_old = np.linspace(0.0, 2.0, n_old)
    x_new = np.linspace(0.0, 2.0, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, target_sum)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline_total = t0 + budget_s

    best_h = None
    best_c5 = float('inf')

    # Initialize from parent
    try:
        h_par = np.array(initial_h_values, dtype=float)
    except NameError:
        h_par = None

    if h_par is not None and len(h_par) > 0:
        n_par = len(h_par)
        h_par = project_feasible(h_par, n_par / 2.0)
        best_h = h_par.copy()
        best_c5 = eval_score(h_par)
    else:
        n_par = 126
        x = np.linspace(0, 2, n_par)
        h_par = np.clip(0.5 + 0.4 * np.sin(3 * np.pi * x) + 0.2 * np.cos(5 * np.pi * x), 0, 1)
        h_par = project_feasible(h_par, n_par / 2.0)
        best_h = h_par.copy()
        best_c5 = eval_score(h_par)

    def time_left():
        return deadline_total - time.time()

    # Ensure even n for symmetric work
    n = len(best_h)
    if n % 2 != 0:
        n += 1
        best_h = interpolate_h(best_h, n, n / 2.0)
        best_c5 = eval_score(best_h)

    m = n // 2
    q_center = extract_q(best_h)

    # Phase 1: Smooth optimization in symmetric subspace
    if time_left() > 80:
        betas = [50, 100, 200, 400, 800, 1200]
        if time_left() < 400:
            betas = betas[:4]
        dq, s = do_smooth_sym(q_center, n, betas, 200, deadline_total - 30)
        if s < best_c5:
            best_c5 = s
            best_h = build_sym_h(dq)
            q_center = dq

    # Phase 2: Epigraph with expansion in symmetric subspace
    if time_left() > 60:
        dq, s = epigraph_expand(q_center, n, max_rounds=4,
                                deadline=deadline_total - 40, top_k_start=25)
        if s < best_c5:
            best_c5 = s
            best_h = build_sym_h(dq)
            q_center = dq

    # Phase 2b: Another round of epigraph
    if time_left() > 40:
        dq, s = epigraph_expand(q_center, n, max_rounds=3,
                                deadline=deadline_total - 30, top_k_start=35)
        if s < best_c5:
            best_c5 = s
            best_h = build_sym_h(dq)
            q_center = dq

    # Phase 3: Full-variable epigraph (escape symmetric subspace)
    if time_left() > 50:
        active = select_active(best_h, n, top_k=30, margin_frac=0.03, neighbor_pad=3)
        h_new, s_new = do_epigraph_full(best_h, n, active, maxiter=80, ftol=1e-10)
        if s_new < best_c5:
            best_c5 = s_new
            best_h = h_new
        # Second full epigraph with more constraints
        if time_left() > 30 and s_new < best_c5 + 1e-6:
            active2 = select_active(best_h, n, top_k=50, margin_frac=0.05, neighbor_pad=4)
            h_new2, s_new2 = do_epigraph_full(best_h, n, active2, maxiter=60, ftol=1e-10)
            if s_new2 < best_c5:
                best_c5 = s_new2
                best_h = h_new2

    # Phase 4: Binary local search in symmetric space
    if time_left() > 40:
        # Snap to nearest binary
        q_bin = (q_center > 0.5).astype(float)
        # Fix sum: ensure exactly m ones (which is automatic for symmetric binary)
        h_bin = build_sym_h(q_bin)
        s_bin = eval_score(h_bin)
        q_start = q_bin.copy()
        if s_bin > best_c5:
            q_start = q_center
        dq, s = binary_local_search(q_start, n, max_flips=5000,
                                    deadline=deadline_total - 20, rng=rng)
        if s < best_c5:
            best_c5 = s
            best_h = build_sym_h(dq)
            q_center = dq

    # Phase 5: Soft binary refinement
    if time_left() > 30:
        dq, s = soft_binary_search(q_center, n, max_iter=2000,
                                   deadline=deadline_total - 15, rng=rng)
        if s < best_c5:
            best_c5 = s
            best_h = build_sym_h(dq)
            q_center = dq

    # Phase 6: Try different resolutions
    if time_left() > 80:
        n_options = []
        for nn in [64, 80, 96, 100, 128, 130, 132]:
            if nn != n and nn >= 32:
                n_options.append(nn)
        # Sort by distance from current n (try close ones first)
        n_options.sort(key=lambda x: abs(x - n))
        for n_try in n_options[:5]:
            if time_left() < 30:
                break
            h_interp = interpolate_h(best_h, n_try, n_try / 2.0)
            s_interp = eval_score(h_interp)
            if n_try % 2 == 0:
                mq = n_try // 2
                q_try = extract_q(h_interp)
                # Quick smooth
                if time_left() > 25:
                    dq, s_sym = do_smooth_sym(q_try, n_try, [150, 400], 100,
                                              deadline_total - 15)
                    h_try = build_sym_h(dq)
                    s_try = eval_score(h_try)
                    # Epigraph
                    if time_left() > 15:
                        dq2, s_ep = epigraph_expand(dq, n_try, max_rounds=2,
                                                    deadline=deadline_total - 10,
                                                    top_k_start=20)
                        if s_ep < s_try:
                            s_try = s_ep
                            h_try = build_sym_h(dq2)
                    if s_try < best_c5:
                        best_c5 = s_try
                        best_h = h_try
                        q_center = extract_q(best_h)

    # Phase 7: Random restarts
    if time_left() > 60:
        n_cur = len(best_h)
        if n_cur % 2 != 0:
            n_cur += 1
            best_h = interpolate_h(best_h, n_cur, n_cur / 2.0)
            best_c5 = eval_score(best_h)
        m_cur = n_cur // 2
        q_cent = extract_q(best_h)

        n_trials = min(12, max(3, int(time_left() // 25)))
        for trial in range(n_trials):
            if time_left() < 20:
                break
            sigma = 0.03 + 0.04 * trial
            q_pert = np.clip(q_cent + sigma * rng.randn(m_cur), 0.0, 1.0)
            if time_left() > 25:
                dq, s = do_smooth_sym(q_pert, n_cur, [200, 500], 80,
                                      deadline_total - 10)
                h_try = build_sym_h(dq)
                s_try = eval_score(h_try)
                if time_left() > 12:
                    dq2, s_ep = epigraph_expand(dq, n_cur, max_rounds=2,
                                                deadline=deadline_total - 8,
                                                top_k_start=20)
                    if s_ep < s_try:
                        s_try = s_ep
                        h_try = build_sym_h(dq2)
                if s_try < best_c5:
                    best_c5 = s_try
                    best_h = h_try
                    q_center = extract_q(best_h)

    # Phase 8: Final polish
    if time_left() > 15 and best_h is not None:
        n_fin = len(best_h)
        if n_fin % 2 == 0:
            q_fin = extract_q(best_h)
            active = select_active(best_h, n_fin, top_k=40, margin_frac=0.01,
                                   neighbor_pad=4)
            dq, s = do_epigraph_sym(q_fin, n_fin, active, maxiter=100, ftol=1e-12)
            if s < best_c5:
                best_c5 = s
                best_h = build_sym_h(dq)

    # Final validation
    if best_h is None or not np.all(np.isfinite(best_h)):
        best_h = h_par if h_par is not None else np.full(126, 0.5)
        best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_c5 = eval_score(best_h)

    return best_h, best_c5, len(best_h)
