def project_onto_feasible(h, target=None):

    n = len(h)
    if target is None:

    h = np.array(h, dtype=np.float64)

        diff = s - target
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h > 0
            if not np.any(mask):
                break
            step = min(diff / np.sum(mask), np.min(h[mask]) * 0.999)
            h[mask] -= step
        else:
            mask = h < 1.0
            if not np.any(mask):
                break
            step = min(-diff / np.sum(mask), (1.0 - np.max(h[mask])) * 0.999)
            h[mask] += step
    # Final correction
    diff = np.sum(h) - target
    if abs(diff) > 1e-14:

        if np.any(mask):
