    F_min = np.min(F)
    shifted = F - F_min
    exp_vals = np.exp(-beta_lag * shifted)
    Z = np.sum(exp_vals)
    weights = exp_vals / Z
    phi = F_min - (1.0/beta_lag) * np.log(Z)
    # Objective J = -phi
    obj = -phi
    # Gradient of J = -sum w_m * G[m,:]
    grad = -np.sum(weights[:, None] * G, axis=0)
    return obj, grad, F, G

        return obj

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr * dx))


def repair_feasible(vec):
    h = np.asarray(vec, dtype=np.float64)
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-12:
        return h.copy()
    if diff > 0:
        order = np.argsort(h)
        rem = diff
        for i in order:
            if rem <= 1e-15:
                break
            add = min(rem, 1.0 - h[i])
            h[i] += add
            rem -= add
    else:
        order = np.argsort(-h)
        rem = -diff
        for i in order:
            if rem <= 1e-15:
                break
            remv = min(rem, h[i])
            h[i] -= remv
            rem -= remv
    diff2 = target - h.sum()
    if abs(diff2) > 1e-14:
        for i in range(n):
            if diff2 > 0 and h[i] < 1.0:
                h[i] += diff2
                break
            elif diff2 < 0 and h[i] > 0.0:
                h[i] += diff2
                break
    return h

def smooth_obj_jac(h, beta_lag, beta_e):
    F, G = compute_F_and_grads(h, beta_e)
    F_min = np.min(F)
    shifted = F - F_min
    exp_vals = np.exp(-beta_lag * shifted)
    Z = np.sum(exp_vals)
    weights = exp_vals / Z
    phi = F_min - (1.0 / beta_lag) * np.log(Z)
    obj = -phi
    grad = -np.sum(weights[:, None] * G, axis=0)
    return obj, grad


def solve_smooth(h_start, beta_lag, beta_e, maxiter=200, tol=1e-10):
    n = len(h_start)
    h0 = repair_feasible(h_start)
    
    def obj_func(x):
        h = np.clip(x, 0.0, 1.0)
        o, _ = smooth_obj_jac(h, beta_lag, beta_e)
        return o
    
    def jac_func(x):
        h = np.clip(x, 0.0, 1.0)
        _, g = smooth_obj_jac(h, beta_lag, beta_e)
        return g
    
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,

import numpy as np
from scipy.optimize import minimize
import time


def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr * dx))


def repair_feasible(vec):
    h = np.asarray(vec, dtype=np.float64)
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-12:
        return h.copy()
    if diff > 0:
        order = np.argsort(h)
        rem = diff
        for i in order:
            if rem <= 1e-15:
                break
            add = min(rem, 1.0 - h[i])
            h[i] += add
            rem -= add
    else:
        order = np.argsort(-h)
        rem = -diff
        for i in order:
            if rem <= 1e-15:
                break
            remv = min(rem, h[i])
            h[i] -= remv
            rem -= remv
    diff2 = target - h.sum()
    if abs(diff2) > 1e-14:
        for i in range(n):
            if diff2 > 0 and h[i] < 1.0:
                h[i] += diff2
                break
            elif diff2 < 0 and h[i] > 0.0:
                h[i] += diff2
                break
    return h


def compute_F_and_grads(h, beta_e):
    n = len(h)
    F = np.zeros(n)
    G = np.zeros((n, n))
    for m in range(n):
        if m == 0:
            A_m = np.dot(h, h)
            F[0] = A_m
            G[0, :] = 2.0 * h
        else:
            A_m = np.dot(h[:n - m], h[m:])
            F[m] = A_m
            G[m, :n - m] += h[m:]
            G[m, m:] += h[:n - m]
            P_m = h[:m].sum()
            S_m = h[n - m:].sum()
            m0 = min(P_m, S_m)
            d = abs(P_m - S_m)
            if d < 1e-12:
                soft_min_val = m0
                w_u = 0.5
                w_v = 0.5
            else:
                bd = beta_e * d
                if bd < 500:
                    soft_min_val = m0 - (1.0 / beta_e) * np.log(1.0 + np.exp(-bd))
                else:
                    soft_min_val = m0
                diff_ps = beta_e * (P_m - S_m)
                if diff_ps > 500:
                    w_u = 0.0
                elif diff_ps < -500:
                    w_u = 1.0
                else:
                    w_u = 1.0 / (1.0 + np.exp(diff_ps))
                w_v = 1.0 - w_u
            F[m] += soft_min_val
            G[m, :m] += w_u
            G[m, n - m:] += w_v
    return F, G


def smooth_obj_jac(h, beta_lag, beta_e):
    F, G = compute_F_and_grads(h, beta_e)
    F_min = np.min(F)
    shifted = F - F_min
    exp_vals = np.exp(-beta_lag * shifted)
    Z = np.sum(exp_vals)
    weights = exp_vals / Z
    phi = F_min - (1.0 / beta_lag) * np.log(Z)
    obj = -phi
    grad = -np.sum(weights[:, None] * G, axis=0)
    return obj, grad


def _solve_obj(x, bl, be):
    h = np.clip(x, 0.0, 1.0)
    o, _ = smooth_obj_jac(h, bl, be)
    return o


def _solve_jac(x, bl, be):
    h = np.clip(x, 0.0, 1.0)
    _, g = smooth_obj_jac(h, bl, be)
    return g


def _sum_con_fun(x):

def solve_smooth(h_start, beta_lag, beta_e, maxiter=200, tol=1e-11):
    n = len(h_start)
    h0 = repair_feasible(h_start)
    bounds = [(0.0, 1.0)] * n

    except Exception:
        return None

def interpolate_to_dimension(vec, new_len):
    n = len(vec)
