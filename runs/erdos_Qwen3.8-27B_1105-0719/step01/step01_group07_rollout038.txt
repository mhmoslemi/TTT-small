import numpy as np
from scipy.optimize import minimize
import time

def exact_objective(h):
    n = len(h)
    if n == 0:
        return float('inf')
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(profile))

def profile_values(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_box_sum(h):
    h = np.array(h, dtype=np.float64).copy()
    n = len(h)
    target = n / 2.0
    h[~np.isfinite(h)] = 0.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            h[mask] = np.minimum(1.0, h[mask] + diff / cnt)
        else:
            mask = h > 1e-14
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            h[mask] = np.maximum(0.0, h[mask] + diff / cnt)
    return np.clip(h, 0.0, 1.0)

# Epigraph formulation
def epigraph_fun(x, n):
    """x = [h_0, ..., h_{n-1}, t], minimize t"""
    return x[-1]

def epigraph_grad(x, n):
    """Gradient of t with respect to x"""
    g = np.zeros(n + 1)
    g[-1] = 1.0
    return g

class EpigraphConstraints:
    def __init__(self, n):
        self.n = n
        self.target = n / 2.0
    
    def fun(self, x):
        h = x[:self.n]
        t = x[-1]
        prof = profile_values(h)
        # Inequality constraints: t - C_k >= 0, i.e., -(t - C_k) <= 0
        # For scipy, we express as >= 0
        return t - prof  # each element should be >= 0
    
    def jac(self, x, jac_arg=None):
        n = self.n
        h = x[:n]
        t = x[-1]
        prof = profile_values(h)
        m = len(prof)  # 2n-1
        # J is m+1 x (n+1)
        # First m rows: d(t - prof_k)/dx
        # Last row: d(sum(h) - target)/dx
        J = np.zeros((m + 1, n + 1))
        dx = 2.0 / n
        # For each lag, compute gradient of C_lag
        # prof[k] corresponds to lag = k - (n-1)
        for k in range(m):
            lag = k - (n - 1)
            # dC_lag/dh_j for each j
            # h[j] appears as h[m+lag] when m=j-lag (valid if max(0,-lag) <= j-lag <= min(n-1, n-1-lag))
            # and as (1-h[m]) when m=j (valid if max(0,-lag) <= j <= min(n-1, n-1-lag))
            lo = max(0, -lag)
            hi = min(n - 1, n - 1 - lag)
            # Contribution from h[j] as first factor: j-lag in [lo, hi] means j in [lo+lag, hi+lag]
            lo1 = lo + lag
            hi1 = hi + lag
            for j in range(max(0, lo1), min(n, hi1 + 1)):
                m_idx = j - lag
                J[k, j] += dx * (1.0 - h[m_idx])
            # Contribution from h[j] as second factor (1-h[j]): j in [lo, hi]
            for j in range(lo, hi + 1):
                J[k, j] -= dx * h[j + lag]
            # d/dt of (t - C_k) = 1
            J[k, n] = 1.0
        # Last row: sum constraint
        J[m, :n] = 1.0  # d(sum(h))/dh_j = 1
        J[m, n] = 0.0
        return J

def epigraph_optimize(h0, deadline, maxiter=100):
    n = len(h0)
    h0 = project_box_sum(h0.copy())
    c0 = exact_objective(h0)
    x0 = np.concatenate([h0, [c0 + 0.01]])
    
    cons = EpigraphConstraints(n)
    
    def cons_fun(x):
        return cons.fun(x)
    
    def cons_jac(x):
        return cons.jac(x)
    
    constraints = [
        {'type': 'ineq', 'fun': cons_fun, 'jac': cons_jac},
        {'type': 'eq', 'fun': lambda x: np.sum(x[:n]) - n/2.0, 'jac': lambda x: np.pad(np.ones(n), (0,1))}
    ]
    
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 0.1)]
    
    try:
        res = minimize(epigraph_fun, x0, args=(n,), method='SLSQP',
                      jac=lambda x: epigraph_grad(x, n),
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-10})
        h_new = project_box_sum(res.x[:n])
        return h_new
    except Exception:
        return h0.copy()

def project_box_sum(h):

    target = n / 2.0
    h[~np.isfinite(h)] = 0.0

            break

            mask = h < 1.0 - 1e-14
            cnt = int(np.sum(mask))
            if cnt == 0:
                break

        else:

def lse_obj_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    m = float(np.max(profile))
    shifted = np.minimum((profile - m) / tau, 0.0)
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    weights = exp_vals / sum_exp
    obj = m + tau * np.log(sum_exp)
    grad = np.zeros(n)
    for j in range(2 * n - 1):
        w = weights[j]
        if w < 1e-13:
            continue
        s = j - (n - 1)
        if s >= 0:
            k1 = np.arange(s, n)
            grad[k1] += w * dx * (1.0 - h[k1 - s])
            k2 = np.arange(0, n - s)
            grad[k2] -= w * dx * h[k2 + s]
        else:
            a = -s
            k1 = np.arange(0, n - a)
            grad[k1] += w * dx * (1.0 - h[k1 + a])
            k2 = np.arange(a, n)
            grad[k2] -= w * dx * h[k2 - a]
    return obj, grad

class LSEFunc:
    def __init__(self, tau):
        self.tau = tau
    def value(self, x):
        return lse_obj_grad(x, self.tau)[0]
    def gradient(self, x):
        return lse_obj_grad(x, self.tau)[1]

class SumConstraint:
    def __init__(self, target):
        self.target = target
    def fun(self, x):
        return np.sum(x) - self.target
    def jac(self, x):
        return np.ones(len(x))

def slsqp_optimize(h0, tau, maxiter=300):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    obj = LSEFunc(tau)
    con = SumConstraint(n / 2.0)
    constraints = [{'type': 'eq', 'fun': con.fun, 'jac': con.jac}]
    try:
        res = minimize(obj.value, h0.copy(), method='SLSQP', jac=obj.gradient,
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        h_new = project_box_sum(np.array(res.x, dtype=np.float64))
        return h_new
    except Exception:
        return h0.copy()

    if n == 0:
        return float('inf')
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(profile))


def profile_values(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx


def project_box_sum(h):
    h = np.array(h, dtype=np.float64).copy()
    n = len(h)
    target = n / 2.0
    h[~np.isfinite(h)] = 0.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            h[mask] = np.minimum(1.0, h[mask] + diff / cnt)
        else:
            mask = h > 1e-14
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            h[mask] = np.maximum(0.0, h[mask] + diff / cnt)
    return np.clip(h, 0.0, 1.0)


def binary_threshold(h):
    n = len(h)
    if n % 2 != 0:
        return None
    m = n // 2
    idx = np.argsort(h)[::-1][:m]
    b = np.zeros(n, dtype=np.float64)
    b[idx] = 1.0
    return b


def greedy_swap(b, deadline, max_rounds=300):
    n = len(b)
    curr_c = exact_objective(b)
    ones = list(np.where(b == 1.0)[0])
    zeros = list(np.where(b == 0.0)[0])
    for rnd in range(max_rounds):
        if time.time() > deadline - 3:
            break
        best_improve = 1e-14
        best_swap = None
        for i in ones:
            for j in zeros:
                b[i] = 0.0
                b[j] = 1.0
                c = exact_objective(b)
                b[i] = 1.0
                b[j] = 0.0
                improve = curr_c - c
                if improve > best_improve:
                    best_improve = improve
                    best_swap = (i, j)
        if best_swap is not None:
            i, j = best_swap
            b[i] = 0.0
            b[j] = 1.0
            curr_c = exact_objective(b)
            ones = list(np.where(b == 1.0)[0])
            zeros = list(np.where(b == 0.0)[0])
        else:
            break
    return b.copy(), curr_c


def sa_binary(b_init, deadline, max_iters=500000, seed=42):
    n = len(b_init)
    b = b_init.copy()
    curr_c = exact_objective(b)
    best_b = b.copy()
    best_c = curr_c
    rng = np.random.RandomState(seed)
    T_start = 0.05
    T_min = 0.0015
    ones = list(np.where(b == 1.0)[0])
    zeros = list(np.where(b == 0.0)[0])
    no_improve = 0
    kick_interval = 8000
    for it in range(max_iters):
        if time.time() > deadline - 3:
            break
        progress = min(1.0, it / (max_iters * 0.85))
        T = T_start * (T_min / T_start) ** progress
        T = max(T, T_min)
        if no_improve > kick_interval:
            T = max(T, T_start * 0.4)
            no_improve = 0
        move_type = rng.random()
        if move_type < 0.6:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            b[i] = 0.0
            b[j] = 1.0
            new_c = exact_objective(b)
            delta = new_c - curr_c
            if delta < 0 or rng.random() < np.exp(-delta / T):
                curr_c = new_c
                ones.remove(i)
                ones.append(j)
                zeros.remove(j)
                zeros.append(i)
                if curr_c < best_c - 1e-15:
                    best_c = curr_c
                    best_b = b.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                b[i] = 1.0
                b[j] = 0.0
        elif move_type < 0.9:
            idx1, idx2 = rng.choice(len(ones), size=2, replace=False)
            jdx1, jdx2 = rng.choice(len(zeros), size=2, replace=False)
            a, c = ones[idx1], ones[idx2]
            d, e = zeros[jdx1], zeros[jdx2]
            b[a] = 0.0
            b[c] = 0.0
            b[d] = 1.0
            b[e] = 1.0
            new_c = exact_objective(b)
            delta = new_c - curr_c
            if delta < 0 or rng.random() < np.exp(-delta / T):
                curr_c = new_c
                ones[idx1] = d
                ones[idx2] = e
                zeros[jdx1] = a
                zeros[jdx2] = c
                if curr_c < best_c - 1e-15:
                    best_c = curr_c
                    best_b = b.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                b[a] = 1.0
                b[c] = 1.0
                b[d] = 0.0
                b[e] = 0.0
        else:
            seg_len = rng.randint(2, min(n // 3, 12))
            i = rng.randint(0, n - seg_len)
            j = i + seg_len
            b[i:j] = b[i:j][::-1]
            new_c = exact_objective(b)
            delta = new_c - curr_c
            if delta < 0 or rng.random() < np.exp(-delta / T):
                curr_c = new_c
                ones = list(np.where(b == 1.0)[0])
                zeros = list(np.where(b == 0.0)[0])
                if curr_c < best_c - 1e-15:
                    best_c = curr_c
                    best_b = b.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                b[i:j] = b[i:j][::-1]
    return best_b, best_c


def lse_obj_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    m = float(np.max(profile))
    shifted = np.minimum((profile - m) / tau, 0.0)
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    weights = exp_vals / sum_exp
    obj = m + tau * np.log(sum_exp)
    grad = np.zeros(n)
    for j in range(2 * n - 1):
        w = weights[j]
        if w < 1e-13:
            continue
        s = j - (n - 1)
        if s >= 0:
            k1 = np.arange(s, n)
            grad[k1] += w * dx * (1.0 - h[k1 - s])
            k2 = np.arange(0, n - s)
            grad[k2] -= w * dx * h[k2 + s]
        else:
            a = -s
            k1 = np.arange(0, n - a)
            grad[k1] += w * dx * (1.0 - h[k1 + a])
            k2 = np.arange(a, n)
            grad[k2] -= w * dx * h[k2 - a]
    return obj, grad


class LSEFunc:
    def __init__(self, tau):
        self.tau = tau

    def value(self, x):
        return lse_obj_grad(x, self.tau)[0]

    def gradient(self, x):
        return lse_obj_grad(x, self.tau)[1]


class SumConstraint:
    def __init__(self, target):
        self.target = target

    def fun(self, x):
        return np.sum(x) - self.target

    def jac(self, x):
        return np.ones(len(x))


def slsqp_optimize(h0, tau, maxiter=200):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    obj = LSEFunc(tau)
    con = SumConstraint(n / 2.0)
    constraints = [{'type': 'eq', 'fun': con.fun, 'jac': con.jac}]
    try:
        res = minimize(obj.value, h0.copy(), method='SLSQP', jac=obj.gradient,
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        h_new = project_box_sum(np.array(res.x, dtype=np.float64))
        return h_new
    except Exception:
        return h0.copy()


def refine_continuous(h0, deadline):
    h = project_box_sum(h0.copy())
    curr_c = exact_objective(h)
    taus = [0.01, 0.004, 0.0015, 0.0005, 0.00015]
    for tau in taus:
        if time.time() > deadline - 5:
            break
        h_new = slsqp_optimize(h, tau, maxiter=150)
        c_new = exact_objective(h_new)
        if c_new < curr_c - 1e-13:
            h = h_new
            curr_c = c_new
    return h, curr_c


def transfer_to_size(h_parent, n_target):
    n_src = len(h_parent)
    if n_src == n_target:
        return project_box_sum(h_parent.copy())
    x_src = (np.arange(n_src) + 0.5) / n_src
    x_tgt = (np.arange(n_target) + 0.5) / n_target
    h_new = np.interp(x_tgt, x_src, h_parent)
    return project_box_sum(h_new)


def make_start(n, rng, mode):
    if mode == 0:
        h = np.zeros(n)
        h[:n // 2] = 1.0
        h = h[rng.permutation(n)]
    elif mode == 1:
        x = np.linspace(0, 1, n)
        freq = rng.choice([2, 3, 4, 5, 6])
        phase = rng.uniform(0, 2 * np.pi)
        h = 0.5 + 0.45 * np.sin(2 * np.pi * freq * x + phase)
        h = np.clip(h, 0, 1)
    elif mode == 2:
        h = np.zeros(n)
        h[::2] = 1.0
    elif mode == 3:
        h = rng.uniform(0, 1, n)
    elif mode == 4:
        h = np.zeros(n)
        block = max(2, n // 8)
        for i in range(0, n, 2 * block):
            h[i:i + block] = 1.0
    elif mode == 5:
        h = np.zeros(n)
        h[:n // 4] = 1.0
        h[3 * n // 4:] = 1.0
    elif mode == 6:
        h = np.zeros(n)
        period = max(4, n // 6)
        fill = period // 2
        for i in range(0, n, period):
            h[i:i + fill] = 1.0
    else:
        h = np.full(n, 0.5)
    return project_box_sum(h)


def random_perturb(h, scale, rng):
    h_new = h + rng.normal(0, scale, len(h))
    return project_box_sum(h_new)


def epigraph_smooth_optimize(h0, deadline):
    """Optimize using only the top few lags as constraints (cutting plane style)."""
    n = len(h0)
    h = project_box_sum(h0.copy())
    curr_c = exact_objective(h)
    for outer in range(3):
        if time.time() > deadline - 5:
            break
        prof = profile_values(h)
        top_lags = np.argsort(prof)[::-1][:10]
        tau = 0.003
        for _ in range(2):
            if time.time() > deadline - 4:
                break
            h_new = slsqp_optimize(h, tau, maxiter=100)
            c_new = exact_objective(h_new)
            if c_new < curr_c - 1e-14:
                h = h_new
                curr_c = c_new
            tau *= 0.5
    return h, curr_c


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    deadline = t_start + budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c = float('inf')

    # Load parent
    parent_h = None
    try:
        parent_h = np.array(globals()['initial_h_values'], dtype=np.float64).copy()
        if len(parent_h) > 0:
            parent_h = project_box_sum(parent_h)
            pc = exact_objective(parent_h)
            if pc < best_c:
                best_c = pc
                best_h = parent_h.copy()
    except Exception:
        pass

    if best_h is None:
        best_h = make_start(48, rng, 0)
        best_c = exact_objective(best_h)

    parent_n = len(parent_h) if parent_h is not None else 48

    # Sizes to explore (even numbers)
    sizes = [48, 50, 52, 54, 56, 60, 64, 44, 40]
    sizes = [s for s in sizes if s % 2 == 0 and 20 <= s <= 80]
    if parent_n in sizes:
        sizes.remove(parent_n)
    sizes = [parent_n] + sizes

    # Phase 1: Generate diverse candidates, continuous refine, convert to binary
    binary_pool = []  # (binary_array, c_value, size)

    for n in sizes:
        if time.time() > deadline - 200:
            break

        cands = []
        if parent_h is not None:
            cands.append(transfer_to_size(parent_h, n))
            cands.append(random_perturb(transfer_to_size(parent_h, n), 0.02, rng))
            cands.append(random_perturb(transfer_to_size(parent_h, n), 0.05, rng))
        for mode in range(8):
            cands.append(make_start(n, rng, mode))

        scored = sorted([(exact_objective(c), c) for c in cands], key=lambda x: x[0])

        # Refine top candidates continuously
        for idx in range(min(3, len(scored))):
            if time.time() > deadline - 150:
                break
            h_ref, c_ref = refine_continuous(scored[idx][1], deadline)
            if c_ref < best_c:
                best_c = c_ref
                best_h = h_ref.copy()
            b = binary_threshold(h_ref)
            if b is not None:
                c_b = exact_objective(b)
                if c_b < best_c:
                    best_c = c_b
                    best_h = b.copy()
                binary_pool.append((b, c_b, n))

        # Binary from raw candidates
        for idx in range(min(5, len(scored))):
            b = binary_threshold(scored[idx][1])
            if b is not None:
                c_b = exact_objective(b)
                if c_b < best_c:
                    best_c = c_b
                    best_h = b.copy()
                binary_pool.append((b, c_b, n))

    # Deduplicate and sort binary pool
    binary_pool.sort(key=lambda x: x[1])
    seen = {}
    deduped = []
    for b, c, n in binary_pool:
        if seen.get(n, 0) < 4:
            deduped.append((b, c, n))
            seen[n] = seen.get(n, 0) + 1
    binary_pool = deduped[:12]

    # Phase 2: Aggressive binary optimization
    for b, c, n in binary_pool:
        if time.time() > deadline - 50:
            break

        # Greedy single swap first
        b_g, c_g = greedy_swap(b.copy(), deadline, max_rounds=200)
        if c_g < best_c:
            best_c = c_g
            best_h = b_g.copy()

        # SA with single swaps (multiple temperature schedules)
        if time.time() < deadline - 40:
            b_sa1, c_sa1 = sa_binary(b_g.copy(), deadline, max_iters=300000, seed=7)
            if c_sa1 < best_c:
                best_c = c_sa1
                best_h = b_sa1.copy()

        if time.time() < deadline - 35:
            b_sa2, c_sa2 = sa_binary(b_g.copy(), deadline, max_iters=300000, seed=17)
            if c_sa2 < best_c:
                best_c = c_sa2
                best_h = b_sa2.copy()

        # SA from the SA result (chain)
        if time.time() < deadline - 20:
            b_sa3, c_sa3 = sa_binary(best_h.copy(), deadline, max_iters=200000, seed=31)
            if c_sa3 < best_c:
                best_c = c_sa3
                best_h = b_sa3.copy()

    # Phase 3: Cross-size exploration from best
    if best_h is not None and time.time() < deadline - 40:
        current_n = len(best_h)
        for n in [48, 50, 52, 54, 56, 60]:
            if n == current_n or time.time() > deadline - 20:
                continue
            h_t = transfer_to_size(best_h, n)
            b_t = binary_threshold(h_t)
            if b_t is not None:
                b_g, c_g = greedy_swap(b_t, deadline, max_rounds=100)
                if c_g < best_c:
                    best_c = c_g
                    best_h = b_g.copy()
                b_sa, c_sa = sa_binary(b_g, deadline, max_iters=200000, seed=99)
                if c_sa < best_c:
                    best_c = c_sa
                    best_h = b_sa.copy()

    # Phase 4: Final polish
    if best_h is not None and time.time() < deadline - 15:
        if np.all((best_h == 0.0) | (best_h == 1.0)):
            # Binary: more SA rounds
            b_final, c_final = sa_binary(best_h.copy(), deadline, max_iters=400000, seed=200)
            if c_final < best_c:
                best_c = c_final
                best_h = b_final.copy()
            # One more greedy pass
            b_g2, c_g2 = greedy_swap(best_h.copy(), deadline, max_rounds=100)
            if c_g2 < best_c:
                best_c = c_g2
                best_h = b_g2.copy()
        else:
            # Continuous: refine
            h_ref, c_ref = epigraph_smooth_optimize(best_h.copy(), deadline)
            if c_ref < best_c:
                best_c = c_ref
                best_h = h_ref.copy()
            b = binary_threshold(best_h)
            if b is not None:
                b_g, c_g = greedy_swap(b, deadline, max_rounds=150)
                b_sa, c_sa = sa_binary(b_g, deadline, max_iters=100000, seed=555)
                if c_sa < best_c:
                    best_c = c_sa
                    best_h = b_sa.copy()

    # Final validation
    best_h = project_box_sum(best_h)
    best_c = exact_objective(best_h)

    return best_h, best_c, len(best_h)
