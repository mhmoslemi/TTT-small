import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time


def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(100):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)


def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def c5_of_h(h, dx):
    return compute_corrs(h).max() * dx


def grad_corr_k(h, k, n):
    offset = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -offset)
    i_hi = min(n - 1, n - 1 - offset)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx + offset] += (1.0 - h[idx])
    grad[idx] += (-h[idx + offset])
    return grad


def gradient_smooth_max(h, dx, beta):
    n = len(h)
    corrs = compute_corrs(h)
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones_like(corrs) / len(corrs)
    else:
        w /= w_sum
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-13:
            continue
        grad += w[k] * grad_corr_k(h, k, n)
    return grad * dx


def smooth_max_value(h, dx, beta):
    return logsumexp(beta * compute_corrs(h)) / beta * dx


def smooth_max_grad(h, dx, beta):
    return gradient_smooth_max(h, dx, beta)


def slsqp_optimize(h0, dx, target_sum, beta, maxiter):
    n = len(h0)
    cons = [{'type': 'eq',
             'fun': lambda x: x.sum() - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(smooth_max_value, h0, args=(dx, beta),
                       jac=lambda x: smooth_max_grad(x, dx, beta),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        h_out = project_feasible(res.x, target_sum)
    except Exception:
        h_out = project_feasible(h0, target_sum)
    return h_out, c5_of_h(h_out, dx)


def coordinate_descent(h0, dx, target_sum, max_rounds, rng):

    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    
    for rnd in range(max_rounds):
        improved = False
        order = rng.permutation(n)
        for j in order:
            old_val = h[j]
            # Compute a_s and c_s for all shifts
            # h_js = h[j+s] if 0<=j+s<n else 0
            # h_jm = h[j-s] if 0<=j-s<n else 0
            corrs = compute_corrs(h)
            # For each shift s (0 to n-1 in 0-index, representing shift -(n-1) to (n-1))
            # Actually let's use the direct correlation array
            # corr[k] where k = 0..2n-2, shift = k-(n-1)
            
            # Coefficient of h[j] in corr[k]:
            # shift s = k - (n-1)
            # h[j] appears as h[i] at i=j if j+s is valid: contributes (1-h[j+s])
            # h[j] appears as h[i+s] at i=j-s if j-s is valid: contributes -h[j-s]
            a = np.zeros(2 * n - 1)
            for s in range(-(n - 1), n):
                k = s + (n - 1)
                h_js = h[j + s] if 0 <= j + s < n else 0.0
                h_jm = h[j - s] if 0 <= j - s < n else 0.0
                a[k] = (1.0 - h_js) - h_jm

def make_binary_start(n, target_sum, rng):
    h = np.zeros(n)
    idx = rng.permutation(n)
    n_ones = int(target_sum)
    if n_ones > 0:
        h[idx[:n_ones]] = 1.0
    frac = target_sum - n_ones
    if frac > 1e-10 and n_ones < n:
        h[idx[n_ones]] = frac
    return h


def make_structured_start(n, target_sum, rng, pattern):
    h = np.zeros(n)
    if pattern == 0:
        # Block: first half is 1, second half is 0
        h[:n // 2] = 1.0
    elif pattern == 1:
        # Alternating blocks
        block = max(1, n // 8)
        for i in range(n):
            if (i // block) % 2 == 0:
                h[i] = 1.0
    elif pattern == 2:
        # Sinusoidal
        x = np.linspace(0, 2, n)
        h = 0.5 + 0.5 * np.sin(np.pi * x + rng.uniform(0, 2 * np.pi))
    elif pattern == 3:
        # Quasi-periodic
        x = np.linspace(0, 2, n)
        freq = rng.uniform(3, 8)
        h = 0.5 + 0.5 * np.cos(2 * np.pi * freq * x)
    else:
        h = make_binary_start(n, target_sum, rng)
    return project_feasible(np.clip(h, 0, 1), target_sum)
