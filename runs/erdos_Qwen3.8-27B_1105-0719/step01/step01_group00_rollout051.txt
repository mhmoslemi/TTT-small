def compute_corrs(h):

            if j + M < n:

def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(100):

        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid

    diff = target_sum - h.sum()

        mask = (h > 1e-10) & (h < 1.0 - 1e-10)

def c5_of_h(h, dx):

def compute_derivs_fast(h, j):
    n = len(h)
    # L ranges from -(n-1) to n-1
    L = np.arange(-(n-1), n)  # shape (2n-1,)
    d = np.zeros(2*n-1)
    
    # For each L, the contribution to d is:
    # If L >= 0: (1-h[j+L]) * [j+L < n] - h[j-L] * [j-L >= 0]
    # If L < 0, M=-L: (1-h[j-M]) * [j >= M] - h[j+M] * [j+M < n]
    
    # Let's handle positive and negative separately
    # For L >= 0:
    idx_pos = L >= 0
    L_pos = L[idx_pos]  # 0, 1, ..., n-1
    v1 = (j + L_pos < n).astype(float)  # [j+L < n]
    v2 = (j - L_pos >= 0).astype(float)  # [j-L >= 0]
    # h[j+L] for valid entries
    h_j_plus_L = np.zeros_like(L_pos, dtype=float)
    mask1 = (j + L_pos < n) & (j + L_pos >= 0)
    h_j_plus_L[mask1] = h[j + L_pos[mask1]]
    h_j_minus_L = np.zeros_like(L_pos, dtype=float)
    mask2 = (j - L_pos >= 0) & (j - L_pos < n)
    h_j_minus_L[mask2] = h[j - L_pos[mask2]]
    d[idx_pos] = (1.0 - h_j_plus_L) * v1 - h_j_minus_L * v2
    
    # For L < 0, M = -L:
    idx_neg = L < 0
    L_neg = L[idx_neg]  # -(n-1), ..., -1
    M = -L_neg  # 1, ..., n-1
    v1_neg = (j >= M).astype(float)
    v2_neg = (j + M < n).astype(float)
    h_j_minus_M = np.zeros_like(M, dtype=float)
    mask1_neg = (j - M >= 0) & (j - M < n)
    h_j_minus_M[mask1_neg] = h[j - M[mask1_neg]]
    h_j_plus_M = np.zeros_like(M, dtype=float)
    mask2_neg = (j + M >= 0) & (j + M < n)
    h_j_plus_M[mask2_neg] = h[j + M[mask2_neg]]
    d[idx_neg] = (1.0 - h_j_minus_M) * v1_neg - h_j_plus_M * v2_neg
    
    return d

def compute_derivs(h, j):
    n = len(h)
    d = np.zeros(2 * n - 1)

            val += (1.0 - h[j + L])
        if 0 <= j - L < n:
            val -= h[j - L]

    return d
