def repair_feasible(vec):
    h = np.asarray(vec, dtype=np.float64)

    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h.copy()
    if diff > 0:
        order = np.argsort(h)
        rem = diff
        for i in order:
            if rem <= 1e-15:
                break
            add = min(rem, 1.0 - h[i])
            h[i] += add
            rem -= add

                break

    """Transfer h to new dimension using CDF-based interpolation."""
    n = len(h)

    # Target CDF values at new cell boundaries
    x_new = np.linspace(0.0, 2.0, new_len + 1)
    x_src = np.linspace(0.0, 2.0, n + 1)
    cdf_new = np.interp(x_new, x_src, cdf)
    # New h values are differences
    h_new = np.diff(cdf_new) / (2.0 / new_len)
    return h_new

    h = x
    values, grads = compute_profile_and_gradients(h)

    shifted = beta * (values - M)

    log_sum = M + (1.0 / beta) * np.log(np.sum(exp_shifted))
    weights = exp_shifted / np.sum(exp_shifted)
    grad = np.dot(weights, grads)
    return log_sum, grad

    n = len(h_start)

        val, _ = soft_objective_and_grad(x, beta)
        return val

        return g

        return np.sum(x) - n / 2.0

        return np.ones(n)

            method='SLSQP', bounds=bounds, constraints=constraints,
            options={'maxiter': maxiter, 'ftol': 1e-16}

    except Exception:
        return None, None
    
    if result.x is not None:
        h_sol = repair_feasible(result.x)
        score = exact_score(h_sol)
        if np.isfinite(score):
            return h_sol, score
    return None, None

where w_r = exp(beta * C_r) / sum exp(beta * C_r)

I can compute this as:
dJ/dh_i = sum_r w_r * dC_r/dh_i

For r = n-1+delta (positive lag delta):
dC_delta/dh_i = dx * (1-h[i+delta]) for i < n-delta (from the h term)
              + (-dx * h[i-delta]) for i >= delta (from the (1-h) term)

Wait, I need to be more careful. Let me re-derive.

C_delta = dx * sum_{i=0}^{n-1-delta} h[i] * (1-h[i+delta]) for delta > 0

dC_delta/dh[j] = dx * (1-h[j+delta]) for j < n-delta (j appears in h[i] term with i=j)
              + dx * (-h[j-delta]) for j >= delta (j appears in h[i+delta] term with i=j-delta)
              = dx * (1-h[j+delta]) * 1_{j < n-delta} - dx * h[j-delta] * 1_{j >= delta}

For delta = 0:
C_0 = dx * sum_i h[i](1-h[i])
dC_0/dh[j] = dx * (1-2h[j])

Now for the soft-max gradient:
dJ/dh[j] = w_{n-1} * dx * (1-2h[j]) + sum_{delta=1}^{n-1} [w_{n-1+delta} + w_{n-1-delta}] * dx * (1-h[j+delta]) * 1_{j < n-delta}
           - sum_{delta=1}^{n-1} [w_{n-1+delta} + w_{n-1-delta}] * dx * h[j-delta] * 1_{j >= delta}

This can be computed in O(n²) total (O(n) per j), which is the same as the matrix approach.

For the purpose of this code, I'll just use the full matrix approach since n is at most ~200 and the matrix is manageable in memory.

import numpy as np
from scipy.optimize import minimize
import time


def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx


def repair_feasible(vec):
    h = np.asarray(vec, dtype=np.float64)
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h.copy()
    if diff > 0:
        order = np.argsort(h)
        rem = diff
        for i in order:
            if rem <= 1e-15:
                break
            add = min(rem, 1.0 - h[i])
            h[i] += add
            rem -= add
    else:
        order = np.argsort(-h)
        rem = -diff
        for i in order:
            if rem <= 1e-15:
                break
            remv = min(rem, h[i])
            h[i] -= remv
            rem -= remv
    diff2 = target - h.sum()
    if abs(diff2) > 1e-14:
        for i in range(n):
            if diff2 > 0 and h[i] < 1.0 - 1e-15:
                h[i] += diff2
                break
            elif diff2 < 0 and h[i] > 1e-15:
                h[i] += diff2
                break
    return h


def compute_profile_and_gradients(h):
    n = len(h)
    dx = 2.0 / n
    n_lags = 2 * n - 1
    values = np.zeros(n_lags)
    grads = np.zeros((n_lags, n))
    for delta in range(0, n):
        if delta == 0:
            values[n - 1] = dx * np.sum(h * (1.0 - h))
            grads[n - 1, :] = dx * (1.0 - 2.0 * h)
        else:
            idx = np.arange(n - delta)
            values[n - 1 + delta] = dx * np.sum(h[idx] * (1.0 - h[idx + delta]))
            grads[n - 1 + delta, idx] += dx * (1.0 - h[idx + delta])
            grads[n - 1 + delta, idx + delta] += -dx * h[idx]
            idx2 = np.arange(delta, n)
            values[n - 1 - delta] = dx * np.sum(h[idx2] * (1.0 - h[idx2 - delta]))
            grads[n - 1 - delta, idx2] += dx * (1.0 - h[idx2 - delta])
            grads[n - 1 - delta, idx2 - delta] += -dx * h[idx2]
    return values, grads


def cdf_transfer(h, new_len):
    n = len(h)
    dx = 2.0 / n
    cdf = np.zeros(n + 1)
    cdf[1:] = np.cumsum(h) * dx
    x_src = np.linspace(0.0, 2.0, n + 1)
    x_new = np.linspace(0.0, 2.0, new_len + 1)
    cdf_new = np.interp(x_new, x_src, cdf)
    h_new = np.diff(cdf_new) / (2.0 / new_len)
    return h_new


def soft_objective_and_grad(x, beta):
    h = x
    values, grads = compute_profile_and_gradients(h)
    M = np.max(values)
    shifted = beta * (values - M)
    shifted = np.clip(shifted, -500.0, 500.0)
    exp_shifted = np.exp(shifted)
    log_sum = M + (1.0 / beta) * np.log(np.sum(exp_shifted))
    wsum = np.sum(exp_shifted)
    weights = exp_shifted / wsum
    grad = np.dot(weights, grads)
    return log_sum, grad


def soft_minimize(h_start, beta, maxiter=200):
    n = len(h_start)
    h0 = repair_feasible(h_start)

    def obj(x):
        val, _ = soft_objective_and_grad(x, beta)
        return val

    def obj_grad(x):
        _, g = soft_objective_and_grad(x, beta)
        return g

    def eq_fun(x):
        return np.sum(x) - n / 2.0

    def eq_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    try:
        result = minimize(
            obj, h0, jac=obj_grad,
            method='SLSQP', bounds=bounds, constraints=constraints,
            options={'maxiter': maxiter, 'ftol': 1e-14}
        )
    except Exception:
        return None, None
    if result.x is not None:
        h_sol = repair_feasible(result.x)
        if np.all(np.isfinite(h_sol)):
            score = exact_score(h_sol)
            if np.isfinite(score):
                return h_sol, score
    return None, None


def active_descent(h, max_steps=60):
    n = len(h)
    h = repair_feasible(h)
    best = h.copy()
    best_score = exact_score(best)
    for step in range(max_steps):
        values, grads = compute_profile_and_gradients(best)
        c = np.max(values)
        n_active = min(25, 2 * n - 1)
        active_idx = np.argsort(values)[::-1][:n_active]
        gaps = np.maximum(c - values[active_idx], 0.0)
        sigma = 0.002 * c + 1e-10
        weights = np.exp(-gaps / sigma)
        wsum = np.sum(weights)
        if wsum < 1e-15:
            break
        weights /= wsum
        direction = -np.dot(weights, grads[active_idx, :])
        direction -= np.mean(direction)
        dnorm = np.linalg.norm(direction)
        if dnorm < 1e-12:
            break
        direction /= dnorm
        improved = False
        for alpha in [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]:
            h_trial = repair_feasible(best + alpha * dnorm * direction)
            score = exact_score(h_trial)
            if score < best_score - 1e-14:
                best = h_trial.copy()
                best_score = score
                improved = True
                break
        if not improved:
            # Try smaller steps
            for alpha in [5e-5, 2e-5, 1e-5, 5e-6]:
                h_trial = repair_feasible(best + alpha * direction)
                score = exact_score(h_trial)
                if score < best_score - 1e-14:
                    best = h_trial.copy()
                    best_score = score
                    improved = True
                    break
        if not improved:
            break
    return best, best_score


def symmetrize(h):
    return (h + h[::-1]) / 2.0
