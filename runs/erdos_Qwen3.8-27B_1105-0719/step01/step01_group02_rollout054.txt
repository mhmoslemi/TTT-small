def lp_minmax_step(h, delta=0.01, max_active=50):
    """Solve the linearized minmax LP to get a descent direction."""
    n = len(h)
    p, grad = profile_and_grad(h)
    C = np.max(p)
    
    # Active set
    active_mask = p >= C - delta
    active_idx = np.where(active_mask)[0]
    if len(active_idx) > max_active:
        # Keep only the top max_active
        top_idx = np.argsort(-p)[0:max_active]
        active_idx = top_idx
    
    A_act = grad[active_idx, :]  # shape (|A|, n)
    
    # Build LP
    # x = [s, d0, d1, ..., d_{n-1}]
    # min s
    # s.t. -s + A_act @ d <= 0  (|A| rows)
    #       d_i <= 1 - h_i     (n rows)
    #      -d_i <= h_i         (n rows)
    #      sum d_i = 0         (1 eq row)
    
    m = len(active_idx)
    # A_ub: (m + 2n) x (1 + n)
    A_ub = np.zeros((m + 2*n, 1 + n))
    b_ub = np.zeros(m + 2*n)
    
    # Active constraints: -s + A_act @ d <= 0
    A_ub[:m, 0] = -1.0
    A_ub[:m, 1:] = A_act
    # b_ub[:m] = 0 (already zero)
    
    # d_i <= 1 - h_i
    A_ub[m:m+n, 1:] = np.eye(n)
    b_ub[m:m+n] = 1.0 - h
    
    # -d_i <= h_i
    A_ub[m+n:m+2*n, 1:] = -np.eye(n)
    b_ub[m+n:m+2*n] = h
    
    # A_eq: sum d_i = 0
    A_eq = np.zeros((1, 1 + n))
    A_eq[0, 1:] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds: s is free (can be negative = improvement), d is unbounded (constrained by box)
    bounds = [(None, None)] + [(None, None)] * n
    
    # Solve
    from scipy.optimize import linprog
    result = linprog(c=np.concatenate([[1.0], np.zeros(n)]),
                     A_ub=A_ub, b_ub=b_ub,
                     A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
    
    if result.status != 0:
        return None, C, p
    
    s_val = result.x[0]
    d = result.x[1:]
    
    return d, C, p

import numpy as np
from scipy.optimize import linprog
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def profile_values(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def profile_gradient(h):
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros((2 * n - 1, n))
    for t in range(2 * n - 1):
        k = t - (n - 1)
        if k >= 0:
            grad[t, k:n] += dx * (1.0 - h[0:n - k])
            grad[t, 0:n - k] -= dx * h[k:n]
        else:
            s = -k
            grad[t, s:n] += dx * (1.0 - h[0:n - s])
            grad[t, 0:n - s] -= dx * h[s:n]
    return grad

def project_box_sum(h, target):
    n = len(h)
    h = np.array(h, dtype=np.float64)
    h = np.clip(np.nan_to_num(h, nan=0.0, posinf=1.0, neginf=0.0), 0.0, 1.0)
    for _ in range(500):
        s = np.sum(h)
        diff = s - target
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h > 0
            if not np.any(mask):
                break
            step = diff / np.sum(mask)
            step = min(step, np.min(h[mask]))
            h[mask] -= step
        else:
            mask = h < 1.0
            if not np.any(mask):
                break
            step = -diff / np.sum(mask)
            step = min(step, 1.0 - np.max(h[mask]))
            h[mask] += step
    h = np.clip(h, 0.0, 1.0)
    return h

def lp_minmax_direction(h, delta=0.01, top_k=30):
    n = len(h)
    p = profile_values(h)
    C = np.max(p)
    active_mask = p >= C - delta
    active_idx = np.where(active_mask)[0]
    if len(active_idx) > top_k:
        active_idx = np.argsort(-p)[0:top_k]
    elif len(active_idx) == 0:
        active_idx = np.argmax(p)
        active_idx = np.array([active_idx])
    A_act = profile_gradient(h)[active_idx, :]
    m = len(A_act)
    A_ub = np.zeros((m + 2 * n, 1 + n))
    b_ub = np.zeros(m + 2 * n)
    A_ub[:m, 0] = -1.0
    A_ub[:m, 1:] = A_act
    A_ub[m:m + n, 1:] = np.eye(n)
    b_ub[m:m + n] = 1.0 - h
    A_ub[m + n:m + 2 * n, 1:] = -np.eye(n)
    b_ub[m + n:m + 2 * n] = h
    A_eq = np.zeros((1, 1 + n))
    A_eq[0, 1:] = 1.0
    b_eq = np.zeros(1)
    c_obj = np.zeros(1 + n)
    c_obj[0] = 1.0
    bounds = [(None, None)] * (1 + n)
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'dual_feasibility_tolerance': 1e-10})
        if res.status != 0:
            return None, C, p, active_idx
        s_val = res.x[0]
        d = res.x[1:]
        return d, C, p, active_idx
    except Exception:
        return None, C, p, active_idx

def line_search_alpha(h0, d, c0, target, n_steps=25):
    n = len(h0)
    best_alpha = 0.0
    best_c = c0
    max_a = 1.0
    for alpha in np.logspace(-9, 0.0, n_steps):
        alpha = min(alpha, max_a)
        h_try = h0 + alpha * d
        h_try = project_box_sum(h_try, target)
        c_try = exact_score(h_try)
        if c_try < best_c - 1e-15:
            best_c = c_try
            best_alpha = alpha
    return best_alpha, best_c

def subgradient_descent(h, tau, lr, n_iter, target):
    n = len(h)
    c = exact_score(h)
    for it in range(n_iter):
        p, grad = profile_values(h), profile_gradient(h)
        C = np.max(p)
        shifted = np.clip((p - C) / tau, -60, 60)
        w = np.exp(shifted)
        w_sum = np.sum(w)
        if w_sum < 1e-300:
            break
        w /= w_sum
        g = w @ grad
        for frac in [1.0, 0.5, 0.25, 0.1]:
            h_new = project_box_sum(h - lr * frac * g, target)
            c_new = exact_score(h_new)
            if c_new < c - 1e-15:
                h = h_new
                c = c_new
                break
    return h, c

def transfer_to_size(parent, target_n):
    n = len(parent)
    if n == target_n:
        return parent.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(target_n) + 0.5) / target_n

def profile_gradient(h):
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros((2 * n - 1, n))
    for t in range(2 * n - 1):
        k = t - (n - 1)
        if k >= 0:
            grad[t, k:n] += dx * (1.0 - h[0:n - k])
            grad[t, 0:n - k] -= dx * h[k:n]
        else:
            s = -k
            grad[t, s:n] += dx * (1.0 - h[0:n - s])
            grad[t, 0:n - s] -= dx * h[s:n]
    return grad
