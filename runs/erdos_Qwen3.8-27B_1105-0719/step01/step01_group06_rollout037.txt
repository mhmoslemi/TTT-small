from scipy.optimize import minimize

def exact_score(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))

def get_profile(h):
    dx = 2.0 / len(h)
    return np.correlate(h, 1.0 - h, mode="full") * dx

def get_lse_gradient(h, beta):
    """Compute gradient of log-sum-exp of profile at temperature beta."""
    n = len(h)
    dx = 2.0 / n
    prof = get_profile(h)
    # Shift for numerical stability
    prof_shifted = prof - prof.max()
    w = np.exp(beta * prof_shifted)
    w = w / w.sum()
    
    # For each lag m (shift), the contribution to gradient at position j is:
    # w[m] * dx * [(1-h[j+m]) * I(0<=j+m<n) - h[j-m] * I(0<=j-m<n)]
    # where m = k - (n-1) for k = 0, ..., 2n-2
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        m = k - (n - 1)  # shift
        wk = w[k] * dx
        if wk < 1e-15:
            continue
        # First term: (1-h[j+m]) for j where 0<=j+m<n, i.e., j in [max(0,-m), min(n-1, n-1-m)]
        # j+m >= 0 => j >= -m
        # j+m < n => j < n-m
        # j >= 0, j < n
        j_start = max(0, -m)
        j_end = min(n, n - m)  # exclusive
        if j_end > j_start:
            grad[j_start:j_end] += wk * (1.0 - h[j_start + m:j_end + m])
        # Second term: -h[j-m] for j where 0<=j-m<n, i.e., j in [max(0,m), min(n, n+m)]
        # j-m >= 0 => j >= m
        # j-m < n => j < n+m
        # j >= 0, j < n
        j_start2 = max(0, m)
        j_end2 = min(n, n + m)
        if j_end2 > j_start2:
            grad[j_start2:j_end2] -= wk * h[j_start2 - m:j_end2 - m]
    
    return grad

def epigraph_solve(h0, t0, n_active=50, maxiter=100, time_limit=20):
    """SLSQP epigraph with dynamic active set."""
    n = len(h0)
    dx = 2.0 / n
    M = 2 * n - 1
    
    def objective(x):
        return x[n]
    
    def obj_grad(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g
    
    def constraints_fun(x):
        h = x[:n]
        t = x[n]
        eq = h.sum() - n / 2.0
        prof = get_profile(h)
        # Select active constraints
        active = np.argsort(prof)[::-1][:n_active]
        ineq = t - prof[active]
        return np.concatenate([[eq], ineq])
    
    def constraints_jac(x):
        h = x[:n]
        prof = get_profile(h)
        active = np.argsort(prof)[::-1][:n_active]
        # Compute gradients of active profile entries
        G = np.zeros((len(active), n))

            js2 = max(0, shift)
            je2 = min(n, n + shift)
            if je2 > js2:
                G[idx, js2:je2] -= dx * h[js2 - shift:je2 - shift]
        J = np.zeros((1 + len(active), n + 1))
        J[0, :n] = 1.0
        J[1:, :n] = -G
        J[1:, n] = 1.0
        return J
    
    bounds = [(0.0, 1.0)] * n + [(0.0, t0 + 1.0)]
    x0 = np.concatenate([h0, [min(t0, exact_score(h0) + 1e-8)]])

            objective, x0, jac=obj_grad,

            options={'maxiter': maxiter, 'ftol': 1e-12}

def lse_gradient(h, beta):
    n = len(h)
    dx = 2.0 / n
    prof = get_profile(h)
    m = prof.max()
    w = np.exp(beta * (prof - m))

    if wsum < 1e-300:
