    Row j: gradient of (t - C_{lag(j)}) with respect to x
    """

            cooldown -= 1
            lr *= 0.99
            continue

        profile = compute_profile(h)
        max_val = np.max(profile)
        # Subgradient: uniform over active set (within 1% of max)
        active = profile >= max_val * 0.99
        w = active / active.sum()

        grad = np.zeros(n)
        for j in range(2 * n - 1):
            if not active[j]:
                continue
            s = j - (n - 1)
            if s >= 0:
                idx1 = np.arange(s, n)
                idx2 = np.arange(0, n - s)
                grad[idx1] += w[j] * (2.0 / n) * (1.0 - h[idx1 - s])
                grad[idx2] -= w[j] * (2.0 / n) * h[idx2 + s]
            else:
                a = -s
                idx1 = np.arange(0, n - a)
                idx2 = np.arange(a, n)
                grad[idx1] += w[j] * (2.0 / n) * (1.0 - h[idx1 + a])
                grad[idx2] -= w[j] * (2.0 / n) * h[idx2 - a]

        v = beta * v + (1.0 - beta) * grad
        h_new = project_box_sum(h - lr * v)
        c_new = exact_objective(h_new)

        if c_new < best_c - 1e-12:
            best_h = h_new.copy()
            best_c = c_new
            h = h_new
            curr_c = c_new
        else:
            cooldown = 1
            lr *= 0.98

    return best_h, best_c


class EpiTopK:
    """Epigraph with top-K constraints + penalty for the rest."""

    def __init__(self, n, K=12, mu=500.0):
        self.n = n
        self.K = K
        self.mu = mu

    def value(self, x):
        h = x[:self.n]
        t = x[self.n]
        profile = compute_profile(h)
        idx = np.argsort(profile)[::-1][:self.K]
        pen_idx = np.argsort(profile)[self.K:]
        top_pen = np.sum(np.maximum(0, profile[idx] - t) ** 2)
        rest_pen = np.sum(np.maximum(0, profile[pen_idx] - t) ** 2)
        return t + self.mu * (top_pen + rest_pen)

    def gradient(self, x):
        n = self.n
        h = x[:n]
        t = x[n]
        profile = compute_profile(h)
        m = 2 * n - 1
        viol = np.maximum(0, profile - t)
        grad = np.zeros(n + 1)
        grad[n] = 1.0 - self.mu * np.sum(viol)

        for j in range(m):
            if viol[j] <= 0:
                continue
            w = self.mu * viol[j]
            s = j - (n - 1)
            if s >= 0:
                idx1 = np.arange(s, n)
                idx2 = np.arange(0, n - s)
                grad[idx1] += w * (2.0 / n) * (1.0 - h[idx1 - s])
                grad[idx2] -= w * (2.0 / n) * h[idx2 + s]
            else:
                a = -s
                idx1 = np.arange(0, n - a)
                idx2 = np.arange(a, n)
                grad[idx1] += w * (2.0 / n) * (1.0 - h[idx1 + a])
                grad[idx2] -= w * (2.0 / n) * h[idx2 - a]
        return grad


def slsqp_epigraph(h0, c0, maxiter=300, mu=200.0):
    n = len(h0)
    t0 = c0 + 0.01
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    obj = EpiTopK(n, K=15, mu=mu)
    con = SumEq(n / 2.0)

    def sum_fun_x(x):
        return np.sum(x[:n]) - n / 2.0

    def sum_jac_x(x):
        g = np.zeros(n + 1)
        g[:n] = 1.0
        return g

    constraints = [{'type': 'eq', 'fun': sum_fun_x, 'jac': sum_jac_x}]
    try:
        res = minimize(obj.value, x0, method='SLSQP', jac=obj.gradient,
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-11})
        h_new = project_box_sum(res.x[:n])
        c_new = exact_objective(h_new)
        if c_new < c0:
            return h_new, c_new
    except Exception:
        pass
    return h0, c0


def binary_threshold(h):
    n = len(h)
    if n % 2 != 0:
        return None
    m = n // 2
    idx = np.argsort(h)[::-1][:m]
    b = np.zeros(n, dtype=np.float64)
    b[idx] = 1.0
    return b


def binary_greedy_swap(b, deadline, max_rounds=200):
    n = len(b)
    curr_c = exact_objective(b)
    ones = np.where(b == 1.0)[0].tolist()
    zeros = np.where(b == 0.0)[0].tolist()
    for rnd in range(max_rounds):
        if time.time() > deadline - 2:
            break
        best_improve = 1e-14
        best_swap = None
        for i in ones:
            for j in zeros:
                b[i] = 0.0
                b[j] = 1.0
                c = exact_objective(b)
                b[i] = 1.0
                b[j] = 0.0
                improve = curr_c - c
                if improve > best_improve:
                    best_improve = improve
                    best_swap = (i, j)
        if best_swap is not None:
            i, j = best_swap
            b[i] = 0.0
            b[j] = 1.0
            curr_c = exact_objective(b)
            ones = np.where(b == 1.0)[0].tolist()
            zeros = np.where(b == 0.0)[0].tolist()
        else:
            break
    return b, curr_c


def binary_sa(b, deadline, t_initial=0.05, t_final=0.001, step=0.999):
    """Simulated annealing on binary solutions."""
    n = len(b)
    b = b.copy()
    curr_c = exact_objective(b)
    best_c = curr_c
    best_b = b.copy()
    rng = np.random.RandomState(12345)
    t = t_initial
    ones = np.where(b == 1.0)[0].tolist()
    zeros = np.where(b == 0.0)[0].tolist()
    max_iters = 50000

    for it in range(max_iters):
        if time.time() > deadline - 2:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        b[i] = 0.0
        b[j] = 1.0
        new_c = exact_objective(b)
        delta = new_c - curr_c
        if delta < 0 or rng.rand() < np.exp(-delta / t):
            curr_c = new_c
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            if curr_c < best_c - 1e-14:
                best_c = curr_c
                best_b = b.copy()
        else:
            b[i] = 1.0
            b[j] = 0.0
        t = max(t_final, t * step)
    return best_b, best_c


def binary_ils(b, deadline, max_rounds=100):
    """Iterated local search on binary solutions."""
    rng = np.random.RandomState(99999)
    n = len(b)
    best_b = b.copy()
    best_c = exact_objective(b)
    curr_b = b.copy()
    curr_c = best_c

    for rnd in range(max_rounds):
        if time.time() > deadline - 3:
            break
        # Local search
        curr_b, curr_c = binary_greedy_swap(curr_b, deadline, max_rounds=50)
        if curr_c < best_c - 1e-14:
            best_c = curr_c
            best_b = curr_b.copy()
        # Perturb: 2-4 random swaps
        n_swaps = rng.randint(2, 5)
        ones = np.where(curr_b == 1.0)[0].tolist()
        zeros = np.where(curr_b == 0.0)[0].tolist()
        for _ in range(n_swaps):
            if not ones or not zeros:
                break
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            curr_b[i] = 0.0
            curr_b[j] = 1.0
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
    return best_b, best_c


def make_start(n, rng, mode):
    if mode == 0:
        h = np.zeros(n)
        h[:n // 2] = 1.0
        h = h[rng.permutation(n)]
    elif mode == 1:
        x = np.linspace(0, 1, n)
        freq = rng.choice([2, 3, 4, 5, 6])
        phase = rng.uniform(0, 2 * np.pi)
        h = 0.5 + 0.45 * np.sin(2 * np.pi * freq * x + phase)
        h = np.clip(h, 0, 1)
    elif mode == 2:
        h = np.zeros(n)
        h[::2] = 1.0
    elif mode == 3:
        h = rng.uniform(0, 1, n)
    elif mode == 4:
        h = np.zeros(n)
        block = max(2, n // 8)
        for i in range(0, n, 2 * block):
            h[i:i + block] = 1.0
    elif mode == 5:
        h = np.zeros(n)
        h[:n // 4] = 1.0
        h[3 * n // 4:] = 1.0
    elif mode == 6:
        # Cyclic-like pattern
        h = np.zeros(n)
        period = max(3, n // 6)
        for i in range(n):
            if i % (2 * period) < period:
                h[i] = 1.0
    else:
        h = np.full(n, 0.5)
    return project_box_sum(h)


def random_perturb(h, scale, rng):
    h_new = h + rng.normal(0, scale, len(h))
    return project_box_sum(h_new)


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    deadline = t_start + budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c = float('inf')

    # Load parent
    parent_h = None
    try:
        parent_h = np.array(globals()['initial_h_values'], dtype=np.float64).copy()
        if len(parent_h) > 0:
            parent_h = project_box_sum(parent_h)
            pc = exact_objective(parent_h)
            if pc < best_c:
                best_c = pc
                best_h = parent_h.copy()
    except Exception:
        pass

    if best_h is None:
        best_h = make_start(48, rng, 0)
        best_c = exact_objective(best_h)

    parent_n = len(parent_h) if parent_h is not None else 48

    # Build size list around parent
    sizes = [parent_n]
    offsets = [0, 4, -4, 8, -8, 12, -12, 16, -16]
    for off in offsets:
        s = parent_n + off
        if 16 <= s <= 100 and s % 2 == 0 and s not in sizes:
            sizes.append(s)

    # Phase 1: Quick LSE refinement from diverse starts at parent size
    n_main = parent_n
    cands = []
    if parent_h is not None:
        cands.append(parent_h.copy())
        cands.append(random_perturb(parent_h, 0.02, rng))
        cands.append(random_perturb(parent_h, 0.05, rng))
    for mode in range(7):
        cands.append(make_start(n_main, rng, mode))

    scored = []
    for c in cands:
        cs = exact_objective(c)
        scored.append((cs, c))
    scored.sort(key=lambda x: x[0])

    # Refine top 3 with LSE
    for idx in range(min(3, len(scored))):
        if time.time() > deadline - 60:
            break
        h0 = scored[idx][1]
        h_ref, c_ref = refine_lse(h0, deadline)
        if c_ref < best_c - 1e-13:
            best_c = c_ref
            best_h = h_ref.copy()

    # Subgradient refinement on current best
    if time.time() < deadline - 50:
        h_sg, c_sg = subgradient_max(best_h.copy(), deadline - 20, max_iter=3000)
        if c_sg < best_c - 1e-13:
            best_c = c_sg
            best_h = h_sg.copy()

    # Phase 2: Epigraph SLSQP on best continuous solution
    if best_h is not None and time.time() < deadline - 40:
        n_ep = len(best_h)
        for mu_try in [100.0, 300.0, 800.0]:
            if time.time() > deadline - 20:
                break
            h_ep, c_ep = slsqp_epigraph(best_h, best_c, maxiter=300, mu=mu_try)
            if c_ep < best_c - 1e-13:
                best_c = c_ep
                best_h = h_ep.copy()

    # Phase 3: Try other sizes with LSE
    for n in sizes:
        if n == parent_n or time.time() > deadline - 40:
            continue
        h_size = transfer_to_size(best_h, n)
        h_ref, c_ref = refine_lse(h_size, deadline)
        if c_ref < best_c - 1e-13:
            best_c = c_ref
            best_h = h_ref.copy()

    # Phase 4: Binary SA + ILS
    if best_h is not None and len(best_h) % 2 == 0:
        # Binary from current best
        b = binary_threshold(best_h)
        if b is not None:
            c_b = exact_objective(b)
            if c_b < best_c - 1e-13:
                best_c = c_b
                best_h = b.copy()
            # Greedy first
            b_g, c_g = binary_greedy_swap(b.copy(), deadline, max_rounds=100)
            if c_g < best_c - 1e-13:
                best_c = c_g
                best_h = b_g.copy()
            # SA
            if time.time() < deadline - 15:
                b_sa, c_sa = binary_sa(b_g.copy(), deadline, t_initial=0.03,
                                       t_final=0.0005, step=0.9995)
                if c_sa < best_c - 1e-13:
                    best_c = c_sa
                    best_h = b_sa.copy()
            # ILS
            if time.time() < deadline - 10:
                b_ils, c_ils = binary_ils(b_g.copy(), deadline, max_rounds=50)
                if c_ils < best_c - 1e-13:
                    best_c = c_ils
                    best_h = b_ils.copy()

    # Phase 5: Binary at other sizes
    for n in sizes:
        if n % 2 != 0 or time.time() > deadline - 15:
            continue
        h_t = transfer_to_size(best_h, n)
        b = binary_threshold(h_t)
        if b is None:
            continue
        b_g, c_g = binary_greedy_swap(b, deadline, max_rounds=80)
        if c_g < best_c - 1e-13:
            best_c = c_g
            best_h = b_g.copy()
        if time.time() < deadline - 8:
            b_sa, c_sa = binary_sa(b_g, deadline, t_initial=0.02,
                                    t_final=0.0003, step=0.9997)
            if c_sa < best_c - 1e-13:
                best_c = c_sa
                best_h = b_sa.copy()

    # Phase 6: Final continuous polish
    if best_h is not None and time.time() < deadline - 8:
        # LSE one more time
        if not np.all((best_h == 0.0) | (best_h == 1.0)):
            h_ref, c_ref = refine_lse(best_h.copy(), deadline)
            if c_ref < best_c - 1e-13:
                best_c = c_ref
                best_h = h_ref.copy()
        # Epigraph one more
        if time.time() < deadline - 6:
            h_ep, c_ep = slsqp_epigraph(best_h, best_c, maxiter=200, mu=500.0)
            if c_ep < best_c - 1e-13:
                best_c = c_ep
                best_h = h_ep.copy()
        # Subgradient one more
        if time.time() < deadline - 4:
            h_sg, c_sg = subgradient_max(best_h.copy(), deadline - 1, max_iter=2000)
            if c_sg < best_c - 1e-13:
                best_c = c_sg
                best_h = h_sg.copy()

    # Final: binary from best continuous if not already binary
    if best_h is not None and len(best_h) % 2 == 0 and time.time() < deadline - 3:
        if not np.all((best_h == 0.0) | (best_h == 1.0)):
            b = binary_threshold(best_h)
            if b is not None:
                b_g, c_g = binary_greedy_swap(b, deadline, max_rounds=100)
                if c_g < best_c - 1e-13:
                    best_c = c_g
                    best_h = b_g.copy()

    # Final validation
    best_h = project_box_sum(best_h)
    best_c = exact_objective(best_h)
    best_n = len(best_h)

    if abs(np.sum(best_h) - best_n / 2.0) > 1e-8:
        best_h = project_box_sum(best_h)
        best_c = exact_objective(best_h)

    return best_h, best_c, best_n
