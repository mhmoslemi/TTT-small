def project_box_sum(h, target=None):
    n = len(h)
    h = np.array(h, dtype=np.float64).copy()
    h = np.nan_to_num(h, nan=0.0, posinf=1.0, neginf=0.0)
    if target is None:
        target = n / 2.0
    h = np.clip(h, 0.0, 1.0)

        diff = s - target
        if abs(diff) < 1e-14:

        if diff > 0:

            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            step = min(diff / cnt, np.min(h[mask]) * 0.999)
            if step <= 0:
                # Try proportional reduction
                h = h * (1.0 - diff / s)
                h = np.clip(h, 0.0, 1.0)

                h = np.clip(h, 0.0, 1.0)
                continue
            h[mask] += step
    # Final correction
