def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0

    if abs(diff) < 1e-14:
        return h
    if diff > 0:

        mid = (lo + hi) / 2
        vals = intercepts + slopes * mid
        idx = np.argmax(vals)
        # Check if there are ties
        max_val = vals[idx]
        active = np.where(np.abs(vals - max_val) < 1e-15)[0]
        if len(active) > 1:
            min_slope = slopes[active].min()
            max_slope = slopes[active].max()
            if min_slope <= 0 <= max_slope:
                return mid
            elif max_slope < 0:
                hi = mid
            else:
                lo = mid
        else:
            if slopes[idx] > 0:
                hi = mid
            elif slopes[idx] < 0:
                lo = mid
            else:
                return mid
        if hi - lo < tol:
            break
    return (lo + hi) / 2

    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx] + 1e-15)
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):

        vals = intercepts + slopes * mid
        max_val = np.max(vals)
        active = np.where(np.abs(vals - max_val) < 1e-10)[0]
        if len(active) > 1:
            ms = slopes[active].min()
            Ms = slopes[active].max()
            if ms <= 0.0 <= Ms:
                return mid
            elif Ms < 0.0:
                hi = mid
            else:
                lo = mid
        else:
            s = slopes[active[0]]
            if s > 0:
                hi = mid
            elif s < 0:
                lo = mid
            else:
                return mid
        if hi - lo < 1e-15:
            break
    return (lo + hi) / 2.0

def coordinate_descent(h, max_sweeps=150, rng=None):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c = exact_score(h)
    stall = 0
    if rng is None:
        rng = np.random.RandomState(0)
    for sweep in range(max_sweeps):
        improved = False
        order = rng.permutation(n)
        for j in order:
            profile = compute_profile(h)
            G = gradient_of_profile(h, n, dx)
            slopes = G[:, j]
            intercepts = profile - slopes * h[j]
            optimal_x = find_1d_min(intercepts, slopes)
            if abs(optimal_x - h[j]) > 1e-12:
                h[j] = optimal_x
                improved = True
        h = repair_feasible(h)
        c = exact_score(h)
        if c < best_c - 1e-13:
            best_c = c
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 30 and not improved:
                break
    return repair_feasible(h), best_c

def slsqp_solve(h0, maxiter=300, tol=1e-10):
    n = len(h0)
    dx = 2.0 / n
    M = 2 * n - 1
    c0 = exact_score(h0)

    def obj(x):
        return x[-1]

    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g

    def eq_fun(x):
        return x[:-1].sum() - n / 2.0

    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g

    def ineq_fun(x):
        h = x[:-1]
        t = x[-1]
        profile = np.correlate(h, 1 - h, mode="full") * dx
        return t - profile

    def ineq_jac(x):
        h = x[:-1]
        G = gradient_of_profile(h, n, dx)
        J = np.zeros((M, n + 1))
        J[:, :-1] = -G
        J[:, -1] = 1.0
        return J

    x0 = np.concatenate([h0, [c0 + 1e-3]])
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 1.0)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': tol})
        if res.x is not None and np.isfinite(res.x).all():
            h_new = repair_feasible(res.x[:-1])
            if h_new.sum() > 0:
                return h_new
    except Exception:
        pass
    return None

def smooth_gradient_descent(h0, max_iters=500, eps_start=0.05, eps_end=0.01):
    n = len(h0)
    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    best_c = exact_score(h)
    for it in range(max_iters):
        t = it / max(max_iters - 1, 1)
        eps = eps_start * (eps_end / eps_start) ** t
        profile = compute_profile(h)
        max_p = np.max(profile)
        exp_p = np.exp((profile - max_p) / eps)
        p = exp_p / np.sum(exp_p)
        G = gradient_of_profile(h, n, dx)
        grad = G.T @ p
        best_eta = 0.0
        best_val = best_c
        for eta in [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]:
            h_try = repair_feasible(h - eta * grad)
            s = exact_score(h_try)
            if s < best_val:
                best_val = s
                best_eta = eta
        if best_eta > 0:
            h = repair_feasible(h - best_eta * grad)
            c = exact_score(h)
            if c < best_c - 1e-14:
                best_c = c
                best_h = h.copy()
    return best_h, best_c

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

def symmetrize(h):
    return (h + h[::-1]) / 2.0

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s - (time.perf_counter() - t_start)

    parent = None
    try:
        parent = np.array(initial_h_values, dtype=np.float64)
    except Exception:
        pass
    if parent is None or len(parent) < 4:
        parent = np.full(66, 0.5)

    parent = repair_feasible(np.clip(parent, 0.0, 1.0))
    best_h = parent.copy()
    best_c = exact_score(parent)
    best_n = len(parent)

    h_sym = repair_feasible(symmetrize(parent))
    c_sym = exact_score(h_sym)
    if c_sym < best_c - 1e-12:
        best_h = h_sym.copy()
        best_c = c_sym

    # Phase 1: Coordinate descent
    if time_left() > 100:
        h_cd, c_cd = coordinate_descent(best_h, max_sweeps=100, rng=rng)
        if c_cd < best_c - 1e-13:
            best_h, best_c = h_cd, c_cd

    # Phase 2: SLSQP polish
    if time_left() > 80:
        h_sol = slsqp_solve(best_h, maxiter=300, tol=1e-11)
        if h_sol is not None:
            c_sol = exact_score(h_sol)
            if c_sol < best_c - 1e-13:
                best_h, best_c = h_sol, c_sol

    # Phase 3: Coordinate descent again
    if time_left() > 80:
        h_cd, c_cd = coordinate_descent(best_h, max_sweeps=100, rng=rng)
        if c_cd < best_c - 1e-13:
            best_h, best_c = h_cd, c_cd

    # Phase 4: Smooth gradient descent
    if time_left() > 100:
        h_sg, c_sg = smooth_gradient_descent(best_h, max_iters=300, eps_start=0.03, eps_end=0.005)
        if c_sg < best_c - 1e-13:
            best_h, best_c = h_sg, c_sg

    # Phase 5: Perturbation restarts
    if time_left() > 120:
        for trial in range(6):
            if time_left() < 30:
                break
            eps = 0.003 + 0.005 * trial
            z = rng.randn(len(best_h))
            z -= z.mean()
            z /= (np.linalg.norm(z) + 1e-15)
            h_pert = repair_feasible(best_h + eps * z)
            h_cd, c_cd = coordinate_descent(h_pert, max_sweeps=80, rng=rng)
            if c_cd < best_c - 1e-13:
                best_h, best_c = h_cd, c_cd
                if time_left() > 40:
                    h_sol = slsqp_solve(best_h, maxiter=200, tol=1e-10)
                    if h_sol is not None:
                        c_sol = exact_score(h_sol)
                        if c_sol < best_c - 1e-13:
                            best_h, best_c = h_sol, c_sol

    # Phase 6: Try different dimensions
    if time_left() > 100:
        n_base = len(best_h)
        candidates = []
        for delta in [-6, -4, -2, 2, 4, 6, 8, 12, -10, 10]:
            d = n_base + delta
            if 30 <= d <= 150:
                candidates.append(d)
        for new_n in candidates[:5]:
            if time_left() < 30:
                break
            h_interp = interpolate_to_n(best_h, new_n)
            h_cd, c_cd = coordinate_descent(h_interp, max_sweeps=80, rng=rng)
            if c_cd < best_c - 1e-13:
                best_h, best_c = h_cd, c_cd
                best_n = new_n
                if time_left() > 40:
                    h_sol = slsqp_solve(best_h, maxiter=200, tol=1e-10)
                    if h_sol is not None:
                        c_sol = exact_score(h_sol)
                        if c_sol < best_c - 1e-13:
                            best_h, best_c = h_sol, c_sol

    # Phase 7: Final coordinate descent + SLSQP
    if time_left() > 60:
        h_cd, c_cd = coordinate_descent(best_h, max_sweeps=100, rng=rng)
        if c_cd < best_c - 1e-14:
            best_h, best_c = h_cd, c_cd

    if time_left() > 40:
        h_sol = slsqp_solve(best_h, maxiter=400, tol=1e-12)
        if h_sol is not None:
            c_sol = exact_score(h_sol)
            if c_sol < best_c - 1e-14:
                best_h, best_c = h_sol, c_sol

    # Phase 8: Final coordinate descent
    if time_left() > 30:
        h_cd, c_cd = coordinate_descent(best_h, max_sweeps=60, rng=rng)
        if c_cd < best_c - 1e-14:
            best_h, best_c = h_cd, c_cd

    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)

    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            final_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 2:
            final_c = float(eval_result[0])
    except Exception:
        pass

    return best_h, final_c, len(best_h)
