    """Compute gradient of P_k with respect to h."""
    n = len(h)
    dx = 2.0 / n
    delta = k - (n - 1)
    g = np.zeros(n)

        for i in range(L):

            g[i + delta] += dx * (1.0 - h[i])

        L = n - d

            g[i] += dx * (1.0 - h[i + d])
            g[i + d] += -dx * h[i]

import numpy as np
import time
from scipy.optimize import minimize

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def get_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def lag_gradient(h, k):
    n = len(h)
    dx = 2.0 / n
    delta = k - (n - 1)
    g = np.zeros(n)
    if delta > 0:
        L = n - delta
        idx_a = np.arange(L)
        idx_b = idx_a + delta
        g[idx_b] += dx * (1.0 - h[idx_a])
        g[idx_a] -= dx * h[idx_b]
    elif delta < 0:
        c = -delta
        L = n - c
        idx_a = np.arange(L)
        idx_b = idx_a + c
        g[idx_a] += dx * (1.0 - h[idx_b])
        g[idx_b] -= dx * h[idx_a]
    else:
        g[:] = dx * (1.0 - 2.0 * h)
    return g

def full_lag_gradients(h):
    n = len(h)
    M = 2 * n - 1
    G = np.zeros((M, n))
    for k in range(M):
        G[k] = lag_gradient(h, k)
    return G

def top_lag_gradients(h, indices):
    n = len(h)
    G = np.zeros((len(indices), n))
    for i, k in enumerate(indices):
        G[i] = lag_gradient(h, k)
    return G

def project_onto_feasible(g, target):
    n = len(g)
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        h = np.clip(g - mid, 0.0, 1.0)
        s = h.sum()
        if s > target:
            lo = mid
        else:

        if hi - lo < 1e-15:

    h = np.clip(g - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-13:

        else:

        remaining = diff
        for idx in order:

            if remaining > 0:

                h[idx] += add
                remaining -= add
            else:

                h[idx] -= rem
                remaining += rem
    return h

def repair_feasible(h, target=None):

    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    n = len(h)

    return project_onto_feasible(h, target)

def projected_gradient_descent(h0, max_iters=500, time_limit=60.0, top_A=12,
                                temperature=0.015, use_momentum=False,
                                min_alpha=1e-7):

    best_h = h.copy()
    best_score = exact_score(h)

    alpha = 0.02
    momentum_v = np.zeros(n)

    for it in range(max_iters):
        if time.perf_counter() - t0 > time_limit:
            break
        if alpha < min_alpha:
            break

        score = float(np.max(prof))
        if score < best_score - 1e-14:
            best_score = score
            best_h = h.copy()

        sorted_idx = np.argsort(prof)[::-1][:A]
        p_active = prof[sorted_idx]
        p_max = p_active.max()
        weights = np.exp((p_active - p_max) / temperature)
        wsum = weights.sum()
        if wsum < 1e-30:

        weights /= wsum

        G_a = top_lag_gradients(h, sorted_idx)
        direction = G_a.T @ weights

            direction = momentum_v

        norm = np.linalg.norm(direction)
        if norm < 1e-14:
            break

                h = h_trial
                alpha = min(alpha * 1.08, 0.06)
                improved = True
                break

        if not improved:
            alpha *= 0.45
            momentum_v *= 0.5

    return best_h, best_score

def solve_epigraph_slsqp(h_start, t_init, maxiter=120, n_constraints=0, beta=0.0):
    n = len(h_start)
    M = 2 * n - 1
    nc = n_constraints if n_constraints > 0 else M
    prof = get_profile(h_start)
    if nc < M:
        active_lags = np.argsort(prof)[::-1][:nc]
    else:
        active_lags = np.arange(M)

    def obj(x):
        h = x[:n]
        t = x[n]
        return t + beta * np.sum(h * (1.0 - h))

    def obj_jac(x):
        h = x[:n]
        g = np.zeros(n + 1)
        g[:n] = beta * (1.0 - 2.0 * h)
        g[n] = 1.0
        return g

    def cons_fun(x):
        h = x[:n]
        t = x[n]
        eq = h.sum() - n / 2.0
        p = np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)
        ineq = t - p[active_lags]
        return np.concatenate([[eq], ineq])

    def cons_jac(x):
        h = x[:n]
        G = top_lag_gradients(h, active_lags)
        J = np.zeros((1 + len(active_lags), n + 1))
        J[0, :n] = 1.0
        J[1:, :n] = -G
        J[1:, n] = 1.0
        return J

    bounds = [(0.0, 1.0)] * n + [(0.0, t_init + 2.0)]
    x0 = np.concatenate([h_start, [t_init]])

    try:
        result = minimize(
            obj, x0, jac=obj_jac,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'ineq', 'fun': cons_fun, 'jac': cons_jac}],
            options={'maxiter': maxiter, 'ftol': 1e-12}
        )
        if result.x is not None and np.all(np.isfinite(result.x)):
            return repair_feasible(result.x[:n])
    except Exception:
        pass
    return None

            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}],
            options={'maxiter': maxiter, 'ftol': 1e-12}
        )
        if result.x is not None and np.all(np.isfinite(result.x)):
            return repair_feasible(result.x)
    except Exception:
        pass
    return None

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t_start = time.perf_counter()

    def tleft():
        return budget_s - (time.perf_counter() - t_start)

    try:
        parent = np.array(initial_h_values, dtype=np.float64).copy()
    except Exception:
        parent = np.full(140, 0.5)

    parent = repair_feasible(parent)
    best_h = parent.copy()
    best_score = exact_score(parent)

    # Phase A: Aggressive projected gradient from parent
    if tleft() > 80:
        tl = min(80, tleft() * 0.2)
        h_pgd, s_pgd = projected_gradient_descent(best_h, max_iters=400, time_limit=tl,
                                                    top_A=12, temperature=0.012,
                                                    use_momentum=False)
        if s_pgd < best_score - 1e-13:
            best_h = h_pgd
            best_score = s_pgd

    # Phase B: PDG with momentum
    if tleft() > 60:
        tl = min(50, tleft() * 0.15)
        h_pgd, s_pgd = projected_gradient_descent(best_h, max_iters=300, time_limit=tl,
                                                    top_A=10, temperature=0.01,
                                                    use_momentum=True)
        if s_pgd < best_score - 1e-13:
            best_h = h_pgd
            best_score = s_pgd

    # Phase C: SLSQP epigraph (all constraints)
    if tleft() > 50:
        mi = min(150, max(40, int(tleft() * 0.1)))
        h_sol = solve_epigraph_slsqp(best_h, best_score + 1e-6, maxiter=mi)
        if h_sol is not None:
            s = exact_score(h_sol)
            if s < best_score - 1e-13:
                best_h = h_sol
                best_score = s

    # Phase D: Quadratic subproblem (weighted active lags)
    if tleft() > 50:
        prof = get_profile(best_h)
        A = 15
        top_idx = np.argsort(prof)[::-1][:A]
        p_top = prof[top_idx]
        w = np.exp((p_top - p_top.max()) / 0.02)
        w /= w.sum()
        h_q = quadratic_subproblem(best_h, w, top_idx, maxiter=80)
        if h_q is not None:
            s = exact_score(h_q)
            if s < best_score - 1e-13:
                best_h = h_q
                best_score = s
        # PDG after quadratic
        if tleft() > 30:
            tl = min(25, tleft() * 0.1)
            h_pgd, s_pgd = projected_gradient_descent(best_h, max_iters=200, time_limit=tl,
                                                        top_A=12, temperature=0.01)
            if s_pgd < best_score - 1e-13:
                best_h = h_pgd
                best_score = s_pgd

    # Phase E: SLSQP with reduced constraints (faster per iteration)
    if tleft() > 40:
        for nc in [50, 80, 120]:
            if tleft() < 20:
                break
            mi = min(80, max(20, int(tleft() * 0.08)))
            h_sol = solve_epigraph_slsqp(best_h, best_score + 1e-6, maxiter=mi, n_constraints=nc)
            if h_sol is not None:
                s = exact_score(h_sol)
                if s < best_score - 1e-13:
                    best_h = h_sol
                    best_score = s

    # Phase F: PDG with various parameters
    if tleft() > 60:
        for A, temp in [(6, 0.008), (20, 0.02), (30, 0.03), (8, 0.015)]:
            if tleft() < 25:
                break
            tl = min(25, tleft() * 0.12)
            h_pgd, s_pgd = projected_gradient_descent(best_h, max_iters=200, time_limit=tl,
                                                        top_A=A, temperature=temp)
            if s_pgd < best_score - 1e-13:
                best_h = h_pgd
                best_score = s_pgd

    # Phase G: Transformations
    if tleft() > 40:
        n = len(best_h)
        h_sym = repair_feasible(0.5 * (best_h + best_h[::-1]))
        s_sym = exact_score(h_sym)
        if s_sym < best_score - 1e-13:
            best_h = h_sym
            best_score = s_sym

        h_comp = repair_feasible(0.5 * (best_h + (1.0 - best_h[::-1])))
        s_comp = exact_score(h_comp)
        if s_comp < best_score - 1e-13:
            best_h = h_comp
            best_score = s_comp

    # Phase H: Dimension exploration
    if tleft() > 80:
        pn = len(best_h)
        dims = sorted(set([d for d in [80, 96, 112, 128, 160, 176, 192]
                           if d != pn and 48 <= d <= 220]))
        for new_n in dims:
            if tleft() < 35:
                break
            h_interp = interpolate_to_dim(best_h, new_n)
            s_interp = exact_score(h_interp)
            tl = min(20, tleft() * 0.12)
            h_opt, s_opt = projected_gradient_descent(h_interp, max_iters=150, time_limit=tl,
                                                        top_A=10, temperature=0.012)
            if s_opt < best_score - 1e-13:
                best_h = h_opt
                best_score = s_opt
            if tleft() > 15:
                h_sol = solve_epigraph_slsqp(best_h, best_score + 1e-6, maxiter=40,
                                              n_constraints=max(30, 2 * len(best_h) - 1))
                if h_sol is not None:
                    s = exact_score(h_sol)
                    if s < best_score - 1e-13:
                        best_h = h_sol
                        best_score = s

    # Phase I: Perturbation + reoptimize
    if tleft() > 50:
        n = len(best_h)
        for p in range(10):
            if tleft() < 20:
                break
            eps = 0.002 + 0.0015 * p
            z = np.random.randn(n)
            z -= z.mean()
            h_pert = repair_feasible(best_h + eps * z)
            tl = min(8, tleft() * 0.06)
            h_opt, s_opt = projected_gradient_descent(h_pert, max_iters=60, time_limit=tl,
                                                        top_A=10, temperature=0.012)
            if s_opt < best_score - 1e-13:
                best_h = h_opt
                best_score = s_opt
                # Follow up with SLSQP
                if tleft() > 15:
                    h_sol = solve_epigraph_slsqp(best_h, best_score + 1e-6, maxiter=40)
                    if h_sol is not None:
                        s = exact_score(h_sol)
                        if s < best_score - 1e-13:
                            best_h = h_sol
                            best_score = s

    # Phase J: Final refinement cycle
    if tleft() > 30:
        for _ in range(3):
            if tleft() < 15:
                break
            tl = min(12, tleft() * 0.2)
            h_pgd, s_pgd = projected_gradient_descent(best_h, max_iters=100, time_limit=tl,
                                                        top_A=8, temperature=0.008)
            if s_pgd < best_score - 1e-13:
                best_h = h_pgd
                best_score = s_pgd
            if tleft() > 10:
                h_sol = solve_epigraph_slsqp(best_h, best_score + 1e-7, maxiter=50)
                if h_sol is not None:
                    s = exact_score(h_sol)
                    if s < best_score - 1e-13:
                        best_h = h_sol
                        best_score = s

    # Final repair and score
    best_h = repair_feasible(best_h)
    final_score = exact_score(best_h)
    return (best_h, final_score, len(best_h))
