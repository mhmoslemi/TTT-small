def trust_region_step2(h_cur, lam=0.03, active_idx=None):
    try:
        import cvxpy as cp
    except ImportError:
        return None, None
    n = len(h_cur)
    cur_score = exact_score(h_cur)
    values, grads = compute_lags(h_cur)
    L = len(values)
    if active_idx is None:
        active_idx = np.arange(L)
    h_new = cp.Variable(n)
    t = cp.Variable()
    lin = values[active_idx] + grads[active_idx, :] @ (h_new - h_cur)
    constraints = [
        lin <= t,
        cp.sum(h_new) == n / 2.0,
        h_new >= 0,
        h_new <= 1
    ]
    obj = cp.Minimize(t + lam * cp.sum_squares(h_new - h_cur))
    prob = cp.Problem(obj, constraints)
    h_new_val = None
    for solver in [cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]:
        try:
            prob.solve(solver=solver, verbose=False)
            if h_new.value is not None:
                h_new_val = np.asarray(h_new.value, dtype=np.float64)
                break
        except Exception:
            continue
    if h_new_val is None:
        return None, None
    best_h = h_cur.copy()
    best_score = cur_score
    for alpha in [1.0, 0.7, 0.5, 0.35, 0.25, 0.15, 0.1, 0.05]:
        h_trial = repair_feasible(h_cur + alpha * (h_new_val - h_cur))
        score = exact_score(h_trial)
        if score < best_score - 1e-15:
            best_h = h_trial
            best_score = score
            break
    if best_score < cur_score - 1e-15:
        return best_h, best_score
    return None, None


def active_descent(h, max_steps=60, time_limit=30.0):
    t_start = time.perf_counter()
    n = len(h)
    h = repair_feasible(h)
    best = h.copy()
    best_score = exact_score(best)
    for step in range(max_steps):
        if time.perf_counter() - t_start > time_limit:
            break
        values, grads = compute_lags(best)
        c = np.max(values)
        n_lags = len(values)
        for n_active in [50, 30, 15, 8, 5]:
            if time.perf_counter() - t_start > time_limit:
                break
            active_idx = np.argsort(values)[::-1][:min(n_active, n_lags)]
            gaps = np.maximum(c - values[active_idx], 0.0)
            temp = 0.002 * c + 1e-12
            weights = np.exp(-gaps / temp)
            wsum = np.sum(weights)
            if wsum < 1e-15:
                continue
            weights /= wsum
            direction = -np.sum(weights[:, None] * grads[active_idx, :], axis=0)
            direction -= np.mean(direction)
            if np.linalg.norm(direction) < 1e-15:
                continue
            improved = False
            for alpha in [0.03, 0.015, 0.008, 0.004, 0.002, 0.001, 0.0005, 0.0002, 0.0001]:
                h_trial = repair_feasible(best + alpha * direction)
                score = exact_score(h_trial)
                if score < best_score - 1e-14:
                    best = h_trial.copy()
                    best_score = score
                    improved = True
                    break
            if improved:
                break
    return best, best_score


def momentum_descent(h, max_steps=300, time_limit=30.0, lr=0.003, momentum=0.85):
    t_start = time.perf_counter()
    n = len(h)
    h = repair_feasible(h).copy()
    best = h.copy()
    best_score = exact_score(best)
    velocity = np.zeros(n)
    eta = 0.001
    for step in range(max_steps):
        if time.perf_counter() - t_start > time_limit:
            break
        values, grads = compute_lags(h)
        c = np.max(values)
        expo = np.exp((values - c) / eta)
        expo[~np.isfinite(expo)] = 0.0
        wsum = np.sum(expo)
        if wsum < 1e-15:
            break
        weights = expo / wsum
        direction = np.sum(weights[:, None] * grads, axis=0)
        direction -= np.mean(direction)
        velocity = momentum * velocity - lr * direction
        h_new = repair_feasible(h + velocity)
        score = exact_score(h_new)
        if score < best_score:
            best = h_new.copy()
            best_score = score
        h = h_new
        eta *= 0.997
    return best, best_score


def coordinate_polish(h, max_rounds=10, time_limit=20.0):
    t_start = time.perf_counter()
    n = len(h)
    h = repair_feasible(h).copy()
    best = h.copy()
    best_score = exact_score(best)
    for rnd in range(max_rounds):
        if time.perf_counter() - t_start > time_limit:
            break
        values, grads = compute_lags(best)
        c = np.max(values)
        top_idx = np.argsort(values)[::-1][:20]
        top_grads = grads[top_idx, :]
        top_w = np.ones(len(top_idx)) / len(top_idx)
        grad_max = np.sum(top_w[:, None] * top_grads, axis=0)
        grad_max -= np.mean(grad_max)
        # Try moving each coordinate in the direction that reduces the max lag
        for i in range(n):
            if time.perf_counter() - t_start > time_limit:
                break
            direction = np.zeros(n)
            direction[i] = -np.sign(grad_max[i]) if grad_max[i] != 0 else 0
            direction -= np.mean(direction)
            for alpha in [0.01, 0.005, 0.002, 0.001, 0.0005]:
                h_trial = repair_feasible(best + alpha * direction)
                score = exact_score(h_trial)
                if score < best_score - 1e-14:
                    best = h_trial
                    best_score = score
                    break
    return best, best_score


def symmetrize(h):
    return (h + h[::-1]) / 2.0


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()
    rng = np.random.RandomState(seed)

    def tl():
        return budget_s - (time.perf_counter() - t_start)

    # Get parent

        parent = np.asarray(initial_h_values, dtype=np.float64)

    else:

    if exact_score(h_sym) < exact_score(best_h) - 1e-13:
        best_h = h_sym

    best_score = exact_score(best_h)

    # Phase 1: Trust-region with various lambda and active sets
    if tl() > 80:

        active_sizes = [None, 100, 60, 40, 25, 15]
        progress = True
        while progress and tl() > 50:
            progress = False
            for lam in lams:
                if tl() < 40:
                    break

                    if tl() < 30:
                        break
                    if n_act is None:
                        act = None
                    else:

                        act = np.argsort(vals)[::-1][:n_act]
                    h_new, score_new = trust_region_step2(best_h, lam=lam, active_idx=act)
                    if h_new is not None and score_new < best_score - 1e-14:
                        best_h = h_new
                        best_score = score_new
                        progress = True
                        break

    # Phase 2: Active descent polish
    if tl() > 30:
        h_ad, score_ad = active_descent(best_h, max_steps=80, time_limit=min(30, tl() * 0.3))
        if score_ad < best_score - 1e-14:
            best_h = h_ad
            best_score = score_ad

    # Phase 3: More trust-region with refined parameters
    if tl() > 60:
        for lam in [0.003, 0.008, 0.015, 0.04, 0.07, 0.1, 0.15, 0.25]:
            if tl() < 35:
                break

                vals, _ = compute_lags(best_h)
                act = np.argsort(vals)[::-1][:n_act]
                h_new, score_new = trust_region_step2(best_h, lam=lam, active_idx=act)
                if h_new is not None and score_new < best_score - 1e-14:
                    best_h = h_new
                    best_score = score_new
                    break

    # Phase 4: Momentum descent
    if tl() > 35:
        h_mom, score_mom = momentum_descent(best_h, max_steps=400, time_limit=min(30, tl() * 0.3), lr=0.002)
        if score_mom < best_score - 1e-14:
            best_h = h_mom
            best_score = score_mom

    # Phase 5: Active descent again
    if tl() > 25:
        h_ad, score_ad = active_descent(best_h, max_steps=60, time_limit=min(20, tl() * 0.4))
        if score_ad < best_score - 1e-14:
            best_h = h_ad
            best_score = score_ad

    # Phase 6: Coordinate polish
    if tl() > 20:
        h_cp, score_cp = coordinate_polish(best_h, max_rounds=5, time_limit=min(15, tl() * 0.4))
        if score_cp < best_score - 1e-14:
            best_h = h_cp
            best_score = score_cp

    # Phase 7: Dimension transfer
    if tl() > 50:
        n_cur = len(best_h)
        dims = set()
        for dn in [5, 10, 15, 20]:

            if s_interp < best_score + 0.005:

                if h_new is not None and score_new < best_score - 1e-14:

                    if score_ad < best_score - 1e-14:

                        best_score = score_ad

    # Phase 8: Final trust-region polish
    if tl() > 40:
        for lam in [0.005, 0.01, 0.02, 0.05, 0.1]:
            if tl() < 20:
                break
            for n_act in [50, 30, 15]:
                if tl() < 15:

            best_score = score_cp

    # Final validation
    best_h = repair_feasible(best_h)
    best_score = exact_score(best_h)
    return best_h, best_score, len(best_h)
