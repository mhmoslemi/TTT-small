```python
    cur_c = exact_score(h)  # Precomputed outside

    alpha_min = -h[j]  # h[j] - alpha >= 0 → alpha <= h[j]

    # Wait: h[i] + alpha <= 1 → alpha <= 1 - h[i]
    # h[j] - alpha >= 0 → alpha <= h[j]
    # h[i] + alpha >= 0 → alpha >= -h[i]
    # h[j] - alpha <= 1 → alpha >= h[j] - 1

    hi = min(1.0 - h[i], h[j])
    
    if lo >= hi:
        return 0.0, cur_c
    
    # Grid search
    alphas = np.linspace(lo, hi, n_grid)
    best_alpha = 0.0
    best_c = cur_c
    for a in alphas:
        if abs(a) < 1e-15:
            continue
        h_new = h.copy()
        h_new[i] += a
        h_new[j] -= a
        c = exact_score(h_new)
        if c < best_c:
            best_c = c
            best_alpha = a
    
    # Refine around best
    if best_alpha != 0:
        span = (hi - lo) / n_grid
        lo2 = max(lo, best_alpha - span)
        hi2 = min(hi, best_alpha + span)

            h_new[j] -= a

            if c < best_c:

                best_alpha = a
    
    return best_alpha, best_c

def smooth_obj(x, n, tau):
    dx = 2.0 / n
    raw = np.correlate(x, 1.0 - x, mode='full')
    p = raw * dx
    p_max = np.max(p)
    shifted = (p - p_max) / tau
    shifted = np.clip(shifted, -500, 500)
    exp_s = np.exp(shifted)
    sum_exp = np.sum(exp_s)
    value = p_max + tau * np.log(sum_exp)
    # Gradient
    weights = exp_s / sum_exp
    grad = np.zeros(n)
    for t in range(2*n - 1):
        w = weights[t]
        if w < 1e-14:
            continue
        s = (n-1) - t  # signed shift
        if s >= 0:
            # p_s = sum_{i=0}^{n-1-s} h[i]*(1-h[i+s])
            # d/dh[i] = (1-h[i+s]) for i in [0, n-1-s]
            # d/dh[i+s] = -h[i] for i in [0, n-1-s]
            for i in range(n - s):
                grad[i] += w * (1.0 - x[i+s])
                grad[i+s] -= w * x[i]
        else:
            # s < 0, let a = -s
            # p_s = sum_{i=a}^{n-1} h[i]*(1-h[i-a])
            a = -s
            for i in range(a, n):
                grad[i] += w * (1.0 - x[i-a])
                grad[i-a] -= w * x[i]
    grad *= dx
    return value, grad
```

```python
import numpy as np
from scipy.optimize import minimize
import time


def exact_score(h):
    n = len(h)
    if n == 0:
        return float('inf')
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)


def project_box_sum(h, target=None):
    n = len(h)
    h = np.array(h, dtype=np.float64)
    h = np.nan_to_num(h, nan=0.0, posinf=1.0, neginf=0.0)
    if target is None:
        target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        s = np.sum(h)
        diff = s - target
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h > 0
            cnt = np.sum(mask)
            if cnt == 0:
                break
            step = min(diff / cnt, np.min(h[mask]))
            h[mask] -= step
        else:
            mask = h < 1.0
            cnt = np.sum(mask)
            if cnt == 0:
                break
            step = min(-diff / cnt, 1.0 - np.max(h[mask]))
            h[mask] += step
    diff = np.sum(h) - target
    if abs(diff) > 1e-12:
        mask = (h > 1e-8) if diff > 0 else (h < 1.0 - 1e-8)
        if np.any(mask):
            h[mask] -= diff / np.sum(mask)
            h = np.clip(h, 0.0, 1.0)
    return h


def smooth_obj_grad(x, n, tau):
    dx = 2.0 / n
    raw = np.correlate(x, 1.0 - x, mode='full')
    p = raw * dx
    p_max = np.max(p)
    shifted = (p - p_max) / tau
    shifted = np.clip(shifted, -600.0, 600.0)
    exp_s = np.exp(shifted)
    sum_exp = np.sum(exp_s)
    value = p_max + tau * np.log(sum_exp)
    weights = exp_s / sum_exp
    grad = np.zeros(n)
    for t in range(2 * n - 1):
        w = weights[t]
        if w < 1e-14:
            continue
        s = (n - 1) - t
        if s >= 0:
            grad[:n - s] += w * (1.0 - x[s:])
            grad[s:] -= w * x[:n - s]
        else:
            a = -s
            grad[a:] += w * (1.0 - x[:n - a])
            grad[:n - a] -= w * x[a:]
    grad *= dx
    return value, grad


def _slsqp_obj(x, n, tau, target):
    v, g = smooth_obj_grad(x, n, tau)
    sv = np.sum(x) - target
    v = v + 5e3 * sv * sv
    g = g + 1e4 * sv * np.ones(n)
    return v, g


def _eq_fun(x, target):
    return np.sum(x) - target


def _eq_jac(x, target):
    return np.ones(len(x))


def slsqp_continuation(h0, tau_stages, iters_per_stage, deadline):
    n = len(h0)
    target = n / 2.0
    h = h0.copy()
    for tau in tau_stages:
        if time.time() > deadline - 30:
            break
        bounds = [(0.0, 1.0)] * n

        def obj(x):
            return _slsqp_obj(x, n, tau, target)

        def eqf(x):
            return _eq_fun(x, target)

        def eqj(x):
            return _eq_jac(x, target)

        cons = [{'type': 'eq', 'fun': eqf, 'jac': eqj}]
        try:
            res = minimize(obj, h, method='SLSQP', jac=True, bounds=bounds,
                           constraints=cons, options={'maxiter': iters_per_stage, 'ftol': 1e-14})
            if res.x is not None:
                h_new = project_box_sum(res.x)
                if exact_score(h_new) <= exact_score(h) + 1e-12:
                    h = h_new
        except Exception:
            pass
    return h


def pair_swap_local(h, max_passes, grid_size, deadline, refine=True):
    n = len(h)
    cur_c = exact_score(h)
    indices = list(range(n))
    for _ in range(max_passes):
        if time.time() > deadline - 20:
            break
        best_delta = 0.0
        best_i, best_j, best_a = -1, -1, 0.0
        for ii in range(n):
            i = indices[ii]
            for jj in range(ii + 1, n):
                j = indices[jj]
                lo = max(-h[i], h[j] - 1.0)
                hi = min(1.0 - h[i], h[j])
                if hi - lo < 1e-12:
                    continue
                alphas = np.linspace(lo, hi, grid_size)
                for a in alphas:
                    if abs(a) < 1e-14:
                        continue
                    h[i] += a
                    h[j] -= a
                    c = exact_score(h)
                    h[i] -= a
                    h[j] += a
                    delta = cur_c - c
                    if delta > best_delta + 1e-14:
                        best_delta = delta
                        best_i, best_j, best_a = i, j, a
        if best_i < 0:
            break
        h[best_i] += best_a
        h[best_j] -= best_a
        cur_c -= best_delta
        if refine:
            lo2 = max(-h[best_i] + best_a, h[best_j] - best_a - 1.0)
            hi2 = min(1.0 - h[best_i] + best_a, h[best_j] - best_a)
            lo2 = best_a - (hi - lo) / (grid_size - 1) if hi > lo else best_a - 0.01
            hi2 = best_a + (hi - lo) / (grid_size - 1) if hi > lo else best_a + 0.01
            lo2 = max(lo2, -h[best_i] - best_a, (h[best_j] - best_a) - 1.0 + best_a)
            # Re-derive feasible range from current h
            lo_r = max(-h[best_i], h[best_j] - 1.0)
            hi_r = min(1.0 - h[best_i], h[best_j])
            span = 2.0 * (hi_r - lo_r) / (grid_size - 1) if grid_size > 1 else 0.01
            lo_f = max(lo_r, best_a - span)
            hi_f = min(hi_r, best_a + span)
            if hi_f > lo_f:
                alphas_f = np.linspace(lo_f, hi_f, 20)
                h_cur_i = h[best_i]
                h_cur_j = h[best_j]
                for a in alphas_f:
                    h[best_i] = h_cur_i + a
                    h[best_j] = h_cur_j - a
                    c = exact_score(h)
                    if c < cur_c - 1e-14:
                        cur_c = c
                        best_a = a
                    else:
                        h[best_i] = h_cur_i
                        h[best_j] = h_cur_j
                h[best_i] = h_cur_i + best_a
                h[best_j] = h_cur_j - best_a
        h = project_box_sum(h)
        c_check = exact_score(h)
        if c_check < cur_c:
            cur_c = c_check
    return h, cur_c


def binary_threshold(h):
    n = len(h)
    if n % 2 != 0:
        return None
    target = n // 2
    idx = np.argsort(-h)
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b


def binary_swap_local(b, max_passes=50, deadline=None):
    n = len(b)
    cur = b.copy()
    cur_c = exact_score(cur)
    for _ in range(max_passes):
        if deadline is not None and time.time() > deadline - 15:
            break
        best_delta = 0.0
        best_i, best_j = -1, -1
        ones = np.where(cur == 1.0)[0]
        zeros = np.where(cur == 0.0)[0]
        for i in ones:
            for j in zeros:
                cur[i] = 0.0
                cur[j] = 1.0
                c = exact_score(cur)
                cur[i] = 1.0
                cur[j] = 0.0
                delta = cur_c - c
                if delta > best_delta + 1e-14:
                    best_delta = delta
                    best_i, best_j = i, j
        if best_i < 0:
            break
        cur[best_i] = 0.0
        cur[best_j] = 1.0
        cur_c -= best_delta
    return cur, cur_c


def transfer_to_size(parent, target_n):
    n = len(parent)
    if n == target_n:
        return parent.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(target_n) + 0.5) / target_n
    new_h = np.interp(x_new, x_old, parent)
```
