def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    """Project y onto {h : 0<=h_i<=1, sum(h)=n/2}"""
    target = n / 2.0
    lo = np.min(y) - 2.0
    hi = np.max(y) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    h = np.clip(y - t, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-10:
        eligible = np.where((h > 1e-6) & (h < 1.0 - 1e-6))[0]
        if len(eligible) > 0:
            h[eligible] += residual / len(eligible)
            h = np.clip(h, 0.0, 1.0)
    return h

def random_binary(n, rng):
    """Generate random binary vector with exactly n/2 ones."""
    assert n % 2 == 0
    h = np.zeros(n)
    ones = rng.choice(n, size=n//2, replace=False)
    h[ones] = 1.0
    return h

    """Generate structured binary patterns."""
    h = np.zeros(n)
    half = n // 2
    if pattern_type == 0:
        # Alternating blocks
        block_size = max(1, n // (2 * rng.randint(2, 8)))
        for i in range(n):

        # Fixed to have exactly half

        if total > half:

            positions = np.where(h == 1)[0]

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    y = np.asarray(y, dtype=float)
    y = np.nan_to_num(y, nan=0.5, posinf=1.0, neginf=0.0)
    lo = float(np.min(y)) - 2.0
    hi = float(np.max(y)) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(y - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    h = np.clip(y - t, 0.0, 1.0)
    residual = target - float(np.sum(h))
    if abs(residual) > 1e-10:
        eligible = np.where((h > 1e-8) & (h < 1.0 - 1e-8))[0]
        if len(eligible) > 0:
            h[eligible] += residual / len(eligible)
            h = np.clip(h, 0.0, 1.0)
    return h

def make_binary(n, rng):
    h = np.zeros(n)
    h[rng.choice(n, size=n//2, replace=False)] = 1.0
    return h

def binary_anneal(h0, rng, t0=0.08, t_min=0.0005, n_iter=80000):
    h = h0.copy()
    n = len(h)
    cur = compute_c5(h)
    best_h = h.copy()
    best = cur
    
    for it in range(n_iter):
        T = t0 * (t_min / t0) ** (it / max(1, n_iter - 1))
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i], h[j] = 0.0, 1.0
        new_c = compute_c5(h)
        d = new_c - cur
        if d < 0 or (T > 1e-18 and np.log(rng.uniform()) < -d / T):
            cur = new_c
            if new_c < best:
                best = new_c
                best_h = h.copy()
        else:
            h[i], h[j] = 1.0, 0.0
    return best_h, best

def binary_hill_climb(h0, max_passes=5):
    h = h0.copy()
    n = len(h)
    cur = compute_c5(h)
    for _ in range(max_passes):
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        improved = False
        for i in ones:
            for j in zeros:
                h[i], h[j] = 0.0, 1.0
                new_c = compute_c5(h)
                if new_c < cur - 1e-12:
                    cur = new_c
                    improved = True
                    break
                else:
                    h[i], h[j] = 1.0, 0.0
            if improved:
                break
        if not improved:
            break
    return h, cur
