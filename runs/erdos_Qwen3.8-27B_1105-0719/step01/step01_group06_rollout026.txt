import numpy as np
import time
from scipy.optimize import minimize, linprog

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))

def get_profile(h):
    dx = 2.0 / len(h)
    return np.correlate(h, 1.0 - h, mode="full") * dx

def build_lag_matrix(h):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    G = np.zeros((M, n))
    for m in range(M):
        d = m - (n - 1)
        if d == 0:
            G[m, :] = dx * (1.0 - 2.0 * h)
        else:
            c = abs(d)
            L = n - c
            if d > 0:
                G[m, c:c+L] += dx * (1.0 - h[0:L])
                G[m, 0:L]   -= dx * h[c:c+L]
            else:
                G[m, 0:L]   += dx * (1.0 - h[c:c+L])
                G[m, c:c+L] -= dx * h[0:L]
    return G

def repair_feasible(h, target=None):
    h = np.array(h, dtype=np.float64).copy()
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    if target is None:
        target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        order = np.argsort(room)[::-1]
        remaining = diff
        for idx in order:
            if remaining <= 1e-15:
                break
            add = min(remaining, room[idx])
            h[idx] += add
            remaining -= add
    else:
        order = np.argsort(h)[::-1]
        remaining = -diff
        for idx in order:
            if remaining <= 1e-15:
                break
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
    return h

def interpolate_to_dim(vec, new_len):
    n = len(vec)
    x_src = (np.arange(n) + 0.5) * 2.0 / n
    x_tgt = (np.arange(new_len) + 0.5) * 2.0 / new_len
    x_ext = np.concatenate([[0.0], x_src, [2.0]])
    f_ext = np.concatenate([[vec[0]], vec, [vec[-1]]])
    h_new = np.interp(x_tgt, x_ext, f_ext)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

def solve_lp_step(h, rad, time_budget=5.0):
    """Solve the linearized Chebyshev LP to get a descent direction."""
    n = len(h)
    M = 2 * n - 1
    p = get_profile(h)
    G = build_lag_matrix(h)
    
    # LP: min theta
    # s.t. G @ x - theta <= G @ h - p
    #      sum(x) = n/2
    #      max(0, h_i - rad) <= x_i <= min(1, h_i + rad)
    #      theta >= 0
    
    A_ub = np.zeros((M, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = G @ h - p
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    bounds = []
    for i in range(n):
        lo = max(0.0, h[i] - rad)
        hi = min(1.0, h[i] + rad)
        bounds.append((lo, hi))
    bounds.append((0.0, np.inf))  # theta >= 0
    
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0  # minimize theta
    
    try:
        t0 = time.perf_counter()
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x is not None:
            x_trial = res.x[:n]
            if time.perf_counter() - t0 < time_budget:
                return x_trial, res.x[n]
    except Exception:
        pass
    return None, None

def line_search(h, direction, best_score, alphas=None):
    """Line search: find best alpha such that exact_score(h + alpha*direction) < best_score."""
    if alphas is None:
        alphas = [1.0, 0.8, 0.6, 0.4, 0.2, 0.1, 0.05, 0.02, 0.01]
    best_h = h
    best_s = best_score
    for a in alphas:
        h_trial = repair_feasible(h + a * direction)
        s = exact_score(h_trial)
        if s < best_s - 1e-12:
            best_s = s
            best_h = h_trial
    return best_h, best_s

def gradient_descent_step(h, beta_reg=0.0):
    """Single gradient descent step on the max of profile."""
    n = len(h)
    p = get_profile(h)
    score = float(np.max(p))
    # Subgradient: weighted average of active lag gradients
    # Use softmax-like weights for smoothness
    margin = 0.01
    p_shifted = p - score + margin
    w = np.exp(p_shifted / margin)
    w /= w.sum()
    
    G = build_lag_matrix(h)
    grad = G.T @ w  # n-vector
    grad -= grad.mean()  # preserve sum
    
    return grad, score, w

def momentum_descent(h, max_iters=100, time_limit=30, beta_reg=0.0):
    """Momentum-based gradient descent."""
    n = len(h)
    h_cur = h.copy()
    v = np.zeros(n)
    best_h = h_cur.copy()
    best_score = exact_score(h_cur)
    momentum = 0.5
    lr = 0.02
    t0 = time.perf_counter()
    
    for it in range(max_iters):
        if time.perf_counter() - t0 > time_limit:
            break
        grad, score, w = gradient_descent_step(h_cur, beta_reg)
        if not np.all(np.isfinite(grad)):
            break
        v = momentum * v + (1 - momentum) * grad
        v -= v.mean()  # preserve sum constraint
        
        # Adaptive step size
        norm_v = np.linalg.norm(v)
        if norm_v < 1e-14:
            break
        step = lr * lr / (norm_v + 1e-10)
        
        h_trial = repair_feasible(h_cur + step * v)
        s_trial = exact_score(h_trial)
        
        if s_trial < best_score - 1e-12:
            best_score = s_trial
            best_h = h_trial.copy()
            h_cur = h_trial
            momentum = min(0.9, momentum * 1.02)
        else:
            h_cur = h_trial  # still move, but don't update best
            momentum = max(0.3, momentum * 0.95)
            lr = max(0.001, lr * 0.9)
    
    return best_h, best_score

import numpy as np
import time
from scipy.optimize import minimize, linprog

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))

def get_profile(h):
    dx = 2.0 / len(h)
    return np.correlate(h, 1.0 - h, mode="full") * dx

def build_lag_matrix(h):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    G = np.zeros((M, n))
    for m in range(M):
        d = m - (n - 1)
        if d == 0:
            G[m, :] = dx * (1.0 - 2.0 * h)
        else:
            c = abs(d)
            L = n - c
            if d > 0:
                G[m, c:c+L] += dx * (1.0 - h[0:L])
                G[m, 0:L] -= dx * h[c:c+L]
            else:
                G[m, 0:L] += dx * (1.0 - h[c:c+L])
                G[m, c:c+L] -= dx * h[0:L]
    return G

def repair_feasible(h, target=None):
    h = np.array(h, dtype=np.float64).copy()
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    if target is None:
        target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        order = np.argsort(room)[::-1]
        remaining = diff
        for idx in order:
            if remaining <= 1e-15:
                break
            add = min(remaining, room[idx])
            h[idx] += add
            remaining -= add
    else:
        order = np.argsort(h)[::-1]
        remaining = -diff
        for idx in order:
            if remaining <= 1e-15:
                break
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
    return h

def interpolate_to_dim(vec, new_len):
    n = len(vec)
    x_src = (np.arange(n) + 0.5) * 2.0 / n
    x_tgt = (np.arange(new_len) + 0.5) * 2.0 / new_len
    x_ext = np.concatenate([[0.0], x_src, [2.0]])
    f_ext = np.concatenate([[vec[0]], vec, [vec[-1]]])
    h_new = np.interp(x_tgt, x_ext, f_ext)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

def solve_lp_chebyshev(h, rad):
    n = len(h)
    M = 2 * n - 1
    p = get_profile(h)
    G = build_lag_matrix(h)
    
    A_ub = np.zeros((M, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = G @ h - p
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    bounds = []
    for i in range(n):
        bounds.append((max(0.0, h[i] - rad), min(1.0, h[i] + rad)))
    bounds.append((0.0, 2.0))

    c_obj[n] = 1.0

    except Exception:
        pass
    return None

def momentum_descent(h, max_iters=200, time_limit=60.0):

    h_cur = h.copy()
    v = np.zeros(n)
    best_h = h_cur.copy()

    lr = 0.01
    t0 = time.perf_counter()
    
    for it in range(max_iters):
        if time.perf_counter() - t0 > time_limit:
            break
        
        p = get_profile(h_cur)
        score = float(np.max(p))
        
        # Softmax weights for active lags
        margin = max(0.005, 0.01 * score)
        p_shift = p - score + margin
        w = np.exp(p_shift / margin)
        w_sum = w.sum()
        if w_sum < 1e-10:
            break
        w /= w_sum
        
        G = build_lag_matrix(h_cur)
        grad = G.T @ w
        grad -= grad.mean()
        
        v = momentum * v + (1.0 - momentum) * grad
        v -= v.mean()
        
        norm_v = np.linalg.norm(v)
        if norm_v < 1e-14:
            break
        
        step = lr * lr / (norm_v + 1e-10)
        h_trial = repair_feasible(h_cur + step * v)
        s_trial = exact_score(h_trial)
        
        if s_trial < best_score - 1e-13:

            h_cur = h_trial

            momentum = max(0.4, momentum - 0.02)
            lr = max(0.001, lr * 0.95)
        
        if it % 50 == 49 and exact_score(h_cur) > best_score + 0.01:
            h_cur = best_h.copy()
            v = np.zeros(n)
    
    return best_h, best_score

def slsqp_epigraph(h_start, t_start, maxiter=80, n_constraints=0):
    n = len(h_start)
    M = 2 * n - 1
    nc = n_constraints if n_constraints > 0 else M
    
    if nc < M:
        prof = get_profile(h_start)
        active_lags = np.argsort(prof)[::-1][:nc]
    else:
        active_lags = np.arange(M)
    
    def obj(x):
        t = x[n]
        return t
    
    def obj_jac(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g
    
    def cons_fun(x):
        h = x[:n]
        t = x[n]
        eq = h.sum() - n / 2.0
        prof = np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)
        ineq = t - prof[active_lags]
        return np.concatenate([[eq], ineq])
    
    def cons_jac(x):
        h = x[:n]
        G = build_lag_matrix(h)
        Ja = G[active_lags, :]
        J = np.zeros((1 + len(active_lags), n + 1))
        J[0, :n] = 1.0
        J[1:, :n] = -Ja
        J[1:, n] = 1.0
        return J
    
    bounds = [(0.0, 1.0)] * n + [(0.0, t_start + 2.0)]
    x0 = np.concatenate([h_start, [t_start]])
    
    try:
        result = minimize(
            obj, x0, jac=obj_jac,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'ineq', 'fun': cons_fun, 'jac': cons_jac}],
            options={'maxiter': maxiter, 'ftol': 1e-12}
        )
        if result.x is not None and np.all(np.isfinite(result.x)):
            return repair_feasible(result.x[:n])
    except Exception:
        pass
    return None

def symmetrize(h):
    h_sym = 0.5 * (h + h[::-1].copy())
    return repair_feasible(h_sym)

def complement_flip(h):
    return repair_feasible(1.0 - h[::-1].copy())
