def project_feasible(x, target_sum):
    """Project x to {h: 0<=h<=1, sum(h)=target_sum} using bisection."""
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-13:
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time


def project_feasible(x, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-13:
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)


def compute_C(h):
    return np.correlate(h, 1.0 - h, mode='full')


def c5_value(h, dx):
    return compute_C(h).max() * dx


def coord_descent(h0, target_sum, max_sweeps=3000):
    n = len(h0)
    lags = np.arange(-(n - 1), n)
    n_corr = 2 * n - 1
    h = project_feasible(h0.astype(float), target_sum)
    C = compute_C(h)
    best_c = C.max()
    best_h = h.copy()

    for sweep in range(max_sweeps):
        changed = False
        for i in range(n):
            old = h[i]
            idx_p = i + lags
            idx_m = i - lags
            mp = (idx_p >= 0) & (idx_p < n)
            mm = (idx_m >= 0) & (idx_m < n)
            slopes = np.zeros(n_corr)
            slopes[mp] += (1.0 - h[idx_p[mp]])
            if mm.any():
                slopes[mm] += (-h[idx_m[mm]])
            lo_t, hi_t = 0.0, 1.0
            for _ in range(30):
                mid_t = (lo_t + hi_t) * 0.5
                vals = C + (mid_t - old) * slopes
                mv = vals.max()
                active = vals >= mv - 1e-14
                if slopes[active].min() > 1e-12:
                    hi_t = mid_t
                elif slopes[active].max() < -1e-12:
                    lo_t = mid_t
                else:
                    break
            t_opt = (lo_t + hi_t) * 0.5
            if abs(t_opt - old) > 1e-14:
                C += (t_opt - old) * slopes
                h[i] = t_opt
                changed = True
        if changed:
            h = project_feasible(h, target_sum)
            C = compute_C(h)
            c_val = C.max()
            if c_val < best_c:
                best_c = c_val
                best_h = h.copy()
        else:
            break
    return best_h, best_c


def subgradient_descent(h0, dx, target_sum, max_iter=15000):
    n = len(h0)
    lags = np.arange(-(n - 1), n)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_value(h, dx)
    alpha = 0.3
    no_improve = 0
    for it in range(max_iter):
        C = compute_C(h)
        max_c = C.max()
        c5_val = max_c * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 4000:
            break
        threshold = max_c * 0.95
        active = C >= threshold
        grad = np.zeros(n)
        for idx in np.where(active)[0]:
            lag = idx - (n - 1)
            ip = np.arange(n) + lag
            im = np.arange(n) - lag
            mp = (ip >= 0) & (ip < n)
            mm = (im >= 0) & (im < n)
            s = np.zeros(n)
            s[mp] += (1.0 - h[ip[mp]])
            if mm.any():
                s[mm] += (-h[im[mm]])
            grad += s
        grad /= max(1.0, active.sum())
        h_new = project_feasible(h - alpha * grad * dx, target_sum)
        if c5_value(h_new, dx) < c5_val:
            h = h_new
            alpha = min(alpha * 1.01, 0.5)
        else:
            alpha *= 0.97
        if alpha < 1e-12:
            break
    return best_h, best_c5


def slsqp_polish(h0, dx, target_sum, beta, maxiter):
    n = len(h0)
    lags = np.arange(-(n - 1), n)
    n_corr = 2 * n - 1

    def obj_val(x):
        return logsumexp(beta * compute_C(x)) / beta * dx

    def obj_grad(x):
        C = compute_C(x)
        shifted = C - C.max()
        w = np.exp(beta * shifted)
        ws = w.sum()
        if ws < 1e-30:
            w = np.ones(n_corr) / n_corr
        else:
            w /= ws
        grad = np.zeros(n)
        for idx in range(n_corr):
            if w[idx] < 1e-15:
                continue
            lag = idx - (n - 1)
            ip = np.arange(n) + lag
            im = np.arange(n) - lag
            mp = (ip >= 0) & (ip < n)
            mm = (im >= 0) & (im < n)
            s = np.zeros(n)
            s[mp] += (1.0 - x[ip[mp]])
            if mm.any():
                s[mm] += (-x[im[mm]])
            grad += w[idx] * s
        return grad * dx

    def cons_fun(x):
        return x.sum() - target_sum

    def cons_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    try:
        res = minimize(obj_val, h0, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        h_out = project_feasible(res.x, target_sum)
    except Exception:
        h_out = project_feasible(h0, target_sum)
    return h_out, c5_value(h_out, dx)


def make_random_start(n, target_sum, rng):
    h = np.zeros(n)
    idx = rng.permutation(n)
    n_ones = int(target_sum)
    h[idx[:n_ones]] = 1.0
    frac = target_sum - n_ones
    if frac > 1e-10 and n_ones < n:
        h[idx[n_ones]] = frac
    return h


def make_block_start(n, target_sum, n_blocks, rng):
    h = np.zeros(n)
    block_len = n // n_blocks
    for b in range(n_blocks):
        start = b * block_len
        end = start + block_len
        half = end - start
        ones = half // 2
        if ones > 0:
            offset = rng.randint(0, half - ones + 1) if half > ones else 0
            h[start + offset:start + offset + ones] = 1.0
    return project_feasible(h, target_sum)


def remaining_time(t0, budget_s):
    return budget_s - (time.time() - t0)
