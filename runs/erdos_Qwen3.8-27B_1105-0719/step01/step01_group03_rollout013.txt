def exact_score(h):
    n = len(h)
    if n < 2:
        return 0.0
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(profile))

def repair_feasible(vec):
    n = len(vec)
    target = n / 2.0
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-13:
        return h
    # Use iterative water-filling
    if diff > 0:
        # Need to add more mass
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx] + 1e-14)
            if add > 0:
                h[idx] += add
                remaining -= add
            if remaining <= 1e-13:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx] + 1e-14)
            if rem > 0:
                h[idx] -= rem
                remaining -= rem
            if remaining <= 1e-13:
                break
    # Final fix

    if abs(diff2) > 1e-14:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
    h = np.clip(h, 0.0, 1.0)
    return h

    h_new = np.interp(xt, full_xs, full_h)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

def sca_lp_propose(h, radius):
    """Solve the SCA LP to get a proposal direction."""
    n = len(h)
    p, G = build_profile_gradients(h)
    M = 2 * n - 1
    target = n / 2.0
    
    # LP variables: z[0..n-1], t
    # Minimize t
    # Subject to:
    #   G[m] @ z - t <= G[m] @ h - p[m] for all m
    #   sum(z) = target
    #   max(0, h[i]-r) <= z[i] <= min(1, h[i]+r)
    
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0  # minimize t
    
    # Inequality constraints: G[m] @ z - t <= G[m]@h - p[m]
    # => G[m] @ z + [-1] * t <= G[m]@h - p[m]
    A_ub = np.zeros((M, n + 1))
    A_ub[:, :-1] = G  # G @ z
    A_ub[:, -1] = -1.0  # -t
    b_ub = G @ h - p
    
    # Equality: sum(z) = target
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    
    # Bounds
    lo = np.maximum(0.0, h - radius)
    hi = np.minimum(1.0, h + radius)
    bounds = list(zip(lo, hi))
    bounds.append((0.0, np.inf))  # t >= 0
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0 and res.x is not None and np.isfinite(res.x).all():
            return res.x[:n], res.x[-1]
    except Exception:
        pass
    return None, None

import numpy as np
from scipy.optimize import linprog
import time


def exact_score(h):
    n = len(h)
    if n < 2:
        return 0.0
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode="full") * dx
    val = float(np.max(profile))
    if not np.isfinite(val):
        return 1e10
    return val


def repair_feasible(vec):
    n = len(vec)
    target = n / 2.0
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx] + 1e-14)
            if add > 0:
                h[idx] += add
                remaining -= add
            if remaining <= 1e-13:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx] + 1e-14)
            if rem > 0:
                h[idx] -= rem
                remaining -= rem
            if remaining <= 1e-13:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-14:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
    h = np.clip(h, 0.0, 1.0)
    return h


def build_profile_gradients(h):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    p = np.correlate(h, 1.0 - h, mode="full") * dx
    G = np.zeros((M, n))
    for m in range(M):
        s = m - (n - 1)
        if s == 0:
            G[m, :] = dx * (1.0 - 2.0 * h)
        elif s > 0:
            G[m, s:n] += dx * (1.0 - h[0:n - s])
            G[m, 0:n - s] += -dx * h[s:n]
        else:
            a = -s
            G[m, 0:n - a] += dx * (1.0 - h[a:n])
            G[m, a:n] += -dx * h[0:n - a]
    return p, G


def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)


def sca_lp_solve(h, radius):
    n = len(h)
    p, G = build_profile_gradients(h)
    M = 2 * n - 1
    target = n / 2.0

    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0

    A_ub = np.zeros((M, n + 1))
    A_ub[:, :-1] = G
    A_ub[:, -1] = -1.0
    b_ub = G @ h - p

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])

    lo = np.maximum(0.0, h - radius)
    hi = np.minimum(1.0, h + radius)
    bounds = list(zip(lo, hi))
    bounds.append((0.0, 1.0 + p.max()))

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 10})
        if res.status == 0 and res.x is not None and np.isfinite(res.x).all():
            return res.x[:n]
    except Exception:
        pass
    return None


def sca_loop(h_init, deadline):
    h = h_init.copy()
    c_cur = exact_score(h)
    c_best = c_cur
    h_best = h.copy()

    radii = [0.25, 0.15, 0.08, 0.04, 0.02, 0.01, 0.005, 0.0025, 0.0012, 0.0006]

        if time.perf_counter() > deadline:
            break
        for inner in range(max_inner):
            if time.perf_counter() > deadline:
                break
            z_prop = sca_lp_solve(h, radius)
            if z_prop is None:
                break
            z_rep = repair_feasible(z_prop)
            if not np.isfinite(z_rep).all():

            accepted = False
            best_alpha_c = c_cur

                accepted = True
                if c_cur < c_best - 1e-12:

            else:

def subgradient_descent(h_init, deadline, max_iters=200):
    h = h_init.copy()
    n = len(h)
    dx = 2.0 / n
    c_cur = exact_score(h)
    c_best = c_cur
    h_best = h.copy()

    for it in range(max_iters):
        if time.perf_counter() > deadline:
            break
        p, G = build_profile_gradients(h)
        c_max = np.max(p)
        thresh = c_max - 1e-5
        active = np.where(p >= thresh)[0]
        if len(active) == 0:
            active = np.array([np.argmax(p)])

        w = p[active] - thresh + 1e-8
        w = w / w.sum()
        grad = G[active].T @ w
        grad -= grad.mean()
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad /= gnorm

        for alpha in [0.008, 0.004, 0.002, 0.001, 0.0005]:
            h_try = h - alpha * grad
            h_try = repair_feasible(h_try)
            c_try = exact_score(h_try)
            if c_try < c_best - 1e-11:
                h = h_try
                c_cur = c_try
                c_best = c_try
                h_best = h_try.copy()
                break

    return h_best, c_best


def coordinate_descent(h_init, deadline, max_rounds=10):
    h = h_init.copy()
    n = len(h)
    c_cur = exact_score(h)
    c_best = c_cur
    h_best = h.copy()
    target = n / 2.0

    for rnd in range(max_rounds):
        if time.perf_counter() > deadline:
            break
        p, G = build_profile_gradients(h)
        c_max = np.max(p)
        improved = False

        for i in range(n):
            if time.perf_counter() > deadline:
                break
            # Try pairing i with a few other coordinates
            for j in [0, n // 2, (n - 1)]:
                if j == i:
                    continue
                # Optimize delta: h[i] += delta, h[j] -= delta
                # Profile change at lag m: (G[m,i] - G[m,j]) * delta
                dg = G[:, i] - G[:, j]
                # Minimize max_m (p[m] + dg[m]*delta)
                # subject to 0 <= h[i]+delta <= 1 and 0 <= h[j]-delta <= 1
                lo_d = max(-h[i], -(1.0 - (target - h[i])))  # simpler below
                lo_d = max(-h[i], h[j] - 1.0)
                hi_d = min(1.0 - h[i], h[j])

                if lo_d >= hi_d:
                    continue

                # Binary search for optimal delta in [lo_d, hi_d]
                best_d = 0.0
                best_c = np.max(p)
                for _ in range(20):
                    mid_lo = lo_d + (best_d - lo_d) * 0.5
                    mid_hi = hi_d + (best_d - hi_d) * 0.5
                    c_lo = np.max(p + dg * lo_d)
                    c_hi = np.max(p + dg * hi_d)
                    if c_lo < c_hi:
                        hi_d = best_d
                    else:
                        lo_d = best_d
                    best_d = 0.5 * (lo_d + hi_d)
                    best_c = np.max(p + dg * best_d)

                h_try = h.copy()
                h_try[i] += best_d
                h_try[j] -= best_d
                h_try = repair_feasible(h_try)
                c_try = exact_score(h_try)
                if c_try < c_best - 1e-12:
                    h = h_try
                    c_cur = c_try
                    c_best = c_try
                    h_best = h_try.copy()
                    improved = True

    return h_best, c_best
