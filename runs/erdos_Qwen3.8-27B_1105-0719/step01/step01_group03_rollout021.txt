def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def waterfill_project(z, target):
    n = len(z)
    lo = z.min() - 1.0
    hi = z.max() + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + 0.5 * (lo + hi), 0.0, 1.0)
    residual = target - h.sum()
    if abs(residual) > 1e-13:
        if residual > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for i in order:
            if residual > 0:
                add = min(residual, 1.0 - h[i])
                h[i] += add
                residual -= add
            else:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-13:
                break
    return h

def build_G(h, n, dx):
    M = 2 * n - 1
    G = np.zeros((M, n))
    c = n - 1
    for d in range(-(n - 1), n):
        m = c + d
        if d == 0:
            G[m, :] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            G[m, 0:n-d] += dx * (1.0 - h[d:n])
            G[m, d:n] -= dx * h[0:n-d]
        else:
            s = -d
            G[m, s:n] += dx * (1.0 - h[0:n-s])
            G[m, 0:n-s] -= dx * h[s:n]
    return G

def interpolate_h(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    return waterfill_project(h_new, new_n / 2.0)

def momentum_descent(h0, max_iters=300, mu=0.3):

def slsqp_solve(h0, maxiter=250):
    n = len(h0)
    dx = 2.0 / n
    M = 2 * n - 1

    def obj(x):
        return x[-1]

    def obj_grad(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g

    def eq_con(x):
        return x[:-1].sum() - n / 2.0

    def eq_con_grad(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g

    def ineq_con(x):
        h = x[:-1]
        t = x[-1]
        return t - compute_profile(h)

    def ineq_con_grad(x):
        h = x[:-1]
        G = build_G(h, n, dx)
        J = np.zeros((M, n + 1))
        J[:, :-1] = -G
        J[:, -1] = 1.0
        return J

    c0 = exact_score(h0)
    x0 = np.concatenate([h0, [c0 + 1e-3]])
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 1.0)]
    constraints = [
        {'type': 'eq', 'fun': eq_con, 'jac': eq_con_grad},
        {'type': 'ineq', 'fun': ineq_con, 'jac': ineq_con_grad}
    ]
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_grad,
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None and np.isfinite(res.x).all():
            h_new = waterfill_project(res.x[:-1], n / 2.0)
            return h_new
    except Exception:
        pass
    return None
