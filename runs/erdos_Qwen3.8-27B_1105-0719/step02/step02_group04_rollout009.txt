    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

        if np.clip(x - mid, 0.0, 1.0).sum() > target_sum:
            lo = mid

            hi = mid

        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.any():
            h[mask] += diff / mask.sum()

def all_grads(h, n):
    """Compute gradients for all lags. Returns array of shape (2n-1, n)"""
    grads = np.zeros((2*n-1, n))
    for d in range(-(n-1), n):
        j = n - 1 + d
        g = np.zeros(n)
        lo1 = max(0, d)
        hi1 = min(n - 1, n - 1 + d)

            g[lo1:hi1 + 1] += 1.0 - h[lo1 - d:hi1 + 1 - d]
        lo2 = max(0, -d)
        hi2 = min(n - 1, n - 1 - d)
        if lo2 <= hi2:
            g[lo2:hi2 + 1] += -h[lo2 + d:hi2 + 1 + d]
        grads[j] = g
    return grads

    Returns (h_new, c5_new, improved)

    thresh = c5_cur - margin

    n_var = n + 1

        A_ub[i, :n] = grads[i]
        A_ub[i, n] = -1.0

import numpy as np
from scipy.optimize import linprog
import time


def eval_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))


def project_to_target(x, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(x - mid, 0.0, 1.0).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.any():
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)


def grad_at_lag(h, d, n):
    g = np.zeros(n)
    lo1 = max(0, d)
    hi1 = min(n - 1, n - 1 + d)
    if lo1 <= hi1:
        g[lo1:hi1 + 1] += 1.0 - h[lo1 - d:hi1 + 1 - d]
    lo2 = max(0, -d)
    hi2 = min(n - 1, n - 1 - d)
    if lo2 <= hi2:
        g[lo2:hi2 + 1] += -h[lo2 + d:hi2 + 1 + d]
    return g


def lp_step(h, n, eps, use_all_lags=True, margin=0.0):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    corr_scaled = corr * dx
    c5_cur = float(corr_scaled.max())

    if use_all_lags:
        active = np.arange(2 * n - 1)
    else:
        thresh = c5_cur - margin
        active = np.where(corr_scaled >= thresh)[0]

    n_act = len(active)
    n_var = n + 1
    c_obj = np.zeros(n_var)
    c_obj[n] = 1.0

    A_ub = np.zeros((n_act, n_var))
    b_ub = np.zeros(n_act)
    for i, j in enumerate(active):
        d = j - (n - 1)
        g = grad_at_lag(h, d, n)
        A_ub[i, :n] = g
        A_ub[i, n] = -1.0
        b_ub[i] = -corr_scaled[j]

    A_eq = np.zeros((1, n_var))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)

    lo_b = np.maximum(-h, -eps)
    hi_b = np.minimum(1.0 - h, eps)
    bounds = list(zip(lo_b, hi_b))
    bounds.append((0.0, c5_cur + 1.0))

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
    except Exception:
        return h, c5_cur, False, c5_cur

    if res.status != 0:
        return h, c5_cur, False, c5_cur

    delta = res.x[:n]
    t_pred = float(res.x[n])

    if t_pred >= c5_cur - 1e-10:
        return h, c5_cur, False, t_pred

    if np.max(np.abs(delta)) < 1e-10:
        return h, c5_cur, False, t_pred

    for alpha in [1.0, 0.8, 0.6, 0.4, 0.2, 0.1, 0.05]:
        h_new = h + alpha * delta
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_to_target(h_new, n / 2.0)
        c5_new = eval_c5(h_new)
        if c5_new < c5_cur - 1e-12:
            return h_new, c5_new, True, t_pred

    return h, c5_cur, False, t_pred


def lp_search(h, n, max_iters=300, eps_init=0.01, time_limit=30.0):
    t_start = time.time()
    eps = eps_init
    c5_cur = eval_c5(h)
    h_cur = h.copy()
    no_improve_count = 0
    max_no_improve = 30

    for it in range(max_iters):
        if time.time() - t_start > time_limit:
            break
        h_new, c5_new, improved, t_pred = lp_step(h_cur, n, eps)
        if improved:
            h_cur = h_new
            c5_cur = c5_new
            no_improve_count = 0
            eps = min(eps * 1.3, 0.1)
        else:
            no_improve_count += 1
            eps *= 0.6
            if eps < 1e-5:
                break
            if no_improve_count > max_no_improve:
                break

    return h_cur, c5_cur


def interp_to(h, n_new):
    n_old = len(h)
    if n_new == n_old:
        return h.copy()
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h)
    return project_to_target(h_new, n_new / 2.0)


def build_sym(q, n):
    m = n // 2
    h = np.empty(n)
    h[:m] = q
    h[m:] = 1.0 - q[::-1]
    return h


def to_sym_q(h, n):
    m = n // 2
    return 0.5 * (h[:m] + 1.0 - h[n - 1:m - 1:-1])


def subgradient_step(h, n, lr):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = corr.max()
    active = np.where(np.abs(corr - mx) < 1e-6)[0]
    g = np.zeros(n)
    for j in active:
        d = j - (n - 1)
        g += grad_at_lag(h, d, n)
    g /= len(active)
    h_new = h - lr * g
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = project_to_target(h_new, n / 2.0)
    return h_new


def subgradient_search(h, n, max_iters=200, time_limit=10.0):
    t_start = time.time()
    lr = 0.02
    c5_cur = eval_c5(h)
    h_cur = h.copy()
    for it in range(max_iters):
        if time.time() - t_start > time_limit:

        h_new = subgradient_step(h_cur, n, lr)
        c5_new = eval_c5(h_new)
        if c5_new < c5_cur - 1e-12:
            h_cur = h_new
            c5_cur = c5_new
            lr = min(lr * 1.05, 0.1)
        else:
            lr *= 0.7
            if lr < 1e-6:
                break
    return h_cur, c5_cur
