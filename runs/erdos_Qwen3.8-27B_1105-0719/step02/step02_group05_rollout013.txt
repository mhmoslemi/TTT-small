from scipy.special import logsumexp
import time

def exact_score_h(h):
    n = len(h)
    dx = 2.0 / n
    corrs = np.correlate(h, 1.0 - h, mode='full')
    return corrs.max() * dx, corrs

def build_h_from_q(q, n):
    m = n // 2
    if n % 2 == 0:
        return np.concatenate([q, 1.0 - q[::-1]])
    else:
        return np.concatenate([q, [0.5], 1.0 - q[::-1]])

def q_from_h(h, n):
    m = n // 2
    return h[:m].copy()

def grad_corr_d(h, d):
    n = len(h)
    g = np.zeros(n)
    if d >= 0:
        if d < n:
            g[d:] += (1.0 - h[:n - d])
            g[:n - d] += (-h[d:])
    else:
        dd = -d
        if dd < n:
            g[:n - dd] += (1.0 - h[dd:])
            g[dd:] += (-h[:n - dd])
    return g

def q_gradient(h, corrs, n, beta=None):
    """Compute gradient of smooth max (or exact max via active set) w.r.t. q"""
    m = n // 2
    dx = 2.0 / n
    if beta is not None and beta > 0:
        shifted = corrs - corrs.max()
        w = np.exp(beta * shifted)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(len(corrs)) / len(corrs)
        else:
            w /= w_sum
    else:
        # Use top-k active
        k_max = np.argmax(corrs)
        w = np.zeros(len(corrs))
        w[k_max] = 1.0
    
    Gq = np.zeros(m)
    for k in range(len(corrs)):
        if w[k] < 1e-14:
            continue
        d = k - (n - 1)
        G = grad_corr_d(h, d)
        Gq += w[k] * (G[:m] - G[m:][::-1])
    return Gq * dx

def smooth_obj_q(q, n, beta):
    h = build_h_from_q(q, n)
    dx = 2.0 / n
    corrs = np.correlate(h, 1.0 - h, mode='full')
    val = logsumexp(beta * corrs) / beta * dx
    grad = q_gradient(h, corrs, n, beta)

                idxs = rng.choice(pop_size, 2, replace=False)
                a, b = idxs
                F = rng.uniform(0.3, 0.8)
                mutant = pop[best_idx] + F * (pop[a] - pop[b])
            else:
                # current-to-best/2
                idxs = rng.choice(pop_size, 2, replace=False)
                a, b = idxs
                F1 = rng.uniform(0.1, 0.4)
                F2 = rng.uniform(0.1, 0.4)
                mutant = pop[i] + F1 * (pop[best_idx] - pop[i]) + F2 * (pop[a] - pop[b])
            
            # Vectorized binomial crossover
            CR = rng.uniform(0.3, 0.7)
            mask = rng.uniform(size=m) < CR
            j0 = rng.randint(m)
            mask[j0] = True
            trial = np.where(mask, mutant, pop[i])
            trial = np.clip(trial, 0, 1)
            
            # Evaluate
            h_trial = build_h_from_q(trial, n)
            s_trial, _ = exact_score_h(h_trial)
            
            if s_trial <= scores[i]:
                pop[i] = trial
                scores[i] = s_trial
                if s_trial < best_score - 1e-12:
                    best_score = s_trial
                    best_q = trial.copy()
                    stagnant = 0
                else:
                    stagnant += 1
            else:
                stagnant += 1
        
        # Diversity injection if stagnant
        if stagnant > pop_size * 2:
            n_inject = pop_size // 4
            for _ in range(n_inject):
                idx = rng.randint(pop_size)
                if rng.random() < 0.5:
                    pop[idx] = pop[idx] + rng.normal(0, 0.1, m)
                else:
                    pop[idx] = q_init + rng.normal(0, 0.15, m)
                pop[idx] = np.clip(pop[idx], 0, 1)
                h = build_h_from_q(pop[idx], n)
                scores[idx], _ = exact_score_h(h)
            stagnant = 0
    
    return best_q, best_score
