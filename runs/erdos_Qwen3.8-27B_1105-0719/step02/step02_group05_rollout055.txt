    for _ in range(100):

        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid

def subgradient_method(h, n, time_budget, max_iter=200000):
    """Subgradient method with projection."""
    h_work = h.copy()
    target = n / 2.0
    dx = 2.0 / n
    t0 = time.time()
    best_h = h_work.copy()
    best_score = exact_score(h_work)
    
    alpha0 = 0.5
    for it in range(max_iter):
        if time_budget - (time.time() - t0) < 0.5:
            break
        
        R = compute_R(h_work)
        d_star = R.argmax()  # index of max correlation
        lag = d_star - (n - 1)
        
        # Gradient of R(lag) w.r.t. h
        g = np.zeros(n)
        if lag >= 0:
            valid_j = np.arange(max(0, n - lag))
            g[valid_j] += (1.0 - h_work[valid_j + lag])
            valid_j2 = np.arange(lag, n)
            g[valid_j2] -= h_work[valid_j2 - lag]
        else:
            ll = -lag
            valid_j = np.arange(ll, n)
            g[valid_j] += (1.0 - h_work[valid_j - ll])
            valid_j2 = np.arange(0, n - ll)
            g[valid_j2] -= h_work[valid_j2 + ll]
        
        alpha = alpha0 / np.sqrt(it + 1)
        h_new = h_work - alpha * g
        h_new = project_feasible(h_new, target)
        h_work = h_new
        
        if it % 500 == 0:
            sc = compute_R(h_work).max() * dx
            if sc < best_score:
                best_score = sc
                best_h = h_work.copy()
    
    sc = compute_R(h_work).max() * dx
    if sc < best_score:
        best_score = sc
        best_h = h_work.copy()
    
    return best_h, best_score

    """Smooth approximation with L-BFGS via subgradients."""
    from scipy.optimize import minimize
    dx = 2.0 / n
    target = n / 2.0
    t0 = time.time()
    h_work = h.copy()
    best_h = h_work.copy()
    best_score = exact_score(h_work)
    
    for beta in [beta_start, int(beta_start * 3), int(beta_end)]:
        if time_budget - (time.time() - t0) < 20:
            break
        
        def obj(x):
            R = compute_R(x)
            return logsumexp(beta * R) / beta * dx
        
        def grad(x):
            R = compute_R(x)
            shifted = R - R.max()
            w = np.exp(beta * shifted)
            ws = w.sum()
            if ws < 1e-30:
                w = np.ones(2 * n - 1) / (2 * n - 1)
            else:
                w /= ws
            g = np.zeros(n)
            for k in range(2 * n - 1):
                if w[k] < 1e-15:
                    continue
                lag = k - (n - 1)
                if lag >= 0:
                    vj = np.arange(0, n - lag)
                    g[vj] += w[k] * (1.0 - x[vj + lag])
                    vj2 = np.arange(lag, n)
                    g[vj2] += w[k] * (-x[vj2 - lag])
                else:
                    ll = -lag
                    vj = np.arange(ll, n)
                    g[vj] += w[k] * (1.0 - x[vj - ll])
                    vj2 = np.arange(0, n - ll)
                    g[vj2] += w[k] * (-x[vj2 + ll])
            return g * dx
        
        try:
            res = minimize(obj, h_work, jac=grad, method='L-BFGS-B',
                          bounds=[(0.0, 1.0)] * n,
                          options={'maxiter': 200, 'ftol': 1e-16})
            h_new = project_feasible(res.x, target)
            sc = exact_score(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new.copy()
                h_work = h_new
        except Exception:
            pass
    
    return best_h, best_score


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)
    
    def time_left():
        return budget_s - (time.time() - t_start)
    
    # Initialize from parent
    h_parent = np.array(initial_h_values, dtype=float)
    n_parent = len(h_parent)
    h_parent = project_feasible(h_parent, n_parent / 2.0)
    
    best_h = h_parent.copy()
    best_c5 = exact_score(h_parent)
    
    # Strategy 1: Anti-symmetric CD on parent n
    if time_left() > 60:
        h1, c5_1 = anti_sym_cd(best_h, n_parent, min(time_left() - 30, 300))
        if c5_1 < best_c5:
            best_c5 = c5_1
            best_h = h1.copy()
    
    # Strategy 2: Free CD on parent n
    if time_left() > 60:
        h2, c5_2 = free_cd(best_h, n_parent, min(time_left() - 30, 300))
        h2 = project_feasible(h2, n_parent / 2.0)
        c5_2 = exact_score(h2)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()
    
    # Strategy 3: Try different n values with anti-symmetric CD
    n_candidates = [64, 80, 96, 112, n_parent, 140, 160, 192]
    n_candidates = sorted(set(n for n in n_candidates if n % 2 == 0))
    
    for n_try in n_candidates:
        if time_left() < 40:
            break
        if n_try == n_parent:
            h_init = best_h.copy()
        else:
            x_orig = np.linspace(0.0, 2.0, len(best_h))
            x_new = np.linspace(0.0, 2.0, n_try)
            h_init = np.interp(x_new, x_orig, best_h)
            h_init = project_feasible(h_init, n_try / 2.0)
        
        # Anti-symmetric CD
        if time_left() > 40:
            h_as, c5_as = anti_sym_cd(h_init, n_try, min(time_left() - 20, 200))
            if c5_as < best_c5:
                best_c5 = c5_as
                best_h = h_as.copy()
        
        # Free CD
        if time_left() > 40:
            h_fc, c5_fc = free_cd(h_init, n_try, min(time_left() - 20, 200))
            h_fc = project_feasible(h_fc, n_try / 2.0)
            c5_fc = exact_score(h_fc)
            if c5_fc < best_c5:
                best_c5 = c5_fc
                best_h = h_fc.copy()
    
    # Strategy 4: Subgradient on best
    if time_left() > 60:
        h_sg, c5_sg = subgradient_method(best_h, len(best_h), min(time_left() - 30, 200))
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()
    
    # Strategy 5: Smooth minimization on best
    if time_left() > 60:
        h_sm, c5_sm = smooth_minimize(best_h, len(best_h), min(time_left() - 30, 200))
        if c5_sm < best_c5:
            best_c5 = c5_sm
            best_h = h_sm.copy()
    
    # Strategy 6: Random restarts with perturbation
    for restart in range(3):
        if time_left() < 60:
            break
        n_r = len(best_h)
        noise = rng.normal(0, 0.05, n_r)
        h_r = np.clip(best_h + noise, 0, 1)
        h_r = project_feasible(h_r, n_r / 2.0)
        
        if time_left() > 40:
            h_r2, c5_r2 = anti_sym_cd(h_r, n_r, min(time_left() - 20, 150))
            if c5_r2 < best_c5:
                best_c5 = c5_r2
                best_h = h_r2.copy()
        
        if time_left() > 40:
            h_r3, c5_r3 = free_cd(h_r, n_r, min(time_left() - 20, 150))
            h_r3 = project_feasible(h_r3, n_r / 2.0)
            c5_r3 = exact_score(h_r3)
            if c5_r3 < best_c5:
                best_c5 = c5_r3
                best_h = h_r3.copy()
    
    # Strategy 7: Final polish - anti-sym CD + free CD alternating
    for _ in range(2):
        if time_left() < 30:
            break
        n_f = len(best_h)
        h_f1, c5_f1 = anti_sym_cd(best_h, n_f, min(time_left() - 15, 100))
        if c5_f1 < best_c5:
            best_c5 = c5_f1
            best_h = h_f1.copy()
        
        if time_left() > 30:
            h_f2, c5_f2 = free_cd(best_h, n_f, min(time_left() - 15, 100))
            h_f2 = project_feasible(h_f2, n_f / 2.0)
            c5_f2 = exact_score(h_f2)
            if c5_f2 < best_c5:
                best_c5 = c5_f2
                best_h = h_f2.copy()
    
    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    n_out = len(best_h)
    best_h = project_feasible(best_h, n_out / 2.0)
    final_score = exact_score(best_h)
    best_c5 = final_score
    
    return best_h, best_c5, n_out
