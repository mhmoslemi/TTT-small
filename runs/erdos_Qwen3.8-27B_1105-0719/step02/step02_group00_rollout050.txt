def exact_score(h):

def compute_profile(h):

def build_gradient_matrix(h, n, dx):
    M = 2 * n - 1
    G = np.zeros((M, n))
    for m in range(M):
        d = m - (n - 1)
        if d == 0:
            G[m, :] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            G[m, d:n] += dx * (1.0 - h[0:n-d])
            G[m, 0:n-d] += -dx * h[d:n]
        else:
            dp = -d
            G[m, 0:n-dp] += dx * (1.0 - h[dp:n])
            G[m, dp:n] += -dx * h[0:n-dp]
    return G

def repair_feasible(vec):

    profile = compute_profile(h)
    T = np.max(profile)
    spread = T - np.median(profile)
    beam = max(0.003, min(0.015, 0.1 * spread))
    
    # Select active rows
    thresh = T - beam
    active = np.where(profile >= thresh)[0]
    if len(active) == 0:
        active = np.array([np.argmax(profile)])
    if len(active) > num_active:
        top_idx = np.argsort(profile[active])[-num_active:]
        active = active[top_idx]

    G_active = G[active, :]
    p_active = profile[active]
    
    # LP: min t + rho * sum(|h_new - h|)
    # s.t. t >= p_m + G_active[m,:] @ (h_new - h) for each active m
    #      sum(h_new) = n/2
    #      h_i - trust_radius <= h_new_i <= h_i + trust_radius
    #      0 <= h_new_i <= 1

    h_dev = cp.Variable(n)  # for |h_new - h|

    upper = np.minimum(1, h + trust_radius)
    constraints.append(h_var >= lower)
    constraints.append(h_var <= upper)
    constraints.append(cp.sum(h_var) == n / 2.0)
    
    if rho > 0:
        constraints.append(h_dev >= h_var - h)
        constraints.append(h_dev >= h - h_var)
        objective = Minimize(t + rho * cp.sum(h_dev))
    else:
        objective = Minimize(t)
    
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.ECOS, verbose=False)

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx] + 1e-15)
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx] + 1e-15)
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

def build_grad_h(h, n, dx):
    M = 2 * n - 1
    G = np.zeros((M, n))
    for m in range(M):
        d = m - (n - 1)
        if d == 0:
            G[m, :] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            G[m, 0:n-d] += -dx * h[d:n]
            G[m, d:n] += dx * (1.0 - h[0:n-d])
        else:
            e = -d
            G[m, 0:n-e] += dx * (1.0 - h[e:n])
            G[m, e:n] += -dx * h[0:n-e]
    return G

def lp_step_antisym(y, n, num_active=25, trust=0.02, rho=0.01):
    m = n // 2
    dx = 2.0 / n
    h = y_to_h(y, n)
    profile = compute_profile(h)
    T = float(np.max(profile))
    spread = T - float(np.median(profile))
    beam = max(0.002, min(0.012, 0.1 * spread))
    thresh = T - beam
    active = np.where(profile >= thresh)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(profile))])
    if len(active) > num_active:
        top_idx = np.argsort(profile[active])[-num_active:]
        active = active[top_idx]
    
    Gh = build_grad_h(h, n, dx)
    Gy_all = grad_y_from_grad_h(Gh, n)
    Gy_active = Gy_all[active, :]
    p_active = profile[active]
    
    y_var = cp.Variable(m)
    t = cp.Variable()
    dev = cp.Variable(m)
    
    cons = []
    for i in range(len(active)):
        cons.append(t >= p_active[i] + Gy_active[i] @ (y_var - y))
    
    lo = np.maximum(0.0, y - trust)
    hi = np.minimum(1.0, y + trust)
    cons.append(y_var >= lo)
    cons.append(y_var <= hi)
    
    if rho > 0:
        cons.append(dev >= y_var - y)
        cons.append(dev >= y - y_var)
        obj = cp.Minimize(t + rho * cp.sum(dev))
    else:
        obj = cp.Minimize(t)
    
    prob = cp.Problem(obj, cons)
    try:
        prob.solve(solver=cp.CLARABEL, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.SCS, verbose=False)
            except Exception:
                return None
    
    if y_var.value is not None and np.isfinite(y_var.value).all():
        return np.clip(y_var.value, 0.0, 1.0)
    return None

            if d < 24 or d % 2 != 0:
                continue
            h_small = interpolate_to_n(best_h, d)
            c_small = exact_score(h_small)
            y_s = h_to_y(h_small, d)
            y_b, c_b = refine_antisym(y_s, d, t_limit=min(15, time_left() - 5))
            if c_b < c_small:
                h_small = y_to_h(y_b, d)
                c_small = c_b
            if c_small < best_c + 1e-10:
                if c_small < best_c - 1e-12 or d < n_cur:
                    if c_small <= best_c + 1e-11:
                        best_h = h_small
                        best_c = c_small
    
    # Final validation
    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)
    
    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            final_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 2:
            final_c = float(eval_result[0])
    except Exception:
        pass
    
    return best_h, final_c, len(best_h)
