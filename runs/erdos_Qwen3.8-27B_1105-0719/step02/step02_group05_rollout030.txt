def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return (np.correlate(h, 1.0 - h, mode='full').max()) * dx

def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = 0.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-13:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def build_h_from_q(q, n):
    m = n // 2
    if n % 2 == 0:
        return np.concatenate([q, 1.0 - q[::-1]])
    else:
        return np.concatenate([q, [0.5], 1.0 - q[::-1]])

def q_from_h(h, n):
    return h[:n // 2].copy()

def grad_corr_d(h, d, n):
    g = np.zeros(n)
    if d >= 0:
        if d < n:
            g[d:] += (1.0 - h[:n - d])
            g[:n - d] += (-h[d:])
    else:
        dd = -d
        if dd < n:
            g[:n - dd] += (1.0 - h[dd:])
            g[dd:] += (-h[:n - dd])
    return g

def q_grad_for_k(h, k, n):
    """Gradient of C_k w.r.t. q"""
    m = n // 2
    d = k - (n - 1)
    G = grad_corr_d(h, d, n)
    return G[:m] - G[m:][::-1]

    for rnd in range(max_rounds):
        if time.time() - slp_q._start > time_limit:
            break

        corrs = compute_corrs(h)
        cur_max = corrs.max() * dx

        t_var = cp.Variable()

The constraint is: C_k(h(q)) ≤ t

Linearized at q_j:
C_k(h(q_j)) + ∇_q C_k(q_j) @ (q - q_j) ≤ t

Rearranging:
∇_q C_k(q_j) @ q ≤ t - C_k(h(q_j)) + ∇_q C_k(q_j) @ q_j

So: gq @ q_var <= t_var - C_k + gq @ best_q

Yes, that's correct.

But wait, I should double-check the sign. C_k(h) = ∑_{i} h[i+d](1-h[i]) * dx.

∂C_k/∂q[j] = ∂C_k/∂h[i] * ∂h[i]/∂q[j]

For h = [q, 1-q[::-1]] (n even, m = n/2):
- h[i] = q[i] for i in [0, m)
- h[i] = 1 - q[m - 1 - (i - m)] = 1 - q[n - 1 - i] for i in [m, n)

So ∂h[i]/∂q[j] = δ_{i,j} for i < m, and = -δ_{i, n-1-j} for i ≥ m.

∂C_k/∂q[j] = (∂C_k/∂h)[j] - (∂C_k/∂h)[n-1-j]

where (∂C_k/∂h) = grad_corr_d(h, d, n) * dx.

So ∂C_k/∂q[j] = (grad_corr_d(h, d, n)[j] - grad_corr_d(h, d, n)[n-1-j]) * dx

def smooth_max_optimize(h0, n, time_limit):
    """Minimize smooth max using L-BFGS-B"""
    dx = 2.0 / n
    target_sum = n / 2.0
    best_h = h0.copy()
    best_score = exact_score(h0)
    
    for beta in [100, 300, 800, 2000, 5000]:
        if time.time() - smooth_max_optimize._start > time_limit:
            break
        
        def smooth_val(x):
            h = np.clip(x, 0, 1)
            corrs = compute_corrs(h)
            return logsumexp(beta * corrs) / beta * dx
        
        def smooth_grad(x):
            h = np.clip(x, 0, 1)
            corrs = compute_corrs(h)
            shifted = corrs - corrs.max()
            w = np.exp(beta * shifted)
            w_sum = w.sum()
            if w_sum < 1e-30:
                w = np.ones(len(corrs)) / len(corrs)
            else:
                w /= w_sum
            g = np.zeros(n)
            for k in range(len(corrs)):
                if w[k] < 1e-14:
                    continue
                d = k - (n - 1)
                g += w[k] * grad_corr_d(h, d, n)
            return g * dx
        
        # Use projection to handle sum constraint
        # Actually, let's use SLSQP with the sum constraint
        x0 = best_h.copy()
        
        def eq_fun(x):
            return x.sum() - target_sum
        
        def eq_jac(x):
            return np.ones(n)
        
        try:
            res = minimize(smooth_val, x0, jac=smooth_grad, method='SLSQP',
                          bounds=[(0,1)]*n,

            h_new = project_feasible(res.x, target_sum)
            s = exact_score(h_new)
            if s < best_score - 1e-13:
                best_score = s
                best_h = h_new
        except:
            pass
    
    return best_h, best_score

        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-13:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)


def build_h_from_q(q, n):
    m = n // 2
    if n % 2 == 0:
        return np.concatenate([q, 1.0 - q[::-1]])
    else:
        return np.concatenate([q, [0.5], 1.0 - q[::-1]])


def q_from_h(h, n):
    return h[:n // 2].copy()


def grad_corr_d(h, d, n):
    g = np.zeros(n)
    if d >= 0:
        if d < n:
            g[d:] += (1.0 - h[:n - d])
            g[:n - d] += (-h[d:])
    else:
        dd = -d
        if dd < n:
            g[:n - dd] += (1.0 - h[dd:])
            g[dd:] += (-h[:n - dd])
    return g


def q_grad_for_k(h, k, n):
    m = n // 2
    d = k - (n - 1)
    G = grad_corr_d(h, d, n)
    return G[:m] - G[m:][::-1]


def slp_q_space(q_init, n, t_start, time_limit, max_rounds=300, max_active=35):
    m = n // 2
    dx = 2.0 / n
    q = q_init.copy()
    h = build_h_from_q(q, n)
    corrs = compute_corrs(h)
    best_q = q.copy()
    best_c5 = corrs.max() * dx

    active_set = set(np.argsort(corrs)[::-1][:12].tolist())

    for rnd in range(max_rounds):
        if time.time() - t_start > time_limit:
            break

        h = build_h_from_q(q, n)
        corrs = compute_corrs(h)

        # Build and solve LP
        q_var = cp.Variable(m)
        t_var = cp.Variable()
        cons = [q_var >= 0, q_var <= 1]

        for k in sorted(active_set):
            C_k = corrs[k] * dx
            gq = q_grad_for_k(h, k, n) * dx
            cons.append(gq @ q_var <= t_var - C_k + gq @ q)

        prob = cp.Problem(cp.Minimize(t_var), cons)
        try:
            prob.solve(solver=cp.CLARABEL)
        except Exception:
            break

        if q_var.value is None or prob.status not in ('optimal', 'optimal_inaccurate'):
            if len(active_set) > 4:
                ranked = sorted(active_set, key=lambda k: -corrs[k])
                active_set.discard(ranked[-1])
            continue

        q_lp = np.clip(q_var.value, 0.0, 1.0)
        t_sol = t_var.value

        # Backtracking line search
        alpha = 1.0
        accepted = False
        for _ls in range(10):
            q_new = (1.0 - alpha) * q + alpha * q_lp
            h_new = build_h_from_q(q_new, n)
            c5_new = compute_corrs(h_new).max() * dx
            if c5_new <= best_c5 + 1e-12:
                accepted = True
                break
            alpha *= 0.5

        if not accepted:
            q_new = q_lp

        h_new = build_h_from_q(q_new, n)
        corrs_new = compute_corrs(h_new)
        c5_new = corrs_new.max() * dx

        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_q = q_new.copy()

        # Check violations
        viol = corrs_new * dx - t_sol
        n_viol = int((viol > 5e-5).sum())

        if n_viol > 0:
            viol_idx = np.where(viol > 1e-4)[0]
            for idx in viol_idx[:6]:
                active_set.add(int(idx))
            if len(active_set) > max_active:
                ranked = sorted(active_set, key=lambda k: -corrs_new[k])
                active_set = set(ranked[:max_active])
            q = q_new
        else:
            binding = [k for k in active_set if corrs_new[k] * dx > t_sol - 5e-5]
            if 2 <= len(binding) < len(active_set):
                active_set = set(binding)
                q = q_new
            elif abs(c5_new - best_c5) < 1e-11:
                break
            else:
                q = q_new
                # Small perturbation to try escaping
                rng_local = np.random.RandomState(rnd)
                pert = rng_local.randn(m) * 1e-4
                q_try = np.clip(q + pert, 0, 1)
                h_try = build_h_from_q(q_try, n)
                c5_try = compute_corrs(h_try).max() * dx
                if c5_try < c5_new:
                    q = q_try
                    best_c5 = c5_try
                    best_q = q.copy()
                else:
                    break

    return best_q, best_c5


def slp_full_space(h_init, n, t_start, time_limit, max_rounds=300, max_active=35):
    dx = 2.0 / n
    target_sum = n / 2.0
    h = h_init.copy()
    corrs = compute_corrs(h)
    best_h = h.copy()
    best_c5 = corrs.max() * dx

    active_set = set(np.argsort(corrs)[::-1][:12].tolist())

    for rnd in range(max_rounds):
        if time.time() - t_start > time_limit:
            break

        corrs = compute_corrs(h)

        h_var = cp.Variable(n)
        t_var = cp.Variable()
        cons = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target_sum]

        for k in sorted(active_set):
            d = k - (n - 1)
            C_k = corrs[k] * dx
            g = grad_corr_d(h, d, n) * dx
            cons.append(g @ h_var <= t_var - C_k + g @ h)

        prob = cp.Problem(cp.Minimize(t_var), cons)
        try:
            prob.solve(solver=cp.CLARABEL)
        except Exception:
            break

        if h_var.value is None or prob.status not in ('optimal', 'optimal_inaccurate'):
            if len(active_set) > 4:
                ranked = sorted(active_set, key=lambda k: -corrs[k])
                active_set.discard(ranked[-1])
            continue

        h_lp = np.clip(h_var.value, 0.0, 1.0)
        h_lp = project_feasible(h_lp, target_sum)
        t_sol = t_var.value

        # Backtracking
        alpha = 1.0
        accepted = False
        for _ls in range(10):
            h_new = (1.0 - alpha) * h + alpha * h_lp
            h_new = project_feasible(h_new, target_sum)
            c5_new = compute_corrs(h_new).max() * dx
            if c5_new <= best_c5 + 1e-12:
                accepted = True
                break
            alpha *= 0.5

        if not accepted:
            h_new = h_lp

        corrs_new = compute_corrs(h_new)
        c5_new = corrs_new.max() * dx

        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()

        viol = corrs_new * dx - t_sol
        n_viol = int((viol > 5e-5).sum())

        if n_viol > 0:
            viol_idx = np.where(viol > 1e-4)[0]
            for idx in viol_idx[:6]:
                active_set.add(int(idx))
            if len(active_set) > max_active:
                ranked = sorted(active_set, key=lambda k: -corrs_new[k])
                active_set = set(ranked[:max_active])
            h = h_new
        else:
            binding = [k for k in active_set if corrs_new[k] * dx > t_sol - 5e-5]
            if 2 <= len(binding) < len(active_set):
                active_set = set(binding)
                h = h_new
            elif abs(c5_new - best_c5) < 1e-11:
                break
            else:
                h = h_new
                break

    return best_h, best_c5


def smooth_max_slsqp(h_init, n, t_start, time_limit):
    dx = 2.0 / n
    target_sum = n / 2.0
    best_h = h_init.copy()
    best_score = exact_score(h_init)

    for beta in [150, 400, 1000, 2500]:
        if time.time() - t_start > time_limit:
            break

        h0 = best_h.copy()

        def obj(x):
            h = np.clip(x, 0, 1)
            corrs = compute_corrs(h)
            return float(logsumexp(beta * corrs) / beta * dx)

        def grad(x):
            h = np.clip(x, 0, 1)
            corrs = compute_corrs(h)
            shifted = corrs - corrs.max()
            w = np.exp(beta * shifted)
            ws = w.sum()
            if ws < 1e-30:
                w = np.ones(len(corrs)) / len(corrs)
            else:
                w /= ws
            g = np.zeros(n)
            for k in range(len(corrs)):
                if w[k] < 1e-14:
                    continue
                d = k - (n - 1)
                g += w[k] * grad_corr_d(h, d, n)
            return g * dx

        def eq_fun(x):
            return x.sum() - target_sum

        def eq_jac(x):
            return np.ones(n)

        try:
            res = minimize(obj, h0, jac=grad, method='SLSQP',
                           bounds=[(0.0, 1.0)] * n,
                           constraints=[{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}],
                           options={'maxiter': 300, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target_sum)
            s = exact_score(h_new)
            if s < best_score - 1e-13:
                best_score = s
                best_h = h_new
        except Exception:
            pass

    return best_h, best_score


def smooth_max_q(q_init, n, t_start, time_limit):
    m = n // 2
    dx = 2.0 / n
    best_q = q_init.copy()
    h = build_h_from_q(best_q, n)
    best_score = compute_corrs(h).max() * dx

    for beta in [150, 400, 1000]:
        if time.time() - t_start > time_limit:
            break

        def obj_q(q):
            h = build_h_from_q(np.clip(q, 0, 1), n)
            corrs = compute_corrs(h)
            return float(logsumexp(beta * corrs) / beta * dx)

        def grad_q(q):
            h = build_h_from_q(np.clip(q, 0, 1), n)
            corrs = compute_corrs(h)
            shifted = corrs - corrs.max()
            w = np.exp(beta * shifted)
            ws = w.sum()
            if ws < 1e-30:
                w = np.ones(len(corrs)) / len(corrs)
            else:
                w /= ws
            Gq = np.zeros(m)
            for k in range(len(corrs)):
                if w[k] < 1e-14:
                    continue
                Gq += w[k] * q_grad_for_k(h, k, n)
            return Gq * dx

        try:
            res = minimize(obj_q, best_q, jac=grad_q, method='L-BFGS-B',
                           bounds=[(0.0, 1.0)] * m,
                           options={'maxiter': 500, 'ftol': 1e-16})
            q_new = np.clip(res.x, 0, 1)
            h_new = build_h_from_q(q_new, n)
            s = compute_corrs(h_new).max() * dx
            if s < best_score - 1e-13:
                best_score = s
                best_q = q_new.copy()
        except Exception:
            pass

    return best_q, best_score


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    target_sum = n0 / 2.0
    h0 = project_feasible(h0, target_sum)

    best_h = h0.copy()
    best_c5 = exact_score(h0)

    def remaining():
        return budget_s - (time.time() - t_start)

    # Phase 1: SLP in q-space on parent
    if n0 % 2 == 0 and remaining() > 120:
        q0 = q_from_h(h0, n0)
        q1, c5_1 = slp_q_space(q0, n0, t_start, remaining() - 30, max_rounds=250)
        h1 = build_h_from_q(q1, n0)
        h1 = project_feasible(h1, target_sum)
        s1 = exact_score(h1)
        if s1 < best_c5:
            best_c5 = s1
            best_h = h1

    # Phase 2: Smooth max in q-space for polish
    if n0 % 2 == 0 and remaining() > 80:
        q_cur = q_from_h(best_h, n0)
        q2, c5_2 = smooth_max_q(q_cur, n0, t_start, remaining() - 20)
        h2 = build_h_from_q(q2, n0)
        h2 = project_feasible(h2, target_sum)
        s2 = exact_score(h2)
        if s2 < best_c5:
            best_c5 = s2
            best_h = h2

    # Phase 3: SLP full space on parent
    if remaining() > 100:
        h3, c5_3 = slp_full_space(best_h, n0, t_start, remaining() - 30, max_rounds=200)
        s3 = exact_score(h3)
        if s3 < best_c5:
            best_c5 = s3
            best_h = h3

    # Phase 4: Try smaller n in full space
    n_candidates = [48, 50, 52, 54, 56, 58, 60, 62, 64, 66, 68, 70, 72, 74, 76]
    for n_try in n_candidates:
        if remaining() < 70:
            break
        ts = n_try / 2.0
        x_orig = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_try)
        h_interp = np.interp(x_new, x_orig, best_h)
        h_interp = project_feasible(h_interp, ts)

        h_trial, c5_trial = slp_full_space(h_interp, n_try, t_start, remaining() - 40, max_rounds=200)
        s_trial = exact_score(h_trial)
        if s_trial < best_c5 - 1e-10:
            best_c5 = s_trial
            best_h = h_trial

        # Also try q-space if even
        if n_try % 2 == 0 and remaining() > 50:
            h_for_q = h_interp
            h_for_q, _ = slp_full_space(h_interp, n_try, t_start, max(1, remaining() - 40), max_rounds=100)
            q_trial = q_from_h(h_for_q, n_try)
            q_res, c5_q = slp_q_space(q_trial, n_try, t_start, max(1, remaining() - 30), max_rounds=150)
            h_q = build_h_from_q(q_res, n_try)
            h_q = project_feasible(h_q, ts)
            s_q = exact_score(h_q)
            if s_q < best_c5 - 1e-10:
                best_c5 = s_q
                best_h = h_q

    # Phase 5: Smooth max polish on best
    n_best = len(best_h)
    if remaining() > 60:
        h5, c5_5 = smooth_max_slsqp(best_h, n_best, t_start, remaining() - 10)
        s5 = exact_score(h5)
        if s5 < best_c5:
            best_c5 = s5
            best_h = h5

    # Phase 6: Final SLP rounds
    n_final = len(best_h)
    if remaining() > 50:
        h6, c5_6 = slp_full_space(best_h, n_final, t_start, max(1, remaining() - 10), max_rounds=100)
        s6 = exact_score(h6)
        if s6 < best_c5:
            best_c5 = s6
            best_h = h6

    if n_final % 2 == 0 and remaining() > 40:
        q_f = q_from_h(best_h, n_final)
        qf, c5_f = slp_q_space(q_f, n_final, t_start, max(1, remaining() - 10), max_rounds=100)
        hf = build_h_from_q(qf, n_final)
        hf = project_feasible(hf, n_final / 2.0)
        sf = exact_score(hf)
        if sf < best_c5:
            best_c5 = sf
            best_h = hf

    # Phase 7: Try a few random perturbations + reoptimize
    if remaining() > 60:
        n_p = len(best_h)
        for trial in range(5):
            if remaining() < 50:
                break
            noise = rng.randn(n_p) * 0.05
            h_pert = project_feasible(best_h + noise, n_p / 2.0)
            h_rp, c5_rp = slp_full_space(h_pert, n_p, t_start, max(1, remaining() - 30), max_rounds=80)
            s_rp = exact_score(h_rp)
            if s_rp < best_c5 - 1e-10:
                best_c5 = s_rp
                best_h = h_rp

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    n_out = len(best_h)
    best_h = project_feasible(best_h, n_out / 2.0)
    final_score = exact_score(best_h)
    best_c5 = min(best_c5, final_score)

    return best_h, best_c5, n_out
