def minimize_1d_minimax(a, b, t_lo=0.0, t_hi=1.0):
    """Minimize max_d (a[d]*t + b[d]) over t in [t_lo, t_hi]."""

    def f(t):
        vals = a * t + b
        return vals.max()
    
    def active_slopes(t):
        vals = a * t + b
        mx = vals.max()
        mask = np.abs(vals - mx) < 1e-12
        return a[mask]

    f_lo = f(t_lo)
    f_hi = f(t_hi)

    slopes_hi = active_slopes(t_hi)

        return t_lo
    if slopes_hi.max() <= 0:  # all active slopes <= 0, min at t_hi

            t_lo = t_mid
        else:
            t_hi = t_mid

    for sweep in range(max_sweeps):
        corrs = compute_corrs(h)
        max_chg = 0.0
        
        for j in order:
            aj = compute_a(j, h, n)
            old = h[j]
            bj = corrs - aj * old
            t_new = solve_1d(aj, bj)
            if abs(t_new - old) > 1e-15:
                d = t_new - old
                corrs += d * aj
                h[j] = t_new
                ad = abs(d)
                if ad > max_chg:
                    max_chg = ad
        
        h = project_feasible(h, target)
        c5 = exact_score(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        if max_chg < tol:
            break
    
    return best_h, best_c5

def symmetrize(h, n):
    """Project h onto the symmetric subspace h[i] + h[n-1-i] = 1."""
    m = n // 2
    h_sym = np.empty(n)
    h_sym[:m] = 0.5 * (h[:m] + 1.0 - h[n-1:n-1-m:-1])
    h_sym[m:] = 1.0 - h_sym[:m][::-1]
    if n % 2 == 1:
        h_sym[m] = 0.5  # middle element
    return np.clip(h_sym, 0.0, 1.0)
