import numpy as np
from scipy.optimize import minimize, linprog
from scipy.special import logsumexp
import time

def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def c5_of_h(h, dx):
    return compute_corrs(h).max() * dx

def grad_corr_k(h, k, n):
    offset = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -offset)
    i_hi = min(n - 1, n - 1 - offset)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx + offset] += (1.0 - h[idx])
    grad[idx] += (-h[idx + offset])
    return grad

def lp_find_direction(h, n, corrs, M, n_active=30):
    """Find descent direction via LP.
    Returns (direction, z_value) or (None, 0) if no improvement.
    """
    # Get active set: top n_active correlations
    sort_idx = np.argsort(corrs)[::-1][:n_active]
    
    # Compute gradients for active set
    grads = np.array([grad_corr_k(h, k, n) for k in sort_idx])  # (n_active, n)
    
    # LP: minimize -z
    # s.t. grad_k · d + z <= M - corr_k  for each active k
    #      sum(d) = 0
    #      -h[i] <= d[i] <= 1-h[i]
    #      z >= 0
    
    c_obj = np.zeros(n + 1)
    c_obj[-1] = -1.0  # minimize -z = maximize z
    
    A_rows = []
    b_rows = []
    for ii, k in enumerate(sort_idx):
        row = np.zeros(n + 1)
        row[:n] = grads[ii]
        row[n] = 1.0
        A_rows.append(row)
        b_rows.append(M - corrs[k])
    
    A_ub = np.array(A_rows)
    b_ub = np.array(b_rows)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    bounds = [(-h[i], 1.0 - h[i]) for i in range(n)]
    bounds.append((0.0, None))
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            d = res.x[:n]
            z = res.x[n]
            # Normalize direction to have max absolute value = 1
            dmax = np.max(np.abs(d))
            if dmax < 1e-12:
                return None, 0.0
            d = d / dmax
            return d, z / dmax  # scaled z
        else:
            return None, 0.0
    except Exception:
        return None, 0.0

def line_search_exact(h, d, dx, target_sum):
    """Find best step t > 0 along direction d.
    Returns (best_h, best_c5, best_t).
    """
    n = len(h)
    # Max step from box constraints
    t_max = np.inf
    for i in range(n):
        if d[i] > 1e-12:
            t_max = min(t_max, (1.0 - h[i]) / d[i])
        elif d[i] < -1e-12:
            t_max = min(t_max, -h[i] / d[i])
    t_max = min(t_max, 2.0)  # cap
    
    if t_max < 1e-12:
        return h, c5_of_h(h, dx), 0.0
    
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    best_t = 0.0
    
    # Try several step sizes
    for frac in [1.0, 0.9, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05]:
        t = frac * t_max
        h_new = h + t * d
        # Project to feasible (should be close due to box constraints)
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_feasible(h_new, target_sum)
        c5_new = c5_of_h(h_new, dx)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
            best_t = t
    
    return best_h, best_c5, best_t

def local_search_lp(h, dx, target_sum, max_iter=2000, t0=None, budget_s=None):
    """LP-guided local search."""
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    corrs = compute_corrs(h)
    M = corrs.max()
    
    no_improve = 0
    n_active = 30
    
    for it in range(max_iter):
        if t0 is not None and budget_s is not None:
            if time.time() - t0 > budget_s - 10:
                break
        
        d, z = lp_find_direction(h, n, corrs, M, n_active)
        
        if d is None or z < 1e-12:
            # Try with more active constraints
            d, z = lp_find_direction(h, n, corrs, M, n_active=50)
            if d is None or z < 1e-12:
                no_improve += 1
                if no_improve > 200:
                    break
                # Random perturbation
                if no_improve % 10 == 0:
                    rng_local = np.random.RandomState(it)
                    d_rand = rng_local.randn(n)
                    d_rand -= d_rand.mean()  # preserve sum
                    dmax = np.max(np.abs(d_rand))
                    d = d_rand / dmax
                    h_try = h + 0.01 * d
                    h_try = project_feasible(h_try, target_sum)
                    c5_try = c5_of_h(h_try, dx)
                    if c5_try < best_c5:
                        best_c5 = c5_try
                        best_h = h_try.copy()
                    h = h_try
                    corrs = compute_corrs(h)
                    M = corrs.max()
                continue
        
        # Line search
        h_new, c5_new, t = line_search_exact(h, d, dx, target_sum)
        
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
            no_improve = 0
            # Accept and continue
            h = h_new
            corrs = compute_corrs(h)
            M = corrs.max()
        elif c5_new < c5_of_h(h, dx) + 1e-14:
            # Marginal: accept anyway to avoid cycling
            h = h_new
            corrs = compute_corrs(h)
            M = corrs.max()
            no_improve += 1
        else:
            no_improve += 1
        
        if no_improve > 500:
            break
    
    return best_h, best_c5

        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()

def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def c5_of_h(h, dx):
    return compute_corrs(h).max() * dx

def grad_corr_k(h, k, n):
    offset = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -offset)
    i_hi = min(n - 1, n - 1 - offset)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx + offset] += (1.0 - h[idx])
    grad[idx] += (-h[idx + offset])
    return grad

def grad_all(h, n, active_idx):
    grads = np.zeros((len(active_idx), n))
    for ii, k in enumerate(active_idx):
        offset = k - (n - 1)
        i_lo = max(0, -offset)
        i_hi = min(n - 1, n - 1 - offset)
        if i_lo > i_hi:
            continue
        idx = np.arange(i_lo, i_hi + 1)
        grads[ii, idx + offset] += (1.0 - h[idx])
        grads[ii, idx] += (-h[idx + offset])
    return grads

def lp_direction(h, n, corrs, M, n_active):
    sort_idx = np.argsort(corrs)[::-1][:n_active]
    grads = grad_all(h, n, sort_idx)
    c_obj = np.zeros(n + 1)
    c_obj[-1] = -1.0
    A_rows = []
    b_rows = []
    for ii in range(len(sort_idx)):
        row = np.zeros(n + 1)
        row[:n] = grads[ii]
        row[n] = 1.0
        A_rows.append(row)
        b_rows.append(M - corrs[sort_idx[ii]])
    A_ub = np.array(A_rows)
    b_ub = np.array(b_rows)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(-h[i], 1.0 - h[i]) for i in range(n)]
    bounds.append((0.0, None))
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.status == 0:
            d = res.x[:n]
            z = res.x[n]
            dmax = np.max(np.abs(d))
            if dmax < 1e-14:
                return None, 0.0
            d = d / dmax
            return d, z / dmax
    except Exception:

    return None, 0.0

def line_search(h, d, dx, target_sum, n_evals=12):
    n = len(h)
    t_max = np.inf
    for i in range(n):
        if d[i] > 1e-14:
            t_max = min(t_max, (1.0 - h[i]) / d[i])
        elif d[i] < -1e-14:
            t_max = min(t_max, -h[i] / d[i])
    t_max = min(t_max, 3.0)
    if t_max < 1e-14:
        return h, c5_of_h(h, dx), 0.0
    c5_cur = c5_of_h(h, dx)
    best_h = h.copy()
    best_c5 = c5_cur
    best_t = 0.0
    fracs = np.linspace(0.02, 1.0, n_evals)
    for frac in fracs:

def pair_transfer_step(h, dx, target_sum, corrs, M):
    n = len(h)
    sort_idx = np.argsort(corrs)[::-1]
    n_act = min(30, len(sort_idx))
    active_idx = sort_idx[:n_act]
    grads = grad_all(h, n, active_idx)
    w = corrs[active_idx]
    w = w / w.sum()
    wg = w @ grads
    a = int(np.argmax(wg))
    b = int(np.argmin(wg))
    if wg[a] - wg[b] < 1e-14:
        return h, c5_of_h(h, dx)
    eps_max = min(h[a], 1.0 - h[b])
    if eps_max < 1e-12:
        return h, c5_of_h(h, dx)
    d = np.zeros(n)
    d[a] = -1.0
    d[b] = 1.0
    return line_search(h, d, dx, target_sum, n_evals=8)

def slsqp_smooth(h0, dx, target_sum, beta, maxiter=800):
    n = len(h0)
    
    def obj(x):
        corrs = compute_corrs(x)
        return logsumexp(beta * corrs) / beta * dx
    
    def grad_obj(x):
        corrs = compute_corrs(x)
        shifted = corrs - corrs.max()
        w = np.exp(beta * shifted)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(len(corrs)) / len(corrs)
        else:
            w /= w_sum
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] > 1e-13:
                grad += w[k] * grad_corr_k(x, k, n)
        return grad * dx
    
    constr = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum,
               'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0, jac=grad_obj, method='SLSQP',
                       bounds=bounds, constraints=constr,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        h_out = project_feasible(res.x, target_sum)
    except Exception:
        h_out = project_feasible(h0, target_sum)
    return h_out, c5_of_h(h_out, dx)

            h_lp, c5_lp = local_search(best_h, dx_try, ts_try, 500, t0, budget_s, use_lp=True)
            if c5_lp < best_c5:
                best_c5 = c5_lp
                best_h = h_lp
    
    # Phase 5: Final intensive polish
    if time.time() - t0 < budget_s - 10 and best_h is not None:
        n_b = len(best_h)
        dx_b = 2.0 / n_b
        ts_b = n_b / 2.0
        h_lp, c5_lp = local_search(best_h, dx_b, ts_b, 2000, t0, budget_s, use_lp=True)
        if c5_lp < best_c5:
            best_c5 = c5_lp
            best_h = h_lp
        if time.time() - t0 < budget_s - 10:
            h_s, c5_s = slsqp_smooth(best_h, dx_b, ts_b, 2000, 800)
            if c5_s < best_c5:
                best_c5 = c5_s
                best_h = h_s
        if time.time() - t0 < budget_s - 8:
            h_lp2, c5_lp2 = local_search(best_h, dx_b, ts_b, 1000, t0, budget_s, use_lp=True)
            if c5_lp2 < best_c5:
                best_c5 = c5_lp2
                best_h = h_lp2
        if time.time() - t0 < budget_s - 8:
            h_pt, c5_pt = local_search(best_h, dx_b, ts_b, 1000, t0, budget_s, use_lp=False)
            if c5_pt < best_c5:
                best_c5 = c5_pt
                best_h = h_pt
    
    return best_h, best_c5, len(best_h)
