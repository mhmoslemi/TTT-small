def project_feasible(x, target_sum):
    """Project x to [0,1]^n with sum = target_sum."""
    n = len(x)
    x = np.clip(np.nan_to_num(x, nan=0.5), 0.0, 1.0)
    # Bisection for the shift
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(x + mid, 0.0, 1.0).sum()
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    h = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)
    # Fix residual
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            # Need to add mass; add to elements not at 1
            order = np.argsort(h)
            for idx in order:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
                if diff <= 1e-14:
                    break
        else:

                h[idx] -= sub
                diff += sub

    """Compute log-sum-exp approximation of max."""
    m = corrs.max()
    if beta < 1:
        return m + np.log(len(corrs)) / beta
    shifted = corrs - m
    w = np.exp(beta * shifted)
    ws = w.sum()
    if ws < 1e-30:
        return m
    return m + np.log(ws) / beta

def smooth_max_grad_h(corrs, h, beta, n):
    """Gradient of smooth max of correlations w.r.t. h."""
    m = corrs.max()
    shifted = corrs - m
    w = np.exp(beta * shifted)
    ws = w.sum()
    if ws < 1e-30:
        w = np.ones(len(corrs)) / len(corrs)
    else:
        w /= ws
    # Gradient of each correlation w.r.t. h
    # corr[k] = sum_i h[i] * (1-h[i+d]) where d = k - (n-1)
    grad = np.zeros(n)
    for k in range(2 * n - 1):

            continue
        d = k - (n - 1)
        i_lo = max(0, -d)
        i_hi = min(n - 1, n - 1 - d)
        if i_lo > i_hi:

        idx = np.arange(i_lo, i_hi + 1)
        # d(corr[k])/d(h[i]) for i in idx: (1 - h[i+d])
        # d(corr[k])/d(h[i+d]) for i in idx: -h[i]
        grad[idx] += w[k] * (1.0 - h[idx + d])
        grad[idx + d] += -w[k] * h[idx]
    return grad

    g *= dx
    g += 2.0 * pen * dx * (h.sum() - n * 0.5)
    return g

def lbgfs_smooth(h0, n, beta, pen, maxiter):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(full_obj, h0.copy(), jac=full_jac,
                      args=(n, beta, pen), method='L-BFGS-B',
                      bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10})
        return project_feasible(res.x, n * 0.5)
    except Exception:
        return h0.copy()

def sg_step(h, n, alpha):
    c = corr_raw(h)
    mx = c.max()
    act = np.where(c >= mx - 1e-7)[0]
    g = np.zeros(n)
    for k in act:
        g += grad_corr_k(h, k, n)
    g /= max(len(act), 1)
    g *= (2.0 / n)
    hnew = h - alpha * g
    return project_feasible(hnew, n * 0.5)

def transfer_n(hold, nnew):
    no = len(hold)
    xo = (np.arange(no) + 0.5) * 2.0 / no
    xn = (np.arange(nnew) + 0.5) * 2.0 / nnew
    hn = np.interp(xn, xo, hold)
    return project_feasible(hn, nnew * 0.5)
