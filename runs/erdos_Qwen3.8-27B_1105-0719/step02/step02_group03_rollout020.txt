def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def c5_of_h(h, dx):
    return compute_corrs(h).max() * dx


def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-10:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
        else:
            h += diff / n
    return np.clip(h, 0.0, 1.0)


def to_binary(h, n):

def steqp_binary(h0, dx, max_sweeps=80):

        ones = np.where(h == 1.0)[0]

        best_improve = 0.0

For a fixed s:
dcorr(s)/dh[i] = (1-h[i-s])*[0≤i-s<n] - h[i+s]*[0≤i+s<n]

Let me define:
term1[i] = (1-h[i-s]) if 0≤i-s<n, else 0
term2[i] = h[i+s] if 0≤i+s<n, else 0

Then dcorr(s)/dh = term1 - term2.

For term1: shift h by s, subtract from 1, pad with 0.
For term2: shift h by -s, pad with 0.

def grad_corr_s(h, s, n):
    grad = np.zeros(n)
    # term1: (1-h[i-s]) for valid i-s
    lo = max(0, s)
    hi = min(n, n + s)
    if lo < hi:
        grad[lo:hi] += (1.0 - h[lo - s:hi - s])
    # term2: -h[i+s] for valid i+s
    lo = max(0, -s)
    hi = min(n, n - s)
    if lo < hi:
        grad[lo:hi] -= h[lo + s:hi + s]
    return grad
