- Try h_new = h + λ*δ for λ = 1, 0.5, 0.25, ...

Actually, since δ is bounded by -h[j]*α ≤ δ[j] ≤ (1-h[j])*α, we have:
h[j] + λ*δ[j] ≥ h[j] + λ*(-h[j]*α) = h[j]*(1-λ*α)
If α ≤ 1 and λ ≤ 1, this is ≥ 0. Good.
h[j] + λ*δ[j] ≤ h[j] + λ*(1-h[j])*α
If α ≤ 1 and λ ≤ 1, this is ≤ h[j] + (1-h[j]) = 1. Good.

So with α ≤ 1 and λ ∈ [0,1], bounds are automatically satisfied.

Let me now think about the active set selection more carefully:

If I only include the top K correlations, the LP direction might reduce those K but increase others. The line search with exact evaluation catches this, but it's inefficient.

Better approach: Include all correlations that are within some fraction of the max. For example, if max = M, include all k where c_k > M - 0.1*M. This gives a more stable active set.

Or even better: Use an "expanding active set" - start with top K, solve LP, check if any previously excluded constraint is violated, add it, re-solve. This is the classical active-set method for LPs.

But since we're doing line search anyway, the simpler approach of just including a generous active set should work.

Let me also think about what happens at a local optimum. At a local optimum, no feasible direction δ (with sum(δ)=0, within bounds) can reduce all active correlations simultaneously. This means the LP is infeasible (or t ≥ current max). In that case, we need to either:
- Reduce α (smaller trust region) - won't help if truly stuck
- Try a different active set
- Try a completely different starting point

For the simulated annealing part:
- Quantize h to M levels (e.g., M=4 or M=8)
- Assert sum is correct
- Moves: pick two indices i, j, increase h[i] by 1/M and decrease h[j] by 1/M (preserves sum)
- Or: swap values at two positions
- Energy: C5 of the quantized vector
- Accept with probability exp(-ΔE/T)

import numpy as np
from scipy.optimize import linprog

def project_feasible(x, target_sum):
    """Project x onto {0<=h<=1, sum(h)=target_sum} using bisection on shift."""
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-10:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def score_h(h, dx):
    """Compute C5 = dx * max(correlate(h, 1-h, full))"""
    return dx * np.correlate(h, 1.0 - h, mode='full').max()

def grad_corr_all(h, n):
    """Compute gradients of all correlations w.r.t. all h[j].
    Returns array of shape (2n-1, n).
    """
    grads = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        offset = k - (n-1)
        i_lo = max(0, -offset)
        i_hi = min(n-1, n-1-offset)
        if i_lo > i_hi:
            continue
        idx = np.arange(i_lo, i_hi+1)
        grads[k, idx+offset] += (1.0 - h[idx])
        grads[k, idx] += (-h[idx+offset])
    return grads

def active_set_lp(h, dx, target_sum, active_indices, alpha):
    """Solve the active-set LP for a descent direction.
    Returns (delta, predicted_t) or (None, None) if infeasible.
    """
    n = len(h)
    corrs = np.correlate(h, 1.0-h, mode='full')
    
    # Build LP:
    # Variables: delta[0..n-1], t
    # Minimize: t
    # Constraints:
    #   For each active k: sum_j grad_k[j] * delta[j] - t <= -corrs[k]
    #   sum_j delta[j] = 0
    #   Bounds: -alpha*h[j] <= delta[j] <= alpha*(1-h[j])
    
    n_vars = n + 1
    c_obj = np.zeros(n_vars)
    c_obj[-1] = 1.0  # minimize t
    
    # Inequality constraints: A_ub @ x <= b_ub
    # For each active k: sum_j grad_k[j]*delta_j - t <= -corrs[k]
    A_ub = np.zeros((len(active_indices), n_vars))
    b_ub = np.zeros(len(active_indices))
    
    for idx, k in enumerate(active_indices):
        offset = k - (n-1)
        i_lo = max(0, -offset)
        i_hi = min(n-1, n-1-offset)
        if i_lo > i_hi:
            continue
        iidx = np.arange(i_lo, i_hi+1)
        A_ub[idx, iidx+offset] = (1.0 - h[iidx])
        A_ub[idx, iidx] = (-h[iidx+offset])
        A_ub[idx, -1] = -1.0
        b_ub[idx] = -corrs[k]
    
    # Equality: sum(delta) = 0
    A_eq = np.zeros((1, n_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds
    bounds = []
    for j in range(n):
        bounds.append((-alpha*h[j], alpha*(1.0-h[j])))
    bounds.append((corrs.max() - 1e-6, None))  # t >= current_max - small
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 5})
        if res.status != 0:
            return None, None
        delta = res.x[:n]
        predicted_t = res.x[-1]
        return delta, predicted_t
    except Exception:
        return None, None

            best_c5 = current_c5
            best_h = h.copy()

        else:

def quantized_sa(h0, dx, target_sum, M, n_iter, t0, budget_s, rng):
    """Simulated annealing on quantized values."""
    n = len(h0)
    # Quantize to M levels: {0, 1/M, 2/M, ..., 1}
    # a_i = round(h_i * M), then adjust to get exact sum
    a = np.round(h0 * M).astype(int)
    a = np.clip(a, 0, M)
    target_a = int(round(target_sum * M))
    # Adjust sum
    diff = target_a - a.sum()
    if diff > 0:
        # Need to increase some entries
        order = np.argsort(-h0)  # largest values first
        for i in range(min(diff, n)):
            j = order[i]
            if a[j] < M:
                a[j] += 1
    elif diff < 0:
        order = np.argsort(h0)  # smallest values first
        for i in range(min(-diff, n)):
            j = order[i]
            if a[j] > 0:
                a[j] -= 1
    
    h_q = a.astype(float) / M
    current_c5 = score_h(h_q, dx)
    best_h = h_q.copy()
    best_c5 = current_c5
    
    T_start = 0.005
    T_end = 1e-5
    cooling = (T_end / T_start) ** (1.0 / max(1, n_iter-1))
    T = T_start
    
    for it in range(n_iter):
        if time.time() - t0 > budget_s - 30:
            break
        
        # Move: pick two indices, transfer one unit
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        if a[i] == M or a[j] == 0:
            continue
        
        # Proposal: a[i] += 1, a[j] -= 1
        a[i] += 1
        a[j] -= 1
        h_new = a.astype(float) / M
        new_c5 = score_h(h_new, dx)
        delta_e = new_c5 - current_c5
        
        if delta_e < 0 or rng.random() < np.exp(-delta_e / T):
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_new.copy()
        else:
            # Reject
            a[i] -= 1
            a[j] += 1
        
        T *= cooling
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import linprog
import time


def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-10:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)


def score_h(h, dx):
    return dx * np.correlate(h, 1.0 - h, mode='full').max()


def solve_active_lp(h, active_indices, alpha):
    n = len(h)
    corrs = np.correlate(h, 1.0 - h, mode='full')
    n_vars = n + 1
    c_obj = np.zeros(n_vars)
    c_obj[-1] = 1.0
    rows = []
    bvals = []
    for k in active_indices:
        offset = k - (n - 1)
        i_lo = max(0, -offset)
        i_hi = min(n - 1, n - 1 - offset)
        if i_lo > i_hi:
            continue
        idx = np.arange(i_lo, i_hi + 1)
        row = np.zeros(n_vars)
        row[idx + offset] = (1.0 - h[idx])
        row[idx] = (-h[idx + offset])
        row[-1] = -1.0
        rows.append(row)
        bvals.append(-corrs[k])
    if len(rows) == 0:
        return None, None
    A_ub = np.array(rows)
    b_ub = np.array(bvals)
    A_eq = np.zeros((1, n_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        bounds.append((-alpha * h[j], alpha * (1.0 - h[j])))
    bounds.append((corrs.max() - 1e-8, None))
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 3})
        if res.status != 0:
            return None, None
        return res.x[:n], res.x[-1]
    except Exception:
        return None, None


def slp_polish(h0, dx, target_sum, max_iter, alpha_init, margin_frac):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = score_h(h, dx)
    alpha = alpha_init
    no_improve = 0
    for it in range(max_iter):
        corrs = np.correlate(h, 1.0 - h, mode='full')
        current_max = corrs.max()
        current_c5 = current_max * dx
        if current_c5 < best_c5 - 1e-14:
            best_c5 = current_c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 25:
                break
        thr = current_max * (1.0 - margin_frac)
        active_mask = corrs >= thr
        top_k = np.argsort(corrs)[-min(50, 2 * n - 1):]
        active_indices = np.unique(np.concatenate([np.where(active_mask)[0], top_k]))
        delta, _ = solve_active_lp(h, active_indices, alpha)
        if delta is None:
            alpha *= 0.5
            if alpha < 1e-5:
                break
            continue
        improved = False
        for lam in [1.0, 0.8, 0.6, 0.4, 0.25, 0.1, 0.05]:
            h_new = h + lam * delta
            h_new = project_feasible(h_new, target_sum)

def quantized_sa(h0, dx, target_sum, M, n_iter, time_limit_fn, rng):

            j = order[i]
            if a[j] < M:
                a[j] += 1

            j = order[i]
            if a[j] > 0:
                a[j] -= 1

        a[j] -= 1

            a[j] += 1

def transfer_grid(h_src, n_target):
    n_src = len(h_src)
    x_src = np.linspace(0, 2, n_src)
    x_tgt = np.linspace(0, 2, n_target)
    return np.interp(x_tgt, x_src, h_src)


def make_symmetric_seed(n, rng):
    h = np.zeros(n)
    target = n // 2
    idx = rng.choice(n, target, replace=False)
    h[idx] = 1.0
    h_sym = 0.5 * (h + 1.0 - h[::-1].copy())
    return project_feasible(h_sym, float(n) / 2.0)


def make_periodic_seed(n, target_sum, period, fill_frac, rng):
    h = np.zeros(n)
    fill = max(1, int(period * fill_frac))
    for i in range(n):
        h[i] = 1.0 if (i % period) < fill else 0.0
    return project_feasible(h, target_sum)


def make_complement_seed(h):
    return project_feasible(1.0 - h, float(len(h)) / 2.0)


def make_perturbed_seed(h, rng, scale):
    return project_feasible(h + scale * (rng.random(len(h)) - 0.5), float(len(h)) / 2.0)
