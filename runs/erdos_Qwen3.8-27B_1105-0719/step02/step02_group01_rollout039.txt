        #             sum(δ) = 0

def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
        else:
            if diff > 0:
                idx = np.argmin(h)
                h[idx] = min(1.0, h[idx] + diff)
            else:
                idx = np.argmax(h)
                h[idx] = max(0.0, h[idx] + diff)
    return np.clip(h, 0.0, 1.0)


def compute_c5(h, dx):
    corrs = np.correlate(h, 1.0 - h, mode='full')
    return corrs.max() * dx, corrs


def compute_influence(h, n):
    A = np.zeros((2 * n - 1, n))
    js = np.arange(n)
    for k in range(2 * n - 1):
        s = k - (n - 1)
        idx_p = js + s
        mask_p = (idx_p >= 0) & (idx_p < n)
        A[k, :] = np.where(mask_p, 1.0, 0.0)
        if mask_p.any():
            A[k, mask_p] -= h[idx_p[mask_p]]
        idx_m = js - s
        mask_m = (idx_m >= 0) & (idx_m < n)
        if mask_m.any():
            A[k, mask_m] -= h[idx_m[mask_m]]
    return A


def solve_1d_max_lines(c, d, dlo, dhi):
    if dhi - dlo < 1e-14:
        return np.max(c + dlo * d), dlo
    f_lo_val = np.max(c + dlo * d)
    f_hi_val = np.max(c + dhi * d)
    t_lo = min(f_lo_val, f_hi_val) - np.max(np.abs(d)) * (dhi - dlo)
    t_lo = max(0.0, t_lo)
    t_hi = max(f_lo_val, f_hi_val)
    for _ in range(50):
        t_mid = (t_lo + t_hi) / 2.0
        fl = dlo
        fh = dhi
        dp_mask = d > 1e-14
        dn_mask = d < -1e-14
        dz_mask = np.abs(d) <= 1e-14
        if dz_mask.any() and np.any(c[dz_mask] > t_mid + 1e-13):
            t_lo = t_mid
            continue
        if dp_mask.any():
            ub = np.min((t_mid - c[dp_mask]) / d[dp_mask])
            fh = min(fh, ub)
        if dn_mask.any():
            lb = np.max((t_mid - c[dn_mask]) / d[dn_mask])
            fl = max(fl, lb)
        if fl <= fh + 1e-13:
            t_hi = t_mid
        else:
            t_lo = t_mid
    t_opt = t_hi
    fl = dlo
    fh = dhi
    dp_mask = d > 1e-14
    dn_mask = d < -1e-14
    if dp_mask.any():
        fh = min(fh, np.min((t_opt - c[dp_mask]) / d[dp_mask]))
    if dn_mask.any():
        fl = max(fl, np.max((t_opt - c[dn_mask]) / d[dn_mask]))
    fl = max(fl, dlo)
    fh = min(fh, dhi)
    if fl > fh:
        if f_lo_val <= f_hi_val:
            return f_lo_val, dlo
        return f_hi_val, dhi
    best_delta = (fl + fh) / 2.0
    actual_val = np.max(c + best_delta * d)
    if f_lo_val < actual_val:
        return f_lo_val, dlo
    if f_hi_val < actual_val:
        return f_hi_val, dhi
    return actual_val, best_delta


def pair_transfer_cd(h, dx, target, corrs, A, max_inner=200, rng=None, n_random=30):
    n = len(h)
    best_c5 = corrs.max() * dx
    improved = False
    for _ in range(max_inner):
        max_c = corrs.max()
        threshold = max_c - max(1e-8, 0.02 * max_c)
        active = corrs >= threshold
        if not active.any():
            active[corrs.argmax()] = True
        w = np.zeros(2 * n - 1)
        w[active] = 1.0 / active.sum()
        g = A.T @ w
        valid_src = h > 1e-9
        valid_dst = h < 1.0 - 1e-9
        if not valid_src.any() or not valid_dst.any():
            break
        g_src = np.where(valid_src, g, 1e30)
        g_dst = np.where(valid_dst, g, -1e30)
        i_star = np.argmin(g_src)
        j_star = np.argmax(g_dst)
        best_impr = 1e-14
        best_delta = 0.0
        best_pair = None
        candidates = [(i_star, j_star)]
        if rng is not None:
            src_idx = np.where(valid_src)[0]
            dst_idx = np.where(valid_dst)[0]
            for _ in range(n_random):
                ii = src_idx[rng.randint(len(src_idx))]
                jj = dst_idx[rng.randint(len(dst_idx))]
                if ii != jj:
                    candidates.append((ii, jj))
        for (ii, jj) in candidates:
            d = A[:, jj] - A[:, ii]
            dlo = -min(1.0 - h[ii], h[jj])
            dhi = min(h[ii], 1.0 - h[jj])
            if dhi - dlo < 1e-12:
                continue
            val, delta_opt = solve_1d_max_lines(corrs, d, dlo, dhi)
            impr = best_c5 - val * dx
            if impr > best_impr:
                best_impr = impr
                best_delta = delta_opt
                best_pair = (ii, jj)
        if best_pair is None:
            break
        ii, jj = best_pair
        h[ii] -= best_delta
        h[jj] += best_delta
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) > 1e-10:
            h = project_feasible(h, target)
        corrs = np.correlate(h, 1.0 - h, mode='full')
        A = compute_influence(h, n)
        new_c5 = corrs.max() * dx
        if new_c5 < best_c5 - 1e-15:
            best_c5 = new_c5
            improved = True
    return h, best_c5, corrs, A, improved


def lp_polish(h, dx, target, corrs, n_steps=40, lam=0.25):
    n = len(h)
    best_h = h.copy()
    best_c5 = corrs.max() * dx
    for _ in range(n_steps):
        max_c = corrs.max()
        n_active = max(15, int(0.25 * (2 * n - 1)))
        top_idx = np.argsort(corrs)[-n_active:]
        A_infl = compute_influence(h, n)
        n_vars = n + 1
        c_obj = np.zeros(n_vars)
        c_obj[-1] = 1.0
        A_ub = np.zeros((n_active, n_vars))
        b_ub = np.zeros(n_active)
        for idx, k in enumerate(top_idx):
            A_ub[idx, :n] = -A_infl[k, :]
            A_ub[idx, -1] = 1.0
            b_ub[idx] = -corrs[k]
        A_eq = np.zeros((1, n_vars))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)
        bounds = []
        for i in range(n):
            bounds.append((-h[i] * lam, (1.0 - h[i]) * lam))
        bounds.append((0.0, None))
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if res.status != 0:
                break
            delta = res.x[:n]
            h_new = h + delta
            h_new = project_feasible(h_new, target)
            new_c5, new_corrs = compute_c5(h_new, dx)
            if new_c5 < best_c5 - 1e-14:
                best_h = h_new
                best_c5 = new_c5
                h = h_new
                corrs = new_corrs
            else:
                lam *= 0.5
                if lam < 1e-7:
                    break
        except Exception:
            break
    return best_h, best_c5


def slsqp_smooth(h0, dx, target, beta, maxiter=500):
    n = len(h0)

    def obj(x):
        corrs = np.correlate(x, 1.0 - x, mode='full')
        return logsumexp(beta * corrs) / beta * dx

    def grad_obj(x):
        corrs = np.correlate(x, 1.0 - x, mode='full')
        shifted = corrs - corrs.max()
        w = np.exp(beta * shifted)
        ws = w.sum()
        if ws < 1e-30:
            w = np.ones(2 * n - 1) / (2 * n - 1)
        else:
            w /= ws
        A = compute_influence(x, n)
        return (A.T @ w) * dx

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0, jac=grad_obj, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        h_out = project_feasible(res.x, target)
        return h_out, compute_c5(h_out, dx)[0]
    except Exception:
        return h0.copy(), compute_c5(h0, dx)[0]


def subgradient_step(h, dx, target, corrs, A, step_size, beta_w=30):
    n = len(h)
    max_c = corrs.max()
    shifted = corrs - max_c
    w = np.exp(beta_w * shifted)
    ws = w.sum()
    w /= ws
    g = A.T @ w * dx
    gnorm = np.linalg.norm(g)
    if gnorm < 1e-15:
        return h, corrs.max() * dx
    h_new = h - step_size * g
    h_new = project_feasible(h_new, target)
    corrs_new = np.correlate(h_new, 1.0 - h_new, mode='full')
    return h_new, corrs_new.max() * dx


def remaining_time(t0, budget_s):
    return budget_s - (time.time() - t0)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    try:
        h = np.array(initial_h_values, dtype=float)
    except NameError:
        n = 126
        h = np.zeros(n)
        h[:n // 2] = 1.0

    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = project_feasible(h, target)

    best_h = h.copy()
    best_c5, best_corrs = compute_c5(h, dx)
    A = compute_influence(h, n)

    # Phase 1: Intensive pair-transfer coordinate descent
    for round_num in range(80):
        if remaining_time(t0, budget_s) < 60:
            break
        h, c5, corrs, A, improved = pair_transfer_cd(
            h, dx, target, corrs, A, max_inner=300, rng=rng, n_random=40)
        if c5 < best_c5 - 1e-14:
            best_h = h.copy()
            best_c5 = c5
            best_corrs = corrs.copy()
        if not improved:
            if round_num > 5:
                break

    # Phase 2: LP polish
    if remaining_time(t0, budget_s) > 45:
        h_lp, c5_lp = lp_polish(best_h, dx, target, best_corrs, n_steps=40, lam=0.25)
        if c5_lp < best_c5 - 1e-14:
            best_h = h_lp.copy()
            best_c5 = c5_lp
            best_corrs = np.correlate(best_h, 1.0 - best_h, mode='full')

    # Phase 3: SLSQP with increasing beta
    if remaining_time(t0, budget_s) > 35:
        for beta in [150, 400, 800, 1500, 3000]:
            if remaining_time(t0, budget_s) < 15:
                break
            h_s, c5_s = slsqp_smooth(best_h, dx, target, beta, 400)
            if c5_s < best_c5 - 1e-14:
                best_h = h_s.copy()
                best_c5 = c5_s
                best_corrs = np.correlate(best_h, 1.0 - best_h, mode='full')

    # Phase 4: More pair-transfer after SLSQP
    if remaining_time(t0, budget_s) > 50:
        A = compute_influence(best_h, n)
        corrs = np.correlate(best_h, 1.0 - best_h, mode='full')
        for round_num in range(60):
            if remaining_time(t0, budget_s) < 25:
                break
            h, c5, corrs, A, improved = pair_transfer_cd(
                best_h, dx, target, best_corrs, A, max_inner=250, rng=rng, n_random=50)
            if c5 < best_c5 - 1e-14:
                best_h = h.copy()
                best_c5 = c5
                best_corrs = corrs.copy()
            if not improved:
                break

    # Phase 5: Subgradient + pair transfer interleaved
    if remaining_time(t0, budget_s) > 40:
        step = 0.5 / np.sqrt(n)
        h_cur = best_h.copy()
        c5_cur = best_c5
        for it in range(3000):
            if remaining_time(t0, budget_s) < 20:
                break
            if it % 20 == 0:
                A_cur = compute_influence(h_cur, n)
                corrs_cur = np.correlate(h_cur, 1.0 - h_cur, mode='full')
                h_cur, c5_sg, corrs_cur, A_cur, _ = pair_transfer_cd(
                    h_cur, dx, target, corrs_cur, A_cur, max_inner=30, rng=rng, n_random=10)
                if c5_sg < best_c5 - 1e-14:
                    best_h = h_cur.copy()
                    best_c5 = c5_sg
                    best_corrs = corrs_cur.copy()
            else:
                A_cur = compute_influence(h_cur, n)
                corrs_cur = np.correlate(h_cur, 1.0 - h_cur, mode='full')
                h_cur, c5_cur = subgradient_step(h_cur, dx, target, corrs_cur, A_cur, step)
                if c5_cur < best_c5 - 1e-14:
                    best_h = h_cur.copy()
                    best_c5 = c5_cur
            step *= 0.9995
            step = max(step, 1e-6)

    # Phase 6: Try different n values
    n_candidates = [100, 110, 120, 130, 140, 150, 160]
    for n_try in n_candidates:
        if remaining_time(t0, budget_s) < 35:
            break
        n_try = n_try // 2 * 2
        if n_try == len(best_h):
            continue
        if n_try < 4:
            continue
        dx_try = 2.0 / n_try
        target_try = n_try / 2.0
        x_orig = np.linspace(0.0, 2.0, len(best_h))
        x_new = np.linspace(0.0, 2.0, n_try)
        h_interp = np.interp(x_new, x_orig, best_h)
        h_interp = project_feasible(h_interp, target_try)
        A_try = compute_influence(h_interp, n_try)
        corrs_try = np.correlate(h_interp, 1.0 - h_interp, mode='full')
        h_opt, c5_opt, corrs_opt, A_opt, _ = pair_transfer_cd(
            h_interp, dx_try, target_try, corrs_try, A_try, max_inner=200, rng=rng, n_random=30)
        if c5_opt < best_c5 - 1e-14:
            best_h = h_opt.copy()
            best_c5 = c5_opt
            best_corrs = corrs_opt.copy()
        if remaining_time(t0, budget_s) > 20:
            h_lp2, c5_lp2 = lp_polish(best_h, 2.0 / len(best_h), len(best_h) / 2.0,
                                      best_corrs, n_steps=20, lam=0.2)
            if c5_lp2 < best_c5 - 1e-14:
                best_h = h_lp2.copy()
                best_c5 = c5_lp2
                best_corrs = np.correlate(best_h, 1.0 - best_h, mode='full')

    # Phase 7: Final intensive polish on best solution
    if remaining_time(t0, budget_s) > 20:
        n_f = len(best_h)
        dx_f = 2.0 / n_f
        target_f = n_f / 2.0
        A_f = compute_influence(best_h, n_f)
        corrs_f = np.correlate(best_h, 1.0 - best_h, mode='full')
        h_f, c5_f, corrs_f, A_f, _ = pair_transfer_cd(
            best_h, dx_f, target_f, corrs_f, A_f, max_inner=400, rng=rng, n_random=60)
        if c5_f < best_c5 - 1e-14:
            best_h = h_f.copy()
            best_c5 = c5_f
            best_corrs = corrs_f

    if remaining_time(t0, budget_s) > 15:
        n_f = len(best_h)
        dx_f = 2.0 / n_f
        target_f = n_f / 2.0
        h_lp3, c5_lp3 = lp_polish(best_h, dx_f, target_f, best_corrs, n_steps=30, lam=0.15)
        if c5_lp3 < best_c5 - 1e-14:
            best_h = h_lp3.copy()
            best_c5 = c5_lp3
            best_corrs = np.correlate(best_h, 1.0 - best_h, mode='full')

    if remaining_time(t0, budget_s) > 10:
        n_f = len(best_h)
        dx_f = 2.0 / n_f
        target_f = n_f / 2.0
        A_f = compute_influence(best_h, n_f)
        corrs_f = np.correlate(best_h, 1.0 - best_h, mode='full')
        h_f2, c5_f2, corrs_f2, A_f2, _ = pair_transfer_cd(
            best_h, dx_f, target_f, corrs_f, A_f, max_inner=300, rng=rng, n_random=80)
        if c5_f2 < best_c5 - 1e-14:
            best_h = h_f2.copy()
            best_c5 = c5_f2

    return best_h, best_c5, len(best_h)
