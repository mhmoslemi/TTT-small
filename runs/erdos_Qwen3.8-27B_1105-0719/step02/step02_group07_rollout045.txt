    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx


def make_sym_h(q, n):
    m = n // 2
    q = np.clip(q, 0.0, 1.0)
    h = np.empty(n)
    h[:m] = q
    h[m:] = 1.0 - q[::-1]
    return h


def make_q_from_h(h, n):
    m = n // 2
    q = 0.5 * (h[:m] + 1.0 - h[n - 1: n - 1 - m: -1])
    return np.clip(q, 0.0, 1.0)


def subgrad_lag(h, k_idx):
    n = len(h)
    dx = 2.0 / n
    d = k_idx - (n - 1)
    g = np.zeros(n)
    i_lo = max(0, -d)
    i_hi = min(n - 1, n - 1 - d)
    if i_lo > i_hi:
        return g
    idx = np.arange(i_lo, i_hi + 1)
    g[idx] += (1.0 - h[idx + d])
    g[idx + d] += (-h[idx])
    return g * dx


def smooth_obj_sym(x, n, beta, active=None):
    h = make_sym_h(x, n)
    corrs = corr_vals(h)
    if active is not None:
        corrs = corrs[active]
    return logsumexp(beta * corrs) / beta


def smooth_grad_sym(x, n, beta, active=None):
    m = n // 2
    h = make_sym_h(x, n)
    corrs = corr_vals(h)
    if active is not None:
        sel = corrs[active]
        shifted = sel - sel.max()
        w = np.exp(beta * shifted)
        ws = w.sum()
        if ws < 1e-30:
            w = np.ones(len(active)) / len(active)
        else:
            w /= ws
    else:
        shifted = corrs - corrs.max()
        w = np.exp(beta * shifted)
        ws = w.sum()
        if ws < 1e-30:
            w = np.ones(2 * n - 1) / (2 * n - 1)
        else:
            w /= ws
        idx = np.arange(2 * n - 1)
    if active is not None:
        idx = active
    else:
        idx = np.arange(2 * n - 1)
    dx = 2.0 / n
    grad = np.zeros(n)
    for j, k_idx in enumerate(idx):
        wj = w[j]
        if wj < 1e-15:
            continue
        d = k_idx - (n - 1)
        i_lo = max(0, -d)
        i_hi = min(n - 1, n - 1 - d)
        if i_lo > i_hi:
            continue
        idx2 = np.arange(i_lo, i_hi + 1)
        grad[idx2] += wj * (1.0 - h[idx2 + d])
        grad[idx2 + d] += wj * (-h[idx2])
    grad *= dx
    return grad[:m] - grad[n - 1: n - 1 - m: -1]


def smooth_optimize_sym(q0, n, beta, maxiter=400, active=None):
    m = n // 2
    try:
        res = minimize(smooth_obj_sym, q0,
                       jac=lambda x: smooth_grad_sym(x, n, beta, active),
                       method='SLSQP', bounds=[(0.0, 1.0)] * m,
                       args=(), options={'maxiter': maxiter, 'ftol': 1e-16})
        h_new = make_sym_h(res.x, n)
        c5 = exact_score(h_new)
        if res.fun < 1e10:
            return res.x, c5
    except Exception:
        pass
    return None, None


class EpiSolver:
    def __init__(self, n, active_lags):
        self.n = n
        self.m = n // 2
        self.dx = 2.0 / n
        self.active_lags = np.array(active_lags, dtype=int)
        self.L = len(self.active_lags)

    def fun(self, x):
        return x[0]

    def jac(self, x):
        g = np.zeros(self.m + 1)
        g[0] = 1.0
        return g

    def con_fun(self, x):
        t = x[0]
        q = x[1:]
        h = make_sym_h(q, self.n)
        corrs = corr_vals(h)
        return t - corrs[self.active_lags]

    def con_jac(self, x):
        q = x[1:]
        h = make_sym_h(q, self.n)
        n = self.n
        m = self.m
        dx = self.dx
        L = self.L
        J = np.zeros((L, m + 1))
        J[:, 0] = 1.0
        for i, lag in enumerate(self.active_lags):
            d = lag - (n - 1)
            i_lo = max(0, -d)
            i_hi = min(n - 1, n - 1 - d)
            G = np.zeros(n)
            if i_lo <= i_hi:
                idx = np.arange(i_lo, i_hi + 1)
                G[idx] += (1.0 - h[idx + d])
                G[idx + d] += (-h[idx])
            G *= dx
            J[i, 1:] = -(G[:m] - G[n - 1: n - 1 - m: -1])
        return J


def solve_epi_sym(q0, n, active_lags, t_init, maxiter=400, tol=1e-14):
    m = n // 2
    solver = EpiSolver(n, active_lags)
    x0 = np.concatenate([[t_init], q0])
    cons = [{'type': 'ineq', 'fun': solver.con_fun, 'jac': solver.con_jac}]
    bounds = [(-1.0, 2.0)] + [(0.0, 1.0)] * m
    try:
        res = minimize(solver.fun, x0, jac=solver.jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': tol})
        h_new = make_sym_h(res.x[1:], n)
        c5 = exact_score(h_new)
        if abs(h_new.sum() - n / 2.0) > 1e-8:
            return None, None
        if c5 > t_init + 0.01:
            return None, None
        return res.x[1:], c5
    except Exception:
        return None, None


def subgradient_full(h0, t_start, t_end, K_init=5, K_max=40):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_score = exact_score(h)
    total_iters = max(1, int(t_end - t_start))
    for it in range(total_iters):
        t_frac = it / max(total_iters - 1, 1)
        alpha = 0.5 / (1.0 + 0.05 * it) if t_frac < 0.5 else 0.1 / (1.0 + 0.02 * (it - total_iters // 2))
        K = min(K_init + int(t_frac * (K_max - K_init)), 2 * n - 1)
        corrs = corr_vals(h)
        top_idx = np.argsort(corrs)[::-1][:K]
        g = np.zeros(n)
        for ki in top_idx:
            g += subgrad_lag(h, ki)
        g /= K
        h_new = project_feasible(h - alpha * g, n / 2.0)
        s = exact_score(h_new)
        if s < best_score:
            best_score = s
            best_h = h_new.copy()
        h = h_new
        if it % 5000 == 0 and it > 0:
            pass
    return best_h, best_score


def active_set_refine(q, n, max_rounds=50, tol=1e-10):
    m = n // 2
    best_q = q.copy()
    best_c5 = exact_score(make_sym_h(q, n))
    for rnd in range(max_rounds):
        h = make_sym_h(best_q, n)
        corrs = corr_vals(h)
        max_val = corrs.max()
        n_act = min(30 + 10 * rnd, 2 * n - 1)
        active = np.argsort(corrs)[::-1][:n_act]
        active = sorted(active.tolist())
        q_new, c5_new = solve_epi_sym(best_q, n, active, t_init=max_val * 1.0001,
                                       maxiter=300)
        if q_new is not None and c5_new < best_c5 - tol:
            best_c5 = c5_new
            best_q = q_new.copy()
        else:
            if rnd > 5:
                break
        if rnd % 10 == 0:
            pass
    return best_q, best_c5


def full_smooth_optimize(h0, n, beta, maxiter=300):
    target_sum = n / 2.0
    dx = 2.0 / n

    def obj(x):
        h = np.clip(x, 0.0, 1.0)
        corrs = corr_vals(h)
        return logsumexp(beta * corrs) / beta

    def obj_grad(x):
        h = np.clip(x, 0.0, 1.0)
        corrs = corr_vals(h)
        shifted = corrs - corrs.max()
        w = np.exp(beta * shifted)
        ws = w.sum()
        if ws < 1e-30:
            w = np.ones(2 * n - 1) / (2 * n - 1)
        else:
            w /= ws
        grad = np.zeros(n)
        for k_idx in range(2 * n - 1):
            wk = w[k_idx]
            if wk < 1e-15:
                continue
            d = k_idx - (n - 1)
            i_lo = max(0, -d)
            i_hi = min(n - 1, n - 1 - d)
            if i_lo > i_hi:
                continue
            idx = np.arange(i_lo, i_hi + 1)
            grad[idx] += wk * (1.0 - h[idx + d])
            grad[idx + d] += wk * (-h[idx])
        return grad * dx

    try:
        res = minimize(obj, h0, jac=obj_grad, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        h_out = project_feasible(res.x, target_sum)
        c5 = exact_score(h_out)
        return h_out, c5
    except Exception:
        return h0, exact_score(h0)


def local_search_sym(q, n, rng, n_trials=30, beta=200):
    m = n // 2
    best_q = q.copy()
    best_c5 = exact_score(make_sym_h(q, n))
    for trial in range(n_trials):
        scale = 0.02 + 0.15 * (trial / max(n_trials - 1, 1))
        q_pert = best_q + scale * rng.randn(m)
        q_pert = np.clip(q_pert, 0.0, 1.0)
        q_new, c5_new = smooth_optimize_sym(q_pert, n, beta, maxiter=150)
        if q_new is not None and c5_new < best_c5 - 1e-10:
            best_c5 = c5_new
            best_q = q_new.copy()
    return best_q, best_c5


def trust_constr_epi(q0, n, active_lags, t_init, maxiter=200):
    m = n // 2
    solver = EpiSolver(n, active_lags)
    x0 = np.concatenate([[t_init], q0])
    cons = [{'type': 'ineq', 'fun': solver.con_fun, 'jac': solver.con_jac}]
    bounds = [(-1.0, 2.0)] + [(0.0, 1.0)] * m
    try:
        res = minimize(solver.fun, x0, jac=solver.jac, method='trust-constr',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'gtol': 1e-10})
        h_new = make_sym_h(res.x[1:], n)
        c5 = exact_score(h_new)
        if abs(h_new.sum() - n / 2.0) > 1e-8:
            return None, None
        if c5 > t_init + 0.01:
            return None, None
        return res.x[1:], c5
    except Exception:
        return None, None


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def remaining():
        return budget_s - (time.time() - t0)

    try:
        h_parent = np.array(initial_h_values, dtype=float)
    except NameError:
        h_parent = 0.5 * np.ones(126)

    n = len(h_parent)
    if n % 2 != 0:
        h_parent = h_parent[:n - 1]
        n = len(h_parent)

    h_best = project_feasible(h_parent, n / 2.0)
    c5_best = exact_score(h_best)
    m = n // 2
    q_best = make_q_from_h(h_best, n)
    print(f"Initial: n={n}, C5={c5_best:.7f}")

    # Phase 1: Quick smooth optimization
    for beta in [100, 300, 800]:
        if remaining() < 20:
            break
        q_new, c5_new = smooth_optimize_sym(q_best, n, beta, maxiter=300)
        if q_new is not None and c5_new < c5_best - 1e-10:
            c5_best = c5_new
            q_best = q_new
            h_best = make_sym_h(q_best, n)
            print(f"  smooth beta={beta}: C5={c5_best:.7f}")

    # Phase 2: Active-set epigraph (SLSQP)
    for rnd in range(15):
        if remaining() < 20:
            break
        h_cur = make_sym_h(q_best, n)
        corrs = corr_vals(h_cur)
        max_c = corrs.max()
        n_act = min(25 + 8 * rnd, 2 * n - 1)
        active = np.argsort(corrs)[::-1][:n_act].tolist()
        active = sorted(active)
        q_new, c5_new = solve_epi_sym(q_best, n, active, t_init=max_c * 1.0001,
                                       maxiter=350)
        if q_new is not None and c5_new < c5_best - 1e-10:
            c5_best = c5_new
            q_best = q_new
            h_best = make_sym_h(q_best, n)
            print(f"  epi slsqp rnd={rnd} (L={n_act}): C5={c5_best:.7f}")
        elif rnd >= 10:
            break

    # Phase 3: Trust-constr epigraph
    if remaining() > 30:
        h_cur = make_sym_h(q_best, n)
        corrs = corr_vals(h_cur)
        active = np.argsort(corrs)[::-1][:50].tolist()
        active = sorted(active)
        q_new, c5_new = trust_constr_epi(q_best, n, active,
                                          t_init=corrs.max() * 1.0001, maxiter=500)
        if q_new is not None and c5_new < c5_best - 1e-10:
            c5_best = c5_new
            q_best = q_new
            h_best = make_sym_h(q_best, n)
            print(f"  trust-constr: C5={c5_best:.7f}")

    # Phase 4: Subgradient method on full space
    if remaining() > 60:
        h_sg = h_best.copy()
        h_sg2, c5_sg = subgradient_full(h_sg, t_start=0, t_end=50000, K_init=5, K_max=50)
        if c5_sg < c5_best - 1e-10:
            c5_best = c5_sg
            h_best = h_sg2
            q_best = make_q_from_h(h_best, n)
            print(f"  subgrad: C5={c5_best:.7f}")
        elif remaining() > 30:
            h_sg3, c5_sg3 = subgradient_full(h_best, t_start=0, t_end=30000, K_init=3, K_max=30)
            if c5_sg3 < c5_best - 1e-10:
                c5_best = c5_sg3
                h_best = h_sg3
                q_best = make_q_from_h(h_best, n)
                print(f"  subgrad2: C5={c5_best:.7f}")

    # Phase 5: Local search in symmetric space
    if remaining() > 40:
        q_new, c5_new = local_search_sym(q_best, n, rng, n_trials=40, beta=300)
        if c5_new < c5_best - 1e-10:
            c5_best = c5_new
            q_best = q_new
            h_best = make_sym_h(q_best, n)
            print(f"  local search: C5={c5_best:.7f}")

    # Phase 6: Another round of active-set after local search
    if remaining() > 25:
        h_cur = make_sym_h(q_best, n)
        corrs = corr_vals(h_cur)
        max_c = corrs.max()
        active = np.argsort(corrs)[::-1][:45].tolist()
        active = sorted(active)
        q_new, c5_new = solve_epi_sym(q_best, n, active, t_init=max_c * 1.0001,
                                       maxiter=400)
        if q_new is not None and c5_new < c5_best - 1e-10:
            c5_best = c5_new
            q_best = q_new
            h_best = make_sym_h(q_best, n)
            print(f"  epi phase6: C5={c5_best:.7f}")

    # Phase 7: Full-space smooth optimization
    if remaining() > 30:
        for beta in [500, 1000]:
            if remaining() < 15:
                break
            h_new, c5_new = full_smooth_optimize(h_best, n, beta, maxiter=300)
            if c5_new < c5_best - 1e-10:
                c5_best = c5_new
                h_best = h_new
                q_best = make_q_from_h(h_best, n)
                print(f"  full smooth beta={beta}: C5={c5_best:.7f}")

    # Phase 8: Try nearby n values
    if remaining() > 50:
        for delta in [-4, 4, -6, 6, -8, 8]:
            if remaining() < 25:
                break
            n_try = n + delta
            if n_try < 60 or n_try % 2 != 0:
                continue
            x_old = np.linspace(0, 2, n, endpoint=False)
            x_new_idx = np.linspace(0, 2, n_try, endpoint=False)
            h_interp = np.interp(x_new_idx, x_old, h_best)
            h_interp = project_feasible(h_interp, n_try / 2.0)
            m_try = n_try // 2
            q_try = make_q_from_h(h_interp, n_try)
            q_new, c5_new = smooth_optimize_sym(q_try, n_try, 300, maxiter=250)
            if q_new is not None:
                q_new, c5_new = smooth_optimize_sym(q_new, n_try, 800, maxiter=250)
            if q_new is not None and c5_new < c5_best - 1e-9:
                c5_best = c5_new
                n = n_try
                m = m_try
                q_best = q_new
                h_best = make_sym_h(q_best, n)
                print(f"  n={n}: C5={c5_best:.7f}")
                if remaining() > 30:
                    h_cur = make_sym_h(q_best, n)
                    corrs = corr_vals(h_cur)
                    active = np.argsort(corrs)[::-1][:40].tolist()
                    active = sorted(active)
                    q_ep, c5_ep = solve_epi_sym(q_best, n, active,
                                                 t_init=corrs.max() * 1.0001,
                                                 maxiter=300)
                    if q_ep is not None and c5_ep < c5_best - 1e-10:
                        c5_best = c5_ep
                        q_best = q_ep
                        h_best = make_sym_h(q_best, n)
                        print(f"  n={n} epi: C5={c5_best:.7f}")

    # Phase 9: Final aggressive active-set cycling
    if remaining() > 30:
        for cycle in range(5):
            if remaining() < 15:
                break
            h_cur = make_sym_h(q_best, n)
            corrs = corr_vals(h_cur)
            max_c = corrs.max()
            # Use slightly larger active set each cycle
            n_act = min(40 + 10 * cycle, 2 * n - 1)
            active = np.argsort(corrs)[::-1][:n_act].tolist()
            active = sorted(active)
            q_new, c5_new = solve_epi_sym(q_best, n, active, t_init=max_c * 1.00005,
                                           maxiter=350, tol=1e-15)
            if q_new is not None and c5_new < c5_best - 1e-12:
                c5_best = c5_new
                q_best = q_new
                h_best = make_sym_h(q_best, n)
                print(f"  final cycle {cycle}: C5={c5_best:.7f}")

    # Phase 10: Final subgradient polish
    if remaining() > 20:
        h_new, c5_new = subgradient_full(h_best, t_start=0, t_end=20000, K_init=3, K_max=30)
        if c5_new < c5_best - 1e-12:
            c5_best = c5_new
            h_best = h_new
            print(f"  final subgrad: C5={c5_best:.7f}")

    # Phase 11: Final full-space polish
    if remaining() > 15:
        h_new, c5_new = full_smooth_optimize(h_best, len(h_best), 800, maxiter=300)
        if c5_new < c5_best - 1e-12:
            c5_best = c5_new
            h_best = h_new
            print(f"  final full: C5={c5_best:.7f}")

    # Final validation
    h_best = np.clip(h_best, 0.0, 1.0)
    h_best = project_feasible(h_best, len(h_best) / 2.0)
    c5_best = exact_score(h_best)
    print(f"Final: C5={c5_best:.7f}, n={len(h_best)}")

    return h_best, c5_best, len(h_best)
