import numpy as np

def project_feasible(x, target_sum):
    """Project x onto {h in [0,1]^n : sum(h) = target_sum} using bisection."""
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
        else:
            h += diff / n
    return np.clip(h, 0.0, 1.0)

def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def c5_of_h(h):
    n = len(h)
    dx = 2.0 / n
    return compute_corrs(h).max() * dx

def compute_pattern(h, p):
    """Compute the pattern vector for position p: how corr changes per unit change in h[p]."""
    n = len(h)
    m = 2 * n - 1
    pattern = np.zeros(m)
    # For offset >= 0 (index in corr array = n-1+offset)
    for offset in range(0, n):
        k = n - 1 + offset
        # p as first index: p in [0, n-1-offset]
        if 0 <= p <= n - 1 - offset:
            pattern[k] += (1.0 - h[p + offset])
        # p as second index: p-offset in [0, n-1-offset], i.e., p in [offset, n-1]
        if offset <= p <= n - 1:
            pattern[k] += (-h[p - offset])
    # For offset < 0 (index in corr array = n-1+offset, where offset = -m)
    for m_val in range(1, n):
        k = n - 1 - m_val
        # p as first index: p = i + m_val for i in [0, n-1-m_val], i.e., p in [m_val, n-1]
        if m_val <= p <= n - 1:
            pattern[k] += (1.0 - h[p - m_val])
        # p as second index: p = i for i in [0, n-1-m_val]
        if p <= n - 1 - m_val:
            pattern[k] += (-h[p + m_val])
    return pattern

    import cvxpy as cp
    n = len(h)
    target = n / 2.0
    h_current = h.copy()
    best_c5 = c5_of_h(h_current)
    best_h = h_current.copy()

            break

        # corr[k] = sum_i h[i] * g_k[i] where g_k[i] = 1 - h[i+off_k]
        # So we need: for each k, g_k[i] = 1 - h[i+off_k] for valid (i, i+off_k)
        
        # Build LP: min c s.t. h @ g_k <= c for all k, sum h = target, 0 <= h <= 1
        h_var = cp.Variable(n)
        c_var = cp.Variable()
        
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target]
        
        for k in range(2 * n - 1):
            off = k - (n - 1)
            # g_k[i] = 1 - h_current[i + off] for valid i
            # valid: 0 <= i < n and 0 <= i + off < n
            i_lo = max(0, -off)
            i_hi = min(n - 1, n - 1 - off)
            if i_lo > i_hi:
                constraints.append(c_var >= 0)  # trivial
                continue
            idx = np.arange(i_lo, i_hi + 1)
            g_k = 1.0 - h_current[idx + off]
            constraints.append(h_var @ g_k <= c_var)

    return np.clip(h, 0.0, 1.0)


def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def c5_of_h(h):
    n = len(h)
    return compute_corrs(h).max() * (2.0 / n)


def swap_steepest(h, dx, max_sweeps, t0, deadline):
    n = len(h)
    best_c5 = c5_of_h(h)
    best_h = h.copy()
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        best_new_c5 = best_c5 - 1e-15
        best_pair = None
        for i in range(n):
            for j in range(i + 1, n):
                if abs(h[i] - h[j]) < 1e-10:
                    continue
                h[i], h[j] = h[j], h[i]
                new_c5 = compute_corrs(h).max() * dx
                h[i], h[j] = h[j], h[i]
                if new_c5 < best_new_c5:
                    best_new_c5 = new_c5
                    best_pair = (i, j)
        if best_pair is None:
            break
        i, j = best_pair
        h[i], h[j] = h[j], h[i]
        best_c5 = best_new_c5
        best_h = h.copy()
    return best_h, best_c5


def swap_sa(h, dx, t0, deadline, seed):
    n = len(h)
    rng = np.random.RandomState(seed)
    c5 = c5_of_h(h)
    best_c5 = c5
    best_h = h.copy()
    temp = 0.015
    no_improve = 0
    patience = 12000
    it = 0
    while time.time() < deadline:
        it += 1
        if it % 3000 == 0:
            temp *= 0.999
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j or abs(h[i] - h[j]) < 1e-10:
            continue
        h[i], h[j] = h[j], h[i]
        new_c5 = compute_corrs(h).max() * dx
        delta = new_c5 - c5
        if delta < -1e-14:
            c5 = new_c5
            no_improve = 0
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            if temp > 1e-12 and rng.random() < np.exp(-max(delta, 0.0) / temp):
                c5 = new_c5
            else:
                h[i], h[j] = h[j], h[i]
                no_improve += 1
        if no_improve > patience:
            break
    return best_h, best_c5


def pair_shift_pass(h, dx, t0, deadline, top_k=400):
    n = len(h)
    best_c5 = c5_of_h(h)
    best_h = h.copy()
    pairs = []
    for i in range(n):
        for j in range(i + 1, n):
            d = abs(h[i] - h[j])
            if d > 1e-6:
                pairs.append((d, i, j))
    pairs.sort(key=lambda x: -x[0])
    pairs = pairs[:top_k]
    for (dval, i, j) in pairs:
        if time.time() > deadline:
            break
        d_min = max(-h[i], h[j] - 1.0) + 1e-12
        d_max = min(1.0 - h[i], h[j]) - 1e-12
        if d_max <= d_min:
            continue
        a, b = d_min, d_max
        for _ in range(35):
            m1 = a + (b - a) / 3.0
            m2 = b - (b - a) / 3.0
            h[i] += m1
            h[j] -= m1
            f1 = compute_corrs(h).max() * dx
            h[i] += (m2 - m1)
            h[j] -= (m2 - m1)
            f2 = compute_corrs(h).max() * dx
            h[i] -= m2
            h[j] += m2
            if f1 < f2:
                b = m2
            else:
                a = m1
        d_opt = (a + b) / 2.0
        h[i] += d_opt
        h[j] -= d_opt
        new_c5 = compute_corrs(h).max() * dx
        if new_c5 < best_c5 - 1e-14:
            best_c5 = new_c5
            best_h = h.copy()
        else:
            h[i] -= d_opt
            h[j] += d_opt
    return best_h, best_c5


def reversal_moves(h, dx, t0, deadline, max_len=18):
    n = len(h)
    best_c5 = c5_of_h(h)
    best_h = h.copy()
    for length in range(2, max_len + 1):
        if time.time() > deadline:
            break
        for i in range(n - length + 1):
            seg = h[i:i + length].copy()
            h[i:i + length] = seg[::-1]
            new_c5 = compute_corrs(h).max() * dx
            if new_c5 < best_c5 - 1e-14:
                best_c5 = new_c5
                best_h = h.copy()
            else:
                h[i:i + length] = seg
    return best_h, best_c5


def rotation_moves(h, dx, t0, deadline, max_len=12):
    n = len(h)
    best_c5 = c5_of_h(h)
    best_h = h.copy()
    for length in range(3, max_len + 1):
        if time.time() > deadline:
            break
        for i in range(n - length + 1):
            seg = h[i:i + length].copy()
            h[i:i + length] = np.roll(seg, -1)
            new_c5 = compute_corrs(h).max() * dx
            if new_c5 < best_c5 - 1e-14:
                best_c5 = new_c5
                best_h = h.copy()
            else:
                h[i:i + length] = seg
    return best_h, best_c5


def interp_to_n(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0.0, 2.0, n_old)
    x_new = np.linspace(0.0, 2.0, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    target = n_new / 2.0
    return project_feasible(h_new, target)


def smooth_obj_val(x, dx, beta):
    corrs = compute_corrs(x)
    m = corrs.max()
    return (m + np.log(np.sum(np.exp(beta * (corrs - m)))) / beta) * dx


def smooth_grad_val(x, dx, beta):
    n = len(x)
    corrs = compute_corrs(x)
    m = corrs.max()
    shifted = beta * (corrs - m)
    w = np.exp(shifted)
    w_sum = w.sum()
    w /= w_sum
    grad = np.zeros(n)
    for k in range(len(corrs)):
        if w[k] < 1e-15:
            continue
        off = k - (n - 1)
        p_lo1 = max(0, -off)
        p_hi1 = min(n - 1, n - 1 - off)
        if p_lo1 <= p_hi1:
            ps = np.arange(p_lo1, p_hi1 + 1)
            grad[ps] += w[k] * (1.0 - x[ps + off])
        p_lo2 = max(0, off)
        p_hi2 = min(n - 1, n - 1 + off)
        if p_lo2 <= p_hi2:
            ps = np.arange(p_lo2, p_hi2 + 1)
            grad[ps] += w[k] * (-x[ps - off])
    return grad * dx


def slsqp_polish(h0, dx, target_sum, beta, maxiter):
    from scipy.optimize import minimize
    n = len(h0)

    def obj(x):
        return smooth_obj_val(x, dx, beta)

    def obj_grad(x):
        return smooth_grad_val(x, dx, beta)

    def cons_fun(x):
        return x.sum() - target_sum

    def cons_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        h_out = project_feasible(res.x, target_sum)
    except Exception:
        h_out = project_feasible(h0, target_sum)
    return h_out, c5_of_h(h_out)


def lp_alternating(h, t0, deadline, max_iter=50):
    try:
        import cvxpy as cp
    except ImportError:
        return h, c5_of_h(h)
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h_cur = h.copy()
    best_c5 = c5_of_h(h_cur)
    best_h = h_cur.copy()
    for it in range(max_iter):
        if time.time() > deadline:
            break
        h_var = cp.Variable(n)
        c_var = cp.Variable()
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target]
        for k in range(2 * n - 1):
            off = k - (n - 1)
            i_lo = max(0, -off)
            i_hi = min(n - 1, n - 1 - off)
            if i_lo > i_hi:
                continue
            idx = np.arange(i_lo, i_hi + 1)
            g_k = 1.0 - h_cur[idx + off]
            constraints.append(h_var @ g_k <= c_var)
        objective = cp.Minimize(c_var)
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.ECOS, max_iters=5000)
        except Exception:
            try:
                prob.solve()
            except Exception:
                break
        if h_var.value is None:
            break
        new_h = np.clip(h_var.value, 0, 1)
        new_h = project_feasible(new_h, target)
        new_c5 = c5_of_h(new_h)
        if new_c5 < best_c5 - 1e-10:
            best_c5 = new_c5
            best_h = new_h.copy()
            h_cur = new_h
        else:
            improved = False
            for alpha in [0.3, 0.5, 0.7]:
                blend = (1.0 - alpha) * h_cur + alpha * new_h
                blend = project_feasible(blend, target)
                bc5 = c5_of_h(blend)
                if bc5 < best_c5:
                    best_c5 = bc5
                    best_h = blend.copy()
                    h_cur = blend
                    improved = True
            if not improved:
                break
    return best_h, best_c5


def random_permutation_score(sorted_vals, n, dx, t0, deadline, seed):
    rng = np.random.RandomState(seed)
    best_c5 = 1e10
    best_h = None
    target = n / 2.0
    it = 0
    while time.time() < deadline:
        it += 1
        perm = rng.permutation(n)
        h = sorted_vals[perm].copy()
        h = project_feasible(h, target)
        c5 = c5_of_h(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        if it > 50000:
            break
    if best_h is None:
        return None, 1e10
    return best_h, best_c5
