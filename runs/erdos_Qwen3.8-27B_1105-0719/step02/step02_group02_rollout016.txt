    """Find delta that minimizes max_k(a[k] + d[k]*delta) subject to delta_lo <= delta <= delta_hi.
    Returns (optimal_delta, minimal_max_value)."""
    # Binary search on V
    # Lower bound: min over delta in [delta_lo, delta_hi] of max_k(a[k] + d[k]*delta)
    # We can get a lower bound as max_k(a[k] + d[k]*delta_lo) and max_k(a[k] + d[k]*delta_hi)
    # Actually, a better lower bound: for each k, the minimum of a[k]+d[k]*delta over [delta_lo, delta_hi]
    # is a[k] + d[k]*delta_lo if d[k] >= 0, or a[k] + d[k]*delta_hi if d[k] < 0
    # The max of these lower bounds is a lower bound on the optimal value.
    
    # Upper bound: max_k(a[k] + d[k]*0) if 0 is in range, else max at closest boundary
    # Better: evaluate at delta=0 (clipped to range)
    
    delta_start = np.clip(0.0, delta_lo, delta_hi)
    V_hi = float(np.max(a + d * delta_start))
    
    # Lower bound
    lb_per_k = np.where(d >= 0, a + d * delta_lo, a + d * delta_hi)
    V_lo = float(np.max(lb_per_k))
    
    if V_hi - V_lo < 1e-15:
        return delta_start, V_hi
    
    for _ in range(60):  # 60 iterations of binary search
        V_mid = (V_lo + V_hi) / 2.0
        # Check feasibility: is there a delta in [delta_lo, delta_hi] with a[k] + d[k]*delta <= V_mid for all k?
        # For d[k] > 0: delta <= (V_mid - a[k]) / d[k]
        # For d[k] < 0: delta >= (V_mid - a[k]) / d[k]

        neg_mask = d < -1e-15
        zero_mask = ~(pos_mask | neg_mask)

            V_lo = V_mid

    offsets = np.arange(-(n-1), n, dtype=int)
    len_corr = 2 * n - 1
    n_improve = 0
    n_no_improve = 0
    
    for it in range(max_iters):
        if time.time() - t0 > time_limit:
            break
        
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        
        # Compute D vector
        idx_jp = j + offsets
        v_jp = (idx_jp >= 0) & (idx_jp < n)
        idx_jm = j - offsets
        v_jm = (idx_jm >= 0) & (idx_jm < n)
        idx_ip = i + offsets
        v_ip = (idx_ip >= 0) & (idx_ip < n)
        idx_im = i - offsets
        v_im = (idx_im >= 0) & (idx_im < n)
        
        D = np.zeros(len_corr)
        D[v_jp] += h[idx_jp[v_jp]]
        D[v_jm] += h[idx_jm[v_jm]]
        D[v_ip] -= h[idx_ip[v_ip]]
        D[v_im] -= h[idx_im[v_im]]
        
        if np.max(np.abs(D)) < 1e-14:
            continue
        
        delta_lo = max(-h[i], h[j] - 1.0)
        delta_hi = min(1.0 - h[i], h[j])
        
        if delta_lo >= delta_hi - 1e-14:
            continue
        
        delta_opt, v_opt = find_optimal_delta(corrs, D, delta_lo, delta_hi)
        
        if v_opt < current_max - 1e-14:
            h[i] = np.clip(h[i] + delta_opt, 0.0, 1.0)
            h[j] = np.clip(h[j] - delta_opt, 0.0, 1.0)
            corrs = corrs + delta_opt * D
            current_max = float(corrs.max())
            n_improve += 1
            
            if current_max < best_max - 1e-14:
                best_max = current_max
                best_h = h.copy()
        else:
            n_no_improve += 1
        
        # Periodic recompute to avoid drift
        if (it + 1) % recompute_every == 0:
            corrs = compute_corrs(h)
            current_max = float(corrs.max())
            if current_max < best_max - 1e-14:
                best_max = current_max
                best_h = h.copy()
        
        # Early stopping if stuck
        if n_no_improve > 50000 and n_improve > 0:
            break
    
    return best_h, best_max

def slsqp_opt(h0, dx, target_sum, beta, maxiter):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        corrs = compute_corrs(x)
        return float(logsumexp(beta * corrs) / beta * dx)
    
    def grad(x):
        corrs = compute_corrs(x)
        shifted = corrs - corrs.max()
        w = np.exp(beta * shifted)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(len(corrs)) / len(corrs)
        else:
            w /= w_sum
        # Gradient of each corr_k w.r.t. h
        # Same as before
        n_local = len(x)
        g = np.zeros(n_local)
        for k in range(len(corrs)):
            if w[k] < 1e-12:
                continue
            offset = k - (n_local - 1)
            if offset == 0:
                g[:] += w[k] * (1.0 - 2.0 * x)
            else:
                i_lo = max(0, -offset)
                i_hi = min(n_local - 1, n_local - 1 - offset)
                if i_lo <= i_hi:
                    idx = np.arange(i_lo, i_hi + 1)
                    g[idx + offset] += w[k] * (1.0 - x[idx])
                    g[idx] += w[k] * (-x[idx + offset])
        return g * dx
    
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum,
             'jac': lambda x: np.ones(n)}]
    
    try:
        res = minimize(obj, h0, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        h_out = project_feasible(res.x, target_sum)
    except Exception:
        h_out = project_feasible(h0, target_sum)
    return h_out, c5_of_h(h_out, dx)
