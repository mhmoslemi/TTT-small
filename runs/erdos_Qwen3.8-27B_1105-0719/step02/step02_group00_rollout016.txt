def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity, kind='mergesort')
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx])

def solve_lp_direction(h, P, G, active_idx, rho=1e-8):
    n = len(h)
    nv = 2 * n + 1
    # x = [u_0,...,u_{n-1}, v_0,...,v_{n-1}, alpha]
    
    # Objective: minimize -alpha
    c_cost = np.zeros(nv)
    c_cost[-1] = -1.0
    
    # Inequality constraints: A_ub @ x <= b_ub
    # 1. For each active m: G[m,:]·(u-v) + alpha <= 0
    #    [G[m,:], -G[m,:], 1] · x <= 0
    rows_ineq = []
    rhs_ineq = []
    for m in active_idx:
        row = np.zeros(nv)
        row[:n] = G[m, :]
        row[n:2*n] = -G[m, :]
        row[-1] = 1.0
        rows_ineq.append(row)
        rhs_ineq.append(0.0)
    
    # 2. Box upper: u_i - v_i <= 1 - h_i + rho
    #    [e_i at u, -e_i at v, 0] · x <= 1 - h_i + rho
    for i in range(n):
        row = np.zeros(nv)
        row[i] = 1.0
        row[n + i] = -1.0
        rows_ineq.append(row)
        rhs_ineq.append(1.0 - h[i] + rho)
    
    # 3. Box lower: -(u_i - v_i) <= h_i + rho
    #    [-e_i at u, e_i at v, 0] · x <= h_i + rho
    for i in range(n):
        row = np.zeros(nv)
        row[i] = -1.0
        row[n + i] = 1.0
        rows_ineq.append(row)
        rhs_ineq.append(h[i] + rho)
    
    A_ub = np.array(rows_ineq)
    b_ub = np.array(rhs_ineq)
    
    # Equality constraints: A_eq @ x = b_eq
    # 1. sum(u_i - v_i) = 0: [1,...,1, -1,...,-1, 0] · x = 0
    # 2. sum(u_i + v_i) = 1: [1,...,1, 1,...,1, 0] · x = 1
    A_eq = np.zeros((2, nv))
    A_eq[0, :n] = 1.0
    A_eq[0, n:2*n] = -1.0
    A_eq[1, :n] = 1.0
    A_eq[1, n:2*n] = 1.0
    b_eq = np.array([0.0, 1.0])
    
    # Bounds: all variables >= 0
    bounds = [(0, None)] * nv
    
    result = linprog(c_cost, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
    
    if result.status == 0 and result.fun < 0:
        alpha = -result.fun
        d = result.x[:n] - result.x[n:2*n]
        if np.all(np.isfinite(d)):
            return d, alpha
    return None, 0.0

C₅(h) = max_k ∫₀² h(x)(1-h(x+k)) dx

For the discrete version with n points:

P[d] = dx * sum_{i=0}^{n-1-|d|} h[i] * (1-h[i+|d|]) + ...

For a binary sequence, this counts positions j where h[j]=0 and h[j+d]=1, multiplied by dx.

import numpy as np
from scipy.optimize import linprog
import time

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity, kind='mergesort')
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx])
            if add > 0:
                h[idx] += add
                capacity[idx] -= add
            remaining -= add
            if remaining <= 1e-14:
                break
    else:
        order = np.argsort(h, kind='mergesort')
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-14:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmin(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def score_exact(h):
    n = len(h)
    if n < 2:
        return float('inf')
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(profile))

def compute_profile_and_grad(h):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    P = np.zeros(M)
    G = np.zeros((M, n))
    for m in range(M):
        d = m - (n - 1)
        if d == 0:
            P[m] = dx * np.sum(h * (1.0 - h))
            G[m, :] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            valid = n - d
            if valid > 0:
                P[m] = dx * np.dot(h[d:], 1.0 - h[:valid])
                G[m, :valid] = -dx * h[d:]
                G[m, d:] += dx * (1.0 - h[:valid])
        else:
            e = -d
            valid = n - e
            if valid > 0:
                P[m] = dx * np.dot(h[:valid], 1.0 - h[e:])
                G[m, :valid] = dx * (1.0 - h[e:])
                G[m, e:] += -dx * h[:valid]
    return P, G

def solve_lp_direction(h, P, G, active_idx, rho=1e-8):
    n = len(h)
    nv = 2 * n + 1
    c_cost = np.zeros(nv)
    c_cost[-1] = -1.0
    rows_ineq = []
    rhs_ineq = []
    for m in active_idx:
        row = np.zeros(nv)
        row[:n] = G[m, :]
        row[n:2*n] = -G[m, :]
        row[-1] = 1.0
        rows_ineq.append(row)
        rhs_ineq.append(0.0)
    for i in range(n):
        row = np.zeros(nv)
        row[i] = 1.0
        row[n + i] = -1.0
        rows_ineq.append(row)
        rhs_ineq.append(1.0 - h[i] + rho)
        row2 = np.zeros(nv)
        row2[i] = -1.0
        row2[n + i] = 1.0
        rows_ineq.append(row2)
        rhs_ineq.append(h[i] + rho)
    A_ub = np.array(rows_ineq)
    b_ub = np.array(rhs_ineq)
    A_eq = np.zeros((2, nv))
    A_eq[0, :n] = 1.0
    A_eq[0, n:2*n] = -1.0
    A_eq[1, :n] = 1.0
    A_eq[1, n:2*n] = 1.0
    b_eq = np.array([0.0, 1.0])
    bounds = [(0.0, None)] * nv
    try:

                         options={'presolve': True, 'time_limit': 5.0})

                             bounds=bounds, method='interior-point')
        except Exception:
            return None, 0.0
    if result.status == 0 and result.fun < -1e-12:
        alpha = -result.fun
        d = result.x[:n] - result.x[n:2*n]
        if np.all(np.isfinite(d)) and np.max(np.abs(d)) > 1e-12:
            return d, float(alpha)
    return None, 0.0

            c_interp = score_exact(h_interp)
            # Quick LP polish
            te_dim = min(t_start + time_left() - 5, time.perf_counter() + 25)
            h_dim, c_dim = local_polish_lp(h_interp, 25, te_dim)
            if c_dim < best_c - 1e-12:
                best_h = h_dim
                best_c = c_dim
                # Re-interpolate back to original dim and polish
                if n_cur != new_n:
                    h_back = interpolate_to_n(best_h, n_cur)
                    te_back = min(t_start + time_left() - 5, time.perf_counter() + 20)
                    h_back2, c_back2 = local_polish_lp(h_back, 20, te_back)
                    if c_back2 < best_c - 1e-12:
                        best_h = h_back2
                        best_c = c_back2

    # Stage 4: Structured seeds from scratch
    if time_left() > 80:
        n_best = len(best_h)
        seeds = []
        # Symmetric/anti-symmetric of current best
        seeds.append(make_symmetric(best_h))
        seeds.append(make_antisymp(best_h))
        # Binary
        seeds.append(make_binary_seed(best_h))
        # Cosine seeds
        for freq in [1, 2, 3]:
            for amp in [0.3, 0.4, 0.5]:
                seeds.append(make_cos_seed(n_best, freq, 0.0, amp))
                seeds.append(make_cos_seed(n_best, freq, np.pi / 3, amp))
        # Block seeds
        for nb in [4, 6, 8, 10]:
            seeds.append(make_block_seed(n_best, nb))
        # Perturbations of best
        for eps in [0.02, 0.05, 0.1]:
            seeds.append(perturb_seeded(best_h, rng, eps))
        for s in seeds:
            if time_left() < 30:
                break
            s = repair_feasible(s)
            if not np.all(np.isfinite(s)):
                continue
            c_s = score_exact(s)
            if c_s > best_c + 0.05:
                continue
            te_s = min(t_start + time_left() - 3, time.perf_counter() + 15)
            h_s, c_s2 = local_polish_lp(s, 15, te_s)
            if c_s2 < best_c - 1e-12:
                best_h = h_s
                best_c = c_s2

    # Stage 5: Final deep LP polish
    if time_left() > 40:
        te_final = t_start + max(5.0, time_left() - 3.0)
        h_fin, c_fin = local_polish_lp(best_h, time_left(), te_final)
        if c_fin < best_c - 1e-14:
            best_h = h_fin
            best_c = c_fin

    # Stage 6: Gradient descent micro-polish
    if time_left() > 15:
        te_fd = t_start + max(3.0, time_left() - 2.0)
        h_fd2, c_fd2 = gradient_descent_fallback(best_h, te_fd)
        if c_fd2 < best_c - 1e-14:
            best_h = h_fd2
            best_c = c_fd2

    # Stage 7: One more LP pass if time remains
    if time_left() > 10:
        te_last = t_start + max(3.0, time_left() - 1.0)
        h_last, c_last = local_polish_lp(best_h, time_left(), te_last)
        if c_last < best_c - 1e-14:
            best_h = h_last
            best_c = c_last

    # Final validation
    best_h = repair_feasible(best_h)
    if not np.all(np.isfinite(best_h)) or len(best_h) < 2:
        best_h = np.full(n_parent, 0.5)
        best_h = repair_feasible(best_h)
    final_c = score_exact(best_h)
    if final_c < best_c - 1e-10:
        best_c = final_c
    else:
        best_c = final_c
    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            if float(eval_result) < best_c:
                best_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 1:
            if np.isfinite(eval_result[0]):
                best_c = min(best_c, float(eval_result[0]))
    except Exception:
        pass
    return best_h, float(best_c), int(len(best_h))
