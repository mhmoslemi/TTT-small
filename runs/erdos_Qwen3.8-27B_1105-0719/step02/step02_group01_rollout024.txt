```python
import numpy as np
from scipy.optimize import linprog
import time

# Helper functions (all top-level, no lambdas)

def project_feasible(x, target_sum):
    """Project x to [0,1]^n with sum = target_sum using bisection."""
    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo+hi)*0.5, 0.0, 1.0)
    # Fix residual
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0 - 1e-10
            if mask.sum() > 0:
                h[mask] += diff / mask.sum()
        else:
            mask = h > 1e-10
            if mask.sum() > 0:
                h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def c5_of_h(h):
    """Compute C5 = max_k correlate(h, 1-h, full) * dx where dx=2/len(h)."""
    dx = 2.0 / len(h)
    return np.correlate(h, 1.0 - h, mode='full').max() * dx

def compute_grad_row(h, off, n):
    """Compute gradient of correlation at offset 'off' w.r.t. h."""
    grad = np.zeros(n)
    n_lo = max(0, -off)
    n_hi = min(n-1, n-1-off)
    if n_lo > n_hi:
        return grad
    # Part 1: h[n+off] factor -> d/dh[j] where j=n+off, n in [n_lo, n_hi]
    idx1 = np.arange(n_lo + off, n_hi + off + 1)
    grad[idx1] += (1.0 - h[idx1 - off])
    # Part 2: (1-h[n]) factor -> d/dh[j] where j=n, n in [n_lo, n_hi]
    idx2 = np.arange(n_lo, n_hi + 1)
    grad[idx2] -= h[idx2 + off]
    return grad

def active_set_lp(h, dx, target_sum, max_iters, alpha0, n_active_frac, time_limit):
    """Active-set LP polish with exact line search."""
    n = len(h)
    h_cur = h.copy()
    c5_cur = c5_of_h(h_cur)
    best_h = h_cur.copy()
    best_c5 = c5_cur
    alpha = alpha0
    t0 = time.time()
    
    for it in range(max_iters):
        if time.time() - t0 > time_limit:
            break
        
        corrs = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        # Select active set
        n_active = max(10, int(n_active_frac * (2*n - 1)))
        top_idx = np.argsort(corrs)[-n_active:]
        
        # Build LP
        # min s
        # s.t. grad_k . delta - s <= -c_k  (for k in active)
        #      sum(delta) = 0
        #      -alpha*h[i] <= delta[i] <= alpha*(1-h[i])
        
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0  # minimize s
        
        A_rows = []
        b_rows = []
        for k in top_idx:
            off = k - (n - 1)
            g = compute_grad_row(h_cur, off, n)
            row = np.zeros(n + 1)
            row[:n] = g
            row[-1] = -1.0
            A_rows.append(row)
            b_rows.append(-corrs[k])
        
        A_ub = np.array(A_rows)
        b_ub = np.array(b_rows)

        A_eq[0, :n] = 1.0
        b_eq = np.array([0.0])

                continue

        except:

            continue

def quantized_sa(h_init, n, M, n_steps, T0, T1, rng, time_limit, t0):
    """Simulated annealing on quantized grid."""
    target_sum_int = n * M // 2  # Must be integer
    if n * M % 2 != 0:
        return None  # Can't do quantized SA
    
    # Initialize from h_init
    a = np.round(h_init * M).astype(int)
    a = np.clip(a, 0, M)
    # Fix sum
    current_sum = a.sum()
    diff = target_sum_int - current_sum
    # Distribute the difference
    if diff > 0:
        free_mask = a < M
        free_idx = np.where(free_mask)[0]
        for _ in range(diff):
            idx = free_idx[rng.randint(len(free_idx))]
            a[idx] = min(M, a[idx] + 1)
            if a[idx] < M:
                continue
            free_mask[idx] = False
            free_idx = np.where(free_mask)[0]
            if len(free_idx) == 0:
                break
    elif diff < 0:
        free_mask = a > 0
        free_idx = np.where(free_mask)[0]
        for _ in range(-diff):
            idx = free_idx[rng.randint(len(free_idx))]
            a[idx] = max(0, a[idx] - 1)
            if a[idx] > 0:
                continue
            free_mask[idx] = False
            free_idx = np.where(free_mask)[0]
            if len(free_idx) == 0:
                break
    
    h = a.astype(float) / M
    c5_cur = c5_of_h(h)
    best_h = h.copy()
    best_c5 = c5_cur
    T = T0
    
    for step in range(n_steps):
        if time.time() - t0 > time_limit:
            break
        # Update temperature
        frac = step / max(1, n_steps - 1)
        T = T0 * (T1 / T0) ** frac
        
        # Random move
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        direction = 1 if rng.random() < 0.5 else -1
        
        new_ai = a[i] + direction
        new_aj = a[j] - direction
        if new_ai < 0 or new_ai > M or new_aj < 0 or new_aj > M:
            continue
        
        a[i] = new_ai
        a[j] = new_aj
        h = a.astype(float) / M
        c5_new = c5_of_h(h)
        
        if c5_new <= c5_cur or rng.random() < np.exp(-(c5_new - c5_cur) / T):
            c5_cur = c5_new
            if c5_cur < best_c5:
                best_c5 = c5_cur
                best_h = h.copy()
        else:
            # Reject
            a[i] = new_ai - direction
            a[j] = new_aj + direction
    
    return best_h, best_c5

def transfer_grid(h_src, n_dst):
    """Transfer h from its current grid to a new grid of size n_dst."""
    n_src = len(h_src)
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    h_dst = np.interp(x_dst, x_src, h_src)
    return project_feasible(h_dst, n_dst / 2.0)

def subgradient_polish(h0, dx, target_sum, max_iter, time_limit, t0):
    """Subgradient method with exact line search."""
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h)
    
    for it in range(max_iter):
        if time.time() - t0 > time_limit:
            break
        corrs = np.correlate(h, 1.0 - h, mode='full')
        c5_val = corrs.max() * dx
        
        # Smooth max weights
        shifted = corrs - corrs.max()
        beta = 50.0 + it * 0.5
        w = np.exp(beta * shifted)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(2*n-1) / (2*n-1)
        else:
            w /= w_sum
        
        grad = np.zeros(n)
        for k in range(2*n-1):
            if w[k] > 1e-12:
                off = k - (n-1)
                grad += w[k] * compute_grad_row(h, off, n)
        grad *= dx
        
        # Line search
        best_step = 0.0
        best_val = c5_val
        base_step = 0.1 / np.linalg.norm(grad)
        for mult in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01]:
            step = base_step * mult
            h_try = h - step * grad
            h_try = project_feasible(h_try, target_sum)
            val = c5_of_h(h_try)
            if val < best_val:
```

```python
def project_feasible(x, target_sum):

    for _ in range(70):

        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)


def c5_of_h(h):
    dx = 2.0 / len(h)
    return np.correlate(h, 1.0 - h, mode='full').max() * dx


def compute_all_grads(h, n):
    grad = np.zeros((2 * n - 1, n))
    one_minus_h = 1.0 - h
    for k in range(2 * n - 1):

def quantized_sa(h_init, n, M, n_steps, T0, T1, rng, time_limit, t0):
    if n * M % 2 != 0:
        return None, None
    target_int = n * M // 2
    a = np.clip(np.round(h_init * M).astype(int), 0, M)
    cur_sum = a.sum()
    diff = target_int - cur_sum
    if diff > 0:
        for _ in range(diff):
            candidates = np.where(a < M)[0]
            if len(candidates) == 0:
                break
            idx = candidates[rng.randint(len(candidates))]
            a[idx] += 1
    elif diff < 0:
        for _ in range(-diff):
            candidates = np.where(a > 0)[0]
            if len(candidates) == 0:
                break
            idx = candidates[rng.randint(len(candidates))]
            a[idx] -= 1

    h = a.astype(float) / M
    c5_cur = c5_of_h(h)
    best_h = h.copy()
    best_c5 = c5_cur
    stagnation = 0

    for step in range(n_steps):
        if time.time() - t0 > time_limit:
            break
        frac = step / max(1, n_steps - 1)
        T = T0 * (T1 / T0) ** frac

        if stagnation > 2000:
            T = max(T, T0 * 0.3)
            stagnation = 0

        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        direction = 1 if rng.random() < 0.5 else -1
        new_ai = a[i] + direction
        new_aj = a[j] - direction
        if new_ai < 0 or new_ai > M or new_aj < 0 or new_aj > M:
            continue

        a[i] = new_ai
        a[j] = new_aj
        h = a.astype(float) / M
        c5_new = c5_of_h(h)

        if c5_new <= c5_cur or rng.random() < np.exp(-(c5_new - c5_cur) / max(T, 1e-9)):
            c5_cur = c5_new
            if c5_cur < best_c5:
                best_c5 = c5_cur
                best_h = h.copy()
                stagnation = 0
            else:
                stagnation += 1
        else:
            a[i] = new_ai - direction
            a[j] = new_aj + direction
            stagnation += 1

    return best_h, best_c5


def transfer_grid(h_src, n_dst):
    n_src = len(h_src)
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    h_dst = np.interp(x_dst, x_src, h_src)
    return project_feasible(h_dst, n_dst / 2.0)


def make_binary_start(n, target_sum, rng):
    h = np.zeros(n)
    idx = rng.permutation(n)
    n_ones = int(target_sum)
    if n_ones > 0:
        h[idx[:n_ones]] = 1.0
    frac = target_sum - n_ones
    if frac > 1e-10 and n_ones < n:
        h[idx[n_ones]] = frac
    return project_feasible(h, target_sum)


def make_periodic_start(n, target_sum, rng, period):
    h = np.zeros(n)
    fill = max(1, int(target_sum / n * period))
    for i in range(n):
        h[i] = 1.0 if (i % period) < fill else 0.0
    return project_feasible(h, target_sum)


def remaining_time(t0, budget_s):
    return budget_s - (time.time() - t0)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')

    # Phase 0: Initialize from parent
    h0 = np.array(initial_h_values, dtype=float).copy()
    n_parent = len(h0)
    ts_parent = n_parent / 2.0
    h0 = project_feasible(h0, ts_parent)
    best_h = h0.copy()
    best_c5 = c5_of_h(h0)

    # Phase 1: LP polish parent (multiple passes with different alpha)
    for alpha_pass in [0.3, 0.15, 0.08]:
        if remaining_time(t0, budget_s) < 40:
            break
        alloc = min(100, remaining_time(t0, budget_s) - 15)
        h_lp, c5_lp = active_set_lp(best_h, ts_parent if len(best_h) == n_parent else len(best_h) / 2.0,
                                     max_iters=400, alpha0=alpha_pass, time_limit=alloc, t0=t0)
        if c5_lp < best_c5:
            best_c5 = c5_lp
            best_h = h_lp

    # Subgradient pass
    if remaining_time(t0, budget_s) > 80:
        n_cur = len(best_h)
        h_sg, c5_sg = subgradient_polish(best_h, n_cur / 2.0, max_iter=8000,
                                         time_limit=min(120, remaining_time(t0, budget_s) - 20), t0=t0)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg

    # Phase 2: Quantized SA on various grid sizes
    sa_configs = []
    for n_try in [48, 64, 80, 96, 112]:
        for M in [4, 8]:
            if n_try * M % 2 == 0:
                sa_configs.append((n_try, M))

    for n_sa, M in sa_configs:
        if remaining_time(t0, budget_s) < 25:
            break
        h_init = transfer_grid(best_h, n_sa)
        T0 = 0.004
        T1 = 1e-6
        n_steps = 20000 if n_sa <= 64 else 12000
        alloc = min(70, remaining_time(t0, budget_s) - 15)
        h_sa, c5_sa = quantized_sa(h_sa_init=h_init, n=n_sa, M=M, n_steps=n_steps,
                                    T0=T0, T1=T1, rng=rng, time_limit=alloc, t0=t0)
        if h_sa is not None and c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa
            # Polish SA result
            if remaining_time(t0, budget_s) > 25:
                h_pol, c5_pol = active_set_lp(h_sa, n_sa / 2.0, max_iters=200,
                                               alpha0=0.2, time_limit=min(40, remaining_time(t0, budget_s) - 10), t0=t0)
                if c5_pol < best_c5:
                    best_c5 = c5_pol
                    best_h = h_pol

    # Phase 3: Diverse continuous starts with LP polish
    for trial in range(12):
        if remaining_time(t0, budget_s) < 20:
            break
        n_try = n_parent
        if trial % 3 == 0:
            h_init = make_binary_start(n_try, n_try / 2.0, rng)
        elif trial % 3 == 1:
            h_init = make_periodic_start(n_try, n_try / 2.0, rng, rng.choice([3, 4, 5, 6]))
        else:
            h_init = project_feasible(best_h + 0.3 * (rng.random(n_try) - 0.5), n_try / 2.0)

        alloc = min(40, remaining_time(t0, budget_s) - 10)
        h_opt, c5_opt = active_set_lp(h_init, n_try / 2.0, max_iters=200,
                                       alpha0=0.25, time_limit=alloc, t0=t0)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
        elif c5_opt < best_c5 + 0.005:
            # Close candidate - try more polish
            if remaining_time(t0, budget_s) > 20:
                h_pol, c5_pol = active_set_lp(h_init, n_try / 2.0, max_iters=150,
                                               alpha0=0.1, time_limit=min(30, remaining_time(t0, budget_s) - 10), t0=t0)
                if c5_pol < best_c5:
                    best_c5 = c5_pol
                    best_h = h_pol

    # Phase 4: Try different grid sizes for the best solution
    n_candidates = []
    for dn in [-16, -8, 8, 12, 16]:
        nc = n_best_n if 'n_best_n' in dir() else n_parent
    n_best_n = len(best_h)
    n_candidates = [n_best_n + d for d in [-16, -8, 8, 16, 24] if n_best_n + d >= 32]
    n_candidates = [x for x in n_candidates if x != n_best_n and x % 2 == 0]

    for n_try in n_candidates:
        if remaining_time(t0, budget_s) < 30:
            break
        h_trans = transfer_grid(best_h, n_try)
        alloc = min(60, remaining_time(t0, budget_s) - 15)
        h_opt, c5_opt = active_set_lp(h_trans, n_try / 2.0, max_iters=300,
                                       alpha0=0.2, time_limit=alloc, t0=t0)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
            # Further polish
            if remaining_time(t0, budget_s) > 20:
                h_pol, c5_pol = active_set_lp(best_h, n_try / 2.0, max_iters=200,
                                               alpha0=0.08, time_limit=min(30, remaining_time(t0, budget_s) - 10), t0=t0)
                if c5_pol < best_c5:
                    best_c5 = c5_pol
                    best_h = h_pol

    # Phase 5: Final intensive polish
    n_fin = len(best_h)
    if remaining_time(t0, budget_s) > 20:
        alloc = min(50, remaining_time(t0, budget_s) - 5)
        h_fin, c5_fin = active_set_lp(best_h, n_fin / 2.0, max_iters=600,
                                       alpha0=0.05, time_limit=alloc, t0=t0)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin

    if remaining_time(t0, budget_s) > 15:
        n_fin = len(best_h)
        h_sg2, c5_sg2 = subgradient_polish(best_h, n_fin / 2.0, max_iter=5000,
                                            time_limit=min(25, remaining_time(t0, budget_s) - 5), t0=t0)
        if c5_sg2 < best_c5:
            best_c5 = c5_sg2
            best_h = h_sg2

    # Final very fine LP
    if remaining_time(t0, budget_s) > 10:
        n_fin = len(best_h)
        h_f2, c5_f2 = active_set_lp(best_h, n_fin / 2.0, max_iters=300,
                                     alpha0=0.03, time_limit=min(15, remaining_time(t0, budget_s) - 2), t0=t0)
        if c5_f2 < best_c5:
            best_c5 = c5_f2
            best_h = h_f2

    # Validate and return
    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_c5 = c5_of_h(best_h)

    return best_h, best_c5, len(best_h)
```
