import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time


def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.5, 1.5
    for _ in range(100):
        mid = (lo + hi) / 2.0
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)


def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full').max() * dx


def corr_vals(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx


def make_sym_h(q, n):
    m = n // 2
    q = np.clip(q, 0.0, 1.0)
    h = np.empty(n)
    h[:m] = q
    h[m:] = 1.0 - q[::-1]
    return h


def make_q_from_h(h, n):
    m = n // 2
    q = 0.5 * (h[:m] + 1.0 - h[n - 1: n - 1 - m: -1])
    return np.clip(q, 0.0, 1.0)


def sym_corr_vals(q, n):
    m = n // 2
    h = make_sym_h(q, n)
    return corr_vals(h)


def sym_grad_lag(q, n, lag_idx):
    """Gradient of correlation at specific lag w.r.t. q (symmetric param)."""
    m = n // 2
    dx = 2.0 / n
    h = make_sym_h(q, n)
    d = lag_idx - (n - 1)
    i_lo = max(0, -d)
    i_hi = min(n - 1, n - 1 - d)
    G = np.zeros(n)
    if i_lo <= i_hi:
        idx = np.arange(i_lo, i_hi + 1)
        G[idx] += (1.0 - h[idx + d])
        G[idx + d] += (-h[idx])
    G *= dx
    return G[:m] - G[n - 1: n - 1 - m: -1]


def active_lags_top(corrs, n_top):
    idx = np.argsort(corrs)[::-1][:n_top]
    return sorted(idx.tolist())


def smooth_obj_and_grad(q, n, beta):
    m = n // 2
    corrs = sym_corr_vals(q, n)
    val = logsumexp(beta * corrs) / beta
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    ws = w.sum()
    if ws < 1e-30:
        w = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        w /= ws
    grad = np.zeros(m)
    for k in range(len(w)):
        if w[k] < 1e-15:
            continue
        grad += w[k] * sym_grad_lag(q, n, k)
    grad *= beta
    return val, grad


def lse_optimize(q0, n, beta, maxiter=500):
    m = n // 2

    def obj_wrapper(x):
        return smooth_obj_and_grad(x, n, beta)[0]

    def grad_wrapper(x):
        return smooth_obj_and_grad(x, n, beta)[1]

    try:
        res = minimize(obj_wrapper, q0, jac=grad_wrapper, method='SLSQP',
                       bounds=[(0.0, 1.0)] * m,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        h_new = make_sym_h(res.x, n)
        c5 = exact_score(h_new)
        return res.x, c5
    except Exception:
        return None, None


def subgradient_step(q, n, step_size, n_active=20):
    """One projected subgradient step on the exact max objective."""
    m = n // 2
    corrs = sym_corr_vals(q, n)
    max_c = corrs.max()
    threshold = max_c - 0.002
    active = np.where(corrs >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(corrs)])
    
    grad = np.zeros(m)
    for lag in active:
        grad += sym_grad_lag(q, n, lag)
    grad /= len(active)
    
    q_new = q - step_size * grad
    q_new = np.clip(q_new, 0.0, 1.0)
    return q_new, max_c


def subgradient_optimize(q0, n, max_iter=2000, step_init=0.02):
    m = n // 2
    q = q0.copy()
    best_q = q.copy()
    best_c = exact_score(make_sym_h(q, n))
    
    for it in range(max_iter):
        step = step_init / (1.0 + 0.01 * it)
        q, cur_c = subgradient_step(q, n, step, n_active=30)
        cur_exact = exact_score(make_sym_h(q, n))
        if cur_exact < best_c:
            best_c = cur_exact
            best_q = q.copy()
    
    return best_q, best_c


def coordinate_descent(q0, n, max_rounds=200, tol=1e-10):
    """Coordinate descent with exact line search on each q_i."""
    m = n // 2
    q = q0.copy()
    best_c = exact_score(make_sym_h(q, n))
    
    order = list(range(m))
    for rnd in range(max_rounds):
        rng2 = np.random.RandomState(rnd + 100)
        rng2.shuffle(order)
        improved = False
        for i in order:
            old_val = q[i]
            # Try a few values around current
            candidates = [old_val]
            for dv in [0.01, -0.01, 0.005, -0.005, 0.02, -0.02, 0.05, -0.05]:
                cand = old_val + dv
                if 0.0 <= cand <= 1.0:
                    candidates.append(cand)
            cameras = sorted(set(cand for cand in candidates if 0.0 <= cand <= 1.0))
            best_cand = old_val
            best_val = best_c
            for c_val in cameras:
                q[i] = c_val
                c = exact_score(make_sym_h(q, n))
                if c < best_val:
                    best_val = c
                    best_cand = c_val
            q[i] = best_cand
            if best_val < best_c - 1e-12:
                best_c = best_val
                improved = True
        if not improved and rnd > 10:
            break
    
    return q, best_c


def nm_polish(q0, n, maxiter=800):
    """Nelder-Mead on the symmetric q parameters."""
    m = n // 2

    def obj(x):
        return exact_score(make_sym_h(x, n))

    try:
        res = minimize(obj, q0, method='Nelder-Mead',
                       options={'maxiter': maxiter, 'fatol': 1e-12,
                                'xatol': 1e-10, 'adaptive': True})
        h_new = make_sym_h(res.x, n)
        c5 = exact_score(h_new)
        return res.x, c5
    except Exception:
        return None, None


def directed_perturb(q, n, rng, scale=0.05):
    """Perturb q in direction that reduces the top correlations."""
    m = n // 2
    corrs = sym_corr_vals(q, n)
    top_lags = np.argsort(corrs)[::-1][:10]
    
    pert = np.zeros(m)
    for lag in top_lags:
        g = sym_grad_lag(q, n, lag)
        pert -= g
    pert = pert / (np.linalg.norm(pert) + 1e-10)
    noise = rng.randn(m) * 0.3 * scale
    q_new = q + scale * pert + noise
    return np.clip(q_new, 0.0, 1.0)

    h_best = project_feasible(h_best, len(h_best) / 2.0)
    c5_best = exact_score(h_best)
    print(f"Final: n={len(h_best)}, C5={c5_best:.7f}")

    return h_best, c5_best, len(h_best)
