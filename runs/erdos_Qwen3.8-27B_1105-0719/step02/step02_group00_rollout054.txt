def repair_feasible(h):
    n = len(h)
    h = np.clip(h.astype(np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        # Need to add mass: add to elements with most room
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx])
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-14:
                break
    else:
        # Need to remove mass: remove from elements with most value
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-14:
                break
    # Final fix
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

import numpy as np
from scipy.optimize import minimize
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def repair_feasible(h):
    n = len(h)
    h = np.clip(h.astype(np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx])
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-14:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-14:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def build_grad_row(h, k, n, dx):
    """Build gradient of P_k with respect to h.
    k is the lag (can be negative, 0, or positive).
    Returns a vector of length n.
    """
    g = np.zeros(n)
    kk = abs(k)
    if kk == 0:
        g[:] = dx * (1.0 - 2.0 * h)
    elif k > 0:
        # P_k = dx * sum_{j=0}^{n-1-k} h[j+k]*(1-h[j])
        # d/dh[m] = dx*(1-h[m-k]) for m>=k, plus dx*(-h[m+k]) for m<n-k
        g[k:] += dx * (1.0 - h[0:n-kk])
        g[0:n-kk] += -dx * h[kk:n]
    else:
        # P_{-k} = dx * sum_{j=0}^{n-1-kk} h[j]*(1-h[j+kk])
        # d/dh[m] = dx*(1-h[m+kk]) for m<n-kk, plus dx*(-h[m-kk]) for m>=kk
        g[0:n-kk] += dx * (1.0 - h[kk:n])
        g[kk:n] += -dx * h[0:n-kk]
    return g

def build_grad_matrix(h):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    G = np.zeros((M, n))
    for k in range(-n+1, n):
        m = k + n - 1  # output index
        G[m] = build_grad_row(h, k, n, dx)
    return G

def active_gradient_step(h, beam=0.005, max_step=0.02, min_step=1e-5, n_steps=8):
    """One step of active-set gradient descent. Returns new h and score."""
    n = len(h)
    dx = 2.0 / n
    profile = compute_profile(h)
    c = np.max(profile)
    
    # Select active set
    thresh = c - beam
    active = np.where(profile >= thresh)[0]
    if len(active) == 0:
        active = np.array([np.argmax(profile)])
    
    # Weight by how close to max
    weights = profile[active] - thresh
    weights = np.maximum(weights, 1e-6)
    weights = weights / weights.sum()
    
    # Compute weighted gradient
    grad = np.zeros(n)
    for m in active:
        k = m - (n - 1)
        grad += weights[np.where(active == m)[0][0]] * build_grad_row(h, k, n, dx)
    
    # Remove mean (to preserve sum constraint approximately)
    grad -= grad.mean()
    
    gnorm = np.linalg.norm(grad)
    if gnorm < 1e-15:
        return h, c
    
    grad /= gnorm
    
    # Line search
    best_h = h
    best_c = c
    step = max_step
    for _ in range(n_steps):
        h_try = h - step * grad
        h_try = repair_feasible(h_try)
        s = exact_score(h_try)
        if s < best_c:
            best_h = h_try
            best_c = s
        step *= 0.5
        if step < min_step:
            break
    
    return best_h, best_c

def active_gradient_loop(h, budget_frac=0.5, max_iters=500):
    """Run active gradient descent for a while."""
    t_start = time.perf_counter()
    best_h = h.copy()
    best_c = exact_score(h)

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

def slsqp_polish(h, maxiter=300):
    """SLSQP optimization."""
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    c0 = exact_score(h)
    
    def obj(x):
        return x[-1]
    
    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    
    def eq_fun(x):
        return x[:-1].sum() - n / 2.0
    
    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g
    
    def ineq_fun(x):
        h_x = x[:-1]
        t = x[-1]
        profile = np.correlate(h_x, 1 - h_x, mode="full") * dx
        return t - profile
    
    def ineq_jac(x):
        h_x = x[:-1]
        G = build_grad_matrix(h_x)
        J = np.zeros((M, n + 1))
        J[:, :-1] = -G
        J[:, -1] = 1.0
        return J
    
    x0 = np.concatenate([h, [c0 + 1e-4]])
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 1.0)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None and np.isfinite(res.x).all():
            h_new = repair_feasible(res.x[:-1])
            c_new = exact_score(h_new)
            if c_new <= c0 + 1e-8:
                return h_new
    except Exception:
        pass
    return None

def smooth_max(profile, temperature=0.005):
    """Smooth approximation of max using log-sum-exp."""
    m = np.max(profile)
    return m + temperature * np.log(np.sum(np.exp((profile - m) / temperature)))

def smooth_gradient_descent(h, max_iters=200, temperature=0.003):

    momentum = 0.3

        # Compute smooth gradient

        weights /= weights.sum()
        
        grad = np.zeros(n)
        for m in range(len(profile)):
            k = m - (n - 1)
            grad += weights[m] * build_grad_row(h, k, n, dx)
        
        grad -= grad.mean()
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad /= gnorm
        
        vel = momentum * vel + grad
        vnorm = np.linalg.norm(vel)
        if vnorm > 0:
            vel /= vnorm
        
        # Backtracking line search
        improved = False
        for a in [alpha, alpha*0.5, alpha*0.25, alpha*0.1]:
            h_try = repair_feasible(h - a * vel)
            s = exact_score(h_try)
            if s < best_c - 1e-14:
                h = h_try
                best_c = s
                improved = True
                break
            elif s < exact_score(h):
                h = h_try
                best_c = min(best_c, s)
        
        if improved:
            alpha = min(alpha * 1.05, 0.03)
        else:
            alpha *= 0.7
            if alpha < 1e-6:
                break
    
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def repair_feasible(h):
    n = len(h)
    h = np.clip(h.astype(np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx])
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-14:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-14:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def build_grad_row(h, k, n, dx):
    g = np.zeros(n)
    kk = abs(k)
    if kk == 0:
        g[:] = dx * (1.0 - 2.0 * h)
    elif k > 0:
        g[kk:] += dx * (1.0 - h[0:n-kk])
        g[0:n-kk] += -dx * h[kk:n]
    else:
        g[0:n-kk] += dx * (1.0 - h[kk:n])
        g[kk:n] += -dx * h[0:n-kk]
    return g

def build_grad_matrix(h):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    G = np.zeros((M, n))
    for k in range(-n+1, n):
        m = k + n - 1
        G[m] = build_grad_row(h, k, n, dx)
    return G

def active_step(h, beam, max_step, max_active=25):
    n = len(h)
    dx = 2.0 / n
    profile = compute_profile(h)
    c = float(np.max(profile))
    thresh = c - beam
    active = np.where(profile >= thresh)[0]
    if len(active) == 0:
        active = np.array([np.argmax(profile)])
    if len(active) > max_active:
        idx = np.argsort(profile[active])[-max_active:]
        active = active[idx]
    weights = profile[active] - thresh
    weights = np.maximum(weights, 1e-6)
    weights = weights / weights.sum()
    G_a = np.zeros((len(active), n))
    for i, m in enumerate(active):
        k = m - (n - 1)
        G_a[i] = build_grad_row(h, k, n, dx)
    grad = G_a.T @ weights
    grad -= grad.mean()
    gnorm = np.linalg.norm(grad)
    if gnorm < 1e-15:
        return h, c
    grad /= gnorm
    best_h = h
    best_c = c
    step = max_step
    for _ in range(7):
        h_try = repair_feasible(h - step * grad)
        s = exact_score(h_try)
        if s < best_c:
            best_h = h_try
            best_c = s
        step *= 0.45
    return best_h, best_c

def active_loop(h, time_budget, max_iters=500):
    t0 = time.perf_counter()
    best_h = h.copy()
    best_c = exact_score(h)
    stall_count = 0
    beams_seq = [0.003, 0.006, 0.01, 0.015, 0.025, 0.04, 0.005, 0.008]
    steps_seq = [0.02, 0.01, 0.005, 0.002, 0.015, 0.008, 0.004, 0.001]
    for it in range(max_iters):
        if time.perf_counter() - t0 > time_budget:
            break

    return best_h, best_c

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

def slsqp_polish(h, maxiter=250):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    c0 = exact_score(h)
    def obj(x):
        return x[-1]
    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    def eq_fun(x):
        return x[:-1].sum() - n / 2.0
    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g
    def ineq_fun(x):
        h_x = x[:-1]
        t = x[-1]
        profile = np.correlate(h_x, 1 - h_x, mode="full") * dx
        return t - profile
    def ineq_jac(x):
        h_x = x[:-1]
        G = build_grad_matrix(h_x)
        J = np.zeros((M, n + 1))
        J[:, :-1] = -G
        J[:, -1] = 1.0
        return J
    x0 = np.concatenate([h, [c0 + 1e-3]])
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 2.0)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None and np.isfinite(res.x).all():
            h_new = repair_feasible(res.x[:-1])
            c_new = exact_score(h_new)
            if c_new <= c0 + 1e-8:
                return h_new
    except Exception:
        pass
    return None

def smooth_grad_descent(h, max_iters=150, temp=0.004):
    n = len(h)
    dx = 2.0 / n
    best_h = h.copy()
    best_c = exact_score(h)
    vel = np.zeros(n)
    alpha = 0.008
    mom = 0.4
    for it in range(max_iters):
        profile = compute_profile(best_h)
        pm = np.max(profile)
        w = np.exp((profile - pm) / temp)
        w /= w.sum()
        grad = np.zeros(n)
        for m in range(len(profile)):
            k = m - (n - 1)
            grad += w[m] * build_grad_row(best_h, k, n, dx)
        grad -= grad.mean()
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        grad /= gn
        vel = mom * vel + grad
        vn = np.linalg.norm(vel)
        if vn > 0:
            vel /= vn
        improved = False
        for a in [alpha, alpha*0.5, alpha*0.25]:
            h_try = repair_feasible(best_h - a * vel)
            s = exact_score(h_try)
            if s < best_c - 1e-14:
                best_h = h_try
                best_c = s
                improved = True
                break
        if improved:
            alpha = min(alpha * 1.03, 0.025)
        else:
            alpha *= 0.6
            if alpha < 1e-6:
                break
    return best_h, best_c

def make_sine_h(n, freq, amp, phase=0.0):
