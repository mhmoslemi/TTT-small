import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time


def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h


def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def c5_of_h(h, dx):
    return float(compute_corrs(h).max() * dx)


def grad_corr_k(h, k, n):
    offset = k - (n - 1)
    grad = np.zeros(n)
    if offset == 0:
        grad[:] = 1.0 - 2.0 * h
    elif offset > 0:
        i_lo, i_hi = 0, n - 1 - offset
        if i_lo <= i_hi:
            idx = np.arange(i_lo, i_hi + 1)
            grad[idx + offset] += (1.0 - h[idx])
            grad[idx] -= h[idx + offset]
    else:
        d = -offset
        i_lo, i_hi = 0, n - 1 - d
        if i_lo <= i_hi:
            idx = np.arange(i_lo, i_hi + 1)
            grad[idx] += (1.0 - h[idx + d])
            grad[idx + d] -= h[idx]
    return grad


def smooth_max_val(h, dx, beta):
    return float(logsumexp(beta * compute_corrs(h)) / beta * dx)


def smooth_max_grad(h, dx, beta):
    n = len(h)
    corrs = compute_corrs(h)
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones_like(corrs) / len(corrs)
    else:
        w /= w_sum
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-10:
            continue
        grad += w[k] * grad_corr_k(h, k, n)
    return grad * dx


def slsqp_smooth(h0, dx, target_sum, beta, maxiter):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum,
             'jac': lambda x: np.ones(n)}]

    def obj(x):
        return smooth_max_val(x, dx, beta)

    def g(x):
        return smooth_max_grad(x, dx, beta)

    try:
        res = minimize(obj, h0, jac=g, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'maxls': 40})
        h_out = project_feasible(res.x, target_sum)
    except Exception:
        h_out = project_feasible(h0, target_sum)
    return h_out, c5_of_h(h_out, dx)


def slsqp_active_iter(h0, dx, target_sum, top_k, maxiter_per_round, n_rounds):
    """Iterative active-set: identify top lags, constrain them, repeat."""
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)

    for rnd in range(n_rounds):
        corrs = compute_corrs(h)
        current_max = corrs.max()
        active_idx = np.argsort(corrs)[-top_k:]

        # Minimize t subject to corr_k(h) <= t for active k, sum(h)=target, 0<=h<=1
        nv = n + 1
        full_bounds = [(0.0, 1.0)] * n + [(0.0, None)]

        def obj(y):
            return y[n]

        def grad_obj(y):
            g = np.zeros(nv)
            g[n] = 1.0
            return g

        cons = [{'type': 'eq',
                 'fun': lambda y: y[:n].sum() - target_sum,
                 'jac': lambda y: np.concatenate([np.ones(n), [0.0]])}]
        for k in active_idx:
            kk = int(k)

            def cfun(y, kk=kk):
                return compute_corrs(y[:n])[kk] - y[n]

            def cjac(y, kk=kk):
                hy = y[:n]
                jj = np.zeros(nv)
                jj[:n] = grad_corr_k(hy, kk, n)
                jj[n] = -1.0
                return jj

            cons.append({'type': 'ineq', 'fun': cfun, 'jac': cjac})

        y0 = np.concatenate([h, [current_max + 1e-5]])
        try:
            res = minimize(obj, y0, jac=grad_obj, method='SLSQP',
                           bounds=full_bounds, constraints=cons,
                           options={'maxiter': maxiter_per_round, 'ftol': 1e-13})
            h_try = project_feasible(res.x[:n], target_sum)
        except Exception:
            h_try = h

        c5_try = c5_of_h(h_try, dx)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()
        h = h_try

    return best_h, best_c5


def subgrad_momentum(h0, dx, target_sum, max_iter=20000, top_m=None,
                     alpha0=0.08, gamma=0.9):
    n = len(h0)
    h = h0.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    alpha = alpha0
    no_improve = 0
    if top_m is None:
        top_m = max(8, n // 8)

    for it in range(max_iter):
        corrs = compute_corrs(h)
        c5_val = corrs.max() * dx

        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 4000:
            break

        sorted_idx = np.argsort(corrs)[-top_m:]
        grad = np.zeros(n)
        for k in sorted_idx:
            grad += grad_corr_k(h, k, n)
        grad /= len(sorted_idx)
        grad *= dx

        v = gamma * v - alpha * grad
        h_new = project_feasible(h + v, target_sum)
        new_c5 = c5_of_h(h_new, dx)

        if new_c5 < c5_val:
            h = h_new
            alpha = min(alpha * 1.008, 0.3)
        else:
            alpha *= 0.93
            v *= 0.4

        if alpha < 1e-15:
            break

    return best_h, best_c5


def make_thue_morse(n, target_sum):
    t = np.zeros(n)
    for i in range(n):
        t[i] = float((i & (i >> 1)).bit_count() & 1)
    return project_feasible(t, target_sum)


def make_m_sequence_start(n, target_sum, rng):
    """Create start from a pseudo-random low-correlation pattern."""
    # Use a simple LFSR-like pattern
    period = 1
    while period < n:
        period *= 2
    # m-sequence of length period-1 using a simple tap
    taps = [0, 3]  # characteristic polynomial x^4 + x + 1
    seq = np.zeros(period - 1)
    state = np.ones(4, dtype=int)
    for i in range(period - 1):
        seq[i] = state[0]
        new_bit = state[taps[0]] ^ state[taps[1]]
        state = np.concatenate([[new_bit], state[:-1]])
    # Tile and truncate to n
    h = np.tile(seq, n // len(seq) + 1)[:n]
    return project_feasible(h, target_sum)


def make_strided_pattern(n, target_sum, stride):
    """Create a pattern with values at regular intervals."""
    h = np.zeros(n)
    positions = list(range(0, n, stride))
    n_ones = n // 2
    if len(positions) < n_ones:
        # Fill remaining with random positions
        remaining = n_ones - len(positions)
        other = [i for i in range(n) if i not in set(positions)]
        positions += other[:remaining]
    h[positions[:n_ones]] = 1.0
    return project_feasible(h, target_sum)


def water_fill_iter(h0, dx, target_sum, n_iter=200, step=0.05):
    """Iteratively push down the maximum correlation."""
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)

    for it in range(n_iter):
        corrs = compute_corrs(h)
        max_idx = np.argmax(corrs)
        max_val = corrs[max_val]

        # Zero out tiny gradient components
        grad = grad_corr_k(h, max_idx, n) * dx
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break

        # Try a small step in the negative gradient direction
        h_new = project_feasible(h - step * grad / grad_norm, target_sum)
        new_c5 = c5_of_h(h_new, dx)

        if new_c5 < best_c5:
            h = h_new
            best_h = h.copy()
            best_c5 = new_c5
        else:
            step *= 0.7
            if step < 1e-10:
                break

    return best_h, best_c5
