def eval_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))


def build_sym(q, n):
    m = n // 2
    h = np.empty(n)
    h[:m] = q
    h[m:] = 1.0 - q[::-1]
    return h


def to_sym_q(h, n):
    m = n // 2
    return 0.5 * (h[:m] + 1.0 - h[n-1:m-1:-1])


def select_active(h, margin=0.002, pad=2):
    n = len(h)
    cv = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    mx = cv.max()
    idx = np.where(cv >= mx - margin)[0]
    padded = set()
    for a in idx:
        for d in range(-pad, pad + 1):
            k = int(a) + d
            if 0 <= k <= 2 * n - 2:
                padded.add(k)
    return np.sort(np.array(padded, dtype=int))


def corr_grad_h(h, d, n):
    g = np.zeros(n)
    lo1, hi1 = max(0, d), min(n - 1, n - 1 + d)
    if lo1 <= hi1:
        g[lo1:hi1 + 1] += 1.0 - h[lo1 - d:hi1 + 1 - d]
    lo2, hi2 = max(0, -d), min(n - 1, n - 1 - d)
    if lo2 <= hi2:
        g[lo2:hi2 + 1] += -h[lo2 + d:hi2 + 1 + d]
    return g


def sym_grad(h, d, n):
    m = n // 2
    gh = corr_grad_h(h, d, n)
    return gh[:m] - gh[n-1:m-1:-1]


def project_to_target(x, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(x - mid, 0.0, 1.0).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.any():
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

    n_old = len(h)
    if n_new == n_old:
        return h.copy()
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h)
    return project_to_target(h_new, n_new / 2.0)


def active_solve_sym(q, n, active_k, target, ridge):
    m = n // 2
    h = build_sym(q, n)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    r = target - c[active_k]
    K = len(active_k)
    P = np.empty((K, m))
    for i, k in enumerate(active_k):
        d = k - (n - 1)
        P[i] = sym_grad(h, d, n) * dx
    A = P @ P.T + ridge * np.eye(K)
    try:
        delta = P.T @ np.linalg.solve(A, r)
    except np.linalg.LinAlgError:
        delta = np.linalg.pinv(P) @ r
    return delta

                            break
                    stall = 0
            elif stall >= 7 and stall % 3 == 0:
                # Random kick
                amp = 0.01 * (1.0 + 0.3 * rng.rand())
                kick = rng.randn(m) * amp
                q_k = np.clip(q + kick, 0, 1)
                h_k = build_sym(q_k, n_cur)
                ck = eval_c5(h_k)
                if ck < best_c5:
                    q, best_h, best_c5 = q_k, h_k, ck
                stall = 0

    # Phase 2: Multi-resolution
    n_base = len(best_h)
    n_cands = set()
    for d in [-18, -12, -6, 0, 6, 12, 18, 24]:
        nc = n_base + d
        if nc >= 48 and nc <= 256 and nc % 2 == 0:
            n_cands.add(nc)
    n_cands = sorted(n_cands)

    for n_try in n_cands:
        if tl() < reserve + 20:
            break
        if n_try == n_base:
            continue
        h_try = interp_to(best_h, n_try)
        q_try = to_sym_q(h_try, n_try)
        h_try = build_sym(q_try, n_try)
        c_try = eval_c5(h_try)

        for it2 in range(300):
            if tl() < reserve + 10:
                break
            c2 = eval_c5(h_try)
            if c2 <= GT:
                break
            gap2 = max(c2 - GT, 1e-6)
            margin2 = max(0.0008, min(0.004, gap2 * 0.5))
            act2 = select_active(h_try, margin2, pad=2)
            imp2 = False
            for rf3 in [0.4, 0.2, 0.1, 0.05]:
                if imp2:
                    break
                tgt3 = c2 - rf3 * gap2
                for rg3 in [1e-4, 1e-3]:
                    if imp2:
                        break
                    dlt = active_solve_sym(q_try, n_try, act2, tgt3, rg3)
                    if np.linalg.norm(dlt) < 1e-14:
                        continue
                    qn, cn = ls_sym(q_try, dlt, n_try, c2, al_fine)
                    if cn < eval_c5(h_try) - 1e-13:
                        q_try, h_try = qn, build_sym(qn, n_try)
                        imp2 = True
            if not imp2:
                break

        c_final_try = eval_c5(h_try)
        if c_final_try < best_c5 - 1e-13:
            best_h, best_c5 = h_try, c_final_try

    # Phase 3: Final fine polish
    if tl() > 20 and best_h is not None:
        n_fin = len(best_h)
        if n_fin % 2 != 0:
            n_fin -= 1
            best_h = project_to_target(best_h[:n_fin], n_fin / 2.0)
            best_c5 = eval_c5(best_h)
        q_fin = to_sym_q(best_h, n_fin)
        stall3 = 0

        for it3 in range(500):
            if tl() < 15:
                break
            if best_c5 <= GT:
                break
            c_fin = best_c5
            gap_fin = max(c_fin - GT, 1e-8)
            margin3 = max(0.0004, min(0.002, gap_fin))
            act3 = select_active(best_h, margin3, pad=3)

            imp3 = False
            for rf4 in [0.3, 0.15, 0.08, 0.04, 0.02]:
                if imp3:
                    break
                tgt4 = c_fin - rf4 * gap_fin
                for rg4 in [1e-5, 1e-4, 1e-3]:
                    if imp3:
                        break
                    dlt4 = active_solve_sym(q_fin, n_fin, act3, tgt4, rg4)
                    if np.linalg.norm(dlt4) < 1e-15:
                        continue
                    qn4, cn4 = ls_sym(q_fin, dlt4, n_fin, c_fin, al_fine)
                    if cn4 < best_c5 - 1e-14:
                        q_fin = qn4
                        best_h = build_sym(q_fin, n_fin)
                        best_c5 = cn4
                        imp3 = True

            if not imp3:
                stall3 += 1
                if stall3 > 10:
                    # Try full space one more time
                    act3f = select_active(best_h, margin3, pad=3)
                    dltf = active_solve_full(best_h, n_fin, act3f, c_fin - 0.1 * gap_fin, 1e-4)
                    hf, cf = ls_full(best_h, dltf, n_fin, c_fin, al_fine)
                    if cf < best_c5 - 1e-14:
                        best_h, best_c5 = hf, cf
                        q_fin = to_sym_q(best_h, n_fin)
                    stall3 = 0

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_target(best_h, len(best_h) / 2.0)
    best_c5 = eval_c5(best_h)

    return best_h, best_c5, len(best_h)
