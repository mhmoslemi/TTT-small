```python
import numpy as np
from scipy.optimize import minimize, linprog
from scipy.special import logsumexp
import time

def project_feasible(x, target_sum):
    """Project x onto {h: 0<=h<=1, sum(h)=target_sum}"""
    n = len(x)
    # Bisection for the shift
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    # Fix residual
    diff = target_sum - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = h < 1.0 - 1e-10
            if mask.sum() > 0:
                h[mask] += diff / mask.sum()
        else:
            mask = h > 1e-10
            if mask.sum() > 0:
                h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def c5_of_h(h, dx):
    return compute_corrs(h).max() * dx

def grad_corr_k(h, k, n):
    """Gradient of the k-th correlation w.r.t. h"""
    offset = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -offset)
    i_hi = min(n - 1, n - 1 - offset)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx + offset] += (1.0 - h[idx])
    grad[idx] -= h[idx + offset]
    return grad

def active_set_minimax(h, dx, target, max_iters, alpha_init, top_frac, time_limit):
    """Active-set LP minimax with line search"""
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    alpha = alpha_init
    t_start = time.time()
    
    for it in range(max_iters):
        if time.time() - t_start > time_limit:
            break
        if alpha < 1e-7:
            break
            
        corrs = compute_corrs(h)
        c5_val = corrs.max() * dx
        
        # Active set: top K correlations
        K = max(10, int(top_frac * (2*n - 1)))
        top_idx = np.argsort(corrs)[-K:]
        
        # Build LP
        n_vars = n + 1  # delta (n) + s (1)
        c_obj = np.zeros(n_vars)
        c_obj[-1] = 1.0  # minimize s
        
        A_ub = []
        b_ub = []
        for k in top_idx:
            g = grad_corr_k(h, k, n)
            row = np.zeros(n_vars)
            row[:n] = dx * g
            row[-1] = -1.0
            A_ub.append(row)
            b_ub.append(-corrs[k] * dx - 1e-12)
        
        A_eq = np.zeros((1, n_vars))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)
        
        bounds = []
        for i in range(n):
            bounds.append((-h[i] * alpha, (1.0 - h[i]) * alpha))
        bounds.append((0.0, None))  # s >= 0
        
        try:
            res = linprog(c_obj, A_ub=np.array(A_ub), b_ub=np.array(b_ub),
                         A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
            if res.status != 0:
                alpha *= 0.5
                continue
        except Exception:
            alpha *= 0.5
            continue
        
        delta = res.x[:n]
        
        # Line search
        improved = False
        for t in [1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125]:
            h_try = project_feasible(h + t * delta, target)
            c5_try = c5_of_h(h_try, dx)
            if c5_try < c5_val - 1e-14:
                h = h_try
                improved = True
                if c5_try < best_c5:
                    best_c5 = c5_try
                    best_h = h.copy()
                break
        
        if improved:
            alpha = min(alpha * 1.1, 0.5)  # slightly increase if working
        else:
            alpha *= 0.5
    
    return best_h, best_c5
```

```python
def quantized_sa(n, M, n_steps, t0, t_end, seed, time_limit=60):
    """Quantized simulated annealing"""
    rng = np.random.RandomState(seed)
    target_quant = n * M // 2  # must be integer
    
    # Initialize: all at M/2
    a = np.full(n, M // 2, dtype=int)
    # Adjust sum
    current_sum = a.sum()
    diff = target_quant - current_sum
    if diff > 0:
        idx = rng.choice(n, diff, replace=False)
        a[idx] += 1
```

```python
import numpy as np
from scipy.optimize import minimize, linprog
from scipy.special import logsumexp
import time


def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = h < 1.0 - 1e-10
            if mask.sum() > 0:
                h[mask] += diff / mask.sum()
        else:
            mask = h > 1e-10
            if mask.sum() > 0:
                h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)


def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def c5_of_h(h, dx):
    return compute_corrs(h).max() * dx


def grad_corr_k(h, k, n):
    offset = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -offset)
    i_hi = min(n - 1, n - 1 - offset)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx + offset] += (1.0 - h[idx])
    grad[idx] -= h[idx + offset]
    return grad


def active_set_minimax(h, dx, target, max_iters, alpha_init, top_frac, time_limit):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    alpha = alpha_init
    t_start = time.time()

    for it in range(max_iters):
        if time.time() - t_start > time_limit:
            break
        if alpha < 5e-7:
            break

        corrs = compute_corrs(h)
        c5_val = corrs.max() * dx
        c5_raw = corrs.max()

        K = max(20, int(top_frac * (2 * n - 1)))
        top_idx = np.argsort(corrs)[-K:]

        n_vars = n + 1
        c_obj = np.zeros(n_vars)
        c_obj[-1] = 1.0

        A_ub = []
        b_ub = []
        for k in top_idx:
            g = grad_corr_k(h, k, n)
            row = np.zeros(n_vars)
            row[:n] = dx * g
            row[-1] = -1.0
            A_ub.append(row)
            b_ub.append(-corrs[k] * dx - 1e-12)

        A_eq = np.zeros((1, n_vars))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)

        bounds = []
        for i in range(n):
            bounds.append((-h[i] * alpha, (1.0 - h[i]) * alpha))
        bounds.append((0.0, None))

        try:
            res = linprog(c_obj, A_ub=np.array(A_ub), b_ub=np.array(b_ub),
                          A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
            if res.status != 0:
                alpha *= 0.7
                continue
        except Exception:
            alpha *= 0.7
            continue

        delta = res.x[:n]
        improved = False
        for t in [1.0, 0.7, 0.5, 0.35, 0.25, 0.15, 0.1, 0.05]:
            h_try = project_feasible(h + t * delta, target)
            c5_try = c5_of_h(h_try, dx)
            if c5_try < c5_val - 1e-14:
                h = h_try
                improved = True
                if c5_try < best_c5:

                    best_h = h.copy()
                break

        if improved:
            alpha = min(alpha * 1.05, 0.5)
        else:
            alpha *= 0.6

    return best_h, best_c5


def slsqp_smooth(h0, dx, target, beta, maxiter):
    n = len(h0)

    def obj(x):
        corrs = compute_corrs(x)
        return logsumexp(beta * corrs) / beta * dx

    def grad_obj(x):
        corrs = compute_corrs(x)
        shifted = corrs - corrs.max()
        w = np.exp(beta * shifted)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones_like(corrs) / len(corrs)
        else:
            w /= w_sum
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] > 1e-13:
                grad += w[k] * grad_corr_k(x, k, n)
        return grad * dx

    def c_fun(x):
        return x.sum() - target

    def c_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': c_fun, 'jac': c_jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0, jac=grad_obj, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        h_out = project_feasible(res.x, target)
    except Exception:
        h_out = project_feasible(h0, target)
    return h_out, c5_of_h(h_out, dx)

    return h
```
