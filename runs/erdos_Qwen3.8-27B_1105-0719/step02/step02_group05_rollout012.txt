def minimize_max_quad(a, b, c):
    """Minimize max_i(a_i*t^2 + b_i*t + c_i) over t in [0,1] using dense grid."""
    t_grid = np.linspace(0.0, 1.0, 501)
    t2 = t_grid * t_grid
    vals = a[None, :] * t2[:, None] + b[None, :] * t_grid[:, None] + c[None, :]
    f_vals = vals.max(axis=1)

    i_lo = max(0, i_best - 2)
    i_hi = min(len(t_grid) - 1, i_best + 2)
    lo = t_grid[i_lo]

class SmoothQObjective:
    def __init__(self, n, beta):
        self.n = n
        self.m = n // 2
        self.beta = beta
        self.dx = 2.0 / n

    def value(self, q):
        h = build_h_from_q(q, self.n)
        corrs = np.correlate(h, 1.0 - h, mode='full')
        return logsumexp(self.beta * corrs) / self.beta * self.dx

    def gradient(self, q):
        n = self.n
        m = self.m
        h = build_h_from_q(q, n)
        corrs = np.correlate(h, 1.0 - h, mode='full')
        shifted = corrs - corrs.max()
        w = np.exp(self.beta * shifted)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(len(corrs)) / len(corrs)
        else:
            w /= w_sum
        # Gradient w.r.t. q: for each correlation at lag d,
        # d(corr_d)/dq[p] = d(corr_d)/dh[p] * 1 + d(corr_d)/dh[n-1-p] * (-1)
        # (since h[n-1-p] = 1-q[p], dh[n-1-p]/dq[p] = -1)
        Gq = np.zeros(m)
        n_full = len(corrs)
        for k in range(n_full):
            if w[k] < 1e-14:
                continue
            d = k - (n - 1)
            G = np.zeros(n)
            if d >= 0:
                if d < n:
                    G[d:] += (1.0 - h[:n - d])
                    G[:n - d] += (-h[d:])
            else:
                dd = -d
                if dd < n:
                    G[:n - dd] += (1.0 - h[dd:])
                    G[dd:] += (-h[:n - dd])
            Gq += w[k] * (G[:m] - G[m:][::-1])
        return Gq * self.dx


def smooth_step(q, n, beta, maxiter=200):
    m = n // 2
    obj = SmoothQObjective(n, beta)
    try:
        res = minimize(obj.value, q, jac=obj.gradient, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * m,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-14})
        q_out = np.clip(res.x, 0.0, 1.0)
        return q_out
    except Exception:
        return q.copy()

    def constraint(self, x, k_idx):

        h = build_h_from_q(q, self.n)
        n = self.n
        k = self.active_k[k_idx]
        d = k - (n - 1)
        if d >= 0:
            i_vals = np.arange(0, n - d)
            val = np.sum(h[i_vals + d] * (1.0 - h[i_vals]))
        else:
            dd = -d
            i_vals = np.arange(dd, n)
            val = np.sum(h[i_vals] * (1.0 - h[i_vals - dd]))
        return t - val * self.dx

    def constraint_grad(self, x, k_idx):
        q = x[1:]
        h = build_h_from_q(q, self.n)
        n = self.n
        m = self.m
        k = self.active_k[k_idx]
        d = k - (n - 1)
        g = np.zeros(1 + m)
        g[0] = 1.0
        G = np.zeros(n)
        if d >= 0:
            if d < n:
                G[d:] += (1.0 - h[:n - d])
                G[:n - d] += (-h[d:])
        else:
            dd = -d
            if dd < n:
                G[:n - dd] += (1.0 - h[dd:])
                G[dd:] += (-h[:n - dd])
        g[1:] = -(G[:m] - G[m:][::-1]) * self.dx
        return g


def active_set_solve(q, n, active_k, t0, maxiter=300):
    m = n // 2
    x0 = np.concatenate([[t0], q])
    ep = EpigraphQSolver(n, active_k)
    cons = []
    for i in range(len(active_k)):
        cons.append({
            'type': 'ineq',
            'fun': lambda x, i=i: ep.constraint(x, i),
            'jac': lambda x, i=i: ep.constraint_grad(x, i)
        })
    bounds = [(0.0, 5.0)] + [(0.0, 1.0)] * m
    try:
        res = minimize(ep.obj, x0, jac=ep.obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        q_out = np.clip(res.x[1:], 0.0, 1.0)
        return q_out, res.x[0]
    except Exception:
        return q.copy(), t0


def active_set_search(q, n, t_start):
    m = n // 2
    dx = 2.0 / n
    h = build_h_from_q(q, n)
    corrs = np.correlate(h, 1.0 - h, mode='full')
    current_max = corrs.max() * dx
    L_init = min(25, 2 * n - 1)
    top_idx = np.argsort(corrs)[::-1][:L_init]
    active_k = sorted(top_idx.tolist())

    best_q = q.copy()
    best_c5 = current_max
    t0 = current_max * (1.0 + 1e-4) + 1e-6

    for rnd in range(12):
        q_new, t_solved = active_set_solve(q, n, active_k, t0, maxiter=250)
        h_new = build_h_from_q(q_new, n)
        corrs_new = np.correlate(h_new, 1.0 - h_new, mode='full')
        c5_new = corrs_new.max() * dx

        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_q = q_new.copy()

        hidden_mask = corrs_new > t_solved + 1e-7
        if hidden_mask.sum() > 0:
            hidden_idx = np.where(hidden_mask)[0]
            new_active = sorted(set(active_k) | set(hidden_idx.tolist()))
            if len(new_active) > 50:
                vals = [corrs_new[kk] for kk in new_active]
                order = np.argsort(vals)[::-1][:50]
                new_active = sorted([new_active[oo] for oo in order])
            active_k = new_active

        t0 = max(t_solved, c5_new) * (1.0 + 1e-5) + 1e-6
        q = q_new

        if c5_new - best_c5 < 1e-12 and hidden_mask.sum() == 0:
            break

    return best_q, best_c5


def project_feasible(x, target_sum):
    n = len(x)
    lo_b, hi_b = -2.0, 2.0
    for _ in range(80):
        mid = (lo_b + hi_b) / 2.0
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo_b = mid
        else:
            hi_b = mid
    h = np.clip(x - (lo_b + hi_b) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)


def random_perturbation(q, rng, scale):
    q_new = q.copy()
    q_new += rng.uniform(-scale, scale, size=len(q_new))
    return np.clip(q_new, 0.0, 1.0)


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)

    def elapsed():
        return time.time() - t_start

    def remaining():
        return budget_s - elapsed()

    # Load parent
    h_parent = np.array(initial_h_values, dtype=float)
    n_parent = len(h_parent)
    m_parent = n_parent // 2

    # Ensure it's even for the anti-symmetric representation
    if n_parent % 2 != 0:
        h_parent = np.delete(h_parent, n_parent // 2)
        n_parent = len(h_parent)
        m_parent = n_parent // 2

    q = q_from_h(h_parent, n_parent).copy()
    n = n_parent
    m = m_parent

    h_init = build_h_from_q(q, n)
    best_c5 = exact_c5(h_init)
    best_q = q.copy()
    best_n = n

    def try_accept(q_candidate, n_candidate):
        nonlocal best_q, best_c5, best_n
        h_cand = build_h_from_q(q_candidate, n_candidate)
        c5_cand = exact_c5(h_cand)
        if c5_cand < best_c5 - 1e-14:
            best_c5 = c5_cand
            best_q = q_candidate.copy()
            best_n = n_candidate
        return c5_cand

    # Phase 1: Coordinate descent on parent grid
    if remaining() > 30:
        q, c5_cd = coord_descent(q, n, n_passes=50, max_time=remaining() - 10, t_start=t_start)
        try_accept(q, n)

    # Phase 2: Smooth step to escape local minimum
    if remaining() > 30:
        for beta in [100, 300, 800]:
            if remaining() < 15:
                break
            q_smooth = smooth_step(best_q, n, beta, maxiter=150)
            try_accept(q_smooth, n)

    # Phase 3: More coordinate descent after smooth step
    if remaining() > 20:
        q, c5_cd2 = coord_descent(best_q, n, n_passes=80, max_time=remaining() - 5, t_start=t_start)
        try_accept(q, n)

    # Phase 4: Random perturbation + coordinate descent (restart)
    n_restarts = 5
    perturb_scales = [0.1, 0.05, 0.15, 0.03, 0.08]
    for r in range(n_restarts):
        if remaining() < 20:
            break
        scale = perturb_scales[r % len(perturb_scales)]
        q_pert = random_perturbation(best_q, rng, scale)
        q_new, c5_new = coord_descent(q_pert, n, n_passes=40,
                                      max_time=min(remaining() - 10, 100), t_start=t_start)
        try_accept(q_new, n)

    # Phase 5: Active-set epigraph
    if remaining() > 30:
        h_cur = build_h_from_q(best_q, best_n)
        corrs_cur = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        t0_es = corrs_cur.max() * (2.0 / best_n) * (1.0 + 1e-3) + 1e-5
        q_es, c5_es = active_set_search(best_q, best_n, t_start)
        try_accept(q_es, best_n)

    # Phase 6: Multi-resolution - try different n values
    n_trials = [60, 66, 72, 84, 96, 110, 126]
    n_trials = [nt for nt in n_trials if nt != best_n and nt % 2 == 0]
    for n_try in sorted(n_trials):
        if remaining() < 30:
            break
        # Interpolate best solution to new grid
        x_old = np.linspace(0.0, 2.0, best_n)
        x_new = np.linspace(0.0, 2.0, n_try)
        h_best = build_h_from_q(best_q, best_n)
        h_interp = np.interp(x_new, x_old, h_best)
        # Enforce anti-symmetry
        m_try = n_try // 2
        q_interp = (h_interp[:m_try] + 1.0 - h_interp[n_try - 1 :: -1][:m_try]) / 2.0
        q_interp = np.clip(q_interp, 0.0, 1.0)

        # Coordinate descent on new grid
        q_new, c5_new = coord_descent(q_interp, n_try, n_passes=40,
                                      max_time=min(remaining() - 20, 80), t_start=t_start)
        c5_test = try_accept(q_new, n_try)

        # If improved, do a smooth step + more CD
        if c5_test < best_c5 + 1e-12 and remaining() > 20:
            q_s = smooth_step(q_new, n_try, 300, maxiter=100)
            try_accept(q_s, n_try)
            q_cd3, _ = coord_descent(best_q if best_n == n_try else q_s,
                                     n_try if best_n == n_try else n_try,
                                     n_passes=30,
                                     max_time=min(remaining() - 10, 60), t_start=t_start)
            try_accept(q_cd3, n_try)

    # Phase 7: Final coordinate descent polish
    if remaining() > 10:
        q_final, _ = coord_descent(best_q, best_n, n_passes=100,
                                   max_time=remaining() - 3, t_start=t_start)
        try_accept(q_final, best_n)

    # Phase 8: One more smooth + CD cycle if time permits
    if remaining() > 20:
        q_s_final = smooth_step(best_q, best_n, 500, maxiter=100)
        try_accept(q_s_final, best_n)
        if remaining() > 10:
            q_cd_final, _ = coord_descent(best_q, best_n, n_passes=60,
                                          max_time=remaining() - 3, t_start=t_start)
            try_accept(q_cd_final, best_n)

    # Final validation
    h_out = build_h_from_q(best_q, best_n)
    h_out = np.clip(h_out, 0.0, 1.0)
    final_c5 = exact_c5(h_out)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return h_out, best_c5, best_n
