def nm_polish(q0, n, maxiter=2000):
    m = n // 2

    def obj(q):
        return exact_score(make_sym_h(q, n))

    simplex = q0 + 0.01 * np.eye(m)
    simplex = np.vstack([q0.reshape(1, -1)] + [s.reshape(1, -1) for s in simplex])
    simplex = np.clip(simplex, 0.0, 1.0)
    try:
        res = minimize(obj, q0, method='Nelder-Mead',
                       options={'maxiter': maxiter, 'fatol': 1e-15, 'xatol': 1e-15,
                                'adaptive': True})
        h_new = make_sym_h(res.x, n)
        c5 = exact_score(h_new)
        return res.x, c5
    except Exception:
        return None, None


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()

    def time_left():
        return budget_s - (time.time() - t0)

    rng = np.random.RandomState(seed)

    try:
        h_parent = np.array(initial_h_values, dtype=float)
    except NameError:
        h_parent = 0.5 * np.ones(126)

    n = len(h_parent)
    if n % 2 != 0:
        h_parent = h_parent[:n - 1]
        n = len(h_parent)

    h_best = project_feasible(h_parent, n / 2.0)
    c5_best = exact_score(h_best)
    m = n // 2
    q_best = make_q_from_h(h_best, n)
    print(f"Initial: n={n}, C5={c5_best:.7f}")

    # Phase 1: L-BFGS-B with increasing beta
    q_curr = q_best.copy()
    for beta in [40, 80, 150, 300, 600, 1200, 2500, 5000, 10000]:
        if time_left() < 25:
            break
        q_new, c5_new = lbeb_optim(q_curr, n, beta, maxiter=600)
        if q_new is not None:
            q_curr = q_new
            if c5_new < c5_best - 1e-12:
                c5_best = c5_new
                q_best = q_new
                h_best = make_sym_h(q_best, n)
                print(f"  L-BFGS-B beta={beta}: C5={c5_best:.7f}")

    # Phase 2: Power mean optimization
    if time_left() > 40:
        for p in [20, 50, 100, 200]:
            if time_left() < 20:
                break
            q_new, c5_new = power_mean_optim(q_best, n, p, maxiter=400)
            if q_new is not None and c5_new < c5_best - 1e-12:
                c5_best = c5_new
                q_best = q_new
                h_best = make_sym_h(q_best, n)
                print(f"  PowerMean p={p}: C5={c5_best:.7f}")

    # Phase 3: Subgradient descent
    if time_left() > 50:
        q_sg, c5_sg = subgrad_descent(q_best, n, n_iter=500, init_step=0.008)
        if c5_sg < c5_best - 1e-12:
            c5_best = c5_sg
            q_best = q_sg
            h_best = make_sym_h(q_best, n)
            print(f"  Subgradient: C5={c5_best:.7f}")
        # Re-run L-BFGS-B from subgradient result
        if time_left() > 20:
            q_new, c5_new = lbeb_optim(q_sg, n, 3000, maxiter=400)
            if q_new is not None and c5_new < c5_best - 1e-12:
                c5_best = c5_new
                q_best = q_new
                h_best = make_sym_h(q_best, n)
                print(f"  Post-SG L-BFGS-B: C5={c5_best:.7f}")

    # Phase 4: Random restarts
    n_restarts = 12
    for trial in range(n_restarts):
        if time_left() < 20:
            break
        scale = 0.02 + 0.15 * (trial / n_restarts)
        q_pert = q_best + scale * rng.randn(m)
        q_pert = np.clip(q_pert, 0.0, 1.0)
        q_new, c5_new = lbeb_optim(q_pert, n, 200, maxiter=300)
        if q_new is not None:
            q_new2, c5_new2 = lbeb_optim(q_new, n, 2000, maxiter=300)
            if q_new2 is not None and c5_new2 < c5_new:
                q_new = q_new2
                c5_new = c5_new2
        if q_new is not None and c5_new < c5_best - 1e-12:
            c5_best = c5_new
            q_best = q_new
            h_best = make_sym_h(q_best, n)
            print(f"  Restart {trial}: C5={c5_best:.7f}")

    # Phase 5: Nelder-Mead polish
    if time_left() > 25:
        q_new, c5_new = nm_polish(q_best, n, maxiter=1500)
        if q_new is not None and c5_new < c5_best - 1e-10:
            c5_best = c5_new
            q_best = q_new
            h_best = make_sym_h(q_best, n)
            print(f"  Nelder-Mead: C5={c5_best:.7f}")

    # Phase 6: Try different n values
    if time_left() > 50:
        for delta in [-4, 4, -2, 2, -6, 6, 8, -8, 10, -10, 12]:
            if time_left() < 25:
                break
            n_try = n + delta
            if n_try < 50 or n_try % 2 != 0:
                continue
            x_old = np.linspace(0, 2, n, endpoint=False)
            x_new = np.linspace(0, 2, n_try, endpoint=False)
            h_interp = np.interp(x_new, x_old, h_best)
            h_interp = project_feasible(h_interp, n_try / 2.0)
            q_try = make_q_from_h(h_interp, n_try)
            q_new, c5_new = lbeb_optim(q_try, n_try, 100, maxiter=300)
            if q_new is not None:
                q_new2, c5_new2 = lbeb_optim(q_new, n_try, 1000, maxiter=400)
                if q_new2 is not None and c5_new2 < c5_new:
                    q_new = q_new2
                    c5_new = c5_new2
            if q_new is not None and c5_new < c5_best - 1e-9:
                c5_best = c5_new
                n = n_try
                m = n // 2
                q_best = q_new
                h_best = make_sym_h(q_best, n)
                print(f"  n={n}: C5={c5_best:.7f}")

    # Phase 7: Final high-iteration L-BFGS-B
    if time_left() > 15:
        q_new, c5_new = lbeb_optim(q_best, n, 8000, maxiter=800)
        if q_new is not None and c5_new < c5_best - 1e-12:
            c5_best = c5_new
            q_best = q_new
            h_best = make_sym_h(q_best, n)
            print(f"  Final L-BFGS-B: C5={c5_best:.7f}")

    # Final validation
    h_best = np.clip(h_best, 0.0, 1.0)
    h_best = project_feasible(h_best, len(h_best) / 2.0)
    c5_best = exact_score(h_best)
    print(f"Final: n={len(h_best)}, C5={c5_best:.7f}")

    return h_best, c5_best, len(h_best)
