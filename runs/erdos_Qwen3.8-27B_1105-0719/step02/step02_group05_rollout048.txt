def project_feasible(x, target_sum):
    """Project x onto {0 ≤ x ≤ 1, sum(x) = target_sum}"""
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2

        else:

    if abs(diff) > 1e-10:

            h = np.clip(h, 0.0, 1.0)
    return h


def grad_corr_d(h, d, n):
    """Gradient of correlation at lag d"""
    g = np.zeros(n)
    if d >= 0:
        if d < n:
            g[d:] += (1.0 - h[:n - d])
            g[:n - d] += (-h[d:])
    else:
        dd = -d
        if dd < n:
            g[:n - dd] += (1.0 - h[dd:])
            g[dd:] += (-h[:n - dd])
    return g


def smooth_value_grad(h, beta, dx):
    """Log-sum-exp smooth approximation of max(corrs)*dx"""
    corrs = compute_corrs(h)
    val = logsumexp(beta * corrs) / beta * dx
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(len(corrs)) / len(corrs)
    else:
        w /= w_sum
    G = np.zeros(len(h))
    n = len(h)
    for k in range(len(corrs)):
        if w[k] < 1e-14:
            continue
        d = k - (n - 1)
        G += w[k] * grad_corr_d(h, d, n)
    return val, G * dx


def smooth_solve(h0, beta, target_sum, maxiter=1000):
    """Minimize smooth approximation using L-BFGS-B"""
    n = len(h0)
    dx = 2.0 / n
    
    def f(x):
        h = np.clip(x, 0.0, 1.0)
        return smooth_value_grad(h, beta, dx)[0]
    
    def g(x):
        h = np.clip(x, 0.0, 1.0)
        return smooth_value_grad(h, beta, dx)[1]
    
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(f, h0, jac=g, method='L-BFGS-B',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12})
        return project_feasible(res.x, target_sum)
    except Exception:
        return project_feasible(h0, target_sum)


def active_corr_value(h, d, n, dx):
    """Compute correlation at specific lag d"""
    if d >= 0:
        if d >= n:
            return 0.0
        return np.sum(h[d:] * (1.0 - h[:n-d])) * dx
    else:
        dd = -d
        if dd >= n:
            return 0.0
        return np.sum(h[dd:] * (1.0 - h[:-dd])) * dx


def active_solve(h0, active_k, target_sum, maxiter=500):

    n = len(h0)
    dx = 2.0 / n
    
    class ActiveSet:
        def __init__(self, n, active_k, target_sum, dx):
            self.n = n
            self.active_k = active_k
            self.target_sum = target_sum
            self.dx = dx
        
        def obj(self, x):
            return x[0]
        
        def obj_grad(self, x):
            g = np.zeros(self.n + 1)
            g[0] = 1.0
            return g
        
        def constraint_k(self, x, idx):
            t = x[0]
            h = x[1:]
            d = self.active_k[idx] - (self.n - 1)
            val = active_corr_value(h, d, self.n, self.dx)
            return t - val
        
        def constraint_k_grad(self, x, idx):
            h = x[1:]
            d = self.active_k[idx] - (self.n - 1)
            g = np.zeros(self.n + 1)
            g[0] = 1.0
            G = grad_corr_d(h, d, self.n)
            g[1:] = -G * self.dx
            return g
        
        def sum_constraint(self, x):
            return x[1:].sum() - self.target_sum
        
        def sum_grad(self, x):
            g = np.zeros(self.n + 1)
            g[1:] = np.ones(self.n)
            return g
    
    ep = ActiveSet(n, active_k, target_sum, dx)
    
    # Initial t: max of active correlations
    corrs = compute_corrs(h0)
    t0 = max(corrs[k] * dx for k in active_k) + 1e-6
    
    x0 = np.concatenate([[t0], h0])
    
    cons = []
    for i in range(len(active_k)):
        cons.append({
            'type': 'ineq',
            'fun': lambda x, i=i: ep.constraint_k(x, i),
            'jac': lambda x, i=i: ep.constraint_k_grad(x, i)
        })
    cons.append({'type': 'eq', 'fun': ep.sum_constraint, 'jac': ep.sum_grad})

                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})

        return h_out, res.x[0]
    except Exception:

        c5_y = exact_score(y)
        if c5_y < best_c5 - 1e-15:
            best_c5 = c5_y
            best_h = y.copy()

        h = h_new

    return best_h, best_c5


def top_k_subgradient_polish(h0, target_sum, n_steps=300, k=3, lr_scale=0.005):
    """Polish using weighted average of top-k subgradients."""
    n = len(h0)
    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    best_c5 = exact_score(h)

    for step in range(n_steps):
        corrs = compute_corrs(h)
        vals = corrs * dx
        top_idx = np.argsort(vals)[::-1][:k]
        top_vals = vals[top_idx]
        # Exponentially weight
        w = np.exp(50.0 * (top_vals - top_vals[0]))
        w /= w.sum()

        G = np.zeros(n)
        for wi, idx in zip(w, top_idx):
            d = idx - (n - 1)
            G += wi * grad_corr_d(h, d, n)
        G *= dx

        lr = lr_scale / (1.0 + step * 0.02)
        h = h - lr * G
        h = project_feasible(h, target_sum)

        c5 = exact_score(h)
        if c5 < best_c5 - 1e-15:
            best_c5 = c5
            best_h = h.copy()

    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()

    def elapsed():
        return time.time() - t_start

    def remaining():
        return budget_s - elapsed()

    h_parent = np.array(initial_h_values, dtype=float)
    n = len(h_parent)
    target_sum = n / 2.0
    h_parent = project_feasible(h_parent, target_sum)

    best_h = h_parent.copy()
    best_c5 = exact_score(h_parent)
    rng = np.random.RandomState(seed)

    # Phase 1: Smooth optimization with increasing beta
    for beta in [40, 80, 150, 300, 600, 1200]:
        if remaining() < 25:
            break
        h_s = smooth_solve(best_h, beta, target_sum, maxiter=500)
        c5_s = exact_score(h_s)
        if c5_s < best_c5 - 1e-14:
            best_c5 = c5_s
            best_h = h_s

    # Phase 2: Subgradient polish on exact max
    if remaining() > 40:
        h_sg, c5_sg = projected_subgradient_polish(best_h, target_sum, n_steps=300, lr_scale=0.015)
        if c5_sg < best_c5 - 1e-14:
            best_c5 = c5_sg
            best_h = h_sg

    # Phase 3: Active-set optimization
    if remaining() > 80:
        h_as, c5_as = active_set_optimize(best_h, target_sum, max_rounds=12, max_active=30, tol=0.002)
        if c5_as < best_c5 - 1e-14:
            best_c5 = c5_as
            best_h = h_as

    # Phase 4: Top-k subgradient polish
    if remaining() > 40:
        h_topk, c5_topk = top_k_subgradient_polish(best_h, target_sum, n_steps=400, k=3, lr_scale=0.004)
        if c5_topk < best_c5 - 1e-14:
            best_c5 = c5_topk
            best_h = h_topk

    # Phase 5: Smooth polish with high beta
    if remaining() > 35:
        for beta in [1500, 3000, 5000]:
            if remaining() < 20:
                break
            h_p = smooth_solve(best_h, beta, target_sum, maxiter=600)
            c5_p = exact_score(h_p)
            if c5_p < best_c5 - 1e-14:
                best_c5 = c5_p
                best_h = h_p

    # Phase 6: Another active-set round with tighter tolerance
    if remaining() > 70:
        h_as2, c5_as2 = active_set_optimize(best_h, target_sum, max_rounds=12, max_active=25, tol=0.001)
        if c5_as2 < best_c5 - 1e-14:
            best_c5 = c5_as2
            best_h = h_as2

    # Phase 7: More subgradient polish
    if remaining() > 40:
        h_sg2, c5_sg2 = projected_subgradient_polish(best_h, target_sum, n_steps=400, lr_scale=0.008)
        if c5_sg2 < best_c5 - 1e-14:
            best_c5 = c5_sg2
            best_h = h_sg2

    # Phase 8: Random perturbations + quick optimization
    if remaining() > 100:
        for trial in range(4):
            if remaining() < 50:
                break
            noise_scale = 0.008 * (trial + 1)
            noise = rng.randn(n) * noise_scale
            h_pert = project_feasible(best_h + noise, target_sum)

            # Quick smooth + subgradient
            h_q = smooth_solve(h_pert, 500, target_sum, maxiter=300)
            h_q, c5_q = projected_subgradient_polish(h_q, target_sum, n_steps=200, lr_scale=0.01)
            c5_q = exact_score(h_q)
            if c5_q < best_c5 - 1e-14:
                best_c5 = c5_q
                best_h = h_q

    # Phase 9: Final active-set + polish
    if remaining() > 50:
        h_fin, c5_fin = active_set_optimize(best_h, target_sum, max_rounds=10, max_active=20, tol=0.001)
        if c5_fin < best_c5 - 1e-14:
            best_c5 = c5_fin
            best_h = h_fin

    if remaining() > 30:
        h_fin2, c5_fin2 = top_k_subgradient_polish(best_h, target_sum, n_steps=300, k=2, lr_scale=0.003)
        if c5_fin2 < best_c5 - 1e-14:
            best_c5 = c5_fin2
            best_h = h_fin2

    if remaining() > 20:
        h_fin3 = smooth_solve(best_h, 5000, target_sum, maxiter=400)
        c5_fin3 = exact_score(h_fin3)
        if c5_fin3 < best_c5 - 1e-14:
            best_c5 = c5_fin3
            best_h = h_fin3

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_feasible(best_h, target_sum)
    final_score = exact_score(best_h)
    if final_score < best_c5:
        best_c5 = final_score

    return best_h, best_c5, len(best_h)
