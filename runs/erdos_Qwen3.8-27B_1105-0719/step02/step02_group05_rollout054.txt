```python
from scipy.special import logsumexp
import time

# All helper functions at top level

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corrs = np.correlate(h, 1.0 - h, mode='full')
    return corrs.max() * dx, corrs

def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def grad_corr_d(h, d, n):
    g = np.zeros(n)
    if d >= 0:
        if d < n:
            g[d:] += (1.0 - h[:n - d])
            g[:n - d] += (-h[d:])
    else:
        dd = -d
        if dd < n:
            g[:n - dd] += (1.0 - h[dd:])
            g[dd:] += (-h[:n - dd])
    return g

def build_h_from_q(q, n):
    m = n // 2
    if n % 2 == 0:
        return np.concatenate([q, 1.0 - q[::-1]])
    else:
        return np.concatenate([q, [0.5], 1.0 - q[::-1]])

def q_from_h(h, n):
    return h[:n // 2].copy()

def smooth_q_value_grad(q, n, beta):
    m = n // 2
    h = build_h_from_q(q, n)
    dx = 2.0 / n
    corrs = compute_corrs(h)
    val = logsumexp(beta * corrs) / beta * dx
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(len(corrs)) / len(corrs)
    else:
        w /= w_sum
    Gq = np.zeros(m)
    for k in range(len(corrs)):
        if w[k] < 1e-14:
            continue
        d = k - (n - 1)
        G = grad_corr_d(h, d, n)
        Gq += w[k] * (G[:m] - G[m:][::-1])
    return val, Gq * dx
```

```python
class ActiveSetQ:
    """Active set solver in q-space for the Erdos problem."""
    def __init__(self, n, active_k):
        self.n = n
        self.m = n // 2
        self.active_k = list(active_k)
        self.dx = 2.0 / n
    
    def obj(self, x):
        return x[0]
    
    def obj_grad(self, x):
        g = np.zeros(self.m + 1)
        g[0] = 1.0
        return g
    
    def _build_h(self, q):
        return build_h_from_q(q, self.n)
    
    def _corr_val(self, h, k):
        d = k - (self.n - 1)
        n = self.n
        if d >= 0:
            return np.sum(h[d:] * (1.0 - h[:n-d])) * self.dx
        else:
            dd = -d
            return np.sum(h[dd:] * (1.0 - h[:n-dd])) * self.dx
    
    def _constraint_fun(self, x, idx):
        t = x[0]
        q = x[1:]
        h = self._build_h(q)
        k = self.active_k[idx]
        return t - self._corr_val(h, k)
    
    def _constraint_jac(self, x, idx):
        q = x[1:]
        h = self._build_h(q)
        k = self.active_k[idx]
        d = k - (self.n - 1)
        G = grad_corr_d(h, d, self.n)
        g = np.zeros(self.m + 1)
        g[0] = 1.0
        g[1:] = -(G[:self.m] - G[self.m:][::-1]) * self.dx
        return g
```

Let me think about the coordinate descent polish. At the optimum, we want to minimize max_k c_k(h) subject to 0≤h≤1, sum(h)=n/2. 

In coordinate descent, we fix all h[j] for j≠i and optimize h[i]. The effect on c_k is:
c_k = Σ_j h[j](1-h[j+d]) 

The terms involving h[i] are:
- h[i](1-h[i+d]) (when i+d is in range)
- h[i-d](1-h[i]) = h[i-d] - h[i-d]*h[i] (when i-d is in range)

So c_k as a function of h[i] is:
c_k(h[i]) = (terms not involving h[i]) + h[i](1-h[i+d]) + h[i-d](1-h[i])
= const + h[i] - h[i]*h[i+d] + h[i-d] - h[i-d]*h[i]
= const + h[i](1 - h[i+d] - h[i-d]) + h[i-d]

This is a quadratic in h[i]: -h[i+d]*h[i]² + (1-h[i+d]-h[i-d])*h[i] + const

Wait, that's not right. Let me redo:
c_k = Σ_j h[j](1-h[j+d])

Terms with h[i]:
- j=i: h[i](1-h[i+d]) if i+d < n
- j=i-d: h[i-d](1-h[i]) if i-d >= 0

So c_k(h[i]) = [Σ_{j≠i, j≠i-d} h[j](1-h[j+d])] + h[i](1-h[i+d]) + h[i-d](1-h[i])

= A_k + h[i] - h[i]*h[i+d] + h[i-d] - h[i-d]*h[i]

where A_k = Σ_{j≠i, j≠i-d} h[j](1-h[j+d])

= A_k + h[i-d] + h[i](1 - h[i+d] - h[i-d])

This is LINEAR in h[i]! (The h[i]*h[i+d] term: h[i+d] is fixed, so this is linear in h[i].)

Wait: h[i](1-h[i+d]) = h[i] - h[i]*h[i+d]. Since h[i+d] is a fixed constant (we're only varying h[i]), this is h[i] * (1-h[i+d]) which IS linear in h[i].

Similarly h[i-d](1-h[i]) = h[i-d] - h[i-d]*h[i] = h[i-d] * (1-h[i]), which is also linear in h[i].

So c_k is linear in h[i] for fixed other coordinates! This means the max of linear functions is convex in h[i], and we can find the optimal h[i] by finding the point where the max is minimized (the "median" in some sense).

More precisely, for each k:
c_k(h[i]) = a_k * h[i] + b_k

where:
a_k = (1 - h[(i+1)%n + d]) * dx if appropriate... 

Hmm, let me be more careful. For a given lag d (corresponding to correlation index k = n-1+d):

c_k = Σ_{j=0}^{n-1-|d|} h[j+|d|] * (1-h[j]) * dx (for d ≥ 0)

The terms involving h[i]:
- h[i] appears as the "first factor" when j+|d| = i, i.e., j = i-|d| (if 0 ≤ i-|d| ≤ n-1-|d|)
- h[i] appears as the "second factor" (inside 1-h[j]) when j = i (if 0 ≤ i ≤ n-1-|d|)

So:
c_k(h[i]) = [terms not involving h[i]] + h[i] * (1-h[i-|d|]) * dx (if valid) + h[i+|d|] * (-h[i]) * dx (if valid)

= const_k + h[i] * [(1-h[i-|d|]) - h[i+|d|]] * dx (with appropriate boundary conditions)

So c_k is indeed linear in h[i], with slope:
slope_k = [(1 - h[i-|d|]) - h[i+|d|]] * dx (with boundary handling)

And the objective max_k c_k is the max of linear functions of h[i], which is convex. We want to minimize this convex function over h[i] ∈ [0,1], subject to the sum constraint.

Without the sum constraint, the optimal h[i] is the point where the "upper envelope" of these lines is minimized, which can be found by binary search on the value.

With the sum constraint, it's more complex, but we can handle it by adjusting h[i] and compensating elsewhere.

Actually, for a simple coordinate descent, let's just ignore the sum constraint temporarily (it will be maintained approximately by the projection step) and optimize each h[i] independently.

For each i, the optimal h[i] minimizes max_k (a_k * h[i] + b_k) over h[i] ∈ [0,1].

This is a 1D convex optimization problem. The minimum occurs where the maximum of the lines is minimized, which is at the intersection of the two "extreme" lines (one with positive slope, one with negative slope).

We can find this by:
1. Computing all (a_k, b_k) for k=0,...,2n-2
2. Finding h[i] that minimizes max_k (a_k * h[i] + b_k)

The minimum of max of linear functions occurs at a vertex, which is the intersection of two lines. We can find it by:
- For each pair of lines, find their intersection
- Check if that point is in [0,1]
- Evaluate the max at all such points
- Take the best

But this is O(n^2) per coordinate, which is too slow for n=126 and 126 coordinates.

Alternative: Use the fact that the minimum of max of linear functions can be found by binary search on the value t. For a given t, the constraint a_k * h[i] + b_k ≤ t for all k defines an interval for h[i]. The optimal h[i] is any point in the intersection of all these intervals (if non-empty), or the point closest to the intersection.

For a_k > 0: h[i] ≤ (t - b_k) / a_k
For a_k < 0: h[i] ≥ (t - b_k) / a_k
For a_k = 0: need b_k ≤ t

So for a given t, the feasible h[i] is [max_{a_k<0} (t-b_k)/a_k, min_{a_k>0} (t-b_k)/a_k] ∩ [0,1].

We want the smallest t such that this interval is non-empty. This can be found by binary search on t.

This is O(n log n) per coordinate (binary search on t, each iteration O(n)), and O(n^2 log n) total for all coordinates. For n=126, that's about 126^2 * 7 ≈ 113,000 operations, which is very fast!

```python
def optimize_single_coord(h, i, n):
    """Find optimal h[i] to minimize max_k c_k."""
    dx = 2.0 / n
    n_lags = 2 * n - 1
    slopes = np.zeros(n_lags)
    intercepts = np.zeros(n_lags)
    
    ad_arr = np.arange(n_lags) - (n - 1)  # -n+1, ..., n-1
    ad_abs = np.abs(ad_arr)
    
    for k in range(n_lags):
        ad = ad_abs[k]
        if ad >= n:
            continue
        s = 1.0
        if i - ad >= 0:
            s -= h[i - ad]
        if i + ad < n:
            s -= h[i + ad]
        slopes[k] = s * dx
        # Current correlation value
        if ad == 0:
            ck = np.sum(h * (1.0 - h)) * dx
        else:
            ck = np.sum(h[ad:] * (1.0 - h[:n-ad])) * dx
        intercepts[k] = ck - h[i] * slopes[k]
    
    # Binary search on t
    # Lower bound: min over x in [0,1] of max_k (slopes[k]*x + intercepts[k])
    # Upper bound: max_k intercepts[k] (value at x=0) or max_k (slopes[k]+intercepts[k]) (value at x=1)
    val0 = np.max(intercepts)
    val1 = np.max(slopes + intercepts)
    lo_t = min(val0, val1)
    hi_t = max(val0, val1)
    
    for _ in range(50):
        mid_t = (lo_t + hi_t) / 2
        # Feasible h[i] must satisfy: slopes[k]*x + intercepts[k] <= mid_t for all k
        # => x <= (mid_t - intercepts[k]) / slopes[k] for slopes[k] > 0
        # => x >= (mid_t - intercepts[k]) / slopes[k] for slopes[k] < 0
        # => intercepts[k] <= mid_t for slopes[k] == 0
        
        pos = slopes > 1e-15
        neg = slopes < -1e-15
        zero = ~(pos | neg)
        
        if zero.any() and intercepts[zero].max() > mid_t:
            hi_t = mid_t
            continue
        
        if pos.any():
            upper = np.min((mid_t - intercepts[pos]) / slopes[pos])
        else:
            upper = np.inf
        
        if neg.any():
            lower = np.max((mid_t - intercepts[neg]) / slopes[neg])
        else:
            lower = -np.inf
        
        # Check intersection with [0,1]
        if lower <= min(upper, 1.0) and max(lower, 0.0) <= upper:
            lo_t = mid_t  # feasible, try smaller t
        else:
            hi_t = mid_t
    
    # At optimal t*, find h[i]
    mid_t = (lo_t + hi_t) / 2
    pos = slopes > 1e-15
    neg = slopes < -1e-15
    
    if pos.any():
        upper = np.min((mid_t - intercepts[pos]) / slopes[pos])
    else:
        upper = np.inf
    if neg.any():
        lower = np.max((mid_t - intercepts[neg]) / slopes[neg])
    else:
        lower = -np.inf
    
    lower = max(lower, 0.0)
    upper = min(upper, 1.0)
    
    if lower > upper:
        # Infeasible at this t, use current value
        return h[i]
    
    # Pick midpoint of feasible interval
    return (lower + upper) / 2.0
```

```python
def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)

    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()

            g[d:] += (1.0 - h[:n - d])
            g[:n - d] += (-h[d:])
    else:
        dd = -d
        if dd < n:
            g[:n - dd] += (1.0 - h[dd:])

    return g

            hidden_idx = np.where(hidden)[0]
            new_active = sorted(set(active_k) | set(hidden_idx.tolist()))
            if len(new_active) > 60:
                vals = [corrs_new[k] for k in new_active]
                order = np.argsort(vals)[::-1][:60]
                new_active = sorted([new_active[i] for i in order])
            active_k = new_active
        t0 = max(t_solved, c5_new) * 1.0001 + 1e-6
        h = h_new
        if abs(c5_new - best_c5) < 1e-13 and not hidden.any():
            break
    return best_h, best_c5

def coord_descent_polish(h0, n, max_rounds=50):
    target_sum = n / 2.0
    dx = 2.0 / n
    n_lags = 2 * n - 1
    h = h0.copy()
    h = project_feasible(h, target_sum)
    best_h = h.copy()
    best_c5 = compute_corrs(h).max() * dx
    for rnd in range(max_rounds):
        corrs = compute_corrs(h) * dx
        any_change = False
        for i in range(n):
            slopes = np.zeros(n_lags)
            for k in range(n_lags):
                ad = abs(k - (n - 1))
                if ad >= n:
                    continue
                s = 1.0
                if i - ad >= 0:
                    s -= h[i - ad]
                if i + ad < n:
                    s -= h[i + ad]
                slopes[k] = s * dx
            intercepts = corrs - h[i] * slopes
            val0 = np.max(intercepts)
            val1 = np.max(slopes + intercepts)
            lo_t = min(val0, val1)
            hi_t = max(val0, val1)
            for _ in range(45):
                mid_t = (lo_t + hi_t) / 2.0
                pos = slopes > 1e-15
                neg = slopes < -1e-15
                zero = ~(pos | neg)
                if zero.any() and np.max(intercepts[zero]) > mid_t + 1e-15:
                    hi_t = mid_t
                    continue
                upper = np.min((mid_t - intercepts[pos]) / slopes[pos]) if pos.any() else 1e10
                lower = np.max((mid_t - intercepts[neg]) / slopes[neg]) if neg.any() else -1e10
                if lower <= min(upper, 1.0) and max(lower, 0.0) <= upper:
                    lo_t = mid_t
                else:
                    hi_t = mid_t
            mid_t = (lo_t + hi_t) / 2.0
            pos = slopes > 1e-15
            neg = slopes < -1e-15
            upper = np.min((mid_t - intercepts[pos]) / slopes[pos]) if pos.any() else 1.0
            lower = np.max((mid_t - intercepts[neg]) / slopes[neg]) if neg.any() else 0.0
            lower = max(lower, 0.0)
            upper = min(upper, 1.0)
            if lower > upper:
                continue
            h_new_i = (lower + upper) / 2.0
            if abs(h_new_i - h[i]) > 1e-12:
                h[i] = h_new_i
                any_change = True
        h = project_feasible(h, target_sum)
        c5_now = compute_corrs(h).max() * dx
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
        if not any_change:
            break
    return best_h, best_c5

def smooth_full_value_grad(h, n, beta):
    dx = 2.0 / n
    corrs = compute_corrs(h)
    val = logsumexp(beta * corrs) / beta * dx
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(len(corrs)) / len(corrs)
    else:
        w /= w_sum
    G = np.zeros(n)
    for k in range(len(corrs)):
        if w[k] < 1e-14:
            continue
        d = k - (n - 1)
        G += w[k] * grad_corr_d(h, d, n)
    return val, G * dx

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    def remaining():
        return budget_s - (time.time() - t_start)
    h_parent = np.array(initial_h_values, dtype=float)
    n_parent = len(h_parent)
    target_sum = n_parent / 2.0
    h_parent = project_feasible(h_parent, target_sum)
    score_parent, _ = exact_score(h_parent)
    best_h = h_parent.copy()
    best_c5 = score_parent

    # Phase 1: Smooth warmup in q-space
    n = n_parent
    m = n // 2
    q_best = q_from_h(h_parent, n)
    for beta in [100, 300, 800, 2000]:
        if remaining() < 40:
            break
        q0 = q_best.copy()
        def q_obj(x):
            return smooth_q_value_grad(x, n, beta)[0]
        def q_grad(x):
            return smooth_q_value_grad(x, n, beta)[1]
        try:
            res = minimize(q_obj, q0, jac=q_grad, method='L-BFGS-B',
                           bounds=[(0.0, 1.0)] * m,
                           options={'maxiter': 400, 'ftol': 1e-16})
            h_res = build_h_from_q(np.clip(res.x, 0, 1), n)
            s_res, _ = exact_score(h_res)
            if s_res < best_c5 - 1e-14:
                best_c5 = s_res
                best_h = h_res
                q_best = np.clip(res.x, 0, 1).copy()
        except Exception:
            pass

    # Phase 2: Active set in q-space
    if remaining() > 100:
        q_es, c5_es = active_set_loop_q(q_best, n, max_rounds=15)
        h_es = build_h_from_q(q_es, n)
        h_es = project_feasible(h_es, target_sum)
        s_es, _ = exact_score(h_es)
        if s_es < best_c5 - 1e-14:
            best_c5 = s_es
            best_h = h_es
            q_best = q_es.copy()

    # Phase 3: Full-space active set (break symmetry)
    if remaining() > 100:
        h_full, c5_full = active_set_loop_full(best_h, n, max_rounds=12)
        s_full, _ = exact_score(h_full)
        if s_full < best_c5 - 1e-14:
            best_c5 = s_full
            best_h = h_full

    # Phase 4: Coordinate descent polish
    if remaining() > 60:
        h_cd, c5_cd = coord_descent_polish(best_h, n, max_rounds=80)
        s_cd, _ = exact_score(h_cd)
        if s_cd < best_c5 - 1e-14:
            best_c5 = s_cd
            best_h = h_cd

    # Phase 5: Second round - active set + coord descent
    if remaining() > 80 and n % 2 == 0:
        q2 = q_from_h(best_h, n)
        q_es2, c5_es2 = active_set_loop_q(q2, n, max_rounds=10)
        h_es2 = build_h_from_q(q_es2, n)
        h_es2 = project_feasible(h_es2, target_sum)
        s_es2, _ = exact_score(h_es2)
        if s_es2 < best_c5 - 1e-14:
            best_c5 = s_es2
            best_h = h_es2
    if remaining() > 60:
        h_cd2, c5_cd2 = coord_descent_polish(best_h, n, max_rounds=60)
        s_cd2, _ = exact_score(h_cd2)
        if s_cd2 < best_c5 - 1e-14:
            best_c5 = s_cd2
            best_h = h_cd2

    # Phase 6: Try nearby n values
    n_candidates = [124, 128, 130, 132, 120, 134, 136, 122]
    for n_try in n_candidates:
        if remaining() < 80:
            break
        ts_try = n_try / 2.0
        dx_try = 2.0 / n_try
        x_orig = np.linspace(0.0, 2.0, len(best_h))
        x_new = np.linspace(0.0, 2.0, n_try)
        h_interp = np.interp(x_new, x_orig, best_h)
        h_interp = project_feasible(h_interp, ts_try)
        h_sm, c5_sm = active_set_loop_full(h_interp, n_try, max_rounds=8)
        s_sm, _ = exact_score(h_sm)
        if s_sm < best_c5 - 1e-10:
            best_c5 = s_sm
            best_h = h_sm
        if remaining() > 50:
            h_cd3, c5_cd3 = coord_descent_polish(best_h, len(best_h), max_rounds=40)
            s_cd3, _ = exact_score(h_cd3)
            if s_cd3 < best_c5 - 1e-10:
                best_c5 = s_cd3
                best_h = h_cd3

    # Phase 7: Final polish on best
    n_final = len(best_h)
    if remaining() > 40:
        h_fp, _ = coord_descent_polish(best_h, n_final, max_rounds=60)
        s_fp, _ = exact_score(h_fp)
        if s_fp < best_c5 - 1e-14:
            best_c5 = s_fp
            best_h = h_fp
    if remaining() > 30 and n_final % 2 == 0:
        q_f = q_from_h(best_h, n_final)
        q_rf, _ = active_set_loop_q(q_f, n_final, max_rounds=8)
        h_rf = build_h_from_q(q_rf, n_final)
        h_rf = project_feasible(h_rf, n_final / 2.0)
        s_rf, _ = exact_score(h_rf)
        if s_rf < best_c5 - 1e-14:
            best_c5 = s_rf
            best_h = h_rf

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    n_out = len(best_h)
    ts_out = n_out / 2.0
    best_h = project_feasible(best_h, ts_out)
    final_score, _ = exact_score(best_h)
    if final_score < best_c5:
        best_c5 = final_score
    return best_h, best_c5, n_out
```
