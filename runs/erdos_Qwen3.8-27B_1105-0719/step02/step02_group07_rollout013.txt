def update_corr_for_pos(h, pos, old_val, new_val, n):
    """Compute change in each correlation when h[pos] changes from old_val to new_val"""
    delta = new_val - old_val
    changes = np.zeros(2*n-1)
    for d in range(2*n-1):
        s = d - (n-1)
        # h[pos] appears as h[j] if j=pos: need 0≤pos+s<n
        # in term h[pos]*(1-h[pos+s]), change is delta*(1-h[pos+s])
        # h[pos] appears as h[j+s] if j+s=pos, j=pos-s: need 0≤pos-s<n
        # in term h[pos-s]*(1-h[pos]), change is -delta*h[pos-s]
        c = 0.0
        if 0 <= pos + s < n:
            c += delta * (1 - h[pos + s])
        if 0 <= pos - s < n:
            c -= delta * h[pos - s]
        changes[d] = c
    return changes * (2.0/n)

    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full') * dx)

    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)

        else:

            qi_old = q[i]

                    best_qi = qi

            g0 = np.linspace(0, 1, grid_size)

                for b in g0:
                    q[i] = a
                    q[i+1] = b
                    h_tmp = make_sym_h(q, n)
                    c5_tmp = exact_score(h_tmp)
                    if c5_tmp < best_c5 - 1e-14:
                        best_c5 = c5_tmp
                        best_q0, best_q1 = a, b
            
            q[i] = best_q0
            q[i+1] = best_q1
            if best_c5 < c5 - 1e-14:
                c5 = best_c5
                improved = True
        
        if not improved:
            break
    
    return q, c5
