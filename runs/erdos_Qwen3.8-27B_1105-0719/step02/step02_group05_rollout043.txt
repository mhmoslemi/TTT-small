import time

def build_h_from_q(q):
    """Build h from anti-symmetric parameter q."""
    m = len(q)
    n = 2 * m
    return np.concatenate([q, 1.0 - q[::-1]])

def q_from_h(h):
    """Extract q from anti-symmetric h (n must be even)."""
    return h[:len(h)//2].copy()

def compute_corrs(h):
    """Compute all correlations."""
    return np.correlate(h, 1.0 - h, mode='full')

def exact_score(h):
    """Compute the C5 objective value."""
    n = len(h)
    dx = 2.0 / n
    corrs = compute_corrs(h)
    return corrs.max() * dx

def project_feasible(x, target_sum):
    """Project x onto [0,1]^n with sum = target_sum."""
    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()

import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time


def build_h_from_q(q):
    m = len(q)
    return np.concatenate([q, 1.0 - q[::-1]])


def q_from_h(h):
    return h[:len(h) // 2].copy()


def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def exact_score(h):
    n = len(h)
    return compute_corrs(h).max() * (2.0 / n)


def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-10:
        mask = (h > 1e-8) & (h < 1.0 - 1e-8)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)


def grad_corr_d(h, d, n):
    g = np.zeros(n)
    if d >= 0:
        if d < n:
            g[d:] += (1.0 - h[:n - d])
            g[:n - d] += (-h[d:])
    else:
        dd = -d
        if dd < n:
            g[:n - dd] += (1.0 - h[dd:])
            g[dd:] += (-h[:n - dd])
    return g


def optimize_coord(h, j, n):
    """Find best value of q[j] given h (anti-symmetric, n even)."""
    a = h[j]
    del_lo = -a
    del_hi = 1.0 - a
    if del_hi - del_lo < 1e-14:
        return a
    corrs = compute_corrs(h)
    qpos = n - 1 - j
    s_arr = np.arange(-(n - 1), n)
    js = j - s_arr
    jp = j + s_arr
    qs = qpos - s_arr
    qp = qpos + s_arr
    m1 = (js >= 0) & (js < n)
    m2 = (jp >= 0) & (jp < n)
    m3 = (qs >= 0) & (qs < n)
    m4 = (qp >= 0) & (qp < n)
    slopes = np.zeros(2 * n - 1)
    slopes[m1] += (1.0 - h[js[m1]])
    slopes[m2] += -h[jp[m2]]
    slopes[m3] += -(1.0 - h[qs[m3]])
    slopes[m4] += h[qp[m4]]
    s_star = n - 1 - 2 * j
    s_neg = 2 * j - (n - 1)
    idx0 = n - 1
    idx_star = s_star + (n - 1)
    idx_neg = s_neg + (n - 1)

def cd_optimize(q, n_total_cd=200):
