                best_jj = new_maxes.argmin()
        if best_ii < 0:
            break
        i = ones[best_ii]
        j = zeros[best_jj]
        h[i] = 0.0
        h[j] = 1.0
        corrs = np.correlate(h, 1.0 - h, mode='full')
        c5 = corrs.max() * dx
        if c5 < best_c5 - 1e-15:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5


def stuck_escape(h, dx, rng, t0=None, deadline=None, max_tries=30):
    n = len(h)
    corrs = np.correlate(h, 1.0 - h, mode='full')
    cur_max = corrs.max()
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    A, B, ones, zeros = compute_delta_matrices(h, n)
    base_maxes = np.full((len(ones), len(zeros)), cur_max + 1.0)
    for ii in range(len(ones)):
        base = corrs + A[ii]
        base_maxes[ii, :] = (base[None, :] + B).max(axis=1)
    worse_by_1 = base_maxes <= cur_max + 1.0 + 1e-14
    candidates = []
    for ii in range(len(ones)):
        for jj in range(len(zeros)):
            if worse_by_1[ii, jj]:
                candidates.append((ii, jj, base_maxes[ii, jj]))
    rng.shuffle(candidates)
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    for ii, jj, _ in candidates[:max_tries]:
        if t0 is not None and deadline is not None and time.time() > deadline:
            break
        h_try = h.copy()
        h_try[ones[ii]] = 0.0
        h_try[zeros[jj]] = 1.0
        h_opt, c5_opt = steepest_descent_binary(h_try, dx, max_steps=300, t0=t0, deadline=deadline)
        if c5_opt < best_c5 - 1e-14:
            best_c5 = c5_opt
            best_h = h_opt
    return best_h, best_c5


def make_golden_init(n, rng):
    h = np.zeros(n)
    target = n // 2
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    positions = set()
    k = 0
    while len(positions) < target and k < n * 10:
        pos = int((k * phi) % n)
        positions.add(pos)
        k += 1
    h[list(positions)] = 1.0
    if h.sum() < target:
        remaining = target - int(h.sum())
        zeros = np.where(h == 0)[0]
        h[zeros[:remaining]] = 1.0
    elif h.sum() > target:
        ones = np.where(h == 1)[0]
        h[ones[-(int(h.sum()) - target):]] = 0.0
    return h


def make_random_init(n, rng):
    h = np.zeros(n)
    target = n // 2
    idx = rng.permutation(n)
    h[idx[:target]] = 1.0
    return h


def make_sine_init(n, rng):
    h = np.zeros(n)
    target = n // 2
    x = np.linspace(0, 2.0, n, endpoint=False)
    freq = rng.uniform(2.0, 8.0)
    phase = rng.uniform(0, 2 * np.pi)
    scores = np.sin(freq * x + phase) + 0.3 * np.sin(2 * freq * x + 2 * phase)
    idx = np.argsort(-scores)
    h[idx[:target]] = 1.0
    return h


def smooth_max_obj(h, dx, beta):
    corrs = np.correlate(h, 1.0 - h, mode='full')
    m = corrs.max()
    return (m + np.log(np.sum(np.exp(beta * (corrs - m)))) / beta) * dx


def smooth_max_grad(h, dx, beta):
    n = len(h)
    corrs = np.correlate(h, 1.0 - h, mode='full')
    m = corrs.max()
    w = np.exp(beta * (corrs - m))
    w /= w.sum()
    lags = np.arange(-(n - 1), n)
    grad = np.zeros(n)
    for k in range(len(corrs)):
        if w[k] < 1e-14:
            continue
        offset = k - (n - 1)
        i_lo = max(0, -offset)
        i_hi = min(n - 1, n - 1 - offset)
        if i_lo > i_hi:
            continue
        idx = np.arange(i_lo, i_hi + 1)
        grad[idx + offset] += (1.0 - h[idx])
        grad[idx] += (-h[idx + offset])
        grad *= 1.0
    grad = np.zeros(n)
    for k in range(len(corrs)):
        if w[k] < 1e-14:
            continue
        offset = k - (n - 1)
        i_lo = max(0, -offset)
        i_hi = min(n - 1, n - 1 - offset)
        if i_lo > i_hi:
            continue
        idx = np.arange(i_lo, i_hi + 1)
        grad[idx + offset] += w[k] * (1.0 - h[idx])
        grad[idx] += w[k] * (-h[idx + offset])
    return grad * dx


def continuous_optimize(h0, dx, target_sum, beta, maxiter, t0=None, deadline=None):
    n = len(h0)

    def obj(x):
        return smooth_max_obj(x, dx, beta)

    def obj_grad(x):
        return smooth_max_grad(x, dx, beta)

    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        h_out = project_feasible(res.x, target_sum)
    except Exception:
        h_out = project_feasible(h0, target_sum)
    return h_out, c5_of_h(h_out, dx)


def binary_sa(h0, dx, rng, max_iters=50000, t0=None, deadline=None):
    n = len(h0)
    target = n // 2
    h = np.zeros(n)
    h[np.argsort(-h0)[:target]] = 1.0
    corrs = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = corrs.max() * dx
    best_h = h.copy()
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    temp = 0.03
    cooling = 0.9998
    no_improve = 0
    for it in range(max_iters):
        if t0 is not None and deadline is not None and time.time() > deadline:
            break
        if it % 3000 == 0:
            temp *= cooling
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        lags = np.arange(-(n - 1), n)
        m = 2 * n - 1
        delta = np.zeros(m)
        vp_i = (i + lags >= 0) & (i + lags < n)
        vm_i = (i - lags >= 0) & (i - lags < n)
        delta[vp_i] += -(1.0 - h[i + lags[vp_i]])
        delta[vm_i] += h[i - lags[vm_i]]
        vp_j = (j + lags >= 0) & (j + lags < n)
        vm_j = (j - lags >= 0) & (j - lags < n)
        delta[vp_j] += (1.0 - h[j + lags[vp_j]])
        delta[vm_j] += -h[j - lags[vm_j]]
        new_corrs = corrs + delta
        new_c5 = new_corrs.max() * dx
        accept = False
        if new_c5 <= best_c5 + 1e-15:
            accept = True
        elif temp > 1e-9 and rng.random() < np.exp(-(new_c5 - best_c5) / temp):
            accept = True
        if accept:
            h[i], h[j] = 0.0, 1.0
            corrs = new_corrs
            if new_c5 < best_c5 - 1e-15:
                best_c5 = new_c5
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
        else:
            no_improve += 1
        if no_improve > 20000:
            break
    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5.0
    rng = np.random.RandomState(seed)

    best_h = np.array(initial_h_values, dtype=float)
    n = len(best_h)
    dx = 2.0 / n
    target_sum = n / 2.0
    best_h = project_feasible(best_h, target_sum)
    best_c5 = c5_of_h(best_h, dx)

    # Phase 1: Binary steepest descent from parent
    h_bin = binary_ize(best_h, n)
    h_sd, c5_sd = steepest_descent_binary(h_bin, dx, max_steps=2000, t0=t0, deadline=deadline)
    if c5_sd < best_c5:
        best_c5 = c5_sd
        best_h = h_sd

    # Phase 2: Stuck escape
    if time.time() < deadline - 60:
        h_esc, c5_esc = stuck_escape(best_h, dx, rng, t0=t0, deadline=deadline, max_tries=40)
        if c5_esc < best_c5:
            best_c5 = c5_esc
            best_h = h_esc

    # Phase 3: More steepest + escape cycles
    for cycle in range(5):
        if time.time() > deadline - 40:
            break
        h_sd2, c5_sd2 = steepest_descent_binary(best_h, dx, max_steps=500, t0=t0, deadline=deadline)
        if c5_sd2 < best_c5:
            best_c5 = c5_sd2
            best_h = h_sd2
        if time.time() > deadline - 40:
            break
        h_esc2, c5_esc2 = stuck_escape(best_h, dx, rng, t0=t0, deadline=deadline, max_tries=25)
        if c5_esc2 < best_c5:
            best_c5 = c5_esc2
            best_h = h_esc2

    # Phase 4: Try different initializations at same n
    inits = []
    inits.append(make_golden_init(n, rng))
    inits.append(make_random_init(n, rng))
    inits.append(make_sine_init(n, rng))
    inits.append(make_random_init(n, rng))
    inits.append(make_sine_init(n, rng))
    for h_init in inits:
        if time.time() > deadline - 60:
            break
        h_sd3, c5_sd3 = steepest_descent_binary(h_init, dx, max_steps=1500, t0=t0, deadline=deadline)
        if c5_sd3 < best_c5 - 1e-6:
            best_c5 = c5_sd3
            best_h = h_sd3
            if time.time() > deadline - 40:
                break
            h_esc3, c5_esc3 = stuck_escape(best_h, dx, rng, t0=t0, deadline=deadline, max_tries=20)
            if c5_esc3 < best_c5:
                best_c5 = c5_esc3
                best_h = h_esc3

    # Phase 5: Simulated annealing polish
    if time.time() > deadline - 200:
        h_sa, c5_sa = binary_sa(best_h, dx, rng, max_iters=80000, t0=t0, deadline=deadline - 30)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa
            h_final, c5_final = steepest_descent_binary(best_h, dx, max_steps=500, t0=t0, deadline=deadline)
            if c5_final < best_c5:
                best_c5 = c5_final
                best_h = h_final

    # Phase 6: Try n=128
    if time.time() > deadline - 120:
        n2 = 128
        dx2 = 2.0 / n2
        ts2 = n2 / 2.0
        h_bin128 = np.zeros(n2)
        h_bin128[np.argsort(-best_h)[:n2 // 2]] = 1.0
        h_sd128, c5_128 = steepest_descent_binary(h_bin128, dx2, max_steps=1500, t0=t0, deadline=deadline)
        if c5_128 < best_c5:
            best_c5 = c5_128
            best_h = h_sd128
            best_n = n2
        if time.time() > deadline - 60:
            h_esc128, c5_esc128 = stuck_escape(best_h, 2.0 / len(best_h), rng, t0=t0, deadline=deadline, max_tries=30)
            if c5_esc128 < best_c5:
                best_c5 = c5_esc128
                best_h = h_esc128

    # Phase 7: Try n=144
    if time.time() > deadline - 100:
        n3 = 144
        dx3 = 2.0 / n3
        ts3 = n3 / 2.0
        x_old = np.linspace(0, 2.0, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2.0, n3, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_bin144 = binary_ize(h_interp, n3)
        h_sd144, c5_144 = steepest_descent_binary(h_bin144, dx3, max_steps=1000, t0=t0, deadline=deadline)
        if c5_144 < best_c5:
            best_c5 = c5_144
            best_h = h_sd144
        if time.time() > deadline - 50:
            h_esc144, c5_e144 = stuck_escape(best_h, 2.0 / len(best_h), rng, t0=t0, deadline=deadline, max_tries=20)
            if c5_e144 < best_c5:
                best_c5 = c5_e144
                best_h = h_esc144

    # Phase 8: Continuous optimization to find better starting point
    if time.time() > deadline - 80:
        n_c = len(best_h)
        dx_c = 2.0 / n_c
        ts_c = n_c / 2.0
        h_cont = project_feasible(best_h.astype(float), ts_c)
        for beta in [200, 500]:
            if time.time() > deadline - 30:
                break
            h_opt, c5_opt = continuous_optimize(h_cont, dx_c, ts_c, beta, 400, t0=t0, deadline=deadline)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
                h_b = binary_ize(best_h, n_c)
                h_bsd, c5_bsd = steepest_descent_binary(h_b, dx_c, max_steps=500, t0=t0, deadline=deadline)
                if c5_bsd < best_c5:
                    best_c5 = c5_bsd
                    best_h = h_bsd

    # Phase 9: Final polish
    if time.time() > deadline - 30:
        n_f = len(best_h)
        dx_f = 2.0 / n_f
        h_f_bin = binary_ize(best_h, n_f)
        h_f_sd, c5_f = steepest_descent_binary(h_f_bin, dx_f, max_steps=500, t0=t0, deadline=deadline)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f_sd

    # Final validation and cleanup
    n_b = len(best_h)
    dx_b = 2.0 / n_b
    ts_b = n_b / 2.0
    best_h = project_feasible(best_h, ts_b)
    best_c5 = c5_of_h(best_h, dx_b)
    if best_h.min() < -1e-10 or best_h.max() > 1.0 + 1e-10:
        best_h = np.clip(best_h, 0.0, 1.0)
        best_h = project_feasible(best_h, ts_b)
        best_c5 = c5_of_h(best_h, dx_b)

    return best_h, best_c5, n_b
