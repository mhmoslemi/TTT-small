    dx = 2.0 / n
    G, profile = build_gradient_matrix(h, n, dx)
    d = cp.Variable(n)
    t = cp.Variable()
    constraints = []
    for m in active_rows:
        constraints.append(G[m, :] @ d <= t - profile[m])
    constraints.append(cp.sum(d) == 0)
    l = np.maximum(-radius, -h)
    u = np.minimum(radius, 1.0 - h)
    constraints.append(d >= l)
    constraints.append(d <= u)

    prob.solve(solver=cp.ECOS)  # or SCS, CLARABEL
    if d.value is not None:
        return d.value, t.value
    return None, None

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx] + 1e-15)
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx] + 1e-15)
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:

            p[idx] = dx * np.sum(h * (1.0 - h))
        elif d > 0:
            p[idx] = dx * np.sum(h[d:n] * (1.0 - h[0:n-d]))
        else:
            e = -d
            p[idx] = dx * np.sum(h[0:n-e] * (1.0 - h[e:n]))
    return G, p

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

def solve_lp_direction(h, n, lags, radius):
    """Solve LP to find direction d minimizing max linearized objective.
    
    min_{d,t} t
    s.t. G_A @ d <= t - p_A
         sum(d) = 0
         l <= d <= u
    """
    dx = 2.0 / n
    G, p = build_gradient_block(h, n, dx, lags)
    M = len(lags)
    
    # Bounds on d
    l = np.maximum(-radius, -h)
    u = np.minimum(radius, 1.0 - h)
    
    # Use cvxpy
    d_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = []
    for m in range(M):
        constraints.append(G[m, :] @ d_var <= t_var - p[m])
    constraints.append(cp.sum(d_var) == 0)
    constraints.append(d_var >= l)
    constraints.append(d_var <= u)
    
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    try:
        prob.solve(solver=cp.ECOS, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.CVXOPT, verbose=False)
        except Exception:
            return None, None
    
    if d_var.value is None:
        return None, None
    
    d_val = np.asarray(d_var.value, dtype=np.float64)
    if not np.isfinite(d_val).all():
        return None, None
    
    return d_val, float(t_var.value) if t_var.value is not None else None

def active_descent_step(h, n, max_iters=100, beam_frac=0.01, top_k=15, step_scales=None):
    """Active-set gradient descent with exact line search."""
    if step_scales is None:
        step_scales = [0.03, 0.02, 0.015, 0.01, 0.007, 0.005, 0.003, 0.002, 0.001, 0.0005]
    
    dx = 2.0 / n
    h = h.copy()
    best_c = exact_score(h)
    
    for it in range(max_iters):
        profile = compute_profile(h)
        c = np.max(profile)
        if c < best_c:
            best_c = c
        
        # Active set
        spread = c - np.median(profile)
        beam = max(0.002, min(0.015, beam_frac * max(spread, 0.001)))
        thresh = c - beam
        active_m = np.where(profile >= thresh)[0]
        if len(active_m) == 0:
            active_m = np.array([np.argmax(profile)])
        if len(active_m) > top_k:
            top_idx = np.argsort(profile[active_m])[-top_k:]
            active_m = active_m[top_idx]
        
        # Convert to lags
        lags = active_m - (n - 1)
        
        # Compute weighted gradient (weights proportional to how "active" each row is)
        G, p = build_gradient_block(h, n, dx, lags)
        w = profile[active_m] - thresh
        w = np.maximum(w, 0.01)
        w = w / w.sum()
        grad = G.T @ w
        grad -= grad.mean()  # sum constraint
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad /= gnorm
        
        # Line search
        improved = False
        for a in step_scales:
            h_try = repair_feasible(h - a * grad)
            s = exact_score(h_try)
            if s < best_c - 1e-13:
                h = h_try
                best_c = s
                improved = True
                break
        if not improved:
            # Try even smaller steps
            for a in [0.0003, 0.0002, 0.0001, 0.00005, 0.00002]:
                h_try = repair_feasible(h - a * grad)
                s = exact_score(h_try)
                if s < best_c - 1e-14:
                    h = h_try
                    best_c = s
                    improved = True
                    break
            if not improved:
                break
    
    return h, best_c

def lp_trust_region_descent(h, n, max_iters=50, beam_frac=0.01, top_k=20, radius_init=0.02):
    """LP trust region method."""
    dx = 2.0 / n
    h = h.copy()
    best_c = exact_score(h)
    radius = radius_init
    
    for it in range(max_iters):
        profile = compute_profile(h)
        c = np.max(profile)
        if c < best_c:
            best_c = c
        
        # Active set
        spread = c - np.median(profile)
        beam = max(0.003, min(0.02, beam_frac * max(spread, 0.002)))
        thresh = c - beam
        active_m = np.where(profile >= thresh)[0]
        if len(active_m) == 0:
            active_m = np.array([np.argmax(profile)])
        if len(active_m) > top_k:
            top_idx = np.argsort(profile[active_m])[-top_k:]
            active_m = active_m[top_idx]
        
        lags = active_m - (n - 1)
        
        # Solve LP
        d, t_pred = solve_lp_direction(h, n, lags, radius)
        if d is None:
            break
        
        # Scale direction
        d_norm = np.linalg.norm(d)
        if d_norm < 1e-10:
            radius *= 0.5
            continue
        d = d / d_norm * radius  # normalize to trust region size
        
        # Line search
        improved = False
        for frac in [1.0, 0.7, 0.5, 0.3, 0.2, 0.1]:
            h_try = repair_feasible(h + frac * d)
            s = exact_score(h_try)
            if s < best_c - 1e-13:
                h = h_try
                best_c = s
                improved = True
                break
        
        if improved:
            radius = min(radius * 1.2, 0.05)
        else:
            radius *= 0.5
            if radius < 0.001:
                break
    
    return h, best_c

def make_symmetric_pattern(n, freq, phase, rng):
    """Create a sinusoidal-based starting pattern."""
    x = np.linspace(0, 2, n, endpoint=False) + 1.0/n
    vals = 0.5 + 0.4 * np.sin(2 * np.pi * freq * x / 2.0 + phase)
    vals = np.clip(vals, 0.01, 0.99)
    return repair_feasible(vals)

def make_block_pattern(n, n_blocks, rng):
    """Create a block-structured pattern."""
    block_size = n // n_blocks
    h = np.zeros(n)
    for b in range(n_blocks):
        start = b * block_size
        end = min((b + 1) * block_size, n)
        h[start:end] = rng.uniform(0.1, 0.9)
    # Make symmetric
    h = (h + h[::-1]) / 2.0
    return repair_feasible(h)

import numpy as np
import time
import cvxpy as cp
from scipy.optimize import linprog, minimize


def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx] + 1e-15)
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx] + 1e-15)
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h


def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))


def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx


def gradient_row(h, n, dx, d):
    g = np.zeros(n)
    if d == 0:
        g[:] = dx * (1.0 - 2.0 * h)
    elif d > 0:
        g[d:n] += dx * (1.0 - h[0:n - d])
        g[0:n - d] += -dx * h[d:n]
    else:
        e = -d
        g[0:n - e] += dx * (1.0 - h[e:n])
        g[e:n] += -dx * h[0:n - e]
    return g


def build_gradient_block(h, n, dx, lags):
    M = len(lags)
    G = np.zeros((M, n))
    p = np.zeros(M)
    for idx, d in enumerate(lags):
        G[idx, :] = gradient_row(h, n, dx, d)
        if d == 0:
            p[idx] = dx * np.sum(h * (1.0 - h))
        elif d > 0:
            p[idx] = dx * np.sum(h[d:n] * (1.0 - h[0:n - d]))
        else:
            e = -d
            p[idx] = dx * np.sum(h[0:n - e] * (1.0 - h[e:n]))
    return G, p


def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)


def antisymmetric_project(h):
    n = len(h)
    if n % 2 != 0:
        return repair_feasible(h)
    half = n // 2
    h_new = np.zeros(n)
    for i in range(half):
        h_new[i] = (h[i] + 1.0 - h[i + half]) / 2.0
        h_new[i + half] = 1.0 - h_new[i]
    return repair_feasible(h_new)


def solve_lp_direction(h, n, lags, radius):
    dx = 2.0 / n
    G, p = build_gradient_block(h, n, dx, lags)
    M = len(lags)
    l = np.maximum(-radius, -h)
    u = np.minimum(radius, 1.0 - h)
    try:
        d_var = cp.Variable(n)
        t_var = cp.Variable()
        cons = []
        for m in range(M):
            cons.append(G[m, :] @ d_var <= t_var - p[m])
        cons.append(cp.sum(d_var) == 0)
        cons.append(d_var >= l)
        cons.append(d_var <= u)
        prob = cp.Problem(cp.Minimize(t_var), cons)
        prob.solve(solver=cp.ECOS, verbose=False)
        if d_var.value is not None and np.isfinite(d_var.value).all():
            d_val = np.asarray(d_var.value, dtype=np.float64)
            t_val = float(t_var.value) if t_var.value is not None else None
            return d_val, t_val
    except Exception:
        pass
    try:
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        A_ub = np.zeros((M, n + 1))
        A_ub[:, :n] = G
        A_ub[:, n] = -1.0
        b_ub = -p
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)
        bounds_list = []
        for i in range(n):
            bounds_list.append((l[i], u[i]))
        bounds_list.append((None, None))
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds_list, method='highs')
        if res.success:
            return res.x[:n], res.x[n]
    except Exception:
        pass
    return None, None


def lp_trust_region_descent(h, n, max_iters, beam_frac, top_k, radius_init, t_limit):
    t0 = time.perf_counter()
    dx = 2.0 / n
    h = h.copy()
    best_c = exact_score(h)
    radius = radius_init
    for it in range(max_iters):
        if time.perf_counter() - t0 > t_limit:
            break
        profile = compute_profile(h)
        c = np.max(profile)
        if c < best_c:
            best_c = c
        spread = c - np.median(profile)
        beam = max(0.003, min(0.02, beam_frac * max(spread, 0.002)))
        thresh = c - beam
        active_m = np.where(profile >= thresh)[0]
        if len(active_m) == 0:
            active_m = np.array([np.argmax(profile)])
        if len(active_m) > top_k:
            top_idx = np.argsort(profile[active_m])[-top_k:]
            active_m = active_m[top_idx]
        lags = active_m - (n - 1)
        d, t_pred = solve_lp_direction(h, n, lags, radius)
        if d is None:
            radius *= 0.5
            if radius < 0.001:
                break
            continue
        d_norm = np.linalg.norm(d)
        if d_norm < 1e-10:
            radius *= 0.5
            continue
        d = d / d_norm * radius
        improved = False
        for frac in [1.0, 0.7, 0.5, 0.3, 0.2, 0.1]:
            h_try = repair_feasible(h + frac * d)
            s = exact_score(h_try)
            if s < best_c - 1e-13:
                h = h_try
                best_c = s
                improved = True
                break
        if improved:
            radius = min(radius * 1.15, 0.05)
        else:
            radius *= 0.5
            if radius < 0.001:
                break
    return h, best_c


def active_descent(h, n, max_iters, beam_frac, top_k, steps, t_limit):

def make_sinusoidal(n, freq, phase):
    x = (np.arange(n) + 0.5) * (2.0 / n)
    vals = 0.5 + 0.35 * np.sin(2.0 * np.pi * freq * x / 2.0 + phase)
    vals = np.clip(vals, 0.05, 0.95)
    return repair_feasible(vals)


def make_blocks(n, n_blocks, rng):
    block_size = n // n_blocks
    h = np.zeros(n)
    for b in range(n_blocks):
        start = b * block_size
        end = min((b + 1) * block_size, n)
        h[start:end] = rng.uniform(0.1, 0.9)
    h = (h + h[::-1]) / 2.0
    return repair_feasible(h)
