    # Let's compute B_d = coefficient of t in corr_d
    # corr_d = sum over valid pairs (j, j+d) of h[j]*(1-h[j+d])
    # If j = i: term is t*(1-h[i+d]) -> contributes (1-h[i+d]) to B_d
    # If j+d = i, i.e., j = i-d: term is h[i-d]*(1-t) -> contributes -h[i-d] to B_d
    
    # So B_d = (1-h[i+d]) if i+d is valid, plus -h[i-d] if i-d is valid
    # (with the understanding that "valid" means within [0, n-1])
    
    old = h[i]
    B = np.zeros(2*n-1)
    # For each lag index k (0 to 2n-2), d = k-(n-1)
    # The term h[i]*(1-h[i+d]) contributes to corr at lag index (n-1+d)
    # Wait, I need to be more careful with the indexing.
    
    # np.correlate(h, 1-h, 'full')[k] = sum_i h[i+k-(n-1)] * (1-h[i]) for valid i
    # = sum over valid i of h[i+k-(n-1)] * (1-h[i])
    
    # Let d = k - (n-1). Then:
    # corr_d = sum over valid i of h[i+d] * (1-h[i])
    # where valid means 0 <= i < n and 0 <= i+d < n.
    
    # Contribution to corr_d from changing h[i]:
    # Case 1: i is the "first" index, i.e., the "i" in the formula: h[i+d]*(1-h[i])
    #   If i+d is valid (0 <= i+d < n), then changing h[i] by delta changes corr_d by -delta*h[i+d]
    # Case 2: i is the "second" index, i.e., i+d = i' for some valid i': h[i']*(1-h[i])... wait
    #   Actually, h[i] appears as h[i+d] when the first index is i-d.
    #   If i-d is valid (0 <= i-d < n), then changing h[i] by delta changes corr_d by delta*(1-h[i-d])
    
    # So B_d = -(h[i+d] if 0<=i+d<n else 0) + (1-h[i-d] if 0<=i-d<n else 0)
    
    for k in range(2*n-1):
        d = k - (n-1)
        coeff = 0.0
        # Case 1: h[i] is the (1-h[i]) factor, i.e., i is the "first" index
        # corr_d has term h[i+d]*(1-h[i]) if 0 <= i+d < n
        if 0 <= i + d < n:
            coeff -= h[i + d]
        # Case 2: h[i] is the h[i+d] factor, i.e., i = i'+d, so i' = i-d
        # corr_d has term h[i]*(1-h[i-d]) if 0 <= i-d < n
        if 0 <= i - d < n:
            coeff += (1.0 - h[i - d])
        B[k] = coeff
    
    # Now corr_d(t) = corr_d(old) + (t - old) * B[k] for each k
    corrs_old = np.correlate(h, 1.0 - h, mode='full')
    # We want to minimize max_k (corrs_old[k] + (t-old)*B[k]) over t in [0,1]
    
    # This is min_t max_k (a_k + t*B[k]) where a_k = corrs_old[k] - old*B[k]
    a = corrs_old - old * B
    
    # The function f(t) = max_k (a_k + t*B[k]) is convex in t.
    # Minimum is where the subgradient contains 0, i.e., there exist k1, k2
    # with a_k1 + t*B_k1 = a_k2 + t*B_k2 = f(t), B_k1 < 0 < B_k2 (or similar).
    
    # Simple approach: ternary search or check all breakpoints.
    # Breakpoints: t = (a_k2 - a_k1) / (B_k1 - B_k2) for pairs (k1, k2) with B_k1 != B_k2
    # But there are O(n^2) breakpoints. For n=126, that's ~16000, which is fine.
    
    # Actually, simpler: since f is convex and piecewise linear, the minimum
    # is at a breakpoint or at the boundary. Let's just do a fine grid + refinement.
    
    # Even simpler: use the fact that the optimum t* satisfies:
    # min over {k: B[k] < 0} of (a[k] - t*B[k]) ... no.
    
    # Let me just do a sorted approach:
    # f(t) = max_k (a_k + B_k * t)
    # The active set changes at breakpoints. The minimum of a convex piecewise-linear
    # function is where the slope changes sign.
    # Sort breakpoints and find where slope goes from negative to positive.
    
    # For efficiency with n=126, let's just evaluate on a fine grid and then refine.
    t_grid = np.linspace(0, 1, 2001)
    vals = np.max(a[None, :] + t_grid[:, None] * B[None, :], axis=1)
    t_star = t_grid[np.argmin(vals)]
    
    # Refine with golden section
    lo, hi = max(0, t_grid[max(0, np.argmin(vals)-1)]) if np.argmin(vals) > 0 else 0.0
    hi = min(1, t_grid[min(len(t_grid)-1, np.argmin(vals)+1)]) if np.argmin(vals) < len(t_grid)-1 else 1.0
    
    def f_of_t(t):
        return np.max(a + t * B)
    
    # Golden section
    gr = (np.sqrt(5) - 1) / 2
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc, fd = f_of_t(c), f_of_t(d)
    for _ in range(50):
        if fc < fd:
            hi, d, fd = d, c, fc
            c = hi - gr * (hi - lo)
            fc = f_of_t(c)
        else:
            lo, c, fc = c, d, fd
            d = lo + gr * (hi - lo)
            fd = f_of_t(d)
    t_star = (lo + hi) / 2
    
    return t_star

def coordinate_descent(h, n_sweeps=10, rng=None):
    """Perform coordinate descent with exact 1D subproblems."""
    n = len(h)
    h = h.copy()
    target_sum = n / 2.0
    
    for sweep in range(n_sweeps):
        # Random order for diversity
        if rng is not None:
            order = rng.permutation(n)
        else:
            order = np.arange(n)
        
        for i in order:
            t_star = coord_subproblem(h, i, n)
            # Update h[i] and fix the sum
            old_val = h[i]
            h[i] = t_star
            diff = target_sum - h.sum()
            if abs(diff) > 1e-12:
                # Distribute among other coordinates
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                # Only adjust coordinates that have room
                room_up = (1.0 - h[mask])
                room_down = h[mask]
                if diff > 0:
                    # Need to increase others
                    adj = np.minimum(room_up, diff / mask.sum())
                    # This is approximate; project at the end
                    h[mask] += diff / mask.sum()
                else:
                    h[mask] += diff / mask.sum()
            # Clip
            h = np.clip(h, 0.0, 1.0)
            # Re-project to fix sum
            h = project_feasible(h, target_sum)
    
    return h

def smooth_obj_and_grad(h, n, beta):
    """Compute logsumexp(beta * corrs)/beta and its gradient."""
    dx = 2.0 / n
    corrs = np.correlate(h, 1.0 - h, mode='full')
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    ws = w.sum()
    if ws < 1e-30:
        w = np.ones(2*n-1) / (2*n-1)
    else:
        w /= ws
    val = (corrs.max() + np.log(ws) / beta) * dx
    
    # Gradient
    grad = np.zeros(n)
    for k in range(2*n-1):
        wk = w[k] * beta * dx
        if wk < 1e-15:
            continue
        d = k - (n-1)
        i_lo = max(0, -d)
        i_hi = min(n-1, n-1-d)
        if i_lo > i_hi:
            continue
        idx = np.arange(i_lo, i_hi+1)
        grad[idx] += wk * (1.0 - h[idx+d])
        grad[idx+d] += wk * (-h[idx])
    return val, grad

def smooth_optimize(h0, n, beta, maxiter=300):
    """Optimize using smooth objective with SLSQP."""
    target_sum = n / 2.0
    
    def obj(x):
        h = project_feasible(x, target_sum)
        corrs = np.correlate(h, 1.0-h, mode='full')
        return logsumexp(beta * corrs) / beta
    
    def grad(x):
        h = project_feasible(x, target_sum)
        return smooth_obj_and_grad(h, n, beta)[1]
    
    try:
        res = minimize(obj, h0, jac=grad, method='SLSQP',
                      bounds=[(0.0, 1.0)] * n,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        h_out = project_feasible(res.x, target_sum)
        return h_out, exact_score(h_out)
    except Exception:
        return h0, exact_score(h0)

One more thought: I should also try a "binary search on t" approach:
- Guess a target C₅ = t.

Actually, for the subgradient method:
- Start with some h.
- Compute all correlations.
- Find the active lag d* (the one with max correlation).
- Compute the gradient of corr_{d*} with respect to h.
- Take a step: h ← h - α * grad(corr_{d*})
- Project to feasible set.
- Repeat.

This is essentially the "cutting plane" or "column generation" approach for minimax.

import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time

def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) / 2, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-10:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def exact_score(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full').max() * (2.0 / n)

def corr_all(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def subgrad_max_corr(h, n):
    """Compute subgradient of max_d corr_d w.r.t. h."""
    dx = 2.0 / n
    corrs_raw = np.correlate(h, 1.0 - h, mode='full')
    max_val = corrs_raw.max()
    active = (corrs_raw >= max_val - 1e-10).astype(float)
    # Average subgradient over active set
    active /= active.sum()
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = active[k]
        if wk < 1e-15:
            continue
        d = k - (n - 1)
        i_lo = max(0, -d)
        i_hi = min(n - 1, n - 1 - d)
        if i_lo > i_hi:
            continue
        idx = np.arange(i_lo, i_hi + 1)
        grad[idx] += wk * (1.0 - h[idx + d])
        grad[idx + d] += wk * (-h[idx])
    return grad * dx

def subgradient_method(h, n, n_iters=200, alpha0=0.01):
    """Subgradient method for the minimax problem."""
    h = h.copy()
    target_sum = n / 2.0
    best_h = h.copy()
    best_c5 = exact_score(h)
    
    for it in range(n_iters):
        alpha = alpha0 / (1 + 0.01 * it)
        g = subgrad_max_corr(h, n)
        h = project_feasible(h - alpha * g, target_sum)
        c5 = exact_score(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5

def coord_subproblem_fast(h, i, n):
    """Fast 1D subproblem for coordinate i."""
    dx = 2.0 / n
    # Compute B vector: coefficient of t in corr_k
    # corr_k = sum over valid i' of h[i'+d] * (1-h[i']) where d = k-(n-1)
    # Changing h[i] affects:
    #   - term where i' = i: h[i+d]*(1-h[i]) -> contributes -h[i+d] to B[k] (if i+d valid)
    #   - term where i'+d = i, i.e., i' = i-d: h[i]*(1-h[i-d]) -> contributes (1-h[i-d]) to B[k] (if i-d valid)
    
    B = np.zeros(2 * n - 1)
    for k in range(2 * n - 1):
        d = k - (n - 1)
        coeff = 0.0
        if 0 <= i + d < n:
            coeff -= h[i + d]
        if 0 <= i - d < n:
            coeff += (1.0 - h[i - d])
        B[k] = coeff
    
    corrs_raw = np.correlate(h, 1.0 - h, mode='full')
    old = h[i]
    a = corrs_raw - old * B
    
    # Minimize max_k (a_k + t * B_k) over t in [0,1]
    # Vectorized evaluation
    t_grid = np.linspace(0, 1, 501)
    vals = np.max(a[None, :] + t_grid[:, None] * B[None, :], axis=1)
    t_star = t_grid[np.argmin(vals)]
    
    # Refine
    def f_of_t(t):
        return np.max(a + t * B)
    
    # Golden section around t_star
    step = 2.0 / 500.0
    lo = max(0.0, t_star - step)
    hi = min(1.0, t_star + step)
    gr = (np.sqrt(5) - 1) / 2
    c = hi - gr * (hi - lo)
    d_pt = lo + gr * (hi - lo)
    fc, fd = f_of_t(c), f_of_t(d_pt)
    for _ in range(30):
        if fc < fd:
            hi, d_pt, fd = d_pt, c, fc
            c = hi - gr * (hi - lo)
            fc = f_of_t(c)
        else:
            lo, c, fc = c, d_pt, fd
            d_pt = lo + gr * (hi - lo)
            fd = f_of_t(d_pt)
    t_star = (lo + hi) / 2
    
    return t_star

def coordinate_descent(h, n_sweeps=10, rng=None, verbose=False):
    n = len(h)
    h = h.copy()
    target_sum = n / 2.0
    best_c5 = exact_score(h)
    
    for sweep in range(n_sweeps):
        order = rng.permutation(n) if rng is not None else np.arange(n)
        improved = False
        for i in order:
            t_star = coord_subproblem_fast(h, i, n)
            old_val = h[i]
            if abs(t_star - old_val) < 1e-10:
                continue
            h[i] = t_star
            # Fix sum
            diff = target_sum - h.sum()
            if abs(diff) > 1e-12:
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                h[mask] += diff / (n - 1)
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target_sum)
            improved = True
        
        c5 = exact_score(h)
        if c5 < best_c5:
            best_c5 = c5
        if verbose:
            print(f"    CD sweep {sweep}: C5={c5:.6f}")
        if not improved:
            break
    
    return h, best_c5
