class SoftMaxProblem:
    def __init__(self, anchor, target_sum, beta, dx, rho=0.0):
        self.n = len(anchor)
        self.target_sum = target_sum
        self.beta = beta
        self.dx = dx
        self.anchor = anchor
        self.rho = rho
    
    def obj(self, x):
        h = x  # no s variable in soft-max
        corrs = compute_raw_corrs(h)
        cd = corrs * self.dx
        val = logsumexp(self.beta * cd) / self.beta
        if self.rho > 0:
            val += self.rho * np.sum((h - self.anchor)**2) / self.n
        return val
    
    def grad(self, x):
        h = x

        cd = corrs * self.dx
        shifted = cd - cd.max()
        w = np.exp(self.beta * shifted)
        w /= w.sum()

import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time

def project_feasible(x, target_sum):
    n = len(x)
    if n == 0:
        return x.copy()
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-13:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_raw_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def c5_exact(h):
    n = len(h)
    return float(compute_raw_corrs(h).max() * 2.0 / n)

def corr_grad(h, k):
    n = len(h)
    grad = np.zeros(n)
    d = k - (n - 1)
    if d == 0:
        grad[:] = 1.0 - 2.0 * h
    elif d > 0:
        m_hi = n - d
        grad[:m_hi] += (1.0 - h[d:])
        grad[d:] += (-h[:m_hi])
    else:
        p = -d
        m_hi = n - p
        grad[p:] += (1.0 - h[:m_hi])
        grad[:m_hi] += (-h[p:])
    return grad

def make_anti_pal(m, v):
    n = 2 * m
    h = np.empty(n)
    h[:m] = np.clip(v, 0, 1)
    h[m:] = 1.0 - h[:m][::-1]
    return h

class SoftMaxProb:
    def __init__(self, anchor, target_sum, beta, rho):
        self.n = len(anchor)
        self.target_sum = target_sum
        self.beta = beta
        self.dx = 2.0 / self.n
        self.anchor = anchor
        self.rho = rho
    def obj(self, x):
        corrs = compute_raw_corrs(x)
        cd = corrs * self.dx
        val = float(logsumexp(self.beta * cd) / self.beta)
        if self.rho > 0:
            val += self.rho * float(np.sum((x - self.anchor)**2)) / self.n
        return val
    def grad(self, x):
        n = self.n
        corrs = compute_raw_corrs(x)
        cd = corrs * self.dx
        shifted = cd - cd.max()
        w = np.exp(self.beta * shifted)
        ws = w.sum()
        if ws < 1e-300:
            w = np.ones(2*n-1) / (2*n-1)
        else:
            w /= ws
        g = np.zeros(n)
        for k in range(2*n-1):
            if w[k] > 1e-15:
                g += w[k] * corr_grad(x, k)
        g *= self.dx
        if self.rho > 0:
            g += 2.0 * self.rho * (x - self.anchor) / self.n
        return g
    def eq_fun(self, x):
        return x.sum() - self.target_sum
    def eq_jac(self, x):
        return np.ones(self.n)

class EpigraphProb:
    def __init__(self, anchor, target_sum, target_raw, rho, active_k):
        self.n = len(anchor)
        self.target_sum = target_sum
        self.target_raw = target_raw
        self.rho = rho
        self.anchor = anchor.copy()
        self.active_k = active_k
    def obj(self, x):
        h = x[:self.n]
        s = x[self.n]
        val = s
        if self.rho > 0:
            val += self.rho * float(np.sum((h - self.anchor)**2))
        return float(val)
    def obj_grad(self, x):
        h = x[:self.n]
        g = np.zeros(self.n + 1)
        g[self.n] = 1.0
        if self.rho > 0:
            g[:self.n] = 2.0 * self.rho * (h - self.anchor)
        return g
    def eq_fun(self, x):
        return x[:self.n].sum() - self.target_sum
    def eq_jac(self, x):
        j = np.zeros(self.n + 1)
        j[:self.n] = 1.0
        return j
    def ineq_fun(self, x):
        h = x[:self.n]
        s = x[self.n]
        corrs = compute_raw_corrs(h)
        return self.target_raw + s - corrs[self.active_k]
    def ineq_jac(self, x):
        h = x[:self.n]
        n = self.n
        ncon = len(self.active_k)
        J = np.zeros((ncon, n + 1))
        for i, k in enumerate(self.active_k):
            J[i, :n] = -corr_grad(h, k)
            J[i, n] = 1.0
        return J

def solve_softmax(h0, target_sum, beta, maxiter=500):
    n = len(h0)
    prob = SoftMaxProb(h0, target_sum, beta, 0.0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': prob.eq_fun, 'jac': prob.eq_jac}]
    try:
        res = minimize(prob.obj, h0, jac=prob.grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        h_out = project_feasible(res.x, target_sum)
    except Exception:
        h_out = project_feasible(h0, target_sum)
    return h_out, c5_exact(h_out)

def solve_epigraph(h0, target_sum, target_c5, rho, maxiter=200, n_active=30):
    n = len(h0)
    target_raw = target_c5 * n / 2.0
    corrs = compute_raw_corrs(h0)
    s0 = max(0.0, corrs.max() - target_raw) + 0.005
    x0 = np.concatenate([h0, [s0]])
    active = np.argsort(corrs)[-n_active:]
    active = np.sort(active)
    prob = EpigraphProb(h0, target_sum, target_raw, rho, active)
    bounds = [(0.0, 1.0)] * n + [(0.0, target_raw * 2.0 + n)]
    cons = [
        {'type': 'eq', 'fun': prob.eq_fun, 'jac': prob.eq_jac},
        {'type': 'ineq', 'fun': prob.ineq_fun, 'jac': prob.ineq_jac}
    ]
    try:
        res = minimize(prob.obj, x0, jac=prob.obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-13})
        h_out = project_feasible(res.x[:n], target_sum)
    except Exception:
        h_out = project_feasible(h0, target_sum)
    return h_out, c5_exact(h_out)

def active_set_descent(h0, target_sum, target_c5, rho, max_rounds=8, n_active=30):
    n = len(h0)
    target_raw = target_c5 * n / 2.0
    h = h0.copy()
    best_h = h.copy()

        corrs = compute_raw_corrs(h)
        active = np.argsort(corrs)[-n_active:]
        active = np.sort(active)
        s0 = max(0.0, corrs[active].max() - target_raw) + 0.005
        x0 = np.concatenate([h, [s0]])
        prob = EpigraphProb(h, target_sum, target_raw, rho, active)
        bounds = [(0.0, 1.0)] * n + [(0.0, target_raw * 2.0 + n)]
        cons = [
            {'type': 'eq', 'fun': prob.eq_fun, 'jac': prob.eq_jac},
            {'type': 'ineq', 'fun': prob.ineq_fun, 'jac': prob.ineq_jac}
        ]
        try:
            res = minimize(prob.obj, x0, jac=prob.obj_grad, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 250, 'ftol': 1e-13})
            h_new = project_feasible(res.x[:n], target_sum)
        except Exception:
            h_new = h
        c5_new = c5_exact(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        if c5_new <= target_c5 + 1e-9:
            h = h_new
            break
        # Check if new lags became active
        corrs_new = compute_raw_corrs(h_new)
        thr = corrs_new.max() * 0.999
        new_active = np.where(corrs_new >= thr)[0]
        if len(new_active) > 0:
            # Include lags near the max
            active_new = np.sort(np.unique(np.concatenate([active, new_active]))[-n_active:])
            active = active_new
        h = h_new
    return best_h, best_c5

def subgrad_topk(h0, target_sum, n_active=15, max_iter=5000):

            no_improve = 0

            no_improve += 1

            h = h_new
            alpha = min(alpha * 1.005, 0.35)
