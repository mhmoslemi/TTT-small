def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)


def corr_vals(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx


def make_sym_h(q, n):
    m = n // 2
    q = np.clip(q, 0.0, 1.0)
    h = np.empty(n)
    h[:m] = q
    h[m:] = 1.0 - q[::-1]
    return h


def make_q_from_h(h, n):
    m = n // 2
    q = 0.5 * (h[:m] + 1.0 - h[n-1:n-1-m:-1])
    return np.clip(q, 0.0, 1.0)


def active_set(corrs, top_k):
    idx = np.argsort(corrs)[::-1][:top_k]
    return sorted(idx.tolist())


def _obj(x, n, beta):
    h = make_sym_h(x, n)
    corrs = corr_vals(h)
    return logsumexp(beta * corrs) / beta


def _grad(x, n, beta):
    h = make_sym_h(x, n)
    corrs = corr_vals(h)
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    ws = w.sum()
    if ws < 1e-30:
        w = np.ones(len(corrs)) / len(corrs)
    else:
        w /= ws
    dx = 2.0 / n
    G = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k] * beta
        if wk < 1e-15:
            continue
        d = k - (n - 1)
        i_lo = max(0, -d)
        i_hi = min(n - 1, n - 1 - d)
        if i_lo > i_hi:
            continue
        idx = np.arange(i_lo, i_hi + 1)
        G[idx] += wk * (1.0 - h[idx + d])
        G[idx + d] += wk * (-h[idx])
    G *= dx
    m = n // 2
    return G[:m] - G[n-1:n-1-m:-1]


def _de_obj(q, n):
    h = make_sym_h(q, n)
    return exact_score(h)


def _epi_fun(x, n):
    return x[0]


def _epi_jac(x, n):
    m = n // 2
    g = np.zeros(m + 1)
    g[0] = 1.0
    return g


def _epi_con_fun(x, n, active):
    h = make_sym_h(x[1:], n)
    corrs = corr_vals(h)
    return x[0] - corrs[np.array(active)]


def _epi_con_jac(x, n, active):
    h = make_sym_h(x[1:], n)
    m = n // 2
    dx = 2.0 / n
    L = len(active)
    J = np.zeros((L, m + 1))
    J[:, 0] = 1.0
    for i, lag in enumerate(active):
        d = lag - (n - 1)
        i_lo = max(0, -d)
        i_hi = min(n - 1, n - 1 - d)
        if i_lo > i_hi:
            continue
        idx = np.arange(i_lo, i_hi + 1)
        G = np.zeros(n)
        G[idx] += (1.0 - h[idx + d])
        G[idx + d] += (-h[idx])
        G *= dx
        J[i, 1:] = -(G[:m] - G[n-1:n-1-m:-1])
    return J


def smooth_lbfgs(x0, n, beta, maxiter=500):
    bounds = [(0.0, 1.0)] * (n // 2)
    res = minimize(
        partial(_obj, n=n, beta=beta),
        x0,
        jac=partial(_grad, n=n, beta=beta),
        method='L-BFGS-B',
        bounds=bounds,
        options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-14}
    )
    h = make_sym_h(res.x, n)
    return res.x, exact_score(h)


def epi_slsqp(q0, n, active, t_init, maxiter=500):
    m = n // 2
    x0 = np.concatenate([[t_init], q0])
    bounds = [(0.0, 2.0)] + [(0.0, 1.0)] * m
    cons = [{'type': 'ineq',
             'fun': partial(_epi_con_fun, n=n, active=active),
             'jac': partial(_epi_con_jac, n=n, active=active)}]
    try:
        res = minimize(partial(_epi_fun, n=n), x0,
                       jac=partial(_epi_jac, n=n),
                       method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        h_new = make_sym_h(res.x[1:], n)
        c5 = exact_score(h_new)
        return res.x[1:], c5
    except Exception:
        return None, None


def epi_tc(q0, n, active, t_init, maxiter=200):
    m = n // 2
    active_arr = np.array(active)
    x0 = np.concatenate([[t_init], q0])
    bounds = Bounds(np.zeros(m + 1), np.concatenate([[2.0], np.ones(m)]))
    con = NonlinearConstraint(
        partial(_epi_con_fun, n=n, active=active_arr),
        0.0, np.inf,
        partial(_epi_con_jac, n=n, active=active_arr)
    )
    try:
        res = trust_constr(partial(_epi_fun, n=n), x0,
                           jac=partial(_epi_jac, n=n),
                           constraints=[con], bounds=bounds,
                           maxiter=maxiter, gtol=1e-12, xtol=1e-14)
        if res.x is not None:
            h_new = make_sym_h(res.x[1:], n)
            c5 = exact_score(h_new)
            return res.x[1:], c5
    except Exception:
        pass
    return None, None


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s - 5.0 - (time.time() - t0)

    try:
        h_init = np.array(initial_h_values, dtype=float)
    except NameError:
        h_init = np.full(126, 0.5)

    n = len(h_init)
    if n % 2 != 0:
        h_init = h_init[:n - 1]
        n = len(h_init)

    h_best = project_feasible(h_init, n // 2.0)
    c5_best = exact_score(h_best)
    m = n // 2
    q_best = make_q_from_h(h_best, n)
    print(f"Initial: n={n}, C5={c5_best:.7f}")

    # Phase 1: Smooth L-BFGS polish with increasing beta
    for beta in [50, 100, 200, 400, 800, 1600]:
        if time_left() < 15:
            break
        q_new, c5_new = smooth_lbfgs(q_best, n, beta, maxiter=400)
        if c5_new < c5_best - 1e-11:
            c5_best = c5_new
            q_best = q_new.copy()
            h_best = make_sym_h(q_best, n)
            print(f"  smooth beta={beta}: C5={c5_best:.7f}")

    # Phase 2: Epigraph active-set (SLSQP)
    for rnd in range(15):
        if time_left() < 15:
            break
        corrs = corr_vals(make_sym_h(q_best, n))
        n_act = min(15 + 12 * rnd, 2 * n - 1)
        active = active_set(corrs, n_act)
        t_init = corrs.max() * 1.000001
        q_new, c5_new = epi_slsqp(q_best, n, active, t_init, maxiter=500)
        if q_new is None or c5_new > c5_best + 1e-5:
            q_new, c5_new = epi_tc(q_best, n, active, t_init, maxiter=150)
        if q_new is not None and c5_new < c5_best - 1e-12:
            c5_best = c5_new
            q_best = q_new.copy()
            h_best = make_sym_h(q_best, n)
            print(f"  epi rnd={rnd} (|A|={n_act}): C5={c5_best:.7f}")

    # Phase 3: Global search via differential evolution
    if time_left() > 150:
        print("  DE global search...")
        de_maxiter = max(20, min(60, int((time_left() * 0.35) / 3)))
        try:
            de_res = differential_evolution(
                partial(_de_obj, n=n),
                bounds=[(0.0, 1.0)] * m,
                seed=seed,
                maxiter=de_maxiter,
                popsize=25,
                tol=1e-6,
                polish=True,
                workers=1,
                init='sohal'
            )
            h_de = make_sym_h(de_res.x, n)
            c5_de = exact_score(h_de)
            print(f"  DE: C5={c5_de:.7f}")
            if c5_de < c5_best - 1e-10:
                c5_best = c5_de
                q_best = de_res.x.copy()
                h_best = h_de
        except Exception as e:
            print(f"  DE failed: {e}")

    # Phase 4: Re-polish after global
    if time_left() > 40:
        for beta in [200, 600, 1200, 2400]:
            if time_left() < 12:
                break
            q_new, c5_new = smooth_lbfgs(q_best, n, beta, maxiter=500)
            if c5_new < c5_best - 1e-11:
                c5_best = c5_new
                q_best = q_new.copy()
                h_best = make_sym_h(q_best, n)
                print(f"  re-polish beta={beta}: C5={c5_best:.7f}")

    # Phase 5: Final epigraph refinement
    for _ in range(5):
        if time_left() < 12:
            break
        corrs = corr_vals(make_sym_h(q_best, n))
        active = active_set(corrs, min(90, 2 * n - 1))
        t_init = corrs.max() * 1.000001
        q_new, c5_new = epi_slsqp(q_best, n, active, t_init, maxiter=600)
        if q_new is None or c5_new > c5_best + 1e-5:
            q_new, c5_new = epi_tc(q_best, n, active, t_init, maxiter=200)
        if q_new is not None and c5_new < c5_best - 1e-12:
            c5_best = c5_new
            q_best = q_new.copy()
            h_best = make_sym_h(q_best, n)
            print(f"  refine: C5={c5_best:.7f}")
        else:
            break

    # Phase 6: Multi-start with diverse perturbations
    n_trials = 80
    for trial in range(n_trials):
        if time_left() < 8:
            break
        # Vary the scale and direction of perturbation
        mode = trial % 4
        if mode == 0:
            scale = 0.02 + 0.25 * (trial / n_trials)
            q_pert = q_best + scale * rng.randn(m)
        elif mode == 1:
            # Targeted: perturb variables most correlated with active lags
            corrs = corr_vals(make_sym_h(q_best, n))
            top_lag = np.argmax(corrs)
            d = top_lag - (n - 1)
            mask = np.zeros(m)
            i_lo = max(0, -d)
            i_hi = min(m - 1, n - 1 - d)
            if i_lo <= i_hi:
                mask[i_lo:i_hi+1] = 1.0
            mask[m-1 - np.arange(max(0, -d), min(m, n-1-d)+1)] = 1.0
            n_active = max(1, int(mask.sum()))
            q_pert = q_best + 0.1 * rng.randn(m) * mask / np.sqrt(n_active)
        elif mode == 2:
            # Smooth perturbation (low frequency)
            noise = rng.randn(m)
            # Simple averaging filter
            kernel = np.ones(5) / 5.0
            noise_f = np.convolve(noise, kernel, mode='same')
            q_pert = q_best + 0.15 * noise_f
        else:
            # Sparse perturbation
            n_change = max(3, m // 10)
            idx_change = rng.choice(m, size=n_change, replace=False)
            q_pert = q_best.copy()
            q_pert[idx_change] += 0.15 * rng.randn(n_change)

        q_pert = np.clip(q_pert, 0.0, 1.0)
        q_new, c5_new = smooth_lbfgs(q_pert, n, 200, maxiter=200)
        if c5_new < c5_best + 0.004:
            q_new2, c5_new2 = smooth_lbfgs(q_new, n, 800, maxiter=300)
            if c5_new2 < c5_best - 1e-11:
                c5_best = c5_new2
                q_best = q_new2.copy()
                h_best = make_sym_h(q_best, n)
                print(f"  trial {trial} (mode={mode}): C5={c5_best:.7f}")
                # Quick epigraph
                if time_left() > 15:
                    corrs2 = corr_vals(h_best)
                    act2 = active_set(corrs2, 50)
                    q_ep, c5_ep = epi_slsqp(q_best, n, act2,
                                            corrs2.max() * 1.000001, maxiter=300)
                    if q_ep is not None and c5_ep < c5_best - 1e-12:
                        c5_best = c5_ep
                        q_best = q_ep.copy()
                        h_best = make_sym_h(q_best, n)

    # Phase 7: Try nearby n values
    if time_left() > 30:
        for delta in [2, -2, 4, -4]:
            if time_left() < 20:
                break
            n_try = n + delta
            if n_try < 40 or n_try % 2 != 0:
                continue
            x_old = np.linspace(0, 2, n, endpoint=False)
            x_new = np.linspace(0, 2, n_try, endpoint=False)
            h_interp = np.interp(x_new, x_old, h_best)
            h_interp = project_feasible(h_interp, n_try // 2.0)
            m_try = n_try // 2
            q_try = make_q_from_h(h_interp, n_try)

            q_new, c5_new = smooth_lbfgs(q_try, n_try, 300, maxiter=300)
            if time_left() > 15 and c5_new < c5_best:
                q_new, c5_new = smooth_lbfgs(q_new, n_try, 1000, maxiter=400)
            if c5_new < c5_best - 1e-9:
                c5_best = c5_new
                n = n_try
                m = m_try
                q_best = q_new.copy()
                h_best = make_sym_h(q_best, n)
                print(f"  n={n}: C5={c5_best:.7f}")

    # Final validation
    h_best = np.clip(h_best, 0.0, 1.0)
    h_best = project_feasible(h_best, len(h_best) // 2.0)
    c5_best = exact_score(h_best)
    print(f"Final: n={len(h_best)}, C5={c5_best:.7f}")

    return h_best, c5_best, len(h_best)
