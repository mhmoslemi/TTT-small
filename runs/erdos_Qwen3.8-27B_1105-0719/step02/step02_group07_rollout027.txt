import numpy as np
from scipy.special import logsumexp
import time


def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-10:
        mask = (h > 1e-8) & (h < 1.0 - 1e-8)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)


def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)


def corr_values(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx


def minimize_1d_convex_pl(a, b, delta_lo, delta_hi, max_iter=45):
    if delta_lo >= delta_hi - 1e-15:
        t = min(max(delta_lo, 0.0), delta_hi) if delta_hi > 0 else delta_lo
        t = np.clip(t, delta_lo, delta_hi)
        return t, (a * t + b).max()
    lo = delta_lo
    hi = delta_hi
    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        vals = a * mid + b
        max_val = vals.max()
        tol = 1e-13 * max(1.0, abs(max_val))
        active = vals >= max_val - tol
        if not np.any(active):
            idx = np.argmax(vals)
            active = np.zeros_like(vals, dtype=bool)
            active[idx] = True
        smin = a[active].min()
        smax = a[active].max()
        if smin <= 0.0 <= smax:
            break
        if smax < 0.0:
            lo = mid
        else:
            hi = mid
    t_star = (lo + hi) / 2.0
    f_star = (a * t_star + b).max()
    return t_star, f_star


def compute_pair_a(h, j, l, n):
    n_corr = 2 * n - 1
    d_vals = np.arange(n_corr) - (n - 1)
    a = np.zeros(n_corr)
    jd = j + d_vals
    m1 = (jd >= 0) & (jd < n)
    a[m1] += (1.0 - h[jd[m1]])
    jd2 = j - d_vals
    m2 = (jd2 >= 0) & (jd2 < n)
    a[m2] -= h[jd2[m2]]
    ld = l + d_vals
    m3 = (ld >= 0) & (ld < n)
    a[m3] -= (1.0 - h[ld[m3]])
    ld2 = l - d_vals
    m4 = (ld2 >= 0) & (ld2 < n)
    a[m4] += h[ld2[m4]]
    return a


def pair_cdc_sweep(h, rng, n_partners=1):
    n = len(h)
    h_cur = h.copy()
    for j in rng.permutation(n):
        for _ in range(n_partners):
            l = rng.randint(n)
            if l == j:
                l = (l + 1) % n
            delta_lo = max(-h_cur[j], h_cur[l] - 1.0)
            delta_hi = min(1.0 - h_cur[j], h_cur[l])
            if delta_lo >= delta_hi - 1e-14:
                continue
            a = compute_pair_a(h_cur, j, l, n)
            corrs = np.correlate(h_cur, 1.0 - h_cur, mode='full') * (2.0 / n)
            delta_star, _ = minimize_1d_convex_pl(a, corrs, delta_lo, delta_hi)
            if abs(delta_star) > 1e-15:
                h_cur[j] = np.clip(h_cur[j] + delta_star, 0.0, 1.0)
                h_cur[l] = np.clip(h_cur[l] - delta_star, 0.0, 1.0)
    return h_cur


def greedy_pair_cdc_sweep(h, rng, n_candidates=4):
    n = len(h)
    h_cur = h.copy()
    dx = 2.0 / n
    for j in rng.permutation(n):
        corrs = np.correlate(h_cur, 1.0 - h_cur, mode='full') * dx
        best_f = corrs.max()
        best_delta = 0.0
        best_l = -1
        for _ in range(n_candidates):
            l = rng.randint(n)
            if l == j:
                l = (l + 1) % n
            delta_lo = max(-h_cur[j], h_cur[l] - 1.0)
            delta_hi = min(1.0 - h_cur[j], h_cur[l])
            if delta_lo >= delta_hi - 1e-14:
                continue
            a = compute_pair_a(h_cur, j, l, n)
            delta_star, f_star = minimize_1d_convex_pl(a, corrs, delta_lo, delta_hi)
            if f_star < best_f - 1e-13:
                best_f = f_star
                best_delta = delta_star
                best_l = l
        if best_l >= 0 and abs(best_delta) > 1e-15:
            h_cur[j] = np.clip(h_cur[j] + best_delta, 0.0, 1.0)
            h_cur[best_l] = np.clip(h_cur[best_l] - best_delta, 0.0, 1.0)
    return h_cur


def smooth_projected_gradient(h0, n, beta, n_steps, rng):
    h = h0.copy()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    d_vals = np.arange(n_corr) - (n - 1)
    target_sum = n / 2.0
    alpha = 1.0 / (beta * 0.02)
    for step in range(n_steps):
        corrs = np.correlate(h, 1.0 - h, mode='full') * dx
        shifted = corrs - corrs.max()
        w = np.exp(beta * shifted)
        ws = w.sum()
        if ws < 1e-30:
            w = np.ones(n_corr) / n_corr
        else:
            w /= ws
        grad = np.zeros(n)
        for k in range(n_corr):
            if w[k] < 1e-15:
                continue
            d = d_vals[k]
            i_lo = max(0, -d)
            i_hi = min(n - 1, n - 1 - d)
            if i_lo > i_hi:
                continue
            idx = np.arange(i_lo, i_hi + 1)
            grad[idx] += w[k] * (1.0 - h[idx + d])
            grad[idx + d] += w[k] * (-h[idx])
        grad *= dx * beta
        h_new = project_feasible(h - alpha * grad, target_sum)
        if exact_score(h_new) < exact_score(h):
            h = h_new
            alpha = min(alpha * 1.05, 2.0 / (beta * 0.01))
        else:
            alpha *= 0.7
    return h
