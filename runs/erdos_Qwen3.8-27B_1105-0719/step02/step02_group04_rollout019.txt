        q_r, c5_r = anneal(q_p, len(best_h), tau_med, t0, budget_s, reserve, maxiter=300, gtol=1e-9)
        h_r = build_h(q_r, len(best_h))
        if c5_r < best_c5:
            best_h, best_c5 = h_r, c5_r
            cur_q = q_r.copy()
        else:
            cur_q = q_r.copy()

    # Phase 3: n-value sweep
    n_center = len(best_h)
    for delta in range(-1, 1):
        for nc in sorted(set([n_center + d for d in range(-24, 25, 2) if n_center + d >= 70])):
            if nc % 2 != 0 or nc == n_center:
                continue
            if time.time() - t0 > budget_s - reserve - 20:
                break
            q_c = interp_q(q_of_h(best_h, len(best_h)), len(best_h), nc)
            tau_s = [0.06, 0.03, 0.015, 0.008, 0.004, 0.002]
            q_n, c5_n = anneal(q_c, nc, tau_s, t0, budget_s, reserve, maxiter=250, gtol=1e-8)
            h_n = build_h(q_n, nc)
            if c5_n < best_c5:
                best_h, best_c5 = h_n, c5_n

    # Phase 4: random restarts
    n_r_base = len(best_h)
    if n_r_base % 2 != 0:
        n_r_base -= 1
    for trial in range(40):
        if time.time() - t0 > budget_s - reserve - 15:
            break
        q_rand = random_q(n_r_base, rng)
        tau_rs = [0.1, 0.05, 0.025, 0.012, 0.006, 0.003]
        q_rr, c5_rr = anneal(q_rand, n_r_base, tau_rs, t0, budget_s, reserve, maxiter=200, gtol=1e-8)
        h_rr = build_h(q_rr, n_r_base)
        if c5_rr < best_c5:
            best_h, best_c5 = h_rr, c5_rr

    # Phase 5: final polish on best
    if time.time() - t0 < budget_s - reserve:
        nf = len(best_h)
        if nf % 2 != 0:
            nf -= 1
        qf = q_of_h(best_h, nf)
        tau_fin = [0.002, 0.001, 0.0005, 0.0003, 0.0001]
        q_fin, c5_fin = anneal(qf, nf, tau_fin, t0, budget_s, reserve, maxiter=600, gtol=1e-10)
        h_fin = build_h(q_fin, nf)
        if c5_fin < best_c5:
            best_h, best_c5 = h_fin, c5_fin

    # Phase 6: one more perturbation + ultra-fine anneal
    if time.time() - t0 < budget_s - reserve - 10:
        q_cur = q_of_h(best_h, len(best_h))
        for sc in [0.01, 0.005, 0.002]:
            if time.time() - t0 > budget_s - reserve - 5:
                break
            q_pt = perturb_q(q_cur, rng, scale=sc)
            tau_u = [0.005, 0.002, 0.001, 0.0005, 0.0002]
            q_u, c5_u = anneal(q_pt, len(best_h), tau_u, t0, budget_s, reserve, maxiter=400, gtol=1e-10)
            h_u = build_h(q_u, len(best_h))
            if c5_u < best_c5:
                best_h, best_c5 = h_u, c5_u
                q_cur = q_u.copy()
            else:
                q_cur = q_u.copy()

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = eval_c5(best_h)
    return best_h, best_c5, len(best_h)
