def coordinate_descent(h, dx, target_sum, max_sweeps=100):
    n = len(h)
    h = h.copy()
    best_c5 = c5_of_h(h, dx)
    
    for sweep in range(max_sweeps):
        improved = False
        for i in range(n):
            old_val = h[i]
            # Try different values of h[i]
            # The objective is convex in h[i], so try a fine grid
            # and also the current value
            candidates = np.linspace(0, 1, 50)
            best_val = old_val
            best_obj = best_c5
            
            for v in candidates:
                h[i] = v
                # Check sum constraint approximately
                obj = c5_of_h(h, dx)
                if obj < best_obj:
                    best_obj = obj
                    best_val = v
            h[i] = best_val
            
            # Project to satisfy sum constraint
            h = project_feasible(h, target_sum)
            
            new_c5 = c5_of_h(h, dx)
            if new_c5 < best_c5 - 1e-15:
                best_c5 = new_c5
                improved = True
        if not improved:
            break
    
    return h, best_c5

    return x[n]


def epiph_jac(x, n):
    g = np.zeros(n + 1)

    corrs = compute_corrs(h)
    return corrs[active_ks] - x[n]

def slsqp_epigraph(h0, dx, target_sum, max_active=12, max_rounds=15, maxiter=400):
    n = len(h0)
    h = h0.copy()

             'jac': partial(epiph_con_jac, n=n, active_ks=active_ks)}

def coord_descent(h0, dx, target_sum, max_sweeps=80, grid_size=40):
    n = len(h0)
    h = h0.copy()
    best_c5 = c5_of_h(h, dx)
    grid = np.linspace(0.0, 1.0, grid_size)

    for sweep in range(max_sweeps):
        improved = False
        for i in range(n):
            old = h[i]
            best_v = old
            best_obj = best_c5
            for v in grid:
                if abs(v - old) < 1e-6:
                    continue
                h[i] = v
                obj = c5_of_h(h, dx)
                if obj < best_obj:
                    best_obj = obj
                    best_v = v
            h[i] = best_v
            h = project_feasible(h, target_sum)
            new_c5 = c5_of_h(h, dx)
            if new_c5 < best_c5 - 1e-15:
                best_c5 = new_c5
                improved = True
        if not improved:
            break
    return h, best_c5

    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new / 2.0)

    elif pattern == 1:
        n_blocks = rng.randint(3, max(4, n // 15))
        block_len = n // n_blocks
        for b in range(n_blocks):
            val = rng.uniform(0.1, 0.9)
            start = b * block_len
            end = min(start + block_len, n)
            h[start:end] = val
    else:

        idx = rng.permutation(n)

        h[idx[:n_ones]] = 1.0
        frac = target_sum - n_ones
        if frac > 1e-10:
            h[idx[n_ones]] = frac
    return project_feasible(h, target_sum)
