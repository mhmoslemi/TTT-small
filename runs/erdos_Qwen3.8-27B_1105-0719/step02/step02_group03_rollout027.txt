```python
def compute_swap_delta(h, i, j, n):
    """Compute delta in correlations for swapping binary h[i]=1 -> 0 and h[j]=0 -> 1."""
    delta = np.zeros(2*n - 1)
    offsets = np.arange(-(n-1), n)
    
    for s in offsets:
        # Terms affected: m=i (if valid), m=j (if valid), m=i-s (if valid and ≠i,j), m=j-s (if valid and ≠i,j)
        change = 0.0
        
        # m = i + s... no wait. m is the first index, m+s is the second.
        # Term at (m, s): h[m]*(1-h[m+s])
        # Affected if m in {i,j} or m+s in {i,j}
        
        # m = i
        if 0 <= i + s <= n - 1:
            if i + s == j:
                # self-like interaction: term is h[i]*(1-h[j]) -> (0)*(1-1) = 0, was 1*(1-0) = 1
                change += 0 - 1  # = -1
            else:
                # h[i] goes 1->0, h[i+s] unchanged
                change += 0 - (1 - h[i+s])  # = -(1-h[i+s]) = h[i+s]-1
        # m = j
        if 0 <= j + s <= n - 1:
            if j + s == i:
                # term is h[j]*(1-h[i]) -> 1*(1-0) = 1, was 0*(1-1) = 0
                change += 1 - 0  # = +1
            else:
                # h[j] goes 0->1
                change += (1 - h[j+s]) - 0  # = 1-h[j+s]
        # m+s = i, so m = i-s
        if 0 <= i - s <= n - 1:
            m = i - s
            if m == i or m == j:
                pass  # already handled above (m==i is the s=0 case, m==j is handled in m=j case)
            else:
                # h[i] goes 1->0, so 1-h[i] goes 0->1
                change += h[m] * 1 - h[m] * 0  # = h[m]
        # m+s = j, so m = j-s
        if 0 <= j - s <= n - 1:
            m = j - s
            if m == i or m == j:
                pass  # already handled
            else:
                # h[j] goes 0->1, so 1-h[j] goes 1->0
                change += 0 - h[m] * 1  # = -h[m]
        
        delta[s + (n-1)] = change
    
    return delta
```

```python
def compute_swap_delta_binary(h, i, j, n):
    delta = np.zeros(2*n - 1)
    n_half = n - 1
    
    for s in range(-(n-1), n):
        if s == 0:
            continue  # no change for binary at s=0
        
        change = 0.0
        
        # m = i, term h[i]*(1-h[i+s])
        if 0 <= i + s <= n - 1:
            if i + s == j:
                change += -1  # 1*(1-0)=1 -> 0*(1-1)=0
            else:
                change += h[i+s] - 1  # 1*(1-h[i+s]) -> 0*(1-h[i+s])
        
        # m = j, term h[j]*(1-h[j+s])
        if 0 <= j + s <= n - 1:
            if j + s == i:
                change += 1  # 0*(1-1)=0 -> 1*(1-0)=1
            else:
                change += 1 - h[j+s]  # 0 -> 1*(1-h[j+s])
        
        # m+s = i, m = i-s, term h[i-s]*(1-h[i])
        m = i - s
        if 0 <= m <= n - 1 and m != i and m != j:
            change += h[m]  # h[m]*0 -> h[m]*1
        
        # m+s = j, m = j-s, term h[j-s]*(1-h[j])
        m = j - s
        if 0 <= m <= n - 1 and m != i and m != j:
            change += -h[m]  # h[m]*1 -> h[m]*0
        
        delta[s + n_half] = change
    
    return delta
```
