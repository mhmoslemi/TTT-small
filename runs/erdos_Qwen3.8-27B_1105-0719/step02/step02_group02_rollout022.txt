def solve_exact_max(h_init, n, dx, target_sum):
    # Variables: [h(0), ..., h(n-1), t]
    # Objective: minimize t
    # Constraints: t - corr_k(h) >= 0 for k = 0, ..., 2n-2
    #              sum(h) - target_sum = 0
    # Bounds: 0 <= h[i] <= 1, t >= 0
    
    def objective(x):
        return x[-1]  # t
    
    def obj_grad(x):
        g = np.zeros(n+1)
        g[-1] = 1.0
        return g
    
    def constraint_fun(x):
        h = x[:n]
        t = x[-1]
        corrs = np.correlate(h, 1.0 - h, mode='full')
        return t - corrs  # all must be >= 0
    
    def constraint_jac(x):
        h = x[:n]
        jac = np.zeros((2*n-1, n+1))
        for k in range(2*n-1):
            g = grad_corr_k(h, k, n)
            jac[k, :n] = -g  # gradient of (t - corr_k) w.r.t. h
            jac[k, n] = 1.0   # gradient w.r.t. t
        return jac
    
    def sum_fun(x):
        return x[:n].sum() - target_sum
    
    def sum_jac(x):
        g = np.zeros(n+1)
        g[:n] = 1.0
        return g
    
    bounds = [(0, 1)] * n + [(0, None)]
    constraints = [
        {'type': 'ineq', 'fun': constraint_fun, 'jac': constraint_jac},
        {'type': 'eq', 'fun': sum_fun, 'jac': sum_jac}
    ]
    
    x0 = np.concatenate([h_init, [max(np.correlate(h_init, 1-h_init, mode='full')) + 0.01]])
    
    res = minimize(objective, x0, jac=obj_grad, method='SLSQP',
                   bounds=bounds, constraints=constraints,
                   options={'maxiter': 500, 'ftol': 1e-12})
    
    h_out = res.x[:n]
    t_out = res.x[-1]
    h_out = project_feasible(h_out, target_sum)
    return h_out, c5_of_h(h_out, dx)

import numpy as np

def project_feasible(x, target_sum):
    n = len(x)
    if n == 0:
        return x
    lo, hi = -1.5, 1.5
    for _ in range(100):
        mid = (lo + hi) * 0.5
        h = np.clip(x - mid, 0.0, 1.0)
        s = h.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-10:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.any():
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h


def c5_of_h(h, dx):
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)


def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def coord_coefficients(h, i, n):
    """For coordinate i, compute coefficient a_k such that corr_k = a_k*h[i] + b_k."""
    a = np.zeros(2 * n - 1)
    for k in range(2 * n - 1):
        d = k - (n - 1)
        c = 0.0
        # h[i] as first factor in pair (i, i+d)
        if max(0, -d) <= i <= min(n - 1, n - 1 - d):
            c += (1.0 - h[i + d])
        # h[i] as second factor in pair (i-d, i)
        if max(d, 0) <= i <= min(n - 1, n - 1 + d):
            c += (-h[i - d])
        a[k] = c
    return a


def coord_optimal_value(h, i, n, corrs):
    """Find t in [0,1] minimizing max_k (a_k*t + b_k) for coordinate i."""
    a = coord_coefficients(h, i, n)
    b = corrs - a * h[i]
    
    # Ternary search on convex function
    lo, hi = 0.0, 1.0
    for _ in range(45):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        if np.max(a * m1 + b) < np.max(a * m2 + b):
            hi = m2
        else:
            lo = m1
    return (lo + hi) * 0.5


def coordinate_descent(h, n, dx, target_sum, n_sweeps=100, seed=42, check_interval=1):
    """Cyclic coordinate descent to minimize max correlation."""
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    h = h.copy()

        for i in order:
            corrs = compute_corrs(h)
