```python
def compute_swap_delta(h, i, j, n):
    """Compute change in all correlations when h[i]:1->0, h[j]:0->1."""
    e = np.arange(-(n-1), n)
    delta = np.zeros(2*n-1)
    
    # Terms involving i (h[i]: 1->0)
    mask_ip = (0 <= i + e) & (i + e < n)
    delta[mask_ip] += h[i + e[mask_ip]] - 1.0
    
    mask_im = (0 <= i - e) & (i - e < n)
    delta[mask_im] += h[i - e[mask_im]]
    
    # Terms involving j (h[j]: 0->1)
    mask_jp = (0 <= j + e) & (j + e < n)
    delta[mask_jp] += 1.0 - h[j + e[mask_jp]]
    
    mask_jm = (0 <= j - e) & (j - e < n)
    delta[mask_jm] -= h[j - e[mask_jm]]
    
    return delta
```

    best_max = corrs.max()
    
    for sweep in range(max_sweeps):
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        
        best_swap = None
        best_new_max = best_max
        
        for i in ones:
            for j in zeros:
                delta = compute_swap_delta(h, i, j, n)
                new_max = (corrs + delta).max()
                if new_max < best_new_max:
                    best_new_max = new_max
                    best_swap = (i, j)
        
        if best_swap is None or best_new_max >= best_max:
            break
        
        i, j = best_swap
        h[i], h[j] = 0.0, 1.0
        corrs = compute_corrs(h)  # Recompute to avoid accumulated errors
        best_max = corrs.max()
    
    return h, int(best_max)

import numpy as np
from scipy.optimize import minimize
import time


def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
        else:
            h += diff / n
    return np.clip(h, 0.0, 1.0)


def c5_of_h(h, dx):
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)


def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_swap_delta(h, i, j, n):
    """O(n) delta for swapping h[i]:1->0, h[j]:0->1 on all correlation values."""
    e = np.arange(-(n - 1), n)
    delta = np.zeros(2 * n - 1)
    ip = i + e
    im = i - e
    jp = j + e
    jm = j - e
    m1 = (ip >= 0) & (ip < n)
    delta[m1] += h[ip[m1]] - 1.0
    m2 = (im >= 0) & (im < n)
    delta[m2] += h[im[m2]]
    m3 = (jp >= 0) & (jp < n)
    delta[m3] += 1.0 - h[jp[m3]]
    m4 = (jm >= 0) & (jm < n)
    delta[m4] -= h[jm[m4]]
    return delta


def binary_best_improve(h0, max_sweeps, n):
    """Greedy best-improvement local search on binary sequences."""
    target = n // 2
    h = np.zeros(n)
    idx = np.argsort(-h0)
    h[idx[:target]] = 1.0
    corrs = compute_corrs(h)
    best_max = float(corrs.max())
    for _ in range(max_sweeps):
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        best_swap = None
        best_new = best_max
        for i in ones:
            for j in zeros:
                d = compute_swap_delta(h, i, j, n)
                nm = float((corrs + d).max())
                if nm < best_new - 1e-12:
                    best_new = nm
                    best_swap = (i, j)
        if best_swap is None:
            break
        i, j = best_swap
        h[i], h[j] = 0.0, 1.0
        corrs = compute_corrs(h)
        best_max = float(corrs.max())
    return h, int(best_max)

    """Simulated annealing for binary sequence optimization."""
    target = n // 2
    h = np.zeros(n)
    idx = np.argsort(-h0)
    h[idx[:target]] = 1.0
    corrs = compute_corrs(h)
    curr_max = float(corrs.max())
    best_max = curr_max
    best_h = h.copy()
    ones = list(np.where(h == 1.0)[0])

    no_imp = 0

        i = ones[rng.randint(len(ones))]
