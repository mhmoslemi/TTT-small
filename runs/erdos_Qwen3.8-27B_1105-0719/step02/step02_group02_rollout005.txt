    """Solve active-set minimax problem."""
    dx = 2.0 / n
    h = h0.copy()
    
    for outer in range(20):
        corrs = compute_corrs(h)
        max_val = corrs.max() * dx
        # Find active lags
        active = np.where(corrs * dx > max_val - tol)[0]
        if len(active) < 2:
            # Use top k
            active = np.argsort(corrs)[-top_k:]
        
        # Define: minimize t subject to corr[k]*dx <= t for k in active
        # Variable: [h, t]
        # This is still non-convex but we can use SLSQP
        
        # Actually, let's just minimize the average of active correlations
        # gradient = average of their individual gradients
        grad = np.zeros(n)
        for k in active:
            grad += corr_grad_k(h, k, n)
        grad /= len(active)
        grad *= dx
        
        # Line search: take step in -grad direction
        for step in [0.01, 0.005, 0.002, 0.001, 0.0005]:
            h_new = project_feasible(h - step * grad, target_sum)
            new_c5 = c5_of_h(h_new)
            if new_c5 < max_val:
                h = h_new
                break

import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time


def project_feasible(x, target_sum):
    n = len(x)
    x = np.asarray(x, dtype=float)
    lo = -3.0
    hi = 3.0
    # Ensure bracket
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-13:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h


def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def c5_of_h(h):
    n = len(h)
    return float(compute_corrs(h).max() * (2.0 / n))


def corr_grad_k(h, k, n):
    """Gradient of corr[k] w.r.t. h where corr = np.correlate(h, 1-h, 'full')."""
    grad = np.zeros(n)
    d = k - (n - 1)
    if d == 0:
        grad[:] = 1.0 - 2.0 * h
    elif d > 0:
        m = np.arange(0, n - d)
        grad[m] += -h[m + d]
        grad[m + d] += (1.0 - h[m])
    else:
        m = np.arange(-d, n)
        grad[m] += -h[m + d]
        grad[m + d] += (1.0 - h[m])
    return grad


class LSEObjective:
    def __init__(self, n, beta):
        self.n = n
        self.beta = beta
        self.dx = 2.0 / n

    def fun(self, x):
        c = compute_corrs(x)
        cd = c * self.dx
        return float(logsumexp(self.beta * cd) / self.beta)

    def jac(self, x):
        c = compute_corrs(x)
        cd = c * self.dx
        max_cd = cd.max()
        w = np.exp(self.beta * (cd - max_cd))
        ws = w.sum()
        if ws < 1e-30:
            w = np.ones_like(c) / len(c)
        else:
            w /= ws
        grad = np.zeros(self.n)
        for k in range(2 * self.n - 1):
            if w[k] < 1e-13:
                continue
            grad += w[k] * corr_grad_k(x, k, self.n)
        return grad * self.dx


class SumConstraintObj:
    def __init__(self, target_sum):
        self.target_sum = target_sum

    def fun(self, x):
        return x.sum() - self.target_sum

    def jac(self, x):
        return np.ones(len(x))


class AvgTopKObjective:
    def __init__(self, n, lags):
        self.n = n
        self.lags = lags
        self.dx = 2.0 / n

    def fun(self, x):
        c = compute_corrs(x)
        return float(c[self.lags].mean() * self.dx)

    def jac(self, x):
        grad = np.zeros(self.n)
        for k in self.lags:
            grad += corr_grad_k(x, k, self.n)
        return (grad / len(self.lags)) * self.dx


def slsqp_lse(h0, n, target_sum, beta, maxiter=500):
    dx = 2.0 / n
    obj = LSEObjective(n, beta)
    cons = SumConstraintObj(target_sum)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj.fun, h0, jac=obj.jac, method='SLSQP',
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': cons.fun, 'jac': cons.jac}],
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        h_out = project_feasible(res.x, target_sum)
    except Exception:
        h_out = project_feasible(h0, target_sum)
    return h_out, c5_of_h(h_out)
