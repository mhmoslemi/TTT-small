        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 5: Diverse structured starts
    if rem() > 60:
        n_best = len(best_h)
        for trial in range(8):
            if rem() < 25:
                break
            n_div = n_best + rng.choice([-4, -2, 0, 2, 4])
            n_div = max(40, min(200, n_div))
            r_div = np.random.RandomState(seed + 7919 * trial)
            h_div = make_structured_init(n_div, r_div)
            # Add sinusoidal structure
            x = np.linspace(0, 2, n_div)
            h_div = 0.5 + 0.25 * np.sin(2 * np.pi * r_div.uniform(1, 4) * x + r_div.uniform(0, 6.28))
            h_div = project_feasible(h_div, n_div / 2.0)
            t5 = min(35, rem() * 0.3)
            h_opt, c5_opt = pairwise_optimize(h_div, t5, rng, n_cand=5)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()

    # Phase 6: Final pairwise refinement with more candidates
    if rem() > 20:
        t6 = min(60, rem() * 0.6)
        h_opt, c5_opt = pairwise_optimize(best_h, t6, rng, n_cand=9)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 7: Very fine pairwise on best
    if rem() > 15:
        t7 = min(30, rem() * 0.5)
        h_opt, c5_opt = pairwise_optimize(best_h, t7, rng, n_cand=11)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    best_h = np.asarray(best_h, dtype=float)
    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_c5 = eval_c5(best_h)
    return best_h, best_c5, len(best_h)
