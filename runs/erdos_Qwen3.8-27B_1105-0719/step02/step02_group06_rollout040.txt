def project_feasible(x, target_sum):
    """Project x to [0,1]^n with sum = target_sum."""
    x = np.clip(x, 0.0, 1.0)
    # Bisection on shift
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(x + mid, 0.0, 1.0).sum()
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    h = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)
    # Fix residual
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        interior = (h > 1e-10) & (h < 1.0 - 1e-10)
        if interior.sum() > 0:
            h[interior] += diff / interior.sum()
    return np.clip(h, 0.0, 1.0)

def transfer_to_n(h_old, n_new):
    """Interpolate h to new resolution."""
    x_old = (np.arange(len(h_old)) + 0.5) * 2.0 / len(h_old)
    x_new = (np.arange(n_new) + 0.5) * 2.0 / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new / 2.0)
