def coord_step_sym(h, j, n, dx, paired):
    """One coordinate descent step for symmetric space.
    h is the full vector, j is the free index (0..m-1), paired is the paired index."""
    h0 = h.copy()
    h0[j] = 0.0
    h0[paired] = 1.0
    h1 = h.copy()
    h1[j] = 1.0
    h1[paired] = 0.0
    corr0 = np.correlate(h0, 1.0 - h0, mode='full') * dx
    corr1 = np.correlate(h1, 1.0 - h1, mode='full') * dx
    a = corr1 - corr0
    b = corr0
    return min_max_lines(a, b)

def coord_step_full(h, i, n, dx):
    """One coordinate descent step for full space."""
    h0 = h.copy()
    h0[i] = 0.0
    h1 = h.copy()
    h1[i] = 1.0
    corr0 = np.correlate(h0, 1.0 - h0, mode='full') * dx
    corr1 = np.correlate(h1, 1.0 - h1, mode='full') * dx
    a = corr1 - corr0
    b = corr0
    return min_max_lines(a, b)

def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def cd_sym(q, n, symmetry, max_sweeps=100, tol=1e-14, rng=None, gauss_southwell=False):
    """Coordinate descent in symmetric space."""
    m = n // 2
    dx = 2.0 / n
    q = q.copy().clip(0, 1)
    
    for sweep in range(max_sweeps):
        h = make_sym_h(q, n) if symmetry == 0 else make_sym_h2(q, n)
        obj = exact_score(h)
        
        if gauss_southwell and rng is not None:
            order = np.argsort(rng.rand(m))  # random order
        else:
            order = np.arange(m)
        
        max_decrease = 0.0
        for j in order:
            if symmetry == 0:
                paired = n - 1 - j
            else:
                paired = m + j
            new_q = coord_step_sym(h, j, n, dx, paired)
            if abs(new_q - q[j]) > 1e-14:
                q[j] = new_q
                max_decrease = max(max_decrease, abs(new_q - q[j]))
        
        h_new = make_sym_h(q, n) if symmetry == 0 else make_sym_h2(q, n)
        obj_new = exact_score(h_new)
        if obj - obj_new < tol:
            break
    
    h = make_sym_h(q, n) if symmetry == 0 else make_sym_h2(q, n)
    return q, exact_score(h)

def cd_full(h, n, max_sweeps=100, tol=1e-14, rng=None, target_sum=None):
    """Coordinate descent in full space with projection."""
    dx = 2.0 / n
    h = h.copy().clip(0, 1)
    target_sum = target_sum if target_sum is not None else n / 2.0
    
    for sweep in range(max_sweeps):
        obj = exact_score(h)
        order = np.arange(n) if rng is None else rng.permutation(n)
        
        for i in order:
            new_val = coord_step_full(h, i, n, dx)
            if abs(new_val - h[i]) > 1e-14:
                h[i] = new_val

        h = project_feasible(h, target_sum)
        h = h.clip(0, 1)
        obj_new = exact_score(h)
        if obj - obj_new < tol:
            break
    
    return h, exact_score(h)
