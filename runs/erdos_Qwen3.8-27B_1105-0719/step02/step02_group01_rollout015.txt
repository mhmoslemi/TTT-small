import numpy as np
from scipy.optimize import linprog
from scipy.special import logsumexp
import time

def project_feasible(x, target_sum):
    """Project x onto {h: 0<=h<=1, sum(h)=target_sum}"""
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def c5_of_h(h, dx):
    """Compute C5 = dx * max(correlate(h, 1-h, 'full'))"""
    return np.correlate(h, 1.0 - h, mode='full').max() * dx

def grad_corr_k(h, k, n):
    """Gradient of correlation k with respect to h.
    k is the index in the 'full' correlation array (0 to 2n-2).
    off = k - (n-1) is the lag.
    """
    off = k - (n - 1)
    grad = np.zeros(n)
    # h[i+off] appears with coefficient (1-h[i])
    # (1-h[i]) appears with coefficient -h[i+off]
    i_lo = max(0, -off)
    i_hi = min(n - 1, n - 1 - off)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    # d/dh[i+off] of h[i+off]*(1-h[i]) = (1-h[i])
    grad[idx + off] += (1.0 - h[idx])
    # d/dh[i] of h[i+off]*(1-h[i]) = -h[i+off]
    grad[idx] += (-h[idx + off])
    return grad

def all_grads(h, n, active_corrs):
    """Compute gradients for active correlations. Returns shape (len(active), n)."""
    grads = np.zeros((len(active_corrs), n))
    for j, k in enumerate(active_corrs):
        grads[j] = grad_corr_k(h, k, n)
    return grads

    h_interp = np.interp(x_tgt, x_src, h_src)
    return h_interp

        else:

def c5_of_h(h, dx):
    return np.correlate(h, 1.0 - h, mode='full').max() * dx

def grad_corr_k(h, k, n):
    off = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -off)
    i_hi = min(n - 1, n - 1 - off)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx + off] += (1.0 - h[idx])
    grad[idx] += (-h[idx + off])
    return grad

def build_and_solve_lp(h, active_idx, corrs, dx, alpha, n):
    K = len(active_idx)
    m_vars = n + 1
    grads = np.zeros((K, n))
    for j, k in enumerate(active_idx):
        grads[j] = grad_corr_k(h, k, n)
    c_obj = np.zeros(m_vars)
    c_obj[-1] = 1.0
    A_ub = np.zeros((K, m_vars))
    for j in range(K):
        A_ub[j, :n] = dx * grads[j]
        A_ub[j, -1] = -1.0
    b_ub = -dx * corrs[active_idx]
    A_eq = np.zeros((1, m_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(-h[i] * alpha, (1.0 - h[i]) * alpha) for i in range(n)]
    bounds.append((0.0, None))
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,

        n_fb = len(best_h)
        dx_fb = 2.0 / n_fb
        ts_fb = n_fb / 2.0
        h_final, c5_final = lp_inner_loop(best_h, dx_fb, ts_fb, 0.02, 150, budget_s, t0)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final
    
    # Final validation
    n_final = len(best_h)
    dx_final = 2.0 / n_final
    ts_final = n_final / 2.0
    best_h = project_feasible(best_h, ts_final)
    best_c5 = c5_of_h(best_h, dx_final)
    
    return best_h, best_c5, n_final
