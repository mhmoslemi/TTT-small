import numpy as np
from scipy.optimize import minimize, linprog
import time

def build_sym_h(q):
    m = len(q)
    h = np.empty(2 * m)
    h[:m] = q
    h[m:] = 1.0 - q[::-1]
    return h

def eval_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def corr_raw(h):
    return np.correlate(h, 1.0 - h, mode='full')

def sym_from_h(h):
    n = len(h)
    m = n // 2
    q = np.empty(m)
    for i in range(m):
        q[i] = 0.5 * (h[i] + (1.0 - h[n - 1 - i]))
    return np.clip(q, 0.0, 1.0)

def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(x + mid, 0.0, 1.0).sum()
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    h = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        interior = (h > 1e-10) & (h < 1.0 - 1e-10)
        if interior.sum() > 0:
            h[interior] += diff / interior.sum()
    return np.clip(h, 0.0, 1.0)

def dh_dq(r, h, n, d):
    m = n // 2
    L = abs(d)
    i = r
    j = n - 1 - r
    if L == 0:
        gi = 1.0 - 2.0 * h[i]
        gj = 1.0 - 2.0 * h[j]
        return gi - gj
    if d > 0:
        gi = 0.0
        if i >= L:
            gi += 1.0 - h[i - L]
        if i + L < n:
            gi += -h[i + L]
        gj = 0.0
        if j >= L:
            gj += 1.0 - h[j - L]
        if j + L < n:
            gj += -h[j + L]
    else:
        gi = 0.0
        if i + L < n:
            gi += 1.0 - h[i + L]
        if i >= L:
            gi += -h[i - L]
        gj = 0.0
        if j + L < n:
            gj += 1.0 - h[j + L]
        if j >= L:
            gj += -h[j - L]
    return gi - gj

def compute_all_grads_q(q, n_full):
    m = n_full // 2
    h = build_sym_h(q)
    n_corr = 2 * n_full - 1
    grads = np.zeros((n_corr, m))
    for k in range(n_corr):
        d = k - (n_full - 1)
        for r in range(m):
            grads[k, r] = dh_dq(r, h, n_full, d)
    return grads, h

def select_active(corrs, margin=0.005, pad=3, max_k=60):
    n = len(corrs)
    max_val = corrs.max()
    threshold = max_val - margin * max(1.0, abs(max_val))
    active = set(np.where(corrs >= threshold)[0].tolist())
    padded = set()
    for k in active:
        for dd in range(-pad, pad + 1):
            if 0 <= k + dd < n:
                padded.add(k + dd)
    active_list = sorted(padded)
    if len(active_list) > max_k:
        scored = sorted([(corrs[k], k) for k in active_list], reverse=True)
        active_list = sorted([k for _, k in scored[:max_k]])
    return active_list

def active_set_lp(q, n_full, active_lags, radius):
    m = n_full // 2
    h = build_sym_h(q)
    dx = 2.0 / n_full
    corrs = corr_raw(h) * dx
    n_corr = 2 * n_full - 1

    A_rows = []
    b_vals = []
    for k in active_lags:
        d = k - (n_full - 1)
        grad_k = np.zeros(m)
        for r in range(m):
            grad_k[r] = dh_dq(r, h, n_full, d)
        rhs = np.dot(grad_k, q) - corrs[k]
        row = np.zeros(m + 1)
        row[:m] = grad_k
        row[m] = -1.0
        A_rows.append(row)
        b_vals.append(rhs)

    A_ub = np.array(A_rows)
    b_ub = np.array(b_vals)
    bounds = []
    for r in range(m):
        lo = max(0.0, q[r] - radius)
        hi = min(1.0, q[r] + radius)
        bounds.append((lo, hi))
    bounds.append((-1.0, 1.0))

    c_obj = np.zeros(m + 1)
    c_obj[m] = 1.0

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.clip(res.x[:m], 0.0, 1.0), res.x[m]
    except Exception:
        pass
    return None, None

def coordinate_sweep_pass(q, n_full, order=None):
    m = n_full // 2
    h = build_sym_h(q)
    dx = 2.0 / n_full
    n_corr = 2 * n_full - 1
    c_base = corr_raw(h) * dx
    current_best = c_base.max()

    if order is None:
        order = list(range(m))

    for r in order:
        h = build_sym_h(q)
        c_base = corr_raw(h) * dx
        g = np.zeros(n_corr)
        for k in range(n_corr):
            d = k - (n_full - 1)
            g[k] = dh_dq(r, h, n_full, d)
        a = c_base - g * q[r]
        b = g

        lo, hi = 0.0, 1.0
        for _ in range(50):
            t_mid = (lo + hi) / 2.0
            vals = a + b * t_mid
            max_val = vals.max()
            active_mask = vals >= max_val - 1e-14
            min_slope = b[active_mask].min()
            max_slope = b[active_mask].max()
            if max_slope < 1e-13:
                hi = t_mid
            elif min_slope > 1e-13:
                lo = t_mid
            else:
                break

        candidates = np.array([0.0, 1.0, lo, hi, q[r]])
        best_t = q[r]
        best_f = float(np.max(a + b * q[r]))
        for t_c in candidates:
            t_c = float(np.clip(t_c, 0.0, 1.0))
            f_c = float(np.max(a + b * t_c))
            if f_c < best_f:
                best_f = f_c
                best_t = t_c
        q[r] = best_t

    q = np.clip(q, 0.0, 1.0)
    return q

def smooth_p_obj(q, n_full, p):
    m = n_full // 2
    h = build_sym_h(q)
    dx = 2.0 / n_full
    corrs = corr_raw(h) * dx
    return float(np.sum(corrs**p)**(1.0 / p))

def smooth_p_obj_grad(q, n_full, p):
    m = n_full // 2
    h = build_sym_h(q)
    dx = 2.0 / n_full
    corrs = corr_raw(h) * dx
    obj = float(np.sum(corrs**p)**(1.0 / p))
    grads = np.zeros((2 * n_full - 1, m))
    for k in range(len(corrs)):
        d = k - (n_full - 1)
        for r in range(m):
            grads[k, r] = dh_dq(r, h, n_full, d)
    weight = corrs**(p - 1) * dx
    grad_q = np.zeros(m)
    for r in range(m):
        grad_q[r] = np.dot(weight, grads[:, r])
    grad_q *= (1.0 / (len(corrs) ** (1.0 / p))) * (len(corrs) ** (1.0 / p))
    norm_factor = obj**(1 - p) * (1.0 / p) * p * (1.0 / len(corrs)**(1.0/p))
    # Simplified: grad = (1/p) * (sum corrs^p)^(1/p - 1) * p * corrs^(p-1) * grad
    # = (sum corrs^p)^(1/p-1) * corrs^(p-1) * grad_k
    scale = obj**(1.0 - p)
    grad_q = np.zeros(m)
    for r in range(m):
        grad_q[r] = scale * np.dot(corrs**(p-1), grads[:, r])
    grad_q *= dx
    return obj, grad_q

def smooth_p_optimize(q0, n_full, p_list, max_per, time_limit):
    m = n_full // 2
    q = np.clip(q0.copy(), 0, 1)
    best_score = eval_score(build_sym_h(q))
    best_q = q.copy()
    bounds = [(0.0, 1.0)] * m

    for p in p_list:
        if time.time() > time_limit:
            break
        try:
            res = minimize(
                lambda xx, pp=p: smooth_p_obj(xx, n_full, pp),
                q,
                jac=lambda xx, pp=p: smooth_p_obj_grad(xx, n_full, pp)[1],
                method='L-BFGS-B', bounds=bounds,
                options={'maxiter': max_per, 'ftol': 1e-14, 'gtol': 1e-9}
            )
            q_new = np.clip(res.x, 0, 1)
        except Exception:
            q_new = q.copy()
        h_new = build_sym_h(q_new)
        sc = eval_score(h_new)
        if sc < best_score:
            best_score = sc
            best_q = q_new.copy()
        q = q_new
    return best_q, best_score

def transfer_to_n(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0, 2.0, n_old)
    x_new = np.linspace(0, 2.0, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new / 2.0)

def main_loop(q, n_full, t_start, budget, active_margin=0.005):

    if abs(best_h.sum() - target_sum) > 1e-9:
        best_h = project_feasible(best_h, target_sum)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = eval_score(best_h)

    return best_h, best_c5, n_out
