```python
from scipy.optimize import minimize

def solve_step_qp(h_current, t_target, active_lags, n):
    """Solve QP: min ||delta||^2 s.t. linearized constraints"""
    # Variables: delta (n)
    # Objective: sum(delta^2)
    # Constraints:
    #   For each active lag d: C_d(h) + grad_d . delta <= t_target
    #   0 <= h[i] + delta[i] <= 1
    #   sum(delta) = 0
    
    c_vals = corr_full(h_current) * (2.0/n)
    
    def objective(delta):
        return np.dot(delta, delta)
    
    def objective_grad(delta):
        return 2.0 * delta
    
    constraints = []
    for d_idx in active_lags:
        d = d_idx - (n-1)
        c_val = c_vals[d_idx]
        g = corr_grad(h_current, d, n) * (2.0/n)
        # c_val + g . delta <= t_target
        constraints.append({
            'type': 'ineq',
            'fun': lambda delta, g=g, c_val=c_val, t=t_target: t_target - c_val - np.dot(g, delta),
            'jac': lambda delta, g=g: -g
        })
    
    bounds = [(-h_current[i], 1.0 - h_current[i]) for i in range(n)]
    
    res = minimize(objective, np.zeros(n), jac=objective_grad,
                   method='SLSQP', bounds=bounds,
                   constraints=[{'type': 'eq', 'fun': lambda delta: np.sum(delta), 'jac': lambda delta: np.ones(n)}] + constraints,
                   options={'maxiter': 100, 'ftol': 1e-12})
    return res.x
```

```python
def lp_step(h, active_indices, n, mu=1.0):
    """Solve LP to find a descent direction."""
    c_vals = corr_full(h) * (2.0/n)
    
    delta = cp.Variable(n)
    t = cp.Variable()
    
    constraints = [sum(delta) == 0]
    constraints += [h + delta >= 0, h + delta <= 1]
```

```python
def project_to_target(x, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(x - mid, 0.0, 1.0).sum() > target_sum:
            lo = mid
        else:
            hi = mid
```

```python
def eval_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_to_target(x, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(x - mid, 0.0, 1.0).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.any():
            h[mask] += diff / mask.sum()

def interp_to_sym(h, n_new):
    """Interpolate h to n_new points and enforce symmetry."""
    n_old = len(h)
    if n_new == n_old:
        h_new = h.copy()
    else:
        x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
        x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)

def logsumexp_obj(h, n, beta):
    """Log-sum-exp of correlation values (smooth approximation to max)."""
    c_vals = corr_full(h) * (2.0/n)
    c_max = c_vals.max()
    return c_max + np.log(np.sum(np.exp(beta * (c_vals - c_max)))) / beta

def logsumexp_model(n, beta, mu_sum, target):
    def fg(z):
        h = z[:n]
        c_vals = corr_full(h) * (2.0/n)
        c_max = c_vals.max()
        f = c_max + np.log(np.sum(np.exp(beta * (c_vals - c_max)))) / beta
        sdev = h.sum() - target
        f += 0.5 * mu_sum * sdev * sdev
        # Gradient
        exp_vals = np.exp(beta * (c_vals - c_max))
        weights = exp_vals / exp_vals.sum()
        g = np.zeros(n)
        dx = 2.0/n
        for idx in range(len(c_vals)):
            if weights[idx] > 1e-15:
                d = idx - (n-1)
                g += weights[idx] * corr_grad_h(h, d, n) * dx
        g *= beta
        g += mu_sum * sdev
        return f, g
    return fg

    h = h_init.copy()

    betas = np.linspace(np.log(beta_start), np.log(beta_end), n_beta)
    
    for log_beta in betas:

        model = logsumexp_model(n, beta, 500.0, target)
        bounds = [(0.0, 1.0)] * n

                          bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-12})

            pass
```

```python
import numpy as np
from scipy.optimize import minimize

def eval_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))


def corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_to_target(x, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(x - mid, 0.0, 1.0).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.any():
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)


def corr_grad_h(h, d, n):
    g = np.zeros(n)
    if d >= 0:
        if n - d > 0:
            g[:n-d] += 1.0 - h[d:]
        if d > 0:
            g[d:] += -h[:n-d]
    else:
        dd = -d
        if n - dd > 0:
            g[dd:] += 1.0 - h[:n-dd]
        if dd > 0:
            g[:n-dd] += -h[dd:]
    return g


def build_sym(q, n):
    m = n // 2
    h = np.empty(n)
    h[:m] = q
    h[m:] = 1.0 - q[::-1]
    return h


def to_sym_q(h, n):
    m = n // 2
    return 0.5 * (h[:m] + 1.0 - h[n-1:m-1:-1])


def sym_grad(h, d, n):
    m = n // 2
    gh = corr_grad_h(h, d, n)
    return gh[:m] - gh[n-1:m-1:-1]


def select_active(c_vals, n, margin=0.01, pad=2):
    mx = c_vals.max()
    thresh = mx - max(1e-10, margin * mx)
    idx = np.where(c_vals >= thresh)[0]
    expanded = set(idx.tolist())
    for a in idx:
        for delta in range(-pad, pad + 1):
            k = int(a) + delta
            if 0 <= k <= 2 * n - 2:
                expanded.add(k)
    return np.sort(np.array(list(expanded), dtype=int))


def lp_step_sym(q, n, active_indices, step_limit=0.05):
    m = n // 2
    h = build_sym(q, n)
    c_vals = corr_full(h) * (2.0 / n)
    dx = 2.0 / n

    delta_q = cp.Variable(m)
    t = cp.Variable()

    constraints = [
        delta_q >= -step_limit,
        delta_q <= step_limit,
        q + delta_q >= 0,
        q + delta_q <= 1
    ]

    for idx in active_indices:
        d = int(idx) - (n - 1)
        gq = sym_grad(h, d, n) * dx
        c_val = float(c_vals[int(idx)])
        constraints.append(gq @ delta_q - t <= -c_val)

    obj = cp.Minimize(t)
    prob = cp.Problem(obj, constraints)

    for solver in [cp.ECOS, cp.SCS, cp.CLARABEL]:
        try:
            prob.solve(solver=solver, max_iters=10000)
            if prob.status == 'optimal' and delta_q.value is not None:
                return np.array(delta_q.value).flatten()
        except Exception:
            continue
    return None

def interp_to_sym(h, n_new):
    n_old = len(h)
    if n_new == n_old:
        h_new = h.copy()
    else:
        x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)

        h_new = np.interp(x_new, x_old, h)
    if n_new % 2 == 0:
        q = to_sym_q(h_new, n_new)
        h_new = build_sym(q, n_new)
        return q
    return None

                best_c5 = c5_fin3

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_target(best_h, len(best_h) / 2.0)
    best_c5 = eval_c5(best_h)

    return best_h, best_c5, len(best_h)
```
