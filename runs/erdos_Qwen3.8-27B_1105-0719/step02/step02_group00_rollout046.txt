            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h


def build_grad_fast(h, n, dx):
    M = 2 * n - 1
    G = np.zeros((M, n))
    center = n - 1
    G[center, :] = dx * (1.0 - 2.0 * h)
    for d in range(1, n):
        m_pos = center + d
        idx1 = np.arange(n - d)
        G[m_pos, idx1] += -dx * h[idx1 + d]
        G[m_pos, idx1 + d] += dx * (1.0 - h[idx1])
        m_neg = center - d
        G[m_neg, idx1] += dx * (1.0 - h[idx1 + d])
        G[m_neg, idx1 + d] += -dx * h[idx1]
    return G


def lp_solve(h_curr, profile, active, dx, trust, rho):
    if not HAS_CVXPY:
        return None
    n = len(h_curr)
    target = n / 2.0
    G = build_grad_fast(h_curr, n, dx)
    G_a = G[active, :]
    b_a = profile[active] - G_a @ h_curr
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    cons = [
        h_var >= 0,
        h_var <= 1,
        cp.sum(h_var) == target,
        h_var >= h_curr - trust,
        h_var <= h_curr + trust,
        G_a @ h_var - t_var <= -b_a
    ]
    obj = cp.Minimize(t_var + rho * cp.norm1(h_var - h_curr))
    prob = cp.Problem(obj, cons)
    for solver in [cp.ECOS, cp.CBC, cp.SCS]:
        try:
            prob.solve(solver=solver, max_iters=10000, abstol=1e-10)
            break
        except Exception:
            continue
    if h_var.value is None:
        return None
    h_sol = np.array(h_var.value).flatten()
    if not np.isfinite(h_sol).all():
        return None
    return h_sol


def grad_step(h, profile, active, n, dx, scales):
    G = build_grad_fast(h, n, dx)
    G_a = G[active, :]
    w = profile[active] - profile[active].min() + 1e-6
    w = w / w.sum()
    grad = G_a.T @ w
    grad -= grad.mean()
    gn = np.linalg.norm(grad)
    if gn < 1e-15:
        return None, None
    grad /= gn
    curr_c = np.max(profile)
    for s in scales:
        h_try = repair_feasible(h - s * grad)
        sc = exact_score(h_try)
        if sc < curr_c - 1e-14:
            return h_try, sc
    return None, None


def interp(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    fx = np.concatenate([[0.0], xs, [2.0]])
    fh = np.concatenate([[h[0]], h, [h[-1]]])
    hn = np.interp(xt, fx, fh)
    return repair_feasible(np.clip(hn, 0, 1))


def active_set_loop(h0, max_iters, t_check, budget_fn):
    """Core LP + gradient active set refinement loop."""
    n = len(h0)
    dx = 2.0 / n
    h = h0.copy()
    best_c = exact_score(h)
    trust = 0.02
    stagnation = 0

    for _ in range(max_iters):
        if budget_fn() < 25:
            break
        profile = compute_profile(h)
        c_max = np.max(profile)
        if c_max < best_c - 1e-14:
            best_c = c_max
            stagnation = 0
        else:
            stagnation += 1

        spread = c_max - np.median(profile)
        beam = max(0.002, min(0.015, 0.15 * spread))
        active = np.where(profile >= c_max - beam)[0]
        if len(active) == 0:
            active = np.array([np.argmax(profile)])
        if len(active) > 18:
            active = active[np.argsort(profile[active])[-18:]]

        accepted = False
        h_lp = lp_solve(h, profile, active, dx, trust, 0.005)
        if h_lp is not None:
            h_lp = repair_feasible(h_lp)
            c_lp = exact_score(h_lp)
            if c_lp < best_c - 1e-14:
                h = h_lp
                best_c = c_lp
                trust = min(trust * 1.25, 0.07)
                accepted = True
            else:
                for alpha in [0.6, 0.4, 0.25]:
                    hb = repair_feasible(alpha * h_lp + (1 - alpha) * h)
                    cb = exact_score(hb)
                    if cb < best_c - 1e-14:
                        h = hb
                        best_c = cb
                        trust = min(trust * 1.1, 0.07)
                        accepted = True
                        break
                if not accepted:
                    trust *= 0.55

        if not accepted:
            hg, cg = grad_step(h, profile, active, n, dx,
                               [trust, trust * 0.5, trust * 0.25, trust * 0.1])
            if hg is not None:
                h = hg
                best_c = cg
                accepted = True
            else:
                # Try wider active set
                for b2 in [beam * 2, beam * 4, beam * 8]:
                    a2 = np.where(profile >= c_max - b2)[0]
                    if len(a2) > 12:
                        a2 = a2[np.argsort(profile[a2])[-12:]]
                    hg, cg = grad_step(h, profile, a2, n, dx,
                                       [0.005, 0.002, 0.001, 0.0005, 0.0002])
                    if hg is not None:
                        h = hg
                        best_c = cg
                        accepted = True
                        break

        if not accepted:
            trust *= 0.4
            if trust < 0.0008 or stagnation > 40:
                break

    return h, best_c


def slsqp_refine(h0, maxiter, tol):
    n = len(h0)
    dx = 2.0 / n
    M = 2 * n - 1

    def obj(x):
        return x[-1]

    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g

    def eq_fun(x):
        return x[:-1].sum() - n / 2.0

    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g

    def ineq_fun(x):
        h = x[:-1]
        return x[-1] - np.correlate(h, 1 - h, mode="full") * dx

    def ineq_jac(x):
        h = x[:-1]
        G = build_grad_fast(h, n, dx)
        J = np.zeros((M, n + 1))
        J[:, :-1] = -G
        J[:, -1] = 1.0
        return J

    t0 = exact_score(h0) + 1e-4
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, t0 + 2.0)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': tol})
        if res.x is not None and np.isfinite(res.x).all():
            hs = repair_feasible(res.x[:-1])
            if np.isfinite(exact_score(hs)):
                return hs
    except Exception:
        pass
    return None


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()
    rng = np.random.RandomState(seed)

    def tl():
        return budget_s + t_start - time.perf_counter()

    # Load parent
    parent = None
    try:
        parent = np.array(initial_h_values, dtype=np.float64)
    except (NameError, Exception):
        pass
    if parent is None or len(parent) < 4:
        parent = np.full(70, 0.5)

    parent = repair_feasible(np.clip(parent, 0.0, 1.0))
    best_h = parent.copy()
    best_c = exact_score(parent)

    # Symmetrize
    h_sym = repair_feasible((parent + parent[::-1]) / 2.0)
    c_sym = exact_score(h_sym)
    if c_sym < best_c - 1e-12:
        best_h, best_c = h_sym, c_sym

    # Phase 1: Main LP active-set loop on parent dimension
    if tl() > 120:
        bh, bc = active_set_loop(best_h, max_iters=300, t_check=1, budget_fn=tl)
        if bc < best_c - 1e-14:
            best_h, best_c = bh, bc

    # Phase 2: Dimension exploration
    if tl() > 150:
        n0 = len(best_h)
        dims = [d for d in range(max(24, n0 - 16), min(150, n0 + 17), 2)
                if d != n0 and d >= 24]
        dims.sort(key=lambda d: abs(d - n0))

        for nd in dims[:10]:
            if tl() < 50:
                break
            hi = interp(best_h, nd)
            ci = exact_score(hi)
            if ci < best_c + 0.02:
                bh, bc = active_set_loop(hi, max_iters=150, t_check=1, budget_fn=tl)
                if bc < best_c - 1e-14:
                    best_h, best_c = bh, bc

    # Phase 3: SLSQP fine-tune on best
    if tl() > 80:
        hs = slsqp_refine(best_h, maxiter=400, tol=1e-12)
        if hs is not None:
            cs = exact_score(hs)
            if cs < best_c - 1e-14:
                best_h, best_c = hs, cs

    # Phase 4: Another LP round after SLSQP
    if tl() > 60:
        bh, bc = active_set_loop(best_h, max_iters=200, t_check=1, budget_fn=tl)
        if bc < best_c - 1e-14:
            best_h, best_c = bh, bc

    # Phase 5: Try shifted/rotated variants
    if tl() > 80:
        n0 = len(best_h)
        for shift in [1, 2, 3, -1, -2]:
            if tl() < 30:
                break
            hs = repair_feasible(np.roll(best_h, shift))
            cs = exact_score(hs)
            if cs < best_c + 0.005:
                bh, bc = active_set_loop(hs, max_iters=80, t_check=1, budget_fn=tl)
                if bc < best_c - 1e-14:
                    best_h, best_c = bh, bc

    # Phase 6: Random perturbations + LP refine
    if tl() > 100:
        for eps in [0.01, 0.02, 0.03, 0.05, 0.08]:
            if tl() < 30:
                break
            z = rng.randn(len(best_h))
            z -= z.mean()
            z /= (np.linalg.norm(z) / len(best_h) + 1e-15)
            hp = repair_feasible(best_h + eps * z)
            cp_ = exact_score(hp)
            if cp_ < best_c + 0.01:
                bh, bc = active_set_loop(hp, max_iters=100, t_check=1, budget_fn=tl)
                if bc < best_c - 1e-14:
                    best_h, best_c = bh, bc

    # Phase 7: Final SLSQP polish
    if tl() > 50:
        hs = slsqp_refine(best_h, maxiter=500, tol=1e-13)
        if hs is not None:
            cs = exact_score(hs)
            if cs < best_c - 1e-14:
                best_h, best_c = hs, cs

    # Phase 8: Final gradient micro-steps
    if tl() > 20:
        n_f = len(best_h)
        dx_f = 2.0 / n_f
        h_f = best_h.copy()
        c_f = best_c
        for _ in range(100):
            if tl() < 8:
                break
            prof = compute_profile(h_f)
            cm = np.max(prof)
            if cm < c_f - 1e-15:
                c_f = cm
            active = np.where(prof >= cm - 0.003)[0]
            if len(active) > 10:
                active = active[np.argsort(prof[active])[-10:]]
            hg, cg = grad_step(h_f, prof, active, n_f, dx_f,
                               [0.001, 0.0005, 0.0002, 0.0001, 5e-5])
            if hg is not None:
                h_f, c_f = hg, cg
            else:
                break
        if c_f < best_c - 1e-14:
            best_h, best_c = h_f, c_f

    # Final validation
    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)

    try:
        er = evaluate_erdos_solution(best_h)
        if isinstance(er, (float, int)):
            final_c = min(final_c, float(er))
        elif isinstance(er, tuple) and len(er) >= 2:
            final_c = min(final_c, float(er[0]))
    except Exception:
        pass

    return best_h, final_c, len(best_h)
