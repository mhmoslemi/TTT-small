def project_feasible(x, target_sum):
    """Project x onto {h: 0<=h<=1, sum(h)=target_sum}"""
    n = len(x)
    lo, hi = -1.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full').max() * dx

def corr_vals(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def weighted_grad(h, dx, weights):
    """Compute weighted sum of gradients of corr_k for all k."""
    n = len(h)
    grad = np.zeros(n)
    n_lags = 2 * n - 1
    for k in range(n_lags):
        wk = weights[k]
        if wk < 1e-15:
            continue
        d = k - (n - 1)
        if d >= 0:
            # corr = sum_{i=0}^{n-1-d} h[i](1-h[i+d])
            # d/dh[j] = (1-h[j+d]) for j in [0, n-1-d] + (-h[j-d]) for j in [d, n-1]
            i_max = n - 1 - d
            for j in range(i_max + 1):
                grad[j] += wk * (1.0 - h[j + d])
            for j in range(d, n):
                grad[j] += wk * (-h[j - d])
        else:
            d2 = -d
            # corr = sum_{i=0}^{n-1-d2} h[i-d2](1-h[i]) = sum_{i=d2}^{n-1} h[i](1-h[i-d2])
            # d/dh[j] = (1-h[j-d2]) for j in [d2, n-1] + (-h[j+d2]) for j in [0, n-1-d2]
            for j in range(d2, n):
                grad[j] += wk * (1.0 - h[j - d2])
            for j in range(n - d2):
                grad[j] += wk * (-h[j + d2])
    return grad * dx

        return x[0]

        g = np.zeros(n + 1)
        g[0] = 1.0
        return g

        t = x[0]
        h = np.clip(x[1:], 0.0, 1.0)
        c = corr_vals(h)
        return t - c[active_lags]

        h = np.clip(x[1:], 0.0, 1.0)
        J = -np.array([lag_grad(h, k) for k in active_lags])  # shape (n_lags, n)

        Jfull[:, 0] = 1.0

        g = np.zeros(n + 1)
        g[1:] = 1.0
        return g

    x0[0] = t0
    x0[1:] = h0
    
    constraints = [
        {'type': 'ineq', 'fun': con_fun, 'jac': con_grad},
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_grad}
    ]
    bounds = [(0.0, 2.0)] + [(0.0, 1.0)] * n

        for i, k in enumerate(al):
            J[i, 1:] = -lag_grad(h, k)
        return J

    def eq_fun(x):
        return x[1:].sum() - target_sum

    def eq_grad(x):
        g = np.zeros(n + 1)
        g[1:] = 1.0
        return g

    x0 = np.zeros(n + 1)
    x0[0] = t_init
    x0[1:] = h0

    constraints = [
        {'type': 'ineq', 'fun': con_fun, 'jac': con_grad},
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_grad}
    ]
    bounds = [(0.0, 2.0)] + [(0.0, 1.0)] * n

    for attempt in range(3):
        try:
            res = minimize(obj, x0, jac=obj_grad, method='SLSQP',
                           bounds=bounds, constraints=constraints,
                           options={'maxiter': maxiter, 'ftol': 1e-14})
            if hasattr(res, 'x') and res.x is not None:
                h_new = project_feasible(res.x[1:], target_sum)
                c5 = exact_score(h_new)
                if c5 < t_init + 0.01:
                    return h_new, c5
                x0 = res.x.copy()
        except Exception:
            break
    return None, None


def local_search_pair(h, n, target_sum, tol=1e-12):
    """Try optimizing pairs of variables to reduce max correlation."""
    n_lags = 2 * n - 1
    c = corr_vals(h)
    max_c = c.max()
    improved = False
    for i in range(n):
        for j in range(i + 1, n):
            s = h[i] + h[j]
            t_lo = max(-s / 2.0, (0.0 - s) / 2.0, -(1.0 - s / 2.0))
            t_hi = min(s / 2.0, (1.0 - s / 2.0))
            if t_hi - t_lo < 1e-10:
                continue
            best_t = h[i] - s / 2.0
            best_val = max_c
            n_samples = 40
            for tt in np.linspace(t_lo, t_hi, n_samples):
                h[i] = s / 2.0 + tt
                h[j] = s / 2.0 - tt
                if h[i] < -1e-10 or h[i] > 1.0 + 1e-10:
                    continue
                if h[j] < -1e-10 or h[j] > 1.0 + 1e-10:
                    continue
                c_new = corr_vals(h)
                mv = c_new.max()
                if mv < best_val - tol:
                    best_val = mv
                    best_t = tt
            h[i] = s / 2.0 + best_t
            h[j] = s / 2.0 - best_t
            h = np.clip(h, 0.0, 1.0)
            if best_val < max_c - tol:
                max_c = best_val
                improved = True
    if not improved:
        h[i] = s / 2.0 + best_t
        h[j] = s / 2.0 - best_t
    return h, max_c, improved


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def elapsed():
        return time.time() - t0

    def remaining():
        return budget_s - elapsed()

    try:
        h = np.array(initial_h_values, dtype=float)
    except NameError:
        h = 0.5 * np.ones(126)

    n = len(h)
    if n % 2 != 0:
        h = h[:n - 1]
        n = len(h)
    target_sum = n / 2.0
    h = project_feasible(h, target_sum)
    c5_best = exact_score(h)
    h_best = h.copy()
    print("Initial: n={}, C5={:.7f}".format(n, c5_best))

    # Phase 1: Full-space smooth optimization (no symmetry constraint)
    h_cur = h_best.copy()
    for beta in [80, 200, 500, 1200, 3000]:
        if remaining() < 30:
            break
        constraints = [{'type': 'eq', 'fun': lambda x: smooth_eq_fun(x, n),
                        'jac': lambda x: smooth_eq_grad(x, n)}]
        try:
            res = minimize(smooth_obj, h_cur, jac=lambda x: smooth_grad(x, n, beta),
                           method='SLSQP', bounds=[(0.0, 1.0)] * n,
                           constraints=constraints,
                           options={'maxiter': 300, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target_sum)
            c5_new = exact_score(h_new)
            if c5_new < c5_best - 1e-9:
                c5_best = c5_new
                h_best = h_new
                h_cur = h_new
                print("  smooth beta={}: C5={:.7f}".format(beta, c5_best))
            else:
                h_cur = h_best.copy()
        except Exception:
            pass

    # Phase 2: Epigraph with active-set expansion (full space)
    for rnd in range(12):
        if remaining() < 25:
            break
        h_cur = h_best.copy()
        c = corr_vals(h_cur)
        max_c = c.max()
        K = min(20 + 8 * rnd, 2 * n - 1)
        active = sorted(np.argsort(c)[::-1][:K].tolist())
        h_new, c5_new = epi_solve(h_cur, n, active, t_init=max_c * 1.0005,
                                   maxiter=400)
        if h_new is not None:
            c_check = corr_vals(h_new)
            actual_max = c_check.max()
            if actual_max < c5_best - 1e-9:
                c5_best = actual_max
                h_best = h_new
                print("  epi rnd={} K={}: C5={:.7f}".format(rnd, K, c5_best))
            elif remaining() < 80:
                break
        # If no improvement, try different active set
        if remaining() < 50:
            break
        # Add some random lags to active set
        c = corr_vals(h_best)
        top_idx = np.argsort(c)[::-1][:K]
        other_idx = np.setdiff1d(np.arange(2 * n - 1), top_idx)
        extra = rng.choice(other_idx, size=min(5, len(other_idx)), replace=False)
        active2 = sorted(np.concatenate([top_idx, extra]).astype(int).tolist())
        h_new2, c5_new2 = epi_solve(h_best, n, active2, t_init=c.max() * 1.0005,
                                     maxiter=300)
        if h_new2 is not None:
            c_check2 = corr_vals(h_new2)
            actual_max2 = c_check2.max()
            if actual_max2 < c5_best - 1e-9:
                c5_best = actual_max2
                h_best = h_new2
                print("  epi2 rnd={} K={}: C5={:.7f}".format(rnd, len(active2), c5_best))

    # Phase 3: Multi-start smooth optimization
    n_starts = 15
    for st in range(n_starts):
        if remaining() < 20:
            break
        scale = 0.02 + 0.08 * (st / max(n_starts - 1, 1))
        h_pert = h_best + scale * rng.randn(n)
        h_pert = project_feasible(h_pert, target_sum)
        for beta in [300, 1000]:
            if remaining() < 15:
                break
            constraints = [{'type': 'eq', 'fun': lambda x: smooth_eq_fun(x, n),
                            'jac': lambda x: smooth_eq_grad(x, n)}]
            try:
                res = minimize(smooth_obj, h_pert, jac=lambda x: smooth_grad(x, n, beta),
                               method='SLSQP', bounds=[(0.0, 1.0)] * n,
                               constraints=constraints,
                               options={'maxiter': 200, 'ftol': 1e-15})
                h_new = project_feasible(res.x, target_sum)
                c5_new = exact_score(h_new)
                if c5_new < c5_best - 1e-9:
                    c5_best = c5_new
                    h_best = h_new
                    print("  start {} beta={}: C5={:.7f}".format(st, beta, c5_best))
                    h_pert = h_new.copy()
                else:
                    h_pert = h_best.copy()
            except Exception:
                pass

    # Phase 4: Multi-resolution search
    if remaining() > 60:
        for delta in [2, -2, 4, -4, 6, -6, 8, -8]:
            if remaining() < 30:
                break
            n_try = n + delta
            if n_try < 60 or n_try % 2 != 0:
                continue
            x_old = np.linspace(0, 2, n, endpoint=False)
            x_new = np.linspace(0, 2, n_try, endpoint=False)
            h_interp = np.interp(x_new, x_old, h_best)
            h_interp = project_feasible(h_interp, n_try / 2.0)
            ts = n_try / 2.0
            for beta in [500, 2000]:
                if remaining() < 20:
                    break
                constraints = [{'type': 'eq', 'fun': lambda x: x.sum() - ts,
                                'jac': lambda x: np.ones(len(x))}]
                try:
                    res = minimize(smooth_obj, h_interp,
                                   jac=lambda x: smooth_grad(x, n_try, beta),
                                   method='SLSQP', bounds=[(0.0, 1.0)] * n_try,
                                   constraints=constraints,
                                   options={'maxiter': 200, 'ftol': 1e-15})
                    h_new = project_feasible(res.x, ts)
                    c5_new = exact_score(h_new)
                    if c5_new < c5_best - 1e-9:
                        c5_best = c5_new
                        h_best = h_new
                        n = n_try
                        target_sum = ts
                        print("  resol n={}: C5={:.7f}".format(n, c5_best))
                        break
                except Exception:
                    pass

    # Phase 5: Final epigraph polish with large active set
    if remaining() > 30:
        c = corr_vals(h_best)
        max_c = c.max()
        K = min(60, 2 * n - 1)
        active = sorted(np.argsort(c)[::-1][:K].tolist())
        h_new, c5_new = epi_solve(h_best, n, active, t_init=max_c * 1.0003,
                                   maxiter=600)
        if h_new is not None:
            actual = exact_score(h_new)
            if actual < c5_best - 1e-9:
                c5_best = actual
                h_best = h_new
                print("  final epi: C5={:.7f}".format(c5_best))

    # Phase 6: One more round of smooth + epi if time permits
    if remaining() > 40:
        for beta in [2000, 5000]:
            if remaining() < 20:
                break
            constraints = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum,
                            'jac': lambda x: np.ones(len(x))}]
            try:
                res = minimize(smooth_obj, h_best,
                               jac=lambda x: smooth_grad(x, n, beta),
                               method='SLSQP', bounds=[(0.0, 1.0)] * n,
                               constraints=constraints,
                               options={'maxiter': 300, 'ftol': 1e-16})
                h_new = project_feasible(res.x, target_sum)
                c5_new = exact_score(h_new)
                if c5_new < c5_best - 1e-9:
                    c5_best = c5_new
                    h_best = h_new
                    print("  polish beta={}: C5={:.7f}".format(beta, c5_best))
            except Exception:
                pass
        if remaining() > 25:
            c = corr_vals(h_best)
            active = sorted(np.argsort(c)[::-1][:40].tolist())
            h_new, c5_new = epi_solve(h_best, n, active, t_init=c.max() * 1.0002,
                                       maxiter=500)
            if h_new is not None:
                actual = exact_score(h_new)
                if actual < c5_best - 1e-9:
                    c5_best = actual
                    h_best = h_new
                    print("  final epi2: C5={:.7f}".format(c5_best))

    # Final validation
    n_final = len(h_best)
    h_best = project_feasible(h_best, n_final / 2.0)
    h_best = np.clip(h_best, 0.0, 1.0)
    c5_best = exact_score(h_best)
    print("Final: n={}, C5={:.7f}".format(n_final, c5_best))

    return h_best, c5_best, n_final
