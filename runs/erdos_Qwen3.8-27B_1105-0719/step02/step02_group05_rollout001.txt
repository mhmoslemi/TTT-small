import numpy as np
from scipy.optimize import minimize, minimize_scalar
from scipy.special import logsumexp
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corrs = np.correlate(h, 1.0 - h, mode='full')
    return corrs.max() * dx, corrs

def build_h_from_q(q, n):
    m = n // 2
    if n % 2 == 0:
        return np.concatenate([q, 1.0 - q[::-1]])
    else:
        return np.concatenate([q, [0.5], 1.0 - q[::-1]])

def q_from_h(h, n):
    m = n // 2
    return h[:m].copy()

def project_feasible(x, target_sum):
    """Project onto {x: 0<=x<=1, sum(x)=target_sum}"""
    n = len(x)
    lo, hi = -1.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    return h

def corr_gradient_d(h, d, n):
    """Gradient of correlation at lag d w.r.t. h"""
    g = np.zeros(n)
    if d >= 0:
        if d < n:
            g[d:] += (1.0 - h[:n - d])
            g[:n - d] += (-h[d:])
    else:
        dd = -d
        if dd < n:
            g[:n - dd] += (1.0 - h[dd:])
            g[dd:] += (-h[:n - dd])
    return g

def q_value_and_grad(q, n, beta=None):
    """Compute value and gradient in q-space. If beta is None, use exact max."""
    m = n // 2
    h = build_h_from_q(q, n)
    dx = 2.0 / n
    corrs = np.correlate(h, 1.0 - h, mode='full')
    
    if beta is None:
        val = corrs.max() * dx
        # Subgradient: gradient of the max lag
        max_idx = np.argmax(corrs)
        d = max_idx - (n - 1)
        G = corr_gradient_d(h, d, n)
        Gq = (G[:m] - G[m:][::-1]) * dx
        return val, Gq
    else:
        shifted = corrs - corrs.max()
        w = np.exp(beta * shifted)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(len(corrs)) / len(corrs)
        else:
            w /= w_sum
        val = logsumexp(beta * corrs) / beta * dx
        Gq = np.zeros(m)
        for k in range(len(corrs)):
            if w[k] < 1e-14:
                continue
            d = k - (n - 1)
            G = corr_gradient_d(h, d, n)
            Gq += w[k] * (G[:m] - G[m:][::-1])
        return val, Gq * dx

def de_q_space(q_init, n, rng, time_limit, pop_size=None):
    """Differential evolution in q-space with exact fitness"""
    m = n // 2
    if pop_size is None:
        pop_size = max(30, min(80, 2 * m))
    
    target_sum = n / 2.0
    
    def fitness(q):
        q = np.clip(q, 0, 1)
        h = build_h_from_q(q, n)
        s, _ = exact_score(h)
        return s
    
    # Initialize population
    pop = np.zeros((pop_size, m))

    for i in range(1, pop_size):
        # Mix of strategies

        if r < 0.3:

        # Sort population
        order = np.argsort(scores)
        pop = pop[order]
        scores = scores[order]
        best_idx = 0
        best_q = pop[0].copy()
        best_score = scores[0]

        if len(np.where(scores < best_score + 1e-8)[0]) > pop_size * 0.7:

            if time.time() - t_start > time_limit:
                break

            while cand == i or cand == best_idx:
                cand = rng.choice(pop_size)
            cand2 = rng.choice(pop_size)
            while cand2 == i or cand2 == cand or cand2 == best_idx:
                cand2 = rng.choice(pop_size)

                pop[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:

def coordinate_descent_q(q0, n, time_limit, max_passes=50):
    """Coordinate descent with golden-section search in q-space"""
    m = n // 2
    q = q0.copy()
    
    def fq(q_in):
        qi = np.clip(q_in, 0, 1)
        h = build_h_from_q(qi, n)
        s, _ = exact_score(h)
        return s
    
    current = fq(q)
    t_start = time.time()
    golden = (np.sqrt(5) - 1) / 2  # ~0.618
    
    for _pass in range(max_passes):
        if time.time() - t_start > time_limit:
            break
        improved = False
        for i in range(m):
            if time.time() - t_start > time_limit:
                break
            # Golden section search on coordinate i
            lo, hi = 0.0, 1.0
            qm1 = q.copy()
            qm2 = q.copy()
            c1 = lo + (1 - golden) * (hi - lo)
            c2 = lo + golden * (hi - lo)
            qm1[i] = c1
            qm2[i] = c2
            f1 = fq(qm1)
            f2 = fq(qm2)
            
            for _iter in range(40):
                if hi - lo < 1e-10:
                    break
                if f1 < f2:
                    hi = c2
                    c2 = c1
                    f2 = f1
                    c1 = lo + (1 - golden) * (hi - lo)
                    qm1[i] = c1
                    f1 = fq(qm1)
                else:
                    lo = c1
                    c1 = c2
                    f1 = f2
                    c2 = lo + golden * (hi - lo)
                    qm2[i] = c2
                    f2 = fq(qm2)
            
            opt_c = (lo + hi) / 2
            q[i] = opt_c
            new_f = fq(q)
            if new_f < current - 1e-12:
                current = new_f
                improved = True
        
        if not improved:
            break
    
    return q, current

def smooth_lbfgs_q(q0, n, beta, time_limit):
    """L-BFGS-B with smooth log-sum-exp objective in q-space"""
    m = n // 2
    
    def q_val(x):
        return q_value_and_grad(x, n, beta)[0]
    
    def q_grad(x):
        return q_value_and_grad(x, n, beta)[1]
    
    try:
        res = minimize(q_val, q0, jac=q_grad, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * m,
                       options={'maxiter': 200, 'ftol': 1e-15,
                                'gtol': 1e-10, 'maxfun': 5000})
        h_res = build_h_from_q(np.clip(res.x, 0, 1), n)
        s_res, _ = exact_score(h_res)
        return np.clip(res.x, 0, 1), s_res
    except Exception:
        return q0.copy(), fq_fallback(q0, n)

def fq_fallback(q, n):
    h = build_h_from_q(np.clip(q, 0, 1), n)
    s, _ = exact_score(h)
    return s

def epigraph_q_solve(q0, n, active_k_list, time_limit):
    """SLSQP epigraph in q-space"""
    m = n // 2
    dx = 2.0 / n
    h = build_h_from_q(q0, n)
    corrs = np.correlate(h, 1.0 - h, mode='full')
    t0 = corrs.max() * dx * 1.001 + 1e-6
    
    x0 = np.concatenate([[t0], q0])
    
    # Precompute active lag parameters
    active_params = []
    for k in active_k_list:
        d = k - (n - 1)
        active_params.append(d)
    
    class EpigraphQSolver:
        def __init__(self):
            self.m = m
            self.n = n
            self.dx = dx
            self.active_params = active_params
        
        def obj(self, x):
            return x[0]
        
        def obj_grad(self, x):
            g = np.zeros(self.m + 1)
            g[0] = 1.0
            return g
        
        def con(self, x, idx):
            t = x[0]
            q = x[1:]
            h = build_h_from_q(q, self.n)
            d = self.active_params[idx]
            n = self.n
            if d >= 0:
                if d < n:
                    val = np.sum(h[d:] * (1.0 - h[:n - d]))
                else:
                    val = 0.0
            else:
                dd = -d
                if dd < n:
                    val = np.sum(h[:n - dd] * (1.0 - h[dd:]))
                else:
                    val = 0.0
            return t - val * self.dx
        
        def con_grad(self, x, idx):
            q = x[1:]
            h = build_h_from_q(q, self.n)
            d = self.active_params[idx]
            n = self.n
            g = np.zeros(self.m + 1)
            g[0] = 1.0
            G = corr_gradient_d(h, d, n)
            Gq = (G[:self.m] - G[self.m:][::-1]) * self.dx
            g[1:] = -Gq
            return g
    
    solver = EpigraphQSolver()
    cons = []
    for i in range(len(active_k_list)):
        cons.append({
            'type': 'ineq',
            'fun': lambda x, i=i: solver.con(x, i),
            'jac': lambda x, i=i: solver.con_grad(x, i)
        })
    
    bounds = [(0.0, 1.0)] + [(0.0, 1.0)] * m
    try:
        res = minimize(solver.obj, x0, jac=solver.obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': 300, 'ftol': 1e-14})
        q_out = np.clip(res.x[1:], 0, 1)
        return q_out
    except Exception:
        return q0.copy()

def active_set_search_q(q0, n, time_limit, max_rounds=15, initial_active=20, max_active=50):
    """Active-set search in q-space"""
    m = n // 2
    dx = 2.0 / n
    q = q0.copy()
    
    h = build_h_from_q(q, n)
    corrs = np.correlate(h, 1.0 - h, mode='full')
    current_max = corrs.max() * dx
    best_q = q.copy()
    best_c5 = current_max
    
    # Initial active set: top correlations
    top_idx = np.argsort(corrs)[::-1][:initial_active]
    active_k = sorted(top_idx.tolist())
    
    t_start = time.time()
    for _round in range(max_rounds):
        if time.time() - t_start > time_limit - 5:
            break
        
        # Solve epigraph
        q_new = epigraph_q_solve(q, n, active_k, time_limit)
        h_new = build_h_from_q(q_new, n)
        corrs_new = np.correlate(h_new, 1.0 - h_new, mode='full')
        c5_new = corrs_new.max() * dx
        
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_q = q_new.copy()
        
        # Check for hidden violations
        hidden_mask = corrs_new * dx > best_c5 + 1e-6
        if hidden_mask.sum() > 0:
            hidden_idx = np.where(hidden_mask)[0]
            new_active = sorted(set(active_k) | set(hidden_idx.tolist()))
            if len(new_active) > max_active:
                # Keep top by correlation value
                vals = [(k, corrs_new[k]) for k in new_active]
                vals.sort(key=lambda x: -x[1])
                new_active = sorted([v[0] for v in vals[:max_active]])
            active_k = new_active
        
        q = q_new
        
        if abs(c5_new - best_c5) < 1e-13 and not hidden_mask.any():
            break
    
    return best_q, best_c5

def interp_h_to_n(h_src, n_dst):
    """Interpolate h from one grid size to another"""
    n_src = len(h_src)
    x_src = np.linspace(0, 2, n_src)
    x_dst = np.linspace(0, 2, n_dst)
    h_dst = np.interp(x_dst, x_src, h_src)
    return h_dst

def coordinate_descent_full(h0, n, target_sum, time_limit, max_passes=20):
    """Coordinate descent in full space"""
    h = h0.copy()
    
    def fh(h_in):
        hi = np.clip(h_in, 0, 1)
        s, _ = exact_score(hi)
        return s
    
    current = fh(h)
    t_start = time.time()
    golden = (np.sqrt(5) - 1) / 2
    
    for _pass in range(max_passes):
        if time.time() - t_start > time_limit:
            break
        improved = False
        for i in range(n):
            if time.time() - t_start > time_limit:
                break
            lo, hi = 0.0, 1.0
            c1 = lo + (1 - golden) * (hi - lo)
            c2 = lo + golden * (hi - lo)
            hm1 = h.copy()
            hm2 = h.copy()
            hm1[i] = c1
            hm2[i] = c2
            f1 = fh(hm1)
            f2 = fh(hm2)
            
            for _iter in range(30):
                if hi - lo < 1e-10:
                    break
                if f1 < f2:
                    hi = c2
                    c2 = c1
                    f2 = f1
                    c1 = lo + (1 - golden) * (hi - lo)
                    hm1[i] = c1
                    f1 = fh(hm1)
                else:
                    lo = c1
                    c1 = c2
                    f1 = f2
                    c2 = lo + golden * (hi - lo)
                    hm2[i] = c2
                    f2 = fh(hm2)
            
            opt_c = (lo + hi) / 2
            h[i] = opt_c
            new_f = fh(h)
            if new_f < current - 1e-12:
                current = new_f
                improved = True
        
        if not improved:
            break
    
    # Final projection
    h = project_feasible(h, target_sum)
    return h, fh(h)

def smooth_full_lbfgs(h0, n, target_sum, beta, time_limit):
    """L-BFGS-B with smooth objective in full space"""
    
    def val_grad(x):
        h = np.clip(x, 0, 1)
        corrs = np.correlate(h, 1.0 - h, mode='full')
        shifted = corrs - corrs.max()
        w = np.exp(beta * shifted)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(len(corrs)) / len(corrs)
        else:
            w /= w_sum
        val = logsumexp(beta * corrs) / beta * (2.0 / n)
        g = np.zeros(n)
        for k in range(len(corrs)):
            if w[k] < 1e-14:
                continue
            d = k - (n - 1)
            g += w[k] * corr_gradient_d(h, d, n)
        return val, g * (2.0 / n)
    
    def obj(x):
        return val_grad(x)[0]
    
    def grad(x):
        return val_grad(x)[1]
    
    # Add penalty for sum constraint
    def obj_pen(x):
        v, g = val_grad(x)
        sum_viol = x.sum() - target_sum
        v += 100.0 * sum_viol ** 2
        g += 200.0 * sum_viol * np.ones(n)
        return v
    
    def grad_pen(x):
        v, g = val_grad(x)
        sum_viol = x.sum() - target_sum
        g += 200.0 * sum_viol * np.ones(n)
        return g
    
    try:
        res = minimize(obj_pen, h0, jac=grad_pen, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': 100, 'ftol': 1e-14, 'gtol': 1e-8})
        h_out = project_feasible(res.x, target_sum)
        return h_out
    except Exception:
        return h0.copy()
