def smooth_q_obj_grad(q, n, beta):

    corrs = compute_corrs(h)

    if ws < 1e-30:

class EpQ:
    def __init__(self, n, active_lags):
        self.n = n
        self.m = n // 2
        self.active = active_lags
        self.dx = 2.0 / n

    def obj(self, x):
        return x[0]

    def obj_grad(self, x):
        g = np.zeros(self.m + 1)
        g[0] = 1.0
        return g

    def con_fun(self, x, i):
        t = x[0]
        q = x[1:]
        h = build_h_from_q(q, self.n)
        k = self.active[i]
        d = k - (self.n - 1)
        n = self.n
        if d >= 0:
            val = np.sum(h[d:] * (1.0 - h[:n - d]))
        else:
            t2 = -d
            val = np.sum(h[t2:] * (1.0 - h[:n - t2]))
        return t - val * self.dx

    def con_grad(self, x, i):
        q = x[1:]
        h = build_h_from_q(q, self.n)
        k = self.active[i]
        d = k - (self.n - 1)
        n = self.n
        m = self.m
        g = np.zeros(m + 1)
        g[0] = 1.0
        Gh = np.zeros(n)
        if d >= 0:
            for jj in range(n - d):
                Gh[jj + d] += (1.0 - h[jj])
                Gh[jj] -= h[jj + d]
        else:
            t2 = -d
            for jj in range(t2, n):
                Gh[jj] += (1.0 - h[jj - t2])
                Gh[jj - t2] -= h[jj]
        if n % 2 == 0:
            g[1:] = -(Gh[:m] - Gh[m:][::-1]) * self.dx
        else:
            g[1:] = -(Gh[:m] - Gh[m + 1:][::-1]) * self.dx
        return g


def solve_epigraph_q(q, n, active_lags, t0, maxiter=200):

                q_pert = np.clip(q_g + rng.randn(len(q_g)) * sigma, 0.0, 1.0)
                if trial % 5 == 4:
                    flip = rng.rand(len(q_g)) < 0.1
                    q_pert[flip] = 1.0 - q_pert[flip]
                q_pert = np.clip(q_pert, 0.0, 1.0)
                h_pert = build_h_from_q(q_pert, n_g)
                s_pert = exact_score(h_pert)
                if s_pert < best_c5:
                    q_cd3 = coordinate_descent(q_pert, n_g, max_sweeps=10)
                    h_cd3 = build_h_from_q(q_cd3, n_g)
                    update(h_cd3, n_g)
                    q_g = q_from_h(best_h, n_g)

    # Phase 6: Final coordinate descent polish
    if n_curr % 2 == 0 and check_time(30):
        n_f = n_curr
        q_f = q_from_h(best_h, n_f)
        q_f2 = coordinate_descent(q_f, n_f, max_sweeps=15)
        h_f2 = build_h_from_q(q_f2, n_f)
        update(h_f2, n_f)

    # Phase 7: Final smooth polish
    if n_curr % 2 == 0 and check_time(30):
        n_p = n_curr
        q_p = q_from_h(best_h, n_p)
        q_p2 = smooth_lbfgs_solve(q_p, n_p, 3000, maxiter=100)
        h_p2 = build_h_from_q(q_p2, n_p)
        update(h_p2, n_p)

    # Final: ensure valid output
    if best_h is None:
        best_h = h0
        best_c5 = exact_score(best_h)
    n_out = len(best_h)
    ts_out = n_out / 2.0
    best_h = project_feasible(np.clip(best_h, 0.0, 1.0), ts_out)
    final_score = exact_score(best_h)
    if final_score < best_c5:
        best_c5 = final_score
    return best_h, best_c5, n_out
