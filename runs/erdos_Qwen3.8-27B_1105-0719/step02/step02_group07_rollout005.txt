import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time


def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)


def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full').max() * dx


def corr_vals(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx


def max_subgradient(h):
    n = len(h)
    dx = 2.0 / n
    corrs = corr_vals(h)
    max_c = corrs.max()
    active = np.where(np.abs(corrs - max_c) < 1e-7)[0]
    grad = np.zeros(n)
    for k in active:
        d = k - (n - 1)
        i_lo = max(0, -d)
        i_hi = min(n - 1, n - 1 - d)
        if i_lo > i_hi:
            continue
        idx = np.arange(i_lo, i_hi + 1)
        grad[idx] += (1.0 - h[idx + d]) * dx
        grad[idx + d] += (-h[idx]) * dx
    grad /= max(len(active), 1)
    return grad, max_c


def smax_gradient(h, beta):
    n = len(h)
    dx = 2.0 / n
    corrs = corr_vals(h)
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    ws = w.sum()
    if ws < 1e-30:
        w = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        w /= ws
    grad = np.zeros(n)

        if wk < 1e-14:
            continue
        d = k - (n - 1)
        i_lo = max(0, -d)
        i_hi = min(n - 1, n - 1 - d)
        if i_lo > i_hi:
            continue
        idx = np.arange(i_lo, i_hi + 1)
        grad[idx] += wk * (1.0 - h[idx + d])
        grad[idx + d] += wk * (-h[idx])
    return grad * dx


def smax_value(h, beta):
    corrs = corr_vals(h)
    return logsumexp(beta * corrs) / beta


def projected_grad_descent(h0, grad_func, max_iter, lr0, use_smax_beta=0):
    n = len(h0)
    target_sum = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_c = exact_score(h)
    cur_obj = smax_value(h, use_smax_beta) if use_smax_beta > 0 else exact_score(h)
    n_stalled = 0

    for it in range(max_iter):
        g = grad_func(h)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break
        lr = lr0 / (1.0 + 0.002 * it)
        for _ in range(50):
            h_new = project_feasible(h - lr * g, target_sum)
            if use_smax_beta > 0:
                obj_new = smax_value(h_new, use_smax_beta)
            else:
                obj_new = exact_score(h_new)
            if obj_new <= cur_obj - 1e-15 or lr < 1e-10:
                break
            lr *= 0.5
        h = h_new
        c = exact_score(h)
        if c < best_c - 1e-12:
            best_c = c
            best_h = h.copy()
            n_stalled = 0
        else:
            n_stalled += 1
        if use_smax_beta > 0:
            cur_obj = smax_value(h, use_smax_beta)
        else:
            cur_obj = c
        if n_stalled > 200:
            break
    return best_h, best_c


def coordinate_rounding(h, n_moves=500, rng=None):
    if rng is None:
        rng = np.random.RandomState(0)
    n = len(h)
    target_sum = n / 2.0
    h = h.copy()
    best_c = exact_score(h)
    for _ in range(n_moves):
        j = rng.randint(n)
        cur = h[j]
        best_val = cur
        for val in [0.0, 1.0]:
            if abs(cur - val) < 1e-8:
                continue
            h_alt = h.copy()
            h_alt[j] = val
            h_alt = project_feasible(h_alt, target_sum)
            c_alt = exact_score(h_alt)
            if c_alt < best_c - 1e-12:
                best_c = c_alt
                best_val = val
        h[j] = best_val
    return h, best_c


def block_optimize(h, block_size, n_blocks, rng):
    n = len(h)
    target_sum = n / 2.0
    h = h.copy()
    best_c = exact_score(h)
    for _ in range(n_blocks):
        start = rng.randint(0, n - block_size + 1)
        mask = np.zeros(n, dtype=bool)
        mask[start:start + block_size] = True
        h_block = h[mask].copy()
        h_fixed_sum = h[~mask].sum()
        target = target_sum - h_fixed_sum

        def obj(x):
            h_full = h.copy()
            h_full[mask] = np.clip(x, 0.0, 1.0)
            h_full = project_feasible(h_full, target_sum)
            return exact_score(h_full)

        try:
            res = minimize(obj, h_block, method='Nelder-Mead',
                           options={'maxiter': 500, 'xatol': 1e-10, 'fatol': 1e-12})
            h_full = h.copy()
            h_full[mask] = np.clip(res.x, 0.0, 1.0)
            h_full = project_feasible(h_full, target_sum)
            c_new = exact_score(h_full)
            if c_new < best_c - 1e-12:
                best_c = c_new
                h = h_full
        except Exception:
            pass
    return h, best_c


def symmetric_h(q, n):
    m = n // 2
    q = np.clip(q, 0.0, 1.0)
    h = np.empty(n)
    h[:m] = q
    h[m:] = 1.0 - q[::-1]
    return h

    m = n // 2
