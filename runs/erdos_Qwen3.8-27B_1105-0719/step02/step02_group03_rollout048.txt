    nv = n + 1  # d_0..d_{n-1}, t

    c = np.zeros(nv)
    c[-1] = 1.0

    # For each active lag: G[k,:] @ d - t <= -R[k]

        A_ub[idx, -1] = -1.0

    A_eq = np.zeros((1, nv))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds: -h <= d <= 1-h, t >= -inf (or t >= 0)

So R[k] is actually linear in h[j] when all other h values are fixed! This makes sense because R[k] is a multilinear function of h (each term is a product of at most 2 h values, and h[j] appears in at most 2 of those terms for a given k).

Wait, h[j] can appear in R[k] in at most 2 ways:
1. As h[i+L] where i = j-L
2. As h[i] where i = j (contributing to (1-h[j]))

So the dependence of R[k] on h[j] is:
- h[j]*(1-h[j-L]) [if j-L valid]: linear in h[j]
- h[j+L]*(1-h[j]) [if j+L valid]: linear in h[j]

Total: h[j]*(1-h[j-L]) + h[j+L]*(1-h[j]) = h[j] - h[j]*h[j-L] + h[j+L] - h[j+L]*h[j]

If both j-L and j+L are valid, this is h[j]*(1-h[j-L]) + h[j+L]*(1-h[j]) which is linear in h[j] (the coefficients depend on other h values but not on h[j] itself... wait, -h[j]*h[j-L] is not linear in h[j] if we consider h[j-L] to potentially equal h[j]).

No wait, h[j-L] and h[j+L] are DIFFERENT coordinates from h[j] (assuming L≠0). So when we vary h[j] and keep everything else fixed, R[k] is indeed linear in h[j].

Therefore, sum_k R[k]^2 is quadratic in h[j] (since each R[k] is linear in h[j], the square is quadratic, and the sum is quadratic).

This means coordinate descent on sum_k R[k]^2 is a sequence of 1D quadratic minimizations, which can be solved exactly!

The coordinate descent update for h[j]:
minimize f(h[j]) = sum_k (a_k + b_k * h[j])^2
where a_k = R[k] - (linear part of R[k] in h[j]) and b_k = dR[k]/dh[j]

f'(h[j]) = 2 * sum_k b_k * (a_k + b_k * h[j]) = 0
→ h[j] = -sum_k a_k * b_k / sum_k b_k^2

Clamped to [0, 1].

This is a very efficient approach! Each coordinate update requires O(n) work (to compute a_k and b_k for all k), and we do n coordinates per sweep, so O(n^2) per sweep.

import numpy as np
from scipy.optimize import linprog
import time


def project_to_sum(h, target_sum):
    n = len(h)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        hp = np.clip(h - mid, 0.0, 1.0)
        if hp.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    hp = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - hp.sum()
    if abs(diff) > 1e-12:
        mask = (hp > 1e-10) & (hp < 1.0 - 1e-10)
        if mask.sum() > 0:
            hp[mask] += diff / mask.sum()
        else:
            hp += diff / n
    return np.clip(hp, 0.0, 1.0)


def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def c5_of_h(h, dx):
    return float(compute_corrs(h).max() * dx)


def get_active_gradients(h, active_lags, n):
    G = np.zeros((len(active_lags), n))
    idx = np.arange(n)
    for ii, k in enumerate(active_lags):
        offset = k - (n - 1)
        jm = idx - offset
        jp = idx + offset
        mask_m = (jm >= 0) & (jm < n)
        mask_p = (jp >= 0) & (jp < n)
        G[ii, mask_m] += (1.0 - h[jm[mask_m]])
        G[ii, mask_p] -= h[jp[mask_p]]
    return G


def solve_lp_direction(h, G, R_active):
    n = len(h)
    nv = n + 1
    c = np.zeros(nv)
    c[-1] = 1.0
    na = len(R_active)
    A_ub = np.zeros((na, nv))
    for ii in range(na):
        A_ub[ii, :n] = G[ii, :]
        A_ub[ii, -1] = -1.0
    b_ub = -R_active
    A_eq = np.zeros((1, nv))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-h[i], -1.0), min(1.0 - h[i], 1.0)) for i in range(n)]
    bounds.append((-np.inf, np.inf))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            d = res.x[:n]
            if np.all(np.isfinite(d)) and np.max(np.abs(d)) <= 1.5:
                return d
    except Exception:
        pass
    return None


def line_search(h, d, dx, n):
    if np.all(np.abs(d) < 1e-14):
        return h.copy(), c5_of_h(h, dx)
    lo_a = 1e-12
    hi_a = 1e10
    for i in range(n):
        if d[i] > 1e-14:
            hi_a = min(hi_a, (1.0 - h[i]) / d[i] * 0.9999)
        elif d[i] < -1e-14:
            hi_a = min(hi_a, -h[i] / d[i] * 0.9999)
    if hi_a < lo_a:
        return h.copy(), c5_of_h(h, dx)
    hi_a = min(hi_a, 1.0)
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    for frac in [0.05, 0.1, 0.2, 0.35, 0.5, 0.7, 0.9]:
        alpha = frac * hi_a
        ht = np.clip(h + alpha * d, 0.0, 1.0)
        ht = project_to_sum(ht, n / 2.0)
        c5t = c5_of_h(ht, dx)
        if c5t < best_c5:
            best_c5 = c5t
            best_h = ht.copy()
    lo_b, hi_b = 0.0, hi_a
    for _ in range(25):
        mid = (lo_b + hi_b) / 2
        if mid < lo_a:
            lo_b = mid
            continue
        ht = np.clip(h + mid * d, 0.0, 1.0)
        ht = project_to_sum(ht, n / 2.0)
        c5t = c5_of_h(ht, dx)
        if c5t < best_c5 - 1e-15:
            best_c5 = c5t
            best_h = ht.copy()
            lo_b = mid
        else:
            hi_b = mid
        if hi_b - lo_b < hi_a * 1e-8:
            break
    return best_h, best_c5


def lp_optimize(h0, dx, max_iters, t0, time_limit):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    no_improve = 0
    margins = [0.4, 0.25, 0.15, 0.1, 0.06, 0.03, 0.015, 0.008]
    total_iters = 0
    for margin_frac in margins:
        stagnant = 0
        for _ in range(max_iters):
            total_iters += 1
            if time.time() - t0 > time_limit:
                return best_h, best_c5
            corrs = compute_corrs(h)
            t_max = corrs.max()
            threshold = t_max * margin_frac
            active_mask = corrs > threshold
            active_lags = np.where(active_mask)[0]
            if len(active_lags) == 0:
                break
            if len(active_lags) > 20:
                top_idx = np.argsort(-corrs[active_lags])[:20]
                active_lags = active_lags[top_idx]
            R_active = corrs[active_lags]
            G = get_active_gradients(h, active_lags, n)
            d = solve_lp_direction(h, G, R_active)
            if d is None:
                break
            d_norm = np.linalg.norm(d)
            if d_norm < 1e-13:
                break
            new_h, new_c5 = line_search(h, d, dx, n)
            if new_c5 < best_c5 - 1e-14:
                best_c5 = new_c5
                best_h = new_h.copy()
                h = new_h.copy()
                no_improve = 0
                stagnant = 0
            else:
                no_improve += 1
                stagnant += 1
            if stagnant > 30:
                break
    return best_h, best_c5


def variance_coordinate_descent(h0, dx, n_sweeps, t0, time_limit):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    for sweep in range(n_sweeps):
        if time.time() - t0 > time_limit:
            break
        corrs = compute_corrs(h)
        for j in range(n):
            grad = np.zeros(2 * n - 1)
            for k in range(2 * n - 1):
                offset = k - (n - 1)
                g = 0.0
                jm = j - offset
                jp = j + offset
                if 0 <= jm < n:
                    g += (1.0 - h[jm])
                if 0 <= jp < n:
                    g -= h[jp]
                grad[k] = g
            a = corrs - grad * h[j]
            S = sum_k_a_b = 0.0
            Sbb = sum_k_b_b = 0.0
            for k in range(2 * n - 1):
                S += a[k] * grad[k]
                Sbb += grad[k] * grad[k]
            if Sbb > 1e-15:
                h_new_j = -S / Sbb
                h_new_j = np.clip(h_new_j, 0.0, 1.0)
            else:
                h_new_j = h[j]
            h[j] = h_new_j
        h = project_to_sum(h, n / 2.0)
        c5_now = c5_of_h(h, dx)
        if c5_now < best_c5:
            best_c5 = c5_now
            best_h = h.copy()
    return best_h, best_c5


def binary_from_continuous(h, n):
    target = int(round(n / 2))
    sorted_idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[sorted_idx[:target]] = 1.0
    return h_bin


def compute_binary_delta(h, a, b, n):
    L = np.arange(-(n - 1), n)
    delta = np.zeros(2 * n - 1)
    mask1 = (a + L >= 0) & (a + L < n)
    delta[mask1] += h[a + L[mask1]]
    mask2 = (a - L >= 0) & (a - L < n)
    delta[mask2] -= (1.0 - h[a - L[mask2]])
    mask3 = (b + L >= 0) & (b + L < n)
    delta[mask3] -= h[b + L[mask3]]
    mask4 = (b - L >= 0) & (b - L < n)
    delta[mask4] += (1.0 - h[b - L[mask4]])
    return delta


def binary_local_search(h0, dx, rng, max_iters=80000, patience=6000):
    n = len(h0)
    target = int(round(n / 2))
    h = np.zeros(n)
    sorted_idx = np.argsort(-h0)
    h[sorted_idx[:target]] = 1.0
    corrs = compute_corrs(h)
    best_c5 = corrs.max() * dx
    best_h = h.copy()
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    no_improve = 0
    temp = 0.03
    cooling = 0.9998
    for it in range(max_iters):
        if it % 3000 == 0:
            temp *= cooling
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        delta = compute_binary_delta(h, i, j, n)
        new_corrs = corrs + delta
        new_c5 = new_corrs.max() * dx
        if new_c5 <= best_c5 + 1e-15:
            h[i], h[j] = 0.0, 1.0
            corrs = new_corrs
            best_c5 = new_c5
            best_h = h.copy()
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            no_improve = 0
        else:
            if temp > 1e-10:
                dv = new_c5 - best_c5
                if rng.random() < np.exp(-dv / temp):
                    h[i], h[j] = 0.0, 1.0
                    corrs = new_corrs
                    ones = np.where(h == 1.0)[0]
                    zeros = np.where(h == 0.0)[0]
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1
        if no_improve > patience:
            break
    return best_h, best_c5


def make_structured_start(n, mode, rng):
    target = n / 2.0
    h = np.zeros(n)
    if mode == 0:
        phi = (1.0 + np.sqrt(5.0)) / 2.0
        for k in range(int(target)):
            pos = int(((k * phi) % 1.0) * n) % n
            h[pos] = 1.0
    elif mode == 1:
        for k in range(int(target)):
            pos = int((k * 0.618033988749895) * n) % n
            h[pos] = 1.0
    elif mode == 2:
        h[::2] = 1.0
        if abs(h.sum() - target) > 0.5:
            if h.sum() < target:
                h[-1] = 1.0
            else:
                h[0] = 0.0
    elif mode == 3:
        idx = rng.permutation(n)
        h[idx[:int(target)]] = 1.0
    else:
        x = np.linspace(0, 4 * np.pi, n)
        h = 0.5 + 0.25 * np.sin(x) + 0.1 * np.sin(3 * x + 1.0)
    return project_to_sum(h, target)


def remaining_time(t0, budget_s):
    return budget_s - (time.time() - t0)
