3. Compute a direction that reduces all active correlations simultaneously
4. Take a step in that direction, projecting to [0,1] and the sum constraint

This is essentially a projected subgradient method.

The subgradient of max_k c_k(h) at h is the subgradient of c_{k*}(h) where k* is an active lag. To reduce the max, we want to move in the negative subgradient direction.

For c_k(h) = dx * sum_i h_i (1 - h_{i+d}), the gradient with respect to h is:
∂c_k/∂h_j = dx * (1 - h_{j+d}) [if j+d valid] - dx * h_{j-d} [if j-d valid]

This is exactly B[j, k] from above!

So the subgradient direction (for the max) is a convex combination of the gradients at active lags:
g = sum_k w_k * grad_k, where w_k >= 0, sum(w_k) = 1, and w_k > 0 only for active k.

To minimize the max, we move in the direction -g (projected).

Let me implement a subgradient method:

```python
def subgradient_method(h_init, n_iters=10000, time_limit=200.0):
    n = len(h_init)
    dx = 2.0 / n
    target_sum = n / 2.0
    h = h_init.copy()
    t0 = time.time()
    
    # Step size: use diminishing step
    # Or Polyak step: (f(x) - f*) / ||g||^2, but we don't know f*
    # Use: alpha_t = c / sqrt(t) for some constant c
    
    c5_history = [exact_score(h)]
    best_h = h.copy()
    best_c5 = c5_history[0]
    
    for it in range(n_iters):
        if time.time() - t0 > time_limit:
            break
        
        c = corr_vals(h)
        current_max = c.max()
        
        if current_max < best_c5:
            best_c5 = current_max
            best_h = h.copy()
        
        # Compute subgradient (use uniform weights on top active lags)
        threshold = current_max - 1e-6
        active = np.where(c >= threshold)[0]
        # Use top few active lags
        top_idx = np.argsort(c)[::-1][:min(len(active), 10)]
        w = np.ones(len(top_idx)) / len(top_idx)
        
        # Compute gradient
        g = np.zeros(n)
        for idx, k in enumerate(top_idx):
            d = k - (n - 1)
            jd = np.arange(0, n) + d
            valid1 = (jd >= 0) & (jd < n)
            g[valid1] += w[idx] * dx * (1.0 - h[jd[valid1]])
            jd2 = np.arange(0, n) - d
            valid2 = (jd2 >= 0) & (jd2 < n)
            g[valid2] -= w[idx] * dx * h[jd2[valid2]]
        
        # Step size
        alpha = 0.01 / (1 + 0.01 * it)
        
        # Move in negative gradient direction
        h_new = h - alpha * g
        h_new = project_feasible(h_new, target_sum)
        
        # Check if improvement
        if exact_score(h_new) < current_max:
            h = h_new
        else:
            # Halve step and try again
            h_new = h - alpha * 0.5 * g
            h_new = project_feasible(h_new, target_sum)
            if exact_score(h_new) < current_max:
                h = h_new
            else:
                h = h_new  # accept anyway (subgradient methods don't always decrease)
    
    return best_h, best_c5
```

class EpiSymSolver:
    def __init__(self, n, active_lags):
        self.n = n
        self.m = n // 2
        self.dx = 2.0 / n
        self.active_lags = np.array(active_lags, dtype=int)

        return x[0]

    def jac(self, x):
        g = np.zeros(self.m + 1)
        g[0] = 1.0
        return g

    def con_fun(self, x):
        t = x[0]
        q = x[1:]
        h = make_sym_h(q, self.n)
        corrs = corr_vals(h)
        return t - corrs[self.active_lags]

    def con_jac(self, x):
        q = x[1:]
        h = make_sym_h(q, self.n)
        n = self.n
        m = self.m
        dx = self.dx
        L = len(self.active_lags)
        J = np.zeros((L, m + 1))
        J[:, 0] = 1.0
        for i, lag in enumerate(self.active_lags):
            d = lag - (n - 1)
            i_lo = max(0, -d)
            i_hi = min(n - 1, n - 1 - d)
            G = np.zeros(n)

                G[idx] += (1.0 - h[idx + d])
                G[idx + d] += (-h[idx])
            G *= dx
            J[i, 1:] = -(G[:m] - G[n - 1:n - 1 - m: -1])
        return J

        res = minimize(opt.fun, q0, jac=opt.jac, method='SLSQP',
                       bounds=[(0.0, 1.0)] * m,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        h_new = make_sym_h(res.x, n)
        c5 = exact_score(h_new)
        if res.fun < 1e10:
            return res.x, c5
    except Exception:
        pass
    return None, None


def smooth_full_optimize(h0, n, beta, maxiter=400):
    opt = SmoothFullOpt(n, beta)
    try:
        res = minimize(opt.fun, h0, jac=opt.jac, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        h_out = project_feasible(res.x, n / 2.0)
        c5 = exact_score(h_out)
        return h_out, c5
    except Exception:
        return h0, exact_score(h0)


def solve_epi_sym(q0, n, active_lags, t_init, maxiter=400):
    m = n // 2
    x0 = np.concatenate([[t_init], q0])
    solver = EpiSymSolver(n, active_lags)
    cons = [{'type': 'ineq', 'fun': solver.con_fun, 'jac': solver.con_jac}]
    bounds = [(0.0, 2.0)] + [(0.0, 1.0)] * m
    try:
        res = minimize(solver.fun, x0, jac=solver.jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        h_new = make_sym_h(res.x[1:], n)
        c5 = exact_score(h_new)
        if abs(h_new.sum() - n / 2.0) > 1e-8:
            return None, None
        return res.x[1:], c5
    except Exception:
        return None, None


def active_lags_top(corrs, n_top):
    idx = np.argsort(corrs)[::-1][:n_top]
    return sorted(idx.tolist())


def coord_descent(h_init, n_sweeps, time_limit, t0, randomize=False, rng=None):
    n = len(h_init)
    dx = 2.0 / n
    target_sum = n / 2.0
    h = h_init.copy()
    d_vals = np.arange(-(n - 1), n)
    c = corr_vals(h)
    best_c5 = c.max()
    best_h = h.copy()
    order = np.arange(n)
    for sweep in range(n_sweeps):
        if time.time() - t0 > time_limit:
            break
        if randomize and rng is not None:
            rng.shuffle(order)
        for j in order:
            B = np.zeros(2 * n - 1)
            jd_plus = j + d_vals
            valid_p = (jd_plus >= 0) & (jd_plus < n)
            B[valid_p] += dx * (1.0 - h[jd_plus[valid_p]])
            jd_minus = j - d_vals
            valid_m = (jd_minus >= 0) & (jd_minus < n)
            B[valid_m] -= dx * h[jd_minus[valid_m]]
            A = c - h[j] * B
            t_opt = solve_1d_minimax(A, B)
            delta = t_opt - h[j]
            if abs(delta) > 1e-14:
                c += delta * B
                h[j] = t_opt
        h = project_feasible(h, target_sum)
        c = corr_vals(h)
        cur_max = c.max()
        if cur_max < best_c5:
            best_c5 = cur_max
            best_h = h.copy()
    return best_h, best_c5


def subgradient_method(h_init, n_iters, time_limit, t0):
    n = len(h_init)
    dx = 2.0 / n
    target_sum = n / 2.0
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = exact_score(h)
    d_vals = np.arange(-(n - 1), n)
    for it in range(n_iters):
        if time.time() - t0 > time_limit:
            break
        c = corr_vals(h)
        current_max = c.max()
        if current_max < best_c5:
            best_c5 = current_max
            best_h = h.copy()
        top_k = min(15, 2 * n - 1)
        top_idx = np.argsort(c)[::-1][:top_k]
        w = np.ones(len(top_idx)) / len(top_idx)
        g = np.zeros(n)
        for ii, k in enumerate(top_idx):
            d = k - (n - 1)
            jd = np.arange(n) + d
            v1 = (jd >= 0) & (jd < n)
            g[v1] += w[ii] * dx * (1.0 - h[jd[v1]])
            jd2 = np.arange(n) - d
            v2 = (jd2 >= 0) & (jd2 < n)
            g[v2] -= w[ii] * dx * h[jd2[v2]]
        alpha = 0.005 / (1.0 + 0.005 * it)
        h_new = h - alpha * g
        h_new = project_feasible(h_new, target_sum)
        if exact_score(h_new) <= current_max + 1e-10:
            h = h_new
        else:
            h_new2 = h - alpha * 0.3 * g
            h_new2 = project_feasible(h_new2, target_sum)
            h = h_new2
    return best_h, best_c5


def try_sym_smooth(q_best, n, betas, t0, budget_s):
    best = (q_best, exact_score(make_sym_h(q_best, n)))
    for beta in betas:
        if time.time() - t0 > budget_s - 20:
            break
        q_new, c5_new = smooth_sym_optimize(best[0], n, beta, maxiter=500)
        if q_new is not None and c5_new < best[1] - 1e-10:
            best = (q_new, c5_new)
    return best


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    try:
        h_parent = np.array(initial_h_values, dtype=float)
    except NameError:
        h_parent = 0.5 * np.ones(126)

    n = len(h_parent)
    if n % 2 != 0:
        h_parent = h_parent[:n - 1]
        n = len(h_parent)

    h_best = project_feasible(h_parent, n / 2.0)
    c5_best = exact_score(h_best)
    m = n // 2
    q_best = make_q_from_h(h_best, n)

    def remaining():
        return budget_s - (time.time() - t0)

    # Phase 1: Coordinate descent (new method)
    h_cd, c5_cd = coord_descent(h_best, n_sweeps=150, time_limit=120, t0=t0)
    if c5_cd < c5_best:
        c5_best = c5_cd
        h_best = h_cd
        q_best = make_q_from_h(h_best, n)

    # Phase 2: Randomized coordinate descent
    if remaining() > 60:
        h_cd2, c5_cd2 = coord_descent(h_best, n_sweeps=100, time_limit=50,
                                      t0=t0, randomize=True, rng=rng)
        if c5_cd2 < c5_best:
            c5_best = c5_cd2
            h_best = h_cd2
            q_best = make_q_from_h(h_best, n)

    # Phase 3: Symmetric smooth SLSQP with increasing beta
    q_best, c5_sym = try_sym_smooth(q_best, n, [60, 120, 250, 500, 1000, 2000], t0, budget_s)
    if c5_sym < c5_best:
        c5_best = c5_sym
        h_best = make_sym_h(q_best, n)

    # Phase 4: Epigraph in symmetric space
    for rnd in range(10):
        if remaining() < 15:
            break
        h_cur = make_sym_h(q_best, n)
        corrs = corr_vals(h_cur)
        n_act = min(30 + 15 * rnd, 2 * n - 1)
        active = active_lags_top(corrs, n_act)
        q_new, c5_new = solve_epi_sym(q_best, n, active, t_init=corrs.max() * 1.0001,
                                      maxiter=400)
        if q_new is not None and c5_new < c5_best - 1e-10:
            c5_best = c5_new
            q_best = q_new
            h_best = make_sym_h(q_best, n)

    # Phase 5: Full-space smooth optimization (break symmetry)
    if remaining() > 30:
        for beta in [200, 500, 1000]:
            if remaining() < 15:
                break
            h_new, c5_new = smooth_full_optimize(h_best, n, beta, maxiter=400)
            if c5_new < c5_best - 1e-10:
                c5_best = c5_new
                h_best = h_new
                q_best = make_q_from_h(h_best, n)

    # Phase 6: Subgradient method for global exploration
    if remaining() > 40:
        h_sg, c5_sg = subgradient_method(h_best, n_iters=3000, time_limit=60, t0=t0)
        if c5_sg < c5_best:
            c5_best = c5_sg
            h_best = h_sg
            q_best = make_q_from_h(h_best, n)

    # Phase 7: Coordinate descent again from improved point
    if remaining() > 40:
        h_cd3, c5_cd3 = coord_descent(h_best, n_sweeps=200, time_limit=60, t0=t0)
        if c5_cd3 < c5_best:
            c5_best = c5_cd3
            h_best = h_cd3
            q_best = make_q_from_h(h_best, n)

    # Phase 8: Final symmetric smooth polish
    if remaining() > 20:
        q_best2, c5_s2 = try_sym_smooth(q_best, n, [500, 1000, 2000, 4000], t0, budget_s)
        if c5_s2 < c5_best:
            c5_best = c5_s2
            h_best = make_sym_h(q_best2, n)
            q_best = q_best2

    # Phase 9: Random perturbations + local optimization
    n_trials = 30
    for trial in range(n_trials):
        if remaining() < 12:
            break
        scale = 0.02 + 0.3 * (trial / max(n_trials, 1))
        q_pert = q_best + scale * rng.randn(m)
        q_pert = np.clip(q_pert, 0.0, 1.0)
        q_new, c5_new = smooth_sym_optimize(q_pert, n, 200, maxiter=200)
        if q_new is not None and c5_new < c5_best + 0.003:
            q_new2, c5_new2 = smooth_sym_optimize(q_new, n, 800, maxiter=300)
            if q_new2 is not None and c5_new2 < c5_best - 1e-10:
                c5_best = c5_new2
                q_best = q_new2
                h_best = make_sym_h(q_best, n)
                if remaining() > 15:
                    h_cur = make_sym_h(q_best, n)
                    corrs = corr_vals(h_cur)
                    active = active_lags_top(corrs, 50)
                    q_ep, c5_ep = solve_epi_sym(q_best, n, active,
                                                 t_init=corrs.max() * 1.0001, maxiter=300)
                    if q_ep is not None and c5_ep < c5_best - 1e-10:
                        c5_best = c5_ep
                        q_best = q_ep
                        h_best = make_sym_h(q_best, n)

    # Phase 10: Try nearby n values
    if remaining() > 50:
        for delta in [2, -2, 4, -4, 6, -6, 8, -8]:
            n_try = n + delta
            if n_try < 40 or n_try % 2 != 0 or remaining() < 20:
                continue
            x_old = np.linspace(0, 2, n, endpoint=False)
            x_new_grid = np.linspace(0, 2, n_try, endpoint=False)
            h_interp = np.interp(x_new_grid, x_old, h_best)
            h_interp = project_feasible(h_interp, n_try / 2.0)
            m_try = n_try // 2
            q_try = make_q_from_h(h_interp, n_try)
            q_new, c5_new = smooth_sym_optimize(q_try, n_try, 300, maxiter=300)
            if q_new is not None:
                q_new, c5_new = smooth_sym_optimize(q_new, n_try, 800, maxiter=300)
            if q_new is not None and c5_new < c5_best - 1e-9:
                c5_best = c5_new
                n = n_try
                m = m_try
                q_best = q_new
                h_best = make_sym_h(q_best, n)

    # Phase 11: Final full-space polish
    if remaining() > 20:
        h_new, c5_new = smooth_full_optimize(h_best, len(h_best), 1000, maxiter=500)
        if c5_new < c5_best - 1e-10:
            c5_best = c5_new
            h_best = h_new

    # Phase 12: Final coordinate descent polish
    if remaining() > 15:
        h_final, c5_final = coord_descent(h_best, n_sweeps=50, time_limit=10, t0=t0)
        if c5_final < c5_best:
            c5_best = c5_final
            h_best = h_final

    # Final validation
    h_best = np.clip(h_best, 0.0, 1.0)
    h_best = project_feasible(h_best, len(h_best) / 2.0)
    c5_best = exact_score(h_best)

    return h_best, c5_best, len(h_best)
