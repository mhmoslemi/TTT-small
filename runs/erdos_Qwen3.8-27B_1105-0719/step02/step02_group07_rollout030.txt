from scipy.optimize import minimize
from scipy.special import logsumexp
import time


def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)


def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full').max() * dx


def corr_vals(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx


def make_sym_h(q, n):
    m = n // 2
    h = np.empty(n)
    h[:m] = q
    h[m:] = 1.0 - q[::-1]
    return h


def make_q_from_h(h, n):
    m = n // 2
    q = 0.5 * (h[:m] + 1.0 - h[m:][::-1])
    return np.clip(q, 0.0, 1.0)


def coord_desc(h, n_sweeps, grid_size, seed, time_limit_fn):
    """Coordinate descent with exact max subproblem."""
    n = len(h)
    dx = 2.0 / n
    target_sum = n / 2.0
    rng = np.random.RandomState(seed)
    h = h.copy()
    best_c5 = exact_score(h)
    
    x_grid = np.linspace(0.0, 1.0, grid_size)
    
    for sweep in range(n_sweeps):
        if time_limit_fn():
            break
        order = rng.permutation(n)
        for i in order:
            old_val = h[i]
            # Compute slopes for all lags
            d_range = np.arange(-(n - 1), n)
            ipd = i + d_range
            imd = i - d_range
            
            slopes = np.zeros(2 * n - 1)
            vf = (ipd >= 0) & (ipd < n)
            slopes[vf] += dx * (1.0 - h[ipd[vf]])
            vs = (imd >= 0) & (imd < n)
            slopes[vs] -= dx * h[imd[vs]]
            
            # Current correlations
            c_cur = np.correlate(h, 1.0 - h, mode='full') * dx
            intercepts = c_cur - slopes * old_val
            
            # Subproblem: min_x max_k (intercepts[k] + slopes[k]*x)
            vals = intercepts[None, :] + slopes[None, :] * x_grid[:, None]
            max_vals = vals.max(axis=1)
            h[i] = x_grid[max_vals.argmin()]
        
        h = project_feasible(h, target_sum)
        c5 = exact_score(h)
        if c5 < best_c5:
            best_c5 = c5
    
    return h, best_c5


def smooth_sym_obj(x, n, beta):
    h = make_sym_h(x, n)
    corrs = corr_vals(h)
    return logsumexp(beta * corrs) / beta


def smooth_sym_grad(x, n, beta):
    m = n // 2
    dx = 2.0 / n
    h = make_sym_h(x, n)
    corrs = corr_vals(h)
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    ws = w.sum()
    if ws < 1e-30:
        w = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        w /= ws
    # Gradient of smoothed max w.r.t. h
    n_lags = 2 * n - 1
    grad = np.zeros(n)
    for k in range(n_lags):
        wk = beta * w[k]
        if wk < 1e-15:
            continue
        d = k - (n - 1)
        i_lo = max(0, -d)
        i_hi = min(n - 1, n - 1 - d)
        if i_lo > i_hi:
            continue
        idx = np.arange(i_lo, i_hi + 1)
        grad[idx] += wk * (1.0 - h[idx + d])
        grad[idx + d] += wk * (-h[idx])
    grad *= dx
    return grad[:m] - grad[m:][::-1]


def smooth_sym_optimize(q0, n, beta, maxiter=400):
    m = n // 2
    try:
        res = minimize(smooth_sym_obj, q0, jac=lambda x: smooth_sym_grad(x, n, beta),
                       method='SLSQP', bounds=[(0.0, 1.0)] * m,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        h_new = make_sym_h(res.x, n)
        c5 = exact_score(h_new)
        if res.fun < 1e10:
            return res.x, c5
    except Exception:
        pass
    return None, None

import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time


def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)


def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full').max() * dx


def corr_vals(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx


def make_sym_h(q, n):
    m = n // 2
    h = np.empty(n)
    h[:m] = q
    h[m:] = 1.0 - q[::-1]
    return h


def make_q_from_h(h, n):
    m = n // 2
    q = 0.5 * (h[:m] + 1.0 - h[m:][::-1])
    return np.clip(q, 0.0, 1.0)


def coord_desc_fast(h_init, n_sweeps, grid_size, seed, deadline):
    n = len(h_init)
    dx = 2.0 / n
    target_sum = n / 2.0
    rng = np.random.RandomState(seed)
    h = h_init.copy()
    best_c5 = exact_score(h)
    best_h = h.copy()
    x_grid = np.linspace(0.0, 1.0, grid_size)
    d_range = np.arange(-(n - 1), n)
    n_lags = 2 * n - 1

    for sweep in range(n_sweeps):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        order = rng.permutation(n)
        for i in order:
            old_val = h[i]
            ipd = i + d_range
            imd = i - d_range
            slopes = np.zeros(n_lags)
            vf = (ipd >= 0) & (ipd < n)
            slopes[vf] = dx * (1.0 - h[ipd[vf]])
            vs = (imd >= 0) & (imd < n)
            slopes[vs] -= dx * h[imd[vs]]
            intercepts = c - slopes * old_val
            vals = intercepts[None, :] + slopes[None, :] * x_grid[:, None]
            max_vals = vals.max(axis=1)
            new_val = x_grid[max_vals.argmin()]
            delta = new_val - old_val
            if abs(delta) > 1e-12:
                c += delta * slopes
                h[i] = new_val
        h = project_feasible(h, target_sum)
        c5 = exact_score(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5


def _sym_obj(x, n, beta):
    h = make_sym_h(x, n)
    corrs = corr_vals(h)
    return logsumexp(beta * corrs) / beta


def _sym_grad(x, n, beta):
    m = n // 2
    dx = 2.0 / n
    h = make_sym_h(x, n)
    corrs = corr_vals(h)
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    ws = w.sum()
    if ws < 1e-30:
        w = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        w /= ws
    n_lags = 2 * n - 1
    grad = np.zeros(n)
    for k in range(n_lags):
        wk = beta * w[k]
        if wk < 1e-15:
            continue
        d = k - (n - 1)
        i_lo = max(0, -d)
        i_hi = min(n - 1, n - 1 - d)
        if i_lo > i_hi:
            continue
        idx = np.arange(i_lo, i_hi + 1)
        grad[idx] += wk * (1.0 - h[idx + d])
        grad[idx + d] += wk * (-h[idx])
    grad *= dx
    return grad[:m] - grad[m:][::-1]


def smooth_sym_optimize(q0, n, beta, maxiter=400):
    m = n // 2
    try:
        res = minimize(_sym_obj, q0, args=(n, beta),
                       jac=_sym_grad, method='SLSQP',
                       bounds=[(0.0, 1.0)] * m,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        h_new = make_sym_h(res.x, n)
        c5 = exact_score(h_new)
        if res.fun < 1e10:
            return res.x, c5
    except Exception:
        pass
    return None, None


def subgradient_method(h_init, n_iters, seed, deadline, alpha0=0.01, decay=0.001):
    n = len(h_init)
    dx = 2.0 / n
    target_sum = n / 2.0
    rng = np.random.RandomState(seed)
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = exact_score(h)
    h_sum = np.zeros(n)
    for t in range(n_iters):
        if time.time() > deadline:
            break
        c = corr_vals(h)
        max_c = c.max()
        # Subgradient of max at h: pick a random active (near-active) lag
        tolerance = 0.02 * dx * n
        active = np.where(c > max_c - tolerance)[0]
        k = rng.choice(active)
        d = k - (n - 1)
        i_lo = max(0, -d)
        i_hi = min(n - 1, n - 1 - d)
        grad = np.zeros(n)
        idx = np.arange(i_lo, i_hi + 1)
        grad[idx] += (1.0 - h[idx + d])
        grad[idx + d] += (-h[idx])
        grad *= dx
        alpha = alpha0 / (1.0 + decay * t)
        h_new = h - alpha * grad
        h = project_feasible(h_new, target_sum)
        h_sum += h
        c5 = exact_score(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    # Also try the average
    h_avg = project_feasible(h_sum / max(n_iters, 1), target_sum)
    c5_avg = exact_score(h_avg)
    if c5_avg < best_c5:
        best_c5 = c5_avg
        best_h = h_avg
    return best_h, best_c5


def _full_obj(x, n):
    h = np.clip(x, 0.0, 1.0)
    return exact_score(h)


def active_lags_top(corrs, n_top):
    idx = np.argsort(corrs)[::-1][:n_top]
    return sorted(idx.tolist())


def scoop_optimize(h0, n, beta, maxiter=300):
    target_sum = n / 2.0

    def obj_fn(x):
        h = np.clip(x, 0.0, 1.0)
        corrs = corr_vals(h)
        return logsumexp(beta * corrs) / beta

    def grad_fn(x):
        h = np.clip(x, 0.0, 1.0)
        corrs = corr_vals(h)
        shifted = corrs - corrs.max()
        w = np.exp(beta * shifted)
        ws = w.sum()
        if ws < 1e-30:
            w = np.ones(2 * n - 1) / (2 * n - 1)
        else:
            w /= ws
        dx = 2.0 / n
        n_lags = 2 * n - 1
        grad = np.zeros(n)
        for k in range(n_lags):
            wk = beta * w[k]
            if wk < 1e-15:
                continue
            d = k - (n - 1)
            i_lo = max(0, -d)
            i_hi = min(n - 1, n - 1 - d)
            if i_lo > i_hi:
                continue
            idx = np.arange(i_lo, i_hi + 1)
            grad[idx] += wk * (1.0 - h[idx + d])
            grad[idx + d] += wk * (-h[idx])
        return grad * dx

    try:
        res = minimize(obj_fn, h0, jac=grad_fn, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        h_out = project_feasible(res.x, target_sum)
        return h_out, exact_score(h_out)
    except Exception:
        return h0, exact_score(h0)


def make_periodic_init(n, period):
    h = np.zeros(n)
    block = n // (2 * period)
    if block < 1:
        block = 1
    for block_idx in range(2 * period):
        start = block_idx * block
        end = min(start + block, n)
        if block_idx % 2 == 0:
            h[start:end] = 1.0
    return project_feasible(h, n / 2.0)
