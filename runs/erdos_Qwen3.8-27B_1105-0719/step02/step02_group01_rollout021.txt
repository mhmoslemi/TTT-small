def grad_corr_k(h, k, n):
    """Gradient of corr_k with respect to h."""
    offset = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -offset)
    i_hi = min(n - 1, n - 1 - offset)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx] += (1.0 - h[idx + offset])
    grad[idx + offset] += (-h[idx])
    return grad

def active_gradient(h, dx, n_active, beta_w):
    """Compute weighted gradient of active correlations."""
    n = len(h)
    corrs = np.correlate(h, 1.0 - h, mode='full')
    max_c = corrs.max()
    # Active set: top correlations
    threshold = max_c - 0.02 * max(1, max_c)  # within 2% of max
    active = corrs >= threshold
    if active.sum() > n_active:
        # Take top n_active
        idx = np.argsort(corrs)[-n_active:]
        active = np.zeros(len(corrs), dtype=bool)
        active[idx] = True
    # Soft weights
    w = np.zeros(len(corrs))
    w[active] = np.exp(beta_w * (corrs[active] - max_c))
    w_sum = w.sum()
    if w_sum < 1e-30:
        w[active] = 1.0 / active.sum()
    else:

    # Compute gradient
    grad = np.zeros(n)
    for k in range(len(corrs)):
        if w[k] > 1e-14:
            grad += w[k] * grad_corr_k(h, k, n)
    return grad * dx, corrs, max_c * dx

        corrs = np.correlate(h, 1.0 - h, mode='full')
        max_c = corrs.max()
        c5_cur = max_c * dx

            best_h = h.copy()

        # (move mass from j to i if g_i > g_j... wait)
        # If we want to DECREASE corr, and grad of corr w.r.t. h_i is g_i,
        # then decreasing h_i (if g_i > 0) helps.
        # Pair: decrease h[j] (g_j > 0, source), increase h[i] (g_i < 0, sink)
        # Change: h[i] += t, h[j] -= t
        # Δcorr ≈ t*(g_i - g_j) -- we want this negative, so g_i < g_j

        g_min_idx = np.argmin(grad)
        g_max_idx = np.argmax(grad)
        
        if h[g_max_idx] < 1e-8 or h[g_min_idx] > 1.0 - 1e-8:
            # Can't do the best pair, try second best
            # Find valid source (high g, h > 0) and sink (low g, h < 1)
            sources = np.where(grad > 0)[0]
            sinks = np.where(grad < 0)[0]
            valid_sources = sources[sources != g_min_idx]  # h > 0 roughly
            valid_sinks = sinks[sinks != g_max_idx]
            
            # Filter by actual capacity
            valid_sources = [i for i in valid_sources if h[i] > 1e-6]
            valid_sinks = [i for i in valid_sinks if h[i] < 1.0 - 1e-6]
            
            if not valid_sources or not valid_sinks:
                step *= 0.5
                continue
            
            best_gap = -inf
            best_si, best_di = None, None
            for s in valid_sources[:10]:
                for d in valid_sinks[:10]:
                    gap = grad[d] - grad[s]  # want most negative
                    if gap < best_gap:
                        best_gap = gap
                        best_si, best_di = s, d
            if best_si is None:
                step *= 0.5
                continue
            src, dst = best_si, best_di
        else:
            src, dst = g_max_idx, g_min_idx

    n = len(h)
    h_cur = h.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    r = r0
    
    for it in range(n_iter):
        corrs = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        c5_cur = corrs.max() * dx
        
        if c5_cur < best_c5 - 1e-15:
            best_c5 = c5_cur
            best_h = h_cur.copy()
        
        # Build LP: min s
        # s.t. dx * sum(g_k[i] * eta[i]) - s <= -dx * corr_k  for all k
        #      sum(eta) = 0
        #      -h[i]*r <= eta[i] <= (1-h[i])*r
        
        n_vars = n + 1  # n eta's + 1 s
        c_obj = np.zeros(n_vars)
        c_obj[-1] = 1.0  # minimize s
        
        A_ub = []
        b_ub = []
        for k in range(2 * n - 1):
            gk = grad_corr_k(h_cur, k, n)
            row = np.zeros(n_vars)
            row[:n] = dx * gk
            row[n] = -1.0
            A_ub.append(row)
            b_ub.append(-dx * corrs[k] - 1e-10)
        
        A_eq = np.zeros((1, n_vars))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)
        
        bounds = [( -h_cur[i] * r, (1.0 - h_cur[i]) * r) for i in range(n)]
        bounds.append((0.0, None))
        
        try:
            res = linprog(c_obj, A_ub=np.array(A_ub), b_ub=np.array(b_ub),
                         A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
            if res.status != 0:
                r *= 0.5
                continue
            eta = res.x[:n]
            h_new = h_cur + eta
            h_new = project_feasible(h_new, target_sum)
            new_c5 = c5_of_h(h_new, dx)
            if new_c5 < c5_cur - 1e-14:
                h_cur = h_new

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.special import logsumexp
import time
from itertools import combinations

def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def c5_of_h(h, dx):
    return np.correlate(h, 1.0 - h, mode='full').max() * dx

def grad_corr_k(h, k, n):
    offset = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -offset)
    i_hi = min(n - 1, n - 1 - offset)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx] += (1.0 - h[idx + offset])
    grad[idx + offset] += (-h[idx])
    return grad

def slsqp_optimize(h0, dx, target_sum, beta, maxiter):
    n = len(h0)
    
    def obj(x):
        corrs = np.correlate(x, 1.0 - x, mode='full')
        return logsumexp(beta * corrs) / beta * dx
    
    def grad_obj(x):
        corrs = np.correlate(x, 1.0 - x, mode='full')
        max_c = corrs.max()
        w = np.exp(beta * (corrs - max_c))
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(len(corrs)) / len(corrs)
        else:
            w /= w_sum
        grad = np.zeros(n)
        for k in range(len(corrs)):
            if w[k] > 1e-14:
                grad += w[k] * grad_corr_k(x, k, n)
        return grad * dx
    
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0, jac=grad_obj, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        h_out = project_feasible(res.x, target_sum)
    except Exception:
        h_out = project_feasible(h0, target_sum)
    return h_out, c5_of_h(h_out, dx)

def full_lp_polish(h, dx, target_sum, n_iter=50, r0=0.3):
    n = len(h)
    h_cur = h.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    r = r0
    
    for it in range(n_iter):
        corrs = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        c5_cur = corrs.max() * dx
        if c5_cur < best_c5 - 1e-15:
            best_c5 = c5_cur
            best_h = h_cur.copy()
        
        n_vars = n + 1
        c_obj = np.zeros(n_vars)
        c_obj[-1] = 1.0
        
        A_ub_list = []
        b_ub_list = []
        for k in range(2 * n - 1):
            gk = grad_corr_k(h_cur, k, n)
            row = np.zeros(n_vars)
            row[:n] = dx * gk
            row[n] = -1.0
            A_ub_list.append(row)
            b_ub_list.append(-dx * corrs[k] - 1e-10)
        
        A_eq = np.zeros((1, n_vars))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)
        
        bounds = [(-h_cur[i] * r, (1.0 - h_cur[i]) * r) for i in range(n)]
        bounds.append((0.0, None))
        
        try:
            res = linprog(c_obj, A_ub=np.array(A_ub_list), b_ub=np.array(b_ub_list),
                         A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
            if res.status != 0:
                r *= 0.5
                if r < 1e-8:
                    break
                continue
            eta = res.x[:n]
            h_new = h_cur + eta
            h_new = project_feasible(h_new, target_sum)
            new_c5 = c5_of_h(h_new, dx)
            if new_c5 < c5_cur - 1e-14:
                h_cur = h_new
                r = max(r * 0.95, 0.05)
            else:
                r *= 0.5
                if r < 1e-8:
                    break
        except Exception:
            r *= 0.5
            if r < 1e-8:
                break
    
    return best_h, best_c5
