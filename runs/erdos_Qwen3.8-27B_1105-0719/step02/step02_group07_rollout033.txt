    h[:m] = q
    h[m:] = 1.0 - q[::-1]

    n = len(h)
    dx = 2.0 / n

def corr_vals(h):

    return np.clip(h, 0.0, 1.0)

    w = np.exp(beta * shifted)

        G[idx+d] += wk * (-h[idx])
    G *= dx
    return G[:m] - G[n-1:n-1-m:-1]

        h = make_sym_h(res.x, n)

def subgrad_step(h, n, n_act=15):
    corrs = np.correlate(h, 1.0-h, mode='full')
    max_c = corrs.max()
    active = np.where(corrs >= max_c - 1e-8)[0]
    if len(active) == 0:
        active = np.array([np.argmax(corrs)])
    if len(active) > n_act:
        idx = np.argsort(corrs[active])[::-1][:n_act]
        active = active[idx]
    dx = 2.0 / n
    G = np.zeros(n)
    for k in active:
        d = k - (n-1)
        i_lo = max(0, -d)
        i_hi = min(n-1, n-1-d)
        idx2 = np.arange(i_lo, i_hi+1)
        G[idx2] += (1.0 - h[idx2+d])
        G[idx2+d] += (-h[idx2])
    G *= dx / len(active)
    return G

import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time


def make_sym_h(q, n):
    m = n // 2
    q = np.clip(q, 0.0, 1.0)
    h = np.empty(n)
    h[:m] = q
    h[m:] = 1.0 - q[::-1]
    return h


def make_q_from_h(h, n):
    m = n // 2
    q = 0.5 * (h[:m] + 1.0 - h[n - 1:n - 1 - m:-1])
    return np.clip(q, 0.0, 1.0)


def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full').max() * dx


def corr_vals(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx


def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -2.0, 3.0
    for _ in range(60):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)


def ss_fun(x, n, beta):
    h = make_sym_h(x, n)
    corrs = corr_vals(h)
    return logsumexp(beta * corrs) / beta


def ss_jac(x, n, beta):
    h = make_sym_h(x, n)
    m = n // 2
    corrs = corr_vals(h)
    dx = 2.0 / n
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    ws = w.sum()
    if ws < 1e-30:
        w = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        w /= ws
    G = np.zeros(n)
    for k in range(2 * n - 1):
        wk = beta * w[k]
        if wk < 1e-14:
            continue
        d = k - (n - 1)
        i_lo = max(0, -d)
        i_hi = min(n - 1, n - 1 - d)
        if i_lo > i_hi:
            continue
        idx = np.arange(i_lo, i_hi + 1)
        G[idx] += wk * (1.0 - h[idx + d])
        G[idx + d] += wk * (-h[idx])
    G *= dx
    return G[:m] - G[n - 1:n - 1 - m:-1]


class SSProb:
    def __init__(self, n, beta):
        self.n = n
        self.beta = beta

    def fun(self, x):
        return ss_fun(x, self.n, self.beta)

    def jac(self, x):
        return ss_jac(x, self.n, self.beta)


def opt_lbfgs(q0, n, beta, maxiter=5000, gtol=1e-12):
    m = n // 2
    p = SSProb(n, beta)
    try:
        res = minimize(p.fun, q0, jac=p.jac, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * m,
                       options={'maxiter': maxiter, 'gtol': gtol, 'ftol': 1e-15,
                                'maxfun': maxiter * 10})
        h = make_sym_h(res.x, n)
        return res.x, exact_score(h)
    except Exception:
        return None, None


def subgrad_step(h, n, n_act=20):
    corrs = np.correlate(h, 1.0 - h, mode='full')
    max_c = corrs.max()
    active = np.where(corrs >= max_c - 1e-8)[0]
    if len(active) == 0:
        active = np.array([np.argmax(corrs)])
    if len(active) > n_act:
        idx = np.argsort(corrs[active])[::-1][:n_act]
        active = active[idx]
    dx = 2.0 / n
    G = np.zeros(n)
    for k in active:
        d = k - (n - 1)
        i_lo = max(0, -d)
        i_hi = min(n - 1, n - 1 - d)
        idx2 = np.arange(i_lo, i_hi + 1)
        G[idx2] += (1.0 - h[idx2 + d])
        G[idx2 + d] += (-h[idx2])
    G *= dx / len(active)
    return G


def subgrad_polish_sym(q, n, n_steps=200, alpha_init=0.015):
    alpha = alpha_init
    m = n // 2
    h = make_sym_h(q, n)
    c5_cur = exact_score(h)
    for _ in range(n_steps):
        G = subgrad_step(h, n)
        grad_q = G[:m] - G[n - 1:n - 1 - m:-1]
        q_new = np.clip(q - alpha * grad_q, 0.0, 1.0)
        h_new = make_sym_h(q_new, n)
        c5_new = exact_score(h_new)
        if c5_new < c5_cur - 1e-13:
            q = q_new
            h = h_new
            c5_cur = c5_new
            alpha = min(alpha * 1.05, 0.05)
        else:
            alpha *= 0.6
        if alpha < 1e-10:
            break
    return q, c5_cur


def subgrad_polish_nonsym(h, n, n_steps=200, alpha_init=0.01):
    alpha = alpha_init
    c5_cur = exact_score(h)
    target_sum = n / 2.0
    for _ in range(n_steps):
        G = subgrad_step(h, n)
        h_new = project_feasible(h - alpha * G, target_sum)
        c5_new = exact_score(h_new)
        if c5_new < c5_cur - 1e-13:
            h = h_new
            c5_cur = c5_new
            alpha = min(alpha * 1.05, 0.05)
        else:
            alpha *= 0.6
        if alpha < 1e-10:
            break
    return h, c5_cur


def coord_descent_sym(q, n, n_sweeps=3, n_trials=8):
    m = n // 2
    h = make_sym_h(q, n)
    best_c5 = exact_score(h)
    for _ in range(n_sweeps):
        improved = False
        for i in range(m):
            q_old = q[i]
            best_qi = q_old
            best_val = best_c5
            for j in range(n_trials):
                frac = j / max(n_trials - 1, 1)
                delta = (frac - 0.5) * 0.1
                qi = q_old + delta
                if qi < 0 or qi > 1:
                    continue
                q[i] = qi
                h = make_sym_h(q, n)
                c5 = exact_score(h)
                if c5 < best_val - 1e-13:
                    best_val = c5
                    best_qi = qi
            if best_qi != q_old:
                q[i] = best_qi
                best_c5 = best_val
                improved = True
        if not improved:
            break
    return q, best_c5


def full_corrs_grad(h, n, weights):
    dx = 2.0 / n
    G = np.zeros(n)
    for k in range(2 * n - 1):
        wk = weights[k]
        if wk < 1e-14:
            continue
        d = k - (n - 1)
        i_lo = max(0, -d)
        i_hi = min(n - 1, n - 1 - d)
        if i_lo > i_hi:
            continue
        idx = np.arange(i_lo, i_hi + 1)
        G[idx] += wk * (1.0 - h[idx + d])
        G[idx + d] += wk * (-h[idx])
    G *= dx
    return G


def line_search_sym(q, n, direction, n_iter=40):
    m = n // 2
    best_t = 0.0
    h0 = make_sym_h(q, n)
    best_f = exact_score(h0)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = -1.0, 1.0
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    hc = make_sym_h(np.clip(q + c * direction, 0, 1), n)
    hd = make_sym_h(np.clip(q + d * direction, 0, 1), n)
    fc = exact_score(hc)
    fd = exact_score(hd)
    for _ in range(n_iter):
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            hc = make_sym_h(np.clip(q + c * direction, 0, 1), n)
            fc = exact_score(hc)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            hd = make_sym_h(np.clip(q + d * direction, 0, 1), n)
            fd = exact_score(hd)
        t_mid = (a + b) / 2.0
        hm = make_sym_h(np.clip(q + t_mid * direction, 0, 1), n)
        fm = exact_score(hm)
        if fm < best_f:
            best_f = fm
            best_t = t_mid
    if best_f < exact_score(h0):
        q_new = np.clip(q + best_t * direction, 0.0, 1.0)
        return q_new, best_f
    return q, best_f
