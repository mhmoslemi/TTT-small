def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def compute_profile(h):
    n = len(h)

    return np.correlate(h, 1 - h, mode="full") * dx

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx])
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-14:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-14:
                break
    # Final fix
    diff2 = target - h.sum()
    if abs(diff2) > 1e-13:
        if diff2 > 0:
            idx = np.argmax(1.0 - h)
        else:
            idx = np.argmax(h)
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def mass_interp(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    # Compute integration intervals
    edges_old = np.linspace(0, 2.0, n + 1)
    edges_new = np.linspace(0, 2.0, new_n + 1)
    h_new = np.zeros(new_n)
    for k in range(new_n):
        lo = edges_new[k]
        hi = edges_new[k + 1]
        # Find overlapping old intervals
        # Old interval j covers [edges_old[j], edges_old[j+1])
        # We need the integral of h over [lo, hi]
        # h is piecewise constant on old intervals
        total = 0.0
        width = hi - lo
        for j in range(n):
            ov_lo = max(lo, edges_old[j])
            ov_hi = min(hi, edges_old[j + 1])
            if ov_hi > ov_lo:
                total += h[j] * (ov_hi - ov_lo)
        h_new[k] = total / width if width > 0 else 0.0
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)
