Now let me think about the symmetric LP more carefully.

In the symmetric case, h[i] = 1 - h[n-1-i] for all i. So we only need to specify h[0], ..., h[m-1] where m = n/2. Then h[m], ..., h[n-1] are determined.

The sum constraint: Σ h[i] = m (which is n/2). Let's verify:
Σ_{i=0}^{n-1} h[i] = Σ_{i=0}^{m-1} h[i] + Σ_{i=m}^{n-1} h[i]
= Σ_{i=0}^{m-1} h[i] + Σ_{i=m}^{n-1} (1 - h[n-1-i])
= Σ_{i=0}^{m-1} h[i] + Σ_{j=0}^{m-1} (1 - h[j])  [substituting j = n-1-i]
= Σ_{i=0}^{m-1} h[i] + m - Σ_{i=0}^{m-1} h[i]
= m = n/2 ✓

So the sum constraint is automatically satisfied.

For the gradient in the symmetric parameterization:
C_k = Σ_{valid i} h[i+k]*(1-h[i])

In terms of q (where h[i] = q[i] for i < m, h[i] = 1-q[n-1-i] for i ≥ m):

The gradient ∂C_k/∂q[j] for j < m:
= ∂C_k/∂h[j] + (contribution from h[n-1-j] = 1-q[j])
= ∂C_k/∂h[j] - ∂C_k/∂h[n-1-j]

So in the symmetric LP, the gradient for variable q[j] is:
g_sym[j] = g_full[j] - g_full[n-1-j]

where g_full is the full gradient.

import numpy as np
import time
from scipy.optimize import linprog

def eval_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def project_to_target(x, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(x - mid, 0.0, 1.0).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-10:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.any():
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def build_sym(q, n):
    m = n // 2
    h = np.empty(n)
    h[:m] = q
    h[m:] = 1.0 - q[::-1]
    return h

def to_sym_q(h, n):
    m = n // 2
    return 0.5 * (h[:m] + 1.0 - h[n-1:m-1:-1])

def grad_lag_full(h, k, n):
    g = np.zeros(n)
    if k >= 0:
        g[k:] += 1.0 - h[:n-k]
        g[:n-k] += -h[k:]
    else:
        kk = -k
        g[:n-kk] += 1.0 - h[kk:]
        g[kk:] += -h[:n-kk]
    return g

def corr_values(h, n):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def select_active(cv, margin=0.01, max_count=50):
    mx = cv.max()
    thresh = mx - max(margin * mx, 1e-9)
    idx = np.where(cv >= thresh)[0]
    if len(idx) > max_count:
        # Keep only the top max_count
        order = np.argsort(cv[idx])[::-1][:max_count]
        idx = idx[order]
    return idx

def lp_solve_sym(q0, n, active_idx, c5_cur):
    """Solve LP in symmetric parameterization.
    q0: current q (length m)
    active_idx: indices in correlation array
    Returns new q or None if failed.
    """
    m = n // 2
    h0 = build_sym(q0, n)
    
    # Build the LP
    # Variables: q[0..m-1], t  (m+1 variables)
    c_obj = np.zeros(m + 1)
    c_obj[m] = 1.0  # minimize t
    
    A_ub_rows = []
    b_ub_vals = []
    
    for idx in active_idx:
        k = idx - (n - 1)
        # Compute C_k(h0)
        g_full = grad_lag_full(h0, k, n)
        # Symmetric gradient: g_sym[j] = g_full[j] - g_full[n-1-j]
        g_sym = g_full[:m] - g_full[n-1:m-1:-1]
        
        # C_k(h0) value
        cv = corr_values(h0, n)
        Cv = cv[idx]
        
        # Linearized: Cv + g_sym^T (q - q0) <= t
        # g_sym^T q - t <= Cv - g_sym^T q0
        rhs = Cv - np.dot(g_sym, q0)
        
        row = np.zeros(m + 1)
        row[:m] = g_sym
        row[m] = -1.0
        A_ub_rows.append(row)
        b_ub_vals.append(rhs)
    
    if not A_ub_rows:
        return None
    
    A_ub = np.array(A_ub_rows)
    b_ub = np.array(b_ub_vals)
    
    bounds = [(0.0, 1.0)] * m + [(0.0, None)]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.clip(res.x[:m], 0.0, 1.0)
    except Exception:
        pass
    return None

def lp_solve_full(h0, n, active_idx, c5_cur):
    """Solve LP in full parameterization."""
    # Variables: h[0..n-1], t  (n+1 variables)
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    A_ub_rows = []
    b_ub_vals = []
    cv = corr_values(h0, n)
    
    for idx in active_idx:
        k = idx - (n - 1)
        g = grad_lag_full(h0, k, n)
        Cv = cv[idx]
        rhs = Cv - np.dot(g, h0)
        
        row = np.zeros(n + 1)
        row[:n] = g
        row[n] = -1.0
        A_ub_rows.append(row)
        b_ub_vals.append(rhs)
    
    if not A_ub_rows:
        return None
    
    A_ub = np.array(A_ub_rows)
    b_ub = np.array(b_ub_vals)
    
    # Equality: sum(h) = n/2
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,

            h_sol = project_to_target(h_sol, n / 2.0)
            return h_sol
    except Exception:
        pass
    return None
