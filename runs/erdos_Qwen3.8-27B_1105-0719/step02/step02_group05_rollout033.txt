from scipy.optimize import minimize
from scipy.special import logsumexp
import time


def build_h_from_q(q, n):
    m = n // 2
    return np.concatenate([q, 1.0 - q[::-1]])


def q_from_h(h):
    return h[:len(h)//2].copy()


def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return compute_corrs(h).max() * dx


def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def coord_desc_q(q, n, max_passes=500):
    m = n // 2
    q = np.clip(q.copy(), 0.0, 1.0)
    h = build_h_from_q(q, n)

    for p in range(max_passes):
        max_change = 0.0
        for i in range(m):
            h0 = h.copy()
            h0[i] = 0.0
            h0[n - 1 - i] = 1.0
            h1 = h.copy()
            h1[i] = 1.0
            h1[n - 1 - i] = 0.0

            c0 = np.correlate(h0, 1.0 - h0, mode='full')
            c1 = np.correlate(h1, 1.0 - h1, mode='full')

            t_new = solve_1d_coord(c1 - c0, c0)
            change = abs(t_new - q[i])
            if change > max_change:
                max_change = change
            q[i] = t_new
            h[i] = t_new
            h[n - 1 - i] = 1.0 - t_new

        if max_change < 1e-14:
            break

    return q


def smooth_val_grad_q(q, n, beta):
    m = n // 2
    h = build_h_from_q(q, n)
    corrs = compute_corrs(h)
    val = logsumexp(beta * corrs) / beta * (2.0 / n)
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(len(corrs)) / len(corrs)
    else:
        w /= w_sum
    grad = np.zeros(m)
    for k in range(len(corrs)):
        if w[k] < 1e-14:
            continue
        d = k - (n - 1)
        G = np.zeros(n)
        if d >= 0:
            G[:n - d] += (1.0 - h[d:])
            G[d:] += (-h[:n - d])
        else:
            dd = -d
            G[dd:] += (1.0 - h[:n - dd])
            G[:n - dd] += (-h[dd:])
        grad += w[k] * (G[:m] - G[m:][::-1])
    return val, grad * (2.0 / n)

            return smooth_val_grad_q(x, n, beta)[1]

    x_old = np.linspace(0.0, 2.0, n_old)
    x_new = np.linspace(0.0, 2.0, n_new)
    h_new = np.interp(x_new, x_old, h)
    return h_new
