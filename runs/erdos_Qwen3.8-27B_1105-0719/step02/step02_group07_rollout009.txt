from scipy.optimize import minimize
from scipy.special import logsumexp
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full').max() * dx

def corr_vals(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def make_sym_h(q, n):
    m = n // 2
    q = np.clip(q, 0.0, 1.0)
    h = np.empty(n)
    h[:m] = q
    h[m:] = 1.0 - q[::-1]
    return h

def make_q_from_h(h, n):
    m = n // 2
    q = 0.5 * (h[:m] + 1.0 - h[n - 1: n - 1 - m: -1])
    return np.clip(q, 0.0, 1.0)

def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    return np.clip(h, 0.0, 1.0)

def weighted_grad_full(h, n, dx, weights):
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = weights[k]
        if wk < 1e-15:
            continue
        d = k - (n - 1)
        i_lo = max(0, -d)
        i_hi = min(n - 1, n - 1 - d)
        if i_lo > i_hi:
            continue
        idx = np.arange(i_lo, i_hi + 1)
        grad[idx] += wk * (1.0 - h[idx + d])
        grad[idx + d] += wk * (-h[idx])
    return grad * dx

def lse_smooth(q, n, beta):
    h = make_sym_h(q, n)
    corrs = corr_vals(h)
    return logsumexp(beta * corrs) / beta

def lse_smooth_grad(q, n, beta):
    m = n // 2
    dx = 2.0 / n
    h = make_sym_h(q, n)
    corrs = corr_vals(h)
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    ws = w.sum()
    if ws < 1e-30:
        w = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        w /= ws
    G = weighted_grad_full(h, n, dx, beta * w)
    return G[:m] - G[n - 1: n - 1 - m: -1]

def lse_optimize(q, n, beta, maxiter=300):
    m = n // 2
    try:
        res = minimize(lse_smooth, q,
                       jac=lambda x: lse_smooth_grad(x, n, beta),
                       method='SLSQP',
                       bounds=[(0.0, 1.0)] * m,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        h_new = make_sym_h(res.x, n)
        c5 = exact_score(h_new)
        return res.x, c5
    except Exception:
        return None, None

def epigraph_solve(q, n, active_lags, t_init, maxiter=300):
    m = n // 2
    dx = 2.0 / n
    L = len(active_lags)
    
    def obj(x):
        return x[0]
    
    def obj_jac(x):
        g = np.zeros(m + 1)
        g[0] = 1.0
        return g
    
    def con_fun(x):
        t = x[0]
        qv = x[1:]
        h = make_sym_h(qv, n)
        corrs = corr_vals(h)
        return t - corrs[active_lags]
    
    def con_jac(x):
        qv = x[1:]
        h = make_sym_h(qv, n)
        n_lags = 2 * n - 1
        J = np.zeros((L, m + 1))
        J[:, 0] = 1.0
        for i, lag in enumerate(active_lags):
            d = lag - (n - 1)
            i_lo = max(0, -d)
            i_hi = min(n - 1, n - 1 - d)
            G = np.zeros(n)
            if i_lo <= i_hi:
                idx = np.arange(i_lo, i_hi + 1)
                G[idx] += (1.0 - h[idx + d])
                G[idx + d] += (-h[idx])
            G *= dx
            J[i, 1:] = -(G[:m] - G[n - 1: n - 1 - m: -1])
        return J
    
    x0 = np.concatenate([[t_init], q])
    cons = [{'type': 'ineq', 'fun': con_fun, 'jac': con_jac}]
    bounds = [(0.0, 2.0)] + [(0.0, 1.0)] * m
    
    try:

                       bounds=bounds, constraints=cons,

import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time


def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full').max() * dx


def corr_vals(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx


def make_sym_h(q, n):
    m = n // 2
    q = np.clip(q, 0.0, 1.0)
    h = np.empty(n)
    h[:m] = q
    h[m:] = 1.0 - q[::-1]
    return h


def make_q_from_h(h, n):
    m = n // 2
    q = 0.5 * (h[:m] + 1.0 - h[n - 1: n - 1 - m: -1])
    return np.clip(q, 0.0, 1.0)


def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    return np.clip(h, 0.0, 1.0)


def weighted_grad_full(h, n, dx, weights):
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = weights[k]
        if wk < 1e-15:
            continue
        d = k - (n - 1)
        i_lo = max(0, -d)
        i_hi = min(n - 1, n - 1 - d)
        if i_lo > i_hi:
            continue
        idx = np.arange(i_lo, i_hi + 1)
        grad[idx] += wk * (1.0 - h[idx + d])
        grad[idx + d] += wk * (-h[idx])
    return grad * dx


def lse_obj(q, n, beta):
    h = make_sym_h(q, n)
    corrs = corr_vals(h)
    return logsumexp(beta * corrs) / beta


def lse_grad(q, n, beta):
    m = n // 2
    dx = 2.0 / n
    h = make_sym_h(q, n)
    corrs = corr_vals(h)
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    ws = w.sum()
    if ws < 1e-30:
        w = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        w /= ws
    G = weighted_grad_full(h, n, dx, beta * w)
    return G[:m] - G[n - 1: n - 1 - m: -1]


def lse_optimize(q, n, beta, maxiter=300):
    m = n // 2
    try:
        res = minimize(lse_obj, q, jac=lambda x: lse_grad(x, n, beta),
                       method='SLSQP', bounds=[(0.0, 1.0)] * m,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        h_new = make_sym_h(res.x, n)
        c5 = exact_score(h_new)
        return res.x, c5
    except Exception:
        return None, None


def epigraph_solve(q, n, active_lags, t_init, maxiter=300):
    m = n // 2
    dx = 2.0 / n
    L = len(active_lags)

    def epi_obj(x):
        return x[0]

    def epi_obj_jac(x):
        g = np.zeros(m + 1)
        g[0] = 1.0
        return g

    def epi_con_fun(x):
        t = x[0]
        qv = x[1:]
        h = make_sym_h(qv, n)
        corrs = corr_vals(h)
        return t - corrs[active_lags]

    def epi_con_jac(x):
        qv = x[1:]
        h = make_sym_h(qv, n)
        J = np.zeros((L, m + 1))
        J[:, 0] = 1.0
        for i, lag in enumerate(active_lags):
            d = lag - (n - 1)
            i_lo = max(0, -d)
            i_hi = min(n - 1, n - 1 - d)
            G = np.zeros(n)
            if i_lo <= i_hi:
                idx = np.arange(i_lo, i_hi + 1)
                G[idx] += (1.0 - h[idx + d])
                G[idx + d] += (-h[idx])
            G *= dx
            J[i, 1:] = -(G[:m] - G[n - 1: n - 1 - m: -1])
        return J

    x0 = np.concatenate([[t_init], q])
    cons = [{'type': 'ineq', 'fun': epi_con_fun, 'jac': epi_con_jac}]
    bounds = [(0.0, 2.0)] + [(0.0, 1.0)] * m
    try:
        res = minimize(epi_obj, x0, jac=epi_obj_jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        h_new = make_sym_h(res.x[1:], n)
        c5 = exact_score(h_new)
        if abs(h_new.sum() - n / 2.0) > 1e-8:

        return None, None

            q_best = q_final
            h_best = make_sym_h(q_best, n)
            print(f"  Final CD: C5={c5_best:.7f}")

    # Phase 8: Full-space (non-symmetric) polish
    if remaining() > 20:
        h_full = h_best.copy()
        h_new, c5_new = full_smooth_optimize(h_full, n, 500, maxiter=300)
        h_new = project_feasible(h_new, n / 2.0)
        c5_new = exact_score(h_new)
        if c5_new < c5_best - 1e-12:
            c5_best = c5_new
            h_best = h_new
            print(f"  Full-space: C5={c5_best:.7f}")

    # Final validation
    h_best = np.clip(h_best, 0.0, 1.0)
    h_best = project_feasible(h_best, len(h_best) / 2.0)
    c5_best = exact_score(h_best)
    print(f"Final: n={len(h_best)}, C5={c5_best:.7f}")
    return h_best, c5_best, len(h_best)
