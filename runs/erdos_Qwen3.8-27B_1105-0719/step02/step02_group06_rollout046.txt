At point h, for each lag r, the correlation is:
R_r(h) = sum_i h[i] * (1 - h[i+d]) where d = r - (n-1)

The gradient w.r.t. h[i] is:
- For i in the "first" array part: ∂R_r/∂h[i] = 1 - h[i+d]
- For i in the "second" array part: ∂R_r/∂h[i] = -h[i-d]

The linearized constraint is:
R_r(h + p) ≈ R_r(h) + grad_r · p ≤ t

minimize t
subject to: grad_r · p + R_r(h) ≤ t  for all active r
         0 ≤ h[i] + p[i] ≤ 1  for all i

def eval_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def corr_raw(h):
    return np.correlate(h, 1.0 - h, mode='full')

def build_sym_h(q):
    m = len(q)
    h = np.empty(2 * m)
    h[:m] = q
    h[m:] = 1.0 - q[::-1]
    return h

def sym_from_h(h):
    n = len(h)
    m = n // 2
    q = np.empty(m)
    for i in range(m):
        q[i] = 0.5 * (h[i] + (1.0 - h[n - 1 - i]))

def project_feasible(x, target_sum):
    n = len(x)
    x = np.clip(x, 0.0, 1.0)
    # Bisection for uniform shift
    lo = -2.0
    hi = 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(x + mid, 0.0, 1.0).sum()
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    h = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(h)
            for idx in order:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
                if diff < 1e-13:

        else:

                sub = min(-diff, h[idx])

def corr_grad_k_full(h, k):

def tangent_lp_sym(q, n_full, active_k, trust_rho):
    """
    Tangent LP for symmetric (antiparallel) case.
    Minimizes t subject to linearized correlation constraints.
    Returns direction p (for q) and the LP objective value.
    """
    m = n_full // 2
    h = build_sym_h(q)
    dx = 2.0 / n_full
    corrs = corr_raw(h)
    
    # Build constraint matrix
    num_constraints = len(active_k)
    A = np.zeros((num_constraints, m + 1))
    b = np.zeros(num_constraints)
    
    for row, k in enumerate(active_k):
        g_h = corr_grad_k_full(h, k)
        g_q = np.zeros(m)
        for r in range(m):
            g_q[r] = g_h[r] - g_h[n_full - 1 - r]
        A[row, :m] = g_q
        A[row, m] = -1.0  # -t term
        b[row] = -corrs[k]  # grad·p - t <= -R_k => grad·p - t + R_k <= 0
    
    # Objective: minimize t
    c_vec = np.zeros(m + 1)
    c_vec[m] = 1.0
    
    # Bounds: q + p in [0,1], ||p||_inf <= rho
    # So p[i] in [-q[i], 1-q[i]] intersected with [-rho, rho]
    lb = np.maximum(-q, -trust_rho)
    ub = np.minimum(1.0 - q, trust_rho)
    bounds = list(zip(lb, ub))
    bounds.append((0.0, corrs.max() + 1.0))
    
    try:
        res = linprog(c_vec, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.status == 0:
            p = res.x[:m]
            t_val = res.x[m]
            return p, t_val
    except Exception:
        pass
    return np.zeros(m), corrs.max()

def transfer_line_search(h, a, b, max_evals=40):
    """
    1D line search transferring mass between indices a and b.
    h[a] += delta, h[b] -= delta (preserves sum).
    """
    n = len(h)
    # Valid delta range
    delta_min = max(-h[a], h[b] - 1.0)
    delta_max = min(1.0 - h[a], -h[b])
    if delta_max < delta_min:
        return h.copy()
    
    best_delta = 0.0
    best_score = eval_score(h)
    best_h = h.copy()
    
    grid = np.linspace(delta_min, delta_max, min(max_evals, 30))
    for d in grid:
        h_try = h.copy()
        h_try[a] += d
        h_try[b] -= d
        if 0 <= h_try[a] <= 1 and 0 <= h_try[b] <= 1:
            s = eval_score(h_try)
            if s < best_score - 1e-14:
                best_score = s
                best_delta = d
                best_h = h_try.copy()
    
    # Refine
    if best_delta != 0 or True:
        step = (delta_max - delta_min) / 200.0
        lo = max(delta_min, best_delta - 3*step)
        hi = min(delta_max, best_delta + 3*step)
        for d in np.linspace(lo, hi, 20):
            h_try = h.copy()
            h_try[a] += d
            h_try[b] -= d
            if 0 <= h_try[a] <= 1 and 0 <= h_try[b] <= 1:
                s = eval_score(h_try)
                if s < best_score - 1e-14:
                    best_score = s
                    best_h = h_try.copy()
    
    return best_h

def transfer_to_n(h_old, n_new):

    x_old = (np.arange(n_old) + 0.5) * 2.0 / n_old

def smooth_max_sym(q, n_full, beta):
    """Smooth max objective and gradient for symmetric case."""
    m = n_full // 2
    h = build_sym_h(q)
    corrs = corr_raw(h) * (2.0 / n_full)
    max_c = corrs.max()
    shifted = corrs - max_c
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(len(corrs)) / len(corrs)
    else:
        w /= w_sum
    val = max_c + np.log(w_sum) / beta
    
    grad_q = np.zeros(m)
    dx = 2.0 / n_full
    for k in range(2 * n_full - 1):
        if w[k] < 1e-12:
            continue
        g_h = corr_grad_k_full(h, k)
        for r in range(m):
            grad_q[r] += w[k] * (g_h[r] - g_h[n_full - 1 - r])
    grad_q *= dx
    return val, grad_q

def gradient_descent_sym(q, n_full, lr=0.01, max_iter=500, time_limit=50.0):
    """Simple gradient descent with the smooth max objective."""
    m = n_full // 2
    best_q = q.copy()
    best_score = eval_score(build_sym_h(q))
    beta = 200.0
    
    for i in range(max_iter):
        if time.time() > time_limit:
            break
        val, grad = smooth_max_sym(best_q, n_full, beta)
        if np.all(abs(grad) < 1e-10):
            break
        # Gradient step with backtracking
        alpha = lr
        for _ in range(20):
            q_try = np.clip(best_q - alpha * grad, 0, 1)
            s = eval_score(build_sym_h(q_try))
            if s <= val + 1e-14:
                break
            alpha *= 0.5
        if alpha < 1e-8:
            break
        q_try = np.clip(best_q - alpha * grad, 0, 1)
        s = eval_score(build_sym_h(q_try))
        if s < best_score:
            best_score = s
            best_q = q_try
    return best_q, best_score

        if score < best_score - 1e-15:
            improvement = best_score - score
            best_h = h_new.copy()
            best_score = score
            trust_rho = min(1.0, trust_rho * 1.4)
            corrs_new = corr_raw(best_h)
            max_act = corrs_new[active].max()
            hidden = np.where(corrs_new > max_act + 1e-9)[0]
            if len(hidden) > 0:
                active = sorted(set(active + list(hidden)))
                if len(active) > 90:
                    active = select_active(corrs_new, margin_frac=0.02, neighbor_pad=4, max_k=90)
        else:
            trust_rho *= 0.6
            if trust_rho < 0.005:
                break
    return best_h, best_score

def smooth_max_sym_val_grad(q, n_full, beta):
    m = n_full // 2
    h = build_sym_h(q)
    dx = 2.0 / n_full
    corrs = corr_raw(h) * dx
    max_c = corrs.max()
    shifted = corrs - max_c
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(len(corrs)) / len(corrs)
    else:
        w /= w_sum
    val = max_c + np.log(w_sum) / beta
    grad_q = np.zeros(m)
    for k in range(2 * n_full - 1):
        if w[k] < 1e-12:
            continue
        g_h = corr_grad_k_full(h, k)
        for r in range(m):
            grad_q[r] += w[k] * (g_h[r] - g_h[n_full - 1 - r])
    grad_q *= dx
    return val, grad_q

def smooth_sym_optimize(q, n_full, beta_list, maxiter_per_beta, time_limit):
    m = n_full // 2
    best_q = np.clip(q.copy(), 0, 1)
    best_score = eval_score(build_sym_h(best_q))
    for beta in beta_list:
        if time.time() > time_limit:
            break
        bounds = [(0.0, 1.0)] * m
        def obj_fn(xx):
            return smooth_max_sym_val_grad(xx, n_full, beta)[0]
        def jac_fn(xx):
            return smooth_max_sym_val_grad(xx, n_full, beta)[1]
        try:
            res = minimize(obj_fn, best_q, jac=jac_fn, method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxiter_per_beta, 'ftol': 1e-15, 'gtol': 1e-10})
            q_new = np.clip(res.x, 0, 1)
        except Exception:
            q_new = best_q.copy()
        h_new = build_sym_h(q_new)
        sc = eval_score(h_new)
        if sc < best_score:
            best_score = sc
            best_q = q_new.copy()
    return best_q, best_score

def transfer_to_n(h_old, n_new):
    n_old = len(h_old)
    x_old = (np.arange(n_old) + 0.5) * 2.0 / n_old
    x_new = (np.arange(n_new) + 0.5) * 2.0 / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new / 2.0)

def make_binary_start(n, rng):
    h = np.zeros(n)
    n_ones = n // 2
    idx = rng.permutation(n)
    h[idx[:n_ones]] = 1.0
    return h

def diversity_coord_sym(q, n_full, rng, num_moves, time_limit):
    m = len(q)
    best_q = q.copy()
    best_score = eval_score(build_sym_h(q))
    for move in range(num_moves):
        if time.time() > time_limit:
            break
        r = rng.randint(m)
        lo = max(0.0, best_q[r] - 0.3)
        hi = min(1.0, best_q[r] + 0.3)
        grid = np.linspace(lo, hi, 15)
        for v in grid:
            q_try = best_q.copy()
            q_try[r] = v
            s = eval_score(build_sym_h(q_try))
            if s < best_score - 1e-15:
                best_score = s
                best_q = q_try.copy()
    return best_q, best_score

def diversity_transfer_full(h, rng, num_moves, time_limit):
    n = len(h)
    best_h = h.copy()
    best_score = eval_score(h)
    for move in range(num_moves):
        if time.time() > time_limit:
            break
        a = rng.randint(n)
        b = rng.randint(n)
        if a == b:
            continue
        d_min = max(-best_h[a], best_h[b] - 1.0)
        d_max = min(1.0 - best_h[a], -best_h[b])
        if d_max < d_min:
            continue
        for d in np.linspace(d_min, d_max, 12):
            h_try = best_h.copy()
            h_try[a] += d
            h_try[b] -= d
            if 0 <= h_try[a] <= 1 and 0 <= h_try[b] <= 1:
                s = eval_score(h_try)
                if s < best_score - 1e-15:
                    best_score = s
                    best_h = h_try.copy()
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')

    try:
        h_parent = np.array(initial_h_values, dtype=float).copy().reshape(-1)
    except (NameError, AttributeError):
        h_parent = None

    if h_parent is not None and len(h_parent) > 10:
        n_p = len(h_parent)
        if abs(np.sum(h_parent) - n_p / 2.0) > 0.5:
            h_parent = project_feasible(h_parent, n_p / 2.0)
        h_parent = np.clip(h_parent, 0.0, 1.0)
        h_parent = project_feasible(h_parent, n_p / 2.0)
        best_h = h_parent.copy()
        best_c5 = eval_score(h_parent)
    else:
        m_p = 63
        x = np.linspace(0, 1, m_p)
        q_p = np.clip(0.5 + 0.3 * np.sin(2 * np.pi * x), 0, 1)
        best_h = build_sym_h(q_p)
        best_c5 = eval_score(best_h)

    n_cur = len(best_h)

    def time_left():
        return budget_s - (time.time() - t0)

    # Phase 1: LP optimization on parent
    if n_cur % 2 == 0 and time_left() > 120:
        q = sym_from_h(best_h)
        tl = time.time() + min(140, time_left() * 0.3)
        q_opt, c5_s = lp_sym_optimize(q, n_cur, max_rounds=50, time_limit=tl)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = build_sym_h(q_opt).copy()

    # Phase 2: Smooth max optimization
    if n_cur % 2 == 0 and time_left() > 100:
        q = sym_from_h(best_h)
        tl = time.time() + min(80, time_left() * 0.2)
        q_opt, c5_g = smooth_sym_optimize(q, n_cur, [100, 250, 500, 1000], 200, tl)
        if c5_g < best_c5:
            best_c5 = c5_g
            best_h = build_sym_h(q_opt).copy()

    # Phase 3: Another LP round
    if n_cur % 2 == 0 and time_left() > 80:
        q = sym_from_h(best_h)
        tl = time.time() + min(100, time_left() * 0.25)
        q_opt, c5_s = lp_sym_optimize(q, n_cur, max_rounds=40, time_limit=tl)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = build_sym_h(q_opt).copy()

    # Phase 4: Full vector LP
    if time_left() > 80:
        tl = time.time() + min(70, time_left() * 0.2)
        h_opt, c5_f = lp_full_optimize(best_h, max_rounds=20, time_limit=tl)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_opt.copy()

    # Phase 5: Multi-resolution
    n_base = len(best_h)
    n_candidates = set()
    for delta in [-8, -6, -4, -3, -2, -1, 1, 2, 3, 4, 6, 8]:
        nc = n_base + delta
        if 48 <= nc <= 200:
            n_candidates.add(nc)
    if n_base // 2 >= 32:
        n_candidates.add(n_base // 2)
    n_candidates = sorted(n_candidates)

    for n_try in n_candidates:
        if time_left() < 40:
            break
        h_try = transfer_to_n(best_h, n_try)
        n_try = len(h_try)
        if n_try % 2 == 0:
            q_try = sym_from_h(h_try)
            if time_left() > 60:
                tl = time.time() + min(45, time_left() * 0.3)
                q_opt, c5_s = lp_sym_optimize(q_try, n_try, max_rounds=25, time_limit=tl)
                if c5_s < best_c5:
                    best_c5 = c5_s
                    best_h = build_sym_h(q_opt).copy()
            if time_left() > 30:
                tl = time.time() + min(25, time_left() * 0.2)
                q_try2 = sym_from_h(best_h)
                q_opt2, c5_g = smooth_sym_optimize(q_try2, n_try, [200, 500], 150, tl)
                if c5_g < best_c5:
                    best_c5 = c5_g
                    best_h = build_sym_h(q_opt2).copy()
        else:
            if time_left() > 50:
                tl = time.time() + min(35, time_left() * 0.25)
                h_opt, c5_f = lp_full_optimize(h_try, max_rounds=15, time_limit=tl)
                if c5_f < best_c5:
                    best_c5 = c5_f
                    best_h = h_opt.copy()

    # Phase 6: Diversity moves
    if len(best_h) % 2 == 0 and time_left() > 40:
        q = sym_from_h(best_h)
        tl = time.time() + min(30, time_left() * 0.2)
        q_opt, c5_d = diversity_coord_sym(q, len(best_h), rng, 10, tl)
        if c5_d < best_c5:
            best_c5 = c5_d
            best_h = build_sym_h(q_opt).copy()

    if time_left() > 40:
        tl = time.time() + min(30, time_left() * 0.2)
        h_opt, c5_d = diversity_transfer_full(best_h, rng, 12, tl)
        if c5_d < best_c5:
            best_c5 = c5_d
            best_h = h_opt.copy()

    # Phase 7: Random restarts
    n_restarts = min(6, max(2, int(time_left() / 90)))
    for i in range(n_restarts):
        if time_left() < 50:
            break
        n_r = n_cur if i % 3 == 0 else max(56, n_cur + int(rng.choice([-6, -3, 3, 6])))
        n_r = min(180, max(50, n_r))
        if n_r % 2 == 0:
            m_r = n_r // 2
            phase = rng.rand() * 6.28
            amp = 0.2 + 0.2 * rng.rand()
            q_r = np.clip(0.5 + amp * np.sin(2 * np.pi * np.linspace(0, 1, m_r) + phase), 0, 1)
            if time_left() > 70:
                tl = time.time() + min(50, time_left() * 0.4)
                q_opt, c5_s = lp_sym_optimize(q_r, n_r, max_rounds=20, time_limit=tl)
                if c5_s < best_c5:
                    best_c5 = c5_s
                    best_h = build_sym_h(q_opt).copy()
        else:
            h_r = make_binary_start(n_r, rng)
            h_r = 0.3 + 0.4 * h_r + 0.12 * rng.randn(n_r)
            h_r = project_feasible(h_r, n_r / 2.0)
            if time_left() > 50:
                tl = time.time() + min(40, time_left() * 0.3)
                h_opt, c5_f = lp_full_optimize(h_r, max_rounds=12, time_limit=tl)
                if c5_f < best_c5:
                    best_c5 = c5_f
                    best_h = h_opt.copy()

    # Phase 8: Final polish
    if time_left() > 25 and best_h is not None:
        n_fin = len(best_h)
        if n_fin % 2 == 0:
            q_fin = sym_from_h(best_h)
            if time_left() > 35:
                tl = time.time() + min(25, time_left() * 0.5)
                q_o, c5_o = lp_sym_optimize(q_fin, n_fin, max_rounds=20, time_limit=tl)
                if c5_o < best_c5:
                    best_c5 = c5_o
                    best_h = build_sym_h(q_o).copy()
        else:
            if time_left() > 25:
                tl = time.time() + min(20, time_left() * 0.5)
                h_o, c5_o = lp_full_optimize(best_h, max_rounds=12, time_limit=tl)
                if c5_o < best_c5:
                    best_c5 = c5_o
                    best_h = h_o.copy()

    # Validate and return
    if best_h is None:
        m_fb = 63
        x_fb = np.linspace(0, 1, m_fb)
        q_fb = np.clip(0.5 + 0.3 * np.sin(2 * np.pi * x_fb), 0, 1)
        best_h = build_sym_h(q_fb)

    best_h = np.asarray(best_h, dtype=float).reshape(-1)
    n_out = len(best_h)
    best_h = project_feasible(best_h, n_out / 2.0)
    best_c5 = eval_score(best_h)

    return best_h, best_c5, n_out
