def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def repair_feasible(h):
    n = len(h)
    h = np.clip(np.asarray(h, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        # Need to add mass: add to positions with most room
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx] + 1e-15)
            h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        # Need to remove mass: remove from positions with most value
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx] + 1e-15)
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    # Final fix
    h = np.clip(h, 0.0, 1.0)
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
            h[idx] = min(1.0, h[idx] + diff2)
        else:
            idx = int(np.argmax(h))
            h[idx] = max(0.0, h[idx] + diff2)
    return h

def build_grad_matrix(h):
    """Build gradient matrix for all lags.
    Returns G of shape (2n-1, n) where G[m, i] = ∂p_m/∂h[i]
    """
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    G = np.zeros((M, n))
    center = n - 1
    
    # Zero lag: p_0 = dx * sum h[i]*(1-h[i])
    G[center, :] = dx * (1.0 - 2.0 * h)
    
    # Positive lags k = 1, ..., n-1
    for k in range(1, n):
        m = center + k
        # p_k = dx * sum_{j=0}^{n-1-k} h[j+k]*(1-h[j])
        # ∂/∂h[i] for i in k..n-1: dx*(1-h[i-k])
        G[m, k:n] = dx * (1.0 - h[0:n-k])
        # ∂/∂h[i] for i in 0..n-1-k: -dx*h[i+k]
        G[m, 0:n-k] += -dx * h[k:n]
        
        # Negative lag -k
        m_neg = center - k
        # p_{-k} = dx * sum_{j=0}^{n-1-k} h[j]*(1-h[j+k])
        # ∂/∂h[i] for i in 0..n-1-k: dx*(1-h[i+k])
        G[m_neg, 0:n-k] = dx * (1.0 - h[k:n])
        # ∂/∂h[i] for i in k..n-1: -dx*h[i-k]
        G[m_neg, k:n] += -dx * h[0:n-k]
    
    return G

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

def lp_trust_region(h, beam=0.005, delta=0.02, top_k=None):
    """Solve LP trust region subproblem.
    Returns (h_new, t_new) or None if solver fails.
    """
    n = len(h)
    M = 2 * n - 1
    profile = compute_profile(h)
    T = np.max(profile)
    
    # Select active set
    thresh = T - beam
    active = np.where(profile >= thresh)[0]
    if len(active) == 0:
        active = np.array([np.argmax(profile)])
    if top_k is not None and len(active) > top_k:
        # Keep only top_k most active
        top_idx = np.argsort(profile[active])[-top_k:]
        active = active[top_idx]
    
    G = build_grad_matrix(h)
    G_active = G[active, :]  # shape (len(active), n)
    p_active = profile[active]
    
    # LP: minimize t
    # s.t. t >= p_m + G_active[m,:] @ (h_new - h) for each active m
    #      0 <= h_new <= 1
    #      sum(h_new) = n/2
    #      |h_new - h|_inf <= delta

    t_var = cp.Variable()

        h_var >= 0,
        h_var <= 1,
        cp.sum(h_var) == n / 2.0,
        h_var >= h - delta,
        h_var <= h + delta,
    ]

            t_var >= p_active[m_idx] + G_active[m_idx, :] @ (h_var - h)
        )

    try:

            return None

        T = np.max(profile)

        if len(active) == 0:
            active = np.array([np.argmax(profile)])

        G_a = G[active, :]
        # Weighted sum of gradients (weight by activity)
        w = profile[active] - thresh + 1e-6
        w = w / w.sum()
        grad = G_a.T @ w
        # Remove mean to maintain sum constraint
        grad -= grad.mean()
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        grad /= gnorm

                best_c = s
                improved = True
                break
    
    return best_h, best_c

import numpy as np
import time
import cvxpy as cp

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def repair_feasible(h):
    n = len(h)
    h = np.clip(np.asarray(h, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx] + 1e-15)
            h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx] + 1e-15)
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    h = np.clip(h, 0.0, 1.0)
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
            h[idx] = min(1.0, h[idx] + diff2)
        else:
            idx = int(np.argmax(h))
            h[idx] = max(0.0, h[idx] + diff2)
    return h

def build_grad_matrix(h):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    G = np.zeros((M, n))
    center = n - 1
    G[center, :] = dx * (1.0 - 2.0 * h)
    for k in range(1, n):
        G[center + k, k:n] = dx * (1.0 - h[0:n - k])
        G[center + k, 0:n - k] += -dx * h[k:n]
        G[center - k, 0:n - k] = dx * (1.0 - h[k:n])
        G[center - k, k:n] += -dx * h[0:n - k]
    return G

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    return repair_feasible(np.clip(h_new, 0.0, 1.0))

def active_descent(h, beam, max_iters, step_base=0.02):
    n = len(h)
    best_h = h.copy()
    best_c = exact_score(h)
    for _ in range(max_iters):
        profile = compute_profile(best_h)
        T = float(np.max(profile))
        thresh = T - beam
        active = np.where(profile >= thresh)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(profile))])
        if len(active) > 20:
            top_idx = np.argsort(profile[active])[-20:]
            active = active[top_idx]
        G = build_grad_matrix(best_h)
        w = profile[active] - thresh + 1e-8
        w /= w.sum()
        grad = G[active, :].T @ w
        grad -= grad.mean()
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        grad /= gnorm
        improved = False
        for alpha in [step_base, step_base*0.5, step_base*0.25, step_base*0.125, step_base*0.0625]:
            h_try = repair_feasible(best_h - alpha * grad)
            s = exact_score(h_try)
            if s < best_c - 1e-14:
                best_h = h_try
                best_c = s
                improved = True
                break
        if not improved:
            break
    return best_h, best_c

def lp_solve_single(h, beam, delta, top_k):
    n = len(h)
    profile = compute_profile(h)
    T = float(np.max(profile))
    thresh = T - beam
    active = np.where(profile >= thresh)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(profile))])
    if top_k is not None and len(active) > top_k:
        top_idx = np.argsort(profile[active])[-top_k:]
        active = active[top_idx]
    G = build_grad_matrix(h)
    G_a = G[active, :]
    p_a = profile[active]
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    cons = [
        h_var >= 0,
        h_var <= 1,
        cp.sum(h_var) == n / 2.0,
        h_var >= h - delta,
        h_var <= h + delta,
    ]
    for i in range(len(active)):
        cons.append(t_var >= p_a[i] + G_a[i, :] @ (h_var - h))
    prob = cp.Problem(cp.Minimize(t_var), cons)
    for solver in [cp.CBC, cp.GLPK, cp.CVXOPT]:
        try:
            prob.solve(solver=solver, max_time=12)
            if prob.status in ('optimal', 'optimal_inaccurate') and h_var.value is not None:
                return np.array(h_var.value, dtype=np.float64)
        except Exception:
            continue
    return None

def lp_trust_region_adaptive(h, beam, delta_init, top_k, max_shrinks=5):
    n = len(h)
    c_cur = exact_score(h)
    delta = delta_init
    for _ in range(max_shrinks):
        if delta < 0.001:
            break
        h_lp = lp_solve_single(h, beam, delta, top_k)
        if h_lp is not None:
            h_lp = repair_feasible(h_lp)
            c_lp = exact_score(h_lp)
            if c_lp < c_cur - 1e-13:
                return h_lp, c_lp, True
            for frac in [0.7, 0.5, 0.3, 0.15]:
                h_blend = repair_feasible(frac * h_lp + (1.0 - frac) * h)
                c_blend = exact_score(h_blend)
                if c_blend < c_cur - 1e-13:
                    return h_blend, c_blend, True
        delta *= 0.5
    return h, c_cur, False

def try_slsqp_epigraph(h0, maxiter=200):
    n = len(h0)
    dx = 2.0 / n
    M = 2 * n - 1
    c0 = exact_score(h0)

    def obj(x):
        return x[-1]

    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g

    def eq_fun(x):
        return x[:-1].sum() - n / 2.0

    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g

    def ineq_fun(x):
        hh = x[:-1]
        t = x[-1]
        prof = np.correlate(hh, 1.0 - hh, mode="full") * dx
        return t - prof

    def ineq_jac(x):
        hh = x[:-1]
        G = build_grad_matrix(hh)
        J = np.zeros((M, n + 1))
        J[:, :-1] = -G
        J[:, -1] = 1.0
        return J

    from scipy.optimize import minimize
    x0 = np.concatenate([h0, [c0 + 1e-4]])
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 2.0)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac},
    ]
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-11})
        if res.x is not None and np.isfinite(res.x).all():
            h_new = repair_feasible(res.x[:-1])
            c_new = exact_score(h_new)
            if c_new <= c0 + 1e-8:
                return h_new
    except Exception:
        pass
    return None
