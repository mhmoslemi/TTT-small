    #   sum(h) - n/2 = 0

from scipy.optimize import minimize
from scipy.special import logsumexp

def project_feasible(x, target_sum):
    n = len(x)
    lo = float(np.min(x)) - 2.0
    hi = float(np.max(x)) + 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-13:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.any():
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h


def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def c5_exact(h):
    n = len(h)
    return float(compute_corrs(h).max() * (2.0 / n))


def raw_corr_grad(h, k, n):
    offset = k - (n - 1)
    grad = np.zeros(n)
    if offset == 0:
        grad[:] = 1.0 - 2.0 * h
    elif offset > 0:
        grad[offset:] += (1.0 - h[:n - offset])
        grad[:n - offset] += (-h[offset:])
    else:
        d = -offset
        grad[:n - d] += (1.0 - h[d:])
        grad[d:] += (-h[:n - d])
    return grad


def smooth_max_obj(h, dx, beta):
    corrs = compute_corrs(h) * dx
    return float(logsumexp(beta * corrs) / beta)


def smooth_max_grad(h, dx, beta):
    n = len(h)
    corrs = compute_corrs(h)
    cd = corrs * dx
    shifted = beta * (cd - cd.max())
    w = np.exp(shifted)
    w /= w.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        grad += w[k] * raw_corr_grad(h, k, n)
    return grad * dx


def slsqp_smoothmax(h0, dx, target_sum, beta, maxiter):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n

    def obj(x):
        return smooth_max_obj(x, dx, beta)

    def objg(x):
        return smooth_max_grad(x, dx, beta)

    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h0, jac=objg, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        h_out = project_feasible(res.x, target_sum)
    except Exception:
        h_out = project_feasible(h0, target_sum)
    return h_out, c5_exact(h_out)


def epigraph_solve(h0, target_raw, n, rho=1.0, maxiter=400, active_lags=None):
    """Solve epigraph: min s + rho/n*||h-h0||^2 s.t. c[k] <= target_raw + s."""
    if active_lags is None:
        active_lags = list(range(2 * n - 1))
    m = len(active_lags)
    bounds_full = [(0.0, 1.0)] * n + [(0.0, target_raw + 10.0)]

    def obj(x):
        h = x[:n]
        s = x[n]
        return s + rho / n * np.dot(h - h0, h - h0)

    def obj_grad(x):
        h = x[:n]
        g = np.empty(n + 1)
        g[:n] = 2.0 * rho / n * (h - h0)
        g[n] = 1.0
        return g

    def eq_fun(x):
        return np.array([x[:n].sum() - n * 0.5])

    def eq_jac(x):
        j = np.zeros((1, n + 1))
        j[0, :n] = 1.0
        return j

    x0_local = h0.copy()

    def ineq_fun(x):
        h = x[:n]
        s = x[n]
        corrs = compute_corrs(h)
        return target_raw + s - corrs[active_lags]

    def ineq_jac(x):
        h = x[:n]
        j = np.zeros((m, n + 1))
        for i, k in enumerate(active_lags):
            j[i, :n] = raw_corr_grad(h, k, n)
            j[i, n] = 1.0
        return j

    cur_max = compute_corrs(h0).max()
    s0 = max(0.0, cur_max - target_raw) + 1e-4
    x0 = np.empty(n + 1)
    x0[:n] = h0
    x0[n] = s0

    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    try:
        res = minimize(obj, x0, jac=obj_grad, method='SLSQP',
                       bounds=bounds_full, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        h_out = project_feasible(res.x[:n], n * 0.5)
    except Exception:
        h_out = project_feasible(h0, n * 0.5)
    return h_out, c5_exact(h_out)


def epigraph_active(h0, target_raw, n, top_m=8, rho=1.0, maxiter=500):
    """Epigraph using only top-m active lags for efficiency."""
    corrs = compute_corrs(h0)
    active = np.argsort(corrs)[-top_m:]
    active = sorted(active.tolist())
    h_out, c5_val = epigraph_solve(h0, target_raw, n, rho=rho,
                                   maxiter=maxiter, active_lags=active)
    # Check if new lags are violated
    corrs_new = compute_corrs(h_out)
    violated = np.where(corrs_new > target_raw + 1e-10)[0]
    new_active = sorted(set(active + violated.tolist()))
    if len(new_active) > len(active) and len(new_active) <= 2 * n - 1:
        h_out2, c5_val2 = epigraph_solve(h_out, target_raw, n, rho=rho,
                                         maxiter=maxiter, active_lags=new_active)
        if c5_val2 < c5_val:
            h_out, c5_val = h_out2, c5_val2
    return h_out, c5_val

    """Build anti-palindromic h from half-vector v of length n//2."""

    h[:m] = v
    h[m + 1:][::-1] = 1.0 - v
    if n % 2 == 1:
        h[m] = 0.5

    else:
        h[m:][::-1] = 1.0 - v[::-1]
    return h

            best_h = h_final

    # Final: one more epigraph squeeze if time allows
    if remaining() > 15:
        n_b = len(best_h)
        target_raw = (best_c5 - 0.00005) * n_b / 2.0
        h_e, c5_e = epigraph_active(best_h, target_raw, n_b,
                                    top_m=8, rho=0.2, maxiter=500)
        if c5_e < best_c5:
            best_c5 = c5_e
            best_h = h_e

    # Final verification
    best_h = project_feasible(best_h, len(best_h) * 0.5)
    final_c5 = c5_exact(best_h)
    if final_c5 < best_c5 + 1e-12:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
