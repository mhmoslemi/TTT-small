def project_feasible(x, target_sum):
    """Project x onto {h: 0<=h[i]<=1, sum(h)=target_sum}."""
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)

        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
        else:
            h += diff / n

def c5_of_h(h, dx):

def compute_delta_binary(h, i, j, n):
    """O(n) delta computation for binary swap i(1->0), j(0->1)."""
    ms = np.arange(-(n-1), n)
    delta = np.zeros(2*n-1)
    v = (i+ms >= 0) & (i+ms <= n-1)
    delta[v] += -(1.0 - h[i+ms[v]])
    v = (j+ms >= 0) & (j+ms <= n-1)
    delta[v] += (1.0 - h[j+ms[v]])
    v = (i-ms >= 0) & (i-ms <= n-1)
    delta[v] += h[i-ms[v]]
    v = (j-ms >= 0) & (j-ms <= n-1)
    delta[v] += -h[j-ms[v]]
    return delta

def smooth_max_obj(h, dx, beta):
    corrs = np.correlate(h, 1.0-h, mode='full')
    m = corrs.max()
    return (m + np.log(np.sum(np.exp(beta*(corrs-m))))/beta) * dx

def grad_corr_k(h, k, n):
    offset = k - (n-1)
    grad = np.zeros(n)
    i_lo = max(0, -offset)
    i_hi = min(n-1, n-1-offset)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi+1)
    grad[idx+offset] += (1.0 - h[idx])
    grad[idx] += (-h[idx+offset])
    return grad

def smooth_max_grad(h, dx, beta):
    n = len(h)
    corrs = np.correlate(h, 1.0-h, mode='full')
    m = corrs.max()

        if w[k] < 1e-14:

        grad += w[k] * grad_corr_k(h, k, n)
    return grad * dx

    except Exception:
        h_out = project_feasible(h0, target_sum)
    return h_out, c5_of_h(h_out, dx)

    return h_out, c5_of_h(h_out, dx)


def to_binary(h, target_count):
    idx = np.argsort(-h)
    b = np.zeros(len(h))
    b[idx[:target_count]] = 1.0
    return b


def binary_steepest(h0, dx, max_sweeps=100):

    h = to_binary(h0, n // 2)
    corrs = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = corrs.max() * dx
    best_h = h.copy()
    for sweep in range(max_sweeps):
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        best_swap = None
        best_new_max = corrs.max()
        for i in ones:
            di = compute_delta_binary(h, i, -1, n)  # placeholder, need j
            # Actually need to try all j
            for j in zeros:
                delta = compute_delta_binary(h, i, j, n)
                new_max = (corrs + delta).max()
                if new_max < best_new_max - 1e-14:
                    best_new_max = new_max
                    best_swap = (i, j)
        if best_swap is None:
            break
        i, j = best_swap
        h[i], h[j] = 0.0, 1.0
        corrs = np.correlate(h, 1.0 - h, mode='full')
        new_c5 = corrs.max() * dx
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
    return best_h, best_c5

Wait, the binary steepest descent is O(n²) per sweep (trying all pairs of ones and zeros). For n=126, that's 63*63 = 3969 evaluations per sweep, each O(n) = O(126). So one sweep is about 500K numpy operations. That should take maybe 10-50ms.

With 100 sweeps, that's 1-5 seconds. That's fine.

But actually, I realize the compute_delta_binary function creates a new arange each time. Let me precompute the ms vector.
