```
3. Return best h found.
```

This is a "linearized minimax" approach. It should work well because:
- The LP is fast (n=126, ~200 constraints).
- The line search ensures monotone improvement.
- Coordinate descent handles the non-smoothness.

Let me also consider: the LP approach is essentially solving the problem to first-order accuracy. For a bilinear objective, the linearization error is O(‖h_new - h0‖²). So the step size should be small initially and can grow as we converge.

```python
import numpy as np
from scipy.optimize import minimize_scalar, linprog
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full').max() * dx

def project_to_sum(h, target):
    """Project h onto the set {x : sum(x) = target, 0 <= x <= 1}"""
    # This is the Euclidean projection onto the intersection of a hyperplane and a box
    # Use bisection on the Lagrange multiplier
    lo, hi = -10.0, 10.0
    for _ in range(100):
        mid = (lo + hi) / 2
        proj = np.clip(h - mid, 0.0, 1.0)
        if proj.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0.0, 1.0)

def coord_desc_sym(q, n, max_sweeps=200, tol=1e-12):
    """Coordinate descent in symmetric space."""
    m = n // 2
    for sweep in range(max_sweeps):
        max_change = 0.0
        for i in range(m):
            v = q[i]
            j = n - 1 - i  # paired index
            
            # Compute slopes for changing h[i] by (x-v) and h[j] by -(x-v)
            h = np.empty(n)
            h[:m] = q
            h[m:] = 1.0 - q[::-1]
            
            # Slope a[k] for lag k = n-1+d
            # Change in c[k] = (x-v) * [(1-h[i-d]) - h[i+d] - (1-h[j-d]) + h[j+d]]
            d_range = np.arange(-(n-1), n)
            # For coordinate i:
            i_d = i - d_range  # i - d for each d
            i_pd = i + d_range  # i + d for each d
            # For coordinate j:
            j_d = j - d_range
            j_pd = j + d_range
            
            slope = np.zeros(2*n-1)
            # (1-h[i-d]) where 0 <= i-d <= n-1
            mask1 = (i_d >= 0) & (i_d <= n-1)
            slope[mask1] += (1.0 - h[i_d[mask1]])
            # -h[i+d] where 0 <= i+d <= n-1
            mask2 = (i_pd >= 0) & (i_pd <= n-1)
            slope[mask2] -= h[i_pd[mask2]]
            # -((1-h[j-d])) where 0 <= j-d <= n-1
            mask3 = (j_d >= 0) & (j_d <= n-1)
            slope[mask3] -= (1.0 - h[j_d[mask3]])
            # +h[j+d] where 0 <= j+d <= n-1
            mask4 = (j_pd >= 0) & (j_pd <= n-1)
            slope[mask4] += h[j_pd[mask4]]
            
            # Current correlations
            c = np.correlate(h, 1.0-h, mode='full')
            
            # c_new[k] = c[k] + (x-v)*slope[k] = (c[k] - v*slope[k]) + x*slope[k]
            b = c - v * slope
            
            # Minimize max_k (b[k] + x*slope[k]) over x in [0,1]
            # Use grid + golden section
            grid = np.linspace(0, 1, 100)
            vals = np.max(b[None, :] + grid[:, None] * slope[None, :], axis=1)
            best_idx = np.argmin(vals)
            
            # Golden section refinement around best grid point
            if best_idx == 0:
                lo, hi = 0.0, grid[1]
            elif best_idx == 99:
                lo, hi = grid[98], 1.0
            else:
                lo, hi = grid[best_idx-1], grid[best_idx+1]
            
            def f1(x):
                return np.max(b + x * slope)
            
            res = minimize_scalar(f1, bounds=(lo, hi), method='bounded',
                                  options={'xatol': 1e-14})
            x_new = res.x
            
            change = abs(x_new - v)
            if change > 1e-14:
                q[i] = x_new
                max_change = max(max_change, change)
        
        if max_change < tol:
            break
    
    h = np.empty(n)
    h[:m] = q
    h[m:] = 1.0 - q[::-1]
    return h

def lp_step(h_current, n, n_active=None):
    """Solve the linearized LP step."""
    m = 2 * n - 1
    c = np.correlate(h_current, 1.0-h_current, mode='full')
    dx = 2.0 / n
    
    if n_active is None:
        n_active = min(60, m)
    
    # Top active lags
    top_idx = np.argsort(c)[::-1][:n_active]
    
    # Build LP: min t s.t. (grad_k · (h - h0) + c_k) <= t, sum(h) = n/2, 0 <= h <= 1
    # Variables: h[0..n-1], t
    # Constraint for each active lag k:
    #   sum_i grad_k[i] * h[i] - t <= -c[k] + sum_i grad_k[i] * h0[i]
    
    # grad_k[i] = (1-h[i-d]) - h[i+d] where d = k-(n-1)
    # Compute gradient matrix for active lags
    G = np.zeros((n_active, n))
    for idx, k in enumerate(top_idx):
        d = k - (n - 1)
        for i in range(n):
            g = 0.0
            if 0 <= i - d <= n - 1:
                g += (1.0 - h_current[i - d])
            if 0 <= i + d <= n - 1:
                g -= h_current[i + d]
            G[idx, i] = g
    
    # LP formulation for linprog:
    # min c^T x
    # s.t. A_ub x <= b_ub
    #      A_eq x = b_eq
    #      bounds
    
    # Variables: [h[0], ..., h[n-1], t]
    # Objective: [0, ..., 0, 1] (minimize t)
    # Inequality: for each active lag k: grad_k · h - t <= -c[k] + grad_k · h0
    #   => grad_k · h - t <= rhs_k
    # Equality: sum(h) = n/2 => [1,1,...,1, 0] · x = n/2
    # Bounds: 0 <= h[i] <= 1, t free (or t >= 0)
    
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    A_ub = np.zeros((n_active, n + 1))
    A_ub[:, :n] = G
    A_ub[:, -1] = -1.0
    
    b_ub = np.array([-c[k] + G[idx, :] @ h_current] for idx, k in enumerate(top_idx))
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    bounds = [(0, 1)] * n + [(0, None)]  # t >= 0
    
    from scipy.optimize import linprog
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.status == 0:
        return res.x[:n], res.x[-1]
    return None, None

def line_search(h0, h1, max_iter=50):
    """Find α ∈ [0,1] that minimizes exact_score(h0 + α*(h1-h0))."""
    def f(alpha):
        h = h0 + alpha * (h1 - h0)
        h = np.clip(h, 0.0, 1.0)
        return exact_score(h)
    
    res = minimize_scalar(f, bounds=(0.0, 1.0), method='bounded',
                          options={'xatol': 1e-8})
    alpha = res.x
    h_new = h0 + alpha * (h1 - h0)
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = project_to_sum(h_new, len(h_new) / 2.0)
    return h_new, f(alpha)
```

```python
def lp_step(h, n, active_indices, n_active_extra=0):
    """Solve linearized LP. Returns (h_new, t_value) or (None, None)."""
    m = 2*n - 1
    c = corr_full(h)
    
    if n_active_extra > 0:
        # Add some additional lags in case
        top_all = np.argsort(c)[::-1]
        combined = list(set(active_indices.tolist()))
        for idx in top_all[:n_active_extra]:
            if idx not in active_indices:
                combined.append(idx)
        active_indices = np.array(combined)
    
    K = len(active_indices)
    
    # Gradient matrix: G[k, i] = d(c_k)/d(h_i) at h
    G = np.zeros((K, n))
    for kk, k in enumerate(active_indices):
        d = k - (n - 1)
        for i in range(n):
            g = 0.0
            if 0 <= i - d <= n - 1:
                g += 1.0 - h[i - d]
            if 0 <= i + d <= n - 1:
                g -= h[i + d]
            G[kk, i] = g
    
    # LP: min t
    # s.t. G[k,:]·h - t <= c[k] - G[k,:]·h  (i.e., linearization rhs)
    #      sum(h) = n/2
    #      0 <= h <= 1
    
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    A_ub = np.zeros((K, n+1))
    A_ub[:, :n] = G
    A_ub[:, -1] = -1.0
    b_ub = c[active_indices] - G @ h  # = c[k] - G[k,:]·h
    
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n/2.0])
    
    bounds = [(0.0, 1.0)] * n + [(c.max() * 0.5, None)]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.status == 0:
            return res.x[:n], res.x[-1]
    except Exception:
        pass
    return None, None
```

```python
def bounded_min_1d(f_obj, args, lo, hi, xtol=1e-10, max_iter=200):
    """Minimize f_obj(x, *args) for x in [lo, hi]."""
    # Golden section search
    gr = (np.sqrt(5) - 1) / 2  # golden ratio conjugate
    a, b = lo, hi
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = f_obj(c, *args)
    fd = f_obj(d, *args)
    for _ in range(max_iter):
        if b - a < xtol:
            break
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = f_obj(c, *args)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = f_obj(d, *args)
    x_opt = (a + b) / 2
    return x_opt, f_obj(x_opt, *args)
```

```python
import numpy as np
from scipy.optimize import linprog
import time


def project_to_sum(h, target):
    lo, hi = -5.0, 5.0
    for _ in range(80):
        mid = (lo + hi) / 2
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0.0, 1.0)


def exact_score(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full').max() * (2.0 / n)


def corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')


def sym_h(q, n):
    m = n // 2
    h = np.empty(n)
    h[:m] = q
    h[m:] = 1.0 - q[::-1]
    return h


def q_from_sym(h, n):
    m = n // 2
    return 0.5 * (h[:m] + 1.0 - h[n - 1: n - m - 1: -1])


def bounded_min_1d(f_obj, args, lo, hi, xtol=1e-12, max_iter=300):
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = lo, hi
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = f_obj(c, *args)
    fd = f_obj(d, *args)
    for _ in range(max_iter):
        if b - a < xtol:
            break
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = f_obj(c, *args)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = f_obj(d, *args)
    x_opt = (a + b) / 2.0
    return x_opt, f_obj(x_opt, *args)


def f_sym_obj(x, b_vec, slope):
    return np.max(b_vec + x * slope)


def f_line_obj(x, h0, direction):
    ht = np.clip(h0 + x * direction, 0.0, 1.0)
    return exact_score(ht)


def f_paired_obj(x, c_cur, slope):
    return np.max(c_cur + x * slope)


def coord_desc_sym(q, n, max_sweeps=500, tol=1e-13):
    m = n // 2
    d_range = np.arange(-(n - 1), n)
    for sweep in range(max_sweeps):
        max_delta = 0.0
        h = sym_h(q, n)
        c = corr_full(h)
        for i in range(m):
            j = n - 1 - i
            v = q[i]
            s = np.zeros(2 * n - 1)
            id_ = i - d_range
            ip = i + d_range
            jd = j - d_range
            jp = j + d_range
            m1 = (id_ >= 0) & (id_ < n)
            m2 = (ip >= 0) & (ip < n)
            m3 = (jd >= 0) & (jd < n)
            m4 = (jp >= 0) & (jp < n)
            s[m1] += (1.0 - h[id_[m1]])
            s[m2] -= h[ip[m2]]
            s[m3] -= (1.0 - h[jd[m3]])
            s[m4] += h[jp[m4]]
            b = c - v * s
            xs = np.linspace(0.0, 1.0, 64)
            vals = np.max(b[None, :] + np.outer(xs, s), axis=1)
            ib = np.argmin(vals)
            if ib == 0:
                lo_g, hi_g = 0.0, xs[1]
            elif ib == 63:
                lo_g, hi_g = xs[-2], 1.0
            else:
                lo_g, hi_g = xs[ib - 1], xs[ib + 1]
            xn, _ = bounded_min_1d(f_sym_obj, (b, s), lo_g, hi_g)
            delta = abs(xn - v)
            if delta > 1e-15:
                q[i] = xn
                c = c + (xn - v) * s
                max_delta = max(max_delta, delta)
        if max_delta < tol:
            break
    return sym_h(q, n)


def compute_gradient_row(h, n, k):
    d = k - (n - 1)
    g = np.zeros(n)
    for i in range(n):
        val = 0.0
        if 0 <= i - d <= n - 1:
            val += 1.0 - h[i - d]
        if 0 <= i + d <= n - 1:
            val -= h[i + d]
        g[i] = val
    return g


def lp_step(h, n, active_indices):
    K = len(active_indices)
    G = np.zeros((K, n))
    c = corr_full(h)
    for kk, k in enumerate(active_indices):
        G[kk] = compute_gradient_row(h, n, k)
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_ub = np.zeros((K, n + 1))
    A_ub[:, :n] = G
    A_ub[:, -1] = -1.0
    b_ub = G @ h - c[active_indices]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
        if res.status == 0:
            return res.x[:n], res.x[-1]
    except Exception:
        pass
    return None, None


def paired_coord_desc(h, n, max_iters, rng):
    c5_cur = exact_score(h)
    d_range = np.arange(-(n - 1), n)
    for _ in range(max_iters):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        vi, vj = h[i], h[j]
        d_lo = max(-vi, vj - 1.0)
        d_hi = min(1.0 - vi, vj)
        if d_lo >= d_hi - 1e-15:
            continue
        c = corr_full(h)
        s = np.zeros(2 * n - 1)
        id_ = i - d_range
        ip = i + d_range
        m1 = (id_ >= 0) & (id_ < n)
        m2 = (ip >= 0) & (ip < n)
        s[m1] += (1.0 - h[id_[m1]])
        s[m2] -= h[ip[m2]]
        jd = j - d_range
        jp = j + d_range
        m3 = (jd >= 0) & (jd < n)
        m4 = (jp >= 0) & (jp < n)
        s[m3] -= (1.0 - h[jd[m3]])
        s[m4] += h[jp[m4]]
        bd, _ = bounded_min_1d(f_paired_obj, (c, s), d_lo, d_hi, xtol=1e-12, max_iter=100)
        if abs(bd) > 1e-13:
            h[i] = np.clip(vi + bd, 0.0, 1.0)
            h[j] = np.clip(vj - bd, 0.0, 1.0)
            c5_t = exact_score(h)
            if c5_t < c5_cur - 1e-13:
                c5_cur = c5_t
    return h, c5_cur
```
