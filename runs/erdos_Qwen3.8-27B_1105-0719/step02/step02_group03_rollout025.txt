            best_obj = new_obj
        else:

    return best_h

def dual_primal_iteration(h, corr, dx, n, step_size=0.1):
    """One iteration of the dual-primal method."""
    # Current h is binary (or use the correlation to determine active set)
    n_corr = len(corr)
    V = corr.max()  # current max correlation
    
    # Update lambda: weight active constraints more
    # Use exponential weighting: lambda_k ∝ exp(beta * C_k)
    beta = max(1.0, 50.0 / dx / V)  # scale beta
    # Actually, use a simpler approach: lambda ∝ C_k^p
    p = 2.0
    lam = corr ** p
    lam_sum = lam.sum()
    if lam_sum > 0:
        lam /= lam_sum
    else:
        lam = np.ones(n_corr) / n_corr
    
    # Compute cost vector c[i] = sum_k lam[k] * dC_k/dh[i]
    c = np.zeros(n)
    for k_idx in range(n_corr):
        if lam[k_idx] < 1e-15:
            continue
        offset = k_idx - (n - 1)
        # dC_k/dh[i] = (1-h[i+offset]) * I(0<=i+offset<n) - h[i-offset] * I(0<=i-offset<n)
        # For binary h: if h[i]=1, dC/dh[i] = -h[i+offset] - (1-h[i-offset])
        # Wait, let me recompute for binary h.
        # dC_k/dh[i] = (1-h[i+offset]) * I(valid) - h[i-offset] * I(valid)
        # For binary h[i] = 1: (1-h[i+offset]) - h[i-offset] = h[i+offset is 0] - h[i-offset is 1]
        #   = (1 - h[i+offset]) - h[i-offset]
        # This is the same formula regardless of h[i]! The derivative is the same.

        # Term 1: (1-h[i+offset]) for i where 0<=i+offset<n
        # Term 2: -h[i-offset] for i where 0<=i-offset<n

        valid1 = (i_range + offset >= 0) & (i_range + offset < n)
        valid2 = (i_range - offset >= 0) & (i_range - offset < n)
        c[valid1] += lam[k_idx] * (1.0 - h[i_range[valid1] + offset])

        mid = (lo + hi) / 2

        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)

def binary_sweep_vectorized(h, dx):
    """One best-improvement sweep, vectorized over the swap pairs."""
    n = len(h)
    corrs = compute_corrs(h)
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    
    # Precompute effect vectors for all positions
    offsets = np.arange(-(n-1), n)

    # effect[p][m] = (1-h[p+off_m])*I(valid) - h[p-off_m]*I(valid)
    effect = np.zeros((n, 2*n-1))
    for p in range(n):
        v1 = (p + offsets >= 0) & (p + offsets < n)
        v2 = (p - offsets >= 0) & (p - offsets < n)
        effect[p, v1] += 1.0 - h[p + offsets[v1]]
        effect[p, v2] += -h[p - offsets[v2]]
    
    # For swap (i in ones, j in zeros): delta = -effect[i] + effect[j]
    # new_corrs = corrs - effect[i] + effect[j]
    # new_max = max(corrs - effect[i] + effect[j])
    
    # Vectorize: for each i in ones, compute (corrs - effect[i]) which is (2n-1) dim
    # Then for each j in zeros, add effect[j] and take max
    
    # Shape: (n_ones, 2n-1) for (corrs - effect[i])
    corrs_removed = corrs[None, :] - effect[ones][:, :]  # (n_ones, 2n-1)
    
    # For each j: new_max[i] = max(corrs_removed[i] + effect[j])
    # Shape: (n_zeros, n_ones, 2n-1) → max over last axis → (n_zeros, n_ones)
    # This is n_zeros * n_ones * (2n-1) = 63 * 63 * 251 ≈ 1 million. OK.
    
    new_maxes = (corrs_removed[None, :, :] + effect[zeros][:, None, :]).max(axis=2)

        ji, ii = np.unravel_index(new_maxes.argmin(), new_maxes.shape)
        i = ones[ii]
        j = zeros[ji]

def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
        else:

        effect[p, v1] += 1.0 - h[p + offsets[v1]]
        effect[p, v2] += -h[p - offsets[v2]]

def dual_primal_binary(h_init, n, dx, max_iter, t0, time_limit):
    n_corr = 2 * n - 1
    lam = np.ones(n_corr) / n_corr
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    for it in range(max_iter):
        if time.time() - t0 > time_limit:
            break
        effect = compute_effect_vectors(h, n)
        c = (lam[None, :] * effect).sum(axis=1)
        n_ones = n // 2
        new_h = np.zeros(n)
        new_h[np.argsort(c)[:n_ones]] = 1.0
        if not np.array_equal(new_h, h):
            h = new_h
            c5 = c5_of_h(h, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        corrs = compute_corrs(h)
        alpha = 0.5 / (1.0 + 0.02 * it)
        lam = lam + alpha * corrs
        lam = np.maximum(lam, 0.0)
        s = lam.sum()
        if s > 1e-15:
            lam /= s
        else:
            lam = np.ones(n_corr) / n_corr
    return best_h, best_c5


def make_binary_structured(n, mode, rng):
    h = np.zeros(n)
    half = n // 2
    if mode == 0:
        h[::2] = 1.0
    elif mode == 1:
        phi = (1.0 + np.sqrt(5.0)) / 2.0
        for k in range(half):
            h[int((k * phi) % 1.0 * n) % n] = 1.0
    elif mode == 2:
        idx = rng.choice(n, size=half, replace=False)
        h[idx] = 1.0
    elif mode == 3:
        period = max(2, n // half)
        for k in range(half):
            h[(k * period + k // 7) % n] = 1.0
    elif mode == 4:
        x = np.arange(n)
        h[np.where((x * 7) % n < n // 2)[0][:half]] = 1.0
    # Ensure exactly half ones
    if h.sum() != half:
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if h.sum() > half:
            to_remove = h.sum() - half
            for idx in ones[:to_remove]:
                h[idx] = 0.0
        else:
            to_add = half - h.sum()
            for idx in zeros[:to_add]:
                h[idx] = 1.0
    return h

        # Final binary check and local search
        h_fin_bin = np.zeros(n_f)
        h_fin_bin[np.argsort(-best_h)[:int(target_f)]] = 1.0
        h_fin, c5_fin = binary_local_search(h_fin_bin, dx_f, 200, t0, t0 + 80)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    # Final validation
    n_final = len(best_h)
    dx_final = 2.0 / n_final
    target_final = n_final / 2.0
    best_h = project_feasible(best_h, target_final)
    best_c5 = c5_of_h(best_h, dx_final)
    # Ensure feasibility
    assert np.all(best_h >= -1e-10) and np.all(best_h <= 1.0 + 1e-10)
    assert abs(best_h.sum() - target_final) < 1e-8

    return best_h, best_c5, n_final
