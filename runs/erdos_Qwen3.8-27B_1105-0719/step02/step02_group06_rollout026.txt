def eval_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def corr_raw(h):
    return np.correlate(h, 1.0 - h, mode='full')

def corr_at_k(h, k):
    n = len(h)
    d = k - (n - 1)
    i_lo = max(0, -d)
    i_hi = min(n - 1, n - 1 - d)
    if i_lo > i_hi:
        return 0.0
    return float(np.sum(h[i_lo:i_hi + 1] * (1.0 - h[i_lo + d:i_hi + 1 + d])))

def corr_grad_k(h, k):
    n = len(h)
    d = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -d)
    i_hi = min(n - 1, n - 1 - d)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx] += (1.0 - h[idx + d])
    grad[idx + d] += (-h[idx])
    return grad

def project_feasible(x, target_sum):

def select_active(corrs_scaled, margin_frac=0.03, neighbor_pad=3, max_k=70):
    max_val = corrs_scaled.max()
    threshold = max_val - max(1e-6, margin_frac * max_val)
    active = set(np.where(corrs_scaled >= threshold)[0].tolist())
    padded = set()
    for k in active:

            if 0 <= k + d < len(corrs_scaled):
                padded.add(k + d)
    active_list = sorted(padded)
    if len(active_list) > max_k:
        scored = sorted([(corrs_scaled[k], k) for k in active_list], reverse=True)
        active_list = sorted([k for _, k in scored[:max_k]])
    return active_list

    A_ub = np.array(A_ub)

    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])

    return None

    for alpha in grid:
        h_new = h + alpha * direction
        if h_new.min() < -1e-10 or h_new.max() > 1 + 1e-10:
            continue
        sc = eval_score(h_new)
        if sc < best_sc:
            best_sc = sc
            best_alpha = alpha
    
    if best_alpha > 0:
        h_new = h + best_alpha * direction
        h_new = project_feasible(h_new, len(h) / 2.0)
        best_sc = eval_score(h_new)
        return h_new, best_sc
    return h, current_score

def coord_descent_step(h, n_pairs=200, rng=None):
    """Try a set of transfer pairs and do line search."""
    n = len(h)
    current_score = eval_score(h)
    
    # Compute correlations to find hot lags
    corrs = corr_raw(h) * (2.0 / n)
    max_val = corrs.max()
    hot_lags = np.where(corrs >= max_val - 0.05 * max_val)[0]
    
    # Find important indices (those appearing in hot lag sums)
    important = set()
    for k in hot_lags:
        d = k - (n - 1)
        i_lo = max(0, -d)
        i_hi = min(n - 1, n - 1 - d)
        important.update(range(i_lo, i_hi + 1))
        important.update(range(i_lo + d, i_hi + d + 1))
    
    # Build pair list
    imp_list = sorted(important)
    pairs = []
    if rng is not None:
        # Mix of random and structured
        for _ in range(n_pairs // 2):
            a = rng.choice(imp_list)
            b = rng.choice(imp_list)
            if a != b:
                pairs.append((a, b))
    for _ in range(n_pairs // 2):
        a = rng.randint(n)
        b = rng.randint(n)
        if a != b:
            pairs.append((a, b))
    
    improved = False
    for (a, b) in pairs:
        if a == b:
            continue
        h_new, sc_new = transfer_line_search(h, a, b, n_grid=48)
        if sc_new < current_score - 1e-12:
            h = h_new
            current_score = sc_new
            improved = True
    
    return h, current_score, improved

def smooth_max_obj_grad_q(q, n_full, beta):
    m = n_full // 2
    h = build_sym_h(q)
    dx = 2.0 / n_full
    corrs = corr_raw(h) * dx
    max_c = corrs.max()
    shifted = corrs - max_c
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(len(corrs)) / len(corrs)
    else:
        w /= w_sum
    val = max_c + np.log(w_sum) / beta
    
    grad_q = np.zeros(m)
    for k in range(2 * n_full - 1):
        if w[k] < 1e-12:
            continue
        g = corr_grad_k(h, k)
        for r in range(m):
            grad_q[r] += w[k] * (g[r] - g[n_full - 1 - r])
    grad_q *= dx
    return val, grad_q

def smooth_sym_optimize(q0, n_full, beta_list, maxiter_per_beta, time_limit):
    m = n_full // 2
    q = np.clip(q0.copy(), 0, 1)
    best_score = eval_score(build_sym_h(q))
    best_q = q.copy()
    
    for beta in beta_list:
        if time.time() > time_limit:
            break
        bounds = [(0.0, 1.0)] * m
        try:
            res = minimize(
                lambda xx: smooth_max_obj_grad_q(xx, n_full, beta)[0],
                q,
                jac=lambda xx: smooth_max_obj_grad_q(xx, n_full, beta)[1],
                method='L-BFGS-B',
                bounds=bounds,
                options={'maxiter': maxiter_per_beta, 'ftol': 1e-14, 'gtol': 1e-10})
            q_new = np.clip(res.x, 0, 1)
        except Exception:
            q_new = q.copy()
        h_new = build_sym_h(q_new)
        sc = eval_score(h_new)
        if sc < best_score:
            best_score = sc
            best_q = q_new.copy()
        q = q_new
    
    return best_q, best_score, build_sym_h(best_q)

def epig_sym_solve(q, n_full, active_k, maxiter=50):
    m = n_full // 2
    h_c = build_sym_h(q)
    dx = 2.0 / n_full
    corrs_c = corr_raw(h_c) * dx
    t0_val = corrs_c.max() + 1e-9
    
    def con_fun(x, kk):
        qq = x[:m]
        t = x[m]
        hh = build_sym_h(qq)
        return t - corr_at_k(hh, kk) * dx
    
    def con_jac(x, kk):
        qq = x[:m]
        hh = build_sym_h(qq)
        g = corr_grad_k(hh, kk)
        j = np.zeros(m + 1)
        for r in range(m):
            j[r] = -dx * (g[r] - g[n_full - 1 - r])
        j[m] = 1.0
        return j
    
    x0 = np.zeros(m + 1)
    x0[:m] = q
    x0[m] = t0_val
    
    bounds = [(0.0, 1.0)] * m + [(0.0, 2.0)]
    constraints = []
    for kk in active_k:
        constraints.append({
            'type': 'ineq',
            'fun': lambda x, k=kk: con_fun(x, k),
            'jac': lambda x, k=kk: con_jac(x, k)
        })
    
    try:
        res = minimize(
            lambda x: x[m],
            x0,
            jac=lambda x: np.concatenate([np.zeros(m), [1.0]]),
            method='SLSQP', bounds=bounds, constraints=constraints,
            options={'maxiter': maxiter, 'ftol': 1e-12})
        q_out = np.clip(res.x[:m], 0, 1)
    except Exception:
        q_out = q.copy()
    
    h_out = build_sym_h(q_out)
    return q_out, h_out, eval_score(h_out)

def epig_full_solve(h, active_k, maxiter=50):
    n = len(h)
    dx = 2.0 / n
    corrs_c = corr_raw(h) * dx
    t0_val = corrs_c.max() + 1e-9
    
    def con_fun(x, kk):
        hh = x[:n]
        t = x[n]
        return t - corr_at_k(hh, kk) * dx
    
    def con_jac(x, kk):
        hh = x[:n]
        g = corr_grad_k(hh, kk)
        j = np.zeros(n + 1)
        j[:n] = -dx * g
        j[n] = 1.0
        return j
    
    def eq_fun(x):
        return np.sum(x[:n]) - n / 2.0
    
    def eq_jac(x):
        j = np.zeros(n + 1)
        j[:n] = 1.0
        return j
    
    x0 = np.zeros(n + 1)
    x0[:n] = h
    x0[n] = t0_val
    
    bounds = [(0.0, 1.0)] * (n + 1)
    constraints = []
    for kk in active_k:
        constraints.append({
            'type': 'ineq',
            'fun': lambda x, k=kk: con_fun(x, k),
            'jac': lambda x, k=kk: con_jac(x, k)
        })
    constraints.append({'type': 'eq', 'fun': eq_fun, 'jac': eq_jac})
    
    try:
        res = minimize(
            lambda x: x[n],
            x0,
            jac=lambda x: np.concatenate([np.zeros(n), [1.0]]),
            method='SLSQP', bounds=bounds, constraints=constraints,
            options={'maxiter': maxiter, 'ftol': 1e-12})
        h_out = project_feasible(res.x[:n], n / 2.0)
    except Exception:
        h_out = h.copy()
    
    return h_out, eval_score(h_out)

def transfer_to_n(h_old, n_new):
    n_old = len(h_old)
    x_old = (np.arange(n_old) + 0.5) * 2.0 / n_old

    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new / 2.0)

import numpy as np
from scipy.optimize import minimize, linprog
import time

def eval_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def corr_raw(h):
    return np.correlate(h, 1.0 - h, mode='full')

def corr_at_k(h, k):
    n = len(h)
    d = k - (n - 1)
    i_lo = max(0, -d)
    i_hi = min(n - 1, n - 1 - d)
    if i_lo > i_hi:
        return 0.0
    return float(np.sum(h[i_lo:i_hi + 1] * (1.0 - h[i_lo + d:i_hi + 1 + d])))

def corr_grad_k(h, k):
    n = len(h)
    d = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -d)
    i_hi = min(n - 1, n - 1 - d)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx] += (1.0 - h[idx + d])
    grad[idx + d] += (-h[idx])
    return grad

def project_feasible(x, target_sum):
    n = len(x)
    x = np.clip(x.astype(float), 0.0, 1.0)
    if abs(x.sum() - target_sum) < 1e-12:
        return x
    lo, hi = -1.5, 1.5
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(x + mid, 0.0, 1.0).sum()
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    h = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    lb = np.maximum(0.0, h - rho)
    ub = np.minimum(1.0, h + rho)
    bounds = list(zip(lb, ub)) + [(0.0, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            return res.x[:n]
    except Exception:
        pass
    return None

def directional_line_search(h, direction, current_score, n_grid=64):
    n = len(h)
    s = direction.sum()
    if abs(s) > 1e-10:
        direction = direction - s / n
    
    if np.max(np.abs(direction)) < 1e-12:
        return h, current_score
    
    lo_a, hi_a = 0.0, 1.0
    mask_pos = direction > 1e-12
    mask_neg = direction < -1e-12
    if mask_pos.any():
        hi_a = min(hi_a, ((1.0 - h[mask_pos]) / direction[mask_pos]).min())
    if mask_neg.any():
        lo_a = max(lo_a, (h[mask_neg] / (-direction[mask_neg])).min())
    
    lo_a = max(0.0, lo_a)
    hi_a = min(2.0, hi_a)
    if hi_a <= lo_a:
        return h, current_score
    
    best_alpha = 0.0
    best_sc = current_score
    grid = np.linspace(lo_a, hi_a, n_grid)
    for alpha in grid:
        h_new = h + alpha * direction
        if h_new.min() < -1e-9 or h_new.max() > 1.0 + 1e-9:
            continue
        sc = eval_score(h_new)
        if sc < best_sc:
            best_sc = sc
            best_alpha = alpha
    
    if best_alpha > 0:
        h_new = h + best_alpha * direction
        h_new = project_feasible(h_new, n / 2.0)
        best_sc = eval_score(h_new)
        return h_new, best_sc
    return h, current_score

def transfer_line_search(h, a, b, n_grid=48):
    n = len(h)
    current_score = eval_score(h)
    lo = max(-h[a], h[b] - 1.0)
    hi = min(1.0 - h[a], -h[b])
    if hi - lo < 1e-12:
        return h, current_score
    best_delta = 0.0
    best_sc = current_score
    grid = np.linspace(lo, hi, n_grid)
    for delta in grid:
        h[a] += delta
        h[b] -= delta
        sc = eval_score(h)
        h[a] -= delta
        h[b] += delta
        if sc < best_sc:
            best_sc = sc
            best_delta = delta
    margin = (hi - lo) / n_grid * 1.5
    lo_r = max(lo, best_delta - margin)
    hi_r = min(hi, best_delta + margin)
    if hi_r > lo_r:
        grid2 = np.linspace(lo_r, hi_r, 24)
        for delta in grid2:
            h[a] += delta
            h[b] -= delta
            sc = eval_score(h)
            h[a] -= delta
            h[b] += delta
            if sc < best_sc:
                best_sc = sc
                best_delta = delta
    if best_delta != 0.0:
        h[a] += best_delta
        h[b] -= best_delta
        h = project_feasible(h, n / 2.0)
        best_sc = eval_score(h)
    return h, best_sc

def coord_descent_step(h, n_pairs=100, rng=None):
    n = len(h)
    current_score = eval_score(h)
    corrs = corr_raw(h) * (2.0 / n)
    max_val = corrs.max()
    hot_lags = np.where(corrs >= max_val - 0.08 * max_val)[0]
    
    important = set()
    for k in hot_lags[:20]:
        d = k - (n - 1)
        i_lo = max(0, -d)
        i_hi = min(n - 1, n - 1 - d)
        important.update(range(i_lo, i_hi + 1))
        important.update(range(i_lo + d, i_hi + d + 1))
    imp_list = sorted(important)
    if len(imp_list) < 4:
        imp_list = list(range(n))
    
    pairs = []
    for _ in range(n_pairs // 2):
        a = int(rng.choice(imp_list))
        b = int(rng.choice(imp_list))
        if a != b:
            pairs.append((a, b))
    for _ in range(n_pairs // 2):
        a = int(rng.randint(n))
        b = int(rng.randint(n))
        if a != b:
            pairs.append((a, b))
    
    improved = False
    for (a, b) in pairs:
        if a == b:
            continue
        h_new, sc_new = transfer_line_search(h, a, b, n_grid=36)
        if sc_new < current_score - 1e-12:
            h = h_new
            current_score = sc_new
            improved = True
    return h, current_score, improved

def smooth_max_obj_grad_q(q, n_full, beta):
    m = n_full // 2
    h = build_sym_h(q)
    dx = 2.0 / n_full
    corrs = corr_raw(h) * dx
    max_c = corrs.max()
    shifted = corrs - max_c
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(len(corrs)) / len(corrs)
    else:
        w /= w_sum
    val = max_c + np.log(w_sum) / beta
    grad_q = np.zeros(m)
    for k in range(2 * n_full - 1):
        if w[k] < 1e-12:
            continue
        g = corr_grad_k(h, k)
        for r in range(m):
            grad_q[r] += w[k] * (g[r] - g[n_full - 1 - r])
    grad_q *= dx
    return val, grad_q

def smooth_sym_optimize(q0, n_full, beta_list, maxiter_per_beta, time_limit):
    m = n_full // 2
    q = np.clip(q0.copy(), 0, 1)
    best_score = eval_score(build_sym_h(q))
    best_q = q.copy()
    for beta in beta_list:
        if time.time() > time_limit:
            break
        bounds = [(0.0, 1.0)] * m
        try:
            res = minimize(
                lambda xx: smooth_max_obj_grad_q(xx, n_full, beta)[0],
                q,
                jac=lambda xx: smooth_max_obj_grad_q(xx, n_full, beta)[1],
                method='L-BFGS-B', bounds=bounds,
                options={'maxiter': maxiter_per_beta, 'ftol': 1e-14, 'gtol': 1e-10})
            q_new = np.clip(res.x, 0, 1)
        except Exception:
            q_new = q.copy()
        h_new = build_sym_h(q_new)
        sc = eval_score(h_new)
        if sc < best_score:
            best_score = sc
            best_q = q_new.copy()
        q = q_new
    return best_q, best_score, build_sym_h(best_q)

def epig_sym_solve(q, n_full, active_k, maxiter=40):
    m = n_full // 2
    h_c = build_sym_h(q)
    dx = 2.0 / n_full
    t0_val = float(corr_raw(h_c).max()) * dx + 1e-9

    def con_fun(x, kk):
        qq = x[:m]
        t = x[m]
        hh = build_sym_h(qq)
        return t - corr_at_k(hh, kk) * dx

    def con_jac(x, kk):
        qq = x[:m]
        hh = build_sym_h(qq)
        g = corr_grad_k(hh, kk)
        j = np.zeros(m + 1)
        for r in range(m):
            j[r] = -dx * (g[r] - g[n_full - 1 - r])
        j[m] = 1.0
        return j

    x0 = np.zeros(m + 1)
    x0[:m] = q
    x0[m] = t0_val
    bounds = [(0.0, 1.0)] * m + [(0.0, 2.0)]
    constraints = [{'type': 'ineq', 'fun': lambda x, k=kk: con_fun(x, k),
                    'jac': lambda x, k=kk: con_jac(x, k)} for kk in active_k]
    try:
        res = minimize(lambda x: x[m], x0,
                       jac=lambda x: np.concatenate([np.zeros(m), [1.0]]),
                       method='SLSQP', bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        q_out = np.clip(res.x[:m], 0, 1)
    except Exception:
        q_out = q.copy()
    h_out = build_sym_h(q_out)
    return q_out, h_out, eval_score(h_out)

def transfer_to_n(h_old, n_new):
    n_old = len(h_old)
    x_old = (np.arange(n_old) + 0.5) * 2.0 / n_old
    x_new = (np.arange(n_new) + 0.5) * 2.0 / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def rem():
        return budget_s - (time.time() - t0)

    best_h = None
    best_c5 = float('inf')

    try:
        h_parent = np.array(initial_h_values, dtype=float).copy().reshape(-1)
    except (NameError, AttributeError):
        h_parent = None

    if h_parent is not None and len(h_parent) > 10:
        n_p = len(h_parent)
        h_parent = np.clip(h_parent, 0.0, 1.0)
        h_parent = project_feasible(h_parent, n_p / 2.0)
        best_h = h_parent.copy()
        best_c5 = eval_score(h_parent)
    else:
        m_p = 63
        x = np.linspace(0, 1, m_p)
        q_p = (0.5 + 0.3 * np.sin(2 * np.pi * x)).clip(0, 1)
        best_h = build_sym_h(q_p)
        best_c5 = eval_score(best_h)

    n_cur = len(best_h)

    # --- LP-based polish (main driver) ---
    def lp_polish(h, max_iter=50, rho_start=0.12, use_all=False):
        nonlocal best_h, best_c5
        n = len(h)
        h = h.copy()
        rho = rho_start
        stall = 0
        for it in range(max_iter):
            if rem() < 30:
                break
            active = None if use_all else select_active(
                corr_raw(h) * (2.0 / n), margin_frac=0.02, neighbor_pad=2, max_k=50)
            z = solve_lp_direction(h, active_k=active, rho=rho, use_all=use_all)
            if z is None:
                break
            direction = z - h
            if np.max(np.abs(direction)) < 1e-9:
                stall += 1
                if stall > 4:
                    break
                rho *= 0.5
                continue
            h_new, sc_new = directional_line_search(h, direction, best_c5, n_grid=56)
            if sc_new < best_c5 - 1e-13:
                best_c5 = sc_new
                best_h = h_new.copy()
                h = h_new
                stall = 0
                rho = min(0.15, rho * 1.15)
            else:
                stall += 1
                rho *= 0.55
                if rho < 0.005:
                    break
        return h, best_c5

    # --- Coordinate descent ---
    def cd_polish(h, max_rounds=25, n_pairs=100):
        nonlocal best_h, best_c5
        h = h.copy()
        for rnd in range(max_rounds):
            if rem() < 30:
                break
            h, sc, improved = coord_descent_step(h, n_pairs=n_pairs, rng=rng)
            if sc < best_c5 - 1e-12:
                best_c5 = sc
                best_h = h.copy()
            if not improved:
                break
        return h, best_c5

    # --- Epigraph SLSQP ---
    def epig_polish(h, max_rounds=15):
        nonlocal best_h, best_c5
        n = len(h)
        h = h.copy()
        for rnd in range(max_rounds):
            if rem() < 40:
                break
            corrs = corr_raw(h) * (2.0 / n)
            active = select_active(corrs, margin_frac=0.03, neighbor_pad=3, max_k=50)
            if n % 2 == 0:
                q = sym_from_h(h)
                q_new, h_new, sc_new = epig_sym_solve(q, n, active, maxiter=45)
            else:
                h_new, sc_new = h.copy(), best_c5
                # Full epigraph fallback
                from scipy.optimize import minimize as _min
                dx = 2.0 / n
                t0v = corrs.max() + 1e-9
                x0 = np.zeros(n + 1)
                x0[:n] = h
                x0[n] = t0v
                cons = []
                for kk in active:
                    def cf(x, k=kk):
                        return x[n] - corr_at_k(x[:n], k) * dx
                    def cj(x, k=kk):
                        g = corr_grad_k(x[:n], k)
                        j = np.zeros(n + 1)
                        j[:n] = -dx * g
                        j[n] = 1.0
                        return j
                    cons.append({'type': 'ineq', 'fun': cf, 'jac': cj})
                def ef(x):
                    return x[:n].sum() - n / 2.0
                def ej(x):
                    j = np.zeros(n + 1)
                    j[:n] = 1.0
                    return j
                cons.append({'type': 'eq', 'fun': ef, 'jac': ej})
                try:
                    res = _min(lambda x: x[n], x0,
                               jac=lambda x: np.concatenate([np.zeros(n), [1.0]]),
                               method='SLSQP', bounds=[(0,1)]*(n+1),
                               constraints=cons, options={'maxiter': 40, 'ftol': 1e-12})
                    h_new = project_feasible(res.x[:n], n / 2.0)
                    sc_new = eval_score(h_new)
                except Exception:
                    h_new, sc_new = h.copy(), best_c5
            if sc_new < best_c5 - 1e-12:
                best_c5 = sc_new
                best_h = h_new.copy()
                h = h_new
            else:
                corrs_new = corr_raw(h_new) * (2.0 / n)
                max_act = corrs_new[active].max()
                hidden = np.where(corrs_new > max_act + 2e-7)[0]
                if len(hidden) > 0:
                    active = sorted(set(active + list(hidden)))
                    h = h_new
                    continue
                break
        return h, best_c5

    # Phase 1: LP polish (full)
    if rem() > 200:
        best_h, best_c5 = lp_polish(best_h, max_iter=35, rho_start=0.12, use_all=True)

    # Phase 2: Coordinate descent
    if rem() > 150:
        best_h, best_c5 = cd_polish(best_h, max_rounds=20, n_pairs=100)

    # Phase 3: Epigraph
    if rem() > 120:
        best_h, best_c5 = epig_polish(best_h, max_rounds=12)

    # Phase 4: Multi-resolution
    n_base = len(best_h)
    n_candidates = sorted(set(
        n_base + d for d in [-6, -4, -3, -2, -1, 1, 2, 3, 4, 6]
        if 44 <= n_base + d <= 200
    ))
    for n_try in n_candidates:
        if rem() < 80:
            break
        h_try = transfer_to_n(best_h, n_try)
        if rem() > 60:
            h_try, _ = lp_polish(h_try, max_iter=20, rho_start=0.10, use_all=True)
        if rem() > 50:
            h_try, _ = epig_polish(h_try, max_rounds=8)
        if rem() > 40:
            h_try, _ = cd_polish(h_try, max_rounds=8, n_pairs=80)

    # Phase 5: Alternating polish loop
    for _ in range(6):
        if rem() < 70:
            break
        best_h, best_c5 = lp_polish(best_h, max_iter=15, rho_start=0.06, use_all=True)
        if rem() < 60:
            break
        best_h, best_c5 = cd_polish(best_h, max_rounds=12, n_pairs=80)
        if rem() < 50:
            break
        best_h, best_c5 = epig_polish(best_h, max_rounds=8)

    # Phase 6: Final high-precision LP
    if rem() > 30:
        best_h, best_c5 = lp_polish(best_h, max_iter=30, rho_start=0.03, use_all=True)
    if rem() > 20:
        best_h, best_c5 = cd_polish(best_h, max_rounds=12, n_pairs=80)

    # Validate
    best_h = np.asarray(best_h, dtype=float).reshape(-1)
    n_out = len(best_h)
    best_h = project_feasible(best_h, n_out / 2.0)
    best_c5 = eval_score(best_h)
    return best_h, best_c5, n_out
