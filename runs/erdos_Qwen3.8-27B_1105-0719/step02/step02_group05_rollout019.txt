def grad_corr_d(h, d, n):
    g = np.zeros(n)
    if d >= 0:
        if d < n:
            g[d:] += (1.0 - h[:n - d])
            g[:n - d] -= h[d:]
    else:
        dd = -d
        if dd < n:
            g[:n - dd] += (1.0 - h[dd:])
            g[dd:] -= h[:n - dd]
    return g

def smooth_max_value_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corrs = compute_corrs(h)
    bcorr = beta * corrs
    mx = bcorr.max()
    val = (mx + np.log(np.exp(bcorr - mx).sum())) / beta * dx
    shifted = bcorr - mx
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(len(corrs)) / len(corrs)
    else:
        w /= w_sum
    grad = np.zeros(n)

        if w[k] < 1e-14:
            continue
        d = k - (n - 1)
        grad += w[k] * grad_corr_d(h, d, n)
    return val, grad * dx

        return x[1:].sum() - self.target_sum

    def sum_grad(self, x):
        g = np.zeros(self.n + 1)
        g[1:] = np.ones(self.n)
        return g


class EpigraphQ:
    def __init__(self, n, active_k):
        self.n = n
        self.active_k = active_k
        self.dx = 2.0 / n
        self.m = n // 2

    def obj(self, x):
        return x[0]

    def obj_grad(self, x):
        g = np.zeros(self.m + 1)
        g[0] = 1.0
        return g

    def constraint_fun(self, x, idx):
        k = self.active_k[idx]
        q = x[1:]
        h = build_h_from_q(q, self.n)
        c = compute_corrs(h)
        return x[0] - c[k] * self.dx

    def constraint_grad(self, x, idx):
        k = self.active_k[idx]
        d = k - (self.n - 1)
        q = x[1:]
        h = build_h_from_q(q, self.n)
        G = grad_corr_d(h, d, self.n)
        Gq = G[:self.m] - G[::-1][:self.m]
        g = np.zeros(self.m + 1)
        g[0] = 1.0
        g[1:] = -Gq * self.dx
        return g


class CorrConFull:
    def __init__(self, ep, idx):
        self.ep = ep
        self.idx = idx

    def fun(self, x):
        return self.ep.constraint_fun(x, self.idx)

    def jac(self, x):
        return self.ep.constraint_grad(x, self.idx)


class SumConFull:
    def __init__(self, ep):
        self.ep = ep

    def fun(self, x):
        return self.ep.sum_fun(x)

    def jac(self, x):
        return self.ep.sum_grad(x)


class CorrConQ:
    def __init__(self, ep, idx):
        self.ep = ep
        self.idx = idx

    def fun(self, x):
        return self.ep.constraint_fun(x, self.idx)

    def jac(self, x):
        return self.ep.constraint_grad(x, self.idx)


def solve_epigraph_full(h0, active_k, target_sum, t0, maxiter=150):
    n = len(h0)
    ep = EpigraphFull(n, active_k, target_sum)
    x0 = np.concatenate([[t0], h0])
    cons = []
    for i in range(len(active_k)):
        cc = CorrConFull(ep, i)
        cons.append({'type': 'ineq', 'fun': cc.fun, 'jac': cc.jac})
    sc = SumConFull(ep)
    cons.append({'type': 'eq', 'fun': sc.fun, 'jac': sc.jac})
    bounds = [(0.0, 1.0)] + [(0.0, 1.0)] * n
    try:
        res = minimize(ep.obj, x0, jac=ep.obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        h_out = project_feasible(res.x[1:], target_sum)
        return h_out, res.x[0]
    except Exception:
        return h0.copy(), t0


def solve_epigraph_q(q0, active_k, t0, n, maxiter=150):
    ep = EpigraphQ(n, active_k)
    m = n // 2
    x0 = np.concatenate([[t0], q0])
    cons = []
    for i in range(len(active_k)):
        cc = CorrConQ(ep, i)
        cons.append({'type': 'ineq', 'fun': cc.fun, 'jac': cc.jac})
    bounds = [(0.0, 2.0)] + [(0.0, 1.0)] * m
    try:
        res = minimize(ep.obj, x0, jac=ep.obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        q_out = np.clip(res.x[1:], 0.0, 1.0)
        return q_out, res.x[0]
    except Exception:
        return q0.copy(), t0


def full_active_search(h0, target_sum, t_start, max_rounds=5):
    n = len(h0)
    dx = 2.0 / n
    h = project_feasible(h0.copy(), target_sum)
    corrs = compute_corrs(h)
    current_max = corrs.max() * dx
    top_k = np.argsort(corrs)[::-1][:min(20, 2 * n - 1)]
    active_k = sorted(top_k.tolist())
    t0 = current_max + 1e-4
    best_h = h.copy()
    best_c5 = current_max

    for rnd in range(max_rounds):
        if time.time() - t_start > budget_s_global - 40:
            break
        h_new, t_solve = solve_epigraph_full(h, active_k, target_sum, t0, maxiter=120)
        h_new = project_feasible(h_new, target_sum)
        corrs_new = compute_corrs(h_new)
        c5_new = corrs_new.max() * dx

        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()

        hidden = corrs_new > t_solve + 1e-7
        if hidden.sum() > 0:
            hidden_idx = np.where(hidden)[0]
            new_active = sorted(set(active_k) | set(hidden_idx.tolist()))
            if len(new_active) > 40:
                vals = np.array([corrs_new[k] for k in new_active])
                top = np.argsort(vals)[::-1][:40]
                new_active = sorted([new_active[i] for i in top])
            active_k = new_active

        t0 = max(t_solve, c5_new) + 1e-5
        h = h_new
        if abs(c5_new - best_c5) < 1e-13 and hidden.sum() == 0:
            break

    return best_h, best_c5


def q_active_search(q0, n, t_start, max_rounds=5):
    m = n // 2
    h = build_h_from_q(q0, n)
    corrs = compute_corrs(h)
    dx = 2.0 / n
    current_max = corrs.max() * dx
    top_k = np.argsort(corrs)[::-1][:min(20, 2 * n - 1)]
    active_k = sorted(top_k.tolist())
    t0 = current_max + 1e-4
    best_q = q0.copy()
    best_c5 = current_max
    q = q0.copy()

    for rnd in range(max_rounds):
        if time.time() - t_start > budget_s_global - 40:
            break
        q_new, t_solve = solve_epigraph_q(q, active_k, t0, n, maxiter=120)
        h_new = build_h_from_q(q_new, n)
        corrs_new = compute_corrs(h_new)
        c5_new = corrs_new.max() * dx

        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_q = q_new.copy()

        hidden = corrs_new > t_solve + 1e-7
        if hidden.sum() > 0:
            hidden_idx = np.where(hidden)[0]
            new_active = sorted(set(active_k) | set(hidden_idx.tolist()))
            if len(new_active) > 40:
                vals = np.array([corrs_new[k] for k in new_active])
                top = np.argsort(vals)[::-1][:40]
                new_active = sorted([new_active[i] for i in top])
            active_k = new_active

        t0 = max(t_solve, c5_new) + 1e-5
        q = q_new
        if abs(c5_new - best_c5) < 1e-13 and hidden.sum() == 0:
            break

    return best_q, best_c5


budget_s_global = 1100


def run(seed=42, budget_s=1100, **kwargs):
    global budget_s_global
    budget_s_global = budget_s
    t_start = time.time()
    rng = np.random.RandomState(seed)

    def elapsed():
        return time.time() - t_start

    def remaining():
        return budget_s - elapsed()

    # Initialize from parent
    h_parent = np.array(initial_h_values, dtype=float)
    n = len(h_parent)
    target_sum = n / 2.0
    h_parent = project_feasible(h_parent, target_sum)

    best_h = h_parent.copy()
    best_c5 = exact_score(best_h)

    # Phase 1: q-space smooth L-BFGS-B with increasing beta
    if n % 2 == 0 and remaining() > 30:
        m = n // 2
        # Check if parent is approximately symmetric; if not, project to symmetric
        q_init = h_parent[:m].copy()
        h_sym = build_h_from_q(q_init, n)
        # Only use symmetric if it's close to parent
        if np.abs(h_sym - h_parent).max() < 0.5:
            q_current = q_init
        else:
            # Use the symmetric part as starting point
            q_current = (h_parent[:m] + (1.0 - h_parent[n-1:m-1:-1])) / 2.0
            q_current = np.clip(q_current, 0.0, 1.0)

        betas_q = [15, 40, 100, 200, 400, 800, 1500, 2500, 4000]
        for beta in betas_q:
            if remaining() < 25:
                break
            so = SmoothQObj(n, beta)
            try:
                res = minimize(so.value, q_current.copy(), jac=so.gradient,
                               method='L-BFGS-B',
                               bounds=[(0.0, 1.0)] * m,
                               options={'maxiter': 250, 'ftol': 1e-15, 'gtol': 1e-10})
                q_new = np.clip(res.x, 0.0, 1.0)
                h_new = build_h_from_q(q_new, n)
                s_new = exact_score(h_new)
                if s_new < best_c5 - 1e-14:
                    best_c5 = s_new
                    best_h = h_new.copy()
                q_current = q_new
            except Exception:
                pass

    # Phase 2: Full-space smooth SLSQP
    if remaining() > 40:
        h_current = project_feasible(best_h.copy(), target_sum)
        betas_full = [30, 80, 200, 500, 1000, 2000]
        for beta in betas_full:
            if remaining() < 30:
                break
            so = SmoothFullObj(beta)
            cons = []
            sc = SumConFull(EpigraphFull(n, [], target_sum))
            cons.append({'type': 'eq', 'fun': sc.fun, 'jac': sc.jac})
            try:
                res = minimize(so.value, h_current.copy(), jac=so.gradient,
                               method='SLSQP',
                               bounds=[(0.0, 1.0)] * n,
                               constraints=cons,
                               options={'maxiter': 150, 'ftol': 1e-12})
                h_new = project_feasible(res.x, target_sum)
                s_new = exact_score(h_new)
                if s_new < best_c5 - 1e-14:
                    best_c5 = s_new
                    best_h = h_new.copy()
                h_current = h_new
            except Exception:
                pass

    # Phase 3: Active-set epigraph in full space
    if remaining() > 50:
        h_es, c5_es = full_active_search(best_h, target_sum, t_start, max_rounds=5)
        s_es = exact_score(h_es)
        if s_es < best_c5 - 1e-14:
            best_c5 = s_es
            best_h = h_es.copy()

    # Phase 4: Active-set epigraph in q-space
    if n % 2 == 0 and remaining() > 40:
        m = n // 2
        q_cur = q_from_h(best_h, n)
        # Ensure symmetry for q-space
        h_sym = build_h_from_q(q_cur, n)
        h_sym = project_feasible(h_sym, target_sum)
        s_sym = exact_score(h_sym)
        if s_sym <= best_c5 + 1e-10:
            q_es, c5_es = q_active_search(q_cur, n, t_start, max_rounds=5)
            h_es = build_h_from_q(q_es, n)
            h_es = project_feasible(h_es, target_sum)
            s_es = exact_score(h_es)
            if s_es < best_c5 - 1e-14:
                best_c5 = s_es
                best_h = h_es.copy()

    # Phase 5: Random perturbation + local reoptimization
    if remaining() > 60:
        n_cur = len(best_h)
        ts_cur = n_cur / 2.0
        n_m = n_cur // 2

        for attempt in range(15):
            if remaining() < 25:
                break
            # Generate perturbation
            noise_scale = 0.03 + 0.04 * (attempt % 3)
            h_pert = best_h + rng.randn(n_cur) * noise_scale
            h_pert = project_feasible(h_pert, ts_cur)

            # Try q-space if possible
            if n_cur % 2 == 0:
                q_pert = q_from_h(h_pert, n_cur)
                q_opt = q_pert.copy()
                for beta in [100, 500]:
                    if remaining() < 15:
                        break
                    so = SmoothQObj(n_cur, beta)
                    try:
                        res = minimize(so.value, q_opt.copy(), jac=so.gradient,
                                       method='L-BFGS-B',
                                       bounds=[(0.0, 1.0)] * n_m,
                                       options={'maxiter': 150, 'ftol': 1e-14})
                        q_opt = np.clip(res.x, 0.0, 1.0)
                    except Exception:
                        pass
                h_new = build_h_from_q(q_opt, n_cur)
                h_new = project_feasible(h_new, ts_cur)
                s_new = exact_score(h_new)
                if s_new < best_c5 - 1e-12:
                    best_c5 = s_new
                    best_h = h_new.copy()

            # Also try full-space from perturbed point
            if remaining() > 15:
                h_full_pert = project_feasible(h_pert, ts_cur)
                # Quick active set solve
                if remaining() > 20:
                    h_fes, c5_fes = full_active_search(h_full_pert, ts_cur, t_start, max_rounds=2)
                    s_fes = exact_score(h_fes)
                    if s_fes < best_c5 - 1e-12:
                        best_c5 = s_fes
                        best_h = h_fes.copy()

    # Phase 6: Resolution search
    if remaining() > 80:
        n_cur = len(best_h)
        candidates_n = []
        for dn in [-6, -4, -2, 2, 4, 6, 8, 12, 16]:
            nn = n_cur + dn
            if 40 <= nn <= 300 and nn % 2 == 0:
                candidates_n.append(nn)
        candidates_n = sorted(set(candidates_n))

        for n_try in candidates_n:
            if remaining() < 50:
                break
            ts_try = n_try / 2.0
            # Interpolate
            x_orig = np.linspace(0.0, 2.0, len(best_h))
            x_new = np.linspace(0.0, 2.0, n_try)
            h_interp = np.interp(x_new, x_orig, best_h)
            h_interp = project_feasible(h_interp, ts_try)
            s_interp = exact_score(h_interp)

            # Quick optimization at this resolution
            if n_try % 2 == 0:
                q_int = q_from_h(h_interp, n_try)
                q_opt = q_int.copy()
                for beta in [50, 200, 500]:
                    if remaining() < 20:
                        break
                    so = SmoothQObj(n_try, beta)
                    try:
                        res = minimize(so.value, q_opt.copy(), jac=so.gradient,
                                       method='L-BFGS-B',
                                       bounds=[(0.0, 1.0)] * (n_try // 2),
                                       options={'maxiter': 150, 'ftol': 1e-14})
                        q_opt = np.clip(res.x, 0.0, 1.0)
                    except Exception:
                        pass
                h_opt = build_h_from_q(q_opt, n_try)
                h_opt = project_feasible(h_opt, ts_try)
                s_opt = exact_score(h_opt)
                if s_opt < best_c5 - 1e-10:
                    best_c5 = s_opt
                    best_h = h_opt.copy()

                # Active set if time permits
                if remaining() > 35 and s_opt < best_c5 + 5e-4:
                    h_aes, c5_aes = full_active_search(h_opt, ts_try, t_start, max_rounds=3)
                    s_aes = exact_score(h_aes)
                    if s_aes < best_c5 - 1e-10:
                        best_c5 = s_aes
                        best_h = h_aes.copy()

    # Phase 7: Final polish - one more round of both spaces
    n_fin = len(best_h)
    ts_fin = n_fin / 2.0

    if remaining() > 30:
        h_fin = project_feasible(best_h.copy(), ts_fin)
        h_fin2, c5_fin2 = full_active_search(h_fin, ts_fin, t_start, max_rounds=3)
        s_fin2 = exact_score(h_fin2)
        if s_fin2 < best_c5 - 1e-14:
            best_c5 = s_fin2
            best_h = h_fin2.copy()

    if n_fin % 2 == 0 and remaining() > 20:
        m_fin = n_fin // 2
        q_fin = q_from_h(best_h, n_fin)
        q_fin2, c5_fin2 = q_active_search(q_fin, n_fin, t_start, max_rounds=3)
        h_fin2 = build_h_from_q(q_fin2, n_fin)
        h_fin2 = project_feasible(h_fin2, ts_fin)
        s_fin2 = exact_score(h_fin2)
        if s_fin2 < best_c5 - 1e-14:
            best_c5 = s_fin2
            best_h = h_fin2.copy()

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    n_out = len(best_h)
    ts_out = n_out / 2.0
    best_h = project_feasible(best_h, ts_out)
    final_score = exact_score(best_h)
    if final_score < best_c5:
        best_c5 = final_score

    return best_h, best_c5, n_out
