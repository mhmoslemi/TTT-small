def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corrs = np.correlate(h, 1.0 - h, mode='full')
    return (corrs.max() * dx, corrs * dx)

def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def build_h_from_q(q, n):
    m = n // 2
    if n % 2 == 0:
        return np.concatenate([q, 1.0 - q[::-1]])
    else:
        return np.concatenate([q, [0.5], 1.0 - q[::-1]])

def q_from_h(h, n):
    m = n // 2
    return h[:m].copy()

def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def grad_corr_d(h, d, n):
    """Gradient of correlation at lag d with respect to h."""
    g = np.zeros(n)
    if d >= 0:
        if d < n:
            g[d:] += (1.0 - h[:n - d])
            g[:n - d] += (-h[d:])
    else:
        dd = -d
        if dd < n:
            g[:n - dd] += (1.0 - h[dd:])
            g[dd:] += (-h[:n - dd])
    return g

def smooth_q_obj(x, n, beta):
    """Smooth objective in q-space using log-sum-exp."""

    h = build_h_from_q(x, n)
    dx = 2.0 / n

def smooth_q_grad(x, n, beta):
    """Gradient of smooth objective in q-space."""
    m = n // 2
    h = build_h_from_q(x, n)
    dx = 2.0 / n
    corrs = compute_corrs(h) * dx

        d = k - (n - 1)
        G = grad_corr_d(h, d, n) * dx

def pnorm_q_grad(x, n, p):
    """Gradient of p-norm objective in q-space."""
    m = n // 2
    h = build_h_from_q(x, n)
    dx = 2.0 / n
    corrs = compute_corrs(h) * dx
    norm = (np.sum(corrs**p) / len(corrs))**(1.0/p)
    # d/ds_k of (sum(s^p)/K)^(1/p) = (s_k^(p-1) / sum(s^p)) * (sum(s^p)/K)^(1/p-1) / K^(1/p)
    # Actually: F = (1/K * sum(s_k^p))^(1/p)

class EpigraphQSolver:
    def __init__(self, n, active_k):
        self.n = n
        self.m = n // 2
        self.active_k = active_k
        self.dx = 2.0 / n

    def _build_h(self, q):
        n = self.n
        m = self.m
        return np.concatenate([q, 1.0 - q[::-1]])

    def obj(self, x):
        return x[0]

    def obj_grad(self, x):
        g = np.zeros(self.m + 1)
        g[0] = 1.0
        return g

    def constraint(self, x, k_idx):
        t = x[0]
        q = x[1:]
        h = self._build_h(q)
        k = self.active_k[k_idx]
        d = k - (self.n - 1)
        n = self.n
        if d >= 0:
            i_vals = np.arange(0, n - d)
            val = np.sum(h[i_vals + d] * (1.0 - h[i_vals]))
        else:
            dd = -d
            i_vals = np.arange(dd, n)
            val = np.sum(h[i_vals] * (1.0 - h[i_vals - dd]))
        return t - val * self.dx

    def constraint_grad(self, x, k_idx):
        q = x[1:]
        h = self._build_h(q)
        k = self.active_k[k_idx]
        d = k - (self.n - 1)
        n = self.n
        g = np.zeros(self.m + 1)
        g[0] = 1.0
        G = grad_corr_d(h, d, n) * self.dx
        g[1:] = -(G[:self.m] - G[self.m:][::-1])
        return g

def solve_epigraph_q(q0, n, active_k, t0, maxiter=300):
    m = n // 2
    x0 = np.concatenate([[t0], q0])
    ep = EpigraphQSolver(n, active_k)
    cons = []
    for i in range(len(active_k)):
        cons.append({
            'type': 'ineq',
            'fun': lambda x, i=i: ep.constraint(x, i),
            'jac': lambda x, i=i: ep.constraint_grad(x, i)
        })
    bounds = [(0.0, 5.0)] + [(0.0, 1.0)] * m
    try:
        res = minimize(ep.obj, x0, jac=ep.obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        q_out = np.clip(res.x[1:], 0.0, 1.0)
        t_out = max(res.x[0], 0.0)

        return q0, t0, -1
