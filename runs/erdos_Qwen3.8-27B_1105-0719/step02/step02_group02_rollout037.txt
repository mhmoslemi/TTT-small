    n_b = len(best_h)
    dx_b = 2.0 / n_b
    ts_b = n_b / 2.0
    if budget_s - elapsed() > 20:
        corrs = np.correlate(best_h, 1.0 - best_h, mode='full')
        top_m = min(2 * n_b - 1, max(15, n_b // 3))
        active = np.argsort(corrs)[-top_m:]
        h_fin, c5_fin = slsqp_active(best_h, dx_b, ts_b, active, 5000, 800)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin

    if budget_s - elapsed() > 15:
        corrs = np.correlate(best_h, 1.0 - best_h, mode='full')
        top_m = min(2 * n_b - 1, max(10, n_b // 4))
        active = np.argsort(corrs)[-top_m:]
        h_fin2, c5_fin2 = slsqp_active(best_h, dx_b, ts_b, active, 10000, 600)
        if c5_fin2 < best_c5:
            best_c5 = c5_fin2
            best_h = h_fin2

    # Final subgradient squeeze
    if budget_s - elapsed() > 10:
        h_sg2, c5_sg2 = subgrad_topk(best_h, dx_b, ts_b, 10000, min(15, n_b // 5))
        if c5_sg2 < best_c5:
            best_c5 = c5_sg2
            best_h = h_sg2

    return best_h, best_c5, len(best_h)
