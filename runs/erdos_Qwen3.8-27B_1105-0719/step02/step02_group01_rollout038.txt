where d_k(i,j) = -[1-h[i+shift_k]]_{valid} + [h[i-shift_k]]_{valid} + [1-h[j+shift_k]]_{valid} - [h[j-shift_k]]_{valid}

And the new C₅ is max_k (corr[k] + t*d_k(i,j)) * dx

To find the optimal t, we need to minimize max_k (corr[k] + t*d_k) over t in [0, t_max].

This is the minimum of a convex function (max of linear functions), which occurs at a point where two or more linear functions are equal, or at a boundary.

For efficiency, I can:
1. Only consider the "active" k's (those near the maximum)
2. Binary search on t using the exact max

Actually, for a cleaner approach, let me just use a "backtracking line search":
1. Start with t = t_max
2. Evaluate C₅ at t
3. If C₅(t) < C₅(0), accept (maybe try smaller t for more improvement)
4. If C₅(t) > C₅(0), halve t and retry

import numpy as np

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corrs = np.correlate(h, 1.0 - h, mode='full')
    return corrs.max() * dx

def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def quantized_sa(h_init, n, M, deadline, rng):

def slsqp_exact(h0, dx, target_sum, beta, maxiter=800):
    from scipy.optimize import minimize
    from scipy.special import logsumexp
    n = len(h0)

    def obj(x):
        c = compute_corrs(x)
        return logsumexp(beta * c) / beta * dx

    def grad_obj(x):
        c = compute_corrs(x)
        shifted = c - c.max()
        w = np.exp(beta * shifted)
        wsum = w.sum()
        if wsum < 1e-30:
            w = np.ones(len(c)) / len(c)
        else:
            w /= wsum
        g = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] < 1e-12:
                continue
            shift = k - (n - 1)
            j_lo = max(0, -shift)
            j_hi = min(n - 1, n - 1 - shift)
            if j_lo > j_hi:
                continue
            js = np.arange(j_lo, j_hi + 1)
            g[js] += w[k] * (1.0 - h_safe(js, x, shift, n))
            m_lo = max(0, j_lo + shift)
            m_hi = min(n - 1, j_hi + shift)
            if m_lo <= m_hi:
                ms = np.arange(m_lo, m_hi + 1)
                for idx_m in range(len(ms)):
                    m = ms[idx_m]
                    if m < j_lo or m > j_hi:
                        g[m] += w[k] * (-x[m - shift])
        return g * dx

    def h_safe(js, x, shift, n):
        return np.clip(x[js + shift], 0, 1)

    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0, jac=grad_obj, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        h_out = project_feasible(res.x, target_sum)
    except Exception:
        h_out = project_feasible(h0, target_sum)
    return h_out, evaluate_c5(h_out)
