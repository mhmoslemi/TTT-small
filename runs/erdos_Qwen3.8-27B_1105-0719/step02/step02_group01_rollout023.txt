import numpy as np
from scipy.optimize import linprog
import time


def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(100):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)


def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def c5_of_h(h, dx):
    return compute_corrs(h).max() * dx


def compute_G(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        shift = k - (n - 1)
        if shift >= 0:
            G[k, :n - shift] += (1.0 - h[shift:])
            G[k, shift:n] += (-h[:n - shift])
        else:
            s = -shift
            G[k, s:] += (1.0 - h[:n - s])
            G[k, :n - s] += (-h[s:])
    return G


def lp_direction(h, target_sum, n_active):
    n = len(h)
    corrs = compute_corrs(h)
    if n_active >= 2 * n - 1:
        top_idx = np.arange(2 * n - 1)
    else:
        top_idx = np.argsort(corrs)[-n_active:]
    G = compute_G(h)
    G_act = G[top_idx, :]
    nv = n + 1
    c_vec = np.zeros(nv)
    c_vec[-1] = -1.0
    A_rows = []
    b_rows = []
    for i in range(min(n_active, 2 * n - 1)):
        row = np.zeros(nv)
        row[:n] = G_act[i]
        row[n] = 1.0
        A_rows.append(row)
        b_rows.append(1e-12)
    for i in range(n):
        row = np.zeros(nv)
        row[i] = -1.0
        A_rows.append(row)
        b_rows.append(h[i])
        row = np.zeros(nv)
        row[i] = 1.0
        A_rows.append(row)
        b_rows.append(1.0 - h[i])
    A_ub = np.array(A_rows)
    b_ub = np.array(b_rows)
    A_eq = np.zeros((1, nv))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, method='highs')
    if res.status != 0:
        return None
    t_val = -res.x[-1]
    if t_val <= 1e-12:
        return None
    return res.x[:n]


def exact_line_search(h, d, dx, target_sum):
    n = len(h)
    alpha_max = 1.0
    for i in range(n):
        if d[i] > 1e-15:
            alpha_max = min(alpha_max, (1.0 - h[i]) / d[i])
        elif d[i] < -1e-15:
            alpha_max = min(alpha_max, h[i] / (-d[i]))
    if alpha_max <= 0:
        return h, c5_of_h(h, dx)
    c5_cur = compute_corrs(h).max() * dx
    best_alpha = 0.0
    best_h = h.copy()
    best_c5 = c5_cur
    alphas = np.concatenate([np.linspace(alpha_max, alpha_max * 0.5, 8),
                             np.geomspace(alpha_max * 0.5, max(1e-8, alpha_max * 1e-6), 20)])
    for alpha in alphas:
        if alpha <= 0 or alpha > alpha_max + 1e-15:
            continue
        h_trial = h + alpha * d
        if h_trial.min() < -1e-10 or h_trial.max() > 1.0 + 1e-10:
            continue
        h_trial = np.clip(h_trial, 0.0, 1.0)
        s = h_trial.sum()
        if abs(s - target_sum) > 1e-12:
            h_trial = project_feasible(h_trial, target_sum)
        c5_trial = compute_corrs(h_trial).max() * dx
        if c5_trial < best_c5 - 1e-16:
            best_c5 = c5_trial
            best_alpha = alpha
            best_h = h_trial.copy()
    return best_h, best_c5


def lp_optimize(h0, dx, target_sum, max_iters, n_active_list, t0, budget_s, margin=30):
    n = len(h0)
    best_h = h0.copy()
    best_c5 = c5_of_h(h0, dx)
    for na in n_active_list:
        h = best_h.copy()
        stuck = 0
        for it in range(max_iters):
            if time.time() - t0 > budget_s - margin:
                return best_h, best_c5
            dh = lp_direction(h, target_sum, na)
            if dh is None:
                stuck += 1
                if stuck > 3:
                    break
                continue
            h_new, c5_new = exact_line_search(h, dh, dx, target_sum)
            c5_cur = c5_of_h(h, dx)
            if c5_new < c5_cur - 1e-15:
                h = h_new
                stuck = 0
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
            else:
                stuck += 1
                if stuck > 5:
                    break
    return best_h, best_c5


def pair_transfer_optimize(h0, dx, target_sum, max_iters, t0, budget_s, margin=20):
    n = len(h0)
    rng = np.random.RandomState(12345)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    no_improve = 0
    for it in range(max_iters):
        if time.time() - t0 > budget_s - margin:
            break
        corrs = compute_corrs(h)
        c5_val = corrs.max() * dx
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > max(500, max_iters // 4):
            break
        G = compute_G(h)
        max_c = corrs.max()
        shifted = corrs - max_c
        beta = 30.0
        w = np.exp(beta * shifted)
        w /= w.sum()
        g = G.T @ w
        good_a = np.where(h > 1e-6)[0]
        good_b = np.where(h < 1.0 - 1e-6)[0]
        if len(good_a) == 0 or len(good_b) == 0:
            break
        top_k = min(8, len(good_a))
        bot_k = min(8, len(good_b))
        top_idx = np.argsort(g[good_a])[-top_k:]
        bot_idx = np.argsort(g[good_b])[:bot_k]
        best_improve = 0.0
        found = False
        for ia in top_idx:
            a = good_a[ia]
            for ib in bot_idx:
                b = good_b[ib]
                if a == b:
                    continue
                max_eps = min(h[a], 1.0 - h[b])
                if max_eps < 1e-8:
                    continue
                for frac in [0.3, 0.6, 1.0]:
                    eps = max_eps * frac
                    h_trial = h.copy()
                    h_trial[a] -= eps
                    h_trial[b] += eps
                    c5_trial = compute_corrs(h_trial).max() * dx
                    if c5_trial < c5_val - best_improve - 1e-16:
                        best_improve = c5_val - c5_trial
                        h[a] -= eps
                        h[b] += eps
                        found = True
                        break
                if found:
                    break
            if found:
                break
        if not found:
            ra = rng.randint(len(good_a))
            rb = rng.randint(len(good_b))
            a = good_a[ra]
            b = good_b[rb]
            if a == b:
                continue
            eps = min(h[a], 1.0 - h[b], 0.05)
            if eps < 1e-8:
                continue
            h[a] -= eps
            h[b] += eps
    return best_h, best_c5


def subgradient_optimize(h0, dx, target_sum, max_iters, t0, budget_s, margin=20):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    no_improve = 0
    alpha = 0.5
    for it in range(max_iters):
        if time.time() - t0 > budget_s - margin:
            break
        corrs = compute_corrs(h)
        c5_val = corrs.max() * dx
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > max(1000, max_iters // 3):
            break
        G = compute_G(h)
        max_c = corrs.max()
        shifted = corrs - max_c
        beta = 15.0
        w = np.exp(beta * shifted)
        w /= w.sum()
        g = G.T @ w
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        step = alpha / (g_norm + 1e-15)
        h_new = h - step * g
        h_new = project_feasible(h_new, target_sum)
        c5_new = c5_of_h(h_new, dx)
        if c5_new > c5_val + 1e-14:
            step *= 0.25
            h_new = h - step * g
            h_new = project_feasible(h_new, target_sum)
            c5_new = c5_of_h(h_new, dx)
            if c5_new > c5_val + 1e-14:
                alpha *= 0.5
                continue
        h = h_new
        alpha = max(alpha * 0.999, 0.01)
    return best_h, best_c5


def random_perturb_search(h0, dx, target_sum, n_trials, t0, budget_s, margin=15):
    n = len(h0)
    rng = np.random.RandomState(999)
    best_h = h0.copy()
    best_c5 = c5_of_h(h0, dx)
    for trial in range(n_trials):
        if time.time() - t0 > budget_s - margin:
            break
        h = best_h.copy()
        n_pert = rng.randint(2, max(3, n // 10))
        idx = rng.choice(n, size=2 * n_pert, replace=False)
        for j in range(n_pert):
            a, b = idx[2 * j], idx[2 * j + 1]
            eps = rng.uniform(0.01, min(h[a], 1.0 - h[b], 0.15))
            h[a] -= eps
            h[b] += eps
        h = project_feasible(h, target_sum)
        c5_trial = c5_of_h(h, dx)
        if c5_trial < best_c5:
            best_c5 = c5_trial
            best_h = h.copy()
    return best_h, best_c5


def interpolate_h(h_src, n_target):
    n_src = len(h_src)
    x_src = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_tgt = np.linspace(0.0, 2.0, n_target, endpoint=False)
    h_interp = np.interp(x_tgt, x_src, h_src)
    return h_interp
