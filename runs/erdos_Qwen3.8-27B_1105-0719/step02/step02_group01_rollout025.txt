import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import lil_matrix, csr_matrix
import time


def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def c5_of_h(h):
    return compute_corrs(h).max() * (2.0 / len(h))


def corr_grad_k(h, k, n):
    shift = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -shift)
    i_hi = min(n - 1, n - 1 - shift)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx] += (1.0 - h[idx + shift])
    grad[idx + shift] += (-h[idx])
    return grad


def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)


def build_and_solve_mccormick_lp(n, dx, target_sum, budget_remaining):
    """Build and solve the McCormick LP relaxation for the Erdos problem."""
    # Variable layout: [h_0..h_{n-1}, t, y_0, y_1, ..., y_{M-1}]
    # For each lag k, for each valid i, one y variable
    lags = []
    for k in range(2 * n - 1):
        shift = k - (n - 1)
        i_lo = max(0, -shift)
        i_hi = min(n - 1, n - 1 - shift)
        lags.append((k, shift, i_lo, i_hi, i_hi - i_lo + 1))
    
    total_y = sum(len_l for _, _, _, _, len_l in lags)
    n_vars = n + 1 + total_y
    t_idx = n
    
    # Build sparse A_ub matrix
    # Constraints:
    # 1) For each lag k: dx*sum(h[i]) - dx*sum(y_i) - t <= 0
    # 2) For each (i,k): y <= h[i]  =>  -h[i] + y <= 0
    # 3) For each (i,k): y <= h[j]  =>  -h[j] + y <= 0
    # 4) h[i] <= 1  =>  h[i] <= 1
    
    n_corr = 2 * n - 1
    n_mcc = 2 * total_y
    n_hbound = n
    n_total_constraints = n_corr + n_mcc + n_hbound
    
    A = lil_matrix((n_total_constraints, n_vars), dtype=np.float64)
    b = np.zeros(n_total_constraints)
    
    row = 0
    # Correlation constraints
    y_ptr = 0
    for k, shift, i_lo, i_hi, cnt in lags:
        for i in range(i_lo, i_hi + 1):
            A[row, i] += dx
            A[row, n + 1 + y_ptr] += -dx
            y_ptr += 1
        A[row, t_idx] = -1.0
        row += 1
    
    # McCormick upper bounds: y <= h[i] and y <= h[j]
    y_ptr = 0
    for k, shift, i_lo, i_hi, cnt in lags:
        for i in range(i_lo, i_hi + 1):
            j = i + shift
            yvar = n + 1 + y_ptr
            # y - h[i] <= 0
            A[row, yvar] = 1.0
            A[row, i] = -1.0
            row += 1
            # y - h[j] <= 0
            A[row, yvar] = 1.0
            A[row, j] = -1.0
            row += 1
            y_ptr += 1
    
    # h[i] <= 1
    for i in range(n):
        A[row, i] = 1.0
        b[row] = 1.0
        row += 1
    
    A = csr_matrix(A)
    
    # Variable bounds: h[i] >= 0, t >= 0, y >= 0
    bounds = [(0.0, 1.0)] * n + [(0.0, None)] + [(0.0, None)] * total_y
    
    # Equality: sum h = target_sum
    A_eq = lil_matrix((1, n_vars), dtype=np.float64)
    for i in range(n):
        A_eq[0, i] = 1.0
    A_eq = csr_matrix(A_eq)
    b_eq = np.array([target_sum])
    
    # Objective: minimize t
    c_obj = np.zeros(n_vars)
    c_obj[t_idx] = 1.0
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': max(10, budget_remaining * 0.4)})
        if res.status == 0:
            return res.x[:n].copy(), res.x[t_idx] / dx
        else:
            return None, None
    except Exception:
        return None, None


def gradient_descent_polish(h_init, n, max_iter, budget_check, tol=1e-14):
    """Polish using gradient descent on the soft-max of correlations."""
    dx = 2.0 / n
    target_sum = n / 2.0
    h = project_feasible(h_init.copy(), target_sum)
    best_h = h.copy()
    best_c5 = c5_of_h(h)
    
    for beta in [100, 300, 800, 2000, 5000]:
        if budget_check() < 15:
            break
        for it in range(max_iter // 5):
            if budget_check() < 10:
                break
            corrs = compute_corrs(h)
            max_corr = corrs.max()
            c5_val = max_corr * dx
            if c5_val < best_c5 - 1e-13:
                best_c5 = c5_val
                best_h = h.copy()
            
            shifted = corrs - max_corr
            w = np.exp(beta * shifted)
            wsum = w.sum()
            if wsum < 1e-30:
                w = np.zeros(len(corrs))
                w[np.argmax(corrs)] = 1.0
            else:
                w /= wsum
            
            grad = np.zeros(n)
            for k in range(len(corrs)):
                if w[k] > 1e-12:
                    grad += w[k] * corr_grad_k(h, k, n)
            grad *= dx
            grad -= grad.mean()
            
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break
            
            step = 0.01
            improved = False
            for _ in range(30):
                h_new = project_feasible(h - step * grad, target_sum)
                c5_new = c5_of_h(h_new)
                if c5_new < c5_val - 1e-15:
                    h = h_new
                    improved = True
                    break
                step *= 0.5
            if not improved:
                break
        h = best_h
    
    return best_h, best_c5

        w = np.zeros(len(corrs))
        w[active] = 1.0 / len(active)
        
        grad = np.zeros(n)
        for k in active:
            grad += w[k] * corr_grad_k(h, k, n)
        grad *= dx
        grad -= grad.mean()
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        h_new = project_feasible(h - step * grad, target_sum)
        c5_new = c5_of_h(h_new)
        if c5_new < c5_val - 1e-14:
            h = h_new
        else:
            step *= 0.5
            if step < 1e-10:
                break
    
    return best_h, best_c5


def pairwise_local_search(h_init, n, max_iter, budget_check):
    """Local search using pairwise mass transfers."""
    dx = 2.0 / n
    target_sum = n / 2.0
    h = project_feasible(h_init.copy(), target_sum)
    best_h = h.copy()
    best_c5 = c5_of_h(h)
    rng = np.random.RandomState(12345)
    stagnant = 0
    
    for it in range(max_iter):
        if budget_check() < 10:
            break
        corrs = compute_corrs(h)
        c5_val = corrs.max() * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            stagnant = 0
        else:
            stagnant += 1
            if stagnant > 2000:
                break
        
        # Try to find a good pair to exchange
        k_max = np.argmax(corrs)
        shift = k_max - (n - 1)
        i_lo = max(0, -shift)
        i_hi = min(n - 1, n - 1 - shift)
        
        improved = False
        # Try targeted moves for the active lag
        if i_lo <= i_hi:
            idx = np.arange(i_lo, i_hi + 1)
            # We want to reduce h[i]*(1-h[i+shift])
            # Move mass from h[i] to h[i+shift] (if h[i] > h[i+shift])
            # or from h[i+shift] to h[i]... actually we want to make them closer to 0 or 1
            for trial in range(30):
                ii = idx[rng.randint(len(idx))]
                jj = ii + shift
                diff = h[ii] - h[jj]
                if abs(diff) < 0.01:
                    continue
                # Push them apart: increase the larger, decrease the smaller
                eps = min(abs(diff) * 0.5, 0.1) * rng.uniform(0.3, 1.0)
                if diff > 0:
                    h_try = h.copy()
                    h_try[ii] = min(1.0, h_try[ii] + eps)
                    h_try[jj] = max(0.0, h_try[jj] - eps)
                else:
                    h_try = h.copy()
                    h_try[jj] = min(1.0, h_try[jj] + eps)
                    h_try[ii] = max(0.0, h_try[ii] - eps)
                h_try = project_feasible(h_try, target_sum)
                c5_try = c5_of_h(h_try)
                if c5_try < c5_val - 1e-13:
                    h = h_try
                    improved = True
                    break
        
        if not improved:
            # Random pair
            for trial in range(20):
                i1 = rng.randint(n)
                i2 = rng.randint(n)
                if i1 == i2:
                    continue
                eps = min(h[i1], 1.0 - h[i2]) * rng.uniform(0.01, 0.1)
                if eps < 1e-5:
                    continue
                h_try = h.copy()
                h_try[i1] -= eps
                h_try[i2] += eps
                h_try = np.clip(h_try, 0, 1)
                h_try = project_feasible(h_try, target_sum)
                c5_try = c5_of_h(h_try)
                if c5_try < c5_val - 1e-13:
                    h = h_try
                    improved = True
                    break
    
    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    
    def remaining():
        return budget_s - (time.time() - t0) - 8.0
    
    # Get parent
    h_parent = np.array(initial_h_values, dtype=float)
    n_parent = len(h_parent)
    best_h = h_parent.copy()
    best_c5 = c5_of_h(best_h)
    
    # Phase 1: Try LP relaxation on parent's n
    if remaining() > 60:
        n_try = n_parent
        dx_try = 2.0 / n_try
        ts_try = n_try / 2.0
        h_lp, c5_lp = build_and_solve_mccormick_lp(n_try, dx_try, ts_try, remaining())
        if h_lp is not None:
            h_lp = project_feasible(h_lp, ts_try)
            c5_actual = c5_of_h(h_lp)
            if c5_actual < best_c5:
                best_c5 = c5_actual
                best_h = h_lp.copy()
    
    # Phase 2: Gradient polish on best solution
    if remaining() > 40:
        h_pol, c5_pol = gradient_descent_polish(best_h, len(best_h), 5000, remaining)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol
    
    # Phase 3: Active set descent
    if remaining() > 30:
        h_as, c5_as = active_set_descent(best_h, len(best_h), 8000, remaining)
        if c5_as < best_c5:
            best_c5 = c5_as
            best_h = h_as
    
    # Phase 4: Pairwise local search
    if remaining() > 40:
        h_pw, c5_pw = pairwise_local_search(best_h, len(best_h), 20000, remaining)
        if c5_pw < best_c5:
            best_c5 = c5_pw
            best_h = h_pw
    
    # Phase 5: Try LP relaxation on different n values
    n_candidates = []
    for nn in [64, 80, 90, 100, 110, 120, 126, 128, 136, 144, 150, 160]:
        if nn != n_parent:
            n_candidates.append(nn)
    
    for n_try in n_candidates:
        if remaining() < 40:
            break
        dx_try = 2.0 / n_try
        ts_try = n_try / 2.0
        
        # Interpolate best solution to new n
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_try)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, ts_try)
        
        # Try LP
        if remaining() > 50:
            h_lp, c5_lp = build_and_solve_mccormick_lp(n_try, dx_try, ts_try, remaining())
            if h_lp is not None:
                h_lp = project_feasible(h_lp, ts_try)
                c5_actual = c5_of_h(h_lp)
                if c5_actual < best_c5:
                    best_c5 = c5_actual
                    best_h = h_lp.copy()
            else:
                h_candidate = h_interp
        else:
            h_candidate = h_interp
        
        # Polish
        if remaining() > 30:
            h_pol, c5_pol = gradient_descent_polish(h_candidate, n_try, 3000, remaining)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol
        if remaining() > 20:
            h_as, c5_as = active_set_descent(best_h, len(best_h), 5000, remaining)
            if c5_as < best_c5:
                best_c5 = c5_as
                best_h = h_as
    
    # Phase 6: Final intensive polish on best
    if remaining() > 50:
        n_b = len(best_h)
        h_pol, c5_pol = gradient_descent_polish(best_h, n_b, 8000, remaining)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol
    
    if remaining() > 30:
        h_as, c5_as = active_set_descent(best_h, len(best_h), 10000, remaining)
        if c5_as < best_c5:
            best_c5 = c5_as
            best_h = h_as
    
    if remaining() > 25:
        h_pw, c5_pw = pairwise_local_search(best_h, len(best_h), 15000, remaining)
        if c5_pw < best_c5:
            best_c5 = c5_pw
            best_h = h_pw
    
    if remaining() > 15:
        h_pol, c5_pol = gradient_descent_polish(best_h, len(best_h), 5000, remaining)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol
    
    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final / 2.0)
    best_c5 = c5_of_h(best_h)
    
    return best_h, best_c5, n_final
