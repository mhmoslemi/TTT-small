- Terms involving h[i] as the "second" factor: contribute -t * h[i-e(k)] for each k where i-e(k) is valid (i.e., (i-e(k)) + e(k) = i)

Wait, let me be more precise. C_k = sum over valid positions p of h[p] * (1 - h[p + e(k)]).

When h[i] changes by +t:
- In term p=i: h[i]*(1-h[i+e(k)]) changes by t*(1-h[i+e(k)]) if i+e(k) is valid
- In term p=i-e(k): h[i-e(k)]*(1-h[i]) changes by -t*h[i-e(k)] if i-e(k) is valid

When h[j] changes by -t:
- In term p=j: h[j]*(1-h[j+e(k)]) changes by -t*(1-h[j+e(k)]) if j+e(k) is valid
- In term p=j-e(k): h[j-e(k)]*(1-h[j]) changes by +t*h[j-e(k)] if j-e(k) is valid

So the total change in C_k is:
ΔC_k = t * [(1-h[i+e(k)]) - h[i-e(k)] - (1-h[j+e(k)]) + h[j-e(k)]] (with appropriate zeroing for invalid indices)

This is linear in t for each k. The max over k of (C_k + ΔC_k) is the max of affine functions of t, which is convex in t.

def smooth_max_obj(h, dx, beta):
    corrs = np.correlate(h, 1.0 - h, mode='full')
    m = corrs.max()
    return m * dx + dx / beta * np.log(np.sum(np.exp(beta * (corrs - m) / dx)))

def smooth_max_grad(h, dx, beta):
    n = len(h)
    corrs = np.correlate(h, 1.0 - h, mode='full')
    m = corrs.max()
    shifted = beta * (corrs - m) / dx
    w = np.exp(shifted)
    w_sum = w.sum()
    w = w / w_sum
    
    grad = np.zeros(n)
    for k in range(len(corrs)):
        if w[k] < 1e-14:
            continue
        e = k - (n - 1)
        i_lo = max(0, -e)
        i_hi = min(n - 1, n - 1 - e)
        if i_lo > i_hi:
            continue
        idx = np.arange(i_lo, i_hi + 1)
        grad[idx + e] += w[k] * (1.0 - h[idx])
        grad[idx] -= w[k] * h[idx + e]
    return grad * dx

        return smooth_max_grad(x, dx, beta)

                a2, b2 = lo_t, hi_t

                    fc = c5_of_h(hc, dx)
                    hd = h_cur.copy()
                    hd[i] += d; hd[j] -= d

                    if fc < fd:

                    else:

                h_new = h_cur.copy()
                h_new[i] += t_opt
                h_new[j] -= t_opt
                c5_new = c5_of_h(h_new, dx)
                if c5_new < best_c5 - 1e-13:
                    h_cur = h_new
                    best_c5 = c5_new
                    improved = True

                break
    
    return h_cur, best_c5

def interpolate_h(h_old, n_old, n_new):

        return h_old.copy()

    x_new = np.linspace(0.0, 2.0, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return h_new

def mixed_sa(h0, dx, rng, time_limit=300):
    """SA with mixed move types."""
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    
    T = 0.001
    no_improve = 0
    patience = 40000
    t0 = time.time()
    count = 0
    
    while time.time() - t0 < time_limit:
        count += 1
        if count % 5000 == 0:
            T *= 0.997
            if no_improve > patience:
                break
        
        # Choose move type
        r = rng.random()
        
        if r < 0.6:
            # Single swap
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            old_i, old_j = h[i], h[j]
            h[i], h[j] = old_j, old_i
        elif r < 0.8:
            # Segment reversal
            a = rng.randint(n)
            b = rng.randint(n)
            if a > b:
                a, b = b, a
            if b - a < 2:
                continue
            seg = h[a:b+1].copy()
            h[a:b+1] = h[a:b+1][::-1]
        else:
            # Subset shuffle: pick 3-5 random positions and shuffle their values
            k = rng.randint(3, 6)
            idx = rng.choice(n, size=k, replace=False)
            vals = h[idx].copy()
            perm = rng.permutation(k)
            h[idx] = vals[perm]
        
        new_c5 = c5_of_h(h, dx)
        
        if new_c5 <= best_c5 + 1e-15:
            best_c5 = new_c5
            best_h = h.copy()
            no_improve = 0
        else:
            delta = new_c5 - best_c5
            if T > 1e-14 and rng.random() < np.exp(-delta / T):
                no_improve = 0
            else:
                # Revert
                if r < 0.6:
                    h[i], h[j] = old_i, old_j
                elif r < 0.8:
                    h[a:b+1] = seg
                else:
                    h[idx] = vals
                no_improve += 1
    
    return best_h, best_c5
