def project_h(x, target_sum):
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) / 2, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def c5_of_h(h, dx):
    corrs = np.correlate(h, 1.0 - h, mode='full')
    return corrs.max() * dx

def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def smooth_max_grad(h, dx, beta):
    n = len(h)
    corrs = compute_corrs(h)
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        w /= w_sum
    grad = np.zeros(n)
    center = n - 1
    for m in range(2 * n - 1):
        wm = w[m]
        if wm < 1e-13:
            continue
        lag = m - center
        if lag == 0:
            grad += wm * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:] += wm * (1.0 - h[:-lag])
            grad[:-lag] += wm * (-h[lag:])
        else:
            d = -lag
            grad[:n - d] += wm * (1.0 - h[d:])
            grad[d:] += wm * (-h[:n - d])
    return grad * dx

def grad_single_lag(h, lag, n):
    """Gradient of correlation at a specific lag."""
    grad = np.zeros(n)
    if lag == 0:
        grad = 1.0 - 2.0 * h
    elif lag > 0:
        grad[lag:] = 1.0 - h[:-lag]
        grad[:-lag] += -h[lag:]
    else:
        d = -lag
        grad[:n - d] = 1.0 - h[d:]
        grad[d:] += -h[:n - d]
    return grad

def lp_active_set(h, dx, target_sum, n_active, alpha):
    """LP-based active set refinement.
    Linearize top correlations, solve LP, take step of size alpha."""
    n = len(h)
    corrs = compute_corrs(h)
    c5_val = corrs.max() * dx
    
    # Select active lags
    indices = np.argsort(corrs)[-n_active:]
    
    # Build LP: min t + tiny * ||h - h0||^2 (approximated by just min t)
    # Variables: h'[0..n-1], t
    # Constraints: grad_k[h'] <= rhs_k for k in active
    #              sum h' = target_sum
    #              h' in [0, 1]
    
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0  # minimize t
    
    A_rows = []
    b_rows = []
    for m in indices:
        lag = m - (n - 1)
        g = grad_single_lag(h, lag, n)
        rhs = -corrs[m] + np.dot(g, h)
        # Constraint: g · h' <= t + rhs  =>  g · h' - t <= rhs
        row = np.zeros(n + 1)
        row[:n] = g
        row[n] = -1.0
        A_rows.append(row)
        b_rows.append(rhs)
    
    A_ub = np.array(A_rows)
    b_ub = np.array(b_rows)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target_sum])
    
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status != 0:
            return h, c5_val
        h_lp = res.x[:n]
        # Take a step: h_new = (1-alpha)*h + alpha*h_lp
        h_new = (1.0 - alpha) * h + alpha * h_lp
        h_new = project_h(h_new, target_sum)
        c5_new = c5_of_h(h_new, dx)
        if c5_new < c5_val:
            return h_new, c5_new
        else:
            return h, c5_val
    except Exception:
        return h, c5_val
