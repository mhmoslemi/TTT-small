import cvxpy as cp

def find_lpdirection(h, active_lags, n, epsilon=0.01):
    d = cp.Variable(n)
    constraints = [
        cp.sum(d) == 0,
        d >= -h + 1e-10,
        d <= 1 - h - 1e-10
    ]
    for k in active_lags:
        offset = k - (n-1)

    if abs(s) > 1e-14:
        d -= s / n
        d = np.clip(d, lo_b, hi_b)
    return d


def solve_lp_direction(h, active_lags, n, epsilon):
    d = cp.Variable(n)
    lo_b = -h + 1e-12
    hi_b = 1.0 - h - 1e-12
    constraints = [
        cp.sum(d) == 0,
        d >= lo_b,
        d <= hi_b
    ]
    for k in active_lags:
        g = grad_corr_k_vec(h, k, n)
        constraints.append(g @ d <= -epsilon)
    prob = cp.Problem(cp.Minimize(cp.sum_squares(d)), constraints)
    try:
        prob.solve(solver=cp.CLARABEL, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
        except Exception:
            return None
    if d.value is None:
        return None
    if not np.all(np.isfinite(d.value)):
        return None
    return d.value


def lp_based_optimize(h0, dx, target_sum, t0, time_limit, max_iters=300):
    n = len(h0)
    h = h0.copy()
    c5_cur = c5_of_h(h, dx)
    best_h = h.copy()
    best_c5 = c5_cur
    no_improve = 0

    for it in range(max_iters):
        if time.time() - t0 > time_limit:
            break
        corrs = compute_corrs(h)
        raw_max = corrs.max()
        sorted_idx = np.argsort(-corrs)

        found_dir = None
        eps_seq = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
        for na in [8, 6, 5, 4, 3, 2, 1]:
            active = list(sorted_idx[:na])
            for eps in eps_seq:
                d = solve_lp_direction(h, active, n, eps)
                if d is not None and np.linalg.norm(d) > 1e-10:
                    found_dir = d
                    break
            if found_dir is not None:
                break

        if found_dir is None:
            no_improve += 1
            if no_improve > 10:
                break
            # Perturb slightly to escape
            rng_local = np.random.RandomState(it * 7 + 3)
            h = project_feasible(h + 0.01 * (rng_local.random(n) - 0.5), target_sum)
            continue

        best_step_h = h
        best_step_c5 = c5_cur
        for alpha in [0.01, 0.02, 0.03, 0.05, 0.08, 0.12, 0.2, 0.35, 0.5]:
            h_try = project_feasible(h + alpha * found_dir, target_sum)
            c5_try = c5_of_h(h_try, dx)
            if c5_try < best_step_c5 - 1e-15:
                best_step_c5 = c5_try
                best_step_h = h_try

        if best_step_c5 < c5_cur - 1e-14:
            h = best_step_h
            c5_cur = best_step_c5
            if c5_cur < best_c5:
                best_c5 = c5_cur
                best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 15:
                break

    return best_h, best_c5


def smooth_max_obj_and_grad(h, dx, beta):
    n = len(h)
    corrs = compute_corrs(h)
    m = corrs.max()
    shifted = corrs - m
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    w = w / w_sum
    obj = (m + np.log(w_sum) / beta) * dx
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        grad += w[k] * grad_corr_k_vec(h, k, n)
    grad *= dx
    return obj, grad


def projected_gradient_step(h, grad, target_sum, step_size):
    d = -grad
    lo_b = -h + 1e-14
    hi_b = 1.0 - h - 1e-14
    d = np.clip(d, lo_b, hi_b)
    s = d.sum()
    d -= s / len(d)
    d = np.clip(d, lo_b, hi_b)
    h_new = project_feasible(h + step_size * d, target_sum)
    return h_new


def smooth_max_optimize(h0, dx, target_sum, t0, time_limit, beta=100, max_iters=500):
    n = len(h0)
    h = h0.copy()
    c5_cur = c5_of_h(h, dx)
    best_h = h.copy()
    best_c5 = c5_cur
    lr = 0.1
    no_improve = 0

    for it in range(max_iters):
        if time.time() - t0 > time_limit:
            break
        obj, grad = smooth_max_obj_and_grad(h, dx, beta)
        h_new = projected_gradient_step(h, grad, target_sum, lr)
        c5_new = c5_of_h(h_new, dx)
        if c5_new < c5_cur - 1e-15:
            h = h_new
            c5_cur = c5_new
            if c5_cur < best_c5:
                best_c5 = c5_cur
                best_h = h.copy()
            no_improve = 0
        else:
            lr *= 0.7
            no_improve += 1
            if no_improve > 30:
                break

    return best_h, best_c5


def compute_delta_binary(h, i, j, n):
    ms = np.arange(-(n - 1), n)
    delta = np.zeros(2 * n - 1)
    v1 = (0 <= i - ms) & (i - ms <= n - 1)
    delta[v1] += -(1.0 - h[i - ms[v1]])
    v2 = (0 <= i + ms) & (i + ms <= n - 1)
    delta[v2] += h[i + ms[v2]]
    v3 = (0 <= j - ms) & (j - ms <= n - 1)
    delta[v3] += (1.0 - h[j - ms[v3]])
    v4 = (0 <= j + ms) & (j + ms <= n - 1)
    delta[v4] += -h[j + ms[v4]]
    return delta


def binary_sa_optimize(h0, dx, t0, time_limit, max_iters=500000):
    n = len(h0)
    target_sum = int(round(n / 2))
    h = np.zeros(n)
    sorted_idx = np.argsort(-h0)
    h[sorted_idx[:target_sum]] = 1.0
    h = project_feasible(h, float(target_sum))

    rng = np.random.RandomState(12345)
    ones = list(np.where(h >= 0.5)[0])
    zeros = list(np.where(h < 0.5)[0])
    # Make it strictly binary
    h = np.where(h >= 0.5, 1.0, 0.0)
    ones = list(np.where(h == 1.0)[0])
    zeros = list(np.where(h == 0.0)[0])

    corrs = compute_corrs(h)
    best_c5 = corrs.max() * dx
    best_h = h.copy()
    cur_c5 = best_c5

    temp = 0.04
    cooling = 0.9998
    no_improve = 0
    patience = 10000

    for it in range(max_iters):
        if time.time() - t0 > time_limit:
            break
        if it % 3000 == 0:
            temp *= cooling

        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        delta = compute_delta_binary(h, i, j, n)
        new_corrs = corrs + delta
        new_c5 = new_corrs.max() * dx

        if new_c5 <= cur_c5 + 1e-15:
            h[i], h[j] = 0.0, 1.0
            corrs = new_corrs
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
            ones.remove(i)
            zeros.remove(j)
            ones.append(j)
            zeros.append(i)
            no_improve = 0
        else:
            if temp > 1e-10 and rng.random() < np.exp(-(new_c5 - cur_c5) / temp):
                h[i], h[j] = 0.0, 1.0
                corrs = new_corrs
                cur_c5 = new_c5
                ones.remove(i)
                zeros.remove(j)
                ones.append(j)
                zeros.append(i)
                no_improve = 0
            else:
                no_improve += 1
                if no_improve > patience:
                    break

    return best_h, best_c5


def binary_greedy_optimize(h0, dx, t0, time_limit, max_sweeps=100):
    n = len(h0)
    h = np.round(h0).astype(float)
    target_sum = int(round(n / 2))
    # Ensure exact count
    if h.sum() > target_sum:
        excess = int(h.sum()) - target_sum
        idx = np.where(h == 1.0)[0]
        rng_g = np.random.RandomState(99)
        for _ in range(excess):
            h[idx[rng_g.randint(len(idx))]] = 0.0
    elif h.sum() < target_sum:
        deficit = target_sum - int(h.sum())
        idx = np.where(h == 0.0)[0]
        rng_g = np.random.RandomState(99)
        for _ in range(deficit):
            h[idx[rng_g.randint(len(idx))]] = 1.0

    ones = list(np.where(h == 1.0)[0])
    zeros = list(np.where(h == 0.0)[0])
    corrs = compute_corrs(h)
    best_c5 = corrs.max() * dx
    best_h = h.copy()

    for sweep in range(max_sweeps):
        if time.time() - t0 > time_limit:
            break
        best_delta_max = corrs.max()
        best_swap = None
        ones_arr = np.array(ones)
        zeros_arr = np.array(zeros)
        for ii in range(len(ones_arr)):
            i = ones_arr[ii]
            for jj in range(len(zeros_arr)):
                j = zeros_arr[jj]
                delta = compute_delta_binary(h, i, j, n)
                new_max = (corrs + delta).max()
                if new_max < best_delta_max - 1e-14:
                    best_delta_max = new_max
                    best_swap = (i, j)
        if best_swap is None:
            break
        i, j = best_swap
        h[i], h[j] = 0.0, 1.0
        corrs = compute_corrs(h)
        c5_now = corrs.max() * dx
        if c5_now < best_c5:
            best_c5 = c5_now
            best_h = h.copy()
        ones = list(np.where(h == 1.0)[0])
        zeros = list(np.where(h == 0.0)[0])

    return best_h, best_c5


def subgradient_optimize(h0, dx, target_sum, t0, time_limit, max_iters=2000):
    n = len(h0)
    h = h0.copy()
    c5_cur = c5_of_h(h, dx)
    best_h = h.copy()
    best_c5 = c5_cur
    lr = 0.05
    no_improve = 0

    for it in range(max_iters):
        if time.time() - t0 > time_limit:
            break
        corrs = compute_corrs(h)
        k_star = int(np.argmax(corrs))
        g = grad_corr_k_vec(h, k_star, n)
        d = project_direction(-g * lr, h, target_sum)
        h_new = project_feasible(h + d, target_sum)
        c5_new = c5_of_h(h_new, dx)
        if c5_new < c5_cur - 1e-15:
            h = h_new
            c5_cur = c5_new
            if c5_cur < best_c5:
                best_c5 = c5_cur
                best_h = h.copy()
            no_improve = 0
        else:
            lr *= 0.8
            no_improve += 1
            if no_improve > 50:
                break

    return best_h, best_c5


def make_wave_start(n, target_sum, freq, phase, rng):
    x = np.linspace(0, 2.0, n)
    raw = 0.5 + 0.3 * np.sin(2 * np.pi * freq * x + phase)
    raw = raw + 0.05 * (rng.random(n) - 0.5)
    return project_feasible(raw, target_sum)


def remaining_time(t0, budget_s):
    return budget_s - (time.time() - t0)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    best_h = np.array(initial_h_values, dtype=float)
    n = len(best_h)
    dx = 2.0 / n
    target_sum = n / 2.0
    best_h = project_feasible(best_h, target_sum)
    best_c5 = c5_of_h(best_h, dx)

    # Phase 1: LP-based optimization (primary, most powerful)
    time_limit1 = budget_s * 0.35
    if remaining_time(t0, budget_s) > time_limit1 + 20:
        h1, c5_1 = lp_based_optimize(best_h, dx, target_sum, t0, time_limit1)
        if c5_1 < best_c5:
            best_c5 = c5_1
            best_h = h1

    # Phase 2: Smooth max gradient (refine)
    time_limit2 = budget_s * 0.15
    if remaining_time(t0, budget_s) > time_limit2 + 10:
        h2, c5_2 = smooth_max_optimize(best_h, dx, target_sum, t0, time_limit2, beta=200)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2
        if remaining_time(t0, budget_s) > 60:
            h2b, c5_2b = smooth_max_optimize(best_h, dx, target_sum, t0, 60, beta=500)
            if c5_2b < best_c5:
                best_c5 = c5_2b
                best_h = h2b

    # Phase 3: Subgradient (different angle)
    time_limit3 = budget_s * 0.10
    if remaining_time(t0, budget_s) > time_limit3 + 10:
        h3, c5_3 = subgradient_optimize(best_h, dx, target_sum, t0, time_limit3)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3

    # Phase 4: LP again from refined point
    time_limit4 = budget_s * 0.15
    if remaining_time(t0, budget_s) > time_limit4 + 10:
        h4, c5_4 = lp_based_optimize(best_h, dx, target_sum, t0, time_limit4)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4

    # Phase 5: Binary SA
    time_limit5 = budget_s * 0.12
    if remaining_time(t0, budget_s) > time_limit5 + 10:
        h5, c5_5 = binary_sa_optimize(best_h, dx, t0, time_limit5)
        if c5_5 < best_c5:
            best_c5 = c5_5
            best_h = h5

    # Phase 6: Binary greedy
    time_limit6 = budget_s * 0.08
    if remaining_time(t0, budget_s) > time_limit6 + 5:
        h6, c5_6 = binary_greedy_optimize(best_h, dx, t0, time_limit6)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6

    # Phase 7: Final LP polish
    time_limit7 = min(60.0, remaining_time(t0, budget_s) - 5)
    if time_limit7 > 15:
        h7, c5_7 = lp_based_optimize(best_h, dx, target_sum, t0, time_limit7)
        if c5_7 < best_c5:
            best_c5 = c5_7
            best_h = h7

    # Phase 8: Try wave starts if time permits
    if remaining_time(t0, budget_s) > 80:
        for freq in [3, 5, 7]:
            if remaining_time(t0, budget_s) < 40:
                break
            h_wave = make_wave_start(n, target_sum, freq, rng.random() * 6.28, rng)
            h_w, c5_w = lp_based_optimize(h_wave, dx, target_sum, t0, 25)
            if c5_w < best_c5:
                best_c5 = c5_w
                best_h = h_w

    # Final validation
    n_b = len(best_h)
    dx_b = 2.0 / n_b
    ts_b = n_b / 2.0
    best_h = project_feasible(best_h, ts_b)
    best_c5 = c5_of_h(best_h, dx_b)

    return best_h, best_c5, n_b
