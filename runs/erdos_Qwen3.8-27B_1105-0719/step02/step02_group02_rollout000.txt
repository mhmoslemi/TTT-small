    strategies.append(h)

    return strategies


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def remaining():
        return budget_s - (time.time() - t0)

    best_h = np.array(initial_h_values, dtype=float)
    n = len(best_h)
    dx = 2.0 / n
    target_sum = n / 2.0
    best_h = project_feasible(best_h, target_sum)
    best_c5 = c5_of_h(best_h, dx)

    # Phase 1: Nesterov subgradient from initial
    if remaining() > 100:
        h_opt, c5_val = nesterov_subgrad(best_h, dx, target_sum, 30000)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_opt

    # Phase 2: SLSQP with increasing beta
    beta_schedule = [50, 100, 200, 500, 1000, 2000, 4000, 8000]
    for beta in beta_schedule:
        if remaining() < 20:
            break
        maxit = 600 if beta < 1000 else 800
        h_opt, c5_val = slsqp_opt(best_h, dx, target_sum, beta, maxit)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_opt

    # Phase 3: Equalize top correlations
    if remaining() > 60:
        for tn in [2, 3, 4, 5]:
            if remaining() < 20:
                break
            h_eq, c5_eq = equalize_top(best_h, dx, target_sum, tn, 200)
            if c5_eq < best_c5:
                best_c5 = c5_eq
                best_h = h_eq

    # Phase 4: More Nesterov after SLSQP
    if remaining() > 80:
        h_opt, c5_val = nesterov_subgrad(best_h, dx, target_sum, 20000)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_opt

    # Phase 5: Final SLSQP polish
    for beta in [10000, 15000, 20000]:
        if remaining() < 15:
            break
        h_opt, c5_val = slsqp_opt(best_h, dx, target_sum, beta, 1000)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_opt

    # Phase 6: Try different n values
    n_candidates = []
    for delta in range(-20, 40, 2):
        nt = n + delta
        if nt >= 40 and nt <= 300:
            n_candidates.append(nt)
    n_candidates = sorted(set(n_candidates))

    for n_try in n_candidates:
        if remaining() < 50:
            break
        dx_try = 2.0 / n_try
        ts_try = n_try / 2.0
        x_old = np.arange(len(best_h)) * dx
        x_new = np.arange(n_try) * dx_try
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, ts_try)

        h_opt, c5_val = slsqp_opt(h_interp, dx_try, ts_try, 200, 400)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_opt
            # Polish at this n
            if remaining() > 60:
                dx_p = 2.0 / len(best_h)
                ts_p = len(best_h) / 2.0
                h_s, c5_s = nesterov_subgrad(best_h, dx_p, ts_p, 10000)
                if c5_s < best_c5:
                    best_c5 = c5_s
                    best_h = h_s
            if remaining() > 40:
                dx_p = 2.0 / len(best_h)
                ts_p = len(best_h) / 2.0
                h_opt2, c5_val2 = slsqp_opt(best_h, dx_p, ts_p, 3000, 700)
                if c5_val2 < best_c5:
                    best_c5 = c5_val2
                    best_h = h_opt2

    # Phase 7: Diverse starts at current best n
    n_b = len(best_h)
    dx_b = 2.0 / n_b
    ts_b = n_b / 2.0
    for trial in range(15):
        if remaining() < 20:
            break
        starts = make_structured_start(n_b, ts_b, rng)
        for h_start in starts:
            if remaining() < 15:
                break
            h_opt, c5_val = slsqp_opt(h_start, dx_b, ts_b, 300, 500)
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h_opt
                if remaining() > 30:
                    h_opt2, c5_val2 = slsqp_opt(h_opt, dx_b, ts_b, 2000, 700)
                    if c5_val2 < best_c5:
                        best_c5 = c5_val2
                        best_h = h_opt2

    # Phase 8: Perturbation + polish cycle
    for trial in range(30):
        if remaining() < 15:
            break
        mag = 0.03 + 0.05 * (trial / 30.0)
        h_pert = best_h + rng.normal(0, mag, len(best_h))
        h_pert = project_feasible(h_pert, target_sum	if False else ts_b)
        h_opt, c5_val = slsqp_opt(h_pert, dx_b, ts_b, 500, 400)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_opt
            if remaining() > 25:
                h_opt2, c5_val2 = slsqp_opt(h_opt, dx_b, ts_b, 3000, 600)
                if c5_val2 < best_c5:
                    best_c5 = c5_val2
                    best_h = h_opt2

    # Final squeeze
    n_b = len(best_h)
    dx_b = 2.0 / n_b
    ts_b = n_b / 2.0
    if remaining() > 20:
        h_final, c5_final = slsqp_opt(best_h, dx_b, ts_b, 15000, 1000)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final

    if remaining() > 15:
        h_final, c5_final = nesterov_subgrad(best_h, dx_b, ts_b, 15000)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final

    if remaining() > 12:
        h_final, c5_final = slsqp_opt(best_h, dx_b, ts_b, 20000, 800)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final

    return best_h, best_c5, len(best_h)
