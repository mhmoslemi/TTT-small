import numpy as np
from scipy.optimize import minimize
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full"))) * dx

def profile_h(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx])
            h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    return repair_feasible(np.clip(h_new, 0.0, 1.0))

def build_gradient_rows(h, rows, n, dx):
    """Build gradient rows for specified lag indices."""
    G = np.zeros((len(rows), n))
    for idx, m in enumerate(rows):
        d = m - (n - 1)
        if d == 0:
            G[idx, :] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            for i in range(n - d):
                G[idx, i + d] += dx * (1.0 - h[i])
                G[idx, i] -= dx * h[i + d]
        else:
            dp = -d
            for j in range(n - dp):
                G[idx, j + dp] += dx * (1.0 - h[j])
                G[idx, j] -= dx * h[j + dp]
    return G

def softmax_gradient(h, beta, lam_sum=50.0):
    """Compute gradient of soft-max of profile + sum penalty."""
    n = len(h)
    dx = 2.0 / n
    P = np.correlate(h, 1 - h, mode="full") * dx
    Pmax = P.max()
    exp_arg = beta * (P - Pmax)
    weights = np.exp(exp_arg)
    S = weights.sum()
    if S < 1e-300:
        weights = np.zeros(len(P))
        weights[np.argmax(P)] = 1.0
    else:
        weights /= S
    # Only need gradient for rows with significant weight
    active = np.where(weights > 0.01 / len(P))[0]
    if len(active) == 0:
        active = np.argsort(weights)[-5:]
    G = build_gradient_rows(h, active, n, dx)
    w_active = weights[active]
    w_active = w_active / w_active.sum()
    grad = G.T @ w_active
    # Sum constraint penalty
    sum_err = h.sum() - n / 2.0
    grad += 2.0 * lam_sum * sum_err / n
    return grad

def projected_descent(h, max_iters=100, momentum=0.3, t_start=None, budget_s=1100):
    """Momentum projected gradient descent on soft-max with decreasing beta."""
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    vel = np.zeros(n)
    
    betas = [50, 100, 200, 400, 800]
    iters_per_beta = max(10, max_iters // len(betas))
    
    for beta in betas:
        alpha = 0.02
        for it in range(iters_per_beta):
            if t_start is not None and time.perf_counter() - t_start > budget_s - 20:
                return h, best_c
            grad = softmax_gradient(h, beta, lam_sum=50.0)
            grad -= grad.mean()
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break
            grad /= gnorm
            vel = momentum * vel + grad
            vnorm = np.linalg.norm(vel)
            if vnorm > 0:
                vel /= vnorm
            improved = False
            for a in [alpha, alpha * 0.5, alpha * 0.25, alpha * 0.1, alpha * 0.05]:
                h_try = repair_feasible(h - a * vel)
                s = exact_score(h_try)
                if s < best_c - 1e-14:
                    h = h_try
                    best_c = s
                    improved = True
                    break
                elif s < exact_score(h):
                    h = h_try
                    c_now = exact_score(h)
                    if c_now < best_c:
                        best_c = c_now
            if improved:
                alpha = min(alpha * 1.05, 0.08)
            else:
                alpha *= 0.6
                if alpha < 1e-7:
                    break
    return h, best_c

def slsqp_active(h0, maxiter=300, active_k=10, tol=1e-12):
    """SLSQP with top-K active constraints."""
    n = len(h0)
    dx = 2.0 / n
    M = 2 * n - 1
    P = profile_h(h0)
    c0 = P.max()
    
    # Select active constraints
    thresh = c0 - max(0.003, 0.05 * (c0 - np.median(P)))
    active = np.where(P >= thresh)[0]
    if len(active) == 0:
        active = np.array([np.argmax(P)])
    if len(active) > active_k:
        top_idx = np.argsort(P[active])[-active_k:]
        active = active[top_idx]
    
    def obj(x):
        return x[-1]
    
    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    
    def eq_fun(x):
        return x[:-1].sum() - n / 2.0
    
    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g
    
    def ineq_fun(x):
        h = x[:-1]
        t = x[-1]
        prof = profile_h(h)
        return t - prof[active]
    
    def ineq_jac(x):
        h = x[:-1]
        G = build_gradient_rows(h, active, n, dx)
        J = np.zeros((len(active), n + 1))
        J[:, :-1] = -G
        J[:, -1] = 1.0
        return J
    
    x0 = np.concatenate([h0, [c0 + 1e-4]])
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 1.0)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': tol})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = repair_feasible(res.x[:-1])
            c_new = exact_score(h_new)
            if c_new <= c0 + 1e-10:
                return h_new
    except Exception:
        pass
    return None

def coordinate_descent(h, max_sweeps=20, t_start=None, budget_s=1100):
    """Coordinate-wise line search to polish the solution."""
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    
    for sweep in range(max_sweeps):
        if t_start is not None and time.perf_counter() - t_start > budget_s - 10:
            break
        improved = False
        for i in range(n):
            if t_start is not None and time.perf_counter() - t_start > budget_s - 10:
                break
            h_save = h[i]
            # Try moving h[i] up or down
            for direction in [1, -1]:
                for frac in [0.01, 0.005, 0.002, 0.001]:
                    delta = direction * frac
                    h_new_val = h_save + delta
                    if h_new_val < -1e-12 or h_new_val > 1.0 + 1e-12:
                        continue
                    h[i] = repair_feasible(h)[i]
                    s = exact_score(h)
                    if s < best_c - 1e-15:
                        best_c = s
                        improved = True
                        break
                    else:
                        h[i] = h_save
                    if s < best_c:
                        h[i] = h_save
                        break
            # After changing h[i], repair
            h = repair_feasible(h)
    return h, best_c

def make_binary_start(n, rng):
    """Create a binary starting point with good structure."""
    h = np.zeros(n)
    # Use a low-discrepancy-like pattern
    # Place ones at positions based on golden ratio
    golden = 0.6180339887
    for i in range(n // 2):
        pos = int((i * golden) % 1.0 * n)
        h[pos] = 1.0
    return repair_feasible(h)
