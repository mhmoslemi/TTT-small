- phi = M + tau * log(sum(exp((C[j]-M)/tau)))
- As tau → 0, phi → max(C)
- As tau → ∞, phi → mean(C) (approximately)
- Always smooth, always provides gradient
- The weights p[j] = exp((C[j]-M)/tau) / Z naturally emphasize near-max constraints

I think LSE should work better because:

2. Always smooth
3. The weighting is automatic and adaptive
4. Can be used with any tau > 0, giving a continuous family of approximations

The downside is that for very small tau, the gradient is dominated by the single max constraint, losing information from near-max constraints. So the annealing schedule is important.

    scores = (C - M) / tau
    scores = np.clip(scores, -200.0, 50.0)
    exp_s = np.exp(scores)
    Z = np.sum(exp_s)
    p = exp_s / Z
    phi = M + tau * np.log(Z)

    target = n / 2.0
    sdev = h.sum() - target
    phi_full = phi + 0.5 * mu_sum * sdev * sdev

    gh = np.zeros(n)
    active = np.where(p > p_thresh)[0]
    for j in active:
        w = p[j] * dx
        d = j - (n - 1)
        i_lo = max(0, -d)
        i_hi = min(n - 1, n - 1 - d)
        if i_lo > i_hi:
            continue
        seg_a = np.arange(i_lo + d, i_hi + d + 1)
        seg_b = np.arange(i_lo, i_hi + 1)
        gh[seg_a] += w * (1.0 - h[seg_b])
        gh[seg_b] += -w * h[seg_a]
    gh += mu_sum * sdev

    return phi_full, gh


def _lse_anneal_sym(q, n, reserve, remaining, budget_s, t0, max_stages=18):
    m = n // 2
    q = q.copy()
    best_q = q.copy()
    best_c5 = eval_c5(build_h(q, n))

    tau_start = 0.12
    tau_end = 0.0003
    ratio = (tau_end / tau_start) ** (1.0 / max_stages)
    tau = tau_start

    maxiter_list = [200, 200, 200, 150, 150, 150, 100, 100, 100, 80, 80, 80, 60, 60, 60, 50, 50, 50]

    for stage in range(max_stages):
        if remaining() < reserve + 10:
            break
        mi = maxiter_list[min(stage, len(maxiter_list) - 1)]
        gtol = 1e-10 if tau < 0.005 else 1e-8

        def obj(z):
            return _grad_lse_sym(z, n, tau)

        try:
            res = minimize(obj, q, jac=True, method='L-BFGS-B',
                           bounds=[(0.0, 1.0)] * m,
                           options={'maxiter': mi, 'gtol': gtol, 'maxls': 40})
            q_new = np.clip(res.x, 0.0, 1.0)
        except Exception:
            q_new = q.copy()

        h_new = build_h(q_new, n)
        c5_new = eval_c5(h_new)

        if c5_new < best_c5 - 1e-12:
            best_q = q_new.copy()
            best_c5 = c5_new
        q = q_new.copy()
        tau *= ratio

    # Extra fine-tuning with very small tau
    if remaining() > reserve + 10:
        for tau_f in [0.0002, 0.0001, 0.00005]:
            if remaining() < reserve + 5:
                break
            def obj_f(z):
                return _grad_lse_sym(z, n, tau_f)
            try:
                res = minimize(obj_f, q, jac=True, method='L-BFGS-B',
                               bounds=[(0.0, 1.0)] * m,
                               options={'maxiter': 80, 'gtol': 1e-11, 'maxls': 30})
                q_new = np.clip(res.x, 0.0, 1.0)
            except Exception:
                q_new = q.copy()
            h_new = build_h(q_new, n)
            c5_new = eval_c5(h_new)
            if c5_new < best_c5 - 1e-12:
                best_q = q_new.copy()
                best_c5 = c5_new
            q = q_new.copy()

    return best_q, best_c5


def _lse_anneal_full(h, n, reserve, remaining, budget_s, t0):
    h = h.copy()
    best_h = h.copy()
    best_c5 = eval_c5(h)
    mu_sum = 500.0

    for tau in [0.06, 0.03, 0.015, 0.008, 0.004, 0.002, 0.001, 0.0005]:
        if remaining() < reserve + 15:
            break

        def obj(z):
            return _grad_lse_full(z, n, tau, mu_sum)

        try:
            res = minimize(obj, h, jac=True, method='L-BFGS-B',
                           bounds=[(0.0, 1.0)] * n,
                           options={'maxiter': 120, 'gtol': 1e-9, 'maxls': 30})
            h_new = np.clip(res.x, 0.0, 1.0)
        except Exception:
            h_new = h.copy()
        h_new = _fix_sum(h_new, n / 2.0)
        c5_new = eval_c5(h_new)
        if c5_new < best_c5 - 1e-12:
            best_h = h_new.copy()
            best_c5 = c5_new
        h = h_new.copy()

    # Increase sum penalty for final enforcement
    if remaining() > reserve + 10:
        mu_sum2 = 5000.0
        def obj2(z):
            return _grad_lse_full(z, n, 0.001, mu_sum2)
        try:
            res = minimize(obj2, h, jac=True, method='L-BFGS-B',
                           bounds=[(0.0, 1.0)] * n,
                           options={'maxiter': 100, 'gtol': 1e-10, 'maxls': 30})
            h_new = np.clip(res.x, 0.0, 1.0)
        except Exception:
            h_new = h.copy()
        h_new = _fix_sum(h_new, n / 2.0)
        c5_new = eval_c5(h_new)
        if c5_new < best_c5 - 1e-12:
            best_h = h_new.copy()
            best_c5 = c5_new

    return best_h, best_c5


def _interp_h(h, n_new):
    n_old = len(h)
    if n_new == n_old:
        return h.copy()
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h)
    return _fix_sum(h_new, n_new / 2.0)


def _make_random_q(m, rng, style=0):
    t = np.linspace(0.0, 1.0, m)
    if style == 0:
        q = 0.5 + 0.3 * np.sin(2.0 * np.pi * rng.uniform(1, 5) * t + rng.uniform(0, 6.28))
        q += 0.1 * np.sin(2.0 * np.pi * rng.uniform(3, 8) * t + rng.uniform(0, 6.28))
    elif style == 1:
        q = 0.5 + 0.25 * np.cos(2.0 * np.pi * rng.uniform(1, 4) * t)
        q += 0.15 * np.sin(4.0 * np.pi * rng.uniform(1, 3) * t + rng.uniform(0, 6.28))
    else:
        q = np.ones(m) * rng.uniform(0.3, 0.7)
        q += 0.2 * (np.sin(2.0 * np.pi * rng.uniform(1, 6) * t + rng.uniform(0, 6.28)))
        q += 0.1 * rng.randn(m)
    return np.clip(q, 0.0, 1.0)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')

    try:
        h_init = np.array(initial_h_values, dtype=float).reshape(-1)
        if h_init.size > 0 and np.all(np.isfinite(h_init)):
            n_init = len(h_init)
            h_init = _fix_sum(h_init, n_init / 2.0)
            c5_init = eval_c5(h_init)
            best_h = h_init.copy()
            best_c5 = c5_init
    except Exception:
        pass

    if best_h is None:
        n_def = 126
        m_def = n_def // 2
        t = np.linspace(0, 1, m_def)
        q = np.clip(0.5 + 0.3 * np.sin(2 * np.pi * 2 * t + 0.5), 0, 1)
        best_h = build_h(q, n_def)
        best_c5 = eval_c5(best_h)

    n_base = len(best_h)
    reserve = 25.0

    def remaining():
        return budget_s - (time.time() - t0)

    # Phase 1: Anti-symmetric LSE on parent n
    if n_base % 2 == 0 and remaining() > reserve + 20:
        q = q_from_h(best_h, n_base)
        h_sym = build_h(q, n_base)
        c5_sym = eval_c5(h_sym)
        if c5_sym < best_c5:
            best_h = h_sym.copy()
            best_c5 = c5_sym
            q = q_from_h(best_h, n_base)

        q_opt, c5_opt = _lse_anneal_sym(q, n_base, reserve, remaining, budget_s, t0)
        if c5_opt < best_c5:
            best_h = build_h(q_opt, n_base)
            best_c5 = c5_opt

    # Phase 2: Full-variable LSE
    if remaining() > reserve + 40:
        h_full, c5_full = _lse_anneal_full(best_h, n_base, reserve, remaining, budget_s, t0)
        if c5_full < best_c5:
            best_h = h_full.copy()
            best_c5 = c5_full

    # Phase 3: Try different n values
    if remaining() > reserve + 20:
        n_base = len(best_h)
        n_candidates = set()
        for delta in [-8, -6, -4, -2, 2, 4, 6, 8, 10, 12]:
            nc = n_base + delta
            if nc >= 40 and nc % 2 == 0:
                n_candidates.add(nc)
        n_candidates = sorted(n_candidates)

        for n_try in n_candidates:
            if remaining() < reserve + 15:
                break
            if n_try == n_base:
                continue
            h_try = _interp_h(best_h, n_try)
            q_try = q_from_h(h_try, n_try)
            q_opt, c5_opt = _lse_anneal_sym(q_try, n_try, reserve, remaining, budget_s, t0)
            h_candidate = build_h(q_opt, n_try)
            c5_final = eval_c5(h_candidate)
            if c5_final < best_c5:
                best_h = h_candidate.copy()
                best_c5 = c5_final

            # Full optimization on this n if promising
            if remaining() > reserve + 20 and c5_final < best_c5 + 0.001:
                h_f, c5_f = _lse_anneal_full(h_candidate, n_try, reserve, remaining, budget_s, t0)
                if c5_f < best_c5:
                    best_h = h_f.copy()
                    best_c5 = c5_f

    # Phase 4: Random restarts
    n_cur = len(best_h)
    if n_cur % 2 != 0:
        n_cur -= 1
    max_restarts = min(60, max(10, int(remaining() / 8.0)))
    for trial in range(max_restarts):
        if remaining() < reserve + 8:
            break
        style = trial % 3
        q_r = _make_random_q(n_cur // 2, rng, style)
        # Perturb toward best for some trials
        if trial % 4 == 0:
            q_best = q_from_h(best_h, n_cur)
            q_r = np.clip(0.5 * q_r + 0.5 * q_best + 0.1 * rng.randn(n_cur // 2), 0, 1)

        q_opt, c5_opt = _lse_anneal_sym(q_r, n_cur, reserve, remaining, budget_s, t0, max_stages=10)
        if c5_opt < best_c5:
            best_h = build_h(q_opt, n_cur).copy()
            best_c5 = c5_opt
            # Polish with full
            if remaining() > reserve + 20:
                h_f, c5_f = _lse_anneal_full(best_h, n_cur, reserve, remaining, budget_s, t0)
                if c5_f < best_c5:
                    best_h = h_f.copy()
                    best_c5 = c5_f

    # Phase 5: Final fine polish on best
    if remaining() > 10 and best_h is not None:
        n_fin = len(best_h)
        if n_fin % 2 == 0:
            q_fin = q_from_h(best_h, n_fin)
            for tau_f in [0.0003, 0.00015, 0.00008, 0.00004]:
                if remaining() < 5:
                    break
                def obj_fin(z):
                    return _grad_lse_sym(z, n_fin, tau_f)
                try:
                    res = minimize(obj_fin, q_fin, jac=True, method='L-BFGS-B',
                                   bounds=[(0.0, 1.0)] * (n_fin // 2),
                                   options={'maxiter': 100, 'gtol': 1e-12, 'maxls': 30})
                    q_fin = np.clip(res.x, 0.0, 1.0)
                except Exception:
                    pass
            h_fin = build_h(q_fin, n_fin)
            c5_fin = eval_c5(h_fin)
            if c5_fin < best_c5:
                best_h = h_fin.copy()
                best_c5 = c5_fin

        # Full final polish
        if remaining() > 8:
            def obj_full_fin(z):
                return _grad_lse_full(z, n_fin, 0.0005, 10000.0)
            try:
                res = minimize(obj_full_fin, best_h, jac=True, method='L-BFGS-B',
                               bounds=[(0.0, 1.0)] * n_fin,
                               options={'maxiter': 80, 'gtol': 1e-11, 'maxls': 20})
                h_fin2 = np.clip(res.x, 0.0, 1.0)
                h_fin2 = _fix_sum(h_fin2, n_fin / 2.0)
                c5_fin2 = eval_c5(h_fin2)
                if c5_fin2 < best_c5:
                    best_h = h_fin2.copy()
                    best_c5 = c5_fin2
            except Exception:
                pass

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _fix_sum(best_h, len(best_h) / 2.0)
    best_c5 = eval_c5(best_h)

    return best_h, best_c5, len(best_h)
