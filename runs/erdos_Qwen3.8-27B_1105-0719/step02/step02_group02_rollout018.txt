def project_feasible(x, target_sum):
    """Project x to [0,1]^n with sum = target_sum using bisection on offset."""
    n = len(x)
    lo = -1.0
    hi = 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    # Fix remaining error
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_c5(h, dx):
    """Compute exact C5 value."""
    corrs = np.correlate(h, 1.0 - h, mode='full')
    return float(corrs.max() * dx)

def corr_grad(h, lag_idx, n):
    """Gradient of corr[lag_idx] with respect to h."""
    d = lag_idx - (n - 1)
    grad = np.zeros(n)
    if d == 0:
        grad[:] = 1.0 - 2.0 * h
    elif d > 0:
        # sum_{i=0}^{n-1-d} h[i]*(1-h[i+d])
        # grad[i] += (1-h[i+d]) for i in [0, n-1-d]
        idx1 = np.arange(0, n - d)
        grad[idx1] += (1.0 - h[idx1 + d])
        # grad[i] += -h[i-d] for i in [d, n-1]
        idx2 = np.arange(d, n)
        grad[idx2] += (-h[idx2 - d])
    else:
        d2 = -d
        # sum_{i=d2}^{n-1} h[i]*(1-h[i-d2])
        # grad[i] += (1-h[i-d2]) for i in [d2, n-1]
        idx1 = np.arange(d2, n)
        grad[idx1] += (1.0 - h[idx1 - d2])

        idx2 = np.arange(0, n - d2)

    return grad

class EpigraphSolver:
    """Solves min s + rho*||h - anchor||^2 s.t. corr_k(h) - target_raw - s <= 0, sum(h)=n/2, 0<=h<=1, s>=0"""
    def __init__(self, h_anchor, target_raw, rho, n):
        self.h_anchor = h_anchor
        self.target_raw = target_raw
        self.rho = rho
        self.n = n
        self.n_vars = n + 1  # h_0..h_{n-1}, s
    
    def objective(self, x):
        h = x[:self.n]
        s = x[self.n]
        return s + self.rho * np.sum((h - self.h_anchor)**2) / self.n
    
    def objective_grad(self, x):
        h = x[:self.n]
        s = x[self.n]
        g = np.zeros(self.n_vars)
        g[:self.n] = 2.0 * self.rho * (h - self.h_anchor) / self.n
        g[self.n] = 1.0
        return g
    
    def constraint_ineq(self, x):
        """Returns vector of (target_raw + s - corr_k) for all k. Must be >= 0."""
        h = x[:self.n]
        s = x[self.n]
        corrs = np.correlate(h, 1.0 - h, mode='full')
        return self.target_raw + s - corrs
    
    def constraint_ineq_jac(self, x):
        h = x[:self.n]
        n = self.n
        L = 2 * n - 1
        jac = np.zeros((L, n + 1))
        for k in range(L):
            g = corr_grad(h, k, n)
            jac[k, :n] = g
            jac[k, n] = 1.0  # derivative of (target_raw + s - corr_k) wrt s is +1
        return jac
    
    def constraint_eq(self, x):
        h = x[:self.n]
        return h.sum() - self.n / 2.0
    
    def constraint_eq_jac(self, x):
        g = np.zeros(self.n_vars)
        g[:self.n] = 1.0
        return g
    
    def solve(self, x0, maxiter):
        n = self.n
        bounds = [(0.0, 1.0)] * n + [(0.0, self.target_raw + 10.0)]
        
        cons = [
            {'type': 'ineq', 'fun': self.constraint_ineq, 'jac': self.constraint_ineq_jac},
            {'type': 'eq', 'fun': self.constraint_eq, 'jac': self.constraint_eq_jac}
        ]
        
        try:
            res = minimize(self.objective, x0, jac=self.objective_grad,
                          method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-12})
            return res
        except Exception:
            return None

class SmoothMaxSolver:
    """Solves min logsumexp(beta * corrs/h) using SLSQP with softmax approximation."""
    def __init__(self, dx, n, target_sum, beta):
        self.dx = dx
        self.n = n
        self.target_sum = target_sum
        self.beta = beta
    
    def objective(self, h):
        corrs = np.correlate(h, 1.0 - h, mode='full') * self.dx
        return float(logsumexp(self.beta * corrs) / self.beta)
    
    def objective_grad(self, h):
        n = self.n
        corrs = np.correlate(h, 1.0 - h, mode='full') * self.dx
        shifted = corrs - corrs.max()
        w = np.exp(self.beta * shifted)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(2*n - 1) / (2*n - 1)
        else:
            w /= w_sum
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] > 1e-12:
                grad += w[k] * corr_grad(h, k, n)
        return grad * self.dx
    
    def constraint_eq(self, h):
        return h.sum() - self.target_sum
    
    def constraint_eq_jac(self, h):
        return np.ones(self.n)
    
    def solve(self, h0, maxiter):
        bounds = [(0.0, 1.0)] * self.n
        cons = [{'type': 'eq', 'fun': self.constraint_eq, 'jac': self.constraint_eq_jac}]
        try:
            res = minimize(self.objective, h0, jac=self.objective_grad,
                          method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-14})
            return project_feasible(res.x, self.target_sum)
        except Exception:
            return project_feasible(h0, self.target_sum)

def build_antipalindromic(v, n):
    """Build anti-palindromic vector from first half v. Requires n even."""
    h = np.zeros(n)
    m = n // 2
    h[:m] = v
    h[m:] = 1.0 - v[::-1]
    return h

def subgradient_step(h, dx, target_sum, step_size, top_m=10):

def epigraph_active_set(h_start, target_raw, active_lags, n, dx, rho, maxiter):
    """Solve epigraph with only active lag constraints."""
    # Build constraint matrix for active lags only
    # Variables: h_0..h_{n-1}, s
    # Minimize: s + rho * ||h - h_start||^2 / n
    # Subject to: corr_k(h) - target_raw - s <= 0 for k in active_lags
    #             sum(h) = n/2
    #             0 <= h[i] <= 1, s >= 0
    
    class ActiveEpigraph:
        def __init__(self, h_anchor, target_raw, active_lags, n, rho):
            self.h_anchor = h_anchor
            self.target_raw = target_raw
            self.active_lags = active_lags
            self.n = n
            self.rho = rho
        
        def obj(self, x):
            h = x[:self.n]
            s = x[self.n]
            return s + self.rho * np.sum((h - self.h_anchor)**2) / self.n
        
        def obj_grad(self, x):
            h = x[:self.n]
            g = np.zeros(self.n + 1)
            g[:self.n] = 2.0 * self.rho * (h - self.h_anchor) / self.n
            g[self.n] = 1.0
            return g
        
        def ineq(self, x):
            h = x[:self.n]
            s = x[self.n]
            corrs = np.correlate(h, 1.0 - h, mode='full')
            return self.target_raw + s - corrs[self.active_lags]
        
        def ineq_jac(self, x):
            h = x[:self.n]
            n = self.n
            k_active = len(self.active_lags)
            jac = np.zeros((k_active, n + 1))
            for i, k in enumerate(self.active_lags):
                jac[i, :n] = corr_grad(h, k, n)
                jac[i, n] = 1.0
            return jac
        
        def eq(self, x):
            return x[:self.n].sum() - self.n / 2.0
        
        def eq_jac(self, x):
            g = np.zeros(self.n + 1)
            g[:self.n] = 1.0
            return g
    
    solver = ActiveEpigraph(h_start, target_raw, active_lags, n, rho)
    bounds = [(0.0, 1.0)] * n + [(0.0, target_raw + 20.0)]
    cons = [
        {'type': 'ineq', 'fun': solver.ineq, 'jac': solver.ineq_jac},

import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time

def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_c5(h, dx):
    corrs = np.correlate(h, 1.0 - h, mode='full')
    return float(corrs.max() * dx)

def get_active_lags(h, n, threshold=0.01):
    """Get lags where correlation is within threshold of maximum."""
    corrs = np.correlate(h, 1.0 - h, mode='full')
    max_c = corrs.max()
    active = np.where(corrs >= max_c - threshold)[0]
    return active, corrs, max_c

def corr_grad(h, lag_idx, n):
    d = lag_idx - (n - 1)
    grad = np.zeros(n)
    if d == 0:
        grad[:] = 1.0 - 2.0 * h
    elif d > 0:
        idx1 = np.arange(0, n - d)
        grad[idx1] += (1.0 - h[idx1 + d])
        idx2 = np.arange(d, n)
        grad[idx2] += (-h[idx2 - d])
    else:
        d2 = -d
        idx1 = np.arange(d2, n)
        grad[idx1] += (1.0 - h[idx1 - d2])
        idx2 = np.arange(0, n - d2)
        grad[idx2] += (-h[idx2 + d2])
    return grad

class SmoothMaxObj:
    def __init__(self, n, dx, beta):
        self.n = n
        self.dx = dx
        self.beta = beta
    
    def fun(self, h):
        corrs = np.correlate(h, 1.0 - h, mode='full') * self.dx
        return float(logsumexp(self.beta * corrs) / self.beta)
    
    def grad(self, h):
        n = self.n
        corrs = np.correlate(h, 1.0 - h, mode='full') * self.dx
        shifted = corrs - corrs.max()
        w = np.exp(self.beta * shifted)
        ws = w.sum()
        if ws < 1e-30:
            w = np.ones(2*n-1) / (2*n-1)
        else:
            w /= ws
        g = np.zeros(n)
        for k in range(2*n-1):
            if w[k] > 1e-10:
                g += w[k] * corr_grad(h, k, n)
        return g * self.dx

class EqConstraint:
    def __init__(self, n, target_sum):
        self.n = n
        self.target_sum = target_sum
    
    def fun(self, h):
        return h.sum() - self.target_sum
    
    def jac(self, h):
        return np.ones(self.n)

class EpigraphObj:
    def __init__(self, h_anchor, target_raw, active_lags, n, rho):
        self.h_anchor = h_anchor
        self.target_raw = target_raw
        self.active_lags = active_lags
        self.n = n
        self.rho = rho
    
    def fun(self, x):
        h = x[:self.n]
        s = x[self.n]
        return s + self.rho * np.sum((h - self.h_anchor)**2) / self.n
    
    def grad(self, x):
        h = x[:self.n]
        g = np.zeros(self.n + 1)
        g[:self.n] = 2.0 * self.rho * (h - self.h_anchor) / self.n
        g[self.n] = 1.0
        return g
    
    def ineq_fun(self, x):

        s = x[self.n]
        corrs = np.correlate(h, 1.0 - h, mode='full')
        return self.target_raw + s - corrs[self.active_lags]

        n = self.n

            jac[i, :n] = corr_grad(h, k, n)
            jac[i, n] = 1.0
        return jac
    
    def eq_fun(self, x):
        return x[:self.n].sum() - self.n / 2.0
    
    def eq_jac(self, x):
        g = np.zeros(self.n + 1)
        g[:self.n] = 1.0
        return g

def solve_smooth_max(h0, n, dx, target_sum, beta, maxiter):
    sm = SmoothMaxObj(n, dx, beta)
    ec = EqConstraint(n, target_sum)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': ec.fun, 'jac': ec.jac}]

                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, target_sum)
    except Exception:
        return project_feasible(h0, target_sum)

def solve_epigraph(h_start, target_raw, active_lags, n, rho, maxiter):

    eo = EpigraphObj(h_start, target_raw, active_lags, n, rho)

    except Exception:

    h[m:] = 1.0 - v[::-1]
    return h

def subgrad_momentum(h0, dx, target_sum, max_iter=15000, time_fn=None):

import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time

def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_c5(h, dx):
    corrs = np.correlate(h, 1.0 - h, mode='full')
    return float(corrs.max() * dx)

def get_active_lags(h, n, frac=0.05):
    corrs = np.correlate(h, 1.0 - h, mode='full')
    max_c = corrs.max()
    active = np.where(corrs >= max_c - frac * max_c)[0]
    return active, corrs, max_c

def corr_grad(h, lag_idx, n):
    d = lag_idx - (n - 1)
    grad = np.zeros(n)
    if d == 0:
        grad[:] = 1.0 - 2.0 * h
    elif d > 0:
        idx1 = np.arange(0, n - d)
        grad[idx1] += (1.0 - h[idx1 + d])
        idx2 = np.arange(d, n)
        grad[idx2] += (-h[idx2 - d])
    else:
        d2 = -d
        idx1 = np.arange(d2, n)
        grad[idx1] += (1.0 - h[idx1 - d2])
        idx2 = np.arange(0, n - d2)
        grad[idx2] += (-h[idx2 + d2])
    return grad

class SmoothMaxObj:
    def __init__(self, n, dx, beta):
        self.n = n
        self.dx = dx
        self.beta = beta

    def fun(self, h):
        corrs = np.correlate(h, 1.0 - h, mode='full') * self.dx
        return float(logsumexp(self.beta * corrs) / self.beta)

    def grad(self, h):
        n = self.n
        corrs = np.correlate(h, 1.0 - h, mode='full') * self.dx
        shifted = corrs - corrs.max()
        w = np.exp(self.beta * shifted)
        ws = w.sum()
        if ws < 1e-30:
            w = np.ones(2 * n - 1) / (2 * n - 1)
        else:
            w /= ws
        g = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] > 1e-10:
                g += w[k] * corr_grad(h, k, n)
        return g * self.dx

class EqConstraint:
    def __init__(self, n, target_sum):
        self.n = n
        self.target_sum = target_sum

    def fun(self, h):
        return h.sum() - self.target_sum

    def jac(self, h):
        return np.ones(self.n)

class EpigraphObj:
    def __init__(self, h_anchor, target_raw, active_lags, n, rho):
        self.h_anchor = h_anchor
        self.target_raw = target_raw
        self.active_lags = active_lags
        self.n = n
        self.rho = rho

    def fun(self, x):
        h = x[:self.n]
        s = x[self.n]
        return s + self.rho * np.sum((h - self.h_anchor) ** 2) / self.n

    def grad(self, x):
        h = x[:self.n]
        g = np.zeros(self.n + 1)
        g[:self.n] = 2.0 * self.rho * (h - self.h_anchor) / self.n
        g[self.n] = 1.0
        return g

    def ineq_fun(self, x):
        h = x[:self.n]
        s = x[self.n]
        corrs = np.correlate(h, 1.0 - h, mode='full')
        return self.target_raw + s - corrs[self.active_lags]

    def ineq_jac(self, x):
        h = x[:self.n]
        n = self.n
        ka = len(self.active_lags)
        jac = np.zeros((ka, n + 1))
        for i, k in enumerate(self.active_lags):
            jac[i, :n] = corr_grad(h, k, n)
            jac[i, n] = 1.0
        return jac

    def eq_fun(self, x):
        return x[:self.n].sum() - self.n / 2.0

    def eq_jac(self, x):
        g = np.zeros(self.n + 1)
        g[:self.n] = 1.0
        return g

def solve_smooth_max(h0, n, dx, target_sum, beta, maxiter):
    sm = SmoothMaxObj(n, dx, beta)
    ec = EqConstraint(n, target_sum)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': ec.fun, 'jac': ec.jac}]
    try:
        res = minimize(sm.fun, h0, jac=sm.grad, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, target_sum)
    except Exception:
        return project_feasible(h0, target_sum)

def solve_epigraph(h_start, target_raw, active_lags, n, rho, maxiter):
    ts = n / 2.0
    eo = EpigraphObj(h_start, target_raw, active_lags, n, rho)
    corrs = np.correlate(h_start, 1.0 - h_start, mode='full')
    s0 = max(0.0, corrs.max() - target_raw) + 0.001
    x0 = np.concatenate([h_start, [s0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, max(target_raw + 20.0, s0 * 3.0))]
    cons = [
        {'type': 'ineq', 'fun': eo.ineq_fun, 'jac': eo.ineq_jac},
        {'type': 'eq', 'fun': eo.eq_fun, 'jac': eo.eq_jac}
    ]
    try:
        res = minimize(eo.fun, x0, jac=eo.grad, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-13})
        return project_feasible(res.x[:n], ts)
    except Exception:
        return project_feasible(h_start, ts)

def build_antipalindromic(v, n):
    h = np.zeros(n)
    m = n // 2
    h[:m] = v
    h[m:] = 1.0 - v[::-1]
    return h

def make_tm_antipalindromic(n):
    m = n // 2
    v = np.zeros(m)
    for i in range(m):
        v[i] = float(bin(i).count('1') % 2)
    return build_antipalindromic(v, n)

def subgrad_momentum(h0, dx, target_sum, max_iter, t0, budget_s, margin=30):
    n = len(h0)
    h = h0.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    alpha = 0.06
    gamma = 0.82
    top_m = max(5, n // 12)
    no_improve = 0

    for it in range(max_iter):
        if budget_s - (time.time() - t0) < margin:
            break
        corrs = np.correlate(h, 1.0 - h, mode='full')
        c5_val = corrs.max() * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 8000:
            break
        top_idx = np.argsort(corrs)[-top_m:]
        grad = np.zeros(n)
        for k in top_idx:
            grad += corr_grad(h, k, n)
        grad /= len(top_idx)
        grad *= dx
        v = gamma * v - alpha * grad
        h_new = project_feasible(h + v, target_sum)
        new_c5 = compute_c5(h_new, dx)
        if new_c5 < c5_val - 1e-14:
            h = h_new
            alpha = min(alpha * 1.003, 0.25)
        else:
            alpha *= 0.94
            v *= 0.3
        if alpha < 1e-12:
            break
    return best_h, best_c5
