    return obj, grad


def slsqp_smooth(h0, dx, target_sum, beta, maxiter=2000):
    n = len(h0)

    def obj(x):
        return logsumexp(beta * np.correlate(x, 1.0 - x, mode='full')) / beta * dx

    def grad_obj(x):
        return smooth_max_obj_and_grad(x, beta)[1]

    def c_fun(x):
        return x.sum() - target_sum

    def c_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': c_fun, 'jac': c_jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0, jac=grad_obj, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        h_out = project_feasible(res.x, target_sum)
    except Exception:
        h_out = project_feasible(h0, target_sum)
    return h_out, c5_of_h(h_out)


def lp_polish_aggressive(h0, dx, target_sum, n_steps=200, tol=1e-14):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h)
    lam = 0.5

    for s in range(n_steps):
        corrs = np.correlate(h, 1.0 - h, mode='full')
        max_c = corrs.max()
        c5_val = max_c * dx

        active_thresh = max_c - 0.05 * max_c
        active_idx = np.where(corrs >= active_thresh)[0]
        if len(active_idx) < 5:
            active_idx = np.argsort(corrs)[-max(10, n // 2):]

        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0

        A_rows = []
        b_rows = []
        for k_idx in active_idx:
            row = np.zeros(n + 1)
            row[:n] = dx * grad_corr_k(h, k_idx, n)
            row[n] = -1.0
            A_rows.append(row)
            b_rows.append(-corrs[k_idx] * dx - 1e-13)

        A_ub = np.array(A_rows)
        b_ub = np.array(b_rows)

        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)

        bounds = []
        for i in range(n):
            bounds.append((-h[i] * lam, (1.0 - h[i]) * lam))
        bounds.append((0.0, None))

        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if res.status != 0:
                lam *= 0.5
                continue
            delta = res.x[:n]
            h_new = project_feasible(h + delta, target_sum)
            new_c5 = c5_of_h(h_new)
            if new_c5 < c5_val - tol:
                h = h_new
                lam = min(lam * 1.2, 1.0)
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h_new.copy()
            else:
                lam *= 0.5
                if lam < 0.01:
                    break
        except Exception:
            break

    return best_h, best_c5


def coordinate_descent(h0, target_sum, n_sweeps=30):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h)

    for sweep in range(n_sweeps):
        for j in range(n):
            old_val = h[j]
            corrs = np.correlate(h, 1.0 - h, mode='full')
            max_c = corrs.max()
            active_thresh = max_c - 0.03 * max_c
            active_idx = np.where(corrs >= active_thresh)[0]

            slopes = []
            intercepts = []
            for k_idx in active_idx:
                offset = k_idx - (n - 1)
                a = 0.0
                b = corrs[k_idx]
                i_lo = max(0, -offset)
                i_hi = min(n - 1, n - 1 - offset)
                if i_lo <= j <= i_hi:
                    b += h[j + offset]
                    a += -(1.0 - h[j])
                if i_lo <= j - offset <= i_hi and 0 <= j - offset < n:
                    b += h[j]
                    a += (1.0 - h[j - offset])
                b -= h[j]
                slopes.append(a)
                intercepts.append(b)

            slopes = np.array(slopes)
            intercepts = np.array(intercepts)

            if len(slopes) == 0:
                continue

            best_t = old_val
            best_val = np.max(slopes * old_val + intercepts)

            candidates = [0.0, 1.0, old_val]
            for i in range(len(slopes)):
                for j2 in range(i + 1, len(slopes)):
                    if abs(slopes[i] - slopes[j2]) < 1e-15:
                        continue
                    t_int = (intercepts[j2] - intercepts[i]) / (slopes[i] - slopes[j2])
                    if 0.0 <= t_int <= 1.0:
                        candidates.append(t_int)

            for t_cand in candidates:
                t_cand = max(0.0, min(1.0, t_cand))
                val = np.max(slopes * t_cand + intercepts)
                if val < best_val - 1e-15:
                    best_val = val
                    best_t = t_cand

            h[j] = best_t
            h = project_feasible(h, target_sum)

        cur_c5 = c5_of_h(h)
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()

    return best_h, best_c5


def projected_subgradient(h0, dx, target_sum, max_iter=5000, beta_w=50.0):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h)
    alpha = 2.0 / (np.linalg.norm(h) + 1.0)
    no_improve = 0

    for it in range(max_iter):
        corrs = np.correlate(h, 1.0 - h, mode='full')
        max_c = corrs.max()
        c5_val = max_c * dx

        if c5_val < best_c5 - 1e-16:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 3000:
            break

        shifted = corrs - corrs.max()
        w = np.exp(beta_w * shifted)
        w_sum = w.sum()
        w /= w_sum

        grad = np.zeros(n)
        for k_idx in range(len(corrs)):
            if w[k_idx] > 1e-10:
                grad += w[k_idx] * grad_corr_k(h, k_idx, n)
        grad *= dx

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        a_polyak = (c5_val - best_c5) / (gnorm ** 2 + 1e-16)
        step = min(alpha, max(a_polyak, 1e-10))

        h_new = project_feasible(h - step * grad, target_sum)
        new_c5 = c5_of_h(h_new)

        tries = 0
        while new_c5 > c5_val + 1e-15 and tries < 5:
            step *= 0.5
            h_new = project_feasible(h - step * grad, target_sum)
            new_c5 = c5_of_h(h_new)
            tries += 1

        h = h_new
        alpha *= (1.0 - 0.2 / (it + 100))
        alpha = max(alpha, 1e-10)

    return best_h, best_c5


def make_periodic_h(n, target_sum, period, fill_frac):
    h = np.zeros(n)
    fill = int(round(fill_frac * period))
    fill = max(1, min(fill, period - 1))
    for i in range(n):
        h[i] = 1.0 if (i % period) < fill else 0.0
    return project_feasible(h, target_sum)


def make_random_h(n, target_sum, rng):
    h = np.zeros(n)
    idx = rng.permutation(n)
    n_ones = int(target_sum)
    h[idx[:n_ones]] = 1.0
    frac = target_sum - n_ones
    if frac > 1e-10:
        h[idx[n_ones]] = frac
    return h


def interpolate_h(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0.0, 2.0, n_old)
    x_new = np.linspace(0.0, 2.0, n_new)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new / 2.0)


def remaining_time(t0, budget_s):
    return budget_s - (time.time() - t0)


def full_optimize(h0, t0, budget_s, time_limit=60):
    n = len(h0)
    dx = 2.0 / n
    target_sum = n / 2.0
    best_h = h0.copy()
    best_c5 = c5_of_h(h0)

    for beta in [200, 500, 1500, 4000]:
        if remaining_time(t0, budget_s) < time_limit:
            break
        h_opt, c5_val = slsqp_smooth(best_h, dx, target_sum, beta, 1500)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_opt

    if remaining_time(t0, budget_s) > time_limit:
        h_lp, c5_lp = lp_polish_aggressive(best_h, dx, target_sum, 150)
        if c5_lp < best_c5:
            best_c5 = c5_lp
            best_h = h_lp

    if remaining_time(t0, budget_s) > time_limit:
        h_cd, c5_cd = coordinate_descent(best_h, target_sum, 20)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd

    if remaining_time(t0, budget_s) > time_limit:
        h_opt, c5_val = slsqp_smooth(best_h, dx, target_sum, 5000, 1000)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_opt
        if remaining_time(t0, budget_s) > time_limit:
            h_lp, c5_lp = lp_polish_aggressive(best_h, dx, target_sum, 100)
            if c5_lp < best_c5:
                best_c5 = c5_lp
                best_h = h_lp

    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in dir() or True:
        try:
            h_parent = np.array(initial_h_values, dtype=float)
        except NameError:
            h_parent = None
    else:
        h_parent = None

    if h_parent is not None and len(h_parent) > 0:
        n_p = len(h_parent)
        h_parent = project_feasible(h_parent, n_p / 2.0)
        best_h = h_parent.copy()
        best_c5 = c5_of_h(h_parent)
    else:
        n_p = 126
        best_h = make_periodic_h(n_p, n_p / 2.0, 9, 0.5)
        best_c5 = c5_of_h(best_h)

    # Phase 1: Polish parent
    if remaining_time(t0, budget_s) > 100:
        best_h, best_c5 = full_optimize(best_h, t0, budget_s, 80)

    # Phase 2: Try n=128 and n=130 (easier targets)
    for n_try in [128, 130, 132]:
        if remaining_time(t0, budget_s) < 80:
            break
        h_interp = interpolate_h(best_h, n_try)
        h_opt, c5_val = full_optimize(h_interp, t0, budget_s, 60)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_opt

    # Phase 3: Multi-start at n=126
    n_cur = len(best_h)
    dx_cur = 2.0 / n_cur
    ts_cur = n_cur / 2.0

    for trial in range(8):
        if remaining_time(t0, budget_s) < 60:
            break
        if trial % 3 == 0:
            h_init = make_random_h(n_cur, ts_cur, rng)
        elif trial % 3 == 1:
            h_init = make_periodic_h(n_cur, ts_cur, rng.choice([7, 8, 9, 10, 11]), 0.45 + 0.1 * rng.random())
        else:
            h_init = project_feasible(best_h + 0.3 * (rng.random(n_cur) - 0.5), ts_cur)

        h_opt, c5_val = full_optimize(h_init, t0, budget_s, 40)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_opt

    # Phase 4: Subgradient + coordinate descent on best
    if remaining_time(t0, budget_s) > 40:
        n_b = len(best_h)
        dx_b = 2.0 / n_b
        ts_b = n_b / 2.0
        h_sg, c5_sg = projected_subgradient(best_h, dx_b, ts_b, 5000)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg

    if remaining_time(t0, budget_s) > 30:
        n_b = len(best_h)
        ts_b = n_b / 2.0
        h_cd, c5_cd = coordinate_descent(best_h, ts_b, 40)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd

    # Phase 5: Final intensive polish
    if remaining_time(t0, budget_s) > 20:
        n_b = len(best_h)
        dx_b = 2.0 / n_b
        ts_b = n_b / 2.0
        best_h, best_c5 = full_optimize(best_h, t0, budget_s, 15)

    # Phase 6: One more LP + coordinate descent round
    if remaining_time(t0, budget_s) > 10:
        n_b = len(best_h)
        dx_b = 2.0 / n_b
        ts_b = n_b / 2.0
        h_lp, c5_lp = lp_polish_aggressive(best_h, dx_b, ts_b, 100)
        if c5_lp < best_c5:
            best_c5 = c5_lp
            best_h = h_lp
        if remaining_time(t0, budget_s) > 8:
            h_cd2, c5_cd2 = coordinate_descent(best_h, ts_b, 30)
            if c5_cd2 < best_c5:
                best_c5 = c5_cd2
                best_h = h_cd2

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_c5 = c5_of_h(best_h)

    return best_h, best_c5, len(best_h)
