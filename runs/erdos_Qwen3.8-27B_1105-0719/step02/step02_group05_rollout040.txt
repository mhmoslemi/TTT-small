def compute_corrs(h):

def grad_corr_d(h, d, n):
    g = np.zeros(n)
    if d >= 0:
        if d < n:
            g[d:] += (1.0 - h[:n - d])
            g[:n - d] += (-h[d:])
    else:
        dd = -d
        if dd < n:
            g[:n - dd] += (1.0 - h[dd:])
            g[dd:] += (-h[:n - dd])
    return g

            if rem() < 30:
                break
            q_new = smooth_q_solve(q_best, n, beta, maxiter=500)
            h_new = build_h_from_q(q_new, n)
            h_new = project_feasible(h_new, target_sum)
            s_new, _ = exact_score(h_new)
            if s_new < best_c5 - 1e-14:
                best_c5 = s_new
                best_h = h_new
                q_best = q_new

    # Phase 4: SLP again after smooth polish
    if rem() > 80:
        h4, c54 = slp_active_phase(best_h, n, t_start, rem())
        if c54 < best_c5:
            best_c5 = c54
            best_h = h4

    # Phase 5: Binary local search
    if rem() > 60:
        hb, cb = binary_local_search(best_h, n, 100, t_start, rem())
        if cb < best_c5:
            best_c5 = cb
            best_h = hb

    # Phase 6: Try different n values
    n_candidates = [124, 128, 120, 130, 116, 132, 104, 96, 140, 144]
    for nc in n_candidates:
        if rem() < 70:
            break
        ts_c = nc / 2.0
        if nc % 2 != 0:
            ts_c = nc / 2.0
        x_orig = np.linspace(0.0, 2.0, len(best_h))
        x_new = np.linspace(0.0, 2.0, nc)
        h_interp = np.interp(x_new, x_orig, best_h)
        h_interp = project_feasible(h_interp, nc / 2.0)

        h_try, c5_try = solve_slp(h_interp, nc, 100, t_start, budget_s - 60, False, 0)
        if rem() < 40:
            continue
        h_try2, c5_try2 = slp_active_phase(h_try, nc, t_start, rem())
        if c5_try2 < c5_try:
            h_try, c5_try = h_try2, c5_try2

        if c5_try < best_c5 - 1e-12:
            best_c5 = c5_try
            best_h = h_try

    # Phase 7: Final SLP polish on best solution
    if rem() > 40:
        n_final = len(best_h)
        h_f, c5_f = solve_slp(best_h, n_final, 200, t_start, budget_s - 30, True, 50)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f

    # Phase 8: Binary search on final best
    if rem() > 30:
        n_final = len(best_h)
        hb2, cb2 = binary_local_search(best_h, n_final, 80, t_start, rem())
        if cb2 < best_c5:
            best_c5 = cb2
            best_h = hb2

    # Phase 9: One more SLP round
    if rem() > 20:
        n_final = len(best_h)
        h_last, c5_last = slp_active_phase(best_h, n_final, t_start, rem())
        if c5_last < best_c5:
            best_c5 = c5_last
            best_h = h_last

    # Final validation
    n_out = len(best_h)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_feasible(best_h, n_out / 2.0)
    final_score, _ = exact_score(best_h)
    if final_score < best_c5:
        best_c5 = final_score

    return best_h, best_c5, n_out
