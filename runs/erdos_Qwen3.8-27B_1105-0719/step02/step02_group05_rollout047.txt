import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time


def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corrs = compute_corrs(h)
    return corrs.max() * dx


def project_to_box_sum(x, target_sum):
    n = len(x)
    lo, hi = -3.0, 3.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        y = np.clip(x - mid, 0.0, 1.0)
        if y.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    y = np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - y.sum()
    if abs(diff) > 1e-14:
        mask = (y > 1e-12) & (y < 1.0 - 1e-12)
        if mask.sum() > 0:
            y[mask] += diff / mask.sum()
            y = np.clip(y, 0.0, 1.0)
    return y


def grad_single_shift(h, d, n):
    g = np.zeros(n)
    if d >= 0:
        if d < n:
            g[d:] += (1.0 - h[:n - d])
            g[:n - d] += (-h[d:])
    else:
        dd = -d
        if dd < n:
            g[:n - dd] += (1.0 - h[dd:])
            g[dd:] += (-h[:n - dd])
    return g


def smooth_obj_grad(h, beta, dx):
    n = len(h)
    corrs = compute_corrs(h)
    val = logsumexp(beta * corrs) / beta * dx
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(len(corrs)) / len(corrs)
    else:
        w /= w_sum
    g = np.zeros(n)
    for k in range(len(corrs)):
        if w[k] < 1e-15:
            continue
        d = k - (n - 1)
        g += w[k] * grad_single_shift(h, d, n)
    return val, g * dx


class SmoothObj:
    def __init__(self, n, beta):
        self.n = n
        self.beta = beta
        self.dx = 2.0 / n
        self.target_sum = n / 2.0

    def value(self, x):

class EpigraphFull:
    def __init__(self, n, active_k):
        self.n = n
        self.active_k = active_k
        self.dx = 2.0 / n
        self.target_sum = n / 2.0

    def obj(self, x):
        return x[0]

    def obj_grad(self, x):
        g = np.zeros(self.n + 1)
        g[0] = 1.0
        return g

    def corr_constraint(self, x, k_idx):
        t = x[0]
        h = x[1:]
        k = self.active_k[k_idx]
        d = k - (self.n - 1)
        n = self.n
        if d >= 0:
            val = np.sum(h[d:] * (1.0 - h[:n - d]))
        else:
            dd = -d
            val = np.sum(h[dd:] * (1.0 - h[:n - dd]))
        return t - val * self.dx

    def corr_constraint_grad(self, x, k_idx):
        h = x[1:]
        k = self.active_k[k_idx]
        d = k - (self.n - 1)
        n = self.n
        g = np.zeros(n + 1)
        g[0] = 1.0
        g[1:] = -grad_single_shift(h, d, n) * self.dx
        return g

    def sum_constraint(self, x):
        return x[1:].sum() - self.target_sum

    def sum_constraint_grad(self, x):
        g = np.zeros(self.n + 1)
        g[1:] = 1.0
        return g

class ConWrapper:
    def __init__(self, ep, k_idx):

        self.k_idx = k_idx

    def fun(self, x):

    return w.fun

    return w.jac


def solve_epigraph(h0, active_k, t0, maxiter=200):
    n = len(h0)
    ep = EpigraphFull(n, active_k)
    x0 = np.concatenate([[t0], h0])
    cons = []
    for i in range(len(active_k)):
        w = ConWrapper(ep, i)
        cons.append({'type': 'ineq', 'fun': w.fun, 'jac': w.jac})
    cons.append({'type': 'eq', 'fun': ep.sum_constraint, 'jac': ep.sum_constraint_grad})
    bounds = [(0.0, 3.0)] + [(0.0, 1.0)] * n
    try:
        res = minimize(ep.obj, x0, jac=ep.obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        h_out = project_to_box_sum(res.x[1:], n / 2.0)
        return h_out, res.x[0], res

        return h0, t0, None

        for beta in [8000, 15000]:
            if remaining() < 20:
                break
            h_f = smooth_optimize(best_h, beta, maxiter=300)
            s_f = exact_score(h_f)
            if s_f < best_c5 - 1e-14:
                best_c5 = s_f
                best_h = h_f

    if remaining() > 30:
        n_cur = len(best_h)
        h_f2, c5_f2 = active_set_refine(best_h, min(remaining() - 5, 50), max_active=80)
        s_f2 = exact_score(h_f2)
        if s_f2 < best_c5 - 1e-14:
            best_c5 = s_f2
            best_h = h_f2

    # Phase 9: Multi-start perturbation
    if remaining() > 60:
        for attempt in range(3):
            if remaining() < 30:
                break
            perturbation = rng.randn(len(best_h)) * 0.02
            h_pert = project_to_box_sum(best_h + perturbation, len(best_h) / 2.0)
            h_pert_sm = smooth_optimize(h_pert, 1000, maxiter=300)
            h_pert_act, _ = active_set_refine(h_pert_sm, min(remaining() - 10, 40), max_active=60)
            s_pert = exact_score(h_pert_act)
            if s_pert < best_c5 - 1e-14:
                best_c5 = s_pert
                best_h = h_pert_act

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    n_out = len(best_h)
    best_h = project_to_box_sum(best_h, n_out / 2.0)
    final_score = exact_score(best_h)
    if final_score < best_c5:
        best_c5 = final_score

    return best_h, best_c5, n_out
