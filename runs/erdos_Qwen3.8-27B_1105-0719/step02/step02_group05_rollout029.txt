import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time


def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full').max() * dx


def project_box_sum(h, target_sum):
    n = len(h)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        c = np.clip(h - mid, 0.0, 1.0)
        if c.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        m = (h > 1e-10) & (h < 1.0 - 1e-10)
        if m.sum() > 0:
            h[m] += diff / m.sum()
        else:
            h += diff / n
    return np.clip(h, 0.0, 1.0)


def corr_grad_lag(h, d, n):
    g = np.zeros(n)
    if d >= 0:
        if d < n:
            g[d:] += (1.0 - h[:n - d])
            g[:n - d] += (-h[d:])
    else:
        dd = -d
        if dd < n:
            g[:n - dd] += (1.0 - h[dd:])
            g[dd:] += (-h[:n - dd])
    return g


def smooth_obj_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corrs = np.correlate(h, 1.0 - h, mode='full')
    val = logsumexp(beta * corrs) / beta * dx
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.ones_like(corrs) / len(corrs)
    else:
        w /= ws
    g = np.zeros(n)
    for k in range(len(corrs)):
        if w[k] < 1e-14:
            continue
        d = k - (n - 1)
        g += w[k] * corr_grad_lag(h, d, n)
    return val, g * dx


def topk_obj_grad(h, k_top):
    n = len(h)
    dx = 2.0 / n
    corrs = np.correlate(h, 1.0 - h, mode='full')
    val = corrs.max() * dx
    top_idx = np.argsort(corrs)[::-1][:k_top]
    top_vals = corrs[top_idx]
    w = top_vals / top_vals.sum()
    g = np.zeros(n)
    for idx, wi in zip(top_idx, w):
        d = idx - (n - 1)
        g += wi * corr_grad_lag(h, d, n)
    return val, g * dx


def optimize_slqp(h0, n, beta, target_sum, maxiter=300):
    def obj(x):
        return smooth_obj_grad(x, beta)[0]

    def obj_grad(x):
        return smooth_obj_grad(x, beta)[1]

    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-13})
        return project_box_sum(res.x, target_sum)
    except Exception:
        return h0.copy()


def projected_gradient(h0, n, target_sum, n_iter, beta_init=500, beta_end=5000):
    h = h0.copy()
    best_h = h.copy()
    best_f = exact_score(h)
    alpha = 0.05
    for it in range(n_iter):
        beta = beta_init + (beta_end - beta_init) * (it / max(1, n_iter - 1))
        f_val, g = smooth_obj_grad(h, beta)
        h_new = project_box_sum(h - alpha * g, target_sum)
        f_new = exact_score(h_new)
        if f_new < f_val + 1e-15:
            h = h_new
            alpha = min(alpha * 1.05, 0.3)
            if f_new < best_f:
                best_f = f_new
                best_h = h.copy()
        else:
            alpha *= 0.7
        if alpha < 1e-9:
            break
    return best_h, best_f

    log_odds_sharp = log_odds * t
    h_new = 1.0 / (1.0 + np.exp(-log_odds_sharp))
    return project_box_sum(h_new, target_sum)

            if time_left() < 60:
                break
            h_mix = project_box_sum(mix * best_h + (1 - mix) * h_rand, n_best / 2.0)
            h_new = optimize_slqp(h_mix, n_best, 300, n_best / 2.0, maxiter=250)
            h_new = optimize_slqp(h_new, n_best, 2000, n_best / 2.0, maxiter=250)
            s = exact_score(h_new)
            if s < best_c5:
                best_c5 = s
                best_h = h_new

    # Phase 6: Final high-beta polish
    for beta in [5000, 10000, 20000]:
        if time_left() < 40:
            break
        h_new = optimize_slqp(best_h, len(best_h), beta, len(best_h) / 2.0,
                              maxiter=500)
        s = exact_score(h_new)
        if s < best_c5:
            best_c5 = s
            best_h = h_new

    # Phase 7: One more sharpen + optimize cycle if time allows
    if time_left() > 80:
        for sharp_t in [5.0, 8.0, 12.0]:
            if time_left() < 50:
                break
            h_sharp = sharpen(best_h, len(best_h) / 2.0, sharp_t)
            h_new = optimize_slqp(h_sharp, len(best_h), 5000, len(best_h) / 2.0,
                                  maxiter=400)
            s = exact_score(h_new)
            if s < best_c5:
                best_c5 = s
                best_h = h_new

    # Phase 8: Projected gradient final refinement
    if time_left() > 40:
        h_pg, s_pg = projected_gradient(best_h, len(best_h), len(best_h) / 2.0,
                                        n_iter=300, beta_init=5000, beta_end=20000)
        if s_pg < best_c5:
            best_c5 = s_pg
            best_h = h_pg

    # Final validation
    best_h = project_box_sum(best_h, len(best_h) / 2.0)
    final_score = exact_score(best_h)
    if final_score < best_c5:
        best_c5 = final_score

    return best_h, best_c5, len(best_h)
