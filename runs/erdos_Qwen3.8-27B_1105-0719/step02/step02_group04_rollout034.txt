For the 1D minimax subproblem:
Given linear functions f_k(x) = c_k + g_k * x for k in active set, find x ∈ [lo, hi] that minimizes max_k f_k(x).

The solution is:
- If all g_k = 0, the solution doesn't matter (x doesn't affect the objective)
- Otherwise, the optimal x is where the two "most constraining" lines cross, clipped to [lo, hi]
- Specifically: the upper envelope of the lines at x is max_k (c_k + g_k * x)
- We want to minimize this upper envelope over x ∈ [lo, hi]
- The minimum of the upper envelope occurs at:
  - A vertex of the upper envelope (intersection of two lines), or
  - A boundary point lo or hi

The upper envelope changes at intersection points. For n lines, there are O(n²) intersection points. We can find the minimum by:
1. For each pair of lines, compute their intersection x*
2. Check if x* is in [lo, hi]
3. At x*, compute the value of all lines and take the max
4. Also check lo and hi
5. Return the x that gives the minimum max value

For |active| ≈ 20-30, this is O(|active|²) per coordinate, which is very fast.

Actually, there's a simpler approach: the optimal x* for minimizing max_k(c_k + g_k*x) over x ∈ [lo, hi] can be found by:
- Among lines with positive slope, find the one that's "highest" (largest intercept/slope ratio)
- Among lines with negative slope, find the one that's "highest" (largest intercept/slope ratio)
- The crossing point of these two gives the candidate
- Clip to [lo, hi]

More precisely:
- Let P = {k : g_k > 0} and N = {k : g_k < 0}
- The upper envelope for x → ∞ is determined by the line in P with the largest g_k
- The upper envelope for x → -∞ is determined by the line in N with the most negative g_k
- The minimum of the upper envelope is at the intersection of:
  - The line in P that dominates the upper envelope from the right
  - The line in N that dominates the upper envelope from the left

    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def project_to_target(x, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        if np.clip(x - mid, 0.0, 1.0).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)

        if mask.any():

    h = np.empty(n)
    h[:m] = q
    h[m:] = 1.0 - q[::-1]
    return h

    return 0.5 * (h[:m] + 1.0 - h[n-1:m-1:-1])

    """Return scaled correlation values for all lags."""
    n = len(h)
    dx = 2.0 / n
    return corr_full(h) * dx

def select_active(corrs, margin=0.01, pad=2, n=None):
    """Select active lags (near maximum)."""
    mx = corrs.max()
    thresh = mx - max(1e-8, margin * mx)
    idx = np.where(corrs >= thresh)[0]
    if n is not None:
        max_idx = 2 * n - 2
    else:
        max_idx = len(corrs) - 1
    for a in list(idx):
        for delta in range(-pad, pad + 1):
            k = a + delta
            if 0 <= k <= max_idx:
                if k not in idx:
                    idx = np.append(idx, k)
    return np.sort(np.unique(idx))

def coord_desc_sym(q, n, active_k, max_sweeps=100, margin=0.02):
    """Coordinate descent on symmetric variable q."""
    m = n // 2
    dx = 2.0 / n
    h = build_sym(q, n)
    corrs = corr_full(h) * dx  # scaled
    best_c5 = corrs.max()
    
    for sweep in range(max_sweeps):
        improved = False
        for j in range(m):
            # Compute coefficient g(d) for each active lag
            # h[j] changes by delta, h[n-1-j] changes by -delta
            # C(d) changes by g(d) * delta where:
            # g(d) = (1-h[j+d])*valid(j+d) - h[j-d]*valid(j-d)
            #        - (1-h[n-1-j+d])*valid(n-1-j+d) + h[n-1-j-d]*valid(n-1-j-d)
            
            a_vals = np.zeros(len(active_k))
            for idx, k in enumerate(active_k):
                d = k - (n - 1)
                g = 0.0
                # term from h[j]
                jd = j + d
                if 0 <= jd < n:
                    g += (1.0 - h[jd])
                jd2 = j - d
                if 0 <= jd2 < n:
                    g -= h[jd2]
                # term from h[n-1-j] = 1-q[j]
                j2 = n - 1 - j
                jd3 = j2 + d
                if 0 <= jd3 < n:
                    g -= (1.0 - h[jd3])
                jd4 = j2 - d
                if 0 <= jd4 < n:
                    g += h[jd4]
                a_vals[idx] = g * dx
            
            c_vals = corrs[active_k]
            lo = -q[j]
            hi = 1.0 - q[j]
            new_delta = solve_1d_minimax(c_vals, a_vals, lo, hi)

                new_delta = solve_1d_minimax(c_vals, a_vals, lo, hi)
                if abs(new_delta) > 1e-14:
                    q[j] = np.clip(q[j] + new_delta, 0.0, 1.0)
                    h[j] = q[j]
                    h[j2] = 1.0 - q[j]
            corrs = corr_full(h) * dx
            new_max = corrs.max()
            if new_max >= best_c5:
                break
        
        active_k = select_active(corrs, margin=margin, pad=3)
    
    return q, corrs.max()


def coord_desc_full(h, n, active_k, max_sweeps=30, margin=0.02, order=None):
    dx = 2.0 / n
    target = n / 2.0
    corrs = corr_full(h) * dx
    best_c5 = corrs.max()
    
    for sweep in range(max_sweeps):
        improved = False
        indices = list(order) if order is not None else list(range(n))
        for i in indices:
            a_vals = np.zeros(len(active_k))
            for idx in range(len(active_k)):
                d = active_k[idx] - (n - 1)
                g = 0.0
                jd = i + d
                if 0 <= jd < n:
                    g += (1.0 - h[jd])
                jd2 = i - d
                if 0 <= jd2 < n:
                    g -= h[jd2]
                a_vals[idx] = g * dx
            
            c_vals = corrs[active_k]
            lo = -h[i]
            hi = 1.0 - h[i]
            if lo >= hi - 1e-15:
                continue
            new_delta = solve_1d_minimax(c_vals, a_vals, lo, hi)
            
            if abs(new_delta) > 1e-14:
                h[i] = np.clip(h[i] + new_delta, 0.0, 1.0)
                # Project to sum constraint
                s = h.sum() - target
                if abs(s) > 1e-10:
                    mask = (h > 1e-10) & (h < 1.0 - 1e-10)
                    if mask.sum() > 0:
                        h[mask] -= s / mask.sum()
                        h = np.clip(h, 0.0, 1.0)
                corrs = corr_full(h) * dx
                if corrs.max() < best_c5 - 1e-14:
                    best_c5 = corrs.max()
                    improved = True
        
        if not improved:
            break
        active_k = select_active(corrs, margin=margin, pad=3)
    
    return h, corrs.max()


def interp_to(h, n_new):
    n_old = len(h)
    if n_new == n_old:
        return h.copy()
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h)
    return project_to_target(h_new, n_new / 2.0)


def lbbfgs_sym(q, n, active_k, mu=1000.0, maxiter=100):
    m = n // 2
    dx = 2.0 / n
    h = build_sym(q, n)
    t0_c5 = eval_c5(h)
    
    def fg(z):
        qz = z[:m]
        t = z[m]
        hz = build_sym(qz, n)
        cv = corr_full(hz) * dx
        vals = cv[active_k]
        exc = np.maximum(0.0, vals - t)
        f = t + 0.5 * mu * np.dot(exc, exc)
        gf = np.zeros(m + 1)
        gf[m] = 1.0 - mu * exc.sum()
        gq = np.zeros(m)
        for i in range(len(active_k)):
            if exc[i] > 0.0:
                d = active_k[i] - (n - 1)
                # Gradient of C(d) wrt q[j]
                # Same as g(d) in coord desc but for all j
                for j in range(m):
                    j2 = n - 1 - j
                    g = 0.0
                    jd = j + d
                    if 0 <= jd < n:
                        g += (1.0 - hz[jd])
                    jd2 = j - d
                    if 0 <= jd2 < n:
                        g -= hz[jd2]
                    jd3 = j2 + d
                    if 0 <= jd3 < n:
                        g -= (1.0 - hz[jd3])
                    jd4 = j2 - d
                    if 0 <= jd4 < n:
                        g += hz[jd4]
                    gq[j] += mu * exc[i] * g * dx
        gf[:m] = gq
        return f, gf
    
    z0 = np.concatenate([q, [t0_c5]])
    bounds = [(0.0, 1.0)] * m + [(0.0, t0_c5 + 0.5)]
    try:
        res = minimize(fg, z0, jac=True, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-10})
        q_opt = np.clip(res.x[:m], 0.0, 1.0)
    except Exception:
        return q, t0_c5
    h_opt = build_sym(q_opt, n)
    return q_opt, eval_c5(h_opt)


def make_random_sym(n, rng, sharpness=0.3):
    m = n // 2
    t = np.linspace(0.0, 1.0, m)
    freq1 = rng.randint(1, 8)
    freq2 = rng.randint(1, 8)
    phase1 = rng.uniform(0, 2 * np.pi)
    phase2 = rng.uniform(0, 2 * np.pi)
    amp1 = rng.uniform(0.1, 0.4)
    amp2 = rng.uniform(0.05, 0.3)
    q = 0.5 + amp1 * np.sin(2 * np.pi * freq1 * t + phase1) + amp2 * np.sin(2 * np.pi * freq2 * t + phase2)
    return np.clip(q, 0.0, 1.0)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    reserve = max(30.0, 0.05 * budget_s)
    
    def time_left():
        return budget_s - (time.time() - t0)
    
    # Initialize
    try:
        h_init = np.array(initial_h_values, dtype=float).reshape(-1)
        if h_init.size > 0 and np.all(np.isfinite(h_init)):
            n_init = len(h_init)
            h_init = project_to_target(h_init, n_init / 2.0)
            best_h = h_init.copy()
            best_c5 = eval_c5(h_init)
    except Exception:
        pass
    
    if best_h is None:
        n_init = 126
        q_init = np.full(n_init // 2, 0.5)
        best_h = build_sym(q_init, n_init)
        best_c5 = eval_c5(best_h)
    
    # Phase 1: Symmetric coordinate descent (main workhorse)
    n = len(best_h)
    if n % 2 == 0:
        q = to_sym_q(best_h, n)
        h_sym = build_sym(q, n)
        c5_sym = eval_c5(h_sym)
        if c5_sym < best_c5:
            best_h = h_sym.copy()
            best_c5 = c5_sym
        else:
            q = to_sym_q(best_h, n)
        
        corrs = get_corrs(best_h)
        active = select_active(corrs, margin=0.03, pad=4)
        
        # Multiple rounds of coordinate descent with tightening
        for round_num in range(15):
            if time_left() < reserve + 20:
                break
            margin = max(0.005, 0.03 * (0.8 ** round_num))
            pad = 3 if round_num < 10 else 5
            active = select_active(get_corrs(best_h), margin=margin, pad=pad)
            order = np.arange(len(q))
            rng.shuffle(order)
            q, c5_val = coord_desc_sym_fast(q, n, active, max_sweeps=30, margin=margin,
                                            order=order, t0=t0, budget_s=budget_s, reserve=reserve)
            h_new = build_sym(q, n)
            c5_new = eval_c5(h_new)
            if c5_new < best_c5 - 1e-14:
                best_h = h_new.copy()
                best_c5 = c5_new
    
    # Phase 2: L-BFGS polish on symmetric
    if time_left() > reserve + 30:
        n = len(best_h)
        if n % 2 == 0:
            q = to_sym_q(best_h, n)
            active = select_active(get_corrs(best_h), margin=0.01, pad=5)
            for mu in [100.0, 500.0, 2000.0, 10000.0]:
                if time_left() < reserve + 10:
                    break
                q, c5_val = lbbfgs_sym(q, n, active, mu=mu, maxiter=80)
                h_new = build_sym(q, n)
                c5_new = eval_c5(h_new)
                if c5_new < best_c5 - 1e-14:
                    best_h = h_new.copy()
                    best_c5 = c5_new
                    active = select_active(get_corrs(best_h), margin=0.01, pad=5)
    
    # Phase 3: Full (non-symmetric) coordinate descent
    if time_left() > reserve + 30:
        n = len(best_h)
        h_full = best_h.copy()
        active_f = select_active(get_corrs(h_full), margin=0.03, pad=4)
        for round_num in range(5):
            if time_left() < reserve + 10:
                break
            order = np.arange(n)
            rng.shuffle(order)
            h_full, c5_f = coord_desc_full(h_full, n, active_f, max_sweeps=20, margin=0.02, order=order)
            c5_f = eval_c5(h_full)
            if c5_f < best_c5 - 1e-14:
                best_h = h_full.copy()
                best_c5 = c5_f
                active_f = select_active(get_corrs(best_h), margin=0.02, pad=4)
    
    # Phase 4: Try different n values
    if time_left() > reserve + 40:
        n_base = len(best_h)
        n_candidates = []
        for delta in [-4, -3, -2, -1, 1, 2, 3, 4, 6, 8, 12]:
            nc = n_base + delta
            if nc >= 40 and nc % 2 == 0:
                n_candidates.append(nc)
        n_candidates = sorted(set(n_candidates))
        
        for n_try in n_candidates:
            if time_left() < reserve + 15:
                break
            if n_try == n_base:
                continue
            h_try = interp_to(best_h, n_try)
            q_try = to_sym_q(h_try, n_try)
            h_try = build_sym(q_try, n_try)
            active_try = select_active(get_corrs(h_try), margin=0.03, pad=4)
            
            # Coordinate descent
            for round_num in range(5):
                if time_left() < reserve + 5:
                    break
                order = np.arange(len(q_try))
                rng.shuffle(order)
                q_try, c5_try = coord_desc_sym_fast(q_try, n_try, active_try, max_sweeps=20,
                                                    margin=0.02, order=order,
                                                    t0=t0, budget_s=budget_s, reserve=reserve)
                h_try = build_sym(q_try, n_try)
                c5_try = eval_c5(h_try)
                if c5_try < best_c5 - 1e-14:
                    best_h = h_try.copy()
                    best_c5 = c5_try
                    active_try = select_active(get_corrs(best_h), margin=0.02, pad=4)
                
                # L-BFGS polish
                if time_left() > reserve + 15:
                    active_try = select_active(get_corrs(h_try), margin=0.01, pad=5)
                    q_try, c5_try2 = lbbfgs_sym(q_try, n_try, active_try, mu=5000.0, maxiter=60)
                    h_try2 = build_sym(q_try, n_try)
                    c5_try2 = eval_c5(h_try2)
                    if c5_try2 < best_c5 - 1e-14:
                        best_h = h_try2.copy()
                        best_c5 = c5_try2
    
    # Phase 5: Random restarts
    if time_left() > reserve + 30:
        n_r = len(best_h)
        if n_r % 2 != 0:
            n_r -= 1
        max_restarts = min(30, max(3, int(time_left() / 20)))
        for trial in range(max_restarts):
            if time_left() < reserve + 10:
                break
            # Start near best with noise
            q_base = to_sym_q(best_h, n_r) if len(best_h) == n_r else to_sym_q(interp_to(best_h, n_r), n_r)
            noise_scale = 0.05 + 0.1 * (trial / max(1, max_restarts))
            q_r = np.clip(q_base + noise_scale * rng.randn(n_r // 2), 0.0, 1.0)
            h_r = build_sym(q_r, n_r)
            active_r = select_active(get_corrs(h_r), margin=0.03, pad=4)
            order = np.arange(n_r // 2)
            rng.shuffle(order)
            q_r, c5_r = coord_desc_sym_fast(q_r, n_r, active_r, max_sweeps=20, margin=0.02,
                                            order=order, t0=t0, budget_s=budget_s, reserve=reserve)
            h_r = build_sym(q_r, n_r)
            c5_r = eval_c5(h_r)
            if c5_r < best_c5 - 1e-14:
                best_h = h_r.copy()
                best_c5 = c5_r
                # Follow-up polish
                if time_left() > reserve + 15:
                    active_r2 = select_active(get_corrs(best_h), margin=0.01, pad=5)
                    q_r2 = to_sym_q(best_h, n_r)
                    q_r2, c5_r2 = lbbfgs_sym(q_r2, n_r, active_r2, mu=5000.0, maxiter=60)
                    h_r2 = build_sym(q_r2, n_r)
                    if eval_c5(h_r2) < best_c5:
                        best_h = h_r2.copy()
                        best_c5 = eval_c5(h_r2)
    
    # Phase 6: Final intensive polish
    if time_left() > 15 and best_h is not None:
        n_fin = len(best_h)
        if n_fin % 2 == 0:
            q_fin = to_sym_q(best_h, n_fin)
            # Tight coordinate descent
            active_fin = select_active(get_corrs(best_h), margin=0.005, pad=4)
            order = np.arange(n_fin // 2)
            q_fin, c5_fin = coord_desc_sym_fast(q_fin, n_fin, active_fin, max_sweeps=30,
                                                margin=0.005, order=order,
                                                t0=t0, budget_s=budget_s, reserve=5)
            h_fin = build_sym(q_fin, n_fin)
            c5_fin = eval_c5(h_fin)
            if c5_fin < best_c5:
                best_h = h_fin.copy()
                best_c5 = c5_fin
        
        if time_left() > 8:
            # One more L-BFGS with very high mu
            if len(best_h) % 2 == 0:
                q_fin2 = to_sym_q(best_h, len(best_h))
                active_fin2 = select_active(get_corrs(best_h), margin=0.003, pad=5)
                q_fin2, c5_fin2 = lbbfgs_sym(q_fin2, len(best_h), active_fin2, mu=50000.0, maxiter=50)
                h_fin2 = build_sym(q_fin2, len(best_h))
                c5_fin2 = eval_c5(h_fin2)
                if c5_fin2 < best_c5:
                    best_h = h_fin2.copy()
                    best_c5 = c5_fin2
    
    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_target(best_h, len(best_h) / 2.0)
    best_c5 = eval_c5(best_h)
    
    return best_h, best_c5, len(best_h)
