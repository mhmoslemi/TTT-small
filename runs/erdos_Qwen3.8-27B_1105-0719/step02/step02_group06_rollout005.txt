import numpy as np
from scipy.optimize import minimize, linprog
import time

def eval_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def corr_raw(h):
    return np.correlate(h, 1.0 - h, mode='full')

def build_sym_h(q):
    m = len(q)
    h = np.empty(2 * m)
    h[:m] = q
    h[m:] = 1.0 - q[::-1]
    return h

def sym_from_h(h):
    n = len(h)
    m = n // 2
    q = 0.5 * (h[:m] + (1.0 - h[n-1:m-1:-1]))
    return np.clip(q, 0.0, 1.0)

def project_feasible(x, target_sum):
    n = len(x)
    x = np.asarray(x, dtype=float)
    x = np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0)
    x = np.clip(x, 0.0, 1.0)
    if abs(x.sum() - target_sum) < 1e-14:
        return x
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(x + mid, 0.0, 1.0).sum()
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    h = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            order = np.argsort(h)
            for i in order:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
                if abs(diff) < 1e-14:
                    break
        else:
            order = np.argsort(-h)
            for i in order:
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub
                if abs(diff) < 1e-14:
                    break
    return np.clip(h, 0.0, 1.0)

def corr_grad_k(h, k):
    n = len(h)
    d = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -d)
    i_hi = min(n - 1, n - 1 - d)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx] += (1.0 - h[idx + d])
    grad[idx + d] -= h[idx]
    return grad

def corr_val_k(h, k):
    n = len(h)
    d = k - (n - 1)
    i_lo = max(0, -d)
    i_hi = min(n - 1, n - 1 - d)
    if i_lo > i_hi:
        return 0.0
    idx = np.arange(i_lo, i_hi + 1)
    return float(np.sum(h[idx] * (1.0 - h[idx + d])))

    return active_list

def transfer_to_n(h_old, n_new):

    x_new = (np.arange(n_new) + 0.5) * 2.0 / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new / 2.0)

def solve_tangent_lp(h0, active_k, rho):
    """Solve LP: min t s.t. grad_ell^T h - t <= grad_ell^T h0 - R_ell(h0)"""
    n = len(h0)
    # Variables: [h (n vars), t (1 var)] total n+1
    # c = [0, ..., 0, 1]
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    A_ub = []
    b_ub = []
    for k in active_k:
        R_ell = corr_val_k(h0, k)
        g = corr_grad_k(h0, k)
        # g^T h - t <= g^T h0 - R_ell
        row = np.zeros(n + 1)
        row[:n] = g
        row[n] = -1.0
        A_ub.append(row)
        b_ub.append(float(g @ h0) - R_ell)
    
    if A_ub:
        A_ub = np.array(A_ub)
        b_ub = np.array(b_ub)
    else:
        A_ub = np.zeros((1, n+1))
        b_ub = np.zeros(1)
    
    # Bounds: h[i] in [max(0, h0[i]-rho), min(1, h0[i]+rho)], t in [0, 1]
    bounds = []
    for i in range(n):
        lo_i = max(0.0, h0[i] - rho)
        hi_i = min(1.0, h0[i] + rho)
        bounds.append((lo_i, hi_i))
    bounds.append((0.0, 1.0))
    
    # Equality: sum(h) = n/2
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
        if res.success:
            h_new = np.clip(res.x[:n], 0.0, 1.0)
            h_new = project_feasible(h_new, n / 2.0)
            return h_new
    except Exception:
        pass
    return h0.copy()

def transfer_line_search(h, a, b, n_grid=40, n_refine=20):
    """Exact 1D line search along h[a]+=delta, h[b]-=delta"""
    x0_a = h[a]
    x0_b = h[b]
    lo = max(-x0_a, x0_b - 1.0)
    hi = min(1.0 - x0_a, -x0_b)
    if hi - lo < 1e-12:
        return h.copy()
    
    best_delta = 0.0
    best_score = eval_score(h)
    
    # Grid search
    deltas = np.linspace(lo, hi, n_grid)
    scores = np.zeros(n_grid)
    h_tmp = h.copy()
    for i, dl in enumerate(deltas):
        h_tmp[a] = x0_a + dl
        h_tmp[b] = x0_b - dl
        scores[i] = eval_score(h_tmp)
    
    idx_best = np.argmin(scores)
    if scores[idx_best] < best_score:
        best_delta = deltas[idx_best]
        best_score = scores[idx_best]
    
    # Local refinement around best
    if idx_best > 0 and idx_best < n_grid - 1:
        local_lo = deltas[max(0, idx_best - 2)]
        local_hi = deltas[min(n_grid - 1, idx_best + 2)]
    else:
        local_lo = deltas[max(0, idx_best - 1)]
        local_hi = deltas[min(n_grid - 1, idx_best + 1)]
    
    local_deltas = np.linspace(local_lo, local_hi, n_refine)
    for dl in local_deltas:
        dl = max(lo, min(hi, dl))
        h_tmp[a] = x0_a + dl
        h_tmp[b] = x0_b - dl
        sc = eval_score(h_tmp)
        if sc < best_score:
            best_score = sc
            best_delta = dl
    
    h[a] = x0_a + best_delta
    h[b] = x0_b - best_delta
    h = np.clip(h, 0.0, 1.0)
    return h

We want: minimize t
subject to: R_ell(h) <= t for all ell in active set (approximated by linearization)

R_ell(h) ≈ R_ell(h0) + g_ell^T (h - h0)

So: R_ell(h0) + g_ell^T (h - h0) <= t
=> g_ell^T h - t <= g_ell^T h0 - R_ell(h0)

The right side: g_ell^T h0 - R_ell(h0)

Let me verify: at h = h0, LHS = g_ell^T h0 - t, RHS = g_ell^T h0 - R_ell(h0)
So: g_ell^T h0 - t <= g_ell^T h0 - R_ell(h0) => -t <= -R_ell(h0) => t >= R_ell(h0) ✓

Good. So the LP is:
- c = [0,...,0, 1] (minimize t)
- A_ub row for k: [g_k, -1], b_ub = g_k^T h0 - R_k(h0)
- A_eq: [1,...,1, 0], b_eq = n/2
- bounds: [max(0, h0[i]-rho), min(1, h0[i]+rho)] for h[i], [0, 1] for t

import numpy as np
from scipy.optimize import linprog
import time

def eval_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def corr_raw(h):
    return np.correlate(h, 1.0 - h, mode='full')

def build_sym_h(q):
    m = len(q)
    h = np.empty(2 * m)
    h[:m] = q
    h[m:] = 1.0 - q[::-1]
    return h

def project_feasible(x, target_sum):
    n = len(x)
    x = np.asarray(x, dtype=float)
    x = np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0)
    x = np.clip(x, 0.0, 1.0)
    if abs(x.sum() - target_sum) < 1e-14:
        return x.copy()
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(x + mid, 0.0, 1.0).sum()
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    h = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            order = np.argsort(h)
            for i in order:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
                if abs(diff) < 1e-14:
                    break
        else:
            order = np.argsort(-h)
            for i in order:
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub
                if abs(diff) < 1e-14:
                    break
    return np.clip(h, 0.0, 1.0)

def corr_val_k(h, k):
    n = len(h)
    d = k - (n - 1)
    i_lo = max(0, -d)
    i_hi = min(n - 1, n - 1 - d)
    if i_lo > i_hi:
        return 0.0
    idx = np.arange(i_lo, i_hi + 1)
    return float(np.sum(h[idx] * (1.0 - h[idx + d])))

def corr_grad_k(h, k):
    n = len(h)
    d = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -d)
    i_hi = min(n - 1, n - 1 - d)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx] += (1.0 - h[idx + d])
    grad[idx + d] -= h[idx]
    return grad

def select_active(corrs, margin_frac=0.03, neighbor_pad=3, max_k=60):
    max_val = corrs.max()
    threshold = max_val - max(2e-6, margin_frac * max_val)
    active = set(np.where(corrs >= threshold)[0].tolist())
    padded = set()
    for k in active:
        for d in range(-neighbor_pad, neighbor_pad + 1):
            if 0 <= k + d < len(corrs):
                padded.add(k + d)
    active_list = sorted(padded)
    if len(active_list) > max_k:
        scored = sorted([(corrs[k], k) for k in active_list], reverse=True)
        active_list = sorted([k for _, k in scored[:max_k]])
    return active_list

def solve_tangent_lp(h0, active_k, rho):
    n = len(h0)
    c = np.zeros(n + 1)
    c[n] = 1.0
    rows = []
    rhs = []
    for k in active_k:
        R_ell = corr_val_k(h0, k)
        g = corr_grad_k(h0, k)
        row = np.zeros(n + 1)
        row[:n] = g
        row[n] = -1.0
        rows.append(row)
        rhs.append(float(g @ h0) - R_ell)
    if rows:
        A_ub = np.array(rows)
        b_ub = np.array(rhs)
    else:
        A_ub = np.array([[0.0] * (n + 1)])
        b_ub = np.array([0.0])
    bounds = []
    for i in range(n):
        lo_i = max(0.0, h0[i] - rho)
        hi_i = min(1.0, h0[i] + rho)
        bounds.append((lo_i, hi_i))
    bounds.append((0.0, 1.5))
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
        if res.success:
            h_new = np.clip(res.x[:n], 0.0, 1.0)
            h_new = project_feasible(h_new, n / 2.0)
            return h_new
    except Exception:
        pass
    return h0.copy()

def transfer_line_search(h, a, b, n_grid=35, n_refine=15):
    x0_a = h[a]
    x0_b = h[b]
    lo = max(-x0_a, x0_b - 1.0)
    hi = min(1.0 - x0_a, -x0_b)
    if hi - lo < 1e-12:
        return h.copy()
    best_delta = 0.0
    best_score = eval_score(h)
    h_tmp = h.copy()
    deltas = np.linspace(lo, hi, n_grid)
    scores = np.empty(n_grid)
    for i, dl in enumerate(deltas):
        h_tmp[a] = x0_a + dl
        h_tmp[b] = x0_b - dl
        scores[i] = eval_score(h_tmp)
    idx_best = np.argmin(scores)
    if scores[idx_best] < best_score:
        best_delta = deltas[idx_best]
        best_score = scores[idx_best]
    if 0 < idx_best < n_grid - 1:
        local_lo = deltas[max(0, idx_best - 1)]
        local_hi = deltas[min(n_grid - 1, idx_best + 1)]
    else:
        local_lo = lo
        local_hi = hi
    for dl in np.linspace(local_lo, local_hi, n_refine):
        dl = max(lo, min(hi, dl))
        h_tmp[a] = x0_a + dl
        h_tmp[b] = x0_b - dl
        sc = eval_score(h_tmp)
        if sc < best_score:
            best_score = sc
            best_delta = dl
    h[a] = x0_a + best_delta
    h[b] = x0_b - best_delta
    return np.clip(h, 0.0, 1.0)

def do_lp_refinement(h, max_rounds=30, rho_start=0.15, rho_decay=0.85, time_limit=None):
    best_h = h.copy()
    best_score = eval_score(h)
    rho = rho_start
    for _ in range(max_rounds):
        if time_limit is not None and time.time() > time_limit:
            break
        dx = 2.0 / len(best_h)
        corrs = corr_raw(best_h) * dx
        active = select_active(corrs, margin_frac=0.02, neighbor_pad=3, max_k=50)
        h_new = solve_tangent_lp(best_h, active, rho)
        sc = eval_score(h_new)
        if sc < best_score - 1e-14:
            best_h = h_new
            best_score = sc
            rho = max(rho * 0.95, 0.01)
        else:
            rho *= rho_decay
            if rho < 0.005:
                break
    return best_h, best_score

def do_transfer_refinement(h, max_rounds=40, n_pairs=8, time_limit=None, seed=0):
    rng = np.random.RandomState(seed)
    n = len(h)
    best_h = h.copy()
    best_score = eval_score(h)
    for rnd in range(max_rounds):
        if time_limit is not None and time.time() > time_limit:
            break
        improved = False
        if rnd % 4 == 0:
            corrs = corr_raw(best_h) * (2.0 / n)
            max_lags = np.where(corrs >= corrs.max() - 0.02 * corrs.max())[0]
            for k in max_lags[:5]:
                d = k - (n - 1)
                i_lo = max(0, -d)
                i_hi = min(n - 1, n - 1 - d)
                if i_hi - i_lo >= 2:
                    for _ in range(n_pairs // 2):
                        a = rng.randint(i_lo, i_hi + 1)
                        b = rng.randint(i_lo, i_hi + 1)
                        if a != b:
                            h_tmp = best_h.copy()
                            h_tmp = transfer_line_search(h_tmp, a, b)
                            sc = eval_score(h_tmp)
                            if sc < best_score - 1e-14:
                                best_h = h_tmp
                                best_score = sc
                                improved = True
        for _ in range(n_pairs):
            a = rng.randint(n)
            b = rng.randint(n)
            if a != b:
                h_tmp = best_h.copy()
                h_tmp = transfer_line_search(h_tmp, a, b)
                sc = eval_score(h_tmp)
                if sc < best_score - 1e-14:
                    best_h = h_tmp
                    best_score = sc
                    improved = True
        if not improved and rnd > 15:
            break
    return best_h, best_score

def combine_lp_transfer(h, time_limit, seed=0):
    best_h = h.copy()
    best_score = eval_score(h)
    steps = min(10, max(2, int((time_limit - time.time()) / 8)))
    for i in range(steps):
        if time.time() > time_limit:
            break
        h_new, sc = do_lp_refinement(best_h, max_rounds=15, rho_start=0.12, time_limit=time.time() + min(25, time_limit - time.time() - 1))
        if sc < best_score - 1e-14:
            best_h = h_new
            best_score = sc
        if time.time() > time_limit:
            break
        h_new, sc = do_transfer_refinement(best_h, max_rounds=20, n_pairs=6, time_limit=time.time() + min(25, time_limit - time.time() - 1), seed=seed + i * 7)
        if sc < best_score - 1e-14:
            best_h = h_new
            best_score = sc
    return best_h, best_score

def transfer_to_n(h_old, n_new):
    n_old = len(h_old)
    x_old = (np.arange(n_old) + 0.5) * 2.0 / n_old
    x_new = (np.arange(n_new) + 0.5) * 2.0 / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new / 2.0)

def make_binary_start(n, rng):
    h = np.zeros(n)
    n_ones = n // 2
    idx = rng.permutation(n)
    h[idx[:n_ones]] = 1.0
    return h

def hill_climb_binary(h, max_swaps=150, rng=None):
    if rng is None:
        rng = np.random.RandomState(42)
    best_h = h.copy()
    best_score = eval_score(h)
    ones = np.where(best_h == 1.0)[0]
    zeros = np.where(best_h == 0.0)[0]
    for _ in range(max_swaps):
        if len(ones) < 1 or len(zeros) < 1:
            break
        a = ones[rng.randint(len(ones))]
        b = zeros[rng.randint(len(zeros))]
        test_h = best_h.copy()
        test_h[a] = 0.0
        test_h[b] = 1.0
        sc = eval_score(test_h)
        if sc < best_score - 1e-14:
            best_h = test_h
            best_score = sc
            ones = np.where(best_h == 1.0)[0]
            zeros = np.where(best_h == 0.0)[0]
    return best_h, best_score
