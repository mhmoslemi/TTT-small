from scipy.optimize import minimize
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx] + 1e-15)
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx] + 1e-15)
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmin(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def build_gradient_matrix(h, n, dx):
    M = 2 * n - 1
    G = np.zeros((M, n))
    for m in range(M):
        d = m - (n - 1)
        if d == 0:
            G[m, :] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            # P_d = sum_{i=d}^{n-1} h[i] * (1-h[i-d])
            # dP_d/dh[i] for i >= d: dx * (1 - h[i-d])
            # dP_d/dh[i] for i < n-d: -dx * h[i+d]
            G[m, d:] += dx * (1.0 - h[0:n-d])
            G[m, 0:n-d] += -dx * h[d:n]
        else:
            dp = -d
            # P_{-d} = sum_{i=0}^{n-d-1} h[i] * (1-h[i+d])
            # dP/dh[i] for i < n-d: dx * (1-h[i+d])
            # dP/dh[i] for i >= d: -dx * h[i-d]
            G[m, 0:n-dp] += dx * (1.0 - h[dp:n])
            G[m, dp:n] += -dx * h[0:n-dp]
    return G

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    # Mass-preserving bin average interpolation
    xs = np.linspace(0, 2, n + 1)
    xt = np.linspace(0, 2, new_n + 1)
    h_interp = np.zeros(new_n)
    for k in range(new_n):
        # Find overlap of [xt[k], xt[k+1]] with each bin [xs[j], xs[j+1]]
        lo_k, hi_k = xt[k], xt[k+1]
        total_mass = 0.0
        total_width = 0.0
        for j in range(n):
            lo_j, hi_j = xs[j], xs[j+1]
            overlap = min(hi_k, hi_j) - max(lo_k, lo_j)
            if overlap > 0:
                total_mass += h[j] * overlap
                total_width += overlap
        h_interp[k] = total_mass / total_width if total_width > 0 else 0.5
    h_interp = np.clip(h_interp, 0.0, 1.0)
    return repair_feasible(h_interp)

def binary_energy(bits):
    """Compute the binary overlap energy for a balanced binary pattern."""
    n = len(bits)
    dx = 2.0 / n
    a = bits.astype(np.float64)
    profile = np.correlate(a, 1.0 - a, mode="full") * dx
    # Use a smoothed maximum to make it differentiable-ish for SA
    c = np.max(profile)
    # Add a penalty for the second and third highest to flatten the profile
    sorted_p = np.sort(profile)[::-1]
    k = min(5, len(sorted_p))
    penalty = np.sum(sorted_p[:k] - c)  # These are 0 or negative
    return c + 0.1 * np.sum(np.maximum(0, sorted_p[:k] - c + 0.01))

def sa_binary_pattern(bits, max_iters=500, T_init=0.5, T_final=0.01):
    """Simulated annealing on binary pattern space."""
    n = len(bits)
    bits = bits.copy()
    current_e = binary_energy(bits)
    best_bits = bits.copy()
    best_e = current_e
    ones_idx = np.where(bits == 1)[0].tolist()
    
    T = T_init
    cooling = (T_final / T_init) ** (1.0 / max(max_iters, 1))
    
    for it in range(max_iters):
        T *= cooling
        # Random exchange move: swap a 1 with a 0
        i = np.random.randint(n)
        j = np.random.randint(n)
        if bits[i] == bits[j]:
            continue
        bits[i], bits[j] = bits[j], bits[i]
        new_e = binary_energy(bits)
        delta = new_e - current_e
        if delta < 0 or np.random.rand() < np.exp(-delta / max(T, 1e-10)):
            current_e = new_e
            if new_e < best_e:
                best_e = new_e
                best_bits = bits.copy()
        else:
            bits[i], bits[j] = bits[j], bits[i]
    
    return best_bits, best_e

    """Extract binary pattern from continuous h."""
    if len(h) != n:
        h = interpolate_to_n(h, n)
    # Select top n/2 positions
    half = n // 2
    idx = np.argsort(h)[::-1][:half]
    bits = np.zeros(n, dtype=np.int64)
    bits[idx] = 1
    return bits

def pattern_to_continuous(bits, h_ref, q=0.0):

    h[bits == 1] = np.maximum(h[bits == 1], 0.3)
    h[bits == 0] = np.minimum(h[bits == 0], 0.7)

import numpy as np
from scipy.optimize import minimize
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx] + 1e-15)
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx] + 1e-15)
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-14:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmin(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def build_gradient_matrix(h, n, dx):
    M = 2 * n - 1
    G = np.zeros((M, n))
    for m in range(M):
        d = m - (n - 1)
        if d == 0:
            G[m, :] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            G[m, d:] += dx * (1.0 - h[0:n-d])
            G[m, 0:n-d] += -dx * h[d:n]
        else:
            dp = -d
            G[m, 0:n-dp] += dx * (1.0 - h[dp:n])
            G[m, dp:n] += -dx * h[0:n-dp]
    return G

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = np.linspace(0, 2, n + 1)
    xt = np.linspace(0, 2, new_n + 1)
    h_interp = np.zeros(new_n)
    for k in range(new_n):
        lo_k, hi_k = xt[k], xt[k+1]
        total_mass = 0.0
        total_width = 0.0
        for j in range(n):
            lo_j, hi_j = xs[j], xs[j+1]
            overlap = min(hi_k, hi_j) - max(lo_k, lo_j)
            if overlap > 0:
                total_mass += h[j] * overlap
                total_width += overlap
        h_interp[k] = total_mass / total_width if total_width > 1e-15 else 0.5
    h_interp = np.clip(h_interp, 0.0, 1.0)
    return repair_feasible(h_interp)

def symmetrize(h):

def antisymmetrize(h):

def slsqp_solve(h0, maxiter=200):
    n = len(h0)
    dx = 2.0 / n
    M = 2 * n - 1
    c0 = exact_score(h0)
    def obj(x):
        return x[-1]
    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    def eq_fun(x):
        return x[:-1].sum() - n / 2.0
    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g
    def ineq_fun(x):
        h = np.clip(x[:-1], 0, 1)
        t = x[-1]
        profile = np.correlate(h, 1 - h, mode="full") * dx
        return t - profile
    def ineq_jac(x):
        h = np.clip(x[:-1], 0, 1)
        G = build_gradient_matrix(h, n, dx)
        J = np.zeros((M, n + 1))
        J[:, :-1] = -G
        J[:, -1] = 1.0
        return J
    x0 = np.concatenate([h0, [c0 + 1e-3]])
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 0.5)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None and np.isfinite(res.x).all():
            h_new = repair_feasible(res.x[:-1])
            c_new = exact_score(h_new)
            if c_new <= c0 + 1e-9:
                return h_new, c_new
    except Exception:
        pass
    return None, None

def binary_energy(bits):
    n = len(bits)
    dx = 2.0 / n
    a = bits.astype(np.float64)
    profile = np.correlate(a, 1.0 - a, mode="full") * dx
    c = float(np.max(profile))
    sorted_p = np.sort(profile)[::-1]
    k = min(8, len(sorted_p))
    penalty = np.sum(np.maximum(0.0, sorted_p[:k] - c + 0.005))
    return c + 0.15 * penalty

def sa_binary_pattern(bits, max_iters=400, T_init=0.4, T_final=0.005, rng=None):
    n = len(bits)
    bits = bits.copy()
    current_e = binary_energy(bits)
    best_bits = bits.copy()
    best_e = current_e
    T = T_init
    cooling = (T_final / T_init) ** (1.0 / max(max_iters, 1))
    for _ in range(max_iters):
        T *= cooling
        i = rng.randint(n)
        j = rng.randint(n)
        if bits[i] == bits[j]:
            continue
        bits[i], bits[j] = bits[j], bits[i]
        new_e = binary_energy(bits)
        delta = new_e - current_e
        if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-10)):
            current_e = new_e
            if new_e < best_e:
                best_e = new_e
                best_bits = bits.copy()
        else:
            bits[i], bits[j] = bits[j], bits[i]
    return best_bits, best_e

            h_lse, c_lse = lse_descent(h_interp, beta=100, max_iters=60)
            if c_lse < best_c - 1e-13:
                best_h = h_lse.copy()
                best_c = c_lse
            h_ase, c_ase = active_set_descent(best_h, max_iters=60, n_active=10)
            if c_ase < best_c - 1e-13:
                best_h = h_ase.copy()
                best_c = c_ase

    # Stage 9: Final SLSQP polish
    if time_left() > 40:
        h_sol, c_sol = slsqp_solve(best_h, maxiter=300)
        if h_sol is not None and c_sol < best_c - 1e-14:
            best_h = h_sol.copy()
            best_c = c_sol

    # Stage 10: Final active-set + LSE
    if time_left() > 20:
        h_ase, c_ase = active_set_descent(best_h, max_iters=200, n_active=10, alpha_init=0.002)
        if c_ase < best_c - 1e-14:
            best_h = h_ase.copy()
            best_c = c_ase
        h_lse, c_lse = lse_descent(best_h, beta=150, max_iters=150, alpha_init=0.002)
        if c_lse < best_c - 1e-14:
            best_h = h_lse.copy()
            best_c = c_lse

    # Final validation
    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)

    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            final_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 2:
            final_c = float(eval_result[0])
    except Exception:
        pass

    return best_h, final_c, len(best_h)
