    """Take a step in the negative subgradient direction of the max overlap."""
    dx = 2.0 / n
    # Compute gradients for each active lag
    grads = [corr_grad_h(h, d, n) for d in active_lags]
    
    # Find lambda that minimizes the norm of the weighted gradient
    # (steepest descent in the subgradient polytope)
    G = np.array(grads)  # shape (K, n)
    
    # Solve: min ||G^T lambda||^2 s.t. lambda >= 0, sum(lambda) = 1
    # This is a QP with K variables
    # K is typically small (5-20)
    
    # Simple approach: uniform weights
    lam = np.ones(len(active_lags)) / len(active_lags)
    direction = -(G.T @ lam)
    
    # Project onto feasible set
    h_new = h + step_size * direction
    h_new = project_to_target(h_new, n/2.0)
    
    return h_new

```python
import cvxpy as cp

def lp_step(h, active_lags, n):
    dx = 2.0 / n
    m = len(h)
    
    # Current correlation values and gradients
    # For each active lag d (0-indexed in correlate output):
    # actual_lag = d - (n-1)
    # C[d] = sum_i h[i] * (1 - h[i + actual_lag]) (with bounds)
    # dC[d]/dh[j] = -sum over contributions
    
    # Build LP
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    
    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n/2.0]
    
    for d in active_lags:
        actual_lag = d - (n - 1)
        # Compute current C[d] and gradient
        C_val = corr_full(h)[d] * dx
        grad = corr_grad_h(h, actual_lag, n)
        # Linearized: C_val + dx * grad @ (h_var - h) <= t_var
        constraints.append(C_val + dx * (grad @ (h_var - h)) <= t_var)
    
    obj = cp.Minimize(t_var)
    prob = cp.Problem(obj, constraints)
    prob.solve(solver=cp.GLPK)  # or GLOP or CBC
    
    if h_var.value is not None:
        h_lp = np.clip(h_var.value, 0, 1)
        h_lp = project_to_target(h_lp, n/2.0)
        return h_lp
    return h
```

from scipy.optimize import minimize
import cvxpy as cp
import time

# ... helper functions ...

def lp_step_sym(h, active_lags, n, max_iter_lp=100):
    """LP step with symmetry constraint h[i] + h[n-1-i] = 1"""
    m = n // 2
    dx = 2.0 / n
    corr_vals = corr_full(h)
    
    # Variables: q[0..m-1] (first half), t
    # h[i] = q[i] for i < m
    # h[i] = 1-q[n-1-i] for i >= m
    
    q = cp.Variable(m)
    t = cp.Variable()
    
    constraints = [q >= 0, q <= 1, cp.sum(q) == m/2.0]
    
    for idx in active_lags:
        lag = idx - (n - 1)
        C_val = corr_vals[idx] * dx
        grad = corr_grad_h(h, lag, n)
        # grad w.r.t. q: for j < m, dh[j]/dq[j] = 1, so ∂C/∂q[j] = grad[j]
        # for j >= m, dh[j]/dq[n-1-j] = -1, so ∂C/∂q[n-1-j] = -grad[j]
        grad_q = np.empty(m)
        grad_q[:m] = grad[:m]
        grad_q[m:] = -grad[n-1:m-1:-1]  # grad[n-1], grad[n-2], ..., grad[m]
        # Linearized: C_val + dx * grad_q @ (q - q_current) <= t
        q_current = h[:m]
        constraints.append(C_val + dx * (grad_q @ (q - q_current)) <= t)
    
    obj = cp.Minimize(t)
    prob = cp.Problem(obj, constraints)

        h_try = (1.0 - alpha) * h + alpha * h_target
        h_try = project_to_target(h_try, n / 2.0)
        c5_try = eval_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try
    return best_h, best_c5


def lbfgs_step(h, active_idx, n, mu, maxiter=200):
    dx = 2.0 / n
    target = n / 2.0

    def fg(z):
        hz = z[:n]
        tv = z[n]
        corv = corr_full(hz) * dx
        vals = corv[active_idx]
        exc = np.maximum(0.0, vals - tv)
        sdev = hz.sum() - target
        f = tv + 0.5 * mu * np.dot(exc, exc) + 0.5 * mu * sdev * sdev
        gf = np.zeros(n + 1)
        gf[n] = 1.0 - mu * exc.sum()
        gh = np.zeros(n)
        for i in range(len(active_idx)):
            if exc[i] > 0.0:
                lag = active_idx[i] - (n - 1)
                gh += mu * exc[i] * corr_grad_h(hz, lag, n)
        gh += mu * sdev
        gf[:n] = gh
        return f, gf

    t0 = eval_c5(h)
    z0 = np.concatenate([h, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, t0 + 1.0)]
    try:
        res = minimize(fg, z0, jac=True, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-11})
        h_opt = project_to_target(np.clip(res.x[:n], 0.0, 1.0), target)
        c5_opt = eval_c5(h_opt)
    except Exception:
        return h.copy(), t0
    return h_opt, c5_opt


def interp_to(h, n_new):
    n_old = len(h)
    if n_new == n_old:
        return h.copy()
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h)
    return project_to_target(h_new, n_new / 2.0)


def remaining_time(t0, budget_s):
    return budget_s - (time.time() - t0)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')

    try:
        h_init = np.array(initial_h_values, dtype=float).reshape(-1)
        if h_init.size > 0 and np.all(np.isfinite(h_init)):
            n_init = len(h_init)
            if n_init % 2 != 0:
                n_init -= 1
                h_init = h_init[:n_init]
            h_init = project_to_target(h_init, n_init / 2.0)
            best_h = h_init.copy()
            best_c5 = eval_c5(h_init)
    except Exception:
        pass

    if best_h is None:
        n_def = 126
        m_def = n_def // 2
        tt = np.linspace(0.0, 1.0, m_def)
        q_def = np.clip(0.5 + 0.3 * np.sin(2.0 * np.pi * 2.0 * tt), 0.0, 1.0)
        best_h = build_sym(q_def, n_def)
        best_c5 = eval_c5(best_h)

    reserve = max(25.0, 0.04 * budget_s)
    solver = get_lp_solver()

    def try_update(h_cand):
        nonlocal best_h, best_c5
        sc = eval_c5(h_cand)
        if sc < best_c5:
            best_h = h_cand.copy()
            best_c5 = sc
            return True
        return False

    # Phase 1: LP-based local optimization (main driver)
    n = len(best_h)
    stall_count = 0
    max_stall = 8

    while remaining_time(t0, budget_s) > reserve + 20:
        corv = corr_full(best_h) * (2.0 / n)
        active = select_active(corv, margin=0.002, pad=2)
        improved = False

        # Try symmetric LP
        if n % 2 == 0:
            h_lp = lp_step_sym(best_h, active, n, solver)
            h_new, c5_new = line_search(best_h, h_lp, best_c5)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
                improved = True
                stall_count = 0

        # Try full LP
        if not improved and remaining_time(t0, budget_s) > reserve + 10:
            h_lp = lp_step_full(best_h, active, n, solver)
            h_new, c5_new = line_search(best_h, h_lp, best_c5)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
                improved = True
                stall_count = 0

        # Try wider active set LP
        if not improved:
            active_wide = select_active(corv, margin=0.008, pad=3)
            if len(active_wide) > len(active):
                h_lp = lp_step_full(best_h, active_wide, n, solver)
                h_new, c5_new = line_search(best_h, h_lp, best_c5)
                if c5_new < best_c5:
                    best_h = h_new
                    best_c5 = c5_new
                    improved = True
                    stall_count = 0

        # Try L-BFGS
        if not improved and remaining_time(t0, budget_s) > reserve + 15:
            active_2 = select_active(corr_full(best_h) * (2.0 / n), margin=0.003, pad=2)
            h_new, c5_new = lbfgs_step(best_h, active_2, n, mu=500.0, maxiter=200)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
                improved = True
                stall_count = 0
            else:
                # Try with higher mu
                h_new, c5_new = lbfgs_step(best_h, active_2, n, mu=5000.0, maxiter=150)
                if c5_new < best_c5:
                    best_h = h_new
                    best_c5 = c5_new
                    improved = True
                    stall_count = 0

        if not improved:
            stall_count += 1
            # Random perturbation to escape local optimum
            if stall_count >= max_stall:
                noise_scale = 0.005 + 0.005 * (stall_count - max_stall)
                h_pert = project_to_target(best_h + noise_scale * rng.randn(n), n / 2.0)
                # Run a few LP steps on perturbed point
                for _ in range(3):
                    if remaining_time(t0, budget_s) < reserve + 5:
                        break
                    corv_p = corr_full(h_pert) * (2.0 / n)
                    act_p = select_active(corv_p, margin=0.003, pad=2)
                    if n % 2 == 0:
                        h_lp = lp_step_sym(h_pert, act_p, n, solver)
                        h_pert, _ = line_search(h_pert, h_lp, eval_c5(h_pert))
                    h_lp = lp_step_full(h_pert, act_p, n, solver)
                    h_pert, _ = line_search(h_pert, h_lp, eval_c5(h_pert))
                if eval_c5(h_pert) < best_c5:
                    best_h = h_pert.copy()
                    best_c5 = eval_c5(h_pert)
                stall_count = 0

    # Phase 2: Try different n values
    n_base = len(best_h)
    n_candidates = sorted(set([n_base - 4, n_base - 2, n_base + 2, n_base + 4,
                                n_base + 6, n_base + 8, n_base + 10]))
    n_candidates = [x for x in n_candidates if x >= 40 and x % 2 == 0]

    for n_try in n_candidates:
        if remaining_time(t0, budget_s) < reserve + 30:
            break
        h_try = interp_to(best_h, n_try)
        n_t = len(h_try)
        corv_t = corr_full(h_try) * (2.0 / n_t)
        act_t = select_active(corv_t, margin=0.003, pad=2)

        for _ in range(5):
            if remaining_time(t0, budget_s) < reserve + 10:
                break
            if n_t % 2 == 0:
                h_lp = lp_step_sym(h_try, act_t, n_t, solver)
                h_try, c5_try = line_search(h_try, h_lp, eval_c5(h_try))
            else:
                h_lp = lp_step_full(h_try, act_t, n_t, solver)
                h_try, c5_try = line_search(h_try, h_lp, eval_c5(h_try))
            act_t = select_active(corr_full(h_try) * (2.0 / n_t), margin=0.003, pad=2)

        h_lp = lp_step_full(h_try, act_t, n_t, solver)
        h_try, c5_try = line_search(h_try, h_lp, eval_c5(h_try))

        if c5_try < best_c5:
            best_h = h_try.copy()
            best_c5 = c5_try
            n_base = n_try

    # Phase 3: Random restarts with diverse initializations
    n_cur = len(best_h)
    max_restarts = min(30, max(3, int(remaining_time(t0, budget_s) / 20.0)))
    for trial in range(max_restarts):
        if remaining_time(t0, budget_s) < reserve + 15:
            break
        m_r = n_cur // 2
        tt = np.linspace(0.0, 1.0, m_r)
        freq = rng.uniform(0.5, 5.0)
        amp = rng.uniform(0.2, 0.45)
        phase = rng.uniform(0, 2 * np.pi)
        q_r = 0.5 + amp * np.sin(2 * np.pi * freq * tt + phase)
        q_r += 0.1 * rng.randn(m_r)
        q_r = np.clip(q_r, 0.0, 1.0)
        h_r = build_sym(q_r, n_cur)
        h_r = project_to_target(h_r, n_cur / 2.0)

        corv_r = corr_full(h_r) * (2.0 / n_cur)
        act_r = select_active(corv_r, margin=0.003, pad=2)
        for _ in range(4):
            if remaining_time(t0, budget_s) < reserve + 5:
                break
            h_lp = lp_step_sym(h_r, act_r, n_cur, solver)
            h_r, _ = line_search(h_r, h_lp, eval_c5(h_r))
            act_r = select_active(corr_full(h_r) * (2.0 / n_cur), margin=0.003, pad=2)
            h_lp = lp_step_full(h_r, act_r, n_cur, solver)
            h_r, _ = line_search(h_r, h_lp, eval_c5(h_r))

        if eval_c5(h_r) < best_c5:
            best_h = h_r.copy()
            best_c5 = eval_c5(h_r)

    # Phase 4: Final intensive polish
    if remaining_time(t0, budget_s) > 20:
        n_fin = len(best_h)
        for mu in [100.0, 500.0, 2000.0, 10000.0]:
            if remaining_time(t0, budget_s) < 10:
                break
            act_fin = select_active(corr_full(best_h) * (2.0 / n_fin), margin=0.001, pad=3)
            h_fin, c5_fin = lbfgs_step(best_h, act_fin, n_fin, mu=mu, maxiter=300)
            if c5_fin < best_c5:
                best_h = h_fin.copy()
                best_c5 = c5_fin

        # Final LP sweep
        for _ in range(10):
            if remaining_time(t0, budget_s) < 5:
                break
            corv_f = corr_full(best_h) * (2.0 / n_fin)
            act_f = select_active(corv_f, margin=0.001, pad=3)
            h_lp = lp_step_full(best_h, act_f, n_fin, solver)
            best_h, best_c5 = line_search(best_h, h_lp, best_c5)

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_target(best_h, len(best_h) / 2.0)
    best_c5 = eval_c5(best_h)

    return best_h, best_c5, len(best_h)
