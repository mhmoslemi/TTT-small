def build_sym(q, n):

def to_sym_q(h, n):

def grad_c_for_lag(h, d, n):
    """Gradient of c[j] w.r.t. h where d = j - (n-1)"""
    g = np.zeros(n)
    lo = max(0, d)
    hi = min(n-1, n-1+d)
    if lo <= hi:
        g[lo:hi+1] += 1.0 - h[lo-d:hi+1-d]
    lo2 = max(0, -d)
    hi2 = min(n-1, n-1-d)
    if lo2 <= hi2:

    return g

def sym_grad(h, d, n):

    m = n // 2

    gq = np.zeros(m)
    gq[:] = gh[:m]
    gq[:] -= gh[n-1:m-1:-1]
    return gq

import numpy as np
from scipy.optimize import linprog, minimize
import time


def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(c))


def corr_values(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx


def build_sym(q, n):
    m = n // 2
    h = np.empty(n)
    h[:m] = q
    h[m:] = 1.0 - q[::-1]
    return h


def to_sym_q(h, n):
    m = n // 2
    return 0.5 * (h[:m] + 1.0 - h[n - 1:m - 1:-1])


def grad_c_for_lag(h, d, n):
    g = np.zeros(n)
    lo = max(0, d)
    hi = min(n - 1, n - 1 + d)
    if lo <= hi:
        g[lo:hi + 1] += 1.0 - h[lo - d:hi + 1 - d]
    lo2 = max(0, -d)
    hi2 = min(n - 1, n - 1 - d)
    if lo2 <= hi2:
        g[lo2:hi2 + 1] += -h[lo2 + d:hi2 + 1 + d]
    return g


def sym_grad(h, d, n):
    m = n // 2
    gh = grad_c_for_lag(h, d, n)
    gq = gh[:m].copy()
    gq -= gh[n - 1:m - 1:-1]
    return gq


def project_to_target(x, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.any():
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

        n_cur = len(best_h)
        m_cur = n_cur // 2
        q_cur = to_sym_q(best_h, n_cur)

        def sym_smax_fg(z):
            qv = z[:m_cur]
            hq = build_sym(qv, n_cur)
            cv = corr_values(hq)
            s_val = 80.0
            e = np.exp(s_val * (cv - cv.max()))
            w = e / e.sum()
            f = cv.max() + np.log(e.sum()) / s_val
            gq = np.zeros(m_cur)
            for idx in np.where(w > 1e-10)[0]:
                d = int(idx) - (n_cur - 1)
                gq += w[idx] * sym_grad(hq, d, n_cur) * (2.0 / n_cur)
            return f, gq

        z0 = q_cur.copy()
        bounds_sym = [(0.0, 1.0)] * m_cur
        try:
            res = minimize(sym_smax_fg, z0, jac=True, method='L-BFGS-B',
                           bounds=bounds_sym, options={'maxiter': 80, 'gtol': 1e-11})
            q_res = np.clip(res.x[:m_cur], 0.0, 1.0)
            h_res = build_sym(q_res, n_cur)
            c5_res = eval_c5(h_res)
            if c5_res < best_c5:
                best_h = h_res.copy()
                best_c5 = c5_res
        except Exception:
            pass

    # Phase 4: Try different n values
    n_base = len(best_h)
    n_candidates = set()
    for delta in [-12, -9, -7, -5, -4, -3, -2, -1, 1, 2, 3, 4, 5, 7, 9, 12, 16]:
        nc = n_base + delta
        if nc >= 24 and nc % 2 == 0 and nc <= 260:
            n_candidates.add(nc)
    n_candidates = sorted(n_candidates)

    for n_try in n_candidates:
        if remaining() < reserve + 18:
            break
        if n_try == n_base:
            continue
        h_try = interp_to(best_h, n_try)

        q_try = to_sym_q(h_try, n_try)
        h_sym_try = build_sym(q_try, n_try)

        q_try, c5_try = slp_sym(q_try, n_try, max_iter=30, R_init=0.04,
                                margin_init=0.02, t0=t0, time_limit=budget_s - reserve - 12)
        h_try = build_sym(q_try, n_try)
        c5_try = eval_c5(h_try)

        if c5_try < best_c5:
            best_h = h_try.copy()
            best_c5 = c5_try

        if remaining() > reserve + 12:
            h_f, c5_f = slp_full(h_try, n_try, max_iter=15, R_init=0.03,
                                 margin_init=0.02, t0=t0, time_limit=budget_s - reserve - 8)
            if c5_f < best_c5:
                best_h = h_f.copy()
                best_c5 = c5_f

    # Phase 5: Random restarts in symmetric space
    n_r = len(best_h)
    if n_r % 2 != 0:
        n_r -= 1
    max_restarts = min(15, max(4, int(remaining() / 25.0)))

    for trial in range(max_restarts):
        if remaining() < reserve + 12:
            break
        m_r = n_r // 2
        t_arr = np.linspace(0.0, 1.0, m_r)
        k1 = 1 + trial % 4
        k2 = 2 + trial % 3
        k3 = 3 + trial % 5
        p1 = rng.uniform(0, 6.28)
        p2 = rng.uniform(0, 6.28)
        p3 = rng.uniform(0, 6.28)
        a1 = rng.uniform(0.1, 0.3)
        a2 = rng.uniform(0.05, 0.15)
        a3 = rng.uniform(0.03, 0.1)
        q_r = np.clip(0.5 + a1 * np.sin(2 * np.pi * k1 * t_arr + p1) +
                      a2 * np.cos(2 * np.pi * k2 * t_arr + p2) +
                      a3 * np.sin(2 * np.pi * k3 * t_arr + p3), 0.0, 1.0)
        h_r = build_sym(q_r, n_r)

        q_r, c5_r = slp_sym(q_r, n_r, max_iter=25, R_init=0.04,
                            margin_init=0.02, t0=t0, time_limit=budget_s - reserve - 10)
        h_r = build_sym(q_r, n_r)
        c5_r = eval_c5(h_r)

        if c5_r < best_c5:
            best_h = h_r.copy()
            best_c5 = c5_r
            if remaining() > reserve + 20:
                q_r2 = to_sym_q(best_h, len(best_h))
                q_r2, c5_r2 = slp_sym(q_r2, len(best_h), max_iter=40, R_init=0.03,
                                      margin_init=0.008, t0=t0, time_limit=budget_s - reserve - 10)
                h_r2 = build_sym(q_r2, len(best_h))
                if eval_c5(h_r2) < best_c5:
                    best_h = h_r2.copy()
                    best_c5 = eval_c5(h_r2)

    # Phase 6: Final aggressive polish
    if remaining() > 15:
        n_fin = len(best_h)
        if n_fin % 2 == 0:
            q_fin = to_sym_q(best_h, n_fin)
            for margin in [0.003, 0.002, 0.0015]:
                if remaining() < 12:
                    break
                q_fin, c5_fin = slp_sym(q_fin, n_fin, max_iter=40, R_init=0.02,
                                        margin_init=margin, t0=t0, time_limit=budget_s - 10)
                h_fin = build_sym(q_fin, n_fin)
                if eval_c5(h_fin) < best_c5:
                    best_h = h_fin.copy()
                    best_c5 = eval_c5(h_fin)

        if remaining() > 12:
            h_f, c5_f = slp_full(best_h, len(best_h), max_iter=25, R_init=0.015,
                                 margin_init=0.005, t0=t0, time_limit=budget_s - 8)
            if c5_f < best_c5:
                best_h = h_f.copy()
                best_c5 = c5_f

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_target(best_h, len(best_h) / 2.0)
    best_c5 = eval_c5(best_h)

    return best_h, best_c5, len(best_h)
