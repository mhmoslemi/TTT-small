import numpy as np
from scipy.optimize import linprog
import time

def eval_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def corr_raw(h):
    return np.correlate(h, 1.0 - h, mode='full')

def build_sym_h(q):
    m = len(q)
    h = np.empty(2 * m)
    h[:m] = q
    h[m:] = 1.0 - q[::-1]
    return h

def sym_from_h(h):
    n = len(h)
    m = n // 2
    q = np.empty(m)
    for i in range(m):
        q[i] = 0.5 * (h[i] + (1.0 - h[n - 1 - i]))
    return np.clip(q, 0.0, 1.0)

def project_feasible(x, target_sum):
    n = len(x)
    x = np.clip(x, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(x + mid, 0.0, 1.0).sum()
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    h = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        interior = (h > 1e-10) & (h < 1.0 - 1e-10)
        if interior.sum() > 0:
            h[interior] += diff / interior.sum()
    return np.clip(h, 0.0, 1.0)

def transfer_to_n(h_old, n_new):
    n_old = len(h_old)
    x_old = (np.arange(n_old) + 0.5) * 2.0 / n_old
    x_new = (np.arange(n_new) + 0.5) * 2.0 / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new / 2.0)

def corr_grad_full(h, k):
    """Gradient of raw_corr[k] w.r.t. h. k is the index in the full correlate output."""
    n = len(h)
    d = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -d)
    i_hi = min(n - 1, n - 1 - d)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    if d == 0:
        grad[idx] = 1.0 - 2.0 * h[idx]
    else:
        grad[idx] += (1.0 - h[idx + d])
        grad[idx + d] += (-h[idx])
    return grad

def select_active(corrs, margin_frac=0.05, max_k=40):
    max_val = corrs.max()
    threshold = max_val - margin_frac * max_val
    active = np.where(corrs >= threshold)[0]
    if len(active) > max_k:
        # Keep the top max_k
        order = np.argsort(corrs[active])[::-1][:max_k]
        active = active[order]
    return active

def solve_tangent_lp_full(h, active, rho=0.3):
    """Solve tangent LP for full h vector. Returns direction d = z - h."""
    n = len(h)
    corrs = corr_raw(h)
    
    # Build LP: minimize t
    # Variables: z[0..n-1], t[n]
    # Constraints: for each active ell: grad_ell^T z - t <= grad_ell^T h - R_ell
    # Bounds: 0 <= z_i <= 1, rho constraints: h_i - rho <= z_i <= h_i + rho
    # Equality: sum(z) = n/2
    
    A_ub = []
    b_ub = []
    for ell in active:
        g = corr_grad_full(h, ell)
        row = np.zeros(n + 1)
        row[:n] = g
        row[n] = -1.0
        A_ub.append(row)
        b_ub.append(float(g @ h - corrs[ell]))
    
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    # Equality: sum(z) = n/2
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = [n / 2.0]
    
    # Bounds for z: max(0, h_i - rho) <= z_i <= min(1, h_i + rho)
    lb = np.maximum(0.0, h - rho)
    ub = np.minimum(1.0, h + rho)
    bounds = list(zip(lb, ub)) + [(0, corrs.max() + 1.0)]
    
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize t
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
        if res.status == 0:
            z = res.x[:n]
            d = z - h
            # Verify sum is approximately preserved
            return d
    except Exception:
        pass
    return None

                lo = a1
                a1 = a2
                f1 = f2
                a2 = lo + gr * (hi - lo)
                f2 = eval_score(np.clip(h + a2 * d, 0, 1))

def pair_transfer_search(h, n_pairs=50, n_evals=10, time_limit=None):

            h_try[b] -= delta

            best_c5 = best_val
            best_h[a] += best_delta

import numpy as np
from scipy.optimize import linprog
import time

def eval_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def corr_raw(h):
    return np.correlate(h, 1.0 - h, mode='full')

def build_sym_h(q):
    m = len(q)
    h = np.empty(2 * m)
    h[:m] = q
    h[m:] = 1.0 - q[::-1]
    return h

def sym_from_h(h):
    n = len(h)
    m = n // 2
    q = np.empty(m)
    for i in range(m):
        q[i] = 0.5 * (h[i] + (1.0 - h[n - 1 - i]))
    return np.clip(q, 0.0, 1.0)

def project_feasible(x, target_sum):
    n = len(x)
    x = np.asarray(x, dtype=float).copy()
    x = np.clip(x, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(x + mid, 0.0, 1.0).sum()
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    h = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        interior = (h > 1e-10) & (h < 1.0 - 1e-10)
        if interior.sum() > 0:
            h[interior] += diff / interior.sum()
    return np.clip(h, 0.0, 1.0)

def transfer_to_n(h_old, n_new):
    n_old = len(h_old)
    x_old = (np.arange(n_old) + 0.5) * 2.0 / n_old
    x_new = (np.arange(n_new) + 0.5) * 2.0 / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new / 2.0)

def corr_grad_full(h, k):
    n = len(h)
    d = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -d)
    i_hi = min(n - 1, n - 1 - d)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    if d == 0:
        grad[idx] = 1.0 - 2.0 * h[idx]
    else:
        grad[idx] += (1.0 - h[idx + d])
        grad[idx + d] += (-h[idx])
    return grad

def select_active(corrs, margin_frac, max_k):
    max_val = corrs.max()
    threshold = max_val - margin_frac * max_val
    active = np.where(corrs >= threshold)[0]
    if len(active) > max_k:
        order = np.argsort(corrs[active])[::-1][:max_k]
        active = active[order]
    return active

def solve_tangent_lp_full(h, active, rho):
    n = len(h)
    corrs = corr_raw(h)
    n_active = len(active)
    A_ub = np.zeros((n_active, n + 1))
    b_ub = np.zeros(n_active)
    for row_idx, ell in enumerate(active):
        g = corr_grad_full(h, ell)
        A_ub[row_idx, :n] = g
        A_ub[row_idx, n] = -1.0
        b_ub[row_idx] = float(g @ h - corrs[ell])
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    lb = np.maximum(0.0, h - rho)
    ub = np.minimum(1.0, h + rho)
    bounds = list(zip(lb, ub))
    bounds.append((0.0, float(corrs.max()) + 2.0))
    c = np.zeros(n + 1)
    c[n] = 1.0
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,

        if res.status == 0:
            z = res.x[:n]
            return z - h, float(res.x[n])
    except Exception:
        pass
    return None, None

        if res.status == 0:
            z = res.x[:m]
            return z - q, float(res.x[m])
    except Exception:
        pass
    return None, None

def line_search_alpha(h, d, alpha_max, n_grid):
    if alpha_max <= 1e-12:
        return 0.0
    best_alpha = 0.0
    best_val = eval_score(h)
    alphas = np.linspace(0.0, alpha_max, n_grid)
    for a in alphas:
        h_try = h + a * d
        if np.any(h_try < -1e-9) or np.any(h_try > 1.0 + 1e-9):
            continue
        v = eval_score(np.clip(h_try, 0, 1))
        if v < best_val:
            best_val = v
            best_alpha = a
    if best_alpha > 0 and best_alpha < alpha_max:
        lo = max(0.0, best_alpha - alpha_max / max(n_grid - 1, 2))
        hi = min(alpha_max, best_alpha + alpha_max / max(n_grid - 1, 2))
        if hi - lo > 1e-10:
            gr_ratio = (np.sqrt(5.0) - 1.0) / 2.0
            a1 = hi - gr_ratio * (hi - lo)
            a2 = lo + gr_ratio * (hi - lo)
            f1 = eval_score(np.clip(h + a1 * d, 0, 1))
            f2 = eval_score(np.clip(h + a2 * d, 0, 1))
            for _ in range(25):
                if f1 < f2:
                    hi = a2
                    a2 = a1
                    f2 = f1
                    a1 = hi - gr_ratio * (hi - lo)
                    f1 = eval_score(np.clip(h + a1 * d, 0, 1))
                else:
                    lo = a1
                    a1 = a2
                    f1 = f2
                    a2 = lo + gr_ratio * (hi - lo)
                    f2 = eval_score(np.clip(h + a2 * d, 0, 1))
            mid = (lo + hi) / 2.0
            v_mid = eval_score(np.clip(h + mid * d, 0, 1))
            if v_mid < best_val:
                best_val = v_mid
                best_alpha = mid
    return best_alpha

def lp_optimize_full(h, max_iter, time_limit, rho, margin_frac, max_k):
    n = len(h)
    best_h = h.copy()
    best_c5 = eval_score(h)
    cur_h = h.copy()
    for it in range(max_iter):
        if time.time() > time_limit:
            break
        corrs = corr_raw(cur_h)
        max_raw = corrs.max()
        active = select_active(corrs, margin_frac, max_k)
        d, lp_t = solve_tangent_lp_full(cur_h, active, rho)
        if d is None:
            break
        if lp_t is not None and lp_t >= max_raw - 1e-9:
            break
        alpha_max = np.inf
        for i in range(n):
            if d[i] > 1e-12:
                alpha_max = min(alpha_max, (1.0 - cur_h[i]) / d[i])
            elif d[i] < -1e-12:
                alpha_max = min(alpha_max, cur_h[i] / (-d[i]))
        alpha_max = min(alpha_max, 1.0)
        if alpha_max <= 1e-12:
            break
        best_alpha = line_search_alpha(cur_h, d, alpha_max, 12)
        if best_alpha < 1e-12:
            break
        h_new = np.clip(cur_h + best_alpha * d, 0.0, 1.0)
        h_new = project_feasible(h_new, n / 2.0)
        c5_new = eval_score(h_new)
        if c5_new < best_c5 - 1e-13:
            best_c5 = c5_new
            best_h = h_new.copy()
            cur_h = h_new
        else:
            half_a = best_alpha * 0.5
            if half_a > 1e-12:
                h_try = np.clip(cur_h + half_a * d, 0.0, 1.0)
                h_try = project_feasible(h_try, n / 2.0)
                c5_try = eval_score(h_try)
                if c5_try < best_c5 - 1e-13:
                    best_c5 = c5_try
                    best_h = h_try.copy()
                    cur_h = h_try
                    continue
            break
    return best_h, best_c5

def lp_optimize_sym(q, max_iter, time_limit, rho, margin_frac, max_k):
    m = len(q)
    n = 2 * m
    best_q = q.copy()
    best_c5 = eval_score(build_sym_h(q))
    cur_q = q.copy()
    for it in range(max_iter):
        if time.time() > time_limit:
            break
        h_cur = build_sym_h(cur_q)
        corrs = corr_raw(h_cur)
        max_raw = corrs.max()
        active = select_active(corrs, margin_frac, max_k)
        d, lp_t = solve_tangent_lp_sym(cur_q, active, rho)
        if d is None:
            break
        if lp_t is not None and lp_t >= max_raw - 1e-9:
            break
        alpha_max = np.inf
        for i in range(m):
            if d[i] > 1e-12:
                alpha_max = min(alpha_max, (1.0 - cur_q[i]) / d[i])
            elif d[i] < -1e-12:
                alpha_max = min(alpha_max, cur_q[i] / (-d[i]))
        alpha_max = min(alpha_max, 1.0)
        if alpha_max <= 1e-12:
            break
        best_alpha = 0.0
        best_val = float(max_raw * 2.0 / n)
        alphas = np.linspace(0.0, alpha_max, 12)
        for a in alphas:
            q_try = np.clip(cur_q + a * d, 0.0, 1.0)
            h_try = build_sym_h(q_try)
            v = eval_score(h_try)
            if v < best_val:
                best_val = v
                best_alpha = a
        if best_alpha > 0 and best_alpha < alpha_max:
            lo = max(0.0, best_alpha - 0.1 * alpha_max)
            hi = min(alpha_max, best_alpha + 0.1 * alpha_max)
            if hi - lo > 1e-10:
                for a in np.linspace(lo, hi, 15):
                    q_try = np.clip(cur_q + a * d, 0.0, 1.0)
                    h_try = build_sym_h(q_try)
                    v = eval_score(h_try)
                    if v < best_val:
                        best_val = v
                        best_alpha = a
        if best_val < best_c5 - 1e-13:
            best_c5 = best_val
            best_q = np.clip(cur_q + best_alpha * d, 0.0, 1.0)
            cur_q = best_q.copy()
        else:
            break
    return best_q, best_c5

def pair_transfer_search(h, n_pairs, n_evals, rng_seed, time_limit):
    n = len(h)
    rng = np.random.RandomState(rng_seed)
    best_h = h.copy()
    best_c5 = eval_score(h)
    cur_h = h.copy()
    for _ in range(n_pairs):
        if time.time() > time_limit:
            break
        a, b = rng.choice(n, 2, replace=False)
        lo = max(-cur_h[a], cur_h[b] - 1.0)
        hi = min(1.0 - cur_h[a], -cur_h[b])
        if hi - lo < 1e-12:
            continue
        deltas = np.linspace(lo, hi, n_evals)
        best_delta = 0.0
        best_val = best_c5
        for delta in deltas:
            h_try = cur_h.copy()
            h_try[a] += delta
            h_try[b] -= delta
            if h_try[a] < -1e-10 or h_try[a] > 1.0 + 1e-10 or h_try[b] < -1e-10 or h_try[b] > 1.0 + 1e-10:
                continue
            v = eval_score(h_try)
            if v < best_val:
                best_val = v
                best_delta = delta
        if best_delta == 0.0:
            continue
        cur_h[a] += best_delta
        cur_h[b] -= best_delta
        if best_val < best_c5 - 1e-13:
            best_c5 = best_val
            best_h = cur_h.copy()
    return best_h, best_c5

def smooth_max_optimize(h, n_betas, maxiter, time_limit):
    n = len(h)
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = eval_score(h)
    cur_h = h.copy()
    for beta in n_betas:
        if time.time() > time_limit:
            break
        prev_c5 = eval_score(cur_h)
        for inner in range(maxiter):
            if time.time() > time_limit:
                break
            corrs = corr_raw(cur_h) * dx
            max_c = corrs.max()
            shifted = corrs - max_c
            w = np.exp(beta * shifted)
            w_sum = w.sum()
            if w_sum < 1e-30:
                w = np.ones(len(corrs)) / len(corrs)
            else:
                w = w / w_sum
            grad = np.zeros(n)
            for k in range(len(corrs)):
                if w[k] < 1e-14:
                    continue
                g = corr_grad_full(cur_h, k)
                grad += w[k] * g
            grad *= dx
            alpha_max = np.inf
            for i in range(n):
                if grad[i] > 1e-14:
                    alpha_max = min(alpha_max, (1.0 - cur_h[i]) / grad[i])
                elif grad[i] < -1e-14:
                    alpha_max = min(alpha_max, cur_h[i] / (-grad[i]))
            alpha_max = min(alpha_max, 1.0)
            if alpha_max <= 1e-12:
                break
            best_alpha = 0.0
            best_val = eval_score(cur_h)
            for a in np.linspace(0.0, alpha_max, 10):
                h_try = np.clip(cur_h - a * grad, 0.0, 1.0)
                h_try = project_feasible(h_try, n / 2.0)
                v = eval_score(h_try)
                if v < best_val:
                    best_val = v
                    best_alpha = a
            if best_alpha < 1e-12:
                break
            cur_h = np.clip(cur_h - best_alpha * grad, 0.0, 1.0)
            cur_h = project_feasible(cur_h, n / 2.0)
            c5_new = eval_score(cur_h)
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                best_h = cur_h.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')

    try:
        h_parent = np.array(initial_h_values, dtype=float).copy().reshape(-1)
    except (NameError, AttributeError):
        h_parent = None

    if h_parent is not None and len(h_parent) > 10:
        n_p = len(h_parent)
        if abs(np.sum(h_parent) - n_p / 2.0) > 0.5:
            h_parent = project_feasible(h_parent, n_p / 2.0)
        h_parent = np.clip(h_parent, 0.0, 1.0)
        h_parent = project_feasible(h_parent, n_p / 2.0)
        best_h = h_parent.copy()
        best_c5 = eval_score(h_parent)
    else:
        n_p = 126
        m_p = n_p // 2
        x = np.linspace(0, 1, m_p)
        q_p = np.clip(0.5 + 0.3 * np.sin(2 * np.pi * x), 0, 1)
        best_h = build_sym_h(q_p)
        best_c5 = eval_score(best_h)

    rem = budget_s - (time.time() - t0)

    # Phase 1: LP exploration (large rho)
    if rem > 250:
        tl = time.time() + min(180, rem * 0.3)
        h_new, c5_new = lp_optimize_full(best_h, max_iter=80, time_limit=tl,
                                         rho=0.3, margin_frac=0.06, max_k=50)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

    rem = budget_s - (time.time() - t0)

    # Phase 2: Pair transfers
    if rem > 200:
        tl = time.time() + min(120, rem * 0.25)
        h_new, c5_new = pair_transfer_search(best_h, n_pairs=300, n_evals=12,
                                             rng_seed=seed, time_limit=tl)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

    rem = budget_s - (time.time() - t0)

    # Phase 3: LP polishing (small rho)
    if rem > 200:
        tl = time.time() + min(150, rem * 0.3)
        h_new, c5_new = lp_optimize_full(best_h, max_iter=100, time_limit=tl,
                                         rho=0.08, margin_frac=0.03, max_k=60)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

    rem = budget_s - (time.time() - t0)

    # Phase 4: Symmetric LP
    n_cur = len(best_h)
    if rem > 150 and n_cur % 2 == 0:
        q_cur = sym_from_h(best_h)
        tl = time.time() + min(100, rem * 0.25)
        q_new, c5_new = lp_optimize_sym(q_cur, max_iter=80, time_limit=tl,
                                        rho=0.15, margin_frac=0.05, max_k=50)
        h_new = build_sym_h(q_new)
        c5_new = eval_score(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

    rem = budget_s - (time.time() - t0)

    # Phase 5: More LP with medium rho
    if rem > 120:
        tl = time.time() + min(90, rem * 0.3)
        h_new, c5_new = lp_optimize_full(best_h, max_iter=80, time_limit=tl,
                                         rho=0.15, margin_frac=0.04, max_k=55)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

    rem = budget_s - (time.time() - t0)

    # Phase 6: Multi-resolution
    n_base = len(best_h)
    n_candidates = []
    for delta in [-8, -6, -4, -3, -2, -1, 1, 2, 3, 4, 6, 8]:
        nc = n_base + delta
        if 48 <= nc <= 200:
            n_candidates.append(nc)
    n_candidates = sorted(set(n_candidates))

    for n_try in n_candidates:
        rem = budget_s - (time.time() - t0)
        if rem < 60:
            break
        h_try = transfer_to_n(best_h, n_try)
        n_try = len(h_try)
        if rem > 80:
            tl = time.time() + min(50, rem * 0.4)
            h_new, c5_new = lp_optimize_full(h_try, max_iter=40, time_limit=tl,
                                             rho=0.2, margin_frac=0.05, max_k=45)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new
            h_try = h_new
        if rem > 40:
            tl = time.time() + min(25, rem * 0.4)
            h_new, c5_new = pair_transfer_search(h_try, n_pairs=100, n_evals=10,
                                                 rng_seed=seed + n_try, time_limit=tl)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new

    rem = budget_s - (time.time() - t0)

    # Phase 7: Diversification
    n_div = 5
    for tri in range(n_div):
        rem = budget_s - (time.time() - t0)
        if rem < 70:
            break
        n_r = n_base if tri % 2 == 0 else max(60, n_base - 4)
        n_r = min(200, max(50, n_r))
        if n_r % 2 == 0:
            m_r = n_r // 2
            x_r = np.linspace(0, 1, m_r)
            freq = (tri + 1) % 3 + 1
            phase = rng.uniform(0, 2 * np.pi)
            amp = 0.15 + 0.1 * tri
            q_r = np.clip(0.5 + amp * np.sin(2 * np.pi * freq * x_r + phase), 0, 1)
            h_r = build_sym_h(q_r)
        else:
            h_r = np.clip(0.5 + 0.25 * rng.randn(n_r), 0, 1)
            h_r = project_feasible(h_r, n_r / 2.0)
        if rem > 90:
            tl = time.time() + min(55, rem * 0.4)
            h_new, c5_new = lp_optimize_full(h_r, max_iter=40, time_limit=tl,
                                             rho=0.3, margin_frac=0.06, max_k=45)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new
            h_r = h_new
        if rem > 40:
            tl = time.time() + min(25, rem * 0.4)
            h_new, c5_new = pair_transfer_search(h_r, n_pairs=80, n_evals=10,
                                                 rng_seed=seed + 100 + tri, time_limit=tl)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new

    rem = budget_s - (time.time() - t0)

    # Phase 8: Final polish - alternating LP and pair transfers
    if rem > 30 and best_h is not None:
        for rnd in range(4):
            rem = budget_s - (time.time() - t0)
            if rem < 15:
                break
            if rnd % 2 == 0:
                tl = time.time() + min(12, rem * 0.4)
                h_new, c5_new = lp_optimize_full(best_h, max_iter=30, time_limit=tl,
                                                 rho=0.05, margin_frac=0.02, max_k=50)
            else:
                tl = time.time() + min(10, rem * 0.4)
                h_new, c5_new = pair_transfer_search(best_h, n_pairs=150, n_evals=10,
                                                     rng_seed=seed + 200 + rnd, time_limit=tl)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new

    # Final validation
    if best_h is None:
        m_fb = 63
        x_fb = np.linspace(0, 1, m_fb)
        q_fb = np.clip(0.5 + 0.3 * np.sin(2 * np.pi * x_fb), 0, 1)
        best_h = build_sym_h(q_fb)

    best_h = np.asarray(best_h, dtype=float).reshape(-1)
    n_out = len(best_h)
    best_h = project_feasible(best_h, n_out / 2.0)
    best_c5 = eval_score(best_h)

    return best_h, best_c5, n_out
