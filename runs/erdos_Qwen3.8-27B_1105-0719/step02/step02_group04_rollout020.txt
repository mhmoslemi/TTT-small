    m = n // 2
    h = build_h_from_q(q, n)
    # Build constraint matrix
    # Variables: [delta_0..delta_{m-1}, t, s_0..s_{m-1}]
    # Total: m + 1 + m = 2m + 1
    # Objective: [0, 0, ..., 0, 1, mu, mu, ..., mu]
    c_obj = np.zeros(2 * m + 1)
    c_obj[m] = 1.0  # t
    c_obj[m+1:] = mu  # s_i
    
    # Constraints: A_ub @ x <= b_ub
    n_constraints = len(active_j) + 2 * m
    A_ub = np.zeros((n_constraints, 2 * m + 1))
    b_ub = np.zeros(n_constraints)
    
    for k, j in enumerate(active_j):
        gq = grad_q(h, j, n)
        A_ub[k, :m] = gq  # gq · delta
        A_ub[k, m] = -1.0  # -t
        b_ub[k] = -C_vals[j]  # <= -C[j]

        A_ub[len(active_j) + i, i] = 1.0

        # -delta_i - s_i <= 0
        A_ub[len(active_j) + m + i, i] = -1.0

import numpy as np
from scipy.optimize import linprog, minimize
import time


def dx_of(n):
    return 2.0 / n


def eval_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx_of(n))


def corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full') * dx_of(len(h))


def build_h_from_q(q, n):
    m = n // 2
    h = np.empty(n)
    h[:m] = q
    h[m:] = 1.0 - q[::-1]
    return h


def q_from_h(h, n):
    m = n // 2
    return 0.5 * (h[:m] + 1.0 - h[n - 1:m - 1:-1])


def grad_corr_row(h, j, n):
    """Gradient of corr_full(h)[j] with respect to h, scaled by dx."""
    d = j - (n - 1)
    n_pts = n
    n_lo = max(0, -d)
    n_hi = min(n_pts - 1, n_pts - 1 - d)
    g = np.zeros(n_pts)
    if n_lo > n_hi:
        return g
    # Term 1: d/dh[t] of h[t] where t = i+d, i in [n_lo, n_hi]
    # t ranges from n_lo+d to n_hi+d
    t_lo1 = max(0, n_lo + d)
    t_hi1 = min(n_pts - 1, n_hi + d)
    if t_lo1 <= t_hi1:
        idx = np.arange(t_lo1, t_hi1 + 1)
        g[idx] += 1.0 - h[idx - d]
    # Term 2: d/dh[t] of (1-h[t]) where t = i, i in [n_lo, n_hi]
    t_lo2 = n_lo
    t_hi2 = n_hi
    idx = np.arange(t_lo2, t_hi2 + 1)
    g[idx] += -h[idx + d]
    return g * dx_of(n_pts)


def grad_q_row(h, j, n):
    """Gradient of corr_full(h)[j] with respect to q (anti-complementary)."""
    m = n // 2
    gh = grad_corr_row(h, j, n)
    return gh[:m] - gh[n - 1:m - 1:-1]


def verify_gradient(h, n, j, verbose=False):
    """Numerically verify gradient for one lag."""
    g_analytic = grad_corr_row(h, j, n)
    eps = 1e-7
    g_num = np.zeros(n)
    for i in range(n):
        h_plus = h.copy()
        h_plus[i] += eps
        h_minus = h.copy()
        h_minus[i] -= eps
        c_plus = np.correlate(h_plus, 1.0 - h_plus, mode='full')[j] * dx_of(n)
        c_minus = np.correlate(h_minus, 1.0 - h_minus, mode='full')[j] * dx_of(n)
        g_num[i] = (c_plus - c_minus) / (2.0 * eps)
    err = np.max(np.abs(g_analytic - g_num))
    if verbose and err > 1e-5:
        print(f"Gradient verification failed for j={j}: err={err}")
    return err


def select_active(C, margin=0.002, pad=2, max_frac=0.15):
    """Select active constraint indices."""
    c_max = np.max(C)
    n_total = len(C)
    thresh = c_max - margin
    idx = np.where(C >= thresh)[0]
    # Add neighbors
    extended = set()
    for a in idx:
        for delta in range(-pad, pad + 1):
            k = a + delta
            if 0 <= k < n_total:
                extended.add(k)
    # Cap the number of active constraints
    idx_sorted = np.array(sorted(extended))
    if len(idx_sorted) > max_frac * n_total:
        # Keep only the top ones
        vals = C[idx_sorted]
        order = np.argsort(-vals)
        idx_sorted = idx_sorted[order[:int(max_frac * n_total)]]
        idx_sorted = np.sort(idx_sorted)
    return idx_sorted


def slp_lp_solve(q, n, active_j, C_vals, R, mu):
    """Solve the SLP linear program. Returns delta or None."""
    m = n // 2
    h = build_h_from_q(q, n)
    # Variables: [delta_0..delta_{m-1}, t, s_0..s_{m-1}]
    n_vars = 2 * m + 1
    c_obj = np.zeros(n_vars)
    c_obj[m] = 1.0
    c_obj[m + 1:] = mu

    n_act = len(active_j)
    n_constr = n_act + 2 * m
    A_ub = np.zeros((n_constr, n_vars))
    b_ub = np.zeros(n_constr)

    for k, j in enumerate(active_j):
        gq = grad_q_row(h, j, n)
        A_ub[k, :m] = gq
        A_ub[k, m] = -1.0
        b_ub[k] = -C_vals[j]

    for i in range(m):
        A_ub[n_act + i, i] = 1.0
        A_ub[n_act + i, m + 1 + i] = -1.0
        b_ub[n_act + i] = 0.0
        A_ub[n_act + m + i, i] = -1.0
        A_ub[n_act + m + i, m + 1 + i] = -1.0
        b_ub[n_act + m + i] = 0.0

    bounds = []
    for i in range(m):
        lo = max(-q[i], -R)
        hi = min(1.0 - q[i], R)
        bounds.append((lo, hi))
    c_max = float(np.max(C_vals))
    bounds.append((0.0, c_max + 0.5))
    for i in range(m):
        bounds.append((0.0, R))

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.status != 0:
            return None
        return res.x[:m]
    except Exception:
        return None


def slp_line_search(q, n, delta, C_cur):
    """Line search: find best lambda that improves objective."""
    m = n // 2
    lambdas = [1.0, 0.8, 0.6, 0.5, 0.4, 0.3, 0.2, 0.1, 0.05]
    best_c = C_cur
    best_q = q.copy()
    best_lam = 0.0
    for lam in lambdas:
        q_try = q + lam * delta
        if np.any(q_try < -1e-10) or np.any(q_try > 1.0 + 1e-10):
            continue
        q_try = np.clip(q_try, 0.0, 1.0)
        h_try = build_h_from_q(q_try, n)
        c_try = eval_c5(h_try)
        if c_try < best_c - 1e-12:
            best_c = c_try
            best_q = q_try.copy()
            best_lam = lam
    return best_q, best_c, best_lam


def slp_optimize(q, n, max_iter=80, R_init=0.04, mu_init=0.03,

    best_c = C_cur

            if no_progress > 8:

            continue

        q_new, c_new, lam = slp_line_search(q_cur, n, delta, C_cur)

def interp_h_to(h_old, n_new):
    """Interpolate h from old grid to new grid, preserving sum."""
    n_old = len(h_old)
    if n_new == n_old:
        return h_old.copy()
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h_old)
    # Project to correct sum
    target = n_new / 2.0
    current = h_new.sum()
    # Water-filling
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        clipped = np.clip(h_new - mid, 0.0, 1.0)
        if clipped.sum() > target:
            lo = mid
        else:
            hi = mid
    h_new = np.clip(h_new - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - h_new.sum()
    if abs(diff) > 1e-10:
        mask = (h_new > 1e-12) & (h_new < 1.0 - 1e-12)
        if mask.any():
            h_new[mask] += diff / mask.sum()
    return np.clip(h_new, 0.0, 1.0)


def full_pen_model_fg(z, n, active_j, mu, mu_sum):
    """Objective and gradient for full-variable penalty method."""
    h = z[:n]
    t = z[n]
    C_arr = corr_full(h)
    vals = C_arr[active_j]
    exc = np.maximum(0.0, vals - t)
    sdev = h.sum() - n / 2.0
    f = t + 0.5 * mu * np.dot(exc, exc) + 0.5 * mu_sum * sdev * sdev
    gf = np.zeros(n + 1)
    gf[n] = 1.0 - mu * exc.sum()
    gh = np.zeros(n)
    for i in range(len(active_j)):
        if exc[i] > 1e-15:
            j = active_j[i]
            gh += mu * exc[i] * grad_corr_row(h, j, n)
    gf[:n] = gh + mu_sum * sdev
    return f, gf


def full_optimize(h0, n, max_iter=150, time_limit=None, t_start=0):
    """Full-variable optimization (breaks symmetry)."""
    active_j = select_active(corr_full(h0), margin=0.005, pad=3)
    t0 = eval_c5(h0)
    z0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, t0 + 1.0)]
    mu_schedule = [50.0, 200.0, 1000.0, 5000.0, 10000.0]
    best_h = h0.copy()
    best_c = t0

    for mu in mu_schedule:
        if time_limit is not None and time.time() - t_start > time_limit:
            break
        class Model:
            def __init__(self, n, aj, mu_col, mu_s):
                self.n = n
                self.aj = aj
                self.mu = mu_col
                self.mu_s = mu_s

            def fg(self, z):
                return full_pen_model_fg(z, self.n, self.aj, self.mu, self.mu_s)

        model = Model(n, active_j, mu, 2000.0)
        try:
            res = minimize(model.fg, z0, jac=True, method='L-BFGS-B',
                           bounds=bounds, options={'maxiter': max_iter // len(mu_schedule),
                                                  'gtol': 1e-10})
            h_new = np.clip(res.x[:n], 0.0, 1.0)
            # Repair sum
            target = n / 2.0
            diff = target - h_new.sum()
            mask = (h_new > 1e-12) & (h_new < 1.0 - 1e-12)
            if mask.any():
                h_new[mask] += diff / mask.sum()
            h_new = np.clip(h_new, 0.0, 1.0)
            c_new = eval_c5(h_new)
            if c_new < best_c - 1e-12:

                best_h = h_new.copy()
                z0 = np.concatenate([h_new, [c_new]])
                active_j = select_active(corr_full(h_new), margin=0.003, pad=3)
        except Exception:
            pass

    return best_h, best_c


def make_sym_q(n, rng, style=0):

        a1 = rng.uniform(0.1, 0.4)
        a2 = rng.uniform(0.05, 0.2)
        q = 0.5 + a1 * np.sin(2 * np.pi * k1 * t + ph1) + a2 * np.cos(2 * np.pi * k2 * t + ph2)
    elif style == 1:

        n_steps = rng.randint(2, 6)
        edges = np.sort(rng.uniform(0, 1, n_steps - 1))

        q = np.zeros(m)
        for i in range(n_steps - 1):

    else:
        k1 = rng.randint(2, 6)
        ph1 = rng.uniform(0, 2 * np.pi)
        a1 = rng.uniform(0.2, 0.45)
        q = 0.5 + a1 * np.sin(np.pi * k1 * t + ph1)
    return np.clip(q, 0.0, 1.0)


def coordinate_refine(h, n, n_steps=200, step_size=0.02, time_limit=None, t_start=0):
    """Simple coordinate descent on full h with anti-complement constraint."""
    m = n // 2
    c_cur = eval_c5(h)
    h_cur = h.copy()
    rng_local = np.random.RandomState(12345)

    for step in range(n_steps):
        if time_limit is not None and time.time() - t_start > time_limit:
            break
        # Pick a random coordinate to perturb
        i = rng_local.randint(0, min(m, 30))
        # Try both positive and negative perturbations
        for sign in [1, -1]:
            h_try = h_cur.copy()
            h_try[i] = np.clip(h_cur[i] + sign * step_size, 0.0, 1.0)
            h_try[n - 1 - i] = 1.0 - h_try[i]
            c_try = eval_c5(h_try)
            if c_try < c_cur - 1e-13:
                h_cur = h_try
                c_cur = c_try
                break
        # Occasionally try a larger step
        if step % 50 == 30:
            i = rng_local.randint(0, min(m, 30))
            for sign in [1, -1]:
                h_try = h_cur.copy()
                h_try[i] = np.clip(h_cur[i] + sign * step_size * 3, 0.0, 1.0)
                h_try[n - 1 - i] = 1.0 - h_try[i]
                c_try = eval_c5(h_try)
                if c_try < c_cur - 1e-13:
                    h_cur = h_try
                    c_cur = c_try
                    break

    return h_cur, c_cur

            best_n = n_parent

    # Second SLP pass with tighter parameters
    if time_ok(40):
        q_try = q_from_h(best_h, best_n)
        q_opt2, c5_opt2 = slp_optimize(q_try, best_n, max_iter=60,
                                       R_init=0.02, mu_init=0.05, margin_init=0.001,
                                       time_limit=budget_s - 40, t_start=t0)
        if c5_opt2 < best_c5:
            best_c5 = c5_opt2
            best_h = build_h_from_q(q_opt2, best_n)

    # Phase 2: Multi-resolution
    n_base = best_n
    n_candidates = sorted(set([
        max(40, n_base - 12),
        max(40, n_base - 6),
        n_base + 6,
        n_base + 12,
        min(256, n_base + 18),
    ]))
    # Filter to even
    n_candidates = [nc for nc in n_candidates if nc % 2 == 0 and 32 <= nc <= 300]

    for n_try in n_candidates:
        if not time_ok(30):
            break
        if n_try == n_base:
            continue
        h_try = interp_h_to(best_h, n_try)
        if n_try % 2 == 0:
            q_try = q_from_h(h_try, n_try)
            c_try = eval_c5(build_h_from_q(q_try, n_try))
            if c_try < best_c5 + 0.01:
                q_opt3, c5_opt3 = slp_optimize(q_try, n_try, max_iter=50,
                                               R_init=0.03, mu_init=0.03, margin_init=0.002,
                                               time_limit=budget_s - 30, t_start=t0)
                if c5_opt3 < best_c5:
                    best_c5 = c5_opt3
                    best_h = build_h_from_q(q_opt3, n_try)
                    best_n = n_try

    # Phase 3: Full-variable optimization (break anti-complementary symmetry)
    if time_ok(80):
        h_full, c5_full = full_optimize(best_h, best_n, max_iter=200,
                                        time_limit=budget_s - 80, t_start=t0)
        if c5_full < best_c5:
            best_c5 = c5_full
            best_h = h_full.copy()
            best_n = len(best_h)

    # Phase 4: Coordinate refinement
    if time_ok(30):
        h_coord, c5_coord = coordinate_refine(best_h, best_n, n_steps=300,
                                              step_size=0.015,
                                              time_limit=budget_s - 20, t_start=t0)
        if c5_coord < best_c5:
            best_c5 = c5_coord
            best_h = h_coord.copy()

    # Phase 5: Random restarts
    max_restarts = int(max(3, remaining() / 25.0))
    for trial in range(max_restarts):
        if not time_ok(20):
            break
        n_r = best_n
        if n_r % 2 != 0:
            n_r -= 1
        for style in [0, 1, 2]:
            q_r = make_sym_q(n_r, rng, style=style)
            h_r = build_h_from_q(q_r, n_r)
            q_opt_r, c5_r = slp_optimize(q_r, n_r, max_iter=40,
                                         R_init=0.05, mu_init=0.05, margin_init=0.005,
                                         time_limit=budget_s - 20, t_start=t0)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = build_h_from_q(q_opt_r, n_r)
                best_n = n_r
                break

    # Phase 6: Final SLP polish with very tight parameters
    if time_ok(20):
        n_fin = best_n
        if n_fin % 2 != 0:
            n_fin -= 1
            best_h = interp_h_to(best_h, n_fin)
        q_fin = q_from_h(best_h, n_fin)
        h_fin_test = build_h_from_q(q_fin, n_fin)
        if eval_c5(h_fin_test) < best_c5:
            best_c5 = eval_c5(h_fin_test)
            best_h = h_fin_test
        q_fin2, c5_fin2 = slp_optimize(q_fin, n_fin, max_iter=80,
                                       R_init=0.01, mu_init=0.1, margin_init=0.0005,
                                       time_limit=budget_s - 10, t_start=t0)
        if c5_fin2 < best_c5:
            best_c5 = c5_fin2
            best_h = build_h_from_q(q_fin2, n_fin)
            best_n = n_fin

    # Final full optimization if time permits
    if time_ok(30):
        h_fin3, c5_fin3 = full_optimize(best_h, len(best_h), max_iter=100,
                                        time_limit=budget_s - 10, t_start=t0)
        if c5_fin3 < best_c5:
            best_c5 = c5_fin3
            best_h = h_fin3.copy()

    # Final validation and repair
    best_h = np.clip(best_h, 0.0, 1.0)
    n_final = len(best_h)
    target = n_final / 2.0
    diff = target - best_h.sum()
    if abs(diff) > 1e-10:
        mask = (best_h > 1e-12) & (best_h < 1.0 - 1e-12)
        if mask.any():
            best_h[mask] += diff / mask.sum()
            best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = eval_c5(best_h)

    return best_h, best_c5, len(best_h)
