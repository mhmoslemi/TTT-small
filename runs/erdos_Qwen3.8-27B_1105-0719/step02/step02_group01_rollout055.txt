def c5_of_h(h, dx):
    corrs = np.correlate(h, 1.0 - h, mode='full')

def grad_corr_k(h, k, n):
    offset = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -offset)
    i_hi = min(n - 1, n - 1 - offset)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx + offset] += (1.0 - h[idx])
    grad[idx] += (-h[idx + offset])
    return grad

        h_pol, c5_pol = active_set_lp_polish(h_pert, dx_b, ts_b,
                                             max_iters=150, time_limit=min(50, rem() - 15),
                                             t0=t0, n_active_start=40, alpha_start=0.12)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol

    # Phase 4: Discrete SA on small grids
    sa_configs = [(64, 8, 40000, 0.004), (80, 6, 30000, 0.005), (96, 8, 25000, 0.004)]
    for n_sa, M_sa, n_it, tau0 in sa_configs:
        if rem() < 70:
            break
        h_sa, c5_sa = discrete_sa_safe(n_sa, M_sa, n_it, tau0, rng, t0, time_limit=min(70, rem() - 20))
        if c5_sa < best_c5 + 0.01:
            h_trans = transfer_grid(h_sa, n_parent)
            h_pol, c5_pol = active_set_lp_polish(h_trans, dx_b, ts_b,
                                                 max_iters=150, time_limit=min(50, rem() - 15),
                                                 t0=t0)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol

    # Phase 5: Try nearby n values
    n_b = len(best_h)
    for dn in [-4, -2, 2, 4, 6, -6]:
        if rem() < 50:
            break
        n_try = n_b + dn
        if n_try < 30 or n_try > 200:
            continue
        h_interp = transfer_grid(best_h, n_try)
        dx_try = 2.0 / n_try
        h_pol, c5_pol = active_set_lp_polish(h_interp, dx_try, n_try / 2.0,
                                             max_iters=120, time_limit=min(45, rem() - 15),
                                             t0=t0, n_active_start=40, alpha_start=0.12)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol

    # Phase 6: Diverse starts + LP
    for trial in range(8):
        if rem() < 45:
            break
        if trial % 3 == 0:
            h_init = make_periodic_start(n_b, 5, 2)
        elif trial % 3 == 1:
            h_init = make_binary_start(n_b, rng)
        else:
            h_init = make_periodic_start(n_b, 7, 3)
        h_pol, c5_pol = active_set_lp_polish(h_init, dx_b, ts_b,
                                             max_iters=100, time_limit=min(35, rem() - 12),
                                             t0=t0, n_active_start=40, alpha_start=0.2)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol

    # Phase 7: Final intensive LP
    if rem() > 25:
        n_b = len(best_h)
        dx_b = 2.0 / n_b
        ts_b = n_b / 2.0
        h_final, c5_final = active_set_lp_polish(best_h, dx_b, ts_b,
                                                  max_iters=300, time_limit=max(10, rem() - 10),
                                                  t0=t0, n_active_start=30, alpha_start=0.05)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final

    # One more tight LP
    if rem() > 15:
        n_b = len(best_h)
        dx_b = 2.0 / n_b
        ts_b = n_b / 2.0
        h_final2, c5_f2 = active_set_lp_polish(best_h, dx_b, ts_b,
                                                max_iters=200, time_limit=max(5, rem() - 5),
                                                t0=t0, n_active_start=25, alpha_start=0.03)
        if c5_f2 < best_c5:
            best_c5 = c5_f2
            best_h = h_final2

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final / 2.0)
    best_c5 = c5_of_h(best_h, 2.0 / n_final)

    if not np.all(np.isfinite(best_h)) or best_c5 > 1.0:
        best_h = np.full(n_parent, 0.5)
        best_c5 = c5_of_h(best_h, 2.0 / n_parent)
        n_final = n_parent

    return best_h, best_c5, n_final
