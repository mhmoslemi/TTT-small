```python
def project_onto_simplex_box(y, target):
    """Project y onto {x : 0 <= x <= 1, sum(x) = target}."""
    # Binary search for lambda
    lo = y.min() - 1
    hi = y.max()
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.clip(y - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) / 2, 0, 1)
    # Fix any remaining error
    diff = target - x.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            order = np.argsort(1 - x)
        else:
            order = np.argsort(x)
        remaining = diff
        for i in order:
            if diff > 0:
                add = min(remaining, 1 - x[i])
            else:
                add = max(remaining, -x[i])
            x[i] += add
            remaining -= add
            if abs(remaining) < 1e-14:
                break
    return x
```

```python
def active_gradient_descent(h, max_iters=1000):
    n = len(h)
    dx = 2.0 / n
    best_c = exact_score(h)
    alpha = 0.01
    momentum = 0.5
    velocity = np.zeros(n)
    
    for it in range(max_iters):
        profile = compute_profile(h)
        c = np.max(profile)
        if c < best_c:
            best_c = c
        
        # Select active set
        spread = c - np.median(profile)
        beam = max(0.003, min(0.01, 0.15 * spread))
        active = np.where(profile >= c - beam)[0]
        if len(active) == 0:
            active = np.array([np.argmax(profile)])
        # Limit active set size
        if len(active) > 20:
            top_idx = np.argsort(profile[active])[-20:]
            active = active[top_idx]
        
        # Compute weighted gradient
        thresh = c - beam
        weights = profile[active] - thresh
        weights = np.maximum(weights, 0.01)
        weights /= weights.sum()
        
        grad = np.zeros(n)
        for m in active:
            grad += weights[np.where(active == m)[0][0]] * gradient_profile_element(m, h, n, dx)
        
        # Remove mean (to be sum-neutral)
        grad -= grad.mean()
        
        norm = np.linalg.norm(grad)
        if norm < 1e-14:
            break
        grad /= norm
        
        # Momentum
        velocity = momentum * velocity + grad
        vnorm = np.linalg.norm(velocity)
        if vnorm > 0:
            velocity /= vnorm
        
        # Line search
        improved = False
        for a in [alpha, alpha*0.5, alpha*0.25, alpha*0.1, alpha*0.05]:
            h_try = project_onto_simplex_box(h - a * velocity, n/2.0)
            # Also enforce trust region
            h_try = np.clip(h_try, max(0, h - 0.05), min(1, h + 0.05))
            h_try = project_onto_simplex_box(h_try, n/2.0)
            s = exact_score(h_try)
            if s < best_c - 1e-12:
                h = h_try
                best_c = s
                improved = True
                alpha = min(alpha * 1.1, 0.05)
                break
        if not improved:
            alpha *= 0.5
            if alpha < 1e-7:
                break
    
    return h, best_c
```

```python
def lp_trust_region_step(h, delta=0.02):
    """Solve LP trust-region subproblem."""
    n = len(h)
    dx = 2.0 / n
    profile = compute_profile(h)
    c = np.max(profile)
    
    # Active set
    beam = max(0.005, 0.02 * (c - np.median(profile)))
    active = np.where(profile >= c - beam)[0]
    if len(active) > 15:
        top_idx = np.argsort(profile[active])[-15:]
        active = active[top_idx]
    
    # Build LP
    # Variables: x (n vars), t (1 var)
    # Minimize t
    # Subject to:
    #   sum(x) = n/2
    #   0 <= x <= 1
    #   |x - h| <= delta (componentwise)
    #   t >= profile[m] + grad_m^T (x - h) for m in active
    
    # In standard LP form: min c^T z s.t. A_ub z <= b_ub, A_eq z = b_eq, bounds
    # z = [x, t], dimension n+1
    
    try:
        from scipy.optimize import linprog
        
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0  # minimize t
        
        constraints_ub = []
        b_ub = []
        
        for m in active:
            g = gradient_profile_element(m, h, n, dx)
            # t - g^T x >= profile[m] - g^T h
            # -(t - g^T x) <= -(profile[m] - g^T h)
            # -g^T x - t <= -(profile[m] - g^T h)
            row = np.zeros(n + 1)
            row[:n] = -g
            row[-1] = -1.0
            constraints_ub.append(row)
            b_ub.append(-(profile[m] - g @ h))
        
        A_ub = np.array(constraints_ub)
        b_ub = np.array(b_ub)
        
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([n / 2.0])
        
        # Bounds: max(0, h_i - delta) <= x_i <= min(1, h_i + delta), t >= 0
        lb = np.maximum(0, h - delta)
        ub = np.minimum(1, h + delta)
        bounds = list(zip(lb, ub))
        bounds.append((0, None))
        
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        
        if res.status == 0:
            x_new = res.x[:n]
            x_new = project_onto_simplex_box(x_new, n/2.0)
            s = exact_score(x_new)
            if s < c - 1e-12:
                return x_new
    except Exception:
        pass
    return None
```

One more thought: the `project_onto_simplex_box` function with bisection might not be perfectly accurate. Let me use a more robust approach: sort the values and find the threshold analytically.

The problem is: find x = clip(y - λ, 0, 1) such that sum(x) = target.

The function f(λ) = sum(clip(y_i - λ, 0, 1)) is piecewise linear and decreasing in λ. We need f(λ) = target.

The breakpoints are at λ = y_i and λ = y_i - 1 for each i. Between breakpoints, f is linear with slope = -(# of i where 0 < y_i - λ < 1).

So we can:
1. Sort the y_i values
2. For each candidate λ (sorted y_i and y_i - 1), compute f(λ)
3. Find the bracketing interval
4. Solve linearly within that interval

This is O(n log n) and exact (up to floating point).

Let me implement this:

```python
                sub = min(remaining, x[idx])
                x[idx] -= sub
                remaining -= sub
                if remaining <= 1e-14:
                    break
    return x

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

def active_gradient_descent(h, max_iters=500, momentum=0.4, max_alpha=0.05, min_alpha=1e-7):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best_c = exact_score(h)
    alpha = 0.01
    velocity = np.zeros(n)
    stagnation = 0

    for it in range(max_iters):
        profile = compute_profile(h)
        c = np.max(profile)
        if c < best_c:
            best_c = c
            stagnation = 0
        else:
            stagnation += 1
        if stagnation > 50:
            break

        spread = max(c - np.median(profile), 1e-6)
        beam = max(0.002, min(0.015, 0.12 * spread))
        active = np.where(profile >= c - beam)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(profile))])
        if len(active) > 25:
            top_idx = np.argsort(profile[active])[-25:]
            active = active[top_idx]

        thresh = c - beam
        weights = profile[active] - thresh
        weights = np.maximum(weights, 0.005)
        weights /= weights.sum()

        grad = np.zeros(n)
        for i, m in enumerate(active):
            grad += weights[i] * gradient_profile_element(m, h, n, dx)
        grad -= grad.mean()

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad /= gnorm

        velocity = momentum * velocity + grad
        vnorm = np.linalg.norm(velocity)
        if vnorm > 1e-14:
            velocity /= vnorm

        improved = False
        for a in [alpha, alpha * 0.5, alpha * 0.25, alpha * 0.1, alpha * 0.05]:
            h_try = h - a * velocity
            h_try = project_onto_simplex_box(h_try, n / 2.0)
            s = exact_score(h_try)
            if s < best_c - 1e-13:
                h = h_try
                best_c = s
                improved = True
                alpha = min(alpha * 1.15, max_alpha)
                break
        if not improved:
            alpha *= 0.6
            if alpha < min_alpha:
                break

    return h, best_c

def lp_trust_region_step(h, delta=0.03, max_active=12):
    n = len(h)
    dx = 2.0 / n
    profile = compute_profile(h)
    c = np.max(profile)
    spread = max(c - np.median(profile), 1e-6)
    beam = max(0.005, 0.03 * spread)
    active = np.where(profile >= c - beam)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(profile))])
    if len(active) > max_active:
        top_idx = np.argsort(profile[active])[-max_active:]
        active = active[top_idx]

    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0

    constraints_ub = []
    b_ub = []
    for m in active:
        g = gradient_profile_element(m, h, n, dx)
        pm = profile[m]
        row = np.zeros(n + 1)
        row[:n] = -g
        row[-1] = -1.0
        constraints_ub.append(row)
        b_ub.append(-(pm - g @ h))

    A_ub = np.array(constraints_ub)
    b_ub = np.array(b_ub)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])

    lb = np.maximum(0.0, h - delta)
    ub = np.minimum(1.0, h + delta)
    bounds = list(zip(lb, ub))
    bounds.append((0.0, None))

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            x_new = res.x[:n]
            x_new = project_onto_simplex_box(x_new, n / 2.0)
            s = exact_score(x_new)
            if s < c - 1e-12:
                return x_new
    except Exception:
        pass
    return None

def symmetrize(h):
    return (h + h[::-1]) / 2.0

def try_slsqp_polish(h, maxiter=200):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    c0 = exact_score(h)

    try:
        x0 = np.concatenate([h, [c0 + 1e-4]])
        bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 1.0)]

        def obj(x):
            return x[-1]

        def obj_jac(x):
            g = np.zeros(n + 1)
            g[-1] = 1.0
            return g

        def eq_fun(x):
            return x[:-1].sum() - n / 2.0

        def eq_jac(x):
            g = np.zeros(n + 1)
            g[:-1] = 1.0
            return g

        def ineq_fun(x):
            t = x[-1]
            prof = np.correlate(x[:-1], 1.0 - x[:-1], mode="full") * dx
            return t - prof

        def ineq_jac(x):
            h_x = x[:-1]
            J = np.zeros((M, n + 1))
            for m in range(M):
                J[m, :n] = -gradient_profile_element(m, h_x, n, dx)
                J[m, -1] = 1.0
            return J

        cons = [
            {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
            {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
        ]
        from scipy.optimize import minimize
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None and np.isfinite(res.x).all():
            h_new = repair_feasible(res.x[:-1])
            c_new = exact_score(h_new)
            if c_new < c0 - 1e-10:
                return h_new
    except Exception:
        pass
    return None

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s - (time.perf_counter() - t_start)

    parent = None
    try:
        parent = np.array(initial_h_values, dtype=np.float64)
    except NameError:
        pass
    except Exception:
        pass

    if parent is None or len(parent) < 4:
        parent = np.full(70, 0.5)

    parent = np.clip(parent, 0.0, 1.0)
    parent = repair_feasible(parent)

    best_h = parent.copy()
    best_c = exact_score(parent)

    h_sym = repair_feasible(symmetrize(parent))
    c_sym = exact_score(h_sym)
    if c_sym < best_c - 1e-12:
        best_h = h_sym
        best_c = c_sym

    # Phase 1: Dimension sweep
    if time_left() > 200:
        n_parent = len(best_h)
        dims = []
        for d in range(36, 130, 2):
            if d != n_parent:
                dims.append(d)
        # Prioritize dimensions close to parent
        dims.sort(key=lambda d: abs(d - n_parent))
        dims = dims[:20]

        for new_n in dims:
            if time_left() < 60:
                break
            h_interp = interpolate_to_n(best_h, new_n)
            c_interp = exact_score(h_interp)
            h_ref, c_ref = active_gradient_descent(h_interp, max_iters=80, momentum=0.4)
            if c_ref < best_c - 1e-10:
                best_h = h_ref
                best_c = c_ref

    # Phase 2: Intensive alternating optimization
    if time_left() > 100:
        n = len(best_h)
        for round_num in range(20):
            if time_left() < 30:
                break
            # Gradient descent
            h_gd, c_gd = active_gradient_descent(best_h, max_iters=200,
                                                 momentum=0.3 + 0.02 * round_num % 5,
                                                 max_alpha=0.04)
            if c_gd < best_c - 1e-12:
                best_h = h_gd
                best_c = c_gd

            # LP step
            if time_left() > 30:
                h_lp = lp_trust_region_step(best_h, delta=0.02 + 0.005 * (round_num % 4))
                if h_lp is not None:
                    c_lp = exact_score(h_lp)
                    if c_lp < best_c - 1e-12:
                        best_h = h_lp
                        best_c = c_lp
                        h_gd2, c_gd2 = active_gradient_descent(best_h, max_iters=100, momentum=0.4)
                        if c_gd2 < best_c - 1e-12:
                            best_h = h_gd2
                            best_c = c_gd2

            # Try symmetrization periodically
            if round_num % 4 == 3:
                h_s = repair_feasible(symmetrize(best_h))
                c_s = exact_score(h_s)
                if c_s < best_c - 1e-12:
                    best_h = h_s
                    best_c = c_s

    # Phase 3: SLSQP polish
    if time_left() > 60:
        h_sol = try_slsqp_polish(best_h, maxiter=300)
        if h_sol is not None:
            c_sol = exact_score(h_sol)
            if c_sol < best_c - 1e-12:
                best_h = h_sol
                best_c = c_sol
                h_gd, c_gd = active_gradient_descent(best_h, max_iters=200, momentum=0.3)
                if c_gd < best_c - 1e-12:
                    best_h = h_gd
                    best_c = c_gd

    # Phase 4: Fine-tuned gradient descent with very small steps
    if time_left() > 40:
        n = len(best_h)
        dx = 2.0 / n
        for _ in range(200):
            if time_left() < 5:
                break
            profile = compute_profile(best_h)
            c = np.max(profile)
            spread = max(c - np.median(profile), 1e-6)
            beam = max(0.001, 0.05 * spread)
            active = np.where(profile >= c - beam)[0]
            if len(active) == 0:
                active = np.array([int(np.argmax(profile))])
            if len(active) > 30:
                top_idx = np.argsort(profile[active])[-30:]
                active = active[top_idx]
            weights = profile[active] - (c - beam)
            weights = np.maximum(weights, 0.001)
            weights /= weights.sum()
            grad = np.zeros(n)
            for i, m in enumerate(active):
                grad += weights[i] * gradient_profile_element(m, best_h, n, dx)
            grad -= grad.mean()
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break
            grad /= gnorm
            improved = False
            for a in [0.002, 0.001, 0.0005, 0.0002, 0.0001]:
                h_try = project_onto_simplex_box(best_h - a * grad, n / 2.0)
                s = exact_score(h_try)
                if s < best_c - 1e-14:
                    best_h = h_try
                    best_c = s
                    improved = True
                    break
            if not improved:
                break

    # Phase 5: Try different dimensions from the improved solution
    if time_left() > 60:
        n_best = len(best_h)
        for offset in [-4, -2, 2, 4, -6, 6, 8, -8]:
            if time_left() < 20:
                break
            new_n = n_best + offset
            if new_n < 24 or new_n > 160:
                continue
            h_interp = interpolate_to_n(best_h, new_n)
            h_ref, c_ref = active_gradient_descent(h_interp, max_iters=150, momentum=0.35)
            if c_ref < best_c - 1e-12:
                best_h = h_ref
                best_c = c_ref

    # Phase 6: Final SLSQP + gradient polish
    if time_left() > 40:
        h_sol = try_slsqp_polish(best_h, maxiter=400)
        if h_sol is not None:
            c_sol = exact_score(h_sol)
            if c_sol < best_c - 1e-12:
                best_h = h_sol
                best_c = c_sol

    if time_left() > 20:
        h_gd, c_gd = active_gradient_descent(best_h, max_iters=300, momentum=0.2, max_alpha=0.02)
        if c_gd < best_c - 1e-13:
            best_h = h_gd
            best_c = c_gd

    # Final validation
    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)

    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            final_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 2:
            final_c = float(eval_result[0])
    except Exception:
        pass

    return best_h, final_c, len(best_h)
```
