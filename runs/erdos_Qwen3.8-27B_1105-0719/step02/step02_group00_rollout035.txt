```python
def lp_direction(h, G_active, alpha_tol=1e-10):
    """Find best descent direction via LP.
    h: current point (n,)
    G_active: gradients of active lags (k, n)
    Returns: (d, alpha) or (None, 0) on failure
    """
    n = len(h)
    k = G_active.shape[0]
    
    # Variables: u (n), v (n), alpha (1) => 2n+1 total
    # Minimize: -alpha => c = [0,...,0, -1]
    c = np.zeros(2 * n + 1)
    c[-1] = -1.0
    
    # Inequality constraints: A_ub @ x <= b_ub
    # 1. For each active lag m: G[m,:]·(u-v) + alpha <= 0
    #    => row: [G[m,:], -G[m,:], 1] @ x <= 0
    rows = []
    rhs = []
    for m in range(k):
        row = np.concatenate([G_active[m], -G_active[m], [1.0]])
        rows.append(row)
        rhs.append(0.0)
    
    # 2. L1 normalization: sum(u) + sum(v) <= 1
    #    => row: [1,...,1, 1,...,1, 0] @ x <= 1
    l1_row = np.ones(2 * n + 1)
    l1_row[-1] = 0.0
    rows.append(l1_row)
    rhs.append(1.0)
    
    # 3. Box constraints: u_i <= (1-h_i), v_i <= h_i
    #    => for each i: u_i <= 1-h_i => row with 1 at position i, rhs = 1-h_i
    #    => for each i: v_i <= h_i => row with 1 at position n+i, rhs = h_i
    for i in range(n):
        row_u = np.zeros(2 * n + 1)
        row_u[i] = 1.0
        rows.append(row_u)
        rhs.append(1.0 - h[i])
        
        row_v = np.zeros(2 * n + 1)
        row_v[n + i] = 1.0
        rows.append(row_v)
        rhs.append(h[i])
    
    A_ub = np.array(rows)
    b_ub = np.array(rhs)
    
    # Equality: sum(u) - sum(v) = 0
    A_eq = np.zeros((1, 2 * n + 1))
    A_eq[0, :n] = 1.0
    A_eq[0, n:2*n] = -1.0
    b_eq = np.array([0.0])
    
    # Bounds: all variables >= 0
    bounds = [(0, None)] * (2 * n + 1)
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.fun < -alpha_tol:
            alpha = -res.fun
            u = res.x[:n]
            v = res.x[n:2*n]
            d = u - v
            # Normalize for stability
            d_norm = np.abs(d).sum()
            if d_norm > 1e-12:
                d = d / d_norm
            return d, alpha
    except Exception:
        pass
    return None, 0.0
```

```python
    n = len(h)
```

```python
import numpy as np
```

```python
import time


def score_exact(h):
    n = len(h)
    if n < 2:
        return float('inf')
    dx = 2.0 / n
    raw = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(raw * dx))


def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx


def build_gradient_matrix(h):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    G = np.zeros((M, n))
    G[n - 1, :] = dx * (1.0 - 2.0 * h)
    for d in range(1, n):
        m = n - 1 + d
        j = np.arange(n - d)
        G[m, j] += -dx * h[j + d]
        G[m, j + d] += dx * (1.0 - h[j])
        m2 = n - 1 - d
        G[m2, j] += dx * (1.0 - h[j + d])
        G[m2, j + d] += -dx * h[j]
    return G


def repair_h(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx])
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-13:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-13:
                break
    # Final fix
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2

    return h


def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    # Piecewise constant interpolation
    dx_old = 2.0 / n
    edges_old = np.linspace(0.0, 2.0, n + 1)
    dx_new = 2.0 / new_n
    edges_new = np.linspace(0.0, 2.0, new_n + 1)
    h_new = np.zeros(new_n)
    for j in range(new_n):
        xl, xr = edges_new[j], edges_new[j + 1]
        # Find overlapping old intervals
        k_start = int(np.ceil((xl - 1e-15) / dx_old))
        k_end = int(np.floor((xr + 1e-15) / dx_old))
        k_start = max(0, k_start)
        k_end = min(n - 1, k_end)
        if k_end < k_start:
            h_new[j] = 0.0
            continue
        overlap = 0.0
        for k in range(k_start, k_end + 1):
            ol = max(xl, edges_old[k])
            orr = min(xr, edges_old[k + 1])
            if orr > ol:
                overlap += h[k] * (orr - ol)
        h_new[j] = overlap / dx_new
    return repair_h(np.clip(h_new, 0.0, 1.0))


def symmetrize(h):
    return repair_h((h + h[::-1]) / 2.0)


def lp_find_direction(h, G, active_idx):
    """Solve LP to find best descent direction.
    Returns (d, alpha) where d is the direction (||d||_1 normalized) or (None, 0).
    """
    n = len(h)
    k = len(active_idx)
    if k == 0:
        return None, 0.0
    
    Ga = G[active_idx, :]  # (k, n)
    
    # Variables: u (n), v (n), alpha (1) => 2n+1
    nv = 2 * n + 1
    c_obj = np.zeros(nv)
    c_obj[-1] = -1.0  # maximize alpha
    
    # Build A_ub, b_ub
    # Active constraints: Ga[m] @ (u-v) + alpha <= 0
    A_rows = np.zeros((k, nv))
    A_rows[:, :n] = Ga
    A_rows[:, n:2*n] = -Ga
    A_rows[:, -1] = 1.0
    b_rows = np.zeros(k)
    
    # L1 norm: sum(u) + sum(v) <= 1
    l1_row = np.zeros(nv)
    l1_row[:2*n] = 1.0
    l1_row = l1_row.reshape(1, -1)
    
    # Box: u_i <= 1-h_i, v_i <= h_i
    box_rows = np.zeros((2*n, nv))
    box_rhs = np.zeros(2*n)
    for i in range(n):
        box_rows[i, i] = 1.0
        box_rhs[i] = 1.0 - h[i]
        box_rows[n+i, n+i] = 1.0
        box_rhs[n+i] = h[i]
    
    A_ub = np.vstack([A_rows, l1_row, box_rows])
    b_ub = np.concatenate([b_rows, [1.0], box_rhs])
    
    # Equality: sum(u) - sum(v) = 0
    A_eq = np.zeros((1, nv))
    A_eq[0, :n] = 1.0
    A_eq[0, n:2*n] = -1.0
    b_eq = np.array([0.0])
    
    bounds = [(0, None)] * nv
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'time_limit': 2.0})
        if res.success:
            alpha = max(0.0, -res.fun)
            if alpha < 1e-12:
                return None, 0.0
            u = res.x[:n]
            v = res.x[n:2*n]
            d = u - v
            l1 = np.abs(d).sum()
            if l1 < 1e-12:
                return None, 0.0
            d = d / l1
            return d, alpha
    except Exception:
        pass
    return None, 0.0


def weighted_grad_direction(h, G, profile, active_idx):
    """Fallback: softmax-weighted gradient of active lags."""
    c = profile[active_idx]
    tau = max(0.005, (c.max() - c.min()) * 0.5)
    w = np.exp((c - c.max()) / tau)
    w = w / w.sum()
    grad = w @ G[active_idx, :]
    grad -= grad.mean()
    norm = np.linalg.norm(grad)
    if norm < 1e-14:
        return None
    return grad / norm
```

```python
    if n < 2:
        return float('inf')
    dx = 2.0 / n
    raw = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(raw * dx))


def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx


def build_gradient_matrix(h):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    G = np.zeros((M, n))
    G[n - 1, :] = dx * (1.0 - 2.0 * h)
    for d in range(1, n):
        j = np.arange(n - d)
        m_pos = n - 1 + d
        G[m_pos, j] += -dx * h[j + d]
        G[m_pos, j + d] += dx * (1.0 - h[j])
        m_neg = n - 1 - d
        G[m_neg, j] += dx * (1.0 - h[j + d])
        G[m_neg, j + d] += -dx * h[j]
    return G


def repair_h(vec):
    vec = np.asarray(vec, dtype=np.float64)
    n = len(vec)
    if n == 0:
        return np.array([0.5])
    if not np.all(np.isfinite(vec)):
        return np.full(n, 0.5)
    h = np.clip(vec, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx])
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-13:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-13:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h


def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    if new_n < 4 or new_n > 200:
        return h.copy()
    dx_old = 2.0 / n
    edges_old = np.linspace(0.0, 2.0, n + 1)
    dx_new = 2.0 / new_n
    edges_new = np.linspace(0.0, 2.0, new_n + 1)
    h_new = np.zeros(new_n)
    for j in range(new_n):
        xl = edges_new[j]
        xr = edges_new[j + 1]
        k_start = max(0, int(np.floor(xl / dx_old + 1e-12)))
        k_end = min(n - 1, int(np.ceil((xr - 1e-12) / dx_old - 1 - 1e-12)))
        k_end = max(k_start, k_end)
        if k_end < k_start:
            h_new[j] = 0.5
            continue
        total = 0.0
        for k in range(k_start, k_end + 1):
            ol = max(xl, edges_old[k])
            orr = min(xr, edges_old[k + 1])
            if orr > ol:
                total += h[k] * (orr - ol)
        h_new[j] = total / dx_new
    return repair_h(np.clip(h_new, 0.0, 1.0))


def symmetrize(h):
    return repair_h((h + h[::-1]) / 2.0)


def lp_find_direction(h, G, active_idx):
    n = len(h)
    k = len(active_idx)
    if k == 0:
        return None, 0.0
    Ga = G[active_idx, :]
    nv = 2 * n + 1
    c_obj = np.zeros(nv)
    c_obj[-1] = -1.0
    A_parts = [Ga.copy()]
    A_parts[-1][:, 0:n] = Ga
    A_parts[-1][:, n:2*n] = -Ga
    A_parts[-1][:, -1] = 1.0
    b_parts = [np.zeros(k)]
    l1_row = np.zeros((1, nv))
    l1_row[0, :2*n] = 1.0
    A_parts.append(l1_row)
    b_parts.append(np.array([1.0]))
    box_rows = np.zeros((2 * n, nv))
    box_rhs = np.zeros(2 * n)
    for i in range(n):
        box_rows[i, i] = 1.0
        box_rhs[i] = max(0.0, 1.0 - h[i])
        box_rows[n + i, n + i] = 1.0
        box_rhs[n + i] = max(0.0, h[i])
    A_parts.append(box_rows)
    b_parts.append(box_rhs)
    A_ub = np.vstack(A_parts)
    b_ub = np.concatenate(b_parts)
    A_eq = np.zeros((1, nv))
    A_eq[0, :n] = 1.0
    A_eq[0, n:2*n] = -1.0
    b_eq = np.array([0.0])
    bounds = [(0.0, None)] * nv
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'time_limit': 3.0})
        if res.success:
            alpha = max(0.0, -res.fun)
            if alpha < 1e-11:
                return None, 0.0
            u = res.x[:n]
            v = res.x[n:2*n]
            d = u - v
            l1 = np.abs(d).sum()
            if l1 < 1e-12:
                return None, 0.0
            d = d / l1
            return d, alpha
    except Exception:
        pass
    return None, 0.0


def weighted_grad_direction(h, G, profile, active_idx):
    c = profile[active_idx]
    tau = max(0.004, (c.max() - c.min()) * 0.5)
    w = np.exp((c - c.max()) / tau)
    w = w / w.sum()
    grad = w @ G[active_idx, :]
    grad -= grad.mean()
    norm = np.linalg.norm(grad)
    if norm < 1e-14:
        return None
    return grad / norm


def line_search(h, d, current_score):
    best_h = h.copy()
    best_s = current_score
    best_t = 0.0
    n_steps = 14
    steps = 1.0 / (1.0 + 0.35 * np.arange(n_steps))
    for t in steps:
        h_try = h + t * d
        h_try = repair_h(h_try)
        s = score_exact(h_try)
        if s < best_s - 1e-15:
            best_s = s
            best_h = h_try
            best_t = t
    return best_h, best_s, best_t


def local_optimize(h, time_limit):
    n = len(h)
    h = h.copy()
    score = score_exact(h)
    best_h = h.copy()
    best_score = score
    stall = 0
    beam = 0.007
    t0 = time.perf_counter()
    while time.perf_counter() - t0 < time_limit:
        profile = compute_profile(h)
        c = float(np.max(profile))
        thresh = c - beam
        active = np.where(profile >= thresh)[0]
        if len(active) > 30:
            top = np.argsort(profile[active])[-30:]
            active = active[top]
        am = int(np.argmax(profile))
        if am not in active:
            active = np.append(active, am)
        G = build_gradient_matrix(h)
        improved = False
        d, alpha = lp_find_direction(h, G, active)
        if d is not None and alpha > 5e-12:
            h_new, s_new, t = line_search(h, d, score)
            if s_new < score - 1e-15:
                h = h_new
                score = s_new
                improved = True
                stall = 0
                beam = max(0.002, beam * 0.92)
        if not improved:
            d2 = weighted_grad_direction(h, G, profile, active)
            if d2 is not None:
                h_new, s_new, t = line_search(h, d2, score)
                if s_new < score - 1e-15:
                    h = h_new
                    score = s_new
                    improved = True
                    stall = 0
        if not improved:
            stall += 1
            beam = min(0.035, beam * 1.4)
            if stall >= 25:
                break
        if score < best_score:
            best_score = score
            best_h = h.copy()
    return best_h, best_score


def apply_perturbation(h, rng, eps):
    z = rng.randn(len(h))
    z -= z.mean()
    return repair_h(h + eps * z)


def try_dimension(h, new_n, time_limit):
    h_interp = interpolate_to_n(h, new_n)
    h_opt, c_opt = local_optimize(h_interp, time_limit)
    return h_opt, c_opt


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()
    rng = np.random.RandomState(seed)

    def elapsed():
        return time.perf_counter() - t_start

    def tl():
        return budget_s - elapsed()

    try:
        parent = np.array(initial_h_values, dtype=np.float64)
    except Exception:
        parent = np.full(70, 0.5)

    if parent is None or len(parent) < 4 or not np.all(np.isfinite(parent)):
        parent = np.full(70, 0.5)

    parent = repair_h(parent)
    best_h = parent.copy()
    best_c = score_exact(parent)

    h_sym = symmetrize(parent)
    c_sym = score_exact(h_sym)
    if c_sym < best_c - 1e-13:
        best_h = h_sym
        best_c = c_sym

    # Phase 1: Deep local optimization
    if tl() > 120:
        h_opt, c_opt = local_optimize(best_h, tl() - 100)
        if c_opt < best_c - 1e-14:
            best_h = h_opt
            best_c = c_opt

    # Phase 2: Dimension exploration
    if tl() > 80:
        n_best = len(best_h)
        ratios = [0.85, 0.92, 1.08, 1.15, 1.25, 1.4]
        dims = set()
        for r in ratios:
            d = int(round(n_best * r))
            d = max(30, min(d, 150))
            if d % 2 == 1:
                d += 1
            dims.add(d)
        dims_sorted = sorted(dims, key=lambda x: abs(x - n_best))
        for new_n in dims_sorted:
            if tl() < 40:
                break
            if new_n == n_best:
                continue
            h_opt, c_opt = try_dimension(best_h, new_n, min(35, tl() - 25))
            if c_opt < best_c - 1e-14:
                best_h = h_opt
                best_c = c_opt
                n_best = new_n

    # Phase 3: More local optimization on best
    if tl() > 60:
        h_opt, c_opt = local_optimize(best_h, tl() - 40)
        if c_opt < best_c - 1e-14:
            best_h = h_opt
            best_c = c_opt

    # Phase 4: Perturbation-based escape
    if tl() > 50:
        for eps in [0.008, 0.015, 0.025, 0.04, 0.06]:
            if tl() < 20:
                break
            h_pert = apply_perturbation(best_h, rng, eps)
            h_opt, c_opt = local_optimize(h_pert, min(18, tl() - 15))
            if c_opt < best_c - 1e-14:
                best_h = h_opt
                best_c = c_opt

    # Phase 5: Symmetry exploitation
    if tl() > 30:
        h_sym2 = symmetrize(best_h)
        c_sym2 = score_exact(h_sym2)
        if c_sym2 < best_c - 1e-14:
            best_h = h_sym2
            best_c = c_sym2
        h_opt, c_opt = local_optimize(best_h, tl() - 10)
        if c_opt < best_c - 1e-14:
            best_h = h_opt
            best_c = c_opt

    # Phase 6: More aggressive perturbations if still time
    if tl() > 30:
        for _ in range(4):
            if tl() < 10:
                break
            for eps in [0.03, 0.05, 0.08]:
                if tl() < 8:
                    break
                h_pert = apply_perturbation(best_h, rng, eps)
                h_opt, c_opt = local_optimize(h_pert, min(12, tl() - 6))
                if c_opt < best_c - 1e-14:
                    best_h = h_opt
                    best_c = c_opt

    # Phase 7: Final deep polish
    if tl() > 25:
        h_opt, c_opt = local_optimize(best_h, tl() - 5)
        if c_opt < best_c - 1e-14:
            best_h = h_opt
            best_c = c_opt

    # Final validation
    best_h = repair_h(best_h)
    final_c = score_exact(best_h)

    try:
        eval_res = evaluate_erdos_solution(best_h)
        if isinstance(eval_res, (float, int)) and np.isfinite(eval_res):
            if eval_res < final_c:
                final_c = float(eval_res)
        elif isinstance(eval_res, tuple) and len(eval_res) >= 1:
            if isinstance(eval_res[0], (float, int)) and np.isfinite(eval_res[0]):
                if eval_res[0] < final_c:
                    final_c = float(eval_res[0])
    except Exception:
        pass

    return best_h, float(final_c), int(len(best_h))
```
