import numpy as np
from scipy.optimize import linprog
import time

def eval_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def corr_raw(h):
    return np.correlate(h, 1.0 - h, mode='full')

def corr_grad_k(h, k):
    """Gradient of correlation lag k with respect to h."""
    n = len(h)
    d = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -d)
    i_hi = min(n - 1, n - 1 - d)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx] += (1.0 - h[idx + d])
    grad[idx + d] -= h[idx]
    return grad

def build_sym_h(q):
    m = len(q)
    h = np.empty(2 * m)
    h[:m] = q
    h[m:] = 1.0 - q[::-1]
    return h

def sym_from_h(h):
    n = len(h)
    m = n // 2
    q = np.empty(m)
    for i in range(m):
        q[i] = 0.5 * (h[i] + (1.0 - h[n - 1 - i]))
    return np.clip(q, 0.0, 1.0)

def project_feasible(x, target_sum):
    n = len(x)
    x = np.asarray(x, dtype=float).copy()
    x[~np.isfinite(x)] = 0.5
    x = np.clip(x, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(90):
        mid = (lo + hi) / 2.0
        s = np.clip(x + mid, 0.0, 1.0).sum()
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    h = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(h)
            for i in order:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
                if abs(diff) < 1e-13:
                    break
        else:
            order = np.argsort(h)[::-1]
            for i in order:
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub
                if abs(diff) < 1e-13:
                    break
    return np.clip(h, 0.0, 1.0)

def tangent_lp_full(h, active_lags, rho=0.15, max_iter_lp=100):
    """Solve tangent LP for full vector h. Returns new h or None if no improvement."""
    n = len(h)
    R = corr_raw(h)
    t0 = R[active_lags].max()
    
    # Build constraint matrix
    # Variables: z[0..n-1], t => n+1 variables
    n_vars = n + 1
    rows = []
    rhs = []
    
    for ell in active_lags:
        g = corr_grad_k(h, ell)
        b_ell = g @ h - R[ell]
        row = np.append(g, -1.0)
        rows.append(row)
        rhs.append(b_ell)
    
    if not rows:
        return None
    
    A_ub = np.array(rows)
    b_ub = np.array(rhs)
    
    # Objective: minimize t
    c = np.zeros(n_vars)
    c[n] = 1.0
    
    # Equality: sum(z) = n/2
    A_eq = np.zeros((1, n_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    # Bounds
    lb = np.maximum(0.0, h - rho)
    ub = np.minimum(1.0, h + rho)
    lb = np.append(lb, 0.0)
    ub = np.append(ub, t0 + 1.0)
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=list(zip(lb, ub)), method='highs')
        if not res.success:
            return None

        return None
    
    z_new = project_feasible(z_new, n / 2.0)

def pair_transfer_search(h, a, b, n_grid=20):
    """Search over delta for pair transfer h[a]+=delta, h[b]-=delta."""
    n = len(h)
    lower = max(-h[a], h[b] - 1.0)
    upper = min(1.0 - h[a], -h[b])
    if upper - lower < 1e-12:
        return h, eval_score(h)
    
    best_h = h.copy()
    best_score = eval_score(h)
    
    deltas = np.linspace(lower, upper, n_grid)
    best_delta = 0.0
    
    for d in deltas:
        h_try = h.copy()
        h_try[a] += d
        h_try[b] -= d
        if h_try[a] < -1e-10 or h_try[a] > 1.0 + 1e-10 or h_try[b] < -1e-10 or h_try[b] > 1.0 + 1e-10:

            best_h = h_try.copy()

    if best_delta != 0.0:
        span = (upper - lower) / n_grid
        for _ in range(10):
            lo_r = best_delta - span
            hi_r = best_delta + span
            lo_r = max(lo_r, lower)
            hi_r = min(hi_r, upper)

                h_try[a] += d

                if h_try[a] < -1e-10 or h_try[a] > 1.0 + 1e-10:

                    best_score = sc
                    best_h = h_try.copy()
                    best_delta = d
            span /= 3.0
            if span < 1e-8:
                break

def lp_optimize_full(h, time_limit, rho_init=0.12, max_rounds=5000):
    """Tangent LP optimization for full vector."""
    n = len(h)
    dx = 2.0 / n
    best_h = h.copy()
    best_score = eval_score(h)
    rho = rho_init
    
    for rnd in range(max_rounds):
        if time.time() > time_limit:
            break
        
        R = corr_raw(best_h)
        max_R = R.max()
        threshold = max_R - 0.02 * max_R
        active = np.where(R >= threshold)[0]
        
        # Build LP
        n_vars = n + 1
        A_rows = []
        b_vals = []
        
        for ell in active:
            g = corr_grad_k(best_h, ell)
            b_ell = g @ best_h - R[ell]
            row = np.zeros(n_vars)
            row[:n] = g
            row[n] = -1.0
            A_rows.append(row)
            b_vals.append(b_ell)
        
        if not A_rows:
            break
        
        A_ub = np.array(A_rows)
        b_ub = np.array(b_vals)
        
        c = np.zeros(n_vars)
        c[n] = 1.0
        
        A_eq = np.zeros((1, n_vars))
        A_eq[0, :n] = 1.0
        b_eq = np.array([n / 2.0])
        
        lb = np.maximum(0.0, best_h - rho)
        ub = np.minimum(1.0, best_h + rho)
        bounds = list(zip(lb, ub))
        bounds.append((0.0, max_R + 1.0))
        
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if not res.success:
                rho *= 0.7
                continue
            z_new = res.x[:n]
        except Exception:
            rho *= 0.7
            continue
        
        # Line search along direction
        direction = z_new - best_h
        best_try = None
        best_try_score = best_score
        
        for alpha in [1.0, 0.7, 0.5, 0.3, 0.15]:
            h_try = best_h + alpha * direction
            h_try = project_feasible(h_try, n / 2.0)
            sc = eval_score(h_try)
            if sc < best_try_score - 1e-14:
                best_try_score = sc
                best_try = h_try
        
        if best_try is not None:
            best_h = best_try
            best_score = best_try_score
            # Slightly increase rho for convergence
            rho = min(0.25, rho * 1.05)
        else:
            # Try to escape: reduce rho
            rho *= 0.7
            if rho < 0.005:

                active2 = np.where(R >= threshold2)[0]
                if len(active2) <= len(active):
                    break
                # Try with expanded active set
                active = active2
                rho = min(0.2, rho * 2.0)
    
    return best_h, best_score

import numpy as np
from scipy.optimize import linprog
import time

def eval_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def corr_raw(h):
    return np.correlate(h, 1.0 - h, mode='full')

def corr_grad_k(h, k):
    n = len(h)
    d = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -d)
    i_hi = min(n - 1, n - 1 - d)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx] += (1.0 - h[idx + d])
    grad[idx + d] -= h[idx]
    return grad

def build_sym_h(q):
    m = len(q)
    h = np.empty(2 * m)
    h[:m] = q
    h[m:] = 1.0 - q[::-1]
    return h

def sym_from_h(h):
    n = len(h)
    m = n // 2
    q = np.empty(m)
    for i in range(m):
        q[i] = 0.5 * (h[i] + (1.0 - h[n - 1 - i]))
    return np.clip(q, 0.0, 1.0)

def project_feasible(x, target_sum):
    n = len(x)
    x = np.asarray(x, dtype=float).copy()
    x[~np.isfinite(x)] = 0.5
    x = np.clip(x, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(90):
        mid = (lo + hi) / 2.0
        s = np.clip(x + mid, 0.0, 1.0).sum()
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    h = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(h)
            for i in order:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
                if abs(diff) < 1e-13:
                    break
        else:
            order = np.argsort(h)[::-1]
            for i in order:
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub
                if abs(diff) < 1e-13:
                    break
    return np.clip(h, 0.0, 1.0)

def transfer_to_n(h_old, n_new):
    n_old = len(h_old)
    x_old = (np.arange(n_old) + 0.5) * 2.0 / n_old
    x_new = (np.arange(n_new) + 0.5) * 2.0 / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new / 2.0)

    n = 2 * m
    dx = 2.0 / n
    best_q = np.clip(q.copy(), 0, 1)
    best_h = build_sym_h(best_q)
    best_score = eval_score(best_h)
    rho = rho_init
    stall_count = 0

        if time.time() > time_limit:
            break

        R = corr_raw(best_h)
        max_R = R.max()
        threshold = max_R - 0.03 * max_R
        active = np.where(R >= threshold)[0]
        if len(active) > 80:
            top_idx = np.argsort(R)[::-1][:80]
            active = top_idx

        n_vars = m + 1
        A_rows = []
        b_vals = []

            gh = gh_cache[ell]
            if m > 1:

            else:

            b_ell = gq @ best_q - R[ell]
            row = np.zeros(n_vars)

            b_ell = g @ best_h - R[ell]
            row = np.zeros(n_vars)
            row[:n] = g
            row[n] = -1.0
            A_rows.append(row)
            b_vals.append(b_ell)

        if not A_rows:
            break

        A_ub = np.array(A_rows)
        b_ub = np.array(b_vals)
        c = np.zeros(n_vars)
        c[n] = 1.0

        A_eq = np.zeros((1, n_vars))
        A_eq[0, :n] = 1.0
        b_eq = np.array([n / 2.0])

        lb = np.maximum(0.0, best_h - rho)
        ub = np.minimum(1.0, best_h + rho)
        bounds = list(zip(lb, ub))
        bounds.append((0.0, max_R + 0.5))

        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if not res.success:
                rho *= 0.8
                stall_count += 1
                if stall_count > 40:
                    break
                continue
            h_cand = res.x[:n]
        except Exception:
            rho *= 0.8
            stall_count += 1
            if stall_count > 40:
                break
            continue

        direction = h_cand - best_h
        improved = False
        for alpha in [1.0, 0.7, 0.5, 0.35, 0.2, 0.1]:
            h_try = best_h + alpha * direction
            h_try = project_feasible(h_try, n / 2.0)
            sc = eval_score(h_try)
            if sc < best_score - 1e-14:
                best_score = sc
                best_h = h_try
                improved = True
                break

        if improved:
            stall_count = 0
            rho = min(0.25, rho * 1.015)
        else:
            stall_count += 1
            rho *= 0.7
            if rho < 0.001:
                break

    return best_h, best_score

def pair_transfer_optimize(h, time_limit, rng, max_rounds=200):
    n = len(h)
    best_h = h.copy()
    best_score = eval_score(h)

    for rnd in range(max_rounds):
        if time.time() > time_limit:
            break

        R = corr_raw(best_h)
        max_ell = np.argmax(R)
        max_R = R[max_ell]
        g = corr_grad_k(best_h, max_ell)

        pair_scores = np.abs(g[:, None] - g[None, :])
        np.fill_diagonal(pair_scores, -1)
        top_pairs = np.argpartition(pair_scores.ravel(), -40)[::-1][:40]

        best_this_round = False
        for pidx in top_pairs:
            if time.time() > time_limit:
                break
            a, b = divmod(pidx, n)
            if a == b:
                continue

            lower = max(-best_h[a], best_h[b] - 1.0)
            upper = min(1.0 - best_h[a], -best_h[b])
            if upper - lower < 1e-12:
                continue

            n_grid = 15
            deltas = np.linspace(lower, upper, n_grid)
            best_delta = 0.0
            best_sc = best_score

            for d in deltas:
                h_try = best_h.copy()
                h_try[a] += d
                h_try[b] -= d
                if h_try[a] < -1e-9 or h_try[a] > 1.0 + 1e-9:
                    continue
                sc = eval_score(h_try)
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    best_delta = d

            if best_delta != 0.0:
                span = (upper - lower) / n_grid
                for _ in range(8):
                    lo_r = max(lower, best_delta - span)
                    hi_r = min(upper, best_delta + span)
                    grid2 = np.linspace(lo_r, hi_r, 15)
                    for d in grid2:
                        h_try = best_h.copy()
                        h_try[a] += d
                        h_try[b] -= d
                        if h_try[a] < -1e-9 or h_try[a] > 1.0 + 1e-9:
                            continue
                        sc = eval_score(h_try)
                        if sc < best_sc - 1e-14:
                            best_sc = sc
                            best_delta = d
                    span /= 3.0
                    if span < 1e-7:
                        break

                h_try = best_h.copy()
                h_try[a] += best_delta
                h_try[b] -= best_delta
                h_try = np.clip(h_try, 0, 1)
                h_try = project_feasible(h_try, n / 2.0)
                sc = eval_score(h_try)
                if sc < best_sc:
                    best_score = sc
                else:
                    best_score = best_sc
                best_h = h_try if eval_score(h_try) <= best_score else best_h
                best_h = np.clip(best_h, 0, 1)
                best_h = project_feasible(best_h, n / 2.0)
                best_sc = eval_score(best_h)
                if best_sc < best_score:
                    best_score = best_sc
                best_this_round = True

        if not best_this_round:
            break

    return best_h, best_score

def perturbed_restart(h, rng, n_pert=10, amp=0.05):
    n = len(h)
    h_pert = h.copy()
    idx = rng.choice(n, size=min(n_pert, n), replace=False)
    h_pert[idx] += amp * (rng.rand(len(idx)) - 0.5)
    h_pert = project_feasible(h_pert, n / 2.0)
    return h_pert

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy().reshape(-1)
    except (NameError, AttributeError):
        n0 = 126
        m0 = n0 // 2
        x = np.linspace(0, 1, m0)
        q0 = 0.5 + 0.3 * np.sin(2 * np.pi * x) + 0.1 * np.sin(4 * np.pi * x)
        q0 = np.clip(q0, 0, 1)
        h0 = build_sym_h(q0)

    n0 = len(h0)
    h0 = project_feasible(h0, n0 / 2.0)
    best_h = h0.copy()
    best_c5 = eval_score(h0)

    # Phase 1: Symmetric LP optimization
    if n0 % 2 == 0:
        q0 = sym_from_h(h0)
        tl1 = t0 + budget_s * 0.22
        q1, c51, h1 = lp_optimize_sym(q0, tl1, rho_init=0.15)
        if c51 < best_c5:
            best_c5 = c51
            best_h = h1.copy()

    # Phase 2: Full LP optimization
    rem = budget_s - (time.time() - t0)
    if rem > 120:
        tl2 = t0 + budget_s * 0.45
        h2, c52 = lp_optimize_full(best_h, tl2, rho_init=0.08)
        if c52 < best_c5:
            best_c5 = c52
            best_h = h2.copy()

    # Phase 3: Pair transfer refinement
    rem = budget_s - (time.time() - t0)
    if rem > 80:
        tl3 = t0 + budget_s * 0.62
        h3, c53 = pair_transfer_optimize(best_h, tl3, rng, max_rounds=120)
        if c53 < best_c5:
            best_c5 = c53
            best_h = h3.copy()

    # Phase 4: Second LP pass on improved solution
    rem = budget_s - (time.time() - t0)
    if rem > 80:
        tl4 = t0 + budget_s * 0.72
        h4, c54 = lp_optimize_full(best_h, tl4, rho_init=0.06)
        if c54 < best_c5:
            best_c5 = c54
            best_h = h4.copy()
        if best_h.size % 2 == 0:
            q4 = sym_from_h(best_h)
            tl4b = t0 + budget_s * 0.78
            q4b, c54b, h4b = lp_optimize_sym(q4, tl4b, rho_init=0.08)
            if c54b < best_c5:
                best_c5 = c54b
                best_h = h4b.copy()

    # Phase 5: Multi-resolution exploration
    rem = budget_s - (time.time() - t0)
    n_base = len(best_h)
    tried = set([n_base])
    deltas = [-4, -2, 2, 4, -6, 6, -8, 8, -10]
    for dlt in deltas:
        rem = budget_s - (time.time() - t0)
        if rem < 40:
            break
        nc = n_base + dlt
        if nc in tried or nc < 48 or nc > 200:
            continue
        tried.add(nc)
        h_try = transfer_to_n(best_h, nc)
        tl5 = time.time() + min(rem * 0.35, 70)
        if nc % 2 == 0:
            q_try = sym_from_h(h_try)
            q_tn, c5_tn, h_tn = lp_optimize_sym(q_try, tl5, rho_init=0.12)
        else:
            h_tn, c5_tn = lp_optimize_full(h_try, tl5, rho_init=0.08)
        if c5_tn < best_c5:
            best_c5 = c5_tn
            best_h = h_tn.copy()
            n_base = len(best_h)

    # Phase 6: Perturbation + reoptimize
    rem = budget_s - (time.time() - t0)
    n_perts = min(4, max(1, int(rem / 60)))
    for pi in range(n_perts):
        rem = budget_s - (time.time() - t0)
        if rem < 40:
            break
        h_pert = perturbed_restart(best_h, rng, n_pert=8, amp=0.04)
        tl6 = time.time() + min(rem * 0.4, 60)
        h6, c56 = lp_optimize_full(h_pert, tl6, rho_init=0.10)
        if c56 < best_c5:
            best_c5 = c56
            best_h = h6.copy()

    # Phase 7: Final polish
    rem = budget_s - (time.time() - t0)
    if rem > 20:
        n_fin = len(best_h)
        if n_fin % 2 == 0:
            q_fin = sym_from_h(best_h)
            tl7 = time.time() + rem * 0.6
            q7, c57, h7 = lp_optimize_sym(q_fin, tl7, rho_init=0.04)
            if c57 < best_c5:
                best_c5 = c57
                best_h = h7.copy()
        if rem > 15:
            h7, c57f = lp_optimize_full(best_h, time.time() + (rem - 15) * 0.5, rho_init=0.03)
            if c57f < best_c5:
                best_c5 = c57f
                best_h = h7.copy()

    # Final validation
    best_h = np.asarray(best_h, dtype=float).reshape(-1)
    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_c5 = eval_score(best_h)

    return best_h, best_c5, len(best_h)
