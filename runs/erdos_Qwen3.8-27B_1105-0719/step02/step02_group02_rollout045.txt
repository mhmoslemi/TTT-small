def project_feasible(x, target_sum):
    """Project x onto {0<=h<=1, sum(h)=target_sum}"""
    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

    s0 = max(0.0, raw_max - target_raw) + 1e-6

        s_max = max(s0 * 3 + 1.0, 5.0)

        h = x[:n]
        s = x[n]
        return s + rho / n * np.sum((h - h0)**2)

        h = x[:n]
        g = np.zeros(n + 1)
        g[:n] = 2.0 * rho / n * (h - h0)
        g[n] = 1.0
        return g
    
    def eq_fun(x):
        return x[:n].sum() - target_sum
    
    def eq_jac(x):
        j = np.zeros(n + 1)
        j[:n] = 1.0
        return j

        h = x[:n]
        s = x[n]
        corrs = compute_corrs(h)
        return target_raw + s - corrs  # >= 0 means feasible

        v = x[:m]
        g = np.zeros(m + 1)
        g[:m] = 2.0 * rho / m * (v - v0)
        g[m] = 1.0
        return g
    
    def ineq_fun(x):
        v = x[:m]
        s = x[m]
        h = build_complement(v, n)
        corrs = compute_corrs(h)
        return target_raw + s - corrs
    
    def ineq_jac(x):
        v = x[:m]
        h = build_complement(v, n)
        J = np.zeros((L, m + 1))
        for k in range(L):
            gk = grad_in_v_space(h, k, n)
            J[k, :m] = -gk
            J[k, m] = 1.0
        return J
    
    bounds = [(0.0, 1.0)] * m + [(0.0, s_max)]
    cons = [{'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}]
    
    try:
        res = minimize(obj, x0, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        v_out = np.clip(res.x[:m], 0.0, 1.0)
    except Exception:
        v_out = np.clip(v0, 0.0, 1.0)
    h_out = build_complement(v_out, n)
    return h_out, c5_of_h(h_out, dx)

def make_complement_start(n, rng, mode='random'):
    """Create a random complement start."""
    m = n // 2
    if mode == 'binary':
        bits = rng.randint(0, 2, m)
        v = bits.astype(float)
    elif mode == 'uniform':
        v = rng.uniform(0, 1, m)
    elif mode == 'bimodal':
        v = np.where(rng.uniform(0, 1, m) < 0.5, 0.2, 0.8)
        v += rng.uniform(-0.1, 0.1, m)
        v = np.clip(v, 0, 1)
    else:
        v = rng.uniform(0, 1, m)
    return v

def subgrad_momentum(h0, dx, target_sum, max_iter=20000):
    """Momentum subgradient descent."""
    n = len(h0)
    h = h0.copy()
    v_mom = np.zeros(n)
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    alpha = 0.1
    gamma = 0.8
    no_improve = 0
    top_m = max(5, n // 8)
    
    for it in range(max_iter):
        corrs = compute_corrs(h)
        max_c = corrs.max()
        c5_val = max_c * dx
        
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 8000:
            break
        
        sorted_idx = np.argsort(corrs)[-top_m:]
        grad = np.zeros(n)
        for k in sorted_idx:
            grad += grad_corr_k(h, k, n)
        grad /= len(sorted_idx)
        grad *= dx
        
        v_mom = gamma * v_mom - alpha * grad
        h_new = project_feasible(h + v_mom, target_sum)
        new_c5 = c5_of_h(h_new, dx)
        
        if new_c5 < c5_val - 1e-16:
            h = h_new
            alpha = min(alpha * 1.005, 0.3)
        else:
            alpha *= 0.93
            v_mom *= 0.3
        
        if alpha < 1e-13:
            break
    
    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    
    def remaining():
        return budget_s - (time.time() - t0)
    
    best_h = None
    best_c5 = float('inf')
    
    # Initialize from parent
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = None
    
    if h0 is not None and len(h0) > 0:
        n = len(h0)
        dx = 2.0 / n
        target_sum = n / 2.0
        h0 = project_feasible(h0, target_sum)
        best_h = h0.copy()
        best_c5 = c5_of_h(h0, dx)
    else:
        # Fallback: create a valid starting point
        n = 64
        dx = 2.0 / n
        target_sum = n / 2.0
        v = rng.randint(0, 2, n // 2).astype(float)
        best_h = build_complement(v, n)
        best_c5 = c5_of_h(best_h, dx)
    
    n_b = len(best_h)
    dx_b = 2.0 / n_b
    ts_b = n_b / 2.0
    
    # Phase 1: Smooth-max SLSQP ladder (quick baseline improvement)
    for beta in [50, 150, 500, 1500, 4000, 10000]:
        if remaining() < 30:
            break
        h_opt, c5_val = slsqp_smooth(best_h, dx_b, ts_b, beta, 500)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_opt
    
    # Phase 2: Complement family exploration
    if remaining() > 150:
        # Try complement optimization at current n (if even) and nearby even n
        n_candidates_comp = []
        n_even = n_b if n_b % 2 == 0 else n_b + 1
        for delta in [-8, -4, 0, 4, 8, 12, 16, 24, 32]:
            nc = n_even + delta
            if 30 <= nc <= 160 and nc % 2 == 0:
                n_candidates_comp.append(nc)
        n_candidates_comp = sorted(set(n_candidates_comp))
        
        for nc in n_candidates_comp:
            if remaining() < 25:
                break
            dx_c = 2.0 / nc
            # Try a few random starts in complement space
            for trial in range(4):
                if remaining() < 15:
                    break
                if trial == 0 and nc == n_even:
                    # Extract v from current best
                    m = nc // 2
                    v = best_h[:m].copy()
                    # Verify complement structure is reasonable
                    h_comp = build_complement(v, nc)
                    c5_comp = c5_of_h(h_comp, dx_c)
                else:
                    mode = ['binary', 'uniform', 'bimodal', 'binary'][trial % 4]
                    v = make_complement_start(nc, rng, mode)
                    h_comp = build_complement(v, nc)
                    c5_comp = c5_of_h(h_comp, dx_c)
                
                # Optimize in complement space
                if remaining() < 15:
                    break
                h_opt, c5_val = slsqp_complement(v, nc, 200, 300)
                if c5_val < best_c5:
                    best_c5 = c5_val
                    best_h = h_opt
                    n_b = nc
                    dx_b = 2.0 / nc
                    ts_b = nc / 2.0
    
    # Phase 3: Epigraph optimization (main new engine)
    if remaining() > 120:
        n_b = len(best_h)
        dx_b = 2.0 / n_b
        ts_b = n_b / 2.0
        raw_max_best = compute_corrs(best_h).max()
        
        # Target schedule: try to get below current best
        # delta in C5 units
        deltas = [0.002, 0.0015, 0.0012, 0.001, 0.0008, 0.0006, 0.0005, 0.0004, 0.0003, 0.0002, 0.00015, 0.0001]
        
        # Build anchor pool
        anchor_pool = [best_h.copy()]
        # Add complement version of best
        if n_b % 2 == 0:
            m = n_b // 2
            v_best = best_h[:m].copy()
            h_comp = build_complement(v_best, n_b)

            if remaining() < 30:

            if target_c5 < 0.37:  # Don't go too far
                break

                    if c5_val < best_c5 - 1e-10:
                        best_c5 = c5_val
                        best_h = h_opt.copy()
