    threshold = max_c - (0.05 * max_c * (1.0 - active_frac))
    active_mask = corrs >= threshold
    n_active = active_mask.sum()
    
    # Ensure at least active_min
    if n_active < active_min:
        sorted_idx = np.argsort(corrs)[::-1]
        active_idx = sorted_idx[:active_min]
    else:
        # Cap at reasonable size
        max_active = max(active_min, int(active_frac * nc))
        if n_active > max_active:
            sorted_idx = np.argsort(corrs)[::-1]
            active_idx = sorted_idx[:max_active]
        else:
            active_idx = np.where(active_mask)[0]
    
    n_act = len(active_idx)
    
    # LP: min s  s.t.  grad_k @ delta - s <= -r_k,  sum(delta)=0,  bounds
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    A_ub = np.zeros((n_act, n + 1))
    b_ub = np.zeros(n_act)
    
    for row, k in enumerate(active_idx):
        g = grad_corr_k(h, k, n)
        A_ub[row, :n] = g
        A_ub[row, n] = -1.0
        b_ub[row] = -corrs[k]
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    bounds = []
    for i in range(n):
        lo_b = -h[i] * alpha
        hi_b = (1.0 - h[i]) * alpha
        bounds.append((lo_b, hi_b))
    bounds.append((0.0, None))
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
    except Exception:
        return None, None
    
    if res.status != 0:
        return None, None
    
    delta = res.x[:n]
    s_pred = res.x[n]
    return delta, s_pred


def lp_polish(h0, target_sum, max_iters, alpha_start, alpha_end,
              active_frac, time_limit):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5, _ = score_h(h)
    t_start = time.time()
    no_improve_count = 0
    
    for it in range(max_iters):
        if time.time() - t_start > time_limit:
            break
        
        frac = float(it) / max(1, max_iters)
        alpha = alpha_start * (alpha_end / alpha_start) ** frac
        alpha = max(alpha, 1e-4)
        
        delta, s_pred = lp_one_step(h, alpha, active_frac, 15)
        if delta is None:
            alpha *= 0.5
            if alpha < 1e-5:
                break
            continue
        
        delta_norm = np.linalg.norm(delta)
        if delta_norm < 1e-13:
            break
        
        # Line search
        best_step = 0.0
        best_trial_c5 = best_c5
        for t in [1.0, 0.75, 0.5, 0.35, 0.25, 0.15, 0.1]:
            h_try = h + t * delta
            h_try = make_feasible(h_try, target_sum)
            c_try, _ = score_h(h_try)
            if c_try < best_trial_c5 - 1e-14:
                best_trial_c5 = c_try
                best_step = t
        
        if best_step > 0:
            h = h + best_step * delta
            h = make_feasible(h, target_sum)
            c_now, _ = score_h(h)
            if c_now < best_c5 - 1e-14:
                best_c5 = c_now
                best_h = h.copy()
                no_improve_count = 0
            else:
                no_improve_count += 1
        else:
            no_improve_count += 1
            alpha *= 0.7
        
        if no_improve_count > 40:
            # Try a shake
            n_shake = max(2, n // 20)
            idx_shake = np.random.RandomState(it * 7 + 3).choice(n, n_shake, replace=False)
            h_shake = h + 0.05 * (np.random.RandomState(it * 13 + 1).randn(n) - 0.5)
            h = make_feasible(h_shake, target_sum)
            no_improve_count = 0
    
    return best_h, best_c5


def coord_descent(h0, target_sum, max_sweeps, time_limit):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5, _ = score_h(h)
    t_start = time.time()
    rng = np.random.RandomState(12345)
    
    for sweep in range(max_sweeps):
        if time.time() - t_start > time_limit:
            break
        
        order = rng.permutation(n)
        improved = False
        
        for i in order:
            if time.time() - t_start > time_limit:
                break
            if h[i] <= 0 or h[i] >= 1:
                continue
            
            # Try a range of new values for h[i], adjusting h[j] to keep sum
            # Pick j as a partner with opposite slack
            if h[i] > 0.5:
                j_mask = h < 0.4
            else:
                j_mask = h > 0.6
            j_candidates = np.where(j_mask)[0]
            if len(j_candidates) == 0:
                j_candidates = np.where(np.abs(h - 0.5) > 0.2)[0]
            if len(j_candidates) == 0:
                continue
            j = j_candidates[rng.randint(len(j_candidates))]
            if i == j:
                continue
            
            old_hi = h[i]
            old_hj = h[j]
            
            for frac_val in [0.1, 0.3, 0.5, 0.7, 0.9]:
                new_hi = old_hi + (0.5 - old_hi) * 2.0 * frac_val
                new_hi = np.clip(new_hi, 0.0, 1.0)
                diff = old_hi - new_hi
                new_hj = np.clip(old_hj + diff, 0.0, 1.0)
                # Correct any rounding
                actual_diff = old_hi + old_hj - new_hi - new_hj
                if abs(actual_diff) > 0:
                    # Redistribute to others minimally
                    pass
                h[i] = new_hi
                h[j] = new_hj
                
                c_try, _ = score_h(h)
                if c_try < best_c5 - 1e-14:
                    best_c5 = c_try
                    best_h = h.copy()
                    improved = True
                else:
                    h[i] = old_hi
                    h[j] = old_hj
        
        if not improved and sweep > 3:
            break
    
    return best_h, best_c5


def sa_discrete(n, M, max_iters, t0, t_end, rng_seed, time_limit):
    target_a = n * M // 2
    a = np.full(n, M // 2, dtype=int)
    
    # Fix sum
    diff = target_a - a.sum()
    if diff > 0:
        for i in range(n):
            if diff <= 0:
                break
            add = min(diff, M - a[i])
            a[i] += add
            diff -= add
    elif diff < 0:
        for i in range(n):
            if diff >= 0:
                break
            sub = min(-diff, a[i])
            a[i] -= sub
            diff += sub
    
    dx = 2.0 / n
    h = a.astype(float) / M
    c_cur = np.correlate(h, 1.0 - h, mode='full').max() * dx
    best_h = h.copy()
    best_c5 = c_cur
    
    t = t0
    cool = (t_end / t0) ** (1.0 / max(1, max_iters))
    t_start = time.time()
    rng_sa = np.random.RandomState(rng_seed)
    
    for it in range(max_iters):
        if time.time() - t_start > time_limit:
            break
        t *= cool
        
        i = rng_sa.randint(n)
        j = rng_sa.randint(n)
        if i == j:
            j = (j + 1) % n
        if a[i] >= M or a[j] <= 0:
            continue
        
        a[i] += 1
        a[j] -= 1
        h_new = a.astype(float) / M
        c_new = np.correlate(h_new, 1.0 - h_new, mode='full').max() * dx
        
        dc = c_new - c_cur
        if dc < 0 or (t > 1e-8 and rng_sa.random() < np.exp(-dc / max(t, 1e-10))):
            c_cur = c_new
            if c_new < best_c5:
                best_c5 = c_new
                best_h = h_new.copy()
        else:
            a[i] -= 1
            a[j] += 1
    
    return best_h, best_c5


def transfer_to_grid(h_src, n_dst, target_sum_dst):
    n_src = len(h_src)
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    h_interp = np.interp(x_dst, x_src, h_src)
    return make_feasible(h_interp, target_sum_dst)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    
    def remaining():
        return budget_s - (time.time() - t0)
    
    # Get starting point
    try:
        h0 = np.array(initial_h_values, dtype=float)
    except Exception:
        h0 = None
    
    if h0 is None or len(h0) < 10:
        h0 = np.full(128, 0.5)
    
    n = len(h0)
    target_sum = n / 2.0
    h0 = make_feasible(h0, target_sum)
    
    best_h = h0.copy()
    best_c5, _ = score_h(h0)
    
    # Phase 1: LP polish on parent grid
    if remaining() > 80:
        h_lp, c5_lp = lp_polish(best_h, target_sum,
                                max_iters=300, alpha_start=0.5, alpha_end=0.01,
                                active_frac=0.35,
                                time_limit=min(remaining() - 60, 350))
        if c5_lp < best_c5:
            best_c5 = c5_lp
            best_h = h_lp
    
    # Phase 2: Second LP pass with different params
    if remaining() > 60:
        h_lp, c5_lp = lp_polish(best_h, target_sum,
                                max_iters=250, alpha_start=0.3, alpha_end=0.005,
                                active_frac=0.5,
                                time_limit=min(remaining() - 50, 250))
        if c5_lp < best_c5:
            best_c5 = c5_lp
            best_h = h_lp
    
    # Phase 3: Coordinate descent
    if remaining() > 40:
        h_cd, c5_cd = coord_descent(best_h, target_sum,
                                    max_sweeps=30,
                                    time_limit=min(remaining() - 30, 100))
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd
            # Follow up with LP
            if remaining() > 40:
                h_lp, c5_lp = lp_polish(best_h, target_sum,
                                        max_iters=100, alpha_start=0.2,
                                        alpha_end=0.01, active_frac=0.4,
                                        time_limit=min(remaining() - 30, 80))
                if c5_lp < best_c5:
                    best_c5 = c5_lp
                    best_h = h_lp
    
    # Phase 4: Discrete SA on small grids
    sa_configs = [
        (64, 4, 40000, 0.012, 1e-5),
        (80, 4, 35000, 0.010, 1e-5),
        (96, 4, 30000, 0.008, 1e-5),
        (64, 8, 30000, 0.006, 1e-5),
        (80, 8, 25000, 0.005, 1e-5),
    ]
    
    for ci, (n_sa, M_sa, sa_iters, sa_t0, sa_t1) in enumerate(sa_configs):
        if remaining() < 40:
            break
        if (n_sa * M_sa) % 2 != 0:
            continue
        ts_sa = n_sa // 2
        
        sa_time = min(remaining() - 40, 100)
        h_sa, c5_sa = sa_discrete(n_sa, M_sa, sa_iters, sa_t0, sa_t1,
                                  seed + ci * 100, sa_time)
        
        if c5_sa < best_c5:
            # Transfer to parent grid and polish
            if remaining() > 50:
                h_trans = transfer_to_grid(h_sa, n, target_sum)
                h_pol, c5_pol = lp_polish(h_trans, target_sum,
                                          max_iters=150, alpha_start=0.4,
                                          alpha_end=0.01, active_frac=0.4,
                                          time_limit=min(remaining() - 30, 100))
                if c5_pol < best_c5:
                    best_c5 = c5_pol
                    best_h = h_pol
                elif c5_sa < best_c5:
                    # Use the SA result on its own grid
                    best_c5 = c5_sa
                    best_h = h_sa
            else:
                best_c5 = c5_sa
                best_h = h_sa
    
    # Phase 5: Try different grid sizes with interpolation from best
    n_candidates = [62, 66, 90, 94, 100, 110, 120, 136, 140]
    for n_try in n_candidates:
        if remaining() < 40:
            break
        if n_try == len(best_h):
            continue
        dx_try = 2.0 / n_try
        ts_try = n_try / 2.0
        h_interp = transfer_to_grid(best_h, n_try, ts_try)
        h_opt, c5_opt = lp_polish(h_interp, ts_try,
                                  max_iters=120, alpha_start=0.3,
                                  alpha_end=0.01, active_frac=0.4,
                                  time_limit=min(remaining() - 30, 80))
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
    
    # Phase 6: Final intensive polish
    if remaining() > 20:
        n_b = len(best_h)
        ts_b = n_b / 2.0
        h_lp, c5_lp = lp_polish(best_h, ts_b,
                                max_iters=200, alpha_start=0.15,
                                alpha_end=0.003, active_frac=0.5,
                                time_limit=min(remaining() - 15, 150))
        if c5_lp < best_c5:
            best_c5 = c5_lp
            best_h = h_lp
    
    if remaining() > 15:
        n_b = len(best_h)
        ts_b = n_b / 2.0
        h_lp, c5_lp = lp_polish(best_h, ts_b,
                                max_iters=100, alpha_start=0.05,
                                alpha_end=0.001, active_frac=0.6,
                                time_limit=min(remaining() - 5, 80))
        if c5_lp < best_c5:
            best_c5 = c5_lp
            best_h = h_lp
    
    if remaining() > 10:
        n_b = len(best_h)
        ts_b = n_b / 2.0
        h_cd, c5_cd = coord_descent(best_h, ts_b,
                                    max_sweeps=15,
                                    time_limit=min(remaining() - 3, 50))
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd
    
    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    final_c5, _ = score_h(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    return best_h, best_c5, len(best_h)
