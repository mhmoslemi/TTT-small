    return grad

def smooth_obj(h, dx, beta):
    corr = corr_array(h)
    m = corr.max()
    shifted = np.clip(corr - m, -50.0 / beta, 0.0)
    w = np.exp(beta * shifted)
    ws = w.sum()
    if ws < 1e-300:
        return m * dx
    return (m + np.log(ws) / beta) * dx

def smooth_grad(h, dx, beta):
    n = len(h)
    corr = corr_array(h)
    m = corr.max()
    shifted = np.clip(corr - m, -50.0 / beta, 0.0)
    w = np.exp(beta * shifted)
    ws = w.sum()
    if ws < 1e-300:
        k_max = int(np.argmax(corr))
        return grad_corr_shift(h, k_max - (n - 1), n) * dx
    w /= ws
    grad = np.zeros(n)
    for k in range(len(corr)):
        if w[k] > 1e-12:
            shift = k - (n - 1)
            grad += w[k] * grad_corr_shift(h, shift, n)
    return grad * dx

def run_slsqp(h0, dx, target, beta, maxiter):
    global _global_n, _global_target
    _global_n = len(h0)
    _global_target = target
    bounds = [(0.0, 1.0)] * _global_n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_start = project_feasible(h0, target)
    try:
        res = minimize(smooth_obj, h_start, args=(dx, beta), jac=smooth_grad,
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        h_out = project_feasible(res.x, target)
    except Exception:
        h_out = h_start
    return h_out, c5_exact(h_out, dx)

def delta_binary(h, i, j, n):
    ms = np.arange(-(n - 1), n)
    valid_i1 = (0 <= i - ms) & (i - ms <= n - 1)
    valid_i2 = (0 <= i + ms) & (i + ms <= n - 1)
    valid_j1 = (0 <= j - ms) & (j - ms <= n - 1)
    valid_j2 = (0 <= j + ms) & (j + ms <= n - 1)
    delta = np.zeros(2 * n - 1)
    delta[valid_i1] += -(1.0 - h[i - ms[valid_i1]])
    delta[valid_i2] += h[i + ms[valid_i2]]
    delta[valid_j1] += (1.0 - h[j - ms[valid_j1]])
    delta[valid_j2] += -h[j + ms[valid_j2]]
    return delta

def binary_sa(h0, dx, rng, max_iters, temp0, cooling, patience):
    n = len(h0)
    target = n // 2
    h = np.zeros(n)
    order = np.argsort(-h0)
    h[order[:target]] = 1.0
    corrs = corr_array(h)
    best_c5 = corrs.max() * dx
    best_h = h.copy()
    ones = list(np.where(h == 1.0)[0])
    zeros = list(np.where(h == 0.0)[0])
    temp = temp0
    no_improve = 0
    for it in range(max_iters):
        if it % 3000 == 0:
            temp *= cooling
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        d = delta_binary(h, i, j, n)
        new_corrs = corrs + d
        new_max = new_corrs.max()
        new_c5 = new_max * dx
        if new_c5 <= best_c5 + 1e-15:
            h[i], h[j] = 0.0, 1.0
            corrs = new_corrs
            best_c5 = new_c5
            best_h = h.copy()
            ones.remove(i)
            zeros.remove(j)
            ones.append(j)
            zeros.append(i)
            no_improve = 0
        else:
            dv = new_c5 - best_c5
            if temp > 1e-10 and rng.random() < np.exp(-dv / temp):
                h[i], h[j] = 0.0, 1.0
                corrs = new_corrs
                ones.remove(i)
                zeros.remove(j)
                ones.append(j)
                zeros.append(i)
                no_improve = 0
            else:
                no_improve += 1
        if no_improve > patience:
            break
    return best_h, best_c5

def binary_greedy(h0, dx, max_sweeps):
    n = len(h0)
    target = n // 2
    h = np.zeros(n)
    order = np.argsort(-h0)
    h[order[:target]] = 1.0
    corrs = corr_array(h)
    best_c5 = corrs.max() * dx
    best_h = h.copy()
    for _ in range(max_sweeps):
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        cur_max = corrs.max()
        best_swap = None
        best_new_max = cur_max
        for i in ones:
            for j in zeros:
                d = delta_binary(h, i, j, n)
                nm = (corrs + d).max()
                if nm < best_new_max - 1e-14:
                    best_new_max = nm
                    best_swap = (i, j)
        if best_swap is None:
            break
        i, j = best_swap
        h[i], h[j] = 0.0, 1.0
        corrs = corr_array(h)
        nc5 = corrs.max() * dx
        if nc5 < best_c5:
            best_c5 = nc5
            best_h = h.copy()
    return best_h, best_c5

def peak_shave(h0, dx, target, max_rounds):
    n = len(h0)
    h = project_feasible(h0, target)
    best_c5 = c5_exact(h, dx)
    best_h = h.copy()
    for rnd in range(max_rounds):
        corr = corr_array(h)
        k_max = int(np.argmax(corr))
        shift = k_max - (n - 1)
        g = -grad_corr_shift(h, shift, n)
        step = 0.1
        improved = False
        for _ in range(30):
            cand = project_feasible(h + step * g, target)
            cand_c5 = c5_exact(cand, dx)
            if cand_c5 < best_c5 - 1e-12:
                h = cand
                best_c5 = cand_c5
                best_h = h.copy()
                improved = True
                break
            step *= 0.5
        if not improved:
            # Try all peak shifts
            top_k = np.argsort(-corr)[:5]
            found = False
            for kk in top_k:
                s2 = int(kk) - (n - 1)
                g2 = -grad_corr_shift(h, s2, n)
                step2 = 0.05
                for _ in range(25):
                    cand2 = project_feasible(h + step2 * g2, target)
                    c2 = c5_exact(cand2, dx)
                    if c2 < best_c5 - 1e-12:
                        h = cand2
                        best_c5 = c2
                        best_h = h.copy()
                        found = True
                        improved = True
                        break
                    step2 *= 0.5
                if found:
                    break
            if not improved:
                break
    return best_h, best_c5

def make_beatty(n, target, frac):
    h = np.zeros(n)
    for k in range(target):
        pos = int((k * frac * n) % n)
        h[pos] = 1.0
    return project_feasible(h, target)

def interp_to_n(h_src, n_src, n_dst, target_dst):
    x_src = np.linspace(0.0, 2.0, n_src)
    x_dst = np.linspace(0.0, 2.0, n_dst)
    h_dst = np.interp(x_dst, x_src, h_src)
    return project_feasible(h_dst, target_dst)

def run_seed(h_start, dx, target, rng, time_budget):
    t0 = time.time()
    best_h = project_feasible(h_start, target)
    best_c5 = c5_exact(best_h, dx)
    
    betas = [30, 80, 200, 500, 1200, 3000, 8000, 20000]
    for b in betas:
        if time.time() - t0 > time_budget:
            break
        h_new, c5_new = run_slsqp(best_h, dx, target, b, 600)
        if c5_new < best_c5 - 1e-10:
            best_c5 = c5_new
            best_h = h_new
    
    # Peak shaving
    if time.time() - t0 < time_budget - 10:
        h_ps, c5_ps = peak_shave(best_h, dx, target, 50)
        if c5_ps < best_c5 - 1e-10:
            best_c5 = c5_ps
            best_h = h_ps
    
    # Binary greedy
    if time.time() - t0 < time_budget - 60:
        h_bg, c5_bg = binary_greedy(best_h, dx, max_sweeps=30)
        if c5_bg < best_c5 - 1e-10:
            best_c5 = c5_bg
            best_h = h_bg
    
    # Binary SA
    if time.time() - t0 < time_budget - 120:
        h_sa, c5_sa = binary_sa(best_h, dx, rng, 200000, 0.04, 0.9995, 6000)
        if c5_sa < best_c5 - 1e-10:
            best_c5 = c5_sa
            best_h = h_sa
    
    # Final SLSQP polish
    if time.time() - t0 < time_budget - 30:
        h_fin, c5_fin = run_slsqp(best_h, dx, target, 50000, 1000)
        if c5_fin < best_c5 - 1e-10:
            best_c5 = c5_fin
            best_h = h_fin
    
    # Final peak shave
    if time.time() - t0 < time_budget - 5:
        h_ps2, c5_ps2 = peak_shave(best_h, dx, target, 30)
        if c5_ps2 < best_c5 - 1e-10:
            best_c5 = c5_ps2
            best_h = h_ps2
    
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)
    
    # Get initial construction
    try:
        h_init = np.array(initial_h_values, dtype=float)
    except NameError:
        h_init = np.full(126, 0.5)
    
    n0 = len(h_init)
    dx0 = 2.0 / n0
    target0 = n0 / 2.0
    h_init = project_feasible(h_init, target0)
    best_c5 = c5_exact(h_init, dx0)
    best_h = h_init.copy()
    
    # Phase 1: Optimize at original n
    alloc1 = 0.55 * budget_s
    h1, c5_1 = run_seed(h_init, dx0, target0, rng, alloc1)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1
    
    elapsed = time.time() - t_start
    remaining = budget_s - elapsed
    
    # Phase 2: Try nearby resolutions
    candidate_ns = []
    for dn in [-4, -2, 2, 4, 6, 8, 12, 16, -6, -8, 10, 20]:
        nn = n0 + dn
        if nn >= 48 and nn % 2 == 0:
            candidate_ns.append(nn)
    candidate_ns = sorted(set(candidate_ns))
    
    per_res = max(20, (remaining * 0.6) / len(candidate_ns))
    
    for nn in candidate_ns:
        if time.time() - t_start > budget_s - 60:
            break
        if nn == n0:
            continue
        dx_n = 2.0 / nn
        target_n = nn / 2.0
        h_interp = interp_to_n(best_h, len(best_h), nn, target_n)
        
        # Quick SLSQP
        h_t, c5_t = run_slsqp(h_interp, dx_n, target_n, 500, 400)
        if c5_t < best_c5 - 1e-8:
            # Do more work on this resolution
            h_t2, c5_t2 = run_seed(h_t, dx_n, target_n, rng, min(per_res, 150))
            if c5_t2 < best_c5:
                best_c5 = c5_t2
                best_h = h_t2
    
    # Phase 3: Multi-start at best n
    n_best = len(best_h)
    dx_best = 2.0 / n_best
    target_best = n_best / 2.0
    remaining2 = budget_s - (time.time() - t_start)
    
    if remaining2 > 80:
        fracs = [0.6180339887, 0.7548776662, 0.5, 0.3819660113]
        for frac in fracs:
            if time.time() - t_start > budget_s - 40:
                break
            h_try = make_beatty(n_best, int(target_best), frac)
            h_res, c5_res = run_slsqp(h_try, dx_best, target_best, 200, 400)
            if c5_res < best_c5 - 1e-8:
                h_res2, c5_res2 = run_peak_and_binary(h_res, dx_best, target_best, rng, remaining2 * 0.3)
                if c5_res2 < best_c5:
                    best_c5 = c5_res2
                    best_h = h_res2
    
    # Phase 4: Final deep optimization at best resolution
    remaining3 = budget_s - (time.time() - t_start)
    if remaining3 > 30:
        h_final, c5_final = run_seed(best_h, dx_best, target_best, rng, remaining3 * 0.8)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final
    
    # Final validation
    n_f = len(best_h)
    dx_f = 2.0 / n_f
    target_f = n_f / 2.0
    best_h = project_feasible(best_h, target_f)
    best_c5 = c5_exact(best_h, dx_f)
    
    return best_h, best_c5, n_f

def run_peak_and_binary(h0, dx, target, rng, time_budget):
    t0 = time.time()
    best_h = project_feasible(h0, target)
    best_c5 = c5_exact(best_h, dx)
    
    # Peak shave
    h_ps, c5_ps = peak_shave(best_h, dx, target, 40)
    if c5_ps < best_c5:
        best_c5 = c5_ps
        best_h = h_ps
    
    # Binary SA
    if time.time() - t0 < time_budget - 10:
        h_sa, c5_sa = binary_sa(best_h, dx, rng, 100000, 0.03, 0.9995, 5000)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa
    
    # Final SLSQP
    if time.time() - t0 < time_budget - 5:
        h_s, c5_s = run_slsqp(best_h, dx, target, 10000, 800)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s
    
    return best_h, best_c5
