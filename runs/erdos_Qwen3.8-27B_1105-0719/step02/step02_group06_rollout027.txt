import numpy as np
from scipy.optimize import minimize
import time


def build_sym_h(q):
    m = len(q)
    h = np.empty(2 * m)
    h[:m] = np.asarray(q, dtype=float)
    h[m:] = 1.0 - np.asarray(q, dtype=float)[::-1]
    return h


def eval_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)


def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    n = len(h)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    h = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        interior = (h > 1e-12) & (h < 1.0 - 1e-12)
        if interior.sum() > 0:
            h[interior] += diff / interior.sum()
    return np.clip(h, 0.0, 1.0)


def sym_from_h(h):
    n = len(h)
    m = n // 2
    q = np.empty(m)
    for i in range(m):
        q[i] = 0.5 * (h[i] + (1.0 - h[n - 1 - i]))
    return np.clip(q, 0.0, 1.0)


def transfer_to_n(h_old, n_new):

    x_old = (np.arange(n_old) + 0.5) * 2.0 / n_old
    x_new = (np.arange(n_new) + 0.5) * 2.0 / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new / 2.0)


def bisect_1d_min(A, B):
    """Find t in [0,1] minimizing max_k(A_k + B_k * t)."""
    lo, hi = 0.0, 1.0
    for _ in range(80):

        fmid = np.max(vals)
        tol = 1e-14 * max(1.0, np.abs(fmid))

        elif b_max < -1e-14:
            lo = mid
        else:

    return (lo + hi) / 2.0


def coord_sweep_q(q, n_full, max_sweeps, rng, deadline):
    """Coordinate descent on symmetric q parameterization."""
    m = n_full // 2
    q = np.asarray(q, dtype=float).copy()
    h = build_sym_h(q)
    n_lags = 2 * n_full - 1
    best_score = eval_score(h)

    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        order = rng.permutation(m)
        for j in order:
            t_old = q[j]
            B = np.zeros(n_lags)
            nm1j = n_full - 1 - j
            for k in range(n_lags):
                d = k - (n_full - 1)
                bk = 0.0
                jd = j + d
                if 0 <= jd < n_full:
                    bk += (1.0 - h[jd])
                jdd = j - d
                if 0 <= jdd < n_full:
                    bk -= h[jdd]
                jd2 = nm1j + d
                if 0 <= jd2 < n_full:
                    bk -= (1.0 - h[jd2])
                jd3 = nm1j - d

    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(len(corrs)) / len(corrs)
    else:
        w = w / w_sum
    val = max_c + np.log(w_sum) / beta

    # Gradient w.r.t. q
    grad_q = np.zeros(m)
    for k in range(len(corrs)):
        if w[k] < 1e-14:
            continue
        d = k - (n_full - 1)
        i_lo = max(0, -d)
        i_hi = min(n_full - 1, n_full - 1 - d)
        if i_lo > i_hi:
            continue
        idx = np.arange(i_lo, i_hi + 1)
        g = np.zeros(n_full)
        g[idx] += (1.0 - h[idx + d])
        g[idx + d] += (-h[idx])
        for r in range(m):
            grad_q[r] += w[k] * (g[r] - g[n_full - 1 - r])
    grad_q *= dx
    return val, grad_q


def smooth_max_obj_fn(x, n_full, beta):
    return smooth_max_obj_grad(x, n_full, beta)[0]


def smooth_max_jac_fn(x, n_full, beta):
    return smooth_max_obj_grad(x, n_full, beta)[1]


def smooth_sym_optimize(q0, n_full, beta_list, maxiter_per_beta, deadline):
    m = n_full // 2
    q = np.clip(np.asarray(q0, dtype=float).copy(), 0, 1)
    best_score = eval_score(build_sym_h(q))
    best_q = q.copy()

    for beta in beta_list:
        if time.time() > deadline:
            break
        bounds = [(0.0, 1.0)] * m
        try:
            res = minimize(smooth_max_obj_fn, q, args=(n_full, beta),
                           jac=smooth_max_jac_fn, method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': maxiter_per_beta, 'ftol': 1e-14, 'gtol': 1e-9,
                                    'maxls': 50})
            q_new = np.clip(res.x, 0, 1)
        except Exception:
            q_new = q.copy()
        h_new = build_sym_h(q_new)
        sc = eval_score(h_new)
        if sc < best_score - 1e-13:
            best_score = sc
            best_q = q_new.copy()
        q = q_new

    return best_q, best_score, build_sym_h(best_q)


def select_active(corrs_scaled, margin_frac=0.03, neighbor_pad=3, max_k=60):
    max_val = corrs_scaled.max()
    threshold = max_val - max(1e-8, margin_frac * max_val)
    active = set(np.where(corrs_scaled >= threshold)[0].tolist())
    padded = set()
    for k in active:
        for d in range(-neighbor_pad, neighbor_pad + 1):
            if 0 <= k + d < len(corrs_scaled):
                padded.add(k + d)
    active_list = sorted(padded)
    if len(active_list) > max_k:
        scored = sorted([(corrs_scaled[k], k) for k in active_list], reverse=True)
        active_list = sorted([k for _, k in scored[:max_k]])
    return active_list


def epig_sym_con_fun(x, k, n_full):
    m = n_full // 2
    q = x[:m]
    t = x[m]
    h = build_sym_h(q)
    dx = 2.0 / n_full
    d = k - (n_full - 1)
    i_lo = max(0, -d)
    i_hi = min(n_full - 1, n_full - 1 - d)
    if i_lo > i_hi:
        return t
    val = np.sum(h[i_lo:i_hi + 1] * (1.0 - h[i_lo + d:i_hi + 1 + d])) * dx
    return t - val


def epig_sym_con_jac(x, k, n_full):
    m = n_full // 2
    q = x[:m]
    h = build_sym_h(q)
    dx = 2.0 / n_full
    d = k - (n_full - 1)
    i_lo = max(0, -d)
    i_hi = min(n_full - 1, n_full - 1 - d)
    j = np.zeros(m + 1)
    j[m] = 1.0
    if i_lo > i_hi:
        return j
    idx = np.arange(i_lo, i_hi + 1)
    g = np.zeros(n_full)
    g[idx] += (1.0 - h[idx + d])
    g[idx + d] += (-h[idx])
    for r in range(m):
        j[r] = -dx * (g[r] - g[n_full - 1 - r])
    return j


def epig_sym_obj_fn(x):
    return x[-1]


def epig_sym_obj_jac(x):
    n = len(x)
    g = np.zeros(n)
    g[-1] = 1.0
    return g


def solve_epigraph_sym(q_center, n_full, active_k, maxiter, ftol):
    m = n_full // 2
    h_c = build_sym_h(q_center)
    dx = 2.0 / n_full
    corrs_c = np.correlate(h_c, 1.0 - h_c, mode='full') * dx
    t0_val = corrs_c.max() + 1e-9

    x0 = np.zeros(m + 1)
    x0[:m] = q_center
    x0[m] = t0_val

    bounds = [(0.0, 1.0)] * m + [(0.0, 2.0)]
    constraints = []
    for k in active_k:
        constraints.append({'type': 'ineq',
                           'fun': lambda x, kk=k: epig_sym_con_fun(x, kk, n_full),
                           'jac': lambda x, kk=k: epig_sym_con_jac(x, kk, n_full)})

    try:
        res = minimize(epig_sym_obj_fn, x0, jac=epig_sym_obj_jac,
                       method='SLSQP', bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': ftol})
        q_out = np.clip(res.x[:m], 0.0, 1.0)
    except Exception:
        q_out = q_center.copy()

    h_out = build_sym_h(q_out)
    return q_out, h_out, eval_score(h_out)


def do_epig_sym_loop(h_input, max_rounds, per_solve_maxiter, deadline):
    n_s = len(h_input)
    if n_s % 2 != 0:
        return h_input, eval_score(h_input)
    m_s = n_s // 2
    dx_s = 2.0 / n_s
    q_s = sym_from_h(h_input)
    best_h = h_input.copy()
    best_c5 = eval_score(h_input)

    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        h_chk = build_sym_h(q_s)
        corrs = np.correlate(h_chk, 1.0 - h_chk, mode='full') * dx_s
        active = select_active(corrs, margin_frac=0.03, neighbor_pad=3, max_k=55)

        q_new, h_new, c5_new = solve_epigraph_sym(q_s, n_s, active, per_solve_maxiter, 1e-11)
        h_new = build_sym_h(np.clip(q_new, 0, 1))
        c5_new = eval_score(h_new)

        if c5_new < best_c5 - 1e-13:
            best_c5 = c5_new
            best_h = h_new.copy()
            q_s = sym_from_h(h_new)
            continue

        corrs_new = np.correlate(h_new, 1.0 - h_new, mode='full') * dx_s
        max_act = corrs_new[active].max() if len(active) > 0 else 0
        hidden = np.where(corrs_new > max_act + 1e-7)[0]
        if len(hidden) > 0:
            active = sorted(set(active + list(hidden)))
            if len(active) > 100:
                active = select_active(corrs_new, margin_frac=0.02, neighbor_pad=3, max_k=100)
            q_s = sym_from_h(h_new)
            continue
        break

    return best_h, best_c5


def binarize_push(h, strength=2.0):
    """Push values towards 0 or 1."""
    n = len(h)
    m = n // 2
    q = sym_from_h(h)
    # Push: values > 0.5 go up, values < 0.5 go down
    q_new = np.where(q > 0.5,
                     np.clip(0.5 + strength * (q - 0.5), 0, 1),
                     np.clip(0.5 + strength * (q - 0.5), 0, 1))
    return q_new


def make_structured_start(n, rng, mode=0):
    """Generate structured initial conditions."""
    m = n // 2
    x = np.linspace(0, 1, m, endpoint=False)
    if mode == 0:
        q = 0.5 + 0.4 * np.sin(2 * np.pi * x + rng.random() * 2 * np.pi)
    elif mode == 1:
        q = (x < rng.uniform(0.3, 0.7)).astype(float)
        q = q + 0.1 * np.sin(4 * np.pi * x)
    elif mode == 2:
        period = rng.randint(3, 8)
        q = (np.arange(m) % period < period // 2).astype(float)
        q = q + 0.05 * rng.randn(m)
    else:
        q = np.ones(m) * 0.5
        q[:m // 2] = 0.8
        q[m // 2:] = 0.2
        q = q + 0.1 * rng.randn(m)
    return np.clip(q, 0, 1)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def rem():
        return budget_s - (time.time() - t0)

    best_h = None
    best_c5 = float('inf')

    # Load parent
    try:
        h_parent = np.array(initial_h_values, dtype=float).copy().reshape(-1)
    except (NameError, AttributeError):
        h_parent = None

    if h_parent is not None and len(h_parent) > 10:
        n_p = len(h_parent)
        if abs(np.sum(h_parent) - n_p / 2.0) > 0.5:
            h_parent = project_feasible(h_parent, n_p / 2.0)
        h_parent = np.clip(h_parent, 0.0, 1.0)
        if n_p % 2 != 0:
            n_p_adj = n_p + (1 if n_p % 2 == 1 else 0)
            h_parent = transfer_to_n(h_parent, n_p_adj)
        best_h = h_parent.copy()
        best_c5 = eval_score(h_parent)
    else:
        n_p = 126
        m_p = n_p // 2
        x = np.linspace(0, 1, m_p, endpoint=False)
        q_p = np.clip(0.5 + 0.35 * np.sin(2 * np.pi * x), 0, 1)
        best_h = build_sym_h(q_p)
        best_c5 = eval_score(best_h)

    n_cur = len(best_h)
    q_cur = sym_from_h(best_h)

    # Phase 1: Coordinate descent (main new method)
    dl1 = t0 + budget_s * 0.25
    q_cur, sc = coord_sweep_q(q_cur, n_cur, 80, rng, dl1)
    h_new = build_sym_h(q_cur)
    if sc < best_c5:
        best_c5 = sc
        best_h = h_new

    # Phase 2: Coordinate descent with perturbations
    if rem() > 150:
        dl2 = t0 + budget_s * 0.40
        for dist in [0.3, 0.5, 0.7, 0.9]:
            if time.time() > dl2:
                break
            q_cur, sc = coord_sweep_perturb(q_cur, n_cur, 8, dist, rng, dl2)
            h_new = build_sym_h(q_cur)
            if sc < best_c5:
                best_c5 = sc
                best_h = h_new

    # Phase 3: Smooth max optimization
    if rem() > 100:
        dl3 = t0 + budget_s * 0.55
        q_cur = sym_from_h(best_h)
        q_opt, c5_s, h_s = smooth_sym_optimize(q_cur, n_cur, [50, 100, 200, 400, 800], 200, dl3)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s

    # Phase 4: Coordinate descent after smooth max
    if rem() > 80:
        q_cur = sym_from_h(best_h)
        dl4 = t0 + budget_s * 0.65
        q_cur, sc = coord_sweep_q(q_cur, n_cur, 60, rng, dl4)
        h_new = build_sym_h(q_cur)
        if sc < best_c5:
            best_c5 = sc
            best_h = h_new

    # Phase 5: Epigraph refinement
    if rem() > 60:
        dl5 = t0 + budget_s * 0.75
        h_new, c5_new = do_epig_sym_loop(best_h, 20, 50, dl5)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

    # Phase 6: More coordinate descent with smaller perturbations
    if rem() > 60:
        q_cur = sym_from_h(best_h)
        dl6 = t0 + budget_s * 0.85
        q_cur, sc = coord_sweep_perturb(q_cur, n_cur, 15, 0.4, rng, dl6)
        h_new = build_sym_h(q_cur)
        if sc < best_c5:
            best_c5 = sc
            best_h = h_new
        # Additional sweeps
        q_cur, sc = coord_sweep_q(q_cur, n_cur, 50, rng, dl6)
        h_new = build_sym_h(q_cur)
        if sc < best_c5:
            best_c5 = sc
            best_h = h_new

    # Phase 7: Multi-resolution
    if rem() > 80:
        n_base = len(best_h)
        n_candidates = []
        for delta in [-6, -4, -2, 2, 4, 6]:
            nc = n_base + delta
            if 60 <= nc <= 200 and nc % 2 == 0:
                n_candidates.append(nc)
        # Also try halved
        if n_base // 2 >= 40:
            n_candidates.append(n_base // 2)
        n_candidates = sorted(set(n_candidates))

        for n_try in n_candidates:
            if rem() < 40:
                break
            h_try = transfer_to_n(best_h, n_try)
            n_try = len(h_try)
            if n_try % 2 != 0:
                continue
            q_try = sym_from_h(h_try)
            # Quick coordinate descent
            dl_r = min(t0 + budget_s * 0.95, time.time() + 40)
            q_try, sc_try = coord_sweep_q(q_try, n_try, 40, rng, dl_r)
            h_try = build_sym_h(q_try)
            if sc_try < best_c5:
                best_c5 = sc_try
                best_h = h_try
            # Smooth max on this resolution
            if rem() > 60:
                q_try, sc_try2, h_try2 = smooth_sym_optimize(q_try, n_try, [100, 300], 150,
                                                             min(t0 + budget_s * 0.95, time.time() + 35))
                if sc_try2 < best_c5:
                    best_c5 = sc_try2
                    best_h = h_try2
                # Coordinate descent on result
                if rem() > 40:
                    q_try2 = sym_from_h(best_h)
                    q_try2, sc_try3 = coord_sweep_q(q_try2, len(best_h), 30, rng,
                                                    min(t0 + budget_s * 0.95, time.time() + 20))
                    h_try3 = build_sym_h(q_try2)
                    if sc_try3 < best_c5:
                        best_c5 = sc_try3
                        best_h = h_try3

    # Phase 8: Diverse structured restarts on best resolution
    if rem() > 60:
        n_best = len(best_h)
        modes = [0, 1, 2, 3]
        for mode in modes:
            if rem() < 30:
                break
            q_start = make_structured_start(n_best, rng, mode)
            q_start, sc_r = coord_sweep_q(q_start, n_best, 50, rng,
                                          min(t0 + budget_s * 0.97, time.time() + 30))
            h_r = build_sym_h(q_start)
            if sc_r < best_c5:
                best_c5 = sc_r
                best_h = h_r

    # Phase 9: Final polish
    if rem() > 20 and best_h is not None:
        n_fin = len(best_h)
        if n_fin % 2 == 0:
            q_fin = sym_from_h(best_h)
            # Final coordinate sweeps
            q_fin, sc_fin = coord_sweep_q(q_fin, n_fin, 100, rng, t0 + budget_s * 0.99)
            h_fin = build_sym_h(q_fin)
            if sc_fin < best_c5:
                best_c5 = sc_fin
                best_h = h_fin
            # Epigraph final
            if rem() > 15:
                h_fin, c5_fin = do_epig_sym_loop(best_h, 10, 40, t0 + budget_s * 0.99)
                if c5_fin < best_c5:
                    best_c5 = c5_fin
                    best_h = h_fin
            # One more coord sweep
            if rem() > 10:
                q_fin = sym_from_h(best_h)
                q_fin, sc_fin = coord_sweep_q(q_fin, n_fin, 50, rng, t0 + budget_s * 0.995)
                h_fin = build_sym_h(q_fin)
                if sc_fin < best_c5:
                    best_c5 = sc_fin
                    best_h = h_fin

    # Final validation
    if best_h is None:
        m_fb = 63
        x_fb = np.linspace(0, 1, m_fb, endpoint=False)
        q_fb = np.clip(0.5 + 0.3 * np.sin(2 * np.pi * x_fb), 0, 1)
        best_h = build_sym_h(q_fb)

    best_h = np.asarray(best_h, dtype=float).reshape(-1)
    n_out = len(best_h)
    best_h = project_feasible(best_h, n_out / 2.0)
    best_c5 = eval_score(best_h)

    return best_h, best_c5, n_out
