def repair_feasible(vec):

    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity, kind='stable')
        remaining = diff

            idx = order[i]
            add = min(remaining, capacity[idx])
            h[idx] += add
            remaining -= add
            if remaining <= 1e-15:
                break
    else:
        order = np.argsort(h, kind='stable')
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-15:

        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)

def symmetrize(h):
    return (h + h[::-1]) / 2.0

def build_gradient_matrix(h, n, dx):
    M = 2 * n - 1
    G = np.zeros((M, n))
    for m in range(M):
        d = m - (n - 1)
        if d == 0:

        elif d > 0:
            G[m, d:] = dx * (1.0 - h[:n-d])
            G[m, :n-d] = -dx * h[d:]
        else:
            dp = -d
            G[m, :n-dp] = dx * (1.0 - h[dp:])
            G[m, dp:] = -dx * h[:n-dp]

        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-15:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def symmetrize(h):
    return (h + h[::-1]) / 2.0

def build_gradient_matrix(h, n, dx):
    M = 2 * n - 1
    G = np.zeros((M, n))
    for m in range(M):
        d = m - (n - 1)
        if d == 0:
            G[m, :] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            G[m, d:] = dx * (1.0 - h[:n - d])
            G[m, :n - d] = -dx * h[d:]
        else:
            dp = -d
            G[m, :n - dp] = dx * (1.0 - h[dp:])
            G[m, dp:] = -dx * h[:n - dp]
    return G

def lse_objective(x, h, beta, lam):
    n = len(h)
    profile = compute_profile(x)
    return logsumexp(beta * profile) / beta + lam * (x.sum() - n / 2.0) ** 2

def lse_gradient(x, h, beta, lam):
    n = len(h)
    profile = compute_profile(x)
    shifted = beta * profile - beta * profile.max()
    w = np.exp(shifted)
    w /= w.sum()
    dx = 2.0 / n
    G = build_gradient_matrix(x, n, dx)
    g = G.T @ w
    g += 2.0 * lam * (x.sum() - n / 2.0) / n
    return g

def lsbfgs_lse(h_start, beta, lam=300.0, maxiter=100, time_limit=25.0):
    n = len(h_start)

    def obj(x):
        return lse_objective(x, h_start, beta, lam)

    def grad(x):
        return lse_gradient(x, h_start, beta, lam)

    t0 = time.perf_counter()
    maxfun = int(min(maxiter * 5, max(200, time_limit * 80)))
    try:
        res = minimize(obj, h_start.copy(), method='L-BFGS-B', jac=grad,
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'maxfun': maxfun,
                                'ftol': 1e-15, 'gtol': 1e-9, 'disp': False})
        if res.x is not None and np.isfinite(res.x).all():
            h_new = repair_feasible(res.x)
            return h_new, exact_score(h_new)
    except Exception:
        pass
    return h_start.copy(), exact_score(h_start)

def slsqp_lse(h_start, beta=200, maxiter=200):
    n = len(h_start)

    def obj(x):
        profile = compute_profile(x)
        return logsumexp(beta * profile) / beta

    def obj_grad(x):
        profile = compute_profile(x)
        shifted = beta * profile - beta * profile.max()
        w = np.exp(shifted)
        w /= w.sum()
        dx = 2.0 / n
        G = build_gradient_matrix(x, n, dx)
        return G.T @ w

    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h_start.copy(), method='SLSQP', jac=obj_grad,
                       bounds=[(0.0, 1.0)] * n, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-13})
        if res.x is not None and np.isfinite(res.x).all():
            h_new = repair_feasible(res.x)
            c_new = exact_score(h_new)
            if c_new <= exact_score(h_start) + 1e-9:
                return h_new
    except Exception:
        pass
    return None

def solve_epigraph(h_start, maxiter=250, active_indices=None):
    n = len(h_start)
    dx = 2.0 / n
    M = 2 * n - 1
    c0 = exact_score(h_start)

    if active_indices is None:
        active = np.arange(M)
    else:
        active = np.asarray(active_indices)

    def obj(x):
        return x[-1]

    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g

    def eq_fun(x):
        return x[:-1].sum() - n / 2.0

    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g

    def ineq_fun(x):
        h = x[:-1]
        t = x[-1]
        profile = np.correlate(h, 1 - h, mode="full") * dx
        return t - profile[active]

    def ineq_jac(x):
        h = x[:-1]
        G = build_gradient_matrix(h, n, dx)
        J = np.zeros((len(active), n + 1))
        J[:, :-1] = -G[active, :]
        J[:, -1] = 1.0
        return J

    x0 = np.concatenate([h_start, [c0 + 1e-5]])
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 1.0)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-13})
        if res.x is not None and np.isfinite(res.x).all():
            h_new = repair_feasible(res.x[:-1])
            c_new = exact_score(h_new)
            if c_new <= c0 + 1e-10:
                return h_new
    except Exception:
        pass
    return None

def active_set_epigraph(h_start, maxiter=200, top_k=10):
    n = len(h_start)
    profile = compute_profile(h_start)
    c = profile.max()
    thresh = c - max(0.005, 4.0 * (c - np.median(profile)))
    active = np.where(profile >= thresh)[0]
    if len(active) > top_k:
        idx = np.argsort(profile[active])[-top_k:]
        active = active[idx]
    return solve_epigraph(h_start, maxiter=maxiter, active_indices=active.tolist())

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

def coordinate_refine(h, max_sweeps=5, time_limit=25.0):
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    t0 = time.perf_counter()
    grid = np.linspace(0.0, 1.0, 30)
    for sweep in range(max_sweeps):
        if time.perf_counter() - t0 > time_limit:
            break
        improved = False
        for i in range(n):
            if time.perf_counter() - t0 > time_limit:
                break
            old_val = h[i]
            best_val = old_val
            best_s = best_c
            for x in grid:
                if abs(x - old_val) < 1e-8:
                    continue
                h[i] = x
                h_r = repair_feasible(h)
                if np.max(np.abs(h_r[i] - x)) > 0.05:
                    h[i] = old_val
                    continue
                s = exact_score(h_r)
                if s < best_s - 1e-14:
                    best_s = s
                    best_val = x
                    best_hr = h_r.copy()
            if best_s < best_c - 1e-14:
                h = best_hr
                best_c = best_s
                improved = True
            else:
                h[i] = old_val
        if not improved:
            break
    return h, best_c

def local_gradient_descent(h, max_iters=80, time_limit=20.0):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best_c = exact_score(h)
    t0 = time.perf_counter()
    alpha = 0.008
    for it in range(max_iters):
        if time.perf_counter() - t0 > time_limit:
            break
        profile = compute_profile(h)
        c = profile.max()
        thresh = c - max(0.004, 4.0 * (c - np.median(profile)))
        active = np.where(profile >= thresh)[0]
        if len(active) == 0:
            active = np.array([np.argmax(profile)])
        if len(active) > 12:
            top = np.argsort(profile[active])[-12:]
            active = active[top]
        G = build_gradient_matrix(h, n, dx)
        G_a = G[active, :]
        w = profile[active] - (c - 0.004)
        w = np.maximum(w, 0.01)
        w /= w.sum()
        grad = G_a.T @ w
        grad -= grad.mean()
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad /= gnorm
        improved = False
        for a in [alpha, alpha * 0.5, alpha * 0.25, alpha * 0.1]:
            h_try = repair_feasible(h - a * grad)
            s = exact_score(h_try)
            if s < best_c - 1e-13:
                h = h_try
                best_c = s
                improved = True
                break
        if not improved:
            alpha *= 0.6
            if alpha < 1e-7:
                break
        else:
            alpha = min(alpha * 1.15, 0.03)
    return h, best_c

def make_structured_start(n, rng):
    h = np.zeros(n)
    block = max(2, n // 8)
    for b in range(n // block + 1):
        s = b * block
        e = min(s + block, n)
        h[s:e] = rng.uniform(0.1, 0.9)
    return repair_feasible(symmetrize(h))

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s + t_start - time.perf_counter()

    try:
        parent = np.array(initial_h_values, dtype=np.float64)
    except (NameError, Exception):
        parent = np.full(64, 0.5)

    parent = np.clip(parent, 0.0, 1.0)
    parent = repair_feasible(parent)
    best_h = parent.copy()
    best_c = exact_score(parent)

    h_sym = repair_feasible(symmetrize(parent))
    c_sym = exact_score(h_sym)
    if c_sym < best_c - 1e-14:
        best_h = h_sym
        best_c = c_sym

    # Phase 1: L-BFGS-B log-sum-exp with increasing beta
    if time_left() > 120:
        betas = [15, 30, 60, 120, 250, 500, 1000, 2000]
        for beta in betas:
            if time_left() < 35:
                break
            tl = min(20.0, max(5.0, (time_left() - 30) / 4.0))
            h_try, c_try = lsbfgs_lse(best_h, beta=beta, lam=400.0,
                                       maxiter=60, time_limit=tl)
            if c_try < best_c - 1e-12:
                best_h = h_try
                best_c = c_try

    # Phase 2: SLSQP log-sum-exp with exact constraint
    if time_left() > 80:
        for beta in [150, 400, 1000]:
            if time_left() < 35:
                break
            h_sol = slsqp_lse(best_h, beta=beta, maxiter=150)
            if h_sol is not None:
                c_sol = exact_score(h_sol)
                if c_sol < best_c - 1e-12:
                    best_h = h_sol
                    best_c = c_sol

    # Phase 3: Full epigraph SLSQP
    if time_left() > 50:
        h_sol = solve_epigraph(best_h, maxiter=250, active_indices=None)
        if h_sol is not None:
            c_sol = exact_score(h_sol)
            if c_sol < best_c - 1e-12:
                best_h = h_sol
                best_c = c_sol

    # Phase 4: Active set epigraph (multiple K values)
    if time_left() > 50:
        for K in [6, 10, 15, 20]:
            if time_left() < 15:
                break
            h_sol = active_set_epigraph(best_h, maxiter=200, top_k=K)
            if h_sol is not None:
                c_sol = exact_score(h_sol)
                if c_sol < best_c - 1e-12:
                    best_h = h_sol
                    best_c = c_sol

    # Phase 5: Local gradient descent
    if time_left() > 25:
        h_try, c_try = local_gradient_descent(best_h, max_iters=80, time_limit=15.0)
        if c_try < best_c - 1e-13:
            best_h = h_try
            best_c = c_try

    # Phase 6: Coordinate refinement
    if time_left() > 20:
        h_try, c_try = coordinate_refine(best_h, max_sweeps=3, time_limit=15.0)
        if c_try < best_c - 1e-13:
            best_h = h_try
            best_c = c_try

    # Phase 7: Dimension exploration
    if time_left() > 70:
        n_best = len(best_h)
        dims = sorted(set(
            [n_best - 4, n_best - 2, n_best + 2, n_best + 4,
             n_best + 6, n_best + 10, n_best + 14,
             max(24, n_best - 8), min(160, n_best + 20),
             n_best * 2 if n_best * 2 <= 200 else n_best,
             n_best // 2 if n_best // 2 >= 20 else n_best,
             40, 44, 48, 52, 56, 60, 64, 68, 72, 76, 80, 84, 88, 92, 96, 100, 104, 108]
        ), key=lambda d: abs(d - n_best) if 16 <= d <= 200 else 999)
        dims = [d for d in dims if 16 <= d <= 200 and d != n_best and d % 2 == 0]

        for new_n in dims[:8]:
            if time_left() < 35:
                break
            h_interp = interpolate_to_n(best_h, new_n)
            # Quick L-BFGS-B
            h_t, c_t = lsbfgs_lse(h_interp, beta=300, lam=300.0,
                                   maxiter=40, time_limit=10.0)
            # Active set epigraph
            h_sol = active_set_epigraph(h_t, maxiter=150, top_k=10)
            if h_sol is not None:
                h_t = h_sol
                c_t = exact_score(h_t)
            if c_t < best_c - 1e-10:
                best_h = h_t
                best_c = c_t

    # Phase 8: Multi-start perturbations
    if time_left() > 60:
        n_b = len(best_h)
        perturbations = []
        for eps in [0.003, 0.008, 0.015, 0.03, 0.05]:
            z = rng.randn(n_b)
            perturbations.append(repair_feasible(symmetrize(best_h + eps * z)))
        for h_p in perturbations:
            if time_left() < 20:
                break
            h_t, c_t = lsbfgs_lse(h_p, beta=500, lam=300.0, maxiter=40, time_limit=12.0)
            h_sol = solve_epigraph(h_t, maxiter=150, active_indices=None)
            if h_sol is not None:
                h_t = h_sol
                c_t = exact_score(h_t)
            if c_t < best_c - 1e-12:
                best_h = h_t
                best_c = c_t

    # Phase 9: Second round of refinement on best
    if time_left() > 50:
        # L-BFGS-B with high beta
        if time_left() > 30:
            h_t, c_t = lsbfgs_lse(best_h, beta=3000, lam=500.0,
                                   maxiter=60, time_limit=15.0)
            if c_t < best_c - 1e-13:
                best_h = h_t
                best_c = c_t
        # SLSQP epigraph
        if time_left() > 25:
            h_sol = solve_epigraph(best_h, maxiter=300, active_indices=None)
            if h_sol is not None:
                c_sol = exact_score(h_sol)
                if c_sol < best_c - 1e-13:
                    best_h = h_sol
                    best_c = c_sol
        # Local gradient
        if time_left() > 15:
            h_t, c_t = local_gradient_descent(best_h, max_iters=60, time_limit=10.0)
            if c_t < best_c - 1e-13:
                best_h = h_t
                best_c = c_t

    # Phase 10: Final micro-refinement
    if time_left() > 15:
        h_t, c_t = local_gradient_descent(best_h, max_iters=50, time_limit=8.0)
        if c_t < best_c - 1e-14:
            best_h = t
            best_c = c_t
        h_t2, c_t2 = coordinate_refine(best_h, max_sweeps=2, time_limit=5.0)
        if c_t2 < best_c - 1e-14:
            best_h = h_t2
            best_c = c_t2

    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)

    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            final_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 2:
            final_c = float(eval_result[0])
    except Exception:
        pass

    return best_h, final_c, len(best_h)
