from scipy.optimize import linprog
import time

def eval_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def corr_C(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def repair_box_sum(h, target):
    n = len(h)
    h = np.asarray(h, dtype=float)
    lo, hi = -3.0, 3.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid

        mask = (h_new > 1e-12) & (h_new < 1.0 - 1e-12)
        if mask.any():
            h_new[mask] += diff / mask.sum()
    return np.clip(h_new, 0.0, 1.0)

def build_sym(q, n):
    m = n // 2
    h = np.empty(n)
    h[:m] = q
    h[m:] = 1.0 - q[::-1]
    return h

def to_sym_q(h, n):
    m = n // 2
    return 0.5 * (h[:m] + 1.0 - h[n-1:m-1:-1])

def grad_cj_scaled(h, j, n):
    """Gradient of C[j] = c_j * dx with respect to h, already scaled by dx"""
    dx = 2.0 / n
    d = j - (n - 1)
    g = np.zeros(n)
    lo = max(0, d)
    hi = min(n - 1, n - 1 + d)
    if lo <= hi:
        g[lo:hi+1] += 1.0 - h[lo-d:hi+1-d]
        g[lo-d:hi+1-d] -= h[lo:hi+1]
    return dx * g

def grad_q_j(h, j, n):
    """Gradient of C[j] with respect to q variables (anti-complementary)"""
    m = n // 2
    gh = grad_cj_scaled(h, j, n)
    gq = gh[:m] - gh[n-1:m-1:-1]
    return gq

def select_active(C, margin_frac=0.01, pad=2):
    """Select active constraints (near-maximum lags)"""
    cmax = C.max()
    thresh = cmax - margin_frac * cmax
    idx = np.where(C >= thresh)[0]
    # Pad with neighbors
    p = 2*n-2
    for a in list(idx):
        for delta in range(-pad, pad+1):
            k = a + delta
            if 0 <= k <= p:
                if k not in idx:
                    idx = np.append(idx, k)
    return np.sort(np.unique(idx))

def slp_sym_step(q, h, C, active, n, R=0.05, move_pen=0.01):
    """One SLP step in the anti-complementary space. Returns new q or None."""
    m = n // 2
    # Build gradient matrix for active constraints
    # Each row is grad_q_j for active j
    G = np.zeros((len(active), m))
    c_vals = np.zeros(len(active))
    for i, j in enumerate(active):
        G[i] = grad_q_j(h, j, n)
        c_vals[i] = C[j]
    
    cmax = C.max()
    
    # LP variables: [delta_0, ..., delta_{m-1}, s_0, ..., s_{m-1}, t]
    # delta: changes to q, s: slack for |delta|, t: new upper bound
    nvar = 2*m + 1
    delta_vars = slice(0, m)
    slack_vars = slice(m, 2*m)
    t_var = 2*m
    
    # Objective: minimize t + move_pen * sum(s_i)
    c_obj = np.zeros(nvar)
    c_obj[t_var] = 1.0
    c_obj[slack_vars] = move_pen
    
    # Inequality constraints: G @ delta <= t - c_vals
    # => G @ delta - t <= -c_vals
    A_ub = np.zeros((len(active), nvar))
    A_ub[:, delta_vars] = G
    A_ub[:, t_var] = -1.0
    b_ub = -c_vals
    
    # Slack constraints: delta_i <= s_i, -delta_i <= s_i
    # => delta_i - s_i <= 0, -delta_i - s_i <= 0
    A_slack = np.zeros((2*m, nvar))
    for i in range(m):
        A_slack[2*i, i] = 1.0
        A_slack[2*i, m+i] = -1.0
        A_slack[2*i+1, i] = -1.0
        A_slack[2*i+1, m+i] = -1.0
    b_slack = np.zeros(2*m)
    
    A_full = np.vstack([A_ub, A_slack])
    b_full = np.concatenate([b_ub, b_slack])
    
    # Bounds
    bounds = []
    for i in range(m):
        lo = max(-q[i], -R)
        hi = min(1.0 - q[i], R)
        if lo < hi:
            bounds.append((lo, hi))
        else:
            bounds.append((0.0, 0.0))
    bounds += [(0.0, 4*R)] * m  # slack bounds
    bounds.append((0.0, cmax + 0.5))  # t bound
    
    result = linprog(c_obj, A_ub=A_full, b_ub=b_full, bounds=bounds, method='highs')
    
    if not result.success:
        return None
    
    delta = np.clip(result.x[delta_vars], -R, R)
    return delta

def eval_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))


def corr_C(h):

    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)


def repair_box_sum(h, target):
    n = len(h)
    h = np.asarray(h, dtype=float)
    lo, hi = -3.0, 3.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h_new = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - h_new.sum()
    if abs(diff) > 1e-14:
        mask = (h_new > 1e-12) & (h_new < 1.0 - 1e-12)
        if mask.any():
            h_new[mask] += diff / mask.sum()
    return np.clip(h_new, 0.0, 1.0)


def build_sym(q, n):
    m = n // 2
    h = np.empty(n)
    h[:m] = q
    h[m:] = 1.0 - q[::-1]
    return h


def to_sym_q(h, n):
    m = n // 2
    return 0.5 * (h[:m] + 1.0 - h[n - 1:m - 1:-1])


def grad_cj_scaled(h, j, n):
    dx = 2.0 / n
    d = j - (n - 1)
    g = np.zeros(n)
    lo = max(0, d)
    hi = min(n - 1, n - 1 + d)
    if lo <= hi:
        g[lo:hi + 1] += 1.0 - h[lo - d:hi + 1 - d]
        g[lo - d:hi + 1 - d] -= h[lo:hi + 1]
    return dx * g


def grad_q_j(h, j, n):
    m = n // 2
    gh = grad_cj_scaled(h, j, n)
    return gh[:m] - gh[n - 1:m - 1:-1]


def select_active(C, margin_frac=0.02, pad=2):
    cmax = C.max()
    thresh = cmax - margin_frac * cmax
    idx = np.where(C >= thresh)[0]
    p = len(C) - 1
    padded = set(idx.tolist())
    for a in idx:
        for delta in range(-pad, pad + 1):
            k = a + delta
            if 0 <= k <= p:
                padded.add(k)
    return np.sort(np.array(padded))

    c_obj[2 * m] = 1.0
    c_obj[m:2 * m] = move_pen
    A_ub = np.zeros((len(active) + 2 * m, nvar))
    A_ub[:len(active), :m] = G
    A_ub[:len(active), 2 * m] = -1.0

    for i in range(m):

        A_ub[len(active) + 2 * i, m + i] = -1.0
        A_ub[len(active) + 2 * i + 1, i] = -1.0
        A_ub[len(active) + 2 * i + 1, m + i] = -1.0

            best_c5 = c_chk

    if time_left() > reserve + 50:
        phase2_deadline = time.time() + min(120.0, 0.18 * (budget_s - (time.time() - t0) - reserve))
        h_f = slp_loop_full(best_h.copy(), R_init=0.02, max_iter=80, t_deadline=phase2_deadline)
        c_f = eval_c5(h_f)
        if c_f < best_c5:
            best_h = h_f.copy()
            best_c5 = c_f

    # Phase 3: Multi-resolution
    if time_left() > reserve + 30:
        n_base = len(best_h)
        n_candidates = set()
        for d in [-8, -6, -4, -3, -2, -1, 1, 2, 3, 4, 6, 8]:
            nc = n_base + d
            if 40 <= nc <= 260 and nc % 2 == 0:
                n_candidates.add(nc)
        n_candidates = sorted(n_candidates)

        for n_try in n_candidates:
            if time_left() < reserve + 15:
                break
            if n_try == n_base:
                continue
            h_try = interp_to(best_h, n_try)
            q_try = to_sym_q(h_try, n_try)
            phase3_deadline = min(time.time() + 40, t0 + budget_s - reserve - 10)
            q_try = slp_loop_sym(q_try, n_try, R_init=0.03, max_iter=50, t_deadline=phase3_deadline)
            h_final = build_sym(q_try, n_try)
            c_final = eval_c5(h_final)
            if c_final < best_c5:
                best_h = h_final.copy()
                best_c5 = c_final

    # Phase 4: Random restarts
    if time_left() > reserve + 25:
        n_r = len(best_h)
        if n_r % 2 != 0:
            n_r -= 1
        max_restarts = min(15, max(2, int((time_left() - reserve) / 20.0)))
        for _ in range(max_restarts):
            if time_left() < reserve + 10:
                break
            q_r = make_random_q(n_r, rng)
            phase4_deadline = min(time.time() + 15, t0 + budget_s - reserve - 5)
            q_r = slp_loop_sym(q_r, n_r, R_init=0.04, max_iter=40, t_deadline=phase4_deadline)
            h_r = build_sym(q_r, n_r)
            c_r = eval_c5(h_r)
            if c_r < best_c5:
                best_h = h_r.copy()
                best_c5 = c_r

    # Phase 5: Final high-precision polish
    if time_left() > 8:
        n_fin = len(best_h)
        if n_fin % 2 == 0:
            q_fin = to_sym_q(best_h, n_fin)
            phase5_deadline = min(time.time() + 30, t0 + budget_s - 4)
            q_fin = slp_loop_sym(q_fin, n_fin, R_init=0.008, max_iter=60, t_deadline=phase5_deadline, move_pen_init=0.003)
            h_pol = build_sym(q_fin, n_fin)
            c_pol = eval_c5(h_pol)
            if c_pol < best_c5:
                best_h = h_pol.copy()
                best_c5 = c_pol
        if time_left() > 5:
            h_fin = slp_loop_full(best_h.copy(), R_init=0.01, max_iter=40, t_deadline=t0 + budget_s - 2)
            c_fin = eval_c5(h_fin)
            if c_fin < best_c5:
                best_h = h_fin.copy()
                best_c5 = c_fin

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = repair_box_sum(best_h, len(best_h) / 2.0)
    best_c5 = eval_c5(best_h)

    return best_h, best_c5, len(best_h)
