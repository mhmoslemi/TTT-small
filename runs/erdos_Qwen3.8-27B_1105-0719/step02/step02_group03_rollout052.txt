def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-10:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
        else:
            h += diff / n
    return np.clip(h, 0.0, 1.0)

def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def c5_of_h(h, dx):
    return float(compute_corrs(h).max() * dx)

def to_binary(h, n):
    m = n // 2
    idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[idx[:m]] = 1.0
    return h_bin

def make_golden(n, m, rng=None):
    h = np.zeros(n)
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    used = set()
    for k in range(m):
        pos = int((k * phi * n) % n)
        while pos in used:
            pos = (pos + 1) % n
        h[pos] = 1.0
        used.add(pos)
    return h

def make_thue_morse(n, m):
    # Generate Thue-Morse sequence and take first m positions as 1s
    # Thue-Morse: t(i) = popcount(i) mod 2
    h = np.zeros(n)
    tm = np.array([bin(i).count('1') % 2 for i in range(n)])
    # Positions where tm = 1 (or 0, whichever gives m positions)
    ones_in_tm = np.sum(tm == 1)
    if ones_in_tm >= m:
        h[tm == 1][:m] = 1.0  # This doesn't work correctly
    # Better: just use the Thue-Morse values directly if n is a power of 2
    # Otherwise, use a modified version
    # Simple approach: use the parity of popcount
    h = (np.array([bin(i).count('1') % 2 for i in range(n)]))
    # Adjust to have exactly m ones
    ones_count = h.sum()
    if ones_count > m:
        # Set some 1s to 0s
        one_indices = np.where(h == 1)[0]
        for idx in one_indices[m:]:
            h[idx] = 0
    elif ones_count < m:
        zero_indices = np.where(h == 0)[0]
        for idx in zero_indices[:m - ones_count]:
            h[idx] = 1
    return h

def make_random(n, m, rng):
    h = np.zeros(n)
    h[rng.choice(n, size=m, replace=False)] = 1.0
    return h

def make_symmetric_golden(n, m, rng=None):
    """Create a symmetric golden-ratio-like pattern."""
    h = np.zeros(n)
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    half = n // 2
    m_half = m // 2
    used = set()
    for k in range(m_half):
        pos = int((k * phi * half) % half)
        while pos in used:
            pos = (pos + 1) % half
        h[pos] = 1.0
        h[n-1-pos] = 1.0
        used.add(pos)
        used.add(n-1-pos)
    # Adjust to have exactly m ones
    ones_count = int(h.sum())
    if ones_count < m:
        zeros = np.where(h == 0)[0]
        for idx in zeros[:m - ones_count]:
            h[idx] = 1.0
    elif ones_count > m:
        ones = np.where(h == 1)[0]
        for idx in ones[m:]:
            h[idx] = 0.0
    return h

            corr_crit = corrs[critical_idx].reshape(1, 1, n_crit)

            da_3d = da.reshape(1, m_zeros, n_crit)
            new_vals = corr_crit + dr_3d + da_3d  # (m_ones, m_zeros, n_crit)
            new_maxes = new_vals.max(axis=2)  # (m_ones, m_zeros)
            
            # Find the best swap
            global_best = new_maxes.min()
            if global_best < max_corr - 1e-10:
                bi, bj = np.unravel_index(new_maxes.argmin(), new_maxes.shape)
                i, j = ones[bi], zeros[bj]
                h[i], h[j] = 0.0, 1.0
                corrs = compute_corrs(h)
                new_c5 = corrs.max() * dx
                if new_c5 < best_c5 - 1e-12:
                    best_c5 = new_c5
                    best_h = h.copy()
                    no_improve = 0
                    continue
                else:
                    no_improve += 1
                    continue
            else:
                no_improve += 1
                # Try random swap
                if rng.random() < 0.1:
                    i = ones[rng.randint(m_ones)]
                    j = zeros[rng.randint(m_zeros)]
                    h[i], h[j] = 0.0, 1.0
                    corrs = compute_corrs(h)
                    if corrs.max() * dx < best_c5:
                        best_c5 = corrs.max() * dx
                        best_h = h.copy()
                        no_improve = 0
                    else:
                        # Revert
                        h[i], h[j] = 1.0, 0.0
                        corrs = compute_corrs(h)
                        no_improve += 1
        else:
            # Too large, use random sampling
            for _ in range(1000):
                i = ones[rng.randint(m_ones)]
                j = zeros[rng.randint(m_zeros)]
                # Compute delta at critical shifts
                delta = np.zeros(n_crit)
                for c, s in enumerate(critical_shifts):
                    d = 0.0
                    if 0 <= i+s <= n-1: d += h[i+s] - 1.0
                    if 0 <= i-s <= n-1: d += h[i-s]
                    if 0 <= j+s <= n-1: d += 1.0 - h[j+s]
                    if 0 <= j-s <= n-1: d += -h[j-s]
                    delta[c] = d
                new_max = (corrs[critical_idx] + delta).max()
                if new_max < max_corr - 1e-10:

import numpy as np
from scipy.optimize import minimize
import time


def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-10:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
        else:
            h += diff / n
    return np.clip(h, 0.0, 1.0)


def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def c5_of_h(h, dx):
    return float(compute_corrs(h).max() * dx)


def to_binary(h, n):
    m = n // 2
    idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[idx[:m]] = 1.0
    return h_bin


def make_golden(n, m):
    h = np.zeros(n)
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    used = np.zeros(n, dtype=bool)
    placed = 0
    k = 0
    while placed < m and k < m * 10:
        pos = int((k * phi * n) % n)
        if not used[pos]:
            h[pos] = 1.0
            used[pos] = True
            placed += 1
        k += 1
    if placed < m:
        for i in range(n):
            if placed >= m:
                break
            if not used[i]:
                h[i] = 1.0
                used[i] = True
                placed += 1
    return h


def make_thue_morse(n, m):
    h = np.zeros(n)
    for i in range(n):
        h[i] = bin(i).count('1') % 2
    ones_count = int(h.sum())
    if ones_count > m:
        one_idx = np.where(h == 1)[0]
        h[one_idx[m:]] = 0.0
    elif ones_count < m:
        zero_idx = np.where(h == 0)[0]
        h[zero_idx[:m - ones_count]] = 1.0
    return h


def make_random(n, m, rng):
    h = np.zeros(n)
    h[rng.choice(n, size=m, replace=False)] = 1.0
    return h


def fast_binary_search(h0, dx, rng, time_limit, t0=None):
    if t0 is None:
        t0 = time.time()
    n = len(h0)
    m = n // 2
    h = to_binary(h0, n).copy()
    corrs = compute_corrs(h)
    best_c5 = corrs.max() * dx
    best_h = h.copy()

    no_improve = 0
    max_no_improve = 8000
    threshold = 1.0

    while no_improve < max_no_improve:
        if time.time() - t0 > time_limit:
            break

        max_corr = corrs.max()
        critical_mask = corrs >= max_corr - threshold
        critical_idx = np.where(critical_mask)[0]
        if len(critical_idx) < 3:
            critical_idx = np.argsort(-corrs)[:5]
        critical_shifts = critical_idx - (n - 1)
        n_crit = len(critical_shifts)

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        m1 = len(ones)
        m0 = len(zeros)

        if m1 == 0 or m0 == 0:
            break

        # Vectorized delta computation for critical shifts
        dr = np.zeros((m1, n_crit))
        da = np.zeros((m0, n_crit))

        for c in range(n_crit):
            s = critical_shifts[c]
            idx_ps = ones + s
            idx_ms = ones - s
            vps = (idx_ps >= 0) & (idx_ps < n)
            vms = (idx_ms >= 0) & (idx_ms < n)
            dr[:, c] = np.where(vps, h[idx_ps] - 1.0, 0.0) + np.where(vms, h[idx_ms], 0.0)

            idx_ps = zeros + s
            idx_ms = zeros - s
            vps = (idx_ps >= 0) & (idx_ps < n)
            vms = (idx_ms >= 0) & (idx_ms < n)
            da[:, c] = np.where(vps, 1.0 - h[idx_ps], 0.0) + np.where(vms, -h[idx_ms], 0.0)

                h[i], h[j] = 0.0, 1.0

def binary_sa(h0, dx, rng, time_limit, t0=None):
    if t0 is None:
        t0 = time.time()
    n = len(h0)
    h = to_binary(h0, n).copy()
    corrs = compute_corrs(h)
    best_c5 = corrs.max() * dx
    best_h = h.copy()

    no_improve = 0
    temp = 0.03
    cooling = 0.9998

    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]

    while no_improve < 15000:
        if time.time() - t0 > time_limit:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break

        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]

        # Compute delta at all shifts efficiently
        # shift = k - (n-1), k in [0, 2n-2]
        s_vals = np.arange(-(n - 1), n)
        # Remove i
        d = np.zeros(2 * n - 1)
        ip = i + s_vals
        im = i - s_vals
        v_p = (ip >= 0) & (ip < n)
        v_m = (im >= 0) & (im < n)
        d += np.where(v_p, h[ip] - 1.0, 0.0) + np.where(v_m, h[im], 0.0)
        # Add j
        jp = j + s_vals
        jm = j - s_vals
        v_p = (jp >= 0) & (jp < n)
        v_m = (jm >= 0) & (jm < n)
        d += np.where(v_p, 1.0 - h[jp], 0.0) + np.where(v_m, -h[jm], 0.0)

        new_corrs = corrs + d
        new_max = new_corrs.max() * dx
        delta_val = new_max - best_c5

        if delta_val < 1e-12:
            h[i], h[j] = 0.0, 1.0
            corrs = new_corrs
            if new_max < best_c5 - 1e-12:
                best_c5 = new_max
                best_h = h.copy()
            no_improve = 0
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            temp *= cooling
        else:
            if temp > 1e-8 and rng.random() < np.exp(-delta_val / temp):
                h[i], h[j] = 0.0, 1.0
                corrs = new_corrs
                if new_max < best_c5:
                    best_c5 = new_max
                    best_h = h.copy()
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
            no_improve += 1

    return best_h, best_c5


def slsqp_smooth(h0, dx, target_sum, beta, maxiter, t0=None, time_limit=60.0):
    if t0 is None:
        t0 = time.time()
    n = len(h0)

    def obj(x):
        if time.time() - t0 > time_limit:
            return 1e10
        corrs = compute_corrs(x)
        m = corrs.max()
        if beta < 10:
            return m * dx
        shifted = beta * (corrs - m)
        # Stable logsumexp
        shifted_max = shifted.max()
        ls = shifted_max + np.log(np.sum(np.exp(shifted - shifted_max)))
        return (m + ls / beta) * dx

    def obj_grad(x):
        if time.time() - t0 > time_limit:
            return np.zeros(n)
        n_local = len(x)
        corrs = compute_corrs(x)
        m = corrs.max()
        shifted = beta * (corrs - m)
        shifted_max = shifted.max()
        w = np.exp(shifted - shifted_max)
        w_sum = w.sum()
        w = w / w_sum
        # Gradient of each corr[k] w.r.t. h
        # corr[k] = sum_i h[i] * (1-h[i+k-(n-1)])
        # d corr[k] / d h[p] = (1-h[p+k-(n-1)]) if p+k-(n-1) valid
        #                     + (-h[p-(n-1-k)]) if p-(n-1-k) valid
        grad = np.zeros(n_local)
        for k in range(len(corrs)):
            if w[k] < 1e-15:
                continue
            shift = k - (n_local - 1)
            g = np.zeros(n_local)
            # Term 1: h[i]*(1-h[i+shift]), derivative w.r.t h[p]:
            #   if p is the i: (1-h[p+shift]) if 0<=p+shift<n
            #   if p is the i+shift, i.e., i=p-shift: -h[p-shift] if 0<=p-shift<n
            idx1 = np.arange(n_local) + shift
            valid1 = (idx1 >= 0) & (idx1 < n_local)
            g[valid1] += (1.0 - x[idx1[valid1]])
            idx2 = np.arange(n_local) - shift
            valid2 = (idx2 >= 0) & (idx2 < n_local)
            g[valid2] += (-x[idx2[valid2]])
            grad += w[k] * g
        return grad * dx

    def cons_fun(x):
        return x.sum() - target_sum

    def cons_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    bounds = [(0.0, 1.0)] * n

    try:
        res = minimize(obj, h0, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        h_out = project_feasible(res.x, target_sum)
    except Exception:
        h_out = project_feasible(h0, target_sum)

    return h_out, c5_of_h(h_out, dx)
