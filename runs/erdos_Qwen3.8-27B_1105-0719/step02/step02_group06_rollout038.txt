```python
from scipy.optimize import minimize, linprog
import time

def eval_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def corr_all(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def build_sym_h(q):
    m = len(q)
    h = np.empty(2 * m)
    h[:m] = q
    h[m:] = 1.0 - q[::-1]
    return h

def sym_from_h(h):
    n = len(h)
    m = n // 2
    q = 0.5 * (h[:m] + 1.0 - h[2*m-1:m-1:-1])
    return np.clip(q, 0.0, 1.0)

def grad_q_for_corrs(q, h, dx, active_k):
    """Compute gradient of active correlations w.r.t. q.
    Returns (m, len(active_k)) matrix where column j is grad_q C_{active_k[j]}.
    """
    m = len(q)
    n = 2 * m
    n_active = len(active_k)
    G = np.zeros((m, n_active))
    for j, k in enumerate(active_k):
        d = k - (n - 1)
        i_lo = max(0, -d)
        i_hi = min(n - 1, n - 1 - d)
        if i_lo > i_hi:
            continue
        idx = np.arange(i_lo, i_hi + 1)
        # g_h[idx] += (1 - h[idx + d])
        # g_h[idx + d] += -h[idx]
        # Map to q: gq[i] = g_h[i] (if i < m), gq[2m-1-i] -= g_h[i] (if i >= m)
        # Part 1: g_h[idx] += (1 - h[idx + d])
        # For i in idx: if i < m: G[i, j] += (1-h[i+d]); else G[2m-1-i, j] -= (1-h[i+d])
        mask1 = idx < m
        if mask1.any():
            G[idx[mask1], j] += (1.0 - h[idx[mask1] + d])
        if (~mask1).any():
            G[2*m-1-idx[~mask1], j] -= (1.0 - h[idx[~mask1] + d])
        # Part 2: g_h[idx + d] += -h[idx]
        idx2 = idx + d
        # For j2 in idx2: if j2 < m: G[j2, j] += (-h[j2-d]); else G[2m-1-j2, j] -= (-h[j2-d])

        if (~mask2).any():
            G[2*m-1-idx2[~mask2], j] -= (-h[idx[~mask2]])
    G *= dx
    return G

def slp_iteration(q, active_k, n_full):
    """One SLP iteration. Returns new q."""
    m = len(q)
    h = build_sym_h(q)
    dx = 2.0 / n_full
    n = n_full
    corrs = corr_all(h)
    
    # Compute gradients for active set
    G = grad_q_for_corrs(q, h, dx, active_k)
    # G shape: (m, len(active_k))
    # C values for active set
    C_active = corrs[active_k]
    
    # LP: min t
    # s.t. G[:, j] @ q_new - t <= -C_active[j] + G[:, j] @ q  for each j
    #      0 <= q_new <= 1
    # Variables: [q_new(0), ..., q_new(m-1), t]  -> total m+1
    # c = [0, ..., 0, 1] (minimize t)
    c_lp = np.zeros(m + 1)
    c_lp[m] = 1.0
    
    # A_ub @ var <= b_ub
    # For each active constraint j:
    # G[:, j] @ q_new - t <= rhs_j
    # where rhs_j = G[:, j] @ q - C_active[j]
    A_ub = np.zeros((len(active_k), m + 1))
    b_ub = np.zeros(len(active_k))
    for j in range(len(active_k)):
        A_ub[j, :m] = G[:, j]
        A_ub[j, m] = -1.0
        b_ub[j] = G[:, j] @ q - C_active[j]
    
    # Bounds: 0 <= q_new[j] <= 1 for all j, t >= corrs.max() - 0.01 (lower bound to prevent t going too low)
    # Actually, just t >= 0 is fine (or we can skip the lower bound on t)
    bounds = [(0.0, 1.0)] * m + [(0.0, None)]
    
    res = linprog(c_lp, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.status != 0:
        return q.copy()
    
    q_new = res.x[:m]
    return np.clip(q_new, 0.0, 1.0)

def slp_line_search(q, q_new, n_full, max_iters=50):
    """Backtracking line search: find alpha to minimize max_k C_k((1-alpha)*q + alpha*q_new)"""
    m = len(q)
    best_alpha = 0.0
    best_val = float('inf')
    
    for it in range(max_iters):
        alpha = min(1.0, 0.5 ** it)
        q_try = (1.0 - alpha) * q + alpha * q_new
        q_try = np.clip(q_try, 0.0, 1.0)
        h_try = build_sym_h(q_try)
        c = eval_score(h_try)
        if c < best_val:
            best_val = c
            best_alpha = alpha
        if alpha < 0.01:
            break
    return q + best_alpha * (q_new - q), best_val

def slp_optimize(q0, n_full, max_rounds=200, margin_frac=0.02, max_active=40):
    """SLP optimization loop."""
    m = len(q0)
    q = np.clip(q0.copy(), 0, 1)
    best_q = q.copy()
    best_c = eval_score(build_sym_h(q))
    
    for rnd in range(max_rounds):
        h = build_sym_h(q)
        corrs = corr_all(h)
        max_c = corrs.max()
        
        # Select active set
        threshold = max_c - margin_frac * max_c
        active = np.where(corrs >= threshold)[0].tolist()
        if len(active) > max_active:
            # Keep the top max_active
            active = sorted(active, key=lambda k: -corrs[k])[:max_active]
        
        # SLP step
        q_new = slp_iteration(q, active, n_full)
        
        # Line search
        q_ls, c_ls = slp_line_search(q, q_new, n_full)
        
        if c_ls < best_c - 1e-14:
            best_c = c_ls
            best_q = q_ls.copy()
            q = q_ls
        
        # Check convergence
        if np.max(np.abs(q_new - q)) < 1e-10:
            break
    
    return best_q, best_c
```

Let me now think about the SLSQP phase more carefully.

For the SLSQP polish, I'll use the epigraph formulation:
min t
s.t. C_k(q) <= t for k in A_active
     0 <= q <= 1

The key is to have the right active set. I'll use an iterative approach:
1. Start with A = top 10 correlations
2. Solve the SLSQP problem
3. Check if any correlation outside A now exceeds t
4. If yes, add it to A and re-solve
5. Repeat until stable

For the SLSQP function, I need to define the constraints. Since I can't use lambdas, I'll use a class:

```python
class EpigraphProblem:
    def __init__(self, active_k, n_full):
        self.active_k = active_k
        self.n_full = n_full
        self.m = n_full // 2
    
    def objective(self, x):
        return x[self.m]  # t
    
    def objective_grad(self, x):
        g = np.zeros(self.m + 1)
        g[self.m] = 1.0
        return g
    
    def constraint(self, x, k_idx):
        q = np.clip(x[:self.m], 0, 1)
        h = build_sym_h(q)
        t = x[self.m]
        n = self.n_full
        d = self.active_k[k_idx] - (n - 1)
        i_lo = max(0, -d)
        i_hi = min(n - 1, n - 1 - d)
        if i_lo > i_hi:
            return t
        idx = np.arange(i_lo, i_hi + 1)
        c_val = np.sum(h[idx] * (1.0 - h[idx + d])) * (2.0 / n)
        return t - c_val
    
    def constraint_grad(self, x, k_idx):
        q = np.clip(x[:self.m], 0, 1)
        h = build_sym_h(q)
        n = self.n_full
        m = self.m
        d = self.active_k[k_idx] - (n - 1)
        i_lo = max(0, -d)
        i_hi = min(n - 1, n - 1 - d)
        if i_lo > i_hi:
            g = np.zeros(m + 1)
            g[m] = 1.0
            return g
        idx = np.arange(i_lo, i_hi + 1)
        # Gradient of C_k w.r.t. h
        g_h = np.zeros(n)
        g_h[idx] += (1.0 - h[idx + d])
        g_h[idx + d] += (-h[idx])
        # Map to q
        gq = g_h[:m] - g_h[2*m-1:m-1:-1]
        g = np.zeros(m + 1)
        g[:m] = -(2.0/n) * gq
        g[m] = 1.0
        return g
```

```python
import numpy as np
from scipy.optimize import minimize, linprog
import time

_g_n = 0
_g_beta = 1.0

def eval_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def corr_all(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def build_sym_h(q):
    m = len(q)
    h = np.empty(2 * m)
    h[:m] = q
    h[m:] = 1.0 - q[::-1]
    return h

def sym_from_h(h):
    n = len(h)
    m = n // 2
    q = 0.5 * (h[:m] + 1.0 - h[2*m-1:m-1:-1])
    return np.clip(q, 0.0, 1.0)

def grad_q_active(h, active_k):
    n = len(h)
    m = n // 2
    dx = 2.0 / n
    na = len(active_k)
    G = np.zeros((m, na))
    for j, k in enumerate(active_k):
        d = k - (n - 1)
        i_lo = max(0, -d)
        i_hi = min(n - 1, n - 1 - d)
        if i_lo > i_hi:
            continue
        idx = np.arange(i_lo, i_hi + 1)
        m1 = idx < m
        if m1.any():
            G[idx[m1], j] += (1.0 - h[idx[m1] + d])
        if (~m1).any():
            G[2*m-1-idx[~m1], j] -= (1.0 - h[idx[~m1] + d])
        idx2 = idx + d
        m2 = idx2 < m
        if m2.any():
            G[idx2[m2], j] += (-h[idx[m2]])
        if (~m2).any():
            G[2*m-1-idx2[~m2], j] -= (-h[idx[~m2]])
    G *= dx
    return G

def smooth_max_needed(q, n_full, beta):
    m = len(q)
    h = build_sym_h(q)
    corrs = corr_all(h)
    max_c = corrs.max()
    shifted = corrs - max_c
    ev = np.exp(beta * shifted)
    ws = ev.sum()
    val = max_c + np.log(ws) / beta
    w = ev / ws
    sig = np.where(w > 1e-10)[0]
    if len(sig) == 0:
        grad = np.zeros(m)
    else:
        G = grad_q_active(h, sig.tolist())
        grad = G @ w[sig]
    return val, grad

def _sm_obj(x):
    v, _ = smooth_max_needed(np.clip(x, 0, 1), _g_n, _g_beta)
    return v

def _sm_grad(x):
    _, g = smooth_max_needed(np.clip(x, 0, 1), _g_n, _g_beta)
    return g

def slp_step(q, active_k, n_full):
    m = len(q)
    h = build_sym_h(q)
    dx = 2.0 / n_full
    corrs = corr_all(h)
    G = grad_q_active(h, active_k)
    C_act = corrs[active_k]
    c_lp = np.zeros(m + 1)
    c_lp[m] = 1.0
    A_ub = np.zeros((len(active_k), m + 1))
    b_ub = np.zeros(len(active_k))
    for j in range(len(active_k)):
        A_ub[j, :m] = G[:, j]
        A_ub[j, m] = -1.0
        b_ub[j] = G[:, j] @ q - C_act[j]
    bounds = [(0.0, 1.0)] * m + [(0.0, None)]
    try:
        res = linprog(c_lp, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.status != 0:
            return q.copy()
        return np.clip(res.x[:m], 0.0, 1.0)
    except Exception:
        return q.copy()

def slp_line_search(q, q_dir, n_full):
    best_a = 0.0
    h_cur = build_sym_h(q)
    best_v = eval_score(h_cur)
    for a in [1.0, 0.6, 0.4, 0.25, 0.15, 0.08, 0.04]:
        q_t = np.clip(q + a * q_dir, 0.0, 1.0)
        c_t = eval_score(build_sym_h(q_t))
        if c_t < best_v - 1e-15:
            best_v = c_t
            best_a = a
    if best_a == 0.0:
        return q, best_v
    return q + best_a * q_dir, best_v

def slp_optimize(q0, n_full, max_rounds=500, time_limit=None):
    m = len(q0)
    q = np.clip(q0.copy(), 0, 1)
    best_q = q.copy()
    best_c = eval_score(build_sym_h(q))
    stall = 0
    margin = 0.04
    
    for rnd in range(max_rounds):
        if time_limit is not None and time.time() > time_limit:
            break
        h = build_sym_h(q)
        corrs = corr_all(h)
        max_c = corrs.max()
        margin = max(0.005, 0.04 * (1.0 - rnd / max_rounds))
        threshold = max_c - margin * max_c
        active = np.where(corrs >= threshold)[0].tolist()
        if len(active) > 50:
            active = sorted(active, key=lambda k: -corrs[k])[:50]
        if len(active) < 3:
            active = np.argsort(corrs)[-5:].tolist()
        
        q_new = slp_step(q, active, n_full)
        q_ls, c_ls = slp_line_search(q, q_new - q, n_full)
        
        if c_ls < best_c - 1e-14:
            best_c = c_ls
            best_q = q_ls.copy()
            stall = 0
        else:
            stall += 1
            if stall > 30:
                q = np.clip(q + 0.01 * np.random.randn(m), 0, 1)
                stall = 0
        q = q_ls
    
    return best_q, best_c

class _EpiCon:
    def __init__(self, k, n_full):
        self.k = k
        self.n_full = n_full
        self.m = n_full // 2
    def fun(self, x):
        q = np.clip(x[:self.m], 0, 1)
        h = build_sym_h(q)
        t = x[self.m]
        n = self.n_full
        d = self.k - (n - 1)
        i_lo = max(0, -d)
        i_hi = min(n - 1, n - 1 - d)
        if i_lo > i_hi:
            return t
        idx = np.arange(i_lo, i_hi + 1)
        cv = np.sum(h[idx] * (1.0 - h[idx + d])) * (2.0 / n)
        return t - cv
    def jac(self, x):
        q = np.clip(x[:self.m], 0, 1)
        h = build_sym_h(q)
        n = self.n_full
        m = self.m
        d = self.k - (n - 1)
        i_lo = max(0, -d)
        i_hi = min(n - 1, n - 1 - d)
        g = np.zeros(m + 1)
        g[m] = 1.0
        if i_lo > i_hi:
            return g
        idx = np.arange(i_lo, i_hi + 1)
        g_h = np.zeros(n)
        g_h[idx] += (1.0 - h[idx + d])
        g_h[idx + d] += (-h[idx])
        g[:m] = -(2.0 / n) * (g_h[:m] - g_h[2*m-1:m-1:-1])
        return g

def _epi_obj(x):
    return x[-1]

def _epi_obj_grad(x):
    g = np.zeros(len(x))
    g[-1] = 1.0
    return g

def slsqp_epigraph(q0, n_full, active_k, maxiter=100):
    m = len(q0)
    h_c = build_sym_h(q0)
    corrs = corr_all(h_c)
    t0 = corrs.max() + 1e-8
    x0 = np.zeros(m + 1)
    x0[:m] = q0
    x0[m] = t0
    bounds = [(0.0, 1.0)] * m + [(0.0, None)]
    constraints = []
    for k in active_k:
        cw = _EpiCon(k, n_full)
        constraints.append({'type': 'ineq', 'fun': cw.fun, 'jac': cw.jac})
    try:
        res = minimize(_epi_obj, x0, jac=_epi_obj_grad,
                       method='SLSQP', bounds=bounds,
                       constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        q_out = np.clip(res.x[:m], 0.0, 1.0)
    except Exception:
        q_out = q0.copy()
    return q_out, eval_score(build_sym_h(q_out))

        q_slp2, c_slp2 = slp_optimize(best_q, n_full, max_rounds=300, time_limit=tl_slp2)
        if c_slp2 < best_c:
            best_c = c_slp2
            best_q = q_slp2.copy()
    
    # Phase 5: Try nearby n values
    if tleft() > 80:
        h_best = build_sym_h(best_q)
        n_base = len(h_best)
        for delta in [2, -2, 4, -4, 6, -6]:
            if tleft() < 40:
                break
            n_try = n_base + delta
            if n_try < 48 or n_try > 200 or n_try % 2 != 0:
                continue
            h_try = transfer_to_n(h_best, n_try)
            q_try = sym_from_h(h_try)
            n_try_full = len(q_try) * 2
            
            if tleft() > 60:
                for beta in [200, 500, 1000]:
                    if tleft() < 15:
                        break
                    _g_beta = beta
                    _g_n = n_try_full
                    try:
                        res = minimize(_sm_obj, q_try, jac=_sm_grad,
                                       method='L-BFGS-B',
                                       bounds=[(0.0, 1.0)] * len(q_try),
                                       options={'maxiter': 200, 'ftol': 1e-17})
                        q_try = np.clip(res.x, 0, 1)
                    except Exception:
                        pass
                    c_try = eval_score(build_sym_h(q_try))
                    if c_try < best_c:
                        best_c = c_try
                        best_q = q_try.copy()
                        n_full = n_try_full
                _g_n = n_full
            
            if tleft() > 40:
                tl_s = time.time() + min(50, tleft() * 0.6)
                q_s2, c_s2 = slp_optimize(q_try, n_try_full, max_rounds=200, time_limit=tl_s)
                if c_s2 < best_c:
                    best_c = c_s2
                    best_q = q_s2.copy()
                    n_full = n_try_full
                _g_n = n_full
    
    # Phase 6: Final polish
    if tleft() > 30:
        _g_n = len(best_q) * 2
        tl_fin = time.time() + min(50, tleft() * 0.7)
        q_f, c_f = epigraph_loop(best_q, len(best_q) * 2, max_rounds=20, time_limit=tl_fin)
        if c_f < best_c:
            best_c = c_f
            best_q = q_f.copy()
        
        if tleft() > 15:
            tl_f2 = time.time() + min(30, tleft() * 0.8)
            q_f2, c_f2 = slp_optimize(best_q, len(best_q) * 2, max_rounds=200, time_limit=tl_f2)
            if c_f2 < best_c:
                best_c = c_f2
                best_q = q_f2.copy()
    
    h_out = build_sym_h(best_q)
    n_out = len(h_out)
    c_out = eval_score(h_out)
    return h_out, c_out, n_out
```
