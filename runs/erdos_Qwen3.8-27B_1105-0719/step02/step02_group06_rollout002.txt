    h = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        interior = (h > 1e-10) & (h < 1.0 - 1e-10)
        if interior.sum() > 0:
            h[interior] += diff / interior.sum()
    return np.clip(h, 0.0, 1.0)

def corr_raw(h):
    return np.correlate(h, 1.0 - h, mode='full')

def smooth_max_q(q, m, beta):
    h = build_sym_h(q)
    n = 2 * m
    dx = 2.0 / n
    corrs = corr_raw(h) * dx
    max_c = corrs.max()
    shifted = corrs - max_c
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(len(corrs)) / len(corrs)
    else:
        w /= w_sum
    val = max_c + np.log(w_sum) / beta
    return val, w

def smooth_max_q_grad(q, m, beta):
    h = build_sym_h(q)
    n = 2 * m
    dx = 2.0 / n
    val, w = smooth_max_q(q, m, beta)
    grad_q = np.zeros(m)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        d = k - (n - 1)
        i_lo = max(0, -d)
        i_hi = min(n - 1, n - 1 - d)
        if i_lo > i_hi:
            continue
        idx = np.arange(i_lo, i_hi + 1)
        g_full = np.zeros(n)
        g_full[idx] += wk * (1.0 - h[idx + d])
        g_full[idx + d] += wk * (-h[idx])
        grad_q += g_full[:m] - g_full[n - 1 - np.arange(m)]
    grad_q *= dx
    return val, grad_q

def active_set_slsq(p, n_full, active_k, maxiter, ftol):
    m = n_full // 2
    h_c = build_sym_h(p)
    dx = 2.0 / n_full
    corrs_c = corr_raw(h_c) * dx
    t0_val = corrs_c.max() + 1e-9
    x0 = np.zeros(m + 1)
    x0[:m] = p
    x0[m] = t0_val
    bounds = [(0.0, 1.0)] * m + [(0.0, 2.0)]
    constraints = []
    for k in active_k:
        d = k - (n_full - 1)
        i_lo = max(0, -d)
        i_hi = min(n_full - 1, n_full - 1 - d)
        idx = np.arange(i_lo, i_hi + 1)
        constraints.append({
            'type': 'ineq',
            'fun': lambda x, kk=k, ii=idx, dd=d: x[m] - np.sum(x[ii] * (1.0 - x[ii + dd])) * dx,
            'jac': lambda x, kk=k, ii=idx, dd=d: _slep_jac(x, ii, dd, dx, m)
        })
    try:
        res = minimize(_slep_obj, x0, jac=_slep_jacob,
                       method='SLSQP', bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': ftol})
        q_out = np.clip(res.x[:m], 0.0, 1.0)
    except Exception:
        q_out = p.copy()
    h_out = build_sym_h(q_out)
    return q_out, eval_score(h_out)

def _slep_obj(x):
    return x[-1]

def _slep_jacob(x):
    n = len(x) - 1
    g = np.zeros(n + 1)
    g[n] = 1.0
    return g

def _slep_jac(x, idx, d, dx, m):
    h = build_sym_h(x[:m])
    g_full = np.zeros(2 * m)
    g_full[idx] = (1.0 - h[idx + d])
    g_full[idx + d] = -h[idx]
    j = np.zeros(m + 1)
    j[:m] = -dx * (g_full[:m] - g_full[2 * m - 1 - np.arange(m)])
    j[m] = 1.0
    return j

def projected_gradient_sym(q, m, beta_list, n_iter, base_lr):
    best_q = q.copy()
    best_val = smooth_max_q(q, m, beta_list[-1])[0]
    momentum = np.zeros(m)
    for beta in beta_list:
        lr = base_lr
        for it in range(n_iter):
            val, grad = smooth_max_q_grad(best_q, m, beta)
            momentum = 0.9 * momentum - lr * grad
            q_new = np.clip(best_q + momentum, 0.0, 1.0)
            val_new = smooth_max_q(q_new, m, beta)[0]
            if val_new < val:
                best_q = q_new
                val = val_new
                if lr < 0.05:
                    lr *= 1.05
            else:
                lr *= 0.7
                momentum = 0.5 * momentum
    return best_q

def transfer_to_n(h_old, n_new):
    n_old = len(h_old)
    x_old = (np.arange(n_old) + 0.5) * 2.0 / n_old
    x_new = (np.arange(n_new) + 0.5) * 2.0 / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new / 2.0)

def select_active(corrs_scaled, margin_frac=0.02, neighbor_pad=3, max_k=60):
    n = len(corrs_scaled)
    max_val = corrs_scaled.max()
    threshold = max_val - max(1e-8, margin_frac * max_val)
    active = set(np.where(corrs_scaled >= threshold)[0].tolist())
    padded = set()
    for k in active:
        for d in range(-neighbor_pad, neighbor_pad + 1):
            if 0 <= k + d < n:
                padded.add(k + d)
    active_list = sorted(padded)
    if len(active_list) > max_k:
        scored = sorted([(corrs_scaled[k], k) for k in active_list], reverse=True)
        active_list = sorted([k for _, k in scored[:max_k]])
    return active_list

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def rem():
        return budget_s - (time.time() - t0)

    best_h = None
    best_c5 = float('inf')

    try:
        h_parent = np.array(initial_h_values, dtype=float).copy().reshape(-1)
    except (NameError, AttributeError):
        h_parent = None

    if h_parent is not None and len(h_parent) > 10:
        n_p = len(h_parent)
        h_parent = np.clip(h_parent, 0.0, 1.0)
        h_parent = project_feasible(h_parent, n_p / 2.0)
        best_h = h_parent.copy()
        best_c5 = eval_score(h_parent)
    else:
        m_fb = 63
        x_fb = np.linspace(0, 1, m_fb)
        q_fb = (0.5 + 0.3 * np.sin(2 * np.pi * x_fb)).clip(0, 1)
        best_h = build_sym_h(q_fb)
        best_c5 = eval_score(best_h)

    n_cur = len(best_h)
    m_cur = n_cur // 2

    # Phase 1: L-BFGS-B smooth max with continuation
    if n_cur % 2 == 0 and rem() > 300:
        q_init = sym_from_h(best_h)
        for beta in [150, 400, 1000, 2500]:
            if rem() < 60:
                break
            def obj_fn(xx):
                return smooth_max_q(xx, m_cur, beta)[0]
            def jac_fn(xx):
                return smooth_max_q_grad(xx, m_cur, beta)[1]
            bounds = [(0.0, 1.0)] * m_cur
            try:
                res = minimize(obj_fn, q_init, jac=jac_fn, method='L-BFGS-B',
                               bounds=bounds,
                               options={'maxiter': 300, 'ftol': 1e-16, 'gtol': 1e-12})
                q_init = np.clip(res.x, 0.0, 1.0)
            except Exception:
                pass
        h_new = build_sym_h(q_init)
        c5_new = eval_score(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 2: Active set SLSQP (symmetric) - iterative
    if n_cur % 2 == 0 and rem() > 200:
        q_s = sym_from_h(best_h)
        for rnd in range(30):
            if rem() < 30:
                break
            h_chk = build_sym_h(q_s)
            corrs = corr_raw(h_chk) * (2.0 / n_cur)
            active = select_active(corrs, margin_frac=0.02, neighbor_pad=3, max_k=55)
            q_new, c5_new = active_set_slsq(q_s, n_cur, active, 80, 1e-13)
            h_new = build_sym_h(q_new)
            c5_new = eval_score(h_new)
            improved = False
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                best_h = h_new.copy()
                q_s = sym_from_h(h_new)
                improved = True
            # Check hidden violations
            corrs_new = corr_raw(h_new) * (2.0 / n_cur)
            max_act = corrs_new[active].max() if len(active) > 0 else 0
            hidden = np.where(corrs_new > max_act + 1e-8)[0]
            if len(hidden) > 0:
                active = sorted(set(active + list(hidden)))
                if len(active) > 100:
                    active = select_active(corrs_new, margin_frac=0.01, neighbor_pad=2, max_k=100)
                continue
            if not improved:
                # Try broader active set
                active_broad = select_active(corrs_new, margin_frac=0.06, neighbor_pad=5, max_k=70)
                if len(active_broad) > len(active):
                    q_nb, c5_nb = active_set_slsq(q_s, n_cur, active_broad, 50, 1e-12)
                    h_nb = build_sym_h(q_nb)
                    c5_nb = eval_score(h_nb)
                    if c5_nb < best_c5 - 1e-14:
                        best_c5 = c5_nb
                        best_h = h_nb.copy()
                        q_s = sym_from_h(h_nb)
                        continue
                break

    # Phase 3: Projected gradient with momentum (symmetric)
    if n_cur % 2 == 0 and rem() > 150:
        q_p = sym_from_h(best_h)
        for beta in [500, 1500, 4000]:
            if rem() < 40:
                break
            q_pg = projected_gradient_sym(q_p, m_cur, [beta], 200, 0.01 / (1.0 + beta * 0.001))
            h_pg = build_sym_h(q_pg)
            c5_pg = eval_score(h_pg)
            if c5_pg < best_c5:
                best_c5 = c5_pg
                best_h = h_pg.copy()
                q_p = q_pg
            else:
                q_p = sym_from_h(best_h)

    # Phase 4: Try nearby n values
    n_base = len(best_h)
    for delta in [-4, -2, -1, 1, 2, 4, 6]:
        if rem() < 80:
            break
        n_try = n_base + delta
        if n_try < 48 or n_try > 200 or n_try % 2 != 0:
            continue
        h_try = transfer_to_n(best_h, n_try)
        n_try = len(h_try)
        m_try = n_try // 2
        q_try = sym_from_h(h_try)
        # Quick L-BFGS-B
        for beta in [300, 1000]:
            if rem() < 40:
                break
            def obj_fn2(xx):
                return smooth_max_q(xx, m_try, beta)[0]
            def jac_fn2(xx):
                return smooth_max_q_grad(xx, m_try, beta)[1]
            bounds2 = [(0.0, 1.0)] * m_try
            try:
                res2 = minimize(obj_fn2, q_try, jac=jac_fn2, method='L-BFGS-B',
                                bounds=bounds2, options={'maxiter': 200, 'ftol': 1e-15})
                q_try = np.clip(res2.x, 0.0, 1.0)
            except Exception:
                pass
        h_try2 = build_sym_h(q_try)
        c5_try = eval_score(h_try2)
        # Quick active set
        if rem() > 50:
            corrs_t = corr_raw(h_try2) * (2.0 / n_try)
            active_t = select_active(corrs_t, margin_frac=0.02, neighbor_pad=3, max_k=50)
            q_at, c5_at = active_set_slsq(q_try, n_try, active_t, 60, 1e-13)
            h_at = build_sym_h(q_at)
            c5_at = eval_score(h_at)
            if c5_at < c5_try:
                c5_try = c5_at
                h_try2 = h_at
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try2.copy()

    # Phase 5: Final high-beta polish on best
    n_fin = len(best_h)
    if n_fin % 2 == 0 and rem() > 30:
        m_fin = n_fin // 2
        q_fin = sym_from_h(best_h)
        for beta in [2000, 5000, 10000]:
            if rem() < 20:
                break
            def obj_fn3(xx):
                return smooth_max_q(xx, m_fin, beta)[0]
            def jac_fn3(xx):
                return smooth_max_q_grad(xx, m_fin, beta)[1]
            bounds3 = [(0.0, 1.0)] * m_fin
            try:
                res3 = minimize(obj_fn3, q_fin, jac=jac_fn3, method='L-BFGS-B',
                                bounds=bounds3, options={'maxiter': 400, 'ftol': 1e-17, 'gtol': 1e-14})
                q_fin = np.clip(res3.x, 0.0, 1.0)
            except Exception:
                pass
        h_fin = build_sym_h(q_fin)
        c5_fin = eval_score(h_fin)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    # Phase 6: One more active set pass
    if n_fin % 2 == 0 and rem() > 20:
        q_fin2 = sym_from_h(best_h)
        h_chk2 = build_sym_h(q_fin2)
        corrs2 = corr_raw(h_chk2) * (2.0 / n_fin)
        active2 = select_active(corrs2, margin_frac=0.005, neighbor_pad=2, max_k=45)
        q_o, c5_o = active_set_slsq(q_fin2, n_fin, active2, 100, 1e-14)
        h_o = build_sym_h(q_o)
        c5_o = eval_score(h_o)
        if c5_o < best_c5:
            best_c5 = c5_o
            best_h = h_o

    # Validate and return
    best_h = np.asarray(best_h, dtype=float).reshape(-1)
    n_out = len(best_h)
    best_h = project_feasible(best_h, n_out / 2.0)
    best_c5 = eval_score(best_h)
    return best_h, best_c5, n_out
