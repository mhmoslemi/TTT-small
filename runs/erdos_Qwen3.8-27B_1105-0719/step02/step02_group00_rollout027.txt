        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx])
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:

        else:
            idx = int(np.argmax(h))

        h = np.clip(h, 0.0, 1.0)

            if j == i:

For the subgradient descent, I need the gradient of each row. Let me precompute which indices each row involves.

For row m with offset delta = m - (n-1):
- delta = 0: all indices, gradient = dx * (1 - 2*h)
- delta > 0: 
  - For index j: 
    - if j >= delta: gradient contribution = dx * (1 - h[j-delta])
    - if j < n-delta: gradient contribution += -dx * h[j+delta]
    (these can overlap)
- delta < 0 (d = -delta):
  - For index j:
    - if j < n-d: gradient contribution = dx * (1 - h[j+d])
    - if j >= d: gradient contribution += -dx * h[j-d]

I can compute the full gradient matrix in O(n²) time (one pass per row, O(n) work per row, M rows = O(n²) total).

def slsqp_solve(h_start, maxiter=200, tol=1e-10):
    n = len(h_start)
    dx = 2.0 / n
    M = 2 * n - 1
    c0 = exact_score(h_start)
    t0 = c0 + 1e-4
    
    def obj(x):
        return x[-1]
    
    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    
    def eq_fun(x):
        return x[:-1].sum() - n / 2.0
    
    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g
    
    def ineq_fun(x):
        hh = x[:-1]
        t = x[-1]
        profile = np.correlate(hh, 1.0 - hh, mode='full') * dx
        return t - profile
    
    def ineq_jac(x):
        hh = x[:-1]
        G = build_grad_matrix(hh, n, dx)
        J = np.zeros((M, n + 1))
        J[:, :-1] = -G
        J[:, -1] = 1.0
        return J
    
    x0 = np.concatenate([h_start, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 2.0)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': tol})
        if res.x is not None and np.isfinite(res.x).all():
            h_new = repair_feasible(res.x[:-1])
            c_new = exact_score(h_new)
            if c_new <= c0 + 1e-8:
                return h_new
    except Exception:
        pass
    return None

    deadline_cd = t_start + min(120, budget_s * 0.12)
    h_cd, c_cd = pairwise_cd(best_h, max_sweeps=100, grid_size=25,
                              rng=rng, deadline=deadline_cd)
    if c_cd < best_c - 1e-14:
        best_h = h_cd
        best_c = c_cd
    
    # Subgradient on best
    if time_left() > 60:
        deadline_sg = t_start + min(200, budget_s * 0.18)
        h_sg, c_sg = subgradient_descent(best_h, max_iters=300, deadline=deadline_sg)
        if c_sg < best_c - 1e-14:
            best_h = h_sg
            best_c = c_sg
    
    # SLSQP polish
    if time_left() > 80:
        h_sol = slsqp_solve(best_h, maxiter=200, tol=1e-10)
        if h_sol is not None:
            c_sol = exact_score(h_sol)
            if c_sol < best_c - 1e-12:
                best_h = h_sol
                best_c = c_sol
    
    # Now explore different dimensions
    n_best = len(best_h)
    dims_to_try = [d for d in dims if d != n_best]
    dims_to_try.sort(key=lambda d: abs(d - n_best))
    
    per_dim_time = max(40, (time_left() - 100) / max(len(dims_to_try[:6]), 1))
    
    for new_n in dims_to_try[:6]:
        if time_left() < 60:
            break
        deadline_dim = time.perf_counter() + per_dim_time
        
        # Interpolate to new dimension
        h_interp = interp_mass(best_h, new_n)
        c_interp = exact_score(h_interp)
        
        # Also try interpolating parent
        h_interp2 = interp_mass(parent, new_n)
        if exact_score(h_interp2) < c_interp:
            h_interp = h_interp2
            c_interp = exact_score(h_interp)
        
        curr_h = h_interp.copy()
        curr_c = c_interp
        
        # Pairwise CD on this dimension
        deadline1 = deadline_dim - 20
        h_r, c_r = pairwise_cd(curr_h, max_sweeps=80, grid_size=20,
                                rng=rng, deadline=deadline1)
        if c_r < curr_c:
            curr_h, curr_c = h_r, c_r
        
        # Subgradient
        if time.perf_counter() < deadline_dim - 10:
            h_r2, c_r2 = subgradient_descent(curr_h, max_iters=100,
                                              deadline=deadline_dim - 10)
            if c_r2 < curr_c:
                curr_h, curr_c = h_r2, c_r2
        
        # SLSQP on this dimension
        if time.perf_counter() < deadline_dim:
            h_sol = slsqp_solve(curr_h, maxiter=150, tol=1e-10)
            if h_sol is not None:
                c_sol = exact_score(h_sol)
                if c_sol < curr_c:
                    curr_h, curr_c = h_sol, c_sol
        
        if curr_c < best_c - 1e-12:
            best_h = curr_h
            best_c = curr_c
    
    # Binary annealing on best
    if time_left() > 80:
        deadline_bin = time.perf_counter() + min(60, time_left() * 0.4)
        h_bin, c_bin = binary_anneal_and_relax(best_h, max_iters=2000,
                                                deadline=deadline_bin, rng=rng)
        if c_bin < best_c - 1e-12:
            best_h = h_bin
            best_c = c_bin
    
    # Final refinement loop
    if time_left() > 40:
        for _ in range(4):
            if time_left() < 20:
                break
            # Quick CD
            deadline_f1 = time.perf_counter() + min(20, time_left() * 0.3)
            h_r, c_r = pairwise_cd(best_h, max_sweeps=50, grid_size=20,
                                    rng=rng, deadline=deadline_f1)
            if c_r < best_c - 1e-14:
                best_h, best_c = h_r, c_r
            # Quick subgradient
            if time_left() < 15:
                break
            deadline_f2 = time.perf_counter() + min(15, time_left() * 0.3)
            h_r2, c_r2 = subgradient_descent(best_h, max_iters=80, deadline=deadline_f2)
            if c_r2 < best_c - 1e-14:
                best_h, best_c = h_r2, c_r2
    
    # Final SLSQP polish
    if time_left() > 30:
        h_sol = slsqp_solve(best_h, maxiter=300, tol=1e-12)
        if h_sol is not None:
            c_sol = exact_score(h_sol)
            if c_sol < best_c - 1e-13:
                best_h = h_sol
                best_c = c_sol
    
    # Very fine CD at the end
    if time_left() > 15:
        deadline_final = time.perf_counter() + min(12, time_left() * 0.5)
        h_r, c_r = pairwise_cd(best_h, max_sweeps=60, grid_size=30,
                                rng=rng, deadline=deadline_final)
        if c_r < best_c - 1e-14:
            best_h, best_c = h_r, c_r
    
    # Final validation
    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)
    
    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            final_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 2:
            final_c = float(eval_result[0])
    except Exception:
        pass
    
    return best_h, final_c, len(best_h)
