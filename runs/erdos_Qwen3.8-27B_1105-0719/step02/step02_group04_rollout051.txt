def corr_grad(h, d, n):
    g = np.zeros(n)
    lo1 = max(0, d)
    hi1 = min(n - 1, n - 1 + d)
    if lo1 <= hi1:
        g[lo1:hi1+1] += 1.0 - h[lo1-d:hi1+1-d]
    lo2 = max(0, -d)
    hi2 = min(n - 1, n - 1 - d)
    if lo2 <= hi2:
        g[lo2:hi2+1] += -h[lo2+d:hi2+1+d]
    return g

                if rem() < reserve + 15:
                    break
                if sym == 'rev':
                    q_t = to_q_rev(h_try, n_try)
                else:
                    q_t = to_q_std(h_try, n_try)
                build_t = build_rev if sym == 'rev' else build_std
                h_t = build_t(q_t, n_try)
                active_t = select_active(h_t, n_try, margin_frac=0.06, pad=2)
                q_t, c5_t, h_t = opt_sym_slp(q_t, n_try, sym, mu_short, active_t, t0, budget_s, reserve + 15)
                if c5_t < best_c5:
                    best_h = h_t
                    best_c5 = c5_t

    # Phase 3: Full non-symmetric optimization
    if rem() > reserve + 50:
        n_cur = len(best_h)
        active_f = select_active(best_h, n_cur, margin_frac=0.06, pad=2)
        h_f, c5_f = opt_full_slp(best_h, n_cur, active_f, [50, 200, 1000, 5000], t0, budget_s, reserve + 20)
        if c5_f < best_c5:
            best_h = h_f
            best_c5 = c5_f

    # Phase 4: Subgradient + coordinate descent polishing
    if rem() > 20:
        n_cur = len(best_h)
        active_p = select_active(best_h, n_cur, margin_frac=0.1, pad=3)
        alpha = 0.002
        for _ in range(50):
            if rem() < 10:
                break
            best_h, best_c5 = subgradient_step(best_h, n_cur, active_p, alpha, t0, budget_s, 10)
            if rem() < 10:
                break
            # Small coordinate descent
            best_h, c5_cd = coord_descend(best_h, n_cur, max_rounds=10, t0=t0, budget=budget_s, reserve=10)
            if c5_cd < best_c5:
                best_c5 = c5_cd
            # Shrink step
            alpha *= 0.7

    # Phase 5: Final intensive SLP polish
    if rem() > 15:
        n_fin = len(best_h)
        if n_fin % 2 == 0:
            for sym in ['rev', 'std']:
                if rem() < 10:
                    break
                if sym == 'rev':
                    q_fin = to_q_rev(best_h, n_fin)
                else:
                    q_fin = to_q_std(best_h, n_fin)
                active_fin = select_active(best_h, n_fin, margin_frac=0.15, pad=4)
                q_fin, c5_fin, h_fin = opt_sym_slp(q_fin, n_fin, sym, [100, 1000, 10000, 100000], active_fin, t0, budget_s, 5)
                if c5_fin < best_c5:
                    best_h = h_fin
                    best_c5 = c5_fin

        if rem() > 10:
            active_fin = select_active(best_h, n_fin, margin_frac=0.1, pad=3)
            h_fin, c5_fin = opt_full_slp(best_h, n_fin, active_fin, [500, 2000, 10000], t0, budget_s, 5)
            if c5_fin < best_c5:
                best_h = h_fin
                best_c5 = c5_fin

    # Final coordinate descent
    if rem() > 8:
        best_h, c5_fin = coord_descend(best_h, len(best_h), max_rounds=50, t0=t0, budget=budget_s, reserve=3)
        if c5_fin < best_c5:
            best_c5 = c5_fin

    # Finalize
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = repair_box_sum(best_h, len(best_h) / 2.0)
    best_c5 = eval_c5(best_h)

    return best_h, best_c5, len(best_h)
