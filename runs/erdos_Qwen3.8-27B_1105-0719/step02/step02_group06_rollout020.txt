```python
def compute_pair_AQ(h, a, b, n):
    """Compute A and Q arrays for pair (a,b)."""
    two_n_1 = 2*n - 1
    A = np.zeros(two_n_1)
    Q = np.zeros(two_n_1)
    
    # Determine the range of d where any of a, b, a±d, b±d could be in range
    # a is first: -a ≤ d ≤ n-1-a
    # a is second: a-(n-1) ≤ d ≤ a
    # b is first: -b ≤ d ≤ n-1-b
    # b is second: b-(n-1) ≤ d ≤ b
    d_min = max(-(n-1), -a, a-(n-1), -b, b-(n-1))
    d_max = min(n-1, n-1-a, a, n-1-b, b)
    
    for d in range(d_min, d_max+1):

        if i_lo <= b <= i_hi: i_set.add(b)
```

```python
def minimize_pair(h, a, b, n, r_base, A, Q, delta_lo, delta_hi):
    """Minimize max_k(r_base + A*delta + Q*delta^2) over [delta_lo, delta_hi]."""
    # Ternary search
    lo, hi = delta_lo, delta_hi
    for _ in range(60):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = (r_base + A*m1 + Q*m1*m1).max()
        f2 = (r_base + A*m2 + Q*m2*m2).max()
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    delta_opt = (lo + hi) / 2.0
    f_opt = (r_base + A*delta_opt + Q*delta_opt*delta_opt).max()
    return delta_opt, f_opt
```
