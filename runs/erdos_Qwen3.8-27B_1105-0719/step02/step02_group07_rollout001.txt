def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h = h.copy()
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)


def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)


def corr_vals(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx


def make_sym_h(q, n):
    m = n // 2
    q = np.clip(q, 0.0, 1.0)
    h = np.empty(n)
    h[:m] = q
    h[m:] = 1.0 - q[::-1]
    return h


def make_q_from_h(h, n):
    m = n // 2
    q = 0.5 * (h[:m] + 1.0 - h[n - 1: n - 1 - m: -1])
    return np.clip(q, 0.0, 1.0)


def full_grad_lag(h, n, d):
    """d/dh of corr lag with offset d (lag index k=n-1+d)."""
    G = np.zeros(n)
    i_lo = max(0, -d)
    i_hi = min(n - 1, n - 1 - d)
    if i_lo > i_hi:
        return G * (2.0 / n)
    idx = np.arange(i_lo, i_hi + 1)
    G[idx] += (1.0 - h[idx + d])
    G[idx + d] += (-h[idx])
    G *= (2.0 / n)
    return G


def weighted_full_grad(h, n, weights):
    n_lags = 2 * n - 1
    G = np.zeros(n)
    for k in range(n_lags):
        wk = weights[k]
        if wk < 1e-15:
            continue
        d = k - (n - 1)
        G += wk * full_grad_lag(h, n, d)
    return G


def sym_grad_active(h, n, corrs, beta):
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    ws = w.sum()
    if ws < 1e-30:
        w = np.ones_like(corrs) / corrs.size
    else:
        w = w / ws
    G = weighted_full_grad(h, n, w)
    m = n // 2
    return G[:m] - G[n - 1: n - 1 - m: -1]


def active_lags_top(corrs, n_top):
    idx = np.argsort(corrs)[::-1][:n_top]
    return sorted(int(i) for i in idx.tolist())


class SmoothSym:
    def __init__(self, n, beta):
        self.n = n
        self.m = n // 2
        self.beta = beta

    def fun(self, q):

        return float(logsumexp(self.beta * corr_vals(h)) / self.beta)

    def jac(self, q):
        h = make_sym_h(q, self.n)
        return sym_grad_active(h, self.n, corr_vals(h), self.beta)


class EpiSymSolver:

        self.m = n // 2
        self.dx = 2.0 / n
        self.active_lags = np.asarray(active_lags, dtype=int)

        return x[0]

        g[0] = 1.0
        return g

        return t - corr_vals(h)[self.active_lags]

        return J

def solve_epi_sym(q0, n, active_lags, t_init, maxiter):

            q_best = q_new
            h_best = h_new
            print("  sym lbfgs beta=%d C5=%.7f" % (beta, c5_best))

    # ---- P2: exact subgradient + line search polish (symmetric) ----
    for trial in range(3):
        if budget_s - elapsed() < 15:
            break
        q_new, c5_new = subgrad_finish_sym(q_best, n, n_iter=1500,
                                           beta_dir=800.0, momentum=0.9)
        if c5_new < c5_best - 1e-10:
            c5_best = c5_new
            q_best = q_new
            h_best = make_sym_h(q_best, n)
            print("  subgrad trial=%d C5=%.7f" % (trial, c5_best))

    # ---- P3: epigraph active-set to enforce equioscillation ----
    for rnd in range(6):
        if budget_s - elapsed() < 15:
            break
        h_cur = make_sym_h(q_best, n)
        corrs = corr_vals(h_cur)
        n_act = min(40 + 12 * rnd, 2 * n - 1)
        active = active_lags_top(corrs, n_act)
        q_new, c5_new = solve_epi_sym(q_best, n, active,
                                      t_init=corrs.max() * 1.0001, maxiter=450)
        if q_new is not None and c5_new < c5_best - 1e-10:
            c5_best = c5_new
            q_best = q_new
            h_best = make_sym_h(q_best, n)
            print("  epi sym rnd=%d L=%d C5=%.7f" % (rnd, n_act, c5_best))

    # ---- P4: restarts in symmetric space to escape local minima ----
    n_trials = 14
    for trial in range(n_trials):
        if budget_s - elapsed() < 18:
            break
        scale = 0.02 + 0.35 * (trial / float(max(n_trials, 1)))
        q_pert = np.clip(q_best + scale * rng.randn(m), 0.0, 1.0)
        q_new = lbfgs_sym(q_pert, n, 1500, maxiter=150, gtol=1e-9)
        q_new = lbfgs_sym(q_new, n, 4000, maxiter=200, gtol=1e-9)
        c5_new = exact_score(make_sym_h(q_new, n))
        if c5_new < c5_best - 1e-10:
            h_states = make_sym_h(q_new, n)
            q_new2, c5_new2 = subgrad_finish_sym(q_new, n, 600, 800.0, 0.9)
            if c5_new2 < c5_best - 1e-10:
                c5_best = c5_new2
                q_best = q_new2
                h_best = make_sym_h(q_best, n)
            else:
                c5_best = c5_new
                q_best = q_new
                h_best = h_states
            print("  restart %d C5=%.7f" % (trial, c5_best))

    # ---- P5: release symmetry, full-space polish ----
    if budget_s - elapsed() > 40:
        h_full = make_sym_h(q_best, n)
        for beta in [1500, 4000]:
            if budget_s - elapsed() < 15:
                break
            fs = FullSmooth(n, beta)
            try:
                res = minimize(fs.fun, h_full, jac=fs.jac, method='L-BFGS-B',
                               bounds=[(0.0, 1.0)] * n,
                               options={'maxiter': 250, 'gtol': 1e-9})
                h_full2 = project_feasible(np.clip(res.x, 0.0, 1.0), n / 2.0)
            except Exception:
                h_full2 = h_full
            c5_full = exact_score(h_full2)
            if c5_full < c5_best - 1e-10:
                c5_best = c5_full
                h_best = h_full2
                h_full = h_full2
                print("  full lbfgs beta=%d C5=%.7f" % (beta, c5_best))

    # ---- P6: resolution transfer to finer grids ----
    if budget_s - elapsed() > 40:
        for delta in [2, 4, 6, -2, 8]:
            if budget_s - elapsed() < 25:
                break
            n_try = n + delta
            if n_try < 60 or n_try % 2 != 0:
                continue
            x_old = np.linspace(0, 2, n, endpoint=False)
            x_new = np.linspace(0, 2, n_try, endpoint=False)
            h_interp = np.interp(x_new, x_old, h_best)
            h_interp = project_feasible(h_interp, n_try / 2.0)
            m_try = n_try // 2
            q_try = make_q_from_h(h_interp, n_try)
            q_try = lbfgs_sym(q_try, n_try, 1500, maxiter=200, gtol=1e-9)
            q_try = lbfgs_sym(q_try, n_try, 4000, maxiter=200, gtol=1e-9)
            q_try, c5_try = subgrad_finish_sym(q_try, n_try, 500, 800.0, 0.9)
            if c5_try < c5_best - 1e-9:
                c5_best = c5_try
                n = n_try
                m = m_try
                q_best = q_try
                h_best = make_sym_h(q_best, n)
                print("  transfer n=%d C5=%.7f" % (n, c5_best))

    # ---- P7: final subgradient + epigraph on the global best ----
    if budget_s - elapsed() > 20:
        q_final = make_q_from_h(h_best, n)
        n = len(h_best)
        m = n // 2
        q_final, c5f = subgrad_finish_sym(q_final, n, 1000, 1500.0, 0.9)
        h_symf = make_sym_h(q_final, n)
        if c5f < exact_score(h_best) - 1e-10:
            h_best, c5_best = h_symf, c5f
        corrs = corr_vals(h_best)
        active = active_lags_top(corrs, min(80, 2 * n - 1))
        qe, ce = solve_epi_sym(make_q_from_h(h_best, n), n, active,
                               t_init=corrs.max() * 1.0001, maxiter=400)
        if qe is not None and ce < exact_score(h_best) - 1e-10:
            h_best, c5_best = make_sym_h(qe, n), ce

    return h_best, float(c5_best), int(len(h_best))
