from scipy.optimize import minimize

    return np.correlate(h, 1.0 - h, mode='full')

def c5_of_h(h, dx):
    return float(compute_corrs(h).max() * dx)

def grad_corr_k(h, k, n):
    ...

def smooth_max_val(h, dx, beta):
    c = compute_corrs(h) * dx
    return float(logsumexp(beta * c) / beta)

def smooth_max_grad(h, dx, beta):
    c = compute_corrs(h) * dx
    shifted = c - c.max()
    w = np.exp(beta * shifted)
    w /= w.sum()
    grad = np.zeros(len(h))
    n = len(h)
    for k in range(len(c)):
        if w[k] < 1e-14:
            continue
        grad += w[k] * grad_corr_k(h, k, n)
    return grad * dx

def solve_smooth_max(h0, dx, target, beta, maxiter, tol):
    """SLSQP with smooth max objective"""
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(n)}]
    
    def obj(x):
        return smooth_max_val(x, dx, beta)
    
    def g(x):
        return smooth_max_grad(x, dx, beta)
    
    try:
        res = minimize(obj, h0, jac=g, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': tol})
        h_out = project_feasible(res.x, target)
    except Exception:
        h_out = project_feasible(h0, target)
    return h_out, c5_of_h(h_out, dx)

def subgrad_linesearch(h0, dx, target, max_iter, top_m, initial_step):
    """Subgradient with Armijo line search"""
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    step = initial_step
    
    for it in range(max_iter):
        c = c5_of_h(h, dx)
        if c < best_c5 - 1e-15:
            best_c5 = c
            best_h = h.copy()
        
        corrs = compute_corrs(h)
        idx = np.argsort(corrs)[-top_m:]
        grad = np.sum([grad_corr_k(h, k, n) for k in idx], axis=0) / len(idx)
        grad *= dx
        
        # Armijo line search
        f0 = c
        c1 = 1e-4
        rho = 0.5
        alpha = step
        improved = False
        for ls in range(30):
            h_trial = project_feasible(h - alpha * grad, target)
            f_trial = c5_of_h(h_trial, dx)
            if f_trial <= f0 - c1 * alpha * np.sum(grad * (h - h0)):  # simplified
                # Actually for subgradient, the condition is:
                # f(x - alpha * g) <= f(x) - c1 * alpha * ||g||^2
                # But since g is subgradient of max, this works
                h = h_trial
                improved = True
                break
            alpha *= rho
        
        if improved:
            step = min(step * 1.1, 0.5)
        else:
            step *= 0.5
        
        if step < 1e-12:
            break
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() >= target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h


def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def c5_of_h(h, dx):
    return float(compute_corrs(h).max() * dx)


def grad_corr_k(h, k, n):
    d = k - (n - 1)
    grad = np.zeros(n)
    if d == 0:
        grad[:] = 1.0 - 2.0 * h
    elif d > 0:
        grad[:n - d] += (1.0 - h[d:])
        grad[d:] += (-h[:n - d])
    else:
        e = -d
        grad[e:] += (1.0 - h[:n - e])
        grad[:n - e] += (-h[e:])
    return grad


def smooth_max_val(h, dx, beta):
    c = compute_corrs(h) * dx
    return float(logsumexp(beta * c) / beta)


def smooth_max_grad(h, dx, beta):
    n = len(h)
    c = compute_corrs(h) * dx
    shifted = c - c.max()
    w = np.exp(beta * shifted)
    ws = w.sum()
    if ws < 1e-30:
        w = np.ones_like(c) / len(c)
    else:
        w /= ws
    grad = np.zeros(n)
    for k in range(len(c)):
        if w[k] > 1e-14:
            grad += w[k] * grad_corr_k(h, k, n)
    return grad * dx

        self.target = target
        self.lags = lags
        self.weights = weights

    def fun(self, x):

        corrs = compute_corrs(x)
        val = 0.0
        for idx, w in zip(self.lags, self.weights):
            val += w * corrs[idx]
        return val * self.dx

        grad = np.zeros(n)
        for idx, w in zip(self.lags, self.weights):
            grad += w * grad_corr_k(x, idx, n)

def solve_with_slsqp(obj, h0, maxiter, ftol=1e-15):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': obj.eq_fun, 'jac': obj.eq_jac}]
    try:
        res = minimize(obj.fun, h0, jac=obj.jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        h_out = project_feasible(res.x, obj.target)
    except Exception:
        h_out = project_feasible(h0, obj.target)
    return h_out, c5_of_h(h_out, 2.0 / n)

    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    step = init_step
    no_improve = 0

    for it in range(max_iter):
        c5_val = c5_of_h(h, dx)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 3000:

        idx_top = np.argsort(corrs)[-top_m:]
        grad = np.zeros(n)
        for k in idx_top:
            grad += grad_corr_k(h, k, n)
        grad /= len(idx_top)
        grad *= dx

        # Armijo line search
        g_norm_sq = np.dot(grad, grad)
        if g_norm_sq < 1e-20:
            break
        f0 = c5_val
        alpha = step
        improved = False
        for _ in range(40):

                break

            step *= 0.5

            break

        return s + self.rho * np.dot(h - self.anchor, h - self.anchor)

    def jac(self, y):
        h = y[:self.n]
        s = y[self.n]
        jac = np.zeros(self.n + 1)
        jac[:self.n] = 2.0 * self.rho * (h - self.anchor)
        jac[self.n] = 1.0
        return jac

    def ineq(self, y):
        h = y[:self.n]
        s = y[self.n]
        corrs = compute_corrs(h) * self.dx
        return s + self.U_target - corrs  # all should be >= 0

    def ineq_jac(self, y):

        n = self.n
        L = 2 * n - 1
        jac = np.zeros((L, n + 1))
        for k in range(L):
            jac[k, :n] = grad_corr_k(h, k, n) * self.dx
            jac[k, n] = 1.0
        return jac

    def eq_fun(self, y):
        return y[:self.n].sum() - self.target

    def eq_jac(self, y):
        row = np.zeros(self.n + 1)
        row[:self.n] = 1.0
        return row

import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time


def project_feasible(x, target_sum):
    n = len(x)
    if n == 0:
        return x
    lo, hi = -3.0, 3.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() >= target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h


def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def c5_of_h(h, dx):
    return float(compute_corrs(h).max() * dx)


def grad_corr_k(h, k, n):
    d = k - (n - 1)
    grad = np.zeros(n)
    if d == 0:
        grad[:] = 1.0 - 2.0 * h
    elif d > 0:
        grad[:n - d] += (1.0 - h[d:])
        grad[d:] += (-h[:n - d])
    else:
        e = -d
        grad[e:] += (1.0 - h[:n - e])
        grad[:n - e] += (-h[e:])
    return grad


class SmoothMaxObj:
    def __init__(self, dx, beta, target):
        self.dx = dx
        self.beta = beta
        self.target = target

    def fun(self, x):
        c = compute_corrs(x) * self.dx
        return float(logsumexp(self.beta * c) / self.beta)

    def jac(self, x):
        n = len(x)
        c = compute_corrs(x) * self.dx
        shifted = c - c.max()
        w = np.exp(self.beta * shifted)
        ws = w.sum()
        if ws < 1e-30:
            w = np.ones_like(c) / len(c)
        else:
            w /= ws
        grad = np.zeros(n)
        for k in range(len(c)):
            if w[k] > 1e-14:
                grad += w[k] * grad_corr_k(x, k, n)
        return grad * self.dx

    def eq_fun(self, x):
        return x.sum() - self.target

    def eq_jac(self, x):
        return np.ones(len(x))


class EpigraphObj:
    def __init__(self, dx, target, U_target, anchor, rho):
        self.dx = dx
        self.target = target
        self.U_target = U_target
        self.anchor = anchor
        self.rho = rho
        self.n = len(anchor)

    def fun(self, y):
        h = y[:self.n]
        s = y[self.n]
        return s + self.rho * np.dot(h - self.anchor, h - self.anchor)

    def jac(self, y):
        h = y[:self.n]
        jac = np.zeros(self.n + 1)
        jac[:self.n] = 2.0 * self.rho * (h - self.anchor)
        jac[self.n] = 1.0
        return jac

    def ineq(self, y):
        h = y[:self.n]
        s = y[self.n]
        corrs = compute_corrs(h) * self.dx
        return s + self.U_target - corrs

    def ineq_jac(self, y):
        h = y[:self.n]
        n = self.n
        L = 2 * n - 1
        jac = np.zeros((L, n + 1))
        for k in range(L):
            jac[k, :n] = grad_corr_k(h, k, n) * self.dx
            jac[k, n] = 1.0
        return jac

    def eq_fun(self, y):
        return y[:self.n].sum() - self.target

    def eq_jac(self, y):
        row = np.zeros(self.n + 1)
        row[:self.n] = 1.0
        return row


class ActiveSetObj:
    def __init__(self, dx, target, lags, weights):
        self.dx = dx
        self.target = target
        self.lags = lags
        self.weights = weights

    def fun(self, x):
        n = len(x)
        corrs = compute_corrs(x)
        val = 0.0
        for idx, w in zip(self.lags, self.weights):
            val += w * corrs[idx]
        return val * self.dx

    def jac(self, x):
        n = len(x)
        grad = np.zeros(n)
        for idx, w in zip(self.lags, self.weights):
            grad += w * grad_corr_k(x, idx, n)
        return grad * self.dx

    def eq_fun(self, x):
        return x.sum() - self.target

    def eq_jac(self, x):
        return np.ones(len(x))


def solve_slsqp(fun, jac, h0, bounds, cons, maxiter, ftol):
    try:
        res = minimize(fun, h0, jac=jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x, res.fun
    except Exception:
        return h0, None


def solve_smooth(h0, dx, target, beta, maxiter, ftol=1e-15):
    n = len(h0)
    obj = SmoothMaxObj(dx, beta, target)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': obj.eq_fun, 'jac': obj.eq_jac}]
    x_out, _ = solve_slsqp(obj.fun, obj.jac, h0, bounds, cons, maxiter, ftol)
    h_out = project_feasible(x_out, target)
    return h_out, c5_of_h(h_out, dx)


def solve_epigraph(h0, dx, target, U_target, rho, maxiter, ftol=1e-14):
    n = len(h0)
    corrs0 = compute_corrs(h0) * dx
    s0 = float(max(0.0, corrs0.max() - U_target)) + 1e-6
    obj = EpigraphObj(dx, target, U_target, h0, rho)
    y0 = np.concatenate([h0, [s0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, s0 + 0.5)]
    cons = [
        {'type': 'eq', 'fun': obj.eq_fun, 'jac': obj.eq_jac},
        {'type': 'ineq', 'fun': obj.ineq, 'jac': obj.ineq_jac}
    ]
    y_out, _ = solve_slsqp(obj.fun, obj.jac, y0, bounds, cons, maxiter, ftol)
    h_out = project_feasible(y_out[:n], target)
    return h_out, c5_of_h(h_out, dx)


def solve_active(h0, dx, target, top_m, maxiter, ftol=1e-14):
    n = len(h0)
    corrs = compute_corrs(h0)
    idx_top = np.argsort(corrs)[-top_m:]
    top_vals = corrs[idx_top]
    w = np.exp(10.0 * (top_vals - top_vals.max()))
    w /= w.sum()
    obj = ActiveSetObj(dx, target, idx_top, w)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': obj.eq_fun, 'jac': obj.eq_jac}]
    x_out, _ = solve_slsqp(obj.fun, obj.jac, h0, bounds, cons, maxiter, ftol)
    h_out = project_feasible(x_out, target)
    return h_out, c5_of_h(h_out, dx)


def subgrad_armijo(h0, dx, target, max_iter, top_m, init_step):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    step = init_step
    no_improve = 0

    for it in range(max_iter):
        c5_val = c5_of_h(h, dx)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 2000:
                break

        corrs = compute_corrs(h)
        idx_top = np.argsort(corrs)[-top_m:]
        grad = np.zeros(n)
        for k in idx_top:
            grad += grad_corr_k(h, k, n)
        grad /= len(idx_top)
        grad *= dx

        g_norm_sq = float(np.dot(grad, grad))
        if g_norm_sq < 1e-22:
            break
        f0 = c5_val
        alpha = step
        improved = False
        for _ in range(40):
            h_new = project_feasible(h - alpha * grad, target)
            f_new = c5_of_h(h_new, dx)
            if f_new <= f0 - 1e-4 * alpha * g_norm_sq:
                h = h_new
                improved = True
                break
            alpha *= 0.5

        if improved:
            step = min(step * 1.03, 0.4)
        else:
            step *= 0.45

        if step < 1e-11:
            break

    return best_h, best_c5


def subgrad_momentum(h0, dx, target, max_iter, top_m, init_alpha, gamma):
    n = len(h0)
    h = h0.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    alpha = init_alpha
    no_improve = 0

    for it in range(max_iter):
        c5_val = c5_of_h(h, dx)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 5000:
                break

        corrs = compute_corrs(h)
        idx_top = np.argsort(corrs)[-top_m:]
        grad = np.zeros(n)
        for k in idx_top:
            grad += grad_corr_k(h, k, n)
        grad /= len(idx_top)
        grad *= dx

        v = gamma * v - alpha * grad
        h_new = project_feasible(h + v, target)
        new_c5 = c5_of_h(h_new, dx)

        if new_c5 < c5_val - 1e-15:
            h = h_new
            alpha = min(alpha * 1.005, 0.3)
        else:
            alpha *= 0.92
            v *= 0.3

        if alpha < 1e-13:
            break

    return best_h, best_c5


def make_pair_complement(m, rng):
    v = rng.uniform(0.05, 0.95, m)
    h = np.zeros(2 * m)
    h[:m] = v
    h[m:] = 1.0 - v[::-1]
    return h


def make_binary_pcomp(n, rng):
    if n % 2 == 0:
        m = n // 2
        v = (rng.rand(m) < 0.5).astype(float)
        h = np.zeros(n)
        h[:m] = v
        h[m:] = 1.0 - v[::-1]
    else:
        m = n // 2
        v = (rng.rand(m) < 0.5).astype(float)
        h = np.zeros(n)
        h[:m] = v
        h[m + 1:] = 1.0 - v[::-1]
        h[m] = 0.5
    return h
