def corr_gradient(h, s, n):
    """Gradient of correlation at shift s w.r.t. h."""
    # s = k - (n-1) where k is the index in the full correlation output
    # corr_s = sum_{n valid} h[n+s] * (1-h[n])
    # valid: max(0, -s) <= n <= min(n-1, n-1-s)
    grad = np.zeros(n)
    i_lo = max(0, -s)
    i_hi = min(n - 1, n - 1 - s)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    # d/dh[idx] of sum h[idx+s]*(1-h[idx]) = -h[idx+s]
    grad[idx] += -h[idx + s]
    # d/dh[idx+s] of sum h[idx+s]*(1-h[idx]) = (1-h[idx])
    grad[idx + s] += (1.0 - h[idx])
    return grad

def active_gradient(h, dx, k_top=30, tau=0.05):

    n = len(h)
    corrs = np.correlate(h, 1.0 - h, mode='full')
    max_c = corrs.max()
    
    # Select active correlations
    threshold = max_c - tau * max(1, max_c)
    active_mask = corrs >= threshold
    if active_mask.sum() < k_top:
        # Take top k_top
        active_idx = np.argsort(corrs)[-k_top:]
        active_mask = np.zeros(len(corrs), dtype=bool)
        active_mask[active_idx] = True

    beta = 50.0
    w = np.exp(beta * shifted)
    w /= w.sum()
    
    grad = np.zeros(n)
    active_indices = np.where(active_mask)[0]
    for i, k in enumerate(active_indices):
        s = k - (n - 1)
        grad += w[i] * corr_gradient(h, s, n)

    """Compute delta for all correlations when h[a]-=t, h[b]+=t. Returns array of length 2n-1."""
    delta = np.zeros(2 * n - 1)
    
    for sign, idx in [(-t, a), (t, b)]:
        # idx appears as i (the "1-h[i]" factor):
        # Valid shifts s where: max(0,-s) <= idx <= min(n-1, n-1-s) and 0 <= idx+s <= n-1
        # Condition 1: idx >= 0 and idx <= n-1 (always true)
        # Condition 2: -s <= idx → s >= -idx; also idx <= n-1-s → s <= n-1-idx
        # Condition 3: 0 <= idx+s → s >= -idx; also idx+s <= n-1 → s <= n-1-idx
        # So: max(-idx, -(n-1)) <= s <= min(n-1-idx, n-1)
        # Since idx in [0, n-1]: s range is [-idx, n-1-idx]
        s_lo = max(-idx, -(n-1))
        s_hi = min(n - 1 - idx, n - 1)
        if s_lo <= s_hi:
            s_vals = np.arange(s_lo, s_hi + 1)
            # Change: sign * h[idx+s] (because (1-h[idx]) changes by sign, so term gets sign*h[idx+s])
            # Wait: if h[idx] changes by sign, then (1-h[idx]) changes by -sign
            # Term h[idx+s]*(1-h[idx]) changes by h[idx+s]*(-sign) = -sign*h[idx+s]

        # i = idx - s, valid when: 0 <= idx-s <= n-1 and max(0,-s) <= idx-s <= min(n-1, n-1-s)
        # Condition: 0 <= idx-s → s <= idx; also idx-s <= n-1 → s >= idx-(n-1)
        # Also: idx-s >= 0 → s <= idx (same); idx-s <= n-1-s → idx <= n-1 (always)
        # And: idx-s >= -s → idx >= 0 (always); idx-s >= 0 → s <= idx
        # So: max(idx-(n-1), -(n-1)) <= s <= min(idx, n-1)
