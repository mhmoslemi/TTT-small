import numpy as np
from scipy.optimize import linprog, minimize
import time

def eval_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def corr_raw(h):
    return np.correlate(h, 1.0 - h, mode='full')

def build_sym_h(q):
    m = len(q)
    h = np.empty(2 * m)
    h[:m] = q
    h[m:] = 1.0 - q[::-1]
    return h

def sym_from_h(h):
    n = len(h)
    m = n // 2
    q = np.empty(m)
    for i in range(m):
        q[i] = 0.5 * (h[i] + (1.0 - h[n - 1 - i]))
    return np.clip(q, 0.0, 1.0)

def project_feasible(x, target_sum):
    x = np.asarray(x, dtype=float).copy()
    x[~np.isfinite(x)] = 0.5
    x = np.clip(x, 0.0, 1.0)
    n = len(x)
    if abs(x.sum() - target_sum) < 1e-12:
        return x
    lo, hi = -n - 1.0, n + 1.0
    for _ in range(90):
        mid = (lo + hi) / 2.0
        s = np.clip(x + mid, 0.0, 1.0).sum()
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    h = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            order = np.argsort(h)
            for i in order:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
                if abs(diff) < 1e-13:
                    break
        else:
            order = np.argsort(h)[::-1]
            for i in order:
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub
                if abs(diff) < 1e-13:
                    break
    return np.clip(h, 0.0, 1.0)

def corr_grad_full(h, k):
    """Gradient of raw correlation at lag index k w.r.t. full h."""
    n = len(h)
    d = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -d)
    i_hi = min(n - 1, n - 1 - d)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    jdx = idx + d
    grad[idx] += (1.0 - h[jdx])
    grad[jdx] += (-h[idx])
    return grad

def solve_tangent_lp_full(h0, active_lags, rho=0.3, t_upper=None):
    """Solve tangent LP for full vector."""
    n = len(h0)
    dx = 2.0 / n
    corrs = corr_raw(h0)
    
    if t_upper is None:
        t_upper = corrs.max() + 1.0
    
    # Variables: z (n) + t (1) = n+1
    # Objective: minimize t
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    # Inequality constraints: for each active lag
    # grad_ell^T z - t <= grad_ell^T h0 - R_ell(h0)
    A_ub = []
    b_ub = []
    for ell in active_lags:
        g = corr_grad_full(h0, ell)
        row = np.zeros(n + 1)
        row[:n] = g
        row[n] = -1.0
        A_ub.append(row)
        b_ub.append(float(np.dot(g, h0) - corrs[ell]))
    
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    # Equality: sum(z) = n/2
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    # Bounds
    bounds_full = []
    for i in range(n):
        lo = max(0.0, h0[i] - rho)
        hi = min(1.0, h0[i] + rho)
        bounds_full.append((lo, hi))
    bounds_full.append((0.0, t_upper))
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds_full, method='highs')
        if res.success:
            return res.x[:n], res.x[n], res.fun
        return h0.copy(), None, None
    except Exception:
        return h0.copy(), None, None

def solve_tangent_lp_sym(q0, active_lags, rho=0.3, t_upper=None):
    """Solve tangent LP for symmetric half vector."""
    m = len(q0)
    n = 2 * m
    h0 = build_sym_h(q0)
    dx = 2.0 / n
    corrs = corr_raw(h0)
    
    if t_upper is None:
        t_upper = corrs.max() + 1.0
    
    # Variables: q (m) + t (1) = m+1
    # Objective: minimize t
    c = np.zeros(m + 1)
    c[m] = 1.0
    
    A_ub = []
    b_ub = []
    for ell in active_lags:
        g = corr_grad_full(h0, ell)
        # grad_q[r] = g[r] - g[n-1-r]
        gq = g[:m].copy()
        for r in range(m):
            gq[r] = g[r] - g[n - 1 - r]
        row = np.zeros(m + 1)
        row[:m] = gq
        row[m] = -1.0
        A_ub.append(row)
        b_ub.append(float(np.dot(gq, q0) - corrs[ell]))
    
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    # No equality needed (symmetry ensures sum)
    bounds_full = []
    for i in range(m):
        lo = max(0.0, q0[i] - rho)
        hi = min(1.0, q0[i] + rho)
        bounds_full.append((lo, hi))
    bounds_full.append((0.0, t_upper))
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                      bounds=bounds_full, method='highs')
        if res.success:
            return res.x[:m], res.x[m], res.fun
        return q0.copy(), None, None
    except Exception:
        return q0.copy(), None, None

def iter_tangent_full(h0, all_lags, max_iter=100, rho_init=0.5, shrink=0.8, time_limit=None):
    """Iterative tangent LP for full vector."""
    n = len(h0)
    dx = 2.0 / n
    h = h0.copy()
    best_score = eval_score(h)
    best_h = h.copy()
    rho = rho_init
    stagnant = 0
    
    for it in range(max_iter):
        if time_limit is not None and time.time() > time_limit:
            break
        
        # Compute active lags
        corrs = corr_raw(h) * dx
        max_val = corrs.max()
        # Use lags within a margin of the max
        margin = max(0.02, 0.05 * max_val)
        active = np.where(corrs >= max_val - margin)[0].tolist()
        
        # Solve LP
        z_new, t_val, obj = solve_tangent_lp_full(h, active, rho=rho)
        if z_new is None or not np.any(z_new != h):
            stagnant += 1
            rho *= shrink
            if stagnant > 10 or rho < 0.01:
                break
            continue
        
        # Take a step: h + alpha * (z_new - h)
        # Try several step sizes
        d = z_new - h
        best_alpha = 0.0
        best_local_score = best_score
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            h_test = np.clip(h + alpha * d, 0.0, 1.0)
            h_test = project_feasible(h_test, n / 2.0)
            sc = eval_score(h_test)
            if sc < best_local_score:
                best_local_score = sc
                best_alpha = alpha
            if alpha == 1.0:
                break
        
        if best_alpha > 0:
            h = project_feasible(np.clip(h + best_alpha * d, 0.0, 1.0), n / 2.0)
            sc = eval_score(h)
            if sc < best_score:
                best_score = sc
                best_h = h.copy()
                stagnant = 0
                # Slightly increase trust region if improving
                rho = min(0.5, rho * 1.1)
            else:
                stagnant += 1
                rho *= shrink
        else:
            stagnant += 1
            rho *= shrink
        
        if stagnant > 15 or rho < 0.005:
            break
    
    return best_h, best_score

def transfer_line_search(h, a, b, n_samples=80):
    """Exact 1D line search along transfer direction h[a]+delta, h[b]-delta."""
    n = len(h)
    dx = 2.0 / n
    h0 = h.copy()
    lower = max(-h0[a], h0[b] - 1.0)
    upper = min(1.0 - h0[a], -h0[b])
    if upper <= lower:
        return h0, eval_score(h0)
    
    deltas = np.linspace(lower, upper, n_samples)
    best_delta = 0.0
    best_score = eval_score(h0)
    
    for d in deltas:
        h_test = h0.copy()
        h_test[a] += d
        h_test[b] -= d
        sc = eval_score(h_test)
        if sc < best_score:
            best_score = sc
            best_delta = d
    
    # Refine around best
    if best_delta != 0.0:
        span = (upper - lower) / n_samples
        lo = best_delta - span
        hi = best_delta + span
        for _ in range(10):
            mid = (lo + hi) / 2.0
            h_mid = h0.copy()
            h_mid[a] += mid
            h_mid[b] -= mid
            sc_mid = eval_score(h_mid)
            h_three = h0.copy()
            h_three[a] += (lo + mid) / 2.0
            h_three[b] -= (lo + mid) / 2.0
            sc_three = eval_score(h_three)
            if sc_three < sc_mid:
                hi = mid
            else:
                lo = mid
    
    h_out = h0.copy()
    h_out[a] += best_delta
    h_out[b] -= best_delta
    return h_out, eval_score(h_out)

def random_transfer_search(h, n_pairs=50, n_samples=60, rng=None):
    """Try random transfer pairs and keep improvements."""
    if rng is None:
        rng = np.random.RandomState(42)
    n = len(h)
    dx = 2.0 / n
    best = h.copy()
    best_score = eval_score(best)
    
    for _ in range(n_pairs):
        a = rng.randint(n)
        b = rng.randint(n)
        if a == b:
            continue
        h_new, sc = transfer_line_search(best, a, b, n_samples=n_samples)
        if sc < best_score - 1e-12:
            best = h_new
            best_score = sc
    
    return best, best_score

def transfer_to_n(h_old, n_new):
    n_old = len(h_old)
    x_old = (np.arange(n_old) + 0.5) * 2.0 / n_old
    x_new = (np.arange(n_new) + 0.5) * 2.0 / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new / 2.0)

    return grad

def transfer_search(h, rng, n_pairs=100, n_grid=60):
    n = len(h)
    best = h.copy()
    best_score = eval_score(best)
    improvements = 0
    for _ in range(n_pairs):
        a = rng.randint(n)
        b = rng.randint(n)
        if a == b:
            continue
        lower = max(-best[a], best[b] - 1.0)
        upper = min(1.0 - best[a], -best[b])
        if upper - lower < 1e-10:
            continue
        deltas = np.linspace(lower, upper, n_grid)
        local_best_d = 0.0
        local_best_sc = best_score
        for d in deltas:
            h_test = best.copy()
            h_test[a] += d
            h_test[b] -= d
            sc = eval_score(h_test)
            if sc < local_best_sc:
                local_best_sc = sc
                local_best_d = d
        if local_best_d != 0.0:
            span = (upper - lower) / n_grid
            lo = max(lower, local_best_d - span)
            hi = min(upper, local_best_d + span)
            for _ in range(8):
                mid1 = lo + (hi - lo) / 3.0
                mid2 = hi - (hi - lo) / 3.0
                h1 = best.copy()
                h1[a] += mid1
                h1[b] -= mid1
                h2 = best.copy()
                h2[a] += mid2
                h2[b] -= mid2
                s1 = eval_score(h1)
                s2 = eval_score(h2)
                if s1 < s2:
                    hi = mid2
                else:
                    lo = mid1
            final_d = (lo + hi) / 2.0
            h_test = best.copy()
            h_test[a] += final_d
            h_test[b] -= final_d
            sc = eval_score(h_test)
            if sc < local_best_sc:
                local_best_sc = sc
                local_best_d = final_d
        if local_best_sc < best_score - 1e-12:
            best[a] += local_best_d
            best[b] -= local_best_d
            best_score = local_best_sc
            improvements += 1
    return best, best_score

    x_new = (np.arange(n_new) + 0.5) * 2.0 / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def rem():
        return budget_s - (time.time() - t0)

    best_h = None
    best_c5 = float('inf')

    try:
        h_parent = np.array(initial_h_values, dtype=float).copy().reshape(-1)
    except (NameError, AttributeError):
        h_parent = None

    if h_parent is not None and len(h_parent) > 10:
        h_parent = project_feasible(np.clip(h_parent, 0.0, 1.0), len(h_parent) / 2.0)
        best_h = h_parent.copy()
        best_c5 = eval_score(h_parent)
    else:
        n_p = 126
        m_p = n_p // 2
        x = np.linspace(0, 1, m_p)
        q_p = (0.5 + 0.3 * np.sin(2 * np.pi * x)).clip(0, 1)
        best_h = build_sym_h(q_p)
        best_c5 = eval_score(best_h)

    # Phase 1: Symmetric tangent LP
    if len(best_h) % 2 == 0 and rem() > 100:
        q_init = sym_from_h(best_h)
        tl = time.time() + min(180, rem() * 0.2)
        q_opt, c5_s, h_s = iter_tangent_sym(q_init, max_iter=400, rho_init=0.4, shrink=0.85, time_limit=tl)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()

    # Phase 2: Full-vector tangent LP (breaks symmetry)
    if rem() > 150:
        tl = time.time() + min(250, rem() * 0.25)
        h_f, c5_f = iter_tangent_full(best_h, max_iter=400, rho_init=0.4, shrink=0.85, time_limit=tl)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Phase 3: Transfer line searches
    if rem() > 100:
        for rnd in range(4):
            if rem() < 20:
                break
            h_new, sc = transfer_search(best_h, rng, n_pairs=120, n_grid=50)
            if sc < best_c5 - 1e-12:
                best_c5 = sc
                best_h = h_new

    # Phase 4: Multi-resolution exploration
    n_base = len(best_h)
    n_candidates = set()
    for delta in [-12, -10, -8, -6, -4, -3, -2, -1, 1, 2, 3, 4, 6, 8, 10, 14, 20, 30]:
        nc = n_base + delta
        if 48 <= nc <= 260:
            n_candidates.add(nc)
    if n_base >= 80:
        n_candidates.add(n_base // 2)
    n_candidates = sorted(n_candidates)

    for n_try in n_candidates:
        if rem() < 60:
            break
        h_try = transfer_to_n(best_h, n_try)
        if len(h_try) % 2 == 0:
            q_try = sym_from_h(h_try)
            if rem() > 120:
                tl = time.time() + min(70, rem() * 0.35)
                q_o, c5_o, h_o = iter_tangent_sym(q_try, max_iter=200, rho_init=0.4, shrink=0.85, time_limit=tl)
                if c5_o < best_c5:
                    best_c5 = c5_o
                    best_h = h_o.copy()
                    h_try = h_o
            if rem() > 60:
                tl = time.time() + min(40, rem() * 0.3)
                h_f, c5_f = iter_tangent_full(h_try, max_iter=150, rho_init=0.35, shrink=0.85, time_limit=tl)
                if c5_f < best_c5:
                    best_c5 = c5_f
                    best_h = h_f.copy()
            if rem() > 30:
                h_new, sc = transfer_search(best_h, rng, n_pairs=60, n_grid=40)
                if sc < best_c5 - 1e-12:
                    best_c5 = sc
                    best_h = h_new
        else:
            if rem() > 60:
                tl = time.time() + min(40, rem() * 0.3)
                h_f, c5_f = iter_tangent_full(h_try, max_iter=150, rho_init=0.35, shrink=0.85, time_limit=tl)
                if c5_f < best_c5:
                    best_c5 = c5_f
                    best_h = h_f.copy()

    # Phase 5: Diverse restarts
    n_restarts = min(6, max(2, int(rem() / 120)))
    for i in range(n_restarts):
        if rem() < 90:
            break
        n_r = len(best_h)
        if i == 0:
            h_r = np.zeros(n_r)
            idx = rng.permutation(n_r)
            h_r[idx[:n_r // 2]] = 1.0
            h_r = 0.1 + 0.7 * h_r + 0.1 * rng.randn(n_r)
        elif i == 1:
            x_r = np.linspace(0, 2, n_r)
            h_r = 0.5 + 0.25 * np.sin(np.pi * x_r) + 0.1 * np.sin(3 * np.pi * x_r + rng.rand() * 6.28)
        elif i == 2:
            h_r = np.zeros(n_r)
            c1 = n_r // 5
            c2 = 4 * n_r // 5
            h_r[c1:c2] = 1.0
            h_r = 0.15 + 0.6 * h_r + 0.08 * rng.randn(n_r)
        elif i == 3:
            x_r = np.linspace(0, 2, n_r)
            h_r = 0.5 + 0.2 * np.sin(2 * np.pi * x_r + rng.rand() * 6.28) + 0.15 * np.sin(4 * np.pi * x_r)
        else:
            h_r = 0.3 + 0.4 * (rng.rand(n_r) > 0.5).astype(float) + 0.1 * rng.randn(n_r)
        h_r = project_feasible(h_r, n_r / 2.0)
        if n_r % 2 == 0 and rem() > 80:
            q_r = sym_from_h(h_r)
            tl = time.time() + min(35, rem() * 0.4)
            q_o, c5_o, h_o = iter_tangent_sym(q_r, max_iter=150, rho_init=0.5, shrink=0.85, time_limit=tl)
            if c5_o < best_c5:
                best_c5 = c5_o
                best_h = h_o.copy()
            if rem() > 40:
                tl = time.time() + min(25, rem() * 0.3)
                h_f, c5_f = iter_tangent_full(h_o if c5_o < best_c5 else h_r, max_iter=100, rho_init=0.35, shrink=0.85, time_limit=tl)
                if c5_f < best_c5:
                    best_c5 = c5_f
                    best_h = h_f.copy()
        elif rem() > 60:
            tl = time.time() + min(30, rem() * 0.4)
            h_f, c5_f = iter_tangent_full(h_r, max_iter=100, rho_init=0.4, shrink=0.85, time_limit=tl)
            if c5_f < best_c5:
                best_c5 = c5_f
                best_h = h_f.copy()

    # Phase 6: Final polish
    if rem() > 25 and best_h is not None:
        n_fin = len(best_h)
        if n_fin % 2 == 0 and rem() > 40:
            q_fin = sym_from_h(best_h)
            tl = time.time() + min(20, rem() * 0.5)
            q_o, c5_o, h_o = iter_tangent_sym(q_fin, max_iter=200, rho_init=0.2, shrink=0.8, time_limit=tl)
            if c5_o < best_c5:
                best_c5 = c5_o
                best_h = h_o.copy()
        if rem() > 25:
            tl = time.time() + min(15, rem() * 0.5)
            h_f, c5_f = iter_tangent_full(best_h, max_iter=200, rho_init=0.15, shrink=0.8, time_limit=tl)
            if c5_f < best_c5:
                best_c5 = c5_f
                best_h = h_f.copy()
        if rem() > 10:
            h_new, sc = transfer_search(best_h, rng, n_pairs=50, n_grid=60)
            if sc < best_c5 - 1e-12:
                best_c5 = sc
                best_h = h_new

    if best_h is None:
        m_fb = 63
        x_fb = np.linspace(0, 1, m_fb)
        q_fb = (0.5 + 0.3 * np.sin(2 * np.pi * x_fb)).clip(0, 1)
        best_h = build_sym_h(q_fb)

    best_h = np.asarray(best_h, dtype=float).reshape(-1)
    n_out = len(best_h)
    best_h = project_feasible(best_h, n_out / 2.0)
    best_c5 = eval_score(best_h)
    return best_h, best_c5, n_out
