import numpy as np

def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
        else:
            h += diff / n
    return np.clip(h, 0.0, 1.0)

def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def c5_of_h(h, dx):
    return float(compute_corrs(h).max() * dx)

def grad_for_lag(h, offset, n):
    """Gradient of R[center+offset] w.r.t. h, where offset is the lag shift."""
    grad = np.zeros(n)
    i_lo = max(0, -offset)
    i_hi = min(n - 1, n - 1 - offset)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx + offset] += (1.0 - h[idx])
    grad[idx] += (-h[idx + offset])
    return grad

def lp_directionfind(h, active_grads, eps_max):
    """
    Find direction d such that:
    - sum(d) = 0
    - g_k · d <= -delta for all active grads g_k
    - |d_i| <= eps_max * min(h_i, 1-h_i)
    
    Returns (d, success)
    """
    n = len(h)
    A_g = np.array(active_grads)  # shape (m, n)
    m = len(active_grads)
    
    # Variables: d[0..n-1]
    # Objective: minimize 0 (feasibility)
    c = np.zeros(n)
    
    # Inequality constraints: A_ub @ d <= b_ub
    # -g_k · d <= delta (we want g_k · d <= -small, i.e., -g_k · d >= small)
    # Actually for linprog: A_ub @ x <= b_ub
    # We want: g_k · d <= -delta  =>  A_ub row = g_k, b_ub = -delta
    
    # But we also need bounds: -eps_i <= d_i <= eps_i
    # And equality: sum(d) = 0
    
    A_ub = A_g  # shape (m, n)
    b_ub = np.full(m, -1e-8)  # want g_k · d <= -1e-8
    
    A_eq = np.array([[np.ones(n)]])
    b_eq = np.array([0.0])
    
    bounds_list = []
    for i in range(n):
        e = eps_max * min(h[i], 1.0 - h[i])
        bounds_list.append((-e, e))
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                     bounds=bounds_list, method='highs')
        if res.status == 0:
            # Scale down to be safe
            d = res.x
            d_norm = np.linalg.norm(d)
            if d_norm > 0:
                d = d / d_norm * eps_max
            # Verify
            for g in active_grads:
                if np.dot(g, d) > 0:
                    return None, False
            return d, True
        else:
            return None, False
    except Exception:
        return None, False

        corrs = compute_corrs(best_h)
        raw_max = corrs.max()

        margin = max(0.5, 0.02 * raw_max)
        active_indices = np.where(corrs > raw_max - margin)[0]

        active_grads = []
        for k_idx in active_indices:
            offset = k_idx - center
            g = grad_for_lag(best_h, offset, n)

                active_grads.append(g)
        
        if len(active_grads) == 0:
            break
        
        # Try LP direction
        found_improvement = False
        for eps in [0.5, 0.2, 0.1, 0.05, 0.01]:
            d, success = lp_directionfind(best_h, active_grads, eps)

                # Line search

                best_obj = c5_of_h(best_h, dx)
                for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
                    cand = project_feasible(best_h + alpha * d, target_sum)

                        best_obj = obj

                    c5 = c5_of_h(best_h, dx)

                    found_improvement = True
                    break

                    if c5 < best_c5:
                        best_c5 = c5
                    found_improvement = True

def slsqp_smooth_max(h0, dx, target_sum, beta, maxiter=500):
    n = len(h0)
    
    def obj(x):
        corrs = compute_corrs(x)
        m = corrs.max()
        # Use log-sum-exp for numerical stability
        shifted = corrs - m
        return (m + np.log(np.sum(np.exp(beta * shifted))) / beta) * dx
    
    def obj_grad(x):
        corrs = compute_corrs(x)
        m = corrs.max()
        shifted = corrs - m
        w = np.exp(beta * shifted)
        w_sum = w.sum()
        w /= w_sum
        grad = np.zeros(n)
        center = n - 1
        for k in range(len(corrs)):
            if w[k] < 1e-14:
                continue
            offset = k - center
            grad += w[k] * grad_for_lag(x, offset, n)
        return grad * dx
    
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum, 
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    
    try:
        res = minimize(obj, h0, jac=obj_grad, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        h_out = project_feasible(res.x, target_sum)
    except Exception:
        h_out = project_feasible(h0, target_sum)
    return h_out, c5_of_h(h_out, dx)

def make_structured_start(n, target_sum, mode=0):
    h = np.zeros(n)
    if mode == 0:
        # Golden ratio
        phi = (1.0 + np.sqrt(5.0)) / 2.0
        for k in range(int(target_sum)):
            pos = int((k * phi * n) % n)
            h[pos] = 1.0
    elif mode == 1:
        # Equally spaced with jitter
        spacing = n / int(target_sum)
        for k in range(int(target_sum)):
            pos = int(k * spacing + 0.5) % n
            h[pos] = 1.0
    elif mode == 2:
        # Two-scale structure
        half = int(target_sum / 2)
        for k in range(half):
            pos = int(2 * k * n / int(target_sum)) % n
            h[pos] = 1.0
        for k in range(int(target_sum) - half):
            pos = int((2 * k + 1) * n / int(target_sum) * 0.7) % n
            if h[pos] == 0:
                h[pos] = 1.0
    return project_feasible(h, target_sum)

def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
        else:
            h += diff / n
    return np.clip(h, 0.0, 1.0)


def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def c5_of_h(h, dx):
    return float(compute_corrs(h).max() * dx)


def grad_for_lag(h, offset, n):
    grad = np.zeros(n)
    i_lo = max(0, -offset)
    i_hi = min(n - 1, n - 1 - offset)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx + offset] += (1.0 - h[idx])
    grad[idx] += (-h[idx + offset])
    return grad


def lp_direction_maxmin(h, active_grads, eps_max):

    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds_list = []
    for i in range(n):
        e = eps_max * min(max(h[i], 1e-10), max(1.0 - h[i], 1e-10))
        bounds_list.append((-e, e))

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,

def momentum_softgrad(h, dx, target_sum, beta, lr, momentum, max_iters=500):
    n = len(h)
    center = n - 1
    h_cur = h.copy()
    v = np.zeros(n)
    beta_cur = beta
    best_h = h_cur.copy()
    best_c5 = c5_of_h(h_cur, dx)

    for it in range(max_iters):
        if it % 50 == 0:
            beta_cur = min(beta * 3.0, 5000.0)
        corrs = compute_corrs(h_cur)
        m = corrs.max()
        shifted = corrs - m
        w = np.exp(beta_cur * shifted)
        w_sum = w.sum()
        w /= w_sum
        grad = np.zeros(n)
        for k in range(len(corrs)):
            if w[k] < 1e-15:
                continue
            offset = k - center
            grad += w[k] * grad_for_lag(h_cur, offset, n)
        grad *= dx
        grad -= grad.mean()
        v = momentum * v + lr * grad
        h_new = project_feasible(h_cur - v, target_sum)
        c5_new = c5_of_h(h_new, dx)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h_cur = h_new
        if np.linalg.norm(v) < 1e-10:
            break

    return best_h, best_c5


def slsqp_smooth_max(h0, dx, target_sum, beta, maxiter=400):
    n = len(h0)
    center = n - 1

    def obj(x):
        corrs = compute_corrs(x)
        m = corrs.max()
        shifted = corrs - m
        return (m + np.log(np.sum(np.exp(beta * shifted))) / beta) * dx

    def obj_grad(x):
        corrs = compute_corrs(x)
        m = corrs.max()
        shifted = corrs - m
        w = np.exp(beta * shifted)
        w_sum = w.sum()
        w /= w_sum
        grad = np.zeros(n)
        for k in range(len(corrs)):
            if w[k] < 1e-15:
                continue
            offset = k - center
            grad += w[k] * grad_for_lag(x, offset, n)
        return grad * dx

    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0, jac=obj_grad, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        h_out = project_feasible(res.x, target_sum)
    except Exception:
        h_out = project_feasible(h0, target_sum)
    return h_out, c5_of_h(h_out, dx)


def binary_local_search(h0, dx, rng, max_iters=200000, patience=6000):
    n = len(h0)
    target_sum = int(round(n / 2))
    h = np.zeros(n)
    sorted_idx = np.argsort(-h0)
    h[sorted_idx[:target_sum]] = 1.0
    corrs = compute_corrs(h)
    best_c5 = corrs.max() * dx
    best_h = h.copy()
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    no_improve = 0
    temp = 0.025
    cooling = 0.99992
    m_vals = np.arange(-(n - 1), n)

    for it in range(max_iters):
        if it % 2000 == 0:
            temp *= cooling
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        change = np.zeros(2 * n - 1)
        valid_im = (i + m_vals >= 0) & (i + m_vals < n)
        change[valid_im] += -(1.0 - h[i + m_vals[valid_im]])
        valid_im2 = (i - m_vals >= 0) & (i - m_vals < n)
        change[valid_im2] += h[i - m_vals[valid_im2]]
        valid_jm = (j + m_vals >= 0) & (j + m_vals < n)
        change[valid_jm] += (1.0 - h[j + m_vals[valid_jm]])
        valid_jm2 = (j - m_vals >= 0) & (j - m_vals < n)
        change[valid_jm2] += -h[j - m_vals[valid_jm2]]
        new_corrs = corrs + change
        new_max = new_corrs.max()
        new_c5 = new_max * dx

        if new_c5 <= best_c5 + 1e-15:
            h[i], h[j] = 0.0, 1.0
            corrs = new_corrs
            best_c5 = new_c5
            best_h = h.copy()
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            no_improve = 0
        else:
            delta_val = new_c5 - best_c5
            if temp > 1e-9 and rng.random() < np.exp(-delta_val / temp):
                h[i], h[j] = 0.0, 1.0
                corrs = new_corrs
                ones = np.where(h == 1.0)[0]
                zeros = np.where(h == 0.0)[0]
                no_improve = 0
            else:
                no_improve += 1
        if no_improve > patience:
            break

    return best_h, best_c5


def make_structured_start(n, target_sum, mode=0, rng=None):
    h = np.zeros(n)
    ts = int(target_sum)
    if mode == 0:
        phi = (1.0 + np.sqrt(5.0)) / 2.0
        seen = set()
        for k in range(n):
            pos = int((k * phi * n) % n)
            if pos not in seen and len(seen) < ts:
                h[pos] = 1.0
                seen.add(pos)
    elif mode == 1:
        spacing = n / ts
        for k in range(ts):
            pos = int(k * spacing + 0.3) % n
            h[pos] = 1.0
    elif mode == 2:
        if rng is not None:
            idx = np.arange(n)
            rng.shuffle(idx)
            h[idx[:ts]] = 1.0
        else:
            h[:ts] = 1.0
    elif mode == 3:
        # Two-scale
        half = ts // 2
        for k in range(half):
            pos = (2 * k * n) // ts
            h[pos % n] = 1.0
        for k in range(ts - half):
            pos = int((2 * k + 1) * n * 0.618 / ts)
            if h[pos % n] == 0:
                h[pos % n] = 1.0
        # Fill remaining
        count = int(h.sum())
        if count < ts:
            for k in range(n):
                if h[k] == 0 and count < ts:
                    h[k] = 1.0
                    count += 1
    return project_feasible(h, target_sum)

        h_lp2, c5_lp2 = active_set_lpData(best_h, dx, n, max_iters=300, margin_frac=0.02)
        if c5_lp2 < best_c5:
            best_c5 = c5_lp2
            best_h = h_lp2

    # Phase 5: Binary SA
    if remaining_time(t0, budget_s) > 100:
        h_sa, c5_sa = binary_local_search(best_h, dx, rng, max_iters=250000, patience=6000)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa

    # Phase 6: SLSQP polish from binary
    if remaining_time(t0, budget_s) > 35:
        h_opt, c5_val = slsqp_smooth_max(best_h, dx, target_sum, 2000, maxiter=800)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_opt

    # Phase 7: LP polish
    if remaining_time(t0, budget_s) > 30:
        h_lp3, c5_lp3 = active_set_lpData(best_h, dx, n, max_iters=300, margin_frac=0.01)
        if c5_lp3 < best_c5:
            best_c5 = c5_lp3
            best_h = h_lp3

    # Phase 8: Try different n values
    for n_try in [96, 80, 64, 112, 144]:
        if remaining_time(t0, budget_s) < 50:
            break
        n2 = n_try
        dx2 = 2.0 / n2
        ts2 = n2 / 2.0
        x_orig = np.linspace(0.0, 2.0, len(best_h), endpoint=False)
        x_new = np.linspace(0.0, 2.0, n2, endpoint=False)
        h_interp = np.interp(x_new, x_orig, best_h)
        h_interp = project_feasible(h_interp, ts2)
        # Optimize
        h_opt2, c5_val2 = slsqp_smooth_max(h_interp, dx2, ts2, 300, maxiter=300)
        if c5_val2 < best_c5:
            if remaining_time(t0, budget_s) > 30:
                h_lp4, c5_lp4 = active_set_lpData(h_opt2, dx2, n2, max_iters=200, margin_frac=0.03)
                if c5_lp4 < c5_val2:
                    c5_val2 = c5_lp4
                    h_opt2 = h_lp4
            if c5_val2 < best_c5:
                best_c5 = c5_val2
                best_h = h_opt2
        # Also try structured starts at this n
        for mode in [0, 1]:
            if remaining_time(t0, budget_s) < 30:
                break
            h_str = make_structured_start(n2, ts2, mode, rng)
            h_s2, c5_s2 = slsqp_smooth_max(h_str, dx2, ts2, 300, maxiter=200)
            if c5_s2 < best_c5:
                best_c5 = c5_s2
                best_h = h_s2

    # Phase 9: Final binary SA on best
    if remaining_time(t0, budget_s) > 80:
        n_b = len(best_h)
        dx_b = 2.0 / n_b
        ts_b = n_b / 2.0
        h_sa2, c5_sa2 = binary_local_search(best_h, dx_b, rng, max_iters=150000, patience=5000)
        if c5_sa2 < best_c5:
            best_c5 = c5_sa2
            best_h = h_sa2

    # Phase 10: Final LP + SLSQP
    n_b = len(best_h)
    dx_b = 2.0 / n_b
    ts_b = n_b / 2.0
    if remaining_time(t0, budget_s) > 30:
        h_opt, c5_val = slsqp_smooth_max(best_h, dx_b, ts_b, 3000, maxiter=1000)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_opt
    if remaining_time(t0, budget_s) > 20:
        h_lp5, c5_lp5 = active_set_lpData(best_h, dx_b, n_b, max_iters=200, margin_frac=0.01)
        if c5_lp5 < best_c5:
            best_c5 = c5_lp5
            best_h = h_lp5

    # Final cleanup
    best_h = project_feasible(best_h, ts_b)
    best_c5 = c5_of_h(best_h, dx_b)

    return best_h, best_c5, len(best_h)
