        return smooth_c5(x, beta)

import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time


def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.5, 1.5
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] = np.clip(h[mask] + diff / mask.sum(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        idx = np.argmin(np.abs(h - 0.5))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h


def c5_exact(h):
    n = len(h)
    dx = 2.0 / n
    corrs = np.correlate(h, 1.0 - h, mode='full')
    return float(corrs.max() * dx)


def smooth_c5(h, beta):
    n = len(h)
    dx = 2.0 / n
    corrs = np.correlate(h, 1.0 - h, mode='full')
    return float(logsumexp(beta * corrs) / beta * dx)


def smooth_c5_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corrs = np.correlate(h, 1.0 - h, mode='full')
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        w /= w_sum
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-12:
            continue
        m = k - (n - 1)
        n_lo = max(0, -m)
        n_hi = min(n - 1, n - 1 - m)
        if n_lo > n_hi:
            continue
        # Term 1: (1-h[i-m]) where i-m in [n_lo, n_hi], i in [n_lo+m, n_hi+m]
        i_lo1 = max(0, n_lo + m)
        i_hi1 = min(n - 1, n_hi + m)
        if i_lo1 <= i_hi1:
            idx = np.arange(i_lo1, i_hi1 + 1)
            grad[idx] += wk * (1.0 - h[idx - m])
        # Term 2: -h[i+m] where i in [n_lo, n_hi] and 0<=i+m<n
        i_lo2 = max(0, n_lo, -m)
        i_hi2 = min(n - 1, n_hi, n - 1 - m)
        if i_lo2 <= i_hi2:
            idx = np.arange(i_lo2, i_hi2 + 1)
            grad[idx] -= wk * h[idx + m]
    return grad * dx


def slsqp_opt(h0, beta, maxiter, ftol=1e-14):
    n = len(h0)
    target_sum = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(smooth_c5, h0, jac=lambda x: smooth_c5_grad(x, beta),
                       args=(beta,), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        h_out = project_feasible(res.x, target_sum)
    except Exception:
        h_out = project_feasible(h0, target_sum)
    return h_out, c5_exact(h_out)


def binary_local_search(h, max_iter=3000, rng=None):

            a = ones[rng.randint(len(ones))]

def make_alternating(n, rng):
    h = np.zeros(n)
    h[::2] = 1.0
    if h.sum() != n / 2:
        h = (np.arange(n) % 4 < 2).astype(float)
    return project_feasible(h, n / 2.0)


def make_block(n, rng):
    h = np.zeros(n)
    h[:n // 2] = 1.0
    if h.sum() != n / 2:
        h = project_feasible(h, n / 2.0)
    return h


def make_sine(n, phase, rng):
    x = np.linspace(0, 2 * np.pi, n, endpoint=False)
    h = ((np.sin(x + phase) + 1) / 2).astype(float)
    return project_feasible(h, n / 2.0)


def make_random_binary(n, rng):
    h = np.zeros(n)
    idx = rng.permutation(n)
    h[idx[:n // 2]] = 1.0
    return h


def interpolate_to_n(h, n_new):
    n_old = len(h)
    x_old = np.arange(n_old) * (2.0 / n_old)
    x_new = np.arange(n_new) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new / 2.0)
