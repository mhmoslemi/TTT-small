    weights = np.maximum(weights, 0.01)
    weights /= weights.sum()
    grad = G_a.T @ weights
    grad -= grad.mean()
    norm = np.linalg.norm(grad)
    if norm < 1e-15:
        return None
    return grad / norm

def kkt_residual(h, active_set):
    """Check if KKT conditions are approximately satisfied."""
    n = len(h)
    dx = 2.0 / n
    G = build_gradient(h, n, dx)
    G_a = G[active_set, :]
    # Project onto sum-0 subspace
    G_proj = G_a - G_a.mean(axis=1, keepdims=True)
    # Check if there exists lambda >= 0, sum=1 such that G_proj.T @ lambda = 0
    # This is: minimize ||G_proj.T @ lambda||^2 s.t. lambda >= 0, sum = 1
    # Equivalent: check if 0 is in the convex hull of rows of G_proj (transposed)
    m = len(active_set)
    if m <= 1:
        return np.max(np.abs(G_proj.T)) if m == 1 else 0.0
    # Simple check: is G_proj.T @ uniform = 0?
    uniform = np.ones(m) / m
    r = G_proj.T @ uniform
    return np.linalg.norm(r)

def line_search(h0, direction, current_score, max_step=0.1, min_step=1e-5):
    n = len(h0)
    best_h = h0.copy()
    best_s = current_score
    steps = [max_step, max_step*0.5, max_step*0.25, max_step*0.125,
             max_step*0.0625, max_step*0.03, max_step*0.01]
    for a in steps:
        h_try = repair_feasible(h0 - a * direction)
        s = exact_score(h_try)
        if s < best_s - 1e-14:
            best_h = h_try
            best_s = s
    return best_h, best_s

def symmetrize(h):
    return (h + h[::-1]) / 2.0

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

def optimize_single(h, t_start, budget_s, max_iters=500):
    """Main optimization loop for a single dimension."""
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c = exact_score(h)
    radius = 0.05
    no_improve_count = 0
    
    for it in range(max_iters):
        elapsed = time.perf_counter() - t_start
        if budget_s - elapsed < 5:
            break
        
        profile = compute_profile(h)
        c = np.max(profile)
        if c < best_c - 1e-14:
            best_c = c
            best_h = h.copy()
            no_improve_count = 0
            radius = min(radius * 1.2, 0.15)
        else:
            no_improve_count += 1
        
        if no_improve_count > 30:
            break
        
        active, T, thresh = get_active_set(profile, top_k=12, beam=None)
        
        # Try LP trust region
        h_lp, t_lp = lp_step(h, active, radius)
        if h_lp is not None:
            h_new, s_new = line_search(h, h - h_lp, c, max_step=0.1)
            if s_new < best_c - 1e-14:
                h = h_new
                continue
        
        # Try gradient direction
        grad = gradient_direction(h, active)
        if grad is not None:
            h_new, s_new = line_search(h, grad, c, max_step=0.05)
            if s_new < best_c - 1e-14:
                h = h_new
                no_improve_count = 0
                continue
        
        # Try with expanded active set
        if no_improve_count > 5:
            active2, _, _ = get_active_set(profile, top_k=20, beam=0.05)
            h_lp2, _ = lp_step(h, active2, radius * 0.5)
            if h_lp2 is not None:
                h_new, s_new = line_search(h, h - h_lp2, c, max_step=0.05)
                if s_new < best_c - 1e-14:
                    h = h_new
                    no_improve_count = 0
                    continue
            grad2 = gradient_direction(h, active2)
            if grad2 is not None:
                h_new, s_new = line_search(h, grad2, c, max_step=0.03)
                if s_new < best_c - 1e-14:
                    h = h_new
                    no_improve_count = 0
                    continue
        
        # Try with contracted active set (top 5 only)
        if no_improve_count > 10:
            active3, _, _ = get_active_set(profile, top_k=5, beam=0.005)
            h_lp3, _ = lp_step(h, active3, radius * 0.3)
            if h_lp3 is not None:
                h_new, s_new = line_search(h, h - h_lp3, c, max_step=0.03)
                if s_new < best_c - 1e-14:
                    h = h_new
                    no_improve_count = 0
                    continue
        
        # Shrink radius
        radius *= 0.8
        if radius < 0.001:
            break
    
    return best_h, best_c

def fine_polish(h, t_start, budget_s, max_iters=200):
    """Very fine local search with small steps."""
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best_c = exact_score(h)
    
    for it in range(max_iters):
        elapsed = time.perf_counter() - t_start
        if budget_s - elapsed < 3:
            break
        
        profile = compute_profile(h)
        c = np.max(profile)
        active, T, _ = get_active_set(profile, top_k=8, beam=0.003)
        
        # Try many small perturbations
        best_step = None
        best_s = best_c
        
        # Gradient-based small steps
        grad = gradient_direction(h, active)
        if grad is not None:
            for a in [0.002, 0.001, 0.0005, 0.0002]:
                h_try = repair_feasible(h - a * grad)
                s = exact_score(h_try)
                if s < best_s - 1e-14:
                    best_s = s
                    best_step = h_try
        
        # LP with very small radius
        h_lp, _ = lp_step(h, active, 0.005)
        if h_lp is not None:
            for alpha in [1.0, 0.5, 0.25, 0.1]:
                h_try = repair_feasible(h + alpha * (h_lp - h))
                s = exact_score(h_try)
                if s < best_s - 1e-14:
                    best_s = s
                    best_step = h_try
        
        if best_step is not None:
            h = best_step
            best_c = best_s
        else:
            break
    
    return h, best_c

def make_structured_start(n, rng, mode=0):
    """Create structured starting points."""
    h = np.zeros(n)
    if mode == 0:
        # Parent-like smooth
        h[:] = 0.5
        for i in range(n):
            h[i] += 0.3 * np.sin(np.pi * i / n + rng.uniform(0, 2*np.pi))
    elif mode == 1:
        # Block structure
        nb = rng.randint(3, 7)
        block = n // nb
        for b in range(nb):
            h[b*block:(b+1)*block] = rng.uniform(0.2, 0.8)
    elif mode == 2:
        # Low discrepancy
        phi = 0.6180339887498949
        for i in range(n):
            h[i] = 0.5 + 0.4 * np.sin(2*np.pi * ((i * phi) % 1.0))
    elif mode == 3:
        # Anti-symmetric around center
        half = n // 2
        h[:half] = rng.uniform(0.1, 0.9, half)
        h[half:] = 1.0 - h[:half][::-1]
        if n % 2 == 1:
            h[half] = 0.5
    else:
        h[:] = rng.uniform(0.3, 0.7, n)
    return repair_feasible(h)

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()
    rng = np.random.RandomState(seed)
    
    def time_left():
        return budget_s - (time.perf_counter() - t_start)
    
    # Load parent
    parent = None
    try:
        parent = np.array(initial_h_values, dtype=np.float64)
    except (NameError, Exception):
        pass
    
    if parent is None or len(parent) < 4 or not np.isfinite(parent).all():
        parent = np.full(70, 0.5)
    
    parent = np.clip(parent, 0.0, 1.0)
    parent = repair_feasible(parent)
    c_parent = exact_score(parent)
    
    best_h = parent.copy()
    best_c = c_parent
    
    # Try symmetrization
    h_sym = repair_feasible(symmetrize(parent))
    c_sym = exact_score(h_sym)
    if c_sym < best_c - 1e-12:
        best_h = h_sym
        best_c = c_sym
    
    n_best = len(best_h)
    
    # Phase 1: Optimize at parent dimension
    if time_left() > 100:
        h_opt, c_opt = optimize_single(best_h, t_start, budget_s, max_iters=200)
        if c_opt < best_c - 1e-14:
            best_h = h_opt
            best_c = c_opt
    
    # Phase 2: Fine polish
    if time_left() > 50:
        h_pol, c_pol = fine_polish(best_h, t_start, budget_s, max_iters=100)
        if c_pol < best_c - 1e-14:
            best_h = h_pol
            best_c = c_pol
    
    # Phase 3: Try different dimensions
    n0 = len(best_h)
    dims_to_try = []
    for d in [n0-4, n0-2, n0+2, n0+4, n0+6, n0+8, n0//2+2, n0*2]:
        if 20 <= d <= 200 and d != n0 and d % 2 == 0:
            dims_to_try.append(d)
    dims_to_try = list(set(dims_to_try))
    dims_to_try.sort(key=lambda d: abs(d - n0))
    
    for new_n in dims_to_try[:6]:
        if time_left() < 60:
            break
        h_interp = interpolate_to_n(best_h, new_n)
        h_opt, c_opt = optimize_single(h_interp, t_start, budget_s, max_iters=150)
        if c_opt < best_c - 1e-14:
            best_h = h_opt
            best_c = c_opt
            # Polish new best
            if time_left() > 40:
                h_pol, c_pol = fine_polish(best_h, t_start, budget_s, max_iters=50)
                if c_pol < best_c - 1e-14:
                    best_h = h_pol
                    best_c = c_pol
    
    # Phase 4: Try structured starts at best dimension
    n_b = len(best_h)
    if time_left() > 80:
        for mode in range(4):
            if time_left() < 30:
                break
            h_struct = make_structured_start(n_b, rng, mode)
            # Blend with best to stay in the basin
            h_blend = 0.5 * best_h + 0.5 * h_struct
            h_blend = repair_feasible(h_blend)
            c_blend = exact_score(h_blend)
            if c_blend < best_c + 0.01:  # Only try if reasonably close
                h_opt, c_opt = optimize_single(h_blend, t_start, budget_s, max_iters=100)
                if c_opt < best_c - 1e-14:
                    best_h = h_opt
                    best_c = c_opt
    
    # Phase 5: Interpolation + re-optimization at finer grid
    if time_left() > 60:
        for scale in [2, 3]:
            new_n = len(best_h) * scale
            if new_n > 250:
                continue
            h_fine = interpolate_to_n(best_h, new_n)
            h_opt, c_opt = optimize_single(h_fine, t_start, budget_s, max_iters=100)
            if c_opt < best_c - 1e-14:
                best_h = h_opt
                best_c = c_opt
    
    # Phase 6: Final intensive polish
    if time_left() > 30:
        h_pol, c_pol = fine_polish(best_h, t_start, budget_s, max_iters=150)
        if c_pol < best_c - 1e-14:
            best_h = h_pol
            best_c = c_pol
    
    # Phase 7: Attempt SLSQP epigraph as final refinement
    if time_left() > 40:
        n_f = len(best_h)
        dx_f = 2.0 / n_f
        M_f = 2 * n_f - 1
        c0_f = exact_score(best_h)
        
        def obj_fn(x):
            return x[-1]
        
        def obj_jac_fn(x):
            g = np.zeros(n_f + 1)
            g[-1] = 1.0
            return g
        
        def eq_fn(x):
            return x[:-1].sum() - n_f / 2.0
        
        def eq_jac_fn(x):
            g = np.zeros(n_f + 1)
            g[:-1] = 1.0
            return g
        
        def ineq_fn(x):
            h = x[:-1]
            t = x[-1]
            prof = np.correlate(h, 1 - h, mode="full") * dx_f
            return t - prof
        
        def ineq_jac_fn(x):
            h = x[:-1]
            G = build_gradient(h, n_f, dx_f)
            J = np.zeros((M_f, n_f + 1))
            J[:, :-1] = -G
            J[:, -1] = 1.0
            return J
        
        x0 = np.concatenate([best_h, [c0_f + 1e-4]])
        bounds = [(0.0, 1.0)] * n_f + [(0.0, None)]
        cons = [
            {'type': 'eq', 'fun': eq_fn, 'jac': eq_jac_fn},
            {'type': 'ineq', 'fun': ineq_fn, 'jac': ineq_jac_fn}
        ]
        try:
            res = minimize(obj_fn, x0, method='SLSQP', jac=obj_jac_fn,
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 300, 'ftol': 1e-12})
            if res.success:
                h_sol = repair_feasible(res.x[:-1])
                c_sol = exact_score(h_sol)
                if c_sol < best_c - 1e-14:
                    best_h = h_sol
                    best_c = c_sol
        except Exception:
            pass
    
    # Phase 8: Final micro-polish
    if time_left() > 15:
        h_pol, c_pol = fine_polish(best_h, t_start, budget_s, max_iters=100)
        if c_pol < best_c - 1e-14:
            best_h = h_pol
            best_c = c_pol
    
    # Final validation
    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)
    
    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            final_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 2:
            final_c = float(eval_result[0])
    except Exception:
        pass
    
    return best_h, final_c, len(best_h)
