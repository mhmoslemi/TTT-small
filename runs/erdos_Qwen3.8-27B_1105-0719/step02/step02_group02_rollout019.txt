def solve_minmax_slsqp(h0, dx, target_sum, maxiter=2000):
    n = len(h0)
    corrs0 = np.correlate(h0, 1-h0, mode='full')
    T0 = corrs0.max()
    
    # Decision variable: [h (n entries), T (1 entry)]
    x0 = np.concatenate([h0, [T0 + 0.01]])
    
    bounds = [(0,1)]*n + [(0, T0+1)]
    
    def objective(x):
        return x[-1]  # minimize T
    
    def objective_grad(x):
        g = np.zeros(n+1)
        g[-1] = 1.0
        return g
    
    # Equality constraint: sum(h) = target_sum
    def eq_fun(x):
        return x[:n].sum() - target_sum
    
    def eq_jac(x):
        g = np.zeros(n+1)
        g[:n] = 1.0
        return g
    
    # Inequality constraints: T - c[k] >= 0 for top M lags
    # We'll use all lags for safety
    
    def ineq_fun(x):
        h = x[:n]
        T = x[-1]
        c = np.correlate(h, 1-h, mode='full')
        return T - c  # all should be >= 0
    
    def ineq_jac(x):
        h = x[:n]
        T = x[-1]
        n_h = len(h)
        g = np.zeros((2*n_h-1, n_h+1))
        for k in range(2*n_h-1):
            offset = k - (n_h-1)
            if offset == 0:
                g[k, :n_h] = 1.0 - 2.0*h
            else:
                i_lo = max(0, -offset)
                i_hi = min(n_h-1, n_h-1-offset)
                if i_lo <= i_hi:
                    idx = np.arange(i_lo, i_hi+1)
                    g[k, idx+offset] = 1.0 - h[idx]
                    g[k, idx] = -h[idx+offset]
            g[k, -1] = 1.0  # gradient of T w.r.t. T is 1
        return g
    
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    
    res = minimize(objective, x0, jac=objective_grad, method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-15})
    
    h_out = np.clip(res.x[:n], 0, 1)
    h_out = project_feasible(h_out, target_sum)
    return h_out, c5_of_h(h_out, dx)

    d = k - (n - 1)
    grad = np.zeros(n)
    if d >= 0:
        # i ranges from 0 to n-1-d
        i_max = n - 1 - d
        # First index contribution: for i in [0, i_max], dc/dh[i] += (1-h[i+d])
        grad[:i_max+1] += (1.0 - h[d:i_max+1+d])  # h[i+d] for i in [0, i_max]
        # Second index contribution: for i in [0, i_max], the second index is i+d, 
        # which ranges from d to n-1. dc/dh[i+d] += -h[i]
        grad[d:] += -h[:i_max+1]  # -h[i] for second index i+d, i in [0,i_max]
    else:

        i_min = d_prime

from scipy.special import logsumexp

def project_feasible(x, target_sum):
    N = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)

            if abs(diff2) > 1e-12:
