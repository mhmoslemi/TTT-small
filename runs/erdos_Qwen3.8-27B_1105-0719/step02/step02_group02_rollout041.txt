```python
    h: binary array of length n

            # Δ_k = sum of up to 4 terms
            # Term 1: i as first index, valid when 0 <= i+k <= n-1
            # Term 2: j as first index, valid when 0 <= j+k <= n-1

                spec3 = mask3 & (i - k == j)
```

- Compute M_i = corrs + P_i (the "base" after removing position i)
- For each j, new_max_j = max(M_i + Q_j + 2*e_{i-j})
- We want argmin_j new_max_j

- Compute M_i + Q_j for all j (this is a matrix of size n_zeros × (2n-1))
- Add the 2*e_{i-j} correction (which is just adding 2 to one specific k for each j)
- Take the max over k for each j

```python
def local_search_binary(h, max_rounds=1000):
    """Steepest descent local search for binary solutions using pair swaps."""
    n = len(h)
    h = (h > 0.5).astype(np.float64)
    
    corrs = np.correlate(h, 1.0 - h, mode='full')
    current_max = corrs.max()
    
    k = np.arange(-(n-1), n)  # shift values
    
    for _ in range(max_rounds):
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        best_max = current_max
        best_i, best_j = -1, -1
        
        for i in ones:
            for j in zeros:
                # Compute delta
                delta = np.zeros(2*n-1)
                
                # P_i contribution (removing i):
                # -C_i: -(1-h[i+k]) for valid i+k
                mask_ic = (i + k >= 0) & (i + k < n)
                delta[mask_ic] -= (1.0 - h[i + k[mask_ic]])
                # B_i: +h[i-k] for valid i-k
                mask_ib = (i - k >= 0) & (i - k < n)
                delta[mask_ib] += h[i - k[mask_ib]]
                
                # Q_j contribution (adding j):
                # C_j: +(1-h[j+k]) for valid j+k
                mask_jc = (j + k >= 0) & (j + k < n)
                delta[mask_jc] += (1.0 - h[j + k[mask_jc]])
                # D_j: -h[j-k] for valid j-k
                mask_jd = (j - k >= 0) & (j - k < n)
                delta[mask_jd] -= h[j - k[mask_jd]]
                
                # Correction at k = i-j
                k_spec = i - j
                if -(n-1) <= k_spec <= n-1:
                    delta[n-1 + k_spec] += 2.0
                
                new_max = (corrs + delta).max()
                if new_max < best_max:
                    best_max = new_max
                    best_i, best_j = i, j
        
        if best_i < 0:
            break  # No improving swap found
        
        h[best_i] = 0.0
        h[best_j] = 1.0
        corrs = np.correlate(h, 1.0 - h, mode='full')
        current_max = corrs.max()
    
    dx = 2.0 / n
    return h, float(current_max * dx)
```

```python
def precompute_contributions(h, n):
    """Precompute P_p and Q_p for all positions p."""
    k = np.arange(-(n-1), n)
    n_corr = 2*n - 1
    P = np.zeros((n, n_corr))  # P[p] = -C_p + B_p (removal contribution)
    Q = np.zeros((n, n_corr))  # Q[p] = C_p + D_p (addition contribution)
    
    for p in range(n):
        # -C_p: -(1-h[p+k]) for valid p+k
        mask_c = (p + k >= 0) & (p + k < n)
        P[p, mask_c] -= (1.0 - h[p + k[mask_c]])
        Q[p, mask_c] += (1.0 - h[p + k[mask_c]])

        mask_b = (p - k >= 0) & (p - k < n)
        P[p, mask_b] += h[p - k[mask_b]]
        Q[p, mask_b] -= h[p - k[mask_b]]
```

```python
def local_search_binary_v2(h, max_rounds=500):

    h = (h > 0.5).astype(np.float64)
    k = np.arange(-(n-1), n)
    n_corr = 2*n - 1
    
    corrs = np.correlate(h, 1.0 - h, mode='full')
    
    for _ in range(max_rounds):
        corrs = np.correlate(h, 1.0 - h, mode='full')
        current_max = corrs.max()
        
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        
        # Precompute P and Q
        P = np.zeros((len(ones), n_corr))
        Q = np.zeros((len(zeros), n_corr))
        
        for idx_i, i in enumerate(ones):
            mask_c = (i + k >= 0) & (i + k < n)
            P[idx_i, mask_c] -= (1.0 - h[i + k[mask_c]])
            mask_b = (i - k >= 0) & (i - k < n)
            P[idx_i, mask_b] += h[i - k[mask_b]]
        
        for idx_j, j in enumerate(zeros):
            mask_c = (j + k >= 0) & (j + k < n)
            Q[idx_j, mask_c] += (1.0 - h[j + k[mask_c]])
            mask_b = (j - k >= 0) & (j - k < n)
            Q[idx_j, mask_b] -= h[j - k[mask_b]]
        
        # Find best swap
        best_max = current_max + 1  # Must improve
        best_ii, best_jj = -1, -1
        
        for ii in range(len(ones)):
            i = ones[ii]
            base = corrs + P[ii]
            # For each zero position j, compute max(base + Q[jj] + 2*e_{i-j})
            # base + Q[jj] is base (broadcast) + Q[jj]
            # For the correction: add 2 at position n-1+(i-j)
            for jj in range(len(zeros)):
                j = zeros[jj]
                candidate = base + Q[jj]
                k_spec = i - j
                if -(n-1) <= k_spec <= n-1:
                    candidate[n-1 + k_spec] += 2.0
                new_max = candidate.max()
                if new_max < best_max:
                    best_max = new_max
                    best_ii, best_jj = ii, jj
        
        if best_ii < 0:
            break
        
        i = ones[best_ii]
        j = zeros[best_jj]
        h[i] = 0.0
        h[j] = 1.0
    
    dx = 2.0 / n
    return h, float(corrs.max() * dx)
```

```python
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time


def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(64):
        mid = (lo + hi) / 2.0
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h


def c5_of_h(h, dx):
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)


def smooth_max(h, dx, beta):
    corrs = np.correlate(h, 1.0 - h, mode='full')
    return float(logsumexp(beta * corrs) / beta * dx)


def slsqp_opt(h0, dx, target_sum, beta, maxiter=500):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    ones_arr = np.ones(n)
    
    def obj(x):
        return smooth_max(x, dx, beta)
    
    def obj_grad(x):
        corrs = np.correlate(x, 1.0 - x, mode='full')
        shifted = corrs - corrs.max()
        w = np.exp(beta * shifted)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones_like(corrs) / len(corrs)
        else:
            w /= w_sum
        grad = np.zeros(n)
        for k_idx in range(len(corrs)):
            if w[k_idx] < 1e-14:
                continue
            k = k_idx - (n - 1)
            # grad_corr_k for shift k
            if k == 0:
                g = 1.0 - 2.0 * x
            else:
                g = np.zeros(n)
                i_lo = max(0, -k)
                i_hi = min(n - 1, n - 1 - k)
                if i_lo <= i_hi:
                    idx = np.arange(i_lo, i_hi + 1)
                    g[idx + k] += (1.0 - x[idx])
                    g[idx] += (-x[idx + k])
            grad += w[k_idx] * g
        return grad * dx
    
    def cons_fun(x):
        return x.sum() - target_sum
    
    def cons_jac(x):
        return ones_arr
    
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    
    try:
        res = minimize(obj, h0, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        h_out = project_feasible(res.x, target_sum)
    except Exception:
        h_out = project_feasible(h0, target_sum)
    return h_out, c5_of_h(h_out, dx)

    """Fast binary local search with pair swaps (steepest descent)."""
    n = len(h)
    h = (h > 0.5).astype(np.float64)
    if rng is None:
        rng = np.random.RandomState(42)
    
    k = np.arange(-(n-1), n)
    n_corr = 2 * n - 1
    
    for _ in range(max_rounds):
        corrs = np.correlate(h, 1.0 - h, mode='full')
        current_max = corrs.max()
        
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        n_ones = len(ones)
        n_zeros = len(zeros)
        
        if n_ones == 0 or n_zeros == 0:
            break
        
        # Precompute P (removal) and Q (addition) contributions
        # P[i] for each one position i
        # Q[j] for each zero position j
        P = np.zeros((n_ones, n_corr))
        Q = np.zeros((n_zeros, n_corr))
        
        for ii in range(n_ones):
            i = ones[ii]
            mc = (i + k >= 0) & (i + k < n)
            P[ii, mc] -= (1.0 - h[i + k[mc]])
            mb = (i - k >= 0) & (i - k < n)
            P[ii, mb] += h[i - k[mb]]
        
        for jj in range(n_zeros):
            j = zeros[jj]
            mc = (j + k >= 0) & (j + k < n)
            Q[jj, mc] += (1.0 - h[j + k[mc]])
            mb = (j - k >= 0) & (j - k < n)
            Q[jj, mb] -= h[j - k[mb]]
        
        # Find best swap
        best_max = current_max  # Must be strictly better
        best_ii, best_jj = -1, -1
        
        for ii in range(n_ones):
            i = ones[ii]
            base = corrs + P[ii]  # shape (n_corr,)
            # candidates[jj] = base + Q[jj] + 2*[k==i-j]
            candidates = base[None, :] + Q  # shape (n_zeros, n_corr)
            # Add corrections
            for jj in range(n_zeros):
                k_spec = i - zeros[jj]
                if -(n-1) <= k_spec <= n-1:
                    candidates[jj, n-1 + k_spec] += 2.0
            new_maxes = candidates.max(axis=1)
            jj_best = new_maxes.argmin()
            if new_maxes[jj_best] < best_max:
                best_max = new_maxes[jj_best]
                best_ii = ii
                best_jj = jj_best
        
        if best_ii < 0:
            break  # Local optimum
        
        h[ones[best_ii]] = 0.0
        h[zeros[best_jj]] = 1.0
    
    dx = 2.0 / n
    return h, c5_of_h(h, dx)

    n = len(h)

    best_h = h.copy()
    best_c5 = current_max * dx
    
    n_iters = max_iters

        zeros = np.where(h == 0.0)[0]

        j = zeros[rng.randint(len(zeros))]
        if i == j:
            continue
        
        # Compute new overlaps after swap
        h[i] = 0.0
        h[j] = 1.0
        new_corrs = np.correlate(h, 1.0 - h, mode='full')
        new_max = new_corrs.max()
        
        delta = (new_max - current_max) * dx
        
        if delta < 0 or rng.rand() < np.exp(-delta / max(temp, 1e-10)):
            current_max = new_max
            corrs = new_corrs
            new_c5 = current_max * dx
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            # Revert
            h[i] = 1.0
            h[j] = 0.0
    
    return best_h, best_c5
```

Wait, the SA is doing a full correlate at each iteration, which is O(n) per iteration (since correlate is O(n²) in general... actually numpy's correlate is O(n * m) where n and m are the lengths. For same-length arrays, it's O(n²). For n=126, that's 16000 operations per correlate call. With 100000 iterations, that's 1.6 billion operations. That's too slow.

I need to use the delta formula for SA too. Let me restructure:

For SA, at each iteration:
1. Pick a random swap (i, j)
2. Compute the delta using the precomputed P and Q (or compute it directly)
3. Compute new_max = max(corrs + delta)
4. Accept or reject
5. If accepted, update corrs = corrs + delta (and update P, Q for the new h)

The issue is that P and Q change after each accepted swap. But we can update them incrementally.

Actually, let's just compute the delta directly for the specific swap:

```python
def compute_swap_delta(h, i, j, n, k, n_corr):
    """Compute the change in all overlaps when swapping h[i] (1->0) and h[j] (0->1)."""
    delta = np.zeros(n_corr)
    
    # P_i: removal of i
    mc = (i + k >= 0) & (i + k < n)
    delta[mc] -= (1.0 - h[i + k[mc]])
    mb = (i - k >= 0) & (i - k < n)
    delta[mb] += h[i - k[mb]]
    
    # Q_j: addition of j
    mc = (j + k >= 0) & (j + k < n)
    delta[mc] += (1.0 - h[j + k[mc]])
    mb = (j - k >= 0) & (j - k < n)
    delta[mb] -= h[j - k[mb]]
    
    # Correction at k = i-j
    k_spec = i - j
    if -(n-1) <= k_spec <= n-1:
        delta[n-1 + k_spec] += 2.0
    
    return delta
```

This is O(n) per swap evaluation. For SA with 100000 iterations and n=126, that's 12.6 million operations, which is fast.
