import numpy as np
from scipy.special import logsumexp
import time


def project_feasible(x, target):
    n = len(x)
    lo, hi = -3.0, 3.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        mask = (h > 1e-9) & (h < 1.0 - 1e-9)
        if mask.sum() > 0:
            h = h.copy()
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)


def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max()) * dx


def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def grad_corr_k(h, k, n):
    d = k - (n - 1)
    grad = np.zeros(n)
    if d >= 0:
        idx = np.arange(n - d)
        grad[idx] += (1.0 - h[idx + d])
        grad[idx + d] += (-h[idx])
    else:
        d2 = -d
        idx = np.arange(n - d2)
        grad[idx] += (-h[idx + d2])
        grad[idx + d2] += (1.0 - h[idx])
    return grad


def grad_smooth_max(h, dx, beta):
    n = len(h)
    corrs = compute_corrs(h)
    scaled = beta * corrs
    shifted = scaled - scaled.max()
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        return np.zeros(n), float(corrs.max()) * dx
    w /= w_sum
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] > 1e-12:
            grad += w[k] * grad_corr_k(h, k, n)
    return grad * dx, float(corrs.max()) * dx


def pgd_nesterov(h0, target, max_iter, beta, mu=0.9):
    n = len(h0)
    dx = 2.0 / n
    h = h0.copy().astype(float)
    h_prev = h.copy()
    best_h = h.copy()
    best_c5 = eval_c5(h)
    L = 4.0  # Lipschitz constant estimate
    t_k = 1.0
    no_improve = 0
    
    for it in range(max_iter):
        h_y = h + (t_k - 1.0) / (t_k + 1.0) * (h - h_prev)
        h_y = np.clip(h_y, 0.0, 1.0)
        grad, c5_curr = grad_smooth_max(h_y, dx, beta)
        
        step = 1.0 / L
        h_new = h_y - step * grad
        h_new = project_feasible(h_new, target)
        
        # Adaptive step
        f_new = eval_c5(h_new)
        tries = 0
        while f_new > c5_curr + 1e-12 and tries < 5:
            L *= 2.0
            step = 1.0 / L
            h_new = h_y - step * grad
            h_new = project_feasible(h_new, target)
            f_new = eval_c5(h_new)
            tries += 1
        
        h_prev = h
        h = h_new
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
        t_k = t_new
        
        if f_new < best_c5 - 1e-14:
            best_c5 = f_new
            best_h = h.copy()
            no_improve = 0
            L = max(1.0, L * 0.95)
        else:
            no_improve += 1
            if no_improve > 3000:
                break
    
    return best_h, best_c5


def coord_descent(h0, target, n_sweeps, grid_size=200, beta_active=50.0):
    n = len(h0)
    dx = 2.0 / n
    h = h0.copy().astype(float)
    best_h = h.copy()
    best_c5 = eval_c5(h)
    
    x_grid = np.linspace(0.0, 1.0, grid_size)
    order = np.random.RandomState(0).permutation(n)
    
    for sweep in range(n_sweeps):
        for j in order:
            corrs = compute_corrs(h)
            # Compute base and coef for each k
            d_range = np.arange(-(n - 1), n)
            coefs = np.zeros(2 * n - 1)
            bases = corrs.copy()
            for kk in range(2 * n - 1):
                d = d_range[kk]
                b = 0.0
                if 0 <= j + d < n:
                    b += (1.0 - h[j + d])
                    bases[kk] -= h[j] * (1.0 - h[j + d])
                if 0 <= j - d < n:
                    b += (-h[j - d])
                    bases[kk] += h[j - d] * (1.0 - h[j])
                coefs[kk] = b
            # Minimize max_k (bases[k] + coef[k]*x) over x in [0,1]
            vals = bases[None, :] + coefs[None, :] * x_grid[:, None]
            obj = vals.max(axis=1)
            j_best = np.argmin(obj)
            x_new = x_grid[j_best]
            if abs(x_new - h[j]) > 1e-10:
                h[j] = x_new
                h = project_feasible(h, target)
        
        c5_val = eval_c5(h)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
    
    return best_h, best_c5


def binary_local_search(h0_bin, max_iters, t_start=1.0, t_end=0.01, rng=None):
    n = len(h0_bin)
    h = h0_bin.copy().astype(float)
    if rng is None:
        rng = np.random.RandomState(0)
    best_h = h.copy()
    best_c5 = eval_c5(h)
    corrs = compute_corrs(h)
    best_corr = corrs.max()
    ones = np.where(h > 0.5)[0].tolist()
    zeros = np.where(h < 0.5)[0].tolist()
    
    it = 0
    while it < max_iters:
        t = t_start * (t_end / t_start) ** (it / max(float(max_iters), 1.0))
        i = rng.choice(len(ones))
        j = rng.choice(len(zeros))
        oi, oj = ones[i], zeros[j]
        
        # Delta evaluation: swap h[oi]=1->0, h[oj]=0->1
        delta = 0
        for kk in range(2 * n - 1):
            d = kk - (n - 1)
            contrib = 0.0
            # h[oi] changes from 1 to 0
            # Term where h[oi] is h[i]: corr loses 1*(1-h[oi+d])
            if 0 <= oi + d < n and oi != oj:
                contrib -= (1.0 - h[oi + d])
            elif 0 <= oi + d < n and oi + d == oj:
                # h[oi+d] is changing from 0 to 1, so this term changes
                # Original: 1*(1-0)=1, New: 0*(1-1)=0, change = -1
                contrib += 0.0  # already handled by the h[oj] change below
            # Term where h[oi] is h[i+d], i.e., i=oi-d:
            if 0 <= oi - d < n and oi - d != oj:
                contrib -= (1.0 - h[oi - d]) * 0  # h[i] stays, (1-h[i+d]) changes
            # Actually let me recompute more carefully
            # corr[kk] = sum_i h[i]*(1-h[i+d])
            # After swap: h_new[oi]=0, h_new[oj]=1
            # Change in corr[kk]:
            # For term i where h[i] changes: i=oi (was 1, now 0): loss = 1*(1-h_new[oi+d])
            # For term i where h[i+d] changes: i+d=oi, i=oi-d (was 1, now 0 in 1-h): gain = h[oi-d]*1
            # For term where h[i] changes: i=oj (was 0, now 1): gain = 1*(1-h_new[oj+d])
            # For term where h[i+d] changes: i+d=oj, i=oj-d (was 0, now 1 in 1-h): loss = h[oj-d]*1
            
            # Let me just compute delta directly
            delta_k = 0.0
            # i=oi: h goes 1->0, so term h[oi]*(1-h[oi+d]) goes from (1-h[oi+d]) to 0
            if 0 <= oi + d < n:
                if oi + d == oj:
                    delta_k += -(1.0 - 0.0) + 0  # was 1*(1-0)=1, now 0*(1-1)=0, delta=-1
                else:
                    delta_k += -(1.0 - h[oi + d])
            # i=oi-d (if valid): h[oi-d]*(1-h[oi]) goes from h[oi-d]*0 to h[oi-d]*1
            if 0 <= oi - d < n and oi - d != oj:
                delta_k += h[oi - d] * 1.0
            elif 0 <= oi - d < n and oi - d == oj:
                # h[oj] also changes, complex - skip or handle separately
                pass
            # i=oj: h goes 0->1, so term h[oj]*(1-h[oj+d]) goes from 0 to (1-h[oj+d])
            if 0 <= oj + d < n:
                if oj + d == oi:
                    delta_k += 0  # h[oj+oi]=h[oi] goes 1->0, (1-h[oi]) goes 0->1
                    # term was 0*(1-1)=0, now 1*(1-0)=1, delta=+1
                    delta_k += 1.0
                else:
                    delta_k += (1.0 - h[oj + d])
            # i=oj-d (if valid): h[oj-d]*(1-h[oj]) goes from h[oj-d]*1 to h[oj-d]*0
            if 0 <= oj - d < n and oj - d != oi:
                delta_k += -h[oj - d]
            elif 0 <= oj - d < n and oj - d == oi:
                pass
            
            # The cross terms when oi and oj are related
            if oi + d == oj:
                # Already handled above
                pass
            if oj + d == oi:
                # Already handled
                pass
            
            delta += delta_k
        
        new_corr_max_estimate = best_corr  # approximation
        if delta < 0 or rng.random() < np.exp(delta / max(t, 1e-10)):
            h[oi] = 0.0
            h[oj] = 1.0
            ones[i] = oj
            zeros[j] = oi
            # Update corrs exactly (expensive but correct)
            new_corrs = compute_corrs(h)
            new_max = new_corrs.max()
            if new_max < best_corr:
                best_h = h.copy()
                best_c5 = eval_c5(h)
            best_corr = new_max
            corrs = new_corrs
        it += 1
    
    return best_h, best_c5


def interpolate_to_n(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_src = np.linspace(0.0, 2.0, n_src + 1)
    x_dst = np.linspace(0.0, 2.0, n_dst + 1)
    h_ext = np.concatenate([h_src, [h_src[-1]]])
    h_new = np.interp(x_dst[:-1], x_src, h_ext)
    return h_new


def active_set_polish(h0, target, n_iters, tol=0.001):
    n = len(h0)
    dx = 2.0 / n
    h = h0.copy().astype(float)
    best_h = h.copy()
    best_c5 = eval_c5(h)
    
    for it in range(n_iters):
        corrs = compute_corrs(h)
        max_val = corrs.max()
        c5_val = max_val * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
        
        # Active set: correlations within tol of max
        active = np.where(corrs >= max_val - 2.0 / (n * tol))[0]
        n_active = len(active)
        if n_active == 0:
            active = np.array([np.argmax(corrs)])
            n_active = 1
        
        # Average gradient over active set
        grad = np.zeros(n)
        for k in active:
            grad += grad_corr_k(h, k, n)
        grad /= n_active
        grad *= dx
        
        # Gradient step with backtracking
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad_normed = grad / gnorm
        
        for alpha in [0.5, 0.25, 0.1, 0.05, 0.02, 0.01]:
            h_trial = project_feasible(h - alpha * grad, target)
            c5_trial = eval_c5(h_trial)
            if c5_trial < c5_val - 1e-14:
                h = h_trial
                break
        else:
            if c5_val < best_c5 - 1e-12:
                best_c5 = c5_val
                best_h = h.copy()
            break
    
    return best_h, best_c5


def make_binary_alternating(n, rng):
    h = np.zeros(n)
    pattern_len = rng.choice([3, 4, 5, 6, 7, 8])
    fill = pattern_len // 2
    for i in range(n):
        h[i] = 1.0 if (i % pattern_len) < fill else 0.0
    return project_feasible(h, n / 2.0)


def make_random_binary(n, rng):
    h = np.zeros(n)
    idx = rng.permutation(n)
    h[idx[:n // 2]] = 1.0
    return h


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    
    # Get parent
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    h0 = project_feasible(h0, n0 / 2.0)
    best_h = h0.copy()
    best_c5 = eval_c5(h0)
    
    def remaining():
        return budget_s - (time.time() - t0)
    
    # Phase 1: Polish parent with Nesterov PGD
    for beta in [100, 300, 800, 2000, 5000]:
        if remaining() < 30:
            break
        h_new, c5_new = pgd_nesterov(best_h, n0 / 2.0, 3000, beta)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new
    
    # Phase 2: Active set polish on parent
    if remaining() > 20:
        h_new, c5_new = active_set_polish(best_h, n0 / 2.0, 500)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new
    
    # Phase 3: Try nearby n values
    n_targets = [128, 120, 130, 132, 136, 140, 118, 122, 124, 126]
    for n_try in n_targets:
        if remaining() < 40:
            break
        if n_try == n0:
            continue
        h_interp = interpolate_to_n(best_h, n_try)
        h_interp = project_feasible(h_interp, n_try / 2.0)
        
        # Polish with PGD
        h_new, c5_new = pgd_nesterov(h_interp, n_try / 2.0, 2000, 500)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new
        
        # Active set polish
        if remaining() > 20:
            h_new2, c5_new2 = active_set_polish(h_new, n_try / 2.0, 300)
            if c5_new2 < best_c5:
                best_c5 = c5_new2
                best_h = h_new2
        
        # Coordinate descent polish
        if remaining() > 15 and c5_new < best_c5 + 0.01:
            h_new3, c5_new3 = coord_descent(h_new, n_try / 2.0, 3)
            if c5_new3 < best_c5:
                best_c5 = c5_new3
                best_h = h_new3
    
    # Phase 4: Binary local search at promising n values
    for n_try in [128, 126, 120, 130]:
        if remaining() < 30:
            break
        # Start from rounded best
        h_bin = (best_h > 0.5).astype(float)
        if n_try != len(best_h):
            h_interp = interpolate_to_n(best_h, n_try)
            h_bin = (h_interp > 0.5).astype(float)
            # Fix to exactly n/2 ones
            n_ones = int(h_bin.sum())
            target_ones = n_try // 2
            if n_ones > target_ones:
                ones_idx = np.where(h_bin == 1.0)[0]
                # Remove some ones
                to_remove = n_ones - target_ones
                perm = rng.permutation(len(ones_idx))
                for ii in range(to_remove):
                    h_bin[ones_idx[perm[ii]]] = 0.0
            elif n_ones < target_ones:
                zeros_idx = np.where(h_bin == 0.0)[0]
                to_add = target_ones - n_ones
                perm = rng.permutation(len(zeros_idx))
                for ii in range(to_add):
                    h_bin[zeros_idx[perm[ii]]] = 1.0
        
        c5_bin = eval_c5(h_bin)
        if c5_bin < 0.42:  # Only worth polishing if reasonably good
            # Lift to continuous and polish
            h_cont = h_bin.astype(float)
            h_cont = project_feasible(h_cont, n_try / 2.0)
            h_new, c5_new = pgd_nesterov(h_cont, n_try / 2.0, 2000, 300)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new
            if remaining() > 15:
                h_new2, c5_new2 = active_set_polish(h_new, n_try / 2.0, 300)
                if c5_new2 < best_c5:
                    best_c5 = c5_new2
                    best_h = h_new2
    
    # Phase 5: Binary search at multiple n values with different starts
    for n_try in [128, 96, 80, 64, 100, 112]:
        if remaining() < 25:
            break
        # Multiple random binary starts
        for trial in range(3):
            if remaining() < 15:
                break
            if trial == 0:
                h_bin = make_random_binary(n_try, rng)
            elif trial == 1:
                h_bin = make_binary_alternating(n_try, rng)
            else:
                h_bin = (np.linspace(0, 1, n_try) > 0.5).astype(float)
                h_bin = project_feasible(h_bin, n_try / 2.0)
                # Add random perturbation
                h_bin = (h_bin + 0.3 * (rng.random(n_try) - 0.5) > 0.5).astype(float)
                # Fix sum
                n_ones = int(h_bin.sum())
                if n_ones != n_try // 2:
                    h_bin = make_random_binary(n_try, rng)
            
            c5_bin = eval_c5(h_bin)
            if c5_bin < 0.40:
                # Do a quick binary local search
                h_bs, c5_bs = binary_local_search(h_bin, min(5000, int(remaining() * 5)), 1.0, 0.01, rng)
                if c5_bs < best_c5:
                    best_c5 = c5_bs
                    best_h = h_bs
                # Lift to continuous
                h_cont = project_feasible(h_bs.astype(float), n_try / 2.0)
                h_new, c5_new = pgd_nesterov(h_cont, n_try / 2.0, 1000, 500)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new
    
    # Phase 6: Final intensive polish on best solution
    if remaining() > 20 and best_h is not None:
        n_b = len(best_h)
        for beta in [2000, 5000, 10000]:
            if remaining() < 10:
                break
            h_new, c5_new = pgd_nesterov(best_h, n_b / 2.0, 2000, beta)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new
        
        if remaining() > 10:
            h_new, c5_new = active_set_polish(best_h, n_b / 2.0, 400)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new
        
        if remaining() > 8:
            h_new, c5_new = coord_descent(best_h, n_b / 2.0, 2, 100)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new
    
    # Final validation
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final / 2.0)
    final_c5 = eval_c5(best_h)
    
    return best_h, final_c5, n_final
