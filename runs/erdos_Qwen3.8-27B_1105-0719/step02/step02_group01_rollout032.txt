def binary_anneal(n, rng, n_iters, t_init=0.04, t_floor=1e-6, cooling=0.9993):

    dx = 2.0 / n

    cur_c5 = c5_raw(h)

    best_c5 = cur_c5

    ones = list(np.where(h == 1.0)[0])
    zeros = list(np.where(h == 0.0)[0])

    T = t_init
    for it in range(n_iters):
        pi = rng.randint(len(ones))
        pj = rng.randint(len(zeros))
        oi = ones[pi]
        oz = zeros[pj]

        h[oi] = 0.0
        h[oz] = 1.0

            zeros[pj] = oi

            h[oz] = 0.0

    zeros = list(np.where(h == 0.0)[0])

    for rnd in range(max_rounds):
        corrs = compute_corrs(h)
        max_k = corrs.argmax()
        offset = max_k - (n - 1)

        best_delta = 0
        best_swap = None
        tried = 0
        for oi in ones:
            if tried > 80:
                break
            for _ in range(3):
                pj = rng.randint(len(zeros))
                oz = zeros[pj]
                if oz == oi:
                    continue
                h[oi] = 0.0
                h[oz] = 1.0
                new_c5 = c5_raw(h)
                h[oi] = 1.0
                h[oz] = 0.0
                delta = cur_c5 - new_c5
                if delta > best_delta:
                    best_delta = delta
                    best_swap = (oi, oz)
                tried += 1

        if best_swap is not None and best_delta > 1e-15:
            oi, oz = best_swap
            h[oi] = 0.0
            h[oz] = 1.0
            cur_c5 += -best_delta
            ones[ones.index(oi)] = oz
            zeros[zeros.index(oz)] = oi
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            break

    return best_h, best_c5


def make_periodic_binary(n, period, fill, rng):
    h = np.zeros(n)
    for i in range(n):
        h[i] = 1.0 if (i % period) < fill else 0.0
    target = n // 2
    cur_sum = int(h.sum())
    if cur_sum > target:
        ones = list(np.where(h == 1.0)[0])
        rng.shuffle(ones)
        for i in range(cur_sum - target):
            h[ones[i]] = 0.0
    elif cur_sum < target:
        zeros = list(np.where(h == 0.0)[0])
        rng.shuffle(zeros)
        for i in range(target - cur_sum):
            h[zeros[i]] = 1.0
    return h


def line_search_polish(h0, dx, target_sum, max_steps=500):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_raw(h)
    step_size = 0.01

    for it in range(max_steps):
        grad = smooth_gradient(h, 500.0, dx)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        grad_unit = grad / gnorm
        h_new = h - step_size * grad_unit
        h_new = project_h(h_new, target_sum)
        new_c5 = c5_raw(h_new)

        if new_c5 < best_c5 - 1e-15:
            best_c5 = new_c5
            best_h = h_new.copy()
            h = h_new
            step_size = min(step_size * 1.2, 0.1)
        else:
            step_size *= 0.7
            if step_size < 1e-8:
                break

    return best_h, best_c5


def lp_polish(h0, dx, target_sum, n_steps=40, lam=0.3):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_raw(h)

    for s in range(n_steps):
        corrs = compute_corrs(h)
        c5_val = corrs.max() * dx
        n_active = max(8, int(0.25 * (2 * n - 1)))
        top_idx = np.argsort(corrs)[-n_active:]

        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        A_rows = []
        b_rows = []
        for k in top_idx:
            row = np.zeros(n + 1)
            row[:n] = dx * grad_corr_k(h, k, n)
            row[n] = -1.0
            A_rows.append(row)
            b_rows.append(-corrs[k] * dx - 1e-13)

        A_ub = np.array(A_rows)
        b_ub = np.array(b_rows)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        bounds = [(-h[i] * lam, (1.0 - h[i]) * lam) for i in range(n)]
        bounds.append((0.0, None))

        from scipy.optimize import linprog
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.zeros(1),
                          bounds=bounds, method='highs')
            if res.status != 0:
                break
            delta = res.x[:n]
            h_new = project_h(h + delta, target_sum)
            new_c5 = c5_raw(h_new)
            if new_c5 < c5_val - 1e-14:
                h = h_new
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h_new.copy()
            else:
                lam *= 0.5
        except Exception:
            break

    return best_h, best_c5


def remaining_time(t0, budget_s):
    return budget_s - (time.time() - t0)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    dx0 = 2.0 / n0
    target0 = n0 / 2.0
    h0 = project_h(h0, target0)

    best_h = h0.copy()
    best_c5 = c5_raw(best_h)

    # Phase 1: SLSQP polishing of initial with increasing beta
    for beta in [100, 200, 400, 800, 1600, 3000, 5000]:
        if remaining_time(t0, budget_s) < 15:
            break
        h_opt, c5_val = slsqp_optimize(best_h, beta, 1500, dx0, target0)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_opt

    # Z-space SLSQP
    if remaining_time(t0, budget_s) > 50:
        z0 = h_to_z(best_h)
        for beta in [500, 1000, 2000, 4000]:
            if remaining_time(t0, budget_s) < 20:
                break
            h_opt, c5_val = slsqp_z_optimize(z0, beta, 1500, dx0, target0)
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h_opt
                z0 = h_to_z(best_h)

    # LP polish
    if remaining_time(t0, budget_s) > 30:
        h_lp, c5_lp = lp_polish(best_h, dx0, target0, 40, 0.25)
        if c5_lp < best_c5:
            best_c5 = c5_lp
            best_h = h_lp

    # Subgradient
    if remaining_time(t0, budget_s) > 80:
        h_sg, c5_sg = subgradient_optimize(best_h, 15000, dx0, target0)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg

    # Line search polish
    if remaining_time(t0, budget_s) > 30:
        h_ls, c5_ls = line_search_polish(best_h, dx0, target0, 400)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls

    # Final SLSQP
    if remaining_time(t0, budget_s) > 20:
        h_opt, c5_val = slsqp_optimize(best_h, 6000, 2000, dx0, target0)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_opt
        if remaining_time(t0, budget_s) > 15:
            h_lp, c5_lp = lp_polish(best_h, dx0, target0, 30, 0.15)
            if c5_lp < best_c5:
                best_c5 = c5_lp
                best_h = h_lp

    # Phase 2: Binary annealing to find better starts at various n
    n_candidates = [64, 80, 96, 112, 126, 128, 144, 160]
    binary_results = []

    for n_try in n_candidates:
        if remaining_time(t0, budget_s) < 40:
            break
        dx_try = 2.0 / n_try
        target_try = n_try // 2.0
        n_restarts = 3
        best_bin_h = None
        best_bin_c5 = float('inf')

        for r in range(n_restarts):
            if remaining_time(t0, budget_s) < 15:
                break
            if r == 0:
                h_init = make_periodic_binary(n_try, 8, 4, rng)
            elif r == 1:
                h_init = make_periodic_binary(n_try, 6, 3, rng)
            else:
                h_init = np.zeros(n_try)
                idx = rng.permutation(n_try)
                h_init[idx[:target_try]] = 1.0

            h_ann, c5_ann = binary_anneal(n_try, rng, 80000, t_init=0.06, cooling=0.9994)
            h_hc, c5_hc = binary_hill_climb(h_ann, n_try, 200, rng)
            if c5_hc < best_bin_c5:
                best_bin_c5 = c5_hc
                best_bin_h = h_hc

        if best_bin_h is not None:
            binary_results.append((best_bin_h, best_bin_c5, n_try))
            if best_bin_c5 < best_c5:
                best_c5 = best_bin_c5
                best_h = best_bin_h.copy()

    # Phase 3: Polish best binary results in continuous space
    for (bin_h, bin_c5, n_b) in binary_results:
        if remaining_time(t0, budget_s) < 30:
            break
        dx_b = 2.0 / n_b
        target_b = n_b / 2.0
        h_cont = bin_h.astype(float)

        h_opt, c5_val = slsqp_optimize(h_cont, 300, 1000, dx_b, target_b)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_opt
            if remaining_time(t0, budget_s) > 20:
                h_opt2, c5_val2 = slsqp_optimize(best_h, 1000, 1000, dx_b, target_b)
                if c5_val2 < best_c5:
                    best_c5 = c5_val2
                    best_h = h_opt2

    # Phase 4: Try interpolating best solution to other grid sizes
    if remaining_time(t0, budget_s) > 40:
        n_best = len(best_h)
        for n_try in [n_best - 2, n_best + 2, n_best + 4, n_best - 4]:
            if n_try < 40 or remaining_time(t0, budget_s) < 25:
                continue
            dx_try = 2.0 / n_try
            target_try = n_try / 2.0
            x_orig = np.linspace(0.0, 2.0, n_best, endpoint=False) + 1.0 / n_best
            x_new = np.linspace(0.0, 2.0, n_try, endpoint=False) + 1.0 / n_try
            h_interp = np.interp(x_new, x_orig, np.concatenate([best_h, best_h[:1]]))
            h_interp = project_h(h_interp, target_try)

            h_opt, c5_val = slsqp_optimize(h_interp, 500, 1000, dx_try, target_try)
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h_opt

    # Phase 5: Final intensive polish
    if remaining_time(t0, budget_s) > 20 and best_h is not None:
        n_f = len(best_h)
        dx_f = 2.0 / n_f
        target_f = n_f / 2.0

        h_opt, c5_val = slsqp_optimize(best_h, 8000, 2000, dx_f, target_f)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_opt

        if remaining_time(t0, budget_s) > 15:
            z_f = h_to_z(best_h)
            h_z, c5_z = slsqp_z_optimize(z_f, 4000, 1500, dx_f, target_f)
            if c5_z < best_c5:
                best_c5 = c5_z
                best_h = h_z

        if remaining_time(t0, budget_s) > 12:
            h_ls, c5_ls = line_search_polish(best_h, dx_f, target_f, 300)
            if c5_ls < best_c5:
                best_c5 = c5_ls
                best_h = h_ls

        if remaining_time(t0, budget_s) > 10:
            h_lp, c5_lp = lp_polish(best_h, dx_f, target_f, 30, 0.1)
            if c5_lp < best_c5:
                best_c5 = c5_lp
                best_h = h_lp

        if remaining_time(t0, budget_s) > 8:
            h_sg, c5_sg = subgradient_optimize(best_h, 5000, dx_f, target_f)
            if c5_sg < best_c5:
                best_c5 = c5_sg
                best_h = h_sg

        if remaining_time(t0, budget_s) > 5:
            h_final, c5_final = slsqp_optimize(best_h, 10000, 1000, dx_f, target_f)
            if c5_final < best_c5:
                best_c5 = c5_final
                best_h = h_final

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    n_final = len(best_h)
    target_final = n_final / 2.0
    best_h = project_h(best_h, target_final)
    final_c5 = c5_raw(best_h)

    return best_h, final_c5, n_final
