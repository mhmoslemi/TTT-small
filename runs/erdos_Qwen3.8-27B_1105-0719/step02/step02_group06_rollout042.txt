from scipy.optimize import minimize, fsolve
import time

def eval_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def corr_raw(h):
    return np.correlate(h, 1.0 - h, mode='full')

def build_sym_h(q):
    m = len(q)
    h = np.empty(2 * m)
    h[:m] = q
    h[m:] = 1.0 - q[::-1]
    return h

def sym_from_h(h):
    n = len(h)
    m = n // 2
    q = np.empty(m)
    for i in range(m):
        q[i] = 0.5 * (h[i] + (1.0 - h[n - 1 - i]))
    return np.clip(q, 0.0, 1.0)

def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.5, 1.5
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(x + mid, 0.0, 1.0).sum()
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    h = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        interior = (h > 1e-12) & (h < 1.0 - 1e-12)
        if interior.sum() > 0:
            h[interior] += diff / interior.sum()
    return np.clip(h, 0.0, 1.0)

def corr_at_k(h, k):
    n = len(h)
    d = k - (n - 1)
    i_lo = max(0, -d)
    i_hi = min(n - 1, n - 1 - d)
    if i_lo > i_hi:
        return 0.0
    return float(np.sum(h[i_lo:i_hi + 1] * (1.0 - h[i_lo + d:i_hi + 1 + d])))

def corr_grad_k(h, k):
    n = len(h)
    d = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -d)
    i_hi = min(n - 1, n - 1 - d)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx] += (1.0 - h[idx + d])
    grad[idx + d] += (-h[idx])
    return grad

def smooth_max_obj_grad_sym(x, n_full, beta):

def select_active(corrs_scaled, margin_frac=0.02, margin_abs=1e-6, neighbor_pad=3, max_k=80):

    max_val = corrs_scaled.max()

def transfer_to_n(h_old, n_new):
    n_old = len(h_old)
    x_old = (np.arange(n_old) + 0.5) * 2.0 / n_old
    x_new = (np.arange(n_new) + 0.5) * 2.0 / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new / 2.0)

def binary_push(h, threshold=0.5):
    """Push values toward 0 or 1"""
    n = len(h)
    h_bin = (h > threshold).astype(float)
    h_bin = project_feasible(h_bin, n / 2.0)
    return h_bin

def partial_binary_push(h, frac=0.3):
    """Push the top frac fraction toward 0 and bottom frac toward 1"""
    n = len(h)
    h_copy = h.copy()
    n_push = int(n * frac / 2)
    # Push smallest toward 0
    idx_smallest = np.argsort(h_copy)[:n_push]
    h_copy[idx_smallest] = 0.0
    # Push largest toward 1
    idx_largest = np.argsort(h_copy)[-n_push:]
    h_copy[idx_largest] = 1.0
    return project_feasible(h_copy, n / 2.0)

import numpy as np
from scipy.optimize import minimize
import time

def eval_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def corr_raw(h):
    return np.correlate(h, 1.0 - h, mode='full')

def build_sym_h(q):
    m = len(q)
    h = np.empty(2 * m)
    h[:m] = q
    h[m:] = 1.0 - q[::-1]
    return h

def sym_from_h(h):
    n = len(h)
    m = n // 2
    q = np.empty(m)
    for i in range(m):
        q[i] = 0.5 * (h[i] + (1.0 - h[n - 1 - i]))
    return np.clip(q, 0.0, 1.0)

def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.5, 1.5
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(x + mid, 0.0, 1.0).sum()
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    h = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        interior = (h > 1e-12) & (h < 1.0 - 1e-12)
        if interior.sum() > 0:
            h[interior] += diff / interior.sum()
    return np.clip(h, 0.0, 1.0)

def corr_at_k(h, k):
    n = len(h)
    d = k - (n - 1)
    i_lo = max(0, -d)
    i_hi = min(n - 1, n - 1 - d)
    if i_lo > i_hi:
        return 0.0
    return float(np.sum(h[i_lo:i_hi + 1] * (1.0 - h[i_lo + d:i_hi + 1 + d])))

def corr_grad_k(h, k):
    n = len(h)
    d = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -d)
    i_hi = min(n - 1, n - 1 - d)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx] += (1.0 - h[idx + d])
    grad[idx + d] += (-h[idx])
    return grad

class SmoothMaxSym:
    def __init__(self, n_full, beta):
        self.n = n_full
        self.beta = beta
        self.dx = 2.0 / n_full
    def fun(self, x):
        h = build_sym_h(x)
        corrs = corr_raw(h) * self.dx
        mc = corrs.max()
        sh = corrs - mc
        w = np.exp(np.clip(self.beta * sh, -500, 0))
        ws = w.sum()
        if ws < 1e-30:
            return mc
        return mc + np.log(ws) / self.beta
    def jac(self, x):
        h = build_sym_h(x)
        corrs = corr_raw(h) * self.dx
        mc = corrs.max()
        sh = corrs - mc
        w = np.exp(np.clip(self.beta * sh, -500, 0))
        ws = w.sum()
        if ws < 1e-30:
            w = np.ones(len(corrs)) / len(corrs)
        else:
            w /= ws
        m = self.n // 2
        grad_q = np.zeros(m)
        for k in range(2 * self.n - 1):
            if w[k] < 1e-14:
                continue
            g = corr_grad_k(h, k)
            for r in range(m):
                grad_q[r] += w[k] * (g[r] - g[self.n - 1 - r])
        grad_q *= self.dx
        return grad_q

def smooth_sym_optimize(q0, n_full, beta_list, maxiter_per_beta, time_limit):
    m = n_full // 2
    q = np.clip(q0.copy(), 0, 1)
    h = build_sym_h(q)
    best_score = eval_score(h)
    best_q = q.copy()
    for beta in beta_list:
        if time.time() > time_limit:
            break
        sm = SmoothMaxSym(n_full, beta)
        bounds = [(0.0, 1.0)] * m
        try:
            res = minimize(sm.fun, q, jac=sm.jac, method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxiter_per_beta, 'ftol': 1e-16, 'gtol': 1e-13, 'maxls': 50})
            q_new = np.clip(res.x, 0, 1)
        except Exception:
            q_new = q.copy()
        h_new = build_sym_h(q_new)
        sc = eval_score(h_new)
        if sc < best_score:
            best_score = sc
            best_q = q_new.copy()
        q = q_new
    return best_q, best_score, build_sym_h(best_q)

def solve_epigraph_sym(q_center, n_full, active_k, maxiter, ftol):

    h_out = build_sym_h(q_out)
    return q_out, h_out, eval_score(h_out)

def select_active(corrs_scaled, margin_frac=0.02, margin_abs=1e-6, neighbor_pad=3, max_k=70):
    n = len(corrs_scaled)
    max_val = corrs_scaled.max()
    threshold = max_val - max(margin_abs, margin_frac * max_val)
    active = set(np.where(corrs_scaled >= threshold)[0].tolist())
    padded = set()
    for k in active:
        for d in range(-neighbor_pad, neighbor_pad + 1):
            if 0 <= k + d < n:
                padded.add(k + d)
    active_list = sorted(padded)
    if len(active_list) > max_k:
        scored = sorted([(corrs_scaled[k], k) for k in active_list], reverse=True)
        active_list = sorted([k for _, k in scored[:max_k]])
    return active_list

def transfer_to_n(h_old, n_new):
    n_old = len(h_old)
    x_old = (np.arange(n_old) + 0.5) * 2.0 / n_old
    x_new = (np.arange(n_new) + 0.5) * 2.0 / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new / 2.0)

def coord_decent_sym(q, n_full, n_sweeps=15, n_iter_per_coord=50, rng=None):
    m = n_full // 2
    dx = 2.0 / n_full
    q = q.copy()
    order = list(range(m))
    for sweep in range(n_sweeps):
        if rng is not None:
            rng.shuffle(order)
        improved = False
        for j in order:
            h = build_sym_h(q)
            corrs = corr_raw(h) * dx
            jp = j
            jm = n_full - 1 - j
            B = np.zeros(2 * n_full - 1)
            for k in range(2 * n_full - 1):
                d = k - (n_full - 1)
                hjp_d = h[jp + d] if 0 <= jp + d < n_full else 0.0
                hjm_d = h[jp - d] if 0 <= jp - d < n_full else 0.0
                hjm_pd = h[jm + d] if 0 <= jm + d < n_full else 0.0
                hjm_md = h[jm - d] if 0 <= jm - d < n_full else 0.0
                B[k] = dx * ((1.0 - hjp_d) - hjm_d - (1.0 - hjm_pd) + hjm_md)
            A = corrs - q[j] * B
            xlo, xhi = 0.0, 1.0
            best_x = q[j]
            best_val = (A + best_x * B).max()
            for _ in range(n_iter_per_coord):
                mid = (xlo + xhi) / 2.0
                vals = A + mid * B
                max_val = vals.max()
                if max_val < best_val - 1e-15:
                    best_val = max_val
                    best_x = mid
                active = np.where(vals >= max_val - 1e-14)[0]
                B_act = B[active]
                if len(B_act) == 0:
                    break
                if np.all(B_act > 1e-15):
                    xhi = mid
                elif np.all(B_act < -1e-15):
                    xlo = mid
                else:
                    best_x = mid
                    best_val = max_val
                    break
            new_x = np.clip(best_x, 0.0, 1.0)
            if abs(new_x - q[j]) > 1e-12:
                q[j] = new_x
                improved = True
        if not improved:
            break
    h = build_sym_h(q)
    return q, h, eval_score(h)
