import numpy as np
from scipy.optimize import linprog
import time


def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
        else:
            h += diff / n
    return np.clip(h, 0.0, 1.0)


def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def c5_of_h(h, dx):
    return compute_corrs(h).max() * dx


def get_offsets(n):
    return np.arange(-(n - 1), n)


def coord_b(k, o, j, n, h):
    # B_k = dc_k/dh_j
    # Term 1: 1_{valid_k(j)} * (1 - h[j+o])
    # valid_k(j): 0 <= j+o < n
    # Term 2: -h[j-o] * 1_{valid_k(j-o)}
    # valid_k(j-o): 0 <= j-o < n and 0 <= (j-o)+o < n => 0 <= j-o < n (since 0<=j<n always)
    v1 = (0 <= j + o) and (j + o < n)
    v2 = (0 <= j - o) and (j - o < n)
    b = 0.0
    if v1:
        b += 1.0 - h[int(j + o)]
    if v2:
        b -= h[int(j - o)]
    return b


def coord_b_1d(k, o, j, n, h, c_k):
    """Return (A, B) such that c_k_new = A + B * t where t is new h[j]."""
    b = coord_b(k, o, j, n, h)
    a = c_k - b * h[j]
    return a, b


def solve_1d_minmax(A, B):
    """Find t in [0,1] minimizing max_k(A[k] + B[k]*t). Returns (t_opt, f_min)."""
    f0 = A.max()
    f1 = (A + B).max()
    
    B_pos = B > 1e-14
    B_neg = B < -1e-14
    
    if not B_neg.any():
        return 1.0, f1
    if not B_pos.any():
        return 0.0, f0
    
    # Interior minimum: binary search on v
    hi = max(f0, f1)
    # Lower bound: dual
    y = np.zeros(len(A))
    y[B_pos] = 1.0 / B_pos.sum()
    y[B_neg] = -1.0 / (-B_neg).sum()
    y /= y.sum()
    lb = y @ A + min(0.0, y @ B)
    lo = max(lb - 1e-10, min(A.min(), (A + B).min()) - 1e-10)
    
    for _ in range(60):
        mid = (lo + hi) / 2
        # Check feasibility: exists t in [0,1] with A_k + B_k*t <= mid for all k
        # B_k > 0: t <= (mid - A_k)/B_k
        # B_k < 0: t >= (mid - A_k)/B_k  [negative B, so (mid-A_k)/B_k is the lower bound]
        # B_k = 0: need mid >= A_k
        B_zero = ~(B_pos | B_neg)
        if B_zero.any() and A[B_zero].max() > mid:
            lo = mid
            continue
        tlo = 0.0
        if B_neg.any():
            tlo = max(tlo, ((mid - A[B_neg]) / B[B_neg]).max())
        thi = 1.0
        if B_pos.any():
            thi = min(thi, ((mid - A[B_pos]) / B[B_pos]).min())
        if tlo <= thi + 1e-15:
            hi = mid
        else:
            lo = mid
    
    v_star = hi
    # Find t

    thi = 1.0

        thi = min(thi, ((v_star - A[B_pos]) / B[B_pos]).min())
    t_opt = (max(0.0, tlo) + min(1.0, thi)) / 2.0
    t_opt = np.clip(t_opt, 0.0, 1.0)
    return t_opt, v_star


def coord_descent_sweep(h, corrs, offsets, n):
    """One full sweep of exact coordinate descent. Returns new h, new corrs, improved flag."""
    improved = False
    h_new = h.copy()
    for j in range(n):
        A = np.empty(2 * n - 1)
        B = np.empty(2 * n - 1)
        for k in range(2 * n - 1):
            o = offsets[k]
            A[k], B[k] = coord_b_1d(k, o, j, n, h_new, corrs[k])
        t_opt, f_val = solve_1d_minmax(A, B)
        if abs(t_opt - h_new[j]) > 1e-12:
            h_new[j] = t_opt
            improved = True
    # Recompute correlations
    new_corrs = compute_corrs(h_new)
    return h_new, new_corrs, improved


def block_coord_descent(h, corrs, offsets, n, n_inner=20):
    """Block coordinate descent: optimize each coordinate n_inner times."""
    for _ in range(n_inner):
        h, corrs, _ = coord_descent_sweep(h, corrs, offsets, n)
    return h, corrs


def binary_steepest_descent(h0, dx, max_sweeps=30, time_limit=None, t0=None):
    """Greedy binary swap: try all (i,j) swaps, apply best."""
    n = len(h0)
    target_sum = int(round(n / 2))
    offsets = get_offsets(n)
    
    h = np.zeros(n)
    sorted_idx = np.argsort(-h0)
    h[sorted_idx[:target_sum]] = 1.0
    
    corrs = compute_corrs(h)
    best_c5 = corrs.max() * dx
    best_h = h.copy()
    
    for sweep in range(max_sweeps):
        if time_limit is not None and t0 is not None:
            if time.time() - t0 > time_limit:
                break
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        best_swap = None
        best_new_max = corrs.max()
        
        for i in ones:
            for j in zeros:
                # Compute delta in correlations
                delta = np.zeros(2 * n - 1)
                for k in range(2 * n - 1):
                    o = offsets[k]
                    vi = (0 <= i + o) and (i + o < n)
                    vi2 = (0 <= i - o) and (i - o < n)
                    vj = (0 <= j + o) and (j + o < n)
                    vj2 = (0 <= j - o) and (j - o < n)
                    # Removing h[i]=1:
                    if vi:
                        delta[k] += -(1.0 - h[i + o])
                    if vi2:
                        delta[k] += h[i - o]
                    # Adding h[j]=1:
                    if vj:
                        delta[k] += (1.0 - h[j + o])
                    if vj2:
                        delta[k] += -h[j - o]
                new_max = (corrs + delta).max()
                if new_max < best_new_max - 1e-14:
                    best_new_max = new_max
                    best_swap = (i, j)
        
        if best_swap is None:
            break
        i, j = best_swap
        h[i], h[j] = 0.0, 1.0
        corrs = compute_corrs(h)
        new_c5 = corrs.max() * dx
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
    
    return best_h, best_c5


def smooth_max_obj(h, dx, beta):
    corrs = compute_corrs(h)
    m = corrs.max()
    return (m + np.log(np.sum(np.exp(beta * (corrs - m)))) / beta) * dx


def smooth_max_grad(h, dx, beta, offsets, n):
    corrs = compute_corrs(h)
    m = corrs.max()
    w = np.exp(beta * (corrs - m))
    w /= w.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        o = offsets[k]
        for j in range(n):
            vi = (0 <= j + o) and (j + o < n)
            vi2 = (0 <= j - o) and (j - o < n)
            g = 0.0
            if vi:
                g += 1.0 - h[j + o]
            if vi2:
                g -= h[j - o]
            grad[j] += w[k] * g
    return grad * dx


def local_slsqp(h0, dx, target_sum, beta, maxiter, n):
    from scipy.optimize import minimize
    offsets = get_offsets(n)
    
    def obj(x):
        return smooth_max_obj(x, dx, beta)
    
    def obj_grad(x):
        return smooth_max_grad(x, dx, beta, offsets, n)
    
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    
    try:
        res = minimize(obj, h0, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        h_out = project_feasible(res.x, target_sum)
    except Exception:
        h_out = project_feasible(h0, target_sum)
    return h_out, c5_of_h(h_out, dx)


def lp_direction(h, corrs, dx, target_sum, n):
    """Solve LP to find direction that reduces max correlation."""
    offsets = get_offsets(n)
    K = 2 * n - 1
    current_max = corrs.max()
    
    # LP: max d
    # s.t. G @ d <= current_max - corrs (all K constraints)
    #      sum(d) == 0
    #      d[j] >= 0 if h[j]==0, d[j] <= 0 if h[j]==1, d[j] >= -h[j] if h[j] in (0,1)
    #      d[j] <= 1-h[j] if h[j] in (0,1)
    
    # G[j,k] = B_k for coordinate j
    G = np.zeros((n, K))
    for j in range(n):
        for k in range(K):
            G[j, k] = coord_b(k, offsets[k], j, n, h)
    
    # Objective: max d = -min(-d), c = -e_n
    c_obj = -np.ones(n)
    
    # Inequality: G @ d <= current_max - corrs
    A_ub = G.astype(float)
    b_ub = (current_max - corrs).astype(float)
    
    # Equality: sum(d) = 0
    A_eq = np.ones((1, n))
    b_eq = np.zeros(1)
    
    # Bounds
    lb = np.zeros(n)
    ub = np.zeros(n)
    for j in range(n):
        if h[j] < 1e-10:
            lb[j], ub[j] = 0.0, np.inf
        elif h[j] > 1.0 - 1e-10:
            lb[j], ub[j] = -np.inf, 0.0
        else:
            lb[j], ub[j] = -h[j], 1.0 - h[j]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq.ravel(), b_eq=b_eq,
                      bounds=list(zip(lb, ub)), method='highs',
                      options={'presolve': True})
        if res.success and res.x is not None:
            return -res.fun, res.x  # (-(-d)) = d value, direction
    except Exception:
        pass
    return None, None


def line_search_from_lp(h, d, corrs, dx, n, target_sum, max_trials=20):

def subgradient_step(h, corrs, dx, n, target_sum, step_size, offsets):
    """Take a subgradient step of the max function."""
    K = 2 * n - 1
    max_idx = np.where(corrs == corrs.max() - 1e-14)[0]
    # Average gradient over active set
    grad_sum = np.zeros(n)
    for k in max_idx:
        o = offsets[k]
        for j in range(n):
            vi = (0 <= j + o) and (j + o < n)
            vi2 = (0 <= j - o) and (j - o < n)
            g = 0.0
            if vi:
                g += 1.0 - h[j + o]
            if vi2:
                g -= h[j - o]
            grad_sum[j] += g
    grad_sum /= len(max_idx)
    
    # Projected subgradient step
    h_new = h - step_size * grad_sum
    h_new = project_feasible(h_new, target_sum)
    return h_new


def make_structured_start(n, target_sum, rng, mode=0):
    h = np.zeros(n)
    if mode == 0:
        phi = (1.0 + np.sqrt(5.0)) / 2.0
        positions = set()
        for k in range(int(target_sum)):
            pos = int((k * phi * n) % n)
            positions.add(pos)
        if len(positions) < int(target_sum):
            # Fill remaining randomly
            all_pos = list(range(n))
            for p in all_pos:
                if len(positions) >= int(target_sum):
                    break
                if p not in positions:
                    positions.add(p)
        for p in positions:
            h[p] = 1.0
    elif mode == 1:
        # Low-discrepancy: use Van der Corput / radical inverse
        for k in range(int(target_sum)):
            val = 0.0
            denom = 1.0
            kk = k + 1
            while kk > 0:
                denom *= 2.0
                val += (kk % 2) / denom
                kk //= 2
            pos = int(val * n) % n
            h[pos] = 1.0
        # Handle collisions
        if h.sum() < int(target_sum):
            missing = int(target_sum) - int(h.sum())
            zero_pos = np.where(h == 0)[0]
            if len(zero_pos) >= missing:
                h[zero_pos[:missing]] = 1.0
    else:
        idx = rng.permutation(n)
        h[idx[:int(target_sum)]] = 1.0
    return project_feasible(h, target_sum)

                break
    
    # Phase 2: LP direction + line search
    if remaining_time(t0, budget_s) > 60:
        for it in range(50):
            if remaining_time(t0, budget_s) < 20:
                break
            d_val, direction = lp_direction(best_h, best_corrs, dx, target_sum, n)
            if direction is None:
                break
            h_trial, c5_trial = line_search_from_lp(best_h, direction, best_corrs, dx, n, target_sum)
            if h_trial is not None:
                best_h = h_trial
                best_c5 = c5_trial
                best_corrs = compute_corrs(best_h)
                # Coordinate descent polish after LP step
                for _ in range(3):
                    if remaining_time(t0, budget_s) < 15:
                        break
                    best_h, best_corrs, _ = coord_descent_sweep(best_h, best_corrs, offsets, n)
                    best_c5 = best_corrs.max() * dx
            else:
                break
    
    # Phase 3: Subgradient descent with polynomial decay
    if remaining_time(t0, budget_s) > 200:
        h_sg = best_h.copy()
        c5_sg = best_c5
        corrs_sg = best_corrs.copy()
        T = 2000
        for t in range(1, T + 1):
            if remaining_time(t0, budget_s) < 30:
                break
            step_size = 0.5 / (t ** 0.6)
            h_sg = subgradient_step(h_sg, corrs_sg, dx, n, target_sum, step_size, offsets)
            corrs_sg = compute_corrs(h_sg)
            c5_sg = corrs_sg.max() * dx
            if t % 100 == 0:
                # Coordinate descent polish
                h_sg, corrs_sg, _ = coord_descent_sweep(h_sg, corrs_sg, offsets, n)
                c5_sg = corrs_sg.max() * dx
            if c5_sg < best_c5:
                best_h = h_sg.copy()
                best_c5 = c5_sg
                best_corrs = corrs_sg.copy()
    
    # Phase 4: Binary steepest descent from current best
    if remaining_time(t0, budget_s) > 150:
        time_left = min(remaining_time(t0, budget_s) - 30, 120)
        h_bin, c5_bin = binary_steepest_descent(best_h, dx, max_sweeps=20,
                                                 time_limit=time_left, t0=t0)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin
            best_corrs = compute_corrs(best_h)
    
    # Phase 5: Coordinate descent from binary
    if remaining_time(t0, budget_s) > 30:
        for _ in range(10):
            if remaining_time(t0, budget_s) < 15:
                break
            best_h, best_corrs, improved = coord_descent_sweep(best_h, best_corrs, offsets, n)
            best_c5 = best_corrs.max() * dx
            if not improved:
                break
    
    # Phase 6: LP + line search again
    if remaining_time(t0, budget_s) > 80:
        for it in range(30):
            if remaining_time(t0, budget_s) < 20:
                break
            d_val, direction = lp_direction(best_h, best_corrs, dx, target_sum, n)
            if direction is None:
                break
            h_trial, c5_trial = line_search_from_lp(best_h, direction, best_corrs, dx, n, target_sum)
            if h_trial is not None:
                best_h = h_trial
                best_c5 = c5_trial
                best_corrs = compute_corrs(best_h)
                for _ in range(2):
                    if remaining_time(t0, budget_s) < 15:
                        break
                    best_h, best_corrs, _ = coord_descent_sweep(best_h, best_corrs, offsets, n)
                    best_c5 = best_corrs.max() * dx
            else:
                break
    
    # Phase 7: Try different initializations
    for mode in range(3):
        if remaining_time(t0, budget_s) < 150:
            break
        h_init = make_structured_start(n, target_sum, rng, mode)
        # Coordinate descent from this start
        corrs_init = compute_corrs(h_init)
        for _ in range(8):
            if remaining_time(t0, budget_s) < 30:
                break
            h_init, corrs_init, improved = coord_descent_sweep(h_init, corrs_init, offsets, n)
            if not improved:
                break
        c5_init = corrs_init.max() * dx
        if c5_init < best_c5:
            best_h = h_init
            best_c5 = c5_init
            best_corrs = corrs_init
            # Continue with LP from here
            if remaining_time(t0, budget_s) > 60:
                for it in range(20):
                    if remaining_time(t0, budget_s) < 20:
                        break
                    d_val, direction = lp_direction(best_h, best_corrs, dx, target_sum, n)
                    if direction is None:
                        break
                    h_trial, c5_trial = line_search_from_lp(best_h, direction, best_corrs, dx, n, target_sum)
                    if h_trial is not None:
                        best_h = h_trial
                        best_c5 = c5_trial
                        best_corrs = compute_corrs(best_h)
                    else:
                        break
    
    # Phase 8: Final coordinate descent polish
    if remaining_time(t0, budget_s) > 20:
        for _ in range(10):
            if remaining_time(t0, budget_s) < 10:
                break
            best_h, best_corrs, improved = coord_descent_sweep(best_h, best_corrs, offsets, n)
            best_c5 = best_corrs.max() * dx
            if not improved:
                break
    
    # Phase 9: SLSQP polish with high beta
    if remaining_time(t0, budget_s) > 80:
        h_opt, c5_val = local_slsqp(best_h, dx, target_sum, 500, 500, n)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_opt
            best_corrs = compute_corrs(best_h)
            # CD polish
            for _ in range(5):
                if remaining_time(t0, budget_s) < 10:
                    break
                best_h, best_corrs, _ = coord_descent_sweep(best_h, best_corrs, offsets, n)
                best_c5 = best_corrs.max() * dx
    
    # Final cleanup
    n_b = len(best_h)
    dx_b = 2.0 / n_b
    ts_b = n_b / 2.0
    best_h = project_feasible(best_h, ts_b)
    best_c5 = c5_of_h(best_h, dx_b)
    
    return best_h, best_c5, len(best_h)
