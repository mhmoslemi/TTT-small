def make_sym_h(q, n):
    m = n // 2
    h = np.empty(n)
    h[:m] = q
    h[m:] = 1.0 - q[::-1]

def all_corrs(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx


def subgrad_sym(q, n):

    h = make_sym_h(q, n)
    corrs = all_corrs(h)
    cmax = corrs.max()
    tol = 1e-6

        active[np.argmax(corrs)] = True

                    best_val = func_val
                    best_d = d_val
            if best_val < cmax - 1e-10:
                q[j] = np.clip(q[j] + best_d, 0.0, 1.0)
                cmax = best_val
                improved = True
        if not improved:
            break
    h = make_sym_h(q, n)
    return q, all_corrs(h).max()


def binary_local_search(h, n, max_sweeps=20):
    m_half = n // 2
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h <= 0.5)[0]
    h = h.copy()
    corrs = all_corrs(h)
    cmax = corrs.max()
    for sweep in range(max_sweeps):
        improved = False
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h <= 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        best_gain = 1e-10
        best_pair = None
        for io in ones:
            for iz in zeros:
                new_h = h.copy()
                new_h[io] = 0.0
                new_h[iz] = 1.0
                c_new = all_corrs(new_h).max()
                gain = cmax - c_new
                if gain > best_gain:
                    best_gain = gain
                    best_pair = (io, iz, c_new)
        if best_pair is not None:
            io, iz, c_new = best_pair
            h[io] = 0.0
            h[iz] = 1.0
            cmax = c_new
            improved = True
        if not improved:
            break
    return h, cmax


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def remaining():
        return budget_s - (time.time() - t0)

    try:
        h_parent = np.array(initial_h_values, dtype=float)
    except NameError:
        h_parent = 0.5 * np.ones(126)

    n_init = len(h_parent)
    if n_init % 2 != 0:
        h_parent = h_parent[:n_init - 1]
    n = len(h_parent)
    m = n // 2

    q = np.mean([h_parent[:m], 1.0 - h_parent[n - 1:n - 1 - m:-1]], axis=0)
    q = np.clip(q, 0.0, 1.0)

    h_best = make_sym_h(q, n)
    c5_best = all_corrs(h_best).max()
    print(f"Start: n={n}, C5={c5_best:.7f}")

    # Phase 1: Projected subgradient with line search
    alpha0 = 0.05
    no_imp = 0
    for it in range(3000):
        if remaining() < 40:
            break
        if it % 500 == 0 and it > 0 and no_imp > 50:
            break
        g_q, ccur = subgrad_sym(q, n)
        q, c_new = line_search(q, n, g_q, ccur, alpha0)
        c_actual = all_corrs(make_sym_h(q, n)).max()
        if c_actual < c5_best - 1e-10:
            c5_best = c_actual
            h_best = make_sym_h(q, n)
            no_imp = 0
        else:
            no_imp += 1
            alpha0 *= 0.995
    print(f"After subgrad1: C5={c5_best:.7f}, iter={it}")

    # Phase 2: Coordinate descent
    q, c5_cd = coord_desc_fast(q, n, max_sweeps=15)
    if c5_cd < c5_best:
        c5_best = c5_cd
        h_best = make_sym_h(q, n)
        print(f"After coord1: C5={c5_best:.7f}")

    # Phase 3: More subgradient with smaller steps
    alpha0 = 0.01
    no_imp = 0
    for it in range(3000):
        if remaining() < 30:
            break
        if no_imp > 100:
            break
        g_q, ccur = subgrad_sym(q, n)
        q, c_new = line_search(q, n, g_q, ccur, alpha0)
        c_actual = all_corrs(make_sym_h(q, n)).max()
        if c_actual < c5_best - 1e-10:
            c5_best = c_actual
            h_best = make_sym_h(q, n)
            no_imp = 0
            alpha0 *= 0.999
        else:
            no_imp += 1
            alpha0 *= 0.99
    print(f"After subgrad2: C5={c5_best:.7f}")

    # Phase 4: Coordinate descent again
    q, c5_cd = coord_desc_fast(q, n, max_sweeps=20)
    if c5_cd < c5_best:
        c5_best = c5_cd
        h_best = make_sym_h(q, n)
        print(f"After coord2: C5={c5_best:.7f}")

    # Phase 5: Multi-start
    n_starts = 25
    for trial in range(n_starts):
        if remaining() < 40:
            break
        if trial % 5 == 0:
            q_try = np.clip(q + 0.08 * rng.randn(m), 0.0, 1.0)
        elif trial % 5 == 1:
            n_ones = m // 2
            idx = rng.permutation(m)[:n_ones]
            q_try = np.zeros(m)
            q_try[idx] = 1.0
            q_try = uniform_filter1d(q_try, size=3)
            q_try = np.clip(q_try, 0.0, 1.0)
        elif trial % 5 == 2:
            q_try = np.clip(0.5 + 0.4 * rng.randn(m), 0.0, 1.0)
            q_try = uniform_filter1d(q_try, size=5)
        elif trial % 5 == 3:
            # Periodic pattern
            period = rng.randint(3, 8)
            base = rng.uniform(0.2, 0.8, period)
            q_try = np.tile(base, m // period + 1)[:m]
            q_try = np.clip(q_try + 0.05 * rng.randn(m), 0.0, 1.0)
        else:
            # Antisymmetric perturbation
            q_try = np.clip(q + 0.15 * rng.randn(m), 0.0, 1.0)

        alpha0 = 0.03
        for it in range(800):
            if remaining() < 20:
                break
            g_try, _ = subgrad_sym(q_try, n)
            q_try, _ = line_search(q_try, n, g_try, all_corrs(make_sym_h(q_try, n)).max(), alpha0)

        q_try, c_try = coord_desc_fast(q_try, n, max_sweeps=5)
        h_try = make_sym_h(q_try, n)
        c_try = all_corrs(h_try).max()

        if c_try < c5_best - 1e-9:
            c5_best = c_try
            h_best = h_try
            q = q_try
            print(f"  MultiStart {trial}: C5={c5_best:.7f}")
            if remaining() > 20:
                alpha0 = 0.01
                for it in range(1000):
                    if remaining() < 15:
                        break
                    g_try, _ = subgrad_sym(q_try, n)
                    q_try, _ = line_search(q_try, n, g_try, all_corrs(make_sym_h(q_try, n)).max(), alpha0)
                q_try, c_try2 = coord_desc_fast(q_try, n, max_sweeps=10)
                h_try = make_sym_h(q_try, n)
                c_try2 = all_corrs(h_try).max()
                if c_try2 < c5_best:
                    c5_best = c_try2
                    h_best = h_try
                    q = q_try

    # Phase 6: Try other n values
    for n_try in [120, 124, 128, 132, 118, 134, 122, 130, 136, 116]:
        if n_try == n or remaining() < 35 or n_try < 40 or n_try % 2 != 0:
            continue
        x_old = np.linspace(0, 2, n, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, h_best, left=0, right=0)
        m_try = n_try // 2
        q_try = np.mean([h_interp[:m_try], 1.0 - h_interp[n_try - 1:n_try - 1 - m_try:-1]], axis=0)
        q_try = np.clip(q_try, 0.0, 1.0)

        alpha0 = 0.03
        for it in range(1000):
            if remaining() < 20:
                break
            g_try, _ = subgrad_sym(q_try, n_try)
            q_try, _ = line_search(q_try, n_try, g_try, all_corrs(make_sym_h(q_try, n_try)).max(), alpha0)

        q_try, c_try = coord_desc_fast(q_try, n_try, max_sweeps=10)
        h_try = make_sym_h(q_try, n_try)
        c_try = all_corrs(h_try).max()

        if c_try < c5_best - 1e-9:
            c5_best = c_try
            h_best = h_try
            n = n_try
            m = m_try
            q = q_try
            print(f"  n={n}: C5={c5_best:.7f}")

    # Phase 7: Final coordinate descent polish
    if remaining() > 15:
        q, c5_cd = coord_desc_fast(q, n, max_sweeps=30)
        if c5_cd < c5_best:
            c5_best = c5_cd
            h_best = make_sym_h(q, n)
            print(f"Final coord: C5={c5_best:.7f}")

    # Phase 8: Final subgradient
    if remaining() > 15:
        alpha0 = 0.005
        for it in range(2000):
            if remaining() < 10:
                break
            g_q, ccur = subgrad_sym(q, n)
            q, c_new = line_search(q, n, g_q, ccur, alpha0)
            c_actual = all_corrs(make_sym_h(q, n)).max()
            if c_actual < c5_best - 1e-10:
                c5_best = c_actual
                h_best = make_sym_h(q, n)

    # Final validation
    h_best = np.clip(h_best, 0.0, 1.0)
    target = n / 2.0
    diff = target - h_best.sum()
    if abs(diff) > 1e-12:
        mask = (h_best > 1e-10) & (h_best < 1.0 - 1e-10)
        if mask.sum() > 0:
            h_best[mask] += diff / mask.sum()
        else:
            h_best += diff / n
    h_best = np.clip(h_best, 0.0, 1.0)
    c5_best = all_corrs(h_best).max()
    print(f"FINAL: n={n}, C5={c5_best:.7f}")
    return h_best, c5_best, len(h_best)
