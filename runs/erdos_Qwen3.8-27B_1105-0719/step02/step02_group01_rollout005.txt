def compute_corrs(h):

def nesterov_subgrad(h0, dx, target_sum, max_iter, beta_w):

def slsqp_direct(h0, dx, target_sum, beta, maxiter):
    n = len(h0)

    def obj(x):
        corrs = compute_corrs(x)
        return logsumexp(beta * corrs) / beta * dx

    def grad_obj(x):
        return gradient_smooth_max(x, dx, beta)

    def c_fun(x):
        return x.sum() - target_sum

    def c_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': c_fun, 'jac': c_jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0, jac=grad_obj, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        h_out = project_feasible(res.x, target_sum)
    except Exception:
        h_out = project_feasible(h0, target_sum)
    return h_out, c5_of_h(h_out, dx)


def make_binary_start(n, target_sum, rng):
    h = np.zeros(n)
    idx = rng.permutation(n)
    n_ones = int(target_sum)
    h[idx[:n_ones]] = 1.0
    frac = target_sum - n_ones
    if frac > 1e-10 and n_ones < n:
        h[idx[n_ones]] = frac
    return h


def make_periodic_start(n, target_sum, rng, period):
    h = np.zeros(n)
    fill = max(1, int(target_sum / n * period))
    for i in range(n):
        h[i] = 1.0 if (i % period) < fill else 0.0
    return project_feasible(h, target_sum)


def make_sturmian_start(n, target_sum, rng):
    h = np.zeros(n)
    alpha = target_sum / n
    phase = rng.uniform(0, 1)
    for i in range(n):
        h[i] = 1.0 if (phase + i * alpha) % 1.0 < alpha else 0.0
    return project_feasible(h, target_sum)


def sigma_a_to_h(sigma_a, n_use):
    n = len(sigma_a)
    if n <= 0:
        return np.ones(2) * 0.5
    n_use = max(2, min(n_use, n))
    x = np.linspace(0.0, 1.0, number=n_use)
    mins = np.diff(np.minimum.accumulate(sigma_a))
    c0 = -mins[0] + (mins >= 0).sum()
    row = c0 + np.minimum.accumulate(mins)
    h_raw = np.zeros((n_use, n_use))
    for jb in range(n_use):
        row_jb = (row[:, jb] + row[jb, :]) / 2.0
        p_raw = np.abs(row_jb) / (2.0 * sigma_a[jb])
        p_mat = np.minimum(np.maximum(p_raw, 0.0), 1.0)
        h_raw[:, jb] = p_mat
    return h_raw[0]


def refined_feasible(h0_int, sigma_a, n, k_keep, target_sum):
    h_base = sigma_a_to_h(sigma_a, n)
    h = h_base.copy()
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        hh = project_feasible(h_base + mid * h0_int, target_sum)
        if hh.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) * 0.5
    for _ in range(30):
        hh = project_feasible(h_base + t * h0_int, target_sum)
        new_t = t + (target_sum - hh.sum()) / (n * (k_keep + 0.5))
        if abs(new_t - t) < 1e-15:
            t = new_t
            break
        t = max(-2.0, min(2.0, new_t))
    return project_feasible(h_base + t * h0_int, target_sum)


def remaining_time(t0, budget_s):
    return budget_s - (time.time() - t0)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')

    def try_update(h_cand, c5_cand):
        nonlocal best_h, best_c5
        if h_cand is None or c5_cand is None:
            return
        if not np.all(np.isfinite(h_cand)):
            return
        h_cand = np.clip(h_cand, 0.0, 1.0)
        c5_cand = c5_of_h(h_cand, 2.0 / len(h_cand))
        if c5_cand < best_c5 - 1e-10:
            best_c5 = c5_cand
            best_h = h_cand.copy()

    # Load parent
    h0 = np.array(initial_h_values, dtype=float)
    n_par = len(h0)
    dx_par = 2.0 / n_par
    ts_par = n_par / 2.0
    h0 = project_feasible(h0, ts_par)
    c5_par = c5_of_h(h0, dx_par)
    try_update(h0, c5_par)

    # Phase 1: Parent polish with stronger LP
    if remaining_time(t0, budget_s) > 60:
        h_lp, c5_lp = lp_polish_full(best_h, dx_par, ts_par, 300, 0.3, 0.85)
        try_update(h_lp, c5_lp)

    if remaining_time(t0, budget_s) > 40:
        for beta in [200, 500, 1000, 2000, 4000]:
            if remaining_time(t0, budget_s) < 15:
                break
            h_opt, c5_val = slsqp_direct(best_h, dx_par, ts_par, beta, 800)
            try_update(h_opt, c5_val)

    if remaining_time(t0, budget_s) > 30:
        h_lp, c5_lp = lp_polish_full(best_h, dx_par, ts_par, 200, 0.2, 0.9)
        try_update(h_lp, c5_lp)

    if remaining_time(t0, budget_s) > 40:
        h_ng, c5_ng = nesterov_subgrad(best_h, dx_par, ts_par, 5000, 50.0)
        try_update(h_ng, c5_ng)

    # Phase 2: Try different n values with diverse starts
    n_candidates = [48, 56, 64, 72, 80, 96, 112, 128, 144, 160, 32, 40]
    n_candidates = sorted(set(n_candidates), key=lambda x: abs(x - n_par))

    for n_try in n_candidates:
        if remaining_time(t0, budget_s) < 20:
            break
        if n_try == n_par:
            continue
        dx_try = 2.0 / n_try
        ts_try = n_try / 2.0

        # Interpolate parent to new grid
        x_old = np.linspace(0.0, 2.0, n_par)
        x_new = np.linspace(0.0, 2.0, n_try)
        h_interp = np.interp(x_new, x_old, h0)
        h_interp = project_feasible(h_interp, ts_try)

        h_opt, c5_val = slsqp_direct(h_interp, dx_try, ts_try, 300, 600)
        try_update(h_opt, c5_val)

        if remaining_time(t0, budget_s) > 15:
            h_lp, c5_lp = lp_polish_full(h_opt, dx_try, ts_try, 150, 0.25, 0.85)
            try_update(h_lp, c5_lp)

    # Phase 3: Diverse starts on parent grid
    for trial in range(25):
        if remaining_time(t0, budget_s) < 15:
            break

        if trial % 6 == 0:
            h_init = make_binary_start(n_par, ts_par, rng)
        elif trial % 6 == 1:
            h_init = make_periodic_start(n_par, ts_par, rng, rng.choice([3, 4, 5, 6, 7, 8]))
        elif trial % 6 == 2:
            h_init = make_sturmian_start(n_par, ts_par, rng)
        elif trial % 6 == 3:
            h_init = project_feasible(best_h + 0.15 * (rng.random(n_par) - 0.5), ts_par)
        elif trial % 6 == 4:
            h_init = make_binary_start(n_par, ts_par, rng)
            h_init = project_feasible(h_init + 0.3 * rng.random(n_par), ts_par)
        else:
            # Complement of current best
            h_init = project_feasible(1.0 - best_h, ts_par)

        h_opt, c5_val = slsqp_direct(h_init, dx_par, ts_par, 200, 400)
        try_update(h_opt, c5_val)

        if remaining_time(t0, budget_s) > 20 and c5_val < best_c5 + 0.01:
            h_opt2, c5_val2 = slsqp_direct(h_opt, dx_par, ts_par, 800, 600)
            try_update(h_opt2, c5_val2)
            if remaining_time(t0, budget_s) > 12:
                h_lp, c5_lp = lp_polish_full(h_opt2, dx_par, ts_par, 100, 0.2, 0.85)
                try_update(h_lp, c5_lp)

    # Phase 4: Sigma-a based construction (from known constructions)
    if remaining_time(t0, budget_s) > 80:
        # Try small grids with sigma-a construction
        for n_sa in [32, 48, 64, 80, 96, 128]:
            if remaining_time(t0, budget_s) < 20:
                break
            ts_sa = n_sa / 2.0
            dx_sa = 2.0 / n_sa

            # Generate sigma_a: use a structured PSD matrix
            # sigma_a[i,j] = max(0, 1 - |i-j|/n) (toeplitz with linear decay)
            idx = np.arange(n_sa)
            sigma_a = np.maximum(0, 1.0 - np.abs(idx[:, None] - idx[None, :]) / float(n_sa))
            # Add small diagonal to ensure positive definiteness
            sigma_a += np.eye(n_sa) * 0.01

            h_sa = sigma_a_to_h(sigma_a, n_sa)
            h_sa = project_feasible(h_sa, ts_sa)
            try_update(h_sa, c5_of_h(h_sa, dx_sa))

            # Polish the sigma-a construction
            if remaining_time(t0, budget_s) > 15:
                h_opt, c5_val = slsqp_direct(h_sa, dx_sa, ts_sa, 300, 500)
                try_update(h_opt, c5_val)
            if remaining_time(t0, budget_s) > 12:
                h_lp, c5_lp = lp_polish_full(h_sa, dx_sa, ts_sa, 100, 0.3, 0.85)
                try_update(h_lp, c5_lp)

    # Phase 5: Aggressive final polish on best solution
    if remaining_time(t0, budget_s) > 20 and best_h is not None:
        n_b = len(best_h)
        dx_b = 2.0 / n_b
        ts_b = n_b / 2.0

        # Multiple LP passes with decreasing alpha
        for alpha_pass in [0.3, 0.2, 0.1, 0.05, 0.02, 0.01]:
            if remaining_time(t0, budget_s) < 8:
                break
            h_lp, c5_lp = lp_polish_full(best_h, dx_b, ts_b, 150, alpha_pass, 0.9)
            try_update(h_lp, c5_lp)

        # High-beta SLSQP
        if remaining_time(t0, budget_s) > 15:
            for beta in [3000, 5000, 8000]:
                if remaining_time(t0, budget_s) < 10:
                    break
                h_opt, c5_val = slsqp_direct(best_h, dx_b, ts_b, beta, 1000)
                try_update(h_opt, c5_val)

        # Nesterov subgradient
        if remaining_time(t0, budget_s) > 20:
            h_ng, c5_ng = nesterov_subgrad(best_h, dx_b, ts_b, 8000, 80.0)
            try_update(h_ng, c5_ng)

        # Final LP
        if remaining_time(t0, budget_s) > 10:
            h_lp, c5_lp = lp_polish_full(best_h, dx_b, ts_b, 200, 0.05, 0.95)
            try_update(h_lp, c5_lp)

    # Phase 6: Try to improve using complement + projection trick
    if remaining_time(t0, budget_s) > 15 and best_h is not None:
        n_b = len(best_h)
        ts_b = n_b / 2.0
        # The complement 1-h has the same C5 (by symmetry of the problem)
        # But it might be in a different basin for the optimizer
        h_comp = 1.0 - best_h
        h_comp = project_feasible(h_comp, ts_b)
        if remaining_time(t0, budget_s) > 10:
            h_opt, c5_val = slsqp_direct(h_comp, 2.0 / n_b, ts_b, 1000, 800)
            try_update(h_opt, c5_val)
            if remaining_time(t0, budget_s) > 8:
                h_lp, c5_lp = lp_polish_full(h_opt, 2.0 / n_b, ts_b, 100, 0.15, 0.9)
                try_update(h_lp, c5_lp)

    # Phase 7: Fine-tune n around best
    if remaining_time(t0, budget_s) > 30 and best_h is not None:
        n_b = len(best_h)
        for dn in [-4, -2, 2, 4, 6, -6]:
            if remaining_time(t0, budget_s) < 12:
                break
            n_try = n_b + dn
            if n_try < 16 or n_try > 200 or n_try % 2 != 0:
                continue
            dx_try = 2.0 / n_try
            ts_try = n_try / 2.0
            x_old = np.linspace(0.0, 2.0, len(best_h))
            x_new = np.linspace(0.0, 2.0, n_try)
            h_interp = np.interp(x_new, x_old, best_h)
            h_interp = project_feasible(h_interp, ts_try)
            h_opt, c5_val = slsqp_direct(h_interp, dx_try, ts_try, 500, 600)
            try_update(h_opt, c5_val)
            if remaining_time(t0, budget_s) > 10:
                h_lp, c5_lp = lp_polish_full(h_opt, dx_try, ts_try, 100, 0.2, 0.9)
                try_update(h_lp, c5_lp)

    # Final: ensure valid output
    if best_h is None:
        best_h = np.ones(n_par) * 0.5
        best_c5 = c5_of_h(best_h, dx_par)

    best_h = np.clip(best_h, 0.0, 1.0)
    n_final = len(best_h)
    # Final feasibility fix
    best_h = project_feasible(best_h, n_final / 2.0)
    best_c5 = c5_of_h(best_h, 2.0 / n_final)

    return best_h, best_c5, n_final
