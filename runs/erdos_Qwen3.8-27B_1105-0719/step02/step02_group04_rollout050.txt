Let me think about what would be most effective:

1. **SLP approach**: At each iteration, linearize the active overlap constraints and solve an LP to find a descent direction. This directly addresses the nonsmooth max structure.

2. **Correct gradients**: The gradient of c_j(h) = Σᵢ hᵢ(1-hᵢ₋d) with respect to h is:
   - ∂c_j/∂h_a = 1 - h_b (for term h_a * (1-h_b))
   - ∂c_j/∂h_b = -h_a (for term h_a * (1-h_b))

3. **Anti-complementary symmetry**: h[i] = q[i], h[n-1-i] = 1-q[i] guarantees sum(n/2) by construction.

Let me think about the LP formulation:
- Variables: delta (step in q-space), t (epigraph), slacks
- Objective: minimize t + penalty * sum(slacks)
- Constraints: For each active lag j:
  - grad_j · delta ≤ t - c_j (linearized constraint)
- Box: -q ≤ delta ≤ 1-q (ensuring 0 ≤ q+delta ≤ 1)
- For each i: |delta_i| ≤ slack_i (L1 penalty on movement)

Actually, let me reconsider. The standard SLP for min-max:
- We want to minimize max_j c_j(h)
- At current point h₀ with value t₀ = max c_j
- Linearize: c_j(h₀ + Δ) ≈ c_j(h₀) + ∇c_j(h₀)·Δ
- LP: min t s.t. c_j(h₀) + ∇c_j(h₀)·Δ ≤ t for all active j
            h₀ + Δ ∈ [0,1]^n, sum(h₀+Δ) = n/2
            ||Δ||_∞ ≤ R (trust region)

For the anti-symmetric case, we work in q-space where sum is automatic.

Let me think about the implementation more carefully:

For the gradient computation:
- c_j = Σ_{i=L_j}^{U_j} h_i * (1 - h_{i-d_j}) where d_j = j - (n-1)
- L_j = max(0, d_j), U_j = min(n-1, n-1+d_j)

For the q-variable chain rule:
- h[i] = q[i] for i < n/2
- h[n-1-i] = 1 - q[i] for i < n/2

So if grad_h is the gradient w.r.t. h:
- grad_q[i] = grad_h[i] - grad_h[n-1-i]

This is because:
- d/dq_i h[i] = 1 (for i < m)
- d/dq_i h[n-1-i] = -1 (for i < m)

So grad_q[i] = grad_h[i] * 1 + grad_h[n-1-i] * (-1) = grad_h[i] - grad_h[n-1-i]

import numpy as np
from scipy.optimize import linprog
from scipy.optimize import minimize
import time

def dx_of(n):
    return 2.0 / n

def corr_scaled(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * dx_of(n)

def eval_c5(h):
    return float(np.max(corr_scaled(h)))

def is_feasible(h, tol=1e-9):
    if not np.all(np.isfinite(h)):
        return False
    if np.min(h) < -tol or np.max(h) > 1.0 + tol:
        return False
    n = len(h)
    if abs(h.sum() * dx_of(n) - 1.0) > tol:
        return False
    return True

def repair_box_sum(h, target_sum):
    """Water-filling to project h onto [0,1]^n with sum = target_sum."""
    n = len(h)
    h = h.copy()
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0.0, 1.0).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    h_new = np.clip(h - lam, 0.0, 1.0)
    diff = target_sum - h_new.sum()
    if abs(diff) > 1e-14:
        mask = (h_new > 1e-12) & (h_new < 1.0 - 1e-12)
        if mask.any():
            h_new[mask] += diff / mask.sum()
    return np.clip(h_new, 0.0, 1.0)

def build_sym(q, n):
    """Build anti-complementary h from q. n must be even, len(q) = n//2."""
    m = n // 2
    h = np.empty(n)
    h[:m] = q
    h[m:] = 1.0 - q[::-1]
    return h

def q_from_h(h, n):
    """Extract q from anti-complementary h."""
    m = n // 2
    return 0.5 * (h[:m] + 1.0 - h[n-1:m-1:-1])

def grad_lag(h, j, n):
    """Exact gradient of c_j (unscaled) w.r.t. h.
    c_j = sum over valid i of h[i] * (1 - h[i-d])
    where d = j - (n-1)
    """
    d = j - (n - 1)
    L = max(0, d)
    U = min(n - 1, n - 1 + d)
    g = np.zeros(n)
    if L > U:
        return g
    i = np.arange(L, U + 1)
    a = i  # h_index
    b = i - d  # complement index
    # For each term h[a] * (1 - h[b]):
    # d/da: 1 - h[b], d/db: -h[a]
    g[a] += (1.0 - h[b])
    g[b] -= h[a]
    return g

def grad_lag_q(h, j, n):
    """Gradient of c_j w.r.t. q variables (anti-complementary)."""
    m = n // 2
    gh = grad_lag(h, j, n)
    gq = np.zeros(m)
    for i in range(m):
        gq[i] = gh[i] - gh[n - 1 - i]
    return gq

def select_active(h, n, margin_abs=0.01, pad=2):
    """Select active lags (near the maximum)."""
    cv = corr_scaled(h)
    mx = cv.max()
    thresh = mx - margin_abs
    idx = np.where(cv >= thresh)[0]
    # Pad
    extended = set()
    for a in idx:
        for delta in range(-pad, pad + 1):
            k = int(a + delta)
            if 0 <= k <= 2 * n - 2:
                extended.add(k)
    return np.sort(np.array(extended, dtype=int))

def slp_step_q(q, n, active_j, R, move_pen, max_lp_iter=200):
    """One SLP step in q-space. Returns (delta, success)."""
    m = n // 2
    h = build_sym(q, n)
    dx = dx_of(n)
    
    # Compute linearization
    c_j = np.array([float(np.sum(h[max(0, j-(n-1)):min(n, n-j+(n-1))] * (1.0 - h[max(0, j-(n-1) - (j-(n-1)):min(n, n-j+(n-1)) -(j-(n-1))]))) for j in active_j])
    
    # Actually let's just use the correlation directly
    cv = corr_scaled(h)
    c_vals = cv[active_j]
    
    # Compute gradients
    A = np.zeros((len(active_j), m))
    for i, j in enumerate(active_j):
        A[i, :] = dx * grad_lag_q(h, j, n)
    
    # LP: min t + move_pen * sum(s)
    # s.t. A @ delta - t <= -c_vals (i.e., c_val + A@delta <= t)
    #      |delta_i| <= s_i
    #      -q_i <= delta_i <= 1-q_i
    #      s_i >= 0
    #      -R <= delta_i <= R
    
    # Variables: [delta (m), t (1), s (m)]
    # Total: 2m + 1
    nvar = 2 * m + 1
    c_obj = np.zeros(nvar)
    c_obj[m] = 1.0  # minimize t
    c_obj[m+1:] = move_pen  # + move_pen * sum(s)
    
    # Constraints: A @ delta - t <= -c_vals
    # => A @ delta + 0*... - 1*t <= -c_vals
    A_ub = np.zeros((len(active_j), nvar))
    A_ub[:, :m] = A
    A_ub[:, m] = -1.0
    b_ub = -c_vals
    
    # |delta_i| <= s_i:
    # delta_i - s_i <= 0  (for each i)
    # -delta_i - s_i <= 0
    n_ineq = len(active_j) + 2 * m
    A_ub2 = np.zeros((n_ineq, nvar))
    b_ub2 = np.zeros(n_ineq)
    
    # First part is already in A_ub
    # delta_i - s_i <= 0
    for i in range(m):
        A_ub2[len(active_j) + i, i] = 1.0
        A_ub2[len(active_j) + i, m + 1 + i] = -1.0
    # -delta_i - s_i <= 0
    for i in range(m):
        A_ub2[len(active_j) + m + i, i] = -1.0
        A_ub2[len(active_j) + m + i, m + 1 + i] = -1.0
    
    A_full = np.vstack([A_ub, A_ub2])
    b_full = np.concatenate([b_ub, b_ub2])
    
    # Bounds
    bounds = []
    for i in range(m):
        lo = max(-q[i], -R)
        hi = min(1.0 - q[i], R)
        if lo > hi:
            lo = hi = 0.0
        bounds.append((lo, hi))
    bounds.append((-R * 2, R * 2 + float(c_vals.max()) + 0.1))  # t
    for i in range(m):
        bounds.append((0.0, R))  # s
    
    try:
        res = linprog(c_obj, A_ub=A_full, b_ub=b_full, bounds=bounds,
                      method='highs', options={'maxiter': max_lp_iter})
        if res.status != 0:
            return None, False
        delta = res.x[:m]
        return delta, True
    except Exception:
        return None, False

def slp_optimize_q(q, n, max_iter=50, R0=0.05, move_pen=0.02, 
                   margin_abs=0.01, time_limit=None, t0=None):
    """SLP optimization in q-space."""
    best_q = q.copy()
    best_c = eval_c5(build_sym(q, n))
    R = R0
    no_progress = 0
    
    for it in range(max_iter):
        if time_limit is not None and (time.time() - t0) > time_limit:
            break
        
        h = build_sym(best_q, n)
        active = select_active(h, n, margin_abs=margin_abs)
        
        delta, ok = slp_step_q(best_q, n, active, R, move_pen)
        if not ok or delta is None:
            no_progress += 1
            R *= 0.7
            if no_progress > 5:
                margin_abs *= 0.5
                no_progress = 0
                if margin_abs < 0.001:
                    break
            continue
        
        # Line search
        best_trial_c = best_c
        best_trial_q = best_q
        for lam in [1.0, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05]:
            q_try = np.clip(best_q + lam * delta, 0.0, 1.0)
            h_try = build_sym(q_try, n)
            c_try = eval_c5(h_try)
            if c_try < best_trial_c - 1e-12:
                best_trial_c = c_try
                best_trial_q = q_try.copy()
        
        if best_trial_c < best_c - 1e-12:
            best_q = best_trial_q
            best_c = best_trial_c
            no_progress = 0
            R = min(0.2, R * 1.3)
        else:
            no_progress += 1
            R *= 0.6
            if no_progress > 8:
                margin_abs *= 0.5
                no_progress = 0
                if margin_abs < 0.0005:
                    break
    
    return best_q, best_c

def slp_optimize_full(h, n, max_iter=30, R0=0.04, move_pen=0.03,
                      margin_abs=0.01, time_limit=None, t0=None):
    """SLP optimization in full h-space (with sum constraint)."""

    R = R0
    no_progress = 0
    target = n / 2.0

        m = n  # variable count
        dx = dx_of(n)
        
        c_vals = cv[active]
        A = np.zeros((len(active), m))
        for i, j in enumerate(active):
            A[i, :] = dx * grad_lag(best_h, j, n)
        
        nvar = 2 * m + 1  # delta, t, s
        c_obj = np.zeros(nvar)
        c_obj[m] = 1.0
        c_obj[m+1:] = move_pen
        
        # Constraints
        n_ineq = len(active) + 2 * m
        A_full = np.zeros((n_ineq, nvar))
        b_full = np.zeros(n_ineq)
        
        # A @ delta - t <= -c_vals
        A_full[:len(active), :m] = A
        A_full[:len(active), m] = -1.0
        b_full[:len(active)] = -c_vals
        
        # delta_i - s_i <= 0
        for i in range(m):
            A_full[len(active) + i, i] = 1.0
            A_full[len(active) + i, m + 1 + i] = -1.0
        # -delta_i - s_i <= 0
        for i in range(m):
            A_full[len(active) + m + i, i] = -1.0
            A_full[len(active) + m + i, m + 1 + i] = -1.0
        
        # Equality: sum(delta) = 0
        A_eq = np.zeros((1, nvar))
        A_eq[0, :m] = 1.0
        b_eq = np.array([0.0])
        
        bounds = []
        for i in range(m):
            lo = max(-best_h[i], -R)
            hi = min(1.0 - best_h[i], R)
            if lo > hi:

            bounds.append((lo, hi))
        bounds.append((-0.5, 0.5 + c_vals.max()))
        for i in range(m):
            bounds.append((0.0, R))
        
        try:
            res = linprog(c_obj, A_ub=A_full, b_ub=b_full, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs', options={'maxiter': 200})
            if res.status != 0:
                no_progress += 1
                R *= 0.7
                continue
            delta = res.x[:m]
        except Exception:
            no_progress += 1
            R *= 0.7
            continue
        
        # Line search
        best_trial_c = best_c
        best_trial_h = best_h
        for lam in [1.0, 0.7, 0.5, 0.3, 0.2, 0.1]:

            h_try = repair_box_sum(h_try, target)
            c_try = eval_c5(h_try)
            if c_try < best_trial_c - 1e-12:

                best_trial_h = h_try.copy()

            best_h = best_trial_h
            best_c = best_trial_c
            no_progress = 0
            R = min(0.15, R * 1.3)
        else:
            no_progress += 1
            R *= 0.6
            if no_progress > 10:
                break
    
    return best_h, best_c

def interp_h(h_old, n_new):
    """Interpolate h to new resolution."""
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    x_old = (np.arange(n_old) + 0.5) * dx_of(n_old)
    x_new = (np.arange(n_new) + 0.5) * dx_of(n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return repair_box_sum(h_new, n_new / 2.0)

def make_fourier_q(n, rng):

import numpy as np

def dx_of(n):
    return 2.0 / n


def corr_scaled(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * dx_of(n)


def eval_c5(h):
    return float(np.max(corr_scaled(h)))


def repair_box_sum(h, target_sum):
    n = len(h)
    h = h.copy()
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0.0, 1.0).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    h_new = np.clip(h - lam, 0.0, 1.0)
    diff = target_sum - h_new.sum()
    if abs(diff) > 1e-14:
        mask = (h_new > 1e-10) & (h_new < 1.0 - 1e-10)
        if mask.any():
            h_new[mask] += diff / mask.sum()
    return np.clip(h_new, 0.0, 1.0)


def build_sym(q, n):
    m = n // 2
    h = np.empty(n)
    h[:m] = q
    h[m:] = 1.0 - q[::-1]
    return h


def q_from_h(h, n):
    m = n // 2
    return 0.5 * (h[:m] + 1.0 - h[n - 1:m - 1:-1])


def grad_lag(h, j, n):
    d = j - (n - 1)
    L = max(0, d)
    U = min(n - 1, n - 1 + d)
    g = np.zeros(n)
    if L > U:
        return g
    i = np.arange(L, U + 1)
    a = i
    b = i - d
    g[a] += (1.0 - h[b])
    g[b] -= h[a]
    return g


def grad_lag_q(h, j, n):
    m = n // 2
    gh = grad_lag(h, j, n)
    gq = np.zeros(m)
    for i in range(m):
        gq[i] = gh[i] - gh[n - 1 - i]
    return gq


def select_active(h, n, margin_abs=0.01, pad=2):
    cv = corr_scaled(h)
    mx = cv.max()
    thresh = mx - margin_abs
    idx = np.where(cv >= thresh)[0]
    extended = set()
    for a in idx:
        for delta in range(-pad, pad + 1):
            k = int(a + delta)
            if 0 <= k <= 2 * n - 2:
                extended.add(k)
    return np.sort(np.array(extended, dtype=int))


def slp_step_q(q, n, active_j, R, move_pen):
    m = n // 2
    h = build_sym(q, n)
    dx = dx_of(n)
    cv = corr_scaled(h)
    c_vals = cv[active_j]

    A = np.zeros((len(active_j), m))
    for i, j in enumerate(active_j):
        A[i, :] = dx * grad_lag_q(h, j, n)

    nvar = 2 * m + 1
    c_obj = np.zeros(nvar)
    c_obj[m] = 1.0
    c_obj[m + 1:] = move_pen

    n_ineq = len(active_j) + 2 * m
    A_full = np.zeros((n_ineq, nvar))
    b_full = np.zeros(n_ineq)

    A_full[:len(active_j), :m] = A
    A_full[:len(active_j), m] = -1.0
    b_full[:len(active_j)] = -c_vals

    row_offset = len(active_j)
    for i in range(m):
        A_full[row_offset + i, i] = 1.0
        A_full[row_offset + i, m + 1 + i] = -1.0

        A_full[row_offset + m + i, i] = -1.0
        A_full[row_offset + m + i, m + 1 + i] = -1.0

    bounds = []

        hi = min(1.0 - q[i], R)

        h_f, c_f = slp_optimize_full(best_h, len(best_h), max_iter=30, R0=0.025,
                                      move_pen=0.025, margin_abs=0.008,
                                      t0=t0, time_limit=remaining() - reserve - 20)
        if c_f < best_c5:
            best_h = h_f.copy()
            best_c5 = c_f

    # Phase 3: Multi-resolution search
    n_base = len(best_h)
    n_cands = set()
    for d in [-12, -8, -6, -4, -3, -2, -1, 1, 2, 3, 4, 6, 8, 12, 16]:
        nc = n_base + d
        if nc >= 44 and nc % 2 == 0:
            n_cands.add(nc)
    n_cands = sorted(n_cands)

    for n_try in n_cands:
        if remaining() < reserve + 18:
            break
        if n_try == n_base:
            continue
        h_try = interp_h(best_h, n_try)
        q_try = q_from_h(h_try, n_try)
        h_try = build_sym(q_try, n_try)
        c_try = eval_c5(h_try)
        if c_try < best_c5:
            best_h = h_try.copy()
            best_c5 = c_try
        if remaining() > reserve + 12:
            q_try, c_val = slp_optimize_q(q_try, n_try, max_iter=25, R0=0.025,
                                           move_pen=0.015, margin_abs=0.008,
                                           t0=t0, time_limit=remaining() - reserve)
            h_new = build_sym(q_try, n_try)
            if c_val < best_c5:
                best_h = h_new.copy()
                best_c5 = c_val

    # Phase 4: Random restarts near best
    max_restarts = min(25, max(3, int(remaining() / 25.0)))
    for trial in range(max_restarts):
        if remaining() < reserve + 10:
            break
        n_r = len(best_h)
        if n_r % 2 != 0:
            n_r -= 1
            best_for_n = interp_h(best_h, n_r)
        else:
            best_for_n = best_h
        q_base = q_from_h(best_for_n, n_r)
        scale = 0.03 + 0.02 * (trial % 3)
        q_r = np.clip(q_base + scale * rng.randn(n_r // 2), 0.0, 1.0)
        h_r = build_sym(q_r, n_r)
        c_r = eval_c5(h_r)
        if c_r < best_c5 + 0.003:
            q_r, c_val = slp_optimize_q(q_r, n_r, max_iter=20, R0=0.03,
                                         move_pen=0.02, margin_abs=0.01,
                                         t0=t0, time_limit=remaining() - reserve)
            h_new = build_sym(q_r, n_r)
            if c_val < best_c5:
                best_h = h_new.copy()
                best_c5 = c_val

    # Phase 5: Final high-precision polish
    if remaining() > 20 and best_h is not None:
        n_fin = len(best_h)
        if n_fin % 2 == 0:
            q_fin = q_from_h(best_h, n_fin)
            q_fin, c_val = slp_optimize_q(q_fin, n_fin, max_iter=40, R0=0.008,
                                           move_pen=0.004, margin_abs=0.002,
                                           t0=t0, time_limit=remaining() - 12)
            h_fin = build_sym(q_fin, n_fin)
            if c_val < best_c5:
                best_h = h_fin.copy()
                best_c5 = c_val

        if remaining() > 15:
            h_fin2, c_fin2 = slp_optimize_full(best_h, len(best_h), max_iter=20,
                                                R0=0.015, move_pen=0.015,
                                                margin_abs=0.004,
                                                t0=t0, time_limit=remaining() - 8)
            if c_fin2 < best_c5:
                best_h = h_fin2.copy()
                best_c5 = c_fin2

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = repair_box_sum(best_h, len(best_h) / 2.0)
    best_c5 = eval_c5(best_h)

    return best_h, best_c5, len(best_h)
