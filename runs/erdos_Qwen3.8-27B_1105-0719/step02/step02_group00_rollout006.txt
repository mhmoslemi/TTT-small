import numpy as np
from scipy.optimize import linprog, minimize
import time

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx] + 1e-15)
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx] + 1e-15)
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    # Final fix
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def exact_score(h):
    return float(np.max(compute_profile(h)))

def build_gradient_rows(h, rows, n, dx):
    """Build gradient rows for specified lag indices.
    rows: array of indices into the profile (0 to 2n-2)
    Returns: |rows| x n matrix
    """
    G = np.zeros((len(rows), n))
    for r, m in enumerate(rows):
        delta = m - (n - 1)
        if delta == 0:
            G[r, :] = dx * (1.0 - 2.0 * h)
        elif delta > 0:
            # P = sum_{i=0}^{n-1-delta} h[i+delta]*(1-h[i])
            # dP/dh[j] = (1-h[j-delta]) for j >= delta
            #           -h[j+delta]     for j <= n-1-delta
            G[r, delta:] += dx * (1.0 - h[:n-delta])
            G[r, :n-delta] += -dx * h[delta:]
        else:
            d = -delta
            # P = sum_{i=0}^{n-1-d} h[i]*(1-h[i+d])
            # dP/dh[j] = (1-h[j+d]) for j <= n-1-d
            #           -h[j-d]     for j >= d
            G[r, :n-d] += dx * (1.0 - h[d:])
            G[r, d:] += -dx * h[:n-d]
    return G

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    # Bin-average interpolation
    edges_old = np.linspace(0, 2, n + 1)
    edges_new = np.linspace(0, 2, new_n + 1)
    # Cumulative integral of h (piecewise constant)
    cum_old = np.concatenate([[0], np.cumsum(h) * (2.0 / n)])
    h_new = np.zeros(new_n)
    for k in range(new_n):
        lo, hi = edges_new[k], edges_new[k + 1]
        width = hi - lo
        # Find overlap with old bins
        # Integral of h over [lo, hi]
        # h is piecewise constant on [edges_old[j], edges_old[j+1]] with value h[j]
        # Find old bins that overlap [lo, hi]
        j_start = max(0, int(np.searchsorted(edges_old, lo, side='right') - 1))
        j_end = int(np.searchsorted(edges_old, hi, side='left'))
        integral = 0.0
        for j in range(j_start, j_end):
            overlap_lo = max(lo, edges_old[j])
            overlap_hi = min(hi, edges_old[j + 1])
            if overlap_hi > overlap_lo:
                integral += h[j] * (overlap_hi - overlap_lo)
        h_new[k] = integral / width
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

def slsqp_epigraph(h_start, maxiter=200, tol=1e-12):
    """SLSQP epigraph formulation."""
    n = len(h_start)
    dx = 2.0 / n
    M = 2 * n - 1
    c0 = exact_score(h_start)
    t0 = c0 + 1e-4
    
    def obj(x):
        return x[-1]
    
    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    
    def eq_fun(x):
        return x[:-1].sum() - n / 2.0
    
    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g
    
    def ineq_fun(x):
        h = x[:-1]
        t = x[-1]
        profile = compute_profile(h)
        return t - profile
    
    def ineq_jac(x):
        h = x[:-1]
        G = build_gradient_rows(h, np.arange(M), n, dx)
        J = np.zeros((M, n + 1))
        J[:, :-1] = -G
        J[:, -1] = 1.0
        return J
    
    x0 = np.concatenate([h_start, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 2.0)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': tol})
        if res.x is not None and np.isfinite(res.x).all():
            h_new = repair_feasible(res.x[:-1])
            c_new = exact_score(h_new)
            if c_new <= c0 + 1e-8:
                return h_new
    except Exception:
        pass
    return None

import numpy as np

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx] + 1e-15)
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx] + 1e-15)
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def exact_score(h):
    return float(np.max(compute_profile(h)))

def build_gradient_rows(h, rows, n, dx):
    G = np.zeros((len(rows), n))
    for r, m in enumerate(rows):
        delta = int(m) - (n - 1)
        if delta == 0:
            G[r, :] = dx * (1.0 - 2.0 * h)
        elif delta > 0:
            G[r, delta:] += dx * (1.0 - h[:n - delta])
            G[r, :n - delta] += -dx * h[delta:]
        else:
            d = -delta
            G[r, :n - d] += dx * (1.0 - h[d:])
            G[r, d:] += -dx * h[:n - d]
    return G

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    edges_old = np.linspace(0, 2, n + 1)
    edges_new = np.linspace(0, 2, new_n + 1)
    h_new = np.zeros(new_n)
    for k in range(new_n):
        lo, hi = edges_new[k], edges_new[k + 1]
        width = hi - lo
        j_start = max(0, int(np.searchsorted(edges_old, lo, side='right')) - 1)
        j_end = int(np.searchsorted(edges_old, hi, side='left'))
        integral = 0.0
        for j in range(j_start, j_end):
            olo = max(lo, edges_old[j])
            ohi = min(hi, edges_old[j + 1])
            if ohi > olo:
                integral += h[j] * (ohi - olo)
        h_new[k] = integral / width
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

def lp_slp(h, max_iters=500, top_k=25):
    """LP-based successive linear programming."""
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best_c = exact_score(h)
    
    for it in range(max_iters):
        profile = compute_profile(h)
        c_cur = float(np.max(profile))
        if c_cur < best_c:
            best_c = c_cur
        
        # Active set: top-k lags
        top_k_actual = min(top_k, 2 * n - 1)
        active = np.argsort(profile)[-top_k_actual:]
        
        G = build_gradient_rows(h, active, n, dx)
        P_active = profile[active]
        
        # LP: min t s.t. G@(u-h) - t <= -P_active, 0<=u<=1, sum(u)=n/2
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        
        A_ub = np.zeros((top_k_actual, n + 1))
        A_ub[:, :n] = G
        A_ub[:, -1] = -1.0
        b_ub = P_active - G @ h
        
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([n / 2.0])
        
        bounds = [(0, 1)] * n + [(0, None)]
        
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        
        if not res.success:
            break
        
        u_lp = res.x[:n]
        d = u_lp - h
        d_norm = np.linalg.norm(d)
        
        if d_norm < 1e-12:
            break

        improved = False
        for alpha in [1.0, 0.8, 0.6, 0.4, 0.3, 0.2, 0.1, 0.05]:

                improved = True
