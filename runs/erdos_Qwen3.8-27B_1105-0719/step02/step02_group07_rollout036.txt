def project_feasible(x, target_sum):

def solve_1d_subproblem(corr, n, old, A, special_idx):
    """
    Solve min_{x in [0,1]} max(f_quad(x), max_k (corr[k] + (x-old)*A[k]))
    where f_quad is the special (k=special_idx, d=0) term:
      f_quad(x) = corr[special_idx] + 2*(x-old)*(1-x-old)
    
    Args:
        corr: current correlation values (length 2n-1)
        n: length of h
        old: current value of h[j]
        A: linear coefficients (length 2n-1)
        special_idx: index of the d=0 correlation (= n-1)
    
    Returns:
        x_opt: optimal value
    """
    # Coarse grid
    x_coarse = np.linspace(0.0, 1.0, 200)
    f_linear = (x_coarse - old)[:, np.newaxis] * A[np.newaxis, :] + corr[np.newaxis, :]
    f_quad = corr[special_idx] + 2.0 * (x_coarse - old) * (1.0 - x_coarse - old)
    f_linear[:, special_idx] = f_quad
    f_max = f_linear.max(axis=1)
    best_coarse = f_max.argmin()
    
    # Fine grid around best

    hi = x_coarse[min(len(x_coarse) - 1, best_coarse + 3)]
