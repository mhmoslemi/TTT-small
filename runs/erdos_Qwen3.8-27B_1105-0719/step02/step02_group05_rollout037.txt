from scipy.optimize import minimize, minimize_scalar
from scipy.special import logsumexp
import time


def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corrs = np.correlate(h, 1.0 - h, mode='full')
    return corrs.max() * dx, corrs


def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)


def build_h_from_q(q, n):
    m = n // 2
    if n % 2 == 0:
        return np.concatenate([q, 1.0 - q[::-1]])
    else:
        return np.concatenate([q, [0.5], 1.0 - q[::-1]])


def q_from_h(h, n):
    m = n // 2
    return h[:m].copy()


def grad_corr_d(h, d):
    n = len(h)
    g = np.zeros(n)
    if d >= 0:
        if d < n:
            g[d:] += (1.0 - h[:n - d])
            g[:n - d] += (-h[d:])
    else:
        dd = -d
        if dd < n:
            g[:n - dd] += (1.0 - h[dd:])
            g[dd:] += (-h[:n - dd])

    return np.correlate(h, 1.0 - h, mode='full')


def q_smooth_val_grad(q, n, beta):
    m = n // 2
    h = build_h_from_q(q, n)
    dx = 2.0 / n
    corrs = compute_corrs(h)
    val = logsumexp(beta * corrs) / beta * dx
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(len(corrs)) / len(corrs)
    else:
        w /= w_sum
    Gq = np.zeros(m)
    for k in range(len(corrs)):
        if w[k] < 1e-14:
            continue
        d = k - (n - 1)
        G = grad_corr_d(h, d)
        Gq += w[k] * (G[:m] - G[m:][::-1])
    return val, Gq * dx


def coord_1d_obj(v, q, i, n, corrs_base, delta_linear, delta_quad):
    val = corrs_base + delta_linear * v + delta_quad * v * v
    return val.max()


def coordinate_descent_polish(q, n, max_rounds=20, time_limit=60.0):
    m = n // 2
    dx = 2.0 / n
    t_start = time.time()
    h = build_h_from_q(q, n)
    corrs = compute_corrs(h)
    best_c5 = corrs.max() * dx
    q_best = q.copy()

    for rnd in range(max_rounds):
        if time.time() - t_start > time_limit:
            break
        improved = False
        for i in range(m):
            j = n - 1 - i
            q_i = q[i]
            h_i = q_i
            h_j = 1.0 - q_i

            # Compute coefficients for R_d(v) = base_d + L_d*v + Q_d*v^2
            delta_linear = np.zeros(2 * n - 1)
            delta_quad = np.zeros(2 * n - 1)

            # Term 1: j=i as first factor: delta_h_i * (1 - h[i+d])
            # Contributes to lag d: v * (1 - h[i+d])
            # In correlation indexing: lag d corresponds to index (n-1+d)
            if 0 <= i + n - 1 + (0) < 2 * n - 1:
                pass  # handled below

            # For each valid lag d:
            # Contribution from h[i] change (delta = v - q_i):
            #   (v - q_i) * (1 - h[i+d])  [if 0<=i+d<n]
            #   -h[i-d] * (v - q_i)       [if 0<=i-d<n]
            # Contribution from h[j] change (delta = -(v-q_i)):
            #   -(v-q_i) * (1-h[j+d])     [if 0<=j+d<n]
            #   +h[j-d] * (v-q_i)         [if 0<=j-d<n]
            #
            # But if i+d == j, then h[i+d] = h[j] = 1-q_i which is changing!
            # In that case: (v)*(1-(1-v)) = v*v, giving a quadratic term.
            # Similarly if j+d == i, or i-d == j, or j-d == i.

            q_orig_i = q[i]
            # Set q[i] = 0 temporarily to get "base"
            q[i] = 0.0
            h_base = build_h_from_q(q, n)
            corrs_base = compute_corrs(h_base)

            # Now compute delta_linear and delta_quad
            # When q[i] goes from 0 to v: h[i] goes 0->v, h[j] goes 1->1-v
            # delta_linear[d] such that R_d = corrs_base[d] + delta_linear[d]*v (+ delta_quad[d]*v^2)

            dl = np.zeros(2 * n - 1)
            dq_arr = np.zeros(2 * n - 1)

            # h[i] = v (was 0): contributes +v to h[i]
            # h[j] = 1-v (was 1): contributes -v to h[j]

            # For each position p with change, compute effect on each lag
            # Position i changes by +v:
            #   As first factor in R_d (lag d = i - s for second factor at s=i-d):
            #     +v * (1 - h_base[i+d]) for valid i+d
            #   As second factor in R_d (lag d = s - i for first factor at s=i-d):
            #     -v * h_base[i-d] for valid i-d

            # Position j changes by -v:
            #   As first factor: -v * (1 - h_base[j+d]) for valid j+d
            #   As second factor: +v * h_base[j-d] for valid j-d

            # Lag d in correlation corresponds to index (n-1+d) in the full array

            # Position i, as first factor: affects lag d where second factor is i+d
            #   R_{-(n-1-i-d)} ... actually let me use the correlation indexing
            #   corr[k] = sum_i h[i] * (1-h[i+k-n+1]) for k = 0..2n-2
            #   So corr[k] involves pairs (i, i+k-n+1)
            #   Lag d = k-n+1, so second factor index = i + d

            # Position i as first factor at "position i":
            #   For lag d: second factor at i+d (if valid)
            #   corr index = n-1+d
            #   Contribution: v * (1 - h_base[i+d])

            # Position i as second factor:
            #   For lag d: first factor at i-d (if valid)
            #   corr index = n-1+d
            #   Contribution: -v * h_base[i-d]  (because the term is h[i-d]*(1-h[i]) and h[i] increases by v)

            # Position j as first factor:
            #   For lag d: second factor at j+d
            #   Contribution: -v * (1 - h_base[j+d])

            # Position j as second factor:
            #   For lag d: first factor at j-d
            #   Contribution: +v * h_base[j-d]

            # Special cases (quadratic):
            # If i+d == j (d = j-i = n-1-2i): h[i+d] is changing, so term is v*(1-(1-v)) = v^2
            #   This replaces the linear term v*(1-h_base[j]) = v*(1-1) = 0
            #   So we add v^2 instead of 0
            # If j+d == i (d = i-j = 2i-n+1): h[j+d] is changing, term is -v*(1-(v)) = -v+v^2
            #   Replaces -v*(1-h_base[i]) = -v*0 = 0
            #   So we add -v + v^2... wait, h_base[i] = 0 (we set q[i]=0), so 1-h_base[i] = 1
            #   Original linear: -v*1 = -v. But actual: -v*(1-v) = -v+v^2. Delta: +v^2
            #   So add v^2 to quadratic, and the linear is already -v (h_base[i]=0, so 1-h_base[i]=1)
            #   Wait: the formula gives -v*(1-h_base[j+d]) = -v*(1-h_base[i]) = -v*(1-0) = -v
            #   Actual: -v*(1-v) = -v+v^2
            #   Difference: +v^2. So add to quadratic.

            # If i-d == j (d = i-j = 2i-n+1): h[i-d] is changing
            #   Term: -v * h[i-d] where h[i-d] = h[j] = 1-v (was 1)
            #   Formula: -v * h_base[i-d] = -v * h_base[j] = -v * 1 = -v
            #   Actual: -v * (1-v) = -v+v^2
            #   Difference: +v^2

            # If j-d == i (d = j-i = n-1-2i): h[j-d] is changing
            #   Term: +v * h[j-d] where h[j-d] = h[i] = v (was 0)
            #   Formula: +v * h_base[j-d] = +v * h_base[i] = +v * 0 = 0
            #   Actual: +v * v = v^2
            #   Difference: +v^2

            # Let me just compute these directly:
            def lag_idx(d):
                return n - 1 + d

            # Position i as first factor
            for d in range(0, n - i):
                # second factor at i+d
                k_idx = lag_idx(d)
                if j == i + d:
                    # h[i+d] = h[j] = 1-v (changing)
                    # Actual: v*(1-(1-v)) = v^2
                    dq_arr[k_idx] += 1.0
                else:
                    dl[k_idx] += (1.0 - h_base[i + d])

            for d in range(0, i + 1):
                # first factor at i-d, second factor at i
                k_idx = lag_idx(d)
                if j == i - d:
                    # h[i-d] = h[j] = 1-v
                    # Actual: -(1-v)*v ... wait
                    # The term is h[i-d]*(1-h[i]) = (1-v)*(1-v) = (1-v)^2
                    # Formula gives: -v*h_base[i-d] = -v*1 = -v (since h_base[j]=1)
                    # Full term: base + change. Base term: h_base[i-d]*(1-h_base[i]) = 1*1 = 1
                    # Actual: (1-v)^2 = 1-2v+v^2
                    # So change = -2v+v^2. Formula gives -v. Difference: -v+v^2
                    dl[k_idx] += -1.0
                    dq_arr[k_idx] += 1.0
                else:
                    dl[k_idx] += -h_base[i - d]

            # Position j as first factor
            for d in range(0, n - j):
                k_idx = lag_idx(d)
                if i == j + d:
                    # h[j+d] = h[i] = v
                    # Actual: -v*(1-v) = -v+v^2
                    # Formula: -v*(1-h_base[i]) = -v*1 = -v (h_base[i]=0)
                    # Difference: +v^2
                    dl[k_idx] += -1.0
                    dq_arr[k_idx] += 1.0
                else:
                    dl[k_idx] += -(1.0 - h_base[j + d])

            for d in range(0, j + 1):
                k_idx = lag_idx(d)
                if i == j - d:
                    # h[j-d] = h[i] = v
                    # Actual: +v*v = v^2
                    # Formula: +v*h_base[i] = 0
                    # Difference: +v^2
                    dq_arr[k_idx] += 1.0
                else:
                    dl[k_idx] += h_base[j - d]

            # Restore q[i]
            q[i] = q_i

            # Now optimize v in [0,1] to minimize max_d (corrs_base[d] + dl[d]*v + dq_arr[d]*v^2) * dx
            # Since dx is constant, minimize max_d (...)

            def obj_1d(v):
                vals = corrs_base + dl * v + dq_arr * v * v
                return vals.max()

            res = minimize_scalar(obj_1d, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-10})
            new_v = res.x
            new_c5 = obj_1d(new_v) * dx
            old_c5 = corrs.max() * dx

            if new_c5 < old_c5 - 1e-15:
                q[i] = new_v
                h = build_h_from_q(q, n)
                corrs = compute_corrs(h)
                improved = True

            if new_c5 * dx < best_c5:
                best_c5 = new_c5 * dx
                q_best = q.copy()

        if not improved:
            break

    return q_best, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)

    def remaining():
        return budget_s - (time.time() - t_start)

    # Initialize from parent
    h_parent = np.array(initial_h_values, dtype=float)
    n_parent = len(h_parent)
    target_sum = n_parent / 2.0
    h_parent = project_feasible(h_parent, target_sum)

    score_parent, _ = exact_score(h_parent)
    best_h = h_parent.copy()
    best_c5 = score_parent

    # Ensure even n for symmetry
    if n_parent % 2 != 0:
        n_even = n_parent + 1
        x_orig = np.linspace(0.0, 2.0, n_parent)
        x_new = np.linspace(0.0, 2.0, n_even)
        h_parent = np.interp(x_new, x_orig, h_parent)
        h_parent = project_feasible(h_parent, n_even / 2.0)
        n_parent = n_even

    n = n_parent
    m = n // 2
    q_best = q_from_h(best_h, n)

    # Phase 1: Continuation L-BFGS-B in q-space
    betas_phase1 = [50, 200, 800, 3000, 10000, 30000]
    for beta in betas_phase1:
        if remaining() < 50:
            break

        def q_val(x):
            return q_smooth_val_grad(x, n, beta)[0]
        def q_grad(x):
            return q_smooth_val_grad(x, n, beta)[1]

        try:
            res = minimize(q_val, q_best, jac=q_grad, method='L-BFGS-B',
                          bounds=[(0.0, 1.0)] * m,
                          options={'maxiter': 2000, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxls': 100})
            q_res = np.clip(res.x, 0.0, 1.0)
            h_res = build_h_from_q(q_res, n)
            s_res, _ = exact_score(h_res)
            if s_res < best_c5 - 1e-15:
                best_c5 = s_res
                best_h = h_res
                q_best = q_res.copy()
        except Exception:
            pass

    # Phase 2: Coordinate descent polish
    if remaining() > 80:
        t_cd_start = time.time()
        time_left_cd = min(remaining() - 20, 150)
        q_cd, c5_cd = coordinate_descent_polish(q_best, n, max_rounds=50,
                                                 time_limit=time_left_cd)
        if c5_cd < best_c5 - 1e-15:
            best_c5 = c5_cd
            best_h = build_h_from_q(q_cd, n)
            q_best = q_cd.copy()

    # Phase 3: More smooth optimization after CD
    for beta in [50000, 100000, 200000]:
        if remaining() < 40:
            break
        def q_val2(x):
            return q_smooth_val_grad(x, n, beta)[0]
        def q_grad2(x):
            return q_smooth_val_grad(x, n, beta)[1]
        try:
            res = minimize(q_val2, q_best, jac=q_grad2, method='L-BFGS-B',
                          bounds=[(0.0, 1.0)] * m,
                          options={'maxiter': 3000, 'ftol': 1e-18,
                                   'gtol': 1e-15, 'maxls': 100})
            q_res = np.clip(res.x, 0.0, 1.0)
            h_res = build_h_from_q(q_res, n)
            s_res, _ = exact_score(h_res)
            if s_res < best_c5 - 1e-15:
                best_c5 = s_res
                best_h = h_res
                q_best = q_res.copy()
        except Exception:
            pass

        # CD after each
        if remaining() > 60:
            q_cd2, c5_cd2 = coordinate_descent_polish(q_best, n, max_rounds=20,
                                                       time_limit=min(60, remaining()-20))
            if c5_cd2 < best_c5 - 1e-15:
                best_c5 = c5_cd2
                best_h = build_h_from_q(q_cd2, n)
                q_best = q_cd2.copy()

    # Phase 4: Grid size optimization
    n_cur = len(best_h)
    for n_try in [n_cur - 4, n_cur - 2, n_cur + 2, n_cur + 4, n_cur + 6,
                  n_cur + 8, n_cur - 6]:
        if n_try < 40 or n_try % 2 != 0:
            continue
        if remaining() < 100:
            break
        x_orig = np.linspace(0.0, 2.0, n_cur)
        x_new = np.linspace(0.0, 2.0, n_try)
        h_interp = np.interp(x_new, x_orig, best_h)
        h_interp = project_feasible(h_interp, n_try / 2.0)
        q_try = q_from_h(h_interp, n_try)
        m_try = n_try // 2

        for beta in [1000, 10000]:
            if remaining() < 40:
                break
            def qv(x):
                return q_smooth_val_grad(x, n_try, beta)[0]
            def qg(x):
                return q_smooth_val_grad(x, n_try, beta)[1]
            try:
                res = minimize(qv, q_try, jac=qg, method='L-BFGS-B',
                              bounds=[(0.0, 1.0)] * m_try,
                              options={'maxiter': 1000, 'ftol': 1e-17,
                                       'gtol': 1e-13, 'maxls': 80})
                q_try = np.clip(res.x, 0.0, 1.0)
            except Exception:
                pass

        # CD polish
        if remaining() > 40:
            q_try, c5_try = coordinate_descent_polish(q_try, n_try, max_rounds=10,
                                                       time_limit=30)
        else:
            c5_try = compute_corrs(build_h_from_q(q_try, n_try)).max() * (2.0/n_try)

        h_try = build_h_from_q(q_try, n_try)
        s_try, _ = exact_score(h_try)

        if s_try < best_c5 - 1e-12:
            best_c5 = s_try
            best_h = h_try
            n_cur = n_try

    # Phase 5: Random perturbation restarts
    n_cur = len(best_h)
    m_cur = n_cur // 2
    q_cur = q_from_h(best_h, n_cur)

    if remaining() > 100:
        pert_scales = [0.005, 0.002, 0.001, 0.0005, 0.0002]
        for ps in pert_scales:
            if remaining() < 50:
                break
            q_pert = q_cur + rng.randn(m_cur) * ps
            q_pert = np.clip(q_pert, 0.0, 1.0)

            for beta in [5000, 50000]:
                if remaining() < 30:
                    break
                def qv3(x):
                    return q_smooth_val_grad(x, n_cur, beta)[0]
                def qg3(x):
                    return q_smooth_val_grad(x, n_cur, beta)[1]
                try:
                    res = minimize(qv3, q_pert, jac=qg3, method='L-BFGS-B',
                                  bounds=[(0.0, 1.0)] * m_cur,
                                  options={'maxiter': 1000, 'ftol': 1e-17,
                                           'gtol': 1e-13, 'maxls': 80})
                    q_pert = np.clip(res.x, 0.0, 1.0)
                except Exception:
                    pass

            if remaining() > 30:
                q_pert, s_pert = coordinate_descent_polish(q_pert, n_cur,
                                                            max_rounds=10, time_limit=20)
            else:
                s_pert = compute_corrs(build_h_from_q(q_pert, n_cur)).max() * (2.0/n_cur)

            if s_pert < best_c5 - 1e-15:
                best_c5 = s_pert
                best_h = build_h_from_q(q_pert, n_cur)
                q_cur = q_pert.copy()

    # Phase 6: Final CD polish
    if remaining() > 30:
        n_final = len(best_h)
        q_final = q_from_h(best_h, n_final)
        time_left = max(10, remaining() - 10)
        q_f2, c5_f2 = coordinate_descent_polish(q_final, n_final, max_rounds=30,
                                                 time_limit=time_left)
        if c5_f2 < best_c5 - 1e-15:
            best_c5 = c5_f2
            best_h = build_h_from_q(q_f2, n_final)

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    n_out = len(best_h)
    ts_out = n_out / 2.0
    best_h = project_feasible(best_h, ts_out)
    final_score, _ = exact_score(best_h)
    if final_score < best_c5:
        best_c5 = final_score

    return best_h, best_c5, n_out
