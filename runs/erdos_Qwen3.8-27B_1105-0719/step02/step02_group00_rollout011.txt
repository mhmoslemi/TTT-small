def repair_feasible(h):
    n = len(h)
    h = np.clip(float64(h), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        # Need to add mass
        capacity = 1.0 - h
        idx = np.argsort(-capacity)
        remaining = diff
        for i in idx:
            add = min(remaining, capacity[i] + 1e-15)
            if add > 0:
                h[i] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        # Need to remove mass
        idx = np.argsort(h)
        remaining = -diff
        for i in idx:
            rem = min(remaining, h[i] + 1e-15)
            if rem > 0:
                h[i] -= rem
            remaining -= rem
            if remaining <= 1e-12:

    if abs(diff2) > 1e-13:
        if diff2 > 0:
            i = int(np.argmax(1.0 - h))
        else:
            i = int(np.argmax(h))
        h[i] += diff2
    return np.clip(h, 0.0, 1.0)

def build_grad_and_profile(h):
    """Build the full gradient matrix and profile for all lags."""
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    profile = np.correlate(h, 1 - h, mode="full") * dx
    G = np.zeros((M, n))
    center = n - 1
    for m in range(M):
        d = m - center
        if d == 0:
            G[m, :] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            # P_d = dx * sum_{i=0}^{n-d-1} h[i+d]*(1-h[i])

            valid = n - d

            G[m, d:] += dx * (1.0 - h[:valid])

            valid = n - e
            G[m, :valid] += dx * (1.0 - h[valid:n])
            G[m, e:] += -dx * h[:valid]
    return profile, G

    """Fast active-set gradient descent."""
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best_c = exact_score(h)
    alpha = 0.05
    velocity = np.zeros(n)
    
    for it in range(max_iters):
        profile, G = build_grad_and_profile(h)
        c = np.max(profile)
        
        # Active set
        thresh = c - max(0.003, 0.05 * (c - np.median(profile)))
        active = np.where(profile >= thresh)[0]
        if len(active) == 0:
            active = np.array([np.argmax(profile)])
        if len(active) > top_k:
            top_idx = np.argsort(profile[active])[-top_k:]
            active = active[top_idx]
        
        # Weighted gradient (weight by how close to max)
        w = profile[active] - thresh + 0.005
        w = np.maximum(w, 0.01)
        w /= w.sum()
        grad = G[active, :].T @ w
        grad -= grad.mean()  # preserve sum constraint
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad /= gnorm
        
        # Line search
        improved = False
        for a in [alpha, alpha*0.5, alpha*0.25, alpha*0.1, alpha*0.05]:

                break

            if alpha < 1e-7:

    return h, best_c

def cvx_linear_step(h, top_k=20, trust_radius=0.05):
    """CVXPY linearized subproblem step."""
    n = len(h)
    dx = 2.0 / n
    profile, G = build_grad_and_profile(h)
    c = np.max(profile)
    
    # Active set
    thresh = c - max(0.003, 0.05 * (c - np.median(profile)))
    active = np.where(profile >= thresh)[0]
    if len(active) == 0:
        active = np.array([np.argmax(profile)])
    if len(active) > top_k:
        top_idx = np.argsort(profile[active])[-top_k:]
        active = active[top_idx]
    
    # Build LP: min t s.t. t >= P_m(x) + G_m(x)^T (z - x) for m in active
    z = cp.Variable(n)
    t = cp.Variable()
    
    constraints = [
        cp.sum(z) == n / 2.0,
        z >= 0,
        z <= 1,
        z >= h - trust_radius,
        z <= h + trust_radius,
    ]
    
    # Linearized active constraints: t >= profile[m] + G[m,:] @ (z - h)
    G_active = G[active, :]
    p_active = profile[active]
    for i in range(len(active)):
        constraints.append(t >= p_active[i] + G_active[i, :] @ (z - h))
    
    obj = cp.Minimize(t + 1e-6 * cp.norm1(z - h))
    prob = cp.Problem(obj, constraints)
    
    try:
        prob.solve(solver=cp.ECOS, max_time=10)
        if z.value is not None and np.isfinite(z.value).all():
            h_new = repair_feasible(z.value)
            s = exact_score(h_new)
            return h_new, s
    except Exception:
        pass
    
    # Try with CLARABEL or SCS

        if z.value is not None and np.isfinite(z.value).all():
            h_new = repair_feasible(z.value)
            s = exact_score(h_new)
            return h_new, s
    except Exception:
        pass

    xs = (np.arange(n) + 0.5) * (2.0 / n)

    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

def symmetrize_anti(h):
    """Symmetrize: h[i] + h[n-1-i] = 1"""
    n = len(h)
    h_sym = np.zeros(n)
    for i in range(n):
        j = n - 1 - i
        h_sym[i] = (h[i] + (1.0 - h[j])) / 2.0
    # Average for double-counted pairs
    h_sym = (h_sym + 1.0 - h_sym[::-1]) / 2.0
    return repair_feasible(h_sym)

    return repair_feasible((h + h[::-1]) / 2.0)

import numpy as np

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def repair_feasible(h):
    n = len(h)
    h = np.clip(np.asarray(h, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        idx = np.argsort(-capacity)
        remaining = diff
        for i in idx:
            add = min(remaining, capacity[i] + 1e-15)
            if add > 0:
                h[i] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        idx = np.argsort(h)
        remaining = -diff
        for i in idx:
            rem = min(remaining, h[i] + 1e-15)
            if rem > 0:
                h[i] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-13:
        if diff2 > 0:
            i = int(np.argmax(1.0 - h))
        else:
            i = int(np.argmax(h))
        h[i] += diff2
    return np.clip(h, 0.0, 1.0)

def build_grad_active(h, active_m):
    """Build gradients only for specified lag indices (0 to 2n-2)."""
    n = len(h)
    dx = 2.0 / n
    center = n - 1
    G = np.zeros((len(active_m), n))
    for row, m in enumerate(active_m):
        d = m - center
        if d == 0:
            G[row, :] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            valid = n - d
            G[row, :valid] += -dx * h[d:n]
            G[row, d:] += dx * (1.0 - h[:valid])
        else:
            e = -d
            valid = n - e
            G[row, :valid] += dx * (1.0 - h[valid:n])
            G[row, e:] += -dx * h[:valid]
    return G

def active_descent(h, max_iters=400, top_k=12, step0=0.06):
    """Active-set gradient descent with adaptive step."""
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    alpha = step0
    for it in range(max_iters):
        profile = compute_profile(h)
        c = np.max(profile)
        spread = c - np.median(profile)
        thresh = c - max(0.002, 0.06 * spread)
        active = np.where(profile >= thresh)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(profile))])
        if len(active) > top_k:
            top_idx = np.argsort(profile[active])[-top_k:]
            active = active[top_idx]
        w = profile[active] - thresh + 0.003
        w = np.maximum(w, 0.01)
        w /= w.sum()
        G_act = build_grad_active(h, active)
        grad = G_act.T @ w
        grad -= grad.mean()
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad /= gnorm
        improved = False
        for a in [alpha, alpha * 0.5, alpha * 0.25, alpha * 0.125]:
            h_try = repair_feasible(h - a * grad)
            s = exact_score(h_try)
            if s < best_c - 1e-14:
                h = h_try
                best_c = s
                alpha = min(alpha * 1.03, 0.12)
                improved = True
                break
        if not improved:
            alpha *= 0.65
            if alpha < 1e-8:
                break
    return h, best_c

def cvx_step(h, top_k=20, radius=0.04, max_time=8):
    """CVXPY linearized minimax step."""
    n = len(h)
    profile = compute_profile(h)
    c = np.max(profile)
    spread = c - np.median(profile)
    thresh = c - max(0.003, 0.06 * spread)
    active = np.where(profile >= thresh)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(profile))])
    if len(active) > top_k:
        top_idx = np.argsort(profile[active])[-top_k:]
        active = active[top_idx]
    G_act = build_grad_active(h, active)
    p_act = profile[active]
    z = cp.Variable(n)
    t = cp.Variable()
    cons = [
        cp.sum(z) == n / 2.0,
        z >= 0.0,
        z <= 1.0,
        z >= h - radius,
        z <= h + radius,
    ]
    for i in range(len(active)):
        cons.append(t >= p_act[i] + G_act[i, :] @ (z - h))
    obj = cp.Minimize(t + 1e-5 * cp.norm1(z - h))
    prob = cp.Problem(obj, cons)
    for solver in [cp.ECOS, cp.SCS, cp.CLARABEL]:
        try:
            prob.solve(solver=solver, max_time=max_time)
            if z.value is not None and np.isfinite(z.value).all():
                h_new = repair_feasible(z.value)
                s = exact_score(h_new)
                if s < c + 1e-10:
                    return h_new, s
        except Exception:
            continue
    return None, None

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

    """Circular shift (simulates translation in x)."""
    n = len(h)
    h_shifted = np.roll(h, k % n)
    return repair_feasible(h_shifted)
