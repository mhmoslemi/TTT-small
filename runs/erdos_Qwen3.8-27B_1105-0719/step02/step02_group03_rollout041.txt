```python
        interior = (h > 1e-10) & (h < 1.0 - 1e-10)
        if interior.sum() > 0:
            h[interior] += diff / interior.sum()
        else:
            h += diff / n
    return np.clip(h, 0.0, 1.0)

def c5_score(h):
    dx = 2.0 / len(h)
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def all_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def grad_corr_single(h, e, n):
    """Gradient of correlation at offset e (where e = k - (n-1)) with respect to h."""
    grad = np.zeros(n)
    i_lo = max(0, -e)
    i_hi = min(n - 1, n - 1 - e)
    if i_lo > i_hi:
        return grad
    # Term 1: j in [i_lo, i_hi], contrib (1 - h[j+e])
    grad[i_lo:i_hi+1] += (1.0 - h[i_lo + e:i_hi + 1 + e])
    # Term 2: j-e in [i_lo, i_hi], contrib -h[j-e]
    j_lo = max(0, i_lo + e)
    j_hi = min(n - 1, i_hi + e)
    if j_lo <= j_hi:
        grad[j_lo:j_hi+1] += -h[j_lo - e:j_hi + 1 - e]
    return grad

def smooth_max_value(h, beta):
    corrs = all_corrs(h)
    m = corrs.max()
    if beta < 1e6 * m + 1:  # avoid overflow
        return m + np.log(np.sum(np.exp(beta * (corrs - m)))) / beta
    return m

def smooth_max_grad(h, beta):
    n = len(h)
    corrs = all_corrs(h)
    m = corrs.max()
    shifted = beta * (corrs - m)
    # Clip for numerical stability
    shifted = np.clip(shifted, -500, 500)
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(len(corrs))
        w[np.argmax(corrs)] = 1.0
    else:
        w /= w_sum
    grad = np.zeros(n)
    for k in range(len(corrs)):
        if w[k] > 1e-12:
            e = k - (n - 1)
            grad += w[k] * grad_corr_single(h, e, n)
    return grad

def slsqp_refine(h0, dx, target_sum, beta, maxiter, time_left):
    n = len(h0)
    def obj(x):
        return smooth_max_value(x, beta) * dx
    def obj_grad(x):
        return smooth_max_grad(x, beta) * dx
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        h_out = project_feasible(res.x, target_sum)
    except Exception:
        h_out = project_feasible(h0, target_sum)
    return h_out, c5_score(h_out)

def binary_delta(h, i, j, n):
    """Compute delta in all correlations when h[i]:1->0, h[j]:0->1."""
    # This is expensive to compute for all k, so we compute it vectorized
    ms = np.arange(-(n - 1), n)  # offsets e
    # When h[i] goes from 1 to 0:
    #   For offset e: contributes to c_e at position i (as h[i]=(1-h[i+e])) -> change: -(1-h[i+e])
    #   For offset -e: h[i] appears as h[j] where j = i+e_orig, so e_orig = -e, h[i] = h[j+e_orig]
    #     contributes -h[i-e_orig]... this is getting confusing.
    # Let me use the direct formula.
    
    # c_k = sum_i h[i]*(1-h[i+e]) where e = k-(n-1)
    # Change in c_k when h[i] changes from 1 to 0:
    #   If i is in the "first" position: loss of (1-h[i+e]) for each valid e
    #   If i is in the "second" position (i = j' + e for some j'): gain of h[j'] for each valid e
    # Similarly for h[j] going from 0 to 1:
    #   If j is in the "first" position: gain of (1-h[j+e])
    #   If j is in the "second" position: loss of h[j-e]
    
    # Vectorized: for each offset e (indexed by k from 0 to 2n-2):
    delta = np.zeros(2 * n - 1)
    for k in range(2 * n - 1):
        e = k - (n - 1)
        # h[i] going 1->0:
        # As first factor: if i_lo <= i <= i_hi, contrib -(1-h[i+e])
        i_lo = max(0, -e)
        i_hi = min(n - 1, n - 1 - e)
        d = 0.0
        if i_lo <= i <= i_hi:
            d -= (1.0 - h[i + e])
        # As second factor: if i_lo <= i-e <= i_hi, contrib -(-h[i-e]) = +h[i-e]
        # Wait: the term is h[i']*(1-h[i'+e]). If i = i'+e, then i' = i-e.
        # The term is h[i-e]*(1-h[i]). When h[i] goes 1->0, (1-h[i]) goes 0->1.
        # So the term changes from h[i-e]*0 to h[i-e]*1, i.e., increases by h[i-e].
        if i_lo <= (i - e) <= i_hi:
            d += h[i - e]
        # h[j] going 0->1:
        # As first factor: if i_lo <= j <= i_hi, contrib +(1-h[j+e])
        if i_lo <= j <= i_hi:
            d += (1.0 - h[j + e])
        # As second factor: if i_lo <= j-e <= i_hi, the term is h[j-e]*(1-h[j]).
        # When h[j] goes 0->1, (1-h[j]) goes 1->0. Term changes from h[j-e]*1 to h[j-e]*0.
        # So decreases by h[j-e].
        if i_lo <= (j - e) <= i_hi:
            d -= h[j - e]
        delta[k] = d
    return delta
```

```python
def binary_steepest(h0, target_sum, max_sweeps, time_limit):
    n = len(h0)
    dx = 2.0 / n
    h = np.zeros(n)
    sorted_idx = np.argsort(-h0)
    h[sorted_idx[:int(target_sum)]] = 1.0
    h = project_feasible(h, target_sum)  # ensure feasibility
    
    corrs = all_corrs(h)
    best_c5 = corrs.max() * dx
    best_h = h.copy()
    t_start = time.time()
    
    for sweep in range(max_sweeps):
        if time.time() - t_start > time_limit:
            break
        ones = np.where(h > 0.999)[0]
        zeros = np.where(h < 0.001)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        best_swap_max = corrs.max()
        best_swap = None
        for i in ones:
            for j in zeros:
                if i == j:
                    continue
                delta = binary_delta_fast(h, int(i), int(j), n)
                new_max = (corrs + delta).max()
                if new_max < best_swap_max - 1e-14:
                    best_swap_max = new_max
                    best_swap = (i, j)
        
        if best_swap is None:
            break
        i, j = best_swap
        h[int(i)] = 0.0
        h[int(j)] = 1.0
        corrs = all_corrs(h)
        c5_val = corrs.max() * dx
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h.copy()
    
    return best_h, best_c5

    zeros = np.where(h < 0.001)[0]
    no_improve = 0
    temp = temp0
    
    for it in range(max_iters):
        if time.time() - t_start > time_limit:
            break
        if it % 3000 == 0:
            temp *= 0.999
        
        i = int(ones[rng.randint(len(ones))])
        j = int(zeros[rng.randint(len(zeros))])
        if i == j:
            continue
        
        delta = binary_delta_fast(h, i, j, n)
        new_corrs = corrs + delta
        new_max = new_corrs.max()
        new_c5 = new_max * dx
        
        if new_c5 <= best_c5 + 1e-15:
            h[i], h[j] = 0.0, 1.0
            corrs = new_corrs
            best_c5 = new_c5
            best_h = h.copy()
            ones = np.where(h > 0.999)[0]
```

```python
    ms = np.arange(-(n-1), n)
    n_km1 = 2*n - 1

    # Effect of flipping h[j] from 0 to 1 (call it pos_effect[j])
    # These are the same formula, just the sign pattern differs
    
    # For position p, the effect of setting it to 0 (if it was 1):
    #   c_k change = -(1-h[p+e]) [as first factor] + h[p-e] [as second factor]
    # For position p, the effect of setting it to 1 (if it was 0):
    #   c_k change = +(1-h[p+e]) [as first factor] - h[p-e] [as second factor]
    
    # So pos_effect[p] = -neg_effect[p]
    # And delta_pair(i,j) = neg_effect[i] + pos_effect[j] + cross = neg_effect[i] - neg_effect[j] + cross
    
    # Precompute neg_effect for all positions
```

```python
            break
        ones = np.where(h > 0.999)[0].astype(int)
        zeros = np.where(h < 0.001)[0].astype(int)
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Precompute effects for all positions
        # effects[p, k] = change in c_k when h[p] goes 1->0
        ms = np.arange(-(n-1), n)
        effects = np.zeros((n, n_km1))
        for p in range(n):
            v1 = (ms >= -p) & (ms <= n - 1 - p)
            v2 = (ms >= p - (n - 1)) & (ms <= p)
            effects[p, v1] = -(1.0 - h[p + ms[v1]])
            effects[p, v2] += h[p - ms[v2]]
        
        best_swap_max = corrs.max()
        best_swap = None
        
        for i in ones:
            for j in zeros:
                # delta = effects[i] - effects[j] + 2 at index (i-j)+(n-1)
                delta = effects[i] - effects[j]
                k_cross = (i - j) + (n - 1)
                if 0 <= k_cross < n_km1:
                    delta[k_cross] += 2.0
                new_max = (corrs + delta).max()
                if new_max < best_swap_max - 1e-14:
                    best_swap_max = new_max
                    best_swap = (i, j)
        
        if best_swap is None:
            break
        i, j = best_swap
        h[i] = 0.0
        h[j] = 1.0
        corrs = all_corrs(h)
        c5_val = corrs.max() * dx
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h.copy()
    
    return best_h, best_c5
```

Now let me also think about the **subgradient descent** method. The idea is:
1. Compute all correlations c_k
2. Find the max M and the set of active indices A = {k : c_k > M - tol}
3. Compute the gradient g = (1/|A|) * sum_{k in A} grad_c_k
4. Take a step: h_new = h - eta * g
5. Project to feasible set
6. If no improvement after several steps, reduce eta

The key advantage over the smooth max is that it directly targets the maximum without approximation error.

However, there's a subtlety: the subgradient of max is not unique at points where multiple c_k achieve the maximum. The "average over active set" is a valid subgradient, but it might not be the most effective direction. An alternative is to use the "most negative" subgradient (the one that gives the largest decrease in the objective), but that requires a QP solve.

For simplicity, I'll use the average subgradient with backtracking line search.

Let me also think about a **hybrid approach**:
- Use the smooth max with moderate beta (say 100-500) for the first few hundred iterations
- Then switch to the exact subgradient method for the remaining iterations

This combines the smoothness of the logarithmic approach with the exactness of the subgradient method.

Now, let me also consider the **multi-bit flip** idea for binary search. Instead of just swapping one 1 and one 0, try swapping 2 ones with 2 zeros (a 4-bit flip). This allows the search to escape from binary local optima that are stuck due to the restriction to pair swaps.

For a 4-bit flip (i1, i2 → 0, j1, j2 → 1), the delta in correlations is:
delta = effects[i1] + effects[i2] - effects[j1] - effects[j2] + cross_terms

The cross terms are more complex (6 pairs of positions can interact), but for most pairs the cross term is zero. We can compute them case by case.

However, the number of 4-bit flips is much larger: C(64,2) * C(64,2) ≈ 10M for n=126. That's too many to try exhaustively. We'd need a heuristic to select which 4-bit flips to try.

For now, let me stick with pair swaps (2-bit flips) and rely on the simulated annealing to escape local optima.

Let me also add one more idea: **coordinate descent**. At each step, for each coordinate i, try the optimal value of h[i] given all other coordinates fixed. The optimal h[i] is:
- If the gradient of the max w.r.t. h[i] is negative, set h[i] as high as possible (but ≤ 1)
- If positive, set as low as possible (but ≥ 0)
- Then adjust the sum constraint.

Actually, this is more like a block coordinate method. For each i, the conditional optimization is:
min_{h[i]} max_k c_k(h) subject to the sum constraint (which couples all coordinates).

This is hard to do exactly, but we can approximate it by:
- Tentatively set h[i] to 0 and 1, compute the resulting max correlation
- Choose the better one (or an interpolation)

This is essentially what the binary search does, but coordinate-wise instead of pair-wise.

I think the pair-wise binary search is more effective because it maintains the sum constraint exactly.
