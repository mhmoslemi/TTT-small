def bundle_step(h_cur, active_lags, n, dx, max_lp_iter=100):
    """Solve the LP subproblem and take a step."""
    m = n - 1
    corrs = corr_vals(h_cur)
    
    # Build the LP
    # Variables: [t, h_0, ..., h_{n-1}]
    # Minimize: t
    # Subject to:
    #   -t + sum_k g_k[i] * h_i <= corrs[k] - g_k^T h_cur  for each active k
    #   -h_i <= 0  for all i
    #   h_i <= 1  for all i
    #   sum(h_i) = n/2
    
    n_vars = n + 1
    c = np.zeros(n_vars)
    c[0] = 1.0  # minimize t
    
    A_ub = np.zeros((len(active_lags) + 2*n, n_vars))
    b_ub = np.zeros(len(active_lags) + 2*n)
    
    for i, k in enumerate(active_lags):
        g_k = grad_corr_k(h_cur, n, dx, k)
        # -t + g_k^T h <= corrs[k] - g_k^T h_cur
        A_ub[i, 0] = -1.0
        A_ub[i, 1:] = g_k
        b_ub[i] = corrs[k] - g_k @ h_cur
    
    # -h_i <= 0
    A_ub[len(active_lags):len(active_lags)+n, 1:] = -np.eye(n)
    # h_i <= 1
    A_ub[len(active_lags)+n:, 1:] = np.eye(n)
    b_ub[len(active_lags)+n:] = 1.0
    
    A_eq = np.zeros((1, n_vars))
    A_eq[0, 1:] = 1.0
    b_eq = np.array([n / 2.0])
    
    # Bounds: t in [0, 2], h_i in [0, 1]
    bounds = [(0, 2)] + [(0, 1)] * n
    
    from scipy.optimize import linprog
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                  bounds=bounds, method='highs')
    
    if res.status != 0:
        return None, None
    
    h_lp = res.x[1:]
    t_lp = res.x[0]
    return h_lp, t_lp

                    continue

                # h[i] ranges from max(0, c-1) to min(1, c)

                f_c = exact_score(h)

    h = h.copy()
    best_c5 = exact_score(h)
    rng = np.random.RandomState(12345)
    for sweep in range(max_sweeps):
        improved = False
        tried = 0
        while tried < max_pairs:
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                tried += 1
                continue
            c = h[i] + h[j]
            if c > 1.0 + 1e-12 or c < -1e-12:
                tried += 1
                continue
            lo = max(0.0, c - 1.0)
            hi = min(1.0, c)
            if hi - lo < 1e-10:
                tried += 1
                continue
            phi = (1.0 + 5.0**0.5) / 2.0
            a, b = lo, hi
            c_gn = b - (b - a) / phi
            d_gn = a + (b - a) / phi
            h[i] = c_gn; h[j] = c - c_gn
            f_c = exact_score(h)
            h[i] = d_gn; h[j] = c - d_gn
            f_d = exact_score(h)
            for _ in range(20):
                if f_c < f_d:
                    b = d_gn; d_gn = c_gn; f_d = f_c
                    c_gn = b - (b - a) / phi
                    h[i] = c_gn; h[j] = c - c_gn
                    f_c = exact_score(h)
                else:
                    a = c_gn; c_gn = d_gn; f_c = f_d
                    d_gn = a + (b - a) / phi
                    h[i] = d_gn; h[j] = c - d_gn
                    f_d = exact_score(h)
            if f_c < f_d:
                h[i] = c_gn; h[j] = c - c_gn
                final = f_c
            else:
                h[i] = d_gn; h[j] = c - d_gn
                final = f_d
            if final < best_c5 - 1e-12:
                best_c5 = final
                improved = True
            tried += 1
        if not improved:
            break
    return h, exact_score(h)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def remaining():
        return budget_s - (time.time() - t0)

    try:
        h_parent = np.array(initial_h_values, dtype=float)
    except NameError:
        h_parent = 0.5 * np.ones(126)

    n = len(h_parent)
    if n % 2 != 0:
        h_parent = h_parent[:n - 1]
        n = len(h_parent)

    h_best = project_feasible(h_parent, n / 2.0)
    c5_best = exact_score(h_best)
    m = n // 2
    q_best = h_best[:m].copy()
    print(f"Initial: n={n}, C5={c5_best:.7f}")

    # Phase 1: Symmetric smooth SLSQP with increasing beta
    for beta in [80, 150, 300, 600, 1200, 2000]:
        if remaining() < 30:
            break
        try:
            res = minimize(smooth_sym_obj, q_best,
                           jac=lambda x: smooth_sym_grad(x, n, beta),
                           method='SLSQP',
                           bounds=[(0.0, 1.0)] * m,
                           options={'maxiter': 600, 'ftol': 1e-18})
            h_new = make_sym_h(res.x, n)
            c5_new = exact_score(h_new)
            if c5_new < c5_best - 1e-11:
                c5_best = c5_new
                q_best = res.x.copy()
                h_best = h_new
                print(f"  sym smooth beta={beta}: C5={c5_best:.7f}")
        except Exception:
            pass

    # Phase 2: Symmetric epigraph SLSQP with multiple active sets
    for n_act in [30, 50, 70, 90, 110]:
        if remaining() < 20:
            break
        h_cur = make_sym_h(q_best, n)
        corrs = corr_vals(h_cur)
        active = active_lags_top(corrs, min(n_act, 2 * n - 1))
        t_init = corrs.max() * 1.00001
        x0 = np.concatenate([[t_init], q_best])
        cons = [{'type': 'ineq',
                 'fun': lambda x: epi_sym_con_fun(x, n, active),
                 'jac': lambda x: epi_sym_con_jac(x, n, active)}]
        bounds = [(0.0, 2.0)] + [(0.0, 1.0)] * m
        try:
            res = minimize(lambda x: x[0], x0,
                           jac=lambda x: epi_sym_jac(x),
                           method='SLSQP', bounds=bounds,
                           constraints=cons,
                           options={'maxiter': 800, 'ftol': 1e-18})
            q_new = res.x[1:]
            h_new = make_sym_h(q_new, n)
            if abs(h_new.sum() - n / 2.0) > 1e-8:
                continue
            c5_new = exact_score(h_new)
            if c5_new < c5_best - 1e-11:
                c5_best = c5_new
                q_best = q_new.copy()
                h_best = h_new
                print(f"  epi sym (act={n_act}): C5={c5_best:.7f}")
        except Exception:
            pass

    # Phase 3: Full-space smooth SLSQP (break symmetry)
    if remaining() > 40:
        h_full = h_best.copy()
        for beta in [200, 500, 1000]:
            if remaining() < 15:
                break
            try:
                res = minimize(lambda x: smooth_full_obj(x, len(x), beta), h_full,
                               jac=lambda x: smooth_full_grad(x, len(x), beta),
                               method='SLSQP',
                               bounds=[(0.0, 1.0)] * n,
                               options={'maxiter': 500, 'ftol': 1e-18})
                h_new = project_feasible(res.x, n / 2.0)
                c5_new = exact_score(h_new)
                if c5_new < c5_best - 1e-11:
                    c5_best = c5_new
                    h_best = h_new
                    h_full = h_new
                    q_best = h_best[:m].copy()
                    print(f"  full smooth beta={beta}: C5={c5_best:.7f}")
            except Exception:
                pass

    # Phase 4: Full-space epigraph SLSQP
    for n_act in [40, 60, 80]:
        if remaining() < 25:
            break
        h_cur = h_best
        corrs = corr_vals(h_cur)
        active = active_lags_top(corrs, min(n_act, 2 * n - 1))
        t_init = corrs.max() * 1.00001
        x0 = np.concatenate([[t_init], h_cur])
        cons_ineq = [{'type': 'ineq',
                      'fun': lambda x: epi_full_con_fun(x, n, active, n / 2.0),
                      'jac': lambda x: epi_full_con_jac(x, n, active)}]
        cons_eq = [{'type': 'eq',
                    'fun': lambda x: x[1:].sum() - n / 2.0,
                    'jac': lambda x: np.concatenate([np.zeros(1), np.ones(n)])}]
        all_cons = cons_ineq + cons_eq
        bounds = [(0.0, 2.0)] + [(0.0, 1.0)] * n
        try:
            res = minimize(lambda x: x[0], x0,
                           jac=lambda x: epi_full_jac(x),
                           method='SLSQP', bounds=bounds,
                           constraints=all_cons,
                           options={'maxiter': 600, 'ftol': 1e-18})
            h_new = project_feasible(res.x[1:], n / 2.0)
            c5_new = exact_score(h_new)
            if c5_new < c5_best - 1e-11:
                c5_best = c5_new
                h_best = h_new
                q_best = h_best[:m].copy()
                print(f"  epi full (act={n_act}): C5={c5_best:.7f}")
        except Exception:
            pass

    # Phase 5: Paired coordinate descent polish
    if remaining() > 30:
        h_cd, c5_cd = paired_cd_polish(h_best, n, max_sweeps=5, max_pairs=500)
        if c5_cd < c5_best - 1e-12:
            c5_best = c5_cd
            h_best = h_cd
            q_best = h_best[:m].copy()
            print(f"  paired CD: C5={c5_best:.7f}")

    # Phase 6: Epigraph with wider active set from best
    for n_act in [100, 120, 140, 160]:
        if remaining() < 15:
            break
        h_cur = h_best
        corrs = corr_vals(h_cur)
        active = active_lags_top(corrs, min(n_act, 2 * n - 1))
        t_init = corrs.max() * 1.000001
        q_cur = h_cur[:m].copy()
        x0 = np.concatenate([[t_init], q_cur])
        cons = [{'type': 'ineq',
                 'fun': lambda x: epi_sym_con_fun(x, n, active),
                 'jac': lambda x: epi_sym_con_jac(x, n, active)}]
        bounds = [(0.0, 2.0)] + [(0.0, 1.0)] * m
        try:
            res = minimize(lambda x: x[0], x0,
                           jac=lambda x: epi_sym_jac(x),
                           method='SLSQP', bounds=bounds,
                           constraints=cons,
                           options={'maxiter': 1000, 'ftol': 1e-18})
            q_new = res.x[1:]
            h_new = make_sym_h(q_new, n)
            if abs(h_new.sum() - n / 2.0) > 1e-8:
                continue
            c5_new = exact_score(h_new)
            if c5_new < c5_best - 1e-11:
                c5_best = c5_new
                q_best = q_new.copy()
                h_best = h_new
                print(f"  epi wide (act={n_act}): C5={c5_best:.7f}")
        except Exception:
            pass

    # Phase 7: More coordinate descent after epigraph
    if remaining() > 20:
        h_cd, c5_cd = paired_cd_polish(h_best, n, max_sweeps=3, max_pairs=400)
        if c5_cd < c5_best - 1e-12:
            c5_best = c5_cd
            h_best = h_cd
            print(f"  paired CD 2: C5={c5_best:.7f}")

    # Phase 8: Try different n values
    if remaining() > 50:
        for delta in [2, -2, 4, -4, 6]:
            n_try = n + delta
            if n_try < 40 or n_try % 2 != 0 or remaining() < 25:
                continue
            x_old = np.linspace(0, 2, n, endpoint=False)
            x_new_pt = np.linspace(0, 2, n_try, endpoint=False)
            h_interp = np.interp(x_new_pt, x_old, h_best)
            h_interp = project_feasible(h_interp, n_try / 2.0)
            m_try = n_try // 2
            q_try = h_interp[:m_try].copy()
            try:
                res = minimize(smooth_sym_obj, q_try,
                               jac=lambda x: smooth_sym_grad(x, n_try, 300),
                               method='SLSQP',
                               bounds=[(0.0, 1.0)] * m_try,
                               options={'maxiter': 400, 'ftol': 1e-16})
                h_new = make_sym_h(res.x, n_try)
                c5_new = exact_score(h_new)
                if c5_new < c5_best - 1e-10:
                    h_cur2 = make_sym_h(res.x, n_try)
                    corrs = corr_vals(h_cur2)
                    active = active_lags_top(corrs, 60)
                    t_init = corrs.max() * 1.00001
                    x0 = np.concatenate([[t_init], res.x])
                    cons = [{'type': 'ineq',
                             'fun': lambda x: epi_sym_con_fun(x, n_try, active),
                             'jac': lambda x: epi_sym_con_jac(x, n_try, active)}]
                    bounds2 = [(0.0, 2.0)] + [(0.0, 1.0)] * m_try
                    res2 = minimize(lambda x: x[0], x0,
                                    jac=lambda x: epi_sym_jac(x),
                                    method='SLSQP', bounds=bounds2,
                                    constraints=cons,
                                    options={'maxiter': 500, 'ftol': 1e-16})
                    c5_new2 = exact_score(make_sym_h(res2.x[1:], n_try))
                    if c5_new2 < c5_best - 1e-10:
                        c5_best = c5_new2
                        n = n_try
                        m = m_try
                        h_best = make_sym_h(res2.x[1:], n)
                        q_best = res2.x[1:].copy()
                        print(f"  n={n}: C5={c5_best:.7f}")
            except Exception:
                pass

    # Phase 9: Final symmetric epigraph with very tight tolerance
    if remaining() > 15:
        h_cur = h_best
        corrs = corr_vals(h_cur)
        active = active_lags_top(corrs, min(80, 2 * n - 1))
        t_init = corrs.max() * 1.0000001
        q_cur = h_cur[:m].copy()
        x0 = np.concatenate([[t_init], q_cur])
        cons = [{'type': 'ineq',
                 'fun': lambda x: epi_sym_con_fun(x, n, active),
                 'jac': lambda x: epi_sym_con_jac(x, n, active)}]
        bounds = [(0.0, 2.0)] + [(0.0, 1.0)] * m
        try:
            res = minimize(lambda x: x[0], x0,
                           jac=lambda x: epi_sym_jac(x),
                           method='SLSQP', bounds=bounds,
                           constraints=cons,
                           options={'maxiter': 2000, 'ftol': 1e-20})
            q_new = res.x[1:]
            h_new = make_sym_h(q_new, n)
            c5_new = exact_score(h_new)
            if c5_new < c5_best - 1e-12:
                c5_best = c5_new
                h_best = h_new
                print(f"  final epi tight: C5={c5_best:.7f}")
        except Exception:
            pass

    # Final coordinate descent
    if remaining() > 10:
        h_cd, c5_cd = paired_cd_polish(h_best, n, max_sweeps=2, max_pairs=300)
        if c5_cd < c5_best - 1e-12:
            c5_best = c5_cd
            h_best = h_cd
            print(f"  final CD: C5={c5_best:.7f}")

    # Final validation
    h_best = np.clip(h_best, 0.0, 1.0)
    h_best = project_feasible(h_best, len(h_best) / 2.0)
    c5_best = exact_score(h_best)
    print(f"Final: n={len(h_best)}, C5={c5_best:.7f}")

    return h_best, c5_best, len(h_best)
