```python
import numpy as np
from scipy.optimize import linprog
import time


def score_exact(h):
    n = len(h)
    if n < 2:
        return float('inf')
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))


def repair_feasible(vec):
    vec = np.asarray(vec, dtype=np.float64)
    n = len(vec)
    if n == 0:
        return np.array([0.5])
    h = np.clip(vec, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity, kind='mergesort')
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx])
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-14:
                break
    else:
        order = np.argsort(h, kind='mergesort')
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-14:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h


def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx


def build_gradient(h):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    G = np.zeros((M, n))
    for r in range(M):
        d = r - (n - 1)
        if d == 0:
            G[r, :] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            G[r, 0:n-d] = -dx * h[d:n]
            G[r, d:n] = G[r, d:n] + dx * (1.0 - h[0:n-d])
        else:
            m = -d
            G[r, 0:n-m] = G[r, 0:n-m] + dx * (1.0 - h[m:n])
            G[r, m:n] = G[r, m:n] - dx * h[0:n-m]
    return G


def lp_direction(h, P, G, active_idx, alpha_tol=1e-10):
    n = len(h)
    nv = 2 * n + 1
    c_obj = np.zeros(nv)
    c_obj[-1] = -1.0

    A_eq = np.zeros((2, nv))
    A_eq[0, 0:n] = 1.0
    A_eq[0, n:2*n] = -1.0
    A_eq[1, 0:2*n] = 1.0
    b_eq = np.array([0.0, 1.0])

    n_active = len(active_idx)
    n_ub = n_active + 2 * n
    A_ub = np.zeros((n_ub, nv))
    b_ub = np.zeros(n_ub)

    row = 0
    for m in active_idx:
        A_ub[row, 0:n] = G[m, :]
        A_ub[row, n:2*n] = -G[m, :]
        A_ub[row, -1] = 1.0
        row += 1

    for i in range(n):
        A_ub[row, i] = 1.0
        A_ub[row, n+i] = -1.0
        b_ub[row] = 1.0 - h[i]
        row += 1
        A_ub[row, i] = -1.0
        A_ub[row, n+i] = 1.0
        b_ub[row] = h[i]
        row += 1

    bounds = [(0, None)] * nv

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0 and np.isfinite(res.fun):
            alpha = max(0.0, -res.fun)
            if alpha < alpha_tol:
                return None, alpha
            d = res.x[0:n] - res.x[n:2*n]
            norm_d = np.sum(np.abs(d))
            if norm_d < 1e-10:
                return None, alpha
            d = d / norm_d
            return d, alpha
        else:
            return None, 0.0
    except Exception:
        return None, 0.0


def lp_polish(h, time_budget, max_iters, rng):
    t0 = time.perf_counter()
    n = len(h)
    h = h.copy()
    best_c = score_exact(h)
    best_h = h.copy()
    stall_count = 0
    beam = 0.008

    for it in range(max_iters):
        if time.perf_counter() - t0 > time_budget:
            break

        P = compute_profile(h)
        c = np.max(P)
        if c < best_c - 1e-14:
            best_c = c
            best_h = h.copy()
        elif stall_count > 15:
            break

        spread = max(0.004, 0.12 * (c - np.median(P)))
        thresh = c - beam
        active = np.where(P >= thresh)[0]
        if len(active) == 0:
            active = np.array([np.argmax(P)])
        if len(active) > 25:
            top_order = np.argsort(P[active])
            active = active[top_order[-25:]]

        G = build_gradient(h)
        d, alpha = lp_direction(h, P, G, active)

        if d is None:
            w = P[active] - thresh
            w = np.maximum(w, 0.005)
            w = w / w.sum()
            grad = G[active, :].T @ w
            grad -= grad.mean()
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break
            d = -grad / gnorm

        improved = False
        scales = [0.4, 0.25, 0.15, 0.08, 0.04, 0.02, 0.01, 0.005, 0.002]
        for scale in scales:
            h_try = repair_feasible(h + scale * d)

            if s < best_c - 1e-13:
                h = h_try
                best_c = s

                improved = True
                break

            stall_count = 0
            beam = max(0.002, beam * 0.92)
        else:
            stall_count += 1
            beam = min(0.02, beam * 1.3)
            if stall_count > 5:
                z = rng.randn(n)
                z -= z.mean()
                z /= np.linalg.norm(z) + 1e-15

                    h = h_try
                    best_c = s
                    best_h = h.copy()
                    stall_count = 0
                elif stall_count > 20:

    return best_h, best_c


def interpolate_mass(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    dx_old = 2.0 / n
    dx_new = 2.0 / new_n
    h_new = np.zeros(new_n)
    for j in range(new_n):
        x0 = j * dx_new
        x1 = (j + 1) * dx_new
        total = 0.0
        overlap = 0.0
        for i in range(n):
            ox0 = i * dx_old
            ox1 = (i + 1) * dx_old
            s = max(x0, ox0)
            e = min(x1, ox1)
            if e > s:
                total += h[i] * (e - s)
                overlap += (e - s)
        h_new[j] = total / overlap if overlap > 0 else 0.5
    return repair_feasible(np.clip(h_new, 0.0, 1.0))


def symmetrize(h):
    return repair_feasible((h + h[::-1]) / 2.0)


def make_block_seed(n, n_blocks, rng):
    h = np.zeros(n)
    block_size = n // n_blocks
    for b in range(n_blocks):
        start = b * block_size
        end = min((b + 1) * block_size, n)
        h[start:end] = rng.uniform(0.1, 0.9)
    return repair_feasible(h)


def make_binary_seed(n, rng):
    vals = rng.randn(n)
    threshold = np.median(vals)
    h = (vals > threshold).astype(np.float64)
    return repair_feasible(h)


def make_fourier_seed(n, freq1, freq2, amp1, amp2, phase, rng):
    x = np.arange(n, dtype=np.float64)
    h = 0.5 + amp1 * np.sin(2.0 * np.pi * freq1 * x / n + phase) + amp2 * np.sin(2.0 * np.pi * freq2 * x / n)
    h = np.clip(h, 0.0, 1.0)
    return repair_feasible(h)

            z = rng.randn(n_b)

            tb = min(12.0, time_left() - 8)
            h_lp, c_lp = lp_polish(h_pert, tb, 50, rng)
            if c_lp < best_c - 1e-12:
                best_h = h_lp
                best_c = c_lp
                n = len(best_h)

    # Stage 5: Block and pattern seeds
    if time_left() > 100:
        n_b = len(best_h)
        seeds = []
        for nb in [2, 3, 4, 5, 6]:
            seeds.append(make_block_seed(n_b, nb, rng))
        for _ in range(3):
            seeds.append(make_binary_seed(n_b, rng))
        for fr in [1, 2, 3, 5]:
            seeds.append(make_fourier_seed(n_b, fr, fr * 2, 0.2, 0.1, rng.uniform(0, 6.28), rng))

        for h_seed in seeds:
            if time_left() < 15:
                break
            c_seed = score_exact(h_seed)
            if c_seed > best_c + 0.05:
                continue
            tb = min(10.0, time_left() - 8)
            h_lp, c_lp = lp_polish(h_seed, tb, 40, rng)
            if c_lp < best_c - 1e-12:
                best_h = h_lp
                best_c = c_lp
                n = len(best_h)

    # Stage 6: Second dimension exploration from improved best
    if time_left() > 100:
        n_cur = len(best_h)
        dims2 = []
        for d in range(max(34, n_cur - 16), min(130, n_cur + 24) + 1, 2):
            if d != n_cur and d >= 34:
                dims2.append(d)
        dims2.sort(key=lambda d: (abs(d - n_cur), d))

        for new_n in dims2[:5]:
            if time_left() < 20:
                break
            h_interp = interpolate_mass(best_h, new_n)
            tb = min(15.0, time_left() - 10)
            h_lp, c_lp = lp_polish(h_interp, tb, 60, rng)
            if c_lp < best_c - 1e-12:
                best_h = h_lp
                best_c = c_lp
                n = len(best_h)
        # Symmetrize and polish
        if time_left() > 20:
            h_sym2 = symmetrize(best_h)
            c_sym2 = score_exact(h_sym2)
            if c_sym2 < best_c - 1e-12:
                best_h = h_sym2
                best_c = c_sym2
            tb = min(10.0, time_left() - 5)
            h_lp, c_lp = lp_polish(best_h, tb, 40, rng)
            if c_lp < best_c - 1e-12:
                best_h = h_lp
                best_c = c_lp

    # Stage 7: Final extended LP polish
    if time_left() > 40:
        tb = min(30.0, time_left() - 10)
        h_lp, c_lp = lp_polish(best_h, tb, 200, rng)
        if c_lp < best_c - 1e-12:
            best_h = h_lp
            best_c = c_lp

    # Stage 8: Very fine local search with tiny steps
    if time_left() > 20:
        n_f = len(best_h)
        dx_f = 2.0 / n_f
        for _ in range(200):
            if time.perf_counter() - t_start > budget_s - 5:
                break
            P_f = compute_profile(best_h)
            c_f = np.max(P_f)
            thresh_f = c_f - 0.003
            active_f = np.where(P_f >= thresh_f)[0]
            if len(active_f) > 12:
                top_f = np.argsort(P_f[active_f])[-12:]
                active_f = active_f[top_f]
            G_f = build_gradient(best_h)
            w_f = P_f[active_f] - thresh_f
            w_f = np.maximum(w_f, 0.005)
            w_f = w_f / w_f.sum()
            grad_f = G_f[active_f, :].T @ w_f
            grad_f -= grad_f.mean()
            gnorm_f = np.linalg.norm(grad_f)
            if gnorm_f < 1e-14:
                break
            d_f = -grad_f / gnorm_f
            found = False
            for sc in [0.002, 0.001, 0.0005, 0.0002, 0.0001]:
                h_try = repair_feasible(best_h + sc * d_f)
                s_try = score_exact(h_try)
                if s_try < best_c - 1e-14:
                    best_h = h_try
                    best_c = s_try
                    found = True
                    break
            if not found:
                d_f2 = lp_direction(best_h, P_f, G_f, active_f)
                if d_f2 is not None:
                    for sc in [0.001, 0.0005, 0.0002]:
                        h_try = repair_feasible(best_h + sc * d_f2[0])
                        s_try = score_exact(h_try)
                        if s_try < best_c - 1e-14:
                            best_h = h_try
                            best_c = s_try
                            found = True
                            break
            if not found:
                break

    # Final validation
    best_h = repair_feasible(best_h)
    final_c = score_exact(best_h)
    if not np.isfinite(final_c) or final_c > 1.0:
        best_h = repair_feasible(np.full(max(34, n), 0.5))
        final_c = score_exact(best_h)

    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            if np.isfinite(eval_result) and float(eval_result) < final_c:
                final_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 1:
            if np.isfinite(eval_result[0]) and float(eval_result[0]) < final_c:
                final_c = float(eval_result[0])
    except Exception:
        pass

    return best_h, float(final_c), int(len(best_h))
```
