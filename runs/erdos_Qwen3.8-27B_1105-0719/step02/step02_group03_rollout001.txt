    grad[idx] -= h[idx + offset]
    return grad

def smooth_max_grad(h, dx, beta):

        return smooth_max_grad(x, dx, beta)

        return x.sum() - target_sum

        return np.ones(n)

    delta = np.zeros(2 * n - 1)
    idx_i1 = i - ms[valid_i1]
    delta[valid_i1] += -(1.0 - h[idx_i1])
    idx_i2 = i + ms[valid_i2]
    delta[valid_i2] += h[idx_i2]
    idx_j1 = j - ms[valid_j1]
    delta[valid_j1] += (1.0 - h[idx_j1])
    idx_j2 = j + ms[valid_j2]
    delta[valid_j2] += -h[idx_j2]
    return delta


def binary_steepest_descent(h, dx, max_sweeps=100, time_limit=60.0):
    t0 = time.time()
    n = len(h)
    target_sum = int(round(n / 2))
    h = np.zeros(n)
    sorted_idx = np.argsort(-h)
    h[sorted_idx[:target_sum]] = 1.0
    corrs = compute_corrs(h)
    best_c5 = corrs.max() * dx
    best_h = h.copy()

    for sweep in range(max_sweeps):
        if time.time() - t0 > time_limit:
            break
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        best_swap = None
        best_new_max = corrs.max()
        for i in ones:
            for j in zeros:
                delta = compute_delta_binary(h, i, j, n)
                new_max = (corrs + delta).max()
                if new_max < best_new_max - 1e-14:
                    best_new_max = new_max
                    best_swap = (i, j)
        if best_swap is None:
            break
        i, j = best_swap
        h[i], h[j] = 0.0, 1.0
        corrs = compute_corrs(h)
        new_c5 = corrs.max() * dx
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
    return best_h, best_c5


def binary_sa(h, dx, rng, max_iters=200000, patience=6000, time_limit=120.0):
    t0 = time.time()
    n = len(h)
    target_sum = int(round(n / 2))
    h = np.zeros(n)
    sorted_idx = np.argsort(-h)
    h[sorted_idx[:target_sum]] = 1.0
    corrs = compute_corrs(h)
    best_c5 = corrs.max() * dx
    best_h = h.copy()
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    no_improve = 0
    temp = 0.03
    cooling = 0.9998

    for it in range(max_iters):
        if time.time() - t0 > time_limit:
            break
        if it % 3000 == 0:
            temp *= cooling
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        delta = compute_delta_binary(h, i, j, n)
        new_corrs = corrs + delta
        new_c5 = new_corrs.max() * dx
        if new_c5 <= best_c5 + 1e-15:
            h[i], h[j] = 0.0, 1.0
            corrs = new_corrs
            best_c5 = new_c5
            best_h = h.copy()
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            no_improve = 0
        else:
            delta_val = new_c5 - best_c5
            if temp > 1e-9 and rng.random() < np.exp(-delta_val / temp):
                h[i], h[j] = 0.0, 1.0
                corrs = new_corrs
                ones = np.where(h == 1.0)[0]
                zeros = np.where(h == 0.0)[0]
                no_improve = 0
            else:
                no_improve += 1
        if no_improve > patience:
            break
    return best_h, best_c5


def binary_block_search(h, dx, rng, max_iters=50000, patience=3000, time_limit=90.0):
    """Try block moves: move a contiguous block of 1s to a new location."""
    t0 = time.time()
    n = len(h)
    target_sum = int(round(n / 2))
    h = np.zeros(n)
    sorted_idx = np.argsort(-h)
    h[sorted_idx[:target_sum]] = 1.0
    corrs = compute_corrs(h)
    best_c5 = corrs.max() * dx
    best_h = h.copy()
    no_improve = 0
    temp = 0.02

    for it in range(max_iters):
        if time.time() - t0 > time_limit:
            break
        if it % 2000 == 0:
            temp *= 0.999
        # Try a single swap (faster) mixed with occasional block
        if rng.random() < 0.7:
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            delta = compute_delta_binary(h, i, j, n)
            new_corrs = corrs + delta
            new_c5 = new_corrs.max() * dx
            if new_c5 <= best_c5 + 1e-15:
                h[i], h[j] = 0.0, 1.0
                corrs = new_corrs
                best_c5 = new_c5
                best_h = h.copy()
                no_improve = 0
            else:
                delta_val = new_c5 - best_c5
                if temp > 1e-9 and rng.random() < np.exp(-delta_val / temp):
                    h[i], h[j] = 0.0, 1.0
                    corrs = compute_corrs(h)
                    no_improve = 0
                else:
                    no_improve += 1
        else:
            # Block move: pick a 1 and try to swap with a 0, 
            # also consider neighbor swaps
            ones = np.where(h == 1.0)[0]
            i = ones[rng.randint(len(ones))]
            # Try swapping i with i+1, i-1, or random
            candidates = []
            if i + 1 < n and h[i+1] == 0:
                candidates.append(i + 1)
            if i - 1 >= 0 and h[i-1] == 0:
                candidates.append(i - 1)
            if not candidates:
                zeros = np.where(h == 0.0)[0]
                candidates = [zeros[rng.randint(len(zeros))]]
            j = candidates[rng.randint(len(candidates))]
            if h[j] == 1.0:
                continue
            delta = compute_delta_binary(h, i, j, n)
            new_corrs = corrs + delta
            new_c5 = new_corrs.max() * dx
            if new_c5 <= best_c5 + 1e-15:
                h[i], h[j] = 0.0, 1.0
                corrs = new_corrs
                best_c5 = new_c5
                best_h = h.copy()
                no_improve = 0
            else:
                delta_val = new_c5 - best_c5
                if temp > 1e-9 and rng.random() < np.exp(-delta_val / temp):
                    h[i], h[j] = 0.0, 1.0
                    corrs = compute_corrs(h)
                    no_improve = 0
                else:
                    no_improve += 1

        if no_improve > patience:
            break
    return best_h, best_c5


def make_wave_start(n, target_sum, freq, phase=0.0):
    x = np.linspace(0, 2.0, n)
    h = 0.5 + 0.4 * np.sin(2 * np.pi * freq * x / 2.0 + phase)
    return project_feasible(h, target_sum)


def make_golden_start(n, target_sum, rng, mode=0):
    h = np.zeros(n)
    ts = int(target_sum)
    if mode == 0:
        phi = (1.0 + np.sqrt(5.0)) / 2.0
        for k in range(ts):
            pos = int((k * phi * n) % n)
            h[pos] = 1.0
    elif mode == 1:
        for k in range(ts):
            pos = int((k * 1.6180339887 * n) % n)
            h[pos] = 1.0
    elif mode == 2:
        for k in range(ts):
            pos = int((k * k * 7 + 3 * k + 1) % n)
            h[pos] = 1.0
    else:
        idx = rng.permutation(n)
        h[idx[:ts]] = 1.0
    return project_feasible(h, target_sum)


def remaining_time(t0, budget_s):
    return budget_s - (time.time() - t0)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    # Initialize from parent
    best_h = np.array(initial_h_values, dtype=float)
    n = len(best_h)
    dx = 2.0 / n
    target_sum = n / 2.0
    best_h = project_feasible(best_h, target_sum)
    best_c5 = c5_of_h(best_h, dx)

    # Phase 1: LP-based local search on the continuous solution
    if remaining_time(t0, budget_s) > 100:
        h_lp, c5_lp = lp_local_search(best_h, dx, target_sum, max_iters=400, time_limit=90.0)
        if c5_lp < best_c5:
            best_c5 = c5_lp
            best_h = h_lp

    # Phase 2: SLSQP polish with high beta
    if remaining_time(t0, budget_s) > 80:
        h_sq, c5_sq = slsqp_polish(best_h, dx, target_sum, beta=500, maxiter=600)
        if c5_sq < best_c5:
            best_c5 = c5_sq
            best_h = h_sq

    # Phase 3: Binary steepest descent from current continuous
    if remaining_time(t0, budget_s) > 120:
        h_bd, c5_bd = binary_steepest_descent(best_h, dx, max_sweeps=80, time_limit=60.0)
        if c5_bd < best_c5:
            best_c5 = c5_bd
            best_h = h_bd

    # Phase 4: LP from best binary (continuous relaxation)
    if remaining_time(t0, budget_s) > 100:
        h_lp2, c5_lp2 = lp_local_search(best_h, dx, target_sum, max_iters=300, time_limit=80.0)
        if c5_lp2 < best_c5:
            best_c5 = c5_lp2
            best_h = h_lp2
        if remaining_time(t0, budget_s) > 50:
            h_bd2, c5_bd2 = binary_steepest_descent(best_h, dx, max_sweeps=60, time_limit=40.0)
            if c5_bd2 < best_c5:
                best_c5 = c5_bd2
                best_h = h_bd2

    # Phase 5: Binary SA with block moves
    if remaining_time(t0, budget_s) > 150:
        h_sa, c5_sa = binary_block_search(best_h, dx, rng, max_iters=200000, patience=5000, time_limit=120.0)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa

    # Phase 6: LP polish again
    if remaining_time(t0, budget_s) > 80:
        h_lp3, c5_lp3 = lp_local_search(best_h, dx, target_sum, max_iters=300, time_limit=70.0)
        if c5_lp3 < best_c5:
            best_c5 = c5_lp3
            best_h = h_lp3
        if remaining_time(t0, budget_s) > 40:
            h_bd3, c5_bd3 = binary_steepest_descent(best_h, dx, max_sweeps=50, time_limit=30.0)
            if c5_bd3 < best_c5:
                best_c5 = c5_bd3
                best_h = h_bd3

    # Phase 7: Try different initializations at same n
    for mode in range(3):
        if remaining_time(t0, budget_s) < 120:
            break
        h_init = make_golden_start(n, target_sum, rng, mode)
        h_opt, c5_val = lp_local_search(h_init, dx, target_sum, max_iters=200, time_limit=40.0)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_opt
        else:
            h_bd, c5_bd = binary_steepest_descent(h_init, dx, max_sweeps=50, time_limit=30.0)
            if c5_bd < best_c5:
                best_c5 = c5_bd
                best_h = h_bd

    # Phase 8: Try wave starts
    for freq in [3, 5, 7]:
        if remaining_time(t0, budget_s) < 80:
            break
        h_wave = make_wave_start(n, target_sum, freq)
        h_opt, c5_val = lp_local_search(h_wave, dx, target_sum, max_iters=150, time_limit=30.0)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_opt
            if remaining_time(t0, budget_s) > 50:
                h_bd, c5_bd = binary_steepest_descent(h_opt, dx, max_sweeps=40, time_limit=30.0)
                if c5_bd < best_c5:
                    best_c5 = c5_bd
                    best_h = h_bd

    # Phase 9: Try n=120 (smaller, potentially better per-sample)
    if remaining_time(t0, budget_s) > 150:
        n2 = 120
        dx2 = 2.0 / n2
        ts2 = n2 / 2.0
        x_orig = np.linspace(0.0, 2.0, n + 1)[:-1]
        x_new = np.linspace(0.0, 2.0, n2)
        h_interp = np.interp(x_new, x_orig, best_h)
        h_interp = project_feasible(h_interp, ts2)
        h_opt2, c5_val2 = lp_local_search(h_interp, dx2, ts2, max_iters=250, time_limit=60.0)
        if c5_val2 < best_c5:
            best_c5 = c5_val2
            best_h = h_opt2
            n = n2
        if remaining_time(t0, budget_s) > 80:
            h_bd4, c5_bd4 = binary_steepest_descent(best_h, 2.0 / len(best_h), max_sweeps=50, time_limit=40.0)
            if c5_bd4 < best_c5:
                best_c5 = c5_bd4
                best_h = h_bd4

    # Phase 10: Try n=132
    if remaining_time(t0, budget_s) > 150:
        n3 = 132
        dx3 = 2.0 / n3
        ts3 = n3 / 2.0
        x_orig = np.linspace(0.0, 2.0, len(best_h) + 1)[:-1]
        x_new = np.linspace(0.0, 2.0, n3)
        h_interp = np.interp(x_new, x_orig, best_h)
        h_interp = project_feasible(h_interp, ts3)
        h_opt3, c5_val3 = lp_local_search(h_interp, dx3, ts3, max_iters=250, time_limit=60.0)
        if c5_val3 < best_c5:
            best_c5 = c5_val3
            best_h = h_opt3
        if remaining_time(t0, budget_s) > 60:
            dx_cur = 2.0 / len(best_h)
            h_bd5, c5_bd5 = binary_steepest_descent(best_h, dx_cur, max_sweeps=50, time_limit=40.0)
            if c5_bd5 < best_c5:
                best_c5 = c5_bd5
                best_h = h_bd5

    # Phase 11: Final aggressive LP + SLSQP
    if remaining_time(t0, budget_s) > 60:
        n_f = len(best_h)
        dx_f = 2.0 / n_f
        ts_f = n_f / 2.0
        h_f, c5_f = lp_local_search(best_h, dx_f, ts_f, max_iters=400, time_limit=40.0)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f
        if remaining_time(t0, budget_s) > 30:
            h_f2, c5_f2 = slsqp_polish(best_h, dx_f, ts_f, beta=1000, maxiter=500)
            if c5_f2 < best_c5:
                best_c5 = c5_f2
                best_h = h_f2

    # Phase 12: Final binary steepest descent
    if remaining_time(t0, budget_s) > 30:
        dx_f = 2.0 / len(best_h)
        h_fb, c5_fb = binary_steepest_descent(best_h, dx_f, max_sweeps=40, time_limit=20.0)
        if c5_fb < best_c5:
            best_c5 = c5_fb
            best_h = h_fb

    # Final validation
    n_b = len(best_h)
    dx_b = 2.0 / n_b
    ts_b = n_b / 2.0
    best_h = project_feasible(best_h, ts_b)
    best_c5 = c5_of_h(best_h, dx_b)

    return best_h, best_c5, int(n_b)
