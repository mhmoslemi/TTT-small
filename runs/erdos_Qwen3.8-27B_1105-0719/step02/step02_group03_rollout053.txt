```python
    G = np.zeros((2*n-1, n))
    for k in range(2*n-1):

        idx = np.arange(n)

        j1 = idx + offset
        m1 = (j1 >= 0) & (j1 < n)
        G[k, m1] += (1.0 - h[j1[m1]])

        j2 = idx - offset
        m2 = (j2 >= 0) & (j2 < n)
        G[k, m2] += (-h[j2[m2]])
    return G

def lp_direction(h, R, G, n):
    # Solve LP: min t s.t. G@d + R <= t, sum(d)=0, -h <= d <= 1-h
    # Variables: d (n) + t (1) = n+1
    # Constraints:
    #   G@d - t*1 <= -R (i.e., G@d + R <= t) -> G@d - t <= -R
    #   sum(d) = 0
    #   -h <= d <= 1-h
    
    c = np.zeros(n+1)
    c[n] = 1.0  # minimize t
    
    # Inequality: G@d - t <= -R  ->  G @ [d;0] + [0; -1] * t <= -R
    A_ub = np.zeros((2*n-1, n+1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -R
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = [0.0]
    
    # Bounds: -h[i] <= d[i] <= 1-h[i], t >= 0
    bounds = [(-h[i], 1-h[i]) for i in range(n)]
    bounds.append((0, None))
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        d = res.x[:n]
        return d
    return None

def line_search(h, d, dx, n, max_alpha=1.0):
    # Find best alpha in [0, max_alpha]
    best_alpha = 0.0
    best_val = c5_of_h(h, dx)
    
    # Compute alpha_max from bounds
    alpha_max = max_alpha
    pos = d > 1e-15
    neg = d < -1e-15
    if pos.any():
        alpha_max = min(alpha_max, ((1-h)[pos] / d[pos]).min())
    if neg.any():
        alpha_max = min(alpha_max, (h[neg] / (-d[neg])).min())
    alpha_max = max(0, min(1.0, alpha_max))
    
    # Try a few alphas
    for alpha in [alpha_max, alpha_max*0.9, alpha_max*0.7, alpha_max*0.5, alpha_max*0.25]:
        if alpha <= 0:
            break
        h_try = h + alpha * d
        h_try = project_feasible(h_try, n/2)
        val = c5_of_h(h_try, dx)
        if val < best_val - 1e-12:
            best_val = val
            best_alpha = alpha
    
    return best_alpha, best_val, h + best_alpha * d if best_alpha > 0 else h
```

```python
def compute_addition_matrix(h, n):
    """For each position j (where h[j]=0), compute addition_j[dd] for all dd."""
    n_zeros = n - int(h.sum())
    zeros = np.where(h == 0.0)[0]
    # addition_j[dd] = (1-h[j+dd])*v(j+dd) - h[j-dd]*v(j-dd)
    # = v(j+dd) - h[j+dd]*v(j+dd) - h[j-dd]*v(j-dd)
    # Since h[j]=0, v(j+dd) is just a bounds indicator
    
    dd = np.arange(-(n-1), n)  # all offsets
    # For each j in zeros, compute addition[j_idx, :] 
    # addition[j_idx, dd_idx] = v(j+dd)*1 - h[j+dd]*v(j+dd) - h[j-dd]*v(j-dd)
    # = v(j+dd)*(1-h[j+dd]) - h[j-dd]*v(j-dd)
    
    # Vectorize: for all j and all dd
    # j + dd: shape (n_zeros, 2n-1)
    j_plus = zeros[:, None] + dd[None, :]  # (n_zeros, 2n-1)
    j_minus = zeros[:, None] - dd[None, :]  # (n_zeros, 2n-1)
    
    v_plus = ((j_plus >= 0) & (j_plus < n)).astype(float)
    v_minus = ((j_minus >= 0) & (j_minus < n)).astype(float)
    
    h_plus = np.zeros_like(v_plus)
    valid_plus = (j_plus >= 0) & (j_plus < n)
    h_plus[valid_plus] = h[j_plus[valid_plus]]
    
    h_minus = np.zeros_like(v_minus)
    valid_minus = (j_minus >= 0) & (j_minus < n)
    h_minus[valid_minus] = h[j_minus[valid_minus]]
    
    addition = v_plus * (1.0 - h_plus) - h_minus * v_minus
    return addition, zeros

def compute_removal_vector(h, i, n):
    """For position i (where h[i]=1), compute removal[dd] for all dd."""
    dd = np.arange(-(n-1), n)
    i_plus = i + dd
    i_minus = i - dd
    
    v_plus = ((i_plus >= 0) & (i_plus < n)).astype(float)
    v_minus = ((i_minus >= 0) & (i_minus < n)).astype(float)
    
    h_plus = np.zeros(2*n-1)
    valid_plus = (i_plus >= 0) & (i_plus < n)
    h_plus[valid_plus] = h[i_plus[valid_plus]]
    
    h_minus = np.zeros(2*n-1)
    valid_minus = (i_minus >= 0) & (i_minus < n)
    h_minus[valid_minus] = h[i_minus[valid_minus]]
    
    # removal[dd] = -(1-h[i+dd])*v(i+dd) + h[i-dd]*v(i-dd)
    removal = -(1.0 - h_plus) * v_plus + h_minus * v_minus
    return removal
```
