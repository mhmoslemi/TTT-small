import numpy as np
from scipy.optimize import minimize, linprog
import time

def exact_score(h):
    n = len(h)
    if n == 0:
        return float('inf')
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def repair_feasible(vec):
    vec = np.asarray(vec, dtype=np.float64)
    n = len(vec)
    h = np.clip(vec, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity, kind='mergesort')
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, max(capacity[idx], 0.0) + 1e-15)
            if add > 0:
                h[idx] += add
                capacity[idx] -= add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h, kind='mergesort')
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx] + 1e-15)
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    # Final fix
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
            h[idx] = min(1.0, h[idx] + diff2)
        else:
            idx = int(np.argmax(h))
            h[idx] = max(0.0, h[idx] + diff2)
        h = np.clip(h, 0.0, 1.0)
    return h

def build_gradient(h, m_list):
    """Build gradient rows for given profile indices."""
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    G = np.zeros((len(m_list), n))
    for idx, m in enumerate(m_list):
        d = m - (n - 1)
        if d == 0:
            G[idx, :] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            # c[d] = sum_{k=d}^{n-1} h[k]*(1-h[k-d])
            # d/dc/dh[i] = (1-h[i-d]) if i >= d, else 0
            #              - h[i+d] if i < n-d, else 0
            if d < n:
                G[idx, d:] += dx * (1.0 - h[:n-d])
                G[idx, :n-d] += -dx * h[d:]
        else:
            dp = -d
            if dp < n:
                # c[-dp] = sum_{k=0}^{n-dp-1} h[k]*(1-h[k+dp])
                # d/dc/dh[i] = (1-h[i+dp]) if i < n-dp, else 0
                #              - h[i-dp] if i >= dp, else 0
                G[idx, :n-dp] += dx * (1.0 - h[dp:])
                G[idx, dp:] += -dx * h[:n-dp]
    return G

def lp_direction(h, profile, active_set, max_active=30):
    """Solve LP to find best descent direction."""
    n = len(h)
    # Select top active constraints
    active = np.array(active_set)
    if len(active) > max_active:
        # Keep only the top ones
        order = np.argsort(profile[active])[::-1]
        active = active[order[:max_active]]
    
    G = build_gradient(h, active)
    
    # LP variables: u[0..n-1], v[0..n-1], alpha
    # x = [u, v, alpha], length 2n+1
    nv = 2 * n + 1
    
    # Objective: minimize -alpha
    c_obj = np.zeros(nv)
    c_obj[-1] = -1.0
    
    # Inequality constraints: A_ub @ x <= b_ub
    # 1) For each active row m: G[m,:] @ (u-v) + alpha <= 0
    #    [G[m,:], -G[m,:], 1] @ x <= 0
    A_ub_list = []
    b_ub_list = []
    for m_idx in range(len(active)):
        row = np.zeros(nv)
        row[:n] = G[m_idx]
        row[n:2*n] = -G[m_idx]
        row[2*n] = 1.0
        A_ub_list.append(row)
        b_ub_list.append(0.0)
    
    # 2) Box direction constraints
    for i in range(n):
        # u[i] - v[i] <= 1 - h[i]
        row = np.zeros(nv)
        row[i] = 1.0
        row[n+i] = -1.0
        A_ub_list.append(row)
        b_ub_list.append(1.0 - h[i])
        
        # v[i] - u[i] <= h[i]
        row = np.zeros(nv)
        row[i] = -1.0
        row[n+i] = 1.0
        A_ub_list.append(row)
        b_ub_list.append(h[i])
    
    A_ub = np.array(A_ub_list)
    b_ub = np.array(b_ub_list)
    
    # Equality constraints: A_eq @ x = b_eq
    # 1) sum(u-v) = 0
    # 2) sum(u+v) = 1
    A_eq = np.zeros((2, nv))
    A_eq[0, :n] = 1.0
    A_eq[0, n:2*n] = -1.0
    A_eq[1, :n] = 1.0
    A_eq[1, n:2*n] = 1.0
    b_eq = np.array([0.0, 1.0])
    
    # Bounds
    bounds = [(0, None)] * (2*n) + [(0, 5.0)]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x is not None and np.isfinite(res.x).all():
            alpha = max(0.0, -res.fun)
            d = res.x[:n] - res.x[n:2*n]
            # Remove mean for sum constraint (should already be ~0)
            d -= d.mean()
            norm_d = np.sum(np.abs(d))
            if norm_d > 1e-12:
                d /= norm_d
                return d, alpha
    except Exception:
        pass
    return None, 0.0

def softmax_gradient_direction(h, profile, tau=None, top_k=20):
    """Compute softmax-weighted gradient direction."""
    n = len(h)
    M = len(profile)
    c_max = np.max(profile)
    if tau is None:
        spread = c_max - np.median(profile)
        tau = max(1e-4, 0.1 * spread)
    
    # Softmax weights
    log_w = (profile - c_max) / tau
    log_w -= np.max(log_w)  # for numerical stability
    w = np.exp(log_w)
    w_sum = w.sum()
    if w_sum < 1e-15:
        w = np.zeros(M)
        w[np.argmax(profile)] = 1.0
    else:
        w /= w_sum
    
    # Get top-k indices for efficiency
    top_idx = np.argsort(profile)[::-1][:top_k]
    
    G = build_gradient(h, top_idx)
    w_sub = w[top_idx]
    w_sum_sub = w_sub.sum()
    if w_sum_sub < 1e-15:
        return None
    
    grad = G.T @ w_sub
    grad -= grad.mean()
    
    gnorm = np.linalg.norm(grad)
    if gnorm < 1e-15:
        return None
    grad /= gnorm
    return grad

def slsqp_polish(h_start, maxiter=300):
    """SLSQP polish using epigraph formulation."""
    n = len(h_start)
    dx = 2.0 / n
    M = 2 * n - 1
    c0 = exact_score(h_start)
    
    def obj(x):
        return x[-1]
    
    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    
    def eq_fun(x):
        return x[:-1].sum() - n / 2.0
    
    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g
    
    def ineq_fun(x):
        h = x[:-1]
        t = x[-1]
        profile = np.correlate(h, 1 - h, mode="full") * dx
        return t - profile
    
    def ineq_jac(x):
        h = x[:-1]
        G = build_gradient(h, list(range(M)))
        J = np.zeros((M, n + 1))
        J[:, :-1] = -G
        J[:, -1] = 1.0
        return J
    
    x0 = np.concatenate([h_start, [c0 + 1e-3]])
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 1.0)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None and np.isfinite(res.x).all():
            h_new = repair_feasible(res.x[:-1])
            c_new = exact_score(h_new)
            if c_new <= c0 + 1e-8:
                return h_new
    except Exception:
        pass
    return None

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    # Piecewise constant interpolation
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)
