    n = len(h0)
    n_ones = n // 2
    h = np.zeros(n)
    # Round to binary
    sorted_idx = np.argsort(-h0)
    h[sorted_idx[:n_ones]] = 1.0
    
    corrs = compute_corrs(h)
    best_c5 = corrs.max() * dx
    best_h = h.copy()
    
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    
    for sweep in range(max_sweeps):
        if time_check is not None and time_check() > time_limit:
            break
        best_swap = None
        best_new_max = corrs.max()
        
        for i in ones:
            for j in zeros:
                # Compute delta for swap i(1->0), j(0->1)
                delta = swap_delta(h, i, j, n)
                new_max = np.max(corrs + delta)
                if new_max < best_new_max - 1e-14:
                    best_new_max = new_max
                    best_swap = (i, j)

        i, j = best_swap
        h[i] = 0.0
        h[j] = 1.0
        corrs = compute_corrs(h)  # Recompute from scratch (safer)

        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]

```python
def swap_delta(h, i, j, n):
    """Compute delta vector for all correlations when h[i]:1->0 and h[j]:0->1."""
    ms = np.arange(-(n-1), n)  # all possible offsets e
    valid_i_plus = (0 <= i + ms) & (i + ms <= n-1)
    valid_i_minus = (0 <= i - ms) & (i - ms <= n-1)
    valid_j_plus = (0 <= j + ms) & (j + ms <= n-1)
    valid_j_minus = (0 <= j - ms) & (j - ms <= n-1)
    
    delta = np.zeros(2*n-1)
    # h[i]: 1->0
    delta[valid_i_plus] += -(1.0 - h[i + ms[valid_i_plus]])
    delta[valid_i_minus] += h[i - ms[valid_i_minus]]
    # h[j]: 0->1
    delta[valid_j_plus] += (1.0 - h[j + ms[valid_j_plus]])
    delta[valid_j_minus] += -h[j - ms[valid_j_minus]]
    return delta
```

import numpy as np
from scipy.optimize import minimize
import time


def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
        else:
            h += diff / n
    return np.clip(h, 0.0, 1.0)


def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def c5_of_h(h, dx):
    return compute_corrs(h).max() * dx


def smooth_max_obj(h, dx, beta):
    corrs = compute_corrs(h)
    m = corrs.max()
    shifted = beta * (corrs - m)
    return (m + np.log(np.sum(np.exp(shifted))) / beta) * dx


def smooth_max_grad(h, dx, beta):
    n = len(h)
    corrs = compute_corrs(h)
    m = corrs.max()
    shifted = beta * (corrs - m)
    w = np.exp(shifted)
    w /= w.sum()
    grad = np.zeros(n)
    for k in range(len(corrs)):
        wk = w[k]
        if wk < 1e-14:
            continue
        e = k - (n - 1)
        if e == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif e > 0:
            v1 = np.arange(0, n - e)
            grad[v1] += wk * (1.0 - h[v1 + e])
            v2 = np.arange(e, n)
            grad[v2] += wk * (-h[v2 - e])
        else:
            ea = -e
            v1 = np.arange(ea, n)
            grad[v1] += wk * (1.0 - h[v1 - ea])
            v2 = np.arange(0, n - ea)
            grad[v2] += wk * (-h[v2 + ea])
    return grad * dx


def slsqp_refine(h0, dx, target_sum, beta, maxiter):
    n = len(h0)

    def obj(x):
        return smooth_max_obj(x, dx, beta)

    def obj_grad(x):
        return smooth_max_grad(x, dx, beta)

    def cons_fun(x):
        return x.sum() - target_sum

    def cons_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    bounds = [(0.0, 1.0)] * n
    h_start = project_feasible(h0, target_sum)
    try:
        res = minimize(obj, h_start, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        h_out = project_feasible(res.x, target_sum)
    except Exception:
        h_out = h_start
    return h_out, c5_of_h(h_out, dx)


def swap_delta(h, i, j, n):
    ms = np.arange(-(n - 1), n)
    vi_p = (i + ms >= 0) & (i + ms <= n - 1)
    vi_m = (i - ms >= 0) & (i - ms <= n - 1)
    vj_p = (j + ms >= 0) & (j + ms <= n - 1)
    vj_m = (j - ms >= 0) & (j - ms <= n - 1)
    delta = np.zeros(2 * n - 1)
    delta[vi_p] += -(1.0 - h[i + ms[vi_p]])
    delta[vi_m] += h[i - ms[vi_m]]
    delta[vj_p] += (1.0 - h[j + ms[vj_p]])
    delta[vj_m] += -h[j - ms[vj_m]]
    return delta


def binary_steepest(h0, dx, max_sweeps=50, t0=None, budget=None):
    n = len(h0)
    n_ones = n // 2
    h = np.zeros(n)
    idx = np.argsort(-h0)
    h[idx[:n_ones]] = 1.0
    corrs = compute_corrs(h)
    best_c5 = corrs.max() * dx
    best_h = h.copy()
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    for sweep in range(max_sweeps):
        if t0 is not None and time.time() - t0 > budget - 10:
            break
        best_sw = None
        best_nm = corrs.max()
        for i in ones:
            for j in zeros:
                d = swap_delta(h, i, j, n)
                nm = np.max(corrs + d)
                if nm < best_nm - 1e-14:
                    best_nm = nm
                    best_sw = (i, j)
        if best_sw is None:
            break
        i, j = best_sw
        h[i] = 0.0
        h[j] = 1.0
        corrs = compute_corrs(h)
        nc5 = corrs.max() * dx
        if nc5 < best_c5:
            best_c5 = nc5
            best_h = h.copy()
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
    return best_h, best_c5

def frank_wolfe(h0, dx, target_sum, beta, max_iter=3000, time_limit=30.0):

def interpolate_to(h_src, n_new):
    n_src = len(h_src)
    x_src = np.linspace(0.0, 2.0, n_src)
    x_new = np.linspace(0.0, 2.0, n_new)
    h_new = np.interp(x_new, x_src, h_src)
    return h_new


def construct_random_binary(n, rng):
    n_ones = n // 2
    h = np.zeros(n)
    h[rng.choice(n, n_ones, replace=False)] = 1.0
    return h


def construct_golden_binary(n, rng):
    n_ones = n // 2
    h = np.zeros(n)
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    positions = set()
    k = 0
    while len(positions) < n_ones:
        pos = int((k * phi * n) % n)
        positions.add(pos)
        k += 1
    for p in positions:
        h[p] = 1.0
    return h
