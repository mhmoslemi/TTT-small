def c5_of_h(h, dx):

def grad_corr_k(h, k, n):
    """Gradient of correlation at lag k with respect to h."""
    offset = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -offset)
    i_hi = min(n - 1, n - 1 - offset)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx + offset] += (1.0 - h[idx])
    grad[idx] += (-h[idx + offset])
    return grad

So H_k is a sparse banded matrix! Specifically, H_k has -1s on the +offset and -offset diagonals.

The quadratic model for the max becomes:
max_k [corr_k(h) + α g_k·d + α²/2 d^T H_k d]

To minimize this, I can use a trust region method:
min_d max_k [corr_k(h) + α g_k·d + α²/2 d^T H_k d]
s.t. ||d|| ≤ Δ, sum(d) = 0, h + d ∈ [0,1]^n

This is still nonconvex (max of quadratics), but for a fixed d, it's easy to evaluate.

Alternatively, I can use the "worst-case" approach:
min_d, t: t
s.t. corr_k(h) + α g_k·d + α²/2 d^T H_k d ≤ t for all k
     sum(d) = 0
     h_i - 1 ≤ d_i ≤ 1 - h_i (to ensure 0 ≤ h_i + d_i ≤ 1)
     ||d||² ≤ Δ²

If I drop the trust region and the quadratic terms (for the active set only), this becomes the LP I had before.

Let me try a different tactic: **use the LP for direction finding, but with a better step rule.**

Instead of just trying fixed fractions, I'll do a proper backtracking line search on the actual objective:

1. Solve LP to get direction d = h_lp - h
2. Start with α = 1
3. While c5(h + αd) > c5(h) * (1 - 1e-4 * α):
     α *= 0.5
4. Update h = h + αd

This is a standard backtracking line search that should work well.

Let me also add a "restart" mechanism: if the LP method stalls (no improvement for several iterations), restart with a different active set or a random perturbation.

Actually, one more thought: the previous code's binary steepest descent (trying all swaps) is O(n²) per sweep, which for n=126 is about 4000 evaluations per sweep. Each evaluation is O(n²) for the correlation (actually O(n) with the full correlate). So it's fast enough.

But what about a continuous version of this? Instead of binary swaps, try small transfers between pairs:
- For each pair (i, j), try h[i] += ε, h[j] -= ε (and vice versa)
- Check if the max correlation decreases
- Accept the best one

This is O(n²) pairs × O(n) evaluation = O(n³) per sweep, which for n=126 is about 2 million operations. That's fast.

Let me include this as a "continuous pair polish" step.

Actually, for the continuous case, the optimal transfer between a pair (i,j) can be found analytically. Given the current corr vector, the change in corr_k due to transferring ε from j to i is:
Δcorr_k = ε * (∂corr_k/∂h_i - ∂corr_k/∂h_j)

We want to choose ε to minimize max_k (corr_k + Δcorr_k). This is a 1D problem that can be solved by bisection.

But doing this for all pairs is expensive. Let me just do a simpler version: for each pair, try a small ε and see if it helps.

import numpy as np
from scipy.optimize import linprog, minimize
import time


def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
        else:
            h += diff / n
    return np.clip(h, 0.0, 1.0)


def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def c5_of_h(h, dx):
    return float(compute_corrs(h).max() * dx)


def grad_corr_single(h, k, n):
    offset = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -offset)
    i_hi = min(n - 1, n - 1 - offset)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx + offset] += (1.0 - h[idx])
    grad[idx] += (-h[idx + offset])
    return grad


def smooth_max_obj(h, dx, beta):
    corrs = compute_corrs(h)
    m = corrs.max()
    return (m + np.log(np.sum(np.exp(beta * (corrs - m)))) / beta) * dx


def smooth_max_grad(h, dx, beta):
    n = len(h)
    corrs = compute_corrs(h)
    m = corrs.max()
    shifted = corrs - m
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    w /= w_sum
    grad = np.zeros(n)
    for k in range(len(corrs)):
        if w[k] < 1e-14:
            continue
        grad += w[k] * grad_corr_single(h, k, n)
    return grad * dx


def slsqp_optimize(h0, dx, target_sum, beta, maxiter):
    n = len(h0)

    def obj(x):
        return smooth_max_obj(x, dx, beta)

    def obj_grad(x):
        return smooth_max_grad(x, dx, beta)

    def cons_fun(x):
        return x.sum() - target_sum

    def cons_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0, jac=obj_grad,
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        h_out = project_feasible(res.x, target_sum)
    except Exception:
        h_out = project_feasible(h0, target_sum)
    return h_out, c5_of_h(h_out, dx)


def solve_lp_direction(h, n, active_indices):
    target_sum = n / 2.0
    corrs = compute_corrs(h)
    m_active = len(active_indices)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((m_active, n + 1))
    b_ub = np.zeros(m_active)
    for idx, k in enumerate(active_indices):
        g = grad_corr_single(h, k, n)
        A_ub[idx, :n] = g
        A_ub[idx, -1] = -1.0
        b_ub[idx] = g @ h - corrs[k]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target_sum])
    bounds = [(0.0, 1.0)] * n + [(None, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None


def lp_active_refine(h0, n, dx, time_limit):
    t_start = time.time()
    h = h0.copy()
    target_sum = n / 2.0
    best_c5 = c5_of_h(h, dx)
    best_h = h.copy()
    no_improve = 0
    margin_schedule = [0.0, 0.02, 0.05, 0.1, 0.15, 0.25, 0.3]
    for it in range(500):
        if time.time() - t_start > time_limit:
            break
        elapsed = time.time() - t_start
        corrs = compute_corrs(h)
        raw_max = corrs.max()
        margin_idx = min(int(elapsed / 10), len(margin_schedule) - 1)
        margin = margin_schedule[margin_idx] * raw_max
        threshold = raw_max - margin
        active = np.where(corrs >= threshold)[0]
        if len(active) < 2:
            active = np.argsort(-corrs)[:3]
        elif len(active) > 20:
            active = np.argsort(-corrs)[:20]
        h_lp = solve_lp_direction(h, n, active)
        if h_lp is None:
            no_improve += 1
            if no_improve > 20:
                margin_idx = min(margin_idx + 1, len(margin_schedule) - 1)
                no_improve = 0
            continue
        direction = h_lp - h
        step_norm = np.linalg.norm(direction)
        if step_norm < 1e-14:
            no_improve += 1
            continue
        found = False
        for alpha in [1.0, 0.5, 0.25, 0.125, 0.0625, 0.03]:
            h_trial = project_feasible(h + alpha * direction, target_sum)
            c5_trial = c5_of_h(h_trial, dx)
            if c5_trial < best_c5 - 1e-14:
                h = h_trial
                best_c5 = c5_trial
                best_h = h.copy()
                no_improve = 0
                found = True
                break
        if not found:
            no_improve += 1
            if no_improve > 30:
                break
    return best_h, best_c5


def continuous_pair_polish(h0, n, dx, time_limit):
    t_start = time.time()
    h = h0.copy()
    target_sum = n / 2.0
    best_c5 = c5_of_h(h, dx)
    best_h = h.copy()
    corrs = compute_corrs(h)
    for sweep in range(200):
        if time.time() - t_start > time_limit:
            break
        best_improve = None
        best_pair = None
        best_eps_val = 0.0
        raw_max = corrs.max()
        active_tol = 0.02 * raw_max
        active_mask = corrs >= (raw_max - active_tol)
        for i in range(n):
            for j in range(i + 1, n):
                if abs(h[i] - h[j]) < 1e-8:
                    continue
                if h[i] < 1e-10 or h[j] > 1.0 - 1e-10:
                    continue
                dgrad = np.zeros(2 * n - 1)
                for k in np.where(active_mask)[0]:
                    dgrad[k] += grad_corr_single(h, k, n)[i] - grad_corr_single(h, k, n)[j]
                active_dgrad = dgrad[active_mask]
                if np.all(active_dgrad >= -1e-12):
                    continue
                for eps in [0.01, 0.005, 0.001, 0.002]:
                    if h[i] < eps or h[j] < eps:
                        continue
                    h[i] -= eps
                    h[j] += eps
                    new_corrs = corrs + eps * dgrad
                    new_max = new_corrs.max()
                    if new_max < raw_max - 1e-12:
                        if best_improve is None or new_max < best_improve:
                            best_improve = new_max
                            best_pair = (i, j)
                            best_eps_val = eps
                    h[i] += eps
                    h[j] -= eps
        if best_pair is not None:
            i, j = best_pair
            h[i] -= best_eps_val
            h[j] += best_eps_val
            h = project_feasible(h, target_sum)
            corrs = compute_corrs(h)
            new_c5 = corrs.max() * dx
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            break
    return best_h, best_c5


def kkt_polish(h0, n, dx, time_limit):
    t_start = time.time()
    h = h0.copy()
    target_sum = n / 2.0
    best_c5 = c5_of_h(h, dx)
    best_h = h.copy()
    for it in range(50):
        if time.time() - t_start > time_limit:
            break
        corrs = compute_corrs(h)
        raw_max = corrs.max()
        threshold = raw_max - 0.005 * raw_max
        active = np.where(corrs >= threshold)[0]
        if len(active) < 2:
            active = np.argsort(-corrs)[:4]
        if len(active) > 12:
            active = np.argsort(-corrs)[:12]
        g_matrix = np.array([grad_corr_single(h, k, n) for k in active])
        free_mask = (h > 0.01) & (h < 0.99)
        if free_mask.sum() < 3:
            break
        g_free = g_matrix[:, free_mask]
        m_a = len(active)
        n_f = free_mask.sum()
        A_mat = np.zeros((n_f + m_a + 1, n_f + m_a + 1))
        b_vec = np.zeros(n_f + m_a + 1)
        for i in range(n_f):
            A_mat[i, :n_f] = g_free[i, :] * 0
        A_mat[:n_f, :n_f] = g_free.T
        A_mat[:n_f, n_f:n_f + m_a] = -np.eye(n_f, m_a)
        A_mat[:n_f, -1] = 1.0
        b_vec[:n_f] = 0.0
        for i in range(m_a):
            row = n_f + i
            A_mat[row, :n_f] = 0.0
            A_mat[row, n_f + i] = 1.0
            A_mat[row, -1] = -1.0
            b_vec[row] = corrs[active[i]]
        row = n_f + m_a
        A_mat[row, :n_f] = 1.0
        A_mat[row, n_f:n_f + m_a] = 0.0
        A_mat[row, -1] = 0.0
        b_vec[row] = target_sum - h[~free_mask].sum()
        try:
            sol = np.linalg.solve(A_mat, b_vec)
            h_free_new = sol[:n_f]
            w_new = sol[n_f:n_f + m_a]
            t_new = sol[-1]
            if np.all(h_free_new > -0.05) and np.all(h_free_new < 1.05) and np.all(w_new > -0.1):
                h_trial = h.copy()
                h_trial[free_mask] = np.clip(h_free_new, 0.0, 1.0)
                h_trial = project_feasible(h_trial, target_sum)
                c5_trial = c5_of_h(h_trial, dx)
                if c5_trial < best_c5 - 1e-14:
                    h = h_trial
                    best_c5 = c5_trial
                    best_h = h.copy()
        except np.linalg.LinAlgError:
            break
    return best_h, best_c5


def make_wave_start(n, target_sum, rng, freq1, freq2, phase1, phase2):
    x = np.linspace(0, 2.0, n, endpoint=False)
    h = 0.5 + 0.3 * np.sin(np.pi * freq1 * x + phase1) + 0.2 * np.sin(np.pi * freq2 * x + phase2)
    return project_feasible(h, target_sum)


def remaining_time(t0, budget_s):
    return budget_s - (time.time() - t0)
