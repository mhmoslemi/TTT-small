def project_feasible(x, target_sum):

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.special import logsumexp
import time


def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full').max() * dx


def corr_vals(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx


def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)


def grad_cols(h, S, n):
    dx = 2.0 / n
    nS = len(S)
    G = np.zeros((2 * n - 1, nS))
    for si in range(nS):
        j = S[si]
        d_lo1 = max(-j, -(n - 1))
        d_hi1 = min(n - 1 - j, n - 1)
        if d_lo1 <= d_hi1:
            d_r = np.arange(d_lo1, d_hi1 + 1)
            k_r = d_r + n - 1
            G[k_r, si] += dx * (1.0 - h[j + d_r])
        d_lo2 = max(j - n + 1, -(n - 1))
        d_hi2 = min(j, n - 1)
        if d_lo2 <= d_hi2:
            d_r = np.arange(d_lo2, d_hi2 + 1)
            k_r = d_r + n - 1
            G[k_r, si] += -dx * h[j - d_r]
    return G


def solve_block_lp(h, S, n):
    S = np.sort(np.asarray(S, dtype=int))
    nS = len(S)
    if nS < 2:
        return h
    dx = 2.0 / n
    n_lags = 2 * n - 1
    target_sum_S = n / 2.0 - (h.sum() - h[S].sum())
    if target_sum_S < -1e-9 or target_sum_S > nS + 1e-9:
        return h
    corrs = np.correlate(h, 1.0 - h, mode='full') * dx
    G = grad_cols(h, S, n)
    A_ub = np.hstack([G, -np.ones((n_lags, 1))])
    b_ub = G @ h[S] - corrs
    A_eq = np.hstack([np.ones((1, nS)), [[0.0]]])
    b_eq = np.array([target_sum_S])
    t_lo = max(0.0, corrs.max() * 0.5)
    bounds = [(0.0, 1.0)] * nS + [(t_lo, 2.0)]
    c_obj = np.zeros(nS + 1)
    c_obj[-1] = 1.0
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 5.0})
        if res.success:
            h_new = h.copy()
            h_new[S] = np.clip(res.x[:nS], 0.0, 1.0)
            return h_new
    except Exception:
        pass
    return h


def smooth_obj(x, n, beta):
    h = np.clip(x, 0.0, 1.0)
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    return logsumexp(beta * c) / beta


def smooth_obj_grad(x, n, beta):
    h = np.clip(x, 0.0, 1.0)
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    shifted = c - c.max()
    w = np.exp(beta * shifted)
    ws = w.sum()
    if ws < 1e-30:
        w = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        w /= ws
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = beta * w[k]

        d = k - (n - 1)

        idx = np.arange(i_lo, i_hi + 1)
        grad[idx] += wk * (1.0 - h[idx + d])
        grad[idx + d] += wk * (-h[idx])
    return grad * dx

        corrs = corr_vals(h)
        top_n = min(40 + 10 * (it // 1000), 2 * n - 1)
        active_lags = np.argsort(corrs)[-top_n:]
        active_d = active_lags - (n - 1)
        involved = set()
        for d in active_d:
            i_lo = max(0, -d)

            else:
                break
        if len(involved) > max_block:
            inv_arr = np.array(list(involved))
            scores = np.minimum(h[inv_arr], 1.0 - h[inv_arr])
