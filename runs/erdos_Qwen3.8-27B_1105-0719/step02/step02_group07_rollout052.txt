For a pair (i, j): the objective is max_k (c_k + a_i[k]*h[i] + a_j[k]*h[j]) where c_k, a_i, a_j are fixed. This is a 2D convex optimization (max of linear functions in 2D). I can solve it with:
- Grid search (30x30 = 900 evaluations)
- Or a specialized 2D algorithm

Let me implement pair coordinate descent as well.

For pair (i, j):
- Coefficient of h[i]: a_i[k] (computed as before)
- Coefficient of h[j]: a_j[k]
- Base: corrs[k] - a_i[k]*h[i] - a_j[k]*h[j] (the correlation without the i,j contributions)

Wait, that's not quite right. Let me think again.

corr[k] = sum_m h[m]*(1-h[m+d]) for valid m.
If I fix all coordinates except i and j, then:
corr[k] = base[k] + h[i]*a_i[k] + h[j]*a_j[k] + cross_term

The cross term: if i and j both appear in the same product h[m]*(1-h[m+d]), there's a cross term. This happens when:
- m=i and m+d=j: h[i]*(1-h[j]) → cross term -h[i]*h[j]
- m=j and m+d=i: h[j]*(1-h[i]) → cross term -h[i]*h[j]
- m=i and m+d=i (trivial, not a cross term)
- etc.

So there can be a cross term -h[i]*h[j] for specific k values. This makes the 2D subproblem NOT linear (it's bilinear). The max of bilinear functions in 2D is non-convex in general.

To handle this, I can:
1. Ignore the cross term (it's usually small and the approximation is often good)
2. Use a local search (grid + refine) that handles the non-convexity
3. Use alternating updates: fix j, optimize i; fix i, optimize j; repeat

Option 3 is simplest and reduces to two 1D problems. But it's essentially the same as doing coordinate descent separately on i and j.

import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time


def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    return np.clip(h, 0.0, 1.0)


def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full').max() * dx


def corr_vec(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx


def make_sym_h(q, n):
    m = n // 2
    h = np.empty(n)
    h[:m] = q
    h[m:] = 1.0 - q[::-1]
    return h


def make_q_from_h(h, n):
    m = n // 2
    q = 0.5 * (h[:m] + 1.0 - h[n - 1:n - 1 - m:-1])
    return np.clip(q, 0.0, 1.0)


def compute_coeff(h, i, n):
    a = np.zeros(2 * n - 1)
    a[n - 1 - i:2 * n - 1 - i] += (1.0 - h)
    a[i:i + n] += -h[::-1]
    return a


def sym_coord_descent(q, n, n_sweeps, grid_size, time_limit=None, t_start=None):
    m = n // 2
    q = q.copy()
    best_obj = np.inf
    improved = True
    actual_sweeps = 0
    for sweep in range(n_sweeps):
        if time_limit is not None and t_start is not None:
            if time.time() - t_start > time_limit:
                break
        h = make_sym_h(q, n)
        corrs = corr_vec(h)
        obj_cur = corrs.max()
        if obj_cur < best_obj:
            best_obj = obj_cur
        for i in range(m):
            j = n - 1 - i
            ai = compute_coeff(h, i, n)
            aj = compute_coeff(h, j, n)
            a_eff = ai - aj
            old_val = q[i]
            t_vals = np.linspace(0.0, 1.0, grid_size)
            deltas = t_vals - old_val
            obj_arr = (corrs + deltas[:, None] * a_eff[None, :]).max(axis=1)
            idx = np.argmin(obj_arr)
            best_t = t_vals[idx]
            lo_b = t_vals[max(0, idx - 1)]
            hi_b = t_vals[min(grid_size - 1, idx + 1)]
            t_fine = np.linspace(lo_b, hi_b, 101)
            d_fine = t_fine - old_val
            obj_fine = (corrs + d_fine[:, None] * a_eff[None, :]).max(axis=1)
            best_t = t_fine[np.argmin(obj_fine)]
            q[i] = best_t
            h[i] = best_t
            h[j] = 1.0 - best_t
            corrs += (best_t - old_val) * a_eff
        actual_sweeps += 1
    return q, actual_sweeps


def nonsym_coord_descent(h, n, n_sweeps, grid_size, time_limit=None, t_start=None):
    h = h.copy()
    target = n / 2.0
    for sweep in range(n_sweeps):
        if time_limit is not None and t_start is not None:
            if time.time() - t_start > time_limit:
                break
        corrs = corr_vec(h)
        for i in range(n):
            a_i = compute_coeff(h, i, n)
            old_val = h[i]
            t_vals = np.linspace(0.0, 1.0, grid_size)
            obj_arr = (corrs + (t_vals - old_val)[:, None] * a_i[None, :]).max(axis=1)
            idx = np.argmin(obj_arr)
            best_t = t_vals[idx]
            lo_b = t_vals[max(0, idx - 1)]
            hi_b = t_vals[min(grid_size - 1, idx + 1)]
            t_fine = np.linspace(lo_b, hi_b, 101)

            h[i] = best_t
            corrs += (best_t - old_val) * a_i
        h = project_feasible(h, target)
    return h


def smooth_sym_obj(x, n, beta):
    h = make_sym_h(x, n)
    corrs = corr_vec(h)
    return logsumexp(beta * corrs) / beta


def smooth_sym_grad(x, n, beta):
    m = n // 2
    h = make_sym_h(x, n)
    corrs = corr_vec(h)
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    ws = w.sum()
    if ws < 1e-30:
        w = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        w /= ws
    G = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        d = k - (n - 1)
        i_lo = max(0, -d)
        i_hi = min(n - 1, n - 1 - d)
        if i_lo > i_hi:
            continue
        idx = np.arange(i_lo, i_hi + 1)
        G[idx] += wk * (1.0 - h[idx + d])
        G[idx + d] += wk * (-h[idx])
    G *= (2.0 / n)
    return G[:m] - G[n - 1:n - 1 - m:-1]


def smooth_sym_optimize(q0, n, beta, maxiter=300):
    m = n // 2
    try:
        res = minimize(smooth_sym_obj, q0.copy(), jac=lambda x: smooth_sym_grad(x, n, beta),
                       method='SLSQP', bounds=[(0.0, 1.0)] * m,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        h_new = make_sym_h(res.x, n)
        c5 = exact_score(h_new)
        return res.x, c5
    except Exception:
        return None, None


def active_lags_top(corrs, n_top):
    idx = np.argsort(corrs)[::-1][:n_top]
    return sorted(idx.tolist())


class EpiSymSolver:
    def __init__(self, n, active_lags):
        self.n = n
        self.m = n // 2
        self.dx = 2.0 / n
        self.active_lags = np.array(active_lags, dtype=int)

    def fun(self, x):
        return x[0]

    def jac(self, x):
        g = np.zeros(self.m + 1)
        g[0] = 1.0
        return g

    def con_fun(self, x):
        t = x[0]
        q = x[1:]
        h = make_sym_h(q, self.n)
        corrs = corr_vec(h)
        return t - corrs[self.active_lags]

    def con_jac(self, x):
        q = x[1:]
        h = make_sym_h(q, self.n)
        n = self.n
        m = self.m
        dx = self.dx
        L = len(self.active_lags)
        J = np.zeros((L, m + 1))
        J[:, 0] = 1.0
        for i, lag in enumerate(self.active_lags):
            d = lag - (n - 1)
            i_lo = max(0, -d)
            i_hi = min(n - 1, n - 1 - d)
            G = np.zeros(n)
            if i_lo <= i_hi:
                idx = np.arange(i_lo, i_hi + 1)
                G[idx] += (1.0 - h[idx + d])
                G[idx + d] += (-h[idx])
            G *= dx
            J[i, 1:] = -(G[:m] - G[n - 1:n - 1 - m:-1])
        return J


def solve_epi_sym(q0, n, active_lags, t_init, maxiter=300):
    m = n // 2
    x0 = np.concatenate([[t_init], q0.copy()])
    solver = EpiSymSolver(n, active_lags)
    cons = [{'type': 'ineq', 'fun': solver.con_fun, 'jac': solver.con_jac}]
    bounds = [(0.0, 2.0)] + [(0.0, 1.0)] * m
    try:
        res = minimize(solver.fun, x0, jac=solver.jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.x[0] < 0:
            return None, None
        h_new = make_sym_h(res.x[1:], n)
        c5 = exact_score(h_new)
        if abs(h_new.sum() - n / 2.0) > 1e-6:
            return None, None
        return res.x[1:], c5
    except Exception:
        return None, None


def make_binary_pattern(n, seed):
    rng = np.random.RandomState(seed)
