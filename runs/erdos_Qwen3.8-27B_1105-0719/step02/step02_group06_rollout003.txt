def eval_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def corr_raw(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_all_gradients(h):
    """Compute gradient matrix for all lags.
    Returns G of shape (2n-1, n) where G[k, i] = dR_k/dh[i]
    and R of shape (2n-1,) where R[k] = raw correlation at lag k.
    """
    n = len(h)
    L = 2 * n - 1
    G = np.zeros((L, n))
    R = np.zeros(L)
    for ell in range(L):
        d = ell - (n - 1)
        i_lo = max(0, -d)
        i_hi = min(n - 1, n - 1 - d)
        if i_lo > i_hi:
            continue
        idx = np.arange(i_lo, i_hi + 1)
        jdx = idx + d
        R[ell] = np.sum(h[idx] * (1.0 - h[jdx]))
        G[ell, idx] += (1.0 - h[jdx])
        G[ell, jdx] += (-h[idx])
    return G, R

def solve_tangent_lp(h, G, R, rho=0.3):
    """Solve the tangent LP to find a step direction.
    Returns z (new point) or None if infeasible.
    """
    n = len(h)
    L = 2 * n - 1
    
    # Objective: minimize t
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    # Inequality constraints: G @ z - t <= G @ h - R
    # i.e., G @ z + [-1] @ t <= G @ h - R
    A_ub = np.hstack([G, -np.ones((L, 1))])
    b_ub = G @ h - R
    
    # Equality: sum(z) = n/2
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    # Bounds: 0 <= z_i <= 1, |z_i - h_i| <= rho, 0 <= t <= upper
    lower = np.maximum(0.0, h - rho)
    upper = np.minimum(1.0, h + rho)
    t_upper = float(R.max()) + rho  # t can't need to be much larger than current max
    bounds = list(zip(lower, upper)) + [(0.0, t_upper)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n]
    return None

def transfer_line_search(h, a, b, n_grid=40):
    """Find best delta for transfer h[a]+delta, h[b]-delta."""
    x_a, x_b = h[a], h[b]
    lo = max(-x_a, x_b - 1.0)
    hi = min(1.0 - x_a, -x_b)
    if hi <= lo:
        return h, 0.0
    
    # Grid search
    grid = np.linspace(lo, hi, n_grid)
    best_val = np.inf
    best_d = 0.0

        h[b] = x_b - d
        v = eval_score(h)
        if v < best_val:

            best_d = d
    
    h[a] = x_a + best_d
    h[b] = x_b - best_d
    
    # Golden section refinement around best_d
    span = (hi - lo) / n_grid
    a_gs = best_d - span
    b_gs = best_d + span
    a_gs = max(a_gs, lo)
    b_gs = min(b_gs, hi)

        f_c1 = _eval_transfer(h, a, b, x_a, x_b, c1)

                c1 = b_gs - gr * (b_gs - a_gs)

                c2 = a_gs + gr * (b_gs - a_gs)
                f_c2 = _eval_transfer(h, a, b, x_a, x_b, c2)
        
        best_d = (a_gs + b_gs) / 2.0
        h[a] = x_a + best_d
        h[b] = x_b - best_d
    
    return h, eval_score(h)

def line_search_lp_step(h, z, n_steps=8):
    """Line search along h -> z. Returns best point and its score."""
    best_h = h.copy()
    best_score = eval_score(h)
    for i in range(1, n_steps + 1):
        alpha = 1.0 / (2 ** (i - 1))
        h_try = h + alpha * (z - h)
        h_try = np.clip(h_try, 0.0, 1.0)
        # Project to satisfy sum constraint
        diff = n_target(h_try) - h_try.sum()
        if abs(diff) > 1e-10:
            h_try = h_try + diff / len(h_try)
            h_try = np.clip(h_try, 0.0, 1.0)
        s = eval_score(h_try)
        if s < best_score:
            best_score = s
            best_h = h_try.copy()
    return best_h, best_score

def n_target(h):

def project_feasible(x, target_sum):
    n = len(x)
    x = np.clip(x, 0.0, 1.0)
    lo = -2.0
    hi = 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(x + mid, 0.0, 1.0).sum()
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    h = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(-h)
            for idx in order:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
                if abs(diff) < 1e-12:

        else:

            for idx in order:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub

def eval_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def corr_raw(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_all_gradients(h):
    n = len(h)
    L = 2 * n - 1
    G = np.zeros((L, n))
    R = np.zeros(L)
    for ell in range(L):
        d = ell - (n - 1)
        i_lo = max(0, -d)
        i_hi = min(n - 1, n - 1 - d)
        if i_lo > i_hi:
            continue
        idx = np.arange(i_lo, i_hi + 1)
        jdx = idx + d
        R[ell] = np.sum(h[idx] * (1.0 - h[jdx]))
        G[ell, idx] += (1.0 - h[jdx])
        G[ell, jdx] += (-h[idx])
    return G, R

def solve_tangent_lp(h, G, R, rho):
    n = len(h)
    L = 2 * n - 1
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_ub = np.hstack([G, -np.ones((L, 1))])
    b_ub = G @ h - R
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    lower = np.maximum(0.0, h - rho)
    upper = np.minimum(1.0, h + rho)
    t_upper = float(R.max()) + 2.0
    bounds = list(zip(lower.tolist(), upper.tolist())) + [(0.0, t_upper)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n]
    return None
