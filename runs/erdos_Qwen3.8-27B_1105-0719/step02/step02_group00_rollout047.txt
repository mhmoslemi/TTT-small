def exact_score(h):
    n = len(h)

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx] + 1e-15)
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx] + 1e-15)
            if rem > 0:

            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def build_gradient_matrix(h, n, dx):
    """Build G matrix where G[m, i] = d(P_m)/d(h[i])"""
    M = 2 * n - 1
    G = np.zeros((M, n))
    for m in range(M):
        d = m - (n - 1)
        if d == 0:
            G[m, :] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            # P_d = dx * sum_{i=0}^{n-d-1} h[i+d]*(1-h[i])
            # d/dh[i] for i in 0..n-d-1: -dx * h[i+d]
            # d/dh[i+d] for i in 0..n-d-1: dx * (1-h[i])
            G[m, 0:n-d] += -dx * h[d:n]
            G[m, d:n] += dx * (1.0 - h[0:n-d])
        else:
            dp = -d
            # P_{-dp} = dx * sum_{i=0}^{n-dp-1} h[i]*(1-h[i+dp])
            # d/dh[i] for i in 0..n-dp-1: dx * (1-h[i+dp])
            # d/dh[i+dp] for i in 0..n-dp-1: -dx * h[i]
            G[m, 0:n-dp] += dx * (1.0 - h[dp:n])
            G[m, dp:n] += -dx * h[0:n-dp]
    return G

    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

def lp_step(h, profile, G, radius, max_active=None):
    """Solve LP trust-region step using cvxpy."""
    n = len(h)
    T = np.max(profile)
    
    # Select active constraints
    beam = max(0.003, 0.05 * (T - np.median(profile)))
    active = np.where(profile >= T - beam)[0]
    if len(active) == 0:
        active = np.array([np.argmax(profile)])
    if max_active is not None and len(active) > max_active:
        top_idx = np.argsort(profile[active])[-max_active:]
        active = active[top_idx]
    
    # Weights: higher for more active constraints
    weights = profile[active] - (T - beam)
    weights = np.maximum(weights, 0.01)
    # Normalize (for regularization only, not for the constraint)
    
    # CVXpy LP
    d = cp.Variable(n)
    t = cp.Variable()
    
    constraints = [
        0 <= h + d,
        h + d <= 1,
        cp.sum(d) == 0,  # sum(h+d) = sum(h) = n/2, so sum(d) = 0
        d <= radius,
        -d <= radius
    ]
    
    # Linearized active constraints: t >= profile[m] + G[m,:] @ d
    for i, m in enumerate(active):
        linearized_val = profile[m] + G[m, :] @ h  # constant part
        # Wait, G[m,:] is already the gradient, so:
        # p_m(h+d) ≈ p_m(h) + G[m,:] @ d = profile[m] + G[m,:] @ d
        constraints.append(t >= profile[m] + G[m, :] @ d)
    
    # Small regularization to avoid degenerate solutions
    objectives = [cp.Minimize(t + 0.01 * cp.norm1(d))]
    
    prob = cp.Problem(objectives[0], constraints)
    
    try:

    if d.value is None or not np.all(np.isfinite(d.value)):

def make_block_pattern(n, n_blocks, rng):
    """Create a block-structured pattern."""
    block_size = n // n_blocks
    h = np.zeros(n)
    for b in range(n_blocks):
        start = b * block_size
        end = min((b + 1) * block_size, n)
        val = rng.uniform(0.1, 0.9)
        h[start:end] = val
    return repair_feasible(h)

def make_symmetric_h(n, h_half, rng, mode='reflect'):

    n_half = n // 2
    y = np.clip(h_half[:n_half], 0, 1)
    if mode == 'reflect':
        h = np.zeros(n)
        h[:n_half] = y
        if n % 2 == 0:

def solve_cvxpy_lp(h, profile, G, radius, max_active=20):
    """Solve the LP trust-region subproblem."""
    n = len(h)
    T = np.max(profile)
    
    beam = max(0.003, 0.05 * (T - np.median(profile)))
    active = np.where(profile >= T - beam)[0]
    if len(active) == 0:
        active = np.array([np.argmax(profile)])
    if max_active is not None and len(active) > max_active:
        top_idx = np.argsort(profile[active])[-max_active:]
        active = active[top_idx]
    
    d = cp.Variable(n)
    t = cp.Variable()
    
    constraints = [
        d <= radius,
        -d <= radius,
        (h + d) >= 0,
        (h + d) <= 1,
        cp.sum(d) == 0
    ]
    
    for m in active:
        constraints.append(t >= profile[m] + G[m, :] @ d)
    
    obj = cp.Minimize(t + 0.005 * cp.sum(cp.abs(d)))
    prob = cp.Problem(obj, constraints)
    
    try:
        prob.solve(solver=cp.ECOS, max_iters=10000)
    except Exception:
        return None
    
    if d.value is None or not np.all(np.isfinite(d.value)):
        return None
    
    return d.value

def build_gradient_matrix(h, n, dx):
