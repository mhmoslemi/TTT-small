    # Actually, let me just use the formula directly:
    # c[k] = sum_n h[n+k_off] * (1-h[n]) where k_off = k-(n-1) and n ranges over valid indices
    # 
    # When h[i] goes from 1 to 0:
    # 1) h[i] appears in h[n+k_off] when n+k_off = i, i.e., n = i-k_off
    #    Old contribution: h[i]*(1-h[i-k_off]) = 1*(1-h[i-k_off])
    #    New contribution: 0*(1-h[i-k_off]) = 0
    #    Delta: -(1-h[i-k_off]) if 0 <= i-k_off < n
    #
    # 2) h[i] appears in (1-h[n]) when n = i
    #    Old contribution: h[i+k_off]*(1-h[i]) = h[i+k_off]*0 = 0
    #    New contribution: h[i+k_off]*(1-0) = h[i+k_off]
    #    Delta: +h[i+k_off] if 0 <= i+k_off < n
    
    # For h[j] going from 0 to 1:
    # 1) h[j] appears in h[n+k_off] when n+k_off = j
    #    Old: 0*(1-h[j-k_off]) = 0
    #    New: 1*(1-h[j-k_off])
    #    Delta: +(1-h[j-k_off]) if 0 <= j-k_off < n
    #
    # 2) h[j] appears in (1-h[n]) when n = j
    #    Old: h[j+k_off]*(1-0) = h[j+k_off]
    #    New: h[j+k_off]*(1-1) = 0
    #    Delta: -h[j+k_off] if 0 <= j+k_off < n
    
    # So total delta for c[k]:
    #   -(1-h[i-k_off]) + h[i+k_off] + (1-h[j-k_off]) - h[j+k_off]
    #   (each term only if the index is valid)
    
    # Let me vectorize this over all k:
    k_off = ms  # ms = arange(-(n-1), n) is the offset for each k
    
    # Term 1: -(1-h[i-ms]) for valid i-ms
    valid_a = (i + ms >= 0) & (i + ms < n)
    delta[valid_a] += -(1.0 - h[i + ms[valid_a]])
    
    # Term 2: +h[i-ms] for valid i-ms  (wait, I need h[i - k_off] where k_off = ms)
    # Actually h[i+k_off] where k_off = ms, so h[i+ms]
    valid_b = (i - ms >= 0) & (i - ms < n)
    # h[i + k_off] = h[i + ms]... wait
    # Let me re-derive. k_off for the k-th correlation entry is: offset = k - (n-1)
    # In 'full' mode, k goes from 0 to 2n-2, and offset = k - (n-1) goes from -(n-1) to n-1
    # So ms = arange(-(n-1), n) gives the offsets.
    #
    # c[k] = sum_n h[n + offset] * (1-h[n])
    #
    # h[i] in h[n+offset]: n + offset = i → n = i - offset
    # Valid: 0 <= i-offset < n
    # Delta from h[i]: 1→0: -(1-h[i-offset])
    valid_c = (i - ms >= 0) & (i - ms < n)
    # I already did this as valid_a with i+ms... let me redo
    # n = i - offset = i - ms
    valid_1 = (i - ms >= 0) & (i - ms < n)
    delta[valid_1] += -(1.0 - h[i - ms[valid_1]])
    
    # h[i] in (1-h[n]): n = i
    # Delta: h was 1, (1-h)=0. Now h=0, (1-h)=1.
    # The term is h[n+offset] * (1-h[n]) = h[i+offset] * (1-h[i])
    # Old: h[i+offset] * 0 = 0
    # New: h[i+offset] * 1 = h[i+offset]
    # Delta: +h[i+offset] if 0 <= i+offset < n
    valid_2 = (i + ms >= 0) & (i + ms < n)
    delta[valid_2] += h[i + ms[valid_2]]
    
    # h[j] from 0 to 1:
    # h[j] in h[n+offset]: n = j-offset
    # Old: 0 * (1-h[j-offset]) = 0
    # New: 1 * (1-h[j-offset])
    # Delta: +(1-h[j-offset])
    valid_3 = (j - ms >= 0) & (j - ms < n)
    delta[valid_3] += (1.0 - h[j - ms[valid_3]])
    
    # h[j] in (1-h[n]): n = j
    # Old: h[j+offset] * (1-0) = h[j+offset]
    # New: h[j+offset] * (1-1) = 0
    # Delta: -h[j+offset]
    valid_4 = (j + ms >= 0) & (j + ms < n)
    delta[valid_4] -= h[j + ms[valid_4]]
    
    return delta

def make_fourier_start(n, target_sum):
    """Create a Fourier-based initial condition."""
    x = np.linspace(0, 2, n, endpoint=False)
    h = np.zeros(n)
    # Try a cosine wave
    h = 0.5 + 0.3 * np.cos(np.pi * x) + 0.1 * np.cos(2 * np.pi * x)
    return project_feasible(h, target_sum)

def make_block_start(n, target_sum, rng):
    """Create a block-structured initial condition."""
    n_blocks = rng.randint(3, 8)
    block_sizes = rng.multinomial(n, np.ones(n_blocks) / n_blocks)
    h = np.zeros(n)
    pos = 0
    for bs in block_sizes:
        if rng.random() > 0.5:
            h[pos:pos+bs] = 1.0
        pos += bs
    return project_feasible(h, target_sum)

def basin_hopping(h0, dx, target_sum, rng, n_attempts, beta, maxiter_slsqp, time_stop):
    """Basin hopping: perturb, optimize, compare."""
    best_h = h0.copy()
    best_c5 = c5_of_h(h0, dx)
    n = len(h0)
    
    for attempt in range(n_attempts):
        if time.time() > time_stop:
            break
        
        # Random perturbation
        eps = rng.uniform(0.01, 0.05, size=n)
        direction = rng.choice([-1, 1], size=n).astype(float)
        h_pert = best_h + eps * direction
        h_pert = project_feasible(h_pert, target_sum)
        
        # Optimize
        h_opt, c5_opt = slsqp_optimize(h_pert, dx, target_sum, beta, maxiter_slsqp)
        
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
    
    return best_h, best_c5
