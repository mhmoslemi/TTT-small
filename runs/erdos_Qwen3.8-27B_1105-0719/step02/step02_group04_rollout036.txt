def eval_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def c5_array(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def project_sum(h, target):
    """Project h to have sum = target while staying in [0,1]"""
    n = len(h)
    # Binary search for the offset
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h_new = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix any residual
    diff = target - h_new.sum()
    if abs(diff) > 1e-12:
        mask = (h_new > 1e-10) & (h_new < 1.0 - 1e-10)
        if mask.any():
            h_new[mask] += diff / mask.sum()

    h[m:] = 1.0 - q[::-1]
    return h

def grad_row(h, lag, n):
    """
    Gradient of the overlap at a given lag index (in the full correlate array)
    with respect to h.
    lag = index in the full array, corresponds to shift d = lag - (n-1)
    """
    d = lag - (n - 1)
    g = np.zeros(n)
    # For the overlap c = sum_i h[i] * (1-h[i-d]) over valid i
    # d = shift, so we're computing sum over i where both i and i-d are in [0, n-1]
    # Valid i: max(0, d) <= i <= min(n-1, n-1+d)
    #         but also i-d >= 0 => i >= d, and i-d <= n-1 => i <= n-1+d
    #         and i >= 0, i <= n-1
    i_lo = max(0, d)
    i_hi = min(n-1, n-1+d)
    for i in range(i_lo, i_hi+1):
        j = i - d
        g[i] += (1.0 - h[j])
        g[j] += (-h[i])
    return g

def sym_grad(h, lag, n):
    """Gradient mapped to q-space for symmetric construction"""
    m = n // 2
    gh = grad_row(h, lag, n)
    # q[a] appears in h[a] with coeff +1 and h[n-1-a] with coeff -1
    # So ∂c/∂q[a] = ∂c/∂h[a] - ∂c/∂h[n-1-a]
    gq = gh[:m] - gh[n-1:m-1:-1]
    return gq

def target_direction_sym(q, h, active, target_val, dx, reg=0.01):
    """
    Find direction in q-space that moves active overlap values toward target_val.
    Returns (direction_dq, residual).
    """
    m = len(q)
    n = len(h)
    cvals = c5_array(h)
    
    # Build the Jacobian rows for active lags (in q-space)
    # J[k, a] = ∂c_val[active[k]] / ∂q[a]
    K = len(active)
    J = np.zeros((K, m))
    for k_idx, lag in enumerate(active):
        J[k_idx, :] = sym_grad(h, lag, n) * dx
    
    # Target: we want J @ dq such that c_vals[active] + J @ dq ≈ target_val
    # So: J @ dq ≈ target_val - c_vals[active]
    rhs = target_val - cvals[active]
    
    # Solve (J^T J + reg * I) dq = J^T rhs
    # This is the least-norm solution with ridge regularization
    A = J.T @ J + reg * np.eye(m)
    b = J.T @ rhs
    try:
        dq = np.linalg.solve(A, b)
    except np.linalg.LinAlgError:
        dq = np.linalg.pinv(A) @ b
    
    return dq

def target_direction_full(h, active, target_val, dx, reg=0.01):
    """
    Find direction in full h-space that moves active overlap values toward target_val.
    Maintains sum constraint by projecting out the mean direction.
    """
    n = len(h)
    cvals = c5_array(h)
    
    K = len(active)
    J = np.zeros((K, n))
    for k_idx, lag in enumerate(active):
        J[k_idx, :] = grad_row(h, lag, n) * dx
    
    rhs = target_val - cvals[active]
    
    # Solve with ridge, then project out the all-ones direction (to maintain sum)
    A = J.T @ J + reg * np.eye(n)
    b = J.T @ rhs
    try:
        dh = np.linalg.solve(A, b)
    except np.linalg.LinAlgError:
        dh = np.linalg.pinv(A) @ b
    
    # Project out the mean (to maintain sum constraint)
    dh -= np.mean(dh)
    
    return dh

    K = len(active)
    if K == 0:

    J = np.zeros((K, m))

    A = J.T @ J + reg * np.eye(m)
    b = J.T @ rhs
    try:
        dq = np.linalg.solve(A, b)
    except:
        dq = np.linalg.pinv(A) @ b
    return dq

    K = len(active)
    if K == 0:
        return np.zeros(m)
    J = np.zeros((K, m))
    for ki, lag in enumerate(active):
        J[ki, :] = grad_to_q(grad_h_full(h, lag, n), n) * dx
    rhs = target - cvals[active]
    A = J.T @ J + reg * np.eye(m)
    b = J.T @ rhs
    try:
        dq = np.linalg.solve(A, b)
    except np.linalg.LinAlgError:
        dq = np.linalg.pinv(A) @ b
    return dq


def target_dir_full(h, active, target, dx, reg=0.01):
    n = len(h)
    cvals = c5_array(h)
    K = len(active)
    if K == 0:
        return np.zeros(n)
    J = np.zeros((K, n))
    for ki, lag in enumerate(active):
        J[ki, :] = grad_h_full(h, lag, n) * dx
    rhs = target - cvals[active]
    A = J.T @ J + reg * np.eye(n)
    b = J.T @ rhs
    try:
        dh = np.linalg.solve(A, b)
    except np.linalg.LinAlgError:
        dh = np.linalg.pinv(A) @ b
    dh -= np.mean(dh)
    return dh


def line_search_sym(q, n, dq, c_cur, alphas=None):
    if alphas is None:
        alphas = [2.0, 1.0, 0.7, 0.5, 0.35, 0.25, 0.15, 0.08, 0.04, 0.02]
    best_q = None
    best_c = c_cur
    for alpha in alphas:
        q_try = np.clip(q + alpha * dq, 0.0, 1.0)
        h_try = build_sym(q_try, n)
        c_try = eval_c5(h_try)
        if c_try < best_c:
            best_q = q_try.copy()
            best_c = c_try
    return best_q, best_c


def line_search_full(h, n, dh, c_cur, alphas=None):
    if alphas is None:
        alphas = [2.0, 1.0, 0.7, 0.5, 0.35, 0.25, 0.15, 0.08, 0.04, 0.02]
    best_h = None
    best_c = c_cur
    for alpha in alphas:
        h_try = np.clip(h + alpha * dh, 0.0, 1.0)
        h_try = project_sum(h_try, n / 2.0)
        c_try = eval_c5(h_try)
        if c_try < best_c:
            best_h = h_try.copy()
            best_c = c_try
    return best_h, best_c


def interp_to(h, n_new):
    n_old = len(h)
    if n_new == n_old:
        return h.copy()
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h)
    return project_sum(h_new, n_new / 2.0)


def random_q_sym(n, rng, scale=0.1):
    m = n // 2
    t = np.linspace(0, 1, m)
    q = 0.5 + scale * (
        rng.uniform(-1, 1) * np.sin(2 * np.pi * t + rng.uniform(0, 6.28))
        + rng.uniform(-1, 1) * np.sin(4 * np.pi * t + rng.uniform(0, 6.28)) * 0.5
    )
    return np.clip(q, 0.0, 1.0)

def obj_smooth(h_flat, n, active, base_val, beta=100.0):
    h_use = np.clip(h_flat[:n], 0.0, 1.0)
    h_use = project_sum(h_use, n / 2.0)
    cv = c5_array(h_use)
    vals = cv[active]
    m = vals.max()
    shifted = vals - m
    return m + np.log(np.exp(beta * shifted).sum() / len(active)) / beta


def obj_smooth_grad(h_flat, n, active, base_val, beta=100.0):
    h_use = np.clip(h_flat[:n], 0.0, 1.0)
    h_use = project_sum(h_use, n / 2.0)
    cv = c5_array(h_use)
    vals = cv[active]
    m = vals.max()
    shifted = vals - m
    exps = np.exp(beta * shifted)
    weights = exps / exps.sum()
    g = np.zeros(n)
    dx = 2.0 / n
    for ki in range(len(active)):
        g += weights[ki] * grad_h_full(h_use, active[ki], n) * dx
    return obj_smooth(h_flat, n, active, base_val, beta), g
