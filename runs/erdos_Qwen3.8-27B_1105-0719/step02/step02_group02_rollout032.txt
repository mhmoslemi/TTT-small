```python
        solved = False

                d = np.linalg.solve(A, -grad_proj)
                solved = True
                break
```

```python
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time


def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-13:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h


def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def c5_of_h(h, dx):
    return float(compute_corrs(h).max() * dx)


def grad_corr_k(h, k, n):
    offset = k - (n - 1)
    grad = np.zeros(n)
    if offset == 0:
        grad[:] = 1.0 - 2.0 * h
        return grad
    i_lo = max(0, -offset)
    i_hi = min(n - 1, n - 1 - offset)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx + offset] += (1.0 - h[idx])
    grad[idx] += (-h[idx + offset])
    return grad


def hessian_corr_k(k, n):
    H = np.zeros((n, n))
    offset = k - (n - 1)
    if offset == 0:
        H[np.arange(n), np.arange(n)] = -2.0
        return H
    i_lo = max(0, -offset)
    i_hi = min(n - 1, n - 1 - offset)
    if i_lo > i_hi:
        return H
    idx = np.arange(i_lo, i_hi + 1)
    jdx = idx + offset
    H[idx, jdx] = -1.0
    H[jdx, idx] = -1.0
    return H


def smooth_max_val(h, dx, beta):
    return float(logsumexp(beta * compute_corrs(h)) / beta * dx)


def smooth_max_grad(h, dx, beta):
    n = len(h)
    corrs = compute_corrs(h)
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones_like(corrs) / len(corrs)
    else:
        w /= w_sum
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        grad += w[k] * grad_corr_k(h, k, n)
    return grad * dx


class SLSQPProblem:
    def __init__(self, dx, beta, target_sum):
        self.dx = dx
        self.beta = beta
        self.target_sum = target_sum

    def obj(self, x):
        return smooth_max_val(x, self.dx, self.beta)

    def grad(self, x):
        return smooth_max_grad(x, self.dx, self.beta)

    def con_fun(self, x):
        return x.sum() - self.target_sum

    def con_jac(self, x):
        return np.ones(len(x))


def slsqp_opt(h0, dx, target_sum, beta, maxiter):
    n = len(h0)
    prob = SLSQPProblem(dx, beta, target_sum)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': prob.con_fun, 'jac': prob.con_jac}]
    try:
        res = minimize(prob.obj, h0, jac=prob.grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        h_out = project_feasible(res.x, target_sum)
    except Exception:
        h_out = project_feasible(h0, target_sum)
    return h_out, c5_of_h(h_out, dx)

            break

        threshold = max_c - active_tol / dx
        active_mask = corrs >= threshold
        n_active = active_mask.sum()
        if n_active == 0:
            active_mask = np.zeros(2 * n - 1, dtype=bool)
            active_mask[np.argmax(corrs)] = True
            n_active = 1

        grad = np.zeros(n)
        for k in np.where(active_mask)[0]:
            grad += grad_corr_k(h, k, n)
        grad /= n_active
        grad *= dx
        grad_proj = grad - (grad.dot(one_vec) / n) * one_vec

        h_new = project_feasible(h - alpha * grad_proj, target_sum)
        new_c5 = c5_of_h(h_new, dx)

        if new_c5 < c5_val - 1e-16:
            h = h_new
            alpha = min(alpha * 1.03, 0.3)
        else:
            alpha *= 0.85

        if alpha < 1e-14:
            break

    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def remaining():
        return budget_s - (time.time() - t0)

    best_h = None
    best_c5 = float('inf')

    h0 = np.array(initial_h_values, dtype=float)
    n = len(h0)
    dx = 2.0 / n
    target_sum = n / 2.0
    h0 = project_feasible(h0, target_sum)
    best_h = h0.copy()
    best_c5 = c5_of_h(h0, dx)

    # Phase 1: SLSQP warmup
    for beta in [30, 80, 200, 500, 1000, 2000, 4000]:
        if remaining() < 12:
            break
        h_opt, c5_val = slsqp_opt(best_h, dx, target_sum, beta, 600)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_opt

    # Phase 2: Newton method
    if remaining() > 100:
        h_newt, c5_newt = newton_method(best_h, dx, target_sum, max_iter=8000)
        if c5_newt < best_c5:
            best_c5 = c5_newt
            best_h = h_newt

    # Phase 3: High-beta SLSQP polish
    if remaining() > 50:
        for beta in [5000, 10000, 20000, 40000]:
            if remaining() < 15:
                break
            h_opt, c5_val = slsqp_opt(best_h, dx, target_sum, beta, 800)
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h_opt

    # Phase 4: Newton again after SLSQP polish
    if remaining() > 60:
        h_newt, c5_newt = newton_method(best_h, dx, target_sum, max_iter=5000, tol_start=0.005, tol_end=0.0005)
        if c5_newt < best_c5:
            best_c5 = c5_newt
            best_h = h_newt

    # Phase 5: Try nearby n values
    n_cur = len(best_h)
    n_candidates = list(range(max(30, n_cur - 15), n_cur + 20))
    n_candidates = sorted(set(n_candidates))
    for n_try in n_candidates:
        if remaining() < 35:
            break
        if n_try < 20:
            continue
        dx_try = 2.0 / n_try
        ts_try = n_try / 2.0
        x_old = np.arange(len(best_h)) * (2.0 / len(best_h))
        x_new = np.arange(n_try) * dx_try
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, ts_try)
        h_opt, c5_val = slsqp_opt(h_interp, dx_try, ts_try, 400, 500)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_opt
            if remaining() > 40:
                dx_p = 2.0 / len(best_h)
                ts_p = len(best_h) / 2.0
                h_newt, c5_newt = newton_method(best_h, dx_p, ts_p, max_iter=4000)
                if c5_newt < best_c5:
                    best_c5 = c5_newt
                    best_h = h_newt
            if remaining() > 25:
                dx_p = 2.0 / len(best_h)
                ts_p = len(best_h) / 2.0
                h_opt2, c5_val2 = slsqp_opt(best_h, dx_p, ts_p, 10000, 600)
                if c5_val2 < best_c5:
                    best_c5 = c5_val2
                    best_h = h_opt2

    # Phase 6: Multi-start with perturbations
    n_b = len(best_h)
    dx_b = 2.0 / n_b
    ts_b = n_b / 2.0
    for trial in range(12):
        if remaining() < 12:
            break
        mag = 0.02 + 0.06 * (trial / 12.0)
        h_pert = best_h + rng.normal(0, mag, n_b)
        h_pert = project_feasible(h_pert, ts_b)
        h_opt, c5_val = slsqp_opt(h_pert, dx_b, ts_b, 300, 400)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_opt
            if remaining() > 25:
                h_newt, c5_newt = newton_method(best_h, dx_b, ts_b, max_iter=3000)
                if c5_newt < best_c5:
                    best_c5 = c5_newt
                    best_h = h_newt
                if remaining() > 15:
                    h_opt2, c5_val2 = slsqp_opt(best_h, dx_b, ts_b, 8000, 500)
                    if c5_val2 < best_c5:
                        best_c5 = c5_val2
                        best_h = h_opt2

    # Phase 7: Binary multi-start
    for trial in range(10):
        if remaining() < 12:
            break
        h_bin = np.zeros(n_b)
        idx = rng.permutation(n_b)
        n_ones = int(ts_b)
        h_bin[idx[:n_ones]] = 1.0
        frac = ts_b - n_ones
        if frac > 1e-10 and n_ones < n_b:
            h_bin[idx[n_ones]] = frac
        h_bin = project_feasible(h_bin, ts_b)
        h_opt, c5_val = slsqp_opt(h_bin, dx_b, ts_b, 300, 350)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_opt
            if remaining() > 20:
                h_newt, c5_newt = newton_method(best_h, dx_b, ts_b, max_iter=3000)
                if c5_newt < best_c5:
                    best_c5 = c5_newt
                    best_h = h_newt

    # Phase 8: Subgradient as backup
    if remaining() > 60:
        h_sg, c5_sg = backtracking_subgrad(best_h, dx_b, ts_b, max_iter=20000)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg

    # Phase 9: Final intensive Newton + SLSQP
    n_b = len(best_h)
    dx_b = 2.0 / n_b
    ts_b = n_b / 2.0
    if remaining() > 30:
        h_newt, c5_newt = newton_method(best_h, dx_b, ts_b, max_iter=10000, tol_start=0.003, tol_end=0.0003)
        if c5_newt < best_c5:
            best_c5 = c5_newt
            best_h = h_newt

    if remaining() > 20:
        for beta in [20000, 50000]:
            if remaining() < 12:
                break
            h_opt, c5_val = slsqp_opt(best_h, dx_b, ts_b, beta, 800)
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h_opt

    # Final Newton squeeze
    if remaining() > 10:
        h_newt, c5_newt = newton_method(best_h, dx_b, ts_b, max_iter=5000, tol_start=0.002, tol_end=0.0002)
        if c5_newt < best_c5:
            best_c5 = c5_newt
            best_h = h_newt

    return best_h, best_c5, len(best_h)
```
