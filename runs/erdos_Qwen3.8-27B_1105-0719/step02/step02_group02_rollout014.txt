def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
            # Second correction
            diff2 = target_sum - h.sum()
            if abs(diff2) > 1e-14:
                mask2 = (h > 1e-12) & (h < 1.0 - 1e-12)
                if mask2.sum() > 0:
                    h[mask2] += diff2 / mask2.sum()
                    h = np.clip(h, 0.0, 1.0)
    return h

def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def c5_of_h(h, dx):
    return float(compute_corrs(h).max() * dx)

def grad_corr_k(h, k, n):
    offset = k - (n - 1)
    grad = np.zeros(n)
    if offset == 0:
        grad[:] = 1.0 - 2.0 * h
        return grad
    i_lo = max(0, -offset)
    i_hi = min(n - 1, n - 1 - offset)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    if offset > 0:
        # corr_k = sum_{i=0}^{n-1-offset} h[i] * (1-h[i+offset])
        grad[idx] += (1.0 - h[idx + offset])  # derivative w.r.t. h[i]
        grad[idx + offset] += (-h[idx])       # derivative w.r.t. h[i+offset]
    else:
        # offset < 0, let d = -offset
        # corr_k = sum_{i=d}^{n-1} h[i-d] * (1-h[i])
        d = -offset
        grad[idx - d] += (1.0 - h[idx])  # derivative w.r.t. h[i-d]
        grad[idx] += (-h[idx - d])       # derivative w.r.t. h[i]
    return grad

            v *= 0.2

        if alpha < 1e-14:
            break
