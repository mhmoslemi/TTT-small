    if res.status != 0:
        return None
    
    q_new = res.x[:m]
    return q_new

    A = np.array(A_rows)
    b = np.array(b_rows)
    c = np.zeros(m + 1)
    c[m] = 1.0
    
    lb = np.maximum(0.0, q - rho)
    ub = np.minimum(1.0, q + rho)
    bounds = list(zip(lb, ub)) + [(0.0, corrs.max() * 1.5)]
    
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if res.status != 0:
        return None
    return res.x[:m]

def lp_full_step(h, n_full, active_k, rho):
    """Solve tangent LP for full vector with sum constraint."""
    n = n_full
    corrs = corr_raw(h)
    
    A_rows = []
    b_rows = []
    for ell in active_k:
        g = corr_grad_k(h, ell)
        A_rows.append(np.append(g, -1.0))
        b_rows.append(g @ h - corrs[ell])
    
    if len(A_rows) == 0:
        return None
    
    A = np.array(A_rows)
    b = np.array(b_rows)
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    lb = np.maximum(0.0, h - rho)
    ub = np.minimum(1.0, h + rho)
    bounds = list(zip(lb, ub)) + [(0.0, corrs.max() * 1.5)]
    
    A_eq = np.ones((1, n))
    b_eq = np.array([n / 2.0])
    
    res = linprog(c, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    if res.status != 0:
        return None
    return res.x[:n]

def smooth_max_sym(q, n_full, beta, maxiter=100):
    """Smoothed max optimization for symmetric half vector."""
    m = n_full // 2
    dx = 2.0 / n_full
    
    def obj(x):
        h = build_sym_h(x)
        corrs = corr_raw(h) * dx
        max_c = corrs.max()
        shifted = corrs - max_c
        w = np.exp(beta * shifted)
        w_sum = w.sum()
        if w_sum < 1e-30:
            return max_c
        return max_c + np.log(w_sum) / beta
    
    def jac(x):
        h = build_sym_h(x)
        corrs = corr_raw(h) * dx
        max_c = corrs.max()
        shifted = corrs - max_c
        w = np.exp(beta * shifted)
        w_sum = w.sum()
        if w_sum < 1e-30:
            g = np.zeros(m)
            for r in range(m):
                g[r] = dx * (1 - 2*h[r])  # only d=0 matters for uniform
            return g
        w /= w_sum
        g = np.zeros(m)
        for ell in range(2 * n_full - 1):
            if w[ell] < 1e-14:
                continue
            g_h = corr_grad_k(h, ell)
            g += w[ell] * (g_h[:m] - g_h[n_full-1::-1][:m])
        g *= dx
        return g
    
    bounds = [(0.0, 1.0)] * m
    try:
        res = minimize(obj, q, jac=jac, method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10})
        return np.clip(res.x, 0, 1)
    except Exception:
        return q.copy()

def transfer_search_sym(q, n_full, n_trials=50, rng=None):

import numpy as np
from scipy.optimize import linprog, minimize
import time

def eval_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def corr_raw(h):
    return np.correlate(h, 1.0 - h, mode='full')

def corr_grad_k(h, k):
    n = len(h)
    d = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -d)
    i_hi = min(n - 1, n - 1 - d)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx] += (1.0 - h[idx + d])
    grad[idx + d] += (-h[idx])
    return grad

def build_sym_h(q):
    m = len(q)
    h = np.empty(2 * m)
    h[:m] = q
    h[m:] = 1.0 - q[::-1]
    return h

def sym_from_h(h):
    n = len(h)
    m = n // 2
    return np.clip(0.5 * (h[:m] + 1.0 - h[n-1::-1][:m]), 0.0, 1.0)

def project_feasible(x, target_sum):
    n = len(x)
    x = np.asarray(x, dtype=float).copy()
    x = np.where(np.isfinite(x), x, 0.5)
    x = np.clip(x, 0.0, 1.0)
    if abs(x.sum() - target_sum) < 1e-12:
        return x
    lo, hi = -2.0, 2.0
    for _ in range(90):
        mid = (lo + hi) / 2.0
        s = np.clip(x + mid, 0.0, 1.0).sum()
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    x = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - x.sum()
    if abs(diff) > 1e-12:
        interior = (x > 1e-10) & (x < 1.0 - 1e-10)
        if interior.sum() > 0:
            x[interior] += diff / interior.sum()
    return np.clip(x, 0.0, 1.0)

def select_active(corrs, margin_frac=0.02, neighbor_pad=2, max_k=40):
    max_val = corrs.max()
    threshold = max_val - max(1e-7, margin_frac * max_val)
    active = set(np.where(corrs >= threshold)[0].tolist())
    padded = set()
    for k in active:
        for d in range(-neighbor_pad, neighbor_pad + 1):
            if 0 <= k + d < len(corrs):
                padded.add(k + d)
    active_list = sorted(padded)
    if len(active_list) > max_k:
        scored = sorted([(corrs[k], k) for k in active_list], reverse=True)
        active_list = sorted([k for _, k in scored[:max_k]])
    return active_list

def line_search_sym(q0, direction, n_full, n_grid=25, n_refine=20):
    c5_0 = eval_score(build_sym_h(q0))
    best_alpha = 0.0
    best_c5 = c5_0
    alphas = np.linspace(0, 1, n_grid)
    for a in alphas:
        q_try = np.clip(q0 + a * direction, 0, 1)
        c5_try = eval_score(build_sym_h(q_try))
        if c5_try < best_c5:
            best_c5 = c5_try
            best_alpha = a
    if best_alpha > 0 and best_alpha < 1:
        i = int(best_alpha * (n_grid - 1))
        lo = alphas[max(0, i - 1)]
        hi = alphas[min(n_grid - 1, i + 1)]
        for _ in range(n_refine):
            m1 = lo + (hi - lo) / 3.0
            m2 = hi - (hi - lo) / 3.0
            c1 = eval_score(build_sym_h(np.clip(q0 + m1 * direction, 0, 1)))
            c2 = eval_score(build_sym_h(np.clip(q0 + m2 * direction, 0, 1)))
            if c1 < c2:
                hi = m2
            else:
                lo = m1
            a_try = (lo + hi) / 2.0
            c_try = eval_score(build_sym_h(np.clip(q0 + a_try * direction, 0, 1)))
            if c_try < best_c5:
                best_c5 = c_try
                best_alpha = a_try
    return best_alpha, best_c5

def lp_sym_step(q, n_full, active_k, rho):
    m = n_full // 2
    h = build_sym_h(q)
    corrs = corr_raw(h)
    A_rows = []
    b_rows = []
    for ell in active_k:
        g_h = corr_grad_k(h, ell)
        g_q = g_h[:m] - g_h[n_full - 1::-1][:m]
        A_rows.append(np.append(g_q, -1.0))
        b_rows.append(g_q @ q - corrs[ell])
    if len(A_rows) == 0:
        return None
    A = np.array(A_rows)
    b = np.array(b_rows)
    c_vec = np.zeros(m + 1)
    c_vec[m] = 1.0
    lb = np.maximum(0.0, q - rho)
    ub = np.minimum(1.0, q + rho)
    bounds = list(zip(lb, ub)) + [(0.0, corrs.max() * 2.0 + 1.0)]
    try:
        res = linprog(c_vec, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    except Exception:
        try:
            res = linprog(c_vec, A_ub=A, b_ub=b, bounds=bounds, method='revised simplex')
        except Exception:
            return None
    if res.status != 0:
        return None
    return res.x[:m]

def lp_full_step(h, n_full, active_k, rho):
    n = n_full
    corrs = corr_raw(h)
    A_rows = []
    b_rows = []
    for ell in active_k:
        g = corr_grad_k(h, ell)
        A_rows.append(np.append(g, -1.0))
        b_rows.append(g @ h - corrs[ell])
    if len(A_rows) == 0:
        return None
    A = np.array(A_rows)
    b = np.array(b_rows)
    c_vec = np.zeros(n + 1)
    c_vec[n] = 1.0
    lb = np.maximum(0.0, h - rho)
    ub = np.minimum(1.0, h + rho)
    bounds = list(zip(lb, ub)) + [(0.0, corrs.max() * 2.0 + 1.0)]
    A_eq = np.ones((1, n))
    b_eq = np.array([n / 2.0])
    try:
        res = linprog(c_vec, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    except Exception:
        try:
            res = linprog(c_vec, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='revised simplex')
        except Exception:
            return None
    if res.status != 0:
        return None
    return res.x[:n]

def smooth_max_sym(q, n_full, beta, maxiter=80):
    m = n_full // 2
    dx = 2.0 / n_full

    def obj(jx):
        h = build_sym_h(jx)
        corrs = corr_raw(h) * dx
        mx = corrs.max()
        sh = corrs - mx
        w = np.exp(beta * sh)
        ws = w.sum()
        if ws < 1e-30:
            return mx
        return mx + np.log(ws) / beta

    def jac(jx):
        h = build_sym_h(jx)
        corrs = corr_raw(h) * dx
        mx = corrs.max()
        sh = corrs - mx
        w = np.exp(beta * sh)
        ws = w.sum()
        if ws < 1e-30:
            g_h = corr_grad_k(h, n_full - 1)
            return dx * (g_h[:m] - g_h[n_full - 1::-1][:m])
        w /= ws
        g = np.zeros(m)
        for ell in range(2 * n_full - 1):
            if w[ell] < 1e-13:
                continue
            g_h = corr_grad_k(h, ell)
            g += w[ell] * (g_h[:m] - g_h[n_full - 1::-1][:m])
        return g * dx

    bounds = [(0.0, 1.0)] * m
    try:
        res = minimize(obj, q, jac=jac, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10})
        return np.clip(res.x, 0, 1)
    except Exception:
        return q.copy()

def coord_search_sym(q, n_full, n_rounds=5, rng=None):
    m = n_full // 2
    if rng is None:
        rng = np.random.RandomState(0)
    c5_cur = eval_score(build_sym_h(q))
    for _ in range(n_rounds):
        improved = False
        order = rng.permutation(m)
        for r in order:
            lo = -q[r]
            hi = 1.0 - q[r]
            if hi - lo < 1e-12:
                continue
            best_d = 0.0
            best_c5 = c5_cur
            deltas = np.linspace(lo, hi, 11)
            for d in deltas:
                q_try = q.copy()
                q_try[r] += d
                cc = eval_score(build_sym_h(q_try))
                if cc < best_c5:
                    best_c5 = cc
                    best_d = d
            if best_d != 0.0:
                a = max(lo, best_d - (hi - lo) * 0.1)
                b = min(hi, best_d + (hi - lo) * 0.1)
                for _ in range(12):
                    m1 = a + (b - a) / 3.0
                    m2 = b - (b - a) / 3.0
                    q1 = q.copy(); q1[r] += m1
                    q2 = q.copy(); q2[r] += m2
                    c1 = eval_score(build_sym_h(q1))
                    c2 = eval_score(build_sym_h(q2))
                    if c1 < c2:
                        b = m2
                    else:
                        a = m1
                dt = (a + b) / 2.0
                qt = q.copy(); qt[r] += dt
                ct = eval_score(build_sym_h(qt))
                if ct < best_c5:
                    best_c5 = ct
                    best_d = dt
            if best_c5 < c5_cur - 1e-13:
                q[r] += best_d
                c5_cur = best_c5
                improved = True
        if not improved:
            break
    return q, c5_cur

def transfer_search_full(h, n_full, n_trials=25, rng=None):
    n = n_full
    if rng is None:
        rng = np.random.RandomState(0)
    c5_cur = eval_score(h)
    for _ in range(n_trials):
        a, b = rng.choice(n, 2, replace=False)
        lo = max(-h[a], h[b] - 1.0)
        hi = min(1.0 - h[a], -h[b])
        if hi - lo < 1e-12:
            continue
        best_d = 0.0
        best_c5 = c5_cur
        for d in np.linspace(lo, hi, 11):
            ht = h.copy()
            ht[a] += d
            ht[b] -= d
            cc = eval_score(ht)
            if cc < best_c5:
                best_c5 = cc
                best_d = d
        if best_d != 0.0:
            ad = max(lo, best_d - (hi - lo) * 0.1)
            bd = min(hi, best_d + (hi - lo) * 0.1)
            for _ in range(12):
                m1 = ad + (bd - ad) / 3.0
                m2 = bd - (bd - ad) / 3.0
                h1 = h.copy(); h1[a] += m1; h1[b] -= m1
                h2 = h.copy(); h2[a] += m2; h2[b] -= m2
                c1 = eval_score(h1)
                c2 = eval_score(h2)
                if c1 < c2:
                    bd = m2
                else:
                    ad = m1
            dt = (ad + bd) / 2.0
            ht = h.copy(); ht[a] += dt; ht[b] -= dt
            ct = eval_score(ht)
            if ct < best_c5:
                best_c5 = ct
                best_d = dt
        if best_c5 < c5_cur - 1e-13:
            h = h.copy()
            h[a] += best_d
            h[b] -= best_d
            c5_cur = best_c5
    return h, c5_cur

def lp_refine_sym(q, n_full, max_iters=200, time_limit=None):
    m = n_full // 2
    c5_cur = eval_score(build_sym_h(q))
    rho = 0.4
    stagnant = 0
    for it in range(max_iters):
        if time_limit is not None and time.time() > time_limit:
            break
        h_cur = build_sym_h(q)
        corrs = corr_raw(h_cur)
        active = select_active(corrs, margin_frac=0.02, neighbor_pad=2, max_k=35)
        q_new = lp_sym_step(q, n_full, active, rho)
        if q_new is None:
            rho *= 0.7
            stagnant += 1
            if stagnant > 15 or rho < 0.005:
                break
            continue
        direction = q_new - q
        if np.linalg.norm(direction) < 1e-12:
            stagnant += 1
            if stagnant > 5:
                break
            rho *= 0.7
            continue
        alpha, c5_try = line_search_sym(q, direction, n_full, n_grid=20, n_refine=18)
        if c5_try < c5_cur - 1e-12:
            q = np.clip(q + alpha * direction, 0, 1)
            c5_cur = c5_try
            stagnant = 0
            if alpha > 0.9:
                rho = min(0.5, rho * 1.15)
        else:
            c5_direct = eval_score(build_sym_h(np.clip(q_new, 0, 1)))
            if c5_direct < c5_cur - 1e-12:
                q = np.clip(q_new, 0, 1)
                c5_cur = c5_direct
                stagnant = 0
            else:
                rho *= 0.75
                stagnant += 1
                if stagnant > 20 or rho < 0.005:
                    break
    return q, c5_cur

def lp_refine_full(h, n_full, max_iters=100, time_limit=None):
    n = n_full
    c5_cur = eval_score(h)
    rho = 0.25
    stagnant = 0
    for it in range(max_iters):
        if time_limit is not None and time.time() > time_limit:
            break
        corrs = corr_raw(h)
        active = select_active(corrs, margin_frac=0.02, neighbor_pad=2, max_k=35)
        h_new = lp_full_step(h, n_full, active, rho)
        if h_new is None:
            rho *= 0.7
            stagnant += 1
            if stagnant > 15 or rho < 0.005:
                break
            continue
        direction = h_new - h
        if np.linalg.norm(direction) < 1e-12:
            stagnant += 1
            if stagnant > 5:
                break
            rho *= 0.7
            continue
        best_a = 0.0
        best_c5 = c5_cur
        for a in [0.1, 0.25, 0.5, 0.75, 1.0]:
            ht = project_feasible(h + a * direction, n / 2.0)
            ct = eval_score(ht)
            if ct < best_c5:
                best_c5 = ct
                best_a = a
        if best_c5 < c5_cur - 1e-12 and best_a > 0:
            h = project_feasible(h + best_a * direction, n / 2.0)
            c5_cur = best_c5
            stagnant = 0
        else:
            rho *= 0.75
            stagnant += 1
            if stagnant > 20 or rho < 0.005:
                break
    return h, c5_cur

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def rem():
        return budget_s - (time.time() - t0)

    best_h = None
    best_c5 = float('inf')

    try:
        h_parent = np.array(initial_h_values, dtype=float).copy().reshape(-1)
    except (NameError, AttributeError):
        h_parent = None

    if h_parent is not None and len(h_parent) >= 20:
        n_p = len(h_parent)
        h_parent = project_feasible(h_parent, n_p / 2.0)
        best_h = h_parent.copy()
        best_c5 = eval_score(h_parent)
    else:
        m_p = 63
        x = np.linspace(0, 1, m_p)
        q_p = np.clip(0.5 + 0.35 * np.sin(2 * np.pi * x) + 0.1 * np.cos(4 * np.pi * x), 0, 1)
        best_h = build_sym_h(q_p)
        best_c5 = eval_score(best_h)

    n = len(best_h)

    # Phase 1: Symmetric smoothing
    if n % 2 == 0 and rem() > 250:
        q = sym_from_h(best_h)
        for beta in [50, 120, 250]:
            if rem() < 60:
                break
            q = smooth_max_sym(q, n, beta, maxiter=60)
            ht = build_sym_h(q)
            ct = eval_score(ht)
            if ct < best_c5:
                best_c5 = ct
                best_h = ht.copy()

    # Phase 2: LP refinement (main driver)
    if n % 2 == 0 and rem() > 250:
        q = sym_from_h(best_h)
        tl = time.time() + min(350, rem() * 0.5)
        q, c5_lp = lp_refine_sym(q, n, max_iters=300, time_limit=tl)
        if c5_lp < best_c5:
            best_c5 = c5_lp
            best_h = build_sym_h(q).copy()

    # Phase 3: Coordinate searches (fine-tuning)
    if n % 2 == 0 and rem() > 150:
        q = sym_from_h(best_h)
        for batch in range(4):
            if rem() < 25:
                break
            q, c5_cs = coord_search_sym(q, n, n_rounds=4, rng=rng)
            if c5_cs < best_c5:
                best_c5 = c5_cs
                best_h = build_sym_h(q).copy()

    # Phase 4: Another LP pass after coordinate refinement
    if n % 2 == 0 and rem() > 150:
        q = sym_from_h(best_h)
        tl = time.time() + min(200, rem() * 0.5)
        q, c5_lp2 = lp_refine_sym(q, n, max_iters=200, time_limit=tl)
        if c5_lp2 < best_c5:
            best_c5 = c5_lp2
            best_h = build_sym_h(q).copy()
        if rem() > 40:
            q, c5_cs2 = coord_search_sym(q, n, n_rounds=3, rng=rng)
            if c5_cs2 < best_c5:
                best_c5 = c5_cs2
                best_h = build_sym_h(q).copy()

    # Phase 5: Multi-resolution exploration
    n_base = len(best_h)
    deltas_to_try = [-4, -2, 2, 4, -6, 6, -8, 8, 12, -12]
    for dn in deltas_to_try:
        if rem() < 80:
            break
        nt = n_base + dn
        if nt < 44 or nt > 220 or nt % 2 != 0:
            continue
        xo = (np.arange(n_base) + 0.5) * 2.0 / n_base
        xn = (np.arange(nt) + 0.5) * 2.0 / nt
        ht = np.interp(xn, xo, best_h)
        ht = project_feasible(ht, nt / 2.0)
        qt = sym_from_h(ht)
        c5_t = eval_score(build_sym_h(qt))
        if rem() > 200:
            qt = smooth_max_sym(qt, nt, 100, maxiter=50)
        if rem() > 100:
            tl2 = time.time() + min(120, rem() * 0.5)
            qt, c5_t = lp_refine_sym(qt, nt, max_iters=100, time_limit=tl2)
        hf = build_sym_h(qt)
        c5_f = eval_score(hf)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = hf.copy()
        elif c5_f < best_c5 + 1e-7 and nt < n_base:
            best_c5 = c5_f
            best_h = hf.copy()

    # Phase 6: Full-vector optimization (break symmetry)
    if rem() > 120:
        nf = len(best_h)
        if nf % 2 != 0:
            nf += 1
            xo = (np.arange(len(best_h)) + 0.5) * 2.0 / len(best_h)
            xn = (np.arange(nf) + 0.5) * 2.0 / nf
            best_h = np.interp(xn, xo, best_h)
            best_h = project_feasible(best_h, nf / 2.0)
            best_c5 = eval_score(best_h)
        hf = best_h.copy()
        tl3 = time.time() + min(100, rem() * 0.5)
        hf, c5_fv = lp_refine_full(hf, nf, max_iters=80, time_limit=tl3)
        if c5_fv < best_c5:
            best_c5 = c5_fv
            best_h = hf.copy()
        if rem() > 40:
            hf2, c5_tf = transfer_search_full(hf, nf, n_trials=20, rng=rng)
            if c5_tf < best_c5:
                best_c5 = c5_tf
                best_h = hf2.copy()

    # Phase 7: Final polish
    if rem() > 25 and best_h is not None:
        nf = len(best_h)
        if nf % 2 == 0:
            qf = sym_from_h(best_h)
            ht = build_sym_h(qf)
            corrs = corr_raw(ht)
            active = select_active(corrs, margin_frac=0.01, neighbor_pad=1, max_k=25)
            try:
                qn = lp_sym_step(qf, nf, active, 0.1)
                if qn is not None:
                    d = qn - qf
                    if np.linalg.norm(d) > 1e-10:
                        a, ct = line_search_sym(qf, d, nf, n_grid=15, n_refine=12)
                        if ct < best_c5:
                            best_c5 = ct
                            best_h = build_sym_h(np.clip(qf + a * d, 0, 1)).copy()
            except Exception:
                pass
        else:
            hfin = best_h.copy()
            corrs = corr_raw(hfin)
            active = select_active(corrs, margin_frac=0.01, neighbor_pad=1, max_k=25)
            hnew = lp_full_step(hfin, nf, active, 0.08)
            if hnew is not None:
                d = hnew - hfin
                for a in [0.25, 0.5, 0.75, 1.0]:
                    ht = project_feasible(hfin + a * d, nf / 2.0)
                    ct = eval_score(ht)
                    if ct < best_c5:
                        best_c5 = ct
                        best_h = ht.copy()
                        break

    # Validate and return
    best_h = np.asarray(best_h, dtype=float).reshape(-1)
    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_c5 = eval_score(best_h)
    return best_h, best_c5, len(best_h)
