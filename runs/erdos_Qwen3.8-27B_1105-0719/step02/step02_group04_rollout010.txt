def active_target_step(q, n, eta, margin, max_active=50):
    """One step of active-set residual targeting in q-space."""
    m = n // 2
    h = build_sym(q, n)
    c_arr = eval_c5_array(h)
    C = c_arr.max()
    J = len(c_arr)  # 2n-1
    
    if C <= 0:
        return q, C, False
    
    # Active set: lags with c[j] >= C - margin
    thresh = C - margin
    active = np.where(c_arr >= thresh)[0]
    if len(active) > max_active:
        # Keep only the top max_active
        top_vals = c_arr[active]
        order = np.argsort(top_vals)[::-1][:max_active]
        active = active[order]
    
    # Also add neighbors
    expanded = set(active.tolist())
    for j in active:
        for delta in [-2, -1, 1, 2]:
            jj = j + delta
            if 0 <= jj < J:
                expanded.add(jj)
    active = np.array(sorted(expanded))
    
    # Build gradient matrix P (active x m)
    P = np.zeros((len(active), m))
    for i, j in enumerate(active):
        d = j - (n - 1)
        gh = corr_grad_row(h, d, n)
        P[i] = gh[:m] - gh[n-1:m-1:-1]
    
    # Target residuals
    target = C - eta
    r = (target - c_arr[active]) / (2.0 / n)  # = (target - c_arr[active]) * n / 2
    
    # Solve min ||delta_q||^2 s.t. P @ delta_q = r
    # Least norm: delta_q = P^T (P P^T + reg*I)^{-1} r
    try:
        PPt = P @ P.T
        reg = 1e-6
        PPt += reg * np.eye(len(active))
        delta_q = P.T @ np.linalg.solve(PPt, r)
    except np.linalg.LinAlgError:
        return q, C, False
    
    if not np.all(np.isfinite(delta_q)):
        return q, C, False
    
    # Line search
    best_q = q
    best_c = C
    best_s = 0.0
    for s in [1.0, 0.7, 0.5, 0.35, 0.2, 0.1, 0.05, 0.02]:
        q_new = np.clip(q + s * delta_q, 0.0, 1.0)
        h_new = build_sym(q_new, n)
        c_new = eval_c5(h_new)
        if c_new < best_c - 1e-12:
            best_c = c_new
            best_q = q_new
            best_s = s
    
    improved = best_c < C - 1e-12
    return best_q, best_c, improved

import numpy as np
from scipy.optimize import minimize
import time


def eval_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))


def eval_c5_array(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)


def build_sym(q, n):
    m = n // 2
    h = np.empty(n)
    h[:m] = q
    h[m:] = 1.0 - q[::-1]
    return h


def to_sym_q(h, n):
    m = n // 2
    return 0.5 * (h[:m] + 1.0 - h[n-1:m-1:-1])


def project_to_target(x, target_sum):
    lo, hi = -1.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(x - mid, 0.0, 1.0).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    offset = (lo + hi) * 0.5
    h = np.clip(x - offset, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)


def corr_grad_row(h, d, n):
    g = np.zeros(n)
    if d >= 0:
        g[d:] += (1.0 - h[:n - d])
        g[:n - d] -= h[d:]
    else:
        k = -d
        g[:n - k] += (1.0 - h[k:])
        g[k:] -= h[:n - k]
    return g


def sym_grad_row(h, d, n):
    m = n // 2
    gh = corr_grad_row(h, d, n)
    return gh[:m] - gh[n - 1:m - 1:-1]


def interp_to(h, n_new):
    n_old = len(h)
    if n_new == n_old:
        return h.copy()
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h)
    return project_to_target(h_new, n_new / 2.0)


def target_step(q, n, eta, margin, max_active=50, ridge=1e-5):
    m = n // 2
    h = build_sym(q, n)
    c_arr = eval_c5_array(h)
    C = c_arr.max()
    J = len(c_arr)

    if C <= 0:
        return q.copy(), C, False

    thresh = C - margin
    active = np.where(c_arr >= thresh)[0]
    if len(active) > max_active:
        top_vals = c_arr[active]
        order = np.argsort(top_vals)[::-1][:max_active]
        active = active[order]

    expanded = set(int(a) for a in active)
    for j in active:
        for delta in [-2, -1, 1, 2]:
            jj = int(j) + delta
            if 0 <= jj < J:
                expanded.add(jj)
    active = np.array(sorted(expanded), dtype=int)

    P = np.zeros((len(active), m))
    dx = 2.0 / n
    for i, j in enumerate(active):
        d = j - (n - 1)
        P[i] = sym_grad_row(h, d, n)

    r = (C - eta - c_arr[active]) / dx

    try:
        PPt = P @ P.T + ridge * np.eye(len(active))
        delta_q = P.T @ np.linalg.solve(PPt, r)
    except np.linalg.LinAlgError:
        try:

        except Exception:
            return q.copy(), C, False

    if not np.all(np.isfinite(delta_q)):
