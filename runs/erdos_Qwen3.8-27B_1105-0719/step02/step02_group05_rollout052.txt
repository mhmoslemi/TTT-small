    corrs = np.correlate(h, 1.0 - h, mode='full')
    return corrs.max() * dx

def project_to_bounds(x, lo=0.0, hi=1.0):
    return np.clip(x, lo, hi)

def build_h_sym(q, n):
    """Build h from symmetric q: h = [q, 1-q_rev] for even n"""
    m = n // 2
    return np.concatenate([q, 1.0 - q[::-1]])

def q_from_h_sym(h, n):
    """Extract q from symmetric h"""
    return h[:n//2].copy()

def compute_corr_single(h, d, n, dx):
    """Compute a single correlation value for shift d"""
    if d >= 0:
        if d >= n:
            return 0.0
        return np.dot(h[d:], 1.0 - h[:n-d]) * dx
    else:
        dd = -d
        if dd >= n:
            return 0.0
        return np.dot(h[dd:], 1.0 - h[:n-dd]) * dx

def grad_corr_single(h, d, n):
    """Gradient of correlation for shift d with respect to h"""
    g = np.zeros(n)
    if d >= 0:

            g[d:] += (1.0 - h[:n-d])
            g[:n-d] += (-h[d:])
    else:
        dd = -d
        if dd < n:
            g[:n-dd] += (1.0 - h[dd:])
            g[dd:] += (-h[:n-dd])
    return g

import cvxpy as cp

def solve_lp_active_set(q_current, n, active_k, t_current):
    """Solve the linearized active set problem (an LP)"""
    m = n // 2
    h = build_h_sym(q_current, n)
    dx = 2.0 / n
    corrs = np.correlate(h, 1.0 - h, mode='full')
    
    q_var = cp.Variable(m)
    t_var = cp.Variable()
    
    constraints = [q_var >= 0, q_var <= 1]
    
    for k in active_k:
        d = k - (n - 1)
        f_k = corrs[k] * dx
        G = grad_corr_single(h, d, n)
        Gq = (G[:m] - G[m:][::-1]) * dx
        # Linearization: f_k(q) ≈ f_k(q_current) + Gq^T (q - q_current)
        # = f_k(q_current) + Gq^T q - Gq^T q_current
        # ≤ t
        constraints.append(f_k + Gq @ q_var - Gq @ q_current <= t_var)
    
    obj = cp.Minimize(t_var)
    prob = cp.Problem(obj, constraints)
    prob.solve(solver=cp.SCS, max_iters=10000, eps=1e-8)
    
    if q_var.value is not None:
        return np.clip(q_var.value, 0, 1), t_var.value
    return q_current, t_current

from scipy.optimize import linprog

def solve_active_lp(q_cur, n, active_k):
    """Solve: min t s.t. f_k_lin(q) ≤ t, 0 ≤ q ≤ 1
    
    Variables: [t, q_0, ..., q_{m-1}] (m+1 vars)
    Constraints: for each k in active_k: -Gq_k^T q + t ≥ f_k(q_cur) - Gq_k^T q_cur
    i.e., Gq_k^T q - t ≤ f_k(q_cur) - Gq_k^T q_cur
    Plus: 0 ≤ q_i ≤ 1
    """
    m = n // 2
    h = build_h_sym(q_cur, n)
    dx = 2.0 / n
    corrs = np.correlate(h, 1.0 - h, mode='full')
    
    # Variables: x = [t, q_0, ..., q_{m-1}]
    # Objective: minimize t → c = [1, 0, 0, ..., 0]
    c = np.zeros(m + 1)
    c[0] = 1.0
    
    # Inequality constraints: A_ub @ x ≤ b_ub
    # For each active k: Gq_k^T q - t ≤ f_k(q_cur) - Gq_k^T q_cur
    # Rewrite: -t + Gq_k^T q ≤ f_k(q_cur) - Gq_k^T q_cur
    # In A_ub form: [-1, Gq_k] @ [t, q] ≤ f_k - Gq_k^T q_cur
    
    A_ub_list = []
    b_ub_list = []
    
    for k in active_k:
        d = k - (n - 1)
        f_k = corrs[k] * dx
        G = grad_corr_single(h, d, n)
        Gq = (G[:m] - G[m:][::-1]) * dx
        row = np.zeros(m + 1)
        row[0] = -1.0
        row[1:] = Gq
        A_ub_list.append(row)
        b_ub_list.append(f_k - Gq @ q_cur)
    
    if A_ub_list:
        A_ub = np.array(A_ub_list)
        b_ub = np.array(b_ub_list)
    else:
        A_ub = None
        b_ub = None
    
    # Bounds: 0 ≤ t, 0 ≤ q_i ≤ 1
    bounds = [(0, None)] + [(0, 1)] * m
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        t_new = res.x[0]
        q_new = res.x[1:]
        return q_new, t_new
    return q_cur, corrs[active_k].max() * dx

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.special import logsumexp
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corrs = np.correlate(h, 1.0 - h, mode='full')
    return corrs.max() * dx

def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def build_h_sym(q, n):
    m = n // 2
    return np.concatenate([q, 1.0 - q[::-1]])

def q_from_h_sym(h, n):
    return h[:n//2].copy()

def grad_corr_single(h, d, n):
    g = np.zeros(n)
    if d >= 0:
        if d < n:
            g[d:] += (1.0 - h[:n-d])
            g[:n-d] += (-h[d:])
    else:
        dd = -d
        if dd < n:
            g[:n-dd] += (1.0 - h[dd:])
            g[dd:] += (-h[:n-dd])
    return g

def grad_corrs_q(h, q, n, d_list):
    """Gradients of correlations for given shifts, projected to q-space"""
    m = n // 2
    result = np.zeros((len(d_list), m))
    for i, d in enumerate(d_list):
        G = grad_corr_single(h, d, n)
        result[i] = G[:m] - G[m:][::-1]
    return result

def solve_active_lp(q_cur, n, active_k):
    m = n // 2
    h = build_h_sym(q_cur, n)
    dx = 2.0 / n
    corrs = compute_corrs(h)
    
    c = np.zeros(m + 1)
    c[0] = 1.0
    
    A_ub_list = []
    b_ub_list = []
    
    for k in active_k:
        d = k - (n - 1)
        f_k = corrs[k] * dx
        G = grad_corr_single(h, d, n)
        Gq = (G[:m] - G[m:][::-1]) * dx
        row = np.zeros(m + 1)
        row[0] = -1.0
        row[1:] = Gq
        A_ub_list.append(row)
        b_ub_list.append(f_k - Gq @ q_cur)
    
    A_ub = np.array(A_ub_list) if A_ub_list else None
    b_ub = np.array(b_ub_list) if b_ub_list else None
    
    bounds = [(0, None)] + [(0, 1)] * m
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success and res.x is not None:
        return np.clip(res.x[1:], 0, 1), res.x[0]
    return q_cur.copy(), corrs[active_k].max() * dx

def smooth_obj_q(q, n, beta, k_list):
    h = build_h_sym(q, n)
    dx = 2.0 / n
    corrs = np.correlate(h, 1.0 - h, mode='full')
    if k_list is not None:
        corrs_use = corrs[k_list]
    else:
        corrs_use = corrs
    return logsumexp(beta * corrs_use) / beta * dx

def smooth_grad_q(q, n, beta, k_list):
    m = n // 2
    h = build_h_sym(q, n)
    dx = 2.0 / n
    corrs = np.correlate(h, 1.0 - h, mode='full')
    if k_list is not None:
        corrs_use = corrs[k_list]
        k_indices = k_list
    else:
        corrs_use = corrs
        k_indices = range(len(corrs))
    
    shifted = beta * (corrs_use - corrs_use.max())
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(len(corrs_use)) / len(corrs_use)
    else:
        w = w / w_sum
    
    Gq = np.zeros(m)
    for i, k in enumerate(k_indices):
        wi = w[i]
        if wi < 1e-15:
            continue
        d = k - (n - 1)
        G = grad_corr_single(h, d, n)
        Gq += wi * (G[:m] - G[m:][::-1])
    return Gq * dx

def smooth_optimize_q(q_init, n, betas, maxiter=300):
    """Continuation method with log-sum-exp smoothing"""
    q = q_init.copy()
    m = n // 2
    
    for beta in betas:
        # Use all correlations for the first few betas, then restrict
        h = build_h_sym(q, n)
        corrs = compute_corrs(h)
        dx = 2.0 / n
        c5 = corrs.max() * dx
        
        # Use correlations above threshold
        threshold = c5 - 0.02
        k_list = [k for k in range(len(corrs)) if corrs[k] * dx >= threshold]
        if len(k_list) < 5:
            k_list = list(np.argsort(corrs)[::-1][:10])
        
        def obj(x):
            return smooth_obj_q(x, n, beta, k_list)
        
        def grad(x):
            return smooth_grad_q(x, n, beta, k_list)
        
        try:
            res = minimize(obj, q, jac=grad, method='L-BFGS-B',
                          bounds=[(0, 1)] * m,
                          options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12})
            if res.fun < smooth_obj_q(q, n, beta, k_list):
                q = np.clip(res.x, 0, 1)
        except Exception:
            pass
    
    return q

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.special import logsumexp
import time


def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corrs = np.correlate(h, 1.0 - h, mode='full')
    return corrs.max() * dx


def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def build_h_sym(q, n):
    m = n // 2
    return np.concatenate([q, 1.0 - q[::-1]])


def q_from_h_sym(h, n):
    return h[:n // 2].copy()


def grad_corr_single(h, d, n):
    g = np.zeros(n)
    if d >= 0:
        if d < n:
            g[d:] += (1.0 - h[:n - d])
            g[:n - d] += (-h[d:])
    else:
        dd = -d
        if dd < n:
            g[:n - dd] += (1.0 - h[dd:])
            g[dd:] += (-h[:n - dd])
    return g


def grad_q_for_shift(h, d, n):
    m = n // 2
    G = grad_corr_single(h, d, n)
    return (G[:m] - G[m:])[::-1] * 0 + (G[:m] - G[m:][::-1])


def project_sum(h, target_sum):
    h = np.clip(h, 0, 1)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-10:
        mask = (h > 1e-8) & (h < 1.0 - 1e-8)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
        else:
            h += diff / len(h)
    return np.clip(h, 0, 1)


def solve_active_lp_sym(q_cur, n, active_k):
    m = n // 2
    h = build_h_sym(q_cur, n)
    dx = 2.0 / n
    corrs = compute_corrs(h)

    c = np.zeros(m + 1)
    c[0] = 1.0

    A_ub_list = []
    b_ub_list = []
    for k in active_k:
        d = k - (n - 1)
        f_k = corrs[k] * dx
        Gq = grad_q_for_shift(h, d, n) * dx
        row = np.zeros(m + 1)
        row[0] = -1.0
        row[1:] = Gq
        A_ub_list.append(row)
        b_ub_list.append(f_k - Gq @ q_cur)

    A_ub = np.array(A_ub_list) if A_ub_list else None
    b_ub = np.array(b_ub_list) if b_ub_list else None
    bounds = [(0, None)] + [(0, 1)] * m

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    except Exception:
        return q_cur.copy(), corrs[active_k].max() * dx, None

    if res.success and res.x is not None:
        return np.clip(res.x[1:], 0, 1), res.x[0], res
    return q_cur.copy(), corrs[active_k].max() * dx, None

    L = min(25, n)

def smooth_obj_q(q, n, beta, k_list):
    h = build_h_sym(q, n)
    dx = 2.0 / n
    corrs = compute_corrs(h)
    corrs_use = corrs[k_list]
    return logsumexp(beta * corrs_use) / beta * dx


def smooth_grad_q(q, n, beta, k_list):
    m = n // 2
    h = build_h_sym(q, n)
    dx = 2.0 / n
    corrs = compute_corrs(h)
    corrs_use = corrs[k_list]

    shifted = beta * (corrs_use - corrs_use.max())
    max_shift = shifted.max()
    w = np.exp(shifted - max_shift)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(len(corrs_use)) / len(corrs_use)
    else:
        w = w / w_sum

    Gq = np.zeros(m)
    for i, k in enumerate(k_list):
        wi = w[i]
        if wi < 1e-15:
            continue
        d = k - (n - 1)
        G = grad_corr_single(h, d, n)
        Gq += wi * (G[:m] - G[m:][::-1])
    return Gq * dx


def smooth_optimize_q(q_init, n, betas, maxiter=200):
    q = q_init.copy()
    m = n // 2

    for beta in betas:
        h = build_h_sym(q, n)
        corrs = compute_corrs(h)
        dx = 2.0 / n
        c5 = corrs.max() * dx

        threshold = c5 - 0.015
        k_list = [k for k in range(n - 1, 2 * n - 1) if corrs[k] * dx >= threshold]
        if len(k_list) < 5:
            k_list = list(np.argsort(corrs)[::-1][:10])

        def obj(x):
            return smooth_obj_q(x, n, beta, k_list)

        def grad(x):
            return smooth_grad_q(x, n, beta, k_list)

        try:
            res = minimize(obj, q, jac=grad, method='L-BFGS-B',
                           bounds=[(0, 1)] * m,
                           options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12})
            if res.fun < smooth_obj_q(q, n, beta, k_list) - 1e-15:
                q = np.clip(res.x, 0, 1)
        except Exception:
            pass

    return q


def make_periodic_h(n, period, duty):

    # Phase 3: SQP in full space (no symmetry)
    if remaining() > 50:
        h_full = best_h.copy()
        n_full = len(h_full)
        h_f, c5_f = sqp_full(h_full, n_full, max_iters=100)
        s_f = exact_score(h_f)
        if s_f < best_c5:
            best_c5 = s_f
            best_h = h_f

    # Phase 4: Multi-start with perturbations
    rng = np.random.RandomState(seed)
    n_curr = len(best_h)
    for trial in range(8):
        if remaining() < 50:
            break

        scale = 0.03 + 0.02 * trial
        h_pert = best_h + rng.randn(n_curr) * scale
        h_pert = project_sum(h_pert, n_curr / 2.0)
        s_pert = exact_score(h_pert)
        start_h = h_pert if s_pert < best_c5 else best_h.copy()

        if n_curr % 2 == 0 and remaining() > 40:
            m = n_curr // 2
            q_pert = start_h[:m].copy()
            q_sq2, c5_sq2 = sqp_sym(q_pert, n_curr, max_iters=60)
            h_sq2 = build_h_sym(q_sq2, n_curr)
            h_sq2 = project_sum(h_sq2, n_curr / 2.0)
            s_sq2 = exact_score(h_sq2)
            if s_sq2 < best_c5:
                best_c5 = s_sq2
                best_h = h_sq2

        # Try full-space SQP from perturbed start
        if remaining() > 30:
            h_pf, c5_pf = sqp_full(start_h, n_curr, max_iters=40)
            s_pf = exact_score(h_pf)
            if s_pf < best_c5:
                best_c5 = s_pf
                best_h = h_pf

    # Phase 5: Try periodic starting points
    for period in [1.0, 0.7, 1.3, 0.5, 1.5]:
        if remaining() < 50:
            break
        duty = 0.5
        h_per = make_periodic_h(n_curr, period, duty)
        if n_curr % 2 == 0 and remaining() > 30:
            m = n_curr // 2
            q_per = h_per[:m].copy()
            q_per = smooth_optimize_q(q_per, n_curr, [100, 300], maxiter=80)
            q_per2, c5_per = sqp_sym(q_per, n_curr, max_iters=80)
            h_per2 = build_h_sym(q_per2, n_curr)
            h_per2 = project_sum(h_per2, n_curr / 2.0)
            s_per = exact_score(h_per2)
            if s_per < best_c5:
                best_c5 = s_per
                best_h = h_per2

    # Phase 6: Try different n values
    n_best = len(best_h)
    for n_try in [64, 90, 100, 130, 140, 150, 160]:
        if remaining() < 50:
            break
        if n_try % 2 != 0:
            continue
        x_orig = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_try)
        h_interp = np.interp(x_new, x_orig, best_h)
        h_interp = project_sum(h_interp, n_try / 2.0)

        m_try = n_try // 2
        q_try = h_interp[:m_try].copy()
        q_try = smooth_optimize_q(q_try, n_try, [80, 200, 500], maxiter=100)
        q_try, c5_try = sqp_sym(q_try, n_try, max_iters=100)
        h_try = build_h_sym(q_try, n_try)
        h_try = project_sum(h_try, n_try / 2.0)
        s_try = exact_score(h_try)

        if s_try < best_c5:
            best_c5 = s_try
            best_h = h_try

        # Also try full-space on this n
        if remaining() > 30:
            h_tf, c5_tf = sqp_full(h_try, n_try, max_iters=60)
            s_tf = exact_score(h_tf)
            if s_tf < best_c5:
                best_c5 = s_tf
                best_h = h_tf

    # Phase 7: Final polish
    n_final = len(best_h)
    if remaining() > 20 and n_final % 2 == 0:
        m = n_final // 2
        q_final = best_h[:m].copy()
        q_final, c5_final = sqp_sym(q_final, n_final, max_iters=100)
        h_final = build_h_sym(q_final, n_final)
        h_final = project_sum(h_final, n_final / 2.0)
        s_final = exact_score(h_final)
        if s_final < best_c5:
            best_c5 = s_final
            best_h = h_final

    if remaining() > 15:
        h_fp, c5_fp = sqp_full(best_h, len(best_h), max_iters=60)
        s_fp = exact_score(h_fp)
        if s_fp < best_c5:
            best_c5 = s_fp
            best_h = h_fp

    # Final validation
    n_out = len(best_h)
    best_h = np.clip(best_h, 0, 1)
    best_h = project_sum(best_h, n_out / 2.0)
    final_c5 = exact_score(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, n_out
