import numpy as np
from scipy.optimize import minimize, linprog
from scipy.special import logsumexp
import time
import math

# Global parameters for SLSQP callbacks
_PARAMS = {}

def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-10:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h = h.copy()
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def full_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, dx):
    return full_corrs(h).max() * dx

def _obj(x):
    dx, beta = _PARAMS['dx'], _PARAMS['beta']
    corrs = full_corrs(x)
    return (logsumexp(beta * corrs) / beta) * dx

def _jac(x):
    dx, beta = _PARAMS['dx'], _PARAMS['beta']
    n = len(x)
    corrs = full_corrs(x)
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones_like(corrs) / len(corrs)
    else:
        w /= w_sum
    grad = np.zeros(n)
    for k in range(len(corrs)):
        wk = w[k]
        if wk < 1e-12:
            continue
        d = k - (n - 1)
        if d >= 0:
            i_lo, i_hi = 0, n - 1 - d
        else:
            i_lo, i_hi = -d, n - 1 + d
        idx = np.arange(i_lo, i_hi + 1)
        grad[idx + d] += (1.0 - x[idx]) * wk
        grad[idx] += (-x[idx + d]) * wk
    return grad * dx

def _cons_fun(x):
    return x.sum() - _PARAMS['target_sum']

def _cons_jac(x):
    return np.ones(len(x))

    _PARAMS['dx'] = dx
    _PARAMS['beta'] = beta
    _PARAMS['target_sum'] = target_sum
