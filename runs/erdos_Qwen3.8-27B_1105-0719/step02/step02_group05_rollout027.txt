def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corrs = np.correlate(h, 1.0 - h, mode='full')
    return corrs.max() * dx, corrs

def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def build_h_from_q(q, n):
    m = n // 2
    if n % 2 == 0:
        return np.concatenate([q, 1.0 - q[::-1]])
    else:
        return np.concatenate([q, [0.5], 1.0 - q[::-1]])

def q_from_h(h, n):
    m = n // 2
    return h[:m].copy()

def grad_corr_d(h, d, n):
    """Gradient of correlation at shift d with respect to h."""
    g = np.zeros(n)
    if d >= 0:
        if d < n:
            g[d:] += (1.0 - h[:n - d])
            g[:n - d] += (-h[d:])
    else:
        dd = -d
        if dd < n:
            g[:n - dd] += (1.0 - h[dd:])
            g[dd:] += (-h[:n - dd])
    return g

def grad_max_approx(h, n, beta):
    """Gradient of the log-sum-exp approximation to the max correlation."""
    dx = 2.0 / n
    corrs = compute_corrs(h)
    shifted = corrs - corrs.max()

            continue

def binary_round(h, n):
    """Round h to nearest binary, maintaining sum constraint."""
    target = n / 2
    # Sort by value, keep top n/2 as 1
    h_rounded = h.copy()
    # Use threshold: values above 0.5 become 1, below become 0

        flip_idx = one_indices[order[:n_flip]]
        h_rounded[flip_idx] = 0

        h_rounded[flip_idx] = 1

        h_rounded[ones] = 1

def active_set_solve(h, n, active_k, t0, target_sum, maxiter=200):

    dx = 2.0 / n
    x0 = np.concatenate([[t0], h])
    
    def obj(x):
        return x[0]
    
    def obj_grad(x):
        g = np.zeros(n + 1)
        g[0] = 1.0
        return g
    
    cons = [{'type': 'eq', 'fun': lambda x: x[1:].sum() - target_sum,
             'jac': lambda x: np.concatenate([[0], np.ones(n)])}]
    
    for ki in active_k:
        d = ki - (n - 1)
        
        def cfun(x, d=d):
            h_x = x[1:]
            if d >= 0:
                i_vals = np.arange(0, n - d)
                val = np.sum(h_x[i_vals + d] * (1.0 - h_x[i_vals]))
            else:
                dd = -d
                i_vals = np.arange(dd, n)
                val = np.sum(h_x[i_vals] * (1.0 - h_x[i_vals - dd]))
            return x[0] - val * dx
        
        def cjac(x, d=d):
            h_x = x[1:]
            g = np.zeros(n + 1)
            g[0] = 1.0
            G = grad_corr_d(h_x, d, n)
            g[1:] = -G * dx
            return g
        
        cons.append({'type': 'ineq', 'fun': cfun, 'jac': cjac})
    
    bounds = [(0.0, 5.0)] + [(0.0, 1.0)] * n
    
    try:

                       bounds=bounds, constraints=cons,

        h_out = project_feasible(res.x[1:], target_sum)

import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time


def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corrs = np.correlate(h, 1.0 - h, mode='full')
    return corrs.max() * dx, corrs


def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] = h[mask] + diff / mask.sum()
    return np.clip(h, 0.0, 1.0)


def build_h_from_q(q, n):
    m = n // 2
    if n % 2 == 0:
        return np.concatenate([q, 1.0 - q[::-1]])
    else:
        return np.concatenate([q, [0.5], 1.0 - q[::-1]])


def q_from_h(h, n):
    m = n // 2
    return h[:m].copy()


def grad_corr_d(h, d, n):
    g = np.zeros(n)
    if d >= 0:
        if d < n:
            g[d:] += (1.0 - h[:n - d])
            g[:n - d] += (-h[d:])
    else:
        dd = -d
        if dd < n:
            g[:n - dd] += (1.0 - h[dd:])
            g[dd:] += (-h[:n - dd])
    return g


def smooth_value_grad(h, n, beta):
    dx = 2.0 / n
    corrs = compute_corrs(h)
    val = logsumexp(beta * corrs) / beta * dx
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(len(corrs)) / len(corrs)
    else:
        w = w / w_sum
    G = np.zeros(n)
    for k in range(len(corrs)):
        if w[k] < 1e-12:
            continue
        d = k - (n - 1)
        G += w[k] * grad_corr_d(h, d, n)
    return val, G * dx


def h_to_q_grad(h_grad, n):
    m = n // 2
    if n % 2 == 0:
        return h_grad[:m] - h_grad[m:][::-1]
    else:
        return h_grad[:m] - h_grad[m + 1:][::-1]


class QSmoothObj:
    def __init__(self, n, beta):
        self.n = n
        self.m = n // 2
        self.beta = beta
        self.dx = 2.0 / n

    def value(self, q):
        q = np.clip(q, 0.0, 1.0)
        h = build_h_from_q(q, self.n)
        corrs = compute_corrs(h)
        return logsumexp(self.beta * corrs) / self.beta * self.dx

    def gradient(self, q):
        q = np.clip(q, 0.0, 1.0)
        h = build_h_from_q(q, self.n)
        _, G = smooth_value_grad(h, self.n, self.beta)
        return h_to_q_grad(G, self.n)


class C:
    def __init__(self, prob, idx):
        self.prob = prob
        self.idx = idx

    def __call__(self, x):
        return self.prob.f(x, self.idx)


class CJ:
    def __init__(self, prob, idx):
        self.prob = prob
        self.idx = idx

    def __call__(self, x):
        return self.prob.g(x, self.idx)


class QEpigraph:
    def __init__(self, n, active_k):
        self.n = n
        self.m = n // 2
        self.active_k = active_k
        self.dx = 2.0 / n

    def obj(self, x):
        return x[0]

    def obj_grad(self, x):
        g = np.zeros(self.m + 1)
        g[0] = 1.0
        return g

    def f(self, x, idx):
        if idx < 0:
            return 0.0
        t = x[0]
        q = np.clip(x[1:], 0.0, 1.0)
        h = build_h_from_q(q, self.n)
        k = self.active_k[idx]
        d = k - (self.n - 1)
        n = self.n
        if d >= 0:
            i_vals = np.arange(0, n - d)
            val = np.sum(h[i_vals + d] * (1.0 - h[i_vals]))
        else:
            dd = -d
            i_vals = np.arange(dd, n)
            val = np.sum(h[i_vals] * (1.0 - h[i_vals - dd]))
        return t - val * self.dx

    def g(self, x, idx):
        if idx < 0:
            return np.zeros(self.m + 1)
        q = np.clip(x[1:], 0.0, 1.0)
        h = build_h_from_q(q, self.n)
        k = self.active_k[idx]
        d = k - (self.n - 1)
        n = self.n
        G = grad_corr_d(h, d, n)
        Gq = h_to_q_grad(G, n) * self.dx
        g = np.zeros(self.m + 1)
        g[0] = 1.0
        g[1:] = -Gq
        return g


def solve_q_epigraph(q0, n, active_k, t0, maxiter=200):
    m = n // 2
    x0 = np.concatenate([[t0], q0])
    ep = QEpigraph(n, active_k)
    cons = []
    for i in range(len(active_k)):
        cons.append({'type': 'ineq', 'fun': C(ep, i), 'jac': CJ(ep, i)})
    bounds = [(0.0, 5.0)] + [(0.0, 1.0)] * m
    try:
        res = minimize(ep.obj, x0, jac=ep.obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        q_out = np.clip(res.x[1:], 0.0, 1.0)
        return q_out, res.x[0], res
    except Exception:
        return q0, t0, None


class FullEpigraph:
    def __init__(self, n, active_k, target_sum):
        self.n = n
        self.active_k = active_k
        self.dx = 2.0 / n
        self.target_sum = target_sum

    def obj(self, x):
        return x[0]

    def obj_grad(self, x):
        g = np.zeros(self.n + 1)
        g[0] = 1.0
        return g

    def f(self, x, idx):
        if idx < 0:
            return x[1:].sum() - self.target_sum
        t = x[0]
        h = np.clip(x[1:], 0.0, 1.0)
        k = self.active_k[idx]
        d = k - (self.n - 1)
        n = self.n
        if d >= 0:
            i_vals = np.arange(0, n - d)
            val = np.sum(h[i_vals + d] * (1.0 - h[i_vals]))
        else:
            dd = -d
            i_vals = np.arange(dd, n)
            val = np.sum(h[i_vals] * (1.0 - h[i_vals - dd]))
        return t - val * self.dx

    def g(self, x, idx):
        if idx < 0:
            g = np.zeros(self.n + 1)
            g[1:] = 1.0
            return g
        h = np.clip(x[1:], 0.0, 1.0)
        k = self.active_k[idx]
        d = k - (self.n - 1)
        n = self.n
        G = grad_corr_d(h, d, n)
        g = np.zeros(self.n + 1)
        g[0] = 1.0
        g[1:] = -G * self.dx
        return g


def solve_full_epigraph(h0, n, active_k, t0, target_sum, maxiter=200):
    x0 = np.concatenate([[t0], h0])
    ep = FullEpigraph(n, active_k, target_sum)
    cons = []
    for i in range(len(active_k)):
        cons.append({'type': 'ineq', 'fun': C(ep, i), 'jac': CJ(ep, i)})
    cons.append({'type': 'eq', 'fun': C(ep, -1), 'jac': CJ(ep, -1)})
    bounds = [(0.0, 5.0)] + [(0.0, 1.0)] * n
    try:
        res = minimize(ep.obj, x0, jac=ep.obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        h_out = project_feasible(res.x[1:], target_sum)
        return h_out, res.x[0], res
    except Exception:
        return h0, t0, None


def local_search_q(q, n, max_rounds=10, delta=0.02):
    m = n // 2
    best_c5, _ = exact_score(build_h_from_q(q, n))
    for rnd in range(max_rounds):
        improved = False
        h = build_h_from_q(q, n)
        corrs = compute_corrs(h)
        max_idx = corrs.argmax()
        max_d = max_idx - (n - 1)

        for k in range(m):
            for di in [delta, -delta, delta * 0.5, -delta * 0.5]:
                q_try = q.copy()
                new_val = q[k] + di
                if new_val < 0 or new_val > 1:
                    continue
                q_try[k] = new_val

                if c5_try < best_c5 - 1e-13:

                    best_c5 = c5_try
                    improved = True

def binary_round_q(q, n):
    h = build_h_from_q(q, n)
    m = n // 2
    h_bin = np.zeros(n)
    ones_needed = n // 2
    sorted_idx = np.argsort(-h)
    for i in range(min(ones_needed, n)):

    if n % 2 == 0:

    else:

            if remaining() < 30:

            q_res = smooth_optimize_q(q_try, n_try, beta, maxiter=200)
            h_res = build_h_from_q(q_res, n_try)
            h_res = project_feasible(h_res, ts_try)
            s_res, _ = exact_score(h_res)
            if s_res < best_c5 - 1e-12:
                best_c5 = s_res
                best_h = h_res.copy()

        # Active set on new grid
        if remaining() > 50 and n_try % 2 == 0:
            m_t = n_try // 2
            q_t = q_from_h(best_h, n_try) if len(best_h) == n_try else q_try
            if len(q_t) != m_t:
                q_t = q_try
            q_res2, c5_res2 = q_active_set_search(q_t, n_try, min(remaining() * 0.4, 100))
            h_res2 = build_h_from_q(q_res2, n_try)
            s_res2, _ = exact_score(h_res2)
            if s_res2 < best_c5 - 1e-12:
                best_c5 = s_res2
                best_h = h_res2.copy()

    # Phase 7: Binary projection and re-optimization
    if remaining() > 40:
        h_bin = binary_round_q(q_best, n)
        h_b = build_h_from_q(h_bin, n)
        s_b, _ = exact_score(h_b)
        if s_b < best_c5:
            best_c5 = s_b
            best_h = h_b.copy()
            q_best = h_bin.copy()
            # Re-optimize from binary
            if remaining() > 30:
                q_res = smooth_optimize_q(q_best, n, 200, maxiter=200)
                h_res = build_h_from_q(q_res, n)
                s_res, _ = exact_score(h_res)
                if s_res < best_c5 - 1e-14:
                    best_c5 = s_res
                    best_h = h_res.copy()
                    q_best = q_res.copy()

    # Phase 8: Final polish - alternating smooth and epigraph
    n_final = len(best_h)
    ts_final = n_final / 2.0
    m_final = n_final // 2
    q_final = q_from_h(best_h, n_final) if n_final % 2 == 0 else None

    for polish_round in range(4):
        if remaining() < 20:
            break
        if q_final is not None:
            q_final = smooth_optimize_q(q_final, n_final, 3000, maxiter=150)
            h_f = build_h_from_q(q_final, n_final)
            s_f, _ = exact_score(h_f)
            if s_f < best_c5 - 1e-14:
                best_c5 = s_f
                best_h = h_f.copy()

            if remaining() > 25:
                q_final2, c5_f2 = q_active_set_search(q_final, n_final, min(remaining() * 0.3, 60))
                h_f2 = build_h_from_q(q_final2, n_final)
                s_f2, _ = exact_score(h_f2)
                if s_f2 < best_c5 - 1e-14:
                    best_c5 = s_f2
                    best_h = h_f2.copy()
                    q_final = q_final2.copy()

        if remaining() > 20:
            h_fp, c5_fp = full_active_set_search(best_h, n_final, min(remaining() * 0.3, 60))
            s_fp, _ = exact_score(h_fp)
            if s_fp < best_c5 - 1e-14:
                best_c5 = s_fp
                best_h = h_fp.copy()
                if n_final % 2 == 0:
                    q_final = q_from_h(best_h, n_final)

    # Local search on final solution
    if remaining() > 15 and len(best_h) % 2 == 0:
        q_f = q_from_h(best_h, len(best_h))
        q_f2, c5_f2 = local_search_q(q_f, len(best_h), max_rounds=8, delta=0.01)
        h_f2 = build_h_from_q(q_f2, len(best_h))
        s_f2, _ = exact_score(h_f2)
        if s_f2 < best_c5 - 1e-14:
            best_c5 = s_f2
            best_h = h_f2.copy()

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    n_out = len(best_h)
    ts_out = n_out / 2.0
    best_h = project_feasible(best_h, ts_out)
    final_score, _ = exact_score(best_h)
    best_c5 = min(best_c5, final_score)

    return best_h, best_c5, n_out
