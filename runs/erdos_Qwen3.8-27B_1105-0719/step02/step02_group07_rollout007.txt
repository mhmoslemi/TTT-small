import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time


def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-10:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)


def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)


def make_sym_h(q, n):
    m = n // 2
    h = np.empty(n)
    h[:m] = q
    h[m:] = 1.0 - q[::-1]
    return h


def make_q_from_h(h, n):
    m = n // 2
    q = 0.5 * (h[:m] + 1.0 - h[n - 1:n - 1 - m:-1])
    return np.clip(q, 0.0, 1.0)


def corr_vals(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx


def grad_corr_lag(h, n, dx, lag):
    d = lag - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -d)
    i_hi = min(n - 1, n - 1 - d)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx] += (1.0 - h[idx + d])
    grad[idx + d] += (-h[idx])
    return grad * dx


def grad_corr_lag_sym(q, n, lag):
    m = n // 2
    dx = 2.0 / n
    h = make_sym_h(q, n)
    G = grad_corr_lag(h, n, dx, lag)
    return G[:m] - G[n - 1:n - 1 - m:-1]


def smooth_sym_obj(x, n, beta):
    h = make_sym_h(x, n)
    cv = corr_vals(h)
    return logsumexp(beta * cv) / beta


def smooth_sym_grad(x, n, beta):
    m = n // 2
    h = make_sym_h(x, n)
    cv = corr_vals(h)
    shifted = cv - cv.max()
    w = np.exp(beta * shifted)
    ws = w.sum()
    if ws < 1e-30:
        w = np.ones(len(cv)) / len(cv)
    else:
        w /= ws
    G = np.zeros(n)
    for k in range(len(cv)):
        if w[k] > 1e-15:
            G += w[k] * grad_corr_lag(h, len(cv) // 2 + len(cv) // 2 + 1 if False else 2 * n - 1, 0, 0, k) if False else w[k] * grad_corr_lag_full(h, n, k)
    return G[:m] - G[n - 1:n - 1 - m:-1]


def grad_corr_lag_full(h, n, k):
    dx = 2.0 / n
    d = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -d)
    i_hi = min(n - 1, n - 1 - d)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx] += (1.0 - h[idx + d])
    grad[idx + d] += (-h[idx])
    return grad * dx


def smooth_sym_obj2(x, n, beta):
    h = make_sym_h(x, n)
    cv = corr_vals(h)
    return logsumexp(beta * cv) / beta


def smooth_sym_grad2(x, n, beta):
    m = n // 2
    h = make_sym_h(x, n)
    cv = corr_vals(h)
    shifted = cv - cv.max()

def smooth_optimize_sym(q0, n, beta, maxiter=500):
    m = n // 2
    try:
        res = minimize(smooth_sym_obj2, q0, jac=lambda x: smooth_sym_grad2(x, n, beta),
                       method='SLSQP', bounds=[(0.0, 1.0)] * m,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        h_new = make_sym_h(res.x, n)
        c5 = exact_score(h_new)
        return res.x, c5
    except Exception:
        return None, None


def active_set_minimax_sym(q0, n, n_active, t_init, maxiter=800):
    m = n // 2
    h_cur = make_sym_h(q0, n)
    cv = corr_vals(h_cur)

    # Select active lags
    if n_active >= 2 * n - 1:
        active = list(range(2 * n - 1))
    else:
        active = sorted(np.argsort(cv)[::-1][:n_active].tolist())

    def obj(x):
        return x[0]

    def obj_grad(x):
        g = np.zeros(m + 1)
        g[0] = 1.0
        return g

    def con_fun(x):
        q = x[1:]
        h = make_sym_h(q, n)
        cv = corr_vals(h)
        return x[0] - cv[active]

    def con_jac(x):
        q = x[1:]
        h = make_sym_h(q, n)
        L = len(active)
        J = np.zeros((L, m + 1))
        J[:, 0] = 1.0
        for i, lag in enumerate(active):
            G = grad_corr_lag_full(h, n, lag)
            J[i, 1:] = -(G[:m] - G[n - 1:n - 1 - m:-1])
        return J

    cons = [{'type': 'ineq', 'fun': con_fun, 'jac': con_jac}]
    bounds = [(0.0, 1.0)] + [(0.0, 1.0)] * m
    x0 = np.concatenate([[t_init], q0])

    try:
        res = minimize(obj, x0, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        h_new = make_sym_h(res.x[1:], n)
        c5 = exact_score(h_new)
        return res.x[1:], c5
    except Exception:
        return None, None


def full_smooth_optimize(h0, n, beta, maxiter=500):
    target_sum = n / 2.0

    def obj(x):
        h = np.clip(x, 0.0, 1.0)
        cv = corr_vals(h)
        return logsumexp(beta * cv) / beta

    def obj_grad(x):
        h = np.clip(x, 0.0, 1.0)
        cv = corr_vals(h)
        shifted = cv - cv.max()
        w = np.exp(beta * shifted)
        ws = w.sum()
        if ws < 1e-30:

        G = np.zeros(n)
        for k in range(len(cv)):
            if w[k] > 1e-14:
                G += w[k] * grad_corr_lag_full(h, n, k)
        return G

                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        h_out = project_feasible(res.x, target_sum)
        c5 = exact_score(h_out)
        return h_out, c5
    except Exception:
        return h0, exact_score(h0)
