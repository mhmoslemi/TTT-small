    # Build LP:
    # minimize t
    # subject to:
    #   grad_k · d - t ≤ 0 for k in active_set
    #   sum(d) = 0

    for idx, k in enumerate(active_set):

        A_ub[idx, :n] = g
        A_ub[idx, n] = -1.0  # grad·d - t ≤ 0

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds: -h[i]/step ≤ d[i] ≤ (1-h[i])/step, t ≥ 0
    bounds = []
    for i in range(n):
        bounds.append((-h[i] / step_size, (1.0 - h[i]) / step_size))
    bounds.append((0.0, None))
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                  bounds=bounds, method='highs')
    
    if res.success:
        d = res.x[:n]
        return d
    return None

    grads = np.zeros((2*n-1, n))

            i_range = np.arange(n - offset)
            grads[k, i_range] += (1.0 - h[i_range + offset])
            grads[k, i_range + offset] += (-h[i_range])
        elif offset < 0:

            grads[k, i_range + offset] += (-h[i_range])

            grads[k, :] = 1.0 - 2.0 * h

    """Solve LP for descent direction."""
    m = min(top_k, 2*n-1)
    top_idx = np.argsort(-corrs)[:m]

    A = np.zeros((m, n))

import numpy as np

def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
        else:
            h += diff / n
    return np.clip(h, 0.0, 1.0)


def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def c5_of_h(h, dx):
    return float(compute_corrs(h).max() * dx)


def grad_at_lag(h, k, n):
    offset = k - (n - 1)
    g = np.zeros(n)
    if offset >= 0:
        j1 = np.arange(n - offset)
        g[j1] += (1.0 - h[j1 + offset])
        j2 = np.arange(offset, n)
        g[j2] -= h[j2 - offset]
    else:
        ao = -offset
        j1 = np.arange(ao, n)
        g[j1] += (1.0 - h[j1 - ao])
        j2 = np.arange(n - ao)
        g[j2] -= h[j2 + ao]
    return g


def lp_direction(h, corrs, n, top_k, step):
    m = min(top_k, 2 * n - 1)
    top_idx = np.argsort(-corrs)[:m]
    A = np.zeros((m, n))
    for idx, k in enumerate(top_idx):
        A[idx] = grad_at_lag(h, k, n)
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_ub = np.hstack((A, -np.ones((m, 1))))
    b_ub = np.zeros(m)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(-h[i] / step, (1.0 - h[i]) / step) for i in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], float(res.x[n])
    except Exception:
        pass
    return None, None


def line_search_alpha(h, d, target_sum, max_alpha, n_trials=30):

def binarize(h, n):
    target = n // 2
    idx = np.argsort(-h)[:target]
    b = np.zeros(n)
    b[idx] = 1.0
    return b


def swap_delta(b, i, j, n):
    delta = np.zeros(2 * n - 1)
    di1 = np.arange(-i, n - i)
    di2 = np.arange(i - (n - 1), i + 1)
    delta[n - 1 + di1] -= (1.0 - b[i + di1])
    delta[n - 1 + di2] += b[i - di2]
    dj1 = np.arange(-j, n - j)
    dj2 = np.arange(j - (n - 1), j + 1)
    delta[n - 1 + dj1] += (1.0 - b[j + dj1])
    delta[n - 1 + dj2] -= b[j - dj2]
    return delta


def binary_local_search(h0, dx, rng, max_iters=200000, patience=10000):
    n = len(h0)
    b = binarize(h0, n)
    corrs = compute_corrs(b)
    best_c5 = float(corrs.max() * dx)
    best_b = b.copy()
    no_improve = 0
    temp = 0.03
    for it in range(max_iters):
        if it % 3000 == 0:
            temp *= 0.999
        ones = np.where(b == 1.0)[0]
        zeros = np.where(b == 0.0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        dd = swap_delta(b, i, j, n)
        new_corrs = corrs + dd
        new_max = float(new_corrs.max() * dx)
        if new_max <= best_c5 + 1e-15:
            b[i], b[j] = 0.0, 1.0
            corrs = new_corrs
            best_c5 = new_max
            best_b = b.copy()
            no_improve = 0
        else:
            dv = new_max - best_c5
            if temp > 1e-9 and rng.random() < np.exp(-dv / temp):
                b[i], b[j] = 0.0, 1.0
                corrs = new_corrs
                no_improve = 0
            else:
                no_improve += 1
        if no_improve > patience:
            break
    return best_b, best_c5


def greedy_binary_swap(h0, dx, max_sweeps=30):

        for i in ones:

            di2 = np.arange(i - (n - 1), i + 1)

            base_d[n - 1 + di1] -= (1.0 - b[i + di1])

                dj2 = np.arange(j - (n - 1), j + 1)
                dd = base_d.copy()
                dd[n - 1 + dj1] += (1.0 - b[j + dj1])
                dd[n - 1 + dj2] -= b[j - dj2]
                nm = float((corrs + dd).max())

                    best_swap = (i, j)

    return best_b, best_c5
