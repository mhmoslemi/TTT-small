import numpy as np
from scipy.optimize import linprog, minimize
from scipy.special import expit
import time

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def corr_raw(h):
    return np.correlate(h, 1.0 - h, mode='full')

def proj_box_sum(x, target):
    n = len(x)
    x = np.clip(np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0), 0.0, 1.0)
    if abs(x.sum() - target) < 1e-12:
        return x
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(x + mid, 0.0, 1.0).sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    v = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)
    r = target - v.sum()
    if abs(r) > 1e-13:
        order = np.argsort(v) if r > 0 else np.argsort(v)[::-1]
        for i in order:
            if r > 0:
                add = min(r, 1.0 - v[i])
                v[i] += add
                r -= add
            else:
                sub = min(-r, v[i])
                v[i] -= sub
                r += sub
            if abs(r) < 1e-15:
                break
    return v

def build_sym(q):
    m = len(q)
    h = np.empty(2 * m)
    h[:m] = q
    h[m:] = 1.0 - q[::-1]
    return h

def sym_from_h(h):
    n = len(h)
    m = n // 2
    q = np.empty(m)
    for i in range(m):
        q[i] = 0.5 * (h[i] + (1.0 - h[n - 1 - i]))
    return np.clip(q, 0.0, 1.0)

def lag_grad_full(h, ell, n):
    """Gradient of lag ell of np.correlate(h, 1-h, 'full') w.r.t. h."""
    d = ell - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -d)
    i_hi = min(n - 1, n - 1 - d)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx] += (1.0 - h[idx + d])
    grad[idx + d] += (-h[idx])
    return grad

def lag_grad_sym(q, ell, n):
    """Gradient of lag ell w.r.t. q for symmetric construction."""
    m = n // 2
    h = build_sym(q)
    gh = lag_grad_full(h, ell, n)
    gq = np.zeros(m)
    for r in range(m):
        gq[r] = gh[r] - gh[n - 1 - r]
    return gq

def lag_val_full(h, ell, n):
    d = ell - (n - 1)
    i_lo = max(0, -d)
    i_hi = min(n - 1, n - 1 - d)
    if i_lo > i_hi:
        return 0.0
    idx = np.arange(i_lo, i_hi + 1)
    return float(np.sum(h[idx] * (1.0 - h[idx + d])))

def select_active(corrs, n, margin_frac=0.03, pad=3, max_k=60):
    max_val = corrs.max()
    threshold = max_val - max(1e-9, margin_frac * max_val)
    active = np.where(corrs >= threshold)[0]
    padded = set()
    for k in active:
        for dd in range(-pad, pad + 1):
            if 0 <= k + dd < n:
                padded.add(k + dd)
    lst = sorted(padded)
    if len(lst) > max_k:
        scored = sorted([(corrs[k], k) for k in lst], reverse=True)
        lst = sorted([k for _, k in scored[:max_k]])
    return lst

def solve_tangent_sym(q0, n_full, active_ells, rho):
    """Solve tangent LP for symmetric case. Returns (q_new, t_new)."""
    m = n_full // 2
    h0 = build_sym(q0)
    dx = 2.0 / n_full
    corrs0 = corr_raw(h0)
    
    # Build constraint matrix
    # Variables: [q_0..q_{m-1}, t]
    nvar = m + 1
    ncon = len(active_ells)
    
    A_ub = np.zeros((ncon, nvar))
    b_ub = np.zeros(ncon)
    
    for ci, ell in enumerate(active_ells):
        gq = lag_grad_sym(q0, ell, n_full)
        R_ell = corrs0[ell]
        rhs = gq @ q0 - R_ell
        A_ub[ci, :m] = gq
        A_ub[ci, m] = -1.0
        b_ub[ci] = rhs
    
    bounds = []
    for i in range(m):
        lo = max(0.0, q0[i] - rho)
        hi = min(1.0, q0[i] + rho)
        bounds.append((lo, hi))
    t_upper = corrs0.max() + 1.0
    bounds.append((0.0, t_upper))
    
    c = np.zeros(nvar)
    c[m] = 1.0
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if not res.success:
        return q0.copy(), corrs0.max()
    
    q_new = np.clip(res.x[:m], 0.0, 1.0)
    return q_new, res.x[m]

def solve_tangent_full(h0, active_ells, rho):
    """Solve tangent LP for full case. Returns (h_new, t_new)."""
    n = len(h0)
    corrs0 = corr_raw(h0)
    
    nvar = n + 1
    ncon = len(active_ells)
    
    A_ub = np.zeros((ncon, nvar))
    b_ub = np.zeros(ncon)
    
    for ci, ell in enumerate(active_ells):
        gh = lag_grad_full(h0, ell, n)
        R_ell = corrs0[ell]
        rhs = gh @ h0 - R_ell
        A_ub[ci, :n] = gh
        A_ub[ci, n] = -1.0
        b_ub[ci] = rhs
    
    # Equality: sum(h) = n/2
    A_eq = np.zeros((1, nvar))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    bounds = []
    for i in range(n):
        lo = max(0.0, h0[i] - rho)
        hi = min(1.0, h0[i] + rho)
        bounds.append((lo, hi))
    t_upper = corrs0.max() + 1.0
    bounds.append((0.0, t_upper))
    
    c = np.zeros(nvar)
    c[n] = 1.0
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    
    if not res.success:
        return h0.copy(), corrs0.max()
    
    h_new = res.x[:n]
    return np.clip(h_new, 0.0, 1.0), res.x[n]

def line_search_step(h0, direction, n, max_alpha=1.0, n_evals=20):
    """Find best alpha in (0, max_alpha] for h0 + alpha * direction."""
    # Ensure the step preserves constraints by checking bounds
    # For full: sum(direction) should be ~0
    # alpha_max from bounds
    if np.any(direction > 0):
        alpha_pos = (1.0 - h0[direction > 0]) / direction[direction > 0]
        max_alpha = min(max_alpha, alpha_pos.min() * 0.999)
    if np.any(direction < 0):
        alpha_neg = (-h0[direction < 0]) / direction[direction < 0]
        max_alpha = min(max_alpha, alpha_neg.min() * 0.999)
    
    if max_alpha < 1e-10:
        return 0.0, eval_c5(h0)
    
    # Grid search
    alphas = np.linspace(0.0, max_alpha, n_evals)
    best_alpha = 0.0
    best_val = eval_c5(h0)
    
    for alpha in alphas[1:]:
        h_try = h0 + alpha * direction
        h_try = np.clip(h_try, 0.0, 1.0)
        val = eval_c5(h_try)
        if val < best_val:
            best_val = val
            best_alpha = alpha
    
    # Local refinement around best
    if best_alpha > 0:
        step = max_alpha / (n_evals * 5)
        for _ in range(5):
            candidates = best_alpha + np.array([-step, -step/2, 0, step/2, step])
            for alpha in candidates:
                if alpha <= 0 or alpha > max_alpha:
                    continue
                h_try = h0 + alpha * direction
                h_try = np.clip(h_try, 0.0, 1.0)
                val = eval_c5(h_try)
                if val < best_val:
                    best_val = val
                    best_alpha = alpha
            step /= 3
    
    return best_alpha, best_val

def pair_transfer_search(h0, n, a, b, n_grid=40):
    """Search over delta for h[a]+=delta, h[b]-=delta."""
    lo = max(-h0[a], h0[b] - 1.0)
    hi = min(1.0 - h0[a], -h0[b])
    if hi - lo < 1e-12:
        return h0.copy(), eval_c5(h0)
    
    deltas = np.linspace(lo, hi, n_grid)
    best_delta = 0.0
    best_val = eval_c5(h0)
    
    for delta in deltas:
        h_try = h0.copy()
        h_try[a] += delta
        h_try[b] -= delta
        h_try = np.clip(h_try, 0.0, 1.0)
        val = eval_c5(h_try)
        if val < best_val:
            best_val = val
            best_delta = delta
    
    # Refine
    if best_delta != 0.0:
        span = hi - lo
        step = span / (n_grid * 5)
        for _ in range(4):
            for frac in [-1.0, -0.5, 0.0, 0.5, 1.0]:
                d_try = best_delta + frac * step
                if d_try < lo or d_try > hi:
                    continue
                h_try = h0.copy()
                h_try[a] += d_try
                h_try[b] -= d_try
                val = eval_c5(h_try)
                if val < best_val:
                    best_val = val
                    best_delta = d_try
            step /= 3
    
    h_new = h0.copy()
    h_new[a] += best_delta
    h_new[b] -= best_delta
    return h_new, best_val

import numpy as np
from scipy.optimize import linprog
import time

def eval_c5(h):

    dx = 2.0 / n

def corr_raw(h):

def proj_box_sum(x, target):

    x = np.clip(np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0), 0.0, 1.0)

        return x.copy()

            hi = mid
        else:
            lo = mid

    r = target - v.sum()
    if abs(r) > 1e-13:
        order = np.argsort(v) if r > 0 else np.argsort(v)[::-1]

                add = min(r, 1.0 - v[i])
                v[i] += add
                r -= add
            else:
                sub = min(-r, v[i])
                v[i] -= sub
                r += sub
            if abs(r) < 1e-15:
                break
    return v

def build_sym(q):
    m = len(q)
    h = np.empty(2 * m)
    h[:m] = q
    h[m:] = 1.0 - q[::-1]
    return h

def sym_from_h(h):
    n = len(h)
    m = n // 2
    q = np.empty(m)
    for i in range(m):
        q[i] = 0.5 * (h[i] + (1.0 - h[n - 1 - i]))
    return np.clip(q, 0.0, 1.0)

def lag_grad_full(h, ell, n):
    d = ell - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -d)
    i_hi = min(n - 1, n - 1 - d)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx] += (1.0 - h[idx + d])
    grad[idx + d] += (-h[idx])
    return grad

def lag_grad_sym(q, ell, n):
    m = n // 2
    h = build_sym(q)
    gh = lag_grad_full(h, ell, n)
    gq = np.zeros(m)
    for r in range(m):
        gq[r] = gh[r] - gh[n - 1 - r]
    return gq

def select_active(corrs, ncols, margin_frac, pad, max_k):

    threshold = max_val - max(1e-10, margin_frac * max_val)
    active = np.where(corrs >= threshold)[0]
    padded = set()
    for k in active:
        for dd in range(-pad, pad + 1):
            if 0 <= k + dd < ncols:
                padded.add(k + dd)
    lst = sorted(padded)
    if len(lst) > max_k:
        scored = sorted([(corrs[k], k) for k in lst], reverse=True)
        lst = sorted([k for _, k in scored[:max_k]])
    return lst

def solve_tangent_sym(q0, n_full, active_ells, rho):

    h0 = build_sym(q0)
    corrs0 = corr_raw(h0)
    nvar = m + 1
    ncon = len(active_ells)
    if ncon == 0:
        return q0.copy(), corrs0.max()

        R_ell = corrs0[ell]
        rhs = gq.dot(q0) - R_ell

def pair_eval(h0, a, b, delta):
    h_try = h0.copy()
    h_try[a] += delta
    h_try[b] -= delta
    return eval_c5(h_try)

def pair_transfer_search(h0, n, a, b, n_grid=35):
    lo = max(-h0[a], h0[b] - 1.0)
    hi = min(1.0 - h0[a], -h0[b])
    if hi - lo < 1e-12:
        return h0.copy(), eval_c5(h0)
    deltas = np.linspace(lo, hi, n_grid)
    best_delta = 0.0
    best_val = eval_c5(h0)
    for delta in deltas:
        val = pair_eval(h0, a, b, delta)
        if val < best_val:
            best_val = val
            best_delta = delta
    if best_delta != 0.0:
        step = (hi - lo) / (n_grid * 4.0)

                    continue
                val = pair_eval(h0, a, b, d_try)
                if val < best_val:
                    best_val = val
                    best_delta = d_try

    h_new = h0.copy()
    h_new[a] += best_delta
    h_new[b] -= best_delta
    h_new = np.clip(h_new, 0.0, 1.0)
    return h_new, best_val

        h_curr = build_sym(best_q)
        corrs = corr_raw(h_curr) * (2.0 / n_full)
        max_c = corrs.max()
        shifted = corrs - max_c
        w = np.exp(beta * shifted)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(len(corrs)) / len(corrs)
        else:
            w /= w_sum
        grad_q = np.zeros(m)
        for k in range(len(corrs)):
            if w[k] < 1e-14:
                continue
            gq = lag_grad_sym(best_q, k, n_full)
            grad_q += w[k] * gq
        grad_q *= (2.0 / n_full)

        def sm_obj(xx):
            h_x = build_sym(xx)
            c_x = corr_raw(h_x) * (2.0 / n_full)
            mc = c_x.max()
            sh = c_x - mc
            ex = np.exp(beta * sh)
            ws = ex.sum()
            if ws < 1e-30:
                return mc
            return mc + np.log(ws) / beta

        tl = time.time() + min(60, max(0, t_limit - time.time()))
        from scipy.optimize import minimize
        try:
            res = minimize(sm_obj, best_q, jac=lambda xx: smooth_max_grad_sym(xx, n_full, beta),
                          method='L-BFGS-B', bounds=[(0.0, 1.0)] * m,
                          options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10, 'maxfun': maxiter * 3})
            q_new = np.clip(res.x, 0.0, 1.0)
        except Exception:
            q_new = best_q.copy()
        h_new = build_sym(q_new)
        sc = eval_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_q = q_new.copy()
    return best_q, best_score

def smooth_max_grad_sym(xx, n_full, beta):
    m = n_full // 2
    h = build_sym(xx)
    corrs = corr_raw(h) * (2.0 / n_full)
    max_c = corrs.max()
    shifted = corrs - max_c
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(len(corrs)) / len(corrs)
    else:
        w /= w_sum
    grad_q = np.zeros(m)
    for k in range(len(corrs)):
        if w[k] < 1e-14:
            continue
        gq = lag_grad_sym(xx, k, n_full)
        grad_q += w[k] * gq
    grad_q *= (2.0 / n_full)
    return grad_q

def coef_con_fun(args):
    x, ell, n, mask = args
    h = x
    d = ell - (n - 1)
    i_lo = max(0, -d)
    i_hi = min(n - 1, n - 1 - d)
    if i_lo > i_hi:
        return 0.0
    idx = np.arange(i_lo, i_hi + 1)
    return float(np.sum(h[idx] * (1.0 - h[idx + d])))

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()

    def rem():
        return budget_s - (time.time() - t0)

    try:
        h_parent = np.array(initial_h_values, dtype=float).copy().reshape(-1)
    except (NameError, AttributeError):
        h_parent = None

    if h_parent is not None and len(h_parent) > 10:
        n = len(h_parent)
        best_h = proj_box_sum(h_parent, n / 2.0)
    else:
        n = 126
        m = n // 2
        x = np.linspace(0, 1, m)
        q = (0.5 + 0.3 * np.sin(2 * np.pi * x)).clip(0, 1)
        best_h = build_sym(q)

    best_c5 = eval_c5(best_h)
    n_best = len(best_h)

    # Phase 1: Symmetric LP tangent iterations
    if n_best % 2 == 0 and rem() > 80:
        n_sym = n_best
        m_sym = n_sym // 2
        q_sym = sym_from_h(best_h)
        h_sym = build_sym(q_sym)

        for it in range(300):
            if rem() < 25:
                break
            corrs = corr_raw(h_sym)
            ncols = 2 * n_sym - 1
            active = select_active(corrs, ncols, 0.025, 3, 55)

            improved = False
            for rho in [0.12, 0.06, 0.03, 0.015]:
                q_new, t_new = solve_tangent_sym(q_sym, n_sym, active, rho)
                h_new = build_sym(q_new)
                c5_new = eval_c5(h_new)

                if c5_new < best_c5 - 1e-12:
                    direction = h_new - h_sym
                    alpha, val_ls = line_search_dir(h_sym, direction, n_sym, n_grid=18)
                    if alpha > 1e-8 and val_ls < c5_new:
                        h_ls = np.clip(h_sym + alpha * direction, 0.0, 1.0)
                        c5_ls = eval_c5(h_ls)
                        if c5_ls < best_c5 - 1e-12:
                            best_c5 = c5_ls
                            best_h = h_ls.copy()
                            h_sym = h_ls.copy()
                            q_sym = sym_from_h(h_ls)
                            improved = True
                            break
                    if c5_new < best_c5 - 1e-12:
                        best_c5 = c5_new
                        best_h = h_new.copy()
                        h_sym = h_new.copy()
                        q_sym = q_new.copy()
                        improved = True
                        break

            if not improved:
                # Check hidden violations
                corrs_chk = corr_raw(h_sym)
                max_act = corrs_chk[active].max()
                hidden = np.where(corrs_chk > max_act + 5e-7)[0]
                if len(hidden) > 0:
                    active2 = select_active(corrs_chk, ncols, 0.02, 3, 70)
                    if len(active2) > len(active):
                        active = active2
                        continue
                # Try expanding
                active_expand = select_active(corrs_chk, ncols, 0.05, 4, 75)
                if len(active_expand) <= len(active) + 1:
                    break
                active = active_expand

        best_c5 = eval_c5(best_h)
        n_best = len(best_h)

    # Phase 2: Smooth max gradient (symmetric) for different basin
    if n_best % 2 == 0 and rem() > 120:
        m_sm = n_best // 2
        q_sm = sym_from_h(best_h)
        t_lim = time.time() + min(100, rem() * 0.4)
        q_res, sc_res = smooth_max_opt_sym(q_sm, n_best, [80, 200, 500], 200, t_lim)
        if sc_res < best_c5:
            best_c5 = sc_res
            best_h = build_sym(q_res).copy()

    # Phase 3: Full-variable LP (break symmetry)
    if rem() > 120:
        n_full = n_best
        h_full = best_h.copy()
        h_full = proj_box_sum(h_full, n_full / 2.0)

        for it in range(150):
            if rem() < 25:
                break
            corrs = corr_raw(h_full)
            ncols = 2 * n_full - 1
            active = select_active(corrs, ncols, 0.025, 3, 55)
            improved = False

            for rho in [0.08, 0.04, 0.02, 0.01]:
                h_new, t_new = solve_tangent_full(h_full, active, rho)
                h_new = proj_box_sum(h_new, n_full / 2.0)
                c5_new = eval_c5(h_new)

                if c5_new < best_c5 - 1e-12:
                    direction = h_new - h_full
                    alpha, val_ls = line_search_dir(h_full, direction, n_full, n_grid=18)
                    if alpha > 1e-8:
                        h_ls = np.clip(h_full + alpha * direction, 0.0, 1.0)
                        h_ls = proj_box_sum(h_ls, n_full / 2.0)
                        c5_ls = eval_c5(h_ls)
                        if c5_ls < best_c5 - 1e-12:
                            best_c5 = c5_ls
                            best_h = h_ls.copy()
                            h_full = h_ls.copy()
                            improved = True
                            break
                    if c5_new < best_c5 - 1e-12:
                        best_c5 = c5_new
                        best_h = h_new.copy()
                        h_full = h_new.copy()
                        improved = True
                        break

            if not improved:
                corrs_chk = corr_raw(h_full)
                active_expand = select_active(corrs_chk, ncols, 0.05, 4, 75)
                if len(active_expand) <= len(active) + 1:
                    break
                active = active_expand

        best_c5 = eval_c5(best_h)
        n_best = len(best_h)

    # Phase 4: Pair-transfer refinement
    if rem() > 60:
        n_pt = n_best
        h_pt = best_h.copy()
        rng_pt = np.random.RandomState(seed + 500)
        stall = 0

        for it in range(2000):
            if rem() < 15:
                break
            # Heuristic: try pairs near active lags
            corrs_pt = corr_raw(h_pt)
            max_idx = int(np.argmax(corrs_pt))
            d_max = max_idx - (n_pt - 1)
            
            if rng_pt.rand() < 0.5 and abs(d_max) < n_pt:
                # Try transfer along the active lag direction
                a = rng_pt.randint(max(0, -d_max), min(n_pt, n_pt - d_max))
                b = a + d_max
                if 0 <= a < n_pt and 0 <= b < n_pt and a != b:
                    h_try, val_try = pair_transfer_search(h_pt, n_pt, a, b, n_grid=25)
                    if val_try < best_c5 - 1e-12:
                        best_c5 = val_try
                        best_h = h_try.copy()
                        h_pt = h_try.copy()
                        stall = 0
                        continue
            # Random pair
            a = rng_pt.randint(n_pt)
            b = rng_pt.randint(n_pt)
            if a == b:
                continue
            h_try, val_try = pair_transfer_search(h_pt, n_pt, a, b, n_grid=20)
            if val_try < best_c5 - 1e-12:
                best_c5 = val_try
                best_h = h_try.copy()
                h_pt = h_try.copy()
                stall = 0
            else:
                stall += 1
            if stall > 100:
                break

    # Phase 5: Multi-resolution exploration
    if rem() > 100:
        n_cur = len(best_h)
        for dn in [-3, -2, -1, 1, 2, 3, 4, -4, 5, -5]:
            if rem() < 35:
                break
            n_try = n_cur + dn
            if n_try < 40 or n_try > 250:
                continue
            h_try = transfer_to_n(best_h, n_try)
            n_try = len(h_try)
            c5_try = eval_c5(h_try)

            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
                n_cur = n_try
                continue

            # Quick LP polish at this resolution
            if rem() > 50:
                if n_try % 2 == 0:
                    m_tr = n_try // 2
                    q_tr = sym_from_h(h_try)
                    corrs_tr = corr_raw(build_sym(q_tr))
                    act_tr = select_active(corrs_tr, 2 * n_try - 1, 0.03, 3, 50)
                    q_tr, _ = solve_tangent_sym(q_tr, n_try, act_tr, 0.05)
                    h_re = build_sym(q_tr)
                    c5_re = eval_c5(h_re)
                    if c5_re < best_c5:
                        best_c5 = c5_re
                        best_h = h_re.copy()
                        n_cur = n_try

    # Phase 6: Restart from different patterns
    if rem() > 80:
        n_rs = len(best_h)
        rng_rs = np.random.RandomState(seed + 7777)

        for rs in range(6):
            if rem() < 40:
                break
            pat = rng_rs.randint(4)
            if pat == 0:
                h_r = make_binary_start(n_rs, seed + 2000 + rs)
                h_r = 0.25 + 0.5 * h_r + 0.15 * rng_rs.randn(n_rs)
            elif pat == 1:
                x_r = np.linspace(0, 1, n_rs)
                h_r = 0.5 + 0.25 * np.sin(2 * np.pi * x_r) + 0.1 * np.sin(4 * np.pi * x_r)
                h_r += 0.1 * rng_rs.randn(n_rs)
            elif pat == 2:
                h_r = np.zeros(n_rs)
                bl = n_rs // 4
                h_r[:bl] = 1.0
                h_r[bl * 2:bl * 3] = 1.0
                h_r += 0.08 * rng_rs.randn(n_rs)
            else:
                x_r = np.linspace(0, 1, n_rs)
                h_r = 0.5 + 0.35 * np.exp(-30 * (x_r - 0.3) ** 2) + 0.2 * np.exp(-30 * (x_r - 0.7) ** 2)
                h_r += 0.08 * rng_rs.randn(n_rs)

            h_r = proj_box_sum(h_r, n_rs / 2.0)
            c5_r = eval_c5(h_r)

            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r.copy()
                continue

            if c5_r < best_c5 + 0.008 and rem() > 35:
                if n_rs % 2 == 0:
                    m_rr = n_rs // 2
                    q_rr = sym_from_h(h_r)
                    corrs_rr = corr_raw(build_sym(q_rr))
                    act_rr = select_active(corrs_rr, 2 * n_rs - 1, 0.03, 3, 50)
                    for rho_rr in [0.1, 0.05, 0.025]:
                        q_nw, _ = solve_tangent_sym(q_rr, n_rs, act_rr, rho_rr)
                        h_nw = build_sym(q_nw)
                        c5_nw = eval_c5(h_nw)
                        if c5_nw < best_c5:
                            best_c5 = c5_nw
                            best_h = h_nw.copy()
                            break

    # Phase 7: Final LP polish with tight trust region
    if rem() > 20:
        n_fin = len(best_h)
        if n_fin % 2 == 0:
            q_fin = sym_from_h(best_h)
            h_fin = build_sym(q_fin)
            corrs_fin = corr_raw(h_fin)
            active_fin = select_active(corrs_fin, 2 * n_fin - 1, 0.015, 2, 45)
            for rho_f in [0.04, 0.02, 0.01, 0.005]:
                q_nw, _ = solve_tangent_sym(q_fin, n_fin, active_fin, rho_f)
                h_nw = build_sym(q_nw)
                c5_nw = eval_c5(h_nw)
                if c5_nw < best_c5 - 1e-13:
                    best_c5 = c5_nw
                    best_h = h_nw.copy()
                    q_fin = sym_from_h(h_nw)
        else:
            h_fin = best_h.copy()
            corrs_fin = corr_raw(h_fin)
            active_fin = select_active(corrs_fin, 2 * n_fin - 1, 0.015, 2, 45)
            for rho_f in [0.03, 0.015, 0.008]:
                h_nw, _ = solve_tangent_full(h_fin, active_fin, rho_f)
                h_nw = proj_box_sum(h_nw, n_fin / 2.0)
                c5_nw = eval_c5(h_nw)
                if c5_nw < best_c5 - 1e-13:
                    best_c5 = c5_nw
                    best_h = h_nw.copy()
                    h_fin = h_nw.copy()

    # Phase 8: Final pair-transfer micro-polish
    if rem() > 10:
        n_mp = len(best_h)
        rng_mp = np.random.RandomState(seed + 4242)
        for _ in range(80):
            if rem() < 8:
                break
            a = rng_mp.randint(n_mp)
            b = rng_mp.randint(n_mp)
            if a == b:
                continue
            h_try, val_try = pair_transfer_search(best_h, n_mp, a, b, n_grid=20)
            if val_try < best_c5 - 1e-12:
                best_c5 = val_try
                best_h = h_try.copy()

    # Final validation
    n_out = len(best_h)
    best_h = proj_box_sum(best_h, n_out / 2.0)
    best_c5 = eval_c5(best_h)
    return best_h, best_c5, n_out
