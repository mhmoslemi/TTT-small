def project_h(x, target):
    lo, hi = -1.0, 1.0

def score(h):
    n = len(h)
    return float(np.correlate(h, 1.0 - h, mode='full').max() * 2.0 / n)


def smooth_obj(h, beta):
    n = len(h)
    corrs = np.correlate(h, 1.0 - h, mode='full')
    return float(logsumexp(beta * corrs) / beta * 2.0 / n)


def smooth_grad(h, beta):
    n = len(h)
    corrs = np.correlate(h, 1.0 - h, mode='full')
    sh = corrs - corrs.max()
    w = np.exp(beta * sh)
    ws = w.sum()
    w = w / ws if ws > 1e-30 else np.ones_like(w) / len(w)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        off = k - (n - 1)
        if off == 0:
            grad[:] += w[k] * (1.0 - 2.0 * h)
        elif off > 0:
            idx = np.arange(0, n - off)
            grad[idx] += w[k] * (1.0 - h[idx + off])
            grad[idx + off] += w[k] * (-h[idx])
        else:
            idx = np.arange(-off, n)
            grad[idx] += w[k] * (1.0 - h[idx + off])
            grad[idx + off] += w[k] * (-h[idx])
    return grad * 2.0 / n


def slsqp_step(h0, target, beta, maxiter):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(n)}]

    def _obj(x):
        return smooth_obj(x, beta)

    def _grad(x):
        return smooth_grad(x, beta)

    try:
        res = minimize(_obj, h0, jac=_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        h_out = project_h(res.x, target)

    except Exception:
        h_out = project_h(h0, target)
    return h_out, score(h_out)

                    if c5_p2 < best_c5:
                        best_c5 = c5_p2
                        best_h = h_p2

    # Phase 6: Final intensive polish
    n_f = len(best_h)
    ts_f = n_f / 2.0
    if remaining() > 20:
        for beta in [3000, 8000, 15000]:
            if remaining() < 15:
                break
            h_f, c5_f = slsqp_step(best_h, ts_f, beta, 600)
            if c5_f < best_c5:
                best_c5 = c5_f
                best_h = h_f

    # Phase 7: Short continuous SA + polish
    if remaining() > 30:
        h_cs2, s_cs2 = continuous_sa(best_h, 100000, 0.02, 0.001, rng, stop)
        if s_cs2 < best_c5:
            best_c5 = s_cs2
            best_h = h_cs2
        if remaining() > 15:
            h_f2, c5_f2 = slsqp_step(best_h, ts_f, 10000, 500)
            if c5_f2 < best_c5:
                best_c5 = c5_f2
                best_h = h_f2

    # Final validation and return
    best_h = project_h(best_h, len(best_h) / 2.0)
    final_c5 = score(best_h)
    # Use the recomputed exact value
    if final_c5 < best_c5:
        best_c5 = final_c5
    return best_h, best_c5, len(best_h)
