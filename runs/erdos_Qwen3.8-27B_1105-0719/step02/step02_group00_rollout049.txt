def compute_profile(h):

def repair_feasible(vec):
    # Clip and enforce sum constraint
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        # Need to add mass - add to positions with most room
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx])
            h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        # Need to remove mass - remove from positions with most mass
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    # Final correction

        else:
            idx = int(np.argmax(h))

        h = np.clip(h, 0.0, 1.0)
    return h

def build_gradient(h, n, dx):
    """Build the gradient matrix G where G[m, :] is grad of P_m w.r.t. h"""
    M = 2 * n - 1
    G = np.zeros((M, n))
    for m in range(M):
        d = m - (n - 1)
        if d == 0:
            G[m, :] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            # P_d = dx * sum_{i=0}^{n-d-1} h[i+d]*(1-h[i])

def lp_trust_region(h, radius=0.03, max_active=25, solver='ECOS'):
    """Solve LP trust region subproblem using cvxpy."""
    import cvxpy as cp
    
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    
    profile = compute_profile(h)
    c_cur = np.max(profile)
    
    # Select active set
    median_p = np.median(profile)
    beam = max(0.003, 0.08 * (c_cur - median_p))
    active = np.where(profile >= c_cur - beam)[0]
    if len(active) > max_active:
        top_idx = np.argsort(profile[active])[-max_active:]
        active = active[top_idx]
    
    # Build gradient for active rows
    G = build_gradient(h, n, dx)
    G_a = G[active, :]  # shape (len(active), n)
    p_a = profile[active]
    
    # LP: minimize t
    # s.t. t >= p_a[j] + G_a[j, :] @ (x - h) for all j in active
    #      sum(x) = n/2
    #      h[i] - radius <= x[i] <= h[i] + radius
    #      0 <= x[i] <= 1
    
    x = cp.Variable(n)
    t = cp.Variable()
    
    constraints = [
        sum(x) == n / 2.0,
        x >= np.maximum(0, h - radius),
        x <= np.minimum(1, h + radius),
        t >= p_a + G_a @ (x - h)  # t >= p_a[j] + G_a[j,:] @ (x-h) for each j
    ]
    
    # Add L1 regularization to encourage small steps
    obj = cp.Minimize(t + 0.01 * cp.norm1(x - h))
    
    prob = cp.Problem(obj, constraints)
    
    try:
        prob.solve(solver=solver)
        if x.value is not None and np.isfinite(x.value).all():
            h_new = repair_feasible(np.clip(x.value, 0, 1))
            c_new = exact_score(h_new)
            if c_new < c_cur - 1e-12:
                return h_new, c_new
    except Exception:
        pass
    
    return None, c_cur

def structured_perturb(h, rng, mode):
    """Generate structured perturbations."""
    n = len(h)
    if mode == 'smooth':
        # Smooth perturbation using low-frequency Fourier modes
        freqs = np.arange(1, min(5, n//4) + 1)
        pert = np.zeros(n)
        for f in freqs:
            amp = rng.uniform(0.01, 0.05) / f
            phase = rng.uniform(0, 2 * np.pi)
            pert += amp * np.sin(2 * np.pi * f * np.arange(n) / n + phase)
        return repair_feasible(h + pert)
    elif mode == 'block':
        # Block perturbation
        n_blocks = rng.randint(2, 5)
        block_size = n // n_blocks
        pert = np.zeros(n)
        for b in range(n_blocks):
            start = b * block_size
            end = min((b+1) * block_size, n)
            pert[start:end] = rng.uniform(-0.03, 0.03)
        pert -= pert.mean()
        return repair_feasible(h + pert)
    elif mode == 'antisym':
        # Anti-symmetric perturbation (preserves sum automatically)
        half = n // 2
        pert = np.zeros(n)
        p = rng.uniform(-0.02, 0.02, half)
        if n % 2 == 0:
            pert[:half] = p
            pert[half:] = -p
        return repair_feasible(h + pert)
    else:
        # Random with zero mean
        z = rng.randn(n)
        z -= z.mean()
        z /= (np.linalg.norm(z) / n)
        return repair_feasible(h + 0.02 * z)

        x <= upper,

    n = len(h)

        amp = eps / (f + 1)
        phase = rng.uniform(0, 2 * np.pi)
        pert += amp * np.sin(2 * np.pi * f * np.arange(n) / n + phase)
    return repair_feasible(h + pert)

def perturb_block(h, rng, eps=0.03):
    n = len(h)

    pert = np.zeros(n)
    for b in range(nb):
        s, e = b * bs, min((b + 1) * bs, n)
        pert[s:e] = rng.uniform(-eps, eps)
    pert -= pert.mean()
