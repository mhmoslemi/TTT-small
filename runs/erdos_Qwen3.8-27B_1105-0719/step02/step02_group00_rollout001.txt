def repair_feasible(vec):

        return h

            if add > 0:
                h[idx] += add
            remaining -= add

    else:

def build_grad_matrix(h, n, dx):
    """Build the gradient matrix for all M=2n-1 lags."""
    M = 2 * n - 1
    G = np.zeros((M, n))
    for m in range(M):
        d = m - (n - 1)
        if d == 0:
            G[m, :] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            # P_m = dx * sum_{i=0}^{n-1-d} h[i+d]*(1-h[i])
            # dP/dh[i] for i < n-d: -dx*h[i+d]
            # dP/dh[i] for i >= d: dx*(1-h[i-d])
            G[m, :n-d] += -dx * h[d:]
            G[m, d:] += dx * (1.0 - h[:n-d])

            G[m, :n-dp] += dx * (1.0 - h[dp:])
            G[m, dp:] += -dx * h[:n-dp]
    return G

def lp_trust_region(h, active_rows, beam, trust_radius, n, dx):
    """Solve LP trust region subproblem using cvxpy."""
    M = 2 * n - 1
    profile = compute_profile(h)
    G = build_grad_matrix(h, n, dx)
    
    # Active rows
    thresh = max(profile) - beam
    if active_rows is None:
        active = np.where(profile >= thresh)[0]
    else:
        active = active_rows
    
    if len(active) == 0:
        active = np.array([np.argmax(profile)])
    
    # Limit active rows
    if len(active) > 40:
        top_idx = np.argsort(profile[active])[-40:]
        active = active[top_idx]
    
    # LP: minimize t
    # s.t. t >= p_m(x) + G[m,:]^T (z - x) for m in active
    #      sum(z) = n/2
    #      0 <= z <= 1
    #      |z - x| <= trust_radius (element-wise)
    
    z = cp.Variable(n)
    t = cp.Variable()
    
    constraints = [
        z >= 0,
        z <= 1,
        cp.sum(z) == n / 2.0,
        z >= h - trust_radius,
        z <= h + trust_radius
    ]
    
    G_active = G[active, :]
    p_active = profile[active]
    
    for i, m in enumerate(active):
        constraints.append(t >= p_active[i] + G_active[i, :] @ (z - h))
    
    obj = cp.Minimize(t)
    prob = cp.Problem(obj, constraints)
    
    try:
        prob.solve(solver=cp.ECOS, max_iters=5000)
        if z.value is not None and np.isfinite(z.value).all():
            return repair_feasible(z.value)
    except Exception:
        pass
    
    return None

import numpy as np

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx] + 1e-15)
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx] + 1e-15)
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmin(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def build_grad_matrix(h, n, dx):
    M = 2 * n - 1
    G = np.zeros((M, n))
    for m in range(M):
        d = m - (n - 1)
        if d == 0:
            G[m, :] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            G[m, :n-d] += -dx * h[d:]
            G[m, d:] += dx * (1.0 - h[:n-d])
        else:
            dp = -d
            G[m, :n-dp] += dx * (1.0 - h[dp:])
            G[m, dp:] += -dx * h[:n-dp]
    return G

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    if new_n < n:
        xs = (np.arange(n) + 0.5) * (2.0 / n)
        xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
        full_xs = np.concatenate([[0.0], xs, [2.0]])
        full_h = np.concatenate([[h[0]], h, [h[-1]]])
        h_new = np.interp(xt, full_xs, full_h)
    else:
        xs = (np.arange(n) + 0.5) * (2.0 / n)
        xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
        full_xs = np.concatenate([[0.0], xs, [2.0]])
        full_h = np.concatenate([[h[0]], h, [h[-1]]])
        h_new = np.interp(xt, full_xs, full_h)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

def symmetrize(h):
    return (h + h[::-1]) / 2.0

def lp_trust_region(h, beam, trust_radius, max_active=35):
    n = len(h)
    dx = 2.0 / n
    profile = compute_profile(h)
    c_max = np.max(profile)
    thresh = c_max - beam
    active = np.where(profile >= thresh)[0]
    if len(active) == 0:
        active = np.array([np.argmax(profile)])
    if len(active) > max_active:
        top_idx = np.argsort(profile[active])[-max_active:]
        active = active[top_idx]
    
    G = build_grad_matrix(h, n, dx)
    G_a = G[active, :]
    p_a = profile[active]
    
    z = cp.Variable(n)
    t = cp.Variable()
    target = n / 2.0
    constraints = [
        z >= 0, z <= 1,
        cp.sum(z) == target,
        z >= h - trust_radius,
        z <= h + trust_radius
    ]
    for i in range(len(active)):
        constraints.append(t >= p_a[i] + G_a[i, :] @ (z - h))
    
    obj = cp.Minimize(t + 1e-6 * cp.norm(z - h, 2)**2)
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=cp.ECOS, max_iters=10000)
        if z.value is not None and np.isfinite(z.value).all():
            return repair_feasible(z.value)
    except Exception:
        pass
    return None

def active_gradient_step(h, beam, max_iters=30):
    n = len(h)
    dx = 2.0 / n
    h_curr = h.copy()
    best_c = exact_score(h_curr)
    
    for it in range(max_iters):
        profile = compute_profile(h_curr)
        c = np.max(profile)
        thresh = c - beam
        active = np.where(profile >= thresh)[0]
        if len(active) == 0:
            active = np.array([np.argmax(profile)])
        if len(active) > 25:
            top_idx = np.argsort(profile[active])[-25:]
            active = active[top_idx]
        
        G = build_grad_matrix(h_curr, n, dx)
        G_a = G[active, :]
        w = profile[active] - thresh
        w = np.maximum(w, 0.01)
        w = w / w.sum()
        grad = G_a.T @ w
        grad -= grad.mean()
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad /= gnorm
        
        improved = False
        for a in [0.02, 0.012, 0.008, 0.005, 0.003, 0.002, 0.001, 0.0006, 0.0003]:
            h_try = repair_feasible(h_curr - a * grad)
            s = exact_score(h_try)
            if s < best_c - 1e-14:
                h_curr = h_try
                best_c = s
                improved = True
                break
        if not improved:
            break
    return h_curr, best_c

def momentum_descent(h, max_iters=100, momentum=0.4):
    n = len(h)
    dx = 2.0 / n

        if c < best_c:

        thresh = c - max(0.003, 3.0 * (c - np.median(profile)))
        active = np.where(profile >= thresh)[0]

            top_idx = np.argsort(profile[active])[-15:]
            active = active[top_idx]
        G = build_grad_matrix(h_curr, n, dx)
        G_a = G[active, :]
        w = profile[active] - thresh
        w = np.maximum(w, 0.01)
        w = w / w.sum()
        grad = G_a.T @ w
        grad -= grad.mean()
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad /= gnorm
        velocity = momentum * velocity + grad
        vnorm = np.linalg.norm(velocity)
        if vnorm > 0:
            velocity /= vnorm
        improved = False
        for a in [alpha, alpha*0.5, alpha*0.25, alpha*0.1]:
            h_try = repair_feasible(h_curr - a * velocity)
            s = exact_score(h_try)
            if s < best_c - 1e-14:
                h_curr = h_try

                improved = True
                alpha = min(alpha * 1.05, 0.04)

    return h_curr, best_c

    pats = []
    x = np.linspace(0, 2, n)
    for freq in [1, 2, 3, 4, 5]:
        for amp in [0.1, 0.2, 0.35]:
            p = 0.5 + amp * np.sin(np.pi * freq * x + rng.uniform(0, 2*np.pi))
            pats.append(repair_feasible(np.clip(p, 0, 1)))
    nblocks = max(2, n // 8)
    block = n // nblocks
    for phase in range(3):
        p = np.zeros(n)

    for _ in range(4):
        p = rng.uniform(0.15, 0.85, n)
        pats.append(repair_feasible(p))
    return pats
