def corr_grad_h(h, d):
    """Gradient of corr at lag d wrt h. d >= 0 or d < 0."""
    n = len(h)
    g = np.zeros(n)
    if d >= 0:
        if n - d > 0:
            g[:n - d] += (1.0 - h[d:n])
        if d < n:
            g[d:] += (-h[:n - d])
    else:
        dd = -d
        if n - dd > 0:
            g[dd:] += (1.0 - h[:n - dd])
            g[:n - dd] += (-h[dd:])
    return g


def smooth_max_grad_q(q, n_full, beta):
    """Compute smooth max of corrs and gradient wrt q (symmetric reduction)."""
    m = n_full // 2
    h = build_sym_h(q)
    dx = 2.0 / n_full
    corrs = corr_raw(h) * dx
    max_c = corrs.max()
    shifted = corrs - max_c
    # Clip to prevent overflow
    shifted = np.maximum(shifted, -50.0 / beta)
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(len(corrs)) / len(corrs)
    else:
        w /= w_sum
    val = max_c + np.log(w_sum) / beta

    # Compute gradient wrt h
    grad_h = np.zeros(n_full)
    for k_idx in range(2 * n_full - 1):
        wk = w[k_idx]
        if wk < 1e-14:
            continue
        d = k_idx - (n_full - 1)
        g = corr_grad_h(h, d)
        grad_h += wk * g
    grad_h *= dx

    # Gradient wrt q
    grad_q = grad_h[:m] - grad_h[n_full - 1:m - 1:-1]
    return val, grad_q


def smooth_max_obj_q(q, n_full, beta):
    return smooth_max_grad_q(q, n_full, beta)[0]


def smooth_max_jac_q(q, n_full, beta):
    return smooth_max_grad_q(q, n_full, beta)[1]


def optimize_smooth(q0, n_full, beta_list, maxiter_per_beta=250):
    m = n_full // 2
    q = np.clip(q0.copy(), 0, 1)
    best_score = eval_score(build_sym_h(q))
    best_q = q.copy()

    for beta in beta_list:
        bounds = [(0.0, 1.0)] * m
        try:
            res = minimize(
                lambda xx: smooth_max_obj_q(xx, n_full, beta),
                q,
                jac=lambda xx: smooth_max_jac_q(xx, n_full, beta),
                method='L-BFGS-B',
                bounds=bounds,
                options={'maxiter': maxiter_per_beta, 'ftol': 1e-18, 'gtol': 1e-12, 'maxls': 40}
            )
            q_new = np.clip(res.x, 0, 1)
        except Exception:
            q_new = q.copy()

        h_new = build_sym_h(q_new)
        sc = eval_score(h_new)
        if sc < best_score:
            best_score = sc
            best_q = q_new.copy()
        q = q_new

    return best_q, best_score, build_sym_h(best_q)


def corr_at_k(h, k):
    """Compute corr at index k of full correlate (without dx)."""
    n = len(h)
    d = k - (n - 1)
    if d >= 0:
        if d >= n:
            return 0.0
        return float(np.sum(h[:n - d] * (1.0 - h[d:n])))
    else:
        dd = -d
        if dd >= n:
            return 0.0
        return float(np.sum(h[dd:n] * (1.0 - h[:n - dd])))


def corr_grad_h_at_k(h, k):
    """Gradient of corr at index k wrt h."""
    n = len(h)
    d = k - (n - 1)
    return corr_grad_h(h, d)


def slsqp_active_set(q, n_full, active_k, maxiter=200, ftol=1e-14):
    """Minimize t s.t. t >= corr[k]*dx for k in active_k, 0<=q<=1."""
    m = n_full // 2
    dx = 2.0 / n_full
    h_c = build_sym_h(q)
    corrs_c = corr_raw(h_c) * dx
    t_start = corrs_c.max() + 1e-10

    x0 = np.zeros(m + 1)
    x0[:m] = q
    x0[m] = t_start

    def obj(x):
        return x[m]

    def obj_grad(x):
        g = np.zeros(m + 1)
        g[m] = 1.0
        return g

    def con_fun(x, kk):
        qv = np.clip(x[:m], 0, 1)
        hv = build_sym_h(qv)
        return x[m] - corr_at_k(hv, kk) * dx

    def con_jac(x, kk):
        qv = np.clip(x[:m], 0, 1)
        hv = build_sym_h(qv)
        g = corr_grad_h_at_k(hv, kk)
        j = np.zeros(m + 1)
        j[:m] = -dx * (g[:m] - g[n_full - 1:m - 1:-1])
        j[m] = 1.0
        return j

    constraints = []
    for kk in active_k:
        constraints.append({'type': 'ineq',
                           'fun': lambda x, k=kk: con_fun(x, k),
                           'jac': lambda x, k=kk: con_jac(x, k)})

    bounds = [(0.0, 1.0)] * m + [(0.0, 2.0)]

    try:
        res = minimize(obj, x0, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': ftol})
        q_out = np.clip(res.x[:m], 0, 1)
    except Exception:
        q_out = q.copy()

    h_out = build_sym_h(q_out)
    return q_out, h_out, eval_score(h_out)


def select_active(corrs_scaled, margin_frac=0.02, margin_abs=1e-5, pad=3, max_k=50):
    max_val = corrs_scaled.max()
    threshold = max_val - max(margin_abs, margin_frac * max_val)
    active = set(np.where(corrs_scaled >= threshold)[0].tolist())
    padded = set()
    for k in active:
        for dd in range(-pad, pad + 1):
            if 0 <= k + dd < len(corrs_scaled):
                padded.add(k + dd)
    active_list = sorted(padded)
    if len(active_list) > max_k:
        scored = sorted([(corrs_scaled[k], k) for k in active_list], reverse=True)
        active_list = sorted([k for _, k in scored[:max_k]])
    return active_list


def transfer_to_n(h_old, n_new):
    n_old = len(h_old)
    x_old = (np.arange(n_old) + 0.5) * 2.0 / n_old
    x_new = (np.arange(n_new) + 0.5) * 2.0 / n_new
    h_new = np.interp(x_new, x_old, h_old, left=h_old[0], right=h_old[-1])
    h_new = np.clip(h_new, 0, 1)
    target = n_new / 2.0
    diff = target - h_new.sum()
    interior = (h_new > 0.01) & (h_new < 0.99)
    if interior.sum() > 0:
        h_new[interior] += diff / interior.sum()
    return np.clip(h_new, 0, 1)


def epigraph_refine(h_in, n_rounds=30, time_limit=None):
    """Iterative active set refinement for symmetric h."""
    n = len(h_in)
    if n % 2 != 0:
        return h_in, eval_score(h_in)
    dx = 2.0 / n
    m = n // 2
    q = sym_from_h(h_in)
    best_score = eval_score(h_in)
    best_q = q.copy()

    for rnd in range(n_rounds):
        if time_limit is not None and time.time() > time_limit:
            break

        h_cur = build_sym_h(q)
        corrs = corr_raw(h_cur) * dx
        active = select_active(corrs, margin_frac=0.03, margin_abs=1e-5, pad=4, max_k=55)

        q_new, h_new, c5_new = slsqp_active_set(q, n, active, maxiter=150, ftol=1e-14)
        h_new = build_sym_h(np.clip(q_new, 0, 1))
        c5_new = eval_score(h_new)

        if c5_new < best_score - 1e-14:
            best_score = c5_new
            best_q = q_new.copy()
            q = q_new
            continue

        # Check hidden violations
        corrs_new = corr_raw(h_new) * dx
        max_active = corrs_new[active].max()
        hidden = np.where(corrs_new > max_active + 5e-8)[0]
        if len(hidden) > 0:
            active = sorted(set(active + list(hidden)))
            if len(active) > 80:
                active = select_active(corrs_new, margin_frac=0.01, margin_abs=1e-6, pad=3, max_k=80)
            q = sym_from_h(h_new)
            continue

        # Try broader active set
        active2 = select_active(corrs_new, margin_frac=0.08, margin_abs=5e-5, pad=5, max_k=70)
        if len(active2) > len(active):
            q_new2, h_new2, c5_new2 = slsqp_active_set(q, n, active2, maxiter=100, ftol=1e-14)
            h_new2 = build_sym_h(np.clip(q_new2, 0, 1))
            c5_new2 = eval_score(h_new2)
            if c5_new2 < best_score - 1e-14:
                best_score = c5_new2
                best_q = q_new2.copy()
                q = q_new2
                continue
        break

    return build_sym_h(best_q), best_score


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def rem():
        return budget_s - (time.time() - t0)

    best_h = None
    best_c5 = float('inf')

    # Load initial construction
    try:
        h_parent = np.array(initial_h_values, dtype=float).copy().reshape(-1)
    except (NameError, AttributeError):
        h_parent = None

    if h_parent is not None and len(h_parent) > 10:
        n_p = len(h_parent)
        if n_p % 2 == 0:
            h_parent = np.clip(h_parent, 0, 1)
            target = n_p / 2.0
            diff = target - h_parent.sum()
            interior = (h_parent > 0.01) & (h_parent < 0.99)
            if interior.sum() > 0:
                h_parent[interior] += diff / interior.sum()
            h_parent = np.clip(h_parent, 0, 1)
            best_h = h_parent.copy()
            best_c5 = eval_score(h_parent)
        else:
            m_p = n_p // 2
            q_p = np.clip(0.5 * (h_parent[:m_p] + (1.0 - h_parent[n_p - 1:m_p - 1:-1])), 0, 1)
            best_h = build_sym_h(q_p)
            best_c5 = eval_score(best_h)
    else:
        m_def = 63
        x_def = np.linspace(0, 1, m_def)
        q_def = np.clip(0.5 + 0.3 * np.sin(2 * np.pi * x_def), 0, 1)
        best_h = build_sym_h(q_def)
        best_c5 = eval_score(best_h)

    n_cur = len(best_h)

    # Phase 1: Smooth max optimization (moderate beta) to polish
    if n_cur % 2 == 0 and rem() > 200:
        q_init = sym_from_h(best_h)
        tl = time.time() + min(180, rem() * 0.3)
        beta_list = [80, 200, 500]
        q_opt, c5_opt, h_opt = optimize_smooth(q_init, n_cur, beta_list, maxiter_per_beta=200)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 2: Epigraph active set refinement (main improvement driver)
    if rem() > 150:
        tl = time.time() + min(350, rem() * 0.45)
        h_ref, c5_ref = epigraph_refine(best_h, n_rounds=40, time_limit=tl)
        if c5_ref < best_c5:
            best_c5 = c5_ref
            best_h = h_ref.copy()

    # Phase 3: Try nearby n values
    n_candidates = []
    for delta in [-8, -6, -4, -2, 2, 4, 6, 8]:
        nc = n_cur + delta
        if 56 <= nc <= 180 and nc % 2 == 0:
            n_candidates.append(nc)
    # Also try some specific values
    for ns in [64, 72, 80, 88, 96, 104, 112, 120, 128, 132, 136, 140]:
        if ns not in n_candidates and 56 <= ns <= 180:
            n_candidates.append(ns)

    n_candidates = sorted(set(n_candidates))

    for n_try in n_candidates:
        if rem() < 80:
            break
        h_try = transfer_to_n(best_h, n_try)
        n_try = len(h_try)
        if n_try % 2 != 0:
            continue
        q_try = sym_from_h(h_try)

        # Quick smooth optimization
        if rem() > 100:
            q_opt, c5_opt, h_opt = optimize_smooth(q_try, n_try, [150, 400], maxiter_per_beta=120)
            if c5_opt < c5_ref if best_h is not h_try else best_c5:
                pass
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                h_try = h_opt
        else:
            h_try = build_sym_h(q_try)

        # Epigraph refinement
        if rem() > 60:
            tl = time.time() + min(120, rem() * 0.5)
            h_ref, c5_ref2 = epigraph_refine(h_try, n_rounds=25, time_limit=tl)
            if c5_ref2 < best_c5:
                best_c5 = c5_ref2
                best_h = h_ref.copy()

    # Phase 4: Random restarts with different structure
    n_restarts = min(8, max(2, int(rem() / 100)))
    for i in range(n_restarts):
        if rem() < 80:
            break
        n_r = n_cur if i % 2 == 0 else max(56, n_cur - 6)
        n_r = min(160, max(52, n_r))
        if n_r % 2 != 0:
            n_r += 1

        m_r = n_r // 2
        x_r = np.linspace(0, 1, m_r)
        # Different parametric families
        if i % 3 == 0:
            q_r = np.clip(0.5 + 0.35 * np.sin(2 * np.pi * x_r + rng.uniform(0, 2)), 0, 1)
        elif i % 3 == 1:
            q_r = np.clip(0.5 + 0.3 * np.cos(4 * np.pi * x_r) + 0.1 * np.sin(6 * np.pi * x_r), 0, 1)
        else:
            q_r = np.clip(0.5 + 0.4 * (x_r - 0.5) + 0.15 * rng.randn(m_r), 0, 1)

        if rem() > 80:
            q_opt, c5_opt, h_opt = optimize_smooth(q_r, n_r, [100, 300], maxiter_per_beta=120)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()

            if c5_opt < best_c5 + 0.003 and rem() > 50:
                tl = time.time() + min(60, rem() * 0.5)
                h_ref, c5_ref3 = epigraph_refine(h_opt, n_rounds=20, time_limit=tl)
                if c5_ref3 < best_c5:
                    best_c5 = c5_ref3
                    best_h = h_ref.copy()

    # Phase 5: Final high-precision polish
    if rem() > 40 and best_h is not None and len(best_h) % 2 == 0:
        n_fin = len(best_h)
        q_fin = sym_from_h(best_h)

        # Very high beta smooth max
        if rem() > 50:
            q_opt, c5_opt, h_opt = optimize_smooth(q_fin, n_fin, [1000, 2000], maxiter_per_beta=300)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
                q_fin = sym_from_h(best_h)

        # Final epigraph with very tight tolerance
        if rem() > 25:
            tl = time.time() + min(30, rem() * 0.6)
            h_ref, c5_ref4 = epigraph_refine(best_h, n_rounds=20, time_limit=tl)
            if c5_ref4 < best_c5:
                best_c5 = c5_ref4
                best_h = h_ref

    # Final: one more targeted epigraph with the most critical lags
    if rem() > 15 and best_h is not None and len(best_h) % 2 == 0:
        n_fin = len(best_h)
        dx_fin = 2.0 / n_fin
        q_fin = sym_from_h(best_h)
        h_chk = build_sym_h(q_fin)
        corrs = corr_raw(h_chk) * dx_fin
        # Use very tight margin to get only the top lags
        active_tight = select_active(corrs, margin_frac=0.005, margin_abs=5e-6, pad=2, max_k=25)
        q_o, h_o, c5_o = slsqp_active_set(q_fin, n_fin, active_tight, maxiter=200, ftol=1e-16)
        h_o = build_sym_h(np.clip(q_o, 0, 1))
        c5_o = eval_score(h_o)
        if c5_o < best_c5:
            best_c5 = c5_o
            best_h = h_o

    # Ensure valid output
    if best_h is None:
        m_fb = 63
        x_fb = np.linspace(0, 1, m_fb)
        q_fb = np.clip(0.5 + 0.3 * np.sin(2 * np.pi * x_fb), 0, 1)
        best_h = build_sym_h(q_fb)

    best_h = np.asarray(best_h, dtype=float).reshape(-1)
    best_c5 = eval_score(best_h)
    return best_h, best_c5, len(best_h)
