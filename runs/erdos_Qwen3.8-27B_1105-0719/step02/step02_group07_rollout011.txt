from scipy.optimize import minimize
from scipy.special import logsumexp

def corr_vals(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def exact_score(h):
    return corr_vals(h).max()

def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def make_sym_h(q, n):
    m = n // 2
    q = np.clip(q, 0.0, 1.0)
    h = np.empty(n)
    h[:m] = q
    h[m:] = 1.0 - q[::-1]
    return h

def make_q_from_h(h, n):

    q = 0.5 * (h[:m] + 1.0 - h[n-1:n-1-m:-1])

def smooth_obj_and_grad(q, n, beta):
    h = make_sym_h(q, n)
    corrs = corr_vals(h)
    obj = logsumexp(beta * corrs) / beta
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    ws = w.sum()
    if ws < 1e-30:
        w = np.ones(2*n-1) / (2*n-1)
    else:
        w /= ws
    # Gradient with respect to h
    n_samples = len(h)
    dx = 2.0 / n_samples
    grad_h = np.zeros(n_samples)
    for k in range(2*n_samples - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        d = k - (n_samples - 1)

        grad_h[idx] += wk * (1.0 - h[idx + d])
        grad_h[idx + d] += wk * (-h[idx])
    grad_h *= dx * beta
    # Gradient with respect to q
    m = n_samples // 2
    grad_q = grad_h[:m] - grad_h[n_samples-1:n_samples-1-m:-1]
    return obj, grad_q

        )

def subgrad_step(q, n, step_size, max_iter=100):
    """Exact subgradient descent on the max correlation."""
    h = make_sym_h(q, n)
    corrs = corr_vals(h)
    for _ in range(max_iter):
        max_idx = np.argmax(corrs)
        k = max_idx
        d = k - (n - 1)
        grad_h = np.zeros(n)
        i_lo = max(0, -d)
        i_hi = min(n - 1, n - 1 - d)
        if i_lo <= i_hi:

            grad_h[idx] += (1.0 - h[idx + d])
            grad_h[idx + d] += (-h[idx])

        m = n // 2

from scipy.optimize import minimize
from scipy.special import logsumexp
import time


def corr_vals(h):

    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx


def exact_score(h):

        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:

        else:

    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)

        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:

    q = np.clip(q, 0.0, 1.0)
    h = np.empty(n)
    h[:m] = q

    q = 0.5 * (h[:m] + 1.0 - h[n-1:n-1-m:-1])

def smooth_obj_and_grad(q, n, beta):
    h = make_sym_h(q, n)
    corrs = corr_vals(h)
    obj = logsumexp(beta * corrs) / beta
    shifted = corrs - corrs.max()

        w = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        w /= ws
    dx = 2.0 / n
    grad_h = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue

        i_lo = max(0, -d)
        i_hi = min(n - 1, n - 1 - d)

        grad_h[idx] += wk * (1.0 - h[idx + d])
        grad_h[idx + d] += wk * (-h[idx])

def smooth_opt_lbfgs(q0, n, beta, maxiter=3000, gtol=1e-10):
    m = n // 2
    try:
        res = minimize(
            smooth_obj_and_grad, q0, args=(n, beta),
            jac=True, method='L-BFGS-B',
            bounds=[(0.0, 1.0)] * m,
            options={'maxiter': maxiter, 'gtol': gtol, 'ftol': 1e-16,
                     'maxls': 50}
        )
        h_new = make_sym_h(res.x, n)
        c5 = exact_score(h_new)
        return res.x, c5
    except Exception:
        return None, None


def subgrad_momentum(q, n, step_size=0.05, max_iter=300, momentum=0.9,
                     smooth_width=0.01):
    m = n // 2
    h = make_sym_h(q, n)
    v = np.zeros(m)
    best_q = q.copy()
    best_c5 = exact_score(h)
    
    for it in range(max_iter):
        corrs = corr_vals(h)
        max_val = corrs.max()
        # Smoothed subgradient: average over near-max lags
        thresh = max_val - smooth_width
        active_mask = corrs >= thresh
        n_active = active_mask.sum()
        if n_active == 0:
            active_mask[np.argmax(corrs)] = True
            n_active = 1
        
        grad_h = np.zeros(n)
        for k in np.where(active_mask)[0]:
            d = k - (n - 1)
            i_lo = max(0, -d)
            i_hi = min(n - 1, n - 1 - d)
            if i_lo > i_hi:
                continue
            idx = np.arange(i_lo, i_hi + 1)
            grad_h[idx] += (1.0 - h[idx + d]) / n_active
            grad_h[idx + d] += (-h[idx]) / n_active
        dx = 2.0 / n
        grad_h *= dx
        grad_q = grad_h[:m] - grad_h[n-1:n-1-m:-1]
        
        # Momentum update
        v = momentum * v - step_size * grad_q
        q_new = np.clip(q + v, 0.0, 1.0)
        h_new = make_sym_h(q_new, n)
        c5_new = exact_score(h_new)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_q = q_new.copy()
            # Keep momentum
        else:
            # Backtrack: reduce step
            step_size *= 0.7
            if step_size < 1e-12:
                break
        
        q = q_new
        h = h_new
    
    return best_q, best_c5


class EpiSolver:
    def __init__(self, n, active_lags):
        self.n = n
        self.m = n // 2
        self.active_lags = np.array(active_lags, dtype=int)
    
    def fun(self, x):
        return x[0]
    
    def jac(self, x):
        g = np.zeros(self.m + 1)
        g[0] = 1.0
        return g
    
    def con_fun(self, x):
        t = x[0]
        q = x[1:]
        h = make_sym_h(q, self.n)
        c = corr_vals(h)
        return t - c[self.active_lags]
    
    def con_jac(self, x):
        q = x[1:]
        h = make_sym_h(q, self.n)
        n = self.n
        m = self.m
        L = len(self.active_lags)
        J = np.zeros((L, m + 1))
        J[:, 0] = 1.0
        for i, lag in enumerate(self.active_lags):
            d = lag - (n - 1)
            i_lo = max(0, -d)
            i_hi = min(n - 1, n - 1 - d)
            Gh = np.zeros(n)
            if i_lo <= i_hi:
                idx = np.arange(i_lo, i_hi + 1)
                Gh[idx] += (1.0 - h[idx + d])
                Gh[idx + d] += (-h[idx])
            Gh *= 2.0 / n
            J[i, 1:] = -(Gh[:m] - Gh[n-1:n-1-m:-1])
        return J


def active_set_solve(q0, n, n_active, t_init, maxiter=400):
    h = make_sym_h(q0, n)
    corrs = corr_vals(h)
    active_idx = np.argsort(corrs)[::-1][:n_active]
    m = n // 2
    solver = EpiSolver(n, active_idx)
    x0 = np.concatenate([[t_init], q0])
    cons = [{'type': 'ineq', 'fun': solver.con_fun, 'jac': solver.con_jac}]
    bounds = [(0.0, 2.0)] + [(0.0, 1.0)] * m
    try:
        res = minimize(solver.fun, x0, jac=solver.jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        h_new = make_sym_h(res.x[1:], n)
        c5 = exact_score(h_new)
        return res.x[1:], c5
    except Exception:
        return None, None


def coordinate_descent(q, n, n_passes=3):
    m = n // 2
    h = make_sym_h(q, n)
    c5 = exact_score(h)
    step_sizes = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
    
    for p in range(n_passes):
        improved = False
        for i in range(m):
            for ss in step_sizes:
                for direction in [1, -1]:
                    q_trial = q.copy()
                    q_trial[i] = np.clip(q[i] + direction * ss, 0.0, 1.0)
                    h_trial = make_sym_h(q_trial, n)

                    if c5_trial < c5 - 1e-13:
                        q = q_trial
                        h = h_trial
                        c5 = c5_trial
                        improved = True
                        break
                if improved:
                    break
    return q, c5

            n_try = n + delta
            if n_try < 40 or n_try % 2 != 0 or remaining() < 25:
                continue
            x_old = np.linspace(0, 2, n, endpoint=False)
            x_new = np.linspace(0, 2, n_try, endpoint=False)
            h_interp = np.interp(x_new, x_old, h_best)
            h_interp = project_feasible(h_interp, n_try / 2.0)
            m_try = n_try // 2
            q_try = make_q_from_h(h_interp, n_try)
            
            q_new, c5_new = smooth_opt_lbfgs(q_try, n_try, 300, maxiter=2000,
                                              gtol=1e-10)
            if q_new is not None:
                q_new2, c5_new2 = smooth_opt_lbfgs(q_new, n_try, 800,
                                                    maxiter=2000, gtol=1e-11)
                if q_new2 is not None and c5_new2 < c5_best - 1e-9:
                    c5_best = c5_new2
                    n = n_try
                    m = m_try
                    q_best = q_new2
                    h_best = make_sym_h(q_best, n)
                    print(f"  n={n}: C5={c5_best:.7f}")
    
    # Phase 10: Final combined polish
    if remaining() > 25:
        q_new, c5_new = smooth_opt_lbfgs(q_best, n, 1500, maxiter=4000, gtol=1e-12)
        if q_new is not None:
            q_best = q_new
            h_best = make_sym_h(q_best, n)
            c5_best = exact_score(h_best)
        q_sg4, c5_sg4 = subgrad_momentum(q_best, n, step_size=0.008,
                                          max_iter=300, momentum=0.6,
                                          smooth_width=0.001)
        if c5_sg4 < c5_best - 1e-12:
            c5_best = c5_sg4
            q_best = q_sg4
        q_cd2, c5_cd2 = coordinate_descent(q_best, n, n_passes=2)
        if c5_cd2 < c5_best - 1e-12:
            c5_best = c5_cd2
            q_best = q_cd2
        h_best = make_sym_h(q_best, n)
        c5_best = exact_score(h_best)
        print(f"  Final polish: C5={c5_best:.7f}")
    
    # Final validation
    h_best = np.clip(h_best, 0.0, 1.0)
    h_best = project_feasible(h_best, len(h_best) / 2.0)
    c5_best = exact_score(h_best)
    
    return h_best, c5_best, len(h_best)
