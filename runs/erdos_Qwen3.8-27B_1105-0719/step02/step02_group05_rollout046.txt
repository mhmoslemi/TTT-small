import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time


def build_h_from_q(q, n):
    m = n // 2
    return np.concatenate([q, 1.0 - q[::-1]])


def q_from_h(h, n):
    return h[:n // 2].copy()


def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def exact_score(h):
    dx = 2.0 / len(h)
    return compute_corrs(h).max() * dx


def exact_score_corrs(h):
    dx = 2.0 / len(h)
    corrs = compute_corrs(h)
    return corrs.max() * dx, corrs


def grad_corr_d_q(h, d, n):
    m = n // 2
    g_h = np.zeros(n)
    if 0 <= d < n:
        g_h[d:] += (1.0 - h[:n - d])
        g_h[:n - d] += (-h[d:])
    elif -n < d < 0:
        dd = -d
        g_h[:n - dd] += (1.0 - h[dd:])
        g_h[dd:] += (-h[:n - dd])
    return g_h[:m] - g_h[n - 1 - np.arange(m)]


def logsumexp_obj_grad_q(q, n, beta):
    m = n // 2
    h = build_h_from_q(q, n)
    dx = 2.0 / n
    corrs = compute_corrs(h)
    val = logsumexp(beta * corrs) / beta * dx
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    ws = w.sum()
    if ws < 1e-30:
        w = np.ones(len(corrs)) / len(corrs)
    else:
        w /= ws
    Gq = np.zeros(m)
    for k in range(len(corrs)):
        if w[k] < 1e-12:
            continue
        d = k - (n - 1)
        Gq += w[k] * grad_corr_d_q(h, d, n)
    return val, Gq * dx


def lbp_solve_q(q0, n, beta, maxiter=2000):
    m = n // 2

    def f(x):
        return logsumexp_obj_grad_q(x, n, beta)[0]

    def g(x):
        return logsumexp_obj_grad_q(x, n, beta)[1]

    try:
        res = minimize(f, q0, jac=g, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * m,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-15})
        return np.clip(res.x, 0.0, 1.0)
    except Exception:
        return q0.copy()

        best_h = h_sym
        best_c5 = c5_sym

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    final_c5 = exact_score(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, n
