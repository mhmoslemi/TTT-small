from scipy.optimize import minimize
from scipy.special import logsumexp
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corrs = np.correlate(h, 1.0 - h, mode='full')
    return (corrs.max() * dx, corrs)

def build_h_from_q(q, n):
    m = n // 2
    if n % 2 == 0:
        return np.concatenate([q, 1.0 - q[::-1]])
    else:
        return np.concatenate([q, [0.5], 1.0 - q[::-1]])

def compute_a_vector(i, h, n):
    """Compute the coefficient vector a_k for coordinate i's change."""
    m = n // 2
    # h[i] changes by +delta, h[n-1-i] changes by -delta
    # For anti-symmetric: h[i] = q[i], h[n-1-i] = 1-q[i]
    # But we work with full h vector
    a = np.zeros(2 * n - 1)
    d_range = np.arange(-(n-1), n)  # d = k - (n-1) for k in 0..2n-2
    
    # Terms from h[i] as first factor: (1-h[i+d]) if 0<=i+d<n
    # Terms from h[i] as second factor: -h[i-d] if 0<=i-d<n
    # Terms from h[n-1-i] as first factor: -(1-h[n-1-i+d]) if 0<=n-1-i+d<n
    # Terms from h[n-1-i] as second factor: +h[n-1-i-d] if 0<=n-1-i-d<n

        val = 0.0
        # h[i] as first: j=i, need 0<=i+d<n
        if 0 <= i + d < n:
            val += (1.0 - h[i + d])
        # h[i] as second: j=i-d, need 0<=i-d<n
        if 0 <= i - d < n:
            val -= h[i - d]
        # h[n-1-i] as first: j=n-1-i, need 0<=n-1-i+d<n
        ji = n - 1 - i
        if 0 <= ji + d < n:
            val -= (1.0 - h[ji + d])
        # h[n-1-i] as second: j=n-1-i-d, need 0<=n-1-i-d<n
        if 0 <= ji - d < n:
            val += h[ji - d]
        a[idx] = val
    return a

def coordinate_search(q_i, a, corr, dx, n_grid=200):
    """Find best delta for coordinate i given coefficient vector a and current correlations."""
    lo = -q_i
    hi = 1.0 - q_i
    # Grid search
    deltas = np.linspace(lo, hi, n_grid)
    # Objective at each delta: max(corr + a * delta) * dx
    # Vectorized: corr[:,None] + a[:,None]*deltas[None,:]
    vals = np.max(corr[:, None] + a[:, None] * deltas[None, :], axis=0) * dx
    best_idx = np.argmin(vals)
    best_val = vals[best_idx]
    best_delta = deltas[best_idx]
    
    # Refine with golden section
    # ... or just use a finer grid around the best
    if 0 < best_idx < n_grid - 1:

        sub_best = np.argmin(vals_sub)
        if vals_sub[sub_best] < best_val:

            best_delta = sub[sub_best]

from scipy.special import logsumexp
import time

    dx = 2.0 / n
    corrs = np.correlate(h, 1.0 - h, mode='full')

def compute_a_vector(i, h, n):
    ji = n - 1 - i
    a = np.zeros(2 * n - 1)
    # d ranges from -(n-1) to n-1, k = d + (n-1)
    # Term 1: (1-h[i+d]) for 0<=i+d<n
    d_min1, d_max1 = -i, n-1-i
    d_min1 = max(d_min1, -(n-1))

        a[k1:k1+(d_max1-d_min1+1)] += (1.0 - h[i+d_min1:i+d_max1+1])

        a[k2:k2+(d_max2-d_min2+1)] -= h[idx]

    hi = 1.0 - q_i
    if lo >= hi:
        return 0.0, corr.max() * dx
    deltas = np.linspace(lo, hi, n_grid)
    # For memory efficiency with large n, chunk the computation

    # Result: max over axis 0 of (2n-1, n_grid) matrix
    # For n=126, 2n-1=251, n_grid=300, so 251*300 = 75K elements, fine

def compute_a_vector(i, h, n):
    ji = n - 1 - i
    a = np.zeros(2 * n - 1)
    d_min1, d_max1 = max(-i, -(n-1)), min(n-1-i, n-1)
    if d_min1 <= d_max1:
        k1 = d_min1 + (n-1)
        a[k1:k1+(d_max1-d_min1+1)] += (1.0 - h[i+d_min1:i+d_max1+1])
    d_min2, d_max2 = max(i-(n-1), -(n-1)), min(i, n-1)
    if d_min2 <= d_max2:
        k2 = d_min2 + (n-1)
        idx = np.arange(i-d_min2, i-d_max2-1, -1)
        a[k2:k2+(d_max2-d_min2+1)] -= h[idx]
    d_min3, d_max3 = max(-ji, -(n-1)), min(n-1-ji, n-1)
    if d_min3 <= d_max3:
        k3 = d_min3 + (n-1)
        a[k3:k3+(d_max3-d_min3+1)] -= (1.0 - h[ji+d_min3:ji+d_max3+1])
    d_min4, d_max4 = max(ji-(n-1), -(n-1)), min(ji, n-1)
    if d_min4 <= d_max4:
        k4 = d_min4 + (n-1)
        idx = np.arange(ji-d_min4, ji-d_max4-1, -1)
        a[k4:k4+(d_max4-d_min4+1)] += h[idx]
    return a


def coord_optimal(q_i, a, corr, dx, n_grid=200):
    lo, hi = -q_i, 1.0 - q_i
    if lo >= hi - 1e-15:
        return 0.0, corr.max() * dx
    deltas = np.linspace(lo, hi, n_grid)
    vals = np.max(corr[:, None] + a[:, None] * deltas[None, :], axis=0) * dx
    bi = np.argmin(vals)
    bv = vals[bi]
    bd = deltas[bi]
    if 0 < bi < n_grid - 1:
        lr = np.linspace(deltas[bi-1], deltas[bi+1], 40)
        vr = np.max(corr[:, None] + a[:, None] * lr[None, :], axis=0) * dx
        ri = np.argmin(vr)
        if vr[ri] < bv - 1e-15:
            bv, bd = vr[ri], lr[ri]
    return bd, bv


def cd_sweep(q, n, corr, dx, rng, n_grid=200):
    m = n // 2
    h = build_h_from_q(q, n)
    n_imp = 0
    order = rng.permutation(m)
    for i in order:
        a = compute_a_vector(i, h, n)
        delta, _ = coord_optimal(q[i], a, corr, dx, n_grid)
        if abs(delta) < 1e-12:
            continue
        best_d = 0.0
        best_val = corr.max() * dx
        for attempt in range(10):
            d_t = delta * (0.6 ** attempt)
            if abs(d_t) < 1e-13:
                break
            nq = q[i] + d_t
            if nq < -1e-12 or nq > 1.0 + 1e-12:
                continue
            q[i] = nq
            h[i] += d_t
            h[n-1-i] -= d_t
            nc = np.correlate(h, 1.0 - h, mode='full')
            nv = nc.max() * dx
            if nv < best_val - 1e-14:
                best_d, best_val = d_t, nv
                corr = nc
                break
            else:
                q[i] -= d_t
                h[i] -= d_t
                h[n-1-i] += d_t
        if best_d != 0.0:
            n_imp += 1
        else:
            h = build_h_from_q(q, n)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return q, corr, corr.max() * dx, n_imp


def smooth_q_val_grad(q, n, beta):
    h = build_h_from_q(q, n)
    dx = 2.0 / n
    corrs = np.correlate(h, 1.0 - h, mode='full')
    val = logsumexp(beta * corrs) / beta * dx
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    ws = w.sum()
    if ws < 1e-30:
        w = np.ones(len(corrs)) / len(corrs)
    else:
        w /= ws
    m = n // 2
    Gq = np.zeros(m)
    for i in range(m):
        a = compute_a_vector(i, h, n)
        Gq[i] = np.dot(w, a)
    return val, Gq * dx


def smooth_lbfgs(q0, n, beta, maxiter=300):
    h0 = build_h_from_q(q0, n)
    dx = 2.0 / n

    def obj(x):
        return smooth_q_val_grad(x, n, beta)[0]

    def grad(x):
        return smooth_q_val_grad(x, n, beta)[1]

    m = n // 2
    try:
        res = minimize(obj, q0, jac=grad, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * m,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-14})
        return np.clip(res.x, 0.0, 1.0)
    except Exception:
        return q0.copy()


def evaluate_n_values(initial_h, best_c5, best_h, best_n, rng, time_limit):
    results = []
    h_ref = best_h
    for n_try in [64, 72, 80, 88, 96, 104, 112, 120, 128, 136, 144]:
        if time.time() - time_limit > 0:
            break
        if n_try == len(h_ref):
            continue
        dx_t = 2.0 / n_try
        ts = n_try / 2.0
        x_orig = np.linspace(0.0, 2.0, len(h_ref), endpoint=False) + dx_t * 0
        x_new = np.linspace(0.0, 2.0, n_try, endpoint=False)
        h_interp = np.interp(x_new, np.linspace(0, 2, len(h_ref), endpoint=False), h_ref)
        m_t = n_try // 2
        q_init = h_interp[:m_t].copy()
        q_init = np.clip(q_init, 0.0, 1.0)
        h_t = build_h_from_q(q_init, n_try)
        corr_t = np.correlate(h_t, 1.0 - h_t, mode='full')
        c5_t = corr_t.max() * dx_t
        for sweep in range(30):
            if time.time() - time_limit > 0:
                break
            q_init, corr_t, c5_t, n_imp = cd_sweep(q_init, n_try, corr_t, dx_t, rng, n_grid=150)
            if n_imp == 0:
                break
        h_t = build_h_from_q(q_init, n_try)
        s_t = np.correlate(h_t, 1.0 - h_t, mode='full').max() * dx_t
        results.append((s_t, h_t, n_try))
    return results


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)

    def remaining():
        return budget_s - (time.time() - t_start)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    m_init = n_init // 2
    q = h_init[:m_init].copy()
    q = np.clip(q, 0.0, 1.0)
    h = build_h_from_q(q, n_init)
    dx = 2.0 / n_init
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = corr.max() * dx
    best_h = h.copy()
    best_n = n_init
    best_q = q.copy()

    # Phase 1: Aggressive coordinate descent on initial n
    n_grid_cd = 250
    for sweep in range(200):
        if remaining() < 30:
            break
        q, corr, c5, n_imp = cd_sweep(q, n_init, corr, dx, rng, n_grid=n_grid_cd)
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_h = build_h_from_q(q, n_init)
            best_q = q.copy()
        if n_imp == 0:
            break

    # Phase 2: Smooth L-BFGS-B with increasing beta
    for beta in [100, 300, 800, 2000, 5000]:
        if remaining() < 60:
            break
        q_smooth = smooth_lbfgs(best_q, n_init, beta, maxiter=500)
        h_s = build_h_from_q(q_smooth, n_init)
        c5_s = np.correlate(h_s, 1.0 - h_s, mode='full').max() * dx
        if c5_s < best_c5 - 1e-14:
            best_c5 = c5_s
            best_h = h_s.copy()
            best_q = q_smooth.copy()
            # Continue with more CD sweeps from new point
            corr_s = np.correlate(h_s, 1.0 - h_s, mode='full')
            for sweep2 in range(50):
                if remaining() < 30:
                    break
                q_smooth, corr_s, c5_s2, n_imp2 = cd_sweep(q_smooth, n_init, corr_s, dx, rng, n_grid=200)
                if c5_s2 < best_c5 - 1e-14:
                    best_c5 = c5_s2
                    best_h = build_h_from_q(q_smooth, n_init)
                    best_q = q_smooth.copy()
                if n_imp2 == 0:
                    break

    # Phase 3: More coordinate descent with different grid resolutions
    for ng in [100, 400, 600]:
        if remaining() < 30:
            break
        corr_cur = np.correlate(build_h_from_q(best_q, n_init), 1.0 - build_h_from_q(best_q, n_init), mode='full')
        for sweep in range(80):
            if remaining() < 20:
                break
            q_cur, corr_cur, c5_cur, n_imp = cd_sweep(best_q, n_init, corr_cur, dx, rng, n_grid=ng)
            if c5_cur < best_c5 - 1e-14:
                best_c5 = c5_cur
                best_h = build_h_from_q(q_cur, n_init)
                best_q = q_cur.copy()
            if n_imp == 0:
                break

    # Phase 4: Try different n values
    t_limit_n = time.time() + min(remaining() * 0.4, 300)
    results_n = evaluate_n_values(h_init, best_c5, best_h, best_n, rng, t_limit_n)
    for s_t, h_t, n_t in results_n:
        if s_t < best_c5 - 1e-12:
            best_c5 = s_t
            best_h = h_t.copy()
            best_n = n_t
            m_t = n_t // 2
            best_q = h_t[:m_t].copy()

    # Phase 5: Refine best with more CD
    n_best = best_n
    dx_best = 2.0 / n_best
    m_best = n_best // 2
    if len(best_h) != n_best:
        best_q = best_h[:m_best].copy()
        best_h = build_h_from_q(best_q, n_best)
    corr_b = np.correlate(best_h, 1.0 - best_h, mode='full')
    for sweep in range(150):
        if remaining() < 20:
            break
        best_q, corr_b, c5_b, n_imp = cd_sweep(best_q, n_best, corr_b, dx_best, rng, n_grid=300)
        if c5_b < best_c5 - 1e-14:
            best_c5 = c5_b
            best_h = build_h_from_q(best_q, n_best)
        if n_imp == 0:
            break

    # Phase 6: Smooth polish
    for beta in [3000, 8000, 15000]:
        if remaining() < 30:
            break
        q_pol = smooth_lbfgs(best_q, n_best, beta, maxiter=400)
        h_pol = build_h_from_q(q_pol, n_best)
        c5_pol = np.correlate(h_pol, 1.0 - h_pol, mode='full').max() * dx_best
        if c5_pol < best_c5 - 1e-14:
            best_c5 = c5_pol
            best_h = h_pol
            best_q = q_pol
            # Quick CD from new point
            corr_pol = np.correlate(h_pol, 1.0 - h_pol, mode='full')
            for s in range(30):
                if remaining() < 15:
                    break
                best_q, corr_pol, c5_pol2, ni = cd_sweep(best_q, n_best, corr_pol, dx_best, rng, n_grid=200)
                if c5_pol2 < best_c5 - 1e-14:
                    best_c5 = c5_pol2
                    best_h = build_h_from_q(best_q, n_best)
                if ni == 0:
                    break

    # Phase 7: Random restarts - perturb and CD
    for restart in range(5):
        if remaining() < 40:
            break
        noise = rng.randn(m_best) * 0.05
        q_r = np.clip(best_q + noise, 0.0, 1.0)
        h_r = build_h_from_q(q_r, n_best)
        c5_r = np.correlate(h_r, 1.0 - h_r, mode='full').max() * dx_best
        corr_r = np.correlate(h_r, 1.0 - h_r, mode='full')
        for sweep in range(40):
            if remaining() < 20:
                break
            q_r, corr_r, c5_r, n_imp_r = cd_sweep(q_r, n_best, corr_r, dx_best, rng, n_grid=200)
            if c5_r < best_c5 - 1e-14:
                best_c5 = c5_r
                best_h = build_h_from_q(q_r, n_best)
                best_q = q_r.copy()
            if n_imp_r == 0:
                break

    # Phase 8: Final CD polish with high resolution grid
    if remaining() > 15:
        corr_fin = np.correlate(best_h, 1.0 - best_h, mode='full')
        q_fin = best_q.copy()
        for ng_f in [500, 800]:
            if remaining() < 10:
                break
            for sweep in range(100):
                if remaining() < 5:
                    break
                q_fin, corr_fin, c5_f, n_if = cd_sweep(q_fin, n_best, corr_fin, dx_best, rng, n_grid=ng_f)
                if c5_f < best_c5 - 1e-14:
                    best_c5 = c5_f
                    best_h = build_h_from_q(q_fin, n_best)
                    best_q = q_fin.copy()
                if n_if == 0:
                    break

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    n_out = len(best_h)
    m_out = n_out // 2
    best_h = build_h_from_q(np.clip(best_h[:m_out], 0, 1), n_out)
    final_score = np.correlate(best_h, 1.0 - best_h, mode='full').max() * (2.0 / n_out)
    if final_score < best_c5:
        best_c5 = final_score

    return best_h, best_c5, n_out
