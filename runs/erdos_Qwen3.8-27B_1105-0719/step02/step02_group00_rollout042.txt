from scipy.optimize import minimize
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx] + 1e-15)
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx] + 1e-15)
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def make_symmetric(h):
    """Make h symmetric: h[i] + h[i+n//2] = 1"""
    n = len(h)
    if n % 2 != 0:
        return None
    half = n // 2
    x = (h[:half] + 1.0 - h[half:]) / 2.0
    x = np.clip(x, 0.0, 1.0)
    result = np.empty(n)
    result[:half] = x
    result[half:] = 1.0 - x
    return result

def profile_from_x(x):
    """Compute profile from symmetric representation x (length n/2)"""
    n = 2 * len(x)
    h = np.empty(n)
    h[:len(x)] = x
    h[len(x):] = 1.0 - x
    return compute_profile(h)

def profile_grad_from_x(x, rows):
    """Compute gradient of profile rows w.r.t. x (symmetric reduced space)"""
    n = 2 * len(x)
    half = len(x)
    dx = 2.0 / n
    h = np.empty(n)
    h[:half] = x
    h[half:] = 1.0 - x
    
    # Build gradient w.r.t. h first
    G = np.zeros((len(rows), n))
    for m in rows:
        d = m - (n - 1)
        if d == 0:
            G[m, :] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            for i in range(d, n):
                G[m, i] += dx * (1.0 - h[i-d])
                G[m, i-d] -= dx * h[i]
        else:
            dp = -d
            for i in range(n - dp):
                G[m, i] += dx * (1.0 - h[i+dp])
                G[m, i+dp] -= dx * h[i]
    
    # Reduce: dP/dx[i] = dP/dh[i] - dP/dh[i+half]
    Gx = G[:, :half] - G[:, half:]
    return Gx

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx] + 1e-15)
            remaining -= add
            h[idx] += add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx] + 1e-15)
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break

        idx = int(np.argmax(1.0 - h)) if diff2 > 0 else int(np.argmin(h))
        h[idx] += diff2

        profile = profile_from_x(x)
        c = float(np.max(profile))
        
        if c < best_c:
            best_c = c
        
        # Active set: top K profile values
        sorted_idx = np.argsort(profile)[::-1]
        K = min(10, len(sorted_idx))
        # Include all within threshold
        thresh = c - max(0.003, 3 * (c - np.median(profile)))
        active = np.where(profile >= thresh)[0]
        if len(active) > K:
            active = sorted_idx[:K]
        active = np.sort(active)
        
        # Compute gradient
        Gx = profile_grad_rows(x, active)
        # Weight by how close to max
        w = profile[active] - (thresh if len(active) > 1 else c - 0.01)
        w = np.maximum(w, 0.01)
        w = w / w.sum()
        grad = Gx.T @ w
        grad -= grad.mean()  # Remove mean (invariance under constant shift)
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad /= gnorm
        
        velocity = momentum * velocity + grad
        vnorm = np.linalg.norm(velocity)
        if vnorm > 0:
            velocity /= vnorm
        
        # Line search
        improved = False
        for a in [alpha, alpha*1.5, alpha*0.5, alpha*0.25, alpha*0.1]:
            x_try = np.clip(x - a * velocity, 0, 1)

            if c_try < best_c - 1e-13:

                best_c = c_try
                improved = True
                break
            elif c_try < c:
                x = x_try
                improved = True
                break
        
        if not improved:
            alpha *= 0.7
            if alpha < 1e-7:
                break
    
    return x, best_c

    half = len(x0)
    n = 2 * half
    dx = 2.0 / n
    
    def objective(x):

        return float(np.max(np.correlate(h, 1-h, mode='full') * dx))
    
    # Use SLSQP with smooth reformulation
    # Actually, let's use the epigraph form
    M = 2 * n - 1
    
    def epigraph_obj(v):
        return v[-1]
    
    def epigraph_grad(v):
        g = np.zeros(half + 1)
        g[-1] = 1.0
        return g
    
    def epigraph_ineq(v):
        x = v[:half]
        t = v[-1]
        h = h_from_x(x)
        profile = np.correlate(h, 1-h, mode='full') * dx
        return t - profile
    
    def epigraph_ineq_jac(v):
        x = v[:half]
        h = h_from_x(x)
        G_full = np.zeros((M, n))
        for m in range(M):
            d = m - (n-1)
            if d == 0:
                G_full[m, :] = dx * (1 - 2*h)
            elif d > 0:
                i = np.arange(n-d)
                G_full[m, i+d] += dx * (1-h[i])

                p = -d
                j = np.arange(n-p)
                G_full[m, j] += dx * (1-h[j+p])
                G_full[m, j+p] -= dx * h[j]
        Gx = G_full[:, :half] - G_full[:, half:]
        J = np.zeros((M, half+1))
        J[:, :half] = -Gx
        J[:, half] = 1.0
        return J
    
    c0 = objective(x0)
    v0 = np.concatenate([x0, [c0 + 1e-4]])
    bounds = [(0,1)]*half + [(0, c0+1)]
    cons = [{'type': 'ineq', 'fun': epigraph_ineq, 'jac': epigraph_ineq_jac}]
    
    try:
        res = minimize(epigraph_obj, v0, method='SLSQP', jac=epigraph_grad,
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.success or res.x is not None:
            x_new = np.clip(res.x[:half], 0, 1)

def exact_score(h):

    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def compute_profile(h):
    n = len(h)

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            add = min(remaining, capacity[order[i]] + 1e-15)
            h[order[i]] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            rem = min(remaining, h[order[i]] + 1e-15)
            h[order[i]] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        idx = int(np.argmax(1.0 - h)) if diff2 > 0 else int(np.argmin(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def h_from_x(x):

    x = (h[:half] + 1.0 - h[half:2*half]) / 2.0

def profile_grad_rows(x, rows):
    n = 2 * len(x)
    half = len(x)
    dx = 2.0 / n
    h = h_from_x(x)
    Gfull = np.zeros((len(rows), n))
    for idx, m in enumerate(rows):
        d = m - (n - 1)
        if d == 0:
            Gfull[idx, :] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            i = np.arange(n - d)
            Gfull[idx, i + d] += dx * (1.0 - h[i])
            Gfull[idx, i] -= dx * h[i + d]
        else:
            p = -d
            j = np.arange(n - p)
            Gfull[idx, j] += dx * (1.0 - h[j + p])
            Gfull[idx, j + p] -= dx * h[j]
    return Gfull[:, :half] - Gfull[:, half:]

                m2 = hi - (hi - lo) / 3.0
                x[i] = m1
                v1 = float(np.max(profile_from_x(x)))
                x[i] = m2
                v2 = float(np.max(profile_from_x(x)))

def slsqp_sym(x0, maxiter=300, deadline=None):
    half = len(x0)
    n = 2 * half
    dx = 2.0 / n
    c0 = float(np.max(profile_from_x(x0)))

    def obj(v):
        return v[-1]

    def obj_jac(v):
        g = np.zeros(half + 1)
        g[-1] = 1.0
        return g

    def ineq_fun(v):
        x = v[:half]
        t = v[-1]
        prof = profile_from_x(x)
        return t - prof

    def ineq_jac(v):
        x = v[:half]
        M = 2 * n - 1
        rows = np.arange(M)
        Gx = profile_grad_rows(x, rows)
        J = np.zeros((M, half + 1))
        J[:, :half] = -Gx
        J[:, half] = 1.0
        return J

    v0 = np.concatenate([x0, [c0 + 1e-4]])
    bounds = [(0.0, 1.0)] * half + [(0.0, c0 + 1.0)]
    cons = [{'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}]
    try:
        res = minimize(obj, v0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None and np.isfinite(res.x).all():
            x_new = np.clip(res.x[:half], 0.0, 1.0)
            c_new = float(np.max(profile_from_x(x_new)))
            if c_new <= c0 + 1e-8:
                return x_new, c_new
    except Exception:
        pass
    return None, None

def interpolate_mass(h_old, new_n):
    n_old = len(h_old)
    if n_old == new_n:
        return h_old.copy()
    edges_old = np.linspace(0, 2.0, n_old + 1)
    edges_new = np.linspace(0, 2.0, new_n + 1)
    h_new = np.zeros(new_n)
    for k in range(new_n):
        lo, hi = edges_new[k], edges_new[k + 1]
        total = 0.0
        width = hi - lo
        if width <= 0:
            continue
        j_start = int(np.searchsorted(edges_old, lo, side='right')) - 1
        j_end = int(np.searchsorted(edges_old, hi, side='left')) - 1
        for j in range(max(0, j_start), min(n_old, j_end + 1)):
            overlap_lo = max(lo, edges_old[j])
            overlap_hi = min(hi, edges_old[j + 1])
            if overlap_hi > overlap_lo:
                total += h_old[j] * (overlap_hi - overlap_lo)
        h_new[k] = total / width
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

def logsumexp_descent(x, tau=50.0, max_iters=200, deadline=None):
    """Gradient descent using log-sum-exp smoothing of the max."""
    half = len(x)
    x = x.copy()
    best_c = float(np.max(profile_from_x(x)))
    alpha = 0.01
    for it in range(max_iters):
        if deadline is not None and time.perf_counter() > deadline:
            break
        profile = profile_from_x(x)
        c = float(np.max(profile))
        if c < best_c:
            best_c = c
        # Softmax weights
        shifted = profile - np.max(profile)
        weights = np.exp(tau * shifted)
        weights /= weights.sum()
        rows = np.arange(len(profile))
        Gx = profile_grad_rows(x, rows)
        grad = Gx.T @ weights
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad /= gnorm
        improved = False
        for a in [alpha, alpha * 0.5, alpha * 0.25, alpha * 0.1]:
            x_try = np.clip(x - a * grad, 0.0, 1.0)
            c_try = float(np.max(profile_from_x(x_try)))
            if c_try < best_c - 1e-13:
                x = x_try
                best_c = c_try
                improved = True
                break
        if not improved:
            alpha *= 0.6
            tau = max(tau * 0.95, 10.0)
            if alpha < 1e-7:
                break
    return x, best_c
