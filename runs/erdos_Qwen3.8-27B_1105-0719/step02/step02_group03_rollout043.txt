        f = 1.0
        r = 0.0
        i = k
        while i > 0:
            f /= base
            r += f * (i % base)
            i //= base

        h[pos] = 1.0

    if ones < target:

        h[zeros[:target - ones]] = 1.0

        h[ones_idx[-(ones - target):]] = 0.0

Actually, I realize I can compute all deltas more efficiently:

For each one-position i, the "removal" part of the delta is:
removal[i, m] = -(1-h[i-m]) [if valid] + h[i+m] [if valid]

For each zero-position j, the "addition" part is:
addition[j, m] = (1-h[j-m]) [if valid] - h[j+m] [if valid]

Then delta[i,j,m] = removal[i,m] + addition[j,m]

            new_max = (corrs + delta).max()
            if new_max < best_max - 1e-10:
                h[i1] = 0.0
                h[j1] = 1.0
                h[i2] = 0.0
                h[j2] = 1.0
                corrs = compute_corrs(h)
                current_max = corrs.max()
                if current_max < best_max:
                    best_max = current_max
                    best_h = h.copy()
                    found_improve = True
                break
        if not found_improve and rnd > 20:
            # Try a single random perturbation to escape
            i = int(ones[rng.randint(n_ones)])
            j = int(zeros[rng.randint(n_zeros)])
            h[i] = 0.0
            h[j] = 1.0
            corrs = compute_corrs(h)
            current_max = corrs.max()
    return best_h, corrs


def slsqp_polish(h, dx, target_sum, maxiter=800):
    n = len(h)

    def smooth_max(x, beta):
        corrs = np.correlate(x, 1.0 - x, mode='full')
        m = corrs.max()
        return (m + np.log(np.sum(np.exp(beta * (corrs - m)))) / beta) * dx

    def smooth_max_grad(x, beta):
        corrs = np.correlate(x, 1.0 - x, mode='full')
        m = corrs.max()
        w = np.exp(beta * (corrs - m))
        w /= w.sum()
        grad = np.zeros(n)
        ms = np.arange(-(n - 1), n)
        for k in range(len(corrs)):
            if w[k] < 1e-14:
                continue
            offset = k - (n - 1)
            i_lo = max(0, -offset)
            i_hi = min(n - 1, n - 1 - offset)
            if i_lo > i_hi:
                continue
            idx = np.arange(i_lo, i_hi + 1)
            grad[idx + offset] += w[k] * (1.0 - x[idx])
            grad[idx] += w[k] * (-x[idx + offset])
        return grad * dx

    best_h = h.copy()
    best_val = c5_of_h(best_h, dx)
    for beta in [100, 500]:
        try:
            res = minimize(
                lambda x: smooth_max(x, beta), h,
                jac=lambda x: smooth_max_grad(x, beta),
                method='SLSQP',
                bounds=[(0.0, 1.0)] * n,
                constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - target_sum,
                              'jac': lambda x: np.ones(n)}],
                options={'maxiter': maxiter, 'ftol': 1e-14}
            )
            h_out = project_feasible(res.x, target_sum)
            val = c5_of_h(h_out, dx)
            if val < best_val:
                best_val = val
                best_h = h_out
        except Exception:
            pass
    return best_h, best_val


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    def remaining():
        return budget_s - (time.time() - t0)

    # Phase 0: Use initial_h_values if available
    try:
        h_init = np.array(initial_h_values, dtype=float)
        n0 = len(h_init)
        dx0 = 2.0 / n0
        ts0 = n0 / 2.0
        h_init = project_feasible(h_init, ts0)
        c5_init = c5_of_h(h_init, dx0)
        best_c5 = c5_init
        best_h = h_init.copy()
        best_n = n0
    except NameError:
        pass

    # Phase 1: Try multiple n values with structured starts
    n_candidates = [128, 160, 200]
    ms_cache = {}

    # Also try to interpolate initial to n=128
    h_128_candidates = []
    try:
        h_init_arr = np.array(initial_h_values, dtype=float)
        x_orig = np.linspace(0.0, 2.0, len(h_init_arr))
        x_128 = np.linspace(0.0, 2.0, 128)
        h_interp = np.interp(x_128, x_orig, h_init_arr)
        h_interp = project_feasible(h_interp, 64.0)
        h_128_candidates.append(h_interp)
    except (NameError, Exception):
        pass

    for n in n_candidates:
        if remaining() < 90:
            break
        dx = 2.0 / n
        target = n // 2
        ms = np.arange(-(n - 1), n)
        ms_cache[n] = ms

        starts = []
        if n == 128:
            starts.extend(h_128_candidates)
        starts.append(golden_ratio_start(n, target))
        starts.append(halton_start(n, target, 2))
        starts.append(halton_start(n, target, 3))
        starts.append(random_start(n, target, rng))

        for h0 in starts:
            if remaining() < 60:
                break
            h0 = np.clip(h0, 0.0, 1.0)
            if abs(h0.sum() - target) > 1:
                h0 = project_feasible(h0, float(target))
            # Project to binary
            h_bin = np.zeros(n)
            sorted_idx = np.argsort(-h0)
            h_bin[sorted_idx[:target]] = 1.0

            # Steepest descent
            time_for_sd = min(90, remaining() * 0.3)
            h_opt, c5_val = steepest_descent(h_bin, dx, ms, max_sweeps=3000, time_limit=time_for_sd)

            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h_opt.copy()
                best_n = n

            # SA from the result
            if remaining() > 120:
                time_for_sa = min(90, remaining() * 0.3)
                h_sa, c5_sa = sa_binary(h_opt, dx, ms, rng,
                                        max_iters=200000, time_limit=time_for_sa, patience=6000)
                if c5_sa < best_c5:
                    best_c5 = c5_sa
                    best_h = h_sa.copy()
                    best_n = n

            # Multi-swap
            if remaining() > 90:
                corrs = compute_corrs(best_h)
                time_for_ms = min(60, remaining() * 0.25)
                h_ms, corrs_ms = multi_swap_search(h_sa, dx, ms, compute_corrs(h_sa),
                                                   rng, max_rounds=80, time_limit=time_for_ms)
                c5_ms = corrs_ms.max() * dx
                if c5_ms < best_c5:
                    best_c5 = c5_ms
                    best_h = h_ms.copy()
                    best_n = n

    # Phase 2: Iterated local search on the best solution found
    if remaining() > 150:
        n_best = best_n
        dx_best = 2.0 / n_best
        ms_best = ms_cache.get(n_best, np.arange(-(n_best - 1), n_best))

        for iteration in range(5):
            if remaining() < 60:
                break
            # Perturb: flip 3-5 random pairs
            h_pert = best_h.copy()
            h_pert_corrs = compute_corrs(h_pert)
            h_pert_max = h_pert_corrs.max()
            n_flips = rng.randint(3, 6)
            ones = np.where(h_pert == 1.0)[0]
            zeros = np.where(h_pert == 0.0)[0]
            for _ in range(n_flips):
                i = int(ones[rng.randint(len(ones))])
                j = int(zeros[rng.randint(len(zeros))])
                if i != j:
                    h_pert[i], h_pert[j] = 0.0, 1.0

            # Local search from perturbed
            time_for_ls = min(80, remaining() * 0.3)
            h_opt, c5_val = steepest_descent(h_pert, dx_best, ms_best,
                                             max_sweeps=3000, time_limit=time_for_ls)
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h_opt.copy()

            # SA from perturbed
            if remaining() > 80:
                time_for_sa = min(60, remaining() * 0.25)
                h_sa, c5_sa = sa_binary(h_pert, dx_best, ms_best, rng,
                                        max_iters=150000, time_limit=time_for_sa, patience=5000)
                if c5_sa < best_c5:
                    best_c5 = c5_sa
                    best_h = h_sa.copy()

    # Phase 3: SLSQP polish on continuous relaxation
    if remaining() > 60:
        n_pol = best_n
        dx_pol = 2.0 / n_pol
        ts_pol = n_pol / 2.0
        h_cont = best_h.copy()
        h_pol, c5_pol = slsqp_polish(h_cont, dx_pol, ts_pol, maxiter=600)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

    # Phase 4: Final binary local search
    if remaining() > 40:
        n_final = best_n
        dx_final = 2.0 / n_final
        ms_final = ms_cache.get(n_final, np.arange(-(n_final - 1), n_final))
        h_bin_final = np.zeros(n_final)
        sorted_idx = np.argsort(-best_h)
        h_bin_final[sorted_idx[:n_final // 2]] = 1.0
        time_fin = min(50, remaining() * 0.4)
        h_fin, c5_fin = steepest_descent(h_bin_final, dx_final, ms_final,
                                         max_sweeps=5000, time_limit=time_fin)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    # Final projection and evaluation
    n_b = len(best_h)
    dx_b = 2.0 / n_b
    ts_b = n_b / 2.0
    best_h = project_feasible(best_h, ts_b)
    best_c5 = c5_of_h(best_h, dx_b)

    return best_h, best_c5, len(best_h)
