    except Exception:
        return None, None


def projected_gradient(q0, n, beta, n_iter=2000, lr_init=0.1):
    m = n // 2
    q = q0.copy()
    lr = lr_init
    for it in range(n_iter):
        g = sym_grad_lse(q, n, beta)
        q_new = np.clip(q - lr * g, 0.0, 1.0)
        f_new = sym_obj_lse(q_new, n, beta)
        f_old = sym_obj_lse(q, n, beta)
        if f_new > f_old + 1e-15:
            lr *= 0.5
            q_new = np.clip(q - lr * g, 0.0, 1.0)
            f_new = sym_obj_lse(q_new, n, beta)
            if f_new > f_old + 1e-15:
                continue
        else:
            if (it + 1) % 100 == 0 and lr < 0.5:
                lr *= 1.1
        q = q_new
    c5 = exact_score(make_sym_h(q, n))
    return q, c5


def coordinate_descent(q0, n, beta, n_sweeps=10):
    m = n // 2
    q = q0.copy()
    best_f = sym_obj_lse(q, n, beta)
    for sweep in range(n_sweps):
        improved = False
        for i in range(m):
            q_old = q[i]
            q[i] = np.clip(q[i] - 0.01 * sym_grad_lse(q, n, beta)[i], 0.0, 1.0)
            f_new = sym_obj_lse(q, n, beta)
            if f_new < best_f - 1e-16:
                best_f = f_new
                improved = True
            else:
                q[i] = q_old
        if not improved:
            break
    c5 = exact_score(make_sym_h(q, n))
    return q, c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def elapsed():
        return time.time() - t0

    def remaining():
        return budget_s - elapsed()

    try:
        h_parent = np.array(initial_h_values, dtype=float)
    except NameError:
        h_parent = 0.5 * np.ones(126)

    n = len(h_parent)
    if n % 2 != 0:
        n -= 1
        h_parent = h_parent[:n]

    m = n // 2
    q_best = make_q_from_h(h_parent, n)
    h_best = make_sym_h(q_best, n)
    c5_best = exact_score(h_best)
    print(f"Initial: n={n}, C5={c5_best:.7f}")

    # Phase 1: L-BFGS-B with increasing beta (main optimization)
    betas = [30, 60, 100, 200, 400, 800, 1200, 1600]
    for beta in betas:
        if remaining() < 80:
            break
        q_new, c5_new = lse_lbfgsb(q_best, n, beta, maxiter=600)
        if q_new is not None and c5_new < c5_best - 1e-12:
            c5_best = c5_new
            q_best = q_new
            h_best = make_sym_h(q_best, n)
            print(f"  LSE beta={beta}: C5={c5_best:.7f}")

    # Phase 2: Top-k focused optimization
    for top_k in [20, 40, 60, 80, 100]:
        if remaining() < 60:
            break
        for beta in [400, 800, 1200]:
            if remaining() < 30:
                break
            q_new, c5_new = topk_lbfgsb(q_best, n, beta, top_k, maxiter=400)
            if q_new is not None and c5_new < c5_best - 1e-12:
                c5_best = c5_new
                q_best = q_new
                h_best = make_sym_h(q_best, n)
                print(f"  TopK k={top_k} beta={beta}: C5={c5_best:.7f}")

    # Phase 3: Projected gradient refinement
    if remaining() > 40:
        for beta in [800, 1500]:
            if remaining() < 20:
                break
            q_new, c5_new = projected_gradient(q_best, n, beta, n_iter=1500, lr_init=0.05)
            if c5_new < c5_best - 1e-12:
                c5_best = c5_new
                q_best = q_new
                h_best = make_sym_h(q_best, n)
                print(f"  ProjGrad beta={beta}: C5={c5_best:.7f}")
                # Follow up with L-BFGS-B
                if remaining() > 30:
                    q_new2, c5_new2 = lse_lbfgsb(q_best, n, min(beta, 1000), maxiter=300)
                    if q_new2 is not None and c5_new2 < c5_best - 1e-12:
                        c5_best = c5_new2
                        q_best = q_new2
                        h_best = make_sym_h(q_best, n)
                        print(f"  ProjGrad+LSE: C5={c5_best:.7f}")

    # Phase 4: Multi-start with varied perturbations
    n_trials = 30
    for trial in range(n_trials):
        if remaining() < 25:
            break
        scale = 0.05 + 0.6 * (trial / max(n_trials, 1))
        q_pert = q_best + scale * rng.randn(m)
        q_pert = np.clip(q_pert, 0.0, 1.0)

        q_new, c5_new = lse_lbfgsb(q_pert, n, 200, maxiter=300)
        if q_new is not None and c5_new < c5_best + 0.003:
            q_new, c5_new = lse_lbfgsb(q_new, n, 600, maxiter=400)
            if q_new is not None and c5_new < c5_best - 1e-12:
                c5_best = c5_new
                q_best = q_new
                h_best = make_sym_h(q_best, n)
                print(f"  MultiStart {trial} (s={scale:.2f}): C5={c5_best:.7f}")
                if remaining() > 20:
                    q_new2, c5_new2 = topk_lbfgsb(q_best, n, 1000, 40, maxiter=300)
                    if q_new2 is not None and c5_new2 < c5_best - 1e-12:
                        c5_best = c5_new2
                        q_best = q_new2
                        h_best = make_sym_h(q_best, n)
                        print(f"  MultiStart {trial}+TopK: C5={c5_best:.7f}")

    # Phase 5: Try different grid sizes
    if remaining() > 60:
        for delta in [-4, -2, 2, 4, 6, 8, -6, -8, 10, 12]:
            if remaining() < 30:
                break
            n_try = n + delta
            if n_try < 40 or n_try % 2 != 0:
                continue
            x_old = np.linspace(0, 2, n, endpoint=False)
            x_new_grid = np.linspace(0, 2, n_try, endpoint=False)
            h_interp = np.interp(x_new_grid, x_old, h_best)
            h_interp = project_feasible(h_interp, n_try / 2.0)
            m_try = n_try // 2
            q_try = make_q_from_h(h_interp, n_try)

            q_new, c5_new = lse_lbfgsb(q_try, n_try, 200, maxiter=300)
            if q_new is not None and remaining() > 15:
                q_new, c5_new = lse_lbfgsb(q_new, n_try, 600, maxiter=300)
            if q_new is not None and c5_new < c5_best - 1e-9:
                c5_best = c5_new
                n = n_try
                m = m_try
                q_best = q_new
                h_best = make_sym_h(q_best, n)
                print(f"  Grid n={n}: C5={c5_best:.7f}")
                if remaining() > 30:
                    q_new2, c5_new2 = topk_lbfgsb(q_best, n, 1000, 40, maxiter=300)
                    if q_new2 is not None and c5_new2 < c5_best - 1e-12:
                        c5_best = c5_new2
                        q_best = q_new2
                        h_best = make_sym_h(q_best, n)
                        print(f"  Grid n={n}+TopK: C5={c5_best:.7f}")

    # Phase 6: Final high-beta polish
    if remaining() > 30:
        for beta in [1000, 1500, 2000, 3000]:
            if remaining() < 15:
                break
            q_new, c5_new = lse_lbfgsb(q_best, n, beta, maxiter=500)
            if q_new is not None and c5_new < c5_best - 1e-13:
                c5_best = c5_new
                q_best = q_new
                h_best = make_sym_h(q_best, n)
                print(f"  FinalPolish beta={beta}: C5={c5_best:.7f}")

    # Phase 7: Subgradient refinement on exact max
    if remaining() > 20:
        q = q_best.copy()
        m = n // 2
        for it in range(3000):
            h = make_sym_h(q, n)
            corrs = corr_vals(h)
            k_star = np.argmax(corrs)
            d = k_star - (n - 1)
            g_full = _laged_grad(h, n, d)
            g = g_full[:m] - g_full[n - 1:n - 1 - m:-1]
            lr = 0.001 / (1.0 + it * 0.01)
            q_new = np.clip(q - lr * g, 0.0, 1.0)
            c5_new = exact_score(make_sym_h(q_new, n))
            if c5_new < c5_best - 1e-14:
                c5_best = c5_new
                q_best = q_new
                h_best = make_sym_h(q_best, n)
                q = q_new
        if c5_best < exact_score(make_sym_h(q_best, n)) + 1e-12:
            print(f"  SubGrad final: C5={c5_best:.7f}")

    # Phase 8: Another round of L-BFGS-B if time permits
    if remaining() > 25:
        q_new, c5_new = lse_lbfgsb(q_best, n, 2000, maxiter=800)
        if q_new is not None and c5_new < c5_best - 1e-13:
            c5_best = c5_new
            q_best = q_new
            h_best = make_sym_h(q_best, n)
            print(f"  FinalRound: C5={c5_best:.7f}")

    # Validation
    h_best = np.clip(h_best, 0.0, 1.0)
    if n % 2 == 0:
        h_best = make_sym_h(make_q_from_h(h_best, n), n)
    else:
        h_best = project_feasible(h_best, n / 2.0)
    c5_best = exact_score(h_best)

    return h_best, c5_best, len(h_best)
