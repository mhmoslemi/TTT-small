import numpy as np
from scipy.optimize import linprog
from scipy.special import logsumexp
import time

def make_sym_h(q, n):
    m = n // 2
    q = np.clip(q, 0.0, 1.0)
    h = np.empty(n)
    h[:m] = q
    h[m:] = 1.0 - q[::-1]
    return h

def make_q_from_h(h, n):
    m = n // 2
    q = 0.5 * (h[:m] + 1.0 - h[n-1:n-1-m:-1])
    return np.clip(q, 0.0, 1.0)

def corr_vals(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def exact_score(h):
    return corr_vals(h).max()

def coord_desc_sweep(q, n, max_sweeps=100, rng=None, verbose=False):
    """Coordinate descent on q in symmetric space."""
    m = n // 2
    dx = 2.0 / n
    best_score = exact_score(make_sym_h(q, n))
    q_cur = q.copy()
    
    for sweep in range(max_sweeps):
        if rng is not None:
            order = rng.permutation(m)
        else:
            order = range(m)
        
        improved = False
        for i in order:
            # Compute corr at q[i]=0 and q[i]=1
            q0 = q_cur.copy()
            q0[i] = 0.0
            q1 = q_cur.copy()
            q1[i] = 1.0
            h0 = make_sym_h(q0, n)
            h1 = make_sym_h(q1, n)
            c0 = corr_vals(h0)
            c1 = corr_vals(h1)
            
            # Find x in [0,1] minimizing max(j)(c0[j] + x*(c1[j]-c0[j]))
            a = c0
            b = c1 - c0
            # Fine grid + refinement
            xs = np.linspace(0, 1, 32)
            vals = np.max(a + b * xs[:, None], axis=1)
            bi = np.argmin(vals)
            if bi == 0:
                x_star = 0.0
            elif bi == len(xs) - 1:
                x_star = 1.0
            else:
                # Refine
                xs_f = np.linspace(xs[bi-1], xs[bi+1], 64)
                vals_f = np.max(a + b * xs_f[:, None], axis=1)
                x_star = xs_f[np.argmin(vals_f)]
            
            if abs(x_star - q_cur[i]) > 1e-10:
                q_cur[i] = x_star
                improved = True
        
        new_score = exact_score(make_sym_h(q_cur, n))
        if new_score < best_score - 1e-14:
            best_score = new_score
        if not improved and sweep > 3:
            break
        if verbose and sweep % 20 == 0:
            print(f"  sweep {sweep}: score={new_score:.8f}")
    
    return q_cur, best_score

def lp_step(q, n, active_lags, t_init):
    """Solve LP with linearized correlations."""
    m = n // 2
    dx = 2.0 / n
    h = make_sym_h(q, n)
    
    # Build LP: min t
    # s.t. dx * sum_i a_{k,i} * h_q[i] <= t for each active lag
    #      0 <= q[i] <= 1
    # Variables: [t, q[0], ..., q[m-1]]
    
    c = np.zeros(m + 1)
    c[0] = 1.0
    
    A_ub = []
    b_ub = []
    
    for lag in active_lags:
        d = lag - (n - 1)
        i_lo = max(0, -d)
        i_hi = min(n - 1, n - 1 - d)
        
        row = np.zeros(m + 1)
        const = 0.0
        
        for i in range(i_lo, i_hi + 1):
            a = 1.0 - h[i + d]  # fixed coefficient
            
            if i < m:
                # h[i] = q[i]
                row[1 + i] += dx * a
            else:
                # h[i] = 1 - q[n-1-i]
                q_idx = n - 1 - i
                row[1 + q_idx] -= dx * a
                const += dx * a
        
        # row[0] * t + row[1:] @ q >= const  →  -row[0]*t - row[1:]@q <= -const
        # i.e., t - dx*sum(a_i * h_q[i]) >= 0
        # row[0] = 1 (for t), row[1+i] = coefficient of q[i]
        row[0] = -1.0  # -t + dx*sum(...) <= 0 → t >= dx*sum(...)
        A_ub.append(row.copy())
        b_ub.append(0.0)
    
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    bounds = [(0, None)] + [(0, 1)] * m
    q0_lp = np.concatenate([[t_init], q])
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'maxiter': 5000})
    
    if res.status == 0:
        q_new = res.x[1:]
        q_new = np.clip(q_new, 0, 1)
        h_new = make_sym_h(q_new, n)
        # Check feasibility
        if abs(h_new.sum() - n/2.0) < 1e-8:
            return q_new
    return None
