from scipy.special import logsumexp

def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(80):

        h = np.clip(x - mid, 0.0, 1.0)

            lo = mid
        else:

        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def c5_of_h(h, dx):
    return compute_corrs(h).max() * dx

def grad_corr_k(h, k, n):
    offset = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -offset)
    i_hi = min(n - 1, n - 1 - offset)
    if i_lo > i_hi:

    grad[idx + offset] += (1.0 - h[idx])
    grad[idx] += (-h[idx + offset])
    return grad

    n = len(h0)

    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    no_improve = 0
    
    for it in range(max_iter):
        corrs = compute_corrs(h)
        max_c = corrs.max()
        c5_val = max_c * dx
        
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 200:
                break
        
        # Identify active set
        n_active = max(5, int(n_active_frac * (2*n - 1)))
        top_idx = np.argsort(corrs)[-n_active:]

        # s.t. corrs[k] + sum_i grad_k[i] * delta_i <= t for k in active
        #      sum delta_i = 0
        #      -h[i]*step <= delta[i] <= (1-h[i])*step
        #      t >= 0
        n_vars = n + 1  # n deltas + 1 t
        c_obj = np.zeros(n_vars)
        c_obj[-1] = 1.0  # minimize t
        
        A_ub = []
        b_ub = []
        for k in top_idx:
            row = np.zeros(n_vars)
            g = grad_corr_k(h, k, n)
            row[:n] = g
            row[n] = -1.0
            A_ub.append(row)
            b_ub.append(-corrs[k])
        
        A_ub = np.array(A_ub)
        b_ub = np.array(b_ub)
        
        A_eq = np.zeros((1, n_vars))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)
        
        bounds = []
        for i in range(n):
            bounds.append((-h[i] * step_size, (1.0 - h[i]) * step_size))
        bounds.append((0.0, None))
        
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs')
            if res.status != 0:
                step_size *= 0.5
                continue
            delta = res.x[:n]
        except Exception:
            break
        
        # Take the step (full step or fraction)
        for frac in [1.0, 0.5, 0.25, 0.1]:

            h_new = project_feasible(h_new, target_sum)
            new_c5 = c5_of_h(h_new, dx)
            if new_c5 < c5_val - 1e-14:
                h = h_new
                break
        else:
            step_size *= 0.7
            if step_size < 1e-6:
                break
    
    return best_h, best_c5

```python
def line_search_quadratic(h, delta, corrs, n, target_sum):
    """Find optimal t in h + t*delta using quadratic model of correlations."""
    n = len(h)
    
    # For each k, compute a_k, b_k, c_k such that corr_k(t) = c_k + b_k*t + a_k*t^2
    # c_k = corrs[k]
    # b_k = grad_corr_k(h, k, n) @ delta
    # a_k = -sum over valid i of delta[i] * delta[i+offset_k]
    
    a = np.zeros(2*n-1)
    b = np.zeros(2*n-1)
    
    for k in range(2*n-1):
        offset = k - (n-1)
        i_lo = max(0, -offset)
        i_hi = min(n-1, n-1-offset)
        if i_lo > i_hi:
            a[k] = 0
            b[k] = 0
            continue
        idx = np.arange(i_lo, i_hi+1)
        b[k] = np.sum(delta[idx] * (1-h[idx+offset]) - h[idx] * delta[idx+offset])
        a[k] = -np.sum(delta[idx] * delta[idx+offset])
    
    c = corrs  # already computed
    
    # Now minimize f(t) = max_k (c_k + b_k*t + a_k*t^2)
    # subject to h + t*delta being feasible: 0 ≤ h[i] + t*delta[i] ≤ 1
    
    # Find feasible range of t
    t_lo, t_hi = 0.0, 1.0
    for i in range(n):
        if delta[i] > 1e-15:
            t_hi = min(t_hi, (1.0 - h[i]) / delta[i])
        elif delta[i] < -1e-15:
            t_lo = max(t_lo, -h[i] / delta[i])
    
    if t_lo >= t_hi:
        return 0.0
    
    # Convex minimization on [t_lo, t_hi]
    # Use bisection on subgradient
    def f_and_grad(t):
        vals = c + b*t + a*t*t
        max_val = vals.max()
        # Subgradient: weighted average of (b_k + 2*a_k*t) for active k
        active = vals >= max_val - 1e-10
        if active.sum() == 0:
            active = np.zeros(len(vals), dtype=bool)
            active[vals.argmax()] = True
        w = np.zeros(len(vals))
        w[active] = 1.0 / active.sum()
        grad = np.sum(w * (b + 2*a*t))
        return max_val, grad
    
    # Bisection
    for _ in range(50):
        t_mid = (t_lo + t_hi) / 2
        _, g = f_and_grad(t_mid)
        if g > 0:
            t_hi = t_mid
        else:
            t_lo = t_mid
    
    t_opt = (t_lo + t_hi) / 2
    return t_opt

Wait, I need to be more careful. The bisection on the subgradient finds where the subgradient changes sign (from negative to positive), which is the minimum of the convex function. But the "subgradient" I'm computing is just one element of the subdifferential. I should use the minimum of the subdifferential (i.e., the minimum of b_k + 2*a_k*t over active k).

Actually for a max of functions, the subdifferential at t is the convex hull of {b_k + 2*a_k*t : k active}. The minimum is in the subdifferential iff min_k (b_k + 2*a_k*t) ≤ 0 ≤ max_k (b_k + 2*a_k*t) for active k.

So the bisection should be:
- Compute s(t) = min over active k of (b_k + 2*a_k*t)
- If s(t) > 0: all subgradients are positive, so increase... wait no.
- If max over active k of (b_k + 2*a_k*t) < 0: all subgradients are negative, minimum is to the right
- If min over active k < 0 < max over active k: we're at the minimum

Let me use:
- d(t) = max over active k of (b_k + 2*a_k*t)  (rightmost subgradient)
- If d(t) < 0: minimum is to the right, t_lo = t
- Else: minimum is to the left or at t, t_hi = t

Actually, the standard approach for minimizing a convex function f(t) with subgradient g(t):
- If g(t) > 0: minimum is to the left (t_hi = t)
- If g(t) < 0: minimum is to the right (t_lo = t)
- If g(t) = 0: we're at the minimum

For the max of quadratics, at any t, the subgradient set is {b_k + 2*a_k*t : k ∈ Active(t)}.
- If the maximum element of this set is < 0: all subgradients are negative → minimum is to the right
- If the minimum element is > 0: all subgradients are positive → minimum is to the left
- Otherwise: 0 is in the subdifferential → we're at the minimum

So:
```python
def bisect_convex(t_lo, t_hi, c, b, a, n_iter=50):
    for _ in range(n_iter):
        t_mid = (t_lo + t_hi) / 2
        vals = c + b*t_mid + a*t_mid*t_mid
        max_val = vals.max()
        active = vals >= max_val - 1e-8
        grads = b + 2*a*t_mid
        g_max = grads[active].max()
        g_min = grads[active].min()
        if g_min > 0:  # all positive, minimum to the left
            t_hi = t_mid
        elif g_max < 0:  # all negative, minimum to the right
            t_lo = t_mid
        else:
            break  # 0 is in subdifferential
    return (t_lo + t_hi) / 2
```

Wait, I need to be careful. The quadratic model is:
corr_k(h + t*delta) = ∑_i (h[i] + t*delta[i]) * (1 - h[i+offset] - t*delta[i+offset])

Let me expand:
= ∑_i [h[i](1-h[i+offset]) - t*h[i]*delta[i+offset] + t*delta[i](1-h[i+offset]) - t²*delta[i]*delta[i+offset]]
= corrs[k] + t*(∑ δ[i](1-h[i+offset]) - ∑ h[i]δ[i+offset]) - t²*∑ δ[i]δ[i+offset]

So: c_k = corrs[k], b_k = ∑_valid [δ[i](1-h[i+offset]) - h[i]δ[i+offset]], a_k = -∑_valid δ[i]δ[i+offset]

Note that a_k ≤ 0 is not guaranteed (it's -∑ δ[i]δ[i+offset] which can be positive or negative).

Actually wait: a_k = -∑ δ[i]*δ[i+offset]. This is negative of a sum of products. It can be positive (if the products are negative) or negative (if products are positive).

OK so the quadratics can open upward or downward. The max of quadratics is still convex (max of convex functions... wait, no! Max of convex functions is convex, but a quadratic opening downward is not convex).

Hmm, if a_k > 0 (opens upward), then c_k + b_k*t + a_k*t² is convex.
If a_k < 0 (opens downward), it's not convex.

So the max of these quadratics might not be convex! In that case, bisection on the subgradient won't work correctly.

Let me reconsider. The actual correlation function corr_k(h) is bilinear in h, so along a line h + t*delta, it IS quadratic in t. But the quadratic can open either way.

However, the max of a set of functions is always the upper envelope, and for our purposes, we just need to find the minimum of this upper envelope over a bounded interval. We can do this by:
1. Dense sampling + local refinement
2. Or: find all pairwise intersections and check the minimum at each interval's minimum point

For 2n-1 ≈ 251 quadratics, finding all pairwise intersections is O(251²) = 63k, which is fine. But it's complex to implement correctly.

A simpler approach: since t is in [0, 1] (or [0, t_max]) and we just need a good approximation, let me use a finer grid + golden section:

```python
def quadratic_line_search(h, delta, corrs, n):

        a[k] = -np.sum(delta[idx] * delta[idx + offset])

    t_max = 1.0
    for i in range(n):
        if delta[i] > 1e-15:
            t_max = min(t_max, (1.0 - h[i]) / delta[i])
    
    if t_max < 1e-12:
        return 0.0
    
    # Coarse grid to find approximate minimum
    ts = np.linspace(0, t_max, 200)
    vals = np.array([np.max(c + b*t + a*t*t) for t in ts])
    t_coarse = ts[vals.argmin()]
    
    # Refine with golden section in neighborhood
    t_lo = max(0, t_coarse - (t_max/100) * 2)
    t_hi = min(t_max, t_coarse + (t_max/100) * 2)
    
    gr = (np.sqrt(5) - 1) / 2
    a_g = t_lo + (1 - gr) * (t_hi - t_lo)
    b_g = t_lo + gr * (t_hi - t_lo)
    f_a = np.max(c + b*a_g + a*a_g*a_g)
    f_b = np.max(c + b*b_g + a*b_g*b_g)
    
    for _ in range(40):
        if f_a < f_b:
            t_hi = b_g
            b_g = a_g
            f_b = f_a
            a_g = t_lo + (1 - gr) * (t_hi - t_lo)
            f_a = np.max(c + b*a_g + a*a_g*a_g)
        else:
            t_lo = a_g
            a_g = b_g
            f_a = f_b
            b_g = t_lo + gr * (t_hi - t_lo)
            f_b = np.max(c + b*b_g + a*b_g*b_g)
    
    return (t_lo + t_hi) / 2

One more thought: for the LP, I should use a **simplifying trick**. Instead of the full LP with n+1 variables, I can use the **projected subgradient** approach:

1. Compute the subgradient of C₅ at h: it's a convex combination of ∇g_k for active k
2. Project this subgradient onto the constraint set {sum = target, 0 ≤ h ≤ 1}
3. Take a step in the negative projected subgradient direction
4. Line search

This avoids the LP entirely and is simpler. But it might be less effective than the LP.

Actually, let me combine both: use the LP for direction finding, and the quadratic line search for step size. This should be the most effective.

```python
import numpy as np

def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def c5_of_h(h, dx):
    return compute_corrs(h).max() * dx

def grad_corr_k(h, k, n):
    offset = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -offset)
    i_hi = min(n - 1, n - 1 - offset)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx + offset] += (1.0 - h[idx])
    grad[idx] += (-h[idx + offset])
    return grad

def quad_coefficients(h, delta, n):
    """Compute a, b such that corr_k(h + t*delta) = c + b_k*t + a_k*t^2"""
    n2 = 2 * n - 1
    a = np.zeros(n2)
    b = np.zeros(n2)
    for k in range(n2):
        offset = k - (n - 1)
        i_lo = max(0, -offset)
        i_hi = min(n - 1, n - 1 - offset)
        if i_lo > i_hi:
            continue
        idx = np.arange(i_lo, i_hi + 1)
        b[k] = np.sum(delta[idx] * (1.0 - h[idx + offset]) - h[idx] * delta[idx + offset])
        a[k] = -np.sum(delta[idx] * delta[idx + offset])
    return a, b

def quad_line_search(h, delta, corrs, n):
    """Find optimal t >= 0 minimizing max corr using quadratic model."""
    a, b = quad_coefficients(h, delta, n)
    c = corrs
    
    t_max = 1.0
    for i in range(len(h)):
        if delta[i] > 1e-15:
            t_max = min(t_max, (1.0 - h[i]) / delta[i])
    
    if t_max < 1e-12:
        return 0.0
    
    # Coarse search
    ts = np.linspace(0, t_max, 100)
    fvals = np.array([np.max(c + b*t + a*t*t) for t in ts])
    t_coarse = ts[fvals.argmin()]
    
    if t_coarse == 0:
        return 0.0
    
    # Golden section refinement
    span = t_max / 50.0
    t_lo = max(0.0, t_coarse - span)
    t_hi = min(t_max, t_coarse + span)
    
    gr = (np.sqrt(5) - 1) / 2
    c1 = t_lo + (1 - gr) * (t_hi - t_lo)
    c2 = t_lo + gr * (t_hi - t_lo)
    f1 = np.max(c + b*c1 + a*c1*c1)
    f2 = np.max(c + b*c2 + a*c2*c2)
    
    for _ in range(30):
        if f1 <= f2:
            t_hi = c2
            c2 = c1
            f2 = f1
            c1 = t_lo + (1 - gr) * (t_hi - t_lo)
            f1 = np.max(c + b*c1 + a*c1*c1)
        else:
            t_lo = c1
            c1 = c2
            f1 = f2
            c2 = t_lo + gr * (t_hi - t_lo)
            f2 = np.max(c + b*c2 + a*c2*c2)
    
    return (t_lo + t_hi) / 2.0

def lp_direction(h, corrs, n, n_active, step_bound):
    """Solve LP to find a descent direction."""
    active_idx = np.argsort(corrs)[-n_active:]
    
    n_vars = n + 1
    c_obj = np.zeros(n_vars)
    c_obj[-1] = 1.0
    
    A_ub = np.zeros((n_active, n_vars))
    b_ub_val = np.zeros(n_active)
    for ii, k in enumerate(active_idx):
        g = grad_corr_k(h, k, n)
        A_ub[ii, :n] = g
        A_ub[ii, n] = -1.0
        b_ub_val[ii] = -corrs[k]
    
    A_eq = np.zeros((1, n_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    bounds = [(-h[i] * step_bound, (1.0 - h[i]) * step_bound) for i in range(n)]
    bounds.append((0.0, None))
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub_val, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
        if res.status == 0:
            return res.x[:n]
    except Exception:
        pass
    return None

def lp_with_line_search(h0, dx, target_sum, max_iter=3000, time_limit=120, n_active_init=30):
    """Main optimization: LP direction + quadratic line search."""
    t_start = time.time()
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    step_bound = 1.0
    no_improve = 0
    
    for it in range(max_iter):
        if time.time() - t_start > time_limit:
            break
        if no_improve > 1000:
            break
        
        corrs = compute_corrs(h)
        c5_val = corrs.max() * dx
        
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive active set size
        n_active = min(n_active_init + no_improve // 50, 80)
        n_active = max(10, n_active)
        
        delta = lp_direction(h, corrs, n, n_active, step_bound)
        if delta is None:
            step_bound *= 0.5
            if step_bound < 1e-4:
                break
            continue
        
        if np.abs(delta).max() < 1e-12:
            step_bound *= 0.5
            continue
        
        # Quadratic line search
        t_opt = quad_line_search(h, delta, corrs, n)
        
        if t_opt < 1e-10:
            step_bound *= 0.7
            continue
        
        # Take the step
        h_new = h + t_opt * delta
        h_new = project_feasible(h_new, target_sum)
        new_c5 = c5_of_h(h_new, dx)
        
        if new_c5 < c5_val - 1e-14:
            h = h_new
        else:
            # Try smaller steps
            accepted = False
            for frac in [0.5, 0.25, 0.125]:
                h_try = h + t_opt * frac * delta
                h_try = project_feasible(h_try, target_sum)
                c5_try = c5_of_h(h_try, dx)
                if c5_try < c5_val - 1e-14:
                    h = h_try
                    accepted = True
                    break
            if not accepted:
                step_bound *= 0.7
                if step_bound < 1e-5:
                    step_bound = 0.1  # reset
    
    return best_h, best_c5

def pair_exchange(h0, dx, target_sum, time_limit=60, rng=None):
    """Pair exchange with gradient-based pair selection."""
    if rng is None:
        rng = np.random.RandomState(42)
    t_start = time.time()
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    no_improve = 0
    
    while time.time() - t_start < time_limit:
        if no_improve > 3000:
            break
        
        corrs = compute_corrs(h)
        c5_val = corrs.max() * dx
        
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        n_active = min(40, 2 * n - 1)
        active_idx = np.argsort(corrs)[-n_active:]
        g_active = np.zeros((n_active, n))
        for ii, k in enumerate(active_idx):
            g_active[ii] = grad_corr_k(h, k, n)
        c_active = corrs[active_idx]
        
        # Try random pairs
        n_tries = 300
        for _ in range(n_tries):
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j or h[i] < 1e-8 or h[j] > 1.0 - 1e-8:
                continue
            
            d = g_active[:, j] - g_active[:, i]
            # Estimate new max for various δ
            lo_d = max(h[j] - 1.0, -0.5)
            hi_d = min(h[i], 0.5)
            
            best_delta = 0.0
            best_est = corrs.max() + 1.0
            
            candidates = np.array([-0.3, -0.1, 0.0, 0.1, 0.3]) * min(hi_d, -lo_d)
            for delta in candidates:
                if delta < lo_d or delta > hi_d or abs(delta) < 1e-10:
                    continue
                est = np.max(c_active + delta * d)
                # Also check non-active correlations (rough bound)
                if est < best_est:
                    best_est = est
                    best_delta = delta
        
            if best_delta != 0 and best_est < corrs.max() - 1e-8:
                h[i] -= best_delta
                h[j] += best_delta
                h = project_feasible(h, target_sum)
                new_c5 = c5_of_h(h, dx)
                if new_c5 < c5_val - 1e-14:
                    c5_val = new_c5
                    corrs = compute_corrs(h)
                    n_active = min(40, 2 * n - 1)
                    active_idx = np.argsort(corrs)[-n_active:]
                    g_active = np.zeros((n_active, n))
                    for ii, k in enumerate(active_idx):
                        g_active[ii] = grad_corr_k(h, k, n)
                    c_active = corrs[active_idx]
                else:
                    # Revert
                    h[i] += best_delta
                    h[j] -= best_delta
                    h = project_feasible(h, target_sum)
    
    return best_h, best_c5

def slsqp_smooth(h0, dx, target_sum, beta, maxiter=1000):
    """SLSQP with smooth max approximation."""
    n = len(h0)
    
    def obj(x):
        corrs = compute_corrs(x)
        return logsumexp(beta * corrs) / beta * dx
    
    def grad_obj(x):
        corrs = compute_corrs(x)
        shifted = corrs - corrs.max()
        w = np.exp(beta * shifted)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones_like(corrs) / len(corrs)
        else:
            w /= w_sum
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] > 1e-13:
                grad += w[k] * grad_corr_k(x, k, n)
        return grad * dx
    
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    
    try:
        res = minimize(obj, h0, jac=grad_obj, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        h_out = project_feasible(res.x, target_sum)
    except Exception:
        h_out = project_feasible(h0, target_sum)
    return h_out, c5_of_h(h_out, dx)
