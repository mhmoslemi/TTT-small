        h = make_sym_h(q, n)
        corrs = corr_vals(h)
        return t - corrs[active_lags]

                idx = np.arange(i_lo, i_hi + 1)
                G = np.zeros(n)
                G[idx] += (1.0 - h[idx + d])
                G[idx + d] += (-h[idx])
            G *= dx
            J[i, 1:] = -(G[:m] - G[n - 1:n - 1 - m:-1])
        return J

    x0 = np.concatenate([[t_init], q0])
    cons = [{'type': 'ineq', 'fun': con_fun, 'jac': con_jac}]
    bounds = [(0.0, 2.0)] + [(0.0, 1.0)] * m
    try:
        res = minimize(obj, x0, jac=obj_jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        h_new = make_sym_h(res.x[1:], n)
        c5 = exact_score(h_new)
        return res.x[1:], c5
    except Exception:
        return None, None


def epi_sym_trust(q0, n, active_lags, t_init, maxiter=200):
    m = n // 2

    def obj(x):
        return x[0]

    def obj_jac(x):
        g = np.zeros(m + 1)
        g[0] = 1.0
        return g

    def con_fun(x):
        t = x[0]
        q = x[1:]
        h = make_sym_h(q, n)
        corrs = corr_vals(h)
        return t - corrs[active_lags]

    def con_jac(x):
        q = x[1:]
        h = make_sym_h(q, n)
        dx = 2.0 / n
        L = len(active_lags)
        J = np.zeros((L, m + 1))
        J[:, 0] = 1.0
        for i, lag in enumerate(active_lags):
            d = lag - (n - 1)
            i_lo = max(0, -d)
            i_hi = min(n - 1, n - 1 - d)
            if i_lo <= i_hi:
                idx = np.arange(i_lo, i_hi + 1)
                G = np.zeros(n)
                G[idx] += (1.0 - h[idx + d])
                G[idx + d] += (-h[idx])
            G *= dx
            J[i, 1:] = -(G[:m] - G[n - 1:n - 1 - m:-1])
        return J

    x0 = np.concatenate([[t_init], q0])
    cons = {'type': 'ineq', 'fun': con_fun, 'jac': con_jac}
    bounds = [(0.0, 2.0)] + [(0.0, 1.0)] * m
    try:
        res = minimize(obj, x0, jac=obj_jac, method='trust-constr',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'gtol': 1e-8, 'xftol': 1e-10})
        h_new = make_sym_h(res.x[1:], n)
        c5 = exact_score(h_new)
        return res.x[1:], c5
    except Exception:
        return None, None


def active_lags_top(corrs, n_top):
    idx = np.argsort(corrs)[::-1][:n_top]
    return sorted(idx.tolist())


def active_lags_above(corrs, threshold):
    idx = np.where(corrs >= threshold)[0]
    return sorted(idx.tolist())


def project_subgrad_step(q, n, beta, lr):
    h = make_sym_h(q, n)
    corrs = corr_vals(h)
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    ws = w.sum()
    if ws < 1e-30:
        w = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        w /= ws
    grad = weighted_grad_sym(h, n, 2.0 / n, w)
    q_new = q - lr * grad
    return np.clip(q_new, 0.0, 1.0)


def subgrad_average(q0, n, beta, n_iter, lr_init):
    m = n // 2
    q = q0.copy()
    q_sum = np.zeros(m)
    best_q = q0.copy()
    best_c = exact_score(make_sym_h(q0, n))
    for i in range(n_iter):
        q_sum += q
        q_avg = q_sum / (i + 1)
        c_avg = exact_score(make_sym_h(q_avg, n))
        if c_avg < best_c:
            best_c = c_avg
            best_q = q_avg.copy()
        lr = lr_init / (1.0 + 0.01 * i)
        q = project_subgrad_step(q, n, beta, lr)
        c_cur = exact_score(make_sym_h(q, n))
        if c_cur < best_c:
            best_c = c_cur
            best_q = q.copy()
    return best_q, best_c


def upscale_h(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return h_new


def interpolate_to_res(old_q, n_old, n_new):
    h_old = make_sym_h(old_q, n_old)
    h_new = upscale_h(h_old, n_new)
    h_new = project_feasible(h_new, n_new / 2.0)
    return make_q_from_h(h_new, n_new)


def full_space_smooth(h0, n, beta, maxiter=300):
    target_sum = n / 2.0

    def obj(x):
        h = np.clip(x, 0.0, 1.0)
        corrs = corr_vals(h)
        return logsumexp(beta * corrs) / beta

    def obj_grad(x):
        h = np.clip(x, 0.0, 1.0)
        corrs = corr_vals(h)
        shifted = corrs - corrs.max()
        w = np.exp(beta * shifted)
        ws = w.sum()
        if ws < 1e-30:
            w = np.ones(2 * n - 1) / (2 * n - 1)
        else:
            w /= ws
        return weighted_grad_sym_full(h, n, 2.0 / n, beta * w)

    try:
        res = minimize(obj, h0, jac=obj_grad, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        h_out = project_feasible(res.x, target_sum)
        c5 = exact_score(h_out)
        return h_out, c5
    except Exception:
        return h0, exact_score(h0)


def weighted_grad_sym_full(h, n, dx, weights):
    grad = np.zeros(n)
    n_lags = 2 * n - 1
    for k in range(n_lags):
        wk = weights[k]
        if wk < 1e-15:
            continue
        d = k - (n - 1)
        i_lo = max(0, -d)
        i_hi = min(n - 1, n - 1 - d)
        if i_lo > i_hi:
            continue
        idx = np.arange(i_lo, i_hi + 1)
        grad[idx] += wk * (1.0 - h[idx + d])
        grad[idx + d] += wk * (-h[idx])
    return grad * dx


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def remaining():
        return budget_s - (time.time() - t0)

    # Load parent
    try:
        h_parent = np.array(initial_h_values, dtype=float)
    except NameError:
        h_parent = 0.5 * np.ones(126)

    n = len(h_parent)
    if n % 2 != 0:
        h_parent = h_parent[:n - 1]
        n = len(h_parent)

    h_best = project_feasible(h_parent, n / 2.0)
    c5_best = exact_score(h_best)
    m = n // 2
    q_best = make_q_from_h(h_best, n)
    n_best = n
    print("Initial: n=%d, C5=%.6f" % (n_best, c5_best))

    # Phase 1: Symmetric smooth at parent resolution
    for beta in [50, 100, 200, 400, 800, 1200]:
        if remaining() < 20:
            break
        q_new, c5_new = smooth_sym_optimize(q_best, n_best, beta, maxiter=400)
        if q_new is not None and c5_new < c5_best - 1e-10:
            c5_best = c5_new
            q_best = q_new
            h_best = make_sym_h(q_best, n_best)
            print("  sym smooth beta=%d: C5=%.7f" % (beta, c5_best))

    # Phase 2: Active-set epigraph at parent resolution
    for rnd in range(10):
        if remaining() < 12:
            break
        h_cur = make_sym_h(q_best, n_best)
        corrs = corr_vals(h_cur)
        max_c = corrs.max()
        n_act = min(30 + 10 * rnd, 2 * n_best - 1)
        active = active_lags_top(corrs, n_act)
        # Try both SLSQP and trust-constr
        q_new, c5_new = epi_sym_slsqp(q_best, n_best, active, t_init=max_c * 1.0001,
                                       maxiter=400)
        if q_new is not None and c5_new < c5_best - 1e-10:
            c5_best = c5_new
            q_best = q_new
            h_best = make_sym_h(q_best, n_best)
            print("  epi slsqp rnd=%d (L=%d): C5=%.7f" % (rnd, n_act, c5_best))
        if remaining() < 12:
            break
        q_new2, c5_new2 = epi_sym_trust(q_best, n_best, active,
                                         t_init=max_c * 1.0001, maxiter=150)
        if q_new2 is not None and c5_new2 < c5_best - 1e-10:
            c5_best = c5_new2
            q_best = q_new2
            h_best = make_sym_h(q_best, n_best)
            print("  epi trust rnd=%d: C5=%.7f" % (rnd, c5_best))

    # Phase 3: Upscale to higher resolution
    target_ns = [150, 170, 200, 220, 256]
    if n_best % 2 == 0:
        for n_try in target_ns:
            if n_try <= n_best or remaining() < 60:
                continue
            if n_try % 2 != 0:
                n_try += 1
            m_try = n_try // 2
            q_try = interpolate_to_res(q_best, n_best, n_try)
            # Smooth optimize at new resolution
            for beta in [100, 300, 600]:
                if remaining() < 20:
                    break
                q_new, c5_new = smooth_sym_optimize(q_try, n_try, beta, maxiter=300)
                if q_new is not None:
                    q_try = q_new
            # Epigraph at new resolution
            if remaining() > 20:
                h_try = make_sym_h(q_try, n_try)
                corrs_try = corr_vals(h_try)
                active_try = active_lags_top(corrs_try, 40)
                q_new, c5_new = epi_sym_slsqp(q_try, n_try, active_try,
                                               t_init=corrs_try.max() * 1.0001,
                                               maxiter=300)
                if q_new is not None:
                    q_try = q_new
            h_try = make_sym_h(q_try, n_try)
            c5_try = exact_score(h_try)
            if c5_try < c5_best - 1e-9:
                c5_best = c5_try
                q_best = q_try
                h_best = h_try
                n_best = n_try
                m = m_try
                print("  upscale n=%d: C5=%.7f" % (n_best, c5_best))

    # Phase 4: Subgradient method at best resolution
    if remaining() > 30:
        q_sg, c5_sg = subgrad_average(q_best, n_best, 500, 2000, lr_init=0.05)
        if c5_sg < c5_best - 1e-10:
            c5_best = c5_sg
            q_best = q_sg
            h_best = make_sym_h(q_best, n_best)
            print("  subgrad: C5=%.7f" % c5_best)

    # Phase 5: More epigraph rounds at best resolution
    for rnd in range(8):
        if remaining() < 12:
            break
        h_cur = make_sym_h(q_best, n_best)
        corrs = corr_vals(h_cur)
        max_c = corrs.max()
        # Use lags that are close to the max
        thresh = max_c - 0.001
        active = active_lags_above(corrs, thresh)
        if len(active) < 5:
            active = active_lags_top(corrs, 30)
        if len(active) > 100:
            active = active_lags_top(corrs, 100)
        q_new, c5_new = epi_sym_slsqp(q_best, n_best, active,
                                       t_init=max_c * 1.0001, maxiter=500)
        if q_new is not None and c5_new < c5_best - 1e-10:
            c5_best = c5_new
            q_best = q_new
            h_best = make_sym_h(q_best, n_best)
            print("  epi refine rnd=%d (L=%d): C5=%.7f" % (rnd, len(active), c5_best))

    # Phase 6: Random restarts
    n_trials = 30
    for trial in range(n_trials):
        if remaining() < 15:
            break
        scale = 0.02 + 0.3 * (trial / max(n_trials, 1))
        q_pert = q_best + scale * rng.randn(m)
        q_pert = np.clip(q_pert, 0.0, 1.0)
        q_new, c5_new = smooth_sym_optimize(q_pert, n_best, 200, maxiter=200)
        if q_new is not None and c5_new < c5_best + 0.003:
            q_new2, c5_new2 = smooth_sym_optimize(q_new, n_best, 600, maxiter=300)
            if q_new2 is not None and c5_new2 < c5_best - 1e-10:
                c5_best = c5_new2
                q_best = q_new2
                h_best = make_sym_h(q_best, n_best)
                print("  restart %d: C5=%.7f" % (trial, c5_best))
                if remaining() > 15:
                    h_cur = make_sym_h(q_best, n_best)
                    corrs = corr_vals(h_cur)
                    active = active_lags_top(corrs, 50)
                    q_ep, c5_ep = epi_sym_slsqp(q_best, n_best, active,
                                                 t_init=corrs.max() * 1.0001,
                                                 maxiter=300)
                    if q_ep is not None and c5_ep < c5_best - 1e-10:
                        c5_best = c5_ep
                        q_best = q_ep
                        h_best = make_sym_h(q_best, n_best)
                        print("  restart %d epi: C5=%.7f" % (trial, c5_best))

    # Phase 7: Full-space polish (breaks symmetry slightly)
    if remaining() > 25:
        for beta in [400, 800]:
            if remaining() < 15:
                break
            h_new, c5_new = full_space_smooth(h_best, n_best, beta, maxiter=300)
            if c5_new < c5_best - 1e-10:
                c5_best = c5_new
                h_best = h_new
                m = n_best // 2
                q_best = make_q_from_h(h_best, n_best)
                print("  full polish beta=%d: C5=%.7f" % (beta, c5_best))

    # Phase 8: Final epigraph at all active lags
    if remaining() > 15:
        h_cur = make_sym_h(q_best, n_best)
        corrs = corr_vals(h_cur)
        max_c = corrs.max()
        thresh = max_c - 0.0005
        active = active_lags_above(corrs, thresh)
        if len(active) > 120:
            active = active_lags_top(corrs, 120)
        if len(active) >= 3:
            q_new, c5_new = epi_sym_slsqp(q_best, n_best, active,
                                           t_init=max_c * 1.0001, maxiter=600)
            if q_new is not None and c5_new < c5_best - 1e-10:
                c5_best = c5_new
                q_best = q_new
                h_best = make_sym_h(q_best, n_best)
                print("  final epi (L=%d): C5=%.7f" % (len(active), c5_best))

    # Phase 9: One more upscale if time allows
    if remaining() > 50:
        n_up = n_best + 40
        if n_up % 2 != 0:
            n_up += 1
        if n_up <= 400:
            m_up = n_up // 2
            q_up = interpolate_to_res(q_best, n_best, n_up)
            for beta in [200, 500]:
                if remaining() < 20:
                    break
                q_new, c5_new = smooth_sym_optimize(q_up, n_up, beta, maxiter=250)
                if q_new is not None:
                    q_up = q_new
            h_up = make_sym_h(q_up, n_up)
            c5_up = exact_score(h_up)
            if c5_up < c5_best - 1e-9:
                c5_best = c5_up
                q_best = q_up
                h_best = h_up
                n_best = n_up
                m = m_up
                print("  final upscale n=%d: C5=%.7f" % (n_best, c5_best))

    # Final validation
    h_best = np.clip(h_best, 0.0, 1.0)
    h_best = project_feasible(h_best, len(h_best) / 2.0)
    c5_final = exact_score(h_best)
    if c5_final < c5_best:
        c5_best = c5_final

    print("Final: n=%d, C5=%.7f" % (len(h_best), c5_best))
    return h_best, c5_best, len(h_best)
