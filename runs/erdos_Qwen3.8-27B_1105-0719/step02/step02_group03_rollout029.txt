    m = corrs.max()
    shifted = corrs - m
    exps = np.exp(beta * shifted)
    sm = m + np.log(exps.sum()) / beta
    w = exps / exps.sum()
    return sm, w

    h = h0.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    alpha = 0.02
    momentum = 0.85
    
    for it in range(max_iter):
        if time_limit is not None and t0 is not None and time.time() - t0 > time_limit:
            break
        
        corrs = compute_corrs(h)
        cur_max = corrs.max()
        cur_c5 = cur_max * dx
        
        # Find active lags (within small tolerance of max)
        active_thresh = cur_max - 0.01 * cur_max
        active_idx = np.where(corrs >= active_thresh)[0]

        for idx in active_idx:
            offset = idx - (n - 1)
            grad += grad_for_lag(h, offset, n)

        h_new = project_feasible(h + v, target_sum)

def compute_corrs(h):

def grad_for_lag(h, offset, n):
    grad = np.zeros(n)
    if offset >= 0:
        if n - offset > 0:
            grad[:n - offset] += (1.0 - h[offset:])
        grad[offset:] -= h[:n - offset]
    else:
        d = -offset
        grad[d:] += (1.0 - h[:n - d])
        grad[:n - d] -= h[d:]
    return grad

                continue
            offset = k - (n - 1)
            grad += w[k] * grad_for_lag(x, offset, n)
        return grad * dx

    def cons_fun(x):
        return x.sum() - target_sum

    def cons_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    bounds = [(0.0, 1.0)] * n

    try:
        res = minimize(obj, h0, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        h_out = project_feasible(res.x, target_sum)
    except Exception:
        h_out = project_feasible(h0, target_sum)
    return h_out, c5_of_h(h_out, dx)


def make_sturmian(n, target_sum, alpha=None):
    if alpha is None:
        alpha = (np.sqrt(5.0) - 1.0) / 2.0
    h = np.zeros(n)
    i = np.arange(n)
    frac = (i * alpha) % 1.0
    h[i] = frac < (float(target_sum) / n)
    count = int(h.sum())
    if count > int(target_sum):
        sorted_idx = np.argsort(-frac)
        h[sorted_idx[int(target_sum):]] = 0.0
    elif count < int(target_sum):
        sorted_idx = np.argsort(frac)
        h[sorted_idx[:int(target_sum - count)]] = 1.0
    return h


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    best_h = np.array(initial_h_values, dtype=float)
    n = len(best_h)
    dx = 2.0 / n
    target_sum = n / 2.0
    best_h = project_feasible(best_h, target_sum)
    best_c5 = c5_of_h(best_h, dx)

    def remaining():
        return budget_s - (time.time() - t0)

    def try_update(h_cand, c5_cand):
        nonlocal best_h, best_c5
        if c5_cand < best_c5:
            best_c5 = c5_cand
            best_h = h_cand.copy()

    # Phase 1: Binary steepest descent from initial (main workhorse)
    h1, c5_1 = steepest_binary(best_h, dx, max_sweeps=500,
                               time_limit=budget_s * 0.20, t0=t0)
    try_update(h1, c5_1)

    # Phase 2: ILS with random perturbations
    if remaining() > budget_s * 0.30:
        h2, c5_2 = iterated_local_search(best_h, dx, rng, n_restarts=40,
                                         time_limit=budget_s * 0.15, t0=t0)
        try_update(h2, c5_2)

    # Phase 3: Continuous subgradient polish
    if remaining() > budget_s * 0.25:
        n_b = len(best_h)
        dx_b = 2.0 / n_b
        ts_b = n_b / 2.0
        h3, c5_3 = subgradient_opt(best_h, dx_b, ts_b,
                                   time_limit=budget_s * 0.15, t0=t0)
        try_update(h3, c5_3)

    # Phase 4: Smooth max SLSQP
    if remaining() > budget_s * 0.20:
        n_b = len(best_h)
        dx_b = 2.0 / n_b
        ts_b = n_b / 2.0
        for beta in [200, 500, 1000]:
            if remaining() < 30:
                break
            h4, c5_4 = smooth_max_slsqp(best_h, dx_b, ts_b, beta, 500,
                                        time_limit=remaining() * 0.1, t0=t0)
            try_update(h4, c5_4)

    # Phase 5: More ILS from best
    if remaining() > budget_s * 0.20:
        h5, c5_5 = iterated_local_search(best_h, dx, rng, n_restarts=60,
                                         time_limit=budget_s * 0.12, t0=t0)
        try_update(h5, c5_5)

    # Phase 6: Try other n values
    for n_try in [128, 132, 140, 120, 144, 136]:
        if remaining() < 80:
            break
        dx_try = 2.0 / n_try
        ts_try = n_try / 2.0
        x_orig = np.linspace(0.0, 2.0, len(best_h))
        x_new = np.linspace(0.0, 2.0, n_try)
        h_interp = np.interp(x_new, x_orig, best_h)
        h_interp = project_feasible(h_interp, ts_try)

        # Steepest descent
        time_alloc = min(remaining() * 0.06, 60)
        h6, c5_6 = steepest_binary(h_interp, dx_try, max_sweeps=300,
                                   time_limit=time_alloc, t0=t0)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()
            # ILS from this
            if remaining() > 60:
                h6b, c5_6b = iterated_local_search(h6, dx_try, rng, n_restarts=15,
                                                   time_limit=remaining() * 0.05, t0=t0)
                try_update(h6b, c5_6b)

    # Phase 7: Sturmian starts with optimization
    for n_try in [126, 128, 132]:
        if remaining() < 50:
            break
        ts_try = int(n_try / 2)
        dx_try = 2.0 / n_try
        alphas = [(np.sqrt(5.0) - 1.0) / 2.0, np.sqrt(2.0) - 1.0,
                  (np.sqrt(3.0) - 1.0) / 2.0, 0.4142, 0.3819, 0.2764]
        for a_try in alphas:
            if remaining() < 20:
                break
            h_st = make_sturmian(n_try, ts_try, a_try)
            time_alloc = min(remaining() * 0.04, 30)
            h7, c5_7 = steepest_binary(h_st, dx_try, max_sweeps=200,
                                       time_limit=time_alloc, t0=t0)
            if c5_7 < best_c5:
                best_c5 = c5_7
                best_h = h7.copy()

    # Phase 8: Final subgradient + SLSQP polish
    if remaining() > 40:
        n_b = len(best_h)
        dx_b = 2.0 / n_b
        ts_b = n_b / 2.0
        h8, c5_8 = subgradient_opt(best_h, dx_b, ts_b,
                                   time_limit=min(remaining() * 0.3, 60), t0=t0)
        try_update(h8, c5_8)

    if remaining() > 30:
        n_b = len(best_h)
        dx_b = 2.0 / n_b
        ts_b = n_b / 2.0
        h9, c5_9 = smooth_max_slsqp(best_h, dx_b, ts_b, 2000, 800,
                                    time_limit=min(remaining() * 0.4, 60), t0=t0)
        try_update(h9, c5_9)

    # Phase 9: Final binary steepest on best
    if remaining() > 30:
        n_b = len(best_h)
        dx_b = 2.0 / n_b
        h10, c5_10 = steepest_binary(best_h, dx_b, max_sweeps=200,
                                     time_limit=min(remaining() * 0.4, 40), t0=t0)
        try_update(h10, c5_10)

    # Final cleanup
    n_b = len(best_h)
    dx_b = 2.0 / n_b
    ts_b = n_b / 2.0
    best_h = project_feasible(best_h, ts_b)
    best_c5 = c5_of_h(best_h, dx_b)

    return best_h, best_c5, len(best_h)
