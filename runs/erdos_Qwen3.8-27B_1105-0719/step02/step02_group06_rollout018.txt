def project_feasible(x, target_sum):
    x = np.asarray(x, dtype=float)
    x = np.where(np.isfinite(x), x, 0.5)
    x = np.clip(x, 0.0, 1.0)
    diff = target_sum - x.sum()
    if abs(diff) < 1e-13:
        return x.copy()
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(x + mid, 0.0, 1.0).sum()
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    h = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(h)
            for idx in order:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
                if diff < 1e-13:

            for idx in order:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
                if -diff < 1e-13:
                    break

def golden_pair_step(h, a, b, n_iter=30):
    """Golden section: minimize max-corr by transferring between h[a] and h[b].
    h[a] += delta, h[b] -= delta. Modifies h in place. Returns improved h."""
    dl = max(-h[a], h[b] - 1.0)
    dh = min(1.0 - h[a], h[b])
    if dh - dl < 1e-14:
        return h
    gr = 0.6180339887498949  # (sqrt(5)-1)/2
    a0, b0 = dl, dh
    c0 = b0 - gr * (b0 - a0)
    d0 = a0 + gr * (b0 - a0)

        h[a] += d
        h[b] -= d
        v = float(raw_corr(h).max())
        h[a] -= d
        h[b] += d
        return v

import numpy as np
import time


def eval_score(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)


def raw_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(x, target_sum):
    x = np.asarray(x, dtype=float)
    x = np.where(np.isfinite(x), x, 0.5)
    x = np.clip(x, 0.0, 1.0)
    diff = target_sum - x.sum()
    if abs(diff) < 1e-13:
        return x.copy()
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(x + mid, 0.0, 1.0).sum()
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    h = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(h)
            for idx in order:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
                if diff < 1e-13:
                    break
        else:
            order = np.argsort(-h)
            for idx in order:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
                if -diff < 1e-13:
                    break
    return np.clip(h, 0.0, 1.0)


def pair_eval(h, a, b, d):
    h[a] += d
    h[b] -= d
    v = float(raw_corr(h).max())
    h[a] -= d
    h[b] += d
    return v


def golden_pair_step(h, a, b, n_iter=30):
    dl = max(-h[a], h[b] - 1.0)
    dh = min(1.0 - h[a], h[b])
    if dh - dl < 1e-14:
        return h
    gr = 0.6180339887498949
    a0, b0 = dl, dh
    c0 = b0 - gr * (b0 - a0)
    d0 = a0 + gr * (b0 - a0)
    fc = pair_eval(h, a, b, c0)
    fd = pair_eval(h, a, b, d0)
    for _ in range(n_iter):
        if fc < fd:
            b0 = d0
            d0 = c0
            fd = fc
            c0 = b0 - gr * (b0 - a0)
            fc = pair_eval(h, a, b, c0)
        else:
            a0 = c0
            c0 = d0
            fc = fd
            d0 = a0 + gr * (b0 - a0)
            fd = pair_eval(h, a, b, d0)
    best_d = (a0 + b0) * 0.5
    flo = pair_eval(h, a, b, dl + 1e-10)
    fhi = pair_eval(h, a, b, dh - 1e-10)
    fmid = pair_eval(h, a, b, best_d)
    if flo < fmid and flo < fhi:
        best_d = dl + 1e-10
    elif fhi < fmid and fhi < flo:
        best_d = dh - 1e-10
    h[a] += best_d
    h[b] -= best_d
    np.clip(h, 0.0, 1.0, out=h)
    return h


def compute_targeted_pairs(h, n, max_pairs=80, rng=None):
    corrs = raw_corr(h)
    max_val = corrs.max()
    if max_val < 1e-15:
        max_val = 1.0
    active_mask = corrs > max_val - 0.03 * max_val
    active_lags = np.where(active_mask)[0]
    pair_scores = {}
    for k in active_lags:
        d = k - (n - 1)
        i_lo = max(0, -d)
        i_hi = min(n - 1, n - 1 - d)
        if i_lo > i_hi:
            continue
        weight = corrs[k] / max_val
        for i in range(i_lo, i_hi + 1):
            j = i + d
            if 0 <= j < n and j != i:
                key = (min(i, j), max(i, j))
                gap = abs(h[i] - h[j])
                score = weight * gap
                if score > pair_scores.get(key, 0.0):
                    pair_scores[key] = score
    if not pair_scores:
        return []
    sorted_items = sorted(pair_scores.items(), key=lambda x: -x[1])
    pairs = [p for p, s in sorted_items[:max_pairs] if s > 0.005]
    if rng is not None:
        for _ in range(max(5, max_pairs // 5)):
            i = rng.randint(n)
            j = rng.randint(n)
            if i != j:
                pairs.append((min(i, j), max(i, j)))
    return pairs


def pairwise_descent(h, max_rounds=300, pairs_per_round=60, time_limit=None,
                     rng=None, n_golden=28, stagnation_limit=30):
    n = len(h)
    h = h.copy()
    best_raw = float(raw_corr(h).max())
    no_improve_count = 0

    for rnd in range(max_rounds):
        if time_limit is not None and time.time() > time_limit:
            break
        corrs = raw_corr(h)
        cur_max = float(corrs.max())
        if cur_max < best_raw - 1e-13:
            best_raw = cur_max

        pairs = compute_targeted_pairs(h, n, max_pairs=pairs_per_round, rng=rng)
        if not pairs:
            break

        if rng is not None:
            rng.shuffle(pairs)

        local_best = cur_max
        for (a, b) in pairs:
            if time_limit is not None and time.time() > time_limit:
                break
            h[a] += 0.0
            h_try = h.copy()
            golden_pair_step(h_try, a, b, n_iter=n_golden)
            sc = float(raw_corr(h_try).max())
            if sc < local_best - 1e-13:
                h = h_try
                local_best = sc

        if local_best < best_raw - 1e-13:
            best_raw = local_best
            no_improve_count = 0
        else:
            no_improve_count += 1
            if no_improve_count >= stagnation_limit and rng is not None:
                # Perturb to escape local optimum
                n_pert = max(3, n // 20)
                for _ in range(n_pert):
                    i = rng.randint(n)
                    j = rng.randint(n)
                    if i != j:
                        delta = rng.uniform(-0.05, 0.05)
                        h[i] += delta
                        h[j] -= delta
                np.clip(h, 0.0, 1.0, out=h)
                h = project_feasible(h, n / 2.0)
                no_improve_count = 0

    return h, best_raw


def interpolate_h(h_old, n_new):
    n_old = len(h_old)
    x_old = (np.arange(n_old) + 0.5) * 2.0 / n_old
    x_new = (np.arange(n_new) + 0.5) * 2.0 / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new / 2.0)


def de_solve_trial(trial, n, F, CR, rng, base):
    trial = project_feasible(trial, n / 2.0)
    for _ in range(8):
        i1 = rng.randint(n)
        j1 = rng.randint(n)
        if i1 != j1:
            golden_pair_step(trial, min(i1, j1), max(i1, j1), n_iter=22)
    return trial


def simple_de(h_init, n_gens=20, pop_size=15, F=0.7, CR=0.4, time_limit=None, rng=None):
    n = len(h_init)
    pop = np.zeros((pop_size, n))
    pop[0] = h_init.copy()
    for i in range(1, pop_size):
        pop[i] = project_feasible(0.5 + 0.35 * rng.randn(n), n / 2.0)
    for i in range(pop_size):
        golden_pair_step(pop[i], rng.randint(n), rng.randint(n), n_iter=15)
        golden_pair_step(pop[i], rng.randint(n), rng.randint(n), n_iter=15)

    scores = np.array([float(raw_corr(pop[i]).max()) for i in range(pop_size)])
    best_idx = int(np.argmin(scores))
    best_h = pop[best_idx].copy()
    best_score = scores[best_idx]

    for gen in range(n_gens):
        if time_limit is not None and time.time() > time_limit:
            break
        for i in range(pop_size):
            if time_limit is not None and time.time() > time_limit:
                break
            candidates = [j for j in range(pop_size) if j != i]
            a = candidates[rng.randint(len(candidates))]
            candidates2 = [j for j in candidates if j != a]
            b = candidates2[rng.randint(len(candidates2))]
            candidates3 = [j for j in candidates2 if j != b]
            c = candidates3[rng.randint(len(candidates3))]

            trial = pop[a].astype(float) + F * (pop[b].astype(float) - pop[c].astype(float))
            mask = rng.rand(n) < CR
            trial[mask] = pop[i][mask]
            trial = de_solve_trial(trial, n, F, CR, rng, pop[i])
            trial = project_feasible(trial, n / 2.0)
            sc = float(raw_corr(trial).max())
            if sc < scores[i]:
                pop[i] = trial
                scores[i] = sc
            if sc < best_score:
                best_score = sc
                best_h = trial.copy()
    return best_h, best_score
