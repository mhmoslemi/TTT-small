def repair_feasible(vec):
    n = len(vec)

    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx] + 1e-15)

            remaining -= add

    else:

    """
    Solve the linearized minimax LP to find a direction.
    Variables: h' (n values), t (scalar)
    Minimize t
    Subject to:
      t - G[m,:]^T h' >= P[m] - G[m,:]^T h  for m in active_set
      sum(h') = n/2
      0 <= h' <= 1
      h - radius <= h' <= h + radius
    """
    n = len(h)
    G_active = gradient[active_set, :]  # (K, n)
    P_active = profile[active_set]  # (K,)
    K = len(active_set)
    
    # Decision variables: [h'(n), t(1)] -> total n+1 variables
    # Objective: minimize t -> c = [0,...,0, 1]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Inequality constraints: A_ub @ x <= b_ub
    # t - G_active[m,:]^T h' >= P[m] - G_active[m,:]^T h
    # -(t - G_active[m,:]^T h') <= -(P[m] - G_active[m,:]^T h)
    # -G_active[m,:]^T h' - t <= G_active[m,:]^T h - P[m]
    
    A_ub = np.zeros((K, n + 1))
    b_ub = np.zeros(K)
    for i, m in enumerate(active_set):
        A_ub[i, :n] = -G_active[i, :]
        A_ub[i, n] = -1.0
        b_ub[i] = G_active[i, :] @ h - P_active[i]
    
    # Bounds: max(0, h - radius) <= h' <= min(1, h + radius), 0 <= t
    lower = np.maximum(0.0, h - radius)
    upper = np.minimum(1.0, h + radius)
    bounds = list(zip(lower, upper))
    bounds.append((0.0, None))  # t >= 0
    
    # Equality: sum(h') = n/2
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    try:

    except Exception:

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx] + 1e-15)
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx] + 1e-15)
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:

            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

import numpy as np
from scipy.optimize import linprog
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx] + 1e-15)
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx] + 1e-15)
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def build_gradient_matrix(h):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    G = np.zeros((M, n))
    for m in range(M):
        d = m - (n - 1)
        if d == 0:
            G[m, :] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            G[m, d:n] += dx * (1.0 - h[0:n-d])
            G[m, 0:n-d] += -dx * h[d:n]
        else:
            dp = -d
            G[m, 0:n-dp] += dx * (1.0 - h[dp:n])
            G[m, dp:n] += -dx * h[0:n-dp]
    return G

def symmetrize(h):
    return (h + h[::-1]) / 2.0

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

def get_active(profile, max_k=20):
    c = np.max(profile)
    med = np.median(profile)
    spread = max(c - med, 1e-4)
    # Adaptive beam
    top_vals = np.sort(profile)[-min(8, len(profile)):]
    flatness = top_vals[-1] - top_vals[0]
    if flatness < 2e-4:
        beam = 3e-4
    else:
        beam = max(3e-3, min(0.02, 0.07 * spread))
    thresh = c - beam
    active = np.where(profile >= thresh)[0]
    if len(active) < 3:
        active = np.argsort(profile)[-3:]
    if len(active) > max_k:
        top_idx = np.argsort(profile[active])[-max_k:]
        active = active[top_idx]
    return active

def lp_direction(h, profile, G, active, radius):
    n = len(h)
    K = len(active)
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_ub = np.zeros((K, n + 1))
    b_ub = np.zeros(K)
    G_a = G[active, :]
    P_a = profile[active]
    A_ub[:K, :n] = -G_a
    A_ub[:K, n] = -1.0
    b_ub = G_a @ h - P_a
    lo = np.maximum(0.0, h - radius)
    hi = np.minimum(1.0, h + radius)
    bounds = list(zip(lo, hi))
    bounds.append((0.0, 5.0))
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n * 0.5])
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def grad_direction(h, profile, G, active):
    n = len(h)
    w = profile[active] - (np.max(profile) - 0.01)
    w = np.maximum(w, 0.01)
    w = w / w.sum()
    g = G[active, :].T @ w
    g -= g.mean()
    ng = np.linalg.norm(g)
    if ng < 1e-15:
        return None
    return g / ng
