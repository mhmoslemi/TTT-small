    m = n // 2

    cv = corr_values(h)
    dx = 2.0 / n

    # Variables: [delta_0, ..., delta_{m-1}, t]
    # n_vars = m + 1
    n_vars = m + 1

    c_obj = np.zeros(n_vars)
    c_obj[m] = 1.0  # minimize t

    # delta_h[k] = delta_q[k] for k < m

import numpy as np
from scipy.optimize import linprog
import time

def eval_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def corr_values(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def grad_for_lag(h, j, n):
    dx = 2.0 / n
    d = j - (n - 1)
    L = max(0, -d)
    U = min(n - 1, n - 1 - d)
    g = np.zeros(n)
    # First factor: h[m] where m-d is in [L, U]
    lo1 = max(0, L + d)
    hi1 = min(n - 1, U + d)
    if lo1 <= hi1:
        g[lo1:hi1+1] += (1.0 - h[lo1-d:hi1+1-d])
    # Second factor: h[m] where m is in [L, U]
    lo2 = L
    hi2 = U
    if lo2 <= hi2:
        g[lo2:hi2+1] -= h[lo2+d:hi2+1+d]
    return dx * g

def build_sym(q, n):
    m = n // 2
    h = np.empty(n)
    h[:m] = q
    h[m:] = 1.0 - q[::-1]
    return h

def to_sym_q(h, n):
    m = n // 2
    return 0.5 * (h[:m] + 1.0 - h[n-1:m-1:-1])

def select_active(cv, margin_frac=0.05, pad=2):
    cmax = cv.max()
    thresh = cmax - margin_frac * cmax
    idx = np.where(cv >= thresh)[0]
    # Pad
    extended = set()
    for a in idx:
        for delta in range(-pad, pad+1):
            if 0 <= a + delta < len(cv):
                extended.add(a + delta)
    return np.sort(list(extended))

def slp_step_sym(q, n, active_lags, R):
    m = n // 2
    h = build_sym(q, n)
    cv = corr_values(h)
    cmax = cv.max()
    
    n_vars = m + 1  # [delta_0, ..., delta_{m-1}, t]
    
    c_obj = np.zeros(n_vars)
    c_obj[m] = 1.0
    
    A_ub = []
    b_ub = []
    
    for j in active_lags:
        gh = grad_for_lag(h, j, n)
        gq = gh[:m] - gh[n-1:m-1:-1]
        row = np.zeros(n_vars)
        row[:m] = gq
        row[m] = -1.0
        A_ub.append(row)
        b_ub.append(-cv[j])
    
    if not A_ub:
        return q, cmax, 0
    
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    bounds = []
    for i in range(m):
        lo = max(-q[i], -R)
        hi = min(1.0 - q[i], R)
        bounds.append((lo, hi))
    bounds.append((0.0, cmax + 0.5))
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if not res.success:
        return q, cmax, 0
    
    delta = res.x[:m]
    
    # Line search
    best_q = q.copy()
    best_c5 = cmax
    for alpha in [1.0, 0.7, 0.5, 0.35, 0.25, 0.15, 0.08, 0.04]:
        q_new = np.clip(q + alpha * delta, 0.0, 1.0)
        h_new = build_sym(q_new, n)
        c5_new = eval_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_q = q_new.copy()
    
    return best_q, best_c5, delta

def slp_run_sym(q, n, max_iters=50, R0=0.08, time_limit=100.0, t_start=0):
    m = n // 2
    h = build_sym(q, n)
    c5 = eval_c5(h)
    R = R0
    stall = 0
    
    for it in range(max_iters):
        if time.time() - t_start > time_limit:
            break
        
        cv = corr_values(h)
        active = select_active(cv, margin_frac=0.05, pad=2)
        
        q_new, c5_new, delta = slp_step_sym(q, n, active, R)
        
        if c5_new < c5 - 1e-12:
            q = q_new
            h = build_sym(q, n)
            c5 = c5_new
            stall = 0
            R = min(0.2, R * 1.2)
        else:
            stall += 1
            R *= 0.7
            if stall > 10:
                break
            if R < 0.005:
                # Try with wider active set
                active2 = select_active(cv, margin_frac=0.15, pad=3)
                q_new, c5_new, _ = slp_step_sym(q, n, active2, max(R, 0.02))
                if c5_new < c5 - 1e-12:
                    q = q_new
                    h = build_sym(q, n)
                    c5 = c5_new
                    stall = 0
                    R = R0
                elif R < 0.002:
                    break
    
    return q, c5

def interp_to(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h)
    return h_new

def slp_step_full(h, n, active_lags, R):
    cv = corr_values(h)
    cmax = cv.max()
    
    n_vars = n + 1  # [delta_0, ..., delta_{n-1}, t]
    
    c_obj = np.zeros(n_vars)
    c_obj[n] = 1.0
    
    A_ub = []
    b_ub = []
    
    for j in active_lags:
        gh = grad_for_lag(h, j, n)
        row = np.zeros(n_vars)
        row[:n] = gh
        row[n] = -1.0
        A_ub.append(row)
        b_ub.append(-cv[j])
    
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    # Equality: sum(delta) = 0
    A_eq = np.zeros((1, n_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    bounds = []
    for i in range(n):
        lo = max(-h[i], -R)
        hi = min(1.0 - h[i], R)
        bounds.append((lo, hi))
    bounds.append((0.0, cmax + 0.5))
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if not res.success:
        return h, cmax
    
    delta = res.x[:n]
    
    best_h = h.copy()
    best_c5 = cmax
    for alpha in [1.0, 0.7, 0.5, 0.35, 0.25, 0.15, 0.08]:
        h_new = np.clip(h + alpha * delta, 0.0, 1.0)
        # Fix sum
        diff = n/2.0 - h_new.sum()
        if abs(diff) > 1e-10:
            mask = (h_new > 1e-10) & (h_new < 1.0 - 1e-10)
            if mask.any():
                h_new[mask] += diff / mask.sum()
                h_new = np.clip(h_new, 0.0, 1.0)
        c5_new = eval_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
    
    return best_h, best_c5

def slp_run_full(h, n, max_iters=30, R0=0.05, time_limit=60.0, t_start=0):
    c5 = eval_c5(h)
    R = R0
    stall = 0
    
    for it in range(max_iters):
        if time.time() - t_start > time_limit:
            break
        
        cv = corr_values(h)
        active = select_active(cv, margin_frac=0.05, pad=2)
        
        h_new, c5_new = slp_step_full(h, n, active, R)
        
        if c5_new < c5 - 1e-12:
            h = h_new
            c5 = c5_new
            stall = 0
            R = min(0.15, R * 1.2)
        else:
            stall += 1
            R *= 0.7
            if stall > 8:
                break
            if R < 0.003:
                break
    
    return h, c5
