```python
    """Compute the vector of changes in correlations when h[p] increases by 1."""
    delta_c = np.zeros(2*n - 1)
    # h[p] appears in c_k with coefficient (1-h[p-(k-(n-1))]) if p is in valid range for i
    for k in range(2*n - 1):
```

```python
def find_optimal_delta(c, alpha, delta_min, delta_max):
    # Minimize max_k (c[k] + delta * alpha[k]) over delta in [delta_min, delta_max]
    # Use subgradient bisection
    
    lo = delta_min
    hi = delta_max
    
    for _ in range(50):
        mid = (lo + hi) / 2
        vals = c + mid * alpha
        max_idx = np.argmax(vals)
        # subgradient is alpha[max_idx] (one of possibly many)
        # If alpha at max is positive, we should go left
        # If negative, go right
        # But there might be multiple active constraints
        active = np.where(np.isclose(vals, vals.max(), atol=1e-12))[0]
        if len(active) == 1:
            if alpha[active[0]] > 1e-14:
                hi = mid
            elif alpha[active[0]] < -1e-14:
                lo = mid
            else:
                break
        else:
            # Multiple active: check if 0 is in the convex hull of their alpha values
            if alpha[active].min() >= -1e-14:
                hi = mid
            elif alpha[active].max() <= 1e-14:
                lo = mid
            else:
                break  # 0 is in the subdifferential
    return (lo + hi) / 2
```

```python
def corr_delta_all(h):
    n = len(h)
    m = 2 * n - 1
    delta_c = np.zeros((n, m))
    # Vectorized: for each p, compute the two contributions
    p_idx = np.arange(n)
    k_idx = np.arange(m)
    # Use broadcasting
    P, K = np.meshgrid(p_idx, k_idx, indexing='ij')
    # First term: (1-h[j]) where j = P - (K-(n-1)), valid if 0<=j<n and K-(n-1)<=P<=K
    J1 = P - (K - (n - 1))
    mask1 = (J1 >= 0) & (J1 < n) & (K - (n - 1) <= P) & (P <= K)
    # But we also need P in [0, n-1] and K-(n-1) <= P <= min(n-1, K)
    # P is already in [0, n-1]
    # K-(n-1) <= P  <=>  P >= K-n+1
    # P <= K
    # P <= n-1 (always true)
    # K-(n-1) <= n-1  <=> K <= 2n-2 (always true for K in [0, 2n-2])
    # So mask1 is: 0 <= J1 < n AND K-n+1 <= P <= K
    mask1 = (J1 >= 0) & (J1 < n) & (P >= K - n + 1) & (P <= K)
    # For valid entries, add (1-h[J1])
    # Clip J1 for safe indexing
    J1_safe = np.clip(J1, 0, n-1)
    delta_c += mask1 * (1.0 - h[J1_safe])
    
    # Second term: -h[I] where I = P + K - (n-1), valid if 0<=I<n
    I = P + K - (n - 1)
    mask2 = (I >= 0) & (I < n)
    I_safe = np.clip(I, 0, n-1)
    delta_c += mask2 * (-h[I_safe])
    
    return delta_c
```

```python
def lp_refine(h, dx, target_sum, n_active=15, max_iters=20):
    n = len(h)
    m = 2 * n - 1
    h = h.copy()
    best_c5 = c5_of_h(h, dx)
    corrs = compute_corrs(h)
    
    for it in range(max_iters):
        delta_c = corr_delta_all(h)
        
        # Select active coordinates by gradient
        beta = 40.0
        shifted = corrs - corrs.max()
        w = np.exp(beta * shifted)
        w /= w.sum()
        grad = np.zeros(n)
        for j in range(n):
            grad[j] = np.dot(w, delta_c[j]) * dx
        
        # Top coordinates by |gradient|
        active = np.argsort(np.abs(grad))[::-1][:n_active]
        m_act = len(active)
        
        # Build LP: variables are [delta_1, ..., delta_m, lambda]
        # Minimize lambda
        # s.t. sum(corrs) + delta^T * dc[:,k] <= lambda for all k
        #      sum(delta_j) = 0
        #      |delta_j| <= bound_j
        
        c_obj = np.zeros(m_act + 1)
        c_obj[-1] = 1.0  # minimize lambda
        
        # Inequality constraints: A_ub @ x <= b_ub
        # corrs[k] + sum_j delta_j * delta_c[active[j], k] <= lambda
        # => sum_j delta_j * delta_c[active[j], k] - lambda <= -corrs[k]
        A_ub = np.zeros((m, m_act + 1))
        for k in range(m):
            A_ub[k, :m_act] = delta_c[active, k]
            A_ub[k, m_act] = -1.0
        b_ub = -corrs
        
        # Equality: sum of deltas = 0
        A_eq = np.zeros((1, m_act + 1))
        A_eq[0, :m_act] = 1.0
        b_eq = np.zeros(1)
        
        # Bounds on delta_j
        bounds = []
        for j in range(m_act):
            idx = active[j]
            lo = -h[idx]
            hi = 1.0 - h[idx]
            bounds.append((lo, hi))
        bounds.append((best_c5 / dx - 1e-10, None))  # lambda >= current max - tiny
        
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs')
            if res.status != 0:
                break
            deltas = res.x[:m_act]
            if np.abs(deltas).max() < 1e-12:
                break
            
            h_new = h.copy()
            for j, idx in enumerate(active):
                h_new[idx] += deltas[j]
            h_new = project_feasible(h_new, target_sum)
            new_c5 = c5_of_h(h_new, dx)
            if new_c5 < best_c5 - 1e-14:
                h = h_new
                best_c5 = new_c5
                corrs = compute_corrs(h)
            else:
                # Try smaller step
                h_new = h.copy()
                for j, idx in enumerate(active):
                    h_new[idx] += 0.5 * deltas[j]
                h_new = project_feasible(h_new, target_sum)
                new_c5 = c5_of_h(h_new, dx)
                if new_c5 < best_c5 - 1e-14:
                    h = h_new
                    best_c5 = new_c5
                    corrs = compute_corrs(h)
                else:
                    break
        except Exception:
            break
    
    return h, best_c5
```

```python
import numpy as np
from scipy.optimize import minimize, linprog
from scipy.special import logsumexp
import time


def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)


def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def c5_of_h(h, dx):
    return compute_corrs(h).max() * dx


def corr_delta_all(h):
    n = len(h)
    m = 2 * n - 1
    P, K = np.meshgrid(np.arange(n), np.arange(m), indexing='ij')
    J1 = P - (K - (n - 1))
    mask1 = (J1 >= 0) & (J1 < n) & (P >= K - n + 1) & (P <= K)
    J1_safe = np.clip(J1, 0, n - 1)
    delta_c = mask1.astype(float) * (1.0 - h[J1_safe])
    I = P + K - (n - 1)
    mask2 = (I >= 0) & (I < n)
    I_safe = np.clip(I, 0, n - 1)
    delta_c += mask2.astype(float) * (-h[I_safe])
    return delta_c


def find_optimal_delta(c, alpha, d_lo, d_hi):
    for _ in range(80):
        if d_hi - d_lo < 1e-13:
            break
        mid = 0.5 * (d_lo + d_hi)
        vals = c + mid * alpha
        mx = vals.max()
        active = vals >= mx - 1e-12
        a_act = alpha[active]
        if a_act.size == 0:
            break
        if a_act.min() > 1e-14:
            d_hi = mid
        elif a_act.max() < -1e-14:
            d_lo = mid
        else:
            break
    return 0.5 * (d_lo + d_hi)


def pair_transfer_optimize(h, dx, target_sum, max_rounds, n_cand, n_random, rng):
    n = len(h)
    h = h.copy()
    best_c5 = c5_of_h(h, dx)
    corrs = compute_corrs(h)
    stagnant = 0
    
    for rnd in range(max_rounds):
        delta_c = corr_delta_all(h)
        
        beta = 40.0
        shifted = corrs - corrs.max()
        w = np.exp(beta * shifted)
        w /= w.sum()
        grad = np.zeros(n)
        for j in range(n):
            grad[j] = np.dot(w, delta_c[j]) * dx
        
        pos = np.argsort(grad, kind='stable')[::-1][:n_cand]
        neg = np.argsort(grad, kind='stable')[:n_cand]
        
        improved = False
        for p in pos:
            for q in neg:
                if p == q:
                    continue
                d_lo = max(-h[p], h[q] - 1.0)
                d_hi = min(1.0 - h[p], h[q])
                if d_lo >= d_hi - 1e-14:
                    continue
                alpha = delta_c[p] - delta_c[q]
                delta_opt = find_optimal_delta(corrs, alpha, d_lo, d_hi)
                if abs(delta_opt) < 1e-13:
                    continue
                h_new = h.copy()
                h_new[p] += delta_opt
                h_new[q] -= delta_opt
                new_c5 = compute_corrs(h_new).max() * dx
                if new_c5 < best_c5 - 1e-14:
                    h = h_new
                    best_c5 = new_c5
                    corrs = compute_corrs(h)
                    improved = True
                    stagnant = 0
                    break
            if improved:
                break
        
        if not improved:
            for _ in range(n_random):
                p = rng.randint(n)
                q = rng.randint(n)
                if p == q:
                    continue
                d_lo = max(-h[p], h[q] - 1.0)
                d_hi = min(1.0 - h[p], h[q])
                if d_lo >= d_hi - 1e-14:
                    continue
                alpha = delta_c[p] - delta_c[q]
                delta_opt = find_optimal_delta(corrs, alpha, d_lo, d_hi)
                if abs(delta_opt) < 1e-13:
                    continue
                h_new = h.copy()
                h_new[p] += delta_opt
                h_new[q] -= delta_opt
                new_c5 = compute_corrs(h_new).max() * dx
                if new_c5 < best_c5 - 1e-14:

                    best_c5 = new_c5

                    stagnant = 0
                    break

def lp_refine(h, dx, target_sum, n_active, max_iters):
    n = len(h)
    m = 2 * n - 1
    h = h.copy()
    best_c5 = c5_of_h(h, dx)
    corrs = compute_corrs(h)
    
    for it in range(max_iters):
        delta_c = corr_delta_all(h)
        
        beta = 40.0
        shifted = corrs - corrs.max()
        w = np.exp(beta * shifted)
        w /= w.sum()
        grad = np.zeros(n)
        for j in range(n):
            grad[j] = np.dot(w, delta_c[j]) * dx
        
        active = np.argsort(np.abs(grad))[::-1][:n_active]
        m_act = len(active)
        
        c_obj = np.zeros(m_act + 1)
        c_obj[-1] = 1.0
        
        A_ub = np.zeros((m, m_act + 1))
        for k in range(m):
            A_ub[k, :m_act] = delta_c[active, k]
            A_ub[k, m_act] = -1.0
        b_ub = -corrs
        
        A_eq = np.zeros((1, m_act + 1))
        A_eq[0, :m_act] = 1.0
        b_eq = np.zeros(1)
        
        bounds = []
        for j in range(m_act):
            idx = active[j]
            bounds.append((-h[idx], 1.0 - h[idx]))
        bounds.append((corrs.max() - 1e-10, None))
        
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs')
            if res.status != 0:
                break
            deltas = res.x[:m_act]
            if np.abs(deltas).max() < 1e-13:
                break
            
            for frac in [1.0, 0.5, 0.25]:
                h_new = h.copy()
                for j, idx in enumerate(active):
                    h_new[idx] += frac * deltas[j]
                h_new = project_feasible(h_new, target_sum)
                new_c5 = c5_of_h(h_new, dx)
                if new_c5 < best_c5 - 1e-14:
                    h = h_new
                    best_c5 = new_c5
                    corrs = compute_corrs(h)
                    break
            else:
                break
        except Exception:
            break
    
    return h, best_c5


def slsqp_step(h0, dx, target_sum, beta, maxiter):
    n = len(h0)
    
    def obj(x):
        corrs = compute_corrs(x)
        return logsumexp(beta * corrs) / beta * dx
    
    def grad_obj(x):
        delta_c = corr_delta_all(x)
        corrs = compute_corrs(x)
        shifted = beta * corrs - logsumexp(beta * corrs)
        w = np.exp(shifted)
        grad = np.zeros(n)
        for j in range(n):
            grad[j] = np.dot(w, delta_c[j]) * dx
        return grad
    
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0, jac=grad_obj, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        h_out = project_feasible(res.x, target_sum)
    except Exception:
        h_out = project_feasible(h0, target_sum)
    return h_out, c5_of_h(h_out, dx)

    return budget_s - (time.time() - t0)
```
