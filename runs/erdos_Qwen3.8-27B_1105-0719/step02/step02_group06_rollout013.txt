**Approach: Projected subgradient / multiplicative weights for minimax**

The idea is:

The key challenge is that the smooth max approach with all lags is O(n²) per gradient evaluation (because we need to compute all 2n-1 correlations). But if we only use active lags, it's O(n * |active|) which is much better.

            beta = max(50, 1000.0 / count_near_max)
        
        # Compute objective and gradient (only over active lags)
        val, grad_q = smooth_obj_grad_sym(q, n_full, active, beta)
        
        # Projected gradient step with backtracking
        lr_new = lr
        for _ in range(20):
            q_trial = np.clip(q - lr_new * grad_q, 0, 1)
            h_trial = build_sym_h(q_trial)
            val_trial, _ = smooth_obj_grad_sym(q_trial, n_full, active, beta)
            if val_trial < val - 0.01 * lr_new * np.sum(np.abs(grad_q) * (q - np.clip(q - lr_new * grad_q, 0, 1))):
                # Sufficient decrease (Armijo)
                break
            lr_new *= 0.5
        
        q = np.clip(q - lr_new * grad_q, 0, 1)
        lr = min(2.0 * lr_new, 5.0)
        
        # Track best
        h = build_sym_h(q)
        c5 = eval_score(h)
        if c5 < best_c5:
            best_c5 = c5
            best_q = q.copy()
    
    return best_q, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def eval_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def corr_raw(h):
    return np.correlate(h, 1.0 - h, mode='full')

def build_sym_h(q):
    m = len(q)
    h = np.empty(2 * m)
    h[:m] = q
    h[m:] = 1.0 - q[::-1]
    return h

def sym_from_h(h):
    n = len(h)
    m = n // 2
    q = np.empty(m)
    for i in range(m):
        q[i] = 0.5 * (h[i] + (1.0 - h[n - 1 - i]))
    return np.clip(q, 0.0, 1.0)

def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.5, 1.5
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(x + mid, 0.0, 1.0).sum()
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    h = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        interior = (h > 1e-10) & (h < 1.0 - 1e-10)
        if interior.sum() > 0:
            h[interior] += diff / interior.sum()
    return np.clip(h, 0.0, 1.0)

def corr_val_k(h, k):
    n = len(h)
    d = k - (n - 1)
    i_lo = max(0, -d)
    i_hi = min(n - 1, n - 1 - d)
    if i_lo > i_hi:
        return 0.0
    return float(np.sum(h[i_lo:i_hi+1] * (1.0 - h[i_lo + d:i_hi + 1 + d])))

def corr_val_k_vec(h, k, idx):
    n = len(h)
    d = k - (n - 1)
    i_lo = max(0, -d)
    i_hi = min(n - 1, n - 1 - d)
    if i_lo > i_hi:
        return 0.0
    return np.sum(h[idx] * (1.0 - h[idx + d]))

def corr_grad_k(h, k):
    n = len(h)
    d = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -d)
    i_hi = min(n - 1, n - 1 - d)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx] += (1.0 - h[idx + d])
    grad[idx + d] += (-h[idx])
    return grad

def select_active(corrs_scaled, margin_frac=0.03, neighbor_pad=3, max_k=60):
    n = len(corrs_scaled)
    max_val = corrs_scaled.max()
    threshold = max_val - max(1e-8, margin_frac * max_val)
    active = set(np.where(corrs_scaled >= threshold)[0].tolist())
    padded = set()
    for k in active:
        for d in range(-neighbor_pad, neighbor_pad + 1):
            if 0 <= k + d < n:
                padded.add(k + d)
    active_list = sorted(padded)
    if len(active_list) > max_k:
        scored = sorted([(corrs_scaled[k], k) for k in active_list], reverse=True)
        active_list = sorted([k for _, k in scored[:max_k]])
    return active_list

def smooth_obj_grad_sym(q, n_full, active_k, beta):
    m = len(q)
    h = build_sym_h(q)
    dx = 2.0 / n_full
    corrs_a = np.zeros(len(active_k))
    for idx, k in enumerate(active_k):
        corrs_a[idx] = corr_val_k(h, k) * dx
    max_c = corrs_a.max()
    shifted = corrs_a - max_c
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(len(active_k)) / len(active_k)
    else:
        w /= w_sum
    val = max_c + np.log(w_sum) / beta
    grad_h = np.zeros(n_full)
    for idx, k in enumerate(active_k):
        if w[idx] < 1e-14:
            continue
        grad_h += w[idx] * corr_grad_k(h, k)
    grad_q = (grad_h[:m] - grad_h[m:][::-1]) * dx
    return val, grad_q

def epig_sym_con_fun(x, k, n_full):
    q = x[:n_full // 2]
    t = x[n_full // 2]
    h = build_sym_h(q)
    dx = 2.0 / n_full
    return t - corr_val_k(h, k) * dx

def epig_sym_con_jac(x, k, n_full):
    m = n_full // 2
    q = x[:m]
    h = build_sym_h(q)
    dx = 2.0 / n_full
    g = corr_grad_k(h, k)
    j = np.zeros(m + 1)
    j[:m] = -dx * (g[:m] - g[m:][::-1])
    j[m] = 1.0
    return j

def epig_sym_obj(x):
    return x[len(x) - 1]

def epig_sym_obj_grad(x):
    n = len(x)
    g = np.zeros(n)
    g[n - 1] = 1.0
    return g

def solve_epigraph_sym(q_center, n_full, active_k, maxiter=80, ftol=1e-12):
    m = n_full // 2
    h_c = build_sym_h(q_center)
    dx = 2.0 / n_full
    corrs_c = corr_raw(h_c) * dx
    t0_val = corrs_c.max() + 2e-8
    x0 = np.zeros(m + 1)
    x0[:m] = q_center
    x0[m] = t0_val
    bounds = [(0.0, 1.0)] * m + [(0.0, 2.0)]
    constraints = []
    for k in active_k:
        constraints.append({'type': 'ineq',
                           'fun': lambda x, kk=k: epig_sym_con_fun(x, kk, n_full),
                           'jac': lambda x, kk=k: epig_sym_con_jac(x, kk, n_full)})
    try:
        res = minimize(epig_sym_obj, x0, jac=epig_sym_obj_grad,
                      method='SLSQP', bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': ftol})
        q_out = np.clip(res.x[:m], 0.0, 1.0)
    except Exception:
        q_out = q_center.copy()
    h_out = build_sym_h(q_out)
    return q_out, h_out, eval_score(h_out)

        corrs_new = corr_raw(h_new) * dx_f
        max_act = corrs_new[active].max() if len(active) > 0 else 0
        hidden = np.where(corrs_new > max_act + 3e-7)[0]
        if len(hidden) > 0:
            active = sorted(set(active + list(hidden)))
            if len(active) > 80:
                active = select_active(corrs_new, margin_frac=0.01, neighbor_pad=3, max_k=80)
            h_f = h_new
            continue
        break
    return h_f, best_c5

def transfer_to_n(h_old, n_new):
    x_old = (np.arange(len(h_old)) + 0.5) * 2.0 / len(h_old)
    x_new = (np.arange(n_new) + 0.5) * 2.0 / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def rem():
        return budget_s - (time.time() - t0)

    best_h = None
    best_c5 = float('inf')

    try:
        h_parent = np.array(initial_h_values, dtype=float).copy().reshape(-1)
    except (NameError, AttributeError):
        h_parent = None

    if h_parent is not None and len(h_parent) > 10:
        n_p = len(h_parent)
        if abs(np.sum(h_parent) - n_p / 2.0) > 0.5:
            h_parent = project_feasible(h_parent, n_p / 2.0)
        h_parent = np.clip(h_parent, 0.0, 1.0)
        best_h = h_parent.copy()
        best_c5 = eval_score(h_parent)
    else:
        m_p = 63
        x = np.linspace(0, 1, m_p)
        q_p = (0.5 + 0.3 * np.sin(2 * np.pi * x)).clip(0, 1)
        best_h = build_sym_h(q_p)
        best_c5 = eval_score(best_h)

    n_cur = len(best_h)

    # Phase 1: Efficient projected gradient on symmetric formulation
    if n_cur % 2 == 0 and rem() > 200:
        q_init = sym_from_h(best_h)
        tl1 = time.time() + min(300, rem() * 0.30)
        q_opt, c5_s = smooth_sym_pg(q_init, n_cur, tl1, max_iter=8000)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = build_sym_h(q_opt).copy()

    # Phase 2: Multiplicative weights refinement
    if n_cur % 2 == 0 and rem() > 200:
        q_init = sym_from_h(best_h)
        tl2 = time.time() + min(250, rem() * 0.25)
        q_opt, c5_s = multiplicative_weights_sym(q_init, n_cur, tl2, max_iter=5000)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = build_sym_h(q_opt).copy()

    # Phase 3: Epigraph SLSQP refinement (symmetric)
    if n_cur % 2 == 0 and rem() > 150:
        tl3 = time.time() + min(200, rem() * 0.25)
        q_init = sym_from_h(best_h)
        q_opt, c5_s = do_epig_sym_loop(best_h, tl3, max_rounds=20, per_solve_maxiter=70)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = build_sym_h(q_opt).copy()

    # Phase 4: Full-variable epigraph (break symmetry)
    if rem() > 120:
        tl4 = time.time() + min(120, rem() * 0.30)
        h_opt, c5_f = do_epig_full_loop(best_h, tl4, max_rounds=12, per_solve_maxiter=50)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_opt.copy()
            n_cur = len(best_h)

    # Phase 5: Multi-resolution exploration
    n_base = len(best_h)
    n_candidates = set()
    for delta in [-6, -4, -3, -2, -1, 1, 2, 3, 4, 6, 8, 12, 16]:
        nc = n_base + delta
        if 48 <= nc <= 200:
            n_candidates.add(nc)
    if n_base // 2 >= 32:
        n_candidates.add(n_base // 2)
    if n_base * 3 // 2 <= 200:
        n_candidates.add(n_base * 3 // 2)
    n_candidates = sorted(n_candidates)

    for n_try in n_candidates:
        if rem() < 60:
            break
        h_try = transfer_to_n(best_h, n_try)
        n_try = len(h_try)
        if n_try % 2 == 0:
            q_try = sym_from_h(h_try)
            if rem() > 100:
                tl5 = time.time() + min(70, rem() * 0.35)
                q_o, c5_o = smooth_sym_pg(q_try, n_try, tl5, max_iter=4000)
                if c5_o < best_c5:
                    best_c5 = c5_o
                    best_h = build_sym_h(q_o).copy()
                    n_base = len(best_h)
            if rem() > 60:
                tl5b = time.time() + min(50, rem() * 0.30)
                q_try = sym_from_h(best_h) if len(best_h) == n_try else sym_from_h(h_try)
                tl5c = time.time() + min(40, rem() * 0.25)
                q_o, c5_o = do_epig_sym_loop(build_sym_h(q_try), tl5c, max_rounds=12, per_solve_maxiter=50)
                if c5_o < best_c5:
                    best_c5 = c5_o
                    best_h = build_sym_h(q_o).copy()
                    n_base = len(best_h)
        else:
            if rem() > 70:
                tl5d = time.time() + min(60, rem() * 0.30)
                h_o, c5_o = do_epig_full_loop(h_try, tl5d, max_rounds=8, per_solve_maxiter=40)
                if c5_o < best_c5:
                    best_c5 = c5_o
                    best_h = h_o.copy()
                    n_base = len(best_h)

    # Phase 6: Random restarts with diverse initialization
    n_restarts = min(8, max(2, int(rem() / 100)))
    for i in range(n_restarts):
        if rem() < 70:
            break
        n_r = n_base if i % 3 == 0 else max(50, min(180, n_base + rng.choice([-6, -3, 3, 6])))
        if n_r % 2 != 0:
            n_r += 1
        n_r = max(48, min(180, n_r))
        m_r = n_r // 2
        if i % 4 == 0:
            x_r = np.linspace(0, 1, m_r, endpoint=False)
            q_r = (0.5 + 0.25 * np.sin(2 * np.pi * 3 * x_r + rng.randn()) * 0.3).clip(0, 1)
        elif i % 4 == 1:
            q_r = np.sort(rng.uniform(0, 1, m_r))
            q_r = project_feasible(q_r, m_r * 0.5)
        elif i % 4 == 2:
            q_r = 0.5 + 0.4 * rng.randn(m_r)
            q_r = np.clip(q_r, 0, 1)
        else:
            x_r = np.linspace(0, 1, m_r, endpoint=False)
            freqs = rng.choice([2, 3, 4, 5], size=3)
            amps = rng.uniform(0.1, 0.3, size=3)
            phs = rng.uniform(0, 2*np.pi, size=3)
            q_r = 0.5 + sum(amps[j] * np.sin(2*np.pi*freqs[j]*x_r + phs[j]) for j in range(3))
            q_r = np.clip(q_r, 0, 1)
        if n_r % 2 == 0:
            tl6 = time.time() + min(50, rem() * 0.40)
            q_o, c5_o = smooth_sym_pg(q_r, n_r, tl6, max_iter=3000)
            if c5_o < best_c5:
                best_c5 = c5_o
                best_h = build_sym_h(q_o).copy()
            if c5_o < best_c5 + 0.003 and rem() > 40:
                tl6b = time.time() + min(30, rem() * 0.30)
                q_o2, c5_o2 = do_epig_sym_loop(build_sym_h(q_o), tl6b, max_rounds=8, per_solve_maxiter=40)
                if c5_o2 < best_c5:
                    best_c5 = c5_o2
                    best_h = build_sym_h(q_o2).copy()

    # Phase 7: Final intensive polish
    if rem() > 30 and best_h is not None:
        n_fin = len(best_h)
        if n_fin % 2 == 0:
            q_fin = sym_from_h(best_h)
            if rem() > 50:
                tl7 = time.time() + min(40, rem() * 0.40)
                q_o, c5_o = smooth_sym_pg(q_fin, n_fin, tl7, max_iter=5000)
                if c5_o < best_c5:
                    best_c5 = c5_o
                    best_h = build_sym_h(q_o).copy()
            if rem() > 25:
                tl7b = time.time() + min(25, rem() * 0.50)
                q_fin = sym_from_h(best_h)
                q_o, c5_o = do_epig_sym_loop(best_h, tl7b, max_rounds=15, per_solve_maxiter=80)
                if c5_o < best_c5:
                    best_c5 = c5_o
                    best_h = build_sym_h(q_o).copy()
        else:
            if rem() > 30:
                tl7c = time.time() + min(25, rem() * 0.50)
                h_o, c5_o = do_epig_full_loop(best_h, tl7c, max_rounds=8, per_solve_maxiter=50)
                if c5_o < best_c5:
                    best_c5 = c5_o
                    best_h = h_o.copy()

    # Final: high-iteration epigraph on best
    if rem() > 15 and best_h is not None:
        n_fin = len(best_h)
        if n_fin % 2 == 0:
            q_fin = sym_from_h(best_h)
            h_chk = build_sym_h(q_fin)
            corrs = corr_raw(h_chk) * (2.0 / n_fin)
            active = select_active(corrs, margin_frac=0.01, neighbor_pad=3, max_k=45)
            q_o, h_o, c5_o = solve_epigraph_sym(q_fin, n_fin, active, 100, 1e-13)
            h_o = build_sym_h(np.clip(q_o, 0, 1))
            c5_o = eval_score(h_o)
            if c5_o < best_c5:
                best_c5 = c5_o
                best_h = h_o

    # Validate and return
    if best_h is None:
        m_fb = 63
        x_fb = np.linspace(0, 1, m_fb)
        q_fb = (0.5 + 0.3 * np.sin(2 * np.pi * x_fb)).clip(0, 1)
        best_h = build_sym_h(q_fb)

    best_h = np.asarray(best_h, dtype=float).reshape(-1)
    n_out = len(best_h)
    best_h = project_feasible(best_h, n_out / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = eval_score(best_h)
    return best_h, best_c5, n_out
