            grad += wm * dx * (1.0 - 2.0 * h)

    sum_err = h.sum() - n / 2.0
    grad += 2.0 * lam * sum_err * np.ones(n) / n
    return loss, grad

def interp_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    fx = np.concatenate([[0.0], xs, [2.0]])
    fh = np.concatenate([[h[0]], h, [h[-1]]])
    hn = np.interp(xt, fx, fh)
    return repair(hn)

def symm(h):
    return (h + h[::-1]) / 2.0

def slsqp_solve(h0, maxiter=300, tol=1e-12, topk=None):
    n = len(h0)
    dx = 2.0 / n
    M = 2 * n - 1
    p0 = profile_arr(h0)
    c0 = p0.max()
    t0 = c0 + 1e-6

    def obj(x):
        return x[-1]

    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g

    def eq_f(x):
        return x[:-1].sum() - n / 2.0

    def eq_j(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g

    # Determine active constraints
    if topk is None:
        active = np.arange(M)
    else:
        idx = np.argsort(p0)[::-1][:topk]
        active = idx

    def ineq_f(x):
        h = x[:-1]
        t = x[-1]
        p = profile_arr(h)
        return t - p[active]

    def ineq_j(x):
        h = x[:-1]
        G = np.zeros((len(active), n))
        for ri, m in enumerate(active):
            d = m - (n - 1)
            if d > 0:
                G[ri, d:n] = dx * (1.0 - h[0:n-d])
                G[ri, 0:n-d] = -dx * h[d:n]
            elif d < 0:
                dp = -d
                G[ri, 0:n-dp] = dx * (1.0 - h[dp:n])
                G[ri, dp:n] = -dx * h[0:n-dp]
            else:
                G[ri, :] = dx * (1.0 - 2.0 * h)
        J = np.zeros((len(active), n + 1))
        J[:, :-1] = -G
        J[:, -1] = 1.0
        return J

    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 1.0)]
    cons = [
        {'type': 'eq', 'fun': eq_f, 'jac': eq_j},
        {'type': 'ineq', 'fun': ineq_f, 'jac': ineq_j},
    ]
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': tol})
        if res.x is not None and np.isfinite(res.x).all():
            h_new = repair(res.x[:-1])
            c_new = exact_score(h_new)
            if c_new <= c0 + 1e-9:
                return h_new
    except Exception:
        pass
    return None

def bfgs_soft_solve(h0, beta_start=100, beta_end=5000, n_outer=8, lam=80.0, maxiter=200):
    """Continuation method: increase beta and re-optimize."""
    n = len(h0)
    h = h0.copy()
    c = exact_score(h)
    betas = np.geomspace(beta_start, beta_end, n_outer)
    for beta in betas:
        def obj(x):
            h_r = np.clip(x, 0, 1)
            l, _ = soft_loss_grad(h_r, beta, lam)
            return l

        def jac(x):
            h_r = np.clip(x, 0, 1)
            _, g = soft_loss_grad(h_r, beta, lam)
            return g

        bounds = [(0.0, 1.0)] * n
        try:
            res = minimize(obj, h, method='L-BFGS-B', jac=jac,
                           bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-8})
            if res.x is not None and np.isfinite(res.x).all():
                h_new = repair(res.x)
                c_new = exact_score(h_new)
                if c_new < c:
                    h = h_new
                    c = c_new
        except Exception:
            pass
    return h, c

def binary_init_from_h(h, rng):
    """Create binary vector from continuous h by thresholding."""
    n = len(h)
    bits = (h > 0.5).astype(float)
    # Fix sum to be n/2
    n_ones = int(bits.sum())
    target = n // 2
    if n_ones > target:
        # Remove excess ones (those closest to 0.5)
        ones = np.where(bits == 1)[0]
        order = np.argsort(h[ones])
        for i in range(n_ones - target):
            bits[ones[order[i]]] = 0
    elif n_ones < target:
        zeros = np.where(bits == 0)[0]
        order = np.argsort(-h[zeros])
        for i in range(target - n_ones):
            bits[zeros[order[i]]] = 1
    return bits

def binary_local_search(bits, rng, max_iters=3000):
    """Local search: swap random 1s with 0s to reduce max overlap."""
    n = len(bits)
    bits = bits.copy()
    best = exact_score(bits)
    ones = np.where(bits == 1)[0]
    zeros = np.where(bits == 0)[0]

    for it in range(max_iters):
        improved = False
        n_try = min(30, len(ones) * len(zeros))
        for _ in range(n_try):
            a = ones[rng.randint(len(ones))]
            b = zeros[rng.randint(len(zeros))]
            bits[a] = 0
            bits[b] = 1
            s = exact_score(bits)
            if s < best - 1e-15:
                best = s
                improved = True
                ones = np.where(bits == 1)[0]
                zeros = np.where(bits == 0)[0]
            else:
                bits[a] = 1
                bits[b] = 0
        if not improved:
            # Try larger moves: flip 2 pairs
            improved2 = False
            for _ in range(10):
                if len(ones) < 2 or len(zeros) < 2:
                    break
                a1 = ones[rng.randint(len(ones))]
                a2 = ones[rng.randint(len(ones))]
                b1 = zeros[rng.randint(len(zeros))]
                b2 = zeros[rng.randint(len(zeros))]
                if a1 == a2 or b1 == b2:
                    continue
                bits[a1] = 0
                bits[a2] = 0
                bits[b1] = 1
                bits[b2] = 1
                s = exact_score(bits)
                if s < best - 1e-15:
                    best = s
                    improved2 = True
                    ones = np.where(bits == 1)[0]
                    zeros = np.where(bits == 0)[0]
                else:
                    bits[a1] = 1
                    bits[a2] = 1
                    bits[b1] = 0
                    bits[b2] = 0
            if not improved2:
                break
    return bits, best

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.perf_counter()

    def time_left():
        return budget_s - (time.perf_counter() - t0)

    rng = np.random.RandomState(seed)

    # Get parent
    parent = None
    try:
        parent = np.array(initial_h_values, dtype=np.float64)
    except (NameError, Exception):
        pass

    if parent is None or len(parent) < 4:
        parent = np.full(70, 0.5)

    n = len(parent)
    parent = np.clip(parent, 0, 1)
    parent = repair(parent)

    # Symmetrize
    h_sym = repair(symm(parent))
    c_sym = exact_score(h_sym)
    c_par = exact_score(parent)

    best_h = parent.copy() if c_par <= c_sym else h_sym.copy()
    best_c = min(c_par, c_sym)

    # Stage 1: BFGS with log-sum-exp continuation on parent dimension
    if time_left() > 100:
        for beta_s in [50, 100, 200]:
            if time_left() < 60:
                break
            h_b, c_b = bfgs_soft_solve(best_h, beta_start=beta_s, beta_end=beta_s*50,
                                        n_outer=6, lam=60.0, maxiter=150)
            if c_b < best_c - 1e-12:
                best_h = h_b.copy()
                best_c = c_b

    # Stage 2: SLSQP with top-k active constraints
    if time_left() > 80:
        for k in [20, 40, None]:
            if time_left() < 40:
                break
            h_s = slsqp_solve(best_h, maxiter=250, tol=1e-12, topk=k)
            if h_s is not None:
                c_s = exact_score(h_s)
                if c_s < best_c - 1e-12:
                    best_h = h_s.copy()
                    best_c = c_s

    # Stage 3: Binary local search
    if time_left() > 120:
        bits = binary_init_from_h(best_h, rng)
        bits, c_bin = binary_local_search(bits, rng, max_iters=2000)
        if c_bin < best_c:
            best_h = bits.copy()
            best_c = c_bin
            # Polish the binary solution with continuous optimization
            if time_left() > 60:
                h_b, c_b = bfgs_soft_solve(best_h, beta_start=50, beta_end=2000,
                                            n_outer=5, lam=60.0, maxiter=100)
                if c_b < best_c - 1e-12:
                    best_h = h_b.copy()
                    best_c = c_b
                h_s = slsqp_solve(best_h, maxiter=300, tol=1e-12, topk=30)
                if h_s is not None:
                    c_s = exact_score(h_s)
                    if c_s < best_c - 1e-12:
                        best_h = s_s if False else h_s.copy()
                        best_c = c_s

    # Stage 4: Dimension exploration
    if time_left() > 150:
        n_cur = len(best_h)
        dims = []
        for d in [n_cur-6, n_cur-4, n_cur-2, n_cur+2, n_cur+4, n_cur+6, n_cur+8, n_cur-8, n_cur+12]:
            if 30 <= d <= 130 and d != n_cur and d % 2 == 0:
                dims.append(d)
        dims = sorted(set(dims), key=lambda x: abs(x - n_cur))

        for nd in dims[:6]:
            if time_left() < 50:
                break
            h_int = interp_to_n(best_h, nd)
            # Try BFGS
            h_b, c_b = bfgs_soft_solve(h_int, beta_start=50, beta_end=3000,
                                        n_outer=5, lam=60.0, maxiter=120)
            c_try = min(c_b, exact_score(h_int))
            h_try = h_b if c_b <= exact_score(h_int) else h_int
            if c_try < best_c - 1e-12:
                best_h = h_try.copy()
                best_c = c_try
            # Try SLSQP on the better of the two
            h_s = slsqp_solve(h_try, maxiter=200, tol=1e-12, topk=25)
            if h_s is not None:
                c_s = exact_score(h_s)
                if c_s < best_c - 1e-12:
                    best_h = h_s.copy()
                    best_c = c_s

    # Stage 5: Perturbation + re-optimize
    if time_left() > 100:
        n_b = len(best_h)
        for eps in [0.01, 0.02, 0.04, 0.06]:
            if time_left() < 40:
                break
            z = rng.randn(n_b)
            z -= z.mean()
            h_pert = repair(symm(best_h + eps * z))
            c_p = exact_score(h_pert)
            if c_p < best_c:
                h_b, c_b = bfgs_soft_solve(h_pert, beta_start=50, beta_end=2000,
                                            n_outer=5, lam=60.0, maxiter=100)
                if c_b < best_c - 1e-12:
                    best_h = h_b.copy()
                    best_c = c_b
            h_s = slsqp_solve(h_pert, maxiter=150, tol=1e-12, topk=20)
            if h_s is not None:
                c_s = exact_score(h_s)
                if c_s < best_c - 1e-12:
                    best_h = h_s.copy()
                    best_c = c_s

    # Stage 6: Final BFGS + SLSQP polish
    if time_left() > 50:
        h_b, c_b = bfgs_soft_solve(best_h, beta_start=200, beta_end=10000,
                                    n_outer=6, lam=100.0, maxiter=200)
        if c_b < best_c - 1e-13:
            best_h = h_b.copy()
            best_c = c_b

    if time_left() > 40:
        h_s = slsqp_solve(best_h, maxiter=400, tol=1e-13, topk=30)
        if h_s is not None:
            c_s = exact_score(h_s)
            if c_s < best_c - 1e-13:
                best_h = h_s.copy()
                best_c = c_s

    # Stage 7: Very fine local search with small gradient steps
    if time_left() > 30:
        n_f = len(best_h)
        dx_f = 2.0 / n_f
        p_f = profile_arr(best_h)
        c_f = p_f.max()
        active = np.where(p_f >= c_f - 0.003)[0]
        if len(active) > 15:
            top_idx = np.argsort(p_f[active])[-15:]
            active = active[top_idx]
        # Build gradient for active set
        grad = np.zeros(n_f)
        w_sum = 0.0
        for m in active:
            d = m - (n_f - 1)
            wm = max(p_f[m] - (c_f - 0.003), 0.001)
            w_sum += wm
            if d > 0:
                grad[d:n_f] += wm * dx_f * (1.0 - best_h[0:n_f-d])
                grad[0:n_f-d] += wm * dx_f * (-best_h[d:n_f])
            elif d < 0:
                dp = -d
                grad[0:n_f-dp] += wm * dx_f * (1.0 - best_h[dp:n_f])
                grad[dp:n_f] += wm * dx_f * (-best_h[0:n_f-dp])
            else:
                grad += wm * dx_f * (1.0 - 2.0 * best_h)
        grad /= (w_sum + 1e-30)
        grad -= grad.mean()
        gn = np.linalg.norm(grad)
        if gn > 1e-14:
            grad /= gn
            for a in [0.002, 0.001, 0.0005, 0.0002, 0.0001]:
                h_try = repair(best_h - a * grad)
                s = exact_score(h_try)
                if s < best_c - 1e-14:
                    best_h = h_try
                    best_c = s
                    break

    # Final validation
    best_h = repair(best_h)
    final_c = exact_score(best_h)

    try:
        ev = evaluate_erdos_solution(best_h)
        if isinstance(ev, (float, int)):
            final_c = min(final_c, float(ev))
        elif isinstance(ev, tuple) and len(ev) >= 2:
            final_c = min(final_c, float(ev[0]))
    except Exception:
        pass

    return best_h, final_c, len(best_h)
