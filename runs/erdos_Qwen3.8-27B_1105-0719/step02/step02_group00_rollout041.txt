        return h.copy()

        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff

            idx = order[i]
            add = min(remaining, capacity[idx])
            h[idx] += add
            remaining -= add
            if remaining <= 1e-12:

    else:

            idx = int(np.argmax(h))
