    c5 = corrs.max()
    
    # Active set: constraints within tol of max
    active = np.where(corrs >= c5 - tol)[0]

    G = np.zeros((len(active), n))

        offset = m - (n-1)

import numpy as np
from scipy.optimize import minimize, linprog
import time


def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
        else:
            h += diff / n
    return np.clip(h, 0.0, 1.0)


def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def c5_of_h(h, dx):
    return compute_corrs(h).max() * dx


def grad_active(h, active_indices):
    n = len(h)
    G = np.zeros((len(active_indices), n))
    for idx, m in enumerate(active_indices):
        offset = m - (n - 1)
        lo1 = max(0, -offset)
        hi1 = min(n, n - offset)
        if hi1 > lo1:
            G[idx, lo1:hi1] += 1.0 - h[lo1 + offset:hi1 + offset]
        lo2 = max(0, offset)
        hi2 = min(n, n + offset)
        if hi2 > lo2:
            G[idx, lo2:hi2] -= h[lo2 - offset:hi2 - offset]
    return G


def smooth_max_obj(h, dx, beta):
    corrs = compute_corrs(h)
    m = corrs.max()
    return (m + np.log(np.sum(np.exp(beta * (corrs - m)))) / beta) * dx


def smooth_max_grad(h, dx, beta):
    n = len(h)
    corrs = compute_corrs(h)
    m = corrs.max()
    shifted = corrs - m
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    w /= w_sum
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        offset = k - (n - 1)
        lo1 = max(0, -offset)
        hi1 = min(n, n - offset)
        if hi1 > lo1:
            grad[lo1:hi1] += w[k] * (1.0 - h[lo1 + offset:hi1 + offset])
        lo2 = max(0, offset)
        hi2 = min(n, n + offset)
        if hi2 > lo2:
            grad[lo2:hi2] -= w[k] * h[lo2 - offset:hi2 - offset]
    return grad * dx


def local_slsqp(h0, dx, target_sum, beta, maxiter):
    n = len(h0)

    def obj(x):
        return smooth_max_obj(x, dx, beta)

    def obj_grad(x):
        return smooth_max_grad(x, dx, beta)

    def cons_fun(x):
        return x.sum() - target_sum

    def cons_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    bounds = [(0.0, 1.0)] * n

    try:
        res = minimize(obj, h0, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        h_out = project_feasible(res.x, target_sum)
    except Exception:
        h_out = project_feasible(h0, target_sum)
    return h_out, c5_of_h(h_out, dx)

    target_sum = h.sum()
