def eval_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))


def corr_values(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)


def project_to_target(x, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(x - mid, 0.0, 1.0).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-13:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.any():
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)


def build_sym(q, n):
    m = n // 2
    h = np.empty(n)
    h[:m] = q
    h[m:] = 1.0 - q[::-1]
    return h


def to_sym_q(h, n):
    m = n // 2
    return 0.5 * (h[:m] + 1.0 - h[n - 1:m - 1:-1])


def grad_h_lag(h, d, n):
    g = np.zeros(n)
    if d >= 0:
        if n - d > 0:
            g[:n - d] += 1.0 - h[d:]
            g[d:] -= h[:n - d]
    else:
        p = -d
        if n - p > 0:
            g[p:] += 1.0 - h[:n - p]
            g[:n - p] -= h[p:]
    return g


def grad_q_lag(h, d, n):
    m = n // 2
    gh = grad_h_lag(h, d, n)
    return gh[:m] - gh[m:][::-1]


def select_active(c_vals, margin=0.004, pad=2, max_k=None):
    mx = c_vals.max()
    thresh = mx - margin
    idx = np.where(c_vals >= thresh)[0]
    if len(idx) == 0:
        idx = np.array([int(np.argmax(c_vals))])
    if max_k is not None and len(idx) > max_k:
        order = np.argsort(c_vals[idx])[::-1][:max_k]
        idx = idx[order]
    idx_set = set(idx.tolist())
    for a in list(idx_set):
        for delta in range(-pad, pad + 1):
            k = a + delta
            if 0 <= k < len(c_vals):
                idx_set.add(k)
    return np.sort(np.array(sorted(idx_set)))


def ls_step(q, n, active_k, target, ridge=0.01):
    m = n // 2
    h = build_sym(q, n)
    G = np.zeros((len(active_k), m))
    for i, k in enumerate(active_k):
        d = k - (n - 1)
        G[i] = grad_q_lag(h, d, n)
    c_act = corr_values(h)[active_k]
    residual = target - c_act
    A = G.T @ G + ridge * np.eye(m)
    b = G.T @ residual
    try:
        delta = np.linalg.solve(A, b)
    except np.linalg.LinAlgError:
        try:
            delta, _, _, _ = np.linalg.lstsq(A, b, rcond=None)
        except Exception:
            delta = np.zeros(m)
    if not np.all(np.isfinite(delta)):
        delta = np.zeros(m)
    return delta

    best_c = c_cur

def interp_to(h, n_new):
    n_old = len(h)

            best_h = h_new.copy()
            best_c5 = c_val

    # Phase 2: Full-variable refinement on parent n
    if time_left() > 80:
        h_f, c_f = opt_full_ls(best_h, len(best_h), max_iter=150, deadline=budget_s - 40)
        if c_f < best_c5:
            best_h = h_f.copy()
            best_c5 = c_f

    # Phase 3: Try different resolutions
    n_base = len(best_h)
    n_candidates = []
    for delta_n in [-12, -8, -6, -4, -2, 2, 4, 6, 8, 10, 14, 18]:
        nc = n_base + delta_n
        if nc >= 32 and nc % 2 == 0:
            n_candidates.append(nc)
    # Also try halved and 3/4
    for frac in [0.5, 0.6, 0.7, 0.75, 1.25, 1.5]:
        nc = int(n_base * frac)
        if nc >= 32 and nc % 2 == 0:
            n_candidates.append(nc)
    n_candidates = sorted(set(n_candidates))

    for n_try in n_candidates:
        if time_left() < 40:
            break
        if n_try == n_base:
            continue
        h_try = interp_to(best_h, n_try)
        if n_try % 2 == 0:
            q_try = to_sym_q(h_try, n_try)
            h_try = build_sym(q_try, n_try)
            c_try = eval_c5(h_try)
            if c_try < best_c5:
                best_h = h_try.copy()
                best_c5 = c_try
                q_try = to_sym_q(best_h, n_try)

            # Optimize at this resolution
            q_opt, c_opt = opt_sym_ls(q_try, n_try, max_iter=150, deadline=budget_s - 40)
            h_opt = build_sym(q_opt, n_try)
            if c_opt < best_c5:
                best_h = h_opt.copy()
                best_c5 = c_opt

            # Full refinement at this n
            if time_left() > 50:
                h_ff, c_ff = opt_full_ls(h_opt, n_try, max_iter=80, deadline=budget_s - 25)
                if c_ff < best_c5:
                    best_h = h_ff.copy()
                    best_c5 = c_ff

    # Phase 4: Random restarts
    n_cur = len(best_h)
    max_restarts = min(20, max(5, int(time_left() / 20)))
    for trial in range(max_restarts):
        if time_left() < 30:
            break
        m_r = n_cur // 2
        t_r = np.linspace(0, 1, m_r, endpoint=False)
        q_r = 0.5 + 0.15 * np.sin(2 * np.pi * rng.randint(1, 6) * t_r + rng.uniform(0, 6.28))
        q_r += 0.08 * np.sin(2 * np.pi * rng.randint(2, 8) * t_r + rng.uniform(0, 6.28))
        q_r = np.clip(q_r, 0.0, 1.0)
        h_r = build_sym(q_r, n_cur)
        c_r = eval_c5(h_r)
        if c_r < best_c5:
            best_h = h_r.copy()
            best_c5 = c_r

        q_opt, c_opt = opt_sym_ls(q_r, n_cur, max_iter=100, deadline=budget_s - 30)
        h_opt = build_sym(q_opt, n_cur)
        if c_opt < best_c5:
            best_h = h_opt.copy()
            best_c5 = c_opt

    # Phase 5: Final polish with aggressive active set
    if time_left() > 20 and best_h is not None:
        n_fin = len(best_h)
        if n_fin % 2 == 0:
            q_fin = to_sym_q(best_h, n_fin)
            c_vals_fin = corr_values(best_h)
            mx_fin = c_vals_fin.max()
            # Include all lags within 2% of max
            active_fin = select_active(c_vals_fin, margin=0.02, pad=4, max_k=60)
            target_fin = mx_fin - 0.001
            for ridge_try in [0.001, 0.005, 0.01, 0.05, 0.1]:
                delta_fin = ls_step(q_fin, n_fin, active_fin, target_fin, ridge=ridge_try)
                q_new, c_new = line_search_sym(q_fin, n_fin, delta_fin, eval_c5(build_sym(q_fin, n_fin)))
                if c_new < best_c5:
                    best_h = build_sym(q_new, n_fin).copy()
                    best_c5 = c_new
                    q_fin = q_new

        # Final full refinement
        if time_left() > 10:
            h_fin, c_fin = opt_full_ls(best_h, len(best_h), max_iter=80, deadline=budget_s - 5)
            if c_fin < best_c5:
                best_h = h_fin.copy()
                best_c5 = c_fin

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_target(best_h, len(best_h) / 2.0)
    best_c5 = eval_c5(best_h)

    return best_h, best_c5, len(best_h)
