def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corrs = np.correlate(h, 1.0 - h, mode='full')
    return corrs.max() * dx, corrs


def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def grad_corr_d(h, d, n):
    g = np.zeros(n)
    if d >= 0:
        if d < n:
            g[d:] += (1.0 - h[:n - d])
            g[:n - d] += (-h[d:])
    else:
        dd = -d
        if dd < n:
            g[:n - dd] += (1.0 - h[dd:])
            g[dd:] += (-h[:n - dd])
    return g


def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)


def smooth_max_value_grad(h, beta, dx):
    corrs = compute_corrs(h)
    val = logsumexp(beta * corrs) / beta * dx

    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(len(corrs)) / len(corrs)

        w /= w_sum
    G = np.zeros(len(h))

            continue
        d = k - (len(h) - 1)
        G += w[k] * grad_corr_d(h, d, len(h))

        step_accepted = False
        for ls_iter in range(40):
            h_trial = np.clip(h + alpha * direction, 0.0, 1.0)
            h_trial = project_feasible(h_trial, target_sum)
            c5_trial = exact_score(h_trial)[0]
            if c5_trial < current_max - 1e-14 * max(1, current_max):
                step_accepted = True
                break
            alpha *= 0.5
        
        if not step_accepted:
            # Try a small step in the steepest descent direction
            # Use the subgradient of the max
            w_sum = 1.0 / len(top_idx)
            G_max = np.zeros(n)
            for k in top_idx:
                d = k - (n - 1)
                G_max += w_sum * grad_corr_d(h, d, n) * dx
            h_trial = project_feasible(h - 0.01 * G_max, target_sum)
            c5_trial = exact_score(h_trial)[0]
            if c5_trial < current_max - 1e-15:
                h = h_trial
                current_max = c5_trial
            else:
                # Stuck - try random perturbation
                h = project_feasible(h + 0.02 * (np.random.randn(n) - 0.5), target_sum)
                current_max = exact_score(h)[0]
            continue
        
        h = h + alpha * direction
        h = project_feasible(h, target_sum)
        current_max = exact_score(h)[0]
        
        if current_max < best_c5 - 1e-14:
            best_c5 = current_max
            best_h = h.copy()
        
        # Early stop if we're at a very good point
        if best_c5 < 0.38080:
            break
    
    return best_h, best_c5


def resample_h(h0, n_new):
    """Resample h to a new grid size."""
    x_orig = np.linspace(0.0, 2.0, len(h0))
    x_new = np.linspace(0.0, 2.0, n_new)
    h_new = np.interp(x_new, x_orig, h0)
    h_new = project_feasible(h_new, n_new / 2.0)
    return h_new


def full_smooth_polish(h0, n, target_sum, betas, time_budget):
    """Multi-beta L-BFGS polish in full space."""
    t_start = time.time()
    h = h0.copy()
    best_h = h.copy()
    best_c5 = exact_score(h)[0]
    
    for beta in betas:
        if time.time() - t_start > time_budget:
            break
        h_new = smooth_lbfgs(h, n, beta, target_sum, maxiter=3000)
        c5_new = exact_score(h_new)[0]
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
    
    return best_h, best_c5


def q_space_optimize(h0, n, time_budget):
    """Optimize in the symmetric q-space."""
    t_start = time.time()
    m = n // 2
    if n % 2 != 0:
        m = (n - 1) // 2
    target_sum = n / 2.0
    
    # Extract q from h (first half)
    if n % 2 == 0:
        q0 = h0[:m].copy()
    else:
        q0 = h0[:m].copy()
    
    def build_h(q):
        if n % 2 == 0:
            return np.concatenate([q, 1.0 - q[::-1]])
        else:
            return np.concatenate([q, [0.5], 1.0 - q[::-1]])
    
    def q_val_grad(x):
        h = build_h(np.clip(x, 0, 1))
        corrs = compute_corrs(h)
        beta_eff = 2000
        val = logsumexp(beta_eff * corrs) / beta_eff * (2.0 / n)
        shifted = corrs - corrs.max()
        w = np.exp(beta_eff * shifted)
        ws = w.sum()
        if ws < 1e-30:
            w = np.ones(len(corrs)) / len(corrs)
        else:
            w /= ws
        Gq = np.zeros(m)
        for k in range(len(corrs)):
            if w[k] < 1e-12:
                continue
            d = k - (n - 1)
            G = grad_corr_d(h, d, n)
            if n % 2 == 0:
                Gq += w[k] * (G[:m] - G[m:][::-1])
            else:
                Gq += w[k] * (G[:m] - G[m+1:][::-1])
        return val, Gq * (2.0 / n)
    
    q = q0.copy()
    best_q = q.copy()
    best_c5 = exact_score(build_h(q))[0]
    
    for _ in range(3):
        if time.time() - t_start > time_budget:
            break
        try:
            res = minimize(lambda x: q_val_grad(x)[0], q,
                         jac=lambda x: q_val_grad(x)[1],
                         method='L-BFGS-B',
                         bounds=[(0.0, 1.0)] * m,
                         options={'maxiter': 3000, 'ftol': 1e-16})
            q_new = np.clip(res.x, 0, 1)
            h_new = build_h(q_new)
            c5_new = exact_score(h_new)[0]
            if c5_new < best_c5 - 1e-15:
                best_c5 = c5_new
                best_q = q_new.copy()
                q = q_new
        except Exception:
            break
    
    return build_h(best_q), best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)
    
    def elapsed():
        return time.time() - t_start
    
    def remaining():
        return budget_s - elapsed()
    
    # Initialize from parent
    h_parent = np.array(initial_h_values, dtype=float)
    n = len(h_parent)
    target_sum = n / 2.0
    h = project_feasible(h_parent, target_sum)
    best_h = h.copy()
    best_c5 = exact_score(h)[0]
    
    # Phase 1: Full-space smooth L-BFGS (no symmetry assumption)
    # Use increasing beta for progressive focusing on the max
    for beta in [300, 800, 2000, 5000]:
        if remaining() < 60:
            break
        h_new = full_smooth_polish(h, n, target_sum, [beta], remaining() - 10)[0]
        c5_new = exact_score(h_new)[0]
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
    
    # Phase 2: Cutting plane in full space
    if remaining() > 150:
        h_cp, c5_cp = cutting_plane_full(h, n, min(300, remaining() - 30))
        if c5_cp < best_c5 - 1e-15:
            best_c5 = c5_cp
            best_h = h_cp.copy()
            h = h_cp
    
    # Phase 3: Symmetric q-space optimization (might find better structure)
    if remaining() > 100:
        h_q, c5_q = q_space_optimize(h, n, min(150, remaining() - 30))
        if c5_q < best_c5 - 1e-15:
            best_c5 = c5_q
            best_h = h_q.copy()
            h = h_q
    
    # Phase 4: More cutting plane from best
    if remaining() > 100:
        h_cp2, c5_cp2 = cutting_plane_full(best_h, n, min(200, remaining() - 30))
        if c5_cp2 < best_c5 - 1e-15:
            best_c5 = c5_cp2
            best_h = h_cp2.copy()
            h = h_cp2
    
    # Phase 5: Try different n values (downsample)
    n_candidates = [n, n - 2, n - 4, n - 6, n - 8, n - 10]
    for n_try in n_candidates:
        if remaining() < 80:
            break
        if n_try <= 0:
            continue
        ts_try = n_try / 2.0
        h_interp = resample_h(best_h, n_try)
        
        # Smooth optimize on this grid
        h_opt = full_smooth_polish(h_interp, n_try, ts_try, [2000, 5000], min(150, remaining() - 20))[0]
        c5_opt = exact_score(h_opt)[0]
        
        # Cutting plane
        if remaining() > 60:
            h_opt2, c5_opt2 = cutting_plane_full(h_opt, n_try, min(100, remaining() - 20))
            if c5_opt2 < c5_opt:
                h_opt = h_opt2
                c5_opt = c5_opt2
        
        if c5_opt < best_c5 - 1e-10:
            best_c5 = c5_opt
            best_h = h_opt.copy()
    
    # Phase 6: Final polish on the best found
    if remaining() > 40:
        n_final = len(best_h)
        ts_final = n_final / 2.0
        h_polished = full_smooth_polish(best_h, n_final, ts_final, [8000], remaining() - 10)[0]
        c5_polished = exact_score(h_polished)[0]
        if c5_polished < best_c5 - 1e-15:
            best_c5 = c5_polished
            best_h = h_polished.copy()
    
    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    n_out = len(best_h)
    ts_out = n_out / 2.0
    best_h = project_feasible(best_h, ts_out)
    final_score, _ = exact_score(best_h)
    best_c5 = min(best_c5, final_score)
    
    return best_h, best_c5, n_out
