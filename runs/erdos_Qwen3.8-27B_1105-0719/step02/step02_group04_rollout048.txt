```python
import numpy as np
from scipy.optimize import linprog
import time

def dx_of(n):
    return 2.0 / n

def corr_vals(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def eval_c5(h):
    return float(np.max(corr_vals(h)))

def corr_grad_row(h, d, n):
    """Gradient of correlation value at lag offset d w.r.t. h."""
    g = np.zeros(n)
    lo1 = max(0, d)
    hi1 = min(n - 1, n - 1 + d)
    if lo1 <= hi1:
        g[lo1:hi1 + 1] += 1.0 - h[lo1 - d:hi1 + 1 - d]
    lo2 = max(0, -d)
    hi2 = min(n - 1, n - 1 - d)
    if lo2 <= hi2:
        g[lo2:hi2 + 1] -= h[lo2 + d:hi2 + 1 + d]
    return g

def build_sym(q, n):
    m = n // 2
    h = np.empty(n)
    h[:m] = q
    h[m:] = 1.0 - q[::-1]
    return h

def to_sym_q(h, n):
    m = n // 2
    return 0.5 * (h[:m] + 1.0 - h[n - 1:m - 1:-1])

def sym_grad_row(h, d, n):
    """Gradient of correlation at lag d w.r.t. q (symmetric parameter)."""
    gh = corr_grad_row(h, d, n)
    m = n // 2
    return gh[:m] - gh[n - 1:m - 1:-1]

def select_active(C, c_max, margin, max_count=50):
    """Select active constraint indices."""
    thresh = c_max - margin
    idx = np.where(C >= thresh)[0]
    if len(idx) > max_count:
        # Keep only top max_count
        top_vals = np.partition(C, -max_count)[-max_count:]
        thresh = top_vals[0] if len(top_vals) > 0 else thresh
        idx = np.where(C >= thresh - 1e-15)[0]
    return idx

def slp_sym_step(q, n, active_idx, R, C):
    """One SLP step in symmetric mode. Returns (delta_q, t_pred) or None."""
    m = n // 2
    h = build_sym(q, n)
    
    # Build gradient matrix A (rows = active constraints, cols = m)
    A_rows = []
    b_vals = []
    for j in active_idx:
        d = j - (n - 1)
        g_q = sym_grad_row(h, d, n)
        A_rows.append(g_q)
        b_vals.append(-C[j])  # a_j · δ ≤ t - C[j] → a_j · δ - t ≤ -C[j]
    
    if not A_rows:
        return None
    
    A = np.array(A_rows)  # shape (K, m)
    b = np.array(b_vals)  # shape (K,)
    
    # LP: minimize t
    # Variables: [δ_0, ..., δ_{m-1}, t]
    # Objective: [0, ..., 0, 1]
    c_obj = np.zeros(m + 1)
    c_obj[-1] = 1.0
    
    # Inequality: A @ δ - t ≤ -C[j] for each j
    # Row: [A[j,:], -1] @ [δ, t] ≤ -C[j]
    A_ub = np.column_stack([A, -np.ones(len(A_rows))])
    b_ub = b  # -C[j]
    
    # Bounds
    lb = np.maximum(-R, -q)
    ub = np.minimum(R, 1.0 - q)
    bounds = list(zip(lb, ub))
    t_lower = 0.0
    t_upper = float(C.max()) + R * np.max(np.abs(A)) + 0.1
    bounds.append((t_lower, t_upper))
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.status != 0:
        return None
    
    delta = res.x[:m]
    t_pred = res.x[m]
    return delta, t_pred

def slp_full_step(h, n, active_idx, R, C):
    """One SLP step in full mode. Returns (delta_h, t_pred) or None."""
    # Build gradient matrix
    A_rows = []
    b_vals = []
    for j in active_idx:
        d = j - (n - 1)
        g_h = corr_grad_row(h, d, n)
        A_rows.append(g_h)
        b_vals.append(-C[j])
    
    if not A_rows:
        return None
    
    A = np.array(A_rows)
    b = np.array(b_vals)
    
    # Variables: [δ_0, ..., δ_{n-1}, t]
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    A_ub = np.column_stack([A, -np.ones(len(A_rows))])
    b_ub = b
    
    # Equality: sum(δ) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    lb = np.maximum(-R, -h)
    ub = np.minimum(R, 1.0 - h)
    bounds = list(zip(lb, ub))
    t_lower = 0.0
    t_upper = float(C.max()) + R * np.max(np.abs(A)) + 0.1
    bounds.append((t_lower, t_upper))

                  bounds=bounds, method='highs')
    
    if res.status != 0:
        return None
    
    delta = res.x[:n]
    t_pred = res.x[n]
    return delta, t_pred

def project_to_target(h, target_sum):
    """Project h to [0,1] with sum = target_sum using bisection."""
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0.0, 1.0).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    result = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    # Fix tiny residual
    diff = target_sum - result.sum()
    if abs(diff) > 1e-14:
        mask = (result > 1e-12) & (result < 1.0 - 1e-12)
        if mask.any():
            result[mask] += diff / mask.sum()
    return np.clip(result, 0.0, 1.0)

def interp_to(h, n_new):
    n_old = len(h)
    if n_new == n_old:
        return h.copy()
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h)
    return project_to_target(h_new, n_new / 2.0)

    m = n // 2
    h = build_sym(q, n)
    C = corr_vals(h)
    c_cur = float(C.max())
    R = R0
    margin = margin0
    no_progress = 0
    
    for it in range(max_iter):
        if time.time() - t0 > budget_s - reserve:
```

```python
import numpy as np
from scipy.optimize import linprog
import time


def dx_of(n):
    return 2.0 / n


def corr_vals(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)


def eval_c5(h):
    return float(np.max(corr_vals(h)))


def corr_grad_row(h, d, n):
    g = np.zeros(n)
    lo1 = max(0, d)
    hi1 = min(n - 1, n - 1 + d)
    if lo1 <= hi1:
        g[lo1:hi1 + 1] += 1.0 - h[lo1 - d:hi1 + 1 - d]
    lo2 = max(0, -d)
    hi2 = min(n - 1, n - 1 - d)
    if lo2 <= hi2:
        g[lo2:hi2 + 1] -= h[lo2 + d:hi2 + 1 + d]
    return g


def build_sym(q, n):
    m = n // 2
    h = np.empty(n)
    h[:m] = q
    h[m:] = 1.0 - q[::-1]
    return h


def to_sym_q(h, n):
    m = n // 2
    return 0.5 * (h[:m] + 1.0 - h[n - 1:m - 1:-1])


def sym_grad_row(h, d, n):
    gh = corr_grad_row(h, d, n)
    m = n // 2
    return gh[:m] - gh[n - 1:m - 1:-1]


def project_to_target(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0.0, 1.0).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    result = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - result.sum()
    if abs(diff) > 1e-14:
        mask = (result > 1e-12) & (result < 1.0 - 1e-12)
        if mask.any():
            result[mask] += diff / mask.sum()
    return np.clip(result, 0.0, 1.0)


def interp_to(h, n_new):
    n_old = len(h)
    if n_new == n_old:
        return h.copy()
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h)
    return project_to_target(h_new, n_new / 2.0)


def select_active(C, c_max, margin, max_count=50):
    thresh = c_max - margin
    idx = np.where(C >= thresh)[0]
    if len(idx) > max_count:
        sorted_idx = np.argsort(C)[::-1][:max_count]
        thresh = C[sorted_idx[-1]]
        idx = np.where(C >= thresh - 1e-15)[0]
    return np.sort(idx)


def solve_slp_sym(q, n, active_idx, R, C):
    m = n // 2
    h = build_sym(q, n)
    K = len(active_idx)
    if K == 0:
        return None
    A = np.empty((K, m))
    b = np.empty(K)
    for i, j in enumerate(active_idx):
        d = j - (n - 1)
        A[i] = sym_grad_row(h, d, n)
        b[i] = -C[j]
    c_obj = np.zeros(m + 1)
    c_obj[-1] = 1.0
    A_ub = np.empty((K, m + 1))
    A_ub[:, :m] = A
    A_ub[:, m] = -1.0
    b_ub = b
    lb = np.maximum(-R, -q)
    ub = np.minimum(R, 1.0 - q)
    bounds = list(zip(lb, ub))
    t_lo = 0.0
    t_hi = float(C.max()) + R * float(np.max(np.abs(A))) + 0.1
    bounds.append((t_lo, t_hi))
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    except Exception:
        return None
    if res.status != 0:
        return None
    return res.x[:m], res.x[m]
```
