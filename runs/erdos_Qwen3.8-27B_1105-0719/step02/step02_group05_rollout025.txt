def grad_corr_single(h, d, n):
    g = np.zeros(n)
    if d >= 0:
        if d < n:
            g[d:] += (1.0 - h[:n - d])
            g[:n - d] += (-h[d:])
    else:
        dd = -d
        if dd < n:
            g[:n - dd] += (1.0 - h[dd:])
            g[dd:] += (-h[:n - dd])
    return g

                while r2 == r1 or r3 == r1 or r2 == r3:
                    r2 = np.random.randint(pop_size)
                    r3 = np.random.randint(pop_size)

            F = np.random.uniform(*f_range)
            mutant = pop[r1] + F * (pop[r2] - pop[r3])
            mutant = np.clip(mutant, 0.0, 1.0)

            # Binomial crossover
            trial = pop[i].copy()
            j_rand = np.random.randint(m)
            for j in range(m):
                if np.random.rand() < cr or j == j_rand:
                    trial[j] = mutant[j]

            trial_score = score_q(trial, n)
            if trial_score <= scores[i]:
                pop[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best_score = trial_score
                    best_q = trial.copy()
                    best_idx = i
            # Occasionally perturb best to maintain diversity
            if gen % 20 == 15 and i == 0:
                noise = np.random.normal(0, 0.05, m)
                perturbed = np.clip(pop[best_idx] + noise, 0, 1)
                ps = score_q(perturbed, n)
                if ps < best_score:
                    best_score = ps
                    best_q = perturbed.copy()

    return best_q, best_score


def project_feasible_full(h, target_sum):
    """Project h to satisfy sum = target_sum and 0<=h<=1."""
    n = len(h)
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2
        h_proj = np.clip(h - mid, 0.0, 1.0)
        if h_proj.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h_proj.sum()
    if abs(diff) > 1e-12:
        mask = (h_proj > 1e-10) & (h_proj < 1.0 - 1e-10)
        if mask.sum() > 0:
            h_proj[mask] += diff / mask.sum()
    return np.clip(h_proj, 0.0, 1.0)


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)

    def remaining():
        return budget_s - (time.time() - t_start)

    def deadline_frac(frac):
        return t_start + frac * budget_s

    # Initialize from parent
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float)
    else:
        # Fallback: create a reasonable initial
        n_parent = 126
        h_parent = 0.5 + 0.3 * np.sin(np.linspace(0, 4 * np.pi, n_parent))
        h_parent = np.clip(h_parent, 0, 1)
        h_parent = project_feasible_full(h_parent, n_parent / 2.0)

    n_parent = len(h_parent)
    h_parent = np.clip(h_parent, 0.0, 1.0)
    h_parent = project_feasible_full(h_parent, n_parent / 2.0)

    best_h = h_parent.copy()
    best_c5 = score_h(best_h)

    # Convert to q-space (symmetric projection)
    # For the symmetric representation, we take the "best" symmetric q
    # by averaging h and its mirror complement
    if n_parent % 2 == 0:
        m = n_parent // 2
        q_parent = (h_parent[:m] + (1.0 - h_parent[m:][::-1])) / 2.0
        q_parent = np.clip(q_parent, 0.0, 1.0)
    else:
        # Make it even by trimming
        n_parent = n_parent - 1
        h_parent = h_parent[:n_parent]
        m = n_parent // 2
        q_parent = (h_parent[:m] + (1.0 - h_parent[m:][::-1])) / 2.0
        q_parent = np.clip(q_parent, 0.0, 1.0)

    # Phase 1: Quick coordinate descent on parent resolution
    n = n_parent
    deadline1 = deadline_frac(0.12)
    q_cd, c5_cd = coord_descent_q(q_parent, n, max_sweeps=15, deadline=deadline1)
    if c5_cd < best_c5:
        best_c5 = c5_cd
        best_h = build_h(q_cd, n)

    # Phase 2: Smooth optimization with increasing beta
    deadline2 = deadline_frac(0.25)
    q_current = q_cd.copy()
    for beta in [100, 300, 800, 2000]:
        if remaining() < 30:
            break
        q_smooth = smooth_q_optimize(q_current, n, beta, deadline=deadline2)
        s_smooth = score_q(q_smooth, n)
        if s_smooth < best_c5:
            best_c5 = s_smooth
            best_h = build_h(q_smooth, n)
            q_current = q_smooth.copy()

    # Coordinate descent after smooth
    q_cd2, c5_cd2 = coord_descent_q(q_current, n, max_sweeps=10, deadline=deadline2)
    if c5_cd2 < best_c5:
        best_c5 = c5_cd2
        best_h = build_h(q_cd2, n)
        q_current = q_cd2.copy()

    # Phase 3: Active-set epigraph
    deadline3 = deadline_frac(0.45)
    q_es, c5_es = active_set_search(q_current, n, max_rounds=6, deadline=deadline3)
    if c5_es < best_c5:
        best_c5 = c5_es
        best_h = build_h(q_es, n)
        q_current = q_es.copy()

    # More coordinate descent
    q_cd3, c5_cd3 = coord_descent_q(q_current, n, max_sweeps=15, deadline=deadline3)
    if c5_cd3 < best_c5:
        best_c5 = c5_cd3
        best_h = build_h(q_cd3, n)
        q_current = q_cd3.copy()

    # Phase 4: Multi-resolution search
    # Try various n values, interpolate best_h to each, optimize
    deadline4 = deadline_frac(0.80)
    target_ns = []
    # Generate even n values around the parent
    for n_try in range(60, 200, 4):
        if n_try % 2 == 0:
            target_ns.append(n_try)
    # Add some specific values
    for n_try in [64, 72, 80, 88, 96, 104, 112, 120, 128, 136, 144, 152, 160, 168, 176, 184, 192]:
        if n_try not in target_ns:
            target_ns.append(n_try)
    target_ns = sorted(set(target_ns))

    # Prioritize n values near the parent
    target_ns.sort(key=lambda x: abs(x - n_parent))

    h_best_for_interp = best_h.copy()

    for n_try in target_ns:
        if remaining() < 40:
            break
        # Interpolate
        h_interp = interpolate_h(h_best_for_interp, len(h_best_for_interp), n_try)
        h_interp = project_feasible_full(h_interp, n_try / 2.0)
        # Convert to symmetric q
        m_try = n_try // 2
        q_try = (h_interp[:m_try] + (1.0 - h_interp[m_try:][::-1])) / 2.0
        q_try = np.clip(q_try, 0.0, 1.0)
        h_try = build_h(q_try, n_try)

        # Quick coordinate descent
        q_opt, c5_opt = coord_descent_q(q_try, n_try, max_sweeps=8, deadline=deadline4)
        c5_opt_check = score_q(q_opt, n_try)
        if c5_opt_check < best_c5 - 1e-10:
            best_c5 = c5_opt_check
            best_h = build_h(q_opt, n_try)
            h_best_for_interp = best_h.copy()
            # Additional epigraph on this promising n
            if remaining() > 60:
                q_es2, c5_es2 = active_set_search(q_opt, n_try, max_rounds=4, deadline=deadline4)
                c5_es2_check = score_q(q_es2, n_try)
                if c5_es2_check < best_c5 - 1e-10:
                    best_c5 = c5_es2_check
                    best_h = build_h(q_es2, n_try)
                    h_best_for_interp = best_h.copy()

    # Phase 5: DE on small grids for global exploration
    deadline5 = deadline_frac(0.92)
    if remaining() > 60:
        for n_de in [48, 64, 80]:
            if remaining() < 40:
                break
            m_de = n_de // 2
            # Create seeds
            seeds = []
            h_de = interpolate_h(best_h, len(best_h), n_de)
            h_de = project_feasible_full(h_de, n_de / 2.0)
            m_de = n_de // 2
            q_de = (h_de[:m_de] + (1.0 - h_de[m_de:][::-1])) / 2.0
            seeds.append(np.clip(q_de, 0, 1))
            # Complement
            seeds.append(1.0 - np.clip(q_de, 0, 1))
            # Reversed
            seeds.append(np.clip(q_de)[::-1].copy())
            # Random
            for _ in range(10):
                seeds.append(np.random.rand(m_de) * 0.4 + 0.3)
            # Step-like
            for shift in [0.3, 0.4, 0.5, 0.6]:
                step_q = np.zeros(m_de)
                step_q[np.arange(m_de) < int(shift * m_de)] = 1.0
                step_q += 0.3 * np.random.randn(m_de)
                seeds.append(np.clip(step_q, 0, 1))

            pop_size = min(40, max(20, m_de))
            best_q_de, best_s_de = run_diff_evolution_q(
                seeds, n_de, pop_size, max_gens=80, deadline=deadline5)

            if best_s_de < best_c5 - 1e-8:
                # Transfer to finer grid
                h_de_best = build_h(best_q_de, n_de)
                for n_fine in [n_parent, 128, 160]:
                    if n_fine <= 0:
                        continue
                    h_fine = interpolate_h(h_de_best, n_de, n_fine)
                    h_fine = project_feasible_full(h_fine, n_fine / 2.0)
                    m_fine = n_fine // 2
                    q_fine = (h_fine[:m_fine] + (1.0 - h_fine[m_fine:][::-1])) / 2.0
                    q_fine = np.clip(q_fine, 0, 1)
                    q_opt_fine, c5_fine = coord_descent_q(q_fine, n_fine, max_sweeps=10, deadline=deadline5)
                    c5_fine_check = score_q(q_opt_fine, n_fine)
                    if c5_fine_check < best_c5 - 1e-10:
                        best_c5 = c5_fine_check
                        best_h = build_h(q_opt_fine, n_fine)

    # Phase 6: Final polish on the best candidate
    deadline6 = t_start + budget_s - 5.0
    n_final = len(best_h)
    if n_final % 2 != 0:
        n_final = n_final - 1
        best_h = best_h[:n_final]

    m_final = n_final // 2
    q_final = (best_h[:m_final] + (1.0 - best_h[m_final:][::-1])) / 2.0
    q_final = np.clip(q_final, 0.0, 1.0)

    # Final coordinate descent
    q_fp, c5_fp = coord_descent_q(q_final, n_final, max_sweeps=20, deadline=deadline6)
    c5_fp = score_q(q_fp, n_final)
    if c5_fp < best_c5:
        best_c5 = c5_fp
        best_h = build_h(q_fp, n_final)
        q_final = q_fp.copy()

    # Final epigraph
    if remaining() > 30:
        q_fs, c5_fs = active_set_search(q_final, n_final, max_rounds=5, deadline=deadline6)
        c5_fs = score_q(q_fs, n_final)
        if c5_fs < best_c5:
            best_c5 = c5_fs
            best_h = build_h(q_fs, n_final)

    # Final smooth polish
    if remaining() > 20:
        for beta in [3000, 8000]:
            if remaining() < 10:
                break
            q_sm = smooth_q_optimize(q_final, n_final, beta, deadline=deadline6)
            s_sm = score_q(q_sm, n_final)
            if s_sm < best_c5:
                best_c5 = s_sm
                best_h = build_h(q_sm, n_final)
                q_final = q_sm.copy()
            q_cd_f, c5_cd_f = coord_descent_q(q_final, n_final, max_sweeps=5, deadline=deadline6)
            if c5_cd_f < best_c5:
                best_c5 = c5_cd_f
                best_h = build_h(q_cd_f, n_final)
                q_final = q_cd_f.copy()

    # Final validation and cleanup
    best_h = np.clip(best_h, 0.0, 1.0)
    n_out = len(best_h)
    final_score = score_h(best_h)
    if final_score < best_c5:
        best_c5 = final_score

    # Ensure feasibility
    if n_out % 2 == 0:
        m_out = n_out // 2
        q_out = (best_h[:m_out] + (1.0 - best_h[m_out:][::-1])) / 2.0
        q_out = np.clip(q_out, 0.0, 1.0)
        best_h = build_h(q_out, n_out)
        final_score = score_h(best_h)
        if final_score < best_c5:
            best_c5 = final_score

    return best_h, best_c5, n_out
