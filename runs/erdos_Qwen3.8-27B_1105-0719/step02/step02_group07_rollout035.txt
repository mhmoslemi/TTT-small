from scipy.optimize import minimize, minimize_scalar
from scipy.special import logsumexp
import time


def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)


def corr_values(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx


def make_sym_h(q):
    m = len(q)
    n = 2 * m
    h = np.empty(n)
    h[:m] = q
    h[m:] = 1.0 - q[::-1]
    return h


def get_q_from_h(h):
    m = len(h) // 2
    q = 0.5 * (h[:m] + 1.0 - h[-1:-1-m:-1])
    return np.clip(q, 0.0, 1.0)


def softmax_obj(q, beta):
    h = make_sym_h(q)
    corrs = corr_values(h)
    return float(logsumexp(beta * corrs) / beta)


def softmax_grad(q, beta):
    m = len(q)
    n = 2 * m
    dx = 2.0 / n
    h = make_sym_h(q)
    corrs = corr_values(h)
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    wsum = w.sum()
    if wsum < 1e-30:
        w = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        w /= wsum

        wk = w[k]
        if wk < 1e-15:
            continue

        i_hi = min(n - 1, n - 1 - d)

        G[idx] += wk * (1.0 - h[idx + d])
        G[idx + d] += wk * (-h[idx])
    G *= dx
    grad_q = G[:m] - G[n-1:n-m-1:-1]

    """One step of gradient descent on the softmax approximation."""
    g = softmax_grad(q, beta)
    q_new = q - alpha * g
    return np.clip(q_new, 0.0, 1.0)


def coordinate_descent(q, n_sweeps=1):
    """Polish q using coordinate descent."""
    m = len(q)
    for sweep in range(n_sweeps):
        for i in range(m):
            # Minimize over q[i]
            def f(t):
                q_temp = q.copy()
                q_temp[i] = t
                h = make_sym_h(q_temp)
                return exact_score(h)
            res = minimize_scalar(f, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-10})
            q[i] = res.x
    return q


def subgradient_optimize(q0, beta, alpha_init, alpha_decay, max_iter):
    """Run subgradient/gradient descent with adaptive step."""
    q = q0.copy()
    best_q = q.copy()
    best_c5 = exact_score(make_sym_h(q))
    for it in range(max_iter):
        g = softmax_grad(q, beta)
        alpha = alpha_init * (alpha_decay ** it)
        q_new = q - alpha * g
        q_new = np.clip(q_new, 0.0, 1.0)
        c5 = exact_score(make_sym_h(q_new))
        if c5 < best_c5:
            best_c5 = c5
            best_q = q_new.copy()
        q = q_new
        if alpha < 1e-8:
            break
    return best_q, best_c5

def exact_score(h):

def corr_values(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx


def make_sym_h(q):
    m = len(q)
    h = np.empty(2 * m)
    h[:m] = q
    h[m:] = 1.0 - q[::-1]
    return h


def get_q_from_h(h):
    m = len(h) // 2
    q = 0.5 * (h[:m] + 1.0 - h[-1:-1-m:-1])
    return np.clip(q, 0.0, 1.0)


def softmax_obj_q(q, beta):
    h = make_sym_h(q)
    corrs = corr_values(h)
    return float(logsumexp(beta * corrs) / beta)


def softmax_grad_q(q, beta):
    m = len(q)
    n = 2 * m
    dx = 2.0 / n
    h = make_sym_h(q)
    corrs = corr_values(h)
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    wsum = w.sum()
    if wsum < 1e-30:
        w = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        w /= wsum
    G = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        d = k - (n - 1)
        i_lo = max(0, -d)
        i_hi = min(n - 1, n - 1 - d)
        if i_lo > i_hi:
            continue
        idx = np.arange(i_lo, i_hi + 1)
        G[idx] += wk * (1.0 - h[idx + d])
        G[idx + d] += wk * (-h[idx])
    G *= dx
    return G[:m] - G[n-1:n-m-1:-1]


def project_sum(arr, target_sum):
    """Project arr onto {0<=x<=1, sum(x)=target_sum} via bisection."""
    n = len(arr)
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(arr - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(arr - (lo + hi) * 0.5, 0.0, 1.0)
    # Fix any small residual
    diff = target_sum - h.sum()
    if abs(diff) > 1e-10:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h


def coordinate_descent_polish(q, n_sweeps=3, t_deadline=None):
    """Polish using exact 1D coordinate descent."""
    m = len(q)
    for sweep in range(n_sweeps):
        if t_deadline is not None and time.time() > t_deadline:
            break

                q_t[idx] = t
                return exact_score(make_sym_h(q_t))
            res = minimize_scalar(_f, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-10})
            q[i] = res.x
    return q
