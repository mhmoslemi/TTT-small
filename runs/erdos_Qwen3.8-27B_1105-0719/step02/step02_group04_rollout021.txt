Actually, one more thought. The SLP approach I described is essentially a cutting-plane method for the max of bilinear functions. It's a valid approach but might be slow to converge to high precision. An alternative is to use a smooth approximation like log-sum-exp:

The gradient of f wrt h is:

where p_j = exp(β*C[j]) / sum_k exp(β*C[k]) (softmax weights).

This is a weighted sum of the individual gradients, with weights proportional to how "active" each constraint is. For large β, only the most active constraints contribute.

This is actually what the previous code was doing with the penalty method, but the log-sum-exp is more principled.

However, the SLP approach has the advantage of being more aggressive in reducing the max (it directly targets the active constraints), while log-sum-exp with moderate β might "smooth over" the difference between active and inactive constraints.

    rhs = []

    # Objective: minimize t
    c_obj = np.zeros(m + 1)
    c_obj[m] = 1.0  # last variable is t

    if res.success:
        delta = res.x[:m]
        t_opt = res.x[m]
        return delta, t_opt
    return None, None

    q = q0.copy()
    h = _build_h(q, n)
    c5_cur = _eval_c5(h)
    C = _corr_full(h)
    R = R0
    no_prog = 0
    for it in range(max_iter):
        if time_limit is not None and t_start is not None:
            if time.time() - t_start > time_limit:
                break
        active = _select_active(C, margin_frac, pad=2)
        delta, t_opt = _slp_step_q(q, h, n, C, active, R)
        if delta is None:
            R *= 0.6
            no_prog += 1
            if no_prog > 12:
                break
            continue
        lam, c5_new = _line_search_q(q, n, delta, c5_cur)
        if lam > 0:
            q = np.clip(q + lam * delta, 0.0, 1.0)
            h = _build_h(q, n)
            c5_cur = c5_new
            C = _corr_full(h)
            no_prog = 0
            R = min(0.15, R * 1.15)
        else:
            no_prog += 1
            R *= 0.65
            if no_prog > 15:
                break
    return q, c5_cur


def _slp_optimize_full(h0, n, max_iter=40, R0=0.03, margin_frac=0.04, time_limit=None, t_start=None):
    h = h0.copy()
    c5_cur = _eval_c5(h)
    C = _corr_full(h)
    R = R0
    no_prog = 0
    for it in range(max_iter):
        if time_limit is not None and t_start is not None:
            if time.time() - t_start > time_limit:
                break
        active = _select_active(C, margin_frac, pad=2)
        delta, t_opt = _slp_step_full(h, n, C, active, R)
        if delta is None:
            R *= 0.6
            no_prog += 1
            if no_prog > 10:
                break
            continue
        lam, c5_new = _line_search_full(h, n, delta, c5_cur)
        if lam > 0:
            h = h + lam * delta
            h = _repair_box_sum(h, n / 2.0)
            c5_cur = c5_new
            C = _corr_full(h)
            no_prog = 0
            R = min(0.1, R * 1.1)
        else:
            no_prog += 1
            R *= 0.65
            if no_prog > 12:
                break
    return h, c5_cur


def _logsumexp_grad_step(h, n, beta, lr, max_iter=30, time_limit=None, t_start=None):
    m = n // 2
    q = _q_from_h(h)
    c5_cur = _eval_c5(h)
    for it in range(max_iter):
        if time_limit is not None and t_start is not None:
            if time.time() - t_start > time_limit:
                break
        h_cur = _build_h(q, n)
        C = _corr_full(h_cur)
        C_shifted = C - C.max()
        weights = np.exp(beta * C_shifted)
        wsum = weights.sum()
        if wsum <= 0:
            break
        p = weights / wsum
        gq = np.zeros(m)
        for j in range(len(C)):
            if p[j] > 1e-8:
                gq += p[j] * _grad_row_q(h_cur, j, n)
        q_new = np.clip(q - lr * gq, 0.0, 1.0)
        h_new = _build_h(q_new, n)
        c5_new = _eval_c5(h_new)
        if c5_new < c5_cur - 1e-12:
            q = q_new
            c5_cur = c5_new
        else:
            lr *= 0.5
            if lr < 1e-8:
                break
    h_final = _build_h(q, n)
    return h_final, _eval_c5(h_final)


def _random_q(n, rng):
    m = n // 2
    t = (np.arange(m) + 0.5) / m
    k1 = rng.randint(1, 7)
    k2 = rng.randint(2, 10)
    k3 = rng.randint(1, 5)
    p1 = rng.uniform(0, 2 * np.pi)
    p2 = rng.uniform(0, 2 * np.pi)
    p3 = rng.uniform(0, 2 * np.pi)
    a1 = rng.uniform(0.1, 0.35)
    a2 = rng.uniform(0.05, 0.2)
    a3 = rng.uniform(0.02, 0.1)
    base = rng.uniform(0.4, 0.6)
    q = base + a1 * np.sin(2 * np.pi * k1 * t + p1) + a2 * np.cos(2 * np.pi * k2 * t + p2) + a3 * np.sin(2 * np.pi * k3 * t + p3)
    return np.clip(q, 0.0, 1.0)


def _verify_grad(h, n, i):
    g = _grad_row_h(h, i, n)
    eps = 1e-7
    C0 = _corr_full(h)[i]
    max_err = 0.0
    for j in range(min(n, 10)):
        hp = h.copy()
        hp[j] = min(1.0, h[j] + eps)
        C1 = _corr_full(hp)[i]
        fd = (C1 - C0) / eps
        max_err = max(max_err, abs(g[j] - fd))
    return max_err


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    reserve = max(20.0, 0.04 * budget_s)

    def _time_left():
        return budget_s - (time.time() - t0)

    def _accept(h_cand, c5_cand):
        nonlocal best_h, best_c5
        if c5_cand < best_c5 - 1e-12:
            best_h = h_cand.copy()
            best_c5 = c5_cand
            return True
        return False

    # Initialize
    try:
        h_init = np.array(initial_h_values, dtype=float).reshape(-1)
        if h_init.size >= 10 and np.all(np.isfinite(h_init)):
            n_init = len(h_init)
            h_init = _repair_box_sum(h_init, n_init / 2.0)
            c5_init = _eval_c5(h_init)
            best_h = h_init.copy()
            best_c5 = c5_init
    except Exception:
        pass

    if best_h is None:
        n_def = 126
        m_def = n_def // 2
        t = (np.arange(m_def) + 0.5) / m_def
        q_def = 0.5 + 0.25 * np.sin(2 * np.pi * 3.0 * t) + 0.1 * np.cos(4 * np.pi * t)
        q_def = np.clip(q_def, 0.0, 1.0)
        best_h = _build_h(q_def, n_def)
        best_c5 = _eval_c5(best_h)

    n_base = len(best_h)

    # Verify gradient on a sample
    if n_base % 2 == 0:
        verify_err = _verify_grad(best_h, n_base, n_base)
        if verify_err > 1e-4:
            pass  # gradient might have orientation issue, but proceed

    # Phase 1: SLP in q-space at parent resolution
    if n_base % 2 == 0:
        q = _q_from_h(best_h)
        h_sym = _build_h(q, n_base)
        c5_sym = _eval_c5(h_sym)
        if c5_sym < best_c5:
            best_h = h_sym.copy()
            best_c5 = c5_sym
            q = _q_from_h(best_h)

        if _time_left() > reserve + 60:
            for mf in [0.02, 0.04, 0.06, 0.08]:
                if _time_left() < reserve + 30:
                    break
                for R0 in [0.03, 0.05, 0.08]:
                    if _time_left() < reserve + 15:
                        break
                    q_out, c5_out = _slp_optimize_q(q, n_base, max_iter=40, R0=R0,
                                                     margin_frac=mf, time_limit=None, t_start=None)
                    h_out = _build_h(q_out, n_base)
                    c5_out = _eval_c5(h_out)
                    if c5_out < best_c5:
                        best_h = h_out.copy()
                        best_c5 = c5_out
                        q = q_out.copy()
                    q = q_out.copy()

    # Phase 2: Log-sum-exp smoothing (escape local minima)
    if _time_left() > reserve + 40:
        for beta in [30.0, 60.0, 100.0, 150.0]:
            if _time_left() < reserve + 15:
                break
            h_lse, c5_lse = _logsumexp_grad_step(best_h, len(best_h), beta=beta, lr=0.02,
                                                  max_iter=25, time_limit=_time_left(), t_start=time.time())
            if c5_lse < best_c5:
                best_h = h_lse.copy()
                best_c5 = c5_lse

        # Re-run SLP after LSE
        if len(best_h) % 2 == 0:
            q = _q_from_h(best_h)
            q_out, c5_out = _slp_optimize_q(q, len(best_h), max_iter=30, R0=0.04,
                                             margin_frac=0.04, time_limit=None, t_start=None)
            h_out = _build_h(q_out, len(best_h))
            c5_out = _eval_c5(h_out)
            if c5_out < best_c5:
                best_h = h_out.copy()
                best_c5 = c5_out

    # Phase 3: Resolution hopping
    n_current = len(best_h)
    n_cands = set()
    for d in [-8, -6, -4, -3, -2, -1, 1, 2, 3, 4, 6, 8, 10, 12]:
        nc = n_current + d
        if nc >= 48 and nc % 2 == 0:
            n_cands.add(nc)
    n_cands.add(max(48, n_current // 2))
    n_cands.add(min(256, n_current + 16))
    if (n_current + 16) % 2 != 0:
        n_cands.add(n_current + 17)
    n_cands = sorted(n_cands)

    for n_try in n_cands:
        if _time_left() < reserve + 20:
            break
        if n_try == n_current:
            continue
        h_try = _interp_h(best_h, n_try)
        if n_try % 2 == 0:
            q_try = _q_from_h(h_try)
            h_try = _build_h(q_try, n_try)
            c5_try = _eval_c5(h_try)
            if c5_try < best_c5:
                best_h = h_try.copy()
                best_c5 = c5_try
            if _time_left() > reserve + 20:
                q_out, c5_out = _slp_optimize_q(q_try, n_try, max_iter=25, R0=0.04,
                                                 margin_frac=0.04, time_limit=None, t_start=None)
                h_out = _build_h(q_out, n_try)
                c5_out = _eval_c5(h_out)
                if c5_out < best_c5:
                    best_h = h_out.copy()
                    best_c5 = c5_out

    # Phase 4: Full-variable SLP (break symmetry)
    if _time_left() > reserve + 40:
        n_fv = len(best_h)
        h_fv = best_h.copy()
        h_fv = _repair_box_sum(h_fv, n_fv / 2.0)
        for mf in [0.03, 0.06]:
            if _time_left() < reserve + 15:
                break
            h_out, c5_out = _slp_optimize_full(h_fv, n_fv, max_iter=30, R0=0.02,
                                                margin_frac=mf, time_limit=None, t_start=None)
            if c5_out < best_c5:
                best_h = h_out.copy()
                best_c5 = c5_out
                h_fv = h_out.copy()

    # Phase 5: Random restarts + local refinement
    max_restarts = min(30, max(3, int(_time_left() / 20.0)))
    n_r = len(best_h)
    if n_r % 2 != 0:
        n_r -= 1
    for trial in range(max_restarts):
        if _time_left() < reserve + 8:
            break
        # Perturb current best
        q_cur = _q_from_h(best_h)
        noise = rng.randn(len(q_cur)) * 0.05
        q_pert = np.clip(q_cur + noise, 0.0, 1.0)
        h_pert = _build_h(q_pert, n_r)
        c5_pert = _eval_c5(h_pert)
        # Optimize from perturbed start
        q_out, c5_out = _slp_optimize_q(q_pert, n_r, max_iter=20, R0=0.05,
                                         margin_frac=0.05, time_limit=None, t_start=None)
        h_out = _build_h(q_out, n_r)
        c5_out = _eval_c5(h_out)
        if c5_out < best_c5:
            best_h = h_out.copy()
            best_c5 = c5_out
        # Also try fully random
        if trial % 3 == 0:
            q_rand = _random_q(n_r, rng)
            q_out2, c5_out2 = _slp_optimize_q(q_rand, n_r, max_iter=15, R0=0.08,
                                               margin_frac=0.08, time_limit=None, t_start=None)
            h_out2 = _build_h(q_out2, n_r)
            c5_out2 = _eval_c5(h_out2)
            if c5_out2 < best_c5:
                best_h = h_out2.copy()
                best_c5 = c5_out2

    # Phase 6: Final polish
    if _time_left() > 10 and best_h is not None:
        n_fin = len(best_h)
        if n_fin % 2 == 0:
            q_fin = _q_from_h(best_h)
            for R0 in [0.02, 0.01, 0.005]:
                if _time_left() < 5:
                    break
                q_fin2, c5_fin2 = _slp_optimize_q(q_fin, n_fin, max_iter=30, R0=R0,
                                                   margin_frac=0.02, time_limit=None, t_start=None)
                h_fin = _build_h(q_fin2, n_fin)
                c5_fin = _eval_c5(h_fin)
                if c5_fin < best_c5:
                    best_h = h_fin.copy()
                    best_c5 = c5_fin
                    q_fin = q_fin2.copy()

        # Final full-variable polish
        if _time_left() > 5:
            h_fin3, c5_fin3 = _slp_optimize_full(best_h, n_fin, max_iter=20, R0=0.01,
                                                  margin_frac=0.03, time_limit=None, t_start=None)
            if c5_fin3 < best_c5:
                best_h = h_fin3.copy()
                best_c5 = c5_fin3

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _repair_box_sum(best_h, len(best_h) / 2.0)
    best_c5 = _eval_c5(best_h)

    return best_h, best_c5, len(best_h)
