```python
def cutting_plane_polish(h, dx, target, n_iter):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = evaluate(h, dx)
    
    for it in range(n_iter):
        corrs = compute_corrs(h)  # shape (n,) after taking unique values
        max_corr = corrs.max()
        c5 = max_corr * dx
        
        if c5 < best_c5 - 1e-15:
            best_c5 = c5
            best_h = h.copy()
        
        # Find active set
        threshold = max_corr - 1e-6
        active = np.where(corrs >= threshold)[0]
        
        # Solve LP: find direction d such that:
        #   sum_j grad_d_j * d_j <= 0 for all active d
        #   sum(d) = 0
        #   -h[i] <= d[i] <= 1-h[i] (box constraints for feasibility)
        #   minimize -sum(d) ??? or just find feasible direction
        
        # Actually, we want to minimize the maximum correlation.
        # Linearized: minimize max_{active d} (corr_d + grad_d · Δ)
        # = minimize t
        #   s.t. grad_d · Δ + corr_d <= t for all active d
        #        sum Δ = 0
        #        -h[i]*alpha <= Δ[i] <= (1-h[i])*alpha for some step size alpha
        
        # Simpler: just try to reduce the max correlation
        # min grad_argmax · Δ (subgradient step)
        # or: min max_{active} grad_d · Δ
        
        # LP: min t
        #     s.t. grad_d · Δ <= t for all d in active
        #          -sum_{active} grad_d · Δ <= t (symmetric, since min could be negative)
        #          sum Δ = 0
        #          |Δ[i]| <= 1 (or scaled by step size)
        
        # Actually simplest: projected subgradient with line search
        # Use the subgradient for the max (uniform over active set)
        grad = np.mean([grad_corr_d(h, d, n) for d in active], axis=0) * dx
        
        # Project gradient onto constraint (remove mean component)
        grad_centered = grad - grad.mean()
        
        # Line search
        for alpha in [0.1, 0.05, 0.01, 0.005, 0.001]:
            h_new = h - alpha * grad_centered
            h_new = project(h_new, target)
            c5_new = evaluate(h_new, dx)
            if c5_new < c5 - 1e-15:
                h = h_new
                break
```

```python
def project_h(x, target):
    """Project x onto {h: 0<=h<=1, sum(h)=target}."""
    lo, hi = x.min() - 1.0, x.max() + 1.0
    for _ in range(60):

        if s > target:
            lo = mid
        else:

    corrs = np.correlate(h, 1.0 - h, mode='full')
    return corrs.max() * dx, corrs

    """Get the n unique correlation values (shifts 0 to n-1)."""
    full = np.correlate(h, 1.0 - h, mode='full')
    # full has 2n-1 elements, centered at index n-1
    # By symmetry, full[n-1+d] = full[n-1-d]
    # So unique values are at indices n-1, n, n+1, ..., 2n-2 (shifts 0, 1, ..., n-1)
    return full[n-1:]  # length n

    Returns array of shape (n, n) where grad[d, j] = d(corr_d)/d(h[j]).
    Actually for efficiency, just compute the weighted sum directly.

        for j in range(d, n):

        # Second term: -h[j+d] for j <= n-1-d
        for j in range(0, n-d):
```
