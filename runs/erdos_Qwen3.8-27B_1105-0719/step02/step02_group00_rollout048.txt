def exact_score(h):

def compute_profile(h):

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        # Need to add diff total
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx])
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-14:
                break
    else:
        # Need to remove -diff total
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-14:
                break
    # Final fix
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def build_gradient_matrix(h, n):
    """Build gradient matrix G where G[m, i] = dP_m/dh[i]."""
    dx = 2.0 / n
    M = 2 * n - 1
    G = np.zeros((M, n))
    for m in range(M):
        d = m - (n - 1)
        if d == 0:
            G[m, :] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            # P_d = dx * sum_{i=0}^{n-d-1} h[i+d]*(1-h[i])
            # dP_d/dh[i] for i < n-d: -dx*h[i+d]
            # dP_d/dh[i+d] for i < n-d: dx*(1-h[i])
            G[m, 0:n-d] += -dx * h[d:n]
            G[m, d:n] += dx * (1.0 - h[0:n-d])
        else:
            e = -d
            # P_{-e} = dx * sum_{i=0}^{n-e-1} h[i]*(1-h[i+e])

            # dP/dh[i+e] for i < n-e: -dx*h[i]
            G[m, 0:n-e] += dx * (1.0 - h[e:n])
            G[m, e:n] += -dx * h[0:n-e]
    return G

def solve_trust_region_cvx(h, active_indices, profile, trust_radius, rho=0.0):
    """Solve the trust-region linearized subproblem using cvxpy."""
    if not HAS_CVXPY:
        return None
    
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    
    # Build gradient matrix
    G = build_gradient_matrix(h, n)
    
    # Active constraints
    A = active_indices
    G_A = G[A, :]  # shape (|A|, n)
    p_A = profile[A]  # current values of active profile entries
    
    # Variables
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    
    # Constraints
    constraints = []
    
    # Trust region bounds
    lo = np.maximum(0.0, h - trust_radius)
    hi = np.minimum(1.0, h + trust_radius)
    constraints.append(h_var >= lo)
    constraints.append(h_var <= hi)
    
    # Sum constraint
    constraints.append(cp.sum(h_var) == n / 2.0)
    
    # Linearized active constraints: t >= p[m] + G[m,:]^T @ (h_var - h)
    # i.e., t - G[m,:]^T @ h_var >= p[m] - G[m,:]^T @ h
    rhs = p_A - G_A @ h  # = G_A @ h - G_A @ h + p_A - ... wait
    # p[m] + G[m,:]^T @ (h_var - h) <= t
    # G[m,:]^T @ h_var - t <= p[m] - G[m,:]^T @ h
    # Actually: t >= p[m] + G_A[m,:]^T @ (h_var - h)
    # => t - G_A[m,:]^T @ h_var >= p[m] - G_A[m,:]^T @ h
    # => G_A[m,:]^T @ h_var - t <= -(p[m] - G_A[m,:]^T @ h) = G_A[m,:]^T @ h - p[m]
    
    rhs_vec = G_A @ h - p_A
    constraints.append(G_A @ h_var - t_var <= rhs_vec)
    
    # Objective
    if rho > 0:
        # Add L1 regularization
        objective = cp.Minimize(t_var + rho * cp.norm1(h_var - h))
    else:
        objective = cp.Minimize(t_var)
    
    prob = cp.Problem(objective, constraints)
    
    try:

        try:

        except Exception:
            try:

            except Exception:
                return None
    
    if prob.status in ['optimal', 'optimal_inaccurate'] and h_var.value is not None:
        return np.asarray(h_var.value, dtype=np.float64).flatten()

    grad -= grad.mean()  # Remove mean to respect sum constraint

import numpy as np

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))


def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx


def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx])
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-14:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-14:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h


def build_gradient_matrix(h, n):
    dx = 2.0 / n
    M = 2 * n - 1
    G = np.zeros((M, n))
    for m in range(M):
        d = m - (n - 1)
        if d == 0:
            G[m, :] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            G[m, 0:n-d] += -dx * h[d:n]
            G[m, d:n] += dx * (1.0 - h[0:n-d])
        else:
            e = -d
            G[m, 0:n-e] += dx * (1.0 - h[e:n])
            G[m, e:n] += -dx * h[0:n-e]
    return G


def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)


def solve_lp_trust(h, profile, active, trust_radius, rho):
    n = len(h)
    dx = 2.0 / n
    G = build_gradient_matrix(h, n)
    G_A = G[active, :]
    p_A = profile[active]

    h_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = []

    lo = np.maximum(0.0, h - trust_radius)
    hi = np.minimum(1.0, h + trust_radius)
    constraints.append(h_var >= lo)
    constraints.append(h_var <= hi)
    constraints.append(cp.sum(h_var) == n / 2.0)

    rhs_vec = G_A @ h - p_A
    constraints.append(G_A @ h_var - t_var <= rhs_vec)

    if rho > 0:
        objective = cp.Minimize(t_var + rho * cp.norm1(h_var - h))
    else:
        objective = cp.Minimize(t_var)

    prob = cp.Problem(objective, constraints)
    solvers_to_try = ['CLARABEL', 'SCS', 'ECOS', 'GLPK', 'CBC']
    for solver_name in solvers_to_try:
        try:
            solver = getattr(cp, solver_name, None)
            if solver is None:
                continue
            prob.solve(solver=solver)
            if prob.status in ['optimal', 'optimal_inaccurate'] and h_var.value is not None:
                return np.asarray(h_var.value, dtype=np.float64).flatten()
        except Exception:
            continue
    return None


def active_grad_direction(h, beam, max_active):
    n = len(h)
    profile = compute_profile(h)
    c = np.max(profile)
    active = np.where(profile >= c - beam)[0]
    if len(active) == 0:
        active = np.array([np.argmax(profile)])
    if len(active) > max_active:
        top_idx = np.argsort(profile[active])[-max_active:]
        active = active[top_idx]
    G = build_gradient_matrix(h, n)
    G_A = G[active, :]
    w = profile[active] - (c - beam)
    w = np.maximum(w, 0.01)
    w = w / w.sum()
    grad = G_A.T @ w
    grad -= grad.mean()
    return grad, profile, c, active
