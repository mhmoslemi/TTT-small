import numpy as np

def eval_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def corr_values(h):
    """Return the full correlation array scaled by dx."""
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def build_sym(q, n):
    m = n // 2
    h = np.empty(n)
    h[:m] = q
    h[m:] = 1.0 - q[::-1]
    return h

def to_sym_q(h, n):
    m = n // 2
    # q[k] = h[k] for k < m
    # Also h[n-1-k] = 1 - q[k], so q[k] = 1 - h[n-1-k]
    # Average: q[k] = 0.5*(h[k] + 1 - h[n-1-k])
    return 0.5 * (h[:m] + 1.0 - h[n-1:m-1:-1])

def corr_grad_row(h, d, n):
    """Compute gradient row for lag d (scaled by dx=2/n)."""
    g = np.zeros(n)
    lo = max(0, d)
    hi = min(n-1, n-1+d)
    if lo <= hi:
        idx = np.arange(lo, hi+1)
        g[idx] += 1.0 - h[idx - d]
        g[idx - d] -= h[idx]
    return g * (2.0 / n)

def q_grad_row(h, d, n):
    """Gradient row for q variables (symmetric)."""
    m = n // 2
    gh = corr_grad_row(h, d, n)
    return gh[:m] - gh[n-1:m-1:-1]

def project_to_target(x, target_sum):
