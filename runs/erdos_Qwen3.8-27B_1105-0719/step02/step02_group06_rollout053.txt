from scipy.optimize import linprog
import time

def eval_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def corr_raw(h):
    return np.correlate(h, 1.0 - h, mode='full')

def corr_grad_full(h, k):
    """Gradient of correlation at lag k (0-indexed in full output) w.r.t. h."""
    n = len(h)
    d = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -d)
    i_hi = min(n - 1, n - 1 - d)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx] += (1.0 - h[idx + d])
    if d != 0:
        grad[idx + d] += (-h[idx])
    else:
        grad[idx] += (-h[idx])  # same indices, add both contributions
    # Wait, for d=0: h[idx]*(1-h[idx]), derivative is 1-2h[idx]
    # My formula gives: (1-h[idx]) + (-h[idx]) = 1-2h[idx]. Correct!
    return grad

def project_box_sum(x, target):
    """Project x onto {0<=x_i<=1, sum(x)=target}."""
    n = len(x)
    x = np.asarray(x, dtype=float).copy()
    x[~np.isfinite(x)] = 0.5
    x = np.clip(x, 0.0, 1.0)

    lo, hi = -1.5, 1.5

        mid = (lo + hi) / 2.0
        s = np.clip(x + mid, 0.0, 1.0).sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    h = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)
    
    # Fix residual
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        if residual > 0:
            order = np.argsort(h)  # ascending, raise smallest first
            for i in order:
                add = min(residual, 1.0 - h[i])

def tangent_lp_full(h, active_k, rho=0.15, t_upper=None):
    """Solve the tangent LP for full vector optimization.
    
    Returns (z, t) where z is the proposed new h and t is the objective value.
    """
    n = len(h)
    dx = 2.0 / n
    L = 2 * n - 1
    
    # Compute raw correlations and their values at active lags
    raw = corr_raw(h)
    
    # Build constraint matrix: for each active lag, grad_ell^T z - t <= grad_ell^T h - R_ell
    # In linprog form: -grad_ell^T z + t >= R_ell - grad_ell^T h
    # Or: grad_ell^T z - t <= grad_ell^T h - R_ell
    # linprog minimizes c^T x subject to A_ub @ x <= b_ub
    # Variable vector: [z_0, ..., z_{n-1}, t]
    
    if t_upper is None:
        t_upper = raw.max() + 1.0
    
    A_rows = []
    b_rows = []
    for k in active_k:
        if k < 0 or k >= L:
            continue
        g = corr_grad_full(h, k)
        rhs = float(g @ h) - raw[k]
        # Constraint: g @ z - t <= rhs
        # Row: [g[0],...,g[n-1], -1] @ [z,t] <= rhs
        row = np.zeros(n + 1)
        row[:n] = g
        row[n] = -1.0
        A_rows.append(row)
        b_rows.append(rhs)
    
    if not A_rows:
        return h.copy(), raw.max()
    
    A_ub = np.array(A_rows)
    b_ub = np.array(b_rows)
    
    # Bounds: [x0-rho, x0+rho] intersect [0,1] for z, [0, t_upper] for t
    lb = np.maximum(0.0, h - rho)
    ub = np.minimum(1.0, h + rho)
    bounds = list(zip(lb, ub)) + [(0.0, t_upper)]
    
    # Equality: sum(z) = n/2
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    # Objective: minimize t
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            z = res.x[:n]
            t_val = res.x[n]
            # Project z to ensure feasibility
            z = project_box_sum(z, n / 2.0)
            return z, t_val
    except Exception:
        pass
    
    return h.copy(), raw.max()

def transfer_line_search(h, a, b, n_evals=25, n_refine=15):
    """Line search on the transfer direction h[a]+=d, h[b]-=d.
    Returns best h and score."""

    lo = max(-h0[a], h0[b] - 1.0)
    hi = min(1.0 - h0[a], -h0[b])

        return h0, eval_score(h0)
    
    best_h = h0.copy()
    best_sc = eval_score(h0)
    
    # Grid search
    deltas = np.linspace(lo, hi, n_evals)
    for d in deltas:
        h[d:]  # no-op, just for clarity
        h[a] = h0[a] + d
        h[b] = h0[b] - d
        sc = eval_score(h)
        if sc < best_sc:
            best_sc = sc
            best_h = h.copy()
            best_d = d
    
    # Refine around best
    if best_sc < eval_score(h0) - 1e-12:
        margin = (hi - lo) / n_evals * 2
        lo2 = max(lo, best_d - margin)
        hi2 = min(hi, best_d + margin)

        for d in deltas2:
            h[a] = h0[a] + d
            h[b] = h0[b] - d
            sc = eval_score(h)
            if sc < best_sc:
                best_sc = sc
                best_h = h.copy()
    
    return best_h, best_sc

import numpy as np
from scipy.optimize import linprog
import time

def eval_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def corr_raw(h):
    return np.correlate(h, 1.0 - h, mode='full')

def corr_grad_full(h, k):
    n = len(h)
    d = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -d)
    i_hi = min(n - 1, n - 1 - d)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx] += (1.0 - h[idx + d])
    grad[idx + d] += (-h[idx])
    return grad

def build_sym_h(q):
    m = len(q)
    h = np.empty(2 * m)
    h[:m] = q
    h[m:] = 1.0 - q[::-1]
    return h

def sym_from_h(h):
    n = len(h)
    m = n // 2
    q = np.empty(m)
    for i in range(m):
        q[i] = 0.5 * (h[i] + (1.0 - h[n - 1 - i]))
    return np.clip(q, 0.0, 1.0)

def project_box_sum(x, target):
    n = len(x)
    x = np.asarray(x, dtype=float).copy()
    x[~np.isfinite(x)] = 0.5
    x = np.clip(x, 0.0, 1.0)
    lo, hi = -1.5, 1.5
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(x + mid, 0.0, 1.0).sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    h = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        if residual > 0:
            order = np.argsort(h)
            for i in order:
                add = min(residual, 1.0 - h[i])
                h[i] += add
                residual -= add
                if residual < 1e-14:
                    break
        else:
            order = np.argsort(-h)
            for i in order:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
                if residual > -1e-14:
                    break
    return np.clip(h, 0.0, 1.0)

def select_active(corrs, margin_frac=0.02, margin_abs=1e-8, pad=3, max_k=80):
    max_val = corrs.max()
    threshold = max_val - max(margin_abs, margin_frac * max_val)
    active = np.where(corrs >= threshold)[0]
    padded = set()
    for k in active:
        for d in range(-pad, pad + 1):
            if 0 <= k + d < len(corrs):
                padded.add(k + d)
    active_list = sorted(padded)
    if len(active_list) > max_k:
        scored = sorted([(corrs[k], k) for k in active_list], reverse=True)
        active_list = sorted([k for _, k in scored[:max_k]])
    return active_list

def tangent_lp_sym(q, active_k, rho=0.15):
    m = len(q)
    h = build_sym_h(q)
    n = 2 * m
    raw = corr_raw(h)
    A_rows = []
    b_rows = []
    for k in active_k:
        if k < 0 or k >= 2 * n - 1:
            continue

        g_q = g_full[:m] - g_full[n - 1 - m:m:-1] if m > 1 else np.array([g_full[0] - g_full[0]])
        rhs = float(g_q @ q) - raw[k]
        row = np.zeros(m + 1)
        row[:m] = g_q
        row[m] = -1.0
        A_rows.append(row)
        b_rows.append(rhs)
    if not A_rows:
        return q.copy()
    A_ub = np.array(A_rows)
    b_ub = np.array(b_rows)
    lb = np.maximum(0.0, q - rho)
    ub = np.minimum(1.0, q + rho)
    t_upper = raw.max() + 1.0
    bounds = list(zip(lb, ub)) + [(0.0, t_upper)]
    c = np.zeros(m + 1)
    c[m] = 1.0
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.status == 0:
            return np.clip(res.x[:m], 0.0, 1.0)
    except Exception:
        pass
    return q.copy()

def tangent_lp_full(h, active_k, rho=0.15):

    if (time.time() - t0) < budget_s - 60:
        h_cur = best_h.copy()
        for rnd in range(6):
            if (time.time() - t0) > budget_s - 20:
                break
            h_new, sc_new = greedy_transfer(h_cur, n_source=12, n_target=12, n_evals=14)
            if sc_new < best_c5 - 1e-13:
                best_c5 = sc_new
                best_h = h_new
                h_cur = h_new
            else:
                # Try with fewer candidates to explore different directions
                h_new2, sc_new2 = greedy_transfer(h_cur, n_source=20, n_target=20, n_evals=10)
                if sc_new2 < best_c5 - 1e-13:
                    best_c5 = sc_new2
                    best_h = h_new2
                    h_cur = h_new2
                else:
                    break

    # === Phase 4: Perturbation and re-optimization ===
    if (time.time() - t0) < budget_s - 80:
        n_pert = len(best_h)
        for attempt in range(4):
            if (time.time() - t0) > budget_s - 30:
                break
            # Random perturbation
            scale = 0.08 * (0.7 ** attempt)
            h_pert = best_h + scale * rng.randn(n_pert)
            h_pert = project_box_sum(h_pert, n_pert / 2.0)
            sc_pert = eval_score(h_pert)
            # Re-optimize from perturbed point
            if n_pert % 2 == 0:
                q_pert = sym_from_h(h_pert)
                if sc_pert < best_c5 + 0.02:
                    q_opt, c5_p, h_p = lp_optimize_sym(q_pert, n_pert, max_rounds=10, t_start=t0)
                    if c5_p < best_c5 - 1e-13:
                        best_c5 = c5_p
                        best_h = h_p
            else:
                if sc_pert < best_c5 + 0.02:
                    h_cur2 = h_pert
                    for _ in range(5):
                        if (time.time() - t0) > budget_s - 15:
                            break
                        raw2 = corr_raw(h_cur2)
                        active2 = select_active(raw2 * (2.0 / n_pert), margin_frac=0.02, pad=3, max_k=70)
                        z2 = tangent_lp_full(h_cur2, active2, rho=0.12)
                        sc2 = eval_score(z2)
                        if sc2 < best_c5 - 1e-13:
                            best_c5 = sc2
                            best_h = z2
                            h_cur2 = z2
                        else:
                            break
                    if eval_score(h_cur2) < best_c5:
                        best_c5 = eval_score(h_cur2)
                        best_h = h_cur2

    # === Phase 5: Multi-resolution exploration ===
    n_base = len(best_h)
    for delta in [-4, -2, 2, 4, 6, -6]:
        if (time.time() - t0) > budget_s - 30:
            break
        nc = n_base + delta
        if 72 <= nc <= 180:
            x_old = (np.arange(n_base) + 0.5) * (2.0 / n_base)
            x_new = (np.arange(nc) + 0.5) * (2.0 / nc)
            h_try = np.interp(x_new, x_old, best_h)
            h_try = project_box_sum(h_try, nc / 2.0)
            if nc % 2 == 0:
                q_try = sym_from_h(h_try)
                q_o, c5_o, h_o = lp_optimize_sym(q_try, nc, max_rounds=10, t_start=t0)
                if c5_o < best_c5:
                    best_c5 = c5_o
                    best_h = h_o
            else:
                h_cur3 = h_try
                for _ in range(5):
                    if (time.time() - t0) > budget_s - 15:
                        break
                    raw3 = corr_raw(h_cur3)
                    active3 = select_active(raw3 * (2.0 / nc), margin_frac=0.02, pad=3, max_k=70)
                    z3 = tangent_lp_full(h_cur3, active3, rho=0.10)
                    sc3 = eval_score(z3)
                    if sc3 < best_c5 - 1e-13:
                        best_c5 = sc3
                        best_h = z3
                        h_cur3 = z3
                    else:
                        break

    # === Phase 6: Final polish ===
    if (time.time() - t0) < budget_s - 10 and best_h is not None:
        n_fin = len(best_h)
        if n_fin % 2 == 0:
            q_fin = sym_from_h(best_h)
            for rho in [0.04, 0.02, 0.01]:
                if (time.time() - t0) > budget_s - 5:
                    break
                raw_f = corr_raw(best_h)
                corrs_f = raw_f * (2.0 / n_fin)
                active_f = select_active(corrs_f, margin_frac=0.015, pad=3, max_k=90)
                q_new = tangent_lp_sym(q_fin, active_f, rho=rho)
                h_new = project_box_sum(build_sym_h(q_new), n_fin / 2.0)
                sc = eval_score(h_new)
                if sc < best_c5 - 1e-14:
                    best_c5 = sc
                    best_h = h_new
                    q_fin = q_new
        else:
            for rho in [0.04, 0.02, 0.01]:
                if (time.time() - t0) > budget_s - 5:
                    break
                raw_f = corr_raw(best_h)
                corrs_f = raw_f * (2.0 / n_fin)
                active_f = select_active(corrs_f, margin_frac=0.015, pad=3, max_k=80)
                z = tangent_lp_full(best_h, active_f, rho=rho)
                sc = eval_score(z)
                if sc < best_c5 - 1e-14:
                    best_c5 = sc
                    best_h = z

    # Final validation
    if best_h is None:
        m_fb = 63
        x_fb = np.linspace(0, 1, m_fb)
        q_fb = (0.5 + 0.3 * np.sin(2 * np.pi * x_fb)).clip(0, 1)
        best_h = build_sym_h(q_fb)

    best_h = np.asarray(best_h, dtype=float).reshape(-1)
    best_h = project_box_sum(best_h, len(best_h) / 2.0)
    best_c5 = eval_score(best_h)

    return best_h, best_c5, len(best_h)
