from scipy.optimize import minimize
from scipy.special import logsumexp
import time

def project_feasible(x, target_sum):
    """Project x onto {0 <= h <= 1, sum(h) = target_sum}"""
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def c5_of_h(h, dx):
    return float(compute_corrs(h).max() * dx)

def grad_corr_k(h, k, n):

    grad = np.zeros(n)
    if offset == 0:
        return 1.0 - 2.0 * h
    i_lo = max(0, -offset)
    i_hi = min(n - 1, n - 1 - offset)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx + offset] += (1.0 - h[idx])
    grad[idx] += (-h[idx + offset])
    return grad

    """Gradient of correlation at index k with respect to h."""
    offset = k - (n - 1)

    i_hi = min(n - 1, n - 1 - offset)

    grad[idx + offset] += (1.0 - h[idx])
    grad[idx] += (-h[idx + offset])

import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time


def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h


def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def c5_of_h(h, dx):
    return float(compute_corrs(h).max() * dx)


def grad_corr_k(h, k, n):
    offset = k - (n - 1)
    if offset == 0:
        return 1.0 - 2.0 * h
    grad = np.zeros(n)
    i_lo = max(0, -offset)
    i_hi = min(n - 1, n - 1 - offset)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx + offset] += (1.0 - h[idx])
    grad[idx] += (-h[idx + offset])
    return grad


def make_pair_full(v, m):
    n = 2 * m
    h = np.empty(n)
    h[:m] = v
    h[m:] = 1.0 - v[m-1::-1]
    return h


def soft_max_obj_pair(v, m, dx, beta):
    h = make_pair_full(v, m)
    n = 2 * m
    corrs = compute_corrs(h) * dx
    return logsumexp(beta * corrs) / beta


def soft_max_grad_pair(v, m, dx, beta, thresh=1e-12):
    h = make_pair_full(v, m)
    n = 2 * m
    corrs = compute_corrs(h) * dx
    shift = corrs - corrs.max()
    w = np.exp(beta * shift)
    ws = w.sum()
    if ws < 1e-30:
        return np.zeros(m)
    w /= ws
    grad = np.zeros(m)
    for k in range(2 * n - 1):
        if w[k] > thresh:
            fg = grad_corr_k(h, k, n)
            grad += w[k] * (fg[:m] - fg[2 * m - 1::-1])
    return grad * dx


def soft_max_obj_full(h, dx, beta):
    corrs = compute_corrs(h) * dx
    return logsumexp(beta * corrs) / beta


def soft_max_grad_full(h, dx, beta, thresh=1e-12):
    n = len(h)
    corrs = compute_corrs(h) * dx
    shift = corrs - corrs.max()
    w = np.exp(beta * shift)
    ws = w.sum()
    if ws < 1e-30:
        return np.zeros(n)
    w /= ws
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] > thresh:
            grad += w[k] * grad_corr_k(h, k, n)
    return grad * dx


def slsqp_full(h0, dx, target_sum, beta, maxiter=500):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n

    def obj(x):
        return soft_max_obj_full(x, dx, beta)

    def jac(x):
        return soft_max_grad_full(x, dx, beta)

    def eq_fun(x):
        return x.sum() - target_sum

    def eq_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    try:
        res = minimize(obj, h0, jac=jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        h_out = project_feasible(res.x, target_sum)
    except Exception:
        h_out = project_feasible(h0, target_sum)
    return h_out, c5_of_h(h_out, dx)


def lbfgsb_pair(v0, m, dx, beta, maxiter=2000):
    n = 2 * m
    bounds = [(0.0, 1.0)] * m

    def obj(v):
        return soft_max_obj_pair(v, m, dx, beta)

    def jac(v):
        return soft_max_grad_pair(v, m, dx, beta)

    try:
        res = minimize(obj, v0, jac=jac, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-9,
                                'maxfun': 20000})
        v_opt = np.clip(res.x, 0.0, 1.0)
    except Exception:
        v_opt = np.clip(v0, 0.0, 1.0)
    h_opt = make_pair_full(v_opt, m)
    return h_opt, c5_of_h(h_opt, dx)


def interp_to_n(h_old, n_old, n_new):
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h_old)
    target = n_new / 2.0
    return project_feasible(h_new, target)

def active_lag_descent_pair(h0, m, dx, top_k=5, maxiter=2000):
    n = 2 * m
    v0 = h0[:m].copy()
    corrs = compute_corrs(h0)
    top_idx = np.argsort(corrs)[-top_k:]

    def obj(v):
        h = make_pair_full(v, m)
        c = compute_corrs(h) * dx
        return c[top_idx].sum()

    def jac(v):
        h = make_pair_full(v, m)
        g = np.zeros(m)
        for k in top_idx:
            fg = grad_corr_k(h, k, n)
            g += fg[:m] - fg[2 * m - 1::-1]
        return g * dx

    bounds = [(0.0, 1.0)] * m
    try:
        res = minimize(obj, v0, jac=jac, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-15})
        v_opt = np.clip(res.x, 0.0, 1.0)
    except Exception:
        v_opt = v0
    h_opt = make_pair_full(v_opt, m)
    return h_opt, c5_of_h(h_opt, dx)
