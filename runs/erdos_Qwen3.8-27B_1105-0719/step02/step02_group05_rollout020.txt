```python
import numpy as np
from scipy.optimize import minimize, minimize_scalar

# Helper functions

def build_h_from_q(q, n):
    m = n // 2
    if n % 2 == 0:
        return np.concatenate([q, 1.0 - q[::-1]])
    else:
        return np.concatenate([q, [0.5], 1.0 - q[::-1]])

def q_from_h(h, n):
    m = n // 2
    return h[:m].copy()

def exact_score_h(h):
    n = len(h)
    dx = 2.0 / n
    corrs = np.correlate(h, 1.0 - h, mode='full')
    return corrs.max() * dx

def exact_score_q(q, n):
    h = build_h_from_q(q, n)
    return exact_score_h(h)

def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def de_objective(q, n):

class QObjective:
    def __init__(self, n):
        self.n = n
    def __call__(self, q):
        return exact_score_q(q, self.n)

def nelder_mead(q0, n, time_limit, t_start):
    obj = QObjective(n)
    m = len(q0)
    # Initial simplex
    x0 = q0.copy()
    res = minimize(obj, x0, method='Nelder-Mead',
                   options={'maxiter': 2000, 'xatol': 1e-10, 'fatol': 1e-12,
                           'adaptive': True, 'initial_simplex': None,
                           'maxfev': 50000})
    h_res = build_h_from_q(np.clip(res.x, 0, 1), n)
    s = exact_score_h(h_res)
    return np.clip(res.x, 0, 1), s


# Coordinate descent
def coordinate_descent(q, n, time_limit, t_start, max_rounds=20):
    m = len(q)
    q = q.copy()
    current_score = exact_score_q(q, n)
    
    for rnd in range(max_rounds):
        if time.time() - t_start > time_limit:
            break
        improved = False
        for i in range(m):
            if time.time() - t_start > time_limit:
                break
            old_val = q[i]
            # Try a range of values
            candidates = [old_val]
            # Add perturbed values
            for delta in [0.02, -0.02, 0.05, -0.05, 0.1, -0.1, 0.01, -0.01]:
                v = old_val + delta
                if 0 <= v <= 1:
                    candidates.append(v)
            # Also try 0 and 1
            candidates.append(0.0)
            candidates.append(1.0)
            candidates.append(0.5)
            
            best_v = old_val
            best_s = current_score
            for v in candidates:
                q[i] = v
                s = exact_score_q(q, n)
                if s < best_s:
                    best_s = s
                    best_v = v
            q[i] = best_v
            if best_s < current_score - 1e-12:
                current_score = best_s
                improved = True
        
        if not improved:
            break
    
    return q, current_score


# Smooth log-sum-exp objective for gradient methods
class SmoothQObj:
    def __init__(self, n, beta):
        self.n = n
        self.beta = beta
        self.dx = 2.0 / n
    
    def __call__(self, q):
        h = build_h_from_q(q, self.n)
        corrs = np.correlate(h, 1.0 - h, mode='full')
        return logsumexp(self.beta * corrs) / self.beta * self.dx
    
    def gradient(self, q):
        n = self.n
        m = n // 2
        h = build_h_from_q(q, n)
        corrs = np.correlate(h, 1.0 - h, mode='full')
        shifted = corrs - corrs.max()
        w = np.exp(self.beta * shifted)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(len(corrs)) / len(corrs)
        else:
            w /= w_sum
        
        G = np.zeros(n)
        for k in range(len(corrs)):
            if w[k] < 1e-14:
                continue
            d = k - (n - 1)
            if d >= 0 and d < n:
                G[d:] += (1.0 - h[:n - d])
                G[:n - d] += (-h[d:])
            elif d < 0:
                dd = -d
                if dd < n:
                    G[:n - dd] += (1.0 - h[dd:])
                    G[dd:] += (-h[:n - dd])
        
        Gq = (G[:m] - G[m:][::-1]) * self.dx
        return Gq


def smooth_solve(q0, n, beta, time_limit, t_start, maxiter=500):
    obj = SmoothQObj(n, beta)
    m = len(q0)
    
    def val(x):
        return obj(x)
    
    def grad(x):
        return obj.gradient(x)
    
    try:
        res = minimize(val, q0, jac=grad, method='L-BFGS-B',
                      bounds=[(0.0, 1.0)] * m,
                      options={'maxiter': maxiter, 'ftol': 1e-15})
        q_out = np.clip(res.x, 0, 1)
        s = exact_score_q(q_out, n)
        return q_out, s
    except:
        return q0, exact_score_q(q0, n)


# Interpolation helper
def interpolate_q(q_src, n_src, n_dst):
    """Interpolate q from source resolution to destination resolution."""
    h_src = build_h_from_q(q_src, n_src)
    x_src = np.linspace(0, 2, n_src, endpoint=False) + 1.0/n_src
    x_dst = np.linspace(0, 2, n_dst, endpoint=False) + 1.0/n_dst
    h_dst = np.interp(x_dst, x_src, h_src)
    m_dst = n_dst // 2
    if n_dst % 2 == 0:
        q_dst = h_dst[:m_dst]
    else:
        q_dst = h_dst[:m_dst]
    return np.clip(q_dst, 0, 1)


# Active set epigraph
class EpigraphQSolver:
    def __init__(self, n, active_k):
        self.n = n
        self.m = n // 2
        self.active_k = active_k
        self.dx = 2.0 / n
    
    def obj(self, x):
        return x[0]
    
    def obj_grad(self, x):
        g = np.zeros(self.m + 1)
        g[0] = 1.0
        return g
    
    def constraint(self, x, k_idx):
        t = x[0]
        q = x[1:]
        h = build_h_from_q(q, self.n)
        k = self.active_k[k_idx]
        d = k - (self.n - 1)
        n = self.n
        if d >= 0:
            val = np.sum(h[d:] * (1.0 - h[:n-d]))
        else:
            dd = -d
            val = np.sum(h[dd:] * (1.0 - h[:n-dd]))
        return t - val * self.dx
    
    def constraint_grad(self, x, k_idx):
        q = x[1:]
        h = build_h_from_q(q, self.n)
        k = self.active_k[k_idx]
        d = k - (self.n - 1)
        n = self.n
        m = self.m
        g = np.zeros(m + 1)
        g[0] = 1.0
        if d >= 0:
            G = np.zeros(n)
            G[d:] += (1.0 - h[:n-d])
            G[:n-d] += (-h[d:])
        else:
            dd = -d
            G = np.zeros(n)
            G[:n-dd] += (1.0 - h[dd:])
            G[dd:] += (-h[:n-dd])
        Gq = (G[:m] - G[m:][::-1]) * self.dx
        g[1:] = -Gq
        return g


def solve_epigraph_q(q0, n, active_k, t0, maxiter=200):
    m = n // 2
    x0 = np.concatenate([[t0], q0])
    ep = EpigraphQSolver(n, active_k)
    
    cons = []
    for i in range(len(active_k)):
        fun_i = lambda x, i=i: ep.constraint(x, i)
        jac_i = lambda x, i=i: ep.constraint_grad(x, i)
        cons.append({'type': 'ineq', 'fun': fun_i, 'jac': jac_i})
    
    bounds = [(0.0, 5.0)] + [(0.0, 1.0)] * m
    try:
        res = minimize(ep.obj, x0, jac=ep.obj_grad, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        q_out = np.clip(res.x[1:], 0, 1)
        return q_out
    except:
        return q0


def active_refine(q0, n, time_limit, t_start):
    """Active set refinement."""
    m = n // 2
    dx = 2.0 / n
    q = q0.copy()
    
    for round_num in range(10):
        if time.time() - t_start > time_limit:
            break
        
        h = build_h_from_q(q, n)
        corrs = np.correlate(h, 1.0 - h, mode='full')
        current_max = corrs.max() * dx
        
        # Select active set
        L = min(30, 2 * n - 1)
        top_idx = np.argsort(corrs)[::-1][:L]
        active_k = sorted(top_idx.tolist())
        
        t0 = current_max * (1.0 + 1e-4) + 1e-6
        q_new = solve_epigraph_q(q, n, active_k, t0, maxiter=200)
        
        h_new = build_h_from_q(q_new, n)
        corrs_new = np.correlate(h_new, 1.0 - h_new, mode='full')
        c5_new = corrs_new.max() * dx
        
        if c5_new < current_max - 1e-14:
            q = q_new
        
        check_time = time.time()
        if check_time - t_start > time_limit:
            break
    return q
```

```python
import numpy as np

def build_h_from_q(q, n):
    m = n // 2
    if n % 2 == 0:
        return np.concatenate([q, 1.0 - q[::-1]])
    else:
        return np.concatenate([q, [0.5], 1.0 - q[::-1]])


def q_from_h(h, n):
    m = n // 2
    return h[:m].copy()


def exact_score_h(h):
    n = len(h)
    dx = 2.0 / n
    corrs = np.correlate(h, 1.0 - h, mode='full')
    return corrs.max() * dx


def exact_score_q(q, n):
    h = build_h_from_q(q, n)
    return exact_score_h(h)


def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)


def interpolate_q(q_src, n_src, n_dst):

class QObj:
    def __init__(self, n):
        self.n = n

    def __call__(self, q):
        return exact_score_q(q, self.n)


def nelder_mead_solve(q0, n, maxfev, t_start, time_limit):
    obj = QObj(n)
    try:
        res = minimize(obj, q0, method='Nelder-Mead',
                       options={'maxiter': maxfev, 'xatol': 1e-9,
                                'fatol': 1e-12, 'adaptive': True})
        if time.time() - t_start > time_limit:
            return q0, exact_score_q(q0, n)
        q_out = np.clip(res.x, 0, 1)
        s = exact_score_q(q_out, n)
        return q_out, s
    except Exception:

class SmoothQObj:
    def __init__(self, n, beta):
        self.n = n
        self.beta = beta
        self.dx = 2.0 / n

    def __call__(self, q):
        h = build_h_from_q(q, self.n)
        corrs = np.correlate(h, 1.0 - h, mode='full')
        return logsumexp(self.beta * corrs) / self.beta * self.dx

    def gradient(self, q):
        n = self.n
        m = n // 2
        h = build_h_from_q(q, n)
        corrs = np.correlate(h, 1.0 - h, mode='full')
        shifted = corrs - corrs.max()
        w = np.exp(self.beta * shifted)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(len(corrs)) / len(corrs)
        else:
            w /= w_sum

        G = np.zeros(n)
        for k in range(len(corrs)):
            if w[k] < 1e-14:
                continue
            d = k - (n - 1)
            if d >= 0 and d < n:
                G[d:] += (1.0 - h[:n - d])
                G[:n - d] += (-h[d:])
            elif d < 0:
                dd = -d
                if dd < n:
                    G[:n - dd] += (1.0 - h[dd:])
                    G[dd:] += (-h[:n - dd])

        return (G[:m] - G[m:][::-1]) * self.dx


def smooth_solve(q0, n, beta, maxiter, t_start, time_limit):
    obj = SmoothQObj(n, beta)
    m = len(q0)

    def val(x):
        return obj(x)

    def grad(x):
        return obj.gradient(x)

    try:
        res = minimize(val, q0, jac=grad, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * m,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        if time.time() - t_start > time_limit:
            return q0, exact_score_q(q0, n)
        q_out = np.clip(res.x, 0, 1)
        s = exact_score_q(q_out, n)
        return q_out, s
    except Exception:
        return q0, exact_score_q(q0, n)


def coordinate_descent(q, n, t_start, time_limit, max_rounds=20):
    m = len(q)
    q = q.copy()
    current_score = exact_score_q(q, n)

    deltas = [0.01, -0.01, 0.02, -0.02, 0.03, -0.03,
              0.05, -0.05, 0.08, -0.08, 0.12, -0.12]

    for rnd in range(max_rounds):
        if time.time() - t_start > time_limit:
            break
        improved = False
        order = list(range(m))
        rng_cd = np.random.RandomState(rnd * 7 + 42)
        rng_cd.shuffle(order)

        for i in order:
            if time.time() - t_start > time_limit:
                break
            old_val = q[i]
            candidates = [old_val, 0.0, 1.0, 0.5]
            for d in deltas:
                v = old_val + d
                if 0.0 <= v <= 1.0:
                    candidates.append(v)

            best_v = old_val
            best_s = current_score
            for v in candidates:
                q[i] = v
                s = exact_score_q(q, n)
                if s < best_s:
                    best_s = s
                    best_v = v

            q[i] = best_v
            if best_s < current_score - 1e-13:
                current_score = best_s
                improved = True

        if not improved:
            break

    return q, current_score


class EpigraphQSolver:
    def __init__(self, n, active_k):
        self.n = n
        self.m = n // 2
        self.active_k = active_k
        self.dx = 2.0 / n

    def obj(self, x):
        return x[0]

    def obj_grad(self, x):
        g = np.zeros(self.m + 1)
        g[0] = 1.0
        return g

    def constraint(self, x, k_idx):
        t = x[0]
        q = x[1:]
        h = build_h_from_q(q, self.n)
        k = self.active_k[k_idx]
        d = k - (self.n - 1)
        n = self.n
        if d >= 0:
            val = np.sum(h[d:] * (1.0 - h[:n - d]))
        else:
            dd = -d
            val = np.sum(h[dd:] * (1.0 - h[:n - dd]))
        return t - val * self.dx

    def constraint_grad(self, x, k_idx):
        q = x[1:]
        h = build_h_from_q(q, self.n)
        k = self.active_k[k_idx]
        d = k - (self.n - 1)
        n = self.n
        m = self.m
        g = np.zeros(m + 1)
        g[0] = 1.0
        G = np.zeros(n)
        if d >= 0 and d < n:
            G[d:] += (1.0 - h[:n - d])
            G[:n - d] += (-h[d:])
        elif d < 0:
            dd = -d
            if dd < n:
                G[:n - dd] += (1.0 - h[dd:])
                G[dd:] += (-h[:n - dd])
        Gq = (G[:m] - G[m:][::-1]) * self.dx
        g[1:] = -Gq
        return g


def solve_epigraph_q(q0, n, active_k, t0, maxiter=200):
    m = n // 2
    x0 = np.concatenate([[t0], q0])
    ep = EpigraphQSolver(n, active_k)

    cons = []
    for i in range(len(active_k)):
        cons.append({
            'type': 'ineq',
            'fun': lambda x, i=i: ep.constraint(x, i),
            'jac': lambda x, i=i: ep.constraint_grad(x, i)
        })

    bounds = [(0.0, 5.0)] + [(0.0, 1.0)] * m
    try:
        res = minimize(ep.obj, x0, jac=ep.obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return np.clip(res.x[1:], 0, 1)
    except Exception:
        return q0

        q_init = np.linspace(1.0, 0.0, m_p)
        h_p = build_h_from_q(q_init, n_p)
        best_c5 = exact_score_h(h_p)
        best_h = h_p
        best_n = n_p

    # Resolutions to try (even only for anti-symmetric form)
    resolutions = [48, 56, 64, 72, 80, 88, 96, 104, 112, 120, 128, 136, 144, 160]
    # Prioritize around parent size and smaller
    def res_priority(n):
        if best_n > 0:
            return abs(n - min(best_n, 128))
        return abs(n - 80)
    resolutions.sort(key=res_priority)

    # Phase 1: Multi-resolution search
    for n in resolutions:
        if time_left() < 25:
            break

        m = n // 2
        if best_h is not None and len(best_h) == n:
            q0 = q_from_h(best_h, n)
        elif best_h is not None:
            q0 = interpolate_q(q_from_h(best_h, best_n), best_n, n)
        else:
            q0 = np.ones(m) * 0.5
        q0 = np.clip(q0, 0, 1)

        cur_q = q0.copy()
        cur_s = exact_score_q(cur_q, n)

        # DE
        if time_left() > 35:
            if m <= 32:
                pop_sz = min(60, max(30, 2 * m))
                max_gen = 150
            elif m <= 48:
                pop_sz = min(50, max(25, int(1.3 * m)))
                max_gen = 120
            else:
                pop_sz = min(40, max(20, int(1.0 * m)))
                max_gen = 80

            de_lim = min(35, time_left() * 0.35)
            q_de, s_de = run_de(cur_q, n, pop_sz, max_gen, rng, de_lim, t_start)
            if s_de < cur_s - 1e-13:
                cur_s = s_de
                cur_q = q_de.copy()

        # Nelder-Mead
        if time_left() > 20:
            maxfev = min(5000, max(1000, int(200 * m)))
            nm_lim = min(25, time_left() * 0.25)
            q_nm, s_nm = nelder_mead_solve(cur_q, n, maxfev, t_start, nm_lim)
            if s_nm < cur_s - 1e-13:
                cur_s = s_nm
                cur_q = q_nm.copy()

        # Smooth L-BFGS-B
        if time_left() > 15:
            for beta in [80, 200, 600, 1500]:
                if time_left() < 8:
                    break
                lim = min(12, time_left() * 0.15)
                q_sm, s_sm = smooth_solve(cur_q, n, beta, 300, t_start, lim)
                if s_sm < cur_s - 1e-13:
                    cur_s = s_sm
                    cur_q = q_sm.copy()

        # Coordinate descent
        if time_left() > 12:
            cd_lim = min(18, time_left() * 0.25)
            q_cd, s_cd = coordinate_descent(cur_q, n, t_start, cd_lim, max_rounds=15)
            if s_cd < cur_s - 1e-13:
                cur_s = s_cd
                cur_q = q_cd.copy()

        # Active set epigraph
        if time_left() > 15:
            act_lim = min(20, time_left() * 0.25)
            q_act = active_refine(cur_q, n, t_start, act_lim)
            s_act = exact_score_q(q_act, n)
            if s_act < cur_s - 1e-13:
                cur_s = s_act
                cur_q = q_act.copy()

        # Final coordinate polish
        if time_left() > 10:
            cd_lim2 = min(12, time_left() * 0.2)
            q_cd2, s_cd2 = coordinate_descent(cur_q, n, t_start, cd_lim2, max_rounds=20)
            if s_cd2 < cur_s - 1e-13:
                cur_s = s_cd2
                cur_q = q_cd2.copy()

        if cur_s < best_c5 - 1e-13:
            best_c5 = cur_s
            best_h = build_h_from_q(cur_q, n)
            best_n = n

    # Phase 2: Cross-pollination - refine best at nearby larger resolutions
    if time_left() > 50 and best_h is not None:
        for dn in [2, 4, 8, 16]:
            n_try = best_n + dn
            if time_left() < 25:
                break
            if n_try > 200 or n_try % 2 != 0:
                continue

            q0 = interpolate_q(q_from_h(best_h, best_n), best_n, n_try)
            cur_q = np.clip(q0, 0, 1)
            cur_s = exact_score_q(cur_q, n_try)

            # Nelder-Mead
            if time_left() > 18:
                maxfev = min(4000, max(800, int(150 * (n_try // 2))))
                nm_lim = min(22, time_left() * 0.3)
                q_nm, s_nm = nelder_mead_solve(cur_q, n_try, maxfev, t_start, nm_lim)
                if s_nm < cur_s - 1e-13:
                    cur_s = s_nm
                    cur_q = q_nm.copy()

            # Coordinate descent
            if time_left() > 12:
                cd_lim = min(15, time_left() * 0.3)
                q_cd, s_cd = coordinate_descent(cur_q, n_try, t_start, cd_lim, max_rounds=20)
                if s_cd < cur_s - 1e-13:
                    cur_s = s_cd
                    cur_q = q_cd.copy()

            # Active set
            if time_left() > 15:
                act_lim = min(18, time_left() * 0.3)
                q_act = active_refine(cur_q, n_try, t_start, act_lim)
                s_act = exact_score_q(q_act, n_try)
                if s_act < cur_s - 1e-13:
                    cur_s = s_act
                    cur_q = q_act.copy()

            if cur_s < best_c5 - 1e-13:
                best_c5 = cur_s
                best_h = build_h_from_q(cur_q, n_try)
                best_n = n_try

    # Phase 3: Intensive final refinement
    if time_left() > 20 and best_h is not None:
        n_f = best_n
        m_f = n_f // 2
        q_f = q_from_h(best_h, n_f)

        # Multiple Nelder-Mead passes
        for attempt in range(4):
            if time_left() < 12:
                break
            if attempt == 0:
                q_try = q_f.copy()
            else:
                q_try = q_f + rng.normal(0, 0.005 * (attempt + 1), m_f)
                q_try = np.clip(q_try, 0, 1)

            maxfev = min(6000, max(2000, int(300 * m_f)))
            nm_lim = min(20, time_left() * 0.3)
            q_nm, s_nm = nelder_mead_solve(q_try, n_f, maxfev, t_start, nm_lim)
            if s_nm < best_c5 - 1e-13:
                best_c5 = s_nm
                best_h = build_h_from_q(q_nm, n_f)
                q_f = q_nm.copy()

        # Active set with extended rounds
        if time_left() > 15:
            act_lim = min(30, time_left() * 0.4)
            q_act = active_refine(q_from_h(best_h, n_f), n_f, t_start, act_lim)
            s_act = exact_score_q(q_act, n_f)
            if s_act < best_c5 - 1e-13:
                best_c5 = s_act
                best_h = build_h_from_q(q_act, n_f)

        # Final coordinate descent with more rounds
        if time_left() > 12:
            cd_lim = min(20, time_left() * 0.4)
            q_final, s_final = coordinate_descent(
                q_from_h(best_h, n_f), n_f, t_start, cd_lim, max_rounds=30)
            if s_final < best_c5 - 1e-13:
                best_c5 = s_final
                best_h = build_h_from_q(q_final, n_f)

        # One more smooth pass at very high beta
        if time_left() > 12:
            lim = min(15, time_left() * 0.3)
            q_sm, s_sm = smooth_solve(q_from_h(best_h, n_f), n_f, 3000, 500, t_start, lim)
            if s_sm < best_c5 - 1e-13:
                best_c5 = s_sm
                best_h = build_h_from_q(q_sm, n_f)
                # Quick coordinate polish
                if time_left() > 8:
                    q_cp, s_cp = coordinate_descent(q_sm, n_f, t_start, min(10, time_left() * 0.3), max_rounds=20)
                    if s_cp < best_c5 - 1e-13:
                        best_c5 = s_cp
                        best_h = build_h_from_q(q_cp, n_f)

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    n_out = len(best_h)

    # Ensure anti-symmetric form if even
    if n_out % 2 == 0:
        m_out = n_out // 2
        q_out = best_h[:m_out]
        h_sym = build_h_from_q(q_out, n_out)
        s_sym = exact_score_h(h_sym)
        if s_sym <= best_c5 + 1e-13:
            best_h = h_sym
            best_c5 = s_sym

    final_score = exact_score_h(best_h)
    if final_score < best_c5:
        best_c5 = final_score

    return best_h, best_c5, n_out
```
