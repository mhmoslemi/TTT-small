import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time


def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corrs = np.correlate(h, 1.0 - h, mode='full')
    return corrs.max() * dx


def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)


def build_h_from_q(q, n):
    m = n // 2
    if n % 2 == 0:
        return np.concatenate([q, 1.0 - q[::-1]])
    else:
        return np.concatenate([q, [0.5], 1.0 - q[::-1]])


def q_from_h(h, n):
    m = n // 2
    return h[:m].copy()


def grad_corr_d(h, d, n):
    g = np.zeros(n)
    if d >= 0:
        if d < n:
            g[d:] += (1.0 - h[:n - d])
            g[:n - d] += (-h[d:])
    else:
        dd = -d
        if dd < n:
            g[:n - dd] += (1.0 - h[dd:])
            g[dd:] += (-h[:n - dd])
    return g


def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def q_smooth_value_grad(q, n, beta):
    m = n // 2
    h = build_h_from_q(q, n)
    dx = 2.0 / n
    corrs = compute_corrs(h)
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        val = corrs.max() * dx
        w = np.zeros(len(corrs))
        w[corrs.argmax()] = 1.0
    else:
        w = w / w_sum
        val = (corrs.max() + np.log(w_sum) / beta) * dx
    Gq = np.zeros(m)
    for k in range(len(corrs)):
        if w[k] < 1e-14:
            continue
        d = k - (n - 1)
        G = grad_corr_d(h, d, n)
        Gq += w[k] * (G[:m] - G[m:][::-1])
    return val, Gq * dx


def q_pnorm_value_grad(q, n, p):
    m = n // 2
    h = build_h_from_q(q, n)
    dx = 2.0 / n
    corrs = compute_corrs(h)
    eps = 1e-14
    c = np.maximum(corrs, eps)
    norm_val = np.power(np.sum(np.power(c, p)), 1.0 / p)
    val = norm_val * dx
    w_k = (c / norm_val) ** (p - 1)
    Gq = np.zeros(m)
    for k in range(len(corrs)):
        if w_k[k] < 1e-14:
            continue
        d = k - (n - 1)
        G = grad_corr_d(h, d, n)
        Gq += w_k[k] * (G[:m] - G[m:][::-1])
    return val, Gq * dx

class FullEpigraphSolver:

def solve_q_epigraph(q, n, active_k, t0, maxiter=200):
    m = n // 2
    solver = QEpigraphSolver(n, active_k)
    x0 = np.concatenate([[t0], q])
    bounds = [(0.0, 2.0)] + [(0.0, 1.0)] * m
    cons = [{'type': 'ineq', 'fun': solver.constraints_fun, 'jac': solver.constraints_jac}]
    try:
        res = minimize(solver.obj, x0, jac=solver.obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        q_out = np.clip(res.x[1:], 0.0, 1.0)
        t_out = float(res.x[0])
        return q_out, t_out
    except Exception:
        return q.copy(), t0

            best_q = q_new.copy()
        hidden = corrs_new * dx > t_solved + 1e-8
        if hidden.sum() > 0:
            hidden_idx = np.where(hidden)[0]
            new_active = sorted(set(active_k) | set(hidden_idx.tolist()))
            if len(new_active) > 70:
                vals = np.array([corrs_new[k] for k in new_active])
                order = np.argsort(vals)[::-1][:70]
                new_active = sorted([new_active[i] for i in order])
            active_k = new_active
        t0 = max(t_solved, c5_new) * 1.001 + 1e-6
        best_q = q_new
        if c5_new >= best_c5 - 1e-13 and hidden.sum() == 0:
            break
    return best_q, best_c5


def run_full_active(h, n, t_limit, t_start):
    target_sum = n / 2.0
    dx = 2.0 / n
    corrs = compute_corrs(h)
    current_max = corrs.max() * dx
    L_init = min(30, 2 * n - 1)
    top_idx = np.argsort(corrs)[::-1][:L_init]
    active_k = sorted(top_idx.tolist())
    best_h = h.copy()
    best_c5 = current_max
    t0 = current_max * 1.01 + 1e-5
    for rnd in range(12):
        if time.time() - t_start > t_limit:
            break
        h_cur = project_feasible(best_h, target_sum)
        corrs_cur = compute_corrs(h_cur)
        cur_max = corrs_cur.max() * dx
        if cur_max < best_c5:
            best_c5 = cur_max
            best_h = h_cur.copy()
        h_new, t_solved = solve_full_epigraph(h_cur, n, active_k, t0, target_sum, maxiter=200)
        h_new = project_feasible(h_new, target_sum)
        corrs_new = compute_corrs(h_new)
        c5_new = corrs_new.max() * dx
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
        hidden = corrs_new * dx > t_solved + 1e-8
        if hidden.sum() > 0:
            hidden_idx = np.where(hidden)[0]
            new_active = sorted(set(active_k) | set(hidden_idx.tolist()))
            if len(new_active) > 60:
                vals = np.array([corrs_new[k] for k in new_active])
                order = np.argsort(vals)[::-1][:60]
                new_active = sorted([new_active[i] for i in order])
            active_k = new_active
        t0 = max(t_solved, c5_new) * 1.001 + 1e-6
        best_h = h_new
        if c5_new >= best_c5 - 1e-13 and hidden.sum() == 0:
            break
    return best_h, best_c5


def run_sa_q(q0, n, t_limit, t_start, rng):
    m = n // 2
    best_q = q0.copy()
    h = build_h_from_q(q0, n)
    best_score = exact_score(h)
    cur_q = q0.copy()
    cur_score = best_score
    T_high = 5e-4
    T_low = 1e-6
    max_iter = 200000
    no_improve_count = 0
    stagnation_limit = 20000
    for it in range(max_iter):
        if time.time() - t_start > t_limit:
            break
        frac = it / float(max_iter)
        T = T_high * (T_low / T_high) ** frac
        # Propose
        move_type = rng.random()
        if move_type < 0.6:
            # Single or few coordinate
            nc = rng.randint(1, 4)
            idx = rng.randint(0, m, size=nc)
            sigma = rng.uniform(0.02, 0.1)
            q_prop = cur_q.copy()
            q_prop[idx] = np.clip(cur_q[idx] + rng.randn(nc) * sigma, 0.0, 1.0)
        elif move_type < 0.85:
            # Block move
            blen = rng.randint(3, min(12, m))
            start = rng.randint(0, max(1, m - blen))
            q_prop = cur_q.copy()
            q_prop[start:start + blen] = np.clip(
                cur_q[start:start + blen] + rng.randn(blen) * 0.05, 0.0, 1.0)
        else:
            # Global scaling perturbation
            q_prop = np.clip(cur_q * (1.0 + rng.randn() * 0.03) + rng.randn(m) * 0.02, 0.0, 1.0)
        h_prop = build_h_from_q(q_prop, n)
        s_prop = exact_score(h_prop)
        delta_s = s_prop - cur_score
        if delta_s < 0 or rng.random() < np.exp(-delta_s / max(T, 1e-15)):
            cur_q = q_prop
            cur_score = s_prop
            if cur_score < best_score - 1e-13:
                best_score = cur_score
                best_q = cur_q.copy()
                no_improve_count = 0
            else:
                no_improve_count += 1
        else:
            no_improve_count += 1
        # Restart from best if stagnant
        if no_improve_count > stagnation_limit:
            cur_q = best_q.copy()
            cur_score = best_score
            no_improve_count = 0
    return best_q, best_score


def run_multistart_q(q0, n, t_limit, t_start, rng):
    m = n // 2
    best_q = q0.copy()
    h = build_h_from_q(q0, n)
    best_score = exact_score(h)
    n_restarts = 8
    iter_budget = t_limit / max(1, n_restarts)
    for r in range(n_restarts):
        if time.time() - t_start > t_limit:
            break
        # Perturb best
        q_pert = np.clip(best_q + rng.randn(m) * rng.uniform(0.05, 0.15), 0.0, 1.0)
        # Quick homotopy from perturbed
        t_sub = time.time()
        sub_limit = min(iter_budget, t_limit - (time.time() - t_start))
        if sub_limit < 5:
            break
        betas_sub = [50, 200, 800]
        q_cur = q_pert.copy()
        for beta in betas_sub:
            if time.time() - t_sub > sub_limit:
                break
            def val_fn(x):
                return q_smooth_value_grad(np.clip(x, 0, 1), n, beta)[0]
            def grad_fn(x):
                return q_smooth_value_grad(np.clip(x, 0, 1), n, beta)[1]
            try:
                res = minimize(val_fn, q_cur, jac=grad_fn, method='L-BFGS-B',
                               bounds=[(0.0, 1.0)] * m,
                               options={'maxiter': 200, 'ftol': 1e-15})
                q_cur = np.clip(res.x, 0.0, 1.0)
            except Exception:
                pass
        h_new = build_h_from_q(q_cur, n)
        s_new = exact_score(h_new)
        if s_new < best_score - 1e-13:
            best_score = s_new
            best_q = q_cur.copy()
    return best_q, best_score


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)
    reserve = 20.0

    def tl():
        return budget_s - (time.time() - t_start) - reserve

    # Initialize from parent
    h_parent = np.array(initial_h_values, dtype=float)
    n_parent = len(h_parent)
    target_sum = n_parent / 2.0
    h_parent = project_feasible(h_parent, target_sum)

    best_h = h_parent.copy()
    best_c5 = exact_score(h_parent)
    q_best = q_from_h(h_parent, n_parent)
    n_best = n_parent

    # Phase 1: Homotopy polishing in q-space
    t_lim = min(180, tl() * 0.22)
    if tl() > 40:
        q1, s1 = run_homotopy_q(q_best, n_parent, t_lim, time.time())
        h1 = build_h_from_q(q1, n_parent)
        h1 = project_feasible(h1, target_sum)
        s1e = exact_score(h1)
        if s1e < best_c5 - 1e-14:
            best_c5 = s1e
            best_h = h1.copy()
            q_best = q1.copy()

    # Phase 2: p-norm polishing
    if tl() > 60:
        t_lim2 = min(100, tl() * 0.12)
        q2, s2 = run_pnorm_q(q_best, n_parent, t_lim2, time.time())
        h2 = build_h_from_q(q2, n_parent)
        h2 = project_feasible(h2, target_sum)
        s2e = exact_score(h2)
        if s2e < best_c5 - 1e-14:
            best_c5 = s2e
            best_h = h2.copy()
            q_best = q2.copy()

    # Phase 3: Active-set epigraph in q-space
    if tl() > 80:
        t_lim3 = min(180, tl() * 0.2)
        q3, s3 = run_active_set_q(q_best, n_parent, t_lim3, time.time())
        h3 = build_h_from_q(q3, n_parent)
        h3 = project_feasible(h3, target_sum)
        s3e = exact_score(h3)
        if s3e < best_c5 - 1e-14:
            best_c5 = s3e
            best_h = h3.copy()
            q_best = q3.copy()

    # Phase 4: Simulated annealing for exploration
    if tl() > 100:
        t_lim4 = min(200, tl() * 0.2)
        q4, s4 = run_sa_q(q_best, n_parent, t_lim4, time.time(), rng)
        h4 = build_h_from_q(q4, n_parent)
        h4 = project_feasible(h4, target_sum)
        s4e = exact_score(h4)
        if s4e < best_c5 - 1e-14:
            best_c5 = s4e
            best_h = h4.copy()
            q_best = q4.copy()

    # Phase 5: Multi-start exploration
    if tl() > 80:
        t_lim5 = min(120, tl() * 0.15)
        q5, s5 = run_multistart_q(q_best, n_parent, t_lim5, time.time(), rng)
        h5 = build_h_from_q(q5, n_parent)
        h5 = project_feasible(h5, target_sum)
        s5e = exact_score(h5)
        if s5e < best_c5 - 1e-14:
            best_c5 = s5e
            best_h = h5.copy()
            q_best = q5.copy()

    # Phase 6: Full-space (non-symmetric) epigraph
    if tl() > 60:
        t_lim6 = min(120, tl() * 0.15)
        h6, s6 = run_full_active(best_h, n_parent, t_lim6, time.time())
        h6 = project_feasible(h6, target_sum)
        s6e = exact_score(h6)
        if s6e < best_c5 - 1e-14:
            best_c5 = s6e
            best_h = h6.copy()
            q_best = q_from_h(h6, n_parent)
