import numpy as np
from scipy.optimize import linprog
import time

def repair_h(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx])
            h[idx] += add
            remaining -= add
            if remaining <= 1e-13:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-13:
                break
    # Final residual fix
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def score_exact(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def compute_gradient_row(h, m, n, dx):
    """Compute gradient of P[m] with respect to h."""
    d = (n - 1) - m
    G = np.zeros(n)
    if d == 0:
        G = (1.0 - 2.0 * h) * dx
    elif d > 0:
        # P[m] = dx * sum_{j=0}^{n-1-d} h[j+d]*(1-h[j])
        # dP/dh[j] for j in [d, n-1]: +dx*(1-h[j-d])
        # dP/dh[j] for j in [0, n-1-d]: -dx*h[j+d]
        G[d:n] += dx * (1.0 - h[0:n-d])
        G[0:n-d] -= dx * h[d:n]
    else:
        e = -d
        # P[m] = dx * sum_{j=0}^{n-1-e} h[j]*(1-h[j+e])
        # dP/dh[j] for j in [0, n-1-e]: +dx*(1-h[j+e])
        # dP/dh[j] for j in [e, n-1]: -dx*h[j-e]
        G[0:n-e] += dx * (1.0 - h[e:n])
        G[e:n] -= dx * h[0:n-e]
    return G

    if len(active) > max_active:
        top_idx = np.argsort(-profile[active])[:max_active]
        active = active[top_idx]
    return active, c

def solve_direction_lp(h, active, n, dx):
    """Solve the LP to find the best descent direction."""
    # Build gradient rows for active set
    G = np.zeros((len(active), n))
    for i, m in enumerate(active):
        G[i] = compute_gradient_row(h, m, n, dx)
    
    # Variables: u(n), v(n), alpha(1) → total 2n+1
    nv = 2 * n + 1
    
    # Objective: minimize -alpha
    c_obj = np.zeros(nv)
    c_obj[-1] = -1.0
    
    # Inequality constraints: G @ (u-v) + alpha <= 0
    A_ub = np.zeros((len(active) + 2 * n, nv))
    b_ub = np.zeros(len(active) + 2 * n)
    
    for i in range(len(active)):
        A_ub[i, 0:n] = G[i]       # u part
        A_ub[i, n:2*n] = -G[i]    # v part
        A_ub[i, -1] = 1.0         # alpha part
        b_ub[i] = 0.0
    
    # Box constraints: u_i - v_i <= 1 - h_i
    for i in range(n):
        A_ub[len(active) + i, i] = 1.0
        A_ub[len(active) + i, n + i] = -1.0
        b_ub[len(active) + i] = 1.0 - h[i]
    
    # Box constraints: -(u_i - v_i) <= h_i → -u_i + v_i <= h_i
    for i in range(n):
        row = len(active) + n + i
        A_ub[row, i] = -1.0
        A_ub[row, n + i] = 1.0
        b_ub[row] = h[i]
    
    # Equality constraints: sum(u) - sum(v) = 0, sum(u) + sum(v) = 1
    A_eq = np.zeros((2, nv))
    A_eq[0, 0:n] = 1.0
    A_eq[0, n:2*n] = -1.0
    A_eq[1, 0:n] = 1.0
    A_eq[1, n:2*n] = 1.0
    b_eq = np.array([0.0, 1.0])
    
    bounds = [(0.0, None)] * (2 * n) + [(0.0, None)]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
    except Exception:
        return None, 0.0
    
    if not res.success:
        return None, 0.0
    
    alpha = max(0.0, -res.fun)
    d = res.x[0:n] - res.x[n:2*n]
    
    # Clean up tiny negatives from solver
    d[np.abs(d) < 1e-12] = 0.0
    d_sum = d.sum()
    if abs(d_sum) > 1e-10:
        d -= d_sum / n
    
    norm = np.abs(d).sum()
    if norm < 1e-10:
        return None, alpha
    
    d = d / norm
    return d, alpha

def interpolate_to_n(h, new_n):

    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])

    h_new = np.interp(xt, full_xs, full_h)

    return repair_h(h_new)

import numpy as np
from scipy.optimize import linprog
import time


def repair_h(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx])
            h[idx] += add
            remaining -= add
            if remaining <= 1e-13:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-13:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h


def score_exact(h):
    n = len(h)
    if n == 0:
        return float('inf')
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))


def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx


def compute_gradient_row(h, m_idx, n, dx):
    d = m_idx - (n - 1)
    G = np.zeros(n)
    if d == 0:
        G[:] = (1.0 - 2.0 * h) * dx
    elif d > 0:
        G[d:n] += dx * (1.0 - h[0:n - d])
        G[0:n - d] -= dx * h[d:n]
    else:
        e = -d
        G[0:n - e] += dx * (1.0 - h[e:n])
        G[e:n] -= dx * h[0:n - e]
    return G


def select_active(profile, max_active=30):
    c = np.max(profile)
    med = np.median(profile)
    beam = max(0.004, 0.06 * (c - med))
    active = np.where(profile >= c - beam)[0]
    if len(active) == 0:
        active = np.array([np.argmax(profile)])
    if len(active) > max_active:
        top = np.argsort(-profile[active])[:max_active]
        active = active[top]
    return active, c


def solve_lp_direction(h, active, n, dx):
    nv = 2 * n + 1
    G = np.zeros((len(active), n))
    for i, m in enumerate(active):
        G[i] = compute_gradient_row(h, m, n, dx)

    c_obj = np.zeros(nv)
    c_obj[-1] = -1.0

    n_rows = len(active) + 2 * n
    A_ub = np.zeros((n_rows, nv))
    b_ub = np.zeros(n_rows)

    for i in range(len(active)):
        A_ub[i, 0:n] = G[i]
        A_ub[i, n:2 * n] = -G[i]
        A_ub[i, -1] = 1.0

    for i in range(n):
        A_ub[len(active) + i, i] = 1.0
        A_ub[len(active) + i, n + i] = -1.0
        b_ub[len(active) + i] = 1.0 - h[i]
        A_ub[len(active) + n + i, i] = -1.0
        A_ub[len(active) + n + i, n + i] = 1.0
        b_ub[len(active) + n + i] = h[i]

    A_eq = np.zeros((2, nv))
    A_eq[0, 0:n] = 1.0
    A_eq[0, n:2 * n] = -1.0
    A_eq[1, 0:n] = 1.0
    A_eq[1, n:2 * n] = 1.0
    b_eq = np.array([0.0, 1.0])

    bounds = [(0.0, None)] * nv

    try:

                      bounds=bounds, method='highs')
    except Exception:
        return None, 0.0

    if not res.success:
        return None, 0.0

    d = res.x[0:n] - res.x[n:2 * n]

    d -= d.mean()
    norm = np.abs(d).sum()
    if norm < 1e-10:
        return None, alpha
    d /= norm
    return d, alpha

        if time.perf_counter() - t_start > time_limit_s:
            break

        profile = compute_profile(h)
        c = np.max(profile)
        if c < best_c - 1e-14:
            best_c = c
            best_h = h.copy()
            stall = 0

        active, c_max = select_active(profile, max_active=30)
        d, alpha = solve_lp_direction(h, active, n, dx)

        if d is None or alpha < 1e-9:
            stall += 1
            if stall >= max_stall:
                break
            # Fallback: weighted gradient on top rows
            top_k = min(10, len(active))
            top_rows = np.argsort(-profile[:])[:top_k]
            w = profile[top_rows] - c_max + 0.01
            w = np.maximum(w, 0.01)
            w /= w.sum()
            G_avg = np.zeros(n)
            for idx, r in enumerate(top_rows):
                G_avg += w[idx] * compute_gradient_row(h, r, n, dx)
            G_avg -= G_avg.mean()
            gnorm = np.linalg.norm(G_avg)
            if gnorm < 1e-14:
                continue
            G_avg /= gnorm
            improved = False
            for scale in [0.05, 0.03, 0.01, 0.005, 0.002]:
                h_try = repair_h(h - scale * G_avg)
                s_try = score_exact(h_try)
                if s_try < c - 1e-14:
                    h = h_try
                    if s_try < best_c:
                        best_c = s_try
                        best_h = h.copy()
                    improved = True
                    break
            if not improved:
                continue

        improved = False
        for scale in [0.4, 0.25, 0.15, 0.08, 0.04, 0.02, 0.01, 0.005]:
            h_try = repair_h(h + scale * d)
            s_try = score_exact(h_try)
            if s_try < c - 1e-14:
                h = h_try
                if s_try < best_c:
                    best_c = s_try
                    best_h = h.copy()
                improved = True
                break

        if not improved:
            for scale in [0.02, 0.01, 0.005]:
                h_try = repair_h(h - scale * d)
                s_try = score_exact(h_try)
                if s_try < c - 1e-14:
                    h = h_try
                    if s_try < best_c:
                        best_c = s_try
                        best_h = h.copy()
                    improved = True
                    break
        if not improved:
            stall += 1
            if stall >= max_stall:
                break
            # Widen active set
            beam_w = max(0.01, 0.12 * (c - np.median(profile)))
            active_w = np.where(profile >= c - beam_w)[0]
            if len(active_w) > 50:
                top_w = np.argsort(-profile[active_w])[:50]
                active_w = active_w[top_w]
            d2, a2 = solve_lp_direction(h, active_w, n, dx)
            if d2 is not None and a2 > 1e-9:
                for scale in [0.3, 0.15, 0.05, 0.02, 0.01]:
                    h_try = repair_h(h + scale * d2)
                    s_try = score_exact(h_try)
                    if s_try < c - 1e-14:
                        h = h_try
                        if s_try < best_c:
                            best_c = s_try
                            best_h = h.copy()
                        break

    return best_h, best_c


def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_h(h_new)


def symmetrize(h):
    return (h + h[::-1]) / 2.0


def make_structural_seed(n, rng):
    """Create a balanced structural seed using quasi-random phases."""
    phi = (np.sqrt(5.0) - 1.0) / 2.0
    x = np.arange(n)
    frac = ((x + 0.5) * phi) % 1.0
    h = (frac > 0.5).astype(np.float64)
    # Adjust to get exactly n/2 ones
    target = n // 2
    cnt = int(h.sum())
    if cnt > target:
        idx = np.where(h == 1)[0]
        rng.shuffle(idx)
        h[idx[:cnt - target]] = 0
    elif cnt < target:
        idx = np.where(h == 0)[0]
        rng.shuffle(idx)
        h[idx[:target - cnt]] = 1
    return repair_h(h)


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()

    def time_left():
        return budget_s - (time.perf_counter() - t_start)

    rng = np.random.RandomState(seed)

    parent = None
    try:
        parent = np.array(initial_h_values, dtype=np.float64)
    except (NameError, Exception):
        pass

    if parent is None or len(parent) < 4 or not np.all(np.isfinite(parent)):
        parent = np.full(70, 0.5)

    parent = repair_h(parent)
    best_h = parent.copy()
    best_c = score_exact(parent)

    # Try symmetrization
    h_sym = repair_h(symmetrize(parent))
    c_sym = score_exact(h_sym)
    if c_sym < best_c - 1e-12:
        best_h = h_sym
        best_c = c_sym

    # Phase 1: Initial LP polish on parent
    if time_left() > 80:
        t1 = min(250, max(50, time_left() * 0.35))
        best_h, best_c = lp_polish(best_h, max_iters=3000, time_limit_s=t1, t_start=t_start)

    # Phase 2: Dimension exploration
    n_best = len(best_h)
    dims = set()
    for d in [n_best - 6, n_best - 4, n_best - 2, n_best + 2, n_best + 4, n_best + 6, n_best + 8, n_best + 10]:
        if 34 <= d <= 130:
            dims.add(d)
    for d in [62, 64, 66, 68, 72, 74, 76, 78, 80, 82, 84, 88]:
        if 34 <= d <= 130:
            dims.add(d)
    dims.discard(n_best)
    dims_list = sorted(dims, key=lambda d: abs(d - n_best))

    for new_n in dims_list[:10]:
        if time_left() < 25:
            break
        h_interp = interpolate_to_n(best_h, new_n)
        # Also try a structural seed at this dimension
        h_struct = make_structural_seed(new_n, rng)
        candidates = [h_interp]
        c_struct = score_exact(h_struct)
        if c_struct < best_c + 0.02:
            candidates.append(h_struct)

        for h_cand in candidates:
            if time_left() < 15:
                break
            budget = min(70, max(20, time_left() * 0.4))
            h_pol, c_pol = lp_polish(h_cand, max_iters=1000, time_limit_s=budget, t_start=t_start)
            if c_pol < best_c - 1e-12:
                best_h = h_pol.copy()
                best_c = c_pol

    # Phase 3: Final deep polish
    if time_left() > 25:
        final_t = max(10, time_left() - 12)
        best_h, best_c = lp_polish(best_h, max_iters=5000, time_limit_s=final_t, t_start=t_start)

    # Phase 4: Quick dimension retry from improved solution
    if time_left() > 30:
        n_now = len(best_h)
        for d in [n_now - 2, n_now + 2, n_now + 4, n_now - 4, n_now + 6]:
            if time_left() < 15:
                break
            if 34 <= d <= 130 and d != n_now:
                h_interp = interpolate_to_n(best_h, d)
                h_pol, c_pol = lp_polish(h_interp, max_iters=500,
                                         time_limit_s=min(25, time_left() - 5), t_start=t_start)
                if c_pol < best_c - 1e-12:
                    best_h = h_pol.copy()
                    best_c = c_pol

    # Final validation
    best_h = repair_h(best_h)
    final_c = score_exact(best_h)

    try:
        er = evaluate_erdos_solution(best_h)
        if isinstance(er, (float, int)):
            final_c = min(final_c, float(er))
        elif isinstance(er, tuple) and len(er) >= 1:
            final_c = min(final_c, float(er[0]))
    except Exception:
        pass

    return best_h, float(final_c), int(len(best_h))
