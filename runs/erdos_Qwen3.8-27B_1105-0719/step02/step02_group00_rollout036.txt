def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity, kind='mergesort')
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx])
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h, kind='mergesort')
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])

        def obj_jac(x):
            g = np.zeros(n + 1)
            g[-1] = 1.0
            return g
        def eq_fun(x):
            return x[:-1].sum() - n / 2.0
        def eq_jac(x):
            g = np.zeros(n + 1)
            g[:-1] = 1.0
            return g
        def ineq_fun(x):
            h_x = x[:-1]
            prof = np.correlate(h_x, 1 - h_x, mode="full") * dx
            return x[-1] - prof[active]
        def ineq_jac(x):
            h_x = x[:-1]
            G = build_gradient(h_x, n, dx)
            J = np.zeros((M_act, n + 1))
            for ii in range(M_act):
                J[ii, :-1] = -G[active[ii], :]
                J[ii, -1] = 1.0
            return J
        
        x0 = np.concatenate([h, [c + 1e-5]])
        bounds = [(0.0, 1.0)] * n + [(0.0, c + 0.5)]
        cons = [
            {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
            {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
        ]
        try:
            res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-13})
            if res.x is not None and np.isfinite(res.x).all():
                h_new = repair_feasible(res.x[:-1])
                c_new = exact_score(h_new)
                if c_new <= c + 1e-10:
                    h = h_new
                    if c_new < best_c - 1e-14:
                        best_c = c_new
                    else:
                        break
                else:
                    break
        except Exception:
            break
    return h, best_c

def lse_projected_gradient(h0, tau_start=0.01, tau_end=0.001, max_iters=200, time_limit=60.0):
    n = len(h0)
    dx = 2.0 / n
    h = h0.copy()
    best_c = exact_score(h)
    t0 = time.perf_counter()
    h_prev = h.copy()
    c_prev = exact_score(h)
    
    for it in range(max_iters):
        if time.perf_counter() - t0 > time_limit:
            break
        tau = tau_start * (tau_end / tau_start) ** (it / max(1, max_iters - 1))
        profile = compute_profile(h)
        m_prof = np.max(profile)
        weights = np.exp((profile - m_prof) / tau)
        weights /= weights.sum()
        G = build_gradient(h, n, dx)
        grad = weights @ G
        grad -= grad.mean()
        
        # Adaptive step size (Barzilai-Borwein style)
        if it > 0:
            s = h - h_prev
            y = compute_profile(h) - compute_profile(h_prev)
            sy = s @ y
            yy = y @ y
            if yy > 1e-20:
                alpha = sy / yy
                alpha = np.clip(alpha, 0.001, 0.1)
            else:
                alpha = 0.01
        else:
            alpha = 0.02
        
        # Try several step sizes
        improved = False
        for a_scale in [1.0, 0.5, 0.25, 0.1, 0.05]:
            a = alpha * a_scale
            h_try = repair_feasible(h - a * grad)
            c_try = exact_score(h_try)
            if c_try < best_c - 1e-14:
                h = h_try
                best_c = c_try
                improved = True
                break
            elif c_try < exact_score(h):
                h = h_try
                improved = True
                break
        if not improved:
            h_prev = h.copy()
            c_prev = exact_score(h)
            if it > 50:
                # Try random perturbation to escape
                rng = np.random.RandomState(it)
                z = rng.randn(n)
                z -= z.mean()
                h_try = repair_feasible(h + 0.005 * z)
                c_try = exact_score(h_try)
                if c_try < best_c - 1e-14:
                    h = h_try
                    best_c = c_try
        else:
            h_prev = h.copy()
            c_prev = exact_score(h)
    return h, best_c

def coordinate_local(h0, max_rounds=20, time_limit=40.0):
    """Optimize individual coordinates with 1D search, repairing after each."""
    n = len(h0)
    dx = 2.0 / n
    h = h0.copy()
    best_c = exact_score(h)
    t0 = time.perf_counter()
    
    for rnd in range(max_rounds):
        if time.perf_counter() - t0 > time_limit:
            break
        improved = False
        for i in range(n):
            if time.perf_counter() - t0 > time_limit:
                break
            old_val = h[i]
            
            def score_fn(val):
                h[i] = val
                s = np.correlate(h, 1 - h, mode="full") * dx
                h[i] = old_val
                return s.max()
            
            res = minimize_scalar(score_fn, bounds=(0.0, 1.0), method='bounded',
                                options={'xatol': 1e-8})
            if res.success and abs(res.x - old_val) > 1e-8:
                h[i] = res.x
                h = repair_feasible(h)
                c_new = exact_score(h)
                if c_new < best_c - 1e-14:
                    best_c = c_new
                    improved = True
        if not improved:
            break
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()
    rng = np.random.RandomState(seed)
    
    def time_left():
        return budget_s - (time.perf_counter() - t_start)
    
    # Get parent
    parent = None
    try:
        parent = np.array(initial_h_values, dtype=np.float64)
    except NameError:
        pass
    except Exception:
        pass
    
    if parent is None or len(parent) < 4:
        parent = np.full(70, 0.5)
    
    parent = np.clip(parent, 0.0, 1.0)
    parent = repair_feasible(parent)
    n_parent = len(parent)
    c_parent = exact_score(parent)
    
    best_h = parent.copy()
    best_c = c_parent
    
    # Phase 1: Try symmetrization
    h_sym = repair_feasible((parent + parent[::-1]) / 2.0)
    c_sym = exact_score(h_sym)
    if c_sym < best_c - 1e-12:
        best_h = h_sym
        best_c = c_sym
    
    # Phase 2: Binary annealing at parent dimension and nearby
    if time_left() > 100:
        dims_to_try = sorted(set([n_parent - 2, n_parent, n_parent + 2]))
        dims_to_try = [d for d in dims_to_try if 20 <= d <= 120 and d % 2 == 0]
        
        for dim in dims_to_try:
            if time_left() < 30:
                break
            t_alloc = min(25.0, time_left() * 0.15)
            for restart in range(3):
                if time_left() < 15:
                    break
                h_bin, c_bin = binary_anneal(dim, n_iter=100000,
                                             seed=seed + restart * 1000 + dim,
                                             time_limit=t_alloc / 3.0)
                # Interpolate to parent dimension and evaluate
                h_interp = interpolate_to_n(h_bin, n_parent)
                c_interp = exact_score(h_interp)
                if c_interp < best_c - 1e-10:
                    best_h = h_interp
                    best_c = c_interp
                # Also try direct binary score (at its own dimension)
                if c_bin < best_c - 1e-10:
                    # Keep track but only update best if dimension matches
                    # Actually update best regardless since we return whatever dimension
                    cand_h = repair_feasible(h_bin)
                    cand_c = exact_score(cand_h)
                    if cand_c < best_c:
                        best_h = cand_h
                        best_c = cand_c
    
    # Phase 3: Also try annealing with initial pattern from parent
    if time_left() > 80:
        h0_bin = (best_h > 0.5).astype(float)
        n_b = len(h0_bin)
        # Ensure exact sum
        ones_count = int(h0_bin.sum())
        if ones_count != n_b // 2:
            idx = np.where(h0_bin == 1.0)[0]
            if len(idx) > n_b // 2:
                remove = idx[n_b // 2:]
                h0_bin[remove] = 0.0
            else:
                zeros = np.where(h0_bin == 0.0)[0]
                add = zeros[:n_b // 2 - len(idx)]
                h0_bin[add] = 1.0
        h_bin2, c_bin2 = binary_anneal(n_b, n_iter=200000,
                                       seed=seed + 9999, time_limit=min(20.0, time_left() * 0.15))
        c_bin2_check = exact_score(h_bin2)
        if c_bin2_check < best_c - 1e-10:
            best_h = h_bin2.copy()
            best_c = c_bin2_check
    
    # Phase 4: Coordinate descent on best
    if time_left() > 60:
        h_cd, c_cd = coordinate_local(best_h, max_rounds=10, time_limit=min(30.0, time_left() * 0.2))
        if c_cd < best_c - 1e-12:
            best_h = h_cd
            best_c = c_cd
    
    # Phase 5: Active-set SLSQP
    if time_left() > 50:
        h_as, c_as = active_set_slsqp(best_h, K=8, max_rounds=20,
                                      maxiter=100, time_limit=min(40.0, time_left() * 0.3))
        if c_as < best_c - 1e-12:
            best_h = h_as
            best_c = c_as
    
    # Phase 6: LSE projected gradient
    if time_left() > 40:
        h_lse, c_lse = lse_projected_gradient(best_h, tau_start=0.005, tau_end=0.0005,
                                             max_iters=150, time_limit=min(30.0, time_left() * 0.25))
        if c_lse < best_c - 1e-12:
            best_h = h_lse
            best_c = c_lse
    
    # Phase 7: Another round of active-set with different K
    if time_left() > 40:
        for K in [5, 12]:
            if time_left() < 20:
                break
            h_as2, c_as2 = active_set_slsqp(best_h, K=K, max_rounds=15,
                                            maxiter=100, time_limit=min(20.0, time_left() * 0.3))
            if c_as2 < best_c - 1e-12:
                best_h = h_as2
                best_c = c_as2
    
    # Phase 8: Try dimension variations of best
    if time_left() > 60:
        n_best = len(best_h)
        for new_n in [n_best - 4, n_best - 2, n_best + 2, n_best + 4]:
            if time_left() < 25:
                break
            if new_n < 20 or new_n > 150 or new_n == n_best:
                continue
            h_interp = interpolate_to_n(best_h, new_n)
            h_cd2, c_cd2 = coordinate_local(h_interp, max_rounds=5,
                                            time_limit=min(10.0, time_left() * 0.15))
            if time_left() > 20:
                h_as3, c_as3 = active_set_slsqp(h_cd2, K=8, max_rounds=10,
                                                maxiter=100, time_limit=min(15.0, time_left() * 0.2))
                c_final_dim = c_as3
                h_final_dim = h_as3
            else:
                c_final_dim = c_cd2
                h_final_dim = h_cd2
            if c_final_dim < best_c - 1e-10:
                best_h = h_final_dim
                best_c = c_final_dim
    
    # Phase 9: Final polish - full SLSQP with all constraints
    if time_left() > 30:
        n = len(best_h)
        dx = 2.0 / n
        M = 2 * n - 1
        
        def obj_full(x):
            return x[-1]
        def obj_full_jac(x):
            g = np.zeros(n + 1)
            g[-1] = 1.0
            return g
        def eq_full(x):
            return x[:-1].sum() - n / 2.0
        def eq_full_jac(x):
            g = np.zeros(n + 1)
            g[:-1] = 1.0
            return g
        def ineq_full(x):
            h_x = x[:-1]
            prof = np.correlate(h_x, 1 - h_x, mode="full") * dx
            return x[-1] - prof
        def ineq_full_jac(x):
            h_x = x[:-1]
            G = build_gradient(h_x, n, dx)
            J = np.zeros((M, n + 1))
            J[:, :-1] = -G
            J[:, -1] = 1.0
            return J
        
        c_curr = exact_score(best_h)
        x0 = np.concatenate([best_h, [c_curr + 1e-5]])
        bounds = [(0.0, 1.0)] * n + [(0.0, c_curr + 0.5)]
        cons = [
            {'type': 'eq', 'fun': eq_full, 'jac': eq_full_jac},
            {'type': 'ineq', 'fun': ineq_full, 'jac': ineq_full_jac}
        ]
        try:
            res = minimize(obj_full, x0, method='SLSQP', jac=obj_full_jac,
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 200, 'ftol': 1e-13})
            if res.x is not None and np.isfinite(res.x).all():
                h_new = repair_feasible(res.x[:-1])
                c_new = exact_score(h_new)
                if c_new < best_c - 1e-13:
                    best_h = h_new
                    best_c = c_new
        except Exception:
            pass
    
    # Phase 10: Final coordinate descent polish
    if time_left() > 20:
        h_final, c_final = coordinate_local(best_h, max_rounds=5, time_limit=min(15.0, time_left() * 0.4))
        if c_final < best_c - 1e-13:
            best_h = h_final
            best_c = c_final
    
    # Final validation
    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)
    
    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            final_c = min(final_c, float(eval_result))
        elif isinstance(eval_result, tuple) and len(eval_result) >= 1:
            if isinstance(eval_result[0], (float, int)):
                final_c = min(final_c, float(eval_result[0]))
    except Exception:
        pass
    
    return best_h, float(final_c), int(len(best_h))
