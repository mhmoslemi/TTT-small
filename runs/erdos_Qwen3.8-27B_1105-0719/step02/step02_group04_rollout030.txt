    m = n // 2
    q = q_init.copy()
    best_c5 = eval_c5(build_h(q, n))
    best_q = q.copy()
    
    for tau in tau_schedule:

    h = build_h(x, n)
    raw = np.correlate(h, 1.0 - h, mode='full')
    C = raw * dx
    C_len = 2 * n - 1
    M = C.max()
    scores = (C - M) / tau
    scores = np.clip(scores, -80.0, 80.0)
    exp_s = np.exp(scores)
    Z = exp_s.sum()
    if Z < 1e-300:
        p = np.ones(C_len) / C_len
    else:
        p = exp_s / Z
    phi = M + tau * np.log(Z)
    grad_h = np.zeros(n)
    for j in range(C_len):
        w = p[j]
        if w < 1e-14:
            continue
        d = j - (n - 1)
        if d >= 0:
            grad_h[d:] += w * (1.0 - h[:n - d])
            grad_h[:n - d] -= w * h[d:]
        else:
            dp = -d
            grad_h[:n - dp] += w * (1.0 - h[dp:])
            grad_h[dp:] -= w * h[:n - dp]
    grad_h *= dx
    gq = grad_h[:m] - grad_h[n-1:m-1:-1]
    return phi, gq


def interpolate_q(q_old, n_old, n_new):
    m_old = n_old // 2
    m_new = n_new // 2
    if m_old == m_new:
        return q_old.copy()
    x_old = (np.arange(m_old) + 0.5) / m_old
    x_new = (np.arange(m_new) + 0.5) / m_new
    q_new = np.interp(x_new, x_old, q_old)
    return np.clip(q_new, 0.0, 1.0)


def make_random_q(n, rng, mode=0):
    m = n // 2
    t = np.linspace(0.0, 1.0, m)
    if mode == 0:
        freq = rng.uniform(1.0, 6.0)
        amp = rng.uniform(0.15, 0.35)
        phase = rng.uniform(0.0, 2.0 * np.pi)
        q = 0.5 + amp * np.sin(2.0 * np.pi * freq * t + phase)
    elif mode == 1:
        freq = rng.uniform(2.0, 10.0)
        amp = rng.uniform(0.1, 0.4)
        phase = rng.uniform(0.0, 2.0 * np.pi)
        q = 0.5 + amp * np.cos(2.0 * np.pi * freq * t + phase)
        q += 0.1 * rng.uniform(-1.0, 1.0, m)
    elif mode == 2:
        f1 = rng.uniform(1.0, 4.0)
        f2 = rng.uniform(3.0, 8.0)
        a1 = rng.uniform(0.1, 0.25)
        a2 = rng.uniform(0.05, 0.15)
        p1 = rng.uniform(0.0, 2.0 * np.pi)
        p2 = rng.uniform(0.0, 2.0 * np.pi)
        q = 0.5 + a1 * np.sin(2.0 * np.pi * f1 * t + p1) + a2 * np.sin(2.0 * np.pi * f2 * t + p2)
    else:
        q = 0.5 + 0.3 * rng.uniform(-1.0, 1.0, m)
    return np.clip(q, 0.0, 1.0)


def run_anneal(q_init, n, t_start, budget, reserve=30.0, max_stages=10):
    m = n // 2
    q = q_init.copy()
    best_q = q.copy()
    best_c5 = eval_c5(build_h(q, n))
    tau_schedule = [0.06, 0.036, 0.022, 0.013, 0.008, 0.0048, 0.0029, 0.0017, 0.0010, 0.0006]
    if max_stages < len(tau_schedule):
        tau_schedule = tau_schedule[:max_stages]
    bounds = [(0.0, 1.0)] * m
    for tau in tau_schedule:
        if time.time() - t_start > budget - reserve:
            break
        try:
            res = minimize(lse_phi_grad_impl, q, args=(n, tau), jac=True,
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': 400, 'gtol': 1e-9, 'maxls': 40})
            q = np.clip(res.x, 0.0, 1.0)
        except Exception:
            pass
        c5 = eval_c5(build_h(q, n))
        if c5 < best_c5 - 1e-13:
            best_c5 = c5
            best_q = q.copy()
    return best_q, best_c5


def coordinate_polish(q_init, n, t_start, budget, reserve=20.0, max_passes=50):
    m = n // 2
    q = q_init.copy()
    best_c5 = eval_c5(build_h(q, n))
    best_q = q.copy()
    h = build_h(q, n)
    step_sizes = [0.04, 0.02, 0.01, 0.005, 0.002, 0.001]
    for spass in range(max_passes):
        improved = False
        for k in range(m):
            if time.time() - t_start > budget - reserve:
                break
            for sp in step_sizes:
                for sign in [1.0, -1.0]:
                    trial = q.copy()
                    new_val = q[k] + sign * sp
                    if new_val < 0.0 or new_val > 1.0:
                        continue
                    trial[k] = new_val
                    h_trial = build_h(trial, n)
                    c5_trial = eval_c5(h_trial)
                    if c5_trial < best_c5 - 1e-12:
                        q = trial
                        h = h_trial
                        best_c5 = c5_trial
                        best_q = trial.copy()
                        improved = True
                        break
                if time.time() - t_start > budget - reserve:
                    break
            if time.time() - t_start > budget - reserve:
                break
        if not improved:
            break
    return best_q, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    reserve = max(30.0, 0.04 * budget_s)

    # Initialize from parent
    try:
        h_init = np.array(initial_h_values, dtype=float).reshape(-1)
        n_init = len(h_init)
        if n_init % 2 != 0:
            n_init -= 1
            h_init = h_init[:n_init]
        q_init = q_from_h(h_init, n_init)
        q_init = np.clip(q_init, 0.0, 1.0)
    except Exception:
        n_init = 126
        m = n_init // 2
        t = np.linspace(0.0, 1.0, m)
        q_init = np.clip(0.5 + 0.25 * np.sin(2.0 * np.pi * 3.0 * t), 0.0, 1.0)

    best_q = q_init.copy()
    best_n = n_init
    best_c5 = eval_c5(build_h(q_init, n_init))

    # Phase 1: Main LSE annealing on parent n
    best_q, c1 = run_anneal(best_q, best_n, t0, budget_s, reserve)
    if c1 < best_c5:
        best_c5 = c1
    best_q = best_q.copy()

    # Phase 2: Try nearby n values
    if time.time() - t0 < budget_s - reserve - 40:
        n_try_list = []
        for delta in [2, -2, 4, -4, 6, -6, 8, -8]:
            nc = best_n + delta
            if nc >= 40 and nc % 2 == 0 and nc not in n_try_list:
                n_try_list.append(nc)
        for nc in n_try_list:
            if time.time() - t0 > budget_s - reserve - 40:
                break
            q_try = interpolate_q(best_q, best_n, nc)
            q_try, c_try = run_anneal(q_try, nc, t0, budget_s, reserve, max_stages=7)
            if c_try < best_c5:
                best_c5 = c_try
                best_q = q_try
                best_n = nc

    # Phase 3: Coordinate polish on best solution
    if time.time() - t0 < budget_s - reserve - 20:
        best_q, c_pol = coordinate_polish(best_q, best_n, t0, budget_s, reserve, max_passes=80)
        if c_pol < best_c5:
            best_c5 = c_pol

    # Phase 4: Random restarts
    if time.time() - t0 < budget_s - reserve - 30:
        n_restarts = int(max(3, (budget_s - reserve - (time.time() - t0)) / 40.0))
        n_restarts = min(n_restarts, 15)
        for trial in range(n_restarts):
            if time.time() - t0 > budget_s - reserve - 30:
                break
            mode = trial % 4
            q_r = make_random_q(best_n, rng, mode=mode)
            q_r, c_r = run_anneal(q_r, best_n, t0, budget_s, reserve, max_stages=8)
            if c_r < best_c5:
                best_c5 = c_r
                best_q = q_r
                # Additional polish
                if time.time() - t0 < budget_s - reserve - 20:
                    best_q, c_r2 = coordinate_polish(best_q, best_n, t0, budget_s, reserve, max_passes=30)
                    if c_r2 < best_c5:
                        best_c5 = c_r2

    # Phase 5: Final fine-tuning LSE
    if time.time() - t0 < budget_s - 15:
        m = best_n // 2
        tau_fine = [0.0004, 0.0002, 0.0001]
        q_cur = best_q.copy()
        bounds = [(0.0, 1.0)] * m
        for tau in tau_fine:
            if time.time() - t0 > budget_s - 10:
                break
            try:
                res = minimize(lse_phi_grad_impl, q_cur, args=(best_n, tau), jac=True,
                             method='L-BFGS-B', bounds=bounds,
                             options={'maxiter': 300, 'gtol': 1e-11, 'maxls': 30})
                q_cur = np.clip(res.x, 0.0, 1.0)
            except Exception:
                pass
            c_cur = eval_c5(build_h(q_cur, best_n))
            if c_cur < best_c5:
                best_c5 = c_cur
                best_q = q_cur.copy()

    # Phase 6: Final coordinate polish
    if time.time() - t0 < budget_s - 5:
        best_q, c_final = coordinate_polish(best_q, best_n, t0, budget_s, reserve=5.0, max_passes=100)
        if c_final < best_c5:
            best_c5 = c_final

    # Final validation
    h_final = build_h(best_q, best_n)
    h_final = np.clip(h_final, 0.0, 1.0)
    best_c5 = eval_c5(h_final)

    return h_final, best_c5, best_n
