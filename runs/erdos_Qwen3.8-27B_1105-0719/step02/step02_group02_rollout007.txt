```python
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time

def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def c5_of_h(h, dx):
    return float(compute_corrs(h).max() * dx)

def grad_corr_k(h, k, n):
    """Gradient of the k-th element of np.correlate(h, 1-h, 'full') with respect to h."""
    offset = k - (n - 1)
    grad = np.zeros(n)
    if offset == 0:
        grad[:] = 1.0 - 2.0 * h
    elif offset > 0:
        # corr = sum_{i=offset}^{n-1} h[i] * (1-h[i-offset])
        # d/dh[p]: (1-h[p-offset]) if p >= offset; -h[p+offset] if p <= n-1-offset
        idx1 = np.arange(offset, n)
        grad[idx1] += (1.0 - h[idx1 - offset])
        idx2 = np.arange(0, n - offset)
        grad[idx2] += (-h[idx2 + offset])
    else:
        d = -offset
        # corr = sum_{i=0}^{n-1-d} h[i] * (1-h[i+d])
        # d/dh[p]: (1-h[p+d]) if p <= n-1-d; -h[p-d] if p >= d
        idx1 = np.arange(0, n - d)
        grad[idx1] += (1.0 - h[idx1 + d])
        idx2 = np.arange(d, n)
        grad[idx2] += (-h[idx2 - d])
    return grad

def smooth_max_val(h, dx, beta):
    return float(logsumexp(beta * compute_corrs(h)) / beta * dx)

def smooth_max_grad(h, dx, beta):
    n = len(h)
    corrs = compute_corrs(h)
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones_like(corrs) / len(corrs)
    else:
        w /= w_sum
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        grad += w[k] * grad_corr_k(h, k, n)
    return grad * dx

# SLSQP smooth-max solver
def slsqp_smooth(h0, dx, target_sum, beta, maxiter):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        return smooth_max_val(x, dx, beta)
    def g(x):
        return smooth_max_grad(x, dx, beta)
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h0, jac=g, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        h_out = project_feasible(res.x, target_sum)
    except Exception:
        h_out = project_feasible(h0, target_sum)
    return h_out, c5_of_h(h_out, dx)
```

```python
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time

def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def c5_of_h(h, dx):
    return float(compute_corrs(h).max() * dx)

def grad_corr_k(h, k, n):
    offset = k - (n - 1)
    grad = np.zeros(n)
    if offset == 0:
        grad[:] = 1.0 - 2.0 * h
    elif offset > 0:
        idx1 = np.arange(offset, n)
        grad[idx1] += (1.0 - h[idx1 - offset])
        idx2 = np.arange(0, n - offset)
        grad[idx2] += (-h[idx2 + offset])
    else:
        d = -offset
        idx1 = np.arange(0, n - d)
        grad[idx1] += (1.0 - h[idx1 + d])
        idx2 = np.arange(d, n)
        grad[idx2] += (-h[idx2 - d])
    return grad

class SmoothMaxProb:
    def __init__(self, dx, beta):
        self.dx = dx
        self.beta = beta
    def f(self, x):
        return float(logsumexp(self.beta * compute_corrs(x)) / self.beta * self.dx)
    def g(self, x):
        n = len(x)
        corrs = compute_corrs(x)
        shifted = corrs - corrs.max()
        w = np.exp(self.beta * shifted)
        ws = w.sum()
        if ws < 1e-30:
            w = np.ones_like(corrs) / len(corrs)
        else:
            w /= ws
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] < 1e-10:
                continue
            grad += w[k] * grad_corr_k(x, k, n)
        return grad * self.dx
    def h_eq(self, x):
        return x.sum() - self.target_sum
    def h_eq_jac(self, x):
        return np.ones(len(x))

class EpigraphProb:
    def __init__(self, n, target_sum, h_anchor, rho):
        self.n = n
        self.target_sum = target_sum
        self.h_anchor = h_anchor
        self.rho = rho
    def f(self, y):
        h = y[:self.n]
        s = y[self.n]
        dev = h - self.h_anchor
        return s + self.rho * np.dot(dev, dev) / self.n
    def g(self, y):
        h = y[:self.n]
        grad = np.empty(self.n + 1)
        dev = h - self.h_anchor
        grad[:self.n] = 2.0 * self.rho * dev / self.n
        grad[self.n] = 1.0
        return grad
    def h_eq(self, y):
        return y[:self.n].sum() - self.target_sum
    def h_eq_jac(self, y):
        j = np.zeros(self.n + 1)
        j[:self.n] = 1.0
        return j
    def c_ineq(self, y):
        h = y[:self.n]
        s = y[self.n]
        return compute_corrs(h) - s
    def c_ineq_jac(self, y):
        h = y[:self.n]
        n = self.n
        n_lags = 2 * n - 1
        J = np.zeros((n_lags, n + 1))
        for k in range(n_lags):
            J[k, :n] = grad_corr_k(h, k, n)
            J[k, n] = -1.0
        return J

def solve_smooth_max(h0, dx, target_sum, beta, maxiter):
    n = len(h0)
    prob = SmoothMaxProb(dx, beta)
    prob.target_sum = target_sum
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': prob.h_eq, 'jac': prob.h_eq_jac}]
    try:
        res = minimize(prob.f, h0, jac=prob.g, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        h_out = project_feasible(res.x, target_sum)
    except Exception:
        h_out = project_feasible(h0, target_sum)
    return h_out, c5_of_h(h_out, dx)

def solve_epigraph(h0, target_sum, rho, maxiter, s_init=None):
    n = len(h0)
    corrs0 = compute_corrs(h0)
    if s_init is None:
        s_init = corrs0.max() + 1e-8
    y0 = np.zeros(n + 1)
    y0[:n] = h0
    y0[n] = s_init
    prob = EpigraphProb(n, target_sum, h0, rho)
    bounds = [(0.0, 1.0)] * n + [(0.0, corrs0.max() + 50.0)]
    cons = [
        {'type': 'eq', 'fun': prob.h_eq, 'jac': prob.h_eq_jac},
        {'type': 'ineq', 'fun': lambda y: -prob.c_ineq(y), 'jac': lambda y: -prob.c_ineq_jac(y)}
    ]
    try:
        res = minimize(prob.f, y0, jac=prob.g, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-13})
        h_out = project_feasible(res.x[:n], target_sum)
    except Exception:
        h_out = project_feasible(h0, target_sum)
    return h_out, c5_of_h(h_out, dx_of_n(n))
```
