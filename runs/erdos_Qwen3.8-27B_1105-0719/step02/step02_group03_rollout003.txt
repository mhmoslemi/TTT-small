def project_feasible(x, target_sum):
    """Project x onto {h in [0,1]^n : sum(h) = target_sum}"""
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
        else:
            h += diff / n
    return np.clip(h, 0.0, 1.0)

def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def c5_of_h(h, dx):
    return compute_corrs(h).max() * dx

def grad_corr_k(h, k, n):
    """Gradient of c_k with respect to h."""
    offset = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -offset)
    i_hi = min(n - 1, n - 1 - offset)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx + offset] += (1.0 - h[idx])
    grad[idx] += (-h[idx + offset])
    return grad

def smooth_max_and_grad(h, dx, beta):
    """Compute smooth max of correlations and its gradient."""
    corrs = compute_corrs(h)
    m = corrs.max()
    shifted = corrs - m
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    w = w / w_sum
    obj = (m + np.log(w_sum) / beta) * dx
    
    grad = np.zeros(len(h))
    n = len(h)
    for k in range(len(corrs)):
        if w[k] < 1e-14:
            continue
        grad += w[k] * grad_corr_k(h, k, n)
    grad *= dx
    return obj, grad

def projected_gradient_descent(h0, dx, target_sum, beta, max_iter, alpha_init):
    """Projected gradient descent with backtracking line search."""
    h = h0.copy().astype(float)
    best_h = h.copy()
    best_val = c5_of_h(h, dx)
    
    alpha = alpha_init
    for it in range(max_iter):
        obj, grad = smooth_max_and_grad(h, dx, beta)
        
        # Backtracking line search
        for _ in range(30):
            h_new = project_feasible(h - alpha * grad, target_sum)
            obj_new = smooth_max_and_grad(h_new, dx, beta)[0]
            if obj_new < obj - 1e-10:
                break
            alpha *= 0.5
        else:
            break
        
        h = h_new
        true_val = c5_of_h(h, dx)
        if true_val < best_val:
            best_val = true_val
            best_h = h.copy()
    
    return best_h, best_val

def binary_steepest_descent(h0, dx, rng, max_sweeps=100):
    """Steepest descent for binary arrangements."""
    n = len(h0)
    target_sum = int(round(n / 2))
    
    h = np.zeros(n)
    sorted_idx = np.argsort(-h0)
    h[sorted_idx[:target_sum]] = 1.0
    
    corrs = compute_corrs(h)
    best_c5 = corrs.max() * dx
    best_h = h.copy()
    
    for sweep in range(max_sweeps):
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        best_swap = None
        best_new_max = corrs.max()
        
        for i in ones:
            for j in zeros:
                # Compute delta for swapping h[i]:1->0 and h[j]:0->1
                delta = _swap_delta(h, i, j, n)
                new_max = (corrs + delta).max()
                if new_max < best_new_max - 1e-14:
                    best_new_max = new_max
                    best_swap = (i, j)
        
        if best_swap is None:
            break
        
        i, j = best_swap
        h[i], h[j] = 0.0, 1.0
        corrs = compute_corrs(h)
        new_c5 = corrs.max() * dx
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
    
    return best_h, best_c5

def _swap_delta(h, i, j, n):
    """Compute change in all correlations when h[i]:1->0 and h[j]:0->1."""
    # h[i] changes from 1 to 0, h[j] changes from 0 to 1
    # For correlation c_k = sum_l h[l] * (1-h[l+k-(n-1)]):
    # Changing h[i] from 1 to 0: delta = -(1-h[i+k-(n-1)]) for valid l=i
    #                                  + h[i] for valid l where l+k-(n-1)=i
    # Wait, let me think more carefully.
    
    # c_k = sum_{l: valid} h[l] * (1 - h[l + k - (n-1)])
    # Let d = k - (n-1), so c_k = sum_{l: 0<=l<n, 0<=l+d<n} h[l]*(1-h[l+d])
    
    # If we change h[i] from 1 to 0:
    #   Terms where l=i: h[i]*(1-h[i+d]) changes from (1-h[i+d]) to 0, delta = -(1-h[i+d]) = -（1-h[i+d])
    #   Terms where l+d=i, i.e., l=i-d: h[i-d]*(1-h[i]) changes from (1-h[i-d]) to 0... 
    #   Wait, h[i] appears in (1-h[i]) when l+d=i, so (1-h[i]) changes from 0 to 1
    #   So delta = h[i-d]*(1-0) - h[i-d]*(1-1) = h[i-d]
    
    # Actually let me just compute it directly:
    # c_k depends on h[l] and h[l+d] for valid l.
    # h[i] appears as:
    #   h[i] in the first factor when l=i (if i+d is valid)
    #   h[i] in (1-h[i]) when l+d=i, i.e., l=i-d (if i-d is valid)
    
    delta = np.zeros(2 * n - 1)
    
    # h[i] changes from 1 to 0:
    # As first factor (l=i): affects c_k where d = k-(n-1), and l=i, l+d=i+d valid
    #   c_k has term h[i]*(1-h[i+d]), changes by -(1-h[i+d]) = h[i+d]-1
    #   This affects lag d = i+d - i... wait, d is the shift.
    #   For shift d, the term at l=i exists if 0<=i<n and 0<=i+d<n
    #   So for each d where 0<=i+d<n (i.e., -i <= d <= n-1-i):
    #     delta at shift d: -(1-h[i+d]) = h[i+d]-1
    #   In correlate indices: k = d + (n-1)
    
    # As second factor (l+d=i, so l=i-d): affects c_k where l=i-d, and 0<=i-d<n
    #   c_k has term h[i-d]*(1-h[i]), h[i] changes 1->0, so (1-h[i]) changes 0->1
    #   delta = h[i-d]*1 - h[i-d]*0 = h[i-d]
    #   For each d where 0<=i-d<n (i.e., i-n+1 <= d <= i):
    #     delta at shift d: h[i-d]
    
    # h[j] changes from 0 to 1:
    # As first factor (l=j): 
    #   For each d where 0<=j+d<n:
    #     delta at shift d: (1-h[j+d]) [was 0, now 1-h[j+d]]
    
    # As second factor (l+d=j, so l=j-d):
    #   For each d where 0<=j-d<n:
    #     delta at shift d: -h[j-d] [h[j] was 0 so (1-h[j])=1, now (1-h[j])=0, so h[j-d]*1 -> h[j-d]*0, delta = -h[j-d]]
    
    # Let's compute using the array of shifts d = -(n-1), ..., 0, ..., n-1
    # Correlate index k corresponds to shift d = k - (n-1)
    
    # For position i (1->0):
    # As first factor: for d in [-i, n-1-i], delta[d] -= (1-h[i+d])
    d_min1 = -i
    d_max1 = n - 1 - i
    if d_min1 <= d_max1:
        d_range = np.arange(d_min1, d_max1 + 1)
        idx = i + d_range  # positions i+d
        delta[d_range + (n-1)] -= (1.0 - h[idx])
    
    # As second factor: for d in [i-n+1, i], delta[d] += h[i-d]
    d_min2 = i - n + 1
    d_max2 = i
    if d_min2 <= d_max2:
        d_range = np.arange(d_min2, d_max2 + 1)
        idx = i - d_range  # positions i-d
        delta[d_range + (n-1)] += h[idx]
    
    # For position j (0->1):
    # As first factor: for d in [-j, n-1-j], delta[d] += (1-h[j+d])
    d_min3 = -j
    d_max3 = n - 1 - j
    if d_min3 <= d_max3:
        d_range = np.arange(d_min3, d_max3 + 1)
        idx = j + d_range
        delta[d_range + (n-1)] += (1.0 - h[idx])
    
    # As second factor: for d in [j-n+1, j], delta[d] -= h[j-d]
    d_min4 = j - n + 1
    d_max4 = j
    if d_min4 <= d_max4:
        d_range = np.arange(d_min4, d_max4 + 1)
        idx = j - d_range
        delta[d_range + (n-1)] -= h[idx]
    
    return delta

def binary_simulated_annealing(h0, dx, rng, max_iters, init_temp, cooling, patience):
    """Simulated annealing for binary arrangements."""
    n = len(h0)
    target_sum = int(round(n / 2))
    
    h = np.zeros(n)
    sorted_idx = np.argsort(-h0)
    h[sorted_idx[:target_sum]] = 1.0
    
    corrs = compute_corrs(h)
    best_c5 = corrs.max() * dx
    best_h = h.copy()
    
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    
    no_improve = 0
    temp = init_temp
    
    for it in range(max_iters):
        if it % 3000 == 0:
            temp *= cooling
        
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        delta = _swap_delta(h, i, j, n)
        new_corrs = corrs + delta
        new_c5 = new_corrs.max() * dx
        
        if new_c5 <= best_c5 + 1e-15:
            h[i], h[j] = 0.0, 1.0
            corrs = new_corrs
            best_c5 = new_c5
            best_h = h.copy()
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            no_improve = 0
        else:
            delta_val = new_c5 - best_c5
            if delta_val > 1e-15:
                prob = np.exp(-delta_val / max(temp, 1e-10))
                if rng.random() < prob:
                    h[i], h[j] = 0.0, 1.0
                    corrs = new_corrs
                    ones = np.where(h == 1.0)[0]
                    zeros = np.where(h == 0.0)[0]
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1
        
        if no_improve > patience:
            break
    
    return best_h, best_c5

def make_binary_start(n, rng, mode=0):
    """Create a binary starting arrangement."""
    h = np.zeros(n)
    k = n // 2
    if mode == 0:
        # Golden ratio
        phi = (1.0 + np.sqrt(5.0)) / 2.0
        for i in range(k):
            pos = int((i * phi) * n) % n
            h[pos] = 1.0
    elif mode == 1:
        # Van der Corput
        for i in range(k):
            # Reverse binary representation
            x = i + 1
            r = 0.0
            f = 1.0
            while x > 0:
                r += f * (x % 2)
                x //= 2
                f /= 2.0
            pos = int(r * n) % n
            h[pos] = 1.0
    elif mode == 2:
        # Uniform with jitter
        positions = np.arange(k) * (n / k) + rng.uniform(-0.3, 0.3, k)
        positions = np.sort(positions)
        for pos in positions:
            h[int(pos) % n] = 1.0
    else:
        # Random
        idx = rng.choice(n, size=k, replace=False)
        h[idx] = 1.0
    
    # Fix if duplicates occurred
    ones_count = h.sum()
    if ones_count != k:
        # Reset and redo
        h = np.zeros(n)
        idx = rng.choice(n, size=k, replace=False)
        h[idx] = 1.0
    
    return project_feasible(h, n / 2.0)
