def solve_lp_scipy(h, beam, trust_radius, max_active):

    if len(active_idx) > max_active:
        top_local = np.argsort(profile[active_idx])[-max_active:]
        active_idx = active_idx[top_local]
    if len(active_idx) == 0:
        active_idx = np.array([np.argmax(profile)])
    
    # Build gradient matrix for active rows
    G = np.zeros((len(active_idx), n))
    for idx, m in enumerate(active_idx):
        d = int(m - (n - 1))
        G[idx] = gradient_row(h, d, n, dx)
    
    b = profile[active_idx] - G @ h
    r = trust_radius
    lower = np.clip(h - r, 0.0, 1.0)
    upper = np.clip(h + r, 0.0, 1.0)
    
    # LP: minimize t
    # x = [h' (n), t (1)]
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    # Constraints: t - G @ h' >= b  ⟹  -t + G @ h' <= -b
    A_ub = np.zeros((len(active_idx), n + 1))
    A_ub[:, :-1] = G
    A_ub[:, -1] = -1.0
    b_ub = -b
    
    # Equality: sum(h') = n/2
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :-1] = 1.0
    b_eq = np.array([n / 2.0])
    
    bounds = list(zip(lower, upper)) + [(-np.inf, np.inf)]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.status != 0:
        return None
    
    h_new = res.x[:n]
    h_new = repair_feasible(h_new)
    return h_new

def repair_feasible(h):
    h = np.clip(np.asarray(h, dtype=np.float64), 0.0, 1.0)
    n = len(h)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx])
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-13:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]

            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-13:
