        b_eq = np.zeros(1)

        bounds = []

            break

        if t_new >= cur_c5 - 1e-10:
            # No descent found
            break
        
        # Apply the direction with exact line search
        # h_new = h + step * delta
        # Find max step such that h_new stays in [0,1]
        max_step = 1.0
        for i in range(n):
            if delta[i] > 1e-15:
                max_step = min(max_step, (1.0 - h[i]) / delta[i])
            elif delta[i] < -1e-15:
                max_step = min(max_step, h[i] / (-delta[i]))

        step = min(max_step, 1.0)
        h_new = h + step * delta
        h_new = project_feasible(h_new, target_sum)
        new_c5 = exact_c5(h_new)

                h_new = project_feasible(h_new, target_sum)
                new_c5 = exact_c5(h_new)
                if new_c5 < cur_c5 - 1e-12:
                    h = h_new
                    break
            else:
                break
    
    return best_h, best_c5
