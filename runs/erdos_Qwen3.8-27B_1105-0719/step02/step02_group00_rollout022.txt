def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for idx in order:
            add = min(remaining, capacity[idx] + 1e-15)
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for idx in order:
            rem = min(remaining, h[idx] + 1e-15)
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def logsumexp_grad(h, beta):
    """Compute gradient of log-sum-exp of profile."""
    n = len(h)
    M = 2 * n - 1
    dx = 2.0 / n
    profile = compute_profile(h)
    Pmax = np.max(profile)
    s = beta * (profile - Pmax)
    s_shifted = s - np.max(s)
    exp_s = np.exp(s_shifted)
    weights = exp_s / exp_s.sum()
    
    # Gradient: sum_m weights[m] * dx * (1 - 2*h[m-i]) for valid m-i
    # Equivalent: for each i, grad[i] = dx * sum_{m: 0<=m-i<n} weights[m] * (1 - 2*h[m-i])
    # Let j = m-i, so m = i+j, and 0<=j<n means i<=m<i+n
    # grad[i] = dx * sum_{j=0}^{n-1} weights[i+j] * (1 - 2*h[j])  (with bounds on i+j < M)
    
    grad = np.zeros(n)
    for i in range(n):
        # j ranges from 0 to n-1, m = i+j ranges from i to i+n-1
        # Need m < M = 2n-1, so i+j < 2n-1, i.e., j < 2n-1-i
        j_max = min(n, M - i)
        m_range = np.arange(i, i + j_max)
        j_range = np.arange(j_max)
        grad[i] = dx * np.sum(weights[m_range] * (1.0 - 2.0 * h[j_range]))
    
    return float(Pmax + np.log(np.sum(exp_s)) / beta), grad

def logsumexp_bfgs(h_start, beta, lam, maxiter=300, time_check=None, t_start=None):
    """Optimize log-sum-exp objective with L-BFGS-B."""
    n = len(h_start)
    
    def obj(x):
        h = np.clip(x, 0.0, 1.0)
        profile = compute_profile(h)
        Pmax = np.max(profile)
        s = beta * (profile - Pmax)
        s_shifted = s - np.max(s)
        log_sum = np.max(s) + np.log(np.sum(np.exp(s_shifted)))
        loss = Pmax + log_sum / beta
        sum_err = h.sum() - n / 2.0
        loss += lam * sum_err * sum_err
        return loss
    
    def obj_grad(x):
        h = np.clip(x, 0.0, 1.0)
        , grad = logsumexp_grad(h, beta)
        sum_err = h.sum() - n / 2.0
        grad += 2.0 * lam * sum_err * np.ones(n)
        return grad
    
    x0 = h_start.copy()
    bounds = [(0.0, 1.0)] * n
    
    try:
        res = minimize(obj, x0, jac=obj_grad, method='L-BFGS-B',
                      bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-6})
        h_new = repair_feasible(res.x)
        return h_new
    except Exception:
        return None

def projected_logsumexp_descent(h_start, beta, max_iters=500, step_init=0.01):
    """Projected gradient descent on log-sum-exp objective."""
    n = len(h_start)
    h = h_start.copy()
    best_c = exact_score(h)
    step = step_init
    momentum = 0.3
    velocity = np.zeros(n)

            j_range = np.arange(j_max)
            grad[i] = dx * np.sum(weights[m_range] * (1.0 - 2.0 * h[j_range]))
        
        # Zero-mean to handle sum constraint implicitly
        grad -= grad.mean()

        h_try = repair_feasible(h + velocity)
        c_try = exact_score(h_try)

            best_c = c_try
            h = h_try

        else:
            step *= 0.7
            if step < 1e-7:

    if ones > target:
        one_idx = np.where(bits > 0.5)[0]
        to_remove = one_idx[ones - target:ones]
        bits[to_remove] = 0.0
    elif ones < target:
        zero_idx = np.where(bits < 0.5)[0]
        to_add = zero_idx[:target - ones]
        bits[to_add] = 1.0
    profile = compute_profile(bits)
    best_score = np.max(profile)
    dx = 2.0 / n
    if rng is None:
        rng = np.random.RandomState(0)
    for it in range(max_iters):
        i = rng.randint(n)
        if bits[i] == 1.0:
            candidates = np.where(bits < 0.5)[0]
            if len(candidates) == 0:
                break
            j = candidates[rng.randint(len(candidates))]
        else:
            candidates = np.where(bits > 0.5)[0]
            if len(candidates) == 0:
                break
            j = candidates[rng.randint(len(candidates))]
        if i == j:
            continue
        bits[i], bits[j] = bits[j], bits[i]
        bits[i] = 1.0
        bits[j] = 0.0
        prof = compute_profile(bits)
        sc = float(np.max(prof))
        if sc < best_score:
            best_score = sc
        else:
            bits[i], bits[j] = bits[j], bits[i]
    return bits.astype(float)

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s - (time.perf_counter() - t_start)

    parent = None
    try:
        parent = np.array(initial_h_values, dtype=np.float64)
    except NameError:
        pass
    except Exception:
        pass

    if parent is None or len(parent) < 4:
        parent = np.full(70, 0.5)
    parent = np.clip(parent, 0.0, 1.0)
    parent = repair_feasible(parent)

    best_h = parent.copy()
    best_c = exact_score(parent)

    h_sym = repair_feasible(symmetrize(parent))
    c_sym = exact_score(h_sym)
    if c_sym < best_c - 1e-13:
        best_h = h_sym
        best_c = c_sym

    n = len(best_h)

    # Stage 1: Log-sum-exp projected descent with multiple betas
    if time_left() > 100:
        for beta in [200, 500, 1000, 2000, 5000]:
            if time_left() < 30:
                break
            h_try, c_try = lse_projected_descent(best_h, beta, max_iters=200, step_init=0.008, momentum=0.4)
            if c_try < best_c - 1e-12:
                best_h = h_try.copy()
                best_c = c_try
            # Also try from a perturbed start
            if time_left() < 20:
                break
            h_pert = repair_feasible(best_h + 0.01 * (rng.randn(n) - rng.randn(n).mean()))
            h_try2, c_try2 = lse_projected_descent(h_pert, beta, max_iters=150, step_init=0.005, momentum=0.5)
            if c_try2 < best_c - 1e-12:
                best_h = h_try2.copy()
                best_c = c_try2

    # Stage 2: L-BFGS-B polishing
    if time_left() > 80:
        for beta in [1000, 3000, 10000]:
            if time_left() < 25:
                break
            h_sol = lse_bfgs(best_h, beta, lam=50.0, maxiter=200)
            if h_sol is not None:
                c_sol = exact_score(h_sol)
                if c_sol < best_c - 1e-12:
                    best_h = h_sol.copy()
                    best_c = c_sol
        # Follow up with projected descent
        if time_left() > 30:
            h_try, c_try = lse_projected_descent(best_h, 5000, max_iters=200, step_init=0.003, momentum=0.5)
            if c_try < best_c - 1e-12:
                best_h = h_try.copy()
                best_c = c_try

    # Stage 3: Active constraint SLSQP
    if time_left() > 80:
        for k in [8, 12, 16]:
            if time_left() < 25:
                break
            h_sol = active_slsqp(best_h, k_active=k, maxiter=200)
            if h_sol is not None:
                c_sol = exact_score(h_sol)
                if c_sol < best_c - 1e-12:
                    best_h = h_sol.copy()
                    best_c = c_sol
            if time_left() > 20:
                h_try, c_try = lse_projected_descent(best_h, 5000, max_iters=100, step_init=0.002, momentum=0.5)
                if c_try < best_c - 1e-12:
                    best_h = h_try.copy()
                    best_c = c_try

    # Stage 4: Dimension exploration
    if time_left() > 120:
        n_cur = len(best_h)
        dims = []
        for d in range(max(30, n_cur - 20), min(130, n_cur + 30), 2):
            if d != n_cur:
                dims.append(d)
        dims.sort(key=lambda d: abs(d - n_cur))
        for new_n in dims[:8]:
            if time_left() < 40:
                break
            h_interp = interpolate_to_n(best_h, new_n)
            # Quick LSE descent at this dimension
            h_try, c_try = lse_projected_descent(h_interp, 1000, max_iters=150, step_init=0.005, momentum=0.4)
            if c_try < best_c - 1e-10:
                best_h = h_try.copy()
                best_c = c_try
                # Polish
                if time_left() > 20:
                    h_sol = lse_bfs = lse_bfgs(h_try, 5000, lam=80.0, maxiter=150)
                    if h_sol is not None:
                        c_sol = exact_score(h_sol)
                        if c_sol < best_c - 1e-12:
                            best_h = h_sol.copy()
                            best_c = c_sol

    # Stage 5: Binary-based initialization
    if time_left() > 150:
        n_b = len(best_h)
        bits_init = (best_h > 0.5).astype(float)
        bits_opt = binary_hill_climb(bits_init, max_iters=3000, rng=rng)
        for smooth_val in [0.05, 0.1, 0.15, 0.2]:
            if time_left() < 30:
                break
            h_bin = binary_to_continuous(bits_opt, n_b, smooth=smooth_val)
            h_try, c_try = lse_projected_descent(h_bin, 800, max_iters=150, step_init=0.01, momentum=0.4)
            if c_try < best_c - 1e-10:
                best_h = h_try.copy()
                best_c = c_try
                if time_left() > 20:
                    h_sol = lse_bfgs(h_try, 5000, lam=50.0, maxiter=150)
                    if h_sol is not None:
                        c_sol = exact_score(h_sol)
                        if c_sol < best_c - 1e-12:
                            best_h = h_sol.copy()
                            best_c = c_sol
        # Try at a different dimension
        if time_left() > 60:
            n_b2 = n_b + 4 if n_b + 4 <= 120 else n_b - 4
            h_b2 = interpolate_to_n(best_h, n_b2)
            bits2 = binary_hill_climb((h_b2 > 0.5).astype(float), max_iters=2000, rng=rng)
            for smooth_val in [0.08, 0.15]:
                if time_left() < 25:
                    break
                h_bin2 = binary_to_continuous(bits2, n_b2, smooth=smooth_val)
                h_try2, c_try2 = lse_projected_descent(h_bin2, 800, max_iters=120, step_init=0.01, momentum=0.4)
                if c_try2 < best_c - 1e-10:
                    best_h = h_try2.copy()
                    best_c = c_try2

    # Stage 6: Multi-restart LSE with random perturbations
    if time_left() > 80:
        n_cur = len(best_h)
        for eps in [0.02, 0.05, 0.1]:
            if time_left() < 20:
                break
            z = rng.randn(n_cur)
            z -= z.mean()
            h_pert = repair_feasible(best_h + eps * z)
            h_try, c_try = lse_projected_descent(h_pert, 1500, max_iters=200, step_init=0.008, momentum=0.35)
            if c_try < best_c - 1e-12:
                best_h = h_try.copy()
                best_c = c_try
                h_sol = lse_bfgs(h_try, 5000, lam=50.0, maxiter=150)
                if h_sol is not None:
                    c_sol = exact_score(h_sol)
                    if c_sol < best_c - 1e-12:
                        best_h = h_sol.copy()
                        best_c = c_sol

    # Stage 7: Alternate LSE high-beta and SLSQP
    if time_left() > 60:
        for _ in range(3):
            if time_left() < 25:
                break
            h_try, c_try = lse_projected_descent(best_h, 8000, max_iters=150, step_init=0.002, momentum=0.6)
            if c_try < best_c - 1e-13:
                best_h = h_try.copy()
                best_c = c_try
            h_sol = active_slsqp(best_h, k_active=10, maxiter=150)
            if h_sol is not None:
                c_sol = exact_score(h_sol)
                if c_sol < best_c - 1e-13:
                    best_h = h_sol.copy()
                    best_c = c_sol

    # Stage 8: Final fine-tuning
    if time_left() > 30:
        h_sol = lse_bfgs(best_h, 20000, lam=100.0, maxiter=300)
        if h_sol is not None:
            c_sol = exact_score(h_sol)
            if c_sol < best_c - 1e-13:
                best_h = h_sol.copy()
                best_c = c_sol
        h_try, c_try = lse_projected_descent(best_h, 10000, max_iters=200, step_init=0.001, momentum=0.7)
        if c_try < best_c - 1e-14:
            best_h = h_try.copy()
            best_c = c_try

    # Final validation
    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)

    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            final_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 2:
            final_c = float(eval_result[0])
    except Exception:
        pass

    return best_h, final_c, len(best_h)
