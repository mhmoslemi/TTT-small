import numpy as np
from scipy.optimize import minimize, linprog, differential_evolution
import time

def eval_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def corr_array(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def build_sym_h(q):
    m = len(q)
    h = np.empty(2 * m)
    h[:m] = q
    h[m:] = 1.0 - q[::-1]
    return h

def sym_from_h(h):
    n = len(h)
    m = n // 2
    q = np.empty(m)
    for i in range(m):
        q[i] = 0.5 * (h[i] + (1.0 - h[n - 1 - i]))
    return np.clip(q, 0.0, 1.0)

def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(x + mid, 0.0, 1.0).sum()
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    h = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        interior = (h > 1e-10) & (h < 1.0 - 1e-10)
        if interior.sum() > 0:
            h[interior] += diff / interior.sum()
    return np.clip(h, 0.0, 1.0)

def grad_q_for_all_lags(q):
    """Compute gradients of all correlations with respect to q.
    Returns array of shape (2*n-1, m) where n=2*m."""
    m = len(q)
    n = 2 * m
    h = build_sym_h(q)
    dx = 2.0 / n
    corrs = corr_array(h)
    num_lags = 2 * n - 1
    grads = np.zeros((num_lags, m))
    for k in range(num_lags):
        d = k - (n - 1)
        i_lo = max(0, -d)
        i_hi = min(n - 1, n - 1 - abs(d))
        if i_lo > i_hi:
            continue
        idx = np.arange(i_lo, i_hi + 1)

        # grad_h[i+d] += -h[i] for i in idx

        gh[idx + d] += (-h[idx])
        # grad_q[r] = gh[r] - gh[n-1-r]
        for r in range(m):
            grads[k, r] = gh[r] - gh[n - 1 - r]
    grads *= dx
    return corrs, grads

def select_active_lags(corrs, margin_frac=0.02, neighbor_pad=3, max_k=35):
    max_val = corrs.max()
    threshold = max_val - max(0.005, margin_frac * max_val)
    active = set(np.where(corrs >= threshold)[0].tolist())
    padded = set()
    for k in active:
        for dd in range(-neighbor_pad, neighbor_pad + 1):
            if 0 <= k + dd < len(corrs):
                padded.add(k + dd)
    active_list = sorted(padded)
    if len(active_list) > max_k:
        scored = sorted([(corrs[k], k) for k in active_list], reverse=True)
        active_list = sorted([k for _, k in scored[:max_k]])
    return active_list

def solve_lp_active(q, corrs, grads, active_lags):
    """Solve the active-set LP to find a direction.
    Variables: q[0..m-1], tau
    min tau
    s.t. grad_k · q - tau <= corrs[k] - grad_k · q  (for active k)
         0 <= q[i] <= 1
    """
    m = len(q)
    n_vars = m + 1
    # Objective: minimize tau
    c = np.zeros(n_vars)
    c[-1] = 1.0
    
    A_ub = []
    b_ub = []
    for k in active_lags:
        row = np.zeros(n_vars)
        row[:m] = grads[k]
        row[-1] = -1.0
        A_ub.append(row)
        b_ub.append(corrs[k] - np.dot(grads[k], q))
    
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    bounds = [(0.0, 1.0)] * m + [(-0.5, 2.0)]  # tau bounds
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x[:m]
    except Exception:
        pass
    return None

def lp_local_refine(q, max_iters=200, time_limit=None, t0=None):

    m = len(q)
    current = np.clip(q.copy(), 0, 1)
    h = build_sym_h(current)
    current_score = eval_score(h)
    
    for it in range(max_iters):
        if t0 is not None and time.time() > time_limit:
            break
        
        corrs, grads = grad_q_for_all_lags(current)
        active = select_active_lags(corrs, margin_frac=0.02, neighbor_pad=3, max_k=35)
        
        q_lp = solve_lp_active(current, corrs, grads, active)
        if q_lp is None:
            break
        
        # Line search along direction
        direction = q_lp - current
        norm_dir = np.linalg.norm(direction)
        if norm_dir < 1e-10:
            break
        
        best_alpha = None
        best_new_score = current_score
        for alpha in [1.0, 0.75, 0.5, 0.25, 0.125, 0.0625]:
            q_try = np.clip(current + alpha * direction, 0, 1)
            h_try = build_sym_h(q_try)
            sc = eval_score(h_try)
            if sc < best_new_score:
                best_new_score = sc
                best_alpha = alpha
        
        if best_alpha is None:
            break
        
        current = np.clip(current + best_alpha * direction, 0, 1)
        current_score = best_new_score
    
    return current, current_score

def coordinate_sweep(q, time_limit=None, t0=None):
    """Exact 1D coordinate descent with golden section search."""
    m = len(q)
    current = q.copy()
    h = build_sym_h(current)
    current_score = eval_score(h)
    
    # Order coordinates by gradient magnitude
    _, grads = grad_q_for_all_lags(current)
    max_corr = np.abs(grads).max(axis=0)
    order = np.argsort(-max_corr)
    
    for r in order:
        if t0 is not None and time.time() > time_limit:
            break
        
        # Golden section search for coordinate r
        gr = 0.6180339887498949
        a, b = 0.0, 1.0
        c = a + (1 - gr) * (b - a)
        d = a + gr * (b - a)
        
        def obj_1d(val):
            q_try = current.copy()
            q_try[r] = val
            h_try = build_sym_h(q_try)
            return eval_score(h_try)
        
        fc = obj_1d(c)
        fd = obj_1d(d)
        
        for _ in range(30):
            if t0 is not None and time.time() > time_limit:
                break
            if fc < fd:
                b = d
                d = c
                fd = fc
                c = a + (1 - gr) * (b - a)
                fc = obj_1d(c)
            else:
                a = c
                c = d
                fc = fd
                d = a + gr * (b - a)
                fd = obj_1d(d)
        
        best_val = (a + b) / 2.0
        fb = obj_1d(best_val)
        
        # Also check the current value
        if current_score < fb:
            best_val = current[r]
            fb = current_score
        
        if fb < current_score - 1e-12:
            current[r] = best_val
            current_score = fb
    
    return current, current_score

def transfer_to_n(h_old, n_new):

def smooth_max_optimize(q, n_full, beta, maxiter, time_limit):
    """Quick smooth max L-BFGS pass."""
    m = n_full // 2
    
    def obj(xx):
        h = build_sym_h(xx)
        corrs = corr_array(h)
        shifted = corrs - corrs.max()
        w = np.exp(beta * shifted)
        w_sum = w.sum()
        if w_sum < 1e-30:
            return corrs.max()
        return corrs.max() + np.log(w_sum) / beta
    
    def jac(xx):
        h = build_sym_h(xx)
        n = n_full
        dx = 2.0 / n
        corrs = np.correlate(h, 1.0 - h, mode='full') * dx
        max_c = corrs.max()
        shifted = corrs - max_c
        w = np.exp(beta * shifted)
        w_sum = w.sum()
        if w_sum < 1e-30:
            return np.zeros(m)
        w /= w_sum
        
        gq = np.zeros(m)
        for k in range(2 * n - 1):
            if w[k] < 1e-14:
                continue
            d = k - (n - 1)
            i_lo = max(0, -d)
            i_hi = min(n - 1, n - 1 - abs(d))
            if i_lo > i_hi:
                continue
            idx = np.arange(i_lo, i_hi + 1)
            gh = np.zeros(n)
            gh[idx] += (1.0 - h[idx + d])
            gh[idx + d] += (-h[idx])
            for r in range(m):
                gq[r] += w[k] * (gh[r] - gh[n - 1 - r])
        gq *= dx
        return gq
    
    try:
        res = minimize(obj, q, jac=jac, method='L-BFGS-B',
                      bounds=[(0, 1)] * m,
                      options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10})
        q_out = np.clip(res.x, 0, 1)
    except Exception:
        q_out = q.copy()
    
    h_out = build_sym_h(q_out)
    return q_out, eval_score(h_out)

def make_structured_start(m, rng, style=0):

    elif style == 2:
        # Block pattern
        n_blocks = rng.randint(3, 7)
        block_size = m // n_blocks
        heights = rng.uniform(0.1, 0.9, n_blocks)
        q = np.repeat(heights, block_size + (1 if np.sum([1 for b in range(n_blocks) if b < m - block_size*n_blocks]) == 0 else m - block_size*n_blocks))
        # Fix length
        if len(q) < m:
            q = np.concatenate([q, np.full(m - len(q), 0.5)])
        q = q[:m]
    elif style == 3:
        # Triangle
        t = np.abs(2 * x - 1)
        q = 0.2 + 0.6 * t
        if rng.rand() > 0.5:

    elif style == 4:
        # Random smooth
        smooth = rng.uniform(0.2, 0.8, min(m, 8))
        x_sm = np.linspace(0, 1, min(m, 8))
        q = np.interp(x, x_sm, smooth)
    else:
        q = 0.5 + 0.3 * rng.randn(m)
    return np.clip(q, 0, 1)

    elif style == 3:
        t = np.abs(2 * x - 1)
        q = 0.15 + 0.7 * t if rng.rand() > 0.5 else 0.85 - 0.7 * t
    else:
        n_pts = min(m, rng.randint(4, 10))
        xp = np.linspace(0, 1, n_pts)
        yp = rng.uniform(0.1, 0.9, n_pts)
        q = np.interp(x, xp, yp)
    return np.clip(q, 0, 1)

def refine_at_resolution(h_in, rng, t0, time_limit):
    n = len(h_in)
    if n % 2 != 0:
        return h_in, eval_score(h_in)
    q = sym_from_h(h_in)
    sc = eval_score(h_in)
    if time.time() > time_limit:
        return h_in, sc
    tl = min(time_limit, time.time() + max(5, (time_limit - time.time()) * 0.25))
    q_w = smooth_max_opt(q, n, beta=200, maxiter=80, t0=t0, time_limit=tl)
    sc_w = eval_score(build_sym_h(q_w))
    if sc_w < sc:
        q = q_w
        sc = sc_w
    if time.time() > time_limit:
        return build_sym_h(q), sc
    tl2 = min(time_limit, time.time() + max(5, (time_limit - time.time()) * 0.35))
    q_lp, sc_lp = lp_refine(q, max_iters=300, t0=t0, time_limit=tl2)
    if sc_lp < sc:
        q = q_lp
        sc = sc_lp
    if time.time() > time_limit:
        return build_sym_h(q), sc
    tl3 = min(time_limit, time.time() + max(3, (time_limit - time.time()) * 0.3))
    q_cs, sc_cs = coord_sweep(q, t0=t0, time_limit=tl3, n_its=30)
    if sc_cs < sc:
        q = q_cs
        sc = sc_cs
    if time.time() > time_limit:
        return build_sym_h(q), sc
    tl4 = min(time_limit, time.time() + max(3, (time_limit - time.time()) * 0.2))
    q_bin, sc_bin = binary_local_search(q, rng, t0=t0, time_limit=tl4, max_rounds=30)
    if sc_bin < sc:
        q = q_bin
        sc = sc_bin
    if sc < sc + 1e-14 and time.time() < time_limit - 5:
        tl5 = min(time_limit, time.time() + 15)
        q_lp2, sc_lp2 = lp_refine(q, max_iters=200, t0=t0, time_limit=tl5)
        if sc_lp2 < sc:
            q = q_lp2
            sc = sc_lp2
        tl6 = min(time_limit, time.time() + 10)
        q_cs2, sc_cs2 = coord_sweep(q, t0=t0, time_limit=tl6, n_its=25)
        if sc_cs2 < sc:
            q = q_cs2
            sc = sc_cs2
    return build_sym_h(q), sc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')

    try:
        h_parent = np.array(initial_h_values, dtype=float).copy().reshape(-1)
    except (NameError, AttributeError):
        h_parent = None

    if h_parent is not None and len(h_parent) > 10:
        if abs(np.sum(h_parent) - len(h_parent) / 2.0) > 0.5:
            h_parent = project_feasible(h_parent, len(h_parent) / 2.0)
        h_parent = np.clip(h_parent, 0.0, 1.0)
        h_parent = project_feasible(h_parent, len(h_parent) / 2.0)
        best_h = h_parent.copy()
        best_c5 = eval_score(h_parent)
    else:
        m_fb = 30
        x = np.linspace(0, 1, m_fb)
        q_fb = 0.5 + 0.3 * np.sin(2 * np.pi * x)
        best_h = build_sym_h(q_fb)
        best_c5 = eval_score(best_h)

    deadline = t0 + budget_s

    # Phase 1: Multi-resolution refinement from parent
    n_targets = [60, 64, 70, 72, 80, 84, 90, 96, 100, 108, 112]
    for n_try in n_targets:
        if time.time() > deadline - 80:
            break
        h_tr = transfer_to_n(best_h, n_try)
        if len(h_tr) % 2 != 0:
            h_tr = transfer_to_n(best_h, n_try - 1)
        n_tr = len(h_tr)
        sc_tr = eval_score(h_tr)
        if sc_tr < best_c5:
            best_c5 = sc_tr
            best_h = h_tr.copy()
        tl = min(deadline - 20, time.time() + max(10, (deadline - time.time()) * 0.35))
        h_r, sc_r = refine_at_resolution(h_tr, rng, t0, tl)
        if sc_r < best_c5:
            best_c5 = sc_r
            best_h = h_r.copy()

    # Phase 2: Structured starts at low resolution
    for m_val in [28, 30, 32, 35, 36, 40, 42]:
        if time.time() > deadline - 60:
            break
        n_val = 2 * m_val
        for style in range(5):
            if time.time() > deadline - 30:
                break
            q_s = make_structured(m_val, rng, style)
            h_s = build_sym_h(q_s)
            sc_s = eval_score(h_s)
            if sc_s < best_c5 + 0.03:
                tl = min(deadline - 15, time.time() + max(5, (deadline - time.time()) * 0.25))
                h_r, sc_r = refine_at_resolution(h_s, rng, t0, tl)
                if sc_r < best_c5:
                    best_c5 = sc_r
                    best_h = h_r.copy()

    # Phase 3: Random binary starts
    for _ in range(8):
        if time.time() > deadline - 40:
            break
        n_r = rng.choice([56, 60, 64, 72, 80])
        m_r = n_r // 2
        pb = rng.permutation(n_r)
        h_rb = np.zeros(n_r)
        h_rb[pb[:m_r]] = 1.0
        h_rb += 0.15 * rng.randn(n_r)
        h_rb = project_feasible(h_rb, n_r / 2.0)
        sc_rb = eval_score(h_rb)
        if sc_rb < best_c5 + 0.04:
            tl = min(deadline - 10, time.time() + max(5, (deadline - time.time()) * 0.2))
            h_r, sc_r = refine_at_resolution(h_rb, rng, t0, tl)
            if sc_r < best_c5:
                best_c5 = sc_r
                best_h = h_r.copy()

    # Phase 4: Final intensive refinement at best resolution
    if best_h is not None and time.time() < deadline - 30:
        n_b = len(best_h)
        if n_b % 2 != 0:
            n_b = n_b - 1
            h_b = transfer_to_n(best_h, n_b)
            best_h = h_b
        # Multiple LP rounds
        q_b = sym_from_h(best_h)
        for rd in range(4):
            if time.time() > deadline - 15:
                break
            tl = min(deadline - 5, time.time() + max(5, (deadline - time.time()) * 0.3))
            q_lp, sc_lp = lp_refine(q_b, max_iters=200, t0=t0, time_limit=tl)
            sc_lp = eval_score(build_sym_h(q_lp))
            if sc_lp < best_c5 - 1e-13:
                q_b = q_lp
                best_c5 = sc_lp
                best_h = build_sym_h(q_b)
        # Coordinate sweep
        if time.time() < deadline - 10:
            tl = min(deadline - 3, time.time() + 15)
            q_cs, sc_cs = coord_sweep(q_b, t0=t0, time_limit=tl, n_its=35)
            sc_cs = eval_score(build_sym_h(q_cs))
            if sc_cs < best_c5:
                best_c5 = sc_cs
                best_h = build_sym_h(q_cs)
        # Binary polish
        if time.time() < deadline - 5:
            tl = min(deadline - 2, time.time() + 10)
            q_bin, sc_bin = binary_local_search(q_b, rng, t0=t0, time_limit=tl, max_rounds=40)
            sc_bin = eval_score(build_sym_h(q_bin))
            if sc_bin < best_c5:
                best_c5 = sc_bin
                best_h = build_sym_h(q_bin)

    # Final validation
    if best_h is None:
        m_fb = 30
        x = np.linspace(0, 1, m_fb)
        q_fb = 0.5 + 0.3 * np.sin(2 * np.pi * x)
        best_h = build_sym_h(q_fb)

    best_h = np.asarray(best_h, dtype=float).reshape(-1)
    n_out = len(best_h)
    best_h = np.clip(best_h, 0.0, 1.0)
    target = n_out / 2.0
    if abs(best_h.sum() - target) > 1e-6:
        best_h = project_feasible(best_h, target)
    best_c5 = eval_score(best_h)
    return best_h, best_c5, n_out
