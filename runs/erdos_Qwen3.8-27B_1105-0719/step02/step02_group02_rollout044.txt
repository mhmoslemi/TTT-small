def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def c5_of_h(h):
    n = len(h)
    dx = 2.0 / n
    return float(compute_corrs(h).max() * dx)

def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.any():
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def grad_all_corrs(h, n, lags):
    """Compute gradients for specified lags. Returns (n, len(lags)) matrix."""
    grads = np.zeros((n, len(lags)))
    for j, k in enumerate(lags):
        offset = k - (n - 1)
        g = np.zeros(n)
        if offset == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            i_lo = max(0, -offset)
            i_hi = min(n - 1, n - 1 - offset)
            if i_lo <= i_hi:
                idx = np.arange(i_lo, i_hi + 1)
                g[idx + offset] += (1.0 - h[idx])
                g[idx] += (-h[idx + offset])
        grads[:, j] = g
    return grads

def lp_descent_step(h, target_sum, active_lags, corrs):
    """Solve LP for one descent step. Returns h_candidate or None."""
    n = len(h)
    grads = grad_all_corrs(h, n, active_lags)
    # grads shape: (n, m)
    
    # LP: min t
    # s.t. for each active lag j: grads[:,j] @ h_new - t <= dot(grads[:,j], h) - corrs[active_lags[j]]
    #      sum(h_new) = target_sum
    #      0 <= h_new <= 1
    
    m = len(active_lags)
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize t (last variable)
    
    A_ub = np.hstack([grads, -np.ones((n, 1))])  # (m, n+1)
    b_ub = grads.T @ h - corrs[active_lags]  # (m,)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target_sum])
    
    bounds = [(0, 1)] * n + [(corrs[active_lags].min(), corrs[active_lags].max() + 5.0)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
        if res.success:
            h_new = project_feasible(res.x[:n], target_sum)
            return h_new
    except Exception:
        pass
    return None

def make_anti_pal(n, rng):
    if n % 2 == 0:

        v = rng.random(m)
        h = np.empty(n)
        h[:m] = v
        h[m:] = 1.0 - v[::-1]

        h[m] = 0.5
        h[m+1:] = 1.0 - v[::-1]
    return h
