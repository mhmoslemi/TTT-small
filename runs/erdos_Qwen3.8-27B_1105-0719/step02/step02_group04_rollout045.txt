    return np.correlate(h, 1.0 - h, mode='full') * dx_of(n)

def grad_row(h, j):
    """Exact gradient of corr_values(h)[j] w.r.t. h."""
    n = len(h)
    d = j - (n - 1)
    dx = dx_of(n)
    g = np.zeros(n)
    L = max(0, d)
    U = min(n - 1, n - 1 + d)
    if L <= U:
        g[L:U+1] += 1.0 - h[L-d:U+1-d]
    lo2 = max(0, -d)
    hi2 = min(n - 1, n - 1 - d)
    if lo2 <= hi2:
        g[lo2:hi2+1] += -h[lo2+d:hi2+1+d]
    return g * dx

def build_sym(q, n):
    m = n // 2
    h = np.empty(n)
    h[:m] = q
    h[m:] = 1.0 - q[::-1]
    return h

def to_sym_q(h, n):
    m = n // 2
    return 0.5 * (h[:m] + 1.0 - h[n-1:m-1:-1])

def sym_grad(q, j, n):
    """Gradient of corr_values(build_sym(q,n))[j] w.r.t. q."""
    h = build_sym(q, n)
    gh = grad_row(h, j)
    m = n // 2
    return gh[:m] - gh[n-1:m-1:-1]

def project_to_target(h, target_sum):
    lo, hi = -1.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo+hi)*0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-13:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def verify_grad(h, j, eps=1e-6):
    """Verify gradient by finite differences."""
    n = len(h)
    g = grad_row(h, j)
    h_pos = h.copy()
    h_neg = h.copy()
    # Pick a few coordinates to check
    coords = np.random.choice(n, min(5, n), replace=False)
    max_err = 0
    for k in coords:
        h_pos[k] += eps
        h_neg[k] -= eps
        c_pos = corr_values(h_pos)[j]
        c_neg = corr_values(h_neg)[j]
        fd = (c_pos - c_neg) / (2*eps)
        max_err = max(max_err, abs(fd - g[k]))
    return max_err

def slp_sym(q, n, max_iter=80, R_init=0.03, margin_init=0.005, t0=time.time(), budget=600):
    """SLP optimization in symmetric mode."""
    m = n // 2
    h = build_sym(q, n)
    c_cur = eval_c5(h)
    R = R_init
    no_progress = 0
    
    for it in range(max_iter):
        if time.time() - t0 > budget - 30:
            break
        
        C = corr_values(h)
        c_max = C.max()
        
        # Active set: constraints within margin of maximum
        margin = max(1e-7, margin_init * c_max)
        active = np.where(C >= c_max - margin)[0]
        
        if len(active) == 0:
            active = np.array([np.argmax(C)])
        
        # Compute gradients
        A = np.array([sym_grad(q, j, n) for j in active])
        c_act = C[active]
        
        # LP: min t
        # s.t. A@delta - t <= -c_act
        #      lower_i <= delta_i <= upper_i
        # Variables: [delta (m), t (1)]
        cvec = np.zeros(m + 1)
        cvec[m] = 1.0  # minimize t
        
        A_ub = np.zeros((len(active), m + 1))
        A_ub[:, :m] = A
        A_ub[:, m] = -1.0
        b_ub = -c_act
        
        lower = np.maximum(-R, -q + 1e-9)
        upper = np.minimum(R, 1.0 - q - 1e-9)
        
        # Ensure feasibility of box
        if np.any(lower > upper):
            # Shrink R
            R *= 0.5
            no_progress += 1
            if no_progress > 30:
                break
            continue
        
        bounds = list(zip(lower, upper)) + [(0.0, c_cur + 0.05)]
        
        try:
            res = linprog(cvec, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        except Exception:
            no_progress += 1
            if no_progress > 30:
                break
            continue
        
        if not res.success or res.x is None:
            no_progress += 1
            R *= 0.7
            if R < 1e-5:
                break
            if no_progress > 30:
                break
            continue
        
        delta = res.x[:m]
        
        # Line search
        improved = False
        for alpha in [1.0, 0.8, 0.6, 0.4, 0.3, 0.2, 0.1]:
            q_trial = q + alpha * delta
            if q_trial.min() < -1e-8 or q_trial.max() > 1.0 + 1e-8:
                continue
            q_trial = np.clip(q_trial, 0.0, 1.0)
            h_trial = build_sym(q_trial, n)
            c_trial = eval_c5(h_trial)
            if c_trial < c_cur - 1e-10:
                q = q_trial
                h = h_trial
                c_cur = c_trial
                improved = True
                no_progress = 0
                break
        
        if not improved:
            no_progress += 1
            R *= 0.6
            if no_progress >= 5:
                margin_init *= 0.7  # Tighter active set
            if no_progress > 35:
                break
    
    return q, c_cur

def slp_full(h, n, max_iter=60, R_init=0.02, margin_init=0.005, t0=time.time(), budget=600):
    """SLP optimization in full (non-symmetric) mode."""
    target = n / 2.0
    c_cur = eval_c5(h)
    R = R_init
    no_progress = 0
    
    for it in range(max_iter):
        if time.time() - t0 > budget - 30:
            break
        
        C = corr_values(h)
        c_max = C.max()
        
        margin = max(1e-7, margin_init * c_max)
        active = np.where(C >= c_max - margin)[0]
        if len(active) == 0:
            active = np.array([np.argmax(C)])
        
        A = np.array([grad_row(h, j) for j in active])
        c_act = C[active]
        
        # LP: min t
        # s.t. A@delta - t <= -c_act
        #      sum(delta) = 0
        #      lower <= delta <= upper
        cvec = np.zeros(n + 1)
        cvec[n] = 1.0
        
        A_ub = np.zeros((len(active), n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = -c_act
        
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([0.0])
        
        lower = np.maximum(-R, -h + 1e-9)
        upper = np.minimum(R, 1.0 - h - 1e-9)
        
        if np.any(lower > upper):
            R *= 0.5
            no_progress += 1
            if no_progress > 30:
                break
            continue
        
        bounds = list(zip(lower, upper)) + [(0.0, c_cur + 0.05)]
        
        try:
            res = linprog(cvec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs')
        except Exception:
            no_progress += 1
            if no_progress > 30:
                break
            continue
        
        if not res.success or res.x is None:
            no_progress += 1
            R *= 0.7
            if no_progress > 30:
                break
            continue
        
        delta = res.x[:n]
        
        improved = False
        for alpha in [1.0, 0.8, 0.6, 0.4, 0.3, 0.2, 0.1]:
            h_trial = h + alpha * delta
            if h_trial.min() < -1e-8 or h_trial.max() > 1.0 + 1e-8:
                continue
            h_trial = np.clip(h_trial, 0.0, 1.0)
            h_trial = project_to_target(h_trial, target)
            c_trial = eval_c5(h_trial)
            if c_trial < c_cur - 1e-10:
                h = h_trial
                c_cur = c_trial
                improved = True
                no_progress = 0
                break
        
        if not improved:
            no_progress += 1
            R *= 0.6
            if no_progress >= 5:
                margin_init *= 0.7
            if no_progress > 35:
                break
    
    return h, c_cur

def interp_h(h_old, n_new):
    n_old = len(h_old)
    if n_new == n_old:
        return h_old.copy()
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_to_target(h_new,
