    target_sum = n / 2.0

        c5_val = corrs.max() * dx

            best_c5 = c5_val
            best_h = h.copy()
            patience = 0
        else:
            patience += 1

            break

        grad = np.zeros(n)
        for k in top_idx:
            grad += grad_corr_k(h, k, n)
        grad *= dx / topm
        
        # Momentum step
        v = gamma * v - alpha * grad
        h_new = project_feasible(h + v, target_sum)
        new_c5 = c5_of_h(h_new)
        
        if new_c5 < c5_val:

            alpha = min(alpha * 1.01, 0.3)

            v *= 0.5

        if it % 1000 == 0:

            gamma = 0.9 + 0.001 * min(it // 1000, 10)
            gamma = min(gamma, 0.99)
        
        if alpha < 1e-14:
            break
    
    return best_h, best_c5

        shifted = corrs - corrs.max()
        w = np.exp(self.beta * shifted)
        ws = w.sum()
        if ws < 1e-30:
            w = np.ones(len(corrs)) / len(corrs)
        else:
            w /= ws
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] < 1e-14:
                continue
            grad += w[k] * grad_corr_k(x, k, n)
        return grad * self.dx

    def eq_f(self, x):
        return x.sum() - self.target_sum

    def eq_g(self, x):
        return np.ones(self.n)


def slsqp_opt(h0, n, beta, maxiter=500):
    dx = 2.0 / n
    target_sum = n / 2.0
    prob = LSEProblem(n, beta)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': prob.eq_f, 'jac': prob.eq_g}]
    try:
        res = minimize(prob.f, h0.copy(), jac=prob.g, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        h_out = project_feasible(res.x, target_sum)
    except Exception:
        h_out = project_feasible(h0, target_sum)
    return h_out, c5_of_h(h_out)


def topk_gradient_descent(h0, n, max_iter=20000, topm_init=30):
    """Gradient descent on top-k correlations with momentum and line search."""
    dx = 2.0 / n
    target_sum = n / 2.0
    h = h0.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = c5_of_h(h)
    alpha = 0.08
    gamma = 0.85
    topm = topm_init
    patience = 0
    
    for it in range(max_iter):
        corrs = compute_corrs(h)
        c5_val = corrs.max() * dx
        
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
        
        if patience > 5000:
            break
        
        # Adaptive topm
        topm = max(3, topm_init - it // 500)
        
        # Compute top-k gradient
        top_idx = np.argsort(corrs)[-topm:]
        grad = np.zeros(n)
        for k in top_idx:
            grad += grad_corr_k(h, k, n)
        grad *= dx / topm
        
        # Momentum update
        v = gamma * v - alpha * grad
        h_new = project_feasible(h + v, target_sum)
        new_c5 = c5_of_h(h_new)
        
        if new_c5 < c5_val - 1e-16:
            h = h_new
            alpha = min(alpha * 1.005, 0.25)
            gamma = min(gamma + 0.0002, 0.95)
        else:
            alpha *= 0.93
            v *= 0.4
        
        if alpha < 1e-14:
            break
    
    return best_h, best_c5


def make_binary_start(n, rng):
    h = np.zeros(n)
    idx = rng.permutation(n)
    n_ones = int(n / 2)
    h[idx[:n_ones]] = 1.0
    return h


def make_sinusoidal_start(n, rng):
    """Create a start with sinusoidal structure."""
    x = np.linspace(0, 2, n, endpoint=False)
    freq = rng.uniform(1, 5)
    phase = rng.uniform(0, 2 * np.pi)
    h = 0.5 + 0.4 * np.sin(freq * np.pi * x + phase)
    h = np.clip(h, 0, 1)
    return project_feasible(h, n / 2.0)


def make_piecewise_start(n, rng):
    """Create a piecewise constant start."""
    n_blocks = rng.randint(4, 12)
    block_size = n // n_blocks
    h = np.zeros(n)
    for b in range(n_blocks):
        start = b * block_size
        end = start + block_size if b < n_blocks - 1 else n
        h[start:end] = rng.uniform(0, 1)
    return project_feasible(h, n / 2.0)
