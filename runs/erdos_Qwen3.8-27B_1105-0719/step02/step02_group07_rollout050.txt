```python
def line_search_1d(a, b, alpha_lo=0, alpha_hi=1):
    """Find alpha in [alpha_lo, alpha_hi] that minimizes max_k(a_k + b_k * alpha)."""
    # This is a 1D convex piecewise-linear minimization
    # Use bisection on the subgradient
    for _ in range(60):
        alpha_mid = 0.5 * (alpha_lo + alpha_hi)
        vals = a + b * alpha_mid
        max_val = vals.max()
        # Subgradient: weighted average of b_k for active k
        active = np.isclose(vals, max_val, atol=1e-10)
        if b[active].min() > 0:
            alpha_hi = alpha_mid  # slope positive, go left
        elif b[active].max() < 0:
            alpha_lo = alpha_mid  # slope negative, go right
        else:
            break  # 0 is in subgradient
    return 0.5 * (alpha_lo + alpha_hi)
```

```python
        n = self.n
        m = self.m
        dx = self.dx
        L = len(self.active_lags)

        c5 = exact_score(h_new)
        if abs(h_new.sum() - n / 2.0) > 1e-8:
            return None, None
        return res.x[1:], c5
    except Exception:
        return None, None


def active_lags_top(corrs, n_top):
    idx = np.argsort(corrs)[::-1][:n_top]
    return sorted(idx.tolist())


def full_smooth_optimize(h0, n, beta, maxiter=300):
    target_sum = n / 2.0

    def obj(x):
        h = np.clip(x, 0.0, 1.0)
        corrs = corr_vals(h)
        return logsumexp(beta * corrs) / beta

    def obj_grad(x):
        h = np.clip(x, 0.0, 1.0)
        corrs = corr_vals(h)
        shifted = corrs - corrs.max()
        w = np.exp(beta * shifted)
        ws = w.sum()
        if ws < 1e-30:
            w = np.ones(len(corrs)) / len(corrs)
        else:
            w /= ws
        return weighted_grad_full(h, n, 2.0 / n, beta * w)

    try:
        res = minimize(obj, h0, jac=obj_grad, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        h_out = project_feasible(res.x, target_sum)
        c5 = exact_score(h_out)
        return h_out, c5
    except Exception:
        return h0, exact_score(h0)


def try_start(q, n, budget_remaining, rng):
    best_q = q.copy()
    best_c5 = exact_score(make_sym_h(q, n))
    dx = 2.0 / n
    m = n // 2

    # Phase A: cyclic coordinate descent
    if budget_remaining() > 10:
        q, c5 = cyclic_coord_descent(q, n, max_sweeps=50, n_evals=12, tol=1e-13)
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_q = q.copy()

    # Phase B: subgradient with line search
    if budget_remaining() > 10:
        q, c5 = subgrad_line_search(best_q, n, dx, max_iters=200)
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_q = q.copy()

    # Phase C: coordinate descent again
    if budget_remaining() > 8:
        q, c5 = cyclic_coord_descent(best_q, n, max_sweeps=80, n_evals=18, tol=1e-14)
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_q = q.copy()

    # Phase D: smooth optimization
    if budget_remaining() > 10:
        for beta in [100, 300, 800]:
            if budget_remaining() < 5:
                break
            q_new, c5_new = smooth_sym_optimize(best_q, n, beta, maxiter=200)
            if q_new is not None and c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                best_q = q_new.copy()

    # Phase E: epigraph
    if budget_remaining() > 12:
        h_cur = make_sym_h(best_q, n)
        corrs = corr_vals(h_cur)
        n_act = min(80, 2 * n - 1)
        active = active_lags_top(corrs, n_act)
        q_new, c5_new = solve_epi_sym(best_q, n, active,
                                      t_init=corrs.max() * 1.0001, maxiter=400)
        if q_new is not None and c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_q = q_new.copy()
        if budget_remaining() > 10:
            h_cur = make_sym_h(best_q, n)
            corrs = corr_vals(h_cur)
            n_act2 = min(120, 2 * n - 1)
            active2 = active_lags_top(corrs, n_act2)
            q_new, c5_new = solve_epi_sym(best_q, n, active2,
                                          t_init=corrs.max() * 1.0001, maxiter=500)
            if q_new is not None and c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                best_q = q_new.copy()

    return best_q, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def remaining():
        return budget_s - (time.time() - t0)

    # Initialize from parent
    try:
        h_parent = np.array(initial_h_values, dtype=float)
    except NameError:
        h_parent = 0.5 * np.ones(126)

    n = len(h_parent)
    if n % 2 != 0:
        h_parent = h_parent[:n - 1]
        n = len(h_parent)

    h_best = project_feasible(h_parent, n / 2.0)
    c5_best = exact_score(h_best)
    m = n // 2
    q_best = make_q_from_h(h_best, n)
    print(f"Initial: n={n}, C5={c5_best:.7f}")

    # Main optimization from parent
    if remaining() > 30:
        q_new, c5_new = try_start(q_best.copy(), n, remaining, rng)
        if c5_new < c5_best - 1e-14:
            c5_best = c5_new
            q_best = q_new
            h_best = make_sym_h(q_best, n)
            print(f"  main start: C5={c5_best:.7f}")

    # Try perturbed starts
    n_perturbs = 12
    for p in range(n_perturbs):
        if remaining() < 15:
            break
        scale = 0.05 + 0.6 * (p / max(n_perturbs, 1))
        q_pert = q_best + scale * rng.randn(m)
        q_pert = np.clip(q_pert, 0.0, 1.0)
        q_new, c5_new = try_start(q_pert, n, remaining, rng)
        if c5_new < c5_best - 1e-14:
            c5_best = c5_new
            q_best = q_new
            h_best = make_sym_h(q_best, n)
            print(f"  perturb {p} (scale={scale:.2f}): C5={c5_best:.7f}")

    # Try structured starts
    if remaining() > 30:
        # Sine-based
        for freq in [1, 2, 3]:
            if remaining() < 15:
                break
            q_struct = 0.5 + 0.4 * np.sin(2 * np.pi * freq * np.arange(m) / m + rng.rand() * 2 * np.pi)
            q_struct = np.clip(q_struct, 0.0, 1.0)
            q_new, c5_new = try_start(q_struct, n, remaining, rng)
            if c5_new < c5_best - 1e-14:
                c5_best = c5_new
                q_best = q_new
                h_best = make_sym_h(q_best, n)
                print(f"  sine freq={freq}: C5={c5_best:.7f}")

        # Binary random
        for _ in range(3):
            if remaining() < 15:
                break
            q_bin = (rng.rand(m) < 0.5).astype(float)
            q_new, c5_new = try_start(q_bin, n, remaining, rng)
            if c5_new < c5_best - 1e-14:
                c5_best = c5_new
                q_best = q_new
                h_best = make_sym_h(q_best, n)
                print(f"  binary: C5={c5_best:.7f}")

    # Try different grid sizes
    if remaining() > 60:
        for delta in [-2, 2, -4, 4]:
            n_try = n + delta
            if n_try < 40 or n_try % 2 != 0 or remaining() < 25:
                continue
            dx_try = 2.0 / n_try
            x_old = np.linspace(0, 2, n, endpoint=False)
            x_new_g = np.linspace(0, 2, n_try, endpoint=False)
            h_interp = np.interp(x_new_g, x_old, h_best)
            h_interp = project_feasible(h_interp, n_try / 2.0)
            m_try = n_try // 2
            q_try = make_q_from_h(h_interp, n_try)
            q_new, c5_new = try_start(q_try, n_try, remaining, rng)
            if c5_new < c5_best - 1e-12:
                c5_best = c5_new
                n = n_try
                m = m_try
                q_best = q_new
                h_best = make_sym_h(q_best, n)
                print(f"  n={n}: C5={c5_best:.7f}")

    # Final polish on best
    if remaining() > 15:
        q_new, c5_new = cyclic_coord_descent(q_best, n, max_sweeps=100, n_evals=25, tol=1e-14)
        if c5_new < c5_best - 1e-14:
            c5_best = c5_new
            q_best = q_new
            h_best = make_sym_h(q_best, n)
            print(f"  final coord: C5={c5_best:.7f}")

    if remaining() > 15:
        q_new, c5_new = subgrad_line_search(q_best, n, 2.0 / n, max_iters=300)
        if c5_new < c5_best - 1e-14:
            c5_best = c5_new
            q_best = q_new
            h_best = make_sym_h(q_best, n)
            print(f"  final subgrad: C5={c5_best:.7f}")

    if remaining() > 15:
        h_cur = make_sym_h(q_best, n)
        corrs = corr_vals(h_cur)
        active = active_lags_top(corrs, min(150, 2 * n - 1))
        q_new, c5_new = solve_epi_sym(q_best, n, active,
                                      t_init=corrs.max() * 1.0001, maxiter=600)
        if q_new is not None and c5_new < c5_best - 1e-14:
            c5_best = c5_new
            q_best = q_new
            h_best = make_sym_h(q_best, n)
            print(f"  final epi: C5={c5_best:.7f}")

    # Full-space polish
    if remaining() > 20:
        h_new, c5_new = full_smooth_optimize(h_best, n, 500, maxiter=300)
        if c5_new < c5_best - 1e-14:
            c5_best = c5_new
            h_best = h_new
            print(f"  final full: C5={c5_best:.7f}")

    # Final validation
    h_best = np.clip(h_best, 0.0, 1.0)
    h_best = project_feasible(h_best, n / 2.0)
    c5_best = exact_score(h_best)
    print(f"Final: n={n}, C5={c5_best:.7f}")

    return h_best, c5_best, n
```
