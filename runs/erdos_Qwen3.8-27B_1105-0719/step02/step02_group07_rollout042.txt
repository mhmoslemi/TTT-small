```python
    """Find optimal q[j] given all other q values fixed."""
    m = n // 2
    dx = 2.0 / n
    v = q[j]
    
    # Compute current correlations
    h = make_sym_h(q, n)

    # q[j] appears in h[j] and h[n-1-j]

    # S_d = (1-h[j+d])_valid - h[j-d]_valid - (1-h[n-1-j+d])_valid + h[n-1-j-d]_valid
    
    S = np.zeros(2*n - 1)
    for k in range(2*n - 1):
        d = k - (n - 1)

        # h[n-1-j-d] valid if 0 <= n-1-j-d < n (n-1-j is second index)
        if 0 <= n - 1 - j - d < n:
            s += h[n - 1 - j - d]
        S[k] = s

    # Binary search on target t

    hi_val = max(val_0, val_1)

    # And it's at least... well, let's just binary search
    hi_val = np.max(corrs)  # current value
    lo_val = 0.0
    
    # Binary search
    for _ in range(60):
        mid = (lo_val + hi_val) / 2
        # Check if there exists x in [0,1] such that corrs[k] + (x-v)*S[k] <= mid for all k
        # For each k: corrs[k] + (x-v)*S[k] <= mid

        # If S[k] < 0: x >= (mid - corrs[k]) / S[k] + v

        else:
```

```python
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time

def make_sym_h(q, n):
    m = n // 2
    h = np.empty(n)
    h[:m] = q
    h[m:] = 1.0 - q[::-1]
    return h

def make_q_from_h(h, n):
    m = n // 2
    q = 0.5 * (h[:m] + 1.0 - h[n-1:n-1-m:-1])
    return np.clip(q, 0.0, 1.0)

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full').max() * dx

def corr_unscaled(h):
    return np.correlate(h, 1.0 - h, mode='full')

def opt_1d(corrs, v, S, n_iters=35):
    """Minimize max_k (corrs[k] + (x-v)*S[k]) over x in [0,1]."""
    pos_mask = S > 1e-14
    neg_mask = S < -1e-14
    zero_mask = ~(pos_mask | neg_mask)
    
    # If any zero-S term exceeds current max, we're stuck
    if np.any(zero_mask):
        if corrs[zero_mask].max() > corrs.max() + 1e-14:
            # This shouldn't happen since corrs.max() >= all corrs
            pass
    
    current_max = corrs.max()
    lo = 0.0
    hi = current_max
    
    for _ in range(n_iters):
        mid = (lo + hi) * 0.5
        # Check: exists x in [0,1] s.t. for all k, corrs[k] + (x-v)*S[k] <= mid
        lb = 0.0
        ub = 1.0
        if np.any(zero_mask) and corrs[zero_mask].max() > mid:
            lo = mid
            continue
        if np.any(pos_mask):
            # x <= (mid - corrs[k]) / S[k] + v for all k with S[k] > 0
            ub = min(ub, np.min((mid - corrs[pos_mask]) / S[pos_mask] + v))
        if np.any(neg_mask):
            # x >= (mid - corrs[k]) / S[k] + v for all k with S[k] < 0
            lb = max(lb, np.max((mid - corrs[neg_mask]) / S[neg_mask] + v))
        
        if lb <= ub:
            hi = mid
        else:
            lo = mid
    
    # Find optimal x
    mid = hi
    lb = 0.0
    ub = 1.0
    if np.any(pos_mask):
        ub = min(ub, np.min((mid - corrs[pos_mask]) / S[pos_mask] + v))
    if np.any(neg_mask):
        lb = max(lb, np.max((mid - corrs[neg_mask]) / S[neg_mask] + v))
    lb = max(lb, 0.0)
    ub = min(ub, 1.0)
    if lb <= ub:
        return (lb + ub) * 0.5
    return v

def compute_S(j, h, n, ds):
    nj = n - 1 - j
    S = np.zeros(len(ds))
    m1 = (j + ds >= 0) & (j + ds < n)
    S[m1] += (1.0 - h[j + ds[m1]])
    m2 = (j - ds >= 0) & (j - ds < n)
    S[m2] -= h[j - ds[m2]]
    m3 = (nj + ds >= 0) & (nj + ds < n)
    S[m3] -= (1.0 - h[nj + ds[m3]])
    m4 = (nj - ds >= 0) & (nj - ds < n)
    S[m4] += h[nj - ds[m4]]
    return S

def coord_descent(q, n, max_sweeps=1000, ds=None, time_fn=None, t_start=None, time_limit=None):
    """Gauss-Seidel coordinate descent in symmetric q-space."""
    m = n // 2
    if ds is None:
        ds = np.arange(-(n-1), n)
    q = q.copy()
    h = make_sym_h(q, n)
    corrs = corr_unscaled(h)
    dx = 2.0 / n
    best_c5 = corrs.max() * dx
    
    for sweep in range(max_sweeps):
        if time_fn is not None:
            if time.time() - t_start > time_limit - 5:
                break
        improved = False
        for j in range(m):
            S = compute_S(j, h, n, ds)
            x_opt = opt_1d(corrs, q[j], S)
            if abs(x_opt - q[j]) > 1e-14:
                old = q[j]
                q[j] = x_opt
                delta = x_opt - old
                h[j] = x_opt
                h[n - 1 - j] = 1.0 - x_opt
                corrs += delta * S
                c5_new = corrs.max() * dx
                if c5_new < best_c5:
                    best_c5 = c5_new
                improved = True
        if not improved:
            break
    
    return q, h, best_c5
```

```python
def subgrad_step(q, n, corrs, h, step_size, top_k=10):
    """Take a subgradient step in q-space."""
    m = n // 2
    ds = np.arange(-(n-1), n)
    dx = 2.0/n
    max_val = corrs.max()
    
    # Find active lags (top k)
    idx = np.argsort(corrs)[::-1][:top_k]
    
    # Compute subgradient: sum of S for active lags
    grad = np.zeros(m)
    for k in idx:
        d = k - (n - 1)
        for j in range(m):
            nj = n - 1 - j
            s = 0.0
            if 0 <= j + d < n:
                s += (1.0 - h[j + d])
            if 0 <= j - d < n:
                s -= h[j - d]
            if 0 <= nj + d < n:
                s -= (1.0 - h[nj + d])
            if 0 <= nj - d < n:
                s += h[nj - d]
            grad[j] += s
    
    grad *= dx
    
    # Step
    q_new = q - step_size * grad
    q_new = np.clip(q_new, 0.0, 1.0)
    
    return q_new
```

For a given lag k (displacement d = k - (n-1)):

The gradient of corr_k with respect to q[j]:

- d(corr_k)/d(h[nj]) = (1-h[nj+d])_valid - h[nj-d]_valid
- d(corr_k)/d(q[j]) = d(corr_k)/d(h[j]) * 1 + d(corr_k)/d(h[nj]) * (-1)

The step size η should be chosen carefully. A common choice is η = α / ||grad|| where α is a constant

```python
def make_sym_h(q, n):
    m = n // 2
    h = np.empty(n)
    h[:m] = q
    h[m:] = 1.0 - q[::-1]
    return h


def make_q_from_h(h, n):
    m = n // 2
    q = 0.5 * (h[:m] + 1.0 - h[n-1:n-1-m:-1])
    return np.clip(q, 0.0, 1.0)


def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full').max() * dx


def corr_unscaled(h):
    return np.correlate(h, 1.0 - h, mode='full')


def opt_1d(corrs, v, S, n_iters=35):
    pos = S > 1e-14
    neg = S < -1e-14
    current_max = corrs.max()
    lo = 0.0
    hi = current_max
    has_pos = np.any(pos)
    has_neg = np.any(neg)
    if has_pos:
        cpos = corrs[pos]
        spos = S[pos]
    if has_neg:
        cneg = corrs[neg]
        sneg = S[neg]
    for _ in range(n_iters):
        mid = (lo + hi) * 0.5
        lb = 0.0
        ub = 1.0
        if has_pos:
            uu = np.min((mid - cpos) / spos + v)
            if uu < ub:
                ub = uu
        if has_neg:
            ll = np.max((mid - cneg) / sneg + v)
            if ll > lb:
                lb = ll
        if lb <= ub:
            hi = mid
        else:
            lo = mid
    mid = hi
    lb = 0.0
    ub = 1.0
    if has_pos:
        uu = np.min((mid - cpos) / spos + v)
        if uu < ub:
            ub = uu
    if has_neg:
        ll = np.max((mid - cneg) / sneg + v)
        if ll > lb:
            lb = ll
    lb = max(lb, 0.0)
    ub = min(ub, 1.0)
    if lb <= ub:
        return (lb + ub) * 0.5
    return v


def compute_S(j, h, n, ds):
    nj = n - 1 - j
    K = len(ds)
    S = np.zeros(K)
    m1 = (j + ds >= 0) & (j + ds < n)
    S[m1] += (1.0 - h[j + ds[m1]])
    m2 = (j - ds >= 0) & (j - ds < n)
    S[m2] -= h[j - ds[m2]]
    m3 = (nj + ds >= 0) & (nj + ds < n)
    S[m3] -= (1.0 - h[nj + ds[m3]])
    m4 = (nj - ds >= 0) & (nj - ds < n)
    S[m4] += h[nj - ds[m4]]
    return S


def coord_descent(q, n, max_sweeps=2000, ds=None, t_start=None, deadline=None,
                  reconverge_every=200, rng=None):
    m = n // 2
    if ds is None:
        ds = np.arange(-(n-1), n)
    q = q.copy()
    h = make_sym_h(q, n)
    corrs = corr_unscaled(h)
    dx = 2.0 / n
    best_c5 = corrs.max() * dx

    for sweep in range(max_sweeps):
        if deadline is not None and time.time() > deadline - 3.0:
            break
        if reconverge_every and sweep % reconverge_every == 0 and sweep > 0:
            h = make_sym_h(q, n)
            corrs = corr_unscaled(h)
        if rng is not None and sweep % 50 == 0:
            order = list(range(m))
            rng.shuffle(order)
        else:
            order = range(m)
        improved = False
        for j in order:
            S = compute_S(j, h, n, ds)
            x_opt = opt_1d(corrs, q[j], S)
            if abs(x_opt - q[j]) > 1e-14:
                old = q[j]
                q[j] = x_opt
                delta = x_opt - old
                h[j] = x_opt
                h[n - 1 - j] = 1.0 - x_opt
                corrs += delta * S
                c5_new = corrs.max() * dx
                if c5_new < best_c5:
                    best_c5 = c5_new
                improved = True
        if not improved:
            break

    h = make_sym_h(q, n)
    best_c5 = exact_score(h)
    return q, h, best_c5


def full_coord_step(h, i, j, corrs, n, ds):
    """Optimally update h[i]+=delta, h[j]-=delta for pair (i,j)."""
    T_i = np.zeros(len(ds))
    T_j = np.zeros(len(ds))
    m1 = (i + ds >= 0) & (i + ds < n)
    T_i[m1] += (1.0 - h[i + ds[m1]])
    m2 = (i - ds >= 0) & (i - ds < n)
    T_i[m2] -= h[i - ds[m2]]
    m3 = (j + ds >= 0) & (j + ds < n)
    T_j[m3] += (1.0 - h[j + ds[m3]])
    m4 = (j - ds >= 0) & (j - ds < n)
    T_j[m4] -= h[j - ds[m4]]
    S = T_i - T_j
    # corr[k] + delta * S[k], delta in [lo, hi]
    lo = max(-h[i], -(1.0 - h[j]))
    hi = min(1.0 - h[i], h[j])
    if lo >= hi:
        return 0.0
    # Binary search on target value
    cur_max = corrs.max()
    v_lo = 0.0
    v_hi = cur_max
    pos = S > 1e-14
    neg = S < -1e-14
    has_pos = np.any(pos)
    has_neg = np.any(neg)
    if has_pos:
        cp = corrs[pos]
        sp = S[pos]
    if has_neg:
        cn = corrs[neg]
        sn = S[neg]
    for _ in range(35):
        mid = (v_lo + v_hi) * 0.5
        lb = lo
        ub = hi
        if has_pos:
            uu = np.min((mid - cp) / sp)
            if uu < ub:
                ub = uu
        if has_neg:
            ll = np.max((mid - cn) / sn)
            if ll > lb:
                lb = ll
        if lb <= ub:
            v_hi = mid
        else:
            v_lo = mid
    mid = v_hi
    lb = lo
    ub = hi
    if has_pos:
        uu = np.min((mid - cp) / sp)
        if uu < ub:
            ub = uu
    if has_neg:
        ll = np.max((mid - cn) / sn)
        if ll > lb:
            lb = ll
    lb = max(lb, lo)
    ub = min(ub, hi)
    if lb <= ub:
        return (lb + ub) * 0.5
    return 0.0


def full_coord_descent(h, n, max_sweeps=100, ds=None, pairs=None,
                       t_start=None, deadline=None, rng=None):
    h = h.copy()
    corrs = corr_unscaled(h)
    dx = 2.0 / n
    best_c5 = corrs.max() * dx
    m = n // 2

    for sweep in range(max_sweeps):
        if deadline is not None and time.time() > deadline - 3.0:
            break
        if reconverge_every_full and sweep % reconverge_every_full == 0 and sweep > 0:
            corrs = corr_unscaled(h)
        improved = False
        if pairs is not None:
            pair_list = pairs
        else:
            if rng is not None:
                idx = list(range(n))
                rng.shuffle(idx)
                pair_list = [(idx[i], idx[(i + m) % n]) for i in range(m)]
            else:
                pair_list = [(i, i + m) for i in range(m)]
        for (i, j) in pair_list:
            delta = full_coord_step(h, i, j, corrs, n, ds)
            if abs(delta) > 1e-14:
                S = np.zeros(len(ds))
                m1 = (i + ds >= 0) & (i + ds < n)
                S[m1] += (1.0 - h[i + ds[m1]])
                m2 = (i - ds >= 0) & (i - ds < n)
                S[m2] -= h[i - ds[m2]]
                m3 = (j + ds >= 0) & (j + ds < n)
                S[m3] -= (1.0 - h[j + ds[m3]])
                m4 = (j - ds >= 0) & (j - ds < n)
                S[m4] += h[j - ds[m4]]
                h[i] += delta
                h[j] -= delta
                corrs += delta * S
                c5_new = corrs.max() * dx
                if c5_new < best_c5:
                    best_c5 = c5_new
                improved = True
        if not improved:
            break

    h = np.clip(h, 0.0, 1.0)
    best_c5 = exact_score(h)
    return h, best_c5

def smooth_max_obj(x, n, beta):
    m = n // 2
    h = make_sym_h(x, n)
    corrs = corr_unscaled(h)
    dx = 2.0 / n
    return logsumexp(beta * corrs) / beta + (h.sum() - n / 2.0) ** 2 * 1e-6


def project_sym(q, n):
    return np.clip(q, 0.0, 1.0)

    best_q = q.copy()
    ds = np.arange(-(n - 1), n)

    print(f"Initial: n={n}, C5={c5_best:.7f}")

    # Phase 1: Coordinate descent from parent
    q, h, c5_cd = coord_descent(q, n, max_sweeps=3000, ds=ds, deadline=deadline,
                                reconverge_every=300, rng=rng)
    if c5_cd < c5_best:
        c5_best = c5_cd
        best_q = q.copy()
        best_h = h.copy()
        print(f"Phase1 CD: C5={c5_best:.7f}")

    # Phase 2: Iterated local search with random perturbations
    n_trials = 80
    for trial in range(n_trials):
        if time.time() > deadline - 30:
            break
        scale = 0.02 + 0.15 * (trial / max(n_trials, 1))
        q_pert = best_q + scale * rng.randn(m)
        q_pert = np.clip(q_pert, 0.0, 1.0)
        q_new, h_new, c5_new = coord_descent(q_pert, n, max_sweeps=800,
                                             ds=ds, deadline=deadline,
                                             reconverge_every=200, rng=rng)
        if c5_new < c5_best - 1e-10:
            c5_best = c5_new
            best_q = q_new.copy()
            best_h = h_new.copy()
            print(f"Phase2 trial {trial}: C5={c5_best:.7f}")

    # Phase 3: Binary threshold + coordinate descent
    if time.time() < deadline - 30:
        h_bin = (best_q > 0.5).astype(float)
        # Adjust to have exactly m ones in q (so n/2 ones in h)
        n_ones = h_bin.sum()
        if n_ones > m:
            # Flip largest values that are 1 to 0
            idx_ones = np.where(h_bin == 1.0)[0]
            rng.shuffle(idx_ones)
            for idx in idx_ones[:int(n_ones - m)]:
                h_bin[idx] = 0.0
        elif n_ones < m:
            idx_zeros = np.where(h_bin == 0.0)[0]
            rng.shuffle(idx_zeros)
            for idx in idx_zeros[:int(m - n_ones)]:
                h_bin[idx] = 1.0
        q_bin = h_bin.copy()
        h_bin_full = make_sym_h(q_bin, n)
        c5_bin = exact_score(h_bin_full)
        q_b2, h_b2, c5_b2 = coord_descent(q_bin, n, max_sweeps=2000,
                                          ds=ds, deadline=deadline,
                                          reconverge_every=300, rng=rng)
        if c5_b2 < c5_best - 1e-10:
            c5_best = c5_b2
            best_q = q_b2.copy()
            best_h = h_b2.copy()
            print(f"Phase3 binary: C5={c5_best:.7f}")

    # Phase 4: Full-space coordinate descent (break symmetry)
    if time.time() < deadline - 60:
        h_full = best_h.copy()
        h_fs, c5_fs = full_coord_descent(h_full, n, max_sweeps=200, ds=ds,
                                         deadline=deadline, rng=rng)

            c5_best = c5_fs
            best_h = h_fs.copy()
            best_q = make_q_from_h(h_full_proj, n)
            print(f"Phase4 full: C5={c5_best:.7f}")

    # Phase 5: More ILS with different perturbation strategies
    n_trials2 = 60
    for trial in range(n_trials2):
        if time.time() > deadline - 20:
            break
        strategy = trial % 4
        if strategy == 0:
            q_pert = best_q + 0.1 * rng.randn(m)
        elif strategy == 1:
            q_pert = best_q.copy()
            idx = rng.choice(m, size=m // 3, replace=False)
            q_pert[idx] = rng.uniform(0, 1, size=len(idx))
        elif strategy == 2:
            q_pert = np.clip(best_q + 0.3 * rng.randn(m), 0, 1)
            q_pert = 0.5 * q_pert + 0.5 * best_q
        else:
            q_pert = 1.0 - best_q[::-1]
            q_pert = np.clip(q_pert, 0, 1)
        q_pert = np.clip(q_pert, 0.0, 1.0)
        q_new, h_new, c5_new = coord_descent(q_pert, n, max_sweeps=500,
                                             ds=ds, deadline=deadline,
                                             reconverge_every=150, rng=rng)
        if c5_new < c5_best - 1e-10:
            c5_best = c5_new
            best_q = q_new.copy()
            best_h = h_new.copy()
            print(f"Phase5 trial {trial} (strat {strategy}): C5={c5_best:.7f}")

    # Phase 6: Try nearby n values
    if time.time() < deadline - 40:
        for delta in [2, -2, 4, -4, 6, -6]:
            if time.time() > deadline - 25:
                break
            n_try = n + delta
            if n_try < 60 or n_try % 2 != 0:
                continue
            dx_old = 2.0 / n
            dx_try = 2.0 / n_try
            x_old = np.linspace(0, 2, n, endpoint=False)
            x_new = np.linspace(0, 2, n_try, endpoint=False)
            h_interp = np.interp(x_new, x_old, best_h)
            m_try = n_try // 2
            q_try = make_q_from_h(h_interp, n_try)
            ds_try = np.arange(-(n_try - 1), n_try)
            q_new, h_new, c5_new = coord_descent(q_try, n_try, max_sweeps=1000,
                                                 ds=ds_try, deadline=deadline,
                                                 reconverge_every=200, rng=rng)
            if c5_new < c5_best - 1e-9:
                c5_best = c5_new
                best_q = q_new.copy()
                best_h = h_new.copy()
                n = n_try
                m = m_try
                ds = ds_try
                print(f"Phase6 n={n}: C5={c5_best:.7f}")

    # Phase 7: Final intensive coordinate descent
    if time.time() < deadline - 15:
        q_final = make_q_from_h(best_h, len(best_h))
        n_final = len(best_h)
        m_final = n_final // 2
        ds_final = np.arange(-(n_final - 1), n_final)
        q_f, h_f, c5_f = coord_descent(q_final, n_final, max_sweeps=5000,
                                       ds=ds_final, deadline=deadline,
                                       reconverge_every=500, rng=rng)
        if c5_f < c5_best - 1e-12:
            c5_best = c5_f
            best_q = q_f.copy()
            best_h = h_f.copy()
            print(f"Phase7 final: C5={c5_best:.7f}")

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    n_final = len(best_h)
    target_sum = n_final / 2.0
    diff = target_sum - best_h.sum()
    if abs(diff) > 1e-10:
        mask = (best_h > 1e-10) & (best_h < 1.0 - 1e-10)
        if mask.sum() > 0:
            best_h[mask] += diff / mask.sum()
        best_h = np.clip(best_h, 0.0, 1.0)

    c5_final = exact_score(best_h)
    if c5_final < c5_best:
        c5_best = c5_final

    return best_h, c5_best, len(best_h)
```
