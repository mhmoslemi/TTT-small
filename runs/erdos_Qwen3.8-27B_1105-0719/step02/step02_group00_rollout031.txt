```python
import numpy as np
from scipy.optimize import minimize
import time

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx])
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    # Final correction
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmin(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def build_grad_matrix(h, n, dx):
    M = 2 * n - 1
    G = np.zeros((M, n))
    # d = 0 row (middle)
    mid = n - 1
    G[mid, :] = dx * (1.0 - 2.0 * h)
    # d > 0: row m = mid + d
    for d in range(1, n):
        m = mid + d
        # P_d = dx * sum_{j=0}^{n-1-d} h[j+d]*(1-h[j])
        # dP/dh[i] = dx*(1-h[i-d]) for i=d..n-1
        #           -dx*h[i+d] for i=0..n-1-d
        G[m, d:] += dx * (1.0 - h[:n-d])       # from h[j+d] term: i=j+d, j=0..n-1-d
        G[m, :n-d] += -dx * h[d:]                # from (1-h[j]) term: i=j, j=0..n-1-d
    # d < 0: row m = mid + d, e = -d
    for e in range(1, n):
        m = mid - e
        # P = dx * sum_{j=0}^{n-1-e} h[j]*(1-h[j+e])
        # dP/dh[i] = dx*(1-h[i+e]) for i=0..n-1-e
        #           -dx*h[i-e] for i=e..n-1
        G[m, :n-e] += dx * (1.0 - h[e:])         # from h[j] term: i=j, j=0..n-1-e
        G[m, e:] += -dx * h[:n-e]                # from (1-h[j+e]) term: i=j+e, j=0..n-1-e
    return G

def soft_max_loss_grad(h, beta, lam_sum):
    """Compute log-sum-exp surrogate and its gradient."""
    n = len(h)
    dx = 2.0 / n
    P = compute_profile(h)
    Pmax = np.max(P)
    s = beta * (P - Pmax)
    # Numerical stability
    if np.min(s) < -500:
        s = np.clip(s, -500, 0)
    exp_s = np.exp(s)
    S = np.sum(exp_s)
    if S < 1e-300:
        loss = Pmax
        weights = np.zeros(len(P))
        weights[np.argmax(P)] = 1.0
    else:
        loss = Pmax + np.log(S) / beta
        weights = exp_s / S
    G = build_grad_matrix(h, n, dx)
    grad = G.T @ weights
    # Add sum penalty gradient
    sum_err = h.sum() - n / 2.0
    grad += 2.0 * lam_sum * sum_err * np.ones(n)
    # Add penalty to loss
    loss += lam_sum * sum_err * sum_err
    return loss, grad
```

    n = len(h0)
    
    def obj(x):
        return soft_max_loss_grad(x, beta, lam_sum)[0]

Let me also think about a completely different approach: **coordinate descent / loopy message passing**.

The objective is max_m P_m(h), where each P_m involves products h[i]*h[j] for specific (i,j) pairs. This has a structure similar to a Markov random field or a factor graph. I could use coordinate descent: iterate over h[i], and for each i, find the best value of h[i] given all other h[j] fixed.

- P_0: involves h[i]*(1-h[i]) - this is a concave function of h[i], maximized at h[i]=0.5
- P_d (d>0): involves h[i]*(1-h[i-d]) and h[i+d]*(1-h[i]) - linear in h[i]
- P_{-d} (d>0): involves h[i]*(1-h[i+d]) and h[i-d]*(1-h[i]) - linear in h[i]

So for each P_m, the dependence on h[i] is at most linear (except for P_0 which is quadratic concave). import numpy as np
from scipy.optimize import minimize
import time

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity, kind='stable')
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx])
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h, kind='stable')
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmin(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def build_grad_matrix(h, n, dx):
    M = 2 * n - 1
    G = np.zeros((M, n))
    mid = n - 1
    G[mid, :] = dx * (1.0 - 2.0 * h)
    for d in range(1, n):
        m = mid + d
        G[m, d:] += dx * (1.0 - h[:n-d])
        G[m, :n-d] += -dx * h[d:]
        m = mid - d
        G[m, :n-d] += dx * (1.0 - h[d:])
        G[m, d:] += -dx * h[:n-d]
    return G

def smooth_obj_grad(x, beta, lam_sum):
    n = len(x)
    dx = 2.0 / n
    P = np.correlate(x, 1.0 - x, mode="full") * dx
    Pmax = float(np.max(P))
    s = beta * (P - Pmax)
    s = np.clip(s, -600.0, 0.0)
    exp_s = np.exp(s)
    S = float(np.sum(exp_s))
    if S < 1e-300:
        loss = Pmax
        weights = np.zeros(len(P))
        weights[int(np.argmax(P))] = 1.0
    else:
        loss = Pmax + np.log(S) / beta
        weights = exp_s / S
    G = build_grad_matrix(x, n, dx)
    grad = G.T @ weights
    sum_err = x.sum() - n / 2.0
    loss += lam_sum * sum_err * sum_err
    grad += 2.0 * lam_sum * sum_err
    return loss, grad

def l_bfgs_optimize(h0, beta, lam_sum, maxfun=2000):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(smooth_obj_grad, h0, method='L-BFGS-B',
                      bounds=bounds, args=(beta, lam_sum),

        if res.x is not None and np.isfinite(res.x).all():
            return repair_feasible(res.x)
    except Exception:
        pass
    return None

def active_set_slsqp(h0, K=8, maxiter=200):

    M = 2 * n - 1

        return x[-1]

            c_new = exact_score(h_new)
            if c_new < c0 + 1e-9:
                return h_new
    except Exception:
        pass
    return None

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)

    h_new = np.interp(xt, full_xs, full_h)

    return repair_feasible(h_new)

    # h is 0/1 with sum = n/2
    return exact_score(h)

def binary_anneal(h_start, n_steps, seed, t_start, time_limit):
    n = len(h_start)
    rng = np.random.RandomState(seed)
    h = (h_start >= 0.5).astype(float)

        one_idx = np.where(h == 1)[0]
        to_remove = rng.choice(one_idx, size=int(ones) - n//2, replace=False)
        h[to_remove] = 0
    elif ones < n // 2:
        zero_idx = np.where(h == 0)[0]
        to_add = rng.choice(zero_idx, size=n//2 - int(ones), replace=False)
        h[to_add] = 1
    h = h.astype(float)
    
    best_h = h.copy()
    best_c = binary_score(h)
    T = 0.05
    decay = 0.999

        # Choose move: flip or swap
        if len(h[h>0.5]) > 1 and rng.random() < 0.5:

            ones_idx = np.where(h > 0.5)[0]
            zeros_idx = np.where(h < 0.5)[0]
            i = ones_idx[rng.randint(len(ones_idx))]
            j = zeros_idx[rng.randint(len(zeros_idx))]
            h[i], h[j] = h[j], h[i]
            c_new = binary_score(h)
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
            elif rng.random() < np.exp(-(c_new - best_c) / T):
                pass  # accept
            else:
                h[i], h[j] = h[j], h[i]  # reject
        else:
            # Single flip won't work (changes sum), so use swap
            ones_idx = np.where(h > 0.5)[0]
            zeros_idx = np.where(h < 0.5)[0]
            if len(ones_idx) > 0 and len(zeros_idx) > 0:
                i = ones_idx[rng.randint(len(ones_idx))]

                    pass
                else:
                    h[i], h[j] = h[j], h[i]
        T *= decay
    
    return best_h, best_c
