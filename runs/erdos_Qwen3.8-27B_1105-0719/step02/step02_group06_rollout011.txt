import numpy as np
from scipy.optimize import minimize, linprog
from scipy.signal import correlate
import time

def build_sym_h(q):
    m = len(q)
    h = np.empty(2 * m)
    h[:m] = q
    h[m:] = 1.0 - q[::-1]
    return h

def sym_from_h(h):
    n = len(h)
    m = n // 2
    q = np.empty(m)
    for i in range(m):
        q[i] = 0.5 * (h[i] + (1.0 - h[n - 1 - i]))
    return np.clip(q, 0.0, 1.0)

def eval_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def corr_raw(h):
    return np.correlate(h, 1.0 - h, mode='full')

def corr_grad_k(h, k):
    n = len(h)
    d = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -d)
    i_hi = min(n - 1, n - 1 - d)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx] += (1.0 - h[idx + d])
    grad[idx + d] += (-h[idx])
    return grad

def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.5, 1.5
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(x + mid, 0.0, 1.0).sum()
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    h = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        interior = (h > 1e-10) & (h < 1.0 - 1e-10)
        if interior.sum() > 0:
            h[interior] += diff / interior.sum()
    return np.clip(h, 0.0, 1.0)

def select_active(corrs, margin_frac=0.03, neighbor_pad=3, max_k=50):
    max_val = corrs.max()
    threshold = max_val - max(1e-9, margin_frac * max_val)
    active = set(np.where(corrs >= threshold)[0].tolist())
    padded = set()
    for k in active:
        for d in range(-neighbor_pad, neighbor_pad + 1):
            if 0 <= k + d < len(corrs):
                padded.add(k + d)
    active_list = sorted(padded)
    if len(active_list) > max_k:
        scored = sorted([(corrs[k], k) for k in active_list], reverse=True)
        active_list = sorted([k for _, k in scored[:max_k]])
    return active_list

def grad_q_for_lag(h, k, m):
    n = len(h)
    g = corr_grad_k(h, k)
    gq = np.zeros(m)
    for r in range(m):
        gq[r] = g[r] - g[n - 1 - r]
    return gq

def lp_trust_region(q0, n_full, active_k, delta, dx):
    """Solve LP trust region subproblem."""
    m = n_full // 2
    h0 = build_sym_h(q0)
    
    # Compute active correlation values and gradients
    A_rows = []
    b_vals = []
    for k in active_k:
        c_k = float(np.sum(h0[max(0, k-n_full+1):n_full-min(0, k-n_full+1)] * 
                           (1.0 - h0[max(0, k):n_full+min(0, k-n_full+1)]))) * dx
        # Actually let me just use the raw corr
        gq = grad_q_for_lag(h0, k, m)
        c_k = float(corr_raw(h0)[k] * dx)
        # c_k + gq · d <= tau  =>  -gq · d + tau >= c_k  =>  gq · d - tau <= -c_k
        A_rows.append(np.concatenate([gq, [-1.0]]))
        b_vals.append(-c_k)
    
    if not A_rows:
        return q0.copy(), 0.0
    
    A = np.array(A_rows)
    b = np.array(b_vals)
    
    # Variables: d (m vars) + tau (1 var)
    # Minimize tau => c = [0,...,0, 1]
    c_obj = np.zeros(m + 1)
    c_obj[-1] = 1.0
    
    # Bounds: -delta <= d[i] <= delta, and 0 <= q0[i] + d[i] <= 1
    # => lower[i] = max(-delta, -q0[i]), upper[i] = min(delta, 1 - q0[i])
    lower = np.maximum(-delta, -q0)
    upper = np.minimum(delta, 1.0 - q0)
    bounds = list(zip(lower, upper))
    bounds.append((float('-inf'), float('inf')))  # tau unbounded
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.status == 0:
            d = res.x[:m]
            tau = res.x[m]
            q_new = np.clip(q0 + d, 0.0, 1.0)
            return q_new, tau
        else:
            return q0.copy(), 2.0
    except Exception:
        return q0.copy(), 2.0

        score_new = eval_score(h_new)

            best_q = q_new.copy()
            q = q_new
            h = h_new

def coordinate_sweep(q, n_full, n_trials=30, max_rounds=5, time_limit=None):
    """Exact coordinate sweep optimization."""
    m = n_full // 2
    q = np.clip(q.copy(), 0, 1)
    h = build_sym_h(q)
    best_score = eval_score(h)
    
    for rnd in range(max_rounds):
        if time_limit is not None and time.time() > time_limit:
            break
        improved = False
        for r in range(m):
            if time_limit is not None and time.time() > time_limit:
                break
            q_orig = q[r]
            # Try values in neighborhood
            lo = max(0.0, q_orig - 0.3)
            hi = min(1.0, q_orig + 0.3)
            candidates = np.linspace(lo, hi, n_trials)
            best_val = q_orig
            best_sc = best_score
            for v in candidates:
                q[r] = v
                h_test = build_sym_h(q)
                sc = eval_score(h_test)
                if sc < best_sc:
                    best_sc = sc
                    best_val = v
            q[r] = best_val
            if best_sc < best_score - 1e-12:
                best_score = best_sc
                h = build_sym_h(q)
                improved = True
        if not improved:
            break
    
    return q, best_score, build_sym_h(q)

def smooth_max_obj_grad(x, n_full, beta):
    m = n_full // 2
    q = x[:m]
    h = build_sym_h(q)
    dx = 2.0 / n_full
    corrs = corr_raw(h) * dx
    max_c = corrs.max()
    shifted = corrs - max_c
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(len(corrs)) / len(corrs)
    else:
        w /= w_sum
    val = max_c + np.log(w_sum) / beta
    grad_q = np.zeros(m)
    for k in range(2 * n_full - 1):
        if w[k] < 1e-12:
            continue
        gq = grad_q_for_lag(h, k, m)
        grad_q += w[k] * gq
    grad_q *= dx
    return val, grad_q

def smooth_optimize(q0, n_full, beta_list, maxiter_per_beta, time_limit):

    best_q = q.copy()

        bounds = [(0.0, 1.0)] * m
        try:
            def obj(xx):
                return smooth_max_obj_grad(xx, n_full, beta)[0]

                return smooth_max_obj_grad(xx, n_full, beta)[1]
            res = minimize(obj, q, jac=jac, method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxiter_per_beta, 'ftol': 1e-15, 'gtol': 1e-10})
            q_new = np.clip(res.x, 0, 1)
        except Exception:
            q_new = q.copy()
        h_new = build_sym_h(q_new)
        sc = eval_score(h_new)
        if sc < best_score:
            best_score = sc
            best_q = q_new.copy()
        q = q_new
    
    return best_q, best_score, build_sym_h(best_q)

import numpy as np
from scipy.optimize import minimize, linprog
import time


def build_sym_h(q):
    m = len(q)
    h = np.empty(2 * m)
    h[:m] = q
    h[m:] = 1.0 - q[::-1]
    return h


def sym_from_h(h):
    n = len(h)
    m = n // 2
    q = np.empty(m)
    for i in range(m):
        q[i] = 0.5 * (h[i] + (1.0 - h[n - 1 - i]))
    return np.clip(q, 0.0, 1.0)


def eval_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)


def corr_raw(h):
    return np.correlate(h, 1.0 - h, mode='full')


def corr_grad_k(h, k):
    n = len(h)
    d = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -d)
    i_hi = min(n - 1, n - 1 - d)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx] += (1.0 - h[idx + d])
    grad[idx + d] += (-h[idx])
    return grad


def grad_q_for_lag(h, k, m):
    n = len(h)
    g = corr_grad_k(h, k)
    gq = np.empty(m)
    for r in range(m):
        gq[r] = g[r] - g[n - 1 - r]
    return gq


def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.5, 1.5
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(x + mid, 0.0, 1.0).sum()
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    h = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        interior = (h > 1e-10) & (h < 1.0 - 1e-10)
        if interior.sum() > 0:
            h[interior] += diff / interior.sum()
    return np.clip(h, 0.0, 1.0)


def select_active(corrs, margin_frac=0.04, neighbor_pad=3, max_k=45):
    max_val = corrs.max()
    threshold = max_val - max(1e-10, margin_frac * max_val)
    active = set(np.where(corrs >= threshold)[0].tolist())
    padded = set()
    for k in active:
        for d in range(-neighbor_pad, neighbor_pad + 1):
            if 0 <= k + d < len(corrs):
                padded.add(k + d)
    active_list = sorted(padded)
    if len(active_list) > max_k:
        scored = sorted([(corrs[k], k) for k in active_list], reverse=True)
        active_list = sorted([k for _, k in scored[:max_k]])
    return active_list


def solve_lp_trust(q0, n_full, active_k, delta):
    m = n_full // 2
    h0 = build_sym_h(q0)
    corrs0 = corr_raw(h0) * (2.0 / n_full)

    A_rows = []
    b_vals = []
    for k in active_k:
        gq = grad_q_for_lag(h0, k, m)
        c_k = corrs0[k]
        A_rows.append(np.concatenate([gq, [-1.0]]))
        b_vals.append(-c_k)

    if not A_rows:
        return q0.copy()

    A = np.array(A_rows)
    b = np.array(b_vals)
    c_obj = np.zeros(m + 1)
    c_obj[-1] = 1.0

    upper = np.minimum(delta, 1.0 - q0)

def smooth_max_obj(x, n_full, beta):
    m = n_full // 2
    q = x[:m]
    h = build_sym_h(q)
    dx = 2.0 / n_full
    corrs = corr_raw(h) * dx
    max_c = corrs.max()
    shifted = corrs - max_c
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        return max_c, np.zeros(m)
    w /= w_sum
    val = max_c + np.log(w_sum) / beta
    grad_q = np.zeros(m)
    for k in range(2 * n_full - 1):
        if w[k] < 1e-12:
            continue
        grad_q += w[k] * grad_q_for_lag(h, k, m)
    grad_q *= dx
    return val, grad_q

    return smooth_max_obj(x, n_full, beta)[0]


def smooth_jac_wrapper(x, n_full, beta):
    return smooth_max_obj(x, n_full, beta)[1]

            q_o, c5_o, h_o = lp_trust_region_loop(q_r, n_r, max_iters=80,
                                                   time_limit=tl, initial_delta=0.04)
            if c5_o < best_c5:
                best_c5 = c5_o
                best_h = h_o.copy()

            if rem() > 30:
                tl = time.time() + min(20, rem() * 0.4)
                q_sweep, c5_sweep, h_sweep = coordinate_sweep(q_o, n_r, n_trials=15,
                                                               max_rounds=2, time_limit=tl)
                if c5_sweep < best_c5:
                    best_c5 = c5_sweep
                    best_h = h_sweep.copy()

    # Phase 7: Final fine polish
    if rem() > 25 and best_h is not None and best_h.size % 2 == 0:
        q_fin = sym_from_h(best_h)
        if rem() > 40:
            tl = time.time() + min(25, rem() * 0.5)
            q_o, c5_o, h_o = lp_trust_region_loop(q_fin, best_h.size, max_iters=80,
                                                   time_limit=tl, initial_delta=0.01)
            if c5_o < best_c5:
                best_c5 = c5_o
                best_h = h_o.copy()

        if rem() > 15:
            q_fin = sym_from_h(best_h)
            tl = time.time() + min(15, rem() * 0.5)
            q_sweep, c5_sweep, h_sweep = coordinate_sweep(q_fin, best_h.size, n_trials=50,
                                                           max_rounds=2, time_limit=tl)
            if c5_sweep < best_c5:
                best_c5 = c5_sweep
                best_h = h_sweep.copy()

    # Validate and return
    if best_h is None:
        m_fb = 63
        x_fb = np.linspace(0, 1, m_fb)
        q_fb = np.clip(0.5 + 0.3 * np.sin(2 * np.pi * x_fb), 0, 1)
        best_h = build_sym_h(q_fb)

    best_h = np.asarray(best_h, dtype=float).reshape(-1)
    n_out = len(best_h)
    best_h = project_feasible(best_h, n_out / 2.0)
    best_c5 = eval_score(best_h)

    return best_h, best_c5, n_out
