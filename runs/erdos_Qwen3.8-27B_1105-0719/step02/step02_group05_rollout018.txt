```python
import numpy as np
from scipy.optimize import minimize, minimize_scalar
from scipy.special import logsumexp
import time

def build_h_from_q(q, n):
    m = n // 2
    if n % 2 == 0:
        return np.concatenate([q, 1.0 - q[::-1]])
    else:
        return np.concatenate([q, [0.5], 1.0 - q[::-1]])

def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def exact_score(h):
    return compute_corrs(h).max() * (2.0 / len(h))

def exact_score_q(q, n):
    h = build_h_from_q(q, n)
    return exact_score(h)

def grad_corr_d(h, d, n):
    """Gradient of correlation at lag d with respect to h."""
    g = np.zeros(n)
    if d == 0:
        g[:] = 1.0 - 2.0 * h
    elif d > 0:
        g[d:] += (1.0 - h[:n-d])
        g[:n-d] -= h[d:]
    else:
        dd = -d
        g[:n-dd] += (1.0 - h[dd:])
        g[dd:] -= h[:n-dd]
    return g

def grad_q(q, n, beta=0, corrs=None):
    """Gradient of smooth objective (or max) in q-space."""
    m = n // 2
    h = build_h_from_q(q, n)
    dx = 2.0 / n
    if corrs is None:
        corrs = compute_corrs(h)
    
    if beta == 0:
        # subgradient at max
        k_max = corrs.argmax()
        d = k_max - (n-1)
        G = grad_corr_d(h, d, n)
    else:
        shifted = corrs - corrs.max()
        w = np.exp(beta * shifted)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(len(corrs)) / len(corrs)
        else:
            w /= w_sum
        G = np.zeros(n)
        for k in range(len(corrs)):
            if w[k] > 1e-14:
                d = k - (n-1)
                G += w[k] * grad_corr_d(h, d, n)
    
    # Transform from h-gradient to q-gradient
    # ∂/∂q[i] = ∂/∂h[i] - ∂/∂h[n-1-i]
    Gq = G[:m] - G[m:][::-1]
    return Gq * dx, h, corrs

def smooth_q_objective(q, n, beta):
    m = n // 2
    h = build_h_from_q(q, n)
    corrs = compute_corrs(h)
    return logsumexp(beta * corrs) / beta * (2.0/n)

def smooth_q_grad_full(q, n, beta):
    val = smooth_q_objective(q, n, beta)
    m = n // 2
    h = build_h_from_q(q, n)
    corrs = compute_corrs(h)
    Gq, _, _ = grad_q(q, n, beta, corrs)
    return val, Gq

def run_de(q_best, n, rng, time_limit, pop_size=40):
    m = n // 2
    # Initialize population
    pop = np.zeros((pop_size, m))
    pop[0] = q_best
    for i in range(1, pop_size):
        pop[i] = np.clip(q_best + rng.normal(0, 0.1, m), 0, 1)
    # Add some random
    pop[-1] = rng.uniform(0, 1, m)
    
    scores = np.array([exact_score_q(pop[i], n) for i in range(pop_size)])
    best_idx = scores.argmin()
    
    t_start = time.time()
    gen = 0
    no_improve = 0
    
    while time.time() - t_start < time_limit:
        gen += 1
        # Adaptive F
        F = 0.5 + 0.3 * rng.random()
        CR = 0.7 + 0.2 * rng.random()
        
        for i in range(pop_size):
            if time.time() - t_start > time_limit:
                break
            # Select 3 different individuals
            a, b, c = rng.choice(pop_size, 3, replace=False)
            while a == b or a == c or b == c:
                a, b, c = rng.choice(pop_size, 3, replace=False)
            
            # Mutation
            v = np.clip(pop[a] + F * (pop[b] - pop[c]), 0, 1)
            
            # Crossover
            trial = pop[i].copy()
            j_rand = rng.randint(m)
            for j in range(m):
                if rng.random() < CR or j == j_rand:
                    trial[j] = v[j]
            
            trial = np.clip(trial, 0, 1)
            trial_score = exact_score_q(trial, n)
            
            if trial_score <= scores[i] + 1e-15:
                pop[i] = trial
                scores[i] = trial_score
                if trial_score < scores[best_idx] - 1e-12:
                    best_idx = i
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1
        
        if no_improve > pop_size * 5:
            break
    
    best_q = pop[best_idx].copy()
    best_c = scores[best_idx]
    return best_q, best_c

# Active set epigraph in q-space
class EpigraphQ:
    def __init__(self, n, active_k):
        self.n = n
        self.m = n // 2
        self.active_k = active_k
        self.dx = 2.0 / n
    
    def obj(self, x):
        return x[0]
    
    def obj_grad(self, x):
        g = np.zeros(self.m + 1)
        g[0] = 1.0
        return g
    
    def constraint(self, x, k_idx):
        t = x[0]
        q = x[1:]
        h = build_h_from_q(q, self.n)
        k = self.active_k[k_idx]
        d = k - (self.n - 1)
        n = self.n
        if d >= 0:
            i_vals = np.arange(0, n - d)
            val = np.sum(h[i_vals + d] * (1.0 - h[i_vals]))
        else:
            dd = -d
            i_vals = np.arange(dd, n)
            val = np.sum(h[i_vals] * (1.0 - h[i_vals - dd]))
        return t - val * self.dx
    
    def constraint_grad(self, x, k_idx):
        q = x[1:]
        h = build_h_from_q(q, self.n)
        k = self.active_k[k_idx]
        d = k - (self.n - 1)
        n = self.n
        g = np.zeros(self.m + 1)
        g[0] = 1.0
        G = grad_corr_d(h, d, n)
        Gq = (G[:self.m] - G[self.m:][::-1]) * self.dx
        g[1:] = -Gq
        return g

def solve_epigraph_q(q0, n, active_k, t0, maxiter=200):
    m = n // 2
    x0 = np.concatenate([[t0], q0])
    ep = EpigraphQ(n, active_k)
    cons = []
    for i in range(len(active_k)):
        cons.append({
            'type': 'ineq',
            'fun': (lambda x, i=i: ep.constraint(x, i)),
            'jac': (lambda x, i=i: ep.constraint_grad(x, i))
        })
    bounds = [(0.0, 5.0)] + [(0.0, 1.0)] * m
    try:
        res = minimize(ep.obj, x0, jac=ep.obj_grad, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-12})
        q_out = np.clip(res.x[1:], 0.0, 1.0)
        return q_out, res.x[0]
    except:
        return q0, t0

def coordinate_descent_q(q0, n, time_limit, max_passes=5):
    m = n // 2
    q = q0.copy()
    current_c = exact_score_q(q, n)
    t_start = time.time()
    
    for p in range(max_passes):
        if time.time() - t_start > time_limit:
            break
        improved = False
        order = list(range(m))
        np.random.shuffle(order)  # use fixed seed for reproducibility... actually use global rng
        for i in order:
            if time.time() - t_start > time_limit:
                break
            # Try to optimize coordinate i
            best_val = q[i]
            best_c = current_c
            # Sample several values
            for trial_val in [0.0, 0.25, 0.5, 0.75, 1.0, q[i]-0.1, q[i]+0.1, q[i]-0.05, q[i]+0.05]:
                if trial_val < 0 or trial_val > 1:
                    continue
                q_old = q[i]
                q[i] = trial_val
                c = exact_score_q(q, n)
                if c < best_c - 1e-14:
                    best_c = c
                    best_val = trial_val
                q[i] = q_old
            
            if best_val != q[i]:
                q[i] = best_val
                current_c = best_c
                improved = True
        
        if not improved:
            break
    
    return q, current_c
```

```python
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time


def build_h_from_q(q, n):
    m = n // 2
    if n % 2 == 0:
        return np.concatenate([q, 1.0 - q[::-1]])
    else:
        return np.concatenate([q, [0.5], 1.0 - q[::-1]])


def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def exact_score(h):
    return compute_corrs(h).max() * (2.0 / len(h))


def exact_score_q(q, n):
    h = build_h_from_q(q, n)
    return exact_score(h)


def grad_corr_d(h, d, n):
    g = np.zeros(n)
    if d == 0:
        g[:] = 1.0 - 2.0 * h
    elif d > 0:
        if d < n:
            g[d:] += (1.0 - h[:n - d])
            g[:n - d] -= h[d:]
    else:
        dd = -d
        if dd < n:
            g[:n - dd] += (1.0 - h[dd:])
            g[dd:] -= h[:n - dd]
    return g


def smooth_q_value(q, n, beta):
    h = build_h_from_q(q, n)
    corrs = compute_corrs(h)
    return logsumexp(beta * corrs) / beta * (2.0 / n)


def smooth_q_gradient(q, n, beta):
    m = n // 2
    h = build_h_from_q(q, n)
    corrs = compute_corrs(h)
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(len(corrs)) / len(corrs)
    else:
        w /= w_sum
    G = np.zeros(n)
    for k in range(len(corrs)):
        if w[k] > 1e-14:
            d = k - (n - 1)
            G += w[k] * grad_corr_d(h, d, n)
    Gq = G[:m] - G[m:][::-1]
    return Gq * (2.0 / n)

    def __init__(self, n, beta):
        self.n = n
        self.beta = beta

    def value(self, x):
        return smooth_q_value(x, self.n, self.beta)

    def gradient(self, x):
        return smooth_q_gradient(x, self.n, self.beta)

    m = n // 2
    so = SmoothObj(n, beta)
    try:
        res = minimize(so.value, q0.copy(), jac=so.gradient, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * m,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-8})
        return np.clip(res.x, 0.0, 1.0)
    except Exception:
        return q0.copy()


class EpigraphQ:
    def __init__(self, n, active_k):
        self.n = n
        self.m = n // 2
        self.active_k = active_k
        self.dx = 2.0 / n

    def value(self, x):
        return x[0]

    def gradient(self, x):
        g = np.zeros(self.m + 1)
        g[0] = 1.0
        return g

    def constraint(self, x, idx):
        t = x[0]
        q = x[1:]
        h = build_h_from_q(q, self.n)
        k = self.active_k[idx]
        d = k - (self.n - 1)
        n = self.n
        if d >= 0:
            val = np.sum(h[d:] * (1.0 - h[:n - d]))
        else:
            dd = -d
            val = np.sum(h[dd:] * (1.0 - h[:n - dd]))
        return t - val * self.dx

    def constraint_grad(self, x, idx):
        q = x[1:]
        h = build_h_from_q(q, self.n)
        k = self.active_k[idx]
        d = k - (self.n - 1)
        n = self.n
        g = np.zeros(self.m + 1)
        g[0] = 1.0
        G = grad_corr_d(h, d, n)
        Gq = (G[:self.m] - G[self.m:][::-1]) * self.dx
        g[1:] = -Gq
        return g

        gen_improved = 0

        for i in range(pop_size):
            if time.time() - t_start > time_limit:
                break
            idxs = rng.choice(pop_size, size=3, replace=False)
            a, b, c = int(idxs[0]), int(idxs[1]), int(idxs[2])
            if a == b or a == c or b == c:
                continue

            v = np.clip(pop[a] + F * (pop[b] - pop[c]), 0, 1)
            trial = pop[i].copy()
            j_rand = rng.randint(m)
            for j in range(m):
                if rng.random() < CR or j == j_rand:
                    trial[j] = v[j]
            trial = np.clip(trial, 0, 1)
            ts = exact_score_q(trial, n)

            if ts <= scores[i] + 1e-15:
                pop[i] = trial
                scores[i] = ts
                gen_improved += 1
                if ts < best_score - 1e-12:
                    best_score = ts
                    best_idx = i
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1

        if no_improve > pop_size * 8:
            break

    return pop[best_idx].copy(), best_score


def coord_descent(q0, n, time_limit, max_passes=4):
    m = n // 2
    q = q0.copy()
    current_c = exact_score_q(q, n)
    t_start = time.time()
    rng_local = np.random.RandomState(0)

    for p in range(max_passes):
        if time.time() - t_start > time_limit:
            break
        improved = False
        order = list(range(m))
        rng_local.shuffle(order)

        for i in order:
            if time.time() - t_start > time_limit:
                break
            best_val = q[i]
            best_c = current_c
            cv = q[i]
            trials = [0.0, 1.0, cv - 0.2, cv - 0.1, cv - 0.05, cv + 0.05, cv + 0.1, cv + 0.2, 0.5]
            for tv in trials:
                if tv < 0 or tv > 1:
                    continue
                q[i] = tv
                c = exact_score_q(q, n)
                if c < best_c - 1e-14:
                    best_c = c
                    best_val = tv
            q[i] = best_val
            if abs(best_val - cv) > 1e-10:
                current_c = best_c
                improved = True

        if not improved:
            break

    return q, current_c


def interp_q(q_src, n_src, n_dst):
    m_dst = n_dst // 2
    h_src = build_h_from_q(q_src, n_src)
    x_src = np.linspace(0, 2, n_src)
    x_dst = np.linspace(0, 2, n_dst)
    h_dst = np.interp(x_dst, x_src, h_src)
    return np.clip(h_dst[:m_dst], 0, 1)


def project_h_to_q(h, n):
    m = n // 2
    return np.clip(h[:m], 0, 1)


def run(seed=42, budget_s=1100, **kwargs):
    t_global_start = time.time()
    rng = np.random.RandomState(seed)

    def remaining():
        return budget_s - (time.time() - t_global_start)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    h_init = np.clip(h_init, 0.0, 1.0)
    best_h = h_init.copy()
    best_c5 = exact_score(best_h)
    m_init = n_init // 2
    q_cur = np.clip(best_h[:m_init], 0.0, 1.0)

    # Phase 1: DE on initial n
    if remaining() > 80:
        de_time = min(remaining() * 0.25, 150)
        q_de, c_de = run_de(q_cur, n_init, rng, de_time, pop_size=45)
        if c_de < best_c5 - 1e-12:
            best_c5 = c_de
            best_h = build_h_from_q(q_de, n_init)
            q_cur = q_de

    # Phase 2: Smooth L-BFGS-B with increasing beta
    for beta in [40, 100, 250, 600, 1500]:
        if remaining() < 25:
            break
        q_s = smooth_solve(q_cur, n_init, beta, maxiter=200)
        c_s = exact_score_q(q_s, n_init)
        if c_s < best_c5 - 1e-12:
            best_c5 = c_s
            best_h = build_h_from_q(q_s, n_init)
            q_cur = q_s

    # Phase 3: Active set epigraph
    if remaining() > 50:
        es_time = min(remaining() * 0.3, 100)
        q_es, c_es = active_set_refine(q_cur, n_init, es_time, max_rounds=7)
        if c_es < best_c5 - 1e-12:
            best_c5 = c_es
            best_h = build_h_from_q(q_es, n_init)
            q_cur = q_es

    # Phase 4: Coordinate descent
    if remaining() > 30:
        cd_time = min(remaining() * 0.2, 50)
        q_cd, c_cd = coord_descent(q_cur, n_init, cd_time, max_passes=4)
        if c_cd < best_c5 - 1e-12:
            best_c5 = c_cd
            best_h = build_h_from_q(q_cd, n_init)
            q_cur = q_cd

    # Phase 5: Second DE round with different parameters
    if remaining() > 60:
        de_time2 = min(remaining() * 0.25, 100)
        q_de2, c_de2 = run_de(q_cur, n_init, rng, de_time2, pop_size=35)
        if c_de2 < best_c5 - 1e-12:
            best_c5 = c_de2
            best_h = build_h_from_q(q_de2, n_init)
            q_cur = q_de2

    # Phase 6: Try different resolutions
    target_sizes = [80, 96, 108, 120]
    for n_try in target_sizes:
        if remaining() < 70:
            break
        m_try = n_try // 2
        q_try = interp_q(q_cur, len(best_h), n_try)

        # Quick smooth
        for beta in [80, 400]:
            if remaining() < 25:
                break
            q_t = smooth_solve(q_try, n_try, beta, maxiter=150)
            c_t = exact_score_q(q_t, n_try)
            if c_t < best_c5 - 1e-12:
                best_c5 = c_t
                best_h = build_h_from_q(q_t, n_try)
                q_try = q_t
            else:
                q_t = q_try

        # DE on this size
        if remaining() > 40:
            q_dt, c_dt = run_de(q_t, n_try, rng, min(remaining() * 0.3, 60), pop_size=30)
            if c_dt < best_c5 - 1e-12:
                best_c5 = c_dt
                best_h = build_h_from_q(q_dt, n_try)
                q_try = q_dt
            else:
                q_try = q_t

        # Active set
        if remaining() > 30:
            q_et, c_et = active_set_refine(q_try, n_try, min(remaining() * 0.3, 50), max_rounds=5)
            if c_et < best_c5 - 1e-12:
                best_c5 = c_et
                best_h = build_h_from_q(q_et, n_try)
                q_try = q_et

    # Phase 7: Final polish on best found
    n_best = len(best_h)
    m_best = n_best // 2
    q_best = np.clip(best_h[:m_best], 0, 1)

    if remaining() > 40:
        for beta in [200, 800, 2000]:
            if remaining() < 15:
                break
            q_s = smooth_solve(q_best, n_best, beta, maxiter=250)
            c_s = exact_score_q(q_s, n_best)
            if c_s < best_c5 - 1e-12:
                best_c5 = c_s
                best_h = build_h_from_q(q_s, n_best)
                q_best = q_s

    if remaining() > 30:
        q_a, c_a = active_set_refine(q_best, n_best, min(remaining() * 0.4, 60), max_rounds=6)
        if c_a < best_c5 - 1e-12:
            best_c5 = c_a
            best_h = build_h_from_q(q_a, n_best)
            q_best = q_a

    if remaining() > 20:
        q_c, c_c = coord_descent(q_best, n_best, min(remaining() * 0.4, 30), max_passes=3)
        if c_c < best_c5 - 1e-12:
            best_c5 = c_c
            best_h = build_h_from_q(q_c, n_best)

    # Final validation and cleanup
    best_h = np.clip(best_h, 0.0, 1.0)
    n_out = len(best_h)
    final_c5 = exact_score(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, n_out
```
