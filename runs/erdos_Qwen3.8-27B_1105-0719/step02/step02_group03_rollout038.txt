            best_val = np.inf

                    flat_idx = vals.argmin()
                    best_i = i_start + flat_idx // len(zeros)
                    best_j = flat_idx % len(zeros)
        else:

            best_val = vals.min()
            flat_idx = vals.argmin()
            best_i = flat_idx // len(zeros)
            best_j = flat_idx % len(zeros)

def make_delta_matrix(h, n):
    """
    Returns D_rem (n/2 × (2n-1)) and D_add (n/2 × (2n-1))
    D_rem[i] = delta corr when ones[i] goes 1→0
    D_add[j] = delta corr when zeros[j] goes 0→1
    """
    nc = 2 * n - 1
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    ks = np.arange(-(n-1), n)
    
    D_rem = np.zeros((len(ones), nc))
    D_add = np.zeros((len(zeros), nc))
    
    for idx, p in enumerate(ones):
        valid_pos = (p + ks >= 0) & (p + ks < n)
        valid_neg = (p - ks >= 0) & (p - ks < n)
        m_pos = ks[valid_pos] + (n-1)
        m_neg = ks[valid_neg] + (n-1)
        D_rem[idx, m_pos] += -(1.0 - h[p + ks[valid_pos]])
        D_rem[idx, m_neg] += h[p - ks[valid_neg]]
