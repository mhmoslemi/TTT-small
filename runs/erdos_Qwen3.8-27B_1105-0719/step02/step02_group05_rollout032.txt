from scipy.optimize import minimize, linprog
from scipy.special import logsumexp
import time

def compute_corrs(h):
    """Compute all correlation values (unscaled)."""
    return np.correlate(h, 1.0 - h, mode='full')

def exact_c5(h):
    """Compute the exact C5 value."""
    n = len(h)
    dx = 2.0 / n
    corrs = compute_corrs(h)
    return corrs.max() * dx

def build_h_from_q(q):
    """Build h from symmetric q representation. n = 2*m."""
    n = 2 * len(q)
    return np.concatenate([q, 1.0 - q[::-1]])

def q_from_h(h):
    """Extract q from h (assuming even n and symmetric)."""
    m = len(h) // 2
    return h[:m].copy()

def score_q(q):
    """Compute C5 for a q vector (symmetric representation)."""
    h = build_h_from_q(q)
    return exact_c5(h)

def cordesc_step(q, coord, n_iter=30):
    """Minimize score_q over q[coord] using ternary search."""
    q0 = q[coord]
    lo, hi = 0.0, 1.0
    # Ternary search
    for _ in range(n_iter):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        q[coord] = m1
        f1 = score_q(q)
        q[coord] = m2
        f2 = score_q(q)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    q[coord] = (lo + hi) / 2.0
    q[coord] = np.clip(q[coord], 0.0, 1.0)
    return score_q(q)

def coordinate_descent(q, max_sweeps=100, time_limit=None, t_start=None):
    """Cyclic coordinate descent in q-space."""
    m = len(q)
    best_score = score_q(q)
    for sweep in range(max_sweeps):
        if time_limit is not None and t_start is not None:
            if time.time() - t_start > time_limit:
                break
        improved = False
        for i in range(m):
            old = q[i]
            new_score = cordesc_step(q, i)
            if new_score < best_score - 1e-14:
                best_score = new_score
                improved = True
            else:
                q[i] = old  # revert if no improvement
        if not improved:
            break
    return q, best_score

                g[dd:] += (-h[:n-dd])

    return val, Gq

def build_h_from_q(q):
    m = len(q)
    return np.concatenate([q, 1.0 - q[::-1]])


def score_q(q):
    h = build_h_from_q(q)
    dx = 2.0 / len(h)
    corrs = np.correlate(h, 1.0 - h, mode='full')
    return corrs.max() * dx


def score_h(h):
    dx = 2.0 / len(h)
    corrs = np.correlate(h, 1.0 - h, mode='full')
    return corrs.max() * dx


def ternary_coord(q, idx, iters=22):
    lo, hi = 0.0, 1.0
    for _ in range(iters):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        q[idx] = m1
        f1 = score_q(q)
        q[idx] = m2
        f2 = score_q(q)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    q[idx] = (lo + hi) / 2.0
    q[idx] = max(0.0, min(1.0, q[idx]))
    return score_q(q)


def coord_desc(q, time_budget, t_start):

def smooth_q_value_grad(q, beta):
    h = build_h_from_q(q)
    dx = 2.0 / len(h)
    n = len(h)
    corrs = np.correlate(h, 1.0 - h, mode='full')
    s = corrs * dx
    z = beta * s
    zmax = z.max()
    zs = z - zmax
    e = np.exp(zs)
    es = e.sum()
    val = (zmax + np.log(es)) / beta
    w = e / es
    G = np.zeros(n)
    for k in range(len(corrs)):
        wk = w[k]
        if wk < 1e-15:
            continue
        d = k - (n - 1)
        if d >= 0 and d < n:
            G[d:] += wk * (1.0 - h[:n - d])
            G[:n - d] += -wk * h[d:]
        elif d < 0:
            dd = -d
            if dd < n:
                G[:n - dd] += wk * (1.0 - h[dd:])
                G[dd:] += -wk * h[:n - dd]
    m = len(q)
    Gq = (G[:m] - G[n - 1:m - 1:-1]) * dx
    return val, Gq
