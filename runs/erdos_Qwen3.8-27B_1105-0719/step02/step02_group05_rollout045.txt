import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time


def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corrs = np.correlate(h, 1.0 - h, mode='full')
    return corrs.max() * dx, corrs


def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)


def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def corr_at_d(h, d, n):
    if d >= 0:
        if d >= n:
            return 0.0
        return np.dot(h[d:], 1.0 - h[:n - d])
    else:
        dd = -d
        if dd >= n:
            return 0.0
        return np.dot(h[dd:], 1.0 - h[:n - dd])


def grad_corr_d(h, d, n):
    g = np.zeros(n)
    if d >= 0:
        if d < n:
            g[d:] += (1.0 - h[:n - d])
            g[:n - d] += (-h[d:])
    else:
        dd = -d
        if dd < n:
            g[:n - dd] += (1.0 - h[dd:])
            g[dd:] += (-h[:n - dd])
    return g


def smooth_value_grad(h, n, beta):
    dx = 2.0 / n
    corrs = compute_corrs(h)
    val = logsumexp(beta * corrs) / beta * dx
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(len(corrs)) / len(corrs)
    else:
        w /= w_sum
    G = np.zeros(n)
    for k in range(len(corrs)):
        if w[k] < 1e-14:
            continue
        d = k - (n - 1)
        G += w[k] * grad_corr_d(h, d, n)
    return val, G * dx


def nesterov_smooth(h0, n, beta, target_sum, maxiter=300, time_limit=None, t_start=None):
    h = h0.copy()
    y = h0.copy()
    best_h = h0.copy()
    best_val = smooth_value_grad(h0, n, beta)[0]
    step = 1.0 / (beta * (2.0 / n))
    
    for it in range(maxiter):
        if time_limit is not None and t_start is not None:
            if time.time() - t_start > time_limit:
                break
        val_y, grad_y = smooth_value_grad(y, n, beta)
        gn = np.linalg.norm(grad_y)
        if gn < 1e-15:
            break
        h_new = project_feasible(y - step * grad_y, target_sum)
        t_k = 1.0 / (1.0 + 0.5 * np.sqrt(1.0 + it * it))
        y = h_new + (t_k - 1.0) / t_k * (h_new - h)
        y = project_feasible(y, target_sum)
        val_h, _ = smooth_value_grad(h_new, n, beta)
        if val_h < best_val:
            best_val = val_h
            best_h = h_new.copy()
        h = h_new
    return best_h, best_val


class EpigraphProblem:
    def __init__(self, n, active_k, target_sum):
        self.n = n
        self.active_k = active_k
        self.dx = 2.0 / n
        self.target_sum = target_sum

    def obj(self, x):
        return x[0]

    def obj_grad(self, x):
        g = np.zeros(self.n + 1)
        g[0] = 1.0
        return g

    def constraint_corr(self, x, k_idx):
        t = x[0]
        h = x[1:]
        k = self.active_k[k_idx]
        d = k - (self.n - 1)
        return t - corr_at_d(h, d, self.n) * self.dx

    def constraint_corr_grad(self, x, k_idx):
        h = x[1:]
        k = self.active_k[k_idx]
        d = k - (self.n - 1)
        g = np.zeros(self.n + 1)
        g[0] = 1.0
        g[1:] = -grad_corr_d(h, d, self.n) * self.dx
        return g

    def constraint_sum(self, x):
        return x[1:].sum() - self.target_sum

    def constraint_sum_grad(self, x):
        g = np.zeros(self.n + 1)
        g[1:] = np.ones(self.n)
        return g


def solve_epigraph(h0, n, active_k, t0, target_sum, maxiter=200):
    ep = EpigraphProblem(n, active_k, target_sum)
    x0 = np.concatenate([[t0], h0])
    cons = []
    for i in range(len(active_k)):
        cons.append({'type': 'ineq', 'fun': ep.constraint_corr, 'jac': ep.constraint_corr_grad, 'args': (i,)})
    cons.append({'type': 'eq', 'fun': ep.constraint_sum, 'jac': ep.constraint_sum_grad})
    bounds = [(0.0, 5.0)] + [(0.0, 1.0)] * n
    res = minimize(ep.obj, x0, jac=ep.obj_grad, method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    h_out = project_feasible(res.x[1:], target_sum)
    return h_out, res.x[0], res


def active_set_optimize(h0, n, target_sum, max_rounds=10, time_limit=None, t_start=None):
    dx = 2.0 / n
    corrs = compute_corrs(h0)
    current_max = corrs.max() * dx
    L = min(25, 2 * n - 1)
    top_idx = np.argsort(corrs)[::-1][:L]
    active_k = sorted(top_idx.tolist())
    h = h0.copy()
    best_h = h.copy()
    best_c5 = current_max
    t0 = current_max * (1.0 + 1e-5) + 1e-7

    for rnd in range(max_rounds):
        if time_limit is not None and t_start is not None:
            if time.time() - t_start > time_limit:
                break
        h = project_feasible(h, target_sum)
        corrs = compute_corrs(h)
        current_max = corrs.max() * dx
        if current_max < best_c5:
            best_c5 = current_max
            best_h = h.copy()

        active_mask = corrs * dx >= current_max - 1e-5
        new_active = sorted(np.where(active_mask)[0].tolist())
        if len(new_active) == 0:
            new_active = active_k
        if len(new_active) > 45:
            vals = [corrs[k] for k in new_active]
            order = np.argsort(vals)[::-1][:45]
            new_active = sorted([new_active[i] for i in order])
        active_k = new_active

        h_new, t_solved, res = solve_epigraph(h, n, active_k, t0, target_sum, maxiter=250)
        h_new = project_feasible(h_new, target_sum)
        corrs_new = compute_corrs(h_new)
        c5_new = corrs_new.max() * dx
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()

        hidden = corrs_new * dx > t_solved + 1e-6
        if hidden.sum() > 0:
            hidden_idx = np.where(hidden)[0]
            merged = sorted(set(active_k) | set(hidden_idx.tolist()))
            if len(merged) > 45:
                vals = [corrs_new[k] for k in merged]
                order = np.argsort(vals)[::-1][:45]
                merged = sorted([merged[i] for i in order])
            active_k = merged

        t0 = max(t_solved, c5_new) * (1.0 + 1e-6) + 1e-7
        h = h_new
        if abs(c5_new - best_c5) < 1e-13 and hidden.sum() == 0:
            break
    return best_h, best_c5


def subgradient_polish(h0, n, target_sum, maxiter=500, time_limit=None, t_start=None):
    """Direct subgradient method on the exact max objective."""
    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_corrs(h).max() * dx
    step = 0.01

    for it in range(maxiter):
        if time_limit is not None and t_start is not None:
            if time.time() - t_start > time_limit:
                break
        corrs = compute_corrs(h)
        max_corr = corrs.max() * dx
        if max_corr < best_c5:
            best_c5 = max_corr
            best_h = h.copy()
        active = np.where(np.abs(corrs * dx - max_corr) < 1e-8)[0]
        G = np.zeros(n)
        for k in active:
            d = k - (n - 1)
            G += grad_corr_d(h, d, n)
        G /= len(active)
        G *= dx
        h_new = project_feasible(h - step * G, target_sum)
        h = h_new
        step *= 0.999
    return best_h, best_c5


def interpolate_h(h_src, n_dst):
    x_src = np.linspace(0.0, 2.0, len(h_src))
    x_dst = np.linspace(0.0, 2.0, n_dst)
    return np.interp(x_dst, x_src, h_src)
