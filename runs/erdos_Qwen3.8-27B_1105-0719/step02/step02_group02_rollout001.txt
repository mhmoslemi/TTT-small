def compute_corrs(h):

def grad_corr_k(h, k, n):

        improved = False
        for _ in range(3):
            for i in range(n):
                if time.time() - t_start > time_limit:
                    break
                old = h[i]
                for delta in [step, -step, step * 0.5, -step * 0.5]:
                    h[i] = np.clip(old + delta, 0.0, 1.0)
                    h = project_feasible(h, target_sum)
                    if c5_of_h(h, dx) < c5_val - 1e-12:
                        c5_val = c5_of_h(h, dx)
                        improved = True
                        break
                h[i] = old
        if not improved:
            step *= 0.7
            if step < 1e-6:
                break

    return best_h, best_c5


def subgrad_momentum(h0, dx, target_sum, max_iter=20000, time_limit=120.0):
    n = len(h0)
    h = h0.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    alpha = 0.1
    gamma = 0.9
    no_improve = 0
    t_start = time.time()
    top_m = max(8, n // 5)

    for it in range(max_iter):
        if time.time() - t_start > time_limit:
            break
        corrs = compute_corrs(h)
        max_c = corrs.max()
        c5_val = max_c * dx

        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 8000:
            break

        sorted_idx = np.argsort(corrs)[-top_m:]
        grad = np.zeros(n)
        for k in sorted_idx:
            grad += grad_corr_k(h, k, n)
        grad /= len(sorted_idx)
        grad *= dx

        v = gamma * v - alpha * grad
        h_new = project_feasible(h + v, target_sum)
        new_c5 = c5_of_h(h_new, dx)

        if new_c5 < c5_val - 1e-16:
            h = h_new
            alpha = min(alpha * 1.005, 0.5)
        else:
            alpha *= 0.93
            v *= 0.4

        if alpha < 1e-14:
            break

    return best_h, best_c5


def binary_round(h, target_sum, rng):
    n = len(h)
    h_bin = (h > 0.5).astype(float)
    n_ones = int(h_bin.sum())
    if n_ones != int(target_sum):
        if n_ones > target_sum:
            one_idx = np.where(h_bin == 1)[0]
            remove = target_sum - n_ones
            for i in range(abs(int(remove))):
                j = rng.randint(len(one_idx))
                h_bin[one_idx[j]] = 0
        else:
            zero_idx = np.where(h_bin == 0)[0]
            add = target_sum - n_ones
            for i in range(abs(int(add))):
                j = rng.randint(len(zero_idx))
                h_bin[zero_idx[j]] = 1
    return project_feasible(h_bin, target_sum)


def make_binary_start(n, target_sum, rng):
    h = np.zeros(n)
    idx = rng.permutation(n)
    n_ones = int(target_sum)
    h[idx[:n_ones]] = 1.0
    frac = target_sum - n_ones
    if frac > 1e-10 and n_ones < n:
        h[idx[n_ones]] = frac
    return project_feasible(h, target_sum)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def remaining():
        return budget_s - (time.time() - t0)

    best_h = None
    best_c5 = float('inf')

    h0 = np.array(initial_h_values, dtype=float)
    n = len(h0)
    dx = 2.0 / n
    target_sum = n / 2.0
    h0 = project_feasible(h0, target_sum)
    best_h = h0.copy()
    best_c5 = c5_of_h(h0, dx)

    # Phase 1: SLSQP with increasing beta
    for beta in [50, 150, 400, 1000, 2500, 5000]:
        if remaining() < 25:
            break
        h_opt, c5_val = slsqp_opt(best_h, dx, target_sum, beta, 600, time_limit=25.0)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_opt

    # Phase 2: Subgradient momentum
    if remaining() > 150:
        h_sg, c5_sg = subgrad_momentum(best_h, dx, target_sum, 25000, time_limit=120.0)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg

    # Phase 3: High-beta SLSQP polish
    if remaining() > 50:
        for beta in [5000, 10000, 20000]:
            if remaining() < 20:
                break
            h_opt, c5_val = slsqp_opt(best_h, dx, target_sum, beta, 800, time_limit=20.0)
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h_opt

    # Phase 4: Coordinate descent polish
    if remaining() > 40:
        h_cd, c5_cd = coord_descent_polish(best_h, dx, target_sum, max_rounds=300, time_limit=35.0)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd

    # Phase 5: Binary rounding + optimize
    if remaining() > 60:
        t_rounded = binary_round(best_h, target_sum, rng)
        h_opt, c5_val = slsqp_opt(t_rounded, dx, target_sum, 500, 600, time_limit=20.0)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_opt
            if remaining() > 25:
                h_opt2, c5_val2 = slsqp_opt(best_h, dx, target_sum, 5000, 800, time_limit=20.0)
                if c5_val2 < best_c5:
                    best_c5 = c5_val2
                    best_h = h_opt2

    # Phase 6: Perturbation + re-optimization
    for trial in range(15):
        if remaining() < 15:
            break
        mag = 0.01 + 0.05 * (trial / 15.0)
        h_pert = best_h + rng.normal(0, mag, len(best_h))
        h_pert = project_feasible(h_pert, target_sum)
        h_opt, c5_val = slsqp_opt(h_pert, dx, target_sum, 300, 400, time_limit=15.0)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_opt
            if remaining() > 30:
                h_opt2, c5_val2 = slsqp_opt(best_h, dx, target_sum, 5000, 700, time_limit=20.0)
                if c5_val2 < best_c5:
                    best_c5 = c5_val2
                    best_h = h_opt2

    # Phase 7: Dimension exploration
    n_b = len(best_h)
    n_candidates = []
    for delta in [-20, -15, -10, -5, 5, 10, 15, 20, 25, 30]:
        n_try = n_b + delta
        if n_try >= 40 and n_try % 2 == 0:
            n_candidates.append(n_try)
    n_candidates = sorted(set(n_candidates))

    for n_try in n_candidates:
        if remaining() < 50:
            break
        dx_try = 2.0 / n_try
        ts_try = n_try / 2.0
        x_old = (np.arange(len(best_h)) + 0.5) * (2.0 / len(best_h))
        x_new = (np.arange(n_try) + 0.5) * (2.0 / n_try)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, ts_try)

        h_opt, c5_val = slsqp_opt(h_interp, dx_try, ts_try, 200, 400, time_limit=15.0)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_opt
            if remaining() > 40:
                dx_p = 2.0 / len(best_h)
                ts_p = len(best_h) / 2.0
                h_sg, c5_sg = subgrad_momentum(best_h, dx_p, ts_p, 10000, time_limit=30.0)
                if c5_sg < best_c5:
                    best_c5 = c5_sg
                    best_h = h_sg
            if remaining() > 30:
                dx_p = 2.0 / len(best_h)
                ts_p = len(best_h) / 2.0
                h_opt2, c5_val2 = slsqp_opt(best_h, dx_p, ts_p, 5000, 700, time_limit=20.0)
                if c5_val2 < best_c5:
                    best_c5 = c5_val2
                    best_h = h_opt2

    # Phase 8: Random binary starts
    n_b = len(best_h)
    dx_b = 2.0 / n_b
    ts_b = n_b / 2.0
    for trial in range(15):
        if remaining() < 15:
            break
        h_bin = make_binary_start(n_b, ts_b, rng)
        h_opt, c5_val = slsqp_opt(h_bin, dx_b, ts_b, 300, 300, time_limit=12.0)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_opt
            if remaining() > 20:
                h_opt2, c5_val2 = slsqp_opt(best_h, dx_b, ts_b, 5000, 600, time_limit=15.0)
                if c5_val2 < best_c5:
                    best_c5 = c5_val2
                    best_h = h_opt2

    # Phase 9: Final intensive polish
    n_final = len(best_h)
    dx_f = 2.0 / n_final
    ts_f = n_final / 2.0

    if remaining() > 20:
        h_cd, c5_cd = coord_descent_polish(best_h, dx_f, ts_f, max_rounds=200, time_limit=15.0)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd

    if remaining() > 20:
        for beta in [10000, 20000, 50000]:
            if remaining() < 12:
                break
            h_final, c5_final = slsqp_opt(best_h, dx_f, ts_f, beta, 1000, time_limit=10.0)
            if c5_final < best_c5:
                best_c5 = c5_final
                best_h = h_final

    if remaining() > 15:
        h_sg, c5_sg = subgrad_momentum(best_h, dx_f, ts_f, 10000, time_limit=12.0)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg

    if remaining() > 10:
        h_cd, c5_cd = coord_descent_polish(best_h, dx_f, ts_f, max_rounds=150, time_limit=8.0)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd

    # Final validation
    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    final_c5 = c5_of_h(best_h, 2.0 / len(best_h))
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
