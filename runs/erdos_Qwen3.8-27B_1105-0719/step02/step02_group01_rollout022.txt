def pair_transfer_optimal(h, a, b, active_indices, corrs):
    """Find optimal δ to transfer from a to b."""
    n = len(h)
    d = []
    for k in active_indices:
        off = k - (n-1)
        # Change in corr[k] per unit δ transferred from a to b
        delta_k = 0.0
        # h[b] increases by δ
        if 0 <= b+off < n:
            delta_k += (1.0 - h[b+off])
        if 0 <= b-off < n:
            delta_k -= h[b-off]
        # h[a] decreases by δ
        if 0 <= a+off < n:
            delta_k -= (1.0 - h[a+off])
        if 0 <= a-off < n:
            delta_k += h[a-off]
        d.append(delta_k)
    
    d = np.array(d)
    c = corrs[active_indices]
    
    # Minimize max_k (c_k + δ * d_k) over δ
    # Feasible range: max(-h[a], 0) ≤ δ ≤ min(1-h[b], 1-h[a]+h[a]) ... 
    # Actually: h[a] - δ ≥ 0 => δ ≤ h[a]
    #           h[b] + δ ≤ 1 => δ ≤ 1 - h[b]
    # So δ ∈ [0, min(h[a], 1-h[b])] (if transferring from a to b)
    # Or δ can be negative (transferring from b to a):
    # δ ∈ [-min(h[b], 1-h[a]), min(h[a], 1-h[b])]
    
    delta_lo = -min(h[b], 1.0 - h[a])

    def f(delta):
        return np.max(c + delta * d)
    
    def slope(delta):
        vals = c + delta * d
        mx = vals.max()
        active = np.where(np.abs(vals - mx) < 1e-10)[0]
        return d[active]  # subgradient
    
    # Try the endpoints and a few interior points

            h_curr[a] -= d_opt
            h_curr[b] += d_opt
            h_curr = project_feasible(h_curr, target_sum)
            new_c5 = c5_of_h(h_curr, dx)
            if new_c5 < c5_val - 1e-14:
                c5_val = new_c5
                improved = True
            else:
                h_curr = h_save

        if not improved:
            break

    return best_h, best_c5


def make_binary_start(n, rng):
    h = np.zeros(n)
    idx = rng.permutation(n)
    h[idx[:int(n / 2)]] = 1.0
    return h


def make_periodic_start(n, rng, period):
    h = np.zeros(n)
    fill = max(1, int(n / 2.0 / n * period))
    for i in range(n):
        if (i % period) < fill:
            h[i] = 1.0
    return project_feasible(h, n / 2.0)


def interpolate_h(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0.0, 2.0, n_old)
    x_new = np.linspace(0.0, 2.0, n_new)
    return np.interp(x_new, x_old, h_old)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def remaining():
        return budget_s - (time.time() - t0)

    h0 = np.array(initial_h_values, dtype=float)
    n = len(h0)
    dx = 2.0 / n
    target_sum = n / 2.0
    h0 = project_feasible(h0, target_sum)

    best_h = h0.copy()
    best_c5 = c5_of_h(h0, dx)

    # Phase 1: Multi-parameter LP search on original n
    param_list = [
        (20, 0.3, 800),
        (30, 0.2, 800),
        (40, 0.15, 800),
        (30, 0.1, 1000),
        (50, 0.1, 800),
        (25, 0.05, 1000),
        (40, 0.05, 800),
        (60, 0.03, 800),
        (30, 0.02, 1000),
        (50, 0.01, 1000),
        (80, 0.01, 600),
    ]
    for na, tr, mi in param_list:
        if remaining() < 15:
            break
        h_opt, c5_val = lp_search(best_h, dx, target_sum, na, tr, mi)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_opt

    # Phase 2: Pair CD polish
    if remaining() > 30:
        h_cd, c5_cd = pair_cd_polish(best_h, dx, target_sum, 80, rng)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd
        if remaining() > 20:
            h_opt, c5_val = lp_search(best_h, dx, target_sum, 40, 0.05, 600)
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h_opt

    # Phase 3: Try different n values
    n_candidates = [110, 112, 116, 118, 120, 122, 124, 126, 128, 130, 132, 134, 136, 138, 140, 144, 148, 150, 152, 156, 160, 164, 168]
    for n_try in n_candidates:
        if remaining() < 20:
            break
        if n_try == len(best_h):
            continue
        dx_try = 2.0 / n_try
        ts_try = n_try / 2.0
        h_interp = interpolate_h(best_h, n_try)
        h_interp = project_feasible(h_interp, ts_try)

        h_opt, c5_val = lp_search(h_interp, dx_try, ts_try, 30, 0.2, 500)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_opt
        if remaining() < 15:
            break
        h_opt2, c5_val2 = lp_search(best_h, 2.0 / len(best_h), len(best_h) / 2.0, 40, 0.1, 600)
        if c5_val2 < best_c5:
            best_c5 = c5_val2
            best_h = h_opt2

    # Phase 4: Random restarts on best n
    n_best = len(best_h)
    dx_best = 2.0 / n_best
    ts_best = n_best / 2.0

    for trial in range(30):
        if remaining() < 15:
            break
        if trial % 4 == 0:
            h_init = make_binary_start(n_best, rng)
        elif trial % 4 == 1:
            h_init = make_periodic_start(n_best, rng, rng.choice([3, 4, 5, 6, 7, 8]))
        elif trial % 4 == 2:
            pert = 0.15 * (rng.random(n_best) - 0.5)
            h_init = project_feasible(best_h + pert, ts_best)
        else:
            pert = 0.05 * (rng.random(n_best) - 0.5)
            h_init = project_feasible(best_h + pert, ts_best)

        h_opt, c5_val = lp_search(h_init, dx_best, ts_best, 30, 0.2, 400)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_opt
            if remaining() > 20:
                h_opt2, c5_val2 = lp_search(h_opt, dx_best, ts_best, 40, 0.1, 400)
                if c5_val2 < best_c5:
                    best_c5 = c5_val2
                    best_h = h_opt2
                if remaining() > 15:
                    h_cd, c5_cd = pair_cd_polish(best_h, dx_best, ts_best, 40, rng)
                    if c5_cd < best_c5:
                        best_c5 = c5_cd
                        best_h = h_cd

    # Phase 5: Final intensive polish
    if remaining() > 10 and best_h is not None:
        n_f = len(best_h)
        dx_f = 2.0 / n_f
        ts_f = n_f / 2.0
        for na, tr, mi in [(30, 0.05, 800), (50, 0.03, 800), (40, 0.02, 1000), (60, 0.01, 800)]:
            if remaining() < 8:
                break
            h_opt, c5_val = lp_search(best_h, dx_f, ts_f, na, tr, mi)
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h_opt
        if remaining() > 10:
            h_cd, c5_cd = pair_cd_polish(best_h, dx_f, ts_f, 60, rng)
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd
            if remaining() > 8:
                h_opt, c5_val = lp_search(best_h, dx_f, ts_f, 50, 0.02, 800)
                if c5_val < best_c5:
                    best_c5 = c5_val
                    best_h = h_opt

    return best_h, best_c5, len(best_h)
