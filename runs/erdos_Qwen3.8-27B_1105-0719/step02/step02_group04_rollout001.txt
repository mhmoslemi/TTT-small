            break
        
        active = select_active(c_arr, margin=margin, pad=2)
        if len(active) > n:
            active = select_active(c_arr, margin=margin * 0.4, pad=1)
        
        # Build full Jacobian
        J = np.zeros((len(active), n))
        for i, k in enumerate(active):
            d = k - (n - 1)
            J[i] = h_grad_row(d, h, n, dx)
        
        # Project out the constraint (sum preservation): use null space of ones
        ones = np.ones(n)
        J_null = J - J.dot(ones) / n * np.ones((len(active), 1))  # remove mean component
        
        target = c_current - eta
        c_active = c_arr[active]
        r = target - c_active
        
        try:
            A = J_null.T @ J_null + lam * np.eye(n)
            b = J_null.T @ r
            delta = np.linalg.solve(A, b)
            delta -= np.mean(delta)  # enforce sum preservation
        except:
            stall += 1
            eta = max(eta * 0.5, 5e-6)
            lam = min(lam * 3, 10.0)
            if stall > 50:
                break
            continue
        
        if not np.all(np.isfinite(delta)) or np.linalg.norm(delta) < 1e-13:
            stall += 1
            eta = max(eta * 0.6, 5e-6)
            if stall > 50:
                break
            continue
        
        best_step_c = c_current
        best_step_h = None
        for alpha in [1.0, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05]:
            h_trial = np.clip(h + alpha * delta, 0.0, 1.0)
            h_trial = project_to_target(h_trial, target_sum)
            c_trial = eval_c5(h_trial)
            if c_trial < best_step_c - 1e-12:
                best_step_c = c_trial
                best_step_h = h_trial
        
        if best_step_h is not None:
            h = best_step_h
            c_current = best_step_c
            stall = 0
            eta = min(eta * 1.3, 0.004)
            lam = max(lam * 0.7, 0.001)
        else:
            stall += 1
            if stall % 4 == 1:
                eta = max(eta * 0.5, 5e-6)
            if stall % 4 == 2:
                margin = min(margin * 1.4, 0.06)
            if stall % 4 == 3:
                lam = min(lam * 2.5, 15.0)
            if stall > 60:
                break
    
    return best_h, best_c

def random_sym_q(n, rng, base_q=None, scale=0.1):
    m = n // 2
    if base_q is not None:
        q = base_q + rng.randn(m) * scale
    else:
        t = np.linspace(0, 1, m)
        a1, a2, a3 = rng.uniform(-0.3, 0.3, 3)
        p1, p2, p3 = rng.uniform(0, 2 * np.pi, 3)
        q = 0.5 + a1 * np.sin(2 * np.pi * t + p1) + a2 * np.sin(4 * np.pi * t + p2) + a3 * np.sin(6 * np.pi * t + p3)
        q += rng.randn(m) * 0.1
    return np.clip(q, 0.0, 1.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0_global = time.time()
    rng = np.random.RandomState(seed)
    reserve = 30.0
    
    # Initialize
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    
    try:
        h_init = np.array(initial_h_values, dtype=float).reshape(-1)
        if h_init.size > 0 and np.all(np.isfinite(h_init)):
            n_init = len(h_init)
            h_init = project_to_target(h_init, n_init / 2.0)
            best_h = h_init.copy()
            best_c5 = eval_c5(h_init)
            best_n = n_init
    except Exception:
        pass
    
    if best_h is None:
        n_def = 126
        m_def = n_def // 2
        t = np.linspace(0, 1, m_def)
        q_def = 0.5 + 0.2 * np.sin(2 * np.pi * 3 * t) + 0.1 * np.sin(4 * np.pi * 5 * t)
        best_h = build_sym(np.clip(q_def, 0, 1), n_def)
        best_c5 = eval_c5(best_h)
        best_n = n_def
    
    def time_left():
        return budget_s - (time.time() - t0_global)
    
    def accept(h_new, c_new):
        nonlocal best_h, best_c5, best_n
        if c_new < best_c5 - 1e-12:
            best_h = h_new.copy()
            best_c5 = c_new
            best_n = len(h_new)
            return True
        return False
    
    # Phase 1: Symmetric active-set optimization on initial n
    n_cur = best_n
    if n_cur % 2 == 0:
        q = to_sym_q(best_h, n_cur)
        h_sym = build_sym(q, n_cur)
        c_sym = eval_c5(h_sym)
        if c_sym < best_c5:
            best_h = h_sym.copy()
            best_c5 = c_sym
        q = to_sym_q(best_h, n_cur)
        
        # Multiple rounds with increasing effort
        for round_num in range(5):
            if time_left() < reserve + 20:
                break
            alloc = min(120.0, time_left() - reserve)
            q, c_val = active_set_optimize(q, n_cur, max_iter=300, time_limit_s=alloc, t0=t0_global + (time.time() - t0_global))
            # Fix: use absolute time
            q, c_val = active_set_optimize(q, n_cur, max_iter=300, time_limit_s=min(100, time_left() - reserve), t0=time.time())
            h_new = build_sym(q, n_cur)
            if c_val < best_c5:
                best_h = h_new.copy()
                best_c5 = c_val
    
    # Phase 2: Try full-variable optimization to break symmetry
    if time_left() > reserve + 40:
        n_f = best_n if best_n % 2 == 0 else best_n - 1
        h_f = best_h if len(best_h) == n_f else interp_to(best_h, n_f)
        h_res, c_res = full_var_optimize(h_f, n_f, max_iter=150, time_limit_s=min(80, time_left() - reserve), t0=time.time())
        accept(h_res, c_res)
    
    # Phase 3: Try different n values
    if time_left() > reserve + 30:
        n_base = best_n
        n_list = sorted(set([
            max(32, n_base - 20),
            max(32, n_base - 12),
            max(32, n_base - 6),
            max(32, n_base - 4),
            max(32, n_base - 2),
            n_base + 2,
            n_base + 4,
            n_base + 6,
            n_base + 12,
            min(256, n_base + 20),
        ]))
        n_list = [x for x in n_list if x % 2 == 0 and x >= 32]
        
        for n_try in n_list:
            if time_left() < reserve + 15:
                break
            if n_try == best_n:
                continue
            h_try = interp_to(best_h, n_try)
            
            if n_try % 2 == 0:
                q_try = to_sym_q(h_try, n_try)
                q_try, c_try = active_set_optimize(q_try, n_try, max_iter=200, time_limit_s=min(40, time_left() - reserve), t0=time.time())
                h_try = build_sym(q_try, n_try)
                c_try = eval_c5(h_try)
                if accept(h_try, c_try):
                    continue
            
            # Also try full-var on this n
            if time_left() > reserve + 15:
                h_r, c_r = full_var_optimize(h_try, n_try, max_iter=80, time_limit_s=min(25, time_left() - reserve), t0=time.time())
                accept(h_r, c_r)
    
    # Phase 4: Random restarts / structured perturbations
    if time_left() > reserve + 10:
        max_restarts = min(30, int(max(0, time_left() - reserve) / 12.0))
        for trial in range(max_restarts):
            if time_left() < reserve + 8:
                break
            n_r = best_n if best_n % 2 == 0 else best_n - 2
            if n_r < 32:
                n_r = 32
            q_base = to_sym_q(best_h, best_n) if best_n % 2 == 0 and best_n == n_r else None
            q_r = random_sym_q(n_r, rng, base_q=q_base, scale=0.05 + 0.05 * rng.rand())
            h_r = build_sym(q_r, n_r)
            q_r, c_r = active_set_optimize(q_r, n_r, max_iter=100, time_limit_s=min(10, time_left() - reserve), t0=time.time())
            h_r = build_sym(q_r, n_r)
            c_r = eval_c5(h_r)
            accept(h_r, c_r)
    
    # Phase 5: Final intensive polish on best
    if time_left() > 15 and best_h is not None:
        n_fin = len(best_h)
        if n_fin % 2 == 0:
            q_fin = to_sym_q(best_h, n_fin)
            q_fin, c_fin = active_set_optimize(q_fin, n_fin, max_iter=400, time_limit_s=min(60, time_left() - 10), t0=time.time())
            h_fin = build_sym(q_fin, n_fin)
            accept(h_fin, eval_c5(h_fin))
        
        if time_left() > 12:
            h_fin2 = best_h if len(best_h) % 2 == 0 else interp_to(best_h, len(best_h) - 1)
            h_res, c_res = full_var_optimize(h_fin2, len(h_fin2), max_iter=100, time_limit_s=min(30, time_left() - 8), t0=time.time())
            accept(h_res, c_res)
    
    # Final validation
    if best_h is None:
        n_fb = 64
        m_fb = n_fb // 2
        t_fb = np.linspace(0, 1, m_fb)
        q_fb = np.clip(0.5 + 0.2 * np.sin(2 * np.pi * 2 * t_fb), 0, 1)
        best_h = build_sym(q_fb, n_fb)
    
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_target(best_h, len(best_h) / 2.0)
    best_c5 = eval_c5(best_h)
    
    return best_h, best_c5, len(best_h)
