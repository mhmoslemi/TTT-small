from scipy.optimize import minimize
import time

def project_feasible(x, target_sum):
    """Project x onto {h: 0<=h<=1, sum(h)=target_sum}."""
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
        else:
            h += diff / n
    return np.clip(h, 0.0, 1.0)

def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def c5_of_h(h, dx):
    return compute_corrs(h).max() * dx

def active_lags(h, tol=1e-10):
    """Return indices of lags at or near the maximum."""
    corrs = compute_corrs(h)
    m = corrs.max()
    return np.where(corrs >= m - tol)[0]

def compute_swap_delta(h, i, j, n):
    """Compute Δc (change in full correlate array) for swap h[i]:1->0, h[j]:0->1."""
    delta = np.zeros(2*n - 1)
    shift_offset = n - 1  # shift s corresponds to index s + shift_offset
    
    # Term 1: -(1-h[i-s]) for s in [i-n+1, i]
    s_min1 = max(i - n + 1, -(n-1))
    s_max1 = min(i, n-1)
    for s in range(s_min1, s_max1 + 1):
        idx_base = i - s
        delta[s + shift_offset] -= (1.0 - h[idx_base])
    
    # Term 2: +h[i+s] for s in [-i, n-i-1]
    s_min2 = max(-i, -(n-1))
    s_max2 = min(n - i - 1, n-1)
    for s in range(s_min2, s_max2 + 1):
        idx_shift = i + s
        delta[s + shift_offset] += h[idx_shift]
    
    # Term 3: +(1-h[j-s]) for s in [j-n+1, j]
    s_min3 = max(j - n + 1, -(n-1))
    s_max3 = min(j, n-1)
    for s in range(s_min3, s_max3 + 1):
        idx_base = j - s
        delta[s + shift_offset] += (1.0 - h[idx_base])
    
    # Term 4: -h[j+s] for s in [-j, n-j-1]
    s_min4 = max(-j, -(n-1))
    s_max4 = min(n - j - 1, n-1)
    for s in range(s_min4, s_max4 + 1):
        idx_shift = j + s
        delta[s + shift_offset] -= h[idx_shift]
    
    return delta
