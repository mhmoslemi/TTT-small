import time

def project_feasible(x, target_sum):
    """Project x onto {h: 0<=h<=1, sum(h)=target_sum} using water-filling."""
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)

    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
        else:
            h += diff / n
    return np.clip(h, 0.0, 1.0)

def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def c5_of_h(h, dx):
    return compute_corrs(h).max() * dx

def compute_sig(h):
    """Compute signature matrix: sig[i, k] = h[i+m_k] + h[i-m_k] with out-of-bounds = 0.
    m_k = k - (n-1) for k = 0, ..., 2n-2."""
    n = len(h)
    m = np.arange(-(n-1), n)  # length 2n-1
    raw1 = np.arange(n)[:, None] + m[None, :]  # (n, 2n-1)

    valid1 = (raw1 >= 0) & (raw1 < n)
    valid2 = (raw2 >= 0) & (raw2 < n)
    clip1 = np.clip(raw1, 0, n-1)
    clip2 = np.clip(raw2, 0, n-1)
    sig = np.where(valid1, h[clip1], 0.0) + np.where(valid2, h[clip2], 0.0)
    return sig

Optimization: we don't need to try all pairs. We can focus on pairs that are "good" for the active lags. For the single most active lag m*, we want sig[i][m*] to be small and sig[j][m*] to be large. So we can:

- Sort the zeros by sig[j][m*] (descending)

Even better: for a single active lag, the optimal move is to take from the position i with smallest sig[i][m*] and give to the position j with largest sig[j][m*]. Then check if this helps or hurts the other active lags.

import numpy as np

def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
        else:
            h += diff / n
    return np.clip(h, 0.0, 1.0)


def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_sig(h):
    n = len(h)
    m = np.arange(-(n - 1), n)
    raw1 = np.arange(n)[:, None] + m[None, :]
    raw2 = np.arange(n)[:, None] - m[None, :]
    valid1 = (raw1 >= 0) & (raw1 < n)
    valid2 = (raw2 >= 0) & (raw2 < n)
    clip1 = np.clip(raw1, 0, n - 1)
    clip2 = np.clip(raw2, 0, n - 1)
    return np.where(valid1, h[clip1], 0.0) + np.where(valid2, h[clip2], 0.0)

def active_set_continuous(h, dx, max_iters=500, eps_tol=1e-12):
    """Exact active-set continuous optimization using sig matrix."""
    n = len(h)
    target_sum = n / 2.0
    h = h.copy().astype(float)
    corrs = compute_corrs(h)
    sig = compute_sig(h)
    best_c5 = corrs.max() * dx
    best_h = h.copy()

    for it in range(max_iters):
        M = corrs.max()
        # Active lags: within 0.05 of max
        active_thresh = M - 0.05
        active_mask = corrs >= active_thresh
        active_idx = np.where(active_mask)[0]

        # Find best single-pair move
        ones_mask = h > 1e-8
        zeros_mask = h < 1.0 - 1e-8
        ones_idx = np.where(ones_mask)[0]
        zeros_idx = np.where(zeros_mask)[0]

        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break

        # Rank candidates: for the worst lag, prefer small sig on ones, large sig on zeros
        worst_k = int(corrs.argmax())
        i_rank = ones_idx[np.argsort(sig[ones_idx, worst_k])[:40]]
        j_rank = zeros_idx[np.argsort(-sig[zeros_idx, worst_k])[:40]]

        best_eps = 0.0
        best_i, best_j = -1, -1
        best_new_max = M

        for i in i_rank:
            for j in j_rank:
                if i == j:
                    continue
                delta = sig[i] - sig[j]
                # We need corr + eps*delta <= M for all k (to not increase max)
                # And we want to actually reduce the max

                if pos_mask.any():
                    eps_limits = (M - corrs[pos_mask]) / delta[pos_mask]

                    if eps_limits.size == 0:
                        continue
                    eps_upper = eps_limits.min()
                else:
                    eps_upper = float('inf')
                # Also bounded by feasibility

                if eps_upper <= 1e-12:
                    continue
                # Check new max
                new_corrs = corrs + eps_upper * delta
                new_max = new_corrs.max()
                # We want new_max < M (strict improvement)
                if new_max < best_new_max - 1e-13:
                    # Can we do better with a larger eps?
                    # Actually we want the eps that makes new_max as small as possible
                    # Try the full eps_upper
                    best_new_max = new_max
                    best_eps = eps_upper
                    best_i, best_j = i, j

        if best_i < 0 or best_eps < eps_tol:
            # Try a broader search: all ones vs all zeros (but limited)
            step_i = max(1, len(ones_idx) // 20)
            step_j = max(1, len(zeros_idx) // 20)
            for ii in range(0, len(i_rank), max(1, len(i_rank) // 10)):
                for jj in range(0, len(j_rank), max(1, len(j_rank) // 10)):
                    i = i_rank[ii]
                    j = j_rank[jj]
                    if i == j:
                        continue
                    delta = sig[i] - sig[j]
                    pos_mask = delta > 1e-12
                    if pos_mask.any():
                        eps_limits = (M - corrs[pos_mask]) / delta[pos_mask]
                        eps_limits = eps_limits[eps_limits > 0]
                        if eps_limits.size == 0:
                            continue
                        eps_upper = eps_limits.min()
                    else:
                        eps_upper = float('inf')
                    eps_upper = min(eps_upper, h[i] * 0.99, (1.0 - h[j]) * 0.99)

                        continue
                    new_corrs = corrs + eps_upper * delta
                    new_max = new_corrs.max()
                    if new_max < best_new_max - 1e-13:
                        best_new_max = new_max
                        best_eps = eps_upper

            if best_i < 0:

        # Take the step
        h[best_i] -= best_eps
        h[best_j] += best_eps
        h = np.clip(h, 0.0, 1.0)

        if abs(s - target_sum) > 1e-14:

def smooth_max_grad(h, dx, beta):
    n = len(h)
    corrs = compute_corrs(h)
    m = corrs.max()
    shifted = corrs - m
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    w = w / w_sum
    grad = np.zeros(n)
    for k in range(len(corrs)):
        if w[k] < 1e-14:
            continue
        offset = k - (n - 1)
        i_lo = max(0, -offset)
        i_hi = min(n - 1, n - 1 - offset)
        if i_lo > i_hi:
            continue
        idx = np.arange(i_lo, i_hi + 1)
        grad[idx + offset] += (1.0 - h[idx]) * w[k]
        grad[idx] += (-h[idx + offset]) * w[k]
    return grad * dx


def slsqp_polish(h0, dx, target_sum, beta, maxiter=800):
    n = len(h0)

    def obj(x):
        corrs = compute_corrs(x)
        m = corrs.max()
        return (m + np.log(np.sum(np.exp(beta * (corrs - m)))) / beta) * dx

    def cons_fun(x):
        return x.sum() - target_sum

    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0, jac=lambda x: smooth_max_grad(x, dx, beta),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        h_out = project_feasible(res.x, target_sum)
    except Exception:
        h_out = project_feasible(h0, target_sum)
    return h_out, c5_of_h(h_out, dx)


def make_structured_init(n, target_count, rng, mode=0):
    h = np.zeros(n)
    if mode == 0:
        phi = (1.0 + np.sqrt(5.0)) / 2.0
        for k in range(target_count):
            pos = int((k * phi * n)) % n
            h[pos] = 1.0
    elif mode == 1:
        # Thue-Morse-like
        for k in range(n):
            c = bin(k).count('1')
            if c % 2 == 0:
                h[k] = 1.0
        # Adjust to exact count
        total = h.sum()
        if total > target_count:
            excess = int(total) - target_count
            idx = np.where(h == 1.0)[0]
            h[idx[:excess]] = 0.0
        elif total < target_count:
            deficit = target_count - int(total)
            idx = np.where(h == 0.0)[0]
            h[idx[:deficit]] = 1.0
    elif mode == 2:
        # Interleaved blocks
        block = max(1, n // (2 * target_count))
        for k in range(target_count):
            start = k * (n // target_count)
            h[start] = 1.0
    else:
        idx = rng.permutation(n)
        h[idx[:target_count]] = 1.0
    return project_feasible(h, target_count)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    # Initialize from parent
    try:
        best_h = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        best_h = np.full(126, 0.5)

    n = len(best_h)
    dx = 2.0 / n
    target_sum = n / 2.0
    best_h = project_feasible(best_h, target_sum)
    best_c5 = c5_of_h(best_h, dx)

    # Phase 1: Binary conversion + greedy local search
    if remaining_time(t0, budget_s) > 60:
        h_bin = to_binary(best_h, n // 2)
        h_g, c5_g = binary_greedy(h_bin, dx, rng, max_sweeps=150, use_2for2=True)
        if c5_g < best_c5:
            best_c5 = c5_g
            best_h = h_g

    # Phase 2: Active-set continuous optimization from best binary
    if remaining_time(t0, budget_s) > 80:
        h_act, c5_act = active_set_continuous(best_h, dx, max_iters=300)
        if c5_act < best_c5:
            best_c5 = c5_act
            best_h = h_act

    # Phase 3: SLSQP with high beta
    if remaining_time(t0, budget_s) > 100:
        h_sl, c5_sl = slsqp_polish(best_h, dx, target_sum, beta=500, maxiter=1000)
        if c5_sl < best_c5:
            best_c5 = c5_sl
            best_h = h_sl

    # Phase 4: More active-set from SLSQP result
    if remaining_time(t0, budget_s) > 60:
        h_act2, c5_act2 = active_set_continuous(best_h, dx, max_iters=300)
        if c5_act2 < best_c5:
            best_c5 = c5_act2
            best_h = h_act2

    # Phase 5: Binary SA with 2-for-2 from current best
    if remaining_time(t0, budget_s) > 200:
        h_bin2 = to_binary(best_h, n // 2)
        h_sa, c5_sa = binary_sa(h_bin2, dx, rng, max_iters=300000, patience=6000,
                                temp0=0.06, cooling=0.99983, use_2for2=True, two_for2_ratio=0.35)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa
            # Polish the new binary result
            if remaining_time(t0, budget_s) > 60:
                h_a3, c5_a3 = active_set_continuous(best_h, dx, max_iters=200)
                if c5_a3 < best_c5:
                    best_c5 = c5_a3
                    best_h = h_a3

    # Phase 6: Try different structured starts
    for mode in range(4):
        if remaining_time(t0, budget_s) < 80:
            break
        h_init = make_structured_init(n, n // 2, rng, mode)
        h_g2, c5_g2 = binary_greedy(h_init, dx, rng, max_sweeps=80, use_2for2=True)
        if c5_g2 < best_c5:
            best_c5 = c5_g2
            best_h = h_g2
            if remaining_time(t0, budget_s) > 50:
                h_a4, c5_a4 = active_set_continuous(best_h, dx, max_iters=200)
                if c5_a4 < best_c5:
                    best_c5 = c5_a4
                    best_h = h_a4

    # Phase 7: Try n=128
    if remaining_time(t0, budget_s) > 120:
        n2 = 128
        dx2 = 2.0 / n2
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n2)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, n2 / 2.0)
        h_b128 = to_binary(h_interp, n2 // 2)
        h_g128, c5_g128 = binary_greedy(h_b128, dx2, rng, max_sweeps=100, use_2for2=True)
        if c5_g128 < best_c5:
            h_a128, c5_a128 = active_set_continuous(h_g128, dx2, max_iters=200)
            if c5_a128 < best_c5:
                best_c5 = c5_a128
                best_h = h_a128

    # Phase 8: Try n=132
    if remaining_time(t0, budget_s) > 120:
        n3 = 132
        dx3 = 2.0 / n3
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n3)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, n3 / 2.0)
        h_b132 = to_binary(h_interp, n3 // 2)
        h_g132, c5_g132 = binary_greedy(h_b132, dx3, rng, max_sweeps=80, use_2for2=True)
        if c5_g132 < best_c5:
            h_a132, c5_a132 = active_set_continuous(h_g132, dx3, max_iters=200)
            if c5_a132 < best_c5:
                best_c5 = c5_a132
                best_h = h_a132

    # Phase 9: Final polish cycle
    if remaining_time(t0, budget_s) > 60:
        h_final_act, c5_final_act = active_set_continuous(best_h, dx if len(best_h) == n else 2.0 / len(best_h), max_iters=300)
        if c5_final_act < best_c5:
            best_c5 = c5_final_act
            best_h = h_final_act

    if remaining_time(t0, budget_s) > 60:
        n_b = len(best_h)
        dx_b = 2.0 / n_b
        h_sl2, c5_sl2 = slsqp_polish(best_h, dx_b, n_b / 2.0, beta=1000, maxiter=800)
        if c5_sl2 < best_c5:
            best_c5 = c5_sl2
            best_h = h_sl2

    # Phase 10: One more binary SA if time
    if remaining_time(t0, budget_s) > 150:
        n_b = len(best_h)
        dx_b = 2.0 / n_b
        h_bin3 = to_binary(best_h, n_b // 2)
        h_sa3, c5_sa3 = binary_sa(h_bin3, dx_b, rng, max_iters=200000, patience=5000,
                                  temp0=0.04, cooling=0.9998, use_2for2=True, two_for2_ratio=0.4)
        if c5_sa3 < best_c5:
            best_c5 = c5_sa3
            best_h = h_sa3
            if remaining_time(t0, budget_s) > 40:
                h_a5, c5_a5 = active_set_continuous(best_h, dx_b, max_iters=150)
                if c5_a5 < best_c5:
                    best_c5 = c5_a5
                    best_h = h_a5

    # Final validation
    n_b = len(best_h)
    dx_b = 2.0 / n_b
    ts_b = n_b / 2.0
    best_h = project_feasible(best_h, ts_b)
    best_c5 = c5_of_h(best_h, dx_b)

    return best_h, best_c5, len(best_h)
