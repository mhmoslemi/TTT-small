    active = set(np.where(corrs >= threshold)[0].tolist())
    padded = set()
    for k in active:
        for d in range(-pad, pad + 1):
            if 0 <= k + d < len(corrs):
                padded.add(k + d)
    active_list = sorted(padded)
    if len(active_list) > max_k:
        scored = sorted([(corrs[k], k) for k in active_list], reverse=True)
        active_list = sorted([k for _, k in scored[:max_k]])
    return active_list

def corr_grad_h(h, k):
    """Gradient of corr_k with respect to h (before dx scaling)."""
    n = len(h)
    d = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -d)
    i_hi = min(n - 1, n - 1 - d)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx] += (1.0 - h[idx + d])
    grad[idx + d] += (-h[idx])
    return grad

def corr_grad_q(q, k, n_full):
    """Gradient of corr_k with respect to q (after dx scaling)."""
    h = build_sym_h(q)
    m = n_full // 2
    dx = 2.0 / n_full
    g_h = corr_grad_h(h, k)
    g_q = dx * (g_h[:m] - g_h[n_full-1:n_full-1-m:-1])
    return g_q

def lp_active_step(q, n_full, active_k, trust_radius=0.1):
    """Solve LP to find best direction within trust region."""
    m = n_full // 2
    h = build_sym_h(q)
    dx = 2.0 / n_full
    corrs = corr_all_scaled(h)
    
    # LP variables: [dq (m), t (1)]
    # Minimize t
    # Subject to: corr_k + grad_k . dq <= t for k in active
    #             -trust_radius <= dq_i <= trust_radius
    #             0 <= q[i] + dq[i] <= 1
    
    c_obj = np.zeros(m + 1)
    c_obj[-1] = 1.0  # minimize t
    
    A_ub = []
    b_ub = []
    
    for k in active_k:
        g = corr_grad_q(q, k, n_full)
        c_val = corrs[k]
        # c_val + g . dq <= t  =>  g . dq - t <= -c_val
        row = np.zeros(m + 1)
        row[:m] = g
        row[-1] = -1.0
        A_ub.append(row)
        b_ub.append(-c_val)
    
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    bounds = []
    for i in range(m):
        lo = max(-q[i], -trust_radius)
        hi = min(1.0 - q[i], trust_radius)
        bounds.append((lo, hi))
    bounds.append((-np.inf, np.inf))  # t is unbounded
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            dq = res.x[:m]
            return dq
    except Exception:
        pass
    return None

def lp_refine(q, n_full, max_rounds=30, time_limit=None):

def coord_descent(q, n_full, max_rounds=15, time_limit=None):
    """Coordinate descent with local 1D search."""
    m = n_full // 2
    h = build_sym_h(q)
    best_score = eval_score(h)
    rng = np.random.RandomState(42)
    
    for rnd in range(max_rounds):
        if time_limit and time.time() > time_limit:
            break
        improved = False
        order = rng.permutation(m)
        
        for r in order:
            if time_limit and time.time() > time_limit:
                break
            # Local search around current value
            qr0 = q[r]
            best_qr = qr0
            best_c = best_score
            
            # Try nearby values
            for step in [0.1, 0.05, 0.02, 0.01, 0.005]:
                for sign in [1, -1]:
                    qr_try = qr0 + sign * step
                    if qr_try < 0 or qr_try > 1:
                        continue
                    q[r] = qr_try
                    h_try = build_sym_h(q)
                    c_try = eval_score(h_try)
                    if c_try < best_c - 1e-14:
                        best_c = c_try
                        best_qr = qr_try
            
            if best_qr != qr0:
                q[r] = best_qr
                # Refine with smaller steps
                for step in [0.002, 0.001, 0.0005]:
                    for sign in [1, -1]:
                        qr_try = q[r] + sign * step
                        if qr_try < 0 or qr_try > 1:
                            continue
                        q[r] = qr_try
                        h_try = build_sym_h(q)
                        c_try = eval_score(h_try)
                        if c_try < best_c - 1e-14:
                            best_c = c_try
                            best_qr = qr_try
                q[r] = best_qr
                best_score = best_c
                improved = True
        
        h = build_sym_h(q)
        best_score = eval_score(h)
        if not improved:
            break
    
    return q, best_score, build_sym_h(q)

def smooth_max_optimize(q, n_full, beta_list, maxiter, time_limit):
    """Smooth max optimization with L-BFGS-B."""
    m = n_full // 2
    best_score = eval_score(build_sym_h(q))
    best_q = q.copy()
    
    for beta in beta_list:
        if time_limit and time.time() > time_limit:
            break
        
        def obj_and_grad(qv):
            h = build_sym_h(qv)
            corrs = corr_all_scaled(h)
            max_c = corrs.max()
            shifted = corrs - max_c
            w = np.exp(np.clip(beta * shifted, -50, 50))
            w_sum = w.sum()
            if w_sum < 1e-30:
                return max_c, np.zeros(m)
            w /= w_sum
            val = max_c + np.log(w_sum) / beta
            
            # Gradient
            gq = np.zeros(m)
            for k in range(len(corrs)):
                if w[k] < 1e-15:
                    continue
                g = corr_grad_q(qv, k, n_full)
                gq += w[k] * g
            return val, gq
        
        def obj_fn(qv):
            return obj_and_grad(qv)[0]
        
        def grad_fn(qv):
            return obj_and_grad(qv)[1]
        
        try:
            res = minimize(obj_fn, q, jac=grad_fn, method='L-BFGS-B',
                          bounds=[(0,1)]*m, options={'maxiter': maxiter, 'ftol': 1e-14})
            q_new = np.clip(res.x, 0, 1)
            h_new = build_sym_h(q_new)
            c_new = eval_score(h_new)
            if c_new < best_score:
                best_score = c_new
                best_q = q_new.copy()
            q = q_new
        except Exception:
            pass
    
    return best_q, best_score, build_sym_h(best_q)

from scipy.optimize import minimize, linprog
import time


def build_sym_h(q):
    m = len(q)
    h = np.empty(2 * m)
    h[:m] = q
    h[m:] = 1.0 - q[::-1]
    return h


def eval_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)


def corr_all_scaled(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx


def sym_from_h(h):
    n = len(h)
    m = n // 2
    q = 0.5 * (h[:m] + (1.0 - h[m:][::-1]))
    return np.clip(q, 0.0, 1.0)


def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(x + mid, 0.0, 1.0).sum()
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    h = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)

    threshold = max_val - max(1e-6, margin * max_val)
    active = set(np.where(corrs >= threshold)[0].tolist())
    padded = set()
    for k in active:
        for d in range(-pad, pad + 1):
            if 0 <= k + d < len(corrs):
                padded.add(k + d)
    active_list = sorted(padded)
    if len(active_list) > max_k:
        scored = sorted([(corrs[k], k) for k in active_list], reverse=True)
        active_list = sorted([k for _, k in scored[:max_k]])
    return active_list


def corr_grad_h(h, k):
    n = len(h)
    d = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -d)
    i_hi = min(n - 1, n - 1 - d)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx] += (1.0 - h[idx + d])
    grad[idx + d] += (-h[idx])
    return grad


def corr_grad_q_full(q, n_full):
    h = build_sym_h(q)
    m = n_full // 2
    dx = 2.0 / n_full
    g_q = np.zeros((2 * n_full - 1, m))
    for k in range(2 * n_full - 1):
        g_h = corr_grad_h(h, k)
        g_q[k] = dx * (g_h[:m] - g_h[n_full - 1:n_full - 1 - m:-1])
    return g_q


def lp_solve_sym(q, n_full, active_k, radius):
    m = n_full // 2
    h = build_sym_h(q)
    dx = 2.0 / n_full
    corrs = np.correlate(h, 1.0 - h, mode='full') * dx

    c_obj = np.zeros(m + 1)
    c_obj[-1] = 1.0

    A_ub = []
    b_ub = []
    for k in active_k:
        g_h = corr_grad_h(h, k)
        g = dx * (g_h[:m] - g_h[n_full - 1:n_full - 1 - m:-1])
        row = np.zeros(m + 1)
        row[:m] = g
        row[-1] = -1.0
        A_ub.append(row)
        b_ub.append(-corrs[k])

    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)

    bounds = []
    for i in range(m):
        lo = max(-q[i], -radius)
        hi = min(1.0 - q[i], radius)
        bounds.append((lo, hi))
    bounds.append((-np.inf, np.inf))

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x[:m]
    except Exception:
        pass
    return None


def lp_refine_sym(q, n_full, max_rounds=40, time_limit=None):
    h = build_sym_h(q)
    best_score = eval_score(h)

    for rnd in range(max_rounds):
        if time_limit and time.time() > time_limit:
            break
        corrs = corr_all_scaled(h)
        improved = False
        for margin in [0.03, 0.06, 0.10]:
            active = select_active(corrs, margin=margin, pad=3, max_k=55)
            for radius in [0.2, 0.1, 0.05, 0.02]:
                dq = lp_solve_sym(q, n_full, active, radius)
                if dq is None:
                    continue
                q_try = np.clip(q + dq, 0.0, 1.0)
                h_try = build_sym_h(q_try)
                c_try = eval_score(h_try)
                if c_try < best_score - 1e-13:
                    best_score = c_try
                    q = q_try.copy()
                    h = h_try
                    improved = True
                    break
            if improved:
                break
        if not improved:
            corrs = corr_all_scaled(h)
            active2 = select_active(corrs, margin=0.15, pad=5, max_k=70)
            dq = lp_solve_sym(q, n_full, active2, 0.03)
            if dq is not None:
                q_try = np.clip(q + dq, 0.0, 1.0)
                h_try = build_sym_h(q_try)
                c_try = eval_score(h_try)
                if c_try < best_score - 1e-13:
                    best_score = c_try
                    q = q_try.copy()
                    h = h_try
                    continue
            break
    return q, best_score, build_sym_h(q)


def coord_descent_sym(q, n_full, max_rounds=20, time_limit=None):
    m = n_full // 2
    h = build_sym_h(q)
    best_score = eval_score(h)
    rng = np.random.RandomState(123)

    for rnd in range(max_rounds):
        if time_limit and time.time() > time_limit:
            break
        improved = False
        order = rng.permutation(m)
        for r in order:
            if time_limit and time.time() > time_limit:
                break
            qr0 = q[r]
            best_qr = qr0
            best_c = best_score
            for step in [0.1, 0.05, 0.02, 0.01, 0.005, 0.002]:
                for sign in [1, -1]:
                    qr_try = qr0 + sign * step
                    if qr_try < -0.01 or qr_try > 1.01:
                        continue
                    qr_clipped = np.clip(qr_try, 0.0, 1.0)
                    if qr_clipped == q[r]:
                        continue
                    q[r] = qr_clipped
                    h_try = build_sym_h(q)
                    c_try = eval_score(h_try)
                    if c_try < best_c - 1e-14:
                        best_c = c_try
                        best_qr = qr_clipped
            if best_qr != qr0:
                q[r] = best_qr
                for step in [0.001, 0.0005, 0.0002]:
                    for sign in [1, -1]:
                        qr_try = q[r] + sign * step
                        qr_clipped = np.clip(qr_try, 0.0, 1.0)
                        if qr_clipped == q[r]:
                            continue
                        q[r] = qr_clipped
                        h_try = build_sym_h(q)
                        c_try = eval_score(h_try)
                        if c_try < best_c - 1e-14:
                            best_c = c_try
                            best_qr = qr_clipped
                q[r] = best_qr
                best_score = best_c
                improved = True
        h = build_sym_h(q)
        best_score = eval_score(h)
        if not improved:
            break
    return q, best_score, build_sym_h(q)


def smooth_obj_func(x, n_full, beta):
    h = build_sym_h(x)
    corrs = corr_all_scaled(h)
    max_c = corrs.max()
    shifted = corrs - max_c
    w = np.exp(np.clip(beta * shifted, -50.0, 50.0))
    w_sum = w.sum()
    if w_sum < 1e-30:
        return max_c
    return max_c + np.log(w_sum) / beta


def smooth_grad_func(x, n_full, beta):
    h = build_sym_h(x)
    dx = 2.0 / n_full
    corrs = np.correlate(h, 1.0 - h, mode='full') * dx
    max_c = corrs.max()
    shifted = corrs - max_c
    w = np.exp(np.clip(beta * shifted, -50.0, 50.0))
    w_sum = w.sum()
    if w_sum < 1e-30:
        return np.zeros(len(x))
    w /= w_sum
    gq = np.zeros(len(x))
    for k in range(len(corrs)):
        if w[k] < 1e-14:
            continue
        g_h = corr_grad_h(h, k)
        gq += w[k] * dx * (g_h[:len(x)] - g_h[n_full - 1:n_full - 1 - len(x):-1])
    return gq


def smooth_max_optimize(q, n_full, beta_list, maxiter, time_limit):
    m = n_full // 2
    best_score = eval_score(build_sym_h(q))
    best_q = q.copy()
    for beta in beta_list:
        if time_limit and time.time() > time_limit:
            break
        try:
            res = minimize(smooth_obj_func, q, args=(n_full, beta),
                          jac=smooth_grad_func,
                          method='L-BFGS-B', bounds=[(0.0, 1.0)] * m,
                          options={'maxiter': maxiter, 'ftol': 1e-14})
            q_new = np.clip(res.x, 0.0, 1.0)
            h_new = build_sym_h(q_new)
            c_new = eval_score(h_new)
            if c_new < best_score:
                best_score = c_new
                best_q = q_new.copy()
            q = q_new
        except Exception:
            pass
    return best_q, best_score, build_sym_h(best_q)


def full_lp_solve(h, n, active_k, radius):
    dx = 2.0 / n
    corrs = np.correlate(h, 1.0 - h, mode='full') * dx
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_ub = []
    b_ub = []
    for k in active_k:
        g = dx * corr_grad_h(h, k)
        row = np.zeros(n + 1)
        row[:n] = g
        row[-1] = -1.0
        A_ub.append(row)
        b_ub.append(-corrs[k])
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds = []
    for i in range(n):
        lo = max(h[i] - radius, 0.0)
        hi = min(h[i] + radius, 1.0)
        bounds.append((lo, hi))
    bounds.append((-np.inf, np.inf))
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None


def full_lp_refine(h, n, max_rounds=20, time_limit=None):
    best_score = eval_score(h)
    for rnd in range(max_rounds):
        if time_limit and time.time() > time_limit:
            break
        corrs = corr_all_scaled(h)
        improved = False
        for margin in [0.03, 0.06, 0.12]:
            active = select_active(corrs, margin=margin, pad=3, max_k=50)
            for radius in [0.15, 0.08, 0.04]:
                h_try = full_lp_solve(h, n, active, radius)
                if h_try is None:
                    continue
                h_try = project_feasible(h_try, n / 2.0)
                c_try = eval_score(h_try)
                if c_try < best_score - 1e-13:
                    best_score = c_try
                    h = h_try
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
    return h, best_score


def make_structure_start(n, rng):
    m = n // 2
    freqs = [1, 2, 3, 4]
    amps = [0.15, 0.1, 0.06, 0.03]
    phases = [rng.uniform(0, 2 * np.pi) for _ in range(len(freqs))]
    x = np.linspace(0, 1, m)
    q = np.zeros(m)
    for i, (f, a, p) in enumerate(zip(freqs, amps, phases)):
        q += a * np.sin(2 * np.pi * f * x + p)
    q = 0.5 + q
    q = np.clip(q, 0.0, 1.0)
    return q


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def rem():
        return budget_s - (time.time() - t0)

    best_h = None
    best_c5 = float('inf')

    try:
        h_parent = np.array(initial_h_values, dtype=float).copy().reshape(-1)
    except (NameError, AttributeError):
        h_parent = None

    if h_parent is not None and len(h_parent) >= 20:
        n_p = len(h_parent)
        if n_p % 2 != 0:
            h_parent = transfer_to_n(h_parent, n_p + 1)
        h_parent = np.clip(h_parent, 0.0, 1.0)
        h_parent = project_feasible(h_parent, len(h_parent) / 2.0)
        best_h = h_parent.copy()
        best_c5 = eval_score(best_h)
    else:
        m = 63
        x = np.linspace(0, 1, m)
        q = np.clip(0.5 + 0.25 * np.sin(2 * np.pi * x) + 0.08 * np.sin(4 * np.pi * x), 0, 1)
        best_h = build_sym_h(q)
        best_c5 = eval_score(best_h)

    n = len(best_h)

    # Phase 1: LP refinement (main new component)
    if rem() > 250:
        q_cur = sym_from_h(best_h)
        tl = time.time() + min(300, rem() * 0.3)
        q_new, c_new, h_new = lp_refine_sym(q_cur, n, max_rounds=50, time_limit=tl)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new.copy()

    # Phase 2: Coordinate descent
    if rem() > 200:
        q_cur = sym_from_h(best_h)
        tl = time.time() + min(200, rem() * 0.25)
        q_new, c_new, h_new = coord_descent_sym(q_cur, n, max_rounds=25, time_limit=tl)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new.copy()

    # Phase 3: Smooth max (polish)
    if rem() > 150:
        q_cur = sym_from_h(best_h)
        tl = time.time() + min(120, rem() * 0.2)
        q_new, c_new, h_new = smooth_max_optimize(q_cur, n, [80, 150, 300], 300, tl)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new.copy()

    # Phase 4: Iterative refinement cycles
    for cyc in range(4):
        if rem() < 120:
            break
        q_cur = sym_from_h(best_h)
        if rem() > 80:
            tl = time.time() + min(60, rem() * 0.3)
            q_new, c_new, h_new = lp_refine_sym(q_cur, n, max_rounds=20, time_limit=tl)
            if c_new < best_c5:
                best_c5 = c_new
                best_h = h_new.copy()
        if rem() > 60:
            q_cur = sym_from_h(best_h)
            tl = time.time() + min(40, rem() * 0.3)
            q_new, c_new, h_new = coord_descent_sym(q_cur, n, max_rounds=15, time_limit=tl)
            if c_new < best_c5:
                best_c5 = c_new
                best_h = h_new.copy()
        if rem() > 40:
            q_cur = sym_from_h(best_h)
            tl = time.time() + min(30, rem() * 0.3)
            q_new, c_new, h_new = smooth_max_optimize(q_cur, n, [200, 500], 200, tl)
            if c_new < best_c5:
                best_c5 = c_new
                best_h = h_new.copy()

    # Phase 5: Resolution exploration
    n_base = len(best_h)
    deltas = list(range(-4, 9, 1)) + [10, 12, -8, -12]
    for delta in deltas:
        if rem() < 70:
            break
        n_try = n_base + delta
        if n_try < 40 or n_try > 280 or n_try % 2 != 0:
            continue
        h_try = transfer_to_n(best_h, n_try)
        c_try = eval_score(h_try)
        if c_try < best_c5 + 0.001:
            q_try = sym_from_h(h_try)
            if rem() > 50:
                tl = time.time() + min(35, rem() * 0.4)
                q_new, c_new, h_new = lp_refine_sym(q_try, n_try, max_rounds=15, time_limit=tl)
                if c_new < best_c5:
                    best_c5 = c_new
                    best_h = h_new.copy()
            if rem() > 25:
                q_try2 = sym_from_h(best_h if c_new < c_try else h_try)
                n_use = len(best_h if c_new < c_try else h_try)
                tl = time.time() + min(20, rem() * 0.4)
                q_new2, c_new2, h_new2 = coord_descent_sym(q_try2, n_use, max_rounds=10, time_limit=tl)
                if c_new2 < best_c5:
                    best_c5 = c_new2
                    best_h = h_new2.copy()

    # Phase 6: Try full (non-symmetric) optimization
    if rem() > 80:
        h_full = best_h.copy()
        tl = time.time() + min(60, rem() * 0.4)
        h_new, c_new = full_lp_refine(h_full, len(h_full), max_rounds=15, time_limit=tl)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new.copy()

    # Phase 7: Diverse starts at same resolution
    n_div = len(best_h)
    for trial in range(5):
        if rem() < 60:
            break
        q_div = make_structure_start(n_div, rng)
        h_div = build_sym_h(q_div)
        c_div = eval_score(h_div)
        if c_div < best_c5 + 0.01:
            if rem() > 40:
                tl = time.time() + min(30, rem() * 0.5)
                q_new, c_new, h_new = lp_refine_sym(q_div, n_div, max_rounds=10, time_limit=tl)
                if c_new < best_c5:
                    best_c5 = c_new
                    best_h = h_new.copy()
                if rem() > 20:
                    tl2 = time.time() + min(15, rem() * 0.5)
                    q_new2, c_new2, h_new2 = coord_descent_sym(sym_from_h(h_new), n_div, max_rounds=8, time_limit=tl2)
                    if c_new2 < best_c5:
                        best_c5 = c_new2
                        best_h = h_new2.copy()

    # Phase 8: Final polishing
    if rem() > 25 and best_h is not None:
        n_fin = len(best_h)
        if n_fin % 2 == 0:
            q_fin = sym_from_h(best_h)
            if rem() > 15:
                tl = time.time() + min(10, rem() * 0.5)
                q_new, c_new, h_new = coord_descent_sym(q_fin, n_fin, max_rounds=15, time_limit=tl)
                if c_new < best_c5:
                    best_c5 = c_new
                    best_h = h_new.copy()
            if rem() > 10:
                q_fin2 = sym_from_h(best_h)
                tl2 = time.time() + min(8, rem() * 0.8)
                q_new2, c_new2, h_new2 = lp_refine_sym(q_fin2, n_fin, max_rounds=8, time_limit=tl2)
                if c_new2 < best_c5:
                    best_c5 = c_new2
                    best_h = h_new2.copy()
        else:
            if rem() > 15:
                tl = time.time() + min(10, rem() * 0.5)
                h_new, c_new = full_lp_refine(best_h, n_fin, max_rounds=10, time_limit=tl)
                if c_new < best_c5:
                    best_c5 = c_new
                    best_h = h_new.copy()

    # Final validation
    if best_h is None:
        m_fb = 63
        x_fb = np.linspace(0, 1, m_fb)
        q_fb = np.clip(0.5 + 0.25 * np.sin(2 * np.pi * x_fb), 0, 1)
        best_h = build_sym_h(q_fb)

    best_h = np.asarray(best_h, dtype=float).reshape(-1)
    n_out = len(best_h)
    best_h = project_feasible(best_h, n_out / 2.0)
    best_c5 = eval_score(best_h)

    return best_h, best_c5, n_out
