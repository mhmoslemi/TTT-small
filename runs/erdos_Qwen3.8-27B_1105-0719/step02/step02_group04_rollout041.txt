import numpy as np
from scipy.optimize import linprog
import time

def dx_val(n):
    return 2.0 / n

def corr_C(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def eval_c5(h):
    return float(np.max(corr_C(h)))

def is_feasible(h, tol=1e-9):
    if h.ndim != 1 or len(h) < 4:
        return False
    if not np.all(np.isfinite(h)):
        return False
    if np.any(h < -tol) or np.any(h > 1.0 + tol):
        return False
    n = len(h)
    if abs(h.sum() - n / 2.0) > tol * n:
        return False
    return True

def repair_feasible(h):
    """Water-filling to project h to [0,1]^n with sum = n/2."""
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    target = n / 2.0
    # Binary search for water level
    lo, hi = -1.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        h_try = np.clip(h - mid, 0.0, 1.0)
        if h_try.sum() > target:
            lo = mid
        else:
            hi = mid
    h_out = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    # Fix residual
    diff = target - h_out.sum()
    if abs(diff) > 1e-14:
        mask = (h_out > 1e-12) & (h_out < 1.0 - 1e-12)
        if mask.any():
            h_out[mask] += diff / mask.sum()
    return np.clip(h_out, 0.0, 1.0)

def build_sym(q, n):
    m = n // 2
    h = np.empty(n)
    h[:m] = q
    h[m:] = 1.0 - q[::-1]
    return h

def q_from_h(h, n):
    m = n // 2
    return 0.5 * (h[:m] + 1.0 - h[n-1:m-1:-1])

def grad_row(h, j, n):
    """Compute gradient of C[j] w.r.t. h, scaled by dx."""
    dx = 2.0 / n
    d = j - (n - 1)
    g = np.zeros(n)
    # First part: h[i] factor, i from max(0,d) to min(n-1, n-1+d)
    lo1 = max(0, d)
    hi1 = min(n - 1, n - 1 + d)
    if lo1 <= hi1:

    # Second part: (1-h[i-d]) factor, k=i-d from max(0,-d) to min(n-1-d, n-1)
    lo2 = max(0, -d)
    hi2 = min(n - 1 - d, n - 1)
    if lo2 <= hi2:
        g[lo2:hi2+1] += -h[lo2+d:hi2+1+d]
    return g * dx

def grad_row_sym(q, h, j, n):
    """Gradient of C[j] w.r.t. q (symmetric parameterization)."""
    m = n // 2
    gh = grad_row(h, j, n)
    gq = gh[:m].copy()
    gq -= gh[n-1:m-1:-1]
    return gq

def select_active(C, c_max, margin_frac=0.01, pad=2):
    """Select active constraint indices."""
    n = len(C)
    thresh = c_max - max(1e-10, margin_frac * c_max)
    idx = np.where(C >= thresh)[0]
    # Pad

    for a in idx:
        for delta in range(-pad, pad+1):
            k = a + delta
            if 0 <= k < n:
                expanded.add(k)
    return np.array(sorted(expanded), dtype=int)

def solve_slp_q(q, n, active_idx, C_vals, R, rho, max_iter_lp=1):
    """Solve one SLP iteration in q-space. Returns delta or None."""
    m = n // 2
    h = build_sym(q, n)
    
    # Build gradient rows for active constraints
    rows = []
    rhs = []
    for j in active_idx:
        gq = grad_row_sym(q, h, j, n)
        rows.append(np.concatenate([gq, np.zeros(m), np.array([-1.0])]))
        rhs.append(-C_vals[j])
    
    if len(rows) == 0:
        return None
    
    A_ub = np.array(rows)
    b_ub = np.array(rhs)
    
    # L1 slack constraints: δ_i - s_i ≤ 0, -δ_i - s_i ≤ 0
    I_m = np.eye(m)
    A_slack = np.zeros((2*m, 2*m + 1))
    A_slack[:m, :m] = I_m
    A_slack[:m, m:2*m] = -I_m
    A_slack[m:2*m, :m] = -I_m
    A_slack[m:2*m, m:2*m] = -I_m
    b_slack = np.zeros(2*m)
    
    A_all = np.vstack([A_ub, A_slack])
    b_all = np.concatenate([rhs, b_slack])
    
    # Objective: t + rho * sum(s_i)
    c_obj = np.zeros(2*m + 1)
    c_obj[m:2*m] = rho
    c_obj[2*m] = 1.0

    for i in range(m):
        lo = max(-q[i], -R)
        hi = min(1.0 - q[i], R)
        bounds.append((lo, hi))
    for i in range(m):
        bounds.append((0.0, R))
    bounds.append((-1.0, C_vals.max() + 1.0))
    
    try:

        if res.status not in [0, 3]:  # 0=optimal, 3=infeasible (relaxed)
            if res.status == 0:
                pass
            else:
                return None
        delta = res.x[:m]
        if not np.all(np.isfinite(delta)):
            return None
        return delta
    except Exception:

def slp_line_search(q, n, delta, C_cur, R):
    """Line search: find best λ that improves C5."""
    lambs = [1.0, 0.8, 0.6, 0.45, 0.3, 0.2, 0.12, 0.07, 0.04]
    best_c = C_cur
    best_lam = 0.0
    best_q = q.copy()
    for lam in lambs:
        q_try = q + lam * delta
        if np.any(q_try < -1e-12) or np.any(q_try > 1.0 + 1e-12):

        c_try = eval_c5(h_try)
        if c_try < best_c - 1e-12:
            best_c = c_try
            best_lam = lam
            best_q = q_try.copy()
    return best_q, best_c, best_lam

import numpy as np

def dx_val(n):
    return 2.0 / n


def corr_C(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)


def eval_c5(h):
    return float(np.max(corr_C(h)))


def repair_feasible(h):
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    target = n * 0.5
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h.copy()
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        h_try = np.clip(h - mid, 0.0, 1.0)
        if h_try.sum() > target:
            lo = mid
        else:
            hi = mid
    h_out = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - h_out.sum()
    if abs(diff) > 1e-13:
        mask = (h_out > 1e-12) & (h_out < 1.0 - 1e-12)
        if mask.any():
            h_out[mask] += diff / mask.sum()
    return np.clip(h_out, 0.0, 1.0)


def build_sym(q, n):
    m = n // 2
    h = np.empty(n)
    h[:m] = q
    h[m:] = 1.0 - q[::-1]
    return h


def q_from_h(h, n):
    m = n // 2
    return 0.5 * (h[:m] + 1.0 - h[n - 1:m - 1:-1])


def grad_row(h, j, n):
    dxv = 2.0 / n
    d = j - (n - 1)
    g = np.zeros(n)
    lo1 = max(0, d)
    hi1 = min(n - 1, n - 1 + d)
    if lo1 <= hi1:
        g[lo1:hi1 + 1] += (1.0 - h[lo1 - d:hi1 + 1 - d])
    lo2 = max(0, -d)
    hi2 = min(n - 1 - d, n - 1)
    if lo2 <= hi2:
        g[lo2:hi2 + 1] += -h[lo2 + d:hi2 + 1 + d]
    return g * dxv


def grad_row_sym(h, j, n):
    m = n // 2
    gh = grad_row(h, j, n)
    gq = gh[:m].copy()
    gq -= gh[n - 1:m - 1:-1]
    return gq


def select_active(C, c_max, margin_frac=0.01, pad=2, max_active=60):
    thresh = c_max - max(1e-10, margin_frac * c_max)
    idx = np.where(C >= thresh)[0]
    if len(idx) > max_active:
        order = idx[np.argsort(C[idx])[::-1]][:max_active]
        idx = order
    expanded = set()
    for a in idx:
        for delta in range(-pad, pad + 1):
            k = int(a) + delta
            if 0 <= k < len(C):
                expanded.add(k)
    return np.array(sorted(expanded), dtype=int)

    m = n // 2
    h = build_sym(q, n)
    rows = []
    rhs = []
    for j in active_idx:
        gq = grad_row_sym(h, j, n)
        rows.append(np.concatenate([gq, np.zeros(m), np.array([-1.0])]))
        rhs.append(-float(C_arr[j]))
    if len(rows) == 0:
        return None
    A_act = np.array(rows)
    b_act = np.array(rhs)

    A_sl[:m, :m] = I_m
    A_sl[:m, m:2 * m] = -I_m
    A_sl[m:2 * m, :m] = -I_m
    A_sl[m:2 * m, m:2 * m] = -I_m
