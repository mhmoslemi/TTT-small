    if d >= 0:

        grad[idx1] += (1.0 - h[idx1 - offset])
        idx2 = np.arange(0, n - offset)
        grad[idx2] += (-h[idx2 + offset])
    else:
        idx1 = np.arange(0, n - offset)
        grad[idx1] += (1.0 - h[idx1 + offset])
        idx2 = np.arange(offset, n)
        grad[idx2] += (-h[idx2 - offset])
    return grad


def full_corr_grad(h, n):
    L = 2 * n - 1
    G = np.zeros((L, n))
    for k in range(L):
        G[k] = corr_grad_k(h, k, n)
    return G


def lse_objective(h, dx, beta):
    corrs = raw_corrs(h) * dx
    return float(logsumexp(beta * corrs) / beta)


def lse_grad(h, dx, beta):
    n = len(h)
    corrs_norm = raw_corrs(h) * dx
    shifted = corrs_norm - corrs_norm.max()
    w = np.exp(beta * shifted)
    ws = w.sum()
    if ws < 1e-30:
        w = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        w /= ws
    grad = np.zeros(n)
    G = full_corr_grad(h, n)
    grad = G.T @ w
    return grad * dx


def slsqp_lse(h0, dx, target_sum, beta, maxiter):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum,
             'jac': lambda x: np.ones(n)}]

    def obj(x):
        return lse_objective(x, dx, beta)

    def g(x):
        return lse_grad(x, dx, beta)

    try:
        res = minimize(obj, h0, jac=g, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        h_out = project_feasible(res.x, target_sum)
    except Exception:
        h_out = project_feasible(h0, target_sum)
    return h_out, c5_exact(h_out, dx)


def build_epi_problem(h_anchor, target_c5, rho, n):
    dx = 2.0 / n
    target_raw = target_c5 * n / 2.0
    anchor = h_anchor.copy()

    class Epi:
        def __init__(s, anchor, target_raw, rho, n):
            s.anchor = anchor
            s.target_raw = target_raw
            s.rho = rho
            s.n = n

        def obj(s, y):
            h = y[:s.n]
            sv = y[s.n]
            dev = h - s.anchor
            return sv + s.rho * np.sum(dev * dev) / s.n

        def obj_jac(s, y):
            h = y[:s.n]
            dev = h - s.anchor
            j = np.zeros(s.n + 1)
            j[:s.n] = 2.0 * s.rho * dev / s.n
            j[s.n] = 1.0
            return j

        def ineq(s, y):
            h = y[:s.n]
            sv = y[s.n]
            c = raw_corrs(h)
            return sv + s.target_raw - c

        def ineq_jac(s, y):

            n = s.n
            L = 2 * n - 1
            J = np.zeros((L, n + 1))
            G = full_corr_grad(h, n)
            J[:, :n] = G
            J[:, n] = 1.0
            return J

        def eq(s, y):

        def eq_jac(s, y):
            j = np.zeros(s.n + 1)
            j[:s.n] = 1.0
            return j

    return Epi(anchor, target_raw, rho, n), target_raw

        m2 = n // 2
        h[:m2] = v[:m2]
        h[m2 + 1:] = 1.0 - v[:m2][::-1]
    return h


def make_binary_anti_pal(n, rng):
    m = n // 2
    v = rng.rand(m)
    v = np.where(v > 0.5, 1.0, 0.0)
    if n % 2 == 0:
        n_ones = m // 2
        if v.sum() != n_ones:
            idx = np.where(v == (1 if v.sum() > n_ones else 0))[0]
            need = int(abs(v.sum() - n_ones))
            v[idx[:need]] = 1.0 - v[idx[:need]]
    return make_anti_palindromic(n, v)


def subgradient_polish(h0, dx, target_sum, max_iter=20000, top_m=5):
    n = len(h0)
    h = h0.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = c5_exact(h, dx)
    alpha = 0.1
    gamma = 0.8
    no_improve = 0

    for it in range(max_iter):
        corrs = raw_corrs(h)
        max_c = corrs.max()
        c5_val = max_c * dx

        if c5_val < best_c5 - 1e-16:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 5000:
            break

        top_idx = np.argsort(corrs)[-top_m:]
        grad = np.zeros(n)
        for k in top_idx:
            grad += corr_grad_k(h, k, n)
        grad /= len(top_idx)
        grad *= dx

        v = gamma * v - alpha * grad
        h_new = project_feasible(h + v, target_sum)
        new_c5 = c5_exact(h_new, dx)

        if new_c5 < c5_val - 1e-17:
            h = h_new
            alpha = min(alpha * 1.005, 0.3)
        else:
            alpha *= 0.93
            v *= 0.3

        if alpha < 1e-14:
            break

    return best_h, best_c5


def subgradient_single_lag(h0, dx, target_sum, max_iter=30000):
    n = len(h0)
    h = h0.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = c5_exact(h, dx)
    alpha = 0.05
    gamma = 0.85
    no_improve = 0

    for it in range(max_iter):
        corrs = raw_corrs(h)
        max_k = corrs.argmax()
        c5_val = corrs[max_k] * dx

        if c5_val < best_c5 - 1e-16:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 8000:
            break

        grad = corr_grad_k(h, max_k, n) * dx
        v = gamma * v - alpha * grad
        h_new = project_feasible(h + v, target_sum)
        new_c5 = c5_exact(h_new, dx)

        if new_c5 < c5_val - 1e-17:
            h = h_new
            alpha = min(alpha * 1.003, 0.2)
        else:
            alpha *= 0.92
            v *= 0.4

        if alpha < 1e-15:
            break

    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def remaining():
        return budget_s - (time.time() - t0)

    h0 = np.array(initial_h_values, dtype=float).ravel()
    n = len(h0)
    dx = 2.0 / n
    target_sum = n / 2.0
    h0 = project_feasible(h0, target_sum)
    best_h = h0.copy()
    best_c5 = c5_exact(h0, dx)

    # Phase 1: SLSQP LSE with moderate betas
    for beta in [50, 200, 800, 3000]:
        if remaining() < 30:
            break
        maxit = 500 if beta < 1000 else 700
        h_opt, c5v = slsqp_lse(best_h, dx, target_sum, beta, maxit)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h_opt

    # Phase 2: Epigraph with top-k constraints
    if remaining() > 100:
        deltas = [0.0005, 0.0003, 0.00015, 0.00008, 0.00004, 0.00002]
        for d in deltas:
            if remaining() < 40:
                break
            target_c5 = best_c5 - d
            if target_c5 < 0.3:
                continue
            for topk in [30, 60]:
                if remaining() < 35:
                    break
                rho = 0.5
                mx = 400
                h_e, c5e = solve_epigraph_topk(best_h, target_c5, rho, n, dx, target_sum, mx, topk)
                if c5e < best_c5:
                    best_c5 = c5e
                    best_h = h_e
                if remaining() < 35:
                    break

    # Phase 3: Full epigraph (all constraints) with smaller target steps
    if remaining() > 80:
        for d in [0.0003, 0.0001, 0.00005, 0.00002]:
            if remaining() < 45:
                break
            target_c5 = best_c5 - d
            if target_c5 < 0.3:
                continue
            rho = 0.3
            mx = 500
            h_e, c5e = solve_epigraph(best_h, target_c5, rho, n, dx, target_sum, mx)
            if c5e < best_c5:
                best_c5 = c5e
                best_h = h_e

    # Phase 4: Subgradient single-lag (aggressive)
    if remaining() > 60:
        h_sg, c5_sg = subgradient_single_lag(best_h, dx, target_sum, 25000)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg

    # Phase 5: Multi-n exploration
    n_base = len(best_h)
    n_candidates = set()
    for nn in [n_base - 10, n_base - 5, n_base, n_base + 5, n_base + 10,
               n_base + 20, n_base + 40, n_base + 60]:
        if 40 <= nn <= 250:
            n_candidates.add(nn)
    n_candidates = sorted(n_candidates)

    for n_try in n_candidates:
        if remaining() < 50:
            break
        if n_try < 30:
            continue
        dx_t = 2.0 / n_try
        ts_t = n_try / 2.0
        x_old = (np.arange(n_base) + 0.5) * (2.0 / n_base)
        x_new = (np.arange(n_try) + 0.5) * (2.0 / n_try)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, ts_t)

        h_opt, c5v = slsqp_lse(h_interp, dx_t, ts_t, 500, 400)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h_opt
            if remaining() > 40:
                h_sg, c5_sg = subgradient_single_lag(best_h, dx_t, ts_t, 10000)
                if c5_sg < best_c5:
                    best_c5 = c5_sg
                    best_h = h_sg

    # Phase 6: Anti-palindromic exploration
    if remaining() > 60:
        n_ap = best_h.shape[0]
        if n_ap % 2 == 1:
            n_ap += 1
        m = n_ap // 2
        dx_ap = 2.0 / n_ap
        ts_ap = n_ap / 2.0

        for trial in range(8):
            if remaining() < 20:
                break
            if trial == 0:
                v = np.full(m, 0.5)
                v = v + rng.normal(0, 0.3, m)
                v = np.clip(v, 0, 1)
            elif trial == 1:
                v = np.where(np.arange(m) % 2 == 0, 0.0, 1.0)
            elif trial == 2:
                v = np.where((np.arange(m) % 4) < 2, 1.0, 0.0)
            else:
                v = rng.rand(m)
                v = np.where(v > 0.5, 1.0, 0.0)
            h_ap = make_anti_palindromic(n_ap, v)
            h_ap = project_feasible(h_ap, ts_ap)
            h_opt, c5v = slsqp_lse(h_ap, dx_ap, ts_ap, 200, 300)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = h_opt
                if remaining() > 30:
                    h_sg, c5_sg = subgradient_single_lag(best_h, dx_ap, ts_ap, 8000)
                    if c5_sg < best_c5:
                        best_c5 = c5_sg
                        best_h = h_sg

    # Phase 7: Perturbation + epigraph
    if remaining() > 50:
        for trial in range(10):
            if remaining() < 20:
                break
            mag = 0.01 + 0.03 * (trial / 10.0)
            h_p = best_h + rng.normal(0, mag, len(best_h))
            h_p = project_feasible(h_p, target_sum)
            target_c5 = best_c5 - 0.0001
            h_e, c5e = solve_epigraph_topk(h_p, target_c5, 0.5, n, dx, target_sum, 350, 40)
            if c5e < best_c5:
                best_c5 = c5e
                best_h = h_e

    # Phase 8: Final high-precision polish
    n_b = len(best_h)
    dx_b = 2.0 / n_b
    ts_b = n_b / 2.0

    if remaining() > 20:
        for beta in [5000, 15000]:
            if remaining() < 15:
                break
            h_f, c5f = slsqp_lse(best_h, dx_b, ts_b, beta, 600)
            if c5f < best_c5:
                best_c5 = c5f
                best_h = h_f

    if remaining() > 15:
        h_sg, c5_sg = subgradient_single_lag(best_h, dx_b, ts_b, 15000)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg

    if remaining() > 12:
        target_c5 = best_c5 - 0.00003
        h_e, c5e = solve_epigraph_topk(best_h, target_c5, 0.2, n_b, dx_b, ts_b, 500, 50)
        if c5e < best_c5:
            best_c5 = c5e
            best_h = h_e

    # Final exact evaluation
    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_c5 = c5_exact(best_h, 2.0 / len(best_h))

    return best_h, best_c5, len(best_h)
