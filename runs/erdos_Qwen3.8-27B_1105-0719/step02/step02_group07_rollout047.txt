def binary_local_search(h, n, max_iter=10000, rng=None):
    """Local search on binary configurations via swaps."""
    h = h.copy()
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    best_c = exact_score(h) * n / 2.0  # work with corr_max directly

        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        # Try swap
        h[i], h[j] = 0, 1
        c = corr_max(h)
        if c < best_c:
