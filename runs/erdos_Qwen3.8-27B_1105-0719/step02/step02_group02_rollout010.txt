import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time


def project_feasible(x, target_sum):
    n = len(x)
    lo = -float(n)
    hi = float(n)
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if diff > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h


def compute_c5(h, dx):
    corrs = np.correlate(h, 1.0 - h, mode='full')
    return float(corrs.max() * dx), corrs


def make_binary_from_continuous(h, target_ones, rng):
    n = len(h)
    h_sorted = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[h_sorted[:target_ones]] = 1.0
    return b


def binary_ils(h_start, n, max_iter, rng, time_limit=None, t0=None):
    """Iterated local search on binary vectors with exactly n/2 ones."""
    target_ones = n // 2
    h = make_binary_from_continuous(h_start, target_ones, rng).copy()
    _, corrs = compute_c5(h, 2.0/n)
    best_c5 = float(corrs.max())  # raw max
    best_h = h.copy()
    
    no_improve = 0
    perturb_size = 4
    total_iter = 0
    
    for it in range(max_iter):
        if time_limit is not None and t0 is not None:
            if time.time() - t0 > time_limit:
                break
        total_iter += 1
        
        # Local search: try swaps, accept improving
        improved = True
        while improved:
            improved = False
            # Try random swaps
            for _ in range(n * 5):
                i = rng.randint(n)
                j = rng.randint(n)
                if h[i] == 1.0 and h[j] == 0.0:
                    h[i] = 0.0
                    h[j] = 1.0
                    corrs_new = np.correlate(h, 1.0 - h, mode='full')

class SmoothProblem:
    def __init__(self, dx, target_sum, beta):
        self.dx = dx
        self.target_sum = target_sum
        self.beta = beta
        self.n = 0  # set later
    
    def set_n(self, n):
        self.n = n
    
    def objective(self, x):
        corrs = np.correlate(x, 1.0 - x, mode='full')
        return logsumexp(self.beta * corrs) / self.beta * self.dx
    
    def gradient(self, x):
        corrs = np.correlate(x, 1.0 - x, mode='full')
        shifted = corrs - corrs.max()
        w = np.exp(self.beta * shifted)
        w /= w.sum()
        # Compute gradient

        grad = np.zeros(n)

            if w[k] < 1e-14:

                    grad[idx2] -= w[k] * x[idx2 - d2]

    def eq_constraint(self, x):
        return x.sum() - self.target_sum
    
    def eq_jacobian(self, x):

        h_out = project_feasible(h0_full, target_sum)
    return h_out


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def remaining():
        return budget_s - (time.time() - t0)

    best_h = None
    best_c5 = float('inf')

    # Initialize from parent
    try:
        h0 = np.array(initial_h_values, dtype=float).ravel()
    except NameError:
        h0 = make_binary(np.random.RandomState(0).rand(126), 63, np.random.RandomState(0))

    n = len(h0)
    dx = 2.0 / n
    target_sum = n / 2.0
    h0 = project_feasible(h0, target_sum)
    best_h = h0.copy()
    best_c5 = c5_score(h0, dx)[0]

    # Phase 1: Quick smooth refinement to get a good starting point
    if remaining() > 100:
        for beta in [50, 200, 800]:
            if remaining() < 30:
                break
            h_opt = slsqp_smooth(best_h, dx, target_sum, beta, 400)
            c5_val = c5_score(h_opt, dx)[0]
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h_opt

    # Phase 2: Binary iterated local search (main new component)
    if remaining() > 200:
        bin_limit = min(250.0, remaining() - 100)
        h_bin, c5_bin = binary_ils(best_h, n, rng, bin_limit, t0)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()
            # Smooth refinement from best binary
            if remaining() > 60:
                h_smooth = project_feasible(h_bin, target_sum)
                for beta in [100, 500, 2000]:
                    if remaining() < 20:
                        break
                    h_opt = slsqp_smooth(h_smooth, dx, target_sum, beta, 400)
                    c5_val = c5_score(h_opt, dx)[0]
                    if c5_val < best_c5:
                        best_c5 = c5_val
                        best_h = h_opt

    # Phase 3: Subgradient with active set
    if remaining() > 100:
        h_sg, c5_sg = subgrad_active(best_h, dx, target_sum, 40000, top_m=3)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg

    # Phase 4: Try pair-complement structure
    if remaining() > 120:
        # Start from current best projected to pair-complement
        m = n // 2
        v_init = best_h[:m].copy()
        # Also try: average of h[:m] and 1-h[m:][::-1] to get closer to PC structure
        v_pc = 0.5 * (v_init + 1.0 - best_h[m:][::-1])
        v_pc = np.clip(v_pc, 0.0, 1.0)
        h_pc = pair_complement(v_pc)
        h_pc = project_feasible(h_pc, target_sum)
        c5_pc = c5_score(h_pc, dx)[0]

        # Also try random binary PC
        h_bin_pc = pair_complement_start(n, rng)
        c5_bin_pc = c5_score(h_bin_pc, dx)[0]

        candidates_pc = [h_pc, h_bin_pc]
        for hc in candidates_pc:
            if remaining() < 30:
                break
            for beta in [100, 500]:
                if remaining() < 20:
                    break
                h_opt = slsqp_pair_comp(hc, dx, target_sum, beta, 350)
                c5_val = c5_score(h_opt, dx)[0]
                if c5_val < best_c5:
                    best_c5 = c5_val
                    best_h = h_opt

    # Phase 5: Multiple SLSQP polish passes with increasing beta
    if remaining() > 80:
        for beta in [500, 2000, 5000, 10000]:
            if remaining() < 25:
                break
            h_opt = slsqp_smooth(best_h, dx, target_sum, beta, 600)
            c5_val = c5_score(h_opt, dx)[0]
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h_opt

    # Phase 6: Subgradient from multiple top-m settings
    if remaining() > 80:
        for tm in [2, 4, 5]:
            if remaining() < 30:
                break
            h_sg, c5_sg = subgrad_active(best_h, dx, target_sum, 30000, top_m=tm)
            if c5_sg < best_c5:
                best_c5 = c5_sg
                best_h = h_sg
                h_opt = slsqp_smooth(best_h, dx, target_sum, 2000, 500)
                c5_val = c5_score(h_opt, dx)[0]
                if c5_val < best_c5:
                    best_c5 = c5_val
                    best_h = h_opt

    # Phase 7: Binary ILS at different sizes, then interpolate back
    if remaining() > 150:
        for n_try in [n - 2, n, n + 2, n + 4]:
            if remaining() < 60:
                break
            if n_try < 30 or n_try > 200:
                continue
            dx_t = 2.0 / n_try
            ts_t = n_try / 2.0
            x_old = (np.arange(len(best_h)) + 0.5) * dx
            x_new = (np.arange(n_try) + 0.5) * (2.0 / n_try)
            h_interp = np.interp(x_new, x_old, best_h)
            h_interp = project_feasible(h_interp, ts_t)

            # Binary ILS at this size
            bin_lim = min(40.0, remaining() - 30)
            if bin_lim > 5:
                h_b, c5_b = binary_ils(h_interp, n_try, rng, bin_lim, time.time())
                if c5_b + (0.380934 - best_c5) * 0.1 < best_c5 * 1.001:
                    h_interp = project_feasible(h_b, ts_t)

            # Smooth refine
            h_opt = slsqp_smooth(h_interp, dx_t, ts_t, 500, 400)
            c5_val = c5_score(h_opt, dx_t)[0]
            if c5_val < best_c5:
                # Interpolate back to original n if needed
                if n_try != n:
                    x_old2 = (np.arange(n_try) + 0.5) * dx_t
                    x_new2 = (np.arange(n) + 0.5) * dx
                    h_back = np.interp(x_new2, x_old2, h_opt)
                    h_back = project_feasible(h_back, target_sum)
                    c5_back = c5_score(h_back, dx)[0]
                    if c5_back < best_c5:
                        best_c5 = c5_back
                        best_h = h_back
                else:
                    best_c5 = c5_val
                    best_h = h_opt
                h_opt2 = slsqp_smooth(best_h, dx, target_sum, 2000, 500)
                c5_v2 = c5_score(h_opt2, dx)[0]
                if c5_v2 < best_c5:
                    best_c5 = c5_v2
                    best_h = h_opt2

    # Phase 8: Perturbation + re-optimization
    for trial in range(15):
        if remaining() < 15:
            break
        mag = 0.01 + 0.03 * (trial / 15.0)
        h_pert = best_h + rng.normal(0, mag, n)
        h_pert = project_feasible(h_pert, target_sum)
        h_opt = slsqp_smooth(h_pert, dx, target_sum, 400, 350)
        c5_val = c5_score(h_opt, dx)[0]
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_opt
            if remaining() > 25:
                h_opt2 = slsqp_smooth(h_opt, dx, target_sum, 3000, 500)
                c5_v2 = c5_score(h_opt2, dx)[0]
                if c5_v2 < best_c5:
                    best_c5 = c5_v2
                    best_h = h_opt2

    # Phase 9: Final intensive subgradient + SLSQP
    if remaining() > 30:
        h_sg, c5_sg = subgrad_active(best_h, dx, target_sum, 20000, top_m=2)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg

    if remaining() > 20:
        for beta in [5000, 15000]:
            if remaining() < 12:
                break
            h_opt = slsqp_smooth(best_h, dx, target_sum, beta, 700)
            c5_val = c5_score(h_opt, dx)[0]
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h_opt

    # Final projection and verification
    best_h = project_feasible(best_h, target_sum)
    final_c5 = c5_score(best_h, dx)[0]
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, n
