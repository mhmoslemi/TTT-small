def find_min_of_lines(corrs, coeffs, lo, hi):
    """Find min of max_i (corrs[i] + coeffs[i] * x) over x in [lo, hi].
    
    Uses the fact that the max of linear functions is a convex PL function.
    """
    # The upper envelope of lines y_i(x) = corrs[i] + coeffs[i] * x
    # Sort by slope
    order = np.argsort(coeffs)
    sorted_coeffs = coeffs[order]
    sorted_corrs = corrs[order]
    
    # Build upper envelope using a stack
    # Each element on the stack is (slope, intercept, x_start)
    # where x_start is the x-value from which this line is on the upper envelope
    stack = []
    for i in range(len(order)):
        m_i = sorted_coeffs[i]
        b_i = sorted_corrs[i]
        x_start = lo
        
        while stack:
            # Check if the new line makes the top of the stack obsolete
            m_top = stack[-1][0]
            b_top = stack[-1][1]
            x_top = stack[-1][2]
            
            if m_i == m_top:
                # Parallel lines: keep the one with higher intercept
                if b_i > b_top:
                    stack.pop()
                    continue
                else:
                    break
            else:
                # Intersection x-coordinate
                x_int = (b_top - b_i) / (m_i - m_top)
                if x_int <= x_top:
                    stack.pop()
                    continue
                else:
                    x_start = x_int
                    break
        
        stack.append((m_i, b_i, x_start))
    
    # Now find the minimum of the upper envelope on [lo, hi]
    # The envelope is piecewise linear and convex
    # The minimum is where the slope changes from negative to positive
    # or at a boundary
    
    if not stack:
        return corrs.max(), 0.0
    
    # Find the vertex where slope >= 0 first
    best_x = lo
    best_val = np.inf
    
    for i in range(len(stack)):
        m_i, b_i, x_start_i = stack[i]
        x_end_i = hi
        if i + 1 < len(stack):
            x_end_i = stack[i + 1][2]
        
        # On [x_start_i, x_end_i], the value is m_i * x + b_i
        # Since we're on the upper envelope and it's convex, the function is
        # decreasing then increasing. The minimum is at the point where slope = 0
        # or at the boundaries.
        
        # Check the left endpoint of this segment
        if x_start_i < lo:
            x_check = lo
        else:
            x_check = x_start_i
        
        if x_check <= hi and x_check >= lo:
            val = m_i * x_check + b_i
            if val < best_val:
                best_val = val
                best_x = x_check
        
        # If slope is negative, keep going left (but we're going left to right)
        # If slope is positive, the minimum is at x_start_i of this segment
        # If slope is 0, any point in this segment is optimal
        if m_i >= 0:
            # The minimum on this segment is at x_start_i (or lo if x_start_i < lo)
            x_min = max(x_start_i, lo)
            if x_min <= hi:
                val = m_i * x_min + b_i
                if val < best_val:
                    best_val = val
                    best_x = x_min
            break
    
    # Also check hi
    # The last line on the envelope
    m_last, b_last, _ = stack[-1]
    val = m_last * hi + b_last
    if val < best_val:
        best_val = val
        best_x = hi
    
    return best_val, best_x

    best_c5 = exact_score(h_parent)
    best_n = n_parent

    def try_update(h_new, c5_new, n_new):
        nonlocal best_h, best_c5, best_n
        if c5_new < best_c5 - 1e-10:
            best_h = h_new.copy()
            best_c5 = c5_new
            best_n = n_new

    # Phase 1: Coordinate descent in q-space on parent grid
    if remaining() > 30:
        q_init = q_from_h(h_parent, n_parent)
        q_opt, c5_opt = coord_descend_q(q_init, n_parent, max_passes=300,
                                         grid_points=200, time_limit=remaining() - 10, t_start=t_start)
        h_opt = build_h_from_q(q_opt, n_parent)
        h_opt = project_feasible(h_opt, n_parent / 2.0)
        c5_exact = exact_score(h_opt)
        try_update(h_opt, c5_exact, n_parent)

    # Phase 2: Try multiple smaller grids
    for n_try in [80, 76, 72, 68, 64, 60, 56, 52, 48]:
        if remaining() < 40:
            break
        h_interp = interpolate_h(best_h, len(best_h), n_try)
        h_interp = project_feasible(h_interp, n_try / 2.0)
        h_as_try, q_try = make_antisymmetric(h_interp, n_try)
        q_opt_s, c5_s = coord_descend_q(q_try, n_try, max_passes=300,
                                         grid_points=200, time_limit=remaining() - 15, t_start=t_start)
        h_opt_s = build_h_from_q(q_opt_s, n_try)
        h_opt_s = project_feasible(h_opt_s, n_try / 2.0)
        c5_exact_s = exact_score(h_opt_s)
        try_update(h_opt_s, c5_exact_s, n_try)

    # Phase 3: Multi-restart with perturbations on the best grid
    n_best = best_n
    if n_best % 2 == 0:
        for restart in range(8):
            if remaining() < 30:
                break
            q_base = q_from_h(best_h, n_best)
            # Random perturbation
            pert = rng.uniform(-0.1, 0.1, size=n_best // 2)
            q_pert = np.clip(q_base + pert, 0, 1)
            q_opt_r, c5_r = coord_descend_q(q_pert, n_best, max_passes=200,
                                             grid_points=200, time_limit=remaining() - 15, t_start=t_start)
            h_opt_r = build_h_from_q(q_opt_r, n_best)
            h_opt_r = project_feasible(h_opt_r, n_best / 2.0)
            c5_exact_r = exact_score(h_opt_r)
            try_update(h_opt_r, c5_exact_r, n_best)

    # Phase 4: Full-space coordinate descent (no anti-symmetric constraint)
    if remaining() > 50:
        h_full = best_h.copy()
        h_opt_f, c5_f = coord_descend_full(h_full, len(h_full), max_passes=150,
                                            grid_points=150, time_limit=remaining() - 15, t_start=t_start)
        h_opt_f = project_feasible(h_opt_f, len(h_opt_f) / 2.0)
        c5_exact_f = exact_score(h_opt_f)
        try_update(h_opt_f, c5_exact_f, len(h_opt_f))

    # Phase 5: Try a few more grid sizes around the best
    if remaining() > 40:
        for n_try in [best_n - 4, best_n - 2, best_n + 2, best_n + 4]:
            if remaining() < 30 or n_try < 40 or n_try > 200:
                continue
            if n_try % 2 != 0:
                n_try += 1
            h_interp = interpolate_h(best_h, len(best_h), n_try)
            h_interp = project_feasible(h_interp, n_try / 2.0)
            h_as_try, q_try = make_antisymmetric(h_interp, n_try)
            q_opt_s, c5_s = coord_descend_q(q_try, n_try, max_passes=200,
                                             grid_points=200, time_limit=remaining() - 15, t_start=t_start)
            h_opt_s = build_h_from_q(q_opt_s, n_try)
            h_opt_s = project_feasible(h_opt_s, n_try / 2.0)
            c5_exact_s = exact_score(h_opt_s)
            try_update(h_opt_s, c5_exact_s, n_try)

    # Phase 6: Final polish - one more coordinate descent pass on best
    if remaining() > 20 and best_n % 2 == 0:
        q_final = q_from_h(best_h, best_n)
        q_opt_p, c5_p = coord_descend_q(q_final, best_n, max_passes=200,
                                         grid_points=300, time_limit=remaining() - 5, t_start=t_start)
        h_opt_p = build_h_from_q(q_opt_p, best_n)
        h_opt_p = project_feasible(h_opt_p, best_n / 2.0)
        c5_exact_p = exact_score(h_opt_p)
        try_update(h_opt_p, c5_exact_p, best_n)

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    n_out = len(best_h)
    best_h = project_feasible(best_h, n_out / 2.0)
    final_score = exact_score(best_h)
    if final_score < best_c5:
        best_c5 = final_score

    return best_h, best_c5, n_out
