import numpy as np
from scipy.optimize import linprog
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx] + 1e-15)
            h[idx] += add
            remaining -= add
            if remaining <= 1e-14:
                break
    else:
        order = np.argsort(-h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx] + 1e-15)
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-14:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def row_gradient(h, d, n, dx):
    g = np.zeros(n)
    if d == 0:
        g[:] = dx * (1.0 - 2.0 * h)
    elif d > 0:
        g[:n - d] -= dx * h[d:]
        g[d:] += dx * (1.0 - h[:n - d])
    else:
        e = -d
        g[:n - e] += dx * (1.0 - h[e:])
        g[e:] -= dx * h[:n - e]
    return g

def fourier_smooth(grad, n, n_keep):
    if n_keep >= n or n_keep < 2:
        return grad
    G = np.fft.rfft(grad)
    G[n_keep:] = 0.0
    return np.fft.irfft(G, n)

def subgradient_descent(h, max_iters=400, K=None, use_fourier=True, n_keep_frac=0.1, seed=42):
    n = len(h)
    dx = 2.0 / n
    center = n - 1
    if K is None:
        K = max(6, n // 8)
    h = h.copy()
    best_c = exact_score(h)
    best_h = h.copy()
    velocity = np.zeros(n)
    momentum = 0.35
    alpha = 1.0
    no_improve = 0
    n_keep = max(3, int(n * n_keep_frac))

    for it in range(max_iters):
        profile = compute_profile(h)
        c = np.max(profile)
        if c < best_c - 1e-13:
            best_c = c
            best_h = h.copy()
            no_improve = 0
            alpha = min(alpha * 1.08, 3.0)
        else:
            no_improve += 1
            alpha *= 0.88
        if no_improve > 80 or alpha < 1e-5:
            break
        top_idx = np.argsort(profile)[-K:]
        w = profile[top_idx] - profile[top_idx].min() + 0.002
        w = w ** 2
        w /= w.sum()
        grad = np.zeros(n)
        for wi, m in zip(w, top_idx):
            d = m - center
            grad += wi * row_gradient(h, d, n, dx)
        grad -= grad.mean()
        if use_fourier:
            grad = fourier_smooth(grad, n, n_keep)
            grad -= grad.mean()
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad /= gnorm
        velocity = momentum * velocity + grad
        vnorm = np.linalg.norm(velocity)
        if vnorm > 1e-14:
            velocity /= vnorm
        improved = False
        for a in [alpha, alpha * 0.5, alpha * 0.25, alpha * 0.1]:
            h_try = repair_feasible(h - a * velocity)
            s = exact_score(h_try)
            if s < best_c - 1e-13:
                h = h_try
                best_c = s
                best_h = h.copy()
                improved = True
                break
        if not improved:
            for a in [alpha * 0.05, alpha * 0.02]:
                h_try = repair_feasible(h - a * velocity)
                s = exact_score(h_try)
                if s < c:
                    h = h_try
                    c_new = exact_score(h)
                    if c_new < best_c:
                        best_c = c_new
                        best_h = h.copy()
                    break
    return best_h, best_c

def lp_trust_region(h, rho, n_active=25):
    n = len(h)
    dx = 2.0 / n
    center = n - 1
    profile = compute_profile(h)
    T = np.max(profile)
    thresh = T - max(0.004, 0.06 * (T - np.median(profile)))
    active = np.where(profile >= thresh)[0]
    if len(active) > n_active:
        active = active[np.argsort(profile[active])[-n_active:]]
    if len(active) == 0:
        active = np.array([np.argmax(profile)])
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_ub = np.zeros((len(active) + 1, n + 1))
    b_ub = np.zeros(len(active) + 1)
    for i, m in enumerate(active):
        d = m - center
        gm = row_gradient(h, d, n, dx)
        A_ub[i, :n] = gm
        A_ub[i, -1] = -1.0
        b_ub[i] = gm @ h - profile[m]
    A_ub[-1, -1] = -1.0
    b_ub[-1] = 0.0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    lb = np.maximum(0.0, h - rho)
    ub = np.minimum(1.0, h + rho)
    bounds = list(zip(lb, ub))
    bounds.append((0.0, T + 0.5))
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0 and res.x is not None:
            return repair_feasible(res.x[:n])
    except Exception:
        pass
    return None

def lp_polish(h, max_iters=40, rho_init=0.06):
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    best_h = h.copy()
    rho = rho_init
    for it in range(max_iters):
        z = lp_trust_region(h, rho, n_active=30)
        if z is None:
            rho *= 0.6
            if rho < 1e-5:
                break
            continue
        found = False
        for beta in [1.0, 0.75, 0.5, 0.3, 0.15]:
            h_try = repair_feasible((1.0 - beta) * h + beta * z)
            s = exact_score(h_try)
            if s < best_c - 1e-12:
                h = h_try
                best_c = s
                best_h = h.copy()
                rho = min(rho * 1.4, 0.15)
                found = True
                break
            elif s < exact_score(h) and beta < 0.2:
                h = h_try
                c_now = exact_score(h)
                if c_now < best_c:
                    best_c = c_now
                    best_h = h.copy()
        if not found:
            rho *= 0.65
            if rho < 1e-5:
                break
    return best_h, best_c

def coordinate_descent(h, n_sweeps=5, n_samples=7):
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    best_h = h.copy()
    for sweep in range(n_sweeps):
        for i in range(n):
            old = h[i]
            candidates = [old]
            for f in [0.8, 0.6, 0.4, 0.2, -0.8, -0.6, -0.4, -0.2, -1.0, 1.0]:
                candidates.append(old + f * 0.05)
            for b in [0.0, 0.5, 1.0]:
                candidates.append(b)
            best_val = old
            best_s = best_c
            for cv in candidates:
                if cv < -0.01 or cv > 1.01:
                    continue
                h[i] = np.clip(cv, 0.0, 1.0)
                h = repair_feasible(h)
                s = exact_score(h)
                if s < best_s - 1e-13:
                    best_s = s
                    best_val = h.copy()
            h = best_val.copy()
            if best_s < best_c:
                best_c = best_s
                best_h = h.copy()
    return best_h, best_c

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

def symmetrize(h):
    return (h + h[::-1]) / 2.0

def perturb(h, rng, eps):
    z = rng.randn(len(h))
    z -= z.mean()
    z /= np.linalg.norm(z)
    return repair_feasible(h + eps * z)
