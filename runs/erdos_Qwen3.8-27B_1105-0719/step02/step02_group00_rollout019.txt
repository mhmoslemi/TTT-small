import numpy as np
from scipy.optimize import linprog
import time

def exact_score(h):
    n = len(h)
    if n < 2:
        return 1.0
    dx = 2.0 / n
    prof = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(prof))

def repair_feasible(vec):
    vec = np.asarray(vec, dtype=np.float64).ravel()
    n = len(vec)
    if n == 0:
        return np.array([0.5])
    h = np.clip(vec, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        cap = 1.0 - h
        order = np.argsort(-cap, kind="mergesort")
        rem = diff
        for i in range(n):
            idx = order[i]
            add = min(rem, cap[idx] + 1e-15)
            if add > 0:
                h[idx] += add
            rem -= add
            if rem <= 1e-13:
                break
    else:
        order = np.argsort(h, kind="mergesort")
        rem = -diff
        for i in range(n):
            idx = order[i]
            sub = min(rem, h[idx] + 1e-15)
            if sub > 0:
                h[idx] -= sub
            rem -= sub
            if rem <= 1e-13:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-14:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
    h = np.clip(h, 0.0, 1.0)
    return h

def compute_profile_and_grad(h):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    P = np.empty(M)
    G = np.zeros((M, n))
    for r in range(M):
        d = (n - 1) - r
        if d >= 0:
            ln = n - d
            P[r] = dx * np.dot(h[d:n], 1.0 - h[:ln])
            G[r, :ln] += -dx * h[d:n]
            G[r, d:n] += dx * (1.0 - h[:ln])
        else:
            e = -d
            ln = n - e
            P[r] = dx * np.dot(h[e:n], 1.0 - h[:ln])
            G[r, e:n] += dx * (1.0 - h[:ln])
            G[r, :ln] += -dx * h[e:n]
    return P, G

def select_active(P, c, beam_frac=0.35, max_active=60):
    spread = max(1e-6, c - np.median(P))
    beam = max(0.003, beam_frac * spread)
    thresh = c - beam
    active = np.where(P >= thresh)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(P))])
    elif len(active) > max_active:
        idx_in = np.argsort(P[active])[-max_active:]
        active = active[idx_in]
    return active

def solve_direction_lp(G, active):
    n = G.shape[1]
    k = len(active)
    nv = 2 * n + 1
    c_obj = np.zeros(nv)
    c_obj[-1] = -1.0
    # Inequality: G[active, i] * d_i + alpha <= 0
    A_ub = np.zeros((k, nv))
    for i, m in enumerate(active):
        A_ub[i, :n] = G[m, :]
        A_ub[i, n:2*n] = -G[m, :]
        A_ub[i, -1] = 1.0
    b_ub = np.zeros(k)
    # Equality: sum(u-v)=0, sum(u+v)=1
    A_eq = np.zeros((2, nv))
    A_eq[0, :n] = 1.0
    A_eq[0, n:2*n] = -1.0
    A_eq[1, :n] = 1.0
    A_eq[1, n:2*n] = 1.0
    b_eq = np.array([0.0, 1.0])
    bounds = [(0.0, None)] * (2 * n) + [(0.0, 10.0)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            alpha = max(0.0, -res.fun)
            u = res.x[:n]
            v = res.x[n:2*n]
            d = u - v
            dnorm = np.sum(np.abs(d))
            if dnorm > 1e-10:
                d /= dnorm
                d -= d.mean()
                return d, alpha
    except Exception:
        pass
    return None, 0.0

def do_line_search(h, d, c_cur, max_steps=12):
    n = len(h)
    best_h = None
    best_c = c_cur
    d_abs = np.abs(d)
    max_step_pos = np.inf
    max_step_neg = np.inf
    mask_pos = d > 1e-12
    mask_neg = d < -1e-12
    if np.any(mask_pos):
        max_step_pos = np.min((1.0 - h[mask_pos]) / d_abs[mask_pos])
    if np.any(mask_neg):
        max_step_neg = np.min(h[mask_neg] / d_abs[mask_neg])
    max_step = min(max_step_pos, max_step_neg, 1.0)
    max_step = max(max_step, 1e-8)
    steps = np.array([0.95, 0.7, 0.5, 0.35, 0.22, 0.12, 0.06, 0.03, 0.015, 0.007, 0.003, 0.001]) * max_step
    for s in steps:
        h_try = h + s * d
        h_try = repair_feasible(h_try)
        c_try = exact_score(h_try)
        if c_try < best_c - 1e-14:
            best_c = c_try
            best_h = h_try.copy()
    return best_h, best_c

def polish_lp(h, time_budget, rng, beam_start=0.35, label=""):
    n = len(h)
    h = repair_feasible(h.copy())
    best_c = exact_score(h)
    cur_c = best_c
    stall = 0
    max_stall = 15
    beam = beam_start
    t0 = time.perf_counter()
    
    while time.perf_counter() - t0 < time_budget and stall < max_stall:
        if time.perf_counter() - t0 > time_budget:
            break
        P, G = compute_profile_and_grad(h)
        c_int = np.max(P)
        active = select_active(P, c_int, beam_frac=beam)
        d, alpha = solve_direction_lp(G, active)
        if d is None or alpha < 1e-12:
            stall += 1
            beam *= 1.4
            if beam > 1.0:
                beam = 1.0
            continue
        h_new, c_new = do_line_search(h, d, cur_c)
        if h_new is not None and c_new < best_c - 1e-13:
            h = h_new
            best_c = c_new
            cur_c = c_new
            stall = 0
            beam = max(0.05, beam * 0.95)
        else:
            cur_c = c_new if c_new < cur_c else cur_c
            if cur_c < best_c:
                best_c = cur_c
                h = repair_feasible(h)
            stall += 1
            beam *= 1.3
            if beam > 1.0:
                beam = 1.0
    return repair_feasible(h), best_c

def interp_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    return repair_feasible(np.clip(h_new, 0.0, 1.0))

def make_binary_seed(n, rng):
    if n % 2 != 0:
        n = n + 1
    ones = n // 2
    h = np.zeros(n)
    idx = rng.choice(n, size=ones, replace=False)
    h[idx] = 1.0
    return repair_feasible(h)

def make_fourier_seed(n, freq, phase, amp, rng):
    x = np.arange(n) / n
    h = 0.5 + amp * np.sin(2 * np.pi * freq * x + phase)
    h = np.clip(h, 0.0, 1.0)
    return repair_feasible(h)

def make_block_seed(n, n_blocks, rng):
    h = np.zeros(n)
    bs = n // n_blocks
    for b in range(n_blocks):
        s, e = b * bs, min((b + 1) * bs, n)
        val = rng.uniform(0.1, 0.9)
        h[s:e] = val
    return repair_feasible(h)
