import numpy as np
from scipy.optimize import linprog
import time

def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = mask.sum()
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h += diff / n
    return np.clip(h, 0.0, 1.0)

def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def c5_of_h(h, dx):
    return float(compute_corrs(h).max() * dx)

def corr_gradients(h, active_ks):
    """Compute gradients of R[k] for k in active_ks.
    Returns array of shape (len(active_ks), n)."""
    n = len(h)
    grads = np.zeros((len(active_ks), n))
    for idx, k in enumerate(active_ks):
        c = k - (n - 1)
        i_lo = max(0, -c)
        i_hi = min(n - 1, n - 1 - c)
        if i_lo > i_hi:
            continue
        i = np.arange(i_lo, i_hi + 1)
        grads[idx, i] += (1.0 - h[i + c])
        if c != 0:
            grads[idx, i + c] -= h[i]
        elif c == 0:
            # When c=0: R = sum h[i]*(1-h[i]), dR/dh[i] = 1-2*h[i]

            # Total: 1-2*h[i] ✓ (the if c!=0 branch handles c=0 case correctly 
            # since i+0 = i, so it adds (1-h[i]) to grads[idx,i] and subtracts h[i] from grads[idx,i])
            pass
    return grads

def lp_descent_direction(h, grads, alpha=0.15):
    """Solve LP to find descent direction.
    min t s.t. grads[k] · d <= t for all k, sum(d) = 0, |d_i| <= alpha * min(h_i, 1-h_i)
    Returns (d, t_opt) or (None, None) if no descent direction.
    """
    n = len(h)
    A_active = len(grads)
    
    # Variables: d_0, ..., d_{n-1}, t
    # Objective: minimize t → c = [0,...,0, 1]
    c_vec = np.zeros(n + 1)
    c_vec[n] = 1.0  # coefficient of t
    
    # Inequality constraints: A_ineq @ x <= b_ineq
    # For each active k: grads[k] · d - t <= 0
    # bounds: -ub_i <= d_i <= ub_i → d_i <= ub_i and -d_i <= ub_i
    
    A_rows = []
    b_rows = []
    
    for k in range(A_active):
        row = np.zeros(n + 1)
        row[:n] = grads[k]
        row[n] = -1.0  # -t
        A_rows.append(row)
        b_rows.append(0.0)
    
    # Upper bounds on d_i
    ub = alpha * np.minimum(h, 1.0 - h)
    # d_i <= ub_i
    for i in range(n):
        row = np.zeros(n + 1)
        row[i] = 1.0
        A_rows.append(row)
        b_rows.append(ub[i])
    # -d_i <= ub_i
    for i in range(n):
        row = np.zeros(n + 1)
        row[i] = -1.0
        A_rows.append(row)
        b_rows.append(ub[i])
    
    A_ineq = np.array(A_rows)
    b_ineq = np.array(b_rows)
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    result = linprog(c_vec, A_ub=A_ineq, b_ub=b_ineq, A_eq=A_eq, b_eq=b_eq,
                     bounds=[(None, None)] * n + [(None, None)],
                     method='highs')
    
    if result.success:
        d = result.x[:n]
        t_opt = result.x[n]
        if t_opt < -1e-10:
            return d, t_opt
    return None, None

def line_search(h, d, max_c5, dx, n_steps=10):
    """Find best step size for h + beta*d."""
    best_c5 = max_c5
    best_beta = 0.0
    for i in range(1, n_steps + 1):
        beta = i / n_steps
        h_try = h + beta * d
        h_try = project_feasible(h_try, h.sum())
        c5_try = c5_of_h(h_try, dx)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_beta = beta
    return best_beta, best_c5, (h + best_beta * d) if best_beta > 0 else h

def binary_swap_delta(h_corrs, h, a, b, n):
    """Compute delta in correlations when flipping h[a]:1→0 and h[b]:0→1.
    h_corrs is the current correlation vector.
    Returns the delta vector (length 2n-1)."""
    delta = np.zeros(2 * n - 1)
    # offset(k) = k - (n-1)
    # For each k (offset c = k-(n-1)):
    # ΔR[k] = -(1-h[a+c]) + h[a-c] + (1-h[b+c]) - h[b-c]
    # (each term only if index is in [0, n-1])
    
    for c in range(-(n-1), n):
        val = 0.0
        # Term 1: -(1-h[a+c]) if 0 <= a+c <= n-1
        if 0 <= a + c <= n - 1:
            val -= (1.0 - h[a + c])
        # Term 2: +h[a-c] if 0 <= a-c <= n-1
        if 0 <= a - c <= n - 1:
            val += h[a - c]
        # Term 3: +(1-h[b+c]) if 0 <= b+c <= n-1
        if 0 <= b + c <= n - 1:
            val += (1.0 - h[b + c])
        # Term 4: -h[b-c] if 0 <= b-c <= n-1
        if 0 <= b - c <= n - 1:
            val -= h[b - c]
        if val != 0.0:
            delta[c + (n - 1)] = val
    
    return delta

def binary_greedy_optimize(h, dx, max_sweeps=50, time_limit=None, t0=None):
    """Greedy binary optimization: try all swaps, apply best."""
    n = len(h)
    target = int(round(n / 2))
    
    h_bin = (h >= 0.5).astype(float)
    # Ensure exactly target ones
    if h_bin.sum() != target:
        sorted_idx = np.argsort(-h)
        h_bin = np.zeros(n)
        h_bin[sorted_idx[:target]] = 1.0
    
    corrs = compute_corrs(h_bin)
    best_c5 = corrs.max() * dx
    best_h = h_bin.copy()
    
    ones = np.where(h_bin == 1.0)[0]
    zeros = np.where(h_bin == 0.0)[0]
    
    for sweep in range(max_sweeps):
        if time_limit is not None and t0 is not None:
            if time.time() - t0 > time_limit:
                break
        
        best_swap = None
        best_new_max = corrs.max()
        
        for i in ones:
            for j in zeros:
                delta = binary_swap_delta(corrs, h_bin, i, j, n)
                new_max = (corrs + delta).max()
                if new_max < best_new_max - 1e-14:
                    best_new_max = new_max
                    best_swap = (i, j)
        
        if best_swap is None:
            break
        
        i, j = best_swap
        h_bin[i], h_bin[j] = 0.0, 1.0
        corrs = compute_corrs(h_bin)
        new_c5 = corrs.max() * dx
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_bin.copy()
            ones = np.where(h_bin == 1.0)[0]
            zeros = np.where(h_bin == 0.0)[0]
        else:
            # Recompute ones/zeros from the modified h_bin
            ones = np.where(h_bin == 1.0)[0]
            zeros = np.where(h_bin == 0.0)[0]
    
    return best_h, best_c5

def random_binary_start(n, rng):
    """Create a random binary vector with n/2 ones."""
    target = n // 2
    idx = rng.choice(n, size=target, replace=False)
    h = np.zeros(n)
    h[idx] = 1.0
    return h

def golden_ratio_start(n, rng):
    """Create binary vector using golden ratio distribution."""
    target = n // 2
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    h = np.zeros(n)
    for k in range(target):
        pos = int((k * phi + rng.uniform(0, 1)) * n) % n
        h[pos] = 1.0
    return h

import numpy as np
from scipy.optimize import linprog
import time


def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = mask.sum()
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h += diff / n
    return np.clip(h, 0.0, 1.0)


def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def c5_of_h(h, dx):
    return float(compute_corrs(h).max() * dx)


def corr_gradients(h, ks):
    n = len(h)
    grads = np.zeros((len(ks), n))
    for idx, k in enumerate(ks):
        c = k - (n - 1)
        i_lo = max(0, -c)
        i_hi = min(n - 1, n - 1 - c)
        if i_lo > i_hi:
            continue
        i = np.arange(i_lo, i_hi + 1)
        grads[idx, i] += (1.0 - h[i + c])
        if c != 0:
            grads[idx, i + c] -= h[i]
        else:
            grads[idx, i] -= h[i]
    return grads


def lp_direction(h, grads, alpha=0.12):
    n = len(h)
    A_active = len(grads)
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    A_rows = []
    b_rows = []
    for kk in range(A_active):
        row = np.zeros(n + 1)
        row[:n] = grads[kk]
        row[n] = -1.0
        A_rows.append(row)
        b_rows.append(0.0)
    for i in range(n):
        row = np.zeros(n + 1)
        row[i] = 1.0
        A_rows.append(row)
        b_rows.append(alpha)
    for i in range(n):
        row = np.zeros(n + 1)
        row[i] = -1.0
        A_rows.append(row)
        b_rows.append(alpha)
    A_ub = np.array(A_rows)
    b_ub = np.array(b_rows)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=[(None, None)] * (n + 1), method='highs',
                      options={'presolve': True})
    except Exception:
        return None, 0.0
    if res.success and res.x[n] < -1e-9:
        d = res.x[:n]
        if np.abs(d).max() > 1e-12:
            return d, float(res.x[n])
    return None, 0.0


def line_search_c5(h, d, target_sum, dx, n_evals=8):
    best_c5 = c5_of_h(h, dx)
    best_h = h.copy()
    for i in range(1, n_evals + 1):
        beta = i / n_evals
        h_try = h + beta * d
        h_try = project_feasible(h_try, target_sum)
        c5_try = c5_of_h(h_try, dx)
        if c5_try < best_c5 - 1e-14:
            best_c5 = c5_try
            best_h = h_try
    return best_h, best_c5

def make_golden_binary(n, rng, noise=0.0):
    target = n // 2
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    pos = []
    for k in range(target):
        p = int((k * phi + rng.uniform(0, 1) + noise) * n) % n
        pos.append(p)
    h = np.zeros(n)
    h[np.array(pos)] = 1.0
    return h


def lp_optimize_loop(h, dx, target_sum, max_iter=500, t0=None, budget_s=1100):
