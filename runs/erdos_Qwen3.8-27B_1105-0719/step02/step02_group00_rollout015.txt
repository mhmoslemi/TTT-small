3. Normalize α

import numpy as np

def _repair(h, n_target=None):
    """Repair h to satisfy 0<=h<=1 and sum(h)=n/2."""
    n = len(h)
    h = np.clip(h.astype(np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            # Need to add mass; add to positions with most room
            order = np.argsort(1.0 - h)  # ascending room
            for i in range(n):
                idx = n - 1 - i  # most room first
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
                if abs(diff) < 1e-14:
                    break
        else:
            # Need to remove mass; remove from positions with most value
            order = np.argsort(h)  # ascending value
            for i in range(n):
                idx = n - 1 - i  # most value first
                rem = min(-diff, h[idx])
                h[idx] -= rem
                diff += rem
                if abs(diff) < 1e-14:
                    break
    # Final fix
    h = np.clip(h, 0.0, 1.0)
    resid = target - h.sum()
    if abs(resid) > 1e-10:
        if resid > 0:
            idx = np.argmin(h)
        else:
            idx = np.argmax(h)
        h[idx] += resid
        h = np.clip(h, 0.0, 1.0)
    return h

def _score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))

def _profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def _grad_lag(h, d, n, dx):
    """Gradient of P_d * dx with respect to h."""
    g = np.zeros(n)
    if d == 0:
        g[:] = dx * (1.0 - 2.0 * h)
    elif d > 0:
        # P_d = sum_{i=d}^{n-1} h[i] * (1-h[i-d])
        # dP_d/dh[i] for i in [d, n-1]: (1 - h[i-d])
        # dP_d/dh[i] for i in [0, n-1-d]: -h[i+d]
        g[d:] += dx * (1.0 - h[:n-d])
        g[:n-d] -= dx * h[d:]
    else:
        d_pos = -d
        # P_d = sum_{i=0}^{n-1-d_pos} h[i] * (1-h[i+d_pos])
        # dP_d/dh[i] for i in [0, n-1-d_pos]: (1 - h[i+d_pos])
        # dP_d/dh[i] for i in [d_pos, n-1]: -h[i-d_pos]
        g[:n-d_pos] += dx * (1.0 - h[d_pos:])
        g[d_pos:] -= dx * h[:n-d_pos]
    return g

def _grad_all(h):
    """Compute gradients for all lags. Returns (2n-1, n) matrix."""
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    G = np.zeros((M, n))
    for j in range(M):
        d = j - (n - 1)
        G[j] = _grad_lag(h, d, n, dx)
    return G

def _softmax_grad(h, beta):
    """Compute the soft-max gradient direction."""
    n = len(h)
    dx = 2.0 / n
    prof = _profile(h)
    # Softmax weights
    shifted = beta * prof - beta * prof.max()
    w = np.exp(shifted)
    w /= w.sum()
    # Weighted gradient
    G = _grad_all(h)
    grad = G.T @ w
    # Project out the mean (to maintain sum constraint)
    grad -= grad.mean()
    return grad, prof.max(), w

def _active_set_grad(h, K=10, epsilon=0.005):
    """Compute gradient using top K active lags with minimal-norm combination."""
    n = len(h)
    prof = _profile(h)
    c = prof.max()
    # Select active lags
    active = np.where(prof >= c - epsilon)[0]
    if len(active) > K:
        top_idx = np.argsort(prof[active])[-K:]
        active = active[top_idx]
    
    G = _grad_all(h)
    G_act = G[active, :]  # shape (|A|, n)
    
    # Find minimal-norm combination: minimize ||G_act^T @ alpha||^2 s.t. sum(alpha)=1, alpha>=0
    # This is equivalent to: find the point in conv{g_i} closest to origin
    # For small |A|, solve the QP
    m = len(active)
    if m == 1:
        alpha = np.ones(1)
    elif m <= 20:
        # QP: min alpha^T (G_act G_act^T) alpha s.t. 1^T alpha = 1, alpha >= 0
        A_q = G_act @ G_act.T  # m x m
        try:
            from scipy.optimize import minimize as _min
            def qp_obj(a):
                return a @ A_q @ a
            def qp_grad(a):
                return 2 * A_q @ a
            x0 = np.ones(m) / m
            bounds_q = [(0, 1)] * m
            cons_q = [{'type': 'eq', 'fun': lambda a: a.sum() - 1, 'jac': lambda a: np.ones(m)}]
            res_q = _min(qp_obj, x0, jac=qp_grad, bounds=bounds_q, constraints=cons_q, method='SLSQP', options={'maxiter': 100, 'ftol': 1e-10})
            if res_q.success and res_q.fun < qp_obj(x0) + 1e-10:
                alpha = np.maximum(res_q.x, 0)
                alpha /= alpha.sum()
            else:
                alpha = np.ones(m) / m
        except:
            alpha = np.ones(m) / m
    else:
        # Use softmax weighting for large active sets
        eps = prof[active].max() - prof[active]
        alpha = np.exp(eps / 0.01)
        alpha /= alpha.sum()
    
    grad = G_act.T @ alpha
    grad -= grad.mean()
    return grad, c, active

def _line_search(h, direction, current_score, max_iter=50):
    """Backtracking line search along -direction."""
    n = len(h)
    best_h = h.copy()
    best_c = current_score
    alpha = 0.1
    for _ in range(max_iter):
        h_try = _repair(h - alpha * direction)
        c_try = _score(h_try)
        if c_try < best_c:
            best_h = h_try
            best_c = c_try
            alpha *= 1.5
        else:
            alpha *= 0.5
            if alpha < 1e-6:
                break
    return best_h, best_c

def _symmetrize(h):
    return (h + h[::-1]) / 2.0

def _interp_mass(h, new_n):
    """Mass-preserving interpolation."""
    n = len(h)
    if n == new_n:
        return h.copy()
    edges_old = np.linspace(0, 2, n + 1)
    edges_new = np.linspace(0, 2, new_n + 1)
    h_new = np.zeros(new_n)
    for k in range(new_n):
        lo, hi = edges_new[k], edges_new[k+1]
        width = hi - lo
        # Find overlapping old bins
        for j in range(n):
            olo, ohi = edges_old[j], edges_old[j+1]
            overlap = min(hi, ohi) - max(lo, olo)
            if overlap > 0:
                h_new[k] += h[j] * overlap / width
    return _repair(h_new)

def _slsqp_solve(h0, maxiter=300, active_K=20):
    """SLSQP with reduced constraint set."""
    n = len(h0)
    dx = 2.0 / n
    M = 2 * n - 1
    c0 = _score(h0)
    
    def objective(x):
        return x[-1]
    
    def obj_grad(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    
    def eq_cons(x):
        return x[:-1].sum() - n / 2.0
    
    def eq_grad(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g
    
    # Pre-compute active set from initial point
    h_init = x_init = h0
    prof_init = _profile(h0)
    c_init = prof_init.max()
    # Use top K lags as constraints
    top_k = np.argsort(prof_init)[::-1][:active_K]
    
    def ineq_cons(x):
        h = x[:-1]
        t = x[-1]
        prof = _profile(h)
        return t - prof[top_k]
    
    def ineq_grad(x):
        h = x[:-1]
        G = _grad_all(h)
        J = np.zeros((active_K, n + 1))
        J[:, :-1] = -G[top_k, :]
        J[:, -1] = 1.0
        return J
    
    x_start = np.concatenate([h0, [c0 + 1e-3]])
    bounds = [(0, 1)] * n + [(0, c0 + 1)]
    cons = [
        {'type': 'eq', 'fun': eq_cons, 'jac': eq_grad},
        {'type': 'ineq', 'fun': ineq_cons, 'jac': ineq_grad}
    ]
    
    try:
        res = minimize(objective, x_start, jac=obj_grad, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None and np.isfinite(res.x).all():
            h_new = _repair(res.x[:-1])
            c_new = _score(h_new)
            if c_new <= c0 + 1e-8:
                return h_new, c_new
    except Exception:
        pass
    return None, c0

def _generate_seed(n, rng, method='random'):
    """Generate a balanced random seed."""
    if method == 'random':
        h = rng.randn(n)
        h = (h - h.min()) / (h.max() - h.min() + 1e-15)
    elif method == 'symmetric':
        half = n // 2
        h_half = rng.rand(half + 1)
        h = np.concatenate([h_half, h_half[-2:0:-1]])
        h = (h - h.min()) / (h.max() - h.min() + 1e-15)
    elif method == 'golden':
        # Golden ratio low-discrepancy
        phi = (1 + np.sqrt(5)) / 2
        h = np.sin(2 * np.pi * (np.arange(n) / phi))
        h = (h - h.min()) / (h.max() - h.min() + 1e-15)
    elif method == 'alternating':
        h = np.zeros(n)
        h[::2] = 1.0
        if n % 2 == 1:
            h[-1] = 0.0
        # Add noise
        h += 0.3 * rng.randn(n)
    else:
        h = np.full(n, 0.5)
    return _repair(h)

    best_h = h.copy()
    
    for it in range(max_iters):
        # Adaptive beta

        beta = beta_start + (beta_end - beta_start) * frac
        
        if method == 'softmax':
