def grad_corr_k(h, k, n):
    """Gradient of correlation k with respect to h."""
    off = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -off)
    i_hi = min(n - 1, n - 1 - off)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    # h[idx + off] * (1 - h[idx])
    # d/dh[idx+off] = (1 - h[idx])
    # d/dh[idx] = -h[idx+off]
    grad[idx + off] += (1.0 - h[idx])
    grad[idx] += (-h[idx + off])
    return grad

def project_feasible(x, target):
    """Project x onto {0 <= h <= 1, sum(h) = target}"""
    n = len(x)
    # Bisection for the shift
    lo, hi = -2.0, 2.0

        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)

    diff = target - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

import numpy as np
from scipy.optimize import linprog
import time


def score_h(h, dx):
    corrs = np.correlate(h, 1.0 - h, mode='full')
    return corrs.max() * dx, corrs


def grad_corr_k(h, k, n):
    off = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -off)
    i_hi = min(n - 1, n - 1 - off)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx + off] += (1.0 - h[idx])
    grad[idx] += (-h[idx + off])
    return grad


def project_feasible(x, target):
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)


def lp_step(h, dx, target_sum, n_active, alpha):
    n = len(h)
    corrs = np.correlate(h, 1.0 - h, mode='full')
    c_max = corrs.max()
    sorted_idx = np.argsort(corrs)[::-1]
    active = sorted_idx[:n_active]
    m = len(active)
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_rows = []
    b_rows = []
    for k in active:
        g = grad_corr_k(h, k, n)
        row = np.zeros(n + 1)
        row[:n] = dx * g
        row[n] = -1.0
        A_rows.append(row)
        b_rows.append(-dx * corrs[k])
    A_ub = np.array(A_rows)
    b_ub = np.array(b_rows)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for i in range(n):
        bounds.append((-h[i] * alpha, (1.0 - h[i]) * alpha))
    bounds.append((0.0, None))
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
    except Exception:
        return None, c_max * dx
    if res.status != 0:
        return None, c_max * dx
    delta = res.x[:n]
    t_pred = res.x[-1]
    lam_max = 1.0
    for i in range(n):
        if delta[i] < -1e-15:
            lam_max = min(lam_max, h[i] / (-delta[i]))
        elif delta[i] > 1e-15:
            lam_max = min(lam_max, (1.0 - h[i]) / delta[i])
    lam_max = min(lam_max, 1.0)
    if lam_max < 1e-15:
        return None, c_max * dx
    best_c5 = c_max * dx
    best_h = None
    for frac in [1.0, 0.8, 0.6, 0.5, 0.4, 0.3, 0.2, 0.1, 0.05]:
        lam = frac * lam_max
        h_new = h + lam * delta
        h_new = project_feasible(h_new, target_sum)
        c5_new, _ = score_h(h_new, dx)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new
    return best_h, best_c5


def lp_polish(h0, dx, target_sum, time_limit, t0):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = score_h(h, dx)[0]
    alpha = 0.3
    n_active = max(30, int(0.4 * (2 * n - 1)))
    no_improve = 0
    it = 0
    while time.time() - t0 < time_limit:
        corrs = np.correlate(h, 1.0 - h, mode='full')
        c_max = corrs.max()
        current_c5 = c_max * dx
        if current_c5 < best_c5 - 1e-14:
            best_c5 = current_c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 40:
            break
        h_new, c5_new = lp_step(h, dx, target_sum, n_active, alpha)
        it += 1
        if h_new is None:
            alpha *= 0.7
            n_active = min(2 * n - 1, int(n_active * 1.3))
            if alpha < 0.01:
                break
            continue
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
            no_improve = 0
            if alpha < 0.3:
                alpha *= 1.1
        else:
            no_improve += 1
            alpha *= 0.8
    return best_h, best_c5


def subgradient_polish(h0, dx, target_sum, max_iter, t0, time_limit):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = score_h(h, dx)[0]
    beta = 30.0
    alpha_step = 1.0 / np.sqrt(n)
    no_improve = 0
    for it in range(max_iter):
        if time.time() - t0 > time_limit:
            break
        corrs = np.correlate(h, 1.0 - h, mode='full')
        c_max = corrs.max()
        c5_val = c_max * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 5000:
            break
        shifted = corrs - corrs.max()
        w = np.exp(beta * shifted)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(len(corrs)) / len(corrs)
        else:
            w /= w_sum
        grad = np.zeros(n)
        for k in range(len(corrs)):
            if w[k] > 1e-12:
                grad += w[k] * grad_corr_k(h, k, n)
        grad *= dx
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        step = alpha_step / max(gnorm, 1e-6)
        step = min(step, 0.5)
        h_new = h - step * grad
        h_new = project_feasible(h_new, target_sum)
        new_c5, _ = score_h(h_new, dx)
        tries = 0
        while new_c5 > c5_val + 1e-13 and tries < 4:
            step *= 0.5
            h_new = h - step * grad
            h_new = project_feasible(h_new, target_sum)
            new_c5, _ = score_h(h_new, dx)
            tries += 1
        h = h_new
        alpha_step *= (1.0 - 0.2 / (it + 100))
        alpha_step = max(alpha_step, 1e-7)
    return best_h, best_c5


def make_binary_start(n, target, rng):
    h = np.zeros(n)
    idx = rng.permutation(n)
    n_ones = int(target)
    if n_ones > 0:
        h[idx[:n_ones]] = 1.0
    frac = target - n_ones
    if frac > 1e-10 and n_ones < n:
        h[idx[n_ones]] = frac
    return h


def make_periodic_start(n, target, rng):
    period = rng.choice([3, 4, 5, 6, 7, 8, 9])
    fill = max(1, int(target / n * period))
    h = np.zeros(n)
    for i in range(n):
        h[i] = 1.0 if (i % period) < fill else 0.0
    return project_feasible(h, target)


def make_random_start(n, target, rng):
    h = rng.random(n)
    return project_feasible(h, target)


def make_perturbed_start(h_best, n_new, target, rng):
    if len(h_best) == n_new:
        h = h_best + 0.15 * (rng.random(n_new) - 0.5)
    else:
        x_old = np.linspace(0, 2, len(h_best))
        x_new = np.linspace(0, 2, n_new)
        h = np.interp(x_new, x_old, h_best)
        h = h + 0.1 * (rng.random(n_new) - 0.5)
    return project_feasible(h, target)


def remaining_time(t0, budget_s):
    return budget_s - (time.time() - t0)
