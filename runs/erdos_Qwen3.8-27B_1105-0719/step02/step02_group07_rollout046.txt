    best_f = exact_score(make_sym_h(q, n))
    for _ in range(n_passes):
        improved = False
        for i in range(m):
            for delta in [0.05, -0.05, 0.02, -0.02, 0.1, -0.1]:
                q_try = best_q.copy()
                q_try[i] = np.clip(q_try[i] + delta, 0.0, 1.0)
                f = exact_score(make_sym_h(q_try, n))
                if f < best_f - 1e-12:
                    best_f = f
                    best_q = q_try
                    improved = True
                    break
        if not improved:
            break
    return best_q, best_f


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()

    def remaining():
        return budget_s - (time.time() - t0)

    rng = np.random.RandomState(seed)

    try:
        h_parent = np.array(initial_h_values, dtype=float)
    except NameError:
        h_parent = 0.5 * np.ones(126)

    n = len(h_parent)
    if n % 2 != 0:
        h_parent = h_parent[:n - 1]
        n = len(h_parent)

    h_best = project_feasible(h_parent, n / 2.0)
    c5_best = exact_score(h_best)
    m = n // 2
    q_best = make_q_from_h(h_best, n)

    # Phase 1: Fast symmetric gradient descent with increasing beta
    for beta, top_k, lr, n_steps in [(5, 20, 0.8, 3000), (20, 15, 0.6, 3000),
                                      (50, 10, 0.5, 3000), (100, 8, 0.4, 3000),
                                      (200, 6, 0.3, 3000), (500, 5, 0.2, 2000)]:
        if remaining() < 10:
            break
        q_new, _ = gd_sym(q_best, n, top_k, beta, n_steps, lr, 0.85)
        c5_new = exact_score(make_sym_h(q_new, n))
        if c5_new < c5_best - 1e-12:
            c5_best = c5_new
            q_best = q_new
            h_best = make_sym_h(q_best, n)

    # Coordinate refinement
    if remaining() > 30:
        q_best, c5_cr = coordinate_refine_sym(q_best, n, n_passes=5)
        if c5_cr < c5_best - 1e-12:
            c5_best = c5_cr
            h_best = make_sym_h(q_best, n)

    # Phase 2: Basin hopping with diverse perturbations
    n_hops = 80
    for hop in range(n_hops):
        if remaining() < 20:
            break
        scale = 0.05 + 0.45 * rng.rand()
        q_pert = q_best.copy()
        mode = rng.rand()
        if mode < 0.25:
            n_flip = max(1, int(0.05 * m + 0.15 * m * rng.rand()))
            flip_idx = rng.choice(m, size=n_flip, replace=False)
            q_pert[flip_idx] = (q_pert[flip_idx] > 0.5).astype(float)
        elif mode < 0.5:
            q_pert += scale * rng.randn(m)
        elif mode < 0.75:
            bs = max(2, m // 6)
            st = rng.randint(0, m - bs)
            q_pert[st:st + bs] += scale * rng.randn(bs)
        else:
            # Swap perturbation
            i, j = rng.choice(m, 2, replace=False)
            tmp = q_pert[i]
            q_pert[i] = q_pert[j]
            q_pert[j] = tmp
            # Add small noise
            q_pert += 0.05 * rng.randn(m)

        q_pert = np.clip(q_pert, 0.0, 1.0)

        # Multi-stage gradient descent from perturbed point
        q_new, _ = gd_sym(q_pert, n, 15, 10, 2000, 0.7, 0.85)
        q_new, _ = gd_sym(q_new, n, 8, 50, 2000, 0.5, 0.85)
        q_new, _ = gd_sym(q_new, n, 5, 200, 1500, 0.3, 0.85)

        c5_new = exact_score(make_sym_h(q_new, n))
        if c5_new < c5_best - 1e-12:
            c5_best = c5_new
            q_best = q_new
            h_best = make_sym_h(q_best, n)

    # Coordinate refinement after hopping
    if remaining() > 20:
        q_best, c5_cr = coordinate_refine_sym(q_best, n, n_passes=5)
        if c5_cr < c5_best - 1e-12:
            c5_best = c5_cr
            h_best = make_sym_h(q_best, n)

    # Phase 3: Full-space polish (drop symmetry)
    if remaining() > 40:
        h_cur = h_best.copy()
        for beta, top_k, lr, nsteps in [(10, 15, 0.6, 2000), (50, 10, 0.4, 2000),
                                         (200, 6, 0.3, 2000)]:
            if remaining() < 15:
                break
            h_new, c5_new = gd_full(h_cur, n, top_k, beta, nsteps, lr, 0.85)
            if c5_new < c5_best - 1e-12:
                c5_best = c5_new
                h_best = h_new
                h_cur = h_new
                q_best = make_q_from_h(h_new, n)

    # Phase 4: Multi-resolution - try nearby n values
    if remaining() > 60:
        for delta in [2, -2, 4, -4, 6, -6, 8, -8]:
            if remaining() < 30:
                break
            n_try = n + delta
            if n_try < 40 or n_try % 2 != 0:
                continue
            x_old = np.linspace(0, 2, n, endpoint=False)
            x_new = np.linspace(0, 2, n_try, endpoint=False)
            h_interp = np.interp(x_new, x_old, h_best)
            h_interp = project_feasible(h_interp, n_try / 2.0)
            m_try = n_try // 2
            q_try = make_q_from_h(h_interp, n_try)

            q_new, _ = gd_sym(q_try, n_try, 12, 20, 2000, 0.6, 0.85)
            q_new, _ = gd_sym(q_new, n_try, 6, 100, 2000, 0.4, 0.85)
            c5_new = exact_score(make_sym_h(q_new, n_try))
            if c5_new < c5_best - 1e-10:
                c5_best = c5_new
                n = n_try
                m = m_try
                q_best = q_new
                h_best = make_sym_h(q_best, n)

    # Phase 5: Final intensive optimization on best
    if remaining() > 30:
        for beta, top_k, lr, nsteps in [(100, 6, 0.3, 3000), (300, 4, 0.2, 3000),
                                         (1000, 3, 0.15, 2000)]:
            if remaining() < 15:
                break
            q_new, _ = gd_sym(q_best, n, top_k, beta, nsteps, lr, 0.9)
            c5_new = exact_score(make_sym_h(q_new, n))
            if c5_new < c5_best - 1e-13:
                c5_best = c5_new
                q_best = q_new
                h_best = make_sym_h(q_best, n)

    # Final coordinate refinement
    if remaining() > 10:
        q_best, c5_cr = coordinate_refine_sym(q_best, n, n_passes=10)
        if c5_cr < c5_best - 1e-13:
            c5_best = c5_cr
            h_best = make_sym_h(q_best, n)

    # Final full-space check
    if remaining() > 10:
        h_new, c5_new = gd_full(h_best, len(h_best), 5, 200, 2000, 0.3, 0.9)
        if c5_new < c5_best - 1e-13:
            c5_best = c5_new
            h_best = h_new

    # Final validation
    h_best = np.clip(h_best, 0.0, 1.0)
    h_best = project_feasible(h_best, len(h_best) / 2.0)
    c5_best = exact_score(h_best)

    return h_best, c5_best, len(h_best)
