    f1 = (base + slopes).max()

def build_h_from_q(q, n):
    m = n // 2
    if n % 2 == 0:
        return np.concatenate([q, 1.0 - q[::-1]])
    return np.concatenate([q, [0.5], 1.0 - q[::-1]])

def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = 0.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corrs = np.correlate(h, 1.0 - h, mode='full')
    return corrs.max() * dx

def coord_slopes(h, i, n):
    """Compute slope of each correlation wrt q[i] (antisymmetric)."""
    A = i
    B = n - 1 - i
    d = np.arange(-(n-1), n)
    slopes = np.zeros(2*n-1)
    
    mask = (d >= A-(n-1)) & (d <= A)
    if mask.any():
        slopes[mask] += (1.0 - h[A - d[mask]])
    
    mask = (d >= -A) & (d <= n-1-A)
    if mask.any():
        slopes[mask] -= h[A + d[mask]]
    
    mask = (d >= B-(n-1)) & (d <= B)
    if mask.any():
        slopes[mask] -= (1.0 - h[B - d[mask]])
    
    mask = (d >= -B) & (d <= n-1-B)
    if mask.any():
        slopes[mask] += h[B + d[mask]]
    
    return slopes

def minimize_max_lines(base, slopes):
    """Find t in [0,1] minimizing max_d(base[d] + t*slopes[d])."""
    f0 = base.max()
    f1 = (base + slopes).max()
    
    act0 = base >= f0 - 1e-12
    if slopes[act0].min() > 1e-14:
        return 0.0
    act1 = (base + slopes) >= f1 - 1e-12
    if slopes[act1].max() < -1e-14:
        return 1.0
    
    lo, hi = 0.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        vals = base + mid * slopes
        fm = vals.max()
        act = vals >= fm - 1e-12
        if slopes[act].min() > 1e-12:
            hi = mid
        elif slopes[act].max() < -1e-12:
            lo = mid
        else:
            break
    return (lo + hi) * 0.5

def coord_descent(q, n, max_sweeps=200, shuffle=False, rng=None):
    """Cyclic coordinate descent in antisymmetric space."""
    m = n // 2
    h = build_h_from_q(q, n)
    corrs = np.correlate(h, 1.0 - h, mode='full')
    best = corrs.max()
    q_work = q.copy()
    
    order = list(range(m))
    
    for sweep in range(max_sweeps):
        improved = False
        if shuffle and rng is not None:
            rng.shuffle(order)
        for i in order:
            slopes = coord_slopes(h, i, n)
            cur = q_work[i]
            base = corrs - cur * slopes
            t_opt = minimize_max_lines(base, slopes)
            if abs(t_opt - cur) > 1e-10:
                delta = t_opt - cur
                corrs_new = corrs + delta * slopes
                if corrs_new.max() < best - 1e-14:
                    best = corrs_new.max()
                    q_work[i] = t_opt
                    h[i] = t_opt
                    h[n-1-i] = 1.0 - t_opt
                    corrs = corrs_new
                    improved = True
        if not improved:
            break
    
    return q_work, best

        # And dc[d]/dq[i] = dc[d]/dh[i] - dc[d]/dh[n-1-i]
        pass
    # Simpler: compute gradient numerically via the slope structure
    Gq = np.zeros(m)
    for i in range(m):
        s = coord_slopes(h, i, n)
        Gq[i] = (w * s).sum() * dx
    return val, Gq


def smooth_q_solve(q0, n, beta, maxiter=300):
    def obj(x):
        return smooth_q_obj_grad(x, n, beta)[0]

    def grad(x):
        return smooth_q_obj_grad(x, n, beta)[1]

    try:
        res = minimize(obj, q0, jac=grad, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * (n // 2),
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return np.clip(res.x, 0.0, 1.0)
    except Exception:
        return q0.copy()


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)

    def remaining():
        return budget_s - (time.time() - t_start)

    h_parent = np.array(initial_h_values, dtype=float)
    n_parent = len(h_parent)
    n = n_parent if n_parent % 2 == 0 else n_parent - 1
    if n != n_parent:
        x_old = np.linspace(0.0, 2.0, n_parent)
        x_new = np.linspace(0.0, 2.0, n)
        h_init = np.interp(x_new, x_old, h_parent)
    else:
        h_init = h_parent.copy()

    m = n // 2
    q_init = h_init[:m].copy()
    h_init = build_h_from_q(q_init, n)

    best_h = h_init.copy()
    best_c5 = exact_score(h_init)
    best_q = q_init.copy()

    # Phase 1: Binary coordinate descent
    if remaining() > 10:
        q_bin, _ = binary_coord_descent(best_q, n, max_sweeps=500)
        h_bin = build_h_from_q(q_bin, n)
        s = exact_score(h_bin)
        if s < best_c5 - 1e-14:
            best_c5 = s
            best_h = h_bin.copy()
            best_q = q_bin.copy()

    # Phase 2: Fractional coordinate descent
    if remaining() > 30:
        q_frac, _ = coord_descent(best_q, n, max_sweeps=300)
        h_frac = build_h_from_q(q_frac, n)
        s = exact_score(h_frac)
        if s < best_c5 - 1e-14:
            best_c5 = s
            best_h = h_frac.copy()
            best_q = q_frac.copy()

    # Phase 3: Smooth optimization (log-sum-exp)
    if remaining() > 60:
        for beta in [200, 500, 1000, 2000]:
            if remaining() < 30:
                break
            q_smooth = smooth_q_solve(best_q, n, beta, maxiter=400)
            h_sm = build_h_from_q(q_smooth, n)
            s = exact_score(h_sm)
            if s < best_c5 - 1e-14:
                best_c5 = s
                best_h = h_sm.copy()
                best_q = q_smooth.copy()
            # Follow up with coordinate descent
            q_cd, _ = coord_descent(q_smooth, n, max_sweeps=200)
            h_cd = build_h_from_q(q_cd, n)
            s = exact_score(h_cd)
            if s < best_c5 - 1e-14:
                best_c5 = s
                best_h = h_cd.copy()
                best_q = q_cd.copy()

    # Phase 4: Perturbation loop with adaptive magnitude
    n_pert = 300
    for attempt in range(n_pert):
        if remaining() < 20:
            break
        frac = attempt / n_pert
        mag = 0.005 + 0.10 * (1.0 - frac) ** 1.5

        q_pert = best_q + rng.randn(m) * mag
        q_pert = np.clip(q_pert, 0.0, 1.0)

        # Binary descent to snap to good basin
        q_b, _ = binary_coord_descent(q_pert, n, max_sweeps=150)

        # Fractional polish
        q_f, _ = coord_descent(q_b, n, max_sweeps=100, shuffle=True, rng=rng)
        h_f = build_h_from_q(q_f, n)
        s = exact_score(h_f)

        if s < best_c5 - 1e-14:
            best_c5 = s
            best_h = h_f.copy()
            best_q = q_f.copy()

    # Phase 5: Try different resolutions
    for n_try in [80, 96, 104, 112, 120, 128, 136, 144, 152, 160, 176, 192]:
        if remaining() < 30:
            break
        if n_try == n:
            continue
        m_try = n_try // 2
        x_old = np.linspace(0.0, 2.0, len(best_h))
        x_new = np.linspace(0.0, 2.0, n_try)
        h_interp = np.interp(x_new, x_old, best_h)
        q_try = h_interp[:m_try].copy()

        q_b, _ = binary_coord_descent(q_try, n_try, max_sweeps=300)
        q_f, _ = coord_descent(q_b, n_try, max_sweeps=150)
        h_f = build_h_from_q(q_f, n_try)
        s = exact_score(h_f)

        if s < best_c5 - 1e-10:
            best_c5 = s
            best_h = h_f.copy()

        # Add perturbations at this resolution
        for p in range(5):
            if remaining() < 20:
                break
            q_p = q_f + rng.randn(m_try) * 0.03
            q_p = np.clip(q_p, 0.0, 1.0)
            q_fb, _ = binary_coord_descent(q_p, n_try, max_sweeps=150)
            q_ff, _ = coord_descent(q_fb, n_try, max_sweeps=100)
            h_ff = build_h_from_q(q_ff, n_try)
            s = exact_score(h_ff)
            if s < best_c5 - 1e-10:
                best_c5 = s
                best_h = h_ff.copy()

    # Phase 6: Final fine-tuning at best resolution
    n_best = len(best_h)
    if remaining() > 15 and n_best % 2 == 0:
        m_best = n_best // 2
        q_fin = best_h[:m_best].copy()
        for attempt in range(100):
            if remaining() < 10:
                break
            mag = 0.003 + 0.01 * (1.0 - attempt / 100)
            q_p = q_fin + rng.randn(m_best) * mag
            q_p = np.clip(q_p, 0.0, 1.0)
            q_f, _ = coord_descent(q_p, n_best, max_sweeps=150, shuffle=True, rng=rng)
            h_f = build_h_from_q(q_f, n_best)
            s = exact_score(h_f)
            if s < best_c5 - 1e-14:
                best_c5 = s
                best_h = h_f.copy()
                q_fin = q_f.copy()

    # Phase 7: One more binary + fractional pass
    if remaining() > 10 and len(best_h) % 2 == 0:
        m_final = len(best_h) // 2
        q_final = best_h[:m_final].copy()
        q_b, _ = binary_coord_descent(q_final, len(best_h), max_sweeps=300)
        q_f, _ = coord_descent(q_b, len(best_h), max_sweeps=200)
        h_f = build_h_from_q(q_f, len(best_h))
        s = exact_score(h_f)
        if s < best_c5 - 1e-14:
            best_c5 = s
            best_h = h_f.copy()

    # Ensure feasibility
    n_out = len(best_h)
    best_h = np.clip(best_h, 0.0, 1.0)
    target = n_out / 2.0
    best_h = project_feasible(best_h, target)
    final_c5 = exact_score(best_h)

    return best_h, final_c5, n_out
