def binary_swap_delta(h, p, q, n):
    delta = np.zeros(n)
    for k in range(1, n):
        # Find affected m values
        affected = set()
        if p <= n - 1 - k:
            affected.add(p)
        if k <= p:
            affected.add(p - k)
        if q <= n - 1 - k:
            affected.add(q)
        if k <= q:
            affected.add(q - k)
        # Remove invalid m values
        affected = {m for m in affected if 0 <= m <= n - 1 - k}
        if not affected:
            continue
        d = 0.0
        for m in affected:
            # Before: h[m+k]*(1-h[m])
            # After: h_new[m+k]*(1-h_new[m])
            # h_new[p] = 0, h_new[q] = 1, others unchanged
            mk = m + k
            h_mk = h[mk] if mk != p else 0.0
            h_m = h[m] if m != p else 0.0
            after = h_mk * (1.0 - h_m)
            before = h[mk] * (1.0 - h[m])
            d += after - before
        delta[k] = d
    return delta
