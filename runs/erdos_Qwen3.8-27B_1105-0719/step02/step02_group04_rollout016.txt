def project_to_target(x, target_sum):

def grad_c(h, d, n):
    """Gradient of c_d with respect to h, at point h."""
    dx = 2.0 / n
    g = np.zeros(n)
    # c_d = sum_{valid n} h[n+d](1-h[n]) * dx
    # d/i: h[i] appears as h[n+d] when n = i-d, and as h[n] when n = i
    # Term 1: i = n+d => n = i-d, valid if 0<=i-d<n
    #   Contribution to g[i]: (1-h[i-d])*dx
    # Term 2: i = n => n = i, valid if 0<=i+d<n
    #   Contribution to g[i]: -h[i+d]*dx
    
    # Vectorized:
    # Term 1: for each i, if 0 <= i-d < n, g[i] += (1-h[i-d])*dx
    lo1 = max(0, d)
    hi1 = min(n, n + d)
    if lo1 < hi1:
        g[lo1:hi1] += (1.0 - h[lo1-d:hi1-d]) * dx
    
    # Term 2: for each i, if 0 <= i+d < n, g[i] += -h[i+d]*dx
    lo2 = max(0, -d)
    hi2 = min(n, n - d)
    if lo2 < hi2:

    return g

def slp_step(h, c, active_k, n, maxiter=100):
    """Sequential Linear Programming step."""
    dx = 2.0 / n
    target = n / 2.0
    
    # Build the LP: min t
    # s.t. grad_d . h' <= t + b_d for active d
    #      0 <= h' <= 1
    #      sum(h') = target
    #
    # Variables: h'[0..n-1], t (total n+1 variables)
    #
    # For scipy.optimize.linprog:
    # c_obj = [0, ..., 0, 1] (minimize t)
    # A_ub @ x <= b_ub
    # A_eq @ x = b_eq
    
    # active_k are indices into the correlation array (0 to 2n-2)
    # Corresponding shift: d = active_k - (n-1)
    
    m = len(active_k)
    # A_ub: m rows (inequality) + box handling
    # For each active d: grad_d . h' - t <= b_d
    # grad_d = grad_c(h, d, n)
    
    A_rows = []
    b_rows = []
    
    for k in active_k:
        d = k - (n - 1)
        g = grad_c(h, d, n)
        b_d = g @ h - c[k]  # This makes: g . h' - t <= b_d equivalent to c_d(h') <= t at linearization
        # Actually: c_d(h') ≈ c_d(h) + g . (h' - h) = g . h' + c_d(h) - g . h
        # So c_d(h') <= t means: g . h' - t <= t - c_d(h) + g . h... 
        # Wait: g . h' + c_d(h) - g . h <= t
        # g . h' - t <= g . h - c_d(h)
        
        A_row = np.zeros(n + 1)
        A_row[:n] = g
        A_row[n] = -1.0
        A_rows.append(A_row)
        b_rows.append(g @ h - c[k])
    
    A_ub = np.array(A_rows)
    b_ub = np.array(b_rows)
    
    # Equality: sum(h') = target
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    
    # Object
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    # Bounds: 0 <= h'[i] <= 1 for i in 0..n-1, t >= 0
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            h_lp = res.x[:n]
            t_lp = res.x[n]
            return h_lp, t_lp
    except Exception:
        pass
    return None, None
