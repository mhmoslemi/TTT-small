def c5_of_h(h, dx):

def smooth_max_val_and_grad(h, dx, beta):
    corrs = np.correlate(h, 1.0 - h, mode='full')
    m = corrs.max()
    shifted = corrs - m
    w = np.exp(beta * shifted)
    w_sum = w.sum()

    else:

    n = len(h)

    for k in range(2 * n - 1):

def _grad_corr_k(h, k, n):
    # Gradient of corr_k with respect to h
    # corr_k = sum_i h[i+k-shift] * (1-h[i]) where shift = n-1
    # The full correlate: result[k] = sum_i h[i] * (1-h[i - (k-(n-1))])
    # Let offset = k - (n-1)
    # result[k] = sum_i h[i] * (1 - h[i - offset]) for valid i
    # d(result[k])/d(h[j]) = (1 - h[j-offset]) - h[j+offset] (for valid indices)
    offset = k - (n - 1)
    grad = np.zeros(n)
    # h[j] appears as first factor: h[j]*(1-h[j-offset]) → d/dh[j] = (1-h[j-offset])
    # h[j] appears as second factor: h[j+offset]*(1-h[j]) → d/dh[j] = -h[j+offset]
    # First factor valid: 0 <= j < n, 0 <= j-offset < n
    lo1 = max(0, offset)
    hi1 = min(n, n + offset)  # exclusive
    if lo1 < hi1:
        idx = np.arange(lo1, hi1)
        grad[idx] += (1.0 - h[idx - offset])
    # Second factor valid: 0 <= j < n, 0 <= j+offset < n
    lo2 = max(0, -offset)
    hi2 = min(n, n - offset)  # exclusive
    if lo2 < hi2:
        idx = np.arange(lo2, hi2)
        grad[idx] += (-h[idx + offset])
    return grad

import numpy as np
from scipy.optimize import minimize, minimize_scalar
from scipy.special import logsumexp
import time


def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
        diff = target_sum - h.sum()
        if abs(diff) > 1e-10:
            h += diff / n
            h = np.clip(h, 0.0, 1.0)
    return h


def c5_of_h(h, dx):
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)


def _grad_corr_k(h, k, n):
    offset = k - (n - 1)
    grad = np.zeros(n)
    lo1 = max(0, offset)
    hi1 = min(n, n + offset)
    if lo1 < hi1:
        idx = np.arange(lo1, hi1)
        grad[idx] += (1.0 - h[idx - offset])
    lo2 = max(0, -offset)
    hi2 = min(n, n - offset)
    if lo2 < hi2:
        idx = np.arange(lo2, hi2)
        grad[idx] += (-h[idx + offset])
    return grad


def _smooth_obj_and_grad(h, dx, beta):
    corrs = np.correlate(h, 1.0 - h, mode='full')
    m = corrs.max()
    shifted = corrs - m
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones_like(corrs) / len(corrs)
    else:
        w /= w_sum
    val = float((m + np.log(w_sum) / beta) * dx)
    n = len(h)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-13:
            continue
        grad += w[k] * _grad_corr_k(h, k, n)
    return val, grad * dx


def _sum_eq(x):
    return x.sum() - len(x) / 2.0


def _sum_eq_jac(x):
    return np.ones(len(x))


def slsqp_beta(h0, dx, target_sum, beta, maxiter=500):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_eq_jac}]

    def obj(x):
        v, _ = _smooth_obj_and_grad(x, dx, beta)
        return v

    def g(x):
        _, gr = _smooth_obj_and_grad(x, dx, beta)
        return gr

    try:
        res = minimize(obj, h0, jac=g, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        h_out = project_feasible(res.x, target_sum)
    except Exception:
        h_out = project_feasible(h0, target_sum)
    return h_out, c5_of_h(h_out, dx)


def _coord_obj(x, a, b, sum_others, target_sum, lam):
    vals = a * x + b
    peak = vals.max()
    s = sum_others + x - target_sum
    return peak + lam * s * s


def coord_descent(h0, dx, target_sum, max_passes=300, lam=1e5):
    n = len(h0)
    h = h0.copy()
    best_c5 = c5_of_h(h, dx)
    best_h = h.copy()
    no_improve = 0

    for p in range(max_passes):
        if p > 50 and no_improve > 10:
            break
        pass_improved = False
        for i in range(n):
            h_save = h[i]
            h[i] = 0.0
            c0 = np.correlate(h, 1.0 - h, mode='full')
            h[i] = 1.0
            c1 = np.correlate(h, 1.0 - h, mode='full')
            h[i] = h_save
            a = c1 - c0
            b = c0
            sum_others = h.sum() - h_save
            res = minimize_scalar(_coord_obj,
                                  args=(a, b, sum_others, target_sum, lam),
                                  bounds=[0.0, 1.0], method='bounded',
                                  options={'xatol': 1e-12})
            h[i] = res.x
            new_c5 = c5_of_h(h, dx)
            if new_c5 < best_c5 - 1e-14:
                best_c5 = new_c5
                best_h = h.copy()
                pass_improved = True
        if not pass_improved:
            no_improve += 1
        else:
            no_improve = 0

    best_h = project_feasible(best_h, target_sum)
    return best_h, c5_of_h(best_h, dx)


def subgrad_momentum(h0, dx, target_sum, max_iter=20000):
    n = len(h0)
    h = h0.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    alpha = 0.1
    gamma = 0.8
    no_improve = 0
    top_m = max(3, n // 15)

    for it in range(max_iter):
        corrs = np.correlate(h, 1.0 - h, mode='full')
        c5_val = corrs.max() * dx
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 8000:
            break
        sorted_idx = np.argsort(corrs)[-top_m:]
        grad = np.zeros(n)

            alpha = min(alpha * 1.002, 0.3)
        else:

            v *= 0.3
        if alpha < 1e-14:
            break
    return best_h, best_c5


def make_binary_start(n, target_sum, rng):
    h = np.zeros(n)
    idx = rng.permutation(n)

    frac = target_sum - n_ones
    if frac > 1e-10 and n_ones < n:
        h[idx[n_ones]] = frac
    return h
