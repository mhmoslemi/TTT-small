    HB = np.where((B >= 0) & (B < n), h[B.clip(0, n - 1)], 0.0)

    max_alpha = 1.0
    if neg_mask.any():
        max_alpha = min(max_alpha, (h[neg_mask] / (-delta[neg_mask])).min())
    if pos_mask.any():
        max_alpha = min(max_alpha, ((1.0 - h[pos_mask]) / delta[pos_mask]).min())
    max_alpha = max(0.0, min(1.0, max_alpha))
    
    if max_alpha < 1e-14:
        return h.copy(), c5_of_h(h, dx)
    
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    
    for frac in [1.0, 0.7, 0.5, 0.3, 0.1]:
        alpha = max_alpha * frac
        if alpha < 1e-14:
            continue
        h_try = h + alpha * delta
        h_try = np.clip(h_try, 0.0, 1.0)
        diff = target_sum - h_try.sum()
        if abs(diff) > 1e-12:
            mask = (h_try > 1e-12) & (h_try < 1.0 - 1e-12)
            if mask.sum() > 0:
                h_try[mask] += diff / mask.sum()
                h_try = np.clip(h_try, 0.0, 1.0)
        c5_try = c5_of_h(h_try, dx)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()
    
    return best_h, best_c5
