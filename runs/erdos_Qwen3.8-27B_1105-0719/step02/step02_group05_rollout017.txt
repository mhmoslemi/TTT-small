def build_h_from_q(q, n):
    m = n // 2
    if n % 2 == 0:

    else:
        return np.concatenate([q, [0.5], 1.0 - q[::-1]])

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corrs = np.correlate(h, 1.0 - h, mode='full')
    return corrs.max() * dx, corrs

def q_to_score(q, n):
    h = build_h_from_q(q, n)
    return exact_score(h)

def grad_corr_d(h, d, n):
    """Gradient of corr at lag d with respect to h."""
    g = np.zeros(n)
    if d >= 0:
        if d < n:
            g[d:] += (1.0 - h[:n - d])
            g[:n - d] += (-h[d:])
    else:
        dd = -d
        if dd < n:
            g[:n - dd] += (1.0 - h[dd:])
            g[dd:] += (-h[:n - dd])
    return g

def q_grad_smooth(q, n, beta):
    """Gradient of smooth log-sum-exp objective w.r.t. q."""
    h = build_h_from_q(q, n)
    dx = 2.0 / n
    corrs = np.correlate(h, 1.0 - h, mode='full')
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(len(corrs)) / len(corrs)
    else:
        w /= w_sum
    m = n // 2
    Gq = np.zeros(m)
    for k in range(len(corrs)):
        if w[k] < 1e-14:
            continue
        d = k - (n - 1)
        G = grad_corr_d(h, d, n)
        Gq += w[k] * (G[:m] - G[m:][::-1])
    return Gq * dx

class EpigraphQ:
    def __init__(self, n, active_k):
        self.n = n
        self.m = n // 2
        self.active_k = active_k
        self.dx = 2.0 / n

    def _build_h(self, q):
        n = self.n
        m = self.m
        if n % 2 == 0:
            return np.concatenate([q, 1.0 - q[::-1]])
        else:
            return np.concatenate([q, [0.5], 1.0 - q[::-1]])

    def obj(self, x):
        return x[0]

    def obj_grad(self, x):
        g = np.zeros(self.m + 1)
        g[0] = 1.0
        return g

    def constraint(self, x, k_idx):
        t = x[0]
        q = x[1:]
        h = self._build_h(q)
        k = self.active_k[k_idx]
        d = k - (self.n - 1)
        n = self.n
        if d >= 0:
            i_vals = np.arange(0, n - d)
            val = np.sum(h[i_vals + d] * (1.0 - h[i_vals]))
        else:
            dd = -d
            i_vals = np.arange(dd, n)
            val = np.sum(h[i_vals] * (1.0 - h[i_vals - dd]))
        val *= self.dx
        return t - val

    def constraint_grad(self, x, k_idx):
        q = x[1:]
        h = self._build_h(q)
        k = self.active_k[k_idx]
        d = k - (self.n - 1)
        n = self.n
        g = np.zeros(self.m + 1)
        g[0] = 1.0
        G = grad_corr_d(h, d, n)
        Gq = (G[:self.m] - G[self.m:][::-1]) * self.dx
        g[1:] = -Gq
        return g

            # Mutation

import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time


def build_h_from_q(q, n):
    m = n // 2
    if n % 2 == 0:
        return np.concatenate([q, 1.0 - q[::-1]])
    else:
        return np.concatenate([q, [0.5], 1.0 - q[::-1]])


def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corrs = np.correlate(h, 1.0 - h, mode='full')
    return corrs.max() * dx, corrs


def q_to_score(q, n):
    h = build_h_from_q(q, n)
    return exact_score(h)


def grad_corr_d(h, d, n):
    g = np.zeros(n)
    if d >= 0:
        if d < n:
            g[d:] += (1.0 - h[:n - d])
            g[:n - d] += (-h[d:])
    else:
        dd = -d
        if dd < n:
            g[:n - dd] += (1.0 - h[dd:])
            g[dd:] += (-h[:n - dd])
    return g


def q_val_smooth(q, n, beta):
    h = build_h_from_q(q, n)
    dx = 2.0 / n
    corrs = np.correlate(h, 1.0 - h, mode='full')
    return logsumexp(beta * corrs) / beta * dx


def q_grad_smooth(q, n, beta):
    h = build_h_from_q(q, n)
    dx = 2.0 / n
    corrs = np.correlate(h, 1.0 - h, mode='full')
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(len(corrs)) / len(corrs)
    else:
        w /= w_sum
    m = n // 2
    Gq = np.zeros(m)
    for k in range(len(corrs)):
        if w[k] < 1e-14:
            continue
        d = k - (n - 1)
        G = grad_corr_d(h, d, n)
        Gq += w[k] * (G[:m] - G[m:][::-1])
    return Gq * dx


class EpigraphQ:
    def __init__(self, n, active_k):
        self.n = n
        self.m = n // 2
        self.active_k = active_k
        self.dx = 2.0 / n

    def _build_h(self, q):
        n = self.n
        m = self.m
        if n % 2 == 0:
            return np.concatenate([q, 1.0 - q[::-1]])
        else:
            return np.concatenate([q, [0.5], 1.0 - q[::-1]])

    def obj(self, x):
        return x[0]

    def obj_grad(self, x):
        g = np.zeros(self.m + 1)
        g[0] = 1.0
        return g

    def constraint(self, x, k_idx):
        t = x[0]
        q = x[1:]
        h = self._build_h(q)
        k = self.active_k[k_idx]
        d = k - (self.n - 1)
        n = self.n
        if d >= 0:
            i_vals = np.arange(0, n - d)
            val = np.sum(h[i_vals + d] * (1.0 - h[i_vals]))
        else:
            dd = -d
            i_vals = np.arange(dd, n)
            val = np.sum(h[i_vals] * (1.0 - h[i_vals - dd]))
        return t - val * self.dx

    def constraint_grad(self, x, k_idx):
        q = x[1:]
        h = self._build_h(q)
        k = self.active_k[k_idx]
        d = k - (self.n - 1)
        n = self.n
        g = np.zeros(self.m + 1)
        g[0] = 1.0
        G = grad_corr_d(h, d, n)
        Gq = (G[:self.m] - G[self.m:][::-1]) * self.dx
        g[1:] = -Gq
        return g


def solve_epigraph_q(q0, n, active_k, t0, maxiter=200):
    m = n // 2
    x0 = np.concatenate([[t0], q0])
    ep = EpigraphQ(n, active_k)
    cons = []
    for i in range(len(active_k)):
        cons.append({
            'type': 'ineq',
            'fun': lambda x, i=i: ep.constraint(x, i),
            'jac': lambda x, i=i: ep.constraint_grad(x, i)
        })
    bounds = [(0.0, 5.0)] + [(0.0, 1.0)] * m
    try:
        res = minimize(ep.obj, x0, jac=ep.obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        q_out = np.clip(res.x[1:], 0.0, 1.0)
        return q_out, res.x[0], res.success
    except Exception:
        return q0, t0, False

def coordinate_descent_q(q0, n, max_passes=5):

    q = q0.copy()

        improved = False
        for i in range(m):
            best_val = c5_cur
            best_qi = q[i]
            # Try range of perturbations
            deltas = [0.005, -0.005, 0.01, -0.01, 0.02, -0.02,
                      0.05, -0.05, 0.1, -0.1, 0.2, -0.2]
            for delta in deltas:

            if s_trial <= scores[i]:
                pop[i] = trial
                scores[i] = s_trial
                if s_trial < best_score - 1e-14:
                    best_score = s_trial
                    best_q = trial.copy()
                    best_idx = i
                    stall_count = 0
                else:
                    stall_count += 1
            else:
                stall_count += 1

        if best_score < prev_best - 1e-8:
            prev_best = best_score
            stall_count = max(stall_count // 2, 0)

    return best_q, best_score


def smooth_lbfgs_q(q0, n, beta, maxiter=400):
    def obj(x):
        return q_val_smooth(x, n, beta)

    def obj_grad(x):
        return q_grad_smooth(x, n, beta)

    try:
        res = minimize(obj, q0, jac=obj_grad, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * len(q0),
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return np.clip(res.x, 0, 1)
    except Exception:
        return q0.copy()


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)

    def remaining():
        return budget_s - (time.time() - t_start)

    # Initialize from parent
    h_parent = np.array(initial_h_values, dtype=float)
    n_parent = len(h_parent)

    # Project to feasible
    ts = n_parent / 2.0
    h_p = np.clip(h_parent, 0, 1)
    diff = ts - h_p.sum()
    if abs(diff) > 1e-10:
        mask = (h_p > 1e-10) & (h_p < 1.0 - 1e-10)
        if mask.sum() > 0:
            h_p[mask] += diff / mask.sum()
    h_p = np.clip(h_p, 0, 1)

    best_h = h_p.copy()
    best_c5, _ = exact_score(h_p)

    # Work in q-space (anti-symmetric)
    n = n_parent if n_parent % 2 == 0 else n_parent - 1
    m = n // 2
    if n_parent == n:
        q0 = h_p[:m].copy()
    else:
        x = np.linspace(0, 2, n_parent)
        x_new = np.linspace(0, 2, n)
        h_interp = np.interp(x_new, x, h_p)
        q0 = h_interp[:m].copy()

    best_q = q0.copy()
    q_s, c5_s = q_to_score(best_q, n)
    if c5_s < best_c5 - 1e-14:
        best_c5 = c5_s
        best_h = build_h_from_q(best_q, n)

    # Phase 1: Smooth L-BFGS with increasing beta
    for beta in [50, 150, 400, 1200, 4000]:
        if remaining() < 25:
            break
        q_new = smooth_lbfgs_q(best_q, n, beta, maxiter=500)
        s_new, _ = q_to_score(q_new, n)
        if s_new < best_c5 - 1e-14:
            best_c5 = s_new
            best_q = q_new
            best_h = build_h_from_q(q_new, n)

    # Phase 2: Active-set epigraph (main refinement)
    if remaining() > 60:
        best_q, best_c5 = active_set_search(best_q, n, max_rounds=15)
        best_h = build_h_from_q(best_q, n)

    # Phase 3: Coordinate descent
    if remaining() > 25:
        best_q, best_c5 = coordinate_descent_q(best_q, n, max_passes=5)
        best_h = build_h_from_q(best_q, n)

    # Phase 4: Another active-set round after coordinate descent
    if remaining() > 50:
        best_q, best_c5 = active_set_search(best_q, n, max_rounds=12)
        best_h = build_h_from_q(best_q, n)

    # Phase 5: Differential evolution (global search)
    if remaining() > 120:
        de_budget = int(min(6000, remaining() * 0.25))
        q_de, c5_de = differential_evolution_q(best_q, n, rng, eval_budget=de_budget)
        s_de, _ = q_to_score(q_de, n)
        if s_de < best_c5 - 1e-12:
            best_c5 = s_de
            best_q = q_de
            best_h = build_h_from_q(q_de, n)
        # Polish DE result
        if remaining() > 40:
            q_pol, c5_pol = active_set_search(q_de, n, max_rounds=10)
            s_pol, _ = q_to_score(q_pol, n)
            if s_pol < best_c5 - 1e-12:
                best_c5 = s_pol
                best_q = q_pol
                best_h = build_h_from_q(q_pol, n)

    # Phase 6: Try smaller n values
    n_list = [64, 68, 72, 76, 80, 84, 88, 92, 96, 100, 104, 108, 112, 116]
    for n_try in n_list:
        if remaining() < 40:
            break
        m_try = n_try // 2
        x_orig = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_try)
        h_interp = np.interp(x_new, x_orig, best_h)
        q_interp = h_interp[:m_try].copy()

        # Quick smooth
        q_sm = smooth_lbfgs_q(q_interp, n_try, 500, maxiter=200)
        s_sm, _ = q_to_score(q_sm, n_try)
        if s_sm < best_c5 - 1e-12:
            best_c5 = s_sm
            best_h = build_h_from_q(q_sm, n_try)

        # Epigraph
        if remaining() > 30:
            q_es, c5_es = active_set_search(q_sm, n_try, max_rounds=10)
            s_es, _ = q_to_score(q_es, n_try)
            if s_es < best_c5 - 1e-12:
                best_c5 = s_es
                best_h = build_h_from_q(q_es, n_try)

    # Phase 7: Final polish on best
    if remaining() > 30:
        n_b = len(best_h)
        if n_b % 2 != 0:
            n_b = n_b - 1
            x = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_b)
            best_h = np.interp(x_new, x, best_h)
        m_b = n_b // 2
        q_b = best_h[:m_b].copy()

        if remaining() > 20:
            q_f = smooth_lbfgs_q(q_b, n_b, 2000, maxiter=400)
            s_f, _ = q_to_score(q_f, n_b)
            if s_f < best_c5 - 1e-14:
                best_c5 = s_f
                best_h = build_h_from_q(q_f, n_b)
                q_b = q_f

        if remaining() > 30:
            q_f2, c5_f2 = active_set_search(q_b, n_b, max_rounds=12)
            s_f2, _ = q_to_score(q_f2, n_b)
            if s_f2 < best_c5 - 1e-14:
                best_c5 = s_f2
                best_h = build_h_from_q(q_f2, n_b)

        if remaining() > 20:
            q_cd2, c5_cd2 = coordinate_descent_q(best_h[:n_b // 2], n_b, max_passes=3)
            s_cd2, _ = q_to_score(q_cd2, n_b)
            if s_cd2 < best_c5 - 1e-14:
                best_c5 = s_cd2
                best_h = build_h_from_q(q_cd2, n_b)

    # Final validation
    n_out = len(best_h)
    ts_out = n_out / 2.0
    best_h = np.clip(best_h, 0, 1)
    diff = ts_out - best_h.sum()
    if abs(diff) > 1e-10:
        mask = (best_h > 1e-8) & (best_h < 1.0 - 1e-8)
        if mask.sum() > 0:
            best_h[mask] += diff / mask.sum()
    best_h = np.clip(best_h, 0, 1)

    final_score, _ = exact_score(best_h)
    return best_h, final_score, n_out
