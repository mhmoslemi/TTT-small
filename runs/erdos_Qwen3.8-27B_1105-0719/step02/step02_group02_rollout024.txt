    """Solve the 1D subproblem for coordinate i."""
    # Compute p_{i,d} for all lags d
    # d ranges from -(n-1) to (n-1), index in corr is d + n - 1
    
    # For lag d > 0: p = I(d<=i)*(1-h[i-d]) - I(i<=n-1-d)*h[i+d]
    # For lag d < 0, d' = -d: p = I(d'<=i)*(1-h[i-d']) - I(i<=n-1-d')*h[i+d']
    # For lag 0: special (quadratic)
    
    corr = np.correlate(h, 1-h, mode='full')
    h_i = h[i]
    
    # Evaluate f(t) at grid points
    ts = np.linspace(0, 1, 200)
    vals = np.zeros((200, 2*n-1))
    
    for d in range(-(n-1), n):
        idx = d + n - 1
        if d == 0:
            # val = corr[idx] + (t - h_i) - (t^2 - h_i^2)
            # = corr[idx] + t - h_i - t^2 + h_i^2
            vals[:, idx] = corr[idx] - h_i + h_i**2 + ts - ts**2
        else:

            if i <= n - 1 - ad:
                p -= h[i + ad]
            vals[:, idx] = corr[idx] + p * (ts - h_i)
    
    f = vals.max(axis=1)
    best_idx = np.argmin(f)
    t_best = ts[best_idx]
    
    # Local refinement
    # ... golden section or fine grid around t_best
    
    return t_best

def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h


def c5_of_h(h, dx):
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)


def coeff_p(h, i, n):
    ds = np.arange(1, n)
    p = np.zeros(n - 1)
    m1 = ds <= i
    m2 = ds <= (n - 1 - i)
    if m1.any():
        p[m1] += 1.0 - h[i - ds[m1]]
    if m2.any():
        p[m2] -= h[i + ds[m2]]
    return p

Wait, I made an error in the corr update for lag 0. Let me redo:
The lag 0 correlation is: sum_j h[j](1-h[j]) = sum_j (h[j] - h[j]^2)
When h[i] changes from h_i to t = h_i + delta:
New lag 0 = old lag 0 + (t - t^2) - (h_i - h_i^2)
= old lag 0 + t - t^2 - h_i + h_i^2
= old lag 0 + delta - (h_i + delta)^2 + h_i^2
= old lag 0 + delta - h_i^2 - 2*h_i*delta - delta^2 + h_i^2
= old lag 0 + delta - 2*h_i*delta - delta^2
= old lag 0 + delta*(1 - 2*h_i) - delta^2

So corr[n-1] += delta*(1 - 2*h_i) - delta^2. That's what I have. Good.

        return float(logsumexp(beta * c) / beta * dx)

        shifted = c - c.max()
        w = np.exp(beta * shifted)
