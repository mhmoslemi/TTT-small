```python
            s = k - (n - 1)  # actual shift
            if s == 0:
                continue
            if s > 0:
                for i in range(n - s):

                        candidates.add((i, i+s))
            else:
                s_neg = -s
                for i in range(s_neg, n):
                    if h[i] == 1 and h[i-s_neg] == 0:
                        candidates.add((i, i-s_neg))
        
        # Try top candidates
        cand_list = list(candidates)
        rng.shuffle(cand_list)
        # Limit to a manageable number
        cand_list = cand_list[:200]
        
        for i, j in cand_list:
            delta = compute_delta_binary(h, i, j, n)
            new_max = (corrs + delta).max()
            if new_max < best_new_max:
                best_new_max = new_max
                best_swap = (i, j)
        
        if best_swap is None:
            break
        
        i, j = best_swap
        h[i] = 0.0
        h[j] = 1.0
        corrs = compute_corrs(h)
        new_c5 = corrs.max() * dx
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
    
    return best_h, best_c5
```

```python
import numpy as np
from scipy.optimize import minimize
import time


def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-10:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h = h.copy()
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)


def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def c5_of_h(h, dx):
    return compute_corrs(h).max() * dx


def compute_delta_binary(h, i, j, n):
    ms = np.arange(-(n - 1), n)
    v1 = (0 <= i - ms) & (i - ms <= n - 1)
    v2 = (0 <= i + ms) & (i + ms <= n - 1)
    v3 = (0 <= j - ms) & (j - ms <= n - 1)
    v4 = (0 <= j + ms) & (j + ms <= n - 1)
    delta = np.zeros(2 * n - 1)
    if v1.any():
        delta[v1] -= (1.0 - h[i - ms[v1]])
    if v2.any():
        delta[v2] += h[i + ms[v2]]
    if v3.any():
        delta[v3] += (1.0 - h[j - ms[v3]])
    if v4.any():
        delta[v4] -= h[j + ms[v4]]
    return delta


def make_binary(h0, n):
    h = np.zeros(n)
    idx = np.argsort(-h0)
    h[idx[:n // 2]] = 1.0
    return h


def make_sturmian(n, alpha):
    h = np.zeros(n)
    s = 0.0
    for i in range(n):
        s += alpha
        if s >= 1.0:
            s -= 1.0
            h[i] = 1.0
    # Ensure exactly n/2 ones
    n1 = int(h.sum())
    if n1 > n // 2:
        ones = np.where(h == 1.0)[0]
        for i in range(n1 - n // 2):
            h[ones[i]] = 0.0
    elif n1 < n // 2:
        zeros = np.where(h == 0.0)[0]
        for i in range(n // 2 - n1):
            h[zeros[i]] = 1.0
    return h


def make_quad_residue(n, p):
    h = np.zeros(n)
    res = set()
    for x in range(1, p):
        res.add((x * x) % p)
    for i in range(n):
        if i % p in res:
            h[i] = 1.0
    n1 = int(h.sum())
    target = n // 2
    if n1 > target:
        ones = np.where(h == 1.0)[0]
        np.random.shuffle(ones)
        for i in range(n1 - target):
            h[ones[i]] = 0.0
    elif n1 < target:
        zeros = np.where(h == 0.0)[0]
        np.random.shuffle(zeros)
        for i in range(target - n1):
            h[zeros[i]] = 1.0
    return h


def sa_local_search(h0, dx, rng, max_iters=500000, patience=10000):
    n = len(h0)
    target = n // 2
    h = make_binary(h0, n)
    corrs = compute_corrs(h)
    best_c5 = corrs.max() * dx
    best_h = h.copy()

    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]

    no_improve = 0
    temp = 0.04
    cool_rate = 0.9995

    for it in range(max_iters):
        if it % 5000 == 0 and temp > 0.0005:
            temp *= cool_rate

        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]

        delta = compute_delta_binary(h, i, j, n)
        new_max = (corrs + delta).max()
        new_c5 = new_max * dx

        if new_c5 <= best_c5 + 1e-14:
            h[i] = 0.0
            h[j] = 1.0
            corrs = corrs + delta
            best_c5 = new_c5
            best_h = h.copy()
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            no_improve = 0
        else:
            if temp > 1e-10:
                val = np.exp(-(new_c5 - best_c5) / temp)
                if rng.random() < val:
                    h[i] = 0.0
                    h[j] = 1.0
                    corrs = corrs + delta
                    ones = np.where(h == 1.0)[0]
                    zeros = np.where(h == 0.0)[0]
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1

        if no_improve > patience:
            break

    return best_h, best_c5


def steepest_descent(h0, dx, max_sweeps=80):
    n = len(h0)
    target = n // 2
    h = make_binary(h0, n)
    corrs = compute_corrs(h)
    best_c5 = corrs.max() * dx
    best_h = h.copy()

    for sweep in range(max_sweeps):
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        current_max = corrs.max()
        best_new_max = current_max
        best_swap = None

        for i in ones:
            for j in zeros:
                delta = compute_delta_binary(h, i, j, n)
                new_max = (corrs + delta).max()
                if new_max < best_new_max - 1e-14:
                    best_new_max = new_max
                    best_swap = (i, j)

        if best_swap is None:
            break

        i, j = best_swap
        h[i] = 0.0
        h[j] = 1.0
        corrs = compute_corrs(h)
        new_c5 = corrs.max() * dx
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()

    return best_h, best_c5


def two_opt_polish(h0, dx, max_rounds=20):
    n = len(h0)
    target = n // 2
    h = make_binary(h0, n)
    corrs = compute_corrs(h)
    best_c5 = corrs.max() * dx
    best_h = h.copy()

    for rnd in range(max_rounds):
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        current_max = corrs.max()

        # Find top single swaps
        swap_scores = []
        for i in ones:
            for j in zeros:
                delta = compute_delta_binary(h, i, j, n)
                new_max = (corrs + delta).max()
                swap_scores.append((new_max, i, j))
        swap_scores.sort()
        top_swaps = swap_scores[:8]

        if top_swaps[0][0] >= current_max - 1e-14:
            break

        # Try single best
        _, i, j = top_swaps[0]
        h[i] = 0.0
        h[j] = 1.0
        corrs = compute_corrs(h)
        if corrs.max() * dx < best_c5:
            best_c5 = corrs.max() * dx
            best_h = h.copy()

        # Try pairs of top swaps (that don't conflict)
        for a in range(len(top_swaps)):
            for b in range(a + 1, len(top_swaps)):
                _, i1, j1 = top_swaps[a]
                _, i2, j2 = top_swaps[b]
                if i1 in (i2, j2) or j1 in (i2, j2) or i1 == i2 or j1 == j2:
                    continue
                h2 = h.copy()
                h2[i1] = 0.0
                h2[j1] = 1.0
                h2[i2] = 0.0
                h2[j2] = 1.0
                c2 = compute_corrs(h2).max() * dx
                if c2 < best_c5 - 1e-14:
                    best_c5 = c2
                    best_h = h2.copy()
                    h = h2.copy()
                    corrs = compute_corrs(h)

    return best_h, best_c5


def smooth_max_obj(x, dx, beta):
    corrs = compute_corrs(x)
    m = corrs.max()
    shifted = corrs - m
    return (m + np.log(np.sum(np.exp(beta * shifted))) / beta) * dx


def smooth_max_grad(x, dx, beta):
    n = len(x)
    corrs = compute_corrs(x)
    m = corrs.max()
    shifted = corrs - m
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    w = w / w_sum

    grad = np.zeros(n)
    for k in range(len(corrs)):
        if w[k] < 1e-14:
            continue
        offset = k - (n - 1)
        grad_k = np.zeros(n)
        i_lo = max(0, -offset)
        i_hi = min(n - 1, n - 1 - offset)
        if i_lo > i_hi:
            continue
        idx = np.arange(i_lo, i_hi + 1)
        grad_k[idx + offset] += (1.0 - x[idx])
        grad_k[idx] += (-x[idx + offset])
        grad += w[k] * grad_k
    return grad * dx


def slsqp_optimize(h0, dx, target_sum, beta, maxiter=600):
    n = len(h0)

    def obj(x):
        return smooth_max_obj(x, dx, beta)

    def obj_grad(x):
        return smooth_max_grad(x, dx, beta)

    def cons_fun(x):
        return x.sum() - target_sum

    def cons_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    bounds = [(0.0, 1.0)] * n

    try:
        res = minimize(obj, h0, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        h_out = project_feasible(res.x, target_sum)
    except Exception:
        h_out = project_feasible(h0, target_sum)
    return h_out, c5_of_h(h_out, dx)


def remaining_time(t0, budget_s):
    return budget_s - (time.time() - t0)
```
