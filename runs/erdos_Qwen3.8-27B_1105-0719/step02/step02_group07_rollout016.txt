    return logsumexp(beta * corrs) / beta

    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)

    if ws < 1e-300:
        return np.ones(len(corrs)) / len(corrs)

    h = make_sym_h(q, n)

    m = n // 2
    dx = 2.0 / n

        return x[0]

        g[0] = 1.0
        return g

        t = x[0]
        q = x[1:]
        h = make_sym_h(q, n)
        corrs = corr_vals(h)
        return t - corrs[t_lags]

        h = make_sym_h(q, n)
        J = np.zeros((L, m + 1))

            G[idx + d] += (-h[idx])
            G *= dx
            J[i, 1:] = -grad_q_from_G(G, n)

import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time


def project_box_sum(x, target):
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)


def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full').max() * dx


def corr_vals(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx


def compute_G(h, weights):
    n = len(h)
    dx = 2.0 / n
    N = 2 * n - 1
    G = np.zeros(n)
    for k in range(N):
        wk = weights[k]
        if wk < 1e-15:
            continue
        d = k - (n - 1)
        i_lo = max(0, -d)
        i_hi = min(n - 1, n - 1 - d)
        if i_lo > i_hi:
            continue
        idx = np.arange(i_lo, i_hi + 1)
        G[idx] += wk * (1.0 - h[idx + d])
        G[idx + d] += wk * (-h[idx])
    return G * dx


def softmax_weights(corrs, beta):
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    ws = w.sum()
    if ws < 1e-300:
        return np.ones(len(corrs)) / len(corrs)
    return w / ws


def sym_h(q, n):
    m = n // 2
    h = np.empty(n)
    h[:m] = q
    h[m:] = 1.0 - q[::-1]
    return h


def q_from_h(h, n):
    m = n // 2
    q = 0.5 * (h[:m] + 1.0 - h[n - 1: n - 1 - m: -1])
    return np.clip(q, 0.0, 1.0)


def sym_grad_q(G, n):
    m = n // 2
    return G[:m] - G[n - 1: n - 1 - m: -1]


def _obj_sym_lse(x, n, beta):
    h = sym_h(x, n)
    c = corr_vals(h)
    return logsumexp(beta * c) / beta


def _jac_sym_lse(x, n, beta):
    h = sym_h(x, n)
    c = corr_vals(h)
    w = softmax_weights(c, beta)
    G = compute_G(h, w)
    return sym_grad_q(G, n)


def _obj_full_lse(x, n, beta):
    h = project_box_sum(x, n / 2.0)
    c = corr_vals(h)
    return logsumexp(beta * c) / beta


def _jac_full_lse(x, n, beta):
    h = project_box_sum(x, n / 2.0)
    c = corr_vals(h)
    w = softmax_weights(c, beta)
    return compute_G(h, w)


class EpiSolver:
    def __init__(self, n, active_lags):
        self.n = n
        self.m = n // 2
        self.dx = 2.0 / n
        self.active_lags = np.array(active_lags, dtype=int)
        self.L = len(active_lags)

    def fun(self, x):
        return x[0]

    def jac(self, x):
        g = np.zeros(self.m + 1)
        g[0] = 1.0
        return g

    def con_fun(self, x):
        t = x[0]
        q = x[1:]
        h = sym_h(q, self.n)
        return t - corr_vals(h)[self.active_lags]

    def con_jac(self, x):
        q = x[1:]
        h = sym_h(q, self.n)
        n = self.n
        m = self.m
        dx = self.dx
        J = np.zeros((self.L, m + 1))
        J[:, 0] = 1.0
        for i, lag in enumerate(self.active_lags):
            d = lag - (n - 1)
            i_lo = max(0, -d)
            i_hi = min(n - 1, n - 1 - d)
            if i_lo > i_hi:
                continue
            G = np.zeros(n)
            idx = np.arange(i_lo, i_hi + 1)
            G[idx] += (1.0 - h[idx + d])
            G[idx + d] += (-h[idx])
            G *= dx
            J[i, 1:] = -sym_grad_q(G, n)
        return J


def slsqp_sym(q0, n, active_lags, t_init, maxiter=300):
    m = n // 2
    solver = EpiSolver(n, active_lags)
    x0 = np.concatenate([[t_init], q0.copy()])
    bounds = [(0.0, 1.0)] + [(0.0, 1.0)] * m
    cons = [{'type': 'ineq', 'fun': solver.con_fun, 'jac': solver.con_jac}]
    try:
        res = minimize(solver.fun, x0, jac=solver.jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.fun < 10:
            return res.x[1:], exact_score(sym_h(res.x[1:], n))
    except Exception:
        pass
    return None, None


def slsqp_full_lse(h0, n, beta, maxiter=300):
    try:
        res = minimize(lambda x: _obj_full_lse(x, n, beta), h0.copy(),
                       jac=lambda x: _jac_full_lse(x, n, beta),
                       method='SLSQP', bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        h_out = project_box_sum(res.x, n / 2.0)
        return h_out, exact_score(h_out)
    except Exception:
        return h0, exact_score(h0)


def coord_descent_step(h, n, k_indices):
    """One pass of coordinate descent on specified indices"""
    target = n / 2.0

        for trial in range(20):
            val = 0.02 * (trial + 1)
            if val > 1.0:

                trial_h = h.copy()
                trial_h[idx] = np.clip(best_val + direction * val, 0.0, 1.0)

                c5_t = exact_score(trial_h)
                if c5_t < best_c5 - 1e-12:
                    best_c5 = c5_t
                    best_val = trial_h[idx]
                    h = trial_h
    return h
