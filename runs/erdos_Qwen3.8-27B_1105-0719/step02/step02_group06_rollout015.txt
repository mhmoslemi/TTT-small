    pop = [h_init.copy()]
    for i in range(pop_size - 1):

            mutant = pop[a] + F * (pop[b] - pop[c])
            # Crossover
            trial = np.zeros(N)
            for j in range(N):
                if rng.rand() < CR or j == rng.randint(N):

                else:
                    trial[j] = pop[i][j]

            trial_score = eval_score(trial)
            
            if trial_score <= scores[i]:
                new_pop.append(trial)
                new_scores.append(trial_score)
            else:
                new_pop.append(pop[i])
                new_scores.append(scores[i])
        
        pop = new_pop
        scores = new_scores
        now_best = np.argmin(scores)
        if scores[now_best] < best_score:
            best_score = scores[now_best]
            best_h = pop[now_best].copy()

        return h, False, 0.0
    
    k_affected = np.where(affected_k)[0]
    R_unaffected = base_r[~affected_k].max() if (~affected_k).any() else 0.0
    
    # Compute A_k, B_k for each affected k
    A_arr = np.zeros(len(k_affected))
    B_arr = np.zeros(len(k_affected))
    base_aff = base_r[k_affected]
    
    for idx, k in enumerate(k_affected):
        d = k - (N-1)
        i_lo = max(0, -d)
        i_hi = min(N-1, N-1-d)
        # Affected i values
        candidates = [a, b, a-d, b-d]
        A_k = 0.0
        B_k = 0.0
        seen = set()
        for ic in candidates:
            if i_lo <= ic <= i_hi and ic not in seen:
                seen.add(ic)
                alpha = 0.0
                beta = 0.0
                if ic == a: alpha = 1.0
                elif ic == b: alpha = -1.0
                ik = ic + d
                if ik == a: beta = 1.0
                elif ik == b: beta = -1.0
                A_k += -alpha * beta
                B_k += alpha * (1.0 - h[ik]) - beta * h[ic]
        A_arr[idx] = A_k
        B_arr[idx] = B_k
    
    # 1D optimization
    def obj(delta):
        vals = base_aff + A_arr * delta * delta + B_arr * delta
        m = vals.max()
        return max(m, R_unaffected)
    
    try:
        res = minimize_scalar(obj, bounds=(d_lo, d_hi), method='bounded',
                              options={'xatol': 1e-10})
        delta_opt = res.x
    except:
        return h, False, 0.0
    
    if abs(delta_opt) < 1e-13:
        return h, False, 0.0
    
    h_new = h.copy()
    h_new[a] = np.clip(h[a] + delta_opt, 0.0, 1.0)
    h_new[b] = np.clip(h[b] - delta_opt, 0.0, 1.0)
    
    # Re-project to maintain sum (should be exact but just in case)
    h_new = project_feasible(h_new, N / 2.0)
    
    return h_new, True, delta_opt

Wait no. C5 = max_k ∫ h(x)(1-h(x+k)) dx. Since ∫ h = 1 and h ∈ [0,1]:
∫ h(x)(1-h(x+k)) dx = ∫_0^{2-k} h(x)dx - ∫_0^{2-k} h(x)h(x+k)dx (for 0<k<2)

The first term depends on k and the distribution of h. The second is the autocorrelation.

For a "two-valued" h (h=0 or 1 almost everywhere), ∫ h(x)(1-h(x+k)) dx = measure of {x : h(x)=1, h(x+k)=0} = measure of (A \ (A-k)) where A = {x: h(x)=1}. This is the measure of points in A that are k away from a point not in A.

So the problem is related to finding a set A of measure 1 (out of [0,2]) that minimizes the maximum "boundary" measure at any shift k.

This is related to the study of "difference sets" or "Bohr sets" in additive combinatorics.

For a step function h that is 1 on a set of measure 1 and 0 elsewhere (on [0,2]), the overlap integral at shift k is the measure of the symmetric difference between A and A+k (intersected with [0,2]). Minimizing the maximum of this over k is a classical problem.

The Erdős constant C_5 is related to the minimum possible maximum overlap for such a function.

def eval_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(x, target_sum):
    x = np.asarray(x, dtype=float).copy()
    bad = ~np.isfinite(x)
    if bad.any():
        x[bad] = 0.5
    x = np.clip(x, 0.0, 1.0)
    cur_sum = x.sum()
    if abs(cur_sum - target_sum) < 1e-13:
        return x
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(x + mid, 0.0, 1.0).sum()
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    h = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - h.sum()
    if diff > 1e-12:
        order = np.argsort(h)
        for idx in order:
            add = min(diff, 1.0 - h[idx])
            if add <= 0: continue
            h[idx] += add
            diff -= add
            if abs(diff) < 1e-13: break
    elif diff < -1e-12:
        order = np.argsort(-h)
        for idx in order:
            sub = min(-diff, h[idx])
            if sub <= 0: continue
            h[idx] -= sub
            diff += sub
            if abs(diff) < 1e-13: break
    return h

def transfer_to_n(h_old, n_new):
    n_old = len(h_old)
    x_old = (np.arange(n_old) + 0.5) * 2.0 / n_old
    x_new = (np.arange(n_new) + 0.5) * 2.0 / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new / 2.0)

def paired_step(h, a, b):
    N = len(h)
    d_lo = max(-h[a], h[b] - 1.0)
    d_hi = min(1.0 - h[a], h[b])
    if d_hi <= d_lo + 1e-11:
        return h, False
    
    base_r = np.correlate(h, 1.0 - h, mode='full')
    
    # Find affected output indices
    affected_mask = np.zeros(2*N-1, dtype=bool)
    for k in range(2*N-1):
        d = k - (N-1)
        i_lo = max(0, -d)
        i_hi = min(N-1, N-1-d)
        if (i_lo <= a <= i_hi) or (i_lo <= b <= i_hi) or (i_lo <= a-d <= i_hi) or (i_lo <= b-d <= i_hi):
            affected_mask[k] = True
    
    if not affected_mask.any():
        return h, False
    
    k_aff = np.where(affected_mask)[0]
    n_aff = len(k_aff)
    
    # Max of unaffected correlations
    if (2*N-1 - n_aff) > 0:
        R_unk = base_r[~affected_mask].max()
    else:
        R_unk = 0.0
    
    base_aff = base_r[k_aff]
    
    # Compute quadratic coefficients for each affected k
    A_arr = np.zeros(n_aff)
    B_arr = np.zeros(n_aff)
    for ii, k in enumerate(k_aff):
        d = k - (N-1)
        i_lo = max(0, -d)
        i_hi = min(N-1, N-1-d)
        A_k = 0.0
        B_k = 0.0
        seen = set()
        for ic in (a, b, a-d, b-d):
            if i_lo <= ic <= i_hi and ic not in seen:
                seen.add(ic)
                alpha = 1.0 if ic == a else (-1.0 if ic == b else 0.0)
                ik = ic + d
                beta = 1.0 if ik == a else (-1.0 if ik == b else 0.0)
                A_k -= alpha * beta
                B_k += alpha * (1.0 - h[ik]) - beta * h[ic]
        A_arr[ii] = A_k
        B_arr[ii] = B_k
    
    def obj(delta):
        vals = base_aff + A_arr * delta * delta + B_arr * delta
        return max(vals.max(), R_unk)
    
    try:

                              options={'xatol': 1e-11})
        delta_opt = res.x
    except Exception:
        return h, False
    
    if abs(delta_opt) < 1e-12:
        return h, False
    
    h[a] += delta_opt
    h[b] -= delta_opt
    h = np.clip(h, 0.0, 1.0)
    
    return h, True

        rng = np.random.RandomState(0)
    N = len(h)

            break
        n_improved = 0

            if time_limit is not None and time.time() > time_limit:
                break
            a = rng.randint(N)
            b = rng.randint(N)
            if a == b:
                continue
            h_new, improved = paired_step(h, a, b)
            if improved:
                h = h_new
                n_improved += 1
        sc = eval_score(h)
        if sc < best_local:
            best_local = sc
        if n_improved == 0:
            break
    
    return h, best_local
