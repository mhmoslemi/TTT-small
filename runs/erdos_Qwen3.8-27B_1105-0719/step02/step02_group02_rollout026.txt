import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time

def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-15) & (h < 1.0 - 1e-15)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def c5_of_h(h):
    n = len(h)
    return float(np.correlate(h, 1.0 - h, mode='full').max() * 2.0 / n)

def smooth_max(h, beta):
    n = len(h)
    corrs = np.correlate(h, 1.0 - h, mode='full')
    return float(logsumexp(beta * corrs) / beta * 2.0 / n)

def smooth_max_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corrs = np.correlate(h, 1.0 - h, mode='full')
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    w /= w.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        m = k - (n - 1)
        if m == 0:
            grad += w[k] * (1.0 - 2.0 * h)
        elif m > 0:
            idx = np.arange(n - m)
            grad[idx] += w[k] * (1.0 - h[idx + m])
            grad[idx + m] += w[k] * (-h[idx])
        else:
            idx = np.arange(-m, n + m)
            grad[idx] += w[k] * (1.0 - h[idx + m])
            grad[idx + m] += w[k] * (-h[idx])
    return grad * dx

def slsqp_opt(h0, beta, maxiter, t0, budget_s, time_limit=30.0):
    n = len(h0)
    target_sum = n / 2.0
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        return smooth_max(x, beta)
    
    def grad(x):
        return smooth_max_grad(x, beta)
    
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum}]
    try:
        res = minimize(obj, h0, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        if (time.time() - t0) > (budget_s - time_limit):

        else:
            h_out = project_feasible(res.x, target_sum)

        h_out = project_feasible(h0, target_sum)
    return h_out, c5_of_h(h_out)

def project_feasible(x, target_sum):
    """Project x onto {0<=h<=1, sum(h)=target_sum}"""
    n = len(x)
    lo, hi = -1.0, 2.0
    for _ in range(80):

        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)

    if abs(diff) > 1e-13:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def c5_of_h(h):
    """Compute C5 for a feasible h"""
    n = len(h)
    return float(np.correlate(h, 1.0 - h, mode='full').max() * 2.0 / n)

def smooth_max_val(h, beta):
    n = len(h)
    corrs = np.correlate(h, 1.0 - h, mode='full')
    return float(logsumexp(beta * corrs) / beta * 2.0 / n)

def smooth_max_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corrs = np.correlate(h, 1.0 - h, mode='full')
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    w = w / w_sum
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        m = k - (n - 1)
        if m == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif m > 0:
            idx = np.arange(n - m)
            grad[idx] += wk * (1.0 - h[idx + m])
            grad[idx + m] += wk * (-h[idx])
        else:
            idx = np.arange(-m, n + m)
            grad[idx] += wk * (1.0 - h[idx + m])
            grad[idx + m] += wk * (-h[idx])
    return grad * dx

def slsqp_opt(h0, beta, maxiter):
    n = len(h0)
    target_sum = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum}]
    try:
        res = minimize(smooth_max_val, h0, jac=lambda x: smooth_max_grad(x, beta),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        h_out = project_feasible(res.x, target_sum)
    except Exception:
        h_out = project_feasible(h0, target_sum)
    return h_out, c5_of_h(h_out)
