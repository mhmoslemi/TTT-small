def eval_c5(h):

def project_to_sum(x, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(x - mid, 0.0, 1.0).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)

    if abs(diff) > 1e-14:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.any():
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def compute_slopes_for_coord(h, j, n):
    """Compute the slope of each correlation value w.r.t. h[j]."""
    slopes = np.zeros(2*n - 1)
    for k in range(2*n - 1):
        d = k - (n - 1)  # actual lag
        slope = 0.0
        if d >= 0:
            if j >= d:
                slope += (1.0 - h[j - d])
            if j + d < n:
                slope -= h[j + d]
        else:
            dd = -d
            if j < n - dd:
                slope += (1.0 - h[j + dd])
            if j >= dd:
                slope -= h[j - dd]
        slopes[k] = slope
    return slopes

def minimax_1d(offsets, slopes):
    """Find x in [0,1] minimizing max_i (offsets[i] + slopes[i]*x)."""
    m = len(offsets)
    # Collect breakpoints
    pts = [0.0, 1.0]
    for i in range(m):
        si = slopes[i]
        oi = offsets[i]
        for j in range(i+1, m):
            sj = slopes[j]
            if abs(si - sj) < 1e-15:
                continue
            x = (offsets[j] - oi) / (si - sj)
            if 0.0 < x < 1.0:
                pts.append(x)
    
    # Evaluate at each breakpoint
    best_x = 0.0
    best_val = float('inf')
    for x in pts:
        val = np.max(offsets + slopes * x)
        if val < best_val:
            best_val = val
            best_x = x
    return best_x, best_val

    m = len(offsets)
    # Use vectorized computation
    # For each pair (i, j), compute intersection
    i_idx, j_idx = np.triu_indices(m, 1)
    valid = np.abs(slopes[i_idx] - slopes[j_idx]) > 1e-15

        j_v = j_idx[valid]

def minimax_1d_ternary(offsets, slopes, n_iter=80):
    """Find x in [0,1] minimizing max_i (offsets[i] + slopes[i]*x) using ternary search."""
    lo, hi = 0.0, 1.0
    for _ in range(n_iter):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = np.max(offsets + slopes * m1)
        f2 = np.max(offsets + slopes * m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    x = (lo + hi) / 2.0
    return x, np.max(offsets + slopes * x)

import numpy as np
import time

def eval_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def project_to_sum(x, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(x - mid, 0.0, 1.0).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.any():
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def compute_slopes_vec(h, j, n):
    slopes = np.zeros(2 * n - 1)
    slopes[n - 1] = 1.0 - 2.0 * h[j]
    d_pos = np.arange(1, n)
    p1 = np.where(j >= d_pos, 1.0 - h[j - d_pos], 0.0)
    p2 = np.where(j + d_pos < n, h[j + d_pos], 0.0)
    slopes[n:] = p1 - p2
    dd_neg = np.arange(n - 1, 0, -1)
    n1 = np.where(j < n - dd_neg, 1.0 - h[j + dd_neg], 0.0)
    n2 = np.where(j >= dd_neg, h[j - dd_neg], 0.0)
    slopes[:n - 1] = n1 - n2
    return slopes

def minimax_1d(offsets, slopes, n_iter=50):
    lo, hi = 0.0, 1.0
    for _ in range(n_iter):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = np.max(offsets + slopes * m1)
        f2 = np.max(offsets + slopes * m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    x = (lo + hi) * 0.5
    return x

def coord_descent(h, rng, max_sweeps=10000, patience=200, t0=None, budget_s=None, reserve=30):
    n = len(h)
    target = n / 2.0
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    best_c5 = float(np.max(corr) * dx)
    best_h = h.copy()
    stale = 0
    
    for sweep in range(max_sweeps):
        # Check time
        if t0 is not None:
            if time.time() - t0 > budget_s - reserve:
                break
        
        order = rng.permutation(n)
        for j in order:
            slopes = compute_slopes_vec(h, j, n)
            offsets = corr - slopes * h[j]
            x_opt = minimax_1d(offsets, slopes)
            delta = x_opt - h[j]
            if abs(delta) > 1e-14:
                h[j] += delta
                corr += slopes * delta
        
        # Project to maintain sum
        h = project_to_sum(h, target)
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5 = float(np.max(corr) * dx)
        
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
            stale = 0
        else:
            stale += 1
            if stale >= patience:
                break
    
    return best_h, best_c5
