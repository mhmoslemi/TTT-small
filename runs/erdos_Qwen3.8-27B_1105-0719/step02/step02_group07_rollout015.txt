import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time


def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)


def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full').max() * dx


def make_sym_h(q, n):
    m = n // 2
    q = np.clip(q, 0.0, 1.0)
    h = np.empty(n)
    h[:m] = q
    h[m:] = 1.0 - q[::-1]
    return h


def make_q_from_h(h, n):
    m = n // 2
    q = 0.5 * (h[:m] + 1.0 - h[n - 1:n - 1 - m:-1])
    return np.clip(q, 0.0, 1.0)


def corr_vals(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx


def smooth_sym_fun_grad(q, n, beta):
    m = n // 2
    dx = 2.0 / n
    h = make_sym_h(q, n)
    corrs = corr_vals(h)
    obj = logsumexp(beta * corrs) / beta
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    ws = w.sum()
    if ws < 1e-30:
        w = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        w /= ws
    g_h = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        d = k - (n - 1)
        i_lo = max(0, -d)
        i_hi = min(n - 1, n - 1 - d)
        if i_lo > i_hi:
            continue
        idx = np.arange(i_lo, i_hi + 1)
        g_h[idx] += wk * (1.0 - h[idx + d])
        g_h[idx + d] += wk * (-h[idx])
    g_h *= dx
    g_q = g_h[:m] - g_h[n - 1:n - 1 - m:-1]
    return obj, g_q


def lbfgs_smooth_sym(q0, n, beta, maxiter=400):
    m = n // 2

    def fun_and_grad(q):
        return smooth_sym_fun_grad(q, n, beta)

    try:

                       bounds=[(0.0, 1.0)] * m,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14,
                                'maxls': 30, 'maxfun': maxiter * 10})
        h_new = make_sym_h(res.x, n)
        c5 = exact_score(h_new)
        return res.x, c5
    except Exception:
        return None, None


def epi_sym_con_fun(x, n, active_lags):
    t = x[0]
    q = x[1:]
    h = make_sym_h(q, n)
    corrs = corr_vals(h)
    return t - corrs[active_lags]


def epi_sym_con_jac(x, n, active_lags):
    q = x[1:]
    h = make_sym_h(q, n)
    m = n // 2
    dx = 2.0 / n
    L = len(active_lags)
    J = np.zeros((L, m + 1))
    J[:, 0] = 1.0
    for i, lag in enumerate(active_lags):
        d = lag - (n - 1)
        i_lo = max(0, -d)
        i_hi = min(n - 1, n - 1 - d)
        G = np.zeros(n)
        if i_lo <= i_hi:
            idx = np.arange(i_lo, i_hi + 1)
            G[idx] += (1.0 - h[idx + d])
            G[idx + d] += (-h[idx])
        G *= dx
        J[i, 1:] = -(G[:m] - G[n - 1:n - 1 - m:-1])
    return J


def epi_sym_fun(x):
    return x[0]


def epi_sym_jac(x):
    m = len(x) - 1
    g = np.zeros(m + 1)
    g[0] = 1.0
    return g


def epi_sym_slsqp(q0, n, active_lags, t_init, maxiter=400):
    m = n // 2
    L = len(active_lags)
    if L == 0:
        return None, None

    def con_fun(x):
        return epi_sym_con_fun(x, n, active_lags)

    def con_jac(x):
        return epi_sym_con_jac(x, n, active_lags)

    x0 = np.concatenate([[t_init], q0])
    cons = [{'type': 'ineq', 'fun': con_fun, 'jac': con_jac}]
    bounds = [(0.0, 1.0)] + [(0.0, 1.0)] * m

    try:
        res = minimize(epi_sym_fun, x0, jac=epi_sym_jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        h_new = make_sym_h(res.x[1:], n)
        c5 = exact_score(h_new)
        return res.x[1:], c5
    except Exception:
        return None, None

            q_best = q_new
            h_best = make_sym_h(q_best, n)
            print(f"  LBFGS beta={beta}: C5={c5_best:.7f}")

    # Phase 2: Active-set epigraph refinement (multiple rounds)
    for rnd in range(15):
        if remaining() < 15:
            break
        h_cur = make_sym_h(q_best, n)
        corrs = corr_vals(h_cur)
        max_c = corrs.max()
        margin = max(0.0005, 0.003 * (1.0 - rnd / 15.0))
        active = get_active_lags(corrs, 40, max_c, margin)
        q_new, c5_new = epi_sym_slsqp(q_best, n, active, t_init=max_c * 1.0001,
                                       maxiter=500)
        if q_new is not None and c5_new < c5_best - 1e-10:
            c5_best = c5_new
            q_best = q_new
            h_best = make_sym_h(q_best, n)
            print(f"  epi sym r{rnd} (L={len(active)}): C5={c5_best:.7f}")

    # Phase 3: Higher beta L-BFGS-B after active set
    for beta in [3000, 6000]:
        if remaining() < 15:
            break
        q_new, c5_new = lbfgs_smooth_sym(q_best, n, beta, maxiter=300)
        if q_new is not None and c5_new < c5_best - 1e-10:
            c5_best = c5_new
            q_best = q_new
            h_best = make_sym_h(q_best, n)
            print(f"  LBFGS2 beta={beta}: C5={c5_best:.7f}")

    # Phase 4: Full-space polish (breaks symmetry slightly)
    if remaining() > 30:
        h_full = make_sym_h(q_best, n)
        for beta in [500, 1500]:
            if remaining() < 15:
                break
            h_new, c5_new = full_space_smooth(h_full, n, beta, maxiter=300)
            if c5_new < c5_best - 1e-10:
                c5_best = c5_new
                h_best = h_new
                h_full = h_new
                print(f"  full beta={beta}: C5={c5_best:.7f}")

    # Phase 5: Perturbation + re-optimize (diverse scales)
    n_trials = 80
    for trial in range(n_trials):
        if remaining() < 8:
            break
        if trial < 30:
            scale = 0.01 + 0.04 * (trial / 30.0)
        elif trial < 60:
            scale = 0.05 + 0.10 * ((trial - 30) / 30.0)
        else:
            scale = 0.15 + 0.25 * ((trial - 60) / 30.0)
        q_pert = q_best + scale * rng.randn(m)
        q_pert = np.clip(q_pert, 0.0, 1.0)

        q_new, c5_new = lbfgs_smooth_sym(q_pert, n, 150, maxiter=150)
        if q_new is not None and c5_new < c5_best + 0.0005:
            if remaining() < 8:
                break
            q_new2, c5_new2 = lbfgs_smooth_sym(q_new, n, 600, maxiter=300)
            if q_new2 is not None and c5_new2 < c5_best - 1e-10:
                c5_best = c5_new2
                q_best = q_new2
                h_best = make_sym_h(q_best, n)
                print(f"  trial {trial} (s={scale:.3f}): C5={c5_best:.7f}")
                if remaining() > 12:
                    h_cur = make_sym_h(q_best, n)
                    corrs = corr_vals(h_cur)
                    max_c = corrs.max()
                    active = get_active_lags(corrs, 30, max_c, 0.001)
                    q_ep, c5_ep = epi_sym_slsqp(q_best, n, active,
                                                 t_init=max_c * 1.0001, maxiter=300)
                    if q_ep is not None and c5_ep < c5_best - 1e-10:
                        c5_best = c5_ep
                        q_best = q_ep
                        h_best = make_sym_h(q_best, n)
                        print(f"  trial {trial} epi: C5={c5_best:.7f}")

    # Phase 6: Resolution sweep
    for delta in [2, -2, 4, -4, 6, -6, 8, -8, 10, -10, 12, -12]:
        if remaining() < 15:
            break
        n_try = n + delta
        if n_try < 40 or n_try % 2 != 0:
            continue
        x_old = np.linspace(0, 2, n, endpoint=False)
        x_new_grid = np.linspace(0, 2, n_try, endpoint=False)
        h_interp = np.interp(x_new_grid, x_old, h_best)
        h_interp = project_feasible(h_interp, n_try / 2.0)
        m_try = n_try // 2
        q_try = make_q_from_h(h_interp, n_try)

        q_new, c5_new = lbfgs_smooth_sym(q_try, n_try, 200, maxiter=300)
        if q_new is not None and remaining() > 10:
            q_new, c5_new = lbfgs_smooth_sym(q_new, n_try, 800, maxiter=300)
        if q_new is not None and remaining() > 10:
            h_cur = make_sym_h(q_new, n_try)
            corrs = corr_vals(h_cur)
            max_c = corrs.max()
            active = get_active_lags(corrs, 30, max_c, 0.002)
            q_ep, c5_ep = epi_sym_slsqp(q_new, n_try, active,
                                         t_init=max_c * 1.0001, maxiter=300)
            if q_ep is not None and c5_ep < c5_new:
                c5_new = c5_ep
                q_new = q_ep
        if q_new is not None and c5_new < c5_best - 1e-9:
            c5_best = c5_new
            n = n_try
            m = m_try
            q_best = q_new
            h_best = make_sym_h(q_best, n)
            print(f"  n={n}: C5={c5_best:.7f}")

    # Phase 7: Final epigraph with many lags
    if remaining() > 20:
        h_cur = make_sym_h(q_best, n)
        corrs = corr_vals(h_cur)
        max_c = corrs.max()
        active = get_active_lags(corrs, 80, max_c, 0.001)
        q_new, c5_new = epi_sym_slsqp(q_best, n, active,
                                       t_init=max_c * 1.0001, maxiter=600)
        if q_new is not None and c5_new < c5_best - 1e-10:
            c5_best = c5_new
            q_best = q_new
            h_best = make_sym_h(q_best, n)
            print(f"  final epi: C5={c5_best:.7f}")

    # Phase 8: One more L-BFGS-B at very high beta
    if remaining() > 15:
        q_new, c5_new = lbfgs_smooth_sym(q_best, n, 10000, maxiter=200)
        if q_new is not None and c5_new < c5_best - 1e-10:
            c5_best = c5_new
            q_best = q_new
            h_best = make_sym_h(q_best, n)
            print(f"  final LBFGS: C5={c5_best:.7f}")

    # Final validation
    h_best = np.clip(h_best, 0.0, 1.0)
    h_best = project_feasible(h_best, len(h_best) / 2.0)
    c5_best = exact_score(h_best)
    print(f"Final: n={len(h_best)}, C5={c5_best:.7f}")

    return h_best, c5_best, len(h_best)
