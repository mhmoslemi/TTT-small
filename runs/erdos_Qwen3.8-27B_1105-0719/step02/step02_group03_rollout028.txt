import numpy as np
from scipy.optimize import minimize, linprog

def project_feasible(x, target_sum):
    # Binary search for the right shift
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    # Fine adjustment
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
        else:
            h += diff / n
    return np.clip(h, 0.0, 1.0)

def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def c5_of_h(h, dx):
    return compute_corrs(h).max() * dx

    grad = np.zeros(n)
    # out[k] = sum_n h[n+k]*(1-h[n]) for valid n
    # ∂/∂h[i]:

    #   -h[i+k] if 0 <= i+k < n [h[i] as h[n], n=i]

    if k >= 0:

        idx = np.arange(k, n)
        grad[idx] += (1.0 - h[idx - k])
    else:

        # i-k < n means i < n+k
        idx = np.arange(0, n + k)  # n+k < n since k < 0
