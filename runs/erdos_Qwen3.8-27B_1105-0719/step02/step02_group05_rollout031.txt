    gr = (np.sqrt(5) - 1) / 2  # golden ratio conjugate ≈ 0.618

    return float(corrs.max() * dx), corrs

def build_h_from_q(q, n):

def project_feasible(x, target_sum):

        else:

class _SmoothQObj:
    def __init__(self, n, beta):
        self.n = n
        self.m = n // 2
        self.dx = 2.0 / n
        self.beta = beta
    def value(self, q):
        h = build_h_from_q(q, self.n)
        corrs = np.correlate(h, 1.0 - h, mode='full')
        b = self.beta
        mx = corrs.max()
        shifted = corrs - mx
        log_s = np.log(np.exp(b * shifted).sum())
        return (mx + log_s / b) * self.dx
    def gradient(self, q):
        h = build_h_from_q(q, self.n)
        corrs = np.correlate(h, 1.0 - h, mode='full')
        b = self.beta
        mx = corrs.max()
        shifted = corrs - mx
        w = np.exp(b * shifted)
        ws = w.sum()
        if ws < 1e-30:
            w = np.ones(len(corrs)) / len(corrs)
        else:
            w /= ws
        n = self.n
        m = self.m
        G = np.zeros(n)
        for k in range(len(corrs)):
            if w[k] < 1e-14:
                continue
            d = k - (n - 1)
            if d >= 0 and d < n:
                G[d:] += (1.0 - h[:n - d])
                G[:n - d] += (-h[d:])
            elif d < 0 and -d < n:
                dd = -d
                G[:n - dd] += (1.0 - h[dd:])
                G[dd:] += (-h[:n - dd])
        return (G[:m] - G[m:][::-1]) * self.dx
