        g = compute_gradient(h_cur, w, n)

        sources = np.where((h_cur > 1e-8) & (g > np.median(g)))[0]
        sinks = np.where((h_cur < 1.0 - 1e-8) & (g < np.median(g)))[0]

        if len(sources) == 0 or len(sinks) == 0:
            break

        src_sample = sources[np.random.choice(len(sources), min(30, len(sources)), replace=False)]
        sink_sample = sinks[np.random.choice(len(sinks), min(30, len(sinks)), replace=False)]

        best_impr = 1e-15
        best_move = None

        for i in src_sample:
            for j in sink_sample:
                if i == j:
                    continue
                t_max = min(h_cur[i], 1.0 - h_cur[j])
                if t_max < 1e-12:
                    continue
                delta = compute_delta(h_cur, i, j, n)
                t_opt, val = optimal_t_corrs(corrs, delta, t_max)
                if t_opt > 1e-10 and val * dx < c5_cur:
                    impr = c5_cur - val * dx
                    if impr > best_impr:
                        best_impr = impr
                        best_move = (i, j, t_opt)

        if best_move is None:
            for _ in range(20):
                i = np.random.choice(np.where(h_cur > 1e-8)[0])
                j = np.random.choice(np.where(h_cur < 1.0 - 1e-8)[0])
                if i == j:
                    continue
                t_max = min(h_cur[i], 1.0 - h_cur[j])
                if t_max < 1e-12:
                    continue
                delta = compute_delta(h_cur, i, j, n)
                t_opt, val = optimal_t_corrs(corrs, delta, t_max)
                if t_opt > 1e-10 and val * dx < c5_cur:
                    impr = c5_cur - val * dx
                    if impr > best_impr:
                        best_impr = impr
                        best_move = (i, j, t_opt)

        if best_move is None:
            continue

        i, j, t = best_move
        h_cur[i] -= t
        h_cur[j] += t
        h_cur = np.clip(h_cur, 0.0, 1.0)
        h_cur = project_feasible(h_cur, target_sum)

    return best_h, best_c5


def quantized_search(h_init, dx, target_sum, M, max_moves, t0, time_limit, rng):
    n = len(h_init)
    q = np.round(h_init * M).astype(int)
    q = np.clip(q, 0, M)
    target_q = int(round(target_sum * M))
    diff = target_q - q.sum()
    if diff > 0:
        idx = np.where(q < M)[0]
        rng.shuffle(idx)
        for k in range(min(diff, len(idx))):
            q[idx[k]] += 1
    elif diff < 0:
        idx = np.where(q > 0)[0]
        rng.shuffle(idx)
        for k in range(min(-diff, len(idx))):
            q[idx[k]] -= 1
    q = np.clip(q, 0, M)
    h = q.astype(float) / M
    h = project_feasible(h, target_sum)
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    q_best = q.copy()
    current_c5 = best_c5
    temp = 0.001

    for mv in range(max_moves):
        if time.time() - t0 > time_limit:
            break
        nonzero = np.where(q > 0)[0]
        nonfull = np.where(q < M)[0]
        if len(nonzero) == 0 or len(nonfull) == 0:
            break
        i = nonzero[rng.randint(len(nonzero))]
        j = nonfull[rng.randint(len(nonfull))]
        if i == j:
            continue
        q_new = q.copy()
        q_new[i] -= 1
        q_new[j] += 1
        h_new = q_new.astype(float) / M
        c5_new = c5_of_h(h_new, dx)
        delta_c5 = c5_new - current_c5
        if delta_c5 < 0 or rng.random() < np.exp(-delta_c5 / max(temp, 1e-10)):
            q = q_new
            h = h_new
            current_c5 = c5_new
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
                q_best = q.copy()
        temp *= 0.9995
        if temp < 1e-6:
            temp = 1e-6

    return best_h, best_c5


def lp_direction(h, dx, target_sum, corrs, n, radius=0.03):
    n_corr = 2 * n - 1
    max_c = corrs.max()
    k_active = max(10, int(0.25 * n_corr))
    top_idx = np.argsort(corrs)

    A_rows = []
    b_rows = []
    for k in range(2 * n - 1):
        if corrs[k] > max_c - 0.01 * max(1.0, max_c):
            offsets = k - (n - 1)
            row = np.zeros(n)
            ip = offsets
            im = -offsets
            for i in range(n):
                contrib = 0.0
                if 0 <= i + offsets < n:
                    contrib += h[i + offsets]
                if 0 <= i - offsets < n:
                    contrib -= (1.0 - h[i - offsets])
                row[i] = contrib
            A_rows.append(row * dx)
            b_rows.append(max_c * dx - corrs[k] * dx - 1e-12)

    if len(A_rows) == 0:
        return None

    A_ub = np.array(A_rows)
    b_ub = np.array(b_rows)
    A_eq = np.zeros((1, n))
    A_eq[0, :] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for i in range(n):
        bounds.append((-h[i] * radius, (1.0 - h[i]) * radius))

    try:
        res = linprog(np.zeros(n), A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 5})
        if res.status == 0:
            h_new = h + res.x
            h_new = project_feasible(h_new, target_sum)
            c5_new = c5_of_h(h_new, dx)
            if c5_new < c5_of_h(h, dx) - 1e-14:
                return h_new
    except Exception:
        pass
    return None


def interpolate_h(h_src, n_src, n_dst):
    x_src = np.linspace(0.0, 2.0, n_src)
    x_dst = np.linspace(0.0, 2.0, n_dst)
    h_dst = np.interp(x_dst, x_src, h_src)
    return h_dst


def remaining_time(t0, budget_s):
    return budget_s - (time.time() - t0)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    reserve = max(10.0, 0.05 * budget_s)
    effective_budget = budget_s - reserve

    best_h = None
    best_c5 = float('inf')

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        n0 = 126
        h0 = np.zeros(n0)
        h0[:n0 // 2] = 1.0

    n_p = len(h0)
    dx_p = 2.0 / n_p
    ts_p = n_p / 2.0
    h0 = project_feasible(h0, ts_p)
    c5_p = c5_of_h(h0, dx_p)
    best_h = h0.copy()
    best_c5 = c5_p

    # Phase 1: Pair transfer on parent
    if remaining_time(t0, budget_s) > reserve + 30:
        time_limit = time.time() + min(remaining_time(t0, budget_s) - reserve - 10, effective_budget * 0.4)
        h1, c5_1 = pair_transfer_optimize(best_h, dx_p, ts_p, 500, time_limit, t0)
        if c5_1 < best_c5:
            best_c5 = c5_1
            best_h = h1

    # Phase 2: Quantized search on best so far
    if remaining_time(t0, budget_s) > reserve + 20:
        n_b = len(best_h)
        dx_b = 2.0 / n_b
        ts_b = n_b / 2.0
        time_limit = time.time() + min(remaining_time(t0, budget_s) - reserve - 8, effective_budget * 0.2)
        h2, c5_2 = quantized_search(best_h, dx_b, ts_b, 4, 30000, t0, time_limit, rng)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2

    # Phase 3: Pair transfer again on potentially improved solution
    if remaining_time(t0, budget_s) > reserve + 15:
        n_b = len(best_h)
        dx_b = 2.0 / n_b
        ts_b = n_b / 2.0
        time_limit = time.time() + min(remaining_time(t0, budget_s) - reserve - 5, effective_budget * 0.2)
        h3, c5_3 = pair_transfer_optimize(best_h, dx_b, ts_b, 400, time_limit, t0)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3

    # Phase 4: LP direction on best
    if remaining_time(t0, budget_s) > reserve + 10:
        n_b = len(best_h)
        dx_b = 2.0 / n_b
        ts_b = n_b / 2.0
        corrs = np.correlate(best_h, 1.0 - best_h, mode='full')
        h_lp = lp_direction(best_h, dx_b, ts_b, corrs, n_b, radius=0.03)
        if h_lp is not None:
            c5_lp = c5_of_h(h_lp, dx_b)
            if c5_lp < best_c5:
                best_c5 = c5_lp
                best_h = h_lp
                # Follow up with pair transfer
                if remaining_time(t0, budget_s) > reserve + 10:
                    time_limit = time.time() + min(remaining_time(t0, budget_s) - reserve - 5, 60)
                    h3, c5_3 = pair_transfer_optimize(best_h, dx_b, ts_b, 200, time_limit, t0)
                    if c5_3 < best_c5:
                        best_c5 = c5_3
                        best_h = h3

    # Phase 5: Try different n values
    n_candidates = sorted(set([max(40, n_p - 20), max(40, n_p - 10), n_p, n_p + 10, n_p + 20,
                                80, 100, 140]))
    n_candidates = [x for x in n_candidates if 40 <= x <= 180]

    for n_try in n_candidates:
        if remaining_time(t0, budget_s) < reserve + 20:
            break
        if n_try == len(best_h):
            continue
        dx_try = 2.0 / n_try
        ts_try = n_try / 2.0
        h_interp = interpolate_h(best_h, len(best_h), n_try)
        h_interp = project_feasible(h_interp, ts_try)

        time_limit = time.time() + min(remaining_time(t0, budget_s) - reserve - 8, effective_budget * 0.15)
        h_new, c5_new = pair_transfer_optimize(h_interp, dx_try, ts_try, 300, time_limit, t0)

        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

        if remaining_time(t0, budget_s) < reserve + 10:
            break
        # Quick quantized refinement
        time_limit2 = time.time() + min(remaining_time(t0, budget_s) - reserve - 5, effective_budget * 0.08)
        h_q, c5_q = quantized_search(h_new, dx_try, ts_try, 4, 15000, t0, time_limit2, rng)
        if c5_q < best_c5:
            best_c5 = c5_q
            best_h = h_q

    # Phase 6: Final intensive polish
    if remaining_time(t0, budget_s) > reserve + 8:
        n_b = len(best_h)
        dx_b = 2.0 / n_b
        ts_b = n_b / 2.0
        time_limit = time.time() + max(5, remaining_time(t0, budget_s) - reserve - 3)
        h_final, c5_final = pair_transfer_optimize(best_h, dx_b, ts_b, 300, time_limit, t0)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final

    # Final validation
    n_out = len(best_h)
    dx_out = 2.0 / n_out
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_feasible(best_h, n_out / 2.0)
    final_c5 = c5_of_h(best_h, dx_out)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
