    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.ones(len(corrs)) / len(corrs)
    else:
        w /= w_sum
    val = max_c + np.log(w_sum) / beta
    
    grad_h = np.zeros(n_full)
    for k in range(2 * n_full - 1):
        if w[k] < 1e-14:
            continue
        d = k - (n_full - 1)
        i_lo = max(0, -d)
        i_hi = min(n_full - 1, n_full - 1 - d)
        if i_lo > i_hi:
            continue
        idx1 = np.arange(i_lo, i_hi + 1)
        idx2 = idx1 + d
        grad_h[idx1] += w[k] * (1.0 - h[idx2])
        grad_h[idx2] += w[k] * (-h[idx1])
    
    grad_q = np.zeros(m)
    for r in range(m):
        grad_q[r] = grad_h[r] - grad_h[n_full - 1 - r]
    grad_q *= dx
    return val, grad_q

def smooth_sym_optimize(q0, beta_list, maxiter_per_beta, time_limit):
    m = len(q0)
    q = np.clip(q0.copy(), 0, 1)
    h = build_sym_h(q)
    best_score = eval_score(h)
    best_q = q.copy()
    
    for beta in beta_list:
        if time.time() > time_limit:
            break
        bounds = [(0.0, 1.0)] * m
        try:
            res = minimize(
                lambda xx: smooth_max_sym_obj_grad(xx, beta)[0],
                q,
                jac=lambda xx: smooth_max_sym_obj_grad(xx, beta)[1],
                method='L-BFGS-B',
                bounds=bounds,
                options={'maxiter': maxiter_per_beta, 'ftol': 1e-16, 'gtol': 1e-12, 'maxfun': 5000})
            q_new = np.clip(res.x, 0, 1)
        except Exception:
            q_new = q.copy()
        h_new = build_sym_h(q_new)
        sc = eval_score(h_new)
        if sc < best_score:
            best_score = sc
            best_q = q_new.copy()
        q = q_new
        if time.time() > time_limit:
            break
    
    return best_q, best_score, build_sym_h(best_q)

def corr_at_k(h, k):
    n = len(h)
    d = k - (n - 1)
    i_lo = max(0, -d)
    i_hi = min(n - 1, n - 1 - d)
    if i_lo > i_hi:
        return 0.0
    idx1 = np.arange(i_lo, i_hi + 1)
    idx2 = idx1 + d
    return float(np.sum(h[idx1] * (1.0 - h[idx2])))

def select_active(corrs, margin_frac=0.03, margin_abs=2e-6, neighbor_pad=4, max_k=80):
    max_val = corrs.max()
    threshold = max_val - max(margin_abs, margin_frac * max_val)
    active = set(np.where(corrs >= threshold)[0].tolist())
    padded = set()
    for k in active:
        for d in range(-neighbor_pad, neighbor_pad + 1):
            if 0 <= k + d < len(corrs):
                padded.add(k + d)
    active_list = sorted(padded)
    if len(active_list) > max_k:
        scored = sorted([(corrs[k], k) for k in active_list], reverse=True)
        active_list = sorted([k for _, k in scored[:max_k]])
    return active_list

def epig_sym_con_fun(x, k, n_full):
    m = n_full // 2
    q = x[:m]
    t = x[m]
    h = build_sym_h(q)
    dx = 2.0 / n_full
    return t - corr_at_k(h, k) * dx

def epig_sym_con_jac(x, k, n_full):
    m = n_full // 2
    q = x[:m]
    h = build_sym_h(q)
    dx = 2.0 / n_full
    d = k - (n_full - 1)
    i_lo = max(0, -d)
    i_hi = min(n_full - 1, n_full - 1 - d)
    grad_h = np.zeros(n_full)
    if i_lo <= i_hi:
        idx1 = np.arange(i_lo, i_hi + 1)
        idx2 = idx1 + d
        grad_h[idx1] = (1.0 - h[idx2])
        grad_h[idx2] = (-h[idx1])
    j = np.zeros(m + 1)
    for r in range(m):
        j[r] = -dx * (grad_h[r] - grad_h[n_full - 1 - r])
    j[m] = 1.0
    return j

def solve_epigraph_sym(q_center, n_full, active_k, maxiter, ftol):
    m = n_full // 2
    h_c = build_sym_h(q_center)
    dx = 2.0 / n_full
    corrs_c = corr_raw(h_c) * dx
    t0_val = corrs_c.max() + 1e-9
    
    x0 = np.zeros(m + 1)
    x0[:m] = q_center
    x0[m] = t0_val
    
    bounds = [(0.0, 1.0)] * m + [(0.0, 2.0)]
    constraints = []
    for k in active_k:
        constraints.append({'type': 'ineq',
                           'fun': lambda x, kk=k: epig_sym_con_fun(x, kk, n_full),
                           'jac': lambda x, kk=k: epig_sym_con_jac(x, kk, n_full)})
    
    try:
        res = minimize(lambda x: x[m], x0,
                      jac=lambda x: np.concatenate([np.zeros(m), [1.0]]),
                      method='SLSQP', bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': ftol})
        q_out = np.clip(res.x[:m], 0.0, 1.0)
    except Exception:
        q_out = q_center.copy()
    
    h_out = build_sym_h(q_out)
    return q_out, h_out, eval_score(h_out)

def transfer_to_n(h_old, n_new):
    n_old = len(h_old)
    x_old = (np.arange(n_old) + 0.5) * 2.0 / n_old
    x_new = (np.arange(n_new) + 0.5) * 2.0 / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new / 2.0)

def coordinate_descent_polish(q, n_full, max_rounds=100, time_limit=None):
    m = len(q)
    h = build_sym_h(q)
    best_score = eval_score(h)
    dx = 2.0 / n_full
    
    for rnd in range(max_rounds):
        if time_limit is not None and time.time() > time_limit:
            break
        improved = False
        for r in range(m):
            if time_limit is not None and time.time() > time_limit:
                break
            best_local = best_score
            best_qr = q[r]
            for trial in range(7):
                delta = (trial - 3) * 0.05
                if abs(delta) < 1e-10:
                    continue
                q_trial = q.copy()
                q_trial[r] = np.clip(q[r] + delta, 0.0, 1.0)
                h_trial = build_sym_h(q_trial)
                sc = eval_score(h_trial)
                if sc < best_local - 1e-12:
                    best_local = sc
                    best_qr = q_trial[r]
            if best_qr != q[r]:
                q[r] = best_qr
                h = build_sym_h(q)
                best_score = best_local
                improved = True
        if not improved:
            # Try smaller steps
            for r in range(m):
                if time_limit is not None and time.time() > time_limit:
                    break
                best_local = best_score
                best_qr = q[r]
                for step in [0.01, -0.01, 0.02, -0.02, 0.005, -0.005]:
                    q_trial = q.copy()
                    q_trial[r] = np.clip(q[r] + step, 0.0, 1.0)
                    h_trial = build_sym_h(q_trial)
                    sc = eval_score(h_trial)
                    if sc < best_local - 1e-12:
                        best_local = sc
                        best_qr = q_trial[r]
                if best_qr != q[r]:
                    q[r] = best_qr
                    h = build_sym_h(q)
                    best_score = best_local
                    improved = True
            if not improved:
                break
    return q, best_score, build_sym_h(q)

def epig_sym_loop(q, n_full, max_rounds=30, per_solve_maxiter=80, time_limit=None):
    m = n_full // 2
    dx = 2.0 / n_full
    h = build_sym_h(q)
    best_score = eval_score(h)
    best_q = q.copy()
    
    for rnd in range(max_rounds):
        if time_limit is not None and time.time() > time_limit:
            break
        h_chk = build_sym_h(q)
        corrs = corr_raw(h_chk) * dx
        active = select_active(corrs, margin_frac=0.03, neighbor_pad=5, max_k=70)
        
        q_new, h_new, c5_new = solve_epigraph_sym(q, n_full, active, per_solve_maxiter, 1e-12)
        q_new = np.clip(q_new, 0, 1)
        h_new = build_sym_h(q_new)
        c5_new = eval_score(h_new)
        
        if c5_new < best_score - 1e-14:
            best_score = c5_new
            best_q = q_new.copy()
            q = q_new
            continue
        
        corrs_new = corr_raw(h_new) * dx
        max_act = corrs_new[np.array(active)].max() if len(active) > 0 else 0
        hidden = np.where(corrs_new > max_act + 1e-7)[0]
        if len(hidden) > 0:
            active = sorted(set(active + list(hidden)))
            if len(active) > 100:
                active = select_active(corrs_new, margin_frac=0.02, neighbor_pad=3, max_k=100)
            q = q_new
            continue
        
        active2 = select_active(corrs_new, margin_frac=0.08, neighbor_pad=7, max_k=90)
        if len(active2) > len(active):
            q_new2, h_new2, c5_new2 = solve_epigraph_sym(q, n_full, active2, 60, 1e-12)
            q_new2 = np.clip(q_new2, 0, 1)
            h_new2 = build_sym_h(q_new2)
            c5_new2 = eval_score(h_new2)
            if c5_new2 < best_score - 1e-14:
                best_score = c5_new2
                best_q = q_new2.copy()
                q = q_new2
                continue
        break
    
    return best_q, best_score, build_sym_h(best_q)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    
    def rem():
        return budget_s - (time.time() - t0)
    
    best_h = None
    best_c5 = float('inf')
    
    try:
        h_parent = np.array(initial_h_values, dtype=float).copy().reshape(-1)
    except (NameError, AttributeError):
        h_parent = None
    
    if h_parent is not None and len(h_parent) > 10:
        n_p = len(h_parent)
        if n_p % 2 != 0:
            n_p -= 1
            h_parent = h_parent[:n_p]
        target_p = n_p / 2.0
        h_parent = project_feasible(h_parent, target_p)
        best_h = h_parent.copy()
        best_c5 = eval_score(h_parent)
    else:
        m_p = 63
        x = np.linspace(0, 1, m_p)
        q_p = np.clip(0.5 + 0.3 * np.sin(2 * np.pi * x), 0, 1)
        best_h = build_sym_h(q_p)
        best_c5 = eval_score(best_h)
    
    n_cur = len(best_h)
    q_cur = sym_from_h(best_h)
    
    # Phase 1: Multi-stage smooth max with many stages
    if rem() > 200:
        beta_list_1 = [20, 40, 80, 150, 300, 600, 1200, 2500]
        time_limit = time.time() + min(280, rem() * 0.30)
        q_opt, c5_s, h_s = smooth_sym_optimize(q_cur, beta_list_1, 300, time_limit)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()
            q_cur = q_opt
    
    # Phase 2: Epigraph symmetric refinement
    if rem() > 150:
        tl = time.time() + min(250, rem() * 0.30)
        q_opt, c5_e, h_e = epig_sym_loop(q_cur, len(best_h), max_rounds=35, per_solve_maxiter=100, time_limit=tl)
        if c5_e < best_c5:
            best_c5 = c5_e
            best_h = h_e.copy()
            q_cur = q_opt
    
    # Phase 3: Coordinate descent polish
    if rem() > 80:
        tl = time.time() + min(100, rem() * 0.25)
        q_cd, c5_cd, h_cd = coordinate_descent_polish(q_cur, len(best_h), max_rounds=80, time_limit=tl)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
            q_cur = q_cd
    
    # Phase 4: Another round of smooth max + epigraph at higher precision
    if rem() > 100:
        beta_list_2 = [500, 1000, 2000, 4000, 8000]
        tl = time.time() + min(120, rem() * 0.30)
        q_opt, c5_s, h_s = smooth_sym_optimize(q_cur, beta_list_2, 250, tl)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()
            q_cur = q_opt
    
    if rem() > 80:
        tl = time.time() + min(100, rem() * 0.35)
        q_opt, c5_e, h_e = epig_sym_loop(q_cur, len(best_h), max_rounds=30, per_solve_maxiter=80, time_limit=tl)
        if c5_e < best_c5:
            best_c5 = c5_e
            best_h = h_e.copy()
            q_cur = q_opt
    
    # Phase 5: Try nearby resolutions
    n_base = len(best_h)
    n_candidates = sorted(set([n_base - 4, n_base - 2, n_base + 2, n_base + 4, n_base + 6]))
    n_candidates = [nc for nc in n_candidates if 48 <= nc <= 200 and nc % 2 == 0]
    
    for n_try in n_candidates:
        if rem() < 60:
            break
        h_try = transfer_to_n(best_h, n_try)
        if len(h_try) != n_try:
            continue
        q_try = sym_from_h(h_try)
        
        if rem() > 80:
            tl = time.time() + min(60, rem() * 0.35)
            q_o, c5_o, h_o = smooth_sym_optimize(q_try, [100, 300, 800, 2000], 200, tl)
            if c5_o < best_c5:
                best_c5 = c5_o
                best_h = h_o.copy()
                q_cur = q_o
        
        if rem() > 50:
            tl = time.time() + min(40, rem() * 0.35)
            q_o, c5_o, h_o = epig_sym_loop(q_try, n_try, max_rounds=20, per_solve_maxiter=60, time_limit=tl)
            if c5_o < best_c5:
                best_c5 = c5_o
                best_h = h_o.copy()
                q_cur = q_o
    
    # Phase 6: Random restarts (moderate perturbation of best solution)
    n_restarts = min(8, max(2, int(rem() / 70)))
    for i in range(n_restarts):
        if rem() < 60:
            break
        
        n_r = len(best_h)
        q_r = q_cur.copy()
        # Perturb
        noise_scale = 0.05 + 0.03 * i
        q_r += noise_scale * rng.randn(len(q_r))
        q_r = np.clip(q_r, 0, 1)
        # Re-project to maintain symmetry constraint (already symmetric by parameterization)
        
        h_r = build_sym_h(q_r)
        # Check sum
        if abs(h_r.sum() - n_r / 2.0) > 0.5:
            q_r = sym_from_h(project_feasible(h_r, n_r / 2.0))
        
        if rem() > 50:
            tl = time.time() + min(40, rem() * 0.5)
            q_o, c5_o, h_o = smooth_sym_optimize(q_r, [200, 600, 1500], 150, tl)
            if c5_o < best_c5:
                best_c5 = c5_o
                best_h = h_o.copy()
                q_cur = q_o
                if rem() > 30:
                    tl2 = time.time() + min(25, rem() * 0.5)
                    q_o2, c5_o2, h_o2 = epig_sym_loop(q_o, n_r, max_rounds=15, per_solve_maxiter=50, time_limit=tl2)
                    if c5_o2 < best_c5:
                        best_c5 = c5_o2
                        best_h = h_o2.copy()
                        q_cur = q_o2
    
    # Phase 7: Final polish - coordinate descent + one more epigraph
    if rem() > 30 and best_h is not None:
        n_fin = len(best_h)
        q_fin = sym_from_h(best_h)
        
        if rem() > 40:
            tl = time.time() + min(30, rem() * 0.5)
            q_cd, c5_cd, h_cd = coordinate_descent_polish(q_fin, n_fin, max_rounds=60, time_limit=tl)
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd.copy()
                q_fin = q_cd
        
        if rem() > 25:
            h_chk = build_sym_h(q_fin)
            corrs = corr_raw(h_chk) * (2.0 / n_fin)
            active = select_active(corrs, margin_frac=0.01, neighbor_pad=4, max_k=50)
            q_o, h_o, c5_o = solve_epigraph_sym(q_fin, n_fin, active, 100, 1e-13)
            q_o = np.clip(q_o, 0, 1)
            h_o = build_sym_h(q_o)
            c5_o = eval_score(h_o)
            if c5_o < best_c5:
                best_c5 = c5_o
                best_h = h_o.copy()
    
    # Validate and return
    if best_h is None:
        m_fb = 63
        x_fb = np.linspace(0, 1, m_fb)
        q_fb = np.clip(0.5 + 0.3 * np.sin(2 * np.pi * x_fb), 0, 1)
        best_h = build_sym_h(q_fb)
    
    best_h = np.asarray(best_h, dtype=float).reshape(-1)
    n_out = len(best_h)
    best_h = project_feasible(best_h, n_out / 2.0)
    best_c5 = eval_score(best_h)
    
    return best_h, best_c5, n_out
