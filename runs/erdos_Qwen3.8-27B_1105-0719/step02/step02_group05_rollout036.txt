Actually let me reconsider. corr[d] = sum_i h[i] * (1-h[i+d]). Each term is h[i] - h[i]*h[i+d]. If one of h[i], h[i+d] is q[i] (or 1-q[something for the symmetric partner]), the term involves q[i] linearly or quadratically.

Let's be more precise. h[i] for i < m is q[i]. h[i] for i >= m is 1 - q[m-1-(i-m)] = 1 - q[n-1-i].

So h[i] = q[i] if i < m, and h[i] = 1 - q[n-1-i] if i >= m.

For corr[d] = sum_{i=0}^{n-1-d} h[i] * (1-h[i+d]) (for d >= 0):
Each term h[i]*(1-h[i+d]) can involve:
- q[a]*(1-q[b]) where a, b might be the same or different
- q[a]*(1-(1-q[b])) = q[a]*q[b]
- (1-q[a])*(1-q[b])
- (1-q[a])*q[b]

So each corr[d] is a quadratic polynomial in the q variables. In particular, as a function of a single q[i] (with all others fixed), each corr[d] is at most a quadratic (and actually at most linear in q[i] plus a constant, since the quadratic term would be q[i]*q[i] = q[i]² which can appear).

Wait, can q[i] appear twice in the same term? h[i]*(1-h[i+d]). If i < m and i+d < m, then it's q[i]*(1-q[i+d]) which is linear in q[i]. If i < m and i+d >= m, then h[i+d] = 1-q[n-1-i-d], so the term is q[i]*q[n-1-i-d], which is linear in q[i]. Similarly for other cases.

So each corr[d] is actually LINEAR in each individual q[i]! That's great for coordinate descent.

So f(q[i]) = max_d (a_d * q[i] + b_d) where a_d and b_d are constants (given all other q's fixed). This is a max of linear functions, which is a convex piecewise-linear function of q[i]. The minimum of a convex piecewise-linear function on [0,1] occurs at either a breakpoint or at an endpoint.

The breakpoints occur where two of the linear functions cross: a_d * q[i] + b_d = a_e * q[i] + b_e, i.e., q[i] = (b_e - b_d) / (a_d - a_e).

So for coordinate descent, for each coordinate i:
1. Compute all a_d and b_d (the coefficients of q[i] in each corr[d])
2. Find all breakpoints in [0, 1]
3. Evaluate f at all breakpoints and at 0 and 1
4. Choose the best value

This is exact and fast! The number of corr terms is 2n-1, so there are at most (2n-1) linear functions, giving at most (2n-1)² breakpoints (but we only need those in [0,1]).

Actually, we can do this more efficiently. The max of linear functions is minimized at a point where either:
- Two constraints are tight (a breakpoint), or
- A breakpoint, or endpoint.

Actually, even simpler: we can just do a 1D golden-section-like search on the convex PL function. Since it's convex, we can use bisection on the subgradient. The subgradient at q is the set of slopes a_d that achieve the max. If the max subgradient > 0, move left; if < 0, move right; if 0 is in the subgradient (i.e., some a_d > 0 and some a_d < 0 achieve the max, or some a_d = 0 achieves the max), we're at the minimum.

def smooth_obj_grad_q(q, n, beta):
    m = n // 2
    h = build_h_from_q(q, n)
    dx = 2.0 / n
    corrs = compute_corrs(h)
    
    val = logsumexp(beta * corrs) / beta * dx
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(len(corrs)) / len(corrs)
    else:
        w /= w_sum
    
    # Gradient w.r.t. h
    # d(corr[k])/dh[j] = (1-h[j-d_k]) * I(0<=j-d_k<n) - h[j+d_k] * I(0<=j+d_k<n)
    # where d_k = k - (n-1)
    
    Gh = np.zeros(n)
    for k in range(len(corrs)):
        if w[k] < 1e-14:
            continue
        d = k - (n - 1)
        # (1-h[j-d]) for j where 0<=j-d<n, i.e., d<=j<n+d
        # This is: for j in [max(0,d), min(n, n+d)), add (1-h[j-d]) * w[k]
        # -h[j+d] for j where 0<=j+d<n, i.e., -d<=j<n-d
        # This is: for j in [max(0,-d), min(n, n-d)), add -h[j+d] * w[k]
        
        if d >= 0:
            # j-d in [0, n-1] means j in [d, n-1+d] intersect [0, n-1] = [d, n-1]
            Gj1 = np.zeros(n)
            Gj1[d:] = (1.0 - h[:n-d])
            # j+d in [0, n-1] means j in [-d, n-1-d] intersect [0,n-1] = [0, n-1-d]
            Gj2 = np.zeros(n)
            Gj2[:n-d] = -h[d:]
        else:
            dd = -d
            # j-d = j+dd in [0,n-1] means j in [-dd, n-1-dd] intersect [0,n-1] = [0, n-1-dd]
            Gj1 = np.zeros(n)
            Gj1[:n-dd] = (1.0 - h[dd:])
            # j+d = j-dd in [0,n-1] means j in [dd, n-1+dd] intersect [0,n-1] = [dd, n-1]
            Gj2 = np.zeros(n)
            Gj2[dd:] = -h[:n-dd]
        
        Gh += w[k] * (Gj1 + Gj2)
    
    Gh *= dx
    
    # Project to q-space
    Gq = Gh[:m] - Gh[m:][::-1]
    
    return val, Gq

    m = n // 2
    q = 0.5 * (h[:m] + 1.0 - h[m:][::-1])
    return np.clip(q, 0.0, 1.0)

    """Exact 1D optimization of q[i]."""
    q_save = q[i]
    q[i] = 0.0; h_a = build_h_from_q(q, n)
    q[i] = 0.5; h_b = build_h_from_q(q, n)
    q[i] = 1.0; h_c = build_h_from_q(q, n)
    q[i] = q_save
    
    ca = compute_corrs(h_a)
    cb = compute_corrs(h_b)
    cc = compute_corrs(h_c)
    
    A = 2.0 * (cb - 0.5*(ca + cc))
    B = (cc - ca) - A
    C = ca

            best_t = t

import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time


def build_h_from_q(q, n):
    m = n // 2
    return np.concatenate([q, 1.0 - q[::-1]])


def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def exact_score(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full').max() * (2.0 / n)


def project_to_q(h, n):
    m = n // 2
    q = 0.5 * (h[:m] + 1.0 - h[m:][::-1])
    return np.clip(q, 0.0, 1.0)


def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)


def coord_optimize(q, i, n):
    q_save = q[i]
    q[i] = 0.0
    h_a = build_h_from_q(q, n)
    q[i] = 0.5
    h_b = build_h_from_q(q, n)
    q[i] = 1.0
    h_c = build_h_from_q(q, n)
    q[i] = q_save

    ca = compute_corrs(h_a)
    cb = compute_corrs(h_b)
    cc = compute_corrs(h_c)

    A = 2.0 * (cb - 0.5 * (ca + cc))
    B = (cc - ca) - A
    C = ca

    best_f = np.inf
    best_t = q_save

    t_grid = np.linspace(0, 1, 51)
    for t in t_grid:
        f = (A * t * t + B * t + C).max()
        if f < best_f:
            best_f = f
            best_t = t

    mask = np.abs(A) > 1e-12
    if mask.any():
        tv = -B[mask] / (2.0 * A[mask])
        tv = tv[(tv >= 0.0) & (tv <= 1.0)]
        for t in tv:
            f = (A * t * t + B * t + C).max()
            if f < best_f:
                best_f = f
                best_t = t

    cur_f = A * q_save * q_save + B * q_save + C
    n_top = min(12, len(cur_f))
    top_idx = np.argsort(cur_f)[::-1][:n_top]
    for ii in range(n_top):
        for jj in range(ii + 1, n_top):
            ki, kj = top_idx[ii], top_idx[jj]
            da = A[ki] - A[kj]
            db = B[ki] - B[kj]
            dc = C[ki] - C[kj]
            if abs(da) > 1e-14:
                disc = db * db - 4.0 * da * dc
                if disc >= 0:
                    sq = np.sqrt(disc)
                    t1 = (-db + sq) / (2.0 * da)
                    t2 = (-db - sq) / (2.0 * da)
                    for t in [t1, t2]:
                        if 0.0 <= t <= 1.0:
                            f = (A * t * t + B * t + C).max()
                            if f < best_f:
                                best_f = f
                                best_t = t
            elif abs(db) > 1e-14:
                t = -dc / db
                if 0.0 <= t <= 1.0:
                    f = (A * t * t + B * t + C).max()
                    if f < best_f:
                        best_f = f
                        best_t = t

    lo = max(0.0, best_t - 0.005)
    hi = min(1.0, best_t + 0.005)
    for t in np.linspace(lo, hi, 41):
        f = (A * t * t + B * t + C).max()
        if f < best_f:
            best_f = f
            best_t = t

    return best_t


def coordinate_descent(q, n, n_passes=5, rng=None):
    m = n // 2
    best_score = np.inf

    for p in range(n_passes):
        if rng is not None:
            order = rng.permutation(m)
        else:
            order = range(m)

        for i in order:
            new_t = coord_optimize(q, i, n)
            if abs(new_t - q[i]) > 1e-14:
                q[i] = new_t

        h = build_h_from_q(q, n)
        score = exact_score(h)
        if score < best_score:
            best_score = score

    h = build_h_from_q(q, n)
    return q, exact_score(h)


def smooth_obj_q(q, n, beta):
    h = build_h_from_q(q, n)
    dx = 2.0 / n
    corrs = compute_corrs(h)
    return logsumexp(beta * corrs) / beta * dx


def smooth_grad_q(q, n, beta):
    m = n // 2
    h = build_h_from_q(q, n)
    dx = 2.0 / n
    corrs = compute_corrs(h)
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    ws = w.sum()
    if ws < 1e-30:
        w = np.ones(len(corrs)) / len(corrs)
    else:
        w /= ws

    Gh = np.zeros(n)
    for k in range(len(corrs)):
        if w[k] < 1e-14:
            continue
        d = k - (n - 1)
        if d >= 0:
            Gh[d:] += w[k] * (1.0 - h[:n - d])
            Gh[:n - d] += w[k] * (-h[d:])
        else:
            dd = -d
            Gh[:n - dd] += w[k] * (1.0 - h[dd:])
            Gh[dd:] += w[k] * (-h[:n - dd])
    Gh *= dx

    Gq = Gh[:m] - Gh[m:][::-1]
    return Gq


def smooth_optimize_q(q, n, beta, maxiter=200):
    m = n // 2
    try:
        res = minimize(smooth_obj_q, q.copy(), args=(n, beta),
                       jac=lambda x: smooth_grad_q(x, n, beta),
                       method='L-BFGS-B', bounds=[(0.0, 1.0)] * m,
                       options={'maxiter': maxiter, 'ftol': 1e-16,
                                'gtol': 1e-10, 'maxls': 40})
        return np.clip(res.x, 0.0, 1.0)
    except Exception:
        return q.copy()


def interpolate_h(h, n_new):
    x_old = np.arange(len(h)) / len(h)
    x_new = np.arange(n_new) / n_new
    return np.interp(x_new, x_old, h)


def multi_start_coordinate(q0, n, rng, n_starts=4, pert_scale=0.03):
    m = n // 2
    best_q = q0.copy()
    best_score = exact_score(build_h_from_q(q0, n))

    for s in range(n_starts):
        if s == 0:
            q_try = q0.copy()
        else:
            q_try = q0 + pert_scale * rng.randn(m)
            q_try = np.clip(q_try, 0.0, 1.0)

        q_try, score = coordinate_descent(q_try, n, n_passes=8, rng=rng)
        if score < best_score:
            best_score = score
            best_q = q_try.copy()

    return best_q, best_score


def active_set_epigraph(q, n, n_active=20, maxiter=150):
    m = n // 2
    h = build_h_from_q(q, n)
    dx = 2.0 / n
    corrs = compute_corrs(h)
    current_max = corrs.max() * dx

    top_idx = np.argsort(corrs)[::-1][:n_active]
    active_k = sorted(top_idx.tolist())

    class ObjFn:
        def __init__(self, n, active_k):
            self.n = n
            self.active_k = active_k
            self.dx = 2.0 / n
            self.m = n // 2

        def obj(self, x):
            return x[0]

        def obj_grad(self, x):
            g = np.zeros(self.m + 1)
            g[0] = 1.0
            return g

        def constraint(self, x, k_idx):
            t = x[0]
            q = x[1:]
            h = build_h_from_q(q, self.n)
            k = self.active_k[k_idx]
            d = k - (self.n - 1)
            n = self.n
            if d >= 0:
                i_vals = np.arange(0, n - d)
                val = np.sum(h[i_vals + d] * (1.0 - h[i_vals]))
            else:
                dd = -d
                i_vals = np.arange(dd, n)
                val = np.sum(h[i_vals] * (1.0 - h[i_vals - dd]))
            return t - val * self.dx

        def constraint_grad(self, x, k_idx):
            q = x[1:]
            h = build_h_from_q(q, self.n)
            k = self.active_k[k_idx]
            d = k - (self.n - 1)
            n = self.n
            m = self.m
            g = np.zeros(m + 1)
            g[0] = 1.0
            if d >= 0:
                i_vals = np.arange(0, n - d)
                G = np.zeros(n)
                G[i_vals + d] += (1.0 - h[i_vals])
                G[i_vals] += (-h[i_vals + d])
            else:
                dd = -d
                i_vals = np.arange(dd, n)
                G = np.zeros(n)
                G[i_vals] += (1.0 - h[i_vals - dd])
                G[i_vals - dd] += (-h[i_vals])
            Gq = G[:m] - G[m:][::-1]
            g[1:] = -Gq * self.dx
            return g

    ep = ObjFn(n, active_k)
    x0 = np.concatenate([[current_max * 1.001 + 1e-6], q])

    cons = []
    for i in range(len(active_k)):
        cons.append({
            'type': 'ineq',
            'fun': lambda x, i=i: ep.constraint(x, i),
            'jac': lambda x, i=i: ep.constraint_grad(x, i)
        })

    bounds = [(0.0, 5.0)] + [(0.0, 1.0)] * m
    try:
        res = minimize(ep.obj, x0, jac=ep.obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        q_out = np.clip(res.x[1:], 0.0, 1.0)
        return q_out
    except Exception:
        return q.copy()
