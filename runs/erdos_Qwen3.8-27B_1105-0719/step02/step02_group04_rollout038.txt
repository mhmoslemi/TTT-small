def eval_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / n))

def project_to_target(x, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(x - mid, 0.0, 1.0).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-10:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.any():
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def build_sym(q, n):

    h = np.empty(n)

    return h

            j2 = j2[mask]
            g[j1] += (1.0 - h[j2])
            g[j2] += (-h[j1])

    # s.t. grad_i · h_new - t <= grad_i · h - cv[i]  for i in active
    #      sum(h_new) = n/2
    #      0 <= h_new <= 1
    
    # x = [h_new (n), t (1)]

    c_obj[n] = 1.0  # minimize t
    
    A_ub = np.zeros((n_act, n + 1))
    b_ub = np.zeros(n_act)
    for i in range(n_act):
        A_ub[i, :n] = grads[i]
        A_ub[i, n] = -1.0
        b_ub[i] = grads[i] @ h - cv[active_k[i]]

        gq = np.zeros(m)

                    j2 = j2[mask]

                    g[j2] += (-h[j1])

def corr_grad(d, h, N):
    """Gradient of c(d) with respect to h, where c(d) = sum_n h[n+d]*(1-h[n])."""
    g = np.zeros(N)
    if d >= 0:
        # First term: (1-h[j-d]) for j in [d, N-1]
        for j in range(d, N):
            g[j] += (1.0 - h[j-d])
        # Second term: -h[j+d] for j in [0, N-1-d]

Variables: q = h[0:N/2] (N/2 free variables), t
Constraints:
- h[i] + h[N-1-i] = 1 (implicit in parameterization)
- ∇c_k · h(q) - t ≤ b_k for each active k
- 0 ≤ q[i] ≤ 1 (which implies 0 ≤ h[N-1-i] = 1-q[i] ≤ 1)

The gradient ∇c_k · h(q) needs to be expressed in terms of q. Since h = [q, 1-q[::-1]], we have:
h[j] = q[j] for j < N/2
h[j] = 1 - q[N-1-j] for j ≥ N/2

So ∇c_k · h = sum_{j<N/2} ∇c_k[j] * q[j] + sum_{j≥N/2} ∇c_k[j] * (1-q[N-1-j])
= sum_{j<N/2} ∇c_k[j] * q[j] + sum_{j≥N/2} ∇c_k[j] - sum_{j≥N/2} ∇c_k[j] * q[N-1-j]
= sum_{j<N/2} ∇c_k[j] * q[j] - sum_{j≥N/2} ∇c_k[j] * q[N-1-j] + sum_{j≥N/2} ∇c_k[j]

Let me define:
gq_k[j] = ∇c_k[j] - ∇c_k[N-1-j] for j = 0, ..., N/2-1
const_k = sum_{j≥N/2} ∇c_k[j]

Then: ∇c_k · h = gq_k · q + const_k

Similarly, c(k) evaluated at the current h can be computed directly.

The LP becomes:
min t
s.t. gq_k · q - t ≤ (∇c_k · h_current) - c(k) for each active k
     0 ≤ q ≤ 1

With x = [q; t]:
min [0,...,0,1]^T x
s.t. [gq_k, -1] x ≤ (∇c_k · h) - c(k) for each k
     -1 ≤ x[i] ≤ 0 for i < N/2 (i.e., 0 ≤ q ≤ 1)
     t unbounded

import numpy as np
from scipy.optimize import minimize, linprog
import time


def eval_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / n))


def project_to_target(x, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(x - mid, 0.0, 1.0).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-10:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.any():
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)


def build_sym(q, n):
    m = n // 2
    h = np.empty(n)
    h[:m] = q
    h[m:] = 1.0 - q[::-1]
    return h


def to_sym_q(h, n):
    m = n // 2
    return 0.5 * (h[:m] + 1.0 - h[n - 1:m - 1:-1])


def corr_grad(d, h, N):
    g = np.zeros(N)
    if d >= 0:
        g[d:] += (1.0 - h[:N - d])
        g[:N - d] += (-h[d:])
    else:
        dd = -d
        g[:N - dd] += (1.0 - h[dd:])
        g[dd:] += (-h[:N - dd])
    return g


def select_active(h, N, n_top=30, margin_frac=0.02):
    cv = np.correlate(h, 1.0 - h, mode='full') * (2.0 / N)
    mx = cv.max()
    thresh = mx - margin_frac * mx
    idx = np.where(cv >= thresh)[0]
    if len(idx) > n_top:
        vals = cv[idx]
        order = np.argsort(vals)[::-1]
        idx = idx[order[:n_top]]
    return np.sort(idx)


def lp_step(h, N, active_k):
    grads = np.array([corr_grad(k - (N - 1), h, N) for k in active_k])
    n_act = len(active_k)
    cv = np.correlate(h, 1.0 - h, mode='full') * (2.0 / N)

    c_obj = np.zeros(N + 1)
    c_obj[N] = 1.0
    A_ub = np.zeros((n_act, N + 1))
    b_ub = np.zeros(n_act)
    for i in range(n_act):
        A_ub[i, :N] = grads[i]
        A_ub[i, N] = -1.0
        b_ub[i] = grads[i] @ h - cv[active_k[i]]

    A_eq = np.zeros((1, N + 1))
    A_eq[0, :N] = 1.0
    b_eq = np.array([N / 2.0])
    bounds = [(0.0, 1.0)] * N + [(None, None)]

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
    except Exception:
        return None
    if res.status != 0:
        return None

    h_lp = np.clip(res.x[:N], 0.0, 1.0)
    best_c5 = eval_c5(h)
    best_h = h.copy()
    for alpha in [1.0, 0.7, 0.5, 0.35, 0.25, 0.15, 0.1, 0.05]:
        h_trial = h + alpha * (h_lp - h)
        h_trial = project_to_target(h_trial, N / 2.0)
        c5_trial = eval_c5(h_trial)
        if c5_trial < best_c5:
            best_c5 = c5_trial
            best_h = h_trial.copy()
    return best_h


def subgrad_step(h, N, active_k):
    cv = np.correlate(h, 1.0 - h, mode='full') * (2.0 / N)
    act_cv = cv[active_k]
    mx = act_cv.max()
    weights = np.maximum(0.0, act_cv - (mx - 0.003))
    if weights.sum() < 1e-10:
        weights = np.ones(len(active_k)) / len(active_k)
    else:
        weights /= weights.sum()
    g_dir = sum(w * corr_grad(k - (N - 1), h, N) for w, k in zip(weights, active_k))
    best_c5 = eval_c5(h)
    best_h = h.copy()
    for alpha in [0.02, 0.01, 0.005, 0.002, 0.001]:
        h_trial = h - alpha * g_dir
        h_trial = project_to_target(h_trial, N / 2.0)
        c5_trial = eval_c5(h_trial)
        if c5_trial < best_c5:
            best_c5 = c5_trial
            best_h = h_trial.copy()
    return best_h


def opt_sym_lbfgs(q0, N, active_k, mu, maxiter=200, gtol=1e-11):
    m = N // 2

    def fg(z):
        q = z[:m]
        t = z[m]
        h = build_sym(q, N)
        cv = np.correlate(h, 1.0 - h, mode='full') * (2.0 / N)
        vals = cv[active_k]
        exc = np.maximum(0.0, vals - t)
        f = t + 0.5 * mu * np.dot(exc, exc)
        gf = np.zeros(m + 1)
        gf[m] = 1.0 - mu * exc.sum()
        gq = np.zeros(m)
        for i in range(len(active_k)):
            if exc[i] > 0.0:
                d = active_k[i] - (N - 1)
                g = corr_grad(d, h, N)
                gq += mu * exc[i] * (g[:m] - g[N - 1:m - 1:-1])
        gf[:m] = gq
        return f, gf

    h0 = build_sym(q0, N)
    t0 = eval_c5(h0)
    z0 = np.concatenate([q0, [t0]])
    bounds = [(0.0, 1.0)] * m + [(0.0, t0 + 0.5)]
    try:
        res = minimize(fg, z0, jac=True, method='L-BFGS-B',
                      bounds=bounds, options={'maxiter': maxiter, 'gtol': gtol})
        q_opt = np.clip(res.x[:m], 0.0, 1.0)
    except Exception:
        return q0, eval_c5(h0)
    h_opt = build_sym(q_opt, N)
    return q_opt, eval_c5(h_opt)


def interp_to(h, n_new):
    n_old = len(h)
    if n_new == n_old:
        return h.copy()
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h)
    return project_to_target(h_new, n_new / 2.0)
