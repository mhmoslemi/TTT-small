import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
from functools import partial
import time


def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(200):
        mid = (lo + hi) * 0.5
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14 and n > 0:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h


def raw_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def c5_of_h(h, dx):
    return float(raw_corrs(h).max() * dx)


def raw_corr_val(h, k, n):
    offset = k - (n - 1)
    i_lo = max(0, -offset)
    i_hi = min(n - 1, n - 1 - offset)
    if i_lo > i_hi:
        return 0.0
    idx = np.arange(i_lo, i_hi + 1)
    if offset == 0:
        return float(np.sum(h[idx] * (1.0 - h[idx])))
    return float(np.sum(h[idx + offset] * (1.0 - h[idx])))


def raw_corr_grad(h, k, n):
    offset = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -offset)
    i_hi = min(n - 1, n - 1 - offset)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    if offset == 0:
        grad[idx] = 1.0 - 2.0 * h[idx]
    else:
        grad[idx + offset] += (1.0 - h[idx])
        grad[idx] += (-h[idx + offset])
    return grad


def corr_grad_h_to_v(g_h, n):
    m = n // 2
    return g_h[:m] - g_h[n - 1 - np.arange(m)]


def anti_pal_build(v, n):
    m = n // 2
    h = np.zeros(n)
    h[:m] = v
    h[m:] = 1.0 - v[m - 1::-1]
    return h


class SmoothMax:
    def __init__(self, dx, target_sum, beta, n):
        self.dx = dx
        self.ts = target_sum
        self.beta = beta
        self.n = n

    def obj(self, x):
        c = raw_corrs(x) * self.dx
        return float(logsumexp(self.beta * c) / self.beta)

    def jac(self, x):
        c = raw_corrs(x) * self.dx
        sh = c - c.max()
        w = np.exp(self.beta * sh)
        ws = w.sum()
        if ws < 1e-30:
            w = np.ones(2 * self.n - 1) / (2 * self.n - 1)
        else:
            w /= ws
        grad = np.zeros(self.n)
        for k in range(2 * self.n - 1):
            if w[k] < 1e-12:
                continue
            grad += w[k] * raw_corr_grad(x, k, self.n)
        return grad * self.dx

    def eq_fun(self, x):
        return x.sum() - self.ts

    def eq_jac(self, x):
        return np.ones(self.n)


class EpigraphFull:
    def __init__(self, n, target_sum, active_k, t_init):
        self.n = n
        self.ts = target_sum
        self.aks = list(active_k)
        self.t_init = t_init

    def obj(self, v):
        return v[self.n]

    def jac(self, v):
        g = np.zeros(self.n + 1)
        g[self.n] = 1.0
        return g

    def ineq_fun(self, v):
        h = v[:self.n]
        t = v[self.n]
        return np.array([t - raw_corr_val(h, k, self.n) for k in self.aks])

    def ineq_jac(self, v):
        h = v[:self.n]
        m = len(self.aks)
        rows = np.zeros((m, self.n + 1))
        for i, k in enumerate(self.aks):
            g = raw_corr_grad(h, k, self.n)
            rows[i, :self.n] = -g
            rows[i, self.n] = 1.0
        return rows

    def eq_fun(self, v):
        return v[:self.n].sum() - self.ts

    def eq_jac(self, v):
        g = np.zeros(self.n + 1)
        g[:self.n] = 1.0
        return g


class EpigraphAP:
    def __init__(self, n, active_k, t_init):
        self.n = n
        self.m = n // 2
        self.aks = list(active_k)
        self.t_init = t_init

    def obj(self, v):
        return v[self.m]

    def jac(self, v):
        g = np.zeros(self.m + 1)
        g[self.m] = 1.0
        return g

    def ineq_fun(self, v):
        h = anti_pal_build(np.clip(v[:self.m], 0.0, 1.0), self.n)
        t = v[self.m]
        return np.array([t - raw_corr_val(h, k, self.n) for k in self.aks])

    def ineq_jac(self, v):
        h = anti_pal_build(np.clip(v[:self.m], 0.0, 1.0), self.n)
        m = len(self.aks)
        rows = np.zeros((m, self.m + 1))
        for i, k in enumerate(self.aks):
            g_h = raw_corr_grad(h, k, self.n)
            g_v = corr_grad_h_to_v(g_h, self.n)
            rows[i, :self.m] = -g_v
            rows[i, self.m] = 1.0
        return rows


def solve_smooth(h0, dx, ts, beta, maxiter=500, ftol=1e-16):
    n = len(h0)
    p = SmoothMax(dx, ts, beta, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': p.eq_fun, 'jac': p.eq_jac}]
    try:
        res = minimize(p.obj, h0, jac=p.jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        h_out = project_feasible(res.x, ts)
    except Exception:
        h_out = project_feasible(h0, ts)
    return h_out, c5_of_h(h_out, dx)


def solve_epigraph_full(h0, dx, ts, active_k, t_init, maxiter=400, ftol=1e-15):
    n = len(h0)
    p = EpigraphFull(n, ts, active_k, t_init)
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    cons = [
        {'type': 'ineq', 'fun': p.ineq_fun, 'jac': p.ineq_jac},
        {'type': 'eq', 'fun': p.eq_fun, 'jac': p.eq_jac},
    ]
    v0 = np.concatenate([h0, [t_init]])
    try:
        res = minimize(p.obj, v0, jac=p.jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        h_out = project_feasible(res.x[:n], ts)
    except Exception:
        h_out = project_feasible(h0, ts)
    return h_out, c5_of_h(h_out, dx)


def solve_epigraph_ap(v0, n, dx, active_k, t_init, maxiter=400, ftol=1e-15):
    p = EpigraphAP(n, active_k, t_init)
    m = n // 2
    bounds = [(0.0, 1.0)] * m + [(0.0, None)]
    cons = [{'type': 'ineq', 'fun': p.ineq_fun, 'jac': p.ineq_jac}]
    v0f = np.concatenate([v0, [t_init]])
    try:
        res = minimize(p.obj, v0f, jac=p.jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        v_out = np.clip(res.x[:m], 0.0, 1.0)
        h_out = anti_pal_build(v_out, n)
    except Exception:
        h_out = anti_pal_build(np.clip(v0, 0.0, 1.0), n)
    return h_out, c5_of_h(h_out, dx)


def get_active_k(h, top_m=25, buffer_raw=0.0):
    corrs = raw_corrs(h)
    max_c = corrs.max()
    thresh = max_c - buffer_raw
    active = np.where(corrs >= thresh)[0]
    if len(active) > top_m:
        order = np.argsort(corrs[active])[::-1][:top_m]
        active = active[order]
    elif len(active) < 5:
        idx_sorted = np.argsort(corrs)[::-1]
        active = idx_sorted[:top_m]
    return active


def multi_start_explore(h_best, dx, ts, rng, n_starts=10, maxiter_each=300):
    n = len(h_best)
    best = h_best.copy()
    best_c5 = c5_of_h(h_best, dx)
    for i in range(n_starts):
        mag = 0.03 + 0.05 * np.sqrt((i + 1) / n_starts)
        h_pert = h_best + rng.normal(0, mag, n)

        if c5_opt < best_c5:
