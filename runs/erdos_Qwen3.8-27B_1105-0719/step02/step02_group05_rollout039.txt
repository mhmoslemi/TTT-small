Let me check: the correlation at lag d is:
f_d(h) = sum_i h[i+d]*(1-h[i]) = sum_i h[i+d] - sum_i h[i+d]*h[i]

The first term is linear in h. The second term is quadratic: sum_i h[i+d]*h[i].

So f_d(h) = linear_term - quadratic_term.

The quadratic term sum_i h[i+d]*h[i] = h^T Q_d h / 2 (where Q_d is a symmetric matrix with 1s at positions (i, i+d) and (i+d, i)).

So f_d(h) = c_d^T h - h^T Q_d h / 2 (approximately).

The constraint f_d(h) <= t becomes: h^T Q_d h / 2 - c_d^T h + t >= 0.

This is not a convex constraint (Q_d is not PSD). So CVXPY won't work directly.

def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return compute_corrs(h).max() * dx

def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-15) & (h < 1.0 - 1e-15)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def smooth_max_value_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corrs = compute_corrs(h)
    cs = corrs * dx
    cs_max = cs.max()
    exp_vals = np.exp(beta * (cs - cs_max))
    exp_sum = exp_vals.sum()
    val = (cs_max + np.log(exp_sum)) / beta
    w = exp_vals / exp_sum
    
    grad = np.zeros(n)
    for k in range(len(corrs)):
        wk = w[k]
        if wk < 1e-13:
            continue
        d = k - (n - 1)
        if d == 0:
            grad[:] += wk * (1.0 - 2.0 * h[:])
        elif d > 0:
            grad[d:] += wk * (1.0 - h[:n-d])
            grad[:n-d] += wk * (-h[d:])
        else:
            dd = -d
            grad[:n-dd] += wk * (1.0 - h[dd:])
            grad[dd:] += wk * (-h[:n-dd])
    
    return val, grad * dx

def nesterov_optimize(h0, beta, n_iter, target_sum, lr0=0.01):
    n = len(h0)
    x = h0.copy()
    y = h0.copy()
    t = 1.0
    best_x = h0.copy()
    best_score = exact_score(h0)
    
    for it in range(n_iter):
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        mom = (t - 1.0) / t_new
        y = project_feasible(x + mom * (x - y), target_sum)
        
        val, grad = smooth_max_value_grad(y, beta)
        step = lr0 / (1.0 + 0.01 * it)
        
        x_new = project_feasible(y - step * grad, target_sum)
        
        # Backtracking: check if we made progress
        s_new = exact_score(x_new)
        if s_new > exact_score(y) + 1e-10:
            step *= 0.5
            x_new = project_feasible(y - step * grad, target_sum)
        
        x = x_new
        t = t_new
        
        if s_new < best_score:
            best_score = s_new
            best_x = x_new.copy()
    
    return best_x, best_score

import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time

def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return compute_corrs(h).max() * dx

def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-15) & (h < 1.0 - 1e-15)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def grad_corr_d(h, d, n):
    g = np.zeros(n)
    if d >= 0:
        if d < n:
            g[d:] += (1.0 - h[:n - d])
            g[:n - d] += (-h[d:])
    else:
        dd = -d
        if dd < n:
            g[:n - dd] += (1.0 - h[dd:])
            g[dd:] += (-h[:n - dd])
    return g

def smooth_max_value_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corrs = compute_corrs(h)
    cs = corrs * dx
    cs_max = cs.max()
    exp_vals = np.exp(beta * (cs - cs_max))
    exp_sum = exp_vals.sum()
    val = (cs_max + np.log(exp_sum)) / beta
    w = exp_vals / exp_sum
    grad = np.zeros(n)
    for k in range(len(corrs)):
        wk = w[k]
        if wk < 1e-13:
            continue
        d = k - (n - 1)
        G = grad_corr_d(h, d, n)
        grad += wk * G
    return val, grad * dx

def nesterov_optimize(h0, beta, n_iter, target_sum, lr0=0.02, max_step=0.5):
    n = len(h0)
    x = h0.copy()
    y = h0.copy()
    t = 1.0
    best_x = h0.copy()
    best_score = exact_score(h0)
    for it in range(n_iter):
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        mom = (t - 1.0) / t_new
        y = project_feasible(x + mom * (x - y), target_sum)
        val, grad = smooth_max_value_grad(y, beta)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break
        step = min(lr0 / (1.0 + it * 0.05), max_step / (grad_norm + 1e-10))
        x_new = project_feasible(y - step * grad, target_sum)
        x = x_new
        t = t_new
        if it % 20 == 0 or it == n_iter - 1:
            s = exact_score(x)
            if s < best_score:
                best_score = s
                best_x = x.copy()
    s_final = exact_score(x)
    if s_final < best_score:
        best_score = s_final
        best_x = x.copy()
    return best_x, best_score

def fista_optimize(h0, beta, n_iter, target_sum, lr0=0.05):
    n = len(h0)
    x = h0.copy()
    z = h0.copy()
    t = 1.0
    best_x = h0.copy()

def perturbation_restart(h0, target_sum, scale, rng, n_iter=300, beta=500):
    n = len(h0)
    noise = rng.randn(n) * scale
    h_pert = project_feasible(h0 + noise, target_sum)
    h_opt, score = nesterov_optimize(h_pert, beta, n_iter, target_sum, lr0=0.03)
    return h_opt, score

def active_set_solve(h0, n, active_k, target_sum, maxiter=200):
    dx = 2.0 / n
    corrs = compute_corrs(h0)
    t0 = corrs.max() * dx + 1e-5

    class EpigraphQ:
        def __init__(self, n, active_k, dx, target_sum):
            self.n = n
            self.active_k = active_k
            self.dx = dx
            self.target_sum = target_sum

        def obj(self, x):
            return x[0]

        def obj_grad(self, x):
            g = np.zeros(self.n + 1)
            g[0] = 1.0
            return g

        def constraint_corr(self, x, k_idx):
            t = x[0]
            h = x[1:]
            k = self.active_k[k_idx]
            d = k - (self.n - 1)
            n = self.n
            if d >= 0:
                i_vals = np.arange(0, n - d)
                val = np.sum(h[i_vals + d] * (1.0 - h[i_vals]))
            else:
                dd = -d
                i_vals = np.arange(dd, n)
                val = np.sum(h[i_vals] * (1.0 - h[i_vals - dd]))
            val *= self.dx
            return t - val

        def constraint_corr_grad(self, x, k_idx):
            h = x[1:]
            k = self.active_k[k_idx]
            d = k - (self.n - 1)
            n = self.n
            g = np.zeros(n + 1)
            g[0] = 1.0
            G = grad_corr_d(h, d, n)
            g[1:] = -G * self.dx
            return g

        def constraint_sum(self, x):
            return x[1:].sum() - self.target_sum

        def constraint_sum_grad(self, x):
            g = np.zeros(self.n + 1)
            g[1:] = np.ones(self.n)
            return g

    ep = EpigraphQ(n, active_k, dx, target_sum)
    x0 = np.concatenate([[t0], h0])
    cons = []
    for i in range(len(active_k)):
        cons.append({'type': 'ineq',
                     'fun': lambda x, i=i: ep.constraint_corr(x, i),
                     'jac': lambda x, i=i: ep.constraint_corr_grad(x, i)})
    cons.append({'type': 'eq', 'fun': ep.constraint_sum, 'jac': ep.constraint_sum_grad})
    bounds = [(0.0, 5.0)] + [(0.0, 1.0)] * n
    try:
        res = minimize(ep.obj, x0, jac=ep.obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        h_out = project_feasible(res.x[1:], target_sum)
        return h_out, res
    except Exception:
        return h0, None

            hidden_idx = np.where(hidden_mask
