    return best_x

def coordinate_descent_incr(h_init, n, n_iters, seed=42, n_grid=60, recompute_every=20):
    rng = np.random.RandomState(seed)
    target_sum = n / 2.0
    h = h_init.copy()
    h = project_feasible(h, target_sum)
    best_h = h.copy()
    best_c5 = exact_score(h)
    corrs = compute_corrs(h)
    best_track_interval = max(1, n_iters // 50)
    
    for it in range(n_iters):
        order = rng.permutation(n)
        for j in order:
            old = h[j]
            best_x = coord_optimize_analytic(j, h, corrs, n, n_grid)
            if best_x != old:
                h[j] = best_x
                corrs[n - 1] += (best_x - old) * (1.0 - best_x - old)
                a = np.zeros(n)
                for d in range(1, n):
                    ad = 0.0
                    if d <= j:
                        ad += 1.0 - h[j - d]
                    if j <= n - 1 - d:
                        ad -= h[j + d]
                    a[d] = ad
                    corrs[n - 1 + d] += ad * (best_x - old)
        
        h = project_feasible(h, target_sum)
        
        if (it + 1) % recompute_every == 0:
            corrs = compute_corrs(h)
        
        if (it + 1) % best_track_interval == 0:
            c5 = exact_score(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    
    h = project_feasible(h, target_sum)
    c5 = exact_score(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    return best_h, best_c5

def smooth_q_value_grad(q, n, beta):
    m = n // 2
    h = np.concatenate([q, 1.0 - q[::-1]])
    dx = 2.0 / n
    corrs = compute_corrs(h)
    val = logsumexp(beta * corrs) / beta * dx
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(len(corrs)) / len(corrs)
    else:
        w /= w_sum
    Gq = np.zeros(m)
    for k in range(len(corrs)):
        if w[k] < 1e-14:
            continue
        d = k - (n - 1)
        if d >= 0:
            if d < n:
                G = np.zeros(n)
                G[d:] += (1.0 - h[:n - d])
                G[:n - d] += (-h[d:])
        else:
            dd = -d
            if dd < n:
                G = np.zeros(n)
                G[:n - dd] += (1.0 - h[dd:])
                G[dd:] += (-h[:n - dd])
        Gq += w[k] * (G[:m] - G[m:][::-1])
    return val, Gq * dx

def build_h_from_q(q, n):
    m = n // 2
    if n % 2 == 0:
        return np.concatenate([q, 1.0 - q[::-1]])
    else:
        return np.concatenate([q, [0.5], 1.0 - q[::-1]])

def q_from_h(h, n):
    m = n // 2
    return h[:m].copy()

def smooth_q_optimize(q0, n, beta, maxiter=500):
    def q_val(x):
        return smooth_q_value_grad(x, n, beta)[0]
    def q_grad(x):
        return smooth_q_value_grad(x, n, beta)[1]
    try:
        res = minimize(q_val, q0, jac=q_grad, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * (n // 2),
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return np.clip(res.x, 0.0, 1.0)
    except Exception:
        return q0

def grad_corr_d(h, d, n):
    g = np.zeros(n)
    if d >= 0:
        if d < n:
            g[d:] += (1.0 - h[:n - d])
            g[:n - d] += (-h[d:])
    else:
        dd = -d
        if dd < n:
            g[:n - dd] += (1.0 - h[dd:])
            g[dd:] += (-h[:n - dd])
    return g

def full_active_search(h0, n, max_rounds=6, max_active=40):
    target_sum = n / 2.0
    dx = 2.0 / n
    corrs = compute_corrs(h0)
    current_max = corrs.max() * dx
    L_init = min(20, 2 * n - 1)
    top_idx = np.argsort(corrs)[::-1][:L_init]
    active_k = sorted(top_idx.tolist())
    h = h0.copy()
    best_h = h.copy()
    best_c5 = current_max
    t0 = current_max * 1.001 + 1e-6

    for rnd in range(max_rounds):
        h = project_feasible(h, target_sum)
        corrs = compute_corrs(h)
        current_max = corrs.max() * dx
        if current_max < best_c5:
            best_c5 = current_max
            best_h = h.copy()

        m_vars = n
        x0 = np.concatenate([[t0], h])

        def obj(x):
            return x[0]
        def obj_grad(x):
            g = np.zeros(n + 1)
            g[0] = 1.0
            return g

        cons = []
        for i, k in enumerate(active_k):
            d = k - (n - 1)
            def c_fun(x, d=d):
                hh = x[1:]
                if d >= 0:
                    iv = np.arange(0, n - d)
                    return x[0] - np.sum(hh[iv + d] * (1.0 - hh[iv])) * dx
                else:
                    dd = -d
                    iv = np.arange(dd, n)
                    return x[0] - np.sum(hh[iv] * (1.0 - hh[iv - dd])) * dx
            def c_jac(x, d=d):
                hh = x[1:]
                g = np.zeros(n + 1)
                g[0] = 1.0
                G = grad_corr_d(hh, d, n)
                g[1:] = -G * dx
                return g
            cons.append({'type': 'ineq', 'fun': c_fun, 'jac': c_jac})

        def sum_fun(x):
            return x[1:].sum() - target_sum
        def sum_jac(x):
            g = np.zeros(n + 1)
            g[1:] = 1.0
            return g
        cons.append({'type': 'eq', 'fun': sum_fun, 'jac': sum_jac})

        bounds = [(0.0, 5.0)] + [(0.0, 1.0)] * n
        try:
            res = minimize(obj, x0, jac=obj_grad, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 200, 'ftol': 1e-13})
            h_new = project_feasible(res.x[1:], target_sum)
        except Exception:
            h_new = h.copy()

        corrs_new = compute_corrs(h_new)
        c5_new = corrs_new.max() * dx
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()

        hidden = corrs_new > res.x[0] + 1e-6 if res is not None else corrs_new > t0 + 1e-6
        if hidden.sum() > 0:
            hidden_idx = np.where(hidden)[0]
            new_active = sorted(set(active_k) | set(hidden_idx.tolist()))
            if len(new_active) > max_active:
                vals = corrs_new[new_active]
                order2 = np.argsort(vals)[::-1][:max_active]
                new_active = sorted(new_active[i] for i in order2)
            active_k = new_active

        t0 = max(res.x[0] if res is not None else t0, c5_new) * 1.0001 + 1e-6
        h = h_new

        if abs(c5_new - best_c5) < 1e-12 and hidden.sum() == 0:
            break

    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()

    def elapsed():
        return time.time() - t_start

    def remaining():
        return budget_s - elapsed()

    rng = np.random.RandomState(seed)

    # Initialize from parent
    h_parent = np.array(initial_h_values, dtype=float)
    n_parent = len(h_parent)
    target_sum = n_parent / 2.0
    h_parent = project_feasible(h_parent, target_sum)
    best_h = h_parent.copy()
    best_c5 = exact_score(h_parent)

    # Phase 1: Coordinate descent in full space (no symmetry)
    n = n_parent
    n_iters_cd = 300
    h_cd, c5_cd = coordinate_descent_incr(best_h, n, n_iters_cd, seed=seed, n_grid=70)
    if c5_cd < best_c5:
        best_c5 = c5_cd
        best_h = h_cd
    if remaining() < 60:
        best_h = project_feasible(best_h, len(best_h) / 2.0)
        return best_h, best_c5, len(best_h)

    # Phase 2: More coordinate descent with different seeds (perturbation)
    for s in range(3):
        if remaining() < 50:
            break
        h_pert = best_h + rng.randn(n) * 0.02
        h_pert = project_feasible(h_pert, target_sum)
        h_cd2, c5_cd2 = coordinate_descent_incr(h_pert, n, 200, seed=seed + s + 100, n_grid=60)
        if c5_cd2 < best_c5:
            best_c5 = c5_cd2
            best_h = h_cd2

    if remaining() < 60:
        best_h = project_feasible(best_h, len(best_h) / 2.0)
        return best_h, best_c5, len(best_h)

    # Phase 3: Smooth optimization in q-space (with symmetry)
    if n % 2 == 0:
        q0 = q_from_h(best_h, n)
        for beta in [200, 800, 2000]:
            if remaining() < 30:
                break
            q_res = smooth_q_optimize(q0, n, beta, maxiter=400)
            h_res = build_h_from_q(q_res, n)
            h_res = project_feasible(h_res, target_sum)
            s_res = exact_score(h_res)
            if s_res < best_c5 - 1e-14:
                best_c5 = s_res
                best_h = h_res
                q0 = q_res.copy()
            else:
                q0 = q_res.copy()

    # Phase 4: Coordinate descent again from updated best
    if remaining() > 40:
        h_cd3, c5_cd3 = coordinate_descent_incr(best_h, n, 200, seed=seed + 999, n_grid=80)
        if c5_cd3 < best_c5:
            best_c5 = c5_cd3
            best_h = h_cd3

    if remaining() < 60:
        best_h = project_feasible(best_h, len(best_h) / 2.0)
        return best_h, best_c5, len(best_h)

    # Phase 5: Active-set epigraph polish
    h_act, c5_act = full_active_search(best_h, n, max_rounds=5, max_active=35)
    if c5_act < best_c5:
        best_c5 = c5_act
        best_h = h_act

    # Phase 6: Try different grid sizes
    for n_try in [64, 80, 96, 112, 128, 140]:
        if remaining() < 50:
            break
        dx_try = 2.0 / nTry if False else 2.0 / n_try
        ts_try = n_try / 2.0
        x_orig = np.linspace(0.0, 2.0, len(best_h))
        x_new = np.linspace(0.0, 2.0, n_try)
        h_interp = np.interp(x_new, x_orig, best_h)
        h_interp = project_feasible(h_interp, ts_try)
        h_sm, c5_sm = coordinate_descent_incr(h_interp, n_try, 150, seed=seed + n_try, n_grid=50)
        if c5_sm < best_c5 - 1e-10:
            best_c5 = c5_sm
            best_h = h_sm
        if n_try % 2 == 0 and remaining() > 40:
            h_act2, c5_act2 = full_active_search(h_sm, n_try, max_rounds=4, max_active=30)
            if c5_act2 < best_c5 - 1e-10:
                best_c5 = c5_act2
                best_h = h_act2

    # Phase 7: Final coordinate descent polish
    if remaining() > 30:
        n_final = len(best_h)
        ts_final = n_final / 2.0
        best_h = project_feasible(best_h, ts_final)
        h_fin, c5_fin = coordinate_descent_incr(best_h, n_final, 300, seed=seed + 7777, n_grid=100)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    n_out = len(best_h)
    ts_out = n_out / 2.0
    best_h = project_feasible(best_h, ts_out)
    final_score = exact_score(best_h)
    if final_score < best_c5:
        best_c5 = final_score

    return best_h, best_c5, n_out
