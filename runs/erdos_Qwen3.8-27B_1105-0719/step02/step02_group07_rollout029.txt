```python
def subgradient_sym(q, n, active_lags=None, top_k=20):
    """Compute subgradient of max corr in q-space for symmetric h."""
    m = n // 2
    dx = 2.0 / n
    h = make_sym_h(q, n)
    corrs = corr_vals(h)
    
    if active_lags is None:
        # Use top-k lags
        idx = np.argsort(corrs)[::-1][:top_k]
        active_lags = idx
    
    # Weight the active lags equally (or proportionally to how close they are to max)
    max_c = corrs.max()
    weights = np.zeros(2*n-1)
    for lag in active_lags:
        # Weight proportional to proximity to max
        w = np.exp(10 * (corrs[lag] - max_c))
        weights[lag] = w
    ws = weights.sum()
    if ws < 1e-30:
        return np.zeros(m)
    weights /= ws
    
    # Compute gradient in h-space
    grad_h = np.zeros(n)
    for lag in range(2*n-1):
        if weights[lag] < 1e-15:
            continue
        d = lag - (n-1)
        i_lo = max(0, -d)
        i_hi = min(n-1, n-1-d)
        if i_lo > i_hi:
            continue
        idx = np.arange(i_lo, i_hi+1)
        grad_h[idx] += weights[lag] * (1.0 - h[idx+d]) * dx
        grad_h[idx+d] += weights[lag] * (-h[idx]) * dx
    
    # Map to q-space
    grad_q = grad_h[:m] - grad_h[n-1:n-1-m:-1]
    return grad_q
```

from scipy.optimize import minimize
from scipy.special import logsumexp
import time

def corr_vals(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def exact_score(h):
    return float(corr_vals(h).max())

def make_sym_h(q, n):
    m = n // 2
    q = np.clip(q, 0.0, 1.0)
    h = np.empty(n)
    h[:m] = q
    h[m:] = 1.0 - q[::-1]
    return h

def make_q_from_h(h, n):
    m = n // 2
    q = 0.5 * (h[:m] + 1.0 - h[n-1:n-1-m:-1])
    return np.clip(q, 0.0, 1.0)

def top_lags(corrs, k):
    idx = np.argsort(corrs)[::-1][:k]
    return sorted(idx.tolist())

class SmoothProblem:
    def __init__(self, n, beta):
        self.n = n
        self.m = n // 2
        self.beta = beta
        self.dx = 2.0 / n
    
    def fun(self, q):
        h = make_sym_h(q, self.n)
        c = corr_vals(h)
        return logsumexp(self.beta * c) / self.beta
    
    def grad(self, q):
        h = make_sym_h(q, self.n)
        c = corr_vals(h)
        s = c - c.max()
        w = np.exp(self.beta * s)
        ws = w.sum()
        if ws < 1e-30:
            w = np.ones(2*self.n-1) / (2*self.n-1)
        else:
            w /= ws
        g = self._weighted_grad(h, w)
        return g[:self.m] - g[self.n-1:self.n-1-self.m:-1]
    
    def _weighted_grad(self, h, w):
        n = self.n
        dx = self.dx
        grad = np.zeros(n)
        for k in range(2*n-1):
            if w[k] < 1e-15:
                continue
            d = k - (n-1)
            i_lo = max(0, -d)
            i_hi = min(n-1, n-1-d)
            if i_lo > i_hi:

            idx = np.arange(i_lo, i_hi+1)
            grad[idx] += w[k] * (1.0 - h[idx+d]) * dx
            grad[idx+d] += w[k] * (-h[idx]) * dx
        return grad

def smooth_optimize(q0, n, beta, maxiter=300):
    prob = SmoothProblem(n, beta)
    res = minimize(prob.fun, q0, jac=prob.grad, method='L-BFGS-B',
                   bounds=[(0.0, 1.0)] * (n//2),
                   options={'maxiter': maxiter, 'ftol': 1e-16})
    return np.clip(res.x, 0.0, 1.0)

    def __init__(self, n, active_lags):
        self.n = n
        self.m = n // 2
        self.dx = 2.0 / n
        self.lags = np.array(active_lags)
    
    def fun(self, x):
        return x[0]
    
    def jac(self, x):
        g = np.zeros(self.m + 1)
        g[0] = 1.0
        return g
    
    def con_fun(self, x):
        t = x[0]
        q = x[1:]
        h = make_sym_h(q, self.n)
        c = corr_vals(h)
        return t - c[self.lags]
    
    def con_jac(self, x):
        q = x[1:]
        h = make_sym_h(q, self.n)
        n = self.n
        m = self.m
        dx = self.dx
        L = len(self.lags)
        J = np.zeros((L, m+1))
        J[:, 0] = 1.0
        for i, lag in enumerate(self.lags):
            d = lag - (n-1)
            i_lo = max(0, -d)
            i_hi = min(n-1, n-1-d)
            G = np.zeros(n)
            if i_lo <= i_hi:
                idx = np.arange(i_lo, i_hi+1)
                G[idx] += (1.0 - h[idx+d]) * dx
                G[idx+d] += (-h[idx]) * dx
            J[i, 1:] = -(G[:m] - G[n-1:n-1-m:-1])
        return J

def epigraph_optimize(q0, n, active_lags, t_init, maxiter=300):
    prob = EpiProblem(n, active_lags)
    x0 = np.concatenate([[t_init], q0])
    cons = [{'type': 'ineq', 'fun': prob.con_fun, 'jac': prob.con_jac}]
    bounds = [(0.0, 2.0)] + [(0.0, 1.0)] * (n//2)
    res = minimize(prob.fun, x0, jac=prob.jac, method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    if res.x[0] >= 0 and not np.any(res.x > 2):
        q = np.clip(res.x[1:], 0.0, 1.0)
        h = make_sym_h(q, n)
        c = exact_score(h)
        if abs(h.sum() - n/2.0) < 1e-8:
            return q, c
    return None, None

def corr_vals(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)


def exact_score(h):
    return float(corr_vals(h).max())


def make_sym_h(q, n):
    m = n // 2
    q = np.clip(q, 0.0, 1.0)
    h = np.empty(n)
    h[:m] = q
    h[m:] = 1.0 - q[::-1]
    return h


def make_q_from_h(h, n):
    m = n // 2
    q = 0.5 * (h[:m] + 1.0 - h[n-1:n-1-m:-1])
    return np.clip(q, 0.0, 1.0)


def top_lags(corrs, k):
    return np.argsort(corrs)[::-1][:k]


def score_q(q, n):
    return exact_score(make_sym_h(q, n))


class SmoothProblem:
    def __init__(self, n, beta):
        self.n = n
        self.m = n // 2
        self.beta = beta
        self.dx = 2.0 / n

    def fun(self, q):
        h = make_sym_h(q, self.n)
        c = corr_vals(h)
        return logsumexp(self.beta * c) / self.beta

    def grad(self, q):
        h = make_sym_h(q, self.n)
        c = corr_vals(h)
        s = c - c.max()
        w = np.exp(self.beta * s)
        ws = w.sum()
        if ws < 1e-30:
            w = np.ones(2 * self.n - 1) / (2 * self.n - 1)
        else:
            w /= ws
        g = self._wgrad(h, w)
        return g[:self.m] - g[self.n-1:self.n-1-self.m:-1]

    def _wgrad(self, h, w):
        n = self.n
        dx = self.dx
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] < 1e-15:
                continue
            d = k - (n - 1)
            i_lo = max(0, -d)
            i_hi = min(n - 1, n - 1 - d)
            if i_lo > i_hi:
                continue
            idx = np.arange(i_lo, i_hi + 1)
            grad[idx] += w[k] * (1.0 - h[idx + d]) * dx
            grad[idx + d] += w[k] * (-h[idx]) * dx
        return grad


def smooth_optimize(q0, n, beta, maxiter=300):
    prob = SmoothProblem(n, beta)
    bounds = [(0.0, 1.0)] * (n // 2)
    res = minimize(prob.fun, q0, jac=prob.grad, method='L-BFGS-B',
                   bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-14})
    return np.clip(res.x, 0.0, 1.0)


class EpiProblem:
    def __init__(self, n, active_lags):
        self.n = n
        self.m = n // 2
        self.dx = 2.0 / n
        self.lags = np.array(active_lags, dtype=int)

    def fun(self, x):
        return x[0]

    def jac(self, x):
        g = np.zeros(self.m + 1)
        g[0] = 1.0
        return g

    def con_fun(self, x):
        t = x[0]
        q = x[1:]
        h = make_sym_h(q, self.n)
        c = corr_vals(h)
        return t - c[self.lags]

    def con_jac(self, x):
        q = x[1:]
        h = make_sym_h(q, self.n)
        n = self.n
        m = self.m
        dx = self.dx
        L = len(self.lags)
        J = np.zeros((L, m + 1))
        J[:, 0] = 1.0
        for i, lag in enumerate(self.lags):
            d = lag - (n - 1)
            i_lo = max(0, -d)
            i_hi = min(n - 1, n - 1 - d)
            G = np.zeros(n)
            if i_lo <= i_hi:
                idx = np.arange(i_lo, i_hi + 1)
                G[idx] += (1.0 - h[idx + d]) * dx
                G[idx + d] += (-h[idx]) * dx
            J[i, 1:] = -(G[:m] - G[n-1:n-1-m:-1])
        return J


def epigraph_optimize(q0, n, active_lags, t_init, maxiter=300):
    prob = EpiProblem(n, active_lags)
    x0 = np.concatenate([[t_init], q0])
    cons = [{'type': 'ineq', 'fun': prob.con_fun, 'jac': prob.con_jac}]
    bounds = [(0.0, 3.0)] + [(0.0, 1.0)] * (n // 2)
    try:
        res = minimize(prob.fun, x0, jac=prob.jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        q = np.clip(res.x[1:], 0.0, 1.0)
        h = make_sym_h(q, n)
        c5 = exact_score(h)
        if abs(h.sum() - n / 2.0) < 1e-8:
            return q, c5
    except Exception:
        pass
    return None, None


def structured_perturb(q, n, rng, mode=0):
    m = len(q)
    q_new = q.copy()
    if mode == 0:
        scale = rng.uniform(0.02, 0.15)
        q_new += scale * rng.randn(m)
    elif mode == 1:
        f = rng.uniform(0.5, 5.0)
        A = rng.uniform(0.03, 0.2)
        phi = rng.uniform(0, 2 * np.pi)
        x = np.linspace(0, 1, m)
        q_new += A * np.sin(2 * np.pi * f * x + phi)
    elif mode == 2:
        i0 = rng.randint(0, m - 10)
        i1 = i0 + rng.randint(10, min(30, m - i0))
        q_new[i0:i1] += rng.uniform(-0.2, 0.2) * rng.randn(i1 - i0)
    elif mode == 3:
        i0 = rng.randint(0, m - 5)
        i1 = i0 + rng.randint(5, min(20, m - i0))
        q_new[i0:i1] = 1.0 - q_new[i0:i1]
    elif mode == 4:
        center = rng.uniform(0.2, 0.8)
        width = rng.uniform(0.05, 0.3)
        A = rng.uniform(-0.25, 0.25)
        x = np.linspace(0, 1, m)
        q_new += A * np.exp(-0.5 * ((x - center) / width) ** 2)
    return np.clip(q_new, 0.0, 1.0)


def subgradient_polish(q, n, n_iter=200, initial_step=0.03):
    m = n // 2
    dx = 2.0 / n
    q_cur = q.copy()
    best_q = q_cur.copy()
    best_c = score_q(best_q, n)
    step = initial_step
    momentum = np.zeros(m)
    beta_mom = 0.85
    stalled = 0

    for it in range(n_iter):
        h = make_sym_h(q_cur, n)
        corrs = corr_vals(h)
        c = corrs.max()
        if c < best_c - 1e-12:
            best_c = c
            best_q = q_cur.copy()
            stalled = 0
        else:
            stalled += 1

        if stalled > 80:
            step *= 0.7
            momentum *= 0.5

        active = top_lags(corrs, 25)
        max_c = corrs.max()
        weights = np.zeros(2 * n - 1)
        for lag in active:
            weights[lag] = np.exp(8.0 * (corrs[lag] - max_c))
        ws = weights.sum()
        if ws < 1e-30:
            break
        weights /= ws

        grad_h = np.zeros(n)
        for k in range(2 * n - 1):
            if weights[k] < 1e-15:
                continue
            d = k - (n - 1)
            i_lo = max(0, -d)
            i_hi = min(n - 1, n - 1 - d)
            if i_lo > i_hi:
                continue
            idx = np.arange(i_lo, i_hi + 1)
            grad_h[idx] += weights[k] * (1.0 - h[idx + d]) * dx
            grad_h[idx + d] += weights[k] * (-h[idx]) * dx

        grad_q = grad_h[:m] - grad_h[n-1:n-1-m:-1]
        momentum = beta_mom * momentum - step * grad_q
        q_cur = np.clip(q_cur + momentum, 0.0, 1.0)

    return best_q, best_c


def anneal_polish(q, n, n_iter=3000, T0=0.01, T_end=0.0005):
    m = n // 2
    q_cur = q.copy()
    best_q = q_cur.copy()
    best_c = score_q(best_q, n)
    c_cur = best_c

    for it in range(n_iter):
        frac = it / max(n_iter - 1, 1)
        T = T0 * (T_end / T0) ** frac
        eps = 0.01 * (1.0 - 0.5 * frac)
        idx = rng_anneal.randint(m)
        q_new = q_cur.copy()
        q_new[idx] += eps * rng_anneal.randn()
        q_new[idx] = np.clip(q_new[idx], 0.0, 1.0)

        c_new = score_q(q_new, n)
        delta = c_new - c_cur
        if delta < 0 or rng_anneal.random() < np.exp(-delta / max(T, 1e-10)):
            q_cur = q_new
            c_cur = c_new
            if c_new < best_c:
                best_c = c_new
                best_q = q_cur.copy()

    return best_q, best_c


rng_anneal = np.random.RandomState(12345)


def full_polish(q, n, time_limit=30):
    t_start = time.time()
    m = n // 2

    h = make_sym_h(q, n)
    corrs = corr_vals(h)
    active = top_lags(corrs, 50)
    q_new, c5_new = epigraph_optimize(q, n, active, corrs.max() * 1.0005, maxiter=400)
    if q_new is not None and c5_new < score_q(q, n):
        q = q_new

    if time.time() - t_start < time_limit - 5:
        q_sub, c5_sub = subgradient_polish(q, n, n_iter=500, initial_step=0.02)
        if c5_sub < score_q(q, n):
            q = q_sub

    if time.time() - t_start < time_limit - 3:
        q_ann, c5_ann = anneal_polish(q, n, n_iter=2000)
        if c5_ann < score_q(q, n):
            q = q_ann

    return q, score_q(q, n)


def interpolate_h(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def remaining():
        return budget_s - (time.time() - t0)

    try:
        h_parent = np.array(initial_h_values, dtype=float)
    except NameError:
        h_parent = 0.5 + 0.4 * np.tanh(np.linspace(-2, 2, 126))

    n = len(h_parent)
    if n % 2 != 0:
        n -= 1
        h_parent = h_parent[:n]

    h_best = np.clip(h_parent, 0.0, 1.0)
    c5_best = exact_score(h_best)
    m = n // 2
    q_best = make_q_from_h(h_best, n)

    # Phase 1: Progressive smooth optimization
    for beta in [40, 80, 150, 300, 600, 1000]:
        if remaining() < 20:
            break
        q_new = smooth_optimize(q_best, n, beta, maxiter=250)
        c5_new = score_q(q_new, n)
        if c5_new < c5_best:
            c5_best = c5_new
            q_best = q_new
            h_best = make_sym_h(q_best, n)

    # Phase 2: Epigraph with increasing active set
    for rnd in range(6):
        if remaining() < 20:
            break
        h_cur = make_sym_h(q_best, n)
        corrs = corr_vals(h_cur)
        n_act = min(25 + 12 * rnd, 2 * n - 1)
        active = top_lags(corrs, n_act)
        q_new, c5_new = epigraph_optimize(q_best, n, active, corrs.max() * 1.0005, maxiter=400)
        if q_new is not None and c5_new < c5_best - 1e-12:
            c5_best = c5_new
            q_best = q_new
            h_best = make_sym_h(q_best, n)
        # Re-optimize with updated active set
        if remaining() > 25:
            corrs2 = corr_vals(make_sym_h(q_best, n))
            active2 = top_lags(corrs2, n_act)
            q_new2, c5_new2 = epigraph_optimize(q_best, n, active2, corrs2.max() * 1.0002, maxiter=300)
            if q_new2 is not None and c5_new2 < c5_best - 1e-12:
                c5_best = c5_new2
                q_best = q_new2
                h_best = make_sym_h(q_best, n)

    # Phase 3: Diverse restarts with local refinement
    n_restarts = 30
    for trial in range(n_restarts):
        if remaining() < 20:
            break
        mode = trial % 5
        scale_factor = 0.5 + 0.5 * (trial / max(n_restarts, 1))
        q_pert = structured_perturb(q_best, n, rng, mode=mode)
        q_pert = np.clip(q_pert, 0.0, 1.0)

        q_loc = smooth_optimize(q_pert, n, 200, maxiter=150)
        c5_loc = score_q(q_loc, n)

        if c5_loc < c5_best + 0.001:
            corrs_loc = corr_vals(make_sym_h(q_loc, n))
            active_loc = top_lags(corrs_loc, 40)
            q_epi, c5_epi = epigraph_optimize(q_loc, n, active_loc, corrs_loc.max() * 1.0005, maxiter=300)
            if q_epi is not None and c5_epi < c5_best - 1e-12:
                c5_best = c5_epi
                q_best = q_epi
                h_best = make_sym_h(q_best, n)

    # Phase 4: Population-based search
    pop_size = 10
    population = [q_best.copy()]
    for i in range(pop_size - 1):
        mode = i % 5
        q_div = structured_perturb(q_best, n, rng, mode=mode)
        population.append(q_div)

    for gen in range(15):
        if remaining() < 30:
            break
        scored = []
        for q in population:
            c = score_q(q, n)
            scored.append((c, q))
        scored.sort(key=lambda x: x[0])
        population = [s[1] for s in scored]
        if population[0][1] is not None:
            c_first = scored[0][0]
            if c_first < c5_best:
                c5_best = c_first
                q_best = population[0].copy()
                h_best = make_sym_h(q_best, n)

        parents = population[:pop_size // 2]
        new_pop = list(parents)
        for i in range(pop_size // 2):
            if remaining() < 15:
                break
            pa = parents[i % len(parents)]
            pb = parents[(i + 2) % len(parents)]
            alpha = rng.uniform(0.3, 0.7)
            q_child = alpha * pa + (1 - alpha) * pb
            mode = rng.randint(0, 5)
            q_child = structured_perturb(q_child, n, rng, mode=mode)
            q_child = smooth_optimize(q_child, n, 150, maxiter=100)
            new_pop.append(q_child)

        if len(new_pop) >= 2:
            q_rand = structured_perturb(q_best, n, rng, mode=rng.randint(0, 5))
            new_pop.append(q_rand)
        population = new_pop[:pop_size]

    # Phase 5: Multi-resolution search
    res_list = [100, 110, 116, 120, 130, 132, 140, 150, 160]
    res_list = [r for r in res_list if r != n and r % 2 == 0 and r >= 60]
    for n_try in res_list:
        if remaining() < 35:
            break
        h_interp = interpolate_h(h_best, n, n_try)
        m_try = n_try // 2
        q_interp = make_q_from_h(h_interp, n_try)

        q_a = smooth_optimize(q_interp, n_try, 100, maxiter=150)
        q_b = smooth_optimize(q_a, n_try, 400, maxiter=200)
        corrs_b = corr_vals(make_sym_h(q_b, n_try))
        active_b = top_lags(corrs_b, 40)
        q_c, c5_c = epigraph_optimize(q_b, n_try, active_b, corrs_b.max() * 1.0005, maxiter=300)
        if q_c is not None:
            q_final_try = q_c
            c5_try = c5_c
        else:
            q_final_try = q_b
            c5_try = score_q(q_final_try, n_try)

        if c5_try < c5_best - 1e-9:
            c5_best = c5_try
            n = n_try
            m = m_try
            q_best = q_final_try
            h_best = make_sym_h(q_best, n)

    # Phase 6: Final intensive polish
    if remaining() > 20:
        q_best, c5_best = full_polish(q_best, n, time_limit=min(remaining() - 5, 60))
        h_best = make_sym_h(q_best, n)

    # Phase 7: One more epigraph round with large active set
    if remaining() > 15:
        corrs = corr_vals(h_best)
        active = top_lags(corrs, min(80, 2 * n - 1))
        q_new, c5_new = epigraph_optimize(q_best, n, active, corrs.max() * 1.0001, maxiter=500)
        if q_new is not None and c5_new < c5_best - 1e-12:
            c5_best = c5_new
            q_best = q_new
            h_best = make_sym_h(q_best, n)

    # Final cleanup
    h_best = np.clip(h_best, 0.0, 1.0)
    c5_best = exact_score(h_best)

    return h_best, c5_best, len(h_best)
