    dx = 2.0 / n
    if d >= 0:
        if d >= n:
            return 0.0
        return np.sum(h[d:] * (1.0 - h[:n-d])) * dx
    else:
        dd = -d
        if dd >= n:
            return 0.0
        return np.sum(h[dd:] * (1.0 - h[:n-dd])) * dx

    g = np.zeros(n)

    if d >= 0:

            g[:n-d] += (-h[d:])
    else:
        dd = -d
        if dd < n:
            g[:n-dd] += (1.0 - h[dd:])
            g[dd:] += (-h[:n-dd])

- After each L-BFGS-B call, project onto the feasible set
- Actually, L-BFGS-B doesn't support custom projections, so we need to handle this differently

For L-BFGS-B, the constraints are only box (lower/upper bounds). So for the full-space problem, we can:
1. Use the smooth objective
2. Use box constraints [0, 1]
3. After convergence, project onto sum(h) = n/2
4. Repeat (use the projection as new starting point)

Or better: incorporate the sum constraint into the objective via a penalty:
f(h) = softmax(corr(h)) + μ * (sum(h) - n/2)²

With L-BFGS-B and bounds [0, 1], this should work.

Let me write the final implementation. I'll focus on:
1. Robust q-space optimization (L-BFGS-B, increasing beta) - this is the main workhorse
2. Full-space optimization with penalty for sum constraint - to check if symmetry is suboptimal
3. Active set refinement - to get the last bit of improvement
4. Multi-resolution - try different n values

import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time


def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corrs = np.correlate(h, 1.0 - h, mode='full')
    return corrs.max() * dx, corrs


def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    mask = (h > 1e-10) & (h < 1.0 - 1e-10)
    if mask.sum() > 0 and abs(diff) > 1e-12:
        h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)


def build_h_from_q(q, n):
    m = n // 2
    return np.concatenate([q, 1.0 - q[::-1]])


def q_from_h(h, n):
    m = n // 2
    return h[:m].copy()


def corr_grad_single(h, d, n):
    g = np.zeros(n)
    if d >= 0:
        if d < n:
            g[d:] += (1.0 - h[:n - d])
            g[:n - d] += (-h[d:])
    else:
        dd = -d
        if dd < n:
            g[:n - dd] += (1.0 - h[dd:])
            g[dd:] += (-h[:n - dd])
    return g


def smooth_q_obj_grad(q, n, beta):
    m = n // 2
    h = build_h_from_q(q, n)
    dx = 2.0 / n
    corrs = np.correlate(h, 1.0 - h, mode='full')
    val = logsumexp(beta * corrs) / beta * dx
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(len(corrs)) / len(corrs)
    else:
        w /= w_sum
    G = np.zeros(n)
    for k in range(len(corrs)):
        if w[k] < 1e-14:
            continue
        d = k - (n - 1)
        G += w[k] * corr_grad_single(h, d, n)
    Gq = (G[:m] - G[m:][::-1]) * dx
    return val, Gq


def smooth_full_obj_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corrs = np.correlate(h, 1.0 - h, mode='full')
    val = logsumexp(beta * corrs) / beta * dx
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(len(corrs)) / len(corrs)
    else:
        w /= w_sum
    G = np.zeros(n)
    for k in range(len(corrs)):
        if w[k] < 1e-14:
            continue
        d = k - (n - 1)
        G += w[k] * corr_grad_single(h, d, n)
    return val, G * dx


def penalty_full_obj_grad(h, n, beta, mu, target_sum):
    val_s, grad_s = smooth_full_obj_grad(h, beta)
    s = h.sum() - target_sum
    val = val_s + mu * s * s
    grad = grad_s + 2.0 * mu * s * np.ones(n)
    return val, grad


def solve_q_lbfgsb(q0, n, beta, maxiter=2000):
    m = n // 2
    try:
        res = minimize(smooth_q_obj_grad, q0, args=(n, beta), jac=True,
                       method='L-BFGS-B', bounds=[(0.0, 1.0)] * m,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10})
        return np.clip(res.x, 0.0, 1.0)
    except Exception:
        return q0.copy()


def solve_full_penalty(h0, n, beta, mu, target_sum, maxiter=1000):
    try:
        res = minimize(penalty_full_obj_grad, h0,
                       args=(n, beta, mu, target_sum), jac=True,
                       method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-8})
        h_out = project_feasible(res.x, target_sum)
        return h_out
    except Exception:
        return project_feasible(h0, target_sum)


def epigraph_q_solve(q0, n, active_k, t0, maxiter=500):
    m = n // 2
    dx = 2.0 / n

    def obj(x):
        return x[0]

    def obj_grad(x):
        g = np.zeros(m + 1)
        g[0] = 1.0
        return g

    def make_cons(k_idx):
        k = active_k[k_idx]
        d = k - (n - 1)

        def cfun(x):
            q = x[1:]
            h = build_h_from_q(q, n)
            if d >= 0:
                val = np.sum(h[d:] * (1.0 - h[:n - d]))
            else:
                dd = -d
                val = np.sum(h[dd:] * (1.0 - h[:n - dd]))
            return x[0] - val * dx

        def cgfun(x):
            q = x[1:]
            h = build_h_from_q(q, n)
            g = np.zeros(m + 1)
            g[0] = 1.0
            G = corr_grad_single(h, d, n)
            Gq = (G[:m] - G[m:][::-1]) * dx
            g[1:] = -Gq
            return g

        return {'type': 'ineq', 'fun': cfun, 'jac': cgfun}

    x0 = np.concatenate([[t0], q0])
    cons = [make_cons(i) for i in range(len(active_k))]
    bounds = [(0.0, 5.0)] + [(0.0, 1.0)] * m
    try:
        res = minimize(obj, x0, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        q_out = np.clip(res.x[1:], 0.0, 1.0)
        return q_out, res.x[0], res
    except Exception:
        return q0.copy(), t0, None


def active_set_refine(q, n, time_limit, max_active=60):
    m = n // 2
    dx = 2.0 / n
    best_q = q.copy()
    h = build_h_from_q(q, n)
    corrs = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = corrs.max() * dx
    t0 = best_c5 * 1.01 + 1e-5

    top_k = min(max_active, len(corrs))
    active_k = sorted(np.argsort(corrs)[::-1][:top_k].tolist())

    for rnd in range(12):
        h = build_h_from_q(q, n)
        corrs = np.correlate(h, 1.0 - h, mode='full')
        cur_max = corrs.max() * dx
        if cur_max < best_c5:
            best_c5 = cur_max
            best_q = q.copy()

        q_new, t_sol, res = epigraph_q_solve(q, n, active_k, t0, maxiter=400)
        h_new = build_h_from_q(q_new, n)
        corrs_new = np.correlate(h_new, 1.0 - h_new, mode='full')
        c5_new = corrs_new.max() * dx

        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_q = q_new.copy()

        hidden = corrs_new > t_sol + 1e-7
        if hidden.sum() > 0:
            hidden_idx = np.where(hidden)[0]
            new_active = sorted(set(active_k) | set(hidden_idx.tolist()))
            if len(new_active) > max_active:
                vals = [corrs_new[kk] for kk in new_active]
                order = np.argsort(vals)[::-1][:max_active]
                new_active = sorted([new_active[i] for i in order])
            active_k = new_active

        t0 = max(t_sol, c5_new) * 1.001 + 1e-6
        q = q_new

        if abs(c5_new - best_c5) < 1e-12 and hidden.sum() == 0:
            break

    return best_q, best_c5
