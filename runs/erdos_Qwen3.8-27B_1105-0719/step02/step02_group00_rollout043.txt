import numpy as np
from scipy.optimize import minimize
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def get_profile(h):
    n = len(h)
    return np.correlate(h, 1 - h, mode="full") * (2.0 / n)

def project(h, n):
    target = n / 2.0
    h = np.asarray(h, dtype=np.float64)
    a, b = h.min() - 1.0, h.max() + 1.0
    for _ in range(60):
        m = 0.5 * (a + b)
        s = np.clip(h - m, 0.0, 1.0).sum()
        if s > target:
            a = m
        else:
            b = m
    hp = np.clip(h - 0.5 * (a + b), 0.0, 1.0)
    diff = target - hp.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - hp) if diff > 0 else np.argsort(hp)
        rem = diff
        for idx in order:
            if rem > 0:
                d = min(rem, 1.0 - hp[idx])
                hp[idx] += d
                rem -= d
            else:
                d = min(-rem, hp[idx])
                hp[idx] -= d
                rem += d
            if abs(rem) < 1e-12:
                break
    return hp

def row_grad(h, d, n, dx):
    g = np.zeros(n)
    if d == 0:
        g[:] = dx * (1.0 - 2.0 * h)
    elif d > 0:
        g[:n - d] += dx * (1.0 - h[:n - d])
        g[d:] += dx * (1.0 - h[:n - d])
    else:
        dp = -d
        g[dp:] += dx * (1.0 - h[dp:])
        g[:n - dp] += dx * (1.0 - h[dp:])
    return g

def lse_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    P = get_profile(h)
    Pmax = P.max()
    s = np.clip(beta * (P - Pmax), -400, 400)
    w = np.exp(s)
    w /= w.sum()
    center = n - 1
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-10:
            continue
        grad += w[k] * row_grad(h, k - center, n, dx)
    return grad - grad.mean()

def soft_nesterov(h0, beta, iters=300, lr0=0.008):
    n = len(h0)
    h = h0.astype(np.float64).copy()
    best_h, best_c = h.copy(), exact_score(h)
    y = h.copy()
    t = 1.0
    lr = lr0
    for it in range(iters):
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        mu = (t - 1.0) / t_new
        y = h + mu * (h - y)
        t = t_new
        grad = lse_grad(y, beta)
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        g_dir = grad / gn
        step = lr
        cur = exact_score(h)
        ok = False
        for _ in range(15):
            h_try = project(h - step * g_dir, n)
            c = exact_score(h_try)
            if c < cur - 1e-14:
                h, cur = h_try, c
                if c < best_c:
                    best_c, best_h = c, h.copy()
                ok = True
                break
            step *= 0.5
        if not ok:
            lr *= 0.75
            if lr < 1e-9:
                break
        else:
            lr = min(step * 1.3, 0.06)
    return best_h, best_c

def slsqp_topk(h0, K=10, iters=300):
    n = len(h0)
    dx = 2.0 / n
    P = get_profile(h0)
    idx = np.argsort(P)[-K:]
    c0 = exact_score(h0)

    def obj_fn(x):
        return x[-1]

    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g

    def eq_fn(x):
        return x[:-1].sum() - n / 2.0

    def eq_jac_fn(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g

    def ineq_fn(x):
        h = x[:-1]
        t = x[-1]
        Pk = np.correlate(h, 1 - h, mode="full") * dx
        return t - Pk[idx]

    def ineq_jac_fn(x):
        h = x[:-1]
        J = np.zeros((K, n + 1))
        c = n - 1
        for i, k in enumerate(idx):
            d = k - c
            J[i, :-1] = -row_grad(h, d, n, dx)
            J[i, -1] = 1.0
        return J

    x0 = np.append(h0.copy(), c0 + 1e-3)
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 0.5)]
    cons = [
        {"type": "eq", "fun": eq_fn, "jac": eq_jac_fn},
        {"type": "ineq", "fun": ineq_fn, "jac": ineq_jac_fn}
    ]
    try:
        res = minimize(obj_fn, x0, method="SLSQP", jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={"maxiter": iters, "ftol": 1e-13})
        if np.isfinite(res.x).all():

            c_new = exact_score(h_new)
            if c_new <= c0 + 1e-8:
                return h_new
    except Exception:
        pass
    return None

    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    fx = np.concatenate([[0.0], xs, [2.0]])
    fv = np.concatenate([[h[0]], h, [h[-1]]])
    return project(np.interp(xt, fx, fv), new_n)

    n = len(h)
    dx = 2.0 / n
    P = get_profile(h)
    k_max = int(np.argmax(P))
    d = k_max - (n - 1)
    g = row_grad(h, d, n, dx)
    g -= g.mean()
    return project(h - alpha * g, n)

def anneal_descent(h0, beta, iters=500, lr0=0.005, temp0=0.01, decay=0.995):
    n = len(h0)
    h = h0.astype(np.float64).copy()
    best_h, best_c = h.copy(), exact_score(h)
    temp = temp0
    for it in range(iters):
        grad = lse_grad(h, beta)
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        g_dir = grad / gn
        step = lr0 * temp
        h_try = project(h - step * g_dir, n)
        c_try = exact_score(h_try)
        c_cur = exact_score(h)
        if c_try < c_cur:
            h = h_try
            if c_try < best_c:
                best_c, best_h = c_try, h.copy()
        else:
            accept = np.exp(-(c_try - c_cur) / max(temp, 1e-10))
            if np.random.random() < accept:
                h = h_try
        temp *= decay
        if temp < 1e-6:
            break
    return best_h, best_c
