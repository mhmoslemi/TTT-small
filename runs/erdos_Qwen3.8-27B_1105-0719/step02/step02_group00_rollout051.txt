                remaining -= add
            if remaining <= 1e-14:

        b_vals.append(h[i])
        row2 = np.zeros(num_vars)
        row2[i] = -1.0
        row2[n+1+i] = -1.0
        A_rows.append(row2)
        b_vals.append(-h[i])

    A_ub = np.array(A_rows)
    b_ub = np.array(b_vals)

    A_eq = np.zeros((1, num_vars))
    A_eq[0, 0:n] = 1.0
    b_eq = np.array([n / 2.0])

    bounds = []
    for i in range(n):
        bounds.append((max(0.0, h[i] - delta), min(1.0, h[i] + delta)))
    bounds.append((0.0, None))
    for i in range(n):
        bounds.append((0.0, None))

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            return repair_feasible(res.x[0:n])
    except Exception:
        pass
    return None

def active_set_from_profile(profile, max_k=20):
    n = len(profile)
    n_half = n // 2 + 1
    c_max = np.max(profile)
    spread = c_max - np.median(profile)
    beam = max(0.003, min(0.015, 0.2 * spread))
    mask = profile >= c_max - beam
    indices = np.where(mask)[0]
    if len(indices) > max_k:
        vals = profile[indices]
        thresh = np.sort(vals)[-max_k]
        indices = indices[profile[indices] >= thresh]
    lags = indices - (len(profile) // 2)
    return lags, c_max

def gradient_direction(h, active_lags, profile, c_max, n, dx):
    spread = c_max - np.median(profile)
    beam = max(0.003, min(0.015, 0.2 * spread))
    mask = np.zeros(len(profile), dtype=bool)
    for lag in active_lags:
        m = lag + (n - 1)
        if 0 <= m < len(profile):
            mask[m] = True
    vals = profile[mask]
    if len(vals) == 0:
        return None
    weights = vals - (c_max - beam)
    weights = np.maximum(weights, 0.01)
    weights /= weights.sum()
    grad = np.zeros(n)
    for w, lag in zip(weights, active_lags):
        grad += w * profile_gradient(h, int(lag), n, dx)
    grad -= grad.mean()
    gnorm = np.linalg.norm(grad)
    if gnorm < 1e-14:
        return None
    return grad / gnorm

def optimize_from_start(h0, time_limit, t_ref):
    n = len(h0)
    dx = 2.0 / n
    best_h = h0.copy()
    best_c = exact_score(best_h)
    no_progress = 0

    for _ in range(500):
        if time.perf_counter() - t_ref > time_limit:
            break
        profile = compute_profile(best_h)
        c_cur = np.max(profile)
        active_lags, _ = active_set_from_profile(profile, max_k=20)

        accepted = False
        deltas = [0.05, 0.03, 0.02, 0.01, 0.005, 0.003]
        for d in deltas:
            h_new = lp_regularized(best_h, active_lags, d, rho=0.005)
            if h_new is not None:
                c_new = exact_score(h_new)
                if c_new < best_c - 1e-13:
                    best_h = h_new
                    best_c = c_new
                    no_progress = 0
                    accepted = True
                    break
            h_new2 = lp_active_step(best_h, active_lags, d, rho=0.0)
            if h_new2 is not None:
                c_new2 = exact_score(h_new2)
                if c_new2 < best_c - 1e-13:
                    best_h = h_new2
                    best_c = c_new2
                    no_progress = 0
                    accepted = True
                    break

        if not accepted:
            gdir = gradient_direction(best_h, active_lags, profile, c_cur, n, dx)
            if gdir is not None:
                for alpha in [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]:
                    h_try = repair_feasible(best_h - alpha * gdir)
                    c_try = exact_score(h_try)
                    if c_try < best_c - 1e-14:
                        best_h = h_try
                        best_c = c_try
                        no_progress = 0
                        accepted = True
                        break

        if not accepted:
            no_progress += 1
            if no_progress >= 5:
                break

    return best_h, best_c

def make_start_pattern(n, kind, seed):
    rng = np.random.RandomState(seed)
    if kind == 0:
        h = np.full(n, 0.5)
    elif kind == 1:
        x = np.linspace(0, 2, n, endpoint=False)
        h = 0.5 + 0.3 * np.sin(2 * np.pi * x)
    elif kind == 2:
        x = np.linspace(0, 2, n, endpoint=False)
        h = 0.5 + 0.35 * np.cos(4 * np.pi * x)
    elif kind == 3:
        block = max(1, n // 8)
        h = np.zeros(n)
        for b in range(8):
            s = min(b * block, n - 1)
            e = min((b + 1) * block, n)
            h[s:e] = 0.5 + 0.3 * np.sin(b * np.pi / 4)
    elif kind == 4:
        h = rng.uniform(0.1, 0.9, n)
    elif kind == 5:
        h = (rng.rand(n) > 0.5).astype(float)
        h = 0.15 + 0.7 * h
    elif kind == 6:
        x = np.linspace(0, 2, n, endpoint=False)
        h = 0.5 + 0.25 * np.sin(3 * np.pi * x) + 0.15 * np.cos(5 * np.pi * x)
    elif kind == 7:
        h = np.zeros(n)
        h[:n//2] = 0.7
        h[n//2:] = 0.3
    else:
        half = n // 2
        y = rng.uniform(0.1, 0.9, half)
        h = np.zeros(n)
        h[:half] = y
        h[half:] = 1.0 - y[::-1]
    return repair_feasible(np.clip(h, 0.0, 1.0))

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()

    def remaining():
        return budget_s - (time.perf_counter() - t_start)

    try:
        parent = np.array(initial_h_values, dtype=np.float64)
    except (NameError, Exception):
        parent = None

    if parent is None or len(parent) < 4:
        parent = np.full(70, 0.5)

    parent = repair_feasible(np.clip(parent, 0.0, 1.0))
    n0 = len(parent)
    best_h = parent.copy()
    best_c = exact_score(best_h)

    h_sym = repair_feasible(symmetrize(parent))
    c_sym = exact_score(h_sym)
    if c_sym < best_c - 1e-12:
        best_h = h_sym.copy()
        best_c = c_sym

    h_rev = repair_feasible(parent[::-1].copy())
    c_rev = exact_score(h_rev)
    if c_rev < best_c - 1e-12:
        best_h = h_rev.copy()
        best_c = c_rev

    # Phase 1: Intensive LP optimization from parent
    t_phase1 = time.perf_counter()
    limit1 = min(remaining() * 0.3, 300)
    best_h, best_c = optimize_from_start(best_h, limit1, t_phase1)

    # Phase 2: Multi-start exploration
    if remaining() > 100:
        n_target = len(best_h)
        starts = []
        for kind in range(8):
            h_s = make_start_pattern(n_target, kind, seed + kind)
            c_s = exact_score(h_s)
            starts.append((c_s, h_s, kind))
        starts.sort(key=lambda x: x[0])

        for c_s, h_s, kind in starts:
            if remaining() < 40:
                break
            if c_s > best_c + 0.05:
                continue
            t_ps = time.perf_counter()
            limit_s = min(remaining() * 0.15, 80)
            h_opt, c_opt = optimize_from_start(h_s, limit_s, t_ps)
            if c_opt < best_c - 1e-12:
                best_h = h_opt
                best_c = c_opt

    # Phase 3: Symmetry-constrained optimization
    if remaining() > 80:
        n_s = len(best_h)
        if n_s % 2 == 0:
            half = n_s // 2
            y = best_h[:half].copy()
            h_sym_struct = np.zeros(n_s)
            h_sym_struct[:half] = y
            h_sym_struct[half:] = 1.0 - y[::-1]
            h_sym_struct = repair_feasible(h_sym_struct)
            c_ss = exact_score(h_sym_struct)
            if c_ss < best_c - 1e-12:
                best_h = h_sym_struct.copy()
                best_c = c_ss
            elif c_ss < best_c + 0.01:
                t_ps = time.perf_counter()
                limit_s = min(remaining() * 0.1, 60)
                h_opt, c_opt = optimize_from_start(h_sym_struct, limit_s, t_ps)
                if c_opt < best_c - 1e-12:
                    best_h = h_opt
                    best_c = c_opt

    # Phase 4: Dimension exploration
    if remaining() > 120:
        n_cur = len(best_h)
        dims = []
        for d in [n_cur - 6, n_cur - 4, n_cur - 2, n_cur + 2, n_cur + 4, n_cur + 6, n_cur + 10]:
            if 24 <= d <= 160 and d != n_cur:
                dims.append(d)
        for new_n in dims:
            if remaining() < 60:
                break
            h_int = interpolate_to_n(best_h, new_n)
            c_int = exact_score(h_int)
            if c_int < best_c + 0.02:
                t_pd = time.perf_counter()
                limit_d = min(remaining() * 0.15, 80)
                h_opt, c_opt = optimize_from_start(h_int, limit_d, t_pd)
                if c_opt < best_c - 1e-12:
                    best_h = h_opt
                    best_c = c_opt

    # Phase 5: Intensive polishing with small steps
    if remaining() > 30:
        t_polish = time.perf_counter()
        n_p = len(best_h)
        dx_p = 2.0 / n_p
        best_cp = best_c.copy()
        bp_h = best_h.copy()

        for _ in range(200):
            if time.perf_counter() - t_polish > remaining() - 10:
                break
            profile = compute_profile(bp_h)
            c_cur = np.max(profile)
            active_lags, _ = active_set_from_profile(profile, max_k=15)

            accepted = False
            for d in [0.01, 0.005, 0.003, 0.002, 0.001, 0.0005]:
                h_new = lp_regularized(bp_h, active_lags, d, rho=0.003)
                if h_new is not None:
                    c_new = exact_score(h_new)
                    if c_new < bp_h_score := exact_score(bp_h):
                        pass
                    if c_new < best_cp - 1e-14:
                        bp_h = h_new
                        best_cp = c_new
                        accepted = True
                        break

            if not accepted:
                gdir = gradient_direction(bp_h, active_lags, profile, c_cur, n_p, dx_p)
                if gdir is None:
                    break
                step_ok = False
                for alpha in [0.005, 0.003, 0.002, 0.001, 0.0005, 0.0002]:
                    h_try = repair_feasible(bp_h - alpha * gdir)
                    c_try = exact_score(h_try)
                    if c_try < best_cp - 1e-15:
                        bp_h = h_try
                        best_cp = c_try
                        step_ok = True
                        break
                if not step_ok:
                    break

        if best_cp < best_c - 1e-14:
            best_h = bp_h
            best_c = best_cp

    # Phase 6: Final micro-optimization
    if remaining() > 15:
        n_f = len(best_h)
        dx_f = 2.0 / n_f
        for _ in range(50):
            if remaining() < 5:
                break
            profile = compute_profile(best_h)
            c_cur = np.max(profile)
            active_lags, _ = active_set_from_profile(profile, max_k=12)
            improved = False
            for d in [0.003, 0.002, 0.001, 0.0005]:
                h_new = lp_active_step(best_h, active_lags, d)
                if h_new is not None:
                    c_new = exact_score(h_new)
                    if c_new < best_c - 1e-15:
                        best_h = h_new
                        best_c = c_new
                        improved = True
                        break
            if not improved:
                break

    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)

    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            final_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 2:
            final_c = min(final_c, float(eval_result[0]))
    except Exception:
        pass

    return best_h, final_c, len(best_h)
