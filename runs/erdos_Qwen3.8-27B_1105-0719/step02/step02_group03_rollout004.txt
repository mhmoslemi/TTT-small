    #   Wait, here i+e = i, so e=0? No. Let me re-index.
    #   If h[i] is in the first factor, then i = m + e for some m, so m = i - e.
    #   The term is h[m+e]*(1-h[m]) = h[i]*(1-h[i-e]).
    #   This is valid when 0 <= i-e < n, i.e., e <= i and e >= i-n+1.
    #   The change is dh * (1-h[i-e]).
    #
    # So total delta for h[i] changing by dh:
    #   For part A (h[i] in second factor): sum over e of -h[i+e]*dh where 0<=i+e<n
    #     = -dh * sum_{e: 0<=i+e<n} h[i+e]
    #     Each such term contributes to correlation at lag e.
    #     For lag e: position i+e in range means e is in [-i, n-1-i].
    #     The output index is p = e + (n-1).
    #     Delta at p = -dh * h[i+e] for e in range.
    #
    #   For part B (h[i] in first factor): sum over e of dh*(1-h[i-e]) where 0<=i-e<n
    #     = dh * sum_{e: 0<=i-e<n} (1-h[i-e])
    #     For lag e: position i-e in range means e is in [i-n+1, i].
    #     The output index is p = e + (n-1).
    #     Delta at p = dh * (1-h[i-e]) for e in range.
    
    # For h[i]: 1->0, dh = -1
    # Part A: delta at output p = -(-1)*h[i+e] = h[i+e] for e in [-i, n-1-i]
    # Part B: delta at output p = (-1)*(1-h[i-e]) = -(1-h[i-e]) = h[i-e]-1 for e in [i-n+1, i]
    
    # For h[j]: 0->1, dh = +1
    # Part A: delta at output p = -(+1)*h[j+e] = -h[j+e] for e in [-j, n-1-j]
    # Part B: delta at output p = (+1)*(1-h[j-e]) = 1-h[j-e] for e in [j-n+1, j]
    
    # Combined delta:
    # From h[i] removal (dh=-1):
    #   Part A: +h[i+e] for output index p = e+(n-1), e in [-i, n-1-i]
    #   Part B: -(1-h[i-e]) for output index p = e+(n-1), e in [i-n+1, i]
    # From h[j] addition (dh=+1):
    #   Part A: -h[j+e] for output index p = e+(n-1), e in [-j, n-1-j]
    #   Part B: +(1-h[j-e]) for output index p = e+(n-1), e in [j-n+1, j]
    
    # Let's implement this:
    # Part A for i: e from -i to n-1-i, output index = e + (n-1)
    e_lo = -i
    e_hi = n - 1 - i
    p_idx = np.arange(e_lo, e_hi + 1) + (n - 1)
    delta[p_idx] += h[i + np.arange(e_lo, e_hi + 1)]  # Part A: +h[i+e]
    
    # Part B for i: e from i-n+1 to i, output index = e + (n-1)
    e_lo = i - n + 1
    e_hi = i
    p_idx = np.arange(e_lo, e_hi + 1) + (n - 1)
    delta[p_idx] += -(1.0 - h[i - np.arange(e_lo, e_hi + 1)])  # Part B: -(1-h[i-e])
    
    # Part A for j: e from -j to n-1-j, output index = e + (n-1)
    e_lo = -j
    e_hi = n - 1 - j
    p_idx = np.arange(e_lo, e_hi + 1) + (n - 1)
    delta[p_idx] += -h[j + np.arange(e_lo, e_hi + 1)]  # Part A: -h[j+e]
    
    # Part B for j: e from j-n+1 to j, output index = e + (n-1)
    e_lo = j - n + 1
    e_hi = j
    p_idx = np.arange(e_lo, e_hi + 1) + (n - 1)
    delta[p_idx] += (1.0 - h[j - np.arange(e_lo, e_hi + 1)])  # Part B: +(1-h[j-e])
    
    return delta

def binary_local_search_v2(h0, dx, rng, max_iters=200000, patience=10000):
    """Simulated annealing on binary vectors with efficient delta computation."""
    n = len(h0)
    target_sum = int(round(n / 2))
    
    h = np.zeros(n)
    sorted_idx = np.argsort(-h0)
    h[sorted_idx[:target_sum]] = 1.0
    
    corrs = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = corrs.max() * dx
    best_h = h.copy()
    
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    
    no_improve = 0
    temp = 0.03
    cooling = 0.9995
    
    for it in range(max_iters):
        if it % 3000 == 0:
            temp *= cooling
        
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        if i == j:
            no_improve += 1
            continue
        
        delta = correlate_delta(h, i, j, n)
        new_corrs_max = (corrs + delta).max()
        new_c5 = new_corrs_max * dx
        
        if new_c5 <= best_c5 + 1e-15:
            h[i], h[j] = 0.0, 1.0
            corrs = corrs + delta
            best_c5 = new_c5
            best_h = h.copy()
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            no_improve = 0
        else:
            delta_val = new_c5 - best_c5
            if temp > 1e-9 and rng.random() < np.exp(-delta_val / temp):
                h[i], h[j] = 0.0, 1.0
                corrs = corrs + delta
                ones = np.where(h == 1.0)[0]
                zeros = np.where(h == 0.0)[0]
                no_improve = 0
            else:
                no_improve += 1
        
        if no_improve > patience:
            break
    
    return best_h, best_c5

def steepest_descent_binary(h0, dx, max_sweeps=30):
    """Try all swaps per sweep, apply the single best."""
    n = len(h0)
    target_sum = int(round(n / 2))
    
    h = np.zeros(n)
    sorted_idx = np.argsort(-h0)
    h[sorted_idx[:target_sum]] = 1.0
    
    corrs = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = corrs.max() * dx
    best_h = h.copy()
    
    for sweep in range(max_sweeps):
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        best_swap = None
        best_new_max = corrs.max()
        
        for i in ones:
            for j in zeros:
                delta = correlate_delta(h, i, j, n)
                new_max = (corrs + delta).max()
                if new_max < best_new_max - 1e-14:
                    best_new_max = new_max
                    best_swap = (i, j)
        
        if best_swap is None:
            break
        
        i, j = best_swap
        h[i], h[j] = 0.0, 1.0
        corrs = np.correlate(h, 1.0 - h, mode='full')
        new_c5 = corrs.max() * dx
        if new_c5 < best_c5 - 1e-15:
            best_c5 = new_c5
            best_h = h.copy()
        else:
            break
    
    return best_h, best_c5

def golden_ratio_binary(n, k, offset=0):
    """Generate binary sequence using golden ratio distribution."""
    h = np.zeros(n)
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    for m in range(k):
        pos = int((m * phi + offset) * n / k) % n
        h[pos] = 1.0
    return h

def random_binary(n, k, rng):
    h = np.zeros(n)
    idx = rng.choice(n, size=k, replace=False)
    h[idx] = 1.0
    return h

def subgradient_step(h, dx, target_sum, step_size, max_iter=1000):
    """Direct subgradient method for min max_k c_k."""
    n = len(h)
    h = h.copy()
    best_c5 = np.inf
    best_h = h.copy()
    
    for it in range(max_iter):
        corrs = np.correlate(h, 1.0 - h, mode='full')
        max_idx = np.argmax(corrs)
        c5 = corrs[max_idx] * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Subgradient: average over all active (near-max) correlations
        threshold = corrs[max_idx] - 0.5
        active = np.where(corrs >= threshold)[0]
        
        grad = np.zeros(n)
        for k in active:
            grad += grad_corr_k(h, k, n)
        grad /= len(active)
        
        # Step in negative gradient direction
        h_new = h - step_size * grad
        h_new = project_feasible(h_new, target_sum)
        
        # Line search
        for alpha in [1.0, 0.5, 0.25, 0.1]:
            h_try = h - alpha * step_size * grad
            h_try = project_feasible(h_try, target_sum)
            c5_try = c5_of_h(h_try, dx)
            if c5_try < c5:
                h = h_try
                break
        else:
            break
        
        step_size *= 0.99
    
    return best_h, best_c5

def generate_starts(n, k, rng):
    """Generate diverse binary starting points."""
    starts = []
    
    # Golden ratio with different offsets
    for off in [0, 0.1, 0.3, 0.5, 0.7]:
        starts.append(golden_ratio_binary(n, k, off))
    
    # Random
    for _ in range(5):
        starts.append(random_binary(n, k, rng))
    
    # Block-based: alternate blocks of different sizes
    # Try period constructions
    for period in [3, 4, 5, 7]:
        h = np.zeros(n)
        # Place ones at positions that are multiples of period, or with some pattern
        for m in range(k):
            pos = (m * period) % n
            h[pos] = 1.0
        if h.sum() == k:
            starts.append(h)
    
    # Symmetric construction: mirror
    half_n = n // 2
    half_k = k // 2
    h_half = np.zeros(half_n)
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    for m in range(half_k):
        pos = int((m * phi) * half_n / half_k) % half_n
        h_half[pos] = 1.0
    h = np.zeros(n)
    h[:half_n] = h_half
    h[half_n:] = h_half[::-1]
    if h.sum() == k:
        starts.append(h)
    else:
        # Fix
        h = np.zeros(n)
        sorted_idx = np.argsort(-h)
        h[sorted_idx[:k]] = 1.0
    
    return starts
