Now, for the symmetric gradient, if I have a function $F(q) = f(h)$ where $h = \text{build\_sym}(q)$, then:
$\frac{\partial F}{\partial q[i]} = \frac{\partial f}{\partial h[i]} + \frac{\partial f}{\partial h[n-1-i]} \cdot \frac{\partial h[n-1-i]}{\partial q[i]}$

Since $h[n-1-i] = 1 - q[i]$, we have $\frac{\partial h[n-1-i]}{\partial q[i]} = -1$.

So: $\frac{\partial F}{\partial q[i]} = \frac{\partial f}{\partial h[i]} - \frac{\partial f}{\partial h[n-1-i]}$

For the full gradient $\nabla_h f$, the symmetric gradient is:
$(\nabla_q F)[i] = (\nabla_h f)[i] - (\nabla_h f)[n-1-i]$

For odd $n$, the middle element $h[m] = 0.5$ is fixed, so it doesn't contribute.

For $f_k = \sum_i h[i](1-h[i-d]) \cdot dx$ where $d = k - (n-1)$:

When $d \geq 0$:
- Valid $i$: from $\max(0, d)$ to $\min(n-1, n-1+d) = n-1$ (since $d \geq 0$)
- So $i$ ranges from $d$ to $n-1$
- $f_k = \sum_{i=d}^{n-1} h[i](1-h[i-d]) \cdot dx$
- $\frac{\partial f_k}{\partial h[j]} = dx \cdot [(1-h[j-d]) \cdot \mathbf{1}[j \geq d] - h[j+d] \cdot \mathbf{1}[j+d \leq n-1]]$
- For $j \in [d, n-1]$: first term is $(1-h[j-d])$
- For $j \in [0, n-1-d]$: second term is $-h[j+d]$

So:
```
g[j] += dx * (1-h[j-d])  for j in [d, n-1]
g[j] -= dx * h[j+d]      for j in [0, n-1-d]
```

In vectorized form:
```
g[d:] += 1-h[:n-d]       # j=d..n-1, j-d=0..n-1-d
g[:n-d] -= h[d:]         # j=0..n-1-d, j+d=d..n-1
```

When $d < 0$ (let $dd = -d > 0$):
- Valid $i$: from $\max(0, -dd) = 0$ to $\min(n-1, n-1-dd) = n-1-dd$
- So $i$ ranges from $0$ to $n-1-dd$
- $f_k = \sum_{i=0}^{n-1-dd} h[i](1-h[i+dd]) \cdot dx$
- $\frac{\partial f_k}{\partial h[j]} = dx \cdot [(1-h[j+dd]) \cdot \mathbf{1}[j+dd \leq n-1] - h[j-dd] \cdot \mathbf{1}[j-dd \geq 0]]$
- For $j \in [0, n-1-dd]$: first term is $(1-h[j+dd])$
- For $j \in [dd, n-1]$: second term is $-h[j-dd]$

In vectorized form:
```
g[:n-dd] += 1-h[dd:]     # j=0..n-1-dd, j+dd=dd..n-1
g[dd:] -= h[:n-dd]       # j=dd..n-1, j-dd=0..n-1-dd
```

```python
import time


def eval_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))


def project_to_target(x, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(x - mid, 0.0, 1.0).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.any():
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)


def corr_vals(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)


def grad_fk(h, d, n):
    dx = 2.0 / n
    g = np.zeros(n)
    if d >= 0:
        g[d:] += 1.0 - h[:n - d]
        g[:n - d] -= h[d:]
    else:

        g[:n - dd] += 1.0 - h[dd:]
        g[dd:] -= h[:n - dd]
    g *= dx
    return g

    m = n // 2
    return gh[:m] - gh[n - 1:m - 1:-1]

            continue
        d = k - nm1
        gh += w[k] * grad_fk(h, d, n)
    gq = sym_grad_full(gh, n)
```

```python
    nm1 = n - 1
    for i in range(len(ak)):
        if exc[i] > 0.0:
            d = int(ak[i]) - nm1
            gh += mu * exc[i] * grad_fk(h, d, n)
    gf = np.zeros(n + 1)
    gf[:n] = gh + mu_sum * sdev
    gf[n] = dfdt
    return f, gf


def opt_lse(q0, n, eps, maxiter=200, gtol=1e-10):
    m = n // 2
    bounds = [(0.0, 1.0)] * m
    try:
        res = minimize(lse_obj_grad, q0, args=(n, eps), jac=True,
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'gtol': gtol, 'ftol': 1e-15})
        return np.clip(res.x, 0.0, 1.0)
    except Exception:
        return q0


def opt_al(q0, n, ak, mu, t_init=None, maxiter=200, gtol=1e-11):
    m = n // 2
    h0 = build_sym(q0, n)
    if t_init is None:
        t_init = eval_c5(h0)
    qt0 = np.concatenate([q0, [t_init]])
    bounds = [(0.0, 1.0)] * m + [(0.0, t_init + 1.0)]
    try:
        res = minimize(al_obj_grad, qt0, args=(n, ak, mu), jac=True,
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'gtol': gtol, 'ftol': 1e-15})
        return np.clip(res.x[:m], 0.0, 1.0)
    except Exception:
        return q0


def opt_full(h0, n, ak, mu, mu_sum=5000.0, maxiter=150, gtol=1e-11):
    t0 = eval_c5(h0)
    z0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, t0 + 1.0)]
    try:
        res = minimize(full_al_obj_grad, z0, args=(n, ak, mu, mu_sum),
                       jac=True, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'gtol': gtol, 'ftol': 1e-15})
        h_opt = np.clip(res.x[:n], 0.0, 1.0)
        h_opt = project_to_target(h_opt, n / 2.0)
        return h_opt
    except Exception:
        return h0


def interp_to_n(h, n_new):
    n_old = len(h)
    if n_new == n_old:
        return h.copy()
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h)
    return project_to_target(h_new, n_new / 2.0)


def remaining_time(t0, budget_s):
    return budget_s - (time.time() - t0)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    reserve = max(20.0, 0.03 * budget_s)
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')

    try:
        h_init = np.array(initial_h_values, dtype=float).reshape(-1)
        if h_init.size > 0 and np.all(np.isfinite(h_init)):
            n_init = len(h_init)
            h_init = project_to_target(h_init, n_init / 2.0)
            best_h = h_init.copy()
            best_c5 = eval_c5(h_init)
    except Exception:
        pass

    if best_h is None:
        n_def = 126
        m_def = n_def // 2
        tt = np.linspace(0.0, 1.0, m_def)
        q_def = 0.5 + 0.15 * np.sin(2.0 * np.pi * 3.0 * tt) + 0.1 * np.sin(2.0 * np.pi * 7.0 * tt)
        q_def = np.clip(q_def, 0.05, 0.95)
        best_h = build_sym(q_def, n_def)
        best_c5 = eval_c5(best_h)

    # Symmetrize
    n_cur = len(best_h)
    if n_cur % 2 != 0:
        n_cur -= 1
        best_h = interp_to_n(best_h, n_cur)
    q = to_sym_q(best_h, n_cur)
    h_sym = build_sym(q, n_cur)
    c5_s = eval_c5(h_sym)
    if c5_s < best_c5:
        best_h = h_sym.copy()
        best_c5 = c5_s

    # Phase 1: LSE cooling for basin attraction
    for eps in [0.05, 0.02, 0.008, 0.003, 0.001]:
        if remaining_time(t0, budget_s) < reserve + 30:
            break
        q = opt_lse(q, n_cur, eps, maxiter=150, gtol=1e-10)
        h_new = build_sym(q, n_cur)
        c5_new = eval_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new

    # Phase 2: Active-set AL with increasing mu
    cv = corr_vals(best_h)
    active = select_active(cv, margin=0.005, pad=4)
    t_cur = best_c5

    for mu in [10.0, 50.0, 200.0, 1000.0, 5000.0, 20000.0, 50000.0]:
        if remaining_time(t0, budget_s) < reserve + 25:
            break
        mi = 150 if mu < 1000 else 100
        q = opt_al(q, n_cur, active, mu, t_init=t_cur, maxiter=mi, gtol=1e-11)
        h_new = build_sym(q, n_cur)
        c5_new = eval_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            t_cur = c5_new
            cv = corr_vals(best_h)
            new_act = select_active(cv, margin=0.003, pad=3)
            active = np.union1d(active, new_act)

    # Phase 3: Tight active-set refinement with high mu
    for _rnd in range(4):
        if remaining_time(t0, budget_s) < reserve + 20:
            break
        cv = corr_vals(best_h)
        act_tight = select_active(cv, margin=0.001, pad=2)
        for mu in [5000.0, 50000.0, 200000.0]:
            if remaining_time(t0, budget_s) < reserve + 12:
                break
            q = opt_al(q, n_cur, act_tight, mu, t_init=best_c5, maxiter=120, gtol=1e-12)
            h_new = build_sym(q, n_cur)
            c5_new = eval_c5(h_new)
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new

    # Phase 4: Try nearby n values
    n_base = len(best_h)
    n_cands = sorted(set([n_base - 6, n_base - 4, n_base - 2, n_base + 2, n_base + 4, n_base + 6, n_base + 8]))
    n_cands = [nc for nc in n_cands if nc >= 40 and nc % 2 == 0]

    for n_try in n_cands:
        if remaining_time(t0, budget_s) < reserve + 30:
            break
        h_try = interp_to_n(best_h, n_try)
        q_try = to_sym_q(h_try, n_try)
        q_try = opt_lse(q_try, n_try, 0.01, maxiter=60, gtol=1e-9)
        h_try = build_sym(q_try, n_try)
        cv_try = corr_vals(h_try)
        act_try = select_active(cv_try, margin=0.003, pad=3)
        for mu in [200.0, 5000.0, 50000.0]:
            if remaining_time(t0, budget_s) < reserve + 15:
                break
            q_try = opt_al(q_try, n_try, act_try, mu, t_init=eval_c5(h_try), maxiter=80, gtol=1e-10)
            h_try = build_sym(q_try, n_try)
            c5_try = eval_c5(h_try)
            if c5_try < best_c5:
                best_h = h_try.copy()
                best_c5 = c5_try
                act_try = select_active(corr_vals(h_try), margin=0.002, pad=3)

    # Phase 5: Full non-symmetric optimization
    if remaining_time(t0, budget_s) > reserve + 40:
        n_f = len(best_h)
        h_f = best_h.copy()
        cv_f = corr_vals(h_f)
        act_f = select_active(cv_f, margin=0.003, pad=3)
        for mu in [100.0, 2000.0, 20000.0]:
            if remaining_time(t0, budget_s) < reserve + 20:
                break
            h_f2 = opt_full(h_f, n_f, act_f, mu, mu_sum=5000.0, maxiter=100, gtol=1e-11)
            c5_f2 = eval_c5(h_f2)
            if c5_f2 < best_c5:
                best_h = h_f2.copy()
                best_c5 = c5_f2
                h_f = h_f2.copy()
                act_f = select_active(corr_vals(h_f), margin=0.002, pad=3)

    # Phase 6: Final high-precision symmetric polish
    if remaining_time(t0, budget_s) > reserve + 10:
        n_fin = len(best_h)
        if n_fin % 2 != 0:
            n_fin -= 1
            best_h = interp_to_n(best_h, n_fin)
        q_fin = to_sym_q(best_h, n_fin)
        cv_fin = corr_vals(best_h)
        act_fin = select_active(cv_fin, margin=0.001, pad=2)
        for mu in [50000.0, 200000.0, 500000.0]:
            if remaining_time(t0, budget_s) < reserve + 8:
                break
            q_fin = opt_al(q_fin, n_fin, act_fin, mu, t_init=best_c5, maxiter=200, gtol=1e-13)
            h_fin = build_sym(q_fin, n_fin)
            c5_fin = eval_c5(h_fin)
            if c5_fin < best_c5:
                best_h = h_fin.copy()
                best_c5 = c5_fin
                act_fin = select_active(corr_vals(h_fin), margin=0.001, pad=2)

    # Phase 7: Very tight final squeeze
    if remaining_time(t0, budget_s) > reserve + 5:
        n_sq = len(best_h)
        if n_sq % 2 != 0:
            n_sq -= 1
            best_h = interp_to_n(best_h, n_sq)
        q_sq = to_sym_q(best_h, n_sq)
        cv_sq = corr_vals(best_h)
        mx = cv_sq.max()
        act_sq = select_active(cv_sq, margin=0.0003, pad=1)
        for mu in [500000.0, 2000000.0]:
            if remaining_time(t0, budget_s) < reserve + 3:
                break
            q_sq = opt_al(q_sq, n_sq, act_sq, mu, t_init=best_c5, maxiter=200, gtol=1e-13)
            h_sq = build_sym(q_sq, n_sq)
            c5_sq = eval_c5(h_sq)
            if c5_sq < best_c5:
                best_h = h_sq.copy()
                best_c5 = c5_sq

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_target(best_h, len(best_h) / 2.0)
    best_c5 = eval_c5(best_h)

    return best_h, best_c5, len(best_h)
```
