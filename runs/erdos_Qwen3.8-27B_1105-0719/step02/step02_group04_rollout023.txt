import numpy as np
import time

def eval_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0-h, mode='full')) * (2.0/n))

def corr_array(h):
    n = len(h)
    return np.correlate(h, 1.0-h, mode='full') * (2.0/n)

def build_sym(q, n):
    m = n // 2
    h = np.empty(n)
    h[:m] = q
    h[m:] = 1.0 - q[::-1]
    return h

def to_sym_q(h, n):
    m = n // 2
    q = 0.5 * (h[:m] + 1.0 - h[n-1:m-1:-1])
    return q

def project_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo+hi)*0.5
        if np.clip(h-mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h-(lo+hi)*0.5, 0, 1)
    diff = target - h.sum()
    mask = (h > 1e-12) & (h < 1.0-1e-12)
    if mask.any() and abs(diff) > 1e-14:
        h[mask] += diff / mask.sum()
    return np.clip(h, 0, 1)

    e = max(0, d)
    u = min(n-1, n-1+d)

    ks1 = np.arange(e, u+1)
    D[ks1] += 1.0 - h[ks1 - d]
    ks2_lo = max(0, e - d)

        D[ks2] -= h[ks2 + d]

            if 0 <= k <= len(cvals)-1:
                padded.add(k)
    return np.sort(np.array(list(padded)))

def solve_target_step(q, P, c_active, target, lam=0.01):
    r = target - c_active

    A = np.vstack([P, np.sqrt(lam) * np.eye(m)])
    b = np.concatenate([r, np.zeros(m)])
    delta, _, _, _ = np.linalg.lstsq(A, b, rcond=None)

            best_c5 = c5_try
            best_q = q_try
    return best_q, best_c5, best_alpha

    else:
