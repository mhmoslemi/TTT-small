def find_optimal_delta(corr, b, lo, hi):
    """Find delta in [lo, hi] that minimizes max_k (corr[k] + delta * b[k])."""
    for _ in range(100):
        m1 = lo + (hi - lo) / 3
        m2 = hi - (hi - lo) / 3
        f1 = (corr + m1 * b).max()
        f2 = (corr + m2 * b).max()
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    delta = (lo + hi) / 2
    return delta

def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
        else:
            h += diff / n
    return np.clip(h, 0.0, 1.0)

    return np.correlate(h, 1.0 - h, mode='full')

def c5_of_h(h, dx):

def compute_sensitivity(h):
    """Compute A[i, l] = sensitivity of corr_l to h[i].
    A[i, l] = (1 - h[i+l])*[0<=i+l<n] - h[i-l]*[0<=i-l<n]
    where l is the lag index in the correlate output.
    
    In np.correlate(h, 1-h, 'full'), output[k] corresponds to lag k-(n-1).
    So for lag d = k-(n-1):
    A[i, k] = (1 - h[i+d])*[0<=i+d<n] - h[i-d]*[0<=i-d<n]
    """
    n = len(h)
    A = np.zeros((n, 2*n-1))
    lags = np.arange(-(n-1), n)  # lags
    for i in range(n):
        # First term: (1-h[i+d]) where 0 <= i+d < n
        valid1 = (i + lags >= 0) & (i + lags < n)
        A[i, valid1] += (1.0 - h[i + lags[valid1]])
        # Second term: -h[i-d] where 0 <= i-d < n
        valid2 = (i - lags >= 0) & (i - lags < n)
        A[i, valid2] += (-h[i - lags[valid2]])
    return A

    # For each i and each lag d:
    # A[i, idx(d)] = (1-h[i+d])*[0<=i+d<n] - h[i-d]*[0<=i+d<n]

    d_idx = lags[None, :]  # (1, 2n-1)
    
    first = i_idx + d_idx  # (n, 2n-1): i+d
    second = i_idx - d_idx  # (n, 2n-1): i-d

    A[valid2] += (-h[second[valid2]])

import numpy as np
from scipy.optimize import minimize
import time


def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
        else:
            h += diff / n
    return np.clip(h, 0.0, 1.0)


def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def c5_of_h(h, dx):
    return compute_corrs(h).max() * dx


def compute_sensitivity(h):
    n = len(h)
    lags = np.arange(-(n - 1), n)
    i_idx = np.arange(n)[:, None]
    d_idx = lags[None, :]
    first = i_idx + d_idx
    second = i_idx - d_idx
    valid1 = (first >= 0) & (first < n)
    valid2 = (second >= 0) & (second < n)
    A = np.zeros((n, 2 * n - 1))
    A[valid1] = (1.0 - h[first[valid1]])
    A[valid2] += (-h[second[valid2]])
    return A


def two_coord_descent(h0, dx, max_sweeps=100, n_iter=40):
    n = len(h0)
    h = h0.copy()
    corrs = compute_corrs(h)
    best_c5 = corrs.max() * dx
    best_h = h.copy()
    no_improve = 0

    for sweep in range(max_sweeps):
        A = compute_sensitivity(h)
        ones = np.where(h > 0.01)[0]
        zeros = np.where(h < 0.99)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        current_max = corrs.max()
        best_improve = 0
        best_i = -1
        best_m = -1
        best_delta = 0.0

        A_zeros = A[zeros]
        for ii in range(len(ones)):
            i = ones[ii]
            hi_bound = np.minimum(h[i], 1.0 - h[zeros])
            valid = hi_bound > 1e-8
            if not valid.any():
                continue
            B = A_zeros[valid] - A[i][None, :]
            n_valid = valid.sum()
            lo = np.zeros(n_valid)
            hi = hi_bound[valid].copy()
            for _ in range(n_iter):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                f1 = (corrs[None, :] + m1[:, None] * B).max(axis=1)
                f2 = (corrs[None, :] + m2[:, None] * B).max(axis=1)
                go_left = f1 < f2
                hi[go_left] = m2[go_left]
                lo[~go_left] = m1[~go_left]
            delta = 0.5 * (lo + hi)
            new_maxes = (corrs[None, :] + delta[:, None] * B).max(axis=1)
            improvements = current_max - new_maxes
            bidx = improvements.argmax()
            if improvements[bidx] > best_improve:
                best_improve = improvements[bidx]
                best_i = i
                best_m = zeros[valid][bidx]
                best_delta = delta[bidx]

        if best_improve < 1e-13 or best_i < 0:
            no_improve += 1
            if no_improve > 5:
                break
            continue
        no_improve = 0
        h[best_i] -= best_delta
        h[best_m] += best_delta
        corrs += best_delta * (A[best_m] - A[best_i])
        new_c5 = corrs.max() * dx
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()

    return best_h, best_c5


def grad_corr_k_vec(h, k, n):
    offset = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -offset)
    i_hi = min(n - 1, n - 1 - offset)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx + offset] += (1.0 - h[idx])
    grad[idx] += (-h[idx + offset])
    return grad


def smooth_max_obj(h, dx, beta):
    corrs = compute_corrs(h)
    m = corrs.max()
    return (m + np.log(np.sum(np.exp(beta * (corrs - m)))) / beta) * dx


def smooth_max_grad(h, dx, beta):
    n = len(h)
    corrs = compute_corrs(h)
    m = corrs.max()
    w = np.exp(beta * (corrs - m))
    w_sum = w.sum()
    if w_sum == 0:
        return np.zeros(n)
    w /= w_sum
    A = compute_sensitivity(h)
    grad = A.T @ w
    return grad * dx


def local_slsqp(h0, dx, target_sum, beta, maxiter):
    n = len(h0)

    def obj(x):
        return smooth_max_obj(x, dx, beta)

    def obj_grad(x):
        return smooth_max_grad(x, dx, beta)

    def cons_fun(x):
        return x.sum() - target_sum

    def cons_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0, jac=obj_grad,
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        h_out = project_feasible(res.x, target_sum)
    except Exception:
        h_out = project_feasible(h0, target_sum)
    return h_out, c5_of_h(h_out, dx)


def subgradient_polish(h0, dx, n_steps=100, alpha_start=0.005):
    n = len(h0)
    h = h0.copy()
    best_c5 = c5_of_h(h, dx)
    best_h = h.copy()
    alpha = alpha_start

    for step in range(n_steps):
        corrs = compute_corrs(h)
        A = compute_sensitivity(h)
        max_val = corrs.max()
        tol = max(0.01, max_val * 0.02)
        active = np.where(corrs >= max_val - tol)[0]
        if len(active) == 0:
            active = np.array([corrs.argmax()])
        subgrad = A[:, active].mean(axis=1)
        h_new = h - alpha * subgrad
        h_new = project_feasible(h_new, n / 2.0)
        c5_new = c5_of_h(h_new, dx)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
        else:
            alpha *= 0.7
        if alpha < 1e-6:
            break
    return best_h, best_c5


def make_quasi_start(n, rng, mode=0):
    h = np.zeros(n)
    target = n // 2
    if mode == 0:
        phi = (1.0 + np.sqrt(5.0)) / 2.0
        for k in range(target):
            pos = int((k * phi * n) % n)
            h[pos] = 1.0
    elif mode == 1:
        idx = rng.permutation(n)
        h[idx[:target]] = 1.0
    elif mode == 2:
        period = max(1, n // (target + 1))
        positions = []
        for k in range(target):
            pos = (k * period + k // period) % n
            if h[pos] == 0:
                h[pos] = 1.0
                positions.append(pos)
        while len(positions) < target:
            pos = rng.randint(n)
            if h[pos] == 0:
                h[pos] = 1.0
                positions.append(pos)
    return project_feasible(h, n / 2.0)


def remaining_time(t0, budget_s):
    return budget_s - (time.time() - t0)
