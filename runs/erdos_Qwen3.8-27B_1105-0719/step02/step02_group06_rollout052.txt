def solve_lp_full(h0, all_grads, raw0, rho, target_sum):
    n = len(h0)
    n_lags = len(raw0)
    n_vars = n + 1
    A = np.zeros((n_lags, n_vars))
    A[:, :n] = all_grads
    A[:, n] = -1.0
    b = all_grads @ h0 - raw0
    c = np.zeros(n_vars)
    c[n] = 1.0
    bounds = []
    for i in range(n):
        bounds.append((max(0.0, h0[i] - rho), min(1.0, h0[i] + rho)))
    bounds.append((0.0, raw0.max() + 2.0))
    A_eq = np.zeros((1, n_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target_sum])
    try:
        res = linprog(c, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            h_new = np.clip(res.x[:n], 0.0, 1.0)
            h_new = project_feasible(h_new, target_sum)
            return h_new
    except Exception:
        pass
    return None

def line_search_sym(q0, direction, n_grid=25):
    if np.abs(direction).max() < 1e-15:
        return 0.0, q0.copy(), eval_score(build_sym_h(q0))
    pos = direction > 1e-15
    neg = direction < -1e-15
    alpha_max = 1.0
    if pos.any():
        alpha_max = min(alpha_max, ((1.0 - q0[pos]) / direction[pos]).min())
    if neg.any():
        alpha_max = min(alpha_max, ((-q0[neg]) / direction[neg]).min())
    alpha_max = max(alpha_max, 1e-10)
    if alpha_max >= 1.0:
        alpha_max = 1.0
    f0 = eval_score(build_sym_h(q0))
    best_f = f0
    best_a = 0.0
    alphas = np.linspace(0.0, alpha_max, n_grid)
    for a in alphas[1:]:
        q_try = q0 + a * direction
        if np.any(q_try < -1e-10) or np.any(q_try > 1.0 + 1e-10):
            continue
        f_try = eval_score(build_sym_h(q_try))
        if f_try < best_f:
            best_f = f_try
            best_a = a
    if best_a > 0 and best_a < alpha_max - 1e-10:
        idx = min(n_grid - 1, max(1, int(round(best_a / alpha_max * (n_grid - 1)))))
        lo = alphas[idx - 1]
        hi = alphas[idx + 1] if idx + 1 < n_grid else alpha_max
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = hi - gr * (hi - lo)
        c2 = lo + gr * (hi - lo)
        f1 = eval_score(build_sym_h(np.clip(q0 + c1 * direction, 0, 1)))
        f2 = eval_score(build_sym_h(np.clip(q0 + c2 * direction, 0, 1)))
        for _ in range(25):
            if hi - lo < 1e-9:
                break
            if f1 < f2:
                hi = c2
                c2 = c1
                f2 = f1
                c1 = hi - gr * (hi - lo)
                f1 = eval_score(build_sym_h(np.clip(q0 + c1 * direction, 0, 1)))
            else:
                lo = c1
                c1 = c2
                f1 = f2
                c2 = lo + gr * (hi - lo)
                f2 = eval_score(build_sym_h(np.clip(q0 + c2 * direction, 0, 1)))
        mid_a = (lo + hi) / 2.0
        f_mid = eval_score(build_sym_h(np.clip(q0 + mid_a * direction, 0, 1)))
        if f_mid < best_f:
            best_f = f_mid
            best_a = mid_a
    return best_a, np.clip(q0 + best_a * direction, 0, 1), best_f

def pair_transfer_sym(q0, a, b, n_grid=25):
    lo = max(-q0[a], -(1.0 - q0[b]))
    hi = min(1.0 - q0[a], q0[b])
    if hi - lo < 1e-10:
        return q0.copy(), float('inf')
    f0 = eval_score(build_sym_h(q0))
    best_f = f0
    best_d = 0.0
    dir_vec = np.zeros(len(q0))
    dir_vec[a] = 1.0
    dir_vec[b] = -1.0
    deltas = np.linspace(lo, hi, n_grid)
    for d in deltas[1:]:
        q_try = q0 + d * dir_vec
        sc = eval_score(build_sym_h(q_try))
        if sc < best_f:
            best_f = sc
            best_d = d
    if best_d != 0.0 and abs(best_d) < hi - lo:
        idx = min(n_grid - 1, max(1, int(round((best_d - lo) / (hi - lo) * (n_grid - 1)))))
        rlo = deltas[idx - 1]
        rhi = deltas[idx + 1] if idx + 1 < n_grid else hi
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = rhi - gr * (rhi - rlo)
        c2 = rlo + gr * (rhi - rlo)
        f1 = eval_score(build_sym_h(np.clip(q0 + c1 * dir_vec, 0, 1)))
        f2 = eval_score(build_sym_h(np.clip(q0 + c2 * dir_vec, 0, 1)))
        for _ in range(20):
            if rhi - rlo < 1e-9:
                break
            if f1 < f2:
                rhi = c2
                c2 = c1
                f2 = f1
                c1 = rhi - gr * (rhi - rlo)
                f1 = eval_score(build_sym_h(np.clip(q0 + c1 * dir_vec, 0, 1)))
            else:
                rlo = c1
                c1 = c2
                f1 = f2
                c2 = rlo + gr * (rhi - rlo)
                f2 = eval_score(build_sym_h(np.clip(q0 + c2 * dir_vec, 0, 1)))
        mid_d = (rlo + rhi) / 2.0
        f_mid = eval_score(build_sym_h(np.clip(q0 + mid_d * dir_vec, 0, 1)))
        if f_mid < best_f:
            best_f = f_mid
            best_d = mid_d
    return np.clip(q0 + best_d * dir_vec, 0, 1), best_f

def transfer_to_n(h_old, n_new):
    n_old = len(h_old)
    x_old = (np.arange(n_old) + 0.5) * 2.0 / n_old
    x_new = (np.arange(n_new) + 0.5) * 2.0 / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    try:
        h_parent = np.array(initial_h_values, dtype=float).copy().reshape(-1)
    except (NameError, AttributeError):
        h_parent = None

    if h_parent is not None and len(h_parent) > 10:
        n = len(h_parent)
        h_parent = np.clip(h_parent, 0.0, 1.0)
        h_parent = project_feasible(h_parent, n / 2.0)
        best_h = h_parent.copy()
    else:
        m = 63
        x = np.linspace(0.0, 1.0, m)
        q_init = np.clip(0.5 + 0.3 * np.sin(2.0 * np.pi * x), 0.0, 1.0)
        best_h = build_sym_h(q_init)

    best_c5 = eval_score(best_h)

    # Phase 1: Symmetric LP optimization with adaptive trust region
    if len(best_h) % 2 == 0:
        q_cur = sym_from_h(best_h)
        n_sym = len(best_h)
        m_sym = n_sym // 2
        stall_count = 0

        for outer_iter in range(200):
            if time.time() - t0 > budget_s - 120:
                break
            rho_list = [0.3, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005]
            if stall_count > 10:
                rho_list = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
            improved = False
            for rho in rho_list:
                if time.time() - t0 > budget_s - 120:
                    break
                raw0 = corr_raw(build_sym_h(q_cur))
                ag = all_grads_q(q_cur)
                q_lp = solve_lp_sym(q_cur, ag, raw0, rho)
                if q_lp is None:
                    continue
                direction = q_lp - q_cur
                if np.abs(direction).max() < 1e-12:
                    continue
                _, q_new, f_new = line_search_sym(q_cur, direction)
                if f_new < best_c5 - 1e-14:
                    best_c5 = f_new
                    best_h = build_sym_h(q_new).copy()
                    q_cur = q_new
                    improved = True
                    break
                elif np.abs(f_new - best_c5) < 1e-10 and f_new <= best_c5 + 1e-12:
                    q_cur = q_new
                    break
            if improved:
                stall_count = 0
            else:
                stall_count += 1
            # Periodic pair transfers
            if stall_count > 3 and stall_count % 2 == 1:
                for pt in range(30):
                    if time.time() - t0 > budget_s - 120:
                        break
                    a = rng.randint(0, m_sym)
                    b = rng.randint(0, m_sym)
                    if a == b:
                        continue
                    q_pt, f_pt = pair_transfer_sym(q_cur, a, b)
                    if f_pt < best_c5 - 1e-14:
                        best_c5 = f_pt
                        best_h = build_sym_h(q_pt).copy()
                        q_cur = q_pt
                        stall_count = 0
                        break

    # Phase 2: Multi-resolution exploration
    n_base = len(best_h)
    n_candidates = []
    for delta in [-6, -4, -3, -2, -1, 1, 2, 3, 4, 6, 8, 12]:
        nc = n_base + delta
        if 40 <= nc <= 220 and nc % 2 == 0:
            n_candidates.append(nc)
    if n_base // 2 >= 30:
        n_candidates.append(n_base // 2)
    n_candidates = sorted(set(n_candidates))

    for n_try in n_candidates:
        if time.time() - t0 > budget_s - 60:
            break
        h_try = transfer_to_n(best_h, n_try)
        n_try = len(h_try)
        q_try = sym_from_h(h_try)
        m_try = len(q_try)
        c5_try = eval_score(h_try)

        # Optimize at this resolution
        for outer_iter in range(50):
            if time.time() - t0 > budget_s - 40:
                break
            raw0 = corr_raw(build_sym_h(q_try))
            ag = all_grads_q(q_try)
            improved_here = False
            for rho in [0.2, 0.1, 0.05, 0.02, 0.01]:
                q_lp = solve_lp_sym(q_try, ag, raw0, rho)
                if q_lp is None:
                    continue
                direction = q_lp - q_try
                if np.abs(direction).max() < 1e-12:
                    continue
                _, q_new, f_new = line_search_sym(q_try, direction)
                if f_new < c5_try - 1e-14:
                    c5_try = f_new
                    q_try = q_new
                    improved_here = True
                    break
                elif f_new <= c5_try + 1e-12:
                    q_try = q_new
                    break
            if not improved_here:
                # Try pair transfers
                pt_improved = False
                for pt in range(20):
                    a = rng.randint(0, m_try)
                    b = rng.randint(0, m_try)
                    if a == b:
                        continue
                    q_pt, f_pt = pair_transfer_sym(q_try, a, b)
                    if f_pt < c5_try - 1e-14:
                        c5_try = f_pt
                        q_try = q_pt
                        pt_improved = True
                        break
                if not pt_improved:
                    break

        h_opt = build_sym_h(q_try)
        c5_opt = eval_score(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 3: Full-vector optimization (break symmetry)
    if time.time() - t0 < budget_s - 80:
        n_full = len(best_h)
        h_cur = best_h.copy()
        target_sum = n_full / 2.0
        best_full_c5 = best_c5

        for outer_iter in range(80):
            if time.time() - t0 > budget_s - 40:
                break
            raw0 = corr_raw(h_cur)
            ag = all_grads_h(h_cur)
            improved = False
            for rho in [0.2, 0.1, 0.05, 0.02, 0.01]:
                h_lp = solve_lp_full(h_cur, ag, raw0, rho, target_sum)
                if h_lp is None:
                    continue
                direction = h_lp - h_cur
                if np.abs(direction).max() < 1e-12:
                    continue
                _, h_new, f_new = line_search_full(h_cur, direction, target_sum)
                if f_new < best_full_c5 - 1e-14:
                    best_full_c5 = f_new
                    h_cur = h_new.copy()
                    improved = True
                    break
                elif f_new <= best_full_c5 + 1e-12:
                    h_cur = h_new.copy()
                    break
            if improved:
                if best_full_c5 < best_c5:
                    best_c5 = best_full_c5
                    best_h = h_cur.copy()
            else:
                # Try random pair transfers in full space
                pt_improved = False
                for pt in range(20):
                    a = rng.randint(0, n_full)
                    b = rng.randint(0, n_full)
                    if a == b:
                        continue
                    dir_v = np.zeros(n_full)
                    dir_v[a] = 1.0
                    dir_v[b] = -1.0
                    lo = max(-h_cur[a], -(1.0 - h_cur[b]))
                    hi = min(1.0 - h_cur[a], h_cur[b])
                    if hi - lo < 1e-10:
                        continue
                    for d in np.linspace(lo, hi, 20):
                        h_try2 = h_cur + d * dir_v
                        if np.any(h_try2 < -1e-10) or np.any(h_try2 > 1.0 + 1e-10):
                            continue
                        f_try2 = eval_score(h_try2)
                        if f_try2 < best_full_c5 - 1e-14:
                            best_full_c5 = f_try2
                            h_cur = project_feasible(h_try2, target_sum)
                            pt_improved = True
                            break
                    if pt_improved:
                        break
                if not pt_improved and best_full_c5 >= best_c5:
                    break

    # Phase 4: Final polish with very small trust region
    if time.time() - t0 < budget_s - 20:
        if len(best_h) % 2 == 0:
            q_fin = sym_from_h(best_h)
            for rho in [0.005, 0.002, 0.001, 0.0005]:
                if time.time() - t0 > budget_s - 10:
                    break
                raw0 = corr_raw(build_sym_h(q_fin))
                ag = all_grads_q(q_fin)
                q_lp = solve_lp_sym(q_fin, ag, raw0, rho)
                if q_lp is None:
                    continue
                direction = q_lp - q_fin
                if np.abs(direction).max() < 1e-14:
                    continue
                _, q_new, f_new = line_search_sym(q_fin, direction, n_grid=40)
                if f_new < best_c5:
                    best_c5 = f_new
                    best_h = build_sym_h(q_new).copy()
                    q_fin = q_new
        else:
            h_fin = best_h.copy()
            target_sum = len(h_fin) / 2.0
            for rho in [0.005, 0.002, 0.001]:
                if time.time() - t0 > budget_s - 10:
                    break
                raw0 = corr_raw(h_fin)
                ag = all_grads_h(h_fin)
                h_lp = solve_lp_full(h_fin, ag, raw0, rho, target_sum)
                if h_lp is None:
                    continue
                direction = h_lp - h_fin
                if np.abs(direction).max() < 1e-14:
                    continue
                _, h_new, f_new = line_search_full(h_fin, direction, target_sum, n_grid=40)
                if f_new < best_c5:
                    best_c5 = f_new
                    best_h = h_new.copy()
                    h_fin = h_new

    # Validate
    best_h = np.asarray(best_h, dtype=float).reshape(-1)
    n_out = len(best_h)
    best_h = project_feasible(best_h, n_out / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = eval_score(best_h)

    return best_h, best_c5, n_out
