    # But for n=251, even O(n²) is fine
    
    # Candidate deltas: lo, hi, and intersections of "upper envelope" lines
    # For simplicity, compute all pairwise intersections and filter
    
    n = len(c)
    candidates = np.array([lo, hi])
    
    # More efficient: use the fact that the minimizer is where the
    # upper envelope changes slope from negative to positive
    # Binary search on delta
    
    def g(delta):
        return np.max(corrs + delta * c)
    
    def subgrad_sign(delta):
        """Return sign of subgradient: -1 if g decreasing, +1 if increasing, 0 if at min."""
        vals = corrs + delta * c
        mx = vals.max()
        active = vals >= mx - 1e-15
        # Subgradient is the set {c[i] for active i}
        # If all active c[i] > 0, g is increasing
        # If all active c[i] < 0, g is decreasing
        # If mixed, we're at the minimum
        if np.all(c[active] > 1e-15):
            return 1  # increasing
        elif np.all(c[active] < -1e-15):
            return -1  # decreasing
        else:
            return 0  # at minimum
    
    # Check boundaries
    best_delta = lo
    best_val = g(lo)
    if g(hi) < best_val:
        best_delta = hi
        best_val = g(hi)
    
    # Binary search
    s = subgrad_sign(lo)
    if s == 0:
        return lo, g(lo)
    
    a, b = lo, hi
    # g is convex, subgradient is monotone
    for _ in range(60):
        mid = (a + b) / 2
        s = subgrad_sign(mid)
        if s == 0:
            a = b = mid
            break
        elif s < 0:
            a = mid  # minimum is to the right
        else:
            b = mid  # minimum is to the left
    
    mid = (a + b) / 2
    val = g(mid)
    if val < best_val:
        best_delta = mid
        best_val = val
    
    return best_delta, best_val

import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full').max() * dx


def corr_values(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx


def make_sym_h(q, n):
    m = n // 2
    h = np.empty(n)
    h[:m] = q
    h[m:] = 1.0 - q[::-1]
    return h


def get_q_from_h(h, n):
    m = n // 2
    q = 0.5 * (h[:m] + 1.0 - h[n-1:n-1-m:-1])
    return np.clip(q, 0.0, 1.0)


def compute_coef(h, n, j):
    i1 = j
    i2 = n - 1 - j
    d = np.arange(-(n-1), n)
    c = np.zeros(2*n-1)
    idx = i1 + d
    mask = (idx >= 0) & (idx < n)
    c[mask] += 1.0 - h[idx[mask]]
    idx = i1 - d
    mask = (idx >= 0) & (idx < n)
    c[mask] += -h[idx[mask]]
    idx = i2 + d
    mask = (idx >= 0) & (idx < n)
    c[mask] += -(1.0 - h[idx[mask]])
    idx = i2 - d
    mask = (idx >= 0) & (idx < n)
    c[mask] += h[idx[mask]]
    return c


def min_1d_max_lines(corrs, c, lo, hi):
    def g(delta):
        return np.max(corrs + delta * c)

    def subgrad_dir(delta):
        vals = corrs + delta * c
        mx = vals.max()
        active = vals >= mx - 1e-13
        cs = c[active]
        if np.all(cs > 1e-13):
            return 1
        elif np.all(cs < -1e-13):
            return -1
        return 0

    best_delta = lo
    best_val = g(lo)
    val_hi = g(hi)
    if val_hi < best_val:
        best_delta = hi
        best_val = val_hi

    if subgrad_dir(lo) >= 0:
        return lo, g(lo)
    if subgrad_dir(hi) <= 0:
        return hi, g(hi)

    a, b = lo, hi
    for _ in range(80):
        mid = (a + b) / 2
        d = subgrad_dir(mid)
        if d == 0:
            a = b = mid
            break
        elif d < 0:
            a = mid
        else:
            b = mid

    delta = (a + b) / 2
    val = g(delta)
    if val < best_val:
        best_delta = delta
        best_val = val
    return best_delta, best_val


def coord_descent_sym(q, n, max_sweeps=300, tol=1e-12, rng=None, random_order=False):

    h = make_sym_h(q, n)
    corrs = corr_values(h)

        max_change = 0.0
        order = list(range(m))
        if random_order and rng is not None:
            rng.shuffle(order)
        for j in order:
            c = compute_coef(h, n, j)
            lo = -q[j]
            hi = 1.0 - q[j]
            delta, _ = min_1d_max_lines(corrs, c, lo, hi)

                h[j] += delta
                h[n-1-j] -= delta
                corrs += delta * c
        if max_change < tol:

    return q, c5


def coord_descent_full(h, n, max_sweeps=100, tol=1e-12, rng=None, target_sum=None):
    if target_sum is None:
        target_sum = n / 2.0
    corrs = corr_values(h)

    for sweep in range(max_sweeps):
        max_change = 0.0
        for j in range(n):

            c = np.zeros(2*n-1)
            idx = j + d
            mask = (idx >= 0) & (idx < n)
            c[mask] += 1.0 - h[idx[mask]]
            idx = j - d
            mask = (idx >= 0) & (idx < n)
            c[mask] += -h[idx[mask]]
            # But we need to maintain sum constraint: compensate with -delta/n spread

            lo = -h[j]
            hi = 1.0 - h[j]

                max_change = abs(delta)

                corrs += delta * c

            h = np.clip(h, 0.0, 1.0)

            break

    return h, c5


def gen_binary_sym(n, rng):
    m = n // 2
    q = np.zeros(m)
    ones_needed = n // 2

    ones_in_q = m // 2

def project_feasible_sym(q, n):
    """Project q to [0,1]^m such that sym h has correct sum."""
    m = n // 2
    # sum of sym h = sum(q) + sum(1-q) = m + (m - 2*sum(q)) = 2m - 2*sum(q) = n - 2*sum(q)
    # We need n - 2*sum(q) = n/2, so sum(q) = n/4 = m/2
    target = m / 2.0
    q = np.clip(q, 0.0, 1.0)
    # Binary search on offset

    for _ in range(60):
        mid = (lo + hi) / 2
        h = np.clip(q - mid, 0.0, 1.0)
        if h.sum() > target:
            lo = mid
        else:
            hi = mid

        mask = (q > 1e-10) & (q < 1.0 - 1e-10)
        if mask.sum() > 0:
            q[mask] += diff / mask.sum()
    return np.clip(q, 0.0, 1.0)


def smooth_sym_obj(x, n, beta):
    h = make_sym_h(x, n)
    corrs = corr_values(h)
    return logsumexp(beta * corrs) / beta


def smooth_sym_grad(x, n, beta):
    m = n // 2
    dx = 2.0 / n
    h = make_sym_h(x, n)
    corrs = corr_values(h)
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    ws = w.sum()
    if ws < 1e-30:
        w = np.ones(2*n-1) / (2*n-1)
    else:
        w /= ws
    # Weighted gradient w.r.t. h
    G = np.zeros(n)
    d_vals = np.arange(2*n-1) - (n-1)
    for ell in range(2*n-1):
        wk = beta * w[ell]
        if wk < 1e-15:
            continue
        d = d_vals[ell]
        i_lo = max(0, -d)
        i_hi = min(n-1, n-1-d)
        if i_lo > i_hi:
            continue
        idx = np.arange(i_lo, i_hi+1)
        G[idx] += wk * (1.0 - h[idx + d])
        G[idx + d] += wk * (-h[idx])
    G *= dx
    return G[:m] - G[n-1:n-1-m:-1]


def smooth_polish(q, n, beta_list, rng=None):
