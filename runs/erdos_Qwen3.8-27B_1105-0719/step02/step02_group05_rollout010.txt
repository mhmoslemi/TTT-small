from scipy.special import logsumexp

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corrs = np.correlate(h, 1.0 - h, mode='full')
    return corrs.max() * dx, corrs


def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            adj = diff / mask.sum()
            h[mask] = np.clip(h[mask] + adj, 0.0, 1.0)
    return np.clip(h, 0.0, 1.0)


def build_h_from_q(q, n):
    m = n // 2
    if n % 2 == 0:
        return np.concatenate([q, 1.0 - q[::-1]])
    else:
        return np.concatenate([q, [0.5], 1.0 - q[::-1]])


def q_from_h(h, n):
    m = n // 2
    return h[:m].copy()


def grad_corr_d(h, d, n):
    g = np.zeros(n)
    if d >= 0:
        if d < n:
            if d == 0:
                g[:] += (1.0 - h) + (-h)
            else:
                g[:n - d] += (1.0 - h[d:])
                g[d:] += (-h[:n - d])
    else:
        dd = -d
        if dd < n:
            if dd == 0:
                g[:] += (1.0 - h) + (-h)
            else:
                g[dd:] += (1.0 - h[:n - dd])
                g[:n - dd] += (-h[dd:])
    return g


def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def smooth_q_val_grad(q, n, beta):
    m = n // 2
    h = build_h_from_q(q, n)
    dx = 2.0 / n
    corrs = compute_corrs(h)
    val = logsumexp(beta * corrs) / beta * dx
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(len(corrs)) / len(corrs)
    else:
        w /= w_sum
    Gq = np.zeros(m)
    for k in range(len(corrs)):
        if w[k] < 1e-15:
            continue
        d = k - (n - 1)
        G = grad_corr_d(h, d, n)
        Gq += w[k] * (G[:m] - G[m:][::-1])
    return val, Gq * dx


class QSmoothObj:
    def __init__(self, n, beta):
        self.n = n
        self.beta = beta

    def value(self, x):
        return smooth_q_val_grad(x, self.n, self.beta)[0]

    def gradient(self, x):
        return smooth_q_val_grad(x, self.n, self.beta)[1]


class QEpigraph:
    def __init__(self, n, active_k):
        self.n = n
        self.m = n // 2
        self.active_k = active_k
        self.dx = 2.0 / n

    def _build_h(self, q):
        return build_h_from_q(q, self.n)

    def obj(self, x):
        return x[0]

    def obj_grad(self, x):
        g = np.zeros(self.m + 1)
        g[0] = 1.0
        return g

    def corr_val(self, x, k_idx):
        t = x[0]
        q = x[1:]
        h = build_h_from_q(q, self.n)
        k = self.active_k[k_idx]
        d = k - (self.n - 1)
        n = self.n
        if d >= 0:
            if d == 0:
                val = np.sum(h * (1.0 - h))
            else:
                val = np.sum(h[d:] * (1.0 - h[:n - d]))
        else:
            dd = -d
            val = np.sum(h[dd:] * (1.0 - h[:n - dd]))
        return t - val * self.dx

    def corr_grad(self, x, k_idx):
        q = x[1:]
        h = build_h_from_q(q, self.n)
        k = self.active_k[k_idx]
        d = k - (self.n - 1)
        n = self.n
        g = np.zeros(self.m + 1)
        g[0] = 1.0
        G = grad_corr_d(h, d, n)
        g[1:] = -(G[:self.m] - G[self.m:][::-1]) * self.dx
        return g


def make_q_constraints(ep):
    cons = []
    for i in range(len(ep.active_k)):
        cons.append({
            'type': 'ineq',
            'fun': partial(ep.corr_val, k_idx=i),
            'jac': partial(ep.corr_grad, k_idx=i)
        })
    return cons


def solve_q_epigraph(q0, n, active_k, t0, maxiter=300):
    m = n // 2
    x0 = np.concatenate([[t0], q0])
    ep = QEpigraph(n, active_k)
    cons = make_q_constraints(ep)
    bounds = [(0.0, 5.0)] + [(0.0, 1.0)] * m
    try:
        res = minimize(ep.obj, x0, jac=ep.obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        q_out = np.clip(res.x[1:], 0.0, 1.0)
        t_out = res.x[0]
        return q_out, t_out, res
    except Exception:
        return q0, t0, None


class FullEpigraph:
    def __init__(self, n, active_k, target_sum):
        self.n = n
        self.active_k = active_k
        self.target_sum = target_sum
        self.dx = 2.0 / n

    def obj(self, x):
        return x[0]

    def obj_grad(self, x):
        g = np.zeros(self.n + 1)
        g[0] = 1.0
        return g

    def corr_val(self, x, k_idx):
        t = x[0]
        h = x[1:]

        n = self.n
        if d >= 0:

            else:
                val = np.sum(h[d:] * (1.0 - h[:n - d]))
        else:
            dd = -d
            val = np.sum(h[dd:] * (1.0 - h[:n - dd]))
        return t - val * self.dx

    def corr_grad(self, x, k_idx):
        h = x[1:]
        k = self.active_k[k_idx]
        d = k - (self.n - 1)
        n = self.n
        g = np.zeros(n + 1)
        g[0] = 1.0
        G = grad_corr_d(h, d, n)
        g[1:] = -G * self.dx
        return g

    def sum_con(self, x):
        return x[1:].sum() - self.target_sum

    def sum_con_grad(self, x):
        g = np.zeros(self.n + 1)
        g[1:] = 1.0
        return g


def make_full_constraints(ep):

def local_perturbation(h, n, n_trials, step_size, target_sum, rng):
    best_h = h.copy()
    best_c5 = exact_score(h)[0]
    for _ in range(n_trials):
        h_try = h.copy()
        mask = (h_try > 0.01) & (h_try < 0.99)
        if mask.sum() < 5:
            continue
        idx = rng.choice(np.where(mask)[0], size=min(5, mask.sum()), replace=False)
        h_try[idx] += rng.uniform(-step_size, step_size, size=len(idx))
        h_try = np.clip(h_try, 0.0, 1.0)
        h_try = project_feasible(h_try, target_sum)
        c5_try, _ = exact_score(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()
            h = h_try.copy()
    return best_h, best_c5
