def grad_corr_d(h, d, n):
    """Gradient of correlation at lag d w.r.t. h"""
    g = np.zeros(n)
    if d >= 0:
        if d < n:
            # c = sum_{i=0}^{n-1-d} h[i]*(1-h[i+d])
            # dc/dh[j] = (1-h[j+d]) for j <= n-1-d, plus -h[j-d] for j >= d
            g[:n-d] += (1.0 - h[d:])  # contribution from h[i] term, i=j, i+d = j+d
            g[d:] += (-h[:n-d])       # contribution from (1-h[i+d]) term, i+d=j, i=j-d

Actually, I recall now that the Erdős minimum overlap problem is different from what I was thinking. Let me re-read the problem statement.

The problem says: C₅ = max_k ∫ h(x)(1-h(x+k)) dx

This is the maximum over ALL k (including k=0!). At k=0, we get ∫ h(x)(1-h(x)) dx = ∫ h(x) dx - ∫ h(x)² dx = 1 - ∫ h(x)² dx.

To minimize this at k=0, we want to maximize ∫ h(x)² dx, which means making h as "peaked" as possible (close to 0 or 1).

But at other k values, we want h(x) and h(x+k) to be correlated (one being 1 implies the other is also 1, or both 0).

This is a tension: we want h to be binary (to minimize k=0 term) but also "spread out" (to minimize other k terms).

For the discrete case with n points:
- At k=0: sum of h[i](1-h[i]) * dx = (sum h[i] - sum h[i]²) * dx = (n/2 - sum h[i]²) * dx
  To minimize: maximize sum h[i]², i.e., make h as binary as possible
  If h is binary: (n/2 - n/2) * dx = 0
  
- At k=1: sum of h[i](1-h[i+1]) * dx (over valid i)
  For binary h, this counts the number of 1→0 transitions at distance 1

So for binary h, the k=0 term is 0, and we need to minimize the maximum number of "mismatches" over all distances.

This is exactly the combinatorial problem I identified earlier!

import numpy as np
from scipy.optimize import minimize
import time


def grad_corr_d(h, d, n):
    g = np.zeros(n)
    if d >= 0:
        if d < n:
            g[:n - d] += (1.0 - h[d:])
            g[d:] += (-h[:n - d])
    else:
        dd = -d
        if dd < n:
            g[dd:] += (1.0 - h[:n - dd])
            g[:n - dd] += (-h[dd:])
    return g


def build_h_from_q(q, n):
    m = n // 2
    if n % 2 == 0:
        return np.concatenate([q, 1.0 - q[::-1]])
    else:
        return np.concatenate([q, [0.5], 1.0 - q[::-1]])


def q_from_h(h, n):
    m = n // 2
    return h[:m].copy()


def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def smooth_max_obj_grad(q, n, beta):
    m = n // 2
    h = build_h_from_q(q, n)
    dx = 2.0 / n
    corrs = compute_corrs(h)
    max_c = corrs.max()
    shifted = corrs - max_c
    exp_s = np.exp(beta * shifted)
    exp_sum = exp_s.sum()
    val = (max_c + np.log(max(exp_sum, 1e-300)) / beta) * dx
    if exp_sum < 1e-30:
        w = np.ones(len(corrs)) / len(corrs)
    else:
        w = exp_s / exp_sum
    Gq = np.zeros(m)
    for k_idx in range(len(corrs)):
        if w[k_idx] < 1e-15:
            continue
        d = k_idx - (n - 1)
        G = grad_corr_d(h, d, n)
        Gq += w[k_idx] * (G[:m] - G[m:][::-1])
    return val, Gq * dx


def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corrs = compute_corrs(h)
    return corrs.max() * dx


def project_to_sum(h, target):
    n = len(h)
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        h_proj = np.clip(h - mid, 0.0, 1.0)
        if h_proj.sum() > target:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h - (lo + hi) / 2, 0.0, 1.0)
    diff = target - h_proj.sum()
    if abs(diff) > 1e-10:
        mask = (h_proj > 1e-12) & (h_proj < 1.0 - 1e-12)
        if mask.sum() > 0:
            h_proj[mask] += diff / mask.sum()
    return np.clip(h_proj, 0.0, 1.0)


def optimize_q(q0, n, betas, maxiter=400):
    m = n // 2
    q = np.clip(q0, 0.0, 1.0).copy()
    h = build_h_from_q(q, n)
    best_score = exact_score(h)
    best_q = q.copy()
    bounds = [(0.0, 1.0)] * m
    for beta in betas:
        def obj_func(x):
            return smooth_max_obj_grad(x, n, beta)[0]

        def grad_func(x):
            return smooth_max_obj_grad(x, n, beta)[1]

        try:
            res = minimize(obj_func, q, jac=grad_func, method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-13})
            q_new = np.clip(res.x, 0.0, 1.0)
            h_new = build_h_from_q(q_new, n)
            s = exact_score(h_new)
            if s < best_score - 1e-15:
                best_score = s
                best_q = q_new.copy()
            q = q_new
        except Exception:
            pass
    return best_q, best_score


def active_set_gradient_step(q, n, active_indices, tau, lr):
    m = n // 2
    h = build_h_from_q(q, n)
    dx = 2.0 / n
    corrs = compute_corrs(h)

    q_new = np.clip(q - lr * Gq * dx, 0.0, 1.0)
    return q_new

                    corrs = compute_corrs(h_x)
                    max_c = corrs.max()
                    shifted = corrs - max_c
                    exp_s = np.exp(beta * shifted)
                    exp_sum = exp_s.sum()
                    if exp_sum < 1e-30:
                        w = np.ones(len(corrs)) / len(corrs)
                    else:
                        w = exp_s / exp_sum
                    G = np.zeros(n_f)
                    dx = 2.0 / n_f
                    for k_idx in range(len(corrs)):
                        if w[k_idx] < 1e-15:
                            continue
                        d = k_idx - (n_f - 1)
                        G += w[k_idx] * grad_corr_d(h_x, d, n_f)
                    return G * dx + 2.0 * mu * (x.sum() - target_f)

                try:
                    res = minimize(full_obj, h_f, jac=full_grad, method='L-BFGS-B',
                                   bounds=[(0.0, 1.0)] * n_f,
                                   options={'maxiter': 300, 'ftol': 1e-15})
                    h_new = project_to_sum(np.clip(res.x, 0.0, 1.0), target_f)
                    s = exact_score(h_new)
                    if s < best_c5 - 1e-14:
                        best_c5 = s
                        best_h = h_new
                        h_f = h_new
                except Exception:
                    pass

    # Phase 7: Final polish - one more smooth max pass
    if time_left() > 30:
        n_fp = len(best_h)
        if n_fp % 2 == 0:
            q_fp = q_from_h(best_h, n_fp)
            betas_fp = [2000, 5000, 10000]
            q_opt, score = optimize_q(q_fp, n_fp, betas_fp, maxiter=400)
            h_opt = build_h_from_q(q_opt, n_fp)
            s = exact_score(h_opt)
            if s < best_c5 - 1e-14:
                best_c5 = s
                best_h = h_opt

    # Final validation
    best_h = project_to_sum(np.clip(best_h, 0.0, 1.0), len(best_h) / 2.0)
    final_score = exact_score(best_h)
    if final_score < best_c5:
        best_c5 = final_score

    return best_h, best_c5, len(best_h)
