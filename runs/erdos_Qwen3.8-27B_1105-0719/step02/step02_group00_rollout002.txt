    grad = G.T @ weights
    grad -= np.sum(grad * h) / np.sum(h) * 0  # Don't project, just subtract mean for sum constraint
    grad -= grad.mean()
    norm = np.linalg.norm(grad)

import numpy as np
from scipy.optimize import minimize
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def repair_feasible(h):
    n = len(h)
    h = np.clip(np.asarray(h, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx] + 1e-15)
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx] + 1e-15)
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def build_grad_rows(h, active_m, n, dx):
    center = n - 1
    G = np.zeros((len(active_m), n))
    for idx, m in enumerate(active_m):
        d = m - center
        if d == 0:
            G[idx, :] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            G[idx, 0:n-d] += -dx * h[d:n]
            G[idx, d:n] += dx * (1.0 - h[0:n-d])
        else:
            dp = -d
            G[idx, 0:n-dp] += dx * (1.0 - h[dp:n])
            G[idx, dp:n] += -dx * h[0:n-dp]
    return G

def grad_step(h, active_m, n, dx, steps=None):
    profile = compute_profile(h)
    c_act = profile[active_m].max()
    G = build_grad_rows(h, active_m, n, dx)
    w = np.maximum(profile[active_m] - (c_act - 0.008), 0.01)
    w /= w.sum()
    grad = G.T @ w
    grad -= grad.mean()
    norm = np.linalg.norm(grad)
    if norm < 1e-15:
        return h, exact_score(h)
    grad /= norm
    if steps is None:
        steps = [0.02, 0.015, 0.01, 0.007, 0.005, 0.003, 0.002, 0.001]
    best_h, best_c = h, exact_score(h)
    for s in steps:
        h_try = repair_feasible(h - s * grad)
        sc = exact_score(h_try)
        if sc < best_c - 1e-13:
            best_h, best_c = h_try, sc
    return best_h, best_c

def lp_solve(h, active_m, n, dx, radius=0.03):
    try:
        import cvxpy as cp
    except ImportError:
        return h, exact_score(h)
    c = exact_score(h)
    profile = compute_profile(h)
    G = build_grad_rows(h, active_m, n, dx)
    p_act = profile[active_m]
    z = cp.Variable(n)
    t = cp.Variable()
    lo = np.maximum(0.0, h - radius)

            solved = True

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    fx = np.concatenate(([0.0], xs, [2.0]))
    fh = np.concatenate(([h[0]], h, [h[-1]]))
    h_new = np.interp(xt, fx, fh)
    return repair_feasible(np.clip(h_new, 0.0, 1.0))

def make_sine_h(n, freq, phase, amp=0.3):
    x = (np.arange(n) + 0.5) * (2.0 / n)
    h = 0.5 + amp * np.sin(np.pi * freq * x + phase)
    return repair_feasible(np.clip(h, 0.01, 0.99))

def make_block_h(n, n_blocks, rng):
    h = np.zeros(n)
    for b in range(n_blocks):
        s = b * n // n_blocks
        e = (b + 1) * n // n_blocks
        h[s:e] = rng.uniform(0.15, 0.85)
    return repair_feasible(h)

def select_active(profile, max_active=20, beam_frac=0.12):
    c = profile.max()
    med = np.median(profile)
    spread = c - med
    beam = max(0.003, min(0.02, beam_frac * spread))

    act = np.where(profile >= thresh)[0]
    if len(act) > max_active:
        order = np.argsort(profile[act])[-max_active:]
        act = act[order]
    return act
