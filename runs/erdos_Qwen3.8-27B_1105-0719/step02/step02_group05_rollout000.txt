import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time


def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corrs = np.correlate(h, 1.0 - h, mode='full')
    return corrs.max() * dx, corrs


def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-10:
        mask = (h > 1e-8) & (h < 1.0 - 1e-8)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)


def build_h_from_q(q, n):
    m = n // 2
    if n % 2 == 0:
        return np.concatenate([q, 1.0 - q[::-1]])
    else:
        return np.concatenate([q, [0.5], 1.0 - q[::-1]])


def q_from_h(h, n):
    m = n // 2
    return h[:m].copy()


def grad_corr_h(h, d, n):
    g = np.zeros(n)
    if d >= 0:
        if d < n:
            g[d:] += (1.0 - h[:n - d])
            g[:n - d] += (-h[d:])
    else:
        dd = -d
        if dd < n:
            g[:n - dd] += (1.0 - h[dd:])
            g[dd:] += (-h[:n - dd])
    return g


def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def q_smooth_value_grad(q, n, beta):
    m = n // 2
    h = build_h_from_q(q, n)
    dx = 2.0 / n
    corrs = compute_corrs(h)
    val = logsumexp(beta * corrs) / beta * dx
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(len(corrs)) / len(corrs)
    else:
        w /= w_sum
    Gq = np.zeros(m)
    for k in range(len(corrs)):
        if w[k] < 1e-15:
            continue
        d = k - (n - 1)
        G = grad_corr_h(h, d, n)
        Gq += w[k] * (G[:m] - G[m:][::-1])
    return val, Gq * dx


def q_coord_grad(q, n, k_idx):
    m = n // 2
    h = build_h_from_q(q, n)
    dx = 2.0 / n
    d = k_idx - (n - 1)
    G = grad_corr_h(h, d, n)
    return (G[:m] - G[m:][::-1]) * dx


class EpigraphQ:
    def __init__(self, n, active_k):
        self.n = n
        self.m = n // 2
        self.active_k = active_k
        self.dx = 2.0 / n

    def _build_h(self, q):
        n = self.n
        m = self.m
        if n % 2 == 0:
            return np.concatenate([q, 1.0 - q[::-1]])
        else:
            return np.concatenate([q, [0.5], 1.0 - q[::-1]])

    def obj(self, x):
        return x[0]

    def obj_grad(self, x):
        g = np.zeros(self.m + 1)
        g[0] = 1.0
        return g

    def constraint(self, x, k_idx):
        t = x[0]
        q = x[1:]
        h = self._build_h(q)
        k = self.active_k[k_idx]
        d = k - (self.n - 1)
        n = self.n
        if d >= 0:
            i_vals = np.arange(0, n - d)
            val = np.sum(h[i_vals + d] * (1.0 - h[i_vals]))
        else:
            dd = -d
            i_vals = np.arange(0, n - dd)
            val = np.sum(h[i_vals] * (1.0 - h[i_vals + dd]))
        return t - val * self.dx

    def constraint_grad(self, x, k_idx):
        q = x[1:]
        h = self._build_h(q)
        k = self.active_k[k_idx]
        d = k - (self.n - 1)
        n = self.n
        g = np.zeros(self.m + 1)
        g[0] = 1.0
        G = grad_corr_h(h, d, n)
        g[1:] = -(G[:self.m] - G[self.m:][::-1]) * self.dx
        return g


def solve_epigraph_q(q0, n, active_k, t0, maxiter=300):
    m = n // 2
    x0 = np.concatenate([[t0], q0.copy()])
    ep = EpigraphQ(n, active_k)
    cons = []
    for i in range(len(active_k)):
        cons.append({
            'type': 'ineq',
            'fun': lambda x, i=i: ep.constraint(x, i),
            'jac': lambda x, i=i: ep.constraint_grad(x, i)
        })
    bounds = [(0.0, 5.0)] + [(0.0, 1.0)] * m
    try:
        res = minimize(ep.obj, x0, jac=ep.obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'eps': 1e-8})
        q_out = np.clip(res.x[1:], 0.0, 1.0)
        return q_out, res.x[0], res
    except Exception:
        return q0, t0, None

        q_interp = q_from_h(h_interp, n_try)
        t_lim = min(remaining() - 30, 80)
        if t_lim < 30:
            continue
        q_ns, c5_ns = active_set_optimize(q_interp, n_try, t_lim)
        h_ns = build_h_from_q(q_ns, n_try)
        s_ns, _ = exact_score(h_ns)
        if s_ns < best_c5 - 1e-12:
            best_c5 = s_ns
            best_h = h_ns

    # Phase 7: Final polish - homotopy + epigraph on best
    n_final = len(best_h)
    if n_final % 2 != 0:
        best_h = best_h[:n_final - 1]
        n_final = len(best_h)
    m_final = n_final // 2
    ts_final = n_final / 2.0
    q_final = q_from_h(best_h, n_final)

    if remaining() > 40:
        for beta in [2000, 6000, 15000]:
            if remaining() < 20:
                break
            h_f = build_h_from_q(q_final, n_final)
            
            def q_val_f(x):
                return q_smooth_value_grad(x, n_final, beta)[0]
            
            def q_grad_f(x):
                return q_smooth_value_grad(x, n_final, beta)[1]
            
            try:
                res = minimize(q_val_f, q_final.copy(), jac=q_grad_f, method='L-BFGS-B',
                               bounds=[(0.0, 1.0)] * m_final,
                               options={'maxiter': 1000, 'ftol': 1e-18, 'gtol': 1e-14})
                q_new = np.clip(res.x, 0.0, 1.0)
                h_new = build_h_from_q(q_new, n_final)
                s_new, _ = exact_score(h_new)
                if s_new < best_c5 - 1e-14:
                    best_c5 = s_new
                    best_h = h_new
                    q_final = q_new.copy()
            except Exception:
                pass

    if remaining() > 30:
        t_lim = min(remaining() - 10, 80)
        q_fp, c5_fp = active_set_optimize(q_final, n_final, t_lim)
        h_fp = build_h_from_q(q_fp, n_final)
        s_fp, _ = exact_score(h_fp)
        if s_fp < best_c5 - 1e-14:
            best_c5 = s_fp
            best_h = h_fp

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    n_out = len(best_h)
    ts_out = n_out / 2.0
    best_h = project_feasible(best_h, ts_out)
    final_score, _ = exact_score(best_h)
    best_c5 = min(best_c5, final_score)

    return best_h, best_c5, n_out
