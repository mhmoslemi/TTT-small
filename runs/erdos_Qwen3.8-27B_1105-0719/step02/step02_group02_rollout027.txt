import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time

def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def c5_of_h(h, dx):
    return float(compute_corrs(h).max() * dx)

def project_feasible(x, target_sum):
    n = len(x)
    lo = x.min() - 1.0
    hi = x.max() + 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def grad_corr_k(h, k, n):
    """Gradient of correlation at lag index k with respect to h."""
    offset = k - (n - 1)
    grad = np.zeros(n)
    if offset == 0:
        grad[:] = 1.0 - 2.0 * h
        return grad
    d = abs(offset)
    if offset > 0:
        # c = sum_{i=0}^{n-1-d} h[i] * (1-h[i+d])
        # d(c)/d(h[j]) = (1-h[j+d]) for j in [0, n-1-d]
        #                - h[j-d] for j in [d, n-1]

            grad[0:i_max+1] += (1.0 - h[d:i_max+1+d])
            grad[d:] += (-h[0:n-d])
    else:

        # c = sum_{j=0}^{n-1-d} h[j+d] * (1-h[j])

        i_max = n - 1 - d

            grad[d:] += (1.0 - h[0:n-d])
            grad[0:i_max+1] += (-h[d:n])
    return grad

def smooth_max_val(h, dx, beta):
    corrs = compute_corrs(h)
    return float(logsumexp(beta * corrs) / beta * dx)

def smooth_max_grad(h, dx, beta):
    n = len(h)
    corrs = compute_corrs(h)
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(len(corrs)) / len(corrs)
    else:
        w /= w_sum
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-10:
            continue
        grad += w[k] * grad_corr_k(h, k, n)
    return grad * dx

def slsqp_smooth(h0, dx, target_sum, beta, maxiter):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{
        'type': 'eq',
        'fun': lambda x: x.sum() - target_sum,
        'jac': lambda x: np.ones(n)
    }]
    
    def obj(x):
        return smooth_max_val(x, dx, beta)
    
    def g(x):
        return smooth_max_grad(x, dx, beta)
    
    try:

def make_pair_complement(v, n):
    """Build full h from first half v (length n//2) using pair complement."""
    m = n // 2
    h = np.zeros(n)
    h[:m] = v
    h[m:] = 1.0 - v[m-1::-1]

    v = (h[:m] + 1.0 - h[m-1::-1]) / 2.0
    return np.clip(v, 0.0, 1.0)

def slsqp_pair_complement(v0, n, dx, maxiter, beta):

def epigraph_solve(h0, n, target_raw, rho, maxiter):
    """Solve epigraph problem: min s + rho/n * ||h - h0||^2
    s.t. corr_k(h) <= target_raw + s for all k
         sum(h) = n/2
         0 <= h <= 1, s >= 0
    """
    m = n  # number of h variables
    total_vars = n + 1  # h + s
    target_sum = n / 2.0
    
    # Bounds
    bounds = [(0.0, 1.0)] * n + [(0.0, target_raw * 2.0)]
    
    def unpack(x):
        h = x[:n]
        s = x[n]
        return h, s
    
    def obj(x):
        h, s = unpack(x)
        return s + (rho / n) * np.sum((h - h0) ** 2)
    
    def obj_grad(x):
        h, s = unpack(x)
        g = np.zeros(total_vars)
        g[:n] = 2.0 * (rho / n) * (h - h0)
        g[n] = 1.0
        return g
    
    # Equality constraint: sum(h) = n/2
    def eq_fun(x):
        return x[:n].sum() - target_sum
    
    def eq_jac(x):
        j = np.zeros(total_vars)
        j[:n] = 1.0
        return j
    
    # Inequality constraints: corr_k(h) - target_raw - s <= 0
    num_corrs = 2 * n - 1
    
    def ineq_fun(x):
        h = x[:n]
        s = x[n]
        corrs = compute_corrs(h)
        return corrs - target_raw - s
    
    def ineq_jac(x):
        h = x[:n]
        J = np.zeros((num_corrs, total_vars))
        for k in range(num_corrs):
            g_k = grad_corr_k(h, k, n)
            J[k, :n] = g_k
            J[k, n] = -1.0
        return J
    
    x0 = np.concatenate([h0, [max(0.0, compute_corrs(h0).max() - target_raw) + 1e-4]])
    
    constraints = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': lambda x: -ineq_fun(x), 'jac': lambda x: -ineq_jac(x)}
    ]
    
    try:
        res = minimize(obj, x0, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        h_out = project_feasible(res.x[:n], target_sum)
    except Exception:
        h_out = h0.copy()
    return h_out, c5_of_h(h_out, 2.0/n)

def project_feasible(x, target_sum):
    n = len(x)
    lo = x.min() - 1.0
    hi = x.max() + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-13:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)

def grad_corr_k(h, k, n):
    offset = k - (n - 1)
    grad = np.zeros(n)
    if offset == 0:
        grad[:] = 1.0 - 2.0 * h
        return grad
    d = abs(offset)
    if d >= n:
        return grad
    i_max = n - 1 - d
    if i_max < 0:
        return grad
    if offset > 0:
        grad[0:i_max + 1] += (1.0 - h[d:i_max + 1 + d])
        grad[d:n] += (-h[0:n - d])
    else:
        grad[d:n] += (1.0 - h[0:n - d])
        grad[0:i_max + 1] += (-h[d:n])
    return grad

        res = minimize(prob.objective, x0, jac=prob.gradient, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        h_out = project_feasible(res.x[:n], n / 2.0)
    except Exception:
        h_out = h0.copy()
    return h_out, c5_of_h(h_out, 2.0 / n)


def make_pair_complement(v, n):
    m = n // 2
    h = np.zeros(n)
    h[:m] = v
    h[m:] = 1.0 - v[m - 1::-1]
    return h


def extract_half(h, n):
    m = n // 2
    v = (h[:m] + 1.0 - h[m - 1::-1]) / 2.0
    return np.clip(v, 0.0, 1.0)


class PairCompProb:
    def __init__(self, n, dx, beta, top_m):
        self.n = n
        self.dx = dx
        self.beta = beta
        self.top_m = top_m
        self.m = n // 2

    def _build_h(self, v):
        return make_pair_complement(v, self.n)

    def objective(self, v):
        h = self._build_h(v)
        corrs = compute_corrs(h)
        if self.top_m is not None and self.top_m < len(corrs):
            idx = np.argpartition(corrs, -self.top_m)[-self.top_m:]
            corrs_sub = corrs[idx]
        else:
            corrs_sub = corrs
        return float(logsumexp(self.beta * corrs_sub) / self.beta * self.dx)

    def gradient(self, v):
        h = self._build_h(v)
        corrs = compute_corrs(h)
        if self.top_m is not None and self.top_m < len(corrs):
            idx = np.argpartition(corrs, -self.top_m)[-self.top_m:]
            top_corrs = corrs[idx]
        else:
            idx = np.arange(len(corrs))
            top_corrs = corrs
        shifted = top_corrs - top_corrs.max()
        w = np.exp(self.beta * shifted)
        ws = w.sum()
        if ws < 1e-30:
            w = np.ones(len(idx)) / len(idx)
        else:
            w /= ws
        g_h = np.zeros(self.n)
        for i, k in enumerate(idx):
            if w[i] > 1e-10:
                g_h += w[i] * grad_corr_k(h, k, self.n)
        g_h *= self.dx
        g_v = g_h[:self.m] - g_h[self.m - 1::-1]
        return g_v


def slsqp_pair_comp(v0, n, dx, maxiter, beta, top_m=None):
    prob = PairCompProb(n, dx, beta, top_m)
    m = n // 2
    bounds = [(0.0, 1.0)] * m
    try:
        res = minimize(prob.objective, v0, jac=prob.gradient, method='SLSQP',
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-14})
        v_opt = np.clip(res.x, 0.0, 1.0)
    except Exception:
        v_opt = v0.copy()
    h_out = make_pair_complement(v_opt, n)
    h_out = project_feasible(h_out, n / 2.0)
    return h_out, c5_of_h(h_out, dx)


def subgrad_momentum(h0, dx, target_sum, max_iter, top_m):
    n = len(h0)
    h = h0.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    alpha = 0.1
    gamma = 0.8
    no_improve = 0

    for it in range(max_iter):
        corrs = compute_corrs(h)
        max_c = corrs.max()
        c5_val = max_c * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 5000:
            break

        top_idx = np.argsort(corrs)[-top_m:]
        grad = np.zeros(n)
        for k in top_idx:
            grad += grad_corr_k(h, k, n)
        grad /= len(top_idx)
        grad *= dx

        v = gamma * v - alpha * grad
        h_new = project_feasible(h + v, target_sum)
        new_c5 = c5_of_h(h_new, dx)

        if new_c5 < c5_val - 1e-15:
            h = h_new
            alpha = min(alpha * 1.002, 0.3)
        else:
            alpha *= 0.95
            v *= 0.3
        if alpha < 1e-12:
            break

    return best_h, best_c5


def make_binary_start(n, rng):
    if n % 2 == 0:
        m = n // 2
        h = np.zeros(n)
        idx = rng.permutation(n)
        h[idx[:m]] = 1.0
    else:
        m = n // 2
        h = np.zeros(n)
        h[m] = 0.5
        remaining = list(range(n))
        remaining.remove(m)
        perm = rng.permutation(len(remaining))
        for i in range(m):
            h[remaining[perm[i]]] = 1.0
    return h


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()

    def remaining():
        return budget_s - (time.time() - t0)

    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')

    h0 = np.array(initial_h_values, dtype=float).ravel()
    n = len(h0)
    dx = 2.0 / n
    target_sum = n / 2.0
    h0 = project_feasible(h0, target_sum)
    best_h = h0.copy()
    best_c5 = c5_of_h(h0, dx)

    # Phase 1: Truncated smooth max with increasing beta
    for beta, tm in [(30, 20), (80, 20), (200, 25), (500, 25), (1000, 30)]:
        if remaining() < 35:
            break
        h_opt, c5_val = slsqp_smooth(best_h, dx, target_sum, beta, 500, top_m=tm)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_opt

    # Phase 2: Full smooth max high beta
    for beta in [3000, 8000, 15000]:
        if remaining() < 30:
            break
        h_opt, c5_val = slsqp_smooth(best_h, dx, target_sum, beta, 600, top_m=None)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_opt

    # Phase 3: Pair complement optimization
    if remaining() > 120 and n % 2 == 0:
        v0 = extract_half(best_h, n)
        for beta, tm in [(200, 25), (800, 25), (3000, 20)]:
            if remaining() < 30:
                break
            h_pc, c5_pc = slsqp_pair_comp(v0, n, dx, 500, beta, top_m=tm)
            if c5_pc < best_c5:
                best_c5 = c5_pc
                best_h = h_pc
            v0 = extract_half(best_h, n)

    # Phase 4: Epigraph polishing
    if remaining() > 150:
        cur_dx = 2.0 / len(best_h)
        cur_n = len(best_h)
        for frac in [0.9995, 0.999, 0.998, 0.997, 0.996, 0.995]:
            if remaining() < 25:
                break
            tgt = best_c5 * frac
            tgt_raw = tgt * cur_n / 2.0
            improved = False
            for rho in [0.1, 0.5, 2.0]:
                if remaining() < 15:
                    break
                h_ep, c5_ep = epigraph_solve(best_h, cur_n, tgt_raw, rho, 350, top_m=25)
                if c5_ep < best_c5:
                    best_c5 = c5_ep
                    best_h = h_ep
                    improved = True
            if not improved:
                continue

    # Phase 5: Momentum subgradient for rough improvement
    if remaining() > 120:
        h_sg, c5_sg = subgrad_momentum(best_h, dx, target_sum, 20000, top_m=15)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg
        # Polish after subgradient
        if remaining() > 40:
            h_opt, c5_val = slsqp_smooth(best_h, dx, target_sum, 2000, 500, top_m=25)
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h_opt

    # Phase 6: Multi-start perturbations
    if remaining() > 120:
        for trial in range(12):
            if remaining() < 20:
                break
            mag = 0.005 + 0.02 * (trial / 12.0)
            h_pert = best_h + rng.normal(0, mag, len(best_h))
            h_pert = project_feasible(h_pert, target_sum)
            h_opt, c5_val = slsqp_smooth(h_pert, dx, target_sum, 300, 350, top_m=20)
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h_opt
                if remaining() > 30:
                    h_opt2, c5_val2 = slsqp_smooth(best_h, dx, target_sum, 5000, 500, top_m=20)
                    if c5_val2 < best_c5:
                        best_c5 = c5_val2
                        best_h = h_opt2

    # Phase 7: Dimension sweep
    if remaining() > 100:
        n_base = len(best_h)
        n_candidates = []
        for delta in [2, 4, 6, 8, 10, 14, 20, 30]:
            n_candidates.append(n_base + delta)
            n_candidates.append(n_base - delta)
        n_candidates = [x for x in n_candidates if 30 <= x <= 200]
        n_candidates = sorted(set(n_candidates))

        for n_try in n_candidates:
            if remaining() < 40:
                break
            dx_try = 2.0 / n_try
            ts_try = n_try / 2.0
            x_old = (np.arange(len(best_h)) + 0.5) * (2.0 / len(best_h))
            x_new = (np.arange(n_try) + 0.5) * (2.0 / n_try)
            h_interp = np.interp(x_new, x_old, best_h)
            h_interp = project_feasible(h_interp, ts_try)

            h_opt, c5_val = slsqp_smooth(h_interp, dx_try, ts_try, 200, 400, top_m=25)
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h_opt
                dx = dx_try
                target_sum = ts_try
                if remaining() > 30:
                    h_opt2, c5_val2 = slsqp_smooth(best_h, dx, target_sum, 5000, 500, top_m=20)
                    if c5_val2 < best_c5:
                        best_c5 = c5_val2
                        best_h = h_opt2

    # Phase 8: Binary starts with pair complement structure
    if remaining() > 60:
        n_cur = len(best_h)
        dx_cur = 2.0 / n_cur
        ts_cur = n_cur / 2.0
        for trial in range(8):
            if remaining() < 15:
                break
            v_bin = (rng.rand(n_cur // 2) > 0.4).astype(float)
            h_bin = make_pair_complement(v_bin, n_cur)
            h_bin = project_feasible(h_bin, ts_cur)
            h_opt, c5_val = slsqp_smooth(h_bin, dx_cur, ts_cur, 200, 300, top_m=20)
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h_opt

    # Phase 9: Final intensive polish
    n_final = len(best_h)
    dx_final = 2.0 / n_final
    ts_final = n_final / 2.0

    if remaining() > 40:
        for beta in [10000, 20000, 50000]:
            if remaining() < 15:
                break
            h_f, c5_f = slsqp_smooth(best_h, dx_final, ts_final, beta, 800, top_m=15)
            if c5_f < best_c5:
                best_c5 = c5_f
                best_h = h_f

    # Phase 10: Final epigraph
    if remaining() > 25:
        for scale in [0.9998, 0.9995, 0.999]:
            if remaining() < 12:
                break
            tgt_raw = (best_c5 * scale) * n_final / 2.0
            for rho in [0.05, 0.3]:
                if remaining() < 10:
                    break
                h_ep, c5_ep = epigraph_solve(best_h, n_final, tgt_raw, rho, 400, top_m=20)
                if c5_ep < best_c5:
                    best_c5 = c5_ep
                    best_h = h_ep

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_c5 = c5_of_h(best_h, 2.0 / len(best_h))

    return best_h, best_c5, len(best_h)
