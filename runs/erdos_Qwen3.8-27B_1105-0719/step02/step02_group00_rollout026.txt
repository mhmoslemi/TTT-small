def compute_profile_and_grad(h):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    P = np.zeros(M)
    G = np.zeros((M, n))
    
    for m in range(M):
        d = m - (n - 1)
        if d == 0:
            P[m] = dx * np.sum(h * (1.0 - h))
            G[m, :] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            # P = dx * sum_{i=0}^{n-1-d} h[i] * (1-h[i+d])
            k = n - 1 - d
            P[m] = dx * np.sum(h[:k+1] * (1.0 - h[d:]))
            # Gradient:
            # For j = 0..k: G[m,j] = dx * (1 - h[j+d])
            G[m, 0:k+1] = dx * (1.0 - h[d:k+d+1])
            # For j = d..n-1: G[m,j] -= dx * h[j-d]
            G[m, d:] -= dx * h[:k+1]
        else:
            e = -d
            k = n - 1 - e
            # P = dx * sum_{i=0}^{k} h[i+e] * (1-h[i])
            P[m] = dx * np.sum(h[e:e+k+1] * (1.0 - h[:k+1]))

            # For j = e..n-1: G[m,j] = dx * (1 - h[j-e])
            G[m, e:] = dx * (1.0 - h[:k+1])
            # For j = 0..k: G[m,j] -= dx * h[j+e]
            G[m, :k+1] -= dx * h[e:e+k+1]

    n = len(h)

def repair_h(vec):
    n = len(vec)
    if n < 1:
        return np.array([0.5])
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity, kind='mergesort')
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx])
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-14:
                break
    else:
        order = np.argsort(h, kind='mergesort')
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-14:
                break

    if abs(diff2) > 1e-15:

        else:
            idx = int(np.argmax(h))
