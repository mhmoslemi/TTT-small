    h[:m] = q
    h[m:] = 1.0 - q[::-1]

    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    return float(c.max())

    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def grad_lag(h, j, n):
    """Gradient of the j-th correlation value (scaled) w.r.t. h."""
    d = j - (n - 1)
    dx = 2.0 / n
    g = np.zeros(n)
    lo1 = max(0, d)
    hi1 = min(n - 1, n - 1 + d)
    if lo1 <= hi1:
        g[lo1:hi1+1] += 1.0 - h[lo1-d:hi1+1-d]
    lo2 = max(0, -d)
    hi2 = min(n - 1, n - 1 - d)
    if lo2 <= hi2:
        g[lo2:hi2+1] -= h[lo2+d:hi2+1+d]
    return dx * g  # scaled

    """Gradient of the j-th correlation value w.r.t. q variables."""
    gh = grad_lag(h, j, n)
    m = n // 2
    return gh[:m] - gh[n-1:m-1:-1]

    mx = c_vals.max()

                all_idx.add(a + delta)
    return np.sort(list(all_idx))

    m = n // 2
    h = build_h(q, n)
    c_vals = corr_scaled(h)

    nvar = 2 * m + 1

    bounds.append((0.0, c_vals.max() + R * 10))  # t bound

        if res.status != 0:

        delta_q = res.x[:m]
        t_opt = res.x[2*m]
        return delta_q, t_opt
    except Exception:

def solve_slp(q, n, active_idx, R, mu=0.01):
    m = n // 2
    h = build_h(q, n)
    c_vals = corr_scaled(h)
    nvar = 2 * m + 1
    c_obj = np.zeros(nvar)
    c_obj[m:2*m] = mu
    c_obj[2*m] = 1.0
    
    A_rows = []
    b_rows = []
    
    for j in active_idx:
        gq = grad_q_for_lag(h, j, n)
        c_j = c_vals[j]
        row = np.zeros(nvar)
        row[:m] = gq
        row[2*m] = -1.0
        A_rows.append(row)
        b_rows.append(-c_j)
    
    for k in range(m):
        row = np.zeros(nvar)
        row[k] = 1.0
        row[m+k] = -1.0
        A_rows.append(row)
        b_rows.append(0.0)
        row2 = -row.copy()
        A_rows.append(row2)
        b_rows.append(0.0)
    
    A_ub = np.array(A_rows)
    b_ub = np.array(b_rows)
    
    bounds = []
    for k in range(m):
        lo = max(-q[k], -R)
        hi = min(1.0 - q[k], R)
        if lo > hi:
            lo = hi = 0.0
        bounds.append((lo, hi))
    slack_ub = R * np.sqrt(m)
    for k in range(m):
        bounds.append((0.0, slack_ub))
    bounds.append((0.0, c_vals.max() + 1.0))
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.status != 0:
            return None, None
        return res.x[:m].copy(), float(res.x[2*m])
    except Exception:
        return None, None

def line_search_slp(q, n, active_idx, delta_q, current_c5):
    """Try various step sizes, return best (q_new, c5_new) or (q, current_c5)."""
    best_q = q.copy()
    best_c5 = current_c5
    for lam in [1.0, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05]:
        q_new = np.clip(q + lam * delta_q, 0.0, 1.0)
        h_new = build_h(q_new, n)
        c5_new = eval_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_q = q_new.copy()
    return best_q, best_c5

def interp_h(h_old, n_new):
    n_old = len(h_old)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return h_new

def build_h(q, n):
    m = n // 2
    h = np.empty(n)
    h[:m] = q
    h[m:] = 1.0 - q[::-1]
    return h

def q_from_h(h):
    n = len(h)
    m = n // 2
    return 0.5 * (h[:m] + 1.0 - h[n-1:m-1:-1])

    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def eval_c5(h):
    return float(cs(h).max())

def grad_h(h, j, n):
    """Scaled gradient of c[j] w.r.t. h."""
    d = j - (n - 1)
    dx = 2.0 / n
    g = np.zeros(n)
    lo1, hi1 = max(0, d), min(n-1, n-1+d)
    if lo1 <= hi1:
        g[lo1:hi1+1] += 1.0 - h[lo1-d:hi1+1-d]
    lo2, hi2 = max(0, -d), min(n-1, n-1-d)
    if lo2 <= hi2:
        g[lo2:hi2+1] -= h[lo2+d:hi2+1+d]
    return dx * g

def grad_q(h, j, n):
    gh = grad_h(h, j, n)
    m = n // 2
    return gh[:m] - gh[n-1:m-1:-1]

def sel_active(c, margin=0.08, pad=2):
    mx = c.max()
    thresh = mx - max(1e-10, margin * mx)
    idx = np.where(c >= thresh)[0]
    s = set(int(x) for x in idx)
    for a in list(s):
        for dd in range(-pad, pad+1):
            if 0 <= a+dd < len(c):
                s.add(a+dd)
    return np.array(sorted(s), dtype=int)

def slp_once(q, n, active, R, mu):
    """One SLP LP solve. Returns (delta_q, t) or (None, None)."""
    m = n // 2
    h = build_h(q, n)
    c = cs(h)
    nvar = 2*m + 1
    # obj: t + mu*sum(s)
    cobj = np.zeros(nvar)
    cobj[m:2*m] = mu
    cobj[2*m] = 1.0
    # constraints
    rows, bvec = [], []
    for j in active:
        gq = grad_q(h, j, n)
        row = np.zeros(nvar)
        row[:m] = gq
        row[2*m] = -1.0
        rows.append(row)
        bvec.append(-c[j])
    for k in range(m):
        r1 = np.zeros(nvar); r1[k]=1; r1[m+k]=-1
        r2 = -r1.copy()
        rows.extend([r1, r2])
        bvec.extend([0.0, 0.0])
    A = np.array(rows)
    b = np.array(bvec)
    bounds = []
    for k in range(m):
        lo = max(-q[k], -R)
        hi = min(1.0-q[k], R)
        if lo > hi: lo = hi = 0.0
        bounds.append((lo, hi))
    su = R * 20
    for _ in range(m):
        bounds.append((0.0, su))
    bounds.append((0.0, c.max() + 2.0))
    try:
        res = linprog(cobj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.status == 0:
            return res.x[:m].copy(), float(res.x[2*m])
    except Exception:
        pass
    return None, None

def ls_step(q, n, dq, c5_cur):
    """Line search. Returns (best_q, best_c5)."""
    bq, bc = q.copy(), c5_cur
    for lam in [1.0, 0.6, 0.4, 0.25, 0.15, 0.1, 0.05, 0.03]:
        qn = np.clip(q + lam*dq, 0.0, 1.0)
        cn = eval_c5(build_h(qn, n))
        if cn < bc - 1e-15:
            bq, bc = qn.copy(), cn
    return bq, bc

def slp_loop(q, n, max_iter, R0, time_fn, reserve):
    """Run SLP loop. Returns (best_q, best_c5)."""
    best_q = q.copy()
    h = build_h(q, n)
    best_c5 = eval_c5(h)
    R = R0
    stagnant = 0
    for _ in range(max_iter):
        if time_fn() < reserve:
            break
        h = build_h(q, n)
        c = cs(h)
        c5_cur = c.max()
        active = sel_active(c, margin=0.08, pad=2)
        dq, t = slp_once(q, n, active, R, 0.01)
        if dq is None:
            R *= 0.5
            if R < 1e-4: break
            continue
        qn, cn = ls_step(q, n, dq, c5_cur)
        if cn < best_c5 - 1e-15:
            q = qn
            best_q = qn.copy()
            best_c5 = cn
            stagnant = 0
            R = min(0.2, R*1.3)
        else:
            stagnant += 1
            if stagnant % 3 == 0:
                R *= 0.6
            if stagnant > 20:
                break
    return best_q, best_c5
