= h[i-k] - h[j-k] + h[i+k] - h[j+k]
= (h[i-k] + h[i+k]) - (h[j-k] + h[j+k])

This confirms my earlier formula. For binary h:
Δcorr[k] = (h[i-k] + h[i+k]) - (h[j-k] + h[j+k])

Where out-of-range indices contribute 0.

So to minimize corr[k*], we want to pick i (from S) with small (h[i-k*] + h[i+k*]) and j (from S̄) with large (h[j-k*] + h[j+k*]).

The values (h[p-k*] + h[p+k*]) can be 0, 1, or 2 for binary h.

Strategy:
- Compute neighbor_count[p] = (h[p-k*] if 0≤p-k*<n else 0) + (h[p+k*] if 0≤p+k*<n else 0) for all p.
- Find i ∈ S with minimum neighbor_count[i] (ideally 0).
- Find j ∈ S̄ with maximum neighbor_count[j] (ideally 2).
- The change in corr[k*] = neighbor_count[i] - neighbor_count[j].
- If neighbor_count[i] = 0 and neighbor_count[j] = 2, we reduce corr[k*] by 2!

But we also need to check that this swap doesn't increase any other lag's correlation above the current max. For that, we need to compute Δcorr[k] for all k, which takes O(n) time per candidate.

Algorithm per iteration:
1. Find k* = argmax corr.
2. Compute neighbor_count for lag k*.
3. Generate candidate swaps: (i, j) where i ∈ S has small neighbor_count and j ∈ S̄ has large neighbor_count. Try a limited number of candidates (top 5-10 for each).
4. For each candidate, compute the full Δcorr vector (O(n) using the formula Δcorr[k] = (h[i-k]+h[i+k]) - (h[j-k]+h[j+k]) for all k).
5. Check if max(corr + Δcorr) < current max.
6. Accept the best valid swap.
7. If no valid swap found, try a "neutral" swap (one that keeps the max the same but might help future improvements).

    jm = j - ks
    d = np.zeros(len(ks))
    m1 = (ip >= 0) & (ip < n)
    m2 = (im >= 0) & (im < n)
    m3 = (jp >= 0) & (jp < n)
    m4 = (jm >= 0) & (jm < n)
    d[m1] += h[ip[m1]]
    d[m2] += h[im[m2]]
    d[m3] -= h[jp[m3]]
    d[m4] -= h[jm[m4]]
    return d


def make_golden(n, ts):
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    h = np.zeros(n)
    seen = set()
    k = 0
    while len(seen) < ts:
        pos = int((k * phi) % 1.0 * n) % n
        if pos not in seen:
            h[pos] = 1.0
            seen.add(pos)
        k += 1
    return h


def make_quadratic_residue_like(n, ts):
    p = n
    h = np.zeros(n)
    residues = set()
    for x in range(1, p):
        residues.add((x * x) % p)
    residues.discard(0)
    h[arr] = 0.0
    for r in sorted(residents)[:ts]:
        h[r % n] = 1.0
    count = int(h.sum())
    if count < ts:
        vals = np.where(h == 0)[0]
        extra = ts - count
        for v in vals[:extra]:
            h[v] = 1.0
    elif count > ts:
        vals = np.where(h == 1)[0]
        remove = count - ts
        for v in vals[:remove]:
            h[v] = 0.0
    return h


def make_random_binary(n, ts, rng):
    idx = rng.permutation(n)
    h = np.zeros(n)
    h[idx[:ts]] = 1.0
    return h


def directed_binary_search(h0, dx, rng, time_limit, n_candidates=10, n_verify=5):
    n = len(h0)
    ts = int(round(n / 2))
    ks = np.arange(-(n - 1), n)

    h = np.zeros(n)
    sorted_idx = np.argsort(-h0)
    h[sorted_idx[:ts]] = 1.0

    corrs = compute_corrs(h)
    current_max = float(corrs.max())
    best_c5 = current_max * dx
    best_h = h.copy()

    t_start = time.time()
    no_improve = 0
    rand_count = 0

    while no_improve < 200000:
        if time.time() - t_start > time_limit:
            break

        k_star_idx = int(np.argmax(corrs))
        k_star = int(ks[k_star_idx])

        nc = np.zeros(n)
        if k_star != 0:
            p1 = np.arange(n) + k_star
            p2 = np.arange(n) - k_star
            m1 = (p1 >= 0) & (p1 < n)
            m2 = (p2 >= 0) & (p2 < n)
            nc[m1] += h[p1[m1]]
            nc[m2] += h[p2[m2]]

        S_idx = np.where(h == 1.0)[0]
        Scomp_idx = np.where(h == 0.0)[0]
        if len(S_idx) == 0 or len(Scomp_idx) == 0:
            break

        S_by_nc = S_idx[np.argsort(nc[S_idx])]
        Scomp_by_nc = Scomp_idx[np.argsort(-nc[Scomp_idx])]

        candidates = []
        nti = min(n_candidates, len(S_by_nc))
        ntj = min(n_candidates, len(Scomp_by_nc))
        for ii in range(nti):
            i = int(S_by_nc[ii])
            for jj in range(ntj):
                j = int(Scomp_by_nc[jj])
                if i == j:
                    continue
                d_ks = nc[i] - nc[j]
                if d_ks < -1e-10:
                    candidates.append((d_ks, i, j))

        candidates.sort()

        best_valid = None
        best_new_max = current_max + 1e-14

        n_verify_actual = min(n_verify, len(candidates))
        for idx_c in range(n_verify_actual):
            d_ks, i, j = candidates[idx_c]
            delta = compute_delta_vec(h, i, j, n, ks)
            new_max = float((corrs + delta).max())
            if new_max < best_new_max:
                best_new_max = new_max
                best_valid = (i, j, delta)

        if best_valid is not None:
            i, j, delta = best_valid
            h[i] = 0.0
            h[j] = 1.0
            corrs = corrs + delta
            current_max = float(corrs.max())
            if current_max * dx < best_c5:
                best_c5 = current_max * dx
                best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 1000 == 0 and rand_count < 500:
                i = int(S_idx[rng.randint(len(S_idx))])
                j = int(Scomp_idx[rng.randint(len(Scomp_idx))])
                delta = compute_delta_vec(h, i, j, n, ks)
                h[i] = 0.0
                h[j] = 1.0
                corrs = corrs + delta
                current_max = float(corrs.max())
                rand_count += 1

    return best_h, best_c5


def shift_local_search(h, dx, time_limit):
    n = len(h)
    corrs = compute_corrs(h)
    current_max = float(corrs.max())
    best_c5 = current_max * dx
    best_h = h.copy()
    ks = np.arange(-(n - 1), n)
    t_start = time.time()
    improved = True

    while improved and (time.time() - t_start) < time_limit:
        improved = False
        h_work = best_h.copy()
        corrs_work = compute_corrs(h_work)
        max_work = float(corrs_work.max())

        for i in range(n):
            if h_work[i] != 1.0:
                continue
            for d in [1, 2, 3, -1, -2, -3]:
                j = i + d
                if 0 <= j < n and h_work[j] == 0.0:
                    delta = compute_delta_vec(h_work, i, j, n, ks)
                    new_max = float((corrs_work + delta).max())
                    if new_max < max_work - 1e-14:
                        h_work[i] = 0.0
                        h_work[j] = 1.0
                        corrs_work = corrs_work + delta
                        max_work = new_max
                        improved = True

        if improved:
            if max_work * dx < best_c5:
                best_c5 = max_work * dx
                best_h = h_work.copy()

    return best_h, best_c5


def slsqp_polish(h0, dx, target_sum, beta, maxiter):
    n = len(h0)

    def obj(x):
        corrs = compute_corrs(x)
        m = corrs.max()
        return (m + np.log(np.sum(np.exp(beta * (corrs - m)))) / beta) * dx

    def obj_grad(x):
        corrs = compute_corrs(x)
        m = corrs.max()
        w = np.exp(beta * (corrs - m))
        w /= w.sum()
        grad = np.zeros(n)
        for k in range(len(corrs)):
            if w[k] < 1e-14:
                continue
            offset = k - (n - 1)
            i_lo = max(0, -offset)
            i_hi = min(n - 1, n - 1 - offset)
            if i_lo > i_hi:
                continue
            idx = np.arange(i_lo, i_hi + 1)
            grad[idx + offset] += (1.0 - x[idx])
            grad[idx] += (-x[idx + offset])
        return w @ np.array([
            np.sum((1.0 - x[np.arange(max(0, -(k-(n-1))) if max(0, -(k-(n-1))) <= min(n-1, n-1-(k-(n-1))) else n,
                                       min(n-1, n-1-(k-(n-1))) + 1)]) * 0)
            for k in range(len(corrs))
        ]) * 0 + _true_grad(x, w, n) * dx

    def _true_grad(x, w, n):
        grad = np.zeros(n)
        for k in range(len(corrs)):
            pass
        return grad

    # Simpler gradient computation
    def obj_grad2(x):
        corrs = compute_corrs(x)
        m = corrs.max()
        w = np.exp(beta * (corrs - m))
        w /= w.sum()
        grad = np.zeros(n)
        n_full = len(corrs)
        for k in range(n_full):
            wk = w[k]
            if wk < 1e-14:
                continue
            offset = k - (n - 1)
            # corr[k] = sum_i h[i+offset]*(1-h[i]) for valid i
            # d/dx[j] corr[k]:
            # If j is in the "h[i+offset]" part: i+offset=j => i=j-offset, contributes (1-h[j-offset])
            # If j is in the "(1-h[i])" part: i=j, contributes -h[j+offset]
            i_val = j = None
            # Contribution from "h[i+offset]" term: when j = i+offset, i.e., i = j-offset
            i_pos = j - offset
            if 0 <= i_pos < n:
                grad[j] += wk * (1.0 - x[i_pos])
            # Contribution from "(1-h[i])" term: when i = j
            jp = j + offset
            if 0 <= jp < n:
                grad[j] -= wk * x[jp]
        return grad * dx

    def cons_fun(x):
        return x.sum() - target_sum

    def cons_jac(x):
        return np.ones(n)

    constraints = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    bounds = [(0.0, 1.0)] * n

    try:
        res = minimize(obj, h0, jac=obj_grad2, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        h_out = project_feasible(res.x, target_sum)
    except Exception:
        h_out = project_feasible(h0, target_sum)
    return h_out, c5_of_h(h_out, dx)


def multi_lag_directed_search(h0, dx, rng, time_limit, top_k=3):
    """Target the top k lags simultaneously."""
    n = len(h0)
    ts = int(round(n / 2))
    ks = np.arange(-(n - 1), n)

    h = np.zeros(n)
    sorted_idx = np.argsort(-h0)
    h[sorted_idx[:ts]] = 1.0

    corrs = compute_corrs(h)
    current_max = float(corrs.max())
    best_c5 = current_max * dx
    best_h = h.copy()

    t_start = time.time()
    no_improve = 0

    while no_improve < 100000:
        if time.time() - t_start > time_limit:
            break

        # Get top lags
        top_indices = np.argsort(-corrs)[:top_k]
        top_lags = ks[top_indices]
        top_vals = corrs[top_indices]

        # Compute combined neighbor count
        nc = np.zeros(n)
        for kl in top_lags:
            if kl == 0:
                continue
            p1 = np.arange(n) + kl
            p2 = np.arange(n) - kl
            m1 = (p1 >= 0) & (p1 < n)
            m2 = (p2 >= 0) & (p2 < n)
            nc[m1] += h[p1[m1]]
            nc[m2] += h[p2[m2]]

        S_idx = np.where(h == 1.0)[0]
        Scomp_idx = np.where(h == 0.0)[0]
        if len(S_idx) == 0 or len(Scomp_idx) == 0:
            break

        S_by_nc = S_idx[np.argsort(nc[S_idx])]
        Scomp_by_nc = Scomp_idx[np.argsort(-nc[Scomp_idx])]

        best_valid = None
        best_new_max = current_max + 1e-14

        nti = min(8, len(S_by_nc))
        ntj = min(8, len(Scomp_by_nc))

        for ii in range(nti):
            i = int(S_by_nc[ii])
            for jj in range(ntj):
                j = int(Scomp_by_nc[jj])
                if i == j:
                    continue
                delta = compute_delta_vec(h, i, j, n, ks)
                new_corrs = corrs + delta
                new_max = float(new_corrs.max())
                if new_max < best_new_max:
                    best_new_max = new_max
                    best_valid = (i, j, delta)

        if best_valid is not None:
            i, j, delta = best_valid
            h[i] = 0.0
            h[j] = 1.0
            corrs = corrs + delta
            current_max = float(corrs.max())
            if current_max * dx < best_c5:
                best_c5 = current_max * dx
                best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 2000 == 0 and no_improve < 20000:
                i = int(S_idx[rng.randint(len(S_idx))])
                j = int(Scomp_idx[rng.randint(len(Scomp_idx))])
                delta = compute_delta_vec(h, i, j, n, ks)
                h[i] = 0.0
                h[j] = 1.0
                corrs = corrs + delta
                current_max = float(corrs.max())

    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    global initial_h_values

    # Start from parent if available
    if 'initial_h_values' in dir() and initial_h_values is not None:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_dx = 2.0 / parent_n
        parent_h = project_feasible(parent_h, parent_n / 2.0)
        parent_c5 = c5_of_h(parent_h, parent_dx)
    else:
        parent_h = None
        parent_c5 = 1.0
        parent_n = 126
        parent_dx = 2.0 / parent_n

    best_h = parent_h
    best_c5 = parent_c5
    best_n = parent_n

    n_values = [128, 126, 120, 112, 104, 96, 88, 80, 72]
    if parent_n in n_values:
        n_values.remove(parent_n)
    n_values.insert(0, parent_n)

    for n in n_values:
        if time.time() - t0 > budget_s - 60:
            break

        dx = 2.0 / n
        ts = n // 2
        remaining = budget_s - (time.time() - t0)

        # Determine time per n based on remaining
        n_remaining = len(n_values) - n_values.index(n)
        time_per_n = max(10, (remaining - 60) / max(1, n_remaining))

        # Generate multiple initializations
        inits = []
        inits.append(make_golden(n, ts))
        inits.append(make_random_binary(n, ts, rng))
        inits.append(make_random_binary(n, ts, rng))

        # If we have a parent, interpolate to this n
        if parent_h is not None and parent_n != n:
            x_old = np.linspace(0, 2, parent_n)
            x_new = np.linspace(0, 2, n)
            h_interp = np.interp(x_new, x_old, parent_h)
            h_interp = project_feasible(h_interp, ts)
            inits.append(h_interp)

        # Try each initialization with directed search
        best_local_h = None
        best_local_c5 = float('inf')

        for init in inits:
            if time.time() - t0 > budget_s - 30:
                break
            init_time = min(time_per_n / len(inits), 15.0)
            if init_time < 2:
                break

            h_bin, c5_bin = directed_binary_search(init, dx, rng, init_time)
            if c5_bin < best_local_c5:
                best_local_c5 = c5_bin
                best_local_h = h_bin

            # Shift local search
            if time.time() - t0 > budget_s - 20:
                break
            h_shift, c5_shift = shift_local_search(h_bin, dx, min(5.0, init_time * 0.5))
            if c5_shift < best_local_c5:
                best_local_c5 = c5_shift
                best_local_h = h_shift

        if best_local_h is not None and best_local_c5 < best_c5:
            best_c5 = best_local_c5
            best_h = best_local_h
            best_n = n

        # Multi-lag search on best
        if best_local_h is not None and time.time() - t0 < budget_s - 40:
            ml_time = min(10.0, (budget_s - (time.time() - t0)) * 0.3)
            h_ml, c5_ml = multi_lag_directed_search(best_local_h, dx, rng, ml_time)
            if c5_ml < best_local_c5:
                best_local_c5 = c5_ml
                best_local_h = h_ml
                if c5_ml < best_c5:
                    best_c5 = c5_ml
                    best_h = best_local_h
                    best_n = n

    # Continuous polish on the best solution
    if time.time() - t0 < budget_s - 20 and best_h is not None:
        n_b = len(best_h)
        dx_b = 2.0 / n_b
        ts_b = n_b / 2.0
        polish_time = min(30.0, budget_s - (time.time() - t0) - 10)
        if polish_time > 5:
            for beta in [200, 500, 1000]:
                if time.time() - t0 > budget_s - 10:
                    break
                h_pol, c5_pol = slsqp_polish(best_h, dx_b, ts_b, beta, 500)
                if c5_pol < best_c5:
                    best_c5 = c5_pol
                    best_h = h_pol

            # Final binary + shift after polish
            if time.time() - t0 < budget_s - 10:
                h_final, c5_final = directed_binary_search(best_h, dx_b, rng,
                                                           min(10.0, budget_s - (time.time() - t0) - 5))
                if c5_final < best_c5:
                    best_c5 = c5_final
                    best_h = h_final

    # Final projection and evaluation
    n_final = len(best_h)
    dx_final = 2.0 / n_final
    best_h = project_feasible(best_h, n_final / 2.0)
    best_c5 = c5_of_h(best_h, dx_final)

    return best_h, best_c5, n_final
