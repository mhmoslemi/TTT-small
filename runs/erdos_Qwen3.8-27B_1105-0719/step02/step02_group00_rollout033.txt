        a, b = edges_new[k], edges_new[k + 1]
        total = 0.0
        for j in range(n):
            lo = max(a, edges_old[j])
            hi = min(b, edges_old[j + 1])
            if hi > lo:
                total += h[j] * (hi - lo)
        h_new[k] = total / (b - a)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

def active_descent(h, max_iters=400, active_tol=0.003):
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    momentum_vec = np.zeros(n)
    momentum = 0.3
    for it in range(max_iters):
        profile = compute_profile(h)
        c = float(profile.max())
        thresh = c - active_tol
        active = np.where(profile >= thresh)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(profile))])
        if len(active) > 20:
            top = np.argsort(profile[active])[-20:]
            active = active[top]
        g = np.zeros(n)
        for m in active:
            g += grad_row(h, m, n)
        g -= g.mean()
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            break
        momentum_vec = momentum * momentum_vec + g
        mv_norm = np.linalg.norm(momentum_vec)
        if mv_norm < 1e-15:
            break
        d = -momentum_vec / mv_norm
        pos_mask = d > 1e-12
        neg_mask = d < -1e-12
        alpha_max = 1.0
        if pos_mask.any():
            alpha_max = min(alpha_max, float(((1.0 - h[pos_mask]) / d[pos_mask]).min()))
        if neg_mask.any():
            alpha_max = min(alpha_max, float((h[neg_mask] / (-d[neg_mask])).min()))
        alpha_max = max(alpha_max, 1e-10)
        alpha_max = min(alpha_max, 0.5)
        best_alpha = 0.0
        best_val = c
        best_h = h
        a_ls, b_ls = 0.0, alpha_max
        for _ in range(22):
            c1 = b_ls - gr * (b_ls - a_ls)
            c2 = a_ls + gr * (b_ls - a_ls)
            h1 = repair_feasible(h + c1 * d)
            h2 = repair_feasible(h + c2 * d)
            f1 = exact_score(h1)
            f2 = exact_score(h2)
            if f1 < best_val:
                best_val = f1
                best_alpha = c1
                best_h = h1.copy()
            if f2 < best_val:
                best_val = f2
                best_alpha = c2
                best_h = h2.copy()
            if f1 < f2:
                b_ls = c2
            else:
                a_ls = c1
            if b_ls - a_ls < 1e-13:
                break
        if best_val >= c - 1e-15:
            break
        h = best_h
        if best_val < best_c:
            best_c = best_val
    return h, best_c

def slsqp_epigraph(h_start, maxiter=200):
    n = len(h_start)
    dx = 2.0 / n
    M = 2 * n - 1
    c0 = exact_score(h_start)
    t0 = c0 + 1e-4
    def obj(x):
        return x[-1]
    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    def eq_fun(x):
        return x[:-1].sum() - n / 2.0
    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g
    def ineq_fun(x):
        h = x[:-1]
        t = x[-1]
        profile = np.correlate(h, 1 - h, mode="full") * dx
        return t - profile
    def ineq_jac(x):
        h = x[:-1]
        J = np.zeros((M, n + 1))
        for m in range(M):
            J[m, :n] = -grad_row(h, m, n)
        J[:, -1] = 1.0
        return J
    x0 = np.concatenate([h_start, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 2.0)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None and np.isfinite(res.x).all():
            h_new = repair_feasible(res.x[:-1])
            c_new = exact_score(h_new)
            if c_new <= c0 + 1e-8:
                return h_new
    except Exception:
        pass
    return None

def binary_seed(n, pattern_type, rng):
    half = n // 2
    h = np.zeros(n)
    if pattern_type == 0:
        idx = np.arange(n)
        vals = (idx * 0.618033988749895) % 1.0
        top = np.argsort(vals)[-half:]
        h[top] = 1.0
    elif pattern_type == 1:
        idx = np.arange(n)
        vals = (idx * 0.4142135623730951) % 1.0
        top = np.argsort(vals)[-half:]
        h[top] = 1.0
    elif pattern_type == 2:
        period = max(3, n // 8)
        for i in range(n):
            if (i // period) % 2 == 0:
                h[i] = 1.0
        ones = int(h.sum())
        if ones > half:
            excess = ones - half
            one_idx = np.where(h == 1.0)[0]
            to_flip = rng.choice(one_idx, excess, replace=False)
            h[to_flip] = 0.0
        elif ones < half:
            deficit = half - ones
            zero_idx = np.where(h == 0.0)[0]
            to_flip = rng.choice(zero_idx, deficit, replace=False)
            h[to_flip] = 1.0
    elif pattern_type == 3:
        h = (np.sin(np.linspace(0, 6 * np.pi, n)) > 0).astype(float)
        ones = int(h.sum())
        if ones > half:
            excess = ones - half
            one_idx = np.where(h == 1.0)[0]
            to_flip = rng.choice(one_idx, excess, replace=False)
            h[to_flip] = 0.0
        elif ones < half:
            deficit = half - ones
            zero_idx = np.where(h == 0.0)[0]
            to_flip = rng.choice(zero_idx, deficit, replace=False)
            h[to_flip] = 1.0
    else:
        perm = rng.permutation(n)
        h[perm[:half]] = 1.0
    return repair_feasible(h)

def coordinate_polish(h, max_iters=100):
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    for it in range(max_iters):
        profile = compute_profile(h)
        c = float(profile.max())
        thresh = c - 0.002
        active = np.where(profile >= thresh)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(profile))])
        G_a = np.array([grad_row(h, m, n) for m in active])
        improved = False
        idx_sorted_desc = np.argsort(-h)
        idx_sorted_asc = np.argsort(h)
        for attempt in range(30):
            i = int(idx_sorted_desc[attempt % n])
            j = int(idx_sorted_asc[attempt % n])
            if i == j:
                continue
            delta = 0.02 * (h[i] - h[j])
            if abs(delta) < 1e-6:
                continue
            for sign in [1.0, -1.0]:
                h_try = h.copy()
                h_try[i] += sign * delta
                h_try[j] -= sign * delta
                if h_try.min() < -1e-10 or h_try.max() > 1 + 1e-10:
                    continue
                h_try = repair_feasible(h_try)
                s = exact_score(h_try)
                if s < best_c - 1e-14:
                    h = h_try
                    best_c = s
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s - (time.perf_counter() - t_start)

    try:
        parent = np.array(initial_h_values, dtype=np.float64)
    except Exception:
        parent = None
    if parent is None or len(parent) < 4:
        parent = np.full(70, 0.5)
    parent = np.clip(parent, 0.0, 1.0)
    parent = repair_feasible(parent)
    n_parent = len(parent)

    best_h = parent.copy()
    best_c = exact_score(parent)

    # Stage 1: Active descent on parent
    if time_left() > 100:
        h_ad, c_ad = active_descent(best_h, max_iters=300, active_tol=0.004)
        if c_ad < best_c:
            best_h, best_c = h_ad, c_ad

    # Stage 2: Try neighboring dimensions
    n_best = len(best_h)
    dims = []
    for dn in [-6, -4, -2, 2, 4, 6, 8, 10, 12, 14, 16]:
        d = n_best + dn
        if d >= 40 and d <= 140 and d % 2 == 0:
            dims.append(d)
    dims.sort(key=lambda x: abs(x - n_best))

    dim_candidates = []
    for new_n in dims[:8]:
        if time_left() < 30:
            break
        h_interp = interp_to_n(best_h, new_n)
        h_ad, c_ad = active_descent(h_interp, max_iters=200, active_tol=0.004)
        if c_ad < best_c:
            best_h, best_c = h_ad, c_ad
        dim_candidates.append((c_ad, h_interp, new_n))

    # Stage 3: SLSQP on best candidates
    for _ in range(3):
        if time_left() < 30:
            break
        h_sol = slsqp_epigraph(best_h, maxiter=150)
        if h_sol is not None:
            c_sol = exact_score(h_sol)
            if c_sol < best_c - 1e-12:
                best_h, best_c = h_sol, c_sol
            h_ad, c_ad = active_descent(best_h, max_iters=100, active_tol=0.002)
            if c_ad < best_c:
                best_h, best_c = h_ad, c_ad

    # Stage 4: Binary seed exploration
    if time_left() > 120:
        n_explore = len(best_h)
        for pt in range(5):
            if time_left() < 20:
                break
            h_bin = binary_seed(n_explore, pt, rng)
            h_blur = (h_bin[:-1] + h_bin[1:]) / 2.0
            h_blur = np.concatenate([h_blur, [h_blur[-1]]])
            # Smooth: mix binary with 0.5
            h_mix = 0.4 * h_bin + 0.6 * np.full(n_explore, 0.5)
            h_mix = repair_feasible(h_mix)
            h_ad, c_ad = active_descent(h_mix, max_iters=150, active_tol=0.005)
            if c_ad < best_c:
                best_h, best_c = h_ad, c_ad

    # Stage 5: More dimension exploration from best
    if time_left() > 80:
        n_b = len(best_h)
        for new_n in [n_b - 4, n_b + 4, n_b + 8, n_b + 12]:
            if time_left() < 25:
                break
            if new_n < 40 or new_n > 160 or new_n % 2 != 0:
                continue
            h_interp = interp_to_n(best_h, new_n)
            h_ad, c_ad = active_descent(h_interp, max_iters=200, active_tol=0.003)
            if c_ad < best_c:
                best_h, best_c = h_ad, c_ad
                if time_left() > 30:
                    h_sol = slsqp_epigraph(best_h, maxiter=100)
                    if h_sol is not None:
                        c_sol = exact_score(h_sol)
                        if c_sol < best_c:
                            best_h, best_c = h_sol, c_sol

    # Stage 6: Fine-tune active descent with tighter tolerance
    if time_left() > 40:
        h_ad, c_ad = active_descent(best_h, max_iters=300, active_tol=0.001)
        if c_ad < best_c:
            best_h, best_c = h_ad, c_ad

    # Stage 7: Coordinate polish
    if time_left() > 30:
        h_cp, c_cp = coordinate_polish(best_h, max_iters=80)
        if c_cp < best_c:
            best_h, best_c = h_cp, c_cp

    # Stage 8: Final SLSQP
    if time_left() > 20:
        h_sol = slsqp_epigraph(best_h, maxiter=200)
        if h_sol is not None:
            c_sol = exact_score(h_sol)
            if c_sol < best_c:
                best_h, best_c = h_sol, c_sol

    # Final polish
    if time_left() > 10:
        h_ad, c_ad = active_descent(best_h, max_iters=100, active_tol=0.0005)
        if c_ad < best_c:
            best_h, best_c = h_ad, c_ad

    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)

    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            final_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 2:
            final_c = float(eval_result[0])
    except Exception:
        pass

    return best_h, final_c, len(best_h)
