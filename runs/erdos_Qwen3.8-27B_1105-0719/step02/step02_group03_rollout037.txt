import numpy as np
from scipy.optimize import minimize
import time

def project_feasible(x, target_sum):
    # Project to [0,1]^n with sum = target_sum
    n = len(x)
    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) / 2, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
        else:
            h += diff / n
    return np.clip(h, 0.0, 1.0)

def c5_of_h(h, dx):
    return np.correlate(h, 1.0 - h, mode='full').max() * dx

def smooth_max_obj(h, dx, beta):
    corrs = np.correlate(h, 1.0 - h, mode='full')
    m = corrs.max()
    shifted = corrs - m
    return (m + np.log(np.sum(np.exp(beta * shifted))) / beta) * dx

def smooth_max_grad(h, dx, beta):
    n = len(h)
    corrs = np.correlate(h, 1.0 - h, mode='full')
    m = corrs.max()
    shifted = corrs - m
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    w /= w_sum
    grad = np.zeros(n)
    for k in range(len(corrs)):
        if w[k] < 1e-10:
            continue
        e = k - (n - 1)
        # c_e = sum_i h[i] * (1-h[i+e])
        # d(c_e)/d(h[j]) = (1-h[j+e]) if j+e in range, plus -h[j-e] if j-e in range
        i_lo = max(0, -e)
        i_hi = min(n - 1, n - 1 - e)
        if i_lo <= i_hi:
            idx = np.arange(i_lo, i_hi + 1)
            # term where h[j] is the first factor: j in [i_lo, i_hi], contribution (1-h[j+e])
            grad[idx] += (1.0 - h[idx + e]) * w[k]
            # term where h[j] is the second factor: j = i+e, i in [i_lo, i_hi], j in [i_lo+e, i_hi+e]
            j_lo = max(0, i_lo + e)
            j_hi = min(n - 1, i_hi + e)
            if j_lo <= j_hi:
                grad[j_lo:j_hi+1] += (-h[j_lo-e:j_hi-e+1]) * w[k]
    return grad * dx

    n = len(h0)
    def obj(x):
        return smooth_max_obj(x, dx, beta)
    def obj_grad(x):
        return smooth_max_grad(x, dx, beta)
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        h_out = project_feasible(res.x, target_sum)
    except:
        h_out = project_feasible(h0, target_sum)
    return h_out, c5_of_h(h_out, dx)

_slsqp_state = {}

def _slsqp_obj(x):
    dx = _slsqp_state['dx']
    beta = _slsqp_state['beta']
    corrs = np.correlate(x, 1.0 - x, mode='full')
    m = corrs.max()
    shifted = corrs - m
    return (m + np.log(np.sum(np.exp(beta * shifted))) / beta) * dx

def _slsqp_grad(x):
    n = len(x)
    dx = _slsqp_state['dx']
    beta = _slsqp_state['beta']
    corrs = np.correlate(x, 1.0 - x, mode='full')
    m = corrs.max()
    shifted = corrs - m
    w = np.exp(beta * shifted)
    w /= w.sum()
    grad = np.zeros(n)
    for k in range(len(corrs)):
        if w[k] < 1e-10:
            continue
        e = k - (n - 1)
        i_lo = max(0, -e)
        i_hi = min(n - 1, n - 1 - e)

            if j_lo <= j_hi:
                grad[j_lo:j_hi+1] += (-x[j_lo-e:j_hi-e+1]) * w[k]
    return grad * dx

    return x.sum() - _slsqp_state['target_sum']

def _slsqp_cons_jac(x):

def swap_effect(h, i, j, n):
    """Compute change in all correlations when h[i]:1→0 and h[j]:0→1."""
    e_vals = np.arange(-(n-1), n)  # length 2n-1
    # For i (1→0): -(1-h[i+e]) + h[i-e]
    ie = i + e_vals
    mask_ie = (ie >= 0) & (ie < n)
    t1 = np.zeros(len(e_vals))
    t1[mask_ie] = (1.0 - h[ie[mask_ie]])
    ie2 = i - e_vals
    mask_ie2 = (ie2 >= 0) & (ie2 < n)
    t2 = np.zeros(len(e_vals))
    t2[mask_ie2] = h[ie2[mask_ie2]]
    delta_i = -t1 + t2
    
    # For j (0→1): (1-h[j+e]) - h[j-e]
    je = j + e_vals
    mask_je = (je >= 0) & (je < n)
    t3 = np.zeros(len(e_vals))
    t3[mask_je] = (1.0 - h[je[mask_je]])
    je2 = j - e_vals
    mask_je2 = (je2 >= 0) & (je2 < n)
    t4 = np.zeros(len(e_vals))
    t4[mask_je2] = h[je2[mask_je2]]
    delta_j = t3 - t4
    
    return delta_i + delta_j
