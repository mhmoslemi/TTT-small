    bounds = [(max(0.0, q0[i] - rho), min(1.0, q0[i] + rho)) for i in range(m)]
    bounds.append((0.0, 2.0))
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        return None, corrs0.max()
    return res.x[:m], float(res.fun)

def solve_tangent_full(h0, n_full, all_k, rho):
    dx = 2.0 / n_full
    corrs0 = corr_raw(h0) * dx
    grads = all_grads_full(h0, all_k)
    A_ub = np.zeros((len(all_k), n_full + 1))
    for ik in range(len(all_k)):
        A_ub[ik, :n_full] = grads[ik]
        A_ub[ik, n_full] = -1.0
    b_ub = np.array([corrs0[all_k[ik]] - grads[ik] @ h0 for ik in range(len(all_k))])
    c = np.zeros(n_full + 1)
    c[n_full] = 1.0
    bounds = [(max(0.0, h0[i] - rho), min(1.0, h0[i] + rho)) for i in range(n_full)]
    bounds.append((0.0, 2.0))
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        return None, corrs0.max()
    return res.x[:n_full], float(res.fun)

def line_search_sym_dir(q0, direction, n_full, max_iters=40):
    best_a = 0.0
    best_s = eval_score(build_sym_h(q0))
    # Try several alphas
    alphas = [0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]
    for a in alphas:
        q_try = np.clip(q0 + a * direction, 0.0, 1.0)
        s = eval_score(build_sym_h(q_try))
        if s < best_s:
            best_s = s
            best_a = a
    if best_a <= 0:
        return 0.0, best_s
    # Golden section around best
    lo = max(0.0, best_a * 0.6)
    hi = best_a * 1.5
    for _ in range(max_iters):
        g1 = hi - 0.382 * (hi - lo)
        g2 = lo + 0.382 * (hi - lo)
        s1 = eval_score(build_sym_h(np.clip(q0 + g1 * direction, 0.0, 1.0)))
        s2 = eval_score(build_sym_h(np.clip(q0 + g2 * direction, 0.0, 1.0)))
        if s1 < s2:
            hi = g2
        else:
            lo = g1
        if hi - lo < 1e-9:
            break
    mid = (lo + hi) / 2.0
    s_mid = eval_score(build_sym_h(np.clip(q0 + mid * direction, 0.0, 1.0)))
    if s_mid < best_s:
        best_s = s_mid
        best_a = mid
    return best_a, best_s

def line_search_full_dir(h0, direction, n_full, max_iters=40):
    best_a = 0.0
    best_s = eval_score(h0)
    alphas = [0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]
    for a in alphas:
        h_try = np.clip(h0 + a * direction, 0.0, 1.0)
        h_try = project_feasible(h_try, n_full / 2.0)
        s = eval_score(h_try)
        if s < best_s:
            best_s = s
            best_a = a
    if best_a <= 0:
        return 0.0, best_s
    lo = max(0.0, best_a * 0.6)
    hi = best_a * 1.5
    for _ in range(max_iters):
        g1 = hi - 0.382 * (hi - lo)
        g2 = lo + 0.382 * (hi - lo)
        s1 = eval_score(project_feasible(np.clip(h0 + g1 * direction, 0.0, 1.0), n_full / 2.0))
        s2 = eval_score(project_feasible(np.clip(h0 + g2 * direction, 0.0, 1.0), n_full / 2.0))
        if s1 < s2:
            hi = g2
        else:
            lo = g1
        if hi - lo < 1e-9:
            break
    mid = (lo + hi) / 2.0
    s_mid = eval_score(project_feasible(np.clip(h0 + mid * direction, 0.0, 1.0), n_full / 2.0))
    if s_mid < best_s:
        best_s = s_mid
        best_a = mid
    return best_a, best_s

def transfer_line_search(h0, a, b, n_full):
    h = h0.copy()
    ha, hb = h[a], h[b]
    lo = max(-ha, hb - 1.0)
    hi = min(1.0 - ha, -hb)
    if hi - lo < 1e-12:
        return h0.copy(), eval_score(h0)
    best_d = 0.0
    best_s = eval_score(h0)
    n_grid = 30
    for ig in range(n_grid + 1):
        d = lo + (hi - lo) * ig / n_grid
        h_try = h0.copy()
        h_try[a] = ha + d
        h_try[b] = hb - d
        s = eval_score(h_try)
        if s < best_s:
            best_s = s
            best_d = d
    if best_d == 0:
        return h0.copy(), best_s
    g_lo = lo + (hi - lo) * max(0, (n_grid - 1) / (n_grid + 1) * (best_d - lo) / (hi - lo) - 1) if hi > lo else lo
    g_hi = lo + (hi - lo) * min(1, (best_d - lo) / (hi - lo) + 1.0 / n_grid) if hi > lo else hi
    g_lo = max(lo, min(hi, best_d - (hi - lo) / n_grid))
    g_hi = min(hi, max(lo, best_d + (hi - lo) / n_grid))
    for _ in range(40):
        m1 = g_hi - 0.382 * (g_hi - g_lo)
        m2 = g_lo + 0.382 * (g_hi - g_lo)
        h1 = h0.copy(); h1[a] = ha + m1; h1[b] = hb - m1
        h2 = h0.copy(); h2[a] = ha + m2; h2[b] = hb - m2
        s1 = eval_score(h1)
        s2 = eval_score(h2)
        if s1 < s2:
            g_hi = m2
        else:
            g_lo = m1
        if g_hi - g_lo < 1e-10:
            break
    best_d = (g_lo + g_hi) / 2.0
    h_best = h0.copy()
    h_best[a] = ha + best_d
    h_best[b] = hb - best_d
    s_best = eval_score(h_best)
    if s_best < best_s:
        best_s = s_best
    return h_best, best_s

def tangent_sym_optimize(q0, n_full, max_iters, time_limit, verbose=False):
    m = n_full // 2
    all_k = list(range(2 * n_full - 1))
    q = np.clip(q0.copy(), 0.0, 1.0)
    best_score = eval_score(build_sym_h(q))
    best_q = q.copy()
    rho = 0.1
    stall = 0
    for it in range(max_iters):
        if time.time() > time_limit:
            break
        q_lp, lp_val = solve_tangent_sym(q, n_full, all_k, rho)
        if q_lp is None:
            rho *= 0.5
            stall += 1
            if stall > 10:
                break
            continue
        direction = q_lp - q
        if np.max(np.abs(direction)) < 1e-12:
            break
        alpha, score = line_search_sym_dir(q, direction, n_full, max_iters=30)
        if alpha > 0 and score < best_score - 1e-14:
            q = np.clip(q + alpha * direction, 0.0, 1.0)
            best_score = score
            best_q = q.copy()
            stall = 0
            rho = min(0.5, rho * 1.2)
        else:
            stall += 1
            rho *= 0.7
            if stall > 15 or rho < 1e-5:
                break
    return best_q, best_score, build_sym_h(best_q)

def tangent_full_optimize(h0, n_full, max_iters, time_limit):
    all_k = list(range(2 * n_full - 1))
    h = project_feasible(h0.copy(), n_full / 2.0)
    best_score = eval_score(h)
    best_h = h.copy()
    rho = 0.1
    stall = 0
    for it in range(max_iters):
        if time.time() > time_limit:
            break
        h_lp, lp_val = solve_tangent_full(h, n_full, all_k, rho)
        if h_lp is None:
            rho *= 0.5
            stall += 1
            if stall > 10:
                break
            continue
        direction = h_lp - h
        # Make zero-sum
        direction -= direction.mean()
        if np.max(np.abs(direction)) < 1e-12:
            break
        alpha, score = line_search_full_dir(h, direction, n_full, max_iters=30)
        if alpha > 0 and score < best_score - 1e-14:
            h = project_feasible(np.clip(h + alpha * direction, 0.0, 1.0), n_full / 2.0)
            best_score = score
            best_h = h.copy()
            stall = 0
            rho = min(0.5, rho * 1.15)
        else:
            stall += 1
            rho *= 0.7
            if stall > 15 or rho < 1e-5:
                break
    return best_h, best_score

def transfer_optimize(h0, n_full, n_pairs, time_limit, rng):
    h = project_feasible(h0.copy(), n_full / 2.0)
    best_score = eval_score(h)
    best_h = h.copy()
    attempts = 0
    while attempts < n_pairs and time.time() < time_limit:
        attempts += 1
        # Pick promising pair using gradient info
        corrs = corr_raw(h) * (2.0 / n_full)
        max_val = corrs.max()
        active_k = np.where(corrs >= max_val - 0.01)[0]
        # Use first active lag for gradient
        k0 = active_k[0]
        gh = corr_grad_k_full(h, k0)
        # Want to increase where gh is negative, decrease where positive
        pos = np.argsort(gh)[:n_full // 4]
        neg = np.argsort(-gh)[:n_full // 4]
        a = int(rng.choice(pos))
        b = int(rng.choice(neg))
        if a == b:
            b = (b + 1) % n_full
        h_new, s_new = transfer_line_search(h, a, b, n_full)
        if s_new < best_score - 1e-14:
            best_score = s_new
            best_h = h_new.copy()
            h = h_new
        else:
            h = h_new  # still use for gradient info
    return best_h, best_score

def transfer_to_n(h_old, n_new):
    n_old = len(h_old)
    x_old = (np.arange(n_old) + 0.5) * 2.0 / n_old
    x_new = (np.arange(n_new) + 0.5) * 2.0 / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def rem():
        return budget_s - (time.time() - t0)

    best_h = None
    best_c5 = float('inf')

    try:
        h_parent = np.array(initial_h_values, dtype=float).copy().reshape(-1)
    except (NameError, AttributeError):
        h_parent = None

    if h_parent is not None and len(h_parent) > 10:
        n_p = len(h_parent)
        if abs(np.sum(h_parent) - n_p / 2.0) > 0.5:
            h_parent = project_feasible(h_parent, n_p / 2.0)
        h_parent = np.clip(h_parent, 0.0, 1.0)
        h_parent = project_feasible(h_parent, n_p / 2.0)
        best_h = h_parent.copy()
        best_c5 = eval_score(h_parent)
    else:
        m_p = 63
        x = np.linspace(0, 1, m_p)
        q_p = np.clip(0.5 + 0.3 * np.sin(2 * np.pi * x), 0, 1)
        best_h = build_sym_h(q_p)
        best_c5 = eval_score(best_h)

    n_cur = len(best_h)

    # Phase 1: Symmetric tangent LP (most efficient)
    if n_cur % 2 == 0 and rem() > 100:
        tl = time.time() + min(300, rem() * 0.4)
        q_init = sym_from_h(best_h)
        q_opt, c5_s, h_s = tangent_sym_optimize(q_init, n_cur, max_iters=500, time_limit=tl)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()

    # Phase 2: Full vector tangent LP
    if rem() > 100:
        tl = time.time() + min(250, rem() * 0.35)
        h_opt, c5_f = tangent_full_optimize(best_h, n_cur, max_iters=300, time_limit=tl)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_opt.copy()

    # Phase 3: Transfer line searches
    if rem() > 60:
        tl = time.time() + min(120, rem() * 0.3)
        h_tr, c5_tr = transfer_optimize(best_h, n_cur, n_pairs=200, time_limit=tl, rng=rng)
        if c5_tr < best_c5:
            best_c5 = c5_tr
            best_h = h_tr.copy()

    # Phase 4: After transfers, refine with tangent LP again
    if rem() > 80 and n_cur % 2 == 0:
        tl = time.time() + min(100, rem() * 0.3)
        q_init = sym_from_h(best_h)
        q_opt, c5_s, h_s = tangent_sym_optimize(q_init, n_cur, max_iters=300, time_limit=tl)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()

    if rem() > 60:
        tl = time.time() + min(80, rem() * 0.3)
        h_opt, c5_f = tangent_full_optimize(best_h, n_cur, max_iters=200, time_limit=tl)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_opt.copy()

    # Phase 5: Multi-resolution exploration
    n_base = len(best_h)
    n_candidates = []
    for delta in [-6, -4, -3, -2, -1, 0, 1, 2, 3, 4, 6, 8]:
        nc = n_base + delta
        if 48 <= nc <= 200 and nc % 2 == 0:
            n_candidates.append(nc)
    if n_base // 2 >= 40:
        n_candidates.append(n_base // 2)
    n_candidates = sorted(set(n_candidates))

    for n_try in n_candidates:
        if rem() < 80:
            break
        h_try = transfer_to_n(best_h, n_try)
        n_try = len(h_try)

        # Symmetric tangent LP at new resolution
        q_try = sym_from_h(h_try)
        tl = time.time() + min(100, rem() * 0.3)
        q_o, c5_o, h_o = tangent_sym_optimize(q_try, n_try, max_iters=200, time_limit=tl)
        if c5_o < best_c5:
            best_c5 = c5_o
            best_h = h_o.copy()

        if rem() > 60:
            tl2 = time.time() + min(60, rem() * 0.3)
            h_o2, c5_o2 = tangent_full_optimize(best_h, len(best_h), max_iters=150, time_limit=tl2)
            if c5_o2 < best_c5:
                best_c5 = c5_o2
                best_h = h_o2.copy()

    # Phase 6: Random restarts at current and nearby resolutions
    n_restarts = max(3, int(rem() / 60))
    for i in range(n_restarts):
        if rem() < 60:
            break
        n_r = n_base if i % 3 == 0 else max(50, n_base + int(rng.choice([-4, -2, 2, 4])))
        n_r = min(180, max(50, n_r))
        if n_r % 2 != 0:
            n_r += 1
        m_r = n_r // 2
        # Diverse initialization
        mode = i % 4
        if mode == 0:
            q_r = np.clip(0.5 + 0.4 * rng.randn(m_r) * 0.3, 0, 1)
        elif mode == 1:
            x_r = np.linspace(0, 1, m_r)
            q_r = np.clip(0.5 + 0.35 * np.sin(2 * np.pi * x_r + rng.uniform(0, 2 * np.pi)), 0, 1)
        elif mode == 2:
            q_r = np.clip(np.random.RandomState(seed + i).rand(m_r), 0, 1)
        else:
            q_r = sym_from_h(transfer_to_n(best_h, n_r))
            q_r = np.clip(q_r + 0.1 * rng.randn(m_r), 0, 1)

        h_r = build_sym_h(q_r)
        tl = time.time() + min(80, rem() * 0.4)
        q_o, c5_o, h_o = tangent_sym_optimize(q_r, n_r, max_iters=150, time_limit=tl)
        if c5_o < best_c5:
            best_c5 = c5_o
            best_h = h_o.copy()

    # Phase 7: Final intensive polish
    if rem() > 30 and best_h is not None:
        n_fin = len(best_h)
        # Transfer to a few sizes and polish
        for n_pol in [n_fin, max(50, n_fin - 2), min(180, n_fin + 2)]:
            if rem() < 25:
                break
            if n_pol % 2 != 0:
                n_pol += 1
            h_pol = transfer_to_n(best_h, n_pol)
            n_pol = len(h_pol)
            q_pol = sym_from_h(h_pol)
            tl = time.time() + min(40, rem() * 0.4)
            q_o, c5_o, h_o = tangent_sym_optimize(q_pol, n_pol, max_iters=200, time_limit=tl)
            if c5_o < best_c5:
                best_c5 = c5_o
                best_h = h_o.copy()

    # Final: transfer back to a good size for output (prefer smaller n)
    # Try to find the best across a few sizes
    final_h = best_h
    final_c5 = best_c5
    final_n = len(best_h)

    if rem() > 30:
        # Try slightly smaller sizes
        for ds in [-4, -2, -6]:
            ns = final_n + ds
            if ns < 48 or ns % 2 != 0:
                continue
            h_chk = transfer_to_n(best_h, ns)
            q_chk = sym_from_h(h_chk)
            tl = time.time() + min(25, rem() * 0.5)
            q_o, c5_o, h_o = tangent_sym_optimize(q_chk, ns, max_iters=100, time_limit=tl)
            if c5_o < final_c5:
                final_c5 = c5_o
                final_h = h_o
                final_n = ns

    # Validate
    final_h = np.asarray(final_h, dtype=float).reshape(-1)
    n_out = len(final_h)
    final_h = project_feasible(final_h, n_out / 2.0)
    final_c5 = eval_score(final_h)

    return final_h, final_c5, n_out
