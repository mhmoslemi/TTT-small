```python
import numpy as np
from scipy.optimize import minimize
import time

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx] + 1e-15)
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx] + 1e-15)
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def lse_objective_and_grad(h, beta):
    """Compute log-sum-exp objective and gradient."""
    P = compute_profile(h)
    Pmax = np.max(P)
    s = beta * (P - Pmax)
    # For numerical stability
    s_max = np.max(s)
    exp_s = np.exp(s - s_max)
    S = np.sum(exp_s)
    if S <= 0 or not np.isfinite(S):
        # Fallback: uniform weights
        w = np.ones(len(P)) / len(P)
    else:
        w = exp_s / S
    loss = Pmax + s_max / beta + np.log(S) / beta
    # Gradient
    n = len(h)
    dx = 2.0 / n
    grad = lse_gradient(h, w, dx)
    return loss, grad

def lse_gradient(h, w, dx):
    """Compute gradient of LSE objective given weights w on profile."""
    n = len(h)
    center = n - 1
    grad = np.zeros(n)
    for lag in range(-n+1, n):
        m = center + lag  # index in profile
        wm = w[m]
        if wm < 1e-15:
            continue
        if lag == 0:
            grad += wm * dx * (1.0 - 2.0 * h)
        elif lag > 0:
            # d = lag > 0: P = dx * sum_{j=0}^{n-1-d} h[j+d]*(1-h[j])
            # dP/dh[i] = dx*((1-h[i-d])*1_{i>=d} - h[i+d]*1_{i+d<n})
            for i in range(n):
                g = 0.0
                if i >= lag:
                    g += (1.0 - h[i - lag])
                if i + lag < n:
                    g -= h[i + lag]
                grad[i] += wm * dx * g
        else:
            # e = -lag > 0: P = dx * sum_{j=0}^{n-1-e} h[j]*(1-h[j+e])
            # dP/dh[i] = dx*((1-h[i+e])*1_{i+e<n} - h[i-e]*1_{i>=e})
            e = -lag
            for i in range(n):
                g = 0.0
                if i + e < n:
                    g += (1.0 - h[i + e])
                if i >= e:
                    g -= h[i - e]
                grad[i] += wm * dx * g
    return grad
```

This is: shift(1-h, d, fill=0) - shift(h, -d, fill=0) where "shift by d" means the result at position i is the original at position i-d.

In numpy:

shift_left = np.zeros(n); shift_left[:-d] = h[d:] → this is h[i+d] for i+d<n

So contrib = shift_right - shift_left = (1-h shifted right by d) - (h shifted left by d)

I can compute this for all d at once! But the weights w differ for each d, so I need:

```python
    for d in range(1, n):
        wm_pos = w[center + d]  # lag +d
        if wm_pos > 1e-15:
            # (1-h[i-d]) for i>=d, minus h[i+d] for i+d<n

            term[d:] -= h[d:]  # -h[i+d] for i < n-d, i.e., term[i] = -h[i+d]

            # And term[i] -= h[i+d] for i+d < n
            term1 = np.zeros(n)
            term1[d:] = 1.0 - h[:-d]  # term1[i] = 1-h[i-d] for i>=d
            term2 = np.zeros(n)
            term2[:n-d] = h[d:]  # term2[i] = h[i+d] for i < n-d
            grad += wm_pos * dx * (term1 - term2)
        
        wm_neg = w[center - d]  # lag -d
        if wm_neg > 1e-15:
            # (1-h[i+d]) for i+d<n, minus h[i-d] for i>=d
            term1 = np.zeros(n)
            term1[:n-d] = 1.0 - h[d:]  # term1[i] = 1-h[i+d] for i < n-d
            term2 = np.zeros(n)
            term2[d:] = h[:-d]  # term2[i] = h[i-d] for i>=d
            grad += wm_neg * dx * (term1 - term2)
    return grad
```
