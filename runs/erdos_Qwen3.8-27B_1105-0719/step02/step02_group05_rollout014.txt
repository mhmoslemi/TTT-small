import numpy as np
from scipy.optimize import minimize, golden
from scipy.special import logsumexp
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corrs = np.correlate(h, 1.0 - h, mode='full')
    return corrs.max() * dx, corrs

def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-13:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def build_h_from_q(q, n):
    m = n // 2
    if n % 2 == 0:
        return np.concatenate([q, 1.0 - q[::-1]])
    else:
        return np.concatenate([q, [0.5], 1.0 - q[::-1]])

def q_from_h(h, n):
    m = n // 2
    return h[:m].copy()

def grad_corr_d(h, d, n):
    g = np.zeros(n)
    if d >= 0:
        if d < n:
            g[d:] += (1.0 - h[:n - d])
            g[:n - d] += (-h[d:])
    else:
        dd = -d
        if dd < n:
            g[:n - dd] += (1.0 - h[dd:])
            g[dd:] += (-h[:n - dd])
    return g

def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def smooth_q_value_grad(q, n, beta):
    m = n // 2
    h = build_h_from_q(q, n)
    dx = 2.0 / n
    corrs = compute_corrs(h)
    val = logsumexp(beta * corrs) / beta * dx
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(len(corrs)) / len(corrs)
    else:
        w /= w_sum
    Gq = np.zeros(m)

        d = k - (n - 1)
        G = grad_corr_d(h, d, n)
        Gq += w[k] * (G[:m] - G[m:][::-1])
    return val, Gq * dx

def smooth_full_value_grad(h, n, beta):

def epigraph_q_objective(x, n, active_k):
    m = n // 2
    t = x[0]
    q = x[1:]
    h = build_h_from_q(q, n)
    dx = 2.0 / n
    corrs = compute_corrs(h)
    # Return t - min over active_k of (corrs[k] - t)...

    # violated = max(0, max over active of (corrs[k]*dx - t))
    active_vals = corrs[active_k] * dx
    return t  # objective is just t

    g = np.zeros(m + 1)

def epigraph_q_constraint(x, n, active_k, k_idx):

def solve_epigraph_q(q0, n, active_k, t0, maxiter=200):
    m = n // 2
    x0 = np.concatenate([[t0], q0])
    cons = []
    for i in range(len(active_k)):
        cons.append({
            'type': 'ineq',
            'fun': lambda x, i=i: epigraph_q_constraint(x, n, active_k, i),
            'jac': lambda x, i=i: epigraph_q_constraint_grad(x, n, active_k, i)
        })
    bounds = [(0.0, 5.0)] + [(0.0, 1.0)] * m
    try:
        res = minimize(epigraph_q_objective, x0, jac=lambda x: epigraph_q_grad(x, n, active_k),

                       options={'maxiter': maxiter, 'ftol': 1e-15})
        q_out = np.clip(res.x[1:], 0.0, 1.0)
        return q_out, res.x[0], res
    except Exception:
        return q0, t0, None

def active_set_qp(q_init, n, max_rounds=15, max_active=40):
    """Active-set method in q-space."""
    m = n // 2
    q = q_init.copy()
    dx = 2.0 / n
    best_q = q.copy()
    
    h = build_h_from_q(q, n)
    corrs = compute_corrs(h)
    best_c5 = corrs.max() * dx
    
    # Initial active set: top correlations
    L = min(max_active, 2 * n - 1)
    top_idx = np.argsort(corrs)[::-1][:L]
    active_k = sorted(top_idx.tolist())
    
    for rnd in range(max_rounds):
        t0 = corrs.max() * dx * (1.0 + 1e-4) + 1e-6
        q_new, t_solved, res = solve_epigraph_q(q, n, active_k, t0, maxiter=200)
        h_new = build_h_from_q(q_new, n)
        corrs_new = compute_corrs(h_new)
        c5_new = corrs_new.max() * dx
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_q = q_new.copy()
        
        # Check hidden violations
        hidden = corrs_new > t_solved / dx + 1e-6  # t_solved is in dx units
        # Actually t_solved should already be in the same units as corrs*dx
        # Let me fix: t is in units of the integral, so t >= corr_k * dx
        hidden = corrs_new * dx > t_solved + 1e-7

        corrs = corrs_new

            break

def coordinate_descent_q(q_init, n, n_sweeps=5, tol=1e-12):
    """Cyclic coordinate descent in q-space with exact 1D minimization."""
    m = n // 2
    dx = 2.0 / n
    q = q_init.copy()
    
    def score_from_q(q_):
        h = build_h_from_q(q_, n)
        return compute_corrs(h).max() * dx
    
    current_best = score_from_q(q)
    order = np.arange(m)
    
    for sweep in range(n_sweps):
        for j in order:
            # Minimize over q[j] in [0,1]
            def f_1d(t):
                q_old = q[j]
                q[j] = t
                h = build_h_from_q(q, n)
                s = compute_corrs(h).max() * dx
                q[j] = q_old
                return s
            
            # Use golden section search
            best_t = q[j]
            best_val = current_best
            
            # Try a few points to get a good bracket
            candidates = [best_t, 0.0, 1.0, 0.5, best_t + 0.1, best_t - 0.1]
            candidates = [max(0.0, min(1.0, c)) for c in candidates]
            candidates = list(set(candidates))
            
            vals = [f_1d(c) for c in candidates]
            best_idx = np.argmin(vals)
            best_t = candidates[best_idx]
            best_val = vals[best_idx]
            
            # Golden section refinement
            if best_val < current_best - 1e-15:
                # Narrow around best_t
                lo = max(0.0, best_t - 0.2)
                hi = min(1.0, best_t + 0.2)
                if lo < best_t < hi:
                    from scipy.optimize import minimize_scalar
                    res = minimize_scalar(f_1d, bounds=(lo, hi), method='bounded',
                                          options={'xatol': 1e-10})
                    if res.fun < best_val:
                        best_t = res.x
                        best_val = res.fun
                
                q[j] = best_t
                current_best = best_val
    
    return q, current_best

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corrs = np.correlate(h, 1.0 - h, mode='full')
    return corrs.max() * dx, corrs


def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-13:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)


def build_h_from_q(q, n):
    m = n // 2
    if n % 2 == 0:
        return np.concatenate([q, 1.0 - q[::-1]])
    else:
        return np.concatenate([q, [0.5], 1.0 - q[::-1]])


def q_from_h(h, n):
    m = n // 2
    return h[:m].copy()


def grad_corr_d(h, d, n):
    g = np.zeros(n)
    if d >= 0:
        if d < n:
            g[d:] += (1.0 - h[:n - d])
            g[:n - d] += (-h[d:])
    else:
        dd = -d
        if dd < n:
            g[:n - dd] += (1.0 - h[dd:])
            g[dd:] += (-h[:n - dd])
    return g


def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def smooth_q_value_grad(q, n, beta):
    m = n // 2
    h = build_h_from_q(q, n)
    dx = 2.0 / n
    corrs = compute_corrs(h)
    val = logsumexp(beta * corrs) / beta * dx
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(len(corrs)) / len(corrs)
    else:
        w /= w_sum
    Gq = np.zeros(m)
    for k in range(len(corrs)):
        if w[k] < 1e-14:
            continue
        d = k - (n - 1)
        G = grad_corr_d(h, d, n)
        Gq += w[k] * (G[:m] - G[m:][::-1])
    return val, Gq * dx

def coordinate_descent_q(q_init, n, n_sweeps=5, verbose=False):
    m = n // 2
    q = q_init.copy()

    def score_q(q_):
        h = build_h_from_q(q_, n)
        return compute_corrs(h).max() * (2.0 / n)

    current_best = score_q(q)
    order = np.arange(m)

    for sweep in range(n_sweeps):
        improvement_this_sweep = 0.0
        for j in order:
            old_val = q[j]

            def f1d(t):
                q[j] = t
                h = build_h_from_q(q, n)
                s = compute_corrs(h).max() * (2.0 / n)
                q[j] = old_val
                return s

            # Coarse sampling
            candidates = np.array([old_val, 0.0, 1.0, 0.5,
                                   old_val + 0.15, old_val - 0.15,
                                   old_val + 0.05, old_val - 0.05])
            candidates = np.clip(candidates, 0.0, 1.0)
            unique_cands = np.unique(candidates)
            vals = np.array([f1d(c) for c in unique_cands])
            best_idx = np.argmin(vals)
            best_t = unique_cands[best_idx]
            best_v = vals[best_idx]

            if best_v < current_best - 1e-14:
                # Refine with bounded minimize
                lo = max(0.0, best_t - 0.15)
                hi = min(1.0, best_t + 0.15)
                if hi - lo > 1e-6:
                    res = minimize_scalar(f1d, bounds=(lo, hi), method='bounded',
                                          options={'xatol': 1e-11})
                    if res.fun < best_v:
                        best_t = res.x
                        best_v = res.fun

                q[j] = best_t
                current_best = best_v
                improvement_this_sweep += 1

        if improvement_this_sweep == 0:
            break

    return q, current_best

    for beta in [80, 200, 500, 1000, 2000]:
        if time.time() - t_loc > time_limit * 0.3:
            break
        q = smooth_solve_q(q, n, beta, maxiter=300)
        h = build_h_from_q(q, n)
        s, _ = exact_score(h)
        if s < best_c5 - 1e-15:
            best_c5 = s

    # Coordinate descent
    if time.time() - t_loc < time_limit * 0.6:
        q, c5_cd = coordinate_descent_q(q, n, n_sweeps=4)
        h = build_h_from_q(q, n)
        s, _ = exact_score(h)
        if s < best_c5 - 1e-15:
            best_c5 = s

    # Active set
    if time.time() - t_loc < time_limit * 0.85:
        q, c5_as = active_set_qp(q, n, max_rounds=10, max_active=30)
        h = build_h_from_q(q, n)
        s, _ = exact_score(h)
        if s < best_c5 - 1e-15:
            best_c5 = s

    # Final coordinate polish
    if time.time() - t_loc < time_limit * 0.95:
        q, _ = coordinate_descent_q(q_from_h(h, n), n, n_sweeps=3)
        h = build_h_from_q(q, n)
        s, _ = exact_score(h)
        if s < best_c5 - 1e-15:
            best_c5 = s

    return q, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)

    def remaining():
        return budget_s - (time.time() - t_start)

    # Load and prepare parent
    h_parent = np.array(initial_h_values, dtype=float)
    n_parent = len(h_parent)
    if n_parent % 2 != 0:
        n_parent -= 1
        h_parent = h_parent[:n_parent]

    m_parent = n_parent // 2
    target_sum = n_parent / 2.0
    h_parent = project_feasible(h_parent, target_sum)

    best_h = h_parent.copy()
    best_c5, _ = exact_score(h_parent)
    q_best = q_from_h(h_parent, n_parent)

    # Phase 1: Main optimization on parent resolution
    t1_limit = min(300, budget_s * 0.35)
    if t1_limit > 60:
        q_opt, c5_opt = optimize_q_pipeline(q_best, n_parent, t1_limit, rng)
        h_opt = build_h_from_q(q_opt, n_parent)
        s_opt, _ = exact_score(h_opt)
        if s_opt < best_c5 - 1e-15:
            best_c5 = s_opt
            best_h = h_opt
            q_best = q_opt.copy()

    # Phase 2: Stochastic restarts - perturb and re-optimize
    for restart in range(3):
        if remaining() < 120:
            break
        # Perturb q with random noise
        noise_scale = [0.02, 0.04, 0.08][min(restart, 2)]
        q_pert = q_best.copy()
        q_pert += noise_scale * rng.randn(m_parent)
        q_pert = np.clip(q_pert, 0.0, 1.0)

        t2_limit = min(80, remaining() * 0.4)
        if t2_limit < 30:
            continue
        q_try, c5_try = optimize_q_pipeline(q_pert, n_parent, t2_limit, rng)
        h_try = build_h_from_q(q_try, n_parent)
        s_try, _ = exact_score(h_try)
        if s_try < best_c5 - 1e-15:
            best_c5 = s_try
            best_h = h_try
            q_best = q_try.copy()

    # Phase 3: Multi-resolution
    resolutions = [96, 112, 128, 144, 160, 80]
    for n_try in resolutions:
        if remaining() < 100:
            break
        if n_try == n_parent:
            continue
        if n_try % 2 != 0:
            continue

        dx_try = 2.0 / n_try
        ts_try = n_try / 2.0

        # Interpolate from best_h
        x_orig = np.linspace(0.0, 2.0, len(best_h), endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_orig, best_h)
        # Handle periodic wrap for interpolation
        h_interp = project_feasible(h_interp, ts_try)

        m_try = n_try // 2
        q_try = q_from_h(h_interp, n_try)

        t3_limit = min(100, remaining() * 0.45)
        if t3_limit < 30:
            continue

        q_opt, c5_opt = optimize_q_pipeline(q_try, n_try, t3_limit, rng)
        h_opt = build_h_from_q(q_opt, n_try)
        h_opt = project_feasible(h_opt, ts_try)
        s_opt, _ = exact_score(h_opt)

        if s_opt < best_c5 - 1e-12:
            best_c5 = s_opt
            best_h = h_opt

    # Phase 4: Full-space active set on best
    if remaining() > 60:
        n_cur = len(best_h)
        h_full, c5_full = full_active_search(best_h, n_cur, max_rounds=6, max_active=25)
        s_full, _ = exact_score(h_full)
        if s_full < best_c5 - 1e-15:
            best_c5 = s_full
            best_h = h_full

    # Phase 5: Final coordinate descent polish
    if remaining() > 30 and len(best_h) % 2 == 0:
        q_fin = q_from_h(best_h, len(best_h))
        q_fin, _ = coordinate_descent_q(q_fin, len(best_h), n_sweeps=5)
        h_fin = build_h_from_q(q_fin, len(best_h))
        s_fin, _ = exact_score(h_fin)
        if s_fin < best_c5 - 1e-15:
            best_c5 = s_fin
            best_h = h_fin

    # Final validation
    n_out = len(best_h)
    ts_out = n_out / 2.0
    best_h = project_feasible(best_h, ts_out)
    final_score, _ = exact_score(best_h)
    best_c5 = min(best_c5, final_score)

    return best_h, best_c5, n_out
