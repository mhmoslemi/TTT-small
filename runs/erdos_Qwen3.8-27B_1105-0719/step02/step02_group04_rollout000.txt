def corr_grad_q(h, q, d, n, m):
    """Gradient of corr_full(h, 1-h)[j] w.r.t. q, where lag d = j-(n-1)."""
    g = np.zeros(n)

        g[:n-d] += -h[d:n]
    else:

        g[:n-e] += (1.0 - h[e:n])
        g[e:n] += -h[:n-e]
    # Map to q-space

def active_set_step(q, n, target_margin=0.001, pad=2, target_offset=0.0005):
    """One active-set least-norm step in q-space."""
    m = n // 2
    h = build_sym(q, n)
    c = corr_array(h)  # length 2n-1, each value is dx * raw_corr
    C = c.max()
    
    # Select active lags
    thresh = C - target_margin
    active = np.where(c >= thresh)[0]
    # Add neighbors
    expanded = set(active.tolist())
    for a in active:
        for delta in range(-pad, pad+1):
            k = a + delta
            if 0 <= k <= 2*n-2:
                expanded.add(k)
    active = np.array(sorted(expanded))
    
    k_active = len(active)
    if k_active == 0 or k_active >= m:
        return q, C, None
    
    # Build Jacobian P (k_active x m)
    P = np.zeros((k_active, m))
    for idx, j in enumerate(active):
        d = j - (n - 1)
        P[idx] = corr_grad_q(h, q, d, n, m)
    
    # Target: reduce all active values to C - target_offset
    target = C - target_offset
    r = c[active] - target  # positive: how much we need to reduce
    
    # Least-norm solution: delta = P^T (P P^T + lambda I)^{-1} (-r)
    # We want P delta = -r (to reduce values)
    lam = 0.01
    PPT = P @ P.T + lam * np.eye(k_active)
    try:
        delta = P.T @ np.linalg.solve(PPT, -r)
    except:
        delta = np.linalg.lstsq(P, -r, rcond=None)[0]
    
    # Line search
    best_q = q
    best_C = C
    alphas = [1.0, 0.7, 0.5, 0.35, 0.2, 0.1, 0.05, 0.02]
    for a in alphas:
        q_new = np.clip(q + a * delta, 0.0, 1.0)
        h_new = build_sym(q_new, n)
        C_new = eval_c5(h_new)
        if C_new < best_C:
            best_C = C_new
            best_q = q_new.copy()
    
    return best_q, best_C, C

    h = build_sym(q, n)
    c = corr_array(h)
    C = c.max()
    active = select_active(c, margin, pad)
    k_a = len(active)
    if k_a == 0 or k_a >= m:
        return q, C, False
    P = np.zeros((k_a, m))
    for idx, j in enumerate(active):
        d = int(j) - (n - 1)
        P[idx] = corr_grad_q(h, d, n, m)
    target = C - target_frac
    r = c[active] - target
    EP = np.eye(k_a)
    PPT = P @ P.T + reg * EP
    try:
        delta = P.T @ np.linalg.solve(PPT, -r)
    except np.linalg.LinAlgError:
        delta = np.linalg.lstsq(P, -r, rcond=None)[0]
    if not np.all(np.isfinite(delta)):
        return q, C, False
    best_q = q
    best_C = C
    improved = False
    for a in alphas:
        q_new = np.clip(q + a * delta, 0.0, 1.0)
        h_new = build_sym(q_new, n)
        C_new = eval_c5(h_new)
        if C_new < best_C - 1e-14:
            best_C = C_new
            best_q = q_new.copy()
            improved = True
    return best_q, best_C, improved

def gradient_descent_step(q, n, lr, active_margin, active_pad):
    m = n // 2
    h = build_sym(q, n)
    c = corr_array(h)
    C = c.max()
    active = select_active(c, active_margin, active_pad)
    k_a = len(active)
    if k_a == 0 or k_a >= m:
        return q, C
    P = np.zeros((k_a, m))
    for idx, j in enumerate(active):
        d = int(j) - (n - 1)
        P[idx] = corr_grad_q(h, d, n, m)
    r = c[active] - C
    target = C - 0.001
    r = c[active] - target
    lam = 0.05
    PPT = P @ P.T + lam * np.eye(k_a)
    try:
        delta = P.T @ np.linalg.solve(PPT, -r)
    except:
        delta = np.zeros(m)
    step = min(1.0, lr / (np.linalg.norm(delta) + 1e-12))
    q_new = np.clip(q + step * delta, 0.0, 1.0)
    h_new = build_sym(q_new, n)
    C_new = eval_c5(h_new)
    if C_new < C:
        return q_new, C_new
    return q, C

def project_to_target(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0.0, 1.0).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    out = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - out.sum()
    if abs(diff) > 1e-12:
        mask = (out > 1e-10) & (out < 1.0 - 1e-10)
        if mask.any():
            out[mask] += diff / mask.sum()
    return np.clip(out, 0.0, 1.0)

def interp_to(h, n_new):
    n_old = len(h)
    if n_new == n_old:
        return h.copy()
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h)
    return project_to_target(h_new, n_new / 2.0)

def full_grad_c(h, j, n):
    d = int(j) - (n - 1)
    return corr_grad_h(h, d, n)

def full_active_set_step(h, n, margin, pad, target_frac, alphas, reg):
    c = corr_array(h)
    C = c.max()
    active = select_active(c, margin, pad)
    k_a = len(active)
    if k_a == 0 or k_a >= n:
        return h, C, False
    P = np.zeros((k_a, n))
    for idx, j in enumerate(active):
        P[idx] = full_grad_c(h, j, n)
    P -= P.mean(axis=1, keepdims=True)
    target = C - target_frac
    r = c[active] - target
    mean_r = r.mean()
    r_centered = r - mean_r
    EP = np.eye(k_a)
    PPT = P @ P.T + reg * EP
    try:
        delta = P.T @ np.linalg.solve(PPT, -r_centered)
    except:
        delta = np.linalg.lstsq(P, -r_centered, rcond=None)[0]
    if not np.all(np.isfinite(delta)):
        return h, C, False
    delta -= delta.mean()
    best_h = h
    best_C = C
    improved = False
    for a in alphas:
        h_new = np.clip(h + a * delta, 0.0, 1.0)
        h_new = project_to_target(h_new, n / 2.0)
        C_new = eval_c5(h_new)
        if C_new < best_C - 1e-14:
            best_C = C_new
            best_h = h_new.copy()
            improved = True
    return best_h, best_C, improved

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')

    try:
        h_init = np.array(initial_h_values, dtype=float).reshape(-1)
        if h_init.size > 0 and np.all(np.isfinite(h_init)):
            n_init = len(h_init)
            h_init = project_to_target(h_init, n_init / 2.0)
            c5_init = eval_c5(h_init)
            best_h = h_init.copy()
            best_c5 = c5_init
    except Exception:
        pass

    if best_h is None:
        n_def = 126
        m_def = n_def // 2
        tt = np.linspace(0.0, 1.0, m_def)
        q_def = np.clip(0.5 + 0.2 * np.sin(2 * np.pi * 2 * tt) + 0.1 * np.sin(4 * np.pi * tt), 0.0, 1.0)
        best_h = build_sym(q_def, n_def)
        best_c5 = eval_c5(best_h)

    def elapsed():
        return time.time() - t0

    def remaining():
        return budget_s - elapsed()

    reserve = 40.0

    # Phase 1: Symmetric active-set optimization
    n_cur = len(best_h)
    if n_cur % 2 != 0:
        n_cur -= 1
        q = to_sym_q(best_h[:n_cur] + best_h[n_cur:][::-1] * 0, n_cur) if False else None
        q = to_sym_q(best_h, len(best_h))[:n_cur // 2]
        best_h = build_sym(q, n_cur)
        best_c5 = eval_c5(best_h)

    if n_cur % 2 == 0:
        q = to_sym_q(best_h, n_cur)
        h_sym = build_sym(q, n_cur)
        c5_sym = eval_c5(h_sym)
        if c5_sym < best_c5:
            best_h = h_sym
            best_c5 = c5_sym
        q = to_sym_q(best_h, n_cur)

        alphas = [1.0, 0.6, 0.4, 0.25, 0.15, 0.08, 0.04, 0.02]
        # Sweep over margins and target fractions
        for margin_val in [0.003, 0.001, 0.0005, 0.0002]:
            for tfrac in [0.003, 0.0015, 0.0008, 0.0004, 0.0002]:
                if remaining() < reserve + 20:
                    break
                for pad_val in [1, 2, 3]:
                    if remaining() < reserve + 10:
                        break
                    for reg_val in [0.001, 0.01, 0.1]:
                        if remaining() < reserve + 5:
                            break
                        for _it in range(15):
                            if remaining() < reserve + 3:
                                break
                            q, c5_val, imp = active_set_step(q, n_cur, margin_val, pad_val, tfrac, alphas, reg_val)
                            if imp:
                                h_new = build_sym(q, n_cur)
                                c5_new = eval_c5(h_new)
                                if c5_new < best_c5:
                                    best_h = h_new
                                    best_c5 = c5_new
                                if c5_new < 0.38070:
                                    break
                            else:
                                break
                        if c5_new < 0.38070:
                            break
                    if best_c5 < 0.38070:
                        break
                if best_c5 < 0.38070:
                    break
            if best_c5 < 0.38070:
                break
        # Final refinement with tight parameters
        for _rnd in range(5):
            if remaining() < reserve + 10:
                break
            q, c5_val, imp = active_set_step(q, n_cur, 0.0001, 2, 0.0001, alphas, 0.001)
            if imp:
                h_new = build_sym(q, n_cur)
                if eval_c5(h_new) < best_c5:
                    best_h = h_new
                    best_c5 = eval_c5(h_new)

    # Phase 2: Full-variable optimization
    if remaining() > reserve + 40:
        n_f = len(best_h)
        h_f = best_h.copy()
        c5_f = best_c5
        alphas_f = [0.5, 0.3, 0.2, 0.1, 0.05, 0.02]
        for margin_val in [0.002, 0.001, 0.0005]:
            if remaining() < reserve + 20:
                break
            for tfrac in [0.002, 0.001, 0.0005]:
                if remaining() < reserve + 10:
                    break
                for _it in range(20):
                    if remaining() < reserve + 5:
                        break
                    h_f, c5_f, imp = full_active_set_step(h_f, n_f, margin_val, 2, tfrac, alphas_f, 0.01)
                    if imp:
                        if c5_f < best_c5:
                            best_h = h_f.copy()
                            best_c5 = c5_f
                        if best_c5 < 0.38070:
                            break
                    else:
                        break
                if best_c5 < 0.38070:
                    break
            if best_c5 < 0.38070:
                break

    # Phase 3: Multi-resolution
    if remaining() > reserve + 30:
        n_base = len(best_h)
        n_list = []
        for delta in [-12, -8, -6, -4, -2, 2, 4, 6, 8, 12, 16]:
            nc = n_base + delta
            if nc >= 40 and nc % 2 == 0:
                n_list.append(nc)
        n_list = sorted(set(n_list))
        for n_try in n_list:
            if remaining() < reserve + 15:
                break
            h_try = interp_to(best_h, n_try)
            c5_try = eval_c5(h_try)
            if c5_try < best_c5:
                best_h = h_try
                best_c5 = c5_try
            if n_try % 2 == 0:
                q_try = to_sym_q(h_try, n_try)
                for margin_val in [0.002, 0.0005]:
                    if remaining() < reserve + 8:
                        break
                    for tfrac in [0.001, 0.0005]:
                        if remaining() < reserve + 5:
                            break
                        for _it in range(10):
                            if remaining() < reserve + 3:
                                break
                            q_try, c5_try, imp = active_set_step(q_try, n_try, margin_val, 2, tfrac, alphas, 0.01)
                            if imp:
                                h_try = build_sym(q_try, n_try)
                                if eval_c5(h_try) < best_c5:
                                    best_h = h_try
                                    best_c5 = eval_c5(h_try)
                            else:
                                break

    # Phase 4: Random perturbations around best
    if remaining() > reserve + 20:
        n_r = len(best_h)
        if n_r % 2 != 0:
            n_r -= 1
        for trial in range(15):
            if remaining() < reserve + 5:
                break
            noise_scale = 0.05 * (1.0 - trial / 20.0)
            if n_r % 2 == 0:
                q_r = to_sym_q(best_h, len(best_h))[:n_r // 2]
                q_r = np.clip(q_r + rng.randn(n_r // 2) * noise_scale, 0.0, 1.0)
                h_r = build_sym(q_r, n_r)
                c5_r = eval_c5(h_r)
                if c5_r < best_c5:
                    best_h = h_r
                    best_c5 = c5_r
                    q_r = to_sym_q(h_r, n_r)
                    for _it in range(5):
                        q_r, c5_r, _ = active_set_step(q_r, n_r, 0.001, 2, 0.001, alphas, 0.01)
                        h_r = build_sym(q_r, n_r)
                        if eval_c5(h_r) < best_c5:
                            best_h = h_r
                            best_c5 = eval_c5(h_r)

    # Phase 5: Final polish on best
    if remaining() > 10 and best_h is not None:
        n_fin = len(best_h)
        if n_fin % 2 == 0:
            q_fin = to_sym_q(best_h, n_fin)
            for margin_val in [0.0003, 0.0001, 0.00005]:
                if remaining() < 8:
                    break
                for _it in range(8):
                    if remaining() < 5:
                        break
                    q_fin, _, imp = active_set_step(q_fin, n_fin, margin_val, 3, 0.0002, alphas, 0.001)
                    if imp:
                        h_fin = build_sym(q_fin, n_fin)
                        if eval_c5(h_fin) < best_c5:
                            best_h = h_fin
                            best_c5 = eval_c5(h_fin)
        if remaining() > 8:
            h_fin, c5_fin, _ = full_active_set_step(best_h, len(best_h), 0.0003, 2, 0.0003, [0.3, 0.15, 0.05], 0.01)
            if c5_fin < best_c5:
                best_h = h_fin
                best_c5 = c5_fin

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_target(best_h, len(best_h) / 2.0)
    best_c5 = eval_c5(best_h)

    return best_h, best_c5, len(best_h)
