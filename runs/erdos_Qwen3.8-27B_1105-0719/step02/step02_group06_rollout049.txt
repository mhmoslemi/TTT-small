def eval_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def corr_raw(h):
    return np.correlate(h, 1.0 - h, mode='full')

def build_sym_h(q):
    m = len(q)
    h = np.empty(2 * m)
    h[:m] = q
    h[m:] = 1.0 - q[::-1]
    return h

def sym_from_h(h):
    n = len(h)
    m = n // 2
    q = 0.5 * (h[:m] + (1.0 - h[m:][::-1]))
    return np.clip(q, 0.0, 1.0)

def project_feasible(x, target_sum):
    n = len(x)
    x = np.asarray(x, dtype=float).copy()
    x[np.isnan(x)] = 0.5
    x = np.clip(x, 0.0, 1.0)
    if abs(x.sum() - target_sum) < 1e-12:
        return x
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(x + mid, 0.0, 1.0).sum()
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    h = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        interior = (h > 1e-10) & (h < 1.0 - 1e-10)
        if interior.sum() > 0:
            h[interior] += diff / interior.sum()
    return np.clip(h, 0.0, 1.0)

def smooth_max_eval_grad(q, n_full, beta):
    m = n_full // 2
    h = build_sym_h(q)
    dx = 2.0 / n_full
    corrs = corr_raw(h) * dx
    max_c = corrs.max()
    shifted = corrs - max_c
    ex = np.exp(beta * shifted)
    w_sum = ex.sum()
    if w_sum < 1e-30:
        ex = np.ones(len(corrs)) / len(corrs)
        w_sum = 1.0
    else:
        ex = ex / w_sum
    val = max_c + np.log(w_sum) / beta
    # Weighted gradient
    grad_h = np.zeros(n_full)

        w = ex[ell]
        if w < 1e-14:

        d = ell - (n_full - 1)

    grad_q = grad_h[:m] - grad_h[n_full-1:n_full-1-m:-1]
    grad_q *= dx
    return val, grad_q

def smooth_sym_optimize(q0, n_full, beta_list, maxiter_per_beta, time_limit):
    m = n_full // 2
    q = np.clip(q0.copy(), 0, 1)
    h = build_sym_h(q)
    best_score = eval_score(h)
    best_q = q.copy()
    
    for beta in beta_list:

            break
        bounds = [(0.0, 1.0)] * m
        def obj(xx):
            return smooth_max_eval_grad(xx, n_full, beta)[0]
        def jac(xx):

def coordinate_line_search(h, n_rounds=50, time_limit=None):
    n = len(h)
    dx = 2.0 / n
    best_score = eval_score(h)
    best_h = h.copy()
    
    t0 = time.time() if time_limit is None else time_limit
    
    for rnd in range(n_rounds):
        if time_limit is not None and time.time() > time_limit:
            break
        improved = False
        corrs = corr_raw(best_h) * dx
        max_c = corrs.max()
        
        # Try each coordinate
        for i in range(n):
            if time_limit is not None and time.time() > time_limit:
                break
            # Try shifting h[i] by various amounts
            # To preserve sum, distribute the change over all other coordinates
            for delta in [-0.3, -0.2, -0.1, -0.05, 0.05, 0.1, 0.2, 0.3]:
                h_trial = best_h.copy()
                h_trial[i] += delta
                if h_trial[i] < -0.01 or h_trial[i] > 1.01:

                    break

            break

def paired_transfer_hill_climb(h, time_limit, max_rounds=50):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    best_h = h.copy()
    best_score = eval_score(best_h)
    
    for rnd in range(max_rounds):
        if time.time() > time_limit:
            break
        improved = False
        corrs = corr_raw(best_h) * dx
        max_c = corrs.max()
        active = np.where(corrs >= max_c - 0.015)[0]
        
        # Compute subgradient
        grad = np.zeros(n)
        for ell in active:
            d = ell - (n - 1)
            i_lo = max(0, -d)
            i_hi = min(n - 1, n - 1 - d)
            if i_lo > i_hi:
                continue
            j_lo = i_lo + d
            j_hi = i_hi + d
            grad[i_lo:i_hi+1] += (1.0 - best_h[j_lo:j_hi+1])
            grad[j_lo:j_hi+1] += (-best_h[i_lo:i_hi+1])
        grad *= dx / len(active)
        
        # For each coordinate, try paired transfer
        for i in range(n):
            if time.time() > time_limit:
                break
            # Direction: decrease grad at i
            if grad[i] > 0:
                # Need to decrease h[i], so increase some h[j]
                # Best j: most negative grad
                j = np.argmin(grad)
                if j == i:
                    j = np.argmin(np.where(np.arange(n) == i, np.inf, grad))
                sign_i = -1
            elif grad[i] < 0:
                # Need to increase h[i], so decrease some h[j]
                j = np.argmax(grad)
                if j == i:
                    j = np.argmax(np.where(np.arange(n) == i, -np.inf, grad))
                sign_i = 1
            else:
                continue
            
            for mag in [0.05, 0.02, 0.01, 0.005]:
                h_trial = best_h.copy()
                h_trial[i] += sign_i * mag
                h_trial[j] -= sign_i * mag
                if h_trial[i] < 0 or h_trial[i] > 1 or h_trial[j] < 0 or h_trial[j] > 1:
                    continue
                sc = eval_score(h_trial)
                if sc < best_score - 1e-14:
                    best_score = sc
                    best_h = h_trial
                    improved = True
                    break
            if improved:
                # Recompute gradient
                break
        
        if not improved:
            break
    
    return best_h, best_score

import numpy as np

import time

def eval_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def corr_raw(h):
    return np.correlate(h, 1.0 - h, mode='full')

def build_sym_h(q):
    m = len(q)
    h = np.empty(2 * m)
    h[:m] = q
    h[m:] = 1.0 - q[::-1]
    return h

def sym_from_h(h):
    n = len(h)
    m = n // 2
    q = 0.5 * (h[:m] + (1.0 - h[m:][::-1]))
    return np.clip(q, 0.0, 1.0)

def project_feasible(x, target_sum):
    n = len(x)
    x = np.asarray(x, dtype=float).copy()
    x = np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0)
    x = np.clip(x, 0.0, 1.0)
    if abs(x.sum() - target_sum) < 1e-12:
        return x
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(x + mid, 0.0, 1.0).sum()
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    h = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        interior = (h > 1e-10) & (h < 1.0 - 1e-10)
        if interior.sum() > 0:
            h[interior] += diff / interior.sum()
    return np.clip(h, 0.0, 1.0)

def smooth_max_eval_grad(q, n_full, beta):
    m = n_full // 2
    h = build_sym_h(q)
    dx = 2.0 / n_full
    corrs = corr_raw(h) * dx
    max_c = corrs.max()
    shifted = corrs - max_c
    ex = np.exp(beta * shifted)
    w_sum = ex.sum()
    if w_sum < 1e-30:
        ex = np.ones(len(corrs)) / len(corrs)
        w_sum = 1.0
    else:
        ex = ex / w_sum
    val = max_c + np.log(w_sum) / beta
    grad_h = np.zeros(n_full)
    for ell in range(2 * n_full - 1):
        w = ex[ell]

    grad_q *= dx
