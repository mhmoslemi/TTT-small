```python
def eval_score(h):
    n = len(h)
    dx = 2.0 / n

def corr_raw(h):
    return np.correlate(h, 1.0 - h, mode='full')

def corr_grad_all(h):
    """Compute gradients of all correlation lags.
    Returns array of shape (2n-1, n)."""
    n = len(h)
    L = 2 * n - 1
    grads = np.zeros((L, n))
    for ell in range(L):
        d = ell - (n - 1)
        i_lo = max(0, -d)
        i_hi = min(n - 1, n - 1 - d)
        if i_lo > i_hi:
            continue
        idx = np.arange(i_lo, i_hi + 1)
        grads[ell, idx] += (1.0 - h[idx + d])
        grads[ell, idx + d] += (-h[idx])
    return grads

def tangent_lp_step(h0, active_lags, trust_region=0.3):
    """Solve tangent LP to find improvement direction.
    Returns (h_new, success)."""
    n = len(h0)
    R = corr_raw(h0)
    L = len(R)

    target = n / 2.0
    
    # Build constraint matrix for active lags
    # For each active lag ell:
    #   sum_i grad[ell,i] * h_new[i] - t <= sum_i grad[ell,i]*h0[i] - R[ell]
    # In standard form for linprog (A_ub @ x <= b_ub):
    #   [grad[ell,:], -1] @ [h_new; t] <= sum_i grad[ell,i]*h0[i] - R[ell]
    
    A_rows = []
    b_vals = []
    for ell in active_lags:
        d = ell - (n - 1)
        i_lo = max(0, -d)
        i_hi = min(n - 1, n - 1 - d)
        if i_lo > i_hi:
            continue
        idx = np.arange(i_lo, i_hi + 1)
        g = np.zeros(n)
        g[idx] += (1.0 - h0[idx + d])
        g[idx + d] += (-h0[idx])
        # g^T h_new - t <= g^T h0 - R[ell]
        row = np.zeros(n + 1)
        row[:n] = g
        row[n] = -1.0
        A_rows.append(row)
        b_vals.append(float(g @ h0) - R[ell])
    
    if not A_rows:
        return h0.copy(), False
    
    A_ub = np.array(A_rows)  # shape (len(active), n+1)
    b_ub = np.array(b_vals)
    
    # Equality: sum(h_new) = n/2
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    
    # Objective: minimize t
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    # Bounds: 0 <= h_new[i] <= 1, t in [0, 2]
    # Also trust region: h0[i] - tr <= h_new[i] <= h0[i] + tr
    bounds = []
    for i in range(n):
        lo_i = max(0.0, h0[i] - trust_region)
        hi_i = min(1.0, h0[i] + trust_region)
        bounds.append((lo_i, hi_i))
    bounds.append((0.0, 2.0))  # t

                      bounds=bounds, method='highs')

    if res.status != 0:
        return h0.copy(), False
    
    h_new = res.x[:n]
    h_new = project_feasible(h_new, target)
    return h_new, True

    """Select active lags based on proximity to maximum."""
    L = len(corrs)
    max_val = corrs.max()

    active = np.where(corrs >= threshold)[0].tolist()
    # Pad with neighbors

    if len(active) > max_k:
        scored = sorted([(corrs[k], k) for k in active], reverse=True)

    return active
```

```python
import numpy as np
from scipy.optimize import linprog
import time


def eval_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)


def corr_raw(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(x, target_sum):
    n = len(x)
    x = np.array(x, dtype=float).copy()
    x = np.where(np.isfinite(x), x, 0.5)
    x = np.clip(x, 0.0, 1.0)
    if abs(x.sum() - target_sum) < 1e-13:
        return x
    lo, hi = -1.5, 1.5
    for _ in range(90):
        mid = (lo + hi) / 2.0
        s = np.clip(x + mid, 0.0, 1.0).sum()
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    h = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        interior = (h > 1e-10) & (h < 1.0 - 1e-10)
        if interior.sum() > 0:
            h[interior] += diff / interior.sum()
    return np.clip(h, 0.0, 1.0)


def grad_for_lag(h, ell):
    n = len(h)
    d = ell - (n - 1)
    g = np.zeros(n)
    i_lo = max(0, -d)
    i_hi = min(n - 1, n - 1 - d)
    if i_lo > i_hi:
        return g
    idx = np.arange(i_lo, i_hi + 1)
    g[idx] += (1.0 - h[idx + d])
    g[idx + d] += (-h[idx])
    return g


def select_active(corrs, margin_frac, margin_abs, max_k):
    L = len(corrs)
    max_val = corrs.max()
    threshold = max_val - max(margin_abs, margin_frac * max_val)
    active = set(np.where(corrs >= threshold)[0].tolist())
    padded = set()
    for k in active:
        for dd in range(-3, 4):
            if 0 <= k + dd < L:
                padded.add(k + dd)
    active = sorted(padded)
    if len(active) > max_k:
        scored = sorted([(corrs[k], k) for k in active], reverse=True)
        active = sorted([k for _, k in scored[:max_k]])
    return active


def tangent_lp(h0, active_lags, trust_region):
    n = len(h0)
    R = corr_raw(h0)
    target = n / 2.0
    A_rows = []
    b_vals = []
    for ell in active_lags:
        d = ell - (n - 1)
        i_lo = max(0, -d)
        i_hi = min(n - 1, n - 1 - d)
        if i_lo > i_hi:
            continue
        idx = np.arange(i_lo, i_hi + 1)
        g = np.zeros(n)
        g[idx] += (1.0 - h0[idx + d])
        g[idx + d] += (-h0[idx])
        row = np.zeros(n + 1)
        row[:n] = g
        row[n] = -1.0
        A_rows.append(row)
        b_vals.append(float(g @ h0) - float(R[ell]))
    if not A_rows:
        return h0.copy(), False
    A_ub = np.array(A_rows)
    b_ub = np.array(b_vals)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    c_vec = np.zeros(n + 1)
    c_vec[n] = 1.0
    bounds = []
    for i in range(n):
        lo_i = max(0.0, h0[i] - trust_region)
        hi_i = min(1.0, h0[i] + trust_region)
        bounds.append((lo_i, hi_i))
    bounds.append((0.0, 2.0))
    try:
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
    except Exception:
        return h0.copy(), False
    if res.status != 0:
        return h0.copy(), False
    h_new = project_feasible(res.x[:n], target)
    return h_new, True


def line_search(h0, h1, n_evals):
    n = len(h0)
    d = h1 - h0
    best_alpha = 0.0
    best_score = eval_score(h0)
    for i in range(1, n_evals + 1):
        alpha = float(i) / float(n_evals)
        h_try = h0 + alpha * d
        if h_try.min() < -1e-8 or h_try.max() > 1.0 + 1e-8:
            continue
        score = eval_score(h_try)
        if score < best_score - 1e-14:
            best_score = score
            best_alpha = alpha
    h_best = h0 + best_alpha * d
    return h_best, best_score


def pair_transfer(h, n_trials, rng):
    n = len(h)
    best_h = h.copy()
    best_score = float(eval_score(h))
    h_work = h.copy()
    for _ in range(n_trials):
        a = int(rng.randint(n))
        b = int(rng.randint(n))
        if a == b:
            continue
        h_w = h_work.copy()
        lo = max(-h_w[a], h_w[b] - 1.0)
        hi = min(1.0 - h_w[a], -h_w[b])
        if hi <= lo + 1e-10:
            continue
        best_local = best_score
        best_delta = 0.0
        for i in range(1, 25):
            delta = lo + (hi - lo) * float(i) / 25.0
            h_w[a] = h_work[a] + delta
            h_w[b] = h_work[b] - delta
            score = eval_score(h_w)
            if score < best_local - 1e-14:
                best_local = score
                best_delta = delta
        if best_delta != 0.0:
            h_work[a] += best_delta
            h_work[b] -= best_delta
            if best_local < best_score:
                best_score = best_local
                best_h = h_work.copy()
    return best_h, best_score


def transfer_res(h_old, n_new):
    n_old = len(h_old)
    x_old = (np.arange(n_old) + 0.5) * 2.0 / n_old
    x_new = (np.arange(n_new) + 0.5) * 2.0 / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new / 2.0)


def make_fourier_init(n, rng):
    x = np.linspace(0.0, 1.0, n)
    h = np.full(n, 0.5)
    n_terms = min(6, max(2, n // 20))
    for _ in range(n_terms):
        freq = int(rng.randint(1, max(2, n // 6)))
        phase = float(rng.uniform(0.0, 2.0 * np.pi))
        amp = float(rng.uniform(-0.25, 0.25))
        h += amp * np.sin(2.0 * np.pi * freq * x + phase)
    h = np.clip(h, 0.0, 1.0)
    return project_feasible(h, n / 2.0)


def make_binary_smooth(n, rng):
    h = np.zeros(n)
    h[:n // 2] = 1.0
    h = rng.permutation(h)
    h = 0.5 * h + 0.5 * project_feasible(np.clip(h + rng.randn(n) * 0.3, 0, 1), n / 2.0)
    return project_feasible(h, n / 2.0)


def lp_optimize(h, max_rounds, time_limit, rng):
    n = len(h)
    current = h.copy()
    current_score = float(eval_score(current))
    best_local = current.copy()
    best_local_score = current_score
    stagnation = 0

    for rnd in range(max_rounds):
        if time.time() > time_limit:
            break
        R = corr_raw(current)
        active = select_active(R, margin_frac=0.02, margin_abs=1e-5, max_k=50)
        improved = False
        for tr in [0.5, 0.25, 0.12, 0.06]:
            h_lp, ok = tangent_lp(current, active, tr)
            if not ok:
                continue
            h_ls, sc_ls = line_search(current, h_lp, n_evals=20)
            if sc_ls < current_score - 1e-13:
                current = h_ls
                current_score = sc_ls
                improved = True
                break
        if not improved:
            current, current_score = pair_transfer(current, 80, rng)
            if current_score <= best_local_score:
                best_local = current.copy()
                best_local_score = current_score
            stagnation += 1
            if stagnation > 6:
                break
        else:
            stagnation = 0
            if current_score < best_local_score:
                best_local = current.copy()
                best_local_score = current_score
    return best_local, best_local_score
```
