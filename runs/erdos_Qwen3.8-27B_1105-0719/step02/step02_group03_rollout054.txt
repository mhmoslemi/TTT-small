```python
import numpy as np
from scipy.optimize import minimize
import time

def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
        else:
            h += diff / n
    return np.clip(h, 0.0, 1.0)

def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def c5_of_h(h, dx):
    return compute_corrs(h).max() * dx

def make_binary(h, target_sum):
    n = len(h)
    ts = int(round(target_sum))
    h_bin = np.zeros(n)
    idx = np.argsort(-h, kind='stable')
    h_bin[idx[:ts]] = 1.0
    return h_bin

def binary_steepest_vec(h, dx, max_sweeps=200, t0=None, time_limit=None):
    """Vectorized steepest descent for binary h."""
    n = len(h)
    ms = np.arange(-(n-1), n)
    corrs = compute_corrs(h)
    best_c5 = corrs.max() * dx
    best_h = h.copy()
    
    for sweep in range(max_sweeps):
        if t0 is not None:
            elapsed = time.time() - t0
            if elapsed > time_limit:
                break
        
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        n_ones = len(ones)
        n_zeros = len(zeros)
        
        if n_ones == 0 or n_zeros == 0:
            break
        
        # Precompute delta matrices
        D = np.zeros((n_ones, 2*n-1))
        for idx_i, i in enumerate(ones):
            v1 = (0 <= i - ms) & (i - ms <= n - 1)
            D[idx_i, v1] += -(1.0 - h[i - ms[v1]])
            v2 = (0 <= i + ms) & (i + ms <= n - 1)
            D[idx_i, v2] += h[i + ms[v2]]
        
        E = np.zeros((n_zeros, 2*n-1))
        for idx_j, j in enumerate(zeros):
            v1 = (0 <= j - ms) & (j - ms <= n - 1)
            E[idx_j, v1] += (1.0 - h[j - ms[v1]])
            v2 = (0 <= j + ms) & (j + ms <= n - 1)
            E[idx_j, v2] += -h[j + ms[v2]]
        
        # Find best swap: minimize max(corrs + D_i + E_j) over all (i,j)
        # Vectorized: (n_ones, n_zeros, 2n-1)
        combined = corrs[None, None, :] + D[:, None, :] + E[None, :, :]
        new_max = combined.max(axis=2)
        
        min_flat = new_max.argmin()
        min_i, min_j = np.unravel_index(min_flat, new_max.shape)
        min_val = new_max[min_i, min_j]
        
        if min_val >= corrs.max() - 1e-14:
            break  # No improvement
        
        i = ones[min_i]
        j = zeros[min_j]
        h[i] = 0.0
        h[j] = 1.0
        corrs = compute_corrs(h)
        new_c5 = corrs.max() * dx
        if new_c5 < best_c5 - 1e-15:
            best_c5 = new_c5
            best_h = h.copy()
    
    return best_h, best_c5
```

This is O(n²) for step 1 and O(n²) for step 2, so O(n²) total. Very fast.

But this doesn't guarantee finding the best 2-swap. It's a heuristic.

4. This targeted 2-swap search should be much more effective than random
