def corr_vector_from_q(q, n):
    h = build_h_from_q(q, n)
    return np.correlate(h, 1.0 - h, mode='full')


def max_corr_from_q(q, n):
    corrs = corr_vector_from_q(q, n)
    return corrs.max() * (2.0 / n), corrs


def optimize_single_coord(q, j, n):
    """Minimize max correlation by optimizing q[j] via grid + refinement."""
    m = n // 2
    dx = 2.0 / n
    old_val = q[j]

    # Coarsely precompute: for each t, corr[d] = A_d + B_d*t + (C_d*t^2 if special)
    # But simpler: just evaluate on grid
    best_t = old_val
    best_score = np.inf

    # 80-point grid
    grid = np.linspace(0.0, 1.0, 80)
    scores_grid = np.empty(80)
    for idx, t in enumerate(grid):
        q[j] = t
        h = build_h_from_q(q, n)
        s = np.correlate(h, 1.0 - h, mode='full').max() * dx
        scores_grid[idx] = s
        if s < best_score:
            best_score = s
            best_t = t

    # Fine refinement: golden section around the grid minimum
    best_idx = np.argmin(scores_grid)
    if best_idx > 0 and best_idx < 79:
        lo = grid[best_idx - 1] if scores_grid[best_idx - 1] < scores_grid[best_idx] else grid[best_idx]
        hi = grid[best_idx + 1] if scores_grid[best_idx + 1] < scores_grid[best_idx] else grid[best_idx]
        # Actually let's just do a fine grid in [grid[best_idx-1], grid[best_idx+1]]
        lo = max(0.0, grid[best_idx - 1])
        hi = min(1.0, grid[best_idx + 1])
        fine_grid = np.linspace(lo, hi, 25)
        for t in fine_grid:
            q[j] = t
            h = build_h_from_q(q, n)
            s = np.correlate(h, 1.0 - h, mode='full').max() * dx
            if s < best_score:
                best_score = s
                best_t = t

    # Even finer golden section
    if 0.0 < best_t < 1.0:
        # Narrow bracket
        step = (hi - lo) / 25.0 if lo < hi else 0.01
        lo2 = max(0.0, best_t - step)
        hi2 = min(1.0, best_t + step)
        for _ in range(20):
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            m1 = hi2 - gr * (hi2 - lo2)
            m2 = lo2 + gr * (hi2 - lo2)
            q[j] = m1
            h = build_h_from_q(q, n)
            s1 = np.correlate(h, 1.0 - h, mode='full').max() * dx
            q[j] = m2
            h = build_h_from_q(q, n)
            s2 = np.correlate(h, 1.0 - h, mode='full').max() * dx
            if s1 < s2:
                hi2 = m2
            else:
                lo2 = m1
            if abs(hi2 - lo2) < 1e-14:
                break
        best_t = (lo2 + hi2) / 2.0
        q[j] = best_t
        h = build_h_from_q(q, n)
        s = np.correlate(h, 1.0 - h, mode='full').max() * dx
        if s < best_score:
            best_score = s

    q[j] = best_t
    return best_score


def coordinate_descent_q(q, n, max_sweeps=100, time_start=None, time_limit=None):
    """Full coordinate descent in q-space."""
    m = n // 2
    best_score, _ = max_corr_from_q(q, n)
    initial_score = best_score

    for sweep in range(max_sweeps):
        if time_limit is not None:
            if time.time() - time_start > time_limit:
                break
        for j in range(m):
            optimize_single_coord(q, j, n)
        new_score, _ = max_corr_from_q(q, n)
        if new_score < best_score:
            best_score = new_score
        if initial_score - best_score < 1e-10 and sweep > 5:
            # Check if we've truly converged
            pass

    return best_score


def q_smooth_value_grad(q, n, beta):
    """Log-sum-exp smooth approximation of max correlation in q-space."""
    m = n // 2
    h = build_h_from_q(q, n)
    dx = 2.0 / n
    corrs = np.correlate(h, 1.0 - h, mode='full')
    val = logsumexp(beta * corrs) / beta * dx
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(len(corrs)) / len(corrs)
    else:
        w /= w_sum

    # Gradient w.r.t. q: for each correlation k with lag d=k-(n-1),
    # d corr_k / dq_j = dx * (dcorr_k/dh_j - dcorr_k/dh_{2m-1-j})
    Gq = np.zeros(m)
    for k in range(len(corrs)):
        if w[k] < 1e-14:
            continue
        d = k - (n - 1)
        # d corr[d] / dh[p] for each p
        # corr[d] = sum_{i in S(d)} h[i]*(1-h[i+d])
        # d/dd[h[i]*(1-h[i+d])] at h:
        #   wrt h[i]: (1-h[i+d])
        #   wrt h[i+d]: -h[i]
        # So dcorr[d]/dh[p] = (1-h[p+d])*[p in S(d)] + (-h[p-d])*[p-d in S(d)]
        # where p in S(d) means 0<=p<n and 0<=p+d<n
        g = np.zeros(n)
        if d >= 0:
            # Valid i: 0 <= i <= n-1-d
            i_range = np.arange(0, n - d)
            # First factor: h[i], i in [0, n-1-d]
            g[i_range] += (1.0 - h[i_range + d])
            # Second factor: h[i+d], i+d in [d, n-1]
            g[d:] += (-h[:n - d])
        else:
            dd = -d
            # Valid i: dd <= i <= n-1
            i_range = np.arange(dd, n)
            # First factor: h[i]
            g[i_range] += (1.0 - h[i_range - dd])
            # Second factor: h[i-dd], i-dd in [0, n-1-dd]
            g[:n - dd] += (-h[dd:])
        # Now Gq[j] += w[k] * dx * (g[j] - g[2m-1-j])
        Gq += w[k] * (g[:m] - g[m:][::-1])
    return val, Gq * dx


def q_smooth_value(q, n, beta):
    return q_smooth_value_grad(q, n, beta)[0]


def q_smooth_grad(q, n, beta):
    return q_smooth_value_grad(q, n, beta)[1]


def epigraph_q_solve(q, n, active_k, t0, maxiter=200):
    """Solve min t s.t. t >= corr_k*dx for k in active_k, q in [0,1]^m."""
    m = n // 2
    dx = 2.0 / n

    def obj(x):
        return x[0]

    def obj_grad(x):
        g = np.zeros(m + 1)
        g[0] = 1.0
        return g

    cons = []
    for i_k, k_idx in enumerate(active_k):
        d = k_idx - (n - 1)

        def con_fun(x, d=d):
            t = x[0]
            qv = x[1:]
            h = build_h_from_q(qv, n)
            corrs = np.correlate(h, 1.0 - h, mode='full')
            return t - corrs[k_idx] * dx

        def con_grad(x, d=d, k_idx=k_idx):
            qv = x[1:]
            h = build_h_from_q(qv, n)
            g = np.zeros(m + 1)
            g[0] = 1.0
            # Gradient of corr[k_idx]*dx w.r.t. q
            if d >= 0:
                i_range = np.arange(0, n - d)
                gh = np.zeros(n)
                gh[i_range] += (1.0 - h[i_range + d])
                gh[d:] += (-h[:n - d])
            else:
                dd = -d
                i_range = np.arange(dd, n)
                gh = np.zeros(n)
                gh[i_range] += (1.0 - h[i_range - dd])
                gh[:n - dd] += (-h[dd:])
            g[1:] = -(gh[:m] - gh[m:][::-1]) * dx
            return g

        cons.append({'type': 'ineq', 'fun': con_fun, 'jac': con_grad})

    x0 = np.concatenate([[t0], q])
    bounds = [(0.0, 2.0)] + [(0.0, 1.0)] * m
    try:
        res = minimize(obj, x0, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        q_out = np.clip(res.x[1:], 0.0, 1.0)
        return q_out, res.x[0]
    except Exception:
        return q.copy(), t0


def subgradient_q(q, n, n_iter=5000, time_start=None, time_limit=None):
    """Subgradient method with averaging for max-corrieration minimization in q-space."""
    m = n // 2
    dx = 2.0 / n
    current = q.copy()
    avg = q.copy()
    best_q = q.copy()
    best_score, _ = max_corr_from_q(q, n)

    # Initial step size
    h = build_h_from_q(q, n)
    corrs = np.correlate(h, 1.0 - h, mode='full')
    max_corr = corrs.max()
    # Estimate Lipschitz constant L ~ n (rough)
    L = n * dx
    T = n_iter
    step_scale = 2.0 * max_corr * L / np.sqrt(T)

    for t in range(T):
        if time_limit is not None and (t % 200 == 0):
            if time.time() - time_start > time_limit:
                break
        h = build_h_from_q(current, n)
        corrs = np.correlate(h, 1.0 - h, mode='full')
        # Pick active lag (randomly from top few)
        top_n = min(5, len(corrs))
        top_idx = np.argsort(corrs)[-top_n:]
        k_pick = top_idx[np.random.randint(top_n)]
        d = k_pick - (n - 1)

        # Gradient of corr[k_pick] w.r.t. h
        gh = np.zeros(n)
        if d >= 0:
            i_range = np.arange(0, n - d)
            gh[i_range] += (1.0 - h[i_range + d])
            gh[d:] += (-h[:n - d])
        else:
            dd = -d
            i_range = np.arange(dd, n)
            gh[i_range] += (1.0 - h[i_range - dd])
            gh[:n - dd] += (-h[dd:])
        # Convert to q-gradient
        gq = (gh[:m] - gh[m:][::-1]) * dx

        # Step
        eta = step_scale / (t + 1)
        current = current - eta * gq
        current = np.clip(current, 0.0, 1.0)
        avg += current / (t + 1)

        if (t + 1) % 500 == 0:
            h_check = build_h_from_q(avg, n)
            s_check = np.correlate(h_check, 1.0 - h_check, mode='full').max() * dx
            if s_check < best_score:
                best_score = s_check
                best_q = avg.copy()
            h_check2 = build_h_from_q(current, n)
            s_check2 = np.correlate(h_check2, 1.0 - h_check2, mode='full').max() * dx
            if s_check2 < best_score:
                best_score = s_check2
                best_q = current.copy()

    return best_q, best_score


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)

    def remaining():
        return budget_s - (time.time() - t_start)

    # Initialize from parent
    h_parent = np.array(initial_h_values, dtype=float)
    n_parent = len(h_parent)
    target_sum = n_parent / 2.0
    h_parent = project_feasible(h_parent, target_sum)

    best_h = h_parent.copy()
    best_c5 = exact_score(h_parent)
    best_n = n_parent

    # Use symmetric reduction on parent (even n)
    n = n_parent
    m = n // 2
    q = q_from_h(h_parent, n)

    def update_best(h_new, c5_new=None):
        nonlocal best_h, best_c5, best_n
        if c5_new is None:
            c5_new = exact_score(h_new)
        if c5_new < best_c5 - 1e-12:
            best_h = h_new.copy()
            best_c5 = c5_new
            best_n = len(h_new)
        return c5_new

    # Phase 1: Homotopy warm-up in q-space
    if remaining() > 200:
        for beta in [30, 60, 120, 250, 500, 1000, 2000]:
            if remaining() < 40:
                break

            def hval(x):
                return q_smooth_value(x, n, beta)

            def hgrad(x):
                return q_smooth_grad(x, n, beta)

            try:
                res = minimize(hval, q, jac=hgrad, method='L-BFGS-B',
                               bounds=[(0.0, 1.0)] * m,
                               options={'maxiter': 300, 'ftol': 1e-16})
                q_res = np.clip(res.x, 0.0, 1.0)
                h_res = build_h_from_q(q_res, n)
                s_res = exact_score(h_res)
                if s_res < best_c5:
                    best_c5 = s_res
                    best_h = h_res.copy()
                    q = q_res.copy()
                elif s_res < 0.3808:
                    q = q_res.copy()
            except Exception:
                pass

    # Phase 2: Coordinate descent (main method)
    if remaining() > 150:
        cd_limit = min(remaining() - 50, 400.0)
        q_cd = q.copy()
        cd_score = coordinate_descent_q(q_cd, n, max_sweeps=500,
                                        time_start=time.time(), time_limit=cd_limit)
        h_cd = build_h_from_q(q_cd, n)
        s_cd = exact_score(h_cd)
        if s_cd < best_c5:
            best_c5 = s_cd
            best_h = h_cd.copy()
            q = q_cd.copy()

    # Phase 3: Active-set epigraph
    if remaining() > 120:
        corrs = corr_vector_from_q(q, n)
        dx = 2.0 / n
        # Select top active lags
        L_active = min(20, 2 * n - 1)
        top_idx = np.argsort(corrs)[::-1][:L_active]
        active_k = sorted(top_idx.tolist())
        t0 = corrs.max() * dx * 1.001 + 1e-5

        for round_num in range(8):
            if remaining() < 30:
                break
            q_new, t_sol = epigraph_q_solve(q, n, active_k, t0, maxiter=150)
            h_new = build_h_from_q(q_new, n)
            corrs_new = np.correlate(h_new, 1.0 - h_new, mode='full')
            s_new = corrs_new.max() * dx

            if s_new < best_c5:
                best_c5 = s_new
                best_h = h_new.copy()
                q = q_new.copy()

            # Update active set
            hidden = corrs_new * dx > t_sol + 1e-7
            if hidden.sum() > 0:
                hidden_idx = np.where(hidden)[0]
                new_active = sorted(set(active_k) | set(hidden_idx.tolist()))
                if len(new_active) > 40:
                    vals = [corrs_new[k] for k in new_active]
                    order = np.argsort(vals)[::-1][:40]
                    new_active = sorted([new_active[i] for i in order])
                active_k = new_active
            t0 = max(t_sol, s_new) * 1.001 + 1e-5
            q = q_new

            if abs(s_new - best_c5) < 1e-12:
                break

        # Coordinate descent after epigraph
        if remaining() > 60:
            cd_score2 = coordinate_descent_q(q, n, max_sweeps=200,
                                             time_start=time.time(), time_limit=min(remaining() - 30, 120.0))
            h_cd2 = build_h_from_q(q, n)
            s_cd2 = exact_score(h_cd2)
            if s_cd2 < best_c5:
                best_c5 = s_cd2
                best_h = h_cd2.copy()

    # Phase 4: Try n=128
    if remaining() > 150:
        n2 = 128
        m2 = n2 // 2
        x_orig = np.linspace(0.0, 2.0, best_n)
        x_new = np.linspace(0.0, 2.0, n2)
        h_interp = np.interp(x_new, x_orig, best_h)
        h_interp = project_feasible(h_interp, n2 / 2.0)
        q2 = q_from_h(h_interp, n2)

        # Quick homotopy
        for beta in [80, 300, 800]:
            if remaining() < 40:
                break
            try:
                res2 = minimize(lambda x: q_smooth_value(x, n2, beta), q2,
                                jac=lambda x: q_smooth_grad(x, n2, beta),
                                method='L-BFGS-B', bounds=[(0.0, 1.0)] * m2,
                                options={'maxiter': 200, 'ftol': 1e-15})
                q2 = np.clip(res2.x, 0.0, 1.0)
            except Exception:
                pass

        # Coordinate descent
        if remaining() > 60:
            coordinate_descent_q(q2, n2, max_sweeps=300,
                                 time_start=time.time(), time_limit=min(remaining() - 30, 120.0))
            h2 = build_h_from_q(q2, n2)
            s2 = exact_score(h2)
            if s2 < best_c5:
                best_c5 = s2
                best_h = h2.copy()
                best_n = n2

    # Phase 5: Subgradient with averaging (for diversity)
    if remaining() > 80:
        n_sub = len(best_h)
        m_sub = n_sub // 2
        q_sub = q_from_h(best_h, n_sub)
        sub_limit = min(remaining() - 30, 60.0)
        q_sub_res, sub_score = subgradient_q(q_sub, n_sub, n_iter=3000,
                                             time_start=time.time(), time_limit=sub_limit)
        h_sub = build_h_from_q(q_sub_res, n_sub)
        s_sub = exact_score(h_sub)
        if s_sub < best_c5:
            best_c5 = s_sub
            best_h = h_sub.copy()
        # Coordinate descent on the subgradient result
        if remaining() > 40:
            coordinate_descent_q(q_sub_res, n_sub, max_sweeps=100,
                                 time_start=time.time(), time_limit=min(remaining() - 20, 40.0))
            h_sub2 = build_h_from_q(q_sub_res, n_sub)
            s_sub2 = exact_score(h_sub2)
            if s_sub2 < best_c5:
                best_c5 = s_sub2
                best_h = h_sub2.copy()

    # Phase 6: More coordinate descent on best
    if remaining() > 30:
        n_best = len(best_h)
        if n_best % 2 == 0:
            m_best = n_best // 2
            q_best = q_from_h(best_h, n_best)
            coordinate_descent_q(q_best, n_best, max_sweeps=300,
                                 time_start=time.time(), time_limit=min(remaining() - 10, 50.0))
            h_final = build_h_from_q(q_best, n_best)
            s_final = exact_score(h_final)
            if s_final < best_c5:
                best_c5 = s_final
                best_h = h_final.copy()

    # Phase 7: Random restart + coordinate descent
    if remaining() > 60:
        n_best = len(best_h)
        if n_best % 2 == 0:
            m_best = n_best // 2
            q_base = q_from_h(best_h, n_best)
            for attempt in range(3):
                if remaining() < 20:
                    break
                pert = rng.normal(0, 0.02, m_best)
                q_pert = np.clip(q_base + pert, 0.0, 1.0)
                coordinate_descent_q(q_pert, n_best, max_sweeps=200,
                                     time_start=time.time(), time_limit=min(remaining() - 10, 20.0))
                h_pert = build_h_from_q(q_pert, n_best)
                s_pert = exact_score(h_pert)
                if s_pert < best_c5:
                    best_c5 = s_pert
                    best_h = h_pert.copy()
                    q_base = q_pert.copy()

    # Final validation
    n_out = len(best_h)
    best_h = project_feasible(best_h, n_out / 2.0)
    final_score = exact_score(best_h)
    if final_score < best_c5:
        best_c5 = final_score
    best_c5 = max(best_c5, final_score)  # don't report better than actual

    return best_h, best_c5, n_out
