def project_feasible(x, target_sum):
    """Project x onto {h: 0 <= h <= 1, sum(h) = target_sum}"""
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    # Fix rounding
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        # Distribute diff among non-boundary elements
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            adj = diff / mask.sum()
            h[mask] = np.clip(h[mask] + adj, 0.0, 1.0)
        else:
            # All at bounds, need to adjust
            # This shouldn't happen if target_sum is feasible
            h = np.clip(h + diff / n, 0.0, 1.0)
    return h

def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def c5_of_h(h, dx):
    return float(compute_corrs(h).max() * dx)

    corrs = compute_corrs(h)
    # Use logsumexp of top-k for a sharper max

    sorted_corrs = np.sort(corrs)[::-1]
    weights = np.exp(np.arange(k) * 2.0)  # exponential weights favoring the max
    weights /= weights.sum()
    return float((weights @ sorted_corrs[:k]) * dx)

def topk_smooth_max_grad(h, dx, k=10):
    """Gradient of top-k smooth max"""
    n = len(h)
    corrs = compute_corrs(h)
    sorted_idx = np.argsort(corrs)[::-1][:k]
    weights = np.exp(np.arange(k) * 2.0)
    weights /= weights.sum()
    
    grad = np.zeros(n)
    for j in range(k):
        ki = sorted_idx[j]
        # Gradient of corr_ki w.r.t. h
        offset = ki - (n - 1)
        g = np.zeros(n)
        i_lo = max(0, -offset)
        i_hi = min(n - 1, n - 1 - offset)
        if i_lo <= i_hi:
            idx = np.arange(i_lo, i_hi + 1)
            g[idx + offset] += (1.0 - h[idx])
            g[idx] += (-h[idx + offset])
        grad += weights[j] * g
    return grad * dx

        t = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0

def project_feasible(x, target_sum):
    n = len(x)

        else:

            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)

        return smooth_max_grad(x, dx, beta)

import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time


def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h


def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def c5_of_h(h, dx):
    return float(compute_corrs(h).max() * dx)


def grad_corr_k(h, k, n):
    offset = k - (n - 1)
    grad = np.zeros(n)
    if offset == 0:
        grad[:] = 1.0 - 2.0 * h
        return grad
    i_lo = max(0, -offset)
    i_hi = min(n - 1, n - 1 - offset)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx + offset] += (1.0 - h[idx])
    grad[idx] += (-h[idx + offset])
    return grad


def smooth_max_val(h, dx, beta):
    corrs = compute_corrs(h)
    return float(logsumexp(beta * corrs) / beta * dx)


def smooth_max_grad(h, dx, beta):
    n = len(h)
    corrs = compute_corrs(h)
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones_like(corrs) / len(corrs)
    else:
        w /= w_sum
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        grad += w[k] * grad_corr_k(h, k, n)
    return grad * dx


def slsqp_opt(h0, dx, target_sum, beta, maxiter):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n

    def obj(x):
        return smooth_max_val(x, dx, beta)

    def g(x):
        return smooth_max_grad(x, dx, beta)

    def cons_fun(x):
        return x.sum() - target_sum

    def cons_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]

    try:
        res = minimize(obj, h0, jac=g, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'maxls': 40})
        h_out = project_feasible(res.x, target_sum)
    except Exception:
        h_out = project_feasible(h0, target_sum)
    c5 = c5_of_h(h_out, dx)
    return h_out, c5


def topk_objective(h, dx, k_val, sharpness):
    corrs = compute_corrs(h)
    sorted_c = np.sort(corrs)[::-1][:k_val]
    w = np.exp(sharpness * np.arange(k_val)[::-1].astype(float))
    w /= w.sum()
    return float(np.dot(w, sorted_c) * dx)


def topk_gradient(h, dx, k_val, sharpness):
    n = len(h)
    corrs = compute_corrs(h)
    sorted_idx = np.argsort(corrs)[::-1][:k_val]
    w = np.exp(sharpness * np.arange(k_val)[::-1].astype(float))
    w /= w.sum()
    grad
 = np.zeros(n)
    for j in range(k_val):
        ki = sorted_idx[j]
        grad += w[j] * grad_corr_k(h, ki, n)
    return grad * dx


def fista_opt(h0, dx, target_sum, max_iter, k_val=25, sharpness=2.5):
    n = len(h0)
    h = h0.copy()
    z = h0.copy()
    t = 1.0
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    alpha = 1.0 / (3.0 * n * dx + 1e-8)
    stagnant = 0

    for it in range(max_iter):
        t_old = t
        t = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        z = h + (t_old - 1.0) / t * (h - z)

        obj_z = topk_objective(z, dx, k_val, sharpness)
        grad = topk_gradient(z, dx, k_val, sharpness)

        alpha_ls = alpha
        for _ in range(25):
            h_new = project_feasible(z - alpha_ls * grad, target_sum)
            obj_new = topk_objective(h_new, dx, k_val, sharpness)
            lin = np.dot(grad, h_new - z) + 0.5 * alpha_ls * np.dot(grad, grad)
            if obj_new <= obj_z + lin:
                break
            alpha_ls *= 0.5
        else:
            h_new = project_feasible(z - alpha_ls * grad, target_sum)

        z = h
        h = h_new
        alpha = alpha_ls

        c5_val = c5_of_h(h, dx)

            best_c5 = c5_val
            best_h = h.copy()
            stagnant = 0
        else:
            stagnant += 1

            break

    return best_h, best_c5

    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    alpha = 0.01

    for it in range(max_iter):
        corrs = compute_corrs(h)
        top_idx = np.argsort(corrs)[::-1][:top_m]
        grad = np.zeros(n)
        for ki in top_idx:
            grad += grad_corr_k(h, ki, n)
        grad /= float(len(top_idx))
        grad *= dx

        improved = False
        for a in [alpha, alpha * 0.5, alpha * 0.2]:
            h_new = project_feasible(h - a * grad, target_sum)
            c5_new = c5_of_h(h_new, dx)
            if c5_new < best_c5 - 1e-15:
                h = h_new
                best_c5 = c5_new
                best_h = h.copy()
                alpha = min(a * 1.1, 0.05)
                improved = True
                break
        if not improved:
            alpha *= 0.9
            if alpha < 1e-6:
                break

    return best_h, best_c5


def make_binary_start(n, target_sum, rng):
    h = np.zeros(n)
    idx = rng.permutation(n)
    n_ones = int(target_sum)
    if n_ones > 0:
        h[idx[:n_ones]] = 1.0
    frac = target_sum - n_ones
    if frac > 1e-10 and n_ones < n:
        h[idx[n_ones]] = frac
    return h


def interpolate_to_n(h_old, n_new):
    x_old = np.arange(len(h_old)) * (2.0 / len(h_old))
    x_new = np.arange(n_new) * (2.0 / n_new)
    return np.interp(x_new, x_old, h_old)


def binary_sa(h0, dx, target_sum, n_iter, rng):
    n = len(h0)
    h = np.zeros(n)
    sorted_idx = np.argsort(h0)[::-1]
    h[sorted_idx[:n // 2]] = 1.0
    c5_cur = c5_of_h(h, dx)
    c5_best = c5_cur
    h_best = h.copy()
    T = 0.03
    T_min = 0.0005

    for it in range(n_iter):
        T = max(T_min, T * 0.9992)
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h_new = h.copy()
        h_new[i] = 0.0
        h_new[j] = 1.0
        c5_new = c5_of_h(h_new, dx)
        delta = c5_new - c5_cur
        if delta < 0 or rng.random() < np.exp(-delta / T):
            h = h_new
            c5_cur = c5_new
            if c5_cur < c5_best:
                c5_best = c5_cur
                h_best = h.copy()
    return h_best, c5_best
