import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time

def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def c5_of_h(h, dx):
    return float(compute_corrs(h).max() * dx)

def grad_corr_k(h, k, n):
    offset = k - (n - 1)
    grad = np.zeros(n)
    j1_lo = max(0, offset)
    j1_hi = min(n - 1, offset + n - 1)
    if j1_lo <= j1_hi:
        idx1 = np.arange(j1_lo, j1_hi + 1)
        grad[idx1] += (1.0 - h[idx1 - offset])
    j2_lo = max(0, -offset)
    j2_hi = min(n - 1, n - offset - 1)
    if j2_lo <= j2_hi:
        idx2 = np.arange(j2_lo, j2_hi + 1)
        grad[idx2] += (-h[idx2 + offset])
    return grad

def smooth_max_val(h, dx, beta):
    corrs = compute_corrs(h)
    return float(logsumexp(beta * corrs) / beta * dx)

def smooth_max_grad(h, dx, beta):
    n = len(h)
    corrs = compute_corrs(h)
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones_like(corrs) / len(corrs)
    else:
        w /= w_sum
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-10:
            continue
        grad += w[k] * grad_corr_k(h, k, n)
    return grad * dx

def slsqp_smooth(h0, dx, target_sum, beta, maxiter):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(smooth_max_val, h0, args=(dx, beta),
                       jac=lambda x: smooth_max_grad(x, dx, beta),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        h_out = project_feasible(res.x, target_sum)
    except Exception:
        h_out = project_feasible(h0, target_sum)
    return h_out, c5_of_h(h_out, dx)

def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def c5_of_h(h, dx):
    return float(compute_corrs(h).max() * dx)

def grad_corr_k(h, k, n):
    offset = k - (n - 1)
    grad = np.zeros(n)
    j1_lo = max(0, offset)
    j1_hi = min(n - 1, offset + n - 1)
    if j1_lo <= j1_hi:
        idx1 = np.arange(j1_lo, j1_hi + 1)
        grad[idx1] += (1.0 - h[idx1 - offset])
    j2_lo = max(0, -offset)
    j2_hi = min(n - 1, n - offset - 1)
    if j2_lo <= j2_hi:
        idx2 = np.arange(j2_lo, j2_hi + 1)
        grad[idx2] += (-h[idx2 + offset])
    return grad

    corrs = compute_corrs(h)
    return float(logsumexp(beta * corrs) / beta * dx)

def smooth_max_grad(h, dx, beta):

    corrs = compute_corrs(h)
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones_like(corrs) / len(corrs)
    else:
        w /= w_sum

def targeted_slsqp(h0, dx, target_sum, target_raw, maxiter):
    """Minimize max(0, corr_k - target_raw) as a penalty approach."""

             'jac': lambda x: np.ones(n)}]

        corrs = compute_corrs(x)
        excess = np.maximum(0.0, corrs - target_raw)
        val = np.sum(excess ** 2) + 0.1 * np.max(corrs)
        return float(val)

    def g(x):
        corrs = compute_corrs(x)
        excess = np.maximum(0.0, corrs - target_raw)
        grad = np.zeros(n)

        grad += 0.1 * grad_corr_k(x, max_k, n)
        return grad
