def select_active(C, margin_frac=0.01, pad=2, max_count=80):
    cmax = C.max()
    thresh = cmax - max(1e-8, margin_frac * cmax)
    idx = np.where(C >= thresh)[0]
    extended = set(idx.tolist())
    for a in idx:
        for delta in range(-pad, pad + 1):
            k = a + delta
            if 0 <= k <= len(C) - 1:
                extended.add(k)
    result = np.sort(np.array(extended))
    if len(result) > max_count:
        order = np.argsort(-C[result])
        result = result[order[:max_count]]
        result = np.sort(result)
    return result

    p1 = rng.uniform(0, 2 * np.pi)
    p2 = rng.uniform(0, 2 * np.pi)
    q = 0.5 + a * np.sin(2 * np.pi * k1 * t + p1) + b * np.cos(2 * np.pi * k2 * t + p2)
    rng_data = rng.rand(m) * 0.05
    q += rng_data - 0.025
    return np.clip(q, 0.0, 1.0)


def refine_slp(q, n, rng, t_deadline):
    best_q, best_c = q.copy(), eval_c5(build_sym(q, n))
    h = build_sym(best_q, n)
    C = corr_c5(h)
    cmax = C.max()
    top_idx = np.argsort(-C)[:5]
    for _ in range(5):
        if time.time() > t_deadline:
            break
        for k_idx in top_idx:
            k = int(k_idx)
            d = k - (n - 1)
            gq = sym_grad_row(h, best_q, d, n)
            gq = gq / (np.max(np.abs(gq)) + 1e-12)
            for step in [0.02, 0.01, 0.005]:
                for sign in [1.0, -1.0]:
                    q_try = np.clip(best_q + sign * step * gq, 0.0, 1.0)
                    h_try = build_sym(q_try, n)
                    c_try = eval_c5(h_try)
                    if c_try < best_c - 1e-12:
                        best_q = q_try.copy()
                        best_c = c_try
                        h = h_try
                        break
    return best_q, best_c


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    reserve = max(20.0, 0.03 * budget_s)

    best_h = None
    best_c5 = float('inf')

    try:
        h_init = np.array(initial_h_values, dtype=float).reshape(-1)
        if h_init.size >= 10 and np.all(np.isfinite(h_init)):
            n_init = len(h_init)
            h_init = repair_box_sum(h_init, n_init / 2.0)
            c5_init = eval_c5(h_init)
            best_h = h_init.copy()
            best_c5 = c5_init
    except Exception:
        pass

    if best_h is None:
        n_def = 126
        m_def = n_def // 2
        tt = (np.arange(m_def) + 0.5) / m_def
        q_def = 0.5 + 0.22 * np.sin(2 * np.pi * 2 * tt) + 0.12 * np.cos(4 * np.pi * tt + 1.0)
        q_def = np.clip(q_def, 0.0, 1.0)
        best_h = build_sym(q_def, n_def)
        best_c5 = eval_c5(best_h)

    def time_ok(needed=10.0):
        return (time.time() - t0) + needed < budget_s - reserve

    def update_best(h_new, c_new):
        nonlocal best_h, best_c5
        if c_new < best_c5 - 1e-12:
            best_h = h_new.copy()
            best_c5 = c_new
            return True
        return False

    n_base = len(best_h)
    verify_gradient(best_h, n_base)

    # Phase 1: SLP on anti-symmetric q at parent n
    if n_base % 2 == 0:
        q = q_from_h(best_h, n_base)
        h_sym = build_sym(q, n_base)
        c_sym = eval_c5(h_sym)
        if c_sym < best_c5:
            best_h = h_sym.copy()
            best_c5 = c_sym
        q = q_from_h(best_h, n_base)

        if time_ok(60):
            q, c_new = slp_sym(q, n_base, max_iters=60, R0=0.05, lam=0.01,
                               margin_frac=0.01, t_deadline=budget_s - reserve - t0 + t0)
            deadline1 = t0 + min(400.0, 0.4 * budget_s)
            if time.time() < deadline1:
                q, c_new = slp_sym(q, n_base, max_iters=80, R0=0.03, lam=0.005,
                                   margin_frac=0.008, t_deadline=deadline1)
            h_new = build_sym(q, n_base)
            update_best(h_new, eval_c5(h_new))

        if time_ok(30):
            q = q_from_h(best_h, n_base)
            q, c_new = refine_slp(q, n_base, rng, t0 + budget_s - reserve)
            h_new = build_sym(q, n_base)
            update_best(h_new, eval_c5(h_new))

    # Phase 2: Multi-resolution SLP
    n_base = len(best_h)
    n_candidates = set()
    for delta in [-12, -8, -6, -4, -2, 2, 4, 6, 8, 12]:
        nc = n_base + delta
        if 32 <= nc <= 300 and nc % 2 == 0:
            n_candidates.add(nc)
    n_candidates.add(n_base)
    n_candidates = sorted(n_candidates)

    for n_try in n_candidates:
        if not time_ok(30):
            break
        if n_try == n_base:
            continue
        h_try = interp_h(best_h, n_try)
        q_try = q_from_h(h_try, n_try)
        h_try = build_sym(q_try, n_try)
        c_try = eval_c5(h_try)
        if c_try < best_c5:
            best_h = h_try.copy()
            best_c5 = c_try
            q_try = q_from_h(best_h, n_try)

        if time_ok(20):
            dl = t0 + budget_s - reserve
            q_try, c_try2 = slp_sym(q_try, n_try, max_iters=30, R0=0.04,
                                    lam=0.01, margin_frac=0.01, t_deadline=dl)
            h_try2 = build_sym(q_try, n_try)
            c_final = eval_c5(h_try2)
            if c_final < best_c5:
                update_best(h_try2, c_final)

    # Phase 3: Full-variable SLP (break anti-symmetry)
    if time_ok(40):
        n_cur = len(best_h)
        dl = t0 + budget_s - reserve
        h_full, c_full = slp_full(best_h, n_cur, max_iters=40, R0=0.03,
                                  lam=0.01, margin_frac=0.01, t_deadline=dl)
        if c_full < best_c5:
            update_best(h_full, c_full)

    # Phase 4: Random restarts
    max_restarts = min(30, max(3, int((budget_s - reserve - (time.time() - t0)) / 20.0)))
    for trial in range(max_restarts):
        if not time_ok(15):
            break
        n_r = len(best_h)
        if n_r % 2 != 0:
            n_r -= 1
        q_r = make_random_q(n_r, rng)
        h_r = build_sym(q_r, n_r)
        c_r = eval_c5(h_r)
        if c_r < best_c5:
            best_h = h_r.copy()
            best_c5 = c_r
        dl = t0 + budget_s - reserve
        q_r2, c_r2 = slp_sym(q_r, n_r, max_iters=25, R0=0.06, lam=0.02,
                             margin_frac=0.02, t_deadline=dl)
        h_r2 = build_sym(q_r2, n_r)
        c_r2 = eval_c5(h_r2)
        if c_r2 < best_c5:
            update_best(h_r2, c_r2)
            if time_ok(20):
                q_r3, c_r3 = refine_slp(q_r2, n_r, rng, t0 + budget_s - reserve)
                h_r3 = build_sym(q_r3, n_r)
                c_r3 = eval_c5(h_r3)
                if c_r3 < best_c5:
                    update_best(h_r3, c_r3)

    # Phase 5: Interpolate best to nearby n and do quick SLP
    n_cur = len(best_h)
    for delta in [-2, -1, 1, 2, 3, 4]:
        if not time_ok(15):
            break
        nc = n_cur + delta
        if nc < 32 or nc > 300 or nc % 2 != 0:
            continue
        h_nc = interp_h(best_h, nc)
        q_nc = q_from_h(h_nc, nc)
        h_nc = build_sym(q_nc, nc)
        c_nc = eval_c5(h_nc)
        if c_nc < best_c5:
            update_best(h_nc, c_nc)
        if time_ok(10):
            q_nc2, c_nc2 = slp_sym(q_nc, nc, max_iters=20, R0=0.03,
                                   lam=0.005, margin_frac=0.008,
                                   t_deadline=t0 + budget_s - reserve)
            h_nc2 = build_sym(q_nc2, nc)
            c_nc2 = eval_c5(h_nc2)
            if c_nc2 < best_c5:
                update_best(h_nc2, c_nc2)

    # Phase 6: Final polish with tight active set
    if time_ok(15) and len(best_h) % 2 == 0:
        n_fin = len(best_h)
        q_fin = q_from_h(best_h, n_fin)
        q_fin, c_fin = slp_sym(q_fin, n_fin, max_iters=30, R0=0.02,
                               lam=0.003, margin_frac=0.005,
                               t_deadline=t0 + budget_s - reserve)
        h_fin = build_sym(q_fin, n_fin)
        c_fin = eval_c5(h_fin)
        if c_fin < best_c5:
            update_best(h_fin, c_fin)

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = repair_box_sum(best_h, len(best_h) / 2.0)
    best_c5 = eval_c5(best_h)

    return best_h, best_c5, len(best_h)
