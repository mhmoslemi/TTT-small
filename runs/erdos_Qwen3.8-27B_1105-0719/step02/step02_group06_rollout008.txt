    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def corr_raw(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(x, target_sum):

            lo = mid

                h[i] += add

def corr_grad_full(h, ell):
    """Gradient of raw_corr[ell] w.r.t. h."""
    n = len(h)
    d = ell - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -d)
    i_hi = min(n - 1, n - 1 - d)
    if i_lo > i_hi:
        return grad
    for i in range(i_lo, i_hi + 1):
        j = i + d
        grad[i] += (1.0 - h[j])
        grad[j] -= h[i]
    return grad

def select_active(corrs, margin_frac=0.03, margin_abs=1e-5, neighbor_pad=3, max_k=60):

def solve_tangent_lp(h0, active_lags, rho=0.15):
    """Solve the tangent LP to get a descent direction."""
    n = len(h0)
    corrs0 = corr_raw(h0)
    
    # Build constraint matrix
    # Variables: [h (n), t (1)]
    # Minimize t: c = [0,...,0, 1]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    A_ub = []
    b_ub = []
    for ell in active_lags:
        g = corr_grad_full(h0, ell)
        R0 = corrs0[ell]
        # Constraint: g^T h - t <= g^T h0 - R0
        row = np.zeros(n + 1)
        row[:n] = g
        row[-1] = -1.0
        A_ub.append(row)
        b_ub.append(np.dot(g, h0) - R0)
    
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    # Bounds: |h[i] - h0[i]| <= rho, 0 <= h[i] <= 1, 0 <= t <= corrs0.max() + 1
    lbs = np.maximum(0.0, h0 - rho)
    ubs = np.minimum(1.0, h0 + rho)
    bounds = list(zip(lbs, ubs))
    t_lo = 0.0
    t_hi = float(corrs0.max()) + 1.0
    bounds.append((t_lo, t_hi))
    
    # Equality: sum(h) = n/2
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = [n / 2.0]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            h_new = res.x[:n]
            return h_new
    except Exception:
        pass
    return None

import numpy as np
from scipy.optimize import linprog
import time

def eval_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def corr_raw(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(x, target_sum):
    n = len(x)
    x = np.array(x, dtype=float).copy()
    x[~np.isfinite(x)] = 0.5
    x = np.clip(x, 0.0, 1.0)
    if abs(x.sum() - target_sum) < 1e-12:
        return x
    lo, hi = -1.5, 1.5
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = np.clip(x + mid, 0.0, 1.0).sum()
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    h = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.argsort(h)
            remaining = diff
            for i in idx:
                add = min(remaining, 1.0 - h[i])
                h[i] += add
                remaining -= add
                if remaining <= 0:
                    break
        else:
            idx = np.argsort(h)[::-1]
            remaining = -diff
            for i in idx:
                sub = min(remaining, h[i])
                h[i] -= sub
                remaining -= sub
                if remaining <= 0:
                    break
    return np.clip(h, 0.0, 1.0)

def corr_grad_full(h, ell):
    n = len(h)
    d = ell - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -d)
    i_hi = min(n - 1, n - 1 - d)
    if i_lo > i_hi:
        return grad
    for i in range(i_lo, i_hi + 1):
        j = i + d
        grad[i] += (1.0 - h[j])
        grad[j] -= h[i]
    return grad

def select_active(corrs, margin_frac=0.03, margin_abs=1e-4, neighbor_pad=3, max_k=60):
    max_val = corrs.max()
    threshold = max_val - max(margin_abs, margin_frac * max_val)
    active = set(np.where(corrs >= threshold)[0].tolist())
    padded = set()
    for k in active:
        for dd in range(-neighbor_pad, neighbor_pad + 1):
            if 0 <= k + dd < len(corrs):
                padded.add(k + dd)
    lst = sorted(padded)
    if len(lst) > max_k:
        scored = sorted([(corrs[k], k) for k in lst], reverse=True)
        lst = sorted([k for _, k in scored[:max_k]])
    return lst

def solve_tangent_lp(h0, active_lags, rho):
    n = len(h0)
    corrs0 = corr_raw(h0)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_rows = []
    b_rows = []
    for ell in active_lags:
        g = corr_grad_full(h0, ell)
        R0 = corrs0[ell]
        row = np.zeros(n + 1)
        row[:n] = g
        row[-1] = -1.0
        A_rows.append(row)
        b_rows.append(np.dot(g, h0) - R0)
    A_ub = np.array(A_rows)
    b_ub = np.array(b_rows)
    lbs = np.maximum(0.0, h0 - rho)
    ubs = np.minimum(1.0, h0 + rho)
    bounds = list(zip(lbs, ubs))
    bounds.append((0.0, float(corrs0.max()) + 2.0))
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def exact_line_search(h0, direction, max_alpha=1.0, n_grid=15, n_gs=25):
    pos_mask = direction > 1e-12
    neg_mask = direction < -1e-12
    alpha_lo = 0.0
    alpha_hi = max_alpha
    if pos_mask.any():
        alpha_hi = min(alpha_hi, float(((1.0 - h0[pos_mask]) / direction[pos_mask]).min()))
    if neg_mask.any():
        alpha_lo = max(alpha_lo, float(((-h0[neg_mask] / direction[neg_mask]).max()))
    if alpha_hi <= alpha_lo + 1e-12:
        return 0.0, eval_score(h0)
    alpha_hi = min(alpha_hi, max_alpha)
    best_alpha = 0.0
    best_score = eval_score(h0)
    alphas = np.linspace(alpha_lo, alpha_hi, n_grid)
    for a in alphas:
        h_try = np.clip(h0 + a * direction, 0.0, 1.0)
        sc = eval_score(h_try)
        if sc < best_score:
            best_score = sc
            best_alpha = a
    idx = int(np.argmin([eval_score(np.clip(h0 + a * direction, 0, 1)) if False else 0 for a in alphas]))
    # Re-do grid properly to find index
    scores = []
    for a in alphas:
        h_try = np.clip(h0 + a * direction, 0.0, 1.0)
        scores.append(eval_score(h_try))
    idx = int(np.argmin(scores))
    lo = alphas[max(0, idx - 1)]
    hi = alphas[min(n_grid - 1, idx + 1)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for _ in range(n_gs):
        m1 = hi - gr * (hi - lo)
        m2 = lo + gr * (hi - lo)
        s1 = eval_score(np.clip(h0 + m1 * direction, 0.0, 1.0))
        s2 = eval_score(np.clip(h0 + m2 * direction, 0.0, 1.0))
        if s1 < s2:
            hi = m2
        else:
            lo = m1
        if abs(hi - lo) < 1e-11:
            break
    a_mid = (lo + hi) / 2.0
    sc_mid = eval_score(np.clip(h0 + a_mid * direction, 0.0, 1.0))
    if sc_mid < best_score:
        best_score = sc_mid
        best_alpha = a_mid
    return best_alpha, best_score

def transfer_line_search(h0, a, b, n_grid=10, n_gs=25):
    delta_lo = max(-h0[a], h0[b] - 1.0)
    delta_hi = min(1.0 - h0[a], -h0[b])
    if delta_hi - delta_lo < 1e-12:
        return 0.0, eval_score(h0)
    best_d = 0.0
    best_s = eval_score(h0)
    deltas = np.linspace(delta_lo, delta_hi, n_grid)
    scores = []
    for d in deltas:
        h_try = h0.copy()
        h_try[a] += d
        h_try[b] -= d
        sc = eval_score(h_try)
        scores.append(sc)
        if sc < best_s:
            best_s = sc
            best_d = d
    idx = int(np.argmin(scores))
    lo = deltas[max(0, idx - 1)]
    hi = deltas[min(n_grid - 1, idx + 1)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for _ in range(n_gs):
        m1 = hi - gr * (hi - lo)
        m2 = lo + gr * (hi - lo)
        h1 = h0.copy()
        h1[a] += m1
        h1[b] -= m1
        h2 = h0.copy()
        h2[a] += m2
        h2[b] -= m2
        s1 = eval_score(h1)
        s2 = eval_score(h2)
        if s1 < s2:
            hi = m2
        else:
            lo = m1
        if abs(hi - lo) < 1e-11:
            break
    d_mid = (lo + hi) / 2.0
    h_m = h0.copy()
    h_m[a] += d_mid
    h_m[b] -= d_mid
    s_mid = eval_score(h_m)
    if s_mid < best_s:
        best_s = s_mid
        best_d = d_mid
    return best_d, best_s

def transfer_polish(h, max_pairs=300, time_limit=None, seed=42):
    n = len(h)
    best_score = eval_score(h)
    best_h = h.copy()
    rng = np.random.RandomState(seed)
    pairs = set()
    for dist in [1, 2, 3, 4, 5, 7, n//4, n//3, n//2, 2*n//3, 3*n//4]:
        for i in range(n):
            j = (i + dist) % n
            if i != j:
                pairs.add((min(i,j), max(i,j)))
    for _ in range(n * 3):
        i, j = int(rng.randint(n)), int(rng.randint(n))
        if i != j:
            pairs.add((min(i,j), max(i,j)))
    plist = list(pairs)[:max_pairs]
    improved = True
    passes = 0
    while improved and passes < 5:
        if time_limit is not None and time.time() > time_limit:
            break
        improved = False
        passes += 1
        for (a, b) in plist:
            if time_limit is not None and time.time() > time_limit:
                break
            d, s = transfer_line_search(best_h, a, b, n_grid=8, n_gs=20)
            if s < best_score - 1e-12:
                best_h = best_h.copy()
                best_h[a] += d
                best_h[b] -= d
                best_score = s
                improved = True
    return best_h, best_score

def transfer_to_n(h_old, n_new):
    n_old = len(h_old)
    x_old = (np.arange(n_old) + 0.5) * 2.0 / n_old
    x_new = (np.arange(n_new) + 0.5) * 2.0 / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new / 2.0)

def lp_optimize_loop(h, max_rounds, time_limit, rho_list=None, k_max=55):
    n = len(h)
    if rho_list is None:
        rho_list = [0.12, 0.08, 0.05, 0.03]
    best_c5 = eval_score(h)
    for rnd in range(max_rounds):
        if time.time() > time_limit:
            break
        corrs_raw = corr_raw(h)
        active = select_active(corrs_raw, margin_frac=0.03, margin_abs=1e-4,
                               neighbor_pad=3, max_k=k_max)
        step_found = False
        for rho in rho_list:
            if time.time() > time_limit:
                break
            h_lp = solve_tangent_lp(h, active, rho)
            if h_lp is None:
                continue
            direction = h_lp - h
            if np.max(np.abs(direction)) < 1e-11:
                continue
            alpha, sc = exact_line_search(h, direction, max_alpha=1.0, n_grid=15, n_gs=25)
            if sc < best_c5 - 1e-12:
                h = np.clip(h + alpha * direction, 0.0, 1.0)
                h = project_feasible(h, n / 2.0)
                best_c5 = eval_score(h)
                step_found = True
                break
        if not step_found:
            active2 = select_active(corrs_raw, margin_frac=0.1, neighbor_pad=5, max_k=80)
            if len(active2) > len(active):
                h_lp = solve_tangent_lp(h, active2, 0.06)
                if h_lp is not None:
                    direction = h_lp - h
                    if np.max(np.abs(direction)) > 1e-10:
                        alpha, sc = exact_line_search(h, direction, max_alpha=1.0, n_grid=12, n_gs=20)
                        if sc < best_c5 - 1e-12:
                            h = np.clip(h + alpha * direction, 0.0, 1.0)
                            h = project_feasible(h, n / 2.0)
                            best_c5 = eval_score(h)
                            continue
            break
    return h, best_c5
