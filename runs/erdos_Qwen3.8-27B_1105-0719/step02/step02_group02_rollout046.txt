```python
class EpigraphProblem:
    def __init__(self, h_anchor, target_raw, rho, top_m_lags, n):
        self.h_anchor = h_anchor
        self.target_raw = target_raw
        self.rho = rho
        self.top_m_lags = top_m_lags
        self.n = n
    
    def objective(self, x):
        h = x[:self.n]
        s = x[self.n]
        return s + self.rho * np.sum((h - self.h_anchor)**2) / self.n
    
    def objective_jac(self, x):
        h = x[:self.n]
        j = np.zeros(self.n + 1)
        j[:self.n] = 2.0 * self.rho * (h - self.h_anchor) / self.n
        j[self.n] = 1.0
        return j
    
    def inequality(self, x):
        h = x[:self.n]
        s = x[self.n]
        corrs = compute_corrs(h)
        return self.target_raw + s - corrs[self.top_m_lags]
        # We want corr_k - target_raw - s <= 0
        # Which is equivalent to target_raw + s - corr_k >= 0
        # SLSQP expects constraints >= 0 for 'ineq'
    
    def inequality_jac(self, x):
        h = x[:self.n]
        j = np.zeros((len(self.top_m_lags), self.n + 1))
        for i, k in enumerate(self.top_m_lags):
            j[i, :self.n] = grad_corr_k(h, k, self.n)
            j[i, self.n] = -1.0
        return j
    
    def equality(self, x):
        return x[:self.n].sum() - self.n / 2.0
    
    def equality_jac(self, x):
        j = np.zeros(self.n + 1)
        j[:self.n] = 1.0
        return j
```
