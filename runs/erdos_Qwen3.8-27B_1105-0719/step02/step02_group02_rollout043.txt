def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def c5_of_h(h, dx):
    return float(compute_corrs(h).max() * dx)

def grad_corr_k(h, k, n):
    offset = k - (n - 1)
    grad = np.zeros(n)
    if offset == 0:
        grad[:] = 1.0 - 2.0 * h
        return grad
    if offset > 0:
        i_lo, i_hi = 0, n - 1 - offset
    else:
        i_lo, i_hi = -offset, n - 1
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    if offset > 0:
        grad[idx] += (1.0 - h[idx + offset])
        grad[idx + offset] += (-h[idx])
    else:
        m = -offset
        grad[idx + m] += (1.0 - h[idx])
        grad[idx] += (-h[idx + m])
    return grad

def smooth_max_grad(h, dx, beta):
    n = len(h)
    corrs = compute_corrs(h)
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones_like(corrs) / len(corrs)
    else:
        w /= w_sum
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-10:
            continue
        grad += w[k] * grad_corr_k(h, k, n)
    return grad * dx

_g_dx = 0.0
_g_beta = 1.0
_g_target = 0.0

    return logsumexp(_g_beta * compute_corrs(x)) / _g_beta * _g_dx

def _slq_grad(x):
    return smooth_max_grad(x, _g_dx, _g_beta)

def _slq_con_fun(x):
    return x.sum() - _g_target

def _slq_con_jac(x):
    return np.ones(len(x))

    global _g_dx, _g_beta, _g_target
    _g_dx = dx
    _g_beta = beta
    _g_target = target_sum
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slq_con_fun, 'jac': _slq_con_jac}]
    try:
        res = minimize(_slq_obj, h0, jac=_slq_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        h_out = project_feasible(res.x, target_sum)
    except Exception:
        h_out = project_feasible(h0, target_sum)
    return h_out, c5_of_h(h_out, dx)

import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time

def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def c5_of_h(h, dx):
    return float(compute_corrs(h).max() * dx)

def grad_corr_k(h, k, n):
    offset = k - (n - 1)
    grad = np.zeros(n)
    if offset == 0:
        grad[:] = 1.0 - 2.0 * h
        return grad
    if offset > 0:
        i_lo, i_hi = 0, n - 1 - offset
    else:
        i_lo, i_hi = -offset, n - 1
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    if offset > 0:
        grad[idx] += (1.0 - h[idx + offset])
        grad[idx + offset] += (-h[idx])
    else:
        m = -offset
        grad[idx + m] += (1.0 - h[idx])
        grad[idx] += (-h[idx + m])
    return grad

def smooth_max_grad(h, dx, beta):
    n = len(h)
    corrs = compute_corrs(h)
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones_like(corrs) / len(corrs)
    else:
        w /= w_sum
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-10:
            continue
        grad += w[k] * grad_corr_k(h, k, n)
    return grad * dx

_g_dx = 0.0
_g_beta = 1.0
_g_target = 0.0

def _slq_obj(x):
    return logsumexp(_g_beta * compute_corrs(x)) / _g_beta * _g_dx

def _slq_grad(x):
    return smooth_max_grad(x, _g_dx, _g_beta)

def _slq_con_fun(x):
    return x.sum() - _g_target

def _slq_con_jac(x):
    return np.ones(len(x))

def slsqp_opt(h0, dx, target_sum, beta, maxiter):
    global _g_dx, _g_beta, _g_target
    _g_dx = dx
    _g_beta = beta
    _g_target = target_sum
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slq_con_fun, 'jac': _slq_con_jac}]
    try:
        res = minimize(_slq_obj, h0, jac=_slq_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        h_out = project_feasible(res.x, target_sum)
    except Exception:
        h_out = project_feasible(h0, target_sum)
    return h_out, c5_of_h(h_out, dx)

def coord_descent_cycle_vec(h, n, t_grid):
    corrs = compute_corrs(h)
    a = np.zeros((n, 2 * n - 1))
    for d in range(1, n):
        kp = n - 1 + d
        kn = n - 1 - d
        a[:n - d, kp] += (1.0 - h[d:n])
        a[d:, kp] += (-h[:n - d])
        a[d:, kn] += (1.0 - h[:n - d])
        a[:n - d, kn] += (-h[d:n])
    a[:, n - 1] = 1.0
    b = corrs[None, :] - a * h[:, None]
    b[:, n - 1] = corrs[n - 1] - h + h * h
    new_h = h.copy()
    chunk = max(1, min(32, n))
    for j0 in range(0, n, chunk):
        j1 = min(j0 + chunk, n)
        vals = b[j0:j1, None, :] + t_grid[None, :, None] * a[j0:j1, None, :]
        vals[:, :, n - 1] = b[j0:j1, n - 1][:, None] + t_grid[None, :] - t_grid[None, :] ** 2
        max_vals = vals.max(axis=2)
        best_i = max_vals.argmin(axis=1)
        new_h[j0:j1] = t_grid[best_i]
    return new_h

def coord_descent_opt(h0, dx, target_sum, max_cycles=1000, time_limit=None):
    n = len(h0)
    h = h0.copy()
    best_c5 = c5_of_h(h, dx)
    best_h = h.copy()
    t_grid = np.linspace(0, 1, 500)
    t_start = time.time()
    stall = 0
    for cycle in range(max_cycles):
        if time_limit is not None and (time.time() - t_start) > time_limit:
            break
        new_h = coord_descent_cycle_vec(h, n, t_grid)
        h = project_feasible(new_h, target_sum)
        c5_val = c5_of_h(h, dx)

            best_h = h.copy()
            stall = 0

                break
    return best_h, best_c5

def subgrad_bb(h0, dx, target_sum, max_iter=5000, time_limit=None):
    """Subgradient with Barzilai-Borwein step size and Nesterov momentum."""
    n = len(h0)
    h = h0.copy()
    y_prev = np.zeros(n)
    s_prev = np.zeros(n)
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    t_start = time.time()
    alpha = 0.1
    mu = 0.7
    no_improve = 0
    for it in range(max_iter):
        if time_limit is not None and (time.time() - t_start) > time_limit:
            break
        corrs = compute_corrs(h)
        c5_val = corrs.max() * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 3000:
                break
        # Softmax weights on top correlations
        shifted = corrs - corrs.max()
        beta_eff = 50.0
        w = np.exp(beta_eff * shifted)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones_like(corrs) / len(corrs)
        else:
            w /= w_sum
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] < 1e-8:
                continue
            grad += w[k] * grad_corr_k(h, k, n)
        grad *= dx
        # BB step size
        y = grad - y_prev
        s = np.zeros(n)
        if it > 0:
            s = h - s_prev
        sy = np.dot(s, y)
        yy = np.dot(y, y)
        ss = np.dot(s, s)
        if yy > 1e-12 and ss > 1e-12:
            if sy > 1e-12:
                alpha = ss / sy
            else:
                alpha = sy / yy
            alpha = np.clip(alpha, 1e-6, 1.0)
        else:
            alpha = 0.05
        # Nesterov-like momentum
        v = mu * (h - s_prev)
        h_new = project_feasible(h - alpha * (grad - 0.3 * (h - s_prev)), target_sum)
        s_prev = h
        y_prev = grad
        h = h_new
    return best_h, best_c5
