import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time


def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)


def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)


def corr_vals(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx


def make_sym_h(q, n):
    m = n // 2
    q = np.clip(q, 0.0, 1.0)
    h = np.empty(n)
    h[:m] = q
    h[m:] = 1.0 - q[::-1]
    return h


def make_q_from_h(h, n):
    m = n // 2
    q = 0.5 * (h[:m] + 1.0 - h[m:][::-1])
    return np.clip(q, 0.0, 1.0)


def sym_obj(x, n, beta):
    h = make_sym_h(x, n)
    corrs = corr_vals(h)
    return float(logsumexp(beta * corrs) / beta)


def sym_jac(x, n, beta):
    m = n // 2
    dx = 2.0 / n
    h = make_sym_h(x, n)
    corrs = corr_vals(h)
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    ws = w.sum()
    if ws < 1e-30:
        w = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        w /= ws
    grad_h = np.zeros(n)
    n_lags = 2 * n - 1
    for k in range(n_lags):
        wk = w[k]
        if wk < 1e-12:
            continue
        d = k - (n - 1)
        j1_lo = max(0, -d)
        j1_hi = min(n - 1, n - 1 - d)
        if j1_lo <= j1_hi:
            idx = np.arange(j1_lo, j1_hi + 1)
            grad_h[idx] += wk * (1.0 - h[idx + d])
        j2_lo = max(0, d)
        j2_hi = min(n - 1, n - 1 + d)
        if j2_lo <= j2_hi:
            idx = np.arange(j2_lo, j2_hi + 1)
            grad_h[idx] += -wk * h[idx - d]
    grad_h *= dx
    return grad_h[:m] - grad_h[n - 1:n - 1 - m:-1]


def run_lbfgs(q0, n, beta, maxiter=2000, gtol=1e-12):
    m = n // 2
    try:
        res = minimize(sym_obj, q0, args=(n, beta), jac=sym_jac,
                       method='L-BFGS-B', bounds=[(0.0, 1.0)] * m,
                       options={'maxiter': maxiter, 'gtol': gtol, 'iprint': -1})
        h_new = make_sym_h(res.x, n)
        c5 = exact_score(h_new)
        return res.x, c5
    except Exception:
        return None, None


def run_slsqp(q0, n, beta, maxiter=800):
    m = n // 2
    try:
        res = minimize(sym_obj, q0, args=(n, beta), jac=sym_jac,
                       method='SLSQP', bounds=[(0.0, 1.0)] * m,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        h_new = make_sym_h(res.x, n)
        c5 = exact_score(h_new)
        return res.x, c5
    except Exception:
        return None, None


def active_lags_top(corrs, n_top):
    idx = np.argsort(corrs)[::-1][:n_top]
    return sorted(idx.tolist())


def epigraph_fun(x, n, active_lags):
    t = x[0]
    q = x[1:]
    h = make_sym_h(q, n)
    corrs = corr_vals(h)
    return t


def epigraph_cons_val(x, n, active_lags):
    t = x[0]
    q = x[1:]
    h = make_sym_h(q, n)
    corrs = corr_vals(h)
    return t - corrs[active_lags]


def epigraph_fun_full(x, n, active_lags, t_penalty):
    t = x[0]
    q = x[1:]
    h = make_sym_h(q, n)
    corrs = corr_vals(h)
    return t + t_penalty * max(0.0, corrs.max() - t)


def run_epi(q0, n, active_lags, t_init, maxiter=500):
    m = n // 2
    x0 = np.concatenate([[t_init], q0])
    cons = [{'type': 'ineq', 'fun': lambda x: epigraph_cons_val(x, n, active_lags),
             'jac': None}]
    bounds = [(0.0, 2.0)] + [(0.0, 1.0)] * m
    
    def epi_obj(x):
        return x[0]
    
    try:
        res = minimize(epi_obj, x0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        h_new = make_sym_h(res.x[1:], n)
        c5 = exact_score(h_new)
        if abs(h_new.sum() - n / 2.0) > 1e-6:
            return None, None
        return res.x[1:], c5
    except Exception:
        return None, None


def coordinate_descent(q0, n, n_sweeps=50, tol=1e-10):
    m = n // 2
    q = q0.copy()
    for sweep in range(n_sweeps):
        improved = False
        for i in range(m):
            best_val = q[i]
            best_obj = None
            h_cur = make_sym_h(q, n)
            c_cur = exact_score(h_cur)
            for trial in [q[i] - 0.1, q[i] - 0.05, q[i] - 0.01,
                          q[i] + 0.01, q[i] + 0.05, q[i] + 0.1]:
                if trial < -0.01 or trial > 1.01:
                    continue
                q[i] = min(1.0, max(0.0, trial))
                h_try = make_sym_h(q, n)
                c_try = exact_score(h_try)
                if best_obj is None or c_try < best_obj:
                    best_obj = c_try
                    best_val = q[i]
            q[i] = best_val
        if sweep % 10 == 0:
            c_now = exact_score(make_sym_h(q, n))
            if sweep > 0 and abs(c_now - c_cur) < tol:
                break
    return q
