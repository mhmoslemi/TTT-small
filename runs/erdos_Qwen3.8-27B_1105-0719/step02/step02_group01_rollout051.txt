    grad[idx + offset] += (1.0 - h[idx])
    grad[idx] -= h[idx + offset]
    return grad

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.special import logsumexp
import time


def make_feasible(x, target_sum):
    n = len(x)
    x = np.asarray(x, dtype=float)
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    s = (lo + hi) / 2.0
    h = np.clip(x - s, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0 - 1e-12
            slack = np.where(mask, 1.0 - h, 0.0)
            total_slack = slack.sum()
            if total_slack > 1e-15:
                h += diff * slack / total_slack
        else:
            mask = h > 1e-12
            cap = np.where(mask, h, 0.0)
            total_cap = cap.sum()
            if total_cap > 1e-15:
                h += diff * cap / total_cap
    return np.clip(h, 0.0, 1.0)


def compute_c5(h, dx):
    corrs = np.correlate(h, 1.0 - h, mode='full')
    return corrs.max() * dx


def grad_corr_k(h, k, n):
    offset = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -offset)
    i_hi = min(n - 1, n - 1 - offset)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx + offset] += (1.0 - h[idx])
    grad[idx] -= h[idx + offset]
    return grad


def lp_minimax(h, dx, target_sum, alpha, max_iters, use_all):
    n = len(h)
    n_corr = 2 * n - 1
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur, dx)
    no_improve = 0

    for it in range(max_iters):
        corrs = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        c5_val = corrs.max() * dx
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_cur.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 50:
                break

        if use_all:
            active = np.arange(n_corr)
        else:
            n_top = min(60, n_corr)
            sorted_idx = np.argsort(corrs)
            active = sorted_idx[-n_top:]

        n_act = len(active)
        A_ub = np.zeros((n_act, n + 1))
        b_ub = np.zeros(n_act)
        for j, k in enumerate(active):
            g = grad_corr_k(h_cur, k, n)
            A_ub[j, :n] = g
            A_ub[j, n] = -1.0
            b_ub[j] = -corrs[k]

        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)

        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0

        bounds = []
        for i in range(n):
            bounds.append((-h_cur[i] * alpha, (1.0 - h_cur[i]) * alpha))
        bounds.append((-1e6, None))

        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if res.status != 0:
                alpha *= 0.5
                if alpha < 1e-8:
                    break
                continue
        except Exception:
            break

        delta = res.x[:n]
        h_new = h_cur + delta
        h_new = make_feasible(h_new, target_sum)
        new_c5 = compute_c5(h_new, dx)

        if new_c5 < c5_val - 1e-14:
            h_cur = h_new
            no_improve = 0
            alpha = min(alpha * 1.05, 0.5)
        else:
            alpha *= 0.4
            if alpha < 1e-8:
                break

        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()

    return best_h, best_c5


def slsqp_smooth(h0, dx, target_sum, beta, maxiter):
    n = len(h0)

    def _obj(x):
        corrs = np.correlate(x, 1.0 - x, mode='full')
        return logsumexp(beta * corrs) / beta * dx

    def _grad_obj(x):
        corrs = np.correlate(x, 1.0 - x, mode='full')
        shifted = corrs - corrs.max()
        w = np.exp(beta * shifted)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones_like(corrs) / len(corrs)
        else:
            w /= w_sum
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] > 1e-13:
                grad += w[k] * grad_corr_k(x, k, n)
        return grad * dx

    def _c_fun(x):
        return x.sum() - target_sum

    def _c_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': _c_fun, 'jac': _c_jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_obj, h0, jac=_grad_obj, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        h_out = make_feasible(res.x, target_sum)
    except Exception:
        h_out = make_feasible(h0, target_sum)
    return h_out, compute_c5(h_out, dx)


def sa_quantized(n, target_sum, rng, M, max_iters, t0, deadline):
    target_a = n * M // 2
    if abs(target_sum * M - target_a) > 0.01:
        return None, None

    dx_g = 2.0 / n
    a = np.full(n, M // 2, dtype=float)
    current_sum = a.sum()
    diff = target_a - current_sum
    if diff > 0:
        idx = rng.permutation(n)
        for i in range(min(int(diff), n)):
            if a[idx[i]] < M:
                a[idx[i]] += 1
    elif diff < 0:
        idx = rng.permutation(n)
        for i in range(min(-int(diff), n)):
            if a[idx[i]] > 0:
                a[idx[i]] -= 1

    h = a / M
    h = make_feasible(h, target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx_g)

    tau = 0.004
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if it % 400 == 0:
            tau *= 0.93

        i1 = rng.randint(n)
        i2 = rng.randint(n)
        if i1 == i2:
            continue

        if a[i1] > 0 and a[i2] < M:
            a[i1] -= 1
            a[i2] += 1
            h_new = a / M
            h_new = make_feasible(h_new, target_sum)
            c5_new = compute_c5(h_new, dx_g)
            if c5_new < best_c5 or rng.random() < np.exp(-(c5_new - best_c5) / max(tau, 1e-12)):
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
            else:
                a[i1] += 1
                a[i2] -= 1
        elif a[i2] > 0 and a[i1] < M:
            a[i2] -= 1
            a[i1] += 1
            h_new = a / M
            h_new = make_feasible(h_new, target_sum)
            c5_new = compute_c5(h_new, dx_g)
            if c5_new < best_c5 or rng.random() < np.exp(-(c5_new - best_c5) / max(tau, 1e-12)):
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
            else:
                a[i2] += 1
                a[i1] -= 1

    return best_h, best_c5


def make_periodic(n, period):
    h = np.zeros(n)
    fill = max(1, period // 2)
    for i in range(n):
        if (i % period) < fill:
            h[i] = 1.0
    return h


def make_random_binary(n, target_sum, rng):
    h = np.zeros(n)
    idx = rng.permutation(n)
    n_ones = int(target_sum)
    h[idx[:n_ones]] = 1.0
    frac = target_sum - n_ones
    if frac > 1e-10 and n_ones < n:
        h[idx[n_ones]] = frac
    return make_feasible(h, target_sum)


def transfer_grid(h_src, n_new):
    n_src = len(h_src)
    x_src = (np.arange(n_src) + 0.5) * 2.0 / n_src
    x_new = (np.arange(n_new) + 0.5) * 2.0 / n_new
    h_new = np.interp(x_new, x_src, h_src)
    return make_feasible(h_new, n_new / 2.0)
