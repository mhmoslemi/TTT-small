Let me think about what specific improvements I can make:

1. **Tangent LP approach**: At a current point h, for each correlation lag, compute the gradient of the raw correlation with respect to h. Then solve an LP that minimizes a scalar t subject to linearized constraints for the active correlations. This gives a direction to move. Then do a line search along that direction.

2. **Pair transfer line search**: For pairs (a,b), search over delta in h[a]+=delta, h[b]-=delta. This is exact and mass-preserving.

3. **Key insight**: The previous approach uses L-BFGS-B on smoothed max and SLSQP on epigraph. The LP approach is different because it uses linear constraints directly, which can be more effective for minimax problems.

Let me think about the implementation:

For the tangent LP:
- At point h (length n), compute raw correlations for all lags
- Select active lags (those near the maximum)
- For each active lag k, compute gradient of raw_corr[k] w.r.t. h
- Set up LP: minimize t
  Subject to: for each active lag k: grad_k · h_new - t <= grad_k · h - raw_corr[k]
  0 <= h_new[i] <= 1
  sum(h_new) = n/2
  h_new[i] in [h[i] - rho, h[i] + rho] (trust region)

- Solve LP, get direction d = h_new - h
- Line search: h(alpha) = h + alpha * d, find best alpha in [0, 1]
- The key is that d preserves sum(h_new) = n/2 exactly (from LP constraint), so h + alpha*d also preserves sum for any alpha.

Actually, the constraint sum(h_new) = n/2 means sum(d) = 0, so sum(h + alpha*d) = sum(h) + alpha * sum(d) = n/2. 

For the line search, I need to be careful about box constraints. If d has both positive and negative components, alpha is bounded by the box constraints.

Let me also think about the pair transfer approach:
- For a pair (a, b), the direction is e_a - e_b

import numpy as np
from scipy.optimize import linprog
import time

def eval_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def corr_raw(h):
    return np.correlate(h, 1.0 - h, mode='full')

def corr_grad_full(h, k):
    """Gradient of raw_corr[k] w.r.t. h[i] for all i."""
    n = len(h)
    d = k - (n - 1)  # shift
    grad = np.zeros(n)
    i_lo = max(0, -d)
    i_hi = min(n - 1, n - 1 - d)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    jdx = idx + d
    grad[idx] += (1.0 - h[jdx])
    grad[jdx] += (-h[idx])
    return grad

def project_to_box_sum(x, target):
    """Project x to [0,1]^n with sum = target."""
    n = len(x)
    x = np.array(x, dtype=float).copy()
    x[~np.isfinite(x)] = 0.5
    x = np.clip(x, 0.0, 1.0)
    if abs(x.sum() - target) < 1e-12:
        return x
    
    # Bisection for uniform shift
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.clip(x + mid, 0, 1).sum() > target:
            hi = mid
        else:
            lo = mid
    c = (lo + hi) / 2.0
    v = np.clip(x + c, 0, 1)
    
    residual = target - v.sum()
    if abs(residual) < 1e-11:
        return v
    
    if residual > 0:
        order = np.argsort(v)
        for i in order:
            add = min(residual, 1.0 - v[i])
            v[i] += add
            residual -= add
            if residual < 1e-13:
                break
    else:
        order = np.argsort(v)[::-1]
        for i in order:
            sub = min(-residual, v[i])
            v[i] -= sub
            residual += sub
            if residual > -1e-13:
                break
    return v

def tangent_lp_step(h, active_k, rho, t_upper):
    """Solve tangent LP at point h with trust region rho.
    Returns new point h_new or None if failed."""
    n = len(h)
    raw = corr_raw(h)
    
    # Select active lags
    if active_k is None:
        max_raw = raw.max()
        threshold = max_raw - max(1e-6, 0.03 * max_raw)
        active_k = np.where(raw >= threshold)[0]
    
    # Build constraint matrix
    # For each active lag k: grad_k · h_new - t <= grad_k · h - raw[k]
    # i.e., grad_k · h_new - t <= grad_k · (h_new - d) - raw[k] ... no
    # Actually: grad_k · h_new - t <= grad_k · h - raw[k]
    # where grad_k is gradient at h
    # Rearranging: grad_k · h_new - t - (grad_k · h - raw[k]) <= 0
    
    m_constraints = len(active_k)
    # A_ub will be m_constraints x (n+1)
    A_ub = np.zeros((m_constraints, n + 1))
    b_ub = np.zeros(m_constraints)
    
    for row, k in enumerate(active_k):
        g = corr_grad_full(h, k)
        A_ub[row, :n] = g
        A_ub[row, n] = -1.0
        b_ub[row] = g @ h - raw[k]
    
    # Box constraints on h: handled by bounds
    # Equality: sum(h_new) = n/2
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    # Bounds: h[i] in [max(0, h[i]-rho), min(1, h[i]+rho)], t in [0, t_upper]
    bounds = []
    for i in range(n):
        lo_i = max(0.0, h[i] - rho)
        hi_i = min(1.0, h[i] + rho)
        bounds.append((lo_i, hi_i))
    bounds.append((0.0, t_upper))
    
    # Objective: minimize t
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            h_new = res.x[:n]
            t_new = res.x[n]
            # Verify box and sum
            h_new = np.clip(h_new, 0, 1)
            # Re-project to be safe
            h_new = project_to_box_sum(h_new, n / 2.0)
            return h_new
    except Exception:
        pass
    return None

def line_search_alpha(h, d, max_alpha=1.0, n_grid=50):
    """Find best alpha in [0, max_alpha] for h + alpha*d.
    d must satisfy sum(d)=0 for mass preservation."""
    n = len(h)
    # Find max feasible alpha from box constraints
    pos = d > 1e-15
    neg = d < -1e-15
    if pos.any():
        alpha_ub = min(alpha_ub, (1.0 - h[pos]) / d[pos]).min()
    if neg.any():
        alpha_ub = min(alpha_ub, (-h[neg]) / d[neg]).min()
    alpha_ub = min(max_alpha, max(0, alpha_ub))
    
    if alpha_ub < 1e-12:
        return 0.0, h
    
    # Grid search
    alphas = np.linspace(0, alpha_ub, n_grid)
    best_alpha = 0.0
    best_score = eval_score(h)
    best_h = h.copy()
    
    for a in alphas:
        if a < 1e-12:
            continue
        h_try = h + a * d
        h_try = np.clip(h_try, 0, 1)
        s = h_try.sum()
        if abs(s - n/2.0) > 1e-8:
            # Shouldn't happen if sum(d)=0
            h_try = project_to_box_sum(h_try, n/2.0)
        sc = eval_score(h_try)
        if sc < best_score:
            best_score = sc
            best_alpha = a
            best_h = h_try.copy()
    
    # Refine around best
    if best_alpha > 0:
        margin = alpha_ub / n_grid
        lo = max(0, best_alpha - margin)
        hi = min(alpha_ub, best_alpha + margin)
        alphas_fine = np.linspace(lo, hi, 20)
        for a in alphas_fine:
            h_try = h + a * d

            sc = eval_score(h_try)
            if sc < best_score:
                best_score = sc
                best_alpha = a
                best_h = h_try.copy()
    
    return best_alpha, best_h

def pair_transfer_search(h, a, b, n_grid=40):
    """Search over delta for h[a]+=delta, h[b]-=delta."""
    n = len(h)
    lo = max(-h[a], h[b] - 1.0)
    hi = min(1.0 - h[a], h[b])
    if hi - lo < 1e-12:
        return h, eval_score(h)
    
    best_score = eval_score(h)
    best_h = h.copy()
    best_delta = 0.0
    
    deltas = np.linspace(lo, hi, n_grid)
    for dt in deltas:
        if abs(dt) < 1e-12:
            continue
        h_try = h.copy()
        h_try[a] += dt
        h_try[b] -= dt
        if h_try[a] < -1e-10 or h_try[a] > 1+1e-10 or h_try[b] < -1e-10 or h_try[b] > 1+1e-10:
            continue
        h_try = np.clip(h_try, 0, 1)
        sc = eval_score(h_try)
        if sc < best_score:
            best_score = sc
            best_h = h_try.copy()
            best_delta = dt
    
    # Refine
    if best_delta != 0:
        margin = (hi - lo) / n_grid
        lo2 = max(lo, best_delta - margin)
        hi2 = min(hi, best_delta + margin)
        deltas_fine = np.linspace(lo2, hi2, 15)

            h_try[a] += dt
            h_try[b] -= dt

            sc = eval_score(h_try)
            if sc < best_score:

                best_h = h_try.copy()
    
    return best_h, best_score

def select_active_k(raw, margin_frac=0.03, neighbor_pad=3, max_k=60):
    max_raw = raw.max()
    threshold = max_raw - max(1e-6, margin_frac * max_raw)
    active = np.where(raw >= threshold)[0]
    padded = set()
    for k in active:
        for d in range(-neighbor_pad, neighbor_pad + 1):
            if 0 <= k + d < len(raw):
                padded.add(k + d)
    active_list = sorted(padded)
    if len(active_list) > max_k:
        scored = sorted([(raw[k], k) for k in active_list], reverse=True)
        active_list = sorted([k for _, k in scored[:max_k]])
    return active_list

import numpy as np
from scipy.optimize import linprog
import time


def eval_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)


def corr_raw(h):
    return np.correlate(h, 1.0 - h, mode='full')


def corr_grad_full(h, k):
    n = len(h)
    d = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -d)
    i_hi = min(n - 1, n - 1 - d)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    jdx = idx + d
    grad[idx] += (1.0 - h[jdx])
    grad[jdx] += (-h[idx])
    return grad


def project_to_box_sum(x, target):
    n = len(x)
    x = np.array(x, dtype=float).copy()
    x[~np.isfinite(x)] = 0.5
    x = np.clip(x, 0.0, 1.0)
    if abs(x.sum() - target) < 1e-12:
        return x
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.clip(x + mid, 0, 1).sum() > target:
            hi = mid
        else:
            lo = mid
    c = (lo + hi) / 2.0
    v = np.clip(x + c, 0, 1)
    residual = target - v.sum()
    if abs(residual) < 1e-11:
        return v
    if residual > 0:
        order = np.argsort(v)
        for i in order:
            add = min(residual, 1.0 - v[i])
            v[i] += add
            residual -= add
            if residual < 1e-13:
                break
    else:
        order = np.argsort(v)[::-1]
        for i in order:
            sub = min(-residual, v[i])
            v[i] -= sub
            residual += sub
            if residual > -1e-13:
                break
    return v


def tangent_lp_step(h, active_k, rho, t_upper):
    n = len(h)
    raw = corr_raw(h)
    m_c = len(active_k)
    A_ub = np.zeros((m_c, n + 1))
    b_ub = np.zeros(m_c)
    for row, k in enumerate(active_k):
        g = corr_grad_full(h, k)
        A_ub[row, :n] = g
        A_ub[row, n] = -1.0
        b_ub[row] = g @ h - raw[k]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    bounds = []
    for i in range(n):
        lo_i = max(0.0, h[i] - rho)
        hi_i = min(1.0, h[i] + rho)
        bounds.append((lo_i, hi_i))
    bounds.append((0.0, t_upper))
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            h_new = np.clip(res.x[:n], 0, 1)
            h_new = project_to_box_sum(h_new, n / 2.0)
            return h_new
    except Exception:
        pass
    return None


def line_search_alpha(h, d, max_alpha, n_grid):
    n = len(h)
    pos = d > 1e-15
    neg = d < -1e-15
    alpha_ub = max_alpha
    if pos.any():

    if neg.any():
        alpha_ub = min(alpha_ub, float((-h[neg] / d[neg]).min()))
    alpha_ub = max(0.0, alpha_ub)

        return 0.0, h.copy()

def pair_transfer_search(h, a, b, n_grid):
    lo = max(-h[a], h[b] - 1.0)
    hi = min(1.0 - h[a], h[b])
    if hi - lo < 1e-12:
        return h, eval_score(h)
    best_score = eval_score(h)
    best_h = h.copy()
    best_delta = 0.0
    deltas = np.linspace(lo, hi, n_grid)
    for dt in deltas:
        if abs(dt) < 1e-12:
            continue
        h_try = h.copy()
        h_try[a] += dt
        h_try[b] -= dt
        h_try = np.clip(h_try, 0, 1)

            best_score = sc
            best_h = h_try.copy()
            best_delta = dt

def select_active_k(raw, margin_frac, neighbor_pad, max_k):
    max_raw = raw.max()
    threshold = max_raw - max(1e-6, margin_frac * max_raw)
    active = np.where(raw >= threshold)[0]
    padded = set()
    for k in active:
        for d in range(-neighbor_pad, neighbor_pad + 1):
            if 0 <= k + d < len(raw):
                padded.add(k + d)
    active_list = sorted(padded)
    if len(active_list) > max_k:
        scored = sorted([(raw[k], k) for k in active_list], reverse=True)
        active_list = sorted([k for _, k in scored[:max_k]])
    return active_list


def transfer_to_n(h_old, n_new):
    n_old = len(h_old)
    x_old = (np.arange(n_old) + 0.5) * 2.0 / n_old
    x_new = (np.arange(n_new) + 0.5) * 2.0 / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return project_to_box_sum(h_new, n_new / 2.0)
