from scipy.optimize import linprog, minimize
from scipy.interpolate import interp1d

def eval_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def corr_raw(h):
    return np.correlate(h, 1.0 - h, mode='full')

def corr_grad_full(h, ell):
    """Gradient of raw correlation at lag index ell (0 to 2n-2)."""
    n = len(h)
    d = ell - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -d)
    i_hi = min(n - 1, n - 1 - d)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    # For d=0, idx+d = idx, so grad[idx] += (1-h[idx]) and grad[idx] += (-h[idx])
    # Net: 1 - 2*h[idx], which is correct
    # For d != 0, idx and idx+d are disjoint
    grad[idx] += (1.0 - h[idx + d])
    grad[idx + d] += (-h[idx])
    return grad

def project_to_box_sum(x, target):
    """Project x to [0,1]^n with sum = target."""
    n = len(x)
    x = np.asarray(x, dtype=float)
    x[~np.isfinite(x)] = 0.5
    x = np.clip(x, 0.0, 1.0)
    if abs(x.sum() - target) < 1e-12:
        return x
    lo = -2.0
    hi = 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(x + mid, 0.0, 1.0).sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    h = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:

def select_active(corrs, margin_frac=0.02, margin_abs=1e-6, max_k=60):
    """Select active correlation lags (near the maximum)."""
    n = len(corrs)
    max_val = corrs.max()
    threshold = max_val - max(margin_abs, margin_frac * max_val)
    active = np.where(corrs >= threshold)[0]

        top_idx = np.argsort(corrs)[::-1][:max_k]
        active = np.sort(top_idx)
    return list(active)

def tangent_lp_step(h, active_lags, rho=0.1, method='highs'):
    """
    Solve tangent LP at point h.
    Returns direction d (step) and predicted t.
    """
    n = len(h)
    corrs = corr_raw(h)
    
    # Build constraint matrix
    # Variables: [d_0, ..., d_{n-1}, t], total n+1
    # For each active lag: [grad_ell, -1] @ [d, t] <= -R_ell
    # Box: -h[i] <= d[i] <= 1-h[i]
    # Mass: sum(d) = 0
    # Trust: |d[i]| <= rho
    
    n_vars = n + 1
    A_ub = []
    b_ub = []
    
    for ell in active_lags:
        g = corr_grad_full(h, ell)
        row = np.zeros(n_vars)
        row[:n] = g
        row[n] = -1.0
        A_ub.append(row)
        b_ub.append(-corrs[ell])
    
    if A_ub:
        A_ub = np.array(A_ub)
        b_ub = np.array(b_ub)
    
    # Bounds: d[i] in [-min(h[i], rho), min(1-h[i], rho)]
    bounds = []
    for i in range(n):
        lo = max(-h[i], -rho)
        hi = min(1.0 - h[i], rho)
        bounds.append((lo, hi))
    bounds.append((0.0, corrs[corrs > 0].max() + 1.0 if (corrs > 0).any() else 1.0))
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n_vars))
    A_eq[0, :n] = 1.0
    b_eq = [0.0]
    
    # Objective: minimize t
    c = np.zeros(n_vars)
    c[n] = 1.0
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method=method)
        if res.success:
            d = res.x[:n]
            t_pred = res.x[n]
            return d, t_pred
    except Exception:
        pass
    return None, None

def transfer_line_search(h, a, b, n_grid=40, n_refine=20):
    """
    Exact 1D line search along transfer direction: h[a]+delta, h[b]-delta.
    Returns best delta and best h.
    """
    delta_lo = max(-h[a], h[b] - 1.0)
    delta_hi = min(1.0 - h[a], -h[b])
    
    if delta_hi <= delta_lo:
        return 0.0, h.copy()
    
    best_delta = 0.0
    best_score = eval_score(h)
    
    # Grid search
    deltas = np.linspace(delta_lo, delta_hi, n_grid)
    scores = []
    for delta in deltas:
        h_new = h.copy()
        h_new[a] += delta
        h_new[b] -= delta
        h_new = np.clip(h_new, 0.0, 1.0)
        sc = eval_score(h_new)
        scores.append(sc)
        if sc < best_score:
            best_score = sc
            best_delta = delta
    
    # Golden section refinement around best
    if best_delta != 0.0 or (deltas[0] != 0.0 and deltas[-1] != 0.0):
        # Find neighborhood
        idx = np.argmin(scores)
        if idx > 0:
            lo = deltas[idx - 1]
        else:
            lo = delta_lo
        if idx < n_grid - 1:
            hi = deltas[idx + 1]
        else:
            hi = delta_hi
        
        # Golden section
        gr = (np.sqrt(5) - 1) / 2
        a_g = lo
        b_g = hi
        c_g = b_g - gr * (b_g - a_g)
        d_g = a_g + gr * (b_g - a_g)
        
        def f(delta):
            h_t = h.copy()
            h_t[a] += delta
            h_t[b] -= delta
            return eval_score(np.clip(h_t, 0.0, 1.0))
        
        fc = f(c_g)
        fd = f(d_g)
        for _ in range(n_refine):
            if fc < fd:
                b_g = d_g
                d_g = c_g
                fd = fc
                c_g = b_g - gr * (b_g - a_g)
                fc = f(c_g)
            else:
                a_g = c_g
                c_g = d_g
                fc = fd
                d_g = a_g + gr * (b_g - a_g)
                fd = f(d_g)
        
        best_delta = (a_g + b_g) / 2.0
        h_best = h.copy()
        h_best[a] += best_delta
        h_best[b] -= best_delta
        best_score = eval_score(np.clip(h_best, 0.0, 1.0))
    
    h_best = h.copy()
    h_best[a] += best_delta
    h_best[b] -= best_delta
    return best_delta, np.clip(h_best, 0.0, 1.0)

def transfer_line_search(h, a, b, n_grid, n_refine, seed):

    delta_hi = min(1.0 - h[a], -h[b])
    if delta_hi <= delta_lo + 1e-10:
        return 0.0, h.copy()
    
    deltas = np.linspace(delta_lo, delta_hi, n_grid)
    scores = np.array([transfer_eval(h, a, b, d) for d in deltas])
    idx = np.argmin(scores)
    best_delta = deltas[idx]
    best_score = scores[idx]
    
    # Also check delta=0
    s0 = eval_score(h)
    if s0 < best_score:
        best_delta = 0.0
        best_score = s0

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = delta_lo
    hi = delta_hi
    if idx > 0:
        lo = deltas[idx - 1]
    if idx < n_grid - 1:
        hi = deltas[idx + 1]
    
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    fc1 = transfer_eval(h, a, b, c1)
    fc2 = transfer_eval(h, a, b, c2)
    
    for _ in range(n_refine):
        if fc1 < fc2:
            hi = c2
            c2 = c1
            fc2 = fc1
            c1 = hi - gr * (hi - lo)
            fc1 = transfer_eval(h, a, b, c1)
        else:
            lo = c1
            c1 = c2
            fc1 = fc2
            c2 = lo + gr * (hi - lo)
            fc2 = transfer_eval(h, a, b, c2)
    
    mid = (lo + hi) / 2.0
    fmid = transfer_eval(h, a, b, mid)
    if fmid < best_score:
        best_delta = mid
        best_score = fmid

    h_best[a] += best_delta
    h_best[b] -= best_delta

def eval_score(h):
    n = len(h)

def corr_raw(h):

def corr_grad_full(h, ell):
    n = len(h)
    d = ell - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -d)
    i_hi = min(n - 1, n - 1 - d)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx] += (1.0 - h[idx + d])
    grad[idx + d] += (-h[idx])
    return grad


def project_to_box_sum(x, target):
    n = len(x)
    x = np.asarray(x, dtype=float).copy()
    x[~np.isfinite(x)] = 0.5

    if abs(diff) > 1e-13:
