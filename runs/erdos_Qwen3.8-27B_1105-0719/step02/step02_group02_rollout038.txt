    """Gradient of corr_k wrt h, where k is index into full correlation output."""

        i_hi = n - 1

    # d/dh[idx+offset] = -h[idx]

class EpigraphSolver:
    def __init__(self, h_anchor, target_raw, rho, n):
        self.h_anchor = h_anchor
        self.target_raw = target_raw
        self.rho = rho
        self.n = n
        self.target_sum = n / 2.0
    
    def objective(self, x):
        h = x[:self.n]
        s = x[self.n]
        reg = self.rho * np.sum((h - self.h_anchor)**2) / self.n
        return s + reg
    
    def objective_grad(self, x):
        h = x[:self.n]
        grad = np.zeros(self.n + 1)
        grad[:self.n] = 2 * self.rho * (h - self.h_anchor) / self.n
        grad[self.n] = 1.0
        return grad
    
    def ineq_constraint(self, x):
        h = x[:self.n]
        s = x[self.n]
        corrs = np.correlate(h, 1.0 - h, mode='full')
        return corrs - self.target_raw - s
    
    def ineq_jac(self, x):
        h = x[:self.n]
        n = self.n
        L = 2 * n - 1
        J = np.zeros((L, n + 1))
        for k in range(L):
            J[k, :n] = corr_grad_k(h, k, n)
            J[k, n] = -1.0
        return J
    
    def eq_constraint(self, x):
        return x[:self.n].sum() - self.target_sum
    
    def eq_jac(self, x):
        J = np.zeros((1, self.n + 1))
        J[0, :self.n] = 1.0
        return J

def all_corr_gradients(h):
    """Return array of shape (2n-1, n) where row k is gradient of corr_k wrt h."""
    n = len(h)
    L = 2*n - 1
    J = np.zeros((L, n))
    for k in range(L):
        d = k - (n-1)
        # Term 1: (1-h[j-d]) for j where 0<=j-d<n
        j1_lo = max(0, d)  # j-d >= 0 => j >= d
        j1_hi = min(n-1, n-1+d)  # j-d < n => j < n+d
        if j1_lo <= j1_hi:
            js = np.arange(j1_lo, j1_hi+1)
            J[k, js] += (1.0 - h[js - d])
        # Term 2: -h[j+d] for j where 0<=j+d<n
        j2_lo = max(0, -d)  # j+d >= 0 => j >= -d
        j2_hi = min(n-1, n-1-d)  # j+d < n => j < n-d
        if j2_lo <= j2_hi:
            js = np.arange(j2_lo, j2_hi+1)
            J[k, js] -= h[js + d]
    return J

def all_corr_grads(h):
    n = len(h)
    L = 2 * n - 1
    J = np.zeros((L, n))
    for k in range(L):
        d = k - (n - 1)

        if lo2 <= hi2:

    return J

def lse_objective(h, dx, beta):
    corrs = compute_raw_corr(h) * dx
    return float(logsumexp(beta * corrs) / beta)

def lse_gradient(h, dx, beta):
    n = len(h)
    corrs = compute_raw_corr(h)
    cd = corrs * dx
    shifted = cd - cd.max()
    w = np.exp(beta * shifted)
    wsum = w.sum()
    if wsum < 1e-30:
        w = np.ones_like(corrs) / len(corrs)
    else:
        w /= wsum
    G = all_corr_grads(h)
    grad = w @ G
    return grad * dx

    n = len(h0)
    
    def obj(x):
        return lse_objective(x, dx, beta)
    
    def grad(x):
        return lse_gradient(x, dx, beta)
    
    def eq(x):
        return x.sum() - target_sum
    
    def eq_jac(x):
        return np.ones((1, n))
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq, 'jac': eq_jac}]

                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-16})
    return project_feasible(res.x, target_sum)

       minimize s + rho*||h-anchor||^2/n
