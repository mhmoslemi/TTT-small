import numpy as np
from scipy.optimize import minimize, linprog
import time


def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
        else:
            h += diff / n
    return np.clip(h, 0.0, 1.0)


def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def c5_of_h(h, dx):
    return compute_corrs(h).max() * dx


def grad_corr_k(h, k, n):
    """Correct gradient of corr[k] w.r.t. h."""
    offset = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, offset)
    i_hi = min(n, n + offset)
    if i_lo >= i_hi:
        return grad
    i = np.arange(i_lo, i_hi)
    grad[i] += (1.0 - h[i - offset])
    grad[i - offset] += (-h[i])
    return grad


def grad_all_corrs(h):
    n = len(h)
    grads = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        grads[k] = grad_corr_k(h, k, n)
    return grads


def active_set_lp(h0, active_indices, target_sum, delta):
    n = len(h0)
    corrs0 = compute_corrs(h0)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_rows = []
    b_vals = []
    for k in active_indices:
        g = grad_corr_k(h0, k, n)
        rhs = np.dot(g, h0) - corrs0[k]
        row = np.zeros(n + 1)
        row[:n] = g
        row[-1] = -1.0
        A_rows.append(row)
        b_vals.append(rhs)
    if A_rows:
        A_ub = np.array(A_rows)
        b_ub = np.array(b_vals)
    else:
        A_ub = np.zeros((0, n + 1))
        b_ub = np.zeros(0)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target_sum])
    lo = np.maximum(0.0, h0 - delta)
    hi = np.minimum(1.0, h0 + delta)
    bounds = list(zip(lo, hi)) + [(0.0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], float(res.x[-1])
    except Exception:
        pass
    return None, None


def smooth_max_obj(h, dx, beta):
    corrs = compute_corrs(h)
    m = corrs.max()
    if beta < 1:
        return m * dx
    return (m + np.log(np.sum(np.exp(beta * (corrs - m)))) / beta) * dx


def smooth_max_grad(h, dx, beta):
    n = len(h)
    corrs = compute_corrs(h)
    m = corrs.max()
    if beta < 1:
        idx = int(np.argmax(corrs))
        return grad_corr_k(h, idx, n) * dx
    shifted = corrs - m
    w = np.exp(beta * shifted)
    w /= w.sum()
    grad = np.zeros(n)
    for k in range(len(corrs)):
        if w[k] < 1e-14:
            continue
        grad += w[k] * grad_corr_k(h, k, n)
    return grad * dx


def local_slsqp(h0, dx, target_sum, beta, maxiter):
    n = len(h0)

    def obj(x):
        return smooth_max_obj(x, dx, beta)

    def obj_grad(x):
        return smooth_max_grad(x, dx, beta)

    def cons_fun(x):
        return x.sum() - target_sum

    def cons_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        h_out = project_feasible(res.x, target_sum)
    except Exception:
        h_out = project_feasible(h0, target_sum)
    return h_out, c5_of_h(h_out, dx)


def compute_delta_binary(h, i, j, n):
    ms = np.arange(-(n - 1), n)
    valid_i1 = (0 <= i - ms) & (i - ms < n)
    valid_i2 = (0 <= i + ms) & (i + ms < n)
    valid_j1 = (0 <= j - ms) & (j - ms < n)
    valid_j2 = (0 <= j + ms) & (j + ms < n)
    delta = np.zeros(2 * n - 1)
    idx_i1 = i - ms[valid_i1]
    delta[valid_i1] += -(1.0 - h[idx_i1])
    idx_i2 = i + ms[valid_i2]
    delta[valid_i2] += h[idx_i2]
    idx_j1 = j - ms[valid_j1]
    delta[valid_j1] += (1.0 - h[idx_j1])
    idx_j2 = j + ms[valid_j2]
    delta[valid_j2] += -h[idx_j2]
    return delta


def binary_local_search(h0, dx, rng, max_iters=200000, patience=6000):
    n = len(h0)
    target_sum = int(round(n / 2))
    h = np.zeros(n)
    sorted_idx = np.argsort(-h0)
    h[sorted_idx[:target_sum]] = 1.0
    corrs = compute_corrs(h)
    best_c5 = corrs.max() * dx
    best_h = h.copy()
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    no_improve = 0
    temp = 0.04
    cooling = 0.9998
    for it in range(max_iters):
        if it % 3000 == 0:
            temp *= cooling
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        delta = compute_delta_binary(h, i, j, n)
        new_corrs = corrs + delta
        new_max = new_corrs.max()
        new_c5 = new_max * dx
        if new_c5 <= best_c5 + 1e-15:
            h[i], h[j] = 0.0, 1.0
            corrs = new_corrs
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            no_improve = 0
        else:
            delta_val = new_c5 - best_c5
            if temp > 1e-10 and rng.random() < np.exp(-delta_val / temp):
                h[i], h[j] = 0.0, 1.0
                corrs = new_corrs
                ones = np.where(h == 1.0)[0]
                zeros = np.where(h == 0.0)[0]
                no_improve = 0
            else:
                no_improve += 1
        if no_improve > patience:
            break
    return best_h, best_c5


def binary_steepest_descent(h0, dx, rng, max_sweeps=30):
    n = len(h0)
    target_sum = int(round(n / 2))
    h = np.zeros(n)
    sorted_idx = np.argsort(-h0)
    h[sorted_idx[:target_sum]] = 1.0
    corrs = compute_corrs(h)
    best_c5 = corrs.max() * dx
    best_h = h.copy()
    for sweep in range(max_sweeps):
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        best_swap = None
        best_new_max = corrs.max()
        for i in ones:
            for j in zeros:
                delta = compute_delta_binary(h, i, j, n)
                new_max = (corrs + delta).max()
                if new_max < best_new_max - 1e-14:
                    best_new_max = new_max
                    best_swap = (i, j)
        if best_swap is None:
            break
        i, j = best_swap
        h[i], h[j] = 0.0, 1.0
        corrs = compute_corrs(h)
        new_c5 = corrs.max() * dx
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
    return best_h, best_c5


def make_structured_start(n, target_sum, mode):
    h = np.zeros(n)
    if mode == 0:
        phi = (1.0 + np.sqrt(5.0)) / 2.0
        for k in range(int(target_sum)):
            pos = int((k * phi * n) % n)
            h[pos] = 1.0
    elif mode == 1:
        period = max(2, int(np.round(n / target_sum)))
        for k in range(int(target_sum)):
            pos = (k * period) % n
            h[pos] = 1.0
    elif mode == 2:
        # Two-scale pattern
        base_period = max(3, int(np.round(n / (target_sum * 2))))
        for k in range(int(target_sum)):
            pos = (k * base_period + (k // 2) * base_period) % n
            h[pos] = 1.0
    else:
        idx = np.random.RandomState(mode * 12345 + 7).permutation(n)
        h[idx[:int(target_sum)]] = 1.0
    return project_feasible(h, target_sum)


def remaining_time(t0, budget_s):
    return budget_s - (time.time() - t0)
