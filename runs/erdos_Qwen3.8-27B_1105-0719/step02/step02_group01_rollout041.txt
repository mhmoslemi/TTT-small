```python
def binary_greedy(h, n, max_sweeps=100):
    """Greedy local search for binary sequence."""
    h = h.copy()
    ones = set(np.where(h == 1)[0])
    zeros = set(np.where(h == 0)[0])
    
    best_c5 = max_corr(h, n) * (2.0/n)
    h_best = h.copy()
    
    for sweep in range(max_sweeps):
        improved = False
        # Try all swaps
        best_delta = 0
        best_swap = None
        for i in list(ones):
            for j in list(zeros):
                # Try swapping i and j
                old_max = max_corr(h, n)
                h[i] = 0
                h[j] = 1
                new_max = max_corr(h, n)
                # Undo
                h[i] = 1
                h[j] = 0
                delta = new_max - old_max
                if delta < best_delta - 1e-15:
                    best_delta = delta
                    best_swap = (i, j)
        if best_swap is not None:
            i, j = best_swap
            h[i] = 0
            h[j] = 1
            ones.remove(i)
            ones.add(j)
            zeros.remove(j)
            zeros.add(i)
            new_c5 = max_corr(h, n) * (2.0/n)
            if new_c5 < best_c5:
                best_c5 = new_c5
                h_best = h.copy()
            improved = True
        if not improved:
            break
    return h_best, best_c5
```

```python
def update_corr_after_swap(corr, h, i, j, n):
    """
    Update correlation array after swapping h[i] (1->0) and h[j] (0->1).
    Only lags k where i or j is in the valid range are affected.
```
