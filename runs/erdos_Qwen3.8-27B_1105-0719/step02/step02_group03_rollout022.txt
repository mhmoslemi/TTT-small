    a2 = ones[best_po[1]]
    b1 = zeros[best_pz[0]]
    b2 = zeros[best_pz[1]]
    delta = G[:, b1] + G[:, b2] - G[:, a1] - G[:, a2]
    new_corr = corr + delta
    new_max = new_corr.max()
    if new_max <= current_max + 1e-12:
        return (a1, a2, b1, b2, new_corr, new_max)
    return None


def deterministic_local_search(h_bin, dx, max_rounds=500):
    n = len(h_bin)
    h = h_bin.copy()
    corr = compute_corrs(h)
    best_c5 = corr.max() * dx
    best_h = h.copy()
    for rnd in range(max_rounds):
        G = compute_gradient_matrix(h)
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        improved = False
        result = find_best_2swap(G, corr, ones, zeros, top_k=5)
        if result is None:
            result = find_best_1swap(G, corr, ones, zeros, top_k=5)
            if result is not None:
                a, b, new_corr, new_max = result
                h[a] = 0.0
                h[b] = 1.0
                corr = new_corr
                improved = True
        else:
            a1, a2, b1, b2, new_corr, new_max = result
            h[a1] = 0.0
            h[a2] = 0.0
            h[b1] = 1.0
            h[b2] = 1.0
            corr = new_corr
            improved = True
        if improved:
            c5_val = corr.max() * dx
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h.copy()
        else:
            # Try with larger top_k
            result = find_best_2swap(G, corr, ones, zeros, top_k=10)
            if result is None:
                result = find_best_1swap(G, corr, ones, zeros, top_k=10)
            if result is not None:
                if len(result) == 5:
                    a1, a2, b1, b2, new_corr, new_max = result
                    h[a1] = 0.0
                    h[a2] = 0.0
                    h[b1] = 1.0
                    h[b2] = 1.0
                else:
                    a, b, new_corr, new_max = result
                    h[a] = 0.0
                    h[b] = 1.0
                corr = new_corr
                c5_val = corr.max() * dx
                if c5_val < best_c5:
                    best_c5 = c5_val
                    best_h = h.copy()
            else:
                break
    return best_h, best_c5


def simulated_annealing_2swap(h_bin, dx, rng, max_iters=200000, patience=15000):
    n = len(h_bin)
    target_count = int(n / 2)
    h = h_bin.copy()
    corr = compute_corrs(h)
    best_c5 = corr.max() * dx
    best_h = h.copy()
    no_improve = 0
    temp = 0.02
    cooling = 0.9998
    for it in range(max_iters):
        if it % 5000 == 0:
            temp *= cooling
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        # Random 2-for-2 swap
        a1 = ones[rng.randint(len(ones))]
        a2 = ones[rng.randint(len(ones))]
        while a2 == a1:
            a2 = ones[rng.randint(len(ones))]
        b1 = zeros[rng.randint(len(zeros))]
        b2 = zeros[rng.randint(len(zeros))]
        while b2 == b1:
            b2 = zeros[rng.randint(len(zeros))]
        G = compute_gradient_matrix(h)
        delta = G[:, b1] + G[:, b2] - G[:, a1] - G[:, a2]
        new_corr = corr + delta
        new_max = new_corr.max()
        new_c5 = new_max * dx
        delta_c5 = new_c5 - best_c5
        if delta_c5 <= 1e-15:
            h[a1] = 0.0
            h[a2] = 0.0
            h[b1] = 1.0
            h[b2] = 1.0
            corr = new_corr
            best_c5 = new_c5
            best_h = h.copy()
            no_improve = 0
        elif temp > 1e-10 and rng.random() < np.exp(-delta_c5 / temp):
            h[a1] = 0.0
            h[a2] = 0.0
            h[b1] = 1.0
            h[b2] = 1.0
            corr = new_corr
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > patience:
            break
        if it % 10000 == 0 and it > 0:
            # Periodic deterministic polish
            G = compute_gradient_matrix(h)
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            for _ in range(10):
                result = find_best_2swap(G, corr, ones, zeros, top_k=5)
                if result is None:
                    result = find_best_1swap(G, corr, ones, zeros, top_k=5)
                if result is None:
                    break
                if len(result) == 5:
                    aa1, aa2, bb1, bb2, new_corr, _ = result
                    h[aa1] = 0.0
                    h[aa2] = 0.0
                    h[bb1] = 1.0
                    h[bb2] = 1.0
                else:
                    aa, bb, new_corr, _ = result
                    h[aa] = 0.0
                    h[bb] = 1.0
                corr = new_corr
                G = compute_gradient_matrix(h)
                ones = np.where(h == 1.0)[0]
                zeros = np.where(h == 0.0)[0]
                c5_val = corr.max() * dx
                if c5_val < best_c5:
                    best_c5 = c5_val
                    best_h = h.copy()
                    no_improve = 0
    return best_h, best_c5


def smooth_max_obj(h, dx, beta):
    corrs = compute_corrs(h)
    m = corrs.max()
    return (m + np.log(np.sum(np.exp(beta * (corrs - m)))) / beta) * dx


def smooth_max_grad(h, dx, beta):
    n = len(h)
    corrs = compute_corrs(h)
    m = corrs.max()
    shifted = corrs - m
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    w /= w_sum
    G = compute_gradient_matrix(h)
    grad = G.T @ w
    return grad * dx


def slsqp_optimize(h0, dx, target_sum, beta, maxiter):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(smooth_max_obj, h0, jac=lambda x: smooth_max_grad(x, dx, beta),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        h_out = project_feasible(res.x, target_sum)
    except Exception:
        h_out = project_feasible(h0, target_sum)
    return h_out, c5_of_h(h_out, dx)


def greedy_construct(n, target_count):
    h = np.zeros(n)
    corr = compute_corrs(h)
    for step in range(int(target_count)):
        best_pos = -1
        best_max = np.inf
        G = compute_gradient_matrix(h)
        for pos in range(n):
            if h[pos] == 0:
                new_corr = corr + G[:, pos]
                new_max = new_corr.max()
                if new_max < best_max:
                    best_max = new_max
                    best_pos = pos
        h[best_pos] = 1.0
        corr = compute_corrs(h)
    return h


def remaining_time(t0, budget_s):
    return budget_s - (time.time() - t0)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    best_h = np.array(initial_h_values, dtype=float)
    n = len(best_h)
    dx = 2.0 / n
    target_sum = n / 2.0
    best_h = project_feasible(best_h, target_sum)
    best_c5 = c5_of_h(best_h, dx)

    # Phase 1: Deterministic local search from initial
    if remaining_time(t0, budget_s) > 100:
        h_bin = to_binary(best_h, n / 2)
        h_det, c5_det = deterministic_local_search(h_bin, dx, max_rounds=500)
        if c5_det < best_c5:
            best_c5 = c5_det
            best_h = h_det

    # Phase 2: SLSQP polish
    if remaining_time(t0, budget_s) > 60:
        h_opt, c5_val = slsqp_optimize(best_h, dx, target_sum, 200, 500)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_opt
        h_opt2, c5_val2 = slsqp_optimize(best_h, dx, target_sum, 1000, 800)
        if c5_val2 < best_c5:
            best_c5 = c5_val2
            best_h = h_opt2

    # Phase 3: SA with 2-for-2 moves
    if remaining_time(t0, budget_s) > 300:
        h_bin = to_binary(best_h, n / 2)
        h_sa, c5_sa = simulated_annealing_2swap(h_bin, dx, rng, max_iters=300000, patience=20000)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa

    # Phase 4: SLSQP after SA
    if remaining_time(t0, budget_s) > 80:
        h_opt, c5_val = slsqp_optimize(best_h, dx, target_sum, 500, 600)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_opt

    # Phase 5: Greedy constructive
    if remaining_time(t0, budget_s) > 200:
        h_greedy = greedy_construct(n, n / 2)
        h_g2, c5_g2 = deterministic_local_search(h_greedy, dx, max_rounds=300)
        if c5_g2 < best_c5:
            best_c5 = c5_g2
            best_h = h_g2
        if remaining_time(t0, budget_s) > 100:
            h_go, c5_go = slsqp_optimize(h_g2, dx, target_sum, 500, 600)
            if c5_go < best_c5:
                best_c5 = c5_go
                best_h = h_go

    # Phase 6: Another SA round with different params
    if remaining_time(t0, budget_s) > 200:
        h_bin = to_binary(best_h, n / 2)
        h_sa2, c5_sa2 = simulated_annealing_2swap(h_bin, dx, rng, max_iters=200000, patience=10000)
        if c5_sa2 < best_c5:
            best_c5 = c5_sa2
            best_h = h_sa2

    # Phase 7: Final deterministic polish
    if remaining_time(t0, budget_s) > 50:
        h_bin = to_binary(best_h, n / 2)
        h_fin, c5_fin = deterministic_local_search(h_bin, dx, max_rounds=300)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin

    # Phase 8: Final SLSQP
    if remaining_time(t0, budget_s) > 40:
        h_opt, c5_val = slsqp_optimize(best_h, dx, target_sum, 2000, 1000)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_opt

    # Phase 9: Try a different n (e.g., 120 or 130)
    if remaining_time(t0, budget_s) > 150:
        for n_try in [120, 130, 126]:
            if remaining_time(t0, budget_s) < 80:
                break
            if n_try == n:
                continue
            dx_t = 2.0 / n_try
            ts_t = n_try / 2.0
            x_orig = np.linspace(0.0, 2.0, len(best_h))
            x_new = np.linspace(0.0, 2.0, n_try)
            h_interp = np.interp(x_new, x_orig, best_h)
            h_interp = project_feasible(h_interp, ts_t)
            h_bin_t = to_binary(h_interp, n_try / 2)
            h_det_t, c5_det_t = deterministic_local_search(h_bin_t, dx_t, max_rounds=200)
            if c5_det_t < best_c5:
                best_c5 = c5_det_t
                best_h = h_det_t
            elif remaining_time(t0, budget_s) > 60:
                h_opt_t, c5_opt_t = slsqp_optimize(h_interp, dx_t, ts_t, 300, 400)
                if c5_opt_t < best_c5:
                    best_c5 = c5_opt_t
                    best_h = h_opt_t

    # Final cleanup
    n_b = len(best_h)
    dx_b = 2.0 / n_b
    ts_b = n_b / 2.0
    best_h = project_feasible(best_h, ts_b)
    best_c5 = c5_of_h(best_h, dx_b)

    return best_h, best_c5, len(best_h)
