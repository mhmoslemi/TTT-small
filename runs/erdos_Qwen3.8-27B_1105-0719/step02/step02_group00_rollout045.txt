import numpy as np

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx])
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    # Final adjustment
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
    return np.clip(h, 0.0, 1.0)

def compute_gradient(h, n, dx, lag):
    """Compute gradient of P_lag with respect to h."""
    g = np.zeros(n)
    if lag == 0:
        g = dx * (1.0 - 2.0 * h)
    elif lag > 0:
        # P = dx * sum_{i=0}^{n-lag-1} h[i+lag] * (1-h[i])
        # dP/dh[i+lag] = dx * (1-h[i]) for i=0..n-lag-1
        # dP/dh[i] = -dx * h[i+lag] for i=0..n-lag-1
        g[lag:n] += dx * (1.0 - h[0:n-lag])
        g[0:n-lag] += -dx * h[lag:n]
    else:
        e = -lag
        # P = dx * sum_{i=0}^{n-e-1} h[i] * (1-h[i+e])
        # dP/dh[i] = dx * (1-h[i+e]) for i=0..n-e-1
        # dP/dh[i+e] = -dx * h[i] for i=0..n-e-1
        g[0:n-e] += dx * (1.0 - h[e:n])
        g[e:n] += -dx * h[0:n-e]
    return g

    """Log-sum-exp approximation of max."""
    p = profile - np.max(profile)
    return np.max(profile) + np.log(np.sum(np.exp(tau * p))) / tau

def trust_region_lp(h, active_indices, G_active, p_active, t0, radius):
    """Solve trust-region LP subproblem using cvxpy."""
    n = len(h)
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    
    # Trust region: box constraint |h - h0| <= radius
    constraints = [
        h_var >= h - radius,
        h_var <= h + radius,
        h_var >= 0,
        h_var <= 1,
        cp.sum(h_var) == n / 2.0
    ]
    
    # Linearized active constraints: t >= p_k + G_k . (h - h0)
    # which is t - G_k . h >= p_k - G_k . h0
    for i, m in enumerate(active_indices):
        # G_active[i] is gradient of P_m at h
        # P_m(h_new) ≈ p_active[i] + G_active[i] @ (h_new - h)
        # constraint: t >= p_active[i] + G_active[i] @ (h_new - h)
        # i.e., t - G_active[i] @ h_new >= p_active[i] - G_active[i] @ h
        constraints.append(t_var - G_active[i] @ h_var >= p_active[i] - G_active[i] @ h)
    
    obj = cp.Minimize(t_var)
    prob = cp.Problem(obj, constraints)
    
    try:

        if h_var.value is not None and np.isfinite(h_var.value).all():
            return h_var.value, t_var.value

        if h_var.value is not None and np.isfinite(h_var.value).all():
            return h_var.value, t_var.value
    except Exception:
        pass
    
    return None, None

def smoothed_lbfgs(h0, tau, maxiter=200):

    n = len(h0)

        p = profile - np.max(profile)
        val = np.max(profile) + np.log(np.sum(np.exp(tau * p))) / tau
        # Penalty for sum constraint

        # Penalty for bounds (soft)
        penalty += 1e4 * np.sum(np.maximum(0, x - 1)**2) + 1e4 * np.sum(np.maximum(0, -x)**2)
        return val + penalty
    
    x = h0.copy()

    t_var = cp.Variable()
    constraints = [
        h_var >= 0,
        h_var <= 1,
        cp.sum(h_var) == n / 2.0
    ]
    for i, m in enumerate(active):
        lag = m - (n - 1)
        g = grad_lag(h0, n, dx, lag)
        constraints.append(t_var >= p_active[i] + g @ (h_var - h0))
    obj = cp.Minimize(t_var + eta * cp.sum_squares(h_var - h0))
    prob = cp.Problem(obj, constraints)
    for solver_name in [cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver_name, max_iters=20000, eps=1e-7)
            if h_var.value is not None and np.isfinite(h_var.value).all():
                return h_var.value
        except Exception:
            continue
    return None

def trust_region_refine(h, max_iters=50, eta_init=0.5, eta_min=0.01, eta_max=50.0):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best_c = exact_score(h)
    eta = eta_init
    stagnation = 0
    for it in range(max_iters):
        profile = compute_profile(h)
        c = np.max(profile)
        if c < best_c:
            best_c = c
            stagnation = 0
        else:
            stagnation += 1
        if stagnation > 15:
            break
        active, weights, c_max = active_set_weights(profile)
        p_act = profile[active]
        h_new = solve_qp_subproblem(h, active, weights, p_act, eta)
        if h_new is None:
            eta *= 2
            if eta > eta_max:
                break
            continue
        h_new = repair_feasible(h_new)
        c_new = exact_score(h_new)
        if c_new < best_c - 1e-14:
            h = h_new
            best_c = c_new
            eta = max(eta * 0.7, eta_min)
        else:
            eta = min(eta * 1.5, eta_max)
            if eta > eta_max:
                break
    return h, best_c

def smoothed_descent(h, tau=50.0, maxiter=300, mu_init=100.0):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0

    def obj(x):
        profile = np.correlate(x, 1 - x, mode="full") * dx
        p = profile - np.max(profile)
        val = np.max(profile) + np.log(np.sum(np.exp(tau * p))) / tau
        val += mu_init * (np.sum(x) - target) ** 2
        return val

    try:
        res = minimize(obj, h, method='L-BFGS-B',
                       bounds=[(0, 1)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-8})
        h_new = repair_feasible(res.x)
        return h_new
    except Exception:
        return None

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

def symmetrize(h):
    return (h + h[::-1]) / 2.0

def subsampled_gradient_descent(h, max_iters=200, subsample=10):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best_c = exact_score(h)
    alpha = 0.5
    for it in range(max_iters):
        profile = compute_profile(h)
        c = np.max(profile)
        active, w, _ = active_set_weights(profile, max_active=subsample)
        G = np.stack([grad_lag(h, n, dx, m - (n-1)) for m in active])
        grad = G.T @ w
        grad -= grad.mean()
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad /= gnorm
        improved = False
        for a in [alpha, alpha*0.5, alpha*0.25, alpha*0.1]:
            h_try = repair_feasible(h - a * grad)
            s = exact_score(h_try)
            if s < best_c - 1e-14:
                h = h_try
                best_c = s
                alpha = min(a * 1.3, 2.0)
                improved = True
                break
        if not improved:
            alpha *= 0.6
            if alpha < 1e-6:
                break
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s - (time.perf_counter() - t_start)

    # Load parent
    parent = None
    try:
        parent = np.array(initial_h_values, dtype=np.float64)
    except NameError:
        pass
    except Exception:
        pass
    if parent is None or len(parent) < 4:
        parent = np.full(70, 0.5)
    parent = repair_feasible(parent)
    best_h = parent.copy()
    best_c = exact_score(parent)
    n = len(best_h)

    # Symmetrize
    h_sym = repair_feasible(symmetrize(parent))
    c_sym = exact_score(h_sym)
    if c_sym < best_c - 1e-13:
        best_h = h_sym
        best_c = c_sym
        n = len(best_h)

    # Phase 1: Trust-region QP on parent dimension
    if time_left() > 120:
        h_tr, c_tr = trust_region_refine(best_h, max_iters=80, eta_init=0.3, eta_min=0.005, eta_max=20.0)
        if c_tr < best_c - 1e-13:
            best_h, best_c = h_tr, c_tr

    # Phase 2: Smoothed descent
    if time_left() > 100:
        for tau in [30.0, 60.0, 150.0]:
            if time_left() < 30:
                break
            h_sm = smoothed_descent(best_h, tau=tau, maxiter=200, mu_init=200.0)
            if h_sm is not None:
                c_sm = exact_score(h_sm)
                if c_sm < best_c - 1e-13:
                    best_h, best_c = h_sm, c_sm

    # Phase 3: More trust-region with tighter parameters
    if time_left() > 100:
        h_tr, c_tr = trust_region_refine(best_h, max_iters=100, eta_init=0.1, eta_min=0.002, eta_max=10.0)
        if c_tr < best_c - 1e-13:
            best_h, best_c = h_tr, c_tr

    # Phase 4: Subsampled gradient descent (different active set sizes)
    if time_left() > 80:
        for subs in [8, 15, 30]:
            if time_left() < 25:
                break
            h_sg, c_sg = subsampled_gradient_descent(best_h, max_iters=150, subsample=subs)
            if c_sg < best_c - 1e-13:
                best_h, best_c = h_sg, c_sg

    # Phase 5: Dimension exploration
    if time_left() > 100:
        n_cur = len(best_h)
        dims = []
        for d in [n_cur-2, n_cur-4, n_cur+2, n_cur+4, n_cur+6, n_cur-8, n_cur*2//1, n_cur//1+2]:
            d2 = d if d % 2 == 0 else d + 1
            if 30 <= d2 <= 150 and d2 != n_cur:
                dims.append(d2)
        dims = sorted(set(dims), key=lambda d: abs(d - n_cur))

        for new_n in dims[:6]:
            if time_left() < 40:
                break
            h_interp = interpolate_to_n(best_h, new_n)
            c_interp = exact_score(h_interp)
            if c_interp < best_c - 1e-12:
                best_h, best_c = h_interp, c_interp
            # Refine at new dimension
            h_tr, c_tr = trust_region_refine(h_interp, max_iters=50, eta_init=0.2, eta_min=0.005, eta_max=10.0)
            if c_tr < best_c - 1e-13:
                best_h, best_c = h_tr, c_tr
            h_sg, c_sg = subsampled_gradient_descent(best_h, max_iters=100, subsample=12)
            if c_sg < best_c - 1e-13:
                best_h, best_c = h_sg, c_sg

    # Phase 6: Random restarts with small perturbations
    if time_left() > 80:
        for eps in [0.003, 0.007, 0.015, 0.03, 0.06]:
            if time_left() < 20:
                break
            for _ in range(2):
                if time_left() < 10:
                    break
                z = rng.randn(len(best_h))
                z -= z.mean()
                h_pert = repair_feasible(best_h + eps * z)
                h_tr, c_tr = trust_region_refine(h_pert, max_iters=40, eta_init=0.15, eta_min=0.005, eta_max=8.0)
                if c_tr < best_c - 1e-13:
                    best_h, best_c = h_tr, c_tr

    # Phase 7: Final intensive trust-region polish
    if time_left() > 60:
        h_tr, c_tr = trust_region_refine(best_h, max_iters=100, eta_init=0.05, eta_min=0.001, eta_max=5.0)
        if c_tr < best_c - 1e-13:
            best_h, best_c = h_tr, c_tr

    # Phase 8: Very fine gradient steps
    if time_left() > 40:
        n_f = len(best_h)
        dx_f = 2.0 / n_f
        for _ in range(80):
            if time_left() < 5:
                break
            profile = compute_profile(best_h)
            c = np.max(profile)
            active, w, _ = active_set_weights(profile, beam_frac=0.002, max_active=12)
            G = np.stack([grad_lag(best_h, n_f, dx_f, m - (n_f-1)) for m in active])
            grad = G.T @ w
            grad -= grad.mean()
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break
            grad /= gnorm
            improved = False
            for a in [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]:
                h_try = repair_feasible(best_h - a * grad)
                s = exact_score(h_try)
                if s < best_c - 1e-14:
                    best_h, best_c = h_try, s
                    improved = True
                    break
            if not improved:
                break

    # Phase 9: Final cvxpy QP with very small eta
    if time_left() > 30:
        h_tr, c_tr = trust_region_refine(best_h, max_iters=50, eta_init=0.02, eta_min=0.001, eta_max=2.0)
        if c_tr < best_c - 1e-14:
            best_h, best_c = h_tr, c_tr

    # Phase 10: Final smoothed pass
    if time_left() > 20:
        h_sm = smoothed_descent(best_h, tau=200.0, maxiter=100, mu_init=500.0)
        if h_sm is not None:
            c_sm = exact_score(h_sm)
            if c_sm < best_c - 1e-14:
                best_h, best_c = h_sm, c_sm

    # Final validation
    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)

    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            final_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 2:
            final_c = float(eval_result[0])
    except Exception:
        pass

    return best_h, final_c, len(best_h)
