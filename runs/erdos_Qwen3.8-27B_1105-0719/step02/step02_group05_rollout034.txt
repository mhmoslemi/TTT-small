Actually, let me try a completely different approach for the refinement: ** iterated projected gradient on the max functional**.

The idea: 
- F(h) = max_k F_k(h) where F_k(h) = sum_i h[i](1-h[i+k-dx]) * dx
- At the current h, find the set of active k's (those achieving the max)
- Compute the subgradient: average of gradients of active F_k's
- Take a projected gradient step
- Repeat

This is essentially the subgradient method for the max of convex functions (which, as I noted, might not be convex, but locally it works like it).

def max_gradient_step(h, n, alpha, max_iter=1000):
    """Iterated projected gradient on the max correlation functional."""
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    
    for it in range(max_iter):
        corrs = compute_corrs(h) * dx
        c5 = corrs.max()
        # Find active set (within tolerance)
        tol = 1e-6
        active = np.where(corrs >= c5 - tol)[0]
        
        # Compute gradient as average of active gradients
        grad = np.zeros(n)
        for k in active:
            d = k - (n-1)
            if d >= 0:
                if d == 0:
                    grad += (1.0 - 2.0*h)
                else:
                    grad[:n-d] += (1.0 - h[d:])
                    grad[d:] += (-h[:n-d])
            else:
                dd = -d
                grad[dd:] += (1.0 - h[:n-dd])
                grad[:n-dd] += (-h[dd:])
        grad /= len(active)
        grad *= dx
        
        # Projected gradient step
        h_new = h - alpha * grad
        h_new = project_feasible(h_new, target)
        
        # Check improvement
        c5_new = exact_score(h_new)[0]
        if c5_new < c5:
            h = h_new
        else:
            alpha *= 0.5
            if alpha < 1e-10:
                break
    
    return h, exact_score(h)[0]

from scipy.special import logsumexp
import time

    corrs = compute_corrs(h)
    val = logsumexp(beta * corrs) / beta * dx
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)

    if ws < 1e-300:

    else:
        w /= ws

def smooth_optimize(h0, n, beta, target, maxiter=500):

        return smooth_max_val_grad(x, n, beta)[0]

        return smooth_max_val_grad(x, n, beta)[1]

    """Gradient of the d-th correlation w.r.t. h."""
    g = np.zeros(n)
    if d >= 0:
        if d == 0:
            g[:] = 1.0 - 2.0 * h
        elif d < n:
            g[:n - d] += (1.0 - h[d:])
            g[d:] += (-h[:n - d])
    else:
        dd = -d
        if dd < n:
            g[dd:] += (1.0 - h[:n - dd])
            g[:n - dd] += (-h[dd:])
    return g

            best_h = h.copy()

        active = np.where(corrs >= c5 - tol)[0]
