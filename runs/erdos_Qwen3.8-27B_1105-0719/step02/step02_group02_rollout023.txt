    # h[idx+offset] appears as second factor with -h[idx]

from scipy.optimize import trust_constr, NonlinearConstraint, LinearConstraint

def active_set_solve(h0, dx, ts, n_active, maxiter, t0, deadline):
    n = len(h0)
    # Current correlations
    corrs = np.correlate(h0, 1.0 - h0, mode='full')
    cmax = corrs.max()
    
    # Select active constraints: those within 5% of max
    threshold = cmax - 0.05 * cmax
    active = np.where(corrs >= threshold)[0]
    if len(active) > n_active:
        active = active[np.argsort(corrs[active])[-n_active:]]
    
    # Define constraint: corr[k] * dx <= t for active k
    # Variables: [h (n), t (1)]
    def constraint_fun(x):
        h = x[:n]
        t = x[n]
        corrs_a = np.correlate(h, 1.0 - h, mode='full')[active]
        return corrs_a * dx - t  # should be <= 0
    
    def constraint_jac(x):
        h = x[:n]
        J = np.zeros((len(active), n + 1))
        for i, k in enumerate(active):
            J[i, :n] = grad_lag(h, k, n) * dx
            J[i, n] = -1.0
        return J
    
    nonlin_con = NonlinearConstraint(constraint_fun, -np.inf, 0.0, 
                                      jac=constraint_jac)
    
    # Equality: sum(h) = ts
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    lin_con = LinearConstraint(A_eq, ts, ts)
    
    # Bounds
    bounds = [(0, 1)] * n + [(0, 2.0)]  # t is unbounded above (well, [0, 2])
    
    def objective(x):
        return x[n]  # minimize t
    
    def objective_grad(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g
    
    x0 = np.concatenate([h0, [cmax * dx]])
    
    try:
        res = trust_constr(objective, x0, bounds=bounds,
                           constraints=[nonlin_con, lin_con],
                           jac=objective_grad,
                           maxiter=maxiter)
        h_out = project_feasible(res.x[:n], ts)
        return h_out
    except:
