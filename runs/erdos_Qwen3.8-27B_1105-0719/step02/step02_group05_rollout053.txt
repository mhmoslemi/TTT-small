def subgradient_descent(h0, n, target_sum, max_iter=10000, learning_rate=0.01):
    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_corrs(h).max() * dx
    
    for t in range(max_iter):
        corrs = compute_corrs(h)
        f_val = corrs.max() * dx
        if f_val < best_c5:
            best_c5 = f_val
            best_h = h.copy()
        
        # Find active set
        threshold = f_val - 1e-4  # lags within 1e-4 of max
        active = np.where(corrs * dx > threshold)[0]
        
        # Compute subgradient
        G = np.zeros(n)
        for k in active:
            d = k - (n - 1)
            G += grad_corr_d(h, d, n)
        G /= len(active)
        
        # Step
        alpha = learning_rate / (1 + t * 0.001)
        h_new = h - alpha * G * dx
        h = project_feasible(h_new, target_sum)
    
    return best_h, best_c5
```

    d_neg = np.arange(0, min(n-1-pos, n-1) + 1)

    # d < 0 part: k = n-1-dd for dd = 1, ..., n-1
    # Positive: (1-h[pos-dd]) for dd = 1, ..., min(pos, n-1)

Actually, one more optimization: instead of computing `compute_corrs(h)` (which is O(n²)) every step, I can maintain it incrementally. When I change h[i] from old to new, the change in corr_k is:
Δcorr_k = (new - old) * ∂corr_k/∂h[i] = (new - old) * g_i[k]

And similarly for h[j]. So I can update the correlations in O(n) instead of O(n²).

def pairwise_coord_descent_fast(h0, n, target_sum, max_rounds=100000, seed=42, time_limit=200):
    dx = 2.0 / n
    m = 2*n - 1
    h = h0.copy()
    corrs = compute_corrs(h)  # O(n²)
    f0 = corrs.max() * dx
    rng = np.random.RandomState(seed)
    t_start = time.time()
    
    no_improve_count = 0
    max_no_improve = 5000
    
    for rnd in range(max_rounds):
        if rnd % 1000 == 0 and time.time() - t_start > time_limit:
            break
        if no_improve_count > max_no_improve:
            break
        
        i = rng.randint(n)
        j = rng.randint(n)
        while j == i:
            j = rng.randint(n)
        i, j = min(i,j), max(i,j)
        
        s = h[i] + h[j]
        x_lo = max(0.0, s - 1.0)
        x_hi = min(1.0, s)
        if x_hi - x_lo < 1e-10:
            continue
        
        # Compute gradients
        g_i = grad_at_position_v2(h, i, n)
        g_j = grad_at_position_v2(h, j, n)
        diff_g = g_i - g_j
        
        # f(x) = max_k (corrs[k] + diff_g[k] * (x - h[i]))
        #     = max_k ((corrs[k] - diff_g[k]*h[i]) + diff_g[k]*x)
        a = corrs - diff_g * h[i]
        
        # Find minimum of max(a + diff_g * x) over x ∈ [x_lo, x_hi]
        # Use the fact that this is convex piecewise linear
        # The minimum is at an intersection of two "upper envelope" lines
        # For efficiency, use a fine grid
        num_cand = 20
        candidates = np.linspace(x_lo, x_hi, num_cand)
        vals = np.max(a[None,:] + diff_g[:,None] * candidates[None,:], axis=1)
        best_idx = np.argmin(vals)
        best_x = candidates[best_idx]
        best_val = vals[best_idx] * dx
        
        if best_val < f0 - 1e-13:
            # Update corrs incrementally
            delta_i = best_x - h[i]
            delta_j = (s - best_x) - h[j]  # = -delta_i
            corrs = corrs + diff_g * delta_i  # since h[j] changes by -delta_i
            # Wait: corr_k changes by g_i[k]*delta_i + g_j[k]*delta_j = g_i[k]*delta_i + g_j[k]*(-delta_i) = (g_i[k]-g_j[k])*delta_i
            # So: corrs += diff_g * delta_i. ✓
            h[i] = best_x
            h[j] = s - best_x
            f0 = best_val
            no_improve_count = 0
        else:
            no_improve_count += 1
    
    return h, compute_corrs(h).max() * dx
```

        best_val = vals[best_idx] * dx

        if best_val < f0 - 1e-14:
            delta = best_x - h[i]
            corrs = corrs + diff_g * delta
            h[i] = best_x
            h[j] = s - best_x
            f0 = best_val

    return h, compute_corrs(h).max() * dx


def try_different_n(h_best, n_best, c5_best, time_limit=120.0):
    results = (h_best, c5_best, n_best)
    t_start = time.time()

    for n_try in [128, 130, 132, 124, 120, 112, 104, 96]:
        if time.time() - t_start > time_limit:
            break
        ts = n_try / 2.0
        x_orig = np.linspace(0.0, 2.0, n_best)
        x_new = np.linspace(0.0, 2.0, n_try)
        h_interp = np.interp(x_new, x_orig, h_best)
        h_interp = project_feasible(h_interp, ts)
        s_interp = exact_score(h_interp)

        h_opt, c5_opt = pairwise_coord_descent(h_interp, n_try, ts,
                                               max_rounds=80000, seed=n_try,
                                               time_limit=max(5, time_limit - (time.time() - t_start)))
        if c5_opt < results[1] - 1e-12:
            results = (h_opt, c5_opt, n_try)

    return results


def run(seed=42, budget_s=1100, **kwargs):
    t_global_start = time.time()

    def elapsed():
        return time.time() - t_global_start

    def remaining():
        return budget_s - elapsed()

    h_parent = np.array(initial_h_values, dtype=float)
    n = len(h_parent)
    target_sum = n / 2.0
    h = project_feasible(h_parent, target_sum)
    best_h = h.copy()
    best_c5 = exact_score(h)

    # Phase 1: Subgradient warm-up
    if remaining() > 100:
        h_sg, c5_sg = subgradient_optimize(h, n, target_sum, max_iter=3000, time_limit=40.0)
        if c5_sg < best_c5 - 1e-14:
            best_c5 = c5_sg
            best_h = h_sg.copy()
            h = h_sg.copy()

    # Phase 2: Main pairwise coordinate descent
    if remaining() > 80:
        t_lim = min(250.0, remaining() - 60.0)
        h_pw, c5_pw = pairwise_coord_descent(h, n, target_sum,
                                             max_rounds=200000, seed=seed,
                                             time_limit=t_lim, num_cand=40)
        if c5_pw < best_c5 - 1e-14:
            best_c5 = c5_pw
            best_h = h_pw.copy()
            h = h_pw.copy()

    # Phase 3: Focused pairwise (target top lags)
    if remaining() > 60:
        t_lim = min(150.0, remaining() - 40.0)
        h_foc, c5_foc = pairwise_focused(h, n, target_sum, time_limit=t_lim, seed=seed + 1)
        if c5_foc < best_c5 - 1e-14:
            best_c5 = c5_foc
            best_h = h_foc.copy()
            h = h_foc.copy()

    # Phase 4: Another round of general pairwise with different seed
    if remaining() > 60:
        t_lim = min(150.0, remaining() - 40.0)
        h_pw2, c5_pw2 = pairwise_coord_descent(h, n, target_sum,
                                               max_rounds=200000, seed=seed + 100,
                                               time_limit=t_lim, num_cand=35)
        if c5_pw2 < best_c5 - 1e-14:
            best_c5 = c5_pw2
            best_h = h_pw2.copy()
            h = h_pw2.copy()

    # Phase 5: Try different n values
    if remaining() > 80:
        h_best, c5_best, n_best = try_different_n(best_h, len(best_h), best_c5,
                                                  time_limit=min(120.0, remaining() - 40.0))
        if c5_best < best_c5 - 1e-12:
            best_c5 = c5_best
            best_h = h_best.copy()

    # Phase 6: Final intensive pairwise on the best solution found
    n_final = len(best_h)
    ts_final = n_final / 2.0
    if remaining() > 30:
        t_lim = min(remaining() - 10.0, 100.0)
        h_final, c5_final = pairwise_coord_descent(best_h, n_final, ts_final,
                                                   max_rounds=200000, seed=seed + 200,
                                                   time_limit=t_lim, num_cand=50)
        if c5_final < best_c5 - 1e-14:
            best_c5 = c5_final
            best_h = h_final.copy()

    # Phase 7: One more focused pass
    if remaining() > 20:
        t_lim = min(remaining() - 5.0, 50.0)
        h_f2, c5_f2 = pairwise_focused(best_h, n_final, ts_final,
                                       time_limit=t_lim, seed=seed + 300)
        if c5_f2 < best_c5 - 1e-14:
            best_c5 = c5_f2
            best_h = h_f2.copy()

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    n_out = len(best_h)
    ts_out = n_out / 2.0
    best_h = project_feasible(best_h, ts_out)
    final_score = exact_score(best_h)
    if final_score < best_c5:
        best_c5 = final_score

    return best_h, best_c5, n_out
