def active_subgradient_descent(h, max_iters=500, alpha_init=0.01):

def nesterov_descent(h, max_iters=300):
    """Nesterov-accelerated projected gradient descent."""
    n = len(h)
    dx = 2.0 / n
    h_cur = h.copy()
    y = h.copy()
    best_h = h_cur.copy()
    best_c = exact_score(h_cur)
    
    for t in range(1, max_iters + 1):
        # Compute gradient at y
        profile = compute_profile(y)
        c = np.max(profile)
        K = min(8, 2*n-1)
        thresh = np.partition(profile, -K)[-K]
        active = np.where(profile >= thresh - 1e-10)[0]
        G = build_gradient_matrix(y, n, dx)
        G_active = G[active, :]
        weights = np.ones(len(active)) / len(active)
        grad = G_active.T @ weights
        grad -= grad.mean()
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Step size: gamma / (t+1)
        gamma = 5.0
        alpha = gamma / (t * t + 1)
        
        # Nesterov momentum
        beta_t = (t - 1) / (t + 1) if t > 1 else 0.0
        
        h_next = project_feasible(y - alpha * grad)
        y = h_next + beta_t * (h_next - h_cur)
        y = np.clip(y, 0, 1)  # Keep y in bounds for next iteration
        
        h_cur = h_next
        s = exact_score(h_cur)
        if s < best_c:
            best_c = s
            best_h = h_cur.copy()
    
    return best_h, best_c

def binary_sa_init(n, rng, n_iter=5000):
    """Generate a good binary starting point using simulated annealing."""
    # Start with random balanced binary
    h_bits = np.zeros(n)
    ones = rng.choice(n, n//2, replace=False)
    h_bits[ones] = 1.0
    
    def bit_score(bits):
        return float(np.max(np.correlate(bits, 1-bits, mode="full")))
    
    current = h_bits.copy()
    current_score = bit_score(current)
    best_bits = current.copy()
    best_score = current_score
    
    T_init = 2.0
    T_final = 0.01
    
    for it in range(n_iter):
        T = T_init * (T_final/T_init) ** (it / n_iter)
        
        # Pick two positions: one with 1, one with 0
        ones_pos = np.where(current == 1)[0]
        zeros_pos = np.where(current == 0)[0]
        if len(ones_pos) == 0 or len(zeros_pos) == 0:
            break
        p = ones_pos[rng.randint(len(ones_pos))]
        q = zeros_pos[rng.randint(len(zeros_pos))]
        
        # Swap
        current[p] = 0
        current[q] = 1
        new_score = bit_score(current)
        delta = new_score - current_score
        
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_bits = current.copy()
        else:
            # Revert
            current[p] = 1
            current[q] = 0
    
    return best_bits

def epigraph_solve(h_start, maxiter=200, tol=1e-10):
    """SLSQP epigraph formulation."""
    n = len(h_start)
    dx = 2.0 / n
    M = 2 * n - 1
    c0 = exact_score(h_start)
    t0 = c0 + 1e-4
    
    def obj(x):
        return x[-1]
    
    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    
    def eq_fun(x):
        return x[:-1].sum() - n / 2.0
    
    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g
    
    def ineq_fun(x):
        h = x[:-1]
        t = x[-1]
        profile = np.correlate(h, 1 - h, mode="full") * dx
        return t - profile
    
    def ineq_jac(x):
        h = x[:-1]
        G = build_gradient_matrix(h, n, dx)
        J = np.zeros((M, n + 1))
        J[:, :-1] = -G
        J[:, -1] = 1.0
        return J
    
    x0 = np.concatenate([h_start, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 2.0)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': tol})
        if res.x is not None and np.isfinite(res.x).all():
