```python
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time

# Global state for optimizer callbacks (avoids lambdas/closures)
_g_n = 0
_g_beta = 1.0
_g_mode = 'sym'  # 'sym' or 'full'
_g_target = 0.0


def _obj_func(x):
    if _g_mode == 'sym':
        return _sym_obj(x, _g_n, _g_beta)[0]
    else:
        return _full_obj(x, _g_n, _g_beta, _g_target)[0]


def _obj_grad(x):
    if _g_mode == 'sym':
        return _sym_obj(x, _g_n, _g_beta)[1]
    else:
        return _full_obj(x, _g_n, _g_beta, _g_target)[1]


def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)


def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full').max() * dx


def corr_vals(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx


def make_sym_h(q, n):
    m = n // 2
    q = np.clip(q, 0.0, 1.0)
    h = np.empty(n)
    h[:m] = q
    h[m:] = 1.0 - q[::-1]
    return h


def make_q_from_h(h, n):
    m = n // 2
    q = 0.5 * (h[:m] + 1.0 - h[n-1:n-1-m:-1])
    return np.clip(q, 0.0, 1.0)


def _corr_grad_h(h, w):
    """Compute sum_k w[k] * grad_h(corr_k) for correlation vector h."""
    n = len(h)
    grad = np.zeros(n)
    for k in range(len(w)):
        if w[k] < 1e-15:
            continue
        d = k - (n - 1)
        i_lo = max(0, -d)
        i_hi = min(n - 1, n - 1 - d)
        if i_lo > i_hi:
            continue
        idx = np.arange(i_lo, i_hi + 1)
        grad[idx] += w[k] * (1.0 - h[idx + d])
        grad[idx + d] += w[k] * (-h[idx])
    return grad


def _sym_obj(x, n, beta):
    m = n // 2
    h = make_sym_h(x, n)
    corrs = corr_vals(h)
    obj = logsumexp(beta * corrs) / beta
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    ws = w.sum()
    if ws < 1e-30:
        w = np.ones(len(corrs)) / len(corrs)
    else:
        w /= ws
    dx = 2.0 / n
    grad_h = _corr_grad_h(h, w)
    grad_q = (grad_h[:m] - grad_h[n-1:n-1-m:-1]) * dx
    return obj, grad_q


def _full_obj(x, n, beta, target_sum):
    """Full-space objective with h[n-1] eliminated: sum(h[0:n-1]) = target_sum - x[n-1]"""
    h = np.empty(n)
    h[:n-1] = x
    h[n-1] = target_sum - x.sum()
    h = np.clip(h, 0.0, 1.0)
    corrs = corr_vals(h)
    obj = logsumexp(beta * corrs) / beta
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    ws = w.sum()
    if ws < 1e-30:
        w = np.ones(len(corrs)) / len(corrs)
    else:
        w /= ws
    dx = 2.0 / n
    grad_h = _corr_grad_h(h, w) * dx
    # grad w.r.t. x[i] = grad_h[i] - grad_h[n-1] for i < n-1
    grad_x = grad_h[:n-1] - grad_h[n-1]
    return obj, grad_x


def subgradient_polish(h, n_iters, step_init=0.01, tol=0.005):
    """Polish using subgradient of the exact max objective."""
    n = len(h)
    target_sum = n / 2.0
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = exact_score(h_cur)
    step = step_init

    for it in range(n_iters):
        h_feas = project_feasible(h_cur, target_sum)
        corrs = corr_vals(h_feas)
        max_c = corrs.max()
        # Active lags: within tol relative to max
        thresh = max_c - tol * max_c
        active = np.where(corrs >= thresh)[0]
        if len(active) == 0:
            active = np.array([np.argmax(corrs)])
        n_lags = len(corrs)
        w = np.zeros(n_lags)
        w[active] = 1.0 / len(active)
        dx = 2.0 / n
        grad_h = _corr_grad_h(h_feas, w) * dx
        h_cur = h_feas - step * grad_h
        h_cur = project_feasible(h_cur, target_sum)
        score = exact_score(h_cur)
        if score < best_score - 1e-12:
            best_score = score
            best_h = h_cur.copy()
        # Adaptive step
        if (it + 1) % 100 == 0:
            if score > best_score:
                step *= 0.7
            else:
                step = min(step * 1.05, 0.1)
        if best_score < 0.37:
            break

    return best_h, best_score


def smooth_sym_lbfgs(q0, n, beta, maxiter=300, ftol=1e-18):
    _g_n = n
    _g_beta = beta
    _g_mode = 'sym'
    m = n // 2
    try:
        res = minimize(_obj_func, q0, jac=_obj_grad, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * m,
                       options={'maxiter': maxiter, 'ftol': ftol})
        h_new = make_sym_h(res.x, n)
        c5 = exact_score(h_new)
        return res.x, c5
    except Exception:
        return None, None


def smooth_full_lbfgs(h0, n, beta, maxiter=300, ftol=1e-18):
    target_sum = n / 2.0
    _g_n = n
    _g_beta = beta
    _g_mode = 'full'
    _g_target = target_sum
    h_feas = project_feasible(h0, target_sum)
    x0 = h_feas[:n-1].copy()
    try:
        res = minimize(_obj_func, x0, jac=_obj_grad, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * (n-1),
                       options={'maxiter': maxiter, 'ftol': ftol})
        h_full = np.empty(n)
        h_full[:n-1] = res.x
        h_full[n-1] = target_sum - res.x.sum()
        h_full = np.clip(h_full, 0.0, 1.0)
        h_full = project_feasible(h_full, target_sum)
        c5 = exact_score(h_full)
        return h_full, c5
    except Exception:
        return None, None


def trial_optimize(h0, n, rng, budget_check):
    """Quick optimization from a starting point."""
    target_sum = n / 2.0
    h_feas = project_feasible(h0, target_sum)
    best_h = h_feas.copy()
    best_c5 = exact_score(best_h)

    if budget_check() < 15:
        return best_h, best_c5

    # Symmetric optimization
    m = n // 2
    q0 = make_q_from_h(best_h, n)
    for beta in [100, 300, 800]:
        if budget_check() < 10:
            break
        q_new, c5_new = smooth_sym_lbfgs(q0, n, beta, maxiter=200)
        if q_new is not None and c5_new < best_c5 - 1e-10:
            best_c5 = c5_new
            best_h = make_sym_h(q_new, n)
            q0 = q_new

    # Subgradient polish
    if budget_check() > 20:
        h_new, c5_new = subgradient_polish(best_h, 500, step_init=0.02)
        if c5_new < best_c5 - 1e-10:
            best_c5 = c5_new
            best_h = h_new

    return best_h, best_c5
```
