```python
def lp_trust_region(q, active_k, radius):
    """Solve LP trust region subproblem.
    Minimize tau subject to:
    corr_k(q) + dx * (grad_k . dq) <= tau for k in active_k
    max(-q[r], -radius) <= dq[r] <= min(1-q[r], radius)
    """
    m = len(q)
    n = 2 * m
    h = build_sym_h(q)
    dx = 2.0 / n
    
    # Variables: [dq_0, ..., dq_{m-1}, tau]
    # c = [0, ..., 0, 1]
    c = np.zeros(m + 1)
    c[m] = 1.0
    
    A_ub = []
    b_ub = []
    
    for k in active_k:
        gq = corr_grad_q(h, k)
        cv = corr_value(h, k) * dx
        row = np.zeros(m + 1)
        row[:m] = dx * gq
        row[m] = -1.0
        A_ub.append(row)
        b_ub.append(-cv)  # dx * grad . dq - tau <= -cv
    
    bounds = []
    for r in range(m):
        lo = max(-q[r], -radius)
        hi = min(1.0 - q[r], radius)
        bounds.append((lo, hi))
    bounds.append((None, None))  # tau unbounded
    
    res = linprog(c, A_ub=np.array(A_ub), b_ub=np.array(b_ub), bounds=bounds, method='highs')
    
    if res.status != 0:
        return None, res
    dq = res.x[:m]
    return dq, res
```

```python
def coord_1d_exact(q, r):
    """Find optimal q[r] by solving max of linear functions exactly."""
    m = len(q)
    n = 2 * m
    h = build_sym_h(q)
    dx = 2.0 / n
    L = 2 * n - 1  # length of full correlation
    
    coeffs = np.array([corr_grad_q(h, k)[r] for k in range(L)])
    # corr_k = coeff[k] * q[r] + (corr_k - coeff[k] * q[r])
    # b_k = corr_k - coeff[k] * q[r]
    corrs = np.array([corr_value(h, k) * dx for k in range(L)])
    b = corrs - coeffs * q[r]
    
    # f(t) = max_k (coeffs[k] * t + b[k])
    # Find min over t in [0, 1]
    # This is a convex PWL function
    # Minimum is where subgradient contains 0
    # Or at a breakpoint
    
    # Efficient: find all breakpoints in [0,1], evaluate f at each
    # Plus endpoints
    # But there could be O(L^2) breakpoints...
    
    # Alternative: use the fact that it's convex, so bisection on subgradient
    # f'(t) from left = max of {coeffs[k] : k active at t from left}
    # Actually for convex PWL, f is the upper envelope of lines
    # The minimum is where the slope changes sign
    
    # Simple approach: ternary search or golden section with exact evaluation
    def f_at(t):
        return float(np.max(coeffs * t + b))
    
    # Grid + refine
    grid = np.linspace(0, 1, 50)
    vals = [f_at(t) for t in grid]
    best_i = np.argmin(vals)
    
    # Golden section refine
    lo = grid[max(0, best_i-1)]
    hi = grid[min(len(grid)-1, best_i+1)]
    for _ in range(40):
        m1 = hi - (hi-lo)*0.618
        m2 = lo + (hi-lo)*0.618
        if f_at(m1) < f_at(m2):
            hi = m2
        else:
            lo = m1
    best_t = (lo+hi)/2
    return best_t, f_at(best_t)
```

```python
    def f_at(t):
        return float(np.max(coeff * t + b))
    
    grid = np.linspace(0, 1, n_grid)
    vals = np.array([f_at(t) for t in grid])
    best_i = int(np.argmin(vals))
    
    # Golden section refinement
    lo = grid[max(0, best_i - 1)]
    hi = grid[min(n_grid - 1, best_i + 1)]
    gr = (np.sqrt(5) - 1) / 2  # golden ratio conjugate
    for _ in range(50):

        m2 = lo + gr * (hi - lo)
        if f_at(m1) < f_at(m2):
            hi = m2
        else:
            lo = m1
    best_t = (lo + hi) / 2
    return best_t
```

```python
def q_1d_solve(q, r, G_col, corrs, q_r_current):
    """Solve 1D subproblem for coordinate r.
    G_col: array of shape (L,), G_col[k] = G[k, r]
    corrs: array of shape (L,), current correlation values
    Returns optimal t for q[r].
    """
    # f(t) = max_k (corrs[k] + G_col[k] * (t - q_r_current))
    #      = max_k (G_col[k] * t + (corrs[k] - G_col[k] * q_r_current))
    b = corrs - G_col * q_r_current
    
    # Find min of max of linear functions over [0, 1]
    # Use grid + golden section
    grid = np.linspace(0, 1, 50)
    f_vals = np.array([np.max(G_col * t + b) for t in grid])
    best_i = int(np.argmin(f_vals))
    
    if best_i == 0:
        return 0.0
    if best_i == len(grid) - 1:
        return 1.0
    
    lo = grid[best_i - 1]
    hi = grid[best_i + 1]
    gr = (np.sqrt(5) - 1) / 2
    for _ in range(50):
        m1 = hi - gr * (hi - lo)
        m2 = lo + gr * (hi - lo)
        if np.max(G_col * m1 + b) < np.max(G_col * m2 + b):
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2.0
```

This is clean and fast. Each call is O(L * 100) ≈ O(250 * 100) = 25K operations.

```python
def lp_trust_step(q, corrs, G, active_k, radius):
    """LP trust region step.
    Minimize tau subject to:
    corrs[k] + G[k,:] . dq <= tau for k in active_k
    0 <= q[r] + dq[r] <= 1
    |dq[r]| <= radius
    """
    m = len(q)
    n_vars = m + 1  # m for dq, 1 for tau
    
    c = np.zeros(n_vars)
    c[m] = 1.0  # minimize tau
    
    A_ub = np.zeros((len(active_k), n_vars))
    b_ub = np.zeros(len(active_k))
    
    for i, k in enumerate(active_k):
        A_ub[i, :m] = G[k, :]
        A_ub[i, m] = -1.0
        b_ub[i] = -corrs[k]
    
    bounds = []
    for r in range(m):
        lo = max(-q[r], -radius)
        hi = min(1.0 - q[r], radius)
        bounds.append((lo, hi))
    bounds.append((None, None))
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.status != 0:
        return None
    return res.x[:m]
```
