def project_to_box_sum(x, target_sum):
    """Project x to [0,1]^n with sum = target_sum."""
    x = np.clip(np.nan_to_num(x), 0.0, 1.0)

import numpy as np
import time

def eval_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_to_box_sum(x, target_sum):
    x = np.clip(np.nan_to_num(x), 0.0, 1.0)
    n = len(x)
    s = x.sum()
    if abs(s - target_sum) < 1e-12:
        return x
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(x + mid, 0, 1).sum() > target_sum:
            hi = mid
        else:
            lo = mid
    x = np.clip(x + (lo + hi) / 2.0, 0, 1)
    diff = target_sum - x.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(x)
            for idx in order:
                add = min(diff, 1.0 - x[idx])
                x[idx] += add
                diff -= add
                if diff < 1e-12:
                    break
        else:
            order = np.argsort(-x)
            for idx in order:
                sub = min(-diff, x[idx])
                x[idx] -= sub
                diff += sub
                if diff > -1e-12:
                    break
    return np.clip(x, 0.0, 1.0)

def transfer_to_n(h_old, n_new):
    n_old = len(h_old)
    x_old = (np.arange(n_old) + 0.5) * 2.0 / n_old
    x_new = (np.arange(n_new) + 0.5) * 2.0 / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return project_to_box_sum(h_new, n_new / 2.0)

    n = len(h)
    # Sort by value, assign top n_ones to 1
    order = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[order[:n_ones]] = 1.0
    return h_bin

def compute_effects(h):
    """Compute E[p] for all p: the change in corr when h[p] goes 1→0."""
    n = len(h)
    E = np.zeros((n, 2 * n - 1))
    for p in range(n):
        # First factor: h[p+d] - 1 for d where 0 <= p+d < n
        d_start = max(0 - p, -(n - 1))
        d_end = min(n - 1 - p, n - 1)
        if d_start <= d_end:
            ds = np.arange(d_start, d_end + 1)
            E[p, ds + (n - 1)] += (h[p + ds] - 1.0)
        # Second factor: h[p-d] for d where 0 <= p-d < n
        d_start2 = max(p - (n - 1), -(n - 1))
        d_end2 = min(p, n - 1)
        if d_start2 <= d_end2:
            ds2 = np.arange(d_start2, d_end2 + 1)
            E[p, ds2 + (n - 1)] += h[p - ds2]
    return E

def binary_local_search(h, rng, max_time=60.0, max_iter=50000):
    """Greedy + annealing local search on binary h."""
    n = len(h)
    h = h.copy()
    C = np.correlate(h, 1.0 - h, mode='full').astype(float)
    current_max = C.max()
    t_start = time.time()
    
    one_positions = np.where(h == 1)[0].tolist()
    zero_positions = np.where(h == 0)[0].tolist()
    
    # Phase 1: Greedy best-improvement
    for it in range(max_iter):
        if time.time() - t_start > max_time:
            break
        if not one_positions or not zero_positions:
            break
        
        E = compute_effects(h)
        best_improve = 0
        best_p, best_q = -1, -1
        best_new_max = current_max

        # The max change in C at any position is bounded by 2 (removing one 1 and adding another)
        # So new_max >= current_max - 2 and new_max <= current_max + 2
        
        for p in one_positions:
            Ep = E[p]
            for q in zero_positions:
                new_C = C + Ep - E[q]
                nm = new_C.max()
                if nm < best_new_max:
                    best_new_max = nm
                    best_p = p
                    best_q = q
                    best_improve = current_max - nm
        
        if best_p < 0 or best_improve < 1e-15:
            break
        
        # Apply swap
        h[best_p] = 0
        h[best_q] = 1
        C += E[best_p] - E[best_q]
        C = np.maximum(C, 0)  # corr should be non-negative for valid h... actually not necessarily
        # Actually corr can be negative? No: h[i]*(1-h[i+d]) >= 0 for h in [0,1]
        # So corr should be >= 0. But numerically it might dip.
        current_max = best_new_max
        one_positions.remove(best_p)
        zero_positions.remove(best_q)
        one_positions.append(best_q)
        zero_positions.append(best_p)
    
    # Phase 2: Annealing
    if time.time() - t_start < max_time - 5:
        T = max(0.001, current_max * 0.01)
        T_min = 0.0001
        cooling = 0.95
        
        n_ones = len(one_positions)
        n_zeros = len(zero_positions)
        
        for it in range(max_iter):
            if time.time() - t_start > max_time:
                break

                break
            if n_ones < 1 or n_zeros < 1:
                break

            # Random pair
            p = one_positions[rng.randint(n_ones)]
            q = zero_positions[rng.randint(n_zeros)]

            new_max = new_C.max()
            delta = new_max - current_max
            
            if delta <= 0 or rng.random() < np.exp(-delta / T):
                h[p] = 0
                h[q] = 1
                C = new_C
                current_max = new_max
                one_positions[one_positions.index(p)] = q
                zero_positions[zero_positions.index(q)] = p
            
            T *= cooling
    
    # Phase 3: Re-greedy
    if time.time() - t_start < max_time - 2:
        one_positions = sorted(np.where(h == 1)[0].tolist())
        zero_positions = sorted(np.where(h == 0)[0].tolist())
        
        for it in range(max_iter):
            if time.time() - t_start > max_time:
                break
            
            E = compute_effects(h)
            best_improve = 0
            best_p, best_q = -1, -1
            best_new_max = current_max
            
            for p in one_positions:
                Ep = E[p]
                for q in zero_positions:
                    new_C = C + Ep - E[q]
                    nm = new_C.max()
                    if nm < best_new_max - 1e-15:

                        best_q = q
                        best_improve = current_max - nm

            current_max = best_new_max
            one_positions.remove(best_p)
            zero_positions.remove(best_q)
            one_positions.append(best_q)
            zero_positions.append(best_p)

def project_to_box_sum(x, target_sum):
    x = np.clip(np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0), 0.0, 1.0)
    if abs(x.sum() - target_sum) < 1e-12:
        return x
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(x + mid, 0, 1).sum() > target_sum:
            hi = mid
        else:
            lo = mid
    x = np.clip(x + (lo + hi) / 2.0, 0, 1)
    diff = target_sum - x.sum()
    if abs(diff) > 1e-12:

The key observation: for a binary h, C_k = |{i : h[i]=1, h[i+d]=0}| where d = k-(N-1). This is the number of "1 followed by 0" at distance d.

If I think of h as a binary string, C_k counts the overlap between the 1-positions of h and the 0-positions of h shifted by d. This is related to the cross-correlation of the indicator sequences.

A classic result: for a binary sequence of length N with N/2 ones, the average value of C_k over all k is... let me compute:
Σ_k C_k = Σ_k Σ_i h[i]*(1-h[i+d_k]) where d_k = k-(N-1)
= Σ_i h[i] * Σ_k (1-h[i+d_k]) [valid]
= Σ_i h[i] * (N - 1 - (number of 1s at distance d from i for valid d))
