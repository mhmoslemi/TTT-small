def golden_search_1d(f, a, b, tol=1e-8, maxiter=100):
    """Minimize f on [a, b] using golden section search"""
    gr = (np.sqrt(5) - 1) / 2  # golden ratio conjugate

    d = a + gr * (b - a)
    fc = f(c)
    fd = f(d)

            d = c

            c = b - gr * (b - a)
            fc = f(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = f(d)

import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time


def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)


def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full').max() * dx


def make_sym_h(q, n):
    m = n // 2
    q = np.clip(q, 0.0, 1.0)
    h = np.empty(n)
    h[:m] = q
    h[m:] = 1.0 - q[::-1]
    return h


def make_q_from_h(h, n):
    m = n // 2
    q = 0.5 * (h[:m] + 1.0 - h[n - 1: n - 1 - m: -1])
    return np.clip(q, 0.0, 1.0)


def corr_vals(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx


def smooth_max_obj(q, n, beta):
    h = make_sym_h(q, n)
    corrs = corr_vals(h)
    return logsumexp(beta * corrs) / beta


def smooth_max_grad(q, n, beta):
    m = n // 2
    dx = 2.0 / n
    h = make_sym_h(q, n)
    corrs = corr_vals(h)
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    ws = w.sum()
    if ws < 1e-30:
        w = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        w /= w.sum()
    w *= beta
    grad_h = np.zeros(n)
    for lag in range(2 * n - 1):
        wk = w[lag]
        if wk < 1e-15:
            continue
        d = lag - (n - 1)
        i_lo = max(0, -d)
        i_hi = min(n - 1, n - 1 - d)
        if i_lo > i_hi:
            continue
        idx = np.arange(i_lo, i_hi + 1)
        grad_h[idx] += wk * (1.0 - h[idx + d])
        grad_h[idx + d] += wk * (-h[idx])
    grad_h *= dx
    grad_q = np.zeros(m)
    grad_q += grad_h[:m]
    grad_q -= grad_h[n - 1: n - 1 - m: -1]
    return grad_q


def subgrad_max(q, n, topk=10):
    """Subgradient of max correlation w.r.t. q (exact top-k)"""
    m = n // 2
    dx = 2.0 / n
    h = make_sym_h(q, n)
    corrs = corr_vals(h)
    idx = np.argsort(corrs)[::-1][:topk]
    weights = np.zeros(2 * n - 1)
    vals = corrs[idx]
    vmin = vals[-1]
    if vmin > 1e-15:
        weights[idx] = vals ** 2
    else:
        weights[idx] = 1.0
    weights /= weights.sum()
    grad_h = np.zeros(n)
    for li in range(len(idx)):
        lag = idx[li]
        wk = weights[li]
        if wk < 1e-15:
            continue
        d = lag - (n - 1)
        i_lo = max(0, -d)
        i_hi = min(n - 1, n - 1 - d)
        if i_lo > i_hi:
            continue
        ii = np.arange(i_lo, i_hi + 1)
        grad_h[ii] += wk * (1.0 - h[ii + d])
        grad_h[ii + d] += wk * (-h[ii])
    grad_h *= dx
    grad_q = np.zeros(m)
    grad_q += grad_h[:m]
    grad_q -= grad_h[n - 1: n - 1 - m: -1]
    return grad_q


def local_smooth_opt(q0, n, beta, maxiter=300):
    m = n // 2
    try:
        res = minimize(lambda x: smooth_max_obj(x, n, beta), q0,
                       jac=lambda x: smooth_max_grad(x, n, beta),
                       method='SLSQP', bounds=[(0.0, 1.0)] * m,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        h_new = make_sym_h(res.x, n)
        c5 = exact_score(h_new)
        if res.x is not None and np.all((res.x >= -1e-10) & (res.x <= 1 + 1e-10)):
            return np.clip(res.x, 0.0, 1.0), c5
    except Exception:
        pass
    return None, None


def subgrad_polish(q0, n, max_iter=300, lr0=0.05):
    m = n // 2
    q = q0.copy()
    c5 = exact_score(make_sym_h(q, n))
    lr = lr0
    patience = 0
    for it in range(max_iter):
        g = subgrad_max(q, n, topk=15)
        gn = np.linalg.norm(g)
        if gn < 1e-12:
            break
        step = lr / gn
        q_new = np.clip(q - step * g, 0.0, 1.0)
        h_new = make_sym_h(q_new, n)
        c5_new = exact_score(h_new)
        if c5_new < c5 - 1e-12:
            q = q_new
            c5 = c5_new
            patience = 0
            lr = min(lr * 1.05, 0.5)
        else:
            patience += 1
            lr *= 0.85
            if patience > 30 or lr < 1e-8:
                break
    return q, c5


def basin_hop_move(q, m, move_type, rng):
    if move_type == 0:
        i, j = rng.choice(m, size=2, replace=False)
        q_new = q.copy()
        q_new[i], q_new[j] = q[j], q[i]
        return q_new
    elif move_type == 1:
        i = rng.randint(m)
        q_new = q.copy()
        q_new[i] = 1.0 - q[i]
        return q_new
    elif move_type == 2:
        length = rng.randint(2, max(3, m // 4))
        start = rng.randint(0, m - length)
        q_new = q.copy()
        q_new[start:start + length] = 1.0 - q_new[start:start + length]
        return q_new
    elif move_type == 3:
        i, j = rng.choice(m, size=2, replace=False)
        q_new = q.copy()
        q_new[i] = 1.0 - q[i]
        q_new[j] = 1.0 - q[j]
        return q_new
    else:
        scale = rng.uniform(0.03, 0.15)
        i = rng.randint(m)
        width = rng.randint(1, 5)
        q_new = q.copy()
        for di in range(-width, width + 1):
            idx = i + di
            if 0 <= idx < m:
                q_new[idx] += scale * np.exp(-di ** 2 / (2.0 * width))
        return np.clip(q_new, 0.0, 1.0)


def try_n_value(n_try, h_source, n_source, max_iter=200):
    x_old = np.linspace(0, 2, n_source, endpoint=False)
    x_new = np.linspace(0, 2, n_try, endpoint=False)

    for beta in [50, 200, 500]:

        if q_new is not None and c5_new < best_c5 - 1e-10:
            best_q = q_new
            best_c5 = c5_new
    q_new, c5_new = subgrad_polish(best_q, n_try, max_iter=100)
    if c5_new < best_c5 - 1e-10:
        best_q = q_new
        best_c5 = c5_new
    return best_q, best_c5, n_try
