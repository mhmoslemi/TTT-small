import numpy as np
from scipy.optimize import linprog
import time

def compute_c5(h, dx):
    corrs = np.correlate(h, 1.0 - h, mode='full')
    return corrs.max() * dx

def grad_corr_k_vec(h, k, n):
    """Compute gradient of correlation at offset k."""
    offset = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -offset)
    i_hi = min(n - 1, n - 1 - offset)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx + offset] += (1.0 - h[idx])
    grad[idx] -= h[idx + offset]
    return grad

def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        h = np.clip(x - mid, 0.0, 1.0)
        s = h.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-10:
        mask = (h > 1e-8) & (h < 1.0 - 1e-8)
        nm = mask.sum()
        if nm > 0:
            adj = diff / nm
            # Check if adjustment keeps within bounds
            max_adj_up = np.minimum(1.0 - h[mask], max(adj, 0))
            max_adj_down = np.minimum(h[mask], max(-adj, 0))
            h[mask] += adj
            h = np.clip(h, 0.0, 1.0)
            # Second pass for residual
            diff2 = target_sum - h.sum()
            if abs(diff2) > 1e-12:
                mask2 = (h > 1e-10) & (h < 1.0 - 1e-10)
                if mask2.sum() > 0:
                    h[mask2] += diff2 / mask2.sum()
                    h = np.clip(h, 0.0, 1.0)
    return h

def lp_polish(h0, dx, target_sum, max_iters=200, alpha_init=0.3, n_active_frac=0.5):

        top_idx = np.argsort(corrs)[-n_active:]

        c_obj[-1] = 1.0

def sa_quantized(h0, dx, target_sum, M, n_steps, t0, budget_s, reserve_s):
    """Quantized simulated annealing."""
    n = len(h0)
    target_int = int(round(target_sum * M))
    # Initialize quantized state
    a = np.round(h0 * M).astype(int)
    a = np.clip(a, 0, M)
    # Fix sum
    diff = target_int - a.sum()
    if diff > 0:
        # Need to increase sum
        order = np.argsort(-h0)  # Start with highest values
        for i in order:
            if diff <= 0:
                break
            inc = min(diff, M - a[i])
            a[i] += inc
            diff -= inc
    elif diff < 0:
        order = np.argsort(h0)
        for i in order:
            if diff >= 0:
                break
            dec = min(-diff, a[i])
            a[i] -= dec
            diff += dec
    
    a = np.clip(a, 0, M).astype(int)
    
    best_h = a.astype(float) / M
    best_c5 = compute_c5(best_h, dx)
    
    t_current = 0.01
    t_final = 0.0001
    cooling = (t_final / t_current) ** (1.0 / max(1, n_steps))
    
    rng = np.random.RandomState(42)
    
    for step in range(n_steps):
        if time.time() - t0 > budget_s - reserve_s:
            break
        
        # Pick two random indices to swap between
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        
        # Determine direction
        if a[i] > 0 and a[j] < M:
            a[i] -= 1
            a[j] += 1
            new_h = a.astype(float) / M
            new_c5 = compute_c5(new_h, dx)
            delta_e = new_c5 - best_c5
            if delta_e < 0 or rng.random() < np.exp(-delta_e / t_current):
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = new_h.copy()
            else:
                a[i] += 1
                a[j] -= 1
            t_current *= cooling
        elif a[j] > 0 and a[i] < M:
            a[j] -= 1
            a[i] += 1
            new_h = a.astype(float) / M
            new_c5 = compute_c5(new_h, dx)
            delta_e = new_c5 - best_c5
            if delta_e < 0 or rng.random() < np.exp(-delta_e / t_current):
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = new_h.copy()
            else:
                a[j] += 1
                a[i] -= 1
            t_current *= cooling
    
    return best_h, best_c5


def make_binary_start(n, rng, pattern=None):
    """Create a binary starting point."""
    h = np.zeros(n)
    if pattern == 'alternating':
        for i in range(0, n, 2):
            h[i] = 1.0
    elif pattern == 'thue_morse':
        for i in range(n):
            # Thue-Morse: parity of number of 1s in binary
            c = bin(i).count('1') & 1
            h[i] = float(c)
    elif pattern == 'random':
        idx = rng.permutation(n)
        h[idx[:n//2]] = 1.0
    elif pattern is not None:
        # Periodic pattern
        p = pattern
        fill = p // 2
        for i in range(n):
            if (i % p) < fill:
                h[i] = 1.0
    else:
        idx = rng.permutation(n)
        h[idx[:n//2]] = 1.0
    
    return project_feasible(h, n / 2.0)


def interpolate_to_grid(h_src, n_dst, target_sum_dst):
    """Interpolate h from its grid to a new grid size."""
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src, target_sum_dst)
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    h_new = np.interp(x_dst, x_src, h_src)
    return project_feasible(h_new, target_sum_dst)
