def active_set_lp(h, dx, target_sum, active_k, step_bound):
    """Solve LP to find best step given active constraints."""
    n = len(h)
    corrs = compute_corrs(h)
    
    # Variables: delta[0..n-1], t
    # Total: n+1 variables
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0  # minimize t
    
    A_ub = []
    b_ub = []
    for k in active_k:
        row = np.zeros(n + 1)
        row[:n] = dx * grad_corr_k(h, k, n)
        row[n] = -1.0
        A_ub.append(row)
        b_ub.append(-corrs[k] * dx)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    bounds = []
    for i in range(n):
        lo = max(-h[i], -step_bound)
        hi = min(1.0 - h[i], step_bound)
        bounds.append((lo, hi))
    bounds.append((0.0, None))
    
    res = linprog(c_obj, A_ub=np.array(A_ub), b_ub=np.array(b_ub),
                  A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    if res.status != 0:
        return None, None
    return res.x, res.fun

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.special import logsumexp
import time


def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)


def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def c5_of_h(h, dx):
    return compute_corrs(h).max() * dx


def grad_corr_k(h, k, n):
    offset = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -offset)
    i_hi = min(n - 1, n - 1 - offset)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx + offset] += (1.0 - h[idx])
    grad[idx] += (-h[idx + offset])
    return grad


def smooth_max_obj(h, dx, beta):
    corrs = compute_corrs(h)
    return logsumexp(beta * corrs) / beta * dx


def smooth_max_grad(h, dx, beta):
    n = len(h)
    corrs = compute_corrs(h)
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones_like(corrs) / len(corrs)
    else:
        w /= w_sum
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] > 1e-13:
            grad += w[k] * grad_corr_k(h, k, n)
    return grad * dx


def slsqp_direct(h0, dx, target_sum, beta, maxiter):
    n = len(h0)

    def obj(x):
        corrs = compute_corrs(x)
        return logsumexp(beta * corrs) / beta * dx

    def grad_obj(x):
        return smooth_max_grad(x, dx, beta)

    def c_fun(x):
        return x.sum() - target_sum

    def c_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': c_fun, 'jac': c_jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0, jac=grad_obj, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        h_out = project_feasible(res.x, target_sum)
    except Exception:
        h_out = project_feasible(h0, target_sum)
    return h_out, c5_of_h(h_out, dx)


def slsqp_z(h0, dx, target_sum, beta, maxiter):
    n = len(h0)

    def sigmoid_2z(z):
        return 1.0 / (1.0 + np.exp(-np.clip(2.0 * z, -500, 500)))

    def obj(z):
        h = sigmoid_2z(z)
        corrs = compute_corrs(h)
        return logsumexp(beta * corrs) / beta * dx

    def grad_obj(z):
        h = sigmoid_2z(z)
        grad_h = smooth_max_grad(h, dx, beta)
        return 2.0 * h * (1.0 - h) * grad_h

    def c_fun(z):
        return sigmoid_2z(z).sum() - target_sum

    def c_jac(z):
        h = sigmoid_2z(z)
        return 2.0 * h * (1.0 - h)

    cons = [{'type': 'eq', 'fun': c_fun, 'jac': c_jac}]
    h0_clip = np.clip(h0, 1e-8, 1.0 - 1e-8)
    z0 = -0.5 * np.log((1.0 - h0_clip) / h0_clip)
    try:
        res = minimize(obj, z0, jac=grad_obj, method='SLSQP',
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        h_out = project_feasible(sigmoid_2z(res.x), target_sum)
    except Exception:
        h_out = project_feasible(h0_clip, target_sum)
    return h_out, c5_of_h(h_out, dx)


def iterative_lp_polish(h0, dx, target_sum, max_iters=200, init_step=0.15):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    step = init_step
    stall_count = 0

    for it in range(max_iters):
        corrs = compute_corrs(h)
        c5_val = corrs.max() * dx
        max_c = corrs.max()

        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            stall_count = 0
        else:
            stall_count += 1
        if stall_count > 40:
            step *= 0.7
            stall_count = 0
            if step < 1e-7:
                break

        # Active set: correlations within tolerance of max
        tol_frac = 0.05
        tol_val = max_c * (1.0 - tol_frac)
        active_mask = corrs >= tol_val
        active_k = np.where(active_mask)[0]

        if len(active_k) > 80:
            active_k = np.argsort(corrs)[-80:]

        # Build LP: min t s.t. dx*sum(grad_k[i]*delta[i]) + corr_k*dx <= t
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        A_rows = []
        b_rows = []
        for k in active_k:
            row = np.zeros(n + 1)
            row[:n] = dx * grad_corr_k(h, k, n)
            row[n] = -1.0
            A_rows.append(row)
            b_rows.append(-corrs[k] * dx - 1e-14)

        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)

        bounds = []
        for i in range(n):
            bounds.append((max(-h[i], -step), min(1.0 - h[i], step)))
        bounds.append((0.0, None))

        try:
            res = linprog(c_obj, A_ub=np.array(A_rows), b_ub=np.array(b_rows),
                          A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
        except Exception:
            break
        if res.status != 0:
            break

        delta = res.x[:n]
        if np.max(np.abs(delta)) < 1e-14:
            break

        h_new = project_feasible(h + delta, target_sum)
        new_c5 = c5_of_h(h_new, dx)

        if new_c5 < c5_val - 1e-14:
            h = h_new
            step = min(step * 1.2, init_step)
        else:
            improved = False
            for frac in [0.5, 0.25, 0.1, 0.05]:
                h_new = project_feasible(h + frac * delta, target_sum)
                new_c5 = c5_of_h(h_new, dx)
                if new_c5 < c5_val - 1e-14:
                    h = h_new
                    improved = True
                    break
            if not improved:
                step *= 0.5
                if step < 1e-8:
                    break

    return best_h, best_c5


def lp_polish_wide(h0, dx, target_sum, n_steps=40, lam=0.25):

def pair_transfer_polish(h0, dx, target_sum, n_top=25, n_iter=3):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)

    for iteration in range(n_iter):
        corrs = compute_corrs(h)
        grad = smooth_max_grad(h, dx, 600.0)

        dec_idx = np.argsort(grad)[-n_top:]
        inc_idx = np.argsort(grad)[:n_top]

        improved = False
        for i in dec_idx:
            for j in inc_idx:
                if i == j:
                    continue
                # Move eps from i to j: h[i]-eps, h[j]+eps
                eps_lo = max(h[i] - 1.0, -h[j])
                eps_hi = min(h[i], 1.0 - h[j])
                if eps_hi - eps_lo < 1e-12:
                    continue

                # d(corr[k])/d(eps) = grad_corr_k[j] - grad_corr_k[i]
                d = np.zeros(2 * n - 1)
                for k in range(2 * n - 1):
                    gk = grad_corr_k(h, k, n)
                    d[k] = gk[j] - gk[i]

                def f_ps(eps):
                    return (corrs + eps * d).max()

                gr = (np.sqrt(5.0) - 1.0) / 2.0
                a, b = eps_lo, eps_hi
                c_pt = b - gr * (b - a)
                d_pt = a + gr * (b - a)
                fc, fd = f_ps(c_pt), f_ps(d_pt)
                for _ in range(50):
                    if b - a < 1e-14:
                        break
                    if fc < fd:
                        b = d_pt
                        d_pt = c_pt
                        fd = fc
                        c_pt = b - gr * (b - a)
                        fc = f_ps(c_pt)
                    else:
                        a = c_pt
                        c_pt = d_pt
                        fc = fd
                        d_pt = a + gr * (b - a)
                        fd = f_ps(d_pt)

                eps_opt = (a + b) / 2.0
                new_c5_rough = (corrs + eps_opt * d).max() * dx
                if new_c5_rough < best_c5 - 1e-14:
                    new_h = h.copy()
                    new_h[i] -= eps_opt
                    new_h[j] += eps_opt
                    new_h = np.clip(new_h, 0.0, 1.0)
                    new_h = project_feasible(new_h, target_sum)
                    new_c5 = c5_of_h(new_h, dx)
                    if new_c5 < best_c5 - 1e-14:
                        h = new_h
                        best_c5 = new_c5
                        best_h = h.copy()
                        corrs = compute_corrs(h)
                        improved = True

        if not improved:
            break

    return best_h, best_c5


def momentum_subgrad(h0, dx, target_sum, max_iter=15000, beta_smooth=30.0):
    n = len(h0)
    h = h0.copy()
    momentum = np.zeros(n)
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    no_improve = 0
    lr = 1.0 / np.sqrt(n)
    beta_mom = 0.9

    for it in range(max_iter):
        corrs = compute_corrs(h)
        c5_val = corrs.max() * dx

        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 5000:
            break

        shifted = corrs - corrs.max()
        w = np.exp(beta_smooth * shifted)
        w /= w.sum()

        grad = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] > 1e-12:
                grad += w[k] * grad_corr_k(h, k, n)
        grad *= dx

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        momentum = beta_mom * momentum - lr * grad / (gnorm + 1e-10)
        h_new = h + momentum
        h_new = project_feasible(h_new, target_sum)
        new_c5 = c5_of_h(h_new, dx)

        tries = 0
        while new_c5 > c5_val + 1e-14 and tries < 4:
            lr *= 0.5
            momentum = beta_mom * momentum - lr * grad / (gnorm + 1e-10)
            h_new = h + momentum
            h_new = project_feasible(h_new, target_sum)
            new_c5 = c5_of_h(h_new, dx)
            tries += 1

        h = h_new

    return best_h, best_c5
