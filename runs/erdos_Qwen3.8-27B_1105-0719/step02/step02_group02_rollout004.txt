    # Find best gamma in [0,1] minimizing smooth_max(h + gamma*(v-h), dx, beta)
    # Use golden section search
    gr = (np.sqrt(5) - 1) / 2
    lo, hi = 0.0, 1.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = smooth_max_val(h + c*(v-h), dx, beta)
    fd = smooth_max_val(h + d*(v-h), dx, beta)

            d = lo + gr*(hi-lo)
            fd = smooth_max_val(h + d*(v-h), dx, beta)

def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def c5_of_h(h, dx):
    return float(compute_corrs(h).max() * dx)

def grad_corr_k(h, k, n):
    offset = k - (n - 1)
    grad = np.zeros(n)
    if offset == 0:
        grad[:] = 1.0 - 2.0 * h
        return grad
    i_lo = max(0, -offset)
    i_hi = min(n - 1, n - 1 - offset)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx + offset] += (1.0 - h[idx])
    grad[idx] += (-h[idx + offset])
    return grad

def smooth_max_val(h, dx, beta):
    return float(logsumexp(beta * compute_corrs(h)) / beta * dx)

def smooth_max_grad(h, dx, beta):
    n = len(h)
    corrs = compute_corrs(h)
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones_like(corrs) / len(corrs)
    else:
        w /= w_sum
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        grad += w[k] * grad_corr_k(h, k, n)
    return grad * dx

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.special import logsumexp
import time


def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h


def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def c5_of_h(h, dx):
    return float(compute_corrs(h).max() * dx)


def grad_corr_k(h, k, n):
    offset = k - (n - 1)
    grad = np.zeros(n)
    if offset == 0:
        grad[:] = 1.0 - 2.0 * h
        return grad
    i_lo = max(0, -offset)
    i_hi = min(n - 1, n - 1 - offset)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx + offset] += (1.0 - h[idx])
    grad[idx] += (-h[idx + offset])
    return grad


def smooth_max_val(h, dx, beta):
    return float(logsumexp(beta * compute_corrs(h)) / beta * dx)


def smooth_max_grad(h, dx, beta):
    n = len(h)
    corrs = compute_corrs(h)
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones_like(corrs) / len(corrs)
    else:
        w /= w_sum
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        grad += w[k] * grad_corr_k(h, k, n)
    return grad * dx


def frank_wolfe(h0, dx, target_sum, beta, max_iter=8000):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    patience = 0

    for t in range(max_iter):
        g = smooth_max_grad(h, dx, beta)
        idx = np.argsort(g)[:int(round(target_sum))]
        v = np.zeros(n)
        v[idx] = 1.0

        if np.allclose(v, h, atol=1e-6):
            break

        gamma = 2.0 / (t + 2.0)
        h_new = h + gamma * (v - h)

        c5_new = c5_of_h(h_new, dx)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            patience = 0
        else:
            patience += 1

        h = h_new
        if patience > 2000:
            break

    return best_h, best_c5


def slp_optimize(h0, dx, top_m, max_iter, step_fracs=None):
    n = len(h0)
    h = h0.copy()
    target_sum = n / 2.0
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)

    if step_fracs is None:
        step_fracs = [1.0, 0.7, 0.5, 0.3, 0.15]

    for it in range(max_iter):
        corrs = compute_corrs(h)
        top_idx = np.argsort(corrs)[-top_m:]

        A_rows = []
        b_vals = []
        for k in top_idx:
            gk = grad_corr_k(h, k, n)
            row = np.concatenate([gk, [-1.0]])
            rhs = corrs[k] - gk @ h
            A_rows.append(row)
            b_vals.append(rhs)

        if len(A_rows) == 0:
            break
        A = np.array(A_rows)
        b = np.array(b_vals)
        c_vec = np.zeros(n + 1)
        c_vec[-1] = 1.0
        bounds = [(0.0, 1.0)] * n + [(None, None)]
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0

def slsqp_opt(h0, dx, target_sum, beta, maxiter):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum,
             'jac': lambda x: np.ones(n)}]

    def obj(x):
        return smooth_max_val(x, dx, beta)

    def g(x):
        return smooth_max_grad(x, dx, beta)

    try:
        res = minimize(obj, h0, jac=g, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        h_out = project_feasible(res.x, target_sum)
    except Exception:
        h_out = project_feasible(h0, target_sum)
    return h_out, c5_of_h(h_out, dx)
