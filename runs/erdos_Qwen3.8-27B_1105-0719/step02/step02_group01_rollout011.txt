    return grad

import numpy as np
from scipy.optimize import linprog
import time


def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    residual = target_sum - h.sum()
    if abs(residual) > 1e-13:
        if residual > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = mask.sum()
        if cnt > 0:
            h[mask] += residual / cnt
    return np.clip(h, 0.0, 1.0)


def c5_of_h(h, dx):
    return np.correlate(h, 1.0 - h, mode='full').max() * dx


def grad_corr_k(h, k, n):
    offset = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -offset)
    i_hi = min(n - 1, n - 1 - offset)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx + offset] += (1.0 - h[idx])
    grad[idx] += (-h[idx + offset])
    return grad


def lp_iterate(h, dx, target_sum, alpha, frac_active):
    n = len(h)
    corrs = np.correlate(h, 1.0 - h, mode='full')
    max_c = corrs.max()
    c5_val = max_c * dx

    n_total = 2 * n - 1
    n_act = max(20, int(frac_active * n_total))
    top_idx = np.argsort(corrs)[-n_act:]
    active_idx = top_idx

    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0

    A_rows = []
    b_rows = []
    for k in active_idx:
        gk = grad_corr_k(h, k, n)
        row = np.zeros(n + 1)
        row[:n] = dx * gk
        row[n] = -1.0
        A_rows.append(row)
        b_rows.append(-dx * corrs[k])

    A_ub = np.array(A_rows)
    b_ub = np.array(b_rows)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)

    bounds = []
    for i in range(n):
        bounds.append((-h[i] * alpha, (1.0 - h[i]) * alpha))
    bounds.append((0.0, None))

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True})
    except Exception:
        return None

    if res.status != 0 or res.x is None:
        return None

    delta = res.x[:n]
    if np.max(np.abs(delta)) < 1e-15:
        return None

    for scale in [1.0, 0.7, 0.5, 0.35, 0.25, 0.15, 0.1]:
        h_new = project_feasible(h + scale * delta, target_sum)
        new_c5 = c5_of_h(h_new, dx)
        if new_c5 < c5_val - 1e-15:
            return (h_new, new_c5)

    return None


def lp_optimize(h0, dx, target_sum, t0, budget, max_time):

    h = h0.copy().astype(float)
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    alpha = 0.3
    frac = 0.35
    stall = 0

    while (time.time() - t0) < (budget - max_time):
        result = lp_iterate(h, dx, target_sum, alpha, frac)
        if result is None:
            stall += 1
            alpha *= 0.6
            frac = min(0.6, frac * 1.2)

                alpha = 0.3
                frac = 0.35

            h = h_new
            stall = 0

            frac = max(0.2, frac * 0.95)
        else:
            h = h_new
            stall += 1

def transfer_grid(h_src, n_target):
    x_old = (np.arange(len(h_src)) + 0.5) * 2.0 / len(h_src)
    x_new = (np.arange(n_target) + 0.5) * 2.0 / n_target
    h_new = np.interp(x_new, x_old, h_src)
    target = n_target / 2.0
    return project_feasible(h_new, target)


def diverse_starts(n, target_sum, rng, n_starts):
    starts = []
    for i in range(n_starts):

        if typ == 0:

        elif typ == 1:

            h = np.zeros(n)
            for j in range(n):

        elif typ == 2:
            h = np.zeros(n)
            b1 = max(1, int(target_sum * rng.uniform(0.2, 0.6)))
            p1 = rng.randint(0, n - b1)
            h[p1:p1 + b1] = 1.0
            remaining = np.where(h == 0)[0]
            b2 = int(target_sum) - b1
            if b2 > 0 and len(remaining) >= b2:
                p2 = remaining[rng.randint(0, len(remaining) - b2 + 1)]
                h[p2:p2 + b2] = 1.0
        elif typ == 3:
            h = np.zeros(n)
            idx = rng.permutation(n)
            h[idx[:int(target_sum)]] = 1.0
        else:
            h = np.ones(n) * 0.5
        h = project_feasible(h, target_sum)
        starts.append(h)
    return starts
