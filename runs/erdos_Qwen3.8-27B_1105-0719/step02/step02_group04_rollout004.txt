        else:

def build_anti(q, n):
    """Build anti-symmetric h from q (first half)."""
    m = n // 2

    h[:m] = q

def logsumexp_obj_grad(q, n, tau):
    """
    Compute log-sum-exp objective and gradient with respect to q.
    q has length m = n//2.
    Phi = M + tau * log(sum_j exp((C_j - M)/tau))
    where C_j = raw_j * dx

    h = build_anti(q, n)
    dx = 2.0 / n
    raw = corr_raw(h)
    C = raw * dx
    M = C.max()

    p = exps / Z  # softmax weights

def project_to_target(x, target_sum):

        t *= ratio

    if err not in sched:
        sched.append(err)
    return sched

        nc = n_base + d
        if 48 <= nc <= 220 and nc % 2 == 0 and nc != n_base:
            n_try_list.append(nc)
    n_try_list = sorted(set(n_try_list))

    for n_try in n_try_list:
        if remaining() < reserve + 30:
            break
        h_try = interp_to_n(best_h, n_try)
        q_try = to_anti_q(h_try, n_try)
        tau_sched_n = make_tau_schedule(0.02, 3e-5, 0.5, count=6)
        q_try, c5_try = smooth_anneal_sym(q_try, n_try, tau_sched_n, 120, reserve, t0, budget_s)
        h_try = build_anti(q_try, n_try)
        if c5_try < best_c5:
            best_h = h_try.copy()
            best_c5 = c5_try

    # Phase 5: Random restarts
    n_r = len(best_h)
    if n_r % 2 != 0:
        n_r -= 1
    m_r = n_r // 2
    max_rest = min(25, max(3, int((remaining() - reserve - 5) / 18.0)))
    q_best_start = to_anti_q(best_h, n_r) if len(best_h) == n_r else None

    for trial in range(max_rest):
        if remaining() < reserve + 10:
            break
        style = trial % 4
        if style == 3 and q_best_start is not None:
            q_r = q_best_start + rng.normal(0, 0.08, m_r)
        else:
            q_r = make_q_random(m_r, rng, style % 4)
        h_r = build_anti(q_r, n_r)
        tau_sched_r = make_tau_schedule(0.02, 3e-5, 0.5, count=5)
        q_r, c5_r = smooth_anneal_sym(q_r, n_r, tau_sched_r, 100, reserve, t0, budget_s)
        h_r = build_anti(q_r, n_r)
        if c5_r < best_c5:
            best_h = h_r.copy()
            best_c5 = c5_r

    # Phase 6: Final deep polish
    if remaining() > reserve + 5 and best_h is not None:
        n_fin = len(best_h)
        if n_fin % 2 != 0:
            n_fin -= 1
        q_fin = to_anti_q(best_h, n_fin)
        tau_fine = make_tau_schedule(0.004, 1e-5, 0.45, count=12)
        q_fin, c5_fin = smooth_anneal_sym(q_fin, n_fin, tau_fine, 400, reserve, t0, budget_s)
        h_fin = build_anti(q_fin, n_fin)
        if c5_fin < best_c5:
            best_h = h_fin.copy()
            best_c5 = c5_fin

    # Phase 7: Final full polish
    if remaining() > reserve + 5 and best_h is not None:
        n_p = len(best_h)
        tau_pol = make_tau_schedule(0.005, 1e-5, 0.45, count=8)
        h_p, c5_p = smooth_anneal_full(best_h, n_p, tau_pol, 200, 5000.0, reserve, t0, budget_s)
        if c5_p < best_c5:
            best_h = h_p.copy()
            best_c5 = c5_p

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    n_final = len(best_h)
    best_h = project_to_target(best_h, n_final / 2.0)
    best_c5 = eval_c5(best_h)

    return best_h, best_c5, n_final
