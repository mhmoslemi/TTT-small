def pair_coordinate_descent(h, dx, target_sum, max_sweeps, beta):
    n = len(h)
    best_c5 = c5_of_h(h)
    
    for sweep in range(max_sweeps):
        improved = False
        # Compute smooth max weights
        corrs = np.correlate(h, 1.0 - h, mode='full')
        m = corrs.max()
        w = np.exp(beta * (corrs - m))
        w /= w.sum()
        
        # For each pair, compute gradient direction
        # grad_i = sum_k w[k] * d(C_k)/d(h[i])
        # d(C_k)/d(h[i]) = (1-h[i+k']) for terms where h[i] appears as first factor
        #                            - h[i-k'] for terms where h[i] appears as second factor
        
        # Actually, let me compute the full gradient of the smooth max
        grad = np.zeros(n)
        for k in range(len(corrs)):
            if w[k] < 1e-14:
                continue
            offset = k - (n - 1)
            i_lo = max(0, -offset)
            i_hi = min(n - 1, n - 1 - offset)
            if i_lo > i_hi:
                continue
            idx = np.arange(i_lo, i_hi + 1)
            grad[idx + offset] += w[k] * (1.0 - h[idx])
            grad[idx] += w[k] * (-h[idx + offset])
        grad *= dx
        
        # Try pairs that are most "opposing" in gradient
        # For pair (i,j): objective change ≈ grad[i]*d + grad[j]*(-d) = (grad[i]-grad[j])*d
        # So we want grad[i] > grad[j] and move d > 0 (increase h[i], decrease h[j])
        
        # Sort by gradient
        order = np.argsort(-grad)
        # Try top pairs
        tried = 0
        for aa in range(min(30, n)):
            for bb in range(aa+1, min(aa+40, n)):
                i = order[aa]
                j = order[bb]
                if i == j:
                    continue
                if grad[i] - grad[j] < 1e-10:
                    continue
                # Line search: h[i]+d, h[j]-d
                # Bounds: d ∈ [-h[i], 1-h[j]]
                d_lo = -h[i]
                d_hi = 1.0 - h[j]
                if d_hi <= d_lo:
                    continue
                # Golden section or ternary search
                best_d = 0.0
                best_val = best_c5
                for t in [0.01, 0.02, 0.05, 0.1, 0.2, 0.3, 0.5]:

                    if d <= 0:

                    h_new = h.copy()
                    h_new[i] += d
                    h_new[j] -= d
                    if h_new[i] > 1.0 or h_new[j] < 0.0:
                        continue
                    val = c5_of_h(h_new)
                    if val < best_val - 1e-12:
                        best_val = val
                        best_d = d
                if best_d > 0:
                    h[i] += best_d
                    h[j] -= best_d
                    h = np.clip(h, 0, 1)
                    best_c5 = best_val
                    improved = True
                tried += 1
                if tried > 200:
                    break
            if tried > 200:
                break
        
        if not improved:
            break
    
    return h, best_c5
