import numpy as np
from scipy.optimize import linprog
import time


def project_feasible(x, target_sum):
    n = len(x)
    x = np.clip(x, -2.0, 2.0)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        cnt = mask.sum()
        if cnt > 0:
            h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)


def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def c5_of_h(h, dx):
    return compute_corrs(h).max() * dx


def grad_corr_k(h, k, n):
    offset = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -offset)
    i_hi = min(n - 1, n - 1 - offset)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    j1 = idx + offset
    grad[j1] += (1.0 - h[idx])
    grad[idx] += (-h[idx + offset])
    return grad


def grad_all_active(h, active_idx, n):
    m = len(active_idx)
    G = np.zeros((m, n))
    for row, k in enumerate(active_idx):
        G[row] = grad_corr_k(h, k, n)
    return G


def active_set_lp_step(h, dx, target_sum, n_active, alpha):
    n = len(h)
    corrs = compute_corrs(h)
    n_total = 2 * n - 1
    current_c5 = corrs.max() * dx

    if n_active >= n_total:
        active_idx = np.arange(n_total)
    else:
        active_idx = np.argsort(corrs)[-n_active:]

    G = grad_all_active(h, active_idx, n)
    m = len(active_idx)

    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0

    A_ub = np.empty((m, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -corrs[active_idx]

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)

    bounds = []
    for i in range(n):
        bounds.append((-h[i] * alpha, (1.0 - h[i]) * alpha))
    bounds.append((-np.inf, np.inf))

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status != 0:
            return h, current_c5, False
        delta = res.x[:n]
    except Exception:
        return h, current_c5, False

    if np.abs(delta).max() < 1e-15:
        return h, current_c5, False

    best_s = 0.0
    best_c5 = current_c5
    best_h = h

    for s in [1.0, 0.7, 0.5, 0.35, 0.2, 0.1, 0.05]:
        h_new = project_feasible(h + s * delta, target_sum)
        c5_new = c5_of_h(h_new, dx)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_s = s
            best_h = h_new
            break

    return best_h, best_c5, best_s > 0
