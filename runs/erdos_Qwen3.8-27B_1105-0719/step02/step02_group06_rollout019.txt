from scipy.optimize import linprog, minimize
import time

def eval_score(h):

    dx = 2.0 / n

    h = np.empty(2 * m)
    h[:m] = q

        s = np.clip(x + mid, 0.0, 1.0).sum()

    idx = np.arange(i_lo, i_hi + 1)

    grad[idx + d] += (-h[idx])
    return grad

    """Compute gradient matrix for active lags w.r.t. q. Returns (|active|, m)."""
    m = n_full // 2

        gh = corr_grad_h(h, k)
        for r in range(m):
            G[i, r] = gh[r] - gh[n_full - 1 - r]

def select_active(corr_arr, margin_frac=0.03, neighbor_pad=3, max_k=50):
    max_val = corr_arr.max()
    threshold = max_val - max(1e-9, margin_frac * max_val)
    active = set(np.where(corr_arr >= threshold)[0].tolist())
    padded = set()
    for k in active:
        for d in range(-neighbor_pad, neighbor_pad + 1):
            if 0 <= k + d < len(corr_arr):
                padded.add(k + d)
    active_list = sorted(padded)
    if len(active_list) > max_k:
        scored = sorted([(corr_arr[k], k) for k in active_list], reverse=True)
        active_list = sorted([k for _, k in scored[:max_k]])
    return active_list

def solve_lp(q, active_k, n_full, delta):
    """Solve LP to minimize max of active linearized correlations."""
    m = n_full // 2
    h = build_sym_h(q)
    dx = 2.0 / n_full
    c_vals = np.array([float(np.correlate(h, 1.0-h, mode='full')[k]) * dx for k in active_k])
    
    G = np.zeros((len(active_k), m))
    for i, k in enumerate(active_k):
        gh = corr_grad_h(h, k)
        for r in range(m):
            G[i, r] = (gh[r] - gh[n_full - 1 - r]) * dx
    
    # Linearization: c_k + G[k,:] @ (q' - q) <= t
    # => G[k,:] @ q' - t <= c_k - G[k,:] @ q
    rhs = c_vals - G @ q
    
    # Build A_ub, b_ub
    n_vars = m + 1
    A_rows = []
    b_list = []
    # Correlation constraints
    for i in range(len(active_k)):
        row = np.zeros(n_vars)
        row[:m] = G[i, :]
        row[m] = -1.0
        A_rows.append(row)
        b_list.append(rhs[i])
    # Trust region: q'[r] - q[r] <= delta
    for r in range(m):
        row = np.zeros(n_vars)
        row[r] = 1.0
        A_rows.append(row)
        b_list.append(q[r] + delta)
    # Trust region: -(q'[r] - q[r]) <= delta
    for r in range(m):
        row = np.zeros(n_vars)
        row[r] = -1.0
        A_rows.append(row)
        b_list.append(-(q[r] - delta))
    
    A_ub = np.array(A_rows)
    b_ub = np.array(b_list)
    c_obj = np.zeros(n_vars)

    bounds = [(0.0, 1.0)] * m + [(0.0, None)]

        if res.status == 0:
            q_new = np.clip(res.x[:m], 0.0, 1.0)

            # So f(v) = max_k (a_k * v + b_k)

            intercepts = np.zeros(len(active_k))

                # where grad_h is the gradient of corr_k w.r.t. h

                # corr_k = sum_i h[i]*(1-h[i+d])
                # Term involving h[r]=v: if r+d < n: v*(1-h[r+d])
                # Term involving h[r] as the (i+d) part: if r-d >= 0: h[r-d]*(1-v)
                # Term involving h[n-1-r]=1-v: if n-1-r+d < n and n-1-r >= 0: (1-v)*(1-h[n-1-r+d])
                # Term involving h[n-1-r] as the (i+d) part: if n-1-r-d >= 0: h[n-1-r-d]*v
                
                slope = 0.0
                intercept = 0.0
                
                # h[r] = v
                if r + d >= 0 and r + d < n_full:
                    slope += (1.0 - h[r + d])
                    intercept += 0.0  # v*(1-h[r+d]) has slope (1-h[r+d]), intercept 0
                if r - d >= 0 and r - d < n_full:
                    slope += (-h[r - d])  # h[r-d]*(1-v) = h[r-d] - h[r-d]*v
                    intercept += h[r - d]
                
                # h[n-1-r] = 1-v
                pos2 = n_full - 1 - r
                if pos2 + d >= 0 and pos2 + d < n_full:
                    # (1-v)*(1-h[pos2+d]) = (1-h[pos2+d]) - v*(1-h[pos2+d])
                    slope += (-(1.0 - h[pos2 + d]))
                    intercept += (1.0 - h[pos2 + d])
                if pos2 - d >= 0 and pos2 - d < n_full:
                    # h[pos2-d]*(1-(1-v)) = h[pos2-d]*v
                    slope += h[pos2 - d]
                    intercept += 0.0

                intercepts[i] = corr_full[k] - slope * q[r]  # so that a*v + b = corr_k when v=q[r]
            
            # Now f(v) = max_i (slopes[i]*v + intercepts[i])
            # Find minimum of this convex piecewise-linear function on [0, 1]
            # Critical points: 0, 1, and intersections of pairs
            
            candidates = [0.0, 1.0]
            for i in range(len(slopes)):
                for j in range(i + 1, len(slopes)):
                    ds = slopes[i] - slopes[j]

                        v_cross = (intercepts[j] - intercepts[i]) / ds
                        if 0.0 <= v_cross <= 1.0:
                            candidates.append(v_cross)
            
            # Also add a few uniform points for safety
            candidates.extend(np.linspace(0.0, 1.0, 5).tolist())

            best_f = float('inf')

                    best_f = f_val
                    best_v = v

        if w[k] < 1e-12:
            continue
        gh = corr_grad_h(h, k)
        for r in range(m):
            grad_q[r] += w[k] * (gh[r] - gh[n_full - 1 - r])
    grad_q *= dx
    return grad_q

def smooth_optimize(q0, n_full, beta_list, maxiter, time_limit):
    m = n_full // 2
    q = np.clip(q0.copy(), 0, 1)
    best_score = eval_score(build_sym_h(q))
    best_q = q.copy()
    for beta in beta_list:
        if time.time() > time_limit:
            break
        _SM['n'] = n_full
        _SM['beta'] = beta
        bounds = [(0.0, 1.0)] * m
        try:
            res = minimize(smooth_obj, q, jac=smooth_jac, method='L-BFGS-B',
                          bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10})
            q_new = np.clip(res.x, 0, 1)
        except Exception:
            q_new = q.copy()
        sc = eval_score(build_sym_h(q_new))
        if sc < best_score:
            best_score = sc
            best_q = q_new.copy()
        q = q_new
    return best_q, best_score

def transfer_to_n(h_old, n_new):
    n_old = len(h_old)
    x_old = (np.arange(n_old) + 0.5) * 2.0 / n_old
    x_new = (np.arange(n_new) + 0.5) * 2.0 / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new / 2.0)

def lp_coord_loop_sym(q, n_full, time_limit, max_rounds=200):
    m = n_full // 2
    best_q = q.copy()
    best_score = eval_score(build_sym_h(q))
    delta = 0.15

    for rnd in range(max_rounds):
        if time.time() > time_limit:
            break
        h = build_sym_h(best_q)
        corr_s = corr_full_scaled(h)
        active = select_active(corr_s, margin_frac=0.04, neighbor_pad=3, max_k=45)

        q_lp = solve_lp_sym(best_q, active, n_full, delta)
        if q_lp is not None:
            sc = eval_score(build_sym_h(q_lp))
            if sc < best_score - 1e-14:
                best_score = sc
                best_q = q_lp.copy()
                if delta > 0.02:
                    delta *= 1.1
                continue

        q_cs = coord_sweep_sym(best_q, active, n_full, num_passes=2)
        sc = eval_score(build_sym_h(q_cs))
        if sc < best_score - 1e-14:
            best_score = sc
            best_q = q_cs.copy()
            delta = 0.15
            continue

        delta *= 0.7
        if delta < 0.005:
            active2 = select_active(corr_full_scaled(build_sym_h(best_q)), margin_frac=0.08, neighbor_pad=5, max_k=60)
            q_cs2 = coord_sweep_sym(best_q, active2, n_full, num_passes=3)
            sc2 = eval_score(build_sym_h(q_cs2))
            if sc2 < best_score - 1e-14:
                best_score = sc2
                best_q = q_cs2.copy()
                delta = 0.1
                continue
            break

    return best_q, best_score

def lp_coord_loop_full(h, time_limit, max_rounds=100):
    n = len(h)
    best_h = h.copy()
    best_score = eval_score(h)
    delta = 0.12

    for rnd in range(max_rounds):
        if time.time() > time_limit:
            break
        corr_s = corr_full_scaled(best_h)
        active = select_active(corr_s, margin_frac=0.04, neighbor_pad=3, max_k=45)

        h_lp = solve_lp_full(best_h, active, delta)
        if h_lp is not None:
            sc = eval_score(h_lp)
            if sc < best_score - 1e-14:
                best_score = sc
                best_h = h_lp.copy()
                if delta > 0.02:
                    delta *= 1.1
                continue

        h_cs = coord_sweep_full(best_h, active, num_passes=2)
        sc = eval_score(h_cs)
        if sc < best_score - 1e-14:
            best_score = sc
            best_h = h_cs.copy()
            delta = 0.12
            continue

        delta *= 0.7
        if delta < 0.005:
            active2 = select_active(corr_full_scaled(best_h), margin_frac=0.08, neighbor_pad=5, max_k=60)
            h_cs2 = coord_sweep_full(best_h, active2, num_passes=3)
            sc2 = eval_score(h_cs2)
            if sc2 < best_score - 1e-14:
                best_score = sc2
                best_h = h_cs2.copy()
                delta = 0.1
                continue
            break

    return best_h, best_score

def make_structured_start(m, rng):
    x = np.linspace(0, 1, m)
    k1 = rng.randint(1, 4)
    k2 = rng.randint(2, 6)
    amp1 = rng.uniform(0.1, 0.4)
    amp2 = rng.uniform(0.05, 0.2)
    ph1 = rng.uniform(0, 2 * np.pi)
    ph2 = rng.uniform(0, 2 * np.pi)
    q = 0.5 + amp1 * np.sin(2 * np.pi * k1 * x + ph1) + amp2 * np.sin(2 * np.pi * k2 * x + ph2)
    return np.clip(q, 0, 1)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def rem():
        return budget_s - (time.time() - t0)

    best_h = None
    best_c5 = float('inf')

    try:
        h_parent = np.array(initial_h_values, dtype=float).copy().reshape(-1)
    except (NameError, AttributeError):
        h_parent = None

    if h_parent is not None and len(h_parent) > 10:
        n_p = len(h_parent)
        if abs(np.sum(h_parent) - n_p / 2.0) > 0.5:
            h_parent = project_feasible(h_parent, n_p / 2.0)
        h_parent = np.clip(h_parent, 0.0, 1.0)
        h_parent = project_feasible(h_parent, n_p / 2.0)
        best_h = h_parent.copy()
        best_c5 = eval_score(h_parent)
    else:
        m_fb = 63
        x_fb = np.linspace(0, 1, m_fb)
        q_fb = 0.5 + 0.3 * np.sin(2 * np.pi * x_fb)
        best_h = build_sym_h(np.clip(q_fb, 0, 1))
        best_c5 = eval_score(best_h)

    n_cur = len(best_h)

    # Phase 1: Smooth max warmup on parent
    if n_cur % 2 == 0 and rem() > 200:
        q_init = sym_from_h(best_h)
        tl = time.time() + min(120, rem() * 0.25)
        q_opt, c5_s = smooth_optimize(q_init, n_cur, [60, 150, 300, 600], 200, tl)
        h_s = build_sym_h(q_opt)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()

    # Phase 2: LP + coordinate sweep on symmetric (main optimization)
    if n_cur % 2 == 0 and rem() > 150:
        q_cur = sym_from_h(best_h)
        tl = time.time() + min(400, rem() * 0.4)
        q_opt, c5_lp = lp_coord_loop_sym(q_cur, n_cur, tl, max_rounds=300)
        h_lp = build_sym_h(q_opt)
        if c5_lp < best_c5:
            best_c5 = c5_lp
            best_h = h_lp.copy()

    # Phase 3: LP + coordinate sweep on full (break symmetry)
    if rem() > 150:
        h_full = best_h.copy()
        h_full = h_full + 0.02 * rng.randn(len(h_full))
        h_full = project_feasible(h_full, len(h_full) / 2.0)
        tl = time.time() + min(200, rem() * 0.3)
        h_opt, c5_full = lp_coord_loop_full(h_full, tl, max_rounds=150)
        if c5_full < best_c5:
            best_c5 = c5_full
            best_h = h_opt.copy()

    # Phase 4: Multi-resolution exploration
    n_base = len(best_h)
    n_candidates = set()
    for delta in [-12, -8, -6, -4, -3, -2, -1, 1, 2, 3, 4, 6, 8, 12, 16]:
        nc = n_base + delta
        if 48 <= nc <= 240:
            n_candidates.add(nc)
    if n_base // 2 >= 32:
        n_candidates.add(n_base // 2)
    if n_base * 3 // 2 <= 240:
        n_candidates.add(n_base * 3 // 2)
    n_candidates = sorted(n_candidates)

    for n_try in n_candidates:
        if rem() < 40:
            break
        h_try = transfer_to_n(best_h, n_try)
        nt = len(h_try)

        if nt % 2 == 0:
            q_try = sym_from_h(h_try)
            if rem() > 80:
                tl = time.time() + min(60, rem() * 0.5)
                q_o, c5_o = lp_coord_loop_sym(q_try, nt, tl, max_rounds=150)
                h_o = build_sym_h(q_o)
                if c5_o < best_c5:
                    best_c5 = c5_o
                    best_h = h_o.copy()
        if rem() > 40:
            tl = time.time() + min(40, rem() * 0.5)
            h_o, c5_o = lp_coord_loop_full(h_try, tl, max_rounds=80)
            if c5_o < best_c5:
                best_c5 = c5_o
                best_h = h_o.copy()

    # Phase 5: Diverse random restarts
    n_restarts = min(12, max(3, int(rem() / 70)))
    for i in range(n_restarts):
        if rem() < 50:
            break
        choices = [len(best_h), max(56, len(best_h) + rng.choice([-8, -4, 4, 8]))]
        n_r = min(220, max(48, int(choices[i % len(choices)])))
        if n_r % 2 != 0:
            n_r += 1

        if n_r % 2 == 0:
            m_r = n_r // 2
            q_r = make_structured_start(m_r, rng)
            q_r[::3] = rng.clip(0, 1, q_r[::3].copy() + 0.3 * rng.randn(m_r // 3))
            q_r = np.clip(q_r, 0, 1)

            if rem() > 70:
                tl = time.time() + min(40, rem() * 0.5)
                q_o, c5_o = lp_coord_loop_sym(q_r, n_r, tl, max_rounds=100)
                h_o = build_sym_h(q_o)
                if c5_o < best_c5:
                    best_c5 = c5_o
                    best_h = h_o.copy()
        else:
            h_r = make_structured_start(n_r // 2, rng)
            h_r = build_sym_h(h_r)
            h_r = 0.3 + 0.4 * h_r + 0.15 * rng.randn(n_r)
            h_r = project_feasible(h_r, n_r / 2.0)
            if rem() > 50:
                tl = time.time() + min(35, rem() * 0.5)
                h_o, c5_o = lp_coord_loop_full(h_r, tl, max_rounds=60)
                if c5_o < best_c5:
                    best_c5 = c5_o
                    best_h = h_o.copy()

    # Phase 6: Final intensive polish on best
    if rem() > 30 and best_h is not None:
        n_fin = len(best_h)
        if n_fin % 2 == 0:
            q_fin = sym_from_h(best_h)
            tl = time.time() + min(30, rem() * 0.7)
            q_o, c5_o = lp_coord_loop_sym(q_fin, n_fin, tl, max_rounds=200)
            h_o = build_sym_h(q_o)
            if c5_o < best_c5:
                best_c5 = c5_o
                best_h = h_o.copy()
        if rem() > 20:
            h_pol = best_h + 0.01 * rng.randn(len(best_h))
            h_pol = project_feasible(h_pol, len(best_h) / 2.0)
            tl = time.time() + min(20, rem() * 0.8)
            h_o, c5_o = lp_coord_loop_full(h_pol, tl, max_rounds=80)
            if c5_o < best_c5:
                best_c5 = c5_o
                best_h = h_o.copy()

    # Validate and return
    if best_h is None:
        m_fb = 63
        x_fb = np.linspace(0, 1, m_fb)
        q_fb = 0.5 + 0.3 * np.sin(2 * np.pi * x_fb)
        best_h = build_sym_h(np.clip(q_fb, 0, 1))

    best_h = np.asarray(best_h, dtype=float).reshape(-1)
    n_out = len(best_h)
    best_h = project_feasible(best_h, n_out / 2.0)
    best_c5 = eval_score(best_h)

    return best_h, best_c5, n_out
