    m = n_full // 2

    _G_N = [n_full]
    _G_K = [0]
    _G_D = [0]
    _G_ILO = [0]
    _G_IHI = [0]

    for rnd in range(max_rounds):
        if time.time() > time_limit:
            break
        h = build_sym_h(best_q)
        corrs = corr_raw(h) * dx
        max_val = corrs.max()
        margin = max(3e-6, 0.015 * max_val)
        active = np.where(corrs >= max_val - margin)[0]
        if len(active) > 50:
            order = np.argsort(-corrs[active])
            active = active[order[:50]]
        if len(active) == 0:
            active = np.array([np.argmax(corrs)])

        cons = []
        for kk in active:
            dd = kk - (n_full - 1)
            ilo = max(0, -dd)
            ihi = min(n_full - 1, n_full - 1 - dd)
            _G_N[0] = n_full
            _G_K[0] = kk
            _G_D[0] = dd
            _G_ILO[0] = ilo
            _G_IHI[0] = ihi
            # Can't use closures, so use a wrapper with a unique id
            # Instead, use a different approach: batch all constraints
            pass

        # Use a batched approach: define one function that checks all constraints
        # and returns the minimum violation (for SLSQP ineq, we need t - c_k >= 0 for each k)
        # SLSQP requires scalar constraints, so we must have one per k.
        # Workaround: use a list of dicts with pre-bound data via a helper class-free approach.
        # We'll use the "args" trick with a global list indexed by position.
        
        # Actually, let's just use the smooth max at very high beta as a substitute for SLSQP.
        # The smooth max at beta=50000 is essentially the max.
        _G_BETA[0] = 50000.0
        bounds = [(0.0, 1.0)] * m
        try:
            res = minimize(_smo_f, best_q, jac=_smo_g, method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-12})
            q_new = np.clip(res.x, 0, 1)
        except Exception:
            break
        h_new = build_sym_h(q_new)
        c5_new = eval_score(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_q = q_new.copy()
        else:
            break

    return best_q, best_c5, build_sym_h(best_q)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()

    def rem():
        return budget_s - (time.time() - t0)

    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    try:
        h_parent = np.array(initial_h_values, dtype=float).copy().reshape(-1)
    except (NameError, AttributeError):
        h_parent = None

    if h_parent is not None and len(h_parent) >= 16:
        n_p = len(h_parent)
        if n_p % 2 == 0:
            h_parent = project_feasible(h_parent, n_p / 2.0)
            best_h = h_parent
            best_c5 = eval_score(h_parent)
            best_n = n_p
        else:
            n_p_even = n_p - 1
            h_parent = h_parent[:n_p_even]
            h_parent = project_feasible(h_parent, n_p_even / 2.0)
            best_h = h_parent
            best_c5 = eval_score(h_parent)
            best_n = n_p_even
    else:
        m0 = 63
        x0 = np.linspace(0, 1, m0)
        q0 = np.clip(0.5 + 0.3 * np.sin(2 * np.pi * x0), 0, 1)
        best_h = build_sym_h(q0)
        best_c5 = eval_score(best_h)
        best_n = len(best_h)

    # Phase 1: Optimize parent n
    if best_n % 2 == 0 and rem() > 200:
        q_init = sym_from_h(best_h)
        tl = time.time() + min(180, rem() * 0.35)
        q_o, c5_o, h_o = continuation_optimize(q_init, best_n,
            [30, 60, 120, 250, 500, 1000, 2000, 4000], 300, tl)
        if c5_o < best_c5:
            best_c5 = c5_o
            best_h = h_o
            best_n = len(h_o)
        if rem() > 80:
            q_o2, c5_o2, h_o2 = continuation_optimize(q_o, best_n,
                [8000, 15000, 30000, 50000], 200, time.time() + 80)
            if c5_o2 < best_c5:
                best_c5 = c5_o2
                best_h = h_o2

    # Phase 2: Subgradient polish
    if rem() > 100 and best_n % 2 == 0:
        q_cur = sym_from_h(best_h)
        q_sp, c5_sp, h_sp = subgrad_polish(q_cur, best_n, 500, time.time() + min(80, rem() * 0.5))
        if c5_sp < best_c5:
            best_c5 = c5_sp
            best_h = h_sp
        # Re-run smooth max from subgradient result
        if rem() > 80:
            q_rb, c5_rb, h_rb = continuation_optimize(q_sp, best_n,
                [2000, 5000, 10000, 20000], 200, time.time() + 80)
            if c5_rb < best_c5:
                best_c5 = c5_rb
                best_h = h_rb

    # Phase 3: Try other n values
    n_targets = []
    if best_n >= 100:
        n_targets = [best_n - 6, best_n - 4, best_n - 2, best_n + 2, best_n + 4, best_n + 6,
                     best_n - 10, best_n + 10, 80, 96, 110, 140, 156, 168]
    else:
        n_targets = [best_n + 6, best_n + 12, 100, 120, 140]
    # Remove duplicates and invalid
    seen = set()
    n_targets_filt = []
    for nt in n_targets:
        if nt % 2 == 0 and 48 <= nt <= 200 and nt != best_n and nt not in seen:
            seen.add(nt)
            n_targets_filt.append(nt)
    n_targets_filt.sort(key=lambda x: abs(x - best_n))

    for n_try in n_targets_filt[:8]:
        if rem() < 60:
            break
        h_try, q_try = transfer_to_n(best_h, n_try)
        tl = time.time() + min(70, rem() * 0.5)
        q_o, c5_o, h_o = continuation_optimize(q_try, n_try,
            [50, 150, 400, 1000, 3000, 8000], 250, tl)
        if c5_o < best_c5:
            best_c5 = c5_o
            best_h = h_o
            best_n = n_try
        if rem() < 60:
            break
        # Quick subgradient
        q_sp, c5_sp, _ = subgrad_polish(q_o, n_try, 200, time.time() + 30)
        if c5_sp < best_c5:
            best_c5 = c5_sp
            best_h = build_sym_h(q_sp)
            best_n = n_try

    # Phase 4: Random restarts
    n_restarts = min(8, max(2, int(rem() / 80)))
    for i in range(n_restarts):
        if rem() < 60:
            break
        n_r = best_n if i % 3 == 0 else max(64, best_n + rng.choice([-8, -4, 4, 8]))
        n_r = min(180, max(56, n_r))
        if n_r % 2 != 0:
            n_r += 1
        m_r = n_r // 2
        style = i % 4
        h_r = random_init(m_r, rng, style)
        q_r = sym_from_h(h_r)
        tl = time.time() + min(50, rem() * 0.4)
        q_o, c5_o, h_o = continuation_optimize(q_r, n_r,
            [40, 100, 300, 800, 2000], 200, tl)
        if c5_o < best_c5:
            best_c5 = c5_o
            best_h = h_o
            best_n = n_r

    # Phase 5: Final aggressive polish
    if rem() > 40 and best_n % 2 == 0:
        q_fin = sym_from_h(best_h)
        # Very high beta smooth max
        tl = time.time() + min(50, rem() * 0.6)
        q_f, c5_f, h_f = continuation_optimize(q_fin, best_n,
            [5000, 10000, 20000, 40000, 60000], 300, tl)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f
        # Subgradient
        if rem() > 20:
            q_sp, c5_sp, h_sp = subgrad_polish(q_f, best_n, 300, time.time() + 25)
            if c5_sp < best_c5:
                best_c5 = c5_sp
                best_h = h_sp

    # Phase 6: One more round of different n
    if rem() > 40:
        for n_try in [best_n - 2, best_n + 2, best_n - 4, best_n + 4]:
            if rem() < 30:
                break
            if n_try < 48 or n_try > 200 or n_try % 2 != 0:
                continue
            h_try, q_try = transfer_to_n(best_h, n_try)
            q_o, c5_o, h_o = continuation_optimize(q_try, n_try,
                [500, 2000, 8000, 20000], 200, time.time() + 25)
            if c5_o < best_c5:
                best_c5 = c5_o
                best_h = h_o
                best_n = n_try

    # Final validation
    if best_h is None:
        m_fb = 63
        x_fb = np.linspace(0, 1, m_fb)
        q_fb = np.clip(0.5 + 0.3 * np.sin(2 * np.pi * x_fb), 0, 1)
        best_h = build_sym_h(q_fb)
    best_h = np.asarray(best_h, dtype=float).reshape(-1)
    n_out = len(best_h)
    best_h = project_feasible(best_h, n_out / 2.0)
    best_c5 = eval_score(best_h)
    return best_h, best_c5, n_out
