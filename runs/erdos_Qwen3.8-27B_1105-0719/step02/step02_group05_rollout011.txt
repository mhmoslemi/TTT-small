def build_h_from_q(q, n):

def q_from_h(h, n):
    m = n // 2

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corrs = np.correlate(h, 1.0 - h, mode='full')
    return corrs.max() * dx

def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def grad_corr_d(h, d, n):
    """Gradient of correlation at lag d with respect to h."""
    g = np.zeros(n)
    if d >= 0:
        if d < n:
            g[d:] += (1.0 - h[:n - d])
            g[:n - d] -= h[d:]
    else:
        dd = -d
        if dd < n:
            g[:n - dd] += (1.0 - h[dd:])
            g[dd:] -= h[:n - dd]
    return g

def q_grad_from_full_grad(G, n):
    """Convert full-space gradient to q-space gradient."""
    m = n // 2
    if n % 2 == 0:
        # h = [q, 1-q[::-1]]
        # dh/dq for i < m: e_i
        # dh/dq for i >= m: -e_{n-1-i} (reversed)
        # So Gq[i] = G[i] - G[n-1-i] for i in 0..m-1
        return G[:m] - G[m:][::-1]
    else:
        # h = [q, 0.5, 1-q[::-1]]
        # h[i] = q[i] for i < m
        # h[m] = 0.5
        # h[m+1+j] = 1 - q[m-1-j] for j = 0..m-1
        # So Gq[i] = G[i] - G[n-1-i] for i in 0..m-1

    if w_sum < 1e-30:
