def compute_corrs(h):

def c5_exact(h):

    return float(compute_corrs(h).max() * dx)

def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def corr_grad_k(h, k, n):
    """Gradient of correlation at output index k with respect to h."""
    d = k - (n - 1)
    grad = np.zeros(n)
    if d == 0:
        grad[:] = 1.0 - 2.0 * h
    elif d > 0:
        # corr = sum_{i=0}^{n-1-d} h[i+d]*(1-h[i])
        # dc/dh[j] = (1-h[j-d])*[d<=j<=n-1] + (-h[j+d])*[0<=j<=n-1-d]

        grad[idx1] += (1.0 - h[idx1 + d])
        idx2 = np.arange(d, n)  # j where second term active
        grad[idx2] += (-h[idx2 - d])
    return grad

    n = len(h)
    L = 2 * n - 1
    J = np.zeros((L, n))
    for k in range(L):
        J[k] = corr_grad_k(h, k, n)
    return J

    """Solve the epigraph NLP: min s + rho*||h-h0||^2 s.t. corr_k(h) <= s + target_raw... 
    
    Actually, let me reformulate. We want to find h such that all corr_k(h) <= s,
    minimizing s. The target gives us a starting point for s.
    
    Variables: y = [h_0, ..., h_{n-1}, s], length n+1
    Objective: s + rho * sum((h - h0)^2)
    Constraints: corr_k(h) - s <= 0 for all k
                 sum(h) - n/2 = 0
    Bounds: 0 <= h[i] <= 1, 0 <= s <= large

    n = len(h0)
    L = 2 * n - 1
    target_sum = n / 2.0
    
    def obj(y):

        s = y[n]
        return s + rho * np.sum((h - h0) ** 2)

        g = np.zeros(n + 1)
        g[:n] = 2.0 * rho * (h - h0)

        corrs = compute_corrs(h)

    def __init__(self, beta, dx, target_sum):
        self.beta = beta
        self.dx = dx
        self.target_sum = target_sum
    def fun(self, x):
        cd = compute_corrs(x) * self.dx
        return float(logsumexp(self.beta * cd))
    def jac(self, x):
        n = len(x)
        cd = compute_corrs(x) * self.dx
        sh = cd - cd.max()
        w = np.exp(self.beta * sh)
        ws = w.sum()
        if ws < 1e-30:
            w[:] = 1.0 / len(cd)
        else:
            w /= ws
        g = np.zeros(n)
        for k in range(2*n-1):
            if w[k] > 1e-13:
                g += w[k] * corr_grad_k(x, k, n)
        return g * self.dx
    def eq_fun(self, x):
        return x.sum() - self.target_sum
    def eq_jac(self, x):
        return np.ones(len(x))

class EpiProb:
    def __init__(self, h_ref, rho, target_sum, lags):
        self.h_ref = np.asarray(h_ref, dtype=float)
        self.rho = rho
        self.target_sum = target_sum
        self.lags = np.asarray(lags, dtype=int)
        self.n = len(self.h_ref)
    def fun(self, y):
        h = y[:self.n]
        s = y[self.n]
        r = self.rho * float(np.dot(h - self.h_ref, h - self.h_ref)) if self.rho > 0 else 0.0
        return s + r
    def jac(self, y):
        n = self.n
        g = np.zeros(n + 1)
        if self.rho > 0:
            g[:n] = 2.0 * self.rho * (y[:n] - self.h_ref)
        g[n] = 1.0
        return g
    def ineq_fun(self, y):
        h = y[:self.n]
        s = y[self.n]
        return s - compute_corrs(h)[self.lags]
    def ineq_jac(self, y):
        h = y[:self.n]
        n = self.n
        m = len(self.lags)
        J = np.zeros((m, n+1))
        for i in range(m):
            J[i, :n] = -corr_grad_k(h, self.lags[i], n)
            J[i, n] = 1.0
        return J
    def eq_fun(self, y):
        return y[:self.n].sum() - self.target_sum
    def eq_jac(self, y):
        g = np.zeros(self.n + 1)
        g[:self.n] = 1.0
        return g

def make_anti_pal(n, rng):
    h = np.zeros(n)
    m = n // 2
    for i in range(m):
        v = rng.rand()
        h[i] = v
        h[n-1-i] = 1.0 - v
    if n % 2 == 1:
        h[m] = 0.5
    return h

def make_binary(n, rng):
    h = np.zeros(n)
    idx = rng.permutation(n)
    h[idx[:n//2]] = 1.0
    return h
