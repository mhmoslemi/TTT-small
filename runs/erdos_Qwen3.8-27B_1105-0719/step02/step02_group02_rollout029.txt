    gamma = 0.9

        max_c = corrs.max()
        c5_val = max_c * dx

def make_structured_start(n, target_sum, rng, pattern='staircase'):

        n_ones = int(target_sum)
        h[idx[:n_ones]] = 1.0
        frac = target_sum - n_ones
        if frac > 1e-10 and n_ones < n:
            h[idx[n_ones]] = frac
    elif pattern == 'staircase':
        # h descends from 1 to 0
        x = np.arange(n) / n
        h = np.clip(1.0 - 2.0 * x, 0.0, 1.0)
    elif pattern == 'bump':

        x = np.arange(n) / n
        center = 0.5
        width = 0.25
        h = np.exp(-((x - center) / width) ** 2 * 5.0)
    elif pattern == 'sinusoidal':

        h = 0.5 + 0.5 * np.sin(2 * np.pi * x)
    elif pattern == 'two_clumps':

        h = np.exp(-((x - 0.25) / 0.1) ** 2 * 3.0) + np.exp(-((x - 0.75) / 0.1) ** 2 * 3.0)

        idx = rng.permutation(n)
        n_ones = int(target_sum)
        h[idx[:n_ones]] = 1.0

def project_feasible(x, target_sum):
    """Project x onto {h: 0<=h<=1, sum(h)=target_sum}"""
    n = len(x)
    lo = x.min() - 1.0
    hi = x.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] = np.clip(h[mask] + diff / mask.sum(), 0.0, 1.0)
    return h


def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def c5_of_h(h, dx):
    return float(np.max(compute_corrs(h)) * dx)


def grad_corr_k(h, k, n):
    """Gradient of corr[k] w.r.t. h"""
    offset = k - (n - 1)
    grad = np.zeros(n)
    if offset == 0:
        grad[:] = 1.0 - 2.0 * h
        return grad
    i_lo = max(0, -offset)
    i_hi = min(n - 1, n - 1 - offset)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx + offset] += (1.0 - h[idx])
    grad[idx] += (-h[idx + offset])
    return grad

        values = np.where(active_mask, corrs, -1e18)
    else:
        values = corrs
    shifted = values - values.max()
    w = np.exp(beta * shifted)
    wsum = w.sum()
    if wsum < 1e-30:
        w = np.ones(len(corrs)) / len(corrs)
    else:
        w /= wsum
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        grad += w[k] * grad_corr_k(h, k, n)
    grad *= dx
    return grad


def slsqp_opt(h0, dx, target_sum, beta, maxiter, active_mask=None):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    ones_vec = np.ones(n)

    def obj_fn(x):
        return smooth_max_val(x, dx, beta, active_mask)

    def g_fn(x):
        return smooth_max_grad(x, dx, beta, active_mask)

    def cons_fn(x):
        return x.sum() - target_sum

    def cons_jac_fn(x):
        return ones_vec

    cons = [{'type': 'eq', 'fun': cons_fn, 'jac': cons_jac_fn}]
    try:
        res = minimize(obj_fn, h0, jac=g_fn, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        h_out = project_feasible(res.x, target_sum)
    except Exception:
        h_out = project_feasible(h0, target_sum)
    return h_out, c5_of_h(h_out, dx)


def coordinate_descent_sweep(h, dx, n_passes=50):
    """Exact coordinate descent: minimize max_k f_k by optimizing each coord"""
    n = len(h)
    h = h.copy()
    dk = 2 * n - 1
    best_c5 = c5_of_h(h, dx)

    for p in range(n_passes):
        improved = False
        for i in range(n):
            # Compute slope of each f_k w.r.t. h[i]
            slopes = np.zeros(dk)
            for k in range(dk):
                slopes[k] = grad_corr_k(h, k, n)[i]
            # f_k(x) = slopes[k]*x + (corrs[k] - slopes[k]*h[i])
            corrs = compute_corrs(h)
            intercepts = corrs - slopes * h[i]

            # Find minimizer of max_k (slopes[k]*x + intercepts[k]) over [0,1]
            x_candidates = np.array([0.0, 1.0])
            # Add intersections
            for k in range(dk):
                for j in range(k + 1, dk):
                    ds = slopes[k] - slopes[j]
                    if abs(ds) < 1e-15:
                        continue
                    x_int = (intercepts[j] - intercepts[k]) / ds
                    if 0.0 <= x_int <= 1.0:
                        x_candidates = np.append(x_candidates, x_int)

            if len(x_candidates) == 0:
                x_candidates = np.array([h[i]])

            best_x = h[i]
            best_v = np.inf
            for x in x_candidates:
                x = float(np.clip(x, 0.0, 1.0))
                v = (slopes * x + intercepts).max()
                if v < best_v - 1e-16:
                    best_v = v
                    best_x = x

            if abs(best_x - h[i]) > 1e-12:
                # Update and check sum constraint
                h[i] = best_x
                improved = True
                # Project to maintain sum
                s = h.sum()
                if abs(s - n / 2.0) > 1e-10:
                    h = project_feasible(h, n / 2.0)

        if not improved:
            break

    return h, c5_of_h(h, dx)


def nesterov_opt(h0, dx, target_sum, beta, max_iter):
    """Nesterov accelerated projected gradient"""
    n = len(h0)
    h = h0.copy()
    y = h0.copy()
    t_param = 1.0
    best_h = h0.copy()
    best_c5 = c5_of_h(h0, dx)
    L = max(beta * 4.0 * dx, 0.01)
    step = 1.0 / L
    no_improve = 0

    for it in range(max_iter):
        c5_y = c5_of_h(y, dx)
        grad = smooth_max_grad(y, dx, beta)
        z = y - step * grad
        h_new = project_feasible(z, target_sum)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_param * t_param)) / 2.0
        alpha = (t_param - 1.0) / t_new
        y = h_new + alpha * (h_new - h)
        h = h_new

        c5_val = c5_of_h(h_new, dx)
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 5000:
            break
        if no_improve % 2000 == 0:
            step *= 0.5

    return best_h, best_c5


def subgrad_momentum(h0, dx, target_sum, max_iter=20000):
    """Subgradient with momentum on top-k active constraints"""
    n = len(h0)
    h = h0.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    alpha = 0.08
    gamma = 0.9
    no_improve = 0
    top_m = max(3, n // 20)

    for it in range(max_iter):
        corrs = compute_corrs(h)
        c5_val = corrs.max() * dx

        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 5000:
            break

        sorted_idx = np.argsort(corrs)[-top_m:]
        grad = np.zeros(n)
        for k in sorted_idx:
            grad += grad_corr_k(h, k, n)
        grad /= len(sorted_idx)
        grad *= dx

        v = gamma * v - alpha * grad
        h_new = project_feasible(h + v, target_sum)
        new_c5 = c5_of_h(h_new, dx)

        if new_c5 < c5_val - 1e-16:
            h = h_new
            alpha = min(alpha * 1.004, 0.25)
        else:
            alpha *= 0.92
            v *= 0.2

        if alpha < 1e-14:
            break

    return best_h, best_c5


def make_structured_start(n, target_sum, rng, pattern):
    h = np.zeros(n)
    if pattern == 'binary':
        idx = rng.permutation(n)
        n_ones = int(target_sum)
        h[idx[:n_ones]] = 1.0
        frac = target_sum - n_ones
        if frac > 1e-10 and n_ones < n:
            h[idx[n_ones]] = frac
    elif pattern == 'staircase':
        x = np.arange(n) / n
        h = np.clip(1.0 - 2.0 * x, 0.0, 1.0)
    elif pattern == 'bump':
        x = np.arange(n) / n
        h = np.exp(-((x - 0.5) / 0.2) ** 2 * 4.0)
    elif pattern == 'sinusoidal':
        x = np.arange(n) / n
        h = 0.5 + 0.5 * np.sin(2 * np.pi * x + 0.5)
    elif pattern == 'two_clumps':
        x = np.arange(n) / n
        h = np.exp(-((x - 0.25) / 0.12) ** 2 * 3.0) + np.exp(-((x - 0.75) / 0.12) ** 2 * 3.0)
        h = np.clip(h, 0.0, 1.0)
    elif pattern == 'random_smooth':
        h = rng.rand(n)
        kernel = np.ones(7) / 7.0
        h = np.convolve(h, kernel, mode='same')
    else:
        idx = rng.permutation(n)
        h[idx[:int(target_sum)]] = 1.0
    return project_feasible(h, target_sum)


def active_set_polish(h, dx, target_sum, max_outer=8):
    """Iteratively identify active set and optimize with focused smooth max"""
    n = len(h)
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)

    for outer in range(max_outer):
        corrs = compute_corrs(best_h)
        max_c = corrs.max()
        # Active: within 0.5% of max
        active_mask = corrs > max_c * 0.995
        if active_mask.sum() < 1:
            active_mask = np.zeros(2 * n - 1, dtype=bool)
            active_mask[np.argmax(corrs)] = True

        h_opt, c5_val = slsqp_opt(best_h, dx, target_sum, 3000, 300, active_mask)
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h_opt
        else:
            break

    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def remaining():
        return budget_s - (time.time() - t0)

    best_h = None
    best_c5 = float('inf')

    # Initialize from provided construction
    h0 = np.array(initial_h_values, dtype=float)
    n = len(h0)
    dx = 2.0 / n
    target_sum = n / 2.0
    h0 = project_feasible(h0, target_sum)
    best_h = h0.copy()
    best_c5 = c5_of_h(h0, dx)

    # Phase 1: Quick SLSQP warmup
    for beta in [20, 80, 300]:
        if remaining() < 25:
            break
        h_opt, c5_val = slsqp_opt(best_h, dx, target_sum, beta, 300)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_opt

    # Phase 2: Coordinate descent (exact, fast, monotone)
    if remaining() > 60:
        n_passes_cd = 80
        h_cd, c5_cd = coordinate_descent_sweep(best_h, dx, n_passes_cd)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd

    # Phase 3: SLSQP with higher beta
    for beta in [1000, 3000, 8000]:
        if remaining() < 25:
            break
        h_opt, c5_val = slsqp_opt(best_h, dx, target_sum, beta, 400)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_opt

    # Phase 4: More coordinate descent
    if remaining() > 50:
        h_cd, c5_cd = coordinate_descent_sweep(best_h, dx, 100)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd

    # Phase 5: Nesterov to potentially escape
    if remaining() > 60:
        h_nes, c5_nes = nesterov_opt(best_h, dx, target_sum, 1500, 15000)
        if c5_nes < best_c5:
            best_c5 = c5_nes
            best_h = h_nes
            # Follow up with CD
            if remaining() > 30:
                h_cd, c5_cd = coordinate_descent_sweep(best_h, dx, 50)
                if c5_cd < best_c5:
                    best_c5 = c5_cd
                    best_h = h_cd

    # Phase 6: Subgradient momentum
    if remaining() > 50:
        h_sg, c5_sg = subgrad_momentum(best_h, dx, target_sum, 15000)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg
            if remaining() > 25:
                h_cd, c5_cd = coordinate_descent_sweep(best_h, dx, 60)
                if c5_cd < best_c5:
                    best_c5 = c5_cd
                    best_h = h_cd

    # Phase 7: Active set polish
    if remaining() > 25:
        h_pol, c5_pol = active_set_polish(best_h, dx, target_sum)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol

    # Phase 8: Try nearby n values
    n_b = len(best_h)
    n_candidates = sorted(set(list(range(max(20, n_b - 15), n_b + 20)) + [40, 50, 60, 70, 80, 90, 100, 110, 130, 140, 150, 160, 180, 200]))

    for n_try in n_candidates:
        if remaining() < 20:
            break
        if n_try < 20:
            continue
        dx_try = 2.0 / n_try
        ts_try = n_try / 2.0
        x_old = np.arange(len(best_h)) * (2.0 / len(best_h))
        x_new = np.arange(n_try) * (2.0 / n_try)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, ts_try)

        # Quick eval
        c5_interp = c5_of_h(h_interp, dx_try)
        if c5_interp > best_c5 + 0.005:
            continue

        h_opt, c5_val = slsqp_opt(h_interp, dx_try, ts_try, 200, 250)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_opt
            if remaining() > 35:
                h_cd, c5_cd = coordinate_descent_sweep(best_h, dx_try, 50)
                if c5_cd < best_c5:
                    best_c5 = c5_cd
                    best_h = h_cd
                if remaining() > 20:
                    h_opt2, c5_val2 = slsqp_opt(best_h, 2.0 / len(best_h), len(best_h) / 2.0, 5000, 400)
                    if c5_val2 < best_c5:
                        best_c5 = c5_val2
                        best_h = h_opt2

    # Phase 9: Diverse starts with CD
    n_b = len(best_h)
    dx_b = 2.0 / n_b
    ts_b = n_b / 2.0
    patterns = ['staircase', 'bump', 'sinusoidal', 'two_clumps', 'random_smooth', 'binary']
    for pattern in patterns:
        if remaining() < 15:
            break
        h_start = make_structured_start(n_b, ts_b, rng, pattern)
        c5_start = c5_of_h(h_start, dx_b)
        if c5_start > best_c5 + 0.02:
            continue
        h_opt, c5_val = slsqp_opt(h_start, dx_b, ts_b, 500, 300)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_opt
            if remaining() > 25:
                h_cd, c5_cd = coordinate_descent_sweep(best_h, 2.0 / len(best_h), 80)
                if c5_cd < best_c5:
                    best_c5 = c5_cd
                    best_h = h_cd
                if remaining() > 15:
                    h_opt2, c5_val2 = slsqp_opt(best_h, 2.0 / len(best_h), len(best_h) / 2.0, 5000, 400)
                    if c5_val2 < best_c5:
                        best_c5 = c5_val2
                        best_h = h_opt2

    # Phase 10: Random perturbations + CD
    for trial in range(12):
        if remaining() < 12:
            break
        mag = 0.01 + 0.06 * (trial / 12.0)
        h_pert = best_h + rng.normal(0, mag, len(best_h))
        h_pert = project_feasible(h_pert, ts_b)
        h_cd, c5_cd = coordinate_descent_sweep(h_pert, 2.0 / len(best_h), 60)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd
            if remaining() > 20:
                h_opt, c5_val = slsqp_opt(best_h, 2.0 / len(best_h), len(best_h) / 2.0, 3000, 400)
                if c5_val < best_c5:
                    best_c5 = c5_val
                    best_h = h_opt
                    h_cd2, c5_cd2 = coordinate_descent_sweep(best_h, 2.0 / len(best_h), 60)
                    if c5_cd2 < best_c5:
                        best_c5 = c5_cd2
                        best_h = h_cd2

    # Phase 11: Final intensive polish
    n_b = len(best_h)
    dx_b = 2.0 / n_b
    ts_b = n_b / 2.0
    if remaining() > 20:
        h_cd, c5_cd = coordinate_descent_sweep(best_h, dx_b, 100)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd

    if remaining() > 15:
        for beta in [10000, 30000]:
            if remaining() < 10:
                break
            h_final, c5_final = slsqp_opt(best_h, dx_b, ts_b, beta, 600)
            if c5_final < best_c5:
                best_c5 = c5_final
                best_h = h_final
                if remaining() > 10:
                    h_cd, c5_cd = coordinate_descent_sweep(best_h, 2.0 / len(best_h), 50)
                    if c5_cd < best_c5:
                        best_c5 = c5_cd
                        best_h = h_cd

    if remaining() > 10:
        h_pol, c5_pol = active_set_polish(best_h, dx_b, ts_b)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol

    # Final evaluation
    n_final = len(best_h)
    dx_final = 2.0 / n_final
    best_c5 = c5_of_h(best_h, dx_final)

    return best_h, best_c5, n_final
