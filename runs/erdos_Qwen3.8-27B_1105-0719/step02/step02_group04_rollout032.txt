import numpy as np
import time

def eval_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def eval_c5_array(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def build_sym(q, n):
    m = n // 2
    h = np.empty(n)
    h[:m] = q
    h[m:] = 1.0 - q[::-1]
    return h

def to_sym_q(h, n):
    m = n // 2
    return 0.5 * (h[:m] + 1.0 - h[m:][::-1])

def project_to_target(x, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        if np.clip(x - mid, 0.0, 1.0).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.any():
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def corr_rows_active(h, active_idx, n):
    """Build gradient rows for active lags w.r.t. h variables."""
    dx = 2.0 / n
    m = n
    G = np.zeros((len(active_idx), m))
    for row, j in enumerate(active_idx):
        d = j - (n - 1)
        # First term: (1-h[i-d]) for i in [max(d,0), min(n-1, n-1+d)]
        lo1 = max(0, d)
        hi1 = min(n - 1, n - 1 + d)
        if lo1 <= hi1:
            G[row, lo1:hi1+1] += (1.0 - h[lo1 - d:hi1 + 1 - d])
        # Second term: -h[i+d] for i in [max(0,-d), min(n-1, n-1-d)]
        lo2 = max(0, -d)
        hi2 = min(n - 1, n - 1 - d)
        if lo2 <= hi2:
            G[row, lo2:hi2+1] += -h[lo2 + d:hi2 + 1 + d]
    G *= dx
    return G

def corr_rows_q(h, active_idx, n):
    """Build gradient rows for active lags w.r.t. q variables (symmetric)."""
    m = n // 2
    Gh = corr_rows_active(h, active_idx, n)
    # Chain rule: d c_j / d q[a] = dh[a]/dq[a] * G_h[a] + dh[2m-1-a]/dq[a] * G_h[2m-1-a]
    # = G_h[a] - G_h[2m-1-a]
    Gq = Gh[:m] - Gh[n-1:m-1:-1]
    return Gq

def select_active(c_arr, margin=0.002, pad=2):
    """Select active set: lags near the maximum."""
    mx = c_arr.max()
    thresh = mx - margin
    idx = np.where(c_arr >= thresh)[0]
    # Add neighbors
    expanded = set()
    for a in idx:
        for delta in range(-pad, pad+1):
            k = a + delta
            if 0 <= k <= 2*len(c_arr) - 3:  # 2n-2 max index... wait c_arr has 2n-1 elements
                expanded.add(k)
    # c_arr has 2*n - 1 elements, indices 0 to 2*n-2
    n = (len(c_arr) + 1) // 2
    expanded = {k for k in expanded if 0 <= k <= 2*n-2}
    return np.sort(np.array(expanded))

def target_solve(G, r, reg=0.01):
    """Solve min ||delta||^2 subject to G @ delta ≈ r, with regularization."""
    # Augmented system: [G; sqrt(reg)*I] @ delta ≈ [r; 0]
    D = G.shape[1]
    I = np.eye(D) * np.sqrt(reg)
    A = np.vstack([G, I])
    b = np.concatenate([r, np.zeros(D)])
    try:
        delta, _, _, _ = np.linalg.lstsq(A, b, rcond=None)
    except np.linalg.LinAlgError:
        delta = np.zeros(D)
    if not np.all(np.isfinite(delta)):
        delta = np.zeros(D)
    return delta

def active_step_sym(q, n, margin=0.002, target_frac=0.7, reg=0.01, max_steps=9):

    n_old = len(h)

    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)

    h_new = np.interp(x_new, x_old, h)

        lo2, hi2 = max(0, -d), min(n - 1, n - 1 - d)
        if lo2 <= hi2:
            G[row, lo2:hi2 + 1] += -h[lo2 + d:hi2 + 1 + d]
    return G * dx

def grad_rows_q(h, active, n):
    m = n // 2
    Gh = grad_rows_h(h, active, n)
    return Gh[:m] - Gh[n - 1:m - 1:-1]

def get_active(c_arr, margin=0.002, pad=2):
    mx = c_arr.max()
    idx = np.where(c_arr >= mx - margin)[0]
    n = (len(c_arr) + 1) // 2
    max_idx = 2 * n - 2
    expanded = set()
    for a in idx:
        a = int(a)
        for dlt in range(-pad, pad + 1):
            k = a + dlt
            if 0 <= k <= max_idx:
                expanded.add(k)
    return np.array(sorted(expanded), dtype=int)

def solve_ls(gmat, r, reg=0.01):
    D = gmat.shape[1]
    A = np.vstack([gmat, np.sqrt(reg) * np.eye(D)])
    b = np.concatenate([r, np.zeros(D)])
    try:
        delta, _, _, _ = np.linalg.lstsq(A, b, rcond=None)
    except Exception:
        delta = np.zeros(D)
    if not np.all(np.isfinite(delta)):
        delta = np.zeros(D)
    return delta

def one_step_sym(q, n, margin=0.002, reg=0.01, eta_scale=0.3):
    m = n // 2
    h = build_sym(q, n)
    c_arr = eval_c5_arr(h)
    c_max = c_arr.max()
    active = get_active(c_arr, margin=margin)
    if c_max <= 0.3808:
        eta = 0.00005
    else:
        eta = max(0.0001, eta_scale * (c_max - 0.38075))
    target = c_max - eta
    G = grad_rows_q(h, active, n)
    r = target - c_arr[active]
    delta = solve_ls(G, r, reg=reg)
    if np.allclose(delta, 0) or not np.all(np.isfinite(delta)):
        return q, c_max, False
    best_q = q
    best_c = c_max
    improved = False
    for ss in [1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125, 0.015625, 0.0078125]:
        nq = np.clip(q + ss * delta, 0.0, 1.0)
        nh = build_sym(nq, n)
        nc = eval_c5(nh)
        if nc < best_c - 1e-12:
            best_c = nc
            best_q = nq
            improved = True
    return best_q, best_c, improved

def interp_h(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    xo = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    xn = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_new = np.interp(xn, xo, h)
    return project_h(h_new, n_new / 2.0)

def rand_q(n, rng):
    m = n // 2
    t = np.linspace(0, 1, m)
    q = 0.5 * np.ones(m)
    for k in range(1, 8):
        amp = 0.25 / (1.0 + 0.5 * k)
        q += amp * np.sin(2.0 * np.pi * k * t + rng.uniform(0, 2.0 * np.pi))
    return np.clip(q, 0.0, 1.0)

def active_loop_sym(q, n, t0, budget_s, reserve, max_iter=200, init_margin=0.002, init_reg=0.01):
    margin = init_margin
    reg = init_reg
    stal = 0
    best_q_local = q.copy()
    best_c_local = eval_c5(build_sym(q, n))
    for _ in range(max_iter):
        if (budget_s - (time.time() - t0)) < reserve:
            break
        q, c_loc, imp = one_step_sym(q, n, margin=margin, reg=reg)
        if imp:
            h_new = build_sym(q, n)
            c_new = eval_c5(h_new)
            if c_new < best_c_local - 1e-12:
                best_c_local = c_new
                best_q_local = q.copy()
            stal = 0
            q = to_sym_q(np.clip(q, 0, 1), n)
        else:
            stal += 1
            if stal == 4:
                margin = min(0.01, margin * 1.4)
                reg = max(0.001, reg * 0.5)
            elif stal == 8:
                margin = init_margin
                reg = init_reg
                stal = 0
    return best_q_local, best_c_local

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    RESERVE = 30.0

    best_h = None
    best_c5 = float('inf')

    try:
        h_init = np.array(initial_h_values, dtype=float).reshape(-1)
        if h_init.size > 0 and np.all(np.isfinite(h_init)):
            n_i = len(h_init)
            h_init = project_h(h_init, n_i / 2.0)
            best_h = h_init
            best_c5 = eval_c5(h_init)
    except Exception:
        pass

    if best_h is None:
        n_d = 126
        m_d = n_d // 2
        t = np.linspace(0, 1, m_d)
        q_d = np.clip(0.5 + 0.22 * np.sin(2 * np.pi * 3 * t) + 0.12 * np.sin(2 * np.pi * 7 * t + 1.0) + 0.06 * np.cos(2 * np.pi * 11 * t + 0.5), 0, 1)
        best_h = build_sym(q_d, n_d)
        best_c5 = eval_c5(best_h)

    def rem():
        return budget_s - (time.time() - t0)

    # Phase 1: Symmetrize and run active-set on current n
    n_cur = len(best_h)
    if n_cur % 2 != 0:
        n_cur -= 1
        best_h = interp_h(best_h, n_cur)
        best_c5 = eval_c5(best_h)

    q = to_sym_q(best_h, n_cur)
    h_s = build_sym(q, n_cur)
    c_s = eval_c5(h_s)
    if c_s < best_c5:
        best_h = h_s
        best_c5 = c_s
        q = to_sym_q(best_h, n_cur)

    q, c_after = active_loop_sym(q, n_cur, t0, budget_s, RESERVE, max_iter=150, init_margin=0.002, init_reg=0.01)
    h_after = build_sym(q, n_cur)
    c_after = eval_c5(h_after)
    if c_after < best_c5:
        best_h = h_after
        best_c5 = c_after

    # Second round with different parameters
    if rem() > RESERVE + 10:
        q = to_sym_q(best_h, n_cur)
        q, c_after2 = active_loop_sym(q, n_cur, t0, budget_s, RESERVE, max_iter=100, init_margin=0.001, init_reg=0.003)
        h_after2 = build_sym(q, n_cur)
        c_after2 = eval_c5(h_after2)
        if c_after2 < best_c5:
            best_h = h_after2
            best_c5 = c_after2

    # Phase 2: Try multiple n values
    n_base = len(best_h)
    n_candidates = sorted(set([max(48, n_base - 12), max(48, n_base - 6), n_base + 6, n_base + 12, n_base + 18]))
    n_candidates = [n for n in n_candidates if n % 2 == 0 and n != n_base and 40 <= n <= 220]

    for n_try in n_candidates:
        if rem() < RESERVE + 10:
            break
        h_t = interp_h(best_h, n_try)
        q_t = to_sym_q(h_t, n_try)
        q_t, c_t = active_loop_sym(q_t, n_try, t0, budget_s, RESERVE, max_iter=80, init_margin=0.002, init_reg=0.01)
        h_t = build_sym(q_t, n_try)
        c_t = eval_c5(h_t)
        if c_t < best_c5 - 1e-12:
            best_h = h_t
            best_c5 = c_t

    # Phase 3: Random restarts
    max_restarts = min(int(rem() / 15), 25)
    for trial in range(max_restarts):
        if rem() < RESERVE + 8:
            break
        n_r = len(best_h)
        if n_r % 2 != 0:
            n_r -= 1
        q_r = rand_q(n_r, rng)
        q_r, c_r = active_loop_sym(q_r, n_r, t0, budget_s, RESERVE, max_iter=30, init_margin=0.003, init_reg=0.05)
        h_r = build_sym(q_r, n_r)
        c_r = eval_c5(h_r)
        if c_r < best_c5 - 1e-12:
            best_h = h_r
            best_c5 = c_r

    # Phase 4: Local random perturbation around best
    if rem() > RESERVE + 15:
        n_p = len(best_h)
        if n_p % 2 == 0:
            q_p = to_sym_q(best_h, n_p)
            for _ in range(int(rem() / 10)):
                if rem() < RESERVE + 8:
                    break
                perturb_scale = rng.choice([0.005, 0.01, 0.02, 0.03])
                q_pert = np.clip(q_p + rng.normal(0, perturb_scale, n_p // 2), 0, 1)
                q_pert, c_pert = active_loop_sym(q_pert, n_p, t0, budget_s, RESERVE, max_iter=30, init_margin=0.002, init_reg=0.01)
                h_pert = build_sym(q_pert, n_p)
                c_pert = eval_c5(h_pert)
                if c_pert < best_c5 - 1e-12:
                    best_h = h_pert
                    best_c5 = c_pert
                    q_p = q_pert.copy()

    # Phase 5: Full (non-symmetric) optimization
    if rem() > RESERVE + 15:
        n_f = len(best_h)
        h_f = best_h.copy()
        c_f = eval_c5(h_f)
        active_f = get_active(eval_c5_arr(h_f), margin=0.001, pad=2)
        st = 0
        mg = 0.002
        rg = 0.01
        while rem() > RESERVE + 8 and st < 15:
            c_af = eval_c5_arr(h_f)
            G_f = grad_rows_h(h_f, active_f, n_f)
            mx_f = c_af.max()
            eta_f = max(0.0001, 0.3 * (mx_f - 0.38075))
            tgt_f = mx_f - eta_f
            r_f = tgt_f - c_af[active_f]
            d_f = solve_ls(G_f, r_f, reg=rg)
            d_f -= np.mean(d_f)
            imp_f = False
            for ss in [0.5, 0.25, 0.125, 0.0625, 0.03125, 0.015625]:
                nh = project_h(h_f + ss * d_f, n_f / 2.0)
                nc = eval_c5(nh)
                if nc < c_f - 1e-12:
                    c_f = nc
                    h_f = nh
                    imp_f = True
                    break
            if imp_f:
                if c_f < best_c5 - 1e-12:
                    best_h = h_f.copy()
                    best_c5 = c_f
                st = 0
                active_f = get_active(eval_c5_arr(best_h), margin=0.001, pad=2)
            else:
                st += 1
                if st == 4:
                    mg *= 1.5
                    active_f = get_active(eval_c5_arr(h_f), margin=mg, pad=2)
                    rg = max(0.001, rg * 0.5)
                elif st == 8:
                    mg = 0.002
                    rg = 0.01
                    active_f = get_active(eval_c5_arr(h_f), margin=0.001, pad=2)
                    st = 0

    # Phase 6: Final tight polish in symmetric space
    if rem() > 12:
        n_fin = len(best_h)
        if n_fin % 2 == 0:
            q_fin = to_sym_q(best_h, n_fin)
            q_fin, c_fin = active_loop_sym(q_fin, n_fin, t0, budget_s, RESERVE, max_iter=60, init_margin=0.0005, init_reg=0.001)
            h_fin = build_sym(q_fin, n_fin)
            c_fin = eval_c5(h_fin)
            if c_fin < best_c5 - 1e-12:
                best_h = h_fin
                best_c5 = c_fin

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_h(best_h, len(best_h) / 2.0)
    best_c5 = eval_c5(best_h)

    return best_h, best_c5, len(best_h)
