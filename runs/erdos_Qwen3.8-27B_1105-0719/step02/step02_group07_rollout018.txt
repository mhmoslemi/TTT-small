        else:
            if n_lags >= 2 * n - 1 - 1:
                break
            h_cur = best_step_h
    return best_h, best_c


class SymSmoothOpt:
    def __init__(self, n, beta):
        self.n = n
        self.beta = beta
        self.m = n // 2
        self.dx = 2.0 / n

    def obj(self, x):
        h = make_sym_h(x, self.n)
        corrs = np.correlate(h, 1.0 - h, mode='full') * self.dx
        return logsumexp(self.beta * corrs) / self.beta

    def grad(self, x):
        h = make_sym_h(x, self.n)
        corrs = np.correlate(h, 1.0 - h, mode='full') * self.dx
        shifted = corrs - corrs.max()
        w = np.exp(self.beta * shifted)
        ws = w.sum()
        if ws < 1e-30:
            w = np.ones(2 * self.n - 1) / (2 * self.n - 1)
        else:
            w /= ws
        grad_full = np.zeros(self.n)
        for k in range(2 * self.n - 1):
            wk = w[k] * self.beta
            if wk < 1e-15:
                continue
            d = k - (self.n - 1)
            i_lo = max(0, -d)
            i_hi = min(self.n - 1, self.n - 1 - d)
            if i_lo > i_hi:
                continue
            idx = np.arange(i_lo, i_hi + 1)
            grad_full[idx] += wk * (1.0 - h[idx + d])
            grad_full[idx + d] += wk * (-h[idx])
        grad_full *= self.dx
        return grad_full[:self.m] - grad_full[self.n - 1:self.n - 1 - self.m:-1]


def smooth_sym_optimize(q0, n, beta, maxiter=500):
    m = n // 2
    solver = SymSmoothOpt(n, beta)
    try:
        res = minimize(solver.obj, q0, jac=solver.grad,
                       method='SLSQP', bounds=[(0.0, 1.0)] * m,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        h_new = make_sym_h(res.x, n)
        c5 = exact_score(h_new)
        return res.x, c5
    except Exception:
        return None, None


class FullSmoothOpt:
    def __init__(self, n, beta):
        self.n = n
        self.beta = beta
        self.dx = 2.0 / n

    def obj(self, x):
        h = np.clip(x, 0.0, 1.0)
        corrs = np.correlate(h, 1.0 - h, mode='full') * self.dx
        return logsumexp(self.beta * corrs) / self.beta

    def grad(self, x):
        h = np.clip(x, 0.0, 1.0)
        corrs = np.correlate(h, 1.0 - h, mode='full') * self.dx
        shifted = corrs - corrs.max()
        w = np.exp(self.beta * shifted)
        ws = w.sum()
        if ws < 1e-30:
            w = np.ones(2 * self.n - 1) / (2 * self.n - 1)
        else:
            w /= ws
        grad = np.zeros(self.n)
        for k in range(2 * self.n - 1):
            wk = w[k] * self.beta
            if wk < 1e-15:
                continue
            d = k - (self.n - 1)
            i_lo = max(0, -d)
            i_hi = min(self.n - 1, self.n - 1 - d)
            if i_lo > i_hi:
                continue
            idx = np.arange(i_lo, i_hi + 1)
            grad[idx] += wk * (1.0 - h[idx + d])
            grad[idx + d] += wk * (-h[idx])
        return grad * self.dx


def full_smooth_optimize(h0, n, beta, maxiter=400):
    solver = FullSmoothOpt(n, beta)
    try:
        res = minimize(solver.obj, h0, jac=solver.grad, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        h_out = project_feasible(res.x, n / 2.0)
        c5 = exact_score(h_out)
        return h_out, c5
    except Exception:
        return h0, exact_score(h0)


def coordinate_descent_polish(h_cur, n, steps=2000, eps=0.02):
    best_h = h_cur.copy()
    best_c = exact_score(h_cur)
    rng = np.random.RandomState(12345)
    for _ in range(steps):
        i = rng.randint(n)
        for delta in [eps, -eps, eps * 0.5, -eps * 0.5]:
            h_try = best_h.copy()
            h_try[i] = np.clip(h_try[i] + delta, 0.0, 1.0)
            h_try = project_feasible(h_try, n / 2.0)
            c_try = exact_score(h_try)
            if c_try < best_c - 1e-12:
                best_c = c_try
                best_h = h_try
                break
    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def remaining():
        return budget_s - (time.time() - t0)

    try:
        h_parent = np.array(initial_h_values, dtype=float)
    except NameError:
        h_parent = 0.5 * np.ones(126)

    n = len(h_parent)
    if n % 2 != 0:
        h_parent = h_parent[:n - 1]
        n = len(h_parent)

    h_best = project_feasible(h_parent, n / 2.0)
    c5_best = exact_score(h_best)
    m = n // 2
    q_best = make_q_from_h(h_best, n)
    print("Initial: n={}, C5={:.7f}".format(n, c5_best))

    # Phase 1: LP-based active set in full space
    if remaining() > 30:
        h_lsa, c5_lsa = lsa_full_space(h_best, n, max_iters=150, init_lags=25, expand=4)
        if c5_lsa < c5_best - 1e-10:
            c5_best = c5_lsa
            h_best = h_lsa
            q_best = make_q_from_h(h_best, n)
            print("  LSA full: C5={:.7f}".format(c5_best))

    # Phase 2: Symmetric smooth optimization
    for beta in [60, 120, 250, 500, 1000]:
        if remaining() < 20:
            break
        q_new, c5_new = smooth_sym_optimize(q_best, n, beta, maxiter=400)
        if q_new is not None and c5_new < c5_best - 1e-10:
            c5_best = c5_new
            q_best = q_new
            h_best = make_sym_h(q_best, n)
            print("  sym smooth beta={}: C5={:.7f}".format(beta, c5_best))

    # Phase 3: LP active set again (full space)
    if remaining() > 25:
        h_lsa, c5_lsa = lsa_full_space(h_best, n, max_iters=200, init_lags=40, expand=5)
        if c5_lsa < c5_best - 1e-10:
            c5_best = c5_lsa
            h_best = h_lsa
            q_best = make_q_from_h(h_best, n)
            print("  LSA full 2: C5={:.7f}".format(c5_best))

    # Phase 4: Full-space smooth optimization (breaks symmetry)
    if remaining() > 30:
        for beta in [200, 500, 1000]:
            if remaining() < 15:
                break
            h_new, c5_new = full_smooth_optimize(h_best, n, beta, maxiter=400)
            if c5_new < c5_best - 1e-10:
                c5_best = c5_new
                h_best = h_new
                print("  full smooth beta={}: C5={:.7f}".format(beta, c5_best))

    # Phase 5: LP active set (full space, more iterations)
    if remaining() > 25:
        h_lsa, c5_lsa = lsa_full_space(h_best, n, max_iters=300, init_lags=50, expand=5)
        if c5_lsa < c5_best - 1e-10:
            c5_best = c5_lsa
            h_best = h_lsa
            print("  LSA full 3: C5={:.7f}".format(c5_best))

    # Phase 6: Multi-start with perturbations
    n_trials = 30
    for trial in range(n_trials):
        if remaining() < 15:
            break
        scale = 0.05 + 0.5 * (trial / max(n_trials, 1))
        h_pert = h_best + scale * rng.randn(n)
        h_pert = project_feasible(h_pert, n / 2.0)
        h_lsa, c5_lsa = lsa_full_space(h_pert, n, max_iters=80, init_lags=30, expand=3)
        if c5_lsa < c5_best - 1e-10:
            c5_best = c5_lsa
            h_best = h_lsa
            print("  trial {}: C5={:.7f}".format(trial, c5_best))

    # Phase 7: Symmetric smooth from perturbed starts
    for trial in range(20):
        if remaining() < 15:
            break
        scale = 0.05 + 0.4 * (trial / 20.0)
        q_pert = q_best + scale * rng.randn(m)
        q_pert = np.clip(q_pert, 0.0, 1.0)
        q_new, c5_new = smooth_sym_optimize(q_pert, n, 300, maxiter=300)
        if q_new is not None:
            q_new2, c5_new2 = smooth_sym_optimize(q_new, n, 800, maxiter=400)
            if q_new2 is not None and c5_new2 < c5_best - 1e-10:
                c5_best = c5_new2
                q_best = q_new2
                h_best = make_sym_h(q_best, n)
                print("  sym trial {}: C5={:.7f}".format(trial, c5_best))

    # Phase 8: Final LP polish
    if remaining() > 20:
        h_lsa, c5_lsa = lsa_full_space(h_best, n, max_iters=300, init_lags=60, expand=4)
        if c5_lsa < c5_best - 1e-10:
            c5_best = c5_lsa
            h_best = h_lsa
            print("  final LSA: C5={:.7f}".format(c5_best))

    # Phase 9: Coordinate descent polish
    if remaining() > 30:
        h_cd, c5_cd = coordinate_descent_polish(h_best, n, steps=3000, eps=0.015)
        if c5_cd < c5_best - 1e-10:
            c5_best = c5_cd
            h_best = h_cd
            print("  coord descent: C5={:.7f}".format(c5_best))

    # Phase 10: Try nearby n values
    if remaining() > 50:
        for delta in [2, -2, 4, -4, 6, -6]:
            if remaining() < 25:
                break
            n_try = n + delta
            if n_try < 60 or n_try % 2 != 0:
                continue
            x_old = np.linspace(0, 2, n, endpoint=False)
            x_new = np.linspace(0, 2, n_try, endpoint=False)
            h_interp = np.interp(x_new, x_old, h_best)
            h_interp = project_feasible(h_interp, n_try / 2.0)
            h_lsa, c5_lsa = lsa_full_space(h_interp, n_try, max_iters=100, init_lags=30, expand=4)
            if c5_lsa < c5_best - 1e-9:
                c5_best = c5_lsa
                n = n_try
                m = n_try // 2
                h_best = h_lsa
                q_best = make_q_from_h(h_best, n)
                print("  n={}: C5={:.7f}".format(n, c5_best))

    # Phase 11: Final full-space LP + smooth
    if remaining() > 20:
        h_lsa, c5_lsa = lsa_full_space(h_best, n, max_iters=200, init_lags=80, expand=3)
        if c5_lsa < c5_best - 1e-10:
            c5_best = c5_lsa
            h_best = h_lsa
            print("  final LSA 2: C5={:.7f}".format(c5_best))

    if remaining() > 15:
        h_new, c5_new = full_smooth_optimize(h_best, n, 1500, maxiter=500)
        if c5_new < c5_best - 1e-10:
            c5_best = c5_new
            h_best = h_new
            print("  final full smooth: C5={:.7f}".format(c5_best))

    # Final validation
    h_best = np.clip(h_best, 0.0, 1.0)
    h_best = project_feasible(h_best, len(h_best) / 2.0)
    c5_best = exact_score(h_best)

    return h_best, c5_best, len(h_best)
