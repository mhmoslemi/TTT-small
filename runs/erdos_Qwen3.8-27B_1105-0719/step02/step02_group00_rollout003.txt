    dx = 2.0 / n
    h = h.copy()
    c = exact_score(h)
    
    for it in range(max_iters):
        profile = compute_profile(h)
        c = max(profile)
        
        # Active set: constraints within tolerance of max
        tol = max(0.001, 0.01 * (c - min(profile)))
        active = np.where(profile >= c - tol)[0]
        
        # Build gradient of active constraints
        G = build_gradient_rows(h, active, dx)  # shape (len(active), n)
        
        # Find best direction: maximize -sum of active constraint gradients
        # subject to sum(direction) = 0 (to maintain sum constraint)
        # and ||direction|| <= 1
        
        # This is equivalent to: minimize G @ d subject to 1^T d = 0, ||d|| <= 1
        # Solution: d = -proj(G^T @ 1) / ||proj(G^T @ 1)||
        # where proj removes the component along 1
        
        g = G.T @ np.ones(len(active))
        g = g - g.mean()  # project out the constant
        norm = np.linalg.norm(g)
        if norm < 1e-14:
            break
        d = -g / norm

import numpy as np

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx] + 1e-15)
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx] + 1e-15)
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    # Final fix
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

    dx = 2.0 / n

def build_gradient_matrix(h, n, dx, rows=None):
    """Build gradient matrix for specified rows (or all rows if None)."""
    M = 2 * n - 1
    if rows is None:
        rows = np.arange(M)
    G = np.zeros((len(rows), n))
    for k, m in enumerate(rows):
        d = m - (n - 1)
        if d == 0:
            G[k, :] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            high = np.arange(d, n)
            low = np.arange(0, n - d)
            G[k, high] += dx * (1.0 - h[low])
            G[k, low] += -dx * h[high]
        else:
            dp = -d
            low = np.arange(0, n - dp)
            high = np.arange(dp, n)
            G[k, low] += dx * (1.0 - h[high])
            G[k, high] += -dx * h[low]

    n = len(h)
    if n == new_n:
        return h.copy()
    # Use cumulative sum approach
    edges_old = np.linspace(0, 2, n + 1)
    edges_new = np.linspace(0, 2, new_n + 1)
    # Cumulative mass
    cum_old = np.concatenate([[0], np.cumsum(h * (2.0/n))])

            if overlap_hi > overlap_lo:

        width = hi - lo

3. Update the active set

For step 2, I can use SLSQP on the sum of active constraints (which is a smooth nonconvex function).

Actually, let me try a simpler approach that might work well:

**Iterated linearization**: 
1. At current h, compute profile P(h)
2. Identify active set A (top K constraints)
3. Linearize P_m around h for m in A: P_m(h + d) ≈ P_m(h) + G_m @ d
4. Solve: minimize max_{m in A} (P_m(h) + G_m @ d) subject to sum(d) = 0, ||d|| ≤ δ
5. Update h = h + d
6. Repeat

Step 4 is a small LP (or QP if we add a regularization). This is essentially a trust-region method with a linearized objective.

The LP in step 4:
minimize t
subject to: G_m @ d + P_m(h) ≤ t for all m in A
           sum(d) = 0
           -δ ≤ d_i ≤ δ for all i (or ||d||_∞ ≤ δ)

This is a linear program with n+1 variables and K+1 n+1 constraints. For n=70 and K=20, this is very fast.

I can solve this with scipy's linprog or even with a simple active-set method.

Let me implement this:

```python
def linearized_descent(h, max_iters=500, K=10, delta_init=0.1):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    c = exact_score(h)
    delta = delta_init
    
    for it in range(max_iters):
        profile = compute_profile(h)
        c = max(profile)
        
        # Active set: top K constraints
        top_k = np.argsort(profile)[-K:]
        top_k = top_k[np.argsort(profile[top_k])]  # sort by value
        
        # Gradient of active constraints
        G = build_gradient_matrix(h, n, dx, rows=top_k)  # (K, n)
        
        # Solve: min t s.t. G @ d + profile[top_k] <= t, sum(d)=0, |d_i| <= delta
        # This is an LP with n+1 variables
        # Variables: [d_0, ..., d_{n-1}, t]
        # Objective: minimize t -> c_obj = [0,...,0,1]
        # Inequalities: G @ d - t <= -profile[top_k] for each row
        #              d_i <= delta, -d_i <= delta
        # Equality: sum(d) = 0
        
        from scipy.optimize import linprog
        
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        
        # Inequalities: G @ d - t <= -profile[top_k]
        A_ub = np.zeros((K, n + 1))
        A_ub[:, :n] = G
        A_ub[:, -1] = -1.0
        b_ub = -profile[top_k]
        
        # Box constraints: |d_i| <= delta
        # d_i <= delta -> handled by bounds
        # -d_i <= delta -> handled by bounds
        
        # Equality: sum(d) = 0
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = [0.0]
        
        bounds = [(-delta, delta)] * n + [(0, None)]
        
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        
        if res.status == 0:
            d = res.x[:n]
            h_new = repair_feasible(h + d)
            c_new = exact_score(h_new)
            if c_new < c - 1e-14:
                h = h_new
                c = c_new
                delta = min(delta * 1.2, 0.3)
            else:
                delta *= 0.7
                if delta < 1e-6:
                    break
        else:
            delta *= 0.5
            if delta < 1e-6:
                break
    
    return h, c
```

def binary_pattern_score(bits, n):
    """Compute the max overlap count for a binary pattern."""
    dx = 2.0 / n
    profile = np.correlate(bits.astype(np.float64), 1 - bits.astype(np.float64), mode="full")
    return float(np.max(profile)) * dx

def anneal_binary(bits, n, rng, max_iters=5000, T_init=1.0, T_final=0.01):
    """Simulated annealing on binary patterns."""
    bits = bits.copy()
    n_ones = bits.sum()
    max_ones = n // 2
    
    # Ensure exact balance
    if n_ones > max_ones:
        one_idx = np.where(bits == 1)[0]
        to_flip = rng.choice(one_idx, n_ones - max_ones, replace=False)
        bits[to_flip] = 0
    elif n_ones < max_ones:
        zero_idx = np.where(bits == 0)[0]
        to_flip = rng.choice(zero_idx, max_ones - n_ones, replace=False)
        bits[to_flip] = 1
    
    current_score = binary_pattern_score(bits, n)
    best_bits = bits.copy()
    best_score = current_score
    
    T = T_init
    cooling = (T_final / T_init) ** (1.0 / max_iters)
    
    for it in range(max_iters):
        T *= cooling
        
        # Random swap: pick one 1 and one 0, swap them
        one_idx = np.where(bits == 1)[0]
        zero_idx = np.where(bits == 0)[0]
        i = one_idx[rng.randint(len(one_idx))]
        j = zero_idx[rng.randint(len(zero_idx))]
        
        bits[i], bits[j] = bits[j], bits[i]
        new_score = binary_pattern_score(bits, n)
        delta = new_score - current_score
        
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best_bits = bits.copy()
        else:
            bits[i], bits[j] = bits[j], bits[i]  # revert
    
    return best_bits, best_score

```python
import numpy as np
from scipy.optimize import minimize, linprog
import time

def repair_feasible(vec, ):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx] + 1e-15)
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx] + 1e-15)
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def build_gradient_matrix(h, n, dx, rows=None):
    M = 2 * n - 1
    if rows is None:
        rows = np.arange(M)
    G = np.zeros((len(rows), n))
    for k, m in enumerate(rows):
        d = int(m) - (n - 1)
        if d == 0:
            G[k, :] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            G[k, d:n] += dx * (1.0 - h[0:n-d])
