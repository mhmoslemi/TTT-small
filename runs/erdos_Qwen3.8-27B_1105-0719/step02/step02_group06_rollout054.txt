import numpy as np
from scipy.optimize import linprog, minimize
import time

def eval_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def corr_raw(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(x, target_sum):
    n = len(x)
    x = np.clip(np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0), 0.0, 1.0)
    if abs(x.sum() - target_sum) < 1e-12:
        return x.copy()
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(x + mid, 0.0, 1.0).sum()
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    v = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target_sum - v.sum()
    if abs(residual) > 1e-13:
        if residual > 0:
            order = np.argsort(v)
            for i in order:
                add = min(residual, 1.0 - v[i])
                v[i] += add
                residual -= add
                if abs(residual) < 1e-13:
                    break
        else:
            order = np.argsort(-v)
            for i in order:
                sub = min(-residual, v[i])
                v[i] -= sub
                residual += sub
                if abs(residual) < 1e-13:
                    break
    return np.clip(v, 0.0, 1.0)

def build_sym_h(q):
    m = len(q)
    h = np.empty(2 * m)
    h[:m] = q
    h[m:] = 1.0 - q[::-1]
    return h

def sym_from_h(h):
    n = len(h)
    m = n // 2
    q = np.empty(m)
    for i in range(m):
        q[i] = 0.5 * (h[i] + (1.0 - h[n - 1 - i]))
    return np.clip(q, 0.0, 1.0)

def get_grad_ell(h, ell):
    """Gradient of raw correlation at lag ell with respect to h."""
    n = len(h)
    d = ell - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -d)
    i_hi = min(n - 1, n - 1 - d)
    if i_lo <= i_hi:
        idx = np.arange(i_lo, i_hi + 1)
        jdx = idx + d
        grad[idx] += (1.0 - h[jdx])
        grad[jdx] += (-h[idx])
    return grad

def get_all_grads(h):
    """Get gradients for all lags. Returns list of gradient vectors."""
    n = len(h)
    L = 2 * n - 1
    grads = []
    for ell in range(L):
        grads.append(get_grad_ell(h, ell))
    return np.array(grads)  # shape (L, n)

def lp_tangent_full(h, rho=0.1, max_iters_check=50):
    """
    Solve the LP tangent subproblem for full vector h.
    Returns (new_h, improved, alpha_used).
    """
    n = len(h)
    L = 2 * n - 1
    corrs = corr_raw(h)
    
    # Build LP
    # Decision variables: z (length n), t (scalar) -> total n+1
    # Minimize t: c = [0,...,0, 1]
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    # Constraints: for each ell, grad_ell^T z - t <= grad_ell^T h - R_ell
    A_ub = np.zeros((L, n + 1))
    b_ub = np.zeros(L)
    
    for ell in range(L):
        g = get_grad_ell(h, ell)
        A_ub[ell, :n] = g
        A_ub[ell, n] = -1.0
        b_ub[ell] = g @ h - corrs[ell]
    
    # Equality: sum(z) = n/2
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    # Bounds: max(0, h_i - rho) <= z_i <= min(1, h_i + rho), 0 <= t
    lb = np.maximum(0.0, h - rho)
    ub = np.minimum(1.0, h + rho)
    bounds = list(zip(lb, ub))
    bounds.append((0.0, None))  # t >= 0
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if not res.success:
        return h.copy(), False, 0.0
    
    z_star = res.x[:n]
    d = z_star - h
    
    # Line search
    best_score = eval_score(h)
    best_alpha = 0.0
    for alpha in [1.0, 0.8, 0.6, 0.4, 0.2]:
        h_try = project_feasible(h + alpha * d, n / 2.0)
        sc = eval_score(h_try)
        if sc < best_score - 1e-14:
            best_score = sc
            best_alpha = alpha
    
    if best_alpha > 0:
        return project_feasible(h + best_alpha * d, n / 2.0), True, best_alpha
    return h.copy(), False, 0.0

def lp_tangent_sym(q, rho=0.1):
    """
    Solve the LP tangent subproblem for antiparallel half vector q.
    """
    m = len(q)
    n = 2 * m
    h = build_sym_h(q)
    L = 2 * n - 1
    corrs = corr_raw(h)
    
    # Get gradients wrt h, then transform to q
    # grad_q[r] = grad_h[r] - grad_h[n-1-r]
    c = np.zeros(m + 1)
    c[m] = 1.0
    
    A_ub = np.zeros((L, m + 1))
    b_ub = np.zeros(L)
    
    for ell in range(L):
        g_h = get_grad_ell(h, ell)
        g_q = np.zeros(m)
        for r in range(m):
            g_q[r] = g_h[r] - g_h[n - 1 - r]
        A_ub[ell, :m] = g_q
        A_ub[ell, m] = -1.0
        b_ub[ell] = g_q @ q - corrs[ell]
    
    # No equality constraint needed (antiparallel ensures sum = n/2)
    # Bounds
    lb = np.maximum(0.0, q - rho)
    ub = np.minimum(1.0, q + rho)
    bounds = list(zip(lb, ub))
    bounds.append((0.0, None))
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if not res.success:
        return q.copy(), False, 0.0
    
    q_star = res.x[:m]
    d = q_star - q
    
    best_score = eval_score(h)
    best_alpha = 0.0
    for alpha in [1.0, 0.8, 0.6, 0.4, 0.2]:
        q_try = np.clip(q + alpha * d, 0, 1)
        h_try = build_sym_h(q_try)
        sc = eval_score(h_try)
        if sc < best_score - 1e-14:
            best_score = sc
            best_alpha = alpha
    
    if best_alpha > 0:
        return np.clip(q + best_alpha * d, 0, 1), True, best_alpha
    return q.copy(), False, 0.0

def transfer_line_search(h, a, b, n_grid=40):
    """
    Search over delta for transfer h[a]+=delta, h[b]-=delta.
    Returns best (h_new, score, delta).
    """
    lo = max(-h[a], h[b] - 1.0)
    hi = min(1.0 - h[a], -h[b])
    if hi - lo < 1e-10:
        return h.copy(), eval_score(h), 0.0
    
    best_score = eval_score(h)
    best_delta = 0.0
    
    # Grid search
    deltas = np.linspace(lo, hi, n_grid)
    for delta in deltas:
        h_try = h.copy()
        h_try[a] += delta
        h_try[b] -= delta
        if h_try[a] < -1e-10 or h_try[a] > 1 + 1e-10 or h_try[b] < -1e-10 or h_try[b] > 1 + 1e-10:
            continue
        h_try = np.clip(h_try, 0, 1)
        sc = eval_score(h_try)
        if sc < best_score:
            best_score = sc
            best_delta = delta
    
    # Refine around best
    if best_delta != 0.0:
        width = (hi - lo) / n_grid
        fine_deltas = np.linspace(best_delta - 2*width, best_delta + 2*width, 20)
        for delta in fine_deltas:
            if delta < lo or delta > hi:
                continue
            h_try = h.copy()
            h_try[a] += delta
            h_try[b] -= delta
            h_try = np.clip(h_try, 0, 1)
            sc = eval_score(h_try)
            if sc < best_score:
                best_score = sc
                best_delta = delta
    
    h_new = h.copy()
    h_new[a] = np.clip(h_new[a] + best_delta, 0, 1)
    h_new[b] = np.clip(h_new[b] - best_delta, 0, 1)
    h_new = project_feasible(h_new, len(h) / 2.0)
    return h_new, best_score, best_delta

def transfer_line_search_sym(q, r, n_grid=30):

            h_try = build_sym_h(q_try)

                    best_h = h_new.copy()
                    n_base = n_try
        elif rem() > 40:
            h_cur = ht
            rho_t = 0.07
            for _ in range(30):
                if rem() < 25: break
                z = lp_full(h_cur, rho_t)
                if z is None: break
                d = z - h_cur
                h_new, sc = line_search_full(h_cur, d)
                if sc < best_c5 - 1e-14:
                    best_c5 = sc
                    best_h = h_new
                    n_base = len(h_new)
                    h_cur = h_new
                else:
                    break

    # Phase 4: Systematic transfer search
    if rem() > 50:
        n_cur = len(best_h)
        devs = np.abs(best_h - 0.5)
        hot = np.argsort(-devs)[:30]
        improved = True
        while improved and rem() > 30:
            improved = False
            for i in range(len(hot)):
                if rem() < 30: break
                for j in range(i+1, min(i+6, len(hot))):
                    ht, sc_t = transfer_full(best_h, hot[i], hot[j], ngrid=40)
                    if sc_t < best_c5 - 1e-14:
                        best_c5 = sc_t
                        best_h = ht
                        improved = True

    # Phase 5: Final polish - alternating sym and full with small rho
    if rem() > 30:
        n_fin = len(best_h)
        if n_fin % 2 == 0:
            q_fin = sym_from_h(best_h)
            for _ in range(50):
                if rem() < 15: break
                z = lp_sym(q_fin, 0.03)
                if z is None: break
                d = z - q_fin
                q_new, sc = line_search_sym(q_fin, d)
                if sc < best_c5 - 1e-14:
                    best_c5 = sc
                    best_h = build_sym_h(q_new).copy()
                    q_fin = q_new
                else:
                    break
        for _ in range(50):
            if rem() < 15: break
            z = lp_full(best_h, 0.025)
            if z is None: break
            d = z - best_h
            h_new, sc = line_search_full(best_h, d)
            if sc < best_c5 - 1e-14:
                best_c5 = sc
                best_h = h_new
            else:
                break

    # Phase 6: Random perturbation restart if still time
    if rem() > 80:
        n_r = len(best_h)
        for _ in range(3):
            if rem() < 40: break
            h_pert = best_h + 0.08 * rng.randn(n_r)
            h_pert = project_feasible(h_pert, n_r * 0.5)
            h_cur = h_pert
            rho_r = 0.06
            for _ in range(30):
                if rem() < 30: break
                z = lp_full(h_cur, rho_r)
                if z is None: break
                d = z - h_cur
                h_new, sc = line_search_full(h_cur, d)
                if sc < best_c5 - 1e-14:
                    best_c5 = sc
                    best_h = h_new
                    h_cur = h_new
                else:
                    break

    # Validate
    best_h = np.asarray(best_h, dtype=float).reshape(-1)
    n_out = len(best_h)
    best_h = project_feasible(best_h, n_out * 0.5)
    best_c5 = eval_score(best_h)
    return best_h, best_c5, n_out
