def corr_grad_h(h, d, n):
    """Gradient of c[d] with respect to h, where d = lag offset."""
    g = np.zeros(n)
    # h[j] appears as first factor: c[d] includes h[j]*(1-h[j+d]) if j+d < n
    # and h[j] appears as second factor: c[d] includes h[j-d]*(1-h[j]) if j-d >= 0

        if 0 <= j + d < n:
            gd += (1.0 - h[j + d])
        if 0 <= j - d < n:
            gd += -h[j - d]
        g[j] = gd
    return g

The key idea I want to implement differently from the previous code:

**LP-based cutting plane approach:**

At each iteration:
1. Compute c[d] for all d (the full correlation)
2. Identify active set A (lags where c[d] is close to max)
3. Solve LP:
   min t
   s.t. Σ_m [∂c_d/∂h[m]] * h[m] ≤ t - (c[d] - Σ_m [∂c_d/∂h[m]] * h[m]) for d in A
        (this is the first-order approximation: c[d] ≈ c[d](h*) + ∇c[d](h*)·(h-h*))
        = c[d](h*) + Σ_m ∂c_d/∂h[m] * (h[m]-h*[m])
        ≤ t
        
   So: Σ_m ∂c_d/∂h[m] * h[m] ≤ t - c[d](h*) + Σ_m ∂c_d/∂h[m] * h*[m]
   
   Or equivalently: Σ_m ∂c_d/∂h[m] * h[m] - t ≤ c[d](h*) - Σ_m ∂c_d/∂h[m] * h*[m]
   
   Note: Σ_m ∂c_d/∂h[m] * h*[m] is the directional derivative of c[d] in the direction h* itself, which is not necessarily related to c[d](h*) in a simple way (since c[d] is not linear).

   Actually, the linearization is:
   c[d](h) ≈ c[d](h*) + ∇c[d](h*)·(h - h*)
            = c[d](h*) + Σ_m ∂c_d/∂h[m] * h[m] - Σ_m ∂c_d/∂h[m] * h*[m]

   So the constraint c[d](h) ≤ t becomes:
   c[d](h*) + Σ_m ∂c_d/∂h[m] * h[m] - Σ_m ∂c_d/∂h[m] * h*[m] ≤ t
   
   Rearranging:
   Σ_m ∂c_d/∂h[m] * h[m] - t ≤ Σ_m ∂c_d/∂h[m] * h*[m] - c[d](h*)

   Let me define: b_d = Σ_m ∂c_d/∂h[m] * h*[m] - c[d](h*)

   And a_d = ∇c[d](h*) (the gradient vector)

   Then the LP is:
   min t
   s.t. a_d · h - t ≤ b_d  for d in A
        Σ h[m] = n/2
        0 ≤ h[m] ≤ 1

4. Solve the LP to get h_new
5. Check if C5(h_new) < C5(h*):
   - If yes: accept, go to step 1
   - If no: try a shorter step. Set h_try = h* + α*(h_new - h*) for α ∈ {0.5, 0.25, 0.125, ...}
     Accept the first one that improves.
   - If no improvement at all: expand the active set and retry.

This is essentially a "linearized" approach that should work well for the nonconvex bilinear problem because:
- Near a local minimum, the first-order approximation is good
- The LP respects all the box and sum constraints exactly
- We always validate with the true objective

The advantage over the previous L-BFGS approach:

- It naturally handles the constraints (box, sum) without projection
- It's exact for the linearized problem, so convergence should be faster

import numpy as np
from scipy.optimize import linprog
import time


def eval_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))


def corr_grad(h, d, n):
    g = np.zeros(n)
    if d >= 0:
        g[d:n] += (1.0 - h[0:n - d])
        g[0:n - d] -= h[d:n]
    else:
        g[0:n + d] += (1.0 - h[-d:n])
        g[-d:n] -= h[0:n + d]
    return g


def correlation_scaled(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)


def select_active_scaled(h, margin=0.005, pad=2):
    cv = correlation_scaled(h)
    mx = cv.max()
    thresh = mx - margin * mx - 1e-10
    idx = np.where(cv >= thresh)[0]
    expanded = set()
    for a in idx:
        for delta in range(-pad, pad + 1):
            k = int(a) + delta
            if 0 <= k <= 2 * len(h) - 2:
                expanded.add(k)
    return sorted(expanded)


def solve_lp(h_star, n, active_indices):
    raw_c = np.correlate(h_star, 1.0 - h_star, mode='full')
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_ub_rows = []
    b_ub_vals = []
    for j in active_indices:
        d = j - (n - 1)
        g = corr_grad(h_star, d, n)
        b_d = float(np.dot(g, h_star)) - float(raw_c[j])
        row = np.empty(n + 1)
        row[:n] = g
        row[n] = -1.0
        A_ub_rows.append(row)
        b_ub_vals.append(b_d)
    A_ub = np.array(A_ub_rows) if A_ub_rows else np.empty((0, n + 1))
    b_ub = np.array(b_ub_vals) if b_ub_vals else np.empty(0)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        return res.x[:n].copy()
    return None


def project_to_target(x, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(x - mid, 0.0, 1.0)) > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
    return np.clip(h, 0.0, 1.0)

def interp_to(h, n_new):
    n_old = len(h)
    if n_new == n_old:
        return h.copy()
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h)
    return project_to_target(h_new, n_new / 2.0)

def lp_optimize(h0, n, max_iters=500, tol=1e-11, t0=None, budget_s=1100.0):

    active = select_active_scaled(h, margin=0.005, pad=2)
    no_improve = 0
    for it in range(max_iters):

        h_new = solve_lp(h, n, active)
        if h_new is None:
            active = select_active_scaled(h, margin=0.05, pad=4)
            continue
        h_new = project_to_target(h_new, n / 2.0)

            best_c5 = c5_new

                    best_h = h.copy()
                    best_c5 = c5_try
                    accepted = True
                    break
        if not accepted:

            if no_improve >= 8:

    h = build_sym(q, n)
    best_h = h.copy()
    best_c5 = eval_c5(h)
    active = select_active_scaled(h, margin=0.005, pad=2)
    no_improve = 0
    for it in range(max_iters):
        if t0 is not None and remaining_time(t0, budget_s) < 30.0:
            break
        h_new = solve_lp(h, n, active)
        if h_new is None:
            active = select_active_scaled(h, margin=0.05, pad=4)
            continue
        q_new = to_sym_q(h_new, n)
        h_new = build_sym(q_new, n)
        c5_new = eval_c5(h_new)
        accepted = False
        if c5_new < best_c5 - tol:
            h = h_new.copy()
            best_h = h.copy()
            best_c5 = c5_new
            accepted = True
        else:
            q_cur = to_sym_q(h, n)
            for alpha in [0.5, 0.25, 0.125]:
                q_try = q_cur + alpha * (q_new - q_cur)
                h_try = build_sym(q_try, n)
                c5_try = eval_c5(h_try)
                if c5_try < best_c5 - tol:
                    h = h_try.copy()
                    best_h = h.copy()
                    best_c5 = c5_try
                    accepted = True
                    break
        if not accepted:
            no_improve += 1
            if no_improve >= 3:
                active = select_active_scaled(h, margin=0.01, pad=1)
                no_improve = 0
            if no_improve >= 10:
                break
        else:
            no_improve = 0
            if it % 15 == 0:
                active = select_active_scaled(h, margin=0.005, pad=2)
    return best_h, best_c5


def small_perturbation(h, n, rng, n_pert=50, max_iters=100):
    idx = rng.choice(n, size=min(n_pert, n), replace=False)
    h_try = h.copy()
    h_try[idx] += rng.uniform(-0.05, 0.05, size=len(idx))
    h_try = project_to_target(h_try, n / 2.0)
    return sym_lp_optimize(h_try, n, max_iters=max_iters)[0]


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')

    try:
        h_init = np.asarray(initial_h_values, dtype=float).reshape(-1)
        if h_init.size > 0 and np.all(np.isfinite(h_init)):
            n_init = len(h_init)
            h_init = project_to_target(h_init, n_init / 2.0)
            best_h = h_init.copy()
            best_c5 = eval_c5(h_init)
    except Exception:
        pass

    if best_h is None:
        n_def = 126
        m_def = n_def // 2
        tt = np.linspace(0.0, 1.0, m_def)
        q_def = np.clip(0.5 + 0.2 * np.sin(2.0 * np.pi * 2.0 * tt) + 0.1 * np.cos(4.0 * np.pi * tt), 0.0, 1.0)
        best_h = build_sym(q_def, n_def)
        best_c5 = eval_c5(best_h)

    n_base = len(best_h)
    reserve = max(20.0, 0.04 * budget_s)

    if n_base % 2 == 0:
        q = to_sym_q(best_h, n_base)
        h_sym = build_sym(q, n_base)
        c5_sym = eval_c5(h_sym)
        if c5_sym < best_c5 - 1e-12:
            best_h = h_sym.copy()
            best_c5 = c5_sym

    if remaining_time(t0, budget_s) > reserve + 10:
        h_acc, c5_acc = lp_optimize(best_h, n_base, max_iters=400, tol=1e-12, t0=t0, budget_s=budget_s)
        if c5_acc < best_c5:
            best_h = h_acc.copy()
            best_c5 = c5_acc

    if remaining_time(t0, budget_s) > reserve + 10 and n_base % 2 == 0:
        h_sym, c5_sym = sym_lp_optimize(best_h, n_base, max_iters=300, tol=1e-12, t0=t0, budget_s=budget_s)
        if c5_sym < best_c5:
            best_h = h_sym.copy()
            best_c5 = c5_sym

    n_candidates = []
    for delta in [-4, -2, -1, 1, 2, 4, 6, 8]:
        nc = n_base + delta
        if nc >= 40 and nc % 2 == 0:
            n_candidates.append(nc)
    n_candidates = sorted(set(n_candidates))

    for n_try in n_candidates:
        if remaining_time(t0, budget_s) < reserve + 20:
            break
        h_try = interp_to(best_h, n_try)
        c5_try = eval_c5(h_try)
        h_opt, c5_opt = lp_optimize(h_try, n_try, max_iters=200, tol=1e-11, t0=t0, budget_s=budget_s)
        if c5_opt < best_c5:
            best_h = h_opt.copy()
            best_c5 = c5_opt
            n_base = n_try

    if remaining_time(t0, budget_s) > reserve + 15 and n_base % 2 == 0:
        h_sym2, c5_sym2 = sym_lp_optimize(best_h, n_base, max_iters=200, tol=1e-12, t0=t0, budget_s=budget_s)
        if c5_sym2 < best_c5:
            best_h = h_sym2.copy()
            best_c5 = c5_sym2

    n_restarts = min(10, max(3, int(remaining_time(t0, budget_s) / 20.0)))
    n_cur = len(best_h)
    if n_cur % 2 != 0:
        n_cur -= 1
    for trial in range(n_restarts):
        if remaining_time(t0, budget_s) < reserve + 10:
            break
        h_pert = small_perturbation(best_h, n_cur, rng, n_pert=max(20, n_cur // 4), max_iters=80)
        c5_pert = eval_c5(h_pert)
        if c5_pert < best_c5:
            best_h = h_pert.copy()
            best_c5 = c5_pert

    if remaining_time(t0, budget_s) > 20:
        n_fin = len(best_h)
        h_fin, c5_fin = lp_optimize(best_h, n_fin, max_iters=300, tol=1e-13, t0=t0, budget_s=budget_s)
        if c5_fin < best_c5:
            best_h = h_fin.copy()
            best_c5 = c5_fin

    if remaining_time(t0, budget_s) > 15 and n_fin % 2 == 0:
        h_fin2, c5_fin2 = sym_lp_optimize(best_h, n_fin, max_iters=200, tol=1e-13, t0=t0, budget_s=budget_s)
        if c5_fin2 < best_c5:
            best_h = h_fin2.copy()
            best_c5 = c5_fin2

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_target(best_h, len(best_h) / 2.0)
    best_c5 = eval_c5(best_h)
    return best_h, best_c5, len(best_h)
