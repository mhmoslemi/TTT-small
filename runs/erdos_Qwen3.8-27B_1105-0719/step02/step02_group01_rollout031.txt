import numpy as np
from scipy.optimize import linprog, minimize
from scipy.special import logsumexp
import time

def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_all_grads(h):
    """Compute gradients of all correlations w.r.t. h.
    Returns G[k, j] = d corr_full[k] / d h[j]
    """
    n = len(h)
    G = np.zeros((2*n-1, n))
    one_minus_h = 1.0 - h
    for k in range(2*n-1):
        offset = k - (n-1)
        i_lo = max(0, -offset)
        i_hi = min(n-1, n-1-offset)
        if i_lo > i_hi:
            continue
        idx = np.arange(i_lo, i_hi+1)
        G[k, idx+offset] += one_minus_h[idx]
        G[k, idx] += -h[idx+offset]
    return G

def c5_of_h(h, dx):
    return compute_corrs(h).max() * dx

def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo+hi)*0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def lp_direction(h, dx, target_sum, G, n_active, max_step=0.5):
    """Find a descent direction using LP on active constraints."""
    n = len(h)
    corrs = compute_corrs(h)
    
    # Select active set: top n_active correlations
    active_idx = np.argsort(corrs)[-n_active:]
    active_corrs = corrs[active_idx]
    active_G = G[active_idx, :]  # shape (n_active, n)
    
    # LP: minimize t
    # s.t. corr[k] + G[k,:] @ delta <= t  for active k
    #      sum(delta) = 0
    #      -h[i] <= delta[i] <= (1-h[i]) * max_step  (limit step size)
    
    # Variables: delta[0..n-1], t
    # Objective: minimize t
    c = np.zeros(n+1)
    c[-1] = 1.0
    
    # Inequality constraints: G_active @ delta - t <= -corrs_active
    # i.e., G_active @ delta - t + corrs_active <= 0
    A_ub = np.zeros((len(active_idx), n+1))
    A_ub[:, :n] = active_G
    A_ub[:, -1] = -1.0
    b_ub = -active_corrs
    
    # Equality: sum(delta) = 0
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds
    bounds = []
    for i in range(n):
        lo = -h[i] * max_step
        hi = (1.0 - h[i]) * max_step
        bounds.append((lo, hi))
    bounds.append((0.0, None))  # t is unbounded above, bounded below by 0
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.status != 0:
        return None, corrs.max()
    
    delta = res.x[:n]
    t_val = res.x[-1]
    return delta, t_val

def exact_line_search(h, dx, target_sum, delta, max_iter=40):
    """Find best step size alpha > 0 such that h + alpha*delta is feasible and improves C5."""
    # delta should already satisfy sum=0 (from LP) and bounds
    # Find max feasible alpha
    # For delta[i] > 0: h[i] + alpha*delta[i] <= 1 => alpha <= (1-h[i])/delta[i]
    # For delta[i] < 0: h[i] + alpha*delta[i] >= 0 => alpha >= -h[i]/delta[i] = h[i]/(-delta[i])

    neg = delta < -1e-15
    
    alpha_max = 1.0
    if pos.any():
        alpha_max = min(alpha_max, ((1.0 - h[pos]) / delta[pos]).min())
    if neg.any():
        alpha_max = min(alpha_max, (h[neg] / (-delta[neg])).min())
    
    alpha_max = max(alpha_max, 0.0)
    if alpha_max <= 0:
        return h, c5_of_h(h, dx), 0.0
    
    base_c5 = c5_of_h(h, dx)
    
    # Bisection on [0, alpha_max] to find the alpha that minimizes C5
    # C5 as a function of alpha is: max_k (corr[k] + alpha * G[k,:] @ delta) * dx
    # This is the max of linear functions of alpha, so it's convex in alpha.
    # We want to find alpha that minimizes this convex function.
    # Use ternary search or gradient-based.
    
    # Actually, since it's convex, we can use the property that the minimum
    # is where the subgradient contains 0, or at a boundary.
    # For simplicity, use golden section search.
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a_lo = 0.0
    a_hi = alpha_max * 0.999
    
    c = a_hi - gr * (a_hi - a_lo)
    d = a_lo + gr * (a_hi - a_lo)
    
    hc = project_feasible(h + c * delta, target_sum)
    hd = project_feasible(h + d * delta, target_sum)
    fc = c5_of_h(hc, dx)
    fd = c5_of_h(hd, dx)
    
    for _ in range(max_iter):
        if fc < fd:
            a_hi = d
            d = c
            fd = fc
            c = a_hi - gr * (a_hi - a_lo)
            hc = project_feasible(h + c * delta, target_sum)
            fc = c5_of_h(hc, dx)
        else:
            a_lo = c
            c = d
            fc = fd
            d = a_lo + gr * (a_hi - a_lo)
            hd = project_feasible(h + d * delta, target_sum)
            fd = c5_of_h(hd, dx)
    
    best_alpha = (a_lo + a_hi) / 2.0
    h_best = project_feasible(h + best_alpha * delta, target_sum)
    return h_best, c5_of_h(h_best, dx), best_alpha

import numpy as np
from scipy.optimize import linprog
import time


def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_all_grads(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    omh = 1.0 - h
    for k in range(2 * n - 1):
        offset = k - (n - 1)
        i_lo = max(0, -offset)
        i_hi = min(n - 1, n - 1 - offset)
        if i_lo > i_hi:
            continue
        idx = np.arange(i_lo, i_hi + 1)
        G[k, idx + offset] += omh[idx]
        G[k, idx] += -h[idx + offset]
    return G


def c5_of_h(h, dx):
    return compute_corrs(h).max() * dx


def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-13:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
        else:
            h += diff / n
    return np.clip(h, 0.0, 1.0)


def lp_find_direction(h, G, n_active, max_step):
    n = len(h)
    corrs = compute_corrs(h)
    
    if n_active >= len(corrs):
        active_idx = np.arange(len(corrs))
    else:
        active_idx = np.argsort(corrs)[-n_active:]
    
    active_G = G[active_idx, :]
    active_corrs = corrs[active_idx]
    
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    A_ub = np.zeros((len(active_idx), n + 1))
    A_ub[:, :n] = active_G
    A_ub[:, -1] = -1.0
    b_ub = -active_corrs
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    bounds = []
    for i in range(n):
        bounds.append((-h[i] * max_step, (1.0 - h[i]) * max_step))
    bounds.append((0.0, None))
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status != 0:
            return None
        return res.x[:n]
    except Exception:
        return None


def exact_line_search(h, dx, target_sum, delta, corrs):
    n = len(h)
    pos = delta > 1e-14
    neg = delta < -1e-14
    alpha_max = 1.0
    if pos.any():
        alpha_max = min(alpha_max, ((1.0 - h[pos]) / delta[pos]).min())
    if neg.any():
        alpha_max = min(alpha_max, (h[neg] / (-delta[neg])).min())
    alpha_max = min(alpha_max, 1.0)
    if alpha_max <= 1e-14:
        return h, c5_of_h(h, dx)
    
    base_c5 = c5_of_h(h, dx)
    
    fracs = np.array([0.1, 0.25, 0.5, 0.75, 1.0])
    best_alpha = 0.0
    best_c5 = base_c5
    
    for f in fracs:
        a = f * alpha_max
        h_try = h + a * delta
        h_try = project_feasible(h_try, target_sum)
        c = c5_of_h(h_try, dx)

            best_c5 = c
            best_alpha = a

            best_c5 = c_mid
            lo = mid
        else:
            hi = mid

    return h_best, c5_of_h(h_best, dx)

def pair_search(h, dx, target_sum, G, corrs, max_pairs=20):
    n = len(h)
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    corrs_max = corrs.max()
    active_idx = np.argsort(corrs)[-15:]
    
    # Find promising pairs: h[i] high, h[j] low, gradient favors transfer
    sorted_h = np.argsort(h)
    high_idx = sorted_h[-25:]
    low_idx = sorted_h[:25]
    
    candidates = []
    for i in high_idx:
        for j in low_idx:
            if i == j:
                continue
            # Change in max active corr for unit transfer from i to j
            change_active = (G[active_idx, j] - G[active_idx, i]).max()
            if change_active < -0.01:
                # Check all corrs roughly
                worst = (corrs + (G[:, j] - G[:, i])).max()
                if worst < corrs_max - 0.01:
                    candidates.append((worst, i, j))
    
    candidates.sort()
    
    for _, i, j in candidates[:max_pairs]:
        delta = np.zeros(n)
        delta[i] = -1.0
        delta[j] = 1.0
        h_new, c5_new = exact_line_search(best_h, dx, target_sum, delta, compute_corrs(best_h))
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new
            corrs = compute_corrs(best_h)
            G = compute_all_grads(best_h)
            corrs_max = corrs.max()
            active_idx = np.argsort(corrs)[-15:]
    
    return best_h, best_c5


def make_periodic(n, target_sum, period, offset=0):
    h = np.zeros(n)
    fill = max(1, int(target_sum / n * period))
    for i in range(n):
        h[i] = 1.0 if ((i + offset) % period) < fill else 0.0
    return project_feasible(h, target_sum)


def make_binary_random(n, target_sum, rng):

    idx = rng.permutation(n)
    n_ones = n // 2
    h[idx[:n_ones]] = 1.0
    return project_feasible(h, target_sum)
