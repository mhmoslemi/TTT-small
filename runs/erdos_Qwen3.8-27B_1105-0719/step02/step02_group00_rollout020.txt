At a local minimum, the KKT conditions say:

The active constraints are those with P_k(h) * dx = t (or close to t).

To find the descent direction:
1. Compute the profile P_k(h) * dx
2. Identify active set A = {k : P_k(h) * dx >= t - epsilon}

4. Project g onto the feasible subspace (orthogonal to the sum constraint and respecting bounds)
5. Take a step in the -g direction (descent direction for the active constraints)
6. Line search to find good step size

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0).copy()
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx])
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff

        else:
            idx = int(np.argmax(h))

def build_gradient_rows(h, rows):
    """Build gradient rows for specified profile indices."""
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    G = np.zeros((len(rows), n))
    for r, m in enumerate(rows):
        d = m - (n - 1)
        if d == 0:
            G[r, :] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            G[r, d:n] += dx * (1.0 - h[0:n-d])
            G[r, 0:n-d] += -dx * h[d:n]
        else:
            dp = -d
            G[r, 0:n-dp] += dx * (1.0 - h[dp:n])
            G[r, dp:n] += -dx * h[0:n-dp]
    return G

def solve_epigraph(h_start, maxiter=300, tol=1e-10):

        return x[-1]

        return g

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    # Mass-preserving interpolation using cumulative sums
    xs_old = np.linspace(0, 2, n + 1)
    xs_new = np.linspace(0, 2, new_n + 1)
    # Cumulative integral of h
    cum_old = np.concatenate([[0], np.cumsum(h) * (2.0/n)])
    # Interpolate cumulative and take differences
    cum_new = np.interp(xs_new, xs_old, cum_old)
    h_new = np.diff(cum_new) / (2.0/new_n)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)
