```python
def project_feasible(x, target_sum):
    """Project x to [0,1]^n with sum = target_sum using bisection on offset."""
    n = len(x)
    lo = -n - 1.0
    hi = n + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def c5_of_h(h, dx):
    return float(compute_corrs(h).max() * dx)

def grad_corr_k(h, k, n):
    """Gradient of correlation index k wrt h."""
    offset = k - (n - 1)
    grad = np.zeros(n)
    if offset == 0:
        grad[:] = 1.0 - 2.0 * h
    elif offset > 0:
        # result[k] = sum_{j=offset}^{n-1} h[j] * (1 - h[j-offset])

        #          + (-h[i+offset]) for i in [0, n-offset-1]
        for i in range(offset, n):
            grad[i] += (1.0 - h[i - offset])
        for i in range(0, n - offset):
            grad[i] += (-h[i + offset])
    else:

        for i in range(0, n - d):
            grad[i] += (1.0 - h[i + d])
        for i in range(d, n):
            grad[i] += (-h[i - d])
    return grad
```

```python
def build_constraint_jac(h, n, target_raw):
    """Build Jacobian of all 2n-1 inequality constraints."""
    L = 2 * n - 1
    J = np.zeros((L, n + 1))
    last_col = -np.ones(L)  # for slack variable s

            J[k, d:n] += (-h[:n-d])
```

```python
def epigraph_solve(h_anchor, target_raw, n, rho, top_m=20, maxiter=300):
    """Solve epigraph problem with constraint selection."""
    dx = 2.0 / n
    target_sum = n / 2.0
    L = 2 * n - 1
    
    # Initialize
    h = h_anchor.copy()
    corrs = compute_corrs(h)
    s0 = max(0.0, corrs.max() - target_raw)
    s = max(s0, 0.0)
    y = np.concatenate([h, [s]])
    
    bounds = [(0.0, 1.0)] * n + [(0.0, max(s0 * 2 + 1, target_raw * 0.1))]
    
    def obj_full(y):
        h_y = y[:n]
        s_y = y[-1]
        return s_y + rho * np.sum((h_y - h_anchor)**2) / n
    
    def jac_full(y):
        h_y = y[:n]
        j = np.zeros(n + 1)
        j[:n] = 2.0 * rho * (h_y - h_anchor) / n
        j[-1] = 1.0
        return j
    
    def eq_fun(y):
        return y[:n].sum() - target_sum
    
    def eq_jac(y):
        j = np.zeros(n + 1)
        j[:n] = 1.0
        return j
    
    # Dynamic inequality constraints
    state = {'h_cur': h_anchor.copy()}
    
    def ineq_fun(y):
        h_y = y[:n]
        s_y = y[-1]
        state['h_cur'] = h_y.copy()
        corrs = compute_corrs(h_y)
        # Select top m most violated
        violations = corrs - target_raw - s_y
        idx = np.argsort(violations)[-top_m:]
        return target_raw + s_y - corrs[idx]  # >= 0 required
    
    def ineq_jac(y):
        h_y = y[:n]
        s_y = y[-1]
        corrs = compute_corrs(h_y)
        violations = corrs - target_raw - s_y
        idx = np.argsort(violations)[-top_m:]
        J = np.zeros((len(idx), n + 1))
        for ii, k in enumerate(idx):
            offset = k - (n - 1)
            if offset == 0:
                J[ii, :n] = -(1.0 - 2.0 * h_y)
            elif offset > 0:
                J[ii, offset:n] -= (1.0 - h_y[:n-offset])
                J[ii, :n-offset] -= (-h_y[offset:n])
            else:
                d = -offset
                J[ii, :n-d] -= (1.0 - h_y[d:n])
                J[ii, d:n] -= (-h_y[:n-d])
            J[ii, -1] = 1.0
        return J
    
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    
    try:
        res = minimize(obj_full, y, jac=jac_full, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        h_out = project_feasible(res.x[:n], target_sum)
    except Exception:
        h_out = project_feasible(h_anchor, target_sum)
    
    return h_out, c5_of_h(h_out, 2.0/n)
```

```python
def linearized_epigraph_step(h_cur, target_raw, n, rho=1.0, step_size=1.0, top_m=30):
    """One linearized epigraph step."""
    target_sum = n / 2.0
    L = 2 * n - 1
    corrs = compute_corrs(h_cur)
    
    # Select top m most violated
    violations = corrs - target_raw
    idx = np.argsort(violations)[-top_m:]
    
    # Build QP: minimize s + rho * ||d||^2
    # subject to: corrs[idx] + G[idx] @ d <= target_raw + s
    #              sum(d) = 0  (since sum(h) already = target_sum)
    #              0 <= h_cur + d <= 1
    #              s >= 0
    
    # G is the gradient matrix for selected constraints
    G = np.zeros((len(idx), n))
    for ii, k in enumerate(idx):
        G[ii] = grad_corr_k(h_cur, k, n)
    
    # QP: min s + rho * d'@d
    # s.t. G@v - v[-1] <= target_raw - corrs[idx]  where v = [d; s]
    #      sum(d) = 0
    #      -h_cur <= d <= 1 - h_cur
    #      s >= 0
    
    # This is a standard QP. Let me formulate for CVXPY.
    import cvxpy as cp
    
    d = cp.Variable(n)
    s = cp.Variable()
    objective = cp.Minimize(s + rho * cp.sum_squares(d))
    constraints = []
    for ii in range(len(idx)):
        constraints.append(corrs[idx[ii]] + G[ii] @ d <= target_raw + s)
    constraints.append(cp.sum(d) == 0)
    constraints.append(d >= -h_cur)
    constraints.append(d <= 1 - h_cur)
    constraints.append(s >= 0)
    
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.ECOS, verbose=False)
    except:
        try:
            prob.solve(solver=cp.SCS, verbose=False)
        except:
            return h_cur, None
    
    if d.value is None:
        return h_cur, None
    
    h_new = h_cur + step_size * d.value
    h_new = project_feasible(h_new, target_sum)
    return h_new, s.value
```
