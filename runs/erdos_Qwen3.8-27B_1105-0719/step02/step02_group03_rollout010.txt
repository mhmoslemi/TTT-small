def exact_line_search_2coord(h, j, m, corrs):
    """Find optimal t for swapping h[j]+t, h[m]-t."""
    n = len(h)

        jmk = j - k  # this is i when i+k=j

    if t_lo >= t_hi:

    if f_hi < f_opt:

        # Wait, this isn't right. Let me pick a random 1 and a random 0.
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        # Compute delta
        g_i = grad_coeff(h, i, ks, n)
        g_j = grad_coeff(h, j, ks, n)
        # For binary swap: h[i] goes 1->0 (change -1), h[j] goes 0->1 (change +1)
        # New corrs = corrs + (-1)*g_i + (+1)*g_j = corrs - g_i + g_j
        new_corrs = corrs - g_i + g_j
        new_max = new_corrs.max()
        
        if new_max < best_c5_raw - 1e-14:
            h[i] = 0.0
            h[j] = 1.0
            corrs = new_corrs
            best_c5_raw = new_max
            best_h = h.copy()
            no_improve = 0
        else:
            if temp > 1e-10 and rng.random() < np.exp(-(new_max - best_c5_raw) / temp):
                h[i] = 0.0
                h[j] = 1.0
                corrs = new_corrs
                no_improve = 0
            else:
                no_improve += 1
        
        if no_improve > patience:
            break
    
    return best_h, best_c5_raw * dx

def smooth_max_obj(h, dx, beta):
    corrs = np.correlate(h, 1.0 - h, mode='full')
    m = corrs.max()
    return (m + np.log(np.sum(np.exp(beta * (corrs - m)))) / beta) * dx
