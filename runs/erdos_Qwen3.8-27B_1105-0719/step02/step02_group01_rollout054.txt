import numpy as np
from scipy.optimize import linprog, minimize
from scipy.special import logsumexp
import time


def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = (h < 1.0 - 1e-10)
        else:
            mask = (h > 1e-10)
        cnt = mask.sum()
        if cnt > 0:
            h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)


def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def c5_of_h(h, dx):
    return float(compute_corrs(h).max() * dx)


def grad_corr_k(h, k, n):
    offset = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -offset)
    i_hi = min(n - 1, n - 1 - offset)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx + offset] += (1.0 - h[idx])
    grad[idx] += (-h[idx + offset])
    return grad


def lp_minimax_optimize(h0, dx, target_sum, t0, stop_time, n_active_init=40):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    alpha = 0.4
    n_active = n_active_init
    no_improve = 0
    max_no_improve = 300

    while time.time() < stop_time:
        if no_improve > max_no_improve:
            break

        h_new, improved = lp_single_step(h, dx, target_sum, n_active, alpha)

        if improved:
            c5_new = c5_of_h(h_new, dx)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            h = h_new
            no_improve = 0
            alpha = min(alpha * 1.15, 0.9)
            if no_improve == 0 and n_active < 150:
                n_active = max(30, n_active - 3)
        else:
            no_improve += 1
            alpha *= 0.65
            n_active = min(n_active + 10, 2 * n - 1)
            if alpha < 1e-5:
                break

    return best_h, best_c5


def lp_single_step(h, dx, target_sum, n_active, alpha):
    n = len(h)
    corrs = compute_corrs(h)
    current_max = corrs.max()

    n_act = min(n_active, 2 * n - 1)
    top_idx = np.argsort(corrs)[-n_act:]

    n_vars = n + 1
    c_obj = np.zeros(n_vars)
    c_obj[-1] = 1.0

    A_rows = []
    b_rows = []
    for k in top_idx:
        g = grad_corr_k(h, k, n)
        row = np.zeros(n_vars)
        row[:n] = g
        row[-1] = -1.0
        A_rows.append(row)
        b_rows.append(-corrs[k])

    if len(A_rows) == 0:
        return h, False

    A_ub = np.array(A_rows)
    b_ub = np.array(b_rows)

    A_eq = np.zeros((1, n_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)

    bounds = []
    for i in range(n):
        bounds.append((-h[i] * alpha, (1.0 - h[i]) * alpha))
    bounds.append((None, None))

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
    except Exception:
        return h, False

    if res.status != 0:
        return h, False

    delta = res.x[:n]
    s_pred = res.x[-1]
    if s_pred >= current_max - 1e-10:
        return h, False

    h_best = h.copy()
    c5_best = c5_of_h(h, dx)

    for frac in [1.0, 0.8, 0.6, 0.4, 0.3, 0.2, 0.1, 0.05]:
        h_cand = h + frac * delta
        h_cand = project_feasible(h_cand, target_sum)
        c5_cand = c5_of_h(h_cand, dx)
        if c5_cand < c5_best:
            c5_best = c5_cand
            h_best = h_cand

    improved = c5_best < c5_of_h(h, dx) - 1e-14
    return h_best, improved


def slsqp_finish(h0, dx, target_sum, beta, maxiter):
    n = len(h0)

    def obj(x):
        corrs = compute_corrs(x)
        return float(logsumexp(beta * corrs)) / beta * dx

    def grad_obj(x):
        corrs = compute_corrs(x)
        shifted = corrs - corrs.max()
        w = np.exp(beta * shifted)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(2 * n - 1) / (2 * n - 1)
        else:
            w /= w_sum
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] > 1e-13:
                grad += w[k] * grad_corr_k(x, k, n)
        return grad * dx

    def c_eq(x):
        return x.sum() - target_sum

    def c_eq_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': c_eq, 'jac': c_eq_jac}]
    bounds = [(0.0, 1.0)] * n

    try:
        res = minimize(obj, h0, jac=grad_obj, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        h_out = project_feasible(res.x, target_sum)
    except Exception:
        h_out = project_feasible(h0, target_sum)

    return h_out, c5_of_h(h_out, dx)


def subgradient_descend(h0, dx, target_sum, max_iter, t0, stop_time):

    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    no_improve = 0
    alpha_step = 1.0 / np.sqrt(n)

    for it in range(max_iter):
        if time.time() > stop_time:
            break

        corrs = compute_corrs(h)
        c5_val = corrs.max() * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 3000:
            break

        shifted = corrs - corrs.max()
        w = np.exp(30.0 * shifted)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(2 * n - 1) / (2 * n - 1)
        else:
            w /= w_sum

        grad = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] > 1e-12:
                grad += w[k] * grad_corr_k(h, k, n)
        grad *= dx

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        polyak = (c5_val - best_c5) / (gnorm ** 2 + 1e-14)
        step = min(alpha_step, max(polyak, 1e-10))

        h_new = h - step * grad
        h_new = project_feasible(h_new, target_sum)
        new_c5 = c5_of_h(h_new, dx)

        tries = 0
        while new_c5 > c5_val + 1e-13 and tries < 4:
            step *= 0.5
            h_new = h - step * grad
            h_new = project_feasible(h_new, target_sum)
            new_c5 = c5_of_h(h_new, dx)
            tries += 1

        h = h_new
        alpha_step *= (1.0 - 0.2 / (it + 100))
        alpha_step = max(alpha_step, 1e-8)

    return best_h, best_c5


def make_binary_start(n, target_sum, rng):
    h = np.zeros(n)
    idx = rng.permutation(n)
    n_ones = int(target_sum)
    if n_ones > 0:
        h[idx[:n_ones]] = 1.0
    frac = target_sum - n_ones
    if frac > 1e-10 and n_ones < n:
        h[idx[n_ones]] = frac
    return project_feasible(h, target_sum)


def make_periodic_start(n, target_sum, rng, period):
    h = np.zeros(n)
    fill = max(1, int(target_sum / n * period))
    for i in range(n):
        h[i] = 1.0 if (i % period) < fill else 0.0
    return project_feasible(h, target_sum)


def make_perturbed_start(base_h, rng, scale):
