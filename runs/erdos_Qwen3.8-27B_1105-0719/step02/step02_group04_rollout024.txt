        if d >= 0:
            grad_h[d:n] += wd * (1.0 - h[0:n - d])
            grad_h[0:n - d] -= wd * h[d:n]
        else:
            a = -d
            grad_h[0:n - a] += wd * (1.0 - h[a:n])
            grad_h[a:n] -= wd * h[0:n - a]
    return c_se, grad_h


def lse_c5_and_grad_q(q, n, tau):
    m = n // 2
    h = build_sym(q, n)
    c_se, grad_h = lse_c5_and_grad_h(h, tau)
    grad_q = grad_h[:m] - grad_h[n - 1:m - 1:-1]
    return c_se, grad_q


class SymLSE:
    def __init__(self, n, tau):
        self.n = n
        self.m = n // 2
        self.tau = tau

    def fg(self, q):
        return lse_c5_and_grad_q(q, self.n, self.tau)


class FullLSE:
    def __init__(self, n, tau, mu_sum):
        self.n = n
        self.tau = tau
        self.mu_sum = mu_sum
        self.target = n / 2.0

    def fg(self, z):
        hh = z
        c_se, grad_h = lse_c5_and_grad_h(hh, self.tau)
        sdev = hh.sum() - self.target
        f = c_se + 0.5 * self.mu_sum * sdev * sdev
        g = grad_h + self.mu_sum * sdev
        return f, g


def opt_sym_lse(q0, n, tau_list, maxiter_per=150, gtol=1e-12):
    m = n // 2
    q = q0.copy()
    bounds = [(0.0, 1.0)] * m
    for tau in tau_list:
        model = SymLSE(n, tau)
        try:
            res = minimize(model.fg, q, jac=True, method='L-BFGS-B',
                           bounds=bounds, options={'maxiter': maxiter_per, 'gtol': gtol})
            q_cand = np.clip(res.x, 0.0, 1.0)
            c_old = eval_c5(build_sym(q, n))
            c_new = eval_c5(build_sym(q_cand, n))
            if c_new <= c_old + 1e-10:
                q = q_cand
        except Exception:
            pass
    return q


def subgrad_exact_h(h, n, alpha, margin_frac=0.005, max_active=40):
    dx = 2.0 / n
    cv = np.correlate(h, 1.0 - h, mode='full') * dx
    mx = cv.max()
    thresh = mx - margin_frac * mx
    active = np.where(cv >= thresh)[0]
    if len(active) > max_active:
        order = np.argsort(cv)[::-1][:max_active]
        active = order
    if len(active) == 0:
        active = np.array([np.argmax(cv)])
    grad = np.zeros(n)
    for j in active:
        d = j - (n - 1)
        wd = dx / len(active)
        if d >= 0:
            grad[d:n] += wd * (1.0 - h[0:n - d])
            grad[0:n - d] -= wd * h[d:n]
        else:
            a = -d
            grad[0:n - a] += wd * (1.0 - h[a:n])
            grad[a:n] -= wd * h[0:n - a]
    h_new = h - alpha * grad
    h_new = project_to_target(h_new, n / 2.0)
    return h_new


def interp_to(h, n_new):
    n_old = len(h)
    if n_new == n_old:
        return h.copy()
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h)
    return project_to_target(h_new, n_new / 2.0)


def rem_time(t0, budget_s):
    return budget_s - (time.time() - t0)


def make_random_q(n, rng, amplitude=0.3):
    m = n // 2
    t = np.linspace(0.0, 1.0, m)
    freqs = rng.randint(1, 8, size=4)
    phases = rng.uniform(0, 2 * np.pi, size=4)
    amps = rng.uniform(0.05, amplitude, size=4)
    q = 0.5 + sum(amps[k] * np.sin(2.0 * np.pi * freqs[k] * t + phases[k]) for k in range(4))
    return np.clip(q, 0.0, 1.0)


def active_set_polish_q(q, n, max_iters=300, margin_frac=0.003, max_active=50):
    m = n // 2
    dx = 2.0 / n
    h = build_sym(q, n)
    alpha = 0.01
    stall_count = 0
    for it in range(max_iters):
        cv = np.correlate(h, 1.0 - h, mode='full') * dx
        mx = cv.max()
        thresh = mx - margin_frac * mx
        active_q = np.where(cv >= thresh)[0]
        if len(active_q) > max_active:
            active_q = np.argsort(cv)[::-1][:max_active]
        if len(active_q) == 0:
            active_q = np.array([np.argmax(cv)])
        # Build gradient in q-space
        grad_q = np.zeros(m)
        for j in active_q:
            d = j - (n - 1)
            wd = dx / len(active_q)
            if d >= 0:
                tmp_h = np.zeros(n)
                tmp_h[d:n] += (1.0 - h[0:n - d])
                tmp_h[0:n - d] -= h[d:n]
            else:
                a = -d
                tmp_h = np.zeros(n)
                tmp_h[0:n - a] += (1.0 - h[a:n])
                tmp_h[a:n] -= h[0:n - a]
            grad_q += wd * (tmp_h[:m] - tmp_h[n - 1:m - 1:-1])
        # Try step
        q_new = np.clip(q - alpha * grad_q, 0.0, 1.0)
        h_new = build_sym(q_new, n)
        c_new = eval_c5(h_new)
        if c_new < mx - 1e-12:
            q = q_new
            h = h_new
            stall_count = 0
            alpha = min(alpha * 1.03, 0.05)
        else:
            alpha *= 0.7
            stall_count += 1
            if alpha < 1e-6 or stall_count > 50:
                break
    return q, eval_c5(build_sym(q, n))


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    reserve = max(40.0, 0.06 * budget_s)

    # Initialize from parent
    try:
        h_init = np.array(initial_h_values, dtype=float).reshape(-1)
        if h_init.size > 0 and np.all(np.isfinite(h_init)):
            n_init = len(h_init)
            h_init = project_to_target(h_init, n_init / 2.0)
            best_h = h_init.copy()
            best_c5 = eval_c5(h_init)
    except Exception:
        pass

    if best_h is None:
        n_def = 126
        m_def = n_def // 2
        tt = np.linspace(0.0, 1.0, m_def)
        q_def = np.clip(0.5 + 0.2 * np.sin(2.0 * np.pi * 2.0 * tt) + 0.1 * np.sin(2.0 * np.pi * 5.0 * tt), 0.0, 1.0)
        best_h = build_sym(q_def, n_def)
        best_c5 = eval_c5(best_h)

    n_parent = len(best_h)

    # Phase 1: Symmetric LSE continuation (primary)
    if n_parent % 2 == 0:
        q = to_sym_q(best_h, n_parent)
        h_sym = build_sym(q, n_parent)
        c_sym = eval_c5(h_sym)
        if c_sym < best_c5:
            best_h = h_sym.copy()
            best_c5 = c_sym

        if rem_time(t0, budget_s) > reserve + 60:
            tau_list_1 = [0.5, 0.35, 0.25, 0.18, 0.13, 0.09, 0.065, 0.05, 0.035, 0.025, 0.018, 0.013, 0.01, 0.007, 0.005, 0.003, 0.002]
            q = opt_sym_lse(q, n_parent, tau_list_1, maxiter_per=200, gtol=1e-12)
            h_new = build_sym(q, n_parent)
            c_new = eval_c5(h_new)
            if c_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c_new

    # Phase 2: Second LSE pass with different schedule
    if rem_time(t0, budget_s) > reserve + 40 and len(best_h) % 2 == 0:
        q2 = to_sym_q(best_h, len(best_h))
        tau_list_2 = [0.4, 0.28, 0.2, 0.14, 0.1, 0.07, 0.05, 0.035, 0.025, 0.018, 0.012, 0.008, 0.005, 0.003, 0.002, 0.001]
        q2 = opt_sym_lse(q2, len(best_h), tau_list_2, maxiter_per=250, gtol=1e-13)
        h2 = build_sym(q2, len(best_h))
        c2 = eval_c5(h2)
        if c2 < best_c5:
            best_h = h2.copy()
            best_c5 = c2

    # Phase 3: Active-set subgradient polish
    if rem_time(t0, budget_s) > reserve + 20 and len(best_h) % 2 == 0:
        n_cur = len(best_h)
        q3 = to_sym_q(best_h, n_cur)
        q3, c3 = active_set_polish_q(q3, n_cur, max_iters=400, margin_frac=0.003, max_active=50)
        h3 = build_sym(q3, n_cur)
        if c3 < best_c5:
            best_h = h3.copy()
            best_c5 = c3

    # Phase 4: Full-variable LSE (break symmetry)
    if rem_time(t0, budget_s) > reserve + 30:
        n_cur = len(best_h)
        h_full = best_h.copy()
        for tau in [0.04, 0.02, 0.012, 0.008, 0.005]:
            if rem_time(t0, budget_s) < reserve + 15:
                break
            model = FullLSE(n_cur, tau, mu_sum=2000.0)
            bounds_f = [(0.0, 1.0)] * n_cur
            try:
                res = minimize(model.fg, h_full, jac=True, method='L-BFGS-B',
                               bounds=bounds_f, options={'maxiter': 80, 'gtol': 1e-10})
                h_cand = np.clip(res.x, 0.0, 1.0)
                h_cand = project_to_target(h_cand, n_cur / 2.0)
                c_cand = eval_c5(h_cand)
                if c_cand < best_c5:
                    best_h = h_cand.copy()
                    best_c5 = c_cand
                    h_full = h_cand
            except Exception:
                pass

    # Phase 5: Multi-n exploration
    n_base = len(best_h)
    n_candidates = sorted(set([n_base + d for d in [-8, -6, -4, -2, 2, 4, 6, 8]
                               if n_base + d >= 40 and (n_base + d) % 2 == 0]))
    for n_try in n_candidates:
        if rem_time(t0, budget_s) < reserve + 20:
            break
        h_try = interp_to(best_h, n_try)
        q_try = to_sym_q(h_try, n_try)
        tau_short = [0.3, 0.2, 0.14, 0.1, 0.07, 0.05, 0.035, 0.025, 0.015, 0.01, 0.007, 0.004]
        q_try = opt_sym_lse(q_try, n_try, tau_short, maxiter_per=100, gtol=1e-11)
        h_try = build_sym(q_try, n_try)
        c_try = eval_c5(h_try)
        if c_try < best_c5:
            best_h = h_try.copy()
            best_c5 = c_try

    # Phase 6: Random restarts with LSE
    n_r = len(best_h)
    max_restarts = min(20, max(3, int(rem_time(t0, budget_s) / 20.0)))
    tau_restart = [0.3, 0.2, 0.12, 0.07, 0.04, 0.025, 0.015, 0.008, 0.004]
    for trial in range(max_restarts):
        if rem_time(t0, budget_s) < reserve + 15:
            break
        q_r = make_random_q(n_r, rng, amplitude=0.25)
        q_r = opt_sym_lse(q_r, n_r, tau_restart, maxiter_per=80, gtol=1e-10)
        h_r = build_sym(q_r, n_r)
        c_r = eval_c5(h_r)
        if c_r < best_c5:
            best_h = h_r.copy()
            best_c5 = c_r
            # Continue optimizing from this point
            if rem_time(t0, budget_s) > reserve + 20:
                q_r2 = to_sym_q(best_h, len(best_h))
                tau_cont = [0.02, 0.012, 0.007, 0.004, 0.002]
                q_r2 = opt_sym_lse(q_r2, len(best_h), tau_cont, maxiter_per=150, gtol=1e-12)
                h_r2 = build_sym(q_r2, len(best_h))
                c_r2 = eval_c5(h_r2)
                if c_r2 < best_c5:
                    best_h = h_r2.copy()
                    best_c5 = c_r2

    # Phase 7: Perturbation restarts around current best
    if rem_time(t0, budget_s) > reserve + 20 and len(best_h) % 2 == 0:
        n_p = len(best_h)
        q_base = to_sym_q(best_h, n_p)
        for p_trial in range(min(10, max(2, int(rem_time(t0, budget_s) / 15.0)))):
            if rem_time(t0, budget_s) < reserve + 12:
                break
            pert = rng.uniform(-0.08, 0.08, size=n_p // 2)
            q_p = np.clip(q_base + pert, 0.0, 1.0)
            tau_pert = [0.1, 0.06, 0.04, 0.025, 0.015, 0.008, 0.004, 0.002]
            q_p = opt_sym_lse(q_p, n_p, tau_pert, maxiter_per=120, gtol=1e-11)
            h_p = build_sym(q_p, n_p)
            c_p = eval_c5(h_p)
            if c_p < best_c5:
                best_h = h_p.copy()
                best_c5 = c_p

    # Phase 8: Final fine polish
    if rem_time(t0, budget_s) > 15 and len(best_h) % 2 == 0:
        n_fin = len(best_h)
        q_fin = to_sym_q(best_h, n_fin)
        tau_final = [0.03, 0.02, 0.012, 0.008, 0.005, 0.003, 0.002, 0.0015, 0.001]
        q_fin = opt_sym_lse(q_fin, n_fin, tau_final, maxiter_per=300, gtol=1e-13)
        h_fin = build_sym(q_fin, n_fin)
        c_fin = eval_c5(h_fin)
        if c_fin < best_c5:
            best_h = h_fin.copy()
            best_c5 = c_fin

    # Final active-set polish
    if rem_time(t0, budget_s) > 10 and len(best_h) % 2 == 0:
        n_f2 = len(best_h)
        q_f2 = to_sym_q(best_h, n_f2)
        q_f2, c_f2 = active_set_polish_q(q_f2, n_f2, max_iters=200, margin_frac=0.002, max_active=60)
        h_f2 = build_sym(q_f2, n_f2)
        if c_f2 < best_c5:
            best_h = h_f2.copy()
            best_c5 = c_f2

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_target(best_h, len(best_h) / 2.0)
    best_c5 = eval_c5(best_h)

    return best_h, best_c5, len(best_h)
