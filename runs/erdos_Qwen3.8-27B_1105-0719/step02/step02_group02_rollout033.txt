5. Accept the best improving swap
6. If no improvement, do a random perturbation

import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time


def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h


def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def c5_of_h(h, dx):
    return float(compute_corrs(h).max() * dx)


def smooth_max_val(h, dx, beta):
    return float(logsumexp(beta * compute_corrs(h)) / beta * dx)


def grad_corr_k(h, k, n):
    offset = k - (n - 1)
    grad = np.zeros(n)
    if offset == 0:
        grad[:] = 1.0 - 2.0 * h
        return grad
    i_lo = max(0, -offset)
    i_hi = min(n - 1, n - 1 - offset)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx + offset] += (1.0 - h[idx])
    grad[idx] += (-h[idx + offset])
    return grad


def smooth_max_grad(h, dx, beta):
    n = len(h)
    corrs = compute_corrs(h)
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones_like(corrs) / len(corrs)
    else:
        w /= w_sum
    grad = np.zeros(n)
    for k in range(len(corrs)):
        if w[k] < 1e-12:
            continue
        grad += w[k] * grad_corr_k(h, k, n)
    return grad * dx


def slsqp_opt(h0, dx, target_sum, beta, maxiter):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n

    def obj(x):
        return smooth_max_val(x, dx, beta)

    def g(x):
        return smooth_max_grad(x, dx, beta)

    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h0, jac=g, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        h_out = project_feasible(res.x, target_sum)
    except Exception:
        h_out = project_feasible(h0, target_sum)
    return h_out, c5_of_h(h_out, dx)


def subgrad_momentum(h0, dx, target_sum, max_iter=20000, top_frac=0.15):
    n = len(h0)
    h = h0.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    alpha = 0.1
    gamma = 0.8
    no_improve = 0
    top_m = max(5, int(n * top_frac))

    for it in range(max_iter):
        corrs = compute_corrs(h)
        max_c = corrs.max()
        c5_val = max_c * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 8000:
            break

        sorted_idx = np.argsort(corrs)[-top_m:]
        grad = np.zeros(n)
        for k in sorted_idx:
            grad += grad_corr_k(h, k, n)
        grad /= len(sorted_idx)
        grad *= dx

        v = gamma * v - alpha * grad
        h_new = project_feasible(h + v, target_sum)
        new_c5 = c5_of_h(h_new, dx)

        if new_c5 < c5_val - 1e-16:
            h = h_new
            alpha = min(alpha * 1.005, 0.35)
        else:
            alpha *= 0.93
            v *= 0.2

        if alpha < 1e-14:
            break

    return best_h, best_c5


def make_binary_from_continuous(h, n, target_sum):
    """Convert continuous h to binary with exact sum."""
    order = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    n_ones = int(target_sum)
    frac = target_sum - n_ones
    h_bin[order[:n_ones]] = 1.0
    if frac > 1e-10 and n_ones < n:
        h_bin[order[n_ones]] = frac
    return h_bin


def binary_local_search(h, n, max_time, rng, perturbation_interval=200):
    """Efficient local search on binary h using swap moves."""
    t_start = time.time()
    h = h.copy()
    ones = np.where(h >= 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    # Make it truly binary for the search
    h[ones] = 1.0
    h[zeros] = 0.0
    m = len(ones)

    corrs = np.correlate(h, 1.0 - h, mode='full')
    max_c = corrs.max()
    best_max_c = max_c
    best_h = h.copy()
    no_improve = 0
    it = 0

    while time.time() - t_start < max_time:
        it += 1
        # Focus on the worst k
        k_star = int(np.argmax(corrs))
        lag = k_star - (n - 1)

        # Find contributing 1-positions for C(k_star)
        # C at index k_star: sum_n h[n+k]*(1-h[n]) where k = lag
        # If lag > 0: n from 0 to n-1-lag, term h[n+lag]*(1-h[n])
        #   Position p in ones contributes if p >= lag (as n+lag=p, n=p-lag) and h[p-lag]=0
        #   Or p <= n-1-lag (as n=p) and h[p+lag]=0... wait
        #   Actually: h[n+lag]*(1-h[n]), position p appears as:
        #     n+lag = p => n = p-lag, valid if 0<=p-lag<=n-1-lag, i.e., lag<=p<=n-1
        #       contribution: 1*(1-h[p-lag]) if h[p-lag]=0
        #     n = p => valid if 0<=p<=n-1-lag, i.e., p<=n-1-lag
        #       contribution: h[p+lag]*(1-1) = 0 (always 0 when h[p]=1)
        #   So position p in ones contributes (1-h[p-lag]) to C(k_star) if lag<=p<=n-1 and h[p-lag]=0

        # For lag < 0: let L = -lag
        #   n from L to n-1, term h[n-L]*(1-h[n])
        #   Position p in ones:
        #     n-L = p => n = p+L, valid if L<=p+L<=n-1, i.e., 0<=p<=n-1-L
        #       contribution: 1*(1-h[p+L]) if h[p+L]=0
        #     n = p => valid if L<=p<=n-1
        #       contribution: h[p-L]*(1-1) = 0

        # In both cases, position p in ones contributes (1-h[p +/- L]) where L=|lag|
        # specifically h[other_end] must be 0

        if abs(lag) == 0:
            continue  # C(0)=0 for binary

        L = abs(lag)
        if lag > 0:
            # p contributes if p >= L, p <= n-1, and h[p-L]==0
            contrib_ones = []
            for idx_p in ones:
                if idx_p >= L and idx_p < n and h[idx_p - L] == 0:
                    contrib_ones.append(idx_p)
        else:
            # p contributes if p <= n-1-L, and h[p+L]==0
            contrib_ones = []
            for idx_p in ones:
                if idx_p < n - L and h[idx_p + L] == 0:
                    contrib_ones.append(idx_p)

        # Try swapping contributing ones with zeros
        best_new_max = max_c
        best_swap = None

        candidates = contrib_ones if len(contrib_ones) > 0 else list(ones)
        # Limit candidates for speed
        if len(candidates) > 20:
            candidates = list(rng.choice(candidates, 20, replace=False))

        for p in candidates:
            for q in zeros:
                # Quick check: will this help k_star?
                if lag > 0:
                    # Removing p from ones at position p: loses contribution to C(k_star) if p>=L and h[p-L]==0
                    # Adding q to ones at position q: gains contribution if q>=L and h[q-L]==0
                    # Net effect on C(k_star): -[p>=L and h[p-L]==0] + [q>=L and h[q-L]==0]
                    lose = 1 if (p >= L and h[p - L] == 0) else 0
                    gain = 1 if (q >= L and q < n and h[q - L] == 0) else 0
                    if gain >= lose:
                        continue  # Won't help k_star, skip
                else:
                    lose = 1 if (p < n - L and h[p + L] == 0) else 0
                    gain = 1 if (q < n - L and h[q + L] == 0) else 0
                    if gain >= lose:
                        continue

                # Do the swap and evaluate
                h[p] = 0
                h[q] = 1
                new_corrs = np.correlate(h, 1.0 - h, mode='full')
                new_max = new_corrs.max()
                h[p] = 1
                h[q] = 0

                if new_max < best_new_max - 1e-14:
                    best_new_max = new_max
                    best_swap = (p, q)

        if best_swap is not None:
            p, q = best_swap
            h[p] = 0
            h[q] = 1
            # Update ones and zeros
            ones = ones[ones != p]
            ones = np.append(ones, q)
            ones.sort()
            zeros = zeros[zeros != q]
            zeros = np.append(zeros, p)
            zeros.sort()
            corrs = np.correlate(h, 1.0 - h, mode='full')
            max_c = corrs.max()
            no_improve = 0

            if max_c < best_max_c - 1e-14:
                best_max_c = max_c
                best_h = h.copy()
        else:
            no_improve += 1
            if no_improve >= perturbation_interval:
                # Random perturbation
                n_pert = min(3, m)
                for _ in range(n_pert):
                    i = rng.randint(m)
                    j = rng.randint(len(zeros))
                    pp, qq = ones[i], zeros[j]
                    h[pp], h[qq] = h[qq], h[pp]
                    ones[i], zeros[j] = qq, pp
                ones.sort()
                zeros.sort()
                corrs = np.correlate(h, 1.0 - h, mode='full')
                max_c = corrs.max()
                no_improve = 0

    # Restore best
    h_best = best_h.copy()
    return h_best, best_max_c
