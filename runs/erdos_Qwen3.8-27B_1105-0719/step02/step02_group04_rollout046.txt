                if c_try < best_c:
                    best_q = q.copy()
                    best_c = c_try
                no_progress = 0
                R = min(0.25, R * 1.15)
                found = True
                break

        if not found:
            no_progress += 1
            R *= 0.6
            if R < 1e-7:
                break

    return best_q, best_c


def slp_full(h, n, R_init, time_limit, t0):
    dx = 2.0 / n
    C = corr_scaled(h)
    c_cur = float(C.max())
    best_h = h.copy()
    best_c = c_cur
    R = R_init
    no_progress = 0
    target = n / 2.0

    while (time.time() - t0) < time_limit and no_progress < 30:
        C = corr_scaled(h)
        c_cur = float(C.max())
        c_max = c_cur

        rel_margin = max(0.003, 0.04)
        active = select_active_slp(C, rel_margin, n_pad=2)

        n_vars = n + 1
        c_obj = np.zeros(n_vars)
        c_obj[n] = 1.0

        A_rows = []
        b_rows = []
        for j in active:
            d = j - (n - 1)
            gh = grad_row(h, d, n) * dx
            row = np.zeros(n_vars)
            row[:n] = gh
            row[n] = -1.0
            A_rows.append(row)
            b_rows.append(-C[j])

        if not A_rows:
            break

        A_ub = np.array(A_rows)
        b_ub = np.array(b_rows)

        A_eq = np.zeros((1, n_vars))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)

        bounds = [(-R, R)] * n + [(0.0, c_max + 0.5)]

        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if not res.success or res.x is None:
                R *= 0.7
                if R < 1e-7:
                    break
                no_progress += 1
                continue
        except Exception:
            R *= 0.7
            if R < 1e-7:
                break
            no_progress += 1
            continue

        delta = res.x[:n]
        t_opt = res.x[n]

        if t_opt >= c_cur - 1e-12:
            no_progress += 1
            R *= 0.6
            if R < 1e-7:
                break
            continue

        found = False
        for lam in [1.0, 0.7, 0.5, 0.3, 0.15, 0.07]:
            h_try = h + lam * delta
            h_try = np.clip(h_try, 0.0, 1.0)
            h_try = repair_box_sum(h_try, target)
            c_try = eval_c5(h_try)
            if c_try < c_cur - 1e-11:
                h = h_try
                c_cur = c_try
                if c_try < best_c:
                    best_h = h.copy()
                    best_c = c_try
                no_progress = 0
                R = min(0.25, R * 1.15)
                found = True
                break

        if not found:
            no_progress += 1
            R *= 0.6
            if R < 1e-7:
                break

    return best_h, best_c


def interp_h(h_old, n_new):
    n_old = len(h_old)
    if n_new == n_old:
        return h_old.copy()
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return repair_box_sum(h_new, n_new / 2.0)


def local_search_sym(q, n, time_limit, t0):
    m = n // 2
    h = build_sym(q, n)
    c_cur = eval_c5(h)
    best_q = q.copy()
    best_c = c_cur
    step_sizes = [0.05, 0.03, 0.02, 0.01, 0.005, 0.003, 0.001]
    rng = np.random.RandomState(123)

    while (time.time() - t0) < time_limit:
        improved = False
        for ss in step_sizes:
            for _ in range(30):
                if (time.time() - t0) >= time_limit:
                    break
                idx = rng.randint(0, m)
                for direction in [-1, 1]:
                    q_try = q.copy()
                    q_try[idx] = np.clip(q_try[idx] + direction * ss, 0.0, 1.0)
                    h_try = build_sym(q_try, n)
                    c_try = eval_c5(h_try)
                    if c_try < c_cur - 1e-11:
                        q = q_try
                        h = h_try
                        c_cur = c_try
                        improved = True
                        if c_try < best_c:
                            best_q = q.copy()
                            best_c = c_try
                        break
        if not improved:
            break

    return best_q, best_c


def polish_lbgfs(q, n, active_k, mu, maxiter=100, gtol=1e-11):
    m = n // 2
    h0 = build_sym(q, n)
    t0 = eval_c5(h0)

    def model_fg(z):
        qq = z[:m]
        t = z[m]
        hh = build_sym(qq, n)
        cv = corr_scaled(hh)
        vals = cv[active_k]
        exc = np.maximum(0.0, vals - t)
        f = t + 0.5 * mu * np.dot(exc, exc)
        gf = np.zeros(m + 1)
        gf[m] = 1.0 - mu * exc.sum()
        gq = np.zeros(m)
        dx = 2.0 / n
        for i in range(len(active_k)):
            if exc[i] > 0.0:
                d = active_k[i] - (n - 1)
                gq += mu * exc[i] * sym_grad(hh, d, n) * dx
        gf[:m] = gq
        return f, gf

    z0 = np.concatenate([q, [t0]])
    bounds = [(0.0, 1.0)] * m + [(0.0, t0 + 0.5)]
    try:
        res = minimize(model_fg, z0, jac=True, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'gtol': gtol})
        q_opt = np.clip(res.x[:m], 0.0, 1.0)
    except Exception:
        return q, t0
    h_opt = build_sym(q_opt, n)
    return q_opt, eval_c5(h_opt)


def make_random_q(n, rng):
    m = n // 2
    t = np.linspace(0.0, 1.0, m)
    a1 = rng.uniform(0.1, 0.3)
    a2 = rng.uniform(0.05, 0.2)
    a3 = rng.uniform(0.02, 0.1)
    k1 = rng.randint(1, 5)
    k2 = rng.randint(2, 7)
    k3 = rng.randint(3, 9)
    p1 = rng.uniform(0, 2 * np.pi)
    p2 = rng.uniform(0, 2 * np.pi)
    p3 = rng.uniform(0, 2 * np.pi)
    base = rng.uniform(0.4, 0.6)
    q = base + a1 * np.sin(2 * np.pi * k1 * t + p1) + \
        a2 * np.cos(2 * np.pi * k2 * t + p2) + \
        a3 * np.sin(2 * np.pi * k3 * t + p3)
    return np.clip(q, 0.0, 1.0)


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)
    reserve = max(30.0, 0.04 * budget_s)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Get initial
    try:
        h_init = np.array(initial_h_values, dtype=float).reshape(-1)
        if h_init.size >= 4 and np.all(np.isfinite(h_init)):
            n_init = len(h_init)
            h_init = repair_box_sum(h_init, n_init / 2.0)
            c5_init = eval_c5(h_init)
            best_h = h_init.copy()
            best_c5 = c5_init
            best_n = n_init
    except Exception:
        pass

    if best_h is None:
        n_def = 126
        m_def = n_def // 2
        t_arr = np.linspace(0.0, 1.0, m_def)
        q_def = np.clip(0.5 + 0.22 * np.sin(2 * np.pi * 2 * t_arr) +
                        0.12 * np.cos(2 * np.pi * 3 * t_arr + 0.5) +
                        0.06 * np.sin(2 * np.pi * 5 * t_arr + 1.0), 0.0, 1.0)
        best_h = build_sym(q_def, n_def)
        best_c5 = eval_c5(best_h)
        best_n = n_def

    def update_best(h_cand, c5_cand):
        nonlocal best_h, best_c5, best_n
        if c5_cand < best_c5 - 1e-12:
            best_h = h_cand.copy()
            best_c5 = c5_cand
            best_n = len(h_cand)

    def time_left():
        return budget_s - (time.time() - t_start) - reserve

    # Phase 1: SLP symmetric on parent
    n_par = len(best_h)
    if n_par % 2 == 0:
        q_par = to_sym_q(best_h, n_par)
        h_sym = build_sym(q_par, n_par)
        c5_sym = eval_c5(h_sym)
        if c5_sym < best_c5:
            best_h = h_sym.copy()
            best_c5 = c5_sym
            n_par = len(best_h)
            q_par = to_sym_q(best_h, n_par)

        p1_limit = 0.35 * budget_s
        q_par, c5_slp = slp_sym(q_par, n_par, R_init=0.06, time_limit=p1_limit, t0=t_start)
        if c5_slp < best_c5:
            best_h = build_sym(q_par, n_par).copy()
            best_c5 = c5_slp

        # Second SLP pass with different R
        if time_left() > 30:
            C_chk = corr_scaled(best_h)
            n_par_c = len(best_h)
            q_start = best_h[:n_par_c // 2].copy() if n_par_c % 2 == 0 else to_sym_q(best_h, n_par_c)
            if n_par_c % 2 == 0:
                q2, c5_2 = slp_sym(q_start, n_par_c, R_init=0.03, time_limit=0.2 * budget_s, t0=t_start)
                if c5_2 < best_c5:
                    best_h = build_sym(q2, n_par_c).copy()
                    best_c5 = c5_2

    # Phase 2: L-BFGS polish
    if time_left() > 20 and best_n % 2 == 0:
        n_c = len(best_h)
        q_c = to_sym_q(best_h, n_c)
        C_c = corr_scaled(best_h)
        active_c = select_active_slp(C_c, 0.02, n_pad=3)
        for mu in [500.0, 2000.0, 10000.0]:
            if time_left() < 15:
                break
            q_c, c5_p = polish_lbgfs(q_c, n_c, active_c, mu, maxiter=80)
            if c5_p < best_c5:
                best_h = build_sym(q_c, n_c).copy()
                best_c5 = c5_p
                active_c = select_active_slp(corr_scaled(best_h), 0.02, n_pad=3)

    # Phase 3: SLP full (break symmetry)
    if time_left() > 60:
        n_f = len(best_h)
        h_f = repair_box_sum(best_h, n_f / 2.0)
        p3_limit = 0.2 * budget_s
        h_f, c5_f = slp_full(h_f, n_f, R_init=0.04, time_limit=p3_limit, t0=t_start)
        if c5_f < best_c5:
            best_h = h_f.copy()
            best_c5 = c5_f

    # Phase 4: Multi-length exploration
    if time_left() > 40:
        n_base = len(best_h)
        candidates = set()
        for delta in [-12, -8, -6, -4, -2, 2, 4, 6, 8, 12, 16, 20]:
            nc = n_base + delta
            if nc >= 40 and nc % 2 == 0:
                candidates.add(nc)
        candidates.add(max(40, n_base // 2))
        n_sorted = sorted(candidates)

        for n_try in n_sorted:
            if time_left() < 25:
                break
            h_try = interp_h(best_h, n_try)
            c5_try = eval_c5(h_try)
            if c5_try < best_c5:
                best_h = h_try.copy()
                best_c5 = c5_try

            if n_try % 2 == 0 and time_left() > 20:
                q_try = to_sym_q(h_try, n_try)
                q_try, c5_s = slp_sym(q_try, n_try, R_init=0.05, time_limit=0.12 * budget_s, t0=t_start)
                if c5_s < best_c5:
                    best_h = build_sym(q_try, n_try).copy()
                    best_c5 = c5_s

    # Phase 5: Local coordinate search
    if time_left() > 20 and best_n % 2 == 0:
        n_ls = len(best_h)
        q_ls = to_sym_q(best_h, n_ls)
        p5_limit = 0.1 * budget_s
        q_ls, c5_ls = local_search_sym(q_ls, n_ls, time_limit=p5_limit, t0=t_start)
        if c5_ls < best_c5:
            best_h = build_sym(q_ls, n_ls).copy()
            best_c5 = c5_ls

    # Phase 6: Random restarts (symmetric SLP)
    if time_left() > 30:
        max_restarts = min(20, max(3, int(time_left() / 20)))
        for trial in range(max_restarts):
            if time_left() < 15:
                break
            n_r = len(best_h)
            if n_r % 2 != 0:
                n_r -= 1
            q_r = make_random_q(n_r, rng)
            active_r = select_active_slp(corr_scaled(build_sym(q_r, n_r)), 0.05, n_pad=2)
            _, c5_r = slp_sym(q_r, n_r, R_init=0.08, time_limit=0.06 * budget_s, t0=t_start)
            if c5_r < best_c5:
                best_h = build_sym(q_r, n_r).copy()
                best_c5 = c5_r

    # Phase 7: Final SLP polish on best
    if time_left() > 10 and best_n % 2 == 0:
        n_fin = len(best_h)
        q_fin = to_sym_q(best_h, n_fin)
        _, c5_fin = slp_sym(q_fin, n_fin, R_init=0.02, time_limit=max(5.0, time_left() * 0.4), t0=t_start)
        if c5_fin < best_c5:
            best_h = build_sym(q_fin, n_fin).copy()
            best_c5 = c5_fin

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = repair_box_sum(best_h, len(best_h) / 2.0)
    best_c5 = eval_c5(best_h)

    return best_h, best_c5, len(best_h)
