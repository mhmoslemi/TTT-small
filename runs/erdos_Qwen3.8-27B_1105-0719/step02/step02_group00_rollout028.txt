def repair_feasible(vec):
    # Clip and fix sum
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        # Need to add mass; add to positions with most room
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx])
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-14:

    else:

def build_gradient_rows(h, n):
    """Build the gradient matrix for all lags.
    Returns G of shape (2n-1, n) where G[m, :] is grad of P_{m-(n-1)} at h."""
    dx = 2.0 / n
    M = 2 * n - 1
    G = np.zeros((M, n))
    center = n - 1

def solve_lp_trust_region(h, active_indices, profile, trust_radius):
    """Solve the LP trust region subproblem.
    minimize t
    subject to: profile[m] + G[m,:]^T * d <= t for m in active
                sum(d) = 0
                -trust_radius <= d_i <= trust_radius
    
    Returns: d (step direction), t_opt
    """
    n = len(h)
    G = build_gradient_rows(h, n)
    G_a = G[active_indices, :]
    p_a = profile[active_indices]
    k = len(active_indices)
    
    # Variables: d (n) + t (1) = n+1 variables
    # Objective: minimize t -> c = [0,...,0, 1]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Inequality constraints: G_a @ d - t <= -p_a
    # i.e., G_a @ d + [-1]*t <= -p_a
    A_ub = np.zeros((k, n + 1))
    A_ub[:, :n] = G_a
    A_ub[:, n] = -1.0
    b_ub = -p_a
    
    # Sum constraint: sum(d) = 0 -> [1,...,1, 0] @ [d; t] = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    # Bounds: -trust_radius <= d_i <= trust_radius, t >= 0
    bounds = [(-trust_radius, trust_radius)] * n + [(0.0, None)]
    
    try:
        from scipy.optimize import linprog
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
        if res.success:
            d = res.x[:n]
            return d, res.x[-1]
    except Exception:
        pass
    return None, None

def gradient_direction(h, active_indices, profile, max_active=20):
    """Compute a weighted gradient direction from active constraints."""
    n = len(h)
    G = build_gradient_rows(h, n)
    
    # Weight by how close each active constraint is to the max
    c_max = np.max(profile)
    weights = np.maximum(profile[active_indices] - (c_max - 0.01), 0.0)
    if weights.sum() < 1e-15:
        weights = np.ones(len(active_indices))
    weights /= weights.sum()
    
    # If too many active, take top ones
    if len(active_indices) > max_active:
        order = np.argsort(profile[active_indices])
        active_indices = active_indices[order[-max_active:]]
        weights = np.ones(len(active_indices)) / len(active_indices)
    
    G_a = G[active_indices, :]
    grad = G_a.T @ weights
    grad -= grad.mean()  # Project onto sum=0 subspace
    
    norm = np.linalg.norm(grad)
    if norm < 1e-15:
        return None
    return grad / norm

            return h_try, s, alpha
        alpha *= 0.5

    return None, None, None

import numpy as np
from scipy.optimize import minimize, linprog
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx])
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-14:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-14:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] = np.clip(h[idx] + diff2, 0.0, 1.0)
    return h

def build_gradient_rows(h, n):
    dx = 2.0 / n
    M = 2 * n - 1
    center = n - 1
    G = np.zeros((M, n))
    for m in range(M):
        d = m - center
        if d == 0:
            G[m, :] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            for i in range(n - d):
                G[m, i] += -dx * h[i + d]
                G[m, i + d] += dx * (1.0 - h[i])
        else:
            e = -d
            for i in range(n - e):
                G[m, i] += dx * (1.0 - h[i + e])
                G[m, i + e] += -dx * h[i]
    return G

def solve_lp_trust(h, active, profile, radius):
    n = len(h)
    G = build_gradient_rows(h, n)
    G_a = G[active, :]
    p_a = profile[active]
    k = len(active)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((k, n + 1))
    A_ub[:, :n] = G_a
    A_ub[:, n] = -1.0
    b_ub = -p_a
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    bounds = [(-radius, radius)] * n + [(0.0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], res.x[-1]
    except Exception:
        pass
    return None, None

def gradient_direction(h, active, profile, max_active=25):
    n = len(h)
    G = build_gradient_rows(h, n)
    c_max = np.max(profile)
    if len(active) > max_active:
        order = np.argsort(profile[active])
        active = active[order[-max_active:]]
    w = np.maximum(profile[active] - (c_max - 0.005), 0.01)
    w /= w.sum()
    G_a = G[active, :]
    grad = G_a.T @ w
    grad -= grad.mean()
    norm = np.linalg.norm(grad)
    if norm < 1e-15:
        return None
    return grad / norm

def line_search_step(h, direction, current_score, init_alpha=0.05):

        s = exact_score(h_try)

        alpha *= 0.5

    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    h_new = np.clip(h_new, 0.0, 1.0)

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s - (time.perf_counter() - t_start)

    # Get parent
    parent = None
    try:
        parent = np.array(initial_h_values, dtype=np.float64)
    except NameError:
        pass
    except Exception:
        pass

    if parent is None or len(parent) < 4:
        parent = np.full(70, 0.5)

    parent = np.clip(parent, 0.0, 1.0)
    parent = repair_feasible(parent)

    best_h = parent.copy()
    best_c = exact_score(parent)

    # Symmetrization
    h_sym = repair_feasible((parent + parent[::-1]) / 2.0)
    c_sym = exact_score(h_sym)
    if c_sym < best_c:
        best_h = h_sym
        best_c = c_sym

    n0 = len(best_h)

    # Phase 1: Aggressive LP optimization at parent dimension
    if time_left() > 120:
        h_opt, c_opt = optimize_at_dim(best_h, time_left, max_iters=800)
        if c_opt < best_c - 1e-14:
            best_h = h_opt
            best_c = c_opt

    # Phase 2: SLSQP polish
    if time_left() > 80:
        h_slq = slsqp_polish(best_h, maxiter=250)
        if h_slq is not None:
            c_slq = exact_score(h_slq)
            if c_slq < best_c - 1e-14:
                best_h = h_slq
                best_c = c_slq

    # Phase 3: More LP optimization after SLSQP
    if time_left() > 80:
        h_opt, c_opt = optimize_at_dim(best_h, time_left, max_iters=600)
        if c_opt < best_c - 1e-14:
            best_h = h_opt
            best_c = c_opt

    # Phase 4: Dimension exploration
    if time_left() > 120:
        n_best = len(best_h)
        dims = []
        for d in range(max(30, n_best - 24), min(160, n_best + 32)):
            if d != n_best and d % 2 == 0:
                dims.append(d)
        dims.sort(key=lambda d: abs(d - n_best))

        for new_n in dims[:20]:
            if time_left() < 30:
                break
            h_interp = interpolate_to_n(best_h, new_n)
            c_interp = exact_score(h_interp)
            if c_interp < best_c + 0.008:
                h_opt, c_opt = optimize_at_dim(h_interp, time_left, max_iters=300)
                if c_opt < best_c - 1e-12:
                    best_h = h_opt
                    best_c = c_opt
                    n_best = new_n

    # Phase 5: Fine-tuning with small trust region
    if time_left() > 60:
        n_t = len(best_h)
        h_cur = best_h.copy()
        for it in range(1000):
            if time_left() < 10:
                break
            profile = compute_profile(h_cur)
            c_cur = np.max(profile)
            if c_cur < best_c - 1e-15:
                best_h = h_cur.copy()
                best_c = c_cur

            spread = c_cur - np.median(profile)
            thresh = c_cur - max(0.001, 0.03 * spread)
            active = np.where(profile >= thresh)[0]
            if len(active) == 0:
                active = np.array([np.argmax(profile)])

            d, _ = solve_lp_trust(h_cur, active, profile, 0.004)
            accepted = False
            if d is not None and np.linalg.norm(d) > 1e-13:
                for frac in [1.0, 0.5, 0.2]:
                    h_try = repair_feasible(h_cur + frac * d)
                    s_try = exact_score(h_try)
                    if s_try < c_cur - 1e-16:
                        h_cur = h_try
                        accepted = True
                        break

            if not accepted:
                grad = gradient_direction(h_cur, active, profile, max_active=30)
                if grad is not None:
                    h_ns, s_ns = line_search_step(h_cur, grad, c_cur, init_alpha=0.01)
                    if h_ns is not None:
                        h_cur = h_ns
                        accepted = True

        c_final = exact_score(h_cur)
        if c_final < best_c:
            best_h = h_cur
            best_c = c_final

    # Phase 6: SLSQP final polish
    if time_left() > 40:
        h_slq = slsqp_polish(best_h, maxiter=400)
        if h_slq is not None:
            c_slq = exact_score(h_slq)
            if c_slq < best_c - 1e-14:
                best_h = h_slq
                best_c = c_slq

    # Phase 7: One more LP round
    if time_left() > 20:
        h_opt, c_opt = optimize_at_dim(best_h, time_left, max_iters=200)
        if c_opt < best_c - 1e-15:
            best_h = h_opt
            best_c = c_opt

    # Final validation
    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)

    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            final_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 2:
            final_c = float(eval_result[0])
    except Exception:
        pass

    return best_h, final_c, len(best_h)
