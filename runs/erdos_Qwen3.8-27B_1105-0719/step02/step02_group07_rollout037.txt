    #   If d >= 0 and p+d < n: term h[p]*(1-h[p+d]) = x*(1-h[p+d])

import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time


def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)


def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full').max() * dx


def make_sym_h(q, n):
    m = n // 2
    q = np.clip(q, 0.0, 1.0)
    h = np.empty(n)
    h[:m] = q
    h[m:] = 1.0 - q[::-1]
    return h


def make_q_from_h(h, n):
    m = n // 2
    q = 0.5 * (h[:m] + 1.0 - h[n - 1: n - 1 - m: -1])
    return np.clip(q, 0.0, 1.0)


def corr_vals(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx


def sym_coord_descent(q, n, max_passes=100, t0=None, time_limit=None):
    m = n // 2
    dx = 2.0 / n
    q = np.clip(q.copy(), 0.0, 1.0)
    h = make_sym_h(q, n)
    corrs = np.correlate(h, 1.0 - h, mode='full') * dx
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    d_range = np.arange(-(n - 1), n)
    n_lags = 2 * n - 1

    for _ in range(max_passes):
        if t0 is not None and time_limit is not None and (time.time() - t0) > time_limit:
            break
        improved = False
        for p in range(m):
            j1 = p
            j2 = n - 1 - p

            s1 = np.zeros(n_lags)
            mask_a = (d_range >= -j1) & (d_range < n - j1)
            if mask_a.any():
                s1[mask_a] = 1.0 - h[j1 + d_range[mask_a]]
            mask_b = (d_range > j1 - n) & (d_range <= j1)
            if mask_b.any():
                s1[mask_b] += -h[j1 - d_range[mask_b]]

            s2 = np.zeros(n_lags)
            mask_a2 = (d_range >= -j2) & (d_range < n - j2)
            if mask_a2.any():
                s2[mask_a2] = 1.0 - h[j2 + d_range[mask_a2]]
            mask_b2 = (d_range > j2 - n) & (d_range <= j2)
            if mask_b2.any():
                s2[mask_b2] += -h[j2 - d_range[mask_b2]]

            slopes = (s1 - s2) * dx
            base = corrs - q[p] * slopes

            a_g, b_g = 0.0, 1.0
            c_g = b_g - gr * (b_g - a_g)
            d_g = a_g + gr * (b_g - a_g)
            fc = np.max(base + c_g * slopes)
            fd = np.max(base + d_g * slopes)
            for _ in range(45):
                if fc < fd:
                    b_g, d_g, fd = d_g, c_g, fc
                    c_g = b_g - gr * (b_g - a_g)
                    fc = np.max(base + c_g * slopes)
                else:
                    a_g, c_g, fc = c_g, d_g, fd
                    d_g = a_g + gr * (b_g - a_g)
                    fd = np.max(base + d_g * slopes)

            new_qp = (a_g + b_g) / 2.0
            delta = new_qp - q[p]
            if abs(delta) > 1e-13:
                q[p] = new_qp
                h[j1] = new_qp
                h[j2] = 1.0 - new_qp
                corrs += delta * slopes
                improved = True
        if not improved:
            break

    return q, exact_score(make_sym_h(q, n))


def sym_block_update(q, n, pair_indices, t0=None, time_limit=None):
    m = n // 2
    dx = 2.0 / n
    q = np.clip(q.copy(), 0.0, 1.0)
    h = make_sym_h(q, n)
    corrs = np.correlate(h, 1.0 - h, mode='full') * dx
    d_range = np.arange(-(n - 1), n)
    n_lags = 2 * n - 1

    def compute_slopes_for(j1, j2):
        s1 = np.zeros(n_lags)
        ma = (d_range >= -j1) & (d_range < n - j1)
        if ma.any():
            s1[ma] = 1.0 - h[j1 + d_range[ma]]
        mb = (d_range > j1 - n) & (d_range <= j1)
        if mb.any():
            s1[mb] += -h[j1 - d_range[mb]]
        s2 = np.zeros(n_lags)
        ma2 = (d_range >= -j2) & (d_range < n - j2)
        if ma2.any():
            s2[ma2] = 1.0 - h[j2 + d_range[ma2]]
        mb2 = (d_range > j2 - n) & (d_range <= j2)
        if mb2.any():
            s2[mb2] += -h[j2 - d_range[mb2]]
        return (s1 - s2) * dx

    for p, qq in pair_indices:
        if t0 is not None and time_limit is not None and (time.time() - t0) > time_limit:
            break
        if p == qq or p >= m or qq >= m:
            continue
        j1, j2 = p, n - 1 - p
        j3, j4 = qq, n - 1 - qq
        sl1 = compute_slopes_for(j1, j2)
        sl2 = compute_slopes_for(j3, j4)

        base = corrs - q[p] * sl1 - q[qq] * sl2

        grid = np.linspace(0.0, 1.0, 25)
        best_obj = np.max(corrs)
        best_x, best_y = q[p], q[qq]
        for x in grid:
            for y in grid:
                val = np.max(base + x * sl1 + y * sl2)
                if val < best_obj:
                    best_obj = val
                    best_x, best_y = x, y

        delta1 = best_x - q[p]
        delta2 = best_y - q[qq]
        if abs(delta1) > 1e-13 or abs(delta2) > 1e-13:
            q[p] = best_x
            q[qq] = best_y
            h[j1] = best_x
            h[j2] = 1.0 - best_x
            h[j3] = best_y
            h[j4] = 1.0 - best_y
            corrs += delta1 * sl1 + delta2 * sl2

    return q, exact_score(make_sym_h(q, n))


def smooth_sym_obj(x, n, beta):
    h = make_sym_h(x, n)
    corrs = corr_vals(h)
    return logsumexp(beta * corrs) / beta


def smooth_sym_grad(x, n, beta):
    m = n // 2
    dx = 2.0 / n
    h = make_sym_h(x, n)
    corrs = corr_vals(h)
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    ws = w.sum()
    if ws < 1e-30:
        w = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        w /= ws
    d_range = np.arange(-(n - 1), n)
    n_lags = 2 * n - 1
    grad_q = np.zeros(m)
    for p in range(m):
        j1 = p
        j2 = n - 1 - p
        s1 = np.zeros(n_lags)
        ma = (d_range >= -j1) & (d_range < n - j1)
        if ma.any():
            s1[ma] = 1.0 - h[j1 + d_range[ma]]
        mb = (d_range > j1 - n) & (d_range <= j1)
        if mb.any():
            s1[mb] += -h[j1 - d_range[mb]]
        s2 = np.zeros(n_lags)
        ma2 = (d_range >= -j2) & (d_range < n - j2)
        if ma2.any():
            s2[ma2] = 1.0 - h[j2 + d_range[ma2]]
        mb2 = (d_range > j2 - n) & (d_range <= j2)
        if mb2.any():
            s2[mb2] += -h[j2 - d_range[mb2]]
        slopes = (s1 - s2) * dx
        grad_q[p] = dot_w_slopes(w, slopes)
    return grad_q


def dot_w_slopes(w, slopes):
    return np.dot(w, slopes)


def smooth_sym_optimize(q0, n, beta, maxiter=300):
    m = n // 2
    try:
        res = minimize(smooth_sym_obj, q0, jac=lambda x: smooth_sym_grad(x, n, beta),
                       method='SLSQP', bounds=[(0.0, 1.0)] * m,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        h_new = make_sym_h(res.x, n)
        c5 = exact_score(h_new)
        return res.x, c5
    except Exception:
        return None, None


def active_lags_top(corrs, n_top):
    idx = np.argsort(corrs)[::-1][:n_top]
    return sorted(idx.tolist())


def solve_epi_sym(q0, n, active_lags, t_init, maxiter=400):
    m = n // 2
    dx = 2.0 / n
    n_lags = 2 * n - 1
    d_range = np.arange(-(n - 1), n)

    def epi_fun(x):
        return x[0]

    def epi_jac(x):
        g = np.zeros(m + 1)
        g[0] = 1.0
        return g

    def epi_con_fun(x):
        t = x[0]
        q = x[1:]
        h = make_sym_h(q, n)
        corrs = np.correlate(h, 1.0 - h, mode='full') * dx
        return t - corrs[active_lags]

    def epi_con_jac(x):
        q = x[1:]
        h = make_sym_h(q, n)
        L = len(active_lags)
        J = np.zeros((L, m + 1))
        J[:, 0] = 1.0
        for i, lag in enumerate(active_lags):
            d = lag - (n - 1)
            row = np.zeros(n)
            mask_a = (d_range == d)
            for p in range(m):
                j1 = p
                j2 = n - 1 - p
                s1 = 0.0
                if 0 <= j1 + d < n:
                    s1 += 1.0 - h[j1 + d]
                if 0 <= j1 - d < n:
                    s1 += -h[j1 - d]
                s2 = 0.0
                if 0 <= j2 + d < n:
                    s2 += 1.0 - h[j2 + d]
                if 0 <= j2 - d < n:
                    s2 += -h[j2 - d]
                J[i, p + 1] = -(s1 - s2) * dx
        return J

    x0 = np.concatenate([[t_init], q0])
    cons = [{'type': 'ineq', 'fun': epi_con_fun, 'jac': epi_con_jac}]
    bounds = [(0.0, 2.0)] + [(0.0, 1.0)] * m
    try:
        res = minimize(epi_fun, x0, jac=epi_jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        h_new = make_sym_h(res.x[1:], n)
        c5 = exact_score(h_new)
        if abs(h_new.sum() - n / 2.0) > 1e-7:
            return None, None
        return res.x[1:], c5
    except Exception:
        return None, None


def full_smooth_optimize(h0, n, beta, maxiter=300):
    target_sum = n / 2.0
    dx = 2.0 / n
    d_range = np.arange(-(n - 1), n)
    n_lags = 2 * n - 1

    def obj(x):
        h = np.clip(x, 0.0, 1.0)
        corrs = np.correlate(h, 1.0 - h, mode='full') * dx
        return logsumexp(beta * corrs) / beta

    def obj_grad(x):
        h = np.clip(x, 0.0, 1.0)
        corrs = np.correlate(h, 1.0 - h, mode='full') * dx
        shifted = corrs - corrs.max()
        w = np.exp(beta * shifted)
        ws = w.sum()
        if ws < 1e-30:
            w = np.ones(n_lags) / n_lags
        else:
            w /= ws
        g = np.zeros(n)
        for p in range(n):
            s = np.zeros(n_lags)
            ma = (d_range >= -p) & (d_range < n - p)
            if ma.any():
                s[ma] = 1.0 - h[p + d_range[ma]]
            mb = (d_range > p - n) & (d_range <= p)
            if mb.any():
                s[mb] += -h[p - d_range[mb]]
            g[p] = np.dot(w, s) * dx
        return g

    try:
        res = minimize(obj, h0, jac=obj_grad, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        h_out = project_feasible(res.x, target_sum)
        c5 = exact_score(h_out)
        return h_out, c5
    except Exception:
        return h0, exact_score(h0)
