def compute_grad_matrix(h, n):
    e_vals = np.arange(-(n-1), n)  # shape (2n-1,)
    # For each i, e: check if i+e and i-e are in [0, n-1]
    # i ranges [0, n-1], e ranges [-(n-1), n-1]
    i_vals = np.arange(n)[:, None]  # shape (n, 1)
    e_expanded = e_vals[None, :]  # shape (1, 2n-1)
    
    i_plus_e = i_vals + e_expanded  # shape (n, 2n-1)
    i_minus_e = i_vals - e_expanded
    
    valid_plus = (i_plus_e >= 0) & (i_plus_e < n)
    valid_minus = (i_minus_e >= 0) & (i_minus_e < n)
    
    # We need h[i+e] where valid, and h[i-e] where valid
    # Use np.where with safe indices
    i_plus_e_safe = np.where(valid_plus, i_plus_e, 0)
    i_minus_e_safe = np.where(valid_minus, i_minus_e, 0)
    
    h_plus = h[i_plus_e_safe]  # shape (n, 2n-1)
    h_minus = h[i_minus_e_safe]
    
    grad_mat = valid_plus * (1.0 - h_plus) - valid_minus * h_minus
    return grad_mat

def slsqp_optimize(h0, dx, target, beta, maxiter, t0, max_time):
    n = len(h0)

    def obj(x):
        return smooth_max_obj(x, dx, beta)

    def obj_grad(x):
        return smooth_max_grad(x, dx, beta)

    def cons_fun(x):
        return x.sum() - target

    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        h_out = project_feasible(res.x, target)
        if time.time() - t0 > max_time:
            return h0, c5_of_h(h0, dx)
    except Exception:
        h_out = h0
    c5_val = c5_of_h(h_out, dx)
    return h_out, c5_val

    # Phase 3: SLSQP with smooth max
    for beta in [100, 300, 800, 2000]:
        if remaining() < 40:
            break
        tm = min(remaining() * 0.25, 180)
        h_sl, c5_sl = slsqp_optimize(best_h, dx, target, beta,
                                     maxiter=500, t0=t0, max_time=tm)
        if c5_sl < best_c5:
            best_c5 = c5_sl
            best_h = h_sl

    # Phase 4: More coordinate descent after SLSQP
    if remaining() > 80:
        h_cd2, c5_cd2, _ = coord_descent_sweeps(best_h, dx, target, t0,
                                                max_time=remaining() * 0.4,
                                                max_sweeps=500)
        if c5_cd2 < best_c5:
            best_c5 = c5_cd2
            best_h = h_cd2

    # Phase 5: Continuous SA
    if remaining() > 120:
        h_sa, c5_sa = sa_continuous(best_h, dx, target, t0,
                                    max_time=remaining() * 0.35,
                                    seed=seed + 1, max_iters=80000)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa

    # Phase 6: Final coordinate descent
    if remaining() > 60:
        h_cd3, c5_cd3, _ = coord_descent_sweeps(best_h, dx, target, t0,
                                                max_time=remaining() * 0.5,
                                                max_sweeps=300)
        if c5_cd3 < best_c5:
            best_c5 = c5_cd3
            best_h = h_cd3

    # Phase 7: Try nearby n values
    for n_try in [n - 2, n + 2, n + 4, n - 4, n + 8]:
        if remaining() < 80:
            break
        if n_try < 40 or n_try > 256:
            continue
        dx_t = 2.0 / n_try
        target_t = n_try / 2.0
        x_old = np.linspace(0.0, 2.0, len(best_h))
        x_new = np.linspace(0.0, 2.0, n_try)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, target_t)
        c5_interp = c5_of_h(h_interp, dx_t)

        h_t, c5_t, _ = coord_descent_sweeps(h_interp, dx_t, target_t, t0,
                                            max_time=min(remaining() * 0.3, 200),
                                            max_sweeps=200)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t

        # SLSQP polish at new n
        if remaining() > 60 and c5_t < best_c5:
            h_sl2, c5_sl2 = slsqp_optimize(h_t, dx_t, target_t, 500,
                                           maxiter=300, t0=t0,
                                           max_time=min(remaining() * 0.2, 100))
            if c5_sl2 < best_c5:
                best_c5 = c5_sl2
                best_h = h_sl2

    # Phase 8: Final polish
    n_b = len(best_h)
    dx_b = 2.0 / n_b
    target_b = n_b / 2.0
    best_h = project_feasible(best_h, target_b)
    best_c5 = c5_of_h(best_h, dx_b)

    if remaining() > 30:
        h_final, c5_final, _ = coord_descent_sweeps(best_h, dx_b, target_b, t0,
                                                    max_time=remaining(),
                                                    max_sweeps=100)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final
            n_b = len(best_h)
            dx_b = 2.0 / n_b
            best_h = project_feasible(best_h, n_b / 2.0)
            best_c5 = c5_of_h(best_h, dx_b)

    return best_h, best_c5, len(best_h)
