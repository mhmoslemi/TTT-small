def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
        else:
            h += diff / n
    return np.clip(h, 0.0, 1.0)

def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def c5_of_h(h, dx):
    return compute_corrs(h).max() * dx

def smooth_max_obj(h, dx, beta):
    corrs = compute_corrs(h)
    m = corrs.max()
    return (m + np.log(np.sum(np.exp(beta * (corrs - m)))) / beta) * dx

def grad_corr_k(h, k, n):
    """Gradient of correlation at lag k with respect to h."""
    offset = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -offset)
    i_hi = min(n - 1, n - 1 - offset)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx + offset] += (1.0 - h[idx])
    grad[idx] += (-h[idx + offset])
    return grad

def smooth_max_grad(h, dx, beta):
    n = len(h)
    corrs = compute_corrs(h)
    m = corrs.max()
    shifted = corrs - m
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    w /= w_sum
    grad = np.zeros(n)
    for k in range(len(corrs)):
        if w[k] < 1e-14:
            continue
        grad += w[k] * grad_corr_k(h, k, n)
    return grad * dx

    def obj_grad(x):
        return smooth_max_grad(x, dx, beta)

        return x.sum() - target_sum

        return np.ones(n)

def compute_delta_binary(h, i, j, n):
    """Compute change in all correlations when h[i]:1->0 and h[j]:0->1."""
    ms = np.arange(-(n - 1), n)
    valid_i1 = (0 <= i - ms) & (i - ms <= n - 1)
    valid_i2 = (0 <= i + ms) & (i + ms <= n - 1)
    valid_j1 = (0 <= j - ms) & (j - ms <= n - 1)
    valid_j2 = (0 <= j + ms) & (j + ms <= n - 1)
    
    delta = np.zeros(2 * n - 1)
    idx_i1 = i - ms[valid_i1]
    delta[valid_i1] += -(1.0 - h[idx_i1])
    idx_i2 = i + ms[valid_i2]
    delta[valid_i2] += h[idx_i2]
    idx_j1 = j - ms[valid_j1]
    delta[valid_j1] += (1.0 - h[idx_j1])
    idx_j2 = j + ms[valid_j2]
    delta[valid_j2] += -h[idx_j2]
    return delta
