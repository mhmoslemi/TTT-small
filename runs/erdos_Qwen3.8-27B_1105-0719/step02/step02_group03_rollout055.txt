                h[ii], h[jj] = 0.0, 1.0
            corrs = compute_corrs(h)
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            no_improve = 0
            T = max(T, 2.0)
    
    return best_h, best_mc * dx


def smooth_max_obj(h, dx, beta):
    corrs = compute_corrs(h)
    m = corrs.max()
    if beta * (corrs.max() - corrs.min()) < 0.01:
        return m * dx
    shifted = corrs - m
    w = np.exp(beta * shifted)
    return (m + np.log(w.sum()) / beta) * dx


def smooth_max_grad(h, dx, beta):
    n = len(h)
    corrs = compute_corrs(h)
    m = corrs.max()
    shifted = corrs - m
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    w = w / w_sum
    grad = np.zeros(n)
    # For each k in the full correlation (length 2n-1), the contribution to grad
    # corr[m] = sum_n h[n] * (1-h[n+lag]) where lag = m-(n-1)
    # d(cor
    # I'll use finite differences for simplicity (slower but correct)
    # Actually, let me compute analytically.
    # corr[m] = sum_i h[i] * (1-h[i+lag(m)]) where lag(m) = m-(n-1)
    # d(corr[m])/d(h[p]) = (1-h[p+lag(m)]) * [p in h[i] slot] - h[p-lag(m)] * [p-lag(m) in h[i] slot]
    # Wait, let me be more careful.
    # corr[m] = sum_i h[i] * (1-h[i+lag])
    # = sum_i h[i] - sum_i h[i]*h[i+lag]
    # d/dh[p]: (1-h[p+lag]) - h[p+lag] - h[p-lag] (if valid)
    # Hmm, this is getting complicated. Let me use the same approach as compute_delta_binary
    # but in a continuous setting.
    
    # Actually, for the continuous case, let me just use numerical gradient:
    # Too slow for large n. Let me use the analytical form.
    
    # corr[m] = sum_i h[i] * (1-h[i+lag(m)])
    # where lag(m) = m - (n-1)
    # d(corr[m])/d(h[p]) = (1-h[p+lag]) if p+lag is valid, minus h[p-lag] if p-lag is valid
    # More precisely:
    # The term h[p] appears in:
    # 1. The first factor: i=p, contributes (1-h[p+lag]) if 0<=p+lag<n
    # 2. The second factor: i+lag=p, i=e-lag, contributes -h[e-lag] * 0 ... 
    #    Wait: h[i]*(1-h[i+lag]). If i+lag=p, i=p-lag, the term is h[p-lag]*(1-h[p]).
    #    d/dh[p] of this term = h[p-lag)*(-1) = -h[p-lag], valid if 0<=p-lag<n.
    # So d(corr[m])/d(h[p]) = (1-h[p+lag(m)]) * [0<=p+lag(m)<n] - h[p-lag(m)] * [0<=p-lag(m)<n]
    
    for m in range(2 * n - 1):
        if w[m] < 1e-14:
            continue
        lag = m - (n - 1)
        # For each p:
        # contribution = w[m] * dx * [ (1-h[p+lag]) if valid - h[p-lag] if valid ]
        # This is O(n) per m, total O(n * 2n) = O(n^2) which is fine for n=128
        
        # Vectorized:
        # Part 1: (1-h[p+lag]) for p where 0<=p+lag<n
        p1_lo = max(0, -lag)
        p1_hi = min(n - 1, n - 1 - lag)
        if p1_lo <= p1_hi:
            grad[p1_lo:p1_hi+1] += w[m] * (1.0 - h[p1_lo + lag : p1_hi + 1 + lag])
        
        # Part 2: -h[p-lag] for p where 0<=p-lag<n
        p2_lo = max(0, lag)
        p2_hi = min(n - 1, n - 1 + lag)
        if p2_lo <= p2_hi:
            grad[p2_lo:p2_hi+1] += -w[m] * h[p2_lo - lag : p2_hi + 1 - lag]
    
    return grad * dx


def continuous_slsqp(h0, dx, target_sum, t0, time_limit, beta=100.0, maxiter=500):
    n = len(h0)
    
    def obj_func(x):
        return smooth_max_obj(x, dx, beta)
    
    def obj_grad(x):
        return smooth_max_grad(x, dx, beta)
    
    def constraint_fun(x):
        return x.sum() - target_sum
    
    def constraint_jac(x):
        return np.ones(n)
    
    constraints = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}]
    bounds = [(0.0, 1.0)] * n
    
    try:
        res = minimize(obj_func, h0, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        h_out = project_feasible(res.x, target_sum)
    except Exception:
        h_out = project_feasible(h0, target_sum)
    
    return h_out, c5_of_h(h_out, dx)


def interpolate_to_n(h_src, n_dst):
    n_src = len(h_src)
    x_src = np.linspace(0.0, 2.0, n_src, endpoint=False) + 1.0 / n_src
    x_dst = np.linspace(0.0, 2.0, n_dst, endpoint=False) + 1.0 / n_dst
    h_dst = np.interp(x_dst, x_src, h_src)
    return h_dst


def to_binary_fix(h, n):
    """Convert to binary with exactly n/2 ones."""
    target = n // 2
    h_bin = (h > 0.5).astype(float)
    cur = h_bin.sum()
    if cur > target:
        # Remove (cur - target) ones, prioritizing those with lowest h value
        excess = int(cur - target)
        one_idx = np.where(h_bin == 1.0)[0]
        order = np.argsort(h[one_idx])
        to_remove = one_idx[order[:excess]]
        h_bin[to_remove] = 0.0
    elif cur < target:
        deficit = int(target - cur)
        zero_idx = np.where(h_bin == 0.0)[0]
        order = np.argsort(-h[zero_idx])
        to_add = zero_idx[order[:deficit]]
        h_bin[to_add] = 1.0
    return h_bin
