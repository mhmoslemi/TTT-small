```python
def project_hypersimplex(h, target):
    """Project h onto {x: 0<=x<=1, sum(x)=target}"""
    n = len(h)
    # Binary search for tau
    lo, hi = h.min() - 1, h.max() + 1
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h - (lo + hi) / 2, 0, 1)
    # Fix rounding
    diff = target - h_proj.sum()
    if abs(diff) > 1e-12:
        # Adjust the element closest to 0.5 (most "interior")
        interior = np.where((h_proj > 1e-8) & (h_proj < 1 - 1e-8))[0]
        if len(interior) > 0:
            # Distribute the difference
            h_proj[interior] += diff / len(interior)
            h_proj = np.clip(h_proj, 0, 1)
    return h_proj
```

import numpy as np
from scipy.optimize import minimize
import time


def project_hypersimplex(h, target):
    n = len(h)
    lo, hi = h.min() - 2.0, h.max() + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h_p = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - h_p.sum()
    if abs(diff) > 1e-12:
        mask = (h_p > 1e-10) & (h_p < 1.0 - 1e-10)
        if mask.sum() > 0:
            h_p[mask] += diff / mask.sum()
        else:
            h_p += diff / n
    return np.clip(h_p, 0.0, 1.0)


def c5_of_h(h, dx):
    return np.correlate(h, 1.0 - h, mode='full').max() * dx


def smooth_max_obj(h, beta):
    corrs = np.correlate(h, 1.0 - h, mode='full')
    m = corrs.max()
    shifted = beta * (corrs - m)
    shifted -= shifted.max()
    return m + np.log(np.exp(shifted).sum()) / beta


def smooth_max_grad(h, beta):
    n = len(h)
    corrs = np.correlate(h, 1.0 - h, mode='full')
    m = corrs.max()
    shifted = beta * (corrs - m)
    shifted -= shifted.max()
    w = np.exp(shifted)
    w_sum = w.sum()
    w /= w_sum
    grad = np.zeros(n)
    for k in range(len(corrs)):
        if w[k] < 1e-14:
            continue
        e = k - (n - 1)
        wk = w[k]
        if e >= 0:
            if e < n:
                grad[:n - e] += wk * (1.0 - h[e:])
                grad[e:] += -wk * h[:n - e]
        else:
            e2 = -e
            if e2 < n:
                grad[e2:] += wk * (1.0 - h[:n - e2])
                grad[:n - e2] += -wk * h[e2:]
    return grad

    n = len(h0)

    def obj(x):
        return smooth_max_obj(x, beta)

    def obj_grad(x):
        return smooth_max_grad(x, beta)

    def cons_fun(x):
        return x.sum() - target_sum

    def cons_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    bounds = [(0.0, 1.0)] * n

    try:
        res = minimize(obj, h0, jac=obj_grad, method='L-BFGS-B',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14})
        h_out = project_hypersimplex(res.x, target_sum)
    except Exception:
        h_out = project_hypersimplex(h0, target_sum)
    return h_out


def compute_delta_binary(h, i, j, n):
    ms = np.arange(-(n - 1), n)
    valid_i1 = (0 <= i - ms) & (i - ms <= n - 1)
    valid_i2 = (0 <= i + ms) & (i + ms <= n - 1)
    valid_j1 = (0 <= j - ms) & (j - ms <= n - 1)
    valid_j2 = (0 <= j + ms) & (j + ms <= n - 1)
    delta = np.zeros(2 * n - 1)
    idx_i1 = i - ms[valid_i1]
    delta[valid_i1] += -(1.0 - h[idx_i1])
    idx_i2 = i + ms[valid_i2]
    delta[valid_i2] += h[idx_i2]
    idx_j1 = j - ms[valid_j1]
    delta[valid_j1] += (1.0 - h[idx_j1])
    idx_j2 = j + ms[valid_j2]
    delta[valid_j2] += -h[idx_j2]
    return delta


def compute_delta_binary2(h, i1, i2, j1, j2, n):
    ms = np.arange(-(n - 1), n)
    delta = np.zeros(2 * n - 1)
    for i in (i1, i2):
        v1 = (0 <= i - ms) & (i - ms <= n - 1)
        v2 = (0 <= i + ms) & (i + ms <= n - 1)
        delta[v1] += -(1.0 - h[i - ms[v1]])
        delta[v2] += h[i + ms[v2]]
    for j in (j1, j2):
        v1 = (0 <= j - ms) & (j - ms <= n - 1)
        v2 = (0 <= j + ms) & (j + ms <= n - 1)
        delta[v1] += (1.0 - h[j - ms[v1]])
        delta[v2] += -h[j + ms[v2]]
    return delta


def binary_sa(h0, dx, rng, max_iters, patience, use_2swap, deadline):
    n = len(h0)
    target = n // 2
    h = np.zeros(n)
    idx_sorted = np.argsort(-h0)
    h[idx_sorted[:target]] = 1.0
    corrs = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = corrs.max() * dx
    best_h = h.copy()
    ones = list(np.where(h == 1.0)[0])
    zeros = list(np.where(h == 0.0)[0])
    no_improve = 0
    temp = 0.08
    t_start = time.time()

    for it in range(max_iters):
        if time.time() > deadline:
            break
        if it % 3000 == 0:
            temp *= 0.999

        if use_2swap and len(ones) >= 2 and len(zeros) >= 2 and (it % 3 == 0):
            i1 = ones[rng.randint(len(ones))]
            i2 = ones[rng.randint(len(ones))]
            if i1 == i2:
                continue
            j1 = zeros[rng.randint(len(zeros))]
            j2 = zeros[rng.randint(len(zeros))]
            if j1 == j2:
                continue
            delta = compute_delta_binary2(h, i1, i2, j1, j2, n)
            new_corrs = corrs + delta
            new_max = new_corrs.max()
            new_c5 = new_max * dx
            accepted = False
            if new_c5 <= best_c5 + 1e-15:
                h[i1], h[i2] = 0.0, 0.0
                h[j1], h[j2] = 1.0, 1.0
                corrs = new_corrs
                best_c5 = new_c5
                best_h = h.copy()
                no_improve = 0
                accepted = True
            elif temp > 1e-10 and rng.random() < np.exp(-(new_c5 - best_c5) / temp):
                h[i1], h[i2] = 0.0, 0.0
                h[j1], h[j2] = 1.0, 1.0
                corrs = new_corrs
                no_improve = 0
                accepted = True
            else:
                no_improve += 1
            if accepted:
                ones = list(np.where(h == 1.0)[0])
                zeros = list(np.where(h == 0.0)[0])
        else:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            delta = compute_delta_binary(h, i, j, n)
            new_corrs = corrs + delta
            new_max = new_corrs.max()
            new_c5 = new_max * dx
            accepted = False
            if new_c5 <= best_c5 + 1e-15:
                h[i] = 0.0
                h[j] = 1.0
                corrs = new_corrs
                best_c5 = new_c5
                best_h = h.copy()
                no_improve = 0
                accepted = True
            elif temp > 1e-10 and rng.random() < np.exp(-(new_c5 - best_c5) / temp):
                h[i] = 0.0
                h[j] = 1.0
                corrs = new_corrs
                no_improve = 0
                accepted = True
            else:
                no_improve += 1
            if accepted:
                ones.remove(i)
                zeros.remove(j)
                ones.append(j)
                zeros.append(i)
                ones.sort()
                zeros.sort()

        if no_improve > patience:
            break

    return best_h, best_c5


def binary_greedy_swap(h0, dx, max_sweeps, deadline):
    n = len(h0)
    target = n // 2
    h = np.zeros(n)
    idx_sorted = np.argsort(-h0)
    h[idx_sorted[:target]] = 1.0
    corrs = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = corrs.max() * dx
    best_h = h.copy()
    ones = list(np.where(h == 1.0)[0])
    zeros = list(np.where(h == 0.0)[0])

    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        found = False
        best_delta_max = corrs.max()
        best_i, best_j = -1, -1
        for i in ones:
            for j in zeros:
                delta = compute_delta_binary(h, i, j, n)
                nm = (corrs + delta).max()
                if nm < best_delta_max - 1e-14:
                    best_delta_max = nm
                    best_i, best_j = i, j
                    found = True
        if not found:
            break
        h[best_i] = 0.0
        h[best_j] = 1.0
        corrs = np.correlate(h, 1.0 - h, mode='full')
        c5 = corrs.max() * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        ones.remove(best_i)
        zeros.remove(best_j)
        ones.append(best_j)
        zeros.append(best_i)
        ones.sort()
        zeros.sort()

    return best_h, best_c5


def make_golden_start(n, target, offset=0):
    h = np.zeros(n)
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    for k in range(int(target)):
        pos = int((k * phi + offset) * n) % n
        h[pos] = 1.0
    return project_hypersimplex(h, float(target))


def make_quadratic_start(n, target):
    h = np.zeros(n)
    p = 2 * target
    if p < 4:
        idx = np.random.RandomState(0).permutation(n)
        h[idx[:target]] = 1.0
        return project_hypersimplex(h, float(target))
    residues = set()
    for a in range(p):
        r = (a * a) % p
        residues.add(r)
    residues.discard(0)
    positions = []
    for r in sorted(residues):
        positions.append(r)
    positions.sort()
    for pos in positions[:int(target)]:
        h[pos % n] = 1.0
    if h.sum() < target:
        remaining = target - int(h.sum())
        available = [i for i in range(n) if h[i] == 0]
        for idx in available[:remaining]:
            h[idx] = 1.0
    return project_hypersimplex(h, float(target))


def interpolate_h(h_src, n_new):
    n_src = len(h_src)
    x_src = np.linspace(0.0, 2.0, n_src)
    x_new = np.linspace(0.0, 2.0, n_new)
    h_new = np.interp(x_new, x_src, h_src)
    return project_hypersimplex(h_new, float(n_new // 2))


def continuous_refine(h0, dx, target_sum, deadline):
    n = len(h0)
    best_h = project_hypersimplex(h0.copy(), target_sum)
    best_c5 = c5_of_h(best_h, dx)

    for beta in [100, 200, 500, 1000, 2000, 5000]:
        if time.time() > deadline - 10:
            break
        t_sub = deadline - 5
        if time.time() > t_sub - 30:
            break
        maxiter = 200
        h_opt = lbfgs_optimize(best_h, dx, target_sum, beta, maxiter, t_sub)
        c5_new = c5_of_h(h_opt, dx)
        if c5_new < best_c5 - 1e-10:
            best_h = h_opt
            best_c5 = c5_new

    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s * 0.95

    rng = np.random.RandomState(seed)

    try:
        parent_h = np.array(initial_h_values, dtype=float)
    except NameError:
        parent_h = None

    best_h = None
    best_c5 = 1.0
    best_n = 0

    def try_candidate(h, n_pts):
        nonlocal best_h, best_c5, best_n
        dx = 2.0 / n_pts
        h = project_hypersimplex(h, float(n_pts // 2))
        c5 = c5_of_h(h, dx)
        if c5 < best_c5 - 1e-12:
            best_h = h.copy()
            best_c5 = c5
            best_n = n_pts
        return c5

    n_sizes = [126, 124, 128, 130, 122, 132, 120, 134]
    if parent_h is not None:
        n_parent = len(parent_h)
        if n_parent not in n_sizes:
            n_sizes.insert(0, n_parent)
        n_sizes = sorted(set(n_sizes))
        for ns in list(n_sizes):
            if ns < 48 or ns > 160:
                n_sizes.remove(ns)

    results = {}

    for n_pts in n_sizes:
        if time.time() > deadline - 60:
            break
        dx = 2.0 / n_pts
        target = float(n_pts // 2)
        n_half = n_pts // 2

        if n_pts > 130 and time.time() > deadline - 90:
            break

        h_init = None
        if parent_h is not None and n_pts != len(parent_h):
            h_init = interpolate_h(parent_h, n_pts)
        elif parent_h is not None:
            h_init = project_hypersimplex(parent_h.copy(), target)
        else:
            h_init = make_golden_start(n_pts, n_half)

        try_candidate(h_init, n_pts)

        t_phase = time.time()
        time_left = deadline - t_phase

        # Phase A: Continuous optimization
        if time_left > 40:
            h_cont, c5_cont = continuous_refine(h_init, dx, target, deadline - 30)
            try_candidate(h_cont, n_pts)

        # Phase B: Binary local search from best continuous
        if time.time() > deadline - 40:
            continue
        h_bin_start = h_cont if 'h_cont' in dir() else h_init
        h_bin_start = project_hypersimplex(h_bin_start, target)

        # Greedy swaps first (fast convergence)
        t_greedy = min(deadline - 50, time.time() + min(30, time_left * 0.15))
        h_g, c5_g = binary_greedy_swap(h_bin_start, dx, max_sweeps=30, deadline=t_greedy)
        try_candidate(h_g, n_pts)

        # SA from greedy result
        t_sa = min(deadline - 20, time.time() + min(60, time_left * 0.3))
        h_sa1, c5_sa1 = binary_sa(h_g, dx, rng, max_iters=200000, patience=5000,
                                  use_2swap=True, deadline=t_sa)
        try_candidate(h_sa1, n_pts)

        # SA from a different start (golden ratio)
        if time.time() > deadline - 40:
            continue
        h_gold = make_golden_start(n_pts, n_half, offset=rng.rand() * 10)
        t_sa2 = min(deadline - 20, time.time() + min(40, time_left * 0.15))
        h_sa2, c5_sa2 = binary_sa(h_gold, dx, rng, max_iters=100000, patience=4000,
                                  use_2swap=True, deadline=t_sa2)
        try_candidate(h_sa2, n_pts)

        # Quadratic residue start
        if time.time() > deadline - 40:
            continue
        h_quad = make_quadratic_start(n_pts, n_half)
        t_sa3 = min(deadline - 15, time.time() + min(30, time_left * 0.1))
        h_sa3, c5_sa3 = binary_sa(h_quad, dx, rng, max_iters=80000, patience=3000,
                                  use_2swap=False, deadline=t_sa3)
        try_candidate(h_sa3, n_pts)

        # Final continuous polish on best binary for this n
        if time.time() > deadline - 20:
            continue
        h_best_n = best_h if best_n == n_pts else h_sa1
        h_polished, c5_pol = continuous_refine(h_best_n, dx, target, deadline - 5)
        try_candidate(h_polished, n_pts)

    # Final: if we have a good continuous result, try one more SA round on it
    if best_h is not None and time.time() > deadline - 30:
        pass  # skip
    else:
        h_final_sa, c5_final = binary_sa(best_h, 2.0 / best_n, rng,
                                         max_iters=50000, patience=3000,
                                         use_2swap=True, deadline=deadline - 3)
        try_candidate(h_final_sa, best_n)

    if best_h is None:
        n_fallback = 126
        h_fallback = make_golden_start(n_fallback, n_fallback // 2)
        best_h = project_hypersimplex(h_fallback, float(n_fallback // 2))
        best_c5 = c5_of_h(best_h, 2.0 / n_fallback)
        best_n = n_fallback

    # Final validation
    best_h = project_hypersimplex(best_h, float(best_n // 2))
    best_c5 = c5_of_h(best_h, 2.0 / best_n)

    return best_h, best_c5, best_n
