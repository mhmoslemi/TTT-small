def project_feasible(x, target_sum):
    """Project x onto {h : 0 <= h <= 1, sum(h) = target_sum}"""
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)

    if abs(diff) > 1e-14:

def compute_corrs(h):

    """Gradient of corr_k(h, 1-h) w.r.t. h"""
    offset = k - (n - 1)

    if offset == 0:

    else:

            grad[idx + offset] += (1.0 - h[idx])
            grad[idx] += (-h[idx + offset])
    return grad

    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        return float(logsumexp(beta * compute_corrs(x)) / beta * dx)
    
    def obj_grad(x):
        corrs = compute_corrs(x)
        shifted = corrs - corrs.max()
        w = np.exp(beta * shifted)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones_like(corrs) / len(corrs)
        else:
            w /= w_sum
        grad = np.zeros(n)
        for k in range(len(corrs)):
            if w[k] < 1e-12:
                continue
            grad += w[k] * grad_corr_k(x, k, n)
        return grad * dx
    
    try:
        res = minimize(obj, h0, jac=obj_grad, method='SLSQP',
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - target_sum,
                                     'jac': lambda x: np.ones(n)}],
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        h_out = project_feasible(res.x, target_sum)
    except Exception:
        h_out = project_feasible(h0, target_sum)
    return h_out, c5_of_h(h_out, dx)

def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid

        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def c5_of_h(h, dx):
    return float(compute_corrs(h).max() * dx)

def grad_corr_k(h, k, n):
    offset = k - (n - 1)
    grad = np.zeros(n)
    if offset == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        i_lo = max(0, -offset)
        i_hi = min(n - 1, n - 1 - offset)
        if i_lo <= i_hi:
            idx = np.arange(i_lo, i_hi + 1)
            grad[idx + offset] += (1.0 - h[idx])
            grad[idx] += (-h[idx + offset])
    return grad

    corrs = compute_corrs(h)
    val = float(logsumexp(beta * corrs) / beta)
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones_like(corrs) / len(corrs)
    else:
        w /= w_sum
    n = len(h)
    grad = np.zeros(n)
    for k in range(len(corrs)):
        if w[k] < 1e-12:
            continue
        grad += w[k] * grad_corr_k(h, k, n)
    return val, grad

    v = np.zeros(n)  # second moment
    best_h = h.copy()
    best_val = c5_of_h(h, dx)

        if time.time() - t_start > time_limit:

        val, grad = smooth_max_and_grad(h, beta)
        grad *= dx

        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        if c5_new < best_val - 1e-15:
            best_val = c5_new
            best_h = h_new.copy()
        
        h = h_new
    
    return best_h, best_val
