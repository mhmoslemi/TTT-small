            no_improve += 1
            alpha *= 0.7
            if alpha < 0.005:
                break
            if no_improve > 50:
                break
    
    return best_h, best_c5


def sa_discrete(n, M, max_iter, t0, budget_s, reserve, rng, h_start=None):
    """Simulated annealing on discrete levels."""
    target = n * M // 2  # sum of a_i should be n*M/2
    
    if target * 2 != n * M:
        return None  # Can't have exact sum with these parameters
    
    # Initialize
    if h_start is not None:
        a = np.round(h_start * M).astype(int)
        a = np.clip(a, 0, M)
        # Fix sum
        diff = target - a.sum()
        # Distribute
        while diff > 0:
            candidates = np.where(a < M)[0]
            if len(candidates) == 0:
                break
            idx = candidates[rng.randint(len(candidates))]
            a[idx] += 1
            diff -= 1
        while diff < 0:
            candidates = np.where(a > 0)[0]
            if len(candidates) == 0:
                break
            idx = candidates[rng.randint(len(candidates))]
            a[idx] -= 1
            diff += 1
    else:
        a = np.full(n, M // 2)
        diff = target - a.sum()
        idx = rng.permutation(n)
        for i in range(len(idx)):
            if diff == 0:
                break
            if diff > 0 and a[idx[i]] < M:
                a[idx[i]] += 1
                diff -= 1
            elif diff < 0 and a[idx[i]] > 0:
                a[idx[i]] -= 1
                diff += 1
    
    h = a.astype(float) / M
    dx = 2.0 / n
    current_c5 = c5_of_h(h, dx)
    best_h = h.copy()
    best_c5 = current_c5
    
    T_start = 0.01
    T_end = 0.0001
    cool_rate = (T_end / T_start) ** (1.0 / max(max_iter, 1))
    
    for it in range(max_iter):
        if time.time() - t0 > budget_s - reserve:
            break
        
        T = T_start * (cool_rate ** it)
        
        # Pick two indices to swap
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        
        # Try to move: a_i += 1, a_j -= 1 (or reverse)
        if a[i] < M and a[j] > 0:
            a[i] += 1
            a[j] -= 1
            h_new = a.astype(float) / M
            new_c5 = c5_of_h(h_new, dx)
            delta_e = new_c5 - current_c5
            if delta_e < 0 or rng.rand() < np.exp(-delta_e / T):
                current_c5 = new_c5
                h = h_new
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.copy()
            else:
                a[i] -= 1
                a[j] += 1
        elif a[i] > 0 and a[j] < M:
            a[i] -= 1
            a[j] += 1
            h_new = a.astype(float) / M
            new_c5 = c5_of_h(h_new, dx)
            delta_e = new_c5 - current_c5
            if delta_e < 0 or rng.rand() < np.exp(-delta_e / T):
                current_c5 = new_c5
                h = h_new
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.copy()
            else:
                a[i] += 1
                a[j] -= 1
        else:
            continue
    
    return best_h, best_c5


def subgradient_min(h_init, dx, target_sum, max_iter, t0, budget_s, reserve):
    """Subgradient method with exact acceptance."""
    n = len(h_init)
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    no_improve = 0
    
    alpha = 1.0 / np.sqrt(n)
    
    for it in range(max_iter):
        if time.time() - t0 > budget_s - reserve:
            break
        
        corrs = compute_corrs(h)
        current_c5 = corrs.max() * dx
        
        if current_c5 < best_c5 - 1e-15:
            best_c5 = current_c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 3000:
            break
        
        # Subgradient: pick the argmax correlation
        k_max = corrs.argmax()
        g = grad_corr_k(h, k_max, n) * dx
        
        # Also blend with top few
        top_k = np.argsort(corrs)[-5:]
        g_smooth = np.zeros(n)
        for k in top_k:
            g_smooth += grad_corr_k(h, k, n)
        g_smooth *= dx / len(top_k)
        
        # Use the subgradient of the max
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            break
        
        step = min(0.5, alpha)
        h_new = h - step * g
        h_new = project_feasible(h_new, target_sum)
        new_c5 = c5_of_h(h_new, dx)
        
        if new_c5 < current_c5:
            h = h_new
            alpha *= 1.01
        else:
            # Try smaller step
            h_new = h - step * 0.5 * g
            h_new = project_feasible(h_new, target_sum)
            new_c5 = c5_of_h(h_new, dx)
            if new_c5 < current_c5:
                h = h_new
            alpha *= 0.95
        
        alpha = max(alpha, 1e-6)
    
    return best_h, best_c5


def make_periodic(n, period, fill, target_sum):
    h = np.zeros(n)
    for i in range(n):
        h[i] = 1.0 if (i % period) < fill else 0.0
    return project_feasible(h, target_sum)


def make_random_binary(n, target_sum, rng):
    h = np.zeros(n)
    idx = rng.permutation(n)
    n_ones = int(round(target_sum))
    h[idx[:n_ones]] = 1.0
    return project_feasible(h, target_sum)


def make_two_block(n, target_sum, rng):
    h = np.zeros(n)
    total_filled = int(round(target_sum))
    b1 = max(1, int(total_filled * rng.uniform(0.3, 0.7)))
    b2 = max(1, total_filled - b1)
    p1 = rng.randint(0, max(1, n - b1))
    h[p1:p1 + b1] = 1.0
    remaining = np.where(h == 0)[0]
    if len(remaining) >= b2:
        p2 = remaining[rng.randint(0, len(remaining) - b2 + 1)]
        h[p2:p2 + b2] = 1.0
    return project_feasible(h, target_sum)


def transfer_grid(h_src, n_target):
    """Interpolate h_src to n_target points."""
    n_src = len(h_src)
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_tgt = (np.arange(n_target) + 0.5) * (2.0 / n_target)
    h_new = np.interp(x_tgt, x_src, h_src)
    target_sum = n_target / 2.0
    return project_feasible(h_new, target_sum)
