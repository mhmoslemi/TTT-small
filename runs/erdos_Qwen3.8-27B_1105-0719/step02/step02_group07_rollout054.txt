def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(70):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:

    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)

            break

        delta = res.x[:n]
        improved = False
        for alpha in [1.0, 0.7, 0.5, 0.3, 0.15, 0.07]:
            h_cand = h + alpha * delta
            h_cand = np.clip(h_cand, 0.0, 1.0)
            h_cand = project_feasible(h_cand, n / 2.0)
            c5_cand = exact_score(h_cand)
            if c5_cand < best_c5 - 1e-10:
                h = h_cand
                best_c5 = c5_cand
                q = make_q_from_h(h, n)
                improved = True
                break

        if not improved:
            break

    q = make_q_from_h(h, n)
    return q, best_c5


def slsqp_sym(q0, n, beta, maxiter=300):
    m = n // 2
    try:
        res = minimize(smooth_sym_obj, q0,
                       jac=lambda x: smooth_sym_grad(x, n, beta),
                       method='SLSQP',
                       bounds=[(0.0, 1.0)] * m,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        h_new = make_sym_h(res.x, n)
        c5 = exact_score(h_new)
        return res.x, c5
    except Exception:
        return None, None


def full_space_smooth(h0, n, beta, maxiter=300):
    target_sum = n / 2.0
    dx = 2.0 / n

    def obj(x):
        h = project_feasible(x, target_sum)
        corrs = corr_raw(h)
        return logsumexp(beta * corrs * dx) / beta

    def obj_grad(x):
        h = project_feasible(x, target_sum)
        corrs = corr_raw(h)
        shifted = corrs * dx - (corrs * dx).max()
        w = np.exp(beta * shifted)
        ws = w.sum()
        if ws < 1e-30:
            w = np.ones(len(corrs)) / len(corrs)
        else:
            w /= ws
        g = np.zeros(n)
        for k in range(len(corrs)):
            if w[k] < 1e-15:
                continue
            d = k - (n - 1)
            i_lo = max(0, -d)
            i_hi = min(n - 1, n - 1 - d)
            if i_lo > i_hi:
                continue
            idx = np.arange(i_lo, i_hi + 1)
            g[idx] += w[k] * (1.0 - h[idx + d])
            g[idx + d] += w[k] * (-h[idx])
        g *= dx
        # Project gradient onto the affine constraint sum(h)=target_sum
        # (approximate: subtract mean of gradient times dx to ensure feasibility direction)
        return g

    try:
        res = minimize(obj, h0, jac=obj_grad, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        h_out = project_feasible(res.x, target_sum)
        c5 = exact_score(h_out)
        return h_out, c5
    except Exception:
        return project_feasible(h0, target_sum), exact_score(h0)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def remaining():
        return budget_s - (time.time() - t0)

    # Get initial solution
    try:
        h_parent = np.array(initial_h_values, dtype=float)
    except NameError:
        h_parent = 0.5 * np.ones(126)

    n = len(h_parent)
    if n % 2 != 0:
        h_parent = h_parent[:n - 1]
        n = len(h_parent)

    h_best = project_feasible(h_parent, n / 2.0)
    c5_best = exact_score(h_best)
    m = n // 2
    q_best = make_q_from_h(h_best, n)
    print(f"Start: n={n}, C5={c5_best:.7f}")

    # Phase 1: Coordinate descent (fast, multiple sweeps)
    if remaining() > 30:
        q_cd, c5_cd = coord_descent_sym(q_best, n, max_sweeps=30)
        if c5_cd < c5_best:
            c5_best = c5_cd
            q_best = q_cd
            h_best = make_sym_h(q_best, n)
            print(f"  CCD: C5={c5_best:.7f}")

    # Phase 2: L-BFGS-B smooth optimization with increasing beta
    for beta in [50, 100, 200, 400, 800, 1500, 3000]:
        if remaining() < 10:
            break
        q_new, c5_new = lbfgsb_sym(q_best, n, beta, maxiter=600)
        if q_new is not None and c5_new < c5_best - 1e-10:
            c5_best = c5_new
            q_best = q_new
            h_best = make_sym_h(q_best, n)
            print(f"  L-BFGS-B beta={beta}: C5={c5_best:.7f}")

    # Phase 3: SLSQP polish
    for beta in [500, 1000, 2000]:
        if remaining() < 15:
            break
        q_new, c5_new = slsqp_sym(q_best, n, beta, maxiter=400)
        if q_new is not None and c5_new < c5_best - 1e-10:
            c5_best = c5_new
            q_best = q_new
            h_best = make_sym_h(q_best, n)
            print(f"  SLSQP beta={beta}: C5={c5_best:.7f}")

    # Phase 4: Active set LP
    for n_act in [20, 40, 60, 80]:
        if remaining() < 15:
            break
        q_new, c5_new = active_set_lp_sym(q_best, n, n_active=n_act, max_rounds=40)
        if c5_new < c5_best - 1e-10:
            c5_best = c5_new
            q_best = q_new
            h_best = make_sym_h(q_best, n)
            print(f"  LP act={n_act}: C5={c5_best:.7f}")

    # Phase 5: CCD again after LP
    if remaining() > 20:
        q_cd2, c5_cd2 = coord_descent_sym(q_best, n, max_sweeps=40)
        if c5_cd2 < c5_best:
            c5_best = c5_cd2
            q_best = q_cd2
            h_best = make_sym_h(q_best, n)
            print(f"  CCD2: C5={c5_best:.7f}")

    # Phase 6: Random restarts in symmetric space
    n_trials = 30
    for trial in range(n_trials):
        if remaining() < 10:
            break
        scale = 0.02 + 0.3 * (trial / max(n_trials - 1, 1))
        q_pert = q_best + scale * rng.randn(m)
        q_pert = np.clip(q_pert, 0.0, 1.0)
        q_new, c5_new = lbfgsb_sym(q_pert, n, 200, maxiter=200)
        if q_new is not None and c5_new < c5_best - 1e-8:
            q_new2, c5_new2 = lbfgsb_sym(q_new, n, 800, maxiter=300)
            if q_new2 is not None and c5_new2 < c5_best - 1e-10:
                c5_best = c5_new2
                q_best = q_new2
                h_best = make_sym_h(q_best, n)
                print(f"  trial {trial}: C5={c5_best:.7f}")
                # Follow up with CCD and LP
                if remaining() > 15:
                    q3, c5_3 = coord_descent_sym(q_best, n, max_sweeps=20)
                    if c5_3 < c5_best:
                        c5_best = c5_3
                        q_best = q3
                        h_best = make_sym_h(q_best, n)
                if remaining() > 15:
                    q4, c5_4 = active_set_lp_sym(q_best, n, n_active=40, max_rounds=20)
                    if c5_4 < c5_best - 1e-10:
                        c5_best = c5_4
                        q_best = q4
                        h_best = make_sym_h(q_best, n)

    # Phase 7: Full space optimization (drop symmetry)
    if remaining() > 30:
        h_full = make_sym_h(q_best, n)
        for beta in [200, 500, 1000]:
            if remaining() < 15:
                break
            h_new, c5_new = full_space_smooth(h_full, n, beta, maxiter=300)
            if c5_new < c5_best - 1e-10:
                c5_best = c5_new
                h_best = h_new
                h_full = h_new
                print(f"  full beta={beta}: C5={c5_best:.7f}")

    # Phase 8: Try different n values
    if remaining() > 60:
        for delta in [2, -2, 4, -4, 6, -6, 8, -8]:
            n_try = n + delta
            if n_try < 50 or n_try % 2 != 0 or remaining() < 25:
                continue
            x_old = np.linspace(0, 2, n, endpoint=False)
            x_new = np.linspace(0, 2, n_try, endpoint=False)
            h_interp = np.interp(x_new, x_old, h_best)
            h_interp = project_feasible(h_interp, n_try / 2.0)
            m_try = n_try // 2
            q_try = make_q_from_h(h_interp, n_try)

            q_new, c5_new = lbfgsb_sym(q_try, n_try, 200, maxiter=300)
            if q_new is not None:
                q_new, c5_new = lbfgsb_sym(q_new, n_try, 800, maxiter=300)
            if q_new is not None and c5_new < c5_best - 1e-9:
                # Polish with CCD
                q_new2, c5_new2 = coord_descent_sym(q_new, n_try, max_sweeps=20)
                if c5_new2 < c5_best - 1e-10:
                    c5_best = c5_new2
                    n = n_try
                    m = m_try
                    q_best = q_new2
                    h_best = make_sym_h(q_best, n)
                    print(f"  n={n}: C5={c5_best:.7f}")

    # Phase 9: Final polish
    if remaining() > 15:
        q_final, c5_final = coord_descent_sym(q_best, n, max_sweeps=50)
        if c5_final < c5_best:
            c5_best = c5_final
            q_best = q_final
            h_best = make_sym_h(q_best, n)
            print(f"  final CCD: C5={c5_best:.7f}")

    if remaining() > 15:
        q_final2, c5_final2 = lbfgsb_sym(q_best, n, 2000, maxiter=500)
        if q_final2 is not None and c5_final2 < c5_best - 1e-10:
            c5_best = c5_final2
            q_best = q_final2
            h_best = make_sym_h(q_best, n)
            print(f"  final L-BFGS: C5={c5_best:.7f}")

    if remaining() > 15:
        q_final3, c5_final3 = active_set_lp_sym(q_best, n, n_active=50, max_rounds=30)
        if c5_final3 < c5_best - 1e-10:
            c5_best = c5_final3
            q_best = q_final3
            h_best = make_sym_h(q_best, n)
            print(f"  final LP: C5={c5_best:.7f}")

    # Final validation
    h_best = np.clip(h_best, 0.0, 1.0)
    h_best = project_feasible(h_best, len(h_best) / 2.0)
    c5_best = exact_score(h_best)
    print(f"Result: n={len(h_best)}, C5={c5_best:.7f}")

    return h_best, c5_best, len(h_best)
