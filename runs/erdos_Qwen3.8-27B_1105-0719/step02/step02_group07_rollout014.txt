```python
def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)


def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full').max() * dx


def make_sym_h(q, n):
    m = n // 2
    q = np.clip(q, 0.0, 1.0)
    h = np.empty(n)
    h[:m] = q
    h[m:] = 1.0 - q[::-1]
    return h

    m = n // 2
    q = 0.5 * (h[:m] + 1.0 - h[n - 1: n - 1 - m: -1])
    return np.clip(q, 0.0, 1.0)


def corr_vals(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

    m = n // 2

        order = np.arange(n // 2)
    for i in order:
        q[i] = optimize_coord(q, n, i)
    return q

    corrs = corr_vals(h)
    return logsumexp(beta * corrs) / beta


def smooth_sym_grad(x, n, beta):
    m = n // 2
    dx = 2.0 / n
    h = make_sym_h(x, n)
    corrs = corr_vals(h)
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)

    else:
        w /= ws

    grad = np.zeros(n)

class EpiSymSolver:
    def __init__(self, n, active_lags):
        self.n = n
        self.m = n // 2
        self.dx = 2.0 / n
        self.active_lags = np.array(active_lags, dtype=int)

    def con_fun(self, x):
        t = x[0]
        q = x[1:]
        h = make_sym_h(q, self.n)
        corrs = corr_vals(h)
        return t - corrs[self.active_lags]

    def con_jac(self, x):
        q = x[1:]
        h = make_sym_h(q, self.n)
        n = self.n
        m = self.m
        dx = self.dx
        L = len(self.active_lags)
        J = np.zeros((L, m + 1))
        J[:, 0] = 1.0
        for i, lag in enumerate(self.active_lags):
            d = lag - (n - 1)
            i_lo = max(0, -d)
            i_hi = min(n - 1, n - 1 - d)
            G = np.zeros(n)
            if i_lo <= i_hi:
                idx = np.arange(i_lo, i_hi + 1)
                G[idx] += (1.0 - h[idx + d])
                G[idx + d] += (-h[idx])
            G *= dx
            J[i, 1:] = -(G[:m] - G[n - 1: n - 1 - m: -1])
        return J
```

```python
def optimize_coord(q, n, i):
    m = n // 2
    q0 = q.copy(); q0[i] = 0.0
    corrs0 = corr_vals(make_sym_h(q0, n))
    q1 = q.copy(); q1[i] = 1.0
    corrs1 = corr_vals(make_sym_h(q1, n))
    qh = q.copy(); qh[i] = 0.5
    corrs_h = corr_vals(make_sym_h(qh, n))
    
    a = corrs1 - corrs0
    b = corrs0
    is_quad = np.abs(corrs_h - (b + 0.5*a)) > 1e-12
    iq = np.where(is_quad)[0] if is_quad.any() else np.array([], dtype=int)
    
    def f(x):
        vals = b + x * a
        if len(iq) > 0:
            vals[iq] = (2*b[iq]*(x-0.5)*(x-1)
                      - 4*corrs_h[iq]*x*(x-1)
                      + 2*corrs1[iq]*x*(x-0.5))
        return vals.max()
    
    # Dense grid
    xs = np.linspace(0, 1, 200)
    fvals = np.array([f(x) for x in xs])
    i_min = np.argmin(fvals)
    
    # Local golden section refinement
    if i_min == 0:
        lo, hi = 0.0, xs[1]
    elif i_min == len(xs) - 1:
        lo, hi = xs[-2], 1.0
    else:
        lo, hi = xs[i_min-1], xs[i_min+1]
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr*(hi - lo)
    d = lo + gr*(hi - lo)
    fc, fd = f(c), f(d)
    for _ in range(50):
        if fc < fd:
            hi = d; d = c; fd = fc
            c = hi - gr*(hi - lo); fc = f(c)
        else:
            lo = c; c = d; fc = fd
            d = lo + gr*(hi - lo); fd = f(d)
    
    return (lo + hi) / 2.0
```

def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)


def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full').max() * dx


def make_sym_h(q, n):
    m = n // 2
    q = np.clip(q, 0.0, 1.0)
    h = np.empty(n)
    h[:m] = q
