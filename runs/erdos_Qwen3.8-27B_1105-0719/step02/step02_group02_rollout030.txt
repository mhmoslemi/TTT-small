        mid = (lo + hi) / 2.0
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            adj = diff / mask.sum()
            h[mask] = np.clip(h[mask] + adj, 0.0, 1.0)
            diff2 = target_sum - h.sum()
            if abs(diff2) > 1e-12:
                mask2 = (h > 0) & (h < 1)
                if mask2.sum() > 0:
                    h[mask2] += diff2 / mask2.sum()
                    h = np.clip(h, 0.0, 1.0)
    return h


def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)


def smooth_max_val(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(logsumexp(beta * corr) / beta * dx)


def grad_corr_single(h, k, n):
    offset = k - (n - 1)
    grad = np.zeros(n)
    if offset == 0:
        grad[:] = 1.0 - 2.0 * h
        return grad
    i_lo = max(0, -offset)
    i_hi = min(n - 1, n - 1 - offset)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx + offset] += (1.0 - h[idx])
    grad[idx] += (-h[idx + offset])
    return grad


def smooth_max_grad(h, dx, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    shifted = corr - corr.max()
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        w /= w_sum
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        grad += w[k] * grad_corr_single(h, k, n)
    return grad * dx


def lbfgs_opt(h0, dx, target_sum, beta, maxiter, ftol):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n

    def obj(x):
        return smooth_max_val(x, dx, beta)

    def g(x):
        return smooth_max_grad(x, dx, beta)

    cons = {'type': 'eq', 'fun': lambda x: x.sum() - target_sum,
            'jac': lambda x: np.ones(n)}
    try:
        res = minimize(obj, h0, jac=g, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': ftol,
                                'gtol': 1e-8, 'maxls': 30})
        h_out = project_feasible(res.x, target_sum)
    except Exception:
        h_out = project_feasible(h0, target_sum)
    return h_out, compute_c5(h_out, dx)


def block_coord_descent(h, dx, target_sum, beta, max_passes=100, tol=1e-12):
    n = len(h)
    h = h.copy()
    best_c5 = compute_c5(h, dx)

    for p in range(max_passes):
        max_change = 0.0
        for i in range(n):
            old_val = h[i]

            def f_1d(t):
                h[i] = t
                return smooth_max_val(h, dx, beta)

            res = minimize_scalar(f_1d, bounds=(0.0, 1.0),
                                  method='bounded',
                                  options={'xatol': 1e-8})
            h[i] = res.x
            change = abs(h[i] - old_val)
            max_change = max(max_change, change)

        h = project_feasible(h, target_sum)
        c5_now = compute_c5(h, dx)
        if c5_now < best_c5:
            best_c5 = c5_now
        if max_change < tol:
            break
    return h, best_c5


def binary_sa(h0, dx, target_sum, n_iter=20000, t_start=0.5, t_end=0.01):
    n = len(h0)
    binary = (h0 > 0.5).astype(float)
    # Ensure exact sum
    n_ones = int(binary.sum())
    target_int = int(target_sum)
    if n_ones > target_int:
        ones_idx = np.where(binary == 1)[0]
        excess = n_ones - target_int
        binary[ones_idx[:excess]] = 0
    elif n_ones < target_int:
        zeros_idx = np.where(binary == 0)[0]
        deficit = target_int - n_ones
        binary[zeros_idx[:deficit]] = 1

    cur_bin = binary.copy()
    cur_c5 = compute_c5(cur_bin, dx)
    best_bin = cur_bin.copy()
    best_c5 = cur_c5

    for it in range(n_iter):
        T = t_start * (t_end / t_start) ** (it / max(n_iter, 1))
        # Pick a random position to flip
        i = it % n
        new_bin = cur_bin.copy()
        if cur_bin[i] > 0:
            # Try to flip 1->0, need to flip a 0->1 elsewhere to maintain sum
            j = (i + 1 + (it * 7) % (n - 1)) % n
            if new_bin[j] == 0:
                new_bin[i] = 0
                new_bin[j] = 1
            else:
                continue
        else:
            j = (i + 1 + (it * 7) % (n - 1)) % n
            if new_bin[j] == 1:
                new_bin[i] = 1
                new_bin[j] = 0
            else:
                continue

        new_c5 = compute_c5(new_bin, dx)
        delta = new_c5 - cur_c5
        if delta < 0 or np.exp(-delta / max(T, 1e-10)) > (it % 1000) / 1000.0:
            cur_bin = new_bin
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_bin = cur_bin.copy()

    # Do continuous relaxation of best binary
    best_h = project_feasible(best_bin * 1.0, target_sum)
    return best_h, best_c5, best_bin


def resample_to_n(h_old, n_new):
    n_old = len(h_old)
    x_old = np.arange(n_old) * (2.0 / n_old)
    x_new = np.arange(n_new) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h_old)
    target = n_new / 2.0
    return project_feasible(h_new, target)


def subgrad_momentum(h0, dx, target_sum, max_iter=20000):
    n = len(h0)
    h = h0.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    alpha = 0.1
    gamma = 0.8
    no_improve = 0
    top_m = max(8, n // 8)

    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = corr.max() * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 8000:
            break

        sorted_idx = np.argsort(corr)[-top_m:]
        grad = np.zeros(n)
        for k in sorted_idx:
            grad += grad_corr_single(h, k, n)
        grad /= len(sorted_idx)
        grad *= dx

        v = gamma * v - alpha * grad
        h_new = project_feasible(h + v, target_sum)
        new_c5 = compute_c5(h_new, dx)

        if new_c5 < c5_val - 1e-16:
            h = h_new
            alpha = min(alpha * 1.005, 0.35)
        else:
            alpha *= 0.95
            v *= 0.4
        if alpha < 1e-13:
            break

    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def remaining():
        return budget_s - (time.time() - t0)

    # Initialize from parent
    h0 = np.array(initial_h_values, dtype=float)
    n = len(h0)
    dx = 2.0 / n
    target_sum = n / 2.0
    h0 = project_feasible(h0, target_sum)
    best_h = h0.copy()
    best_c5 = compute_c5(h0, dx)

    # Phase 1: L-BFGS-B with increasing beta (continuation)
    betas = [30, 80, 200, 500, 1200, 3000, 6000, 12000]
    for beta in betas:
        if remaining() < 30:
            break
        h_opt, c5_val = lbfgs_opt(best_h, dx, target_sum, beta, 400, 1e-14)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_opt

    # Phase 2: Block coordinate descent (key new method)
    if remaining() > 200:
        for beta in [500, 2000, 8000]:
            if remaining() < 50:
                break
            h_bcd, c5_bcd = block_coord_descent(best_h, dx, target_sum, beta,
                                                 max_passes=60)
            if c5_bcd < best_c5:
                best_c5 = c5_bcd
                best_h = h_bcd

    # Phase 3: Subgradient momentum
    if remaining() > 150:
        h_sg, c5_sg = subgrad_momentum(best_h, dx, target_sum, 25000)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg

    # Phase 4: Binary local search with SA
    if remaining() > 120:
        h_bin, c5_bin, bin_pattern = binary_sa(best_h, dx, target_sum,
                                                n_iter=15000)
        if c5_bin < best_c5:
            # Polish the binary solution
            h_pol, c5_pol = lbfgs_opt(h_bin, dx, target_sum, 500, 300, 1e-14)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol
            elif c5_bin < best_c5:
                best_c5 = c5_bin
                best_h = h_bin
        # Also try some random binary patterns
        for trial in range(5):
            if remaining() < 30:
                break
            h_rand = np.zeros(n)
            idx = rng.permutation(n)
            h_rand[idx[:int(target_sum)]] = 1.0
            h_b2, c5_b2, _ = binary_sa(h_rand, dx, target_sum, n_iter=8000)
            if c5_b2 < best_c5:
                h_pol, c5_pol = lbfgs_opt(h_b2, dx, target_sum, 500, 200, 1e-14)
                if c5_pol < best_c5:
                    best_c5 = c5_pol
                    best_h = h_pol

    # Phase 5: More block coordinate refinement
    if remaining() > 100:
        for beta in [5000, 15000]:
            if remaining() < 30:
                break
            h_bcd, c5_bcd = block_coord_descent(best_h, dx, target_sum, beta,
                                                 max_passes=80)
            if c5_bcd < best_c5:
                best_c5 = c5_bcd
                best_h = h_bcd

    # Phase 6: Dimension exploration
    n_b = len(best_h)
    dim_candidates = []
    for dn in [-12, -8, -6, -4, -2, 2, 4, 6, 8, 12]:
        nt = n_b + dn
        if nt >= 40 and nt % 2 == 0:
            dim_candidates.append(nt)
    dim_candidates = sorted(set(dim_candidates))

    for n_try in dim_candidates:
        if remaining() < 45:
            break
        dx_try = 2.0 / n_try
        ts_try = n_try / 2.0
        h_interp = resample_to_n(best_h, n_try)
        h_opt, c5_val = lbfgs_opt(h_interp, dx_try, ts_try, 800, 300, 1e-14)
        if c5_val < best_c5:
            # Polish with block coord
            if remaining() > 35:
                h_bc, c5_bc = block_coord_descent(h_opt, dx_try, ts_try, 5000, max_passes=40)
                if c5_bc < c5_val:
                    c5_val = c5_bc
                    h_opt = h_bc
            best_c5 = c5_val
            best_h = h_opt

    # Phase 7: Perturbation + re-optimization
    for trial in range(10):
        if remaining() < 20:
            break
        mag = 0.01 + 0.03 * (trial / 10.0)
        h_pert = best_h + rng.normal(0, mag, len(best_h))
        h_pert = project_feasible(h_pert, target_sum)
        n_p = len(h_pert)
        dx_p = 2.0 / n_p
        ts_p = n_p / 2.0
        h_opt, c5_val = lbfgs_opt(h_pert, dx_p, ts_p, 1000, 300, 1e-14)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_opt

    # Phase 8: Final intensive polish
    n_f = len(best_h)
    dx_f = 2.0 / n_f
    ts_f = n_f / 2.0
    if remaining() > 30:
        for beta in [8000, 15000, 25000]:
            if remaining() < 15:
                break
            h_final, c5_final = lbfgs_opt(best_h, dx_f, ts_f, beta, 600, 1e-15)
            if c5_final < best_c5:
                best_c5 = c5_final
                best_h = h_final

    if remaining() > 20:
        h_bcd, c5_bcd = block_coord_descent(best_h, dx_f, ts_f, 20000, max_passes=100)
        if c5_bcd < best_c5:
            best_c5 = c5_bcd
            best_h = h_bcd

    if remaining() > 15:
        h_sg, c5_sg = subgrad_momentum(best_h, dx_f, ts_f, 15000)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg

    # Final validation and return
    best_h = project_feasible(best_h, len(best_h) / 2.0)
    final_c5 = compute_c5(best_h, 2.0 / len(best_h))
    if final_c5 > best_c5 + 1e-10:
        # Something went wrong, use the safer value
        final_c5 = best_c5

    return best_h, final_c5, len(best_h)
