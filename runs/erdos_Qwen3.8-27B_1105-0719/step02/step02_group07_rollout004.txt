    """Minimize exact_score along coordinate i of q."""
    m = n // 2
    q0 = q.copy()
    q0[i] = 0.0  # will be set in the loop
    
    def phi(t):
        q0[i] = np.clip(q[i] + t, 0.0, 1.0)
        h = make_sym_h(q0, n)
        return exact_score(h)
    
    # Golden section
    a, b = t_range
    gr = (np.sqrt(5) - 1) / 2
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = phi(c)
    fd = phi(d)
    for _ in range(n_golden):
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = phi(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = phi(d)
    
    t_best = (a + b) / 2
    q_new = q.copy()
    q_new[i] = np.clip(q[i] + t_best, 0.0, 1.0)
    h_new = make_sym_h(q_new, n)
    return q_new, exact_score(h_new)

Actually, for a single coordinate change, only the correlations involving index i or n-1-i change. Let me precompute the base correlations and then update them.

When q[i] changes by δ:
- h[i] changes by δ
- h[n-1-i] changes by -δ

For correlation at lag d (displacement):
- If i is in the valid range for lag d: corr_d changes by δ*(1-h[i+d])
- If i+d is in the valid range for lag d: corr_d changes by -h[i]*(-δ) = h[i]*δ... 

Wait, let me be more careful.

corr_d = Σ_{valid j} h[j] * (1-h[j+d])

If h[i] changes by δ (i < m):
- Terms where j = i: h[i] changes to h[i]+δ, so the term changes by δ*(1-h[i+d]) (if i+d is valid)
- Terms where j+d = i, i.e., j = i-d: the term (1-h[j+d]) = (1-h[i]) changes to (1-h[i]-δ), so the term changes by -h[j]*δ = -h[i-d]*δ (if i-d is valid)

Similarly for h[n-1-i] changing by -δ (n-1-i >= m):
- Terms where j = n-1-i: h[n-1-i] changes to h[n-1-i]-δ, term changes by -δ*(1-h[n-1-i+d]) (if valid)
- Terms where j+d = n-1-i, i.e., j = n-1-i-d: term changes by h[j]*δ (if valid)

So for each lag d, at most 4 terms change. This means I can update all correlations in O(n) time per evaluation (iterate over the 2n-1 lags and update at most 4 terms each).

import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time

def make_sym_h(q, n):
    m = n // 2
    q = np.clip(q, 0.0, 1.0)
    h = np.empty(n)
    h[:m] = q
    h[m:] = 1.0 - q[::-1]
    return h

def make_q_from_h(h, n):
    m = n // 2
    q = 0.5 * (h[:m] + 1.0 - h[m:][::-1])
    return np.clip(q, 0.0, 1.0)

def corr_vals(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def exact_score(h):
    return corr_vals(h).max()

def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def sym_smooth_optimize(q0, n, beta, maxiter=500):
    m = n // 2
    def obj(q):
        return sym_obj(q, n, beta)
    def grad(q):
        return sym_grad(q, n, beta)
    try:
        res = minimize(obj, q0, jac=grad, method='SLSQP',
                       bounds=[(0.0, 1.0)] * m,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        h_new = make_sym_h(res.x, n)
        c5 = exact_score(h_new)
        return res.x, c5
    except Exception:
        return None, None

def sym_obj(q, n, beta):
    h = make_sym_h(q, n)
    c = corr_vals(h)
    return logsumexp(beta * c) / beta

def sym_grad(q, n, beta):
    m = n // 2
    dx = 2.0 / n
    h = make_sym_h(q, n)
    c = corr_vals(h)
    shifted = c - c.max()
    w = np.exp(beta * shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.ones(len(c)) / len(c)
    else:
        w /= ws
    gh = np.zeros(n)
    n_lags = 2 * n - 1
    for k in range(n_lags):
        wk = w[k]
        if wk < 1e-14:
            continue
        d = k - (n - 1)
        i_lo = max(0, -d)
        i_hi = min(n - 1, n - 1 - d)
        if i_lo > i_hi:
            continue
        idx = np.arange(i_lo, i_hi + 1)
        gh[idx] += wk * dx * (1.0 - h[idx + d])
        gh[idx + d] += wk * dx * (-h[idx])
    gq = gh[:m] - gh[n-1:m-1:-1]
    return gq

def coord_descent(q0, n, n_passes=5, n_golden=30):
    """Coordinate descent with exact line search."""
    m = n // 2
    q = q0.copy()
    h = make_sym_h(q, n)
    c5 = exact_score(h)
    for p in range(n_passes):
        for i in range(m):
            q0_save = q[i]
            def phi(t):
                q[i] = np.clip(q0_save + t, 0.0, 1.0)
                h_t = make_sym_h(q, n)
                return exact_score(h_t)
            # Find bracket
            t_lo, t_hi = -1.0, 1.0
            f_lo, f_mid, f_hi = phi(-0.5), phi(0.0), phi(0.5)
            # Simple golden section in [-1, 1]
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            a, b = -1.0, 1.0
            c_pt = b - gr * (b - a)
            d_pt = a + gr * (b - a)
            fc = phi(c_pt)
            fd = phi(d_pt)
            for _ in range(n_golden):
                if fc < fd:
                    b = d_pt
                    d_pt = c_pt
                    fd = fc
                    c_pt = b - gr * (b - a)
                    fc = phi(c_pt)
                else:
                    a = c_pt
                    c_pt = d_pt
                    fc = fd
                    d_pt = a + gr * (b - a)
                    fd = phi(d_pt)
            t_best = (a + b) / 2.0
            q[i] = np.clip(q0_save + t_best, 0.0, 1.0)
            h = make_sym_h(q, n)
            c5_new = exact_score(h)
            if c5_new < c5 - 1e-12:
                c5 = c5_new
    return q, c5
