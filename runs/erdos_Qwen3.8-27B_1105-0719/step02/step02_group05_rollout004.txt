    scores = np.zeros(pop_size)

    n_seeds = min(len(seeds), pop_size)
    for i in range(pop_size):
        if i < n_seeds:
            pop[i] = np.clip(seeds[i], 0.0, 1.0)
        else:
            pop[i] = rng.uniform(0, 1, m)
        h = build_h_from_q(pop[i], n)
        scores[i] = np.correlate(h, 1.0 - h, mode='full').max() * dx

    best_score = scores.min()
    best_q = pop[scores.argmin()].copy()
    no_improve = 0

    while time.time() - t_start < time_limit:
        improved_this_gen = False
        idx_best = scores.argmin()
        for i in range(pop_size):
            if time.time() - t_start > time_limit:
                break
            strategy = rng.integers(3)
            a, b, c = rng.choice(pop_size, 3, replace=False)
            if strategy == 0:
                F = rng.uniform(0.3, 0.9)
                v = pop[a] + F * (pop[b] - pop[c])
            elif strategy == 1:
                F = rng.uniform(0.2, 0.8)
                v = pop[idx_best] + F * (pop[a] - pop[b])
            else:
                F = rng.uniform(0.2, 0.6)
                v = pop[i] + F * (pop[idx_best] - pop[i]) + F * (pop[a] - pop[b])
            v = np.clip(v, 0.0, 1.0)
            CR = rng.uniform(0.3, 0.8)
            j_rand = rng.integers(m)
            trial = pop[i].copy()
            mask = rng.random(m) < CR
            mask[j_rand] = True
            trial[mask] = v[mask]
            trial = np.clip(trial, 0.0, 1.0)
            h = build_h_from_q(trial, n)
            trial_score = np.correlate(h, 1.0 - h, mode='full').max() * dx
            if trial_score <= scores[i]:
                pop[i] = trial
                scores[i] = trial_score
                if trial_score < best_score - 1e-14:
                    best_score = trial_score
                    best_q = trial.copy()
                    no_improve = 0
                    improved_this_gen = True
                else:
                    no_improve += 1
        if not improved_this_gen:
            no_improve += pop_size
        if no_improve > 4 * pop_size:
            idx = np.argsort(scores)[pop_size // 2:]
            for j in idx:
                pop[j] = np.clip(pop[j] + rng.normal(0, 0.15, m), 0, 1)
            no_improve = 0
            idx_best = scores.argmin()

    return best_q, best_score

def polish_q(q0, n, t_start, time_limit, rng):
    m = n // 2
    best_q = q0.copy()
    best_c5, _ = exact_score_q(q0, n)

    for beta in [100, 300, 800, 2000]:
        if time.time() - t_start > time_limit - 5:
            break
        q_smooth = smooth_q_solve(best_q, n, beta, maxiter=300)
        s, _ = exact_score_q(q_smooth, n)
        if s < best_c5 - 1e-14:
            best_c5 = s
            best_q = q_smooth

    if time.time() - t_start < time_limit - 10:
        q_cd, c5_cd = coordinate_descent_q(best_q, n, max_passes=2,
                                           t_start=t_start, time_limit=time_limit - 5)
        if c5_cd < best_c5 - 1e-14:
            best_c5 = c5_cd
            best_q = q_cd

    if time.time() - t_start < time_limit - 15:
        q_as, c5_as = active_set_q(best_q, n, max_rounds=6, max_active=40)
        if c5_as < best_c5 - 1e-14:
            best_c5 = c5_as
            best_q = q_as

    if time.time() - t_start < time_limit - 10:
        for beta in [500, 1500]:
            if time.time() - t_start > time_limit - 5:
                break
            q_smooth = smooth_q_solve(best_q, n, beta, maxiter=200)
            s, _ = exact_score_q(q_smooth, n)
            if s < best_c5 - 1e-14:
                best_c5 = s
                best_q = q_smooth

    if time.time() - t_start < time_limit - 10:
        q_cd2, c5_cd2 = coordinate_descent_q(best_q, n, max_passes=1,
                                             t_start=t_start, time_limit=time_limit - 3)
        if c5_cd2 < best_c5 - 1e-14:
            best_c5 = c5_cd2
            best_q = q_cd2

    return best_q, best_c5

def generate_seeds(base_q, n, rng, num_seeds=10):
    m = n // 2
    seeds = []
    seeds.append(base_q.copy())
    seeds.append(1.0 - base_q.copy())
    seeds.append(base_q[::-1].copy())
    seeds.append(1.0 - base_q[::-1].copy())
    for _ in range(4):
        sig = rng.choice([0.02, 0.05, 0.1, 0.2])
        seeds.append(np.clip(base_q + rng.normal(0, sig, m), 0, 1))
    for _ in range(num_seeds - 8):
        if num_seeds > 8:
            seeds.append(rng.uniform(0, 1, m))
    return seeds[:num_seeds]

def interpolate_q(src_q, src_n, target_n):
    src_m = src_n // 2
    tgt_m = target_n // 2
    if src_m == tgt_m:
        return src_q.copy()
    x_src = np.linspace(0, 1, src_m)
    x_tgt = np.linspace(0, 1, tgt_m)
    return np.interp(x_tgt, x_src, src_q)

def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)

    def elapsed():
        return time.time() - t_start

    def remaining():
        return budget_s - elapsed()

    # Initialize from parent
    try:
        h_parent = np.array(initial_h_values, dtype=float)
        n_parent = len(h_parent)
    except (NameError, UnboundLocalError):
        n_parent = 126
        h_parent = np.clip(0.5 + 0.3 * np.sin(np.linspace(0, 4 * np.pi, n_parent)), 0, 1)

    target_sum = n_parent / 2.0
    h_parent = project_feasible(h_parent, target_sum)
    score_parent, _ = exact_score_h(h_parent)
    best_h = h_parent.copy()
    best_c5 = score_parent
    best_n = n_parent

    # Convert to q-space for even n
    if n_parent % 2 == 0:
        base_q = q_from_h(h_parent, n_parent)
    else:
        n_parent = 126
        x_orig = np.linspace(0, 2, len(h_parent))
        x_new = np.linspace(0, 2, n_parent)
        h_parent = np.interp(x_new, x_orig, h_parent)
        h_parent = project_feasible(h_parent, n_parent / 2.0)
        base_q = q_from_h(h_parent, n_parent)
        score_parent, _ = exact_score_h(h_parent)
        best_h = h_parent.copy()
        best_c5 = score_parent
        best_n = n_parent

    # Try initial transforms
    transforms = [base_q, 1.0 - base_q, base_q[::-1], 1.0 - base_q[::-1]]
    for tq in transforms:
        tq = np.clip(tq, 0, 1)
        s, _ = exact_score_q(tq, n_parent)
        if s < best_c5 - 1e-14:
            best_c5 = s
            best_h = build_h_from_q(tq, n_parent)
            base_q = tq

    n = n_parent
    m = n // 2

    # Phase 1: Quick smooth polish on parent
    if remaining() > 50:
        for beta in [100, 300, 800]:
            if remaining() < 15:
                break
            q_s = smooth_q_solve(base_q, n, beta, maxiter=300)
            s, _ = exact_score_q(q_s, n)
            if s < best_c5 - 1e-14:
                best_c5 = s
                best_h = build_h_from_q(q_s, n)
                base_q = q_s

    # Phase 2: Active set epigraph on parent size
    if remaining() > 60:
        q_as, c5_as = active_set_q(base_q, n, max_rounds=8, max_active=45)
        if c5_as < best_c5 - 1e-14:
            best_c5 = c5_as
            best_h = build_h_from_q(q_as, n)
            base_q = q_as

    # Phase 3: Coordinate descent
    if remaining() > 30:
        q_cd, c5_cd = coordinate_descent_q(best_h if len(best_h) == n else base_q, n,
                                           max_passes=2, t_start=t_start,
                                           time_limit=remaining() - 10)
        if c5_cd < best_c5 - 1e-14:
            best_c5 = c5_cd
            best_h = build_h_from_q(q_cd, n)
            base_q = q_cd

    # Phase 4: Multi-resolution DE exploration
    # Try several sizes
    sizes_to_try = []
    if n <= 80:
        sizes_to_try = sorted(set([48, 62, 64, 72, 80, 96, n] + list(range(40, 80, 4))))
    elif n <= 140:
        sizes_to_try = sorted(set([48, 64, 80, 96, 112, 126, 128, 144]))
    else:
        sizes_to_try = sorted(set([64, 80, 96, 112, 126, 128, 144, 160]))

    # Filter: only even sizes, and not too many
    sizes_to_try = [s for s in sizes_to_try if s % 2 == 0 and s >= 40]
    # Cap the list
    sizes_to_try = sizes_to_try[:12]

    de_best_per_size = {}

    for sz in sizes_to_try:
        if remaining() < 45:
            break
        sz_m = sz // 2

        # Interpolate current best to this size
        if len(base_q) == sz_m:
            seed_q = base_q.copy()
        elif best_n % 2 == 0:
            seed_q = interpolate_q(base_q, best_n, sz)
        else:
            x_old = np.linspace(0, 1, m)
            x_new = np.linspace(0, 1, sz_m)
            seed_q = np.interp(x_new, x_old, base_q)
        seed_q = np.clip(seed_q, 0, 1)

        # Generate diverse seeds
        seeds = generate_seeds(seed_q, sz, rng, num_seeds=8)

        # Determine DE time budget for this size
        de_time = min(remaining() * 0.35, 60.0)
        if sz <= 64:
            de_time = min(de_time, 35.0)
        elif sz <= 96:
            de_time = min(de_time, 50.0)

        if de_time < 10:
            de_time = 10

        # Run DE
        de_q_best, de_c5 = de_q(seeds, sz, t_start=elapsed() + t_start,
                                 time_limit=de_time, rng=rng)

        # Polish the DE result
        if remaining() > 25:
            pol_q, pol_c5 = polish_q(de_q_best, sz, t_start, remaining() - 5)
            if pol_c5 < de_c5:
                de_c5 = pol_c5
                de_q_best = pol_q

        de_best_per_size[sz] = (de_q_best, de_c5)

        if de_c5 < best_c5 - 1e-12:
            best_c5 = de_c5
            best_h = build_h_from_q(de_q_best, sz)
            best_n = sz
            base_q = de_q_best

    # Phase 5: Re-interpolate best to parent size and try to combine
    if remaining() > 40 and best_n != n_parent:
        # Interpolate best back to parent size
        best_q_interp = interpolate_q(q_from_h(best_h, best_n), best_n, n_parent)
        best_q_interp = np.clip(best_q_interp, 0, 1)
        s_interp, _ = exact_score_q(best_q_interp, n_parent)

        # Also try at the best_n
        best_q_at_best_n = q_from_h(best_h, best_n)

        # Polish at parent size
        if remaining() > 30:
            pol = polish_q(best_q_interp, n_parent, t_start, remaining() - 10)
            s_pol, _ = exact_score_q(pol[0], n_parent)
            if s_pol < best_c5 - 1e-14:
                best_c5 = s_pol
                best_h = build_h_from_q(pol[0], n_parent)
                best_n = n_parent

    # Phase 6: Final intensive polish on the best
    if remaining() > 30:
        best_q = q_from_h(best_h, best_n)
        final_polish = polish_q(best_q, best_n, t_start, remaining() - 5)
        if final_polish[1] < best_c5 - 1e-14:
            best_c5 = final_polish[1]
            best_h = build_h_from_q(final_polish[0], best_n)
            best_n = len(best_h)

    # Phase 7: One more coordinate descent pass
    if remaining() > 15 and best_n % 2 == 0:
        best_q = q_from_h(best_h, best_n)
        q_final, c5_final = coordinate_descent_q(best_q, best_n, max_passes=2,
                                                 t_start=t_start, time_limit=remaining() - 3)
        if c5_final < best_c5 - 1e-14:
            best_c5 = c5_final
            best_h = build_h_from_q(q_final, best_n)
            best_n = len(best_h)

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    n_out = len(best_h)
    ts_out = n_out / 2.0
    # Ensure feasibility
    if abs(best_h.sum() - ts_out) > 1e-10:
        best_h = project_feasible(best_h, ts_out)
    final_score, _ = exact_score_h(best_h)
    if final_score < best_c5:
        best_c5 = final_score
    best_c5 = final_score

    return best_h, best_c5, n_out
