1. Identify which correlations are affected by changing q[r]
2. For those correlations, compute their slope and intercept as functions of alpha
3. The max is the upper envelope of these linear functions
4. Find the minimum of this upper envelope (which occurs at a breakpoint or at a boundary)

The upper envelope of these lines has its minimum at:
- A boundary (alpha = 0 or alpha = 1), or
- An intersection point of two lines

def eval_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def corr_raw(h):
    return np.correlate(h, 1.0 - h, mode='full')

def build_sym_h(q):
    m = len(q)
    h = np.empty(2 * m)
    h[:m] = q
    h[m:] = 1.0 - q[::-1]
    return h

def sym_from_h(h):
    n = len(h)
    m = n // 2
    q = np.empty(m)
    for i in range(m):
        q[i] = 0.5 * (h[i] + (1.0 - h[n - 1 - i]))
    return np.clip(q, 0.0, 1.0)

def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.5, 1.5
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(x + mid, 0.0, 1.0).sum()
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    h = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        interior = (h > 1e-10) & (h < 1.0 - 1e-10)
        if interior.sum() > 0:
            h[interior] += diff / interior.sum()
    return np.clip(h, 0.0, 1.0)

def corr_at_k(h, k):
    """Compute correlation at lag index k (0-indexed in full correlation output)."""
    n = len(h)
    d = k - (n - 1)
    i_lo = max(0, -d)
    i_hi = min(n - 1, n - 1 - d)
    if i_lo > i_hi:
        return 0.0
    return float(np.sum(h[i_lo:i_hi + 1] * (1.0 - h[i_lo + d:i_hi + 1 + d])))

def corr_grad_k(h, k):
    """Gradient of correlation at lag k with respect to h."""
    n = len(h)
    d = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -d)
    i_hi = min(n - 1, n - 1 - d)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx] += (1.0 - h[idx + d])
    grad[idx + d] += (-h[idx])
    return grad

def grad_q_from_h(h, k):
    """Gradient of correlation at lag k with respect to q (symmetric parameterization)."""
    n = len(h)
    m = n // 2
    g_h = corr_grad_k(h, k)
    g_q = np.zeros(m)
    for r in range(m):
        g_q[r] = g_h[r] - g_h[n - 1 - r]
    return g_q

    padded = set()
    for k in active:
        for d in range(-neighbor_pad, neighbor_pad + 1):

                padded.add(k + d)

        scored = sorted([(corrs_scaled[k], k) for k in active_list], reverse=True)

    return active_list

def lp_trust_region(q0, n_full, active_k, delta):
    """Solve LP trust-region subproblem for the active-set piecewise-linear model."""
    m = n_full // 2
    h0 = build_sym_h(q0)
    dx = 2.0 / n_full
    corrs = corr_raw(h0) * dx
    
    # Build LP: minimize t
    # subject to: g_k^T q - t <= corrs[k] - g_k^T q0  for each active k
    #             0 <= q[i] <= 1
    #             |q[i] - q0[i]| <= delta  (trust region)
    
    # Number of variables: m (for q) + 1 (for t) = m + 1
    # c vector: [0, ..., 0, 1] (minimize t)
    c_vec = np.zeros(m + 1)
    c_vec[m] = 1.0
    
    A_ub = []
    b_ub = []
    
    for k in active_k:
        g_k = grad_q_from_h(h0, k) * dx
        # g_k^T q - t <= corrs[k] - g_k^T q0
        # g_k^T q - t + g_k^T q0 <= corrs[k]
        # Actually: corrs[k] ≈ corrs[k] + g_k^T (q - q0)
        # We want: corrs[k] + g_k^T (q - q0) <= t
        # i.e.: g_k^T q - t <= corrs[k] - g_k^T q0
        row = np.zeros(m + 1)
        row[:m] = g_k
        row[m] = -1.0
        A_ub.append(row)
        b_ub.append(corrs[k] - g_k @ q0)
    
    # Trust region: q0[i] - delta <= q[i] <= q0[i] + delta
    # This becomes bounds on q variables
    bounds = []
    for i in range(m):
        lo = max(0.0, q0[i] - delta)
        hi = min(1.0, q0[i] + delta)
        bounds.append((lo, hi))
    # t is unbounded below (or bounded by -inf effectively, but let's say [0, 2])
    bounds.append((0.0, 2.0))
    
    if not A_ub:
        return q0.copy()
    
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    try:
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            q_new = res.x[:m]
            return np.clip(q_new, 0.0, 1.0)
    except Exception:
        pass
    return q0.copy()

    m = n_full // 2

    best_q = q.copy()
    delta = delta0

            continue

def smooth_max_obj_grad(x, n_full, beta):
    m = n_full // 2
    q = x[:m]
    h = build_sym_h(q)
    dx = 2.0 / n_full
    corrs = corr_raw(h) * dx
    max_c = corrs.max()
    shifted = corrs - max_c
    w = np.exp(np.clip(beta * shifted, -50, 50))
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(len(corrs)) / len(corrs)
    else:
        w /= w_sum
    val = max_c + np.log(w_sum) / beta
    grad_q = np.zeros(m)
    for k in range(2 * n_full - 1):
        if w[k] < 1e-12:
            continue
        g = grad_q_from_h(h, k) * dx
        grad_q += w[k] * g
    return val, grad_q

import numpy as np
from scipy.optimize import minimize, linprog
import time

_smooth_state = {}

def eval_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def corr_raw(h):
    return np.correlate(h, 1.0 - h, mode='full')

def build_sym_h(q):
    m = len(q)
    h = np.empty(2 * m)
    h[:m] = q
    h[m:] = 1.0 - q[::-1]
    return h

def sym_from_h(h):
    n = len(h)
    m = n // 2
    q = np.empty(m)
    for i in range(m):
        q[i] = 0.5 * (h[i] + (1.0 - h[n - 1 - i]))
    return np.clip(q, 0.0, 1.0)

def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.5, 1.5
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(x + mid, 0.0, 1.0).sum()
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    h = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        interior = (h > 1e-10) & (h < 1.0 - 1e-10)
        if interior.sum() > 0:
            h[interior] += diff / interior.sum()
    return np.clip(h, 0.0, 1.0)

def corr_at_k(h, k):
    n = len(h)
    d = k - (n - 1)
    i_lo = max(0, -d)
    i_hi = min(n - 1, n - 1 - d)
    if i_lo > i_hi:
        return 0.0
    return float(np.sum(h[i_lo:i_hi + 1] * (1.0 - h[i_lo + d:i_hi + 1 + d])))

def corr_grad_h(h, k):
    n = len(h)
    d = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -d)
    i_hi = min(n - 1, n - 1 - d)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx] += (1.0 - h[idx + d])
    grad[idx + d] += (-h[idx])
    return grad

def grad_q_from_h(h, k):
    n = len(h)
    m = n // 2
    g_h = corr_grad_h(h, k)
    g_q = np.zeros(m)
    for r in range(m):
        g_q[r] = g_h[r] - g_h[n - 1 - r]
    return g_q

def select_active(corrs_scaled, margin_frac=0.03, neighbor_pad=4, max_k=70):
    n = len(corrs_scaled)
    max_val = corrs_scaled.max()
    threshold = max_val - max(1e-8, margin_frac * max_val)
    active = set(np.where(corrs_scaled >= threshold)[0].tolist())
    padded = set()
    for k in active:
        for d in range(-neighbor_pad, neighbor_pad + 1):
            if 0 <= k + d < n:
                padded.add(k + d)
    active_list = sorted(padded)
    if len(active_list) > max_k:
        scored = sorted([(corrs_scaled[k], k) for k in active_list], reverse=True)
        active_list = sorted([k for _, k in scored[:max_k]])
    return active_list

    x_old = (np.arange(n_old) + 0.5) * 2.0 / n_old

def smooth_max_obj_grad(x, n_full, beta):
    m = n_full // 2
    q = x[:m]
    h = build_sym_h(q)
    dx = 2.0 / n_full
    corrs = corr_raw(h) * dx
    max_c = corrs.max()

    w = np.exp(np.clip(beta * shifted, -50, 50))

    if w_sum < 1e-30:

    else:
        w = w / w_sum
    val = max_c + np.log(w_sum) / beta

            continue
        g = grad_q_from_h(h, k) * dx
        grad_q += w[k] * g
    return val, grad_q

        _smooth_state['beta'] = beta
        bounds = [(0.0, 1.0)] * m

        sc = eval_score(build_sym_h(q_new))

            best_q = q_new.copy()
        q = q_new

def lp_solve(q0, n_full, active_k, delta):
    m = n_full // 2
    h0 = build_sym_h(q0)
    dx = 2.0 / n_full
    corrs = corr_raw(h0) * dx
    c_vec = np.zeros(m + 1)
    c_vec[m] = 1.0
    A_ub_rows = []
    b_ub_vals = []
    for k in active_k:
        g_k = grad_q_from_h(h0, k) * dx
        row = np.zeros(m + 1)
        row[:m] = g_k
        row[m] = -1.0
        A_ub_rows.append(row)
        b_ub_vals.append(corrs[k] - g_k @ q0)
    bounds = []
    for i in range(m):
        lo = max(0.0, q0[i] - delta)
        hi = min(1.0, q0[i] + delta)
        bounds.append((lo, hi))
    bounds.append((0.0, 2.0))
    if not A_ub_rows:
        return q0.copy()
    A_ub = np.array(A_ub_rows)
    b_ub = np.array(b_ub_vals)
    try:
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.clip(res.x[:m], 0.0, 1.0)
    except Exception:
        pass
    return q0.copy()

            q = q_new
            h = h_new

        else:
            delta *= shrink

                break

    best_score = eval_score(build_sym_h(q))
    best_q = q.copy()

            s2 = eval_score(build_sym_h(q2))

            else:

        q_new = best_q.copy()

            best_q = q_new.copy()

        m_r = n_r // 2
        q_r = sym_from_h(h_r)
        if rem() > 40:
            tl = time.time() + min(40, rem() * 0.6)
            q_o, c5_o, h_o = lp_trust_region_loop(q_r, n_r, max_rounds=50, delta0=0.5, time_limit=tl)
            if c5_o < best_c5:
                best_c5 = c5_o
                best_h = h_o.copy()
            q_use = q_o if c5_o <= 0.4 else q_r
        else:
            q_use = q_r
        if rem() > 25:
            tl2 = time.time() + min(25, rem() * 0.5)
            q_o2, c5_o2, h_o2 = coordinate_sweep(q_use, n_r, time_limit=tl2, max_evals=100)
            if c5_o2 < best_c5:
                best_c5 = c5_o2
                best_h = h_o2.copy()

    # Phase 7: Final polish
    if rem() > 25 and best_h is not None:
        n_fin = len(best_h)
        if n_fin % 2 == 0:
            q_fin = sym_from_h(best_h)
            if rem() > 35:
                tl = time.time() + min(25, rem() * 0.5)
                q_o, c5_o, h_o = lp_trust_region_loop(q_fin, n_fin, max_rounds=40, delta0=0.1, time_limit=tl)
                if c5_o < best_c5:
                    best_c5 = c5_o
                    best_h = h_o.copy()
            if rem() > 15:
                q_fin2 = sym_from_h(best_h)
                tl2 = time.time() + min(15, rem() * 0.5)
                q_o2, c5_o2, h_o2 = coordinate_sweep(q_fin2, n_fin, time_limit=tl2, max_evals=80)
                if c5_o2 < best_c5:
                    best_c5 = c5_o2
                    best_h = h_o2.copy()

    # Validate and return
    if best_h is None:
        m_fb = 63
        x_fb = np.linspace(0, 1, m_fb)
        q_fb = np.clip(0.5 + 0.3 * np.sin(2 * np.pi * x_fb), 0, 1)
        best_h = build_sym_h(q_fb)

    best_h = np.asarray(best_h, dtype=float).reshape(-1)
    n_out = len(best_h)
    best_h = project_feasible(best_h, n_out / 2.0)
    best_c5 = eval_score(best_h)
    return best_h, best_c5, n_out
