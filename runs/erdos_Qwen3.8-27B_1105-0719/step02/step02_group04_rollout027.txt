lo2 = max(0,-1) = 0, hi2 = min(3, 3-1) = 2.
g[0:3] += -h[0+1:3+1] = -h[1:4] = [-h[1], -h[2], -h[3]]

So previous: g = [-h[1], 1-h[0]-h[1], 1-h[1]-h[2], 1-h[2]]

Correct: g = [-h[1], 1-h[0]-h[2], 1-h[1], 0]

They're different! The previous code is wrong. This is a significant bug.

Now, the key question is: does the correct gradient help? The previous code was using L-BFGS-B on a smoothed objective with the wrong gradient, which means it was optimizing a different (incorrect) objective. Fixing the gradient should help.

Now let me design the SLP approach:

1. At each iteration, compute the exact C array.
2. Select active constraints (those near the maximum).
3. For each active constraint, compute the correct gradient row.
4. Solve an LP: minimize t subject to g_j^T * delta <= t - c_j for each active j, with box constraints on delta and a step-size limit.
5. Take a step of size lambda (line search) and evaluate exactly.
6. Accept if the exact objective improved.

For the symmetric builder, I need to map the gradient from h-space to q-space.

If h[0:m] = q and h[m:2m] = 1 - q[::-1], then:
- For i in [0, m): h[i] = q[i], so d(h[i])/d(q[k]) = δ(i,k)
- For i in [m, 2m): h[i] = 1 - q[2m-1-i], so d(h[i])/d(q[k]) = -δ(2m-1-i, k) = -δ(i, 2m-1-k)

So if g is the gradient in h-space, the gradient in q-space is:
gq[k] = g[k] - g[2m-1-k] for k in [0, m)

Wait: 
- g[k] contributes +1 to gq[k] (from h[k] = q[k])
- g[2m-1-k] contributes -1 to gq[k] (from h[2m-1-k] = 1-q[k])

So gq[k] = g[k] - g[2m-1-k]. ✓

Now let me write the full implementation. I'll use scipy.optimize.linprog for the LP step.

Actually, let me think about whether to use linprog or a simpler approach. The LP has:
- m variables for delta (q-space)
- 1 variable for t
- m active constraints: g_j(q)^T * delta <= t - c_j
- Box constraints on delta: |delta_i| <= R (trust region)
- Optionally: sum constraint (but for symmetric builder, sum is automatic)

So the LP has m+1 variables and m + 2m = 3m constraints (active + box). For m ~ 63, this is very small.

- The LP finds the best descent direction that simultaneously reduces all active constraints
- This is more principled for a min-max problem

import numpy as np
from scipy.optimize import linprog
import time

def eval_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def repair_box_sum(h, target_sum):
    """Water-filling projection to [0,1]^n with sum = target_sum."""
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -5.0, 5.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    h_new = np.clip(h - lam, 0.0, 1.0)
    diff = target_sum - h_new.sum()
    if abs(diff) > 1e-13:
        mask = (h_new > 1e-12) & (h_new < 1.0 - 1e-12)
        if mask.any():
            h_new[mask] += diff / mask.sum()
    return np.clip(h_new, 0.0, 1.0)

def build_sym(q, n):
    m = n // 2
    h = np.empty(n)
    h[:m] = q
    h[m:] = 1.0 - q[::-1]
    return h

def to_sym_q(h, n):
    m = n // 2
    return 0.5 * (h[:m] + 1.0 - h[2*m-1:m-1:-1])

def corr_grad_h(h, d, n):
    """Correct gradient of corr[p] wrt h, where d = p - (n-1)."""
    g = np.zeros(n)
    # First term: (1-h[j-d]) for j in [max(0,d), min(n-1, n-1-d)]
    lo1 = max(0, d)
    hi1 = min(n - 1, n - 1 - d)
    if lo1 <= hi1:
        g[lo1:hi1 + 1] += 1.0 - h[lo1 - d:hi1 + 1 - d]
    # Second term: -h[j+d] for j+d in V, i.e., j in [max(0-d,0), ...] 
    # j+d in [max(0,d), min(n-1, n-1-d)]
    # => j in [max(0,d)-d, min(n-1,n-1-d)-d] ∩ [0, n-1]
    lo2 = max(0, max(0, d) - d)
    hi2 = min(n - 1, min(n - 1, n - 1 - d) - d)
    if lo2 <= hi2:
        g[lo2:hi2 + 1] += -h[lo2 + d:hi2 + 1 + d]
    return g

def sym_grad_q(h, d, n):
    """Gradient of corr[p] wrt q variables, where h = build_sym(q, n)."""
    m = n // 2
    gh = corr_grad_h(h, d, n)
    gq = gh[:m] - gh[2*m-1:m-1:-1]
    return gq

def select_active(h, n, margin=0.01, pad=2):
    cv = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    mx = cv.max()
    thresh = mx - max(1e-8, margin * mx)
    idx = np.where(cv >= thresh)[0]
    expanded = set(idx)
    for a in idx:
        for delta in range(-pad, pad + 1):
            k = a + delta
            if 0 <= k <= 2 * n - 2:
                expanded.add(k)
    return np.sort(np.array(list(expanded)))

def slp_step_sym(q, n, active_k, R, move_pen=0.01, max_iter=50):
    """One SLP iteration in symmetric q-space."""
    m = n // 2
    h = build_sym(q, n)
    dx = 2.0 / n
    cv = np.correlate(h, 1.0 - h, mode='full') * dx
    c_active = cv[active_k]
    c_max = cv.max()
    
    # Build LP: min t + move_pen * sum(|delta_i|)
    # Variables: [delta_0..delta_{m-1}, s_0..s_{m-1}, t]
    # s_i >= 0, s_i >= delta_i, s_i >= -delta_i
    # Active constraints: g_j^T delta <= t - c_j  =>  g_j^T delta - t <= -c_j
    # Box: |delta_i| <= R
    
    dim = m
    n_vars = 3 * m + 1  # delta (m), slack (m), t (1)
    
    # Objective: [0..., move_pen..., move_pen..., 1]
    c_obj = np.zeros(n_vars)
    c_obj[dim:2*dim] = move_pen
    c_obj[2*dim:3*dim] = move_pen
    c_obj[3*dim] = 1.0  # t coefficient
    
    # Inequality constraints A_ub @ x <= b_ub
    A_rows = []
    b_rows = []
    
    # Active constraints: g_j^T delta - t <= -c_j
    for i in range(len(active_k)):
        k = active_k[i]
        d = k - (n - 1)
        gq = sym_grad_q(h, d, n) * dx  # scaled gradient
        row = np.zeros(n_vars)
        row[:m] = gq
        row[3*dim] = -1.0  # -t
        A_rows.append(row)
        b_rows.append(-c_active[i])
    
    # delta_i - s_i <= 0
    for i in range(m):
        row = np.zeros(n_vars)
        row[i] = 1.0
        row[dim + i] = -1.0
        A_rows.append(row)
        b_rows.append(0.0)
    
    # -delta_i - s_i <= 0
    for i in range(m):
        row = np.zeros(n_vars)
        row[i] = -1.0
        row[dim + i] = -1.0
        A_rows.append(row)
        b_rows.append(0.0)
    
    A_ub = np.array(A_rows)
    b_ub = np.array(b_rows)
    
    # Bounds
    bounds = []
    for i in range(m):
        bounds.append((max(-R, -q[i]), min(R, 1.0 - q[i])))
    for i in range(m):
        bounds.append((0.0, 4 * R))
    bounds.append((0.0, c_max + 1.0))
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    except Exception:
        return None, c_max, 0.0
    
    if res.status != 0:
        return None, c_max, 0.0
    
    delta = res.x[:m]
    t_val = res.x[3*dim]
    return delta, t_val, res.fun

    """SLP search in symmetric q-space."""
    m = n // 2
    h = build_sym(q, n)
    c_cur = eval_c5(h)
    best_q = q.copy()
    best_c = c_cur
    R = R0
    no_progress = 0
    move_pen = 0.01
    
    for it in range(max_iters):
        if time.time() - t0 > total_budget:
            break
        if len(active_k) == 0:
            active_k = select_active(h, n)
        
        result = slp_step_sym(q, n, active_k, R, move_pen)
        if result[0] is None:
            no_progress += 1
            R *= 0.7
            if no_progress > 5:
                break
            continue
        
        delta, t_pred, _ = result
        
        # Line search
        improved = False
        for lam in [1.0, 0.7, 0.5, 0.3, 0.2, 0.1]:
            q_new = np.clip(q + lam * delta, 0.0, 1.0)
