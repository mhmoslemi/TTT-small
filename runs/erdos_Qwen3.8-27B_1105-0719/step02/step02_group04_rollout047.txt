from scipy.optimize import linprog
import time

def eval_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def corr_scaled(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def grad_corr(h, j, n):
    """Gradient of c_j = sum h_i*(1-h_{i-d}) w.r.t. h, scaled by dx."""
    dx = 2.0 / n
    d = j - (n - 1)
    g = np.zeros(n)
    L = max(0, d)
    U = min(n - 1, n - 1 + d)
    if L <= U:
        g[L:U+1] += (1.0 - h[L-d:U+1-d])
    L2 = max(0, -d)
    U2 = min(n - 1, n - 1 - d)
    if L2 <= U2:
        g[L2:U2+1] += -h[L2+d:U2+1+d]
    return dx * g

def grad_corr_q(h, j, n):
    """Gradient of c_j w.r.t. q variables (anti-complementary builder)."""
    m = n // 2
    gh = grad_corr(h, j, n)
    return gh[:m] - gh[n-1:m-1:-1]

    h = np.empty(n)

    if max_count is not None and len(idx) > max_count:
        # Keep only the largest

        top_k = np.argsort(vals)[::-1][:max_count]
        idx = np.sort(idx[top_k])
    return idx

import numpy as np
from scipy.optimize import linprog
import time


def eval_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))


def corr_scaled(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)


def grad_corr(h, j, n):
    dx = 2.0 / n
    d = j - (n - 1)
    g = np.zeros(n)
    L = max(0, d)
    U = min(n - 1, n - 1 + d)
    if L <= U:
        g[L:U + 1] += (1.0 - h[L - d:U + 1 - d])
    L2 = max(0, -d)
    U2 = min(n - 1, n - 1 - d)
    if L2 <= U2:
        g[L2:U2 + 1] += -h[L2 + d:U2 + 1 + d]
    return dx * g


def grad_corr_q(h, j, n):
    m = n // 2
    gh = grad_corr(h, j, n)
    return gh[:m] - gh[n - 1:m - 1:-1]


def build_sym(q, n):
    m = n // 2
    h = np.empty(n)
    h[:m] = q
    h[m:] = 1.0 - q[::-1]
    return h


def q_from_h(h, n):
    m = n // 2
    return 0.5 * (h[:m] + 1.0 - h[n - 1:m - 1:-1])


def project_to_target(x, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(x - mid, 0.0, 1.0).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)


def select_active(C_arr, margin_frac=0.01, pad=2, max_count=None):
    mx = C_arr.max()
    thresh = mx - margin_frac * mx
    idx = np.where(C_arr >= thresh)[0]
    padded = set()
    for a in idx:
        for delta in range(-pad, pad + 1):
            k = int(a + delta)
            if 0 <= k < len(C_arr):
                padded.add(k)
    idx = np.sort(np.array(sorted(padded), dtype=int))
    if max_count is not None and len(idx) > max_count:
        vals = C_arr[idx]
        top_k = np.argsort(vals)[::-1][:max_count]
        idx = np.sort(idx[top_k])
    return idx


def slp_sym(q, n, R=0.05, max_iter=60, time_limit=None, t0=None, margin=0.015):
    m = n // 2
    cur_q = q.copy()
    cur_h = build_sym(cur_q, n)
    cur_c = eval_c5(cur_h)
    no_progress = 0

    for it in range(max_iter):
        if time_limit is not None and time.time() - t0 > time_limit:
            break

        C_arr = corr_scaled(cur_h)
        c_max = C_arr.max()
        active = select_active(C_arr, margin_frac=margin, pad=2, max_count=80)
        n_act = len(active)
        if n_act == 0:
            break

        A_ub = np.zeros((n_act, m + 1))
        b_ub = np.zeros(n_act)
        for idx, j in enumerate(active):
            a_j = grad_corr_q(cur_h, int(j), n)
            A_ub[idx, :m] = a_j
            A_ub[idx, m] = -1.0
            b_ub[idx] = -C_arr[int(j)]

        c_obj = np.zeros(m + 1)
        c_obj[m] = 1.0
        bounds = [(-R, R)] * m + [(0.0, 1.0)]

        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        except Exception:
            break

        if not res.success:
            break

        delta = res.x[:m]
        t_pred = res.x[m]

        if t_pred >= cur_c - 1e-10:
            no_progress += 1
            R = R * 0.6
            if R < 0.0005:
                break
            continue

        best_lambda = 0.0
        best_c = cur_c
        for lam in [1.0, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05, 0.02]:
            q_new = np.clip(cur_q + lam * delta, 0.0, 1.0)
            h_new = build_sym(q_new, n)
            c_new = eval_c5(h_new)
            if c_new < best_c - 1e-13:
                best_c = c_new
                best_lambda = lam

        if best_lambda > 0:
            cur_q = np.clip(cur_q + best_lambda * delta, 0.0, 1.0)
            cur_h = build_sym(cur_q, n)

            no_progress = 0

        else:
            no_progress += 1

def interp_to(h, n_new):
    n_old = len(h)
    if n_new == n_old:
        return h.copy()
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h)
    return project_to_target(h_new, n_new / 2.0)


def verify_gradient(h, n, j, tol=1e-6):
    g = grad_corr(h, j, n)
    eps = 1e-7
    idx_test = [0, n // 4, n // 2, 3 * n // 4, n - 1]
    for idx in idx_test:
        if idx < 0 or idx >= n:
            continue
        h_plus = h.copy()
        h_plus[idx] = min(1.0, h_plus[idx] + eps)
        h_minus = h.copy()
        h_minus[idx] = max(0.0, h_minus[idx] - eps)
        c_plus = corr_scaled(h_plus)[j]
        c_minus = corr_scaled(h_minus)[j]
        fd = (c_plus - c_minus) / (2.0 * eps)
        if abs(fd - g[idx]) > tol * max(1.0, abs(g[idx])):
            return False
    return True
