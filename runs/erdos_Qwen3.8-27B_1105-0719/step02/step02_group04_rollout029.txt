        return np.linalg.solve(AtA, Atb)
    except np.linalg.LinAlgError:
        return np.linalg.lstsq(AtA, Atb, rcond=None)[0]

def max_alpha_sym(q, delta):
    alpha = 1e6
    for i in range(len(q)):
        if delta[i] > 1e-14:
            a = (1.0 - q[i]) / delta[i]
            if a < alpha:
                alpha = a
        elif delta[i] < -1e-14:
            a = q[i] / (-delta[i])
            if a < alpha:
                alpha = a
    return min(alpha, 5.0)

def single_target_step_q(q, n, margin, eta, lam, n_steps=10):
    m = n // 2
    dx = 2.0 / n
    h = build_sym(q, n)
    c_arr = eval_c5_array(h)
    c_cur = float(c_arr.max())
    k_act = select_active(c_arr, margin)
    A_mat = np.zeros((len(k_act), m))
    for idx, j in enumerate(k_act):
        d = int(j) - (n - 1)
        gh = corr_grad_h_vec(h, d, n)
        A_mat[idx, :] = dx * qmap_grad(gh, n)
    z = c_cur - eta
    r = z - c_arr[k_act]
    delta = solve_ridge(A_mat, r, lam)
    if not np.all(np.isfinite(delta)):
        return None, None
    dm = np.max(np.abs(delta))
    if dm < 1e-15:
        return None, None
    alpha_max = max_alpha_sym(q, delta)
    if alpha_max < 1e-10:
        return None, None
    best_q = None
    best_c5 = c_cur
    for i in range(n_steps):
        alpha = alpha_max * (0.6 ** i)
        if alpha < 1e-7:
            break
        q_try = q + alpha * delta
        q_try = np.clip(q_try, 0.0, 1.0)
        h_try = build_sym(q_try, n)
        c5_try = eval_c5(h_try)
        if c5_try < best_c5 - 1e-13:
            best_c5 = c5_try
            best_q = q_try.copy()
    return best_q, best_c5

def single_target_step_full(h, margin, eta, lam, n_steps=10):
    n = len(h)
    dx = 2.0 / n
    c_arr = eval_c5_array(h)
    c_cur = float(c_arr.max())
    k_act = select_active(c_arr, margin)
    A_mat = np.zeros((len(k_act), n))
    for idx, j in enumerate(k_act):
        d = int(j) - (n - 1)
        A_mat[idx, :] = dx * corr_grad_h_vec(h, d, n)
    z = c_cur - eta
    r = z - c_arr[k_act]
    delta = solve_ridge(A_mat, r, lam)
    if not np.all(np.isfinite(delta)):
        return None, None
    delta -= np.mean(delta)
    dm = np.max(np.abs(delta))
    if dm < 1e-15:
        return None, None
    alpha_max = 1.0 / dm
    best_h = None
    best_c5 = c_cur
    for i in range(n_steps):
        alpha = alpha_max * (0.5 ** i)
        if alpha < 1e-8:
            break
        h_try = h + alpha * delta
        h_try = project_to_target(h_try, n / 2.0)
        c5_try = eval_c5(h_try)
        if c5_try < best_c5 - 1e-13:
            best_c5 = c5_try
            best_h = h_try.copy()
    return best_h, best_c5

def run_sym_loop(q, n, t0, deadline, reserve):
    best_q = q.copy()
    best_c5 = eval_c5(build_sym(q, n))
    margin = 0.004
    eta = 0.0015
    lam = 0.1
    fail = 0
    iter_count = 0
    while time.time() < deadline - reserve:
        q_new, c5_new = single_target_step_q(best_q, n, margin, eta, lam)
        iter_count += 1
        if c5_new is not None and c5_new < best_c5 - 1e-12:
            best_q = q_new
            best_c5 = c5_new
            fail = 0
            eta = min(eta * 1.05, 0.006)
        else:
            fail += 1
            if fail == 2:
                eta = max(eta * 0.6, 0.0003)
            elif fail == 4:
                margin = min(margin * 1.8, 0.025)
                lam = lam * 3.0 if lam < 8.0 else 0.05
            elif fail >= 7:
                margin = 0.004
                eta = 0.001
                lam = 0.1
                fail = 0
    return best_q, best_c5

def run_full_loop(h, t0, deadline, reserve):
    best_h = h.copy()
    best_c5 = eval_c5(h)
    margin = 0.004
    eta = 0.0015
    lam = 0.3
    fail = 0
    while time.time() < deadline - reserve:
        h_new, c5_new = single_target_step_full(best_h, margin, eta, lam)
        if c5_new is not None and c5_new < best_c5 - 1e-12:
            best_h = h_new
            best_c5 = c5_new
            fail = 0
        else:
            fail += 1
            if fail == 2:
                eta = max(eta * 0.6, 0.0003)
            elif fail == 4:
                margin = min(margin * 1.8, 0.025)
                lam = lam * 3.0 if lam < 8.0 else 0.1
            elif fail >= 7:
                margin = 0.004
                eta = 0.001
                lam = 0.3
                fail = 0
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    reserve = max(25.0, 0.04 * budget_s)
    best_h = None
    best_c5 = float('inf')
    try:
        h_init = np.array(initial_h_values, dtype=float).reshape(-1)
        if h_init.size > 0 and np.all(np.isfinite(h_init)):
            n_init = len(h_init)
            h_init = project_to_target(h_init, n_init / 2.0)
            best_h = h_init.copy()
            best_c5 = eval_c5(h_init)
    except Exception:
        pass
    if best_h is None:
        n_def = 120
        m_def = n_def // 2
        tt = np.linspace(0.0, 1.0, m_def)
        q_def = np.clip(0.5 + 0.22*np.sin(2*np.pi*3*tt + 0.5) + 0.12*np.sin(2*np.pi*5*tt + 1.3), 0.0, 1.0)
        best_h = build_sym(q_def, n_def)
        best_c5 = eval_c5(best_h)
    # Phase 1: Symmetric targeting on initial n
    n1 = len(best_h)
    if n1 % 2 == 0:
        q1 = to_sym_q(best_h, n1)
        h_sym_chk = build_sym(q1, n1)
        c5_chk = eval_c5(h_sym_chk)
        if c5_chk < best_c5:
            best_h = h_sym_chk
            best_c5 = c5_chk
        phase1_end = t0 + 0.38 * budget_s
        q1, c5_1 = run_sym_loop(q1, n1, t0, phase1_end, reserve)
        h_new1 = build_sym(q1, n1)
        if c5_1 < best_c5:
            best_h = h_new1
            best_c5 = c5_1
    # Phase 2: Multi-n exploration
    if time.time() < budget_s - reserve - 40:
        n_base = len(best_h)
        n_list = []
        for dlt in [-8, -6, -4, -2, 2, 4, 6, 8]:
            nc = n_base + dlt
            if nc >= 36 and nc % 2 == 0:
                n_list.append(nc)
        x_old = (np.arange(n_base) + 0.5) * (2.0 / n_base)
        for nc in n_list:
            if time.time() >= budget_s - reserve - 25:
                break
            x_new = (np.arange(nc) + 0.5) * (2.0 / nc)
            h_int = np.interp(x_new, x_old, best_h)
            h_int = project_to_target(h_int, nc / 2.0)
            q_int = to_sym_q(h_int, nc)
            h_sym = build_sym(q_int, nc)
            c5_int = eval_c5(h_sym)
            sb = min(20.0, max(5.0, budget_s - (time.time() - t0) - reserve - 15))
            q_res, c5_res = run_sym_loop(q_int, nc, t0, time.time() + sb, reserve + 5)
            h_res = build_sym(q_res, nc)
            if c5_res < best_c5:
                best_h = h_res
                best_c5 = c5_res
    # Phase 3: Full-variable targeting
    if time.time() < budget_s - reserve - 30:
        phase3_end = time.time() + min(0.15 * budget_s, budget_s - (time.time() - t0) - reserve - 20)
        h_f, c5_f = run_full_loop(best_h, t0, phase3_end, reserve)
        if c5_f < best_c5:
            best_h = h_f
            best_c5 = c5_f
    # Phase 4: Random restarts
    if time.time() < budget_s - reserve:
        n_r = len(best_h)
        if n_r % 2 != 0:
            n_r -= 1
        trials = int(max(3, min(25, (budget_s - (time.time() - t0) - reserve) / 12.0)))
        for tr in range(trials):
            if time.time() >= budget_s - reserve - 8:
                break
            m_r = n_r // 2
            tt = np.linspace(0, 1, m_r)
            ncomp = rng.randint(2, 5)
            q_r = np.zeros(m_r)
            for c in range(ncomp):
                amp = rng.uniform(0.05, 0.35) / (c + 1)
                freq = rng.choice([1, 2, 3, 4, 5, 6, 7])
                ph = rng.uniform(0, 2 * np.pi)
                q_r += amp * np.sin(2 * np.pi * freq * tt + ph)
            q_r = 0.5 + q_r
            q_r = np.clip(q_r, 0.0, 1.0)
            h_r = build_sym(q_r, n_r)
            c5_r = eval_c5(h_r)
            if c5_r < 0.385:
                sb = min(12.0, max(3.0, budget_s - (time.time() - t0) - reserve - 5))
                q_r2, c5_r2 = run_sym_loop(q_r, n_r, t0, time.time() + sb, reserve + 3)
                h_r2 = build_sym(q_r2, n_r)
                if c5_r2 < best_c5:
                    best_h = h_r2
                    best_c5 = c5_r2
    # Phase 5: Final tight polish
    if time.time() < budget_s - 12 and best_h is not None:
        n_fin = len(best_h)
        if n_fin % 2 == 0:
            q_fin = to_sym_q(best_h, n_fin)
            margin = 0.003
            eta = 0.0005
            lam = 0.05
            fail_ct = 0
            while time.time() < budget_s - 10:
                q_new, c5_new = single_target_step_q(q_fin, n_fin, margin, eta, lam, n_steps=14)
                if c5_new is not None and c5_new < best_c5 - 1e-13:
                    q_fin = q_new
                    best_h = build_sym(q_fin, n_fin)
                    best_c5 = c5_new
                    fail_ct = 0
                else:
                    fail_ct += 1
                    if fail_ct == 2:
                        eta = max(eta * 0.5, 0.0002)
                    elif fail_ct == 4:
                        margin = min(margin * 1.5, 0.015)
                    elif fail_ct >= 8:
                        break
    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_target(best_h, len(best_h) / 2.0)
    best_c5 = eval_c5(best_h)
    return best_h, best_c5, len(best_h)
