From the earlier analysis:

g_d(h) = dx * sum_{i=0}^{n-1-|d|} h[i+|d|](1-h[i])

∂g_d/∂h[j] = dx * [∂/∂h[j] sum_{i} h[i+|d|] - ∂/∂h[j] sum_{i} h[i+|d|]h[i]]

First term: ∂/∂h[j] sum_i h[i+|d|] = 1 if |d| ≤ j ≤ n-1, else 0. (Valid range: i = j-|d| must satisfy 0 ≤ i ≤ n-1-|d|, so |d| ≤ j ≤ n-1.)

Second term: ∂/∂h[j] sum_i h[i+|d|]h[i]:
- i = j: ∂/∂h[j] h[j+|d|]h[j] = h[j+|d|] (if j ≤ n-1-|d|)
- i+|d| = j, i.e., i = j-|d|: ∂/∂h[j] h[j]h[j-|d|] = h[j-|d|] (if |d| ≤ j ≤ n-1-|d|)

So:
∂g_d/∂h[j] = dx * [I(|d| ≤ j ≤ n-1) - I(j ≤ n-1-|d|)*h[j+|d|] - I(|d| ≤ j ≤ n-1-|d|)*h[j-|d|]]

For the symmetric parameterization, h[n-1-j] = 1-h[j], so:
∂g_d/∂q[j] = ∂g_d/∂h[j] - ∂g_d/∂h[n-1-j] (since h[n-1-j] = 1-q[j])

def eval_c5(h):

    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)


def corr_array(h):

    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx


def subgrad_sym(q, n):

    dx = 2.0 / n

def coord_descent_sym(q, n, max_rounds=50, deterministic=True):
    m = n // 2
    c5 = eval_c5(make_sym_h(q, n))
    for rnd in range(max_rounds):
        improved = False
        for j in range(m):
            cur = q[j]
            best = cur
            best_c5 = c5
            if deterministic:
                candidates = np.unique(np.clip(np.concatenate([
                    [cur],
                    np.linspace(0.0, 1.0, 13),
                    [max(0.0, cur - 0.02), max(0.0, cur - 0.005),
                     min(1.0, cur + 0.02), min(1.0, cur + 0.005)],
                    [0.0, 1.0]
                ]), 0.0, 1.0))
            else:
                candidates = np.unique(np.clip(np.concatenate([
                    [cur],
                    np.random.uniform(0.0, 1.0, 12),
                    [0.0, 1.0]
                ]), 0.0, 1.0))
            for val in candidates:
                if abs(val - cur) < 1e-12:
                    continue
                q[j] = val
                c5_t = eval_c5(make_sym_h(q, n))
                if c5_t < best_c5:
                    best_c5 = c5_t
                    best = val
            q[j] = best
            if best != cur:
                c5 = best_c5
                improved = True
        if not improved:
            break
    return q, c5


def subgrad_coord_descent(q, n, max_iters=200, lr_init=0.01):
    m = n // 2
    c5 = eval_c5(make_sym_h(q, n))
    lr = lr_init
    for it in range(max_iters):
        g, k_star = subgrad_sym(q, n)
        q_new = q - lr * g
        q_new = np.clip(q_new, 0.0, 1.0)
        c5_new = eval_c5(make_sym_h(q_new, n))
        if c5_new < c5:
            q = q_new
            c5 = c5_new
            lr = min(lr * 1.1, 0.1)
        else:
            lr *= 0.7
            if lr < 1e-8:
                break
    return q, c5


def smooth_sym_obj(x, n, beta):
    h = make_sym_h(x, n)
    corrs = corr_array(h)
    return float(logsumexp(beta * corrs) / beta)


def smooth_sym_grad(x, n, beta):
    m = n // 2
    dx = 2.0 / n
    h = make_sym_h(x, n)
    corrs = corr_array(h)
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    ws = w.sum()
    if ws < 1e-30:
        w = np.ones(len(corrs)) / len(corrs)
    else:
        w /= ws
    n_lags = len(corrs)
    grad_h = np.zeros(n)
    for k in range(n_lags):
        wk = beta * w[k]
        if wk < 1e-15:
            continue
        d = k - (n - 1)
        ad = abs(d)
        if ad == 0:
            grad_h += wk * dx * (1.0 - 2.0 * h)
        else:
            grad_h[ad:] += wk * dx
            grad_h[ad:] -= wk * dx * h[:n - ad]
            grad_h[:n - ad] -= wk * dx * h[ad:n]
    return grad_h[:m] - grad_h[n-1:n-1-m:-1]


def slsqp_refine(q, n, beta=500, maxiter=300):
    m = n // 2
    try:
        res = minimize(smooth_sym_obj, q, jac=lambda x: smooth_sym_grad(x, n, beta),
                       method='SLSQP', bounds=[(0.0, 1.0)] * m,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        h_new = make_sym_h(res.x, n)
        c5_new = eval_c5(h_new)
        return res.x, c5_new
    except Exception:
        return None, None


def epigraph_solve(q, n, active_lags, t_init, maxiter=400):
    m = n // 2
    dx = 2.0 / n
    L = len(active_lags)

    def epi_fun(x):
        return x[0]

    def epi_jac(x):
        g = np.zeros(m + 1)
        g[0] = 1.0
        return g

    def epi_con_fun(x):
        t = x[0]
        q_x = x[1:]
        h = make_sym_h(q_x, n)
        corrs = corr_array(h)
        return t - corrs[active_lags]

    def epi_con_jac(x):
        q_x = x[1:]
        h = make_sym_h(q_x, n)
        J = np.zeros((L, m + 1))
        J[:, 0] = 1.0
        for i, lag in enumerate(active_lags):
            d = lag - (n - 1)
            ad = abs(d)
            G = np.zeros(n)
            if ad == 0:
                G[:] = dx * (1.0 - 2.0 * h)
            else:
                G[ad:] += dx
                G[ad:] -= dx * h[:n - ad]
                G[:n - ad] -= dx * h[ad:n]
            J[i, 1:] = -(G[:m] - G[n-1:n-1-m:-1])
        return J

    x0 = np.concatenate([[t_init], q])
    cons = [{'type': 'ineq', 'fun': epi_con_fun, 'jac': epi_con_jac}]
    bounds = [(0.0, 2.0)] + [(0.0, 1.0)] * m
    try:
        res = minimize(epi_fun, x0, jac=epi_jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        h_new = make_sym_h(res.x[1:], n)
        c5_new = eval_c5(h_new)
        if abs(h_new.sum() - n / 2.0) > 1e-8:
            return None, None
        return res.x[1:], c5_new
    except Exception:
        return None, None


def interpolate_h(h_old, n_old, n_new):
    dx_old = 2.0 / n_old
    x_old = (np.arange(n_old) + 0.5) * dx_old
    dx_new = 2.0 / n_new
    x_new = (np.arange(n_new) + 0.5) * dx_new
    h_new = np.interp(x_new, x_old, h_old)
    return np.clip(h_new, 0.0, 1.0)


def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)


def full_space_refine(h0, n, max_rounds=20):

    target = n / 2.0
    h = h0.copy()
    c5 = eval_c5(h)

        for j in range(n):

                    q_best = q_best
                    h_best = make_sym_h(q_best, n)
                print(f"  SLSQP beta={beta}: C5={c5_best:.8f}")

    # Phase 4: Epigraph with active set
    if remaining() > 30:
        h_cur = make_sym_h(q_best, n)
        corrs = corr_array(h_cur)
        max_val = corrs.max()
        n_act = min(40, 2 * n - 1)
        active = sorted(np.argsort(corrs)[::-1][:n_act].tolist())
        q_new, c5_new = epigraph_solve(q_best, n, active, t_init=max_val * 1.001, maxiter=300)
        if q_new is not None and c5_new < c5_best - 1e-10:
            c5_best = c5_new
            q_best = q_new
            h_best = make_sym_h(q_best, n)
            print(f"  Epigraph: C5={c5_best:.8f}")
            q_best, c5_cd = coord_descent_sym(q_best, n, max_rounds=10, deterministic=True)
            if c5_cd < c5_best:
                c5_best = c5_cd
                q_best = q_best
                h_best = make_sym_h(q_best, n)

    # Phase 5: Multi-start with random perturbations
    n_starts = 60
    for trial in range(n_starts):
        if remaining() < 15:
            break
        scale = 0.02 + 0.3 * (trial / max(n_starts - 1, 1))
        q_pert = q_best + scale * rng.randn(m_best)
        q_pert = np.clip(q_pert, 0.0, 1.0)
        q_new, c5_new = coord_descent_sym(q_pert, n, max_rounds=15, deterministic=True)
        if c5_new < c5_best - 1e-10:
            # Polish
            q_pol, c5_pol = slsqp_refine(q_new, n, beta=500, maxiter=150)
            if q_pol is not None and c5_pol < c5_new:
                c5_new = c5_pol
                q_new = q_pol
            q_new, c5_new = coord_descent_sym(q_new, n, max_rounds=15, deterministic=True)
            if c5_new < c5_best - 1e-10:
                c5_best = c5_new
                q_best = q_new
                h_best = make_sym_h(q_best, n)
                m_best = n // 2
                print(f"  Start {trial}: C5={c5_best:.8f}")

    # Phase 6: Try different resolutions
    if remaining() > 60:
        for n_try in [n + 10, n - 10, n + 20, n - 20, 200, 250, 300, 160, 220]:
            if remaining() < 25:
                break
            if n_try < 40 or n_try % 2 != 0:
                continue
            if n_try == n_best:
                continue
            h_interp = interpolate_h(h_best, n_best, n_try)
            h_interp = project_feasible(h_interp, n_try / 2.0)
            q_try = make_q_from_h(h_interp)
            q_new, c5_new = coord_descent_sym(q_try, n_try, max_rounds=15, deterministic=True)
            if c5_new < c5_best - 1e-9:
                q_pol, c5_pol = slsqp_refine(q_new, n_try, beta=400, maxiter=150)
                if q_pol is not None and c5_pol < c5_new:
                    c5_new = c5_pol
                    q_new = q_pol
                q_new, c5_new = coord_descent_sym(q_new, n_try, max_rounds=20, deterministic=True)
                if c5_new < c5_best - 1e-9:
                    c5_best = c5_new
                    n_best = n_try
                    m_best = n_try // 2
                    q_best = q_new
                    h_best = make_sym_h(q_best, n_best)
                    print(f"  n={n_try}: C5={c5_best:.8f}")

    # Phase 7: Full-space coordinate descent (break symmetry)
    if remaining() > 30:
        h_full = h_best.copy()
        target = n_best / 2.0
        improved_full = 0
        for rnd in range(5):
            if remaining() < 10:
                break
            h_imp = h_full.copy()
            for j in range(n_best):
                cur = h_full[j]
                best_local = cur
                best_h = h_full.copy()
                best_c5 = eval_c5(h_full)
                for val in np.unique(np.clip(np.array([cur - 0.05, cur - 0.02, cur + 0.02, cur + 0.05, 0.0, 1.0]), 0.0, 1.0)):
                    if abs(val - cur) < 1e-12:
                        continue
                    h_full[j] = val
                    h_p = project_feasible(h_full, target)
                    c5_t = eval_c5(h_p)
                    if c5_t < best_c5:
                        best_c5 = c5_t
                        best_h = h_p.copy()
                        best_local = val
                h_full = best_h.copy()
            if eval_c5(h_full) < eval_c5(h_imp) - 1e-12:
                improved_full += 1
        if eval_c5(h_full) < c5_best - 1e-10:
            c5_best = eval_c5(h_full)
            h_best = h_full
            q_best = make_q_from_h(h_best)
            print(f"  Full-space: C5={c5_best:.8f}")

    # Phase 8: Final polish - coordinate descent + SLSQP
    if remaining() > 20:
        q_best, c5_cd = coord_descent_sym(q_best, n_best, max_rounds=20, deterministic=True)
        if c5_cd < c5_best:
            c5_best = c5_cd
            h_best = make_sym_h(q_best, n_best)
        if remaining() > 15:
            q_new, c5_new = slsqp_refine(q_best, n_best, beta=800, maxiter=200)
            if q_new is not None:
                q_pol, c5_pol = coord_descent_sym(q_new, n_best, max_rounds=15, deterministic=True)
                if c5_pol < c5_best:
                    c5_best = c5_pol
                    q_best = q_pol
                    h_best = make_sym_h(q_best, n_best)
                    print(f"  Final polish: C5={c5_best:.8f}")

    # Phase 9: Epigraph final
    if remaining() > 20:
        corrs = corr_array(make_sym_h(q_best, n_best))
        active = sorted(np.argsort(corrs)[::-1][:min(50, 2 * n_best - 1)].tolist())
        q_new, c5_new = epigraph_solve(q_best, n_best, active,
                                        t_init=corrs.max() * 1.0005, maxiter=400)
        if q_new is not None and c5_new < c5_best - 1e-10:
            c5_best = c5_new
            q_best = q_new
            q_best, c5_cd = coord_descent_sym(q_best, n_best, max_rounds=10, deterministic=True)
            if c5_cd < c5_best:
                c5_best = c5_cd
            h_best = make_sym_h(q_best, n_best)
            print(f"  Final epi: C5={c5_best:.8f}")

    # Final validation
    h_best = np.clip(h_best, 0.0, 1.0)
    if len(h_best) % 2 == 0:
        h_best = project_feasible(h_best, len(h_best) / 2.0)
    c5_final = eval_c5(h_best)
    if c5_final < c5_best:
        c5_best = c5_final
    print(f"  Final: n={len(h_best)}, C5={c5_best:.8f}")
    return h_best, c5_best, len(h_best)
