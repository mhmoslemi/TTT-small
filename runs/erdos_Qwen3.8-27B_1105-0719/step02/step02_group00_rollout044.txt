import numpy as np
import time
from scipy.optimize import minimize

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx] + 1e-15)
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx] + 1e-15)
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full")) * dx)

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def build_gradient_rows(h, active_indices):
    n = len(h)
    dx = 2.0 / n
    center = n - 1
    M = len(active_indices)
    G = np.zeros((M, n))
    for row, m in enumerate(active_indices):
        d = m - center
        if d == 0:
            G[row, :] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            G[row, 0:n-d] += -dx * h[d:n]
            G[row, d:n] += dx * (1.0 - h[0:n-d])
        else:
            e = -d
            G[row, 0:n-e] += dx * (1.0 - h[e:n])
            G[row, e:n] += -dx * h[0:n-e]
    return G

def symmetrize_reverse(h):
    return (h + h[::-1]) / 2.0

def symmetrize_half(h):
    n = len(h)
    half = n // 2
    if n % 2 == 0:
        return np.concatenate([h[:half], 1.0 - h[half-1::-1]])[:n]
    else:
        out = np.zeros(n)
        out[:half] = h[:half]
        out[half] = 0.5
        out[half+1:] = 1.0 - h[half-1::-1]
        return out

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    if new_n <= 0:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

def lp_trust_region(h, radius, beam_frac, max_active=30, rho=0.0):
    try:
        import cvxpy as cp
    except ImportError:
        return None, None
    n = len(h)
    dx = 2.0 / n
    profile = compute_profile(h)
    T = np.max(profile)
    spread = T - np.median(profile)
    beam = max(0.002, min(0.02, beam_frac * spread)) if spread > 1e-10 else 0.005
    thresh = T - beam
    active = np.where(profile >= thresh)[0]
    if len(active) == 0:
        active = np.array([np.argmax(profile)])
    if len(active) > max_active:
        top_idx = np.argsort(profile[active])[::-1][:max_active]
        active = active[top_idx]
    G = build_gradient_rows(h, active)
    p_active = profile[active]
    p_active = p_active - G @ h
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    l = np.maximum(0.0, h - radius)
    u = np.minimum(1.0, h + radius)
    constraints = [
        t_var >= p_active + G @ h_var,
        cp.sum(h_var) == n / 2.0,
        h_var >= l,
        h_var <= u
    ]
    if rho > 0:
        obj = cp.Minimize(t_var + rho * cp.sum(cp.abs(h_var - h)))
    else:
        obj = cp.Minimize(t_var)
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=cp.ECOS, max_iter=10000, abstol=1e-10, reltol=1e-10)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, eps=1e-10, max_iters=10000)
        except Exception:
            try:
                prob.solve(solver=cp.CLARABEL)
            except Exception:
                return None, None
    if prob.status not in ("optimal", "optimal_inaccurate"):
        return None, None
    h_sol = h_var.value
    if h_sol is None or not np.isfinite(h_sol).all():
        return None, None
    h_sol = np.clip(h_sol, 0.0, 1.0)
    h_sol = repair_feasible(h_sol)
    return h_sol, t_var.value if t_var.value is not None else None

def weighted_gradient_step(h, beam_frac=0.08, max_active=20, scales=None):
    n = len(h)
    profile = compute_profile(h)
    T = np.max(profile)
    spread = T - np.median(profile)
    beam = max(0.002, min(0.02, beam_frac * spread)) if spread > 1e-10 else 0.005
    thresh = T - beam
    active = np.where(profile >= thresh)[0]
    if len(active) == 0:
        active = np.array([np.argmax(profile)])
    if len(active) > max_active:
        top_idx = np.argsort(profile[active])[::-1][:max_active]
        active = active[top_idx]
    G = build_gradient_rows(h, active)
    w = profile[active] - thresh
    w = np.maximum(w, 0.01)
    w = w / w.sum()
    grad = G.T @ w
    grad -= grad.mean()
    gnorm = np.linalg.norm(grad)
    if gnorm < 1e-15:
        return None, None
    grad /= gnorm
    if scales is None:
        scales = [0.03, 0.02, 0.01, 0.006, 0.003, 0.001, 0.0005]
    best_h = None
    best_s = None
    for s in scales:
        h_try = repair_feasible(h - s * grad)
        sc = exact_score(h_try)
        if sc < best_s if best_s is not None else True:
            if best_s is None or sc < best_s:
                best_s = sc
                best_h = h_try
    return best_h, best_s

def momentum_descent(h, max_iters=100, momentum=0.4):
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    velocity = np.zeros(n)
    alpha = 0.015
    for it in range(max_iters):
        profile = compute_profile(h)
        c = np.max(profile)
        if c < best_c:
            best_c = c
        spread = c - np.median(profile)
        beam = max(0.002, min(0.015, 0.08 * spread)) if spread > 1e-10 else 0.005
        thresh = c - beam
        active = np.where(profile >= thresh)[0]
        if len(active) == 0:
            active = np.array([np.argmax(profile)])
        if len(active) > 15:
            top_idx = np.argsort(profile[active])[::-1][:15]
            active = active[top_idx]
        G = build_gradient_rows(h, active)
        w = profile[active] - thresh
        w = np.maximum(w, 0.01)
        w = w / w.sum()
        grad = G.T @ w
        grad -= grad.mean()
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad /= gnorm
        velocity = momentum * velocity + grad
        vnorm = np.linalg.norm(velocity)
        if vnorm > 0:
            velocity /= vnorm
        improved = False
        for a in [alpha, alpha * 0.5, alpha * 0.25, alpha * 0.1]:
            h_try = repair_feasible(h - a * velocity)
            s = exact_score(h_try)
            if s < best_c - 1e-13:
                h = h_try
                best_c = s
                improved = True
                break
        if not improved:
            alpha *= 0.7
            if alpha < 1e-7:
                break
        else:
            alpha = min(alpha * 1.05, 0.04)
    return h, best_c

                h_gw, c_gw = weighted_gradient_step(h_pat, beam_frac=0.1, max_active=15)
                if h_gw is not None and c_gw < c_pat - 1e-13:
                    h_pat = h_gw.copy()
                    c_pat = c_gw
                else:
                    break
            if time_left() > 30 and c_pat < best_c + 0.05:
                h_lp, _ = lp_trust_region(h_pat, radius=0.02, beam_frac=0.08, max_active=25, rho=0.0)
                if h_lp is not None:
                    c_lp = exact_score(h_lp)
                    if c_lp < c_pat:
                        h_pat = h_lp.copy()
                        c_pat = c_lp
            if c_pat < best_c - 1e-10:
                best_h = h_pat.copy()
                best_c = c_pat

    # Phase 8: Fine-tune with very small steps
    if time_left() > 50:
        for _ in range(100):
            if time_left() < 10:
                break
            h_gw, c_gw = weighted_gradient_step(best_h, beam_frac=0.04, max_active=12,
                                               scales=[0.005, 0.003, 0.002, 0.001, 0.0005, 0.0002])
            if h_gw is not None and c_gw < best_c - 1e-14:
                best_h = h_gw.copy()
                best_c = c_gw
            else:
                # Try LP with very small radius
                h_lp, _ = lp_trust_region(best_h, radius=0.005, beam_frac=0.04, max_active=15, rho=0.01)
                if h_lp is not None:
                    c_lp = exact_score(h_lp)
                    if c_lp < best_c - 1e-14:
                        best_h = h_lp.copy()
                        best_c = c_lp
                    else:
                        break
                else:
                    break

    # Phase 9: Final momentum polish
    if time_left() > 30:
        h_md, c_md = momentum_descent(best_h, max_iters=100, momentum=0.3)
        if c_md < best_c - 1e-12:
            best_h = h_md.copy()
            best_c = c_md

    # Phase 10: One more LP pass with tight radius
    if time_left() > 20:
        h_lp, _ = lp_trust_region(best_h, radius=0.008, beam_frac=0.06, max_active=30, rho=0.0)
        if h_lp is not None:
            c_lp = exact_score(h_lp)
            if c_lp < best_c - 1e-12:
                best_h = h_lp.copy()
                best_c = c_lp

    # Final validation
    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)

    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            final_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 2:
            final_c = float(eval_result[0])
    except Exception:
        pass

    return best_h, final_c, len(best_h)
