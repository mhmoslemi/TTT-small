The problem is to find a step function h: [0,2] → [0,1] that minimizes C₅ = max_k ∫ h(x)(1-h(x+k)) dx, subject to ∫₀² h(x) dx = 1.

The previous solution achieved C₅ = 0.380938, and we need to beat the record of 0.38085.

Key observations:
1. The current approach uses symmetric (antiparallel) construction h[i] = q[i], h[n-1-i] = 1-q[n-1-i]
2. It uses smooth max optimization, epigraph SLSQP, multi-resolution, and random restarts

Let me think about what could be different:

1. **Tangent LP approach**: Instead of SLSQP with nonlinear constraints, use linear programming with first-order Taylor approximations of the correlation lags. This gives a linear subproblem that can be solved efficiently and gives a coordinated descent direction.

2. **Pair-transfer line search**: Instead of only optimizing through the symmetric half-vector, use direct 2-point mass transfers that preserve the sum exactly. This can find improvements that the symmetric constraint prevents.

3. **Different starting points**: Try various structured initial conditions beyond just the parent.

4. **Coordinate descent on the full vector**: After finding a good symmetric solution, try breaking the symmetry with coordinate-level optimization.

Let me design the algorithm:

Phase 1: Start from parent (if available), project to feasible set.
Phase 2: Run tangent LP local optimization on the symmetric half-vector.
Phase 3: Run pair-transfer line searches on the full vector.
Phase 4: Try multi-resolution transfers and repeated optimization.
Phase 5: Random restarts with diverse initial conditions.

The key innovation from the previous code:

import numpy as np
from scipy.optimize import minimize, linprog
import time

def eval_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def corr_raw(h):
    return np.correlate(h, 1.0 - h, mode='full')

def build_sym_h(q):
    m = len(q)
    h = np.empty(2 * m)
    h[:m] = q
    h[m:] = 1.0 - q[::-1]
    return h

def sym_from_h(h):
    n = len(h)
    m = n // 2
    q = np.empty(m)
    for i in range(m):
        q[i] = 0.5 * (h[i] + (1.0 - h[n - 1 - i]))
    return np.clip(q, 0.0, 1.0)

def project_feasible(x, target_sum):
    n = len(x)
    x = np.clip(np.nan_to_num(x, nan=0.5), 0.0, 1.0)
    if abs(x.sum() - target_sum) < 1e-12:
        return x
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(x + mid, 0.0, 1.0).sum()
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    h = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        interior = (h > 1e-10) & (h < 1.0 - 1e-10)
        if interior.sum() > 0:
            h[interior] += diff / interior.sum()
    return np.clip(h, 0.0, 1.0)

def corr_at_k(h, k):
    n = len(h)
    d = k - (n - 1)
    i_lo = max(0, -d)
    i_hi = min(n - 1, n - 1 - d)
    if i_lo > i_hi:
        return 0.0
    return float(np.sum(h[i_lo:i_hi + 1] * (1.0 - h[i_lo + d:i_hi + 1 + d])))

def corr_grad_k(h, k):
    n = len(h)
    d = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -d)
    i_hi = min(n - 1, n - 1 - d)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx] += (1.0 - h[idx + d])
    grad[idx + d] -= h[idx]
    return grad

def select_active(corrs_scaled, margin_frac=0.03, margin_abs=2e-6, neighbor_pad=4, max_k=70):
    n = len(corrs_scaled)
    max_val = corrs_scaled.max()
    threshold = max_val - max(margin_abs, margin_frac * max_val)
    active = set(np.where(corrs_scaled >= threshold)[0].tolist())
    padded = set()
    for k in active:
        for d in range(-neighbor_pad, neighbor_pad + 1):
            if 0 <= k + d < n:
                padded.add(k + d)
    active_list = sorted(padded)
    if len(active_list) > max_k:
        scored = sorted([(corrs_scaled[k], k) for k in active_list], reverse=True)
        active_list = sorted([k for _, k in scored[:max_k]])
    return active_list

def transfer_to_n(h_old, n_new):
    n_old = len(h_old)
    x_old = (np.arange(n_old) + 0.5) * 2.0 / n_old
    x_new = (np.arange(n_new) + 0.5) * 2.0 / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new / 2.0)

def tangent_lp_sym(q, n_full, active_k, trust_radius=0.3, t_upper=None):
    """Solve tangent LP on symmetric half-vector."""
    m = n_full // 2
    h = build_sym_h(q)
    dx = 2.0 / n_full
    
    # Compute raw corrs and gradients for active lags
    corrs = np.zeros(len(active_k))
    grads = np.zeros((len(active_k), m))
    
    for ai, k in enumerate(active_k):
        corrs[ai] = corr_at_k(h, k)
        g_full = corr_grad_k(h, k)
        for r in range(m):
            grads[ai, r] = g_full[r] - g_full[n_full - 1 - r]
    
    if t_upper is None:
        t_upper = corrs.max() + 1.0
    
    # LP: minimize t
    # vars: [q_new (m), t] -> total m+1 vars
    c_obj = np.zeros(m + 1)
    c_obj[m] = 1.0  # minimize t
    
    # Inequality constraints: grad_ai @ q_new - t <= grad_ai @ q - corr_ai + (corrections)
    # Actually: m_ai(q_new) = corr_ai + grad_ai @ (q_new - q)
    # We want m_ai(q_new) <= t for all active ai
    # => grad_ai @ q_new - t <= grad_ai @ q - corr_ai
    
    A_ub = np.zeros((len(active_k), m + 1))
    b_ub = np.zeros(len(active_k))
    for ai in range(len(active_k)):
        A_ub[ai, :m] = grads[ai]
        A_ub[ai, m] = -1.0
        b_ub[ai] = np.dot(grads[ai], q) - corrs[ai]
    
    # Bounds: q in [max(0, q_i - rho), min(1, q_i + rho)], t in [0, t_upper]
    lb = np.maximum(0.0, q - trust_radius)
    ub = np.minimum(1.0, q + trust_radius)
    bounds = list(zip(lb, ub)) + [(0.0, t_upper)]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            q_new = res.x[:m]
            t_val = res.x[m]
            # Check if improvement (reduction in t relative to current max)
            return q_new, t_val
    except Exception:
        pass
    return q.copy(), corrs.max()

def tangent_lp_full(h, active_k, trust_radius=0.15, t_upper=None):
    """Solve tangent LP on full vector with mass constraint."""
    n = len(h)
    dx = 2.0 / n
    
    corrs = np.zeros(len(active_k))
    grads = np.zeros((len(active_k), n))
    
    for ai, k in enumerate(active_k):
        corrs[ai] = corr_at_k(h, k)
        grads[ai] = corr_grad_k(h, k)
    
    if t_upper is None:
        t_upper = corrs.max() + 1.0
    
    # variables: [h_new (n), t] -> total n+1
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    A_ub = np.zeros((len(active_k), n + 1))
    b_ub = np.zeros(len(active_k))
    for ai in range(len(active_k)):
        A_ub[ai, :n] = grads[ai]
        A_ub[ai, n] = -1.0
        b_ub[ai] = np.dot(grads[ai], h) - corrs[ai]
    
    lb = np.maximum(0.0, h - trust_radius)
    ub = np.minimum(1.0, h + trust_radius)
    bounds = list(zip(lb, ub)) + [(0.0, t_upper)]
    
    # Equality: sum(h_new) = n/2
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], res.x[n]
    except Exception:
        pass
    return h.copy(), corrs.max()

def pair_transfer_line_search(h, a, b, n_evals=80, refine_depth=3):
    """1D line search over delta for h[a] += delta, h[b] -= delta."""
    lower = max(-h[a], h[b] - 1.0)
    upper = min(1.0 - h[a], -h[b])
    if upper <= lower:
        return h, eval_score(h)
    
    best_delta = 0.0
    best_score = eval_score(h)
    
    # Grid search
    deltas = np.linspace(lower, upper, n_evals)
    scores = np.zeros(n_evals)
    for i, d in enumerate(deltas):
        h_trial = h.copy()
        h_trial[a] += d
        h_trial[b] -= d
        scores[i] = eval_score(h_trial)
    
    best_idx = np.argmin(scores)
    best_delta = deltas[best_idx]
    best_score = scores[best_idx]
    
    if best_score >= best_score - 1e-13:  # no improvement from grid
        return h, best_score
    
    # Refine with golden section
    for _ in range(refine_depth):
        # Narrow around best_idx
        lo = max(0, best_idx - 3)
        hi = min(n_evals - 1, best_idx + 3)
        sub_deltas = np.linspace(deltas[lo], deltas[hi], 40)
        sub_scores = np.zeros(len(sub_deltas))
        for i, d in enumerate(sub_deltas):
            h_trial = h.copy()
            h_trial[a] += d
            h_trial[b] -= d
            sub_scores[i] = eval_score(h_trial)
        best_sub = np.argmin(sub_scores)
        best_delta = sub_deltas[best_sub]
        best_score = sub_scores[best_sub]
    
    h_new = h.copy()
    h_new[a] += best_delta
    h_new[b] -= best_delta
    return h_new, best_score

def smooth_max_obj_grad_full(h, beta):
    """Smooth max of correlation lags with gradient."""
    n = len(h)
    dx = 2.0 / n
    corrs = corr_raw(h) * dx
    max_c = corrs.max()
    shifted = corrs - max_c
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(len(corrs)) / len(corrs)
    else:
        w /= w_sum
    val = max_c + np.log(w_sum) / beta
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        g = corr_grad_k(h, k)
        grad += w[k] * g
    grad *= dx
    return val, grad

def smooth_sym_optimize(q0, n_full, beta_list, maxiter_per_beta, time_limit):
    m = n_full // 2
    q = np.clip(q0.copy(), 0, 1)
    h = build_sym_h(q)
    best_score = eval_score(h)
    best_q = q.copy()

    for beta in beta_list:
        if time.time() > time_limit:
            break
        bounds = [(0.0, 1.0)] * m
        
        def obj_fn(xx):
            return smooth_max_obj_sym(xx, n_full, beta)
        
        def jac_fn(xx):
            return smooth_max_jac_sym(xx, n_full, beta)
        
        try:
            res = minimize(obj_fn, q, jac=jac_fn, method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxiter_per_beta, 'ftol': 1e-15, 'gtol': 1e-10})
            q_new = np.clip(res.x, 0, 1)
        except Exception:
            q_new = q.copy()
        h_new = build_sym_h(q_new)
        sc = eval_score(h_new)
        if sc < best_score:
            best_score = sc
            best_q = q_new.copy()
        q = q_new

    return best_q, best_score, build_sym_h(best_q)

import numpy as np
from scipy.optimize import minimize, linprog
import time

def eval_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def corr_raw(h):
    return np.correlate(h, 1.0 - h, mode='full')

def build_sym_h(q):
    m = len(q)
    h = np.empty(2 * m)
    h[:m] = q
    h[m:] = 1.0 - q[::-1]
    return h

def sym_from_h(h):
    n = len(h)
    m = n // 2
    q = np.empty(m)
    for i in range(m):
        q[i] = 0.5 * (h[i] + (1.0 - h[n - 1 - i]))
    return np.clip(q, 0.0, 1.0)

def project_feasible(x, target_sum):
    n = len(x)
    x = np.clip(np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0), 0.0, 1.0)
    if abs(x.sum() - target_sum) < 1e-12:
        return x.copy()
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(x + mid, 0.0, 1.0).sum()
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    h = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        interior = (h > 1e-10) & (h < 1.0 - 1e-10)
        if interior.sum() > 0:
            h[interior] += diff / interior.sum()
    return np.clip(h, 0.0, 1.0)

def corr_at_k(h, k):
    n = len(h)
    d = k - (n - 1)
    i_lo = max(0, -d)
    i_hi = min(n - 1, n - 1 - d)
    if i_lo > i_hi:
        return 0.0
    return float(np.sum(h[i_lo:i_hi + 1] * (1.0 - h[i_lo + d:i_hi + 1 + d])))

def corr_grad_k(h, k):
    n = len(h)
    d = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -d)
    i_hi = min(n - 1, n - 1 - d)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx] += (1.0 - h[idx + d])
    grad[idx + d] -= h[idx]
    return grad

def select_active(corrs_scaled, margin_frac=0.03, neighbor_pad=4, max_k=70):
    n = len(corrs_scaled)
    max_val = corrs_scaled.max()
    threshold = max_val - max(2e-6, margin_frac * max_val)
    active = set(np.where(corrs_scaled >= threshold)[0].tolist())
    padded = set()
    for k in active:
        for d in range(-neighbor_pad, neighbor_pad + 1):
            if 0 <= k + d < n:
                padded.add(k + d)
    active_list = sorted(padded)
    if len(active_list) > max_k:
        scored = sorted([(corrs_scaled[k], k) for k in active_list], reverse=True)
        active_list = sorted([k for _, k in scored[:max_k]])
    return active_list

def transfer_to_n(h_old, n_new):
    n_old = len(h_old)
    x_old = (np.arange(n_old) + 0.5) * 2.0 / n_old
    x_new = (np.arange(n_new) + 0.5) * 2.0 / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new / 2.0)

def tangent_lp_sym(q, n_full, rho, lag_list=None):
    """Solve tangent LP on symmetric half-vector. Returns (q_new, t_pred, success)."""
    m = n_full // 2
    h = build_sym_h(q)
    n = n_full

    if lag_list is None:
        corrs_s = corr_raw(h) * (2.0 / n)
        lag_list = select_active(corrs_s, margin_frac=0.05, neighbor_pad=3, max_k=80)

    if len(lag_list) == 0:
        return q.copy(), 0.0, False

    # Build constraint matrix
    corrs = np.zeros(len(lag_list))
    grads = np.zeros((len(lag_list), m))

    for ai, k in enumerate(lag_list):
        corrs[ai] = corr_at_k(h, k)
        g_full = corr_grad_k(h, k)
        for r in range(m):
            grads[ai, r] = g_full[r] - g_full[n - 1 - r]

    t_upper = corrs.max() + 1.0

    # LP: minimize t s.t. grads @ q_new - t <= grads @ q - corrs, box + trust
    c_obj = np.zeros(m + 1)
    c_obj[m] = 1.0

    A_ub = np.empty((len(lag_list), m + 1))
    b_ub = np.empty(len(lag_list))
    for ai in range(len(lag_list)):
        A_ub[ai, :m] = grads[ai]
        A_ub[ai, m] = -1.0
        b_ub[ai] = np.dot(grads[ai], q) - corrs[ai]

    lb = np.maximum(0.0, q - rho)
    ub = np.minimum(1.0, q + rho)
    bounds = list(zip(lb, ub)) + [(0.0, t_upper)]

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x[:m], float(res.x[m]), True
    except Exception:
        pass
    return q.copy(), corrs.max(), False

def sym_tangent_iterate(q, n_full, max_rounds=50, rho_init=0.3, time_limit=None):

        if rho < 0.02:
            rho = 0.02

        q_new, t_pred, success = tangent_lp_sym(best_q, n_full, rho)
        if not success or np.allclose(q_new, best_q, atol=1e-10):
            break

        # Line search along direction
        direction = q_new - best_q
        norm_d = np.linalg.norm(direction)
        if norm_d < 1e-12:
            break

        best_alpha = 0.0
        for alpha in [0.25, 0.5, 0.75, 1.0, 1.2, 1.5]:
            q_try = np.clip(best_q + alpha * direction, 0, 1)
            s = eval_score(build_sym_h(q_try))
            if s < best_c5 - 1e-14:
                best_alpha = alpha
                best_c5 = s
                best_q = q_try

            # Try smaller step

                    best_alpha = alpha
                    best_c5 = s
                    best_q = q_try
                    break

                break

                break

    return best_h, best_c5

def smooth_max_obj_sym(q, n_full, beta):
    h = build_sym_h(q)
    n = n_full
    dx = 2.0 / n
    corrs = corr_raw(h) * dx
    max_c = corrs.max()
    shifted = corrs - max_c
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(len(corrs)) / len(corrs)
    else:
        w /= w_sum
    return max_c + np.log(w_sum) / beta

def smooth_max_jac_sym(q, n_full, beta):
    m = n_full // 2
    h = build_sym_h(q)
    n = n_full
    dx = 2.0 / n
    corrs = corr_raw(h) * dx
    max_c = corrs.max()
    shifted = corrs - max_c
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(len(corrs)) / len(corrs)
    else:
        w /= w_sum
    grad_q = np.zeros(m)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        g = corr_grad_k(h, k)
        for r in range(m):
            grad_q[r] += w[k] * (g[r] - g[n - 1 - r])
    grad_q *= dx
    return grad_q

def smooth_sym_optimize(q0, n_full, beta_list, maxiter_per_beta, time_limit):
    m = n_full // 2
    q = np.clip(q0.copy(), 0, 1)
    best_score = eval_score(build_sym_h(q))
    best_q = q.copy()

    for beta in beta_list:
        if time.time() > time_limit:
            break
        bounds = [(0.0, 1.0)] * m
        try:
            res = minimize(lambda xx: smooth_max_obj_sym(xx, n_full, beta),
                          q, jac=lambda xx: smooth_max_jac_sym(xx, n_full, beta),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': maxiter_per_beta, 'ftol': 1e-15, 'gtol': 1e-10})
            q_new = np.clip(res.x, 0, 1)
        except Exception:
            q_new = q.copy()
        h_new = build_sym_h(q_new)
        sc = eval_score(h_new)
        if sc < best_score:
            best_score = sc
            best_q = q_new.copy()
        q = q_new

    return best_q, best_score, build_sym_h(best_q)

def pair_transfer_search(h, n_pairs_to_try=30, n_evals=50, time_limit=None):
    """Try gradient-informed pair transfers on full vector."""
    n = len(h)
    best_h = h.copy()
    best_c5 = eval_score(best_h)

    for _ in range(n_pairs_to_try):
        if time_limit is not None and time.time() > time_limit:
            break
        corrs = corr_raw(best_h) * (2.0 / n)
        max_k = int(np.argmax(corrs))
        d = max_k - (n - 1)
        i_lo = max(0, -d)
        i_hi = min(n - 1, n - 1 - d)
        if i_lo > i_hi:
            break
        idx = np.arange(i_lo, i_hi + 1)

        g = np.zeros(n)
        g[idx] += (1.0 - best_h[idx + d])
        g[idx + d] -= best_h[idx]

        pos_cands = np.where(g > 0.005)[0]
        neg_cands = np.where(g < -0.005)[0]

        if len(pos_cands) == 0 or len(neg_cands) == 0:
            a, b = tuple(np.sort(rng_global.choice(n, 2, replace=False)))
            a, b = int(a), int(b)
        else:
            a = int(pos_cands[np.argmax(g[pos_cands])])
            b = int(neg_cands[np.argmin(g[neg_cands])])
            if a == b:
                a2 = int(pos_cands[np.argmax(g[pos_cands])])
                b = int(neg_cands[np.argmin(g[neg_cands])])
                if a == b:
                    continue

        lower = max(-best_h[a], best_h[b] - 1.0)
        upper = min(1.0 - best_h[a], -best_h[b])
        if upper - lower < 1e-6:
            continue

        deltas = np.linspace(lower, upper, n_evals)
        best_delta = 0.0
        best_s = best_c5
        for dval in deltas:
            h_try = best_h.copy()
            h_try[a] += dval
            h_try[b] -= dval
            s = eval_score(h_try)
            if s < best_s:
                best_s = s
                best_delta = dval

        if best_s < best_c5 - 1e-13:
            best_h[a] += best_delta
            best_h[b] -= best_delta
            best_c5 = best_s

    return best_h, best_c5

rng_global = np.random.RandomState(0)

def make_structured_start(n, rng, variant=0):
    """Create diverse structured initial vectors."""
    m = n // 2
    if n % 2 == 0:
        if variant == 0:
            # Binary random
            h = np.zeros(n)
            h[rng.permutation(n)[:n // 2]] = 1.0
        elif variant == 1:
            # Sine wave
            x = np.linspace(0, 1, m)
            q = np.clip(0.5 + 0.35 * np.sin(2 * np.pi * x + rng.uniform(0, 2)), 0, 1)
            h = build_sym_h(q)
        elif variant == 2:
            # Multi-frequency
            x = np.linspace(0, 1, m)
            q = 0.5
            for _ in range(2):
                q += rng.uniform(0.1, 0.2) * np.sin(2 * np.pi * rng.uniform(1, 5) * x + rng.uniform(0, 6))
            h = build_sym_h(np.clip(q, 0, 1))
        elif variant == 3:
            # Block
            nblocks = rng.randint(3, 7)
            bsz = max(1, m // nblocks)
            q = np.zeros(m)
            for bi in range(m // bsz + 1):
                s = bi * bsz
                e = min(s + bsz, m)
                q[s:e] = rng.choice([0.0, 0.3, 0.5, 0.7, 1.0])
            h = build_sym_h(q)
        elif variant == 4:
            # Smooth random
            h = 0.5 + 0.25 * np.sin(2 * np.pi * rng.uniform(0.5, 3) * np.linspace(0, 1, n))
            h += 0.1 * rng.randn(n)
        else:
            h = 0.5 + 0.3 * rng.randn(n)
    else:
        h = 0.5 + 0.3 * rng.randn(n)

    return project_feasible(h, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    global rng_global
    t0 = time.time()
    rng = np.random.RandomState(seed)
    rng_global = np.random.RandomState(seed + 1)

    def rem():
        return budget_s - (time.time() - t0)

    best_h = None
    best_c5 = float('inf')

    try:
        h_parent = np.array(initial_h_values, dtype=float).copy().reshape(-1)
    except (NameError, AttributeError):
        h_parent = None

    if h_parent is not None and len(h_parent) > 10:
        h_parent = np.clip(h_parent, 0.0, 1.0)
        h_parent = project_feasible(h_parent, len(h_parent) / 2.0)
        best_h = h_parent.copy()
        best_c5 = eval_score(best_h)
    else:
        n_p = 126
        m_p = n_p // 2
        x = np.linspace(0, 1, m_p)
        q_p = np.clip(0.5 + 0.3 * np.sin(2 * np.pi * x), 0, 1)
        best_h = build_sym_h(q_p)
        best_c5 = eval_score(best_h)

    # Phase 1: Smooth max L-BFGS-B
    n_cur = len(best_h)
    if n_cur % 2 == 0 and rem() > 180:
        q_init = sym_from_h(best_h)
        tl = time.time() + min(100, rem() * 0.2)
        q_opt, c5_s, h_s = smooth_sym_optimize(q_init, n_cur, [60, 120, 250, 500], 200, tl)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()

    # Phase 2: Symmetric tangent LP
    if n_cur % 2 == 0 and rem() > 150:
        q_init = sym_from_h(best_h)
        tl = time.time() + min(150, rem() * 0.25)
        q_opt, c5_s = sym_tangent_iterate(q_init, n_cur, max_rounds=60, rho_init=0.3, time_limit=tl)
        h_s = build_sym_h(q_opt)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()

    # Phase 3: Full-vector tangent LP (break symmetry)
    if rem() > 120:
        tl = time.time() + min(100, rem() * 0.2)
        h_opt, c5_f = full_tangent_iterate(best_h, max_rounds=30, rho_init=0.12, time_limit=tl)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_opt

    # Phase 4: Pair transfers
    if rem() > 80:
        tl = time.time() + min(80, rem() * 0.15)
        h_opt, c5_p = pair_transfer_search(best_h, n_pairs_to_try=50, n_evals=50, time_limit=tl)
        if c5_p < best_c5:
            best_c5 = c5_p
            best_h = h_opt

    # Phase 5: Multi-resolution pipeline
    n_base = len(best_h)
    n_cands = []
    for delta in [-12, -8, -6, -4, -2, 2, 4, 6, 8, 12, 16, -16]:
        nc = n_base + delta
        if 40 <= nc <= 260 and nc % 2 == 0:
            n_cands.append(nc)
    if n_base // 2 >= 30:
        n_cands.append(n_base // 2)
    n_cands = sorted(set(n_cands))

    for n_try in n_cands:
        if rem() < 60:
            break

        h_try = transfer_to_n(best_h, n_try)
        q_try = sym_from_h(h_try)

        # Smooth optimize
        if rem() > 80:
            tl = time.time() + min(50, rem() * 0.3)
            q_o, c5_o, h_o = smooth_sym_optimize(q_try, n_try, [100, 300], 150, tl)
            if c5_o < best_c5:
                best_c5 = c5_o
                best_h = h_o.copy()
                q_try = sym_from_h(best_h)

        # Tangent LP
        if rem() > 50:
            tl = time.time() + min(40, rem() * 0.25)
            q_o2, c5_o2 = sym_tangent_iterate(q_try, n_try, max_rounds=35, rho_init=0.25, time_limit=tl)
            h_o2 = build_sym_h(q_o2)
            if c5_o2 < best_c5:
                best_c5 = c5_o2
                best_h = h_o2.copy()

        # Full tangent on transferred
        if rem() > 40:
            h_src = best_h if best_c5 < eval_score(h_try) else h_try
            h_src = transfer_to_n(h_src, n_try)
            tl = time.time() + min(30, rem() * 0.2)
            h_o3, c5_o3 = full_tangent_iterate(h_src, max_rounds=20, rho_init=0.1, time_limit=tl)
            if c5_o3 < best_c5:
                best_c5 = c5_o3
                best_h = h_o3

    # Phase 6: Diverse restarts
    n_restarts = min(12, max(3, int(rem() / 70)))
    for i in range(n_restarts):
        if rem() < 60:
            break

        n_r = n_base
        if i % 3 == 1:
            n_r = max(56, n_base - 8)
        elif i % 3 == 2:
            n_r = min(200, n_base + 8)
        if n_r % 2 != 0:
            n_r += 1
        n_r = max(48, min(220, n_r))

        variant = i % 6
        h_r = make_structured_start(n_r, rng, variant)

        q_r = sym_from_h(h_r)
        if rem() > 70:
            tl = time.time() + min(50, rem() * 0.4)
            q_o, c5_o, h_o = smooth_sym_optimize(q_r, n_r, [80, 200, 400], 150, tl)
            if c5_o < best_c5:
                best_c5 = c5_o
                best_h = h_o.copy()

        if rem() > 40:
            tl = time.time() + min(30, rem() * 0.3)
            q_cur = sym_from_h(best_h if best_c5 < eval_score(h_r) else h_r)
            n_cur2 = len(q_cur) * 2
            q_o2, c5_o2 = sym_tangent_iterate(q_cur, n_cur2, max_rounds=25, rho_init=0.2, time_limit=tl)
            h_o2 = build_sym_h(q_o2)
            if c5_o2 < best_c5:
                best_c5 = c5_o2
                best_h = h_o2

        # Full tangent on this restart
        if rem() > 30:
            h_src = best_h if best_c5 < eval_score(h_r) else h_r
            tl = time.time() + min(20, rem() * 0.2)
            h_o3, c5_o3 = full_tangent_iterate(h_src, max_rounds=15, rho_init=0.1, time_limit=tl)
            if c5_o3 < best_c5:
                best_c5 = c5_o3
                best_h = h_o3

    # Phase 7: Final combined polish
    if rem() > 30 and best_h is not None:
        n_fin = len(best_h)
        # Smooth
        if n_fin % 2 == 0 and rem() > 20:
            q_fin = sym_from_h(best_h)
            tl = time.time() + min(15, rem() * 0.3)
            q_o, c5_o, h_o = smooth_sym_optimize(q_fin, n_fin, [500, 1000], 200, tl)
            if c5_o < best_c5:
                best_c5 = c5_o
                best_h = h_o

        # Tangent
        if n_fin % 2 == 0 and rem() > 15:
            q_fin = sym_from_h(best_h)
            tl = time.time() + min(12, rem() * 0.3)
            q_o, c5_o = sym_tangent_iterate(q_fin, n_fin, max_rounds=20, rho_init=0.1, time_limit=tl)
            h_o = build_sym_h(q_o)
            if c5_o < best_c5:
                best_c5 = c5_o
                best_h = h_o

        # Full tangent
        if rem() > 10:
            tl = time.time() + min(10, rem() * 0.3)
            h_o, c5_o = full_tangent_iterate(best_h, max_rounds=10, rho_init=0.08, time_limit=tl)
            if c5_o < best_c5:
                best_c5 = c5_o
                best_h = h_o

    # Final validation
    if best_h is None:
        m_fb = 63
        x_fb = np.linspace(0, 1, m_fb)
        q_fb = np.clip(0.5 + 0.3 * np.sin(2 * np.pi * x_fb), 0, 1)
        best_h = build_sym_h(q_fb)

    best_h = np.asarray(best_h, dtype=float).reshape(-1)
    n_out = len(best_h)
    best_h = project_feasible(best_h, n_out / 2.0)
    best_c5 = eval_score(best_h)

    return best_h, best_c5, n_out
