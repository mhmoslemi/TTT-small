    """Feasibility repair: clip to [0,1] and adjust sum to n/2."""
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        # Need to add mass
        capacity = 1.0 - h
        order = np.argsort(-capacity, kind='mergesort')
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx])
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-14:
                break
    else:
        # Need to subtract mass
        order = np.argsort(h, kind='mergesort')
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            if rem > 0:

            remaining -= rem

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_gradient(h):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    G = np.zeros((M, n))
    G[n-1] = dx * (1.0 - 2.0 * h)
    for k in range(1, n):
        G[n-1+k, k:] += dx * (1.0 - h[:n-k])
        G[n-1+k, :n-k] += -dx * h[k:]
        G[n-1-k, :n-k] += dx * (1.0 - h[k:])
        G[n-1-k, k:] += -dx * h[:n-k]
    return G

def lp_descent_direction(h, G, active_rows):
    n = len(h)
    A = len(active_rows)
    if A == 0:
        return None, 0.0
    c = np.zeros(n + 1)
    c[n] = -1.0  # maximize t (minimize -t)
    A_ub = np.zeros((A + 2 * n, n + 1))
    b_ub = np.zeros(A + 2 * n)
    A_ub[:A, :n] = G[active_rows, :]
    A_ub[:A, n] = 1.0
    A_ub[A:A+n, :n] = -np.eye(n)
    b_ub[A:A+n] = h
    A_ub[A+n:A+2*n, :n] = np.eye(n)
    b_ub[A+n:A+2*n] = 1.0 - h
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(None, None)] * n + [(-1.0, 1.0)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x is not None:
            d = res.x[:n].copy()
            t_val = -res.x[n]
            d -= d.mean()
            if np.all(np.isfinite(d)) and np.linalg.norm(d) > 1e-12:
                return d, t_val
    except Exception:
        pass
    return None, 0.0

def line_search(h, d, best_c):
    n = len(h)
    max_step = 0.08
    pos_mask = d > 1e-10
    neg_mask = d < -1e-10
    if np.any(pos_mask):
        max_step = min(max_step, 0.95 * np.min((1.0 - h[pos_mask]) / d[pos_mask]))
    if np.any(neg_mask):
        max_step = min(max_step, 0.95 * np.min(-h[neg_mask] / d[neg_mask]))
    if max_step < 1e-10:
        return h, best_c, False
    best_h = h.copy()
    for frac in [1.0, 0.75, 0.55, 0.4, 0.3, 0.2, 0.12, 0.07, 0.04]:
        step = max_step * frac
        h_try = repair_h(h + step * d)
        c_try = score_exact(h_try)
        if c_try < best_c - 1e-14:
            best_h = h_try
            best_c = c_try
            break
    return best_h, best_c, best_c < score_exact(h) - 1e-14

def polish_lp(h, time_budget, seed=42):
    n = len(h)
    h = repair_h(h.copy())
    best_c = score_exact(h)
    t_end = time.time() + time_budget
    beam = 0.015
    stall = 0

    while time.time() < t_end - 1.0:
        if time.time() > t_end - 1.0:
            break
        profile = compute_profile(h)
        c_now = np.max(profile)
        if c_now < best_c - 1e-12:
            best_c = c_now
        thresh = c_now - beam
        active = np.where(profile >= thresh)[0]
        if len(active) == 0:
            active = np.array([np.argmax(profile)])
        if len(active) > 50:
            order = np.argsort(profile[active])[-50:]
            active = active[order]
        G = compute_gradient(h)
        d, t_val = lp_descent_direction(h, G, active)
        if d is not None and t_val > 1e-8:
            h_new, c_new, improved = line_search(h, d, best_c)
            if improved:
                h = h_new
                best_c = c_new
                stall = 0
                beam = max(0.003, beam * 0.92)
            else:
                stall += 1
                beam = max(0.002, beam * 0.9)
        else:
            stall += 1
            # Fallback: weighted gradient direction
            weights = np.maximum(profile[active] - thresh, 0.01)
            weights /= weights.sum()
            grad = G[active].T @ weights
            grad -= grad.mean()
            gnorm = np.linalg.norm(grad)
            if gnorm > 1e-12:
                d_fb = -grad / gnorm
                h_new, c_new, improved = line_search(h, d_fb, best_c)
                if improved:
                    h = h_new
                    best_c = c_new
                    stall = 0
                    beam = max(0.003, beam * 0.95)
        if stall > 8:
            break
    h = repair_h(h)
    best_c = score_exact(h)
    return h, best_c

def polish_slsqp(h_start, maxiter=300):
    n = len(h_start)
    dx = 2.0 / n
    M = 2 * n - 1
    c0 = score_exact(h_start)

    def obj(x):
        return x[-1]

    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g

    def eq_fun(x):
        return x[:-1].sum() - n / 2.0

    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g

    def ineq_fun(x):
        h = x[:-1]
        t = x[-1]
        profile = np.correlate(h, 1.0 - h, mode='full') * dx
        return t - profile

    def ineq_jac(x):
        h = x[:-1]
        G = compute_gradient(h)
        J = np.zeros((M, n + 1))
        J[:, :-1] = -G
        J[:, -1] = 1.0
        return J

    x0 = np.concatenate([h_start, [c0 + 1e-4]])
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 2.0)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = repair_h(res.x[:-1])
            c_new = score_exact(h_new)
            if c_new <= c0 + 1e-8:
                return h_new
    except Exception:
        pass
    return None

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    return repair_h(np.clip(h_new, 0.0, 1.0))

def make_binary_seed(n, rng):
    indices = rng.choice(n, size=n // 2, replace=False)
    h = np.zeros(n)
    h[indices] = 1.0
    return repair_h(h)

def make_symmetric_seed(n, rng):
    half = n // 2
    h_half = rng.uniform(0.2, 0.8, size=half)
    if n % 2 == 0:
        h = np.concatenate([h_half, h_half[::-1]])
    else:
        h = np.concatenate([h_half, [0.5], h_half[::-1]])
    return repair_h(h)

def make_block_seed(n, n_blocks, rng):
    h = np.zeros(n)
    block_size = n // n_blocks
    for b in range(n_blocks):
        start = b * block_size
        end = min((b + 1) * block_size, n)
        h[start:end] = rng.uniform(0.0, 1.0)
    return repair_h(h)

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()

    def time_left():
        return budget_s - (time.time() - t_start)

    rng = np.random.RandomState(seed)

    parent = None
    try:
        parent = np.array(initial_h_values, dtype=np.float64)
    except Exception:
        pass

    if parent is None or len(parent) < 10:
        parent = np.full(70, 0.5)
    parent = repair_h(np.clip(parent, 0.0, 1.0))
    c_parent = score_exact(parent)

    best_h = parent.copy()
    best_c = c_parent
    n0 = len(best_h)

    # Stage 1: LP polish on parent
    if time_left() > 30:
        alloc = min(120, time_left() * 0.3)
        h_new, c_new = polish_lp(best_h, alloc, seed=seed)
        if c_new < best_c - 1e-13:
            best_h, best_c = h_new, c_new

    # Stage 2: SLSQP polish
    if time_left() > 60:
        h_sol = polish_slsqp(best_h, maxiter=400)
        if h_sol is not None:
            c_sol = score_exact(h_sol)
            if c_sol < best_c - 1e-12:
                best_h, best_c = h_sol, c_sol
                # Follow up with LP
                if time_left() > 60:
                    h_new, c_new = polish_lp(best_h, min(60, time_left() * 0.2), seed=seed+1)
                    if c_new < best_c - 1e-13:
                        best_h, best_c = h_new, c_new

    # Stage 3: Symmetric variant
    h_sym = repair_h((best_h + best_h[::-1]) / 2.0)
    c_sym = score_exact(h_sym)
    if c_sym < best_c - 1e-12:
        best_h, best_c = h_sym, c_sym
        if time_left() > 40:
            h_new, c_new = polish_lp(best_h, min(40, time_left() * 0.2), seed=seed+2)
            if c_new < best_c - 1e-13:
                best_h, best_c = h_new, c_new

    # Stage 4: Dimension exploration
    if time_left() > 80:
        n_cur = len(best_h)
        dims = sorted(set([
            n_cur - 4, n_cur - 2, n_cur + 2, n_cur + 4,
            n_cur + 8, n_cur - 8, n_cur + 16, n_cur - 16
        ]))
        dims = [d for d in dims if 30 <= d <= 180 and d != n_cur]
        dims.sort(key=lambda d: abs(d - n_cur))

        for new_n in dims[:6]:
            if time_left() < 40:
                break
            h_interp = interpolate_to_n(best_h, new_n)
            c_interp = score_exact(h_interp)
            if time_left() > 30:
                h_new, c_new = polish_lp(h_interp, min(40, time_left() * 0.25), seed=seed + 10 + new_n)
                if c_new < best_c - 1e-13:
                    best_h, best_c = h_new, c_new
            elif c_interp < best_c - 1e-13:
                best_h, best_c = h_interp, c_interp

    # Stage 5: Binary seeds
    if time_left() > 60:
        n_try = max(40, min(len(best_h), 80))
        for i in range(4):
            if time_left() < 25:
                break
            h_bin = make_binary_seed(n_try, rng)
            c_bin = score_exact(h_bin)
            if time_left() > 30:
                h_new, c_new = polish_lp(h_bin, min(35, time_left() * 0.3), seed=seed + 100 + i)
                if c_new < best_c - 1e-13:
                    best_h, best_c = h_new, c_new

    # Stage 6: Random continuous seeds
    if time_left() > 50:
        n_try = len(best_h)
        for i in range(3):
            if time_left() < 25:
                break
            h_rand = repair_h(rng.uniform(0.1, 0.9, size=n_try))
            if time_left() > 30:
                h_new, c_new = polish_lp(h_rand, min(30, time_left() * 0.3), seed=seed + 200 + i)
                if c_new < best_c - 1e-13:
                    best_h, best_c = h_new, c_new

    # Stage 7: Symmetric random seeds
    if time_left() > 50:
        n_try = len(best_h)
        for i in range(3):
            if time_left() < 25:
                break
            h_rand = make_symmetric_seed(n_try, rng)
            if time_left() > 30:
                h_new, c_new = polish_lp(h_rand, min(30, time_left() * 0.3), seed=seed + 300 + i)
                if c_new < best_c - 1e-13:
                    best_h, best_c = h_new, c_new

    # Stage 8: Block seeds
    if time_left() > 50:
        n_try = len(best_h)
        for nb in [4, 6, 8]:
            if time_left() < 25:
                break
            h_blk = make_block_seed(n_try, nb, rng)
            if time_left() > 30:
                h_new, c_new = polish_lp(h_blk, min(25, time_left() * 0.3), seed=seed + 400 + nb)
                if c_new < best_c - 1e-13:
                    best_h, best_c = h_new, c_new

    # Stage 9: Final polish on global best
    if time_left() > 30:
        alloc = min(60, time_left() * 0.4)
        h_new, c_new = polish_lp(best_h, alloc, seed=seed + 500)
        if c_new < best_c - 1e-14:
            best_h, best_c = h_new, c_new

    # Stage 10: SLSQP final
    if time_left() > 40:
        h_sol = polish_slsqp(best_h, maxiter=500)
        if h_sol is not None:
            c_sol = score_exact(h_sol)
            if c_sol < best_c - 1e-13:
                best_h, best_c = h_sol, c_sol

    # Stage 11: One more LP polish
    if time_left() > 20:
        h_new, c_new = polish_lp(best_h, min(20, time_left() * 0.5), seed=seed + 600)
        if c_new < best_c - 1e-14:
            best_h, best_c = h_new, c_new

    # Final validation
    best_h = repair_h(best_h)
    final_c = score_exact(best_h)

    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            if float(eval_result) < final_c - 1e-9:
                final_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 2:
            if float(eval_result[0]) < final_c - 1e-9:
                final_c = float(eval_result[0])
    except Exception:
        pass

    return best_h, float(final_c), int(len(best_h))
