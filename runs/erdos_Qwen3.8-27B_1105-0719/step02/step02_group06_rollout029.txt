```python
import numpy as np
from scipy.optimize import minimize
import time

def eval_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def corr_raw(h):
    return np.correlate(h, 1.0 - h, mode='full')

def build_sym_h(q):
    m = len(q)
    h = np.empty(2 * m)
    h[:m] = q
    h[m:] = 1.0 - q[::-1]
    return h

def sym_from_h(h):
    n = len(h)
    m = n // 2
    q = 0.5 * (h[:m] + 1.0 - h[m:][::-1])
    return np.clip(q, 0.0, 1.0)

def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(x + mid, 0.0, 1.0).sum()
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    h = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        interior = (h > 1e-10) & (h < 1.0 - 1e-10)
        if interior.sum() > 0:
            h[interior] += diff / interior.sum()
    return np.clip(h, 0.0, 1.0)

def corr_at_k(h, k):
    """Compute correlation at full-correlation index k (shift d = k - (n-1))"""
    n = len(h)
    d = k - (n - 1)
    i_lo = max(0, -d)
    i_hi = min(n - 1, n - 1 - d)
    if i_lo > i_hi:
        return 0.0
    return float(np.sum(h[i_lo:i_hi+1] * (1.0 - h[i_lo+d:i_hi+1+d])))

def corr_grad_k(h, k):
    """Gradient of corr_at_k with respect to h"""
    n = len(h)
    d = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -d)
    i_hi = min(n - 1, n - 1 - d)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx] += (1.0 - h[idx + d])
    grad[idx + d] += (-h[idx])
    return grad
```

```python
def select_active(corrs, margin_frac=0.02, margin_abs=5e-5, neighbor_pad=3, max_k=100):

        q = q_new
        if it % 500 == 0:
            lr = lr_init * 0.7 ** (it // 500)
    return best_q, best_score

def rng_idx(it, n):
    return it % n

def transfer_to_n(h_old, n_new):
    n_old = len(h_old)
    x_old = (np.arange(n_old) + 0.5) * 2.0 / n_old
    x_new = (np.arange(n_new) + 0.5) * 2.0 / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new / 2.0)

def epig_full_con_fun(x, k, n):
    h = x[:n]
    t = x[n]
    dx = 2.0 / n
    return t - corr_at_k(h, k) * dx

def epig_full_con_jac(x, k, n):
    h = x[:n]
    dx = 2.0 / n
    d = k - (n - 1)
    i_lo = max(0, -d)
    i_hi = min(n - 1, n - 1 - d)
    gh = np.zeros(n)
    if i_lo <= i_hi:
        idx_arr = np.arange(i_lo, i_hi + 1)
        gh[idx_arr] += (1.0 - h[idx_arr + d])
        gh[idx_arr + d] -= h[idx_arr]
    j = np.zeros(n + 1)
    j[:n] = -dx * gh
    j[n] = 1.0
    return j

def epig_full_eq(x, n):
    return np.sum(x[:n]) - n / 2.0

def epig_full_eq_jac(x, n):
    j = np.zeros(n + 1)
    j[:n] = 1.0
    return j

def epig_full_obj(x):
    return x[-1]

def epig_full_obj_grad(x):
    g = np.zeros(len(x))
    g[-1] = 1.0
    return g

def solve_epigraph_full(h_center, active_k, maxiter=60, ftol=1e-12):
    n = len(h_center)
    dx = 2.0 / n
    corrs_c = corr_all(h_center)
    t0_val = corrs_c.max() + 1e-8

    x0 = np.zeros(n + 1)
    x0[:n] = h_center
    x0[n] = t0_val

    bounds = [(0.0, 1.0)] * (n + 1)
    constraints = []
    for k in active_k:
        constraints.append({
            'type': 'ineq',
            'fun': partial(epig_full_con_fun, k=k, n=n),
            'jac': partial(epig_full_con_jac, k=k, n=n)
        })
    constraints.append({
        'type': 'eq',
        'fun': partial(epig_full_eq, n=n),
        'jac': partial(epig_full_eq_jac, n=n)
    })

    try:
        res = minimize(epig_full_obj, x0, jac=epig_full_obj_grad,
                       method='SLSQP', bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': ftol})
        h_out = res.x[:n]
    except Exception:
        h_out = h_center.copy()
    h_out = project_feasible(h_out, n / 2.0)
    return h_out, eval_score(h_out)

def smooth_full_obj(h, beta):
    n = len(h)
    c = corr_all(h)
    mc = c.max()
    shifted = c - mc
    w = np.exp(beta * shifted)
    ws = w.sum()
    if ws < 1e-30:
        return mc
    return mc + np.log(ws) / beta

def smooth_full_jac(h, beta):
    n = len(h)
    c = corr_all(h)
    mc = c.max()
    shifted = c - mc
    w = np.exp(beta * shifted)
    ws = w.sum()
    if ws < 1e-30:
        w = np.ones(len(c)) / len(c)
    else:
        w /= ws
    dx = 2.0 / n
    G = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        d = k - (n - 1)
        i_lo = max(0, -d)
        i_hi = min(n - 1, n - 1 - d)
        if i_lo > i_hi:
            continue
        idx_arr = np.arange(i_lo, i_hi + 1)
        G[idx_arr] += w[k] * (1.0 - h[idx_arr + d])
        G[idx_arr + d] -= w[k] * h[idx_arr]
    return G * dx

def solve_smooth_full(h0, n, beta_list, maxiter_per_beta, time_limit):
    h = h0.copy()
    best_score = eval_score(h)
    best_h = h.copy()
    bounds = [(0.0, 1.0)] * n
    for beta in beta_list:
        if time.time() > time_limit:
            break
        try:
            res = minimize(
                smooth_full_obj, h,
                jac=smooth_full_jac,
                args=(beta,),
                method='L-BFGS-B',
                bounds=bounds,
                options={'maxiter': maxiter_per_beta, 'ftol': 1e-16, 'gtol': 1e-10, 'maxls': 40}
            )
            h_new = np.clip(res.x, 0, 1)
            h_new = project_feasible(h_new, n / 2.0)
        except Exception:
            h_new = h.copy()
        sc = eval_score(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    np_rng = np.random.RandomState(seed)

    def rem():
        return budget_s - (time.time() - t0)

    best_h = None
    best_c5 = float('inf')

    try:
        h_parent = np.array(initial_h_values, dtype=float).copy().reshape(-1)
    except (NameError, AttributeError):
        h_parent = None

    if h_parent is not None and len(h_parent) > 10:
        n_p = len(h_parent)
        h_parent = project_feasible(h_parent, n_p / 2.0)
        best_h = h_parent.copy()
        best_c5 = eval_score(h_parent)
    else:
        m_p = 63
        x = np.linspace(0, 1, m_p)
        q_p = np.clip(0.5 + 0.3 * np.sin(2 * np.pi * x), 0, 1)
        best_h = build_sym_h(q_p)
        best_c5 = eval_score(best_h)

    n_cur = len(best_h)
    m_cur = n_cur // 2

    # Phase 1: Smooth max L-BFGS-B on symmetric
    if n_cur % 2 == 0 and rem() > 150:
        q_init = sym_from_h(best_h)
        tl = time.time() + min(200, rem() * 0.3)
        q_opt, c5_s = solve_smooth_sym(q_init, n_cur, [60, 120, 250, 500, 1000], 300, tl)
        h_s = build_sym_h(q_opt)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s

    # Phase 2: Coordinate descent on symmetric
    if n_cur % 2 == 0 and rem() > 100:
        q_cd = sym_from_h(best_h)
        tl = time.time() + min(80, rem() * 0.2)
        q_cd = coord_descent_sym(q_cd, n_cur, n_sweeps=20, n_cands=50, time_limit=tl)
        h_cd = build_sym_h(q_cd)
        c5_cd = eval_score(h_cd)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd

    # Phase 3: Epigraph refinement (symmetric) with iterative active set
    if n_cur % 2 == 0 and rem() > 120:
        q_ep = sym_from_h(best_h)
        for rnd in range(30):
            if rem() < 30:
                break
            h_chk = build_sym_h(q_ep)
            corrs = corr_all(h_chk)
            active = select_active(corrs, margin_frac=0.015, margin_abs=3e-5, neighbor_pad=4, max_k=90)
            q_new, c5_new = solve_epigraph_sym(q_ep, n_cur, active, maxiter=80, ftol=1e-13)
            h_new = build_sym_h(q_new)
            c5_new = eval_score(h_new)
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                best_h = h_new
                q_ep = sym_from_h(h_new)
                continue
            # Check hidden violations
            corrs_new = corr_all(h_new)
            max_act = corrs_new[active].max() if len(active) > 0 else 0
            hidden = np.where(corrs_new > max_act + 1e-7)[0]
            if len(hidden) > 0:
                active2 = sorted(set(active + list(hidden)))
                if len(active2) > 120:
                    active2 = select_active(corrs_new, margin_frac=0.01, margin_abs=2e-5, neighbor_pad=5, max_k=120)
                q_ep = sym_from_h(h_new)
                q_new2, c5_new2 = solve_epigraph_sym(q_ep, n_cur, active2, maxiter=60, ftol=1e-12)
                h_new2 = build_sym_h(q_new2)
                c5_new2 = eval_score(h_new2)
                if c5_new2 < best_c5 - 1e-14:
                    best_c5 = c5_new2
                    best_h = h_new2
                    q_ep = sym_from_h(h_new2)
                continue
            break

    # Phase 4: Coordinate descent again (after epigraph)
    if n_cur % 2 == 0 and rem() > 60:
        q_cd2 = sym_from_h(best_h)
        tl = time.time() + min(50, rem() * 0.3)
        q_cd2 = coord_descent_sym(q_cd2, n_cur, n_sweeps=25, n_cands=60, time_limit=tl)
        h_cd2 = build_sym_h(q_cd2)
        c5_cd2 = eval_score(h_cd2)
        if c5_cd2 < best_c5:
            best_c5 = c5_cd2
            best_h = h_cd2

    # Phase 5: Nonsymmetric optimization
    if rem() > 100:
        n_ns = n_cur
        h_ns = best_h.copy()
        # Smooth max on full h
        tl = time.time() + min(80, rem() * 0.4)
        h_ns, c5_ns = solve_smooth_full(h_ns, n_ns, [80, 200, 500], 200, tl)
        if c5_ns < best_c5:
            best_c5 = c5_ns
            best_h = h_ns
        # Epigraph on full h
        if rem() > 60:
            h_chk = best_h
            corrs = corr_all(h_chk)
            active = select_active(corrs, margin_frac=0.02, margin_abs=5e-5, neighbor_pad=3, max_k=70)
            h_new, c5_new = solve_epigraph_full(h_chk, active, maxiter=70, ftol=1e-12)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new

    # Phase 6: Multi-resolution exploration
    n_base = len(best_h)
    n_candidates = []
    for delta in [-6, -4, -2, 0, 2, 4, 6, 8, 10]:
        nc = n_base + delta
        if 48 <= nc <= 200:
            n_candidates.append(nc)
    if n_base // 2 >= 32:
        n_candidates.append(n_base // 2)
    n_candidates = sorted(set(n_candidates))

    for n_try in n_candidates:
        if rem() < 40:
            break
        h_try = transfer_to_n(best_h, n_try)
        n_try = len(h_try)
        if n_try % 2 == 0:
            q_try = sym_from_h(h_try)
            if rem() > 70:
                tl = time.time() + min(50, rem() * 0.4)
                q_o, c5_o = solve_smooth_sym(q_try, n_try, [100, 300, 600], 200, tl)
                h_o = build_sym_h(q_o)
                if c5_o < best_c5:
                    best_c5 = c5_o
                    best_h = h_o
                q_try = sym_from_h(h_o)
            if rem() > 40:
                tl = time.time() + min(30, rem() * 0.5)
                q_try = coord_descent_sym(q_try, n_try, n_sweeps=15, n_cands=40, time_limit=tl)
                h_o = build_sym_h(q_try)
                c5_o = eval_score(h_o)
                if c5_o < best_c5:
                    best_c5 = c5_o
                    best_h = h_o
        else:
            if rem() > 60:
                tl = time.time() + min(40, rem() * 0.5)
                h_o, c5_o = solve_smooth_full(h_try, n_try, [100, 300], 150, tl)
                if c5_o < best_c5:
                    best_c5 = c5_o
                    best_h = h_o

    # Phase 7: Subgradient method for global-ish search
    if n_cur % 2 == 0 and rem() > 40:
        q_sg = sym_from_h(best_h)
        tl = time.time() + min(40, rem() * 0.5)
        q_sg, c5_sg = subgradient_sym(q_sg, n_cur, n_iter=5000, lr_init=0.005, momentum=0.95, time_limit=tl)
        h_sg = build_sym_h(q_sg)
        # Polish with coordinate descent
        tl2 = time.time() + min(20, max(5, rem() - 10))
        q_sg = coord_descent_sym(q_sg, n_cur, n_sweeps=10, n_cands=30, time_limit=tl2)
        h_sg = build_sym_h(q_sg)
        c5_sg = eval_score(h_sg)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg

    # Phase 8: Final polish
    if rem() > 20 and best_h is not None:
        n_fin = len(best_h)
        if n_fin % 2 == 0:
            q_fin = sym_from_h(best_h)
            if rem() > 30:
                tl = time.time() + min(15, rem() * 0.5)
                q_fin, c5_o = solve_smooth_sym(q_fin, n_fin, [800, 1500], 200, tl)
                h_o = build_sym_h(q_fin)
                c5_o = eval_score(h_o)
                if c5_o < best_c5:
                    best_c5 = c5_o
                    best_h = h_o
            if rem() > 15:
                q_fin = sym_from_h(best_h)
                tl = time.time() + min(10, rem() * 0.5)
                q_fin = coord_descent_sym(q_fin, n_fin, n_sweeps=10, n_cands=50, time_limit=tl)
                h_o = build_sym_h(q_fin)
                c5_o = eval_score(h_o)
                if c5_o < best_c5:
                    best_c5 = c5_o
                    best_h = h_o
            # Final epigraph
            if rem() > 10:
                q_fin = sym_from_h(best_h)
                h_chk = build_sym_h(q_fin)
                corrs = corr_all(h_chk)
                active = select_active(corrs, margin_frac=0.008, margin_abs=1e-5, neighbor_pad=4, max_k=80)
                q_o, c5_o = solve_epigraph_sym(q_fin, n_fin, active, maxiter=100, ftol=1e-14)
                h_o = build_sym_h(q_o)
                c5_o = eval_score(h_o)
                if c5_o < best_c5:
                    best_c5 = c5_o
                    best_h = h_o
        else:
            if rem() > 20:
                h_chk = best_h
                corrs = corr_all(h_chk)
                active = select_active(corrs, margin_frac=0.01, margin_abs=2e-5, neighbor_pad=4, max_k=60)
                h_o, c5_o = solve_epigraph_full(h_chk, active, maxiter=80, ftol=1e-13)
                if c5_o < best_c5:
                    best_c5 = c5_o
                    best_h = h_o

    # Validate
    if best_h is None:
        m_fb = 63
        x_fb = np.linspace(0, 1, m_fb)
        q_fb = np.clip(0.5 + 0.3 * np.sin(2 * np.pi * x_fb), 0, 1)
        best_h = build_sym_h(q_fb)

    best_h = np.asarray(best_h, dtype=float).reshape(-1)
    n_out = len(best_h)
    best_h = project_feasible(best_h, n_out / 2.0)
    best_c5 = eval_score(best_h)

    return best_h, best_c5, n_out
```
