from scipy.optimize import minimize
from scipy.special import logsumexp
import time


def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h


def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def c5_of_h(h, dx):
    return float(compute_corrs(h).max() * dx)


def grad_corr_k(h, k, n):
    offset = k - (n - 1)
    grad = np.zeros(n)
    if offset == 0:
        grad[:] = 1.0 - 2.0 * h
        return grad
    i_lo = max(0, -offset)
    i_hi = min(n - 1, n - 1 - offset)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx + offset] += (1.0 - h[idx])
    grad[idx] += (-h[idx + offset])
    return grad


def smooth_max_val(h, dx, beta):
    return float(logsumexp(beta * compute_corrs(h)) / beta * dx)


def smooth_max_grad(h, dx, beta):
    n = len(h)
    corrs = compute_corrs(h)
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones_like(corrs) / len(corrs)
    else:
        w /= w_sum
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        grad += w[k] * grad_corr_k(h, k, n)
    return grad * dx


def slsqp_opt(h0, dx, target_sum, beta, maxiter):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum,
             'jac': lambda x: np.ones(n)}]

    def obj(x):
        return smooth_max_val(x, dx, beta)

    def g(x):
        return smooth_max_grad(x, dx, beta)

    try:
        res = minimize(obj, h0, jac=g, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        h_out = project_feasible(res.x, target_sum)
    except Exception:
        h_out = project_feasible(h0, target_sum)
    return h_out, c5_of_h(h_out, dx)

def slsqp_opt(h0, dx, target_sum, beta, maxiter):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum,
             'jac': lambda x: np.ones(n)}]

    def obj(x):
        return smooth_max_val(x, dx, beta)

    def g(x):
        return smooth_max_grad(x, dx, beta)

    try:
        res = minimize(obj, h0, jac=g, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        h_out = project_feasible(res.x, target_sum)
    except Exception:

            h_cd2, c5_cd2 = cd_optimize(best_h, target_sum, max_sweeps=100, top_m=14)
            if c5_cd2 < best_c5:
                best_c5 = c5_cd2
                best_h = h_cd2

    # Phase 3: Subgradient momentum
    if remaining() > 100:
        h_sg, c5_sg = subgrad_momentum(best_h, dx, target_sum, 20000)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg
        h_cd3, c5_cd3 = cd_optimize(best_h, target_sum, max_sweeps=150, top_m=14)
        if c5_cd3 < best_c5:
            best_c5 = c5_cd3
            best_h = h_cd3

    # Phase 4: Multistart with perturbations
    for trial in range(30):
        if remaining() < 15:
            break
        mag = 0.01 + 0.06 * (trial / 30.0)
        h_pert = best_h + rng.normal(0, mag, len(best_h))
        h_pert = project_feasible(h_pert, target_sum)
        h_cd4, c5_cd4 = cd_optimize(h_pert, target_sum, max_sweeps=80, top_m=12)
        if c5_cd4 < best_c5:
            best_c5 = c5_cd4
            best_h = h_cd4
            if remaining() > 25:
                h_opt, c5_val = slsqp_opt(best_h, dx, target_sum, 1500, 400)
                if c5_val < best_c5:
                    best_c5 = c5_val
                    best_h = h_opt
                h_cd5, c5_cd5 = cd_optimize(best_h, target_sum, max_sweeps=80, top_m=14)
                if c5_cd5 < best_c5:
                    best_c5 = c5_cd5
                    best_h = h_cd5

    # Phase 5: Try different n values
    n_current = len(best_h)
    n_candidates = []
    for offset in range(-20, 30):
        n_try = n_current + offset
        if 20 <= n_try <= 300:
            n_candidates.append(n_try)
    for extra in [40, 50, 60, 70, 80, 90, 100, 110, 126, 130, 140, 150, 160, 180, 200]:
        if extra not in n_candidates:
            n_candidates.append(extra)
    n_candidates = sorted(set(n_candidates))

    for n_try in n_candidates:
        if remaining() < 40:
            break
        dx_try = 2.0 / n_try
        ts_try = n_try / 2.0
        x_old = np.arange(n_current) * (2.0 / n_current)
        x_new = np.arange(n_try) * dx_try
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, ts_try)

        h_opt, c5_val = cd_optimize(h_interp, ts_try, max_sweeps=60, top_m=12)
        if c5_val < best_c5:
            if remaining() > 30:
                h_opt2, c5_val2 = slsqp_opt(h_opt, dx_try, ts_try, 500, 400)
                if c5_val2 < c5_val:
                    h_opt, c5_val = h_opt2, c5_val2
            h_cd6, c5_cd6 = cd_optimize(h_opt, ts_try, max_sweeps=80, top_m=14)
            if c5_cd6 < c5_val:
                h_opt, c5_val = h_cd6, c5_cd6
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h_opt
                n_current = n_try

    # Phase 6: Structured starts at best n
    n_best = len(best_h)
    dx_best = 2.0 / n_best
    ts_best = n_best / 2.0
    for trial in range(20):
        if remaining() < 12:
            break
        h_struct = make_structured_start(n_best, ts_best, rng)
        h_opt, c5_val = cd_optimize(h_struct, ts_best, max_sweeps=60, top_m=12)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_opt
            if remaining() > 20:
                h_opt2, c5_val2 = slsqp_opt(best_h, dx_best, ts_best, 1000, 500)
                if c5_val2 < best_c5:
                    best_c5 = c5_val2
                    best_h = h_opt2

    # Phase 7: Final intensive polish
    n_final = len(best_h)
    dx_final = 2.0 / n_final
    ts_final = n_final / 2.0
    if remaining() > 30:
        h_cd7, c5_cd7 = cd_optimize(best_h, ts_final, max_sweeps=200, top_m=16)
        if c5_cd7 < best_c5:
            best_c5 = c5_cd7
            best_h = h_cd7
    if remaining() > 20:
        for beta in [3000, 8000, 15000]:
            if remaining() < 12:
                break
            h_opt, c5_val = slsqp_opt(best_h, dx_final, ts_final, beta, 600)
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h_opt
    if remaining() > 15:
        h_cd8, c5_cd8 = cd_optimize(best_h, ts_final, max_sweeps=200, top_m=16)
        if c5_cd8 < best_c5:
            best_c5 = c5_cd8
            best_h = h_cd8
    if remaining() > 10:
        h_sg2, c5_sg2 = subgrad_momentum(best_h, dx_final, ts_final, 10000)
        if c5_sg2 < best_c5:
            best_c5 = c5_sg2
            best_h = h_sg2
        h_cd9, c5_cd9 = cd_optimize(best_h, ts_final, max_sweeps=100, top_m=16)
        if c5_cd9 < best_c5:
            best_c5 = c5_cd9
            best_h = h_cd9

    return best_h, best_c5, len(best_h)
