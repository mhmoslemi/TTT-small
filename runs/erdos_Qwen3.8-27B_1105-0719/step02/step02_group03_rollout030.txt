def swap_delta_binary(h, i, j, n):
    ms = np.arange(-(n-1), n)
    delta = np.zeros(2*n-1)
    
    # Linear terms from position i (delta=-1):
    valid_i_neg = (i - ms >= 0) & (i - ms < n)
    idx_i_neg = i - ms[valid_i_neg]
    delta[valid_i_neg] += (h[idx_i_neg] - 1.0)
    
    valid_i_pos = (i + ms >= 0) & (i + ms < n)
    idx_i_pos = i + ms[valid_i_pos]
    delta[valid_i_pos] += h[idx_i_pos]
    
    # Linear terms from position j (delta=+1):
    valid_j_neg = (j - ms >= 0) & (j - ms < n)
    idx_j_neg = j - ms[valid_j_neg]
    delta[valid_j_neg] += (1.0 - h[idx_j_neg])
    
    valid_j_pos = (j + ms >= 0) & (j + ms < n)
    idx_j_pos = j + ms[valid_j_pos]
    delta[valid_j_pos] += -h[idx_j_pos]
    
    # Cross terms:
    L1 = j - i  # cross term at this lag
    L2 = i - j  # cross term at this lag (negative of L1)
    idx1 = L1 + (n-1)
    idx2 = L2 + (n-1)
    if 0 <= idx1 < 2*n-1:
        delta[idx1] += 1.0
    if 0 <= idx2 < 2*n-1:
        delta[idx2] += 1.0
    
    # L=0 correction:
    delta[n-1] -= 2.0
    
    return delta

- The effect depends on the local pattern around i and j.

import numpy as np
import time

def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def make_binary_from_continuous(h, target_sum):
    n = len(h)
    idx = np.argsort(-h)
    result = np.zeros(n)
    result[idx[:int(target_sum)]] = 1.0
    return result

def compute_c5(h, dx):
    corrs = np.correlate(h, 1.0 - h, mode='full')
    return corrs.max() * dx

def full_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def swap_delta_binary(h, i, j, n):
    ms = np.arange(-(n-1), n)
    delta = np.zeros(2*n-1)
    
    valid_i_neg = (i - ms >= 0) & (i - ms < n)
    if valid_i_neg.any():
        idx = i - ms[valid_i_neg]
        delta[valid_i_neg] += (h[idx] - 1.0)
    
    valid_i_pos = (i + ms >= 0) & (i + ms < n)
    if valid_i_pos.any():
        idx = i + ms[valid_i_pos]
        delta[valid_i_pos] += h[idx]
    
    valid_j_neg = (j - ms >= 0) & (j - ms < n)
    if valid_j_neg.any():
        idx = j - ms[valid_j_neg]
        delta[valid_j_neg] += (1.0 - h[idx])
    
    valid_j_pos = (j + ms >= 0) & (j + ms < n)
    if valid_j_pos.any():
        idx = j + ms[valid_j_pos]
        delta[valid_j_pos] += -h[idx]
    
    # Cross terms
    L1 = j - i
    idx1 = L1 + (n-1)
    if 0 <= idx1 < 2*n-1:
        delta[idx1] += 1.0
    L2 = i - j
    idx2 = L2 + (n-1)
    if 0 <= idx2 < 2*n-1:
        delta[idx2] += 1.0
    
    # L=0 correction
    delta[n-1] -= 2.0
    
    return delta

def binary_steepest_descent(h0, dx, max_sweeps=100, t0=None, budget_s=None):
    n = len(h0)
    h = h0.copy()
    corrs = full_corrs(h)
    current_max = corrs.max()
    
    for sweep in range(max_sweeps):
        if t0 is not None and time.time() - t0 > budget_s - 10:
            break
        
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        best_improvement = 1e-10
        best_swap = None
        best_new_corrs = None
        
        for i in ones:
            for j in zeros:
                delta = swap_delta_binary(h, i, j, n)
                new_corrs = corrs + delta
                new_max = new_corrs.max()
                improvement = current_max - new_max
                if improvement > best_improvement:
                    best_improvement = improvement
                    best_swap = (i, j)
                    best_new_corrs = new_corrs
        
        if best_swap is None:
            break
        
        i, j = best_swap
        h[i], h[j] = 0.0, 1.0
        corrs = best_new_corrs
        current_max = corrs.max()
    
    c5 = current_max * dx
    return h, c5

def make_good_binary(n, rng, mode=0):
    """Create a good initial binary sequence."""
    h = np.zeros(n)
    half = n // 2
    if mode == 0:
        # Golden ratio distribution
        phi = (1.0 + np.sqrt(5.0)) / 2.0
        for k in range(half):
            pos = int((k * phi * n) % n)
            h[pos] = 1.0
        # Fix if we have duplicates
        # Actually with the modular arithmetic, we might get duplicates
        # Let's use a different approach
        h = np.zeros(n)
        positions = np.zeros(half, dtype=int)
        for k in range(half):
            positions[k] = int((k * phi * n) % n)
        # Remove duplicates
        unique_positions = np.unique(positions)
        if len(unique_positions) < half:
            # Fill remaining randomly
            used = set(unique_positions.tolist())
            remaining = [p for p in range(n) if p not in used]
            fill = rng.choice(remaining, half - len(unique_positions), replace=False)
            all_positions = np.concatenate([unique_positions, fill])
        else:
            all_positions = positions
        h[all_positions] = 1.0
    elif mode == 1:
        # Alternating with perturbation
        for k in range(half):
            pos = (2*k) % n
            h[pos] = 1.0
        # Randomly swap a few
        for _ in range(n//10):
            i = rng.randint(n)
            j = rng.randint(n)
            if h[i] == 1 and h[j] == 0:
                h[i], h[j] = h[j], h[i]
    elif mode == 2:
        # Random
        idx = rng.permutation(n)
        h[idx[:half]] = 1.0
    return h
