  - Find optimal value of q[r] in [0, 1] that minimizes max_k corr_k
  - This is a 1D minimax problem: minimize max_k f_k(q[r]) where each f_k is linear in q[r]
  - The optimum is where two (or more) constraints cross
  - Can be solved analytically or by bisection + checking

Actually for a 1D minimax of linear functions, the optimum is at a point where at least two functions achieve the maximum simultaneously. So:
- Compute all f_k(v) = corr_k at q with q[r] = v, for v in [0,1]
- Each f_k is linear in v: f_k(v) = a_k * v + b_k
- The minimax is: min_v max_k (a_k * v + b_k)
- This is a convex piecewise-linear function of v
- The minimum is where the upper envelope has a "valley"

I can solve this by:
1. For each k, compute the value at v=0 and v=1
2. The upper envelope at v=0 is max_k b_k
3. The upper envelope at v=1 is max_k (a_k + b_k)
4. Use binary search: find v where the number of "active" (top) constraints changes

Actually simpler: just do a golden section search or ternary search on the 1D function max_k f_k(v). Since it's convex, this works.

- For each pair (i, j), find the intersection v* of lines i and j
- Check if at v*, lines i and j are both at or above all other lines
- Among valid intersections, pick the one with lowest max value
- Also check endpoints v=0 and v=1

    m = n // 2
    dx = 2.0 / n

    # The second term (1-h[m]) is affected when m=j, contributing -h[j+k] if 0<=j+k<n

    j1_hi = min(n - 1, k + n - 1)

    # Term 2: -h[j+k] for j in [-k, n-1-k] ∩ [0, n-1]

    return grad_q

    active = set(np.where(corrs >= threshold)[0].tolist())
    # Add neighbors

    active = padded
    # If too many, keep top

        active = set(k for _, k in scored[:max_lags])
    return sorted(active)

def solve_lp(q, active_lags, radius):
    """Solve the LP for one trust region step."""
    m = len(q)
    n = 2 * m
    dx = 2.0 / n
    h = build_sym_h(q)
    corrs = corr_raw(h) * dx
    
    # Build LP
    c_obj = np.zeros(m + 1)
    c_obj[m] = 1.0
    
    A_rows = []
    b_rows = []
    for k in active_lags:
        g = compute_grad_q(h, k)
        row = np.zeros(m + 1)
        row[:m] = g
        row[m] = -1.0
        rhs = np.dot(g, q) - corrs[k]
        A_rows.append(row)
        b_rows.append(rhs)
    
    A_ub = np.array(A_rows)
    b_ub = np.array(b_rows)
    
    bounds = []
    for r in range(m):
        lo_r = max(0.0, q[r] - radius)
        hi_r = min(1.0, q[r] + radius)
        bounds.append((lo_r, hi_r))
    bounds.append((-1.0, 1.0))  # t bounds
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x[:m], res.x[m]
    except Exception:
        pass
    return q.copy(), corrs.max()

def solve_1d_minimax(a, b, v_lo=0.0, v_hi=1.0):
    """Find v in [v_lo, v_hi] that minimizes max_k (a[k]*v + b[k])."""
    # Binary search on the derivative
    # f(v) = max_k (a[k]*v + b[k])
    # f'(v) = a[argmax_k (a[k]*v + b[k])]  (when unique)
    
    lo, hi = v_lo, v_hi
    for _ in range(80):
        mid = (lo + hi) / 2.0
        vals = a * mid + b
        k_star = np.argmax(vals)
        slope = a[k_star]
        if slope < -1e-15:
            # f is decreasing at mid, optimum is to the right
            lo = mid
        elif slope > 1e-15:
            # f is increasing at mid, optimum is to the left
            hi = mid
        else:
            break
    v_opt = (lo + hi) / 2.0
    return v_opt, float(np.max(a * v_opt + b))

import numpy as np
from scipy.optimize import linprog, minimize
import time

def eval_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def corr_raw(h):
    return np.correlate(h, 1.0 - h, mode='full')

def build_sym_h(q):
    m = len(q)
    h = np.empty(2 * m)
    h[:m] = q
    h[m:] = 1.0 - q[::-1]
    return h

def sym_from_h(h):
    n = len(h)
    m = n // 2
    q = 0.5 * (h[:m] + (1.0 - h[m:][::-1]))
    return np.clip(q, 0.0, 1.0)

def project_feasible(x, target_sum):
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.clip(x + mid, 0.0, 1.0).sum()
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    h = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-13:
        interior = (h > 1e-10) & (h < 1.0 - 1e-10)
        if interior.sum() > 0:
            h[interior] += diff / interior.sum()
    return np.clip(h, 0.0, 1.0)

def grad_q_at_k(h, k):
    n = len(h)
    m = n // 2
    dx = 2.0 / n
    d = k - (n - 1)
    grad_h = np.zeros(n)
    j1_lo = max(0, d)
    j1_hi = min(n - 1, n - 1 + d)
    if j1_lo <= j1_hi:
        indices = np.arange(j1_lo, j1_hi + 1)
        grad_h[indices] += (1.0 - h[indices - d])
    j2_lo = max(0, -d)
    j2_hi = min(n - 1, n - 1 - d)
    if j2_lo <= j2_hi:
        indices = np.arange(j2_lo, j2_hi + 1)
        grad_h[indices] -= h[indices + d]
    return dx * (grad_h[:m] - grad_h[n - 1:n - 1 - m:-1])

def select_active(corrs, max_k=30, margin=0.004, pad=3):
    n = len(corrs)
    M = corrs.max()
    thresh = M - margin
    active = set(np.where(corrs >= thresh)[0].tolist())
    padded = set()
    for k in active:
        for dd in range(-pad, pad + 1):
            if 0 <= k + dd < n:
                padded.add(k + dd)
    lst = sorted(padded)
    if len(lst) > max_k:
        scored = sorted(((corrs[k], k) for k in lst), reverse=True)
        lst = sorted(k for _, k in scored[:max_k])
    return lst

def solve_lp_step(q, active, radius):
    m = len(q)
    n = 2 * m
    dx = 2.0 / n
    h = build_sym_h(q)
    corrs = corr_raw(h) * dx
    c_obj = np.zeros(m + 1)
    c_obj[m] = 1.0
    A_rows = []
    b_rows = []
    for k in active:
        g = grad_q_at_k(h, k)
        row = np.zeros(m + 1)
        row[:m] = g
        row[m] = -1.0
        rhs = float(np.dot(g, q)) - corrs[k]
        A_rows.append(row)
        b_rows.append(rhs)
    A_ub = np.array(A_rows)
    b_ub = np.array(b_rows)
    bounds = []
    for r in range(m):
        lo_r = max(0.0, q[r] - radius)
        hi_r = min(1.0, q[r] + radius)
        bounds.append((lo_r, hi_r))
    bounds.append((-1.0, 1.0))
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.clip(res.x[:m], 0.0, 1.0), float(res.x[m])
    except Exception:
        pass
    return q.copy(), float(corrs.max())

def solve_1d_minimax(a, b):
    lo = 0.0
    hi = 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        vals = a * mid + b
        k_star = int(np.argmax(vals))
        slope = a[k_star]
        if slope < -1e-15:
            lo = mid
        elif slope > 1e-15:
            hi = mid
        else:
            break
    v_opt = (lo + hi) / 2.0
    f_opt = float(np.max(a * v_opt + b))
    return v_opt, f_opt

def smooth_max_obj_grad(q, n_full, beta):

    h = build_sym_h(q)
    dx = 2.0 / n_full
    corrs = corr_raw(h) * dx

    shifted = corrs - max_c
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(len(corrs)) / len(corrs)
    else:
        w /= w_sum
    val = max_c + np.log(w_sum) / beta

                q = q_new2
                if radius < 0.5:
                    radius = min(0.5, radius * 1.2)
            else:
                radius *= 0.4
                if radius < 1e-6:
                    break
    # Final coord descent
    if time.time() < time_limit - 5:
        q, c5 = coord_descend(q, max_rounds=10, time_limit=time_limit)
        if c5 < best:
            best = c5
            best_q = q.copy()
    return best_q, best

def transfer_h(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    x_old = (np.arange(n_old) + 0.5) * 2.0 / n_old
    x_new = (np.arange(n_new) + 0.5) * 2.0 / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new / 2.0)

def make_osc_start(m, rng, freq=2.0, amp=0.3):
    x = np.linspace(0, 1, m)
    q = 0.5 + amp * np.sin(2 * np.pi * freq * x + rng.uniform(0, 2 * np.pi))
    return np.clip(q, 0.0, 1.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def rem():
        return budget_s - (time.time() - t0)

    best_h = None
    best_c5 = float('inf')

    h_parent = None
    try:
        h_parent = np.array(initial_h_values, dtype=float).copy().reshape(-1)
    except (NameError, AttributeError):
        h_parent = None

    if h_parent is not None and len(h_parent) > 10:
        if abs(np.sum(h_parent) - len(h_parent) / 2.0) > 0.5:
            h_parent = project_feasible(h_parent, len(h_parent) / 2.0)
        best_h = np.clip(h_parent, 0.0, 1.0)
        best_h = project_feasible(best_h, len(best_h) / 2.0)
        best_c5 = eval_score(best_h)
    else:
        m_fb = 63
        x = np.linspace(0, 1, m_fb)
        q_fb = np.clip(0.5 + 0.3 * np.sin(2 * np.pi * x), 0, 1)
        best_h = build_sym_h(q_fb)
        best_c5 = eval_score(best_h)

    n_cur = len(best_h)

    # Phase 1: Smooth max warmup
    if n_cur % 2 == 0 and rem() > 100:
        q_init = sym_from_h(best_h)
        tl = time.time() + min(80, rem() * 0.15)
        q_opt, c5_s = smooth_max_optimize(q_init, n_cur, [100, 300], 200, tl)
        h_s = build_sym_h(q_opt)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()

    # Phase 2: LP + coordinate descent main loop
    if n_cur % 2 == 0 and rem() > 100:
        q_w = sym_from_h(best_h)
        tl2 = time.time() + min(400, rem() * 0.4)
        q_opt, c5_lp = lp_coord_loop(q_w, tl2, max_rounds=100)
        h_opt = build_sym_h(q_opt)
        if c5_lp < best_c5:
            best_c5 = c5_lp
            best_h = h_opt.copy()
        else:
            # Still run coord descent on current best
            if rem() > 50:
                q_c = sym_from_h(best_h)
                tl3 = time.time() + min(60, rem() * 0.2)
                q_c, c5_c = coord_descend(q_c, max_rounds=15, time_limit=tl3)
                h_c = build_sym_h(q_c)
                if c5_c < best_c5:
                    best_c5 = c5_c
                    best_h = h_c.copy()

    # Phase 3: Multi-resolution exploration
    n_base = len(best_h)
    n_candidates = set()
    for delta in [-12, -8, -6, -4, -3, -2, -1, 1, 2, 3, 4, 6, 8, 12]:
        nc = n_base + delta
        if 48 <= nc <= 200:
            n_candidates.add(nc)
    for nc in [48, 54, 60, 66, 72, 78, 84, 90, 96, 102, 108, 114, 120, 132, 138, 144]:
        n_candidates.add(nc)
    n_candidates = sorted(n_candidates)

    for n_try in n_candidates:
        if rem() < 40:
            break
        if n_try % 2 != 0:
            continue
        if n_try >= n_base and n_try != n_base:
            continue
        h_try = transfer_h(best_h, n_try)
        n_try = len(h_try)
        if n_try % 2 != 0:
            continue
        q_try = sym_from_h(h_try)
        c5_try = eval_score(build_sym_h(q_try))

        # Quick LP+coord on this resolution
        if rem() > 60:
            tl_r = time.time() + min(60, rem() * 0.3)
            q_r, c5_r = lp_coord_loop(q_try, tl_r, max_rounds=40)
            h_r = build_sym_h(q_r)
            c5_r = eval_score(h_r)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r.copy()

    # Phase 4: Diverse restarts
    n_restarts = min(8, max(2, int(rem() / 80)))
    for i in range(n_restarts):
        if rem() < 50:
            break
        n_r = len(best_h) if i % 2 == 0 else max(56, len(best_h) - 6)
        n_r = min(160, max(48, n_r))
        if n_r % 2 != 0:
            n_r -= 1
        m_r = n_r // 2
        q_r = make_osc_start(m_r, rng, freq=rng.choice([1.0, 2.0, 3.0, 4.0]),
                            amp=rng.uniform(0.2, 0.4))
        if rem() > 80:
            tl_x = time.time() + min(50, rem() * 0.4)
            q_x, c5_x = lp_coord_loop(q_r, tl_x, max_rounds=30)
            h_x = build_sym_h(q_x)
            c5_x = eval_score(h_x)
            if c5_x < best_c5:
                best_c5 = c5_x
                best_h = h_x.copy()

    # Phase 5: Final polish
    if rem() > 20 and best_h is not None:
        n_fin = len(best_h)
        if n_fin % 2 == 0:
            q_fin = sym_from_h(best_h)
            if rem() > 40:
                tl_f = time.time() + min(30, rem() * 0.5)
                q_f, c5_f = lp_coord_loop(q_fin, tl_f, max_rounds=30)
                h_f = build_sym_h(q_f)
                c5_f = eval_score(h_f)
                if c5_f < best_c5:
                    best_c5 = c5_f
                    best_h = h_f.copy()
            if rem() > 15:
                q_fin2 = sym_from_h(best_h)
                q_fin2, c5_fin = coord_descend(q_fin2, max_rounds=12, time_limit=time.time() + 10)
                h_fin = build_sym_h(q_fin2)
                if c5_fin < best_c5:
                    best_c5 = c5_fin
                    best_h = h_fin.copy()

    # Validate and return
    if best_h is None:
        m_fb = 63
        x = np.linspace(0, 1, m_fb)
        q_fb = np.clip(0.5 + 0.3 * np.sin(2 * np.pi * x), 0, 1)
        best_h = build_sym_h(q_fb)

    best_h = np.asarray(best_h, dtype=float).reshape(-1)
    n_out = len(best_h)
    best_h = project_feasible(best_h, n_out / 2.0)
    best_c5 = eval_score(best_h)
    return best_h, best_c5, n_out
