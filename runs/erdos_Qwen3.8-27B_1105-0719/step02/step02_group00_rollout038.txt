def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    # Adjust sum
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)  # indices with most capacity first
        remaining = diff
        for idx in order:
            add = min(remaining, 1.0 - h[idx])
            h[idx] += add
            remaining -= add
            if remaining <= 1e-14:
                break
    else:
        order = np.argsort(h)  # indices with least h first
        remaining = -diff
        for idx in order:
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-14:
                break
    # Final correction
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = np.argmax(1.0 - h)
        else:
            idx = np.argmax(h)
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

import numpy as np
from scipy.optimize import minimize
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        remaining = diff
        for idx in order:
            add = min(remaining, 1.0 - h[idx])
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-13:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for idx in order:
            rem = min(remaining, h[idx])
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-13:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-14:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] = np.clip(h[idx] + diff2, 0.0, 1.0)
    return h

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    return repair_feasible(np.clip(h_new, 0.0, 1.0))

def pen_coord_descent(h, max_sweeps=80, lam_start=50.0, lam_end=200.0, seed=42):
    """Coordinate descent with adaptive penalty on sum constraint."""
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    target = n / 2.0
    center = n - 1
    rng = np.random.RandomState(seed)
    best_c = exact_score(h)
    
    for sweep in range(max_sweeps):
        lam = lam_start + (lam_end - lam_start) * sweep / max(1, max_sweeps - 1)
        order = rng.permutation(n)
        
        for i in order:
            x_old = h[i]
            h[i] = 0.0
            P0 = np.correlate(h, 1 - h, mode="full") * dx
            h[i] = 1.0
            P1 = np.correlate(h, 1 - h, mode="full") * dx
            h[i] = x_old
            B = P1 - P0
            S_other = h.sum() - x_old
            t = target - S_other
            
            # Minimize max(P0 + B*x, with quadratic at center) + lam*(x-t)^2 over x in [0,1]
            grid = np.linspace(0, 1, 120)
            best_x = x_old
            best_f = np.inf
            for xg in grid:
                P_vals = P0 + B * xg
                P_vals[center] = P0[center] + dx * xg * (1.0 - xg)
                f_val = np.max(P_vals) + lam * (xg - t) ** 2
                if f_val < best_f:

                    best_x = xg

        if c < best_c:

def build_gradient_matrix(h, n, dx):
    M = 2 * n - 1
    G = np.zeros((M, n))
    for m in range(M):
        d = m - (n - 1)
        if d == 0:
            G[m, :] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            G[m, d:n] += dx * (1.0 - h[0:n-d])
            G[m, 0:n-d] += -dx * h[d:n]
        else:
            dp = -d
            G[m, 0:n-dp] += dx * (1.0 - h[dp:n])
            G[m, dp:n] += -dx * h[0:n-dp]
    return G

def solve_epigraph(h_start, maxiter=300):
    n = len(h_start)
    dx = 2.0 / n
    M = 2 * n - 1
    c0 = exact_score(h_start)
    
    def obj(x):
        return x[-1]
    
    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    
    def eq_fun(x):
        return x[:-1].sum() - n / 2.0
    
    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g
    
    def ineq_fun(x):
        h = x[:-1]
        t = x[-1]
        profile = np.correlate(h, 1 - h, mode="full") * dx
        return t - profile
    
    def ineq_jac(x):
        h = x[:-1]
        G = build_gradient_matrix(h, n, dx)
        J = np.zeros((M, n + 1))
        J[:, :-1] = -G
        J[:, -1] = 1.0
        return J
    
    x0 = np.concatenate([h_start, [c0 + 1e-6]])
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 1.0)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None and np.isfinite(res.x).all():
            h_new = repair_feasible(res.x[:-1])
            c_new = exact_score(h_new)
            if c_new <= c0 + 1e-9:

def lagrangian_coord_descent(h, max_sweeps=200, seed=42):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    target = n / 2.0
    center = n - 1
    lam = 0.0
    rng = np.random.RandomState(seed)
    
    for sweep in range(max_sweeps):
        order = rng.permutation(n)
        for i in order:
            x_old = h[i]
            # Compute P0 and P1 (profile with h[i]=0 and h[i]=1)
            h[i] = 0.0
            P0 = np.correlate(h, 1-h, mode="full") * dx
            h[i] = 1.0
            P1 = np.correlate(h, 1-h, mode="full") * dx
            h[i] = x_old
            B = P1 - P0
            
            # For d=0: P_0(x) = P0[center] + dx*x*(1-x)
            # For d≠0: P_d(x) = P0[m] + B[m]*x
            # Objective: max over all d of P_d(x) + lam*x
            # (the constant lam*(S_other - target) doesn't matter for argmin)
            
            # Grid search
            grid = np.linspace(0, 1, 200)
            best_x = x_old
            best_f = np.inf
            for xg in grid:
                P_vals = P0 + B * xg
                P_vals[center] = P0[center] + dx * xg * (1.0 - xg)
                f_val = np.max(P_vals) + lam * xg
                if f_val < best_f:
                    best_f = f_val
                    best_x = xg
            
            h[i] = best_x
        
        # Update lambda
        sum_h = h.sum()
        err = sum_h - target
        lam += 0.5 * err / n  # subgradient step
    
    return repair_feasible(h)
