    h[:m] = q
    h[m:] = 1.0 - q[::-1]
    return h

    q = np.empty(m)

def project_feasible(x, target_sum):

def _smmax(h, dx, beta):

    mx = c.max()
    ws = np.exp(beta * (c - mx)).sum()

import numpy as np
from scipy.optimize import minimize
import time

def eval_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def build_sym_h(q):
    m = len(q)
    h = np.empty(2 * m)
    h[:m] = q
    h[h.shape[0] // 2:] = 1.0 - q[::-1]
    return h

def sym_from_h(h):
    n = len(h)
    m = n // 2
    q = np.empty(m)
    for i in range(m):
        q[i] = 0.5 * (h[i] + (1.0 - h[n - 1 - i]))
    return np.clip(q, 0.0, 1.0)

def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.5, 1.5
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(x + mid, 0.0, 1.0).sum()
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    h = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        interior = (h > 1e-10) & (h < 1.0 - 1e-10)
        if interior.sum() > 0:
            h[interior] += diff / interior.sum()
    return np.clip(h, 0.0, 1.0)

def _smmax(h, dx, beta):
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    mx = c.max()
    ws = np.exp(beta * (c - mx)).sum()
    if ws < 1e-30:
        return mx
    return mx + np.log(ws) / beta

def cd_sweep(q, n_full, beta, max_rounds=200):
    m = n_full // 2
    dx = 2.0 / n_full
    gr = 0.38196601125  # (3-sqrt(5))/2, the golden section ratio
    h = build_sym_h(q)
    best_actual = eval_score(h)

    for rnd in range(max_rounds):
        any_change = False
        for r in range(m):
            sr = n_full - 1 - r
            old_val = h[r]
            lo, hi = 0.0, 1.0
            c1 = lo + gr * (hi - lo)
            c2 = hi - gr * (hi - lo)

            h[r] = c1; h[sr] = 1.0 - c1
            v1 = _smmax(h, dx, beta)
            h[r] = c2; h[sr] = 1.0 - c2
            v2 = _smmax(h, dx, beta)

            for _ in range(35):
                if v1 <= v2:
                    hi = c2; c2 = c1; v2 = v1
                    c1 = lo + gr * (hi - lo)
                    h[r] = c1; h[sr] = 1.0 - c1
                    v1 = _smmax(h, dx, beta)
                else:
                    lo = c1; c1 = c2; v1 = v2
                    c2 = hi - gr * (hi - lo)
                    h[r] = c2; h[sr] = 1.0 - c2
                    v2 = _smmax(h, dx, beta)

            best_a = (lo + hi) / 2.0
            h[r] = best_a; h[sr] = 1.0 - best_a
            q[r] = best_a
            if abs(best_a - old_val) > 1e-12:
                any_change = True

        sc = eval_score(h)
        if sc < best_actual:
            best_actual = sc
        if not any_change:
            break

    return q, h, best_actual

def _epig_con_fun(x, k, n_full):
    m = n_full // 2
    q = x[:m]
    t = x[m]
    h = build_sym_h(q)
    dx = 2.0 / n_full
    d = k - (n_full - 1)
    i_lo = max(0, -d)
    i_hi = min(n_full - 1, n_full - 1 - d)
    if i_lo > i_hi:
        return t
    val = np.sum(h[i_lo:i_hi+1] * (1.0 - h[i_lo+d:i_hi+1+d])) * dx
    return t - val

def _epig_con_jac(x, k, n_full):
    m = n_full // 2
    q = x[:m]
    h = build_sym_h(q)
    dx = 2.0 / n_full
    d = k - (n_full - 1)
    i_lo = max(0, -d)
    i_hi = min(n_full - 1, n_full - 1 - d)
    j = np.zeros(m + 1)
    j[m] = 1.0
    if i_lo <= i_hi:
        idx = np.arange(i_lo, i_hi + 1)
        g = np.zeros(m + 1)
        # Gradient w.r.t. h[idx] is (1-h[idx+d]), w.r.t. h[idx+d] is -h[idx]
        # Map back to q: h[i] = q[i] for i<m, h[i] = 1-q[n-1-i] for i>=m
        for i in idx:
            gi = 1.0 - h[i + d]
            gj = -h[i]
            # h[i] depends on q: if i < m: h[i]=q[i], else h[i]=1-q[n-1-i]
            if i < m:
                g[i] += gi
            else:
                qi = n_full - 1 - i
                g[qi] -= gi
            if (i + d) < m:
                g[i + d] += gj
            else:
                qi = n_full - 1 - (i + d)
                g[qi] -= gj
        j[:m] = -dx * g
    return j

def _epig_obj(x):
    return x[-1]

def _epig_obj_jac(x):
    g = np.zeros(len(x))
    g[-1] = 1.0
    return g

def solve_epigraph_sym(q_center, n_full, active_k, maxiter=100):
    m = n_full // 2
    h_c = build_sym_h(q_center)
    dx = 2.0 / n_full
    corrs_c = np.correlate(h_c, 1.0 - h_c, mode='full') * dx
    t0_val = corrs_c.max() + 1e-8

    x0 = np.zeros(m + 1)
    x0[:m] = q_center
    x0[m] = t0_val

    bounds = [(0.0, 1.0)] * m + [(0.0, 2.0)]
    constraints = []
    for k in active_k:
        constraints.append({
            'type': 'ineq',
            'fun': _epig_con_fun_k(n_full, k),
            'jac': _epig_con_jac_k(n_full, k)
        })

    try:
        res = minimize(_epig_obj, x0, jac=_epig_obj_jac,
                      method='SLSQP', bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-12})
        q_out = np.clip(res.x[:m], 0.0, 1.0)
    except Exception:
        q_out = q_center.copy()

    h_out = build_sym_h(q_out)
    return q_out, h_out, eval_score(h_out)

def _epig_con_fun_k(n_full, k):
    return lambda x: _epig_con_fun(x, k, n_full)

def _epig_con_jac_k(n_full, k):
    return lambda x: _epig_con_jac(x, k, n_full)

def select_active(corrs, margin_frac=0.03, neighbor_pad=4, max_k=70):
    max_val = corrs.max()
    threshold = max_val - max(2e-6, margin_frac * max_val)
    active = set(np.where(corrs >= threshold)[0].tolist())
    n = len(corrs)
    padded = set()
    for k in active:
        for dd in range(-neighbor_pad, neighbor_pad + 1):
            if 0 <= k + dd < n:
                padded.add(k + dd)
    active_list = sorted(padded)
    if len(active_list) > max_k:
        scored = sorted([(corrs[k], k) for k in active_list], reverse=True)
        active_list = sorted([k for _, k in scored[:max_k]])
    return active_list

def transfer_to_n(h_old, n_new):
    n_old = len(h_old)
    x_old = (np.arange(n_old) + 0.5) * 2.0 / n_old
    x_new = (np.arange(n_new) + 0.5) * 2.0 / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new / 2.0)

def full_epig_con_fun(x, k, n):
    h = x[:n]
    t = x[n]
    dx = 2.0 / n
    d = k - (n - 1)
    i_lo = max(0, -d)
    i_hi = min(n - 1, n - 1 - d)
    if i_lo > i_hi:
        return t
    val = np.sum(h[i_lo:i_hi+1] * (1.0 - h[i_lo+d:i_hi+1+d])) * dx
    return t - val

def full_epig_con_jac(x, k, n):
    h = x[:n]
    dx = 2.0 / n
    d = k - (n - 1)
    i_lo = max(0, -d)
    i_hi = min(n - 1, n - 1 - d)
    j = np.zeros(n + 1)
    j[n] = 1.0
    if i_lo <= i_hi:
        idx = np.arange(i_lo, i_hi + 1)
        j[idx] += -dx * (1.0 - h[idx + d])
        j[idx + d] += -dx * (-h[idx])
    return j

def full_eq_fun(x, n):
    return np.sum(x[:n]) - n / 2.0

def full_eq_jac(x, n):
    j = np.zeros(n + 1)
    j[:n] = 1.0
    return j

def _full_obj(x):
    return x[-1]

def _full_obj_jac(x):
    g = np.zeros(len(x))
    g[-1] = 1.0
    return g

def solve_epigraph_full(h_center, active_k, maxiter=60):
    n = len(h_center)
    dx = 2.0 / n
    corrs_c = np.correlate(h_center, 1.0 - h_center, mode='full') * dx
    t0_val = corrs_c.max() + 1e-8

    x0 = np.zeros(n + 1)
    x0[:n] = h_center
    x0[n] = t0_val

    bounds = [(0.0, 1.0)] * (n + 1)
    constraints = []
    for k in active_k:
        constraints.append({
            'type': 'ineq',
            'fun': lambda xx, kk=k, nn=n: full_epig_con_fun(xx, kk, nn),
            'jac': lambda xx, kk=k, nn=n: full_epig_con_jac(xx, kk, nn)
        })
    constraints.append({
        'type': 'eq',
        'fun': lambda xx: full_eq_fun(xx, n),
        'jac': lambda xx: full_eq_jac(xx, n)
    })

    try:
        res = minimize(_full_obj, x0, jac=_full_obj_jac,
                      method='SLSQP', bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-12})
        h_out = res.x[:n]
    except Exception:
        h_out = h_center.copy()

    h_out = project_feasible(h_out, n / 2.0)
    return h_out, eval_score(h_out)

        elif i % 3 == 2:
            n_r = min(180, n_base + 4)
        n_r = n_r if n_r % 2 == 0 else n_r + 1

        m_r = n_r // 2
        x_r = np.linspace(0, 1, m_r)
        # Different initial patterns
        if i % 4 == 0:
            q_r = np.clip(0.5 + 0.3 * np.sin(2*np.pi*x_r + rng.uniform(0, 2*np.pi)), 0, 1)
        elif i % 4 == 1:
            q_r = np.clip(0.5 + 0.2 * np.cos(4*np.pi*x_r) + 0.15*np.sin(2*np.pi*x_r), 0, 1)
        else:
            q_r = np.clip(0.5 + rng.uniform(-0.3, 0.3, m_r), 0, 1)

        h_r = build_sym_h(q_r)
        sc_r = eval_score(h_r)

        if sc_r < best_c5 + 0.01 and rem() > 50:
            for beta in [100, 500]:
                if rem() < 30:
                    break
                q_r, h_r, sc_r = cd_sweep(q_r, n_r, beta, max_rounds=80)
                if sc_r < best_c5:
                    best_c5 = sc_r
                    best_h = h_r.copy()

    # Phase 6: Final high-beta CD polish
    if rem() > 30:
        n_fin = len(best_h)
        if n_fin % 2 == 0:
            q_fin = sym_from_h(best_h)
            for beta in [2000, 5000, 10000]:
                if rem() < 15:
                    break
                q_fin, h_fin, sc_fin = cd_sweep(q_fin, n_fin, beta, max_rounds=100)
                if sc_fin < best_c5:
                    best_c5 = sc_fin
                    best_h = h_fin.copy()

            # Final epigraph
            if rem() > 20:
                q_fin = sym_from_h(best_h)
                h_chk = build_sym_h(q_fin)
                corrs = np.correlate(h_chk, 1.0 - h_chk, mode='full') * (2.0 / n_fin)
                active = select_active(corrs, margin_frac=0.005, neighbor_pad=2, max_k=40)
                q_out, h_out, c5_out = solve_epigraph_sym(q_fin, n_fin, active, maxiter=120)
                h_out = build_sym_h(np.clip(q_out, 0, 1))
                c5_out = eval_score(h_out)
                if c5_out < best_c5:
                    best_c5 = c5_out
                    best_h = h_out

    # Return
    if best_h is None:
        m_fb = 63
        x_fb = np.linspace(0, 1, m_fb)
        q_fb = np.clip(0.5 + 0.25 * np.sin(2*np.pi*x_fb), 0, 1)
        best_h = build_sym_h(q_fb)

    best_h = np.asarray(best_h, dtype=float).reshape(-1)
    n_out = len(best_h)
    best_h = project_feasible(best_h, n_out / 2.0)
    best_c5 = eval_score(best_h)

    return best_h, best_c5, n_out
