                no_prog = 0
                R = min(0.12, R * 1.12)
            else:
                no_prog += 1
                if no_prog >= 3:
                    R = max(0.003, R * 0.6)
                    no_prog = 0
                if R < 0.004:
                    break

    # Phase 4: Perturbation + re-optimize
    if tl() > reserve + 30:
        for trial in range(80):
            if tl() < reserve + 10:
                break
            scale = rng.uniform(0.003, 0.02)
            h_p = best_h + rng.randn(len(best_h)) * scale
            h_p = repair_sum(h_p, len(best_h) / 2.0)
            c_p = eval_c5(h_p)
            if c_p < best_c5 - 1e-12:
                best_h = h_p.copy()
                best_c5 = c_p

    # Phase 5: Final symmetric SLP polish on best
    n_fin = len(best_h)
    if n_fin % 2 == 0 and tl() > reserve + 10:
        q_fin = q_from_h(best_h, n_fin)
        h_fin = build_h(q_fin, n_fin)
        c_fin = eval_c5(h_fin)
        if c_fin <= best_c5 + 1e-12:
            best_h = h_fin
            best_c5 = c_fin
        R = 0.03
        for it in range(200):
            if tl() < reserve + 5:
                break
            q_fin, c_new, improved = slp_step_sym(q_fin, n_fin, R=R, margin=0.001)
            h_new = build_h(q_fin, n_fin)
            if c_new < best_c5 - 1e-12:
                best_h = h_new.copy()
                best_c5 = c_new
                R = min(0.06, R * 1.1)
            else:
                if R > 0.005:
                    R *= 0.7
                else:
                    break

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = repair_sum(best_h, len(best_h) / 2.0)
    best_c5 = eval_c5(best_h)
    return best_h, best_c5, len(best_h)
