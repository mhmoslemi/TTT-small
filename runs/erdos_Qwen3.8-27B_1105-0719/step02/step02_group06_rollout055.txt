def eval_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def corr_raw(h):
    return np.correlate(h, 1.0 - h, mode='full')

def build_sym_h(q):
    m = len(q)
    h = np.empty(2 * m)
    h[:m] = q
    h[m:] = 1.0 - q[::-1]
    return h

def sym_from_h(h):
    n = len(h)
    m = n // 2
    q = np.empty(m)
    for i in range(m):
        q[i] = 0.5 * (h[i] + (1.0 - h[n - 1 - i]))
    return np.clip(q, 0.0, 1.0)

def project_feasible(x, target_sum):
    n = len(x)
    x = np.clip(np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0), 0.0, 1.0)
    if abs(x.sum() - target_sum) < 1e-13:
        return x
    lo, hi = -2.0, 2.0

        if s > target_sum:
            hi = mid
        else:
            lo = mid
    c = (lo + hi) / 2.0
    v = np.clip(x + c, 0.0, 1.0)
    r = target_sum - v.sum()
    if abs(r) > 1e-12:
        if r > 0:
            idx = np.argsort(v)
        else:
            idx = np.argsort(-v)
        for i in idx:
            if r > 0:
                amt = min(r, 1.0 - v[i])
                v[i] += amt
                r -= amt
            else:
                amt = min(-r, v[i])
                v[i] -= amt
                r += amt
            if abs(r) < 1e-13:

def corr_grad_full(h, k):
    """Gradient of raw_corr[k] w.r.t. h (full vector)."""
    n = len(h)
    d = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -d)
    i_hi = min(n - 1, n - 1 - d)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx] += (1.0 - h[idx + d])
    grad[idx + d] += (-h[idx])
    return grad

def select_active(corrs, margin_frac=0.02, neighbor_pad=3, max_k=60):
    max_val = corrs.max()
    threshold = max_val - max(1e-8, margin_frac * max_val)
    active = set(np.where(corrs >= threshold)[0].tolist())
    padded = set()
    for k in active:
        for d in range(-neighbor_pad, neighbor_pad + 1):
            if 0 <= k + d < len(corrs):
                padded.add(k + d)
    active_list = sorted(padded)
    if len(active_list) > max_k:
        scored = sorted([(corrs[k], k) for k in active_list], reverse=True)
        active_list = sorted([k for _, k in scored[:max_k]])
    return active_list

def subgradient_sym(q, n_full, step_size, n_iter=50):
    """Subgradient method for symmetric case."""
    m = n_full // 2
    q = np.clip(q.copy(), 0.0, 1.0)
    best_q = q.copy()
    best_score = eval_score(build_sym_h(q))
    
    for it in range(n_iter):
        h = build_sym_h(q)
        corrs = corr_raw(h)
        max_c = corrs.max()
        
        # Active set
        threshold = max_c - max(1e-8, 0.03 * max_c)
        active = np.where(corrs >= threshold)[0]
        
        # Weighted subgradient
        shifted = corrs - max_c
        weights = np.exp(50.0 * shifted)
        weights = weights / weights.sum()
        
        # Compute subgradient
        sg = np.zeros(m)
        for ell in active:
            if weights[ell] < 1e-12:
                continue
            g_full = corr_grad_full(h, ell)
            g_q = g_full[:m] - g_full[n_full - 1:-m - 1:-1]
            sg += weights[ell] * g_q
        
        # Step
        lr = step_size / (1.0 + 0.1 * it)
        q_new = q - lr * sg
        q_new = np.clip(q_new, 0.0, 1.0)
        
        sc = eval_score(build_sym_h(q_new))
        if sc < best_score:
            best_score = sc
            best_q = q_new.copy()
            q = q_new
        else:
            # Halve step size
            step_size *= 0.9
            q_new = q - lr * 0.5 * sg
            q_new = np.clip(q_new, 0.0, 1.0)
            sc = eval_score(build_sym_h(q_new))
            if sc < best_score:
                best_score = sc
                best_q = q_new.copy()
                q = q_new
    
    return best_q, best_score

import numpy as np

import time

def eval_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def corr_raw(h):
    return np.correlate(h, 1.0 - h, mode='full')

def build_sym_h(q):
    m = len(q)
    h = np.empty(2 * m)
    h[:m] = q
    h[m:] = 1.0 - q[::-1]
    return h

def sym_from_h(h):
    n = len(h)
    m = n // 2
    q = np.empty(m)
    for i in range(m):
        q[i] = 0.5 * (h[i] + (1.0 - h[n - 1 - i]))
    return np.clip(q, 0.0, 1.0)

def project_feasible(x, target_sum):
    n = len(x)
    x = np.clip(np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0), 0.0, 1.0)
    diff = target_sum - x.sum()
    if abs(diff) < 1e-13:
        return x
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(x + mid, 0.0, 1.0).sum()
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    c = (lo + hi) / 2.0
    v = np.clip(x + c, 0.0, 1.0)
    r = target_sum - v.sum()
    if abs(r) > 1e-12:
        if r > 0:
            idx = np.argsort(v)
        else:
            idx = np.argsort(-v)
        for i in idx:
            if r > 0:
                amt = min(r, 1.0 - v[i])
                v[i] += amt
                r -= amt
            else:
                amt = min(-r, v[i])
                v[i] -= amt
                r += amt
            if abs(r) < 1e-14:
                break
    return np.clip(v, 0.0, 1.0)

def corr_grad_full(h, k):
    n = len(h)
    d = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -d)
    i_hi = min(n - 1, n - 1 - d)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx] += (1.0 - h[idx + d])
    grad[idx + d] += (-h[idx])
    return grad

def select_active(corrs, margin_frac=0.02, neighbor_pad=3, max_k=60):
    max_val = float(corrs.max())
    threshold = max_val - max(1e-8, margin_frac * max_val)
    active = set(np.where(corrs >= threshold)[0].tolist())
    padded = set()
    for k in active:
        for d in range(-neighbor_pad, neighbor_pad + 1):
            if 0 <= k + d < len(corrs):
                padded.add(k + d)
    active_list = sorted(padded)
    if len(active_list) > max_k:
        scored = sorted([(corrs[k], k) for k in active_list], reverse=True)
        active_list = sorted([k for _, k in scored[:max_k]])
    return active_list

def eval_score(h):
    n = len(h)
    dx = 2.0 / n

def corr_raw(h):

def build_sym_h(q):
    m = len(q)

    h[m:] = 1.0 - q[::-1]

def sym_from_h(h):
    n = len(h)

    q = np.empty(m)
    for i in range(m):
        q[i] = 0.5 * (h[i] + (1.0 - h[n - 1 - i]))

def project_feasible(x, target_sum):
    n = len(x)
    x = np.clip(np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0), 0.0, 1.0)
    diff = target_sum - x.sum()
    if abs(diff) < 1e-13:
        return x
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(x + mid, 0.0, 1.0).sum()
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    c = (lo + hi) / 2.0
    v = np.clip(x + c, 0.0, 1.0)
    r = target_sum - v.sum()
    if abs(r) > 1e-12:
        if r > 0:
            idx = np.argsort(v)
        else:
            idx = np.argsort(-v)
        for i in idx:
            if r > 0:
                amt = min(r, 1.0 - v[i])
                v[i] += amt
                r -= amt
            else:
                amt = min(-r, v[i])
                v[i] -= amt
                r += amt
            if abs(r) < 1e-14:
                break
    return np.clip(v, 0.0, 1.0)

def corr_grad_full(h, k):
    n = len(h)
    d = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -d)
    i_hi = min(n - 1, n - 1 - d)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx] += (1.0 - h[idx + d])
    grad[idx + d] += (-h[idx])
    return grad

def select_active(corrs, margin_frac=0.02, neighbor_pad=3, max_k=60):
    max_val = float(corrs.max())
    threshold = max_val - max(1e-8, margin_frac * max_val)
    active = set(np.where(corrs >= threshold)[0].tolist())
    padded = set()
    for k in active:
        for d in range(-neighbor_pad, neighbor_pad + 1):
            if 0 <= k + d < len(corrs):
                padded.add(k + d)
    active_list = sorted(padded)
    if len(active_list) > max_k:
        scored = sorted([(corrs[k], k) for k in active_list], reverse=True)
        active_list = sorted([k for _, k in scored[:max_k]])
    return active_list

        dl = min(t0 + budget_s, time.time() + rem() * 0.25)
        q_opt, sc = optimize_at_n(q, best_n, dl, rem() * 0.25)
        if sc < best_c5:
            best_c5 = sc
            best_h = build_sym_h(q_opt)
            best_n = len(best_h)

    # Phase 2: Multi-resolution exploration
    n_base = best_n
    n_candidates = set()
    for d in [-12, -8, -6, -4, -3, -2, -1, 1, 2, 3, 4, 6, 8, 12, 16]:
        nc = n_base + d
        if 60 <= nc <= 200 and nc % 2 == 0:
            n_candidates.add(nc)
    if n_base // 2 >= 48:
        n_candidates.add(n_base // 2)
    if n_base * 3 // 2 <= 200 and n_base * 3 // 2 % 2 == 0:
        n_candidates.add(n_base * 3 // 2)
    n_candidates = sorted(n_candidates)

    for nc in n_candidates:
        if rem() < 40:
            break
        h_try = transfer_to_n(best_h, nc)
        q_try = sym_from_h(h_try)
        per_budget = min(80, rem() * 0.6 / max(1, len(n_candidates)))
        dl = min(t0 + budget_s, time.time() + per_budget)
        q_opt, sc = optimize_at_n(q_try, nc, dl, per_budget)
        if sc < best_c5:
            best_c5 = sc
            best_h = build_sym_h(q_opt)
            best_n = nc

    # Phase 3: Full (non-symmetric) LP on best
    if rem() > 60:
        n_f = len(best_h)
        dl = min(t0 + budget_s, time.time() + min(80, rem() * 0.3))
        h_f, sc_f = lp_full_optimize(best_h, 0.15, 200, dl)
        if sc_f < best_c5:
            best_c5 = sc_f
            best_h = h_f
            best_n = n_f
        # Second pass with smaller rho
        if rem() > 30 and sc_f < best_c5 + 1e-10:
            dl2 = min(t0 + budget_s, time.time() + min(40, rem() * 0.2))
            h_f2, sc_f2 = lp_full_optimize(best_h, 0.05, 150, dl2)
            if sc_f2 < best_c5:
                best_c5 = sc_f2
                best_h = h_f2
                best_n = n_f

    # Phase 4: Diverse restarts
    n_restarts = min(8, max(2, int(rem() / 80)))
    for i in range(n_restarts):
        if rem() < 50:
            break
        n_r = best_n if i % 3 == 0 else max(72, best_n + int(rng.choice([-8, -4, 4, 8])))
        n_r = min(180, max(64, n_r))
        if n_r % 2 != 0:
            n_r += 1
        m_r = n_r // 2

        if i % 2 == 0:
            q_r = make_fourier_q(m_r, rng)
        else:
            q_r = make_binary_q(m_r, rng)

        h_r = build_sym_h(q_r)
        sc_r = eval_score(h_r)

        # Quick LP optimization
        per_budget = min(60, rem() * 0.7 / max(1, n_restarts - i))
        dl = min(t0 + budget_s, time.time() + per_budget)
        q_opt, sc_opt = lp_sym_optimize(q_r, n_r, 0.4, 150, dl)
        if sc_opt < best_c5:
            best_c5 = sc_opt
            best_h = build_sym_h(q_opt)
            best_n = n_r
        elif sc_opt < sc_r - 0.002 and rem() > 40:
            # Polish
            dl2 = min(t0 + budget_s, time.time() + per_budget * 0.5)
            q_opt2, sc_opt2 = smooth_max_sym(q_opt, n_r, [200, 400], 150, dl2)
            if sc_opt2 < best_c5:
                best_c5 = sc_opt2
                best_h = build_sym_h(q_opt2)
                best_n = n_r

    # Phase 5: Final polish
    if rem() > 20 and best_h is not None:
        n_fin = len(best_h)
        if n_fin % 2 == 0 and rem() > 25:
            q_fin = sym_from_h(best_h)
            dl = min(t0 + budget_s, time.time() + min(25, rem() * 0.5))
            q_o, sc_o = lp_sym_optimize(q_fin, n_fin, 0.08, 100, dl)
            if sc_o < best_c5:
                best_c5 = sc_o
                best_h = build_sym_h(q_o)
                best_n = n_fin
        if rem() > 15:
            dl = min(t0 + budget_s, time.time() + min(15, rem() * 0.5))
            h_o, sc_o = lp_full_optimize(best_h, 0.05, 80, dl)
            if sc_o < best_c5:
                best_c5 = sc_o
                best_h = h_o
                best_n = len(h_o)

    # Validate and return
    if best_h is None:
        m_fb = 63
        q_fb = make_fourier_q(m_fb, rng)
        best_h = build_sym_h(q_fb)

    best_h = np.asarray(best_h, dtype=float).reshape(-1)
    n_out = len(best_h)
    target = float(n_out) / 2.0
    if abs(best_h.sum() - target) > 0.01:
        best_h = project_feasible(best_h, target)
    else:
        best_h = np.clip(best_h, 0.0, 1.0)
        diff = target - best_h.sum()
        if abs(diff) > 1e-10:
            best_h = project_feasible(best_h, target)
    best_c5 = eval_score(best_h)

    return best_h, best_c5, n_out
