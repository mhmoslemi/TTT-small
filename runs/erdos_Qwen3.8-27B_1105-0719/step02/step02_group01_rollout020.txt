def project_feasible(x, target_sum):
    """Project x onto {h: 0≤h≤1, sum(h)=target_sum}"""
    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) / 2, 0.0, 1.0)
    # Fix sum exactly
    diff = target_sum - h.sum()
    if abs(diff) > 1e-10:
        # Distribute among interior points
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def c5_of_h(h, dx):
    return compute_corrs(h).max() * dx

def grad_corr_k(h, k, n):
    """Gradient of corr[k] with respect to h."""
    offset = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -offset)
    i_hi = min(n - 1, n - 1 - offset)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx] += (1.0 - h[idx + offset])
    # Second term: -h[j-offset] for j where j-offset is in [i_lo, i_hi]
    # j = i + offset, so j ranges from i_lo+offset to i_hi+offset
    j_lo = max(0, i_lo + offset)
    j_hi = min(n - 1, i_hi + offset)
    if j_lo <= j_hi:
        grad[j_lo:j_hi+1] += (-h[j_lo - offset: j_hi - offset + 1])
    return grad

def grad_all_corrs(h, n):
    """Compute gradients for all correlations. Returns array of shape (2n-1, n)."""
    grads = np.zeros((2*n-1, n))
    for k in range(2*n-1):

**Pair Transfer Algorithm:**
1. Compute all correlations
2. Find the top M (e.g., 20) correlations (the "active" set)
3. For each active k, compute B_k = grad_corr_k(h, k, n) (the gradient of corr[k])
4. Find the pair (i, j) that maximizes min_k (B_k[i] - B_k[j]) over active k

5. Transfer a small amount of mass from j to i
6. Check exact improvement
7. Repeat

For step 4, the brute force is O(n² * M). For n=126, M=20: 126² * 20 = 317,520 operations. That's fast.

But we can do better: for each i, compute min_k B_k[i] (the "bottleneck" gradient at position i). Then the best pair is (i*, j*) where i* argmax min_k B_k[i] and j* argmin max_k B_k[j]. Wait, that's not quite right either.

        h_opt, c5_val = slsqp_smooth(best_h, dx, target_sum, beta, 1500)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_opt

    # Phase 2: LP polish
    if remaining_time(t0, budget_s) > 30:
        h_lp, c5_lp = lp_polish(best_h, dx, target_sum, 60, 0.3)
        if c5_lp < best_c5:
            best_c5 = c5_lp
            best_h = h_lp

    # Phase 3: Pair transfer polish (main improvement)
    if remaining_time(t0, budget_s) > 60:
        h_pt, c5_pt = pair_transfer_polish(best_h, dx, target_sum, 15000)
        if c5_pt < best_c5:
            best_c5 = c5_pt
            best_h = h_pt

    # Phase 4: SLSQP again with very high beta
    if remaining_time(t0, budget_s) > 30:
        h_opt, c5_val = slsqp_smooth(best_h, dx, target_sum, 5000, 1200)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_opt

    # Phase 5: More pair transfers
    if remaining_time(t0, budget_s) > 40:
        h_pt, c5_pt = pair_transfer_polish(best_h, dx, target_sum, 15000)
        if c5_pt < best_c5:
            best_c5 = c5_pt
            best_h = h_pt

    # Phase 6: LP + pair transfer cycle
    for cycle in range(3):
        if remaining_time(t0, budget_s) < 30:
            break
        h_lp, c5_lp = lp_polish(best_h, dx, target_sum, 40, 0.2)
        if c5_lp < best_c5:
            best_c5 = c5_lp
            best_h = h_lp
        if remaining_time(t0, budget_s) < 15:
            break
        h_pt, c5_pt = pair_transfer_polish(best_h, dx, target_sum, 8000)
        if c5_pt < best_c5:
            best_c5 = c5_pt
            best_h = h_pt

    # Phase 7: Try nearby n values by interpolation
    n_try_list = [124, 128, 130, 120, 132, 136, 122, 140]
    for n_try in n_try_list:
        if remaining_time(t0, budget_s) < 30:
            break
        if n_try == n:
            continue
        dx_try = 2.0 / n_try
        ts_try = n_try / 2.0
        x_orig = np.linspace(0.0, 2.0, len(best_h))
        x_new = np.linspace(0.0, 2.0, n_try)
        h_interp = np.interp(x_new, x_orig, best_h)
        h_interp = project_feasible(h_interp, ts_try)

        h_opt, c5_val = slsqp_smooth(h_interp, dx_try, ts_try, 500, 800)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_opt
            if remaining_time(t0, budget_s) > 20:
                h_pt, c5_pt = pair_transfer_polish(best_h, 2.0 / len(best_h),
                                                   len(best_h) / 2.0, 5000)
                if c5_pt < best_c5:
                    best_c5 = c5_pt
                    best_h = h_pt

    # Phase 8: Final intensive polish on best solution
    if remaining_time(t0, budget_s) > 15 and best_h is not None:
        n_b = len(best_h)
        dx_b = 2.0 / n_b
        ts_b = n_b / 2.0

        h_pt, c5_pt = pair_transfer_polish(best_h, dx_b, ts_b, 10000)
        if c5_pt < best_c5:
            best_c5 = c5_pt
            best_h = h_pt

        if remaining_time(t0, budget_s) > 12:
            h_opt, c5_val = slsqp_smooth(best_h, dx_b, ts_b, 8000, 1000)
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h_opt

        if remaining_time(t0, budget_s) > 10:
            h_pt, c5_pt = pair_transfer_polish(best_h, dx_b, ts_b, 8000)
            if c5_pt < best_c5:
                best_c5 = c5_pt
                best_h = h_pt

        if remaining_time(t0, budget_s) > 8:
            h_lp, c5_lp = lp_polish(best_h, dx_b, ts_b, 30, 0.15)
            if c5_lp < best_c5:
                best_c5 = c5_lp
                best_h = h_lp

    return best_h, best_c5, len(best_h)
