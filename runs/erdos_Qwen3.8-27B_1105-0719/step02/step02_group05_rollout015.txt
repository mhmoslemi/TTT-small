from scipy.special import logsumexp

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corrs = np.correlate(h, 1.0 - h, mode='full')
    return corrs.max() * dx, corrs

    m = n // 2
    q = q.copy()
    h = build_h_from_q(q, n)
    current_score, _ = exact_score(h)
    best_q = q.copy()
    best_score = current_score

        if t0 is not None and time.time() - t0 > time_limit:
            break

        for r in range(m):

            q_fixed = q.copy()
            res = minimize_scalar(q_coordinate_obj, bounds=(0.0, 1.0), 
                                   args=(r, q_fixed, n), method='bounded',
                                   options={'xatol': 1e-10})
            if res.fun < current_score - 1e-14:

                current_score = res.fun
                improved = True
                if current_score < best_score:

                    best_q = q.copy()
        if not improved:
