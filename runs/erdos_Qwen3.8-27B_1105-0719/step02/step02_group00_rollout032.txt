    mid = n - 1
    G[mid, :] = dx * (1.0 - 2.0 * h)

        G[m, d:] += dx * (1.0 - h[:-d])
        G[m, :n-d] -= dx * h[d:]

    # 3. v[i] - u[i] <= h[i]  (n rows)

        A_ub[n_active + i, n + i] = -1.0
        b_ub[n_active + i] = 1.0 - h[i]
    
    # Box lower: v[i] - u[i] <= h[i]  (i.e., -(u[i] - v[i]) <= h[i])
    for i in range(n):
        A_ub[n_active + n + i, i] = -1.0
        A_ub[n_active + n + i, n + i] = 1.0
        b_ub[n_active + n + i] = h[i]
    
    # Equality constraints: A_eq @ x = b_eq
    # 1. sum(u) - sum(v) = 0
    # 2. sum(u) + sum(v) = 1
    
    A_eq = np.zeros((2, nvar))
    b_eq = np.zeros(2)
    A_eq[0, :n] = 1.0
    A_eq[0, n:2*n] = -1.0
    b_eq[0] = 0.0
    A_eq[1, :n] = 1.0
    A_eq[1, n:2*n] = 1.0
    b_eq[1] = 1.0
    
    # Bounds
    bounds = [(0, None)] * (2 * n) + [(0, max_alpha)]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            alpha = max(0.0, -res.fun)
            d = res.x[:n] - res.x[n:2*n]
            # Normalize d
            norm_d = np.sum(np.abs(d))
            if norm_d > 1e-10:
                d = d / norm_d
            else:
                return None, 0.0
            # Remove mean to ensure sum=0 more precisely
            d -= np.mean(d)
            return d, alpha
        else:
            return None, 0.0
    except Exception:
        return None, 0.0

def interpolate_to_n(h, new_n):
    """Interpolate h from current dimension to new_n using mass-conserving method."""
    n = len(h)
    if n == new_n:
        return h.copy()
    # Piecewise constant interpolation
    h_new = np.zeros(new_n)
    dx_old = 2.0 / n
    dx_new = 2.0 / new_n
    for j in range(new_n):
        x_start = j * dx_new
        x_end = (j + 1) * dx_new
        # Find overlapping old cells
        k_start = max(0, int(np.ceil(x_start / dx_old - 1e-12)))
        k_end = min(n, int(np.floor(x_end / dx_old + 1e-12)))
        if k_end > k_start:
            # Compute overlap
            total = 0.0
            for k in range(k_start, k_end):
                ov_start = max(x_start, k * dx_old)
                ov_end = min(x_end, (k + 1) * dx_old)
                if ov_end > ov_start:
                    total += h[k] * (ov_end - ov_start)
            h_new[j] = total / dx_new
    return repair_h(h_new)

    return repair_h((h + h[::-1]) / 2.0)

import numpy as np
from scipy.optimize import linprog
import time

def repair_h(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity, kind='mergesort')
        remaining = diff
        for i in range(n):
            idx = int(order[i])
            add = min(remaining, capacity[idx])
            if add > 1e-15:
                h[idx] += add
                remaining -= add
            if remaining <= 1e-14:
                break
    else:
        order = np.argsort(h, kind='mergesort')
        remaining = -diff
        for i in range(n):
            idx = int(order[i])
            rem = min(remaining, h[idx])
            if rem > 1e-15:
                h[idx] -= rem
                remaining -= rem
            if remaining <= 1e-14:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-12:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] = np.clip(h[idx] + diff2, 0.0, 1.0)
    return h

def score_exact(h):
    n = len(h)
    if n < 2:
        return float('inf')
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(profile))

    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def compute_gradient_matrix(h):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    G = np.zeros((M, n))
    mid = n - 1
    G[mid, :] = dx * (1.0 - 2.0 * h)
    for d in range(1, n):
        G[mid + d, d:] += dx * (1.0 - h[:-d])
        G[mid + d, :n-d] -= dx * h[d:]
        G[mid - d, :n-d] += dx * (1.0 - h[d:])
        G[mid - d, d:] -= dx * h[:n-d]
    return G

    A_ub[:n_act, -1] = 1.0

        A_ub[n_act + i, n + i] = -1.0
        b_ub[n_act + i] = 1.0 - h[i]
        A_ub[n_act + n + i, i] = -1.0
        A_ub[n_act + n + i, n + i] = 1.0
        b_ub[n_act + n + i] = h[i]
    A_eq = np.zeros((2, nvar))

    A_eq[0, :n] = 1.0

    A_eq[1, :n] = 1.0

    dx_old = 2.0 / n
    dx_new = 2.0 / new_n
    h_new = np.zeros(new_n)
    for j in range(new_n):
        xs = j * dx_new
        xe = (j + 1) * dx_new
        k_start = max(0, int(np.floor(xs / dx_old)))
        k_end = min(n, int(np.ceil(xe / dx_old)))
        total = 0.0
        for k in range(k_start, k_end):
            os_ = max(xs, k * dx_old)
            oe = min(xe, (k + 1) * dx_old)
            if oe > os_:
                total += h[k] * (oe - os_)
        h_new[j] = total / dx_new
    return repair_h(h_new)

def symmetrize(h):
    return repair_h((h + h[::-1]) / 2.0)

def lp_local_search(h, time_budget):
    best_h = h.copy()
    best_c = score_exact(h)
    t0 = time.perf_counter()
    beam = 0.006
    stall = 0
    max_stall = 25
    min_beam = 0.001
    max_beam = 0.03
    
    for _ in range(2000):
        if time.perf_counter() - t0 > time_budget:
            break
        profile = compute_profile(best_h)
        c = np.max(profile)
        if c < best_c - 1e-13:
            best_c = c
        spread = c - np.median(profile)
        thresh = c - max(beam, 0.3 * spread)
        active = np.where(profile >= thresh)[0]
        if len(active) == 0:
            active = np.array([np.argmax(profile)])
        if len(active) > 35:
            top_idx = np.argsort(profile[active])[-35:]
            active = active[top_idx]
        G = compute_gradient_matrix(best_h)
        d, alpha = solve_direction_lp(best_h, G, active)
        if d is None or alpha < 1e-9:
            stall += 1
            if stall >= max_stall:
                break
            beam = min(max_beam, beam * 1.3)
            continue
        improved = False
        steps = [1.0, 0.6, 0.4, 0.25, 0.15, 0.08, 0.04, 0.02, 0.01]
        for step in steps:
            h_try = repair_h(best_h + step * d)
            c_try = score_exact(h_try)
            if c_try < best_c - 1e-13:
                best_h = h_try
                best_c = c_try
                improved = True
                stall = 0
                beam = max(min_beam, beam * 0.85)
                break
        if not improved:
            stall += 1
            if stall >= max_stall:
                break
    return best_h, best_c

def weighted_gradient_direction(h, G, active_indices):
    profile = compute_profile(h)
    thresh = np.max(profile) - 0.01
    weights = np.maximum(profile[active_indices] - thresh, 0.001)
    weights = weights / weights.sum()
    grad = G[active_indices, :].T @ weights
    grad -= grad.mean()
    norm = np.linalg.norm(grad)
    if norm < 1e-14:
        return None
    return -grad / norm

def gradient_local_search(h, time_budget):
    best_h = h.copy()
    best_c = score_exact(h)
    t0 = time.perf_counter()
    stall = 0
    for _ in range(500):
        if time.perf_counter() - t0 > time_budget:
            break
        profile = compute_profile(best_h)
        c = np.max(profile)
        if c < best_c - 1e-13:
            best_c = c
        active = np.where(profile >= c - 0.01)[0]
        if len(active) == 0:
            active = np.array([np.argmax(profile)])
        if len(active) > 25:
            top_idx = np.argsort(profile[active])[-25:]
            active = active[top_idx]
        G = compute_gradient_matrix(best_h)
        d = weighted_gradient_direction(best_h, G, active)
        if d is None:
            break
        improved = False
        for step in [0.05, 0.03, 0.02, 0.01, 0.005, 0.002]:
            h_try = repair_h(best_h + step * d)
            c_try = score_exact(h_try)
            if c_try < best_c - 1e-13:
                best_h = h_try
                best_c = c_try
                improved = True
                break
        if not improved:
            stall += 1
            if stall >= 30:
                break
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s - (time.perf_counter() - t_start)

    parent = None
    try:
        parent = np.array(initial_h_values, dtype=np.float64).copy()
    except (NameError, Exception):
        pass

    if parent is None or len(parent) < 4 or not np.isfinite(parent).all():
        n_fb = 70
        parent = np.full(n_fb, 0.5)

    parent = np.clip(parent, 0.0, 1.0)
    parent = repair_h(parent)
    best_h = parent.copy()
    best_c = score_exact(parent)

    h_sym = symmetrize(parent)
    c_sym = score_exact(h_sym)
    if c_sym < best_c - 1e-12:
        best_h = h_sym
        best_c = c_sym

    # Stage 1: LP local search from best
    if time_left() > 30:
        tb = min(180, time_left() * 0.35)
        h_out, c_out = lp_local_search(best_h, tb)
        if c_out < best_c - 1e-12:
            best_h = h_out
            best_c = c_out

    # Stage 2: Gradient-based search from best
    if time_left() > 30:
        tb = min(60, time_left() * 0.2)
        h_out, c_out = gradient_local_search(best_h, tb)
        if c_out < best_c - 1e-12:
            best_h = h_out
            best_c = c_out

    # Stage 3: Perturbed starts
    if time_left() > 80:
        n = len(best_h)
        n_pert = 6
        for i in range(n_pert):
            if time_left() < 25:
                break
            eps = 0.005 + 0.01 * i
            z = rng.randn(n)
            z -= z.mean()
            h_pert = repair_h(best_h + eps * z)
            h_sp = symmetrize(h_pert)
            c_p = score_exact(h_sp)
            c_p2 = score_exact(h_pert)
            h_use = h_sp if c_p <= c_p2 else h_pert
            c_use = min(c_p, c_p2)
            if c_use < best_c + 0.005:
                tb = min(80, time_left() * 0.25)
                h_out, c_out = lp_local_search(h_use, tb)
                if c_out < best_c - 1e-12:
                    best_h = h_out
                    best_c = c_out

    # Stage 4: Dimension exploration
    if time_left() > 100:
        n_best = len(best_h)
        ratios = [0.8, 0.85, 0.9, 0.95, 1.05, 1.1, 1.2, 1.3, 1.5]
        dims = []
        for r in ratios:
            d = int(round(n_best * r / 2.0)) * 2
            if 24 <= d <= 160 and d != n_best:
                dims.append(d)
        dims = sorted(set(dims), key=lambda d: abs(d - n_best))
        for new_n in dims[:5]:
            if time_left() < 50:
                break
            h_interp = interpolate_to_n(best_h, new_n)
            tb = min(120, time_left() * 0.3)
            h_out, c_out = lp_local_search(h_interp, tb)
            if c_out < best_c - 1e-12:
                best_h = h_out
                best_c = c_out
                # Follow up with gradient search
                if time_left() > 30:
                    h_out2, c_out2 = gradient_local_search(best_h, min(40, time_left() * 0.2))
                    if c_out2 < best_c - 1e-12:
                        best_h = h_out2
                        best_c = c_out2

    # Stage 5: Another LP pass on best
    if time_left() > 40:
        tb = min(100, time_left() * 0.4)
        h_out, c_out = lp_local_search(best_h, tb)
        if c_out < best_c - 1e-12:
            best_h = h_out
            best_c = c_out

    # Stage 6: Structured seeds
    if time_left() > 60:
        n = len(best_h)
        # Deterministic quasi-random binary seeds
        for i in range(4):
            if time_left() < 20:
                break
            bits = np.zeros(n)
            golden = 0.618033988749895
            phase = i * 0.25
            vals = np.array([((j + 1) * golden + phase) % 1.0 for j in range(n)])
            order = np.argsort(vals)
            bits[order[:n//2]] = 1.0
            bits = repair_h(bits)
            # Smooth
            kernel = np.ones(3) / 3.0
            bits_smooth = np.convolve(bits, kernel, mode='same')
            bits_smooth = repair_h(np.clip(bits_smooth, 0, 1))
            tb = min(50, time_left() * 0.2)
            h_out, c_out = lp_local_search(bits_smooth, tb)
            if c_out < best_c - 1e-12:
                best_h = h_out
                best_c = c_out

    # Stage 7: Fine polish - very tight active set
    if time_left() > 20:
        for beam_val in [0.002, 0.001, 0.0005]:
            if time_left() < 5:
                break
            profile = compute_profile(best_h)
            c = np.max(profile)
            active = np.where(profile >= c - beam_val)[0]
            if len(active) == 0:
                active = np.array([np.argmax(profile)])
            G = compute_gradient_matrix(best_h)
            d, alpha = solve_direction_lp(best_h, G, active[:15])
            if d is not None and alpha > 1e-7:
                for step in [0.01, 0.005, 0.003, 0.001, 0.0005]:
                    h_try = repair_h(best_h + step * d)
                    c_try = score_exact(h_try)
                    if c_try < best_c - 1e-13:
                        best_h = h_try
                        best_c = c_try
                        break

    # Stage 8: Final gradient polish
    if time_left() > 15:
        h_out, c_out = gradient_local_search(best_h, min(20, time_left() * 0.5))
        if c_out < best_c - 1e-12:
            best_h = h_out
            best_c = c_out

    # Final validation
    best_h = repair_h(best_h)
    final_c = score_exact(best_h)
    if not np.isfinite(final_c) or final_c > 0.5:
        n_fb = len(best_h)
        best_h = repair_h(np.full(n_fb, 0.5))
        final_c = score_exact(best_h)

    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)) and np.isfinite(eval_result):
            final_c = min(final_c, float(eval_result))
        elif isinstance(eval_result, (tuple, list)) and len(eval_result) >= 1:
            v = eval_result[0]
            if np.isfinite(v):
                final_c = min(final_c, float(v))
    except Exception:
        pass

    return best_h, float(final_c), int(len(best_h))
