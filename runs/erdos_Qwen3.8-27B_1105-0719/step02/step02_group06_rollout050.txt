def eval_score(h):

def project_to_box_sum(x, target):

def corr_lag_grad(h, ell):

def transfer_to_n(h_old, n_new):
    n_old = len(h_old)
    x_old = (np.arange(n_old) + 0.5) * 2.0 / n_old
    x_new = (np.arange(n_new) + 0.5) * 2.0 / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return project_to_box_sum(h_new, n_new / 2.0)

import numpy as np
from scipy.optimize import linprog
import time

def eval_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def corr_raw(h):
    return np.correlate(h, 1.0 - h, mode='full')

def build_sym_h(q):
    m = len(q)
    h = np.empty(2 * m)
    h[:m] = q
    h[m:] = 1.0 - q[::-1]
    return h

def sym_from_h(h):
    n = len(h)
    m = n // 2
    q = np.empty(m)
    for i in range(m):
        q[i] = 0.5 * (h[i] + (1.0 - h[n - 1 - i]))
    return np.clip(q, 0.0, 1.0)

def project_to_box_sum(x, target):
    n = len(x)
    x = np.asarray(x, dtype=float).copy()
    x[~np.isfinite(x)] = 0.5
    x = np.clip(x, 0.0, 1.0)
    if abs(x.sum() - target) < 1e-13:
        return x
    low, high = -2.0, 2.0
    for _ in range(80):
        mid = (low + high) / 2.0
        s = np.clip(x + mid, 0.0, 1.0).sum()
        if s > target:
            high = mid
        else:
            low = mid
    v = np.clip(x + (low + high) / 2.0, 0.0, 1.0)
    r = target - v.sum()
    if abs(r) > 1e-12:
        if r > 0:
            order = np.argsort(v)
            for i in order:
                add = min(r, 1.0 - v[i])
                v[i] += add
                r -= add
                if r < 1e-14:
                    break
        else:
            order = np.argsort(-v)
            for i in order:
                sub = min(-r, v[i])
                v[i] -= sub
                r += sub
                if r > -1e-14:
                    break
    return np.clip(v, 0.0, 1.0)

def corr_lag_grad(h, ell):
    n = len(h)
    d = ell - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -d)
    i_hi = min(n - 1, n - 1 - d)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx] += (1.0 - h[idx + d])
    grad[idx + d] += (-h[idx])
    return grad

def select_active_lags(corrs_scaled, margin_frac=0.03, margin_abs=3e-6, pad=4, max_k=70):
    max_val = corrs_scaled.max()
    threshold = max_val - max(margin_abs, margin_frac * max_val)
    active = np.where(corrs_scaled >= threshold)[0]
    padded = set()
    for k in active:
        for dd in range(-pad, pad + 1):
            if 0 <= k + dd < len(corrs_scaled):
                padded.add(k + dd)
    active_list = sorted(padded)
    if len(active_list) > max_k:
        scored = sorted([(corrs_scaled[k], k) for k in active_list], reverse=True)
        active_list = sorted([k for _, k in scored[:max_k]])
    return active_list

def line_search_1d(h0, direction, n_grid=40):

    a_lo = 0.0

def solve_tangent_lp_sym(q0, active_lags, rho=0.3):
    m = len(q0)
    n = 2 * m
    h0 = build_sym_h(q0)
    corrs0 = corr_raw(h0)

    A_rows = []
    b_vals = []

    for ell in active_lags:
        gh = corr_lag_grad(h0, ell)
        gq = np.empty(m)
        for r in range(m):
            gq[r] = gh[r] - gh[n - 1 - r]
        R_ell = corrs0[ell]
        row = np.zeros(m + 1)
        row[:m] = gq
        row[m] = -1.0
        A_rows.append(row)
        b_vals.append(float(np.dot(gq, q0)) - R_ell)

    A_ub = np.array(A_rows)
    b_ub = np.array(b_vals)

    c = np.zeros(m + 1)
    c[m] = 1.0

    lb = np.maximum(0.0, q0 - rho)
    ub = np.minimum(1.0, q0 + rho)
    bounds = list(zip(lb, ub))
    bounds.append((0.0, 10.0))

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    except Exception:
        return None

    if not res.success:
        return None
    return res.x[:m]

    c[n] = 1.0

    lb = np.maximum(0.0, h0 - rho)
    ub = np.minimum(1.0, h0 + rho)
    bounds = list(zip(lb, ub))
    bounds.append((0.0, 10.0))

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
    except Exception:
        return None

    if not res.success:
        return None
    return res.x[:n]

def tangent_lp_optimize_sym(q0, max_iters=80, rho_init=0.35, time_limit=None):
    m = len(q0)
    n = 2 * m
    q = np.clip(q0.copy(), 0, 1)
    h = build_sym_h(q)
    best_score = eval_score(h)
    best_q = q.copy()
    rho = rho_init
    stagnant = 0

    for it in range(max_iters):
        if time_limit is not None and time.time() > time_limit:
            break

        corrs = corr_raw(h)
        corrs_scaled = corrs * (2.0 / n)

        if n <= 160:
            active = list(range(2 * n - 1))
        else:
            active = select_active_lags(corrs_scaled, margin_frac=0.04, margin_abs=3e-6, pad=5, max_k=90)

        z = solve_tangent_lp_sym(q, active, rho)
        if z is None:
            rho *= 0.7
            stagnant += 1
            if stagnant > 5:
                break
            continue

        direction_q = z - q
        if np.abs(direction_q).max() < 1e-12:
            rho *= 0.7
            stagnant += 1
            if stagnant > 5:
                break
            continue

        h_dir = np.empty(n)
        h_dir[:m] = direction_q
        h_dir[m:] = -direction_q[::-1]

        alpha, score = line_search_1d(h, h_dir)

        if score < best_score - 1e-13:
            q = np.clip(q + alpha * direction_q, 0, 1)
            h = build_sym_h(q)
            best_score = score
            best_q = q.copy()
            rho = min(rho * 1.15, 0.5)
            stagnant = 0
        else:
            rho *= 0.75
            stagnant += 1
            if stagnant > 8:
                break

    return best_q, best_score, build_sym_h(best_q)

def tangent_lp_optimize_full(h0, max_iters=80, rho_init=0.3, time_limit=None):
    n = len(h0)
    h = project_to_box_sum(np.clip(h0.copy(), 0, 1), n / 2.0)
    best_score = eval_score(h)
    best_h = h.copy()
    rho = rho_init
    stagnant = 0

    for it in range(max_iters):
        if time_limit is not None and time.time() > time_limit:
            break

        corrs = corr_raw(h)
        corrs_scaled = corrs * (2.0 / n)

        if n <= 160:
            active = list(range(2 * n - 1))
        else:
            active = select_active_lags(corrs_scaled, margin_frac=0.04, margin_abs=3e-6, pad=5, max_k=80)

        z = solve_tangent_lp_full(h, active, rho)
        if z is None:
            rho *= 0.7
            stagnant += 1
            if stagnant > 5:
                break
            continue

        direction = z - h
        if np.abs(direction).max() < 1e-12:
            rho *= 0.7
            stagnant += 1
            if stagnant > 5:
                break
            continue

        alpha, score = line_search_1d(h, direction)

        if score < best_score - 1e-13:
            h_new = project_to_box_sum(h + alpha * direction, n / 2.0)
            best_score = score
            best_h = h_new.copy()
            h = h_new
            rho = min(rho * 1.15, 0.45)
            stagnant = 0
        else:
            rho *= 0.75
            stagnant += 1
            if stagnant > 8:
                break

    return best_h, best_score

def pair_transfer_search(h, max_pairs=600, n_eval=25, time_limit=None):
    n = len(h)
    best_h = h.copy()
    best_score = eval_score(h)

    corrs = corr_raw(h) * (2.0 / n)
    max_val = corrs.max()
    active = np.where(corrs >= max_val - max(0.03, 0.02 * max_val))[0]
    if len(active) == 0:
        active = np.argsort(-corrs)[:10]

    grad_sum = np.zeros(n)
    for ell in active[:25]:
        grad_sum += corr_lag_grad(h, ell)

    idx_top = np.argsort(-np.abs(grad_sum))[:min(n, 50)]

    pairs_tried = 0
    for i in idx_top:
        if pairs_tried >= max_pairs:
            break
        if time_limit is not None and time.time() > time_limit:
            break
        for j in idx_top:
            if i == j:
                continue
            if pairs_tried >= max_pairs:
                break
            pairs_tried += 1

            if h[i] <= 1e-12 and h[j] >= 1.0 - 1e-12:
                continue

            a_lo = max(1e-4, -h[i], h[j] - 1.0 + 1e-12)
            a_hi = min(1.0 - h[i] - 1e-12, h[j] - 1e-12)
            if a_hi <= a_lo:
                continue

            alphas = np.linspace(a_lo, a_hi, n_eval)
            for a in alphas:
                h[i] += a
                h[j] -= a
                if h[i] < -1e-10 or h[i] > 1.0 + 1e-10 or h[j] < -1e-10 or h[j] > 1.0 + 1e-10:
                    h[i] -= a
                    h[j] += a
                    continue
                s = eval_score(h)
                if s < best_score:
                    best_score = s
                    best_h = h.copy()
                h[i] -= a
                h[j] += a

    return best_h, best_score

def coordinate_descent(h, max_rounds=20, time_limit=None):
    """Try moving each coordinate by a small amount (compensated by nearest)."""
    n = len(h)
    best_score = eval_score(h)
    h_curr = h.copy()

    for rnd in range(max_rounds):
        if time_limit is not None and time.time() > time_limit:
            break
        improved = False
        for i in range(n):
            if time_limit is not None and time.time() > time_limit:
                break
            # Try to move h[i] up or down, compensating with a "free" coordinate
            best_move = 0.0
            best_s = best_score
            # Find a good compensation index
            comp = (i + n // 2) % n
            if comp == i:
                comp = (i + 1) % n

            for sign in [1.0, -1.0]:
                # h[i] sign*d, h[comp] -sign*d
                if sign > 0:
                    d_max = min(1.0 - h_curr[i], h_curr[comp])
                else:
                    d_max = min(h_curr[i], 1.0 - h_curr[comp])
                if d_max < 1e-6:
                    continue
                for d in [0.1, 0.15, 0.2, 0.3, 0.4, d_max * 0.5, d_max * 0.8]:
                    if d < 1e-6 or d > d_max:
                        continue
                    h_curr[i] += sign * d
                    h_curr[comp] -= sign * d
                    if h_curr[i] < -1e-10 or h_curr[i] > 1.0 + 1e-10:
                        h_curr[i] -= sign * d
                        h_curr[comp] += sign * d
                        continue
                    if h_curr[comp] < -1e-10 or h_curr[comp] > 1.0 + 1e-10:
                        h_curr[i] -= sign * d
                        h_curr[comp] += sign * d
                        continue
                    s = eval_score(h_curr)
                    if s < best_s:
                        best_s = s
                        best_move = sign * d
                    h_curr[i] -= sign * d
                    h_curr[comp] += sign * d

            if best_move != 0.0:
                h_curr[i] += best_move
                h_curr[comp] -= best_move
                h_curr = np.clip(h_curr, 0, 1)
                best_score = best_s
                improved = True
        if not improved:
            break

    return h_curr, best_score

def transfer_to_n(h_old, n_new):
    n_old = len(h_old)
    x_old = (np.arange(n_old) + 0.5) * 2.0 / n_old
    x_new = (np.arange(n_new) + 0.5) * 2.0 / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return project_to_box_sum(h_new, n_new / 2.0)

def make_fourier_start(n, seed=0):
    rng = np.random.RandomState(seed)
    x = np.arange(n) / n
    h = 0.5 * np.ones(n)
    for k in range(1, 7):
        amp = rng.uniform(0.05, 0.35)
        phase = rng.uniform(0, 2 * np.pi)
        h += amp * np.sin(2 * np.pi * k * x + phase)
    return project_to_box_sum(h, n / 2.0)

def make_block_start(n, n_blocks, seed=0):
    rng = np.random.RandomState(seed)
    vals = rng.uniform(0.05, 0.95, n_blocks)
    h = np.repeat(vals, n // n_blocks + 1)[:n]
    return project_to_box_sum(h, n / 2.0)

def make_trig_start(n, freq1, freq2, seed=0):
    x = (np.arange(n) + 0.5) / n
    h = 0.5 + 0.3 * np.sin(2 * np.pi * freq1 * x) + 0.15 * np.sin(2 * np.pi * freq2 * x)
    return project_to_box_sum(h, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def rem():
        return budget_s - (time.time() - t0)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Get parent
    try:
        h_parent = np.array(initial_h_values, dtype=float).copy().reshape(-1)
    except (NameError, AttributeError):
        h_parent = None

    if h_parent is not None and len(h_parent) > 10:
        n_p = len(h_parent)
        h_parent = project_to_box_sum(h_parent, n_p / 2.0)
        best_h = h_parent.copy()
        best_c5 = eval_score(h_parent)
        best_n = n_p
    else:
        n_p = 126
        best_h = make_trig_start(n_p, 3, 7, seed)
        best_c5 = eval_score(best_h)
        best_n = n_p

    n_cur = len(best_h)

    # Phase 1: Symmetric tangent LP on parent
    if n_cur % 2 == 0 and rem() > 180:
        q_init = sym_from_h(best_h)
        tl = time.time() + min(150, rem() * 0.25)
        q_opt, c5_s, h_s = tangent_lp_optimize_sym(q_init, max_iters=60, rho_init=0.35, time_limit=tl)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()
            best_n = len(h_s)

    # Phase 2: Pair transfer
    if rem() > 120:
        tl = time.time() + min(100, rem() * 0.15)
        h_pt, c5_pt = pair_transfer_search(best_h, max_pairs=800, n_eval=25, time_limit=tl)
        if c5_pt < best_c5:
            best_c5 = c5_pt
            best_h = h_pt.copy()
            best_n = len(h_pt)

    # Phase 3: Full tangent LP
    if rem() > 150:
        tl = time.time() + min(180, rem() * 0.25)
        h_opt, c5_f = tangent_lp_optimize_full(best_h, max_iters=60, rho_init=0.3, time_limit=tl)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_opt.copy()
            best_n = len(h_opt)

    # Phase 4: Another round of pair transfer + coordinate descent
    if rem() > 100:
        tl = time.time() + min(80, rem() * 0.2)
        h_pt, c5_pt = pair_transfer_search(best_h, max_pairs=1000, n_eval=30, time_limit=tl)
        if c5_pt < best_c5:
            best_c5 = c5_pt
            best_h = h_pt.copy()
            best_n = len(h_pt)

    # Phase 5: Coordinate descent polish
    if rem() > 60:
        tl = time.time() + min(50, rem() * 0.2)
        h_cd, c5_cd = coordinate_descent(best_h, max_rounds=15, time_limit=tl)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
            best_n = len(h_cd)

    # Phase 6: Multi-resolution
    n_base = len(best_h)
    n_candidates = sorted(set([n_base - 8, n_base - 4, n_base - 2, n_base, n_base + 2, n_base + 4, n_base + 8]))
    n_candidates = [nc for nc in n_candidates if 50 <= nc <= 200]

    for n_try in n_candidates:
        if rem() < 60:
            break
        if n_try == len(best_h):
            continue
        h_try = transfer_to_n(best_h, n_try)

        if n_try % 2 == 0 and rem() > 90:
            q_try = sym_from_h(h_try)
            tl = time.time() + min(60, rem() * 0.35)
            q_o, c5_o, h_o = tangent_lp_optimize_sym(q_try, max_iters=40, rho_init=0.35, time_limit=tl)
            if c5_o < best_c5:
                best_c5 = c5_o
                best_h = h_o.copy()
                best_n = len(h_o)
                # Also try pair transfer at this resolution
                if rem() > 40:
                    tl2 = time.time() + min(30, rem() * 0.3)
                    h_pt, c5_pt = pair_transfer_search(h_o, max_pairs=500, n_eval=20, time_limit=tl2)
                    if c5_pt < best_c5:
                        best_c5 = c5_pt
                        best_h = h_pt.copy()
                        best_n = len(h_pt)

        if rem() > 50 and n_try % 2 != 0:
            tl = time.time() + min(40, rem() * 0.4)
            h_o, c5_o = tangent_lp_optimize_full(h_try, max_iters=30, rho_init=0.25, time_limit=tl)
            if c5_o < best_c5:
                best_c5 = c5_o
                best_h = h_o.copy()
                best_n = len(h_o)

    # Phase 7: Diverse restarts
    n_restarts = min(12, max(4, int(rem() / 70)))
    start_strategies = []
    for i in range(n_restarts):
        n_r = len(best_h) + rng.choice([-8, -6, -4, -2, 2, 4, 6, 8])
        n_r = min(190, max(50, n_r))
        if n_r % 2 != 0:
            n_r += 1
        strat = i % 4
        start_strategies.append((n_r, strat))

    for (n_r, strat) in start_strategies:
        if rem() < 55:
            break

        if strat == 0:
            h_r = make_fourier_start(n_r, seed + 100)
        elif strat == 1:
            h_r = make_block_start(n_r, n_blocks=rng.choice([4, 5, 6, 8]), seed=seed + 200)
        elif strat == 2:
            f1 = rng.choice([2, 3, 4, 5])
            f2 = rng.choice([5, 7, 9, 11])
            h_r = make_trig_start(n_r, f1, f2, seed)
        else:
            h_r = transfer_to_n(best_h, n_r)
            h_r = np.clip(h_r + 0.15 * rng.randn(n_r), 0, 1)
            h_r = project_to_box_sum(h_r, n_r / 2.0)

        if n_r % 2 == 0 and rem() > 80:
            q_r = sym_from_h(h_r)
            tl = time.time() + min(55, rem() * 0.5)
            q_o, c5_o, h_o = tangent_lp_optimize_sym(q_r, max_iters=50, rho_init=0.35, time_limit=tl)
            if c5_o < best_c5:
                best_c5 = c5_o
                best_h = h_o.copy()
                best_n = len(h_o)
            elif c5_o < best_c5 + 0.003 and rem() > 40:
                # Close enough, try pair transfer
                tl2 = time.time() + min(30, rem() * 0.4)
                h_pt, c5_pt = pair_transfer_search(h_o, max_pairs=600, n_eval=25, time_limit=tl2)
                if c5_pt < best_c5:
                    best_c5 = c5_pt
                    best_h = h_pt.copy()
                    best_n = len(h_pt)

        if rem() > 40:
            tl = time.time() + min(35, rem() * 0.5)
            h_o, c5_o = tangent_lp_optimize_full(h_r, max_iters=35, rho_init=0.25, time_limit=tl)
            if c5_o < best_c5:
                best_c5 = c5_o
                best_h = h_o.copy()
                best_n = len(h_o)

    # Phase 8: Final intensive polish on best
    if rem() > 25 and best_h is not None:
        n_fin = len(best_h)

        # Symmetric LP with tight trust region
        if n_fin % 2 == 0 and rem() > 30:
            q_fin = sym_from_h(best_h)
            tl = time.time() + min(20, rem() * 0.4)
            q_o, c5_o, h_o = tangent_lp_optimize_sym(q_fin, max_iters=30, rho_init=0.15, time_limit=tl)
            if c5_o < best_c5:
                best_c5 = c5_o
                best_h = h_o.copy()
                best_n = len(h_o)

        # Pair transfer
        if rem() > 20:
            tl = time.time() + min(15, rem() * 0.5)
            h_pt, c5_pt = pair_transfer_search(best_h, max_pairs=800, n_eval=30, time_limit=tl)
            if c5_pt < best_c5:
                best_c5 = c5_pt
                best_h = h_pt.copy()
                best_n = len(h_pt)

    # Phase 9: One final full LP pass
    if rem() > 15 and best_h is not None:
        n_fin = len(best_h)
        tl = time.time() + min(12, rem() * 0.6)
        h_o, c5_o = tangent_lp_optimize_full(best_h, max_iters=20, rho_init=0.15, time_limit=tl)
        if c5_o < best_c5:
            best_c5 = c5_o
            best_h = h_o.copy()
            best_n = len(h_o)

    # Final validation
    if best_h is None:
        m_fb = 63
        x_fb = np.arange(m_fb) / m_fb
        q_fb = np.clip(0.5 + 0.25 * np.sin(6 * np.pi * x_fb) + 0.1 * np.sin(14 * np.pi * x_fb), 0, 1)
        best_h = build_sym_h(q_fb)

    best_h = np.asarray(best_h, dtype=float).reshape(-1)
    n_out = len(best_h)
    best_h = project_to_box_sum(best_h, n_out / 2.0)
    best_c5 = eval_score(best_h)

    return best_h, best_c5, n_out
