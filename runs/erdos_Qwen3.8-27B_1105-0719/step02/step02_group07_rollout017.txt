import numpy as np
import time

def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-10:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def corr_vals(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def make_sym_h(q, n):
    m = n // 2
    q = np.clip(q, 0.0, 1.0)
    h = np.empty(n)
    h[:m] = q
    h[m:] = 1.0 - q[::-1]
    return h

def make_q_from_h(h, n):
    m = n // 2
    q = 0.5 * (h[:m] + 1.0 - h[n-1:n-1-m:-1])
    return np.clip(q, 0.0, 1.0)

def cd_step_sym_vec(q, i, n, h, corrs):
    """Vectorized 1D CD step for symmetric parameterization."""
    j = n - 1 - i
    d_vals = np.arange(2*n-1) - (n-1)
    
    # Compute masks
    v1 = (i + d_vals >= 0) & (i + d_vals < n)
    v2 = (i - d_vals >= 0) & (i - d_vals < n)
    v3 = (j + d_vals >= 0) & (j + d_vals < n)
    v4 = (j - d_vals >= 0) & (j - d_vals < n)
    nz = (d_vals != 0)
    
    old = np.zeros(2*n-1)
    n_const = np.zeros(2*n-1)
    n_slope = np.zeros(2*n-1)
    n_quad = np.zeros(2*n-1)
    
    # d=0
    idx0 = n - 1  # d_vals[n-1] = 0
    old[idx0] = h[i]*(1-h[i]) + h[j]*(1-h[j])
    n_slope[idx0] = 2.0
    n_quad[idx0] = -2.0
    
    # d≠0
    v1n = v1 & nz
    v2n = v2 & nz
    v3n = v3 & nz
    v4n = v4 & nz
    
    if v1n.any():
        old[v1n] += h[i] * (1.0 - h[i + d_vals[v1n]])
        n_slope[v1n] += 1.0 - h[i + d_vals[v1n]]
    if v2n.any():
        old[v2n] += h[i - d_vals[v2n]] * (1.0 - h[i])
        n_const[v2n] += h[i - d_vals[v2n]]
        n_slope[v2n] -= h[i - d_vals[v2n]]
    if v3n.any():
        old[v3n] += h[j] * (1.0 - h[j + d_vals[v3n]])
        n_const[v3n] += 1.0 - h[j + d_vals[v3n]]
        n_slope[v3n] -= 1.0 - h[j + d_vals[v3n]]
    if v4n.any():
        old[v4n] += h[j - d_vals[v4n]] * (1.0 - h[j])
        n_slope[v4n] += h[j - d_vals[v4n]]
    
    base = corrs - old + n_const
    slp = n_slope
    qd = n_quad

def cd_sweep_sym(q, n, n_sweeps=20):
    m = n // 2
    for sweep in range(n_sweeps):
        h = make_sym_h(q, n)
        corrs = corr_vals(h)
        max_change = 0.0
        for i in range(m):
            new_t = cd_step_sym_vec(q, i, n, h, corrs)
            delta = new_t - q[i]
            max_change = max(max_change, abs(delta))
            q[i] = new_t
        if max_change < 1e-10:
            break
    return q

def golden_ratio_h(n, alpha=None):
    if alpha is None:
        alpha = (1.0 + np.sqrt(5.0)) / 2.0
    x = np.arange(n) * (2.0 / n)
    h = (np.mod(x * alpha, 1.0) < 0.5).astype(float)
    return project_feasible(h, n / 2.0)

def random_binary_h(n, seed):
    rng = np.random.RandomState(seed)
    h = (rng.rand(n) < 0.5).astype(float)
    return project_feasible(h, n / 2.0)

def subgrad_step(h, n, step_size=0.01, tol=0.01):
    """Subgradient step on the max correlation."""
    corrs = corr_vals(h)
    max_c = corrs.max()
    active = corrs > max_c - tol
    max_idx = np.argmax(corrs)
    
    # Subgradient of c_d for the active d's
    d_vals = np.arange(2*n-1) - (n-1)
    sg = np.zeros(n)
    
    for k_idx in np.where(active)[0]:
        d = d_vals[k_idx]
        if d == 0:
            sg[:] += (1.0 - 2.0 * h)
        else:
            # c_d = sum_{j: valid} h[j](1-h[j+d])
            # d/dh[i] of c_d:
            #   if 0<=i+d<n: (1-h[i+d])
            #   if 0<=i-d<n: -h[i-d]
