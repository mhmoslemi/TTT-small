For even n = 2m:
- h[i] + h[n-1-i] = 1 for i = 0, ..., n-1

def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s >= target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h


def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def c5_of_h(h, dx):
    return float(compute_corrs(h).max() * dx)


def grad_corr_k(h, k, n):
    offset = k - (n - 1)
    grad = np.zeros(n)
    if offset == 0:
        grad[:] = 1.0 - 2.0 * h
        return grad
    i_lo = max(0, -offset)
    i_hi = min(n - 1, n - 1 - offset)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx + offset] += (1.0 - h[idx])
    grad[idx] += (-h[idx + offset])
    return grad


class SmoothMaxProb:
    def __init__(self, dx, target_sum, beta):
        self.dx = dx
        self.target_sum = target_sum
        self.beta = beta

    def f(self, x):
        corrs = compute_corrs(x) * self.dx
        return float(logsumexp(self.beta * corrs) / self.beta)

    def g(self, x):
        n = len(x)
        corrs = compute_corrs(x) * self.dx
        shifted = corrs - corrs.max()
        w = np.exp(self.beta * shifted)
        ws = w.sum()
        if ws < 1e-30:
            w = np.ones(len(corrs)) / len(corrs)
        else:
            w /= ws
        grad = np.zeros(n)
        for k in range(len(corrs)):
            if w[k] < 1e-12:
                continue
            grad += w[k] * grad_corr_k(x, k, n)
        return grad * self.dx

        return x.sum() - self.target_sum

        return np.ones(len(x))

    # Phase 3: Binary push
    if remaining() > 80:
        h_bp, c5_bp = binary_push(best_h, dx, target_sum)
        if c5_bp < best_c5:
            best_c5 = c5_bp
            best_h = h_bp
            # Polish after binary push
            if remaining() > 30:
                h_pol, c5_pol = solve_active_set(best_h, dx, target_sum,
                                                  np.argsort(compute_corrs(best_h))[::-1][:8], 0.03, 300)
                if c5_pol < best_c5:
                    best_c5 = c5_pol
                    best_h = h_pol

    # Phase 4: Try complement-symmetric starting points
    if remaining() > 50:
        h_comp = make_complement_start(best_h, n)
        h_comp = project_feasible(h_comp, target_sum)
        c5_comp = c5_of_h(h_comp, dx)
        if c5_comp < best_c5 + 0.05:
            h_new, c5_new = solve_active_set(h_comp, dx, target_sum,
                                              np.argsort(compute_corrs(h_comp))[::-1][:8], 0.05, 300)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new

    # Phase 5: Random binary starts
    if remaining() > 60:
        for trial in range(8):
            if remaining() < 15:
                break
            h_bin = make_binary_start(n, rng)
            h_opt, c5_opt = solve_smooth(h_bin, dx, target_sum, 200, 300)
            if c5_opt < best_c5 + 0.02:
                h_opt2, c5_opt2 = solve_active_set(h_opt, dx, target_sum,
                                                    np.argsort(compute_corrs(h_opt))[::-1][:8], 0.05, 250)
                if c5_opt2 < best_c5:
                    best_c5 = c5_opt2
                    best_h = h_opt2

    # Phase 6: Dimension exploration
    if remaining() > 100:
        n_cur = len(best_h)
        dx_cur = 2.0 / n_cur
        n_tries = sorted(set([max(40, n_cur - 20), n_cur - 10, n_cur, n_cur + 10,
                              n_cur + 20, max(40, n_cur - 30), n_cur + 30]))
        for n_try in n_tries:
            if remaining() < 30:
                break
            if n_try < 30 or n_try > 200:
                continue
            dx_try = 2.0 / n_try
            ts_try = n_try / 2.0
            x_old = (np.arange(len(best_h)) + 0.5) * (2.0 / len(best_h))
            x_new = (np.arange(n_try) + 0.5) * (2.0 / n_try)
            h_interp = np.interp(x_new, x_old, best_h)
            h_interp = project_feasible(h_interp, ts_try)
            h_opt, c5_opt = solve_smooth(h_interp, dx_try, ts_try, 200, 300)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
            if c5_opt < best_c5 + 0.01:
                if remaining() > 25:
                    top_l = np.argsort(compute_corrs(h_opt))[::-1][:8]
                    h_opt2, c5_opt2 = solve_active_set(h_opt, dx_try, ts_try, top_l, 0.05, 300)
                    if c5_opt2 < best_c5:
                        best_c5 = c5_opt2
                        best_h = h_opt2

    # Phase 7: Final intensive polish
    if remaining() > 30:
        n_b = len(best_h)
        dx_b = 2.0 / n_b
        ts_b = n_b / 2.0
        corrs = compute_corrs(best_h)
        for m_val in [6, 10, 5]:
            if remaining() < 15:
                break
            top_l = np.argsort(corrs)[::-1][:m_val]
            h_new, c5_new = solve_active_set(best_h, dx_b, ts_b, top_l, 0.01, 500)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new
                corrs = compute_corrs(best_h)

    if remaining() > 20:
        h_sg, c5_sg = active_subgrad(best_h, dx_b, ts_b, 6, 5000, 0.03)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg

    if remaining() > 15:
        h_new, c5_new = solve_smooth(best_h, 2.0 / len(best_h), len(best_h) / 2.0, 5000, 600)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

    # Final verification
    best_h = project_feasible(best_h, len(best_h) / 2.0)
    final_c5 = c5_of_h(best_h, 2.0 / len(best_h))

    return best_h, final_c5, len(best_h)
