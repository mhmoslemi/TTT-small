from scipy.optimize import minimize

def build_h(q, n):
    m = n // 2
    h = np.empty(n)
    h[:m] = q
    h[m:] = 1.0 - q[::-1]

    return 0.5 * (h[:m] + 1.0 - h[n - 1 - np.arange(m)][::-1])

    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

    g = np.zeros(n)
    if d >= 0:

        g[d:] += (1.0 - h[:nd])

    else:

    p = exp_s / Z
    phi = M + tau * np.log(Z)
    gh = np.zeros(n)
    for j in range(2 * n - 1):
        w = p[j]
        if w < 1e-15:
            continue
        d = j - (n - 1)
        if d >= 0:
            nd = n - d
            gh[d:] += w * (1.0 - h[:nd])
            gh[:nd] -= w * h[d:]
        else:
            dp = -d
            nd = n - dp
            gh[:nd] += w * (1.0 - h[dp:])
            gh[dp:] -= w * h[:nd]
    gh *= dx
    gq = gh[:m] - gh[n - 1 - np.arange(m)][::-1]
    return phi, gq


def full_lse_obj_grad(h, n, tau, lam):
    dx = 2.0 / n
    raw = np.correlate(h, 1.0 - h, mode='full')
    C = raw * dx
    M = C.max()
    scores = (C - M) / tau
    scores = np.clip(scores, -50.0, 50.0)
    exp_s = np.exp(scores)
    Z = exp_s.sum()
    p = exp_s / Z
    phi = M + tau * np.log(Z)
    sdev = h.sum() - n / 2.0
    phi += 0.5 * lam * sdev * sdev
    gh = np.zeros(n)
    for j in range(2 * n - 1):
        w = p[j]
        if w < 1e-15:
            continue
        d = j - (n - 1)
        if d >= 0:
            nd = n - d
            gh[d:] += w * (1.0 - h[:nd])
            gh[:nd] -= w * h[d:]
        else:
            dp = -d
            nd = n - dp
            gh[:nd] += w * (1.0 - h[dp:])
            gh[dp:] -= w * h[:nd]
    gh *= dx
    gh += lam * sdev
    return phi, gh


def project_to_sum(h, target):
    h = h.copy()
    for _ in range(50):
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
        diff = target - s
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        if not mask.any():
            h += diff / len(h)
        else:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)


def interp_h(h, n_new):
    n_old = len(h)
    if n_new == n_old:
        return h.copy()
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h)
    return project_to_sum(h_new, n_new / 2.0)


def optimize_anti(q0, n, tau_list, max_iter, t_start, t_limit):
    m = n // 2
    bounds = [(0.0, 1.0)] * m
    q = np.clip(q0.copy(), 0.0, 1.0)
    best_q = q0.copy()
    best_c5 = eval_c5(build_h(q0, n))

    for tau in tau_list:
        if time.time() > t_limit:
            break
        elapsed = time.time() - t_start
        remaining = t_limit - time.time()
        if remaining < 2.0:
            break
        mi = min(max_iter, max(50, int(remaining * 80)))

        def fun_g(x):
            phi, g = lse_obj_grad_q(x, n, tau)
            return phi, g

        try:
            res = minimize(fun_g, q, jac=True, method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': mi, 'gtol': 1e-9, 'iprint': -1})
            q = np.clip(res.x, 0.0, 1.0)
        except Exception:
            pass
        h = build_h(q, n)
        c5 = eval_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_q = q.copy()
    return best_q, best_c5


def optimize_full(h0, n, tau_list, max_iter, t_start, t_limit, lam=500.0):
    bounds = [(0.0, 1.0)] * n
    h = np.clip(h0.copy(), 0.0, 1.0)
    best_h = h0.copy()
    best_c5 = eval_c5(h0)

    for tau in tau_list:
        if time.time() > t_limit:
            break
        remaining = t_limit - time.time()
        if remaining < 2.0:
            break
        mi = min(max_iter, max(50, int(remaining * 60)))

        def fun_g(x):
            phi, g = full_lse_obj_grad(x, n, tau, lam)
            return phi, g

        try:
            res = minimize(fun_g, h, jac=True, method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': mi, 'gtol': 1e-9, 'iprint': -1})
            h = np.clip(res.x, 0.0, 1.0)
            h = project_to_sum(h, n / 2.0)
        except Exception:
            pass
        c5 = eval_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5


def make_structured_q(m, rng):
    t = np.linspace(0.0, 1.0, m)
    k1 = rng.randint(1, 6)
    k2 = rng.randint(1, 6)
    p1 = rng.uniform(0, 2 * np.pi)
    p2 = rng.uniform(0, 2 * np.pi)
    a1 = rng.uniform(0.1, 0.4)
    a2 = rng.uniform(0.05, 0.25)
    q = 0.5 + a1 * np.sin(2 * np.pi * k1 * t + p1) + a2 * np.sin(2 * np.pi * k2 * t + p2)
    return np.clip(q, 0.0, 1.0)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    reserve = 20.0

    best_h = None
    best_c5 = float('inf')

    try:
        h_init = np.array(initial_h_values, dtype=float).reshape(-1)
        if h_init.size > 0 and np.all(np.isfinite(h_init)):
            n_init = len(h_init)
            h_init = np.clip(h_init, 0.0, 1.0)
            h_init = project_to_sum(h_init, n_init / 2.0)
            best_h = h_init.copy()
            best_c5 = eval_c5(h_init)
    except Exception:
        pass

    if best_h is None:
        n = 126
        q = 0.5 + 0.2 * np.sin(2 * np.pi * 2.0 * np.linspace(0, 1, n // 2))
        best_h = build_h(np.clip(q, 0, 1), n)
        best_c5 = eval_c5(best_h)

    n_base = len(best_h)

    # Phase 1: Anti-symmetric optimization on base n
    if n_base % 2 == 0:
        q0 = q_from_h(best_h, n_base)
        tau_main = [0.08, 0.04, 0.02, 0.01, 0.005, 0.0025, 0.0012]
        t_lim = t0 + budget_s - reserve - 120.0
        q_opt, c5_opt = optimize_anti(q0, n_base, tau_main, 400, t0, t_lim)
        h_opt = build_h(q_opt, n_base)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 2: Fine-tune with smaller tau on base n
    if n_base % 2 == 0 and time.time() < t0 + budget_s - reserve - 30:
        q0 = q_from_h(best_h, n_base)
        tau_fine = [0.0006, 0.0003, 0.00015]
        t_lim = t0 + budget_s - reserve - 30.0
        q_opt, c5_opt = optimize_anti(q0, n_base, tau_fine, 300, t0, t_lim)
        h_opt = build_h(q_opt, n_base)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 3: Try different n values
    n_candidates = []
    for delta in [-6, -4, -2, 2, 4, 6, 10, 14]:
        nc = n_base + delta
        if nc >= 40 and nc % 2 == 0:
            n_candidates.append(nc)
    n_candidates = sorted(set(n_candidates))

    for n_try in n_candidates:
        if time.time() > t0 + budget_s - reserve - 15:
            break
        h_try = interp_h(best_h, n_try)
        q_try = q_from_h(h_try, n_try)
        tau_sched = [0.04, 0.02, 0.01, 0.005, 0.002]
        t_lim = min(t0 + budget_s - reserve - 10, time.time() + 60.0)
        q_opt, c5_opt = optimize_anti(q_try, n_try, tau_sched, 250, t0, t_lim)
        h_opt = build_h(q_opt, n_try)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 4: Full (non-antisymmetric) optimization
    if time.time() < t0 + budget_s - reserve - 20:
        n_cur = len(best_h)
        tau_full = [0.03, 0.015, 0.008, 0.004]
        t_lim = t0 + budget_s - reserve - 15.0
        h_opt, c5_opt = optimize_full(best_h, n_cur, tau_full, 200, t0, t_lim, lam=1000.0)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 5: Random restarts
    n_best = len(best_h)
    if n_best % 2 != 0:
        n_best -= 1
    max_restarts = min(30, max(3, int((t0 + budget_s - reserve - 10 - time.time()) / 8.0)))
    for trial in range(max_restarts):
        if time.time() > t0 + budget_s - reserve - 5:
            break
        q_rand = make_structured_q(n_best // 2, rng)
        tau_short = [0.05, 0.02, 0.01, 0.004]
        t_lim = min(t0 + budget_s - reserve - 5, time.time() + 15.0)
        q_opt, c5_opt = optimize_anti(q_rand, n_best, tau_short, 150, t0, t_lim)
        h_opt = build_h(q_opt, n_best)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 6: Final deep polish on best
    n_fin = len(best_h)
    if time.time() < t0 + budget_s - reserve + 5 and n_fin % 2 == 0:
        q_fin = q_from_h(best_h, n_fin)
        tau_final = [0.001, 0.0005, 0.0002, 0.0001]
        t_lim = t0 + budget_s - reserve + 5.0
        q_opt, c5_opt = optimize_anti(q_fin, n_fin, tau_final, 300, t0, t_lim)
        h_opt = build_h(q_opt, n_fin)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_sum(best_h, len(best_h) / 2.0)
    best_c5 = eval_c5(best_h)

    return best_h, best_c5, len(best_h)
