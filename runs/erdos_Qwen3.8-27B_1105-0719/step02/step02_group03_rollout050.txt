import time

def project_feasible(x, target_sum):

    lo, hi = -2.0, 2.0

        h = np.clip(x - mid, 0.0, 1.0)

        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
        else:
            h += diff / n
    return np.clip(h, 0.0, 1.0)

def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def c5_of_h(h, dx):
    return compute_corrs(h).max() * dx

def binary_to_binary(h0, n, target_count):
    """Convert a real-valued h to binary with exactly target_count ones."""
    h_bin = np.zeros(n)
    sorted_idx = np.argsort(-h0)
    h_bin[sorted_idx[:target_count]] = 1.0
    return h_bin

def compute_delta_swap(h, i, j, n):
    """O(n) computation of delta in correlations for swap h[i]:1->0, h[j]:0->1."""
    ms = np.arange(-(n-1), n)
    delta = np.zeros(2*n-1)
    
    vi1 = (i - ms >= 0) & (i - ms < n)
    delta[vi1] -= (1.0 - h[i - ms[vi1]])
    
    vi2 = (i + ms >= 0) & (i + ms < n)
    delta[vi2] += h[i + ms[vi2]]
    
    vj1 = (j - ms >= 0) & (j - ms < n)
    delta[vj1] += (1.0 - h[j - ms[vj1]])
    
    vj2 = (j + ms >= 0) & (j + ms < n)
    delta[vj2] -= h[j + ms[vj2]]
    
    return delta

def tabu_sa_search(h0, dx, rng, max_iters=200000, tabu_len=30, 
                   t0_temp=0.03, t_min=0.001, patience=5000):
    """Tabu search + simulated annealing for binary h."""
    n = len(h0)
    target_count = int(round(n / 2))
    
    h = binary_to_binary(h0, n, target_count)
    corrs = compute_corrs(h)
    best_c5 = corrs.max() * dx
    best_h = h.copy()
    
    ones = set(np.where(h == 1.0)[0].tolist())
    zeros = set(np.where(h == 0.0)[0].tolist())
    
    tabu_list = {}  # (i,j) -> iteration when it becomes untabu
    best_iter = 0
    no_improve = 0
    temp = t0_temp
    
    for it in range(max_iters):
        if no_improve > patience:
            break
            
        # Pick a random swap
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = list(ones)[rng.randint(len(ones))]
        j = list(zeros)[rng.randint(len(zeros))]
        
        # Check tabu
        key = (min(i,j), max(i,j))
        is_tabu = key in tabu_list and tabu_list[key] > it
        
        delta = compute_delta_swap(h, i, j, n)
        new_corrs = corrs + delta
        new_max = new_corrs.max()
        new_c5 = new_max * dx
        
        accept = False
        if new_c5 <= best_c5 + 1e-15:
            accept = True
        elif not is_tabu:
            delta_val = new_c5 - corrs.max() * dx
            if delta_val < 0:
                accept = True
            elif temp > t_min:
                accept = rng.random() < np.exp(-delta_val / temp)
        
        if accept:
            h[i], h[j] = 0.0, 1.0
            ones.discard(i)
            ones.add(j)
            zeros.discard(j)
            zeros.add(i)
            corrs = new_corrs
            tabu_list[key] = it + tabu_len
            
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
                best_iter = it
                no_improve = 0
            else:
                no_improve += 1
        
        else:
            no_improve += 1
        
        if it % 1000 == 0:
            temp *= 0.999

```python
def steepest_descent_sweep(h, corrs, n, max_sweeps=50):
    """Try all swaps, pick the best one per sweep."""
    best_val = corrs.max()
    best_h = h.copy()
    best_corrs = corrs.copy()
    
    for sweep in range(max_sweeps):
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 1.0)[0]  # BUG: should be h == 0
        zeros = np.where(h == 0.0)[0]
        
        best_swap_val = best_val
        best_swap = None
        best_new_corrs = None
        
        ms = np.arange(-(n-1), n)
        n_corr = 2*n - 1
        
        for i in ones:
            # Effect of removing position i (h[i]: 1->0)
            base_delta = np.zeros(n_corr)
            vi1 = (i - ms >= 0) & (i - ms < n)
            base_delta[vi1] -= (1.0 - h[i - ms[vi1]])
            vi2 = (i + ms >= 0) & (i + ms < n)
            base_delta[vi2] += h[i + ms[vi2]]
            
            # Effect of adding position j (h[j]: 0->1), vectorized over j
            # For each j: 
            #   vj1: (j - ms >= 0) & (j - ms < n) -> delta += (1 - h[j-ms])
            #   vj2: (j + ms >= 0) & (j + ms < n) -> delta -= h[j+ms]
            
            for j in zeros:
                extra = np.zeros(n_corr)
                vj1 = (j - ms >= 0) & (j - ms < n)
                extra[vj1] += (1.0 - h[j - ms[vj1]])
                vj2 = (j + ms >= 0) & (j + ms < n)
                extra[vj2] -= h[j + ms[vj2]]
                
                new_corrs = corrs + base_delta + extra
                new_max = new_corrs.max()
                
                if new_max < best_swap_val - 1e-14:
                    best_swap_val = new_max
                    best_swap = (i, j)
                    best_new_corrs = new_corrs
        
        if best_swap is None:
            break
        
        i, j = best_swap
        h[i] = 0.0
        h[j] = 1.0
        corrs = best_new_corrs
        if best_swap_val < best_val:
            best_val = best_swap_val
            best_h = h.copy()
    
    return best_h, best_corrs
```
