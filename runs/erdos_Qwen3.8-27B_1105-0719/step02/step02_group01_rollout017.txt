Actually, let me look at this more carefully. For arrays a and b of length n:
np.correlate(a, b, 'full')[k] = sum_i a[i] * b[i - (k - (n-1))]

where the sum is over i where both indices are valid (0 to n-1).

where shift = k - (n-1)

Let j = i - shift, so i = j + shift:
result[k] = sum_j h[j+shift] * (1 - h[j])
where j ranges over values where both j and j+shift are in [0, n-1].

Gradient with respect to h[m]:
- If m appears as j+shift: contribution is (1 - h[j]) = (1 - h[m-shift])
- If m appears as j: contribution is -h[j+shift] = -h[m+shift]

So grad[m] = sum over valid shifts where m is a valid index in the product.

This matches the previous code's grad_corr_k function.

from scipy.optimize import linprog
from scipy.special import logsumexp
import time


def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-8) & (h < 1.0 - 1e-8)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)


def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def c5_of_h(h, dx):
    return compute_corrs(h).max() * dx


def grad_corr_k(h, k, n):
    shift = k - (n - 1)
    i_lo = max(0, -shift)
    i_hi = min(n - 1, n - 1 - shift)
    if i_lo > i_hi:
        return np.zeros(n)
    idx = np.arange(i_lo, i_hi + 1)
    grad = np.zeros(n)
    grad[idx + shift] += (1.0 - h[idx])
    grad[idx] += (-h[idx + shift])
    return grad


def smooth_max_obj(x, dx, beta):
    corrs = compute_corrs(x)
    return logsumexp(beta * corrs) / beta * dx


def smooth_max_grad(x, dx, beta):
    n = len(x)
    corrs = compute_corrs(x)
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(len(corrs)) / len(corrs)
    else:
        w /= w_sum
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] > 1e-14:
            grad += w[k] * grad_corr_k(x, k, n)
    return grad * dx


def projected_grad_opt(h0, dx, target_sum, betas, iters_per_beta, time_limit):

            grad = smooth_max_grad(h, dx, beta)

            if gnorm < 1e-14:

            step = max(step, 1e-10)

def lp_direction(h, dx, n_active, radius):
    n = len(h)
    corrs = compute_corrs(h)
    max_c = corrs.max()
    top_idx = np.argsort(corrs)[-n_active:]

    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_rows = []
    b_rows = []
    for k in top_idx:
        g = grad_corr_k(h, k, n)
        row = np.zeros(n + 1)
        row[:n] = dx * g
        row[n] = -1.0
        A_rows.append(row)
        b_rows.append(-corrs[k] - 1e-14)

    A_ub = np.array(A_rows)
    b_ub = np.array(b_rows)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(-h[i] * radius, (1.0 - h[i]) * radius) for i in range(n)]
    bounds.append((0.0, None))

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            return res.x[:n]
    except Exception:
        pass
    return None


def lp_polish(h, dx, target_sum, max_iters, time_limit):
    n = len(h)
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    radius = 0.15
    n_active = max(8, min(120, int(0.35 * (2 * n - 1))))
    t0 = time.time()
    no_improve = 0

    for it in range(max_iters):
        if time.time() - t0 > time_limit:
            break
        if radius < 1e-7:
            break
        if no_improve > 40:

        delta = lp_direction(h, dx, n_active, radius)
        if delta is None:
            radius *= 0.5
            no_improve += 1
            continue

        dnorm = np.linalg.norm(delta)
        if dnorm < 1e-12:
            no_improve += 1
            continue

        # Exact line search along LP direction
        cur_c5 = c5_of_h(h, dx)
        best_frac = 0.0
        best_c5_ls = cur_c5
        fracs = [1.0, 0.7, 0.5, 0.3, 0.15, 0.05]
        for frac in fracs:
            h_try = h + frac * delta
            h_try = project_feasible(h_try, target_sum)
            c5_try = c5_of_h(h_try, dx)
            if c5_try < best_c5_ls - 1e-15:
                best_c5_ls = c5_try
                best_frac = frac

        if best_frac > 0:
            h = project_feasible(h + best_frac * delta, target_sum)
            cur_c5 = c5_of_h(h, dx)
            if cur_c5 < best_c5:

                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
            # Adaptively grow active set

            radius = min(0.15, radius * 1.1)
        else:
            radius *= 0.5
            no_improve += 1
            n_active = max(5, n_active - 2)

def pair_polish(h, dx, target_sum, max_rounds, time_limit):
    n = len(h)
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    t0 = time.time()

    for rnd in range(max_rounds):
        if time.time() - t0 > time_limit:
            break
        corrs = compute_corrs(h)
        max_c = corrs.max()
        c5_cur = max_c * dx

        # Weighted gradient from active constraints
        active_thresh = max_c - 0.02
        active_mask = corrs >= active_thresh
        if active_mask.sum() < 3:
            top_k = np.argsort(corrs)[-10:]
            active_mask = np.zeros(len(corrs), dtype=bool)
            active_mask[top_k] = True

        active_indices = np.where(active_mask)[0]
        shifted = corrs[active_indices] - corrs[active_indices].max()
        weights = np.exp(200.0 * shifted)
        weights /= weights.sum()

        grad = np.zeros(n)
        for wi, k in zip(weights, active_indices):
            grad += wi * grad_corr_k(h, k, n)
        grad *= dx

        # Find promising pairs
        src_mask = h > 1e-6
        sink_mask = h < 1.0 - 1e-6
        if not src_mask.any() or not sink_mask.any():
            break

        src_idx = np.where(src_mask)[0]
        sink_idx = np.where(sink_mask)[0]
        # Top sources (high gradient = should decrease)
        src_rank = src_idx[np.argsort(grad[src_idx])[-15:]]
        # Top sinks (low gradient = should increase)
        sink_rank = sink_idx[np.argsort(grad[sink_idx])[:15]]

        improved = False
        for si in src_rank:
            for sj in sink_rank:
                if si == sj or grad[si] <= grad[sj]:
                    continue
                max_t = min(h[si], 1.0 - h[sj])
                if max_t < 1e-8:
                    continue
                best_t = 0.0
                best_c5_try = c5_cur
                for frac in [0.5, 1.0, 0.25, 0.75, 0.15, 0.9, 0.4, 0.6]:
                    t = frac * max_t
                    h_try = h.copy()
                    h_try[si] -= t
                    h_try[sj] += t
                    c5_try = c5_of_h(h_try, dx)
                    if c5_try < best_c5_try:
                        best_c5_try = c5_try
                        best_t = t
                if best_t > 0 and best_c5_try < c5_cur - 1e-15:
                    h[si] -= best_t
                    h[sj] += best_t
                    h = project_feasible(h, target_sum)
                    c5_cur = c5_of_h(h, dx)
                    if c5_cur < best_c5:
                        best_c5 = c5_cur
                        best_h = h.copy()
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
    return best_h, best_c5


def worst_target_polish(h, dx, target_sum, max_rounds, time_limit):
    """Target the single worst correlation directly."""
    n = len(h)
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    t0 = time.time()

    for rnd in range(max_rounds):
        if time.time() - t0 > time_limit:
            break
        corrs = compute_corrs(h)
        k_worst = np.argmax(corrs)
        c5_cur = corrs.max() * dx

        # Gradient of the worst correlation
        g = grad_corr_k(h, k_worst, n) * dx

        # Also consider nearby correlations to avoid making them worse
        nearby = np.argsort(-np.abs(corrs - corrs.max()))[:5]
        g_combined = g.copy()
        for k in nearby:
            if k != k_worst:
                g_combined += 0.5 * grad_corr_k(h, k, n) * dx

        # Move in direction that reduces worst correlation
        src_mask = (h > 1e-6) & (g_combined > 0.01)
        sink_mask = (h < 1.0 - 1e-6) & (g_combined < -0.01)
        if not src_mask.any() or not sink_mask.any():
            break

        src_idx = np.where(src_mask)[0]
        sink_idx = np.where(sink_mask)[0]
        # Pick the most extreme source and sink
        si = src_idx[np.argmax(g_combined[src_idx])]
        sj = sink_idx[np.argmin(g_combined[sink_idx])]

        max_t = min(h[si], 1.0 - h[sj])
        if max_t < 1e-8:
            break

        best_t = 0.0
        best_c5_try = c5_cur
        for frac in [0.3, 0.5, 0.7, 1.0, 0.15, 0.85]:
            t = frac * max_t
            h_try = h.copy()
            h_try[si] -= t
            h_try[sj] += t
            c5_try = c5_of_h(h_try, dx)
            if c5_try < best_c5_try:
                best_c5_try = c5_try
                best_t = t

        if best_t > 0 and best_c5_try < c5_cur - 1e-15:
            h[si] -= best_t
            h[sj] += best_t
            h = project_feasible(h, target_sum)
            if c5_of_h(h, dx) < best_c5:
                best_c5 = c5_of_h(h, dx)
                best_h = h.copy()
        else:
            break
    return best_h, best_c5


def make_binary_random(n, rng):
    h = np.zeros(n)
    idx = rng.permutation(n)
    h[idx[:n // 2]] = 1.0
    return h


def make_periodic(n, period):
    fill = period // 2
    if fill < 1:
        fill = 1
    h = np.zeros(n)
    pattern = np.zeros(period)
    pattern[:fill] = 1.0
    h = np.tile(pattern, n // period + 1)[:n]
    return project_feasible(h, n / 2.0)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def remaining():
        return budget_s - (time.time() - t0)

    # Get parent solution
    h0 = np.array(initial_h_values, dtype=float).copy()
    n = len(h0)
    dx = 2.0 / n
    target_sum = n / 2.0
    h0 = project_feasible(h0, target_sum)

    best_h = h0.copy()
    best_c5 = c5_of_h(h0, dx)

    # Phase 1: Projected gradient with smooth max (warmup)
    if remaining() > 100:
        h_opt, c5_opt = projected_grad_opt(
            best_h, dx, target_sum,
            betas=[50, 200, 600, 1500],
            iters_per_beta=300,
            time_limit=min(60, remaining() * 0.35))
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 2: Aggressive LP polishing
    if remaining() > 60:
        h_lp, c5_lp = lp_polish(best_h, dx, target_sum,
                                max_iters=500, time_limit=min(50, remaining() * 0.35))
        if c5_lp < best_c5:
            best_c5 = c5_lp
            best_h = h_lp

    # Phase 3: Pair polish
    if remaining() > 30:
        h_pp, c5_pp = pair_polish(best_h, dx, target_sum,
                                  max_rounds=300, time_limit=min(25, remaining() * 0.25))
        if c5_pp < best_c5:
            best_c5 = c5_pp
            best_h = h_pp

    # Phase 4: Worst-target polish
    if remaining() > 20:
        h_wt, c5_wt = worst_target_polish(best_h, dx, target_sum,
                                          max_rounds=100, time_limit=min(15, remaining() * 0.2))
        if c5_wt < best_c5:
            best_c5 = c5_wt
            best_h = h_wt

    # Phase 5: Alternate LP + pair + worst-target cycles
    for cycle in range(4):
        if remaining() < 15:
            break
        h_lp, c5_lp = lp_polish(best_h, dx, target_sum,
                                max_iters=200, time_limit=min(15, remaining() * 0.3))
        if c5_lp < best_c5:
            best_c5 = c5_lp
            best_h = h_lp
        if remaining() < 10:
            break
        h_pp, c5_pp = pair_polish(best_h, dx, target_sum,
                                  max_rounds=150, time_limit=min(10, remaining() * 0.25))
        if c5_pp < best_c5:
            best_c5 = c5_pp
            best_h = h_pp
        if remaining() < 8:
            break
        h_wt, c5_wt = worst_target_polish(best_h, dx, target_sum,
                                          max_rounds=80, time_limit=min(8, remaining() * 0.2))
        if c5_wt < best_c5:
            best_c5 = c5_wt
            best_h = h_wt

    # Phase 6: Try different n values
    n_candidates = [64, 80, 96, 112, 128, 144, 160]
    for n_try in n_candidates:
        if remaining() < 25:
            break
        if n_try == n:
            continue
        dx_try = 2.0 / n_try
        ts_try = n_try / 2.0

        x_orig = 0.5 * dx + dx * np.arange(n)
        x_new = 0.5 * dx_try + dx_try * np.arange(n_try)
        h_interp = np.interp(x_new, x_orig, best_h)
        h_interp = project_feasible(h_interp, ts_try)

        # Quick projected gradient
        h_opt, c5_opt = projected_grad_opt(
            h_interp, dx_try, ts_try,
            betas=[200, 800],
            iters_per_beta=200,
            time_limit=min(8, remaining() * 0.3))

        if remaining() > 10:
            h_lp, c5_lp = lp_polish(h_opt, dx_try, ts_try,
                                    max_iters=100, time_limit=min(8, remaining() * 0.25))
            if c5_lp < c5_opt:
                c5_opt = c5_lp
                h_opt = h_lp
        if remaining() > 8:
            h_pp, c5_pp = pair_polish(h_opt, dx_try, ts_try,
                                      max_rounds=80, time_limit=min(6, remaining() * 0.2))
            if c5_pp < c5_opt:
                c5_opt = c5_pp
                h_opt = h_pp

        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 7: Diverse starts on best n
    best_n = len(best_h)
    best_dx = 2.0 / best_n
    best_ts = best_n / 2.0
    for trial in range(15):
        if remaining() < 12:
            break
        if trial % 3 == 0:
            h_init = make_binary_random(best_n, rng)
        elif trial % 3 == 1:
            h_init = make_periodic(best_n, rng.choice([4, 5, 6, 7, 8, 9, 10]))
        else:
            h_init = project_feasible(best_h + 0.25 * (rng.random(best_n) - 0.5), best_ts)

        h_opt, c5_opt = projected_grad_opt(
            h_init, best_dx, best_ts,
            betas=[100, 500, 1200],
            iters_per_beta=200,
            time_limit=min(6, remaining() * 0.3))

        if remaining() > 8:
            h_lp, c5_lp = lp_polish(h_opt, best_dx, best_ts,
                                    max_iters=80, time_limit=min(5, remaining() * 0.25))
            if c5_lp < c5_opt:
                c5_opt = c5_lp
                h_opt = h_lp
        if remaining() > 6:
            h_pp, c5_pp = pair_polish(h_opt, best_dx, best_ts,
                                      max_rounds=60, time_limit=min(4, remaining() * 0.2))
            if c5_pp < c5_opt:
                c5_opt = c5_pp
                h_opt = h_pp

        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Final intensive polish
    if remaining() > 8:
        h_lp, c5_lp = lp_polish(best_h, best_dx, best_ts,
                                max_iters=200, time_limit=min(6, remaining() * 0.5))
        if c5_lp < best_c5:
            best_c5 = c5_lp
            best_h = h_lp
    if remaining() > 4:
        h_pp, c5_pp = pair_polish(best_h, best_dx, best_ts,
                                  max_rounds=100, time_limit=min(3, remaining() * 0.5))
        if c5_pp < best_c5:
            best_c5 = c5_pp
            best_h = h_pp

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    final_dx = 2.0 / len(best_h)
    final_c5 = c5_of_h(best_h, final_dx)

    return best_h, final_c5, len(best_h)
