import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time

def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
        else:
            h += diff / n
    return np.clip(h, 0.0, 1.0)

def c5_of_h(h, dx):
    return np.correlate(h, 1.0 - h, mode='full').max() * dx

def smooth_max_val(corrs, beta):
    m = corrs.max()
    if beta < 1:
        return m
    shifted = corrs - m
    return m + np.log(np.sum(np.exp(beta * shifted)) / np.exp(beta * m)) / beta
    # Actually: m + log(sum(exp(beta*(corr-m))))/beta

def smooth_max_obj(h, dx, beta):
    corrs = np.correlate(h, 1.0 - h, mode='full')
    m = corrs.max()
    shifted = corrs - m
    val = m + np.log(np.sum(np.exp(beta * shifted))) / beta
    return val * dx

def smooth_max_grad(h, dx, beta):
    n = len(h)
    corrs = np.correlate(h, 1.0 - h, mode='full')
    m = corrs.max()
    shifted = corrs - m
    with np.errstate(over='ignore'):
        w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        # Fallback: uniform over max
        w = np.zeros(len(corrs))
        w[corrs == m] = 1.0 / (corrs == m).sum()
    else:
        w /= w_sum
    # Compute weighted gradient
    grad = np.zeros(n)
    ms = np.arange(-(n-1), n)  # lags
    for idx_k, k in enumerate(ms):

def swap_delta(h, i, j, n):
    """Compute delta in correlations for swapping h[i]:1->0, h[j]:0->1."""
    ms = np.arange(-(n-1), n)  # lags
    delta = np.zeros(2*n-1)
    
    # Contribution from h[i]: 1->0
    # -(1-h[i-k]) if 0 <= i-k < n
    idx1 = i - ms
    valid1 = (idx1 >= 0) & (idx1 < n)
    delta[valid1] += -(1.0 - h[idx1[valid1]])
    
    # +h[i+k] if 0 <= i+k < n
    idx2 = i + ms
    valid2 = (idx2 >= 0) & (idx2 < n)
    delta[valid2] += h[idx2[valid2]]
    
    # Contribution from h[j]: 0->1
    # +(1-h[j-k]) if 0 <= j-k < n
    idx3 = j - ms
    valid3 = (idx3 >= 0) & (idx3 < n)
    delta[valid3] += (1.0 - h[idx3[valid3]])
    
    # -h[j+k] if 0 <= j+k < n
    idx4 = j + ms
    valid4 = (idx4 >= 0) & (idx4 < n)
    delta[valid4] += -h[idx4[valid4]]
    
    return delta

import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time


def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
        else:
            h += diff / n
    return np.clip(h, 0.0, 1.0)


def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def c5_of_h(h, dx):
    return compute_corrs(h).max() * dx


def smooth_max_val(h, dx, beta):
    corrs = compute_corrs(h)
    m = corrs.max()
    shifted = corrs - m
    return (m + np.log(np.sum(np.exp(beta * shifted))) / beta) * dx


def smooth_max_grad(h, dx, beta):
    n = len(h)
    corrs = compute_corrs(h)
    m = corrs.max()
    shifted = corrs - m
    with np.errstate(over='ignore'):
        w = np.exp(beta * shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(corrs))
        w[corrs >= m - 1e-14] = 1.0 / np.sum(corrs >= m - 1e-14)
    else:
        w /= ws
    grad = np.zeros(n)
    ms = np.arange(-(n - 1), n)
    for idx_k in range(len(ms)):
        wk = w[idx_k]
        if wk < 1e-15:
            continue
        k = ms[idx_k]
        if k >= 0:
            kk = int(k)
            if kk > 0:
                grad[kk:] += wk * (1.0 - h[:n - kk])
                grad[:n - kk] += wk * (-h[kk:])
            else:
                grad[:] += wk * (1.0 - h[:])
                grad[:] += wk * (-h[:])
        else:
            kp = int(-k)
            if kp > 0:
                grad[:n - kp] += wk * (1.0 - h[kp:])
                grad[kp:] += wk * (-h[:n - kp])
            else:
                grad[:] += wk * (1.0 - h[:])
                grad[:] += wk * (-h[:])
    return grad * dx


def lbfgs_refine(h0, dx, target, max_time=60.0, beta=500):
    n = len(h0)
    t_start = time.time()
    penalty_mu = 100.0

    def obj_pen(x):
        c = smooth_max_val(x, dx, beta)
        s = x.sum() - target
        return c + penalty_mu * s * s

    def grad_pen(x):
        g = smooth_max_grad(x, dx, beta)
        s = x.sum() - target
        return g + 2.0 * penalty_mu * s * np.ones(n)

    x = h0.copy()
    bounds = [(0.0, 1.0)] * n
    best_x = x.copy()
    best_val = obj_pen(x)

    for trial in range(3):
        elapsed = time.time() - t_start
        if elapsed > max_time - 5:
            break
        res = minimize(obj_pen, x, jac=grad_pen, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': 300, 'ftol': 1e-18, 'gtol': 1e-12,
                                'maxls': 50})
        x = res.x
        h_proj = project_feasible(x, target)
        val = obj_pen(h_proj)
        if val < best_val:
            best_val = val
            best_x = h_proj.copy()
        x = h_proj
        if time.time() - t_start > max_time - 10:
            break

    return project_feasible(best_x, target)


def slsqp_refine(h0, dx, target, betas, max_time=60.0):
    n = len(h0)
    t_start = time.time()
    h = h0.copy()
    best_h = h0.copy()
    best_c5 = c5_of_h(h0, dx)

    for beta in betas:
        elapsed = time.time() - t_start
        if elapsed > max_time - 10:
            break

        def obj(x):
            return smooth_max_val(x, dx, beta)

        def obj_g(x):
            return smooth_max_grad(x, dx, beta)

        def cons_fun(x):
            return x.sum() - target

        def cons_jac(x):
            return np.ones(n)

        cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
        bounds = [(0.0, 1.0)] * n

        try:
            res = minimize(obj, h, jac=obj_g, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 500, 'ftol': 1e-18})
            h_new = project_feasible(res.x, target)
        except Exception:
            h_new = h.copy()

        c5_new = c5_of_h(h_new, dx)
        if c5_new < best_c5:

        h = h_new

def swap_delta_fast(h, i, j, n):
    ms = np.arange(-(n - 1), n)
    delta = np.zeros(2 * n - 1)
    idx1 = i - ms
    v1 = (idx1 >= 0) & (idx1 < n)
    delta[v1] -= (1.0 - h[idx1[v1]])
    idx2 = i + ms
    v2 = (idx2 >= 0) & (idx2 < n)
    delta[v2] += h[idx2[v2]]
    idx3 = j - ms
    v3 = (idx3 >= 0) & (idx3 < n)
    delta[v3] += (1.0 - h[idx3[v3]])
    idx4 = j + ms
    v4 = (idx4 >= 0) & (idx4 < n)
    delta[v4] -= h[idx4[v4]]
    return delta

                    h[i], h[j] = 0.0, 1.0
                    corrs = new_corrs
                    ones = np.where(h == 1.0)[0]
                    zeros = np.where(h == 0.0)[0]
                    no_improve = 0
                else:
                    no_improve += 1

        if no_improve > patience:
            break

    return best_h, best_c5


def directed_binary_search(h0, dx, rng, max_iters=100000, time_limit=60.0):
    """Focus swaps on positions contributing to the worst lag."""
    n = len(h0)
    t_start = time.time()
    target = n // 2
    h = np.zeros(n)
    h[np.argsort(-h0)[:target]] = 1.0
    corrs = compute_corrs(h)
    best_c5 = corrs.max() * dx
    best_h = h.copy()
    temp = 0.02
    no_improve = 0

    for it in range(max_iters):
        if time.time() - t_start > time_limit - 2:
            break
        if it % 5000 == 0:
            temp *= 0.999
        if temp < 1e-10:
            break

        k_star = np.argmax(corrs)
        ms = np.arange(-(n - 1), n)
        k_val = ms[k_star]

        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        delta = swap_delta_fast(h, i, j, n)
        new_c5 = (corrs + delta).max() * dx

        if new_c5 <= best_c5 + 1e-15:
            h[i], h[j] = 0.0, 1.0
            corrs = compute_corrs(h)
            best_c5 = new_c5
            best_h = h.copy()
            no_improve = 0
        else:
            d_val = new_c5 - best_c5
            if rng.random() < np.exp(-d_val / temp):
                h[i], h[j] = 0.0, 1.0
                corrs = compute_corrs(h)
                no_improve = 0
            else:
                no_improve += 1

        if no_improve > 6000:
            break

    return best_h, best_c5


def projected_gradient(h0, dx, target, max_iters=50000, time_limit=60.0):
    """Direct subgradient method on the non-smooth max."""
    n = len(h0)
    t_start = time.time()
    h = h0.copy()
    best_h = h0.copy()
    best_c5 = c5_of_h(h0, dx)
    step = 0.01
    lr_decay = 0.9995

    for it in range(max_iters):
        if time.time() - t_start > time_limit - 2:
            break
        if it % 100 == 0:
            step *= lr_decay
            if step < 1e-6:
                break

        corrs = compute_corrs(h)
        k_star = np.argmax(corrs)
        ms = np.arange(-(n - 1), n)
        k = ms[k_star]

        grad = np.zeros(n)
        if k >= 0:
            kk = int(k)
            if kk > 0:
                grad[kk:] = (1.0 - h[:n - kk])
                grad[:n - kk] += -h[kk:]
            else:
                grad = 1.0 - 2.0 * h
        else:
            kp = int(-k)
            if kp > 0:
                grad[:n - kp] = (1.0 - h[kp:])
                grad[kp:] += -h[:n - kp]

        h_new = h - step * grad
        h_new = project_feasible(h_new, target)
        c5_new = c5_of_h(h_new, dx)

        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new

    return best_h, best_c5


def try_new_resolution(h_best, n_new, rng, time_limit=60.0):
    """Interpolate to new resolution and optimize."""
    n_old = len(h_best)
    dx_new = 2.0 / n_new
    target_new = n_new / 2.0
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_interp = np.interp(x_new, x_old, h_best)
    h_interp = project_feasible(h_interp, target_new)
    c5_interp = c5_of_h(h_interp, dx_new)

    if time_limit < 20:
        return h_interp, c5_interp, n_new

    t_s = time.time()
    h_opt, c5_opt = slsqp_refine(h_interp, dx_new, target_new,
                                  [100, 300, 800],
                                  max_time=max(10, time_limit * 0.5))
    if time.time() - t_s < time_limit * 0.4:
        h_bin = np.zeros(n_new)
        h_bin[np.argsort(-h_opt)[:n_new // 2]] = 1.0
        h_bs, c5_bs = binary_steepest(h_bin, dx_new, max_sweeps=30,
                                      time_limit=max(5, time_limit * 0.25))
        if c5_bs < c5_opt:
            c5_opt = c5_bs
            h_opt = h_bs

    return h_opt, c5_opt, n_new


def make_rand_start(n, rng):
    h = np.zeros(n)
    idx = rng.permutation(n)
    h[idx[:n // 2]] = 1.0
    return project_feasible(h, n / 2.0)


def make_quasirand_start(n, rng, seed_offset=0):
    h = np.zeros(n)
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    for k in range(n // 2):
        pos = int((k * phi + seed_offset) * n) % n
        h[pos] = 1.0
    return project_feasible(h, n / 2.0)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def remaining():
        return budget_s - (time.time() - t0)

    best_h = np.array(initial_h_values, dtype=float)
    n_best = len(best_h)
    dx_best = 2.0 / n_best
    target_best = n_best / 2.0
    best_h = project_feasible(best_h, target_best)
    best_c5 = c5_of_h(best_h, dx_best)

    # Phase 1: SLSQP refinement on initial resolution
    if remaining() > 80:
        h_opt, c5_opt = slsqp_refine(best_h, dx_best, target_best,
                                     [50, 100, 200, 500],
                                     max_time=min(70, remaining() * 0.5))
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 2: Binary steepest descent
    if remaining() > 60:
        h_bin = np.zeros(n_best)
        h_bin[np.argsort(-best_h)[:n_best // 2]] = 1.0
        h_bs, c5_bs = binary_steepest(h_bin, dx_best, max_sweeps=80,
                                      time_limit=min(50, remaining() * 0.4))
        if c5_bs < best_c5:
            best_c5 = c5_bs
            best_h = project_feasible(h_bs, target_best)

    # Phase 3: Binary SA
    if remaining() > 100:
        h_sa, c5_sa = binary_sa(best_h, dx_best, rng, max_iters=200000,
                                time_limit=min(80, remaining() * 0.5),
                                temp_init=0.04, cool=0.9997)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = project_feasible(h_sa, target_best)

    # Phase 4: L-BFGS-B continuous refinement
    if remaining() > 60:
        h_lbg = lbfgs_refine(best_h, dx_best, target_best,
                             max_time=min(50, remaining() * 0.5), beta=200)
        c5_lbg = c5_of_h(h_lbg, dx_best)
        if c5_lbg < best_c5:
            best_c5 = c5_lbg
            best_h = h_lbg

    # Phase 5: More SLSQP with higher beta
    if remaining() > 60:
        h_opt, c5_opt = slsqp_refine(best_h, dx_best, target_best,
                                     [500, 1000, 2000],
                                     max_time=min(50, remaining() * 0.5))
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 6: Directed binary search
    if remaining() > 60:
        h_dir, c5_dir = directed_binary_search(best_h, dx_best, rng,
                                               max_iters=80000,
                                               time_limit=min(50, remaining() * 0.5))
        if c5_dir < best_c5:
            best_c5 = c5_dir
            best_h = project_feasible(h_dir, target_best)

    # Phase 7: Projected gradient
    if remaining() > 50:
        h_pg, c5_pg = projected_gradient(best_h, dx_best, target_best,
                                         max_iters=30000,
                                         time_limit=min(40, remaining() * 0.5))
        if c5_pg < best_c5:
            best_c5 = c5_pg
            best_h = h_pg

    # Phase 8: Binary SA multi (double swaps)
    if remaining() > 80:
        h_m, c5_m = binary_sa_multi(best_h, dx_best, rng, max_iters=200000,
                                     time_limit=min(70, remaining() * 0.5))
        if c5_m < best_c5:
            best_c5 = c5_m
            best_h = project_feasible(h_m, target_best)

    # Phase 9: Try nearby resolutions
    n_candidates = []
    for dn in [-6, -4, -2, 2, 4, 6, 8, 10, 12, 14, 16, 20, 24, 28, 32, 40]:
        nn = n_best + dn
        if nn >= 48 and nn <= 250 and nn % 2 == 0:
            n_candidates.append(nn)
    n_candidates = sorted(set(n_candidates))

    # Sort by expected ease (nearby first)
    n_candidates.sort(key=lambda x: abs(x - n_best))

    for nn in n_candidates[:8]:
        if remaining() < 40:
            break
        alloc = min(50, remaining() * 0.4)
        h_new, c5_new, n_new = try_new_resolution(best_h, nn, rng, time_limit=alloc)
        if c5_new < best_c5 - 1e-6:
            best_c5 = c5_new
            best_h = h_new
            n_best = n_new
            dx_best = 2.0 / n_best
            target_best = n_best / 2.0
            # Continue refining at the new resolution
            if remaining() > 60:
                h_o2, c5_o2 = slsqp_refine(best_h, dx_best, target_best,
                                           [200, 500, 1000],
                                           max_time=min(40, remaining() * 0.4))
                if c5_o2 < best_c5:
                    best_c5 = c5_o2
                    best_h = h_o2
                if remaining() > 30:
                    h_b2 = np.zeros(n_best)
                    h_b2[np.argsort(-best_h)[:n_best // 2]] = 1.0
                    h_bs2, c5_bs2 = binary_steepest(h_b2, dx_best,
                                                     max_sweeps=40,
                                                     time_limit=min(25, remaining() * 0.3))
                    if c5_bs2 < best_c5:
                        best_c5 = c5_bs2
                        best_h = project_feasible(h_bs2, target_best)

    # Phase 10: Try random restarts at current best resolution
    if remaining() > 80:
        for attempt in range(3):
            if remaining() < 30:
                break
            h_rand = make_quasirand_start(n_best, rng, seed_offset=attempt * 7.3)
            h_r_opt, c5_r = slsqp_refine(h_rand, dx_best, target_best,
                                         [100, 300],
                                         max_time=min(25, remaining() * 0.4))
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r_opt

    # Phase 11: Final polish
    if remaining() > 40:
        h_final, c5_final = slsqp_refine(best_h, dx_best, target_best,
                                         [1000, 3000],
                                         max_time=min(30, remaining() * 0.6))
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final

    if remaining() > 25:
        h_l2 = lbfgs_refine(best_h, dx_best, target_best,
                            max_time=min(20, remaining() * 0.7), beta=2000)
        c5_l2 = c5_of_h(h_l2, dx_best)
        if c5_l2 < best_c5:
            best_c5 = c5_l2
            best_h = h_l2

    # Final validation
    n_f = len(best_h)
    dx_f = 2.0 / n_f
    target_f = n_f / 2.0
    best_h = project_feasible(best_h, target_f)
    best_c5 = c5_of_h(best_h, dx_f)

    return best_h, best_c5, n_f
