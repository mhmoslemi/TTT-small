def smooth_obj_grad(q, n, tau):
    h = build_sym(q, n)
    raw = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    C = raw * dx
    M = C.max()
    scores = (C - M) / tau
    # Clip for numerical stability
    scores = np.clip(scores, -100, 100)
    exp_scores = np.exp(scores)
    Z = exp_scores.sum()
    p = exp_scores / Z
    Phi = M + tau * np.log(Z)
    
    # Gradient
    g_h = np.zeros(n)
    # For each lag d, weight = dx * p[d + n - 1]
    for d in range(-(n-1), n):
        w = dx * p[d + n - 1]
        if w < 1e-15:
            continue
        # First term: (1 - h[j-d]) for valid j
        if d >= 0:
            j_start, j_end = d, n
            s = slice(j_start, j_end)
            g_h[s] += w * (1.0 - h[j_start - d:j_end - d])
        else:
            j_start, j_end = 0, n + d
            s = slice(j_start, j_end)
            g_h[s] += w * (1.0 - h[j_start - d:j_end - d])
        # Second term: -h[j+d] for valid j
        if d >= 0:
            j_start, j_end = 0, n - d
            s = slice(j_start, j_end)
            g_h[s] += w * (-h[j_start + d:j_end + d])
        else:
            j_start, j_end = -d, n
            s = slice(j_start, j_end)
            g_h[s] += w * (-h[j_start + d:j_end + d])
    
    # Transform to q gradient
    m = n // 2
    g_q = g_h[:m] - g_h[n-1:m-1:-1]
    
    return Phi, g_q

    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def build_sym(q, n):
    m = n // 2
    h = np.empty(n)
    h[:m] = q
    h[m:] = 1.0 - q[::-1]
    return h

def q_of_h(h, n):
    m = n // 2
    q = 0.5 * (h[:m] + 1.0 - h[n-1:m-1:-1])
    return np.clip(q, 0.0, 1.0)

def smooth_phi_grad(q, n, tau):
    """Compute the log-sum-exp smoothed objective and gradient w.r.t. q."""
    h = build_sym(q, n)
    raw = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    C = raw * dx
    M = C.max()
    scores = (C - M) / tau
    scores = np.clip(scores, -500, 500)
    exp_s = np.exp(scores)
    Z = exp_s.sum()
    if Z < 1e-300:
        p = np.full_like(C, 1.0 / len(C))
    else:
        p = exp_s / Z
    Phi = M + tau * np.log(max(Z, 1e-300))
    
    # Compute gradient w.r.t. h
    g_h = np.zeros(n)
    for j in range(len(C)):
        w = dx * p[j]
        if w < 1e-18:
            continue
        d = j - (n - 1)
        # Contribution to g_h[p]:
        # (1 - h[p-d]) * I(valid first) - h[p+d] * I(valid second)
        # For d >= 0:
        #   first: d <= p <= n-1, contributes (1-h[p-d])
        #   second: 0 <= p <= n-1-d, contributes -h[p+d]
        # For d < 0 (let d' = -d):
        #   first: 0 <= p <= n-1-d', contributes (1-h[p+d'])
        #   second: d' <= p <= n-1, contributes -h[p-d']
        if d >= 0:
            if d < n:
                # First term: p from d to n-1
                g_h[d:n] += w * (1.0 - h[0:n-d])
            # Second term: p from 0 to n-1-d
            if d > 0:
                g_h[0:n-d] += w * (-h[d:n])
            else:
                g_h[0:n] += w * (-h[0:n])
        else:
            dp = -d
            # First term: p from 0 to n-1-dp
            if dp < n:
                g_h[0:n-dp] += w * (1.0 - h[dp:n])
            # Second term: p from dp to n-1
            if dp > 0:
                g_h[dp:n] += w * (-h[0:n-dp])
    
    # Transform to q gradient
    m = n // 2
    g_q = g_h[:m] - g_h[n-1:m-1:-1]
    
    return Phi, g_q

def opt_smooth(q0, n, tau_list, max_per_stage=100, gtol=1e-8):
    """Anneal through tau values, optimizing at each stage."""
    q = q0.copy()
    bounds = [(0.0, 1.0)] * len(q)
    
    for tau in tau_list:
        def obj(z):
            phi, grad = smooth_phi_grad(z, n, tau)
            return phi, grad
        try:
            res = minimize(obj, q, jac=True, method='L-BFGS-B',
                          bounds=bounds, options={'maxiter': max_per_stage, 'gtol': gtol})
            q_new = np.clip(res.x, 0.0, 1.0)
            # Accept if exact objective improved or is comparable
            h_old = build_sym(q, n)
            h_new = build_sym(q_new, n)
            c5_old = eval_c5(h_old)
            c5_new = eval_c5(h_new)
            if c5_new <= c5_old + 1e-6:  # Allow small degradation for progress
                q = q_new
        except Exception:
            pass
    
    return q, eval_c5(build_sym(q, n))

def project_to_target(h, target_sum):
    """Project h to have sum = target_sum while staying in [0,1]."""
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        clipped = np.clip(h - mid, 0.0, 1.0)
        if clipped.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    result = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    # Fix small numerical error
    diff = target_sum - result.sum()
    if abs(diff) > 1e-12:
        mask = (result > 1e-10) & (result < 1.0 - 1e-10)
        if mask.sum() > 0:
            result[mask] += diff / mask.sum()
    return np.clip(result, 0.0, 1.0)

def opt_full(h0, n, tau_list, max_per_stage=80, gtol=1e-8):
    """Full optimization without symmetry constraint, using penalty for integral."""
    target = n / 2.0
    h = h0.copy()
    
    def obj(z):
        h_local = np.clip(z, 0.0, 1.0)
        # Soft projection
        raw = np.correlate(h_local, 1.0 - h_local, mode='full')
        dx = 2.0 / n
        C = raw * dx
        M = C.max()
        scores = (C - M) / tau
        scores = np.clip(scores, -500, 500)
        exp_s = np.exp(scores)
        Z = exp_s.sum()
        p = exp_s / max(Z, 1e-300)
        phi = M + tau * np.log(max(Z, 1e-300))
        # Penalty for sum constraint
        sdev = h_local.sum() - target
        phi += 1000.0 * sdev * sdev
        
        # Gradient
        g_h = np.zeros(n)
        for j in range(len(C)):
            w = dx * p[j]
            if w < 1e-18:
                continue
            d = j - (n - 1)
            if d >= 0:
                if d < n:
                    g_h[d:n] += w * (1.0 - h_local[0:n-d])
                if d > 0:
                    g_h[0:n-d] += w * (-h_local[d:n])
                else:
                    g_h[0:n] += w * (-h_local[0:n])
            else:
                dp = -d
                if dp < n:
                    g_h[0:n-dp] += w * (1.0 - h_local[dp:n])
                if dp > 0:
                    g_h[dp:n] += w * (-h_local[0:n-dp])
        g_h += 2000.0 * sdev
        
        # Boundary handling: push back if at boundary
        at_lo = (h_local <= 0.0) & (g_h < 0)
        at_hi = (h_local >= 1.0) & (g_h > 0)
        g_h[at_lo] = 0.0
        g_h[at_hi] = 0.0
        
        return phi, g_h
    
    for i, tau in enumerate(tau_list):
        try:
            res = minimize(obj, h, jac=True, method='L-BFGS-B',
                          bounds=[(0.0, 1.0)] * n,
                          options={'maxiter': max_per_stage, 'gtol': gtol})
            h_new = np.clip(res.x, 0.0, 1.0)
            h_new = project_to_target(h_new, target)
            c5_new = eval_c5(h_new)
            c5_old = eval_c5(h)
            if c5_new <= c5_old + 1e-6:
                h = h_new
        except Exception:
            pass
    
    return h, eval_c5(h)

def interp_to(h, n_new):
    n_old = len(h)
    if n_new == n_old:
        return h.copy()
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h)
    return project_to_target(h_new, n_new / 2.0)

def make_random_q(n, rng):

        q = np.zeros(m)

            start = int(k * m / n_steps)
            end = int((k+1) * m / n_steps)
            q[start:end] = rng.uniform(0.1, 0.9)
        # Smooth slightly

            kernel = np.ones(3) / 3.0
            q = np.convolve(q, kernel, mode='same')

            kernel = np.ones(kernel_size) / kernel_size
            q = np.convolve(q, kernel, mode='same')
    return np.clip(q, 0.0, 1.0)

                           bounds=bounds, options={'maxiter': max_per_stage, 'gtol': gtol})
            q_new = np.clip(res.x, 0.0, 1.0)
            c5_old = eval_c5(build_sym(q, n))
            c5_new = eval_c5(build_sym(q_new, n))
            if c5_new <= c5_old + 5e-7:
                q = q_new
        except Exception:
            pass
    h_final = build_sym(q, n)
    return q, eval_c5(h_final)


def full_phi_grad(h, n, tau, target):
    raw = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    C = raw * dx
    M = C.max()
    scores = (C - M) / tau
    scores = np.clip(scores, -500.0, 500.0)
    exp_s = np.exp(scores)
    Z = exp_s.sum()
    p = exp_s / max(Z, 1e-300)
    phi = M + tau * np.log(max(Z, 1e-300))
    sdev = h.sum() - target
    phi += 500.0 * sdev * sdev
    g_h = np.zeros(n)
    for j in range(len(C)):
        w = dx * p[j]
        if w < 1e-18:
            continue
        d = j - (n - 1)
        if d >= 0:
            if d < n:
                g_h[d:n] += w * (1.0 - h[0:n-d])
            if d > 0:
                g_h[0:n-d] -= w * h[d:n]
            else:
                g_h[:] -= w * h[:]
        else:
            dp = -d
            if dp < n:
                g_h[0:n-dp] += w * (1.0 - h[dp:n])
            if dp > 0:
                g_h[dp:n] -= w * h[0:n-dp]
    g_h += 1000.0 * sdev
    at_lo = (h <= 1e-15) & (g_h < 0)
    at_hi = (h >= 1.0 - 1e-15) & (g_h > 0)
    g_h[at_lo] = 0.0
    g_h[at_hi] = 0.0
    return phi, g_h


def opt_full(h0, n, tau_list, max_per_stage=80, gtol=1e-9):
    target = n / 2.0
    h = h0.copy()
    bounds = [(0.0, 1.0)] * n
    for tau in tau_list:
        def obj(z):
            return full_phi_grad(z, n, tau, target)
        try:
            res = minimize(obj, h, jac=True, method='L-BFGS-B',
                           bounds=bounds, options={'maxiter': max_per_stage, 'gtol': gtol})
            h_new = np.clip(res.x, 0.0, 1.0)
            h_new = project_to_target(h_new, target)
            c5_new = eval_c5(h_new)
            c5_old = eval_c5(h)
            if c5_new <= c5_old + 5e-7:
                h = h_new
        except Exception:
            pass
    return h, eval_c5(h)


def project_to_target(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0.0, 1.0).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    result = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - result.sum()
    if abs(diff) > 1e-12:
        mask = (result > 1e-10) & (result < 1.0 - 1e-10)
        if mask.sum() > 0:
            result[mask] += diff / mask.sum()
    return np.clip(result, 0.0, 1.0)


def interp_to(h, n_new):
    n_old = len(h)
    if n_new == n_old:
        return h.copy()
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h)
    return project_to_target(h_new, n_new / 2.0)


def make_random_q(n, rng):
    m = n // 2
    t = np.linspace(0.0, 1.0, m)
    mode = rng.randint(5)
    if mode == 0:
        f1 = rng.uniform(1, 4)
        f2 = rng.uniform(4, 8)
        p1 = rng.uniform(0, 2 * np.pi)
        p2 = rng.uniform(0, 2 * np.pi)
        a1 = rng.uniform(0.15, 0.35)
        a2 = rng.uniform(0.05, 0.2)
        q = 0.5 + a1 * np.sin(2 * np.pi * f1 * t + p1) + a2 * np.sin(2 * np.pi * f2 * t + p2)
    elif mode == 1:
        n_steps = rng.randint(3, 8)
        q = np.zeros(m)
        for k in range(n_steps):
            s = int(k * m / n_steps)
            e = int((k + 1) * m / n_steps)
            q[s:e] = rng.uniform(0.05, 0.95)
        if m > 6:
            ksz = max(3, m // 15)
            ker = np.ones(ksz) / ksz
            q = np.convolve(q, ker, mode='same')
    elif mode == 2:
        c = rng.uniform(0.2, 0.8)
        w = rng.uniform(0.08, 0.35)
        q = 0.5 + 0.4 * np.exp(-((t - c) / w) ** 2)
        q += 0.05 * np.sin(2 * np.pi * rng.uniform(3, 7) * t)
    elif mode == 3:
        q = np.ones(m) * 0.5
        n_bumps = rng.randint(2, 5)
        for _ in range(n_bumps):
            c = rng.uniform(0, 1)
            w = rng.uniform(0.05, 0.2)
            amp = rng.uniform(-0.3, 0.3)
            q += amp * np.exp(-((t - c) / w) ** 2)
    else:
        q = 0.5 + 0.3 * (2 * rng.rand(m) - 1)
        if m > 6:
            ksz = max(3, m // 12)
            ker = np.ones(ksz) / ksz
            q = np.convolve(q, ker, mode='same')
    return np.clip(q, 0.0, 1.0)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')

    def remaining():
        return budget_s - (time.time() - t0)

    def try_accept(h_cand):
        nonlocal best_h, best_c5
        c = eval_c5(h_cand)
        if c < best_c5 - 1e-12:
            best_h = h_cand.copy()
            best_c5 = c
            return True
        return False

    # Initialize from parent
    try:
        h_init = np.array(initial_h_values, dtype=float).reshape(-1)
        if h_init.size > 0 and np.all(np.isfinite(h_init)):
            n_init = len(h_init)
            h_init = project_to_target(h_init, n_init / 2.0)
            best_h = h_init.copy()
            best_c5 = eval_c5(h_init)
    except Exception:
        pass

    if best_h is None:
        n_def = 126
        m_def = n_def // 2
        t = np.linspace(0, 1, m_def)
        q_def = 0.5 + 0.28 * np.sin(2 * np.pi * 2.5 * t) + 0.08 * np.sin(2 * np.pi * 5.0 * t)
        best_h = build_sym(np.clip(q_def, 0, 1), n_def)
        best_c5 = eval_c5(best_h)

    n_parent = len(best_h)
    reserve = max(20.0, 0.1 * budget_s)

    # Phase 1: Anti-symmetric optimization with multi-stage annealing
    if n_parent % 2 == 0:
        q = q_of_h(best_h, n_parent)
        h_sym = build_sym(q, n_parent)
        if eval_c5(h_sym) < best_c5:
            best_h = h_sym.copy()
            best_c5 = eval_c5(best_h)
            q = q_of_h(best_h, n_parent)

        # Multi-pass annealing
        tau_passes = [
            [0.08, 0.05, 0.03, 0.018, 0.01, 0.006],
            [0.01, 0.006, 0.0035, 0.002, 0.001, 0.0006],
            [0.003, 0.0018, 0.001, 0.0006, 0.0003, 0.00015],
        ]
        iters_per_stage = [150, 200, 300]

        for pi in range(3):
            if remaining() < reserve + 30:
                break
            tl = tau_passes[pi]
            mi = iters_per_stage[pi]
            q, c5_val = opt_smooth(q, n_parent, tl, max_per_stage=mi, gtol=1e-10)
            h_new = build_sym(q, n_parent)
            if c5_val < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_val

    # Phase 2: Full (non-symmetric) optimization
    if remaining() > reserve + 50 and len(best_h) >= 50:
        n_cur = len(best_h)
        tau_f = [0.05, 0.03, 0.015, 0.008, 0.004, 0.002, 0.001]
        h_f, c5_f = opt_full(best_h, n_cur, tau_f, max_per_stage=100, gtol=1e-9)
        if c5_f < best_c5:
            best_h = h_f.copy()
            best_c5 = c5_f

    # Phase 3: Try different n values
    n_base = len(best_h)
    deltas = [-8, -6, -4, -3, -2, -1, 1, 2, 3, 4, 6, 8, 10, 14]
    n_cands = set()
    for d in deltas:
        nc = n_base + d
        if nc >= 48 and nc % 2 == 0:
            n_cands.add(nc)
    n_cands = sorted(n_cands)

    for n_try in n_cands:
        if remaining() < reserve + 20:
            break
        h_try = interp_to(best_h, n_try)
        q_try = q_of_h(h_try, n_try)
        h_try = build_sym(q_try, n_try)
        tau_n = [0.04, 0.02, 0.01, 0.005, 0.0025, 0.001, 0.0005]
        q_try, c5_try = opt_smooth(q_try, n_try, tau_n, max_per_stage=120, gtol=1e-9)
        h_try = build_sym(q_try, n_try)
        if c5_try < best_c5:
            best_h = h_try.copy()
            best_c5 = c5_try
            # Deeper refinement at this n
            if remaining() > reserve + 15:
                tau_deep = [0.005, 0.003, 0.0015, 0.0008, 0.0004]
                q_deep = q_of_h(best_h, len(best_h))
                q_deep, c5_deep = opt_smooth(q_deep, len(best_h), tau_deep,
                                             max_per_stage=200, gtol=1e-10)
                h_deep = build_sym(q_deep, len(best_h))
                if c5_deep < best_c5:
                    best_h = h_deep.copy()
                    best_c5 = c5_deep
        if remaining() > reserve + 12:
            h_f2, c5_f2 = opt_full(h_try, n_try, [0.03, 0.015, 0.008, 0.004],
                                   max_per_stage=70, gtol=1e-8)
            if c5_f2 < best_c5:
                best_h = h_f2.copy()
                best_c5 = c5_f2

    # Phase 4: Random restarts
    n_r = len(best_h)
    if n_r % 2 != 0:
        n_r -= 1
    max_restarts = min(25, max(4, int(remaining() / 25.0)))

    for trial in range(max_restarts):
        if remaining() < reserve + 8:
            break
        q_r = make_random_q(n_r, rng)
        h_r = build_sym(q_r, n_r)
        tau_r = [0.04, 0.02, 0.01, 0.005, 0.0025, 0.001]
        q_r, c5_r = opt_smooth(q_r, n_r, tau_r, max_per_stage=80, gtol=1e-8)
        h_r = build_sym(q_r, n_r)
        if c5_r < best_c5:
            best_h = h_r.copy()
            best_c5 = c5_r
            # Deepen the new best
            if remaining() > reserve + 20:
                q_r2 = q_of_h(best_h, len(best_h))
                tau_r2 = [0.008, 0.004, 0.002, 0.001, 0.0005]
                q_r2, c5_r2 = opt_smooth(q_r2, len(best_h), tau_r2,
                                         max_per_stage=180, gtol=1e-10)
                h_r2 = build_sym(q_r2, len(best_h))
                if c5_r2 < best_c5:
                    best_h = h_r2.copy()
                    best_c5 = c5_r2

    # Phase 5: Perturbation-based local search on best solution
    if remaining() > reserve + 10 and best_h is not None:
        n_pert = len(best_h)
        if n_pert % 2 == 0:
            q_pert = q_of_h(best_h, n_pert)
            for _ in range(min(5, int(remaining() / 15.0))):
                if remaining() < reserve + 5:
                    break
                # Small random perturbation
                noise = rng.randn(len(q_pert)) * 0.02
                q_p = np.clip(q_pert + noise, 0.0, 1.0)
                tau_p = [0.005, 0.003, 0.0015, 0.0008, 0.0004]
                q_p, c5_p = opt_smooth(q_p, n_pert, tau_p, max_per_stage=100, gtol=1e-10)
                h_p = build_sym(q_p, n_pert)
                if c5_p < best_c5:
                    best_h = h_p.copy()
                    best_c5 = c5_p
                    q_pert = q_of_h(best_h, n_pert)

    # Phase 6: Final deep polish
    if remaining() > 12 and best_h is not None:
        n_fin = len(best_h)
        if n_fin % 2 == 0:
            q_fin = q_of_h(best_h, n_fin)
            tau_fin = [0.004, 0.002, 0.001, 0.0005, 0.00025, 0.0001]
            q_fin, c5_fin = opt_smooth(q_fin, n_fin, tau_fin, max_per_stage=250, gtol=1e-11)
            h_fin = build_sym(q_fin, n_fin)
            if c5_fin < best_c5:
                best_h = h_fin.copy()
                best_c5 = c5_fin

        if remaining() > 8:
            h_fin2, c5_fin2 = opt_full(best_h, len(best_h),
                                       [0.008, 0.004, 0.002, 0.001, 0.0005],
                                       max_per_stage=120, gtol=1e-10)
            if c5_fin2 < best_c5:
                best_h = h_fin2.copy()
                best_c5 = c5_fin2

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_target(best_h, len(best_h) / 2.0)
    best_c5 = eval_c5(best_h)

    return best_h, best_c5, len(best_h)
