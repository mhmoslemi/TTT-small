    t0 = corrs0.max()

```python
def solve_active_epigraph(h0, n, target_sum, maxiter, active_count=10, tol=1e-10):
    """
    Solve min t s.t. corr_k <= t for active k, 0<=h<=1, sum(h)=target_sum
    with iterative active set management.
    """
    h = h0.copy()
    dx = 2.0 / n
    
    for outer in range(20):
        corrs = compute_corrs(h)
        t_current = corrs.max()
        
        # Select active set: top 'active_count' + any within tol of max
        sorted_k = np.argsort(corrs)[::-1]
        active = sorted_k[:active_count]
        # Add any within tol
        near_max = np.where(client > t_current - tol)[0]
        active = np.unique(np.concatenate([active, near_max]))
        
        M = len(active)
        
        # Set up SLSQP problem
        # Variables: x = [h(0), ..., h(n-1), t]
        
        def obj(x):
            return x[n]
        
        def obj_grad(x):
            g = np.zeros(n+1)
            g[n] = 1.0
            return g
        
        active_list = active.tolist()
        
        def ineq_fun(x):
            h_x = x[:n]
            t_x = x[n]
            c = compute_corrs(h_x)
            return np.array([t_x - c[k] for k in active_list])
        
        def ineq_jac(x):
            h_x = x[:n]
            jac = np.zeros((M, n+1))
            for i, k in enumerate(active_list):
                g = grad_corr_k(h_x, k, n)
                jac[i, :n] = -g
                jac[i, n] = 1.0
            return jac
        
        bounds = [(0.0, 1.0)] * n + [(0.0, t_current + 1.0)]
        cons = [
            {'type': 'eq', 'fun': lambda x: x[:n].sum() - target_sum,
             'jac': lambda x: np.append(np.ones(n), 0.0)},
            {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
        ]
        
        x0 = np.append(h.copy(), t_current)
        
        try:
            res = minimize(obj, x0, jac=obj_grad, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-15})
            h_new = np.clip(res.x[:n], 0, 1)
            h_new = project_feasible(h_new, target_sum)
            t_new = max(res.x[n], 0)
        except Exception:
            h_new = h.copy()
            t_new = t_current
        
        c5_new = c5_of_h(h_new, dx)
        c5_old = c5_of_h(h, dx)
        
        if c5_new < c5_old:
            h = h_new
        else:
            # No improvement, check for constraint violations
            corrs_new = compute_corrs(h)
            violated = np.where(corrs_new > t_current + tol)[0]
            if len(violated) == 0:
                break
            # Add violated constraints
            active = np.unique(np.concatenate([active, violated]))
            active_count = min(active_count + 5, 2*n - 1)
    
    return h, c5_of_h(h, dx)

```python
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time


def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-10:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h


def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def c5_of_h(h, dx):
    return float(compute_corrs(h).max() * dx)


def grad_corr_k(h, k, n):
    offset = k - (n - 1)
    grad = np.zeros(n)
    if offset == 0:
        grad[:] = 1.0 - 2.0 * h
        return grad
    i_lo = max(0, -offset)
    i_hi = min(n - 1, n - 1 - offset)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx + offset] += (1.0 - h[idx])
    grad[idx] += (-h[idx + offset])
    return grad


def smooth_max_val(h, dx, beta):
    corrs = compute_corrs(h)
    return float(logsumexp(beta * corrs) / beta * dx)


def smooth_max_grad(h, dx, beta):
    n = len(h)
    corrs = compute_corrs(h)
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones_like(corrs) / len(corrs)
    else:
        w /= w_sum
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        grad += w[k] * grad_corr_k(h, k, n)
    return grad * dx


def slsqp_smoothmax(h0, dx, target_sum, beta, maxiter):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        return smooth_max_val(x, dx, beta)
    
    def g(x):
        return smooth_max_grad(x, dx, beta)
    
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum,
             'jac': lambda x: np.ones(n)}]
    
    try:
        res = minimize(obj, h0, jac=g, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        h_out = project_feasible(res.x, target_sum)
    except Exception:
        h_out = project_feasible(h0, target_sum)
    return h_out, c5_of_h(h_out, dx)


def solve_active_epigraph(h0, dx, target_sum, maxiter_per_solve=300, 
                          init_active=10, max_active=40, outer_iters=15,
                          tol=1e-12):
    """
    Active-set epigraph solver:
    min t s.t. corr_k <= t for k in active_set, 0<=h<=1, sum(h)=target_sum
    with iterative constraint generation.
    """
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    
    for outer in range(outer_iters):
        corrs = compute_corrs(h)
        t_cur = corrs.max()
        c5_cur = float(t_cur * dx)
        
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h.copy()
        
        # Select active set
        sorted_k = np.argsort(corrs)[::-1]
        active = sorted_k[:init_active]
        # Add near-max constraints
        near = np.where(corrs >= t_cur * (1 - 1e-4) - 1e-10)[0]
        active = np.unique(np.concatenate([active, near]))
        if len(active) > max_active:
            active = sorted_k[:max_active]
        
        M = len(active)
        active_list = active.tolist()
        
        # SLSQP setup
        def obj(x):
            return x[n]
        
        def obj_grad(x):
            g = np.zeros(n + 1)
            g[n] = 1.0
            return g
        
        def ineq_fun(x):
            h_x = x[:n]
            t_x = x[n]
            c = np.correlate(h_x, 1.0 - h_x, mode='full')
            return t_x - c[active]
        
        def ineq_jac(x):
            h_x = x[:n]
            jac = np.zeros((M, n + 1))
            for i, k in enumerate(active_list):
                g = grad_corr_k(h_x, k, n)
                jac[i, :n] = -g
                jac[i, n] = 1.0
            return jac
        
        bounds = [(0.0, 1.0)] * n + [(0.0, t_cur + 2.0)]
        cons = [
            {'type': 'eq', 'fun': lambda x: x[:n].sum() - target_sum,
             'jac': lambda x: np.append(np.ones(n), 0.0)},
            {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
        ]
        
        x0 = np.append(h.copy(), t_cur)
        
        try:
            res = minimize(obj, x0, jac=obj_grad, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_solve, 'ftol': 1e-14})
            h_new = np.clip(res.x[:n], 0.0, 1.0)
            h_new = project_feasible(h_new, target_sum)
        except Exception:
            h_new = h.copy()
        
        c5_new = c5_of_h(h_new, dx)
        
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
        elif c5_new < c5_cur - 1e-14:
            h = h_new
        else:
            # Check for constraint violations
            corrs_new = compute_corrs(h)
            violated = np.where(corrs_new > corrs[active].max() + 1e-8)[0]
            if len(violated) == 0:
                break
            # Expand active set
            active = np.unique(np
