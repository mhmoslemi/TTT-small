import numpy as np
from scipy.optimize import linprog
import time

def eval_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / len(h))

def build_sym(q, n):
    m = n // 2
    h = np.empty(n)
    h[:m] = q
    h[m:] = 1.0 - q[::-1]
    return h

def to_sym_q(h, n):
    m = n // 2
    return 0.5 * (h[:m] + 1.0 - h[n-1:m-1:-1])

def project_to_target(x, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(x - mid, 0.0, 1.0).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.any():
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def grad_h_for_lag(h, d, n):
    """Compute gradient of correlation at lag d with respect to h."""
    g = np.zeros(n)
    lo1 = max(0, d)
    hi1 = min(n - 1, n - 1 + d)
    if lo1 <= hi1:
        g[lo1:hi1+1] += 1.0 - h[lo1-d:hi1+1-d]
    lo2 = max(0, -d)
    hi2 = min(n - 1, n - 1 - d)
    if lo2 <= hi2:
        g[lo2:hi2+1] += -h[lo2+d:hi2+1+d]
    return g

def grad_q_for_lag(h, d, n):
    """Compute gradient of correlation at lag d with respect to q (symmetric vars)."""
    gh = grad_h_for_lag(h, d, n)
    m = n // 2
    return gh[:m] - gh[n-1:m-1:-1]

def select_active(C, margin_frac=0.02, pad=2):
    """Select active set: indices where C is within margin of max, plus neighbors."""
    mx = C.max()
    thresh = mx - max(1e-10, margin_frac * mx)
    idx = np.where(C >= thresh)[0]
    expanded = set()
    for a in idx:
        for delta in range(-pad, pad+1):
            if 0 <= a + delta <= len(C) - 1:
                expanded.add(a + delta)
    return np.sort(np.array(list(expanded), dtype=int))

def slp_sym(q, n, active, R, move_pen=0.0, max_iter=50, eps=1e-10):
    """SLP on symmetric subspace. Returns best q and c5 found."""
    m = n // 2
    dx = 2.0 / n
    h = build_sym(q, n)
    C = corr_full(h)
    c_cur = C.max()
    best_q = q.copy()
    best_c = c_cur
    trust = R
    
    for it in range(max_iter):
        # Recompute active set
        h = build_sym(q, n)
        C = corr_full(h)
        c_cur = C.max()
        if c_cur < best_c:
            best_c = c_cur
            best_q = q.copy()
        active = select_active(C)
        
        # Build LP
        # Variables: delta (m), t (1)
        # Total: m + 1
        nvar = m + 1
        c_obj = np.zeros(nvar)
        c_obj[m] = 1.0  # minimize t
        
        A_ub = []
        b_ub = []
        for j in active:
            d = j - (n - 1)
            gq = grad_q_for_lag(h, d, n) * dx
            # Constraint: dx * gq @ delta <= t - C[j]
            # i.e., dx * gq @ delta - t <= -C[j]
            row = np.zeros(nvar)
            row[:m] = gq
            row[m] = -1.0
            A_ub.append(row)
            b_ub.append(-C[j])
        
        if not A_ub:
            break
        
        A_ub = np.array(A_ub)
        b_ub = np.array(b_ub)
        
        bounds = []
        for i in range(m):
            lo = max(-q[i], -trust)
            hi = min(1.0 - q[i], trust)
            if lo > hi:
                lo = hi = 0
            bounds.append((lo, hi))
        bounds.append((0.0, c_cur + 1.0))
        
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        except Exception:
            break
        
        if res.status != 0:
            break
        
        delta = res.x[:m]
        t_new = res.x[m]
        
        # Line search
        accepted = False
        for alpha in [1.0, 0.7, 0.5, 0.3, 0.2, 0.1]:
            q_new = np.clip(q + alpha * delta, 0.0, 1.0)
            h_new = build_sym(q_new, n)
            c_new = eval_c5(h_new)
            if c_new < c_cur - eps:
                q = q_new
                c_cur = c_new
                accepted = True
                trust = min(2.0 * trust, 0.2)
                break
        
        if accepted:
            if c_cur < best_c:
                best_c = c_cur
                best_q = q.copy()
        else:
            trust *= 0.5
            if trust < 1e-6:
                break
    
    return best_q, best_c

def slp_full(h, n, active, R, move_pen=0.0, max_iter=30, eps=1e-10):

    dx = 2.0 / n
    C = corr_full(h)
    c_cur = C.max()
    best_h = h.copy()
    best_c = c_cur
    trust = R
    target = n / 2.0

        nvar = n + 1

            gh = grad_h_for_lag(h, d, n) * dx
            row = np.zeros(nvar)
            row[:n] = gh
            row[n] = -1.0
            A_ub.append(row)
            b_ub.append(-C[j])

        A_ub = np.array(A_ub)
        b_ub = np.array(b_ub)
        
        bounds = []

def interp_h(h_old, n_new):
    n_old = len(h_old)
    if n_new == n_old:
        return h_old.copy()
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_to_target(h_new, n_new / 2.0)

import numpy as np

def eval_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))


def corr_scaled(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)


def build_sym(q, n):
    m = n // 2
    h = np.empty(n)
    h[:m] = q
    h[m:] = 1.0 - q[::-1]
    return h


def to_sym_q(h, n):
    m = n // 2
    return 0.5 * (h[:m] + 1.0 - h[n - 1:m - 1:-1])


def project_to_target(x, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(x - mid, 0.0, 1.0).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.any():
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)


def grad_h_lag(h, d, n):
    g = np.zeros(n)
    lo1 = max(0, d)
    hi1 = min(n - 1, n - 1 + d)
    if lo1 <= hi1:
        g[lo1:hi1 + 1] += 1.0 - h[lo1 - d:hi1 + 1 - d]
    lo2 = max(0, -d)
    hi2 = min(n - 1, n - 1 - d)
    if lo2 <= hi2:
        g[lo2:hi2 + 1] += -h[lo2 + d:hi2 + 1 + d]
    return g


def grad_q_lag(h, d, n):
    gh = grad_h_lag(h, d, n)
    m = n // 2
    return gh[:m] - gh[n - 1:m - 1:-1]


def select_active(C, margin_abs=0.005, pad=2):
    mx = C.max()
    thresh = mx - margin_abs
    idx = set(np.where(C >= thresh)[0].tolist())
    expanded = set()
    for a in idx:
        for delta in range(-pad, pad + 1):
            if 0 <= a + delta <= len(C) - 1:
                expanded.add(a + delta)
    return np.sort(np.array(list(expanded), dtype=int))


def slp_sym(q, n, R0, max_iter=50, margin_abs=0.005, eps=1e-11):

    best_c = c_cur

        if len(active) == 0:

        if remaining() > reserve + 15:
            q = to_sym_q(best_h, n_base)
            q_gd, c5_gd = gradient_descent_sym(q, n_base, steps=300, lr=0.001, margin_abs=0.003)
            if c5_gd < best_c5:
                best_c5 = c5_gd
                best_h = build_sym(q_gd, n_base).copy()
            # SLP after GD
            if remaining() > reserve + 10:
                q = to_sym_q(best_h, n_base)
                q_slp2, c5_slp2 = slp_sym(q, n_base, 0.02, max_iter=40, margin_abs=0.003)
                if c5_slp2 < best_c5:
                    best_c5 = c5_slp2
                    best_h = build_sym(q_slp2, n_base).copy()

    # Phase 2: Multiple n values
    if remaining() > reserve + 25:
        n_base = len(best_h)
        candidates = set()
        for delta in [-12, -10, -8, -6, -5, -4, -3, -2, -1, 1, 2, 3, 4, 5, 6, 8, 10, 12]:
            nc = n_base + delta
            if nc >= 50 and nc % 2 == 0 and nc <= 260:
                candidates.add(nc)
        for nc in [84, 96, 102, 108, 114, 120, 126, 132, 138, 144, 150, 156, 162]:
            if nc >= 50 and nc <= 260:
                candidates.add(nc)
        n_list = sorted(candidates)

        for n_try in n_list:
            if remaining() < reserve + 12:
                break
            if n_try == n_base:
                continue
            h_try = interp_h(best_h, n_try)
            q_try = to_sym_q(h_try, n_try)
            h_try = build_sym(q_try, n_try)
            c5_try = eval_c5(h_try)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()

            for R0 in [0.05, 0.025]:
                if remaining() < reserve + 8:
                    break
                q_s, c5_s = slp_sym(q_try, n_try, R0, max_iter=35, margin_abs=0.004)
                if c5_s < best_c5:
                    best_c5 = c5_s
                    best_h = build_sym(q_s, n_try).copy()
                    q_try = q_s

            if remaining() > reserve + 8:
                q_gd2, c5_gd2 = gradient_descent_sym(q_try, n_try, steps=150, lr=0.001, margin_abs=0.003)
                if c5_gd2 < best_c5:
                    best_c5 = c5_gd2
                    best_h = build_sym(q_gd2, n_try).copy()

    # Phase 3: Full (non-symmetric) SLP
    if remaining() > reserve + 20:
        n_cur = len(best_h)
        h_f, c5_f = slp_full(best_h, n_cur, 0.025, max_iter=30, margin_abs=0.004)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

        if remaining() > reserve + 12:
            h_f2, c5_f2 = slp_full(best_h, len(best_h), 0.015, max_iter=25, margin_abs=0.003)
            if c5_f2 < best_c5:
                best_c5 = c5_f2
                best_h = h_f2.copy()

    # Phase 4: Random restarts
    if remaining() > reserve + 10:
        n_r = len(best_h)
        if n_r % 2 != 0:
            n_r -= 1
        q_best = to_sym_q(best_h, n_r) if n_r == len(best_h) else to_sym_q(interp_h(best_h, n_r), n_r)
        max_restarts = min(15, max(3, int(remaining() / 25.0)))
        for trial in range(max_restarts):
            if remaining() < reserve + 8:
                break
            if trial < max_restarts // 2:
                q_r = q_best + rng.randn(n_r // 2) * 0.04
            else:
                q_r = make_fallback_q(n_r, seed=seed + 1000 + trial)
            q_r = np.clip(q_r, 0.0, 1.0)
            c5_r = eval_c5(build_sym(q_r, n_r))
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = build_sym(q_r, n_r).copy()
            q_s, c5_s = slp_sym(q_r, n_r, 0.04, max_iter=25, margin_abs=0.004)
            if c5_s < best_c5:
                best_c5 = c5_s
                best_h = build_sym(q_s, n_r).copy()
                q_best = q_s

    # Phase 5: Final fine-grained SLP
    if remaining() > 12 and best_h is not None:
        n_fin = len(best_h)
        if n_fin % 2 == 0:
            q_fin = to_sym_q(best_h, n_fin)
            for R_final in [0.01, 0.005, 0.003]:
                if remaining() < 8:
                    break
                q_f, c5_f = slp_sym(q_fin, n_fin, R_final, max_iter=30, margin_abs=0.002)
                if c5_f < best_c5:
                    best_c5 = c5_f
                    best_h = build_sym(q_f, n_fin).copy()
                    q_fin = q_f

        if remaining() > 8:
            h_pf, c5_pf = slp_full(best_h, len(best_h), 0.008, max_iter=20, margin_abs=0.002)
            if c5_pf < best_c5:
                best_c5 = c5_pf
                best_h = h_pf.copy()

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_target(best_h, len(best_h) / 2.0)
    best_c5 = eval_c5(best_h)

    return best_h, best_c5, len(best_h)
