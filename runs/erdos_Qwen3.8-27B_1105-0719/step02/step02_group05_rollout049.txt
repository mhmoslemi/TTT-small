        cfun, cjac = ep.make_constraint(i)
        cons.append({'type': 'ineq', 'fun': cfun, 'jac': cjac})
    bounds = [(0.0, 5.0)] + [(0.0, 1.0)] * m
    x0 = np.concatenate([[t0], q0])
    try:
        res = minimize(ep.obj, x0, jac=ep.obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return np.clip(res.x[1:], 0, 1), res.x[0]
    except Exception:
        return q0.copy(), t0


def solve_full_epigraph(h0, n, active_k, t0, target_sum, maxiter=500):
    ep = FullEpigraph(n, active_k, target_sum)
    cons = []
    for i in range(len(active_k)):
        cfun, cjac = ep.make_constraint(i)
        cons.append({'type': 'ineq', 'fun': cfun, 'jac': cjac})
    cons.append({'type': 'eq', 'fun': ep.sum_constraint, 'jac': ep.sum_constraint_grad})
    bounds = [(0.0, 5.0)] + [(0.0, 1.0)] * n
    x0 = np.concatenate([[t0], h0])
    try:
        res = minimize(ep.obj, x0, jac=ep.obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(np.clip(res.x[1:], 0, 1), target_sum), res.x[0]
    except Exception:
        return project_feasible(h0, target_sum), t0


def q_true_max(q, n):
    h = build_h_from_q(q, n)
    return exact_score(h)[0]


def full_true_max(h, n):
    return exact_score(h)[0]


def coordinate_descent_q(q, n, n_sweeps=3, tol=1e-14):
    m = n // 2
    for sweep in range(n_sweeps):
        improved = False
        for i in range(m):
            best_val = q_true_max(q, n)
            for delta in [0.005, -0.005, 0.001, -0.001, 0.01, -0.01, 0.0005, -0.0005]:
                q_try = q.copy()
                q_try[i] = np.clip(q[i] + delta, 0, 1)
                v_try = q_true_max(q_try, n)
                if v_try < best_val - tol:
                    best_val = v_try
                    q = q_try.copy()
                    improved = True
        if not improved:
            break
    return q


def coordinate_descent_full(h, n, target_sum, n_sweeps=2, tol=1e-14):
    for sweep in range(n_sweeps):
        improved = False
        for i in range(0, n, 2):
            best_val = full_true_max(h, n)
            for delta in [0.005, -0.005, 0.001, -0.001, 0.01, -0.01]:
                h_try = h.copy()
                h_try[i] = np.clip(h[i] + delta, 0, 1)
                v_try = full_true_max(h_try, n)
                if v_try < best_val - tol:
                    best_val = v_try
                    h = h_try.copy()
                    improved = True
        if not improved:
            break
    return h


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)

    def elapsed():
        return time.time() - t_start

    def remaining():
        return budget_s - elapsed()

    # Initialize from parent
    h_parent = np.array(initial_h_values, dtype=float)
    n = len(h_parent)
    m = n // 2
    target_sum = n / 2.0
    h_parent = project_feasible(h_parent, target_sum)

    score_parent, _ = exact_score(h_parent)
    best_h = h_parent.copy()
    best_c5 = score_parent
    q_best = q_from_h(h_parent, n)

    # Phase 1: Variable-smooth L-BFGS-B in q-space with aggressive schedule
    betas_q = [15, 30, 60, 120, 250, 500, 1000, 2000, 4000, 8000, 16000]
    for beta in betas_q:
        if remaining() < 25:
            break
        so = QSmoothObj(n, beta)
        bounds = [(0.0, 1.0)] * m
        try:
            res = minimize(so.value, q_best, jac=so.grad, method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': 10000, 'ftol': 1e-22, 'xtol': 1e-18, 'gtol': 1e-16})
            q_new = np.clip(res.x, 0, 1)
            h_new = build_h_from_q(q_new, n)
            s_new, _ = exact_score(h_new)
            if s_new < best_c5:
                best_c5 = s_new
                best_h = h_new.copy()
                q_best = q_new.copy()
        except Exception:
            pass

    # Phase 2: Active-set SLSQP in q-space
    if remaining() > 80:
        h_curr = build_h_from_q(q_best, n)
        corrs = compute_corrs(h_curr)
        dx = 2.0 / n
        current_max = corrs.max() * dx
        for n_active in [6, 10, 14, 18, 22, 28, 34]:
            if remaining() < 40:
                break
            top_idx = np.argsort(corrs)[::-1][:n_active]
            active_k = sorted(top_idx.tolist())
            t0 = current_max * 1.0005 + 1e-8
            q_new, t_sol = solve_q_epigraph(q_best, n, active_k, t0, maxiter=1000)
            h_new = build_h_from_q(q_new, n)
            s_new, corrs_new = exact_score(h_new)
            if s_new < best_c5:
                best_c5 = s_new
                best_h = h_new.copy()
                q_best = q_new.copy()
                corrs = corrs_new
                current_max = s_new
            hidden = corrs_new > t_sol + 1e-7
            if hidden.sum() > 0:
                new_active = sorted(set(active_k) | set(np.where(hidden)[0].tolist()))
                if len(new_active) > 45:
                    idx = sorted(new_active, key=lambda k: -corrs_new[k])[:45]
                    active_k = sorted(idx)

    # Phase 3: Coordinate descent polish in q-space
    if remaining() > 30:
        q_cd = coordinate_descent_q(q_best, n, n_sweeps=5)
        h_cd = build_h_from_q(q_cd, n)
        s_cd, _ = exact_score(h_cd)
        if s_cd < best_c5:
            best_c5 = s_cd
            best_h = h_cd.copy()
            q_best = q_cd.copy()

    # Phase 4: Random restarts
    if remaining() > 200:
        restart_sigmas = [0.03, 0.015, 0.008, 0.004, 0.002, 0.001]
        for sigma in restart_sigmas:
            if remaining() < 50:
                break
            n_restarts = 4 if sigma > 0.01 else 6
            for r in range(n_restarts):
                if remaining() < 25:
                    break
                q_pert = np.clip(q_best + rng.randn(m) * sigma, 0, 1)
                for beta in [400, 2000]:
                    if remaining() < 15:
                        break
                    so = QSmoothObj(n, beta)
                    bounds = [(0.0, 1.0)] * m
                    try:
                        res = minimize(so.value, q_pert, jac=so.grad, method='L-BFGS-B',
                                       bounds=bounds,
                                       options={'maxiter': 3000, 'ftol': 1e-20, 'xtol': 1e-16})
                        q_try = np.clip(res.x, 0, 1)
                        h_try = build_h_from_q(q_try, n)
                        s_try, _ = exact_score(h_try)
                        if s_try < best_c5:
                            best_c5 = s_try
                            best_h = h_try.copy()
                            q_best = q_try.copy()
                            q_pert = q_try.copy()
                    except Exception:
                        pass

    # Phase 5: Full-space smooth optimization (break symmetry)
    if remaining() > 120:
        h_full = best_h.copy()
        for beta in [300, 1000, 3000, 8000]:
            if remaining() < 25:
                break
            fs = FullSmoothObj(n, beta, target_sum)
            bounds = [(0.0, 1.0)] * n
            try:
                res = minimize(fs.value, h_full, jac=fs.grad, method='L-BFGS-B',
                               bounds=bounds,
                               options={'maxiter': 5000, 'ftol': 1e-20, 'xtol': 1e-16})
                h_new = project_feasible(np.clip(res.x, 0, 1), target_sum)
                s_new, _ = exact_score(h_new)
                if s_new < best_c5:
                    best_c5 = s_new
                    best_h = h_new.copy()
                h_full = h_new.copy()
            except Exception:
                h_full = project_feasible(h_full, target_sum)

    # Phase 6: Full-space active-set SLSQP
    if remaining() > 80:
        h_curr = best_h.copy()
        corrs = compute_corrs(h_curr)
        current_max = corrs.max() * (2.0 / n)
        for n_active in [6, 10, 14, 18]:
            if remaining() < 40:
                break
            top_idx = np.argsort(corrs)[::-1][:n_active]
            active_k = sorted(top_idx.tolist())
            t0 = current_max * 1.0005 + 1e-8
            h_new, t_sol = solve_full_epigraph(h_curr, n, active_k, t0, target_sum, maxiter=800)
            s_new, corrs_new = exact_score(h_new)
            if s_new < best_c5:
                best_c5 = s_new
                best_h = h_new.copy()
                h_curr = h_new.copy()
                corrs = corrs_new
                current_max = s_new
            hidden = corrs_new > t_sol + 1e-7
            if hidden.sum() > 0:
                new_active = sorted(set(active_k) | set(np.where(hidden)[0].tolist()))
                if len(new_active) > 35:
                    idx = sorted(new_active, key=lambda k: -corrs_new[k])[:35]
                    active_k = sorted(idx)

    # Phase 7: Coordinate descent in full space
    if remaining() > 20:
        h_cd2 = coordinate_descent_full(best_h, n, target_sum, n_sweeps=3)
        s_cd2, _ = exact_score(h_cd2)
        if s_cd2 < best_c5:
            best_c5 = s_cd2
            best_h = h_cd2.copy()

    # Phase 8: Try different n values
    if remaining() > 100:
        for n_try in [124, 128, 132, 136, 140, 144]:
            if remaining() < 50:
                break
            ts_try = n_try / 2.0
            x_orig = np.linspace(0.0, 2.0, len(best_h))
            x_new = np.linspace(0.0, 2.0, n_try)
            h_interp = np.interp(x_new, x_orig, best_h)
            h_interp = project_feasible(h_interp, ts_try)
            if n_try % 2 == 0:
                q_interp = q_from_h(h_interp, n_try)
                for beta in [500, 3000]:
                    if remaining() < 20:
                        break
                    so = QSmoothObj(n_try, beta)
                    bounds = [(0.0, 1.0)] * (n_try // 2)
                    try:
                        res = minimize(so.value, q_interp, jac=so.grad, method='L-BFGS-B',
                                       bounds=bounds,
                                       options={'maxiter': 3000, 'ftol': 1e-20, 'xtol': 1e-16})
                        q_try = np.clip(res.x, 0, 1)
                        h_try = build_h_from_q(q_try, n_try)
                        s_try, _ = exact_score(h_try)
                        if s_try < best_c5:
                            best_c5 = s_try
                            best_h = h_try.copy()
                    except Exception:
                        pass

    # Phase 9: Final ultra-precise polish
    n_final = len(best_h)
    ts_final = n_final / 2.0

    if remaining() > 25 and n_final % 2 == 0:
        q_fin = q_from_h(best_h, n_final)
        so = QSmoothObj(n_final, 20000)
        bounds = [(0.0, 1.0)] * (n_final // 2)
        try:
            res = minimize(so.value, q_fin, jac=so.grad, method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': 20000, 'ftol': 1e-22, 'xtol': 1e-18})
            q_fin = np.clip(res.x, 0, 1)
            h_fin = build_h_from_q(q_fin, n_final)
            s_fin, _ = exact_score(h_fin)
            if s_fin < best_c5:
                best_c5 = s_fin
                best_h = h_fin.copy()
        except Exception:
            pass

    if remaining() > 15:
        q_cd3 = coordinate_descent_q(q_from_h(best_h, n_final), n_final, n_sweeps=8, tol=1e-16)
        h_cd3 = build_h_from_q(q_cd3, n_final)
        s_cd3, _ = exact_score(h_cd3)
        if s_cd3 < best_c5:
            best_c5 = s_cd3
            best_h = h_cd3.copy()

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    n_out = len(best_h)
    ts_out = n_out / 2.0
    best_h = project_feasible(best_h, ts_out)
    final_score, _ = exact_score(best_h)
    best_c5 = min(best_c5, final_score)

    return best_h, best_c5, n_out
