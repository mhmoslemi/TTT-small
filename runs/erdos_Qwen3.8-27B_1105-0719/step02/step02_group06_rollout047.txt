Wait, let me be more precise. The raw correlation at lag k (shift d) is:
R_k = sum_{i=i_lo}^{i_hi} h[i] * (1 - h[i+d])

If I change h[a] by +delta and h[b] by -delta (where a=j, b=n-1-j):
- Term i=a: h[a]*(1-h[a+d]) → (h[a]+delta)*(1-h[a+d]) = old + delta*(1-h[a+d])
- Term i=a+d (if in range): h[a+d]*(1-h[a]) → h[a+d]*(1-h[a]-delta) = old - delta*h[a+d]
- Term i=b: h[b]*(1-h[b+d]) → (h[b]-delta)*(1-h[b+d]) = old - delta*(1-h[b+d])
- Term i=b+d (if in range): h[b+d]*(1-h[b]) → h[b+d]*(1-h[b]+delta) = old + delta*h[b+d]

So the change in R_k is:
delta_R_k = delta * [(1-h[a+d])_in_range - h[a+d]_in_range - (1-h[b+d])_in_range + h[b+d]_in_range]

Where _in_range means the term is only counted if the index is in [i_lo, i_hi].

3. Evaluating at breakpoints and at the boundaries of the feasible interval

import numpy as np
from scipy.optimize import minimize
import time

def eval_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def corr_raw(h):
    return np.correlate(h, 1.0 - h, mode='full')

def build_sym_h(q):
    m = len(q)
    h = np.empty(2 * m)
    h[:m] = q
    h[m:] = 1.0 - q[::-1]
    return h

def sym_from_h(h):
    n = len(h)
    m = n // 2
    q = 0.5 * (h[:m] + 1.0 - h[n-1:m-1:-1])
    return np.clip(q, 0.0, 1.0)

def project_feasible(x, target_sum):
    n = len(x)
    x = np.asarray(x, dtype=float).copy()
    x[~np.isfinite(x)] = 0.5
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(x + mid, 0.0, 1.0).sum()
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    h = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-13:
        interior = (h > 1e-12) & (h < 1.0 - 1e-12)
        if interior.sum() > 0:
            h[interior] += diff / interior.sum()
    return np.clip(h, 0.0, 1.0)

def transfer_to_n(h_old, n_new):
    n_old = len(h_old)
    x_old = (np.arange(n_old) + 0.5) * 2.0 / n_old
    x_new = (np.arange(n_new) + 0.5) * 2.0 / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new / 2.0)

def smooth_max_sym_obj(x, n_full, beta):
    h = build_sym_h(x)
    dx = 2.0 / n_full
    corrs = corr_raw(h) * dx
    max_c = corrs.max()
    shifted = corrs - max_c
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(len(corrs)) / len(corrs)
    else:
        w /= w_sum
    val = max_c + np.log(w_sum) / beta
    grad_q = np.zeros(len(x))
    for k in range(len(corrs)):
        if w[k] < 1e-14:
            continue
        d = k - (n_full - 1)
        i_lo = max(0, -d)
        i_hi = min(n_full - 1, n_full - 1 - d)

        gh[jdx] += (-h[idx])
        grad_q += w[k] * (gh[:len(x)] - gh[n_full-1: len(x)-1:-1])
    grad_q *= dx
    return val, grad_q

def select_active(corrs, margin_frac=0.04, margin_abs=1e-5, neighbor_pad=5, max_k=60):

    threshold = max_val - max(margin_abs, margin_frac * max_val)
    active = set(np.where(corrs >= threshold)[0].tolist())
    padded = set()
    for k in active:
        for dd in range(-neighbor_pad, neighbor_pad + 1):
            if 0 <= k + dd < len(corrs):
                padded.add(k + dd)
    active_list = sorted(padded)
    if len(active_list) > max_k:
        scored = sorted([(corrs[k], k) for k in active_list], reverse=True)
        active_list = sorted([k for _, k in scored[:max_k]])
    return active_list

    t0_val = corrs_c.max() + 1e-8

    x0 = np.zeros(n + 1)
    x0[:n] = h_center
    x0[n] = t0_val

    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    constraints = []
    for k in active_k:
        constraints.append({'type': 'ineq', 'fun': lambda x, kk=k: epig_full_con_fun(x, kk, n),
                           'jac': lambda x, kk=k: epig_full_con_jac(x, kk, n)})
    constraints.append({'type': 'eq', 'fun': lambda x: epig_full_eq(x, n),
                       'jac': lambda x: epig_full_eq_jac(x, n)})
    try:
        res = minimize(epig_full_obj, x0, jac=epig_full_obj_grad,
                      method='SLSQP', bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': ftol})
        h_out = np.clip(res.x[:n], 0.0, 1.0)
    except Exception:
        h_out = h_center.copy()
    h_out = project_feasible(h_out, n / 2.0)
    return h_out, eval_score(h_out)

def coord_descent_sym(q, n_full, n_passes=5, grid_size=100):
    m = len(q)
    dx = 2.0 / n_full
    q = q.copy()
    h = build_sym_h(q)
    best_sc = (corr_raw(h) * dx).max()

    for _pass in range(n_passes):
        improved = False
        for j in range(m):
            old = q[j]
            grid = np.linspace(0.0, 1.0, grid_size)
            best_local = best_sc
            best_qj = old
            for qj in grid:
                if abs(qj - old) < 1e-10:
                    continue
                q[j] = qj
                h[j] = qj
                h[n_full - 1 - j] = 1.0 - qj
                sc = (corr_raw(h) * dx).max()
                if sc < best_local - 1e-14:
                    best_local = sc
                    best_qj = qj
            if best_qj != old:
                lo = max(0.0, best_qj - 1.0/grid_size)
                hi = min(1.0, best_qj + 1.0/grid_size)
                gr = (np.sqrt(5.0) - 1.0) / 2.0
                for _ in range(25):
                    m1 = hi - gr * (hi - lo)
                    m2 = lo + gr * (hi - lo)
                    q[j] = m1; h[j] = m1; h[n_full-1-j] = 1.0-m1
                    s1 = (corr_raw(h) * dx).max()
                    q[j] = m2; h[j] = m2; h[n_full-1-j] = 1.0-m2
                    s2 = (corr_raw(h) * dx).max()
                    if s1 < s2:
                        hi = m2
                    else:
                        lo = m1
                best_qj = (lo + hi) / 2.0
            q[j] = best_qj
            h[j] = best_qj
            h[n_full - 1 - j] = 1.0 - best_qj
            sc = (corr_raw(h) * dx).max()
            if sc < best_sc - 1e-14:
                best_sc = sc
                improved = True
        if not improved:
            break
    return q, best_sc

def coord_descent_full(h, n_passes=3, grid_size=80):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best_sc = (corr_raw(h) * dx).max()
    target = n / 2.0

    for _pass in range(n_passes):
        improved = False
        order = np.random.permutation(n) if _pass > 0 else np.arange(n)
        for a in order[:n]:
            old_a = h[a]
            grid = np.linspace(0.0, 1.0, grid_size)
            best_local = best_sc
            best_val = old_a
            for va in grid:
                if abs(va - old_a) < 1e-10:
                    continue
                h[a] = va
                # need to adjust another element to keep sum
                # skip for now - just use as-is and project
                sc = (corr_raw(h) * dx).max()
                if sc < best_local - 1e-14:
                    best_local = sc
                    best_val = va
            if best_val != old_a:
                h[a] = best_val
                h = project_feasible(h, target)
                sc = eval_score(h)
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    improved = True
        if not improved:
            break
    return h, best_sc

def subgrad_sym_step(q, n_full, step_size, n_active=15):
    h = build_sym_h(q)
    dx = 2.0 / n_full
    corrs = corr_raw(h) * dx
    max_c = corrs.max()
    active_threshold = max_c - 0.02
    active = np.where(corrs >= active_threshold)[0]
    if len(active) == 0:
        active = np.where(corrs >= max_c - 1e-6)[0]
    if len(active) == 0:
        active = np.array([np.argmax(corrs)])
    if len(active) > n_active:
        order = np.argsort(corrs[active])[::-1]
        active = active[order[:n_active]]

    grad = np.zeros(len(q))
    for k in active:
        d = k - (n_full - 1)
        i_lo = max(0, -d)
        i_hi = min(n_full - 1, n_full - 1 - d)
        if i_lo > i_hi:
            continue
        idx = np.arange(i_lo, i_hi + 1)
        jdx = idx + d
        gh = np.zeros(n_full)
        gh[idx] += (1.0 - h[jdx])
        gh[jdx] += (-h[idx])
        grad += (gh[:len(q)] - gh[n_full-1: len(q)-1:-1])
    grad /= len(active)

    # Line search
    best_q = q.copy()
    best_sc = max_c
    for alpha in [step_size, step_size*0.5, step_size*0.25, step_size*0.1, step_size*0.05]:
        q_new = np.clip(q - alpha * grad, 0.0, 1.0)
        h_new = build_sym_h(q_new)
        sc = eval_score(h_new)
        if sc < best_sc:
            best_sc = sc
            best_q = q_new
    return best_q, best_sc

def optimize_sym_full(q0, n, time_deadline, do_full=False):
    m = n // 2
    q = np.clip(q0.copy(), 0.0, 1.0)
    best_q = q.copy()
    best_sc = eval_score(build_sym_h(q))

    # Phase A: Smooth max L-BFGS-B
    betas = [80, 200, 500, 1000]
    for beta in betas:
        if time.time() > time_deadline:
            break
        bounds = [(0.0, 1.0)] * m
        try:
            res = minimize(smooth_max_sym_f, q, jac=smooth_max_sym_j,
                          args=(n, beta), method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': 400, 'ftol': 1e-16, 'gtol': 1e-12})
            q_new = np.clip(res.x, 0.0, 1.0)
        except Exception:
            q_new = q.copy()
        sc = eval_score(build_sym_h(q_new))
        if sc < best_sc:
            best_sc = sc
            best_q = q_new.copy()
        q = q_new

    # Phase B: Epigraph refinement
    if time.time() < time_deadline - 10:
        h_chk = build_sym_h(q)
        corrs = corr_raw(h_chk) * (2.0 / n)
        active = select_active(corrs, margin_frac=0.04, neighbor_pad=5, max_k=55)
        q_new, h_new, sc_new = solve_epig_sym(q, n, active, maxiter=100, ftol=1e-13)
        if sc_new < best_sc:
            best_sc = sc_new
            best_q = q_new.copy()
            q = q_new

        # Second round with updated active set
        if time.time() < time_deadline - 10:
            h_chk = build_sym_h(q)
            corrs = corr_raw(h_chk) * (2.0 / n)
            active2 = select_active(corrs, margin_frac=0.02, neighbor_pad=3, max_k=45)
            q_new2, h_new2, sc_new2 = solve_epig_sym(q, n, active2, maxiter=80, ftol=1e-14)
            if sc_new2 < best_sc:
                best_sc = sc_new2
                best_q = q_new2.copy()
                q = q_new2

    # Phase C: Coordinate descent
    if time.time() < time_deadline - 5:
        q, sc = coord_descent_sym(q, n, n_passes=4, grid_size=120)
        if sc < best_sc:
            best_sc = sc
            best_q = q.copy()

    # Phase D: Subgradient refinement
    if time.time() < time_deadline - 5:
        q = best_q.copy()
        h = build_sym_h(q)
        current_sc = eval_score(h)
        step = 0.02
        for _ in range(200):
            if time.time() > time_deadline:
                break
            q, sc = subgrad_sym_step(q, n, step, n_active=20)
            if sc < current_sc - 1e-15:
                current_sc = sc
                if current_sc < best_sc:
                    best_sc = current_sc
                    best_q = q.copy()
            else:
                step *= 0.7
                if step < 1e-5:
                    break

    # Phase E: Final coordinate descent
    if time.time() < time_deadline - 3:
        q, sc = coord_descent_sym(best_q, n, n_passes=3, grid_size=100)
        if sc < best_sc:
            best_sc = sc
            best_q = q.copy()

    return best_q, best_sc

def optimize_full(h0, time_deadline):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_sc = eval_score(h)

    betas = [80, 250, 600]
    for beta in betas:
        if time.time() > time_deadline:
            break
        bounds = [(0.0, 1.0)] * n
        try:
            res = minimize(smooth_max_full_f, h, jac=smooth_max_full_j,
                          args=(n, beta), method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': 300, 'ftol': 1e-15, 'gtol': 1e-11})
            h_new = np.clip(res.x, 0.0, 1.0)
            h_new = project_feasible(h_new, n / 2.0)
        except Exception:
            h_new = h.copy()
        sc = eval_score(h_new)
        if sc < best_sc:
            best_sc = sc
            best_h = h_new.copy()
        h = h_new

    # Epigraph full
    if time.time() < time_deadline - 15:
        corrs = corr_raw(h) * (2.0 / n)
        active = select_active(corrs, margin_frac=0.04, neighbor_pad=4, max_k=50)
        h_new, sc_new = solve_epig_full(h, active, maxiter=70, ftol=1e-12)
        if sc_new < best_sc:
            best_sc = sc_new
            best_h = h_new.copy()
            h = h_new

    # Coordinate descent full
    if time.time() < time_deadline - 5:
        h, sc = coord_descent_full(h, n_passes=2, grid_size=60)
        if sc < best_sc:
            best_sc = sc
            best_h = h.copy()

    return best_h, best_sc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def rem():
        return budget_s - (time.time() - t0)

    best_h = None
    best_c5 = float('inf')

    # Get initial solution
    try:
        h_parent = np.array(initial_h_values, dtype=float).copy().reshape(-1)
    except (NameError, AttributeError):
        h_parent = None

    if h_parent is not None and len(h_parent) > 20:
        n_p = len(h_parent)
        h_parent = project_feasible(h_parent, n_p / 2.0)
        h_parent = np.clip(h_parent, 0.0, 1.0)
        best_h = h_parent.copy()
        best_c5 = eval_score(best_h)
    else:
        m = 63
        x = np.linspace(0.0, 1.0, m)
        q = np.clip(0.5 + 0.3 * np.sin(2.0 * np.pi * x), 0, 1)
        best_h = build_sym_h(q)
        best_c5 = eval_score(best_h)

    n_main = len(best_h)

    # Determine n candidates
    n_candidates = set()
    for d in [0, -2, -4, -6, 2, 4, 6, 8, -8, 10, -10]:
        nc = n_main + d
        if 80 <= nc <= 200 and nc % 2 == 0:
            n_candidates.add(nc)
    # Also try n_main if even
    if n_main % 2 == 0:
        n_candidates.add(n_main)
    n_candidates = sorted(n_candidates)

    # Sort by proximity to n_main (prefer staying close)
    n_sorted = sorted(n_candidates, key=lambda x: abs(x - n_main))

    # Time allocation: give more time to n_main
    time_per_n = max(60, min(200, (budget_s * 0.7) / len(n_sorted)))

    for n_try in n_sorted:
        if rem() < 50:
            break

        # Get starting point for this n
        if n_try == n_main and best_h is not None:
            q0 = sym_from_h(best_h)
        else:
            h_tr = transfer_to_n(best_h, n_try)
            q0 = sym_from_h(h_tr)

        # Optimize
        time_lim = time.time() + min(time_per_n, rem() - 20)
        q_opt, c5_opt = optimize_sym_full(q0, n_try, time_lim)
        h_opt = build_sym_h(q_opt)
        c5_opt = eval_score(h_opt)

        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Try full-vector optimization on best
    if rem() > 80:
        n_f = len(best_h)
        h_full_start = best_h.copy()
        time_lim = time.time() + min(120, rem() - 30)
        h_opt_full, c5_full = optimize_full(h_full_start, time_lim)
        if c5_full < best_c5:
            best_c5 = c5_full
            best_h = h_opt_full

    # Try a different starting point: step function
    if rem() > 80:
        n_sf = n_main if n_main % 2 == 0 else n_main + 1
        m_sf = n_sf // 2
        # Step: h = 1 on [0,1), 0 on [1,2)
        q_sf = np.zeros(m_sf)
        # First half is [0,1), so q should be 1 where h is 1
        # h[0:m] = q, h[m:2m] = 1-q[::-1]
        # For step at x=1: h[i]=1 for (i+0.5)*2/n < 1, i.e., i < n/2 = m
        # So q = ones(m), and 1-q[::-1] = zeros(m). That gives h = [1,...,1,0,...,0]
        # But sum = m, need sum = n/2 = m. OK!
        q_sf = np.ones(m_sf) * 0.5  # start with uniform, will optimize
        # Actually a better start: h = 1 on [0, 0.75), 0 on [0.75, 1] in [0,1] coords
        # In [0,2]: h=1 on [0,1.5), 0 on [1.5,2]
        # sum = 1.5/2 * 2... wait sum should be 1 (integral = 1)
        # sum(h)*dx = 1, so sum(h) = n/2
        # Step at position p: h=1 for x<p, h=0 for x>=p
        # sum = p/2 * n / 2... let me just use a smooth start
        x_sf = np.linspace(0, 1, m_sf)
        q_sf = np.clip(1.0 - 2.0 * x_sf, 0, 1)  # linear ramp from 1 to 0
        time_lim = time.time() + min(100, rem() - 20)
        q_opt, c5_opt = optimize_sym_full(q_sf, n_sf, time_lim)
        h_opt = build_sym_h(q_opt)
        c5_opt = eval_score(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Final polish: coordinate descent on best
    if rem() > 15 and best_h is not None:
        n_fin = len(best_h)
        if n_fin % 2 == 0:
            q_fin = sym_from_h(best_h)
            q_fin, sc_fin = coord_descent_sym(q_fin, n_fin, n_passes=5, grid_size=150)
            h_fin = build_sym_h(q_fin)
            sc_fin = eval_score(h_fin)
            if sc_fin < best_c5:
                best_c5 = sc_fin
                best_h = h_fin
        else:
            h_fin, sc_fin = coord_descent_full(best_h, n_passes=3, grid_size=80)
            if sc_fin < best_c5:
                best_c5 = sc_fin
                best_h = h_fin

    # Final epigraph pass
    if rem() > 10 and best_h is not None:
        n_fin = len(best_h)
        if n_fin % 2 == 0:
            q_fin = sym_from_h(best_h)
            h_chk = build_sym_h(q_fin)
            corrs = corr_raw(h_chk) * (2.0 / n_fin)
            active = select_active(corrs, margin_frac=0.01, neighbor_pad=3, max_k=40)
            q_o, h_o, c5_o = solve_epig_sym(q_fin, n_fin, active, maxiter=120, ftol=1e-14)
            h_o = build_sym_h(np.clip(q_o, 0, 1))
            c5_o = eval_score(h_o)
            if c5_o < best_c5:
                best_c5 = c5_o
                best_h = h_o

    # Validate and return
    best_h = np.asarray(best_h, dtype=float).reshape(-1)
    n_out = len(best_h)
    best_h = project_feasible(best_h, n_out / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = eval_score(best_h)

    return best_h, best_c5, n_out
