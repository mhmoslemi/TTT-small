    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx + offset] += (1.0 - h[idx])
    grad[idx] -= h[idx + offset]
    return grad


def solve_lp_push(h, dx, active_idx, alpha):
    n = len(h)
    corrs = np.correlate(h, 1.0 - h, mode='full')
    m = len(active_idx)
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_ub = np.zeros((m, n + 1))
    b_ub = np.zeros(m)
    for j, k in enumerate(active_idx):
        g = grad_corr_k(h, k, n)
        A_ub[j, :n] = dx * g
        A_ub[j, n] = -1.0
        b_ub[j] = -dx * corrs[k]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for i in range(n):
        bounds.append((-alpha * h[i], alpha * (1.0 - h[i])))
    bounds.append((0.0, None))
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status != 0:
            return None
        return h + res.x[:n]
    except Exception:
        return None


def lp_polish(h0, dx, target_sum, max_iters, time_limit, t0, alpha_init, act_frac):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    alpha = alpha_init
    no_improve = 0
    n_corrs = 2 * n - 1
    n_active = max(15, int(act_frac * n_corrs))

    for it in range(max_iters):
        if time.time() - t0 > time_limit:
            break
        corrs = np.correlate(h, 1.0 - h, mode='full')
        cur_c5 = corrs.max() * dx
        if cur_c5 < best_c5 - 1e-14:
            best_c5 = cur_c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 150:
            break
        top_idx = np.argsort(corrs)[-n_active:]
        h_cand = solve_lp_push(h, dx, top_idx, alpha)
        if h_cand is None:
            alpha *= 0.6
            if alpha < 1e-7:
                break
            continue
        h_cand = make_feasible(h_cand, target_sum)
        new_c5 = evaluate_c5(h_cand)
        if new_c5 < cur_c5 - 1e-14:
            h = h_cand
            alpha = min(0.25, alpha * 1.03)
        else:
            improved = False
            for frac in [0.5, 0.25, 0.1]:
                h_try = h + frac * (h_cand - h)
                h_try = make_feasible(h_try, target_sum)
                tc5 = evaluate_c5(h_try)
                if tc5 < cur_c5 - 1e-15:
                    h = h_try
                    improved = True
                    break
            if not improved:
                alpha *= 0.6
                if alpha < 1e-7:
                    break
    return best_h, best_c5


def subgrad_polish(h0, dx, target_sum, max_iter, time_limit, t0):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    avg_h = np.zeros(n)
    no_improve = 0

    for it in range(max_iter):
        if time.time() - t0 > time_limit:
            break
        corrs = np.correlate(h, 1.0 - h, mode='full')
        cur_c5 = corrs.max() * dx
        if cur_c5 < best_c5 - 1e-15:
            best_c5 = cur_c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 3000:
            break
        sorted_idx = np.argsort(corrs)[::-1]
        g = np.zeros(n)
        n_near = 0
        for k in sorted_idx:
            if corrs[k] >= corrs[sorted_idx[0]] - 0.02:
                g += grad_corr_k(h, k, n) * dx
                n_near += 1
                if n_near >= 8:
                    break
        if n_near > 0:
            g /= n_near
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-14:
            break
        step = 0.5 / (np.sqrt(it + 50.0))
        h = make_feasible(h - step * g, target_sum)
        avg_h = avg_h * (it / (it + 1.0)) + h / (it + 1.0)

    avg_h = make_feasible(avg_h, target_sum)
    avg_c5 = evaluate_c5(avg_h)
    if avg_c5 < best_c5:
        best_h = avg_h
        best_c5 = avg_c5
    return best_h, best_c5


def transfer_grid(h_src, n_dst):
    n_src = len(h_src)
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    h_interp = np.interp(x_dst, x_src, h_src)
    target = n_dst / 2.0
    return make_feasible(h_interp, target)


def lp_polish_multi(h0, dx, target_sum, time_limit, t0):
    n = len(h0)
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    alphas = [0.15, 0.10, 0.06, 0.03, 0.015]
    fracs = [0.3, 0.4, 0.5, 0.6, 0.7]
    max_iters_list = [200, 200, 250, 300, 400]
    for i in range(len(alphas)):
        if time.time() - t0 > time_limit - 5:
            break
        h_out, c5_out = lp_polish(best_h, dx, target_sum, max_iters_list[i],
                                  time_limit, t0, alphas[i], fracs[i])
        if c5_out < best_c5:
            best_c5 = c5_out
            best_h = h_out
    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')

    def rem():
        return budget_s - (time.time() - t0)

    # Initialize from parent
    h0 = np.array(initial_h_values, dtype=float)
    n = len(h0)
    dx = 2.0 / n
    target_sum = n / 2.0
    h0 = make_feasible(h0, target_sum)
    c5_0 = evaluate_c5(h0)
    best_h = h0.copy()
    best_c5 = c5_0

    # Phase 1: Active-set LP polishing on parent grid
    if rem() > 30:
        h_out, c5_out = lp_polish_multi(best_h, dx, target_sum, rem() - 20, t0)
        if c5_out < best_c5:
            best_c5 = c5_out
            best_h = h_out

    # Phase 2: Subgradient to explore
    if rem() > 40:
        h_sg, c5_sg = subgrad_polish(best_h, dx, target_sum, 5000, rem() - 20, t0)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg
            # Polish the improved point
            if rem() > 30:
                h_out, c5_out = lp_polish_multi(best_h, dx, target_sum, rem() - 15, t0)
                if c5_out < best_c5:
                    best_c5 = c5_out
                    best_h = h_out

    # Phase 3: More LP polishing with different parameters
    if rem() > 30:
        h_out, c5_out = lp_polish(best_h, dx, target_sum, 500, rem() - 10, t0, 0.05, 0.5)
        if c5_out < best_c5:
            best_c5 = c5_out
            best_h = h_out

    # Phase 4: Try different grid sizes
    n_candidates = [64, 80, 96, 100, 110, 128, 132, 140]
    for n_try in n_candidates:
        if rem() < 40:
            break
        if n_try == n:
            continue
        dx_try = 2.0 / n_try
        ts_try = n_try / 2.0
        h_trans = transfer_grid(best_h, n_try)
        c5_trans = evaluate_c5(h_trans)
        h_out, c5_out = lp_polish_multi(h_trans, dx_try, ts_try, rem() - 20, t0)
        if c5_out < best_c5:
            best_c5 = c5_out
            best_h = h_out

    # Phase 5: Final intensive polish on best grid
    if rem() > 20 and best_h is not None:
        n_b = len(best_h)
        dx_b = 2.0 / n_b
        ts_b = n_b / 2.0
        h_out, c5_out = lp_polish_multi(best_h, dx_b, ts_b, rem() - 10, t0)
        if c5_out < best_c5:
            best_c5 = c5_out
            best_h = h_out

    # Phase 6: Final subgradient + polish
    if rem() > 30:
        n_b = len(best_h)
        dx_b = 2.0 / n_b
        ts_b = n_b / 2.0
        h_sg2, c5_sg2 = subgrad_polish(best_h, dx_b, ts_b, 3000, rem() - 20, t0)
        if c5_sg2 < best_c5:
            best_c5 = c5_sg2
            best_h = h_sg2
            if rem() > 15:
                h_out, c5_out = lp_polish(best_h, dx_b, ts_b, 300, rem() - 5, t0, 0.04, 0.5)
                if c5_out < best_c5:
                    best_c5 = c5_out
                    best_h = h_out

    # Final validation
    n_final = len(best_h)
    dx_final = 2.0 / n_final
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = make_feasible(best_h, n_final / 2.0)
    best_c5 = evaluate_c5(best_h)

    return best_h, best_c5, n_final
