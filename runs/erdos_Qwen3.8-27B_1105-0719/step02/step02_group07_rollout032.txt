def corr_grad_one(h, d, dx):
    """Gradient of g_k at lag d"""
    n = len(h)
    grad = np.zeros(n)
    i_lo = max(0, -d)
    i_hi = min(n-1, n-1-d)

        idx = np.arange(i_lo, i_hi+1)
        grad[idx] += (1.0 - h[idx+d])
        grad[idx+d] += (-h[idx])
    return grad * dx

def sym_smooth_grad(q, n, beta):

    h = make_sym_h(q, n)

    shifted = corrs - corrs.max()

def fw_step(h):
    """Frank-Wolfe linearized step: find vertex of feasible polytope in direction of negative subgradient"""
    n = len(h)
    corrs = corr_vals(h)
    max_c = corrs.max()
    # Active set: lags within tolerance of max
    tol = 0.01
    active = np.where(corrs >= max_c - tol)[0]
    if len(active) == 0:
        active = [np.argmax(corrs)]
    # Subgradient
    dx = 2.0 / n
    g_sub = np.zeros(n)
    for k in active:
        d = k - (n-1)
        g_sub += corr_grad_one(h, d, dx)
    g_sub /= len(active)
    # LP: min g_sub · x s.t. 0 ≤ x ≤ 1, sum(x) = n/2
    # Sort and fill
    sorted_idx = np.argsort(g_sub)
    x_new = np.zeros(n)
    x_new[sorted_idx[:n//2]] = 1.0
    # Verify sum constraint (should be exact since n/2 is integer)
    return x_new

But I can optimize by not calling correlate every time. Since changing q[j] only affects h[j] and h[m+j] (wait, h[m:] = 1 - q[::-1], so h[m + (m-1-j)] = 1 - q[j], i.e., h[2m-1-j] = 1-q[j] for j in [0,m)).

So changing q[j] affects h[j] and h[2m-1-j] = h[n-1-j].

The change in correlation at lag d is:
ΔC_d = dx * [(new_hj - old_hj) * (1 - h[j+d]) * I(valid) + (new_h2 - old_h2) * (1 - h[n-1-j+d]) * I(valid)
            + h[j-d] * (old_hj - new_hj) * I(valid) + h[n-1-j-d] * (old_h2 - new_h2) * I(valid)]

where new_hj = trial, old_hj = q[j], new_h2 = 1-trial, old_h2 = 1-q[j].

This is O(1) per lag, so O(n) total per trial. Much faster!

Let me implement this optimized version:

def sym_coord_fast(q, n, n_grid=50, max_sweeps=200):

    h = make_sym_h(q, n)

    best_c5_raw = corrs.max()
    best_q = q.copy()
    
    grid = np.linspace(0, 1, n_grid)
    
    for sweep in range(max_sweeps):
        improved = False
        for j in range(m):
            old_qj = q[j]
            # h[j] and h[n-1-j] are the affected positions
            i1 = j
            i2 = n - 1 - j
            
            best_val = best_c5_raw
            best_new = old_qj
            
            for trial in grid:
                if abs(trial - old_qj) < 1e-10:
                    continue
                # New values
                new_h_i1 = trial
                new_h_i2 = 1.0 - trial
                old_h_i1 = h[i1]
                old_h_i2 = h[i2]
                
                # Update correlations
                # For each lag k (corresponding to shift d = k - (n-1)):
                # C[k] involves h[a]*h[b] where b-a = d (or a-b = d)
                # Specifically: C[k] = sum_i h[i] * h[i+d] for valid i (this is autocorrelation part)
                # But we need cross-correlation with (1-h)
                # corr[k] = sum_i h[i] * (1-h[i+d]) for valid i
                #         = sum_i h[i] - sum_i h[i]*h[i+d]
                
                # Change in corr[k] when h[i1] changes from old_h_i1 to new_h_i1:
                # The term h[i1]*(1-h[i1+d]) changes by (new-old)*(1-h[i1+d])
                # The term h[i1-d]*(1-h[i1]) changes by -h[i1-d]*(new-old)
                # Similarly for i2.
                
                # This is getting complex. Let me just recompute correlate.
                h[i1] = new_h_i1
                h[i2] = new_h_i2
                new_corrs = np.correlate(h, 1-h, mode='full')
                val = new_corrs.max()
                if val < best_val:
                    best_val = val
                    best_new = trial
                    corrs = new_corrs
                else:
                    h[i1] = old_h_i1
                    h[i2] = old_h_i2
            
            q[j] = best_new
            h[i1] = best_new
            h[i2] = 1.0 - best_new
            if best_new != old_qj:
                improved = True
                best_c5_raw = corrs.max()
        
        if best_c5_raw < best_c5_raw:  # always true, just for logic
            pass
        if not improved:
            break
    
    h_best = make_sym_h(best_q, n)
    return best_q, exact_score(h_best)

import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time

def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(100):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full').max() * dx

def make_sym_h(q, n):
    m = n // 2
    q = np.clip(q, 0.0, 1.0)
    h = np.empty(n)
    h[:m] = q
    h[m:] = 1.0 - q[::-1]
    return h

def make_q_from_h(h, n):
    m = n // 2
    q = 0.5 * (h[:m] + 1.0 - h[n-1:n-1-m:-1])
    return np.clip(q, 0.0, 1.0)

def corr_vals(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def corr_grad_one(h, d, dx):
    n = len(h)
    grad = np.zeros(n)
    i_lo = max(0, -d)
    i_hi = min(n-1, n-1-d)
    if i_lo <= i_hi:
        idx = np.arange(i_lo, i_hi+1)
        grad[idx] += (1.0 - h[idx+d])
        grad[idx+d] += (-h[idx])
    return grad * dx

def sym_smooth_obj(q, n, beta):
    h = make_sym_h(q, n)
    corrs = corr_vals(h)
    return logsumexp(beta * corrs) / beta

def sym_smooth_grad(q, n, beta):
    m = n // 2
    h = make_sym_h(q, n)
    corrs = corr_vals(h)
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    ws = w.sum()
    if ws < 1e-30:
        w = np.ones(len(corrs)) / len(corrs)
    else:
        w /= ws
    dx = 2.0 / n
    g = np.zeros(m)
    for k in range(len(corrs)):
        wk = w[k]
        if wk < 1e-15:
            continue
        d = k - (n-1)
        gn = corr_grad_one(h, d, dx)
        g += wk * (gn[:m] - gn[n-1:n-1-m:-1])
    return beta * g

def lbfgsb_smooth(q0, n, beta, maxiter=5000):
    m = n // 2
    try:
        res = minimize(sym_smooth_obj, q0, jac=lambda x: sym_smooth_grad(x, n, beta),
                       method='L-BFGS-B', bounds=[(0.0, 1.0)]*m,
                       options={'maxiter': maxiter, 'gtol': 1e-10, 'maxfun': 10000})
        if res.fun < 1e10:
            h_new = make_sym_h(res.x, n)
            c5 = exact_score(h_new)
            return res.x, c5
    except Exception:
        pass
    return None, None

def coord_descent_sym(q, n, n_grid=40, max_sweeps=500, tol=1e-14):
    """Coordinate descent in symmetric q-space. Sum constraint auto-satisfied."""
    m = n // 2
    h = make_sym_h(q, n)
    corrs_raw = np.correlate(h, 1.0 - h, mode='float64')
    best_raw = corrs_raw.max()
    best_q = q.copy()
    grid = np.linspace(0.0, 1.0, n_grid)
    
    prev_best = best_raw + 1
    stall_count = 0
    
    for sweep in range(max_sweeps):
        improved = False
        for j in range(m):
            old_qj = q[j]
            i1 = j
            i2 = n - 1 - j
            best_val = best_raw
            best_new = old_qj
            h[i1] = old_qj
            h[i2] = 1.0 - old_qj
            
            for trial in grid:
                if abs(trial - old_qj) < 1e-12:
                    continue
                h[i1] = trial
                h[i2] = 1.0 - trial
                nc = np.correlate(h, 1.0 - h, mode='float64')
                val = nc.max()
                if val < best_val - 1e-15:
                    best_val = val
                    best_new = trial
                    corrs_raw = nc
            q[j] = best_new
            h[i1] = best_new
            h[i2] = 1.0 - best_new
            if best_new != old_qj:
                improved = True
                best_raw = corrs_raw.max()
        
        if not improved:
            break
        if best_raw < prev_best - tol:
            prev_best = best_raw
            best_q = q.copy()
            stall_count = 0
        else:
            stall_count += 1
            if stall_count > 3:
                break
    
    h_best = make_sym_h(best_q, n)
    return best_q, exact_score(h_best)

def full_coord_descent(h, n, n_grid=30, max_sweeps=100):
    """Coordinate descent in full h-space with projection."""
    target = n / 2.0
    best_c5 = exact_score(h)
    best_h = h.copy()
    grid = np.linspace(0.0, 1.0, n_grid)
    
    for sweep in range(max_sweeps):
        improved = False
        for j in range(n):
            old = h[j]
            best_val = best_c5
            best_new = old
            for trial in grid:
                if abs(trial - old) < 1e-12:
                    continue
                h[j] = trial
                h_p = project_feasible(h, target)
                val = exact_score(h_p)
                if val < best_val - 1e-15:
                    best_val = val
                    best_new = trial
                    h = h_p
            h[j] = best_new
            h = project_feasible(h, target)
            if best_new != old:
                improved = True
                best_c5 = exact_score(h)
                best_h = h.copy()
        if not improved:
            break
    
    return best_h, exact_score(best_h)

def fw_linearized_step(h, n):
    """Frank-Wolfe style: find vertex in direction of negative subgradient."""
    corrs = corr_vals(h)
    max_c = corrs.max()
    tol = 0.02
    active = np.where(corrs >= max_c - tol)[0]
    if len(active) == 0:
        active = np.array([np.argmax(corrs)])
    dx = 2.0 / n
    g_sub = np.zeros(n)
    for k in active:
        d = k - (n-1)
        g_sub += corr_grad_one(h, d, dx)
    g_sub /= len(active)
    sorted_idx = np.argsort(g_sub)
    x_new = np.zeros(n)
    x_new[sorted_idx[:n//2]] = 1.0
    return x_new
