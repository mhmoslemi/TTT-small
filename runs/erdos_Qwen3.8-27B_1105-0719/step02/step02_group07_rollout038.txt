One more optimization: I can vectorize the gradient computation using the fact that for a fixed h, the correlations are:
corr[k] = sum_i h[i] * (1-h[i+d]) * dx where d = k - (n-1)

The gradient of corr[k] w.r.t. h[j] is:
- (1-h[j+d]) * dx if j is in valid range (0 <= j < n, 0 <= j+d < n)
- -h[j-d] * dx if j-d is in valid range (0 <= j-d < n, 0 <= j < n)
- Sum of both if both conditions hold

This can be computed as:
G[j] = dx * (sum_k w[k] * (1-h[j+d_k]) * I(j+d_k valid) - sum_k w[k] * h[j-d_k] * I(j-d_k valid))

where d_k = k - (n-1).

For a fixed j, d_k ranges over -(n-1) to n-1 as k ranges from 0 to 2n-2.
- j + d_k is valid when 0 <= j + d_k < n, i.e., -j <= d_k < n-j
- j - d_k is valid when 0 <= j - d_k < n, i.e., -(n-1-j) < d_k <= j, i.e., d_k > j-n and d_k <= j

So:
G[j] = dx * (sum_{k: -j <= d_k < n-j, w[k]>0} w[k]*(1-h[j+d_k]) - sum_{k: j-n < d_k <= j, w[k]>0} w[k]*h[j-d_k])

This is a convolution-like operation. I could potentially use FFT for speed, but for n=126 it's not worth the complexity. The loop version is fine.

import numpy as np
from scipy.optimize import minimize

def make_sym_h(q, n):
    m = n // 2
    h = np.empty(n)
    h[:m] = q
    h[m:] = 1.0 - q[::-1]
    return h


def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)


def corr_vals(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx


def sym_grad_from_full(G, n):
    m = n // 2
    return G[:m] - G[n-1:n-1-m:-1]


def _fun_smooth_sym(x, n, beta):
    h = make_sym_h(x, n)
    corrs = corr_vals(h)
    dx = 2.0 / n
    mc = corrs.max()
    shifted = beta * (corrs - mc)
    exps = np.exp(shifted)
    s = exps.sum()
    if s < 1e-300:
        f = mc
        w = np.zeros(2 * n - 1)
        w[np.argmax(corrs)] = 1.0
    else:
        f = mc + np.log(s) / beta
        w = exps / s
    G = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        wk = w[k]
        d = k - (n - 1)
        i_lo = max(0, -d)
        i_hi = min(n - 1, n - 1 - d)
        if i_lo <= i_hi:
            idx = np.arange(i_lo, i_hi + 1)
            G[idx] += wk * (1.0 - h[idx + d])
            G[idx + d] += wk * (-h[idx])
    G *= dx
    return f, sym_grad_from_full(G, n)

        q_new, c5 = _opt_lbfgs(q_start, n, 80, maxiter=100)
        if c5 < c5_best - 1e-9:
            q_new2, c5_2 = _opt_lbfgs(q_new, n, 500, maxiter=250)
            if c5_2 < c5_best - 1e-11:
                q_p, c5_p = _subgrad_polish(q_new2, n, max_iters=200)
                if c5_p < c5_2:
                    c5_2 = c5_p
                    q_new2 = q_p
                c5_best = c5_2
                q_best = q_new2
                h_best = make_sym_h(q_best, n)
                print(f"  S{trial}: {c5_best:.7f}")

    # Phase 7: Try different n values
    if remaining() > 30:
        for delta in [2, -2, 4, -4]:
            n_try = n + delta
            if n_try < 60 or n_try % 2 != 0 or remaining() < 18:
                continue
            x_old = np.linspace(0, 2, n, endpoint=False)
            x_new_pts = np.linspace(0, 2, n_try, endpoint=False)
            h_interp = np.interp(x_new_pts, x_old, h_best)
            m_try = n_try // 2
            q_try = np.clip(0.5 * (h_interp[:m_try] + 1.0 - h_interp[n_try - 1:n_try - 1 - m_try:-1]), 0.0, 1.0)
            q_new, c5 = _opt_lbfgs(q_try, n_try, 100, maxiter=200)
            if c5 < c5_best - 1e-9:
                q_new2, c5_2 = _opt_lbfgs(q_new, n_try, 600, maxiter=300)
                if c5_2 < c5_best - 1e-11:
                    c5_best = c5_2
                    n = n_try
                    m = m_try
                    q_best = q_new2
                    h_best = make_sym_h(q_best, n)
                    print(f"  n={n}: {c5_best:.7f}")

    # Final polish
    if remaining() > 10:
        q_p, c5_p = _subgrad_polish(q_best, n, max_iters=300, top_k=3, alpha0=0.01)
        if c5_p < c5_best - 1e-11:
            c5_best = c5_p
            q_best = q_p
            h_best = make_sym_h(q_best, n)
            print(f"  Final subgrad: {c5_best:.7f}")

    h_best = np.clip(h_best, 0.0, 1.0)
    c5_best = exact_score(h_best)
    print(f"Done: C5={c5_best:.7f}, n={len(h_best)}")
    return h_best, c5_best, len(h_best)
