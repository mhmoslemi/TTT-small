import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time

def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-13:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def c5_of_h(h, dx):
    return float(compute_corrs(h).max() * dx)

def grad_corr_k_vec(h, k, n):
    m = k - (n - 1)
    grad = np.zeros(n)
    if m == 0:
        grad[:] = 1.0 - 2.0 * h
        return grad
    if m > 0:
        j1 = np.arange(m, n)
        grad[j1] += (1.0 - h[j1 - m])
        j2 = np.arange(0, n - m)
        grad[j2] += (-h[j2 + m])
    else:
        d = -m
        j1 = np.arange(0, n - d)
        grad[j1] += (1.0 - h[j1 + d])
        j2 = np.arange(d, n)
        grad[j2] += (-h[j2 - d])
    return grad

def smooth_max_val(h, dx, beta):
    corrs = compute_corrs(h)
    return float(logsumexp(beta * corrs) / beta * dx)

def smooth_max_grad(h, dx, beta):
    n = len(h)
    corrs = compute_corrs(h)
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(len(corrs)) / len(corrs)
    else:
        w /= w_sum
    grad = np.zeros(n)
    for k in range(len(corrs)):
        if w[k] > 1e-10:
            grad += w[k] * grad_corr_k_vec(h, k, n)
    return grad * dx

def slsqp_opt(h0, dx, target_sum, beta, maxiter):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        return smooth_max_val(x, dx, beta)
    def g(x):
        return smooth_max_grad(x, dx, beta)
    def con_fun(x):
        return x.sum() - target_sum
    def con_jac(x):
        return np.ones(n)
    try:
        res = minimize(obj, h0, jac=g, method='SLSQP',
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': con_fun, 'jac': con_jac}],
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        h_out = project_feasible(res.x, target_sum)
    except Exception:
        h_out = project_feasible(h0, target_sum)
    return h_out, c5_of_h(h_out, dx)

def coordinate_push_top(h, dx, target_sum, top_m=3, blend=1.0):
    """For each top shift, compute exact per-coordinate optimum, blend, project."""
    n = len(h)
    corrs = compute_corrs(h)
    top_idx = np.argsort(corrs)[-top_m:]
    h_pushed = np.zeros(n)
    for k in top_idx:
        m = k - (n - 1)
        if m == 0:
            h_k = h.copy()
        else:
            c = np.zeros(n)
            if m > 0:
                j1 = np.arange(m, n)
                c[j1] += (1.0 - h[j1 - m])
                j2 = np.arange(0, n - m)
                c[j2] += (-h[j2 + m])
            else:
                d = -m
                j1 = np.arange(0, n - d)
                c[j1] += (1.0 - h[j1 + d])
                j2 = np.arange(d, n)
                c[j2] += (-h[j2 - d])
            h_k = np.where(c < c.mean(), 1.0, 0.0)
            diff = target_sum - h_k.sum()
            if diff > 0:
                zeros = np.where(h_k < 0.5)[0]
                if len(zeros) > 0:
                    order = zeros[np.argsort(c[zeros])]
                    add = min(diff, 1.0)
                    h_k[order[0]] = min(1.0, h_k[order[0]] + add)
                    rem = diff - add
                    for j in order[1:]:
                        if rem <= 0:
                            break
                        a = min(rem, 1.0 - h_k[j])
                        h_k[j] += a
                        rem -= a
            elif diff < 0:
                ones = np.where(h_k > 0.5)[0]
                if len(ones) > 0:
                    order = ones[np.argsort(-c[ones])]
                    sub = min(-diff, 1.0)
                    h_k[order[0]] = max(0.0, h_k[order[0]] - sub)
                    rem = -diff - sub
                    for j in order[1:]:
                        if rem <= 0:
                            break
                        a = min(rem, h_k[j])
                        h_k[j] -= a
                        rem -= a
        h_pushed += h_k
    h_pushed /= top_m
    h_new = blend * h_pushed + (1.0 - blend) * h
    h_new = project_feasible(h_new, target_sum)
    return h_new

def subgrad_adaptive(h0, dx, target_sum, max_iter=40000, top_m=3):
    n = len(h0)
    h = h0.copy()
    best_h = h0.copy()
    best_c5 = c5_of_h(h0, dx)
    v = np.zeros(n)
    gamma = 0.9
    alpha = 0.1
    no_improve = 0
    pause = 0

    for it in range(max_iter):
        corrs = compute_corrs(h)
        C5 = corrs.max() * dx
        if C5 < best_c5 - 1e-14:
            best_c5 = C5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 15000:
            break

        top_idx = np.argsort(corrs)[-top_m:]
        grad = np.zeros(n)
        for k in top_idx:
            grad += grad_corr_k_vec(h, k, n)
        grad /= top_m
        grad *= dx

        v = gamma * v - alpha * grad
        h_new = project_feasible(h + v, target_sum)
        c5_new = c5_of_h(h_new, dx)

        if c5_new < C5 - 1e-15:
            h = h_new
            alpha = min(alpha * 1.005, 0.5)
            pause = 0
        else:
            alpha *= 0.93
            v *= 0.4
            pause += 1
            if pause > 500:
                h = best_h.copy()
                v = np.zeros(n)
                alpha = 0.05
                pause = 0

        if alpha < 1e-12:
            break

    return best_h, best_c5

def simulated_anneal(h0, dx, target_sum, max_iter=30000, rng=None):
    n = len(h0)
    if rng is None:
        rng = np.random.RandomState(0)
    h = h0.copy()
    best_h = h0.copy()
    best_c5 = c5_of_h(h0, dx)
    T_init = 0.005
    T_final = 0.0001
    n_coord = max(1, n // 30)

    for it in range(max_iter):
        frac = it / max_iter
        T = T_init * (T_final / T_init) ** frac
        corrs = compute_corrs(h)
        C5 = corrs.max() * dx

        if C5 < best_c5 - 1e-14:
            best_c5 = C5
            best_h = h.copy()

        idx = rng.choice(n, n_coord, replace=False)
        delta = rng.uniform(-0.3, 0.3, n_coord)
        h_new = h.copy()
        h_new[idx] = np.clip(h_new[idx] + delta, 0.0, 1.0)
        h_new = project_feasible(h_new, target_sum)
        C5_new = c5_of_h(h_new, dx)
        dC = C5_new - C5
        if dC < 0 or rng.rand() < np.exp(-dC / T):
            h = h_new

    return best_h, best_c5

def make_start(n, target_sum, rng, style):
    h = np.zeros(n)
    if style == 0:
        h = rng.uniform(0, 1, n)
    elif style == 1:
        bs = rng.randint(3, 12)
        for i in range(n):
            h[i] = 1.0 if (i // bs) % 2 == 0 else 0.0
    elif style == 2:
        freq = rng.uniform(3, 25)
        phase = rng.uniform(0, 6.28)
        h = 0.5 + 0.45 * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
    elif style == 3:
        h = (rng.uniform(0, 1, n) > 0.5).astype(float)
        nf = max(1, n // 6)
        idx = rng.choice(n, nf, replace=False)
        h[idx] = rng.uniform(0.2, 0.8, nf)
    elif style == 4:
        n1 = int(target_sum)
        idx = rng.choice(n, n1, replace=False)
        h[idx] = 1.0
    else:
        x = np.linspace(0, 1, n)
        h = (x < target_sum / n).astype(float)
        h += rng.normal(0, 0.15, n)
    return project_feasible(h, target_sum)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def remaining():
        return budget_s - (time.time() - t0)

    best_h = None
    best_c5 = float('inf')

    h0 = np.array(initial_h_values, dtype=float)
    n = len(h0)
    dx = 2.0 / n
    target_sum = n / 2.0
    h0 = project_feasible(h0, target_sum)
    best_h = h0.copy()
    best_c5 = c5_of_h(h0, dx)

    # Phase 1: SLSQP with increasing beta
    for beta in [50, 200, 1000, 5000]:
        if remaining() < 25:
            break
        h_opt, c5_val = slsqp_opt(best_h, dx, target_sum, beta, 1200)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_opt

    # Phase 2: Coordinate push cycling (new)
    if remaining() > 80:
        for _ in range(8):
            if remaining() < 15:
                break
            h_push = coordinate_push_top(best_h, dx, target_sum, top_m=4, blend=1.0)
            c5_push = c5_of_h(h_push, dx)
            if c5_push < best_c5:
                best_c5 = c5_push
                best_h = h_push
            for blend in [0.7, 0.5, 0.3]:
                h_blend = coordinate_push_top(best_h, dx, target_sum, top_m=3, blend=blend)
                c5_blend = c5_of_h(h_blend, dx)
                if c5_blend < best_c5:
                    best_c5 = c5_blend
                    best_h = h_blend

    # Phase 3: Subgradient with momentum
    if remaining() > 80:
        h_sg, c5_sg = subgrad_adaptive(best_h, dx, target_sum, max_iter=40000, top_m=3)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg

    # Phase 4: SLSQP high-beta polish
    if remaining() > 40:
        for beta in [5000, 20000]:
            if remaining() < 20:
                break
            h_opt, c5_val = slsqp_opt(best_h, dx, target_sum, beta, 1500)
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h_opt

    # Phase 5: Coordinate push + subgrad alternation
    if remaining() > 60:
        for rnd in range(6):
            if remaining() < 20:
                break
            h_push = coordinate_push_top(best_h, dx, target_sum, top_m=5, blend=0.8)
            c5_p = c5_of_h(h_push, dx)
            if c5_p < best_c5:
                best_c5 = c5_p
                best_h = h_push
            h_sg, c5_sg = subgrad_adaptive(best_h, dx, target_sum, max_iter=15000, top_m=3)
            if c5_sg < best_c5:
                best_c5 = c5_sg
                best_h = h_sg

    # Phase 6: n sweep
    n_best = len(best_h)
    n_cands = sorted(set(
        list(range(max(40, n_best - 15), n_best + 35, 2)) +
        [40, 50, 60, 70, 80, 90, 100, 110, 130, 140, 150, 160, 180, 200]
    ))
    for n_try in n_cands:
        if remaining() < 35:
            break
        if n_try < 20:
            continue
        dx_t = 2.0 / n_try
        ts_t = n_try / 2.0
        x_old = np.linspace(0, 2, n_best, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_int = project_feasible(np.interp(x_new, x_old, best_h), ts_t)
        h_opt, c5_v = slsqp_opt(h_int, dx_t, ts_t, 500, 600)
        if c5_v < best_c5:
            best_c5 = c5_v
            best_h = h_opt
            if remaining() > 40:
                h_sg, c5_sg = subgrad_adaptive(best_h, dx_t, ts_t, max_iter=20000, top_m=3)
                if c5_sg < best_c5:
                    best_c5 = c5_sg
                    best_h = h_sg
            if remaining() > 30:
                h_opt2, c5_v2 = slsqp_opt(best_h, dx_t, ts_t, 5000, 800)
                if c5_v2 < best_c5:
                    best_c5 = c5_v2
                    best_h = h_opt2

    # Phase 7: Diverse starts at best n
    n_b = len(best_h)
    dx_b = 2.0 / n_b
    ts_b = n_b / 2.0
    for trial in range(25):
        if remaining() < 15:
            break
        style = trial % 6
        h_s = make_start(n_b, ts_b, rng, style)
        h_opt, c5_v = slsqp_opt(h_s, dx_b, ts_b, 300, 500)
        if c5_v < best_c5:
            best_c5 = c5_v
            best_h = h_opt
            if remaining() > 25:
                h_sg, c5_sg = subgrad_adaptive(best_h, dx_b, ts_b, max_iter=20000, top_m=3)
                if c5_sg < best_c5:
                    best_c5 = c5_sg
                    best_h = h_sg

    # Phase 8: Simulated annealing (new)
    if remaining() > 40:
        h_sa, c5_sa = simulated_anneal(best_h, dx_b, ts_b, max_iter=20000, rng=rng)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa
            if remaining() > 20:
                h_opt, c5_v = slsqp_opt(best_h, dx_b, ts_b, 5000, 1000)
                if c5_v < best_c5:
                    best_c5 = c5_v
                    best_h = h_opt

    # Phase 9: Final intensive polish
    n_f = len(best_h)
    dx_f = 2.0 / n_f
    ts_f = n_f / 2.0
    if remaining() > 25:
        for beta in [10000, 30000]:
            if remaining() < 15:
                break
            h_f, c5_f = slsqp_opt(best_h, dx_f, ts_f, beta, 2000)
            if c5_f < best_c5:
                best_c5 = c5_f
                best_h = h_f

    if remaining() > 15:
        h_as = coordinate_push_top(best_h, dx_f, ts_f, top_m=4, blend=0.9)
        c5_as = c5_of_h(h_as, dx_f)
        if c5_as < best_c5:
            best_c5 = c5_as
            best_h = h_as

    if remaining() > 10:
        h_sg, c5_sg = subgrad_adaptive(best_h, dx_f, ts_f, max_iter=15000, top_m=2)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg

    return best_h, best_c5, len(best_h)
