def eval_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def build_sym(q, n):
    m = n // 2
    h = np.empty(n)
    h[:m] = q
    h[m:] = 1.0 - q[::-1]
    return h

def to_sym_q(h, n):
    m = n // 2
    return 0.5 * (h[:m] + 1.0 - h[m:][::-1])

def grad_corr(h, d, n):
    """Gradient of correlation at lag d with respect to h."""
    g = np.zeros(n)
    if d >= 0:
        # C_d = sum_{i=d}^{n-1} h[i](1-h[i-d])
        # dC_d/dh[x] = (1-h[x-d]) for x in [d, n-1]
        #              - h[x+d] for x in [0, n-1-d]
        if d == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[d:] += (1.0 - h[:n-d])
            g[:n-d] -= h[d:]
    else:
        # d < 0, let d' = -d
        dp = -d
        # C_{-dp} = sum_{i=0}^{n-1-dp} h[i](1-h[i+dp])
        # dC/dh[x] = (1-h[x+dp]) for x in [0, n-1-dp]
        #            - h[x-dp] for x in [dp, n-1]
        g[:n-dp] += (1.0 - h[dp:])
        g[dp:] -= h[:n-dp]
    return g

def grad_q_from_h(gh, n):
    """Transform gradient from h-space to q-space for anti-symmetric rep."""
    m = n // 2

    m = n // 2

import numpy as np
from scipy.optimize import minimize
import time


def eval_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))


def build_sym(q, n):
    m = n // 2
    h = np.empty(n)
    h[:m] = q
    h[m:] = 1.0 - q[::-1]
    return h


def to_sym_q(h, n):
    m = n // 2
    return 0.5 * (h[:m] + 1.0 - h[m:][::-1])


def grad_corr(h, d, n):
    g = np.zeros(n)
    if d == 0:
        g[:] = 1.0 - 2.0 * h
    elif d > 0:
        g[d:] += (1.0 - h[:n - d])
        g[:n - d] -= h[d:]
    else:
        dp = -d
        g[:n - dp] += (1.0 - h[dp:])
        g[dp:] -= h[:n - dp]
    return g


def grad_q_from_h(gh, n):
    m = n // 2
    return gh[:m] - gh[m:][::-1]


def select_active(h, n, margin=0.005, pad=3):
    cv = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    mx = cv.max()
    thresh = mx - max(1e-8, margin * mx)
    idx = set(np.where(cv >= thresh)[0])
    for a in list(idx):
        for delta in range(-pad, pad + 1):
            k = a + delta
            if 0 <= k <= 2 * n - 2:
                idx.add(k)
    return np.array(sorted(idx), dtype=int)


def interp_to(h, n_new):
    n_old = len(h)
    if n_new == n_old:
        return h.copy()
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    return np.interp(x_new, x_old, h)

class LSEModel:
    def __init__(self, n, tau):
        self.n = n
        self.m = n // 2
        self.tau = tau
        self.dx = 2.0 / n

    def fg(self, z):
        q = z[:self.m]
        h = build_sym(q, self.n)
        C = np.correlate(h, 1.0 - h, mode='full') * self.dx
        M = C.max()
        scores = (C - M) / self.tau
        scores = np.clip(scores, -400, 400)
        exp_s = np.exp(scores)
        Z = exp_s.sum()
        if Z < 1e-300:
            p = np.ones(2 * self.n - 1) / (2 * self.n - 1)
        else:
            p = exp_s / Z
        phi = M + self.tau * np.log(max(Z, 1e-300))
        gh = np.zeros(self.n)
        for j in range(2 * self.n - 1):
            w = p[j]
            if w < 1e-14:
                continue
            d = j - (self.n - 1)
            gh += w * grad_corr(h, d, self.n)
        gh *= self.dx
        gq = grad_q_from_h(gh, self.n)
        return phi, gq


class ActiveModel:
    def __init__(self, n, active_k, mu):
        self.n = n
        self.m = n // 2
        self.ak = active_k
        self.mu = mu
        self.dx = 2.0 / n

    def fg(self, z):
        q = z[:self.m]
        t = z[self.m]
        h = build_sym(q, self.n)
        cv = np.correlate(h, 1.0 - h, mode='full') * self.dx
        vals = cv[self.ak]
        exc = np.maximum(0.0, vals - t)
        f = t + 0.5 * self.mu * np.dot(exc, exc)
        gq = np.zeros(self.m)

                d = self.ak[i] - (self.n - 1)
                gq += self.mu * exc[i] * grad_q_from_h(grad_corr(h, d, self.n), self.n)
        gf = np.zeros(self.m + 1)
        gf[self.m] = 1.0 - self.mu * exc.sum()
        gf[:self.m] = gq
        return f, gf
