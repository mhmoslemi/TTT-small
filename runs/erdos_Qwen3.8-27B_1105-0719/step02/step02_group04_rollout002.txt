```
def solve_qp_step(h, active, cv, eps, solver):
    n = len(h)
    G = np.stack([corr_grad_h(h, int(j-(n-1)), n) for j in active])
    rhs = cv[active]
    d = cp.Variable(n)
    t = cp.Variable()
    cons = [cp.sum(d) == 0.0]
    # G d <= t - rhs  => G d - t <= -rhs
    cons.append(G @ d - t <= -rhs)
    cons.append(d >= -h)
    cons.append(d <= 1.0 - h)
    obj = cp.Minimize(t + 0.5*eps*cp.sum_squares(d))
    prob = cp.Problem(obj, cons)
    try:
        prob.solve(solver=solver)
    except Exception:
        try:
            prob.solve()
        except Exception:
            return None
    if d.value is None:
        return None
    return np.clip(d.value, -h, 1.0-h)
```

```python
        mid = 0.5 * (lo + hi)
        if np.clip(x - mid, 0.0, 1.0).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.any():
            h = h.copy()
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return np.clip(h, 0.0, 1.0)


def remaining_time(t0, budget_s):
    return budget_s - (time.time() - t0)


# ---------------- analytic gradient of C_j w.r.t. h ----------------
def corr_grad_h(h, d, n):
    """Gradient (w.r.t. h) of the (unscaled) full-correlation entry at lag offset d."""
    g = np.zeros(n)
    lo1 = max(0, d)
    hi1 = min(n - 1, n - 1 + d)
    if lo1 <= hi1:
        g[lo1:hi1 + 1] += 1.0 - h[lo1 - d:hi1 + 1 - d]
    lo2 = max(0, -d)
    hi2 = min(n - 1, n - 1 - d)
    if lo2 <= hi2:
        g[lo2:hi2 + 1] += -h[lo2 + d:hi2 + 1 + d]
    return g


# ---------------- active lags ----------------
def select_active(h, n, margin_frac=0.02, pad=2, cap=80):
    dx = 2.0 / n
    cv = corr_full(h) * dx
    mx = cv.max()
    thresh = mx - margin_frac * mx
    idx = np.where(cv >= thresh)[0]
    order = idx[np.argsort(-cv[idx])]
    idx = order[:cap]
    out = set(int(x) for x in idx)
    for a in list(out):
        for k in range(a - pad, a + pad + 1):
            if 0 <= k <= 2 * n - 2:
                out.add(int(k))
    arr = np.array(sorted(out), dtype=int)
    if len(arr) > cap + 4 * pad:
        keep = np.argsort(-corr_full(h) * dx)[0]  # fallback safe
    return arr[:cap + 8]


def build_sym(q, n):
    m = n // 2
    h = np.empty(n)
    h[:m] = q
    h[m:] = 1.0 - q[::-1]
    return h


def to_sym_q(h, n):
    m = n // 2
    return 0.5 * (h[:m] + 1.0 - h[n - 1:m - 1:-1])


def interp_to(h, n_new):
    n_old = len(h)
    if n_new == n_old:
        return h.copy()
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h)
    return project_to_target(h_new, n_new / 2.0)


# ---------------- QP linearized (Chebyshev) step ----------------
def solve_qp_step(h, active, cv, eps, solver):
    n = len(h)
    rows = []
    for j in active:
        d = int(j) - (n - 1)
        rows.append(corr_grad_h(h, d, n))
    G = np.asarray(rows, dtype=float)
    rhs = cv[active]
    dvar = cp.Variable(n)
    tvar = cp.Variable()
    cons = [cp.sum(dvar) == 0.0]
    cons.append(G @ dvar - tvar <= -rhs)
    cons.append(dvar >= -h)
    cons.append(dvar <= 1.0 - h)
    obj = cp.Minimize(tvar + 0.5 * eps * cp.sum_squares(dvar))
    prob = cp.Problem(obj, cons)
    dv = None
    try:
        prob.solve(solver=solver)
        if dvar.value is not None:
            dv = np.asarray(dvar.value, dtype=float)
    except Exception:
        dv = None
    if dv is None:
        try:
            prob.solve()
            if dvar.value is not None:
                dv = np.asarray(dvar.value, dtype=float)
        except Exception:
            dv = None
    if dv is None or not np.all(np.isfinite(dv)):
        return None
    return np.clip(dv, -h, 1.0 - h)


def line_search(h, dvec, cur_c5):
    n = len(h)
    best_a = 0.0
    best_v = cur_c5
    best_h = None
    for a in (1.0, 0.6, 0.35, 0.2, 0.1, 0.05, 0.02):
        hn = np.clip(h + a * dvec, 0.0, 1.0)
        hn = project_to_target(hn, n / 2.0)
        v = eval_c5(hn)
        if v < best_v - 1e-12:
            best_v = v
            best_a = a
            best_h = hn
    return best_h, best_v


# ---------------- linearized active-set local search ----------------
def lp_local_search(h, t0, budget_s, reserve, solver='ECOS', max_iter=800):
    n = len(h)
    best = h.copy()
    bc = eval_c5(best)
    eps = 1e-3
    margins = (0.05, 0.03, 0.02, 0.01, 0.006, 0.003, 0.0015, 0.0008)
    stall = 0
    rng_l = np.random.RandomState(1000)
    for it in range(max_iter):
        if remaining_time(t0, budget_s) < reserve + 5.0:
            break
        cv = corr_full(best)
        cmax_s = (cv * (2.0 / n)).max()
        mf = margins[min(stall, len(margins) - 1)]
        active = select_active(best, n, margin_frac=mf, pad=2, cap=80)
        if len(active) == 0:
            break
        dvec = solve_qp_step(best, active, cv, eps, solver)
        took = 0
        if dvec is not None and np.max(np.abs(dvec)) > 1e-9:
            hn2, v2 = line_search(best, dvec, bc)
            if hn2 is not None:
                best = hn2
                bc = v2
                stall = 0
                took = 1
        if took == 0:
            stall += 1
            if stall >= 3 and rng_l.rand() < 0.3:
                delta = rng_l.randn(n) * 0.01
                hp = project_to_target(np.clip(best + delta, 0.0, 1.0), n / 2.0)
                vp = eval_c5(hp)
                if vp < bc:
                    best = hp
                    bc = vp
                    stall = 0
            if stall >= len(margins) + 3:
                eps = max(1e-6, eps * 0.5)
                stall = 0
    return best, bc, n


# ---------------- smoothed-max L-BFGS polish (parent idea, compact) ----------------
def smooth_fg(z, n, active, mu, mu_sum, target):
    h = z[:n]
    t = z[n]
    cv = corr_full(h) * (2.0 / n)
    vals = cv[active]
    exc = np.maximum(0.0, vals - t)
    sdev = h.sum() - target
    f = t + 0.5 * mu * float(np.dot(exc, exc)) + 0.5 * mu_sum * sdev * sdev
    gf = np.zeros(n + 1)
    gf[n] = 1.0 - mu * exc.sum()
    gh = np.zeros(n)
    for i in range(len(active)):
        if exc[i] > 0.0:
            d = int(active[i]) - (n - 1)
            gh += mu * exc[i] * corr_grad_h(h, d, n)
    gf[:n] = gh + mu_sum * sdev
    return f, gf


class SmoothModel:
    def __init__(self, n, active, mu, mu_sum):
        self.n = n
        self.active = active
        self.mu = mu
        self.mu_sum = mu_sum
        self.target = n / 2.0

    def fg(self, z):
        return smooth_fg(z, self.n, self.active, self.mu, self.mu_sum, self.target)


def smooth_polish(h0, n, active, mu=2000.0, mu_sum=3000.0, maxiter=200, gtol=1e-11):
    t0c = eval_c5(h0)
    model = SmoothModel(n, active, mu, mu_sum)
    z0 = np.concatenate([h0, [t0c]])
    bounds = [(0.0, 1.0)] * n + [(0.0, t0c + 0.5)]
    try:
        res = minimize(model.fg, z0, jac=True, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'gtol': gtol})
        h_opt = np.clip(res.x[:n], 0.0, 1.0)
        h_opt = project_to_target(h_opt, n / 2.0)
    except Exception:
        return h0, eval_c5(h0)
    return h_opt, eval_c5(h_opt)


def make_random_sym(n, rng):
    m = n // 2
    t = np.linspace(0.0, 1.0, m)
    q = 0.5 + 0.3 * np.sin(2.0 * np.pi * 3.0 * t + rng.uniform(0, 6.28))
    return np.clip(q, 0.0, 1.0)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')

    try:
        h_init = np.asarray(initial_h_values, dtype=float).reshape(-1)
        if h_init.size > 0 and np.all(np.isfinite(h_init)):
            ni = len(h_init)
            h_init = project_to_target(h_init, ni / 2.0)
            best_h = h_init.copy()
            best_c5 = eval_c5(h_init)
    except Exception:
        best_h = None

    if best_h is None:
        nd = 126
        md = nd // 2
        tt = np.linspace(0.0, 1.0, md)
        qd = np.clip(0.5 + 0.25 * np.sin(2.0 * np.pi * 3.0 * tt), 0.0, 1.0)
        best_h = build_sym(qd, nd)
        best_c5 = eval_c5(best_h)

    reserve = max(20.0, 0.05 * budget_s)

    def try_accept(hc, cc):
        nonlocal best_h, best_c5
        if cc < best_c5 - 1e-12:
            best_h = hc.copy()
            best_c5 = cc
            return True
        return False

    # ---- Phase 1: linearized active-set search on parent scale ----
    if remaining_time(t0, budget_s) > reserve + 20.0:
        h1, c1, n1 = lp_local_search(best_h, t0, budget_s, reserve, solver='ECOS', max_iter=1400)
        try_accept(h1, c1)

    # ---- Phase 2: smoothed-max L-BFGS polish ----
    if remaining_time(t0, budget_s) > reserve + 20.0:
        nc = len(best_h)
        act = select_active(best_h, nc, margin_frac=0.01, pad=3, cap=100)
        for mu in (500.0, 3000.0, 10000.0):
            if remaining_time(t0, budget_s) < reserve + 15.0:
                break
            hf, cf = smooth_polish(best_h, nc, act, mu=mu, mu_sum=3000.0,
                                   maxiter=200, gtol=1e-12)
            if cf < best_c5:
                best_h = hf.copy()
                best_c5 = cf
                act = select_active(best_h, nc, margin_frac=0.005, pad=3, cap=100)

    # ---- Phase 3: re-run linearized search after polish ----
    if remaining_time(t0, budget_s) > reserve + 25.0:
        h3, c3, n3 = lp_local_search(best_h, t0, budget_s, reserve, solver='SCS', max_iter=900)
        try_accept(h3, c3)

    # ---- Phase 4: try a couple of nearby discretizations ----
    nb = len(best_h)
    n_cands = set()
    for delta in (-6, -4, -2, 2, 4, 6, 10):
        nc_ = nb + delta
        if nc_ >= 40 and nc_ % 2 == 0:
            n_cands.add(nc_)
    n_cands.add(max(40, nb // 2))
    for nc_ in sorted(n_cands):
        if remaining_time(t0, budget_s) < reserve + 30.0:
            break
        if nc_ == nb:
            continue
        ht = interp_to(best_h, nc_)
        ht, ct, _ = lp_local_search(ht, t0, budget_s, reserve, solver='ECOS', max_iter=400)
        if ct < best_c5 and remaining_time(t0, budget_s) > reserve + 15.0:
            act2 = select_active(ht, nc_, margin_frac=0.01, pad=3, cap=100)
            hf, cf = smooth_polish(ht, nc_, act2, mu=3000.0, mu_sum=3000.0, maxiter=150)
            ct = min(ct, cf)
            if cf < ct:
                ht = hf
        try_accept(ht, ct)

    # ---- Phase 5: a few random symmetric restarts (escape) ----
    max_restarts = max(2, int(remaining_time(t0, budget_s) / 25.0))
    max_restarts = min(max_restarts, 10)
    for _ in range(max_restarts):
        if remaining_time(t0, budget_s) < reserve + 20.0:
            break
        nr = len(best_h)
        if nr % 2 != 0:
            nr -= 1
        qr = make_random_sym(nr, rng)
        hr = build_sym(qr, nr)
        hr, cr, _ = lp_local_search(hr, t0, budget_s, reserve, solver='ECOS', max_iter=250)
        try_accept(hr, cr)

    # ---- Phase 6: final tight linearized + polish on best ----
    if remaining_time(t0, budget_s) > 15.0 and best_h is not None:
        nf_ = len(best_h)
        hf_, cf_, _ = lp_local_search(best_h, t0, budget_s, 5.0, solver='ECOS', max_iter=400)
        try_accept(hf_, cf_)
        if remaining_time(t0, budget_s) > 10.0:
            actf = select_active(best_h, nf_, margin_frac=0.004, pad=4, cap=120)
            hf2, cf2 = smooth_polish(best_h, nf_, actf, mu=10000.0, mu_sum=5000.0,
                                     maxiter=200, gtol=1e-13)
            try_accept(hf2, cf2)

    # final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_target(best_h, len(best_h) / 2.0)
    best_c5 = eval_c5(best_h)
    return best_h, float(best_c5), int(len(best_h))
```
