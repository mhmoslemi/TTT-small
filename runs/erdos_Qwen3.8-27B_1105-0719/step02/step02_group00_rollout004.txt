def repair_feasible(vec):

    else:

def vandercorput_pattern(n):
    """Van der Corput sequence for balanced pattern."""
    half = n // 2
    vals = np.zeros(n)

def active_polish(h, max_iters=200, time_limit=30):
    """Active-set gradient descent with adaptive steps."""
    n = len(h)
    dx = 2.0 / n
    t_start = time.perf_counter()
    h = h.copy()
    best_c = exact_score(h)
    velocity = np.zeros(n)
    alpha = 0.005
    momentum = 0.5
    
    for it in range(max_iters):
        if time.perf_counter() - t_start > time_limit:
            break
        profile = compute_profile(h)
        c = np.max(profile)
        if c < best_c:
            best_c = c
        
        # Active set: top constraints
        thresh = c - max(0.003, 4 * (c - np.median(profile)))
        active = np.where(profile >= thresh)[0]
        if len(active) == 0:
            active = np.array([np.argmax(profile)])
        if len(active) > 20:
            top_idx = np.argsort(profile[active])[-20:]
            active = active[top_idx]
        
        # Compute gradient for active constraints
        # d/dh_i of profile[m] where delta = m-(n-1)
        grad = np.zeros(n)
        for m in active:
            d = m - (n - 1)
            w = (profile[m] - thresh)
            if d == 0:
                grad += w * dx * (1.0 - 2.0 * h)
            elif d > 0:
                hi = np.arange(d, n)
                lo = np.arange(0, n - d)
                grad[hi] += w * dx * (1.0 - h[lo])
                grad[lo] -= w * dx * h[hi]
            else:
                dp = -d
                lo = np.arange(0, n - dp)
                hi = np.arange(dp, n)
                grad[lo] += w * dx * (1.0 - h[hi])
                grad[hi] -= w * dx * h[lo]
        
        grad -= grad.mean()
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad /= gnorm
        
        velocity = momentum * velocity + grad
        vnorm = np.linalg.norm(velocity)
        if vnorm > 0:
            velocity /= vnorm
        
        # Adaptive step
        improved = False
        for a in [alpha, alpha * 0.5, alpha * 0.25, alpha * 0.1]:
            h_try = repair_feasible(h - a * velocity)

                h = h_try

                improved = True
                break
        if not improved:
            alpha *= 0.7
            if alpha < 1e-7:
                break
        else:

    return h, best_c

        if res.x is not None and np.isfinite(res.x).all():

            if c_new < c0 + 1e-8:

def interpolate_to_n(h, new_n):
    """Mass-preserving interpolation."""
    n = len(h)
    if n == new_n:
        return h.copy()
    # Bin-average interpolation
    edges_old = np.linspace(0, 2, n + 1)
    edges_new = np.linspace(0, 2, new_n + 1)
    h_new = np.zeros(new_n)
    for k in range(new_n):
        lo = edges_new[k]
        hi = edges_new[k + 1]
        width = hi - lo
        total = 0.0
        for j in range(n):
            overlap = min(hi, edges_old[j + 1]) - max(lo, edges_old[j])
            if overlap > 0:
                total += h[j] * overlap
        h_new[k] = total / width if width > 0 else 0
    return repair_feasible(np.clip(h_new, 0, 1))
