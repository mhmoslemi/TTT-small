import numpy as np
from scipy.optimize import linprog
import time

def dx_of(n):
    return 2.0 / n

def corr_C(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * dx_of(n)

def eval_c5(h):
    return float(np.max(corr_C(h)))

def repair_box_sum(h, target=None):
    n = len(h)
    if target is None:
        target = n / 2.0
    lo, hi = h.min() - 2.0, h.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h_new = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - h_new.sum()
    if abs(diff) > 1e-14:
        mask = (h_new > 1e-12) & (h_new < 1.0 - 1e-12)
        if mask.any():
            h_new[mask] += diff / mask.sum()
    return np.clip(h_new, 0.0, 1.0)

def build_h_from_q(q, n):
    m = n // 2
    h = np.empty(n)
    h[:m] = q
    h[m:] = 1.0 - q[::-1]
    return h

def q_from_h(h, n):
    m = n // 2
    return 0.5 * (h[:m] + 1.0 - h[n - 1:m - 1:-1])

def grad_h_for_lag(h, d, n):
    g = np.zeros(n)
    L = max(0, -d)
    U = min(n - 1, n - 1 - d)
    j_lo = L + d
    j_hi = U + d
    if j_lo <= j_hi:
        g[j_lo:j_hi + 1] += 1.0 - h[j_lo - d:j_hi + 1 - d]
    if L <= U:
        g[L:U + 1] += -h[L + d:U + 1 + d]
    return g

def grad_q_from_gh(gh, n):
    m = n // 2
    return gh[:m] - gh[n - 1:m - 1:-1]

def select_active(C, cmax, margin=0.005, pad=2, max_count=80):
    n_pts = len(C)
    thresh = cmax - margin
    idx = np.where(C >= thresh)[0]
    if len(idx) == 0:
        idx = np.array([np.argmax(C)])
    extended = set()
    for a in idx:
        for delta in range(-pad, pad + 1):
            k = a + delta
            if 0 <= k < n_pts:
                extended.add(k)
    if len(extended) > max_count:
        arr = np.array(sorted(extended))
        order = np.argsort(-C[arr])
        selected = arr[order[:max_count]]
        extended = set(selected.tolist())
    return np.array(sorted(extended), dtype=int)

    dx = dx_of(n)
    best_q = q.copy()

    h = build_h_from_q(q, n)
    C = corr_C(h)
    cmax = C.max()
    no_progress = 0
    R_cur = R

    for it in range(max_iter):
        if time.time() - t_start > time_limit:
            break
        if time.time() - t_start < 0:
            break
        elapsed = time.time() - t_start
        if elapsed > time_limit:
            break

        C = corr_C(h)
        cmax = C.max()

        active = select_active(C, cmax, margin=margin, pad=2, max_count=min(80, 2 * n))
        if len(active) == 0:
            active = np.array([np.argmax(C)])

        # Build LP
        # min t s.t. grads[i]·delta - t <= -C[j_i] for each active, box on delta
        A_rows = []
        b_rows = []
        for j in active:
            d = int(j) - (n - 1)
            gh = grad_h_for_lag(h, d, n)
            gq = grad_q_from_gh(gh, n) * dx
            A_rows.append(np.concatenate([gq, [-1.0]]))

            hi = min(R_cur, 1.0 - q[i])

        c_obj[m] = 1.0  # minimize t

            if R_cur < 1e-6:
                break
            continue

            continue

            R_cur *= 0.6

            if R_cur < 1e-6:
                break

def interp_to(h, n_new):
    n_old = len(h)
    if n_new == n_old:
        return h.copy()
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h)
    return repair_box_sum(h_new, n_new / 2.0)
