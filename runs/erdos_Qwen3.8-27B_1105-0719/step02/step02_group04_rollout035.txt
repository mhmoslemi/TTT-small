    h = np.empty(n)
    h[:m] = q
    h[m:] = 1.0 - q[::-1]

    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def compute_phi_grad(q, n, tau):
    """Compute log-sum-exp objective and gradient w.r.t. q."""
    m = n // 2
    h = build_anti(q, n)
    C = corr_scaled(h)
    M = C.max()
    scores = (C - M) / tau
    # For numerical stability, clip very negative values
    scores = np.clip(scores, -500, 500)

        p = np.full(2*n - 1, 1.0 / (2*n - 1))
    else:
        phi = M + tau * np.log(Z)

    return phi, grad_q

    m = n // 2

        return compute_phi_grad(q, n, tau)
    
    bounds = [(0.0, 1.0)] * m
    try:
        res = minimize(obj_and_grad, q0.copy(), jac=True, method='L-BFGS-B',
                      bounds=bounds, options={'maxiter': maxiter, 'gtol': gtol})
        q_opt = np.clip(res.x, 0.0, 1.0)
    except Exception:
        q_opt = q0.copy()
    return q_opt

from scipy.optimize import minimize
import time

def eval_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def build_anti(q, n):
    m = n // 2
    h = np.empty(n)
    h[:m] = q
    h[m:] = 1.0 - q[::-1]
    return h

    m = n // 2
    return 0.5 * (h[:m] + 1.0 - h[n-1:m-1:-1])

def corr_scaled(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def fs_phi_grad_anti(q, n, tau):
    """Log-sum-exp objective and gradient for anti-symmetric representation."""
    m = n // 2
    dx = 2.0 / n
    h = build_anti(q, n)
    C = corr_scaled(h)
    M = C.max()
    scores = (C - M) / tau
    scores = np.clip(scores, -100, 100)
    exp_scores = np.exp(scores)
    Z = exp_scores.sum()
    if Z < 1e-15:
        return M, np.zeros(m)
    phi = M + tau * np.log(Z)
    p = exp_scores / Z

    g = np.zeros(n)
    for d in range(-(n-1), n):
        w = p[d + n - 1]
        if w < 1e-14:
            continue
        if d >= 0:
            i_start, i_end = d, n
        else:
            i_start, i_end = 0, n + d
        length = i_end - i_start
        if length <= 0:
            continue

    grad_q = g[:m] - g[n-1:m-1:-1]

                q_r = lsbfgs_safe(obj_rq, q_r, bd_best, 150, 1e-8)
                h_r = build_anti(q_r, best_n)
                c5_r = eval_c5(h_r)
                if c5_r < best_c5 - 1e-13:
                    best_c5 = c5_r
                    best_h = h_r.copy()
                    best_q = q_r.copy()
                    # Deeper polish from improvement
                    if rem() > reserve + 8:
                        for tau2 in [0.0005, 0.0002]:
                            if rem() < reserve + 2:
                                break
                            q_r = lsbfgs_safe(obj_rq, q_r, bd_best, 250, 1e-10)
                            h_r2 = build_anti(q_r, best_n)
                            c5_r2 = eval_c5(h_r2)
                            if c5_r2 < best_c5 - 1e-13:
                                best_c5 = c5_r2
                                best_h = h_r2.copy()
                                best_q = q_r.copy()

    # Phase 5: Full-variable polish on best
    if rem() > reserve + 8:
        h_pol = best_h.copy()
        n_pol = len(h_pol)
        h_pol = project_sum(h_pol, n_pol)
        bd_pol = [(0.0, 1.0)] * n_pol
        for tau in [0.001, 0.0005, 0.0002]:
            if rem() < reserve + 2:
                break
            lam = 5000.0

            def obj_pol(h):
                return fs_phi_grad_full(h, n_pol, tau, lam)

            h_pol = lsbfgs_safe(obj_pol, h_pol, bd_pol, 300, 1e-10)
            h_pol = np.clip(h_pol, 0, 1)
            h_pol = project_sum(h_pol, n_pol)
            c5_pol = eval_c5(h_pol)
            if c5_pol < best_c5 - 1e-13:
                best_c5 = c5_pol
                best_h = h_pol.copy()
                best_n = n_pol
                if n_pol % 2 == 0:
                    best_q = q_from_h(h_pol, n_pol)

    # Phase 6: Final anti-symmetric micro-polish
    if rem() > 4 and best_q is not None:
        n_fin = best_n
        if n_fin % 2 == 0:
            q_fin = best_q.copy()
            bd_fin = [(0.0, 1.0)] * (n_fin // 2)
            for tau in [0.0003, 0.0001]:
                if rem() < 2:
                    break

                def obj_fin(q):
                    return fs_phi_grad_anti(q, n_fin, tau)

                q_fin = lsbfgs_safe(obj_fin, q_fin, bd_fin, 500, 1e-11)
                h_fin = build_anti(q_fin, n_fin)
                c5_fin = eval_c5(h_fin)
                if c5_fin < best_c5 - 1e-13:
                    best_c5 = c5_fin
                    best_h = h_fin.copy()
                    best_q = q_fin.copy()

    # Final validation
    best_h = np.clip(best_h, 0, 1)
    n_out = len(best_h)
    if n_out % 2 == 0 and best_q is not None and len(best_q) == n_out // 2:
        best_h = build_anti(best_q, n_out)
    else:
        best_h = project_sum(best_h, n_out)
    final_c5 = eval_c5(best_h)
    if not np.isfinite(final_c5) or final_c5 > 0.5:
        n_fb = 126
        best_h = build_anti(np.full(n_fb // 2, 0.5), n_fb)
        final_c5 = eval_c5(best_h)
        n_out = n_fb

    return best_h, final_c5, n_out
