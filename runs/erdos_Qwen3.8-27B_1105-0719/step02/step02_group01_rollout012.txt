def make_feasible(x, target_sum):
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        h = np.clip(x - mid, 0.0, 1.0)
        s = h.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            slack = 1.0 - h
            ts = slack.sum()
            if ts > 1e-12:
                h += (diff / ts) * slack
        else:
            cap = h.copy()
            tc = cap.sum()
            if tc > 1e-12:
                h += (diff / tc) * cap
    return np.clip(h, 0.0, 1.0)

def compute_c5(h, dx):
    corrs = np.correlate(h, 1.0 - h, mode='full')
    return corrs.max() * dx

def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def grad_corr_k(h, k, n):
    off = k - (n - 1)
    grad = np.zeros(n)
    if off >= 0:
        # Valid i: 0 <= i < n and 0 <= i+off < n => 0 <= i < n-off
        i_max = n - off
        if i_max <= 0:
            return grad
        idx = np.arange(i_max)
        # d/dc h[i+off] += (1 - h[i])
        grad[off:off+i_max] += (1.0 - h[:i_max])
        # d/dc h[i] += -h[i+off]
        grad[:i_max] += (-h[off:off+i_max])
    else:
        d = -off
        i_max = n - d
        if i_max <= 0:
            return grad
        idx = np.arange(i_max)
        # d/dc h[i+off] = h[i-d] += (1 - h[i])
        grad[:i_max] += (1.0 - h[d:d+i_max])
        # d/dc h[i] += -h[i+off] = -h[i-d]
        grad[d:d+i_max] += (-h[:i_max])
    return grad
