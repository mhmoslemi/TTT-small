    return float(np.max(raw_corrs(h)) * 2.0 / n)

def grad_corr_k(h, k, n):
    offset = k - (n - 1)
    grad = np.zeros(n)
    if offset == 0:
        grad[:] = 1.0 - 2.0 * h
    elif offset > 0:
        m = n - offset
        i1 = np.arange(m)
        i2 = i1 + offset
        grad[i1] += 1.0 - h[i2]
        grad[i2] -= h[i1]
    else:
        d = -offset
        i1 = np.arange(d, n)
        i2 = i1 - d
        grad[i1] += 1.0 - h[i2]
        grad[i2] -= h[i1]
    return grad

    """Return (2n-1, n) array of gradients for all lags."""
    L = 2 * n - 1
    G = np.zeros((L, n))
    for k in range(L):
        G[k] = grad_corr_k(h, k, n)
    return G

    corrs = raw_corrs(h)
    return float(logsumexp(beta * corrs) / beta)

def lse_grad(h, beta, n):
    corrs = raw_corrs(h)
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    w /= w.sum()
    grad = np.zeros(n)
    for k in range(len(corrs)):
        if w[k] > 1e-10:

    return grad

        return lse_grad(x, beta, n) * dx
