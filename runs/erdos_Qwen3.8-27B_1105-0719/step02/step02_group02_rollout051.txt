def project_feasible(x, target_sum):
    n = len(x)
    if n == 0:
        return x
    lo = float(np.min(x)) - n - 1.0
    hi = float(np.max(x)) + n + 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h


def compute_c5(h):

    corrs = np.correlate(h, 1.0 - h, mode='full')
    return float(corrs.max() * dx), corrs


def corr_grad_k(h, k, n):

        grad[:] = 1.0 - 2.0 * h

def smooth_max_c5(h, beta):
    n = len(h)
    dx = 2.0 / n
    corrs = np.correlate(h, 1.0 - h, mode='full')
    return float(logsumexp(beta * corrs) / beta * dx)


def smooth_max_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corrs = np.correlate(h, 1.0 - h, mode='full')
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones_like(corrs) / len(corrs)
    else:
        w /= w_sum
    grad = np.zeros(n)
    active = np.where(w > 1e-12)[0]
    for k in active:
        grad += w[k] * corr_grad_k(h, k, n)
    return grad * dx


def slsqp_smooth(h0, target_sum, beta, maxiter):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum,
             'jac': lambda x: np.ones(n)}]
    try:

def active_subgrad(h0, target_sum, max_iter, top_k=4, init_alpha=0.08):

    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    best_c5 = float(np.correlate(h, 1.0 - h, 'full').max() * dx)
    alpha = init_alpha
    momentum = 0.88
    v = np.zeros(n)
    no_improve = 0

    for it in range(max_iter):
        corrs = np.correlate(h, 1.0 - h, mode='full')
        max_raw = float(corrs.max())
        c5_val = max_raw * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0

            no_improve += 1

            break

        top_idx = np.argsort(corrs)[-top_k:]
        top_vals = corrs[top_idx]
        spread = top_vals - top_vals[-2]
        weights = np.maximum(spread, 0.05)
        weights /= weights.sum()

            grad += wk * corr_grad_k(h, k, n)
        grad *= dx

        new_raw = float(np.correlate(h_new, 1.0 - h_new, mode='full').max())

            h = h_new
            alpha = min(alpha * 1.008, 0.25)
        else:
            alpha *= 0.94
            v *= 0.4

            break

    m = n // 2
    h = np.zeros(n)
    if n % 2 == 0:
        h[:m] = v
        h[m:] = 1.0 - v[::-1]
    else:
        h[:m] = v
        h[m] = 0.5
        h[m + 1:] = 1.0 - v[::-1]
    return h

        if remaining() < 25:
            break
        h_opt = slsqp_smooth(best_h, target_sum, beta, 500)
        c5_val, _ = compute_c5(h_opt)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h_opt

    # Phase 2: Active subgradient (focus on top lags)
    if remaining() > 60:
        n_iter = min(25000, int(remaining() * 25))
        h_sg, c5_sg = active_subgrad(best_h, target_sum, n_iter, top_k=4, init_alpha=0.06)
        if c5_sg < best_c5 - 1e-14:
            best_c5 = c5_sg
            best_h = h_sg

    # Phase 3: SLSQP after subgradient
    if remaining() > 30:
        for beta in [3000, 10000, 30000]:
            if remaining() < 20:
                break
            h_opt = slsqp_smooth(best_h, target_sum, beta, 500)
            c5_val, _ = compute_c5(h_opt)
            if c5_val < best_c5 - 1e-14:
                best_c5 = c5_val
                best_h = h_opt

    # Phase 4: Local perturbation search
    if remaining() > 40:
        h_lp, c5_lp = local_perturb_search(best_h, target_sum, rng, n_trials=150)
        if c5_lp < best_c5 - 1e-14:
            best_c5 = c5_lp
            best_h = h_lp
        # SLSQP polish after perturbation
        if remaining() > 20:
            h_opt = slsqp_smooth(best_h, target_sum, 5000, 400)
            c5_val, _ = compute_c5(h_opt)
            if c5_val < best_c5 - 1e-14:
                best_c5 = c5_val
                best_h = h_opt

    # Phase 5: Active subgradient with different parameters
    if remaining() > 80:
        h_sg2, c5_sg2 = active_subgrad(best_h, target_sum, min(20000, int(remaining() * 20)),
                                       top_k=3, init_alpha=0.04)
        if c5_sg2 < best_c5 - 1e-14:
            best_c5 = c5_sg2
            best_h = h_sg2
        if remaining() > 20:
            h_opt = slsqp_smooth(best_h, target_sum, 10000, 400)
            c5_val, _ = compute_c5(h_opt)
            if c5_val < best_c5 - 1e-14:
                best_c5 = c5_val
                best_h = h_opt

    # Phase 6: Complement parameterization
    if remaining() > 40:
        h_comp = optimize_complement(best_h, target_sum, 2000, 400)
        c5_comp, _ = compute_c5(h_comp)
        if c5_comp < best_c5 - 1e-14:
            best_c5 = c5_comp
            best_h = h_comp
            if remaining() > 20:
                h_opt = slsqp_smooth(best_h, target_sum, 5000, 300)
                c5_val, _ = compute_c5(h_opt)
                if c5_val < best_c5 - 1e-14:
                    best_c5 = c5_val
                    best_h = h_opt

    # Phase 7: Dimension sweep
    n_b = len(best_h)
    n_candidates = sorted(set([
        max(60, n_b - 20), max(60, n_b - 10),
        n_b + 2, n_b + 4, n_b + 6,
        n_b + 10, n_b + 20, n_b + 40,
        80, 100, 120, 140, 160, 180, 200, 220
    ]))
    n_candidates = [x for x in n_candidates if 40 <= x <= 260]

    for n_try in n_candidates:
        if remaining() < 25:
            break
        ts_try = n_try / 2.0
        x_old = (np.arange(len(best_h)) + 0.5) * (2.0 / len(best_h))
        x_new = (np.arange(n_try) + 0.5) * (2.0 / n_try)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, ts_try)

        h_opt = slsqp_smooth(h_interp, ts_try, 800, 250)
        c5_val, _ = compute_c5(h_opt)
        if c5_val < best_c5 - 1e-12:
            best_c5 = c5_val
            best_h = h_opt
            if remaining() > 35:
                h_opt2 = slsqp_smooth(h_opt, ts_try, 3000, 300)
                c5_val2, _ = compute_c5(h_opt2)
                if c5_val2 < best_c5 - 1e-14:
                    best_c5 = c5_val2
                    best_h = h_opt2
                if remaining() > 30:
                    h_sg3, c5_sg3 = active_subgrad(best_h, best_h.size / 2.0,
                                                    min(8000, int(remaining() * 15)),
                                                    top_k=4, init_alpha=0.05)
                    if c5_sg3 < best_c5 - 1e-14:
                        best_c5 = c5_sg3
                        best_h = h_sg3

    # Phase 8: Diverse restarts
    n_b = len(best_h)
    ts_b = n_b / 2.0
    for trial in range(25):
        if remaining() < 15:
            break
        if trial % 5 == 0:
            h_pert = best_h + rng.normal(0, 0.02, n_b)
        elif trial % 5 == 1:
            h_pert = best_h + rng.normal(0, 0.05, n_b)
        elif trial % 5 == 2:
            h_pert = best_h.copy()
            flip_idx = rng.choice(n_b, max(2, n_b // 20), replace=False)
            h_pert[flip_idx] = 1.0 - h_pert[flip_idx]
        elif trial % 5 == 3:
            # Smooth perturbation
            h_pert = best_h.copy()
            for _ in range(3):
                i = rng.randint(1, n_b - 1)
                amt = 0.03 * rng.choice([-1, 1])
                h_pert[i] += amt
                h_pert[i - 1] += 0.3 * amt
                h_pert[i + 1] += 0.3 * amt
        else:
            # Scale perturbation
            scale = 1.0 + 0.05 * rng.randn()
            h_pert = best_h * scale

        h_pert = project_feasible(h_pert, ts_b)
        h_opt = slsqp_smooth(h_pert, ts_b, 500, 250)
        c5_val, _ = compute_c5(h_opt)
        if c5_val < best_c5 - 1e-12:
            best_c5 = c5_val
            best_h = h_opt
            if remaining() > 25:
                h_opt2 = slsqp_smooth(h_opt, ts_b, 3000, 400)
                c5_val2, _ = compute_c5(h_opt2)
                if c5_val2 < best_c5 - 1e-14:
                    best_c5 = c5_val2
                    best_h = h_opt2
                if remaining() > 20:
                    h_sg4, c5_sg4 = active_subgrad(best_h, ts_b, min(8000, int(remaining() * 12)),
                                                    top_k=4, init_alpha=0.05)
                    if c5_sg4 < best_c5 - 1e-14:
                        best_c5 = c5_sg4
                        best_h = h_sg4

    # Phase 9: Final intensive polish
    n_b = len(best_h)
    ts_b = n_b / 2.0
    if remaining() > 20:
        for beta in [8000, 20000, 50000]:
            if remaining() < 12:
                break
            h_opt = slsqp_smooth(best_h, ts_b, beta, 600)
            c5_val, _ = compute_c5(h_opt)
            if c5_val < best_c5 - 1e-14:
                best_c5 = c5_val
                best_h = h_opt

    if remaining() > 50:
        h_sg5, c5_sg5 = active_subgrad(best_h, ts_b, min(12000, int(remaining() * 18)),
                                       top_k=3, init_alpha=0.03)
        if c5_sg5 < best_c5 - 1e-14:
            best_c5 = c5_sg5
            best_h = h_sg5

    if remaining() > 15:
        h_lp2, c5_lp2 = local_perturb_search(best_h, ts_b, rng, n_trials=80)
        if c5_lp2 < best_c5 - 1e-14:
            best_c5 = c5_lp2
            best_h = h_lp2

    if remaining() > 12:
        h_opt = slsqp_smooth(best_h, ts_b, 50000, 500)
        c5_val, _ = compute_c5(h_opt)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h_opt

    # Final validation and cleanup
    best_h = project_feasible(best_h, len(best_h) / 2.0)
    final_c5, _ = compute_c5(best_h)
    best_c5 = min(best_c5, final_c5)

    return best_h, float(best_c5), int(len(best_h))
