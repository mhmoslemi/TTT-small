def grad_corr_k(h, k, n):
    # Gradient of correlation at lag k
    offset = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -offset)
    i_hi = min(n - 1, n - 1 - offset)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx + offset] += (1.0 - h[idx])
    grad[idx] += (-1.0) * h[idx + offset]
    return grad

def lp_descent_direction(h, dx, target_sum, active_margin, step_bound):
    n = len(h)
    m = 2*n - 1
    corrs = compute_corrs(h)
    max_corr = corrs.max()
    
    # Active set
    active = np.where(corrs > max_corr - active_margin)[0]
    
    # LP: minimize t
    # subject to: G_active * d <= t (where G_active are gradients of active correlations)
    #             sum(d) = 0
    #             -h[i]*step_bound <= d[i] <= (1-h[i])*step_bound
    
    # Variables: [d_0, ..., d_{n-1}, t]
    # c = [0, ..., 0, 1] (minimize t)
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Inequality constraints: A_ub * x <= b_ub
    # For each active correlation k: grad_k · d - t <= 0
    A_ub_rows = []
    b_ub_rows = []
    for k in active:
        g = grad_corr_k(h, k, n)
        row = np.append(g, -1.0)
        A_ub_rows.append(row)
        b_ub_rows.append(0.0)
    
    # sum(d) = 0 → two inequalities: sum(d) <= 0 and -sum(d) <= 0
    # Actually, use equality constraint
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    A_ub = np.array(A_ub_rows) if A_ub_rows else np.zeros((0, n+1))
    b_ub = np.array(b_ub_rows) if b_ub_rows else np.zeros(0)
    
    # Bounds
    bounds = []
    for i in range(n):
        lo = -h[i] * step_bound
        hi = (1.0 - h[i]) * step_bound
        bounds.append((lo, hi))
    bounds.append((None, None))  # t is unbounded

def exact_line_search(h, d, dx):
    n = len(h)
    corrs = compute_corrs(h)
    m = len(corrs)
    
    # Max feasible alpha
    alpha_max = 1e10
    for i in range(n):
        if d[i] > 1e-15:
            alpha_max = min(alpha_max, (1.0 - h[i]) / d[i])
        elif d[i] < -1e-15:
            alpha_max = min(alpha_max, h[i] / (-d[i]))
    alpha_max = min(alpha_max, 1.0)  # cap at 1
    
    # Compute grad_k · d for all k
    gdot = np.zeros(m)
    for k in range(m):
        g = grad_corr_k(h, k, n)
        gdot[k] = g @ d
    
    # Breakpoints: alpha where corr[k1] + alpha*gdot[k1] = corr[k2] + alpha*gdot[k2]
    # alpha = (corrs[k2] - corrs[k1]) / (gdot[k1] - gdot[k2])
    breakpoints = [0.0, alpha_max]
    for i in range(m):
        for j in range(i+1, m):
            denom = gdot[i] - gdot[j]
            if abs(denom) > 1e-15:
                alpha = (corrs[j] - corrs[i]) / denom
                if 0 < alpha < alpha_max:
                    breakpoints.append(alpha)
    
    breakpoints = sorted(set(breakpoints))
    
    best_alpha = 0.0
    best_val = np.inf
    for alpha in breakpoints:
        new_corrs = corrs + alpha * gdot
        val = new_corrs.max()
        if val < best_val:
            best_val = val
            best_alpha = alpha
    
    return best_alpha, best_val * dx

    n = len(x)

    m = 2 * n - 1
    corrs = compute_corrs(h)

    # Minimize t
    # s.t. G_active @ d - t * 1 <= 0  (for each active k)
    #      sum(d) = 0
    #      -h[i]*step_bound <= d[i] <= (1-h[i])*step_bound

    c[-1] = 1.0

    A_ub[:, :n] = G_act
    A_ub[:, -1] = -1.0
    b_ub = np.zeros(n_act)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)

    bounds.append((None, None))

                  bounds=bounds, method='highs')

            h[i] = 1.0

        if best_swap is None:
            break

        i, j = best_swap
        h[i] = 0.0
        h[j] = 1.0
        corrs = best_new_corrs
        cur_c5 = corrs.max() * dx
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()

    return best_h, best_c5


def binary_sa(h0, dx, rng, t0, budget_s, min_time_left=40.0, max_iters=500000, patience=10000):
    n = len(h0)
    target_sum = int(round(n / 2.0))

    h = np.zeros(n)
    sorted_idx = np.argsort(-h0)
    h[sorted_idx[:target_sum]] = 1.0

    corrs = compute_corrs(h)
    best_c5 = corrs.max() * dx
    best_h = h.copy()

    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]

    no_improve = 0
    temp = 0.03
    cooling = 0.9998

    for it in range(max_iters):
        if time.time() - t0 > budget_s - min_time_left:
            break

        if it % 3000 == 0:
            temp *= cooling

        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]

        old_hi = h[i]
        h[i] = 1.0 - old_hi
        h[j] = 1.0 - h[j]
        new_corrs = compute_corrs(h)
        new_max = new_corrs.max()
        new_c5 = new_max * dx

        if new_c5 <= best_c5 + 1e-14:
            best_c5 = new_c5
            best_h = h.copy()
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            no_improve = 0
        else:
            delta_c5 = new_c5 - best_c5
            if temp > 1e-10 and rng.random() < np.exp(-delta_c5 / temp):
                ones = np.where(h == 1.0)[0]
                zeros = np.where(h == 0.0)[0]
                no_improve = 0
            else:
                h[i] = old_hi
                h[j] = 1.0 - h[j]
                no_improve += 1

        if no_improve > patience:
            break

    return best_h, best_c5


def make_wave_start(n, target_sum, freq=1.0, phase=0.0):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = 0.5 * (1.0 + np.cos(2.0 * np.pi * freq * x + phase))
    return project_feasible(h, target_sum)


def make_golden_start(n, target_sum):
    h = np.zeros(n)
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    for k in range(int(target_sum)):
        pos = int((k * phi) % n)
        h[pos] = 1.0
    return project_feasible(h, target_sum)


def make_mixed_start(n, target_sum, rng):
    h = np.zeros(n)
    n_ones = int(target_sum)
    half_ones = n_ones // 2
    gold_positions = []
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    for k in range(half_ones):
        pos = int((k * phi * 2.0) % n)
        if h[pos] == 0.0:
            h[pos] = 1.0
    remaining = n_ones - int(h.sum())
    available = np.where(h == 0.0)[0]
    if remaining > 0 and len(available) > 0:
        chosen = rng.choice(available, size=min(remaining, len(available)), replace=False)
        h[chosen] = 1.0
    return project_feasible(h, target_sum)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    best_h = np.array(initial_h_values, dtype=float)
    n = len(best_h)
    dx = 2.0 / n
    target_sum = n / 2.0
    best_h = project_feasible(best_h, target_sum)
    best_c5 = c5_of_h(best_h, dx)

    # Phase 1: LP-based optimization from parent
    h_res, c5_res = lp_optimize_phase(best_h, dx, target_sum, t0, budget_s, min_time_left=60.0)
    if c5_res < best_c5:
        best_c5 = c5_res
        best_h = h_res

    # Phase 2: Binary steepest descent
    if time.time() - t0 < budget_s - 80:
        h_bin, c5_bin = binary_steepest(best_h, dx, t0, budget_s, min_time_left=80.0, max_sweeps=100)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin

    # Phase 3: LP from binary result
    if time.time() - t0 < budget_s - 60:
        h_cont = project_feasible(best_h, target_sum)
        h_res2, c5_res2 = lp_optimize_phase(h_cont, dx, target_sum, t0, budget_s, min_time_left=50.0)
        if c5_res2 < best_c5:
            best_c5 = c5_res2
            best_h = h_res2

    # Phase 4: Binary SA
    if time.time() - t0 < budget_s - 100:
        h_sa, c5_sa = binary_sa(best_h, dx, rng, t0, budget_s, min_time_left=100.0,
                                max_iters=300000, patience=8000)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa

    # Phase 5: More LP refinement
    if time.time() - t0 < budget_s - 40:
        h_cont2 = project_feasible(best_h, target_sum)
        h_res3, c5_res3 = lp_optimize_phase(h_cont2, dx, target_sum, t0, budget_s, min_time_left=30.0)
        if c5_res3 < best_c5:
            best_c5 = c5_res3
            best_h = h_res3

    # Phase 6: Different starting points
    starts = []
    starts.append(make_wave_start(n, target_sum, freq=1.0, phase=0.0))
    starts.append(make_wave_start(n, target_sum, freq=2.0, phase=0.5))
    starts.append(make_golden_start(n, target_sum))
    starts.append(make_mixed_start(n, target_sum, rng))

    for s_start in starts:
        if time.time() - t0 > budget_s - 60:
            break
        h_try, c5_try = lp_optimize_phase(s_start, dx, target_sum, t0, budget_s, min_time_left=60.0)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try

    # Phase 7: Final binary polish
    if time.time() - t0 < budget_s - 30:
        h_final_bin, c5_final_bin = binary_steepest(best_h, dx, t0, budget_s, min_time_left=20.0, max_sweeps=50)
        if c5_final_bin < best_c5:
            best_c5 = c5_final_bin
            best_h = h_final_bin

    # Final LP polish
    if time.time() - t0 < budget_s - 20:
        h_final = project_feasible(best_h, target_sum)
        h_fp, c5_fp = lp_optimize_phase(h_final, dx, target_sum, t0, budget_s, min_time_left=15.0)
        if c5_fp < best_c5:
            best_c5 = c5_fp
            best_h = h_fp

    # Final cleanup
    n_b = len(best_h)
    dx_b = 2.0 / n_b
    ts_b = n_b / 2.0
    best_h = project_feasible(best_h, ts_b)
    best_c5 = c5_of_h(best_h, dx_b)

    return best_h, best_c5, n_b
