def project_feasible(x, target_sum):
    """Project x onto {h : 0 <= h_i <= 1, sum(h) = target_sum}"""
    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    # Fix sum
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        # Distribute among non-bound entries
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
        # If still off (all at bounds), adjust one entry
        if abs(h.sum() - target_sum) > 1e-12:
            # Find entry with most room
            if diff > 0:
                room = 1.0 - h
            else:
                room = h
            j = np.argmax(room)
            h[j] += min(diff, room[j]) if diff > 0 else max(diff, -h[j])
    return h

def grad_corr_k(h, k, n):
    """Gradient of correlation k with respect to h."""
    offset = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -offset)
    i_hi = min(n - 1, n - 1 - offset)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx + offset] += (1.0 - h[idx])
    grad[idx] += (-h[idx + offset])
    return grad

def lp_active_set(h, dx, target_sum, alpha, n_active):
    """Active-set LP: minimize predicted max correlation."""
    n = len(h)
    corrs = np.correlate(h, 1.0 - h, mode='full')
    
    # Select active set: top n_active correlations
    top_idx = np.argsort(corrs)[-n_active:]

    # s.t. dx * grad_k . delta - t <= -dx * corr_k  for k in active
    #      sum(delta) = 0
    #      -alpha * h_i <= delta_i <= alpha * (1-h_i)
    #      t >= 0

    c_obj[n] = 1.0

    b_ub = []

        row[n] = -1.0
        A_ub.append(row)
        b_ub.append(-dx * corrs[k])
    
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    A_eq = np.zeros((1, n_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    bounds = []
    for i in range(n):
        lo_b = -alpha * h[i]
        hi_b = alpha * (1.0 - h[i])
        bounds.append((lo_b, hi_b))
    bounds.append((0.0, None))
    
    try:

For a pairwise move: increase h[i] by δ and decrease h[j] by δ (preserving mass). The change in correlation k is:
Δr_k = δ * (∂r_k/∂h[i] - ∂r_k/∂h[j])

We want to minimize the maximum, so we want to find (i, j, δ) that minimizes max_k (r_k + δ * (∂r_k/∂h[i] - ∂r_k/∂h[j])).

This is a one-dimensional problem for each pair (i,j), and we can solve it by finding the δ that minimizes the maximum.

But there are O(n²) pairs, which for n=126 is about 8000 pairs. For each pair, we need to evaluate the maximum over all correlations, which is O(n) work. Total: O(n³) per iteration = about 2M operations. That's fast enough to do many iterations.

Actually, this is a great approach! Let me think about it more carefully.

For a fixed pair (i,j), the objective as a function of δ is:
f(δ) = max_k [r_k + δ * (g_k(i) - g_k(j))]

where g_k(i) = ∂r_k/∂h[i].

This is the maximum of linear functions of δ, which is a convex piecewise-linear function. The minimum can be found by:
1. Computing the slope for each k: s_k = g_k(i) - g_k(j)
2. The minimum of max_k (r_k + δ*s_k) occurs where the two largest "lines" cross (or at a boundary).

Actually, it's simpler: we can use ternary search on δ ∈ [-min(h[i], h[j]), min(1-h[i], 1-h[j])], since f(δ) is convex.

Or even better: the minimum of max of linear functions is at a point where at least two of the lines are equal and are the maximum. We can find this by:
- Computing all the "breakpoints" where the maximum changes
- Checking the objective at each breakpoint and at the boundaries

But the simplest approach is: for each pair (i,j), do a few ternary search steps or just evaluate at a few candidate δ values (like the point where the top-2 correlations would equalize).

Actually, the most efficient approach for a given pair is:
1. Compute s_k = g_k(i) - g_k(j) for all k
2. The problem is: min_δ max_k (r_k + δ*s_k)
3. This is a 1D convex problem. We can solve it exactly.

The optimal δ* satisfies: there exist k1, k2 such that r_{k1} + δ*·s_{k1} = r_{k2} + δ*·s_{k2} = max, and for all other k, r_k + δ*·s_k ≤ max.

In practice, we can just do a few bisection/ternary steps.

        h[i] += best_delta
        h[j] -= best_delta
        h = np.clip(h, 0.0, 1.0)
        h = project_feasible(h, target_sum)

        for k in range(n_corr):
            G[k, :] = grad_corr_k(h, k, n)

def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-13:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def c5_of_h(h, dx):
    return np.correlate(h, 1.0 - h, mode='full').max() * dx

    offset = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -offset)
    i_hi = min(n - 1, n - 1 - offset)

        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx + offset] +=
    (1.0 - h[idx])
    grad[idx] -= h[idx + offset]
    return grad
