from scipy.optimize import minimize, minimize_scalar
from scipy.special import logsumexp
import time

def exact_score(h):

    dx = 2.0 / n
    corrs = np.correlate(h, 1.0 - h, mode='full')
    return corrs.max() * dx

def build_h(q, n):
    m = n // 2
    if n % 2 == 0:
        return np.concatenate([q, 1.0 - q[::-1]])
    return np.concatenate([q, [0.5], 1.0 - q[::-1]])

def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def coord_descent(q, n, n_sweeps=100, grid_size=21, refine=True):
    m = n // 2
    dx = 2.0 / n
    q = q.copy()
    
    for sweep in range(n_sweeps):
        improved = False
        for p in range(m):
            best_val = q[p]
            best_score = exact_score(build_h(q, n))
            
            # Grid search
            candidates = np.linspace(0, 1, grid_size)
            best_cand_score = best_score
            for c in candidates:
                q[p] = c
                s = exact_score(build_h(q, n))
                if s < best_cand_score - 1e-12:
                    best_cand_score = s
                    best_val = c
            q[p] = best_val
            
            # Refine with golden section if needed
            if refine and 0 < best_val < 1:
                # Find bracket
                lo = max(0, best_val - 1.0/grid_size)
                hi = min(1, best_val + 1.0/grid_size)
                if hi > lo:
                    def obj_1d(x):
                        q[p] = x
                        return exact_score(build_h(q, n))
                    res = minimize_scalar(obj_1d, bounds=(lo, hi), method='bounded',
                                        options={'xatol': 1e-6})
                    if res.fun < best_cand_score - 1e-12:
                        q[p] = res.x
                        best_cand_score = res.fun
            
            s = exact_score(build_h(q, n))
            if s < best_score - 1e-12:
                improved = True
                best_score = s
        
        if not improved:
            break
    
    return q, exact_score(build_h(q, n))

def exact_score(h):
    n = len(h)
