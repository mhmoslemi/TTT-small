import numpy as np
from scipy.optimize import linprog
import time

def dx_val(n):
    return 2.0 / n

def corr_full(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * dx_val(n)

def eval_c5(h):
    return float(np.max(corr_full(h)))

def is_feasible(h, tol=1e-9):
    if len(h) < 4:
        return False
    if not np.all(np.isfinite(h)):
        return False
    if np.any(h < -tol) or np.any(h > 1.0 + tol):
        return False
    n = len(h)
    if abs(h.sum() * dx_val(n) - 1.0) > tol * 10:
        return False
    return True

def repair_h(h):
    """Project h to [0,1]^n with sum = n/2 using water filling."""
    n = len(h)
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(200):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h_new = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - h_new.sum()
    if abs(diff) > 1e-13:
        mask = (h_new > 1e-10) & (h_new < 1.0 - 1e-10)
        if mask.any():
            h_new[mask] += diff / mask.sum()
    return np.clip(h_new, 0.0, 1.0)

def build_sym(q, n):
    m = n // 2
    h = np.empty(n)
    h[:m] = q
    h[m:] = 1.0 - q[::-1]
    return h

def to_sym_q(h, n):
    m = n // 2
    return 0.5 * (h[:m] + 1.0 - h[n-1:m-1:-1])

def grad_h_for_lag(h, lag_idx, n):
    """Compute gradient of corr[lag_idx] with respect to h.
    corr_full(h)[j] = sum_i h[i] * (1-h[i-d]) * dx where d = j - (n-1)
    """
    d = lag_idx - (n - 1)
    dx = dx_val(n)
    g = np.zeros(n)
    # Valid range: 0 <= i < n and 0 <= i-d < n
    i_lo = max(0, d)
    i_hi = min(n - 1, n - 1 + d)
    if i_lo <= i_hi:
        # For each i in [i_lo, i_hi]:
        # term = h[i] * (1 - h[i-d])
        # d(term)/d(h[i]) = 1 - h[i-d]
        # d(term)/d(h[i-d]) = -h[i]
        idx = np.arange(i_lo, i_hi + 1)
        g[idx] += (1.0 - h[idx - d]) * dx
        g[idx - d] += (-h[idx]) * dx
    return g

def grad_q_for_lag(h, q, lag_idx, n):
    """Chain rule: grad_q[j] = grad_h[j] - grad_h[n-1-j]"""
    m = n // 2
    gh = grad_h_for_lag(h, lag_idx, n)
    gq = gh[:m] - gh[n-1:m-1:-1]
    return gq

def verify_gradient(h, n, test_lag=0):
    """Verify analytical gradient against numerical gradient."""
    m = n // 2
    q = to_sym_q(h, n)
    eps = 1e-7
    # Numerical gradient of C[lag] wrt q
    num_gq = np.zeros(m)
    for i in range(m):
        q_plus = q.copy()
        q_plus[i] += eps
        h_plus = build_sym(q_plus, n)
        num_gq[i] = (np.correlate(h_plus, 1.0 - h_plus, mode='full')[test_lag] * dx_val(n)
                      - np.correlate(h, 1.0 - h, mode='full')[test_lag] * dx_val(n)) / eps
    # Analytical gradient
    ana_gq = grad_q_for_lag(h, q, test_lag, n)
    max_err = np.max(np.abs(num_gq - ana_gq))
    return max_err

def select_active(h, n, margin_frac=0.02, pad=2):
    """Select active lag indices (those near the maximum)."""
    c = corr_full(h)
    cmax = c.max()
    thresh = cmax - max(1e-8, margin_frac * cmax)
    idx = np.where(c >= thresh)[0]
    # Pad neighbors
    extended = set()
    for a in idx:
        for delta in range(-pad, pad + 1):
            k = int(a) + delta
            if 0 <= k <= 2 * n - 2:
                extended.add(k)
    return np.sort(np.array(list(extended), dtype=int))

def solve_slp_sym(q, n, active_lags, trust, max_iter=50, move_pen=0.0):
    """SLP on the symmetric subspace. Returns improved q and C5."""
    m = n // 2
    h = build_sym(q, n)
    c5 = eval_c5(h)
    c_vals = corr_full(h)
    
    best_q = q.copy()
    best_c5 = c5
    R = trust
    no_progress = 0
    
    for iteration in range(max_iter):
        # Recompute active lags
        c_vals = corr_full(h)
        cmax = c_vals.max()
        active = select_active(h, n, margin_frac=0.015, pad=1)
        
        # Compute gradients for active lags
        grads = np.zeros((len(active), m))
        c_active = c_vals[active]
        for i, lag in enumerate(active):
            grads[i] = grad_q_for_lag(h, q, int(lag), n)
        
        # Build LP:
        # Variables: [δ_0, ..., δ_{m-1}, t]
        # Minimize: t + move_pen * sum(|δ_i|)
        # Constraints:
        #   grads[i] · δ - t <= -c_active[i]  for each i
        #   -min(q, R) <= δ_i <= min(1-q_i, R)  for each i
        #   t >= 0 (or some lower bound)
        
        n_vars = m + 1
        # Objective: minimize t + move_pen * (u_i + v_i) for L1
        # Actually, let's keep it simple: just minimize t, use trust region as constraint
        c_obj = np.zeros(n_vars)
        c_obj[m] = 1.0  # coefficient of t
        
        # Inequality constraints: A_ub @ x <= b_ub
        # For each active lag: grads[i] @ δ - t <= -c_active[i]
        A_ub = np.zeros((len(active), n_vars))
        b_ub = np.zeros(len(active))
        for i in range(len(active)):
            A_ub[i, :m] = grads[i]
            A_ub[i, m] = -1.0
            b_ub[i] = -c_active[i]

        bounds = []
        for i in range(m):
            lo = max(-q[i], -R)
            hi = min(1.0 - q[i], R)
            bounds.append((lo, hi))
        bounds.append((0.0, cmax + 0.1))  # t
        
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        except Exception:
            break
        
        if not res.success or res.x is None:
            break
        
        delta = res.x[:m]

        best_local_c5 = c5
        best_local_q = q.copy()
        for alpha in [1.0, 0.7, 0.5, 0.3, 0.1, 0.05]:

            c5_new = eval_c5(h_new)
            if c5_new < best_local_c5 - 1e-12:
                best_local_c5 = c5_new
                best_local_q = q_new.copy()
        
        if best_local_c5 < best_q and False:  # never true, just for structure
            pass
        
        if best_local_c5 < c5 - 1e-12:

            h = build_sym(q, n)
            c5 = best_local_c5
            best_q = q.copy()
            best_c5 = c5
            R = min(0.2, R * 1.3)
            no_progress = 0
        else:
            R *= 0.6
            no_progress += 1
            if R < 0.001 or no_progress > 15:

    return best_q, best_c5

def interp_to(h, n_new):
    n_old = len(h)
    if n_new == n_old:
        return h.copy()
    x_old = (np.arange(n_old) + 0.5) * dx_val(n_old)
    x_new = (np.arange(n_new) + 0.5) * dx_val(n_new)
    h_new = np.interp(x_new, x_old, h)
    return repair_h(h_new)
