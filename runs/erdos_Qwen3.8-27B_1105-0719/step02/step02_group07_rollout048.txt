                        pass
                    # More carefully:
                    # f_k = sum over valid i of h[i+k]*(1-h[i])
                    # d/dh[p] f_k: h[p] appears as h[i+k] when p=i+k (i=p-k), or as h[i] when p=i
                    # Term 1: h[p] = h[i+k] where i = p-k, valid if i_lo <= p-k <= i_hi and 0<=p<k+n
                    #   contribution: (1-h[p-k])
                    # Term 2: h[p] = h[i] where i = p, valid if i_lo <= p <= i_hi
                    #   contribution: -h[p+k]
                    if i_lo <= i - k <= i_hi and 0 <= i < n:
                        gi += (1.0 - h[i - k])
                    if i_lo <= i <= i_hi and 0 <= i + k < n:
                        gi += (-h[i + k])
                    gi *= dx
                    if i_lo <= j - k <= i_hi and 0 <= j < n:
                        gj += (1.0 - h[j - k])
                    if i_lo <= j <= i_hi and 0 <= j + k < n:
                        gj += (-h[j + k])
                    gj *= dx
                gdiff[k] = gi - gj
            # Find best delta for this pair
            # F(delta) = max_k (corrs[k] + delta * gdiff[k])
            # delta range: transfer from i to j: delta in [-(h[j]), min(h[i], 1-h[j])]
            #              transfer from j to i: delta in [-(h[i]), min(h[j], 1-h[i])]
            d_lo = -h[j]
            d_hi = min(h[i], 1.0 - h[j])
            if d_hi <= d_lo:
                continue
            # Golden section search for min of max of linear functions
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            a, b = d_lo, d_hi
            c_ = b - gr * (b - a)
            d_ = a + gr * (b - a)
            fc = np.max(corrs + c_ * gdiff)
            fd = np.max(corrs + d_ * gdiff)
            for _ in range(25):
                if fc < fd:
                    b = d_
                    d_ = c_
                    fd = fc
                    c_ = b - gr * (b - a)
                    fc = np.max(corrs + c_ * gdiff)
                else:
                    a = c_
                    c_ = d_
                    fc = fd
                    d_ = a + gr * (b - a)
                    fd = np.max(corrs + d_ * gdiff)
            delta_best = (a + b) / 2.0
            if abs(delta_best) < 1e-6:
                continue
            # Verify with exact score
            h_new = h.copy()
            h_new[i] += delta_best
            h_new[j] -= delta_best
            if h_new[i] < -1e-10 or h_new[i] > 1 + 1e-10 or h_new[j] < -1e-10 or h_new[j] > 1 + 1e-10:
                continue
            h_new = np.clip(h_new, 0.0, 1.0)
            h_new = project_feasible(h_new, n / 2.0)
            c_new = exact_score(h_new)
            if c_new < best_c - 1e-12:
                h = h_new
                best_c = c_new
                corrs = all_corrs(h)
                improved = True
    return h, best_c


def subgradient_with_restarts(h0, n, n_restarts=4, iters_per_restart=2000, top_k_list=None):
    if top_k_list is None:
        top_k_list = [1, 2, 3, 5, 10, 15, 20]
    h = h0.copy()
    best_c = exact_score(h0)
    best_h = h0.copy()
    rng = np.random.RandomState(123)
    for r in range(n_restarts):
        for tk in top_k_list:
            h_new, c_new = subgradient_descent(h, n, max_iters=iters_per_restart // len(top_k_list),
                                               top_k=tk, alpha0=0.01)
            if c_new < best_c - 1e-10:
                best_c = c_new
                best_h = h_new
                h = h_new
        # Perturb for next restart
        h_pert = h + 0.05 * rng.randn(n)
        h = project_feasible(h_pert, n / 2.0)
        c = exact_score(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    return best_h, best_c


def coordinate_descent_polish(h0, n, max_iters=500):
    h = h0.copy()
    best_c = exact_score(h)
    target = n / 2.0
    for it in range(max_iters):
        improved = False
        # Try adjusting pairs
        for i in range(n):
            for j in range(i + 1, min(i + 10, n)):
                # Try small transfers
                for delta in [0.005, 0.01, 0.02, 0.05, -0.005, -0.01, -0.02, -0.05]:
                    hi_new = h[i] + delta
                    hj_new = h[j] - delta
                    if 0 <= hi_new <= 1 and 0 <= hj_new <= 1:
                        h_old_i, h_old_j = h[i], h[j]
                        h[i], h[j] = hi_new, hj_new
                        c = exact_score(h)
                        if c < best_c - 1e-12:
                            best_c = c
                            improved = True
                        else:
                            h[i], h[j] = h_old_i, h_old_j
        if not improved:
            break
    return h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def remaining():
        return budget_s - (time.time() - t0)

    # Get initial
    try:
        h_parent = np.array(initial_h_values, dtype=float)
    except NameError:
        h_parent = 0.5 * np.ones(126)

    n = len(h_parent)
    if n % 2 != 0:
        h_parent = h_parent[:n - 1]
        n = len(h_parent)

    h_best = project_feasible(h_parent, n / 2.0)
    c5_best = exact_score(h_best)
    print(f"Initial: n={n}, C5={c5_best:.7f}")

    # Phase 1: Subgradient descent
    if remaining() > 100:
        h_new, c_new = subgradient_descent(h_best, n, max_iters=3000, top_k=3, alpha0=0.02)
        if c_new < c5_best:
            c5_best = c_new
            h_best = h_new
            print(f"  SG top3: C5={c5_best:.7f}")

        h_new, c_new = subgradient_descent(h_best, n, max_iters=3000, top_k=5, alpha0=0.015)
        if c_new < c5_best:
            c5_best = c_new
            h_best = h_new
            print(f"  SG top5: C5={c5_best:.7f}")

        h_new, c_new = subgradient_descent(h_best, n, max_iters=2000, top_k=10, alpha0=0.01)
        if c_new < c5_best:
            c5_best = c_new
            h_best = h_new
            print(f"  SG top10: C5={c5_best:.7f}")

    # Phase 2: Pair move local search
    if remaining() > 100:
        h_new, c_new = pair_move_search(h_best, n, max_sweeps=10, n_candidates=50)
        if c_new < c5_best:
            c5_best = c_new
            h_best = h_new
            print(f"  Pair search: C5={c5_best:.7f}")

    # Phase 3: More subgradient with restarts
    if remaining() > 150:
        h_new, c_new = subgradient_with_restarts(h_best, n, n_restarts=3,
                                                  iters_per_restart=1500,
                                                  top_k_list=[1, 3, 5, 10])
        if c_new < c5_best:
            c5_best = c_new
            h_best = h_new
            print(f"  SG restarts: C5={c5_best:.7f}")

    # Phase 4: Pair move again
    if remaining() > 80:
        h_new, c_new = pair_move_search(h_best, n, max_sweeps=15, n_candidates=50)
        if c_new < c5_best:
            c5_best = c_new
            h_best = h_new
            print(f"  Pair search 2: C5={c5_best:.7f}")

    # Phase 5: Try different n values
    if remaining() > 100:
        for n_try in [n - 2, n + 2, n + 4, n - 4]:
            if n_try < 40 or n_try % 2 != 0 or remaining() < 30:
                continue
            x_old = np.linspace(0, 2.0, n, endpoint=False)
            x_new = np.linspace(0, 2.0, n_try, endpoint=False)
            h_interp = np.interp(x_new, x_old, h_best)
            h_interp = project_feasible(h_interp, n_try / 2.0)
            h_new, c_new = subgradient_descent(h_interp, n_try, max_iters=1500, top_k=5, alpha0=0.01)
            if c_new < c5_best - 1e-8:
                h_new, c_new = pair_move_search(h_new, n_try, max_sweeps=5)
                if c_new < c5_best - 1e-8:
                    c5_best = c_new
                    h_best = h_new
                    n = n_try
                    print(f"  n={n}: C5={c5_best:.7f}")

    # Phase 6: Final subgradient polish
    if remaining() > 40:
        h_new, c_new = subgradient_descent(h_best, n, max_iters=2000, top_k=3, alpha0=0.005)
        if c_new < c5_best:
            c5_best = c_new
            h_best = h_new
            print(f"  Final SG: C5={c5_best:.7f}")

    # Phase 7: Final pair search
    if remaining() > 30:
        h_new, c_new = pair_move_search(h_best, n, max_sweeps=10)
        if c_new < c5_best:
            c5_best = c_new
            h_best = h_new
            print(f"  Final pairs: C5={c5_best:.7f}")

    # Final cleanup
    h_best = project_feasible(h_best, len(h_best) / 2.0)
    c5_best = exact_score(h_best)
    print(f"Final: C5={c5_best:.7f}, n={len(h_best)}")
    return h_best, c5_best, len(h_best)
