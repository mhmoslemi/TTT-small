```python
def compute_effect_matrix(h):
    n = len(h)
    e = np.arange(-(n-1), n)  # offsets, length 2n-1
    # For each i and offset e, compute (1-h[i+e]) if valid, minus h[i-e] if valid
    E = np.zeros((n, len(e)))
    for i in range(n):
        # i+e valid: 0 <= i+e < n
        valid_plus = (i + e >= 0) & (i + e < n)
        idx_plus = (i + e)[valid_plus]
        E[i, valid_plus] += (1 - h[idx_plus])
        # i-e valid: 0 <= i-e < n
        valid_minus = (i - e >= 0) & (i - e < n)
        idx_minus = (i - e)[valid_minus]
        E[i, valid_minus] -= h[idx_minus]
    return E
```

```python
def compute_effect_matrix(h):
    n = len(h)
    e = np.arange(-(n-1), n)  # shape (2n-1,)
    # i ranges over 0..n-1, e ranges over -(n-1)..n-1
    i = np.arange(n)[:, None]  # shape (n, 1)
    e_row = e[None, :]  # shape (1, 2n-1)
    
    # i + e: shape (n, 2n-1)
    ie = i + e_row
    valid_plus = (ie >= 0) & (ie < n)
    # Safe index
    ie_clipped = np.clip(ie, 0, n-1)
    E_plus = np.where(valid_plus, 1 - h[ie_clipped], 0)

    im = i - e_row
    valid_minus = (im >= 0) & (im < n)
    im_clipped = np.clip(im, 0, n-1)
    E_minus = np.where(valid_minus, h[im_clipped], 0)
```

```python
        f_hi = np.max(C[None, None, :] + hi[:, :, None] * D, axis=2)
```

```python
        new_c5 = C.max() * dx
```

g = (1/|A|) * sum_{k∈A} E[:, k]

This is a vector of length n. Then:
h_new = project_feasible(h - η * g, n/2)

where η is the step size.

This is a "smoothed" version of the subgradient method, using the top ε-fraction of correlations instead of just the maximum. It should converge faster than the pure subgradient method and is simpler than the pairwise optimization.

```python
def smooth_subgradient_optimize(h, max_iters=10000, epsilon=0.01, eta0=0.01, deadline=None):
    n = len(h)
    dx = 2.0 / n

    best_c5 = c5_score(h)
    
    for it in range(max_iters):
        if deadline and time.time() > deadline:
            break
        
        C = np.correlate(h, 1.0 - h, mode='full')
        C_max = C.max()
        active = C >= C_max - epsilon
        n_active = active.sum()
        
        if n_active == 0:
            break
        
        E = compute_E(h)
        g = E[:, active].mean(axis=1)

            h = h_new
        else:
            h = h_new  # Still take the step (for exploration)
            # Or: don't take the step (pure descent)
```

```python
def pairwise_anneal(h, max_iters=50000, T0=0.01, cooling=0.999, deadline=None):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    C = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = C.max() * dx
    best_h = h.copy()
    T = T0
    
    rng = np.random.RandomState(0)
    
    for it in range(max_iters):
        if deadline and time.time() > deadline:
            break
        if it % 1000 == 0 and it > 0:
            T *= cooling
        
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        
        E = compute_E(h)
        D_ij = E[i] - E[j]
        
        delta_lo = max(-h[i], h[j] - 1)
        delta_hi = min(1 - h[i], h[j])
        if delta_lo >= delta_hi:
            continue
        
        # Random delta in feasible range
        delta = delta_lo + rng.random() * (delta_hi - delta_lo)
        
        new_C = C + delta * D_ij
        new_max = new_C.max() * dx
        delta_f = new_max - best_c5
        
        if delta_f < 0 or rng.random() < np.exp(-delta_f / max(T, 1e-10)):
            h[i] += delta
            h[j] -= delta
            h[i] = np.clip(h[i], 0, 1)
            h[j] = np.clip(h[j], 0, 1)
            C = new_C
            new_c5 = C.max() * dx
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
    
    return best_h, best_c5
```

This is similar to the simulated annealing in the previous code, but for continuous variables and with the exact objective. It's slower than the deterministic pairwise optimization (random pair selection vs. best pair), but it can escape local optima.
