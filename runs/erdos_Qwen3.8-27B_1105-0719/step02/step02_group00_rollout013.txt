import numpy as np
from scipy.optimize import minimize
import time

def _repair(h):
    """Repair h to satisfy 0<=h<=1 and sum(h)=n/2."""
    n = len(h)
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        # Need to add mass: add to positions with most room
        order = np.argsort(-(1.0 - h))
        for i in order:
            if diff <= 0:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        # Need to remove mass: remove from positions with most value
        order = np.argsort(h)
        for i in order:
            if diff >= 0:
                break
            rem = min(-diff, h[i])
            h[i] -= rem
            diff += rem
    # Final adjustment
    residual = target - h.sum()
    if abs(residual) > 1e-15:
        # Find an index where we can adjust
        if residual > 0:
            idx = np.argmax(1.0 - h)
        else:
            idx = np.argmin(h)
        h[idx] += residual
        h = np.clip(h, 0.0, 1.0)
    return h

def _score(h):

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h):
    n = len(h)
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(-(1.0 - h))
        for i in order:
            if diff <= 0: break
            add = min(diff, 1.0 - h[i] + 1e-15)
            if add > 0:
                h[i] += add
                diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= 0: break
            rem = min(-diff, h[i] + 1e-15)
            if rem > 0:
                h[i] -= rem
                diff += rem
    residual = target - h.sum()
    if abs(residual) > 1e-14:
        if residual > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmin(h))
        h[idx] += residual
        h = np.clip(h, 0.0, 1.0)
    return h

def _score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def _profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def _grad_rows(h, rows):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((len(rows), n))
    for r, m in enumerate(rows):
        d = m - (n - 1)
        if d == 0:
            G[r] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            G[r, :n-d] += dx * (1.0 - h[d:n])
            G[r, d:] += -dx * h[:n-d]
        else:
            dp = -d
            G[r, dp:] += dx * (1.0 - h[:n-dp])
            G[r, :n-dp] += -dx * h[dp:]
    return G

def _interp(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = np.arange(n + 1) * (2.0 / n)
    xt = np.arange(new_n + 1) * (2.0 / new_n)
    hc = np.concatenate([[0.0], h, [0.0]])
    c_old = np.cumsum(hc) * (2.0 / n)
    c_new = np.interp(xt, xs, c_old)
    h_new = np.diff(c_new) / (2.0 / new_n)
    return _repair(h_new)

def _binary_from_continuous(h, rng):
    n = len(h)
    half = n // 2
    idx = np.argsort(-h)[:half]
    bits = np.zeros(n, dtype=np.float64)
    bits[idx] = 1.0
    return bits

def _balanced_random_bits(n, rng):

    bits = np.zeros(n, dtype=np.float64)
    idx = rng.choice(n, size=half, replace=False)

    return bits

def _binary_score(bits):

    prof = np.correlate(bits, 1.0 - bits, mode="full") * dx

def _anneal_binary(bits, max_iters, rng, temp0=0.5, temp1=0.01):
    n = len(bits)
    half = n // 2
    best_bits = bits.copy()

def _active_descent(h, max_iters=300, top_k=8, step_decay=0.9, alpha0=0.05):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best_c = _score(h)
    alpha = alpha0
    stall = 0
    
    for it in range(max_iters):
        prof = _profile(h)
        c = float(np.max(prof))
        if c < best_c:
            best_c = c
            stall = 0
        else:
            stall += 1
        
        if stall > 50 and alpha < 1e-6:
            break
        
        # Active set: top lags
        threshold = c - max(0.003, 3 * (c - np.median(prof)))
        active = np.where(prof >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(prof))])
        if len(active) > top_k:
            top_idx = np.argsort(prof[active])[-top_k:]
            active = active[top_idx]
        
        G = _grad_rows(h, active)
        weights = prof[active] - threshold
        weights = np.maximum(weights, 0.01)
        weights /= weights.sum()
        grad = G.T @ weights
        grad -= grad.mean()  # remove null direction (sum constraint)
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad /= gnorm
        
        improved = False
        for a in [alpha, alpha*0.5, alpha*0.25, alpha*0.1]:
            h_try = _repair(h - a * grad)
            s = _score(h_try)
            if s < best_c - 1e-13:
                h = h_try
                best_c = s
                improved = True
                break
            elif s < c:
                h = h_try
                improved = True
                break
        
        if not improved:
            alpha *= step_decay
            if alpha < 1e-7:
                break
        
    return h, best_c

def _slsqp_solve(h0, maxiter=200):
    n = len(h0)
    dx = 2.0 / n
    M = 2 * n - 1
    c0 = _score(h0)
    t0 = c0 + 0.001
    
    def obj(x):
        return x[-1]
    
    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    
    def eq_fun(x):
        return x[:-1].sum() - n / 2.0
    
    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g
    
    def ineq_fun(x):
        h = x[:-1]
        t = x[-1]
        prof = np.correlate(h, 1 - h, mode="full") * dx
        return t - prof
    
    def ineq_jac(x):
        h = x[:-1]
        G = _grad_rows(h, np.arange(M))
        J = np.zeros((M, n + 1))
        J[:, :-1] = -G
        J[:, -1] = 1.0
        return J
    
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 1.0)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None and np.isfinite(res.x).all():
            h_new = _repair(res.x[:-1])
            c_new = _score(h_new)
            if c_new <= c0 + 1e-6:
                return h_new
    except Exception:
        pass
    return None

        pass
    
    return best_h, final_c, len(best_h)
