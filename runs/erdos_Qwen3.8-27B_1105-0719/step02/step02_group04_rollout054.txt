The SLP approach avoids this by directly solving the constraint satisfaction problem. The LP says: "find the maximum reduction in t that is consistent with all active constraints being satisfied in the linearized model." This is a cleaner problem.

- If the active set is too large, the LP might be infeasible (no direction can reduce all constraints simultaneously)
- If the active set is too small, the step might activate new constraints

The solution is to adaptively manage the active set:
- Start with a small active set (just the max)
- If the step fails (exact C5 doesn't improve), add more constraints
- If the LP is infeasible, remove some constraints (the least active ones)

This is similar to the active-set method in constrained optimization.

            if no_progress > 10: break
            continue

        best_c5_local = c5

        else:
            hi = mid
    out = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - out.sum()
    if abs(diff) > 1e-12:
        mask = (out > 1e-12) & (out < 1.0 - 1e-12)
        if mask.any():
            out[mask] += diff / mask.sum()
    return np.clip(out, 0.0, 1.0)

def interp_feasible(h_old, n_new):
    n_old = len(h_old)
    if n_new == n_old:
        return h_old.copy()
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_to_target(h_new, n_new / 2.0)

def slp_sym(q, n, max_iter=200, time_limit=200, t0=None):
    m = n // 2
    h = build_sym(q, n)
    c5 = eval_c5(h)
    c = corr_scaled(h)
    dx = 2.0 / n
    margin = 0.01
    trust = 0.06
    no_prog = 0
    best_q = q.copy()
    best_c5 = c5

    for it in range(max_iter):
        if t0 is not None and (time.time() - t0) > time_limit:
            break
        c = corr_scaled(h)
        c5 = float(c.max())
        if c5 < best_c5:
            best_c5 = c5
            best_q = q.copy()
        active = select_active(c, margin, pad=2)
        G = np.zeros((len(active), m))
        for idx, j in enumerate(active):
            G[idx] = grad_lag_q(h, int(j), n) * dx
        delta, t_new = solve_lp_sym(G, c[active], q, trust, m)
        if delta is None:
            margin = max(0.002, margin * 0.5)
            no_prog += 1
            if no_prog > 12:
                margin = min(0.05, margin * 2.0)
                trust = max(0.02, trust * 0.7)
                no_prog = 0
            continue
        best_alpha = 0.0
        best_c5_ls = c5
        for alpha in [1.0, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05]:
            q_try = np.clip(q + alpha * delta, 0.0, 1.0)
            h_try = build_sym(q_try, n)
            cv = eval_c5(h_try)
            if cv < best_c5_ls - 1e-14:
                best_c5_ls = cv
                best_alpha = alpha
        if best_alpha == 0.0:
            no_prog += 1
            margin = min(0.04, margin * 1.3)
            trust = max(0.01, trust * 0.75)
            if no_prog > 15:
                break
        else:
            q = np.clip(q + best_alpha * delta, 0.0, 1.0)
            h = build_sym(q, n)
            no_prog = 0
            if best_alpha >= 0.8:
                trust = min(0.15, trust * 1.2)
            margin = max(0.004, 0.015 * c5)
    return best_q, best_c5

def slp_full(h, n, max_iter=150, time_limit=150, t0=None):
    h = project_to_target(h.copy(), n / 2.0)
    c5 = eval_c5(h)
    c = corr_scaled(h)
    dx = 2.0 / n
    margin = 0.01
    trust = 0.05
    no_prog = 0
    best_h = h.copy()
    best_c5 = c5

    for it in range(max_iter):
        if t0 is not None and (time.time() - t0) > time_limit:
            break
        c = corr_scaled(h)
        c5 = float(c.max())
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        active = select_active(c, margin, pad=2)
        G = np.zeros((len(active), n))
        for idx, j in enumerate(active):
            G[idx] = grad_lag_full(h, int(j)) * dx
        delta, t_new = solve_lp_full(G, c[active], h, trust, n)
        if delta is None:
            margin = max(0.002, margin * 0.5)
            no_prog += 1
            if no_prog > 12:
                margin = min(0.05, margin * 2.0)
                trust = max(0.01, trust * 0.7)
                no_prog = 0
            continue
        best_alpha = 0.0
        best_c5_ls = c5
        for alpha in [1.0, 0.7, 0.5, 0.3, 0.2, 0.1]:
            h_try = h + alpha * delta
            h_try = project_to_target(h_try, n / 2.0)
            cv = eval_c5(h_try)
            if cv < best_c5_ls - 1e-14:
                best_c5_ls = cv

        if best_alpha == 0.0:

            margin = min(0.04, margin * 1.3)
            trust = max(0.01, trust * 0.75)
            if no_prog > 15:
                break
        else:
            h = project_to_target(h + best_alpha * delta, n / 2.0)
            no_prog = 0
            if best_alpha >= 0.8:
                trust = min(0.12, trust * 1.2)
            margin = max(0.004, 0.015 * c5)
    return best_h, best_c5

def make_random_q(n, rng):
    m = n // 2
    t = np.linspace(0.0, 1.0, m)
    k1 = rng.randint(1, 6)
    k2 = rng.randint(2, 9)
    p1 = rng.uniform(0, 2 * np.pi)
    p2 = rng.uniform(0, 2 * np.pi)
    a1 = rng.uniform(0.1, 0.3)
    a2 = rng.uniform(0.05, 0.2)
    q = 0.5 + a1 * np.sin(2 * np.pi * k1 * t + p1) + a2 * np.cos(2 * np.pi * k2 * t + p2)
    return np.clip(q, 0.0, 1.0)

def remaining_time(t0, budget_s):
    return max(0.0, budget_s - (time.time() - t0))

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')

    # Try to use initial_h_values
    try:
        h_init = np.array(initial_h_values, dtype=float).reshape(-1)
        if h_init.size > 4 and np.all(np.isfinite(h_init)):
            n_init = len(h_init)
            h_init = project_to_target(h_init, n_init / 2.0)
            if np.all(h_init >= -1e-10) and np.all(h_init <= 1.0 + 1e-10):
                h_init = np.clip(h_init, 0.0, 1.0)
                h_init = project_to_target(h_init, n_init / 2.0)
                best_h = h_init.copy()
                best_c5 = eval_c5(h_init)
    except Exception:
        pass

    if best_h is None:
        n_def = 126
        m_def = n_def // 2
        t = np.linspace(0.0, 1.0, m_def)
        q_def = np.clip(0.5 + 0.25 * np.sin(2.0 * np.pi * 3.0 * t), 0.0, 1.0)
        best_h = build_sym(q_def, n_def)
        best_c5 = eval_c5(best_h)

    reserve = max(30.0, 0.05 * budget_s)

    # Phase 1: SLP on parent n (symmetric)
    n_base = len(best_h)
    if n_base % 2 != 0:
        n_base -= 1
    q_base = to_sym_q(best_h, n_base + (0 if len(best_h) % 2 == 0 else 1))
    if len(best_h) % 2 == 0:
        n_work = len(best_h)
    else:
        n_work = len(best_h) - 1
    q_base = to_sym_q(best_h[:n_work], n_work)
    h_sym = build_sym(q_base, n_work)
    c5_sym = eval_c5(h_sym)
    if c5_sym < best_c5:
        best_h = h_sym.copy()
        best_c5 = c5_sym

    # SLP symmetric on parent
    time_left = remaining_time(t0, budget_s)
    t1 = t0
    q_opt, c5_opt = slp_sym(q_base, n_work, max_iter=300, time_limit=min(350, 0.4 * budget_s), t0=t0)
    h_new = build_sym(q_opt, n_work)
    if c5_opt < best_c5:
        best_h = h_new.copy()
        best_c5 = c5_opt

    # Second pass with wider margin
    if remaining_time(t0, budget_s) > reserve + 60:
        q_opt2, c5_opt2 = slp_sym(q_opt, n_work, max_iter=200, time_limit=min(200, 0.2 * budget_s), t0=time.time())
        h_new2 = build_sym(q_opt2, n_work)
        if c5_opt2 < best_c5:
            best_h = h_new2.copy()
            best_c5 = c5_opt2

    # Phase 2: Try nearby n values with SLP
    n_candidates = set()
    n_base_cur = len(best_h)
    for delta in [-8, -6, -4, -3, -2, -1, 1, 2, 3, 4, 6, 8, 12]:
        nc = n_base_cur + delta
        if 40 <= nc <= 300 and nc % 2 == 0:
            n_candidates.add(nc)
    n_candidates = sorted(n_candidates)

    for n_try in n_candidates:
        if remaining_time(t0, budget_s) < reserve + 30:
            break
        h_try = interp_feasible(best_h, n_try)
        q_try = to_sym_q(h_try, n_try)
        h_try = build_sym(q_try, n_try)
        c5_try_init = eval_c5(h_try)
        q_try, c5_try = slp_sym(q_try, n_try, max_iter=150, time_limit=60, t0=time.time())
        h_try = build_sym(q_try, n_try)
        if c5_try < best_c5:
            best_h = h_try.copy()
            best_c5 = c5_try

    # Phase 3: Full-variable SLP (break symmetry)
    if remaining_time(t0, budget_s) > reserve + 60:
        n_fin = len(best_h)
        h_full = best_h.copy()
        h_full, c5_full = slp_full(h_full, n_fin, max_iter=100, time_limit=min(150, 0.15 * budget_s), t0=time.time())
        if c5_full < best_c5:
            best_h = h_full.copy()
            best_c5 = c5_full

    # Phase 4: More n values with full SLP
    if remaining_time(t0, budget_s) > reserve + 40:
        n_base2 = len(best_h)
        for delta in [-4, -2, 2, 4]:
            if remaining_time(t0, budget_s) < reserve + 25:
                break
            nc = n_base2 + delta
            if nc < 40 or nc > 300 or nc % 2 != 0:
                continue
            h_int = interp_feasible(best_h, nc)
            h_f, c5_f = slp_full(h_int, nc, max_iter=80, time_limit=50, t0=time.time())
            if c5_f < best_c5:
                best_h = h_f.copy()
                best_c5 = c5_f

    # Phase 5: Random restarts
    max_restarts = min(20, max(3, int(remaining_time(t0, budget_s) / 20.0)))
    for trial in range(max_restarts):
        if remaining_time(t0, budget_s) < reserve + 15:
            break
        n_r = len(best_h)
        if n_r % 2 != 0:
            n_r -= 1
        q_r = make_random_q(n_r, rng)
        h_r = build_sym(q_r, n_r)
        q_r, c5_r = slp_sym(q_r, n_r, max_iter=80, time_limit=30, t0=time.time())
        h_r = build_sym(q_r, n_r)
        if c5_r < best_c5:
            best_h = h_r.copy()
            best_c5 = c5_r

    # Phase 6: Final polish with tight tolerance
    if remaining_time(t0, budget_s) > 15:
        n_fin = len(best_h)
        if n_fin % 2 == 0:
            q_fin = to_sym_q(best_h, n_fin)
            q_fin, c5_fin = slp_sym(q_fin, n_fin, max_iter=200, time_limit=60, t0=time.time())
            h_fin = build_sym(q_fin, n_fin)
            if c5_fin < best_c5:
                best_h = h_fin.copy()
                best_c5 = c5_fin
        if remaining_time(t0, budget_s) > 10:
            h_fin2, c5_fin2 = slp_full(best_h, len(best_h), max_iter=80, time_limit=40, t0=time.time())
            if c5_fin2 < best_c5:
                best_h = h_fin2.copy()
                best_c5 = c5_fin2

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_target(best_h, len(best_h) / 2.0)
    best_c5 = eval_c5(best_h)

    return best_h, best_c5, len(best_h)
