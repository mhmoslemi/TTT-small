def project_feasible(x, target_sum):
    """Project x onto {0<=h<=1, sum(h)=target_sum}"""
    n = len(x)

        h = np.clip(x - mid, 0.0, 1.0)

            lo = mid
        else:
            hi = mid

import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time


def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-13:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = mask.sum()
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h


def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def c5_exact(h):
    n = len(h)
    return float(compute_corrs(h).max() * (2.0 / n))


def grad_corr_k(h, k, n):
    grad = np.zeros(n)
    d = k - (n - 1)
    if d == 0:
        grad[:] = 1.0 - 2.0 * h
    elif d > 0:
        i_hi = n - 1 - d
        if i_hi >= 0:
            idx = np.arange(0, i_hi + 1)
            grad[idx + d] += (1.0 - h[idx])
            grad[idx] += (-h[idx + d])
    else:
        i_lo = -d
        idx = np.arange(i_lo, n)
        grad[idx + d] += (1.0 - h[idx])
        grad[idx] += (-h[idx + d])
    return grad


# LSE-based SLSQP
def lse_obj(x, dx, beta):
    c = compute_corrs(x) * dx
    return float(logsumexp(beta * c) / beta)


def lse_obj_grad(x, dx, beta):
    n = len(x)
    c = compute_corrs(x) * dx
    shifted = c - c.max()
    w = np.exp(beta * shifted)
    ws = w.sum()
    if ws < 1e-30:
        w = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        w /= ws
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] > 1e-14:
            g += w[k] * grad_corr_k(x, k, n)
    return g


def slsqp_lse_solve(h0, dx, target_sum, beta, maxiter):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n

    def _obj(x):
        return lse_obj(x, dx, beta)

    def _grad(x):
        return lse_obj_grad(x, dx, beta)

    def _cons_fun(x):
        return x.sum() - target_sum

    def _cons_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    try:
        res = minimize(_obj, h0, jac=_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        h_out = project_feasible(res.x, target_sum)
    except Exception:
        h_out = project_feasible(h0, target_sum)
    return h_out, c5_exact(h_out)

def focused_slsqp_solve(h0, dx, target_sum, top_k, maxiter, anchor_w):
    """Minimize s s.t. corr_k(h) <= s for top-K lags, sum(h)=target, 0<=h<=1.
    Objective: s + anchor_w * ||h - h0||^2 / n
    """
    n = len(h0)
    corrs = compute_corrs(h0)
    raw_max = corrs.max()
    active = np.argsort(corrs)[-top_k:]

    x0 = np.concatenate([h0, [raw_max + 1e-6]])
    bounds = [(0.0, 1.0)] * n + [(0.0, raw_max + 10.0)]

    def _obj(x):
        h = x[:n]
        s = x[-1]
        return s + anchor_w * np.sum((h - h0) ** 2) / n

    def _grad(x):
        h = x[:n]
        g = np.zeros(n + 1)
        g[:n] = 2.0 * anchor_w * (h - h0) / n
        g[-1] = 1.0
        return g

    def _ineq(x):
        h = x[:n]
        s = x[-1]
        c = compute_corrs(h)
        return s - c[active]  # >= 0

    def _ineq_jac(x):
        h = x[:n]
        jac = np.zeros((top_k, n + 1))
        for i, k in enumerate(active):
            gk = grad_corr_k(h, k, n)
            jac[i, :n] = -gk
            jac[i, -1] = 1.0
        return jac

    def _eq_fun(x):
        return x[:n].sum() - target_sum

    def _eq_jac(x):
        j = np.zeros(n + 1)
        j[:n] = 1.0
        return j

    cons = [
        {'type': 'ineq', 'fun': _ineq, 'jac': _ineq_jac},
        {'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac},
    ]

    try:
        res = minimize(_obj, x0, jac=_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-13})
        h_out = project_feasible(res.x[:n], target_sum)
    except Exception:
        h_out = project_feasible(h0, target_sum)
    return h_out, c5_exact(h_out)


# Anti-palindromic optimizer
def build_antipal(v, n):
    """Build full anti-palindromic vector from first half v. n must be even."""
    h = np.empty(n)
    m = n // 2
    h[:m] = v
    h[m:] = 1.0 - v[::-1]
    return h


def slsqp_antipal_solve(h_full, dx, target_sum, beta, maxiter):
    """Optimize within anti-palindromic subspace."""
    n = len(h_full)
    m = n // 2
    v0 = h_full[:m].copy()

    bounds = [(0.0, 1.0)] * m

    def _obj(v):
        h = build_antipal(v, n)
        return lse_obj(h, dx, beta)

    def _grad(v):
        h = build_antipal(v, n)
        full_grad = lse_obj_grad(h, dx, beta)
        # Need gradient of objective w.r.t. v
        # h[i] = v[i] for i < m
        # h[i] = 1 - v[n-1-i] for i >= m
        # d(obj)/dv[i] = d(obj)/dh[i] + d(obj)/dh[n-1-i] * (-1)
        g = np.zeros(m)
        for i in range(m):
            g[i] = full_grad[i] - full_grad[n - 1 - i]
        return g

    # No equality constraint needed (sum is automatically n/2)
    cons = []

    try:
        res = minimize(_obj, v0, jac=_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        v_out = np.clip(res.x, 0.0, 1.0)
        h_out = build_antipal(v_out, n)
    except Exception:
        h_out = h_full.copy()
    return h_out, c5_exact(h_out)


def interpolate_to_dim(h_old, n_new):
    """Interpolate h_old to n_new points."""
    n_old = len(h_old)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return h_new
