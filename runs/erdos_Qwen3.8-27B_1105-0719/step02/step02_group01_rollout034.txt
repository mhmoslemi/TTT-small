from scipy.optimize import minimize
from scipy.special import logsumexp
import time

def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def c5_of_h(h, dx):
    return compute_corrs(h).max() * dx

def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def grad_corr_k(h, k, n):
    offset = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -offset)
    i_hi = min(n - 1, n - 1 - offset)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx + offset] += (1.0 - h[idx])
    grad[idx] += (-h[idx + offset])
    return grad

def active_gradient(h, dx, n_active, beta_w):
    """Compute weighted gradient of active correlations."""
    n = len(h)
    corrs = compute_corrs(h)
    max_c = corrs.max()
    # Weight: softmax over active correlations
    shifted = corrs - max_c
    w = np.exp(beta_w * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(len(corrs)) / len(corrs)
    else:
        w /= w_sum
    grad = np.zeros(n)
    for k in range(len(corrs)):
        if w[k] > 1e-12:
            grad += w[k] * grad_corr_k(h, k, n)
    return grad * dx, corrs, max_c

def slsqp_logsumexp(h0, dx, target_sum, beta, maxiter):
    n = len(h0)
    
    def obj(x):
        c = compute_corrs(x)
        return logsumexp(beta * c) / beta * dx
    
    def grad_obj(x):
        g, _, _ = active_gradient(x, dx, 2*n-1, beta)
        return g
    
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0, jac=grad_obj, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        h_out = project_feasible(res.x, target_sum)
    except Exception:
        h_out = project_feasible(h0, target_sum)

def slsqp_epigraph(h0, dx, target_sum, n_active, maxiter):
    n = len(h0)
    
    def make_consq(active_idx):
        def fun(x):
            h = x[:-1]
            t = x[-1]
            c = compute_corrs(h)
            return t - c[active_idx]
        return fun
    
    def active_fun(x, active_idx):
        h = x[:-1]
        t = x[-1]
        c = compute_corrs(h)
        return t - c[active_idx]
    
    def solve_with_active(h_init, active_idx, t_init):
        m = len(active_idx)
        
        def obj(x):
            return x[-1] * dx
        
        def grad_obj(x):
            g = np.zeros(len(x))
            g[-1] = dx
            return g
        
        def con_ineq(x):
            h = x[:-1]
            t = x[-1]
            c = compute_corrs(h)
            return t - c[active_idx]
        
        def con_eq(x):
            return x[:-1].sum() - target_sum
        
        def con_eq_jac(x):
            j = np.zeros(len(x))
            j[:-1] = 1.0
            return j
        
        constraints = [
            {'type': 'ineq', 'fun': con_ineq},
            {'type': 'eq', 'fun': con_eq, 'jac': con_eq_jac}
        ]
        bounds = [(0.0, 1.0)] * n + [(0.0, None)]
        x0 = np.concatenate([h_init, [t_init]])
        
        try:
            res = minimize(obj, x0, jac=grad_obj, method='SLSQP',
                           bounds=bounds, constraints=constraints,
                           options={'maxiter': maxiter, 'ftol': 1e-14})
            h_out = project_feasible(res.x[:-1], target_sum)
            return h_out
        except Exception:
            return project_feasible(h_init, target_sum)
    
    # Iteratively update active set
    h = h0.copy()
    t_init = c5_of_h(h, dx)
    
    for round_ in range(3):
        corrs = compute_corrs(h)
        sorted_idx = np.argsort(corrs)[::-1]
        active_idx = sorted_idx[:n_active]
        t_init = corrs[active_idx[0]]  # the max in active set
        
        h = solve_with_active(h, active_idx, t_init)
    
    return h, c5_of_h(h, dx)

def combinatorial_search(h0, dx, target_sum, max_iter, t0, budget_s, safety):
    """Gradient-guided combinatorial local search on near-binary solutions."""
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    
    # Identify binary vs fractional
    # We'll work in the continuous space but bias towards binary moves
    
    it = 0
    stall = 0
    T = 0.05 * best_c5  # temperature
    
    while it < max_iter:
        if time.time() - t0 > budget_s - safety:
            break
        
        corrs = compute_corrs(h)
        cur_c5 = corrs.max() * dx
        
        if cur_c5 < best_c5 - 1e-14:
            best_c5 = cur_c5
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            T *= 0.999
            if stall > 5000:
                break
        
        # Compute active gradient
        max_c = corrs.max()
        shifted = corrs - max_c
        w = np.exp(300.0 * shifted)
        w /= w.sum()
        
        grad = np.zeros(n)
        for k in range(2*n - 1):
            if w[k] > 1e-12:
                grad += w[k] * grad_corr_k(h, k, n)
        grad *= dx
        
        # Gradient descent step with projection
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Adaptive step size
        step = (cur_c5 - best_c5) / (gnorm**2 + 1e-15)
        step = min(step, 0.1)
        
        h_new = h - step * grad
        h_new = project_feasible(h_new, target_sum)
        
        new_c5 = c5_of_h(h_new, dx)
        
        # Accept if better, or with SA probability
        delta = new_c5 - cur_c5
        if delta < 0 or (T > 1e-12 and np.random.random() < np.exp(-delta / T)):
            h = h_new
        else:
            # Try smaller step
            h_new2 = h - 0.5 * step * grad
            h_new2 = project_feasible(h_new2, target_sum)
            new_c5_2 = c5_of_h(h_new2, dx)
            if new_c5_2 < cur_c5:
                h = h_new2
            else:
                stall += 1
        
        it += 1
    
    return best_h, best_c5

def binary_hill_climb(h0, dx, target_sum, max_iter, t0, budget_s, safety):
    """Hill climbing on binary (or near-binary) solutions."""
    n = len(h0)
    # Round to binary preserving mass
    h = h0.copy()
    # Sort by distance to 0.5, round the most extreme first
    h = project_feasible(h, target_sum)
    
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    
    for it in range(max_iter):
        if time.time() - t0 > budget_s - safety:
            break
        
        cur_c5 = best_c5
        improved = False
        
        # Try all single moves: move mass from i to j
        # Bias: try moves suggested by gradient first
        corrs = compute_corrs(h)
        max_c = corrs.max()
        shifted = corrs - max_c
        w = np.exp(200.0 * shifted)
        w /= w.sum()
        
        grad = np.zeros(n)
        for k in range(2*n - 1):
            if w[k] > 1e-12:
                grad += w[k] * grad_corr_k(h, k, n)
        grad *= dx
        
        # Try top-K gradient-descending moves
        sources = np.where(h > 0.01)[0]
        sinks = np.where(h < 0.99)[0]
        
        if len(sources) == 0 or len(sinks) == 0:
            break
        
        # Pick top sources (high gradient) and top sinks (low gradient)
        src_rank = np.argsort(-grad[sources])[:20]
        sink_rank = np.argsort(grad[sinks])[:20]
        
        # Random moves too
        n_random = 5
        src_rand = rng_choice(sources, min(n_random, len(sources)))
        sink_rand = rng_choice(sinks, min(n_random, len(sinks)))
        
        best_move = None
        best_move_c5 = cur_c5
        
        candidate_pairs = []
        for si in src_rank:
            for sj in sink_rank:
                candidate_pairs.append((sources[si], sinks[sj]))
        for si in src_rand:
            for sj in sink_rand:
                candidate_pairs.append((si, sj))
        
        for i, j in candidate_pairs:
            if i == j:
                continue
            h_trial = h.copy()
            # Move amount: limited by h[i] and (1-h[j])
            amt = min(h[i], 1.0 - h[j])
            if amt < 1e-6:
                continue
            # Try full move first
            h_trial[i] -= amt
            h_trial[j] += amt
            h_trial = project_feasible(h_trial, target_sum)
            c5_trial = c5_of_h(h_trial, dx)
            if c5_trial < best_move_c5 - 1e-14:
                best_move_c5 = c5_trial
                best_move = h_trial.copy()
        
        if best_move is not None:
            h = best_move
            best_h = h.copy()
            best_c5 = best_move_c5
            improved = True
        
        if not improved and it > 100:
            # Try random perturbation to escape local minimum
            h = project_feasible(h + 0.05 * (np.random.random(n) - 0.5), target_sum)
    
    return best_h, best_c5

def rng_choice(arr, k):
    return np.random.choice(arr, size=min(k, len(arr)), replace=False)

def pair_coordinate_descent(h, dx, target_sum, max_rounds=100):
    """Coordinate descent varying pairs of variables."""
    n = len(h)
    h = h.copy()
    best_c5 = c5_of_h(h, dx)
    
    for round_ in range(max_rounds):
        improved = False
        corrs = compute_corrs(h)
        
        # For each pair (m, p), try to improve
        # Skip pairs that are both binary (h[i] is 0 or 1 for both)
        for m in range(n):
            for p in range(m+1, n):
                # Only consider if at least one is fractional
                if h[m] < 1e-8 and h[p] < 1e-8:
                    continue
                if h[m] > 1-1e-8 and h[p] > 1-1e-8:
                    continue
                
                S = h[m] + h[p]
                if S < 1e-10 or S > 2.0 - 1e-10:
                    continue
                
                # Compute d_k = a_k(m) - a_k(p) for all k
                # a_k(m) = (1-h[m+k])*[m+k<n] + (-h[m-k])*[m-k>=0]
                d = np.zeros(2*n-1)
                for k in range(2*n-1):
                    offset = k - (n-1)
                    a_m = 0.0
                    a_p = 0.0
                    # h[m] contribution

                        # term: h[m] * (1 - h[m+offset])
                        a_m += (1.0 - h[m + offset])

                        # term: h[m-offset] * (1 - h[m])
                        a_m += (-h[m - offset])
                    # h[p] contribution

                        a_p += (1.0 - h[p + offset])
                    if p - offset >= 0 and p - offset < n:
                        a_p += (-h[p - offset])

```python
def minimax_1d(c, d, u_lo, u_hi):
    """Minimize max_k (c[k] + d[k]*u) over u in [u_lo, u_hi]."""
    def f(u):
        return (c + d * u).max()
    
    # Check endpoints
    f_lo = f(u_lo)
    f_hi = f(u_hi)
    best_u = u_lo if f_lo <= f_hi else u_hi
    best_f = min(f_lo, f_hi)
    
    # Split into positive and negative slopes
    pos_mask = d > 1e-14
    neg_mask = d < -1e-14
    
    if not pos_mask.any() or not neg_mask.any():
        # Function is monotone
        if pos_mask.any() and not neg_mask.any():
            return u_lo, f_lo  # increasing → min at u_lo
        elif neg_mask.any() and not pos_mask.any():
            return u_hi, f_hi  # decreasing → min at u_hi
        else:
            return 0.0, f(0.0)  # constant
    
    # Binary search for intersection
    lo, hi = u_lo, u_hi
    for _ in range(60):
        mid = (lo + hi) / 2
        g = (c[pos_mask] + d[pos_mask] * mid).max()  # increasing function
        h_val = (c[neg_mask] + d[neg_mask] * mid).max()  # decreasing function
        if g < h_val:
            lo = mid
        else:
            hi = mid
    
    u_star = (lo + hi) / 2
    f_star = f(u_star)
    
    if f_star < best_f:
        return u_star, f_star
    else:
        return best_u, best_f
```
