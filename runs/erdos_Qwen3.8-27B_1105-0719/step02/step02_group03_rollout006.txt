```python
def compute_swap_delta(h, i, j, n):
    """Compute the change in all correlations when h[i]:1->0 and h[j]:0->1."""
    #_offsets e range from -(n-1) to n-1
    e = np.arange(-(n-1), n)
    
    delta = np.zeros(2*n-1)
    
    # Effect of h[i]: 1->0
    # Term 1: -(1-h[i+e]) where i+e in [0, n-1]
    valid1 = (i + e >= 0) & (i + e <= n-1)
    delta[valid1] -= (1 - h[i + e[valid1]])
    
    # Term 2: +h[i-e] where i-e in [0, n-1]
    valid2 = (i - e >= 0) & (i - e <= n-1)
    delta[valid2] += h[i - e[valid2]]
    
    # Effect of h[j]: 0->1
    # Term 3: +(1-h[j+e]) where j+e in [0, n-1]
    valid3 = (j + e >= 0) & (j + e <= n-1)
    delta[valid3] += (1 - h[j + e[valid3]])
    
    # Term 4: -h[j-e] where j-e in [0, n-1]
    valid4 = (j - e >= 0) & (j - e <= n-1)
    delta[valid4] -= h[j - e[valid4]]
    
    return delta
```

```python
import numpy as np
from scipy.optimize import minimize
import time


def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
        else:
            h += diff / n
    return np.clip(h, 0.0, 1.0)


def c5_of_h(h, dx):
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)


def thue_morse_binary(n):
    tm = np.zeros(n, dtype=np.int32)
    for i in range(1, n):
        tm[i] = tm[i >> 1] ^ (i & 1)
    return tm


def vander_corput_scores(n, base=2):
    dc = np.zeros(n)
    for i in range(n):
        f = 0.0
        denom = 1.0
        num = i
        while num > 0:
            denom *= base
            f += (num % base) / denom
            num //= base
        dc[i] = f
    return dc


def golden_ratio_binary(n, m, shift=0):
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    h = np.zeros(n)
    for k in range(m):
        pos = int(((k + shift) * phi) % n)
        h[pos] = 1.0
    return h


def make_binary(h, m):
    h = np.zeros(len(h))
    idx = np.argsort(-h)
    h[idx[:m]] = 1.0
    return h


def compute_swap_delta(h, i, j, offset_arr, valid_masks):
    n = len(h)
    e = offset_arr
    delta = np.zeros(2 * n - 1)
    v1 = (i + e >= 0) & (i + e <= n - 1)
    delta[v1] -= (1.0 - h[i + e[v1]])
    v2 = (i - e >= 0) & (i - e <= n - 1)
    delta[v2] += h[i - e[v2]]
    v3 = (j + e >= 0) & (j + e <= n - 1)
    delta[v3] += (1.0 - h[j + e[v3]])
    v4 = (j - e >= 0) & (j - e <= n - 1)
    delta[v4] -= h[j - e[v4]]
    return delta


def binary_local_search(h_start, dx, rng, max_iters=500000, patience=15000, num_samples=20):
    n = len(h_start)
    m = n // 2
    h = make_binary(h_start, m)
    offset_arr = np.arange(-(n - 1), n)
    corrs = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = corrs.max() * dx
    best_h = h.copy()
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    no_improve = 0
    best_iter = 0
    for it in range(max_iters):
        if it % 20000 == 0 and it > 0:
            if no_improve > patience:
                break
        trials = rng.choice(len(ones), size=min(num_samples, len(ones)), replace=False)
        trials_z = rng.choice(len(zeros), size=min(num_samples, len(zeros)), replace=False)
        best_delta = None
        best_new_max = corrs.max()
        for ti in trials:
            for tj in trials_z:
                i = ones[ti]
                j = zeros[tj]
                delta = compute_swap_delta(h, i, j, offset_arr, None)
                new_max = (corrs + delta).max()
                if new_max < best_new_max - 1e-14:
                    best_new_max = new_max
                    best_delta = (i, j, delta)
        if best_delta is not None:
            i, j, delta = best_delta
            h[i], h[j] = 0.0, 1.0
            corrs = corrs + delta
            new_c5 = corrs.max() * dx
            best_c5 = new_c5
            best_h = h.copy()
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            no_improve = 0
            best_iter = it
        else:
            no_improve += 1
    return best_h, best_c5


def binary_sa(h_start, dx, rng, max_iters=300000, patience=10000):
    n = len(h_start)
    m = n // 2
    h = make_binary(h_start, m)
    offset_arr = np.arange(-(n - 1), n)
    corrs = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = corrs.max() * dx
    best_h = h.copy()
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    no_improve = 0
    temp = 0.03
    for it in range(max_iters):
        if it % 5000 == 0:
            temp *= 0.97
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        delta = compute_swap_delta(h, i, j, offset_arr, None)
        new_corrs = corrs + delta
        new_c5 = new_corrs.max() * dx
        if new_c5 <= best_c5 + 1e-15:
            h[i], h[j] = 0.0, 1.0
            corrs = new_corrs
            best_c5 = new_c5
            best_h = h.copy()
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            no_improve = 0
        else:
            if temp > 1e-9 and rng.random() < np.exp(-(new_c5 - best_c5) / temp):
                h[i], h[j] = 0.0, 1.0
                corrs = new_corrs
                ones = np.where(h == 1.0)[0]
                zeros = np.where(h == 0.0)[0]
                no_improve = 0
            else:
                no_improve += 1
        if no_improve > patience:
            break
    return best_h, best_c5


def smooth_max_obj(x, dx, beta, corrs_cache=None):
    corrs = np.correlate(x, 1.0 - x, mode='full')
    m = corrs.max()
    shifted = corrs - m
    if beta > 0:
        log_sum = np.log(np.sum(np.exp(beta * shifted)))
        obj = (m + log_sum / beta) * dx
    else:
        obj = m * dx
    return obj


def smooth_max_grad(x, dx, beta):
    n = len(x)
    corrs = np.correlate(x, 1.0 - x, mode='full')
    m = corrs.max()
    shifted = corrs - m
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    w /= w_sum
    grad = np.zeros(n)
    offset_arr = np.arange(-(n - 1), n)
    for k in range(len(corrs)):
        if w[k] < 1e-15:
            continue
        e = offset_arr[k]
        wk = w[k]
        v1 = (offset_arr >= -e) & (offset_arr + e >= 0) & (offset_arr + e <= n - 1)
        idx1 = offset_arr[v1]
        v1b = (idx1 >= 0) & (idx1 <= n - 1)
        grad[idx1[v1b]] += wk * (1.0 - x[idx1[v1b] + e])
        v2 = (offset_arr <= -e) & (offset_arr - e >= 0) & (offset_arr - e <= n - 1)
        idx2 = offset_arr[v2]
        grad[idx2] += wk * (-x[idx2 - e])
    return grad * dx


def grad_corr_single(x, e, n):
    grad = np.zeros(n)
    for i in range(n):
        if i + e >= 0 and i + e <= n - 1:
            grad[i] += (1.0 - x[i + e])
        if i - e >= 0 and i - e <= n - 1:
            grad[i] += -x[i - e]
    return grad


def smooth_max_grad_eff(x, dx, beta):
    n = len(x)
    corrs = np.correlate(x, 1.0 - x, mode='full')
    m = corrs.max()
    shifted = corrs - m
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum == 0:
        return np.zeros(n)
    w /= w_sum
    grad = np.zeros(n)
    for k in range(len(corrs)):
        if w[k] < 1e-14:
            continue
        e = k - (n - 1)
        wk = w[k]
        for i in range(n):
            if 0 <= i + e <= n - 1:
                grad[i] += wk * (1.0 - x[i + e])
            if 0 <= i - e <= n - 1:
                grad[i] += wk * (-x[i - e])
    return grad * dx


def slsqp_refine(h0, dx, target, beta, maxiter=500):
    n = len(h0)

    def obj_fn(x):
        return smooth_max_obj(x, dx, beta)

    def grad_fn(x):
        return smooth_max_grad_eff(x, dx, beta)

    def con_fun(x):
        return x.sum() - target

    def con_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_fn, h0, jac=grad_fn, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        h_out = project_feasible(res.x, target)
    except Exception:
        h_out = project_feasible(h0, target)
    return h_out, c5_of_h(h_out, dx)


def frank_wolfe_refine(h0, dx, target, max_iter=2000, tol=1e-10):
    n = len(h0)
    h = h0.copy()
    offset_arr = np.arange(-(n - 1), n)
    best_val = c5_of_h(h, dx)
    for it in range(max_iter):
        corrs = np.correlate(h, 1.0 - h, mode='full')
        max_idx = np.argmax(corrs)
        e = offset_arr[max_idx]
        g = grad_corr_single(h, e, n)
        # Linear minimization: find z in [0,1]^n, sum=z=n/2 minimizing g·z
        # Optimal: z_i = 1 if g_i is among the smallest n/2, else 0
        sorted_g = np.argsort(g)
        z = np.zeros(n)
        z[sorted_g[:n // 2]] = 1.0
        # Line search
        alpha = 1.0
        for _ in range(30):
            h_new = (1 - alpha) * h + alpha * z
            h_new = project_feasible(h_new, target)
            val = c5_of_h(h_new, dx)
            if val <= best_val:
                h = h_new
                best_val = val
                break
            alpha /= 2
        else:
            break
        if best_val < c5_of_h(h0, dx) - tol:
            pass
    return h, best_val
```
