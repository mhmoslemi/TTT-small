            G += w[k] * grad_corr_d(h, d, self.n)
        return (G[:self.m] - G[self.m:][::-1]) * self.dx

class EpigraphQ:
    def __init__(self, n, active_k):
        self.n = n
        self.m = n // 2
        self.active_k = active_k
        self.dx = 2.0 / n

    def obj(self, x):
        return x[0]

    def obj_grad(self, x):
        g = np.zeros(self.m + 1)
        g[0] = 1.0
        return g

    def cons_fun(self, x, i):
        t = x[0]
        q = x[1:]
        h = build_h_from_q(q, self.n)
        k = self.active_k[i]
        d = k - (self.n - 1)
        n = self.n
        if d >= 0:
            val = np.sum(h[d:] * (1.0 - h[:n - d]))
        else:
            dd = -d
            val = np.sum(h[dd:] * (1.0 - h[:n - dd]))
        return t - val * self.dx

    def cons_grad(self, x, i):
        q = x[1:]
        h = build_h_from_q(q, self.n)
        k = self.active_k[i]
        d = k - (self.n - 1)
        n = self.n
        g = np.zeros(self.m + 1)
        g[0] = 1.0
        G = grad_corr_d(h, d, n)
        g[1:] = -(G[:self.m] - G[self.m:][::-1]) * self.dx
        return g

def solve_epigraph_q(q0, n, active_k, t0, maxiter=200):
    m = n // 2
    x0 = np.concatenate([[t0], q0])
    ep = EpigraphQ(n, active_k)
    cons = []
    for i in range(len(active_k)):
        cons.append({
            'type': 'ineq',
            'fun': lambda x, i=i: ep.cons_fun(x, i),
            'jac': lambda x, i=i: ep.cons_grad(x, i)
        })
    bounds = [(0.0, 5.0)] + [(0.0, 1.0)] * m
    try:
        res = minimize(ep.obj, x0, jac=ep.obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return np.clip(res.x[1:], 0.0, 1.0), res.x[0], res
    except Exception:
        return q0, t0, None

def active_set_search(q0, n):
    m = n // 2
    q = q0.copy()
    dx = 2.0 / n
    h = build_h_from_q(q, n)
    corrs = compute_corrs(h)
    current_max = corrs.max() * dx

    L = min(40, 2 * n - 1)
    top_idx = np.argsort(corrs)[::-1][:L]
    active_k = sorted(top_idx.tolist())

    best_q = q.copy()
    best_c5 = current_max
    t0 = current_max + 1e-5

    for _ in range(15):
        h = build_h_from_q(q, n)
        corrs = compute_corrs(h)
        cur = corrs.max() * dx
        if cur < best_c5:
            best_c5 = cur
            best_q = q.copy()

        q_new, t_solv, _ = solve_epigraph_q(q, n, active_k, t0, maxiter=150)
        h_new = build_h_from_q(q_new, n)
        corrs_new = compute_corrs(h_new)
        c5_new = corrs_new.max() * dx

        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_q = q_new.copy()

        hidden = np.where(corrs_new > t_solv + 1e-7)[0]
        if len(hidden) > 0:
            new_active = sorted(set(active_k) | set(hidden.tolist()))
            if len(new_active) > 60:
                vals = corrs_new[new_active]
                keep = np.argsort(vals)[::-1][:60]
                new_active = sorted(new_active[i] for i in keep)
            active_k = new_active

        t0 = max(t_solv, c5_new) + 1e-5
        q = q_new
        if abs(c5_new - best_c5) < 1e-12 and len(hidden) == 0:
            break

    return best_q, best_c5

class FullEpigraph:
    def __init__(self, n, active_k, target_sum):
        self.n = n
        self.active_k = active_k
        self.dx = 2.0 / n
        self.target_sum = target_sum

    def obj(self, x):
        return x[0]

    def obj_grad(self, x):
        g = np.zeros(self.n + 1)
        g[0] = 1.0
        return g

    def cons_corr(self, x, i):
        t = x[0]
        h = x[1:]
        k = self.active_k[i]
        d = k - (self.n - 1)
        n = self.n
        if d >= 0:
            val = np.sum(h[d:] * (1.0 - h[:n - d]))
        else:
            dd = -d
            val = np.sum(h[dd:] * (1.0 - h[:n - dd]))
        return t - val * self.dx

    def cons_corr_grad(self, x, i):
        h = x[1:]
        k = self.active_k[i]
        d = k - (self.n - 1)
        n = self.n
        g = np.zeros(n + 1)
        g[0] = 1.0
        G = grad_corr_d(h, d, n)
        g[1:] = -G * self.dx
        return g

    def cons_sum(self, x):
        return x[1:].sum() - self.target_sum

    def cons_sum_grad(self, x):
        g = np.zeros(self.n + 1)
        g[1:] = 1.0
        return g

def solve_full_epigraph(h0, n, active_k, t0, target_sum, maxiter=150):
    x0 = np.concatenate([[t0], h0])
    ep = FullEpigraph(n, active_k, target_sum)
    cons = []
    for i in range(len(active_k)):
        cons.append({
            'type': 'ineq',
            'fun': lambda x, i=i: ep.cons_corr(x, i),
            'jac': lambda x, i=i: ep.cons_corr_grad(x, i)
        })
    cons.append({'type': 'eq', 'fun': ep.cons_sum, 'jac': ep.cons_sum_grad})
    bounds = [(0.0, 5.0)] + [(0.0, 1.0)] * n
    try:
        res = minimize(ep.obj, x0, jac=ep.obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        h_out = project_feasible(res.x[1:], target_sum)
        return h_out, res.x[0], res
    except Exception:
        return project_feasible(h0, target_sum), t0, None

def full_active_search(h0, n):
    target_sum = n / 2.0
    dx = 2.0 / n
    corrs = compute_corrs(h0)
    current_max = corrs.max() * dx

    L = min(30, 2 * n - 1)
    top_idx = np.argsort(corrs)[::-1][:L]
    active_k = sorted(top_idx.tolist())

    h = h0.copy()
    best_h = h0.copy()
    best_c5 = current_max
    t0 = current_max + 1e-5

    for _ in range(10):
        h = project_feasible(h, target_sum)
        corrs = compute_corrs(h)
        cur = corrs.max() * dx
        if cur < best_c5:
            best_c5 = cur
            best_h = h.copy()

        h_new, t_solv, _ = solve_full_epigraph(h, n, active_k, t0, target_sum, maxiter=120)
        h_new = project_feasible(h_new, target_sum)
        corrs_new = compute_corrs(h_new)
        c5_new = corrs_new.max() * dx

        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()

        hidden = np.where(corrs_new > t_solv + 1e-7)[0]
        if len(hidden) > 0:
            new_active = sorted(set(active_k) | set(hidden.tolist()))
            if len(new_active) > 50:
                vals = corrs_new[new_active]
                keep = np.argsort(vals)[::-1][:50]
                new_active = sorted(new_active[i] for i in keep)
            active_k = new_active

        t0 = max(t_solv, c5_new) + 1e-5
        h = h_new
        if abs(c5_new - best_c5) < 1e-12 and len(hidden) == 0:
            break

    return best_h, best_c5

def interpolate_h(h_old, n_new):
    x_old = np.linspace(0.0, 2.0, len(h_old))
    x_new = np.linspace(0.0, 2.0, n_new)
    return np.interp(x_new, x_old, h_old)

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)

    def elapsed():
        return time.time() - t_start

    def remaining():
        return budget_s - elapsed()

    # Initialize
    h_parent = np.array(initial_h_values, dtype=float)
    n_parent = len(h_parent)
    target_sum = n_parent / 2.0
    h_parent = project_feasible(h_parent, target_sum)
    h_parent = np.clip(h_parent, 0.0, 1.0)

    best_c5 = exact_score(h_parent)
    best_h = h_parent.copy()
    best_n = n_parent

    # Project to symmetric form
    n = n_parent
    m = n // 2
    q0 = q_from_h(h_parent, n)
    h_sym = build_h_from_q(q0, n)
    score_sym = exact_score(h_sym)
    if score_sym < best_c5:
        best_c5 = score_sym
        best_h = h_sym.copy()
    q_best = q0.copy()

    # Phase 1: Homotopy in q-space
    for beta in [40, 100, 300, 800, 2000, 5000]:
        if remaining() < 25:
            break
        obj = QSmooth(n, beta)
        try:
            res = minimize(obj.value, q_best, jac=obj.gradient,
                           method='L-BFGS-B',
                           bounds=[(0.0, 1.0)] * m,
                           options={'maxiter': 200, 'ftol': 1e-16,
                                    'gtol': 1e-10, 'maxfun': 5000})
            q_new = np.clip(res.x, 0.0, 1.0)
            h_new = build_h_from_q(q_new, n)
            s_new = exact_score(h_new)
            if s_new < best_c5 - 1e-14:
                best_c5 = s_new
                best_h = h_new.copy()
                q_best = q_new.copy()
        except Exception:
            pass

    # Phase 2: Simulated Annealing in q-space
    if remaining() > 100:
        T_start = 5e-4
        T_end = 1e-6
        # Estimate total steps based on time
        # Each step: build_h + correlate + max ≈ 0.1-0.5ms for n=126
        # Estimate 3000 steps/sec, so in 300s we get ~900k steps
        # But let's be conservative: aim for 200k-500k steps
        sa_time = min(remaining() - 50, 350)
        
        q_sa = q_best.copy()
        h_sa = build_h_from_q(q_sa, n)
        score_sa = exact_score(h_sa)
        best_q_sa = q_sa.copy()
        best_score_sa = score_sa
        
        # Adaptive parameters
        noise_base = 0.05  # initial noise scale (fraction of [0,1])
        steps = 0
        max_steps = int(sa_time * 2000)  # ~2000 evals/sec target
        cooling = (T_end / T_start) ** (1.0 / max(max_steps, 1))
        
        for step in range(max_steps):
            if remaining() < 60:
                break
            T = T_start * (cooling ** step)
            noise = noise_base * (T / T_start) ** 0.5
            
            q_prop = q_sa.copy()
            r = rng.random()
            if r < 0.35:
                idx = rng.randint(m)
                q_prop[idx] += rng.randn() * noise
            elif r < 0.7:
                sz = rng.randint(2, 8)
                sz = min(sz, m)
                st = rng.randint(0, max(1, m - sz))
                q_prop[st:st+sz] += rng.randn(sz) * noise
            else:
