The gradient at h₀ is:

subject to: for each active k: R_k(h₀) + ∇R_k(h₀)·δ ≤ t

            h₀[j] + δ[j] ≥ 0
            Σ δ[j] = 0 (mass preservation)

import numpy as np
from scipy.optimize import minimize, linprog
import time

def eval_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def corr_raw(h):
    return np.correlate(h, 1.0 - h, mode='full')

def build_sym_h(q):
    m = len(q)
    h = np.empty(2 * m)
    h[:m] = q
    h[m:] = 1.0 - q[::-1]
    return h

def sym_from_h(h):
    n = len(h)
    m = n // 2
    q = np.empty(m)
    for i in range(m):
        q[i] = 0.5 * (h[i] + (1.0 - h[n - 1 - i]))
    return np.clip(q, 0.0, 1.0)

def project_feasible(x, target_sum):
    n = len(x)
    x = np.clip(np.nan_to_num(x), 0.0, 1.0)
    if abs(x.sum() - target_sum) < 1e-12:
        return x.copy()
    lo, hi = -1.5, 1.5
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(x + mid, 0.0, 1.0).sum()
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    h = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(h)
        else:
            order = np.argsort(-h)
        remaining = abs(diff)
        sign = 1 if diff > 0 else -1
        for idx in order:
            if remaining < 1e-12:
                break
            if sign > 0:
                inc = min(remaining, 1.0 - h[idx])
                h[idx] += inc
                remaining -= inc
            else:
                dec = min(remaining, h[idx])
                h[idx] -= dec
                remaining -= dec
    return np.clip(h, 0.0, 1.0)

def corr_grad_full(h, k):
    n = len(h)
    d = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -d)
    i_hi = min(n - 1, n - 1 - d)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx] += (1.0 - h[idx + d])
    grad[idx + d] += (-h[idx])
    return grad

def transfer_line_search(h, a, b, n_grid=20):
    """1D line search along transfer direction h[a]+delta, h[b]-delta"""
    lower = max(-h[a], h[b] - 1.0)
    upper = min(1.0 - h[a], -h[b])
    if upper - lower < 1e-10:
        return h.copy(), eval_score(h)
    
    best_delta = 0.0
    best_score = eval_score(h)
    
    # Grid search
    deltas = np.linspace(lower, upper, n_grid)
    for delta in deltas:
        h_trial = h.copy()
        h_trial[a] += delta
        h_trial[b] -= delta
        h_trial = np.clip(h_trial, 0, 1)
        sc = eval_score(h_trial)
        if sc < best_score:
            best_score = sc
            best_delta = delta
    
    if best_delta == 0.0:
        return h.copy(), best_score
    
    # Refine with golden section around best_delta
    # Find the neighborhood
    grid_step = (upper - lower) / (n_grid - 1)
    lo_gs = max(lower, best_delta - grid_step)
    hi_gs = min(upper, best_delta + grid_step)
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for _ in range(30):
        m1 = hi_gs - gr * (hi_gs - lo_gs)
        m2 = lo_gs + gr * (hi_gs - lo_gs)
        h1 = h.copy(); h1[a] += m1; h1[b] -= m1; h1 = np.clip(h1, 0, 1)
        h2 = h.copy(); h2[a] += m2; h2[b] -= m2; h2 = np.clip(h2, 0, 1)
        s1 = eval_score(h1)
        s2 = eval_score(h2)
        if s1 < s2:
            hi_gs = m2
        else:
            lo_gs = m1
        if hi_gs - lo_gs < 1e-10:
            break
    
    best_delta = (lo_gs + hi_gs) / 2.0
    h_best = h.copy()
    h_best[a] += best_delta
    h_best[b] -= best_delta
    h_best = np.clip(h_best, 0, 1)
    best_score = eval_score(h_best)
    
    if best_score < eval_score(h):
        return h_best, best_score
    return h.copy(), best_score

def lp_local_optimize_full(h, n_active=30, rho=0.3, max_iters=50, time_limit=None):
    """LP-based local optimization for full vector."""
    n = len(h)
    h = h.copy()
    target = n / 2.0
    best_c5 = eval_score(h)
    
    for it in range(max_iters):
        if time_limit is not None and time.time() > time_limit:
            break
        
        corrs = corr_raw(h)
        # Select active lags
        sorted_idx = np.argsort(-corrs)[:n_active]
        
        # Build LP
        # Variables: delta (n), t (1) -> total n+1
        # Minimize t
        # Subject to:
        #   For each active k: sum_j grad_k[j] * delta[j] - t <= -corrs[k]
        #   Actually: corrs[k] + sum_j grad_k[j]*delta[j] <= t
        #   => sum_j grad_k[j]*delta[j] - t <= -corrs[k]
        #   0 <= h[j] + delta[j] <= 1
        #   sum delta[j] = 0
        #   |delta[j]| <= rho
        
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0  # minimize t
        
        A_ub = np.zeros((n_active, n + 1))
        b_ub = np.zeros(n_active)
        
        for i, k in enumerate(sorted_idx):
            g = corr_grad_full(h, k)
            A_ub[i, :n] = g
            A_ub[i, n] = -1.0
            b_ub[i] = -corrs[k]
        
        # Box constraints: -h[j] <= delta[j] <= 1-h[j]
        # These go into bounds
        
        # |delta[j]| <= rho
        bounds = []
        for j in range(n):
            lo = max(-h[j], -rho)
            hi = min(1.0 - h[j], rho)
            bounds.append((lo, hi))
        bounds.append((0.0, 10.0))  # t >= 0
        
        # Equality: sum(delta) = 0
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)
        
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs')
            if res.success:
                delta = res.x[:n]
                # Line search along delta
                h_trial = h + delta
                h_trial = project_feasible(h_trial, target)
                
                # Try the full step and some fractions
                best_alpha = 0.0
                best_sc = eval_score(h)
                for alpha in [1.0, 0.5, 0.25, 0.125]:
                    h_try = h + alpha * (h_trial - h)
                    h_try = project_feasible(h_try, target)
                    sc = eval_score(h_try)
                    if sc < best_sc:
                        best_sc = sc
                        best_alpha = alpha
                
                if best_alpha > 0 and best_sc < best_c5 - 1e-12:
                    h = h + best_alpha * (h_trial - h)
                    h = project_feasible(h, target)
                    best_c5 = eval_score(h)
                else:
                    break
            else:
                break
        except Exception:
            break
    
    return h, best_c5
