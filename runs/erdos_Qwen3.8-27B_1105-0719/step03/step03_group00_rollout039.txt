    corrs = np.correlate(h, 1.0 - h, mode='full')
    cmax = corrs.max()
    shifted = (corrs - cmax) / T
    exps = np.exp(shifted)
    weights = exps / exps.sum()
    grad = np.zeros(n)
    for k in range(len(corrs)):

        idx = np.arange(i_lo, i_hi + 1)
        w = weights[k]
        grad[idx + off] += w * (1.0 - h[idx])
        grad[idx] += -w * h[idx + off]
    return grad * dx

def pair_reoptimize(h, i, j, n, dx, tol=1e-10):
    """Optimize h[i] and h[j] keeping h[i]+h[j] constant."""
    s = h[i] + h[j]
    if s < 0 or s > 1:
        s = np.clip(s, 0, 1)
    
    # For each k, corr[k] = c_k + d_k * t where h[i]=t, h[j]=s-t
    # corr[k] = sum_m h[m]*(1-h[m+off])
    # Only terms involving h[i] or h[j] change
    # d_k = contribution of h[i] - contribution of h[j] (since h[j] = s-t)
    
    # Actually, let me just evaluate corr at two points and interpolate
    # Since corr is linear in t, we can compute it at t=0 and t=s
    
    h_old_i, h_old_j = h[i], h[j]
    
    # Evaluate at t = 0 (h[i]=0, h[j]=s)
    h[i], h[j] = 0.0, s
    corrs_0 = np.correlate(h, 1.0 - h, mode='full')
    
    # Evaluate at t = s (h[i]=s, h[j]=0)
    h[i], h[j] = s, 0.0
    corrs_s = np.correlate(h, 1.0 - h, mode='full')
    
    # corrs(t) = corrs_0 + (t/s) * (corrs_s - corrs_0) for s > 0
    # We want to minimize max_k corrs(t) * dx
    # i.e., minimize max_k [corrs_0[k] + (t/s) * (corrs_s[k] - corrs_0[k])]
    
    if s < 1e-12:
        h[i], h[j] = 0.0, 0.0
        return h
    
    delta = (corrs_s - corrs_0) / s  # derivative w.r.t. t
    base = corrs_0  # value at t=0
    
    # Minimize max_k [base[k] + t * delta[k]] over t in [0, s]
    # This is a 1D convex minimization
    
    # The optimal t is where the subgradient contains 0
    # Subgradient at t: {sum of weights_k * delta[k] where k are argmax}
    # More practically: bisection on the derivative
    
    lo_t, hi_t = 0.0, s
    
    def max_at_t(tt):
        return np.max(base + tt * delta)
    
    def subgrad_low(tt):
        vals = base + tt * delta
        mx = vals.max()
        active = vals >= mx - 1e-12
        return delta[active].min()
    
    def subgrad_high(tt):
        vals = base + tt * delta
        mx = vals.max()
        active = vals >= mx - 1e-12
        return delta[active].max()
    
    # Bisection
    for _ in range(60):
        mid_t = (lo_t + hi_t) / 2
        sv_lo = subgrad_low(mid_t)
        sv_hi = subgrad_high(mid_t)
        if sv_lo > 1e-14:
            hi_t = mid_t  # derivative positive, move left
        elif sv_hi < -1e-14:
            lo_t = mid_t  # derivative negative, move right
        else:
            break
    
    t_opt = (lo_t + hi_t) / 2

def pair_reopt_fast(h, i, j, n, corrs):
    """Optimize h[i] and h[j] keeping sum fixed. Uses precomputed correlations.
    Returns new h[i], h[j] and new corrs."""
    s = h[i] + h[j]
    a, b = h[i], h[j]
    
    # Valid range for h[i] = a + δ, h[j] = b - δ
    # Need: 0 <= a+δ <= 1 and 0 <= b-δ <= 1
    # So: -a <= δ <= 1-a and b-1 <= δ <= b
    d_lo = max(-a, b - 1.0)
    d_hi = min(1.0 - a, b)
    
    if d_lo >= d_hi:
        return a, b, corrs
    
    # Compute Δcorr[k] / δ for each k
    # Δcorr[k] = δ * coef[k] where coef[k] = [(1-h[i+off]) - h[i-off]] - [(1-h[j+off]) - h[j-off]]
    coef = np.zeros(2 * n - 1)
    
    for k in range(2 * n - 1):
        off = k - (n - 1)
        c_val = 0.0
        # h[i] as m in h[m]*(1-h[m+off])
        if 0 <= i + off < n:
            c_val += (1.0 - h[i + off])
        # h[i] as (1-h[m+off]) where m+off = i, i.e., m = i-off
        if 0 <= i - off < n:
            c_val += (-h[i - off])
        # h[j] as m
        if 0 <= j + off < n:
            c_val -= (1.0 - h[j + off])
        # h[j] as (1-h[m+off])
        if 0 <= j - off < n:
            c_val -= (-h[j - off])
        coef[k] = c_val
    
    # Now minimize max_k (corrs[k] + δ * coef[k]) over δ in [d_lo, d_hi]
    # Bisection
    lo_d, hi_d = d_lo, d_hi
    for _ in range(80):
        mid_d = (lo_d + hi_d) / 2.0
        vals = corrs + mid_d * coef
        mx = vals.max()
        active = np.where(vals >= mx - 1e-10)[0]
        sv_min = coef[active].min()
        sv_max = coef[active].max()
        if sv_min > 1e-14:
            hi_d = mid_d
        elif sv_max < -1e-14:
            lo_d = mid_d
        else:
            break
    
    d_opt = (lo_d + hi_d) / 2.0
    new_i = a + d_opt
    new_j = b - d_opt
    new_corrs = corrs + d_opt * coef
    return new_i, new_j, new_corrs

    dx = 2.0 / n
    ones = np.where(h_bin == 1)[0]
    zeros = np.where(h_bin == 0)[0]
    if len(ones) > target_count:
        to_flip = ones[len(ones) - (len(ones) - target_count):]
        for idx in to_flip:
            h_bin[idx] = 0
        ones = np.where(h_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]
    elif len(ones) < target_count:
        to_flip = zeros[:target_count - len(ones)]
        for idx in to_flip:
            h_bin[idx] = 1
        ones = np.where(h_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]
    best_bin = h_bin.copy()
    best_c5 = c5_of_h(h_bin, dx)
    cur_c5 = best_c5
    T = 0.04
    T_min = 0.0005
    cooling = 0.9998
    rng = np.random.RandomState(seed)
    for it in range(n_iters):
        if time.time() - t0 > budget_s - 5:
            break
        if it % 500 == 0:
            T = max(T_min, T * cooling)
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h_bin[i] = 0
        h_bin[j] = 1
        new_c5 = c5_of_h(h_bin, dx)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            cur_c5 = new_c5
            ones = np.where(h_bin == 1)[0]
            zeros = np.where(h_bin == 0)[0]
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_bin = h_bin.copy()
        else:
            h_bin[i] = 1
            h_bin[j] = 0
    return best_bin, best_c5

def transfer_grid(h_src, n_target):
    n_src = len(h_src)
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_tgt = (np.arange(n_target) + 0.5) * (2.0 / n_target)
    return np.interp(x_tgt, x_src, h_src)

def perturb_and_optimize(h, n, target_sum, t0, budget_s, rng, scale=0.05):
    h_p = h + scale * rng.standard_normal(n)
    h_p = project_feasible(h_p, target_sum)
    h_opt, c5_opt = pair_cd(h_p, n, t0, budget_s, n_sweeps=40, n_pairs=60, seed=int(time.time()*100)%10000)
    return h_opt, c5_opt

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def tl():
        return budget_s - (time.time() - t0)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = np.full(128, 0.5)

    n = len(h0)
    dx = 2.0 / n
    target_sum = n / 2.0
    h0 = project_feasible(h0, target_sum)
    best_h = h0.copy()
    best_c5 = c5_of_h(h0, dx)

    # Phase 1: Intensive pair CD
    if tl() > 80:
        h1, c51 = pair_cd(best_h, n, t0, budget_s, n_sweeps=100, n_pairs=100, seed=seed)
        if c51 < best_c5:
            best_c5 = c51
            best_h = h1

    # Phase 2: LSE-Adam
    if tl() > 80:
        h2, c52 = lse_adam(best_h, n, target_sum, t0, budget_s, seed=seed+1)
        if c52 < best_c5:
            best_c5 = c52
            best_h = h2
        if tl() > 30:
            h2b, c52b = pair_cd(best_h, n, t0, budget_s, n_sweeps=60, n_pairs=80, seed=seed+2)
            if c52b < best_c5:
                best_c5 = c52b
                best_h = h2b

    # Phase 3: Perturbation + re-optimization
    if tl() > 60:
        for r in range(5):
            if tl() < 30:
                break
            sc = 0.08 / (r + 1)
            h_p, c5_p = perturb_and_optimize(best_h, n, target_sum, t0, budget_s, rng, scale=sc)
            if c5_p < best_c5:
                best_c5 = c5_p
                best_h = h_p

    # Phase 4: Binary SA
    if tl() > 60:
        h_bin, c5_bin = binary_sa(best_h, n, t0, budget_s, n_iters=20000, seed=seed+10)
        if c5_bin < best_c5 + 0.005:
            h_bin_cont = project_feasible(h_bin.astype(float), target_sum)
            h_br, c5_br = pair_cd(h_bin_cont, n, t0, budget_s, n_sweeps=60, n_pairs=80, seed=seed+20)
            if c5_br < best_c5:
                best_c5 = c5_br
                best_h = h_br

    # Phase 5: Multi-grid exploration
    n_base = len(best_h)
    n_candidates = sorted(set([
        n_base - 4, n_base - 2, n_base + 2, n_base + 4,
        96, 104, 112, 120, 128, 136, 144, 152, 160, 168, 176, 184
    ]))
    n_candidates = [x for x in n_candidates if 64 <= x <= 224 and x % 2 == 0 and x != n_base]

    for n_try in n_candidates:
        if tl() < 25:
            break
        dx_t = 2.0 / n_try
        ts_t = n_try / 2.0
        h_t = transfer_grid(best_h, n_try)
        h_t = project_feasible(h_t, ts_t)
        h_opt, c5_opt = pair_cd(h_t, n_try, t0, budget_s, n_sweeps=50, n_pairs=70, seed=seed+n_try)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 6: Final intensive pair CD on best
    n_b = len(best_h)
    ts_b = n_b / 2.0
    if tl() > 20:
        h_f, c5_f = pair_cd(best_h, n_b, t0, budget_s, n_sweeps=200, n_pairs=60, seed=seed+100)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f

    # Phase 7: Final LSE polish
    if tl() > 15:
        h_f2, c5_f2 = lse_adam(best_h, n_b, ts_b, t0, budget_s, seed=seed+200)
        if c5_f2 < best_c5:
            best_c5 = c5_f2
            best_h = h_f2
        if tl() > 10:
            h_f3, c5_f3 = pair_cd(best_h, n_b, t0, budget_s, n_sweeps=150, n_pairs=40, seed=seed+300)
            if c5_f3 < best_c5:
                best_c5 = c5_f3
                best_h = h_f3

    # Phase 8: Very fine polish
    n_final_base = len(best_h)
    ts_final_base = n_final_base / 2.0
    if tl() > 10:
        h_fin, c5_fin = pair_cd(best_h, n_final_base, t0, budget_s, n_sweeps=300, n_pairs=30, seed=seed+400)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin

    # Final validation
    n_final = len(best_h)
    dx_final = 2.0 / n_final
    ts_final = n_final / 2.0
    best_h = project_feasible(best_h, ts_final)
    best_c5 = c5_of_h(best_h, dx_final)

    return best_h, best_c5, n_final
