So the optimal t for a given pair is where the subgradient contains 0, i.e., where there exist at least two lags that are simultaneously maximal and their g_d values straddle 0.

This is a much better approach! I can:
1. For each pair (i,j), compute g_d for all d
2. Find the t that minimizes max_d (F_d(0) + t * g_d) subject to 0 <= h[i]+t <= 1 and 0 <= h[j]-t <= 1
3. Accept if improvement

For a convex piecewise linear function, the minimum is at a "kink point" where two or more lines are simultaneously maximal. I can find this by:
- Computing all F_d(0) + t * g_d for the feasible range of t

import numpy as np
from scipy.optimize import minimize
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx])
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-14:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmin(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def pair_line_search(h, i, j, n_evals=50):
    """Given pair (i,j), find optimal t minimizing max profile."""
    n = len(h)
    dx = 2.0 / n
    a = h[i]
    b = h[j]
    # Constraints: 0 <= a+t <= 1, 0 <= b-t <= 1
    t_lo = max(-a, b - 1.0)
    t_hi = min(1.0 - a, b)
    if t_hi - t_lo < 1e-12:
        return h, exact_score(h)
    
    # Compute gradient of each F_d with respect to t
    # F_d = sum_x h[x]*(1-h[x+d])
    # dF_d/dt = (1-h[i+d]) - h[i-d] - (1-h[j+d]) + h[j-d] (for valid indices)
    M = 2 * n - 1
    g = np.zeros(M)
    for d in range(-n+1, n):
        idx = d + (n - 1)  # position in profile array
        val = 0.0
        # Contribution from h[i] = h[i] -> h[i]+t
        # F_d includes term h[x]*(1-h[x+d]) 
        # If x = i: contributes (1-h[i+d]) * dt (if i+d valid)
        # If x+d = i, i.e., x = i-d: contributes -h[i-d] * dt (if i-d valid)
        if i + d >= 0 and i + d < n:
            val += (1.0 - h[i + d])
        if i - d >= 0 and i - d < n:
            val += -h[i - d]
        # Contribution from h[j] = h[j] -> h[j]-t
        if j + d >= 0 and j + d < n:
            val += -(1.0 - h[j + d])
        if j - d >= 0 and j - d < n:
            val += h[j - d]
        g[idx] = val * dx
    
    F0 = compute_profile(h)
    
    # f(t) = max_d (F0[d] + t * g[d]) is convex in t
    # Find minimum using golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0  # golden ratio
    
    def f_of_t(t):
        return float(np.max(F0 + t * g))
    
    # Golden section search
    lo = t_lo
    hi = t_hi
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f_of_t(c)
    fd = f_of_t(d)
    
    for _ in range(n_evals):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f_of_t(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f_of_t(d)
        if hi - lo < 1e-10:
            break
    
    t_opt = (lo + hi) / 2.0
    h_new = h.copy()
    h_new[i] = a + t_opt
    h_new[j] = b - t_opt
    h_new = repair_feasible(h_new)
    c_new = exact_score(h_new)
    return h_new, c_new

def active_pair_descent(h, max_rounds=100, pairs_per_round=20, rng=None):
    """Iteratively apply pair line search to active lags."""
    n = len(h)
    best_c = exact_score(h)
    h = h.copy()
    no_improve = 0
    
    for rnd in range(max_rounds):
        profile = compute_profile(h)
        c = float(np.max(profile))
        if c + 1e-14 < best_c:
            best_c = c
            h_best = h.copy()
            no_improve = 0
        else:
            h_best = h.copy()
            no_improve += 1
            if no_improve > 5:
                break
        
        # Find active lags
        margin = max(0.002, 3.0 * (c - float(np.median(profile))))
        active = np.where(profile >= c - margin)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(profile))])
        
        # Generate candidate pairs
        pairs = []
        for lag_idx in active[:min(8, len(active))]:
            d = lag_idx - (n - 1)
            if d > 0:
                # Lag d: F_d = sum_{x=0}^{n-1-d} h[x]*(1-h[x+d])
                # High h[x] and low h[x+d] contribute most
                for x in range(n - d):
                    contrib = h[x] * (1 - h[x + d])
                    if contrib > 0.01:
                        pairs.append((x, x + d))
            elif d < 0:
                dp = -d
                for x in range(n - dp):
                    contrib = h[x + dp] * (1 - h[x])
                    if contrib > 0.01:
                        pairs.append((x, x + dp))
            
            # Also add random pairs
            if rng is not None:
                for _ in range(3):
                    a = rng.randint(n)
                    b = rng.randint(n)
                    if a != b:
                        pairs.append((a, b))
        
        if not pairs:
            break
        
        # Sample pairs
        if len(pairs) > pairs_per_round:
            if rng is not None:
                idx = rng.choice(len(pairs), size=pairs_per_round, replace=False)
                pairs = [pairs[k] for k in idx]
            else:
                pairs = pairs[:pairs_per_round]
        
        improved = False
        for (pi, pj) in pairs:
            if pi == pj:
                continue
            h_new, c_new = pair_line_search(h, pi, pj)
            if c_new < best_c - 1e-12:
                h = h_new
                best_c = c_new
                improved = True
        
        if not improved:
            no_improve += 1
            if no_improve > 10:
                break
    
    h_best = h.copy()

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = np.linspace(0, 2, n + 1)
    xt = np.linspace(0, 2, new_n + 1)
    h_interp = np.zeros(new_n)
    for k in range(new_n):
        lo_k, hi_k = xt[k], xt[k + 1]
        total_mass = 0.0
        total_width = 0.0
        for j in range(n):
            lo_j, hi_j = xs[j], xs[j + 1]
            overlap = min(hi_k, hi_j) - max(lo_k, lo_j)
            if overlap > 0:
                total_mass += h[j] * overlap
                total_width += overlap
        h_interp[k] = total_mass / total_width if total_width > 1e-15 else 0.5
    h_interp = np.clip(h_interp, 0.0, 1.0)
    return repair_feasible(h_interp)

def antisym_to_full(y):
    """Convert anti-symmetric half-vector to full vector."""
    half = len(y)
    n = 2 * half
    h = np.zeros(n)
    h[:half] = y
    h[half:] = 1.0 - y[::-1]
    return h

def full_to_antisym(h):
    """Extract anti-symmetric half from full vector."""
    n = len(h)
    half = n // 2
    y = (h[:half] + 1.0 - h[half:][::-1]) / 2.0
    return np.clip(y, 0.0, 1.0)

def active_set_descent(h, max_iters=200, n_active=12, alpha_init=0.005):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()

        G_a = np.zeros((len(active), n))
        for r, m in enumerate(active):
            d = m - (n - 1)

            elif d > 0:
                G_a[r, :n-d] += dx * (1.0 - h[d:n])
                G_a[r, d:n] += -dx * h[0:n-d]
            else:
                dp = -d
                G_a[r, dp:n] += dx * (1.0 - h[0:n-dp])
                G_a[r, 0:n-dp] += -dx * h[dp:n]
        w = np.ones(len(active)) / len(active)
        grad = G_a.T @ w
        grad -= grad.mean()
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad /= gnorm
        improved = False
        for a in [alpha, alpha*0.5, alpha*0.25, alpha*0.1]:

            s = exact_score(h_try)

                best_c = s
                improved = True
                break
            elif s < c - 1e-14:
                h = h_try
                improved = True
                break
        if not improved:

def lse_descent(h, beta=80.0, max_iters=150, alpha_init=0.004):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best_c = exact_score(h)
    alpha = alpha_init
    stuck = 0
    for it in range(max_iters):
        profile = compute_profile(h)
        scaled = beta * (profile - float(np.max(profile)))
        weights = np.exp(scaled)
        wsum = weights.sum()
        if wsum < 1e-300:
            break
        weights /= wsum
        # Build full gradient matrix
        M = 2 * n - 1
        G = np.zeros((M, n))
        for m in range(M):
            d = m - (n - 1)
            if d == 0:
                G[m, :] = dx * (1.0 - 2.0 * h)
            elif d > 0:
                G[m, :n-d] += dx * (1.0 - h[d:n])
                G[m, d:n] += -dx * h[0:n-d]
            else:
                dp = -d
                G[m, dp:n] += dx * (1.0 - h[0:n-dp])
                G[m, 0:n-dp] += -dx * h[dp:n]
        grad = G.T @ weights
        grad -= grad.mean()
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad /= gnorm
        improved = False
        for a in [alpha, alpha*0.5, alpha*0.25, alpha*0.1]:

            s = exact_score(h_try)
            if s < best_c - 1e-14:

                improved = True
                alpha = min(alpha * 1.1, 0.02)

                break

def slsqp_antisym(h0, maxiter=200):
    """SLSQP on anti-symmetric parameterization."""
    n = len(h0)
    half = n // 2
    dx = 2.0 / n
    M = 2 * n - 1
    
    y = full_to_antisym(h0)
    h_check = antisym_to_full(y)
    c0 = exact_score(h_check)
    
    def obj(z):
        h = antisym_to_full(z)
        h = np.clip(h, 0, 1)
        return z[half]  # epigraph variable
    
    def obj_jac(z):
        g = np.zeros(half + 1)
        g[half] = 1.0
        return g
    
    def ineq_fun(z):
        h = antisym_to_full(z)
        h = np.clip(h, 0, 1)
        profile = np.correlate(h, 1 - h, mode="full") * dx
        return z[half] - profile
    
    def ineq_jac(z):
        h = antisym_to_full(z)
        h = np.clip(h, 0, 1)
        # Gradient of F_d w.r.t. y
        # h[i] = y[i] for i < half
        # h[half + i] = 1 - y[half-1-i]
        # So dh[i]/dy[j] = delta(i,j) for i < half
        # dh[half+i]/dy[j] = -delta(half-1-i, j)
        J = np.zeros((M, half + 1))
        for m in range(M):
            d = m - (n - 1)
            row = np.zeros(n)
            if d == 0:
                row[:] = dx * (1.0 - 2.0 * h)
            elif d > 0:
                row[:n-d] += dx * (1.0 - h[d:n])
                row[d:n] += -dx * h[0:n-d]
            else:
                dp = -d
                row[dp:n] += dx * (1.0 - h[0:n-dp])
                row[0:n-dp] += -dx * h[dp:n]
            # Map from h-gradient to y-gradient
            # y[j] appears in h[j] as +1 and in h[2*half-1-j] = h[n-1-j] as -1
            y_grad = np.zeros(half)
            for j in range(half):
                y_grad[j] = row[j] - row[n-1-j]
            J[m, :half] = y_grad
            J[m, half] = 1.0
        return J
    
    x0 = np.concatenate([y, [c0 + 1e-3]])
    bounds = [(0.0, 1.0)] * half + [(0.0, c0 + 0.5)]
    cons = [{'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}]
    
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None and np.isfinite(res.x).all():
            h_new = antisym_to_full(res.x[:half])
            h_new = np.clip(h_new, 0, 1)
            c_new = exact_score(h_new)
            if c_new <= c0 + 1e-9:
                return h_new, c_new
    except Exception:
        pass
    return None, None

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            add = min(remaining, capacity[order[i]])
            if add > 0:
                h[order[i]] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            rem = min(remaining, h[order[i]])
            if rem > 0:
                h[order[i]] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-14:
        idx = int(np.argmax(1.0 - h)) if diff2 > 0 else int(np.argmin(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)

    no_improve = 0

        active_top = active_sorted[:min(6, len(active_top) if 'active_top' in dir() else len(active_sorted))]

                break

        if len(pairs) > pairs_per_round:
            if rng is not None:
                sel = rng.choice(len(pairs), size=pairs_per_round, replace=False)
                pairs = [pairs[k] for k in sel]
            else:
                pairs = pairs[:pairs_per_round]

        improved = False
        for (pi, pj) in pairs:
            if pi == pj:
                continue
            h_new, c_new = pair_line_search(h, pi, pj)
            if c_new < best_c - 1e-12:
                h = h_new
                best_c = c_new
                improved = True
        if not improved and no_improve > 4:
            break

    return h, best_c


def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = np.linspace(0, 2, n + 1)
    xt = np.linspace(0, 2, new_n + 1)
    h_interp = np.zeros(new_n)
    for k in range(new_n):
        lo_k, hi_k = xt[k], xt[k + 1]
        total_mass = 0.0
        total_width = 0.0
        for j in range(n):
            lo_j, hi_j = xs[j], xs[j + 1]
            overlap = min(hi_k, hi_j) - max(lo_k, lo_j)
            if overlap > 0:
                total_mass += h[j] * overlap
                total_width += overlap
        h_interp[k] = total_mass / total_width if total_width > 1e-15 else 0.5
    h_interp = np.clip(h_interp, 0.0, 1.0)
    return repair_feasible(h_interp)


def antisym_to_full(y):
    half = len(y)
    n = 2 * half
    h = np.zeros(n)
    h[:half] = y
    h[half:] = 1.0 - y[::-1]
    return h


def full_to_antisym(h):
    n = len(h)
    half = n // 2
    y = (h[:half] + 1.0 - h[half:][::-1]) / 2.0
    return np.clip(y, 0.0, 1.0)


def antisym_profile(y):
    n = 2 * len(y)
    dx = 2.0 / n
    h = antisym_to_full(y)
    h = np.clip(h, 0.0, 1.0)
    return np.correlate(h, 1 - h, mode="full") * dx, h


def lse_descent(h, beta=80.0, max_iters=120, alpha_init=0.004):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best_c = exact_score(h)
    alpha = alpha_init
    stuck = 0
    for it in range(max_iters):
        profile = compute_profile(h)
        scaled = beta * (profile - float(np.max(profile)))
        weights = np.exp(scaled)
        wsum = weights.sum()
        if wsum < 1e-300:
            break
        weights /= wsum
        M = 2 * n - 1
        G = np.zeros((M, n))
        for m in range(M):
            d = m - (n - 1)
            if d == 0:
                G[m, :] = dx * (1.0 - 2.0 * h)
            elif d > 0:
                G[m, :n-d] += dx * (1.0 - h[d:n])
                G[m, d:n] += -dx * h[0:n-d]
            else:
                dp = -d
                G[m, dp:n] += dx * (1.0 - h[0:n-dp])
                G[m, 0:n-dp] += -dx * h[dp:n]
        grad = G.T @ weights
        grad -= grad.mean()
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad /= gnorm
        improved = False
        for a in [alpha, alpha * 0.5, alpha * 0.25, alpha * 0.1]:
            h_try = repair_feasible(h - a * grad)
            s = exact_score(h_try)
            if s < best_c - 1e-14:
                h = h_try
                best_c = s
                improved = True
                alpha = min(alpha * 1.08, 0.02)
                stuck = 0
                break
        if not improved:
            alpha *= 0.5
            stuck += 1
            if stuck > 30 or alpha < 1e-9:
                break
    return h, best_c


def active_set_descent(h, max_iters=150, n_active=12, alpha_init=0.004):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best_c = exact_score(h)
    alpha = alpha_init
    stuck = 0
    for it in range(max_iters):
        profile = compute_profile(h)
        c = float(np.max(profile))
        if c + 1e-14 < best_c:
            best_c = c
            stuck = 0
            alpha = min(alpha * 1.08, 0.025)
        margin = max(0.002, 4.0 * (c - float(np.median(profile))))
        active = np.where(profile >= c - margin)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(profile))])
        if len(active) > n_active:
            top_idx = np.argsort(profile[active])[-n_active:]
            active = active[top_idx]
        G_a = np.zeros((len(active), n))
        for r, m in enumerate(active):
            d = m - (n - 1)
            if d == 0:
                G_a[r, :] = dx * (1.0 - 2.0 * h)
            elif d > 0:
                G_a[r, :n-d] += dx * (1.0 - h[d:n])
                G_a[r, d:n] += -dx * h[0:n-d]
            else:
                dp = -d
                G_a[r, dp:n] += dx * (1.0 - h[0:n-dp])
                G_a[r, 0:n-dp] += -dx * h[dp:n]
        w = np.ones(len(active)) / len(active)
        grad = G_a.T @ w
        grad -= grad.mean()
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad /= gnorm
        improved = False
        for a in [alpha, alpha * 0.5, alpha * 0.25, alpha * 0.1]:
            h_try = repair_feasible(h - a * grad)
            s = exact_score(h_try)
            if s < best_c - 1e-14:
                h = h_try
                best_c = s
                improved = True
                break
        if not improved:
            alpha *= 0.55
            stuck += 1
            if stuck > 40 or alpha < 1e-9:
                break
    return h, best_c


def slsqp_antisym(h0, maxiter=250):
    n = len(h0)
    half = n // 2
    dx = 2.0 / n
    M = 2 * n - 1
    y = full_to_antisym(h0)
    h_check = np.clip(antisym_to_full(y), 0, 1)
    c0 = exact_score(h_check)

    def obj(z):
        return z[half]

    def obj_jac(z):
        g = np.zeros(half + 1)
        g[half] = 1.0
        return g

    def ineq_fun(z):
        h = np.clip(antisym_to_full(z), 0, 1)
        profile = np.correlate(h, 1 - h, mode="full") * dx
        return z[half] - profile

    def ineq_jac(z):
        h = np.clip(antisym_to_full(z), 0, 1)
        J = np.zeros((M, half + 1))
        for m in range(M):
            d = m - (n - 1)
            row = np.zeros(n)
            if d == 0:
                row[:] = dx * (1.0 - 2.0 * h)
            elif d > 0:
                row[:n-d] += dx * (1.0 - h[d:n])
                row[d:n] += -dx * h[0:n-d]
            else:
                dp = -d
                row[dp:n] += dx * (1.0 - h[0:n-dp])
                row[0:n-dp] += -dx * h[dp:n]
            y_grad = np.zeros(half)
            for jj in range(half):
                y_grad[jj] = row[jj] - row[n - 1 - jj]
            J[m, :half] = y_grad
            J[m, half] = 1.0
        return J

    x0 = np.concatenate([y, [c0 + 1e-3]])
    bounds = [(0.0, 1.0)] * half + [(0.0, c0 + 0.5)]
    cons = [{'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}]
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None and np.isfinite(res.x).all():
            h_new = np.clip(antisym_to_full(res.x[:half]), 0, 1)
            c_new = exact_score(h_new)
            if c_new <= c0 + 1e-9:
                return h_new, c_new
    except Exception:
        pass
    return None, None


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()
    rng = np.random.RandomState(seed)
    np.random.seed(seed)

    def time_left():
        return budget_s + t_start - time.perf_counter()

    parent = None
    try:
        parent = np.array(initial_h_values, dtype=np.float64)
    except NameError:
        pass
    except Exception:
        pass
    if parent is None or len(parent) < 4:
        parent = np.full(70, 0.5)

    parent = np.clip(parent, 0.0, 1.0)
    parent = repair_feasible(parent)
    n_parent = len(parent)

    best_h = parent.copy()
    best_c = exact_score(parent)

    # Stage 1: Pair line descent (primary method)
    if time_left() > 100:
        h_p, c_p = active_pair_descent(best_h, max_rounds=60, pairs_per_round=18, rng=rng)
        if c_p < best_c - 1e-13:
            best_h, best_c = h_p, c_p

    # Stage 2: LSE + active set polish
    if time_left() > 100:
        h_lse, c_lse = lse_descent(best_h, beta=70, max_iters=80)
        if c_lse < best_c - 1e-13:
            best_h, best_c = h_lse, c_lse
        h_ase, c_ase = active_set_descent(best_h, max_iters=80, n_active=10)
        if c_ase < best_c - 1e-13:
            best_h, best_c = h_ase, c_ase
        h_p, c_p = active_pair_descent(best_h, max_rounds=40, pairs_per_round=15, rng=rng)
        if c_p < best_c - 1e-13:
            best_h, best_c = h_p, c_p

    # Stage 3: Multi-resolution exploration
    if time_left() > 120:
        dims = set()
        for factor in [0.7, 0.75, 0.8, 0.85, 0.9, 0.95, 1.0, 1.05, 1.1, 1.15, 1.25, 1.4]:
            d = int(round(n_parent * factor / 2)) * 2
            d = max(30, min(d, 180))
            dims.add(d)
        dims.discard(n_parent)
        dims_list = sorted(dims, key=lambda x: abs(x - n_parent))
        for new_n in dims_list[:6]:
            if time_left() < 40:
                break
            h_interp = interpolate_to_n(best_h, new_n)
            h_p, c_p = active_pair_descent(h_interp, max_rounds=35, pairs_per_round=14, rng=rng)
            if c_p < best_c - 1e-13:
                best_h, best_c = h_p, c_p
            h_lse, c_lse = lse_descent(best_h, beta=90, max_iters=40)
            if c_lse < best_c - 1e-13:
                best_h, best_c = h_lse, c_lse
            h_ase, c_ase = active_set_descent(best_h, max_iters=40, n_active=10)
            if c_ase < best_c - 1e-13:
                best_h, best_c = h_ase, c_ase

    # Stage 4: Anti-symmetric SLSQP
    if time_left() > 70:
        h_sol, c_sol = slsqp_antisym(best_h, maxiter=250)
        if h_sol is not None and c_sol < best_c - 1e-12:
            best_h, best_c = h_sol, c_sol
            h_p, c_p = active_pair_descent(best_h, max_rounds=30, pairs_per_round=12, rng=rng)
            if c_p < best_c - 1e-13:
                best_h, best_c = h_p, c_p

    # Stage 5: Extended pair descent
    if time_left() > 70:
        h_p, c_p = active_pair_descent(best_h, max_rounds=80, pairs_per_round=25, rng=rng)
        if c_p < best_c - 1e-13:
            best_h, best_c = h_p, c_p

    # Stage 6: Perturbation restarts
    if time_left() > 80:
        n_b = len(best_h)
        for eps in [0.004, 0.008, 0.015, 0.025]:
            if time_left() < 25:
                break
            z = rng.randn(n_b)
            z -= z.mean()
            h_pert = repair_feasible(best_h + eps * z)
            h_p, c_p = active_pair_descent(h_pert, max_rounds=35, pairs_per_round=14, rng=rng)
            if c_p < best_c - 1e-12:
                best_h, best_c = h_p, c_p
                h_ase, c_ase = active_set_descent(best_h, max_iters=30, n_active=10)
                if c_ase < best_c - 1e-13:
                    best_h, best_c = h_ase, c_ase

    # Stage 7: LSE with various betas
    for beta in [40, 120, 200]:
        if time_left() < 18:
            break
        h_lse, c_lse = lse_descent(best_h, beta=beta, max_iters=60)
        if c_lse < best_c - 1e-13:
            best_h, best_c = h_lse, c_lse
            h_p, c_p = active_pair_descent(best_h, max_rounds=20, pairs_per_round=12, rng=rng)
            if c_p < best_c - 1e-13:
                best_h, best_c = h_p, c_p

    # Stage 8: Final resolution adjust
    if time_left() > 40:
        n_final = len(best_h)
        for dn in [-4, -2, 2, 4, 6]:
            if time_left() < 15:
                break
            new_n = n_final + dn
            if new_n < 30 or new_n > 180 or new_n % 2 != 0:
                continue
            h_interp = interpolate_to_n(best_h, new_n)
            h_p, c_p = active_pair_descent(h_interp, max_rounds=25, pairs_per_round=12, rng=rng)
            if c_p < best_c - 1e-13:
                best_h, best_c = h_p, c_p
            h_ase, c_ase = active_set_descent(best_h, max_iters=30, n_active=8)
            if c_ase < best_c - 1e-13:
                best_h, best_c = h_ase, c_ase

    # Stage 9: Final SLSQP + pair
    if time_left() > 40:
        h_sol, c_sol = slsqp_antisym(best_h, maxiter=300)
        if h_sol is not None and c_sol < best_c - 1e-14:
            best_h, best_c = h_sol, c_sol
        h_p, c_p = active_pair_descent(best_h, max_rounds=40, pairs_per_round=20, rng=rng)
        if c_p < best_c - 1e-14:
            best_h, best_c = h_p, c_p

    # Stage 10: Final active set polish
    if time_left() > 15:
        h_ase, c_ase = active_set_descent(best_h, max_iters=100, n_active=10, alpha_init=0.002)
        if c_ase < best_c - 1e-14:
            best_h, best_c = h_ase, c_ase
        h_p, c_p = active_pair_descent(best_h, max_rounds=20, pairs_per_round=15, rng=rng)
        if c_p < best_c - 1e-14:
            best_h, best_c = h_p, c_p

    # Final validation
    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)
    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            final_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 2:
            final_c = float(eval_result[0])
    except Exception:
        pass

    return best_h, final_c, len(best_h)
