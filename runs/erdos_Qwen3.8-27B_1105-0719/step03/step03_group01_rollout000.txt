import numpy as np
from scipy.optimize import linprog
import time


def project_feasible(x, target_sum):
    n = len(x)
    x = np.clip(x.astype(float), -5.0, 5.0)
    lo, hi = -5.0, 5.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        cnt = mask.sum()
        if cnt > 0:
            h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def c5_value(h, dx):
    return float(compute_corr(h).max() * dx)


def lse_obj_and_grad(h, beta, n):
    corr = compute_corr(h)
    mc = corr.max()
    shifted = beta * (corr - mc)
    shifted = np.clip(shifted, -400, 400)
    exp_s = np.exp(shifted)
    s = exp_s.sum()
    w = exp_s / s
    obj = (mc + np.log(s) / beta) * (2.0 / n)
    grad = np.zeros(n)
    for p in range(len(corr)):
        wp = w[p]
        if wp < 1e-14:
            continue
        k = p - (n - 1)
        i1s = max(0, k)
        i1e = min(n, k + n)
        if i1s < i1e:
            idx = np.arange(i1s, i1e)
            grad[idx] += wp * (1.0 - h[idx - k])
        i2s = max(0, -k)
        i2e = min(n, n - k)
        if i2s < i2e:
            idx2 = np.arange(i2s, i2e)
            grad[idx2] += wp * (-h[idx2 + k])
    return obj, grad


def lse_projected_opt(h_init, dx, target_sum, beta, t0, max_time, max_iter=3000):
    n = len(h_init)
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = c5_value(h, dx)
    step = 0.5
    no_improve = 0
    for it in range(max_iter):
        if time.time() - t0 > max_time:
            break
        obj, grad = lse_obj_and_grad(h, beta, n)
        # Project gradient: remove component along (1,1,...,1)
        grad_proj = grad - np.mean(grad)
        # Backtracking line search
        found = False
        for _ in range(30):
            h_new = project_feasible(h - step * grad_proj, target_sum)
            c5_new = c5_value(h_new, dx)
            if c5_new < best_c5:
                h = h_new
                best_h = h_new.copy()
                best_c5 = c5_new
                no_improve = 0
                step = min(step * 1.1, 2.0)
                found = True
                break
            step *= 0.7
        if not found:
            no_improve += 1
            if no_improve > 100:
                break
    return best_h, best_c5


def lp_step_active(h, dx, target_sum, n_active, alpha):
    n = len(h)
    corr = compute_corr(h)
    n_total = 2 * n - 1
    current_c5 = corr.max() * dx
    if n_active >= n_total:
        active_idx = np.arange(n_total)
    else:
        active_idx = np.argsort(corr)[-n_active:]
    m = len(active_idx)
    # Build gradient matrix
    G = np.zeros((m, n))
    for row, p in enumerate(active_idx):
        k = p - (n - 1)
        i1s = max(0, k)
        i1e = min(n, k + n)
        if i1s < i1e:
            idx = np.arange(i1s, i1e)
            G[row, idx] += (1.0 - h[idx - k])
        i2s = max(0, -k)
        i2e = min(n, n - k)
        if i2s < i2e:
            idx2 = np.arange(i2s, i2e)
            G[row, idx2] += (-h[idx2 + k])
    # LP: min tau s.t. G*delta - tau*1 <= -corr[active], 1'*delta = 0, |delta_i| <= alpha*(bound)
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_ub = np.empty((m, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -corr[active_idx]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for i in range(n):
        lo_b = max(-h[i] * alpha, -alpha)
        hi_b = min((1.0 - h[i]) * alpha, alpha)
        bounds.append((lo_b, hi_b))
    bounds.append((-np.inf, np.inf))
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status != 0:
            return h, current_c5, False
        delta = res.x[:n]
    except Exception:
        return h, current_c5, False
    if np.abs(delta).max() < 1e-15:

    best_s = 0.0

        c5_new = c5_value(h_new, dx)

            best_c5 = c5_new
            best_s = s
            best_h = h_new
            if s == 1.0:
                break
    return best_h, best_c5, best_s > 0


def lp_optimize(h, dx, target_sum, t0, max_time, alpha_start=0.2, n_active_frac=0.35):
    n = len(h)
    n_total = 2 * n - 1
    n_active = max(10, int(n_active_frac * n_total))
    best_h = h.copy()

    alpha = alpha_start
    no_improve = 0
    while time.time() - t0 < max_time:
        if alpha < 0.002:
            break
        h_new, c5_new, improved = lp_step_active(h, dx, target_sum, n_active, alpha)
        if improved and c5_new < best_c5 - 1e-14:

            best_c5 = c5_new
            no_improve = 0

            elif no_improve == 60:
                n_active = max(15, int(n_total * 0.3))
                alpha = max(alpha, 0.18)
            if no_improve > 120:
                break
    return best_h, best_c5


def sa_quantized(h_init, n, M, max_iters, seed, t0, max_time):
