import numpy as np
from scipy.optimize import linprog, minimize
import time

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.argsort(1.0 - h)
        for i in idx:
            cap = 1.0 - h[i]
            add = min(diff, cap)
            h[i] += add
            diff -= add
            if diff <= 1e-14:
                break
    else:
        idx = np.argsort(h)
        for i in idx:
            rem = min(-diff, h[i])
            h[i] -= rem
            diff += rem
            if diff >= -1e-14:
                break
    # Final fix
    r = target - h.sum()
    if abs(r) > 1e-15:
        if r > 0:
            i = int(np.argmax(1.0 - h))
        else:
            i = int(np.argmax(h))
        h[i] = np.clip(h[i] + r, 0.0, 1.0)
    return h

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def grad_row(h, m, n):
    dx = 2.0 / n
    d = m - (n - 1)
    if d == 0:
        return dx * (1.0 - 2.0 * h).copy()
    elif d > 0:
        g = np.zeros(n)
        g[d:] += dx * (1.0 - h[:-d])
        g[:-d] -= dx * h[d:]
        return g
    else:
        p = -d
        g = np.zeros(n)
        g[:-p] += dx * (1.0 - h[p:])
        g[p:] -= dx * h[:-p]
        return g

def lp_descent(h, delta, top_k=20):
    """LP trust region step. Returns (direction, predicted_tau, current_max) or (None, None, current_max)."""
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    profile = compute_profile(h)
    cur_max = float(profile.max())
    
    # Select top-k active lags
    active = np.argsort(profile)[-top_k:]
    
    # Build LP: min tau s.t. G_m · d - tau <= -P_m, sum(d)=0, bounds on d
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    n_active = len(active)
    A_ub = np.zeros((n_active, n + 1))
    b_ub = np.zeros(n_active)
    for idx, m in enumerate(active):
        A_ub[idx, :n] = grad_row(h, int(m), n)
        A_ub[idx, -1] = -1.0
        b_ub[idx] = -profile[int(m)]
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    bounds = []
    for i in range(n):
        lo = max(-delta, -h[i] + 1e-12)
        hi = min(delta, 1.0 - h[i] - 1e-12)
        if lo >= hi:
            lo, hi = 0.0, 0.0
        bounds.append((lo, hi))
    bounds.append((0.0, cur_max + 1.0))
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs-ds')
        if res.success and res.x is not None:
            d = res.x[:n]
            tau_pred = float(res.x[-1])
            return d, tau_pred, cur_max
    except Exception:
        pass
    return None, None, cur_max

def line_search(h, d, cur_score, max_alpha=1.0):
    """Backtracking line search. Returns (best_alpha, best_h, best_score)."""
    best_alpha = 0.0
    best_h = h.copy()
    best_val = cur_score
    
    # Try a sequence of alphas
    alphas = [max_alpha, 0.8*max_alpha, 0.6*max_alpha, 0.4*max_alpha,
              0.25*max_alpha, 0.15*max_alpha, 0.1*max_alpha, 0.05*max_alpha, 0.02*max_alpha]
    
    for a in alphas:
        h_try = repair_feasible(h + a * d)
        c_try = exact_score(h_try)
        if c_try < best_val - 1e-15:
            best_val = c_try
            best_alpha = a
            best_h = h_try
            if c_try < cur_score * 0.999:  # significant improvement, stop early
                break
    
    return best_alpha, best_h, best_val

def coordinate_descent(h, max_sweeps=50, tol=1e-14):
    """Coordinate descent with exact line search per coordinate (mass-preserving pair)."""
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    
    for sweep in range(max_sweeps):
        improved = False
        # Try each coordinate: move mass to/from the "best" other coordinate
        # Simpler: for each i, try adjusting h[i] and compensating on h[j]
        # Even simpler: for each i, find best h[i] with all else fixed
        # But that doesn't preserve sum... so use pairs.
        
        # Pairwise: for each i, find j that most benefits from exchange
        profile = compute_profile(h)
        # Find the lag with max profile
        m_star = int(np.argmax(profile))
        g_star = grad_row(h, m_star, n)
        
        # For each pair (i, j) with h[i] > h[j], try exchange
        # Sort by "exchange benefit" = g_star[i] - g_star[j] (positive means reducing i helps)
        # Actually, moving mass from i to j changes P_m by -g_star[i]*amt + g_star[j]*amt
        # So we want g_star[i] > g_star[j] for improvement
        
        # Try top pairs
        idx = np.argsort(g_star)  # most negative first (benefit from gaining mass)
        for attempt in range(min(n, 30)):
            i = int(idx[attempt])  # candidate to give mass
            # Find j to receive mass
            j_candidates = np.argsort(g_star)  # most negative g means most benefit from receiving
            j = int(j_candidates[attempt])
            if i == j:
                continue
            if h[i] <= h[j]:
                continue
            # Try transferring mass from i to j
            max_transfer = (h[i] - h[j]) * 0.5
            if max_transfer < 1e-8:
                continue
            for fraction in [1.0, 0.7, 0.5, 0.3, 0.1]:
                amt = max_transfer * fraction
                h_try = h.copy()
                h_try[i] -= amt
                h_try[j] += amt
                c_try = exact_score(h_try)
                if c_try < best_c - tol:
                    h = h_try
                    best_c = c_try
                    improved = True
                    break
            if improved:
                break
        
        if not improved:
            break
    
    return h, best_c

def active_subgradient_descent(h, max_iters=200, tol=1e-14):
    """Subgradient method with adaptive step for the max function."""
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    step = 0.05
    momentum = np.zeros(n)
    mu = 0.3
    
    for it in range(max_iters):
        profile = compute_profile(h)
        c = float(profile.max())
        
        # Weighted subgradient (Tikhonov-like)
        weights = np.exp(50.0 * (profile - c) / max(np.std(profile), 1e-10))
        weights /= weights.sum()
        
        g = np.zeros(n)
        for m in range(len(profile)):
            if weights[m] > 1e-8:
                g += weights[m] * grad_row(h, m, n)
        
        # Remove mean for mass preservation
        g -= g.mean()
        
        # Momentum
        momentum = mu * momentum + g
        d = -momentum
        d -= d.mean()  # ensure mass preservation
        
        dnorm = np.linalg.norm(d)
        if dnorm < 1e-15:
            break
        d = d / dnorm
        
        # Line search
        best_alpha = 0.0
        best_h = h
        best_val = c
        for alpha in [step, 0.8*step, 0.6*step, 0.4*step, 0.2*step, 0.1*step]:
            h_try = repair_feasible(h + alpha * d)
            c_try = exact_score(h_try)
            if c_try < best_val - tol:
                best_val = c_try
                best_alpha = alpha
                best_h = h_try
                break
        
        if best_alpha > 0:
            h = best_h
            if best_val < best_c:
                best_c = best_val
            step = min(step * 1.1, 0.2)
        else:
            step *= 0.7
            if step < 0.001:
                break
    
    return h, best_c

def interp_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    edges_old = np.linspace(0.0, 2.0, n + 1)
    edges_new = np.linspace(0.0, 2.0, new_n + 1)
    h_new = np.zeros(new_n)
    for k in range(new_n):
        a, b = edges_new[k], edges_new[k+1]
        total = 0.0
        for j in range(n):
            lo = max(a, edges_old[j])
            hi = min(b, edges_old[j+1])
            if hi > lo:
                total += h[j] * (hi - lo)
        h_new[k] = total / (b - a)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

def slsqp_optimize(h_start, maxiter=300):
    """SLSQP with epigraph formulation."""
    n = len(h_start)
    dx = 2.0 / n
    M = 2 * n - 1
    c0 = exact_score(h_start)
    
    def obj(x):
        return x[-1]
    
    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    
    def con_fun(x):
        h = x[:-1]
        t = x[-1]
        profile = np.correlate(h, 1 - h, mode="full") * dx
        return t - profile
    
    def con_jac(x):
        h = x[:-1]
        J = np.zeros((M, n + 1))
        for m in range(M):
            J[m, :n] = -grad_row(h, m, n)
        J[:, -1] = 1.0
        return J
    
    def eq_fun(x):
        return x[:-1].sum() - n / 2.0
    
    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g
    
    x0 = np.concatenate([h_start, [c0 + 1e-4]])
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 1.0)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': con_fun, 'jac': con_jac}
    ]
    
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,

        if res.x is not None and np.isfinite(res.x).all():

            c_new = exact_score(h_new)
            if c_new <= c0 + 1e-10:
                return h_new, c_new
    except Exception:
        pass
    return None, c0

    noise -= noise.mean()
    h_new = h + noise
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = repair_feasible(h_new)
    return h_new
