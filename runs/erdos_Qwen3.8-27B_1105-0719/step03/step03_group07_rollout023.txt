def golden_section_min(f, a, b, tol=1e-12, max_iter=100):
    """Minimize convex f on [a,b] using golden section search."""
    invphi = (np.sqrt(5) - 1) / 2  # ~0.618
    c = b - invphi * (b - a)
    d = a + invphi * (b - a)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if b - a < tol:
            break
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - invphi * (b - a)
            fc = f(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + invphi * (b - a)
            fd = f(d)
    return f((a + b) / 2), (a + b) / 2

    G = np.zeros((len(rows), n))

        cand_dims.sort(key=lambda d: abs(d - n_best))
        
        for new_n in cand_dims[:10]:
            if not time_ok(30):
                break
            h_interp = interpolate_to_n(best_h, new_n)
            # Quick pairwise at new dimension
            h_pw, c_pw = pairwise_sweep(h_interp, max_sweeps=15, t_start=t_start,
                                        budget_s=budget_s, margin=20)
            c = c_pw
            if c < best_c - 1e-11:
                # SLSQP polish at this dimension
                sol = slsqp_active(h_pw, maxiter=250, active_k=20, tol=1e-12)
                if sol is not None:
                    c_sq = exact_score(sol)
                    if c_sq < c:
                        h_pw = sol
                        c = c_sq
                # More pairwise
                h_pw2, c_pw2 = pairwise_sweep(h_pw, max_sweeps=15, t_start=t_start,
                                              budget_s=budget_s, margin=15)
                if c_pw2 < c:
                    h_pw = h_pw2
                    c = c_pw2
                if c < best_c - 1e-11:
                    best_h = h_pw
                    best_c = c
    
    # Phase 5: Binary-inspired starts
    if time_ok(100):
        n_b = len(best_h)
        for trial in range(4):
            if not time_ok(30):
                break
            h_bin = make_binary_start(n_b, np.random.RandomState(seed + 200 + trial))
            # Interpolate between binary and current best
            for mix in [0.3, 0.5, 0.7]:
                if not time_ok(20):
                    break
                h_mix = repair_feasible(mix * h_bin + (1 - mix) * best_h)
                h_pw, c_pw = pairwise_sweep(h_mix, max_sweeps=15, t_start=t_start,
                                            budget_s=budget_s, margin=15)
                if c_pw < best_c - 1e-11:
                    sol = slsqp_active(h_pw, maxiter=200, active_k=20, tol=1e-12)
                    if sol is not None and exact_score(sol) < c_pw:
                        h_pw = sol
                        c_pw = exact_score(h_pw)
                    if c_pw < best_c - 1e-11:
                        best_h = h_pw
                        best_c = c_pw
    
    # Phase 6: Final intensive refinement
    if time_ok(60):
        h_pw, c_pw = pairwise_sweep(best_h, max_sweeps=40, t_start=t_start,
                                    budget_s=budget_s, margin=20)
        if c_pw < best_c - 1e-14:
            best_h = h_pw
            best_c = c_pw
        if time_ok(30):
            sol = slsqp_active(best_h, maxiter=400, active_k=25, tol=1e-13)
            if sol is not None:
                c = exact_score(sol)
                if c < best_c - 1e-14:
                    best_h = sol
                    best_c = c
                    h_pw, c_pw = pairwise_sweep(best_h, max_sweeps=30, t_start=t_start,
                                                budget_s=budget_s, margin=10)
                    if c_pw < best_c - 1e-14:
                        best_h = h_pw
                        best_c = c_pw
    
    # Phase 7: Very fine pairwise (high iteration count)
    if time_ok(30):
        n = len(best_h)
        h = best_h.copy()
        best_c2 = exact_score(h)
        # Focus on pairs near the center (most impactful)
        center = n // 2
        for sweep in range(3):
            if not time_ok(10):
                break
            for p in range(center - 10, center + 11):
                if p < 0 or p >= n:
                    continue
                for q in range(max(0, p - 15), min(n, p + 16)):
                    if p == q:
                        continue
                    t_lo = max(-h[p], h[q] - 1.0)
                    t_hi = min(1.0 - h[p], h[q])
                    if t_hi - t_lo < 1e-12:
                        continue
                    t_best, f_best = golden_min_pair(h, p, q, max_iter=50)
                    if f_best < best_c2 - 1e-14:
                        h[p] += t_best
                        h[q] -= t_best
                        h = np.clip(h, 0.0, 1.0)
                        h = repair_feasible(h)
                        best_c2 = f_best
        if best_c2 < best_c - 1e-14:
            best_h = h
            best_c = best_c2
    
    # Final: try symmetrized version
    if time_ok(10):
        h_sym = repair_feasible(symmetrize(best_h))
        c_sym = exact_score(h_sym)
        if c_sym < best_c - 1e-14:
            best_h = h_sym
            best_c = c_sym
    
    # Final validation
    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)
    
    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            final_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 1:
            if np.isfinite(eval_result[0]):
                final_c = float(eval_result[0])
    except Exception:
        pass
    
    return best_h, final_c, len(best_h)
