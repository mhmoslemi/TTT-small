import numpy as np
from scipy.optimize import minimize, linprog
import cvxpy as cp
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def grad_matrix(h):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    G = np.zeros((M, n))
    G[n-1, :] = dx * (1.0 - 2.0 * h)
    for d in range(1, n):
        G[n-1+d, d:n] += dx * (1.0 - h[0:n-d])
        G[n-1+d, 0:n-d] += -dx * h[d:n]
        G[n-1-d, 0:n-d] += dx * (1.0 - h[d:n])
        G[n-1-d, d:n] += -dx * h[0:n-d]
    return G

def project_feasible(y):
    n = len(y)
    target = n / 2.0
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    result = np.clip(y - lam, 0.0, 1.0)
    return result

def repair_feasible(h):
    h = np.clip(h.astype(np.float64), 0.0, 1.0)
    n = len(h)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, 1.0 - h[idx])
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)[::-1]
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        idx = int(np.argmax(1.0 - h)) if diff2 > 0 else int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    return repair_feasible(np.clip(h_new, 0.0, 1.0))

def lse_obj_and_grad(x, beta):
    h = np.clip(x, 1e-12, 1.0 - 1e-12)
    P = compute_profile(h)
    bP = beta * P
    bP_max = bP.max()
    exp_bP = np.exp(bP - bP_max)
    Z = exp_bP.sum()
    obj = bP_max + np.log(Z)
    weights = exp_bP / Z
    G = grad_matrix(h)
    grad = G.T @ weights
    return obj, grad

def get_active_set(P, top_k, beam_frac=0.06):
    C = P.max()
    spread = C - P.min()
    beam = max(0.001, beam_frac * spread)
    active = np.where(P >= C - beam)[0]
    if len(active) > top_k:
        top = np.argsort(P[active])[-top_k:]
        active = active[top]
    if len(active) < 3:
        active = np.argsort(P)[-min(10, len(P)):]
    return active

def qp_subproblem(h, active_idx, radius, gamma):
    n = len(h)
    P = compute_profile(h)
    G = grad_matrix(h)
    G_a = G[active_idx, :]
    P_a = P[active_idx]
    C = P.max()
    
    delta = cp.Variable(n)
    t = cp.Variable()
    obj = cp.Minimize(t + gamma * cp.sum_squares(delta))
    cons = [
        G_a @ delta <= t - P_a,
        cp.sum(delta) == 0,
        delta >= -np.minimum(h, radius),
        delta <= np.minimum(1.0 - h, radius),
    ]
    prob = cp.Problem(obj, cons)
    for solver in [cp.ECOS, cp.SCS, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if prob.status in ['optimal', 'optimal_inaccurate']:
                if delta.value is not None:
                    d = np.array(delta.value).flatten()
                    if np.isfinite(d).all() and np.abs(d).max() < radius + 1e-6:
                        return d, float(t.value) if t.value is not None else C
        except Exception:
            continue
    return None, None

def lp_fallback(h, active_idx, radius):
    n = len(h)
    dx = 2.0 / n
    P = compute_profile(h)
    G = grad_matrix(h)
    G_a = G[active_idx, :]
    P_a = P[active_idx]
    C = P.max()
    
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_ub = np.zeros((len(active_idx), n + 1))
    A_ub[:, :n] = G_a
    A_ub[:, -1] = -1.0
    b_ub = -(P_a - C - 1e-10)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    lower = np.maximum(-h, -radius)
    upper = np.minimum(1.0 - h, radius)
    bounds = list(zip(lower, upper))
    bounds.append((0.0, None))
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'maxiter': 2000})
        if res.success and res.x is not None:
            d = res.x[:n]
            if np.isfinite(d).all():
                return d
    except Exception:
        pass
    return None

def gradient_direction(h, active_idx, P):
    G = grad_matrix(h)
    G_a = G[active_idx, :]
    P_a = P[active_idx]
    C = P.max()
    weights = np.maximum(P_a - (C - 0.1 * (C - P.min())), 0.01)
    weights /= weights.sum()
    grad = G_a.T @ weights
    grad -= grad.mean()
    norm = np.linalg.norm(grad)
    if norm < 1e-15:
        return None
    return grad / norm

def slsqp_lse(h0, beta, t_start, budget_s, max_iter=300):
    n = len(h0)
    def obj_fn(x):
        o, _ = lse_obj_and_grad(x, beta)
        return o
    def jac_fn(x):
        _, g = lse_obj_and_grad(x, beta)
        return g
    def eq_fn(x):
        return [x.sum() - n / 2.0]
    def eq_jac_fn(x):
        return np.ones((1, n))
    
    x0 = h0.copy()
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_fn, 'jac': eq_jac_fn}]
    try:
        res = minimize(obj_fn, x0, method='SLSQP', jac=jac_fn,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-14})
        if res.x is not None and np.isfinite(res.x).all():
            h_sol = repair_feasible(res.x)
            return h_sol
    except Exception:
        pass
    return None

def slsqp_max(h0, t_start, budget_s, max_iter=200):
    n = len(h0)
    M = 2 * n - 1
    def obj_fn(x):
        return exact_score(x)
    def eq_fn(x):
        return [x.sum() - n / 2.0]
    def eq_jac_fn(x):
        return np.ones((1, n))
    x0 = h0.copy()
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_fn, 'jac': eq_jac_fn}]
    try:
        res = minimize(obj_fn, x0, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-15})
        if res.x is not None and np.isfinite(res.x).all():
            h_sol = repair_feasible(res.x)
            return h_sol
    except Exception:
        pass
    return None

def lse_momentum_optimize(h, t_start, budget_s, max_iters=400):

def fine_polish(h, t_start, budget_s, max_iters=150):

        C = P.max()
        active = np.argsort(P)[-8:]
        d, _ = qp_subproblem(h, active, 0.004, 100.0)
        improved = False
        if d is not None and np.abs(d).max() > 1e-12:
            for alpha in [1.0, 0.5, 0.25]:
                h_try = repair_feasible(h + alpha * d)
                s = exact_score(h_try)

                    h = h_try
                    best_c = s

def make_smooth_start(n, rng, mode):
    h = np.zeros(n)
    if mode == 0:
        h[:] = 0.5
        h += 0.25 * np.sin(np.pi * np.arange(n) / n + rng.uniform(0, 2 * np.pi))
    elif mode == 1:
        nb = rng.randint(3, 7)
        block = n // nb
        for b in range(nb):
            h[b*block:(b+1)*block] = rng.uniform(0.25, 0.75)
    elif mode == 2:
        phi = 0.6180339887498949
        for i in range(n):
            h[i] = 0.5 + 0.35 * np.sin(2 * np.pi * ((i * phi) % 1.0))
    elif mode == 3:
        half = n // 2
        h[:half] = rng.uniform(0.15, 0.85, half)
        h[half:] = 1.0 - h[:half][::-1]
    else:
        h[:] = 0.5 + 0.2 * np.sin(2 * np.pi * np.arange(n) / n * rng.randint(1, 4))
    return repair_feasible(h)
