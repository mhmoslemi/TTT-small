def pair_line_search(h, i, j, G, dx):
    n = len(h)
    M = 2*n - 1
    P = np.correlate(h, 1-h, mode="full") * dx
    delta = G[:, i] - G[:, j]
    t_low = max(-h[i], h[j] - 1.0)
    t_high = min(1.0 - h[i], h[j])
    if t_high - t_low < 1e-15:
        return h, float(np.max(P))
    # Grid search
    n_grid = 64
    ts = np.linspace(t_low, t_high, n_grid)
    scores = np.maximum(P + ts[0:1] * delta, axis=1) if n_grid == 1 else None
    # Actually, need to compute max over m for each t
    # P + t * delta has shape (n_grid, M)
    candidates = P[None, :] + np.outer(ts, delta)
    scores = np.max(candidates, axis=1)
    best_idx = np.argmin(scores)
    t_best = ts[best_idx]
    # Refine
    if best_idx > 0 and best_idx < n_grid - 1:
        lo, hi = ts[best_idx-1], ts[best_idx+1]
        for _ in range(30):
            mid1 = lo + (hi - lo) / 3
            mid2 = hi - (hi - lo) / 3
            s1 = float(np.max(P + mid1 * delta))
            s2 = float(np.max(P + mid2 * delta))
            if s1 < s2:
                hi = mid2
            else:
                lo = mid1
        t_best = (lo + hi) / 2
    # Apply
    h_new = h.copy()
    h_new[i] += t_best
    h_new[j] -= t_best
    h_new = np.clip(h_new, 0, 1)
    s_new = exact_score(h_new)
    return h_new, s_new

import numpy as np
from scipy.optimize import minimize
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx] + 1e-15)
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx] + 1e-15)
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-14:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmin(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def build_gradient_matrix(h, n, dx):
    M = 2 * n - 1
    G = np.zeros((M, n))
    for d in range(-n + 1, n):
        m = d + n - 1
        if d == 0:
            G[m, :] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            G[m, d:] += dx * (1.0 - h[0:n-d])
            G[m, 0:n-d] += -dx * h[d:n]
        else:
            dp = -d
            G[m, 0:n-dp] += dx * (1.0 - h[dp:n])
            G[m, dp:n] += -dx * h[0:n-dp]
    return G

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = np.linspace(0, 2, n + 1)
    xt = np.linspace(0, 2, new_n + 1)
    h_interp = np.zeros(new_n)
    for k in range(new_n):
        lo_k, hi_k = xt[k], xt[k+1]
        total_mass = 0.0
        total_width = 0.0
        for j in range(n):
            lo_j, hi_j = xs[j], xs[j+1]
            overlap = min(hi_k, hi_j) - max(lo_k, lo_j)
            if overlap > 0:
                total_mass += h[j] * overlap
                total_width += overlap
        h_interp[k] = total_mass / total_width if total_width > 1e-15 else 0.5
    h_interp = np.clip(h_interp, 0.0, 1.0)
    return repair_feasible(h_interp)

def pair_line_search(h, G, i, j, c_current):
    """Exact 1D minimization along pair (i,j): h[i]+=t, h[j]-=t."""
    n = len(h)
    M = 2 * n - 1
    P = compute_profile(h)
    delta = G[:, i] - G[:, j]
    t_low = max(-h[i], h[j] - 1.0)
    t_high = min(1.0 - h[i], h[j])
    if t_high - t_low < 1e-14:
        return h, c_current
    # Grid search
    n_grid = 50
    ts = np.linspace(t_low, t_high, n_grid)
    # P + t*delta for each t
    candidates = P[None, :] + np.outer(ts, delta)
    scores = np.max(candidates, axis=1)
    best_idx = int(np.argmin(scores))
    t_best = ts[best_idx]
    # Refine with ternary search
    if 0 < best_idx < n_grid - 1:
        lo = ts[best_idx - 1]
        hi = ts[best_idx + 1]
        for _ in range(25):
            mid1 = lo + (hi - lo) / 3.0
            mid2 = hi - (hi - lo) / 3.0
            s1 = float(np.max(P + mid1 * delta))
            s2 = float(np.max(P + mid2 * delta))
            if s1 < s2:
                hi = mid2
            else:
                lo = mid1
        t_best = (lo + hi) / 2.0
    # Validate
    s_at_tbest = float(np.max(P + t_best * delta))
    if s_at_tbest >= c_current - 1e-13:
        return h, c_current
    h_new = h.copy()
    h_new[i] = h[i] + t_best
    h_new[j] = h[j] - t_best
    h_new = np.clip(h_new, 0.0, 1.0)
    # Sum is preserved by construction (h[i]+t + h[j]-t = h[i]+h[j])
    s_new = exact_score(h_new)
    if s_new < c_current - 1e-14:
        return h_new, s_new
    return h, c_current

def lse_descent(h, beta=80.0, max_iters=100, alpha_init=0.004):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best_c = exact_score(h)
    alpha = alpha_init
    stuck = 0
    
    for it in range(max_iters):
        P = compute_profile(h)
        scaled = beta * P
        max_s = np.max(scaled)
        weights = np.exp(scaled - max_s)
        wsum = weights.sum()
        if wsum < 1e-300:
            break
        weights /= wsum
        
        G = build_gradient_matrix(h, n, dx)
        grad = G.T @ weights  # (n,)
        grad -= grad.mean()  # remove constant component (preserves sum)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad /= gnorm
        
        improved = False
        for factor in [1.0, 0.5, 0.25, 0.1, 0.05]:
            a = alpha * factor
            h_try = repair_feasible(h - a * grad)
            s = exact_score(h_try)
            if s < best_c - 1e-14:
                h = h_try
                best_c = s
                improved = True
                alpha = min(alpha * 1.05, 0.02)
                stuck = 0
                break
        if not improved:
            alpha *= 0.5
            stuck += 1
            if stuck > 30 or alpha < 1e-9:
                break
    
    return h, best_c

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx] + 1e-15)
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx] + 1e-15)
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-14:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmin(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def build_gradient_matrix(h, n, dx):
