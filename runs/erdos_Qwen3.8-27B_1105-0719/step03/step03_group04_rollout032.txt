- A boundary point (0 or 1)

We can find it by:

2. For each pair of "active" lines (those that achieve the max at some point), find their intersection
3. The optimal x is the one that gives the smallest max value

```python
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def repair_feasible(h):
    n = len(h)
    h = np.clip(np.asarray(h, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, 1.0 - h[idx])
            h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)[::-1]
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    return h

def build_grad_matrix(h):
    """Build the M x n gradient matrix G where G[k,m] = d(profile_k)/d(h_m)."""
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    G = np.zeros((M, n))
    for k in range(M):
        d = k - (n - 1)  # offset
        if d >= 0:
            # h[m] * (1-h[m+d]) for m = 0..n-1-d
            # d/dh[m] = (1-h[m+d]) for m in range(n-d)
            # Also h[m-d] * (1-h[m]) for m = d..n-1
            # d/dh[m] = -h[m-d] for m in range(d, n)
            for m in range(n - d):
                G[k, m] += dx * (1.0 - h[m + d])
            for m in range(d, n):
                G[k, m] += -dx * h[m - d]
        else:
            e = -d
            # h[m] * (1-h[m-e]) for m = e..n-1
            # d/dh[m] = (1-h[m-e]) for m in range(e, n)
            # Also h[m+e] * (1-h[m]) for m = 0..n-1-e
            # d/dh[m] = -h[m+e] for m in range(n-e)
            for m in range(e, n):
                G[k, m] += dx * (1.0 - h[m - e])
            for m in range(n - e):
                G[k, m] += -dx * h[m + e]
    return G
```

```python
def grad_softmax_beta(h, w, beta_unused):
    """Compute gradient of soft-max objective. w = softmax(beta*profile)."""
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    grad = np.zeros(n)
    
    # For each m, compute:
    # Part 1: sum over k where j1 = m+k-n+1 is valid
    # Part 2: sum over k where i2 = m-k+n-1 is valid
    
    # Vectorized approach:
    # For part 1: for each m, j1 = m+k-n+1 must be in [0, n-1]
    # So k must be in [n-1-m, 2n-2-m], intersected with [0, M-1]
    
    # Let's use the offset d = k-(n-1) formulation
    # For offset d (from -(n-1) to n-1), k = n-1+d
    # Part 1: h[m]*(1-h[m+d]) contributes dx*(1-h[m+d]) to G[n-1+d, m] if 0<=m+d<=n-1
    # Part 2: h[m-d]*(1-h[m]) contributes -dx*h[m-d] to G[n-1+d, m] if 0<=m-d<=n-1
    
    # So: grad[m] = dx * sum_d w[n-1+d] * (1-h[m+d]) * I[0<=m+d<=n-1]
    #              - dx * sum_d w[n-1+d] * h[m-d] * I[0<=m-d<=n-1]
    
    # This is a convolution!
    # Part 1: grad1[m] = dx * sum_d w[n-1+d] * (1-h[m+d])
    #   = dx * ((1-h) * w reversed and flipped) [m]  ... it's a cross-correlation
    
    # Let me just do it with the offset loop since M is small
    for d in range(-(n-1), n):
        k = n - 1 + d
        wk = w[k]
        if wk < 1e-15:
            continue
        # Part 1: m+d in [0, n-1] => m in [-d, n-1-d], intersect [0, n-1]
        m_lo = max(0, -d)
        m_hi = min(n-1, n-1-d)
        if m_lo <= m_hi:
            ms = np.arange(m_lo, m_hi+1)
            grad[ms] += dx * wk * (1.0 - h[ms + d])
        # Part 2: m-d in [0, n-1] => m in [d, n-1+d], intersect [0, n-1]
        m_lo2 = max(0, d)
        m_hi2 = min(n-1, n-1+d)
        if m_lo2 <= m_hi2:
            ms = np.arange(m_lo2, m_hi2+1)
            grad[ms] += -dx * wk * h[ms - d]
    
    return grad
```

```python
import numpy as np
from scipy.optimize import minimize, linprog
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def repair_feasible(h):
    n = len(h)
    h = np.clip(np.asarray(h, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, 1.0 - h[idx])
            h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)[::-1]
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    return h

def build_grad_matrix(h):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    G = np.zeros((M, n))
    center = n - 1
    G[center, :] = dx * (1.0 - 2.0 * h)
    for d in range(1, n):
        G[center + d, 0:n-d] += -dx * h[d:n]
        G[center + d, d:n] += dx * (1.0 - h[0:n-d])
        G[center - d, 0:n-d] += dx * (1.0 - h[d:n])
        G[center - d, d:n] += -dx * h[0:n-d]
    return G

def grad_weighted(h, w):
    """Compute sum_k w[k] * G[k,:] efficiently."""
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    grad = np.zeros(n)
    for d in range(0, n):
        wk_pos = w[n - 1 + d] if n - 1 + d < M else 0.0
        wk_neg = w[n - 1 - d] if n - 1 - d >= 0 else 0.0
        if d == 0:
            if abs(wk_pos) > 1e-16:
                grad += dx * wk_pos * (1.0 - 2.0 * h)
        else:
            if abs(wk_pos) > 1e-16:
                grad[0:n-d] += -dx * wk_pos * h[d:n]
                grad[d:n] += dx * wk_pos * (1.0 - h[0:n-d])
            if abs(wk_neg) > 1e-16:
                grad[0:n-d] += dx * wk_neg * (1.0 - h[d:n])
                grad[d:n] += -dx * wk_neg * h[0:n-d]
    return grad

def softmax_obj_grad(h, beta, mu):
    n = len(h)
    profile = compute_profile(h)
    shifted = beta * profile - np.max(beta * profile)
    exp_shifted = np.exp(shifted)
    w = exp_shifted / np.sum(exp_shifted)
    obj = (np.max(beta * profile) + np.log(np.sum(exp_shifted))) / beta
    grad = grad_weighted(h, w)
    if mu > 0:
        grad += mu * (h.sum() - n / 2.0)
    return obj, grad

def coordinate_optimal(h, m):
    """Find optimal x for h[m] minimizing max_k profile_k, given other coords fixed."""
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    profile = compute_profile(h)
    a = np.zeros(M)
    for k in range(M):
        d = k - (n - 1)
        ak = 0.0
        j1 = m + d
        if 0 <= j1 < n:
            ak += dx * (1.0 - h[j1])
        i2 = m - d
        if 0 <= i2 < n:
            ak += -dx * h[i2]
        a[k] = ak
    b = profile - a * h[m]
    lo, hi = 0.0, 1.0
    for _ in range(80):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = float(np.max(a * m1 + b))
        f2 = float(np.max(a * m2 + b))
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2.0

def softmax_lbfgs(h0, beta, t_start, deadline, maxiter=200):
    n = len(h0)
    h = h0.copy()
    h = repair_feasible(h)
    best_h = h.copy()
    best_c = exact_score(h)
    mu = 10.0
    for it in range(maxiter):
        if time.perf_counter() > deadline:
            break
        def obj_fn(x):
            xr = np.clip(x, 0.0, 1.0)
            prof = compute_profile(xr)
            shifted = beta * prof - np.max(beta * prof)
            val = (np.max(beta * prof) + np.log(np.sum(np.exp(shifted)))) / beta
            s = xr.sum() - n / 2.0
            val += 0.5 * mu * s * s
            return val

        def jac_fn(x):
            xr = np.clip(x, 0.0, 1.0)
            prof = compute_profile(xr)
            shifted = beta * prof - np.max(beta * prof)
            ws = np.exp(shifted)
            ws = ws / np.sum(ws)
            g = grad_weighted(xr, ws)
            g += mu * (xr.sum() - n / 2.0)
            return np.where((x > 0) & (x < 1), g, 0.0)

        try:
            res = minimize(obj_fn, h, jac=jac_fn, method='L-BFGS-B',
                           bounds=[(0.0, 1.0)] * n,
                           options={'maxiter': maxiter, 'ftol': 1e-14})
            if res.x is not None:
                h_new = repair_feasible(res.x)
                c_new = exact_score(h_new)
                if c_new < best_c - 1e-14:
                    best_c = c_new
                    best_h = h_new.copy()
                h = h_new
        except Exception:
            break
        mu *= 5.0
        if mu > 1e6:
            break
    return best_h, best_c

def coordinate_descent(h, t_start, deadline, max_sweeps=50):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c = exact_score(h)
    order = np.arange(n)
    for sweep in range(max_sweeps):
        if time.perf_counter() > deadline:
            break
        np.random.shuffle(order)
        for m in order:
            x_opt = coordinate_optimal(h, m)
            if abs(x_opt - h[m]) > 1e-14:
                h[m] = x_opt
        h = repair_feasible(h)
        c = exact_score(h)
        if c < best_c - 1e-14:
            best_c = c
            best_h = h.copy()
        elif sweep > 5:
            break
    return best_h, best_c

def subgradient_descent(h, t_start, deadline, max_iters=3000):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c = exact_score(h)
    step = 0.01
    momentum = 0.7
    vel = np.zeros(n)
    no_improve = 0
    for it in range(max_iters):
        if time.perf_counter() > deadline:
            break
        profile = compute_profile(h)
        C = float(np.max(profile))
        if C < best_c - 1e-14:
            best_c = C
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 100:
            step *= 0.85
            if step < 1e-7:
                break
        spread = C - np.min(profile)
        if spread < 1e-10:
            break
        tau = max(0.002, 0.06 * spread)
        thresh = C - tau
        active = np.where(profile >= thresh)[0]
        if len(active) == 0:
            active = np.array([np.argmax(profile)])
        prof_a = profile[active]
        w = (prof_a - prof_a.min() + 1e-10)
        w = w / w.sum()
        G_a = build_grad_matrix(h)[active, :]
        grad = (G_a.T @ w)
        grad -= grad.mean()
        vel = momentum * vel + (1 - momentum) * grad
        norm = np.linalg.norm(vel)
        if norm < 1e-14:
            break
        vel_normed = vel / norm
        h_try = repair_feasible(h - step * vel_normed)
        c_try = exact_score(h_try)
        if c_try < C - 1e-15:
            h = h_try
        else:
            h_try2 = repair_feasible(h - step * 0.5 * vel_normed)
            c_try2 = exact_score(h_try2)
            if c_try2 < C - 1e-15:
                h = h_try2
                step *= 0.9
            else:
                step *= 0.9
    return best_h, best_c

def interpolate_h(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

def optimize_full(h, t_start, deadline):
    n = len(h)
    if deadline - time.perf_counter() < 15:
        return h, exact_score(h)
    best_h = h.copy()
    best_c = exact_score(h)
    # L-BFGS with increasing beta
    for beta in [20.0, 50.0, 120.0, 300.0, 800.0]:
        if deadline - time.perf_counter() < 25:
            break
        h_s, c_s = softmax_lbfgs(best_h, beta, t_start, deadline, maxiter=150)
        if c_s < best_c - 1e-14:
            best_c = c_s
            best_h = h_s
    # Coordinate descent
    if deadline - time.perf_counter() > 30:
        h_cd, c_cd = coordinate_descent(best_h, t_start, deadline, max_sweeps=40)
        if c_cd < best_c - 1e-14:
            best_c = c_cd
            best_h = h_cd
    # Subgradient
    if deadline - time.perf_counter() > 30:
        h_sg, c_sg = subgradient_descent(best_h, t_start, deadline, max_iters=2000)
        if c_sg < best_c - 1e-14:

    # Final coordinate polish
    if deadline - time.perf_counter() > 20:

        if c_cd2 < best_c - 1e-14:
            best_c = c_cd2
            best_h = h_cd2
    return best_h, best_c
```
