def repair_feasible(vec):

            idx = order[i]

            h[idx] += add
            remaining -= add

    else:

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx] + 1e-15)
            h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx] + 1e-15)
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break

    if abs(diff2) > 1e-15:

        else:
            idx = int(np.argmax(h))

    bounds = [(0.0, 1.0)] * n + [(0.0, t0 + 2.0)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': tol})
        if res.x is not None and np.isfinite(res.x).all():
            hs = repair_feasible(res.x[:-1])
            cs = exact_score(hs)
            if np.isfinite(cs):
                return hs, cs
    except Exception:
        pass
    return None, None

def interp_h(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    fx = np.concatenate([[0.0], xs, [2.0]])
    fh = np.concatenate([[h[0]], h, [h[-1]]])
    hn = np.interp(xt, fx, fh)
    return repair_feasible(np.clip(hn, 0, 1))

def gradient_line_search_loop(h0, budget_fn, min_remaining=30):
    h = h0.copy()
    curr = exact_score(h)
    n = len(h)
    temperature = 0.004
    min_temp = 0.0008
    stagnant = 0
    rng_local = np.random.RandomState(777)
    while budget_fn() > min_remaining:
        grad = smooth_gradient(h, temperature)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            direction = rng_local.randn(n)
            direction -= direction.mean()
            direction /= (np.linalg.norm(direction) + 1e-15)
        else:
            direction = -grad / gnorm
            direction -= direction.mean()
        h_new, s_new = line_search_1d(h, direction, curr, max_alpha=0.4)
        if h_new is not None and s_new < curr - 1e-13:
            h = h_new
            curr = s_new
            stagnant = 0
            temperature = max(min_temp, temperature * 0.97)
        else:
            stagnant += 1
            temperature = min(0.02, temperature * 1.4)
            if stagnant > 30:
                d_rand = rng_local.randn(n) * 0.01
                d_rand -= d_rand.mean()
                d_rand /= (np.linalg.norm(d_rand) + 1e-15)
                h_new, s_new = line_search_1d(h, d_rand, curr, max_alpha=0.15)
                if h_new is not None and s_new < curr - 1e-13:
                    h = h_new
                    curr = s_new
                    stagnant = 0
                    temperature = max(min_temp, temperature * 0.9)
            if stagnant > 150:
                break
    return h, curr

def binary_local_search(h_bin, max_iters=2000, budget_fn=None):
    n = len(h_bin)
    h = h_bin.copy()
    curr = exact_score(h)
    rng_ls = np.random.RandomState(42)
    stalled = 0
    for it in range(max_iters):
        if budget_fn is not None and budget_fn() < 8:
            break
        if stalled > 100:
            break
        zeros = np.where(h < 0.5)[0]
        ones = np.where(h > 0.5)[0]
        if len(zeros) == 0 or len(ones) == 0:
            break
        n_try = min(500, len(zeros) * len(ones))
        improved = False
        for _ in range(n_try):
            i = zeros[rng_ls.randint(len(zeros))]
            j = ones[rng_ls.randint(len(ones))]
            h[i], h[j] = 1.0, 0.0
            s = exact_score(h)
            if s < curr - 1e-10:
                curr = s
                improved = True
                stalled = 0
                break
            else:
                h[i], h[j] = 0.0, 1.0
        if not improved:
            stalled += 1
    return h, curr

def make_binary(h, n_target=None):
    if n_target is None:
        n_target = len(h)
    else:
        h = interp_h(h, n_target)
    n = n_target
    target = n // 2
    order = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[order[:target]] = 1.0
    return h_bin

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()
    rng = np.random.RandomState(seed)

    def tl():
        return budget_s - (time.perf_counter() - t_start)

    parent = None
    try:
        parent = np.array(initial_h_values, dtype=np.float64)
    except (NameError, Exception):
        pass
    if parent is None or len(parent) < 4:
        parent = np.full(70, 0.5)
    parent = repair_feasible(np.clip(parent, 0.0, 1.0))
    best_h = parent.copy()
    best_c = exact_score(parent)

    # Symmetrize
    h_sym = repair_feasible((parent + parent[::-1]) / 2.0)
    c_sym = exact_score(h_sym)
    if c_sym < best_c - 1e-12:
        best_h, best_c = h_sym, c_sym

    # Phase 1: Coordinate descent (exact per-coordinate optimization)
    if tl() > 200:
        bh, bc = coord_descent(best_h, max_sweeps=15, tol=1e-13, budget_fn=tl)
        if bc < best_c - 1e-14:
            best_h, best_c = bh, bc

    # Phase 2: Gradient + exact line search
    if tl() > 150:
        bh, bc = gradient_line_search_loop(best_h, tl, min_remaining=50)
        if bc < best_c - 1e-14:
            best_h, best_c = bh, bc

    # Phase 3: SLSQP
    if tl() > 120:
        hs, cs = slsqp_refine(best_h, maxiter=500, tol=1e-12)
        if hs is not None and cs < best_c - 1e-14:
            best_h, best_c = hs, cs

    # Phase 4: Coordinate descent again (after SLSQP)
    if tl() > 100:
        bh, bc = coord_descent(best_h, max_sweeps=10, tol=1e-13, budget_fn=tl)
        if bc < best_c - 1e-14:
            best_h, best_c = bh, bc

    # Phase 5: Binary pattern exploration
    if tl() > 100:
        n0 = len(best_h)
        h_bin = make_binary(best_h)
        hb, cb = binary_local_search(h_bin, max_iters=3000, budget_fn=tl)
        if tl() > 60:
            bh, bc = coord_descent(hb, max_sweeps=8, tol=1e-13, budget_fn=tl)
            if bc < best_c - 1e-14:
                best_h, best_c = bh, bc
            bh2, bc2 = gradient_line_search_loop(hb, tl, min_remaining=30)
            if bc2 < best_c - 1e-14:
                best_h, best_c = bh2, bc2
        # Random binary starts
        for trial in range(4):
            if tl() < 40:
                break
            h_rand = rng.rand(n0)
            h_rb = make_binary(h_rand)
            hb, cb = binary_local_search(h_rb, max_iters=1500, budget_fn=tl)
            if tl() > 30:
                bh, bc = coord_descent(hb, max_sweeps=6, tol=1e-13, budget_fn=tl)
                if bc < best_c - 1e-14:
                    best_h, best_c = bh, bc

    # Phase 6: Dimension exploration
    if tl() > 80:
        n0 = len(best_h)
        dims = []
        for d in range(max(40, n0 - 16), min(130, n0 + 17), 2):
            if d != n0 and d >= 40:
                dims.append(d)
        dims.sort(key=lambda d: abs(d - n0))
        for nd in dims[:5]:
            if tl() < 35:
                break
            hi = interp_h(best_h, nd)
            ci = exact_score(hi)
            if ci < best_c + 0.008:
                bh, bc = coord_descent(hi, max_sweeps=8, tol=1e-13, budget_fn=tl)
                if bc < best_c - 1e-14:
                    best_h, best_c = bh, bc
                bh2, bc2 = gradient_line_search_loop(hi, tl, min_remaining=25)
                if bc2 < best_c - 1e-14:
                    best_h, best_c = bh2, bc2

    # Phase 7: Final SLSQP polish
    if tl() > 50:
        hs, cs = slsqp_refine(best_h, maxiter=600, tol=1e-13)
        if hs is not None and cs < best_c - 1e-14:
            best_h, best_c = hs, cs

    # Phase 8: Micro coordinate descent
    if tl() > 20:
        bh, bc = coord_descent(best_h, max_sweeps=5, tol=1e-14, budget_fn=tl)
        if bc < best_c - 1e-14:
            best_h, best_c = bh, bc

    # Phase 9: Very small gradient steps
    if tl() > 10:
        h_f = best_h.copy()
        c_f = best_c
        temp = 0.002
        for _ in range(30):
            if tl() < 4:
                break
            grad = smooth_gradient(h_f, temp)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-13:
                break
            d = -grad / gnorm
            h_new, s_new = line_search_1d(h_f, d, c_f, max_alpha=0.03)
            if h_new is not None and s_new < c_f - 1e-13:
                h_f, c_f = h_new, s_new
            else:
                temp *= 0.7
                if temp < 1e-6:
                    break
        if c_f < best_c - 1e-14:
            best_h, best_c = h_f, c_f

    # Final validation
    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)
    try:
        er = evaluate_erdos_solution(best_h)
        if isinstance(er, (float, int)):
            final_c = min(final_c, float(er))
        elif isinstance(er, tuple) and len(er) >= 2:
            final_c = min(final_c, float(er[0]))
    except Exception:
        pass
    return best_h, final_c, len(best_h)
