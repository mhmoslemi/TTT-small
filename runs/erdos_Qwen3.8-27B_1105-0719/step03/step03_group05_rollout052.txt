import numpy as np
from scipy.optimize import minimize
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        capacity_idx = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = capacity_idx[i]
            add = min(remaining, capacity[idx])
            h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    return h

def gradient_of_profile_k(h, k_full, n):
    """Gradient of profile[k_full] with respect to h.
    k_full is the index in the 'full' correlate output (0 to 2n-2).
    The actual shift is d = k_full - (n-1).
    """
    g = np.zeros(n)
    d = k_full - (n - 1)
    if d >= 0:
        # profile = sum_{i=0}^{n-1-d} h[i] * (1 - h[i+d])
        for i in range(n - d):
            g[i] += (1 - h[i+d])
            g[i+d] += (-h[i])
    else:
        # d < 0, let dp = -d
        dp = -d
        # profile = sum_{i=dp}^{n-1} h[i] * (1 - h[i-dp])
        for i in range(dp, n):
            g[i] += (1 - h[i-dp])
            g[i-dp] += (-h[i])
    return g

def batch_gradient(h, active_indices):
    """Compute weighted sum of gradients for active profile indices."""
    n = len(h)
    g = np.zeros(n)
    for m in active_indices:
        g += gradient_of_profile_k(h, m, n)
    return g

# For efficiency, let's vectorize the batch gradient
def batch_gradient_fast(h, active_indices, n):
    g = np.zeros(n)
    for m in active_indices:
        d = m - (n - 1)
        if d >= 0:
            if n - d > 0:
                g[:n-d] += (1 - h[d:])
                g[d:] += (-h[:n-d])
        else:
            dp = -d
            g[dp:] += (1 - h[:n-dp])
            g[:n-dp] += (-h[dp:])
    return g

def nesterov_subgradient(h, max_iters=500, betas=[50, 100], rng=None):
    """Nesterov-accelerated projected subgradient method for min max_k f_k(h)."""
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    y = h.copy()  # Nesterov's y
    best_h = h.copy()
    best_c = exact_score(h)
    
    for beta in betas:
        for it in range(max_iters):
            profile = compute_profile(y)
            c = float(np.max(profile))
            
            # Softmax weights
            scaled = beta * profile
            shift = np.max(scaled)
            w = np.exp(scaled - shift)
            w /= w.sum()
            
            # Subgradient (weighted sum of active gradients)
            g = np.zeros(n)
            for m in range(len(profile)):
                if w[m] < 1e-10:
                    continue
                d = m - (n - 1)
                if d >= 0:
                    if n - d > 0:
                        g[:n-d] += w[m] * (1 - h[d:])
                        g[d:] += w[m] * (-h[:n-d])
                else:
                    dp = -d
                    g[dp:] += w[m] * (1 - h[:n-dp])
                    g[:n-dp] += w[m] * (-h[dp:])
            
            # Project gradient onto constraint (sum h = n/2)
            g -= g.mean()
            
            # Step size (diminishing)
            step = 0.5 / np.sqrt(it + 1)
            
            # Nesterov update
            y_new = h - step * g
            t_k = (1 + np.sqrt(1 + 4 * (it+1)**2)) / 2
            t_prev = (1 + np.sqrt(1 + 4 * it**2)) / 2 if it > 0 else 1
            beta_nest = (t_k - 1) / t_prev
            y = y_new + beta_nest * (y_new - h)
            h = repair_feasible(y)
            # Actually for Nesterov, we should keep y in the feasible set too
            y = repair_feasible(y)
            
            c_h = exact_score(h)
            if c_h < best_c:
                best_c = c_h
                best_h = h.copy()
    
    return best_h, best_c

            remaining -= rem
            if remaining <= 1e-12:

def profile_gradient(h, m, n):
    """Gradient of profile[m] w.r.t. h, where m is index in full correlate."""
    d = m - (n - 1)
    g = np.zeros(n)
    if d >= 0:
        if n - d > 0:
            g[:n-d] += (1.0 - h[d:])
            g[d:] += (-h[:n-d])
    else:
        dp = -d
        g[dp:] += (1.0 - h[:n-dp])
        g[:n-dp] += (-h[dp:])
    return g

def weighted_subgradient(h, weights):
    """Weighted sum of profile gradients."""
    n = len(h)
    g = np.zeros(n)
    for m, w in enumerate(weights):
        if w < 1e-12:
            continue
        d = m - (n - 1)
        if d >= 0:
            if n - d > 0:
                g[:n-d] += w * (1.0 - h[d:])
                g[d:] += w * (-h[:n-d])
        else:
            dp = -d
            g[dp:] += w * (1.0 - h[:n-dp])
            g[:n-dp] += w * (-h[dp:])
    return g

def lse_descent_v2(h, beta=80, max_iters=200, alpha_init=0.003):
    """LSE descent with better step size scheduling."""
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    alpha = alpha_init
    stuck = 0
    for it in range(max_iters):
        profile = compute_profile(h)
        c = float(np.max(profile))
        scaled = beta * profile
        max_s = np.max(scaled)
        w = np.exp(scaled - max_s)
        wsum = w.sum()
        w /= wsum
        g = weighted_subgradient(h, w)
        g -= g.mean()  # project onto sum(h) = const
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-14:
            break
        g /= gnorm
        improved = False
        for a in [alpha, alpha*0.5, alpha*0.25, alpha*0.1, alpha*0.05, alpha*0.02]:
            h_try = repair_feasible(h - a * g)
            s = exact_score(h_try)
            if s < best_c - 1e-14:
                h = h_try
                best_c = s
                improved = True
                alpha = min(alpha * 1.05, 0.015)
                stuck = 0
                break
            elif s < c - 1e-14:
                h = h_try
                improved = True
                break
        if not improved:
            alpha *= 0.5
            stuck += 1
            if stuck > 50 or alpha < 1e-10:
                break
    return h, best_c

def coordinate_descent(h, max_rounds=50, pair_stride=1):
    """Pair-wise coordinate descent: for each pair (i,j), optimize along e_i - e_j."""
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    
    for rnd in range(max_rounds):
        improved = False
        for i in range(0, n, pair_stride):
            for j in range(i+1, n, pair_stride):
                h_i = h[i]
                h_j = h[j]
                s = h_i + h_j
                # t = new h[i], then h[j] = s - t
                # Feasible: max(0, s-1) <= t <= min(1, s)
                t_lo = max(0.0, s - 1.0)
                t_hi = min(1.0, s)
                if t_hi - t_lo < 1e-12:
                    continue
                # Try several values of t
                best_t = h_i
                best_local = best_c
                # Sample points
                n_samples = 11
                for frac in range(n_samples):
                    t_try = t_lo + (t_hi - t_lo) * frac / (n_samples - 1)
                    h[i] = t_try
                    h[j] = s - t_try
                    c_try = exact_score(h)
                    if c_try < best_local - 1e-15:
                        best_local = c_try
                        best_t = t_try
                # Also try the current value
                h[i] = h_i
                h[j] = h_j
                if best_t != h_i:
                    h[i] = best_t
                    h[j] = s - best_t
                    if best_local < best_c - 1e-15:
                        best_c = best_local
                        improved = True
                    # Reset if not actually better than original
                    else:
                        h[i] = h_i
                        h[j] = h_j
        if not improved:
            break
    return h, best_c

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = np.linspace(0, 2, n + 1)
    xt = np.linspace(0, 2, new_n + 1)
    h_interp = np.zeros(new_n)
    for k in range(new_n):
        lo_k, hi_k = xt[k], xt[k+1]
        total_mass = 0.0
        total_width = 0.0
        for j in range(n):
            lo_j, hi_j = xs[j], xs[j+1]
            overlap = min(hi_k, hi_j) - max(lo_k, lo_j)
            if overlap > 0:
                total_mass += h[j] * overlap
                total_width += overlap
        h_interp[k] = total_mass / total_width if total_width > 1e-15 else 0.5
    h_interp = np.clip(h_interp, 0.0, 1.0)
    return repair_feasible(h_interp)

import numpy as np
from scipy.optimize import minimize
import time


def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))


def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx


def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, 1.0 - h[idx])
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    return h


def prof_grad_one(h, m, n):
    d = m - (n - 1)
    g = np.zeros(n)
    if d >= 0:
        if n - d > 0:
            g[:n - d] += (1.0 - h[d:])
            g[d:] += (-h[:n - d])
    else:
        dp = -d
        g[dp:] += (1.0 - h[:n - dp])
        g[:n - dp] += (-h[dp:])
    return g


def weighted_subgrad(h, weights, n):
    g = np.zeros(n)
    for m in range(len(weights)):
        w = weights[m]
        if w < 1e-12:
            continue
        d = m - (n - 1)
        if d >= 0:
            if n - d > 0:
                g[:n - d] += w * (1.0 - h[d:])
                g[d:] += w * (-h[:n - d])
        else:
            dp = -d
            g[dp:] += w * (1.0 - h[:n - dp])
            g[:n - dp] += w * (-h[dp:])
    return g


def lse_descent(h, beta=80.0, max_iters=200, alpha_init=0.003):
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    alpha = alpha_init
    stuck = 0
    for it in range(max_iters):
        profile = compute_profile(h)
        c = float(np.max(profile))
        scaled = beta * profile
        mx = np.max(scaled)
        w = np.exp(scaled - mx)
        ws = w.sum()
        if ws < 1e-300:
            break
        w /= ws
        g = weighted_subgrad(h, w, n)
        g -= g.mean()
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-14:
            break
        g /= gnorm
        improved = False
        for a in [alpha, alpha * 0.5, alpha * 0.25, alpha * 0.1, alpha * 0.05, alpha * 0.02]:
            h_try = repair_feasible(h - a * g)
            s = exact_score(h_try)
            if s < best_c - 1e-14:
                h = h_try
                best_c = s
                improved = True
                alpha = min(alpha * 1.06, 0.012)
                stuck = 0
                break
            elif s < c - 1e-14:
                h = h_try
                improved = True
                break
        if not improved:
            alpha *= 0.5
            stuck += 1
            if stuck > 50 or alpha < 1e-10:
                break
    return h, best_c


def coordinate_descent(h, max_rounds=20, n_samples=15):
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    for rnd in range(max_rounds):
        improved_any = False
        for i in range(n):
            for j in range(i + 1, n):
                a = h[i]
                b = h[j]
                s = a + b
                t_lo = max(0.0, s - 1.0)
                t_hi = min(1.0, s)
                if t_hi - t_lo < 1e-10:
                    continue
                best_t = a
                best_local = best_c
                for frac in range(n_samples):
                    t_val = t_lo + (t_hi - t_lo) * frac / (n_samples - 1)
                    if abs(t_val - a) < 1e-10:
                        continue
                    h[i] = t_val
                    h[j] = s - t_val
                    c_try = exact_score(h)
                    if c_try < best_local - 1e-15:
                        best_local = c_try
                        best_t = t_val
                h[i] = a
                h[j] = b
                if best_t != a:
                    h[i] = best_t
                    h[j] = s - best_t
                    c_new = exact_score(h)
                    if c_new < best_c - 1e-15:
                        best_c = c_new
                        improved_any = True
                    else:
                        h[i] = a
                        h[j] = b
        if not improved_any:
            break
    return h, best_c


def slsqp_solve(h0, maxiter=300, t_offset=0.001):
    n = len(h0)
    dx = 2.0 / n
    M = 2 * n - 1
    c0 = exact_score(h0)

    def obj(x):
        return x[-1]

    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g

    def eq_fun(x):
        return x[:-1].sum() - n / 2.0

    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g

    def ineq_fun(x):
        h = np.clip(x[:-1], 0, 1)
        t = x[-1]
        profile = np.correlate(h, 1 - h, mode="full") * dx
        return t - profile

    def ineq_jac(x):
        h = np.clip(x[:-1], 0, 1)
        J = np.zeros((M, n + 1))
        for m in range(M):
            d = m - (n - 1)
            if d >= 0:
                if n - d > 0:
                    J[m, :n - d] = -(1.0 - h[d:])
                    J[m, d:] = -h[:n - d]
            else:
                dp = -d
                J[m, dp:] = -(1.0 - h[:n - dp])
                J[m, :n - dp] = -h[dp:]
            J[m, -1] = 1.0
        return J

    x0 = np.concatenate([h0, [c0 + t_offset]])
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 0.5)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.x is not None and np.isfinite(res.x).all():
            h_new = repair_feasible(res.x[:-1])
            c_new = exact_score(h_new)
            if c_new <= c0 + 1e-10:
                return h_new, c_new
    except Exception:
        pass
    return None, None


def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = np.linspace(0, 2, n + 1)
    xt = np.linspace(0, 2, new_n + 1)
    h_interp = np.zeros(new_n)
    for k in range(new_n):
        lo_k, hi_k = xt[k], xt[k + 1]
        total_mass = 0.0
        total_width = 0.0
        for j in range(n):
            lo_j, hi_j = xs[j], xs[j + 1]
            overlap = min(hi_k, hi_j) - max(lo_k, lo_j)
            if overlap > 0:
                total_mass += h[j] * overlap
                total_width += overlap
        h_interp[k] = total_mass / total_width if total_width > 1e-15 else 0.5
    h_interp = np.clip(h_interp, 0.0, 1.0)
    return repair_feasible(h_interp)


def symmetrize_complement(h):
    n = len(h)
    h_rev = h[::-1].copy()
    h_sym = (h + (1.0 - h_rev)) / 2.0
    return repair_feasible(h_sym)


def momentum_descent(h, max_iters=300, momentum=0.85, lr_init=0.005):
    n = len(h)
    h = h.copy()
    v = np.zeros(n)
    best_c = exact_score(h)
    best_h = h.copy()
    for it in range(max_iters):
        profile = compute_profile(h)
        c = float(np.max(profile))
        m_max = int(np.argmax(profile))
        # Use top few for stability
        top_k = min(5, len(profile))
        sorted_idx = np.argsort(profile)[::-1][:top_k]
        g = np.zeros(n)
        for m in sorted_idx:
            g += prof_grad_one(h, m, n)
        g /= top_k
        g -= g.mean()
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-14:
            break
        g /= gnorm
        lr = lr_init / (1.0 + 0.02 * it)
        v = momentum * v - lr * g
        h_new = repair_feasible(h + v)
        s = exact_score(h_new)
        if s < best_c:
            best_c = s
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c


def random_pair_greedy(h, n_iters=5000, rng=None):
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    deltas = np.array([0.002, 0.005, 0.01, 0.02, 0.05, -0.002, -0.005, -0.01, -0.02, -0.05])
    for _ in range(n_iters):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = deltas[rng.randint(len(deltas))]
        new_i = h[i] + d
        new_j = h[j] - d
        if new_i < 0 or new_i > 1 or new_j < 0 or new_j > 1:
            continue
        h[i] = new_i
        h[j] = new_j
        c_try = exact_score(h)
        if c_try < best_c - 1e-14:
            best_c = c_try
        else:
            h[i] -= d
            h[j] += d
    return h, best_c


def active_set_descent(h, max_iters=200, n_active=12, alpha_init=0.004):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best_c = exact_score(h)
    alpha = alpha_init
    stuck = 0
    for it in range(max_iters):
        profile = compute_profile(h)
        c = float(np.max(profile))
        margin = max(0.001, 3.0 * (c - float(np.median(profile))))
