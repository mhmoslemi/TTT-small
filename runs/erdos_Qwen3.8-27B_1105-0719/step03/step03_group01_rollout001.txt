    G = np.zeros(n)
    for k in range(m):
        d = k - (n - 1)
        G += w[k] * grad_one_corr(h, d, n)
    G -= G.mean()
    norm = np.abs(G).max()
    if norm < 1e-15:
        return G, corrs, 0.0
    return G, corrs, norm


def topk_direction(h, n, k):
    corrs = compute_corrs(h)
    m = len(corrs)
    top_idx = np.argsort(corrs)[-k:]
    G = np.zeros(n)
    for kidx in top_idx:
        d = kidx - (n - 1)
        G += grad_one_corr(h, d, n)
    G /= k
    G -= G.mean()
    norm = np.abs(G).max()
    return G, corrs, norm


def line_search_exact(h, v, dx, target_sum, alpha_max, n_points=20):
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    if alpha_max <= 0:
        return best_h, best_c5, 0.0
    alphas = np.linspace(0, alpha_max, n_points)
    for alpha in alphas:
        h_new = project_feasible(h + alpha * v, target_sum)
        c5_new = c5_of_h(h_new, dx)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new
    return best_h, best_c5, best_h is not h


def get_alpha_max(h, v):
    pos = v > 1e-15
    neg = v < -1e-15
    alpha_max = np.inf
    if pos.any():
        alpha_max = min(alpha_max, ((1.0 - h[pos]) / v[pos]).min())
    if neg.any():
        alpha_max = min(alpha_max, (h[neg] / (-v[neg])).min())
    if not np.isfinite(alpha_max) or alpha_max < 0:
        alpha_max = 0.0
    return alpha_max


def lse_optimize(h, dx, target_sum, t0, deadline, T_start=5.0, T_end=0.05,
                 T_decay=0.97, max_no_improve=60):
    n = len(h)
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    T = T_start
    no_imp = 0
    while time.time() < deadline and T > T_end:
        G, corrs, norm = lse_direction(h, n, T)
        if norm < 1e-14:
            T *= T_decay
            continue
        v = G / norm
        alpha_max = get_alpha_max(h, v)
        if alpha_max < 1e-12:
            T *= T_decay
            no_imp += 1
            continue
        h_new, c5_new, _ = line_search_exact(h, v, dx, target_sum, alpha_max)
        if c5_new < best_c5 - 1e-14:
            h = h_new
            best_c5 = c5_new
            best_h = h_new.copy()
            no_imp = 0
            T = max(T_end, T * 0.995)
        else:
            no_imp += 1
            T *= T_decay
            if no_imp > max_no_improve:
                break
    return best_h, best_c5


def topk_optimize(h, dx, target_sum, t0, deadline, k=30, max_no_improve=40):
    n = len(h)
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    no_imp = 0
    while time.time() < deadline:
        G, corrs, norm = topk_direction(h, n, k)
        if norm < 1e-14:
            break
        v = G / norm
        alpha_max = get_alpha_max(h, v)
        if alpha_max < 1e-12:
            no_imp += 1
            k = min(2 * n - 1, k + 5)
            if no_imp > max_no_improve:
                break
            continue
        h_new, c5_new, _ = line_search_exact(h, v, dx, target_sum, alpha_max)
        if c5_new < best_c5 - 1e-14:
            h = h_new
            best_c5 = c5_new
            best_h = h_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp == 20:
                k = max(10, k - 10)
            if no_imp > max_no_improve:
                break
    return best_h, best_c5


def active_lp_step(h, dx, target_sum, n_active, alpha_bound):
    n = len(h)
    corrs = compute_corrs(h)
    n_total = 2 * n - 1
    current_c5 = corrs.max() * dx
    if n_active >= n_total:
        active_idx = np.arange(n_total)
    else:
        active_idx = np.argsort(corrs)[-n_active:]
    m = len(active_idx)
    G = np.zeros((m, n))
    for row, k in enumerate(active_idx):
        d = k - (n - 1)
        G[row] = grad_one_corr(h, d, n)
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_ub = np.empty((m, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -corrs[active_idx]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for i in range(n):
        bounds.append((-h[i] * alpha_bound, (1.0 - h[i]) * alpha_bound))
    bounds.append((-np.inf, np.inf))
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status != 0:
            return h, current_c5, False
        delta = res.x[:n]
    except Exception:
        return h, current_c5, False
    if np.abs(delta).max() < 1e-15:
        return h, current_c5, False
    best_h = h
    best_c5 = current_c5
    for s in [1.0, 0.7, 0.5, 0.35, 0.2, 0.1, 0.05]:
        h_new = project_feasible(h + s * delta, target_sum)
        c5_new = c5_of_h(h_new, dx)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new
            break
    return best_h, best_c5, best_c5 < current_c5 - 1e-14


def lp_optimize(h, dx, target_sum, t0, deadline, alpha_start=0.2,
                n_active_frac=0.35, alpha_decay=0.985, min_alpha=0.003,
                max_no_improve=50):
    n = len(h)
    n_total = 2 * n - 1
    n_active = max(10, int(n_active_frac * n_total))
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    alpha = alpha_start
    no_imp = 0
    while time.time() < deadline and alpha > min_alpha:
        h_new, c5_new, improved = active_lp_step(h, dx, target_sum, n_active, alpha)
        if improved and c5_new < best_c5 - 1e-14:
            h = h_new
            best_c5 = c5_new
            best_h = h_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            alpha *= alpha_decay
            if no_imp == 25:
                n_active = min(n_total, n_active + 10)
                alpha = max(alpha, 0.1)
            if no_imp > max_no_improve:
                break
    return best_h, best_c5


def sa_quantized(h_init, n, M, max_iters, seed, t0_temp=0.003, t_end=1e-6):
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    h_q = np.round(h_init * M) / M
    h_q = project_feasible(h_q, n / 2.0)
    a = np.round(h_q * M).astype(np.int64)
    target_a = n * M // 2
    # Fix sum
    cur_sum = a.sum()
    for _ in range(50000):
        if cur_sum == target_a:
            break
        if cur_sum > target_a:
            i = rng.randint(n)
            if a[i] > 0:
                a[i] -= 1
                cur_sum -= 1
        else:
            i = rng.randint(n)
            if a[i] < M:
                a[i] += 1
                cur_sum += 1
    h_cur = a.astype(float) / M
    cur_c5 = c5_of_h(h_cur, dx)
    best_c5 = cur_c5
    best_a = a.copy()
    temp = t0_temp
    for it in range(max_iters):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j or a[i] == 0 or a[j] == M:
            continue
        a[i] -= 1
        a[j] += 1
        h_new = a.astype(float) / M
        c5_new = c5_of_h(h_new, dx)
        delta_e = c5_new - cur_c5
        if delta_e < 0 or (temp > t_end and rng.random() < np.exp(-delta_e / temp)):
            cur_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_a = a.copy()
        else:
            a[i] += 1
            a[j] -= 1
        temp = max(t_end, temp * 0.9997)
    best_h = project_feasible(best_a.astype(float) / M, n / 2.0)
    return best_h, best_c5


def perturb_h(h, target_sum, sigma, rng):
    h_pert = h + sigma * rng.randn(len(h))
    return project_feasible(h_pert, target_sum)


def try_n_value(n_try, h_best, dx_best, c5_best, t0, deadline, seed):
    target_try = n_try / 2.0
    dx_try = 2.0 / n_try
    x_orig = np.linspace(0.0, 2.0, len(h_best))
    x_new = np.linspace(0.0, 2.0, n_try)
    h_interp = np.interp(x_new, x_orig, h_best)
    h_interp = project_feasible(h_interp, target_try)
    c5_init = c5_of_h(h_interp, dx_try)
    remaining = deadline - time.time()
    if remaining < 10:
        if c5_init < c5_best:
            return h_interp, c5_init
        return None, c5_best
    # LSE optimize
    dt1 = time.time() + min(remaining * 0.5, 60)
    h_opt, c5_opt = lse_optimize(h_interp, dx_try, target_try, t0, dt1, T_start=3.0)
    # Top-k optimize
    dt2 = time.time() + min(remaining * 0.3, 40)
    h_opt2, c5_opt2 = topk_optimize(h_opt, dx_try, target_try, t0, dt2, k=20)
    # LP optimize
    dt3 = time.time() + min(remaining * 0.2, 30)
    h_opt3, c5_opt3 = lp_optimize(h_opt2, dx_try, target_try, t0, dt3, alpha_start=0.15)
    best_local = (h_opt3, c5_opt3)
    if c5_opt < c5_opt3:
        best_local = (h_opt, c5_opt)
    if c5_opt2 < best_local[1]:
        best_local = (h_opt2, c5_opt2)
    if c5_init < best_local[1]:
        best_local = (h_interp, c5_init)
    if best_local[1] < c5_best:
        return best_local
    return None, c5_best


def combined_optimize(h, dx, target_sum, deadline, n, seed):
    """Run LSE, topk, and LP in sequence, keeping the best."""
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    # LSE phase
    dt_lse = time.time() + min((deadline - time.time()) * 0.4, 200)
    h1, c5_1 = lse_optimize(h, dx, target_sum, time.time(), dt_lse, T_start=4.0, T_end=0.1)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1
    # Top-k phase
    dt_topk = time.time() + min((deadline - time.time()) * 0.3, 100)
    h2, c5_2 = topk_optimize(best_h, dx, target_sum, time.time(), dt_topk, k=25)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2
    # LP phase
    dt_lp = time.time() + min((deadline - time.time()) * 0.3, 100)
    h3, c5_3 = lp_optimize(best_h, dx, target_sum, time.time(), dt_lp, alpha_start=0.15, n_active_frac=0.4)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3
    # Repeat with perturbation
    if time.time() < deadline - 20:
        h_pert = perturb_h(best_h, target_sum, 0.03, rng)
        dt2 = min(deadline - time.time() - 5, 60)
        h4, c5_4 = lse_optimize(h_pert, dx, target_sum, time.time(), time.time() + dt2, T_start=2.0, T_end=0.05)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4
    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n = len(h0)
    dx = 2.0 / n
    target_sum = n / 2.0
    h0 = project_feasible(h0, target_sum)
    best_h = h0.copy()
    best_c5 = c5_of_h(h0, dx)

    # Phase 1: Combined optimization on parent (50% of budget)
    dl1 = t0 + budget_s * 0.45
    best_h, best_c5 = combined_optimize(best_h, dx, target_sum, dl1, n, seed)

    # Phase 2: Multi-start perturbations (20%)
    dl2 = t0 + budget_s * 0.65
    sigmas = [0.04, 0.02, 0.06, 0.015, 0.035, 0.025]
    for p, sigma in enumerate(sigmas):
        if time.time() > dl2:
            break
        remaining = dl2 - time.time()
        if remaining < 15:
            break
        h_pert = perturb_h(best_h, target_sum, sigma, rng)
        sub_dl = time.time() + min(remaining * 0.6, 40)
        h_opt, c5_opt = combined_optimize(h_pert, dx, target_sum, sub_dl, n, seed + 100 + p)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 3: Quantized SA (15%)
    dl3 = t0 + budget_s * 0.80
    sa_configs = [(4, 20000, 0.004), (8, 15000, 0.003), (4, 25000, 0.006)]
    for s, (M, n_iters, t_alpha) in enumerate(sa_configs):
        if time.time() > dl3:
            break
        h_sa, c5_sa = sa_quantized(best_h, n, M, n_iters, seed + 500 + s * 11, t0_temp=t_alpha)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa
        # Polish SA result
        if time.time() < dl3 - 10:
            h_opt, c5_opt = lse_optimize(h_sa, dx, target_sum, time.time(),
                                         time.time() + min(dl3 - time.time() - 5, 30), T_start=2.0)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt

    # Phase 4: Try different n values (10%)
    dl4 = t0 + budget_s * 0.92
    n_candidates = [120, 124, 128, 130, 132, 116, 134, 125, 127]
    for n_try in n_candidates:
        if time.time() > dl4:
            break
        if n_try == n:
            continue
        remaining = dl4 - time.time()
        if remaining < 12:
            break
        h_new, c5_new = try_n_value(n_try, best_h, dx, best_c5, t0,
                                    time.time() + min(remaining * 0.6, 30), seed + n_try)
        if h_new is not None and c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

    # Phase 5: Final intensive polish (8%)
    dl5 = t0 + budget_s - 5
    n_b = len(best_h)
    dx_b = 2.0 / n_b
    ts_b = n_b / 2.0
    if time.time() < dl5:
        # Multiple LSE passes with different T
        for T_s in [3.0, 1.5, 0.5]:
            if time.time() > dl5 - 5:
                break
            sub_dl = time.time() + min(dl5 - time.time() - 3, 30)
            h_opt, c5_opt = lse_optimize(best_h, dx_b, ts_b, time.time(), sub_dl,
                                         T_start=T_s, T_end=T_s * 0.02, T_decay=0.96)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
        # Top-k with small k
        if time.time() < dl5 - 10:
            sub_dl = time.time() + min(dl5 - time.time() - 3, 20)
            h_opt, c5_opt = topk_optimize(best_h, dx_b, ts_b, time.time(), sub_dl, k=15)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
        # Final tiny perturbation + LSE
        if time.time() < dl5 - 8:
            h_tiny = perturb_h(best_h, ts_b, 0.01, rng)
            sub_dl = time.time() + min(dl5 - time.time() - 2, 15)
            h_opt, c5_opt = lse_optimize(h_tiny, dx_b, ts_b, time.time(), sub_dl,
                                         T_start=1.0, T_end=0.02, T_decay=0.95)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    n_final = len(best_h)
    ts_final = n_final / 2.0
    best_h = project_feasible(best_h, ts_final)
    dx_final = 2.0 / n_final
    best_c5 = c5_of_h(best_h, dx_final)

    return best_h, best_c5, n_final
