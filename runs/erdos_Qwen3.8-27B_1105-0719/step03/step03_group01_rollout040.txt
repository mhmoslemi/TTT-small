```python

        else:
            mask = h > 1e-10
        cnt = mask.sum()
        if cnt > 0:
            h[mask] += diff / cnt

def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def c5_of_h(h, dx):
    return float(compute_corrs(h).max() * dx)

def grad_corr_at(h, d, n):
    """Compute gradient of c_d with respect to h, where d is the offset."""
    p = np.arange(n)
    idx1 = p - d
    mask1 = (idx1 >= 0) & (idx1 < n)
    idx2 = p + d
    mask2 = (idx2 >= 0) & (idx2 < n)
    grad = np.zeros(n)
    if mask1.any():
        grad[mask1] = 1.0 - h[idx1[mask1]]
    if mask2.any():
        grad[mask2] += -h[idx2[mask2]]
    return grad

def nlp_obj(x):
    return x[0]

def nlp_obj_grad(x):
    g = np.zeros(len(x))
    g[0] = 1.0
    return g

def nlp_eq_fun(x):
    target = (len(x) - 1) / 2.0
    return x[1:].sum() - target

def nlp_eq_jac(x):
    n = len(x) - 1
    j = np.zeros(len(x))
    j[1:] = 1.0
    return j

def nlp_ineq_fun_all(x):
    n = len(x) - 1
    dx = 2.0 / n
    h = x[1:]

def nlp_ineq_jac_all(x):
    n = len(x) - 1
    dx = 2.0 / n
    h = x[1:]
    m = 2 * n - 1
    J = np.empty((m, n + 1))
    J[:, 0] = 1.0
    for k in range(m):
        d = k - (n - 1)
        J[k, 1:] = -dx * grad_corr_at(h, d, n)

    for row, k in enumerate(k_indices):

        J[row, 1:] = -dx * grad_corr_at(h, d, n)

def solve_nlp(h_init, dx, target_sum, t0, max_time, use_active=False, top_k=80,
              maxiter=300, ftol=1e-11):
    n = len(h_init)
    c_init = c5_of_h(h_init, dx)
    
    if time.time() - t0 > max_time:
        return h_init, c_init
    
    x0 = np.zeros(n + 1)
    x0[0] = c_init + 1e-4
    x0[1:] = h_init.copy()
    
    # Bounds: t in [c_init - 0.01, c_init + 0.01], h in [0, 1]
    t_lo = max(0.0, c_init - 0.02)
    t_hi = min(0.5, c_init + 0.02)
    bounds = [(t_lo, t_hi)] + [(0.0, 1.0)] * n
    
    constraints = [{'type': 'eq', 'fun': nlp_eq_fun, 'jac': nlp_eq_jac}]
    
    if use_active:
        global _active_idx
        corrs = compute_corrs(h_init)
        _active_idx = np.argsort(corrs)[-top_k:]
        constraints.append({'type': 'ineq', 'fun': nlp_ineq_fun_active, 'jac': nlp_ineq_jac_active})
    else:
        constraints.append({'type': 'ineq', 'fun': nlp_ineq_fun_all, 'jac': nlp_ineq_jac_all})
    
    try:
        result = minimize(nlp_obj, x0, jac=nlp_obj_grad,
                        method='SLSQP', bounds=bounds,
                        constraints=constraints,
                        options={'maxiter': maxiter, 'ftol': ftol, 'maxfun': maxiter * 20})
        h_new = np.clip(result.x[1:], 0.0, 1.0)
        h_new = project_feasible(h_new, target_sum)
        c_new = c5_of_h(h_new, dx)
        if c_new < c_init - 1e-12:
            return h_new, c_new
    except Exception:
        pass
    
    return h_init, c_init

def progressive_nlp(h_init, dx, target_sum, t0, max_time, k_schedule=None):
    """Run NLP with progressively more constraints."""
    if k_schedule is None:

    best_c5 = c5_of_h(h_init, dx)
    n = len(h_init)
```

```python
import numpy as np
from scipy.optimize import minimize
import time

_active_idx = None

def project_feasible(x, target_sum):
    n = len(x)
    x = np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        cnt = mask.sum()
        if cnt > 0:
            h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def c5_of_h(h, dx):
    return float(compute_corrs(h).max() * dx)

def grad_corr_at(h, d, n):
    p = np.arange(n)
    idx1 = p - d
    mask1 = (idx1 >= 0) & (idx1 < n)
    idx2 = p + d
    mask2 = (idx2 >= 0) & (idx2 < n)
    grad = np.zeros(n)
    if mask1.any():
        grad[mask1] = 1.0 - h[idx1[mask1]]
    if mask2.any():
        grad[mask2] += -h[idx2[mask2]]
    return grad

def nlp_obj(x):
    return x[0]

def nlp_obj_grad(x):
    g = np.zeros(len(x))
    g[0] = 1.0
    return g

def nlp_eq_fun(x):
    target = (len(x) - 1) / 2.0
    return x[1:].sum() - target

def nlp_eq_jac(x):
    n = len(x) - 1
    j = np.zeros(len(x))
    j[1:] = 1.0
    return j

def nlp_ineq_fun_all(x):
    n = len(x) - 1
    dx = 2.0 / n
    h = x[1:]
    return x[0] - dx * compute_corrs(h)

def nlp_ineq_jac_all(x):
    n = len(x) - 1
    dx = 2.0 / n
    h = x[1:]
    m = 2 * n - 1
    J = np.empty((m, n + 1))
    J[:, 0] = 1.0
    for k in range(m):
        d = k - (n - 1)
        J[k, 1:] = -dx * grad_corr_at(h, d, n)
    return J

def nlp_ineq_fun_active(x):
    n = len(x) - 1
    dx = 2.0 / n
    h = x[1:]
    corrs = compute_corrs(h)
    return x[0] - dx * corrs[_active_idx]

def nlp_ineq_jac_active(x):
    n = len(x) - 1
    dx = 2.0 / n
    h = x[1:]
    k_indices = _active_idx
    m = len(k_indices)
    J = np.empty((m, n + 1))
    J[:, 0] = 1.0
    for row, k in enumerate(k_indices):
        d = k - (n - 1)
        J[row, 1:] = -dx * grad_corr_at(h, d, n)
    return J

def solve_nlp(h_init, dx, target_sum, t0, max_time, use_active=False, top_k=80,
              maxiter=250, ftol=1e-11):
    n = len(h_init)
    c_init = c5_of_h(h_init, dx)
    global _active_idx

    if time.time() - t0 > max_time:
        return h_init, c_init

    x0 = np.zeros(n + 1)
    x0[0] = c_init + 1e-4
    x0[1:] = h_init.copy()

    t_lo = max(0.0, c_init - 0.05)
    t_hi = min(0.55, c_init + 0.02)
    bounds = [(t_lo, t_hi)] + [(0.0, 1.0)] * n

    constraints = [{'type': 'eq', 'fun': nlp_eq_fun, 'jac': nlp_eq_jac}]

    if use_active:
        corrs = compute_corrs(h_init)
        _active_idx = np.argsort(corrs)[-top_k:]
        constraints.append({'type': 'ineq', 'fun': nlp_ineq_fun_active, 'jac': nlp_ineq_jac_active})
    else:
        constraints.append({'type': 'ineq', 'fun': nlp_ineq_fun_all, 'jac': nlp_ineq_jac_all})

    try:
        result = minimize(nlp_obj, x0, jac=nlp_obj_grad,
                         method='SLSQP', bounds=bounds,
                         constraints=constraints,
                         options={'maxiter': maxiter, 'ftol': ftol, 'maxfun': maxiter * 20})
        h_new = np.clip(result.x[1:], 0.0, 1.0)
        h_new = project_feasible(h_new, target_sum)
        c_new = c5_of_h(h_new, dx)
        if c_new < c_init - 1e-13:
            return h_new, c_new
    except Exception:
        pass
    return h_init, c_init

def progressive_nlp(h_init, dx, target_sum, t0, max_time, k_schedule=None):
    global _active_idx
    if k_schedule is None:

    best_h = h_init.copy()

def line_polish(h, dx, target_sum, rng, t0, max_time, max_rounds=100):
    n = len(h)
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)

    for rnd in range(max_rounds):
        if time.time() - t0 > max_time:
            break
        scale = max(0.005, 0.08 * (1.0 - rnd / max_rounds))
        v = rng.randn(n) * scale
        v -= v.mean()

        pos_mask = v > 1e-12
        neg_mask = v < -1e-12
        alpha_max = 10.0
        if pos_mask.any():
            alpha_max = min(alpha_max, np.min((1.0 - h[pos_mask]) / v[pos_mask]))
        if neg_mask.any():
            alpha_max = min(alpha_max, np.min(h[neg_mask] / (-v[neg_mask])))
        if alpha_max <= 1e-6:
            continue

        best_alpha = 0.0
        best_c5_iter = best_c5
        for i in range(10):
            alpha = alpha_max * (i + 1) / 11.0
            h_try = project_feasible(h + alpha * v, target_sum)
            c5_try = c5_of_h(h_try, dx)
            if c5_try < best_c5_iter:
                best_c5_iter = c5_try
                best_alpha = alpha

        if best_alpha > 0 and best_c5_iter < best_c5 - 1e-12:
            h = project_feasible(h + best_alpha * v, target_sum)
            best_h = h.copy()
            best_c5 = best_c5_iter
        elif rnd > 40:
            break
    return best_h, best_c5

def pair_move_polish(h, dx, target_sum, rng, t0, max_time, max_rounds=200):
    n = len(h)
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)

    for rnd in range(max_rounds):
        if time.time() - t0 > max_time:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        avail = min(h[i], 1.0 - h[j])
        if avail < 1e-6:
            continue
        amt = avail * (0.01 + 0.3 * rng.random())
        h[i] -= amt
        h[j] += amt
        c5_try = c5_of_h(h, dx)
        if c5_try < best_c5 - 1e-13:
            best_c5 = c5_try
            best_h = h.copy()
        else:
            h[i] += amt
            h[j] -= amt
    return best_h, best_c5

def try_resolution(h_best, best_c5, n_try, t0, max_time, seed):

    elif c5_init < best_c5:

    return None, best_c5
```
