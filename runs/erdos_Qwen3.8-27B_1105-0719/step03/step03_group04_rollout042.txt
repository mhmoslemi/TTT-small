    profile = compute_profile(h)
    m = np.max(profile)
    return m + tau * np.log(np.sum(np.exp((profile - m) / tau)))

    """Gradient of the log-sum-exp approximation."""
    G = grad_matrix(h)
    profile = compute_profile(h)
    m = np.max(profile)
    weights = np.exp((profile - m) / tau)
    weights /= weights.sum()
    return G.T @ weights

def projected_descent_smooth(h, tau, n_iter=200, lr=None):

    n = len(h)
    h_cur = h.copy()
    h_prev = h.copy()
    best = h.copy()
    best_c = exact_score(h)
    
    if lr is None:
        lr = 1.0 / (np.linalg.norm(grad_matrix(h), ord=np.inf) + 1e-10)
        lr = min(lr, 0.1)
    
    for it in range(n_iter):
        # Nesterov extrapolation
        h_y = h_cur + ((it) / (it + 3.0)) * (h_cur - h_prev)

        g = logsumexp_gradient(repair(h_y), tau)

        g -= g.mean()

            best = h_new.copy()

        return x[-1]

    profile = compute_profile(h)
    G_a = G[active, :]
    p_a = profile[active]

    A_ub[:, -1] = -1.0

    A_eq = np.zeros((1, n + 1))

    bounds = [(-radius, radius)] * n + [(0, None)]
    try:

                      bounds=bounds, method='highs', options={'maxiter': 5000})
        if res.success and np.isfinite(res.x).all():

    except Exception:
        pass
    return None

def subgrad_direction(h, active):
    G = grad_matrix(h)
    profile = compute_profile(h)
    c = np.max(profile)
    G_a = G[active, :]
    # Weight by how close to max (emphasize the highest)
    weights = np.maximum(profile[active] - (c - 0.01), 0.0)
    if weights.sum() < 1e-15:
        weights = np.ones(len(active))
    weights /= weights.sum()
    grad = G_a.T @ weights
    grad -= grad.mean()
    norm = np.linalg.norm(grad)
    if norm < 1e-15:

    return grad / norm

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def compute_profile(h):

def grad_matrix(h):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    center = n - 1
    G = np.zeros((M, n))
    G[center, :] = dx * (1.0 - 2.0 * h)
    for d in range(1, n):
        G[center + d, d:] += dx * (1.0 - h[:n - d])
        G[center + d, :n - d] += -dx * h[d:]
        G[center - d, :n - d] += dx * (1.0 - h[d:])
        G[center - d, d:] += -dx * h[:n - d]
    return G

def repair(h):
    n = len(h)
    h = np.clip(h.astype(np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        remaining = diff
        for idx in order:
            add = min(remaining, 1.0 - h[idx])
            h[idx] += add
            remaining -= add
            if remaining <= 1e-14:
                break
    else:
        order = np.argsort(h)[::-1]
        remaining = -diff
        for idx in order:
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-14:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
    return np.clip(h, 0.0, 1.0)

def lp_step(h, active, radius):
    n = len(h)
    G = grad_matrix(h)
    profile = compute_profile(h)
    G_a = G[active, :]
    p_a = profile[active]
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_ub = np.zeros((len(active), n + 1))
    A_ub[:, :n] = G_a
    A_ub[:, -1] = -1.0
    b_ub = -p_a
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(-radius, radius)] * n + [(0, None)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'maxiter': 5000})
        if res.success and np.isfinite(res.x).all():
            return repair(h + res.x[:n])
    except Exception:
        pass
    return None

def subgrad_direction(h, active):
    G = grad_matrix(h)
    profile = compute_profile(h)
    c = np.max(profile)
    G_a = G[active, :]
    weights = np.maximum(profile[active] - (c - 0.01), 0.0)
    if weights.sum() < 1e-15:
        weights = np.ones(len(active))
    weights /= weights.sum()
    grad = G_a.T @ weights
    grad -= grad.mean()
    norm = np.linalg.norm(grad)
    if norm < 1e-15:
        return None
    return grad / norm

        return x[-1]

        g[-1] = 1.0

        return np.array([x[:-1].sum() - n / 2.0])
