    n = len(h)
    M = 2 * n - 1

def compute_G(h):
    n = len(h)
    M = 2 * n - 1
    G = np.zeros((n, M))
    for i in range(n):
        # First factor: k ∈ [i-n+1, i], n_idx = i-k ranges from n-1 down to 0
        ks = np.arange(i - n + 1, i + 1)
        n_idxs = i - ks  # n-1, n-2, ..., 0
        G[i, ks + n - 1] += (1.0 - h[n_idxs])
        # Second factor: k ∈ [-i, n-1-i], idx = i+k ranges from 0 to n-1
        ks2 = np.arange(-i, n - i)
        idxs = i + ks2
        G[i, ks2 + n - 1] -= h[idxs]
    return G

        return 0.5 * (d_lo + d_hi)

        else:

        # Sort by proximity to best
        dims_to_try.sort(key=lambda d: abs(d - n_best))

        for new_n in dims_to_try[:10]:
            if not time_ok(60):
                break
            h_interp = interpolate_to_n(best_h, new_n)
            # Quick pairwise descent
            h_opt, c_opt = pairwise_descent(h_interp, max_sweeps=60, t_start=t_start,
                                            budget_s=budget_s, margin=60, rng=rng)
            if c_opt < best_c - 1e-10:
                best_h = h_opt
                best_c = c_opt
                best_n = new_n
                # SLSQP polish at this dimension
                if time_ok(40):
                    sol = slsqp_active(best_h, maxiter=200, active_k=12, tol=1e-11)
                    if sol is not None:
                        c = exact_score(sol)
                        if c < best_c - 1e-13:
                            best_h = sol
                            best_c = c
                    # More pairwise
                    if time_ok(30):
                        h2, c2 = pairwise_descent(best_h, max_sweeps=50, t_start=t_start,
                                                  budget_s=budget_s, margin=30, rng=rng)
                        if c2 < best_c - 1e-13:
                            best_h = h2
                            best_c = c2

    # Phase 4: New starting points
    if time_ok(120):
        n_try = len(best_h)
        starts = []
        # Symmetric random
        for trial in range(3):
            starts.append(make_symmetric_start(n_try, np.random.RandomState(seed + 200 + trial)))
        # Binary
        for trial in range(3):
            starts.append(make_binary_start(n_try, np.random.RandomState(seed + 300 + trial)))
        # Mix with current best
        for mix in [0.3, 0.5, 0.7]:
            z = np.random.RandomState(seed + 400).randn(n_try)
            z -= z.mean()
            starts.append(repair_feasible(mix * best_h + (1 - mix) * 0.5))

        for h0 in starts[:6]:
            if not time_ok(40):
                break
            h_opt, c_opt = pairwise_descent(h0, max_sweeps=50, t_start=t_start,
                                            budget_s=budget_s, margin=40, rng=rng)
            if c_opt < best_c - 1e-10:
                best_h = h_opt
                best_c = c_opt
                if time_ok(30):
                    sol = slsqp_active(best_h, maxiter=200, active_k=12, tol=1e-11)
                    if sol is not None:
                        c = exact_score(sol)
                        if c < best_c - 1e-13:
                            best_h = sol
                            best_c = c

    # Phase 5: Final intensive pairwise descent
    if time_ok(100):
        h1, c1 = pairwise_descent(best_h, max_sweeps=150, t_start=t_start,
                                  budget_s=budget_s, margin=60, rng=rng)
        if c1 < best_c - 1e-14:
            best_h = h1
            best_c = c1

    # Phase 6: SLSQP with more active constraints
    if time_ok(80):
        sol = slsqp_active(best_h, maxiter=300, active_k=18, tol=1e-12)
        if sol is not None:
            c = exact_score(sol)
            if c < best_c - 1e-13:
                best_h = sol
                best_c = c
        # Pairwise again
        if time_ok(40):
            h1, c1 = pairwise_descent(best_h, max_sweeps=80, t_start=t_start,
                                      budget_s=budget_s, margin=40, rng=rng)
            if c1 < best_c - 1e-14:
                best_h = h1
                best_c = c1

    # Phase 7: Try a few more dimensions near the best
    if time_ok(60):
        n_best = len(best_h)
        for d in [n_best - 2, n_best + 2, n_best - 4, n_best + 4]:
            if d < 20 or d > 150 or d % 2 != 0:
                continue
            if not time_ok(30):
                break
            h_interp = interpolate_to_n(best_h, d)
            h_opt, c_opt = pairwise_descent(h_interp, max_sweeps=60, t_start=t_start,
                                            budget_s=budget_s, margin=30, rng=rng)
            if c_opt < best_c - 1e-10:
                best_h = h_opt
                best_c = c_opt

    # Final: one more SLSQP + pairwise
    if time_ok(40):
        sol = slsqp_active(best_h, maxiter=200, active_k=10, tol=1e-13)
        if sol is not None:
            c = exact_score(sol)
            if c < best_c - 1e-14:
                best_h = sol
                best_c = c
    if time_ok(25):
        h1, c1 = pairwise_descent(best_h, max_sweeps=50, t_start=t_start,
                                  budget_s=budget_s, margin=15, rng=rng)
        if c1 < best_c - 1e-14:
            best_h = h1
            best_c = c1

    # Final validation
    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)

    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            final_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 1:
            if np.isfinite(eval_result[0]):
                final_c = float(eval_result[0])
    except Exception:
        pass

    return best_h, final_c, len(best_h)
