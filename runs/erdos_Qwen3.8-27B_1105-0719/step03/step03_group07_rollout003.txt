    n = len(h0)
    dx = 2.0 / n
    c0 = exact_score(h0)
    
    def obj(x):
        return x[-1]
    
    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    
    def eq_fun(x):
        return x[:-1].sum() - n / 2.0
    
    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g
    
    def ineq_fun(x):
        h = x[:-1]
        t = x[-1]
        prof = profile_h(h)
        return t - prof[active_lags]
    
    def ineq_jac(x):
        h = x[:-1]
        G = compute_gradients(h, active_lags)
        J = np.zeros((len(active_lags), n + 1))
        J[:, :-1] = -G
        J[:, -1] = 1.0
        return J
    
    x0 = np.concatenate([h0, [c0 + 1e-5]])
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 1.0)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': tol})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = repair_feasible(res.x[:-1])
            c_new = exact_score(h_new)
            if c_new <= c0 + 1e-10:
                return h_new
    except Exception:
        pass
    return None

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    return repair_feasible(np.clip(h_new, 0.0, 1.0))

def symmetrize(h):
    return (h + h[::-1].copy()) / 2.0

def coordinate_descent(h, max_sweeps=15, t_start=None, budget_s=1100):
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    for sweep in range(max_sweeps):
        if t_start is not None and time.perf_counter() - t_start > budget_s - 10:
            break
        improved = False
        for i in range(n):
            if t_start is not None and time.perf_counter() - t_start > budget_s - 10:
                break
            h_save = h[i]
            for direction in [1, -1]:
                for frac in [0.02, 0.01, 0.005, 0.002, 0.001]:
                    delta = direction * frac
                    h_new_val = h_save + delta
                    if h_new_val < -1e-12 or h_new_val > 1.0 + 1e-12:
                        continue
                    h[i] = h_new_val
                    h = repair_feasible(h)
                    s = exact_score(h)
                    if s < best_c - 1e-15:
                        best_c = s
                        improved = True
                        break
                    else:
                        h[i] = h_save
                else:
                    continue
                break
            h[i] = h[i] if np.isfinite(h[i]) else h_save
            h = repair_feasible(h)
        if not improved:
            break
    return h, best_c

def lp_search_loop(h, max_iters=2000, t_start=None, budget_s=1100):
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    active_window = 0.003
    stall_count = 0
    max_stall = 50
    
    for it in range(max_iters):
        if t_start is not None and time.perf_counter() - t_start > budget_s - 30:
            break
        P = profile_h(h)
        c_max = P.max()
        if c_max < best_c:
            best_c = c_max
            h = repair_feasible(h)
        
        thresh = c_max - active_window
        active = np.where(P >= thresh)[0]
        if len(active) == 0:
            active = np.array([np.argmax(P)])
        
        direction = lp_leveling_direction(h, active, step_bound=0.5)
        if direction is None:
            top_k = min(len(active), 10)
            top_idx = np.argsort(P[active])[-top_k:]
            top_active = active[top_idx]
            direction = lp_leveling_direction(h, top_active, step_bound=0.8)
        
        if direction is None:
            sol = slsqp_leveling(h, active[:20], maxiter=100, tol=1e-11)
            if sol is not None:
                c_sol = exact_score(sol)
                if c_sol < best_c - 1e-12:
                    h = sol
                    best_c = c_sol
                    stall_count = 0
                else:
                    stall_count += 1
            else:
                stall_count += 1
            if stall_count >= max_stall:
                break
            continue
        
        norm = np.linalg.norm(direction)
        if norm < 1e-10:
            stall_count += 1
            if stall_count >= max_stall:
                break
            continue
        direction = direction / norm
        
        best_step = 0
        best_c_try = exact_score(h)
        for alpha in [0.3, 0.2, 0.15, 0.1, 0.07, 0.05, 0.03, 0.02, 0.01, 0.005]:
            h_try = repair_feasible(h + alpha * direction)
            c_try = exact_score(h_try)
            if c_try < best_c_try:
                best_c_try = c_try
                best_step = alpha
        
        if best_step > 0 and best_c_try < best_c - 1e-13:
            h = repair_feasible(h + best_step * direction)
            best_c = best_c_try
            stall_count = 0
            active_window = max(0.001, active_window * 0.95)
        else:
            if best_step > 0:
                h = repair_feasible(h + best_step * direction)
                c_now = exact_score(h)
                if c_now < best_c:
                    best_c = c_now
            stall_count += 1
            active_window = min(0.01, active_window * 1.2)
            if stall_count >= max_stall:
                break
    return h, best_c

def soft_max_descent(h, max_iters=100, t_start=None, budget_s=1100):
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    vel = np.zeros(n)
    momentum = 0.4
    
    for beta in [30, 60, 120, 250]:
        alpha = 0.05
        for it in range(max_iters // 4):
            if t_start is not None and time.perf_counter() - t_start > budget_s - 20:
                return h, best_c
            P = profile_h(h)
            Pmax = P.max()
            exp_arg = beta * (P - Pmax)
            exp_arg = np.clip(exp_arg, -500, 0)
            weights = np.exp(exp_arg)
            S = weights.sum()
            if S < 1e-300:
                weights = np.zeros(len(P))
                weights[np.argmax(P)] = 1.0
            else:
                weights /= S
            active = np.where(weights > 0.005)[0]
            if len(active) == 0:
                active = np.argsort(weights)[-5:]
            G = compute_gradients(h, active)
            w_a = weights[active]
            w_a = w_a / w_a.sum()
            grad = G.T @ w_a
            grad -= grad.mean()
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break
            grad /= gnorm
            vel = momentum * vel + grad
            vnorm = np.linalg.norm(vel)
            if vnorm > 0:
                vel /= vnorm
            improved = False
            for a in [alpha, alpha*0.5, alpha*0.25, alpha*0.1]:
                h_try = repair_feasible(h - a * vel)
                s = exact_score(h_try)
                if s < best_c - 1e-14:
                    h = h_try
                    best_c = s
                    improved = True
                    break
                elif s < exact_score(h) * (1 - 1e-8):
                    h = h_try
                    if exact_score(h) < best_c:
                        best_c = exact_score(h)
            if improved:
                alpha = min(alpha * 1.05, 0.12)
            else:
                alpha *= 0.6
                if alpha < 1e-7:
                    break
    return h, best_c

def block_coordinate_descent(h, block_size=7, max_rounds=50, t_start=None, budget_s=1100):
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    n_blocks = (n + block_size - 1) // block_size
    
    for rnd in range(max_rounds):
        if t_start is not None and time.perf_counter() - t_start > budget_s - 15:
            break
        improved = False
        for b in range(n_blocks):
            if t_start is not None and time.perf_counter() - t_start > budget_s - 15:
                break
            start = b * block_size
            end = min(start + block_size, n)
            n_block = end - start
            if n_block < 2:
                continue
            h_block_save = h[start:end].copy()
            for i in range(n_block):
                for direction in [1, -1]:
                    for frac in [0.03, 0.015, 0.007, 0.003, 0.001]:
                        delta = direction * frac
                        new_val = h[start + i] + delta
                        if new_val < -1e-12 or new_val > 1.0 + 1e-12:
                            continue
                        h[start + i] = new_val
                        h = repair_feasible(h)
                        s = exact_score(h)
                        if s < best_c - 1e-15:
                            best_c = s
                            improved = True
                            break
                        else:
                            h[start + i] = h_block_save[i]
                    else:
                        continue
                    break
            h[start:end] = repair_feasible(h)[start:end]
        if not improved:
            break
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()
    rng = np.random.RandomState(seed)
    
    def time_ok(margin=20):
        return time.perf_counter() - t_start < budget_s - margin
    
    parent = None
    try:
        parent = np.array(initial_h_values, dtype=np.float64)
    except NameError:
        pass
    except Exception:
        pass
    
    if parent is None or len(parent) < 4:
        parent = np.full(70, 0.5)
    
    parent = np.clip(parent, 0.0, 1.0)
    parent = repair_feasible(parent)
    
    best_h = parent.copy()
    best_c = exact_score(parent)
    
    if time_ok():
        h_sym = repair_feasible(symmetrize(parent))
        c_sym = exact_score(h_sym)
        if c_sym < best_c - 1e-13:
            best_h = h_sym
            best_c = c_sym
    
    n = len(best_h)
    
    # Phase 1: LP-based active constraint leveling (main workhorse)
    if time_ok(100):
        h_lp, c_lp = lp_search_loop(best_h, max_iters=3000,
                                    t_start=t_start, budget_s=budget_s)
        if c_lp < best_c - 1e-14:
            best_h = h_lp
            best_c = c_lp
    
    # Phase 2: SLSQP active-set polish
    if time_ok(80):
        P = profile_h(best_h)
        c_max = P.max()
        thresh = c_max - 0.004
        active = np.where(P >= thresh)[0]
        if len(active) > 25:
            top_idx = np.argsort(P[active])[-25:]
            active = active[top_idx]
        sol = slsqp_leveling(best_h, active, maxiter=300, tol=1e-12)
        if sol is not None:
            c_sol = exact_score(sol)
            if c_sol < best_c - 1e-13:
                best_h = sol
                best_c = c_sol
    
    # Phase 3: Another LP round with tighter window
    if time_ok(80):
        h_lp2, c_lp2 = lp_search_loop(best_h, max_iters=2000,
                                      t_start=t_start, budget_s=budget_s)
        if c_lp2 < best_c - 1e-14:
            best_h = h_lp2
            best_c = c_lp2
    
    # Phase 4: Soft-max descent for different basin
    if time_ok(80):
        h_sm, c_sm = soft_max_descent(best_h, max_iters=80,
                                      t_start=t_start, budget_s=budget_s)
        if c_sm < best_c - 1e-13:
            best_h = h_sm
            best_c = c_sm
    
    # Phase 5: SLSQP with wider active set
    if time_ok(60):
        P = profile_h(best_h)
        c_max = P.max()
        thresh = c_max - 0.008
        active = np.where(P >= thresh)[0]
        if len(active) > 30:
            top_idx = np.argsort(P[active])[-30:]
            active = active[top_idx]
        sol = slsqp_leveling(best_h, active, maxiter=400, tol=1e-12)
        if sol is not None:
            c_sol = exact_score(sol)
            if c_sol < best_c - 1e-13:
                best_h = sol
                best_c = c_sol
    
    # Phase 6: Dimension exploration
    if time_ok(120):
        n_best = len(best_h)
        cand_dims = []
        for d in range(max(16, n_best - 15), min(150, n_best + 15)):
            if d % 2 == 0 and d != n_best:
                cand_dims.append(d)
        cand_dims.sort(key=lambda d: abs(d - n_best))
        
        for new_n in cand_dims[:10]:
            if not time_ok(30):
                break
            h_interp = interpolate_to_n(best_h, new_n)
            h_trial, c_trial = lp_search_loop(h_interp, max_iters=800,
                                              t_start=t_start, budget_s=budget_s)
            if c_trial < best_c - 5e-5:
                best_h = h_trial
                best_c = c_trial
                # Further polish at this dimension
                if time_ok(30):
                    h_pol, c_pol = lp_search_loop(best_h, max_iters=1500,
                                                  t_start=t_start, budget_s=budget_s)
                    if c_pol < best_c - 1e-14:
                        best_h = h_pol
                        best_c = c_pol
                P2 = profile_h(best_h)
                c_max2 = P2.max()
                thresh2 = c_max2 - 0.004
                active2 = np.where(P2 >= thresh2)[0]
                if len(active2) > 25:
                    top_idx2 = np.argsort(P2[active2])[-25:]
                    active2 = active2[top_idx2]
                sol2 = slsqp_leveling(best_h, active2, maxiter=200, tol=1e-12)
                if sol2 is not None:
                    c_sol2 = exact_score(sol2)
                    if c_sol2 < best_c - 1e-13:
                        best_h = sol2
                        best_c = c_sol2
    
    # Phase 7: Block coordinate descent
    if time_ok(60):
        best_h, best_c = block_coordinate_descent(best_h, block_size=5,
                                                  max_rounds=30,
                                                  t_start=t_start, budget_s=budget_s)
    
    # Phase 8: Final LP + SLSQP
    if time_ok(40):
        h_final, c_final = lp_search_loop(best_h, max_iters=500,
                                          t_start=t_start, budget_s=budget_s)
        if c_final < best_c - 1e-14:
            best_h = h_final
            best_c = c_final
    
    if time_ok(20):
        P = profile_h(best_h)
        c_max = P.max()
        thresh = c_max - 0.003
        active = np.where(P >= thresh)[0]
        if len(active) > 20:
            top_idx = np.argsort(P[active])[-20:]
            active = active[top_idx]
        sol = slsqp_leveling(best_h, active, maxiter=200, tol=1e-13)
        if sol is not None:
            c_sol = exact_score(sol)
            if c_sol < best_c - 1e-14:
                best_h = sol
                best_c = c_sol
    
    # Phase 9: Coordinate descent fine-tune
    if time_ok(15):
        best_h, best_c = coordinate_descent(best_h, max_sweeps=10,
                                            t_start=t_start, budget_s=budget_s)
    
    # Final validation
    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)
    
    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            final_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 1:
            if np.isfinite(eval_result[0]):
                final_c = float(eval_result[0])
    except Exception:
        pass
    
    return best_h, final_c, len(best_h)
