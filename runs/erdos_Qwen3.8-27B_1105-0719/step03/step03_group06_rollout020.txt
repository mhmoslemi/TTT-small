    n = len(h)
    M = 2 * n - 1
    h = h.copy()

        # s.t. G[:,j] * d_j - t <= -p_m  for all m

        #      -R <= d_j <= min(R, 1-h_j)  (tighter upper)
        #      max(-R, h_j - 1) <= d_j     (tighter lower)

import numpy as np

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(-(1.0 - h))
        remaining = diff
        for i in range(n):
            idx = order[i]
            cap = 1.0 - h[idx]
            if cap <= 1e-15:
                continue
            add = min(remaining, cap)
            h[idx] += add
            remaining -= add
            if remaining <= 1e-14:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            if h[idx] <= 1e-15:
                continue
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-14:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            candidates = np.where(h < 1.0 - 1e-10)[0]
        else:
            candidates = np.where(h > 1e-10)[0]
        if len(candidates) > 0:
            idx = candidates[len(candidates) // 2]
            h[idx] = float(np.clip(h[idx] + diff2, 0.0, 1.0))
    return h

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def grad_row(h, m, n):
    dx = 2.0 / n
    d = m - (n - 1)
    if d == 0:
        return dx * (1.0 - 2.0 * h)
    elif d > 0:
        g = np.zeros(n)
        g[d:] += dx * (1.0 - h[:-d])
        g[:-d] -= dx * h[d:]
        return g
    else:
        p = -d
        g = np.zeros(n)
        g[:-p] += dx * (1.0 - h[p:])
        g[p:] -= dx * h[:-p]
        return g

def lse_descent(h, betas, max_iters_per_beta=40):
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    for beta in betas:
        for it in range(max_iters_per_beta):

            c = float(profile.max())
            z = beta * (profile - c)
            exp_z = np.exp(z)
            s = exp_z.sum()
            if s < 1e-300:
                w = np.zeros(len(profile))
                w[np.argmax(profile)] = 1.0
            else:
                w = exp_z / s
            g = np.zeros(n)
            for m in range(len(profile)):
                if w[m] > 1e-10:
                    g += w[m] * grad_row(h, m, n)
            g -= g.mean()
            gnorm = np.linalg.norm(g)
            if gnorm < 1e-14:
                break
            d = -g / gnorm
            pos_mask = d > 1e-12
            neg_mask = d < -1e-12
            alpha_max = 0.08
            if pos_mask.any():
                alpha_max = min(alpha_max, float(((1.0 - h[pos_mask]) / d[pos_mask]).min()))
            if neg_mask.any():
                alpha_max = min(alpha_max, float((h[neg_mask] / (-d[neg_mask])).min()))
            alpha_max = max(alpha_max, 1e-10)
            best_alpha = 0.0

def trust_region_slp(h, R_init=0.05, max_iters=50):
    n = len(h)
    M = 2 * n - 1
    h = h.copy()
    best_c = exact_score(h)
    R = R_init
    stall = 0
    for it in range(max_iters):
        profile = compute_profile(h)
        c = float(profile.max())
        G = np.zeros((M, n))
        for m in range(M):
            G[m] = grad_row(h, m, n)
        c_lp = np.zeros(n + 1)
        c_lp[-1] = 1.0
        A_ineq = np.zeros((M, n + 1))
        A_ineq[:, :n] = G
        A_ineq[:, -1] = -1.0
        b_ineq = -profile
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)
        bounds = []
        for j in range(n):
            lo = max(-R, -h[j])
            hi = min(R, 1.0 - h[j])
            bounds.append((lo, hi))
        bounds.append((0.0, c + 1.0))
        try:
            res = linprog(c_lp, A_ub=A_ineq, b_ub=b_ineq,
                          A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
        except Exception:
            R *= 0.5
            if R < 1e-7:
                break
            continue
        if not res.success:
            R *= 0.5
            if R < 1e-7:
                break
            continue
        d_new = res.x[:n]
        best_alpha = 0.0
        best_val = c
        best_h = h
        for alpha in [1.0, 0.6, 0.35, 0.18, 0.09]:
            h_try = repair_feasible(h + alpha * d_new)
            val = exact_score(h_try)
            if val < best_val:
                best_val = val
                best_alpha = alpha
                best_h = h_try
        if best_val > c - 1e-12:
            R *= 0.45
            stall += 1
            if R < 1e-7 or stall > 15:
                break
            continue
        h = best_h
        if best_val < best_c:
            best_c = best_val
        stall = 0
        if best_alpha >= 0.5:
            R = min(R * 1.25, 0.12)
    return h, best_c

def slsqp_epigraph(h_start, maxiter=200):
    n = len(h_start)
    dx = 2.0 / n
    M = 2 * n - 1
    c0 = exact_score(h_start)
    t0 = c0 + 1e-3

    def obj(x):
        return x[-1]

    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g

    def eq_fun(x):
        return x[:-1].sum() - n / 2.0

    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g

    def ineq_fun(x):
        h = x[:-1]
        t = x[-1]
        profile = np.correlate(h, 1 - h, mode="full") * dx
        return t - profile

    def ineq_jac(x):
        h = x[:-1]
        J = np.zeros((M, n + 1))
        for m in range(M):
            J[m, :n] = -grad_row(h, m, n)
        J[:, -1] = 1.0
        return J

    x0 = np.concatenate([h_start, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 2.0)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.x is not None and np.isfinite(res.x).all():
            h_new = repair_feasible(res.x[:-1])
            c_new = exact_score(h_new)
            if c_new <= c0 + 1e-9:
                return h_new, c_new
    except Exception:
        pass
    return None, c0

def interp_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    edges_old = np.linspace(0.0, 2.0, n + 1)
    edges_new = np.linspace(0.0, 2.0, new_n + 1)
    h_new = np.zeros(new_n)
    for k in range(new_n):
        a, b = edges_new[k], edges_new[k + 1]
        total = 0.0
        for j in range(n):
            lo = max(a, edges_old[j])
            hi = min(b, edges_old[j + 1])
            if hi > lo:
                total += h[j] * (hi - lo)
        h_new[k] = total / (b - a)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

def pair_exchange_polish(h, max_rounds=80):
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    profile_cache = compute_profile(h)
    for rnd in range(max_rounds):
        improved = False
        for shrink in [0.02, 0.01, 0.005, 0.002]:
            for i in range(n):
                for j in range(i + 1, n):
                    if h[i] <= shrink or h[j] <= shrink:
                        continue
                    delta = shrink * (1.0 if h[i] > h[j] else -1.0)
                    if abs(h[i] - h[j]) < 2.0 * shrink + 1e-8:
                        continue
                    h_try = h.copy()
                    h_try[i] -= delta
                    h_try[j] += delta
                    if h_try.min() < -1e-12 or h_try.max() > 1.0 + 1e-12:
                        continue
                    h_try = repair_feasible(h_try)
                    val = exact_score(h_try)
                    if val < best_c - 1e-13:
                        h = h_try
                        best_c = val
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        if not improved:
            break
    return h, best_c
