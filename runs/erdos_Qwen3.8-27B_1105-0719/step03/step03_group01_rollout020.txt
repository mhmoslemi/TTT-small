            no_improve += 1
            if no_improve > 200:
                h = project_feasible(h + 0.03 * np.random.randn(n), target)
                v_acc = np.zeros(n)
                no_improve = 0

    return best_h, best_c5


def pair_swap_optimize(h, n, t0, max_time, max_iter=20000):
    """Pair-swap local search guided by LSE weights."""
    best_h = h.copy()
    best_c5 = c5_raw(h, n)
    target = n / 2.0
    cur_h = h.copy()
    cur_c5 = best_c5
    no_improve = 0

    for it in range(max_iter):
        if time.time() - t0 > max_time:
            break
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j:
            continue
        if cur_h[i] >= cur_h[j]:
            continue
        # Try swapping a small amount
        amt = min(cur_h[i], 1.0 - cur_h[j]) * 0.5
        if amt < 1e-10:
            continue
        cur_h[i] -= amt
        cur_h[j] += amt
        c5_try = c5_raw(cur_h, n)
        if c5_try < cur_c5 - 1e-15:
            cur_c5 = c5_try
            no_improve = 0
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = cur_h.copy()
        else:
            cur_h[i] += amt
            cur_h[j] -= amt
            no_improve += 1
            if no_improve > 5000:
                break

    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    np.random.seed(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n = len(h0)
    target = n / 2.0
    h0 = project_feasible(h0, target)

    best_h = h0.copy()
    best_c5 = c5_raw(h0, n)

    # Phase 1: LSE optimization - broad temperature
    remaining = budget_s * 0.35 - (time.time() - t0)
    if remaining > 15:
        h_opt, c5_opt = lse_optimize(best_h, n, t0, t0 + remaining,
                                      T_s=3.0, T_e=0.1, max_iter=8000, mu=0.7)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 2: LSE optimization - medium temperature
    remaining = budget_s * 0.55 - (time.time() - t0)
    if remaining > 15:
        h_opt, c5_opt = lse_optimize(best_h, n, t0, t0 + remaining,
                                      T_s=1.5, T_e=0.05, max_iter=8000, mu=0.75)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 3: LSE optimization - low temperature (fine tuning)
    remaining = budget_s * 0.7 - (time.time() - t0)
    if remaining > 15:
        h_opt, c5_opt = lse_optimize(best_h, n, t0, t0 + remaining,
                                      T_s=0.5, T_e=0.02, max_iter=6000, mu=0.8)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 4: Pair-swap polish
    remaining = budget_s * 0.8 - (time.time() - t0)
    if remaining > 20:
        h_opt, c5_opt = pair_swap_optimize(best_h, n, t0, t0 + remaining,
                                            max_iter=30000)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 5: Multi-start perturbations
    remaining = budget_s * 0.9 - (time.time() - t0)
    if remaining > 30:
        sigmas = [0.04, 0.08, 0.02, 0.06, 0.1, 0.03]
        for p in range(len(sigmas)):
            if time.time() - t0 > budget_s * 0.9:
                break
            sigma = sigmas[p]
            h_pert = project_feasible(best_h + sigma * np.random.randn(n), target)
            dt = budget_s * 0.9 - (time.time() - t0)
            if dt < 10:
                break
            h_opt, c5_opt = lse_optimize(h_pert, n, t0, t0 + dt * 0.7,
                                          T_s=1.0, T_e=0.03, max_iter=4000, mu=0.7)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt

    # Phase 6: Try different n values
    for n_try in [120, 128, 132, 112, 140]:
        if time.time() - t0 > budget_s * 0.95:
            break
        dt = budget_s * 0.95 - (time.time() - t0)
        if dt < 12:
            break
        x_o = np.linspace(0.0, 2.0, n + 1)[:-1]
        x_n = np.linspace(0.0, 2.0, n_try + 1)[:-1]
        h_i = np.interp(x_n, x_o, best_h)
        h_i = project_feasible(h_i, n_try / 2.0)
        c5_i = c5_raw(h_i, n_try)
        if dt > 18:
            h_o, c5_o = lse_optimize(h_i, n_try, t0, t0 + dt - 4,
                                      T_s=1.5, T_e=0.05, max_iter=3000, mu=0.7)
            if c5_o < best_c5:
                best_c5 = c5_o
                best_h = h_o
        elif c5_i < best_c5:
            best_c5 = c5_i
            best_h = h_i

    # Phase 7: Final very-low-T polish on best
    if time.time() - t0 < budget_s - 5:
        n_b = len(best_h)
        dt = budget_s - 5 - (time.time() - t0)
        if dt > 5:
            h_o, c5_o = lse_optimize(best_h, n_b, t0, t0 + dt,
                                      T_s=0.3, T_e=0.005, max_iter=int(min(dt * 20, 4000)), mu=0.85)
            if c5_o < best_c5:
                best_c5 = c5_o
                best_h = h_o

    # Final validation
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final / 2.0)
    best_c5 = c5_raw(best_h, n_final)

    return best_h, best_c5, n_final
