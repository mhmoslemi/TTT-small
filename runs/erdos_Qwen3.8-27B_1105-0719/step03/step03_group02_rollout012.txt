Specifically, B[n-1+j] = sum_i d[i]*d[i+j] for j >= 0 (positive lags)
And B[n-1-j] = sum_i d[i]*d[i-j] for j >= 1 (negative lags, which equals B[n-1+j] by symmetry)

So B_k in our notation (where k is the correlation index 0 to 2n-2) corresponds to B[k] from np.correlate(d, d, 'full').

1. Compute A = np.array([grad_corr_k(h, k, n) @ d for k in range(2*n-1)])
2. Compute B = np.correlate(d, d, 'full')
3. corr_k(alpha) = corrs[k] + alpha*A[k] - alpha^2*B[k]
4. Find min of max_k corr_k(alpha) over feasible alpha

- Refine with local search

```python
import numpy as np
from scipy.optimize import minimize, linprog
from scipy.special import logsumexp
import time


def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)


def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def c5_of_h(h, dx):
    return compute_corrs(h).max() * dx


def grad_corr_k(h, k, n):
    offset = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -offset)
    i_hi = min(n - 1, n - 1 - offset)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx + offset] += (1.0 - h[idx])
    grad[idx] += (-h[idx + offset])
    return grad


def all_grads_dot_d(h, d, n):
    return np.array([grad_corr_k(h, k, n) @ d for k in range(2 * n - 1)])


def exact_line_search(h, d, dx, n, max_iter_gs=80):
    pos_mask = d > 1e-14
    neg_mask = d < -1e-14
    alpha_lo = 0.0
    alpha_hi = 1.0
    if pos_mask.any():
        a_max_pos = ((1.0 - h[pos_mask]) / d[pos_mask]).min()
        alpha_hi = min(alpha_hi, a_max_pos)
    if neg_mask.any():
        a_min_neg = (-h[neg_mask] / d[neg_mask]).min()
        alpha_lo = max(alpha_lo, a_min_neg)
    if alpha_lo >= alpha_hi - 1e-14:
        return 0.0

    def obj(alpha):
        return compute_corrs(h + alpha * d).max()

    n_grid = 300
    alphas = np.linspace(alpha_lo, alpha_hi, n_grid)
    vals = np.empty(n_grid)
    for i in range(n_grid):
        vals[i] = obj(alphas[i])
    best_idx = np.argmin(vals)

    if best_idx == 0:
        a, b = alpha_lo, alphas[1]
    elif best_idx == n_grid - 1:
        a, b = alphas[-2], alpha_hi
    else:
        a = alphas[best_idx - 1]
        b = alphas[best_idx + 1]

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = b - gr * (b - a)
    dp = a + gr * (b - a)
    fc = obj(c)
    fd = obj(dp)
    for _ in range(max_iter_gs):
        if b - a < 1e-13:
            break
        if fc < fd:
            b = dp
            dp = c
            fd = fc
            c = b - gr * (b - a)
            fc = obj(c)
        else:
            a = c
            c = dp
            fc = fd
            dp = a + gr * (b - a)
            fd = obj(dp)

    alpha_opt = (a + b) / 2.0
    return alpha_opt


def subgradient_direction(h, corrs, n, top_frac=0.15):
    max_c = corrs.max()
    tol = max_c * (1.0 - 0.08)
    active = np.where(corrs >= tol)[0]
    if len(active) == 0:
        active = np.array([np.argmax(corrs)])
    if len(active) > 60:
        active = np.argsort(corrs)[-60:]
    avg_grad = np.zeros(n)
    for k in active:
        avg_grad += grad_corr_k(h, k, n)
    avg_grad /= len(active)
    avg_grad -= avg_grad.mean()
    if np.linalg.norm(avg_grad) < 1e-14:
        return None
    return -avg_grad / np.linalg.norm(avg_grad)


def targeted_direction(h, corrs, n):
    k_star = np.argmax(corrs)
    g = grad_corr_k(h, k_star, n)
    g -= g.mean()
    if np.linalg.norm(g) < 1e-14:
        return None
    return -g / np.linalg.norm(g)


def random_direction(n, rng):
    d = rng.randn(n)
    d -= d.mean()
    norm = np.linalg.norm(d)
    if norm < 1e-14:
        return None
    return d / norm


def structured_direction(h, n, rng):
    m = rng.randint(1, max(2, n // 4))
    d = np.sin(2.0 * np.pi * m * np.arange(n) / n)
    d += 0.5 * np.cos(4.0 * np.pi * m * np.arange(n) / n)
    d -= d.mean()
    norm = np.linalg.norm(d)
    if norm < 1e-14:
        return None
    return d / norm


def direct_ls_optimize(h0, dx, target_sum, n, max_iters=500, time_fn=None, time_limit=60):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    stall = 0

    for it in range(max_iters):
        if time_fn is not None and time_fn() < time_limit:
            break

        corrs = compute_corrs(h)
        c5_val = corrs.max() * dx

        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

        if stall > 100:
            break

        # Generate and try multiple directions
        dirs = []
        d_sub = subgradient_direction(h, corrs, n)
        if d_sub is not None:
            dirs.append(d_sub)
        d_tgt = targeted_direction(h, corrs, n)
        if d_tgt is not None:
            dirs.append(d_tgt)
        if it % 3 == 0:
            d_rand = random_direction(n, np.random.RandomState(it * 7919 + n))
            if d_rand is not None:
                dirs.append(d_rand)
        if it % 5 == 0:
            d_str = structured_direction(h, n, np.random.RandomState(it * 104729 + n))
            if d_str is not None:
                dirs.append(d_str)

        improved = False
        for d in dirs:
            alpha = exact_line_search(h, d, dx, n)
            if abs(alpha) < 1e-14:
                continue
            h_new = h + alpha * d
            h_new = project_feasible(h_new, target_sum)
            c5_new = c5_of_h(h_new, dx)
            if c5_new < c5_val - 1e-13:
                h = h_new
                c5_val = c5_new
                improved = True
                break

        if not improved:
            # Try smaller steps
            for scale in [0.5, 0.25, 0.1]:
                for d in dirs[:2]:
                    d_s = d * scale
                    alpha = exact_line_search(h, d_s, dx, n)
                    if abs(alpha) < 1e-14:
                        continue
                    h_new = h + alpha * d_s
                    h_new = project_feasible(h_new, target_sum)
                    c5_new = c5_of_h(h_new, dx)
                    if c5_new < c5_val - 1e-13:
                        h = h_new
                        c5_val = c5_new
                        improved = True
                        break
                if improved:
                    break

    return best_h, best_c5


def pair_transfer_exact(h0, dx, target_sum, n_top=20, n_iter=5):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)

    for iteration in range(n_iter):
        corrs = compute_corrs(h)
        max_c = corrs.max()
        tol = max_c * 0.95
        active = np.where(corrs >= tol)[0]
        if len(active) > 40:
            active = np.argsort(corrs)[-40:]

        grad_active = np.zeros(n)
        for k in active:
            grad_active += grad_corr_k(h, k, n)

        dec_idx = np.argsort(grad_active)[-n_top:]
        inc_idx = np.argsort(grad_active)[:n_top]

        improved = False
        for i in dec_idx:
            for j in inc_idx:
                if i == j:
                    continue
                eps_hi = min(h[i], 1.0 - h[j])
                if eps_hi < 1e-12:
                    continue
                d = np.zeros(n)
                d[i] = -1.0
                d[j] = 1.0
                alpha = exact_line_search(h, d, dx, n, max_iter_gs=50)
                if alpha < 1e-12:
                    continue
                h_new = h + alpha * d
                h_new = project_feasible(h_new, target_sum)
                c5_new = c5_of_h(h_new, dx)
                if c5_new < best_c5 - 1e-14:
                    h = h_new
                    best_c5 = c5_new
                    best_h = h.copy()
                    corrs = compute_corrs(h)
                    improved = True

        if not improved:
            break

    return best_h, best_c5


def slsqp_direct(h0, dx, target_sum, beta, maxiter):
    n = len(h0)

    def obj(x):
        corrs = compute_corrs(x)
        return logsumexp(beta * corrs) / beta * dx

    def grad_obj(x):
        corrs = compute_corrs(x)
        shifted = corrs - corrs.max()
        w = np.exp(beta * shifted)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones_like(corrs) / len(corrs)
        else:
            w /= w_sum
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] > 1e-13:
                grad += w[k] * grad_corr_k(x, k, n)
        return grad * dx

    def c_fun(x):
        return x.sum() - target_sum

    def c_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': c_fun, 'jac': c_jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0, jac=grad_obj, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        h_out = project_feasible(res.x, target_sum)
    except Exception:
        h_out = project_feasible(h0, target_sum)
    return h_out, c5_of_h(h_out, dx)


def lp_polish(h0, dx, target_sum, max_iters=100, init_step=0.1):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    step = init_step
    stall = 0

    for it in range(max_iters):
        corrs = compute_corrs(h)
        c5_val = corrs.max() * dx
        max_c = corrs.max()

        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 30:
            step *= 0.6
            if step < 1e-7:
                break

        tol_val = max_c * 0.92
        active_k = np.where(corrs >= tol_val)[0]
        if len(active_k) > 50:
            active_k = np.argsort(corrs)[-50:]

        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        A_rows = []
        b_rows = []
        for k in active_k:
            row = np.zeros(n + 1)
            row[:n] = dx * grad_corr_k(h, k, n)
            row[n] = -1.0
            A_rows.append(row)
            b_rows.append(-corrs[k] * dx - 1e-14)

        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)

        bounds = []
        for i in range(n):
            bounds.append((max(-h[i], -step), min(1.0 - h[i], step)))
        bounds.append((0.0, None))

        try:
            res = linprog(c_obj, A_ub=np.array(A_rows), b_ub=np.array(b_rows),
                          A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
        except Exception:
            break
        if res.status != 0:
            break

        delta = res.x[:n]
        if np.max(np.abs(delta)) < 1e-14:
            break

        h_new = project_feasible(h + delta, target_sum)
        new_c5 = c5_of_h(h_new, dx)

        if new_c5 < c5_val - 1e-14:
            h = h_new
        else:
            improved = False
            for frac in [0.5, 0.25, 0.1]:
                h_new = project_feasible(h + frac * delta, target_sum)
                new_c5 = c5_of_h(h_new, dx)
                if new_c5 < c5_val - 1e-14:
                    h = h_new
                    improved = True
                    break
            if not improved:
                step *= 0.5
                if step < 1e-8:
                    break

    return best_h, best_c5


def full_optimize(h0, dx, target_sum, time_left_fn, min_time=15):
    n = len(h0)
    best_h = h0.copy()
    best_c5 = c5_of_h(h0, dx)

    # SLSQP with increasing beta
    for beta in [200, 800, 3000, 8000]:
        if time_left_fn() < min_time:
            break
        h_opt, c5_val = slsqp_direct(best_h, dx, target_sum, beta, 1000)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_opt

    # Direct line search
    if time_left_fn() > min_time + 10:
        h_ls, c5_ls = direct_ls_optimize(best_h, dx, target_sum, n,
                                         max_iters=300, time_fn=time_left_fn,
                                         time_limit=min_time)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls

    # LP polish
    if time_left_fn() > min_time + 5:
        h_lp, c5_lp = lp_polish(best_h, dx, target_sum, max_iters=80, init_step=0.08)
        if c5_lp < best_c5:
            best_c5 = c5_lp
            best_h = h_lp

    # More line search
    if time_left_fn() > min_time + 10:
        h_ls2, c5_ls2 = direct_ls_optimize(best_h, dx, target_sum, n,
                                           max_iters=200, time_fn=time_left_fn,
                                           time_limit=min_time)
        if c5_ls2 < best_c5:
            best_c5 = c5_ls2
            best_h = h_ls2

    # Pair transfer
    if time_left_fn() > min_time:
        h_pt, c5_pt = pair_transfer_exact(best_h, dx, target_sum, n_top=15, n_iter=4)
        if c5_pt < best_c5:
            best_c5 = c5_pt
            best_h = h_pt

    # Final SLSQP
    if time_left_fn() > min_time:
        h_f, c5_f = slsqp_direct(best_h, dx, target_sum, 10000, 1500)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f

    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s - (time.time() - t0)

    best_h = None

    # Phase 1: Main optimization from initial

        h_opt2, c5_val2 = full_optimize(h_try, dx, target_sum, time_left, min_time=25)
        if c5_val2 < best_c5:
            best_c5 = c5_val2
            best_h = h_opt2

    # Phase 3: Different n values
    n_candidates = [120, 130, 140, 150, 160, 170, 180, 200, 220, 250]
    n_candidates = [nn for nn in n_candidates if nn != n and nn % 2 == 0]
    n_candidates.sort(key=lambda x: abs(x - n))

    for n_try in n_candidates:
        if time_left() < 50:
            break
        dx_try = 2.0 / n_try
        ts_try = n_try / 2.0
        x_orig = np.linspace(0.0, 2.0, len(best_h), endpoint=False) + 1.0 / len(best_h)
        x_new = np.linspace(0.0, 2.0, n_try, endpoint=False) + 1.0 / n_try
        h_interp = np.interp(x_new, x_orig, best_h)
        h_interp = project_feasible(h_interp, ts_try)
        h_opt3, c5_val3 = full_optimize(h_interp, dx_try, ts_try, time_left, min_time=30)
        if c5_val3 < best_c5:
            best_c5 = c5_val3
            best_h = h_opt3

    if time_left() > 30 and best_h is not None:

        dx_b = 2.0 / n_b
        ts_b = n_b / 2.0
        # More direct line search iterations

                                               time_limit=15)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final
        h_pt, c5_pt = pair_transfer_exact(best_h, dx_b, ts_b, n_top=25, n_iter=5)
        if c5_pt < best_c5:
            best_c5 = c5_pt
            best_h = h_pt
        h_lp, c5_lp = lp_polish(best_h, dx_b, ts_b, max_iters=100, init_step=0.06)
        if c5_lp < best_c5:
            best_c5 = c5_lp
            best_h = h_lp

    # Phase 5: Ultra-fine pair transfer
    if time_left() > 10 and best_h is not None:
        n_b = len(best_h)
        dx_b = 2.0 / n_b
        ts_b = n_b / 2.0
        h_pt2, c5_pt2 = pair_transfer_exact(best_h, dx_b, ts_b, n_top=30, n_iter=6)
        if c5_pt2 < best_c5:
            best_c5 = c5_pt2
            best_h = h_pt2

    # Final validation
    if best_h is None:
        best_h = h0
        best_c5 = c5_of_h(h0, dx)
    else:
        n_final = len(best_h)
        best_h = project_feasible(best_h, n_final / 2.0)
        best_c5 = c5_of_h(best_h, 2.0 / n_final)

    return best_h, best_c5, len(best_h)
```
