So the total smoothed gradient is:
G = sum_k u_k * grad_k

where u_k are the LSE weights.

Then we center: v = G - mean(G) (to preserve the sum constraint).

The step size is limited by box constraints:
- For v[i] > 0: alpha_max_i = (1 - h[i]) / v[i]
- For v[i] < 0: alpha_max_i = h[i] / (-v[i])
- alpha_max = min of all these

We can do a line search along h + alpha * v for alpha in [0, alpha_max].

def lse_optimize(h, dx, target_sum, t0, max_time):
    n = len(h)
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    
    T = 0.1  # initial temperature
    T_decay = 0.99
    
    while time.time() - t0 < max_time:
        raw = np.correlate(h, 1.0 - h, mode='full')
        weights = lse_weights(raw, T)
        
        # Compute gradient
        grad = np.zeros(n)
        for p in range(2*n - 1):
            k = p - (n - 1)
            u = weights[p]
            if u < 1e-8:
                continue
            if k >= 0:
                for i in range(n - k):
                    grad[i] += u * (1.0 - h[i + k])
                    grad[i + k] += -u * h[i]
            else:
                for i in range(-k, n):
                    grad[i] += u * (1.0 - h[i + k])
                    grad[i + k] += -u * h[i]
        
        # Center (zero sum)
        grad -= grad.mean()
        
        if np.abs(grad).max() < 1e-12:
            break
        
        # Step size limited by box
        # alpha * grad must keep h + alpha*grad in [0,1]
        # For grad[i] > 0: alpha <= (1-h[i])/grad[i]
        # For grad[i] < 0: alpha <= h[i]/(-grad[i])
        pos_mask = grad > 1e-15
        neg_mask = grad < -1e-15
        alpha_max = 1.0  # default
        if pos_mask.any():
            alpha_max = min(alpha_max, (1.0 - h[pos_mask]) / grad[pos_mask]).min()
        if neg_mask.any():
            alpha_max = min(alpha_max, h[neg_mask] / (-grad[neg_mask])).min()
        
        if alpha_max < 1e-15:
            T *= T_decay
            continue
        
        # Line search
        best_alpha = 0.0
        best_c5_local = c5_of_h(h, dx)
        for frac in [1.0, 0.7, 0.5, 0.3, 0.1, 0.05]:
            a = frac * alpha_max
            h_new = h + a * grad
            h_new = project_feasible(h_new, target_sum)
            c5_new = c5_of_h(h_new, dx)
            if c5_new < best_c5_local - 1e-14:
                best_c5_local = c5_new
                best_alpha = a
                h = h_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                break
        
        if best_alpha == 0.0:
            T *= T_decay
        else:
            T = max(0.01, T * 0.995)
        
        if T < 0.01:
            # Try a few more steps with very low T
            T = 0.01
    
    return best_h, best_c5

import numpy as np
import time

def project_feasible(x, target_sum):
    """Project x onto {h: 0<=h_i<=1, sum(h)=target_sum}"""
    n = len(x)
    x = np.nan_to_num(x, nan=0.5, posinf=5.0, neginf=-5.0)
    x = np.clip(x, -10.0, 10.0)
    lo, hi = -10.0, 10.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            cap = 1.0 - h
            s = cap.sum()
            if s > 1e-10:
                h += cap * (diff / s)
        else:
            s = h.sum()
            if s > 1e-10:
                h -= h * (abs(diff) / s)
    return np.clip(h, 0.0, 1.0)

def c5_of_h(h, dx):
    raw = np.correlate(h, 1.0 - h, mode='full')
    return raw.max() * dx

def lse_weights(raw, T):
    m = len(raw)
    rmax = raw.max()
    w = np.exp((raw - rmax) / T)
    w += 1.0 / (16.0 * m)
    w /= w.sum()
    return w

def lse_gradient(h, weights, n):
    """Compute LSE-weighted gradient of all correlation terms."""
    grad = np.zeros(n)
    for p in range(2 * n - 1):
        k = p - (n - 1)
        u = weights[p]
        if u < 1e-10:
            continue
        if k >= 0:
            idx = np.arange(n - k)
            grad[idx] += u * (1.0 - h[idx + k])
            grad[idx + k] += -u * h[idx]
        else:
            idx = np.arange(-k, n)
            grad[idx] += u * (1.0 - h[idx + k])
            grad[idx + k] += -u * h[idx]
    return grad

def lse_descent(h, dx, target_sum, t0, max_time, 
                T_init=0.05, T_min=0.005, T_decay=0.995,
                momentum=0.5, max_no_improve=200):
    """LSE-based gradient descent with momentum."""
    n = len(h)
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    v = np.zeros(n)  # momentum buffer
    T = T_init
    no_improve = 0
    it = 0
    
    while time.time() - t0 < max_time:
        raw = np.correlate(h, 1.0 - h, mode='full')
        weights = lse_weights(raw, T)
        g = lse_gradient(h, weights, n)
        g -= g.mean()  # center for sum constraint
        
        if np.abs(g).max() < 1e-14:
            T = max(T_min, T * 0.9)
            if T == T_min:
                break
            continue
        
        # Update momentum
        v = momentum * v + (1.0 - momentum) * g
        
        # Compute max step from box constraints
        pos = v > 1e-15
        neg = v < -1e-15
        alpha_max = 1.0
        if pos.any():
            a_pos = (1.0 - h[pos]) / v[pos]
            if a_pos.min() < alpha_max:
                alpha_max = a_pos.min()
        if neg.any():
            a_neg = h[neg] / (-v[neg])
            if a_neg.min() < alpha_max:
                alpha_max = a_neg.min()
        
        if alpha_max < 1e-14:
            v *= 0.5  # dampen momentum
            T = max(T_min, T * 0.95)
            no_improve += 1
            if no_improve > max_no_improve:
                break
            continue

        improved = False
        for frac in [1.0, 0.8, 0.6, 0.4, 0.2, 0.1, 0.05]:

                    best_h = h_new.copy()

            v *= 0.8  # dampen

        if cur_sum > target_a:

                a[i] += 1
                cur_sum += 1

    h_rev = project_feasible(best_h[::-1].copy(), target_sum)
    variants.append(h_rev)
    # Perturbations
    for sig in [0.03, 0.06, 0.01, 0.10]:
        h_p = project_feasible(best_h + sig * rng.randn(n), target_sum)
        variants.append(h_p)
    
    for h_v in variants:
        if time.time() > phase2_end:
            break
        t_max = min(phase2_end, time.time() + budget_s * 0.06)
        h_opt, c5_opt = lse_descent(h_v, dx, target_sum, t0, t_max - t0,
                                     T_init=0.05, T_min=0.005, T_decay=0.995,
                                     momentum=0.5, max_no_improve=200)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
    
    # Phase 3: SA on quantized grid
    phase3_end = t0 + budget_s * 0.78
    sa_configs = [
        (4, 30000, 0),
        (8, 20000, 1),
        (4, 25000, 2),
        (8, 15000, 3),
    ]
    for (M, n_it, idx) in sa_configs:
        if time.time() > phase3_end:
            break
        t_max = min(phase3_end, time.time() + budget_s * 0.07)
        h_sa, c5_sa = sa_quantized(n, M, n_it, seed + 100 + idx, t0, t_max)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa
        else:
            # Try LSE polish on SA result
            if time.time() < phase3_end - 5:
                h_opt, c5_opt = lse_descent(h_sa, dx, target_sum, t0,
                                             min(phase3_end - 5, time.time() + 30) - t0,
                                             T_init=0.03, T_min=0.003,
                                             T_decay=0.995, momentum=0.5,
                                             max_no_improve=100)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt
    
    # Phase 4: Try different n values
    phase4_end = t0 + budget_s * 0.90
    x_orig = np.linspace(0.0, 2.0, n)
    for n_try in [100, 112, 120, 128, 140, 144]:
        if time.time() > phase4_end:
            break
        if n_try == n:
            continue
        x_new = np.linspace(0.0, 2.0, n_try)
        h_interp = np.interp(x_new, x_orig, best_h)
        h_interp = project_feasible(h_interp, n_try / 2.0)
        dx_try = 2.0 / n_try
        c5_init = c5_of_h(h_interp, dx_try)
        if c5_init < best_c5:
            best_c5 = c5_init
            best_h = h_interp
            n = n_try
            dx = dx_try
            target_sum = n / 2.0
        # Try LSE on this
        if time.time() < phase4_end - 10:
            t_max = min(phase4_end, time.time() + 20)
            h_opt, c5_opt = lse_descent(h_interp, dx_try, n_try / 2.0, t0, t_max - t0,
                                         T_init=0.05, T_min=0.005,
                                         T_decay=0.995, momentum=0.5,
                                         max_no_improve=150)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
                n = n_try
                dx = dx_try
                target_sum = n / 2.0
    
    # Phase 5: Final intensive polish
    if time.time() < budget_s - 5:
        dx_f = 2.0 / len(best_h)
        ts_f = len(best_h) / 2.0
        remaining = budget_s - 5 - (time.time() - t0)
        
        # Multiple LSE passes
        for (T_i, mom) in [(0.02, 0.5), (0.05, 0.3), (0.01, 0.7)]:
            if remaining < 10:
                break
            t_max = time.time() + min(remaining * 0.3, 40)
            h_opt, c5_opt = lse_descent(best_h, dx_f, ts_f, t0, t_max - t0,
                                         T_init=T_i, T_min=0.002,
                                         T_decay=0.998, momentum=mom,
                                         max_no_improve=300)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
            remaining = budget_s - 5 - (time.time() - t0)
                
        # Tiny perturbation
        if remaining > 5:
            h_tiny = project_feasible(best_h + 0.005 * rng.randn(len(best_h)), ts_f)
            t_max = time.time() + min(remaining * 0.5, 15)
            h_opt, c5_opt = lse_descent(h_tiny, dx_f, ts_f, t0, t_max - t0,
                                         T_init=0.02, T_min=0.002,
                                         T_decay=0.998, momentum=0.5,
                                         max_no_improve=200)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
    
    # Final validation
    n_final = len(best_h)
    dx_final = 2.0 / n_final
