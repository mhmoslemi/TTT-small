def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx])
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff

        else:
            idx = int(np.argmax(h))

            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] = np.clip(h[idx] + diff2, 0.0, 1.0)
    return h


def two_valued_alpha_score(B, n):
    """Compute C[k]*dx for all k as a quadratic in alpha for a balanced binary mask."""
    dx = 2.0 / n
    center = n - 1
    Bf = B.astype(np.float64)
    corr_bb = np.correlate(Bf, Bf, mode="full")
    prefix = np.zeros(n + 1)
    for i in range(n):
        prefix[i + 1] = prefix[i] + Bf[i]
    total_ones = int(round(prefix[n]))
    M = 2 * n - 1
    # For each correlate index m (shift k = m - center):
    # k >= 0: C[center+k] = dx * sum_{i=0}^{n-1-k} h[i+k](1-h[i])
    #   = dx * [a(1-a)*(n11+n00) + (1-a)^2 * n10 + a^2 * n01]
    # k < 0 (k'=-k): C[center-k'] = dx * sum_{i=0}^{n-1-k'} h[i](1-h[i+k'])
    #   = dx * [a(1-a)*(n11+n00) + a^2 * n10 + (1-a)^2 * n01]
    # Coeffs: C[k] = dx*(A*a^2 + Bc*a + C)
    A_arr = np.zeros(M)
    B_arr = np.zeros(M)
    C_arr = np.zeros(M)
    for k in range(-n + 1, n):
        m = center + k
        if k >= 0:
            nk = n - k
            n11 = corr_bb[center + k]
            sum_first = prefix[nk]
            sum_second = prefix[n] - prefix[k]
            n10 = sum_first - n11
            n01 = sum_second - n11
            n00 = nk - n11 - n10 - n01
            # (1-a)^2*n10 + a^2*n01 + a(1-a)*(n11+n00)
            # = n10 - 2*n10*a + n10*a^2 + n01*a^2 + (n11+n00)*a - (n11+n00)*a^2
            A_arr[m] = dx * (n10 + n01 - n11 - n00)
            B_arr[m] = dx * (-2.0 * n10 + n11 + n00)
            C_arr[m] = dx * n10
        else:
            kp = -k
            nk = n - kp
            n11 = corr_bb[center - kp]
            sum_first = prefix[nk]
            sum_second = prefix[n] - prefix[kp]
            n10 = sum_first - n11
            n01 = sum_second - n11
            n00 = nk - n11 - n10 - n01
            # a^2*n10 + (1-a)^2*n01 + a(1-a)*(n11+n00)
            A_arr[m] = dx * (n10 + n01 - n11 - n00)
            B_arr[m] = dx * (2.0 * n01 - n11 - n00)
            C_arr[m] = dx * n01
    return A_arr, B_arr, C_arr


def solve_two_valued(A_arr, B_arr, C_arr):
    """Find alpha in [0,1] minimizing max_k (A[k]*a^2 + B[k]*a + C[k])."""
    def objective(a):
        vals = A_arr * a * a + B_arr * a + C_arr
        return np.max(vals)
    res = minimize_scalar(objective, bounds=(0.01, 0.99), method='bounded',
                          options={'xatol': 1e-10})
    if res.success:
        return res.x, res.fun
    return 0.5, objective(0.5)


def two_valued_binary_score(B):
    """Compute the optimal two-valued score for a balanced binary mask."""
    n = len(B)
    A_arr, B_arr, C_arr = two_valued_alpha_score(B, n)
    alpha, score = solve_two_valued(A_arr, B_arr, C_arr)
    h = np.where(B > 0.5, alpha, 1.0 - alpha).astype(np.float64)
    h = repair_feasible(h)
    actual = exact_score(h)
    return actual, h


def anneal_binary(B, n, max_iters, t_start_fn, time_budget):
    """Simulated annealing on balanced binary mask. Returns best B and score."""
    B = B.copy().astype(np.int8)
    ones = np.where(B == 1)[0]
    zeros = np.where(B == 0)[0]
    current_score = two_valued_binary_score(B)[0]
    best_score = current_score
    best_B = B.copy()
    rng = np.random.RandomState(42 + n)
    T = 0.005
    cool = 0.97
    for it in range(max_iters):
        if time_budget() < 5:
            break
        if it % 50 == 0 and it > 0:
            T *= cool
            if T < 1e-6:
                T = 1e-6
        i1 = rng.randint(len(ones))
        i2 = rng.randint(len(zeros))
        B[ones[i1]] = 0
        B[zeros[i2]] = 1
        new_score = two_valued_binary_score(B)[0]
        delta = new_score - current_score
        if delta < 0 or rng.rand() < np.exp(-delta / (T + 1e-15)):
            current_score = new_score
            ones[i1] = zeros[i2]
            zeros[i2] = ones[i1]
            if current_score < best_score:
                best_score = current_score
                best_B = B.copy()
        else:
            B[ones[i1]] = 1
            B[zeros[i2]] = 0
    return best_B, best_score


def pnorn_gradient(h, n, dx, p):
    """Compute gradient of (sum C[k]^p)^(1/p) with respect to h."""
    profile = compute_profile(h)
    M = 2 * n - 1
    center = n - 1
    weights = profile ** (p - 1)
    weights = weights / (np.sum(weights) + 1e-30)
    grad = np.zeros(n)
    for k in range(-n + 1, n):
        m = center + k
        w = weights[m]
        if w < 1e-15:
            continue
        if k >= 0:
            nk = n - k
            for i in range(nk):
                grad[i + k] += dx * w * (1 - h[i])
                grad[i] += dx * w * (-h[i + k])
        else:
            kp = -k
            nk = n - kp
            for i in range(nk):
                grad[i] += dx * w * (1 - h[i + kp])
                grad[i + kp] += dx * w * (-h[i])
    profile_val = np.sum(profile ** p) ** (1.0 / p)
    grad *= profile_val / (p * np.mean(weights) + 1e-30)
    return grad


def projected_gradient_step(h, grad, step_size):
    h_new = h - step_size * grad
    return repair_feasible(h_new)


def pnorn_optimize(h0, max_iters, time_fn, time_budget):
    """Optimize using p-norm gradient descent."""
    n = len(h0)
    dx = 2.0 / n
    h = h0.copy()
    best_c = exact_score(h)
    p = 20
    step = 0.01
    stagnation = 0
    for it in range(max_iters):
        if time_budget() < 10:
            break
        curr_c = exact_score(h)
        if curr_c < best_c - 1e-14:
            best_c = curr_c
            stagnation = 0
        else:
            stagnation += 1
        if stagnation > 30:
            break
        grad = pnorn_gradient(h, n, dx, p)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        grad /= gnorm
        accepted = False
        for s in [step, step * 0.5, step * 0.25, step * 0.1]:
            h_try = projected_gradient_step(h, grad, s)
            c_try = exact_score(h_try)
            if c_try < best_c - 1e-14:
                h = h_try
                best_c = c_try
                accepted = True
                step = min(step * 1.1, 0.05)
                break
        if not accepted:
            step *= 0.6
            if step < 1e-5:
                break
        # Increase p over time
        if it % 50 == 0 and it > 0:
            p = min(p * 2, 200)
    return h, best_c


def active_set_lp_refine(h0, max_iters, time_budget):
    """Improved active set LP refinement."""
    if not HAS_CVXPY:
        return h0, exact_score(h0)
    n = len(h0)
    dx = 2.0 / n
    h = h0.copy()
    best_c = exact_score(h)
    target = n / 2.0
    trust = 0.01
    stagnation = 0
    for it in range(max_iters):
        if time_budget() < 15:
            break
        profile = compute_profile(h)
        c_max = np.max(profile)
        if c_max < best_c - 1e-14:
            best_c = c_max
            stagnation = 0
        else:
            stagnation += 1
        if stagnation > 50:
            break
        spread = c_max - np.median(profile)
        beam = max(0.001, min(0.02, 0.2 * spread))
        active = np.where(profile >= c_max - beam)[0]
        if len(active) == 0:
            active = np.array([np.argmax(profile)])
        if len(active) > 25:
            active = active[np.argsort(profile[active])[-25:]]
        # Build gradient matrix for active constraints
        M = len(active)
        G = np.zeros((M, n))
        center = n - 1
        for ii, m in enumerate(active):
            k = m - center
            if k >= 0:
                nk = n - k
                for i in range(nk):
                    G[ii, i + k] += -dx * h[i]
                    G[ii, i] += dx * (1 - h[i + k])
            else:
                kp = -k
                nk = n - kp
                for i in range(nk):
                    G[ii, i] += -dx * h[i + kp]
                    G[ii, i + kp] += dx * (1 - h[i])
        # LP: minimize t s.t. G @ h_new <= profile[active] + G @ (h_new - h) approx
        # Linearize: C[k] ≈ C[k]_curr + G[k] @ (h_new - h)
        G_a = G
        b_lin = profile[active] - G_a @ h
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        cons = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == target,
            h_var >= h - trust,
            h_var <= h + trust,
            G_a @ h_var + b_lin <= t_var,
        ]
        obj = cp.Minimize(t_var + 0.001 * cp.svm_hinge(t_var - c_max + 0.001))
        # Actually simpler: minimize t + rho*||h_new - h||_1
        obj = cp.Minimize(t_var + 0.005 * cp.norm1(h_var - h))
        prob = cp.Problem(obj, cons)
        solved = False
        for solver in [cp.ECOS, cp.SCS, cp.CBC]:
            try:
                prob.solve(solver=solver, max_iters=5000, abstol=1e-10)
                solved = True
                break
            except Exception:
                continue
        if not solved or h_var.value is None:
            trust *= 0.6
            if trust < 0.0005:
                break
            continue
        h_sol = np.array(h_var.value).flatten()
        if not np.isfinite(h_sol).all():
            trust *= 0.6
            continue
        h_sol = repair_feasible(h_sol)
        c_sol = exact_score(h_sol)
        if c_sol < best_c - 1e-14:
            h = h_sol
            best_c = c_sol
            trust = min(trust * 1.15, 0.04)
        else:
            for alpha in [0.5, 0.3, 0.15]:
                hb = repair_feasible(alpha * h_sol + (1 - alpha) * h)
                cb = exact_score(hb)
                if cb < best_c - 1e-14:
                    h = hb
                    best_c = cb
                    trust = min(trust * 1.05, 0.04)
                    break
            else:
                trust *= 0.5
    return h, best_c


def slsqp_refine(h0, maxiter, tol):
    n = len(h0)
    dx = 2.0 / n
    M = 2 * n - 1

    def obj(x):
        return x[-1]

    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g

    def eq_fun(x):
        return x[:-1].sum() - n / 2.0

    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g

    def ineq_fun(x):
        h = x[:-1]
        return x[-1] - np.correlate(h, 1 - h, mode="full") * dx

    def ineq_jac(x):
        h = x[:-1]
        center = n - 1
        J = np.zeros((M, n + 1))
        J[:, -1] = 1.0
        for k in range(-n + 1, n):
            m = center + k
            if k >= 0:
                nk = n - k
                for i in range(nk):
                    J[m, i + k] -= dx * (1 - h[i])
                    J[m, i] -= dx * (-h[i + k])
            else:
                kp = -k
                nk = n - kp
                for i in range(nk):
                    J[m, i] -= dx * (1 - h[i + kp])
                    J[m, i + kp] -= dx * (-h[i])
        return -J

    t0 = exact_score(h0) + 1e-3
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, t0 + 1.0)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac},
    ]
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': tol})
        if res.x is not None and np.isfinite(res.x).all():
            hs = repair_feasible(res.x[:-1])
            if np.isfinite(exact_score(hs)):
                return hs
    except Exception:
        pass
    return None


def interp(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_new if False else 2.0 / new_n)
    fx = np.concatenate([[0.0], xs, [2.0]])
    fh = np.concatenate([[h[0]], h, [h[-1]]])
    hn = np.interp(xt, fx, fh)
    return repair_feasible(np.clip(hn, 0, 1))


def micro_gradient_polish(h0, max_iters, time_budget):
    """Very fine gradient steps for final polish."""
    n = len(h0)
    dx = 2.0 / n
    h = h0.copy()
    best_c = exact_score(h)
    p = 50
    for it in range(max_iters):
        if time_budget() < 5:
            break
        profile = compute_profile(h)
        c_max = np.max(profile)
        if c_max < best_c - 1e-15:
            best_c = c_max
        active = np.where(profile >= c_max - 0.002)[0]
        if len(active) == 0:
            break
        if len(active) > 15:
            active = active[np.argsort(profile[active])[-15:]]
        # Compute subgradient direction
        center = n - 1
        grad = np.zeros(n)
        for m in active:
            k = m - center
            w = (profile[m] - profile[active].min() + 1e-8)
            if k >= 0:
                nk = n - k
                for i in range(nk):
                    grad[i + k] += -dx * w * h[i]
                    grad[i] += dx * w * (1 - h[i + k])
            else:
                kp = -k
                nk = n - kp
                for i in range(nk):
                    grad[i] += -dx * w * h[i + kp]
                    grad[i + kp] += dx * w * (1 - h[i])
        grad -= grad.mean()
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        grad /= gn
        accepted = False
        for s in [0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-5, 2e-5]:
            h_try = repair_feasible(h - s * grad)
            c_try = exact_score(h_try)
            if c_try < best_c - 1e-15:
                h = h_try
                best_c = c_try
                accepted = True
                break
        if not accepted:
            break
    return h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()
    rng = np.random.RandomState(seed)

    def time_budget():
        return budget_s - (time.perf_counter() - t_start)

    # Load parent
    parent = None
    try:
        parent = np.array(initial_h_values, dtype=np.float64)
    except (NameError, Exception):
        pass
    if parent is None or len(parent) < 4:
        parent = np.full(70, 0.5)
    parent = repair_feasible(np.clip(parent, 0.0, 1.0))
    best_h = parent.copy()
    best_c = exact_score(parent)

    # Try symmetrized version
    h_sym = repair_feasible((parent + parent[::-1]) / 2.0)
    c_sym = exact_score(h_sym)
    if c_sym < best_c - 1e-12:
        best_h, best_c = h_sym, c_sym

    n_parent = len(best_h)

    # Phase 1: Two-valued binary search at multiple resolutions
    if time_budget() > 200:
        for nd in [40, 50, 60, n_parent, 80]:
            if time_budget() < 60:
                break
            if nd == n_parent:
                B = (best_h > 0.5).astype(np.int8)
                # Ensure balanced
                ones = np.sum(B)
                target_ones = nd // 2
                if ones > target_ones:
                    idx = np.where(B == 1)[0]
                    rng.shuffle(idx)
                    for i in range(ones - target_ones):
                        B[idx[i]] = 0
                elif ones < target_ones:
                    idx = np.where(B == 0)[0]
                    rng.shuffle(idx)
                    for i in range(target_ones - ones):
                        B[idx[i]] = 1
                # Anneal
                max_ann = 200
                best_B, best_bs = anneal_binary(B, nd, max_ann, t_start, time_budget)
                sc, h_tv = two_valued_binary_score(best_B)
                if sc < best_c - 1e-10:
                    best_h, best_c = h_tv, sc
            else:
                # Random binary starting pattern
                n_half = nd // 2
                B = np.zeros(nd, dtype=np.int8)
                idx = rng.permutation(nd)[:n_half]
                B[idx] = 1
                # Also try structured pattern (alternating blocks)
                for pattern_type in range(3):
                    if time_budget() < 30:
                        break
                    if pattern_type == 0:
                        # Random
                        B_try = np.zeros(nd, dtype=np.int8)
                        idx_try = rng.permutation(nd)[:n_half]
                        B_try[idx_try] = 1
                    elif pattern_type == 1:
                        # Alternating blocks of size 2
                        B_try = np.zeros(nd, dtype=np.int8)
                        for i in range(0, nd, 4):
                            B_try[i:i + 2] = 1
                        # Balance
                        s = np.sum(B_try)
                        if s > n_half:
                            ones_idx = np.where(B_try == 1)[0]
                            for i in range(s - n_half):
                                B_try[ones_idx[i]] = 0
                        elif s < n_half:
                            zeros_idx = np.where(B_try == 0)[0]
                            for i in range(n_half - s):
                                B_try[zeros_idx[i]] = 1
                    else:
                        # Periodic with period 5: 11000 pattern
                        B_try = np.tile([1, 1, 0, 0, 0], nd // 5 + 1)[:nd].astype(np.int8)
                        s = np.sum(B_try)
                        if s > n_half:
                            ones_idx = np.where(B_try == 1)[0]
                            for i in range(s - n_half):
                                B_try[ones_idx[i]] = 0
                        elif s < n_half:
                            zeros_idx = np.where(B_try == 0)[0]
                            for i in range(n_half - s):
                                B_try[zeros_idx[i]] = 1
                    if np.sum(B_try) != n_half:
                        continue
                    max_ann = 300
                    best_B_try, best_bs_try = anneal_binary(B_try, nd, max_ann, t_start, time_budget)
                    sc_try, h_tv_try = two_valued_binary_score(best_B_try)
                    if sc_try < best_c - 1e-10:
                        best_h, best_c = h_tv_try, sc_try

    # Phase 2: p-norm gradient optimization from best solution
    if time_budget() > 150:
        h_pn, c_pn = pnorn_optimize(best_h, max_iters=200, time_fn=time.perf_counter,
                                    time_budget=time_budget)
        if c_pn < best_c - 1e-14:
            best_h, best_c = h_pn, c_pn

    # Phase 3: Active-set LP refinement
    if time_budget() > 100:
        h_lp, c_lp = active_set_lp_refine(best_h, max_iters=200, time_budget=time_budget)
        if c_lp < best_c - 1e-14:
            best_h, best_c = h_lp, c_lp

    # Phase 4: Try different resolutions and interp back
    if time_budget() > 120:
        n_cur = len(best_h)
        for nd in [n_cur - 10, n_cur - 5, n_cur + 5, n_cur + 10]:
            if time_budget() < 40:
                break
            if nd < 20 or nd > 120:
                continue
            h_interp = interp(best_h, nd)
            c_interp = exact_score(h_interp)
            if c_interp < best_c + 0.05:
                h_pn2, c_pn2 = pnorn_optimize(h_interp, max_iters=100, time_fn=time.perf_counter,
                                              time_budget=time_budget)
                if c_pn2 < best_c - 1e-14:
                    best_h, best_c = h_pn2, c_pn2
                h_lp2, c_lp2 = active_set_lp_refine(best_h, max_iters=100, time_budget=time_budget)
                if c_lp2 < best_c - 1e-14:
                    best_h, best_c = h_lp2, c_lp2

    # Phase 5: SLSQP refinement
    if time_budget() > 80:
        hs = slsqp_refine(best_h, maxiter=500, tol=1e-12)
        if hs is not None:
            cs = exact_score(hs)
            if cs < best_c - 1e-14:
                best_h, best_c = hs, cs

    # Phase 6: Another round of LP + p-norm
    if time_budget() > 60:
        h_pn3, c_pn3 = pnorn_optimize(best_h, max_iters=150, time_fn=time.perf_counter,
                                      time_budget=time_budget)
        if c_pn3 < best_c - 1e-14:
            best_h, best_c = h_pn3, c_pn3
        h_lp3, c_lp3 = active_set_lp_refine(best_h, max_iters=150, time_budget=time_budget)
        if c_lp3 < best_c - 1e-14:
            best_h, best_c = h_lp3, c_lp3

    # Phase 7: SLSQP again
    if time_budget() > 50:
        hs2 = slsqp_refine(best_h, maxiter=600, tol=1e-13)
        if hs2 is not None:
            cs2 = exact_score(hs2)
            if cs2 < best_c - 1e-14:
                best_h, best_c = hs2, cs2

    # Phase 8: Micro gradient polish
    if time_budget() > 20:
        h_micro, c_micro = micro_gradient_polish(best_h, max_iters=200, time_budget=time_budget)
        if c_micro < best_c - 1e-15:
            best_h, best_c = h_micro, c_micro

    # Phase 9: Try random perturbations + quick refine
    if time_budget() > 50:
        for eps in [0.01, 0.02, 0.03, 0.05]:
            if time_budget() < 20:
                break
            z = rng.randn(len(best_h))
            z -= z.mean()
            z /= (np.linalg.norm(z) / len(best_h) + 1e-15)
            hp = repair_feasible(best_h + eps * z)
            cp_ = exact_score(hp)
            if cp_ < best_c + 0.01:
                h_lp4, c_lp4 = active_set_lp_refine(hp, max_iters=80, time_budget=time_budget)
                if c_lp4 < best_c - 1e-14:
                    best_h, best_c = h_lp4, c_lp4

    # Final SLSQP
    if time_budget() > 30:
        hs3 = slsqp_refine(best_h, maxiter=400, tol=1e-13)
        if hs3 is not None:
            cs3 = exact_score(hs3)
            if cs3 < best_c - 1e-14:
                best_h, best_c = hs3, cs3

    # Final micro polish
    if time_budget() > 10:
        h_final, c_final = micro_gradient_polish(best_h, max_iters=100, time_budget=time_budget)
        if c_final < best_c - 1e-15:
            best_h, best_c = h_final, c_final

    # Final validation
    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)

    try:
        er = evaluate_erdos_solution(best_h)
        if isinstance(er, (float, int)):
            final_c = min(final_c, float(er))
        elif isinstance(er, tuple) and len(er) >= 2:
            final_c = min(final_c, float(er[0]))
    except Exception:
        pass

    return best_h, final_c, len(best_h)
