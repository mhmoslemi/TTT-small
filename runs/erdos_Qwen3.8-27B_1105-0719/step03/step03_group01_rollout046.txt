Let me also add a technique I haven't seen in the previous code: **exploiting the bilinear structure for exact line search**.

Given a direction d (from SLSQP or gradient), the correlation at h + α*d is:
corr_k(h + α*d) = sum (h_i + α*d_i)(1 - h_j - α*d_j) for valid pairs
= corr_k(h) + α * grad_corr_k(h)_k + α² * (d_i * d_j summed appropriately)

The α² term is quadratic in α, so I can find the exact minimum of each active correlation along the line! This is much better than the bisection/line search the previous program used.

Specifically, for a bilinear function f(h) = sum a_ij * h_i * h_j + linear terms:
f(h + α*d) = f(h) + α * (∇f·d) + α² * (d^T A d)

where A is the bilinear form matrix. For our correlation:
f(α) = corr_k(h + α*d) = corr_k(h) + α * g_k + α² * q_k

where g_k = ∇corr_k(h) · d and q_k = sum over valid pairs (i,j) of d_i * d_j (with appropriate signs).

Actually, let me be more precise. corr_k = sum valid h_{n+d} (1-h_n) = sum valid h_{n+d} - sum valid h_{n+d} h_n.

d(corr_k)/dα = sum valid d_{n+d} - sum valid (d_{n+d} h_n + h_{n+d} d_n)
= sum valid d_{n+d} - sum valid d_{n+d}(h_n + α d_n) - sum valid (h_{n+d} + α d_{n+d}) d_n [at α]

Hmm, this is getting complicated. Let me just compute it directly:

corr_k(h + α*d) = sum valid (h_{n+d} + α d_{n+d})(1 - h_n - α d_n)
= sum valid [(h_{n+d} + α d_{n+d})(1-h_n) - α d_{n+d}(h_n + α d_n)]
= sum valid (h_{n+d})(1-h_n) + α sum valid d_{n+d}(1-h_n) - α sum valid d_{n+d} h_n - α² sum valid d_{n+d} d_n
= corr_k(h) + α [sum valid d_{n+d}(1-h_n) - sum valid d_{n+d} h_n] - α² sum valid d_{n+d} d_n
= corr_k(h) + α [∇corr_k(h)·d] - α² sum valid d_{n+d} d_n

So it's quadratic in α! The coefficient of α² is always negative (or zero), meaning corr_k is concave in α along any direction. This is because the bilinear term -h_{n+d}*h_n has a negative Hessian contribution.

For the epigraph problem, we want the largest α such that all constraints are satisfied:

def project_feasible(x, target_sum):
    n = len(x)
    x = np.array(x, dtype=float)
    x = np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0)
    x = np.clip(x, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        cnt = mask.sum()
        if cnt > 0:
            h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)


def compute_c5(h, dx):
    corrs = np.correlate(h, 1.0 - h, mode='full')
    return float((corrs * dx).max())


def grad_corr_k(h, k, n):
    d = k - (n - 1)
    grad = np.zeros(n)
    if d == 0:
        grad[:] = 1.0 - 2.0 * h
        return grad
    if d > 0:
        grad[:n - d] += -h[d:]
        grad[d:] += 1.0 - h[:n - d]
    else:
        dp = -d
        grad[dp:] += -h[:n - dp]
        grad[:n - dp] += 1.0 - h[dp:]
    return grad


def quadratic_line_coeff(h, direction, k, n):
    """Compute coefficients for corr_k(h + alpha*direction) = c0 + c1*alpha + c2*alpha^2.
    
    corr_k(h + a*dir) = corr_k(h) + a*(grad . dir) - a^2 * sum(valid pairs) dir[i]*dir[j]
    The c2 coefficient is -sum_valid dir[n+d]*dir[n] which is always <= 0.
    """
    d = k - (n - 1)
    g = grad_corr_k(h, k, n)
    c1 = g.dot(direction)
    c0 = np.sum(np.correlate(h, 1.0 - h, mode='full')[k])
    if d == 0:
        c2 = -np.sum(direction**2)
    elif d > 0:
        c2 = -np.sum(direction[:n - d] * direction[d:])
    else:
        dp = -d
        c2 = -np.sum(direction[:n - dp] * direction[dp:])
    return c0, c1, c2


def slsqp_solve(h0, dx, target_sum, t_margin=1e-4, n_active=None, maxiter=150, maxfun=5000):
    """Solve epigraph min t s.t. t >= dx*corr[p] for active p, sum(h)=target, 0<=h<=1."""
    n = len(h0)
    n_total = 2 * n - 1
    corrs = np.correlate(h0, 1.0 - h, mode='full') if False else np.correlate(h0, 1.0 - h0, mode='full')
    c5_init = (corrs * dx).max()

    if n_active is None or n_active >= n_total:
        active = np.arange(n_total)
    else:
        active = np.argsort(corrs)[::-1][:n_active]

    m = len(active)
    t0_val = c5_init + t_margin
    x0 = np.concatenate([[t0_val], h0])

    def objective(x):
        return x[0]

    def grad_obj(x):
        g = np.zeros(n + 1)
        g[0] = 1.0
        return g

    def constraints_inq(x):
        t = x[0]
        h = np.clip(x[1:], 0.0, 1.0)
        hs = h.sum()
        # scale to maintain sum (SLSQP will handle equality separately, but keep feasible)
        return t - dx * np.correlate(h, 1.0 - h, mode='full')[active]

    def jac_inq(x):

        h = np.clip(x[1:], 0.0, 1.0)
        J = np.zeros((m, n + 1))
        J[:, 0] = 1.0
        for i, ak in enumerate(active):
            J[i, 1:] = -dx * grad_corr_k(h, ak, n)
        return J

    def kend(x):
        h = x[1:]
        return h.sum() - target_sum

    def jake(x):
        J = np.zeros(n + 1)
        J[1:] = 1.0
        return J

    cons = [{'type': 'ineq', 'fun': constraints_inq, 'jac': jac_inq},
            {'type': 'eq', 'fun': kend, 'jac': jake}]

    bounds = [(0.0, 0.6)] + [(0.0, 1.0)] * n

    try:

            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r
            dt_rem = budget_s - 10 - (time.time() - t0)

        # Final quadratic polish
        if dt_rem > 5:
            h_pol, c5_pol = quadratic_polish(best_h, dx_b, ts_b,
                                             n_steps=min(20, int(dt_rem)),
                                             n_active=10)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol

    # Final validation and repair
    n_final = len(best_h)
    dx_final = 2.0 / n_final
    ts_final = n_final / 2.0
    best_h = project_feasible(best_h, ts_final)
    best_c5 = compute_c5(best_h, dx_final)

    return best_h, best_c5, n_final
