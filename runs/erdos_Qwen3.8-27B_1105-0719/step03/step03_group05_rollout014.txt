import numpy as np
from scipy.optimize import linprog
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx] + 1e-15)
            h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx] + 1e-15)
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-14:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmin(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def build_gradient(h):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    G = np.zeros((M, n))
    G[n - 1, :] = dx * (1.0 - 2.0 * h)
    for d in range(1, n):
        G[n - 1 + d, d:] += dx * (1.0 - h[:n - d])
        G[n - 1 + d, :n - d] += -dx * h[d:]
        G[n - 1 - d, :n - d] += dx * (1.0 - h[d:])
        G[n - 1 - d, d:] += -dx * h[:n - d]
    return G

def verify_gradient(h):
    n = len(h)
    dx = 2.0 / n
    G = build_gradient(h)
    P = compute_profile(h)
    rng_chk = np.random.RandomState(0)

        actual = P_pert - P
        predicted = G @ d
        if np.max(np.abs(actual - predicted)) > 1e-8 * (1 + np.max(np.abs(P))):

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    edges_old = np.linspace(0, 2, n + 1)
    edges_new = np.linspace(0, 2, new_n + 1)
    h_interp = np.zeros(new_n)
    for k in range(new_n):
        lo, hi = edges_new[k], edges_new[k + 1]
        mass = 0.0
        width = 0.0
        for j in range(n):
            olo = max(lo, edges_old[j])
            ohi = min(hi, edges_old[j + 1])
            if ohi > olo:
                mass += h[j] * (ohi - olo)
                width += (ohi - olo)
        h_interp[k] = mass / width if width > 1e-15 else 0.5
    return repair_feasible(np.clip(h_interp, 0, 1))

def lp_minimax_direction(h, D=0.5):
    n = len(h)
    M = 2 * n - 1
    P = compute_profile(h)
    G = build_gradient(h)
    c_cur = float(np.max(P))
    nv = n + 1
    c_obj = np.zeros(nv)
    c_obj[-1] = 1.0
    A_ub = np.zeros((M, nv))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -P.copy()
    A_eq = np.zeros((1, nv))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(-D, D)] * n + [(-2.0, 2.0)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'dual_feasibility_tolerance': 1e-10})
        if res.status == 0:
            d = res.x[:n].copy()
            d = d - d.mean()
            t_pred = res.x[-1]
            if t_pred < c_cur - 1e-11:
                return d, t_pred
    except Exception:
        pass
    return None, None

def subgradient_descent(h, current_best, max_iters=50, alpha=0.01):
    n = len(h)
    h = h.copy()
    best_c = current_best
    for _ in range(max_iters):
        P = compute_profile(h)
        c = float(np.max(P))
        if c < best_c:
            best_c = c
        margin = max(0.003, 3.0 * (c - float(np.median(P))))
        active = np.where(P >= c - margin)[0]
        weights = np.maximum(0, P[active] - (c - margin))
        if weights.sum() < 1e-15:
            weights = np.ones(len(active))
        weights /= weights.sum()
        G = build_gradient(h)
        G_a = G[active, :]
        grad = G_a.T @ weights
        grad -= grad.mean()
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        grad /= gn
        a_try = alpha
        improved = False
        for _ in range(6):
            h_try = repair_feasible(h - a_try * grad)
            s = exact_score(h_try)
            if s < best_c - 1e-14:
                h = h_try
                best_c = s
                improved = True
                alpha = min(alpha * 1.15, 0.04)
                break
            elif s < exact_score(h) - 1e-14:
                h = h_try
                improved = True
                break
            a_try *= 0.5
        if not improved:
            alpha *= 0.6
            if alpha < 1e-8:
                break
    return h, best_c

                    h[i] += delta
                    h[j] -= delta
                    s = exact_score(h)
                    if s < best_pair_s - 1e-14:
                        best_pair_s = s
                        best_move = (i, j, delta)
                    h[i] -= delta
                    h[j] += delta
                    tries += 1
                    if tries >= max_pairs * 5:

                if tries >= max_pairs * 5:

    n_p = len(best_h)
    h_as = np.zeros(n_p)
    for i in range(n_p):
        h_as[i] = (best_h[i] + (1.0 - best_h[n_p - 1 - i])) / 2.0
    h_as = repair_feasible(h_as)
    c_as = exact_score(h_as)
    if c_as < best_c - 1e-12:
        best_h = h_as
        best_c = c_as

    # Phase 1: Main LP local descent
    if time_left() > 400:
        h2, c2 = local_optimize(best_h, best_c, min(350, time_left() - 100), rng)
        if c2 < best_c - 1e-14:
            best_h = h2
            best_c = c2

    # Phase 2: Resolution exploration
    if time_left() > 250:
        n_cur = len(best_h)
        candidates_n = []
        for delta in [-8, -6, -4, -2, 2, 4, 6, 8, 10, 12]:
            nn = n_cur + delta
            if 30 <= nn <= 140 and nn % 2 == 0:
                candidates_n.append(nn)
        # Also try some "round" numbers
        for nn in [40, 50, 56, 60, 64, 68, 72, 76, 80, 84, 90, 96, 100]:
            if 30 <= nn <= 140 and nn % 2 == 0 and nn != n_cur:
                candidates_n.append(nn)
        candidates_n = list(set(candidates_n))
        candidates_n.sort(key=lambda x: abs(x - n_cur))

        for nn in candidates_n[:8]:
            if time_left() < 80:
                break
            h_interp = interpolate_to_n(best_h, nn)
            c_interp = exact_score(h_interp)
            h2, c2 = local_optimize(h_interp, c_interp, min(55, time_left() - 50), rng)
            if c2 < best_c - 1e-14:
                best_h = h2
                best_c = c2

    # Phase 3: Second LP polish at best resolution
    if time_left() > 150:
        h2, c2 = local_optimize(best_h, best_c, min(130, time_left() - 50), rng)
        if c2 < best_c - 1e-14:
            best_h = h2
            best_c = c2

    # Phase 4: Try neighboring resolutions of current best
    if time_left() > 100:
        n_best = len(best_h)
        for delta in [-2, 2, -4, 4]:
            if time_left() < 50:
                break
            nn = n_best + delta
            if nn < 30 or nn > 140 or nn % 2 != 0:
                continue
            h_interp = interpolate_to_n(best_h, nn)
            h2, c2 = local_optimize(h_interp, exact_score(h_interp), min(40, time_left() - 30), rng)
            if c2 < best_c - 1e-14:
                best_h = h2
                best_c = c2

    # Phase 5: Final fine-tuning
    if time_left() > 50:
        h2, c2 = local_optimize(best_h, best_c, min(40, time_left() - 15), rng)
        if c2 < best_c - 1e-14:
            best_h = h2
            best_c = c2

    # Final: try to reduce resolution while maintaining score
    if time_left() > 30:
        n_final = len(best_h)
        for delta in [-2, -4, -6]:
            if time_left() < 20:
                break
            nn = n_final + delta
            if nn < 26 or nn % 2 != 0:
                continue
            h_trial = interpolate_to_n(best_h, nn)
            c_trial = exact_score(h_trial)
            h2, c2 = local_optimize(h_trial, c_trial, min(20, time_left() - 10), rng)
            if c2 < best_c - 1e-14:
                best_h = h2
                best_c = c2

    # Final validation
    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)

    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            final_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 2:
            final_c = float(eval_result[0])
    except Exception:
        pass

    return best_h, final_c, len(best_h)
