import numpy as np
import cvxpy as cp
from scipy.optimize import minimize
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def repair_feasible(h):
    n = len(h)
    h = np.clip(np.asarray(h, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, 1.0 - h[idx])
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-13:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-13:
                break
    # Final fix
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        idx = int(np.argmax(1.0 - h)) if diff2 > 0 else int(np.argmin(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def build_gradient(h, n, dx):
    """Build M x n gradient matrix where M = 2n-1."""
    M = 2 * n - 1
    G = np.zeros((M, n))
    center = n - 1
    # d = 0
    G[center, :] = dx * (1.0 - 2.0 * h)
    # d > 0
    for d in range(1, n):
        row = np.zeros(n)
        # i from 0 to n-1-d, j = i+d
        # ∂/∂h[i] += dx*(1-h[i+d]) for i in [0, n-d-1]
        row[0:n-d] += dx * (1.0 - h[d:n])
        # ∂/∂h[j] += -dx*h[i-d] for j in [d, n-1] (where i=j-d)
        row[d:n] -= dx * h[0:n-d]
        G[center + d, :] = row
    # d < 0, let e = -d
    for e in range(1, n):
        d = -e
        row = np.zeros(n)
        # i from e to n-1, j = i-e
        # ∂/∂h[i] += dx*(1-h[i-e]) for i in [e, n-1]
        row[e:n] += dx * (1.0 - h[0:n-e])
        # ∂/∂h[j] += -dx*h[j+e] for j in [0, n-e-1]
        row[0:n-e] -= dx * h[e:n]
        G[center - e, :] = row
    return G

def get_active_set(profile, top_k=50, beam=None):
    T = np.max(profile)
    spread = T - np.min(profile)
    if beam is None:
        beam = max(0.001, 0.05 * spread)
    active = np.where(profile >= T - beam)[0]
    if len(active) == 0:
        active = np.array([np.argmax(profile)])
    if len(active) > top_k:
        idx = np.argsort(profile[active])[-top_k:]
        active = active[idx]
    return active, T

def solve_qp_step(h, active, radius, gamma):
    """Solve the QP subproblem. Returns (delta, predicted_t) or (None, None)."""
    n = len(h)
    dx = 2.0 / n
    G = build_gradient(h, n, dx)
    profile = compute_profile(h)
    G_a = G[active, :]
    P_a = profile[active]
    
    delta = cp.Variable(n)
    t = cp.Variable()
    
    lower = np.maximum(-radius, -h)
    upper = np.minimum(radius, 1.0 - h)
    
    constraints = [
        G_a @ delta <= t - P_a,
        cp.sum(delta) == 0,
        delta >= lower,
        delta <= upper,
        t >= 0
    ]
    
    objective = cp.Minimize(t + gamma * cp.sum_squares(delta))
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.CLARAB, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, verbose=False)
        except Exception:
            return None, None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    if delta.value is None:
        return None, None
    
    delta_val = np.asarray(delta.value, dtype=np.float64)
    t_val = float(t.value) if t.value is not None else 0.0
    return delta_val, t_val

def line_search(h, direction, current_c, max_step=1.0):
    """Line search along direction, returns (best_h, best_c)."""
    best_h = h.copy()
    best_c = current_c
    for alpha in [max_step, max_step*0.7, max_step*0.5, max_step*0.35, 
                  max_step*0.2, max_step*0.1, max_step*0.05, max_step*0.02]:
        h_try = repair_feasible(h + alpha * direction)
        c_try = exact_score(h_try)
        if c_try < best_c - 1e-14:
            best_h = h_try
            best_c = c_try
    return best_h, best_c

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:

    xs = (np.arange(n) + 0.5) * (2.0 / n)

    h_new = np.interp(xt, xs, h)

import numpy as np
import cvxpy as cp
from scipy.optimize import minimize
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def repair_feasible(h):
    n = len(h)
    h = np.clip(np.asarray(h, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, 1.0 - h[idx])
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-13:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-13:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        idx = int(np.argmax(1.0 - h)) if diff2 > 0 else int(np.argmin(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def build_gradient(h, n, dx):
    M = 2 * n - 1
    G = np.zeros((M, n))
    center = n - 1
    G[center, :] = dx * (1.0 - 2.0 * h)
    for d in range(1, n):
        row = np.zeros(n)
        row[0:n-d] += dx * (1.0 - h[d:n])
        row[d:n] -= dx * h[0:n-d]
        G[center + d, :] = row
        row2 = np.zeros(n)
        row2[d:n] += dx * (1.0 - h[0:n-d])
        row2[0:n-d] -= dx * h[d:n]
        G[center - d, :] = row2
    return G

def get_active_set(profile, top_k=50, beam=None):
    T = float(np.max(profile))
    spread = T - float(np.min(profile))
    if beam is None:
        beam = max(0.002, 0.06 * spread)
    active = np.where(profile >= T - beam)[0]
    if len(active) == 0:
        active = np.array([np.argmax(profile)])
    if len(active) > top_k:
        idx = np.argsort(profile[active])[-top_k:]
        active = active[idx]
    return active, T

def solve_qp(h, active, radius, gamma):
    n = len(h)
    dx = 2.0 / n
    G = build_gradient(h, n, dx)
    profile = compute_profile(h)
    G_a = G[active, :]
    P_a = profile[active]
    delta = cp.Variable(n)
    t = cp.Variable()
    lower = np.maximum(-radius, -h)
    upper = np.minimum(radius, 1.0 - h)
    constraints = [G_a @ delta <= t - np.asarray(P_a), cp.sum(delta) == 0, delta >= lower, delta <= upper, t >= 0]
    objective = cp.Minimize(t + gamma * cp.sum_squares(delta))
    prob = cp.Problem(objective, constraints)
    solvers_to_try = ['SCS', 'ECOS', 'CVXOPT', 'PDLP', 'CLARAB']
    for sname in solvers_to_try:
        try:
            prob.solve(solver=cp.SCS if sname == 'SCS' else None) if False else None
        except Exception:
            continue
    solved = False
    for sname in ('SCS', 'ECOS', 'CVXOPT', 'PDLP'):
        try:
            prob.solve(solver=sname, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                solved = True
                break
        except Exception:
            continue
    if not solved:
        return None, None
    if delta.value is None:
        return None, None
    delta_val = np.asarray(delta.value, dtype=np.float64).flatten()
    if not np.isfinite(delta_val).all():
        return None, None
    t_val = float(t.value) if t.value is not None else 0.0
    return delta_val, t_val

def smooth_max_grad(h, active, profile, beta=50.0):
    n = len(h)
    dx = 2.0 / n
    G = build_gradient(h, n, dx)
    G_a = G[active, :]
    P_a = profile[active].copy()
    P_a = P_a - np.max(P_a)
    weights = np.exp(beta * P_a)
    wsum = weights.sum()
    if wsum < 1e-300:
        weights = np.ones(len(active)) / len(active)
    else:
        weights /= wsum
    grad = G_a.T @ weights
    grad -= grad.mean()
    return grad

def line_search(h, direction, c_cur, alphas=None):
    if alphas is None:
        alphas = [1.0, 0.7, 0.5, 0.35, 0.2, 0.1, 0.05, 0.02]
    best_h = h.copy()
    best_c = c_cur
    for a in alphas:
        h_try = repair_feasible(h + a * direction)
        c_try = exact_score(h_try)
        if c_try < best_c - 1e-14:
            best_h = h_try
            best_c = c_try
    return best_h, best_c

def symmetrize(h):
    return repair_feasible((h + h[::-1]) / 2.0)

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    h_new = np.interp(xt, xs, h)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)
