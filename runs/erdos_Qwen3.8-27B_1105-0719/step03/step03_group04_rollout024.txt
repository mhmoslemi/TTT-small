def exact_score(h):
    n = len(h)

    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def repair_feasible(h):
    n = len(h)
    h = np.clip(np.asarray(h, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        # Need to add mass; add to positions with most room
        order = np.argsort(1.0 - h)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, 1.0 - h[idx])
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        # Need to remove mass; remove from positions with most mass
        order = np.argsort(h)[::-1]
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    # Final fix
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def symmetrize(h):
    return (h + 1.0 - h[::-1]) / 2.0

    n = len(h)
    h = np.clip(np.asarray(h, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, 1.0 - h[idx])
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)[::-1]
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break

        else:
            idx = int(np.argmax(h))

    return h

    """Compute gradient of log-sum-exp smoothed max of profile."""
    n = len(h)
    dx = 2.0 / n
    profile = np.correlate(h, 1 - h, mode="full") * dx

        grad[d:] += W[d] * (1.0 - h[:n-d])

    return grad * dx

def solve_lse_slsqp(h0, tau, t_init=None, maxiter=500):
    """Solve min t s.t. t >= L_tau(h), sum(h)=n/2, 0<=h<=1 using SLSQP."""
    n = len(h0)
    h0 = h0.copy()
    
    if t_init is None:
        t_init = exact_score(h0) + 1e-4
    
    def obj(x):
        return x[-1]
    
    def obj_grad(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    
    def eq_constraint(x):
        return x[:-1].sum() - n / 2.0
    
    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g
    
    def ineq_constraint(x):
        return x[-1] - lse_value(x[:-1], tau)
    
    def ineq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = -lse_gradient(x[:-1], tau)
        g[-1] = 1.0
        return g
    
    x0 = np.concatenate([h0, [t_init]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]

        {'type': 'ineq', 'fun': ineq_constraint, 'jac': ineq_jac}
    ]

                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-12})
    
    if res.success and np.isfinite(res.x).all():
        h_sol = repair_feasible(res.x[:-1])
        return h_sol, exact_score(h_sol)

    p_max = profile.max()
    exp_p = np.exp((profile - p_max) / tau)
    weights = exp_p / exp_p.sum()
    center = n - 1
    W = np.zeros(n)
    for k in range(2 * n - 1):
        d = abs(k - center)
        W[d] += weights[k]
    grad = np.zeros(n)
    for d in range(n):
        if W[d] < 1e-15:
            continue
        grad[d:] += W[d] * (1.0 - h[:n - d])
        grad[:n - d] -= W[d] * h[d:]
    return grad * dx

def solve_slsqp(h0, tau, maxiter=500, ftol=1e-14):
    n = len(h0)
    t_init = exact_score(h0) + 1e-4

    def obj(x):
        return x[-1]

    def obj_grad(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g

    def eq_c(x):
        return x[:-1].sum() - n / 2.0

    def eq_j(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g

    def ineq_c(x):
        return x[-1] - lse_value(x[:-1], tau)

    def ineq_j(x):
        g = np.zeros(n + 1)
        g[:-1] = -lse_gradient(x[:-1], tau)
        g[-1] = 1.0
        return g

    x0 = np.concatenate([h0, [t_init]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    cons = [
        {'type': 'eq', 'fun': eq_c, 'jac': eq_j},
        {'type': 'ineq', 'fun': ineq_c, 'jac': ineq_j}
    ]
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_grad,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        if res.success and np.isfinite(res.x).all():
            h_sol = repair_feasible(res.x[:-1])
            return h_sol, exact_score(h_sol)
    except Exception:
        pass
    return None, None

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    return repair_feasible(np.clip(h_new, 0.0, 1.0))

def coordinate_polish(h, max_rounds=50):
    n = len(h)
    dx = 2.0 / n
    best_c = exact_score(h)
    for rnd in range(max_rounds):
        improved = False
        profile = np.correlate(h, 1 - h, mode="full") * dx
        active = set(np.where(profile > best_c - 0.005)[0].tolist())
        for i in range(n):
            if (time.perf_counter() - t_start_global) > budget_global - 10:
                break
            for delta in [0.02, -0.02, 0.01, -0.01, 0.005, -0.005]:
                h_try = h.copy()
                h_try[i] += delta
                if h_try[i] < 0 or h_try[i] > 1:
                    continue
                h_try = repair_feasible(h_try)
                c_try = exact_score(h_try)
                if c_try < best_c - 1e-14:
                    h = h_try
                    best_c = c_try
                    improved = True
                    break
        if not improved:
            break
    return h, best_c

def projected_gradient_step(h, tau, lr, n_steps=50):
    n = len(h)
    best_c = exact_score(h)
    best_h = h.copy()
    for step in range(n_steps):
        g = lse_gradient(h, tau)
        g_sum = g.sum()
        g_zero = g - g_sum / n
        h_new = h - lr * g_zero
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = repair_feasible(h_new)
        c_new = exact_score(h_new)
        if c_new < best_c - 1e-14:
            best_h = h_new
            best_c = c_new
            h = h_new
            lr *= 1.1
        else:
            lr *= 0.5
        if lr < 1e-8:
            break
    return best_h, best_c

t_start_global = 0.0
budget_global = 1100.0

def run(seed=42, budget_s=1100, **kwargs):
    global t_start_global, budget_global
    t_start_global = time.perf_counter()
    budget_global = budget_s
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s - (time.perf_counter() - t_start_global)

    parent = None
    try:
        parent = np.array(initial_h_values, dtype=np.float64)
    except (NameError, Exception):
        pass

    if parent is None or len(parent) < 4 or not np.isfinite(parent).all():
        parent = np.full(70, 0.5)

    parent = np.clip(parent, 0.0, 1.0)
    parent = repair_feasible(parent)
    best_h = parent.copy()
    best_c = exact_score(parent)

    h_sym = repair_feasible(symmetrize(best_h))
    c_sym = exact_score(h_sym)
    if c_sym < best_c - 1e-12:
        best_h = h_sym
        best_c = c_sym

    # Phase 1: Annealed LSE SLSQP
    taus = [0.15, 0.08, 0.04, 0.02, 0.01, 0.005, 0.003, 0.002, 0.001]
    for tau in taus:
        if time_left() < 25:
            break
        h_res, c_res = solve_slsqp(best_h, tau, maxiter=400, ftol=1e-14)
        if h_res is not None and c_res < best_c - 1e-12:
            best_h = h_res
            best_c = c_res

    # Symmetrize and retry
    h_sym = repair_feasible(symmetrize(best_h))
    c_sym = exact_score(h_sym)
    if c_sym < best_c - 1e-12:
        best_h = h_sym
        best_c = c_sym

    # Phase 2: Second anneal
    if time_left() > 50:
        for tau in [0.002, 0.001, 0.0005, 0.0002]:
            if time_left() < 15:
                break
            h_res, c_res = solve_slsqp(best_h, tau, maxiter=500, ftol=1e-15)
            if h_res is not None and c_res < best_c - 1e-12:
                best_h = h_res
                best_c = c_res

    # Phase 3: Projected gradient with small tau
    if time_left() > 40:
        for tau in [0.005, 0.002, 0.001]:
            if time_left() < 15:
                break
            h_pg, c_pg = projected_gradient_step(best_h, tau, lr=0.05, n_steps=80)
            if c_pg < best_c - 1e-14:
                best_h = h_pg
                best_c = c_pg

    # Phase 4: Try different dimensions
    n0 = len(best_h)
    if time_left() > 80:
        dims = [n0 - 2, n0 + 2, n0 - 4, n0 + 4, n0 - 6, n0 + 6, n0 + 8]
        dims = [d for d in dims if 20 <= d <= 200 and d % 2 == 0 and d != n0]
        for new_n in dims:
            if time_left() < 40:
                break
            h_interp = interpolate_to_n(best_h, new_n)
            c_int = exact_score(h_interp)
            for tau in [0.05, 0.02, 0.01, 0.005, 0.002]:
                if time_left() < 20:
                    break
                h_res, c_res = solve_slsqp(h_interp, tau, maxiter=300)
                if h_res is not None:
                    h_interp = h_res
                    c_int = c_res
            if c_int < best_c - 1e-12:
                best_h = h_interp
                best_c = c_int

    # Phase 5: Random restarts
    if time_left() > 60:
        for trial in range(8):
            if time_left() < 20:
                break
            scale = [0.03, 0.05, 0.08, 0.12][trial % 4]
            h_pert = best_h + rng.normal(0, scale, len(best_h))
            h_pert = repair_feasible(h_pert)
            h_pert = repair_feasible(symmetrize(h_pert))
            for tau in [0.02, 0.01, 0.005, 0.002, 0.001]:
                if time_left() < 12:
                    break
                h_res, c_res = solve_slsqp(h_pert, tau, maxiter=250)
                if h_res is not None:
                    h_pert = h_res
            c_fin = exact_score(h_pert)
            h_fin = repair_feasible(symmetrize(h_pert))
            c_fin2 = exact_score(h_fin)
            if c_fin2 < c_fin:
                h_pert, c_fin = h_fin, c_fin2
            if c_fin < best_c - 1e-12:
                best_h = h_pert
                best_c = c_fin

    # Phase 6: Coordinate polish
    if time_left() > 20:
        best_h, best_c = coordinate_polish(best_h, max_rounds=30)

    # Phase 7: Final LSE polish
    if time_left() > 15:
        for tau in [0.001, 0.0005]:
            if time_left() < 8:
                break
            h_res, c_res = solve_slsqp(best_h, tau, maxiter=400, ftol=1e-15)
            if h_res is not None and c_res < best_c - 1e-14:
                best_h = h_res
                best_c = c_res

    # Phase 8: One last coordinate polish
    if time_left() > 10:
        best_h, best_c = coordinate_polish(best_h, max_rounds=20)

    # Final validation
    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)

    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            final_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 2:
            final_c = float(eval_result[0])
    except Exception:
        pass

    return best_h, final_c, len(best_h)
