    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = b - gr * (b - a)
    d = a + gr * (b - a)

import numpy as np
from scipy.optimize import minimize

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        cap = 1.0 - h
        order = np.argsort(-cap)
        rem = diff
        for i in range(n):
            idx = order[i]
            add = min(rem, cap[idx] + 1e-15)
            h[idx] += add
            rem -= add
            if rem <= 1e-12:
                break
    else:
        order = np.argsort(h)
        rem = -diff
        for i in range(n):
            idx = order[i]
            remv = min(rem, h[idx] + 1e-15)
            h[idx] -= remv
            rem -= remv
            if rem <= 1e-12:
                break
    return np.clip(h, 0.0, 1.0)

def anti_project(h):
    n = len(h)
    m = n // 2
    right = h[n-1:n-m-1:-1]
    y = (h[:m] + 1.0 - right) / 2.0
    return np.clip(y, 0.0, 1.0)

def anti_reconstruct(y, n):
    m = n // 2
    h = np.empty(n)
    h[:m] = y
    h[m:] = 1.0 - y[::-1]
    return h

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = np.linspace(0, 2.0, n + 1)
    h_cum = np.concatenate([[0.0], np.cumsum(h * (2.0/n))])
    xq = np.linspace(0, 2.0, new_n + 1)
    hnew = np.zeros(new_n)
    for k in range(new_n):
        a, b = xq[k], xq[k+1]
        # Find overlap with each old bin
        total = 0.0
        width = b - a
        # Vectorized: find old bins that overlap [a, b]
        old_starts = xs[:-1]
        old_ends = xs[1:]
        ov_start = np.maximum(old_starts, a)
        ov_end = np.minimum(old_ends, b)
        ov = np.maximum(0.0, ov_end - ov_start)
        total = np.sum(h * ov)
        hnew[k] = total / width if width > 0 else 0.5
    return repair_feasible(np.clip(hnew, 0.0, 1.0))

def build_gradient_matrix(h, n, dx):
    M = 2 * n - 1
    G = np.zeros((M, n))
    G[n-1, :] = dx * (1.0 - 2.0 * h)
    for d in range(1, n):
        q = n - 1 + d
        G[q, d:] += dx * (1.0 - h[:n-d])
        G[q, :n-d] += -dx * h[d:]
        q2 = n - 1 - d
        G[q2, :n-d] += dx * (1.0 - h[d:])
        G[q2, d:] += -dx * h[:n-d]
    return G

def golden_section_min(f, a, b, tol=1e-10, max_iter=60):
    """Minimize f on [a,b] using golden section. Assumes f is unimodal."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if abs(b - a) < tol:
            break
        if fc <= fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = f(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = f(d)
    mid = (a + b) / 2.0
    fm = f(mid)
    # Check endpoints
    fa, fb = f(a), f(b)
    if fa < fm:
        return a, fa
    if fb < fm:
        return b, fb
    return mid, fm

def single_line_search(h, j, n):
    """Exact line search for moving coordinate j (and its mirror n-1-j)."""
    j2 = n - 1 - j
    t_min = -h[j]
    t_max = 1.0 - h[j]
    if t_max - t_min < 1e-13:
        return h, exact_score(h)
    
    def f(t):
        h[j] += t
        h[j2] -= t
        s = exact_score(h)
        h[j] -= t
        h[j2] += t
        return s
    
    best_t, best_s = golden_section_min(f, t_min, t_max)
    if best_s < exact_score(h) - 1e-14:
        h[j] += best_t
        h[j2] -= best_t
        return h, best_s
    return h, exact_score(h)

def pair_line_search(h, i, j, n):
    """Line search for pair move: h[i]+=t, h[n-1-i]-=t, h[j]-=t, h[n-1-j]+=t."""
    i2 = n - 1 - i
    j2 = n - 1 - j
    if i == j or i == j2 or i2 == j or i2 == j2:
        return h, exact_score(h)
    
    t_min = max(-h[i], h[j] - 1.0)
    t_max = min(1.0 - h[i], h[j])
    if t_max - t_min < 1e-13:
        return h, exact_score(h)
    
    def f(t):
        h[i] += t; h[i2] -= t
        h[j] -= t; h[j2] += t
        s = exact_score(h)
        h[i] -= t; h[i2] += t
        h[j] += t; h[j2] -= t
        return s
    
    # Since might not be convex, use grid + refinement
    n_grid = 15
    grid = np.linspace(t_min, t_max, n_grid)
    best_val = np.inf
    best_idx = 0
    for idx, t in enumerate(grid):
        v = f(t)
        if v < best_val:
            best_val = v
            best_idx = idx
    # Refine
    lo = grid[max(0, best_idx - 1)]
    hi = grid[min(n_grid - 1, best_idx + 1)]
    best_t, best_s = golden_section_min(f, lo, hi, tol=1e-10, max_iter=40)
    
    if best_s < exact_score(h) - 1e-14:
        h[i] += best_t; h[i2] -= best_t
        h[j] -= best_t; h[j2] += best_t
        return h, best_s
    return h, exact_score(h)

def lse_anti(h, n, beta=80.0, max_iters=100):
    """LSE descent in anti-symmetric reduced coordinates."""
    m = n // 2
    dx = 2.0 / n
    best_c = exact_score(h)
    alpha = 0.01
    stuck = 0
    
    for it in range(max_iters):

        c = float(np.max(profile))

        wsum = w.sum()

            w = np.zeros_like(w)
            w[np.argmax(profile)] = 1.0
        else:
            w /= wsum

        g_full = G.T @ w  # shape (n,)
        
        # Reduced gradient: g_y[j] = g_full[j] - g_full[n-1-j]
        # g_full[n-1-j] for j=0..m-1: indices n-1, n-2, ..., n-m = m
        # So g_full[n-1-j] = g_full[m + (m-1-j)] = g_full[m:][m-1-j] = g_full[m:][::-1][j]
        g_red = g_full[:m] - g_full[m:][::-1]
        g_red -= g_red.mean()

        if gn < 1e-14:
            break
        g_red /= gn

        for a in [alpha, alpha*0.5, alpha*0.25, alpha*0.1]:

                    best_c = s

                    alpha = min(alpha * 1.15, 0.05)

                    break
                elif s < c - 1e-14:
                    h = h_try
                    improved = True
                    break
        if not improved:
            alpha *= 0.5
            stuck += 1
            if stuck > 40 or alpha < 1e-10:
                break
    
    return h, best_c
