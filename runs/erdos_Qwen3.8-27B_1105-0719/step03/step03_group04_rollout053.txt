    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:

    else:

def repair_feasible(h):

    target = n / 2.0
    diff = target - h.sum()

        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, 1.0 - h[idx])
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)[::-1]
        remaining = -diff
        for i in range(n):

        h = np.clip(h, 0.0, 1.0)

def pair_direction_score(G, P, p, q):

    delta = G[:, q] - G[:, p]
    # Predicted new profile: P + t * delta

    # Linear approximation: find t that minimizes max(P + t*delta)
    # This is a 1D convex optimization (max of linear functions)

    # and some increase (where delta_k > 0)

    return delta, max(0, improvement)

    return best_h, best_c

def pair_optimize(h, t_start, budget_s, max_iters=500000):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c = exact_score(h)
    stagnation = 0
    step_scale = 1.0  # Global step scale (starts at 1, shrinks over time)
    
    for it in range(max_iters):
        if time.perf_counter() - t_start > budget_s - 5:
            break
        
        P = compute_profile(h)
        c = float(P.max())
        
        if c < best_c - 1e-15:
            best_c = c
            best_h = h.copy()
            stagnation = 0
        else:
            stagnation += 1
        
        if stagnation > 50000:
            break
        
        G = build_gradient(h)
        M = 2 * n - 1
        
        # Adapt step scale based on stagnation
        if stagnation > 1000:
            step_scale = max(0.1, step_scale * 0.99)
        
        # Generate pairs
        pairs = generate_pairs(h, G, P, n, it, step_scale)
        
        # Test pairs
        improved = False
        for p, q, t_pred in pairs:
            mu = min(h[p], 1.0 - h[q])
            if mu <= 1e-12:
                continue
            # Test several t values around t_pred
            t_candidates = get_t_candidates(t_pred, mu)
            for t in t_candidates:
                h[p] -= t
                h[q] += t
                c_new = exact_score(h)
                if c_new < best_c - 1e-14:
                    best_c = c_new
                    best_h = h.copy()
                    improved = True
                    break
                h[p] += t
                h[q] -= t
            if improved:
                break
        
        if not improved and stagnation > 500:
            # Escape: small random perturbation
            rng = np.random.RandomState(it * 7 + 13)
            pert = rng.randn(n) * 0.002 * step_scale
            pert -= pert.mean()
            h_try = repair_feasible(h + pert)
            c_try = exact_score(h_try)
            if c_try < best_c:
                h = h_try
            elif c_try < c + 0.001:
                h = h_try  # Accept even if not better than best (for exploration)
    
    return best_h, best_c

The KKT conditions at the optimum:
- min t
- s.t. sum(h) = n/2, t >= P_k(h) for all k, 0 <= h <= 1

At the optimum with active set A (constraints at equality: t = P_k for k in A):
- Gradient of t = 0 (trivially satisfied since t is the variable we're minimizing)
- For active constraints: ∂P_k/∂h = λ_k * ∂(sum h)/∂h + μ (where μ is for box)
  Actually: ∇_h L = Σ_{k∈A} λ_k ∇P_k - ν * 1 + box_mult = 0
  where L = t - Σ λ_k (t - P_k) - ν(sum(h) - n/2)
  
  Simplifying: Σ_{k∈A} λ_k ∇P_k + ν * 1 = 0 (ignoring box for now)
  
  And: λ_k >= 0 for all k in A

So I need to find:
- A set A of constraints
- Weights λ_k >= 0, sum = 1 (normalization)
- Such that: G[A,:].T @ λ = ν * 1 (the gradient of active constraints, weighted sum equals a constant vector)

This means: G[A,:].T @ λ should be a constant vector (all elements equal).

import numpy as np
from scipy.optimize import minimize
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def repair_feasible(h):
    n = len(h)
    h = np.clip(np.asarray(h, dtype=np.float64), 0.0, 1.0).copy()
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, 1.0 - h[idx])
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)[::-1]
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def build_gradient(h):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    G = np.zeros((M, n))
    for d in range(-(n - 1), n):
        k = n - 1 + d
        i_start = max(0, -d)
        i_end = min(n - 1, n - 1 - d)
        if i_start > i_end:
            continue
        i_arr = np.arange(i_start, i_end + 1)
        j1 = i_arr
        j2 = i_arr + d
        G[k, j1] += dx * (1.0 - h[j2])
        G[k, j2] += -dx * h[j1]
    return G

def get_t_candidates(t_pred, mu):
    fracs = [0.02, 0.05, 0.1, 0.2, 0.35, 0.5, 0.7, 0.9, 1.0]
    t_vals = []
    for f in fracs:
        t = mu * f * t_pred
        if 0 < t <= mu:
            t_vals.append(t)
    t_vals.append(t_pred * mu)
    unique_t = []
    for t in sorted(t_vals):
        if 0 < t <= mu and (not unique_t or abs(t - unique_t[-1]) > mu * 0.01):
            unique_t.append(t)
    return unique_t[:8]
