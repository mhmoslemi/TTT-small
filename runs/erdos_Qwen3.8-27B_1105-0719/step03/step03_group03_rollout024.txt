- Sort h
- The top n/2 get pushed toward 1, the bottom n/2 get pushed toward 0
- h_new[i] = h[i] + t * (1 if rank(i) < n/2 else -1)
- Then repair

This is like a discrete version of "prepare for binary" while maintaining the sum.

Key differences:
1. LSE smooth gradient (global pressure on all near-max lags)
2. Binary pushing (exploits likely binary structure)

import numpy as np
from scipy.optimize import minimize

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))


def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx


def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx] + 1e-15)
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx] + 1e-15)
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmin(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h


def build_grad(h):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    center = n - 1
    G = np.zeros((M, n))
    for m in range(n):
        for d in range(-(n-1), n):
            k = center + d
            j = m - d
            i = m + d
            val = 0.0
            if 0 <= j < n:
                val += (1.0 - h[j])
            if 0 <= i < n:
                val -= h[i]
            G[k, m] = dx * val
    return G


def lse_gradient(h, eta):
    """Compute smoothed gradient using log-sum-exp weighting."""
    profile = compute_profile(h)
    cmax = np.max(profile)
    z = (profile - cmax) / eta
    z -= z.max()  # numerical stability
    w = np.exp(z)
    w /= w.sum()
    G = build_grad(h)
    grad = G.T @ w
    # Remove mean to respect sum constraint
    grad -= grad.mean()
    return grad, profile, cmax


def lse_optimize(h0, budget_fn, etas=None, max_iters=500):
    """Optimize using LSE gradient with adaptive step size."""
    if etas is None:
        etas = [0.08, 0.05, 0.03, 0.02, 0.012, 0.007, 0.004, 0.002]
    
    h = h0.copy()
    best_c = exact_score(h)
    n = len(h)
    
    for eta in etas:
        if budget_fn() < 30:
            break
        alpha = 0.01
        for it in range(max_iters):
            if budget_fn() < 30:
                break
            grad, profile, cmax = lse_gradient(h, eta)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break
            
            # Line search
            found = False
            for _ in range(20):
                h_try = repair_feasible(h - alpha * grad)
                c_try = exact_score(h_try)
                if c_try < best_c - 1e-14:
                    h = h_try
                    best_c = c_try
                    alpha = min(alpha * 1.3, 0.1)
                    found = True
                    break
                alpha *= 0.5
            if not found:
                # Try small random step in gradient direction
                alpha_small = max(alpha, 0.0001)
                h_try = repair_feasible(h - alpha_small * grad)
                c_try = exact_score(h_try)
                if c_try < best_c - 1e-14:
                    h = h_try
                    best_c = c_try
                else:
                    break
            if it > 0 and it % 100 == 0:
                # Check stagnation
                pass
        # After each eta stage, check if we should continue
        if budget_fn() < 50:
            break
    
    return h, best_c


def binary_push(h, strength=0.01):
    """Push values toward 0 or 1 while maintaining sum constraint."""
    n = len(h)
    target = n / 2.0
    h_new = h.copy()
    # Values above 0.5 get pushed up, below get pushed down
    above = h > 0.5
    below = ~above
    h_new[above] += strength
    h_new[below] -= strength
    return repair_feasible(h_new)


def pairwise_cd(h0, budget_fn, max_sweeps=50):
    """Pairwise cyclic coordinate descent maintaining sum constraint."""
    n = len(h0)
    h = h0.copy()
    best_c = exact_score(h)
    dx = 2.0 / n
    M = 2 * n - 1
    center = n - 1
    
    for sweep in range(max_sweeps):
        if budget_fn() < 30:
            break
        improved = False
        
        # Try all pairs in a fixed order
        for m in range(n):
            if budget_fn() < 30:
                break
            for mp in [m + 1, m + n // 2, m - 1]:
                mp = mp % n
                if mp == m:
                    continue
                
                # Compute coefficients for this pair
                # P[k] = sum of terms; changing h[m] by +δ and h[mp] by -δ
                # The change in P[k]:
                # For each pair (i,j) contributing to P[k]:
                #   if i==m: term changes by (1-h[j])*δ (h[m] increases)
                #   if j==m: term changes by -h[i]*δ
                #   if i==mp: term changes by -(1-h[j])*δ (h[mp] decreases)
                #   if j==mp: term changes by +h[i]*δ
                # Combined: ΔP[k] = δ * (contrib_m - contrib_mp)
                
                # Let a[k] = dP[k]/dh[m] and b[k] = dP[k]/dh[mp]
                # ΔP[k] = δ * (a[k] - b[k])
                # We want to minimize max_k (P[k] + δ*(a[k]-b[k]))
                # subject to 0 ≤ h[m]+δ ≤ 1 and 0 ≤ h[mp]-δ ≤ 1
                
                a = np.zeros(M)
                b = np.zeros(M)
                for d in range(-(n-1), n):
                    k = center + d
                    j_m = m - d
                    i_m = m + d
                    j_mp = mp - d
                    i_mp = mp + d
                    ak = 0.0
                    bk = 0.0
                    if 0 <= j_m < n:
                        ak += (1.0 - h[j_m])
                    if 0 <= i_m < n:
                        ak -= h[i_m]
                    if 0 <= j_mp < n:
                        bk += (1.0 - h[j_mp])
                    if 0 <= i_mp < n:
                        bk -= h[i_mp]
                    a[k] = dx * ak
                    b[k] = dx * bk
                
                # ΔP[k] = δ * (a[k] - b[k])
                diff_ab = a - b
                # Feasible range of δ:
                # h[m] + δ ∈ [0,1] → δ ∈ [-h[m], 1-h[m]]
                # h[mp] - δ ∈ [0,1] → δ ∈ [h[mp]-1, h[mp]]
                delta_lo = max(-h[m], h[mp] - 1.0)
                delta_hi = min(1.0 - h[m], h[mp])
                
                if delta_lo >= delta_hi - 1e-15:
                    continue
                
                # Minimize max_k (P[k] + δ*diff_ab[k]) over δ ∈ [delta_lo, delta_hi]
                # This is a 1D convex piecewise linear function
                # The minimum is at a breakpoint or at an endpoint
                
                profile = compute_profile(h)
                base = profile  # P[k]
                # f(δ) = max_k (base[k] + δ * diff_ab[k])
                
                # Try endpoints and a few breakpoints
                candidates = [delta_lo, delta_hi, 0.0]
                # Add breakpoints: where base[i] + δ*diff_ab[i] = base[j] + δ*diff_ab[j]
                # δ = (base[j] - base[i]) / (diff_ab[i] - diff_ab[j])
                # Only consider a few to keep it fast
                top_idx = np.argsort(base)[-10:]
                for ii in range(len(top_idx)):
                    for jj in range(ii+1, len(top_idx)):
                        i_ = top_idx[ii]
                        j_ = top_idx[jj]
                        denom = diff_ab[i_] - diff_ab[j_]
                        if abs(denom) > 1e-15:
                            db = (base[j_] - base[i_]) / denom
                            if delta_lo < db < delta_hi:
                                candidates.append(db)
                
                best_delta = 0.0
                best_val = np.max(base)
                for dc in candidates:
                    dc = np.clip(dc, delta_lo, delta_hi)
                    val = np.max(base + dc * diff_ab)
                    if val < best_val - 1e-15:
                        best_val = val
                        best_delta = dc
                
                if best_delta != 0.0 and best_val < best_c - 1e-14:
                    h[m] += best_delta
                    h[mp] -= best_delta
                    best_c = min(best_c, exact_score(h))
                    improved = True
        
        if not improved:
            break
    
    return h, best_c


def interp(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    fx = np.concatenate([[0.0], xs, [2.0]])
    fh = np.concatenate([[h[0]], h, [h[-1]]])
    hn = np.interp(xt, fx, fh)
    return repair_feasible(np.clip(hn, 0, 1))


def slsqp_refine(h0, maxiter=300, tol=1e-12):
    n = len(h0)
    dx = 2.0 / n
    M = 2 * n - 1
    
    def obj(x):
        return x[-1]
    
    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    
    def eq_fun(x):
        return x[:-1].sum() - n / 2.0
    
    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g
    
    def ineq_fun(x):
        h = x[:-1]
        return x[-1] - np.correlate(h, 1 - h, mode="full") * dx
    
    def ineq_jac(x):
        h = x[:-1]
        G = build_grad(h)
        J = np.zeros((M, n + 1))
        J[:, :-1] = -G
        J[:, -1] = 1.0
        return J
    
    t0 = exact_score(h0) + 1e-4
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, t0 + 2.0)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': tol})
        if res.x is not None and np.isfinite(res.x).all():
            hs = repair_feasible(res.x[:-1])
            if np.isfinite(exact_score(hs)):
                return hs
    except Exception:
        pass
    return None


def trust_region_step(h, active, dx, trust, rho):
    """Solve a trust region subproblem."""
    if not HAS_CVXPY:
        return None
    n = len(h)
    target = n / 2.0
    G = build_grad(h)
    G_a = G[active, :]
    profile = compute_profile(h)
    b_a = profile[active] - G_a @ h
    
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    cons = [
        h_var >= 0,
        h_var <= 1,
        cp.sum(h_var) == target,
        h_var >= h - trust,
        h_var <= h + trust,
        G_a @ h_var - t_var <= -b_a
    ]
    obj = cp.Minimize(t_var + rho * cp.sum((h_var - h) ** 2))
    prob = cp.Problem(obj, cons)
    for solver in [cp.ECOS, cp.SCS, cp.CBC]:
        try:
            prob.solve(solver=solver, max_iters=10000)
            break
        except Exception:
            continue
    if h_var.value is None:
        return None
    h_sol = np.array(h_var.value).flatten()
    if not np.isfinite(h_sol).all():
        return None
    return h_sol

ΔP[center+d] = dx * [-(1-h[m-d])*I(0≤m-d<n) + h[m+d]*I(0≤m+d<n) + (1-h[m'-d])*I(0≤m'-d<n) - h[m'+d]*I(0≤m'+d<n)]

This can be computed in O(n) per d, so O(n) total for all lags (since each lag has at most 4 non-zero terms from the swap).

- m+d is valid: d ∈ [-m, n-1-m]

def binary_sa(h_binary, budget_fn, max_sweeps=200, T_start=0.05, T_end=0.005):
    """Simulated annealing on binary vectors."""
    n = len(h_binary)
    h = h_binary.copy()
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h <= 0.5)[0]
    profile = compute_profile(h)
    curr_c = np.max(profile)
    best_c = curr_c
    best_h = h.copy()
    
    for sweep in range(max_sweeps):
        if budget_fn() < 30:
            break
        T = T_start * (T_end / T_start) ** (sweep / max_sweeps)
        
        # Shuffle order of ones
        rng_order = np.random.permutation(len(ones))
        
        for idx in rng_order:
            if budget_fn() < 30:
                break
            m = ones[idx]
            # Try swapping with a random zero
            mp_idx = np.random.randint(len(zeros))
            mp = zeros[mp_idx]
            
            # Compute change in profile
            dx = 2.0 / n
            center = n - 1
            delta_profile = np.zeros(len(profile))
            
            for d in range(-(n-1), n):
                k = center + d
                val = 0.0
                j1 = m - d
                if 0 <= j1 < n:
                    val -= (1.0 - h[j1])
                i1 = m + d
                if 0 <= i1 < n:
                    val += h[i1]
                j2 = mp - d
                if 0 <= j2 < n:
                    val += (1.0 - h[j2])
                i2 = mp + d
                if 0 <= i2 < n:
                    val -= h[i2]
                delta_profile[k] = dx * val
            
            new_profile = profile + delta_profile
            new_c = np.max(new_profile)
            
            if new_c < curr_c or (np.random.random() < np.exp(-(new_c - curr_c) / T)):
                # Accept
                h[m] = 0
                h[mp] = 1
                ones = np.delete(ones, idx)
                zeros = np.delete(zeros, mp_idx)
                zeros = np.append(zeros, m)
                profile = new_profile
                curr_c = new_c
                if curr_c < best_c:
                    best_c = curr_c
                    best_h = h.copy()
    
    return best_h, best_c
```

def compute_delta(n, h, m, mp, dx):
    center = n - 1
    d = np.arange(-(n-1), n)
    k = center + d  # all lag indices
    
    val = np.zeros(2*n-1)
    
    # Term 1: -(1-h[m-d]) for valid m-d
    j1 = m - d
    mask1 = (j1 >= 0) & (j1 < n)
    val[mask1] -= (1.0 - h[j1[mask1]])
    
    # Term 2: +h[m+d] for valid m+d
    i1 = m + d
    mask2 = (i1 >= 0) & (i1 < n)
    val[mask2] += h[i1[mask2]]
    
    # Term 3: +(1-h[mp-d]) for valid mp-d
    j2 = mp - d
    mask3 = (j2 >= 0) & (j2 < n)
    val[mask3] += (1.0 - h[j2[mask3]])
    
    # Term 4: -h[mp+d] for valid mp+d
    i2 = mp + d
    mask4 = (i2 >= 0) & (i2 < n)
    val[mask4] -= h[i2[mask4]]
    
    return val * dx
```
