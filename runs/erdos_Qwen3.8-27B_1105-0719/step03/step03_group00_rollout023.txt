def project_box_sum(x, target):
    n = len(x)
    if n == 0:
        return x.copy()
    x = np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)

                h[mask] = np.minimum(h[mask] + add, 1.0)

def c5_of_h(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * dx

def weighted_grad(h, w, dx):
    """Compute gradient of weighted sum of correlations.
    w has length 2*n-1, aligned with corr indices.
    Returns gradient of length n."""
    n = len(h)
    grad = np.zeros(n)
    
    # d=0 term
    w0 = w[n-1]
    grad += dx * w0 * (1.0 - 2.0 * h)
    
    # d>0 terms
    for d in range(1, n):
        u = dx * w[n-1+d]  # positive lag weight
        v = dx * w[n-1-d]  # negative lag weight
        s = u + v
        if s < 1e-15:
            continue
        # Positive lag contribution
        grad[:n-d] += u - s * h[d:]
        # Negative lag contribution
        grad[d:] += v - s * h[:n-d]
    
    return grad

def softmax_weights(corr, alpha):
    """Compute softmax weights from correlations."""
    x = corr / alpha
    x = x - x.max()
    p = np.exp(x)
    s = p.sum()
    if s < 1e-300:
        return np.ones(len(corr)) / len(corr)
    return p / s

def grad_step(h, w, dx, target, step_size, rng=None):
    """One gradient descent step on weighted objective."""
    n = len(h)
    g = weighted_grad(h, w, dx)
    # Remove mean to respect sum constraint (project gradient onto sum-zero subspace)
    g -= g.mean()
    h_new = h - step_size * g
    h_new = project_box_sum(h_new, target)
    return h_new

def softmax_continuation(h, dx, target, alphas, steps_per_alpha, t0, budget_s, rng):

def pairwise_move(h, i, j, target, dx):
    """Optimal move shifting mass between h[i] and h[j].
    Returns (new_h, new_c5) or (None, None) if no improvement."""
    n = len(h)
    old_c5 = c5_of_h(h, dx)
    
    # We can shift mass from i to j or vice versa
    # h[i] + h[j] = constant (to maintain sum)
    total = h[i] + h[j]
    
    # Try several values of h[i] (and h[j] = total - h[i])
    # h[i] must be in [max(0, total-1), min(1, total)]
    lo = max(0.0, total - 1.0)
    hi = min(1.0, total)

def random_pairwise_refine(h, dx, target, n_moves, t0, budget_s, rng):

def cosine_modes(n, L):
    """Generate L cosine modes of length n, centered and normalized."""
    j = np.arange(n) + 0.5
    modes = np.zeros((L, n))
    for r in range(1, L + 1):
        modes[r-1] = np.cos(np.pi * r * j / n)
        modes[r-1] -= modes[r-1].mean()
        norm = np.linalg.norm(modes[r-1])
        if norm > 1e-12:
            modes[r-1] /= norm
    return modes

import numpy as np
from scipy.optimize import linprog
import time

def project_box_sum(x, target):
    n = len(x)
    x = np.array(x, dtype=float)
    x = np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0)
    lo = -3.0
    hi = 3.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    err = target - h.sum()
    if abs(err) > 1e-10:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += err / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def c5_of_h(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * dx

def grad_corr_k(h, k, n):
    off = k - (n - 1)
    grad = np.zeros(n)
    if off == 0:
        return 1.0 - 2.0 * h
    if off > 0:
        d = off
        grad[:n-d] += (1.0 - h[d:])
        grad[d:] += (-h[:n-d])
    else:
        d = -off
        grad[d:] += (1.0 - h[:n-d])
        grad[:n-d] += (-h[d:])
    return grad

def build_lp_direction(h, active_idx, corrs, dx, alpha, n):
    K = len(active_idx)
    if K == 0:
        return None
    m_vars = n + 1
    c_obj = np.zeros(m_vars)
    c_obj[-1] = 1.0
    A_ub = np.zeros((K, m_vars))
    b_ub = np.zeros(K)
    for j, k in enumerate(active_idx):
        g = grad_corr_k(h, k, n)
        A_ub[j, :n] = dx * g
        A_ub[j, -1] = -1.0
        b_ub[j] = -dx * corrs[k]
    A_eq = np.zeros((1, m_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(-h[i] * alpha, (1.0 - h[i]) * alpha) for i in range(n)]
    bounds.append((0.0, None))
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.status == 0:
            return res.x[:n]
    except Exception:
        pass
    return None

def lp_inner_loop(h, dx, target, alpha, n_inner, budget_s, t0, min_time=3.0):
    n = len(h)
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    cur = h.copy()
    cur_c5 = best_c5

    for _ in range(n_inner):
        if budget_s - (time.time() - t0) < min_time:
            break
        corrs = np.correlate(cur, 1.0 - cur, mode='full')
        max_c = corrs.max()
        margin = max(0.15, 0.025 * max_c)
        n_top = max(15, min(45, int(0.2 * (2 * n - 1))))
        top_idx = np.argsort(corrs)[-n_top:]
        active_mask = corrs >= (max_c - margin)
        active_mask[top_idx] = True
        active_idx = np.where(active_mask)[0]
        if len(active_idx) < 5:
            active_idx = top_idx

        delta = build_lp_direction(cur, active_idx, corrs, dx, alpha, n)
        if delta is None:
            alpha *= 0.5
            if alpha < 0.005:
                break
            continue

        improved = False
        for t in [1.0, 0.8, 0.6, 0.5, 0.4, 0.3, 0.2, 0.15, 0.1, 0.05, 0.02]:
            h_c = cur + t * delta
            h_c = project_box_sum(h_c, target)
            new_c5 = c5_of_h(h_c, dx)
            if new_c5 < cur_c5 - 1e-14:
                cur = h_c
                cur_c5 = new_c5
                improved = True
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h_c.copy()
                break
        if not improved:
            alpha *= 0.6
            if alpha < 0.005:
                break
        else:
            alpha = min(0.35, alpha * 1.04)

    return best_h, best_c5

def softmax_weights(corr, alpha):
    x = corr / alpha
    x = x - x.max()
    p = np.exp(x)
    s = p.sum()
    if s < 1e-300:
        return np.ones(len(corr)) / len(corr)
    return p / s

def weighted_grad(h, w, dx):
    n = len(h)
    grad = np.zeros(n)
    w0 = w[n-1]
    grad += dx * w0 * (1.0 - 2.0 * h)
    for d in range(1, n):
        u = dx * w[n-1+d]
        v = dx * w[n-1-d]
        s = u + v
        if s < 1e-15:
            continue
        grad[:n-d] += u - s * h[d:]
        grad[d:] += v - s * h[:n-d]
    return grad

def softmax_descent(h, dx, target, alphas, steps_per_alpha, t0, budget_s, rng):
    n = len(h)
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    cur = h.copy()
    cur_c5 = best_c5

    for alpha in alphas:
        for _ in range(steps_per_alpha):
            if budget_s - (time.time() - t0) < 5.0:
                break
            corr = np.correlate(cur, 1.0 - cur, mode='full')
            w = softmax_weights(corr, alpha)
            g = weighted_grad(cur, w, dx)
            g -= g.mean()
            gnorm = np.linalg.norm(g)
            if gnorm < 1e-12:
                break
            step = min(0.08, 0.4 / gnorm)
            h_new = project_box_sum(cur - step * g, target)
            new_c5 = c5_of_h(h_new, dx)
            if new_c5 < cur_c5 - 1e-14:
                cur = h_new
                cur_c5 = new_c5
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = cur.copy()
            else:
                h_new2 = project_box_sum(cur - step * 0.3 * g, target)
                new_c52 = c5_of_h(h_new2, dx)
                if new_c52 < cur_c5 - 1e-14:
                    cur = h_new2
                    cur_c5 = new_c52
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = cur.copy()

    return best_h, best_c5

def binary_explore(h, dx, target, rng, t0, budget_s, n_rounds=8):
    n = len(h)
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    target_int = int(target)

    for r in range(n_rounds):
        if budget_s - (time.time() - t0) < 20.0:
            break
        theta = rng.uniform(0.25, 0.75)
        h_bin = (h > theta).astype(float)
        n_ones = int(h_bin.sum())
        if n_ones > target_int:
            ones = np.where(h_bin == 1)[0]
            order = np.argsort(h[ones])
            for idx in order[:n_ones - target_int]:
                h_bin[ones[idx]] = 0
        elif n_ones < target_int:
            zeros = np.where(h_bin == 0)[0]
            order = np.argsort(-h[zeros])
            for idx in order[:target_int - n_ones]:
                h_bin[zeros[idx]] = 1
        h_bin = project_box_sum(h_bin, target)

        for t_mix in [1.0, 0.7, 0.5, 0.3]:
            h_try = (1.0 - t_mix) * best_h + t_mix * h_bin
            h_try = project_box_sum(h_try, target)
            c5_try = c5_of_h(h_try, dx)
            if c5_try < best_c5:
                h_opt, c5_opt = lp_inner_loop(h_try, dx, target, 0.25, 40, budget_s, t0)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt
                break

    return best_h, best_c5

def pairwise_refine(h, dx, target, n_moves, t0, budget_s, rng):
    n = len(h)
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)

    for _ in range(n_moves):
        if budget_s - (time.time() - t0) < 5.0:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        total = best_h[i] + best_h[j]
        lo = max(0.0, total - 1.0)
        hi = min(1.0, total)
        if hi - lo < 1e-6:
            continue
        best_t = None
        best_c5_pair = best_c5
        for t_val in np.linspace(lo + 1e-6, hi - 1e-6, 7):
            h_try = best_h.copy()
            h_try[i] = t_val
            h_try[j] = np.clip(total - t_val, 0.0, 1.0)
            c5_try = c5_of_h(h_try, dx)
            if c5_try < best_c5_pair:
                best_c5_pair = c5_try
                best_t = t_val
        if best_t is not None and best_c5_pair < best_c5:
            best_h[i] = best_t
            best_h[j] = np.clip(total - best_t, 0.0, 1.0)
            best_c5 = best_c5_pair

    return best_h, best_c5

def cosine_subspace_refine(h, dx, target, L, n_iters, t0, budget_s, rng):
    n = len(h)
    j = np.arange(n) + 0.5
    B = np.zeros((L, n))
    for r in range(1, L + 1):
        B[r-1] = np.cos(np.pi * r * j / n)
        B[r-1] -= B[r-1].mean()
        norm = np.linalg.norm(B[r-1])
        if norm > 1e-12:
            B[r-1] /= norm

    h0 = h - 0.5
    coeffs = B @ h0
    best_coeffs = coeffs.copy()
    best_c5 = c5_of_h(h, dx)
    cur_coeffs = coeffs.copy()
    lr = 0.05

    for _ in range(n_iters):
        if budget_s - (time.time() - t0) < 5.0:
            break
        h_cur = project_box_sum(0.5 + B.T @ cur_coeffs, target)
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        w = softmax_weights(corr, 0.08)
        g_h = weighted_grad(h_cur, w, dx)
        g_c = B @ g_h
        cur_coeffs -= lr * g_c
        h_new = project_box_sum(0.5 + B.T @ cur_coeffs, target)
        new_c5 = c5_of_h(h_new, dx)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_coeffs = cur_coeffs.copy()
            lr = min(lr * 1.08, 0.3)
        else:
            cur_coeffs = best_coeffs.copy()
            lr *= 0.7
            if lr < 0.002:
                break

    best_h = project_box_sum(0.5 + B.T @ best_coeffs, target)
    best_c5 = c5_of_h(best_h, dx)
    return best_h, best_c5

def transfer_grid(h_src, n_target):
