def exact_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def exact_score(h):

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx])
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-13:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-13:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-14:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmin(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def build_gradient_matrix_fast(h, dx):

def solve_lp_direction(h, D=0.5, active_only=False, margin=0.01):
    n = len(h)
    dx = 2.0 / n
    P = exact_profile(h)
    c = float(np.max(P))
    G = build_gradient_matrix_fast(h, dx)
    M = 2 * n - 1
    
    if active_only:
        active = np.where(P >= c - margin)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(P))])
        rows = active
    else:
        rows = np.arange(M)
    
    n_rows = len(rows)
    # Variables: d_0, ..., d_{n-1}, s
    # Minimize s
    # Subject to: G[m,:].d - s <= -P[m] for m in rows
    #             sum(d) = 0
    #             -D <= d_i <= D
    #             s free (or s >= 0)
    
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0  # minimize s
    
    A_ub = np.zeros((n_rows, n + 1))
    b_ub = np.zeros(n_rows)
    for i, m in enumerate(rows):
        A_ub[i, :n] = G[m, :]
        A_ub[i, -1] = -1.0
        b_ub[i] = -P[m]
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    bounds_lp = [(-D, D)] * n + [(None, None)]
    
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds_lp, method='highs')
    
    if result.success:
        d = result.x[:n]
        s_val = result.x[-1]
        # Ensure sum is zero (numerical)
        d = d - d.mean()
        gnorm = np.linalg.norm(d)
        if gnorm < 1e-12:
            return None, None, c, P
        return d, s_val, c, P
    return None, None, c, P

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = np.linspace(0, 2, n + 1)
    xt = np.linspace(0, 2, new_n + 1)
    h_interp = np.zeros(new_n)
    for k in range(new_n):
        lo_k, hi_k = xt[k], xt[k+1]
        total_mass = 0.0
        total_width = 0.0
        for j in range(n):
            lo_j, hi_j = xs[j], xs[j+1]
            overlap = min(hi_k, hi_j) - max(lo_k, lo_j)
            if overlap > 0:
                total_mass += h[j] * overlap
                total_width += overlap
        h_interp[k] = total_mass / total_width if total_width > 1e-15 else 0.5
    return repair_feasible(np.clip(h_interp, 0.0, 1.0))

            h_lp, c_lp = lp_descent(h_interp, max_iters=50, D=0.4, use_active=False)
            if c_lp < best_c - 1e-13:
                best_h, best_c = h_lp, c_lp
                h_pm, c_pm = pair_move_descent(best_h, max_iters=40, rng=rng)
                if c_pm < best_c - 1e-13:
                    best_h, best_c = h_pm, c_pm
    
    # Phase 7: Final focused LP with small steps
    if time_left() > 30:
        for D in [0.1, 0.15, 0.2]:
            if time_left() < 10:
                break
            h_lp, c_lp = lp_descent(best_h, max_iters=80, D=D, use_active=True, active_margin=0.003)
            if c_lp < best_c - 1e-14:
                best_h, best_c = h_lp, c_lp
    
    # Phase 8: Final pair polish
    if time_left() > 15:
        h_pm, c_pm = pair_move_descent(best_h, max_iters=100, rng=rng)
        if c_pm < best_c - 1e-14:
            best_h, best_c = h_pm, c_pm
    
    # Final validation
    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)
    
    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            final_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 2:
            final_c = float(eval_result[0])
    except Exception:
        pass
    
    return best_h, final_c, len(best_h)
