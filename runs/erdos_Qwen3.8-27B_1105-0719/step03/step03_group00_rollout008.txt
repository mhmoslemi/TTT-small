```python
        wm = w[n-1-d]  # weight for corr[n-1-d] = sum h[i]*(1-h[i+d])
        
        if wd > 0:
            grad[d:] += wd * (1 - h[:K])
            grad[:K] -= wd * h[d:]
        
        if wm > 0:
            grad[:K] += wm * (1 - h[d:])
            grad[d:] -= wm * h[:K]
    
    return grad
```

This changes if i = p (contribution: h[p+d]*(1-h[p]) changes) or i+d = p (i = p-d, contribution: h[p]*(1-h[p-d]) changes) or i = q or i+d = q.

```python
import numpy as np
```

```python
def project_feasible(x, target_sum):
    n = len(x)
    x = np.where(np.isfinite(x), x, 0.5)
    lo, hi = x.min() - 2.0, x.max() + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def c5_of_h(h):

def lq_gradient(h, corr, q):
    n = len(h)
    maxc = max(corr.max(), 1e-30)
    a = np.maximum(corr / maxc, 0.0)
    if q > 1:
        w = a ** (q - 1)
    else:
        w = np.ones(2 * n - 1)
    grad = np.zeros(n)
    w0 = w[n - 1]
    if w0 > 0:
        grad += w0 * (1.0 - 2.0 * h)

        wd = w[n - 1 + d]

        if wd > 0:
            grad[d:] += wd * (1.0 - h[:K])
            grad[:K] -= wd * h[d:]
        if wm > 0:
            grad[:K] += wm * (1.0 - h[d:])
            grad[d:] -= wm * h[:K]
    return grad
```
