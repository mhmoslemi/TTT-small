def grad_profile(h, lags):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((len(lags), n))
    for idx, m in enumerate(lags):
        d = m - (n - 1)
        if d == 0:

        elif d > 0:

            G[idx, i] -= dx * h[i + d]

            dp = -d
            j = np.arange(n - dp)
            G[idx, j + dp] += dx * (1.0 - h[j])
            G[idx, j] -= dx * h[j + dp]

            h_opt = slsqp_iterative(h_interp, max_rounds=8, maxiter_per=150,
                                    t_start=t_start, budget_s=budget_s, margin=30)
            c_opt = exact_score(h_opt)
            if c_opt < best_c - 1e-10:
                best_h = h_opt
                best_c = c_opt
                # Polish at this dimension
                if time_ok(30):
                    h_pol = slsqp_iterative(best_h, max_rounds=10, maxiter_per=200,
                                            t_start=t_start, budget_s=budget_s, margin=20)
                    c_pol = exact_score(h_pol)
                    if c_pol < best_c - 1e-13:
                        best_h = h_pol
                        best_c = c_pol

    # Phase 6: Second round of subgradient from best
    if time_ok(100):
        h_sg2, c_sg2 = subgradient_avg(best_h, max_iters=2000, t_start=t_start,
                                       budget_s=budget_s, margin=60)
        if c_sg2 < best_c - 1e-13:
            best_h = h_sg2
            best_c = c_sg2

    # Phase 7: Final SLSQP polish
    if time_ok(60):
        h_final = slsqp_iterative(best_h, max_rounds=10, maxiter_per=250,
                                  t_start=t_start, budget_s=budget_s, margin=20)
        c_final = exact_score(h_final)
        if c_final < best_c - 1e-14:
            best_h = h_final
            best_c = c_final

    # Phase 8: Coordinate descent fine-tune
    if time_ok(30):
        best_h, best_c = coordinate_polish(best_h, max_sweeps=10,
                                           t_start=t_start, budget_s=budget_s, margin=10)

    # Final validation
    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)

    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            final_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 1:
            if np.isfinite(eval_result[0]):
                final_c = float(eval_result[0])
    except Exception:
        pass

    return best_h, final_c, len(best_h)
