def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx])
            h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx])
            h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    return repair_feasible(np.clip(h_new, 0.0, 1.0))

def slsqp_active(h0, t_start, budget_s, margin=30, maxiter=400, active_k=15, tol=1e-11):
    n = len(h0)
    dx = 2.0 / n
    M = 2 * n - 1
    P = np.correlate(h0, 1 - h0, mode="full") * dx
    c0 = P.max()

    thresh = c0 - max(0.002, 0.04 * (c0 - np.median(P)))
    active = np.where(P >= thresh)[0]
    if len(active) == 0:
        active = np.array([np.argmax(P)])
    if len(active) > active_k:
        top_idx = np.argsort(P[active])[-active_k:]
        active = active[top_idx]

    def grad_row_for_lag(h, m_idx):
        d = m_idx - (n - 1)
        row = np.zeros(n)
        if d == 0:
            row[:] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            for i in range(n - d):
                row[i + d] += dx * (1.0 - h[i])
                row[i] -= dx * h[i + d]
        else:
            dp = -d
            for j in range(n - dp):
                row[j + dp] += dx * (1.0 - h[j])
                row[j] -= dx * h[j + dp]
        return row

    def obj(x):
        return x[-1]

    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g

    def eq_fun(x):
        return x[:-1].sum() - n / 2.0

    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g

    def ineq_fun(x):
        h = x[:-1]
        t = x[-1]
        prof = np.correlate(h, 1 - h, mode="full") * dx
        return t - prof[active]

    def ineq_jac(x):
        h = x[:-1]
        J = np.zeros((len(active), n + 1))
        for idx, m in enumerate(active):
            J[idx, :-1] = -grad_row_for_lag(h, m)
            J[idx, -1] = 1.0
        return J

    x0 = np.concatenate([h0, [c0 + 1e-4]])
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 1.0)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': tol})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = repair_feasible(res.x[:-1])
            c_new = exact_score(h_new)
            if c_new <= c0 + 1e-10:
                return h_new
    except Exception:
        pass
    return None

def build_gradient_rows(h, rows, n, dx):
    G = np.zeros((len(rows), n))
    for idx, m in enumerate(rows):
        d = m - (n - 1)
        if d == 0:
            G[idx, :] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            for i in range(n - d):
                G[idx, i + d] += dx * (1.0 - h[i])
                G[idx, i] -= dx * h[i + d]
        else:
            dp = -d
            for j in range(n - dp):
                G[idx, j + dp] += dx * (1.0 - h[j])
                G[idx, j] -= dx * h[j + dp]
    return G

def projected_descent(h, t_start, budget_s, margin=30, max_iters=150, momentum=0.4):
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    vel = np.zeros(n)
    betas = [40, 80, 160, 320, 640]
    iters_per_beta = max(8, max_iters // len(betas))

    for beta in betas:
        alpha = 0.015
        for it in range(iters_per_beta):
            if time.perf_counter() - t_start > budget_s - margin:
                return h, best_c
            P = np.correlate(h, 1 - h, mode="full") * (2.0 / n)
            Pmax = P.max()
            exp_arg = beta * (P - Pmax)
            weights = np.exp(exp_arg)
            S = weights.sum()
            if S < 1e-300:
                weights = np.zeros(len(P))
                weights[np.argmax(P)] = 1.0
            else:
                weights /= S
            active = np.where(weights > 0.005 / len(P))[0]
            if len(active) == 0:
                active = np.argsort(weights)[-5:]
            G = build_gradient_rows(h, active, n, 2.0 / n)
            w_a = weights[active]
            w_a = w_a / w_a.sum()
            grad = G.T @ w_a
            sum_err = h.sum() - n / 2.0
            grad += 2.0 * 30.0 * sum_err / n
            grad -= grad.mean()
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break
            vel = momentum * vel + grad / gnorm
            vnorm = np.linalg.norm(vel)
            if vnorm > 0:
                vel /= vnorm
            improved = False
            for a in [alpha, alpha * 0.5, alpha * 0.25, alpha * 0.1]:

                if s < best_c - 1e-14:
                    h = h_try
                    best_c = s
                    improved = True
                    break
                elif s < exact_score(h):
                    h = h_try
            if improved:
                alpha = min(alpha * 1.05, 0.06)
            else:
                alpha *= 0.6
                if alpha < 1e-8:
                    break
    return h, best_c

            h_interp, c_interp = pairwise_sweep(h_interp, t_start, budget_s, margin=40,
                                                 max_sweeps=30, random_frac=0.5, seed=seed + new_n)
            sol = slsqp_active(h_interp, t_start, budget_s, margin=40, maxiter=200, active_k=12, tol=1e-11)
            if sol is not None:
                c = exact_score(sol)
                if c < best_c - 1e-10:
                    best_h = sol
                    best_c = c
                    if time_ok(30):
                        best_h, best_c = pairwise_sweep(best_h, t_start, budget_s, margin=40,
                                                         max_sweeps=30, random_frac=None, seed=seed + new_n + 100)
                        sol2 = slsqp_active(best_h, t_start, budget_s, margin=40, maxiter=200, active_k=14, tol=1e-12)
                        if sol2 is not None:
                            c2 = exact_score(sol2)
                            if c2 < best_c - 1e-13:
                                best_h = sol2
                                best_c = c2
            elif c_interp < best_c - 1e-10:
                best_h = h_interp
                best_c = c_interp

    # Phase 7: Random restart perturbations + pairwise
    if time_ok(50):
        n = len(best_h)
        rng = np.random.RandomState(seed + 500)
        for trial in range(4):
            if not time_ok(20):
                break
            eps = 0.01 * (1 + trial * 0.5)
            z = rng.randn(n)
            z -= z.mean()
            h_pert = repair_feasible(best_h + eps * z)
            h_pert, c_pert = pairwise_sweep(h_pert, t_start, budget_s, margin=40,
                                             max_sweeps=25, random_frac=0.4, seed=seed + 600 + trial)
            if c_pert < best_c - 1e-11:
                best_h = h_pert
                best_c = c_pert
                sol = slsqp_active(best_h, t_start, budget_s, margin=30, maxiter=200, active_k=12, tol=1e-11)
                if sol is not None:
                    c = exact_score(sol)
                    if c < best_c - 1e-13:
                        best_h = sol
                        best_c = c

    # Phase 8: Final pairwise polish (all pairs, many sweeps)
    if time_ok(30):
        best_h, best_c = pairwise_sweep(best_h, t_start, budget_s, margin=15,
                                         max_sweeps=100, random_frac=None, seed=seed + 999)

    # Phase 9: Final SLSQP
    if time_ok(20):
        sol = slsqp_active(best_h, t_start, budget_s, margin=10, maxiter=300, active_k=15, tol=1e-12)
        if sol is not None:
            c = exact_score(sol)
            if c < best_c - 1e-14:
                best_h = sol
                best_c = c

    # Final validation
    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)

    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            final_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 1:
            if np.isfinite(eval_result[0]):
                final_c = float(eval_result[0])
    except Exception:
        pass

    return best_h, final_c, len(best_h)
