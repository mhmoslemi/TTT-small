    i_lo = max(0, -d)
    i_hi = min(n-1, n-1-d)
    if i_lo > i_hi:

    g[idx + d] += (-h[idx])
    return g

import numpy as np
from scipy.optimize import linprog
import time

def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -5.0, 5.0
    for _ in range(60):
        mid = (lo + hi) / 2
        if np.clip(x - mid, 0.0, 1.0).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-10:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def c5_eval(h):
    return np.correlate(h, 1.0 - h, mode='full').max() * (2.0 / len(h))

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def grad_lag(h, d, n):
    g = np.zeros(n)
    i_lo = max(0, -d)
    i_hi = min(n - 1, n - 1 - d)
    if i_lo > i_hi:
        return g
    idx = np.arange(i_lo, i_hi + 1)
    g[idx] += (1.0 - h[idx + d])
    g[idx + d] += (-h[idx])
    return g

def smooth_obj_grad(h, tau, n, dx):
    corr = compute_corr(h)
    scaled = corr / tau
    m = scaled.max()
    if m > 50:
        scaled = scaled - m
    exp_s = np.exp(scaled)
    w = exp_s / exp_s.sum()
    obj = dx * tau * (m + np.log(exp_s.sum()))
    grad = np.zeros(n)
    for k in range(len(corr)):
        if w[k] < 1e-14:
            continue
        d = k - (n - 1)
        i_lo = max(0, -d)
        i_hi = min(n - 1, n - 1 - d)
        if i_lo > i_hi:
            continue
        idx = np.arange(i_lo, i_hi + 1)
        grad[idx] += w[k] * (1.0 - h[idx + d])
        grad[idx + d] += w[k] * (-h[idx])
    grad *= dx
    return obj, grad

def lp_direction(h, corr, active_idx, dx, alpha, n):
    K = len(active_idx)
    m_vars = n + 1
    grads = np.zeros((K, n))
    for j, k in enumerate(active_idx):
        d = k - (n - 1)
        grads[j] = grad_lag(h, d, n)
    c_obj = np.zeros(m_vars)
    c_obj[-1] = 1.0
    A_ub = np.zeros((K, m_vars))
    for j in range(K):
        A_ub[j, :n] = dx * grads[j]
        A_ub[j, -1] = -1.0
    b_ub = -dx * corr[active_idx]
    A_eq = np.zeros((1, m_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(-h[i] * alpha, (1.0 - h[i]) * alpha) for i in range(n)]
    bounds.append((0.0, None))
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status != 0:
            return None
        return res.x[:n]
    except Exception:
        return None

def line_search(cur, delta, target, cur_c5):

    best_h = cur
