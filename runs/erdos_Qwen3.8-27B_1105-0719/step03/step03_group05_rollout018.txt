def smooth_reflect(h, k=2):

def repair_feasible(vec):

def _build_gradient(h, n, dx):
    M = 2 * n - 1
    G = np.zeros((M, n))

            G[m, 0:n - d] += dx * (1.0 - h[d:n])
            G[m, d:n] += -dx * h[0:n - d]
        else:
            dp = -d
            G[m, dp:n] += dx * (1.0 - h[0:n - dp])
            G[m, 0:n - dp] += -dx * h[dp:n]
    return G

def smooth_reflect(h, k=2):

                best_c = c_pair
    
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()
    rng = np.random.RandomState(seed)
    
    def time_left():
        return budget_s + t_start - time.perf_counter()
    
    # Get parent
    parent = None
    try:
        parent = np.array(initial_h_values, dtype=np.float64)
    except NameError:
        pass
    except Exception:
        pass
    
    if parent is None or len(parent) < 4:
        parent = np.full(70, 0.5)
    
    parent = np.clip(parent, 0.0, 1.0)
    parent = repair_feasible(parent)
    c_parent = exact_score(parent)
    
    best_h = parent.copy()
    best_c = c_parent
    
    # Quick initial polish
    if time_left() > 150:
        h_lse, c_lse = lse_descent(best_h, beta=60, max_iters=40, alpha_init=0.003)
        if c_lse < best_c - 1e-13:
            best_h = h_lse.copy()
            best_c = c_lse
        h_ase, c_ase = active_set_descent(best_h, max_iters=40, n_active=10)
        if c_ase < best_c - 1e-13:
            best_h = h_ase.copy()
            best_c = c_ase
    
    n_parent = len(best_h)
    
    # Resolution exploration
    if time_left() > 400:
        dims = set()
        for factor in [0.85, 0.9, 0.95, 1.05, 1.1, 1.15, 1.2, 1.3]:
            d = int(round(n_parent * factor / 2)) * 2
            d = max(24, min(d, 160))
            dims.add(d)
        dims.discard(n_parent)
        dims_list = sorted(dims, key=lambda x: abs(x - n_parent))
        
        for new_n in dims_list[:5]:
            if time_left() < 150:
                break
            h_interp = interpolate_to_n(best_h, new_n)
            h_lse, c_lse = lse_descent(h_interp, beta=80, max_iters=50)
            if c_lse < best_c - 1e-13:
                best_h = h_lse.copy()
                best_c = c_lse
            h_ase, c_ase = active_set_descent(best_h, max_iters=40, n_active=10)
            if c_ase < best_c - 1e-13:
                best_h = h_ase.copy()
                best_c = c_ase
    
    # Core: Pair descent (main new algorithm)
    if time_left() > 250:
        h_pair, c_pair = pair_descent(best_h, max_rounds=100, n_pairs=40, grid_size=45, rng=rng, time_fn=time_left, min_time=100)
        if c_pair < best_c - 1e-13:
            best_h = h_pair.copy()
            best_c = c_pair
        # LSE polish after pair descent
        if time_left() > 150:
            h_lse, c_lse = lse_descent(best_h, beta=100, max_iters=60, alpha_init=0.003)
            if c_lse < best_c - 1e-13:
                best_h = h_lse.copy()
                best_c = c_lse
    
    # SLSQP
    if time_left() > 120:
        h_sol, c_sol = slsqp_solve(best_h, maxiter=200)
        if h_sol is not None and c_sol < best_c - 1e-12:
            best_h = h_sol.copy()
            best_c = c_sol
            # Polish after SLSQP
            if time_left() > 80:
                h_pair, c_pair = pair_descent(best_h, max_rounds=40, n_pairs=35, grid_size=40, rng=rng, time_fn=time_left, min_time=40)
                if c_pair < best_c - 1e-13:
                    best_h = h_pair.copy()
                    best_c = c_pair
    
    # More aggressive pair descent with larger grid
    if time_left() > 200:
        h_pair, c_pair = pair_descent(best_h, max_rounds=120, n_pairs=50, grid_size=55, rng=rng, time_fn=time_left, min_time=80)
        if c_pair < best_c - 1e-13:
            best_h = h_pair.copy()
            best_c = c_pair
        h_ase, c_ase = active_set_descent(best_h, max_iters=80, n_active=10)
        if c_ase < best_c - 1e-13:
            best_h = h_ase.copy()
            best_c = c_ase
    
    # Multi-start exploration
    if time_left() > 150:
        best_h, best_c = multi_start_explore(best_h, best_c, rng, time_left, min_time=60)
    
    # Final SLSQP
    if time_left() > 70:
        h_sol, c_sol = slsqp_solve(best_h, maxiter=300)
        if h_sol is not None and c_sol < best_c - 1e-12:
            best_h = h_sol.copy()
            best_c = c_sol
    
    # Final refinement cycle
    if time_left() > 50:
        h_pair, c_pair = pair_descent(best_h, max_rounds=80, n_pairs=50, grid_size=50, rng=rng, time_fn=time_left, min_time=20)
        if c_pair < best_c - 1e-14:
            best_h = h_pair.copy()
            best_c = c_pair
        h_ase, c_ase = active_set_descent(best_h, max_iters=100, n_active=12, alpha_init=0.002)
        if c_ase < best_c - 1e-14:
            best_h = h_ase.copy()
            best_c = c_ase
        h_lse, c_lse = lse_descent(best_h, beta=150, max_iters=80, alpha_init=0.002)
        if c_lse < best_c - 1e-14:
            best_h = h_lse.copy()
            best_c = c_lse
    
    # Very final pair sweep
    if time_left() > 25:
        h_pair, c_pair = pair_descent(best_h, max_rounds=60, n_pairs=60, grid_size=60, rng=rng, time_fn=time_left, min_time=5)
        if c_pair < best_c - 1e-14:
            best_h = h_pair.copy()
            best_c = c_pair
    
    # Final validation
    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)
    
    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            final_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 2:
            final_c = float(eval_result[0])
    except Exception:
        pass
    
    return best_h, final_c, len(best_h)
