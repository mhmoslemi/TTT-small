import numpy as np
from scipy.optimize import minimize
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx] + 1e-15)
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx] + 1e-15)
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-14:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmin(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = np.linspace(0, 2, n + 1)
    xt = np.linspace(0, 2, new_n + 1)
    h_interp = np.zeros(new_n)
    for k in range(new_n):
        lo_k, hi_k = xt[k], xt[k+1]
        total_mass = 0.0
        total_width = 0.0
        for j in range(n):
            lo_j, hi_j = xs[j], xs[j+1]
            overlap = min(hi_k, hi_j) - max(lo_k, lo_j)
            if overlap > 0:
                total_mass += h[j] * overlap
                total_width += overlap
        h_interp[k] = total_mass / total_width if total_width > 1e-15 else 0.5
    h_interp = np.clip(h_interp, 0.0, 1.0)
    return repair_feasible(h_interp)

def smooth_reflect(h, width=5):
    if width <= 1 or len(h) <= width:
        return h.copy()
    half = width // 2
    h_pad = np.concatenate([h[half:0:-1], h, h[-2:-half-2:-1]])
    kernel = np.array([1.0 - abs(k - half) / (half + 0.5) for k in range(width)])
    kernel /= kernel.sum()
    h_smooth = np.convolve(h_pad, kernel, mode='valid')
    return repair_feasible(h_smooth)

def build_gradient_matrix(h, n, dx):
    M = 2 * n - 1
    G = np.zeros((M, n))
    for m in range(M):
        d = m - (n - 1)
        if d == 0:
            G[m, :] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            G[m, d:] += dx * (1.0 - h[0:n-d])
            G[m, 0:n-d] += -dx * h[d:n]
        else:
            dp = -d
            G[m, 0:n-dp] += dx * (1.0 - h[dp:n])
            G[m, dp:n] += -dx * h[0:n-dp]
    return G

def build_anti_h(y, half):
    n = 2 * half
    h = np.zeros(n)
    h[:half] = y
    h[half:] = 1.0 - y[::-1]
    return h

def active_set_descent(h, max_iters=200, n_active=12, alpha_init=0.005):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best_c = exact_score(h)
    alpha = alpha_init
    stuck = 0
    for it in range(max_iters):
        profile = compute_profile(h)
        c = float(np.max(profile))
        if c + 1e-14 < best_c:
            best_c = c
            stuck = 0
            alpha = min(alpha * 1.08, 0.025)
        margin = max(0.002, 4.0 * (c - float(np.median(profile))))
        active = np.where(profile >= c - margin)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(profile))])
        if len(active) > n_active:
            idx_sorted = np.argsort(profile[active])
            active = active[idx_sorted[-n_active:]]
        G = build_gradient_matrix(h, n, dx)
        G_a = G[active, :]
        w = np.ones(len(active)) / len(active)
        grad = G_a.T @ w
        grad -= grad.mean()
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad /= gnorm
        improved = False
        for a in [alpha, alpha*0.5, alpha*0.25, alpha*0.1, alpha*0.05]:
            h_try = repair_feasible(h - a * grad)
            s = exact_score(h_try)
            if s < best_c - 1e-14:
                h = h_try
                best_c = s
                improved = True
                break
            elif s < c - 1e-14:
                h = h_try
                improved = True
                break
        if not improved:
            alpha *= 0.55
            stuck += 1
            if stuck > 40 or alpha < 1e-9:
                break
    return h, best_c

def lse_descent(h, beta=80.0, max_iters=150, alpha_init=0.004):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best_c = exact_score(h)
    alpha = alpha_init
    stuck = 0
    for it in range(max_iters):
        profile = compute_profile(h)
        scaled = beta * profile
        max_s = np.max(scaled)
        shifted = scaled - max_s
        weights = np.exp(shifted)
        wsum = weights.sum()
        if wsum < 1e-300:
            break
        weights /= wsum
        G = build_gradient_matrix(h, n, dx)
        grad = G.T @ weights
        grad -= grad.mean()
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad /= gnorm
        improved = False
        for a in [alpha, alpha*0.5, alpha*0.25, alpha*0.1]:
            h_try = repair_feasible(h - a * grad)
            s = exact_score(h_try)

                improved = True
                break

def pair_line_search(h, i, j, n_grid=21):
    a, b = h[i], h[j]
    t_low = max(-a, b - 1.0)
    t_high = min(1.0 - a, b)
    if t_high - t_low < 1e-12:
        return None, None
    h_work = h.copy()
    c_current = float(np.max(compute_profile(h_work)))
    best_t = 0.0
    best_c = c_current
    ts = np.linspace(t_low, t_high, n_grid)
    for t in ts:
        h_work[i] = a + t
        h_work[j] = b - t
        c = float(np.max(compute_profile(h_work)))
        if c < best_c:
            best_c = c
            best_t = t
    if best_c >= c_current - 1e-14:
        return None, None
    ts_fine = np.linspace(t_low, t_high, n_grid * 4)
    for t in ts_fine:
        h_work[i] = a + t
        h_work[j] = b - t
        c = float(np.max(compute_profile(h_work)))
        if c < best_c:
            best_c = c
            best_t = t
    h_work[i] = a + best_t
    h_work[j] = b - best_t
    h_work = np.clip(h_work, 0, 1)
    c_final = exact_score(h_work)
    if c_final < c_current - 1e-12:
        return h_work, c_final
    return None, None

def slsqp_solve(h0, maxiter=200):
    n = len(h0)
    dx = 2.0 / n
    M = 2 * n - 1
    c0 = exact_score(h0)

    def obj(x):
        return x[-1]

    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g

    def eq_fun(x):
        return x[:-1].sum() - n / 2.0

    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g

    def ineq_fun(x):
        h = np.clip(x[:-1], 0, 1)
        t = x[-1]
        profile = np.correlate(h, 1 - h, mode="full") * dx
        return t - profile

    def ineq_jac(x):
        h = np.clip(x[:-1], 0, 1)
        G = build_gradient_matrix(h, n, dx)
        J = np.zeros((M, n + 1))
        J[:, :-1] = -G
        J[:, -1] = 1.0
        return J

    x0 = np.concatenate([h0, [c0 + 1e-3]])
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 0.5)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None and np.isfinite(res.x).all():
            h_new = repair_feasible(res.x[:-1])
            c_new = exact_score(h_new)
            if c_new <= c0 + 1e-9:
                return h_new, c_new
    except Exception:
        pass
    return None, None

        dims = set()
        for factor in [0.8, 0.85, 0.9, 0.95, 1.0, 1.05, 1.1, 1.15, 1.2, 1.3, 1.4]:
            d = int(round(n_cur * factor / 2)) * 2
            d = max(24, min(d, 160))
            dims.add(d)
        dims.discard(n_cur)
        dims_list = sorted(dims, key=lambda x: abs(x - n_cur))
        for new_n in dims_list[:8]:
            if time_left() < 50:
                break
            h_interp = interpolate_to_n(best_h, new_n)
            h_smooth = smooth_reflect(h_interp, width=5)
            # Try both interpolated and smoothed
            for h_start in [h_interp, h_smooth]:
                h_lse, c_lse = lse_descent(h_start, beta=80, max_iters=60)
                if c_lse < best_c - 1e-13:
                    best_h = h_lse.copy()
                    best_c = c_lse
                h_ase, c_ase = active_set_descent(best_h, max_iters=60, n_active=10)
                if c_ase < best_c - 1e-13:
                    best_h = h_ase.copy()
                    best_c = c_ase
                h_pd, c_pd = pair_descent(best_h, max_rounds=15, rng=rng, time_limit=15, t_start=t_start)
                if c_pd < best_c - 1e-13:
                    best_h = h_pd.copy()
                    best_c = c_pd

    # Stage 3: Main hybrid optimization loop
    if time_left() > 120:
        for outer in range(6):
            if time_left() < 60:
                break
            h_pd, c_pd = pair_descent(best_h, max_rounds=30, n_inc=10, n_dec=10, rng=rng, time_limit=30, t_start=t_start)
            if c_pd < best_c - 1e-13:
                best_h = h_pd.copy()
                best_c = c_pd
            h_ase, c_ase = active_set_descent(best_h, max_iters=150, n_active=12, alpha_init=0.004)
            if c_ase < best_c - 1e-13:
                best_h = h_ase.copy()
                best_c = c_ase
            h_lse, c_lse = lse_descent(best_h, beta=100, max_iters=100, alpha_init=0.003)
            if c_lse < best_c - 1e-13:
                best_h = h_lse.copy()
                best_c = c_lse

    # Stage 4: SLSQP full
    if time_left() > 80:
        h_sol, c_sol = slsqp_solve(best_h, maxiter=300)
        if h_sol is not None and c_sol < best_c - 1e-12:
            best_h = h_sol.copy()
            best_c = c_sol
            h_pd, c_pd = pair_descent(best_h, max_rounds=20, rng=rng, time_limit=20, t_start=t_start)
            if c_pd < best_c - 1e-13:
                best_h = h_pd.copy()
                best_c = c_pd

    # Stage 5: SLSQP anti-symmetric
    if time_left() > 60:
        n_b = len(best_h)
        half_b = n_b // 2
        y_best = get_anti_y(best_h)
        h_sol_a, c_sol_a = slsqp_antisym(y_best, maxiter=300)
        if h_sol_a is not None and c_sol_a < best_c - 1e-12:
            best_h = h_sol_a.copy()
            best_c = c_sol_a
            h_pd, c_pd = pair_descent(best_h, max_rounds=20, rng=rng, time_limit=15, t_start=t_start)
            if c_pd < best_c - 1e-13:
                best_h = h_pd.copy()
                best_c = c_pd

    # Stage 6: Multi-start with perturbations
    if time_left() > 100:
        n_b = len(best_h)
        starts = [best_h.copy()]
        for eps in [0.003, 0.008, 0.015, 0.03]:
            z = rng.randn(n_b)
            z -= z.mean()
            starts.append(repair_feasible(best_h + eps * z))
        z_sym = (rng.randn(n_b) + rng.randn(n_b)[::-1]) / 2.0
        z_sym -= z_sym.mean()
        starts.append(repair_feasible(best_h + 0.01 * z_sym))
        for h0 in starts:
            if time_left() < 30:
                break
            h_pd, c_pd = pair_descent(h0, max_rounds=20, rng=rng, time_limit=15, t_start=t_start)
            if c_pd < best_c - 1e-13:
                best_h = h_pd.copy()
                best_c = c_pd
            h_lse, c_lse = lse_descent(h0 if c_pd >= best_c else best_h, beta=80, max_iters=80)
            if c_lse < best_c - 1e-13:
                best_h = h_lse.copy()
                best_c = c_lse
            h_ase, c_ase = active_set_descent(best_h, max_iters=80, n_active=10)
            if c_ase < best_c - 1e-13:
                best_h = h_ase.copy()
                best_c = c_ase

    # Stage 7: Resolution fine-tuning
    if time_left() > 60:
        n_final = len(best_h)
        for dn in [-4, -2, 2, 4, 6]:
            if time_left() < 25:
                break
            new_n = n_final + dn
            if new_n < 24 or new_n > 160 or new_n % 2 != 0:
                continue
            h_interp = interpolate_to_n(best_h, new_n)
            h_pd, c_pd = pair_descent(h_interp, max_rounds=20, rng=rng, time_limit=15, t_start=t_start)
            if c_pd < best_c - 1e-13:
                best_h = h_pd.copy()
                best_c = c_pd
            h_ase, c_ase = active_set_descent(best_h, max_iters=80, n_active=10)
            if c_ase < best_c - 1e-13:
                best_h = h_ase.copy()
                best_c = c_ase
            h_lse, c_lse = lse_descent(best_h, beta=120, max_iters=80)
            if c_lse < best_c - 1e-13:
                best_h = h_lse.copy()
                best_c = c_lse

    # Stage 8: Intensive pair descent with different parameters
    if time_left() > 50:
        h_pd, c_pd = pair_descent(best_h, max_rounds=50, n_inc=12, n_dec=12, rng=rng, time_limit=40, t_start=t_start)
        if c_pd < best_c - 1e-13:
            best_h = h_pd.copy()
            best_c = c_pd

    # Stage 9: Varying active set sizes
    if time_left() > 40:
        for n_act in [6, 8, 15, 20, 30]:
            if time_left() < 15:
                break
            h_ase, c_ase = active_set_descent(best_h, max_iters=100, n_active=n_act, alpha_init=0.003)
            if c_ase < best_c - 1e-13:
                best_h = h_ase.copy()
                best_c = c_ase
            h_pd, c_pd = pair_descent(best_h, max_rounds=15, rng=rng, time_limit=10, t_start=t_start)
            if c_pd < best_c - 1e-13:
                best_h = h_pd.copy()
                best_c = c_pd

    # Stage 10: Varying LSE beta
    if time_left() > 30:
        for beta in [40, 150, 250, 400]:
            if time_left() < 12:
                break
            h_lse, c_lse = lse_descent(best_h, beta=beta, max_iters=100, alpha_init=0.003)
            if c_lse < best_c - 1e-13:
                best_h = h_lse.copy()
                best_c = c_lse

    # Stage 11: Final SLSQP polish
    if time_left() > 35:
        h_sol, c_sol = slsqp_solve(best_h, maxiter=400)
        if h_sol is not None and c_sol < best_c - 1e-14:
            best_h = h_sol.copy()
            best_c = c_sol

    # Stage 12: Final pair + active set
    if time_left() > 20:
        h_pd, c_pd = pair_descent(best_h, max_rounds=30, n_inc=10, n_dec=10, rng=rng, time_limit=12, t_start=t_start)
        if c_pd < best_c - 1e-14:
            best_h = h_pd.copy()
            best_c = c_pd
        h_ase, c_ase = active_set_descent(best_h, max_iters=200, n_active=10, alpha_init=0.002)
        if c_ase < best_c - 1e-14:
            best_h = h_ase.copy()
            best_c = c_ase
        h_lse, c_lse = lse_descent(best_h, beta=150, max_iters=150, alpha_init=0.002)
        if c_lse < best_c - 1e-14:
            best_h = h_lse.copy()
            best_c = c_lse

    # Final validation
    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)

    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            final_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 2:
            final_c = float(eval_result[0])
    except Exception:
        pass

    return best_h, final_c, len(best_h)
