Let me also think about **what happens at the optimum**. At the optimum of the min-max problem, the KKT conditions say:
- There exist λ_k ≥ 0 for each active constraint k, with ∑λ_k = 1
- The gradient of the objective (which is ∑λ_k * grad_k) is in the normal cone of the feasible set at h*

- If 0 < h*_j < 1: ∑λ_k * G[k,j] = μ (the Lagrange multiplier for the mass constraint)
- If h*_j = 0: ∑λ_k * G[k,j] ≥ μ
- If h*_j = 1: ∑λ_k * G[k,j] ≤ μ

This gives us a way to check proximity to optimality and to identify which lags should be active.

    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def build_gradient(h, n, dx):
    M = 2 * n - 1
    G = np.zeros((M, n))
    center = n - 1
    G[center, :] = dx * (1.0 - 2.0 * h)
    for d in range(1, n):
        G[center + d, 0:n-d] += dx * (1.0 - h[d:n])
        G[center + d, d:n] += -dx * h[0:n-d]
        G[center - d, d:n] += dx * (1.0 - h[0:n-d])
        G[center - d, 0:n-d] += -dx * h[d:n]
    return G

def repair_feasible(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, 1.0 - h[idx])
            h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)[::-1]
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    # Final fix
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
    return np.clip(h, 0.0, 1.0)

def lp_step_full(h, radius, use_all=True, beam=0.01):
    """Full LP step using all (or active) lags."""
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    center = n - 1
    profile = compute_profile(h)
    T = profile.max()
    
    # Select constraints
    if use_all:
        active = np.arange(M)
    else:
        thresh = T - beam
        active = np.where(profile >= thresh)[0]
        if len(active) == 0:
            active = np.array([np.argmax(profile)])
    
    G = build_gradient(h, n, dx)
    G_a = G[active, :]
    p_a = profile[active]
    
    # LP: minimize t
    # s.t. G_a[k,:] · Δh + p_a[k] ≤ t  for each k in active
    #      sum(Δh) = 0
    #      -min(h, radius) ≤ Δh ≤ min(1-h, radius)
    
    n_vars = n + 1  # Δh (n vars) + t (1 var)
    c_obj = np.zeros(n_vars)
    c_obj[-1] = 1.0
    
    # Inequality: G_a[k,:] · Δh - t ≤ -p_a[k]
    A_ub = np.zeros((len(active), n_vars))
    A_ub[:, :n] = G_a
    A_ub[:, -1] = -1.0
    b_ub = -p_a
    
    # Equality: sum(Δh) = 0
    A_eq = np.zeros((1, n_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    lower = np.maximum(-h, -radius)
    upper = np.minimum(1.0 - h, radius)
    bounds = list(zip(lower, upper))
    bounds.append((0.0, None))  # t ≥ 0
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'maxiter': 5000})
        if res.success and res.x is not None:
            delta = res.x[:n]
            h_new = h + delta
            h_new = repair_feasible(h_new)
            return h_new, float(res.x[-1])
    except Exception:
        pass
    return None, None

def subgradient_step(h, active_set, step_size):
    """Move in the negative weighted subgradient direction."""
    n = len(h)
    dx = 2.0 / n
    G = build_gradient(h, n, dx)
    profile = compute_profile(h)
    T = profile.max()
    
    # Weights: proportional to how close each active constraint is to T
    p_a = profile[active_set]
    weights = np.maximum(p_a - (T - 0.1 * (T - p_a.min())), 0.01)
    weights /= weights.sum()
    
    grad = G[active_set, :].T @ weights
    grad -= grad.mean()  # Remove mass constraint component
    norm = np.linalg.norm(grad)
    if norm < 1e-15:
        return h.copy()
    
    h_new = h - step_size * grad / norm * np.linalg.norm(h)  # Scale appropriately
    return repair_feasible(h_new)

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

            no_improve = 0

        else:

        else:
            h_new, t_new = lp_step_full(h, radius * 0.5, use_all=False, beam=0.02)
        
        if h_new is not None and exact_score(h_new) < best_c:
            h = h_new
        else:
            # Try subgradient
            active = np.where(profile >= c - 0.02)[0]
            for ss in [0.01, 0.005, 0.002]:
                h_sg = subgradient_step(h, active, ss)
                if exact_score(h_sg) < best_c:

                    break
            else:

                if radius < 0.002:

    return best_h, best_c

import numpy as np
from scipy.optimize import linprog, minimize
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def build_gradient(h, n, dx):
    M = 2 * n - 1
    G = np.zeros((M, n))
    center = n - 1
    G[center, :] = dx * (1.0 - 2.0 * h)
    for d in range(1, n):
        G[center + d, 0:n-d] += dx * (1.0 - h[d:n])
        G[center + d, d:n] += -dx * h[0:n-d]
        G[center - d, d:n] += dx * (1.0 - h[0:n-d])
        G[center - d, 0:n-d] += -dx * h[d:n]
    return G

def repair_feasible(h):
    n = len(h)
    h = np.clip(np.asarray(h, dtype=np.float64), 0.0, 1.0).copy()
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, 1.0 - h[idx])
            h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)[::-1]
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
    return np.clip(h, 0.0, 1.0)

def symmetrize(h):
    return (h + h[::-1]) / 2.0

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

def lp_step(h, radius, use_all=True, beam=0.015):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    profile = compute_profile(h)
    T = profile.max()
    if use_all:
        active = np.arange(M)
    else:
        thresh = T - beam
        active = np.where(profile >= thresh)[0]
        if len(active) == 0:
            active = np.array([np.argmax(profile)])
    G = build_gradient(h, n, dx)
    G_a = G[active, :]
    p_a = profile[active]
    n_vars = n + 1
    c_obj = np.zeros(n_vars)
    c_obj[-1] = 1.0
    A_ub = np.zeros((len(active), n_vars))
    A_ub[:, :n] = G_a
    A_ub[:, -1] = -1.0
    b_ub = -p_a
    A_eq = np.zeros((1, n_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    lower = np.maximum(-h, -radius)
    upper = np.minimum(1.0 - h, radius)
    bounds = list(zip(lower, upper))
    bounds.append((0.0, None))
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'maxiter': 5000})
        if res.success and res.x is not None and np.isfinite(res.x).all():
            delta = res.x[:n]
            h_new = repair_feasible(h + delta)
            return h_new
    except Exception:
        pass
    return None

def smooth_max_obj(h, beta):
    n = len(h)
    dx = 2.0 / n
    profile = np.correlate(h, 1 - h, mode="full") * dx
    max_p = profile.max()
    shifted = profile - max_p
    return max_p + np.log(np.sum(np.exp(beta * shifted))) / beta

def smooth_max_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    profile = np.correlate(h, 1 - h, mode="full") * dx
    max_p = profile.max()
    shifted = beta * (profile - max_p)
    weights = np.exp(shifted) / np.sum(np.exp(shifted))
    G = build_gradient(h, n, dx)
    grad = G.T @ weights
    grad -= grad.mean()
    return grad, profile.max()

def slsqp_polish(h, t_start, time_limit):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    c0 = exact_score(h)
    def obj_fn(x):
        return x[-1]
    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    def eq_fn(x):
        return x[:-1].sum() - n / 2.0
    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g
    def ineq_fn(x):
        h = x[:-1]
        t = x[-1]
        prof = np.correlate(h, 1 - h, mode="full") * dx
        return t - prof
    def ineq_jac(x):
        h = x[:-1]
        G = build_gradient(h, n, dx)
        J = np.zeros((M, n + 1))
        J[:, :-1] = -G
        J[:, -1] = 1.0
        return J
    x0 = np.concatenate([h, [c0 + 5e-4]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    cons = [
        {'type': 'eq', 'fun': eq_fn, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fn, 'jac': ineq_jac}
    ]
    try:
        res = minimize(obj_fn, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': 800, 'ftol': 1e-16})
        if res.success and np.isfinite(res.x).all():
            h_sol = repair_feasible(res.x[:-1])
            return h_sol
    except Exception:
        pass
    return None

def transfer_moves(h, n_trials, rng, eps=0.02):
    n = len(h)
    best_h = h.copy()
    best_c = exact_score(h)
    for _ in range(n_trials):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        delta = eps * (2 * rng.rand() - 1)
        if h[i] + delta < 0 or h[i] + delta > 1:

        if h[j] - delta < 0 or h[j] - delta > 1:

        s = exact_score(h_try)

                best_c = c_sol

    if time_left() > 10:
        h_opt, c_opt = optimize_lp(best_h, t_start, t_start + min(20, time_left()-3), max_iters=100)
        if c_opt < best_c - 1e-14:
            best_h = h_opt
            best_c = c_opt

    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)

    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            final_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 2:
            final_c = float(eval_result[0])
    except Exception:
        pass

    return best_h, final_c, len(best_h)
