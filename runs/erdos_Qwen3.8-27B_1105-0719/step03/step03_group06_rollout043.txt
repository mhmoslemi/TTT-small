def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(-(1.0 - h))
        remaining = diff
        for i in range(n):
            idx = order[i]
            cap = 1.0 - h[idx]
            if cap <= 1e-15:
                continue
            add = min(remaining, cap)
            h[idx] += add
            remaining -= add
            if remaining <= 1e-14:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            if h[idx] <= 1e-15:
                continue
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-14:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:

out[m] = c_{m-(n-1)} = Σ_i h[i] * (1-h)[i-(m-(n-1))] = Σ_i h[i] * (1-h)[i-m+n-1]

Let d = m - (n-1). Then out[m] = Σ_i h[i] * (1-h)[i-d]

For validity: 0 ≤ i < n and 0 ≤ i-d < n, i.e., d ≤ i < d+n.
Combined: max(0, d) ≤ i < min(n, d+n) = n (since d < n for valid m).
So: max(0, d) ≤ i ≤ n-1.

Case d ≥ 0: i ranges from d to n-1.
out[m] = Σ_{i=d}^{n-1} h[i] * (1-h[i-d]) = Σ_{i=d}^{n-1} h[i] - Σ_{i=d}^{n-1} h[i]*h[i-d]

Case d < 0: let p = -d > 0. i ranges from 0 to n-1-p.
out[m] = Σ_{i=0}^{n-1-p} h[i] * (1-h[i+p]) = Σ_{i=0}^{n-1-p} h[i] - Σ_{i=0}^{n-1-p} h[i]*h[i+p]

Now the gradient:
For d ≥ 0: out[m] = Σ_{i=d}^{n-1} h[i](1-h[i-d])
∂/∂h[j] = (1-h[j-d])*1_{d≤j≤n-1} - h[j+d]*1_{0≤j+d≤n-1, d≤j+d≤n-1}
= (1-h[j-d])*1_{j≥d} - h[j+d]*1_{j≤n-1-d}

So: g[j] = (1-h[j-d])*1_{j≥d} - h[j+d]*1_{j≤n-1-d} (times dx)

import numpy as np
from scipy.optimize import minimize
import time


def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(-(1.0 - h))
        remaining = diff
        for i in range(n):
            idx = order[i]
            cap = 1.0 - h[idx]
            if cap <= 1e-15:
                continue
            add = min(remaining, cap)
            h[idx] += add
            remaining -= add
            if remaining <= 1e-14:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            if h[idx] <= 1e-15:
                continue
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-14:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] = float(np.clip(h[idx] + diff2, 0.0, 1.0))
    return h


def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))


def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx


def grad_row(h, m, n):
    dx = 2.0 / n
    d = m - (n - 1)
    g = np.zeros(n)
    if d == 0:
        g[:] = dx * (1.0 - 2.0 * h)
    elif d > 0:
        g[d:] += dx * (1.0 - h[:-d])
        g[:-d] -= dx * h[d:]
    else:
        p = -d
        g[:-p] += dx * (1.0 - h[p:])
        g[p:] -= dx * h[:-p]
    return g


def build_qp_matrices(n, rho):
    B = np.zeros((n, n))
    w = np.zeros(n)
    for i in range(n):
        for j in range(n):
            m = j - i + (n - 1)
            B[i, j] = rho[m]
            w[i] += rho[m]
    return B, w


def solve_qp(B, w, n, h0):
    def obj(h):
        return w @ h - h @ B @ h

    def grad(h):
        return w - 2.0 * B @ h

    bounds = [(0.0, 1.0)] * n
    ones = np.ones(n)

    def eq_fun(h):
        return h.sum() - n / 2.0

    def eq_jac(h):
        return ones

    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    res = minimize(obj, h0, jac=grad, method='L-BFGS-B',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': 500, 'ftol': 1e-16})
    return repair_feasible(res.x)


def binary_project(h):
    n = len(h)
    target = n // 2
    idx = np.argsort(-h)[:target]
    b = np.zeros(n)
    b[idx] = 1.0
    return b


def lse_descent(h, max_iters=300, temp_start=20.0, temp_end=20000.0, n_stages=12):
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    best_h = h.copy()
    temps = np.geomspace(temp_start, temp_end, n_stages)
    iters_per = max(3, max_iters // n_stages)
    momentum_vec = np.zeros(n)
    mom = 0.35
    for temp in temps:
        for it in range(iters_per):
            profile = compute_profile(h)
            c = float(profile.max())
            shifted = profile - profile.max()
            exp_vals = np.exp(temp * shifted)
            ts = exp_vals.sum()
            if ts < 1e-30:
                break
            weights = exp_vals / ts
            g = np.zeros(n)
            M = 2 * n - 1
            for m in range(M):
                if weights[m] > 1e-8:
                    g += weights[m] * grad_row(h, m, n)
            g -= g.mean()
            gnorm = np.linalg.norm(g)
            if gnorm < 1e-15:
                break
            momentum_vec = mom * momentum_vec + g
            mvn = np.linalg.norm(momentum_vec)
            if mvn < 1e-15:
                break
            d = -momentum_vec / mvn
            best_val = c
            best_h_step = h.copy()
            best_alpha = 0.0
            for frac in [0.4, 0.25, 0.15, 0.1, 0.06, 0.03, 0.015, 0.005]:
                h_try = repair_feasible(h + frac * d)
                val = exact_score(h_try)
                if val < best_val - 1e-15:
                    best_val = val
                    best_alpha = frac
                    best_h_step = h_try.copy()
            if best_alpha == 0.0:
                break
            h = best_h_step
            if best_val < best_c:
                best_c = best_val
                best_h = h.copy()
    return best_h, best_c


def game_search(h_init, n, time_limit=120.0):
    t0 = time.perf_counter()
    M = 2 * n - 1
    best_h = h_init.copy()
    best_c = exact_score(h_init)
    rho = np.ones(M) / M
    betas = [10, 30, 80, 200, 500]
    for beta in betas:
        if time.perf_counter() - t0 > time_limit:
            break
        for it in range(25):
            if time.perf_counter() - t0 > time_limit:
                break
            B, w = build_qp_matrices(n, rho)
            h_new = solve_qp(B, w, n, best_h)
            profile = compute_profile(h_new)
            c = float(profile.max())
            if c < best_c:
                best_c = c
                best_h = h_new.copy()
            hb = binary_project(h_new)
            cb = exact_score(hb)
            if cb < best_c:
                best_c = cb
                best_h = hb
            pmax = profile.max()
            rho = rho * np.exp(beta * (profile - pmax))
            rs = rho.sum()
            if rs > 0 and np.isfinite(rs):
                rho /= rs
            else:
                rho = np.ones(M) / M
    return best_h, best_c


def active_descent_tight(h, max_iters=200, tol=0.001):
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    best_h = h.copy()
    mv = np.zeros(n)
    mom = 0.3
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for it in range(max_iters):
        profile = compute_profile(h)
        c = float(profile.max())
        thresh = c - tol
        active = np.where(profile >= thresh)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(profile))])
        if len(active) > 25:
            top = np.argsort(profile[active])[-25:]
            active = active[top]
        g = np.zeros(n)
        for m in active:
            g += grad_row(h, m, n)
        g -= g.mean()
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            break
        mv = mom * mv + g
        mvn = np.linalg.norm(mv)
        if mvn < 1e-15:
            break
        d = -mv / mvn
        pos = d > 1e-12
        neg = d < -1e-12
        a_max = 1.0
        if pos.any():
            a_max = min(a_max, float(((1.0 - h[pos]) / d[pos]).min()))
        if neg.any():
            a_max = min(a_max, float((h[neg] / (-d[neg])).min()))
        a_max = max(a_max, 1e-10)
        a_max = min(a_max, 0.5)
        best_val = c
        best_h_step = h.copy()
        a_lo, a_hi = 0.0, a_max
        for _ in range(20):
            c1 = a_hi - gr * (a_hi - a_lo)
            c2 = a_lo + gr * (a_hi - a_lo)
            h1 = repair_feasible(h + c1 * d)
            h2 = repair_feasible(h + c2 * d)
            f1 = exact_score(h1)
            f2 = exact_score(h2)
            if f1 < best_val:
                best_val = f1
                best_h_step = h1.copy()
            if f2 < best_val:
                best_val = f2
                best_h_step = h2.copy()
            if f1 < f2:
                a_hi = c2
            else:
                a_lo = c1
            if a_hi - a_lo < 1e-14:
                break
        if best_val >= c - 1e-15:
            break
        h = best_h_step
        if best_val < best_c:
            best_c = best_val
            best_h = h.copy()
    return best_h, best_c


def interp_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    edges_old = np.linspace(0.0, 2.0, n + 1)
    edges_new = np.linspace(0.0, 2.0, new_n + 1)
    h_new = np.zeros(new_n)
    for k in range(new_n):
        a, b = edges_new[k], edges_new[k + 1]
        total = 0.0
        for j in range(n):
            lo = max(a, edges_old[j])
            hi = min(b, edges_old[j + 1])
            if hi > lo:
                total += h[j] * (hi - lo)
        h_new[k] = total / (b - a)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)


def coordinate_polish(h, max_iters=60):
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    for it in range(max_iters):
        profile = compute_profile(h)
        c = float(profile.max())
        thresh = c - 0.002
        active = np.where(profile >= thresh)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(profile))])
        improved = False
        idx_desc = np.argsort(-h)
        idx_asc = np.argsort(h)
        for attempt in range(20):
            i = int(idx_desc[attempt % n])
            j = int(idx_asc[attempt % n])
            if i == j:
                continue
            delta = 0.01 * (h[i] - h[j])
            if abs(delta) < 1e-6:
                continue
            for sign in [1.0, -1.0]:
                h_try = h.copy()
                h_try[i] += sign * delta
                h_try[j] -= sign * delta
                if h_try.min() < -1e-10 or h_try.max() > 1 + 1e-10:
                    continue
                h_try = repair_feasible(h_try)
                s = exact_score(h_try)
                if s < best_c - 1e-14:
                    h = h_try
                    best_c = s
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
    return h, best_c


def binary_local_search(h, rng, max_swaps=5000):
    n = len(h)
    h = binary_project(h)
    best_c = exact_score(h)
    target = n // 2
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    t0 = time.perf_counter()
    for _ in range(max_swaps):
        if time.perf_counter() - t0 > 30:
            break
        i = int(ones[rng.randint(len(ones))])
        j = int(zeros[rng.randint(len(zeros))])
        if i == j:
            continue
        h[i] = 0.0
        h[j] = 1.0
        s = exact_score(h)
        if s < best_c - 1e-14:
            best_c = s
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
        else:
            h[i] = 1.0
            h[j] = 0.0
    return h, best_c
