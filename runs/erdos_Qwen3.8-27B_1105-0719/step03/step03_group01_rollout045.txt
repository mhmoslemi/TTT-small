            lo = mid
        else:
            hi = mid

    Returns array of shape (2n-1, n+1) where row k is
    [1, -dx * d(c_k)/dh[0], ..., -dx * d(c_k)/dh[n-1]]

    m = 2 * n - 1
    J = np.zeros((m, n + 1))
    J[:, 0] = 1.0  # coefficient of t

        offset = k - (n - 1)

            grad[:n-d] -= h[d:]

import numpy as np
from scipy.optimize import minimize
import time


def project_feasible(x, target_sum):
    n = len(x)
    x = np.clip(x, -2.0, 2.0)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        cnt = mask.sum()
        if cnt > 0:
            h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def c5_of_h(h, dx):
    return float(np.max(compute_corr(h)) * dx)


def build_jacobian(h, dx):
    n = len(h)
    m = 2 * n - 1
    J = np.zeros((m, n))
    nrm1 = n - 1
    for k in range(m):
        offset = k - nrm1
        if offset >= 0:
            J[k, :n - offset] += (1.0 - h[offset:])
            J[k, offset:] -= h[:n - offset]
        else:
            d = -offset
            J[k, d:] += (1.0 - h[:n - d])
            J[k, :n - d] -= h[d:]
    J *= dx
    return J


def slsqp_solve(h, dx, target_sum, maxiter=150, ftol=1e-10, n_active=None):
    n = len(h)
    m = 2 * n - 1
    corrs = compute_corr(h) * dx
    current_c5 = float(corrs.max())

    if n_active is None or n_active >= m:
        active_idx = np.arange(m)
    else:
        active_idx = np.argsort(corrs)[-n_active:]
    n_act = len(active_idx)

    t0_val = current_c5 + 1e-7
    x0 = np.concatenate(([t0_val], h))

    def obj_func(x):
        return x[0]

    def obj_grad(x):
        g = np.zeros(n + 1)
        g[0] = 1.0
        return g

    def ineq_fun(x):
        h_x = x[1:]
        c = compute_corr(h_x) * dx
        return x[0] - c[active_idx]

    def ineq_jac(x):
        h_x = x[1:]
        n_local = len(h_x)
        J_full = build_jacobian(h_x, dx)
        J_act = J_full[active_idx]
        jac = np.zeros((n_act, n_local + 1))
        jac[:, 0] = 1.0
        jac[:, 1:] = -J_act
        return jac

    def eq_fun(x):
        return np.sum(x[1:]) - target_sum

    def eq_jac(x):
        J = np.zeros(n + 1)
        J[1:] = 1.0
        return J

    bounds = [(max(0.0, current_c5 - 0.02), 0.5)] + [(0.0, 1.0)] * n

        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac},

def slsqp_loop(h, dx, target_sum, t0, time_limit, max_rounds=8):

def active_set_lp_step(h, dx, target_sum, n_active, alpha):

def lp_optimize(h, dx, target_sum, time_limit, alpha_start=0.2,
