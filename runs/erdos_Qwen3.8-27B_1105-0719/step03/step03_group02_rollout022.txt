import numpy as np
from scipy.optimize import minimize, linprog
from scipy.special import logsumexp
import time


def project_feasible(x, target_sum):
    """Project x onto {0<=h<=1, sum(h)=target_sum} via bisection on shift."""
    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)


def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def c5_of_h(h, dx):
    return compute_corrs(h).max() * dx


def grad_corr_k(h, k, n):
    """Gradient of corr[k] with respect to h, where k is the index in the full correlation array."""
    offset = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -offset)
    i_hi = min(n - 1, n - 1 - offset)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx + offset] += (1.0 - h[idx])
    grad[idx] += (-h[idx + offset])
    return grad


def all_grad_matrix(h, n):
    """Compute gradient matrix: grad_mat[k, i] = d corr[k] / d h[i]"""
    mat = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        mat[k] = grad_corr_k(h, k, n)
    return mat


def lp_step(h, dx, target_sum, corrs, active_k, trust_radius):
    """One LP step: find delta that reduces max active correlation."""
    n = len(h)
    # Build gradient rows for active k's
    A_rows = []
    b_rows = []
    for k in active_k:
        gk = grad_corr_k(h, k, n)
        row = np.zeros(n + 1)
        row[:n] = dx * gk
        row[n] = -1.0
        A_rows.append(row)
        b_rows.append(-corrs[k] * dx - 1e-15)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    bounds = []
    for i in range(n):
        bounds.append((max(-h[i], -trust_radius), min(1.0 - h[i], trust_radius)))
    bounds.append((None, None))  # t is free
    
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    try:
        res = linprog(c_obj, A_ub=np.array(A_rows), b_ub=np.array(b_rows),
                     A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    except Exception:
        return None, None
    
    if res.status != 0:
        return None, None
    
    delta = res.x[:n]
    t_val = res.x[n]
    return delta, t_val

def slsqp_smooth(h0, dx, target_sum, beta, maxiter=1000):
    """SLSQP with smooth max objective."""
    n = len(h0)
    
    class SlsqpProb:
        def __init__(self, dx, target_sum, beta):
            self.dx = dx
            self.target_sum = target_sum
            self.beta = beta
        
        def obj(self, x):
            corrs = compute_corrs(x)
            return logsumexp(self.beta * corrs) / self.beta * self.dx
        
        def grad(self, x):
            n = len(x)
            corrs = compute_corrs(x)
            shifted = corrs - corrs.max()
            w = np.exp(self.beta * shifted)
            w_sum = w.sum()
            if w_sum < 1e-30:
                w = np.ones_like(corrs) / len(corrs)
            else:
                w /= w_sum
            grad = np.zeros(n)
            for k in range(2 * n - 1):
                if w[k] > 1e-13:
                    grad += w[k] * grad_corr_k(x, k, n)
            return grad * self.dx
        
        def cfun(self, x):
            return x.sum() - self.target_sum
        
        def cjac(self, x):
            return np.ones(len(x))
    
    prob = SlsqpProb(dx, target_sum, beta)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': prob.cfun, 'jac': prob.cjac}]
    
    try:
        res = minimize(prob.obj, h0, jac=prob.grad, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        h_out = project_feasible(res.x, target_sum)
    except Exception:
        h_out = project_feasible(h0, target_sum)
    
    return h_out, c5_of_h(h_out, dx)

                    if actual_c5 < best_c5 - 1e-14:

def triple_transfer(h0, dx, target_sum, n_top=15, n_rounds=2):
    """Transfer between pairs of pairs: (i1->j1) and (i2->j2) simultaneously."""
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    
    for rnd in range(n_rounds):
        corrs = compute_corrs(h)
        shifted = corrs - corrs.max()
        w = np.exp(80.0 * shifted)
        w /= w.sum()
        
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] > 1e-10:
                grad += w[k] * grad_corr_k(h, k, n)
        grad *= dx
        
        dec_idx = np.argsort(grad)[-n_top:]
        inc_idx = np.argsort(grad)[:n_top]
        
        improved = False
        # Try pairs of transfers: (i1->j1) and (i2->j2)
        for idx_a in range(min(n_top, len(dec_idx))):
            for idx_b in range(idx_a + 1, min(n_top, len(dec_idx))):
                i1, i2 = dec_idx[idx_a], dec_idx[idx_b]
                for j_idx in range(min(n_top, len(inc_idx))):
                    j1 = inc_idx[j_idx]
                    if i1 == j1 or i2 == j1:
                        continue
                    # Direction: -e_i1 - e_i2 + 2*e_j1 (move 1 from i1, 1 from i2, 2 to j1)
                    # Actually let's do: -eps*e_i1 - eps*e_i2 + 2*eps*e_j1
                    # Or simpler: two simultaneous pair transfers with same eps
                    # h[i1] -= eps, h[i2] -= eps, h[j1] += 2*eps
                    # Constraint: eps <= min(h[i1], h[i2]), 2*eps <= 1 - h[j1]
                    eps_hi = min(h[i1], h[i2], (1.0 - h[j1]) / 2.0)
                    eps_lo = 0.0
                    if eps_hi < 1e-12:
                        continue
                    
                    d = np.zeros(2 * n - 1)
                    for k in range(2 * n - 1):
                        gk = grad_corr_k(h, k, n)
                        d[k] = -gk[i1] - gk[i2] + 2.0 * gk[j1]
                    
                    # Linear in eps, so optimum is at endpoint or where max changes
                    # f(eps) = max(corrs + eps * d) is convex in eps
                    # Minimum is where the two largest terms are equal
                    val_at_0 = corrs.max()
                    val_at_hi = (corrs + eps_hi * d).max()
                    
                    if val_at_hi >= val_at_0:
                        continue
                    
                    # Bisect for the minimum of max(corrs + eps * d)
                    lo, hi = eps_lo, eps_hi
                    mid = (lo + hi) / 2
                    f_mid = (corrs + mid * d).max()
                    for _ in range(30):
                        hi2 = (lo + hi) / 2
                        f_hi2 = (corrs + hi2 * d).max()
                        lo2 = (hi2 + hi) / 2
                        f_lo2 = (corrs + lo2 * d).max()
                        if f_hi2 <= f_lo2:
                            lo = hi2
                        else:
                            hi = lo2
                    eps_opt = (lo + hi) / 2
                    new_c5 = (corrs + eps_opt * d).max() * dx
                    
                    if new_c5 < best_c5 - 1e-14:
                        new_h = np.array(h, dtype=float)
                        new_h[i1] -= eps_opt
                        new_h[i2] -= eps_opt
                        new_h[j1] += 2.0 * eps_opt
                        new_h = project_feasible(new_h, target_sum)
                        actual_c5 = c5_of_h(new_h, dx)
                        if actual_c5 < best_c5 - 1e-14:
                            h = new_h
                            best_c5 = actual_c5
                            best_h = h.copy()
                            corrs = compute_corrs(h)
                            improved = True
        
        if not improved:
            break
    
    return best_h, best_c5


def subgradient_descent(h0, dx, target_sum, max_iter=5000, beta=50.0, lr_init=0.1):
    """Subgradient method with momentum for the max correlation."""
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    momentum = np.zeros(n)
    lr = lr_init
    beta_mom = 0.85
    no_improve = 0
    
    for it in range(max_iter):
        corrs = compute_corrs(h)
        c5_val = corrs.max() * dx
        
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 2000:
                break
        
        # Compute subgradient: weighted average of active correlation gradients
        shifted = corrs - corrs.max()
        w = np.exp(beta * shifted)
        w /= w.sum()
        
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] > 1e-12:
                grad += w[k] * grad_corr_k(h, k, n)
        grad *= dx
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Momentum update
        momentum = beta_mom * momentum - lr * grad / (gnorm + 1e-10)
        h_new = h + momentum
        h_new = project_feasible(h_new, target_sum)
        new_c5 = c5_of_h(h_new, dx)
        
        # Backtracking
        tries = 0
        while new_c5 > c5_val + 1e-12 and tries < 5:
            lr *= 0.5
            momentum = beta_mom * momentum - lr * grad / (gnorm + 1e-10)
            h_new = h + momentum
            h_new = project_feasible(h_new, target_sum)
            new_c5 = c5_of_h(h_new, dx)
            tries += 1
        
        h = h_new
    
    return best_h, best_c5

def make_structured_seed(n, target_sum, pattern_type='alternating'):
    """Create a structured initial seed."""
    h = np.zeros(n)
    if pattern_type == 'alternating':
        for i in range(n):
            h[i] = 1.0 if i % 2 == 0 else 0.0
    elif pattern_type == 'block3':
        # Period 3: 1, 0, 0 repeating
        for i in range(n):
            h[i] = 1.0 if i % 3 == 0 else 0.0
        # Adjust to get correct sum
    elif pattern_type == 'sin':
        # Sinusoidal pattern
        x = np.linspace(0, 2 * np.pi * n / 10, n)
        h = 0.5 + 0.4 * np.sin(x)
    elif pattern_type == 'random_balance':
        rng = np.random.RandomState(12345)
        h = rng.randint(0, 2, size=n).astype(float)
        # Adjust to get exactly n/2 ones
        ones_mask = h > 0.5
        n_ones = ones_mask.sum()
        if n_ones > n // 2:
            flip_idx = np.where(ones_mask)[0]
            rng.shuffle(flip_idx)
            for idx in flip_idx[:n_ones - n // 2]:
                h[idx] = 0.0
        elif n_ones < n // 2:
            flip_idx = np.where(~ones_mask)[0]
            rng.shuffle(flip_idx)
            for idx in flip_idx[:n // 2 - n_ones]:
                h[idx] = 1.0
    elif pattern_type == 'sturmian':
        # Sturmian word with slope 1/2
        phi = 0.5
        alpha = 0.3
        for i in range(n):
            h[i] = 1.0 if np.floor((i + 1) * phi + alpha) > np.floor(i * phi + alpha) else 0.0
    return project_feasible(h, target_sum)
