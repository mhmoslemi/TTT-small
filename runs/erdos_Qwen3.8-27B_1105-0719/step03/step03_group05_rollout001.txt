def solve_lp_direction(h, G, D=0.5):
    """Solve LP to find best minimax descent direction.
    Minimize t subject to G@b - t <= 0, sum(b)=0, -D <= b <= D
    Returns direction b and predicted t.
    """
    n = len(h)
    M = G.shape[0]
    # Variables: b_0, ..., b_{n-1}, t (total n+1)
    # c = [0, 0, ..., 0, 1] (minimize t)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    # Inequality: G@b - t <= 0 => G@b + (-1)*t <= 0
    # A_ub @ [b, t] <= b_ub
    A_ub = np.zeros((M, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = np.zeros(M)
    # Equality: sum(b) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = [0.0]
    # Bounds: -D <= b_i <= D, t free
    bounds = [(-D, D)] * n + [(None, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                  bounds=bounds, method='highs')
    if res.status == 0:
        b_dir = res.x[:n]
        t_pred = res.x[-1]
        return b_dir, t_pred
    return None, None

def line_search(h, direction, max_evals=50):
    """Exact line search along h + t*direction.
    direction should have sum=0.
    """
    n = len(h)
    # Compute safe step bounds
    pos_mask = direction > 1e-12
    neg_mask = direction < -1e-12
    t_max = np.inf
    if np.any(pos_mask):
        t_max = min(t_max, np.min((1.0 - h[pos_mask]) / direction[pos_mask]))
    if np.any(neg_mask):
        t_max = min(t_max, np.min(-h[neg_mask] / direction[neg_mask]))
    t_max = max(t_max, 1e-12)
    # Sample points
    ts = np.linspace(0, t_max, max_evals)
    best_t = 0
    best_score = exact_score(h)
    for t in ts:
        h_try = h + t * direction
        h_try = np.clip(h_try, 0, 1)
        if abs(h_try.sum() - n/2) > 1e-6:
            continue
        s = exact_score(h_try)
        if s < best_score:
            best_score = s
            best_t = t
    # Refine around best_t
    if best_t > 0:
        lo = max(0, best_t * 0.8)
        hi = min(t_max, best_t * 1.2)
        ts2 = np.linspace(lo, hi, 30)
        for t in ts2:
            h_try = h + t * direction

        m_neg = n - 1 - d
        G[m_pos, d:] += dx * (1.0 - h[:n - d])
        G[m_pos, :n - d] += -dx * h[d:]
        G[m_neg, :n - d] += dx * (1.0 - h[d:])
        G[m_neg, d:] += -dx * h[:n - d]
    return G

def solve_lp_direction(G, P, D):
    n = G.shape[1]
    M = G.shape[0]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((M, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -P.copy()
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(-D, D)] * n + [(None, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            return res.x[:n], res.x[-1]
    except Exception:
        pass
    return None, None

def solve_lp_active(G, P, active_idx, D):
    n = G.shape[1]
    ga = G[active_idx, :]
    pa = P[active_idx]
    M = len(active_idx)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((M, n + 1))
    A_ub[:, :n] = ga
    A_ub[:, n] = -1.0
    b_ub = -pa.copy()
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(-D, D)] * n + [(None, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            return res.x[:n], res.x[-1]
    except Exception:
        pass
    return None, None

def line_search_1d(h, direction, t_max, n_samples=40):
    n = len(h)
    best_t = 0.0
    best_s = exact_score(h)
    ts = np.linspace(0, t_max, n_samples)
    for t in ts:
        h_try = h + t * direction
        if h_try.min() < -1e-9 or h_try.max() > 1 + 1e-9:
            continue
        if abs(h_try.sum() - n / 2) > 1e-6:
            continue
        s = exact_score(h_try)
        if s < best_s:
            best_s = s
            best_t = t
    if best_t <= 0:
        return h, best_s
    lo = max(0.0, best_t * 0.7)
    hi = min(t_max, best_t * 1.3)
    if hi - lo < 1e-15:
        hi = t_max
    ts2 = np.linspace(lo, hi, 30)
    for t in ts2:
        h_try = h + t * direction
        if h_try.min() < -1e-9 or h_try.max() > 1 + 1e-9:
            continue
        s = exact_score(h_try)
        if s < best_s:
            best_s = s
            best_t = t
    h_new = h + best_t * direction
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = repair_feasible(h_new)
    return h_new, best_s

def safe_t_max(h, direction):
    t_max = 1.0
    pos = direction > 1e-12
    neg = direction < -1e-12
    if np.any(pos):
        t_max = min(t_max, np.min((1.0 - h[pos]) / direction[pos]))
    if np.any(neg):
        t_max = min(t_max, np.min(-h[neg] / direction[neg]))
    return max(t_max * 0.99, 1e-8)

def pair_move(h, i, j, sign, n_samples=30):
    n = len(h)
    d = np.zeros(n)
    d[i] = sign
    d[j] = -sign
    if sign > 0:
        t_max = min(1.0 - h[i], h[j])
    else:
        t_max = min(h[i], 1.0 - h[j])
    if t_max < 1e-12:
        return h, exact_score(h)
    best_t = 0.0
    best_s = exact_score(h)
    ts = np.linspace(0, t_max, n_samples)
    for t in ts:
        h_try = h.copy()
        h_try[i] += sign * t
        h_try[j] -= sign * t
        s = exact_score(h_try)
        if s < best_s:
            best_s = s
            best_t = t
    if best_t > 0:
        lo = max(0.0, best_t * 0.7)
        hi = min(t_max, best_t * 1.3)
        for t in np.linspace(lo, hi, 20):
            h_try = h.copy()
            h_try[i] += sign * t
            h_try[j] -= sign * t
            s = exact_score(h_try)
            if s < best_s:
                best_s = s
                best_t = t
        h_new = h.copy()
        h_new[i] += sign * best_t
        h_new[j] -= sign * best_t
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = repair_feasible(h_new)
        return h_new, best_s
    return h, best_s

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = np.linspace(0, 2, n + 1)
    xt = np.linspace(0, 2, new_n + 1)
    h_interp = np.zeros(new_n)
    for k in range(new_n):
        lo_k, hi_k = xt[k], xt[k + 1]
        total_mass = 0.0
        total_width = 0.0
        for j in range(n):
            lo_j, hi_j = xs[j], xs[j + 1]
            overlap = min(hi_k, hi_j) - max(lo_k, lo_j)
            if overlap > 0:
                total_mass += h[j] * overlap
                total_width += overlap
        h_interp[k] = total_mass / total_width if total_width > 1e-15 else 0.5
    h_interp = np.clip(h_interp, 0.0, 1.0)
    return repair_feasible(h_interp)

def lp_descent(h, max_iters=200, D_values=None, active_frac=0.15):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    c = exact_score(h)
    if D_values is None:
        D_values = [0.3, 0.5, 0.8, 1.2]
    best_c = c
    best_h = h.copy()
    stall = 0
    for it in range(max_iters):
        profile = exact_profile(h)
        c = float(np.max(profile))
        if c < best_c - 1e-14:
            best_c = c
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 50:
                break
        G = build_gradient_matrix(h, n, dx)
        c_max = float(np.max(profile))
        margin = max(0.001, active_frac * (c_max - float(np.min(profile))))
        active = np.where(profile >= c_max - margin)[0]
        found_improve = False
        for D in D_values:
            if found_improve:
                break
            d_dir, t_pred = solve_lp_direction(G, profile, D)
            if d_dir is None:
                continue
            d_dir = d_dir - d_dir.mean()
            dnrm = np.linalg.norm(d_dir)
            if dnrm < 1e-12:
                continue
            d_dir /= dnrm * D
            t_max = safe_t_max(h, d_dir)
            h_new, s_new = line_search_1d(h, d_dir, t_max, n_samples=35)
            if s_new < best_c - 1e-12:
                h = h_new
                best_c = s_new
                best_h = h.copy()
                found_improve = True
        if not found_improve:
            d_dir, t_pred = solve_lp_active(G, profile, active, D_values[0])
            if d_dir is not None:
                d_dir = d_dir - d_dir.mean()
                dnrm = np.linalg.norm(d_dir)
                if dnrm > 1e-12:
                    d_dir /= dnrm * D_values[0]
                    t_max = safe_t_max(h, d_dir)
                    h_new, s_new = line_search_1d(h, d_dir, t_max, n_samples=30)
                    if s_new < best_c - 1e-12:
                        h = h_new
                        best_c = s_new
                        best_h = h.copy()
        if not found_improve:
            profile = exact_profile(h)
            m_star = int(np.argmax(profile))
            g_active = G[m_star, :]
            sorted_idx = np.argsort(g_active)
            for trial in range(6):
                i = sorted_idx[trial]
                j = sorted_idx[-1 - trial]
                if i == j:
                    continue
                if g_active[i] >= g_active[j]:
                    continue
                h_new, s_new = pair_move(h, i, j, 1.0, n_samples=25)
                if s_new < best_c - 1e-12:
                    h = h_new
                    best_c = s_new
                    best_h = h.copy()
        if best_c <= 0.38080:
            break
    return best_h, best_c

def perturb_restart(h, rng, sigma=0.02):
    n = len(h)
    z = rng.randn(n)
    z -= z.mean()
    h_new = h + sigma * z
    return repair_feasible(h_new)

def try_resolutions(h, dims, budget_check):
    best_h = h.copy()
    best_c = exact_score(h)
    for new_n in dims:
        if budget_check() < 30:
            break
        if new_n == len(h) or new_n < 24 or new_n > 140 or new_n % 2 != 0:
            continue
        h_interp = interpolate_to_n(h, new_n)
        h_opt, c_opt = lp_descent(h_interp, max_iters=80, D_values=[0.3, 0.5])
        if c_opt < best_c - 1e-12:
            best_h = h_opt.copy()
            best_c = c_opt
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s - (time.perf_counter() - t_start)

    def check_budget(threshold=20):
        return time_left() > threshold

    parent = None
    try:
        parent = np.array(initial_h_values, dtype=np.float64)
    except NameError:
        pass
    except Exception:
        pass

    if parent is None or len(parent) < 4:
        parent = np.full(70, 0.5)

    parent = np.clip(parent, 0.0, 1.0)
    parent = repair_feasible(parent)
    best_h = parent.copy()
    best_c = exact_score(parent)

    # Try symmetric/antisymmetric transforms
    h_sym = repair_feasible((parent + parent[::-1]) / 2.0)
    c_sym = exact_score(h_sym)
    if c_sym < best_c - 1e-12:
        best_h = h_sym
        best_c = c_sym

    n_cur = len(best_h)

    # Main LP descent
    if check_budget(150):
        best_h, best_c = lp_descent(best_h, max_iters=200, D_values=[0.2, 0.4, 0.6, 1.0])

    # Pair move refinement
    if check_budget(100):
        n = len(best_h)
        dx = 2.0 / n
        G = build_gradient_matrix(best_h, n, dx)
        profile = exact_profile(best_h)
        c_max = float(np.max(profile))
        active = np.where(profile >= c_max - 0.005)[0]
        for m_idx in active[:5]:
            g_active = G[m_idx, :]
            sorted_idx = np.argsort(g_active)
            for trial in range(10):
                if not check_budget(5):
                    break
                i = sorted_idx[trial]
                j = sorted_idx[-1 - trial]
                if i == j:
                    continue
                if g_active[i] >= g_active[j]:
                    continue
                h_new, s_new = pair_move(best_h, i, j, 1.0, n_samples=30)
                if s_new < best_c - 1e-13:
                    best_h = h_new
                    best_c = s_new

    # More LP iterations with different D
    if check_budget(100):
        for D_set in [[0.1, 0.2, 0.3], [0.5, 0.8, 1.5], [0.2, 0.5, 0.9]]:
            if not check_budget(30):
                break
            h_new, c_new = lp_descent(best_h, max_iters=60, D_values=D_set, active_frac=0.05)
            if c_new < best_c - 1e-12:
                best_h = h_new
                best_c = c_new

    # Resolution exploration
    if check_budget(80):
        n_cur = len(best_h)
        dims = set()
        for dn in [-6, -4, -3, -2, -1, 1, 2, 3, 4, 6]:
            d = n_cur + dn
            if d >= 24 and d <= 140 and d % 2 == 0:
                dims.add(d)
        for factor in [0.8, 0.9, 1.1, 1.2]:
            d = int(round(n_cur * factor / 2)) * 2
            d = max(24, min(d, 140))
            dims.add(d)
        dims.discard(n_cur)
        dims_list = sorted(dims, key=lambda x: abs(x - n_cur))
        for new_n in dims_list[:5]:
            if not check_budget(30):
                break
            h_interp = interpolate_to_n(best_h, new_n)
            h_opt, c_opt = lp_descent(h_interp, max_iters=60, D_values=[0.3, 0.5, 0.8])
            if c_opt < best_c - 1e-12:
                best_h = h_opt
                best_c = c_opt

    # Perturbation + restart
    if check_budget(120):
        for sigma in [0.01, 0.015, 0.025, 0.04, 0.06]:
            if not check_budget(25):
                break
            h_pert = perturb_restart(best_h, rng, sigma=sigma)
            h_opt, c_opt = lp_descent(h_pert, max_iters=80, D_values=[0.3, 0.5, 0.8])
            if c_opt < best_c - 1e-12:
                best_h = h_opt
                best_c = c_opt

    # Random direction kicks
    if check_budget(60):
        n = len(best_h)
        for trial in range(15):
            if not check_budget(5):
                break
            z = rng.randn(n)
            z -= z.mean()
            z /= (np.linalg.norm(z) + 1e-15) * 0.5
            t_max = safe_t_max(best_h, z)
            h_new, s_new = line_search_1d(best_h, z, t_max, n_samples=25)
            if s_new < best_c - 1e-12:
                best_h = h_new
                best_c = s_new
                h_opt, c_opt = lp_descent(best_h, max_iters=40, D_values=[0.3, 0.5])
                if c_opt < best_c - 1e-12:
                    best_h = h_opt
                    best_c = c_opt

    # Final focused LP descent
    if check_budget(40):
        for D_set in [[0.1, 0.15, 0.2], [0.05, 0.1, 0.25], [0.3, 0.4, 0.6]]:
            if not check_budget(15):
                break
            h_new, c_new = lp_descent(best_h, max_iters=100, D_values=D_set, active_frac=0.03)
            if c_new < best_c - 1e-14:
                best_h = h_new
                best_c = c_new

    # Final pair move polish
    if check_budget(20):
        n = len(best_h)
        dx = 2.0 / n
        G = build_gradient_matrix(best_h, n, dx)
        profile = exact_profile(best_h)
        c_max = float(np.max(profile))
        active = np.where(profile >= c_max - 0.003)[0]
        for m_idx in active[:8]:
            if not check_budget(3):
                break
            g_active = G[m_idx, :]
            sorted_idx = np.argsort(g_active)
            for trial in range(8):
                i = sorted_idx[trial]
                j = sorted_idx[-1 - trial]
                if i == j:
                    continue
                if g_active[i] >= g_active[j]:
                    continue
                h_new, s_new = pair_move(best_h, i, j, 1.0, n_samples=20)
                if s_new < best_c - 1e-14:
                    best_h = h_new
                    best_c = s_new

    # Final validation
    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)

    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            final_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 2:
            final_c = float(eval_result[0])
    except Exception:
        pass

    return best_h, final_c, len(best_h)
