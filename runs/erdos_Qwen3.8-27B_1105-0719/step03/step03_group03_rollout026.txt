    n = len(h)
    dx = 2.0 / n

    P_max = np.max(P)
    z = (P - P_max) / eta

            idx = int(np.argmax(x))

    cons = [
        h_var >= 0,
        h_var <= 1,
        cp.sum(h_var) == target,
        h_var >= h_curr - trust,
        h_var <= h_curr + trust,
        G_a @ h_var - t_var <= -b_a,
    ]
    obj = cp.Minimize(t_var + rho * cp.norm1(h_var - h_curr))
    prob = cp.Problem(obj, cons)
    h_sol = None

def interp_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    fx = np.concatenate([[0.0], xs, [2.0]])
    fh = np.concatenate([[h[0]], h, [h[-1]]])
    hn = np.interp(xt, fx, fh)
    return project_feasible(np.clip(hn, 0, 1), new_n)

        g[-1] = 1.0
        return g
    
    def eq_fun(x):
        return x[:-1].sum() - n / 2.0
    
    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g
    
    def ineq_fun(x):
        h = x[:-1]
        return x[-1] - np.correlate(h, 1 - h, mode="full") * dx
    
    def ineq_jac(x):
        h = x[:-1]
        G = build_grad_matrix(h, n, dx)
        J = np.zeros((M, n + 1))
        J[:, :-1] = -G
        J[:, -1] = 1.0
        return J
    
    t0 = exact_score(h0) + 1e-4
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, t0 + 2.0)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac},
    ]
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-13})
        if res.x is not None and np.isfinite(res.x).all():
            hs = project_feasible(res.x[:-1], n)
            sc = exact_score(hs)
            if np.isfinite(sc):
                return hs
    except Exception:
        pass
    return None


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()
    rng = np.random.RandomState(seed)
    
    def rem():
        return budget_s + t_start - time.perf_counter()
    
    # Load parent
    parent = None
    try:
        parent = np.array(initial_h_values, dtype=np.float64)
    except (NameError, Exception):
        pass
    if parent is None or len(parent) < 4:
        parent = np.full(70, 0.5)
    
    n_par = len(parent)
    parent = project_feasible(np.clip(parent, 0.0, 1.0), n_par)
    best_h = parent.copy()
    best_c = exact_score(parent)
    
    # Symmetrize
    h_sym = project_feasible((parent + parent[::-1]) / 2.0, n_par)
    c_sym = exact_score(h_sym)
    if c_sym < best_c - 1e-12:
        best_h, best_c = h_sym, c_sym
    
    # Phase 1: LSE optimization (main improvement engine)
    eta_schedule = [0.08, 0.04, 0.02, 0.012, 0.007, 0.004, 0.0025, 0.0015, 0.001, 0.0006]
    if rem() > 100:
        h_lse, c_lse = lse_optimize(best_h, eta_schedule, max_iters_per_eta=200, time_fn=rem)
        if c_lse < best_c - 1e-14:
            best_h, best_c = h_lse, c_lse
    
    # Phase 2: Active-set LP refinement
    if rem() > 150:
        h_as, c_as = active_set_loop(best_h, max_iters=300, time_fn=rem)
        if c_as < best_c - 1e-14:
            best_h, best_c = h_as, c_as
    
    # Phase 3: Alternate LSE and active-set
    for rnd in range(3):
        if rem() < 60:
            break
        h_lse2, c_lse2 = lse_optimize(best_h, [0.003, 0.002, 0.001, 0.0005], 
                                       max_iters_per_eta=150, time_fn=rem)
        if c_lse2 < best_c - 1e-14:
            best_h, best_c = h_lse2, c_lse2
        if rem() < 40:
            break
        h_as2, c_as2 = active_set_loop(best_h, max_iters=100, time_fn=rem)
        if c_as2 < best_c - 1e-14:
            best_h, best_c = h_as2, c_as2
    
    # Phase 4: Multi-resolution exploration
    if rem() > 120:
        n0 = len(best_h)
        dims = []
        for d in [n0 - 2, n0 + 2, n0 - 4, n0 + 4, n0 - 6, n0 + 6, n0 + 8, n0 - 8, n0 + 10, n0 + 12]:
            if d >= 20 and d <= 200 and d != n0:
                dims.append(d)
        dims = sorted(set(dims), key=lambda d: abs(d - n0))
        
        for nd in dims[:8]:
            if rem() < 50:
                break
            hi = interp_to_n(best_h, nd)
            ci = exact_score(hi)
            if ci < best_c + 0.005:
                h_lse3, c_lse3 = lse_optimize(hi, [0.02, 0.01, 0.005, 0.002, 0.001],
                                              max_iters_per_eta=100, time_fn=rem)
                if c_lse3 < best_c - 1e-14:
                    best_h, best_c = h_lse3, c_lse3
                if rem() < 30:
                    break
                h_as3, c_as3 = active_set_loop(best_h, max_iters=80, time_fn=rem)
                if c_as3 < best_c - 1e-14:
                    best_h, best_c = h_as3, c_as3
    
    # Phase 5: Random perturbation restarts
    if rem() > 100:
        n_cur = len(best_h)
        for trial in range(12):
            if rem() < 30:
                break
            eps = rng.choice([0.02, 0.04, 0.06, 0.1, 0.15])
            z = rng.randn(n_cur)
            z -= z.mean()
            z /= (np.linalg.norm(z) + 1e-15)
            hp = project_feasible(best_h + eps * z, n_cur)
            cp_ = exact_score(hp)
            if cp_ < best_c + 0.01:
                h_r, c_r = lse_optimize(hp, [0.03, 0.015, 0.008, 0.004, 0.002],
                                        max_iters_per_eta=80, time_fn=rem)
                if c_r < best_c - 1e-14:
                    best_h, best_c = h_r, c_r
    
    # Phase 6: Binary projection exploration
    if rem() > 60:
        n_cur = len(best_h)
        h_bin = binary_project(best_h, n_cur)
        c_bin = exact_score(h_bin)
        if c_bin < best_c + 0.002:
            h_lse4, c_lse4 = lse_optimize(h_bin, [0.05, 0.02, 0.01, 0.005, 0.002, 0.001],
                                          max_iters_per_eta=100, time_fn=rem)
            if c_lse4 < best_c - 1e-14:
                best_h, best_c = h_lse4, c_lse4
    
    # Phase 7: SLSQP polish
    if rem() > 80 and len(best_h) <= 150:
        hs = slsqp_polish(best_h, maxiter=300, time_fn=rem)
        if hs is not None:
            cs = exact_score(hs)
            if cs < best_c - 1e-14:
                best_h, best_c = hs, cs
    
    # Phase 8: Final LSE + active-set refinement
    if rem() > 40:
        h_lse5, c_lse5 = lse_optimize(best_h, [0.004, 0.002, 0.001, 0.0005],
                                       max_iters_per_eta=200, time_fn=rem)
        if c_lse5 < best_c - 1e-14:
            best_h, best_c = h_lse5, c_lse5
    if rem() > 25:
        h_as5, c_as5 = active_set_loop(best_h, max_iters=150, time_fn=rem)
        if c_as5 < best_c - 1e-14:
            best_h, best_c = h_as5, c_as5
    
    # Phase 9: Final SLSQP
    if rem() > 30 and len(best_h) <= 120:
        hs2 = slsqp_polish(best_h, maxiter=200, time_fn=rem)
        if hs2 is not None:
            cs2 = exact_score(hs2)
            if cs2 < best_c - 1e-14:
                best_h, best_c = hs2, cs2
    
    # Final validation
    best_h = project_feasible(best_h, len(best_h))
    final_c = exact_score(best_h)
    
    try:
        er = evaluate_erdos_solution(best_h)
        if isinstance(er, (float, int)):
            final_c = min(final_c, float(er))
        elif isinstance(er, tuple) and len(er) >= 2:
            final_c = min(final_c, float(er[0]))
    except Exception:
        pass
    
    return best_h, final_c, len(best_h)
