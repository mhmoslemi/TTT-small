            if time.perf_counter() > deadline - 15:
                break
            h_struct = make_structured(n_b, rng, mode)
            h_blend = repair(0.6 * best_h + 0.4 * h_struct)
            c_blend = score(h_blend)
            if c_blend < best_c + 0.02:
                h_opt4, c_opt4 = optimize_pairs(h_blend, rng, deadline, max_stagnation=150)
                if c_opt4 < best_c - 1e-14:
                    best_h = h_opt4
                    best_c = c_opt4

    # Phase 6: Final pair optimization round
    if time.perf_counter() < deadline - 10:
        h_final, c_final = optimize_pairs(best_h, rng, deadline, max_stagnation=200)
        if c_final < best_c - 1e-14:
            best_h = h_final
            best_c = c_final

    # Phase 7: Final SLSQP
    if time.perf_counter() < deadline - 5:
        h_f, c_f = slsqp_polish(best_h, deadline)
        if c_f < best_c - 1e-14:
            best_h = h_f
            best_c = c_f

    # Final validation
    best_h = repair(best_h)
    final_c = score(best_h)

    try:
        ev = evaluate_erdos_solution(best_h)
        if isinstance(ev, (float, int)):
            final_c = float(ev)
        elif isinstance(ev, tuple) and len(ev) >= 2:
            final_c = float(ev[0])
    except Exception:
        pass

    if final_c > best_c + 1e-10:
        final_c = best_c

    return best_h, final_c, len(best_h)
