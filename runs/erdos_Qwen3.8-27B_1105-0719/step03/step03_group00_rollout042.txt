import numpy as np
import time

def project_box_sum(x, target):
    n = len(x)
    x = np.nan_to_num(x, nan=0.5)
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(x - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if mask.any():
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def c5_eval(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full').max() * (2.0 / n)

def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def soft_max_weights(corr, alpha):
    scaled = corr / alpha
    m = scaled.max()
    p = np.exp(scaled - m)
    s = p.sum()
    return p / s if s > 1e-300 else np.full(len(corr), 1.0/len(corr))

def weighted_grad(h, w, dx, n):
    grad = np.zeros(n)
    w0 = w[n-1]
    if w0 > 1e-18:
        grad += w0 * (1.0 - 2.0 * h)
    for d in range(1, n):
        wp = w[n-1+d]
        wn = w[n-1-d]
        if wp + wn < 1e-18:
            continue
        rem = n - d
        grad[:rem] += (wn - wp) * h[d:] + wn
        grad[d:] += wp - (wp + wn) * h[:rem]
    return dx * grad

def adam_softmax_optimize(h, dx, target, n, alpha_schedule, steps_per_alpha, lr, t0, budget_s):
    """Optimize using soft-max weights with Adam and continuation on alpha."""
    best_h = h.copy()
    best_c5 = c5_eval(h)
    
    m = np.zeros(n)  # first moment
    v = np.zeros(n)  # second moment
    t = 0
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    
    for alpha in alpha_schedule:
        for step in range(steps_per_alpha):
            if budget_s - (time.time() - t0) < 5.0:
                return best_h, best_c5
            
            corr = compute_corrs(h)
            w = soft_max_weights(corr, alpha)
            grad = weighted_grad(h, w, dx, n)
            
            # Remove mean for sum constraint
            grad -= grad.mean()
            
            t += 1
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad**2
            m_hat = m / (1 - beta1**t)
            v_hat = v / (1 - beta2**t)
            
            h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
            h_new = project_box_sum(h_new, target)
            
            c5_new = c5_eval(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            h = h_new
    
    return best_h, best_c5

def coord_grad(h, i, n):
    g = np.zeros(2*n - 1)
