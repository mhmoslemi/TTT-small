- Sort c: c_(1) <= c_(2) <= ... <= c_(n)
- If s is integer: v_i = 1 for the s positions with smallest c, 0 otherwise
- If s is not integer: partial fill at the boundary

1. Compute gradient G of the smoothed max (or active set gradient)
2. Solve LMO: v = binary vector with n/2 ones at positions of smallest G
3. Line search on (1-γ)h + γv for γ in [0, 1]
4. Accept if improves

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.special import logsumexp
import time

def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - 0.5*(lo + hi), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def c5_of_h(h, dx):
    return compute_corrs(h).max() * dx

def grad_corr_k(h, k, n):
    offset = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -offset)
    i_hi = min(n - 1, n - 1 - offset)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx + offset] += (1.0 - h[idx])
    grad[idx] += (-h[idx + offset])
    return grad

def smooth_max_obj(h, dx, beta):
    corrs = compute_corrs(h)
    return logsumexp(beta * corrs) / beta * dx

def smooth_max_grad(h, dx, beta):
    n = len(h)
    corrs = compute_corrs(h)
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones_like(corrs) / len(corrs)
    else:
        w /= w_sum
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] > 1e-13:
            grad += w[k] * grad_corr_k(h, k, n)
    return grad * dx

def lmo_vertex(G, n, target_sum):
    """Solve min G^T v s.t. 0<=v<=1, sum v = target_sum.
    For integer target_sum, the vertex solution is: set v=1 at positions of smallest G."""
    v = np.zeros(n)
    k_select = int(round(target_sum))
    # Select k_select positions with smallest G
    idx = np.argsort(G)[:k_select]
    v[idx] = 1.0
    return v

def lmo_line_search(h, v, dx, target_sum, max_iter=100):
    """Line search on (1-gamma)*h + gamma*v"""
    n = len(h)
    best_gamma = 0.0
    best_c5 = c5_of_h(h, dx)
    a, b = 0.0, 1.0
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c_pt = b - gr * (b - a)
    d_pt = a + gr * (b - a)
    
    def f(gamma):
        h_new = (1 - gamma) * h + gamma * v
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_feasible(h_new, target_sum)
        return c5_of_h(h_new, dx)
    
    fc, fd = f(c_pt), f(d_pt)
    for _ in range(max_iter):
        if b - a < 1e-8:
            break
        if fc < fd:
            b = d_pt
            d_pt = c_pt
            fd = fc
            c_pt = b - gr * (b - a)
            fc = f(c_pt)
        else:
            a = c_pt
            c_pt = d_pt
            fc = fd
            d_pt = a + gr * (b - a)
            fd = f(d_pt)
    
    best_gamma = 0.5 * (a + b)
    h_new = (1 - best_gamma) * h + best_gamma * v
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = project_feasible(h_new, target_sum)
    new_c5 = c5_of_h(h_new, dx)
    return h_new, new_c5

def epigraph_slsqp(h0, dx, target_sum, n_active=64, maxiter=2000):
    """Epigraph formulation: min tau s.t. tau >= dx*corrs(k) for active k, sum h = target_sum, 0<=h<=1"""
    n = len(h0)
    corrs = compute_corrs(h0)
    # Select active set: top n_active correlations
    active_k = np.argsort(corrs)[-n_active:]
    
    def obj(z):
        return z[-1]
    
    def obj_grad(z):
        g = np.zeros(len(z))
        g[-1] = 1.0
        return g
    
    cons = []
    for k in active_k:
        def make_con(k):
            def con_fun(z):
                h = z[:n]
                return z[-1] - dx * compute_corrs(h)[k]
            def con_jac(z):
                h = z[:n]
                j = np.zeros(len(z))
                j[-1] = 1.0
                j[:n] = -dx * grad_corr_k(h, k, n)
                return j
            return {'type': 'ineq', 'fun': con_fun, 'jac': con_jac}
        cons.append(make_con(k))
    
    def eq_fun(z):
        return z[:n].sum() - target_sum
    def eq_jac(z):
        j = np.zeros(len(z))
        j[:n] = 1.0
        return j
    cons.append({'type': 'eq', 'fun': eq_fun, 'jac': eq_jac})
    
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    z0 = np.append(h0, c5_of_h(h0, dx))
    
    try:
        res = minimize(obj, z0, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        h_out = project_feasible(res.x[:n], target_sum)
    except Exception:
        h_out = project_feasible(h0, target_sum)
    return h_out, c5_of_h(h_out, dx)

def iterative_epigraph(h0, dx, target_sum, max_rounds=20, n_active=64):
    """Iteratively solve epigraph with changing active set"""
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    
    for rnd in range(max_rounds):
        h_new, c5_new = epigraph_slsqp(h, dx, target_sum, n_active, 1500)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
        else:
            # Try with different active set size
            h_new2, c5_new2 = epigraph_slsqp(h, dx, target_sum, min(128, 2*n-1), 1000)
            if c5_new2 < best_c5 - 1e-15:
                best_c5 = c5_new2
                best_h = h_new2.copy()
                h = h_new2
            else:
                break
    return best_h, best_c5

def lmo_frank_wolfe(h0, dx, target_sum, max_iter=200, beta=200):
    """Frank-Wolfe style: LMO + line search"""
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    no_improve = 0
    
    for it in range(max_iter):
        corrs = compute_corrs(h)
        c5_val = corrs.max() * dx
        
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 50:
            # Try with higher beta
            beta *= 2
            no_improve = 0
            if beta > 10000:
                break
        
        # Compute smooth gradient
        G = smooth_max_grad(h, dx, beta)
        
        # LMO: find vertex minimizing G^T v
        v = lmo_vertex(G, n, target_sum)
        
        # Line search
        h_new, c5_new = lmo_line_search(h, v, dx, target_sum)
        
        if c5_new < c5_val - 1e-14:
            h = h_new
        else:
            # Try partial step
            for frac in [0.5, 0.25, 0.1]:
                h_try = (1 - frac) * h + frac * v
                h_try = np.clip(h_try, 0.0, 1.0)
                h_try = project_feasible(h_try, target_sum)
                c5_try = c5_of_h(h_try, dx)
                if c5_try < c5_val - 1e-14:
                    h = h_try
                    break
            else:
                break
    
    return best_h, best_c5

def iterative_lp_polish(h0, dx, target_sum, max_iters=200, init_step=0.15):

        except Exception:
            break
        if res.status != 0:
            break

        delta = res.x[:n]
        if np.max(np.abs(delta)) < 1e-14:

        h_new = project_feasible(h + delta, target_sum)
        new_c5 = c5_of_h(h_new, dx)

def pair_transfer_polish(h0, dx, target_sum, n_top=25, n_iter=3):

        w = np.exp(beta * shifted)

        else:
            w /= w_sum
        grad = np.zeros(n)
        for k in range(2 * n - 1):

                grad += w[k] * grad_corr_k(h, k, n)

        dec_idx = np.argsort(grad)[-n_top:]
        inc_idx = np.argsort(grad)[:n_top]

        improved = False
        for i in dec_idx:
            for j in inc_idx:
                if i == j:
                    continue
                eps_lo = max(0.0, -h[j])
                eps_hi = min(h[i], 1.0 - h[j])
                if eps_hi - eps_lo < 1e-12:

                d = np.zeros(2 * n - 1)
                for k in range(2 * n - 1):
                    gk = grad_corr_k(h, k, n)
                    d[k] = gk[j] - gk[i]

                # Minimize max(corrs + eps*d) over eps
                a, b = eps_lo, eps_hi
                gr = (np.sqrt(5.0) - 1.0) / 2.0
                c_pt = b - gr * (b - a)
                d_pt = a + gr * (b - a)
                
                def f_ps(eps):

                        d_pt = c_pt
                        fd = fc
                        c_pt = b - gr * (b - a)
                        fc = f_ps(c_pt)
                    else:
                        a = c_pt
                        c_pt = d_pt
                        fc = fd
                        d_pt = a + gr * (b - a)
                        fd = f_ps(d_pt)

                eps_opt = 0.5 * (a + b)
                new_c5_rough = (corrs + eps_opt * d).max() * dx

    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)


def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def c5_of_h(h, dx):
    return compute_corrs(h).max() * dx


def grad_corr_k(h, k, n):
    offset = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -offset)
    i_hi = min(n - 1, n - 1 - offset)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx + offset] += (1.0 - h[idx])
    grad[idx] += (-h[idx + offset])
    return grad


def smooth_max_obj(h, dx, beta):
    corrs = compute_corrs(h)
    return logsumexp(beta * corrs) / beta * dx


def smooth_max_grad(h, dx, beta):
    n = len(h)
    corrs = compute_corrs(h)
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones_like(corrs) / len(corrs)
    else:
        w /= w_sum
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] > 1e-13:
            grad += w[k] * grad_corr_k(h, k, n)
    return grad * dx


class SumFun:
    def __init__(self, n, target_sum):
        self.n = n
        self.target_sum = target_sum
    def __call__(self, x):
        return x.sum() - self.target_sum


class SumJac:
    def __init__(self, n):
        self.n = n
    def __call__(self, x):
        return np.ones(self.n)


class EpiIneqFun:
    def __init__(self, active_k, n, dx):
        self.active_k = active_k
        self.n = n
        self.dx = dx
    def __call__(self, z):
        h = z[:self.n]
        corrs = compute_corrs(h)
        return z[-1] - self.dx * corrs[self.active_k]


class EpiIneqJac:
    def __init__(self, active_k, n, dx):
        self.active_k = active_k
        self.n = n
        self.dx = dx
    def __call__(self, z):
        h = z[:self.n]
        m = len(self.active_k)
        mvar = self.n + 1
        J = np.zeros((m, mvar))
        J[:, -1] = 1.0
        for row, k in enumerate(self.active_k):
            J[row, :self.n] = -self.dx * grad_corr_k(h, k, self.n)
        return J


def epi_obj(z):
    return z[-1]


def epi_obj_grad(z):
    g = np.zeros(len(z))
    g[-1] = 1.0
    return g


def epigraph_solve(h0, dx, target_sum, n_active=64, maxiter=2000):
    n = len(h0)
    corrs = compute_corrs(h0)
    active_k = np.argsort(corrs)[-n_active:]

    e_fun = EpiIneqFun(active_k, n, dx)
    e_jac = EpiIneqJac(active_k, n, dx)

    ineq_con = NonlinearConstraint(e_fun, 0.0, np.inf, jac=e_jac)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    eq_con = LinearConstraint(A_eq, target_sum, target_sum)

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    z0 = np.append(h0, c5_of_h(h0, dx))

    try:
        res = minimize(epi_obj, z0, jac=epi_obj_grad, method='trust-constr',
                       constraints=[ineq_con, eq_con], bounds=bounds,
                       options={'maxiter': maxiter, 'gtol': 1e-10})
        h_out = project_feasible(res.x[:n], target_sum)
    except Exception:
        h_out = project_feasible(h0, target_sum)
    return h_out, c5_of_h(h_out, dx)


def iterative_epigraph(h0, dx, target_sum, max_rounds=15, n_active=64):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)

    for rnd in range(max_rounds):
        for na in [n_active, min(2 * n_active, 2 * n - 1)]:
            h_new, c5_new = epigraph_solve(h, dx, target_sum, na, 1500)
            if c5_new < best_c5 - 1e-15:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new
    return best_h, best_c5


def lmo_vertex(G, n, target_sum):
    v = np.zeros(n)
    k_select = int(round(target_sum))
    idx = np.argsort(G)[:k_select]
    v[idx] = 1.0
    return v


def lmo_line_search_val(gamma, h, v, dx, target_sum):
    h_new = (1.0 - gamma) * h + gamma * v
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = project_feasible(h_new, target_sum)
    return c5_of_h(h_new, dx)


def lmo_line_search(h, v, dx, target_sum):
    n = len(h)
    best_c5 = c5_of_h(h, dx)
    a, b = 0.0, 1.0
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c_pt = b - gr * (b - a)
    d_pt = a + gr * (b - a)
    fc = lmo_line_search_val(c_pt, h, v, dx, target_sum)
    fd = lmo_line_search_val(d_pt, h, v, dx, target_sum)
    for _ in range(60):
        if b - a < 1e-8:
            break
        if fc < fd:
            b = d_pt
            d_pt = c_pt
            fd = fc
            c_pt = b - gr * (b - a)
            fc = lmo_line_search_val(c_pt, h, v, dx, target_sum)
        else:
            a = c_pt
            c_pt = d_pt
            fc = fd
            d_pt = a + gr * (b - a)
            fd = lmo_line_search_val(d_pt, h, v, dx, target_sum)
    best_gamma = 0.5 * (a + b)
    h_new = (1.0 - best_gamma) * h + best_gamma * v
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = project_feasible(h_new, target_sum)
    new_c5 = c5_of_h(h_new, dx)
    return h_new, new_c5


def lmo_frank_wolfe(h0, dx, target_sum, max_iter=300, beta=200):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    no_improve = 0

    for it in range(max_iter):
        corrs = compute_corrs(h)
        c5_val = corrs.max() * dx

        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 80:
            beta = min(beta * 2, 20000)
            no_improve = 0
            if beta >= 20000:
                break

        G = smooth_max_grad(h, dx, beta)
        v = lmo_vertex(G, n, target_sum)
        h_new, c5_new = lmo_line_search(h, v, dx, target_sum)

        if c5_new < c5_val - 1e-14:
            h = h_new
        else:
            improved = False
            for frac in [0.5, 0.25, 0.1, 0.05]:
                h_try = (1.0 - frac) * h + frac * v
                h_try = np.clip(h_try, 0.0, 1.0)
                h_try = project_feasible(h_try, target_sum)
                c5_try = c5_of_h(h_try, dx)
                if c5_try < c5_val - 1e-14:
                    h = h_try
                    improved = True
                    break
            if not improved:
                break

    return best_h, best_c5


def iterative_lp_polish(h0, dx, target_sum, max_iters=300, init_step=0.15):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    step = init_step
    stall_count = 0

    for it in range(max_iters):
        corrs = compute_corrs(h)
        c5_val = corrs.max() * dx
        max_c = corrs.max()

        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            stall_count = 0
        else:
            stall_count += 1
        if stall_count > 50:
            step *= 0.6
            stall_count = 0
            if step < 1e-7:
                break

        tol_frac = 0.05
        tol_val = max_c * (1.0 - tol_frac)
        active_mask = corrs >= tol_val
        active_k = np.where(active_mask)[0]
        if len(active_k) > 80:
            active_k = np.argsort(corrs)[-80:]

        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        A_rows = []
        b_rows = []
        for k in active_k:
            row = np.zeros(n + 1)
            row[:n] = dx * grad_corr_k(h, k, n)
            row[n] = -1.0
            A_rows.append(row)
            b_rows.append(-corrs[k] * dx - 1e-14)

        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)

        bounds = []
        for i in range(n):
            bounds.append((max(-h[i], -step), min(1.0 - h[i], step)))
        bounds.append((0.0, None))

        try:
            res = linprog(c_obj, A_ub=np.array(A_rows), b_ub=np.array(b_rows),
                          A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
        except Exception:
            break
        if res.status != 0:
            break

        delta = res.x[:n]
        if np.max(np.abs(delta)) < 1e-14:
            break

        h_new = project_feasible(h + delta, target_sum)
        new_c5 = c5_of_h(h_new, dx)

        if new_c5 < c5_val - 1e-14:
            h = h_new
            step = min(step * 1.15, init_step)
        else:
            improved = False
            for frac in [0.5, 0.25, 0.1, 0.05]:
                h_new = project_feasible(h + frac * delta, target_sum)
                new_c5 = c5_of_h(h_new, dx)
                if new_c5 < c5_val - 1e-14:
                    h = h_new
                    improved = True
                    break
            if not improved:
                step *= 0.5
                if step < 1e-8:
                    break

    return best_h, best_c5


def pair_transfer_polish(h0, dx, target_sum, n_top=25, n_iter=4):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)

    for iteration in range(n_iter):
        corrs = compute_corrs(h)
        beta = 800.0
        shifted = corrs - corrs.max()
        w = np.exp(beta * shifted)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones_like(corrs) / len(corrs)
        else:
            w /= w_sum
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] > 1e-12:
                grad += w[k] * grad_corr_k(h, k, n)
        grad *= dx

        dec_idx = np.argsort(grad)[-n_top:]
        inc_idx = np.argsort(grad)[:n_top]

        improved = False
        for i in dec_idx:
            for j in inc_idx:
                if i == j:
                    continue
                eps_lo = 0.0
                eps_hi = min(h[i], 1.0 - h[j])
                if eps_hi - eps_lo < 1e-12:
                    continue

                d = np.zeros(2 * n - 1)
                for k in range(2 * n - 1):
                    gk = grad_corr_k(h, k, n)
                    d[k] = gk[j] - gk[i]

                a, b = eps_lo, eps_hi
                gr = (np.sqrt(5.0) - 1.0) / 2.0
                c_pt = b - gr * (b - a)
                d_pt = a + gr * (b - a)

                fc = (corrs + c_pt * d).max()
                fd = (corrs + d_pt * d).max()
                for _ in range(45):
                    if b - a < 1e-14:
                        break
                    if fc < fd:
                        b = d_pt
                        d_pt = c_pt
                        fd = fc
                        c_pt = b - gr * (b - a)
                        fc = (corrs + c_pt * d).max()
                    else:
                        a = c_pt
                        c_pt = d_pt
                        fc = fd
                        d_pt = a + gr * (b - a)
                        fd = (corrs + d_pt * d).max()

                eps_opt = 0.5 * (a + b)
                new_c5_rough = (corrs + eps_opt * d).max() * dx
                if new_c5_rough < best_c5 - 1e-14:
                    new_h = h.copy()
                    new_h[i] -= eps_opt
                    new_h[j] += eps_opt
                    new_h = np.clip(new_h, 0.0, 1.0)
                    new_h = project_feasible(new_h, target_sum)
                    new_c5 = c5_of_h(new_h, dx)
                    if new_c5 < best_c5 - 1e-14:
                        h = new_h
                        best_c5 = new_c5
                        best_h = h.copy()
                        corrs = compute_corrs(h)
                        improved = True

        if not improved:
            break

    return best_h, best_c5


def slsqp_smoothed(h0, dx, target_sum, beta, maxiter):
    n = len(h0)
    sf = SumFun(n, target_sum)
    sj = SumJac(n)
    cons = [{'type': 'eq', 'fun': sf, 'jac': sj}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(smooth_max_obj, h0, jac=smooth_max_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       args=(dx, beta),
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        h_out = project_feasible(res.x, target_sum)
    except Exception:
        h_out = project_feasible(h0, target_sum)
    return h_out, c5_of_h(h_out, dx)


def full_optimize(h0, dx, target_sum, time_left_fn, label=""):
    n = len(h0)
    best_h = h0.copy()
    best_c5 = c5_of_h(h0, dx)

    for beta in [100, 300, 800, 2000, 5000]:
        if time_left_fn() < 20:
            break
        h_opt, c5_val = slsqp_smoothed(best_h, dx, target_sum, beta, 1500)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_opt

    if time_left_fn() > 35:
        h_ep, c5_ep = iterative_epigraph(best_h, dx, target_sum, max_rounds=8, n_active=64)
        if c5_ep < best_c5:
            best_c5 = c5_ep
            best_h = h_ep

    if time_left_fn() > 30:
        h_fw, c5_fw = lmo_frank_wolfe(best_h, dx, target_sum, max_iter=200, beta=300)
        if c5_fw < best_c5:
            best_c5 = c5_fw
            best_h = h_fw

    if time_left_fn() > 25:
        h_lp, c5_lp = iterative_lp_polish(best_h, dx, target_sum, max_iters=300, init_step=0.12)
        if c5_lp < best_c5:
            best_c5 = c5_lp
            best_h = h_lp

    if time_left_fn() > 15:
        h_pt, c5_pt = pair_transfer_polish(best_h, dx, target_sum, n_top=20, n_iter=3)
        if c5_pt < best_c5:
            best_c5 = c5_pt
            best_h = h_pt

    if time_left_fn() > 30:
        h_f, c5_f = slsqp_smoothed(best_h, dx, target_sum, 5000, 2000)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f
    if time_left_fn() > 25:
        h_ep2, c5_ep2 = iterative_epigraph(best_h, dx, target_sum, max_rounds=6, n_active=96)
        if c5_ep2 < best_c5:
            best_c5 = c5_ep2
            best_h = h_ep2
    if time_left_fn() > 20:
        h_lp2, c5_lp2 = iterative_lp_polish(best_h, dx, target_sum, max_iters=200, init_step=0.08)
        if c5_lp2 < best_c5:
            best_c5 = c5_lp2
            best_h = h_lp2
    if time_left_fn() > 12:
        h_pt2, c5_pt2 = pair_transfer_polish(best_h, dx, target_sum, n_top=15, n_iter=3)
        if c5_pt2 < best_c5:
            best_c5 = c5_pt2
            best_h = h_pt2

    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s - (time.time() - t0)

    best_h = None
    best_c5 = float('inf')

    h0 = np.array(initial_h_values, dtype=float)
    n = len(h0)
    dx = 2.0 / n
    target_sum = n / 2.0
    h0 = project_feasible(h0, target_sum)

    h_opt, c5_val = full_optimize(h0, dx, target_sum, time_left, "main")
    best_h = h_opt
    best_c5 = c5_val

    if time_left() > 50:
        for beta_val in [50, 200, 1000]:
            if time_left() < 35:
                break
            G = smooth_max_grad(best_h, dx, beta_val)
            v = lmo_vertex(G, n, target_sum)
            for mix in [0.3, 0.6]:
                if time_left() < 30:
                    break
                h_mix = (1.0 - mix) * best_h + mix * v
                h_mix = np.clip(h_mix, 0.0, 1.0)
                h_mix = project_feasible(h_mix, target_sum)
                h_opt2, c5_val2 = full_optimize(h_mix, dx, target_sum, time_left, "lmo")
                if c5_val2 < best_c5:
                    best_c5 = c5_val2
                    best_h = h_opt2

    for trial in range(10):
        if time_left() < 35:
            break
        if trial % 3 == 0:
            perturb = 0.04 * (rng.random(n) - 0.5)
        elif trial % 3 == 1:
            perturb = 0.12 * (rng.random(n) - 0.5)
        else:
            idx = rng.choice(n, size=max(5, n // 8), replace=False)
            perturb = np.zeros(n)
            perturb[idx] = rng.choice([-1, 1], size=len(idx)) * 0.08
        h_try = project_feasible(best_h + perturb, target_sum)
        h_opt3, c5_val3 = full_optimize(h_try, dx, target_sum, time_left, "perturb")
        if c5_val3 < best_c5:
            best_c5 = c5_val3
            best_h = h_opt3

    n_candidates = []
    for nn in range(100, 190):
        if nn % 2 == 0:
            n_candidates.append(nn)
    n_candidates.sort(key=lambda x: abs(x - len(best_h)))
    n_candidates = n_candidates[:12]

    for n_try in n_candidates:
        if time_left() < 40:
            break
        if n_try == len(best_h):
            continue
        dx_try = 2.0 / n_try
        ts_try = n_try / 2.0
        x_orig = np.linspace(0.0, 2.0, len(best_h), endpoint=False) + 1.0 / len(best_h)
        x_new = np.linspace(0.0, 2.0, n_try, endpoint=False) + 1.0 / n_try
        h_interp = np.interp(x_new, x_orig, best_h)
        h_interp = project_feasible(h_interp, ts_try)
        h_opt4, c5_val4 = full_optimize(h_interp, dx_try, ts_try, time_left, "n" + str(n_try))
        if c5_val4 < best_c5:
            best_c5 = c5_val4
            best_h = h_opt4

    if time_left() > 30 and best_h is not None:
        n_b = len(best_h)
        dx_b = 2.0 / n_b
        ts_b = n_b / 2.0
        h_final, c5_final = full_optimize(best_h, dx_b, ts_b, time_left, "final")
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final

    if time_left() > 15 and best_h is not None:
        n_b = len(best_h)
        dx_b = 2.0 / n_b
        ts_b = n_b / 2.0
        h_pt, c5_pt = pair_transfer_polish(best_h, dx_b, ts_b, n_top=35, n_iter=6)
        if c5_pt < best_c5:
            best_c5 = c5_pt
            best_h = h_pt
    if time_left() > 10 and best_h is not None:
        n_b = len(best_h)
        dx_b = 2.0 / n_b
        ts_b = n_b / 2.0
        h_lp3, c5_lp3 = iterative_lp_polish(best_h, dx_b, ts_b, max_iters=150, init_step=0.05)
        if c5_lp3 < best_c5:
            best_c5 = c5_lp3
            best_h = h_lp3

    if best_h is None:
        best_h = h0
        best_c5 = c5_of_h(h0, 2.0 / len(h0))
    else:
        n_final = len(best_h)
        best_h = project_feasible(best_h, n_final / 2.0)
        best_c5 = c5_of_h(best_h, 2.0 / n_final)

    return best_h, best_c5, len(best_h)
