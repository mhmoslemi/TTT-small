def compute_pairwise_grad(h):
    """Compute gradient of each overlap component with respect to h.
    Returns G where G[k, j] = d overlap_k / d h[j]."""
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    G = np.zeros((M, n))
    for s in range(-(n-1), n):
        k = s + (n - 1)
        for j in range(n):
            g = 0.0
            if s >= 0:
                if j <= n - 1 - s:
                    g += (1.0 - h[j + s])
                if j >= s:
                    g -= h[j - s]
            else:
                sp = -s
                if j >= sp:
                    g += (1.0 - h[j - sp])
                if j <= n - 1 - sp:
                    g -= h[j + sp]
            G[k, j] = g * dx
    return G


def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx] + 1e-15)
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx] + 1e-15)
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h


def interp_to(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    fx = np.concatenate([[0.0], xs, [2.0]])
    fh = np.concatenate([[h[0]], h, [h[-1]]])
    hn = np.interp(xt, fx, fh)
    return repair_feasible(np.clip(hn, 0, 1))


def subgradient_descent(h0, max_iters, budget_fn, learning_rate=0.01):
    """Subgradient descent with projection onto feasible set."""
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c = exact_score(h)
    lr = learning_rate
    
    for it in range(max_iters):
        if budget_fn() < 30:
            break
        profile, grad = compute_profile_and_grad(h)
        c_max = profile[np.argmax(profile)]
        
        # Also use weighted average of gradients of near-max components
        threshold = c_max - 0.005
        active = np.where(profile >= threshold)[0]
        if len(active) > 0:
            weights = profile[active] - threshold + 0.001
            weights = weights / weights.sum()
            # Average gradient
            G = compute_pairwise_grad(h)
            grad_avg = (weights[:, None] * G[active, :]).sum(axis=0)
        else:
            grad_avg = grad
        
        # Project gradient onto constraint: grad - mean(grad) ensures sum is preserved
        grad_avg = grad_avg - grad_avg.mean()
        
        # Line search
        best_h_try = h
        best_c_try = c_max
        for alpha in [lr, lr*0.5, lr*0.25, lr*0.1, lr*0.05]:
            h_try = repair_feasible(h - alpha * grad_avg)
            c_try = exact_score(h_try)
            if c_try < best_c - 1e-12:
                best_h_try = h_try
                best_c_try = c_try
                break

            h = best_h_try
            best_c = best_c_try
            best_h = h.copy()
            lr = min(lr * 1.1, 0.05)
        else:
            lr *= 0.6
            if lr < 1e-6:
                break
    
    return best_h, best_c

                    improved = True
                    break

                    h[i] -= delta

def slsqp_refine(h0, maxiter, tol):
    """SLSQP refinement for final polish."""
    n = len(h0)
    dx = 2.0 / n
    M = 2 * n - 1
    
    def obj(x):
        return x[-1]
    
    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    
    def eq_fun(x):
        return x[:-1].sum() - n / 2.0
    
    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g
    
    def ineq_fun(x):
        h = x[:-1]
        return x[-1] - np.correlate(h, 1 - h, mode="full") * dx
    
    def ineq_jac(x):
        h = x[:-1]
        G = compute_pairwise_grad(h)
        J = np.zeros((M, n + 1))
        J[:, :-1] = -G
        J[:, -1] = 1.0
        return J
    
    t0 = exact_score(h0) + 1e-4
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, t0 + 2.0)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': tol})
        if res.x is not None and np.isfinite(res.x).all():
            hs = repair_feasible(res.x[:-1])
            if np.isfinite(exact_score(hs)):
                return hs
    except Exception:
        pass
    return None

    """Compute gradient of overlap at shift s (where s is in [-(n-1), n-1])."""
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    if s >= 0:

            idx = order[i]
            add = min(remaining, capacity[idx] + 1e-15)
            if add > 0:
                h[idx] += add
            remaining -= add

```python
def compute_all_grads(h):
    """Compute gradient of all overlap components. Returns G of shape (2n-1, n)."""
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    G = np.zeros((M, n))
    
    for s in range(-(n-1), n):
        k = s + (n - 1)
        if s >= 0:
            # overlap_k = sum_{i=0}^{n-1-s} h[i]*(1-h[i+s]) * dx
            # d/dh[j]:
            # if j in [0, n-1-s]: += (1 - h[j+s]) * dx
            # if j in [s, n-1]: -= h[j-s] * dx
            # Combined:
            # j in [0, n-1-s]: += (1-h[j+s])
            # j in [s, n-1-s]: -= h[j-s] (both terms)
            # j in [n-s, n-1] (i.e., j > n-1-s): -= h[j-s]
            
            valid_first = np.arange(0, n - s)  # indices 0 to n-1-s
            G[k, valid_first] += dx * (1.0 - h[valid_first + s])
            valid_second = np.arange(s, n)  # indices s to n-1
            G[k, valid_second] -= dx * h[valid_second - s]
        else:
            sp = -s
            # overlap = sum_{i=sp}^{n-1} h[i]*(1-h[i-sp]) * dx
            # d/dh[j]:
            # if j in [sp, n-1]: += (1 - h[j-sp]) * dx
            # if j in [0, n-1-sp]: -= h[j+sp] * dx
            valid_first = np.arange(sp, n)
            G[k, valid_first] += dx * (1.0 - h[valid_first - sp])
            valid_second = np.arange(0, n - sp)
            G[k, valid_second] -= dx * h[valid_second + sp]
    
    return G
```

This is O(n^2) total but with vectorized inner operations, so it's fast.

import numpy as np
from scipy.optimize import minimize

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))


def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx


def compute_all_grads(h):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    G = np.zeros((M, n))
    for s in range(-(n - 1), n):
        k = s + (n - 1)
        if s >= 0:
            vf = np.arange(0, n - s)
            G[k, vf] += dx * (1.0 - h[vf + s])
            vs = np.arange(s, n)
            G[k, vs] -= dx * h[vs - s]
        else:
            sp = -s
            vf = np.arange(sp, n)
            G[k, vf] += dx * (1.0 - h[vf - sp])
            vs = np.arange(0, n - sp)
            G[k, vs] -= dx * h[vs + sp]
    return G


def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx] + 1e-15)
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx] + 1e-15)
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h


def interp_to(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    fx = np.concatenate([[0.0], xs, [2.0]])
    fh = np.concatenate([[h[0]], h, [h[-1]]])
    hn = np.interp(xt, fx, fh)
    return repair_feasible(np.clip(hn, 0, 1))


def lse_anneal(h0, budget_fn, max_inner=50):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c = exact_score(h)
    betas = [20, 40, 80, 150, 300, 600, 1200]
    mom = 0.7

    for beta in betas:
        if budget_fn() < 40:
            break
        v = np.zeros(n)
        for _ in range(max_inner):
            if budget_fn() < 30:
                break
            profile = compute_profile(h)
            max_p = profile.max()
            weights = np.exp(beta * (profile - max_p))
            total = weights.sum()
            if total < 1e-30:
                break
            weights = weights / total
            G = compute_all_grads(h)
            grad = (weights[:, None] * G).sum(axis=0)
            grad -= grad.mean()
            v = mom * v + (1.0 - mom) * grad
            c_curr = exact_score(h)
            accepted = False
            for lr in [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]:
                h_try = repair_feasible(h - lr * v)
                c_try = exact_score(h_try)
                if c_try < c_curr - 1e-13:
                    h = h_try
                    if c_try < best_c - 1e-14:
                        best_c = c_try
                        best_h = h.copy()
                    accepted = True
                    break
            if not accepted:
                v *= 0.5
    return best_h, best_c


def trust_constr_solve(h0, maxiter=300, tol=1e-12):
    n = len(h0)
    dx = 2.0 / n
    M = 2 * n - 1

    def obj(x):
        return x[-1]

    def obj_grad(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g

    def eq_fun(x):
        return x[:-1].sum() - n / 2.0

    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g

    def ineq_fun(x):
        h = x[:-1]
        return x[-1] - np.correlate(h, 1 - h, mode="full") * dx

    def ineq_jac(x):
        h = x[:-1]
        G = compute_all_grads(h)
        J = np.zeros((M, n + 1))
        J[:, :-1] = -G
        J[:, -1] = 1.0
        return J

    t0 = exact_score(h0) + 1e-4
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, t0 + 2.0)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    # Try trust-constr first
    try:
        res = minimize(obj, x0, method='trust-constr', jac=obj_grad,
                       bounds=bounds, constraints=[
                           {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
                           {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
                       ],
                       options={'maxiter': maxiter, 'gtol': tol, 'xtol': tol})
        if res.x is not None and np.isfinite(res.x).all():
            hs = repair_feasible(res.x[:-1])
            if np.isfinite(exact_score(hs)):
                return hs
    except Exception:
        pass
    # Fallback SLSQP
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_grad,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': tol})
        if res.x is not None and np.isfinite(res.x).all():
            hs = repair_feasible(res.x[:-1])
            if np.isfinite(exact_score(hs)):
                return hs
    except Exception:
        pass
    return None

def subgrad_refine(h0, max_iters=200, budget_fn=None):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c = exact_score(h)
    for _ in range(max_iters):
        if budget_fn is not None and budget_fn() < 20:
            break
        profile = compute_profile(h)
        c_max = profile.max()
        ham = c_max - 0.003
        active = np.where(profile >= ham)[0]
        if len(active) == 0:
            active = np.array([np.argmax(profile)])
        G = compute_all_grads(h)
        w = profile[active] - ham + 0.001
        w = w / w.sum()
        grad = (w[:, None] * G[active, :]).sum(axis=0)
        grad -= grad.mean()
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        grad /= gn
        accepted = False
        for s in [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]:
            h_try = repair_feasible(h - s * grad)
            c_try = exact_score(h_try)
            if c_try < best_c - 1e-13:
                h = h_try
                best_c = c_try
                best_h = h.copy()
                accepted = True
                break
        if not accepted:
            break
    return best_h, best_c
