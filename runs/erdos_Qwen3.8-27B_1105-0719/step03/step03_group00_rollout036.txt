import numpy as np
import time

def project_feasible(x, target_sum):
    n = len(x)
    x = np.where(np.isfinite(x), x, 0.5)
    if target_sum <= 0:
        return np.zeros(n)
    if target_sum >= n:
        return np.ones(n)
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def c5_of_h(h, dx):
    return np.correlate(h, 1.0 - h, mode='full').max() * dx

def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def softmax_weights(corr, alpha):
    scaled = np.asarray(corr, dtype=np.float64) / alpha
    m = scaled.max()
    p = np.exp(scaled - m)
    s = p.sum()
    if s < 1e-300:
        return np.ones(len(corr)) / len(corr)
    return p / s

def weighted_gradient(h, w, dx):
    N = len(h)
    grad = np.zeros(N)
    # m = 0
    grad[:] += w[N-1] * (1.0 - 2.0 * h)
    for m in range(1, N):
        wm = w[N-1+m]
        if wm > 1e-14:
            grad[0:N-m] += wm * (1.0 - h[m:N])
            grad[m:N]   += wm * (-h[0:N-m])
        wp = w[N-1-m]
        if wp > 1e-14:
            grad[m:N]   += wp * (1.0 - h[0:N-m])
            grad[0:N-m] += wp * (-h[m:N])
    grad *= dx
    return grad

def softmax_descent(h, dx, target_sum, alphas, max_iters, rng, budget_s, t0):

    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    cur = h.copy()
    cur_c5 = best_c5
    
    for alpha in alphas:
        for it in range(max_iters):
            if budget_s - (time.time() - t0) < 5.0:
                return best_h, best_c5
            
            corr = compute_corrs(cur)
            w = softmax_weights(corr, alpha)
            grad = weighted_gradient(cur, w, dx)
            # Zero-mean direction
            grad -= grad.mean()
            
            # Step size search
            gnorm = np.linalg.norm(grad)

                break

def project_feasible(x, target_sum):

        else:
            hi = mid
