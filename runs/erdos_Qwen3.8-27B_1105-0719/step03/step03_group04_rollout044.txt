    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def exact_score(h):
    return float(np.max(compute_profile(h)))

def build_grad_matrix(h, n, dx):
    M = 2 * n - 1
    G = np.zeros((M, n))
    center = n - 1
    G[center, :] = dx * (1.0 - 2.0 * h)
    for d in range(1, n):
        G[center + d, d:n] += dx * (1.0 - h[0:n-d])
        G[center + d, 0:n-d] += -dx * h[d:n]
        G[center - d, d:n] += dx * (1.0 - h[0:n-d])
        G[center - d, 0:n-d] += -dx * h[d:n]
    return G

def project_feasible(x, target_sum):
    x = np.asarray(x, dtype=np.float64).copy()
    n = len(x)
    lo = x.min() - target_sum - 1.0
    hi = x.max() - 0.0 + 1.0
    for _ in range(200):

        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    t = 0.5 * (lo + hi)
    result = np.clip(x - t, 0.0, 1.0)
    diff = target_sum - result.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            order = np.argsort(1.0 - result)
        else:
            order = np.argsort(result)[::-1]
        remaining = abs(diff)

            idx = order[i]

                add = min(remaining, 1.0 - result[idx])
                result[idx] += add
            else:

                result[idx] -= rem
            remaining -= min(abs(diff) if diff > 0 else -diff, 1.0)

    return result

    pmax = profile.max()
    shifted = (profile - pmax) / tau

    weights = np.exp(shifted)
    weights /= weights.sum()
    G = build_grad_matrix(h, n, dx)
    grad = G.T @ weights

    # Solve: min ||G_a^T w||^2 s.t. w >= 0, sum(w) = 1

        return x[-1]
    
    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    
    def constraints_fun(x):
        h_val = x[:-1]
        t = x[-1]
        # Only use selected constraints for speed
        prof_sel = compute_profile(h_val)[indices]
        return t - prof_sel
    
    x0 = np.concatenate([h, [c0 + 1e-6]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    
    # Full constraint set (all M lags) via inequality
    def all_ineq(x):
        h_val = x[:-1]
        t = x[-1]
        prof = compute_profile(h_val)
        return t - prof
    
    def all_ineq_jac(x):
        h_val = x[:-1]
        G_full = build_grad_matrix(h_val, n, dx)
        J = np.zeros((M, n + 1))
        J[:, :-1] = -G_full
        J[:, -1] = 1.0
        return J
    
    cons = [
        {'type': 'eq', 'fun': lambda x: x[:-1].sum() - target_sum,
         'jac': lambda x: np.concatenate([np.ones(n), [0.0]])},
        {'type': 'ineq', 'fun': all_ineq, 'jac': all_ineq_jac}
    ]
    
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': 500, 'ftol': 1e-14})
        if res.x is not None and np.isfinite(res.x).all():
            h_sol = project_feasible(res.x[:-1], target_sum)
            c_sol = exact_score(h_sol)
            return h_sol, c_sol
    except Exception:
        pass
    return h, c0

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_feasible(h_new, new_n / 2.0)

def symmetrize(h):
    return (h + h[::-1]) / 2.0

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()
    rng = np.random.RandomState(seed)
    
    def time_left():
        return budget_s - (time.perf_counter() - t_start)
    
    # Load parent
    parent = None
    try:
        parent = np.array(initial_h_values, dtype=np.float64)
    except (NameError, Exception):
        pass
    
    if parent is None or len(parent) < 4 or not np.isfinite(parent).all():
        n_init = 70
        parent = np.full(n_init, 0.5)
    
    n_parent = len(parent)
    parent = np.clip(parent, 0.0, 1.0)
    target_parent = n_parent / 2.0
    parent = project_feasible(parent, target_parent)
    c_parent = exact_score(parent)
    
    best_h = parent.copy()
    best_c = c_parent
    
    # Try symmetrization
    h_sym = project_feasible(symmetrize(parent), target_parent)
    c_sym = exact_score(h_sym)
    if c_sym < best_c - 1e-12:
        best_h = h_sym
        best_c = c_sym
    
    # Phase 1: Annealed LSE at parent dimension
    if time_left() > 200:
        h_opt, c_opt = annealed_lse_optimize(best_h, t_start, t_start + time_left() - 50, target_parent, max_iters=3000)
        if c_opt < best_c - 1e-14:
            best_h = h_opt
            best_c = c_opt
    
    # Phase 2: SLSQP epigraph polish
    if time_left() > 80:
        h_slq, c_slq = slsqp_epigraph(best_h, t_start, t_start + time_left() - 20, target_parent, max_active=15)
        if c_slq < best_c - 1e-14:
            best_h = h_slq
            best_c = c_slq
    
    # Phase 3: Coordinate descent polish
    if time_left() > 60:
        h_cd, c_cd = coordinate_descent_polish(best_h, t_start, t_start + time_left() - 10, target_parent, max_rounds=300)
        if c_cd < best_c - 1e-14:
            best_h = h_cd
            best_c = c_cd
    
    # Phase 4: Try nearby even dimensions
    n0 = len(best_h)
    dims_to_try = []
    for d in range(max(20, n0 - 10), min(200, n0 + 15), 2):
        if d % 2 == 0 and d != n0:
            dims_to_try.append(d)
    # Sort by proximity to n0
    dims_to_try.sort(key=lambda d: abs(d - n0))
    
    for new_n in dims_to_try[:8]:
        if time_left() < 80:
            break
        t_dim = time_left() - 15
        h_interp = interpolate_to_n(best_h, new_n)
        target_new = new_n / 2.0
        
        # Optimize at new dimension
        h_opt, c_opt = annealed_lse_optimize(h_interp, t_start, t_start + t_dim, target_new, max_iters=1500)
        
        # SLSQP polish
        if time_left() > 40:
            h_slq, c_slq = slsqp_epigraph(h_opt, t_start, t_start + time_left() - 10, target_new, max_active=10)
            if c_slq < c_opt - 1e-14:
                h_opt = h_slq
                c_opt = c_slq
        
        if c_opt < best_c - 1e-14:
            best_h = h_opt
            best_c = c_opt
    
    # Phase 5: Multi-start at best dimension with blending
    n_best = len(best_h)
    if time_left() > 100:
        for attempt in range(5):
            if time_left() < 40:
                break
            # Create perturbed version
            scale = 0.02 + 0.01 * attempt
            perturb = rng.randn(n_best) * scale
            perturb -= perturb.mean()
            h_start = project_feasible(best_h + perturb, n_best / 2.0)
            c_start = exact_score(h_start)
            
            if c_start <= best_c + 0.002:
                t_sub = time_left() - 10
                h_opt, c_opt = annealed_lse_optimize(h_start, t_start, t_start + t_sub, n_best / 2.0, max_iters=1000)
                if c_opt < best_c - 1e-14:
                    best_h = h_opt
                    best_c = c_opt
    
    # Phase 6: Final SLSQP with more iterations
    if time_left() > 50:
        n_f = len(best_h)
        h_slq2, c_slq2 = slsqp_epigraph(best_h, t_start, t_start + time_left() - 5, n_f / 2.0, max_active=20)
        if c_slq2 < best_c - 1e-14:
            best_h = h_slq2
            best_c = c_slq2
    
    # Phase 7: Ultra-fine coordinate polish
    if time_left() > 30:
        n_f = len(best_h)
        h_cd2, c_cd2 = coordinate_descent_polish(best_h, t_start, t_start + time_left() - 3, n_f / 2.0, max_rounds=500)
        if c_cd2 < best_c - 1e-14:
            best_h = h_cd2
            best_c = c_cd2
    
    # Phase 8: One more LSE pass with very small tau
    if time_left() > 40:
        n_f = len(best_h)
        h_lse2, c_lse2 = annealed_lse_optimize(best_h, t_start, t_start + time_left() - 5, n_f / 2.0, max_iters=2000)
        if c_lse2 < best_c - 1e-14:
            best_h = h_lse2
            best_c = c_lse2
    
    # Final: ensure feasibility
    best_h = project_feasible(best_h, len(best_h) / 2.0)
    final_c = exact_score(best_h)
    
    # Validate with evaluator
    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            final_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 2:
            final_c = float(eval_result[0])
    except Exception:
        pass
    
    return best_h, final_c, len(best_h)
