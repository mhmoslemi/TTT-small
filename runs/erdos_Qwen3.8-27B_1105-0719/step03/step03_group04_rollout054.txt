import numpy as np
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def compute_gradient(h):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    G = np.zeros((M, n))
    center = n - 1
    G[center, :] = dx * (1.0 - 2.0 * h)
    for d in range(1, n):
        G[center + d, 0:n-d] += -dx * h[d:n]
        G[center + d, d:n] += dx * (1.0 - h[0:n-d])
        G[center - d, 0:n-d] += dx * (1.0 - h[d:n])
        G[center - d, d:n] += -dx * h[0:n-d]
    return G

def optimal_alpha(c, d, alpha_max):
    if alpha_max <= 1e-15:
        return 0.0
    lo, hi = 0.0, alpha_max
    for _ in range(45):
        mid = (lo + hi) * 0.5
        vals = c + mid * d
        m_star = int(np.argmax(vals))
        if d[m_star] < 0:
            lo = mid
        elif d[m_star] > 0:
            hi = mid
        else:
            return mid
    return (lo + hi) * 0.5

def repair_feasible(h):
    n = len(h)
    h = np.clip(h.astype(np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
    else:
        order = np.argsort(h)[::-1]
    remaining = abs(diff)
    sign = 1.0 if diff > 0 else -1.0
    for i in range(n):
        idx = order[i]
        if sign > 0:
            add = min(remaining, 1.0 - h[idx])
        else:
            add = min(remaining, h[idx])
        if add > 0:
            h[idx] += sign * add
        remaining -= add
        if remaining <= 1e-13:
            break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
    return np.clip(h, 0.0, 1.0)

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

def optimize_2point(h, t_start, deadline, max_iters=1000, top_k=5, tol=1e-10):
    n = len(h)

    best_h = h.copy()

    best_c = float(np.max(c))
    no_improve = 0

    for it in range(max_iters):
        if time.perf_counter() > deadline:
            break

        G = compute_gradient(h)

        m0 = int(np.argmax(c))

        best_improve = tol
        best_move = None
        best_new_c = None

        feasible_mask = (1.0 - h) > 1e-10

        for i in range(n):

def multi_angle_polish(h, t_start, deadline, rng, max_rounds=50):
    """Polish by trying moves from multiple "angles" (different active sets)."""
    n = len(h)
    best_h = h.copy()
    best_c = exact_score(h)

    for round_i in range(max_rounds):
        if time.perf_counter() > deadline:
            break
        h_cur = best_h.copy()
        c = compute_profile(h_cur)
        G = compute_gradient(h_cur)

        # Find top lags
        top_lags = np.argsort(c)[::-1][:10]

        # For each top lag, try moves that reduce it
        improved = False
        for m in top_lags:
            if time.perf_counter() > deadline:
                break
            Gm = G[m, :]
            # Find i with large positive Gm[i] (reducing h[i] helps)
            # and j with large negative Gm[j] (increasing h[j] helps)
            # Move: h[i] -= a, h[j] += a → delta_c[m] = a*(Gm[j]-Gm[i])
            # Want Gm[j] - Gm[i] < 0, i.e., Gm[i] > Gm[j]
            sources = np.where(h > 1e-10)[0]
            dests = np.where(h < 1 - 1e-10)[0]
            if len(sources) == 0 or len(dests) == 0:
                continue
            # Best source: max Gm[i], best dest: min Gm[j]
            src_rank = sources[np.argsort(-Gm[sources])][:3]
            dst_rank = dests[np.argsort(Gm[dests])][:3]
            for i in src_rank:
                for j in dst_rank:
                    if i == j:
                        continue
                    alpha_max = min(h_cur[i], 1.0 - h_cur[j])
                    if alpha_max <= 1e-12:
                        continue
                    d_all = G[:, j] - G[:, i]
                    alpha = optimal_alpha(c, d_all, alpha_max)
                    if alpha <= 1e-12:
                        continue
                    for f in [1.0, 0.5, 1.5]:
                        a = alpha * f
                        if a <= 1e-12 or a > alpha_max:
                            continue
                        h_cur[i] -= a
                        h_cur[j] += a
                        sc = float(np.max(compute_profile(h_cur)))
                        if sc < best_c - 1e-13:
                            best_c = sc
                            best_h = h_cur.copy()
                            improved = True
                        h_cur[i] += a
                        h_cur[j] -= a
        if not improved:
            break

    return best_h, best_c
