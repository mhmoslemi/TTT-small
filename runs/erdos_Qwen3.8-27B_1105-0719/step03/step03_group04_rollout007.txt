    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

    m = np.max(P)
    return m + np.log(np.sum(np.exp(β * (P - m)))) / β

    P = compute_profile(h)

    m = np.max(P)
    weights = np.exp(β * (P - m))
    weights /= weights.sum()
    # Gradient = weighted sum of individual gradients

    grad = G.T @ weights
