def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx])
            h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:

        remaining = -diff
        for i in range(n):

            g[i] -= dx * h[i + d]
    else:
        dp = -d
        for j in range(n - dp):
            g[j + dp] += dx * (1.0 - h[j])
            g[j] -= dx * h[j + dp]
    return g

def lp_direction(h, active_indices, n, dx):

    # Variables: d[0..n-1], t
    # n+1 variables
    # Objective: minimize t -> c = [0,...,0,1]

    c[-1] = 1.0

        g = build_gradient_row(h, m, n, dx)
        A_ub[idx, :n] = g
        A_ub[idx, -1] = -1.0

    A_eq[0, :n] = 1.0

    except Exception:
        pass
    return None, None

            add = min(remaining, capacity[idx])
            h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break

        remaining = -diff

            idx = order[i]
            rem = min(remaining, h[idx])

    d = m - (n - 1)

        for j in range(n - dp):
