import numpy as np
from scipy.optimize import linprog, minimize
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full"))) * dx

def profile_h(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx])
            h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    return h

def grad_all(h):
    """Compute gradient of all (2n-1) correlation values w.r.t. h.
    Returns array of shape (2n-1, n)."""
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    G = np.zeros((M, n))
    # Lag d = 0 (index n-1)
    G[n-1, :] = dx * (1.0 - 2.0 * h)
    # Positive lags d = 1, ..., n-1 (index n-1+d)
    for d in range(1, n):
        # out[n-1+d] = sum_{i=0}^{n-1-d} h[i]*(1-h[i+d])
        # d/dh[j]: (1-h[j+d]) if j <= n-1-d, minus h[j-d] if j >= d
        idx = n - 1 + d
        for j in range(n):
            g = 0.0
            if j <= n - 1 - d:
                g += (1.0 - h[j + d])
            if j >= d:
                g -= h[j - d]
            G[idx, j] = dx * g
    # Negative lags d = -1, ..., -(n-1) (index n-1+d = n-1-dp)
    for dp in range(1, n):
        # out[n-1-dp] = sum_{i=dp}^{n-1} h[i]*(1-h[i-dp])
        # = sum_{j=0}^{n-1-dp} h[j+dp]*(1-h[j])
        # d/dh[m]: (1-h[m-dp]) if m >= dp, minus h[m+dp] if m <= n-1-dp
        idx = n - 1 - dp
        for j in range(n):
            g = 0.0
            if j >= dp:
                g += (1.0 - h[j - dp])
            if j <= n - 1 - dp:
                g -= h[j + dp]
            G[idx, j] = dx * g
    return G

            remaining -= add

        try:
            res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
        except Exception:
            break
        
        if res.status != 0 or res.x is None:
            break
        
        d = res.x[:n] - h
        if np.linalg.norm(d) < 1e-12:
            break
        
        improved = False
        for alpha in [1.0, 0.5, 0.3, 0.15, 0.08, 0.04]:
            h_try = repair_feasible(h + alpha * d)
            s = exact_score(h_try)
            if s < best_c - 1e-13:
                h = h_try
                best_h = h.copy()
                best_c = s
                improved = True
                break
        if not improved:
            break
    
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()
    
    def elapsed():
        return time.perf_counter() - t_start
    
    def time_ok(margin=30):
        return elapsed() < budget_s - margin
    
    parent = None
    try:
        parent = np.array(initial_h_values, dtype=np.float64)
    except (NameError, Exception):
        pass
    
    if parent is None or len(parent) < 4:
        parent = np.full(70, 0.5)
    
    parent = np.clip(parent, 0.0, 1.0)
    parent = repair_feasible(parent)
    
    best_h = parent.copy()
    best_c = exact_score(parent)
    
    # Phase 1: Full SLP from parent (many iterations)
    if time_ok(60):
        h_new, c_new = slp_optimize(best_h, max_iters=300,
                                    time_fn=elapsed, time_limit=budget_s - 60)
        if c_new < best_c - 1e-12:
            best_h, best_c = h_new, c_new
    
    # Phase 2: Targeted pair moves
    if time_ok(40):
        h_new, c_new = targeted_pair_moves(best_h, n_sweeps=30,
                                           time_fn=elapsed, time_limit=budget_s - 40)
        if c_new < best_c - 1e-12:
            best_h, best_c = h_new, c_new
    
    # Phase 3: Another full SLP round
    if time_ok(60):
        h_new, c_new = slp_optimize(best_h, max_iters=300,
                                    time_fn=elapsed, time_limit=budget_s - 60)
        if c_new < best_c - 1e-12:
            best_h, best_c = h_new, c_new
    
    # Phase 4: Active-set SLP with perturbation
    if time_ok(50):
        n = len(best_h)
        rng = np.random.RandomState(seed)
        for trial in range(3):
            if not time_ok(15):
                break
            z = rng.randn(n) * 0.01
            z -= z.mean()
            h_pert = repair_feasible(best_h + z)
            h_new, c_new = slp_with_active(h_pert, active_k=40, max_iters=150,
                                           time_fn=elapsed, time_limit=budget_s - 15)
            if c_new < best_c - 1e-12:
                best_h, best_c = h_new, c_new
    
    # Phase 5: Dimension exploration
    if time_ok(100):
        n_cur = len(best_h)
        cand_dims = []
        for d in range(max(30, n_cur - 10), min(120, n_cur + 20)):
            if d % 2 == 0 and d != n_cur:
                cand_dims.append(d)
        cand_dims.sort(key=lambda x: abs(x - n_cur))
        
        for new_n in cand_dims[:6]:
            if not time_ok(30):
                break
            h_interp = interpolate_to_n(best_h, new_n)
            h_new, c_new = slp_optimize(h_interp, max_iters=200,
                                        time_fn=elapsed, time_limit=budget_s - 20)
            if c_new < best_c - 1e-12:
                best_h, best_c = h_new, c_new
                # Polish at this dimension
                if time_ok(20):
                    h_new2, c_new2 = targeted_pair_moves(best_h, n_sweeps=20,
                                                         time_fn=elapsed,
                                                         time_limit=budget_s - 10)
                    if c_new2 < best_c - 1e-12:
                        best_h, best_c = h_new2, c_new2
    
    # Phase 6: Final SLP rounds
    if time_ok(60):
        h_new, c_new = slp_optimize(best_h, max_iters=250, reheat_sigma=0.003,
                                    time_fn=elapsed, time_limit=budget_s - 30)
        if c_new < best_c - 1e-13:
            best_h, best_c = h_new, c_new
    
    # Phase 7: Final pair moves
    if time_ok(30):
        h_new, c_new = targeted_pair_moves(best_h, n_sweeps=30,
                                           time_fn=elapsed, time_limit=budget_s - 10)
        if c_new < best_c - 1e-13:
            best_h, best_c = h_new, c_new
    
    # Phase 8: Very fine SLP polish
    if time_ok(30):
        h_new, c_new = slp_optimize(best_h, max_iters=200, reheat_sigma=0.001,
                                    time_fn=elapsed, time_limit=budget_s - 5)
        if c_new < best_c - 1e-14:
            best_h, best_c = h_new, c_new
    
    # Final validation
    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)
    
    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            final_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 1:
            if np.isfinite(eval_result[0]):
                final_c = float(eval_result[0])
    except Exception:
        pass
    
    return best_h, final_c, len(best_h)
