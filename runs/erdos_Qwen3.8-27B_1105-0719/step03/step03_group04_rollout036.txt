import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def repair_feasible(h):
    n = len(h)
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        # Need to add mass; add to coordinates with most room
        order = np.argsort(1.0 - h)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, 1.0 - h[idx])
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        # Need to remove mass; remove from coordinates with most mass
        order = np.argsort(h)[::-1]
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    h = np.clip(h, 0.0, 1.0)
    return h

def build_gradient(h, n, dx):
    """Compute the gradient matrix G where G[k, i] = dP(k)/dh[i] * dx"""
    M = 2 * n - 1
    G = np.zeros((M, n))
    center = n - 1
    # P(k) = sum_m h[m] * (1-h[m-k+center]) * dx (approximately)
    # dP(k)/dh[i] = (1-h[i-k+center]) * dx - h[i+k-center] * dx (when indices valid)

    #   First: m=i, so i-k+center must be in [0,n-1], contribution: (1-h[i-k+center])*dx
    #   Second: m-k+center=i, so m=i+k-center, which must be in [0,n-1], contribution: -h[i+k-center]*dx
    for k in range(M):

            # Second term: h[i+shift] * (1-h[i]) → derivative w.r.t. h[i] is -h[i+shift]

Let me think about the **pairwise swap** approach more carefully.

Given current h with profile P and score c = max(P):
For a swap (i, j, α): h[i] -= α, h[j] += α (or vice versa)
The new profile is P'(k) = P(k) + α * (G(k,j) - G(k,i))
The new score is max_k P'(k)

We want to minimize this over α (and over choice of i, j, and sign).

For a fixed (i, j), the function f(α) = max_k [P(k) + α*(G(k,j) - G(k,i))] is convex in α. We can find its minimum by:
1. Computing v = G(:,j) - G(:,i) (a vector of length M=2n-1)
2. Finding α that minimizes max_k [P(k) + α*v(k)]
3. This is the same as finding the α that minimizes the maximum of M lines through (0, P(k)) with slope v(k)

The minimum of max of lines can be found by noting that at the optimum, at least two lines achieve the maximum (or we're at a boundary). We can use the fact that this is equivalent to:
min α: max_k [P(k) + α*v(k)]

Let t*(α) = max_k [P(k) + α*v(k)]
At the minimum, the subgradient 0 ∈ ∂t*(α) = conv{v(k) : k ∈ argmax}.
So we need 0 to be in the convex hull of {v(k) : P(k) + α*v(k) = t*(α)}.

A practical approach: 
- The optimal α is one of the "breakpoints" of the upper envelope of the lines
- Breakpoints occur where two lines intersect: P(k1) + α*v(k1) = P(k2) + α*v(k2)
  → α = (P(k2) - P(k1)) / (v(k1) - v(k2))
- There are O(M²) breakpoints, which is O(n²) ≈ 5000 for n=70
- We can compute all breakpoints, filter to valid α range, evaluate t* at each, and take the minimum

This is feasible! For each pair (i,j), we compute v = G(:,j) - G(:,i) (O(M) = O(n)), then find the optimal α in O(M²) time. Total per pair: O(n²). Total for all pairs: O(n⁴) ≈ 24 million for n=70. That might be a bit slow but doable.

def repair_feasible(h):
    n = len(h)
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, 1.0 - h[idx])
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)[::-1]
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break

def compute_profile(h):

def repair_feasible(h):
    n = len(h)
    h = np.clip(np.asarray(h, dtype=np.float64).copy(), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, 1.0 - h[idx])
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)[::-1]
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    return np.clip(h, 0.0, 1.0)

def build_gradient_rows(h, n, dx, rows):

        s = int(k) - center

def interpolate_to_n(h, new_n):

        if c_opt < best_c - 1e-14:
            best_h, best_c = h_opt, c_opt

    # Phase 4: Try different dimensions
    n0 = len(best_h)
    dims = []
    for d in range(max(30, n0 - 12), min(130, n0 + 16), 2):
        if d != n0:
            dims.append(d)
    dims.sort(key=lambda d: abs(d - n0))

    for new_n in dims[:10]:
        if time_left() < 50:
            break
        h_interp = interpolate_to_n(best_h, new_n)
        h_opt, c_opt = optimize_point(h_interp, t_start, budget_s, max_iters=120, seed=seed + new_n)
        if c_opt < best_c - 1e-14:
            best_h, best_c = h_opt, c_opt
        if time_left() > 30:
            h_pol, c_pol = slsqp_polish(best_h, t_start, budget_s, max_iter=200)
            if c_pol < best_c - 1e-14:
                best_h, best_c = h_pol, c_pol

    # Phase 5: Random restarts from best
    for restart in range(8):
        if time_left() < 30:
            break
        n_r = len(best_h)
        scale = 0.01 + 0.02 * restart * 0.3
        h_try = repair_feasible(best_h + rng.randn(n_r) * scale)
        c_try = exact_score(h_try)
        if c_try < best_c + 0.005:
            h_opt, c_opt = optimize_point(h_try, t_start, budget_s, max_iters=80, seed=seed + 100 + restart)
            if c_opt < best_c - 1e-14:
                best_h, best_c = h_opt, c_opt

    # Phase 6: Try a few structured starting points
    if time_left() > 50:
        n_s = len(best_h)
        for mode in range(3):
            if time_left() < 25:
                break
            h_struct = np.zeros(n_s)
            if mode == 0:
                h_struct[:] = 0.5 + 0.25 * np.sin(2 * np.pi * np.arange(n_s) / n_s + rng.uniform(0, 6))
            elif mode == 1:
                h_struct[:n_s // 2] = 0.7
                h_struct[n_s // 2:] = 0.3
                h_struct += rng.randn(n_s) * 0.05
            else:
                h_struct[:] = 0.5
                h_struct[::2] += 0.2
                h_struct[1::2] -= 0.2
            h_struct = repair_feasible(h_struct)
            c_struct = exact_score(h_struct)
            if c_struct < best_c + 0.02:
                h_opt, c_opt = optimize_point(h_struct, t_start, budget_s, max_iters=100, seed=seed + 200 + mode)
                if c_opt < best_c - 1e-14:
                    best_h, best_c = h_opt, c_opt

    # Phase 7: Final SLSQP + local polish on best
    if time_left() > 25:
        h_pol, c_pol = slsqp_polish(best_h, t_start, budget_s, max_iter=300)
        if c_pol < best_c - 1e-14:
            best_h, best_c = h_pol, c_pol

    if time_left() > 15:
        h_opt, c_opt = optimize_point(best_h, t_start, budget_s, max_iters=100, seed=seed + 999)
        if c_opt < best_c - 1e-14:
            best_h, best_c = h_opt, c_opt

    # Final validation
    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)

    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            final_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 2:
            final_c = float(eval_result[0])
    except Exception:
        pass

    return best_h, final_c, len(best_h)
