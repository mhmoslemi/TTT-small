def grad_row(h, m, n):
    dx = 2.0 / n
    d = m - (n - 1)
    if d == 0:
        return dx * (1.0 - 2.0 * h)
    elif d > 0:
        g = np.zeros(n)
        g[d:] += dx * (1.0 - h[:-d])
        g[:-d] -= dx * h[d:]
        return g
    else:
        p = -d
        g = np.zeros(n)
        g[:-p] += dx * (1.0 - h[p:])

        return g

def full_gradient_matrix(h):

def lse_gradient(h, beta):

    w = exp_z / exp_z.sum()

    G = full_gradient_matrix(h)
    g = w @ G  # shape (n,)
    g -= g.mean()  # Remove mean to stay on the hyperplane sum(h) = n/2
    return g

def trust_region_slp(h, R_init=0.05, max_slp_iters=50):

    h = h.copy()

    edges_new = np.linspace(0.0, 2.0, new_n + 1)
    h_new = np.zeros(new_n)

            if hi > lo:

        g[-1] = 1.0
        return g

        return x[:-1].sum() - n / 2.0

        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g

    x0 = np.concatenate([h_start, [t0]])

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(-(1.0 - h))
        remaining = diff
        for i in range(n):
            idx = order[i]
            cap = 1.0 - h[idx]
            if cap <= 1e-15:
                continue
            add = min(remaining, cap)
            h[idx] += add
            remaining -= add
            if remaining <= 1e-14:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            if h[idx] <= 1e-15:
                continue
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-14:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] = float(np.clip(h[idx] + diff2, 0.0, 1.0))
    return h

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def grad_row(h, m, n):
    dx = 2.0 / n
    d = m - (n - 1)
    if d == 0:
        return dx * (1.0 - 2.0 * h)
    elif d > 0:
        g = np.zeros(n)
        g[d:] += dx * (1.0 - h[:-d])
        g[:-d] -= dx * h[d:]
        return g
    else:
        p = -d
        g = np.zeros(n)
        g[:-p] += dx * (1.0 - h[p:])
        g[p:] -= dx * h[:-p]
        return g

def full_gradient_matrix(h):
    n = len(h)
    M = 2 * n - 1
    G = np.zeros((M, n))
    for m in range(M):
        G[m] = grad_row(h, m, n)
    return G

def lse_value(h, beta):
    profile = compute_profile(h)
    c = profile.max()
    z = beta * (profile - c)
    return c + logsumexp(z) / beta

def lse_grad(h, beta):
    n = len(h)
    profile = compute_profile(h)
    c = profile.max()
    z = beta * (profile - c)
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    w = exp_z / exp_z.sum()
    G = full_gradient_matrix(h)
    g = w @ G
    g -= g.mean()
    return g

def interp_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    eo = np.linspace(0.0, 2.0, n + 1)
    en = np.linspace(0.0, 2.0, new_n + 1)
    hn = np.zeros(new_n)
    for k in range(new_n):
        a, b = en[k], en[k + 1]
        tot = 0.0
        for j in range(n):
            lo, hi = max(a, eo[j]), min(b, eo[j + 1])
            if hi > lo:
                tot += h[j] * (hi - lo)
        hn[k] = tot / (b - a)
    hn = np.clip(hn, 0.0, 1.0)
    return repair_feasible(hn)

def slsqp_epigraph(h_start, maxiter=200):

        return t - prof

            best_h, best_c = h_ad, c_ad

    # Phase 4: SLSQP LSE (smooth)
    if tleft() > 60:
        h_sol = slsqp_lse(best_h, beta=1500, maxiter=200)
        if h_sol is not None:
            c_sol = exact_score(h_sol)
            if c_sol < best_c:
                best_h, best_c = h_sol, c_sol

    # Phase 5: SLSQP epigraph (exact)
    if tleft() > 50:
        h_sol = slsqp_epigraph(best_h, maxiter=200)
        if h_sol is not None:
            c_sol = exact_score(h_sol)
            if c_sol < best_c:
                best_h, best_c = h_sol, c_sol

    # Phase 6: Dimension exploration
    n_best = len(best_h)
    dims = sorted([d for d in range(n_best - 10, n_best + 21, 2) if 40 <= d <= 160 and d != n_best])
    for dn in dims[:8]:
        if tleft() < 40:
            break
        hi = interp_to_n(best_h, dn)
        hc, cc = lse_descent(hi, [80, 400, 2000], max_iters_per_beta=20)
        if cc < best_c:
            best_h, best_c = hc, c
        ha, ca = active_descent(best_h, max_iters=100, tol=0.002)
        if ca < best_c:
            best_h, best_c = ha, ca
        if tleft() > 20:
            hs, cs2 = trust_region_slp(best_h, r_init=0.03, max_iters=25)
            if cs2 < best_c:
                best_h, best_c = hs, cs2

    # Phase 7: Perturbation escape
    if tleft() > 80:
        hp2, cp2 = perturb_refine(best_h, rng, n_pert=12, scale=0.04)
        if cp2 < best_c:
            best_h, best_c = hp2, cp2
        hp3, cp3 = perturb_refine(best_h, rng, n_pert=10, scale=0.015)
        if cp3 < best_c:
            best_h, best_c = hp3, cp3

    # Phase 8: Refine best
    if tleft() > 50:
        h_lse2, c_lse2 = lse_descent(best_h, [500, 2000, 8000], max_iters_per_beta=20)
        if c_lse2 < best_c:
            best_h, best_c = h_lse2, c_lse2
        h_slp2, c_slp2 = trust_region_slp(best_h, r_init=0.02, max_iters=35)
        if c_slp2 < best_c:
            best_h, best_c = h_slp2, c_slp2

    # Phase 9: SLSQP LSE high beta
    if tleft() > 35:
        h_sol2 = slsqp_lse(best_h, beta=5000, maxiter=150)
        if h_sol2 is not None:
            c_sol2 = exact_score(h_sol2)
            if c_sol2 < best_c:
                best_h, best_c = h_sol2, c_sol2

    # Phase 10: Final SLSQP epigraph
    if tleft() > 25:
        h_sol3 = slsqp_epigraph(best_h, maxiter=250)
        if h_sol3 is not None:
            c_sol3 = exact_score(h_sol3)
            if c_sol3 < best_c:
                best_h, best_c = h_sol3, c_sol3

    # Phase 11: Ultra-fine active descent
    if tleft() > 15:
        h_af, c_af = active_descent(best_h, max_iters=200, tol=0.0005)
        if c_af < best_c:
            best_h, best_c = h_af, c_af

    # Phase 12: Micro perturbation
    if tleft() > 8:
        hm, cm = perturb_refine(best_h, rng, n_pert=8, scale=0.005)
        if cm < best_c:
            best_h, best_c = hm, cm

    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)

    try:
        ev = evaluate_erdos_solution(best_h)
        if isinstance(ev, (float, int)):
            final_c = float(ev)
        elif isinstance(ev, tuple) and len(ev) >= 2:
            final_c = float(ev[0])
    except Exception:
        pass

    return best_h, final_c, len(best_h)
