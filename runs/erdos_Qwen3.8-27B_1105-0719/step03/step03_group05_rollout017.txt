def repair_feasible(h):
    n = len(h)
    h = np.clip(np.asarray(h, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        # Need to add mass
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx] + 1e-15)
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        # Need to remove mass
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx] + 1e-15)
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    # Final fix
    diff2 = target - h.sum()
    if abs(diff2) > 1e-14:
        h[int(np.argmin(1.0 - h) if diff2 > 0 else np.argmax(h))] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def solve_lp_direction(h, D=0.3):

    dx = 2.0 / n
    M = 2 * n - 1
    P = compute_profile(h)
    c = float(np.max(P))
    G = build_gradient_matrix(h)
    
    # Variables: d_0, ..., d_{n-1}, S
    # Minimize S
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    # Constraints: G @ d - S <= -P
    # i.e., [G | -1] @ [d; S] <= -P
    A_ub = np.hstack([G, -np.ones((M, 1))])
    b_ub = -P
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds: -D <= d_i <= D, S >= -1 (or 0)
    bounds = [(-D, D)] * n + [(-c, c + 1)]  # S bounded
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            d = res.x[:n]
            d -= d.mean()  # ensure sum = 0 (numerical)
            return d, res.x[-1]
    except Exception:
        pass
    return None, None

import numpy as np
from scipy.optimize import linprog, minimize
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def repair_feasible(h):
    n = len(h)
    h = np.clip(np.asarray(h, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx] + 1e-15)
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx] + 1e-15)
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-13:
        idx = int(np.argmax(1.0 - h)) if diff2 > 0 else int(np.argmin(h))
        h[idx] += np.clip(diff2, -(h[idx]), 1.0 - h[idx])
    return np.clip(h, 0.0, 1.0)

def build_gradient_matrix(h):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    G = np.zeros((M, n))
    for d in range(-(n - 1), n):
        m = n - 1 + d
        if d >= 0:
            G[m, :n - d] += dx * (1.0 - h[d:n])
            G[m, d:n] += -dx * h[0:n - d]
        else:
            r = -d
            G[m, r:n] += dx * (1.0 - h[0:n - r])
            G[m, 0:n - r] += -dx * h[r:n]
    return G

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = np.linspace(0, 2, n + 1)
    xt = np.linspace(0, 2, new_n + 1)
    h_interp = np.zeros(new_n)
    for k in range(new_n):
        lo_k, hi_k = xt[k], xt[k + 1]
        total_mass = 0.0
        total_width = 0.0
        for j in range(n):
            lo_j, hi_j = xs[j], xs[j + 1]
            overlap = min(hi_k, hi_j) - max(lo_k, lo_j)
            if overlap > 0:
                total_mass += h[j] * overlap
                total_width += overlap
        h_interp[k] = total_mass / total_width if total_width > 1e-15 else 0.5
    h_interp = np.clip(h_interp, 0.0, 1.0)
    return repair_feasible(h_interp)

def symmetrize(h):
    return repair_feasible((h + h[::-1]) / 2.0)

def antisymmetrize(h):
    n = len(h)
    h_out = np.zeros(n)
    for i in range(n):
        h_out[i] = (h[i] + (1.0 - h[n - 1 - i])) / 2.0
    return repair_feasible(h_out)

    else:

    else:
        alpha_lo = -1.0

    best_c = exact_score(base)
    best_h = base.copy()
    best_a = 0.0
    lo = max(alpha_lo - 1e-8, -2.0)
    hi = min(alpha_hi + 1e-8, 2.0)
    if hi <= lo:

    alphas = np.linspace(lo, hi, n_samples)

                    best_h = h1.copy()
                    best_a = a1

                    best_a = a2

                break

def select_pair_indices(h, direction, n_pairs=8):
    n = len(h)
    d = direction - direction.mean()
    score_pos = d
    score_neg = -d
    top_pos = np.argsort(-score_pos)[:n_pairs * 2]
    top_neg = np.argsort(-score_neg)[:n_pairs * 2]
    pairs = []
    for i in top_pos[:n_pairs]:
        for j in top_neg[:n_pairs]:
            if i != j:
                pairs.append((int(i), int(j)))
                break
    if len(pairs) < n_pairs:
        for i in top_pos:
            for j in top_neg:
                if i != j and (i, j) not in pairs:
                    pairs.append((int(i), int(j)))
                    if len(pairs) >= n_pairs * 2:
                        break
            if len(pairs) >= n_pairs * 2:
                break

                        stuck = 0
        if not improved:
            h_try, c_try = line_search(h, -direction if direction is not None else np.zeros(n), n_samples=15)
            if c_try < best_c - 1e-13:
                h = h_try
                best_c = c_try
                stuck = 0
                continue
            stuck += 1
            if stuck > 30:
                break
        D *= 1.02
    return h, best_c

def pair_local_optimize(h, n_pairs_per_iter=6, max_rounds=100, time_limit_s=None):
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    t0 = time.perf_counter()
    for rnd in range(max_rounds):
        if time_limit_s is not None and (time.perf_counter() - t0) > time_limit_s - 5:
            break
        P = compute_profile(h)
        G = build_gradient_matrix(h)
        c = float(np.max(P))
        margin = max(0.002, 2.0 * (c - float(np.median(P))))
        active = np.where(P >= c - margin)[0]
        if len(active) == 0:
            active = np.array([np.argmax(P)])
        w = np.ones(len(active)) / len(active)
        grad = G[active, :].T @ w
        grad -= grad.mean()
        pairs = select_pair_indices(h, grad, n_pairs=n_pairs_per_iter)
        improved = False
        for (i, j) in pairs:
            if time_limit_s is not None and (time.perf_counter() - t0) > time_limit_s - 3:
                break
            h_try, c_try = pair_move(h, i, j, n_samples=25)
            if c_try < best_c - 1e-13:
                h = h_try
                best_c = c_try
                improved = True
        if not improved and rnd > 0:
            broken = 0
            for (i, j) in pairs[:3]:
                h_try, c_try = pair_move(h, i, j, n_samples=40)
                if c_try < best_c - 1e-13:
                    h = h_try
                    best_c = c_try
                    broken = 1
                    break
            if broken == 0 and rnd > 10:
                break
    return h, best_c

def lse_descent(h, beta=80.0, max_iters=100, alpha_init=0.004, time_limit_s=None):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best_c = exact_score(h)
    alpha = alpha_init
    stuck = 0
    t0 = time.perf_counter()
    for it in range(max_iters):
        if time_limit_s is not None and (time.perf_counter() - t0) > time_limit_s - 5:
            break
        profile = compute_profile(h)
        scaled = beta * profile
        max_s = np.max(scaled)
        shifted = scaled - max_s
        weights = np.exp(shifted)
        wsum = weights.sum()
        if wsum < 1e-300:
            break
        weights /= wsum
        G = build_gradient_matrix(h)
        grad = G.T @ weights
        grad -= grad.mean()
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad /= gnorm
        improved = False
        for a in [alpha, alpha * 0.5, alpha * 0.25, alpha * 0.1]:
            h_try = repair_feasible(h - a * grad)
            s = exact_score(h_try)
            if s < best_c - 1e-14:
                h = h_try
                best_c = s
                improved = True
                alpha = min(alpha * 1.08, 0.02)
                stuck = 0
                break
            elif s < float(np.max(compute_profile(h))) - 1e-14:
                h = h_try
                improved = True
                break
        if not improved:
            alpha *= 0.5
            stuck += 1
            if stuck > 25 or alpha < 1e-9:
                break
    return h, best_c

def active_set_descent(h, max_iters=200, n_active=12, alpha_init=0.004, time_limit_s=None):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best_c = exact_score(h)
    alpha = alpha_init
    stuck = 0
    t0 = time.perf_counter()
    for it in range(max_iters):
        if time_limit_s is not None and (time.perf_counter() - t0) > time_limit_s - 5:
            break
        profile = compute_profile(h)
        c = float(np.max(profile))
        if c + 1e-14 < best_c:
            best_c = c
            stuck = 0
            alpha = min(alpha * 1.08, 0.02)
        margin = max(0.002, 4.0 * (c - float(np.median(profile))))
        active = np.where(profile >= c - margin)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(profile))])
        if len(active) > n_active:
            top_idx = np.argsort(profile[active])[-n_active:]
            active = active[top_idx]
        G = build_gradient_matrix(h)
        G_a = G[active, :]
        w = np.ones(len(active)) / len(active)
        grad = G_a.T @ w
        grad -= grad.mean()
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad /= gnorm
        improved = False
        for a in [alpha, alpha * 0.5, alpha * 0.25, alpha * 0.1]:
            h_try = repair_feasible(h - a * grad)
            s = exact_score(h_try)
            if s < best_c - 1e-14:
                h = h_try
                best_c = s
                improved = True
                break
            elif s < c - 1e-14:
                h = h_try
                improved = True
                break
        if not improved:
            alpha *= 0.55
            stuck += 1
            if stuck > 35 or alpha < 1e-9:
                break
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s + t_start - time.perf_counter()

    parent = None
    try:
        parent = np.array(initial_h_values, dtype=np.float64)
    except NameError:
        pass
    except Exception:
        pass

    if parent is None or len(parent) < 4:
        parent = np.full(70, 0.5)

    parent = np.clip(parent, 0.0, 1.0)
    parent = repair_feasible(parent)
    n_parent = len(parent)
    c_parent = exact_score(parent)

    best_h = parent.copy()
    best_c = c_parent

    h_sym = symmetrize(parent)
    c_sym = exact_score(h_sym)
    if c_sym < best_c - 1e-12:
        best_h = h_sym
        best_c = c_sym

    h_asym = antisymmetrize(parent)
    c_asym = exact_score(h_asym)
    if c_asym < best_c - 1e-12:
        best_h = h_asym
        best_c = c_asym

    # Phase 1: LSE + active set warm start
    if time_left() > 100:
        h2, c2 = lse_descent(best_h, beta=60, max_iters=80, time_limit_s=min(time_left() - 20, 60))
        if c2 < best_c - 1e-13:
            best_h = h2
            best_c = c2
        h3, c3 = active_set_descent(best_h, max_iters=80, n_active=10, time_limit_s=min(time_left() - 20, 60))
        if c3 < best_c - 1e-13:
            best_h = h3
            best_c = c3

    # Phase 2: LP-based optimization (main new method)
    if time_left() > 60:
        h4, c4 = lp_local_optimize(best_h, max_iters=200, D=0.35, time_limit_s=min(time_left() - 20, 200))
        if c4 < best_c - 1e-13:
            best_h = h4
            best_c = c4
        h5, c5 = pair_local_optimize(best_h, n_pairs_per_iter=6, max_rounds=50, time_limit_s=min(time_left() - 20, 100))
        if c5 < best_c - 1e-13:
            best_h = h5
            best_c = c5
        h6, c6 = lp_local_optimize(best_h, max_iters=100, D=0.25, time_limit_s=min(time_left() - 20, 100))
        if c6 < best_c - 1e-13:
            best_h = h6
            best_c = c6

    # Phase 3: Try different resolutions
    if time_left() > 80:
        n_b = len(best_h)
        resolution_candidates = []
        for factor in [0.8, 0.85, 0.9, 0.95, 1.05, 1.1, 1.15, 1.2]:
            d = int(round(n_b * factor / 2)) * 2
            d = max(24, min(d, 140))
            if d != n_b:
                resolution_candidates.append(d)
        # Deduplicate
        seen = set()
        unique_dims = []
        for d in sorted(resolution_candidates, key=lambda x: abs(x - n_b)):
            if d not in seen:
                seen.add(d)
                unique_dims.append(d)
        for new_n in unique_dims[:4]:
            if time_left() < 40:
                break
            h_interp = interpolate_to_n(best_h, new_n)
            h_l, cl = lp_local_optimize(h_interp, max_iters=100, D=0.35, time_limit_s=min(time_left() - 30, 80))
            if cl < best_c - 1e-13:
                best_h = h_l
                best_c = cl
            h_p, cp = pair_local_optimize(best_h, n_pairs_per_iter=5, max_rounds=30, time_limit_s=min(time_left() - 30, 50))
            if cp < best_c - 1e-13:
                best_h = h_p
                best_c = cp

    # Phase 4: Multi-start with diverse perturbations
    if time_left() > 80:
        n_b = len(best_h)
        starts = [best_h.copy()]
        for eps in [0.005, 0.01, 0.02, 0.04]:
            z = rng.randn(n_b)
            z -= z.mean()
            starts.append(repair_feasible(best_h + eps * z))
        z_s = rng.randn(n_b)
        z_s = symmetrize(z_s * 0.5 + 0.5) - 0.5
        z_s -= z_s.mean()
        starts.append(repair_feasible(best_h + 0.01 * z_s))
        z_a = antisymmetrize(np.clip(rng.randn(n_b) * 0.3 + 0.5, 0, 1))
        z_a = z_a - z_a.mean()
        starts.append(repair_feasible(best_h + 0.008 * z_a))
        for h0 in starts:
            if time_left() < 35:
                break
            h_sol, c_sol = lp_local_optimize(h0, max_iters=120, D=0.3, time_limit_s=min(time_left() - 25, 60))
            if c_sol < best_c - 1e-12:
                best_h = h_sol
                best_c = c_sol
            h_p, c_p = pair_local_optimize(best_h, n_pairs_per_iter=5, max_rounds=30, time_limit_s=min(time_left() - 25, 40))
            if c_p < best_c - 1e-12:
                best_h = h_p
                best_c = c_p

    # Phase 5: Focused pair moves with different strategies
    if time_left() > 40:
        h_p, c_p = pair_local_optimize(best_h, n_pairs_per_iter=10, max_rounds=80, time_limit_s=min(time_left() - 20, 80))
        if c_p < best_c - 1e-13:
            best_h = h_p
            best_c = c_p

    # Phase 6: LSE with different betas
    for beta in [50, 100, 150]:
        if time_left() < 25:
            break
        h_l, cl = lse_descent(best_h, beta=beta, max_iters=80, time_limit_s=min(time_left() - 15, 40))
        if cl < best_c - 1e-13:
            best_h = h_l
            best_c = cl

    # Phase 7: Active set with various n_active
    for n_act in [6, 10, 15]:
        if time_left() < 20:
            break
        h_a, ca = active_set_descent(best_h, max_iters=100, n_active=n_act, time_limit_s=min(time_left() - 10, 40))
        if ca < best_c - 1e-13:
            best_h = h_a
            best_c = ca

    # Phase 8: Final LP polish with smaller D
    if time_left() > 30:
        h_f, cf = lp_local_optimize(best_h, max_iters=150, D=0.15, time_limit_s=min(time_left() - 15, 60))
        if cf < best_c - 1e-14:
            best_h = h_f
            best_c = cf

    # Phase 9: Final pair polish
    if time_left() > 20:
        h_pp, cpp = pair_local_optimize(best_h, n_pairs_per_iter=8, max_rounds=40, time_limit_s=min(time_left() - 10, 30))
        if cpp < best_c - 1e-14:
            best_h = h_pp
            best_c = cpp

    # Phase 10: Final LSE + active set
    if time_left() > 15:
        h_ll, cl = lse_descent(best_h, beta=120, max_iters=80, alpha_init=0.002, time_limit_s=min(time_left() - 8, 20))
        if cl < best_c - 1e-14:
            best_h = h_ll
            best_c = cl
        h_aa, ca = active_set_descent(best_h, max_iters=100, n_active=8, alpha_init=0.002, time_limit_s=min(time_left() - 5, 15))
        if ca < best_c - 1e-14:
            best_h = h_aa
            best_c = ca

    # Final validation
    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)

    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            final_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 2:
            final_c = float(eval_result[0])
    except Exception:
        pass

    return best_h, final_c, len(best_h)
