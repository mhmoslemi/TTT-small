def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def repair_feasible(h):
    n = len(h)
    h = np.clip(np.asarray(h, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, 1.0 - h[idx])
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)[::-1]
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def build_gradient(h, n, dx):
    M = 2 * n - 1
    G = np.zeros((M, n))
    center = n - 1
    G[center, :] = dx * (1.0 - 2.0 * h)
    for d in range(1, n):
        G[center + d, 0:n-d] += -dx * h[d:n]
        G[center + d, d:n] += dx * (1.0 - h[0:n-d])
        G[center - d, 0:n-d] += dx * (1.0 - h[d:n])
        G[center - d, d:n] += -dx * h[0:n-d]

def get_active(profile, beam, top_k):
    T = np.max(profile)
    thresh = T - beam
    active = np.where(profile >= thresh)[0]
    if len(active) == 0:
        active = np.array([np.argmax(profile)])
    if len(active) > top_k:
        top_idx = np.argsort(profile[active])[-top_k:]
        active = active[top_idx]
    return active, T

def lp_step(h, active_set, radius):
    n = len(h)
    dx = 2.0 / n
    G = build_gradient(h, n, dx)
    profile = compute_profile(h)
    M = len(active_set)
    G_a = G[active_set, :]
    p_a = profile[active_set]
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_ub = np.zeros((M, n + 1))
    A_ub[:, :n] = G_a
    A_ub[:, -1] = -1.0
    b_ub = -p_a
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    lower = np.maximum(-h, -radius)
    upper = np.minimum(1.0 - h, radius)
    bounds = list(zip(lower, upper))
    bounds.append((0.0, None))
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'maxiter': 3000})
        if res.success and res.x is not None and np.isfinite(res.x).all():
            h_new = repair_feasible(h + res.x[:n])
            return h_new, res.x[-1]
    except Exception:
        pass
    return None, None

    dims = sorted(set(dims), key=lambda d: abs(d - n0))
    
    for new_n in dims[:5]:
        if tl() < 70:
            break
        h_interp = interpolate_to_n(best_h, new_n)
        h_opt, c_opt = main_optimize(h_interp, t_start, budget_s, max_iters=150)
        if c_opt < best_c - 1e-14:
            best_h, best_c = h_opt, c_opt
            if tl() > 40:
                h_ps, c_ps = pair_swap_search(best_h, t_start, budget_s, max_rounds=80, max_pairs=600)
                if c_ps < best_c - 1e-14:
                    best_h, best_c = h_ps, c_ps
    
    # Phase 6: SLSQP polish
    if tl() > 60:
        h_sls, c_sls = lsqp_polish(best_h, t_start, budget_s)
        if c_sls < best_c - 1e-14:
            best_h, best_c = h_sls, c_sls
    
    # Phase 7: Final intensive pair-swap + gradient
    if tl() > 40:
        h7, c7 = pair_swap_search(best_h, t_start, budget_s, max_rounds=100, max_pairs=1200)
        if c7 < best_c - 1e-14:
            best_h, best_c = h7, c7
        h8, c8 = gradient_polish(best_h, t_start, budget_s, max_iters=200)
        if c8 < best_c - 1e-14:
            best_h, best_c = h8, c8
    
    # Phase 8: Very fine micro-polish
    if tl() > 20:
        h9, c9 = pair_swap_search(best_h, t_start, budget_s, max_rounds=80, max_pairs=1500)
        if c9 < best_c - 1e-14:
            best_h, best_c = h9, c9
    
    # Final validation
    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)
    
    try:
        ev = evaluate_erdos_solution(best_h)
        if isinstance(ev, (float, int)):
            final_c = float(ev)
        elif isinstance(ev, tuple) and len(ev) >= 2:
            final_c = float(ev[0])
    except Exception:
        pass
    
    return best_h, final_c, len(best_h)
