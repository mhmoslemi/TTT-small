def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx


def exact_score(h):
    return float(np.max(compute_profile(h)))


def compute_grad_weighted(h, weights, n, dx):
    """Compute weighted sum of gradients of f_k for all lags.
    weights has length 2*n-1, indexed same as the profile.
    Returns gradient vector of length n.
    """
    center = n - 1
    g = np.zeros(n)
    # Center lag (d=0): f_0 = sum h[i]*(1-h[i])
    g += weights[center] * dx * (1.0 - 2.0 * h)
    for d in range(1, n):
        w_pos = weights[center + d] * dx
        w_neg = weights[center - d] * dx
        if abs(w_pos) < 1e-15 and abs(w_neg) < 1e-15:
            continue
        # Lag +d: f = sum_{i=0}^{n-1-d} h[i+d]*(1-h[i])
        # grad: for j in [0, n-1-d]: -h[j+d]
        #       for j in [d, n-1]: (1-h[j-d])
        #       overlap [d, n-1-d]: both apply
        if w_pos != 0:
            g[:n-d] += w_pos * (-h[d:])       # second term: -h[j+d] for j in [0, n-1-d]
            g[d:] += w_pos * (1.0 - h[:n-d])   # first term: (1-h[j-d]) for j in [d, n-1]
        # Lag -d: f = sum_{i=0}^{n-1-d} h[i]*(1-h[i+d])
        # grad: for j in [0, n-1-d]: (1-h[j+d])
        #       for j in [d, n-1]: -h[j-d]
        if w_neg != 0:
            g[:n-d] += w_neg * (1.0 - h[d:])    # first term: (1-h[j+d]) for j in [0, n-1-d]
            g[d:] += w_neg * (-h[:n-d])         # second term: -h[j-d] for j in [d, n-1]
    return g


def safe_subtract_mean(v):

    return v - v.mean()


def line_search(h, v, alpha_lo, alpha_hi, max_evals=30):

    Uses grid + golden section refinement.
    """

        return 0.0, exact_score(h)

    best_score = exact_score(h)

    scores = []

        s = exact_score(h + a * v)
        scores.append(s)
        if s < best_score:
            best_score = s
            best_alpha = a
    scores = np.array(scores)
    
    # Golden section refinement around best
    i_best = int(np.argmin(scores))
    lo = alphas[max(0, i_best - 1)]
    hi = alphas[min(len(alphas) - 1, i_best + 1)]
    
    gr = (np.sqrt(5) - 1) / 2
    a1 = hi - gr * (hi - lo)
    a2 = lo + gr * (hi - lo)
    f1 = exact_score(h + a1 * v)
    f2 = exact_score(h + a2 * v)
    
    for _ in range(20):

            a2 = a1
            f2 = f1

        else:

            a1 = a2
            f1 = f2

        best_alpha = mid

def soft_subgrad_step(h, temperature, momentum_prev, momentum_beta):
    """One step of soft-subgradient with momentum and line search."""
    n = len(h)
    dx = 2.0 / n
    profile = compute_profile(h)
    
    # Softmax weights
    shifted = profile - profile.max()
    weights = np.exp(shifted / temperature)
    weights /= weights.sum()
    
    # Compute weighted gradient
    g = compute_grad_weighted(h, weights, n, dx)
    
    # Apply momentum
    v = -g + momentum_beta * momentum_prev
    v = safe_subtract_mean(v)  # ensure sum = 0
    
    if np.linalg.norm(v) < 1e-15:
        return h, exact_score(h), np.zeros_like(h)
    
    # Line search
    alpha_lo, alpha_hi = compute_alpha_bounds(h, v)
    # Limit step size
    max_step = 0.1
    alpha_lo = max(alpha_lo, -max_step)
    alpha_hi = min(alpha_hi, max_step)
    
    if alpha_hi <= alpha_lo:
        return h, exact_score(h), v
    
    best_alpha, best_score = line_search(h, v, alpha_lo, alpha_hi)
    h_new = h + best_alpha * v
    h_new = np.clip(h_new, 0.0, 1.0)
    
    return h_new, best_score, v


def repair(h, n):
    """Repair h to be feasible: [0,1] and sum = n/2."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        # Need to add mass; add to positions with most room
        room = 1.0 - h
        order = np.argsort(-room)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, room[idx])
            h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        # Need to remove mass; remove from positions with most mass
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    # Final fix
    h = np.clip(h, 0.0, 1.0)
    diff2 = target - h.sum()
    if abs(diff2) > 1e-14:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h


def interp_to_n(h, new_n):

    n = len(h)

    fh = np.concatenate([[h[0]], h, [h[-1]]])
    hn = np.interp(xt, fx, fh)
    return repair(hn, new_n)

Actually, for SLSQP, the ineq constraint should be: `g(x) >= 0` where `g(x) = t - max_k P_k(h)`. But this is non-smooth. A common trick is to use `t - P_k(h) >= 0` for all k, which gives M constraints. The Jacobian would be M x (n+1).

def ineq_fun(x):
    h = x[:-1]
    t = x[-1]
    return t - np.correlate(h, 1 - h, mode="full") * dx

def ineq_jac(x):
    h = x[:-1]
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    # Compute gradient of each P_k with respect to h
    G = np.zeros((M, n))
    center = n - 1
    G[center, :] = dx * (1.0 - 2.0 * h)
    for d in range(1, n):
        G[center + d, :n-d] += -dx * h[d:]
        G[center + d, d:] += dx * (1.0 - h[:n-d])
        G[center - d, :n-d] += dx * (1.0 - h[d:])
        G[center - d, d:] += -dx * h[:n-d]
    J = np.zeros((M, n + 1))
    J[:, :-1] = -G  # d(t - P_k)/dh = -dP_k/dh
    J[:, -1] = 1.0  # d(t - P_k)/dt = 1
    return J

def repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        idx = np.argsort(1.0 - h)
        remaining = diff
        for i in range(n):
            add = min(remaining, 1.0 - h[idx[i]])
            h[idx[i]] += add
            remaining -= add
            if remaining <= 1e-13:
                break
    else:
        idx = np.argsort(h)
        remaining = -diff
        for i in range(n):
            rem = min(remaining, h[idx[i]])
            h[idx[i]] -= rem
            remaining -= rem
            if remaining <= 1e-13:

        else:
            i = int(np.argmax(h))

def grad_profile(h):
