        c_obj[-1] = 1.0
        A_ub = np.empty((m, n + 1))
        A_ub[:, :n] = G
        A_ub[:, n] = -1.0
        b_ub = -corrs[active_idx]
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)
        bounds = [(-h[i] * alpha, (1.0 - h[i]) * alpha) for i in range(n)]
        bounds.append((-np.inf, np.inf))
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if res.status != 0:
                alpha *= alpha_decay
                no_improve += 1
                continue
            delta = res.x[:n]
        except Exception:
            alpha *= alpha_decay
            no_improve += 1
            continue
        if np.abs(delta).max() < 1e-15:
            alpha *= alpha_decay
            no_improve += 1
            continue
        improved = False
        for s in [1.0, 0.7, 0.5, 0.35, 0.2, 0.1, 0.05]:
            h_new = project_feasible(h + s * delta, target_sum)
            c5_new = c5_of_h(h_new, dx)
            if c5_new < best_c5 - 1e-14:
                h = h_new
                best_c5 = c5_new
                best_h = h_new.copy()
                improved = True
                break
        if improved:
            no_improve = 0
            alpha = alpha_start * 0.7
        else:
            no_improve += 1
            alpha *= alpha_decay
            if no_improve == 20:
                n_active = min(n_total, n_active + 10)
                alpha = max(alpha, 0.15)
            if no_improve > max_no_improve:
                break
    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n = len(h0)
    dx = 2.0 / n
    target_sum = n / 2.0
    h0 = project_feasible(h0, target_sum)
    best_h = h0.copy()
    best_c5 = c5_of_h(h0, dx)

    # Phase 1: SA exploration (25% of budget)
    sa_time = budget_s * 0.25
    sa_t1 = t0 + sa_time
    sa_seeds = [seed, seed + 1, seed + 2, seed + 3, seed + 4]
    for si in range(len(sa_seeds)):
        if time.time() > sa_t1:
            break
        dt = sa_t1 - time.time()
        if dt < 15:
            break
        pert_sigma = [0.03, 0.05, 0.08, 0.02, 0.06][si]
        h_start = project_feasible(best_h + pert_sigma * rng.randn(n), target_sum)
        h_sa, c5_sa = sa_continuous(h_start, dx, target_sum, t0, dt, seed=sa_seeds[si],
                                    t_start=0.003, cool=0.99999)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa

    # Phase 2: LSE subgradient optimization (30% of budget)
    lse_time = budget_s * 0.30
    lse_t1 = t0 + sa_time + lse_time
    lse_configs = [
        (50.0, 500.0, 0.12, 0.005, 2000, 0.3),
        (30.0, 300.0, 0.15, 0.008, 2000, 0.4),
        (100.0, 1000.0, 0.10, 0.003, 1500, 0.25),
        (20.0, 800.0, 0.15, 0.005, 2500, 0.35),
    ]
    for ci, (b_s, b_e, s_s, s_e, n_it, n_af) in enumerate(lse_configs):
        if time.time() > lse_t1:
            break
        dt = lse_t1 - time.time()
        if dt < 10:
            break
        h_lse, c5_lse = lse_subgradient_optimize(
            best_h, dx, target_sum, t0, dt,
            beta_start=b_s, beta_end=b_e,
            step_start=s_s, step_end=s_e,
            max_iter=n_it, n_active_frac=n_af)
        if c5_lse < best_c5:
            best_c5 = c5_lse
            best_h = h_lse

    # Phase 3: LP polish (25% of budget)
    lp_time = budget_s * 0.25
    lp_t1 = t0 + sa_time + lse_time + lp_time
    lp_configs = [
        (0.3, 0.30, 0.98, 80),
        (0.2, 0.40, 0.985, 100),
        (0.15, 0.25, 0.97, 60),
        (0.25, 0.45, 0.98, 80),
    ]
    for li, (a_s, n_af, a_d, m_ni) in enumerate(lp_configs):
        if time.time() > lp_t1:
            break
        dt = lp_t1 - time.time()
        if dt < 8:
            break
        h_lp, c5_lp = lp_polish(best_h, dx, target_sum, t0, dt,
                                alpha_start=a_s, n_active_frac=n_af,
                                alpha_decay=a_d, max_no_improve=m_ni)
        if c5_lp < best_c5:
            best_c5 = c5_lp
            best_h = h_lp

    # Phase 4: More SA from best (10% of budget)
    sa2_time = budget_s * 0.10
    sa2_t1 = t0 + budget_s - 30
    for si in range(4):
        if time.time() > sa2_t1:
            break
        dt = sa2_t1 - time.time()
        if dt < 10:
            break
        sigma = [0.01, 0.02, 0.015, 0.005][si]
        h_start = project_feasible(best_h + sigma * rng.randn(n), target_sum)
        h_sa, c5_sa = sa_continuous(h_start, dx, target_sum, t0, dt,
                                    seed=seed + 100 + si,
                                    t_start=0.001, cool=0.99999)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa

    # Phase 5: Final LP polish (remaining time)
    dt_final = t0 + budget_s - 10 - time.time()
    if dt_final > 5:
        h_lp, c5_lp = lp_polish(best_h, dx, target_sum, t0, dt_final,
                                alpha_start=0.1, n_active_frac=0.4,
                                alpha_decay=0.99, min_alpha=0.001, max_no_improve=50)
        if c5_lp < best_c5:
            best_c5 = c5_lp
            best_h = h_lp

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_feasible(best_h, n / 2.0)
    best_c5 = c5_of_h(best_h, dx)
    return best_h, best_c5, n
