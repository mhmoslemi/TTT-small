def project_feasible(x, target_sum):

    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(60):

        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid

    slack_start = n
    q_idx = n + K

def project_feasible(x, target_sum):
    n = len(x)

    for _ in range(60):

    offset = k - (n - 1)
    grad = np.zeros(n)

        b_ub[j] = -corrs[k]
    A_eq = np.zeros((1, n_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(-h[i]*alpha, (1.0-h[i])*alpha) for i in range(n)]
    bounds += [(0.0, None)] * K
    bounds.append((None, None))
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.status != 0:
            return None
        d = res.x[:n]
        if np.linalg.norm(d) < 1e-15:
            return None
        return d / np.linalg.norm(d)
    except Exception:
        return None

def solve_hard_lp(h, dx, active_idx, corrs, alpha, n):
    K = len(active_idx)
    if K == 0:
        return None
    n_vars = n + 1
    q_idx = n
    c_obj = np.zeros(n_vars)
    c_obj[q_idx] = 1.0
    A_ub = np.zeros((K, n_vars))
    b_ub = np.zeros(K)
    for j, k in enumerate(active_idx):
        g = grad_lag(h, k, n)
        A_ub[j, :n] = dx * g
        A_ub[j, q_idx] = -1.0
        b_ub[j] = -corrs[k]
    A_eq = np.zeros((1, n_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(-h[i]*alpha, (1.0-h[i])*alpha) for i in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.status != 0:
            return None
        d = res.x[:n]
        if np.linalg.norm(d) < 1e-15:
            return None
        return d / np.linalg.norm(d)
    except Exception:
        return None

def solve_weighted_lp(h, dx, active_idx, corrs, alpha, n):
    K = len(active_idx)
    if K == 0:
        return None
    w = corrs[active_idx].copy()
    w = np.maximum(w, 0)
    w_sum = w.sum()
    if w_sum < 1e-15:
        w = np.ones(K) / K
    else:
        w = w / w_sum
    c_obj = np.zeros(n)
    for j, k in enumerate(active_idx):
        c_obj += dx * w[j] * grad_lag(h, k, n)
    A_eq = np.zeros((1, n))
    A_eq[0, :] = 1.0
    b_eq = np.zeros(1)
    bounds = [(-h[i]*alpha, (1.0-h[i])*alpha) for i in range(n)]
    try:
        res = linprog(c_obj, A_eq=A_eq, b_eq=b_eq, bounds=bounds,
                      method='highs', options={'presolve': True})
        if res.status != 0:
            return None
        d = res.x
        if np.linalg.norm(d) < 1e-15:
            return None
        return d / np.linalg.norm(d)
    except Exception:
        return None

def line_search_accept(cur, cur_c5, direction, dx, target, step_size):
    best_h = cur
    best_c5 = cur_c5
    for t in [1.0, 0.8, 0.6, 0.5, 0.4, 0.3, 0.2, 0.15, 0.1, 0.07, 0.04, 0.02]:
        h_new = cur + t * step_size * direction
        h_new = project_feasible(h_new, target)
        c5_new = c5_of_h(h_new, dx)
        if c5_new < best_c5 - 1e-14:
            best_h = h_new
            best_c5 = c5_new
            break
    return best_h, best_c5

def lp_inner(h, dx, target, alpha, n_iter, budget_s, t0):
    n = len(h)
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    cur = h.copy()
    cur_c5 = best_c5
    cur_alpha = alpha
    stagnant = 0
    for it in range(n_iter):
        if budget_s - (time.time() - t0) < 5.0:
            break
        corrs = corr_array(cur)
        max_c = corrs.max()
        margin = max(0.01 * dx * n, 2e-6)
        threshold = max_c - margin
        active_mask = corrs >= threshold
        top20 = np.argsort(corrs)[-20:]
        active_mask[top20] = True
        active_idx = np.where(active_mask)[0]
        direction = None
        mode = it % 4
        if mode == 0:
            direction = solve_soft_lp(cur, dx, active_idx, corrs, cur_alpha, n)
        elif mode == 1:
            direction = solve_hard_lp(cur, dx, active_idx, corrs, cur_alpha, n)
        elif mode == 2:
            direction = solve_soft_lp(cur, dx, active_idx, corrs, cur_alpha * 0.6, n)
        else:
            direction = solve_weighted_lp(cur, dx, active_idx, corrs, cur_alpha, n)
        if direction is None:
            cur_alpha *= 0.7
            stagnant += 1
            if stagnant > 6:
                break
            continue
        new_h, new_c5 = line_search_accept(cur, cur_c5, direction, dx, target, cur_alpha)
        if new_c5 < cur_c5 - 1e-14:
            cur = new_h
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = new_h.copy()
            stagnant = 0
            cur_alpha = min(0.35, cur_alpha * 1.08)
        else:
            stagnant += 1
            cur_alpha *= 0.7
            if stagnant > 8:
                break
    return best_h, best_c5

def pair_move(h, dx, target, n, rng, n_pairs):
    cur_h = h.copy()
    cur_c5 = c5_of_h(h, dx)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for _ in range(n_pairs):
        p, q = rng.choice(n, size=2, replace=False)
        a, b = cur_h[p], cur_h[q]
        L = max(-a, b - 1.0)
        U = min(1.0 - a, b)
        if U - L < 1e-12:
            continue
        n_samp = 13
        t_vals = np.linspace(L, U, n_samp)
        best_t = 0.0
        best_val = cur_c5
        for t in t_vals:
            if abs(t) < 1e-14:
                continue
            cur_h[p] = a + t
            cur_h[q] = b - t
            val = c5_of_h(cur_h, dx)
            if val < best_val:
                best_val = val
                best_t = t
        if best_t == 0.0:
            cur_h[p] = a
            cur_h[q] = b
            continue
        idx = np.argmin(np.abs(t_vals - best_t))
        t_lo = t_vals[max(0, idx - 1)]
        t_hi = t_vals[min(n_samp - 1, idx + 1)]
        if t_hi - t_lo < 1e-12:
            cur_h[p] = a + best_t
            cur_h[q] = b - best_t
            continue
        for _ in range(28):
            t1 = t_hi - gr * (t_hi - t_lo)
            t2 = t_lo + gr * (t_hi - t_lo)
            cur_h[p] = a + t1
            cur_h[q] = b - t1
            f1 = c5_of_h(cur_h, dx)
            cur_h[p] = a + t2
            cur_h[q] = b - t2
            f2 = c5_of_h(cur_h, dx)
            if f1 < f2:
                t_hi = t2
            else:
                t_lo = t1
        t_best = (t_lo + t_hi) / 2.0
        cur_h[p] = a + t_best
        cur_h[q] = b - t_best
        final_val = c5_of_h(cur_h, dx)
        if final_val < cur_c5 - 1e-12:
            cur_c5 = final_val
        else:
            cur_h[p] = a
            cur_h[q] = b
    return cur_h, cur_c5

def neighbor_pair_move(h, dx, target, n, rng, n_pairs):
    cur_h = h.copy()
    cur_c5 = c5_of_h(h, dx)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for _ in range(n_pairs):
        p = rng.randint(n)
        if rng.rand() < 0.5:
            q = (p + 1) % n
        else:
            q = (p - 1) % n
        a, b = cur_h[p], cur_h[q]
        L = max(-a, b - 1.0)
        U = min(1.0 - a, b)
        if U - L < 1e-12:
            continue
        n_samp = 11
        t_vals = np.linspace(L, U, n_samp)
        best_t = 0.0
        best_val = cur_c5
        for t in t_vals:
            if abs(t) < 1e-14:
                continue
            cur_h[p] = a + t
            cur_h[q] = b - t
            val = c5_of_h(cur_h, dx)
            if val < best_val:
                best_val = val
                best_t = t
        if best_t == 0.0:
            cur_h[p] = a
            cur_h[q] = b
            continue
        idx = np.argmin(np.abs(t_vals - best_t))
        t_lo = t_vals[max(0, idx - 1)]
        t_hi = t_vals[min(n_samp - 1, idx + 1)]
        for _ in range(22):
            t1 = t_hi - gr * (t_hi - t_lo)
            t2 = t_lo + gr * (t_hi - t_lo)
            cur_h[p] = a + t1
            cur_h[q] = b - t1
            f1 = c5_of_h(cur_h, dx)
            cur_h[p] = a + t2
            cur_h[q] = b - t2
            f2 = c5_of_h(cur_h, dx)
            if f1 < f2:
                t_hi = t2
            else:
                t_lo = t1
        t_best = (t_lo + t_hi) / 2.0
        cur_h[p] = a + t_best
        cur_h[q] = b - t_best
        final_val = c5_of_h(cur_h, dx)
        if final_val < cur_c5 - 1e-12:
            cur_c5 = final_val
        else:
            cur_h[p] = a
            cur_h[q] = b
    return cur_h, cur_c5

def perturb_reopt(h, dx, target, rng, n_rounds, base_alpha, budget_s, t0):
    n = len(h)
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    for r in range(n_rounds):
        if budget_s - (time.time() - t0) < 20.0:
            break
        scale = max(0.005, base_alpha / (r + 1))
        pert = rng.standard_normal(n)
        pert -= pert.mean()
        h_p = h + scale * pert
        h_p = project_feasible(h_p, target)
        h_opt, c5_opt = lp_inner(h_p, dx, target, base_alpha, 60, budget_s, t0)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            h = h_opt.copy()
    return best_h, best_c5

def transfer_grid(h_src, n_target):
    n_src = len(h_src)
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_tgt = (np.arange(n_target) + 0.5) * (2.0 / n_target)
    return np.interp(x_tgt, x_src, h_src)

def binary_round(h, target, rng):
    n = len(h)
    h_bin = (h > 0.5).astype(float)
    diff = target - h_bin.sum()
    if diff > 0:
        frac = np.argsort(h)[::-1]
        count = int(diff)
        for i in range(min(count, n)):
            h_bin[frac[i]] = 1.0
        if count >= n:
            pass
    elif diff < 0:
        frac = np.argsort(h)
        count = int(-diff)
        for i in range(min(count, n)):
            h_bin[frac[i]] = 0.0
    h_bin = project_feasible(h_bin, target)
    return h_bin

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    
    def tl():
        return budget_s - (time.time() - t0)
    
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = np.full(128, 0.5)
    
    n = len(h0)
    dx = 2.0 / n
    target_sum = n / 2.0
    h0 = project_feasible(h0, target_sum)
    best_h = h0.copy()
    best_c5 = c5_of_h(h0, dx)
    
    # Phase 1: LP polish with multiple alpha levels
    for alpha_init in [0.25, 0.15, 0.1, 0.07, 0.05, 0.03]:
        if tl() < 25:
            break
        h_new, c5_new = lp_inner(best_h, dx, target_sum, alpha_init, 150, budget_s, t0)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new
    
    # Phase 2: Intensive pair moves
    if tl() > 80:
        h_pm, c5_pm = pair_move(best_h, dx, target_sum, n, rng, 2000)
        if c5_pm < best_c5:
            best_c5 = c5_pm
            best_h = h_pm
        if tl() > 30:
            h_new, c5_new = lp_inner(best_h, dx, target_sum, 0.08, 100, budget_s, t0)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new
    
    # Phase 3: Neighbor pair moves
    if tl() > 60:
        h_nm, c5_nm = neighbor_pair_move(best_h, dx, target_sum, n, rng, 3000)
        if c5_nm < best_c5:
            best_c5 = c5_nm
            best_h = h_nm
        if tl() > 20:
            h_new, c5_new = lp_inner(best_h, dx, target_sum, 0.06, 80, budget_s, t0)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new
    
    # Phase 4: More pair moves
    if tl() > 80:
        h_pm2, c5_pm2 = pair_move(best_h, dx, target_sum, n, rng, 3000)
        if c5_pm2 < best_c5:
            best_c5 = c5_pm2
            best_h = h_pm2
    
    # Phase 5: Binary rounding + re-optimize
    if tl() > 60:
        h_bin = binary_round(best_h, target_sum, rng)
        c5_bin = c5_of_h(h_bin, dx)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin
        # Re-optimize from binary
        h_opt, c5_opt = lp_inner(h_bin, dx, target_sum, 0.15, 100, budget_s, t0)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
        h_pr, c5_pr = pair_move(best_h, dx, target_sum, n, rng, 1000)
        if c5_pr < best_c5:
            best_c5 = c5_pr
            best_h = h_pr
    
    # Phase 6: Perturbation + re-optimization
    if tl() > 80:
        h_pert, c5_pert = perturb_reopt(best_h, dx, target_sum, rng, 8, 0.12, budget_s, t0)
        if c5_pert < best_c5:
            best_c5 = c5_pert
            best_h = h_pert
        if tl() > 30:
            h_new, c5_new = lp_inner(best_h, dx, target_sum, 0.06, 100, budget_s, t0)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new
            h_pr, c5_pr = pair_move(best_h, dx, target_sum, n, rng, 1500)
            if c5_pr < best_c5:
                best_c5 = c5_pr
                best_h = h_pr
    
    # Phase 7: Multi-resolution exploration
    n_base = len(best_h)
    candidates = set()
    for dn in range(-32, 33, 4):
        nt = n_base + dn
        if 48 <= nt <= 200 and nt % 2 == 0 and nt != n_base:
            candidates.add(nt)
    for fixed in [80, 96, 104, 112, 120, 128, 136, 144, 152, 160, 168, 176, 184, 192]:
        if fixed != n_base:
            candidates.add(fixed)
    candidates = sorted(candidates)
    
    for n_try in candidates:
        if tl() < 35:
            break
        dx_t = 2.0 / n_try
        ts_t = n_try / 2.0
        h_interp = transfer_grid(best_h, n_try)
        h_interp = project_feasible(h_interp, ts_t)
        h_opt, c5_opt = lp_inner(h_interp, dx_t, ts_t, 0.15, 80, budget_s, t0)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
            n_base = n_try
            if tl() > 30:
                h_pr, c5_pr = pair_move(best_h, dx_t, ts_t, n_try, rng, 1500)
                if c5_pr < best_c5:
                    best_c5 = c5_pr
                    best_h = h_pr
                    h_new, c5_new = lp_inner(best_h, dx_t, ts_t, 0.08, 60, budget_s, t0)
                    if c5_new < best_c5:

                        best_h = h_new
    
    # Phase 8: Final intensive polish on best grid
    n_b = len(best_h)
    dx_b = 2.0 / n_b

            best_c5 = c5_f2
            best_h = h_f2
    
    if tl() > 30:
        h_nm2, c5_nm2 = neighbor_pair_move(best_h, dx_b, ts_b, n_b, rng, 3000)
        if c5_nm2 < best_c5:
            best_c5 = c5_nm2
            best_h = h_nm2
            if tl() > 15:
                h_f3, c5_f3 = lp_inner(best_h, dx_b, ts_b, 0.02, 100, budget_s, t0)
                if c5_f3 < best_c5:

                    best_h = h_f3

    if tl() > 40:
        h_pert2, c5_pert2 = perturb_reopt(best_h, dx_b, ts_b, rng, 6, 0.05, budget_s, t0)

    if tl() > 20:

    best_h = project_feasible(best_h, ts_final)
    best_c5 = c5_of_h(best_h, dx_final)
    
    return best_h, best_c5, n_final
