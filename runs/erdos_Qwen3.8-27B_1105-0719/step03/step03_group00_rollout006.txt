def project_feasible(z, target):
    n = len(z)
    lo = -2.0
    hi = 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        h = np.clip(z - mid, 0.0, 1.0)
        if h.sum() > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def c5_of_h(h, dx):
    return np.correlate(h, 1.0 - h, mode='full').max() * dx

def corr_of_h(h):
    return np.correlate(h, 1.0 - h, mode='full')

def grad_lag(h, k, n):
    off = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -off)
    i_hi = min(n - 1, n - 1 - off)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx + off] += (1.0 - h[idx])
    grad[idx] += (-h[idx + off])
    return grad

def solve_lp_soft(h, dx, top_idx, n, alpha, rho_q, rho_s, corr_vals):
    """LP with soft slacks for selected lags."""
    M = len(top_idx)
    # Variables: delta (n), s (M), q (1) -> total n + M + 1
    c_obj = np.zeros(n + M + 1)
    c_obj[-1] = rho_q
    c_obj[n:n+M] = rho_s  # slack penalties
    
    # Constraints: dx * grad_k · delta + s_j - q <= -r_{a_j}
    # Rearranged: dx * grad_k · delta + s_j - q + r_{a_j} <= 0
    A = np.zeros((M, n + M + 1))
    b = np.zeros(M)
    for j, k in enumerate(top_idx):
        g = grad_lag(h, k, n)
        A[j, :n] = dx * g
        A[j, n + j] = 1.0   # s_j
        A[j, -1] = -1.0      # -q
        b[j] = -corr_vals[k]
    
    # Equality: sum(delta) = 0
    A_eq = np.zeros((1, n + M + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    bounds = [(-h[i] * alpha, (1.0 - h[i]) * alpha) for i in range(n)]
    bounds += [(0.0, None)] * M
    bounds.append((0.0, None))
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.status != 0:
            return None
        return res.x[:n]
    except Exception:
        return None

import numpy as np
from scipy.optimize import linprog
import time

def project_feasible(z, target):
    n = len(z)
    lo = -2.0
    hi = 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        h = np.clip(z - mid, 0.0, 1.0)
        if h.sum() > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def c5_of_h(h, dx):
    return np.correlate(h, 1.0 - h, mode='full').max() * dx

def corr_of_h(h):
    return np.correlate(h, 1.0 - h, mode='full')

def grad_lag(h, k, n):
    off = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -off)
    i_hi = min(n - 1, n - 1 - off)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx + off] += (1.0 - h[idx])
    grad[idx] += (-h[idx + off])
    return grad

def solve_lp_soft(h, dx, top_idx, n, alpha, rho_q, rho_s, corr_vals):
    M = len(top_idx)
    c_obj = np.zeros(n + M + 1)
    c_obj[-1] = rho_q
    c_obj[n:n+M] = rho_s
    A = np.zeros((M, n + M + 1))
    b = np.zeros(M)
    for j, k in enumerate(top_idx):
        g = grad_lag(h, k, n)
        A[j, :n] = dx * g
        A[j, n + j] = 1.0
        A[j, -1] = -1.0
        b[j] = -corr_vals[k]
    A_eq = np.zeros((1, n + M + 1))
    A_eq[0, :n] = 1.0
    bounds = [(-h[i] * alpha, (1.0 - h[i]) * alpha) for i in range(n)]
    bounds += [(0.0, None)] * M
    bounds.append((0.0, None))
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=np.zeros(1),
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.status != 0:
            return None
        return res.x[:n]
    except Exception:
        return None

def solve_lp_hard(h, dx, top_idx, n, alpha, corr_vals):
    M = len(top_idx)
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A = np.zeros((M, n + 1))
    b = np.zeros(M)
    for j, k in enumerate(top_idx):
        g = grad_lag(h, k, n)
        A[j, :n] = dx * g
        A[j, -1] = -1.0
        b[j] = -corr_vals[k]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    bounds = [(-h[i] * alpha, (1.0 - h[i]) * alpha) for i in range(n)]
    bounds.append((0.0, None))
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=np.zeros(1),
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.status != 0:
            return None
        return res.x[:n]
    except Exception:
        return None

def smooth_gradient_direction(h, dx, corr_vals, lam):
    n = len(h)
    max_r = corr_vals.max()
    if max_r <= 0:
        return None
    weights = np.exp(lam * (corr_vals - max_r) / max_r)
    total_w = weights.sum()
    if total_w < 1e-10:
        return None
    weights = weights / total_w
    grad = np.zeros(n)
    for k in range(len(corr_vals)):
        if weights[k] > 1e-8:
            grad += weights[k] * grad_lag(h, k, n)
    grad -= grad.mean()
    return grad

def line_search(cur, direction, dx, target, cur_c5):
    best_h = cur
    best_c5 = cur_c5
    scales = [1.0, 0.8, 0.6, 0.5, 0.4, 0.3, 0.2, 0.15, 0.1, 0.06, 0.04, 0.02]

            best_h = h_try

def greedy_pair_refine(h, dx, target, max_rounds, budget_s, t0):

            break

        top_k = np.argsort(corr)[-25:]

        for k in top_k:
            off = k - (n - 1)
            i_lo = max(0, -off)
            i_hi = min(n - 1, n - 1 - off)
            if i_lo <= i_hi:
                active_coords.update(range(i_lo, min(i_hi + 1, n)))
                if off != 0:
                    active_coords.update(range(max(0, i_lo + off), min(n, i_hi + off + 1)))
        a_list = sorted(active_coords)[:35]
        if len(a_list) < 2:
            break
        improved = False
        for p_i in range(len(a_list)):
            if budget_s - (time.time() - t0) < 3.0:

            for q_i in range(p_i + 1, len(a_list)):
                q = a_list[q_i]
                old_p = best_h[p]
                old_q = best_h[q]
                t_lo = max(-old_p, old_q - 1.0) + 1e-8
                t_hi = min(1.0 - old_p, old_q) - 1e-8
                if t_hi <= t_lo:
                    continue
                t_vals = np.linspace(t_lo, t_hi, 13)
                best_t = 0.0
                best_c5_local = best_c5
                for t in t_vals:
                    best_h[p] = old_p + t
                    best_h[q] = old_q - t
                    c5_try = c5_of_h(best_h, dx)
                    if c5_try < best_c5_local:
                        best_c5_local = c5_try

                if best_t != 0.0:
                    best_h[p] = old_p + best_t
                    best_h[q] = old_q - best_t
                    best_c5 = best_c5_local

                else:
