import cvxpy as cp

def qp_trust_region(h, profile, G, C, active_indices, radius, gamma):
    n = len(h)
    delta = cp.Variable(n)
    r = cp.Variable()
    
    # Objective: minimize r + gamma * ||delta||^2
    obj = cp.Minimize(r + gamma * cp.sum_squares(delta))
    
    # Constraints
    constraints = [
        r >= C - 2 * radius,  # r must be at least somewhat below C
        cp.sum(delta) == 0,
        delta >= -radius,
        delta <= radius,
        h + delta >= 0,
        h + delta <= 1,
    ]
    
    # Linearized profile constraints
    G_a = G[active_indices, :]
    P_a = profile[active_indices]
    constraints.append(G_a @ delta <= r - P_a)
    
    cp.Problem(obj, constraints).solve(solver=cp.MOSEK)
    
    if delta.value is not None:
        return delta.value
    return None

import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def repair_feasible(h):
    n = len(h)
    h = np.clip(np.asarray(h, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, 1.0 - h[idx])
            h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)[::-1]
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    # Final fix
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def build_gradient(h):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    G = np.zeros((M, n))
    center = n - 1
    
    # Zero lag
    G[center, :] = dx * (1.0 - 2.0 * h)
    
    # Positive lags d = 1, ..., n-1
    for d in range(1, n):
        row = np.zeros(n)
        # First index contribution: (1-h[q-d]) for q in [d, n-1]
        row[d:] = (1.0 - h[:n-d])
        # Second index contribution: -h[q+d] for q in [0, n-1-d]
        row[:n-d] -= h[d:]
        G[center + d, :] = dx * row
    
    # Negative lags d = -1, ..., -(n-1), i.e., e = 1, ..., n-1
    for e in range(1, n):
        row = np.zeros(n)
        # First index contribution: (1-h[q+e]) for q in [0, n-1-e]
        row[:n-e] = (1.0 - h[e:])
        # Second index contribution: -h[q-e] for q in [e, n-1]
        row[e:] -= h[:n-e]
        G[center - e, :] = dx * row
    
    return G

def get_active_set(profile, top_k=15, beam=None):
    T = np.max(profile)
    if beam is None:
        spread = T - np.median(profile)
        beam = max(0.003, min(0.015, 0.15 * spread))
    thresh = T - beam
    active = np.where(profile >= thresh)[0]
    if len(active) == 0:
        active = np.array([np.argmax(profile)])
    if len(active) > top_k:
        top_idx = np.argsort(profile[active])[-top_k:]
        active = active[top_idx]
    return active, T

def solve_qp(h, G, profile, active, radius, gamma):
    n = len(h)
    C = np.max(profile)
    
    delta = cp.Variable(n)
    r = cp.Variable()
    
    obj = cp.Minimize(r + gamma * cp.sum_squares(delta))
    
    constraints = [
        cp.sum(delta) == 0,
        delta >= -radius,
        delta <= radius,
    ]
    
    # Keep h + delta in [0, 1]
    lower_h = np.maximum(-radius, -h)
    upper_h = np.minimum(radius, 1.0 - h)
    constraints.append(delta >= lower_h)
    constraints.append(delta <= upper_h)
    
    # Linearized constraints: G[active] @ delta <= r - P[active]
    G_a = G[active, :]
    P_a = profile[active]
    constraints.append(G_a @ delta <= r - P_a)
    
    # r lower bound to encourage decrease
    r_lower = max(0.0, C - 2.0 * radius * 10)
    constraints.append(r >= r_lower)
    
    prob = cp.Problem(obj, constraints)
    
    # Try solvers in order
    for solver in [cp.MOSEK, cp.ECOS, cp.CVXOPT, cp.PDLP]:
        try:
            prob.solve(solver=solver)
            if prob.status == cp.OPTIMAL and delta.value is not None:
                return delta.value
        except Exception:
            continue
    
    return None

def line_search(h, direction, current_score, max_step=1.0):
    n = len(h)
    best_h = h.copy()
    best_s = current_score
    alphas = [max_step, 0.7*max_step, 0.5*max_step, 0.3*max_step,
              0.2*max_step, 0.1*max_step, 0.05*max_step, 0.02*max_step]
    for a in alphas:
        h_try = repair_feasible(h + a * direction)
        s = exact_score(h_try)
        if s < best_s - 1e-14:
            best_h = h_try
            best_s = s
    return best_h, best_s

def optimize_qp(h, t_start, budget_s, max_iters=300):

    h = h.copy()
    best_h = h.copy()
    best_c = exact_score(h)
    radius = 0.08
    gamma = 5.0
    no_improve = 0
    
    for it in range(max_iters):
        elapsed = time.perf_counter() - t_start
        if budget_s - elapsed < 10:
            break
        
        profile = compute_profile(h)
        c = float(np.max(profile))
        
        if c < best_c - 1e-14:
            best_c = c
            best_h = h.copy()
            no_improve = 0
        
        # Get active set
        active, T = get_active_set(profile, top_k=20, beam=None)
        
        # Build gradient
        G = build_gradient(h)
        
        # Solve QP
        delta = solve_qp(h, G, profile, active, radius, gamma)
        
        if delta is not None and np.linalg.norm(delta) > 1e-12:
            h_new, s_new = line_search(h, delta, c, max_step=1.0)
            if s_new < c - 1e-14:
                h = h_new
                no_improve = 0
                radius = min(radius * 1.1, 0.15)
                continue
        
        # Try with larger active set
        if no_improve > 5:
            active2, _ = get_active_set(profile, top_k=30, beam=0.03)
            delta2 = solve_qp(h, G, profile, active2, radius * 0.5, gamma)
            if delta2 is not None and np.linalg.norm(delta2) > 1e-12:
                h_new, s_new = line_search(h, delta2, c, max_step=0.5)
                if s_new < c - 1e-14:
                    h = h_new
                    no_improve = 0
                    continue
        
        # Try gradient-ascent direction (weighted)
        if no_improve > 10:
            weights = np.maximum(profile[active] - 0.95 * T, 0.01)
            weights /= weights.sum()
            grad = G[active, :].T @ weights
            grad -= grad.mean()
            norm = np.linalg.norm(grad)
            if norm > 1e-14:
                grad /= norm
                h_new, s_new = line_search(h, -grad, c, max_step=0.03)
                if s_new < c - 1e-14:
                    h = h_new
                    no_improve = 0
                    continue
        
        no_improve += 1
        radius *= 0.85
        gamma *= 1.1
        
        if no_improve > 40 or radius < 0.002:
            break
    
    return best_h, best_c

def symmetrize(h):
    return (h + h[::-1]) / 2.0

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

def compute_profile(h):

            remaining -= add

        remaining = -diff

        row[d:] = (1.0 - h[:n-d])

        G[center + d, :] = dx * row
        row2 = np.zeros(n)
        row2[:n-d] = (1.0 - h[d:])
        row2[d:] -= h[:n-d]

import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time


def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))


def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx


def repair_feasible(h):
    n = len(h)
    h = np.clip(np.asarray(h, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, 1.0 - h[idx])
            h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)[::-1]
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h


def build_gradient(h):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    G = np.zeros((M, n))
    center = n - 1
    G[center, :] = dx * (1.0 - 2.0 * h)
    for d in range(1, n):
        row = np.zeros(n)
        row[d:] = (1.0 - h[:n - d])
        row[:n - d] -= h[d:]
        G[center + d, :] = dx * row
        row2 = np.zeros(n)
        row2[:n - d] = (1.0 - h[d:])
        row2[d:] -= h[:n - d]
        G[center - d, :] = dx * row2
    return G


def get_active(profile, top_k=20, beam=None):
    T = float(np.max(profile))
    if beam is None:
        spread = T - float(np.median(profile))
        beam = max(0.003, 0.12 * spread)
    thresh = T - beam
    active = np.where(profile >= thresh)[0]
    if len(active) == 0:
        active = np.array([np.argmax(profile)])
    if len(active) > top_k:
        top_idx = np.argsort(profile[active])[-top_k:]
        active = active[top_idx]
    return active


def solve_qp(h, G, profile, active, radius, gamma):
    n = len(h)
    C = float(np.max(profile))
    delta = cp.Variable(n)
    r = cp.Variable()
    G_a = G[active, :]
    P_a = profile[active]
    lower = np.maximum(-radius, -h)
    upper = np.minimum(radius, 1.0 - h)
    obj = cp.Minimize(r + gamma * cp.sum_squares(delta))
    cons = [
        cp.sum(delta) == 0,
        delta >= lower,
        delta <= upper,
        G_a @ delta <= r - P_a,
        r >= 0.0,
    ]
    prob = cp.Problem(obj, cons)
    for solver in [cp.MOSEK, cp.ECOS, cp.CVXOPT]:
        try:
            prob.solve(solver=solver)
            if prob.status == cp.OPTIMAL and delta.value is not None:
                return np.asarray(delta.value, dtype=np.float64)
        except Exception:
            continue
    return None


def line_search(h, direction, current_score, max_step=1.0):
    best_h = h.copy()
    best_s = current_score
    alphas = [max_step, 0.7 * max_step, 0.5 * max_step, 0.35 * max_step,
              0.2 * max_step, 0.1 * max_step, 0.05 * max_step, 0.02 * max_step]
    for a in alphas:
        h_try = repair_feasible(h + a * direction)
        s = exact_score(h_try)
        if s < best_s - 1e-14:
            best_h = h_try
            best_s = s
    return best_h, best_s

            weights = np.maximum(profile[active3] - 0.9 * float(np.max(profile)), 0.01)
            weights /= weights.sum()
            grad = G[active3, :].T @ weights
            grad -= grad.mean()
            ng = np.linalg.norm(grad)
            if ng > 1e-14:
                grad /= ng
                h_new, s_new = line_search(h, -grad, c, max_step=0.05)
                if s_new < c - 1e-14:
                    h = h_new
                    no_improve = 0
                    accepted = True

        if not accepted:
            no_improve += 1
            radius *= 0.8
            gamma = min(gamma * 1.15, 50.0)

    return best_h, best_c


def slsqp_polish(h_init, t_start, budget_s, maxiter=200):
    n = len(h_init)
    M = 2 * n - 1
    c0 = exact_score(h_init)
    x0 = np.concatenate([h_init, [c0 + 1e-5]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    def obj_fn(x):
        return x[-1]

    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g

    def eq_fn(x):
        return x[:-1].sum() - n / 2.0

    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g

    def ineq_fn(x):
        h = x[:-1]
        t = x[-1]
        prof = compute_profile(h)
        return t - prof

    def ineq_jac(x):
        h = x[:-1]
        G = build_gradient(h)
        J = np.zeros((M, n + 1))
        J[:, :-1] = -G
        J[:, -1] = 1.0
        return J

    cons = [
        {'type': 'eq', 'fun': eq_fn, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fn, 'jac': ineq_jac}
    ]
    try:
        res = minimize(obj_fn, x0, jac=obj_jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-13})
        if res.x is not None and np.isfinite(res.x).all():
            h_sol = repair_feasible(res.x[:-1])
            c_sol = exact_score(h_sol)
            return h_sol, c_sol
    except Exception:
        pass
    return None, None


def symmetrize(h):
    return (h + h[::-1]) / 2.0


def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)


def make_random_start(n, rng, base_h=None):
    h = np.zeros(n)
    if base_h is not None:
        h = interpolate_to_n(base_h, n).copy()
        h += 0.08 * rng.randn(n)
    else:
        h[:] = 0.5 + 0.3 * rng.randn(n)
    return repair_feasible(h)
