import numpy as np
from scipy.optimize import minimize, linprog
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full"))) * dx

def profile_h(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def repair_feasible(vec):
    """Project onto {h : 0 <= h <= 1, sum(h) = n/2} using water-filling."""
    n = len(vec)
    target = n / 2.0
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    
    # Water-filling: find lambda such that sum(clamp(vec + lambda, 0, 1)) = target
    lo, hi = -1.0, 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(vec + mid, 0.0, 1.0))
        if s > target:
            hi = mid
        else:
            lo = mid
    h = np.clip(vec + (lo + hi) / 2.0, 0.0, 1.0)
    
    # Final correction
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0 - 1e-12
            if mask.any():
                add = min(diff, mask.sum() * (1.0 - h[mask]).max())
                h[mask] += add / mask.sum()
        else:
            mask = h > 1e-12
            if mask.any():
                rem = min(-diff, h[mask].sum())
                h[mask] -= rem / mask.sum()
    h = np.clip(h, 0.0, 1.0)
    return h

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full"))) * dx

def profile_h(h):

    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx])
            h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            mask = h < 1.0 - 1e-10
            if mask.any():
                h[mask] += diff2 / mask.sum()
        else:
            mask = h > 1e-10
            if mask.any():
                h[mask] += diff2 / mask.sum()
    return np.clip(h, 0.0, 1.0)

def compute_gradients(h, dx):
    n = len(h)
    M = 2 * n - 1
    G = np.zeros((M, n))
    G[n - 1] = dx * (1.0 - 2.0 * h)

def nesterov_smooth(h0, beta_schedule, n_total=200, t_start=None, budget_s=1100, margin=30):

    t_val = 1.0
    M = 2 * n - 1

    for it in range(total_steps):
        if t_start is not None and time.perf_counter() - t_start > budget_s - margin:
            break
        frac = it / float(total_steps)
        beta = beta_schedule[0] + (beta_schedule[1] - beta_schedule[0]) * frac
        P = profile_h(h)
        Pmax = P.max()
        exp_arg = beta * (P - Pmax)
        np.clip(exp_arg, -500, 500, out=exp_arg)
        weights = np.exp(exp_arg)
        S = weights.sum()
        if S < 1e-300:
            weights = np.zeros(M)
            weights[np.argmax(P)] = 1.0
        else:
            weights /= S
        G = compute_gradients(h, dx)
        grad = G.T @ weights
        lr = 0.15 / sqrt_beta(beta, n, dx)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0

        y = h + momentum * (h - prev_h)

        if c_new < best_c:

        prev_h = h
        t_val = t_new
        h = h_new
    
    return h, best_c

def sqrt_beta(beta, n, dx):
    return max(beta * dx * n * 2.0, 1e-6)

def beatty_h(n):
    h = np.zeros(n)

    k = n // 2
    for i in range(k):
        pos = int((i + 0.5) * alpha * n) % n
        h[pos] = 1.0
    return repair_feasible(h)

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    return repair_feasible(np.clip(h_new, 0.0, 1.0))

        if h_lp is None:
            break
        h_lp = np.clip(h_lp, 0.0, 1.0)
        h_lp = repair_feasible(h_lp)
        improved = False
        for alpha in [1.0, 0.6, 0.4, 0.25, 0.15, 0.08, 0.04]:
            h_try = h + alpha * (h_lp - h)
            h_try = repair_feasible(h_try)
            c_try = exact_score(h_try)
            if c_try < best_c - 1e-15:
                h = h_try
                best_c = c_try
                improved = True
                break
        if not improved:
            if it > 3:
                break
    
    return h, best_c

def pair_correction(h, max_pairs=200, t_start=None, budget_s=1100, margin=30):
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    
    P = profile_h(h)
    c_max = P.max()
    bottleneck = np.where(P >= c_max - 0.001)[0]
    
    for _ in range(max_pairs):
        if t_start is not None and time.perf_counter() - t_start > budget_s - margin:
            break
        improved = False
        for bi in bottleneck[:5]:
            m = bi - (n - 1)
            if m == 0:
                continue
            abs_m = abs(m)
            for i in range(min(n - abs_m, 30)):
                j = i + abs_m
                if m > 0:
                    ii, jj = j, i
                else:
                    ii, jj = i, j
                v1, v2 = h[ii], h[jj]
                if abs(v1 - v2) < 1e-6:
                    continue
                for frac in [0.3, 0.15, 0.08, 0.04]:
                    delta = (v2 - v1) * frac
                    h_try = h.copy()
                    h_try[ii] = v1 + delta
                    h_try[jj] = v2 - delta
                    if h_try[ii] < -1e-10 or h_try[ii] > 1 + 1e-10:
                        continue
                    if h_try[jj] < -1e-10 or h_try[jj] > 1 + 1e-10:
                        continue
                    h_try = repair_feasible(h_try)
                    c_try = exact_score(h_try)
                    if c_try < best_c - 1e-14:
                        h = h_try
                        best_c = c_try
                        improved = True
                        break
        if not improved:
            break
    
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()
    rng = np.random.RandomState(seed)
    
    def time_ok(margin=20):
        return time.perf_counter() - t_start < budget_s - margin
    
    parent = None
    try:
        parent = np.array(initial_h_values, dtype=np.float64)
    except Exception:
        pass
    
    if parent is None or len(parent) < 4:
        parent = np.full(70, 0.5)
    
    parent = np.clip(parent, 0.0, 1.0)
    parent = repair_feasible(parent)
    
    best_h = parent.copy()
    best_c = exact_score(parent)
    
    # Phase 1: SLP from parent
    if time_ok(80):
        best_h, best_c = slp_optimize(best_h, max_iters=40, t_start=t_start,
                                       budget_s=budget_s, margin=40)
    
    # Phase 2: Active SLP for polishing
    if time_ok(60):
        best_h, best_c = active_slp(best_h, active_k=12, max_iters=30,
                                     t_start=t_start, budget_s=budget_s, margin=40)
    
    # Phase 3: Nesterov on soft-max
    if time_ok(80):
        h_nes, c_nes = nesterov_smooth(best_h, [50, 300], n_total=150,
                                        t_start=t_start, budget_s=budget_s, margin=40)
        if c_nes < best_c - 1e-13:
            best_h = h_nes
            best_c = c_nes
    
    # Phase 4: SLP polish after Nesterov
    if time_ok(60):
        best_h, best_c = slp_optimize(best_h, max_iters=30, t_start=t_start,
                                       budget_s=budget_s, margin=40)
    
    # Phase 5: Pair correction
    if time_ok(40):
        best_h, best_c = pair_correction(best_h, max_pairs=100,
                                          t_start=t_start, budget_s=budget_s, margin=30)
    
    # Phase 6: Try Beatty and other structured starts
    if time_ok(120):
        n_cur = len(best_h)
        structured_starts = []
        # Beatty
        h_b = beatty_h(n_cur)
        structured_starts.append(h_b)
        # Perturbed Beatty
        z = rng.randn(n_cur)
        z -= z.mean()
        structured_starts.append(repair_feasible(0.7 * h_b + 0.3 * best_h + 0.02 * z))
        # Golden ratio direct
        alpha = (np.sqrt(5.0) + 1.0) / 2.0
        h_g = np.zeros(n_cur)
        for i in range(n_cur // 2):
            pos = int(i / alpha) % n_cur
            h_g[pos] = 1.0
        structured_starts.append(repair_feasible(h_g))
        
        for h0 in structured_starts:
            if not time_ok(30):
                break
            h_res, c_res = slp_optimize(h0, max_iters=25, t_start=t_start,
                                         budget_s=budget_s, margin=30)
            if c_res < best_c - 1e-12:
                best_h = h_res
                best_c = c_res
            h_res2, c_res2 = active_slp(h_res, active_k=10, max_iters=20,
                                         t_start=t_start, budget_s=budget_s, margin=30)
            if c_res2 < best_c - 1e-12:
                best_h = h_res2
                best_c = c_res2
    
    # Phase 7: Dimension exploration
    if time_ok(150):
        n_cur = len(best_h)
        dims_to_try = []
        for d in range(max(40, n_cur - 8), min(110, n_cur + 15)):
            if d != n_cur and d % 2 == 0:
                dims_to_try.append(d)
        # Sort by proximity to current
        dims_to_try.sort(key=lambda d: abs(d - n_cur))
        
        for new_n in dims_to_try[:6]:
            if not time_ok(30):
                break
            h_interp = interpolate_to_n(best_h, new_n)
            h_res, c_res = slp_optimize(h_interp, max_iters=25,
                                         t_start=t_start, budget_s=budget_s, margin=30)
            if c_res < best_c - 1e-11:
                best_h = h_res
                best_c = c_res
                # Polish
                h_res2, c_res2 = active_slp(h_res, active_k=12, max_iters=25,
                                             t_start=t_start, budget_s=budget_s, margin=30)
                if c_res2 < best_c - 1e-12:
                    best_h = h_res2
                    best_c = c_res2
                h_res3, c_res3 = slp_optimize(h_res2, max_iters=20,
                                               t_start=t_start, budget_s=budget_s, margin=30)
                if c_res3 < best_c - 1e-13:
                    best_h = h_res3
                    best_c = c_res3
    
    # Phase 8: Random restarts from diverse starts
    if time_ok(100):
        n_cur = len(best_h)
        for trial in range(5):
            if not time_ok(20):
                break
            # Random binary
            h_bin = random_binary_h(n_cur, np.random.RandomState(seed + 500 + trial))
            # Blend with best
            for mix in [0.5, 0.7]:
                if not time_ok(15):
                    break
                h_mix = repair_feasible(mix * h_bin + (1 - mix) * best_h)
                h_res, c_res = slp_optimize(h_mix, max_iters=20,
                                             t_start=t_start, budget_s=budget_s, margin=20)
                if c_res < best_c - 1e-11:
                    best_h = h_res
                    best_c = c_res
    
    # Phase 9: Final SLP + active SLP + pair correction
    if time_ok(60):
        best_h, best_c = slp_optimize(best_h, max_iters=30, t_start=t_start,
                                       budget_s=budget_s, margin=30)
    if time_ok(30):
        best_h, best_c = active_slp(best_h, active_k=15, max_iters=25,
                                     t_start=t_start, budget_s=budget_s, margin=20)
    if time_ok(20):
        best_h, best_c = pair_correction(best_h, max_pairs=100,
                                          t_start=t_start, budget_s=budget_s, margin=10)
    
    # Final: one more SLP pass
    if time_ok(25):
        best_h, best_c = slp_optimize(best_h, max_iters=20, t_start=t_start,
                                       budget_s=budget_s, margin=5)
    
    # Final validation
    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)
    
    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            final_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 1:
            if np.isfinite(eval_result[0]):
                final_c = float(eval_result[0])
    except Exception:
        pass
    
    return best_h, final_c, len(best_h)
