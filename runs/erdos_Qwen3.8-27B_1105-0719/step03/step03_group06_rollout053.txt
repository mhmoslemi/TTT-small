    edges_old = np.linspace(0.0, 2.0, n + 1)
    edges_new = np.linspace(0.0, 2.0, new_n + 1)
    h_new = np.zeros(new_n)
    for k in range(new_n):
        a, b = edges_new[k], edges_new[k + 1]
        total = 0.0
        for j in range(n):
            lo = max(a, edges_old[j])
            hi = min(b, edges_old[j + 1])
            if hi > lo:
                total += h[j] * (hi - lo)
        h_new[k] = total / (b - a)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

def trlp_full(h, delta, t_start, deadline, min_remaining=5):
    """Full trust-region LP step using all constraints."""
    n = len(h)
    M = 2 * n - 1
    P = compute_profile(h)
    c = float(P.max())
    if (time.perf_counter() - t_start) > deadline - min_remaining:
        return h, c, False
    G = build_gradient_matrix(h, n)
    # LP: min t s.t. G@d - t <= -P, sum(d)=0, box on d
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_ub = np.zeros((M, n + 1))
    A_ub[:, :n] = G
    A_ub[:, -1] = -1.0
    b_ub = -P
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for i in range(n):
        lo = max(-delta, -h[i] - 1e-12)
        hi = min(delta, 1.0 - h[i] + 1e-12)
        bounds.append((lo, hi))
    bounds.append((0.0, c + 2.0))
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and np.isfinite(res.x).all():
            d = res.x[:n]
            h_new = repair_feasible(h + d)
            c_new = exact_score(h_new)
            if c_new < c - 1e-13:
                return h_new, c_new, True
    except Exception:
        pass
    return h, c, False

def trlp_subset(h, delta, n_active=40, t_start=0, deadline=999999):
    """Trust-region LP with subset of active constraints."""
    n = len(h)
    M = 2 * n - 1
    P = compute_profile(h)
    c = float(P.max())
    if (time.perf_counter() - t_start) > deadline - 5:
        return h, c, False
    active = np.argsort(-P)[:n_active]
    G_a = np.array([grad_row(h, m, n) for m in active])
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_ub = np.zeros((len(active), n + 1))
    A_ub[:, :n] = G_a
    A_ub[:, -1] = -1.0
    b_ub = -P[active]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for i in range(n):
        lo = max(-delta, -h[i] - 1e-12)
        hi = min(delta, 1.0 - h[i] + 1e-12)
        bounds.append((lo, hi))
    bounds.append((0.0, c + 2.0))
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and np.isfinite(res.x).all():
            d = res.x[:n]
            h_new = repair_feasible(h + d)
            c_new = exact_score(h_new)
            if c_new < c - 1e-13:
                return h_new, c_new, True
    except Exception:
        pass
    return h, c, False

def lp_descent(h, max_iters=300, initial_delta=0.035, t_start=0, deadline=999999):
    """Iterative LP trust-region descent with adaptive delta."""
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    delta = initial_delta
    stall = 0
    for it in range(max_iters):
        if (time.perf_counter() - t_start) > deadline - 10:
            break
        # Alternate between full and subset LP
        if it % 3 == 2:
            h_new, c_new, improved = trlp_full(h, delta, t_start, deadline)
        else:
            h_new, c_new, improved = trlp_subset(h, delta, n_active=50, t_start=t_start, deadline=deadline)
        if improved:
            h = h_new
            if c_new < best_c:
                best_c = c_new
            stall = 0
            delta = min(delta * 1.15, 0.25)
        else:
            stall += 1
            delta *= 0.65
            if delta < 5e-5 or stall > 20:
                break
    return h, best_c

def subgrad_descent(h, max_iters=500, init_step=0.04, t_start=0, deadline=999999):
    """Subgradient descent with adaptive step size and multiple active sets."""
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    step = init_step
    mum = 0
    for it in range(max_iters):
        if (time.perf_counter() - t_start) > deadline - 10:
            break
        P = compute_profile(h)
        c = float(P.max())
        # Use top-k active with k varying
        k = 5 + (it % 10)
        active = np.argsort(-P)[:k]
        g = np.zeros(n)
        for m in active:
            g += grad_row(h, m, n)
        g -= g.mean()
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            break
        step = 0.08 / (np.sqrt((it + 1)) * 0.1 + 1.0)
        step = max(step, 0.005)
        step = min(step, 0.12)
        d = -g / gnorm
        # Try multiple alphas
        best_a, best_val, best_h = 0.0, c, h
        for a in [step, step * 0.7, step * 0.5, step * 1.3]:
            h_try = repair_feasible(h + a * d)
            c_try = exact_score(h_try)
            if c_try < best_val:
                best_val = c_try
                best_a = a
                best_h = h_try
        if best_a > 0:
            h = best_h
            if best_val < best_c:
                best_c = best_val
            mum = 0
        else:
            mum += 1
            if mum > 30:
                break
            step *= 0.85
    return h, best_c

def prox_grad(h, max_iters=400, t_start=0, deadline=999999):
    """Proximal gradient with Nesterov momentum."""
    n = len(h)
    h = h.copy()
    h_prev = h.copy()
    best_c = exact_score(h)
    step = 0.06
    mom = 0.5
    for it in range(max_iters):
        if (time.perf_counter() - t_start) > deadline - 10:
            break
        y = h + mom * (h - h_prev)
        P = compute_profile(y)
        c_y = float(P.max())
        active = np.argsort(-P)[:12]
        g = np.zeros(n)
        for m in active:
            g = grad_row(y, m, n) if m == active[0] else g + grad_row(y, m, n)
        g -= g.mean()
        h_new = repair_feasible(y - step * g)
        c_new = exact_score(h_new)
        if c_new < c_y:
            mom_new = min(0.8, (1 + mom) / 2)
        else:
            h_new = h
            mom_new = mom * 0.9
            step *= 0.9
        h_prev = h
        h = h_new
        mom = mom_new
        if c_new < best_c:
            best_c = c_new
        if step < 1e-4:
            break
    return h, best_c

def binary_seed(n, rng, mode=0):
    half = n // 2
    h = np.zeros(n)
    if mode == 0:
        idx = np.arange(n)
        vals = (idx * 0.618033988749895) % 1.0
        top = np.argsort(vals)[-half:]
        h[top] = 1.0
    elif mode == 1:
        ones = [i for i in range(n) if bin(i).count('1') % 2 == 0]
        if len(ones) >= half:
            h[ones[:half]] = 1.0
        else:
            h[ones] = 1.0
            zeros = np.where(h == 0)[0]
            h[rng.choice(zeros, half - len(ones), replace=False)] = 1.0
    elif mode == 2:
        freq = 3.0 + rng.random() * 5.0
        h = (np.sin(2 * np.pi * freq * np.arange(n) / n + rng.random() * 6.28) > 0).astype(float)
        ones = int(h.sum())
        if ones > half:
            oi = np.where(h == 1.0)[0]
            h[rng.choice(oi, ones - half, replace=False)] = 0.0
        elif ones < half:
            zi = np.where(h == 0.0)[0]
            h[rng.choice(zi, half - ones, replace=False)] = 1.0
    elif mode == 3:
        # Fibonacci word style
        fib = [0]
        for i in range(60):
            fib = fib + [1 - x for x in fib]
        h[:n] = np.array(fib[:n], dtype=float)
        ones = int(h.sum())
        if ones > half:
            oi = np.where(h == 1.0)[0]
            h[rng.choice(oi, ones - half, replace=False)] = 0.0
        elif ones < half:
            zi = np.where(h == 0.0)[0]
            h[rng.choice(zi, half - ones, replace=False)] = 1.0
    elif mode == 4:
        # Weighted golden ratio with offset
        offset = rng.random()
        idx = np.arange(n)
        vals = ((idx + offset) * 0.618033988749895) % 1.0
        top = np.argsort(vals)[-half:]
        h[top] = 1.0
    else:
        perm = rng.permutation(n)
        h[perm[:half]] = 1.0
    return repair_feasible(h)

def multi_start_lp(h_best, best_c, n, rng, t_start, deadline, n_trials=4):
    """Run LP descent from multiple random perturbations of h_best."""
    h = h_best.copy()
    bc = best_c
    for trial in range(n_trials):
        if (time.perf_counter() - t_start) > deadline - 30:
            break
        # Generate perturbation
        noise = rng.randn(n) * 0.08
        noise -= noise.mean()
        h_pert = repair_feasible(h + noise)
        h_ad, c_ad = lp_descent(h_pert, max_iters=120, initial_delta=0.04,
                                t_start=t_start, deadline=deadline)
        if c_ad < bc:
            h, bc = h_ad, c_ad
    return h, bc

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()
    rng = np.random.RandomState(seed)

    # Get parent
    try:
        parent = np.array(initial_h_values, dtype=np.float64)
    except Exception:
        parent = None
    if parent is None or len(parent) < 4:
        parent = np.full(70, 0.5)
    parent = np.clip(parent, 0.0, 1.0)
    parent = repair_feasible(parent)
    n_parent = len(parent)

    best_h = parent.copy()
    best_c = exact_score(parent)

    # Phase 1: LP descent on parent
    if (time.perf_counter() - t_start) < budget_s - 200:
        h_ad, c_ad = lp_descent(best_h, max_iters=200, initial_delta=0.03,
                                t_start=t_start, deadline=budget_s - 150)
        if c_ad < best_c:
            best_h, best_c = h_ad, c_ad

    # Phase 2: Proximal gradient refinement
    if (time.perf_counter() - t_start) < budget_s - 100:
        h_pg, c_pg = prox_grad(best_h, max_iters=200, t_start=t_start, deadline=budget_s - 80)
        if c_pg < best_c:
            best_h, best_c = h_pg, c_pg

    # Phase 3: Dimension exploration
    n_b = len(best_h)
    dim_list = []
    for dn in [-6, -4, -2, 2, 4, 6, 8, 10]:
        d = n_b + dn
        if 44 <= d <= 120:
            dim_list.append(d)
    dim_list.sort(key=lambda x: abs(x - n_b))
    for new_n in dim_list[:6]:
        if (time.perf_counter() - t_start) > budget_s - 80:
            break
        h_interp = interp_to_n(best_h, new_n)
        h_ad, c_ad = lp_descent(h_interp, max_iters=120, initial_delta=0.035,
                                t_start=t_start, deadline=budget_s - 60)
        if c_ad < best_c:
            best_h, best_c = h_ad, c_ad
            n_b = new_n

    # Phase 4: Binary seeds with LP optimization
    n_explore = len(best_h)
    for mode in range(5):
        if (time.perf_counter() - t_start) > budget_s - 60:
            break
        h_bin = binary_seed(n_explore, rng, mode)
        # Try direct LP on binary
        h_ad, c_ad = lp_descent(h_bin, max_iters=100, initial_delta=0.03,
                                t_start=t_start, deadline=budget_s - 40)
        if c_ad < best_c:
            best_h, best_c = h_ad, c_ad
        # Try softened version
        for alpha in [0.3, 0.5, 0.7]:
            if (time.perf_counter() - t_start) > budget_s - 40:
                break
            h_soft = repair_feasible(alpha * h_bin + (1 - alpha) * np.full(n_explore, 0.5))
            h_ad, c_ad = lp_descent(h_soft, max_iters=80, initial_delta=0.04,
                                    t_start=t_start, deadline=budget_s - 30)
            if c_ad < best_c:
                best_h, best_c = h_ad, c_ad

    # Phase 5: Multi-start from best
    if (time.perf_counter() - t_start) < budget_s - 80:
        best_h, best_c = multi_start_lp(best_h, best_c, len(best_h), rng,
                                        t_start, budget_s - 30, n_trials=5)

    # Phase 6: Subgradient refinement
    if (time.perf_counter() - t_start) < budget_s - 50:
        h_sg, c_sg = subgrad_descent(best_h, max_iters=400, init_step=0.03,
                                     t_start=t_start, deadline=budget_s - 30)
        if c_sg < best_c:
            best_h, best_c = h_sg, c_sg

    # Phase 7: Fine LP with small delta
    if (time.perf_counter() - t_start) < budget_s - 20:
        h_ad, c_ad = lp_descent(best_h, max_iters=300, initial_delta=0.015,
                                t_start=t_start, deadline=budget_s - 10)
        if c_ad < best_c:
            best_h, best_c = h_ad, c_ad

    # Phase 8: Final subgradient polish
    if (time.perf_counter() - t_start) < budget_s - 5:
        h_sg, c_sg = subgrad_descent(best_h, max_iters=300, init_step=0.01,
                                     t_start=t_start, deadline=budget_s - 2)
        if c_sg < best_c:
            best_h, best_c = h_sg, c_sg

    # Final
    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)

    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            if eval_result < final_c + 1e-9:
                final_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 2:
            if float(eval_result[0]) < final_c + 1e-9:
                final_c = float(eval_result[0])
    except Exception:
        pass

    return best_h, final_c, len(best_h)
