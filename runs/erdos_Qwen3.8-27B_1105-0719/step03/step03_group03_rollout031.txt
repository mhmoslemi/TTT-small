```python
def compute_profile(h):

    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        # Need to add diff total
        slack = 1.0 - h
        order = np.argsort(-slack)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, slack[idx])
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining < 1e-15:
                break
    else:
        # Need to remove -diff total
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining < 1e-15:
                break
    # Final fix
    diff2 = target - h.sum()
    if abs(diff2) > 1e-12:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    G = np.zeros((M, n))
    center = n - 1
    G[center, :] = dx * (1.0 - 2.0 * h)

        idx1 = np.arange(n - d)
        G[m_pos, idx1] += -dx * h[idx1 + d]
        G[m_pos, idx1 + d] += dx * (1.0 - h[idx1])
        m_neg = center - d
        G[m_neg, idx1] += dx * (1.0 - h[idx1 + d])
        G[m_neg, idx1 + d] += -dx * h[idx1]
    return G

    n = len(h)
    profile = compute_profile(h)
    G = build_grad(h)
    
    z = profile / eta
    z -= z.max()
    exp_z = np.exp(z)
    p = exp_z / exp_z.sum()
    
    grad = G.T @ p

    grad -= grad.mean()
    return grad, profile, p

    best_h = h.copy()
    best_c = exact_score(h)

            if time_limit_fn() < 5:

            if gnorm < 1e-14:

            direction = -momentum
            direction -= direction.mean()
            
            # Backtracking line search

            best_trial = None
            for alpha_mult in [1.0, 0.5, 0.25, 0.125, 0.0625]:

                h_trial = repair_feasible(h + alpha * direction)
                c_trial = exact_score(h_trial)
                if c_trial < curr_c - 1e-13:
                    best_trial = (h_trial, c_trial, alpha)
                    break
            
            if best_trial is not None:
                h, c_new, _ = best_trial
                if c_new < best_c - 1e-14:
                    best_c = c_new
                    best_h = h.copy()
                step = min(step * 1.05, eta * 2.0)
            else:
                step *= 0.5
                if step < eta * 0.01:

        # After each eta, check if we should reset momentum
        h = best_h.copy()
        # Re-evaluate
        c_check = exact_score(h)
        if c_check < best_c:
            best_c = c_check
    
    return best_h, best_c
```

```python
def solve_1d_minimax(a, b):
    """Solve min_{x in [0,1]} max_k (b[k] + a[k]*x)
    Returns x* and the optimal value."""
    K = len(a)
    
    # Find all breakpoints
    # x_ik = (b[k] - b[i]) / (a[i] - a[k])
    # Vectorized:
    a_diff = a[:, None] - a[None, :]  # K x K
    b_diff = b[None, :] - b[:, None]  # K x K (b[k] - b[i])
    
    with np.errstate(divide='ignore', invalid='ignore'):
        x_pts = b_diff / a_diff  # where a[i] != a[k]
    
    # Filter to [0,1], finite
    mask = np.isfinite(x_pts) & (x_pts >= 0) & (x_pts <= 1) & (np.abs(a_diff) > 1e-15)
    x_candidates = x_pts[mask]
    
    # Add boundaries
    x_candidates = np.concatenate([x_candidates, [0.0, 1.0]])
    x_candidates = np.unique(x_candidates)  # remove dupes
    
    # Evaluate f(x) = max_k(b[k] + a[k]*x) at each candidate
    # Shape: (K, N_candidates)
    f_vals = b[:, None] + a[:, None] * x_candidates[None, :]
    f_max = f_vals.max(axis=0)
    
    # Find best
    idx = np.argmin(f_max)
    return x_candidates[idx], f_max[idx]
```

```python
def lse_step(h, eta, momentum, mu):
    """One LSE gradient step with Nesterov momentum."""
    n = len(h)
    profile = compute_profile(h)
    G = build_grad(h)
    
    # LSE weights
    z = profile / eta
    z -= z.max()
    exp_z = np.exp(z)
    p = exp_z / exp_z.sum()
    
    # Gradient
    grad = G.T @ p
    grad -= grad.mean()  # project to tangent space
    
    # Nesterov momentum
    momentum = mu * momentum + grad
    direction = -momentum
    direction -= direction.mean()
    
    return direction, momentum, profile
```

```python
def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    prof = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(prof))


def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx


def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        slack = 1.0 - h
        order = np.argsort(-slack)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, slack[idx])
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining < 1e-14:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining < 1e-14:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-11:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h


def build_grad(h):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    G = np.zeros((M, n))
    center = n - 1
    G[center, :] = dx * (1.0 - 2.0 * h)
    for d in range(1, n):
        m_pos = center + d
        G[m_pos, :n - d] += -dx * h[d:]
        G[m_pos, d:] += dx * (1.0 - h[:n - d])
        m_neg = center - d
        G[m_neg, :n - d] += dx * (1.0 - h[d:])
        G[m_neg, d:] += -dx * h[:n - d]
    return G


def solve_1d_minimax(a, b):
    K = len(a)
    if K == 0:
        return 0.5, 0.0
    a_diff = a[:, None] - a[None, :]
    b_diff = b[None, :] - b[:, None]
    with np.errstate(divide='ignore', invalid='ignore'):
        x_pts = b_diff / a_diff
    mask = np.isfinite(x_pts) & (x_pts >= 0.0) & (x_pts <= 1.0) & (np.abs(a_diff) > 1e-14)
    xc = x_pts[mask]
    xc = np.concatenate([xc, [0.0, 1.0]])
    xc = np.unique(np.round(xc, 14))
    if len(xc) == 0:
        return 0.0, float(np.max(b))
    fvals = b[:, None] + a[:, None] * xc[None, :]
    fmax = fvals.max(axis=0)
    idx = int(np.argmin(fmax))
    return float(xc[idx]), float(fmax[idx])


def coord_descent(h0, n_sweeps, time_check, rng):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c = exact_score(h)
    for sw in range(n_sweeps):
        if time_check() < 10:
            break
        order = rng.permutation(n)
        for j in order:
            profile = compute_profile(h)
            G = build_grad(h)
            a = G[:, j]
            b = profile - a * h[j]
            x_star, _ = solve_1d_minimax(a, b)
            if x_star >= 0.0 and x_star <= 1.0:
                h[j] = x_star
        h = repair_feasible(h)
        c = exact_score(h)
        if c < best_c - 1e-14:
            best_c = c
            best_h = h.copy()
    return best_h, best_c


def lse_descent(h0, etas, iters_per_eta, time_check):
    h = h0.copy()
    best_h = h.copy()
    best_c = exact_score(h)
    for eta in etas:
        momentum = np.zeros_like(h)
        step = 0.5 * eta
        for it in range(iters_per_eta):
            if time_check() < 8:
                break
            profile = compute_profile(h)
            G = build_grad(h)
            z = profile / eta
            z -= z.max()
            exp_z = np.exp(z)
            p = exp_z / exp_z.sum()
            grad = G.T @ p
            grad -= grad.mean()
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break
            momentum = 0.8 * momentum + 0.2 * grad
            direction = -momentum
            direction -= direction.mean()
            dn = np.linalg.norm(direction)
            if dn < 1e-14:
                break
            curr_c = exact_score(h)
            accepted = False
            for mult in [1.0, 0.7, 0.5, 0.35, 0.2, 0.1]:
                alpha = step * mult
                h_try = repair_feasible(h + alpha * direction)
                c_try = exact_score(h_try)
                if c_try < curr_c - 1e-13:
                    h = h_try
                    if c_try < best_c - 1e-14:
                        best_c = c_try
                        best_h = h.copy()
                    step = min(step * 1.1, 3.0 * eta)
                    accepted = True
                    break
            if not accepted:
                step *= 0.6
                if step < 0.01 * eta:
                    break
        h = best_h.copy()
    return best_h, best_c


def interp_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    fx = np.concatenate([[0.0], xs, [2.0]])
    fh = np.concatenate([[h[0]], h, [h[-1]]])
    hn = np.interp(xt, fx, fh)
    return repair_feasible(np.clip(hn, 0.0, 1.0))


def slsqp_polish(h0, maxiter, time_check):
    n = len(h0)
    dx = 2.0 / n
    M = 2 * n - 1
    t0_init = exact_score(h0) + 1e-3

    def slsqp_obj(x):
        return x[-1]

    def slsqp_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g

    def slsqp_eq(x):
        return x[:-1].sum() - n / 2.0

    def slsqp_eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g

    def slsqp_ineq(x):
        hv = x[:-1]
        prof = np.correlate(hv, 1.0 - hv, mode="full") * dx
        return x[-1] - prof

    def slsqp_ineq_jac(x):
        hv = x[:-1]
        G = build_grad(hv)
        J = np.zeros((M, n + 1))
        J[:, :-1] = -G
        J[:, -1] = 1.0
        return J

    x0 = np.concatenate([h0, [t0_init]])
    bounds = [(0.0, 1.0)] * n + [(0.0, t0_init + 1.0)]
    cons = [
        {'type': 'eq', 'fun': slsqp_eq, 'jac': slsqp_eq_jac},
        {'type': 'ineq', 'fun': slsqp_ineq, 'jac': slsqp_ineq_jac}
    ]
    try:
        res = minimize(slsqp_obj, x0, method='SLSQP', jac=slsqp_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.x is not None and np.isfinite(res.x).all():
            hs = repair_feasible(res.x[:-1])
            if np.isfinite(exact_score(hs)):
                return hs
    except Exception:
        pass
    return None
```
