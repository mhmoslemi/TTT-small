def slsqp_optimize(h0, dx, target_sum, beta, maxiter):
    n = len(h0)

    def obj(x):
        corrs = compute_corrs(x)
        return logsumexp(beta * corrs) / beta * dx

    def grad_obj(x):
        return smooth_max_grad(x, dx, beta)

    def c_fun(x):
        return x.sum() - target_sum

    def c_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': c_fun, 'jac': c_jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0, jac=grad_obj, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        h_out = project_feasible(res.x, target_sum)
    except Exception:
        h_out = project_feasible(h0, target_sum)
    return h_out, c5_of_h(h_out, dx)


def exact_subgrad_line_search(h, dx, target_sum, lr_init, max_backtrack=20):
    """Exact line search along the negative subgradient of the true max."""
    n = len(h)
    corrs = compute_corrs(h)
    c5_val = corrs.max() * dx
    k_star = np.argmax(corrs)
    g = grad_corr_k(h, k_star, n) * dx
    
    lr = lr_init
    h_cur = h.copy()
    for _ in range(max_backtrack):
        h_new = project_feasible(h_cur - lr * g, target_sum)
        new_c5 = c5_of_h(h_new, dx)
        if new_c5 < c5_val - 1e-15:
            return h_new, new_c5
        lr *= 0.5
        if lr < 1e-10:
            break
    return h_cur, c5_val

def lp_active_set(h, dx, target_sum, n_active=50, step_bound=0.15):
    """LP polish with active set constraints."""
    n = len(h)
    corrs = compute_corrs(h)
    max_c = corrs.max()
    
    top_idx = np.argsort(corrs)[-n_active:]
    
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_rows = []
    b_rows = []
    for k in top_idx:
        row = np.zeros(n + 1)
        row[:n] = dx * grad_corr_k(h, k, n)
        row[n] = -1.0
        A_rows.append(row)
        b_rows.append(-corrs[k] * dx)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)

    bounds = []
    for i in range(n):
        bounds.append((max(-h[i], -step_bound), min(1.0 - h[i], step_bound)))
    bounds.append((0.0, None))

    try:
        res = linprog(c_obj, A_ub=np.array(A_rows), b_ub=np.array(b_rows),
                      A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    except Exception:
        return h, c5_of_h(h, dx)
    if res.status != 0:
        return h, c5_of_h(h, dx)

    delta = res.x[:n]

        return h, c5_of_h(h, dx)

    h_new = project_feasible(h + delta, target_sum)
    return h_new, c5_of_h(h_new, dx)

def pairwise_exchange(h, dx, target_sum, n_top=20, n_passes=3):

    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)

    for _ in range(n_passes):
        corrs = compute_corrs(h)
        grad = smooth_max_grad(h, dx, 100.0)
        dec_idx = np.argsort(grad)[-n_top:]
        inc_idx = np.argsort(grad)[:n_top]
        improved = False
        
        for i in dec_idx:
            for j in inc_idx:
                if i == j:
                    continue
                # d(corr[k])/d(eps) for moving eps from i to j

                for k in range(2 * n - 1):
                    gk = grad_corr_k(h, k, n)

                        a, c_pt, fc = c_pt, d_pt, fd

def simulated_annealing(h0, dx, target_sum, t0, t1, n_iter):

    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.special import logsumexp
import time


def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)


def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def c5_of_h(h, dx):
    return compute_corrs(h).max() * dx


def grad_corr_k(h, k, n):
    offset = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -offset)
    i_hi = min(n - 1, n - 1 - offset)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx + offset] += (1.0 - h[idx])
    grad[idx] += (-h[idx + offset])
    return grad


def all_gradients(h, n):
    m = 2 * n - 1
    G = np.zeros((m, n))
    for k in range(m):
        G[k] = grad_corr_k(h, k, n)
    return G


def smooth_max_grad(h, dx, beta):
    n = len(h)
    corrs = compute_corrs(h)
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones_like(corrs) / len(corrs)
    else:
        w /= w_sum
    G = all_gradients(h, n)
    return (w @ G) * dx


def slsqp_optimize(h0, dx, target_sum, beta, maxiter):
    n = len(h0)

    def obj(x):
        corrs = compute_corrs(x)
        return logsumexp(beta * corrs) / beta * dx

    def grad_obj(x):
        return smooth_max_grad(x, dx, beta)

    def c_fun(x):
        return x.sum() - target_sum

    def c_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': c_fun, 'jac': c_jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0, jac=grad_obj, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        h_out = project_feasible(res.x, target_sum)
    except Exception:
        h_out = project_feasible(h0, target_sum)
    return h_out, c5_of_h(h_out, dx)

    # Final SLSQP + Benders + Pairwise
    if time_left_fn() > 12:
        h_f, c5_f = slsqp_optimize(best_h, dx, target_sum, 10000, 500)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f
    if time_left_fn() > 10:
        h_b2, c5_b2 = iterative_benders(best_h, dx, target_sum, max_iters=80, time_left_fn=time_left_fn)
        if c5_b2 < best_c5:
            best_c5 = c5_b2
            best_h = h_b2
    if time_left_fn() > 8:
        h_pw2, c5_pw2 = pairwise_exchange(best_h, dx, target_sum, n_top=20, n_passes=3)
        if c5_pw2 < best_c5:
            best_c5 = c5_pw2
            best_h = h_pw2

    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s - (time.time() - t0)

    best_h = None
    best_c5 = float('inf')

    # Main optimization
    h0 = np.array(initial_h_values, dtype=float)
    n = len(h0)
    dx = 2.0 / n
    target_sum = n / 2.0
    h0 = project_feasible(h0, target_sum)

    h_opt, c5_val = full_optimize(h0, dx, target_sum, time_left, "main")
    best_h = h_opt
    best_c5 = c5_val

    # Diversified restarts with varying perturbation
    for trial in range(20):
        if time_left() < 35:
            break
        if trial < 5:
            scale = 0.03
        elif trial < 12:
            scale = 0.10
        else:
            scale = 0.20
        perturb = scale * (rng.random(n) - 0.5)
        # Also try structured perturbations
        if trial % 4 == 3:
            block_size = n // 6
            for b in range(6):
                start = b * block_size
                end = start + block_size if b < 5 else n
                perturb[start:end] += 0.05 * (rng.random() - 0.5)
        h_try = project_feasible(best_h + perturb, target_sum)
        h_opt2, c5_val2 = full_optimize(h_try, dx, target_sum, time_left, "pert")
        if c5_val2 < best_c5:
            best_c5 = c5_val2
            best_h = h_opt2

    # Try different resolutions
    for n_try in [120, 132, 146, 160, 172, 180]:
        if time_left() < 45:
            break
        if n_try == len(best_h):
            continue
        dx_try = 2.0 / n_try
        ts_try = n_try / 2.0
        x_orig = np.linspace(0, 2, len(best_h), endpoint=False) + 1.0 / len(best_h)
        x_new = np.linspace(0, 2, n_try, endpoint=False) + 1.0 / n_try
        h_interp = np.interp(x_new, x_orig, best_h)
        h_interp = project_feasible(h_interp, ts_try)
        h_opt3, c5_val3 = full_optimize(h_interp, dx_try, ts_try, time_left, "n%d" % n_try)
        if c5_val3 < best_c5:
            best_c5 = c5_val3
            best_h = h_opt3

    # Final intensive polish
    if time_left() > 25 and best_h is not None:
        n_b = len(best_h)
        dx_b = 2.0 / n_b
        ts_b = n_b / 2.0
        for rnd in range(4):
            if time_left() < 12:
                break
            h_pw, c5_pw = pairwise_exchange(best_h, dx_b, ts_b, n_top=25, n_passes=3)
            if c5_pw < best_c5:
                best_c5 = c5_pw
                best_h = h_pw
            h_b, c5_b = iterative_benders(best_h, dx_b, ts_b, max_iters=60, time_left_fn=time_left)
            if c5_b < best_c5:
                best_c5 = c5_b
                best_h = h_b
            h_ls, c5_ls = exact_subgrad_ls(best_h, dx_b, ts_b, 0.005, top_k=5)
            if c5_ls < best_c5:
                best_c5 = c5_ls
                best_h = h_ls

    # Ensure feasibility
    if best_h is None:
        best_h = h0
        best_c5 = c5_of_h(h0, 2.0 / len(h0))
    else:
        n_final = len(best_h)
        best_h = project_feasible(best_h, n_final / 2.0)
        best_c5 = c5_of_h(best_h, 2.0 / n_final)

    return best_h, best_c5, len(best_h)
