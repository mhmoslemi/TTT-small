    n = len(h)

    best_c5 = c5_of_h(h, dx)

        corrs = compute_corrs(h)
        c5_val = corrs.max() * dx

        beta = 1000.0
        shifted = corrs - corrs.max()
        w = np.exp(beta * shifted)
        w /= w.sum()

        grad = np.zeros(n)

import numpy as np

def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)


def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def c5_of_h(h, dx):
    return compute_corrs(h).max() * dx


def grad_corr_k(h, k, n):
    offset = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -offset)
    i_hi = min(n - 1, n - 1 - offset)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx + offset] += (1.0 - h[idx])
    grad[idx] += (-h[idx + offset])
    return grad


def smooth_max_grad(h, dx, beta):
    n = len(h)
    corrs = compute_corrs(h)
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones_like(corrs) / len(corrs)
    else:
        w /= w_sum
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] > 1e-12:
            grad += w[k] * grad_corr_k(h, k, n)
    return grad * dx


def exact_line_search(h, d, dx, target_sum, n, n_coarse=40, n_fine=20):
    """Line search along direction d (sum(d)=0) to minimize C5."""
    d_norm = np.linalg.norm(d)
    if d_norm < 1e-15:
        return h.copy(), c5_of_h(h, dx)
    
    # Find alpha_max
    alpha_max = 1.0 / d_norm  # normalize so max step is 1
    d_scaled = d / d_norm
    for i in range(n):
        if d_scaled[i] < -1e-15:
            a = -h[i] / d_scaled[i]
            alpha_max = min(alpha_max, a)
        elif d_scaled[i] > 1e-15:
            a = (1.0 - h[i]) / d_scaled[i]
            alpha_max = min(alpha_max, a)
    
    if alpha_max <= 1e-10:
        return h.copy(), c5_of_h(h, dx)
    alpha_max = min(alpha_max, 5.0)
    
    # Coarse grid
    alphas = np.linspace(0, alpha_max, n_coarse)
    best_a = 0.0
    best_c5 = c5_of_h(h, dx)
    for a in alphas:
        h_try = h + a * d_scaled
        h_try = project_feasible(h_try, target_sum)
        c5_try = c5_of_h(h_try, dx)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_a = a
    
    # Fine grid around best
    if best_a > 1e-10 and best_a < alpha_max - 1e-10:
        margin = min(best_a, alpha_max - best_a) * 0.5
        lo_f = max(0.0, best_a - margin)
        hi_f = min(alpha_max, best_a + margin)
        alphas_fine = np.linspace(lo_f, hi_f, n_fine)
        for a in alphas_fine:
            h_try = h + a * d_scaled
            h_try = project_feasible(h_try, target_sum)
            c5_try = c5_of_h(h_try, dx)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_a = a
    
    h_best = h + best_a * d_scaled
    h_best = project_feasible(h_best, target_sum)
    return h_best, best_c5

def pair_descent_opt(h0, dx, target_sum, time_left_fn, max_rounds=200, n_pairs=15):

def lp_polish_opt(h0, dx, target_sum, time_left_fn, max_iters=200, init_step=0.15):

        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0

        b_rows = []
        for k in active_k:

            row[:n] = dx * grad_corr_k(h, k, n)

def slsqp_optimize(h0, dx, target_sum, time_left_fn, beta=2000, maxiter=1000):
    """SLSQP with smooth max objective."""
    n = len(h0)
    
    def obj(x):
        corrs = compute_corrs(x)
        return logsumexp(beta * corrs) / beta * dx
    
    def grad_obj(x):
        return smooth_max_grad(x, dx, beta)
    
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0, jac=grad_obj, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        h_out = project_feasible(res.x, target_sum)
    except Exception:
        h_out = project_feasible(h0, target_sum)
    return h_out, c5_of_h(h_out, dx)


def random_direction_linesearch(h0, dx, target_sum, time_left_fn, n_tries=50, rng=None):

def make_structured_start(n, target_sum, rng, pattern_type=0):
    """Create structured initial conditions."""
    h = np.zeros(n)
    if pattern_type == 0:
        # Beta distribution
        h = rng.beta(1.5, 1.5, size=n)
    elif pattern_type == 1:
        # Periodic
        period = int(n / 3)
        fill = max(1, int(target_sum / n * period))
        for i in range(n):
            h[i] = 1.0 if (i % period) < fill else 0.0
        h += 0.1 * rng.randn(n)
    elif pattern_type == 2:
        # Bump
        center = n // 2
        width = n // 4
        h = np.exp(-0.5 * ((np.arange(n) - center) / width) ** 2)
        h += 0.1 * rng.randn(n)
    else:
        # Uniform with noise
        h = np.full(n, 0.5)
        h += 0.2 * rng.randn(n)
    return project_feasible(h, target_sum)

        if time_left() < 35:
            break
        dx_try = 2.0 / n_try
        ts_try = n_try / 2.0
        x_orig = np.linspace(0.0, 2.0, len(best_h), endpoint=False) + 1.0 / len(best_h)
        x_new = np.linspace(0.0, 2.0, n_try, endpoint=False) + 1.0 / n_try
        h_interp = np.interp(x_new, x_orig, best_h)
        h_interp = project_feasible(h_interp, ts_try)
        
        h_opt3, c5_opt3 = full_optimize(h_interp, dx_try, ts_try, time_left, f"n{n_try}")
        if c5_opt3 < best_c5:
            best_c5 = c5_opt3
            best_h = h_opt3
    
    # Try structured starts at original n
    for ptype in range(4):
        if time_left() < 30:
            break
        h_struct = make_structured_start(n0, ts0, rng, ptype)
        # Blend with current best
        h_blend = project_feasible(0.5 * h_struct + 0.5 * best_h, ts0)
        h_s, c5_s = full_optimize(h_blend, dx0, ts0, time_left, f"struct{ptype}")
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s
    
    # Final intensive polish
    if time_left() > 20 and best_h is not None:
        n_b = len(best_h)
        dx_b = 2.0 / n_b
        ts_b = n_b / 2.0
        
        h_f, c5_f = subgradient_linesearch_opt(best_h, dx_b, ts_b, time_left, max_iters=400)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f
        
        if time_left() > 15:
            h_lp, c5_lp = lp_polish_opt(best_h, dx_b, ts_b, time_left, max_iters=150, init_step=0.06)
            if c5_lp < best_c5:
                best_c5 = c5_lp
                best_h = h_lp
        
        if time_left() > 10:
            h_p, c5_p = pair_descent_opt(best_h, dx_b, ts_b, time_left, max_rounds=100, n_pairs=20)
            if c5_p < best_c5:
                best_c5 = c5_p
                best_h = h_p
        
        if time_left() > 8:
            h_s, c5_s = subgradient_hard_opt(best_h, dx_b, ts_b, time_left, max_iters=200)
            if c5_s < best_c5:
                best_c5 = c5_s
                best_h = h_s
    
    # Random direction exploration on best
    if time_left() > 10 and best_h is not None:
        n_b = len(best_h)
        dx_b = 2.0 / n_b
        ts_b = n_b / 2.0
        h_r, c5_r = random_direction_linesearch(best_h, dx_b, ts_b, time_left, n_tries=30, rng=rng)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r
    
    # Final validation
    if best_h is None:
        best_h = h0
        best_c5 = c5_of_h(h0, 2.0 / len(h0))
    else:
        n_final = len(best_h)
        best_h = project_feasible(best_h, n_final / 2.0)
        best_c5 = c5_of_h(best_h, 2.0 / n_final)
    
    return best_h, best_c5, len(best_h)
