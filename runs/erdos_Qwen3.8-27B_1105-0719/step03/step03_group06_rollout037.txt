def interp_to_n(h, new_n):

def quadratic_residue_seed(n, rng):
    """Generate a binary seed using quadratic residues."""
    h = np.zeros(n)
    # Find a prime p >= n
    p = n
    while True:
        if all(p % i != 0 for i in range(2, int(p**0.5) + 1)):
            break
        p += 1
    if p <= 2:
        p = 3
    # Quadratic residues mod p
    residues = set()
    for x in range(1, p):
        residues.add((x * x) % p)
    # Map to our n positions
    for i in range(n):
        if i in residues:
            h[i] = 1.0
    return repair_feasible(h)

        best_val = c
        best_h = h
        a_ls, b_ls = 0.0, alpha_max
        for _ in range(18):
            c1 = b_ls - gr * (b_ls - a_ls)
            c2 = a_ls + gr * (b_ls - a_ls)
            h1 = project_feasible(h + c1 * d)
            h2 = project_feasible(h + c2 * d)
            f1 = exact_score(h1)
            f2 = exact_score(h2)
            if f1 < best_val:
                best_val = f1
                best_alpha = c1
                best_h = h1.copy()
            if f2 < best_val:
                best_val = f2
                best_alpha = c2
                best_h = h2.copy()
            if f1 < f2:
                b_ls = c2
            else:
                a_ls = c1
            if b_ls - a_ls < 1e-13:
                break
        if best_val >= c - 1e-15:
            break
        h = best_h
        if best_val < best_c:
            best_c = best_val
    return h, best_c

def interp_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    edges_old = np.linspace(0.0, 2.0, n + 1)
    edges_new = np.linspace(0.0, 2.0, new_n + 1)
    h_new = np.zeros(new_n)
    for k in range(new_n):
        a, b = edges_new[k], edges_new[k + 1]
        total = 0.0
        for j in range(n):
            lo = max(a, edges_old[j])
            hi = min(b, edges_old[j + 1])
            if hi > lo:
                total += h[j] * (hi - lo)
        h_new[k] = total / (b - a)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_feasible(h_new)

def quadratic_residue_seed(n):
    h = np.zeros(n)
    p = n + 1
    while True:
        if p > 1 and all(p % i != 0 for i in range(2, int(p**0.5) + 1)):
            break
        p += 1
    residues = set((x * x) % p for x in range(1, p))
    for i in range(n):
        if i in residues:
            h[i] = 1.0
    return project_feasible(h)

def msequence_seed(n, rng):
    p = n
    p = max(p, 7)
    while p % 2 == 0:
        p -= 1
    h = np.zeros(n)
    state = 1
    for i in range(n):
        h[i] = 1.0 if (state & 1) == 1 else 0.0
        nb = (state >> p - 1) ^ ((state * 7) & 1)
        state = ((state << 1) | nb) & ((1 << p) - 1)
    return project_feasible(h)

def smooth_seed(n, rng):
    x = np.linspace(0, 2.0, n, endpoint=False)
    h = np.zeros(n)
    for f in range(1, 6):
        amp = 1.0 / f
        phase = rng.uniform(0, 2 * np.pi)
        h += amp * np.sin(2 * np.pi * f * x / 2.0 + phase)
    h = h - h.min()
    rng_val = h.max() - h.min()
    if rng_val > 1e-10:
        h = h / rng_val
    return project_feasible(0.3 + 0.7 * h)

def pair_polish(h, t_start, max_time):
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    for it in range(500):
        if time.perf_counter() - t_start > max_time:
            break
        profile = compute_profile(h)
        c = float(profile.max())
        hot = np.argsort(-profile)[:5]
        improved = False
        for m in hot:
            g_m = grad_row(h, m, n)
            desc = np.argsort(-g_m)
            asc = np.argsort(g_m)
            for k in range(min(20, n)):
                i = int(desc[k])
                j = int(asc[n - 1 - k])
                if i == j:
                    continue
                for delta in [0.02, 0.05, 0.01]:
                    if h[i] < delta or h[j] > 1.0 - delta:
                        continue
                    h_try = h.copy()
                    h_try[i] -= delta
                    h_try[j] += delta
                    h_try = project_feasible(h_try)
                    s = exact_score(h_try)
                    if s < best_c - 1e-14:
                        h = h_try
                        best_c = s
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        if not improved:
            break
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()
    rng = np.random.RandomState(seed)

    def time_left():
        return max(0.0, budget_s - (time.perf_counter() - t_start))

    try:
        parent = np.array(initial_h_values, dtype=np.float64)
    except Exception:
        parent = None
    if parent is None or len(parent) < 4 or not np.all(np.isfinite(parent)):
        parent = np.full(70, 0.5)
    parent = np.clip(parent, 0.0, 1.0)
    parent = project_feasible(parent)
    n_parent = len(parent)

    best_h = parent.copy()
    best_c = exact_score(parent)

    # Stage 1: LSE multi-temperature descent from parent
    if time_left() > 80:
        h_lse, c_lse = lse_descent_multi(best_h, t_start, min(150, time_left() * 0.25),
                                         temp_schedule=[50.0, 200.0, 800.0, 3000.0, 10000.0, 30000.0])
        if c_lse < best_c:
            best_h, best_c = h_lse, c_lse

    # Stage 2: Active descent on best
    if time_left() > 60:
        h_ad, c_ad = active_descent(best_h, t_start, min(80, time_left() * 0.15),
                                    max_iters=400, active_tol=0.003)
        if c_ad < best_c:
            best_h, best_c = h_ad, c_ad

    # Stage 3: Diverse starting points
    seeds = []
    # Quadratic residue
    h_qr = quadratic_residue_seed(n_parent)
    seeds.append(project_feasible(0.5 * h_qr + 0.5 * best_h))
    # M-sequence
    h_ms = msequence_seed(n_parent, rng)
    seeds.append(project_feasible(0.4 * h_ms + 0.6 * best_h))
    # Smooth
    seeds.append(smooth_seed(n_parent, rng))
    # Random perturbation
    h_r = project_feasible(best_h + rng.normal(0, 0.15, n_parent))
    seeds.append(h_r)
    # Anti-pattern
    h_anti = project_feasible(1.0 - best_h + 0.1 * rng.normal(0, 0.1, n_parent))
    seeds.append(h_anti)

    for s_idx, h_seed in enumerate(seeds):
        if time_left() < 40:
            break
        t_alloc = min(60, time_left() * 0.15)
        h_lse, c_lse = lse_descent_multi(h_seed, t_start, t_alloc,
                                         temp_schedule=[100.0, 500.0, 2000.0, 8000.0, 20000.0])
        if c_lse < best_c:
            best_h, best_c = h_lse, c_lse
        if time_left() < 30:
            continue
        h_ad, c_ad = active_descent(best_h, t_start, min(30, time_left() * 0.15),
                                    max_iters=200, active_tol=0.002)
        if c_ad < best_c:
            best_h, best_c = h_ad, c_ad

    # Stage 4: Dimension exploration
    n_best = len(best_h)
    dim_list = []
    for dn in [-6, -4, -2, 2, 4, 6, 8, 10, 12, 14, -8, -10]:
        d = n_best + dn
        if 40 <= d <= 140 and d % 2 == 0 and d != n_best:
            dim_list.append(d)
    dim_list.sort(key=lambda x: abs(x - n_best))

    for new_n in dim_list[:10]:
        if time_left() < 30:
            break
        h_interp = interp_to_n(best_h, new_n)
        t_alloc = min(40, time_left() * 0.12)
        h_lse, c_lse = lse_descent_multi(h_interp, t_start, t_alloc,
                                         temp_schedule=[100.0, 400.0, 1500.0, 5000.0, 15000.0])
        if c_lse < best_c:
            best_h, best_c = h_lse, c_lse
            if time_left() > 20:
                h_ad, c_ad = active_descent(best_h, t_start, min(25, time_left() * 0.2),
                                            max_iters=150, active_tol=0.002)
                if c_ad < best_c:
                    best_h, best_c = h_ad, c_ad
        else:
            h_ad, c_ad = active_descent(h_interp, t_start, min(20, time_left() * 0.1),
                                        max_iters=100, active_tol=0.003)
            if c_ad < best_c:
                best_h, best_c = h_ad, c_ad

    # Stage 5: Pair polish
    if time_left() > 25:
        h_pp, c_pp = pair_polish(best_h, t_start, min(25, time_left() * 0.3))
        if c_pp < best_c:
            best_h, best_c = h_pp, c_pp

    # Stage 6: Final fine active descent
    if time_left() > 15:
        h_ad, c_ad = active_descent(best_h, t_start, min(20, time_left() * 0.6),
                                    max_iters=300, active_tol=0.001)
        if c_ad < best_c:
            best_h, best_c = h_ad, c_ad

    # Stage 7: Very fine LSE
    if time_left() > 10:
        h_lse, c_lse = lse_descent_multi(best_h, t_start, min(15, time_left() * 0.7),
                                         temp_schedule=[2000.0, 8000.0, 30000.0, 100000.0])
        if c_lse < best_c:
            best_h, best_c = h_lse, c_lse

    # Final
    best_h = project_feasible(best_h)
    final_c = exact_score(best_h)

    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            final_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 2:
            final_c = float(eval_result[0])
    except Exception:
        pass

    return best_h, final_c, len(best_h)
