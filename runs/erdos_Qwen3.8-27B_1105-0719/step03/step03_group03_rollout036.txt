def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full")) * dx)

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx])
            h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    # final fix
    diff2 = target - h.sum()
    if abs(diff2) > 1e-14:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

            valid_i = np.arange(0, n - d)

def random_direction(n, rng, smoothness=1):
    """Generate a random zero-sum direction."""
    v = rng.randn(n)
    v += rng.randn(n) * 0  # just to vary
    # Apply smoothing
    for _ in range(smoothness):
        v = 0.5 * (v + np.roll(v, 1))
    v -= v.mean()
    norm = np.linalg.norm(v)
    if norm < 1e-14:
        return None
    return v / norm

def bump_direction(n, center, width, rng):
    """Generate a smooth bump-centered direction."""
    x = np.arange(n)
    v = np.exp(-0.5 * ((x - center) / width)**2)
    v -= v.mean()
    norm = np.linalg.norm(v)
    if norm < 1e-14:
        return None
    if rng.rand() < 0.5:
        v = -v
    return v / norm

def interp_to_n(h, new_n):
    """Interpolate h from current n to new_n."""
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    fx = np.concatenate([[0.0], xs, [2.0]])
    fh = np.concatenate([[h[0]], h, [h[-1]]])
    hn = np.interp(xt, fx, fh)
    return repair_feasible(np.clip(hn, 0, 1))

import numpy as np
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full")) * dx)

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            add = min(remaining, capacity[order[i]])
            h[order[i]] += add
            remaining -= add
            if remaining <= 1e-13:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            rem = min(remaining, h[order[i]])
            h[order[i]] -= rem
            remaining -= rem
            if remaining <= 1e-13:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-14:
        if diff2 > 0:
            h[int(np.argmax(1.0 - h))] += diff2
        else:
            h[int(np.argmax(h))] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def feasible_interval(h, v):
    pos = v > 1e-14
    neg = v < -1e-14
    a_min = 0.0
    a_max = 0.0
    if pos.any():
        a_max = float(np.min((1.0 - h[pos]) / v[pos]))
    if neg.any():
        a_min = float(np.max(-h[neg] / v[neg]))
    return a_min, a_max

def ray_search(h, v, n_coarse=150, n_fine=80, n_ultra=40):
    n = len(h)
    v = np.asarray(v, dtype=np.float64) - np.mean(v)
    vm = np.abs(v).max()
    if vm < 1e-14:
        return None, None
    v = v / vm  # normalize to unit max

    a_min, a_max = feasible_interval(h, v)
    if a_max - a_min < 1e-13:
        return None, None

    # Compute A, B, C via three evaluations
    A = np.correlate(h, 1.0 - h, mode="full")
    span = a_max - a_min
    a1 = a_min + 0.25 * span
    a2 = a_min + 0.75 * span

    h1 = h + a1 * v
    h2 = h + a2 * v
    P1 = np.correlate(h1, 1.0 - h1, mode="full")
    P2 = np.correlate(h2, 1.0 - h2, mode="full")

    d1 = P1 - A
    d2 = P2 - A
    C = (d2 / a2 - d1 / a1) / (a2 - a1)
    B = d1 / a1 - a1 * C

    # Dense search
    alphas = np.linspace(a_min, a_max, n_coarse)
    obj = np.max(A[None, :] + alphas[:, None] * B[None, :] + alphas[:, None]**2 * C[None, :], axis=1)
    bi = int(np.argmin(obj))
    best_a = alphas[bi]
    best_v = obj[bi]

    # Refine
    if bi > 0 and bi < n_coarse - 1:
        lo, hi = alphas[bi - 1], alphas[bi + 1]
    elif bi == 0:
        lo, hi = a_min, alphas[1]
    else:
        lo, hi = alphas[-2], a_max

    for nf in [n_fine, n_ultra]:
        alphas_r = np.linspace(lo, hi, nf)
        obj_r = np.max(A[None, :] + alphas_r[:, None] * B[None, :] + alphas_r[:, None]**2 * C[None, :], axis=1)
        bi_r = int(np.argmin(obj_r))
        best_a = alphas_r[bi_r]
        best_v = obj_r[bi_r]
        if bi_r > 0 and bi_r < nf - 1:
            lo, hi = alphas_r[bi_r - 1], alphas_r[bi_r + 1]
        elif bi_r == 0:
            lo, hi = lo, alphas_r[1]
        else:
            lo, hi = alphas_r[-2], hi

    h_new = repair_feasible(h + best_a * v)
    score = exact_score(h_new)
    return h_new, score

def grad_direction(h, profile, top_k=6):
    n = len(h)
    M = 2 * n - 1
    center = n - 1
    top = np.argsort(profile)[::-1][:top_k]
    w = np.zeros(M)
    for i, idx in enumerate(top):
        w[idx] = 1.0 / (i + 1.0)
    w /= w.sum()

    g = np.zeros(n)
    for k in range(M):
        wk = w[k]
        if wk == 0:
            continue
        d = k - center
        if d >= 0:
            j = np.arange(0, n - d)
            g[j] += wk * (1.0 - h[j + d])
            g[j + d] += -wk * h[j]
        else:
            dd = -d
            j = np.arange(dd, n)
            g[j] += wk * (1.0 - h[j - dd])
            j2 = np.arange(0, n - dd)
            g[j2] += -wk * h[j2 + dd]

    g -= g.mean()
    norm = np.linalg.norm(g)
    if norm < 1e-14:
        return None
    return -g / norm

def random_dir(n, rng, smooth=1):
    v = rng.randn(n)
    for _ in range(smooth):
        v = 0.5 * (v + np.concatenate([[v[-1]], v[:-1]]))
    v -= v.mean()
    norm = np.linalg.norm(v)
    if norm < 1e-14:
        return None
    return v / norm

def bump_dir(n, rng):
    c = rng.randint(0, n)
    w = rng.uniform(2.0, max(4.0, n / 3.0))
    x = np.arange(n)
    v = np.exp(-0.5 * ((x - c) / w) ** 2)
    if rng.rand() < 0.5:
        c2 = rng.randint(0, n)
        v -= np.exp(-0.5 * ((x - c2) / w) ** 2)
    v -= v.mean()
    norm = np.linalg.norm(v)
    if norm < 1e-14:
        return None
    return v / norm

def block_swap_dir(h, n, rng):
    """Transfer mass from a high block to a low block."""
    # Split into two halves, find high and low blocks
    half = n // 2
    left_mean = h[:half].mean()
    right_mean = h[half:].mean()
    v = np.zeros(n)
    if left_mean > right_mean:
        src_start = rng.randint(0, max(1, half - 5))
        src_len = rng.randint(3, min(8, half))
        dst_start = rng.randint(half, max(half + 1, n - 5))
        dst_len = rng.randint(3, min(8, n - half))
    else:
        dst_start = rng.randint(0, max(1, half - 5))
        dst_len = rng.randint(3, min(8, half))
        src_start = rng.randint(half, max(half + 1, n - 5))
        src_len = rng.randint(3, min(8, n - half))
    src_end = min(src_start + src_len, n)
    dst_end = min(dst_start + dst_len, n)
    v[src_start:src_end] = 1.0
    v[dst_start:dst_end] = -1.0
    v -= v.mean()
    norm = np.linalg.norm(v)
    if norm < 1e-14:
        return None
    return v / norm

def interp_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    fx = np.concatenate([[0.0], xs, [2.0]])
    fh = np.concatenate([[h[0]], h, [h[-1]]])
    hn = np.interp(xt, fx, fh)
    return repair_feasible(np.clip(hn, 0.0, 1.0))

def improve_local(h, tl, rng, max_rounds=200, verbose=False):
    n = len(h)
    h = h.copy()
    score = exact_score(h)
    profile = compute_profile(h)
    best_h = h.copy()
    best_score = score
    stagnation = 0

    for rnd in range(max_rounds):
        if tl() < 15:
            break

        # Generate candidate directions
        dirs = []
        d_grad = grad_direction(h, profile, top_k=5)
        if d_grad is not None:
            dirs.append(d_grad)
        d_grad2 = grad_direction(h, profile, top_k=10)
        if d_grad2 is not None and d_grad is not None:
            dirs.append(0.5 * (d_grad + d_grad2) / (0.5 * np.linalg.norm(d_grad + d_grad2) + 1e-15))
            dirs.append(d_grad2)

        for _ in range(3):
            d = random_dir(n, rng, smooth=rng.randint(0, 2))
            if d is not None:
                dirs.append(d)
        for _ in range(2):
            d = bump_dir(n, rng)
            if d is not None:
                dirs.append(d)
        if rnd % 3 == 0:
            d = block_swap_dir(h, n, rng)
            if d is not None:
                dirs.append(d)

        # Try all directions, keep best
        best_dir_h = None
        best_dir_score = score

        for d in dirs:
            if tl() < 10:
                break
            hn, sn = ray_search(h, d, n_coarse=120, n_fine=60, n_ultra=30)
            if hn is not None and sn < best_dir_score - 1e-13:
                best_dir_score = sn
                best_dir_h = hn

        if best_dir_h is not None:
            h = best_dir_h
            score = best_dir_score
            profile = compute_profile(h)
            if score < best_score - 1e-13:
                best_h = h.copy()
                best_score = score
                stagnation = 0
            else:
                stagnation += 1
        else:
            stagnation += 1

        # Escape mechanism
        if stagnation > 20:
            v = rng.randn(n)
            v -= v.mean()
            v /= (np.linalg.norm(v) + 1e-15)
            escaped = False
            for eps in [0.04, 0.07, 0.1, 0.15]:
                hp = repair_feasible(h + eps * v)
                cp = exact_score(hp)
                if cp < best_score + 0.01:
                    h = hp
                    score = cp
                    profile = compute_profile(h)
                    stagnation = 0
                    escaped = True
                    break
            if not escaped:
                # Try binarized restart
                hb = repair_feasible((h > 0.5).astype(float))
                cb = exact_score(hb)
                if cb < best_score + 0.02:
                    h = hb
                    score = cb
                    profile = compute_profile(h)
                    stagnation = 0

    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.perf_counter()
    rng = np.random.RandomState(seed)

    def tl():
        return budget_s - (time.perf_counter() - t0)

    # Load parent
    parent = None
    try:
        parent = np.array(initial_h_values, dtype=np.float64)
    except (NameError, Exception):
        pass
    if parent is None or len(parent) < 4:
        parent = np.full(70, 0.5)

    parent = repair_feasible(np.clip(parent, 0.0, 1.0))
    best_h = parent.copy()
    best_c = exact_score(parent)

    # Symmetrize
    hs = repair_feasible((parent + parent[::-1]) / 2.0)
    cs = exact_score(hs)
    if cs < best_c - 1e-12:
        best_h, best_c = hs, cs

    # Phase 1: Main local improvement
    if tl() > 120:
        bh, bc = improve_local(best_h, tl, rng, max_rounds=300)
        if bc < best_c - 1e-14:
            best_h, best_c = bh, bc

    # Phase 2: Multi-resolution exploration
    if tl() > 150:
        n0 = len(best_h)
        dim_list = sorted(set(d for d in range(max(30, n0 - 16), min(160, n0 + 17), 2)))
        # Sort by distance from n0
        dim_list.sort(key=lambda d: abs(d - n0))

        for nd in dim_list:
            if tl() < 60:
                break
            if nd == n0:
                continue
            hi = interp_to_n(best_h, nd)
            ci = exact_score(hi)
            if ci < best_c + 0.025:
                bh, bc = improve_local(hi, tl, rng, max_rounds=120)
                if bc < best_c - 1e-14:
                    best_h, best_c = bh, bc

    # Phase 3: Another local improvement round
    if tl() > 80:
        bh, bc = improve_local(best_h, tl, rng, max_rounds=200)
        if bc < best_c - 1e-14:
            best_h, best_c = bh, bc

    # Phase 4: Random perturbation restarts
    if tl() > 100:
        for trial in range(8):
            if tl() < 40:
                break
            n_cur = len(best_h)
            v = rng.randn(n_cur)
            v -= v.mean()
            v /= (np.linalg.norm(v) + 1e-15)
            for eps in [0.04, 0.07, 0.1, 0.14, 0.2]:
                if tl() < 30:
                    break
                hp = repair_feasible(best_h + eps * v)
                cp = exact_score(hp)
                if cp < best_c + 0.012:
                    bh, bc = improve_local(hp, tl, rng, max_rounds=80)
                    if bc < best_c - 1e-14:
                        best_h, best_c = bh, bc
                    break

    # Phase 5: Try binarized variants
    if tl() > 60:
        n_cur = len(best_h)
        for thresh in [0.3, 0.4, 0.5, 0.6, 0.7]:
            if tl() < 25:
                break
            hb = (best_h > thresh).astype(float)
            hb = repair_feasible(hb)
            cb = exact_score(hb)
            if cb < best_c + 0.02:
                bh, bc = improve_local(hb, tl, rng, max_rounds=80)
                if bc < best_c - 1e-14:
                    best_h, best_c = bh, bc

    # Phase 6: Final fine polish
    if tl() > 40:
        n_cur = len(best_h)
        h_f = best_h.copy()
        c_f = best_c
        prof_f = compute_profile(h_f)
        for _ in range(80):
            if tl() < 10:
                break
            d = grad_direction(h_f, prof_f, top_k=8)
            if d is not None:
                hn, sn = ray_search(h_f, d, n_coarse=300, n_fine=100, n_ultra=60)
                if hn is not None and sn < c_f - 1e-14:
                    h_f, c_f = hn, sn
                    prof_f = compute_profile(h_f)
                    continue
            d2 = random_dir(n_cur, rng, smooth=2)
            if d2 is not None:
                hn, sn = ray_search(h_f, d2, n_coarse=200, n_fine=80, n_ultra=40)
                if hn is not None and sn < c_f - 1e-14:
                    h_f, c_f = hn, sn
                    prof_f = compute_profile(h_f)
                    continue
            break
        if c_f < best_c - 1e-14:
            best_h, best_c = h_f, c_f

    # Final validation
    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)

    try:
        er = evaluate_erdos_solution(best_h)
        if isinstance(er, (float, int)):
            final_c = min(final_c, float(er))
        elif isinstance(er, tuple) and len(er) >= 2:
            final_c = min(final_c, float(er[0]))
    except Exception:
        pass

    return best_h, final_c, len(best_h)
