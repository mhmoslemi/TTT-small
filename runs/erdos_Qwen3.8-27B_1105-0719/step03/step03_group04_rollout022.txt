```python
    for m in range(M):
        k = m - (n - 1)
        for j in range(n):
            val = 0.0
            # First term: h[j] in h[n+k] when n=j-k, valid if 0<=j-k<n
            if 0 <= j - k < n:
                val += (1.0 - h[j - k])
            # Second term: h[j] in (1-h[n]) when n=j, valid if 0<=j+k<n
            if 0 <= j + k < n:
                val -= h[j + k]
            G[m, j] = dx * val
    return profile, G
```

```python
    h[:-1] = x
    h[-1] = n/2.0 - x.sum()

    if h[-1] < 0 or h[-1] > 1:
        return 1e10

    profile = np.correlate(h, 1-h, mode="full") * dx
    # Numerically stable log-sum-exp
    max_p = np.max(profile)
    if max_p == -np.inf:
        return 0
    lse = max_p + tau * np.log(np.sum(np.exp((profile - max_p) / tau)))
```

```python
            remaining -= add

def pairwise_line_search(h, j, k, n_steps=20):
    """Optimize the line h[j]+t, h[k]-t for optimal t."""
    n = len(h)
    dx = 2.0/n
    best_t = 0.0
    best_c = exact_score(h)
    
    # Determine valid range of t
    t_min = max(-h[j], -(1-h[k]))
    t_max = min(1-h[j], h[k])
    if t_max <= t_min:
        return h, best_c
    
    # Sample t values
    ts = np.linspace(t_min, t_max, n_steps)
    for t in ts:
        if abs(t) < 1e-15:
            continue
        h_try = h.copy()
        h_try[j] += t
        h_try[k] -= t
        h_try = np.clip(h_try, 0, 1)
        c_try = exact_score(h_try)
        if c_try < best_c:
            best_c = c_try
            best_t = t
    
    if best_t != 0:
        h[j] += best_t
        h[k] -= best_t
        h = np.clip(h, 0, 1)
    
    return h, best_c

def coordinate_polish(h, max_rounds=50, time_limit=None, t_start=None):
    """Iteratively polish via pairwise line searches."""
    n = len(h)
    best_c = exact_score(h)
    
    for rnd in range(max_rounds):
        if time_limit is not None and t_start is not None:
            if time.perf_counter() - t_start > time_limit:

        improved = False
        for j in range(n):
            for k in range(j+1, n):

                if c < best_c:
                    best_c = c
                    improved = True
```

The Frank-Wolfe (conditional gradient) method works as follows:
- At each iteration, solve a linearization subproblem
- Take a step toward the solution

For our problem:
- Variables: (h, t) with 0<=h<=1, sum(h)=n/2, t >= max(profile)
- Linearize the constraints: at current (h_k, t_k), the violated constraints are those with profile_k > t_k.
- Linear subproblem: minimize t subject to linearized constraints.

This is essentially what the LP trust region does, but without the trust region. The step length is chosen by line search.

Let me also think about what "chevron" or "neighborhood" search means here. If we perturb h[j] by +ε and h[k] by -ε (to preserve sum), the change in profile at lag m is:
Δprofile_m = dx * [(1-h[j-k_m])*ε - h[j+k_m]*ε - (1-h[k-k_m])*ε + h[k+k_m]*ε]
(simplified, where k_m is the lag corresponding to output index m)

This is linear in ε, so we can find the optimal ε by checking where the max is minimized.

Actually, for a pair swap (j, δ) and (k, -δ), the profile change is:
Δprofile_m = dx * ε * [G[m,j] - G[m,k]]

where G is the Jacobian matrix. So the new profile is:
profile_m + dx * ε * [G[m,j] - G[m,k]]

We want to find ε that minimizes max_m (profile_m + dx * ε * [G[m,j] - G[m,k]]).

This is a 1D minimization of a piecewise linear function! We can solve it exactly by finding the breakpoints.

The breakpoints occur where two components of (profile_m + dx * ε * d_m) cross, where d_m = G[m,j] - G[m,k].

So: profile_a + ε*d_a = profile_b + ε*d_b => ε = (profile_b - profile_a)/(d_a - d_b)

We find all such breakpoints, sort them, and check the max at each interval's endpoints.

This gives an exact 1D optimization for each pair! That's much better than sampling.

Let me implement this:

```python
def exact_pair_optimal(h, j, k):
    """Find optimal ε for h[j]+=ε, h[k]-=ε that minimizes max profile."""
    n = len(h)
    dx = 2.0/n
    prof = compute_profile(h)
    _, G = profile_jacobian(h)
    d = G[:, j] - G[:, k]  # direction in profile space
    
    # Valid range of ε
    eps_min = max(-h[j], h[k] - 1.0)  # h[j]+eps >= 0, h[k]-eps <= 1
    eps_max = min(1.0 - h[j], h[k])    # h[j]+eps <= 1, h[k]-eps >= 0
    
    if eps_max <= eps_min:
        return 0.0
    
    # Profile as function of eps: prof + eps * d
    # Find breakpoints where (prof_a + eps*d_a) = (prof_b + eps*d_b)
    # eps = (prof_b - prof_a) / (d_a - d_b)
    
    M = len(prof)
    breakpoints = [eps_min, 0.0, eps_max]
    
    # For efficiency, only check a subset of breakpoints
    # When d and d_b differ significantly
    non_zero = np.where(np.abs(d) > 1e-10)[0]
    if len(non_zero) > 50:
        non_zero = non_zero[np.argsort(np.abs(d[non_zero]))[-50:]]
    
    for a in non_zero:
        for b in non_zero:
            if a == b:
                continue
            denom = d[a] - d[b]
            if abs(denom) < 1e-15:
                continue
            eps_bp = (prof[b] - prof[a]) / denom
            if eps_min < eps_bp < eps_max:
                breakpoints.append(eps_bp)
    
    breakpoints = sorted(set([max(eps_min, min(eps_max, bp)) for bp in breakpoints]))
    
    # Evaluate at each breakpoint and at the endpoints
    best_eps = 0.0
    best_val = np.max(prof)
    for eps in breakpoints:
        val = np.max(prof + eps * d)
        if val < best_val:
            best_val = val
            best_eps = eps
    
    return best_eps
```

Wait, this is O(M^2) per pair evaluation due to the double loop over breakpoints. For M=139, that's about 19321 per pair, and n(n-1)/2 = 2415 pairs. Total: ~47 million operations per round. That might be okay for a few rounds.

Actually, I can optimize this. Instead of all pairs of breakpoints, I just need to find the ε that minimizes max_m (prof_m + ε*d_m). This is a 1D minimization of a convex piecewise-linear function (convex because it's the max of linear functions).

For a convex piecewise-linear function, the minimum is at a point where the subgradient contains 0, i.e., where there's a "kink" that straddles 0. We can find it by:
1. Computing all breakpoints in O(M^2)
2. Binary searching for the minimum in O(log(M^2)) evaluations

Or more simply, since the function is convex, we can use the fact that the minimum is at the point where the derivative changes sign. The derivative (where it exists) is d_m* for the active m (the one achieving the max).

Actually, for a max of linear functions, the minimum can be found by:
- The function f(ε) = max_m (prof_m + ε*d_m)
- f is convex piecewise linear
- The minimum is where 0 is in the subgradient
- Subgradient at ε is the convex hull of {d_m : m in active(ε)}
- So minimum is where min_active(d_m) <= 0 <= max_active(d_m)

This can be found by a bisection method in O(M log(precision)) time!

```python
def exact_pair_optimal_fast(h, j, k, tol=1e-10):
    n = len(h)
    dx = 2.0/n
    prof = compute_profile(h)
    _, G = profile_jacobian(h)
    d = G[:, j] - G[:, k]
    
    eps_min = max(-h[j], h[k] - 1.0)
    eps_max = min(1.0 - h[j], h[k])
    if eps_max <= eps_min + 1e-15:
        return 0.0
    
    # Bisect for the minimum of f(eps) = max_m(prof_m + eps*d_m)
    lo, hi = eps_min, eps_max
    for _ in range(60):  # ~1e-18 precision
        mid = (lo + hi) / 2
        f_mid = prof + mid * d
        active_max = np.max(f_mid)
        # Subgradient: d_m for m where f_mid[m] == active_max
        mask = np.abs(f_mid - active_max) < 1e-12
        if not np.any(mask):
            mask = [np.argmax(f_mid)]
        sg_min = np.min(d[mask])
        sg_max = np.max(d[mask])
        if sg_min > 1e-14:
            # f is increasing, go left
            hi = mid
        elif sg_max < -1e-14:
            # f is decreasing, go right
            lo = mid
        else:
            # 0 is in subgradient, found minimum
            break
    
    opt_eps = (lo + hi) / 2
    # Verify
    f_opt = np.max(prof + opt_eps * d)
    f_zero = np.max(prof)
    if f_opt < f_zero:
        return opt_eps
    return 0.0
```

```python
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def repair_feasible(h):
    n = len(h)
    h = np.clip(np.asarray(h, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, 1.0 - h[idx])
            h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)[::-1]
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx_arr = np.where(h < 1.0)[0]
            if len(idx_arr) > 0:
                idx = idx_arr[np.argmax(1.0 - h[idx_arr])]
                h[idx] += diff2
        else:
            idx_arr = np.where(h > 0.0)[0]
            if len(idx_arr) > 0:
                idx = idx_arr[np.argmax(h[idx_arr])]
                h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def profile_jacobian(h):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    profile = np.correlate(h, 1 - h, mode="full") * dx
    G = np.zeros((M, n))
    for m in range(M):
        k = m - (n - 1)
        js = max(0, k)
        je = min(n, n + k)
        if js < je:
            G[m, js:je] += dx * (1.0 - h[js-k:je-k])
        js2 = max(0, -k)
        je2 = min(n, n - k)
        if js2 < je2:
            G[m, js2:je2] -= dx * h[js2+k:je2+k]
    return profile, G

def slsqp_optimize(h0, maxiter=500, ftol=1e-14, time_limit=None, t_start=None):

    def _obj(x):

        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g

    def _eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0

        h = x[:-1]
        t = x[-1]
        prof = np.correlate(h, 1 - h, mode="full") * (2.0 / n)
        return t - prof

        _, G = profile_jacobian(h)

        J[:, :-1] = -G
        J[:, -1] = 1.0
        return J
    
    x0 = np.concatenate([h0, [c0 + 1e-3]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    cons = [
        {'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac},
        {'type': 'ineq', 'fun': _ineq_fun, 'jac': _ineq_jac}
    ]
    
    try:

            c_sol = exact_score(h_sol)
            if c_sol < c0 - 1e-15:
                return h_sol, c_sol
    except Exception:

    return h0.copy(), c0
```

Now let me think about the pairwise descent more carefully. I want to avoid the O(n^2) all-pairs sweep being too slow. Let me instead do:

1. Compute the current profile and its "active set" (top k lags)
2. For the active lags, find which pair (j,k) would most reduce the max

Actually, a simpler heuristic: the gradient of the max function (subgradient) points in the direction that increases the active lags. So the negative subgradient (projected to sum-0 subspace) is a descent direction. For the pair swap, we want the pair that most aligns with this direction.

The subgradient of max_k profile_k at h is:
sg = sum_{k in active} w_k * G[:,k] (column vector, the gradient of profile_k w.r.t. h)

where w_k >= 0, sum w_k = 1.

The projected descent direction is: d = sg - mean(sg) (project to sum-0 subspace).

For a pair swap (j, +ε) and (k, -ε), the directional derivative is: ε * (d[j] - d[k]).

To minimize this, we want d[j] - d[k] to be as negative as possible, i.e., d[j] << d[k].

So: find j with minimum d[j] and k with maximum d[k]. Apply swap.
