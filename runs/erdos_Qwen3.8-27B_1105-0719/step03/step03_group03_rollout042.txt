    """Project v onto {0<=x<=1, sum(x)=n/2} using water-filling."""
    target = n / 2.0
    # Binary search for lambda
    lo, hi = -1.0, 2.0  # lambda range
    for _ in range(100):
        mid = (lo + hi) / 2.0
        h = np.clip(v - mid, 0.0, 1.0)
        s = h.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - (lo+hi)/2.0, 0.0, 1.0)
    # Fix any tiny numerical issues
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        idx = int(np.argmax(1.0 - h)) if diff > 0 else int(np.argmax(h))
        h[idx] += diff
        h = np.clip(h, 0, 1)
    return h

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def project_feasible(v, n):
    target = n / 2.0
    lo, hi = -2.0, 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h = np.clip(v - mid, 0.0, 1.0)
        s = h.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        idx = int(np.argmax(1.0 - h)) if diff > 0 else int(np.argmax(h))
        h[idx] = np.clip(h[idx] + diff, 0, 1)
    return h

def build_gradient(h, n, dx):
    """Compute the gradient matrix G where G[k,i] = dp[k]/dh[i]."""
    M = 2 * n - 1
    G = np.zeros((M, n))
    center = n - 1
    G[center, :] = dx * (1.0 - 2.0 * h)
    for d in range(1, n):
        idx = np.arange(n - d)
        G[center + d, idx] += -dx * h[idx + d]
        G[center + d, idx + d] += dx * (1.0 - h[idx])
        G[center - d, idx] += dx * (1.0 - h[idx + d])
        G[center - d, idx + d] += -dx * h[idx]
    return G

def weighted_gradient(h, weights, n, dx):
    """Compute weighted sum of gradients."""
    G = build_gradient(h, n, dx)
    grad = G.T @ weights
    grad -= grad.mean()  # center to respect sum constraint

            c_try = exact_score(h_try)
            if c_try < best_c - 1e-14:

                best_c = c_try
                best_h = h.copy()
                improved = True
                break

                    improved = True

def interp_to(h, new_n):

    if n == new_n:

def pair_descent(h, max_iters, budget_fn):
    """Coordinate pair descent: try h[i]+=eps, h[j]-=eps for all pairs."""
    n = len(h)

    for it in range(max_iters):

                if h[j] <= 0.0:

                for eps in [0.01, 0.005, 0.002, 0.001, 0.0005]:
                    h_try = h.copy()
                    h_try[i] = min(1.0, h[i] + eps)

                    # Fix sum

                        k = int(np.argmax(1.0 - h_try)) if diff > 0 else int(np.argmax(h_try))
                        h_try[k] = np.clip(h_try[k] + diff, 0, 1)
                    c_try = exact_score(h_try)
                    if c_try < best_c - 1e-14:
                        h = h_try
                        best_c = c_try
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        if not improved:
            break
    
    return h, best_c

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))


def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx


def project_feasible(v, n):
    target = n / 2.0
    lo, hi = -2.0, 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(v - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        idx = int(np.argmax(1.0 - h)) if diff > 0 else int(np.argmax(h))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h


def build_grad(h, n, dx):
    M = 2 * n - 1
    G = np.zeros((M, n))
    center = n - 1
    G[center, :] = dx * (1.0 - 2.0 * h)
    for d in range(1, n):
        idx = np.arange(n - d)
        G[center + d, idx] = -dx * h[idx + d]
        G[center + d, idx + d] = dx * (1.0 - h[idx])
        G[center - d, idx] = dx * (1.0 - h[idx + d])
        G[center - d, idx + d] = -dx * h[idx]
    return G


def mw_descent(h0, max_iters, budget_fn, beta_range=(5.0, 80.0)):
    n = len(h0)
    dx = 2.0 / n
    h = h0.copy()
    best_c = exact_score(h)
    best_h = h.copy()
    stagnant = 0
    
    for t in range(max_iters):
        if budget_fn() < 40:
            break
        profile = compute_profile(h)
        c_max = float(profile.max())

            best_h = h.copy()
            stagnant = 0
        else:
            stagnant += 1
            if stagnant > 30:
                break

        beta = beta_range[0] + (beta_range[1] - beta_range[0]) * frac
        w = np.exp(beta * (profile - profile.max()))
        w /= w.sum()

        grad = G.T @ w

            h_try = project_feasible(h - alpha * grad, n)
            c_try = exact_score(h_try)
            if c_try < best_c - 1e-14:

            else:

def lp_active_set(h0, profile, active, dx, trust, rho, budget_fn):
    if not HAS_CVXPY:
        return None
    n = len(h0)
    target = n / 2.0
    G = build_grad(h0, n, dx)
    G_a = G[active, :]
    b_a = profile[active] - G_a @ h0
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    cons = [
        h_var >= 0,
        h_var <= 1,
        cp.sum(h_var) == target,
        h_var >= h0 - trust,
        h_var <= h0 + trust,
    ]
    for i in range(len(active)):
        cons.append(G_a[i, :] @ h_var - t_var <= -b_a[i])
    obj = cp.Minimize(t_var + rho * cp.norm1(h_var - h0))
    prob = cp.Problem(obj, cons)
    for solver in [cp.ECOS, cp.SCS, cp.CBC]:
        try:
            prob.solve(solver=solver, max_iters=5000, abstol=1e-10)
            break
        except Exception:
            continue
    if h_var.value is None:
        return None
    h_sol = np.array(h_var.value).flatten()
    if not np.isfinite(h_sol).all():
        return None
    return project_feasible(h_sol, n)

def slsqp_refine(h0, maxiter, tol, budget_fn):
    n = len(h0)
    dx = 2.0 / n
    M = 2 * n - 1
    
    def obj(x):
        return x[-1]
    
    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    
    def eq_fun(x):
        return x[:-1].sum() - n / 2.0
    
    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g
    
    def ineq_fun(x):
        h = x[:-1]
        return x[-1] - np.correlate(h, 1 - h, mode="full") * dx
    
    def ineq_jac(x):
        h = x[:-1]
        G = build_grad(h, n, dx)
        J = np.zeros((M, n + 1))
        J[:, :-1] = -G
        J[:, -1] = 1.0
        return J
    
    t0 = exact_score(h0) + 1e-3
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, t0 + 1.0)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': tol})
        if res.x is not None and np.isfinite(res.x).all():

            if np.isfinite(exact_score(hs)):

        pass

            c_try = exact_score(h_try)
            if c_try < best_c - 1e-14:
                h = h_try
                best_c = c_try
                best_h = h.copy()
                improved = True
                break
            elif c_try < c_max - 1e-14:
                h = h_try
                improved = True
        if not improved:
            break
    return best_h, best_c


def lp_solve_step(h0, profile, active, dx, trust, rho):
    if not HAS_CVXPY:
        return None
    n = len(h0)
    target = n / 2.0
    G = build_grad(h0, n, dx)
    G_a = G[active, :]
    b_a = profile[active] - G_a @ h0
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    cons = [
        h_var >= 0,
        h_var <= 1,
        cp.sum(h_var) == target,
        h_var >= h0 - trust,
        h_var <= h0 + trust,
    ]
    for i in range(len(active)):
        cons.append(G_a[i, :] @ h_var - t_var <= -b_a[i])
    obj = cp.Minimize(t_var + rho * cp.norm1(h_var - h0))
    prob = cp.Problem(obj, cons)
    for solver in [cp.ECOS, cp.SCS, cp.CBC]:
        try:
            prob.solve(solver=solver, max_iters=5000, abstol=1e-10)
            break
        except Exception:
            continue
    if h_var.value is None:
        return None
    h_sol = np.array(h_var.value).flatten()
    if not np.isfinite(h_sol).all():
        return None
    return project_feasible(h_sol, n)


def active_set_loop(h0, max_iters, budget_fn):
    n = len(h0)
    dx = 2.0 / n
    h = h0.copy()
    best_c = exact_score(h)
    best_h = h.copy()
    trust = 0.03
    stag = 0
    for _ in range(max_iters):
        if budget_fn() < 40:
            break
        profile = compute_profile(h)
        c_max = float(profile.max())
        if c_max < best_c - 1e-15:
            best_c = c_max
            best_h = h.copy()
            stag = 0
        else:
            stag += 1
        spread = c_max - np.median(profile)
        beam = max(0.003, min(0.02, 0.25 * spread))
        active = np.where(profile >= c_max - beam)[0]
        if len(active) == 0:
            active = np.array([np.argmax(profile)])
        if len(active) > 30:
            active = active[np.argsort(profile[active])[-30:]]
        accepted = False
        h_lp = lp_solve_step(h, profile, active, dx, trust, 0.002)
        if h_lp is not None:
            c_lp = exact_score(h_lp)
            if c_lp < best_c - 1e-14:
                h = h_lp
                best_c = c_lp
                best_h = h.copy()
                trust = min(trust * 1.3, 0.08)
                accepted = True
            else:
                for alpha in [0.7, 0.5, 0.3, 0.15]:
                    hb = project_feasible(alpha * h_lp + (1 - alpha) * h, n)
                    cb = exact_score(hb)
                    if cb < best_c - 1e-14:
                        h = hb
                        best_c = cb
                        best_h = h.copy()
                        accepted = True
                        break
                if not accepted:
                    trust *= 0.5
        if not accepted:
            w = np.exp(30.0 * (profile - c_max))
            w /= w.sum()
            G = build_grad(h, n, dx)
            grad = G.T @ w
            grad -= grad.mean()
            gn = np.linalg.norm(grad)
            if gn > 1e-14:
                grad /= gn
                for s in [trust, trust * 0.5, trust * 0.25, 0.003, 0.001, 0.0005]:
                    h_try = project_feasible(h - s * grad, n)
                    c_try = exact_score(h_try)
                    if c_try < best_c - 1e-14:
                        h = h_try
                        best_c = c_try
                        best_h = h.copy()
                        accepted = True
                        break
        if not accepted:
            trust *= 0.4
            if trust < 0.001 or stag > 50:
                break
    return best_h, best_c


def targeted_peaks(h, max_rounds, budget_fn):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best_c = exact_score(h)
    for _ in range(max_rounds):
        if budget_fn() < 30:
            break
        profile = compute_profile(h)
        c_max = float(profile.max())
        if c_max < best_c - 1e-15:
            best_c = c_max
        improved_any = False
        k_star = int(np.argmax(profile))
        d_star = k_star - (n - 1)
        if d_star == 0:
            for i in range(n):
                if 0.05 < h[i] < 0.95:
                    for eps in [0.005, 0.002, 0.001]:
                        h_try = h.copy()
                        h_try[i] -= eps
                        h_try[(i + 1) % n] += eps
                        h_try = project_feasible(h_try, n)
                        c_try = exact_score(h_try)
                        if c_try < best_c - 1e-14:
                            h = h_try
                            best_c = c_try
                            improved_any = True
                            break
                    if improved_any:
                        break
        elif d_star > 0:
            for i in range(n - d_star):
                if h[i + d_star] > h[i] + 0.02:
                    for eps in [0.005, 0.002, 0.001]:
                        move = min(eps, h[i + d_star] - h[i])
                        h_try = h.copy()
                        h_try[i + d_star] -= move
                        h_try[i] += move
                        c_try = exact_score(h_try)
                        if c_try < best_c - 1e-14:
                            h = h_try
                            best_c = c_try
                            improved_any = True
                            break
                    if improved_any:
                        break
        elif d_star < 0:
            da = -d_star
            for i in range(da, n):
                if h[i] > h[i - da] + 0.02:
                    for eps in [0.005, 0.002, 0.001]:
                        move = min(eps, h[i] - h[i - da])
                        h_try = h.copy()
                        h_try[i] -= move
                        h_try[i - da] += move
                        c_try = exact_score(h_try)
                        if c_try < best_c - 1e-14:
                            h = h_try
                            best_c = c_try
                            improved_any = True
                            break
                    if improved_any:
                        break
        if not improved_any:
            break
    return h, best_c


def slsqp_refine(h0, maxiter, tol):
    n = len(h0)
    dx = 2.0 / n
    M = 2 * n - 1

    def obj(x):
        return x[-1]

    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g

    def eq_fun(x):
        return x[:-1].sum() - n / 2.0

    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g

    def ineq_fun(x):
        h = x[:-1]
        return x[-1] - np.correlate(h, 1 - h, mode="full") * dx

    def ineq_jac(x):
        h = x[:-1]
        G = build_grad(h, n, dx)
        J = np.zeros((M, n + 1))
        J[:, :-1] = -G
        J[:, -1] = 1.0
        return J

    t0 = exact_score(h0) + 1e-3
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, t0 + 2.0)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': tol})
        if res.x is not None and np.isfinite(res.x).all():
            hs = project_feasible(res.x[:-1], n)
            if np.isfinite(exact_score(hs)):
                return hs
    except Exception:
        pass
    return None


def interp_to(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    fx = np.concatenate([[0.0], xs, [2.0]])
    fh = np.concatenate([[h[0]], h, [h[-1]]])
    hn = np.interp(xt, fx, fh)
    return project_feasible(hn, new_n)


def full_optimize(h_init, n_target, budget_fn):
    if budget_fn() < 80:
        return h_init, exact_score(h_init)
    h = interp_to(h_init, n_target)
    c = exact_score(h)
    if budget_fn() > 120:
        h1, c1 = lse_descent(h, max_iters=400, budget_fn=budget_fn,
                             beta_lo=5.0, beta_hi=80.0)
        if c1 < c:
            h, c = h1, c1
    if budget_fn() > 100:
        h2, c2 = active_set_loop(h, max_iters=80, budget_fn=budget_fn)
        if c2 < c:
            h, c = h2, c2
    if budget_fn() > 80:
        h3, c3 = lse_descent(h, max_iters=400, budget_fn=budget_fn,
                             beta_lo=40.0, beta_hi=200.0)
        if c3 < c:
            h, c = h3, c3
    if budget_fn() > 60:
        h4, c4 = targeted_peaks(h, max_rounds=50, budget_fn=budget_fn)
        if c4 < c:
            h, c = h4, c4
    if budget_fn() > 50:
        hs = slsqp_refine(h, maxiter=300, tol=1e-12)
        if hs is not None:
            cs = exact_score(hs)
            if cs < c:
                h, c = hs, cs
    return h, c


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()
    rng = np.random.RandomState(seed)

    def tl():
        return budget_s + t_start - time.perf_counter()

    parent = None
    try:
        parent = np.array(initial_h_values, dtype=np.float64)
    except (NameError, Exception):
        pass
    if parent is None or len(parent) < 4:
        parent = np.full(70, 0.5)

    n0 = len(parent)
    parent = project_feasible(parent, n0)
    best_h = parent.copy()
    best_c = exact_score(parent)

    h_sym = project_feasible((parent + parent[::-1]) / 2.0, n0)
    c_sym = exact_score(h_sym)
    if c_sym < best_c - 1e-12:
        best_h, best_c = h_sym, c_sym

    # Main optimization at parent dimension
    if tl() > 250:
        h1, c1 = full_optimize(best_h, n0, tl)
        if c1 < best_c:
            best_h, best_c = h1, c1

    # Explore nearby dimensions
    dims_to_try = [n0 + 2, n0 - 2, n0 + 4, n0 - 4, n0 + 6, n0 - 6,
                   n0 + 8, n0 - 8, n0 + 10, n0 - 10]
    for nd in dims_to_try:
        if tl() < 120 or nd < 20 or nd > 200:
            continue
        h2, c2 = full_optimize(best_h, nd, tl)
        if c2 < best_c:
            best_h, best_c = h2, c2

    # Random restart perturbation
    if tl() > 150:
        n_best = len(best_h)
        for eps in [0.02, 0.05, 0.1, 0.15]:
            if tl() < 80:
                break
            z = rng.randn(n_best)
            z -= z.mean()
            z /= (np.linalg.norm(z) / n_best + 1e-15)
            hp = project_feasible(best_h + eps * z, n_best)
            cp_ = exact_score(hp)
            if cp_ < best_c + 0.02:
                h3, c3 = full_optimize(hp, n_best, tl)
                if c3 < best_c:
                    best_h, best_c = h3, c3

    # Final polish
    if tl() > 60:
        n_best = len(best_h)
        hs = slsqp_refine(best_h, maxiter=500, tol=1e-13)
        if hs is not None:
            cs = exact_score(hs)
            if cs < best_c:
                best_h, best_c = hs, cs

    if tl() > 40:
        n_best = len(best_h)
        hf, cf = lse_descent(best_h, max_iters=300, budget_fn=tl,
                             beta_lo=50.0, beta_hi=250.0)
        if cf < best_c:
            best_h, best_c = hf, cf

    if tl() > 30:
        n_best = len(best_h)
        ht, ct = targeted_peaks(best_h, max_rounds=50, budget_fn=tl)
        if ct < best_c:
            best_h, best_c = ht, ct

    # Final validation
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    final_c = exact_score(best_h)

    try:
        er = evaluate_erdos_solution(best_h)
        if isinstance(er, (float, int)):
            final_c = min(final_c, float(er))
        elif isinstance(er, tuple) and len(er) >= 2:
            final_c = min(final_c, float(er[0]))
    except Exception:
        pass

    return best_h, final_c, n_final
