import numpy as np
from scipy.optimize import linprog
import time

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(-(1.0 - h))
        remaining = diff
        for i in range(n):
            idx = order[i]
            cap = 1.0 - h[idx]
            if cap <= 1e-15:
                continue
            add = min(remaining, cap)
            h[idx] += add
            remaining -= add
            if remaining <= 1e-14:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            if h[idx] <= 1e-15:
                continue
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-14:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] = float(np.clip(h[idx] + diff2, 0.0, 1.0))
    return h

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def grad_row(h, m, n):
    dx = 2.0 / n
    d = m - (n - 1)
    if d == 0:
        return dx * (1.0 - 2.0 * h)
    elif d > 0:
        g = np.zeros(n)
        g[d:] += dx * (1.0 - h[:-d])
        g[:-d] -= dx * h[d:]
        return g
    else:
        p = -d
        g = np.zeros(n)
        g[:-p] += dx * (1.0 - h[p:])
        g[p:] -= dx * h[:-p]
        return g

def grad_matrix(h, n):
    M = 2 * n - 1
    G = np.zeros((M, n))
    for m in range(M):
        G[m] = grad_row(h, m, n)
    return G

def trlp_step(h, delta, max_row_limit=None):
    n = len(h)
    M = 2 * n - 1
    P = compute_profile(h)  # shape (M,)
    G = grad_matrix(h, n)  # shape (M, n)
    
    # LP: minimize tau
    # s.t. G @ d - tau <= -P  (M constraints)
    #      sum(d) = 0          (1 equality)
    #      -min(h, delta) <= d_i <= min(1-h, delta)
    
    # If too many constraints, subsample
    if max_row_limit is not None and M > max_row_limit:
        # Keep the most important rows (highest P values)
        top_idx = np.argsort(-P)[:max_row_limit]
        P_use = P[top_idx]
        G_use = G[top_idx]
    else:
        P_use = P
        G_use = G
    
    M_use = len(P_use)
    
    # c: objective = [0, ..., 0, 1] (minimize tau)
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    # A_ub: [G_use, -1] @ [d, tau] <= -P_use
    A_ub = np.zeros((M_use, n + 1))
    A_ub[:, :n] = G_use
    A_ub[:, n] = -1.0
    b_ub = -P_use
    
    # A_eq: [1, ..., 1, 0] @ [d, tau] = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds on d_i: max(-delta, -h_i) <= d_i <= min(delta, 1-h_i)
    lo = np.maximum(-delta, -h)
    hi = np.minimum(delta, 1.0 - h)
    bounds = list(zip(lo, hi)) + [(None, None)]  # tau unbounded
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if not res.success or res.x is None:
        return None, None
    
    d = res.x[:n]
    tau = res.x[n]
    return d, tau

def line_search(h, d, delta, current_c):
    n = len(h)
    best_alpha = 0.0
    best_val = current_c
    best_h = h.copy()
    
    # Try a range of alphas
    alphas = [delta, 0.7*delta, 0.5*delta, 0.3*delta, 0.2*delta, 0.1*delta, 0.05*delta]
    for a in alphas:
        h_try = repair_feasible(h + a * d)
        c_try = exact_score(h_try)
        if c_try < best_val:
            best_val = c_try
            best_alpha = a
            best_h = h_try
    return best_h, best_alpha, best_val

def trlp_optimize(h_init, max_iters=200, initial_delta=0.05, time_deadline=None):
    h = h_init.copy()
    c = exact_score(h)
    best_h = h.copy()
    best_c = c
    delta = initial_delta
    stall = 0
    max_stall = 15
    
    for it in range(max_iters):
        if time_deadline is not None and (time.perf_counter() - time_deadline) > 0:
            break
        
        d, tau = trlp_step(h, delta)
        if d is None:
            # LP failed, reduce delta
            delta *= 0.5
            stall += 1
            if stall > max_stall or delta < 1e-4:
                break
            continue
        
        d_norm = np.linalg.norm(d)
        if d_norm < 1e-12:
            delta *= 0.5
            stall += 1
            if stall > max_stall or delta < 1e-4:
                break
            continue
        
        # Line search
        h_new, alpha, c_new = line_search(h, d, delta, c)
        
        if c_new < best_c - 1e-14:
            best_h = h_new.copy()
            best_c = c_new
            h = h_new
            c = c_new
            stall = 0
            # Increase delta slightly if we made good progress
            if alpha >= 0.8 * delta:
                delta = min(delta * 1.3, 0.3)
            else:
                delta = min(delta * 1.1, 0.3)
        elif c_new < c - 1e-14:
            # Local improvement but not global best
            h = h_new
            c = c_new
            stall = 0
        else:
            # No improvement
            stall += 1
            delta *= 0.5
            if stall > max_stall or delta < 1e-4:
                break
    
    return best_h, best_c

def interp_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    edges_old = np.linspace(0.0, 2.0, n + 1)
    edges_new = np.linspace(0.0, 2.0, new_n + 1)
    h_new = np.zeros(new_n)
    for k in range(new_n):
        a, b = edges_new[k], edges_new[k + 1]
        total = 0.0
        for j in range(n):
            lo = max(a, edges_old[j])
            hi = min(b, edges_old[j + 1])
            if hi > lo:
                total += h[j] * (hi - lo)
        h_new[k] = total / (b - a)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

def coordinate_polish(h, max_iters=200, time_deadline=None):
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    
    for it in range(max_iters):
        if time_deadline is not None and (time.perf_counter() - time_deadline) > 0:
            break
        
        profile = compute_profile(h)
        c = float(profile.max())
        thresh = c - 0.003
        active = np.where(profile >= thresh)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(profile))])
        
        # Compute gradient of active constraints
        g = np.zeros(n)
        for m in active:
            g += grad_row(h, m, n)
        g -= g.mean()  # preserve sum
        
        # Try paired coordinate moves
        improved = False
        order_desc = np.argsort(-h)
        order_asc = np.argsort(h)
        
        for attempt in range(min(20, n)):
            i = int(order_desc[attempt % n])
            j = int(order_asc[attempt % n])
            if i == j:
                continue
            # Try moving mass from i to j
            for frac in [0.1, 0.05, 0.02]:
                delta_move = frac * (h[i] - h[j])
                if abs(delta_move) < 1e-8:
                    continue
                h_try = h.copy()
                h_try[i] -= delta_move
                h_try[j] += delta_move
                if h_try.min() < -1e-10 or h_try.max() > 1 + 1e-10:
                    continue
                h_try = repair_feasible(h_try)
                s = exact_score(h_try)
                if s < best_c - 1e-14:
                    h = h_try
                    best_c = s
                    improved = True
                    break
            if improved:
                break
        
        if not improved:
            break
    
    return h, best_c

def golden_seed(n, rng):
    h = np.zeros(n)
    half = n // 2
    idx = np.arange(n)
    vals = (idx * 0.618033988749895) % 1.0
    top = np.argsort(vals)[-half:]
    h[top] = 1.0
    # Smooth slightly
    h_smooth = 0.3 * h + 0.7 * np.full(n, 0.5)
    return repair_feasible(h_smooth)

def sine_seed(n, freq, phase, rng):
    x = np.linspace(0, 2.0, n, endpoint=False) + 2.0/n
    h = 0.5 + 0.3 * np.sin(np.pi * freq * x + phase)
    h = np.clip(h, 0.0, 1.0)
    return repair_feasible(h)

def random_smooth_seed(n, rng):
    # Generate a smooth random function
    base = rng.uniform(0.2, 0.8, n)
    # Smooth with a small window
    for _ in range(3):
        base = np.convolve(base, np.ones(3)/3.0, mode='same')
    return repair_feasible(base)

def active_descent(h, max_iters=400, active_tol=0.003, time_deadline=None):
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    gr_phi = (np.sqrt(5.0) - 1.0) / 2.0
    momentum_vec = np.zeros(n)
    momentum = 0.3
    
    for it in range(max_iters):
        if time_deadline is not None and (time.perf_counter() - time_deadline) > 0:
            break
        
        profile = compute_profile(h)
        c = float(profile.max())
        thresh = c - active_tol
        active = np.where(profile >= thresh)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(profile))])
        if len(active) > 20:
            top = np.argsort(profile[active])[-20:]
            active = active[top]
        
        g = np.zeros(n)
        for m in active:
            g += grad_row(h, m, n)
        g -= g.mean()
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            break
        
        momentum_vec = momentum * momentum_vec + g
        mv_norm = np.linalg.norm(momentum_vec)
        if mv_norm < 1e-15:
            break
        d = -momentum_vec / mv_norm
        
        pos_mask = d > 1e-12
        neg_mask = d < -1e-12
        alpha_max = 1.0
        if pos_mask.any():
            alpha_max = min(alpha_max, float(((1.0 - h[pos_mask]) / d[pos_mask]).min()))
        if neg_mask.any():
            alpha_max = min(alpha_max, float((h[neg_mask] / (-d[neg_mask])).min()))
        alpha_max = max(alpha_max, 1e-10)
        alpha_max = min(alpha_max, 0.5)
        
        # Golden section line search
        best_alpha = 0.0
        best_val = c
        best_h = h
        a_ls, b_ls = 0.0, alpha_max
        for _ in range(20):
            c1 = b_ls - gr_phi * (b_ls - a_ls)
            c2 = a_ls + gr_phi * (b_ls - a_ls)
            h1 = repair_feasible(h + c1 * d)
            h2 = repair_feasible(h + c2 * d)
            f1 = exact_score(h1)
            f2 = exact_score(h2)
            if f1 < best_val:
                best_val = f1
                best_alpha = c1
                best_h = h1.copy()
            if f2 < best_val:
                best_val = f2
                best_alpha = c2
                best_h = h2.copy()
            if f1 < f2:
                b_ls = c2
            else:
                a_ls = c1
            if b_ls - a_ls < 1e-14:
                break
        
        if best_val >= c - 1e-15:
            break
        h = best_h
        if best_val < best_c:
            best_c = best_val
    
    return h, best_c

def repair_feasible(vec):

    target = n / 2.0

    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(-(1.0 - h))
        remaining = diff
        for i in range(n):
            idx = order[i]
            cap = 1.0 - h[idx]
            if cap <= 1e-15:
                continue
            add = min(remaining, cap)
            h[idx] += add
            remaining -= add
            if remaining <= 1e-14:
                break
    else:
        order = np.argsort(h)
        remaining = -diff

            rem = min(remaining, h[idx])

def grad_row(h, m, n):

    elif d > 0:

        g[d:] += dx * (1.0 - h[:-d])
        g[:-d] -= dx * h[d:]
        return g
    else:
        p = -d
        g = np.zeros(n)
        g[:-p] += dx * (1.0 - h[p:])
        g[p:] -= dx * h[:-p]
        return g

    M = 2 * n - 1
    P = compute_profile(h)
    G = grad_matrix(h, n)
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    A_ub = np.zeros((M, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -P
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    lo = np.maximum(-delta, -h)
    hi = np.minimum(delta, 1.0 - h)
    bounds = list(zip(lo, hi)) + [(None, None)]
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')

        h_lp, c_lp = trlp_optimize(best_h, max_iters=80, initial_delta=0.01, deadline=dl)
        if c_lp < best_c:
            best_h, best_c = h_lp, c_lp

    # Phase 8: Final active descent
    if time_left() > 5:
        dl = time.perf_counter() + min(5, time_left() * 0.5)
        h_ad, c_ad = active_descent(best_h, max_iters=100, active_tol=0.001, deadline=dl)
        if c_ad < best_c:
            best_h, best_c = h_ad, c_ad

    # Final: try a few more dimension tweaks if time allows
    if time_left() > 8:
        n_b = len(best_h)
        for new_n in [n_b - 2, n_b + 2]:
            if time_left() < 5 or new_n < 30 or new_n > 150:
                continue
            h_interp = interp_to_n(best_h, new_n)
            dl = time.perf_counter() + min(4, time_left() * 0.5)
            h_lp, c_lp = trlp_optimize(h_interp, max_iters=40, initial_delta=0.015, deadline=dl)
            if c_lp < best_c:
                best_h, best_c = h_lp, c_lp

    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)

    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            val = float(eval_result)
            if np.isfinite(val) and val < final_c:
                final_c = val
        elif isinstance(eval_result, tuple) and len(eval_result) >= 2:
            val = float(eval_result[0])
            if np.isfinite(val) and val < final_c:
                final_c = val
    except Exception:
        pass

    return best_h, final_c, len(best_h)
