import numpy as np
from scipy.optimize import minimize, linprog
from scipy.special import logsumexp
import time

def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def c5_of_h(h, dx):
    return compute_corrs(h).max() * dx

def grad_corr_k(h, k, n):
    offset = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -offset)
    i_hi = min(n - 1, n - 1 - offset)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx + offset] += (1.0 - h[idx])
    grad[idx] += (-h[idx + offset])
    return grad

def smooth_max_obj(h, dx, beta):
    corrs = compute_corrs(h)
    return logsumexp(beta * corrs) / beta * dx

def smooth_max_grad(h, dx, beta):
    n = len(h)
    corrs = compute_corrs(h)
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones_like(corrs) / len(corrs)
    else:
        w /= w_sum
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] > 1e-13:
            grad += w[k] * grad_corr_k(h, k, n)
    return grad * dx

def slsqp_direct(h0, dx, target_sum, beta, maxiter):
    n = len(h0)
    
    def obj(x):
        corrs = compute_corrs(x)
        return logsumexp(beta * corrs) / beta * dx
    
    def grad_obj(x):
        return smooth_max_grad(x, dx, beta)
    
    def c_fun(x):
        return x.sum() - target_sum
    
    def c_jac(x):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': c_fun, 'jac': c_jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0, jac=grad_obj, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        h_out = project_feasible(res.x, target_sum)
    except Exception:
        h_out = project_feasible(h0, target_sum)
    return h_out, c5_of_h(h_out, dx)

def exact_1d_convex_pl(corr, d, alpha_lo, alpha_hi, n_iter=80):
    """Minimize max_k (corr[k] + alpha * d[k]) for alpha in [alpha_lo, alpha_hi]."""
    # Convex piecewise-linear function -> ternary search works well
    for _ in range(n_iter):
        m1 = alpha_lo + (alpha_hi - alpha_lo) / 3.0
        m2 = alpha_hi - (alpha_hi - alpha_lo) / 3.0
        f1 = (corr + m1 * d).max()
        f2 = (corr + m2 * d).max()
        if f1 < f2:
            alpha_hi = m2
        else:
            alpha_lo = m1
    alpha = (alpha_lo + alpha_hi) / 2.0
    return alpha, (corr + alpha * d).max()

def lp_direction(h, dx, target_sum, corrs, n_active, step):
    """Find LP direction to reduce max correlation."""
    n = len(h)
    max_c = corrs.max()
    
    # Active set: top correlations
    active_k = np.argsort(corrs)[-n_active:]
    
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_rows = []
    b_rows = []
    for k in active_k:
        row = np.zeros(n + 1)
        row[:n] = dx * grad_corr_k(h, k, n)
        row[n] = -1.0
        A_rows.append(row)
        b_rows.append(-corrs[k] * dx - 1e-13)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)

    bounds = []
    for i in range(n):
        bounds.append((max(-h[i], -step), min(1.0 - h[i], step)))
    bounds.append((0.0, None))

    try:
        res = linprog(c_obj, A_ub=np.array(A_rows), b_ub=np.array(b_rows),
                      A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    except Exception:
        return None
    if res.status != 0:
        return None
    return res.x[:n]

def lp_with_line_search(h0, dx, target_sum, max_iters=500, n_active=60, init_step=0.15):
    """LP direction + exact 1D line search."""
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    step = init_step
    stall = 0

    for it in range(max_iters):
        corrs = compute_corrs(h)
        c5_val = corrs.max() * dx

        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 50:
            step *= 0.7
            stall = 0
            if step < 1e-8:
                break

        d = lp_direction(h, dx, target_sum, corrs, n_active, step)
        if d is None:
            break
        if np.max(np.abs(d)) < 1e-14:
            break

        # Find feasible alpha range
        # h + alpha*d must satisfy 0 <= h_i + alpha*d_i <= 1
        alpha_lo = 0.0
        alpha_hi = 1.0
        for i in range(n):
            if d[i] < -1e-15:
                alpha_hi = min(alpha_hi, h[i] / (-d[i]))
            elif d[i] > 1e-15:
                alpha_lo = max(alpha_lo, (h[i] - 1.0) / d[i])
        alpha_lo = max(alpha_lo, 0.0)
        alpha_hi = min(alpha_hi, 1.0)
        
        if alpha_hi <= alpha_lo:
            step *= 0.5
            if step < 1e-8:
                break
            continue

        # Exact 1D convex optimization
        alpha_opt, f_opt = exact_1d_convex_pl(corrs, d, alpha_lo, alpha_hi, 60)
        
        # Only apply if alpha_opt > 0 (we want to move forward)
        if alpha_opt > 1e-10:
            h_new = project_feasible(h + alpha_opt * d, target_sum)
            new_c5 = c5_of_h(h_new, dx)
            if new_c5 < best_c5 - 1e-15:
                h = h_new
                step = min(step * 1.1, init_step)

    return best_h, best_c5

def pair_exact(h0, dx, target_sum, n_top=30, n_iter=10):
    """Pair transfer with exact 1D convex optimization."""
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)

    for it in range(n_iter):
        corrs = compute_corrs(h)
        
        # Compute gradient for selecting pairs
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            c_k = corrs[k]
            if c_k > corrs.max() * 0.9:
                grad += c_k * grad_corr_k(h, k, n)
        
        dec_idx = np.argsort(grad)[-n_top:]
        inc_idx = np.argsort(grad)[:n_top]

        improved = False
        for i in dec_idx:
            for j in inc_idx:
                if i == j:
                    continue
                
                # Direction: h[i] += eps, h[j] -= eps
                d_corrs = np.zeros(2 * n - 1)
                for k in range(2 * n - 1):
                    gk = grad_corr_k(h, k, n)
                    d_corrs[k] = gk[i] - gk[j]
                
                # Feasible range for eps
                eps_lo = max(-h[i], h[j] - 1.0)  # h[i]+eps >= 0, h[j]-eps <= 1
                eps_hi = min(1.0 - h[i], h[j])   # h[i]+eps <= 1, h[j]-eps >= 0
                
                if eps_hi - eps_lo < 1e-12:
                    continue
                
                eps_opt, f_opt = exact_1d_convex_pl(corrs, d_corrs, eps_lo, eps_hi, 50)
                
                if abs(eps_opt) > 1e-12:
                    new_h = h.copy()
                    new_h[i] += eps_opt
                    new_h[j] -= eps_opt
                    new_h = np.clip(new_h, 0.0, 1.0)
                    new_h = project_feasible(new_h, target_sum)
                    new_c5 = c5_of_h(new_h, dx)
                    if new_c5 < best_c5 - 1e-14:
                        h = new_h
                        best_c5 = new_c5
                        best_h = h.copy()
                        corrs = compute_corrs(h)
                        improved = True

        if not improved:
            break

    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.special import logsumexp
import time


def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)


def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def c5_of_h(h, dx):
    return compute_corrs(h).max() * dx


def grad_corr_k(h, k, n):
    offset = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -offset)
    i_hi = min(n - 1, n - 1 - offset)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx + offset] += (1.0 - h[idx])
    grad[idx] += (-h[idx + offset])
    return grad


def grad_corr_all(h, n):
    """Precompute gradients for all k."""
    m = 2 * n - 1
    G = np.zeros((m, n))
    for k in range(m):
        G[k] = grad_corr_k(h, k, n)
    return G


def smooth_max_grad(h, dx, beta):
    n = len(h)
    corrs = compute_corrs(h)
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones_like(corrs) / len(corrs)
    else:
        w /= w_sum
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] > 1e-13:
            grad += w[k] * grad_corr_k(h, k, n)
    return grad * dx


def slsqp_direct(h0, dx, target_sum, beta, maxiter):
    n = len(h0)

    def obj(x):
        corrs = compute_corrs(x)
        return logsumexp(beta * corrs) / beta * dx

    def grad_obj(x):
        return smooth_max_grad(x, dx, beta)

    def c_fun(x):
        return x.sum() - target_sum

    def c_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': c_fun, 'jac': c_jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0, jac=grad_obj, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        h_out = project_feasible(res.x, target_sum)
    except Exception:
        h_out = project_feasible(h0, target_sum)
    return h_out, c5_of_h(h_out, dx)


def exact_1d_convex_pl(corr, d, alpha_lo, alpha_hi, n_iter=70):
    """Minimize max_k (corr[k] + alpha * d[k]), convex PL in alpha."""
    for _ in range(n_iter):
        m1 = alpha_lo + (alpha_hi - alpha_lo) / 3.0
        m2 = alpha_hi - (alpha_hi - alpha_lo) / 3.0
        f1 = (corr + m1 * d).max()
        f2 = (corr + m2 * d).max()
        if f1 <= f2:
            alpha_hi = m2
        else:
            alpha_lo = m1
    alpha = (alpha_lo + alpha_hi) / 2.0
    return alpha, (corr + alpha * d).max()


def feasible_alpha_range(h, d):
    """Find alpha range where h + alpha*d stays in [0,1]."""
    alpha_lo = 0.0
    alpha_hi = 1e10
    for i in range(len(h)):
        if d[i] < -1e-15:
            alpha_hi = min(alpha_hi, h[i] / (-d[i]))
        elif d[i] > 1e-15:
            alpha_lo = max(alpha_lo, (h[i] - 1.0) / d[i])
    alpha_lo = max(alpha_lo, 0.0)
    alpha_hi = min(alpha_hi, 1.0)
    if alpha_hi < alpha_lo:
        alpha_lo, alpha_hi = alpha_hi, alpha_lo
    return alpha_lo, alpha_hi


def lp_polish_exact(h0, dx, target_sum, max_iters=800, n_active=50, init_step=0.2):
    """LP direction + exact 1D convex line search."""
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    step = init_step
    stall = 0

    for it in range(max_iters):
        corrs = compute_corrs(h)
        c5_val = corrs.max() * dx

        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 60:
            step *= 0.65
            stall = 0
            if step < 1e-9:
                break

        active_k = np.argsort(corrs)[-n_active:]

        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        A_rows = []
        b_rows = []
        for k in active_k:
            row = np.zeros(n + 1)
            row[:n] = dx * grad_corr_k(h, k, n)
            row[n] = -1.0
            A_rows.append(row)
            b_rows.append(-corrs[k] * dx - 1e-13)

        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)

        bounds = []
        for i in range(n):
            bounds.append((max(-h[i], -step), min(1.0 - h[i], step)))
        bounds.append((0.0, None))

        try:
            res = linprog(c_obj, A_ub=np.array(A_rows), b_ub=np.array(b_rows),
                          A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
        except Exception:
            break
        if res.status != 0:
            break

        d = res.x[:n]
        if np.max(np.abs(d)) < 1e-14:
            break

        alpha_lo, alpha_hi = feasible_alpha_range(h, d)
        if alpha_hi <= alpha_lo + 1e-12:
            step *= 0.5
            if step < 1e-9:
                break
            continue

        alpha_opt, f_opt = exact_1d_convex_pl(corrs, d, alpha_lo, alpha_hi, 60)

        if alpha_opt > 1e-10:
            h_new = project_feasible(h + alpha_opt * d, target_sum)
            new_c5 = c5_of_h(h_new, dx)
            if new_c5 < best_c5 - 1e-15:
                h = h_new
                step = min(step * 1.15, init_step)
        else:
            step *= 0.7
            if step < 1e-9:
                break

    return best_h, best_c5


def pair_transfer_exact(h0, dx, target_sum, n_top=25, n_iter=8):
    """Pair transfer with exact convex 1D optimization."""
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    m = 2 * n - 1

    for it in range(n_iter):
        corrs = compute_corrs(h)
        G = grad_corr_all(h, n)

        # Weighted gradient: emphasize top correlations
        w = np.zeros(m)
        top_frac = 0.85
        thr = np.quantile(corrs, top_frac)
        w[corrs >= thr] = (corrs[corrs >= thr] - thr + 0.01)
        w = w / (w.sum() + 1e-30)

        grad = (G * w[:, None]).sum(axis=0)

        dec_idx = np.argsort(grad)[-n_top:]
        inc_idx = np.argsort(grad)[:n_top]

        improved = False
        for i in dec_idx:
            for j in inc_idx:
                if i == j:
                    continue
                # Direction: h[i] += eps, h[j] -= eps
                d_corrs = G[:, i] - G[:, j]

                eps_lo = max(-h[i], h[j] - 1.0)
                eps_hi = min(1.0 - h[i], h[j])
                if eps_hi - eps_lo < 1e-12:
                    continue

                eps_opt, f_opt = exact_1d_convex_pl(corrs, d_corrs, eps_lo, eps_hi, 50)

                if abs(eps_opt) > 1e-12:
                    new_h = h.copy()
                    new_h[i] += eps_opt
                    new_h[j] -= eps_opt
                    new_h = np.clip(new_h, 0.0, 1.0)
                    new_h = project_feasible(new_h, target_sum)
                    new_c5 = c5_of_h(new_h, dx)
                    if new_c5 < best_c5 - 1e-14:
                        h = new_h
                        best_c5 = new_c5
                        best_h = h.copy()
                        corrs = compute_corrs(h)
                        G = grad_corr_all(h, n)
                        improved = True

        if not improved:
            break

    return best_h, best_c5


def slsqp_z(h0, dx, target_sum, beta, maxiter):
    n = len(h0)

    def sigmoid_2z(z):
        return 1.0 / (1.0 + np.exp(-np.clip(2.0 * z, -500, 500)))

    def obj(z):
        h = sigmoid_2z(z)
        corrs = compute_corrs(h)
        return logsumexp(beta * corrs) / beta * dx

    def grad_obj(z):
        h = sigmoid_2z(z)
        grad_h = smooth_max_grad(h, dx, beta)
        return 2.0 * h * (1.0 - h) * grad_h

    def c_fun(z):
        return sigmoid_2z(z).sum() - target_sum

    def c_jac(z):
        h = sigmoid_2z(z)
        return 2.0 * h * (1.0 - h)

    cons = [{'type': 'eq', 'fun': c_fun, 'jac': c_jac}]
    h0_clip = np.clip(h0, 1e-8, 1.0 - 1e-8)
    z0 = -0.5 * np.log((1.0 - h0_clip) / h0_clip)
    try:
        res = minimize(obj, z0, jac=grad_obj, method='SLSQP',
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        h_out = project_feasible(sigmoid_2z(res.x), target_sum)
    except Exception:
        h_out = project_feasible(h0_clip, target_sum)
    return h_out, c5_of_h(h_out, dx)


def momentum_subgrad(h0, dx, target_sum, max_iter=8000, beta_smooth=50.0):

def optimize_at_n(h_seed, n_target, time_left_fn):
    """Optimize at a specific n by interpolating the seed."""
    dx = 2.0 / n_target
    target_sum = n_target / 2.0

    if len(h_seed) == n_target:
        h0 = project_feasible(h_seed.copy(), target_sum)
    else:
        x_orig = np.linspace(0.0, 2.0, len(h_seed), endpoint=False) + 1.0 / len(h_seed)
        x_new = np.linspace(0.0, 2.0, n_target, endpoint=False) + 1.0 / n_target
        h0 = np.interp(x_new, x_orig, h_seed)
        h0 = project_feasible(h0, target_sum)

    h_opt, c5_val = full_optimize(h0, dx, target_sum, time_left_fn)
    return h_opt, c5_val, n_target
