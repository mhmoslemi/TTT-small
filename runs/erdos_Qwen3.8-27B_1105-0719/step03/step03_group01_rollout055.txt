def pair_move_optimal(h, dx, i, j):
    """Find optimal transfer amount from i to j, return (new_h, new_c5, a_opt)"""
    n = len(h)
    n_total = 2 * n - 1
    corrs = np.correlate(h, 1.0 - h, mode='full')
    
    a_min = max(h[i] - 1.0, -h[j])
    a_max = min(h[i], 1.0 - h[j])
    
    if a_max <= a_min:
        return h, corrs.max() * dx, 0.0
    
    # Compute slopes for each lag
    alphas = np.zeros(n_total)
    for k in range(n_total):
        d = k - (n - 1)
        alpha = 0.0
        # i as first factor
        if 0 <= i + d < n:
            alpha -= (1.0 - h[i + d])
        # i as second factor
        if 0 <= i - d < n:
            alpha += h[i - d]
        # j as first factor
        if 0 <= j + d < n:
            alpha += (1.0 - h[j + d])
        # j as second factor
        if 0 <= j - d < n:
            alpha -= h[j - d]
        alphas[k] = alpha
    
    # Binary search on t
    t_lo = corrs.min()  # lower bound for t
    t_hi = corrs.max()  # upper bound (at a=0)
    
    # Tighten upper bound
    vals_at_min = corrs + alphas * a_min
    vals_at_max = corrs + alphas * a_max
    t_hi = max(t_hi, vals_at_min.max(), vals_at_max.max())
    # Actually we want the minimum, so t_hi should be an upper bound on the minimum
    # The minimum is at most max(corrs) (at a=0)
    t_hi = corrs.max()
    
    for _ in range(60):
        t_mid = (t_lo + t_hi) / 2.0
        # Check feasibility: exists a in [a_min, a_max] with beta_d + alpha_d * a <= t_mid for all d
        lower = a_min
        upper = a_max
        feasible = True
        for k in range(n_total):
            if abs(alphas[k]) < 1e-15:
                if corrs[k] > t_mid + 1e-15:
                    feasible = False
                    break
            elif alphas[k] > 0:
                ub = (t_mid - corrs[k]) / alphas[k]
                if ub < upper:
                    upper = ub
            else:
                lb = (t_mid - corrs[k]) / alphas[k]
                if lb > lower:
                    lower = lb
        if not feasible or lower > upper + 1e-12:
            t_lo = t_mid
        else:
            t_hi = t_mid
    
    t_opt = (t_lo + t_hi) / 2.0
    # Find optimal a for t_opt (use t_hi which is feasible)
    t_use = t_hi
    lower = a_min
    upper = a_max
    for k in range(n_total):
        if abs(alphas[k]) < 1e-15:
            continue
        elif alphas[k] > 0:
            ub = (t_use - corrs[k]) / alphas[k]
            if ub < upper:
                upper = ub
        else:
            lb = (t_use - corrs[k]) / alphas[k]
            if lb > lower:
                lower = lb
    a_opt = (lower + upper) / 2.0
    a_opt = max(a_min, min(a_max, a_opt))
    
    h_new = h.copy()
    h_new[i] -= a_opt
    h_new[j] += a_opt
    h_new = np.clip(h_new, 0.0, 1.0)
    c5_new = c5_of_h(h_new, dx)
    return h_new, c5_new, a_opt

    # Change in corr_k = -a * contribs[i][k] + a * contribs[j][k] = a * (contribs[j][k] - contribs[i][k])
    alphas = contribs[j] - contribs[i]

    n_total = 2 * n - 1
    d = np.arange(-(n-1), n)  # offsets
    
    # For index i as first factor: valid when 0 <= i+d < n
    mask_fi = (i + d >= 0) & (i + d < n)
    # For index i as second factor: valid when 0 <= i-d < n
    mask_si = (i - d >= 0) & (i - d < n)
    # For index j as first factor
    mask_fj = (j + d >= 0) & (j + d < n)
    # For index j as second factor
    mask_sj = (j - d >= 0) & (j - d < n)
    
    alphas = np.zeros(n_total)
    alphas[mask_fi] -= (1.0 - h[i + d[mask_fi]])

    alphas[mask_fj] += (1.0 - h[j + d[mask_fj]])
    alphas[mask_sj] -= h[j - d[mask_sj]]
    
    return alphas

def compute_G(h, n):
    """Compute G matrix: G[k, p] = [0<=p+d<n]*(1-h[p+d]) - [0<=p-d<n]*h[p-d]"""
    n_total = 2 * n - 1
    G = np.zeros((n_total, n))
    d_arr = np.arange(-(n-1), n)
    p_arr = np.arange(n)
    
    # For each d, compute the contribution
    # G[k, p] where k = d + n - 1
    # First factor: p + d in [0, n) → G[k, p] += (1 - h[p+d])
    # Second factor: p - d in [0, n) → G[k, p] -= h[p-d]
    
    for d in range(-(n-1), n):
        k = d + (n-1)
        # First: indices p where 0 <= p+d < n
        valid_f = (p_arr + d >= 0) & (p_arr + d < n)
        G[k, valid_f] += (1.0 - h[p_arr[valid_f] + d])
        # Second: indices p where 0 <= p-d < n
        valid_s = (p_arr - d >= 0) & (p_arr - d < n)
        G[k, valid_s] -= h[p_arr[valid_s] - d]
    return G

    h_new[i] -= a_opt
    h_new[j] += a_opt
    h_new = np.clip(h_new, 0.0, 1.0)
    return h_new, t_hi * dx

def coordinate_descent(h, dx, max_sweeps, seed, t0, time_limit):
    n = len(h)
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() - t0 > time_limit:
            break
        corrs = compute_corrs(h)
        cur_c5 = corrs.max() * dx
        sweep_best = cur_c5
        perm = rng.permutation(n)
        for a in range(n):
            if time.time() - t0 > time_limit:
                break
            for b in range(a + 1, n):
                i, j = int(perm[a]), int(perm[b])
                h_new, c5_new = pair_move_optimal(h, dx, i, j, corrs)
                if h_new is not None and c5_new < sweep_best - 1e-13:
                    h = h_new
                    corrs = compute_corrs(h)
                    sweep_best = c5_new
        if sweep_best < best_c5 - 1e-13:
            best_c5 = sweep_best
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall >= 3:
                break
    return best_h, best_c5

def slsqp_obj(x, dx, n):
    return x[0]

def slsqp_jac(x, dx, n):
    j = np.zeros(n + 1)
    j[0] = 1.0
    return j

def slsqp_ineq_fun(x, dx, n):
    t = x[0]
    h = x[1:]
    corrs = compute_corrs(h)
    return t - dx * corrs

def slsqp_ineq_jac(x, dx, n):
    h = x[1:]
    n_total = 2 * n - 1
    J = np.zeros((n_total, n + 1))
    J[:, 0] = 1.0
    p_arr = np.arange(n)
    for d in range(-(n - 1), n):
        k = d + (n - 1)
        vf = (p_arr + d >= 0) & (p_arr + d < n)
        vs = (p_arr - d >= 0) & (p_arr - d < n)
        g = np.zeros(n)
        if vf.any():
            g[vf] = (1.0 - h[p_arr[vf] + d])
        if vs.any():
            g[vs] += (-h[p_arr[vs] - d])
        J[k, 1:] = -dx * g
    return J

def slsqp_eq_fun(x, dx, n):
    return x[1:].sum() - n / 2.0

def slsqp_eq_jac(x, dx, n):
    j = np.zeros(n + 1)
    j[1:] = 1.0
    return j

def slsqp_optimize(h0, dx, t0, time_limit, maxiter=150):
    n = len(h0)
    c0 = c5_of_h(h0, dx)
    x0 = np.zeros(n + 1)
    x0[0] = c0 + 1e-4
    x0[1:] = h0
    bounds = [(0.0, 0.6)] + [(0.0, 1.0)] * n
    cons = [
        {'type': 'eq', 'fun': lambda x: slsqp_eq_fun(x, dx, n), 'jac': lambda x: slsqp_eq_jac(x, dx, n)},
        {'type': 'ineq', 'fun': lambda x: slsqp_ineq_fun(x, dx, n), 'jac': lambda x: slsqp_ineq_jac(x, dx, n)}
    ]
    try:
        res = minimize(slsqp_obj, x0, jac=lambda x: slsqp_jac(x, dx, n),
                       bounds=bounds, constraints=cons,
                       method='SLSQP', options={'maxiter': maxiter, 'ftol': 1e-10})
        h_new = project_feasible(res.x[1:], n / 2.0)
        c5_new = c5_of_h(h_new, dx)
        return h_new, c5_new
    except Exception:
        return h0, c0

def interpolate_h(h_old, n_new):
    n_old = len(h_old)
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return h_new

def try_resolution(h_best, c5_best, n_new, t0, time_limit, seed):
    dx_new = 2.0 / n_new
    h_interp = interpolate_h(h_best, n_new)
    h_interp = project_feasible(h_interp, n_new / 2.0)
    c5_init = c5_of_h(h_interp, dx_new)
    if time.time() - t0 > time_limit - 20:
        if c5_init < c5_best:
            return h_interp, c5_init
        return None, c5_best
    h_cd, c5_cd = coordinate_descent(h_interp, dx_new, 30, seed, t0, time_limit - 30)
    if c5_cd < c5_best:
        return h_cd, c5_cd
    if c5_init < c5_best:
        return h_interp, c5_init
    return None, c5_best

def perturb(h, sigma, seed):
    rng = np.random.RandomState(seed)
    target = len(h) / 2.0
    h_p = h + sigma * rng.randn(len(h))
    return project_feasible(h_p, target)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    hard_stop = t0 + budget_s - 5.0

    h0 = np.array(initial_h_values, dtype=float)
    n = len(h0)
    dx = 2.0 / n
    target = n / 2.0
    h0 = project_feasible(h0, target)
    best_h = h0.copy()
    best_c5 = c5_of_h(h0, dx)

    # Phase 1: Coordinate descent from parent
    for attempt in range(4):
        if time.time() > hard_stop - 200:
            break
        tl = min(hard_stop - 200, t0 + budget_s * (0.12 + 0.08 * attempt))
        s = seed + attempt * 13
        h_cd, c5_cd = coordinate_descent(best_h, dx, 50, s, t0, tl)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd

    # Phase 2: SLSQP polish
    for attempt in range(3):
        if time.time() > hard_stop - 150:
            break
        tl = min(hard_stop - 150, t0 + budget_s * (0.3 + 0.05 * attempt))
        h_sl, c5_sl = slsqp_optimize(best_h, dx, t0, tl, maxiter=120)
        if c5_sl < best_c5:
            best_c5 = c5_sl
            best_h = h_sl
        # CD after SLSQP
        if time.time() < hard_stop - 80:
            h_cd2, c5_cd2 = coordinate_descent(h_sl, dx, 20, seed + 100 + attempt, t0, hard_stop - 80)
            if c5_cd2 < best_c5:
                best_c5 = c5_cd2
                best_h = h_cd2

    # Phase 3: Perturbation multi-start
    sigmas = [0.03, 0.05, 0.02, 0.04, 0.01, 0.06]
    for pi, sig in enumerate(sigmas):
        if time.time() > hard_stop - 100:
            break
        h_p = perturb(best_h, sig, seed + 200 + pi * 7)
        tl = min(hard_stop - 100, t0 + budget_s * (0.55 + 0.04 * pi))
        h_cd, c5_cd = coordinate_descent(h_p, dx, 30, seed + 300 + pi, t0, tl)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd

    # Phase 4: Resolution changes
    n_candidates = [120, 124, 128, 132, 136, 126, 122, 130]
    for nc in n_candidates:
        if time.time() > hard_stop - 60:
            break
        if nc == n:
            continue
        tl = min(hard_stop - 60, t0 + budget_s * 0.85)
        h_new, c5_new = try_resolution(best_h, best_c5, nc, t0, tl, seed + nc)
        if h_new is not None and c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new
            n = len(h_new)
            dx = 2.0 / n

    # Phase 5: Final intensive CD
    if time.time() < hard_stop - 10:
        n_b = len(best_h)
        dx_b = 2.0 / n_b
        h_final, c5_final = coordinate_descent(best_h, dx_b, 40, seed + 999, t0, hard_stop - 10)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final

    # Final validation
    n_final = len(best_h)
    dx_final = 2.0 / n_final
    best_h = project_feasible(best_h, n_final / 2.0)
    best_c5 = c5_of_h(best_h, dx_final)
    return best_h, best_c5, n_final
