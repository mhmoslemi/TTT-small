def repair_feasible(vec, target=None):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    if not np.isfinite(h).all():
        h[~np.isfinite(h)] = 0.5
    if target is None:

    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:

        remaining = diff

            cap = 1.0 - h[idx]
            if cap <= 0:
                break

            h[idx] += add

    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            if h[idx] <= 1e-15:
                continue
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining < 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-13:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] = float(np.clip(h[idx] + diff2, 0.0, 1.0))
    return h


def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))


def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx


def grad_one_row(h, m):
    n = len(h)
    dx = 2.0 / n
    d = m - (n - 1)
    g = np.zeros(n)
    if d == 0:
        g[:] = dx * (1.0 - 2.0 * h)
    elif d > 0:
        g[d:] += dx * (1.0 - h[:n - d])
        g[:n - d] -= dx * h[d:]
    else:
        p = -d
        g[:n - p] += dx * (1.0 - h[p:])
        g[p:] -= dx * h[:n - p]
    return g


def lp_direction(h, active, profile, delta):
    n = len(h)
    K = len(active)
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_ub = np.zeros((K, n + 1))
    b_ub = np.zeros(K)
    for i, m in enumerate(active):
        g = grad_one_row(h, m)
        A_ub[i, :n] = g
        A_ub[i, -1] = -1.0
        b_ub[i] = -profile[m]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for i in range(n):
        lo = max(-delta, -h[i] + 1e-10)
        hi = min(delta, 1.0 - h[i] - 1e-10)
        if lo > hi:
            lo, hi = -1e-10, 1e-10
        bounds.append((lo, hi))
    bounds.append((0.0, None))
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs-ds')
        if res.success and res.x is not None:
            return res.x[:n]
    except Exception:
        pass
    return None


def gradient_direction(h, active, step_size):
    n = len(h)
    g = np.zeros(n)
    for m in active:
        g += grad_one_row(h, m)
    g -= g.mean()
    gnorm = np.linalg.norm(g)
    if gnorm < 1e-14:
        return None
    return -step_size * g / gnorm


def optimize_lp(h_init, max_iters=2000, deadline=None, delta0=0.08, tol_scale=0.02):
    n = len(h_init)
    h = h_init.copy()
    best_c = exact_score(h)
    delta = delta0
    stall = 0

    for it in range(max_iters):
        if deadline is not None and time.perf_counter() > deadline:
            break
        profile = compute_profile(h)
        c = float(profile.max())

        thresh = c - max(0.0005, tol_scale * c)
        active = np.where(profile >= thresh)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(profile))])
        if len(active) > 40:
            top = np.argsort(profile[active])[-40:]
            active = active[top]

        d = lp_direction(h, active, profile, delta)
        if d is None:
            d = gradient_direction(h, active, delta)
        if d is None:
            stall += 1
            if stall > 15:
                delta *= 0.5
                if delta < 5e-5:
                    break
            continue

        h_new = repair_feasible(h + d)
        c_new = exact_score(h_new)

        if c_new < c - 1e-13:
            h = h_new
            if c_new < best_c:
                best_c = c_new
            stall = 0
            if c_new < c - 0.3 * delta * 0.005:
                delta = min(delta * 1.15, 0.35)
        else:
            improved = False
            for frac in [0.5, 0.25, 0.1, 0.05]:
                h_try = repair_feasible(h + frac * d)
                c_try = exact_score(h_try)
                if c_try < c - 1e-13:
                    h = h_try
                    if c_try < best_c:
                        best_c = c_try
                    improved = True
                    stall = 0
                    break
            if not improved:
                stall += 1
                delta *= 0.6
                if delta < 5e-5:
                    break

    return h, best_c


def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    edges_old = np.linspace(0.0, 2.0, n + 1)
    edges_new = np.linspace(0.0, 2.0, new_n + 1)
    h_new = np.zeros(new_n)
    for k in range(new_n):
        a, b = edges_new[k], edges_new[k + 1]
        total = 0.0
        for j in range(n):
            lo = max(a, edges_old[j])
            hi = min(b, edges_old[j + 1])
            if hi > lo:
                total += h[j] * (hi - lo)
        h_new[k] = total / (b - a)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)


def golden_seed(n, shift=0.0):
    phi = 0.618033988749895
    idx = np.arange(n, dtype=np.float64)
    vals = (idx * phi + shift) % 1.0
    half = n // 2
    top = np.argsort(vals)[-half:]
    h = np.zeros(n)
    h[top] = 1.0
    return repair_feasible(h)


def blended_seed(n, shift=0.0, blend=0.2):
    h_bin = golden_seed(n, shift)
    h = (1.0 - blend) * h_bin + blend * 0.5
    return repair_feasible(h)


def nesterov_polish(h, max_iters=300, deadline=None, tol=0.004):
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    beta = 0.85
    y = h.copy()
    v = np.zeros(n)
    step = 0.03

    for it in range(max_iters):
        if deadline is not None and time.perf_counter() > deadline:
            break
        profile = compute_profile(y)
        c = float(profile.max())
        thresh = c - tol
        active = np.where(profile >= thresh)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(profile))])
        if len(active) > 25:
            top = np.argsort(profile[active])[-25:]
            active = active[top]
        g = np.zeros(n)
        for m in active:
            g += grad_one_row(y, m)
        g -= g.mean()
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-14:
            break
        v = beta * v - step * g / gnorm
        y = repair_feasible(h + v)
        h = y
        step *= 0.999
        c_h = exact_score(h)
        if c_h < best_c:
            best_c = c_h
    return h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s - (time.perf_counter() - t_start)

    def deadline():
        return t_start + budget_s - 5.0

    try:
        parent = np.array(initial_h_values, dtype=np.float64)
    except Exception:
        parent = None

    if parent is None or len(parent) < 4:
        parent = blended_seed(70, 0.0, 0.15)
    parent = np.clip(parent, 0.0, 1.0)
    parent = repair_feasible(parent)

    best_h = parent.copy()
    best_c = exact_score(parent)

    # Phase 1: LP optimization on parent
    dl = deadline()
    h1, c1 = optimize_lp(best_h, max_iters=1500, deadline=dl, delta0=0.08)
    if c1 < best_c:
        best_h, best_c = h1, c1

    # Phase 2: Nesterov polish
    if time_left() > 60:
        dl = deadline()
        h2, c2 = nesterov_polish(best_h, max_iters=200, deadline=dl, tol=0.003)
        if c2 < best_c:
            best_h, best_c = h2, c2

    # Phase 3: Tighter LP
    if time_left() > 50:
        dl = deadline()
        h3, c3 = optimize_lp(best_h, max_iters=2000, deadline=dl, delta0=0.04, tol_scale=0.01)
        if c3 < best_c:
            best_h, best_c = h3, c3

    # Phase 4: Dimension exploration
    n_cur = len(best_h)
    dims_to_try = []
    for dn in [-4, -2, 2, 4, 6, 8, -6, 10, -8, 12]:
        d = n_cur + dn
        if 40 <= d <= 140:
            dims_to_try.append(d)

    for new_n in dims_to_try[:8]:
        if time_left() < 40:
            break
        h_interp = interpolate_to_n(best_h, new_n)
        dl = deadline()
        h_try, c_try = optimize_lp(h_interp, max_iters=800, deadline=dl, delta0=0.06)
        if c_try < best_c:
            best_h, best_c = h_try, c_try
            # Also try to go back to n_cur
            if len(best_h) != n_cur:
                h_back = interpolate_to_n(best_h, n_cur)
                dl2 = deadline()
                h_back, c_back = optimize_lp(h_back, max_iters=500, deadline=dl2, delta0=0.05)
                if c_back < best_c:
                    best_h, best_c = h_back, c_back

    # Phase 5: Multi-start perturbation
    if time_left() > 120:
        n_exp = len(best_h)
        for trial in range(8):
            if time_left() < 30:
                break
            noise_scale = 0.03 + 0.02 * trial
            h_pert = repair_feasible(best_h + noise_scale * rng.randn(n_exp))
            # Blend with golden seed for diversity
            h_gold = golden_seed(n_exp, shift=trial * 0.137)
            mix = 0.3 * h_pert + 0.7 * h_gold
            h_pert = repair_feasible(mix)
            dl = deadline()
            h_try, c_try = optimize_lp(h_pert, max_iters=600, deadline=dl, delta0=0.08)
            if c_try < best_c:
                best_h, best_c = h_try, c_try

    # Phase 6: Very tight final LP
    if time_left() > 30:
        dl = deadline()
        h6, c6 = optimize_lp(best_h, max_iters=3000, deadline=dl, delta0=0.02, tol_scale=0.005)
        if c6 < best_c:
            best_h, best_c = h6, c6

    # Phase 7: Final nesterov
    if time_left() > 20:
        dl = deadline()
        h7, c7 = nesterov_polish(best_h, max_iters=200, deadline=dl, tol=0.001)
        if c7 < best_c:
            best_h, best_c = h7, c7

    # Final repair
    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)

    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            final_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 2:
            c_eval = float(eval_result[0])
            if c_eval < final_c + 1e-9:
                final_c = c_eval
    except Exception:
        pass

    return best_h, final_c, len(best_h)
