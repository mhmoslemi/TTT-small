def grad_corr_k(h, k, n):
    """Gradient of corr at lag k (index in full correlation) with respect to h."""
    off = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -off)
    i_hi = min(n, n - off)
    if i_lo >= i_hi:
        return grad
    idx = np.arange(i_lo, i_hi)
    grad[idx + off] += (1.0 - h[idx])
    grad[idx] += -h[idx + off]
    return grad

def pair_optimize(h, p, q, cur_c5_raw, n_grid=25):

def subgradient_step(h, target_sum, n_active=30, step_size=0.02):
    """Take a subgradient step on the max correlation."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = corr.max()
    # Find active lags (within margin of max)
    margin = max(0.05 * max_c, 0.01)
    active = np.where(corr >= max_c - margin)[0]
    if len(active) == 0:
        active = np.array([np.argmax(corr)])
    # Cap active set
    if len(active) > n_active:
        active = active[np.argsort(corr[active])[-n_active:]]

    grad = np.zeros(n)
    for k in active:
