def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        order = np.argsort(-(1.0 - h))
        remaining = diff
        for i in range(n):
            idx = order[i]
            cap = 1.0 - h[idx]
            if cap <= 1e-15:
                continue
            add = min(remaining, cap)
            h[idx] += add
            remaining -= add
            if remaining <= 1e-14:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            if h[idx] <= 1e-15:
                continue
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-14:
                break
    # Final fix
    diff2 = target - h.sum()
    if abs(diff2) > 1e-14:
        mask = (h > 1e-10) & (h < 1 - 1e-10)
        if mask.any():
            h[mask] += diff2 / mask.sum()
            h = np.clip(h, 0.0, 1.0)

            idx = int(np.argmax(1.0 - h)) if diff2 > 0 else int(np.argmax(h))
            h[idx] = float(np.clip(h[idx] + diff2, 0.0, 1.0))
    return h

def exact_score(h):
    n = len(h)

def compute_profile(h):

def grad_row(h, m, n):
    dx = 2.0 / n
    d = m - (n - 1)
    if d == 0:
        return dx * (1.0 - 2.0 * h)
    elif d > 0:
        g = np.zeros(n)
        g[:-d] -= dx * h[d:]
        g[d:] += dx * (1.0 - h[:-d])
        return g
    else:
        p = -d
        g = np.zeros(n)
        g[p:] -= dx * h[:-p]
        g[:-p] += dx * (1.0 - h[p:])
        return g

def interp_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    x_old = np.linspace(0, 2, n + 1)
    x_new = np.linspace(0, 2, new_n + 1)
    # For each new cell, compute the average of old values weighted by overlap
    h_new = np.zeros(new_n)
    for k in range(new_n):
        a, b = x_new[k], x_new[k+1]
        total = 0.0
        for j in range(n):
            lo = max(a, x_old[j])
            hi = min(b, x_old[j+1])
            if hi > lo:
                total += h[j] * (hi - lo)
        h_new[k] = total / (b - a)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

def slsqp_epigraph(h_start, maxiter=200):
    n = len(h_start)
    dx = 2.0 / n
    M = 2 * n - 1
    c0 = exact_score(h_start)
    t0 = c0 + 1e-4
    
    def obj(x):
        return x[-1]
    
    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    
    def eq_fun(x):
        return x[:-1].sum() - n / 2.0
    
    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g
    
    def ineq_fun(x):
        h = x[:-1]
        t = x[-1]
        profile = np.correlate(h, 1 - h, mode="full") * dx
        return t - profile
    
    def ineq_jac(x):
        h = x[:-1]
        J = np.zeros((M, n + 1))
        for m in range(M):
            J[m, :n] = -grad_row(h, m, n)
        J[:, -1] = 1.0
        return J
    
    x0 = np.concatenate([h_start, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 2.0)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-13})
        if res.x is not None and np.isfinite(res.x).all():
            h_new = repair_feasible(res.x[:-1])
            c_new = exact_score(h_new)
            if c_new <= c0 + 1e-9:
                return h_new
    except Exception:
        pass
    return None

def golden_binary_seed(n, shift=0.0):
    """Create a binary seed using golden ratio spacing."""
    half = n // 2
    h = np.zeros(n)
    phi = (1 + np.sqrt(5)) / 2
    idx = np.arange(n)
    vals = ((idx + shift) * (phi % 1)) % 1.0
    top = np.argsort(vals)[-half:]
    h[top] = 1.0
    return repair_feasible(h)

def smooth_binary(h, window=2):
    """Apply box filter smoothing."""
    n = len(h)
    kernel = np.ones(window) / window
    h_smooth = np.convolve(h, kernel, mode='same')
    return repair_feasible(np.clip(h_smooth, 0, 1))

def crossover(h_a, h_b, t=0.5):
    """Convex combination of two feasible solutions of same length."""
    if len(h_a) != len(h_b):
        return None
    h_mix = (1 - t) * h_a + t * h_b
    return repair_feasible(h_mix)

def lp_trust_region(h, delta=0.03, max_active=40):
    """
    Linearized trust-region LP: minimize tau s.t. P_m + grad_m · d <= tau
    for active m, sum(d) = 0, |d_i| <= delta, h_i + d_i in [0,1].
    """
    n = len(h)
    profile = compute_profile(h)
    c = float(profile.max())
    thresh = c - 0.005
    active = np.where(profile >= thresh)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(profile))])
    if len(active) > max_active:
        top = np.argsort(profile[active])[-max_active:]
        active = active[top]
    
    M = len(active)
    # Build constraint matrix: [G_row | -1] @ [d; tau] <= -P_row
    A = np.zeros((M, n + 1))
    b = np.zeros(M)
    for i, m in enumerate(active):
        A[i, :n] = grad_row(h, m, n)
        A[i, n] = -1.0
        b[i] = -profile[m]
    
    # Bound on d
    lo_d = np.maximum(-delta, -h + 1e-10)
    hi_d = np.minimum(delta, 1.0 - h - 1e-10)
    
    # Objective: minimize tau
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    bounds_lp = list(zip(lo_d, hi_d)) + [(0, c + 1.0)]
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds_lp,
                      method='highs-ds',
                      options={'presolve': True, 'dual_feasibility_tol': 1e-8})
        if res.success and res.x is not None:
            d = res.x[:n]
            # Project d to ensure sum(d) ~ 0 (mass constraint)
            d -= d.mean()
            h_new = repair_feasible(h + d)
            c_new = exact_score(h_new)
            if c_new < c - 1e-12:
                return h_new, c_new
    except Exception:
        pass
    return None, c

        A[i, :n] = grad_row(h, m, n)
        A[i, n] = -1.0
        b_ub[i] = -profile[m]

    lo_d = np.maximum(-delta, -h + 1e-10)
    hi_d = np.minimum(delta, 1.0 - h - 1e-10)
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    bounds_lp = list(zip(lo_d, hi_d)) + [(0.0, c + 1.0)]

    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds_lp,
                      method='highs-ds',
                      options={'presolve': True})
        if res.success and res.x is not None:
            d = res.x[:n]
            d -= d.mean()
            h_new = repair_feasible(h + d)
            c_new = exact_score(h_new)
            if c_new < c - 1e-12:
                return h_new, c_new
    except Exception:
        pass
    return None, c

def golden_binary_seed(n, shift=0.0):
    half = n // 2
    h = np.zeros(n)
    phi = 0.618033988749895
    idx = np.arange(n)
    vals = ((idx + shift) * phi) % 1.0
    top = np.argsort(vals)[-half:]
    h[top] = 1.0
    return repair_feasible(h)

def crossover(h_a, h_b, t=0.5):
    if len(h_a) != len(h_b):
        return None
    h_mix = (1 - t) * h_a + t * h_b
    return repair_feasible(h_mix)

def coordinate_polish(h, max_iters=60):
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    for it in range(max_iters):
        improved = False
        profile = compute_profile(h)
        c = float(profile.max())
        thresh = c - 0.003
        active = np.where(profile >= thresh)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(profile))])
        G = np.array([grad_row(h, m, n) for m in active])
        idx_desc = np.argsort(-h)
        idx_asc = np.argsort(h)
        tried = 0
        for attempt in range(40):
            i = int(idx_desc[attempt % n])
            j = int(idx_asc[attempt % n])
            if i == j:
                continue
            delta = 0.015 * (h[i] - h[j])
            if abs(delta) < 1e-7:
                continue
            for sign in [1.0, -1.0]:
                h_try = h.copy()
                h_try[i] += sign * delta
                h_try[j] -= sign * delta
                if h_try.min() < -1e-10 or h_try.max() > 1 + 1e-10:
                    continue
                h_try = repair_feasible(h_try)
                s = exact_score(h_try)
                if s < best_c - 1e-14:
                    h = h_try
                    best_c = s
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s - (time.perf_counter() - t_start)

    try:
        parent = np.array(initial_h_values, dtype=np.float64)
    except Exception:
        parent = None
    if parent is None or len(parent) < 4:
        parent = np.full(70, 0.5)
    parent = np.clip(parent, 0.0, 1.0)
    parent = repair_feasible(parent)
    n_orig = len(parent)

    best_h = parent.copy()
    best_c = exact_score(parent)

    # Stage 1: Aggressive active descent on parent
    if time_left() > 100:
        h_ad, c_ad = active_descent_v2(best_h, max_iters=500, tol_init=0.005, tol_min=0.0002)
        if c_ad < best_c:
            best_h, best_c = h_ad, c_ad

    # Stage 2: LP trust region (large step)
    if time_left() > 60:
        h_lp, c_lp = lp_trust_region(best_h, delta=0.05)
        if h_lp is not None and c_lp < best_c:
            best_h, best_c = h_lp, c_lp
            if time_left() > 40:
                h_ad, c_ad = active_descent_v2(best_h, max_iters=300, tol_init=0.003, tol_min=0.0002)
                if c_ad < best_c:
                    best_h, best_c = h_ad, c_ad

    # Stage 3: Dimension sweep (prefer smaller)
    n_best = len(best_h)
    dims_to_try = []
    for dn in [-8, -6, -4, -2, 2, 4, 6, 8, 10, 12]:
        d = n_best + dn
        if 30 <= d <= 130:
            dims_to_try.append(d)
    dims_to_try.sort(key=lambda x: (x if x < n_best else 200 + x, abs(x - n_best)))

    secondary = []
    for new_n in dims_to_try[:8]:
        if time_left() < 35:
            break
        h_interp = interp_to_n(best_h, new_n)
        h_ad, c_ad = active_descent_v2(h_interp, max_iters=300, tol_init=0.004, tol_min=0.0003)
        if c_ad < best_c:
            best_h, best_c = h_ad, c_ad
            n_best = new_n
        else:
            secondary.append((c_ad, h_ad.copy(), new_n))

    # Stage 4: SLSQP
    for _ in range(2):
        if time_left() < 30:
            break
        h_sol = slsqp_epigraph(best_h, maxiter=180)
        if h_sol is not None:
            c_sol = exact_score(h_sol)
            if c_sol < best_c - 1e-12:
                best_h, best_c = h_sol, c_sol
            if time_left() > 20:
                h_ad, c_ad = active_descent_v2(best_h, max_iters=150, tol_init=0.002, tol_min=0.0002)
                if c_ad < best_c:
                    best_h, best_c = h_ad, c_ad

    # Stage 5: Population exploration
    if time_left() > 90:
        n_exp = len(best_h)
        pool = [(best_c, best_h.copy(), n_exp)]

        for pt in range(5):
            if time_left() < 25:
                break
            if pt == 0:
                h_seed = golden_binary_seed(n_exp, 0.3)
            elif pt == 1:
                h_seed = golden_binary_seed(n_exp, 2.7)
            elif pt == 2:
                h_raw = 0.5 + 0.35 * np.sin(np.linspace(0, 5 * np.pi, n_exp) + 0.4)
                h_seed = repair_feasible(np.clip(h_raw, 0, 1))
            elif pt == 3:
                h_raw = 0.5 + 0.3 * np.sin(np.linspace(0, 9 * np.pi, n_exp))
                h_seed = repair_feasible(np.clip(h_raw, 0, 1))
            else:
                h_seed = repair_feasible(np.clip(0.5 + rng.randn(n_exp) * 0.25, 0, 1))
            h_ad, c_ad = active_descent_v2(h_seed, max_iters=200, tol_init=0.005, tol_min=0.001)
            pool.append((c_ad, h_ad.copy(), n_exp))

        pool.sort(key=lambda x: x[0])

        # Crossover
        for i in range(min(4, len(pool) - 1)):
            if time_left() < 20:
                break
            for j in range(i + 1, min(i + 3, len(pool))):
                if time_left() < 15:
                    break
                if pool[i][2] == pool[j][2]:
                    for t_mix in [0.3, 0.5, 0.7]:
                        h_cross = crossover(pool[i][1], pool[j][1], t_mix)
                        if h_cross is None:
                            continue
                        h_ad, c_ad = active_descent_v2(h_cross, max_iters=120, tol_init=0.004, tol_min=0.001)
                        if c_ad < best_c:
                            best_h, best_c = h_ad.copy(), c_ad
                        pool.append((c_ad, h_ad.copy(), len(h_ad)))

        # Also try crossover with secondary
        for sc in secondary[:2]:
            if time_left() < 20:
                break
            if sc[2] == n_exp:
                h_cross = crossover(best_h, sc[1], 0.3)
            else:
                h_cross = crossover(best_h, interp_to_n(sc[1], n_exp), 0.3)
            if h_cross is not None:
                h_ad, c_ad = active_descent_v2(h_cross, max_iters=150, tol_init=0.003, tol_min=0.001)
                if c_ad < best_c:
                    best_h, best_c = h_ad, c_ad

    # Stage 6: LP again with different delta
    if time_left() > 35:
        for delta in [0.02, 0.04, 0.07]:
            if time_left() < 12:
                break
            h_lp, c_lp = lp_trust_region(best_h, delta=delta)
            if h_lp is not None and c_lp < best_c:
                best_h, best_c = h_lp, c_lp
                if time_left() > 15:
                    h_ad, c_ad = active_descent_v2(best_h, max_iters=200, tol_init=0.002, tol_min=0.0003)
                    if c_ad < best_c:
                        best_h, best_c = h_ad, c_ad

    # Stage 7: Final tight active descent
    if time_left() > 25:
        h_ad, c_ad = active_descent_v2(best_h, max_iters=400, tol_init=0.003, tol_min=0.0001)
        if c_ad < best_c:
            best_h, best_c = h_ad, c_ad

    # Stage 8: Coordinate polish
    if time_left() > 20:
        h_cp, c_cp = coordinate_polish(best_h, max_iters=50)
        if c_cp < best_c:
            best_h, best_c = h_cp, c_ad

    # Stage 9: Final SLSQP
    if time_left() > 15:
        h_sol = slsqp_epigraph(best_h, maxiter=200)
        if h_sol is not None:
            c_sol = exact_score(h_sol)
            if c_sol < best_c:
                best_h, best_c = h_sol, c_sol

    # Stage 10: Ultra-fine polish
    if time_left() > 8:
        h_ad, c_ad = active_descent_v2(best_h, max_iters=200, tol_init=0.002, tol_min=0.0001)
        if c_ad < best_c:
            best_h, best_c = h_ad, c_ad

    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)

    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            final_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 2:
            final_c = float(eval_result[0])
    except Exception:
        pass

    return best_h, final_c, len(best_h)
