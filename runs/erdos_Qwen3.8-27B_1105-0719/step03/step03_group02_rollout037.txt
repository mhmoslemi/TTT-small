def threshold_penalty_gradient(h, dx, target, n_active=None):
    """Gradient of sum of squared excesses above target."""
    n = len(h)
    corrs = compute_corrs(h)
    c5_vals = corrs * dx
    excess = np.maximum(0, c5_vals - target)
    active = excess > 0
    if not np.any(active):
        return np.zeros(n), 0.0
    if n_active and active.sum() > n_active:
        # Only use the worst
        worst_idx = np.where(active)[0]
        worst_idx = worst_idx[np.argsort(c5_vals[worst_idx])[-n_active:]]
        active = np.zeros(len(c5_vals), dtype=bool)
        active[worst_idx] = True
    
    grad = np.zeros(n)
    penalty_val = 0.0
    for k in np.where(active)[0]:
        w = 2.0 * (c5_vals[k] - target) * dx
        grad += w * grad_corr_k(h, k, n)
        penalty_val += (c5_vals[k] - target) ** 2
    
    return grad, penalty_val

def binary_local_search(b, n, rng, max_evals=100000):
    """Local search on binary vector. Returns (best_b, best_max_corr)."""
    b = b.astype(np.float64).copy()
    corrs = compute_corrs(b)
    best_max = corrs.max()
    ones = set(np.where(b == 1)[0].tolist())
    zeros = set(np.where(b == 0)[0].tolist())
    
    for _ in range(max_evals // 50):  # 50 random swaps per "iteration"
        improved = False
        for _ in range(50):
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = rng.choice(list(zeros))
            j = rng.choice(list(ones))
            
            # Swap: i becomes 1, j becomes 0
            b[i] = 1.0
            b[j] = 0.0
            new_corrs = compute_corrs(b)
            new_max = new_corrs.max()
            
            if new_max < best_max:
                best_max = new_max
                ones.add(i)
                zeros.discard(i)
                zeros.add(j)
                ones.discard(j)
                improved = True
            else:
                # Revert
                b[i] = 0.0
                b[j] = 1.0
                ones.discard(i)
                zeros.add(i)
                zeros.discard(j)
                ones.add(j)
        
        if not improved:
            break
    
    return b, best_max

def make_balanced_binary(n, rng):
    """Create random balanced binary vector."""
    idx = rng.choice(n, size=n//2, replace=False)

def make_periodic_binary(n, rng, period, fill):
    """Create periodic binary vector with given period and fill ratio."""
    b = np.zeros(n)
    for i in range(n):
        if (i % period) < fill:
            b[i] = 1.0
    # Adjust to be exactly balanced
    ones_count = b.sum()
    target = n // 2
    if ones_count > target:
        excess_idx = np.where(b == 1)[0]
        remove = rng.choice(excess_count, size=int(ones_count - target), replace=False)
        b[remove] = 0.0
    elif ones_count < target:
        deficit_idx = np.where(b == 0)[0]
        add = rng.choice(deficit_count, size=int(target - ones_count), replace=False)
        b[add] = 1.0
    return b

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.special import logsumexp
import time

def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def c5_of_h(h, dx):
    return compute_corrs(h).max() * dx

def grad_corr_k(h, k, n):
    s = k - (n - 1)
    grad = np.zeros(n)
    r1_lo = max(0, -s)
    r1_hi = min(n - 1, n - 1 - s)
    r2_lo = max(0, s)
    r2_hi = min(n - 1, n - 1 + s)
    if r1_lo <= r1_hi:
        idx = np.arange(r1_lo, r1_hi + 1)
        grad[idx] += (1.0 - h[idx + s])
    if r2_lo <= r2_hi:
        idx = np.arange(r2_lo, r2_hi + 1)
        grad[idx] += (-h[idx - s])
    return grad

def compute_full_jacobian(h):
    """Compute J[k,j] = d(corr[k])/d(h[j]) for all k, j."""
    n = len(h)
    J = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        J[k] = grad_corr_k(h, k, n)
    return J

def smooth_max_gradient(h, dx, beta):
    n = len(h)
    corrs = compute_corrs(h)
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones_like(corrs) / len(corrs)
    else:
        w /= w_sum
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] > 1e-13:
            grad += w[k] * grad_corr_k(h, k, n)
    return grad * dx

def slsqp_optimize(h0, dx, target_sum, beta, maxiter=1500):
    n = len(h0)
    def obj(x):
        corrs = compute_corrs(x)
        return float(logsumexp(beta * corrs) / beta) * dx
    def grad_obj(x):
        return smooth_max_gradient(x, dx, beta)
    def c_fun(x):
        return x.sum() - target_sum
    def c_jac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': c_fun, 'jac': c_jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0, jac=grad_obj, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        h_out = project_feasible(res.x, target_sum)
    except Exception:
        h_out = project_feasible(h0, target_sum)
    return h_out, c5_of_h(h_out, dx)

def pair_transfer_greedy(h0, dx, target_sum, max_sweeps=50, min_improve=1e-14):
    """Greedy pair transfer: for each i, find best j, do exact optimal transfer."""
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()

        c5_val = corrs.max() * dx
        J = compute_full_jacobian(h)
        sweep_improved = False

        for i in range(n):
            if time.time() - getattr(pair_transfer_greedy, '_t0', time.time()) > 400:
                return best_h, best_c5
            best_j = -1
            best_eps = 0.0
            best_new_max = float('inf')
            for j in range(n):
                if i == j:
                    continue
                a = J[:, i] - J[:, j]
                eps_lo = max(-h[i], h[j] - 1.0)
                eps_hi = min(1.0 - h[i], h[j])

                    continue
                a_t, b_t = eps_lo, eps_hi
                c_t = b_t - gr * (b_t - a_t)
                d_t = a_t + gr * (b_t - a_t)
                fc = (corrs + c_t * a).max()
                fd = (corrs + d_t * a).max()
                for _ in range(40):
                    if b_t - a_t < 1e-14:
                        break
                    if fc < fd:
                        b_t = d_t
                        d_t = c_t
                        fd = fc
                        c_t = b_t - gr * (b_t - a_t)
                        fc = (corrs + c_t * a).max()
                    else:
                        a_t = c_t
                        c_t = d_t
                        fc = fd
                        d_t = a_t + gr * (b_t - a_t)
                        fd = (corrs + d_t * a).max()
                eps_opt = (a_t + b_t) / 2.0

                if new_max < best_new_max:

def binary_local_search(b, n, rng, max_rounds=2000):
    b = b.astype(np.float64).copy()
    corrs = compute_corrs(b)
    best_max = corrs.max()
    ones_list = np.where(b == 1)[0].tolist()
    zeros_list = np.where(b == 0)[0].tolist()
    for _ in range(max_rounds):
        improved = False
        for _ in range(30):
            i = rng.choice(zeros_list)
            j = rng.choice(ones_list)
            b[i] = 1.0
            b[j] = 0.0
            new_corrs = compute_corrs(b)
            new_max = new_corrs.max()
            if new_max < best_max:
                best_max = new_max
                ones_list.append(i)
                zeros_list.remove(i)
                zeros_list.append(j)
                ones_list.remove(j)
                corrs = new_corrs
                improved = True
            else:
                b[i] = 0.0
                b[j] = 1.0
                ones_list.remove(i)
                zeros_list.append(i)
                zeros_list.remove(j)
                ones_list.append(j)
        if not improved:
            break
    return b, best_max

def make_balanced_binary(n, rng):
    idx = rng.choice(n, size=n // 2, replace=False)

def make_periodic_binary(n, rng, period, fill):

        n_add = target - ones_count
        add = rng.choice(deficit_idx, size=min(n_add, len(deficit_idx)), replace=False)

        ts_b = n_b / 2.0
        for beta in [2000, 5000, 10000]:
            if not check_time(15):
                break
            h_opt, c5_val = slsqp_optimize(best_h, dx_b, ts_b, beta, 1500)
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h_opt

    # Phase 8: Final pair transfer
    if check_time(20):
        n_b = len(best_h)
        dx_b = 2.0 / n_b
        ts_b = n_b / 2.0
        pair_transfer_greedy._t0 = time.time()
        h_pt2, c5_pt2 = pair_transfer_greedy(best_h, dx_b, ts_b, max_sweeps=15)
        delattr(pair_transfer_greedy, '_t0')
        if c5_pt2 < best_c5:
            best_c5 = c5_pt2
            best_h = h_pt2

    # Phase 9: Final fine LP
    if check_time(15):
        n_b = len(best_h)
        dx_b = 2.0 / n_b
        ts_b = n_b / 2.0
        h_fine, c5_fine = lp_polish_full(best_h, dx_b, ts_b, max_iters=200, init_step=0.03)
        if c5_fine < best_c5:
            best_c5 = c5_fine
            best_h = h_fine

    # Final validation
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final / 2.0)
    best_c5 = c5_of_h(best_h, 2.0 / n_final)
    return best_h, best_c5, n_final
