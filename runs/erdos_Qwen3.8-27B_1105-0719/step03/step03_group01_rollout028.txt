def project_feasible(x, target_sum):
    """Project x onto [0,1]^n with sum = target_sum using bisection on shift."""
    n = len(x)
    x = np.clip(x, -3.0, 3.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        cnt = mask.sum()
        if cnt > 0:
            h[mask] += diff / cnt

def compute_corrs(h):

        if cur_sum > target_a:
            i = rng.randint(n)
            if a[i] > 0:
                a[i] -= 1
                cur_sum -= 1
        else:
            i = rng.randint(n)
            if a[i] < M:
                a[i] += 1
                cur_sum += 1
    h = a.astype(float) / M
    c = compute_corrs(h).copy()
    cur_obj = float(np.max(c))
    best_obj = cur_obj
    best_a = a.copy()
    T = T_start
    it = 0
    while it < max_iters:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j or a[i] == 0 or a[j] == M:
            it += 1
            continue
        a[i] -= 1
        a[j] += 1
        h_new = a.astype(float) / M
        c_new = compute_corrs(h_new)
        new_obj = float(np.max(c_new))
        delta = new_obj - cur_obj
        if delta < -1e-12 or (T > T_end and rng.random() < np.exp(-delta / T)):
            cur_obj = new_obj
            c = c_new
            h = h_new
            if new_obj < best_obj:
                best_obj = new_obj
                best_a = a.copy()
        else:
            a[i] += 1
            a[j] -= 1
        T = max(T_end, T * cooling)
        it += 1
    best_h = best_a.astype(float) / M
    return project_feasible(best_h, n / 2.0), best_obj * dx

def lp_polish(h, dx, target_sum, t0, max_time, n_active=50, alpha=0.2):
    n = len(h)
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    no_improve = 0
    for it in range(300):
        if time.time() - t0 > max_time:
            break
        if no_improve > 50:
            break
        corrs = compute_corrs(best_h)
        active_idx = np.argsort(corrs)[-n_active:]
        G = np.zeros((len(active_idx), n))
        for row, p in enumerate(active_idx):
            o = p - (n - 1)
            for i in range(n):
                g = 0.0
                if 0 <= i + o < n:
                    g += (1.0 - best_h[i + o])
                if 0 <= i - o < n:
                    g += -best_h[i - o]
                G[row, i] = g
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        A_ub = np.empty((len(active_idx), n + 1))
        A_ub[:, :n] = G
        A_ub[:, n] = -1.0
        b_ub = -corrs[active_idx]
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)
        bounds = [(-best_h[i] * alpha, (1.0 - best_h[i]) * alpha) for i in range(n)]
        bounds.append((-np.inf, np.inf))
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if res.status != 0:
                no_improve += 1
                continue
            delta = res.x[:n]
        except Exception:
            no_improve += 1
            continue
        if np.abs(delta).max() < 1e-14:
            no_improve += 1
            continue
        improved = False
        for s in [1.0, 0.7, 0.5, 0.3, 0.15, 0.05]:
            h_new = project_feasible(best_h + s * delta, target_sum)
            c5_new = c5_of_h(h_new, dx)
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                best_h = h_new
                improved = True
                break
        if improved:
            no_improve = 0
        else:
            no_improve += 1
    return best_h, best_c5

def greedy_hill_climb(h, dx, target_sum, n, max_iters, seed):
    rng = np.random.RandomState(seed)
    h_cur = h.copy()
    c = compute_corrs(h_cur)
    cur_obj = float(np.max(c))
    best_obj = cur_obj
    best_h = h_cur.copy()
    for it in range(max_iters):
        a = rng.randint(n)
        b = rng.randint(n)
        if a == b:
            continue
        mf = min(h_cur[a], 1.0 - h_cur[b])
        mb = min(h_cur[b], 1.0 - h_cur[a])
        if rng.random() < 0.5:
            if mf < 1e-9:
                continue
            d = mf * rng.uniform(0.3, 1.0)
        else:
            if mb < 1e-9:
                continue
            d = -mb * rng.uniform(0.3, 1.0)
        c_old = c.copy()
        ha, hb = h_cur[a], h_cur[b]
        update_corrs_transfer(c, h_cur, a, b, d, n)
        new_obj = float(np.max(c))
        if new_obj < cur_obj - 1e-13:
            cur_obj = new_obj
            if new_obj < best_obj:
                best_obj = new_obj
                best_h = h_cur.copy()
        else:
            h_cur[a] = ha
            h_cur[b] = hb
            c[:] = c_old
    return best_h, best_obj * dx

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n = len(h0)
    dx = 2.0 / n
    target_sum = n / 2.0
    h0 = project_feasible(h0, target_sum)
    best_h = h0.copy()
    best_c5 = c5_of_h(h0, dx)

    # Verify incremental update correctness
    h_test = best_h.copy()
    c_direct = compute_corrs(h_test).copy()
    a_t, b_t = 0, n - 1
    d_t = 0.01
    h_inc = h_test.copy()
    c_inc = c_direct.copy()
    update_corrs_transfer(c_inc, h_inc, a_t, b_t, d_t, n)
    c_ref = compute_corrs(h_inc)
    assert np.max(np.abs(c_inc - c_ref)) < 1e-8, "Incremental update mismatch"

    # Phase 1: LP polish (15% budget)
    t_p1 = budget_s * 0.12
    for alpha, nact in [(0.25, 40), (0.15, 60), (0.1, 80)]:
        if time.time() - t0 > t_p1:
            break
        h_opt, c5_opt = lp_polish(best_h, dx, target_sum, t0, time.time() - t0 + (t_p1 - (time.time() - t0)),
                                  n_active=nact, alpha=alpha)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 2: Continuous SA (45% budget)
    t_p2 = budget_s * 0.55
    sa_configs = [
        (800000, 4.0, 0.002, 0.99990),
        (800000, 2.0, 0.001, 0.99992),
        (1000000, 1.0, 0.0005, 0.99994),
        (800000, 0.5, 0.0003, 0.99995),
        (1000000, 0.2, 0.0002, 0.99996),
    ]
    for ci, (n_it, Ts, Te, cool) in enumerate(sa_configs):
        if time.time() - t0 > t_p2:
            break
        sa_seed = seed + 5000 + ci * 313
        h_sa, c5_sa = sa_continuous(n, n_it, sa_seed, best_h, Ts, Te, cool, dx, target_sum)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa
        # Quick polish after each SA
        if time.time() - t0 < t_p2:
            h_opt, c5_opt = lp_polish(h_sa, dx, target_sum, t0, time.time() + 15, n_active=60, alpha=0.15)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt

    # Phase 3: Quantized SA (15% budget)
    t_p3 = budget_s * 0.70
    for M, n_it, Ts, Te, cool in [(8, 500000, 3.0, 0.01, 0.9998), (16, 400000, 2.0, 0.005, 0.99985)]:
        if time.time() - t0 > t_p3:
            break
        h_sa, c5_sa = sa_quantized(n, M, n_it, seed + M * 101, best_h, Ts, Te, cool, dx)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa
        if time.time() - t0 < t_p3:
            h_opt, c5_opt = lp_polish(h_sa, dx, target_sum, t0, time.time() + 10, n_active=50, alpha=0.2)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt

    # Phase 4: Hill climbing (10% budget)
    t_p4 = budget_s * 0.82
    for hc_seed in [seed + 7001, seed + 7002, seed + 7003]:
        if time.time() - t0 > t_p4:
            break
        h_hc, c5_hc = greedy_hill_climb(best_h, dx, target_sum, n, 200000, hc_seed)
        if c5_hc < best_c5:
            best_c5 = c5_hc
            best_h = h_hc
        h_opt, c5_opt = lp_polish(best_h, dx, target_sum, t0, time.time() + 10, n_active=80, alpha=0.1)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 5: Try different n (10% budget)
    t_p5 = budget_s * 0.92
    for n_try in [120, 132, 112, 144]:
        if time.time() - t0 > t_p5:
            break
        x_orig = np.linspace(0.0, 2.0, len(best_h))
        x_new = np.linspace(0.0, 2.0, n_try)
        h_interp = np.interp(x_new, x_orig, best_h)
        h_interp = project_feasible(h_interp, n_try / 2.0)
        dx_try = 2.0 / n_try
        c5_try = c5_of_h(h_interp, dx_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_interp
        elif time.time() - t0 < t_p5 - 20:
            h_sa, c5_sa = sa_continuous(n_try, 200000, seed + n_try, h_interp, 1.5, 0.001, 0.9999, dx_try, n_try / 2.0)
            if c5_sa < best_c5:
                best_c5 = c5_sa
                best_h = h_sa
                n = n_try
                dx = dx_try
                target_sum = n / 2.0

    # Phase 6: Final intensive polish (remaining time)
    if time.time() - t0 < budget_s - 8:
        n_f = len(best_h)
        dx_f = 2.0 / n_f
        ts_f = n_f / 2.0
        for alpha, nact in [(0.3, 50), (0.2, 70), (0.1, 100), (0.05, 150)]:
            if time.time() - t0 > budget_s - 8:
                break
            h_opt, c5_opt = lp_polish(best_h, dx_f, ts_f, t0, time.time() - t0 + (budget_s - 8 - (time.time() - t0)),
                                      n_active=nact, alpha=alpha)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
        if time.time() - t0 < budget_s - 12:
            n_sa_final = min(500000, int((budget_s - 12 - (time.time() - t0)) * 8000))
            h_sa, c5_sa = sa_continuous(n_f, max(10000, n_sa_final), seed + 99999, best_h, 0.3, 0.0005, 0.99997, dx_f, ts_f)
            if c5_sa < best_c5:
                best_c5 = c5_sa
                best_h = h_sa
            h_opt, c5_opt = lp_polish(best_h, dx_f, ts_f, t0, time.time() + 8, n_active=100, alpha=0.08)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    n_final = len(best_h)
    ts_final = n_final / 2.0
    best_h = project_feasible(best_h, ts_final)
    dx_final = 2.0 / n_final
    best_c5 = c5_of_h(best_h, dx_final)
    return best_h, best_c5, n_final
