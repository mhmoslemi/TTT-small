def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        # Need to add mass: prioritize adding to elements with most room
        order = np.argsort(-(1.0 - h))
        remaining = diff
        for i in range(n):
            idx = order[i]
            cap = 1.0 - h[idx]
            if cap <= 1e-15:
                continue
            add = min(remaining, cap)
            h[idx] += add
            remaining -= add
            if remaining <= 1e-14:
                break
    else:
        # Need to remove mass: prioritize removing from elements with most excess
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            if h[idx] <= 1e-15:
                continue
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-14:
                break
    # Final fix
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] = float(np.clip(h[idx] + diff2, 0.0, 1.0))
    return h

def grad_row(h, m, n):

    profile = compute_profile(h)
    dx = 2.0 / n

    # Also: sum(d) = 0 (equality)

    # Also: 0 <= h_j + d_j <= 1 (bounds)

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(-(1.0 - h))
        remaining = diff
        for i in range(n):
            idx = order[i]
            cap = 1.0 - h[idx]
            if cap <= 1e-15:
                continue
            add = min(remaining, cap)
            h[idx] += add
            remaining -= add
            if remaining <= 1e-14:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            if h[idx] <= 1e-15:
                continue
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-14:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] = float(np.clip(h[idx] + diff2, 0.0, 1.0))
    return h


def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))


def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx


def grad_row(h, m, n):
    dx = 2.0 / n
    d = m - (n - 1)
    if d == 0:
        return dx * (1.0 - 2.0 * h)
    elif d > 0:
        g = np.zeros(n)
        g[d:] += dx * (1.0 - h[:-d])
        g[:-d] -= dx * h[d:]
        return g
    else:
        p = -d
        g = np.zeros(n)
        g[:-p] += dx * (1.0 - h[p:])
        g[p:] -= dx * h[:-p]
        return g


def lse_weighted_gradient(h, beta):
    n = len(h)
    profile = compute_profile(h)
    c_max = profile.max()
    z = beta * (profile - c_max)
    z = np.clip(z, -50, 50)
    exp_z = np.exp(z)
    w_sum = exp_z.sum()
    if w_sum < 1e-30:
        w = np.zeros(len(profile))
        w[int(np.argmax(profile))] = 1.0
    else:
        w = exp_z / w_sum
    g = np.zeros(n)
    for m in range(len(profile)):
        if w[m] > 1e-12:
            g += w[m] * grad_row(h, m, n)
    return g


def lse_descent(h, max_iters=150, beta_start=5.0, beta_end=150.0, lr=0.008):
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    best_h = h.copy()
    for it in range(max_iters):
        frac = it / max(max_iters - 1, 1)
        beta = beta_start * (beta_end / beta_start) ** frac
        g = lse_weighted_gradient(h, beta)
        g = g - g.mean()
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-14:
            break
        d = -g / gnorm
        alpha = lr
        c_curr = exact_score(h)
        improved = False
        for _ in range(20):
            h_try = repair_feasible(h + alpha * d)
            c_try = exact_score(h_try)
            if c_try < c_curr - 1e-13:
                h = h_try
                improved = True
                break
            alpha *= 0.5
        if not improved:
            break
        if exact_score(h) < best_c:

def trust_region_step(h, R):
    n = len(h)
    M = 2 * n - 1
    profile = compute_profile(h)
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_ub = np.zeros((M, n + 1))
    for m in range(M):
        A_ub[m, :n] = grad_row(h, m, n)
        A_ub[m, n] = -1.0
    b_ub = -profile
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    lower = np.maximum(-R, -h)
    upper = np.minimum(R, 1.0 - h)
    bounds = list(zip(lower, upper))
    bounds.append((None, None))
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x is not None:
            return res.x[:n]
    except Exception:
        pass
    return None


def trust_region_descent(h, max_iters=80, R_start=0.04, R_shrink=0.65, R_min=0.002):

    best_h = h.copy()
    R = R_start
    for it in range(max_iters):

        if d is None:
            R *= R_shrink

            continue

def interp_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    x_old = np.linspace(0.0, 2.0, n + 1)
    x_new = np.linspace(0.0, 2.0, new_n + 1)
    h_new = np.zeros(new_n)
    for k in range(new_n):
        a, b = x_new[k], x_new[k + 1]
        total = 0.0
        for j in range(n):
            lo = max(a, x_old[j])
            hi = min(b, x_old[j + 1])
            if hi > lo:
                total += h[j] * (hi - lo)
        h_new[k] = total / (b - a)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)


def slsqp_epigraph(h_start, maxiter=150):
    n = len(h_start)
    dx = 2.0 / n
    M = 2 * n - 1
    c0 = exact_score(h_start)

    def obj(x):
        return x[-1]

    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g

    def eq_fun(x):
        return x[:-1].sum() - n / 2.0

    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g

    def ineq_fun(x):
        h = x[:-1]
        t = x[-1]
        profile = np.correlate(h, 1 - h, mode="full") * dx
        return t - profile

    def ineq_jac(x):
        h = x[:-1]
        J = np.zeros((M, n + 1))
        for m in range(M):
            J[m, :n] = -grad_row(h, m, n)
        J[:, -1] = 1.0
        return J

    x0 = np.concatenate([h_start, [c0 + 1e-3]])
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 1.0)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None and np.isfinite(res.x).all():
            h_new = repair_feasible(res.x[:-1])
            c_new = exact_score(h_new)
            if c_new <= c0 + 1e-8:
                return h_new
    except Exception:
        pass
    return None


def active_descent_v2(h, max_iters=300):

            break
        d = -momentum_vec / mv_norm
        pos_mask = d > 1e-12
        neg_mask = d < -1e-12
        alpha_max = 1.0
        if pos_mask.any():
            alpha_max = min(alpha_max, float(((1.0 - h[pos_mask]) / d[pos_mask]).min()))
        if neg_mask.any():
            alpha_max = min(alpha_max, float((h[neg_mask] / (-d[neg_mask])).min()))
        alpha_max = max(alpha_max, 1e-10)
        alpha_max = min(alpha_max, 0.5)
        best_alpha = 0.0
        best_val = c
        best_h_ls = h.copy()
        a_ls, b_ls = 0.0, alpha_max
        for _ in range(18):
            c1 = b_ls - gr_coef * (b_ls - a_ls)
            c2 = a_ls + gr_coef * (b_ls - a_ls)
            h1 = repair_feasible(h + c1 * d)
            h2 = repair_feasible(h + c2 * d)
            f1 = exact_score(h1)
            f2 = exact_score(h2)
            if f1 < best_val:
                best_val = f1
                best_alpha = c1

            if f2 < best_val:

            if f1 < f2:
