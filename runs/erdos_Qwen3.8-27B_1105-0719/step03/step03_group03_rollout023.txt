    """Project vec onto {h: 0≤h≤1, sum(h)=target_sum}."""
    h = np.clip(vec, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    # Adjust: add diff/n to all, then clip, iterate
    for _ in range(100):
        h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)
        new_diff = target_sum - h.sum()
        if abs(new_diff) < 1e-12:
            return h
        # Only adjust the non-saturated components
        mask = (h > 0) & (h < 1)
        if mask.sum() == 0:
            break
        h[mask] += new_diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
        diff = target_sum - h.sum()
        if abs(diff) < 1e-12:
            break
    return h

def project_feasible(vec, target_sum=None):
    n = len(vec)
    if target_sum is None:
        target_sum = n / 2.0
    h = np.clip(vec, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        mask = (h > 0) & (h < 1)
        if mask.sum() == 0:
            h = np.clip(h + diff/len(h), 0, 1)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_grad(h, beta):
    """Compute soft-max approximation and its gradient."""
    n = len(h)
    dx = 2.0 / n
    M = 2*n - 1
    profile = compute_profile(h)
    
    # Soft max for numerical stability
    m = profile.max()
    shifted = beta * (profile - m)
    shifted -= shifted.max()  # additional stability
    exp_s = np.exp(shifted)
    w = exp_s / exp_s.sum()
    val = m + (shifted.max() + np.log(exp_s.sum())) / beta
    
    # Gradient via the G matrix
    G = build_G(h, n, dx)
    grad = G.T @ w
    return val, grad

def build_G(h, n, dx):
    """Build the gradient matrix G where G[d, j] = d(profile[d])/d(h[j])."""
    M = 2*n - 1
    G = np.zeros((M, n))
    center = n - 1
    # d=0: profile[0] = sum h[i](1-h[i])*dx, d/dh[j] = (1-2h[j])*dx
    G[center, :] = dx * (1.0 - 2.0 * h)
    for d in range(1, n):
        idx = np.arange(n - d)
        # Positive shift (center + d): sum_{i=0}^{n-d-1} h[i](1-h[i+d])
        # d/dh[i] = (1-h[i+d])*dx for i < n-d
        # d/dh[i+d] = -h[i]*dx for i < n-d
        G[center+d, idx] += dx * (1.0 - h[idx + d])
        G[center+d, idx + d] += -dx * h[idx]
        # Negative shift (center - d): sum_{i=d}^{n-1} h[i](1-h[i-d])
        # d/dh[i] = (1-h[i-d])*dx
        # d/dh[i-d] = -h[i]*dx
        G[center-d, idx + d] += dx * (1.0 - h[idx])
        G[center-d, idx] += -dx * h[idx + d]
    return G

def projected_gradient_step(h, grad, step_size):
    """Take a projected gradient step."""
    n = len(h)
    h_new = h - step_size * grad
    return project_feasible(h_new)

def pgd_optimize(h0, beta, max_iter=500, lr=None):
    """Optimize using projected gradient with smoothed max."""

    h = h0.copy()

            best_h = h.copy()
