    n = len(h)
    M = 2 * n - 1
    R = np.correlate(h, 1 - h, mode="full")

    Rdx = R * dx  # actual C5 values for each lag

    else:

                G[k, i] += dx * (1.0 - h[i - d])

                G[k, i] += dx * (1.0 - h[i + dp])
            for i in range(dp, n):
                G[k, i] += -dx * h[i - dp]

    n = len(vec)

    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx])
            h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff

    M = 2 * n - 1
    dx = 2.0 / n
    G = np.zeros((M, n))
    for k in range(M):
        d = k - (n - 1)
        if d >= 0:

                G[k, d:] += dx * (1.0 - h[:n-d])

    return G

import numpy as np
from scipy.optimize import linprog, minimize
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full"))) * dx

def profile_h(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx])
            h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    return h

def grad_all(h, n):
    M = 2 * n - 1
    dx = 2.0 / n
    G = np.zeros((M, n))
    for k in range(M):
        d = k - (n - 1)
        if d >= 0:
            if d > 0:
                G[k, :n-d] += -dx * h[d:]
                G[k, d:] += dx * (1.0 - h[:n-d])
            else:
                G[k, :] += dx * (1.0 - 2.0 * h)
        else:
            dp = -d
            G[k, :n-dp] += dx * (1.0 - h[dp:])
            G[k, dp:] += -dx * h[:n-dp]
    return G

def slp_step(h, step_size):
    n = len(h)
    M = 2 * n - 1
    R = np.correlate(h, 1 - h, mode="full")
    dx = 2.0 / n
    Rdx = R * dx
    G = grad_all(h, n)
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_ub = np.zeros((M, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = Rdx - G @ h
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    lb = np.maximum(0.0, h - step_size)
    ub = np.minimum(1.0, h + step_size)
    bounds = list(zip(lb, ub)) + [(0.0, Rdx.max() + 2.0)]
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status != 0:
        return None
    return res.x[:n]

def slp_optimize(h0, t_start, budget_s, max_iters=500):
    n = len(h0)
    h = h0.copy()
    best_c = exact_score(h)
    margin = 20
    step_schedules = [
        [0.05, 0.03, 0.02, 0.01],
        [0.03, 0.02, 0.01, 0.005],
        [0.02, 0.01, 0.005, 0.002],
        [0.01, 0.005, 0.002, 0.001],
    ]
    for phase in range(min(4, max(1, max_iters // 50))):
        schedule = step_schedules[min(phase, len(step_schedules) - 1)]
        for it in range(max_iters // len(step_schedules)):
            if time.perf_counter() - t_start > budget_s - margin:
                return h, best_c
            improved = False
            for ss in schedule:
                if time.perf_counter() - t_start > budget_s - margin:
                    break

                if x_new is None:

                d = x_new - h

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    return repair_feasible(np.clip(h_new, 0.0, 1.0))

def slsqp_minimax(h0, t_start, budget_s, maxiter=300):
    n = len(h0)
    dx = 2.0 / n
    M = 2 * n - 1
    P = profile_h(h0)
    c0 = P.max()
    thresh = c0 - max(0.003, 0.05 * (c0 - np.median(P)))
    active = np.where(P >= thresh)[0]
    if len(active) == 0:
        active = np.array([np.argmax(P)])
    if len(active) > 15:
        top_idx = np.argsort(P[active])[-15:]
        active = active[top_idx]

    def obj(x):
        return x[-1]

    def eq_fun(x):
        return x[:-1].sum() - n / 2.0

    def ineq_fun(x):
        h_x = x[:-1]
        t_val = x[-1]
        prof = profile_h(h_x)
        return t_val - prof[active]

    def ineq_jac(x):
        h_x = x[:-1]
        G = grad_all(h_x, n)
        return np.column_stack([-G[active], np.ones(len(active))])

    x0 = np.concatenate([h0, [c0 + 1e-4]])
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 1.0)]
    cons = [
        {'type': 'eq', 'fun': eq_fun},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    try:
        res = minimize(obj, x0, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = repair_feasible(res.x[:-1])
            c_new = exact_score(h_new)
            if c_new <= c0 + 1e-10:
                return h_new, c_new
    except Exception:
        pass
    return None, c0

def subgradient_step(h, n_active=10, step_size=0.05):
    n = len(h)
    P = profile_h(h)
    cmax = P.max()
    gap = cmax - P.min()
    threshold = cmax - max(0.002, 0.1 * gap)
    active = np.where(P >= threshold)[0]
    if len(active) > n_active:
        active = np.argsort(P[active])[-n_active:]
    G = grad_all(h, n)
    g_avg = G[active].mean(axis=0)
    g_avg -= g_avg.mean()
    gnorm = np.linalg.norm(g_avg)
    if gnorm < 1e-15:
        return h.copy()
    direction = -g_avg / gnorm
    h_new = repair_feasible(h + step_size * direction)
    return h_new

                h_s, c_s = slp_optimize(h_b, t_start, budget_s, max_iters=400)
                if c_s < best_c - 1e-13:
                    best_h = h_s
                    best_c = c_s
                else:
                    best_h = h_b
                    best_c = c_b

    # Phase 5: Multi-start SLP with structured perturbations
    if time.perf_counter() - t_start < budget_s - 400:
        perturbation_types = [
            "random", "symmetric", "block", "alternating"
        ]
        for pt_idx, pt in enumerate(perturbation_types[:4]):
            if time.perf_counter() - t_start >= budget_s - 300:
                break
            eps = 0.04 + 0.02 * pt_idx
            if pt == "random":
                z = rng.randn(n)
                z -= z.mean()
                h_p = repair_feasible(best_h + eps * z)
            elif pt == "symmetric":
                z = rng.randn(n)
                z = (z + z[::-1]) / 2.0
                z -= z.mean()
                h_p = repair_feasible(best_h + eps * z)
            elif pt == "block":
                z = np.zeros(n)
                half = n // 2
                z[:half] = eps
                z[half:] = -eps
                z -= z.mean()
                h_p = repair_feasible(best_h + z)
            else:
                z = np.zeros(n)
                z[0::2] = eps
                z[1::2] = -eps
                z -= z.mean()
                h_p = repair_feasible(best_h + z)
            h_s, c_s = slp_optimize(h_p, t_start, budget_s, max_iters=300)
            if c_s < best_c - 1e-13:
                best_h = h_s
                best_c = c_s

    # Phase 6: Try different dimensions
    if time.perf_counter() - t_start < budget_s - 300:
        n_best = len(best_h)
        cand_ns = sorted(set([n_best - 2, n_best + 2, n_best - 4, n_best + 4,
                              n_best + 6, n_best - 6]),
                         key=lambda x: abs(x - n_best))
        for new_n in cand_ns[:5]:
            if time.perf_counter() - t_start >= budget_s - 200:
                break
            if new_n < 30 or new_n > 200:
                continue
            h_interp = interpolate_to_n(best_h, new_n)
            h_s, c_s = slp_optimize(h_interp, t_start, budget_s, max_iters=500)
            if c_s < best_c - 1e-13:
                best_h = h_s
                best_c = c_s

    # Phase 7: Subgradient descent from best (different direction finding)
    if time.perf_counter() - t_start < budget_s - 200:
        h_sg, c_sg = subgradient_optimize(best_h, t_start, budget_s, max_iters=200)
        if c_sg < best_c - 1e-14:
            best_h = h_sg
            best_c = c_sg

    # Phase 8: Combined SLP + SLSQP final polish
    if time.perf_counter() - t_start < budget_s - 150:
        h_slp, c_slp = slp_optimize(best_h, t_start, budget_s, max_iters=400)
        if c_slp < best_c - 1e-14:
            best_h = h_slp
            best_c = c_slp
    if time.perf_counter() - t_start < budget_s - 80:
        h_sqp, c_sqp = slsqp_minimax(best_h, t_start, budget_s, maxiter=300)
        if h_sqp is not None and c_sqp < best_c - 1e-14:
            best_h = h_sqp
            best_c = c_sqp
    if time.perf_counter() - t_start < budget_s - 40:
        h_slp, c_slp = slp_optimize(best_h, t_start, budget_s, max_iters=200)
        if c_slp < best_c - 1e-14:
            best_h = h_slp
            best_c = c_slp

    # Final validation
    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)

    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            final_c = float(eval_result)
        elif isinstance(eval_result, (tuple, list)) and len(eval_result) >= 1:
            if np.isfinite(eval_result[0]):
                final_c = float(eval_result[0])
    except Exception:
        pass

    return best_h, final_c, len(best_h)
