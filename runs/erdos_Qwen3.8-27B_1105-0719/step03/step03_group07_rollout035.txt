import numpy as np
from scipy.optimize import linprog, minimize
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full"))) * dx

def profile_h(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def repair_feasible(h):
    n = len(h)
    h = np.clip(h.astype(np.float64).copy(), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)  # most room first
        remaining = diff
        for i in order:
            add = min(remaining, 1.0 - h[i])
            h[i] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)  # smallest first
        remaining = -diff
        for i in order:
            rem = min(remaining, h[i])
            h[i] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    # Final fix
    diff2 = target - h.sum()
    if abs(diff2) > 1e-14:
        idx = int(np.argmax(1.0 - h)) if diff2 > 0 else int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def overlap_gradients(h, lags):
    """Compute gradient rows for given lag indices.
    lags: array of lag values (0 to 2n-2, index into full correlate output).
    Returns: array of shape (len(lags), n)
    """
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((len(lags), n))
    for idx, m in enumerate(lags):
        d = m - (n - 1)  # actual shift
        if d >= 0:
            # O = sum_{i=d}^{n-1} h[i]*(1-h[i-d]) * dx
            # dO/dh[j] = (1-h[j-d])*dx for d<=j<=n-1, minus h[j+d]*dx for j<=n-1-d
            for j in range(d, n):
                G[idx, j] += dx * (1.0 - h[j - d])
            for j in range(0, n - d):
                G[idx, j] -= dx * h[j + d]
        else:
            dp = -d
            # O = sum_{i=0}^{n-1-dp} h[i]*(1-h[i+dp]) * dx
            # dO/dh[j] = (1-h[j+dp])*dx for j<=n-1-dp, minus h[j-dp]*dx for dp<=j<=n-1
            for j in range(0, n - dp):
                G[idx, j] += dx * (1.0 - h[j + dp])
            for j in range(dp, n):
                G[idx, j] -= dx * h[j - dp]
    return G

def lp_direction(h, active_lags, n_vars=None):
    """Find LP direction that decreases all active overlaps simultaneously.
    Returns direction g and optimal t, or None if infeasible/no improvement.
    """
    n = len(h)
    if n_vars is None:
        n_vars = n
    G = overlap_gradients(h, active_lags)  # shape (m, n)
    m = len(active_lags)
    
    # Variables: g[0], ..., g[n-1], t  => n+1 variables
    # Minimize t = e^T x where x = [g; t]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Inequality constraints: G @ g <= t  =>  G @ g - t <= 0
    # In matrix form: [G, -1] @ x <= 0
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = G
    A_ub[:, -1] = -1.0
    b_ub = np.zeros(m)
    
    # Equality: sum(g) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds: -1 <= g[j] <= 1, t unbounded below, bounded above by 0
    bounds = [(-1.0, 1.0)] * n + [(None, 0.0)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0 and res.fun < -1e-10:
            g = res.x[:n]
            return g, res.fun
        elif res.status == 0:
            return None, res.fun
        else:
            return None, None
    except Exception:
        return None, None

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    return repair_feasible(h_new)

def line_search(h, g, best_c, max_alpha=1.0):
    """Find best alpha along direction g."""
    n = len(h)
    # Check if g stays in [0,1]
    if np.any(h + max_alpha * g < -0.01) or np.any(h + max_alpha * g > 1.01):
        # Scale down
        alphas = []
        for j in range(n):
            if g[j] > 0 and h[j] < 1.0:
                alphas.append((1.0 - h[j]) / g[j])
            elif g[j] < 0 and h[j] > 0.0:
                alphas.append(h[j] / (-g[j]))
        if alphas:
            max_alpha = max(0.0, min(alphas) * 0.99)
        else:
            return h, best_c
    
    best_h = h
    best_s = best_c
    # Try a range of alphas
    alphas = np.linspace(0.02, min(max_alpha, 0.5), 10)
    for a in alphas:
        h_try = repair_feasible(h + a * g)
        s = exact_score(h_try)
        if s < best_s:
            best_s = s
            best_h = h_try
    # Also try larger steps
    for a in [0.5, 0.8, max_alpha * 0.9, max_alpha]:
        if a <= 0:
            continue
        h_try = repair_feasible(h + a * g)
        s = exact_score(h_try)
        if s < best_s:
            best_s = s
            best_h = h_try
    return best_h, best_s

def lp_direction_optimize(h, max_iters=50, t_start=None, budget_s=1100, K=15):
    """Iterative LP direction optimization."""
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    
    for it in range(max_iters):
        if t_start is not None and time.perf_counter() - t_start > budget_s - 30:
            break
        
        P = profile_h(h)
        c_max = P.max()
        
        # Select active lags: those within a tolerance of the max
        tol = max(0.005, 0.1 * (c_max - np.median(P)))
        active_mask = P >= c_max - tol

import numpy as np
from scipy.optimize import linprog
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full"))) * dx

def profile_h(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def repair_feasible(h):
    n = len(h)
    h = np.clip(h.astype(np.float64).copy(), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        remaining = diff
        for i in order:
            add = min(remaining, 1.0 - h[i])
            h[i] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in order:
            rem = min(remaining, h[i])
            h[i] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-14:
        idx = int(np.argmax(1.0 - h)) if diff2 > 0 else int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def overlap_gradients(h, lag_indices):
    """Compute gradient of each overlap w.r.t. h for given lag indices.
    lag_indices: indices into the full correlate output (0 to 2n-2).
    Returns: (len(lag_indices), n) array.
    """
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((len(lag_indices), n))
    for idx, m in enumerate(lag_indices):
        d = m - (n - 1)
        if d >= 0:
            for j in range(d, n):
                G[idx, j] += dx * (1.0 - h[j - d])
            for j in range(0, n - d):
                G[idx, j] -= dx * h[j + d]
        else:
            dp = -d
            for j in range(0, n - dp):
                G[idx, j] += dx * (1.0 - h[j + dp])
            for j in range(dp, n):
                G[idx, j] -= dx * h[j - dp]
    return G

def find_lp_direction(h, lag_indices):
    """Solve LP to find direction decreasing all specified overlaps.
    Returns (direction, t_value) or (None, t_value).
    """
    n = len(h)
    G = overlap_gradients(h, lag_indices)
    m = len(lag_indices)
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = np.zeros(m)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    bounds = [(-1.0, 1.0)] * n + [(None, 0.0)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g = res.x[:n]
            return g, float(res.fun)
        return None, None
    except Exception:
        return None, None

def line_search_alpha(h, g, best_c):
    """Find best step size along direction g."""
    n = len(h)
    max_a = 1.0
    for j in range(n):
        if g[j] > 0.01:
            max_a = min(max_a, (1.0 - h[j]) / g[j])

    return best_h, best_s

                    break
                continue
        
        stalled = 0

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)

    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    return repair_feasible(h_new)

import numpy as np
from scipy.optimize import linprog
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full"))) * dx

def profile_h(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def repair_feasible(h):
    n = len(h)
    h = np.clip(h.astype(np.float64).copy(), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        remaining = diff
        for i in order:
            add = min(remaining, 1.0 - h[i])
            h[i] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in order:
            rem = min(remaining, h[i])
            h[i] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-14:
        idx = int(np.argmax(1.0 - h)) if diff2 > 0 else int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def overlap_gradients(h, lag_indices):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((len(lag_indices), n))
    for idx, m in enumerate(lag_indices):
        d = m - (n - 1)
        if d >= 0:
            for j in range(d, n):
                G[idx, j] += dx * (1.0 - h[j - d])
            for j in range(0, n - d):
                G[idx, j] -= dx * h[j + d]
        else:
            dp = -d
            for j in range(0, n - dp):
                G[idx, j] += dx * (1.0 - h[j + dp])
            for j in range(dp, n):
                G[idx, j] -= dx * h[j - dp]
    return G

def find_lp_direction(h, lag_indices):
    n = len(h)
    G = overlap_gradients(h, lag_indices)
    m = len(lag_indices)
    if m == 0:
        return None, None
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = np.zeros(m)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(-1.0, 1.0)] * n + [(None, 0.0)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            return res.x[:n], float(res.fun)
        return None, None
    except Exception:
        return None, None

def line_search_alpha(h, g, best_c):
    n = len(h)
    max_a = 2.0
    for j in range(n):
        if g[j] > 0.001:
            max_a = min(max_a, (1.0 - h[j]) / g[j])
        elif g[j] < -0.001:
            max_a = min(max_a, h[j] / (-g[j]))
    max_a = max(0.01, min(max_a * 0.98, 2.0))
    best_h = h
    best_s = best_c
    n_pts = 12
    alphas = np.linspace(0.01, min(max_a, 0.4), n_pts)

        if a <= 0 or a > max_a:

        s = exact_score(h_try)
        if s < best_s - 1e-15:
            best_s = s
            best_h = h_try
    return best_h, best_s

def lp_optimize(h, max_iters=60, K=15, t_start=None, budget_s=1100, rng=None):

        spread = c_max - np.percentile(P, 20)
        tol = max(0.002, 0.25 * spread)
        active = np.where(P >= c_max - tol)[0]
        if len(active) > K:
            active = active[np.argsort(P[active])[-K:]]
        g, t = find_lp_direction(h, active)
        if g is None or t >= -1e-9:
            found = False
            for k2 in [min(8, len(active)), 5, 3, 2]:
                if len(active) <= k2:
                    continue
                small = active[np.argsort(P[active])][-k2:]

                    found = True
                    break
            if not found:
                stalled += 1
                if stalled > 8:
                    break
                continue
        stalled = 0
        h, best_c = line_search_alpha(h, g, best_c)
    return h, best_c

    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)

                s = exact_score(h_fixed)
