def nesterov_descent(h0, max_iters, t_start, budget_s, beta_schedule=None):

        lr = 0.1 / (beta + 1)

                return best_h, best_c

            # Project gradient (remove mean to stay in sum-constraint affine space)
            grad_c = grad - grad.mean()
            
            # Nesterov update
            y = h + momentum * vel
            vel = -lr * grad_c  # gradient of the approximation
            # Actually for Nesterov: 
            # y_{k+1} = x_{k+1} + (t_{k+1}-1)/t_{k+2} * (x_{k+1} - x_k)
            # Simplified: momentum-based
            vel = momentum * vel + (-lr * grad_c)
            h_new = project_feasible(h + vel)
            
            c_new = exact_score(h_new)
            if c_new < best_c:
                best_h = h_new.copy()
                best_c = c_new
            if c_new < exact_score(h) + 1e-12:
                h = h_new
            else:
                # Step back
                lr *= 0.5
                vel = np.zeros(n)
    
    return best_h, best_c

                if nv < -1e-12 or nv > 1.0 + 1e-12:
                    continue
                h[i] = nv
                h = repair_feasible(h)
                s = exact_score(h)
                if s < best_c - 1e-14:
                    best_c = s
                    improved = True
                    break
                else:
                    h[i] = h_save
            h = repair_feasible(h)
        if not improved:
            break
    return h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()
    rng = np.random.RandomState(seed)

    def time_ok(margin=35):
        return time.perf_counter() - t_start < budget_s - margin

    parent = None
    try:
        parent = np.array(initial_h_values, dtype=np.float64)
    except (NameError, Exception):
        pass

    if parent is None or len(parent) < 4:
        parent = np.full(70, 0.5)

    parent = repair_feasible(np.clip(parent, 0.0, 1.0))
    best_h = parent.copy()
    best_c = exact_score(parent)

    # Phase 1: Optimize at parent dimension with smooth descent + SLSQP
    if time_ok(120):
        h_md, c_md = smooth_descent(best_h, max_iters=120, t_start=t_start, budget_s=budget_s)
        if c_md < best_c:
            best_h = h_md
            best_c = c_md
        h_sol = multi_slsqp(best_h, rounds=5, active_k=16, maxiter=300)
        c_sol = exact_score(h_sol)
        if c_sol < best_c:
            best_h = h_sol
            best_c = c_sol

    # Phase 2: Dimension exploration - interpolate to various sizes
    if time_ok(200):
        n_parent = len(best_h)
        dims = sorted(set(
            list(range(max(36, n_parent - 16), n_parent, 2)) +
            list(range(n_parent + 2, min(136, n_parent + 22), 2))
        ))
        for new_n in dims[:12]:
            if not time_ok(25):
                break
            h_interp = interpolate_to_n(best_h, new_n)
            h_md, c_md = smooth_descent(h_interp, max_iters=60, t_start=t_start, budget_s=budget_s)
            h_sol = multi_slsqp(h_md, rounds=3, active_k=14, maxiter=200)
            c_sol = exact_score(h_sol)
            if c_sol < best_c - 1e-10:
                best_h = h_sol
                best_c = c_sol

    # Phase 3: Structured restarts at best dimension
    if time_ok(150):
        n_best = len(best_h)
        starts = []
        # Binary starts with different structure
        for s in range(4):
            h_bin = binary_start(n_best, seed + 200 + s * 7)
            starts.append(h_bin)
        # Periodic starts
        for k in [3, 4, 5, 6, 8]:
            if n_best % k == 0:
                h_per = periodic_start(n_best, k)
                c_per = exact_score(h_per)
                if c_per < best_c + 0.05:
                    starts.append(h_per)
        # Mixed with parent
        for mix in [0.4, 0.6]:
            for s in range(2):
                h_bin = binary_start(n_best, seed + 500 + s * 13)
                h_mix = repair_feasible(mix * h_bin + (1 - mix) * best_h)
                starts.append(h_mix)

        for h0 in starts:
            if not time_ok(15):
                break
            h_md, c_md = smooth_descent(h0, max_iters=50, t_start=t_start, budget_s=budget_s)
            h_sol = multi_slsqp(h_md, rounds=3, active_k=12, maxiter=150)
            c_sol = exact_score(h_sol)
            if c_sol < best_c - 1e-10:
                best_h = h_sol
                best_c = c_sol

    # Phase 4: Perturbation multi-start from best
    if time_ok(100):
        n_best = len(best_h)
        for trial in range(6):
            if not time_ok(15):
                break
            eps = 0.005 * (trial + 1)
            z = rng.randn(n_best)
            z -= z.mean()
            z /= (np.linalg.norm(z) + 1e-10)
            h_pert = repair_feasible(best_h + eps * z)
            h_md, c_md = smooth_descent(h_pert, max_iters=60, t_start=t_start, budget_s=budget_s)
            h_sol = multi_slsqp(h_md, rounds=4, active_k=14, maxiter=200)
            c_sol = exact_score(h_sol)
            if c_sol < best_c - 1e-12:
                best_h = h_sol
                best_c = c_sol

    # Phase 5: Coordinate descent polish
    if time_ok(40):
        best_h, best_c = coordinate_descent(best_h, max_sweeps=4,
                                            t_start=t_start, budget_s=budget_s)

    # Phase 6: Final SLSQP with tight tolerance
    if time_ok(30):
        h_sol = multi_slsqp(best_h, rounds=3, active_k=12, maxiter=300)
        c_sol = exact_score(h_sol)
        if c_sol < best_c - 1e-13:
            best_h = h_sol
            best_c = c_sol

    # Phase 7: One more dimension attempt if time allows
    if time_ok(60):
        n_best = len(best_h)
        for new_n in [n_best - 4, n_best - 2, n_best + 2, n_best + 4]:
            if new_n < 30 or new_n > 130 or new_n % 2 != 0:
                continue
            if not time_ok(20):
                break
            h_interp = interpolate_to_n(best_h, new_n)
            h_md, c_md = smooth_descent(h_interp, max_iters=40, t_start=t_start, budget_s=budget_s)
            h_sol = multi_slsqp(h_md, rounds=3, active_k=14, maxiter=200)
            c_sol = exact_score(h_sol)
            if c_sol < best_c - 1e-10:
                best_h = h_sol
                best_c = c_sol
                best_h, best_c = coordinate_descent(best_h, max_sweeps=3,
                                                     t_start=t_start, budget_s=budget_s)

    # Final validation
    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)

    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            final_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 1:
            if np.isfinite(eval_result[0]):
                final_c = float(eval_result[0])
    except Exception:
        pass

    return best_h, final_c, len(best_h)
