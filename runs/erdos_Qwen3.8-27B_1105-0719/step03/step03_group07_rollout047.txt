import numpy as np
from scipy.optimize import minimize
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full"))) * dx

def profile_h(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx])
            h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    return h

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    return repair_feasible(np.clip(h_new, 0.0, 1.0))

def compute_gradient_matrix(h):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    G = np.zeros((M, n))
    for m in range(M):
        d = m - (n - 1)
        if d == 0:
            G[m, :] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            for i in range(n - d):
                G[m, i + d] += dx * (1.0 - h[i])
                G[m, i] -= dx * h[i + d]
        else:
            dp = -d
            for j in range(n - dp):
                G[m, j + dp] += dx * (1.0 - h[j])
                G[m, j] -= dx * h[j + dp]
    return G

def softmax_descent(h, t_start, budget_s, max_iters=200):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best_c = exact_score(h)
    vel = np.zeros(n)
    
    betas = [30, 60, 120, 250, 500]
    for beta in betas:
        alpha = 0.015
        momentum = 0.4
        for it in range(max_iters // len(betas)):
            if time.perf_counter() - t_start > budget_s - 25:
                return h, best_c
            P = profile_h(h)
            Pmax = P.max()
            exp_arg = beta * (P - Pmax)
            exp_arg -= exp_arg.max()
            weights = np.exp(exp_arg)
            S = weights.sum()
            if S < 1e-300:
                weights = np.zeros(len(P))
                weights[np.argmax(P)] = 1.0
            else:
                weights /= S
            G = compute_gradient_matrix(h)
            active = np.where(weights > 1.0 / len(P))[0]
            if len(active) == 0:
                active = np.argsort(weights)[-4:]
            w_act = weights[active]
            w_act /= w_act.sum()
            grad = G[active].T @ w_act
            sum_err = h.sum() - n / 2.0
            grad += 40.0 * sum_err / n
            grad -= grad.mean()
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break
            grad /= gnorm
            vel = momentum * vel + grad
            vnorm = np.linalg.norm(vel)
            if vnorm > 0:
                vel /= vnorm
            improved = False
            for a in [alpha, alpha * 0.5, alpha * 0.25, alpha * 0.1]:
                h_try = repair_feasible(h - a * vel)
                s = exact_score(h_try)
                if s < best_c - 1e-15:
                    h = h_try
                    best_c = s
                    improved = True
                    break
                elif s < exact_score(h) - 1e-16:
                    h = h_try
                    c_now = exact_score(h)
                    if c_now < best_c:
                        best_c = c_now
            if improved:
                alpha = min(alpha * 1.03, 0.06)
            else:
                alpha *= 0.55
                if alpha < 1e-8:
                    break
    return h, best_c

                       options={'maxiter': maxiter, 'ftol': tol})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = repair_feasible(res.x[:-1])
            c_new = exact_score(h_new)
            if c_new <= c0 + 5e-10:
                return h_new
    except Exception:
        pass
    return None

def pairwise_cd(h, t_start, budget_s, max_sweeps=8):
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    
    for sweep in range(max_sweeps):
        if time.perf_counter() - t_start > budget_s - 15:
            break
        G = compute_gradient_matrix(h)
        P = profile_h(h)
        Pmax = P.max()
        active = np.where(P >= Pmax - 0.005)[0]
        G_active = G[active]
        
        for i in range(n):
            if time.perf_counter() - t_start > budget_s - 15:
                break
            grad_i = G_active[:, i]
            for j in range(i + 1, n):
                if time.perf_counter() - t_start > budget_s - 15:
                    break
                diff_grad = grad_i - G_active[:, j]
                max_change = np.max(diff_grad)
                if max_change > 1e-6:
                    for delta in [0.005, 0.003, 0.001]:
                        if h[i] + delta > 1.0 or h[j] - delta < 0.0:
                            continue
                        h[i] += delta
                        h[j] -= delta
                        s = exact_score(h)
                        if s < best_c - 1e-15:
                            best_c = s
                            break
                        else:
                            h[i] -= delta
                            h[j] += delta
                        if s >= best_c:
                            break
    return h, best_c

def subgradient_nesterov(h, t_start, budget_s, max_iters=300):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c = exact_score(h)
    y = h.copy()
    vel = np.zeros(n)
    t_mom = 1.0
    t_count = 1
    
    for it in range(max_iters):
        if time.perf_counter() - t_start > budget_s - 20:
            break
        t_count += 1
        t_mom = (1 + np.sqrt(1 + 4 * t_count * t_count)) / 2
        
        P = profile_h(y)
        Pmax = P.max()
        beta = 20.0 + 3.0 * it
        exp_arg = beta * (P - Pmax)
        exp_arg -= exp_arg.max()
        weights = np.exp(exp_arg)
        S = weights.sum()
        if S < 1e-300:
            weights = np.zeros(len(P))
            weights[np.argmax(P)] = 1.0
        else:
            weights /= S
        
        active = np.where(weights > 1.0 / len(P))[0]
        if len(active) == 0:
            active = np.argsort(weights)[-3:]
        w_act = weights[active]
        w_act /= w_act.sum()
        G = compute_gradient_matrix(y)
        grad = G[active].T @ w_act
        sum_err = y.sum() - n / 2.0
        grad += 30.0 * sum_err / n
        
        alpha = 0.02 / (1.0 + 0.01 * it)
        h_new = repair_feasible(y - alpha * grad)
        vel = (t_count - 1) / t_mom * vel
        y = h_new + (t_count - 1) / t_mom * (h_new - h)
        y = repair_feasible(y)
        h = h_new
        
        s = exact_score(h)
        if s < best_c - 1e-15:
            best_c = s
            best_h = h.copy()
    
    return best_h, best_c

def try_dimension(h_current, dim, t_start, budget_s):
    if dim == len(h_current):
        return h_current, exact_score(h_current)
    h_interp = interpolate_to_n(h_current, dim)
    sol = slsqp_solve(h_interp, maxiter=200, active_k=12, tol=1e-11)
    if sol is not None:
        c = exact_score(sol)
        if c < exact_score(h_interp):
            h_interp = sol
    h_interp, c = softmax_descent(h_interp, t_start, budget_s, max_iters=60)
    sol = slsqp_solve(h_interp, maxiter=250, active_k=15, tol=1e-12)
    if sol is not None:
        c2 = exact_score(sol)
        if c2 < c:
            h_interp = sol
            c = c2
    return h_interp, c

def symmetrize(h):
    return (h + h[::-1].copy()) / 2.0

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()
    rng = np.random.RandomState(seed)
    
    def time_ok(margin=15):
        return time.perf_counter() - t_start < budget_s - margin
    
    parent = None
    try:
        parent = np.array(initial_h_values, dtype=np.float64)
    except NameError:
        pass
    except Exception:
        pass
    
    if parent is None or len(parent) < 4:
        parent = np.full(60, 0.5)
    
    parent = np.clip(parent, 0.0, 1.0)
    parent = repair_feasible(parent)
    
    best_h = parent.copy()
    best_c = exact_score(parent)
    n0 = len(best_h)
    
    # Phase 1: Quick SLSQP at parent dimension
    if time_ok(100):
        sol = slsqp_solve(best_h, maxiter=300, active_k=15, tol=1e-11)
        if sol is not None:
            c = exact_score(sol)
            if c < best_c:
                best_h, best_c = sol, c
        if time_ok(80):
            sol = slsqp_wider(best_h, maxiter=300, active_k=20, tol=1e-12)
            if sol is not None:
                c = exact_score(sol)
                if c < best_c:
                    best_h, best_c = sol, c
    
    # Phase 2: Nesterov subgradient
    if time_ok(150):
        h_ns, c_ns = subgradient_nesterov(best_h, t_start, budget_s, max_iters=150)
        if c_ns < best_c - 1e-14:
            best_h, best_c = h_ns, c_ns
    
    # Phase 3: SLSQP after subgradient
    if time_ok(120):
        sol = slsqp_solve(best_h, maxiter=350, active_k=15, tol=1e-12)
        if sol is not None:
            c = exact_score(sol)
            if c < best_c - 1e-14:
                best_h, best_c = sol, c
    
    # Phase 4: Dimension exploration
    if time_ok(200):
        n_best = len(best_h)
        dims_to_try = []
        for d in range(max(24, n_best - 20), min(130, n_best + 30)):
            if d % 2 == 0 and d != n_best and d >= 24:
                dims_to_try.append(d)
        # Also try some structurally different dimensions
        for d in [30, 36, 40, 48, 56, 64, 80, 96, 100, 108, 120]:
            if d % 2 == 0 and d not in dims_to_try and d >= 24:
                dims_to_try.append(d)
        dims_to_try.sort(key=lambda d: abs(d - n_best))
        
        for new_n in dims_to_try[:12]:
            if not time_ok(30):
                break
            h_try, c_try = try_dimension(best_h, new_n, t_start, budget_s)
            if c_try < best_c - 1e-12:
                best_h, best_c = h_try, c_try
                # Polish at this dimension
                if time_ok(40):
                    sol = slsqp_wider(best_h, maxiter=300, active_k=20, tol=1e-12)
                    if sol is not None:
                        c2 = exact_score(sol)
                        if c2 < best_c - 1e-14:
                            best_h, best_c = sol, c2
    
    # Phase 5: Multi-start SLSQP with perturbations
    if time_ok(150):
        n = len(best_h)
        for trial in range(4):
            if not time_ok(25):
                break
            eps = 0.005 + 0.01 * trial
            z = rng.randn(n)
            z -= z.mean()
            h_pert = repair_feasible(best_h + eps * z)
            # Symmetric variant
            h_pert_sym = repair_feasible(symmetrize(h_pert))
            for h0 in [h_pert, h_pert_sym]:
                if not time_ok(15):
                    break
                for ak in [10, 15, 20]:
                    if not time_ok(10):
                        break
                    sol = slsqp_solve(h0, maxiter=250, active_k=ak, tol=1e-11)
                    if sol is not None:
                        c = exact_score(sol)
                        if c < best_c - 1e-13:
                            best_h, best_c = sol, c
    
    # Phase 6: Another subgradient round
    if time_ok(100):
        h_ns, c_ns = subgradient_nesterov(best_h, t_start, budget_s, max_iters=100)
        if c_ns < best_c - 1e-14:
            best_h, best_c = h_ns, c_ns
        if time_ok(60):
            sol = slsqp_solve(best_h, maxiter=350, active_k=15, tol=1e-12)
            if sol is not None:
                c = exact_score(sol)
                if c < best_c - 1e-14:
                    best_h, best_c = sol, c
    
    # Phase 7: Pairwise coordinate descent
    if time_ok(60):
        best_h, best_c = pairwise_cd(best_h, t_start, budget_s, max_sweeps=6)
    
    # Phase 8: Final polish with tight tolerance
    if time_ok(30):
        sol = slsqp_solve(best_h, maxiter=400, active_k=10, tol=1e-13)
        if sol is not None:
            c = exact_score(sol)
            if c < best_c - 1e-15:
                best_h, best_c = sol, c
        if time_ok(15):
            sol = slsqp_wider(best_h, maxiter=300, active_k=18, tol=1e-13)
            if sol is not None:
                c = exact_score(sol)
                if c < best_c - 1e-15:
                    best_h, best_c = sol, c
    
    # Final validation
    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)
    
    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            final_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 1:
            if np.isfinite(eval_result[0]):
                final_c = float(eval_result[0])
    except Exception:
        pass
    
    return best_h, final_c, len(best_h)
