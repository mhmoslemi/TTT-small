    else:

    G = compute_gradient_profile(h)
    if active is not None:
        G = G[active, :]
    grad = G.T @ weights
    return grad, weights

    profile = compute_profile(h)
    pmax = profile.max()
    shifted = (profile - pmax) / tau
    return pmax + tau * np.log(np.sum(np.exp(shifted - shifted.max())))

import numpy as np
from scipy.optimize import minimize, linprog

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def compute_gradient_matrix(h):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    G = np.zeros((M, n))
    center = n - 1
    G[center, :] = dx * (1.0 - 2.0 * h)
    for d in range(1, n):
        G[center + d, d:n] += dx * (1.0 - h[0:n-d])
        G[center + d, 0:n-d] += -dx * h[d:n]
        G[center - d, d:n] += dx * (1.0 - h[0:n-d])
        G[center - d, 0:n-d] += -dx * h[d:n]
    return G

def waterfill_project(h, target=None):
    n = len(h)
    if target is None:
        target = n / 2.0
    h = np.asarray(h, dtype=np.float64)
    lo = h.min() - 1.5

        t = (lo + hi) / 2.0
        s = np.clip(h - t, 0.0, 1.0).sum()
        if s > target:
            lo = t
        else:
            hi = t
    t = (lo + hi) / 2.0
    result = np.clip(h - t, 0.0, 1.0)
    diff = target - result.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            mask = result < 1.0 - 1e-12
            if mask.any():

            if mask.any():
                idx = np.where(mask)[0][np.argmin(result[mask])]
                result[idx] = max(result[idx] + diff, 0.0)
    return np.clip(result, 0.0, 1.0)

def repair_feasible(h):

def lse_smoothed_grad(h, tau, active_idx=None):
    n = len(h)
    profile = compute_profile(h)
    if active_idx is not None:
        p = profile[active_idx]
        G = compute_gradient_matrix(h)[active_idx, :]
    else:
        p = profile
        G = compute_gradient_matrix(h)
    pmax = p.max()
    shifted = (p - pmax) / tau
    shifted -= shifted.max()
    exp_s = np.exp(shifted)
    w = exp_s / exp_s.sum()
    grad = G.T @ w
    return grad, w

def lse_value(h, tau, active_idx=None):

        p = profile[active_idx]
    else:
        p = profile
    pmax = p.max()

    return pmax + tau * np.log(np.sum(np.exp(shifted - shifted.max())))

def backtracking_line_search(h0, grad, func, max_iter=30):
    n = len(h0)

def slsqp_epigraph(h, t_start, budget_s, top_k=40, maxiter=800):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    profile = compute_profile(h)
    if top_k is not None and top_k < M:
        idx = np.argsort(profile)[-top_k:]
    else:
        idx = np.arange(M)
    K = len(idx)

    def obj_fn(x):
        return x[-1]

    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g

    def eq_fn(x):
        return x[:-1].sum() - n / 2.0

    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g

    def ineq_fn(x):
        h_v = x[:-1]
        t = x[-1]
        prof = compute_profile(h_v)
        return t - prof[idx]

    def ineq_jac(x):
        h_v = x[:-1]
        G = compute_gradient_matrix(h_v)
        J = np.zeros((K, n + 1))
        J[:, :-1] = -G[idx, :]
        J[:, -1] = 1.0
        return J

    c0 = exact_score(h)
    x0 = np.zeros(n + 1)
    x0[:-1] = h
    x0[-1] = c0 + 0.005
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    cons = [
        {'type': 'eq', 'fun': eq_fn, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fn, 'jac': ineq_jac}
    ]

    try:
        res = minimize(obj_fn, x0, jac=obj_jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.success and np.isfinite(res.x).all():

            if c_sol < c0:
                return h_sol, c_sol
    except Exception:
        pass

    if top_k is not None and top_k > 10:

    for _ in range(3):
        h = rng.uniform(0.1, 0.9, n)
        # Smooth
        from scipy.ndimage import uniform_filter1d
        try:
            h = uniform_filter1d(h, size=max(2, n // 10))
        except Exception:
            pass
        starts.append(waterfill_project(h))
    # Symmetric
    half = n // 2
    h_half = rng.uniform(0.1, 0.9, half)
    h_sym = np.concatenate([h_half, h_half[::-1]])
    if n % 2 == 1:
        h_sym = np.insert(h_sym, half, 0.5)
    starts.append(waterfill_project(h_sym[:n]))
    return starts

def lp_active_step(h, active, radius):
    n = len(h)
    dx = 2.0 / n
    G = compute_gradient_matrix(h)[active, :]
    profile = compute_profile(h)
    p_a = profile[active]
    M = len(active)

    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_ub = np.zeros((M, n + 1))
    A_ub[:, :n] = G
    A_ub[:, -1] = -1.0
    b_ub = -p_a
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    lower = np.maximum(-h, -radius)
    upper = np.minimum(1.0 - h, radius)
    bounds = list(zip(lower, upper))
    bounds.append((0.0, None))

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'maxiter': 3000})
        if res.success and res.x is not None and np.isfinite(res.x).all():
            h_new = waterfill_project(h + res.x[:n])
            return h_new
    except Exception:
        pass
    return None

def active_set_optimize(h, t_start, budget_s, max_iters=200):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c = exact_score(h)
    radius = 0.03
    no_improve = 0

    for it in range(max_iters):
        elapsed = time.perf_counter() - t_start
        if budget_s - elapsed < 5.0:
            break

        profile = compute_profile(h)
        pmax = profile.max()
        c = pmax
        if c < best_c - 1e-14:
            best_c = c
            best_h = h.copy()
            no_improve = 0
            radius = min(radius * 1.3, 0.1)
        else:
            no_improve += 1
        if no_improve > 50:
            break

        # Get active set
        spread = pmax - profile.min()
        beam = max(0.001, 0.1 * spread)
        active = np.where(profile >= pmax - beam)[0]
        if len(active) > 15:
            active = np.argsort(profile)[-15:]

        # LP step
        h_lp = lp_active_step(h, active, radius)
        if h_lp is not None:
            direction = h - h_lp
            # Line search along direction
            best_s = c
            best_step_h = h.copy()
            for alpha in [1.0, 0.7, 0.5, 0.3, 0.15]:
                h_try = waterfill_project(h - alpha * direction)
                s_try = exact_score(h_try)
                if s_try < best_s - 1e-15:
                    best_s = s_try
                    best_step_h = h_try
            if best_s < c - 1e-15:
                h = best_step_h
                if best_s < best_c - 1e-14:
                    best_c = best_s
                    best_h = h.copy()
                continue

        # Gradient step
        grad, _ = lse_smoothed_grad(h, 0.01, active)
        grad -= grad.mean()
        h_new, s_new = backtracking_line_search(h, grad, exact_score)
        if s_new < c - 1e-15:
            h = h_new
            if s_new < best_c - 1e-14:
                best_c = s_new
                best_h = h.copy()
            no_improve = 0
            continue

        radius *= 0.75
        if radius < 0.001:
            break

    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s - (time.perf_counter() - t_start)

    # Load parent
    parent = None
    try:
        parent = np.array(initial_h_values, dtype=np.float64)
    except (NameError, Exception):
        pass

    if parent is None or len(parent) < 4 or not np.isfinite(parent).all():
        parent = np.full(70, 0.5)

    parent = waterfill_project(parent)
    n_p = len(parent)
    c_p = exact_score(parent)
    best_h = parent.copy()
    best_c = c_p

    # Phase 1: Annealed descent at parent dimension
    if time_left() > 150:
        for tau_i in [0.08, 0.04, 0.02]:
            if time_left() < 40:
                break
            h_opt, c_opt = annealed_descent(best_h, t_start, budget_s,
                                            tau_init=tau_i, max_iters=300,
                                            rng=rng)
            if c_opt < best_c - 1e-14:
                best_h = h_opt
                best_c = c_opt

    # Phase 2: Active set optimization
    if time_left() > 100:
        h_opt, c_opt = active_set_optimize(best_h, t_start, budget_s, max_iters=200)
        if c_opt < best_c - 1e-14:
            best_h = h_opt
            best_c = c_opt

    # Phase 3: SLSQP epigraph
    if time_left() > 80:
        for tk in [35, 20, 15, 10]:
            if time_left() < 30:
                break
            h_sol, c_sol = slsqp_epigraph(best_h, t_start, budget_s,
                                          top_k=tk, maxiter=600)
            if c_sol < best_c - 1e-14:
                best_h = h_sol
                best_c = c_sol

    # Phase 4: Dimension exploration
    n_b = len(best_h)
    dims = []
    for delta in [-8, -6, -4, -2, 2, 4, 6, 8, 10, 12]:
        d = n_b + delta
        if 24 <= d <= 200 and d % 2 == 0 and d != n_b:
            dims.append(d)
    for special in [n_b // 2 + 1, n_b * 3 // 2]:
        if 24 <= special <= 200 and special % 2 == 0 and special != n_b:
            dims.append(special)
    dims = sorted(set(dims), key=lambda d: abs(d - n_b))

    for new_n in dims[:8]:
        if time_left() < 70:
            break
        h_interp = interpolate_h(best_h, new_n)
        # Quick annealed descent
        h_opt, c_opt = annealed_descent(h_interp, t_start, budget_s,
                                        tau_init=0.05, tau_min=0.003,
                                        max_iters=200, rng=rng)
        # Active set
        if time_left() > 40:
            h_opt2, c_opt2 = active_set_optimize(h_opt, t_start, budget_s,
                                                 max_iters=100)
            if c_opt2 < c_opt:
                h_opt, c_opt = h_opt2, c_opt2
        if c_opt < best_c - 1e-14:
            best_h = h_opt
            best_c = c_opt
            # SLSQP polish
            if time_left() > 30:
                h_sol, c_sol = slsqp_epigraph(best_h, t_start, budget_s,
                                              top_k=20, maxiter=400)
                if c_sol < best_c - 1e-14:
                    best_h = h_sol
                    best_c = c_sol

    # Phase 5: Multi-start at best dimension
    n_b = len(best_h)
    if time_left() > 120:
        starts = make_starts(n_b, rng)
        for s in starts:
            if time_left() < 35:
                break
            h_blend = 0.6 * best_h + 0.4 * s
            h_blend = waterfill_project(h_blend)
            c_bl = exact_score(h_blend)
            if c_bl < best_c + 0.015:
                h_opt, c_opt = annealed_descent(h_blend, t_start, budget_s,
                                                tau_init=0.03, max_iters=150,
                                                rng=rng)
                if c_opt < best_c - 1e-14:
                    best_h = h_opt
                    best_c = c_opt
                    if time_left() > 30:
                        h_sol, c_sol = slsqp_epigraph(best_h, t_start, budget_s,
                                                      top_k=15, maxiter=300)
                        if c_sol < best_c - 1e-14:
                            best_h = h_sol
                            best_c = c_sol

    # Phase 6: Final SLSQP with full constraints
    if time_left() > 50:
        h_sol, c_sol = slsqp_epigraph(best_h, t_start, budget_s,
                                      top_k=None, maxiter=1000)
        if c_sol < best_c - 1e-14:
            best_h = h_sol
            best_c = c_sol

    # Phase 7: Final micro active-set polish
    if time_left() > 25:
        h_opt, c_opt = active_set_optimize(best_h, t_start, budget_s, max_iters=100)
        if c_opt < best_c - 1e-14:
            best_h = h_opt
            best_c = c_opt

    # Phase 8: Tiny SLSQP
    if time_left() > 20:
        h_sol, c_sol = slsqp_epigraph(best_h, t_start, budget_s,
                                      top_k=8, maxiter=500)
        if c_sol < best_c - 1e-14:
            best_h = h_sol
            best_c = c_sol

    # Final
    best_h = waterfill_project(best_h)
    final_c = exact_score(best_h)

    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            final_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 2:
            final_c = float(eval_result[0])
    except Exception:
        pass

    return best_h, final_c, len(best_h)
