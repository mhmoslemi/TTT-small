    if off == 0:

def solve_soft_minmax_lp(h, dx, active_idx, corrs, alpha, rho_q, rho_s):
    """Solve soft minmax LP to find a descent direction."""
    n = len(h)
    M = len(active_idx)
    # Variables: delta (n), s (M), q (1) -> total n + M + 1
    m_vars = n + M + 1
    
    # Objective: minimize rho_q * q + rho_s * sum(s_j)
    c_obj = np.zeros(m_vars)
    c_obj[n:n+M] = rho_s
    c_obj[-1] = rho_q
    
    # Constraints: r_j + dx * grad_j . delta + s_j <= q
    # Rearranged: dx * grad_j . delta + s_j - q <= -r_j
    A_ub = np.zeros((M, m_vars))
    b_ub = np.zeros(M)
    
    for j, k in enumerate(active_idx):
        g = grad_lag(h, k, n)
        A_ub[j, :n] = dx * g
        A_ub[j, n+j] = 1.0  # s_j
        A_ub[j, -1] = -1.0  # -q
        b_ub[j] = -corrs[k]
    
    # Equality: sum(delta) = 0
    A_eq = np.zeros((1, m_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds: -h_i*alpha <= delta_i <= (1-h_i)*alpha, s_j >= 0, q free
    bounds = [(-h[i]*alpha, (1.0-h[i])*alpha) for i in range(n)]
    bounds.extend([(0.0, None)] * M)
    bounds.append((None, None))
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.status != 0:
            return None
        return res.x[:n]
    except Exception:
        return None

        return grad

    grad[idx] += (1.0 - h[idx + off])
    grad[idx + off] += (-h[idx])
    return grad

        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.status != 0:
            return None
        return res.x[:n]
    except Exception:
        return None


def solve_linear_sum_lp(h, dx, active_idx, alpha):
    n = len(h)
    c_obj = np.zeros(n)
    for k in active_idx:
        g = grad_lag(h, k, n)
        c_obj += dx * g
    A_eq = np.zeros((1, n))
    A_eq[0, :] = 1.0
    b_eq = np.zeros(1)
    bounds = [(-h[i] * alpha, (1.0 - h[i]) * alpha) for i in range(n)]
    try:
        res = linprog(c_obj, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.status != 0:
            return None
        return res.x
    except Exception:
        return None


def line_search_accept(h_cur, c5_cur, delta, dx, target_sum):
    scales = [1.0, 0.8, 0.6, 0.5, 0.4, 0.3, 0.25, 0.2, 0.15, 0.1, 0.07, 0.04, 0.02]
    best_h = None
    best_c5 = c5_cur
    for t in scales:
        h_new = project_feasible(h_cur + t * delta, target_sum)
        c5_new = c5_of_h(h_new, dx)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new
    return best_h, best_c5


def lp_polish(h, dx, target_sum, n_iters, alpha_start, budget_s, t0,
              margin_frac, rho_q, rho_s):
    n = len(h)
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    alpha = alpha_start
    fail = 0
    for _ in range(n_iters):
        if budget_s - (time.time() - t0) < 10.0:
            break
        corrs = corr_of_h(best_h)
        active_idx = compute_active_set(corrs, n, margin_frac, 30, 100)
        delta = solve_soft_minmax_lp(best_h, dx, active_idx, corrs,
                                     alpha, rho_q, rho_s)
        if delta is None:
            delta = solve_linear_sum_lp(best_h, dx, active_idx, alpha)
        if delta is None:
            alpha *= 0.7
            fail += 1
            if fail > 25:
                break
            continue
        h_new, c5_new = line_search_accept(best_h, best_c5, delta, dx, target_sum)
        if h_new is not None:
            best_h = h_new
            best_c5 = c5_new
            alpha = min(0.5, alpha * 1.08)
            fail = 0
        else:
            alpha *= 0.7
            fail += 1
            if fail > 25:
                break
    return best_h, best_c5


def pair_refine(h, dx, target_sum, n_moves, rng, budget_s, t0):
    n = len(h)
    best_h = h.copy()
    best_c5 = c5_of_h(best_h, dx)
    for _ in range(n_moves):
        if budget_s - (time.time() - t0) < 5.0:
            break
        if rng.random() < 0.4:
            sorted_idx = np.argsort(best_h)
            p = sorted_idx[rng.randint(0, n // 2)]
            q = sorted_idx[rng.randint(n // 2, n)]
        else:
            p = rng.randint(n)
            q = rng.randint(n)
        if p == q:
            continue
        a = best_h[p]
        b = best_h[q]
        L = max(-a, b - 1.0)
        U = min(1.0 - a, b)
        if U - L < 1e-10:
            continue
        t_samples = np.linspace(L, U, 17)
        best_t = 0.0
        best_val = best_c5
        for t in t_samples:
            if abs(t) < 1e-12:
                continue
            h_trial = best_h.copy()
            h_trial[p] = a + t
            h_trial[q] = b - t
            h_trial = project_feasible(h_trial, target_sum)
            val = c5_of_h(h_trial, dx)
            if val < best_val - 1e-12:
                best_val = val
                best_t = t
        if best_t == 0.0:
            continue
        span = (U - L) / 8.0
        t_lo = max(L, best_t - span)
        t_hi = min(U, best_t + span)
        t_fine = np.linspace(t_lo, t_hi, 15)
        for t in t_fine:
            if abs(t) < 1e-12:
                continue
            h_trial = best_h.copy()
            h_trial[p] = a + t
            h_trial[q] = b - t
            h_trial = project_feasible(h_trial, target_sum)
            val = c5_of_h(h_trial, dx)
            if val < best_val - 1e-12:
                best_val = val
                best_t = t
        if best_t != 0.0 and best_val < best_c5 - 1e-12:
            best_h[p] = a + best_t
            best_h[q] = b - best_t
            best_h = project_feasible(best_h, target_sum)
            best_c5 = c5_of_h(best_h, dx)
    return best_h, best_c5


def triple_refine(h, dx, target_sum, n_moves, rng, budget_s, t0):
    n = len(h)
    best_h = h.copy()
    best_c5 = c5_of_h(best_h, dx)
    for _ in range(n_moves):
        if budget_s - (time.time() - t0) < 5.0:
            break
        idx = rng.choice(n, size=3, replace=False)
        p, q, r = idx[0], idx[1], idx[2]
        a, b, c = best_h[p], best_h[q], best_h[r]
        # Move mass from one to another: h[p]+=t, h[q]-=t (standard pair)
        # or h[p]+=t, h[q]-=t/s, h[r]+=t*(1-1/s)
        # Simplify: just do pair moves within the triple
        pairs = [(p, q), (p, r), (q, r)]
        for px, qx in pairs:
            av, bv = best_h[px], best_h[qx]
            Lo = max(-av, bv - 1.0)
            Hi = min(1.0 - av, bv)
            if Hi - Lo < 1e-10:
                continue
            t_best = 0.0
            v_best = best_c5
            ts = np.linspace(Lo, Hi, 11)
            for t in ts:
                if abs(t) < 1e-12:
                    continue
                ht = best_h.copy()
                ht[px] = av + t
                ht[qx] = bv - t
                ht = project_feasible(ht, target_sum)
                v = c5_of_h(ht, dx)
                if v < v_best - 1e-12:
                    v_best = v
                    t_best = t
            if t_best != 0.0 and v_best < best_c5 - 1e-12:
                best_h[px] = av + t_best
                best_h[qx] = bv - t_best
                best_h = project_feasible(best_h, target_sum)
                best_c5 = c5_of_h(best_h, dx)
    return best_h, best_c5


def block_flip(h, dx, target_sum, rng, n_flips, budget_s, t0):
    n = len(h)
    best_h = h.copy()
    best_c5 = c5_of_h(best_h, dx)
    for _ in range(n_flips):
        if budget_s - (time.time() - t0) < 5.0:
            break
        length = rng.randint(3, max(4, n // 4))
        start = rng.randint(0, n - length + 1)
        h_trial = best_h.copy()
        h_trial[start:start + length] = 1.0 - h_trial[start:start + length]
        h_trial = project_feasible(h_trial, target_sum)
        v = c5_of_h(h_trial, dx)
        if v < best_c5 - 1e-6:
            best_h = h_trial
            best_c5 = v
    return best_h, best_c5


def perturb_reopt(h, dx, target_sum, rng, n_rounds, budget_s, t0):
    n = len(h)
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    for r in range(n_rounds):
        if budget_s - (time.time() - t0) < 25.0:
            break
        scale = max(0.03, 0.18 / (r + 1))
        pert = rng.standard_normal(n)
        pert -= pert.mean()
        h_p = project_feasible(best_h + scale * pert, target_sum)
        h_opt, c5_opt = lp_polish(h_p, dx, target_sum, 60, 0.2,
                                  budget_s, t0, 0.15, 0.3, 5.0)
        c5_opt2 = c5_opt
        if budget_s - (time.time() - t0) > 15.0:
            h_opt2, c5_opt2 = pair_refine(h_opt, dx, target_sum, 80, rng,
                                          budget_s, t0)
        if c5_opt2 < best_c5:
            best_c5 = c5_opt2
            best_h = h_opt if c5_opt2 == c5_opt else h_opt2
    return best_h, best_c5


def transfer_grid(h_src, n_target):
    n_src = len(h_src)
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_tgt = (np.arange(n_target) + 0.5) * (2.0 / n_target)
    return np.interp(x_tgt, x_src, h_src)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = np.full(126, 0.5)

    n = len(h0)
    dx = 2.0 / n
    target_sum = n / 2.0
    h0 = project_feasible(h0, target_sum)
    best_h = h0.copy()
    best_c5 = c5_of_h(h0, dx)

    # Phase 1: Multi-config LP polish
    configs = [
        (0.3, 0.3, 5.0, 0.15),
        (0.2, 0.25, 8.0, 0.12),
        (0.15, 0.35, 5.0, 0.10),
        (0.1, 0.3, 10.0, 0.08),
        (0.08, 0.2, 6.0, 0.06),
    ]
    for (a_s, rq, rs, mf) in configs:
        if budget_s - (time.time() - t0) < 20.0:
            break
        h_new, c5_new = lp_polish(best_h, dx, target_sum, 100, a_s,
                                  budget_s, t0, mf, rq, rs)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

    # Phase 2: Pair and triple refinement
    if budget_s - (time.time() - t0) > 30.0:
        h_new, c5_new = pair_refine(best_h, dx, target_sum, 400, rng,
                                    budget_s, t0)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

    if budget_s - (time.time() - t0) > 30.0:
        h_new, c5_new = triple_refine(best_h, dx, target_sum, 200, rng,
                                      budget_s, t0)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

    # Phase 3: More LP after pair moves
    if budget_s - (time.time() - t0) > 30.0:
        h_new, c5_new = lp_polish(best_h, dx, target_sum, 120, 0.12,
                                  budget_s, t0, 0.10, 0.3, 6.0)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

    # Phase 4: More pairs
    if budget_s - (time.time() - t0) > 30.0:
        h_new, c5_new = pair_refine(best_h, dx, target_sum, 400, rng,
                                    budget_s, t0)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

    # Phase 5: Block flips + perturbations
    if budget_s - (time.time() - t0) > 40.0:
        h_new, c5_new = block_flip(best_h, dx, target_sum, rng, 50,
                                   budget_s, t0)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

    if budget_s - (time.time() - t0) > 50.0:
        h_new, c5_new = perturb_reopt(best_h, dx, target_sum, rng, 10,
                                      budget_s, t0)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

    # Phase 6: Grid size exploration
    n_base = len(best_h)
    n_candidates = sorted(set([n_base + d for d in range(-20, 21, 2)
                                if 64 <= n_base + d <= 240]))
    for n_try in n_candidates:
        if budget_s - (time.time() - t0) < 25.0:
            break
        if n_try == n_base:
            continue
        dx_t = 2.0 / n_try
        ts_t = n_try / 2.0
        h_interp = project_feasible(transfer_grid(best_h, n_try), ts_t)
        h_opt, c5_opt = lp_polish(h_interp, dx_t, ts_t, 60, 0.2,
                                  budget_s, t0, 0.12, 0.3, 5.0)
        if c5_opt < best_c5 and budget_s - (time.time() - t0) > 15.0:
            h_opt2, c5_opt2 = pair_refine(h_opt, dx_t, ts_t, 100, rng,
                                          budget_s, t0)
            if c5_opt2 < best_c5:
                best_c5 = c5_opt2
                best_h = h_opt2
                n_base = n_try
            elif c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
                n_base = n_try

    # Phase 7: Final intensive polish on best grid
    n_f = len(best_h)
    dx_f = 2.0 / n_f
    ts_f = n_f / 2.0

    if budget_s - (time.time() - t0) > 20.0:
        h_new, c5_new = lp_polish(best_h, dx_f, ts_f, 150, 0.1,
                                  budget_s, t0, 0.08, 0.3, 5.0)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

    if budget_s - (time.time() - t0) > 15.0:
        h_new, c5_new = pair_refine(best_h, dx_f, ts_f, 300, rng,
                                    budget_s, t0)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

    if budget_s - (time.time() - t0) > 12.0:
        h_new, c5_new = lp_polish(best_h, dx_f, ts_f, 100, 0.06,
                                  budget_s, t0, 0.05, 0.4, 12.0)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

    if budget_s - (time.time() - t0) > 8.0:
        h_new, c5_new = pair_refine(best_h, dx_f, ts_f, 200, rng,
                                    budget_s, t0)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

    # Phase 8: Ultra-fine final LP
    if budget_s - (time.time() - t0) > 8.0:
        h_new, c5_new = lp_polish(best_h, dx_f, ts_f, 80, 0.04,
                                  budget_s, t0, 0.04, 0.5, 15.0)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

    # Final validation
    n_final = len(best_h)
    dx_final = 2.0 / n_final
    ts_final = n_final / 2.0
    best_h = project_feasible(best_h, ts_final)
    best_c5 = c5_of_h(best_h, dx_final)

    return best_h, best_c5, n_final
