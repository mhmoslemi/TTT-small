def repair_feasible(vec):
    """Project onto {0<=h<=1, sum(h)=n/2}"""
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:

    for _ in range(100):
        if abs(diff) < 1e-14:
            break
        if diff > 0:

            mask = room > 1e-12
            if not mask.any():

            add = min(diff / mask.sum(), room[mask])
            h[mask] += add

def smooth_grad_descent(h0, budget_fn=None, max_iters=5000):
    n = len(h0)
    dx = 2.0 / n
    M = 2 * n - 1
    h = h0.copy()
    y = h0.copy()
    best_h = h0.copy()
    best_c = exact_score(h0)
    eps = 0.01
    t = 0
    
    for t in range(max_iters):
        if budget_fn is not None and budget_fn() < 20:
            break
        alpha = 2.0 / (t + 2)
        y = (1 - alpha) * h + alpha * y
        y = repair_feasible(y)
        
        profile = compute_profile(y)
        shifted = profile / eps
        shifted -= shifted.max()
        weights = np.exp(shifted)
        weights /= weights.sum()
        
        # Compute gradient: sum_d w_d * [dx*(1-h[i+d]) - dx*h[i-d]]
        grad = np.zeros(n)
        active_d = np.where(weights > 1e-8)[0]
        for di in active_d:
            d = di - (n - 1)
            w = weights[di]
            for i in range(n):
                if 0 <= i+d < n:
                    grad[i] += w * dx * (1.0 - y[i+d])
                if 0 <= i-d < n:
                    grad[i] -= w * dx * y[i-d]
        
        # Subgradient step with diminishing step size
        eta = 0.1 / (np.sqrt(t + 1) + 1)
        h_new = repair_feasible(y - eta * grad)
        c_new = exact_score(h_new)
        
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new
    
    return best_h, best_c

def interp_to(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    fx = np.concatenate([[0.0], xs, [2.0]])
    fh = np.concatenate([[h[0]], h, [h[-1]]])
    hn = np.interp(xt, fx, fh)
    return repair_feasible(np.clip(hn, 0, 1))

    n_parent = len(parent)
    best_h = parent.copy()
    best_c = exact_score(parent)
    
    # Phase 1: Binary local search from parent rounding
    if tl() > 60:
        h_bin = round_to_binary(best_h)
        c_bin = exact_score(h_bin)
        if c_bin < best_c + 0.05:
            h_b, c_b = binary_local_search(h_bin, max_iters=3000, budget_fn=tl)
            if c_b < best_c - 1e-14:
                best_h, best_c = h_b, c_b
            # Near-binary refinement
            if tl() > 40:
                h_nb, c_nb = near_binary_refine(h_b, n_free=10, budget_fn=tl)
                if c_nb < best_c - 1e-14:

    if tl() > 80:
        for attempt in range(30):
            if tl() < 20:
                break
            # Start from random binary
            n_try = n_parent
            perm = rng.permutation(n_try)
            h_rand = np.zeros(n_try)
            h_rand[perm[:n_try//2]] = 1.0
            h_r, c_r = binary_local_search(h_rand, max_iters=2000, budget_fn=tl)
            if c_r < best_c - 1e-14:
                best_h, best_c = h_r, c_r
                # Refine
                if tl() > 30:
                    h_nb, c_nb = near_binary_refine(h_r, n_free=8, budget_fn=tl)
                    if c_nb < best_c - 1e-14:
                        best_h, best_c = h_nb, c_nb
            elif c_r < best_c + 0.02:
                # Close, try harder
                h_r2, c_r2 = binary_local_search(h_r, max_iters=3000, budget_fn=tl)

    if tl() > 60:
        for nd in [n_parent - 2, n_parent + 2, n_parent - 4, n_parent + 4,
                   n_parent - 6, n_parent + 6]:
            if tl() < 15 or nd < 30 or nd > 120 or nd % 2 != 0:
                continue
            # Interpolate current best to new dimension
            h_i = interp_to(best_h, nd)

            for _ in range(5):
                if tl() < 10:
                    break
                h_b, c_b = binary_local_search(h_bin_i, max_iters=2000, budget_fn=tl)
                if c_b < best_c - 1e-14:
                    best_h, best_c = h_b, c_b
                    h_bin_i = h_b
                else:
                    # Random restart at this dimension
                    perm = rng.permutation(nd)
                    h_rand = np.zeros(nd)
                    h_rand[perm[:nd//2]] = 1.0
                    h_b, c_b = binary_local_search(h_rand, max_iters=1500, budget_fn=tl)

        if c_s < best_c - 1e-14:

            if c_nb < best_c - 1e-14:

    # Phase 6: SLSQP fine-tune
    if tl() > 40:
        n = len(best_h)
        dx = 2.0 / n
        M = 2 * n - 1
        
        def slsqp_obj(x):
            h_t = np.clip(x, 0, 1)
            prof = np.correlate(h_t, 1 - h_t, mode="full") * dx
            return np.max(prof)
        
        def slsqp_jac(x):

            prof = np.correlate(h_t, 1 - h_t, mode="full") * dx
            cm = np.max(prof)

            if len(active) == 0:

            grad = np.zeros(n)
            for di in active:

                w = 1.0 / len(active)
                for i in range(n):
                    if 0 <= i+d < n:
                        grad[i] += w * dx * (1.0 - h_t[i+d])
                    if 0 <= i-d < n:
                        grad[i] -= w * dx * h_t[i-d]
            return grad

                     'jac': lambda x: np.ones(n)}]

            if res.x is not None:
                hs = repair_feasible(res.x)
                cs = exact_score(hs)
                if cs < best_c - 1e-14:
                    best_h, best_c = hs, cs

            pass

    if tl() > 30:

        if c_bf < best_c + 0.01:

        h_cd = best_h.copy()
        c_cd = exact_score(h_cd)
        for it in range(200):
            if tl() < 5:

            for i in range(n):

                # Try moving h[i] in small steps (compensate with another coordinate)
                for delta in [0.01, -0.01, 0.005, -0.005, 0.02, -0.02]:
                    new_val = old + delta
                    if new_val < 0 or new_val > 1:
