    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):

            add = min(remaining, capacity[idx] + 1e-15)
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx] + 1e-15)
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmin(h))
        h[idx] = np.clip(h[idx] + diff2, 0.0, 1.0)
    return h

def ray_search(h, v, n_grid=200):
    """Find alpha in feasible range minimizing max profile along ray h + alpha*v."""
    n = len(h)
    lo, hi = ray_feasible_range(h, v)
    if hi - lo < 1e-12:
        return None, None
    A, B, C = ray_profile_coeffs(h, v)
    
    # Grid search
    alphas = np.linspace(lo, hi, n_grid)
    # Compute max profile at each alpha
    # profile[s] = A[s] + alpha*B[s] + alpha^2*C[s]
    # For efficiency, compute in batches
    scores = np.zeros(n_grid)
    for k, a in enumerate(alphas):
        prof = A + a * B + a * a * C
        scores[k] = np.max(prof)
    
    k_best = np.argmin(scores)
    best_alpha = alphas[k_best]
    best_score = scores[k_best]
    
    # Local refinement
    if 0 < k_best < n_grid - 1:
        a_lo = alphas[k_best - 1]
        a_hi = alphas[k_best + 1]
        # finer grid
        fine_alphas = np.linspace(a_lo, a_hi, 100)
        fine_scores = np.zeros(100)
        for k, a in enumerate(fine_alphas):
            prof = A + a * B + a * a * C
            fine_scores[k] = np.max(prof)
        kf = np.argmin(fine_scores)
        best_alpha = fine_alphas[kf]
        best_score = fine_scores[kf]
    
    # Further refinement
    if 0 < kf < 99:
        a_lo2 = fine_alphas[kf-1]
        a_hi2 = fine_alphas[kf+1]
        fine2 = np.linspace(a_lo2, a_hi2, 50)
        s2 = np.max(A[None,:] + fine2[:,None]*B[None,:] + fine2[:,None]**2*C[None,:], axis=1)
        kf2 = np.argmin(s2)
        best_alpha = fine2[kf2]
        best_score = s2[kf2]
    
    h_new = h + best_alpha * v
    h_new = repair_feasible(h_new)
    actual = exact_score(h_new)
    return h_new, actual

    """Compute gradient of sum_s w[s]*R(s) w.r.t. h.
    w is array of length 2n-1, indexed by lag offset from center."""
    n = len(h)
    M = 2*n - 1
    center = n - 1
    g = np.zeros(n)
    for s_offset in range(M):
        s = s_offset - center
        if abs(w[s_offset]) < 1e-15:

            g[i] += w[s_offset] * (1 - h[i + s])
            g[i + s] -= w[s_offset] * h[i]
        else:
            i = np.arange(-s, n)
            g[i] += w[s_offset] * (1 - h[i + s])
            g[i + s] -= w[s_offset] * h[i]
    return g

import numpy as np
from scipy.optimize import minimize
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def compute_profile(h):
    return np.correlate(h, 1.0 - h, mode="full")

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        cap = 1.0 - h
        order = np.argsort(-cap)
        rem = diff
        for i in range(n):
            idx = order[i]
            add = min(rem, cap[idx] + 1e-15)
            if add > 0:
                h[idx] += add
            rem -= add
            if rem <= 1e-12:
                break
    else:
        order = np.argsort(h)
        rem = -diff
        for i in range(n):
            idx = order[i]
            sub = min(rem, h[idx] + 1e-15)
            if sub > 0:
                h[idx] -= sub
            rem -= sub
            if rem <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        idx = int(np.argmax(1.0 - h)) if diff2 > 0 else int(np.argmin(h))
        h[idx] = np.clip(h[idx] + diff2, 0.0, 1.0)
    return h

def ray_coeffs(h, v):
    n = len(h)
    M = 2 * n - 1
    center = n - 1
    A = compute_profile(h)
    B = np.zeros(M)
    C = np.zeros(M)
    for s in range(-(n-1), n):
        if s >= 0:
            i = np.arange(0, n - s)
        else:
            i = np.arange(-s, n)
        B[center + s] = np.sum(v[i] - h[i] * v[i + s] - h[i + s] * v[i])
        C[center + s] = np.sum(v[i] * v[i + s])
    return A, B, C

def feasible_range(h, v):
    nz = np.where(np.abs(v) > 1e-14)[0]
    if len(nz) == 0:
        return 0.0, 0.0
    lo = 0.0
    hi = 0.0
    for i in nz:
        if v[i] > 0:
            lo = max(lo, -h[i] / v[i])
            hi = min(hi, (1.0 - h[i]) / v[i])
        else:
            lo = max(lo, (1.0 - h[i]) / v[i])
            hi = min(hi, -h[i] / v[i])
    return lo, hi

def ray_minimax(A, B, C, lo, hi, n_coarse=200, n_fine=100):
    """Minimize max_s (A[s] + alpha*B[s] + alpha^2*C[s]) over alpha in [lo, hi]."""
    if hi - lo < 1e-12:
        return lo
    
    # Coarse grid
    alphas = np.linspace(lo, hi, n_coarse)
    # Vectorized: shape (n_coarse, M)
    vals = A[None, :] + alphas[:, None] * B[None, :] + alphas[:, None]**2 * C[None, :]
    scores = np.max(vals, axis=1)
    k = np.argmin(scores)
    
    # Fine refinement
    if k == 0:
        a_lo, a_hi = lo, alphas[1]
    elif k == n_coarse - 1:
        a_lo, a_hi = alphas[-2], hi
    else:
        a_lo, a_hi = alphas[k-1], alphas[k+1]
    
    fine = np.linspace(a_lo, a_hi, n_fine)
    vals2 = A[None, :] + fine[:, None] * B[None, :] + fine[:, None]**2 * C[None, :]
    scores2 = np.max(vals2, axis=1)
    kf = np.argmin(scores2)
    
    # Even finer
    if kf > 0 and kf < n_fine - 1:
        a_lo2, a_hi2 = fine[kf-1], fine[kf+1]
        fine2 = np.linspace(a_lo2, a_hi2, 50)
        vals3 = A[None, :] + fine2[:, None] * B[None, :] + fine2[:, None]**2 * C[None, :]
        scores3 = np.max(vals3, axis=1)
        best_a = fine2[np.argmin(scores3)]
    else:
        best_a = fine[kf]
    
    return best_a

def do_ray_search(h, v, n_coarse=200):
    """Perform ray search along direction v. Returns (h_new, score) or (None, inf)."""
    # Ensure v has zero sum
    v = v - v.mean()
    # Normalize
    vnorm = np.max(np.abs(v))
    if vnorm < 1e-14:
        return None, np.inf
    v = v / vnorm  # scale so max abs is 1
    
    lo, hi = feasible_range(h, v)
    if hi - lo < 1e-12:
        return None, np.inf
    
    A, B, C = ray_coeffs(h, v)
    best_a = ray_minimax(A, B, C, lo, hi, n_coarse)
    
    h_new = h + best_a * v
    h_new = repair_feasible(h_new)
    score = exact_score(h_new)
    return h_new, score

def compute_grad(h, w):
    n = len(h)
    M = 2 * n - 1
    center = n - 1
    g = np.zeros(n)
    for idx in range(M):
        s = idx - center
        ws = w[idx]
        if abs(ws) < 1e-15:
            continue
        if s >= 0:
            k1 = np.arange(0, n - s)
            g[k1] += ws * (1.0 - h[k1 + s])
            k2 = np.arange(s, n)
            g[k2] -= ws * h[k2 - s]
        else:
            k1 = np.arange(-s, n)
            g[k1] += ws * (1.0 - h[k1 + s])
            k2 = np.arange(0, n + s)
            g[k2] -= ws * h[k2 - s]
    return g

def make_gradient_dir(h, profile, n_top=8):
    """Create a descent direction based on active lags."""
    n = len(h)
    M = 2 * n - 1
    c_max = np.max(profile)
    # Weight lags by how close they are to max
    threshold = c_max - 0.01
    active = np.where(profile >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(profile)])
    if len(active) > n_top:
        # Keep top n_top
        order = np.argsort(profile[active])[-n_top:]
        active = active[order]
    
    # Equal weights for active lags
    w = np.zeros(M)
    w[active] = 1.0 / len(active)
    
    g = compute_grad(h, w)
    v = -(g - g.mean())  # negative gradient, zero-sum
    return v

def make_random_dir(n, rng):
    v = rng.randn(n)
    v -= v.mean()
    vn = np.max(np.abs(v))
    if vn < 1e-14:
        return None
    return v / vn

def make_smooth_dir(n, rng, k_range=(1, 8)):
    k = rng.randint(k_range[0], k_range[1] + 1)
    phase = rng.uniform(0, 2 * np.pi)
    v = np.sin(2 * np.pi * k * np.arange(n) / n + phase)
    v -= v.mean()
    vn = np.max(np.abs(v))
    if vn < 1e-14:
        return None
    return v / vn

def make_swap_dir(n, rng, h):
    """Direction that swaps mass between a high and low coordinate."""
    sorted_idx = np.argsort(h)
    # Pick from top and bottom halves
    i = rng.randint(n // 2, n)  # high value
    j = rng.randint(0, n // 2)  # low value
    v = np.zeros(n)
    v[i] = -1.0
    v[j] = 1.0
    return v

def make_block_dir(n, rng):
    """Block transfer direction."""
    n_blocks = rng.randint(2, 5)
    block_size = n // n_blocks
    v = np.zeros(n)
    # Transfer from one block to another
    b1 = rng.randint(0, n_blocks)
    b2 = rng.randint(0, n_blocks - 1)
    if b2 == b1:
        b2 = (b1 + 1) % n_blocks
    start1 = b1 * block_size
    start2 = b2 * block_size
    v[start1:start1+block_size] = -1.0
    v[start2:start2+block_size] = 1.0
    v -= v.mean()
    vn = np.max(np.abs(v))
    if vn < 1e-14:
        return None
    return v / vn

def interp_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    fx = np.concatenate([[0.0], xs, [2.0]])
    fh = np.concatenate([[h[0]], h, [h[-1]]])
    hn = np.interp(xt, fx, fh)
    return repair_feasible(np.clip(hn, 0, 1))

def block_average(h, factor):
    """Average blocks of 'factor' consecutive values. n must be divisible by factor."""
    n = len(h)
    if n % factor != 0:
        return None
    new_n = n // factor
    h_new = h.reshape(new_n, factor).mean(axis=1)
    return repair_feasible(np.clip(h_new, 0, 1))

import numpy as np
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def compute_profile(h):
    return np.correlate(h, 1.0 - h, mode="full")

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        cap = 1.0 - h
        order = np.argsort(-cap)
        rem = diff
        for i in range(n):
            idx = order[i]
            add = min(rem, cap[idx] + 1e-15)
            if add > 0:
                h[idx] += add
            rem -= add
            if rem <= 1e-12:
                break
    else:
        order = np.argsort(h)
        rem = -diff
        for i in range(n):
            idx = order[i]
            sub = min(rem, h[idx] + 1e-15)
            if sub > 0:
                h[idx] -= sub
            rem -= sub
            if rem <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        idx = int(np.argmax(1.0 - h)) if diff2 > 0 else int(np.argmin(h))
        h[idx] = np.clip(h[idx] + diff2, 0.0, 1.0)
    return h

def ray_coeffs(h, v):
    n = len(h)
    M = 2 * n - 1
    center = n - 1
    A = compute_profile(h)
    B = np.zeros(M)
    C = np.zeros(M)
    for s in range(-(n-1), n):
        if s >= 0:
            i = np.arange(0, n - s)
        else:
            i = np.arange(-s, n)
        B[center + s] = np.sum(v[i] - h[i] * v[i + s] - h[i + s] * v[i])
        C[center + s] = np.sum(v[i] * v[i + s])
    return A, B, C

def feasible_range(h, v):
    nz = np.where(np.abs(v) > 1e-14)[0]
    if len(nz) == 0:
        return 0.0, 0.0
    lo = 0.0
    hi = 0.0
    for i in nz:
        if v[i] > 0:
            lo = max(lo, -h[i] / v[i])
            hi = min(hi, (1.0 - h[i]) / v[i])
        else:
            lo = max(lo, (1.0 - h[i]) / v[i])
            hi = min(hi, -h[i] / v[i])
    return lo, hi

def ray_minimax(A, B, C, lo, hi):
    if hi - lo < 1e-12:
        return lo
    # Coarse grid
    alphas = np.linspace(lo, hi, 200)
    vals = A[None, :] + alphas[:, None] * B[None, :] + (alphas[:, None]) ** 2 * C[None, :]
    scores = np.max(vals, axis=1)
    k = int(np.argmin(scores))
    # Fine refinement
    if k == 0:
        a_lo, a_hi = lo, min(alphas[1], lo + (hi-lo)*0.05)
    elif k == 199:
        a_lo, a_hi = max(alphas[198], hi - (hi-lo)*0.05), hi
    else:
        a_lo, a_hi = alphas[k-1], alphas[k+1]
    fine = np.linspace(a_lo, a_hi, 80)
    vals2 = A[None, :] + fine[:, None] * B[None, :] + (fine[:, None]) ** 2 * C[None, :]
    scores2 = np.max(vals2, axis=1)
    kf = int(np.argmin(scores2))
    # Ultra fine
    if kf > 0 and kf < 79:
        a_lo2, a_hi2 = fine[kf-1], fine[kf+1]
        ultra = np.linspace(a_lo2, a_hi2, 40)
        vals3 = A[None, :] + ultra[:, None] * B[None, :] + (ultra[:, None]) ** 2 * C[None, :]
        scores3 = np.max(vals3, axis=1)
        return float(ultra[int(np.argmin(scores3))])
    return float(fine[kf])

def do_ray_search(h, v, n_coarse=200):
    v = np.asarray(v, dtype=np.float64) - np.mean(np.asarray(v, dtype=np.float64))
    vnorm = np.max(np.abs(v))
    if vnorm < 1e-14:
        return None, np.inf
    v = v / vnorm
    lo, hi = feasible_range(h, v)
    if hi - lo < 1e-12:
        return None, np.inf
    A, B, C = ray_coeffs(h, v)
    best_a = ray_minimax(A, B, C, lo, hi)
    h_new = h + best_a * v
    h_new = repair_feasible(h_new)
    score = exact_score(h_new)
    return h_new, score

def compute_grad(h, w):
    n = len(h)
    M = 2 * n - 1
    center = n - 1
    g = np.zeros(n)
    for idx in range(M):
        s = idx - center
        ws = w[idx]
        if abs(ws) < 1e-15:
            continue
        if s >= 0:
            k1 = np.arange(0, n - s)
            g[k1] += ws * (1.0 - h[k1 + s])
            k2 = np.arange(s, n)
            g[k2] -= ws * h[k2 - s]
        else:
            k1 = np.arange(-s, n)
            g[k1] += ws * (1.0 - h[k1 + s])
            k2 = np.arange(0, n + s)
            g[k2] -= ws * h[k2 - s]
    return g

def make_gradient_dir(h, profile, n_top=8):
    n = len(h)
    M = 2 * n - 1
    c_max = np.max(profile)
    threshold = c_max - max(0.005, 0.05 * c_max)
    active = np.where(profile >= threshold)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(profile))])
    if len(active) > n_top:
        order = np.argsort(profile[active])[-n_top:]
        active = active[order]
    w = np.zeros(M)
    w[active] = 1.0 / len(active)
    g = compute_grad(h, w)
    v = -(g - g.mean())
    return v

def make_random_dir(n, rng):
    v = rng.randn(n).astype(np.float64)
    v -= v.mean()
    vn = np.max(np.abs(v))
    if vn < 1e-14:

    return v / vn

    v -= v.mean()
    vn = np.max(np.abs(v))
    if vn < 1e-14:

    v[i] = -1.0
    v[j] = 1.0

def make_block_dir(n, rng):
    n_blocks = rng.randint(2, 5)

    v = np.zeros(n)
    b1 = int(rng.randint(0, n_blocks))

def interp_to_n(h, new_n):
