def grad_max_step(h, dx, target_sum, alpha):
    """Take a step in the negative subgradient direction of the max."""
    n = len(h)
    corrs = np.correlate(h, 1.0 - h, mode='full')
    m = 2 * n - 1
    # Find active set (within tolerance of max)
    cmax = corrs.max()
    tol = 1e-4
    active = np.where(corrs >= cmax - tol)[0]
    # Average gradient of active constraints
    g = np.zeros(n)
    ooh = 1.0 - h
    for p in active:
        d = p - (n - 1)
        if d >= 0:
            g[d:] += ooh[:n-d]
            g[:n-d] -= h[d:]
        else:
            ad = -d
            g[:n-ad] += ooh[ad:]
            g[ad:] -= h[:n-ad]
    g /= len(active)
    # Project gradient to feasible direction (zero-sum)
    g -= g.mean()
    # Line search
    best_alpha = 0
    best_c = c5_of_h(h, dx)
    for a in [alpha, alpha*0.5, alpha*0.25, alpha*0.1]:
        h_new = project_feasible(h - a * g, target_sum)
        c_new = c5_of_h(h_new, dx)
        if c_new < best_c:
            best_c = c_new
            best_alpha = a
    if best_alpha > 0:
        return project_feasible(h - best_alpha * g, target_sum), best_c
    return h, best_c

import numpy as np
from scipy.optimize import minimize
import time


def project_feasible(x, target_sum):
    n = len(x)
    if n == 0:
        return x.copy()
    x = np.nan_to_num(np.asarray(x, dtype=float), nan=0.5, posinf=1.0, neginf=0.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        cnt = mask.sum()
        if cnt > 0:
            h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)


def c5_of_h(h, dx):
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)


def compute_grad_rows(h):
    n = len(h)
    m = 2 * n - 1
    ooh = 1.0 - h
    grads = np.zeros((m, n))
    for p in range(m):
        d = p - (n - 1)
        if d >= 0:
            grads[p, d:n] += ooh[:n - d]
            grads[p, :n - d] -= h[d:]
        else:
            ad = -d
            grads[p, :n - ad] += ooh[ad:]
            grads[p, ad:n] -= h[:n - ad]
    return grads


def slsqp_solve(h_init, target_sum, t_end, maxiter=150, maxfun=3000):
    n = len(h_init)
    m = 2 * n - 1
    dx = 2.0 / n
    c0 = c5_of_h(h_init, dx)

    x0 = np.zeros(n + 1)
    x0[0] = c0 + 2e-4
    x0[1:] = h_init

    def obj(x):
        return x[0]

    def ineq(x):
        h = x[1:]
        return x[0] - np.correlate(h, 1.0 - h, mode='full')

    def ineq_jac(x):
        h = x[1:]
        n = len(h)
        m = 2 * n - 1
        J = np.zeros((m, n + 1))
        J[:, 0] = 1.0
        ooh = 1.0 - h
        for p in range(m):
            d = p - (n - 1)
            if d >= 0:
                J[p, d + 1:n + 1] -= ooh[:n - d]
                J[p, 1:n - d + 1] += h[d:]
            else:
                ad = -d
                J[p, 1:n - ad + 1] -= ooh[ad:]
                J[p, ad + 1:n + 1] += h[:n - ad]
        return J

    def eq(x):
        return x[1:].sum() - target_sum

    def eq_jac(x):
        J = np.zeros(n + 1)
        J[1:] = 1.0
        return J

    bounds = [(0.0, 0.55)] + [(0.0, 1.0)] * n
    constraints = [
        {'type': 'ineq', 'fun': ineq, 'jac': ineq_jac},
        {'type': 'eq', 'fun': eq, 'jac': eq_jac},
    ]

    try:
        res = minimize(obj, x0, method='SLSQP', bounds=bounds,
                       constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-10, 'maxfun': maxfun})
    except Exception:
        return h_init, c0

    if res.x is not None and np.all(np.isfinite(res.x)):
        h_new = project_feasible(res.x[1:], target_sum)
        c_new = c5_of_h(h_new, dx)
        if c_new < c0 + 1e-12:
            return h_new, c_new
    return h_init, c0

def pair_exchange_search(h, dx, target_sum, n_pairs=200, max_rounds=5, seed=0):
    """Greedy pair-exchange local search."""
    rng = np.random.RandomState(seed)
    n = len(h)
    best_h = h.copy()
    best_c = c5_of_h(h, dx)

    for rnd in range(max_rounds):
        improved = False
        for _ in range(n_pairs):
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            # Try transfers in both directions
            for direction in [1, -1]:
                for frac in [0.1, 0.05, 0.02]:
                    amt = frac * direction * min(best_h[i], 1.0 - best_h[j]) if direction > 0 \
                        else -frac * direction * min(best_h[j], 1.0 - best_h[i])
                    if direction > 0:
                        amt = frac * min(best_h[i], 1.0 - best_h[j])
                        if amt < 1e-6:
                            break
                        h_try = best_h.copy()
                        h_try[i] -= amt
                        h_try[j] += amt
                    else:
                        amt = frac * min(best_h[j], 1.0 - best_h[i])
                        if amt < 1e-6:
                            break
                        h_try = best_h.copy()
                        h_try[j] -= amt
                        h_try[i] += amt
                    c_try = c5_of_h(h_try, dx)
                    if c_try < best_c - 1e-12:
                        best_h = h_try
                        best_c = c_try
                        improved = True
                        break
        if not improved:
            break
    return best_h, best_c


def perturb_and_refine(h_best, c5_best, n, dx, target_sum, t0, deadline, rng, n_rounds=8):
    """Perturb the best solution and refine with SLSQP."""
    for r in range(n_rounds):
        if time.time() > deadline:
            break
        sigma = 0.01 + 0.04 * rng.random()
        h_pert = project_feasible(h_best + sigma * rng.randn(n), target_sum)
        t_end = min(deadline, time.time() + (deadline - time.time()) / (n_rounds - r))
        h_new, c_new = slsqp_solve(h_pert, target_sum, t_end, maxiter=80, maxfun=2000)
        if c_new < c5_best:
            h_best = h_new
            c5_best = c_new
        if time.time() > deadline:
            break
    return h_best, c5_best


def try_resolution(n_try, h_best, c5_best, dx_best, t0, deadline, seed):
    """Try a different number of points."""
    n_cur = len(h_best)
    dx_try = 2.0 / n_try
    ts_try = n_try / 2.0
    # Interpolate
    x_orig = (np.arange(n_cur) + 0.5) / n_cur
    x_new = (np.arange(n_try) + 0.5) / n_try
    if n_try > n_cur:
        h_interp = np.interp(x_new, x_orig, h_best)
    else:
        h_interp = np.interp(x_new, x_orig, h_best)
    h_interp = project_feasible(h_interp, ts_try)
    c_init = c5_of_h(h_interp, dx_try)

    if time.time() < deadline - 15:
        t_end = min(deadline - 5, time.time() + max(10, (deadline - time.time()) * 0.6))
        h_opt, c_opt = slsqp_solve(h_interp, ts_try, t_end, maxiter=80, maxfun=2000)
        if c_opt < c_init:
            h_interp, c_init = h_opt, c_opt

    if c_init < c5_best:
        return h_interp, c_init
    return None, c5_best
