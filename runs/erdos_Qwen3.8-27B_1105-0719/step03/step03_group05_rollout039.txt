This is getting complex. The key point is that for fixed s, each f_d(c) is a quadratic in c (with coefficient -C_d on c²). The max of these quadratics in c can be minimized by finding the c where the active constraints are balanced.

If C_d > 0 (positive autocorrelation), f_d is concave in c, and the max is at the boundary (c=0 or c=max).
If C_d < 0 (negative autocorrelation), f_d is convex in c, and the max could be in the interior.

For the max to be minimized in the interior, we need the active constraints to have C_d < 0 (so they're convex in c).

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def repair_feasible(h):
    n = len(h)
    h = np.clip(np.asarray(h, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, 1.0 - h[idx])
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-14:
        idx = int(np.argmax(1.0 - h)) if diff2 > 0 else int(np.argmin(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def two_cluster_profile_c(s, c, n, delta, C):
    """Fast profile evaluation for h = 0.5 + c*s using precomputed delta, C."""
    profile = np.zeros(2 * n - 1)
    f0 = n * (0.25 - c * c)
    profile[n - 1] = f0
    dx = 2.0 / n
    for d in range(1, n):
        base = 0.25 * (n - d) - c * c * C[d]
        prof_plus = base + 0.5 * c * delta[d]
        prof_minus = base - 0.5 * c * delta[d]
        profile[n - 1 + d] = prof_plus
        profile[n - 1 - d] = prof_minus
    return profile * dx

def compute_delta_C(s, n):
    """Compute delta_d = B_d - A_d and C_d (autocorrelation) for binary s (±1, mean 0)."""
    delta = np.zeros(n)
    C = np.zeros(n)
    cumsum = np.cumsum(s)
    for d in range(1, n):
        A_d = cumsum[n - 1 - d]
        B_d = cumsum[n - 1] - cumsum[d - 1]
        delta[d] = B_d - A_d
        C[d] = np.dot(s[0:n-d], s[d:n])
    C[0] = n
    return delta, C

        return float(np.max(prof))
    gr = (np.sqrt(5) - 1) / 2

    c1 = b - gr * (b - a)

            c2 = a + gr * (b - a)
            f2 = f(c2)

def lse_descent(h, beta=80.0, max_iters=200, alpha_init=0.004):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best_c = exact_score(h)
    alpha = alpha_init
    stuck = 0
    M = 2 * n - 1
    for it in range(max_iters):
        profile = compute_profile(h)
        c_cur = float(np.max(profile))
        scaled = beta * profile
        shifted = scaled - np.max(scaled)
        weights = np.exp(shifted)
        wsum = weights.sum()
        if wsum < 1e-300:
            break
        weights /= wsum
        # Compute gradient of the weighted sum of profile terms
        grad = np.zeros(n)
        for m in range(M):
            w = weights[m]
            if w < 1e-10:
                continue
            d = m - (n - 1)
            if d == 0:
                grad += w * dx * (1.0 - 2.0 * h)
            elif d > 0:
                grad[d:] += w * dx * (1.0 - h[0:n-d])
                grad[0:n-d] += -w * dx * h[d:n]
            else:
                dp = -d
                grad[0:n-dp] += w * dx * (1.0 - h[dp:n])
                grad[dp:n] += -w * dx * h[0:n-dp]
        grad -= grad.mean()
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad /= gnorm
        improved = False
        for a in [alpha, alpha*0.5, alpha*0.25, alpha*0.1]:
            h_try = repair_feasible(h - a * grad)
            s = exact_score(h_try)
            if s < best_c - 1e-14:

                improved = True
                alpha = min(alpha * 1.15, 0.03)
                stuck = 0
                break
            elif s < c_cur - 1e-14:
                h = h_try
                improved = True
                break
        if not improved:
            alpha *= 0.5
            stuck += 1

                break
    return h, best_c

        c_cur = float(np.max(profile))
        # Pick a random active constraint
        active = np.where(profile >= c_cur - 0.001)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(profile))])
        m = active[np.random.randint(len(active))]
        d = m - (n - 1)
        grad = np.zeros(n)
        if d == 0:
            grad = dx * (1.0 - 2.0 * h)
        elif d > 0:
            grad[d:] = dx * (1.0 - h[0:n-d])
            grad[0:n-d] = -dx * h[d:n]
        else:
            dp = -d
            grad[0:n-dp] = dx * (1.0 - h[dp:n])
            grad[dp:n] = -dx * h[0:n-dp]
        grad -= grad.mean()
        v = momentum * v + (1 - momentum) * grad
        vnorm = np.linalg.norm(v)
        if vnorm < 1e-14:
            break
        v /= vnorm
        step = alpha
        improved = False
        for a in [step, step*0.5, step*0.25]:
            h_try = repair_feasible(h - a * v)
            s = exact_score(h_try)
            if s < best_c - 1e-14:
                h = h_try
                best_c = s
                improved = True
                stuck = 0
                break
            elif s < c_cur - 1e-14:
                h = h_try
                improved = True
                break
        if not improved:
            alpha *= 0.9
            stuck += 1
            if stuck > 100 or alpha < 1e-8:
                break
    return h, best_c

def sa_two_cluster(s, n, max_iters=3000, T_init=1.0, T_final=0.01, rng=None):
    """Simulated annealing on binary patterns with c re-optimized."""
    s = s.copy()
    c_cur, e_cur = two_cluster_max_c(s, n)
    best_s = s.copy()
    best_c = c_cur
    best_e = e_cur
    T = T_init
    cooling = (T_final / T_init) ** (1.0 / max(max_iters, 1))
    # Precompute delta, C for fast updates
    delta, C = compute_delta_C(s, n)
    for it in range(max_iters):
        T *= cooling
        i = rng.randint(n)
        j = rng.randint(n)
        if s[i] == s[j]:
            continue
        # Swap s[i] and s[j]
        s[i], s[j] = s[j], s[i]
        # Recompute delta and C efficiently
        delta_new, C_new = compute_delta_C(s, n)
        c_new, e_new = two_cluster_max_c(s, n)
        delta, C = delta_new, C_new
        de = e_new - e_cur
        if de < 0 or rng.rand() < np.exp(-de / max(T, 1e-10)):
            e_cur = e_new
            c_cur = c_new
            if e_new < best_e:
                best_e = e_new
                best_c = c_new
                best_s = s.copy()
        else:
            s[i], s[j] = s[j], s[i]
    return best_s, best_c, best_e

def lfsr_sequence(m):
    """Generate a maximal-length LFSR sequence of length 2^m - 1 as ±1."""
    n = (1 << m) - 1
    # Use a known primitive polynomial for m=6 (length 63)
    # For m=6: x^6 + x + 1 (taps at positions 6 and 1)
    taps = {6, 1} if m == 6 else {m, m//2}  # generic fallback
    state = 1
    seq = []
    for _ in range(n):
        seq.append(1 if state & 1 else -1)
        feedback = 1
        for t in taps:
            feedback ^= (state >> t) & 1
        state = (state >> 1) | (feedback << (m - 1))
    return np.array(seq, dtype=np.float64)

def balance_sequence(s, n_target):
    """Adjust a ±1 sequence to have exactly n/2 +1s and n/2 -1s, length n_target."""
    n = len(s)
    if n == n_target:
        # Balance by flipping
        num_pos = int(np.sum(s > 0))
        target_pos = n_target // 2
        if num_pos > target_pos:
            # Flip some +1 to -1
            pos_idx = np.where(s > 0)[0]
            to_flip = pos_idx[:num_pos - target_pos]
            s[to_flip] = -1
        elif num_pos < target_pos:
            neg_idx = np.where(s < 0)[0]
            to_flip = neg_idx[:target_pos - num_pos]
            s[to_flip] = 1
        return s
    elif n > n_target:
        # Truncate and rebalance
        s = s[:n_target]
        num_pos = int(np.sum(s > 0))
        target_pos = n_target // 2
        if num_pos > target_pos:
            pos_idx = np.where(s > 0)[0]
            rng_tmp = np.random.RandomState(0)
            rng_tmp.shuffle(pos_idx)
            s[pos_idx[:num_pos - target_pos]] = -1
        elif num_pos < target_pos:
            neg_idx = np.where(s < 0)[0]
            rng_tmp = np.random.RandomState(0)
            rng_tmp.shuffle(neg_idx)
            s[neg_idx[:target_pos - num_pos]] = 1
        return s
    else:
        # Extend
        s_new = np.zeros(n_target)
        s_new[:n] = s
        remaining = n_target - n
        num_pos = int(np.sum(s > 0))
        target_pos = n_target // 2
        need_pos = target_pos - num_pos
        if need_pos > 0:
            s_new[n:n+need_pos] = 1
        elif need_pos < 0:
            s_new[n:n+(-need_pos)] = -1
        # Fill rest randomly balanced
        rest = remaining - abs(need_pos)
        half = rest // 2
        s_new[n+abs(need_pos):n+abs(need_pos)+half] = 1
        s_new[n+abs(need_pos)+half:n+abs(need_pos)+rest] = -1
        return s_new

def slsqp_solve(h0, maxiter=100):
    n = len(h0)
    dx = 2.0 / n
    M = 2 * n - 1
    c0 = exact_score(h0)
    def obj(x):
        return x[-1]
    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    def eq_fun(x):
        return x[:-1].sum() - n / 2.0
    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g
    def ineq_fun(x):
        h = np.clip(x[:-1], 0, 1)
        t = x[-1]
        profile = np.correlate(h, 1 - h, mode="full") * dx
        return t - profile
    def ineq_jac(x):
        h = np.clip(x[:-1], 0, 1)
        J = np.zeros((M, n + 1))
        for m in range(M):
            d = m - (n - 1)
            if d == 0:
                J[m, :-1] = -dx * (1.0 - 2.0 * h)
            elif d > 0:
                J[m, d:n] = -dx * (1.0 - h[0:n-d])
                J[m, 0:n-d] = dx * h[d:n]
            else:
                dp = -d
                J[m, 0:n-dp] = -dx * (1.0 - h[dp:n])
                J[m, dp:n] = dx * h[0:n-dp]
            J[m, -1] = 1.0
        return J
    x0 = np.concatenate([h0, [c0 + 1e-3]])
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 0.5)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None and np.isfinite(res.x).all():
            h_new = repair_feasible(res.x[:-1])
            c_new = exact_score(h_new)
            if c_new <= c0 + 1e-9:
                return h_new, c_new
    except Exception:
        pass
    return None, None

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = np.linspace(0, 2, n + 1)
    xt = np.linspace(0, 2, new_n + 1)
    h_interp = np.zeros(new_n)
    for k in range(new_n):
        lo_k, hi_k = xt[k], xt[k+1]
        total_mass = 0.0
        total_width = 0.0
        for j in range(n):
            lo_j, hi_j = xs[j], xs[j+1]
            overlap = min(hi_k, hi_j) - max(lo_k, lo_j)
            if overlap > 0:
                total_mass += h[j] * overlap
                total_width += overlap
        h_interp[k] = total_mass / total_width if total_width > 1e-15 else 0.5
    h_interp = np.clip(h_interp, 0.0, 1.0)
    return repair_feasible(h_interp)

def symmetrize(h):
    return (h + h[::-1]) / 2.0

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()
    rng = np.random.RandomState(seed)
    np.random.seed(seed)

    def time_left():
        return budget_s + t_start - time.perf_counter()

    # Get parent
    parent = None
    try:
        parent = np.array(initial_h_values, dtype=np.float64)
    except NameError:
        pass
    except Exception:
        pass

    if parent is None or len(parent) < 4:
        parent = np.full(70, 0.5)

    parent = np.clip(parent, 0.0, 1.0)
    parent = repair_feasible(parent)
    best_h = parent.copy()
    best_c = exact_score(parent)

    # Symmetrize parent
    h_sym = repair_feasible(symmetrize(parent))
    c_sym = exact_score(h_sym)
    if c_sym < best_c - 1e-12:
        best_h = h_sym
        best_c = c_sym

    n_parent = len(best_h)

    # === Stage 1: Two-cluster search (NEW) ===
    if time_left() > 200:
        # Try LFSR-based starting patterns
        n_test = n_parent
        # Get LFSR sequence of length 63, adjust to n_test
        lfsr_s = lfsr_sequence(6)  # length 63
        s_init = balance_sequence(lfsr_s, n_test)
        
        # Also try random balanced sequences
        s_rand = np.zeros(n_test)
        half = n_test // 2
        s_rand[:half] = 1
        s_rand[half:] = -1
        rng.shuffle(s_rand)
        
        # Optimized c for each
        for s0 in [s_init, s_rand]:
            if time_left() < 50:
                break
            c_opt, c5_val = two_cluster_max_c(s0, n_test)
            h_val = 0.5 + c_opt * s0
            h_val = repair_feasible(h_val)
            c5_actual = exact_score(h_val)
            if c5_actual < best_c - 1e-12:
                best_h = h_val.copy()
                best_c = c5_actual
            
            # SA on the two-cluster family
            if time_left() > 100:
                s_best, c_best_tc, e_best_tc = sa_two_cluster(s0, n_test, 
                                                               max_iters=2000,
                                                               T_init=0.8, 
                                                               T_final=0.01, rng=rng)
                h_tc = 0.5 + c_best_tc * s_best
                h_tc = repair_feasible(h_tc)
                c5_tc = exact_score(h_tc)
                if c5_tc < best_c - 1e-12:
                    best_h = h_tc.copy()
                    best_c = c5_tc

    # === Stage 2: LSE descent from best ===
    if time_left() > 100:
        h_lse, c_lse = lse_descent(best_h, beta=60, max_iters=100)
        if c_lse < best_c - 1e-13:
            best_h = h_lse.copy()
            best_c = c_lse
        h_ase, c_ase = active_set_descent(best_h, max_iters=100, n_active=10)
        if c_ase < best_c - 1e-13:
            best_h = h_ase.copy()
            best_c = c_ase

    # === Stage 3: Subgradient with momentum ===
    if time_left() > 80:
        h_sg, c_sg = subgradient_descent(best_h, max_iters=300, alpha_init=0.015, momentum=0.4)
        if c_sg < best_c - 1e-13:
            best_h = h_sg.copy()
            best_c = c_sg
        h_lse2, c_lse2 = lse_descent(best_h, beta=100, max_iters=80)
        if c_lse2 < best_c - 1e-13:
            best_h = h_lse2.copy()
            best_c = c_lse2

    # === Stage 4: Resolution exploration ===
    if time_left() > 100:
        n_cur = len(best_h)
        trial_ns = set()
        for factor in [0.85, 0.9, 0.95, 1.05, 1.1, 1.15, 1.2, 1.3]:
            d = int(round(n_cur * factor / 2)) * 2
            d = max(28, min(d, 140))
            trial_ns.add(d)
        trial_ns.discard(n_cur)
        
        for new_n in sorted(trial_ns, key=lambda x: abs(x - n_cur))[:5]:
            if time_left() < 40:
                break
            h_interp = interpolate_to_n(best_h, new_n)
            h_new, c_new = lse_descent(h_interp, beta=80, max_iters=60)
            if c_new < best_c - 1e-13:
                best_h = h_new.copy()
                best_c = c_new
            h_new2, c_new2 = active_set_descent(best_h, max_iters=60, n_active=10)
            if c_new2 < best_c - 1e-13:
                best_h = h_new2.copy()
                best_c = c_new2
            # Two-cluster at new resolution
            if time_left() > 60:
                s_rand2 = np.zeros(new_n)
                half2 = new_n // 2
                s_rand2[:half2] = 1
                s_rand2[half2:] = -1
                rng.shuffle(s_rand2)
                s_b2, c_b2, _ = sa_two_cluster(s_rand2, new_n, max_iters=1000, T_init=0.6, T_final=0.01, rng=rng)
                h_b2 = repair_feasible(0.5 + c_b2 * s_b2)
                c5_b2 = exact_score(h_b2)
                if c5_b2 < best_c - 1e-12:
                    best_h = h_b2.copy()
                    best_c = c5_b2
                h_lse3, c_lse3 = lse_descent(best_h, beta=80, max_iters=50)
                if c_lse3 < best_c - 1e-13:
                    best_h = h_lse3.copy()
                    best_c = c_lse3

    # === Stage 5: SLSQP ===
    if time_left() > 70:
        h_sol, c_sol = slsqp_solve(best_h, maxiter=150)
        if h_sol is not None and c_sol < best_c - 1e-12:
            best_h = h_sol.copy()
            best_c = c_sol
            h_ase, c_ase = active_set_descent(best_h, max_iters=60, n_active=10)
            if c_ase < best_c - 1e-13:
                best_h = h_ase.copy()
                best_c = c_ase

    # === Stage 6: Multi-start with diverse perturbations ===
    if time_left() > 80:
        n_b = len(best_h)
        starts = [best_h.copy()]
        for eps in [0.005, 0.01, 0.02, 0.03]:
            z = rng.randn(n_b)
            z -= z.mean()
            starts.append(repair_feasible(best_h + eps * z))
        # Symmetric
        z = rng.randn(n_b)
        z_sym = symmetrize(z)
        z_sym -= z_sym.mean()
        starts.append(repair_feasible(best_h + 0.015 * z_sym))
        
        for h0 in starts:
            if time_left() < 30:
                break
            h_lse_s, c_lse_s = lse_descent(h0, beta=80, max_iters=50)
            if c_lse_s < best_c - 1e-13:
                best_h = h_lse_s.copy()
                best_c = c_lse_s
            h_sg_s, c_sg_s = subgradient_descent(best_h, max_iters=100, alpha_init=0.01, momentum=0.3)
            if c_sg_s < best_c - 1e-13:
                best_h = h_sg_s.copy()
                best_c = c_sg_s

    # === Stage 7: Focused active set with varying sizes ===
    for n_act in [5, 8, 12, 16, 20]:
        if time_left() < 20:
            break
        h_ase, c_ase = active_set_descent(best_h, max_iters=100, n_active=n_act, alpha_init=0.003)
        if c_ase < best_c - 1e-14:
            best_h = h_ase.copy()
            best_c = c_ase

    # === Stage 8: LSE with varying beta ===
    for beta in [40, 100, 150, 200]:
        if time_left() < 15:
            break
        h_lse, c_lse = lse_descent(best_h, beta=beta, max_iters=80, alpha_init=0.003)
        if c_lse < best_c - 1e-14:
            best_h = h_lse.copy()
            best_c = c_lse

    # === Stage 9: Final two-cluster refinement ===
    if time_left() > 60:
        n_final = len(best_h)
        # Extract binary pattern from current best
        half = n_final // 2
        idx_top = np.argsort(best_h)[::-1][:half]
        s_extract = np.zeros(n_final)
        s_extract[idx_top] = 1
        s_extract[np.where(~np.isin(np.arange(n_final), idx_top))[0]] = -1
        # SA on this pattern
        s_bf, c_bf, _ = sa_two_cluster(s_extract, n_final, max_iters=2000, T_init=0.5, T_final=0.005, rng=rng)
        h_bf = repair_feasible(0.5 + c_bf * s_bf)
        c5_bf = exact_score(h_bf)
        if c5_bf < best_c - 1e-12:
            best_h = h_bf.copy()
            best_c = c5_bf
        # Refine continuously from best of (old best, two-cluster)
        h_lse_f, c_lse_f = lse_descent(best_h, beta=120, max_iters=80, alpha_init=0.002)
        if c_lse_f < best_c - 1e-14:
            best_h = h_lse_f.copy()
            best_c = c_lse_f

    # === Stage 10: Final SLSQP polish ===
    if time_left() > 40:
        h_sol, c_sol = slsqp_solve(best_h, maxiter=200)
        if h_sol is not None and c_sol < best_c - 1e-14:
            best_h = h_sol.copy()
            best_c = c_sol

    # === Stage 11: Final active set + subgradient ===
    if time_left() > 20:
        h_ase, c_ase = active_set_descent(best_h, max_iters=150, n_active=10, alpha_init=0.002)
        if c_ase < best_c - 1e-14:
            best_h = h_ase.copy()
            best_c = c_ase
        h_sg_f, c_sg_f = subgradient_descent(best_h, max_iters=200, alpha_init=0.005, momentum=0.5)
        if c_sg_f < best_c - 1e-14:
            best_h = h_sg_f.copy()
            best_c = c_sg_f

    # Final validation
    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)

    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            final_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 2:
            final_c = float(eval_result[0])
    except Exception:
        pass

    return best_h, final_c, len(best_h)
