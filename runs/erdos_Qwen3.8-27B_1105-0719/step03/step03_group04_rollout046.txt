import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def repair_feasible(h):
    n = len(h)
    h = np.clip(np.asarray(h, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, 1.0 - h[idx])
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)[::-1]
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-14:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def build_gradient(h, n, dx):
    M = 2 * n - 1
    G = np.zeros((M, n))
    for d in range(-n + 1, n):
        row = np.zeros(n)
        if d >= 0:
            valid_i = range(0, n - d)
            for i in valid_i:
                row[i] += dx * (1.0 - h[i + d])
                row[i + d] -= dx * h[i]
        else:
            e = -d
            valid_i = range(e, n)
            for i in valid_i:
                row[i] += dx * (1.0 - h[i - e])
                row[i - e] -= dx * h[i]
        G[n - 1 + d] = row
    return G

def build_gradient_fast(h, n, dx):
    M = 2 * n - 1
    G = np.zeros((M, n))
    center = n - 1
    G[center] = dx * (1.0 - 2.0 * h)
    for d in range(1, n):
        row = np.zeros(n)
        row[:n-d] += dx * (1.0 - h[d:])
        row[d:] -= dx * h[:n-d]
        G[center + d] = row
        row2 = np.zeros(n)
        row2[d:] += dx * (1.0 - h[:n-d])
        row2[:n-d] -= dx * h[d:]
        G[center - d] = row2
    return G

def get_active_set(profile, C, beam=None, top_k=15):
    if beam is None:
        spread = C - np.median(profile)
        beam = max(0.003, min(0.015, 0.15 * spread))
    thresh = C - beam
    active = np.where(profile >= thresh)[0]
    if len(active) == 0:
        active = np.array([np.argmax(profile)])
    if len(active) > top_k:
        top_idx = np.argsort(profile[active])[-top_k:]
        active = active[top_idx]
    return active

def symmetrize(h):
    return (h + h[::-1]) / 2.0

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

def qp_trust_region(h, active, radius, gamma, n, dx, G):
    m = n // 2
    h_half = h[:m]
    G_half = G[:, :m] + G[:, m:]
    h_full = np.zeros(n)
    h_full[:m] = h_half
    h_full[m:] = h_half[::-1]
    profile = compute_profile(h_full)
    C = np.max(profile)
    p_act = profile[active]
    G_act = G_half[active, :]
    M_act = len(active)
    
    delta = cp.Variable(m)
    r = cp.Variable()
    lower_d = np.maximum(-h_half, -radius)
    upper_d = np.minimum(1.0 - h_half, radius)
    
    constraints = [
        r >= 0,
        sum(delta) == 0,
        delta >= lower_d,
        delta <= upper_d
    ]
    for k in range(M_act):
        constraints.append(p_act[k] + G_act[k] @ delta <= r)
    
    objective = cp.Minimize(r + gamma * cp.sum_squares(delta))
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.ECOS, max_iter=5000, feastol=1e-10, abstol=1e-10, reltol=1e-10)
        if prob.status in ['optimal', 'optimal_inaccurate'] and delta.value is not None:
            return np.asarray(delta.value).flatten(), r.value, C
    except Exception:
        pass
    try:
        prob2 = cp.Problem(objective, constraints)
        prob2.solve(solver=cp.SCS, max_iters=10000, eps=1e-9)
        if prob2.status in ['optimal', 'optimal_inaccurate'] and delta.value is not None:
            return np.asarray(delta.value).flatten(), r.value, C
    except Exception:
        pass
    return None, None, C

def apply_half_step(h, delta_half, alpha):
    n = len(h)
    m = n // 2
    full_delta = np.zeros(n)
    full_delta[:m] = delta_half
    full_delta[m:] = delta_half[::-1]
    h_new = h + alpha * full_delta
    return repair_feasible(h_new)

def lse_gradient(h, beta):
    n = len(h)
    dx = 2.0 / n
    profile = compute_profile(h)
    C = np.max(profile)
    shifted = beta * (profile - C)
    shifted = np.clip(shifted, -500, 0)
    exps = np.exp(shifted)
    w = exps / exps.sum()
    G = build_gradient_fast(h, n, dx)
    grad = G.T @ w
    grad -= grad.mean()
    return grad, C

def make_smooth_start(n, rng, mode=0):
    h = np.zeros(n)
    if mode == 0:
        h[:] = 0.5
        h += 0.3 * np.sin(np.pi * np.arange(n) / n + rng.uniform(0, 2 * np.pi))
    elif mode == 1:
        nb = rng.randint(3, 8)
        block = n // nb
        for b in range(nb):
            h[b*block:(b+1)*block] = rng.uniform(0.2, 0.8)
        if nb * block < n:
            h[nb*block:] = rng.uniform(0.2, 0.8)
    elif mode == 2:
        phi = 0.6180339887498949
        for i in range(n):
            h[i] = 0.5 + 0.35 * np.sin(2 * np.pi * ((i * phi) % 1.0))
    elif mode == 3:
        h = rng.uniform(0.1, 0.9, n)
        h = (h + h[::-1]) / 2.0
    else:
        h[:] = 0.5 + 0.25 * np.sin(2 * np.pi * rng.randint(1, 5) * np.arange(n) / n)
    return repair_feasible(h)

def lse_optimize(h, t_start, budget_s, max_iters=100):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c = exact_score(h)
    
    for it in range(max_iters):
        if time.perf_counter() - t_start > budget_s - 10:
            break
        beta = 5.0 + 10.0 * (it / max_iters)
        grad, C = lse_gradient(h, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        grad = grad / gnorm
        best_step = None
        for alpha in [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]:
            h_try = repair_feasible(h - alpha * grad)
            s = exact_score(h_try)
            if s < best_c - 1e-13:
                best_step = h_try
                best_c = s
                best_h = h_try.copy()
        if best_step is None:
            for alpha in [0.0002, 0.0001, 0.00005]:
                h_try = repair_feasible(h - alpha * grad)
                s = exact_score(h_try)
                if s < best_c - 1e-14:
                    best_step = h_try
                    best_c = s
                    best_h = h_try.copy()
                    break
        if best_step is not None:
            h = best_step
        else:
            break
    return best_h, best_c

def slsqp_polish(h, t_start, budget_s, max_iter=200):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    h = h.copy()
    c0 = exact_score(h)
    
    def obj_fn(x):
        return x[-1]
    
    def eq_fn(x):
        return x[:-1].sum() - n / 2.0
    
    def ineq_fn(x):
        hv = x[:-1]
        t = x[-1]
        prof = np.correlate(hv, 1 - hv, mode="full") * dx
        return t - prof
    
    def ineq_jac(x):
        hv = x[:-1]
        G = build_gradient_fast(hv, n, dx)
        J = np.zeros((M, n + 1))
        J[:, :-1] = -G
        J[:, -1] = 1.0
        return J
    
    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g
    
    x0 = np.concatenate([h, [c0 + 1e-3]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    cons = [
        {'type': 'eq', 'fun': eq_fn, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fn, 'jac': ineq_jac}
    ]
    
    try:
        res = minimize(obj_fn, x0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-12})
        if res.success or res.x is not None:
            h_sol = repair_feasible(np.clip(res.x[:-1], 0, 1))
            c_sol = exact_score(h_sol)
            if c_sol < c0:
                return h_sol, c_sol
    except Exception:
        pass
    return h, c0

def optimize_with_qp(h, t_start, budget_s, max_iters=300):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c = exact_score(h)
    radius = 0.04
    gamma = 1.0 / (2.0 * radius)
    no_improve = 0
    
    for it in range(max_iters):
        if time.perf_counter() - t_start > budget_s - 15:
            break
        
        profile = compute_profile(h)
        C = np.max(profile)
        if C < best_c - 1e-14:
            best_c = C
            best_h = h.copy()
            no_improve = 0
            radius = min(radius * 1.15, 0.12)
            gamma = 1.0 / (2.0 * radius)
        else:
            no_improve += 1
            if no_improve > 40:
                break
        
        active = get_active_set(profile, C, beam=None, top_k=14)
        G = build_gradient_fast(h, n, dx)
        
        delta, r_opt, _ = qp_trust_region(h, active, radius, gamma, n, dx, G)
        if delta is not None:
            accepted = False
            for alpha in [1.0, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05]:
                h_try = apply_half_step(h, delta, alpha)
                s = exact_score(h_try)
                if s < C - 1e-13:
                    h = h_try
                    accepted = True
                    break
            if not accepted:
                radius *= 0.7
                gamma = 1.0 / (2.0 * radius)
            else:
                if no_improve > 3:
                    radius = min(radius * 1.1, 0.15)
                    gamma = 1.0 / (2.0 * radius)
        else:
            grad, _ = lse_gradient(h, 8.0)
            gnorm = np.linalg.norm(grad)
            if gnorm > 1e-12:
                grad /= gnorm
                for alpha in [0.01, 0.005, 0.002, 0.001]:
                    h_try = repair_feasible(h - alpha * grad)
                    s = exact_score(h_try)
                    if s < C - 1e-13:
                        h = h_try
                        no_improve = max(0, no_improve - 2)
                        break
            radius *= 0.8
            gamma = 1.0 / (2.0 * radius)
            if radius < 0.001:
                break
    
    return best_h, best_c
