import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx] + 1e-15)
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx] + 1e-15)
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-14:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmin(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_gradient(h, d, n):
    """Compute gradient of f_d (correlation at lag d) w.r.t. h"""
    grad = np.zeros(n)
    if d == 0:
        grad[:] = 1 - 2*h
    elif d > 0:
        grad[:n-d] += (1 - h[d:])
        grad[d:] -= h[:n-d]
    else:
        dp = -d
        grad[dp:] += (1 - h[:n-dp])
        grad[:n-dp] -= h[dp:]
    return grad

def compute_all_gradients(h, n):
    """Compute gradients for all lags. Shape: (2n-1, n)"""
    G = np.zeros((2*n-1, n))
    for m in range(2*n-1):
        d = m - (n-1)
        G[m] = compute_gradient(h, d, n)
    return G

def lp_linearization_step(h, active_indices, n, solver='SCS'):
    """Solve the LP linearization step"""
    profile_unscaled = np.correlate(h, 1-h, mode="full")  # unscaled
    G = compute_all_gradients(h, n)
    
    h_var = cp.Variable(n)
    t = cp.Variable()
    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n/2]
    
    for idx in active_indices:
        g = G[idx]
        f_val = profile_unscaled[idx]
        constraints.append(f_val + g @ (h_var - h) <= t)
    
    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS)

        h_new = repair_feasible(h_new)
        return h_new
    return None
