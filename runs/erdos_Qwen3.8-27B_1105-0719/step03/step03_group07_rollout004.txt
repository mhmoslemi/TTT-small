import numpy as np
from scipy.optimize import minimize
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full"))) * dx

def profile_h(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def compute_gradient_for_lag(h, m, n, dx):
    """Compute gradient of P[m] with respect to h.
    m is the lag index in [0, 2n-2], corresponding to shift d = m - (n-1)."""
    d = m - (n - 1)
    grad = np.zeros(n)
    if d == 0:
        grad[:] = dx * (1.0 - 2.0 * h)
    elif d > 0:
        # P[m] = dx * sum_{i=0}^{n-1-d} h[i+d] * (1-h[i])
        # dP/dh[i] for i in [0, n-1-d] as first factor of (1-h[i]): -dx * h[i+d]
        # dP/dh[j] for j in [d, n-1] as h[j] in h[i+d]: dx * (1-h[j-d])
        for i in range(n - d):
            grad[i] -= dx * h[i + d]
            grad[i + d] += dx * (1.0 - h[i])
    else:
        dp = -d
        # P[m] = dx * sum_{i=dp}^{n-1} h[i] * (1-h[i-dp])
        for i in range(dp, n):
            grad[i] += dx * (1.0 - h[i - dp])
            grad[i - dp] -= dx * h[i]
    return grad

def subgradient_max(h, delta=0.0):
    """Compute subgradient of max(P) at h, using active set within delta of max."""
    n = len(h)
    dx = 2.0 / n
    P = profile_h(h)
    Pmax = P.max()
    active = np.where(P >= Pmax - delta)[0]
    if len(active) == 0:
        active = np.array([np.argmax(P)])
    grad = np.zeros(n)
    for m in active:
        grad += compute_gradient_for_lag(h, m, n, dx)
    grad /= len(active)
    return grad, len(active)

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    # Redistribute
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx])
            h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def projected_subgradient(h, delta=0.005, max_iters=500, t_start=None, budget_s=1100):
    """Projected subgradient descent with active-set leveling."""
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    alpha = 0.01
    patience = 0
    
    for it in range(max_iters):
        if t_start is not None and time.perf_counter() - t_start > budget_s - 20:
            break
            
        grad, n_active = subgradient_max(h, delta)
        # Remove the mean component (project onto sum=0 direction)
        grad -= grad.mean()
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Backtracking line search
        improved = False
        for a_factor in [1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125, 0.015625]:
            a = alpha * a_factor
            h_try = repair_feasible(h - a * grad)
            s = exact_score(h_try)
            if s < best_c:
                h = h_try
                best_c = s
                improved = True
                break

            Pmax = P.max()
            # Set delta to include lags within a reasonable band

def pairwise_transfer(h, t_start=None, budget_s=1100, max_pairs=50):
    """Try pairwise mass transfers that reduce the max overlap."""
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)

    P = profile_h(h)

    for trial in range(max_pairs):

            break

        if d > 0:
            contrib = h[d:] * (1 - h[:-d])

        else:

        w_idx = np.argmax(contrib)

            s = exact_score(h_try)
            if s < best_c - 1e-14:

For binary h (indicator of a set S with |S| = n/2), we want to minimize max_k |S ∩ (S^c shifted by k)|. 

This is related to the concept of "difference sets" and "Steinhaus problem".

Actually, for the discrete problem, a quadratic phase e^{2πi * k²/n} has the property that its autocorrelation is flat (like a Zadoff-Chu sequence). This might be useful!

Let me try: h[i] = 0.5 + a * cos(2π * i² / n + phase)

This is a chirp. The autocorrelation of a chirp is a triangle (Baker's property), which might give a good distribution of the overlap profile.

Let me also think about the "leveling" approach more carefully. At the optimum (or near it), several lags should be at approximately the same maximum value. The subgradient approach naturally tries to level them. But the issue is that the problem is nonconvex (bilinear), so the subgradient might not be sufficient.

A potentially better approach: **Bundle method**. At each iteration, maintain a collection of linear models (cuts) of each P[m] at previous iterates. Then solve a QP to find the next point. This is more robust than pure subgradient.

But implementing a full bundle method is complex. Let me stick with the subgradient + SLSQP approach but make it more effective.

Another idea: **Project the subgradient onto the feasible direction cone**. The feasible directions at h are those d with:
- sum(d) = 0 (to maintain the sum constraint)
- d[i] ≥ 0 when h[i] = 0
- d[i] ≤ 0 when h[i] = 1

So the projected subgradient should be the projection of -grad onto this cone. For a first approximation, just clipping and removing the mean is fine, but a proper projection would be better.

- Adjust d so that sum(d) = 0

This is a simple box-constrained projection with a linear equality constraint.

import numpy as np
from scipy.optimize import minimize
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full"))) * dx

def profile_h(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def grad_lag(h, m, n, dx):
    d = m - (n - 1)
    g = np.zeros(n)
    if d == 0:
        g[:] = dx * (1.0 - 2.0 * h)
    elif d > 0:
        for i in range(n - d):
            g[i] -= dx * h[i + d]
            g[i + d] += dx * (1.0 - h[i])
    else:
        dp = -d
        for i in range(dp, n):
            g[i] += dx * (1.0 - h[i - dp])
            g[i - dp] -= dx * h[i]
    return g

def active_subgradient(h, delta):
    n = len(h)
    dx = 2.0 / n
    P = profile_h(h)
    Pmax = P.max()
    active = np.where(P >= Pmax - delta)[0]
    if len(active) == 0:
        active = np.array([np.argmax(P)])
    g = np.zeros(n)
    for m in active:
        g += grad_lag(h, m, n, dx)
    g /= len(active)
    return g, len(active), P, Pmax

def project_feasible_dir(h, d):
    n = len(h)
    d = d.copy()
    d[np.where(h <= 1e-12)[0]] = max(0, d[np.where(h <= 1e-12)[0]])
    d[np.where(h >= 1 - 1e-12)[0]] = min(0, d[np.where(h >= 1 - 1e-12)[0]])
    d -= d.mean()
    return d

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx])
            h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def subgradient_method(h, t_start, budget_s, max_iters=400):
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    alpha = 0.008
    delta = 0.004
    patience = 0
    best_delta = delta
    
    for it in range(max_iters):
        if t_start is not None and time.perf_counter() - t_start > budget_s - 20:
            break
        g, n_act, P, Pmax = active_subgradient(h, delta)
        d = project_feasible_dir(h, -g)
        gnorm = np.linalg.norm(d)
        if gnorm < 1e-14:
            break
        d /= gnorm
        
        improved = False
        for a_factor in [1.0, 0.7, 0.5, 0.35, 0.25, 0.15, 0.1, 0.05]:
            a = alpha * a_factor
            h_try = repair_feasible(h + a * d)
            s = exact_score(h_try)
            if s < best_c:
                h = h_try
                best_c = s
                improved = True
                break
        if improved:
            patience = 0
            alpha = min(alpha * 1.08, 0.04)

            alpha *= 0.75
            if patience % 10 == 5:
                delta = min(0.02, delta * 1.5)
            if alpha < 1e-6 or patience > 50:

        if patience == 0 and it % 30 == 29:
            delta = best_delta
    
    return h, best_c

def slsqp_solve(h0, maxiter=300, active_k=12, tol=1e-12):
    n = len(h0)
    dx = 2.0 / n
    P = profile_h(h0)
    c0 = P.max()
    span = c0 - np.min(P)
    thresh = c0 - max(0.002, 0.08 * span)
    active = np.where(P >= thresh)[0]
    if len(active) == 0:
        active = np.array([np.argmax(P)])
    if len(active) > active_k:
        top_idx = np.argsort(P[active])[-active_k:]
        active = active[top_idx]
    
    def obj(x):
        return x[-1]
    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    def eq_fun(x):
        return x[:-1].sum() - n / 2.0
    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g
    def ineq_fun(x):
        h = x[:-1]
        t = x[-1]
        prof = profile_h(h)
        return t - prof[active]
    def ineq_jac(x):
        h = x[:-1]
        J = np.zeros((len(active), n + 1))
        for idx, m in enumerate(active):
            J[idx, :-1] = -grad_lag(h, m, n, dx)
        J[:, -1] = 1.0
        return J
    
    x0 = np.concatenate([h0, [c0 + 1e-4]])
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 1.0)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': tol})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = repair_feasible(res.x[:-1])
            c_new = exact_score(h_new)
            if c_new <= c0 + 1e-10:
                return h_new
    except Exception:
        pass
    return None

def interp_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    return repair_feasible(np.clip(h_new, 0.0, 1.0))
