```python
def slp_step(h, all_lags=True, eps=1e-4):
    n = len(h)
    dx = 2.0 / n
    P = np.correlate(h, 1 - h, mode="full") * dx
    C = P.max()
    
    if all_lags:
        active = np.arange(len(P))
    else:
        thresh = C - max(eps, 0.05 * (C - np.median(P)))
        active = np.where(P >= thresh)[0]
    
    # Build gradient matrix G where G[d, k] = ∂P[d]/∂h[k]
    M = 2 * n - 1
    G = np.zeros((len(active), n))
    for idx, m in enumerate(active):
        d = m - (n - 1)
        if d == 0:
            G[idx, :] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            for i in range(n - d):
                G[idx, i + d] += dx * (1.0 - h[i])
                G[idx, i] -= dx * h[i + d]
        else:
            dp = -d
            for j in range(n - dp):
                G[idx, j + dp] += dx * (1.0 - h[j])
                G[idx, j] -= dx * h[j + dp]
    
    # LP: minimize t
    # s.t. t - G[d,:] @ x >= P[d]_0 - G[d,:] @ h  for each d
    #      0 <= x <= 1
    #      sum(x) = n/2
    
    # Variables: [x_0, ..., x_{n-1}, t]
    # c = [0, ..., 0, 1]
    # A_ub: for each d, -G[d,:] x + t <= -(P[d]_0 - G[d,:] @ h)
    #        => -G[d,:] x + t <= G[d,:] @ h - P[d]_0
    # A_eq: sum(x) = n/2 => [1,...,1,0] x = n/2
    # bounds: 0 <= x_i <= 1, t >= 0 (or t >= C - some bound)
    
    c_vec = np.zeros(n + 1)
    c_vec[-1] = 1.0
    
    A_ub = np.zeros((len(active), n + 1))
    A_ub[:, :-1] = -G
    A_ub[:, -1] = 1.0
    b_ub = G @ h - P[active]
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :-1] = 1.0
    b_eq = np.array([n / 2.0])
    
    bounds = [(0, 1)] * n + [(0, None)]
    
    from scipy.optimize import linprog
    res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.status == 0:
        x_new = res.x[:-1]
        t_new = res.x[-1]
        return x_new, t_new
    return None
```

```python
import numpy as np
from scipy.optimize import minimize, linprog
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full"))) * dx

def profile_h(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx])
            h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def build_gradient_matrix(h, lag_indices):
    """Build gradient matrix for specified lag indices.
    Returns G where G[idx, k] = dP[lag_indices[idx]]/dh[k]"""
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((len(lag_indices), n))
    for idx, m in enumerate(lag_indices):
        d = m - (n - 1)
        if d == 0:
            G[idx, :] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            for i in range(n - d):
                G[idx, i + d] += dx * (1.0 - h[i])
                G[idx, i] -= dx * h[i + d]
        else:
            dp = -d
            for j in range(n - dp):
                G[idx, j + dp] += dx * (1.0 - h[j])
                G[idx, j] -= dx * h[j + dp]
    return G

def slp_solve(h, all_lags=True, tol=1e-12):
    """One SLP iteration: solve the linearized minimax LP."""
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    P = np.correlate(h, 1 - h, mode="full") * dx
    C = P.max()
    
    if all_lags:
        active = np.arange(M)
    else:
        thresh = C - 0.01
        active = np.where(P >= thresh)[0]
        if len(active) > 30:
            active = np.argsort(P)[::-1][:30]
    
    G = build_gradient_matrix(h, active)
    
    # LP: minimize t
    # s.t. t - G[d,:] @ x >= P[d] - G[d,:] @ h   (i.e., t >= P[d] + G[d,:]@(x-h))
    #      0 <= x <= 1
    #      sum(x) = n/2
    #
    # In standard form for linprog (minimize c@var s.t. A_ub@var <= b_ub):
    # -G[d,:] @ x + t <= G[d,:] @ h - P[d]
    
    nvar = n + 1
    c_vec = np.zeros(nvar)
    c_vec[-1] = 1.0
    
    A_ub = np.zeros((len(active), nvar))
    A_ub[:, :-1] = -G
    A_ub[:, -1] = 1.0
    b_ub = (G @ h) - P[active]
    
    A_eq = np.zeros((1, nvar))
    A_eq[0, :-1] = 1.0
    b_eq = np.array([n / 2.0])
    
    bounds = [(0, 1)] * n + [(0, None)]
    
    res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.status == 0 and res.x is not None:
        return res.x[:-1], res.x[-1]
    return None, None

def slp_optimize(h0, max_iter=200, t_start=None, budget_s=1100, all_lags=True):
    """SLP optimization with backtracking."""
    h = h0.copy()
    best_c = exact_score(h)
    best_h = h.copy()
    
    for it in range(max_iter):
        if t_start is not None and time.perf_counter() - t_start > budget_s - 30:
            break
        
        x_lp, t_lp = slp_solve(h, all_lags=all_lags)
        if x_lp is None:
            break
        
        # Backtracking
        improved = False
        for alpha in [1.0, 0.8, 0.6, 0.4, 0.2, 0.1, 0.05, 0.02]:
            h_try = (1 - alpha) * h + alpha * x_lp
            h_try = repair_feasible(h_try)
            c_try = exact_score(h_try)
            if c_try < best_c - 1e-14:
                h = h_try
                best_c = c_try
                best_h = h.copy()
                improved = True
                break
            elif c_try < exact_score(h):
                h = h_try
                c_now = exact_score(h)
                if c_now < best_c:
                    best_c = c_now
                    best_h = h.copy()
                improved = True
                break
        
        if not improved:
            # Try with all lags if we weren't already
            if not all_lags:
                x_lp, t_lp = slp_solve(h, all_lags=True)
                if x_lp is not None:
                    for alpha in [0.1, 0.05, 0.02]:
                        h_try = repair_feasible((1 - alpha) * h + alpha * x_lp)
                        c_try = exact_score(h_try)
                        if c_try < best_c - 1e-14:
                            h = h_try
                            best_c = c_try
                            best_h = h.copy()
                            improved = True
                            break
            if not improved:
                break
    
    return best_h, best_c

def pair_descent(h, max_sweeps=5, t_start=None, budget_s=1100):
    """Pair coordinate descent: for each pair (i,j), optimize h[i], h[j] jointly."""
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    
    for sweep in range(max_sweeps):
        if t_start is not None and time.perf_counter() - t_start > budget_s - 15:
            break
        improved = False
        # Random subset of pairs for speed
        rng = np.random.RandomState(sweep * 1000 + 42)
        pairs = [(rng.randint(n), rng.randint(n)) for _ in range(min(100, n*n//4))]
        # Also include some structured pairs
        for i in range(0, n, max(1, n//20)):
            pairs.append((i, (i + n//2) % n))
        
        for (i, j) in pairs:
            if i == j:
                continue
            if t_start is not None and time.perf_counter() - t_start > budget_s - 15:
                break
            s = h[i] + h[j]  # Must preserve this
            # Try h[i] = s/2 + delta, h[j] = s/2 - delta for various delta
            h_save = h.copy()
            best_local_c = best_c
            best_delta = 0.0
            for delta in np.linspace(-s/2 * 0.9, s/2 * 0.9, 11):
                h[i] = np.clip(s/2 + delta, 0, 1)
                h[j] = s - h[i]
                if h[j] < -1e-12 or h[j] > 1 + 1e-12:
                    continue
                h[j] = np.clip(h[j], 0, 1)
                c = exact_score(h)
                if c < best_local_c:
                    best_local_c = c
                    best_delta = h[i] - s/2
            if best_local_c < best_c - 1e-14:
                best_c = best_local_c
                improved = True
            else:
                h = h_save
    
    return h, best_c

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)

def binary_anneal(h0, n_steps=200, t_start=None, budget_s=1100):
    """Simulated annealing on binarized structure."""
    n = len(h0)

    best_c = exact_score(h)
    best_h = h.copy()
    rng = np.random.RandomState(123)

    T_final = 0.001
    
    for step in range(n_steps):
        if t_start is not None and time.perf_counter() - t_start > budget_s - 15:
            break
        T = T_init * (T_final / T_init) ** (step / n_steps)
        
        # Random perturbation
        idx = rng.choice(n)
        delta = rng.choice([-1, 1]) * (0.02 + 0.03 * T)

        h[idx] = np.clip(h[idx] + delta, 0, 1)
        h = repair_feasible(h)

            best_h = h.copy()

            if rng.random() < np.exp(-(c_new - best_c) / max(T, 1e-10)):
                pass  # accept
            else:
                h[idx] = h_save
                h = repair_feasible(h)
    
    return best_h, best_c
```

```python
import numpy as np
from scipy.optimize import linprog
import time


def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full"))) * dx


def profile_h(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx


def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx])
            h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h


def build_gradient_matrix(h, lag_indices):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((len(lag_indices), n))
    for idx, m in enumerate(lag_indices):
        d = int(m) - (n - 1)
        if d == 0:
            G[idx, :] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            for i in range(n - d):
                G[idx, i + d] += dx * (1.0 - h[i])
                G[idx, i] -= dx * h[i + d]
        else:
            dp = -d
            for j in range(n - dp):
                G[idx, j + dp] += dx * (1.0 - h[j])
                G[idx, j] -= dx * h[j + dp]
    return G


def slp_one_step(h, trust=0.15):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    P = np.correlate(h, 1 - h, mode="full") * dx

    active = np.arange(M)
    G = build_gradient_matrix(h, active)

    nvar = n + 1
    c_vec = np.zeros(nvar)
    c_vec[-1] = 1.0

    n_ineq = M + 2 * n
    A_ub = np.zeros((n_ineq, nvar))
    b_ub = np.zeros(n_ineq)

    A_ub[:M, :-1] = -G
    A_ub[:M, -1] = 1.0
    b_ub[:M] = (G @ h) - P

    A_ub[M:M+n, :-1] = np.eye(n)
    b_ub[M:M+n] = h + trust
    A_ub[M+n:2*M+n, :-1] = -np.eye(n)
    b_ub[M+n:2*M+n] = -h + trust

    A_eq = np.zeros((1, nvar))
    A_eq[0, :-1] = 1.0
    b_eq = np.array([n / 2.0])

    bounds = [(max(0.0, h[i] - trust), min(1.0, h[i] + trust))] * n + [(0, None)]

    try:
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            return res.x[:-1], res.x[-1]
    except Exception:
        pass
    return None, None


def slp_optimize(h0, max_iter=200, t_start=None, budget_s=1100, trust=0.15):
    h = h0.copy()
    best_c = exact_score(h)
    best_h = h.copy()
    trust_cur = trust

    for it in range(max_iter):
        if t_start is not None and time.perf_counter() - t_start > budget_s - 25:
            break
        if best_c - exact_score(h) < -1e-10:
            h = best_h.copy()
            best_c = exact_score(h)

        x_lp, t_lp = slp_one_step(h, trust_cur)
        if x_lp is None:
            trust_cur *= 0.5
            if trust_cur < 0.005:
                break
            continue

        improved = False
        for alpha in [1.0, 0.7, 0.5, 0.3, 0.15, 0.08, 0.04]:
            h_try = (1.0 - alpha) * h + alpha * x_lp
            h_try = repair_feasible(h_try)
            c_try = exact_score(h_try)
            if c_try < best_c - 1e-14:
                h = h_try
                best_c = c_try
                best_h = h.copy()
                improved = True
                trust_cur = min(trust_cur * 1.2, 0.3)
                break
            elif c_try < exact_score(h) + 1e-12:
                h = h_try
                c_now = exact_score(h)
                if c_now < best_c:
                    best_c = c_now
                    best_h = h.copy()
                improved = True
                break
        if not improved:
            trust_cur *= 0.6
            if trust_cur < 0.003:
                break

    return best_h, best_c


def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    return repair_feasible(np.clip(h_new, 0.0, 1.0))


def softmax_descent(h0, beta=60, max_iter=100, t_start=None, budget_s=1100):
    n = len(h0)
    h = h0.copy()
    best_c = exact_score(h)
    best_h = h.copy()
    vel = np.zeros(n)

    for it in range(max_iter):
        if t_start is not None and time.perf_counter() - t_start > budget_s - 20:
            break
        P = np.correlate(h, 1 - h, mode="full") * dx if False else None
        P = profile_h(h)
        Pmax = P.max()
        exp_arg = beta * (P - Pmax)
        exp_arg -= exp_arg.max()
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum < 1e-300:
            break
        w /= w_sum

        active = np.where(w > 0.01 / len(P))[0]
        if len(active) == 0:
            active = np.argsort(w)[-5:]
        G = build_gradient_matrix(h, active)
        wa = w[active]
        wa /= wa.sum()
        grad = G.T @ wa

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad /= gnorm
        vel = 0.4 * vel + grad
        vnorm = np.linalg.norm(vel)
        if vnorm > 0:
            vel /= vnorm

        improved = False
        for a in [0.04, 0.02, 0.01, 0.005, 0.002]:
            h_try = repair_feasible(h - a * vel)
            c_try = exact_score(h_try)
            if c_try < best_c - 1e-14:
                h = h_try
                best_c = c_try
                best_h = h.copy()
                improved = True
                break
        if not improved:
            break
    return best_h, best_c


def pair_descent(h0, max_sweeps=8, t_start=None, budget_s=1100, seed=0):
    n = len(h0)
    h = h0.copy()
    best_c = exact_score(h)
    rng = np.random.RandomState(seed)

    for sweep in range(max_sweeps):
        if t_start is not None and time.perf_counter() - t_start > budget_s - 10:
            break
        improved = False
        n_pairs = min(150, n * n // 3)
        pairs = []
        for _ in range(n_pairs):
            i = rng.randint(n)
            j = rng.randint(n)
            if i != j:
                pairs.append((i, j))
        step = max(1, n // 25)
        for i in range(0, n, step):
            pairs.append((i, (i + n // 2) % n))

        for (i, j) in pairs:
            if i == j:
                continue
            if t_start is not None and time.perf_counter() - t_start > budget_s - 10:
                break
            s = h[i] + h[j]
            h_save_i = h[i]
            h_save_j = h[j]
            lo = max(0.0, s - 1.0)
            hi = min(1.0, s)
            if lo >= hi:
                continue
            best_local_c = best_c
            best_val = h_save_i
            for frac in np.linspace(0.0, 1.0, 9):
                h[i] = lo + frac * (hi - lo)
                h[j] = s - h[i]
                if h[j] < -1e-12 or h[j] > 1.0 + 1e-12:
                    continue
                c = exact_score(h)
                if c < best_local_c:
                    best_local_c = c
                    best_val = h[i]
            if best_local_c < best_c - 1e-14:
                h[i] = best_val
                h[j] = s - best_val
                best_c = best_local_c
                improved = True
            else:
                h[i] = h_save_i
                h[j] = h_save_j
        if not improved:
            break
    return h, best_c


def make_structured_start(n, ptype, rng):
    h = np.zeros(n)
    if ptype == 0:
        h[0:n:2] = 1.0
        for _ in range(n // 8):
            i = rng.randint(0, n - 1)
            if h[i] != h[i + 1]:
                h[i], h[i + 1] = h[i + 1], h[i]
    elif ptype == 1:
        blen = max(1, n // 12)
        for start in range(0, n, blen * 2):
            h[start:start + blen] = 1.0
    elif ptype == 2:
        golden = 0.6180339887
        for k in range(n // 2):
            pos = int((k * golden) % 1.0 * n)
            h[pos] = 1.0
    elif ptype == 3:
        tm = np.zeros(n, dtype=np.int64)
        for i in range(1, n):
            tm[i] = tm[i // 2] ^ (i % 2)
        h[tm == 1] = 1.0
        n_ones = int(h.sum())
        target = n // 2
        if n_ones < target:
            zeros = np.where(h == 0)[0]
            rng.shuffle(zeros)
            h[zeros[:target - n_ones]] = 1.0
        elif n_ones > target:
            ones = np.where(h == 1)[0]
            rng.shuffle(ones)
            h[ones[:n_ones - target]] = 0.0
    elif ptype == 4:
        h[0:n:3] = 1.0
        h[1:n:3] = 0.5
        for _ in range(n // 6):
            i = rng.randint(n)
            h[i] = rng.choice([0.0, 1.0])
    else:
        idx = rng.choice(n, n // 2, replace=False)
        h[idx] = 1.0
    return repair_feasible(h)


def binary_anneal_step(h0, n_steps=400, t_start=None, budget_s=1100, seed=0):
    n = len(h0)
    h = h0.copy()
    best_c = exact_score(h)
    best_h = h.copy()
    rng = np.random.RandomState(seed)
    T_init = 0.015
    T_final = 0.0005

    for step in range(n_steps):
        if t_start is not None and time.perf_counter() - t_start > budget_s - 10:
            break
        T = T_init * (T_final / T_init) ** (step / max(1, n_steps - 1))
        idx = rng.randint(n)
        delta = rng.choice([-1, 1]) * (0.03 + 0.04 * T)
        h_save = h[idx]
        h[idx] = np.clip(h[idx] + delta, 0, 1)
        h = repair_feasible(h)
        c_new = exact_score(h)
        if c_new < best_c:
            best_c = c_new
            best_h = h.copy()
        else:
            if rng.random() < np.exp(-(c_new - best_c) / max(T, 1e-12)):
                pass
            else:
                h[idx] = np.clip(h_save, 0, 1)
                h = repair_feasible(h)
    return best_h, best_c
```
