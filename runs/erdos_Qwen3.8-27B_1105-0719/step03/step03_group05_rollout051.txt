import numpy as np
from scipy.optimize import minimize
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-10:
        return h
    # Distribute the correction
    if diff > 0:
        mask = h < 1.0
        if mask.sum() == 0:
            return h
        # Add proportionally to available capacity
        cap = (1.0 - h) * mask
        total_cap = cap.sum()
        if total_cap > 1e-15:
            h += diff * cap / total_cap
        else:
            idx = int(np.argmax(1.0 - h))
            h[idx] += min(diff, 1.0 - h[idx])
    else:
        mask = h > 0.0
        if mask.sum() == 0:
            return h
        h += diff * h / h.sum()
    h = np.clip(h, 0.0, 1.0)
    # Fix any remaining discrepancy
    diff2 = target - h.sum()
    if abs(diff2) > 1e-12:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
            h[idx] = min(1.0, h[idx] + diff2)
        else:
            idx = int(np.argmin(h))
            h[idx] = max(0.0, h[idx] + diff2)
    return h

def precompute_gradient_matrix(h):
    """Compute G[i,k] = derivative of profile[k] w.r.t. h[i]."""
    n = len(h)
    M = 2 * n - 1
    dx = 2.0 / n
    G = np.zeros((n, M))
    for i in range(n):
        for k in range(M):
            g = 0.0
            # h[i] as first factor: h[n+k] where n+k=i, so n=i-k
            if 0 <= i - k < n:
                g += (1.0 - h[i - k])
            # h[i] as second factor: (1-h[n]) where n=i, so h[n+k]=h[i+k]
            if 0 <= i + k < n:
                g += (-h[i + k])
            G[i, k] = g * dx
    return G

def find_best_delta(P0, L, lo, hi):
    """Find δ in [lo,hi] minimizing max_k(P0[k] + δ*L[k])."""
    if hi - lo < 1e-14:
        return 0.0
    # Convex piecewise-linear function. Use successive bisection on slope.
    # Evaluate at a grid, then refine.
    best_c = np.inf
    best_d = (lo + hi) / 2
    
    # Coarse grid
    for t in np.linspace(0, 1, 32):
        d = lo + t * (hi - lo)
        c = np.max(P0 + d * L)
        if c < best_c:
            best_c = c
            best_d = d
    
    # Refine
    rng_width = (hi - lo) / 32
    for _ in range(4):
        d_lo = max(lo, best_d - rng_width)
        d_hi = min(hi, best_d + rng_width)
        for t in np.linspace(0, 1, 16):
            d = d_lo + t * (d_hi - d_lo)
            c = np.max(P0 + d * L)
            if c < best_c:
                best_c = c
                best_d = d
        rng_width = (d_hi - d_lo) / 16
    
    return best_d

    n = len(h)
    h = h.copy()
    best_c = exact_score(h)

        if sweep % 10 == 0:
            G = precompute_gradient_matrix(h)
            P0 = compute_profile(h)
        
        total_improvement = 0.0
        order = np.arange(n)
        if rng is not None:
            rng.shuffle(order)
        
        for ii in range(n):
            i = order[ii]
            for jj in range(ii + 1, n):
                j = order[jj]
                s = h[i] + h[j]
                lo = max(0.0, s - 1.0)
                hi = min(1.0, s)
                if hi - lo < 1e-12:
                    continue
                
                L = G[i] - G[j]
                best_d = find_best_delta(P0, L, lo, hi)

            break
    
    return h, best_c

But the issue is that after changing h[i] and h[j], the gradient matrix G changes (because G depends on h). So I need to update G efficiently.

When h changes at positions i and j, how does G change?

That's O(M) entries for each changed position, O(2M) total per pair update.

import numpy as np
from scipy.optimize import minimize
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        cap = 1.0 - h
        total = cap.sum()
        if total > 1e-15:
            h += diff * cap / total
    else:
        total = h.sum()
        if total > 1e-15:
            h += diff * h / total
    h = np.clip(h, 0.0, 1.0)
    diff2 = target - h.sum()
    if abs(diff2) > 1e-12:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
            h[idx] = min(1.0, h[idx] + diff2)
        else:
            idx = int(np.argmin(h))
            h[idx] = max(0.0, h[idx] + diff2)
    return h

def build_G(h):
    n = len(h)
    M = 2 * n - 1
    dx = 2.0 / n
    G = np.zeros((n, M))
    for i in range(n):
        # h[i] as first factor h[n+k]: n = i-k, valid if 0<=i-k<n
        valid1 = (i >= 0) & (i - np.arange(M) >= 0) & (i - np.arange(M) < n)
        # Actually let me just do it element-wise
        ks = np.arange(M)
        # First term: (1-h[i-k]) if 0 <= i-k < n
        idx1 = i - ks
        mask1 = (idx1 >= 0) & (idx1 < n)
        # Second term: -h[i+k] if 0 <= i+k < n
        idx2 = i + ks
        mask2 = (idx2 >= 0) & (idx2 < n)
        
        G[i, mask1] += (1.0 - h[idx1[mask1]]) * dx
        G[i, mask2] += (-h[idx2[mask2]]) * dx
    return G

def find_best_delta(P0, L, lo, hi):
    if hi - lo < 1e-14:
        return 0.0
    best_c = np.inf
    best_d = (lo + hi) / 2
    # Coarse
    for t in np.linspace(0, 1, 32):
        d = lo + t * (hi - lo)
        c = float(np.max(P0 + d * L))
        if c < best_c:
            best_c = c
            best_d = d
    # Refine
    w = (hi - lo) / 32
    for _ in range(4):
        dlo = max(lo, best_d - w)
        dhi = min(hi, best_d + w)
        for t in np.linspace(0, 1, 16):
            d = dlo + t * (dhi - dlo)
            c = float(np.max(P0 + d * L))
            if c < best_c:
                best_c = c
                best_d = d
        w = (dhi - dlo) / 16
    return best_d

def paired_descent(h, max_sweeps=80, rng=None, verbose=False):
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    for sweep in range(max_sweeps):
        G = build_G(h)
        P0 = compute_profile(h)
        total_improvement = 0.0
        for i in range(n):
            for j in range(i + 1, n):
                s = h[i] + h[j]
                lo = max(0.0, s - 1.0)
                hi = min(1.0, s)
                if hi - lo < 1e-12:
                    continue
                L = G[i] - G[j]
                best_d = find_best_delta(P0, L, lo, hi)
                if abs(best_d) < 1e-14:
                    continue
                old_i, old_j = h[i], h[j]
                h[i] = np.clip(h[i] + best_d, 0.0, 1.0)
                h[j] = np.clip(h[j] - best_d, 0.0, 1.0)
                # Recompute
                P0 = compute_profile(h)
                G = build_G(h)
                c = float(np.max(P0))
                if c < best_c - 1e-13:
                    total_improvement += best_c - c
                    best_c = c
        if total_improvement < 1e-12:
            break
    return h, best_c

def lse_descent(h, beta=80.0, max_iters=200, alpha_init=0.004):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best_c = exact_score(h)
    alpha = alpha_init
    stuck = 0
    for it in range(max_iters):
        profile = compute_profile(h)
        scaled = beta * profile
        max_s = np.max(scaled)
        weights = np.exp(scaled - max_s)
        wsum = weights.sum()
        if wsum < 1e-300:
            break
        weights /= wsum
        G = build_G(h)
        grad = G.T @ weights
        grad -= grad.mean()
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad /= gnorm
        improved = False
        for a in [alpha, alpha*0.5, alpha*0.25, alpha*0.1]:
            h_try = repair_feasible(h - a * grad)
            s = exact_score(h_try)
            if s < best_c - 1e-14:
                h = h_try
                best_c = s
                improved = True
                alpha = min(alpha * 1.1, 0.02)
                stuck = 0
                break
        if not improved:
            alpha *= 0.5
            stuck += 1
            if stuck > 30 or alpha < 1e-9:
                break
    return h, best_c

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = np.linspace(0, 2, n + 1)
    xt = np.linspace(0, 2, new_n + 1)
    h_interp = np.zeros(new_n)
    for k in range(new_n):
        total_mass = 0.0
        total_width = 0.0
        for j in range(n):
            overlap = min(xs[j+1], xt[k+1]) - max(xs[j], xt[k])
            if overlap > 0:
                total_mass += h[j] * overlap
                total_width += overlap
        h_interp[k] = total_mass / total_width if total_width > 1e-15 else 0.5
    return repair_feasible(np.clip(h_interp, 0.0, 1.0))
