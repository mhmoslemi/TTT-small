```python
def pair_line_optimize(h, i, j, n_iter=40):
    """Find optimal t for h[i]+=t, h[j]-=t using ternary search."""
    n = len(h)
    dx = 2.0 / n
    a, b = h[i], h[j]
    # Feasible t range
    lo = max(-a, b - 1.0)
    hi = min(1.0 - a, b)
    if hi - lo < 1e-14:
        return h, exact_score(h)
    
    def obj(t):
        h[i] = a + t
        h[j] = b - t
        return exact_score(h)
    
    # Ternary search
    for _ in range(n_iter):
        m1 = lo + (hi - lo) / 3
        m2 = hi - (hi - lo) / 3
        f1 = obj(m1)
        f2 = obj(m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    
    t_opt = (lo + hi) / 2
    h[i] = a + t_opt
    h[j] = b - t_opt
    return h, exact_score(h)
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx] + 1e-15)
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx] + 1e-15)
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-14:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmin(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = np.linspace(0, 2, n + 1)
    xt = np.linspace(0, 2, new_n + 1)
    h_interp = np.zeros(new_n)
    for k in range(new_n):
        lo_k, hi_k = xt[k], xt[k + 1]
        total_mass = 0.0
        total_width = 0.0
        for j in range(n):
            lo_j, hi_j = xs[j], xs[j + 1]
            overlap = min(hi_k, hi_j) - max(lo_k, lo_j)
            if overlap > 0:
                total_mass += h[j] * overlap
                total_width += overlap
        h_interp[k] = total_mass / total_width if total_width > 1e-15 else 0.5
    h_interp = np.clip(h_interp, 0.0, 1.0)
    return repair_feasible(h_interp)

def antisymmetrize(h):
    n = len(h)
    h_out = np.zeros(n)
    for i in range(n):
        h_out[i] = (h[i] + (1.0 - h[n - 1 - i])) / 2.0
    return repair_feasible(h_out)

def build_gradient_matrix(h, n, dx):
    M = 2 * n - 1
    G = np.zeros((M, n))
    for m in range(M):
        d = m - (n - 1)
        if d == 0:
            G[m, :] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            G[m, d:] += dx * (1.0 - h[0:n - d])
            G[m, 0:n - d] += -dx * h[d:n]
        else:
            dp = -d
            G[m, 0:n - dp] += dx * (1.0 - h[dp:n])
            G[m, dp:n] += -dx * h[0:n - dp]
    return G

def lse_descent(h, beta=80.0, max_iters=200, alpha_init=0.004):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best_c = exact_score(h)
    alpha = alpha_init
    stuck = 0
    for it in range(max_iters):
        profile = compute_profile(h)
        c = float(np.max(profile))
        scaled = beta * (profile - c)
        weights = np.exp(scaled)
        wsum = weights.sum()
        if wsum < 1e-300:
            break
        weights /= wsum
        G = build_gradient_matrix(h, n, dx)
        grad = G.T @ weights
        grad -= grad.mean()
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad /= gnorm
        improved = False
        for a in [alpha, alpha * 0.5, alpha * 0.25, alpha * 0.1]:
            h_try = repair_feasible(h - a * grad)
            s = exact_score(h_try)
            if s < best_c - 1e-14:
                h = h_try
                best_c = s
                improved = True
                alpha = min(alpha * 1.1, 0.025)
                stuck = 0
                break
            elif s < c - 1e-14:
                h = h_try
                improved = True
                break
        if not improved:
            alpha *= 0.5
            stuck += 1
            if stuck > 30 or alpha < 1e-9:
                break
    return h, best_c

def pair_line_search(h, i, j, n_iter=45):
    n = len(h)
    a, b = h[i], h[j]
    lo = max(-a, b - 1.0) + 1e-12
    hi = min(1.0 - a, b) - 1e-12
    if hi - lo < 1e-10:
        return h.copy(), 1e18
    h_work = h.copy()

    def obj_at(t):
        h_work[i] = a + t
        h_work[j] = b - t
        return exact_score(h_work)

    f_lo = obj_at(lo)
    f_hi = obj_at(hi)
    f_mid = obj_at((lo + hi) / 2.0)
    best_f = min(f_lo, f_hi, f_mid)
    best_t = (lo + hi) / 2.0
    if f_lo < best_f:
        best_t = lo
    elif f_hi < best_f:
        best_t = hi

    for _ in range(n_iter):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = obj_at(m1)
        f2 = obj_at(m2)
        if f1 < f2:
            hi = m2
            if f1 < best_f:
                best_f = f1
                best_t = m1
        else:
            lo = m1
            if f2 < best_f:
                best_f = f2
                best_t = m2

    h_work[i] = a + best_t
    h_work[j] = b - best_t
    final_f = exact_score(h_work)
    h_work[i] = a + best_t
    h_work[j] = b - best_t
    return h_work, final_f

def select_pairs_from_profile(h, profile, n_pairs_max=30):

    M = len(profile)

    if len(active) == 0:
        active = np.array([int(np.argmax(profile))])
    n_act = min(len(active), 15)
    if len(active) > n_act:
        idx = np.argsort(profile[active])[-n_act:]
        active = active[idx]

    candidates_pos = []
    candidates_neg = []
    dx = 2.0 / n
    for m in active:
        d = m - (n - 1)
        if d >= 0:
            for k in range(n - d):
                val = h[k] * (1.0 - h[k + d])
                if val > 0.01:
                    candidates_pos.append((k, val * (1.0 - h[k + d])))
                    candidates_neg.append((k + d, val * h[k]))
        else:
            dp = -d
            for k in range(n - dp):
                val = h[k + dp] * (1.0 - h[k])
                if val > 0.01:
                    candidates_pos.append((k + dp, val * (1.0 - h[k])))
                    candidates_neg.append((k, val * h[k]))

    candidates_pos.sort(key=lambda x: -x[1])
    candidates_neg.sort(key=lambda x: -x[1])

    seen = set()
    pairs = []
    pos_candidates = [x[0] for x in candidates_pos[:40]]
    neg_candidates = [x[0] for x in candidates_neg[:40]]

    for i in pos_candidates:
        for j in neg_candidates:
            if i != j and (i, j) not in seen:
                seen.add((i, j))
                pairs.append((i, j))
                if len(pairs) >= n_pairs_max:
                    return pairs

    rng_local = np.random.RandomState(0)
    while len(pairs) < min(n_pairs_max, 20):
        i = rng_local.randint(n)
        j = rng_local.randint(n)
        if i != j and (i, j) not in seen:
            seen.add((i, j))
            pairs.append((i, j))
    return pairs

def slsqp_solve(h0, maxiter=200):
    n = len(h0)
    dx = 2.0 / n
    M = 2 * n - 1
    c0 = exact_score(h0)

    def obj(x):
        return x[-1]

    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g

    def eq_fun(x):
        return x[:-1].sum() - n / 2.0

    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g

    def ineq_fun(x):
        h = np.clip(x[:-1], 0, 1)
        t = x[-1]
        profile = np.correlate(h, 1 - h, mode="full") * dx
        return t - profile

    def ineq_jac(x):
        h = np.clip(x[:-1], 0, 1)
        G = build_gradient_matrix(h, n, dx)
        J = np.zeros((M, n + 1))
        J[:, :-1] = -G
        J[:, -1] = 1.0
        return J

    x0 = np.concatenate([h0, [c0 + 1e-3]])
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 0.5)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None and np.isfinite(res.x).all():
            h_new = repair_feasible(res.x[:-1])
            c_new = exact_score(h_new)
            if c_new <= c0 + 1e-9:
                return h_new, c_new
    except Exception:
        pass
    return None, None

def slsqp_antisym(h0, maxiter=150):
    n = len(h0)
    half = n // 2
    y = np.array([(h0[i] + 1.0 - h0[n - 1 - i]) / 2.0 for i in range(half)])
    y = np.clip(y, 0, 1)
    dx = 2.0 / n
    M = 2 * n - 1

    def reconstruct(y_vec):
        h_full = np.zeros(n)
        for i in range(half):
            h_full[i] = y_vec[i]
            h_full[n - 1 - i] = 1.0 - y_vec[i]
        return h_full

    def profile_from_y(yv):
        h_full = reconstruct(yv)
        return np.correlate(h_full, 1.0 - h_full, mode="full") * dx

    c0 = float(np.max(profile_from_y(y)))

    def obj(x):
        return x[-1]

    def obj_jac(x):
        g = np.zeros(half + 1)
        g[-1] = 1.0
        return g

    def ineq_fun(x):
        yv = np.clip(x[:-1], 0, 1)
        t = x[-1]
        p = profile_from_y(yv)
        return t - p

    x0 = np.concatenate([y, [c0 + 1e-3]])
    bounds = [(0.0, 1.0)] * half + [(0.0, c0 + 0.5)]
    cons = [{'type': 'ineq', 'fun': ineq_fun}]
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-11})
        if res.x is not None and np.isfinite(res.x).all():
            y_new = np.clip(res.x[:-1], 0, 1)
            h_new = reconstruct(y_new)
            h_new = repair_feasible(h_new)
            c_new = exact_score(h_new)
            if c_new <= c0 + 1e-9:
                return h_new, c_new
    except Exception:
        pass
    return None, None

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s + t_start - time.perf_counter()

    parent = None
    try:
        parent = np.array(initial_h_values, dtype=np.float64)
    except NameError:
        pass
    except Exception:
        pass

    if parent is None or len(parent) < 4:
        parent = np.full(70, 0.5)

    parent = np.clip(parent, 0.0, 1.0)
    parent = repair_feasible(parent)
    c_parent = exact_score(parent)

    best_h = parent.copy()
    best_c = c_parent

    h_asym = antisymmetrize(parent)
    c_asym = exact_score(h_asym)
    if c_asym < best_c - 1e-12:
        best_h = h_asym.copy()
        best_c = c_asym

    # Stage 1: Initial LSE + active set
    if time_left() > 100:
        h_lse, c_lse = lse_descent(best_h, beta=60, max_iters=80)
        if c_lse < best_c - 1e-13:
            best_h = h_lse.copy()
            best_c = c_lse
        h_ase, c_ase = active_set_descent(best_h, max_iters=80, n_active=10)
        if c_ase < best_c - 1e-13:
            best_h = h_ase.copy()
            best_c = c_ase

    # Stage 2: Pair-line search (main new contribution)
    if time_left() > 150:
        n = len(best_h)
        for round_idx in range(300):
            if time_left() < 30:
                break
            profile = compute_profile(best_h)
            pairs = select_pairs_from_profile(best_h, profile, n_pairs_max=20)
            improved_this_round = False
            for (i, j) in pairs:
                if time_left() < 25:
                    break
                h_try, f_try = pair_line_search(best_h, i, j, n_iter=40)
                if f_try < best_c - 1e-12:
                    best_h = h_try
                    best_c = f_try
                    improved_this_round = True
                    profile = compute_profile(best_h)
                    pairs = select_pairs_from_profile(best_h, profile, n_pairs_max=15)
            if not improved_this_round:
                # Try random pairs as escape
                for _ in range(10):
                    if time_left() < 25:
                        break
                    i = int(rng.randint(n))
                    j = int(rng.randint(n))
                    if i == j:
                        continue
                    h_try, f_try = pair_line_search(best_h, i, j, n_iter=30)
                    if f_try < best_c - 1e-12:
                        best_h = h_try
                        best_c = f_try
                        improved_this_round = True
                if not improved_this_round:
                    # Small perturbation to escape
                    eps = 0.003
                    z = rng.randn(n)
                    z -= z.mean()
                    h_pert = repair_feasible(best_h + eps * z)
                    profile = compute_profile(h_pert)
                    pairs = select_pairs_from_profile(h_pert, profile, n_pairs_max=10)
                    found = False
                    for (i, j) in pairs:
                        h_try, f_try = pair_line_search(h_pert, i, j, n_iter=35)
                        if f_try < best_c - 1e-12:
                            best_h = h_try
                            best_c = f_try
                            found = True
                            break
                    if not found:
                        break

    # Stage 3: LSE + active set after pair search
    if time_left() > 80:
        h_lse, c_lse = lse_descent(best_h, beta=100, max_iters=80)
        if c_lse < best_c - 1e-13:
            best_h = h_lse.copy()
            best_c = c_lse
        h_ase, c_ase = active_set_descent(best_h, max_iters=100, n_active=10)
        if c_ase < best_c - 1e-13:
            best_h = h_ase.copy()
            best_c = c_ase

    # Stage 4: Anti-symmetric SLSQP
    if time_left() > 80:
        h_sol, c_sol = slsqp_antisym(best_h, maxiter=150)
        if h_sol is not None and c_sol < best_c - 1e-12:
            best_h = h_sol.copy()
            best_c = c_sol
            # Polish with pair search
            if time_left() > 40:
                profile = compute_profile(best_h)
                pairs = select_pairs_from_profile(best_h, profile, n_pairs_max=15)
                for (i, j) in pairs:
                    if time_left() < 20:
                        break
                    h_try, f_try = pair_line_search(best_h, i, j, n_iter=35)
                    if f_try < best_c - 1e-12:
                        best_h = h_try
                        best_c = f_try

    # Stage 5: Multi-resolution exploration
    if time_left() > 120:
        n_best = len(best_h)
        try_ns = set()
        for factor in [0.8, 0.85, 0.9, 0.95, 1.0, 1.05, 1.1, 1.15, 1.2, 1.3, 1.4]:
            d = int(round(n_best * factor / 2)) * 2
            d = max(24, min(d, 160))
            try_ns.add(d)
        try_ns.discard(n_best)
        ns_list = sorted(try_ns, key=lambda x: abs(x - n_best))
        for new_n in ns_list[:8]:
            if time_left() < 50:
                break
            h_interp = interpolate_to_n(best_h, new_n)
            h_lse, c_lse = lse_descent(h_interp, beta=80, max_iters=60)
            c_try = c_lse
            h_try = h_lse
            if c_try > best_c:
                h_try = h_interp
                c_try = exact_score(h_try)
            # Pair search at new resolution
            if time_left() > 35:
                profile = compute_profile(h_try)
                pairs = select_pairs_from_profile(h_try, profile, n_pairs_max=15)
                for (i, j) in pairs:
                    if time_left() < 20:
                        break
                    ht, ft = pair_line_search(h_try, i, j, n_iter=30)
                    if ft < c_try - 1e-12:
                        h_try = ht
                        c_try = ft
            if c_try < best_c - 1e-12:
                best_h = h_try.copy()
                best_c = c_try

    # Stage 6: SLSQP full
    if time_left() > 60:
        h_sol, c_sol = slsqp_solve(best_h, maxiter=200)
        if h_sol is not None and c_sol < best_c - 1e-12:
            best_h = h_sol.copy()
            best_c = c_sol
            h_ase, c_ase = active_set_descent(best_h, max_iters=60, n_active=10)
            if c_ase < best_c - 1e-13:
                best_h = h_ase.copy()
                best_c = c_ase

    # Stage 7: More pair search with different pair selection
    if time_left() > 60:
        n = len(best_h)
        profile = compute_profile(best_h)
        c = float(np.max(profile))
        # Use gradient-based pair selection
        dx = 2.0 / n
        G = build_gradient_matrix(best_h, n, dx)
        margin = max(0.002, 3.0 * (c - float(np.median(profile))))
        active = np.where(profile >= c - margin)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(profile))])
        n_act = min(len(active), 20)
        if len(active) > n_act:
            top_idx = np.argsort(profile[active])[-n_act:]
            active = active[top_idx]
        grad = G[active, :].mean(axis=0)
        grad -= grad.mean()
        # Pairs: increase where grad<0, decrease where grad>0
        neg_idx = np.where(grad < -1e-6)[0]
        pos_idx = np.where(grad > 1e-6)[0]
        neg_idx = neg_idx[np.argsort(grad[neg_idx])][:20]
        pos_idx = pos_idx[np.argsort(-grad[pos_idx])][:20]
        for i in neg_idx:
            for j in pos_idx:
                if i == j:
                    continue
                if time_left() < 20:
                    break
                h_try, f_try = pair_line_search(best_h, int(i), int(j), n_iter=35)
                if f_try < best_c - 1e-12:
                    best_h = h_try
                    best_c = f_try
                    profile = compute_profile(best_h)

    # Stage 8: Final polish - multiple methods
    if time_left() > 40:
        h_lse, c_lse = lse_descent(best_h, beta=150, max_iters=100, alpha_init=0.002)
        if c_lse < best_c - 1e-14:
            best_h = h_lse.copy()
            best_c = c_lse
        h_ase, c_ase = active_set_descent(best_h, max_iters=150, n_active=8, alpha_init=0.003)
        if c_ase < best_c - 1e-14:
            best_h = h_ase.copy()
            best_c = c_ase

    # Stage 9: Final pair search polish
    if time_left() > 20:
        n = len(best_h)
        profile = compute_profile(best_h)
        pairs = select_pairs_from_profile(best_h, profile, n_pairs_max=15)
        for (i, j) in pairs:
            if time_left() < 15:
                break
            h_try, f_try = pair_line_search(best_h, i, j, n_iter=40)
            if f_try < best_c - 1e-13:
                best_h = h_try
                best_c = f_try

    # Final SLSQP if time
    if time_left() > 35:
        h_sol, c_sol = slsqp_solve(best_h, maxiter=150)
        if h_sol is not None and c_sol < best_c - 1e-14:
            best_h = h_sol.copy()
            best_c = c_sol

    # Final validation
    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)

    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            final_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 2:
            final_c = float(eval_result[0])
    except Exception:
        pass

    return best_h, final_c, len(best_h)
```
