The LSE approach: Instead of just looking at the max, we use:
s_beta = (1/beta) * log(sum_m exp(beta * p_m))

At high beta, this approximates the max. The gradient is:
∇s_beta = sum_m w_m * ∇p_m, where w_m = exp(beta*(p_m - c)) / sum_j exp(beta*(p_j - c))

This means we push down ALL lags in proportion to their relative height, which is much better than pushing down just the top few.

For the trust-region LP:
- Variables: d (length n) and t (scalar)
- Minimize t
- Subject to: p_m(h) + G[m,:]·d <= t for all m
- sum(d) = 0
- max(|d_j|) <= R (trust region, or box constraints h_j + d_j in [0,1])

This is a linear program with O(n) variables and O(n) constraints (plus bounds), solvable quickly with scipy or cvxpy.

```python
import numpy as np
from scipy.optimize import minimize, linprog
import time

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(-(1.0 - h))
        remaining = diff
        for i in range(n):
            idx = order[i]
            cap = 1.0 - h[idx]
            if cap <= 1e-15:
                continue
            add = min(remaining, cap)
            h[idx] += add
            remaining -= add
            if remaining <= 1e-14:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            if h[idx] <= 1e-15:
                continue
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-14:
                break
    return h

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def grad_row(h, m, n):
    dx = 2.0 / n
    d = m - (n - 1)
    if d == 0:
        return dx * (1.0 - 2.0 * h)
    elif d > 0:
        g = np.zeros(n)
        g[d:] += dx * (1.0 - h[:-d])
        g[:-d] -= dx * h[d:]
        return g
    else:
        p = -d
        g = np.zeros(n)
        g[:-p] += dx * (1.0 - h[p:])
        g[p:] -= dx * h[:-p]
        return g

def full_gradient(h):
    n = len(h)
    M = 2 * n - 1
    g = np.zeros(n)
    for m in range(M):
        g += grad_row(h, m, n)
    return g

def lse_weighted_gradient(h, beta):
    n = len(h)
    M = 2 * n - 1
    profile = compute_profile(h)
    c = profile.max()
    shifted = np.clip(beta * (profile - c), -500, 500)
    exp_shifted = np.exp(shifted)
    weights = exp_shifted / exp_shifted.sum()
    g = np.zeros(n)
    for m in range(M):
        w = weights[m]
        if w > 1e-12:
            g += w * grad_row(h, m, n)
    return g

def lse_descent(h, beta_schedule, max_iters_per_beta=30):
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    best_h = h.copy()
    
    for beta in beta_schedule:
        for it in range(max_iters_per_beta):
            g = lse_weighted_gradient(h, beta)
            # Project gradient to sum=0 subspace
            g = g - g.mean()
            gnorm = np.linalg.norm(g)
            if gnorm < 1e-14:
                break
            d = -g / gnorm
            # Find max step size
            pos_mask = d > 1e-12
            neg_mask = d < -1e-12
            alpha_max = 1.0
            if pos_mask.any():
                alpha_max = min(alpha_max, float(((1.0 - h[pos_mask]) / d[pos_mask]).min()))
            if neg_mask.any():
                alpha_max = min(alpha_max, float((h[neg_mask] / (-d[neg_mask])).min()))
            alpha_max = max(alpha_max, 1e-10)
            alpha_max = min(alpha_max, 0.3)
            
            # Backtracking line search
            best_alpha = 0.0
            best_val = exact_score(h)
            best_h_try = h
            for frac in [1.0, 0.8, 0.6, 0.4, 0.2, 0.1, 0.05]:
                a = alpha_max * frac
                h_try = repair_feasible(h + a * d)
                val = exact_score(h_try)
                if val < best_val - 1e-14:
                    best_val = val
                    best_alpha = a
                    best_h_try = h_try
                if val < best_val:  # should not happen but safe
                    break
            if best_alpha == 0.0:
                break
            h = best_h_try
            if best_val < best_c:
                best_c = best_val
                best_h = h.copy()
    return best_h, best_c

def trust_region_lp(h, R):
    """Solve trust-region LP for exact max."""
    n = len(h)
    M = 2 * n - 1
    profile = compute_profile(h)
    
    # Build gradient matrix
    # G[m, j] = partial p_m / partial h_j
    # We'll build it row by row
    G = np.zeros((M, n))
    for m in range(M):
        G[m, :] = grad_row(h, m, n)
    
    # Variables: [d_1, ..., d_n, t]
    # Minimize t
    c_vec = np.zeros(n + 1)
    c_vec[-1] = 1.0
    
    # Inequality constraints: G @ d - t <= -p_m
    A_ub = np.zeros((M, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -profile
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds
    bounds = []
    for j in range(n):
        lo = max(-R, -h[j])
        hi = min(R, 1.0 - h[j])
        bounds.append((lo, hi))
    bounds.append((0.0, 2.0 / n + 1.0))
    
    try:
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
        if res.success:
            d = res.x[:n]
            return h + d, R
    except Exception:
        pass
    return None, R

def trust_region_descent(h, max_iters=50, R_init=0.05, R_min=0.001):
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    best_h = h.copy()
    R = R_init
    
    for it in range(max_iters):
        h_new, R = trust_region_lp(h, R)
        if h_new is None:
            R *= 0.5
            if R < R_min:
                break
            continue
        h_new = repair_feasible(h_new)
        c_new = exact_score(h_new)
        
        if c_new < best_c - 1e-14:
            h = h_new
            best_c = c_new
            best_h = h.copy()
            # Slightly increase R on success
            R = min(R * 1.2, 0.15)
        else:
            # Try smaller step
            improved = False
            for frac in [0.5, 0.25, 0.1]:
                h_try = repair_feasible(h + frac * (h_new - h))
                c_try = exact_score(h_try)
                if c_try < best_c - 1e-14:
                    h = h_try
                    best_c = c_try
                    best_h = h.copy()
                    improved = True
                    break
            if not improved:
                R *= 0.5
                if R < R_min:
                    break
    return best_h, best_c

def interp_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    x_old = np.linspace(0, 2, n + 1)
    x_new = np.linspace(0, 2, new_n + 1)
    h_new = np.zeros(new_n)
    for k in range(new_n):
        a, b = x_new[k], x_new[k + 1]
        total = 0.0
        for j in range(n):
            lo = max(a, x_old[j])
            hi = min(b, x_old[j + 1])
            if hi > lo:
                total += h[j] * (hi - lo)
        h_new[k] = total / (b - a)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

def slsqp_epigraph(h_start, maxiter=200):
    n = len(h_start)
    dx = 2.0 / n
    M = 2 * n - 1
    c0 = exact_score(h_start)
    t0 = c0 + 1e-4
    
    def obj(x):
        return x[-1]
    
    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    
    def eq_fun(x):
        return x[:-1].sum() - n / 2.0
    
    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g
    
    def ineq_fun(x):
        h = x[:-1]
        t = x[-1]
        profile = np.correlate(h, 1 - h, mode="full") * dx
        return t - profile
    
    def ineq_jac(x):
        h = x[:-1]
        J = np.zeros((M, n + 1))
        for m in range(M):
            J[m, :n] = -grad_row(h, m, n)
        J[:, -1] = 1.0
        return J
    
    x0 = np.concatenate([h_start, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 2.0)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None and np.isfinite(res.x).all():
            h_new = repair_feasible(res.x[:-1])
            c_new = exact_score(h_new)
            if c_new <= c0 + 1e-8:
                return h_new
    except Exception:
        pass
    return None

def active_descent(h, max_iters=300, active_tol=0.003):
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    best_h = h.copy()
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    momentum_vec = np.zeros(n)
    momentum = 0.3
    
    for it in range(max_iters):
        profile = compute_profile(h)
        c = float(profile.max())
        thresh = c - active_tol
        active = np.where(profile >= thresh)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(profile))])
        if len(active) > 25:
            top = np.argsort(profile[active])[-25:]
            active = active[top]
        
        g = np.zeros(n)
        for m in active:
            g += grad_row(h, m, n)
        g -= g.mean()
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            break
        
        momentum_vec = momentum * momentum_vec + g
        mv_norm = np.linalg.norm(momentum_vec)
        if mv_norm < 1e-15:
            break
        d = -momentum_vec / mv_norm
        
        pos_mask = d > 1e-12
        neg_mask = d < -1e-12
        alpha_max = 1.0
        if pos_mask.any():
            alpha_max = min(alpha_max, float(((1.0 - h[pos_mask]) / d[pos_mask]).min()))
        if neg_mask.any():
            alpha_max = min(alpha_max, float((h[neg_mask] / (-d[neg_mask])).min()))
        alpha_max = max(alpha_max, 1e-10)
        alpha_max = min(alpha_max, 0.5)
        
        best_alpha = 0.0
        best_val = c
        best_h_try = h
        a_ls, b_ls = 0.0, alpha_max
        for _ in range(20):
            c1 = b_ls - gr * (b_ls - a_ls)
            c2 = a_ls + gr * (b_ls - a_ls)
            h1 = repair_feasible(h + c1 * d)
            h2 = repair_feasible(h + c2 * d)
            f1 = exact_score(h1)
            f2 = exact_score(h2)
            if f1 < best_val:
                best_val = f1
                best_alpha = c1
                best_h_try = h1.copy()
            if f2 < best_val:
                best_val = f2
                best_alpha = c2
                best_h_try = h2.copy()
            if f1 < f2:
                b_ls = c2
            else:
                a_ls = c1
            if b_ls - a_ls < 1e-13:
                break
        if best_val >= c - 1e-15:
            break
        h = best_h_try
        if best_val < best_c:
            best_c = best_val
            best_h = h.copy()
    return best_h, best_c
```

```python
import numpy as np
from scipy.optimize import minimize, linprog
import time

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(-(1.0 - h))
        remaining = diff
        for i in range(n):
            idx = order[i]
            cap = 1.0 - h[idx]
            if cap <= 1e-15:
                continue
            add = min(remaining, cap)
            h[idx] += add
            remaining -= add
            if remaining <= 1e-14:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            if h[idx] <= 1e-15:
                continue
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-14:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-14:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] = float(np.clip(h[idx] + diff2, 0.0, 1.0))
    return h

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def grad_row(h, m, n):
    dx = 2.0 / n
    d = m - (n - 1)
    if d == 0:
        return dx * (1.0 - 2.0 * h)
    elif d > 0:
        g = np.zeros(n)
        g[d:] += dx * (1.0 - h[:-d])
        g[:-d] -= dx * h[d:]
        return g
    else:
        p = -d
        g = np.zeros(n)
        g[:-p] += dx * (1.0 - h[p:])
        g[p:] -= dx * h[:-p]
        return g

def lse_weighted_gradient(h, beta):
    n = len(h)
    M = 2 * n - 1
    profile = compute_profile(h)
    c = profile.max()
    shifted = np.clip(beta * (profile - c), -500.0, 500.0)
    exp_shifted = np.exp(shifted)
    weights = exp_shifted / exp_shifted.sum()
    g = np.zeros(n)
    for m in range(M):
        w = weights[m]
        if w > 1e-12:
            g += w * grad_row(h, m, n)
    return g

def lse_descent(h, beta_schedule, max_iters_per_beta=25):
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    best_h = h.copy()
    for beta in beta_schedule:
        for it in range(max_iters_per_beta):
            g = lse_weighted_gradient(h, beta)
            g = g - g.mean()
            gnorm = np.linalg.norm(g)
            if gnorm < 1e-14:
                break
            d = -g / gnorm
            pos_mask = d > 1e-12
            neg_mask = d < -1e-12
            alpha_max = 1.0
            if pos_mask.any():

            if neg_mask.any():

            alpha_max = max(alpha_max, 1e-10)
            alpha_max = min(alpha_max, 0.25)
            best_alpha = 0.0

def trust_region_lp(h, R):
    n = len(h)
    M = 2 * n - 1
    profile = compute_profile(h)
    G = np.zeros((M, n))
    for m in range(M):
        G[m, :] = grad_row(h, m, n)
    c_vec = np.zeros(n + 1)
    c_vec[-1] = 1.0
    A_ub = np.zeros((M, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -profile
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for j in range(n):
        lo = max(-R, -h[j])
        hi = min(R, 1.0 - h[j])
        bounds.append((lo, hi))
    bounds.append((0.0, 2.0 / n + 0.5))
    try:
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
        if res.success:
            d = res.x[:n]
            return h + d
    except Exception:
        pass
    return None

    best_h = h.copy()

            best_c = c_new
            best_h = h.copy()
            R = min(R * 1.15, 0.12)

def active_descent(h, max_iters=300, active_tol=0.003):

    h = h.copy()
    best_c = exact_score(h)
    best_h = h.copy()

    momentum_vec = np.zeros(n)
    momentum = 0.3
    for it in range(max_iters):
        profile = compute_profile(h)
        c = float(profile.max())
        thresh = c - active_tol

        if len(active) == 0:

        if len(active) > 25:

            active = active[top]

            g += grad_row(h, m, n)
        g -= g.mean()

        if gnorm < 1e-15:
            break

        mv_norm = np.linalg.norm(momentum_vec)
        if mv_norm < 1e-15:
            break

        pos_mask = d > 1e-12
        neg_mask = d < -1e-12
        alpha_max = 1.0
        if pos_mask.any():
            alpha_max = min(alpha_max, float(((1.0 - h[pos_mask]) / d[pos_mask]).min()))
        if neg_mask.any():

        alpha_max = max(alpha_max, 1e-10)

            c2 = a_ls + gr * (b_ls - a_ls)

            h2 = repair_feasible(h + c2 * d)
            f1 = exact_score(h1)
            f2 = exact_score(h2)
            if f1 < best_val:
                best_val = f1
                best_alpha = c1
                best_h_try = h1.copy()
            if f2 < best_val:
                best_val = f2
                best_alpha = c2
                best_h_try = h2.copy()
            if f1 < f2:
                b_ls = c2
            else:
                a_ls = c1
            if b_ls - a_ls < 1e-13:
                break

            break

        if best_val < best_c:

            best_h = h.copy()
    return best_h, best_c

def interp_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    x_old = np.linspace(0, 2, n + 1)
    x_new = np.linspace(0, 2, new_n + 1)
    h_new = np.zeros(new_n)
    for k in range(new_n):
        a, b = x_new[k], x_new[k + 1]
        total = 0.0
        for j in range(n):
            lo = max(a, x_old[j])
            hi = min(b, x_old[j + 1])
            if hi > lo:
                total += h[j] * (hi - lo)
        h_new[k] = total / (b - a)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

def slsqp_epigraph(h_start, maxiter=150):
    n = len(h_start)
    dx = 2.0 / n
    M = 2 * n - 1
    c0 = exact_score(h_start)
    t0 = c0 + 1e-4

    def obj(x):
        return x[-1]

    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g

    def eq_fun(x):
        return x[:-1].sum() - n / 2.0

    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g

    def ineq_fun(x):
        h = x[:-1]
        t = x[-1]
        profile = np.correlate(h, 1 - h, mode="full") * dx
        return t - profile

    def ineq_jac(x):
        h = x[:-1]
        J = np.zeros((M, n + 1))
        for m in range(M):
            J[m, :n] = -grad_row(h, m, n)
        J[:, -1] = 1.0
        return J

    x0 = np.concatenate([h_start, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 2.0)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None and np.isfinite(res.x).all():
            h_new = repair_feasible(res.x[:-1])
            c_new = exact_score(h_new)
            if c_new <= c0 + 1e-8:
                return h_new
    except Exception:
        pass
    return None

def structured_start(n, seed):
    rng = np.random.RandomState(seed)
    half = n // 2
    patterns = []
    # Pattern 0: golden ratio
    idx = np.arange(n)
    vals = (idx * 0.618033988749895) % 1.0
    top = np.argsort(vals)[-half:]
    h0 = np.zeros(n)
    h0[top] = 1.0
    patterns.append(h0)
    # Pattern 1: alternating blocks
    period = max(4, n // 7)
    h1 = np.zeros(n)
    for i in range(n):
        if (i // period) % 2 == 0:
            h1[i] = 1.0
    ones = int(h1.sum())
    if ones > half:
        one_idx = np.where(h1 == 1.0)[0]
        to_flip = rng.choice(one_idx, ones - half, replace=False)
        h1[to_flip] = 0.0
    elif ones < half:
        zero_idx = np.where(h1 == 0.0)[0]
        to_flip = rng.choice(zero_idx, half - ones, replace=False)
        h1[to_flip] = 1.0
    patterns.append(h1)
    # Pattern 2: sinusoidal
    h2 = 0.5 + 0.4 * np.sin(2 * np.pi * np.arange(n) / n * 3)
    h2 = np.clip(h2, 0.0, 1.0)
    h2 = repair_feasible(h2)
    patterns.append(h2)
    # Pattern 3: two-frequency
    h3 = 0.5 + 0.3 * np.sin(np.arange(n) * 2 * np.pi / n * 2) + 0.15 * np.sin(np.arange(n) * 2 * np.pi / n * 5)
    h3 = np.clip(h3, 0.0, 1.0)
    h3 = repair_feasible(h3)
    patterns.append(h3)
    return patterns

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s - (time.perf_counter() - t_start)

    try:
        parent = np.array(initial_h_values, dtype=np.float64)
    except Exception:
        parent = None
    if parent is None or len(parent) < 4:
        parent = np.full(70, 0.5)
    parent = np.clip(parent, 0.0, 1.0)
    parent = repair_feasible(parent)
    n_parent = len(parent)

    best_h = parent.copy()
    best_c = exact_score(parent)

    # Stage 1: LSE annealing
    if time_left() > 80:
        beta_schedule = [15, 30, 60, 120, 250, 500, 1000, 2000, 4000]
        h_lse, c_lse = lse_descent(best_h, beta_schedule, max_iters_per_beta=20)
        if c_lse < best_c:
            best_h, best_c = h_lse, c_lse

    # Stage 2: Active descent
    if time_left() > 50:
        h_ad, c_ad = active_descent(best_h, max_iters=300, active_tol=0.003)
        if c_ad < best_c:
            best_h, best_c = h_ad, c_ad

    # Stage 3: Trust-region LP
    if time_left() > 30:
        h_tr, c_tr = trust_region_descent(best_h, max_iters=50, R_init=0.05, R_min=0.002)
        if c_tr < best_c:
            best_h, best_c = h_tr, c_tr

    # Stage 4: Combined LSE + active descent at high beta
    if time_left() > 60:
        h_lse2, c_lse2 = lse_descent(best_h, [500, 1000, 2000, 5000, 10000], max_iters_per_beta=15)
        if c_lse2 < best_c:
            best_h, best_c = h_lse2, c_lse2
        if time_left() > 30:
            h_ad2, c_ad2 = active_descent(best_h, max_iters=200, active_tol=0.002)
            if c_ad2 < best_c:
                best_h, best_c = h_ad2, c_ad2

    # Stage 5: Trust region again
    if time_left() > 25:
        h_tr2, c_tr2 = trust_region_descent(best_h, max_iters=40, R_init=0.025, R_min=0.001)
        if c_tr2 < best_c:
            best_h, best_c = h_tr2, c_tr2

    # Stage 6: Dimension exploration
    n_best = len(best_h)
    dims_to_try = []
    for dn in [-6, -4, -2, 2, 4, 6, 8, 10, 12, 14, 16, 20]:
        d = n_best + dn
        if 40 <= d <= 160 and d % 2 == 0:
            dims_to_try.append(d)
    dims_to_try.sort(key=lambda x: abs(x - n_best))

    for new_n in dims_to_try[:8]:
        if time_left() < 20:
            break
        h_interp = interp_to_n(best_h, new_n)
        h_ad, c_ad = active_descent(h_interp, max_iters=200, active_tol=0.003)
        if c_ad < best_c:
            best_h, best_c = h_ad, c_ad
            if time_left() > 15:
                h_lse_d, c_lse_d = lse_descent(best_h, [200, 500, 1000], max_iters_per_beta=10)
                if c_lse_d < best_c:
                    best_h, best_c = h_lse_d, c_lse_d
                h_tr_d, c_tr_d = trust_region_descent(best_h, max_iters=20, R_init=0.03)
                if c_tr_d < best_c:
                    best_h, best_c = h_tr_d, c_tr_d

    # Stage 7: Structured starts in best dimension
    n_cur = len(best_h)
    if time_left() > 80:
        patterns = structured_start(n_cur, seed)
        for pi in range(len(patterns)):
            if time_left() < 15:
                break
            h_pat = repair_feasible(patterns[pi])
            # Smooth mix
            h_mix = 0.5 * h_pat + 0.5 * best_h
            h_mix = np.clip(h_mix, 0.0, 1.0)
            h_mix = repair_feasible(h_mix)
            h_ad, c_ad = active_descent(h_mix, max_iters=150, active_tol=0.004)
            if c_ad < best_c:
                best_h, best_c = h_ad, c_ad
            else:
                # Try pure binary smoothed
                h_smooth = 0.3 * h_pat + 0.7 * np.full(n_cur, 0.5)
                h_smooth = np.clip(h_smooth, 0.0, 1.0)
                h_smooth = repair_feasible(h_smooth)
                h_ad2, c_ad2 = active_descent(h_smooth, max_iters=150, active_tol=0.004)
                if c_ad2 < best_c:
                    best_h, best_c = h_ad2, c_ad2

    # Stage 8: SLSQP polish
    if time_left() > 30:
        h_sol = slsqp_epigraph(best_h, maxiter=150)
        if h_sol is not None:
            c_sol = exact_score(h_sol)
            if c_sol < best_c - 1e-12:
                best_h, best_c = h_sol, c_sol

    # Stage 9: Final LSE + active descent
    if time_left() > 30:
        h_lse3, c_lse3 = lse_descent(best_h, [2000, 5000, 10000], max_iters_per_beta=15)
        if c_lse3 < best_c:
            best_h, best_c = h_lse3, c_lse3
        h_ad3, c_ad3 = active_descent(best_h, max_iters=300, active_tol=0.001)
        if c_ad3 < best_c:
            best_h, best_c = h_ad3, c_ad3

    # Stage 10: Final trust region
    if time_left() > 15:
        h_tr3, c_tr3 = trust_region_descent(best_h, max_iters=30, R_init=0.015, R_min=0.0005)
        if c_tr3 < best_c:
            best_h, best_c = h_tr3, c_tr3

    # Stage 11: Final SLSQP
    if time_left() > 20:
        h_sol = slsqp_epigraph(best_h, maxiter=100)
        if h_sol is not None:
            c_sol = exact_score(h_sol)
            if c_sol < best_c:
                best_h, best_c = h_sol, c_sol

    # Final active descent with very tight tolerance
    if time_left() > 10:
        h_final, c_final = active_descent(best_h, max_iters=200, active_tol=0.0005)
        if c_final < best_c:
            best_h, best_c = h_final, c_final

    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)

    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            final_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 2:
            final_c = float(eval_result[0])
    except Exception:
        pass

    return best_h, final_c, len(best_h)
```
