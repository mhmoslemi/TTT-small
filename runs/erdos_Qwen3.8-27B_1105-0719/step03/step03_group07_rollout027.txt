    """Minimize max(P + t*g) over t in [lo, hi] using ternary search."""
    def f(t):
        return np.max(P + t * g)
    
    for _ in range(n_iter):
        m1 = lo + (hi - lo) / 3
        m2 = hi - (hi - lo) / 3
        if f(m1) < f(m2):
            hi = m2
        else:
            lo = m1
    t = (lo + hi) / 2
    return t, f(t)

import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full"))) * dx

def profile_h(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def build_all_gradients(h):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    G = np.zeros((M, n))
    for m in range(M):
        d = m - (n - 1)
        if d == 0:
            G[m, :] = dx * (1.0 - 2.0 * h)
        else:
            dp = abs(d)
            G[m, dp:] += dx * (1.0 - h[:n-dp])
            G[m, :n-dp] -= dx * h[dp:]
    return G

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx])
            h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    return h

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    return repair_feasible(np.clip(h_new, 0.0, 1.0))

    """One SLP iteration: solve LP and return proposed h_new."""
    n = len(h)
    P = profile_h(h)
    G = build_all_gradients(h)
    M = len(P)
    
    rhs = -P + G @ h
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    A = np.hstack([G, -np.eye(M)])
    b = rhs
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    bounds = [(0, 1)] * n + [(0, 1.0)]
    
    res = linprog(c, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.status == 0:
        return res.x[:n]
    return None

            c_try = exact_score(h_try)
            if c_try < best_c - 1e-15:
                h = h_try
                best_c = c_try
                improved = True
                break
        
        if not improved:
            break
    
    return h, best_c

            lo = max(-h[i], h[j] - 1.0)

                if f1 < f2:

                    a = m1

            val = np.max(P + delta * g_diff)

            improve = P.max() - val

        h = repair_feasible(h)

            h[i] -= best_delta

        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx])
            h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    return repair_feasible(np.clip(h_new, 0.0, 1.0))

def slp_solve(h):
    n = len(h)
    P = profile_h(h)
    G = build_all_gradients(h)
    M = len(P)
    rhs = -P + G @ h
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A = np.hstack([G, -np.eye(M)])
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    res = linprog(c, A_ub=A, b_ub=rhs, A_eq=A_eq, b_eq=[n / 2.0],
                  bounds=bounds, method='highs')
    if res.status == 0:
        return res.x[:n], float(res.fun)
    return None, None

def slp_minimax(h, max_iters=300, t_start=None, budget_s=1100):
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    stall = 0
    for it in range(max_iters):
        if t_start is not None and time.perf_counter() - t_start > budget_s - 15:
            break
        h_new, t_lp = slp_solve(h)
        if h_new is None:
            break
        improved = False
        P = profile_h(h)
        cur_max = P.max()
        if t_lp >= cur_max - 1e-12:
            stall += 1
            if stall > 20:
                break
            continue
        for alpha in [1.0, 0.7, 0.5, 0.3, 0.1, 0.05, 0.01]:
            h_try = repair_feasible(np.clip(h + alpha * (h_new - h), 0, 1))
            c_try = exact_score(h_try)
            if c_try < best_c - 1e-15:
                h = h_try
                best_c = c_try
                improved = True
                stall = 0
                break
        if not improved:
            stall += 1
            if stall > 30:
                break
    return h, best_c

def stochastic_descent(h, max_iters=200, t_start=None, budget_s=1100):
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    rng = np.random.RandomState(123)
    step = 0.05
    for it in range(max_iters):
        if t_start is not None and time.perf_counter() - t_start > budget_s - 15:
            break
        P = profile_h(h)
        m_star = np.argmax(P)
        G_star = build_all_gradients(h)[m_star]
        # Move in direction -G_star (projected)
        # But we need to preserve sum, so center the gradient
        G_centered = G_star - G_star.mean()
        h_try = repair_feasible(np.clip(h - step * G_centered, 0, 1))
        c_try = exact_score(h_try)
        if c_try < best_c - 1e-15:
            h = h_try
            best_c = c_try
            step = min(step * 1.1, 0.2)
        else:
            step *= 0.7
            if step < 1e-6:
                break
    return h, best_c

def pair_polish(h, max_iters=100, t_start=None, budget_s=1100):
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    for it in range(max_iters):
        if t_start is not None and time.perf_counter() - t_start > budget_s - 15:
            break
        P = profile_h(h)
        G = build_all_gradients(h)
        cur_max = P.max()
        m_star = int(np.argmax(P))
        G_star = G[m_star]
        # Find best pair that reduces P[m_star]
        pos = np.where(G_star > 0.005)[0]
        neg = np.where(G_star < -0.005)[0]
        if len(pos) == 0 or len(neg) == 0:
            break
        best_improve = 1e-12
        best_move = None
        # Sample up to 100 pairs
        pairs = []
        for i in pos[:15]:
            for j in neg[:15]:
                pairs.append((i, j))
        rng_local = np.random.RandomState(it)
        if len(pairs) > 200:
            idx = rng_local.choice(len(pairs), 200, replace=False)
            pairs = [pairs[k] for k in idx]
        for (i, j) in pairs:
            g_diff = G[:, i] - G[:, j]
            lo = max(-h[i], h[j] - 1.0)
            hi = min(1.0 - h[i], h[j])
            if hi <= lo + 1e-10:
                continue
            # Quick ternary search
            a, b = lo, hi
            for _ in range(30):
                m1 = a + (b - a) / 3.0
                m2 = b - (b - a) / 3.0
                if np.max(P + m1 * g_diff) < np.max(P + m2 * g_diff):
                    b = m2
                else:
                    a = m1
            delta = (a + b) / 2.0
            new_max = np.max(P + delta * g_diff)
            improve = cur_max - new_max
            if improve > best_improve:
                best_improve = improve
                best_move = (i, j, delta)
        if best_move is None:
            break
        i, j, delta = best_move
        h[i] += delta
        h[j] -= delta
        h = repair_feasible(h)
        c_new = exact_score(h)
        if c_new >= best_c - 1e-15:
            break
        best_c = c_new
    return h, best_c
