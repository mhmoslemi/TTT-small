    # d = 0

Actually, since each corr_d changes linearly in δ, the maximum of all corr_d + δ * coeff_d is a piecewise linear function of δ. The optimal δ that minimizes the max is where two or more correlations cross.

This is actually a 1D minimax problem! We can solve it by finding the δ that minimizes max_d (corr_d + δ * c_d) where c_d are the coefficients above.

This is equivalent to: find δ that minimizes max_d (a_d + δ * b_d) where a_d = corr_d and b_d = c_d.

This can be solved by binary search on the target value T: for a given T, we need corr_d + δ*c_d <= T for all d, which gives:
- If c_d > 0: δ <= (T - corr_d) / c_d
- If c_d < 0: δ >= (T - corr_d) / c_d
- If c_d = 0: need corr_d <= T

The optimal δ is the one that minimizes T subject to these constraints AND the box constraints (0 <= h[i]+δ <= 1, 0 <= h[j]-δ <= 1).

This is a 1D feasibility problem that can be solved by binary search on T, and for each T, the feasible δ interval is [max lower bounds, min upper bounds].

```python
def pairwise_move(h, i, j, n):
    """Try to improve by moving δ from j to i (h[i]+=δ, h[j]-=δ)"""
    # Compute coefficients for all lags
    # For lag d (offset), coefficient is:
    # (1-h[i+d])*[valid] - h[i-d]*[valid] - (1-h[j+d])*[valid] + h[j-d]*[valid]
    
    # Current correlations
    corr = np.correlate(h, 1-h, mode='full')
    K = len(corr)  # 2n-1
    
    # For each lag k (0 to K-1), offset d = k - (n-1)
    # coeff[k] = contribution to corr[k] per unit δ
    # This is a vector we can compute
    
    # Actually, let's compute it directly
    idx = np.arange(K) - (n-1)  # offsets d
    # For position i: contributes (1-h[i+d]) if 0<=i+d<n, and -h[i-d] if 0<=i-d<n
    # For position j: contributes -(1-h[j+d]) if 0<=j+d<n, and +h[j-d] if 0<=j-d<n
    
    # coeff for each d:
    # part_i(d) = (1-h[i+d]) * [0<=i+d<n] - h[i-d] * [0<=i-d<n]
    # part_j(d) = -(1-h[j+d]) * [0<=j+d<n] + h[j-d] * [0<=j-d<n]
    # coeff(d) = part_i(d) + part_j(d)
    
    # Compute efficiently
    d = np.arange(-(n-1), n)  # all offsets
    # part_i
    part_i = np.zeros(len(d))
    valid1 = (i + d >= 0) & (i + d < n)
    part_i[valid1] += (1 - h[i + d[valid1]])
    valid2 = (i - d >= 0) & (i - d < n)
    part_i[valid2] -= h[i - d[valid2]]
    
    # part_j
    part_j = np.zeros(len(d))
    valid3 = (j + d >= 0) & (j + d < n)
    part_j[valid3] -= (1 - h[j + d[valid3]])
    valid4 = (j - d >= 0) & (j - d < n)
    part_j[valid4] += h[j - d[valid4]]
    
    coeff = part_i + part_j
    
    # Now find δ that minimizes max(corr + δ*coeff)
    # subject to: 0 <= h[i]+δ <= 1 and 0 <= h[j]-δ <= 1
    # i.e., -h[i] <= δ <= 1-h[i] and h[j]-1 <= δ <= h[j]
    d_lo = max(-h[i], h[j]-1)
    d_hi = min(1-h[i], h[j])
    
    if d_lo >= d_hi:
        return 0, 0  # no feasible move
    
    # 1D minimax: minimize max_k (corr[k] + δ*coeff[k]) for δ in [d_lo, d_hi]
    # The function f(δ) = max_k (corr[k] + δ*coeff[k]) is convex in δ
    # (max of linear functions is convex)
    # So we can use ternary search or find where subgradient crosses 0.
    
    # Simple approach: evaluate at a grid of points
    # Or use the fact that optimum is at a crossing of two lines
    # For now, use a fine grid
    n_grid = 200
    delta_grid = np.linspace(d_lo, d_hi, n_grid)
    values = np.max(corr[None, :] + delta_grid[:, None] * coeff[None, :], axis=1)
    best_idx = np.argmin(values)
    best_delta = delta_grid[best_idx]
    
    return best_delta, values[best_idx]
```

3. If the subgradient (sum of coeff[k] for active k) crosses 0, find the crossing
4. Otherwise, move to the boundary and repeat

import numpy as np
import time

def project_box_sum(x, target):
    n = len(x)
    x = np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = x.min() - 2.0
    hi = x.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target:
            lo = mid
        else:
            hi = mid
    s = (lo + hi) * 0.5
    h = np.clip(x - s, 0.0, 1.0)
    err = target - h.sum()
    if abs(err) > 1e-12:
        if err > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if mask.sum() > 0:
            h[mask] += err / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def c5_of_h(h, dx):
    return np.correlate(h, 1.0 - h, mode='full').max() * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def softmax_weights(corr, alpha, rng=None):
    scaled = corr / alpha
    m = scaled.max()
    p = np.exp(scaled - m)
    if rng is not None:
        noise = 1.0 + 0.03 * rng.standard_normal(len(p))
        p = p * noise
        p = np.clip(p, 1e-10, None)
    s = p.sum()
    if s < 1e-30:
        return np.ones(len(corr)) / len(corr)
    return p / s

def weighted_grad(h, w, dx):
    n = len(h)
    grad = np.zeros(n)
    w0 = w[n - 1]
    if w0 > 1e-15:
        grad += w0 * (1.0 - 2.0 * h)
    for d in range(1, n):
        wp = w[n - 1 + d]
        if wp > 1e-15:
            grad[:n - d] += wp * (1.0 - h[d:])
            grad[d:] -= wp * h[:n - d]
        wn = w[n - 1 - d]
        if wn > 1e-15:
            grad[d:] += wn * (1.0 - h[:n - d])
            grad[:n - d] -= wn * h[d:]
    return dx * grad

def pairwise_move(h, i, j, n, corr):
    K = len(corr)
    d = np.arange(-(n - 1), n)
    coeff = np.zeros(K)
    v1 = (i + d >= 0) & (i + d < n)
    coeff[v1] += (1.0 - h[i + d[v1]])
    v2 = (i - d >= 0) & (i - d < n)
    coeff[v2] -= h[i - d[v2]]
    v3 = (j + d >= 0) & (j + d < n)
    coeff[v3] -= (1.0 - h[j + d[v3]])
    v4 = (j - d >= 0) & (j - d < n)
    coeff[v4] += h[j - d[v4]]
    d_lo = max(-h[i], h[j] - 1.0)
    d_hi = min(1.0 - h[i], h[j])
    if d_lo >= d_hi - 1e-12:
        return 0.0, corr.max()
    G = 60
    deltas = np.linspace(d_lo, d_hi, G)
    vals = (corr[None, :] + deltas[:, None] * coeff[None, :]).max(axis=1)
    bi = np.argmin(vals)
    return deltas[bi], vals[bi]

def subgradient_phase(h, dx, target, rng, budget_s, t0, n_steps=400):
    n = len(h)
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    alpha = 0.08
    lr = 0.15
    momentum = np.zeros(n)
    beta = 0.75
    for step in range(n_steps):
        if budget_s - (time.time() - t0) < 20:
            break
        corr = compute_corr(best_h)
        w = softmax_weights(corr, alpha, rng)
        g = weighted_grad(best_h, w, dx)
        g -= g.mean()
        momentum = beta * momentum + (1.0 - beta) * g
        h_new = project_box_sum(best_h + lr * momentum, target)
        c5_new = c5_of_h(h_new, dx)
        if c5_new < best_c5 - 1e-12:
            best_h = h_new
            best_c5 = c5_new
            lr = min(0.4, lr * 1.01)
        else:
            lr *= 0.9
        alpha = max(0.005, alpha * 0.985)
    return best_h, best_c5

def coordinate_phase(h, dx, target, rng, budget_s, t0, n_iter=300, top_k=8):
    n = len(h)
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    for it in range(n_iter):
        if budget_s - (time.time() - t0) < 12:
            break
        corr = compute_corr(best_h)
        max_val = corr.max()
        w = softmax_weights(corr, 0.04, rng)
        g = weighted_grad(best_h, w, dx)
        g -= g.mean()
        inc = np.argsort(g)[:top_k]
        dec = np.argsort(g)[::-1][:top_k]
        best_move = None
        best_val = max_val
        for i in inc:
            for j in dec:
                if i == j:
                    continue
                delta, val = pairwise_move(best_h, i, j, n, corr)
                if val < best_val - 1e-12:
                    best_val = val
                    best_move = (i, j, delta)
        if best_move is None:
            inc_b = np.argsort(g)[:3 * top_k]
            dec_b = np.argsort(g)[::-1][top_k:4 * top_k]
            for i in inc_b:
                for j in dec_b:
                    if i == j:
                        continue
                    delta, val = pairwise_move(best_h, i, j, n, corr)
                    if val < best_val - 1e-12:
                        best_val = val
                        best_move = (i, j, delta)
            if best_move is None:
                if rng.random() < 0.3 and n > 10:
                    i = rng.randint(n)
                    j = rng.randint(n)
                    if i != j:
                        delta, val = pairwise_move(best_h, i, j, n, corr)
                        if val < best_val - 1e-12:
                            best_move = (i, j, delta)
            if best_move is None:
                break
        i, j, delta = best_move
        best_h[i] += delta
        best_h[j] -= delta
        best_h = project_box_sum(best_h, target)
        new_c5 = c5_of_h(best_h, dx)
        if new_c5 < best_c5:
            best_c5 = new_c5
    return best_h, best_c5

def binary_local_search(h, dx, target, rng, budget_s, t0, n_rounds=8):
    n = len(h)
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    for rnd in range(n_rounds):
        if budget_s - (time.time() - t0) < 12:
            break
        cur = (best_h > 0.5).astype(float)
        cur = project_box_sum(cur, target)
        cur_c5 = c5_of_h(cur, dx)
        if cur_c5 > best_c5 + 0.005:
            continue
        ones = np.where(cur > 0.5)[0]
        zeros = np.where(cur < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            continue
        improved = True
        max_sweeps = 20
        while improved and max_sweeps > 0:
            improved = False
            max_sweeps -= 1
            if budget_s - (time.time() - t0) < 10:
                break
            order_i = np.zeros(len(ones), dtype=int)
            order_i[:] = ones
            rng.shuffle(order_i)
            order_j = np.zeros(len(zeros), dtype=int)
            order_j[:] = zeros
            rng.shuffle(order_j)
            for i in order_i[:min(80, len(order_i))]:
                if budget_s - (time.time() - t0) < 10:
                    break
                for j in order_j[:min(80, len(order_j))]:
                    cur[i] = 0.0
                    cur[j] = 1.0
                    val = c5_of_h(cur, dx)
                    if val < cur_c5 - 1e-10:
                        cur_c5 = val
                        improved = True
                        if val < best_c5:
                            best_c5 = val
                            best_h = cur.copy()
                        ones = np.where(cur > 0.5)[0]
                        zeros = np.where(cur < 0.5)[0]
                        break
                    else:
                        cur[i] = 1.0
                        cur[j] = 0.0
    return best_h, best_c5

def transfer_grid(h_src, n_target):
    n_src = len(h_src)
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_tgt = (np.arange(n_target) + 0.5) * (2.0 / n_target)
    return np.interp(x_tgt, x_src, h_src)

def perturb_reopt(h, dx, target, rng, n_rounds, budget_s, t0):
    n = len(h)
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    for r in range(n_rounds):
        if budget_s - (time.time() - t0) < 20:
            break
        scale = max(0.02, 0.08 / (r + 1))
        pert = rng.standard_normal(n)
        pert -= pert.mean()
        h_p = project_box_sum(h + scale * pert, target)
        h_sm, c5_sm = subgradient_phase(h_p, dx, target, rng, budget_s, t0, n_steps=100)
        h_cd, c5_cd = coordinate_phase(h_sm, dx, target, rng, budget_s, t0, n_iter=80, top_k=6)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd
            h = h_cd.copy()
    return best_h, best_c5
