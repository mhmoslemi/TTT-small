import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full"))) * dx

def profile_h(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx])
            h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    return h

def pairwise_sweep(h, rng, max_pairs=None, t_start=None, budget_s=1100, margin=30):
    """Do one full pairwise sweep in random order."""
    n = len(h)
    pairs = [(i, j) for i in range(n) for j in range(i+1, n)]
    rng.shuffle(pairs)
    if max_pairs:
        pairs = pairs[:max_pairs]
    
    c0 = exact_score(h)
    improved_count = 0
    
    for p, q in pairs:
        if t_start is not None and time.perf_counter() - t_start > budget_s - margin:
            break
        c_val = h[p] + h[q]
        lo = max(0.0, c_val - 1.0)
        hi = min(1.0, c_val)
        if hi - lo < 1e-10:
            continue
        
        h_work = h.copy()
        
        def f_obj(t):
            h_work[p] = t
            h_work[q] = c_val - t
            return exact_score(h_work)
        
        try:
            res = minimize_scalar(f_obj, bounds=(lo, hi), method='bounded',
                                  options={'xatol': 1e-8, 'maxiter': 30})
            if res.fun < c0 - 1e-10:
                h[p] = res.x
                h[q] = c_val - res.x
                h = np.clip(h, 0.0, 1.0)
                c0 = exact_score(h)
                improved_count += 1
        except Exception:
            pass
    
    return h, c0, improved_count

def slsqp_active(h0, maxiter=300, active_k=12, tol=1e-11, t_start=None, budget_s=1100, margin=30):
    """SLSQP with top-K active constraints."""
    n = len(h0)
    dx = 2.0 / n
    M = 2 * n - 1
    P = profile_h(h0)
    c0 = P.max()
    
    thresh = c0 - max(0.003, 0.05 * (c0 - np.median(P)))
    active = np.where(P >= thresh)[0]
    if len(active) > active_k:
        active = active[np.argsort(P[active])[-active_k:]]
    
    def obj(x):
        return x[-1]
    
    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    
    def eq_fun(x):
        return x[:-1].sum() - n / 2.0
    
    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g
    
    def ineq_fun(x):
        h = x[:-1]
        t = x[-1]
        prof = profile_h(h)
        return t - prof[active]
    
    def ineq_jac(x):
        h = x[:-1]
        dh = x[:-1]
        # Gradient of P[m] w.r.t. h
        G = np.zeros((len(active), n))
        for idx, m in enumerate(active):
            d = m - (n - 1)
            if d == 0:
                G[idx, :] = dx * (1.0 - 2.0 * dh)
            elif d > 0:
                for i in range(n - d):
                    G[idx, i + d] += dx * (1.0 - dh[i])
                    G[idx, i] -= dx * dh[i + d]
            else:
                dp = -d
                for j in range(n - dp):
                    G[idx, j + dp] += dx * (1.0 - dh[j])
                    G[idx, j] -= dx * dh[j + dp]
        J = np.zeros((len(active), n + 1))
        J[:, :-1] = -G
        J[:, -1] = 1.0
        return J
    
    x0 = np.concatenate([h0, [c0 + 1e-4]])
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 1.0)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': tol})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = repair_feasible(res.x[:-1])
            c_new = exact_score(h_new)
            if c_new <= c0 + 1e-10:
                return h_new
    except Exception:
        pass
    return None

def symmetrize(h):
    return (h + h[::-1].copy()) / 2.0

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    return repair_feasible(np.clip(h_new, 0.0, 1.0))

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full"))) * dx

def profile_h(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx])
            h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    return h

def pairwise_sweep(h, rng, t_start=None, budget_s=1100, margin=40, max_sweeps=1, tol=1e-10):
    """Multiple pairwise sweeps with random ordering."""
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
    
    for sweep in range(max_sweeps):
        if t_start is not None and time.perf_counter() - t_start > budget_s - margin:
            break
        rng.shuffle(pairs)
        improved_any = False
        for p, q in pairs:
            if t_start is not None and time.perf_counter() - t_start > budget_s - margin:
                break
            c_val = h[p] + h[q]
            lo = max(0.0, c_val - 1.0)
            hi = min(1.0, c_val)
            if hi - lo < 1e-10:
                continue
            h_work = h.copy()
            
            def f_obj(t):
                h_work[p] = t
                h_work[q] = c_val - t
                return exact_score(h_work)
            
            try:
                res = minimize_scalar(f_obj, bounds=(lo, hi), method='bounded',
                                      options={'xatol': 1e-9, 'maxiter': 35})
                if res.fun < best_c - tol:
                    h[p] = max(0.0, min(1.0, res.x))
                    h[q] = max(0.0, min(1.0, c_val - res.x))
                    best_c = exact_score(h)
                    improved_any = True
            except Exception:
                pass
        if not improved_any:
            break
    return h, best_c

def slsqp_active(h0, maxiter=300, active_k=15, tol=1e-12, t_start=None, budget_s=1100, margin=30):
    n = len(h0)
    dx = 2.0 / n
    M = 2 * n - 1
    P = profile_h(h0)
    c0 = P.max()
    
    thresh = c0 - max(0.002, 0.04 * (c0 - np.median(P)))
    active = np.where(P >= thresh)[0]
    if len(active) > active_k:
        active = active[np.argsort(P[active])[-active_k:]]
    
    def obj(x):
        return x[-1]
    
    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    
    def eq_fun(x):
        return x[:-1].sum() - n / 2.0
    
    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g
    
    def ineq_fun(x):
        h = x[:-1]
        t = x[-1]
        prof = profile_h(h)
        return t - prof[active]
    
    def ineq_jac(x):
        dh = x[:-1]
        G = np.zeros((len(active), n))
        for idx, m in enumerate(active):
            d = m - (n - 1)
            if d == 0:
                G[idx, :] = dx * (1.0 - 2.0 * dh)
            elif d > 0:
                for i in range(n - d):
                    G[idx, i + d] += dx * (1.0 - dh[i])
                    G[idx, i] -= dx * dh[i + d]
            else:
                dp = -d
                for j in range(n - dp):
                    G[idx, j + dp] += dx * (1.0 - dh[j])
                    G[idx, j] -= dx * dh[j + dp]
        J = np.zeros((len(active), n + 1))
        J[:, :-1] = -G
        J[:, -1] = 1.0
        return J
    
    x0 = np.concatenate([h0, [c0 + 1e-4]])
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 1.0)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    if t_start is not None and time.perf_counter() - t_start > budget_s - margin:
        return None

    except Exception:
        pass

def symmetrize(h):
    return (h + h[::-1].copy()) / 2.0

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    return repair_feasible(np.clip(h_new, 0.0, 1.0))

def softmax_direction(h, beta, lam=100.0, top_k=20):
    """Compute a descent direction using top-k softmax of profile."""
    n = len(h)
    dx = 2.0 / n
    P = profile_h(h)
    Pmax = P.max()
    exp_arg = np.clip(beta * (P - Pmax), -50, 0)
    weights = np.exp(exp_arg)
    S = weights.sum()
    if S < 1e-300:
        weights = np.zeros(len(P))
        weights[np.argmax(P)] = 1.0
    else:
        weights /= S
    active = np.argsort(weights)[-top_k:]
    G = np.zeros((len(active), n))
    for idx, m in enumerate(active):
        d = m - (n - 1)
        if d == 0:
            G[idx, :] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            for i in range(n - d):
                G[idx, i + d] += dx * (1.0 - h[i])
                G[idx, i] -= dx * h[i + d]
        else:
            dp = -d
            for j in range(n - dp):
                G[idx, j + dp] += dx * (1.0 - h[j])
                G[idx, j] -= dx * h[j + dp]
    w_active = weights[active]
    w_active = w_active / w_active.sum()
    grad = G.T @ w_active
    sum_err = h.sum() - n / 2.0
    grad += 2.0 * lam * sum_err / n
    return grad

def projected_subgrad(h, t_start, budget_s, margin=40):
    """Projected subgradient with momentum and backtracking."""
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    vel = np.zeros(n)
    momentum = 0.5
    
    for beta in [30, 60, 120, 240, 480]:
        alpha = 0.03
        for it in range(40):
            if t_start is not None and time.perf_counter() - t_start > budget_s - margin:
                return h, best_c
            grad = softmax_direction(h, beta, lam=80.0, top_k=15)
            grad -= grad.mean()
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break
            grad /= gnorm
            vel = momentum * vel + grad
            vnorm = np.linalg.norm(vel)
            if vnorm > 0:
                vel /= vnorm
            improved = False
            for a in [alpha, alpha*0.5, alpha*0.25, alpha*0.1, alpha*0.05]:
                h_try = repair_feasible(h - a * vel)
                s = exact_score(h_try)
                if s < best_c - 1e-13:
                    h = h_try
                    best_c = s
                    improved = True
                    break
                elif s < exact_score(h) - 1e-14:
                    h = h_try
                    best_c = min(best_c, s)
            if improved:
                alpha = min(alpha * 1.03, 0.06)
            else:
                alpha *= 0.6
                if alpha < 1e-8:
                    break
    return h, best_c
