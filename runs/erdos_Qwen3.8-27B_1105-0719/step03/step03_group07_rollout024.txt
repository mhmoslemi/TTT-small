import numpy as np
from scipy.optimize import linprog, minimize
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full"))) * dx

def compute_profile(h):
    """Compute R(d) for all lags."""
    return np.correlate(h, 1 - h, mode="full")

def compute_all_gradients(h):
    """Compute ∇R(d) for all d, returned as array of shape (2n-1, n)."""
    n = len(h)
    M = 2 * n - 1
    G = np.zeros((M, n))
    # d = -(n-1), ..., 0, ..., n-1
    # output index i corresponds to d = i - (n-1)
    for i in range(M):
        d = i - (n - 1)
        if d == 0:
            G[i, :] = 1.0 - 2.0 * h
        elif d > 0:
            # ∂R(d)/∂h_i = (1-h_{i-d}) * 1{i >= d} - h_{i+d} * 1{i <= n-1-d}
            G[i, d:] = 1.0 - h[:n-d]  # (1 - h_{i-d}) for i >= d
            G[i, :n-d] -= h[d:]  # -h_{i+d} for i <= n-1-d
        else:
            dp = -d
            # ∂R(-dp)/∂h_i = (1-h_{i+dp}) * 1{i <= n-1-dp} - h_{i-dp} * 1{i >= dp}
            G[i, :n-dp] = 1.0 - h[dp:]  # (1 - h_{i+dp}) for i <= n-1-dp
            G[i, dp:] -= h[:n-dp]  # -h_{i-dp} for i >= dp
    return G

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        # Need to add mass: add to entries with most room
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for idx in order:
            add = min(remaining, capacity[idx])
            h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        # Need to remove mass: remove from entries with most mass
        order = np.argsort(h)
        remaining = -diff
        for idx in order:
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    # Final fix
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def lp_direction(h, max_step=0.05):
    """Solve LP to find best descent direction.
    Returns (direction, predicted_improvement) or (None, 0).
    """
    n = len(h)
    M = 2 * n - 1
    R = compute_profile(h)
    T0 = R.max()
    
    # Compute gradients
    G = compute_all_gradients(h)  # shape (M, n)
    
    # LP: minimize T_new
    # s.t. G[i] @ delta - T_new <= -R[i] for all i
    #      sum(delta) = 0
    #      -h[i] + (1-max_step)*h[i] ... wait, let me use bounds properly
    #      delta[i] >= -min(h[i], max_step)
    #      delta[i] <= min(1-h[i], max_step)
    
    # Variables: [delta_0, ..., delta_{n-1}, T_new]
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize T_new
    
    # Inequality constraints: G[i] @ delta - T_new <= -R[i]
    A_ub = np.hstack([G, -np.ones((M, 1))])
    b_ub = -R
    
    # Equality: sum(delta) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds
    lo = -np.minimum(h, max_step)
    hi = np.minimum(1.0 - h, max_step)
    bounds = list(zip(lo, hi)) + [(0.0, T0 + 1.0)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs',
                  options={'presolve': True})
    
    if res.success:
        delta = res.x[:n]
        T_new = res.x[-1]
        return delta, T_new
    return None, None

def lp_iterate(h, max_iters=100, max_step=0.03, t_start=None, budget_s=1100):
    """Iterate the LP-based descent."""
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    best_h = h.copy()
    
    for it in range(max_iters):
        if t_start is not None and time.perf_counter() - t_start > budget_s - 30:
            break
        
        delta, T_pred = lp_direction(h, max_step=max_step)
        if delta is None:
            break
        
        # Check predicted improvement
        current_max = compute_profile(h).max()
        if T_pred >= current_max - 1e-10:
            # No improvement predicted; try smaller step
            max_step *= 0.5
            if max_step < 1e-6:
                break
            continue
        
        # Take the step
        h_new = repair_feasible(h + delta)
        c_new = exact_score(h_new)
        
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            # Reset step size if we made progress
            max_step = min(max_step * 1.2, 0.05)
        else:
            # Try half step
            h_half = repair_feasible(h + 0.5 * delta)
            c_half = exact_score(h_half)
            if c_half < best_c - 1e-14:
                best_c = c_half
                best_h = h_half.copy()
                h = h_half
            else:
                max_step *= 0.5
                if max_step < 1e-7:
                    break
    
    return best_h, best_c
