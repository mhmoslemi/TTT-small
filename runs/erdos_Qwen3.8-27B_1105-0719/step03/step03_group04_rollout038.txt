def pair_move_optimal_alpha(profile, G_delta, alpha_max):
    """
    Given current profile C and the change vector delta = G[:,i] - G[:,j],
    find alpha in [0, alpha_max] that minimizes max_k (C[k] + alpha * delta[k]).
    
    Since f(alpha) = max_k(C[k] + alpha*delta[k]) is convex piecewise-linear,
    the minimum is at a breakpoint or at a boundary.
    """
    n_lags = len(profile)
    
    # If alpha_max is 0, nothing to do
    if alpha_max <= 0:
        return 0.0
    
    # The function f(alpha) = max_k (C[k] + alpha * delta[k])
    # at alpha=0: f(0) = max(C)
    # at alpha=alpha_max: f(alpha_max) = max(C + alpha_max * delta)
    
    # For the convex PWL function, the minimum is where the subgradient
    # changes from negative to positive.
    # Subgradient at alpha: {delta[k] : C[k] + alpha*delta[k] = f(alpha)}
    
    # Strategy: find the point where the "average slope" of active lines is 0.
    # More precisely, find alpha where there exist active k1, k2 with
    # delta[k1] < 0 < delta[k2] and both are at the maximum.
    
    # Practical approach: evaluate at breakpoints from top lines.
    
    C = profile
    d = G_delta
    
    # Get the top lines (by intercept C[k])
    # The upper envelope is determined by the top few lines
    top_n = min(30, n_lags)
    top_idx = np.argsort(C)[-top_n:]
    
    # Compute breakpoints between pairs of top lines
    breakpoints = [0.0, alpha_max]
    for i in range(len(top_idx)):
        for j in range(i+1, len(top_idx)):
            k1, k2 = top_idx[i], top_idx[j]
            if abs(d[k1] - d[k2]) > 1e-15:
                bp = (C[k1] - C[k2]) / (d[k2] - d[k1])
                if 0 < bp < alpha_max:
                    breakpoints.append(bp)
    
    # Also add breakpoints with the current max line and others
    max_idx = np.argmax(C)
    for k in range(n_lags):
        if abs(d[max_idx] - d[k]) > 1e-15:
            bp = (C[max_idx] - C[k]) / (d[k] - d[max_idx])
            if 0 < bp < alpha_max:
                breakpoints.append(bp)
    
    breakpoints = sorted(set(breakpoints))
    
    # Evaluate f at each breakpoint and boundaries
    best_alpha = 0.0
    best_val = np.max(C)  # f(0)
    
    for a in breakpoints:
        f_a = np.max(C + a * d)

Wait, the binary search approach for the convex PWL function isn't quite right because the function isn't smooth. Let me use a different approach.

For a convex PWL function f(α) = max_k (C[k] + α*Δ[k]), the minimum is at a point where the subgradient contains 0. The subgradient at α is the convex hull of {Δ[k] : C[k] + α*Δ[k] = f(α)}.

So the minimum is at α* where:
- There exist k1, k2 with C[k1] + α*Δ[k1] = C[k2] + α*Δ[k2] = f(α*)
- Δ[k1] < 0 and Δ[k2] > 0 (or one of them = 0)

This gives: α* = (C[k1] - C[k2]) / (Δ[k2] - Δ[k1])

But we don't know which (k1, k2) pair is active. However, we know that at the minimum, the active lines include at least one with positive slope and one with negative slope.

A practical approach: 
1. Compute all lines L_k(α) = C[k] + α*Δ[k]
2. The minimum of max(L_k) is the lowest point of the upper envelope
3. This occurs at a vertex of the upper envelope, which is an intersection of two lines

So I need to find all intersections of pairs of lines, evaluate f at each intersection (and at boundaries), and take the minimum.

import numpy as np
from scipy.optimize import linprog
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def build_gradient(h):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    G = np.zeros((M, n))
    center = n - 1
    G[center, :] = (1.0 - 2.0 * h)  # will multiply by dx at the end
    for d in range(1, n):
        G[center + d, :n-d] += -h[d:n]
        G[center + d, d:n] += (1.0 - h[:n-d])
        G[center - d, :n-d] += (1.0 - h[d:n])
        G[center - d, d:n] += -h[:n-d]
    return G * dx

def repair_feasible(h):
    n = len(h)
    h = np.clip(h.astype(np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, 1.0 - h[idx])
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)[::-1]
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    return h

def pair_move_alpha(profile, delta, alpha_max):
    """Find alpha in [0, alpha_max] minimizing max_k(profile[k] + alpha*delta[k])."""
    if alpha_max <= 1e-15:
        return 0.0, profile.max()
    
    C = profile
    d = delta
    n_lags = len(C)
    
    # Get top lines
    top_n = min(25, n_lags)
    top_idx = np.argsort(C)[-top_n:]
    
    # Find candidate breakpoints
    candidates = [0.0, alpha_max]
    
    # Intersections among top lines
    for i in range(len(top_idx)):
        for j in range(i+1, len(top_idx)):
            k1, k2 = top_idx[i], top_idx[j]
            dd = d[k2] - d[k1]
            if abs(dd) > 1e-14:
                bp = (C[k1] - C[k2]) / dd
                if 1e-10 < bp < alpha_max - 1e-10:
                    candidates.append(bp)
    
    # Intersections of max line with all others
    kmax = int(np.argmax(C))
    for k in range(n_lags):
        if k == kmax:
            continue
        dd = d[k] - d[kmax]
        if abs(dd) > 1e-14:
            bp = (C[kmax] - C[k]) / dd
            if 1e-10 < bp < alpha_max - 1e-10:
                candidates.append(bp)
    
    # Evaluate at all candidates
    best_val = C.max()
    best_alpha = 0.0
    for a in candidates:
        if a < 0 or a > alpha_max:
            continue
        val = np.max(C + a * d)
        if val < best_val - 1e-15:
            best_val = val
            best_alpha = a
    
    return best_alpha, best_val

def pair_descent(h, t_start, time_limit, max_iters=5000):

    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c = exact_score(h)
    stuck = 0
    
    for it in range(max_iters):

        c = profile.max()

            stuck = 0
        else:
            stuck += 1

        w = np.ones(len(active)) / len(active)
        grad = G[active, :].T @ w
        grad -= grad.mean()

        best_improvement = 1e-14
        best_move = None
        
        # Top positive and negative gradient coordinates
        pos_order = np.argsort(grad)[::-1][:8]
        neg_order = np.argsort(grad)[:8]

            for j in neg_order:
                if i == j:

                delta = G[:, i] - G[:, j]
                alpha_max = min(h[i], 1.0 - h[j])

def lp_trust_region(h, active, radius, t_start, time_limit):
    """LP trust region step."""
    n = len(h)
    dx = 2.0 / n
    G = build_gradient(h)
    profile = compute_profile(h)
    
    G_a = G[active, :]
    p_a = profile[active]
    
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    A_ub = np.zeros((len(active), n + 1))
    A_ub[:, :n] = G_a
    A_ub[:, -1] = -1.0
    b_ub = -p_a
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    lower = np.maximum(-h, -radius)
    upper = np.minimum(1.0 - h, radius)
    bounds = list(zip(lower, upper))
    bounds.append((0.0, None))
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'maxiter': 3000})
        if res.success and res.x is not None:
            step = res.x[:n]
            # Line search
            best_alpha = 0.0
            best_val = exact_score(h)
            for a in [1.0, 0.5, 0.25, 0.1, 0.05]:
                h_try = repair_feasible(h + a * step)
                val = exact_score(h_try)
                if val < best_val:
                    best_val = val
                    best_alpha = a
            if best_alpha > 0:
                return repair_feasible(h + best_alpha * step), best_val
    except Exception:
        pass
    return None, None

def build_gradient(h):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    G = np.zeros((M, n))
    center = n - 1
    
    # Lag 0
    G[center, :] = 1.0 - 2.0 * h
    
    for d in range(1, n):
        # Lag +d (index center + d)
        # G[center+d, j] = (1-h[j-d]) for j >= d, minus h[j+d] for j <= n-1-d
        G[center + d, d:] += (1.0 - h[:n-d])      # j from d to n-1: (1-h[j-d]) = 1-h[0:n-d]
        G[center + d, :n-d] += -h[d:]              # j from 0 to n-1-d: -h[j+d] = -h[d:n]
        
        # Lag -d (index center - d)
        # G[center-d, j] = (1-h[j+d]) for j <= n-1-d, minus h[j-d] for j >= d
        G[center - d, :n-d] += (1.0 - h[d:])       # j from 0 to n-1-d: (1-h[j+d]) = 1-h[d:n]
        G[center - d, d:] += -h[:n-d]              # j from d to n-1: -h[j-d] = -h[0:n-d]
    
    return G * dx

                    if new_c < c - 1e-14:
                        h[i] -= a
                        h[j] += a
                        break
                else:
                    continue
                break
            else:
                if stuck > 200:
                    break

    return best_h, best_c

def greedy_pair_descent(h, t_start, time_limit, max_iters=20000):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c = exact_score(h)
    stuck = 0

    for it in range(max_iters):
        if time.perf_counter() - t_start > time_limit:
            break

        profile = compute_profile(h)
        c = float(profile.max())

        if c < best_c - 1e-14:
            best_c = c
            best_h = h.copy()
            stuck = 0
        else:
            stuck += 1
            if stuck > 500:
                break

        G = build_gradient(h)
        T = c
        active = np.where(profile >= T - 0.02)[0]
        if len(active) > 20:
            active = np.argsort(profile)[-20:]

        w = np.ones(len(active)) / len(active)
        grad = G[active, :].T @ w
        grad -= grad.mean()

        pos_order = np.argsort(grad)[::-1][:10]
        neg_order = np.argsort(grad)[:10]

        best_improvement = 1e-14
        best_move = None

        for i in pos_order:
            for j in neg_order:
                if i == j:
                    continue
                am = min(h[i], 1.0 - h[j])
                if am < 1e-10:
                    continue
                delta = G[:, i] - G[:, j]
                for frac in [0.25, 0.5, 0.75, 1.0]:
                    a = frac * am
                    new_c = float(np.max(profile + a * delta))
                    if c - new_c > best_improvement:
                        best_improvement = c - new_c
                        best_move = (i, j, a)

        if best_move is not None:
            i, j, a = best_move
            h[i] -= a
            h[j] += a
        else:
            # Try reverse direction
            for i in neg_order[:5]:
                for j in pos_order[:5]:
                    if i == j:
                        continue
                    am = min(h[i], 1.0 - h[j])
                    if am < 1e-10:
                        continue
                    delta = G[:, i] - G[:, j]
                    for frac in [0.5, 1.0]:
                        a = frac * am
                        new_c = float(np.max(profile + a * delta))
                        if new_c < c - 1e-14:
                            h[i] -= a
                            h[j] += a
                            best_move = (i, j, a)
                            break
                    if best_move:
                        break
                if best_move:
                    break
            if best_move is None:
                if stuck > 100:
                    break

    return best_h, best_c

def lp_trust_region(h, active, radius, t_start, time_limit):
    n = len(h)
    G = build_gradient(h)
    profile = compute_profile(h)
    G_a = G[active, :]
    p_a = profile[active]

    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_ub = np.zeros((len(active), n + 1))
    A_ub[:, :n] = G_a
    A_ub[:, -1] = -1.0
    b_ub = -p_a
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    lower = np.maximum(-h, -radius)
    upper = np.minimum(1.0 - h, radius)
    bounds = list(zip(lower, upper))
    bounds.append((0.0, None))

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'maxiter': 3000})
        if res.success and res.x is not None and np.isfinite(res.x).all():
            step = res.x[:n]
            best_alpha = 0.0
            best_val = exact_score(h)
            for a in [1.0, 0.8, 0.6, 0.4, 0.2, 0.1, 0.05]:
                h_try = repair_feasible(h + a * step)
                val = exact_score(h_try)
                if val < best_val - 1e-15:
                    best_val = val
                    best_alpha = a
            if best_alpha > 0:
                return repair_feasible(h + best_alpha * step), best_val
    except Exception:
        pass
    return None, None

def fine_polish(h, t_start, time_limit, max_iters=5000):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c = exact_score(h)
    stuck = 0

    for it in range(max_iters):
        if time.perf_counter() - t_start > time_limit:
            break

        profile = compute_profile(h)
        c = float(profile.max())
        if c < best_c - 1e-15:
            best_c = c
            best_h = h.copy()
            stuck = 0
        else:
            stuck += 1
            if stuck > 200:
                break

        G = build_gradient(h)
        T = c
        active = np.where(profile >= T - 0.005)[0]
        if len(active) > 15:
            active = np.argsort(profile)[-15:]

        # Try all pairs with small steps
        best_improvement = 1e-15
        best_move = None
        for i in range(n):
            if h[i] < 1e-8:
                continue
            for j in range(n):
                if i == j:
                    continue
                am = min(h[i], 1.0 - h[j])
                if am < 1e-8:
                    continue
                delta = G[:, i] - G[:, j]
                a = 0.3 * am
                new_c = float(np.max(profile + a * delta))
                if c - new_c > best_improvement:
                    best_improvement = c - new_c
                    best_move = (i, j, a)
                a = 0.1 * am
                if a > 1e-10:
                    new_c2 = float(np.max(profile + a * delta))
                    if c - new_c2 > best_improvement:
                        best_improvement = c - new_c2
                        best_move = (i, j, a)

        if best_move is not None:
            i, j, a = best_move
            h[i] -= a
            h[j] += a
        else:
            break

    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()
    np.random.seed(seed)

    def time_left():
        return budget_s - (time.perf_counter() - t_start)

    try:
        parent = np.array(initial_h_values, dtype=np.float64)
    except (NameError, Exception):
        parent = np.full(70, 0.5)

    parent = np.clip(parent, 0.0, 1.0)
    parent = repair_feasible(parent)

    best_h = parent.copy()
    best_c = exact_score(parent)

    # Phase 1: Greedy pair descent (fast, many iterations)
    if time_left() > 200:
        h1, c1 = greedy_pair_descent(best_h, t_start, budget_s - 200, max_iters=5000)
        if c1 < best_c - 1e-14:
            best_h, best_c = h1, c1

    # Phase 2: Full pair sweep (more thorough)
    if time_left() > 150:
        h2, c2 = pair_sweep(best_h, t_start, budget_s - 150, max_passes=50000)
        if c2 < best_c - 1e-14:
            best_h, best_c = h2, c2

    # Phase 3: LP trust region
    if time_left() > 100:
        n = len(best_h)
        profile = compute_profile(best_h)
        T = float(profile.max())
        for beam in [0.02, 0.01, 0.005, 0.002]:
            if time_left() < 30:
                break
            active = np.where(profile >= T - beam)[0]
            if len(active) > 15:
                active = np.argsort(profile)[-15:]
            for radius in [0.05, 0.02, 0.01, 0.005, 0.002, 0.001]:
                if time_left() < 15:
                    break
                h_lp, c_lp = lp_trust_region(best_h, active, radius, t_start, budget_s - 10)
                if h_lp is not None and c_lp < best_c - 1e-14:
                    best_h, best_c = h_lp, c_lp
                    profile = compute_profile(best_h)
                    T = float(profile.max())
                    active = np.where(profile >= T - beam)[0]
                    if len(active) > 15:
                        active = np.argsort(profile)[-15:]

    # Phase 4: More pair sweeps (after LP)
    if time_left() > 80:
        h4, c4 = pair_sweep(best_h, t_start, budget_s - 80, max_passes=50000)
        if c4 < best_c - 1e-14:
            best_h, best_c = h4, c4

    # Phase 5: Fine polish
    if time_left() > 50:
        h5, c5 = fine_polish(best_h, t_start, budget_s - 50, max_iters=5000)
        if c5 < best_c - 1e-14:
            best_h, best_c = h5, c5

    # Phase 6: Another LP round
    if time_left() > 40:
        n = len(best_h)
        profile = compute_profile(best_h)
        T = float(profile.max())
        active = np.where(profile >= T - 0.003)[0]
        if len(active) > 12:
            active = np.argsort(profile)[-12:]
        for radius in [0.005, 0.002, 0.001, 0.0005]:
            if time_left() < 10:
                break
            h_lp, c_lp = lp_trust_region(best_h, active, radius, t_start, budget_s - 5)
            if h_lp is not None and c_lp < best_c - 1e-14:
                best_h, best_c = h_lp, c_lp

    # Phase 7: Final fine polish
    if time_left() > 20:
        h7, c7 = fine_polish(best_h, t_start, budget_s - 5, max_iters=3000)
        if c7 < best_c - 1e-14:
            best_h, best_c = h7, c7

    # Phase 8: Try dimension neighbors
    n_best = len(best_h)
    if time_left() > 30:
        for new_n in [n_best - 2, n_best + 2]:
            if time_left() < 15:
                break
            if new_n < 20 or new_n > 300:
                continue
            xs = (np.arange(n_best) + 0.5) * (2.0 / n_best)
            xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
            h_interp = np.interp(xt, xs, best_h)
            h_interp = repair_feasible(h_interp)
            h_dim, c_dim = greedy_pair_descent(h_interp, t_start, budget_s - 10, max_iters=2000)
            if c_dim < best_c - 1e-14:
                best_h, best_c = h_dim, c_dim

    # Final validation
    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)

    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            final_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 2:
            final_c = float(eval_result[0])
    except Exception:
        pass

    return best_h, final_c, len(best_h)
