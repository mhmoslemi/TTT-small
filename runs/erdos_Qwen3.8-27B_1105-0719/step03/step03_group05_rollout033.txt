def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:

        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx] + 1e-15)
            if add > 0:

            remaining -= add
            if remaining <= 1e-12:
                break
    else:

        remaining = -diff
        for i in range(n):

            rem = min(remaining, h[idx] + 1e-15)

                h[idx] -= rem
            remaining -= rem

                break

        else:
            idx = int(np.argmin(h))

Actually, let me also think about a completely different approach that might be more powerful:

**Subgradient method with Polyak step size:**

The problem: minimize f(h) = max_m P_m(h) where P_m(h) = sum_n h[n]*(1-h)[n-m+(n-1)]*dx.

Subgradient of f at h: any convex combination of {∇P_m(h) : m ∈ argmax P_m(h)}.

Polyak step size: step_k = (f(h_k) - f*) / ||g_k||² where f* is the optimal value (which we don't know). We can use f* = 0 (lower bound) or f* = some estimate.

Actually, since we don't know f*, we can use a decreasing step size: step_k = c / sqrt(k) for some c.

With averaging: h_avg = (1/K) * sum_{k=1}^{K} h_k.

This is a standard technique for subgradient methods. It's guaranteed to converge (for convex problems) and often works well for non-convex problems too.

The key advantage: it's very simple and can be run for a long time without getting stuck (unlike gradient descent which can stall at non-smooth points of the max function).

Let me add this as another component.

**Subgradient method implementation:**

```python
def subgradient_method(h0, max_iters=10000, c_step=0.1):
    n = len(h0)
    dx = 2.0 / n
    h = h0.copy()
    h_avg = h0.copy()
    best_c = exact_score(h)
    best_h = h0.copy()
    
    for k in range(1, max_iters + 1):
        profile = compute_profile(h)
        c = profile.max()
        m_active = np.where(profile >= c - 1e-10)[0]
        
        # Subgradient: average gradient of active constraints
        G = build_gradient_matrix(h, n, dx)
        g = G[m_active].mean(axis=0)  # shape (n,)
        
        # Remove mean to preserve sum constraint
        g -= g.mean()
        
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        
        # Step size
        step = c_step / np.sqrt(k)
        step = min(step, 0.1)  # cap
        
        h_new = h - step * g / gn
        h_new = repair_feasible(h_new)
        
        h_avg = h_avg + (h_new - h_avg) / k
        h = h_new
        
        c_new = exact_score(h)
        if c_new < best_c:
            best_c = c_new
            best_h = h.copy()
        
        c_avg = exact_score(repair_feasible(h_avg))
        if c_avg < best_c:
            best_c = c_avg
            best_h = repair_feasible(h_avg).copy()
    
    return best_h, best_c
```

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx] + 1e-15)
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx] + 1e-15)
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-14:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmin(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def build_gradient_matrix(h, n, dx):
    M = 2 * n - 1
    G = np.zeros((M, n))
    for m in range(M):
        d = m - (n - 1)
        if d == 0:
            G[m, :] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            G[m, d:] += dx * (1.0 - h[0:n-d])
            G[m, 0:n-d] += -dx * h[d:n]
        else:
            dp = -d
            G[m, 0:n-dp] += dx * (1.0 - h[dp:n])
            G[m, dp:n] += -dx * h[0:n-dp]
    return G

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = np.linspace(0, 2, n + 1)
    xt = np.linspace(0, 2, new_n + 1)
    h_interp = np.zeros(new_n)
    for k in range(new_n):
        lo_k, hi_k = xt[k], xt[k + 1]
        total_mass = 0.0
        total_width = 0.0
        for j in range(n):
            lo_j, hi_j = xs[j], xs[j + 1]
            overlap = min(hi_k, hi_j) - max(lo_k, lo_j)
            if overlap > 0:
                total_mass += h[j] * overlap
                total_width += overlap
        h_interp[k] = total_mass / total_width if total_width > 1e-15 else 0.5
    h_interp = np.clip(h_interp, 0.0, 1.0)
    return repair_feasible(h_interp)
