def repair_feasible(h):
    n = len(h)
    h = np.clip(np.asarray(h, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, 1.0 - h[idx])
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)[::-1]
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))

    obj = cp.Minimize(r + gamma * cp.sum_squares(delta))
    prob = cp.Problem(obj, constraints)
    
    for solver in [cp.CLARABEL, cp.SCS, cp.ECOS, cp.MOSEK, cp.GUROBI]:
        try:
            prob.solve(solver=solver)
            if prob.status in ['optimal', 'optimal_inaccurate']:
                if delta.value is not None:
                    return np.array(delta.value, dtype=np.float64), float(r.value)
            break
        except Exception:
            continue
    return None

def lp_step_fallback(h, active, radius):
    from scipy.optimize import linprog
    n = len(h)
    dx = 2.0 / n
    profile = compute_profile(h)
    G = build_gradient(h, n, dx)
    G_a = G[active, :]
    p_a = profile[active]
    
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    A_ub = np.zeros((len(active), n + 1))
    A_ub[:, :n] = G_a
    A_ub[:, -1] = -1.0
    b_ub = -p_a
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    lower = np.maximum(-h, -radius)
    upper = np.minimum(1.0 - h, radius)
    bounds = list(zip(lower, upper))
    bounds.append((0.0, None))
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'maxiter': 3000})
        if res.success and res.x is not None:
            return res.x[:n]
    except Exception:
        pass
    return None

def optimize_local(h, t_start, deadline, max_iters=300):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c = exact_score(h)
    radius = 0.04
    gamma = 25.0
    no_improve = 0
    use_qp = True
    
    for it in range(max_iters):
        elapsed = time.perf_counter() - t_start
        if deadline - elapsed < 8:
            break
        
        profile = compute_profile(h)
        c = float(np.max(profile))
        if c < best_c - 1e-13:
            best_c = c
            best_h = h.copy()
            no_improve = 0
            radius = min(radius * 1.15, 0.12)
        else:
            no_improve += 1
        
        if no_improve > 35:
            break
        if radius < 0.0005:
            break
        
        T = c
        spread = T - (np.percentile(profile, 20) + 1e-15)
        band = max(0.001, 0.08 * spread)
        active = np.where(profile >= T - band)[0]
        if len(active) > 50:
            top_idx = np.argsort(profile[active])[-50:]
            active = active[top_idx]
        if len(active) < 3:
            active = np.argsort(profile)[-5:]
        
        took_step = False
        
        if use_qp:
            result = qp_step_cvxpy(h, active, radius, gamma)
            if result is not None:
                delta, r_pred = result
                h_new, s_new = line_search_plus(h, delta, c)
                if s_new < c - 1e-13:
                    h = h_new
                    took_step = True
            
            if not took_step and no_improve > 8:
                gamma = max(gamma * 0.8, 5.0)
                result = qp_step_cvxpy(h, active, radius * 1.5, gamma)
                if result is not None:
                    delta, r_pred = result
                    h_new, s_new = line_search_plus(h, delta, c)
                    if s_new < c - 1e-13:
                        h = h_new
                        took_step = True
        
        if not took_step:
            delta_lp = lp_step_fallback(h, active, radius * 0.8)
            if delta_lp is not None:
                h_new, s_new = line_search_plus(h, delta_lp, c)
                if s_new < c - 1e-13:
                    h = h_new
                    took_step = True
        
        if not took_step:
            gs = subgradient_direction(h, active, profile)
            if gs is not None:
                h_new, s_new = line_search(h, gs, c, max_step=radius)
                if s_new < c - 1e-13:
                    h = h_new
                    took_step = True
        
        if took_step:
            radius = min(radius * 1.1, 0.15)
            gamma = min(gamma * 1.05, 50.0)
        else:
            radius *= 0.75
            gamma *= 0.9
    
    return best_h, best_c

def slsqp_epigraph(h, t_start, deadline, max_iter=500):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    c0 = exact_score(h)
    
    def objective(x):
        return x[-1]
    
    def objective_grad(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    
    def eq_constraint(x):
        return x[:-1].sum() - n / 2.0
    
    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g
    
    def ineq_constraint(x):
        hh = x[:-1]
        t = x[-1]
        prof = np.correlate(hh, 1 - hh, mode="full") * dx
        return t - prof
    
    def ineq_jac(x):
        hh = x[:-1]
        G = build_gradient(hh, n, dx)
        J = np.zeros((M, n + 1))
        J[:, :-1] = -G
        J[:, -1] = 1.0
        return J
    
    x0 = np.concatenate([h, [c0 + 1e-4]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    cons = [
        {'type': 'eq', 'fun': eq_constraint, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_constraint, 'jac': ineq_jac},
    ]
    
    try:
        res = minimize(objective, x0, method='SLSQP', jac=objective_grad,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-14})
        if res.x is not None:
            h_sol = repair_feasible(res.x[:-1])
            c_sol = exact_score(h_sol)
            return h_sol, c_sol
    except Exception:
        pass
    return h, c0

def make_random_start(n, rng, mode=0):
    h = np.zeros(n)
    if mode == 0:
        h[:] = 0.5 + 0.3 * np.sin(np.pi * np.arange(n) / n + rng.uniform(0, 2*np.pi))
    elif mode == 1:
        nb = rng.randint(3, 6)
        block = n // nb
        for b in range(nb):
            h[b*block:(b+1)*block] = rng.uniform(0.2, 0.8)
    elif mode == 2:
        phi = 0.6180339887498949
        for i in range(n):
            h[i] = 0.5 + 0.35 * np.sin(2*np.pi * (i * phi % 1.0))
    else:
        h[:] = rng.uniform(0.3, 0.7, n)
    return repair_feasible(h)

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()
    rng = np.random.RandomState(seed)
    
    try:
        parent = np.asarray(initial_h_values, dtype=np.float64)
    except (NameError, Exception):
        parent = np.full(70, 0.5)
    
    if len(parent) < 4 or not np.isfinite(parent).all():
        parent = np.full(70, 0.5)
    
    parent = np.clip(parent, 0.0, 1.0)
    parent = repair_feasible(parent)
    
    best_h = parent.copy()
    best_c = exact_score(parent)
    n0 = len(parent)
    
    h_sym = repair_feasible(symmetrize(parent))
    c_sym = exact_score(h_sym)
    if c_sym < best_c - 1e-12:
        best_h, best_c = h_sym, c_sym
    
    h_comp = repair_feasible(1.0 - parent)
    c_comp = exact_score(h_comp)
    if c_comp < best_c - 1e-12:
        best_h, best_c = h_comp, c_comp
    
    # Phase 1: QP trust region on parent
    phase1_deadline = t_start + budget_s * 0.30
    h1, c1 = optimize_local(best_h, t_start, phase1_deadline, max_iters=200)
    if c1 < best_c - 1e-13:
        best_h, best_c = h1, c1
    
    # Phase 2: SLSQP epigraph on current best
    if time.perf_counter() - t_start < budget_s * 0.55:
        h2, c2 = slsqp_epigraph(best_h, t_start, t_start + budget_s * 0.55, max_iter=300)
        if c2 < best_c - 1e-13:
            best_h, best_c = h2, c2
    
    # Phase 3: Try nearby dimensions
    dims_to_try = []
    for d in range(max(20, n0-10), min(120, n0+15), 2):
        if d != n0 and d >= 20:
            dims_to_try.append(d)
    rng_dims = rng.choice(dims_to_try, size=min(5, len(dims_to_try)), replace=False)
    
    phase3_deadline = t_start + budget_s * 0.75
    for new_n in sorted(rng_dims, key=lambda d: abs(d - n0)):
        if time.perf_counter() - t_start > phase3_deadline:
            break
        h_interp = interpolate_to_n(best_h, int(new_n))
        h_opt, c_opt = optimize_local(h_interp, t_start, phase3_deadline, max_iters=100)
        if c_opt < best_c - 1e-13:
            best_h, best_c = h_opt, c_opt
            if time.perf_counter() - t_start < phase3_deadline - 20:
                h_sl, c_sl = slsqp_epigraph(best_h, t_start, phase3_deadline, max_iter=200)
                if c_sl < best_c - 1e-13:
                    best_h, best_c = h_sl, c_sl
    
    # Phase 4: Random restarts
    if time.perf_counter() - t_start < budget_s * 0.85:
        phase4_deadline = t_start + budget_s * 0.90
        n_b = len(best_h)
        for attempt in range(4):
            if time.perf_counter() - t_start > phase4_deadline - 30:
                break
            mode = attempt % 4
            h_rand = make_random_start(n_b, rng, mode)
            h_blend = 0.4 * best_h + 0.6 * h_rand
            h_blend = repair_feasible(h_blend)
            c_blend = exact_score(h_blend)
            if c_blend < best_c + 0.02:
                h_opt, c_opt = optimize_local(h_blend, t_start, phase4_deadline, max_iters=80)
                if c_opt < best_c - 1e-13:
                    best_h, best_c = h_opt, c_opt
    
    # Phase 5: Final SLSQP polish
    if time.perf_counter() - t_start < budget_s - 20:
        h5, c5 = slsqp_epigraph(best_h, t_start, t_start + budget_s - 5, max_iter=400)
        if c5 < best_c - 1e-13:
            best_h, best_c = h5, c5
    
    # Phase 6: Quick local polish
    if time.perf_counter() - t_start < budget_s - 10:
        h6, c6 = optimize_local(best_h, t_start, t_start + budget_s - 2, max_iters=50)
        if c6 < best_c - 1e-13:
            best_h, best_c = h6, c6
    
    # Final: try symmetrization
    h_final_sym = repair_feasible(symmetrize(best_h))
    c_final_sym = exact_score(h_final_sym)
    if c_final_sym < best_c - 1e-12:
        best_h, best_c = h_final_sym, c_final_sym
    
    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)
    
    try:
        ev = evaluate_erdos_solution(best_h)
        if isinstance(ev, (float, int)):
            final_c = float(ev)
        elif isinstance(ev, tuple) and len(ev) >= 2:
            final_c = float(ev[0])
    except Exception:
        pass
    
    return best_h, final_c, len(best_h)
