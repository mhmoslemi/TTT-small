def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx] + 1e-15)
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx] + 1e-15)
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:

def slsqp_solve(h0, maxiter=200):
    n = len(h0)
    dx = 2.0 / n
    M = 2 * n - 1
    c0 = exact_score(h0)
    
    def obj(x):
        return x[-1]
    
    def eq_fun(x):
        return x[:-1].sum() - n / 2.0
    
    def ineq_fun(x):
        h = np.clip(x[:-1], 0, 1)
        t = x[-1]
        profile = np.correlate(h, 1 - h, mode="full") * dx
        return t - profile
    
    x0 = np.concatenate([h0, [c0 + 1e-3]])
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 0.5)]
    cons = [
        {'type': 'eq', 'fun': eq_fun},
        {'type': 'ineq', 'fun': ineq_fun}
    ]
    try:
        res = minimize(obj, x0, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None and np.isfinite(res.x).all():
            h_new = repair_feasible(res.x[:-1])
            c_new = exact_score(h_new)
            if c_new <= c0 + 1e-9:
                return h_new, c_new
    except Exception:
        pass
    return None, None

    t_start = time.perf_counter()
    rng = np.random.RandomState(seed)
    
    def time_left():
        return budget_s + t_start - time.perf_counter()
    
    # Get parent
    parent = None
    try:
        parent = np.array(initial_h_values, dtype=np.float64)
    except NameError:
        pass
    except Exception:
        pass
    
    if parent is None or len(parent) < 4:
        parent = np.full(70, 0.5)
    
    parent = np.clip(parent, 0.0, 1.0)
    parent = repair_feasible(parent)
    n_parent = len(parent)
    
    best_h = parent.copy()
    best_c = exact_score(parent)
    
    # Anti-symmetric projection
    y_best = full_to_reduced(parent, n_parent)
    h_anti = reduced_to_full(y_best, n_parent)
    c_anti = exact_score(h_anti)
    if c_anti < best_c - 1e-12:
        best_h = h_anti.copy()
        best_c = c_anti
    
    # Phase 1: Anti-symmetric line search optimization
    # Interleaved single and pair coordinate searches with gradient guidance
    n = n_parent
    m = n // 2
    stuck_count = 0
    max_stuck = 8
    resolution_change_every = 12
    
    pass_count = 0
    while time_left() > 120:
        pass_count += 1
        
        # Compute gradient for direction selection
        grad, c_cur = reduced_gradient(y_best, n)
        gabs = np.abs(grad)
        
        # Prioritized coordinate order by gradient magnitude
        coord_order = np.argsort(-gabs)
        
        improved_any = False
        
        # Single coordinate line searches
        for coord in coord_order:
            if time_left() < 60:
                break
            if gabs[coord] < 1e-6:
                continue
            y_new, delta = line_search_single(y_best, n, coord, n_samples=80)
            if delta < -1e-12:
                y_best = y_new
                improved_any = True
        
        # Pair line searches (top gradient coordinates)
        n_pairs = min(15, m * (m - 1) // 2)
        pair_indices = []
        for i in range(min(8, m)):
            for j in range(min(8, m)):
                if i != j:
                    pair_indices.append((int(coord_order[i]), int(coord_order[j])))
        rng.shuffle(pair_indices)
        pair_indices = pair_indices[:n_pairs]
        
        for (i, j) in pair_indices:
            if time_left() < 60:
                break
            y_new, delta = line_search_pair(y_best, n, i, j, n_samples=50)
            if delta < -1e-12:
                y_best = y_new
                improved_any = True
        
        if improved_any:
            stuck_count = 0
        else:
            stuck_count += 1
        
        # Periodic LSE descent in reduced space
        if stuck_count % 3 == 0 and time_left() > 100:
            y_new, c_new = lse_reduced_fast(y_best, n, beta=60, max_iters=80, alpha_init=0.01)
            if c_new < best_c - 1e-13:
                y_best = y_new
                best_h = reduced_to_full(y_best, n)
                best_c = c_new
                stuck_count = 0
        
        # Resolution change periodically
        if stuck_count >= max_stuck and time_left() > 150:
            # Try nearby resolutions
            for dn in [-2, 2, -4, 4]:
                if time_left() < 60:
                    break
                new_n = n + dn
                if new_n < 30 or new_n > 120 or new_n % 2 != 0:
                    continue
                h_interp = interpolate_to_n(reduced_to_full(y_best, n), new_n)
                y_new = full_to_reduced(h_interp, new_n)
                c_try = score_reduced(y_new, new_n)
                if c_try < best_c - 1e-10:
                    y_best = y_new
                    n = new_n
                    m = n // 2
                    best_h = reduced_to_full(y_best, n)
                    best_c = c_try
                    stuck_count = 0
                    break
            if stuck_count >= max_stuck:
                # Reset: try random perturbation in reduced space
                pert = rng.randn(m) * 0.03
                y_pert = np.clip(y_best + pert, 0.0, 1.0)
                c_pert = score_reduced(y_pert, n)
                if c_pert < best_c + 0.005:
                    y_best = y_pert
                stuck_count = 0
            else:
                stuck_count = 0
    
    # Update best from anti-symmetric
    c_anti = score_reduced(y_best, n)
    if c_anti < best_c - 1e-13:
        best_h = reduced_to_full(y_best, n)
        best_c = c_anti
    
    # Phase 2: Full-space active set descent on best
    if time_left() > 60:
        h_new, c_new = active_set_descent_full(best_h, max_iters=150, n_active=10, alpha_init=0.003)
        if c_new < best_c - 1e-13:
            best_h = h_new
            best_c = c_new
    
    # Phase 3: SLSQP polish
    if time_left() > 50:
        h_sol, c_sol = slsqp_solve(best_h, maxiter=150)
        if h_sol is not None and c_sol < best_c - 1e-12:
            best_h = h_sol
            best_c = c_sol
            # Re-project to anti-symmetric and continue
            n_cur = len(best_h)
            y_new = full_to_reduced(best_h, n_cur)
            c_anti2 = score_reduced(y_new, n_cur)
            if c_anti2 < best_c - 1e-13:
                best_h = reduced_to_full(y_new, n_cur)
                best_c = c_anti2
            # Quick LSE in reduced space
            if time_left() > 40:
                y_new2, c_new2 = lse_reduced_fast(y_new, n_cur, beta=100, max_iters=60)
                if c_new2 < best_c - 1e-13:
                    best_h = reduced_to_full(y_new2, n_cur)
                    best_c = c_new2
    
    # Phase 4: Multi-resolution final check
    if time_left() > 40:
        n_cur = len(best_h)
        for dn in [-2, 2, -4, 4, -6, 6]:
            if time_left() < 20:
                break
            new_n = n_cur + dn
            if new_n < 30 or new_n > 120 or new_n % 2 != 0:
                continue
            h_interp = interpolate_to_n(best_h, new_n)
            y_new = full_to_reduced(h_interp, new_n)
            c_try = score_reduced(y_new, new_n)
            if c_try < best_c - 1e-10:
                # Polish at this resolution
                y_pol, c_pol = lse_reduced_fast(y_new, new_n, beta=100, max_iters=50)
                if c_pol < best_c - 1e-12:
                    best_h = reduced_to_full(y_pol, new_n)
                    best_c = c_pol
    
    # Phase 5: Final active set + gradient descent
    if time_left() > 30:
        h_new, c_new = active_set_descent_full(best_h, max_iters=100, n_active=8, alpha_init=0.002)
        if c_new < best_c - 1e-13:
            best_h = h_new
            best_c = c_new
    
    # Final validation
    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)
    
    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            final_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 2:
            final_c = float(eval_result[0])
    except Exception:
        pass
    
    return best_h, final_c, len(best_h)
