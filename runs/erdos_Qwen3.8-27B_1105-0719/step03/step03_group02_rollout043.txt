def compute_gradient_matrix(h, n):
    """Compute G[i,k] = d(corr[k])/d(h[i]) for all i, k."""
    G = np.zeros((n, 2*n-1))
    for k in range(2*n-1):
        s = k - (n-1)
        for i in range(n):
            g = 0.0
            if 0 <= i+s < n:
                g += 1.0 - h[i+s]
            if 0 <= i-s < n:
                g -= h[i-s]
            G[i, k] = g
    return G

            h[mask] += diff / mask.sum()

    return np.correlate(h, 1.0 - h, mode='full')

    return compute_corrs(h).max() * dx

        s = k - (n-1)
        for i in range(n):
            g = 0.0
            if 0 <= i+s < n:
                g += 1.0 - h[i+s]
            if 0 <= i-s < n:
                g -= h[i-s]
            G[i, k] = g
    return G

    if hi - lo < 1e-14:
        return (lo + hi) * 0.5
    gr = 0.6180339887498949
    a, b = lo, hi
    c = b - gr * (b - a)
    d2 = a + gr * (b - a)
    fc = np.max(corr + c * d)
    fd = np.max(corr + d2 * d)

            c = d2
            fc = fd
            d2 = a + gr * (b - a)

import numpy as np

def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.0, 1.5
    for _ in range(80):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)


def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def c5_of_h(h, dx):
    return compute_corrs(h).max() * dx


def compute_G_vec(h, n):
    G = np.zeros((n, 2 * n - 1))
    for s in range(-(n - 1), n):
        k = s + (n - 1)
        i_min1 = max(0, -s)
        i_max1 = min(n - 1, n - 1 - s)
        if i_max1 >= i_min1:
            G[i_min1:i_max1 + 1, k] += 1.0 - h[i_min1 + s:i_max1 + 1 + s]
        i_min2 = max(0, s)
        i_max2 = min(n - 1, n - 1 + s)
        if i_max2 >= i_min2:
            G[i_min2:i_max2 + 1, k] -= h[i_min2 - s:i_max2 + 1 - s]
    return G


def golden_min_1d(corr, d, lo, hi, n_iter=30):
    if hi - lo < 1e-14:
        return (lo + hi) * 0.5
    gr = 0.6180339887498949
    a, b = lo, hi
    c = b - gr * (b - a)
    d2 = a + gr * (b - a)
    fc = np.max(corr + c * d)
    fd = np.max(corr + d2 * d)
    for _ in range(n_iter):
        if b - a < 1e-15:
            break
        if fc < fd:
            b = d2
            d2 = c
            fd = fc
            c = b - gr * (b - a)
            fc = np.max(corr + c * d)
        else:
            a = c
            c = d2
            fc = fd
            d2 = a + gr * (b - a)
            fd = np.max(corr + d2 * d)
    return (a + b) * 0.5


def pair_transfer_optimize(h0, dx, target_sum, time_check, max_rounds=5000, n_sample=800):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    corr = compute_corrs(h)
    best_c5 = corr.max() * dx
    stall = 0
    rng = np.random.RandomState(12345)
    ii_all, jj_all = np.triu_indices(n, 1)
    n_pairs = len(ii_all)

    for rnd in range(max_rounds):
        if time_check() < 30:
            break
        G = compute_G_vec(h, n)
        c_max = corr.max()

        tol = 1e-7 * max(1.0, c_max)
        active = np.where(corr >= c_max - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(corr)])

        if n_pairs <= n_sample:
            sel = np.arange(n_pairs)
        else:
            sel = rng.choice(n_pairs, n_sample, replace=False)

        D_act = G[ii_all[sel], active] - G[jj_all[sel], active]
        min_D = D_act.min(axis=1)
        max_D = D_act.max(axis=1)
        improv = (min_D > 1e-12) | (max_D < -1e-12)
        imp_idx = sel[improv]

        if len(imp_idx) == 0:
            stall += 1
            if stall > 30:
                break
            continue

        candidates = []
        for idx in imp_idx:
            i = ii_all[idx]
            j = jj_all[idx]
            eps_lo = max(-h[i], h[j] - 1.0)
            eps_hi = min(1.0 - h[i], h[j])
            if eps_hi - eps_lo < 1e-12:
                continue
            d = G[i, :] - G[j, :]
            eps = golden_min_1d(corr, d, eps_lo, eps_hi, 30)
            if abs(eps) > 1e-11:
                new_max = np.max(corr + eps * d)
                imp_val = c_max - new_max
                if imp_val > 5e-11:
                    candidates.append((imp_val, i, j, eps))

        if not candidates:
            stall += 1
            if stall > 30:
                break
            continue

        candidates.sort(key=lambda x: -x[0])
        K = min(15, len(candidates))
        for imp_val, i, j, eps in candidates[:K]:
            h[i] += eps
            h[j] -= eps

        h = project_feasible(h, target_sum)
        corr = compute_corrs(h)
        new_c5 = corr.max() * dx

        if new_c5 < best_c5 - 1e-15:
            best_c5 = new_c5
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 30:
                break

    return best_h, best_c5

def slsqp_optimize(h0, dx, target_sum, beta, maxiter):
    n = len(h0)

    def obj(x):
        corrs = compute_corrs(x)
        return logsumexp(beta * corrs) / (beta * dx) * dx

    def grad_obj(x):
        corrs = compute_corrs(x)
        shifted = corrs - corrs.max()
        w = np.exp(beta * shifted)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones_like(corrs) / len(corrs)
        else:
            w /= w_sum
        G = compute_G_vec(x, n)
        grad = (w[None, :] * G).sum(axis=1) * dx
        return grad

    def c_fun(x):
        return x.sum() - target_sum

    def c_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': c_fun, 'jac': c_jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0, jac=grad_obj, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        h_out = project_feasible(res.x, target_sum)
    except Exception:
        h_out = project_feasible(h0, target_sum)
    return h_out, c5_of_h(h_out, dx)


def penalty_descent(h0, dx, target_sum, time_check, n_target=300, lr_init=0.005):
    """Threshold penalty descent to push all correlations below a target."""
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    lr = lr_init

    for epoch in range(n_target):
        if time_check() < 20:
            break
        target = best_c5 - epoch * (best_c5 * 0.3 / n_target)
        target = max(target, best_c5 * 0.7)

        for it in range(50):
            if time_check() < 20:
                break
            corr = compute_corrs(h)
            excess = np.maximum(0.0, corr * dx - target)
            If_total = (excess ** 2).sum()
            if If_total < 1e-20:
                break
            G = compute_G_vec(h, n)
            grad = (2.0 * excess[None, :] * G).sum(axis=1) * dx
            h = project_feasible(h - lr * grad, target_sum)
            c5_now = c5_of_h(h, dx)
            if c5_now < best_c5 - 1e-15:
                best_c5 = c5_now
                best_h = h.copy()

        lr *= 0.95

    return best_h, best_c5
