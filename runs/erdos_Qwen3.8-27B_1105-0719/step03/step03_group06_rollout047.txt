Minimize tau

The bounds on d_i are:
lo_i = max(-delta, -h_i)
hi_i = min(delta, 1 - h_i)

import numpy as np
from scipy.optimize import linprog, minimize
import time

def repair_feasible(vec, target=None):
    n = len(vec)
    if target is None:
        target = n / 2.0
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    # Handle non-finite
    h[~np.isfinite(h)] = 0.5
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(-(1.0 - h))
        remaining = diff
        for i in range(n):
            idx = order[i]
            cap = 1.0 - h[idx]
            if cap <= 1e-15:
                continue
            add = min(remaining, cap)
            h[idx] += add
            remaining -= add
            if remaining <= 1e-14:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            if h[idx] <= 1e-15:
                continue
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-14:
                break
    # Final correction
    r = target - h.sum()
    if abs(r) > 1e-13:
        idx = int(np.argmax(1.0 - h)) if r > 0 else int(np.argmax(h))
        h[idx] = float(np.clip(h[idx] + r, 0.0, 1.0))
    return h

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def grad_profile(h, n):
    """Compute gradient matrix G of shape (M, n) where M = 2n-1"""
    dx = 2.0 / n
    M = 2 * n - 1
    G = np.zeros((M, n))
    for m in range(M):
        d = m - (n - 1)
        if d == 0:
            G[m, :] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            G[m, d:] += dx * (1.0 - h[:-d])
            G[m, :-d] -= dx * h[d:]
        else:
            p = -d
            G[m, :-p] += dx * (1.0 - h[p:])
            G[m, p:] -= dx * h[:-p]
    return G

def verify_gradients(h, n, eps=1e-7):
    """Verify gradient formula via finite differences"""
    dx = 2.0 / n
    M = 2 * n - 1
    G = grad_profile(h, n)
    P = np.correlate(h, 1 - h, mode="full") * dx
    max_err = 0.0
    for m in range(M):
        for i in range(n):
            h2 = h.copy()
            h2[i] += eps
            P2 = np.correlate(h2, 1 - h2, mode="full") * dx
            fd = (P2[m] - P[m]) / eps
            if abs(fd - G[m, i]) > 1e-4:
                max_err = max(max_err, abs(fd - G[m, i]))
    return max_err

def lp_trust_step(h, c, delta=0.05):
    """Solve linearized trust region LP. Returns (d, tau) or (None, None)"""
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    P = np.correlate(h, 1 - h, mode="full") * dx
    G = grad_profile(h, n)
    
    # Variables: [d_0, ..., d_{n-1}, tau]
    # Minimize tau
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    # Constraints: G_m · d - tau <= -P_m  for all m
    # i.e., -G_m · d + tau >= P_m
    # In linprog form (A_ub @ x <= b_ub): G_m · d - tau <= -P_m
    A_ub = np.zeros((M, n + 1))
    A_ub[:, :n] = G
    A_ub[:, -1] = -1.0
    b_ub = -P
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    # Bounds on d_i: max(-delta, -h_i) <= d_i <= min(delta, 1-h_i)
    bounds = []
    for i in range(n):
        lo = max(-delta, -h[i])
        hi = min(delta, 1.0 - h[i])
        bounds.append((lo, hi))
    bounds.append((0.0, c + 2.0))  # tau >= 0, upper loose
    # Actually tau can be anything, but let's keep it bounded
    bounds[-1] = (-1.0, c + 2.0)
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x is not None:
            d = res.x[:n]
            tau = res.x[-1]
            if np.isfinite(d).all() and np.isfinite(tau):
                return d, tau
    except Exception:
        pass
    return None, None

def lp_local_optimize(h, max_iters=200, delta_init=0.05, time_limit_fn=None):
    """Local optimization using LP trust region steps"""
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    delta = delta_init
    stall = 0
    max_stall = 15
    
    for it in range(max_iters):
        if time_limit_fn is not None and time_limit_fn() < 5:
            break
        
        d, tau = lp_trust_step(h, best_c, delta)
        if d is None:
            # Fallback: try smaller delta
            delta *= 0.5
            if delta < 1e-5:
                break
            stall += 1
            if stall > max_stall:
                break
            continue
        
        # Try the step with line search
        improved = False
        h_new = h
        c_new = best_c
        
        # Try full step first, then fractions
        for alpha in [1.0, 0.7, 0.5, 0.3, 0.15, 0.05]:
            h_try = repair_feasible(h + alpha * d)
            c_try = exact_score(h_try)
            if c_try < c_new - 1e-14:
                h_new = h_try
                c_new = c_try
                improved = True
        if improved:
            h = h_new
            if c_new < best_c:
                best_c = c_new
            stall = 0
            # Adjust delta
            if alpha >= 0.5:
                delta = min(2.0 * delta, 0.4)
            elif alpha >= 0.15:
                delta = min(1.3 * delta, 0.3)
            else:
                delta = max(0.7 * delta, 0.01)
        else:
            delta *= 0.6
            stall += 1
            if delta < 1e-5 or stall > max_stall:
                break
    
    return h, best_c

def gradient_descent_fallback(h, max_iters=100, tol=0.003):
    """Simple gradient descent fallback when LP fails"""
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    momentum_vec = np.zeros(n)
    momentum = 0.3
    
    for it in range(max_iters):
        profile = compute_profile(h)
        c = float(profile.max())
        thresh = c - tol
        active = np.where(profile >= thresh)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(profile))])
        if len(active) > 20:
            top = np.argsort(profile[active])[-20:]
            active = active[top]
        
        g = np.zeros(n)
        for m in active:
            d = m - (n - 1)
            if d == 0:
                g += 2.0 / n * (1.0 - 2.0 * h)
            elif d > 0:
                g[d:] += (2.0 / n) * (1.0 - h[:-d])
                g[:-d] -= (2.0 / n) * h[d:]
            else:
                p = -d
                g[:-p] += (2.0 / n) * (1.0 - h[p:])
                g[p:] -= (2.0 / n) * h[:-p]
        
        g -= g.mean()
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            break
        
        momentum_vec = momentum * momentum_vec + g
        mv_norm = np.linalg.norm(momentum_vec)
        if mv_norm < 1e-15:
            break
        d = -momentum_vec / mv_norm
        
        pos_mask = d > 1e-12
        neg_mask = d < -1e-12
        alpha_max = 0.5
        if pos_mask.any():
            alpha_max = min(alpha_max, float(((1.0 - h[pos_mask]) / d[pos_mask]).min()))
        if neg_mask.any():
            alpha_max = min(alpha_max, float((h[neg_mask] / (-d[neg_mask])).min()))
        alpha_max = max(alpha_max, 1e-10)
        
        best_alpha = 0.0
        best_val = c
        best_h = h
        a_ls, b_ls = 0.0, alpha_max
        for _ in range(15):
            c1 = b_ls - gr * (b_ls - a_ls)
            c2 = a_ls + gr * (b_ls - a_ls)
            h1 = repair_feasible(h + c1 * d)
            h2 = repair_feasible(h + c2 * d)
            f1 = exact_score(h1)
            f2 = exact_score(h2)
            if f1 < best_val:
                best_val = f1
                best_alpha = c1
                best_h = h1.copy()
            if f2 < best_val:
                best_val = f2
                best_alpha = c2
                best_h = h2.copy()
            if f1 < f2:
                b_ls = c2
            else:
                a_ls = c1
            if b_ls - a_ls < 1e-13:
                break
        
        if best_val >= c - 1e-15:
            break
        h = best_h
        if best_val < best_c:
            best_c = best_val
    
    return h, best_c

def interp_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    edges_old = np.linspace(0.0, 2.0, n + 1)
    edges_new = np.linspace(0.0, 2.0, new_n + 1)
    h_new = np.zeros(new_n)
    for k in range(new_n):
        a, b = edges_new[k], edges_new[k + 1]
        total = 0.0
        for j in range(n):
            lo = max(a, edges_old[j])
            hi = min(b, edges_old[j + 1])
            if hi > lo:
                total += h[j] * (hi - lo)
        h_new[k] = total / (b - a)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

def crossover(h_a, h_b, t):
    """Convex combination - automatically feasible if both are feasible and same length"""
    h = (1 - t) * h_a + t * h_b
    return repair_feasible(h)

def binary_seed(n, pattern_type, rng):
    half = n / 2.0
    h = np.zeros(n)
    if pattern_type == 0:
        idx = np.arange(n)
        vals = (idx * 0.618033988749895) % 1.0
        top = np.argsort(vals)[-int(half):]
        h[top] = 1.0
    elif pattern_type == 1:
        idx = np.arange(n)
        vals = (idx * 0.4142135623730951) % 1.0
        top = np.argsort(vals)[-int(half):]
        h[top] = 1.0
    elif pattern_type == 2:
        period = max(3, n // 8)
        for i in range(n):
            if (i // period) % 2 == 0:
                h[i] = 1.0
        ones = h.sum()
        if ones > half:
            excess = int(ones - half)
            one_idx = np.where(h == 1.0)[0]
            to_flip = rng.choice(one_idx, min(excess, len(one_idx)), replace=False)
            h[to_flip] = 0.0
        elif ones < half:
            deficit = int(half - ones)
            zero_idx = np.where(h == 0.0)[0]
            to_flip = rng.choice(zero_idx, min(deficit, len(zero_idx)), replace=False)
            h[to_flip] = 1.0
    elif pattern_type == 3:
        h = (np.sin(np.linspace(0, 6 * np.pi, n)) > 0).astype(float)
        ones = h.sum()
        if ones > half:
            excess = int(ones - half)
            one_idx = np.where(h == 1.0)[0]
            to_flip = rng.choice(one_idx, min(excess, len(one_idx)), replace=False)
            h[to_flip] = 0.0
        elif ones < half:
            deficit = int(half - ones)
            zero_idx = np.where(h == 0.0)[0]
            to_flip = rng.choice(zero_idx, min(deficit, len(zero_idx)), replace=False)
            h[to_flip] = 1.0
    elif pattern_type == 4:
        # Golden ratio blocking
        idx = np.arange(n)
        phase = idx / n * 2.0 * np.pi
        h = (np.sin(phase * 5.0 + 0.5) > 0.1).astype(float)
    else:
        perm = rng.permutation(n)
        h[perm[:int(half)]] = 1.0
    
    h = repair_feasible(h)
    return h

def smooth_binary(h, window=2):
    """Apply box smoothing then repair"""
    n = len(h)
    h_smooth = np.zeros(n)
    for i in range(n):
        lo = max(0, i - window)
        hi = min(n, i + window + 1)
        h_smooth[i] = h[lo:hi].mean()
    return repair_feasible(h_smooth)

def repair_feasible(vec, target=None):
    n = len(vec)
    if target is None:
        target = n / 2.0
    h = np.asarray(vec, dtype=np.float64).copy()
    h[~np.isfinite(h)] = 0.5
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(-(1.0 - h))
        remaining = diff
        for i in range(n):
            idx = order[i]
            cap = 1.0 - h[idx]
            if cap <= 1e-15:
                continue
            add = min(remaining, cap)
            h[idx] += add
            remaining -= add
            if remaining <= 1e-14:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            if h[idx] <= 1e-15:
                continue
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-14:
                break
    r = target - h.sum()
    if abs(r) > 1e-13:
        idx = int(np.argmax(1.0 - h)) if r > 0 else int(np.argmax(h))
        h[idx] = float(np.clip(h[idx] + r, 0.0, 1.0))
    return h


def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))


def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx


def grad_profile(h, n):
    dx = 2.0 / n
    M = 2 * n - 1
    G = np.zeros((M, n))
    for m in range(M):
        d = m - (n - 1)
        if d == 0:
            G[m, :] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            G[m, d:] += dx * (1.0 - h[:-d])
            G[m, :-d] -= dx * h[d:]
        else:
            p = -d
            G[m, :-p] += dx * (1.0 - h[p:])
            G[m, p:] -= dx * h[:-p]
    return G


def lp_trust_step(h, c, delta=0.05):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    P = np.correlate(h, 1 - h, mode="full") * dx
    G = grad_profile(h, n)
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_ub = np.zeros((M, n + 1))
    A_ub[:, :n] = G
    A_ub[:, -1] = -1.0
    b_ub = -P
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    bounds = []
    for i in range(n):
        lo = max(-delta, -h[i])
        hi = min(delta, 1.0 - h[i])
        bounds.append((lo, hi))
    bounds.append((-1.0, c + 2.0))
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
        if res.success and res.x is not None and np.isfinite(res.x).all():
            return res.x[:n], res.x[-1]
    except Exception:
        pass
    return None, None

def interp_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    edges_old = np.linspace(0.0, 2.0, n + 1)
    edges_new = np.linspace(0.0, 2.0, new_n + 1)
    h_new = np.zeros(new_n)
    for k in range(new_n):
        a, b = edges_new[k], edges_new[k + 1]
        total = 0.0
        for j in range(n):
            lo = max(a, edges_old[j])
            hi = min(b, edges_old[j + 1])
            if hi > lo:
                total += h[j] * (hi - lo)
        h_new[k] = total / (b - a)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)


def crossover_mix(h_a, h_b, t):

        vals = (idx * 0.4142135623730951) % 1.0
        top = np.argsort(vals)[-int(half):]
        h[top] = 1.0
    elif pattern_type == 2:
        period = max(3, n // 8)
        for i in range(n):
            if (i // period) % 2 == 0:
                h[i] = 1.0
    elif pattern_type == 3:
        h = (np.sin(np.linspace(0, 6 * np.pi, n)) > 0).astype(float)
    elif pattern_type == 4:
        h = (np.sin(5.0 * np.pi * idx / n + 0.7) > 0.1).astype(float)
    elif pattern_type == 5:
        freq = 3.0 + rng.random() * 4.0
        phase = rng.random() * 2.0 * np.pi
        h = (np.sin(freq * np.pi * idx / n + phase) > 0).astype(float)
    else:
        perm = rng.permutation(n)
        h[perm[:int(half)]] = 1.0
    s = h.sum()
    if s != half:
        if s > half:
            excess = int(s - half)
            one_idx = np.where(h > 0.5)[0]
            if len(one_idx) > excess:
                h[rng.choice(one_idx, excess, replace=False)] = 0.0
        else:
            deficit = int(half - s)
            zero_idx = np.where(h < 0.5)[0]
            if len(zero_idx) > deficit:
                h[rng.choice(zero_idx, min(deficit, len(zero_idx)), replace=False)] = 1.0
    return repair_feasible(h)
