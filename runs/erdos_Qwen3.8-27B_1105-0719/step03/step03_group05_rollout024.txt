def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full")) * dx)

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx] + 1e-15)
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx] + 1e-15)
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-14:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmin(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = np.linspace(0, 2, n + 1)
    xt = np.linspace(0, 2, new_n + 1)
    h_interp = np.zeros(new_n)
    for k in range(new_n):
        lo_k, hi_k = xt[k], xt[k+1]
        total_mass = 0.0
        total_width = 0.0
        for j in range(n):
            lo_j, hi_j = xs[j], xs[j+1]
            overlap = min(hi_k, hi_j) - max(lo_k, lo_j)
            if overlap > 0:
                total_mass += h[j] * overlap
                total_width += overlap
        h_interp[k] = total_mass / total_width if total_width > 1e-15 else 0.5
    h_interp = np.clip(h_interp, 0.0, 1.0)

def coordinate_descent(h, max_sweeps=40, rng=None, time_fn=None, time_limit=0):

    n_pairs = len(all_pairs)
    order = np.arange(n_pairs)
    for sweep in range(max_sweeps):
        if time_fn is not None and time_fn() < time_limit:

        for pi in order:

            if dhi - dlo < 1e-12:

            f_cur = float(np.max(profile))

    c_final = exact_score(h)
    if c_final < best_c:
        best_c = c_final

            break
        grad /= gnorm
        improved = False
        for a in [alpha, alpha*0.5, alpha*0.25, alpha*0.1]:
            h_try = repair_feasible(h - a * grad)
            s = exact_score(h_try)
            if s < best_c - 1e-14:
                h = h_try
                best_c = s
                improved = True
                alpha = min(alpha * 1.08, 0.02)
                stuck = 0
                break
        if not improved:
            alpha *= 0.5
            stuck += 1
            if stuck > 30 or alpha < 1e-10:
                break
    return h, best_c

def slsqp_solve(h0, maxiter=200):
    n = len(h0)
    dx = 2.0 / n
    M = 2 * n - 1
    c0 = exact_score(h0)
    def obj(x):
        return x[-1]
    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    def eq_fun(x):
        return x[:-1].sum() - n / 2.0
    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g
    def ineq_fun(x):
        h = np.clip(x[:-1], 0, 1)
        t = x[-1]
        profile = np.correlate(h, 1 - h, mode="full") * dx
        return t - profile
    def ineq_jac(x):
        h = np.clip(x[:-1], 0, 1)
        G = np.zeros((M, n))
        for m in range(M):
            d = m - (n - 1)
            if d == 0:
                G[m, :] = dx * (1.0 - 2.0 * h)
            elif d > 0:
                G[m, d:] += dx * (1.0 - h[0:n-d])
                G[m, 0:n-d] += -dx * h[d:n]
            else:
                dp = -d
                G[m, 0:n-dp] += dx * (1.0 - h[dp:n])
                G[m, dp:n] += -dx * h[0:n-dp]
        J = np.zeros((M, n + 1))
        J[:, :-1] = -G
        J[:, -1] = 1.0
        return J
    x0 = np.concatenate([h0, [c0 + 1e-3]])
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 0.5)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-13})
        if res.x is not None and np.isfinite(res.x).all():
            h_new = repair_feasible(res.x[:-1])
            c_new = exact_score(h_new)
            if c_new <= c0 + 1e-9:
                return h_new, c_new
    except Exception:
        pass
    return None, None

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s + t_start - time.perf_counter()

    parent = None
    try:
        parent = np.array(initial_h_values, dtype=np.float64)
    except NameError:
        pass
    except Exception:
        pass

    if parent is None or len(parent) < 4:
        parent = thue_morse_init(70)

    parent = np.clip(parent, 0.0, 1.0)
    parent = repair_feasible(parent)

    best_h = parent.copy()
    best_c = exact_score(parent)

    # Try symmetries
    h_sym = symmetrize(parent)
    c_sym = exact_score(h_sym)
    if c_sym < best_c - 1e-12:
        best_h = h_sym
        best_c = c_sym

    n_parent = len(best_h)

    # Stage 1: Coordinate descent at parent resolution
    if time_left() > 200:
        h_cd, c_cd = coordinate_descent(best_h, max_sweeps=50, rng=rng,
                                        time_fn=time_left, time_limit=60)
        if c_cd < best_c - 1e-13:
            best_h = h_cd.copy()
            best_c = c_cd

    # LSE polish
    if time_left() > 150:
        h_lse, c_lse = lse_descent(best_h, beta=80, max_iters=100,
                                   time_fn=time_left, time_limit=60)
        if c_lse < best_c - 1e-13:
            best_h = h_lse.copy()
            best_c = c_lse

    # Stage 2: Resolution exploration
    if time_left() > 300:
        dims_to_try = []
        for f in [0.8, 0.85, 0.9, 0.95, 1.05, 1.1, 1.15, 1.25, 1.4]:
            d = int(round(n_parent * f / 2)) * 2
            d = max(36, min(d, 160))
            if d != n_parent:
                dims_to_try.append(d)
        # Also try fixed good dimensions
        for d in [48, 56, 64, 72, 80, 96, 112, 128]:
            if d != n_parent and d not in dims_to_try:
                dims_to_try.append(d)
        dims_to_try = sorted(set(dims_to_try), key=lambda x: abs(x - n_parent))

        for new_n in dims_to_try[:8]:
            if time_left() < 80:
                break
            h_interp = interpolate_to_n(best_h, new_n)
            h_cd, c_cd = coordinate_descent(h_interp, max_sweeps=35, rng=rng,
                                            time_fn=time_left, time_limit=40)
            if c_cd < best_c - 1e-13:
                best_h = h_cd.copy()
                best_c = c_cd
                n_parent = new_n
            # LSE from this
            if time_left() > 80:
                h_lse, c_lse = lse_descent(best_h, beta=100, max_iters=60,
                                           time_fn=time_left, time_limit=30)
                if c_lse < best_c - 1e-13:
                    best_h = h_lse
                    best_c = c_lse

    # Stage 3: Thue-Morse based starts
    if time_left() > 150:
        n_try = len(best_h)
        for tm_n in [n_try, n_try - 2, n_try + 2, n_try + 4]:
            if time_left() < 40:
                break
            if tm_n < 36 or tm_n > 160 or tm_n % 2 != 0:
                continue
            h_tm = thue_morse_init(tm_n)
            # Also try complement
            h_tm_comp = repair_feasible(1.0 - h_tm)
            for h_init in [h_tm, h_tm_comp]:
                if time_left() < 30:
                    break
                h_cd, c_cd = coordinate_descent(h_init, max_sweeps=30, rng=rng,
                                                time_fn=time_left, time_limit=25)
                if c_cd < best_c - 1e-13:
                    best_h = h_cd.copy()
                    best_c = c_cd

    # Stage 4: SLSQP on best
    if time_left() > 80:
        h_sol, c_sol = slsqp_solve(best_h, maxiter=200)
        if h_sol is not None and c_sol < best_c - 1e-12:
            best_h = h_sol.copy()
            best_c = c_sol
            # Polish with coordinate descent
            h_cd, c_cd = coordinate_descent(best_h, max_sweeps=20, rng=rng,
                                            time_fn=time_left, time_limit=30)
            if c_cd < best_c - 1e-13:
                best_h = h_cd
                best_c = c_cd

    # Stage 5: Multi-start with perturbations
    if time_left() > 120:
        n_b = len(best_h)
        starts = [best_h.copy()]
        for eps in [0.003, 0.008, 0.015, 0.03, 0.06]:
            z = rng.randn(n_b)
            z -= z.mean()
            starts.append(repair_feasible(best_h + eps * z))
        # Symmetric perturbation
        z = rng.randn(n_b)
        z_s = (z + z[::-1]) * 0.5
        z_s -= z_s.mean()
        starts.append(repair_feasible(best_h + 0.01 * z_s))

        for h0 in starts[:6]:
            if time_left() < 35:
                break
            h_cd, c_cd = coordinate_descent(h0, max_sweeps=25, rng=rng,
                                            time_fn=time_left, time_limit=25)
            if c_cd < best_c - 1e-13:
                best_h = h_cd.copy()
                best_c = c_cd
            if time_left() > 40:
                h_sol, c_sol = slsqp_solve(h_cd, maxiter=100)
                if h_sol is not None and c_sol < best_c - 1e-12:
                    best_h = h_sol
                    best_c = c_sol

    # Stage 6: More coordinate descent at different resolutions from best
    if time_left() > 100:
        n_b = len(best_h)
        for dn in [-4, -2, 2, 4, 6, 8]:
            if time_left() < 30:
                break
            new_n = n_b + dn
            if new_n < 36 or new_n > 180 or new_n % 2 != 0:
                continue
            h_interp = interpolate_to_n(best_h, new_n)
            h_cd, c_cd = coordinate_descent(h_interp, max_sweeps=30, rng=rng,
                                            time_fn=time_left, time_limit=25)
            if c_cd < best_c - 1e-13:
                best_h = h_cd
                best_c = c_cd

    # Stage 7: Final LSE + coordinate descent polish
    if time_left() > 40:
        h_lse, c_lse = lse_descent(best_h, beta=120, max_iters=80,
                                   time_fn=time_left, time_limit=20)
        if c_lse < best_c - 1e-13:
            best_h = h_lse
            best_c = c_lse
        h_cd, c_cd = coordinate_descent(best_h, max_sweeps=30, rng=rng,
                                        time_fn=time_left, time_limit=15)
        if c_cd < best_c - 1e-13:
            best_h = h_cd
            best_c = c_cd

    # Stage 8: Final SLSQP
    if time_left() > 35:
        h_sol, c_sol = slsqp_solve(best_h, maxiter=300)
        if h_sol is not None and c_sol < best_c - 1e-14:
            best_h = h_sol
            best_c = c_sol

    # Final validation
    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)

    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            final_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 2:
            final_c = float(eval_result[0])
    except Exception:
        pass

    return best_h, final_c, len(best_h)
