def epigraph_slsqp(h0, dx, target_sum, active_set, maxiter=500):
    n = len(h0)
    
    def unpack(z):
        h = z[:n]
        tau = z[n]
        return h, tau
    
    def obj(z):
        h, tau = unpack(z)
        return tau
    
    def grad_obj(z):
        g = np.zeros(n+1)
        g[n] = 1.0
        return g
    
    def cons_func(z):
        h, tau = unpack(z)
        h_clip = np.clip(h, 0, 1)
        corrs = compute_corrs(h_clip)
        # tau - dx*corrs[k] >= 0 for k in active_set
        return tau - dx * corrs[active_set]
    
    def cons_grad(z):
        h, tau = unpack(z)
        h_clip = np.clip(h, 0, 1)
        g = np.zeros((len(active_set), n+1))
        for idx, k in enumerate(active_set):
            g[idx, n] = 1.0
            g[idx, :n] = -dx * grad_corr_k(h_clip, k, n)
        return g
    
    def eq_func(z):
        h, tau = unpack(z)
        return h.sum() - target_sum
    
    def eq_grad(z):
        g = np.zeros(n+1)
        g[:n] = 1.0
        return g
    
    z0 = np.concatenate([h0, [c5_of_h(h0, dx) + 0.01]])
    bounds = [(0, 1)] * n + [(0, None)]
    constraints = [
        {'type': 'ineq', 'fun': cons_func, 'jac': cons_grad},
        {'type': 'eq', 'fun': eq_func, 'jac': eq_grad}
    ]
    
    res = minimize(obj, z0, jac=grad_obj, method='SLSQP',
                   bounds=bounds, constraints=constraints,
                   options={'maxiter': maxiter, 'ftol': 1e-16})
    
    h_out = project_feasible(res.x[:n], target_sum)
    return h_out, c5_of_h(h_out, dx)

def project_feasible(x, target_sum):
    # Project x onto {0 <= h_i <= 1, sum(h) = target_sum}
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) / 2, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def c5_of_h(h, dx):
    return compute_corrs(h).max() * dx

def grad_corr_k(h, k, n):
    """Gradient of corrs[k] w.r.t. h"""
    offset = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -offset)
    i_hi = min(n - 1, n - 1 - offset)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx + offset] += (1.0 - h[idx])
    grad[idx] += (-h[idx + offset])
    return grad

def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-13:
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)


def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def c5_of_h(h, dx):
    return compute_corrs(h).max() * dx


def grad_corr_k(h, k, n):
    offset = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -offset)
    i_hi = min(n - 1, n - 1 - offset)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx + offset] += (1.0 - h[idx])
    grad[idx] += (-h[idx + offset])
    return grad


def epigraph_solve(h0, dx, target_sum, n_active=20, maxiter=800):

    top_k = np.sort(top_k)
    
    z0 = np.concatenate([h0, [corrs.max() * dx + 0.005]])
    
    def obj(z):
        return z[n]
    
    def grad_obj(z):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g
    
    def ineq_func(z):

        tau = z[n]

    def ineq_grad(z):
        h = np.clip(z[:n], 0.0, 1.0)
        g = np.zeros((len(top_k), n + 1))

            g[idx, n] = 1.0
            g[idx, :n] = -dx * grad_corr_k(h, k, n)
        return g
    
    def eq_func(z):
        return z[:n].sum() - target_sum
    
    def eq_grad(z):
        g = np.zeros(n + 1)
        g[:n] = 1.0
        return g
    
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constraints = [
        {'type': 'ineq', 'fun': ineq_func, 'jac': ineq_grad},
        {'type': 'eq', 'fun': eq_func, 'jac': eq_grad}
    ]
    
    try:
        res = minimize(obj, z0, jac=grad_obj, method='SLSQP',
                       bounds=bounds, constraints=constraints,

    except Exception:
        h_out = project_feasible(h0, target_sum)
    
    return h_out, c5_of_h(h_out, dx)
