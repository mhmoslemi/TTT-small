```python
def binary_score(bits, n):
    """Compute max disagreement for binary bitmap of length n."""
    h = (bits & ((1 << n) - 1))  # ensure n bits
    max_disc = 0
    for d in range(1, n):
        # h[i]=1 and h[i+d]=0 means: h has bit i set, h>>d has bit i unset
        # Valid i: 0 <= i < n-d
        valid_mask = (1 << (n-d)) - 1
        disc = (h & ~(h >> d) & valid_mask).bit_count()

    return max_disc
```

```python
def get_grad_for_lag(h, off, n):
    """Compute gradient of c_k with respect to h, where off = k - (n-1)."""
    grad = np.zeros(n)
    # c_k = sum_{i: i valid, i+off valid} h[i] * (1-h[i+off])
    # dc_k/dh[j] = (1-h[j+off]) if j+off in [0,n-1] and j in valid first-index range
    #             - h[j-off] if j-off in [0,n-1] and j-off in valid first-index range

    i_lo = max(0, -off)
    i_hi = min(n-1, n-1-off)
    
    # First term: j = i, so j in [i_lo, i_hi], contributes (1-h[j+off])
    if i_lo <= i_hi:

        grad[a] += (1.0 - h[a + off])

    j_lo = max(0, i_lo + off)
```

```python
            # = min_t max_k (corrs[k] + t*g[k])  (since cur[i]*g[k] cancels... wait no)
            # 
            # Actually: c_k(t) = c_k(cur) + (t - cur[i]) * g[k]
            # = (c_k(cur) - cur[i]*g[k]) + t*g[k]
            # Let a_k = c_k(cur) - cur[i]*g[k], b_k = g[k]
            # Then c_k(t) = a_k + t*b_k
            # We want: min_{t in [0,1]} max_k (a_k + t*b_k)
            
            a = corrs - cur[i] * g
            b = g
            
            # Find t* = argmin max_k (a_k + t*b_k) for t in [0,1]
            # Use ternary search (function is convex)
            lo, hi = 0.0, 1.0
            for _ in range(50):
                m1 = lo + (hi-lo)/3
                m2 = hi - (hi-lo)/3
                f1 = np.max(a + m1*b)
                f2 = np.max(a + m2*b)
                if f1 < f2:
                    hi = m2
                else:
                    lo = m1
            t_opt = (lo+hi)/2
            
            # Check if this improves
            if abs(t_opt - cur[i]) > 1e-10:
                cur[i] = t_opt
                # Project to maintain sum
                cur = project_feasible(cur, target_sum)
                cur_c5 = c5_of_h(cur, dx)
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = cur.copy()
        
    return best_h, best_c5
```

```python
def compute_coefficients(h, i, n):
    """Compute g[k] = dc_k/dh[i] for all k."""
    g = np.zeros(2*n-1)
    
    # Term 1: off in [-i, n-1-i], k in [n-1-i, 2n-2-i]
    k1_lo = n-1-i
    k1_hi = 2*n-2-i
    if k1_lo <= k1_hi:
        offs1 = np.arange(k1_lo - (n-1), k1_hi - (n-1) + 1)
        g[k1_lo:k1_hi+1] += (1.0 - h[i + offs1])
    
    # Term 2: off in [i-n+1, i], k in [i, i+n-1]
    k2_lo = i
    k2_hi = i + n - 1
    if k2_lo <= k2_hi:
        offs2 = np.arange(k2_lo - (n-1), k2_hi - (n-1) + 1)
        g[k2_lo:k2_hi+1] += (-h[i - offs2])
    
    return g
```

This is O(n) per coordinate, fully vectorized.

```python
def binary_search_t(a, b, n_iter=50):
    """Find t in [0,1] minimizing max_k(a_k + t*b_k)."""
    lo, hi = 0.0, 1.0
    for _ in range(n_iter):
        m1 = lo + (hi-lo)/3.0
        m2 = hi - (hi-lo)/3.0
        if np.max(a + m1*b) < np.max(a + m2*b):
            hi = m2
        else:
            lo = m1
    return (lo+hi)/2.0
```
