    n = len(h)
    m = n // 2
    y = np.zeros(m)
    for j in range(m):
        y[j] = 0.5 * (h[j] + 1.0 - h[n-1-j])
    y = np.clip(y, 0.0, 1.0)
    h_new = np.empty(n)
    h_new[:m] = y
    h_new[m:] = 1.0 - y[::-1]
    return h_new

def anti_descent_single(h, max_iters=200, n_scan=25):
    """Coordinate descent in anti-symmetric reduced space"""
    n = len(h)
    m = n // 2
    h = h.copy()
    # Ensure anti-symmetric
    h = anti_project(h)
    best_c = exact_score(h)
    improved = True
    it = 0
    while improved and it < max_iters:
        improved = False
        it += 1
        for k in range(m):
            y_k = h[k]
            # Scan the interval [0, 1] for y[k]
            # h[k] = y_k, h[n-1-k] = 1 - y_k
            ts = np.linspace(0.0, 1.0, n_scan)

            for t in ts:
                if abs(t - y_k) < 1e-6:

                h[k] = t
                h[n-1-k] = 1.0 - t

                    best_t = t
            if best_s < best_c - 1e-14:

                h[n-1-k] = 1.0 - best_t
                best_c = best_s
                improved = True
            else:

                h[n-1-k] = 1.0 - y_k
    return h, best_c

import numpy as np
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full")) * dx)

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx] + 1e-15)
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx] + 1e-15)
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    return np.clip(h, 0.0, 1.0)

def anti_project(h):
    n = len(h)
    m = n // 2
    y = 0.5 * (h[:m] + 1.0 - h[n-1:n-m-1:-1])
    y = np.clip(y, 0.0, 1.0)
    h_new = np.empty(n)
    h_new[:m] = y
    h_new[m:] = 1.0 - y[::-1]
    return h_new

def build_G(h, n, dx):
    M = 2 * n - 1
    G = np.zeros((M, n))
    for q in range(M):
        d = q - (n - 1)
        if d == 0:
            G[q, :] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            G[q, d:] += dx * (1.0 - h[:n-d])
            G[q, :n-d] += -dx * h[d:]
        else:
            p = -d
            G[q, :n-p] += dx * (1.0 - h[p:])
            G[q, p:] += -dx * h[:n-p]
    return G

def golden_1d_ms(base_vec, slope_vec, a, b, max_iter=50, tol=1e-12):
    phi = (np.sqrt(5.0) - 1.0) / 2.0
    if b - a < tol:
        return 0.5 * (a + b)
    c = b - phi * (b - a)
    d = a + phi * (b - a)
    fc = np.max(base_vec + c * slope_vec)
    fd = np.max(base_vec + d * slope_vec)
    for _ in range(max_iter):
        if abs(b - a) < tol:
            break
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - phi * (b - a)
            fc = np.max(base_vec + c * slope_vec)
        else:
            a = c
            c = d
            fc = fd
            d = a + phi * (b - a)
            fd = np.max(base_vec + d * slope_vec)
    return 0.5 * (a + b)

def anti_coord_descent(h, max_sweeps=200):
    n = len(h)
    m = n // 2
    dx = 2.0 / n
    h = anti_project(h.copy())
    best_c = exact_score(h)
    for sweep in range(max_sweeps):
        G = build_G(h, n, dx)
        G_red = G[:, :m] - G[:, n-1:n-m-1:-1]
        base = compute_profile(h)
        any_imp = False
        for k in range(m):
            slopes = G_red[:, k]
            t0 = h[k]
            u_opt = golden_1d_ms(base, slopes, -t0, 1.0 - t0)
            t_opt = t0 + u_opt
            if abs(t_opt - t0) > 1e-10:
                h[k] = t_opt
                h[n-1-k] = 1.0 - t_opt
                new_c = exact_score(h)
                if new_c < best_c - 1e-14:
                    best_c = new_c
                    any_imp = True
                else:
                    h[k] = t0
                    h[n-1-k] = 1.0 - t0
        if not any_imp:
            break
    return h, best_c
