1. Compute gradient g

The projection is the key challenge. The feasible set is {h : 0 <= h[i] <= 1, sum(h) = n/2}.

1. Find lambda such that sum(clip(z - lambda, 0, 1)) = n/2
2. Return clip(z - lambda, 0, 1)

def project_feasible(x, target_sum):
    """Project x onto {h: 0<=h[i]<=1, sum(h)=target_sum}"""
    n = len(x)
    # Bisection on lambda
    lo = x.min() - 2.0
    hi = x.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h = np.clip(x - mid, 0.0, 1.0)
        s = h.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    # Fix residual
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def project_feasible(x, target_sum):
    n = len(x)
    lo = x.min() - 2.0
    hi = x.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def c5_of_h(h, dx):
    return np.correlate(h, 1.0 - h, mode='full').max() * dx

def smooth_pow_gradient(h, q):
    n = len(h)
    R = np.correlate(h, 1.0 - h, mode='full')
    R_safe = np.maximum(R, 0.0)
    max_R = max(R_safe.max(), 1e-30)
    R_norm = R_safe / max_R
    w = np.zeros_like(R)
    pos = R_norm > 1e-15
    w[pos] = R_norm[pos] ** (q - 1)
    g = np.zeros(n)
    w0 = w[n - 1]
    g += w0 * (1.0 - 2.0 * h)
    for d in range(1, n):
        K = n - d
        wp = w[n - 1 - d]
        wm = w[n - 1 + d]
        if wp > 1e-15:
            g[:K] += wp * (1.0 - h[d:])
            g[d:] += wp * (-h[:K])
        if wm > 1e-15:
            g[d:] += wm * (1.0 - h[:K])
            g[:K] += wm * (-h[d:])
    g *= q
    g -= g.mean()
    return g

            best_c5 = c5_opt
            best_h = h_opt
            h = h_opt.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def tl():
        return budget_s - (time.time() - t0)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = np.full(160, 0.5)

    n = len(h0)
    dx = 2.0 / n
    target_sum = n / 2.0
    h0 = project_feasible(h0, target_sum)
    best_h = h0.copy()
    best_c5 = c5_of_h(h0, dx)

    # Phase 1: Moderate-q smooth gradient (broad improvements)
    if tl() > 60:
        h1, c5_1 = smooth_pow_descent(best_h, dx, target_sum, [4, 6, 8, 12], 400, t0, budget_s, 50.0)
        if c5_1 < best_c5:
            best_c5 = c5_1
            best_h = h1

    # Phase 2: Higher-q smooth gradient (focused)
    if tl() > 50:
        h2, c5_2 = smooth_pow_descent(best_h, dx, target_sum, [16, 24, 32], 400, t0, budget_s, 40.0)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2

    # Phase 3: Pair descent for exact local improvements
    if tl() > 40:
        h3, c5_3 = pair_descent(best_h, dx, target_sum, rng, t0, budget_s, 30.0, n_passes=100, n_tries_per_pass=400)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3

    # Phase 4: Very high-q smooth gradient (fine tuning)
    if tl() > 40:
        h4, c5_4 = smooth_pow_descent(best_h, dx, target_sum, [48, 64, 96, 128], 500, t0, budget_s, 30.0)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4

    # Phase 5: More pair descent
    if tl() > 30:
        h5, c5_5 = pair_descent(best_h, dx, target_sum, rng, t0, budget_s, 25.0, n_passes=100, n_tries_per_pass=500)
        if c5_5 < best_c5:
            best_c5 = c5_5
            best_h = h5

    # Phase 6: Perturbation + re-optimization
    if tl() > 50:
        h6, c5_6 = perturbed_reopt(best_h, dx, target_sum, rng, t0, budget_s, 40.0, n_pert=10)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6

    # Phase 7: Grid size exploration
    if tl() > 50:
        n_cur = len(best_h)
        n_candidates = sorted(set([
            n_cur - 20, n_cur - 10, n_cur + 10, n_cur + 20,
            120, 130, 140, 150, 170, 180, 190, 200, 210, 220
        ]))
        n_candidates = [nc for nc in n_candidates if 80 <= nc <= 240 and nc % 2 == 0]
        for nc in n_candidates:
            if tl() < 35:
                break
            if nc == n_cur:
                continue
            dx_t = 2.0 / nc
            ts_t = nc / 2.0
            h_interp = project_feasible(transfer_grid(best_h, nc), ts_t)
            h_opt, c5_opt = smooth_pow_descent(h_interp, dx_t, ts_t, [8, 16, 32, 64], 300, t0, budget_s, 25.0)
            if tl() > 20:
                h_opt2, c5_opt2 = pair_descent(h_opt, dx_t, ts_t, rng, t0, budget_s, 20.0, n_passes=50, n_tries_per_pass=300)
                if c5_opt2 < c5_opt:
                    h_opt, c5_opt = h_opt2, c5_opt2
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt

    # Phase 8: Final intensive polish on best grid
    if tl() > 20:
        n_f = len(best_h)
        dx_f = 2.0 / n_f
        ts_f = n_f / 2.0
        h8, c5_8 = smooth_pow_descent(best_h, dx_f, ts_f, [64, 128, 256], 300, t0, budget_s, 15.0)
        if c5_8 < best_c5:
            best_c5 = c5_8
            best_h = h8

    # Phase 9: Final pair polish
    if tl() > 15:
        n_f = len(best_h)
        dx_f = 2.0 / n_f
        ts_f = n_f / 2.0
        h9, c5_9 = pair_descent(best_h, dx_f, ts_f, rng, t0, budget_s, 10.0, n_passes=100, n_tries_per_pass=500)
        if c5_9 < best_c5:
            best_c5 = c5_9
            best_h = h9

    # Final validation
    n_final = len(best_h)
    dx_final = 2.0 / n_final
    ts_final = n_final / 2.0
    best_h = project_feasible(best_h, ts_final)
    best_c5 = c5_of_h(best_h, dx_final)

    return best_h, best_c5, n_final
