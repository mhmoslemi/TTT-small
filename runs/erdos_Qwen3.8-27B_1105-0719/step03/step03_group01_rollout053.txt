import numpy as np
from scipy.optimize import minimize

def project_feasible(x, target_sum):
    """Project x onto {h : 0<=h_i<=1, sum(h)=target_sum}"""
    n = len(x)
    x = np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0)
    # Bisection on lambda: sum(clip(x - lambda, 0, 1)) = target_sum
    lo, hi = -1.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h = np.clip(x - lam, 0.0, 1.0)
    # Fix remaining error
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        cnt = mask.sum()
        if cnt > 0:
            h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * dx, corr

    return corr.max() * dx

def grad_corr_all(h):
    """Compute gradients of all correlation values w.r.t. h.
    Returns array of shape (2n-1, n) where row k is d(c_k)/dh."""
    n = len(h)
    m = 2 * n - 1
    grads = np.zeros((m, n))
    # For each lag d = k - (n-1):
    for d in range(-(n-1), n):
        k = d + (n - 1)
        if d >= 0:
            # c_k = sum_{j=0}^{n-1-d} h[j] * (1-h[j+d])
            # d/dh[p]: 
            #   If 0<=p<=n-1-d: (1-h[p+d]) [from h[j] term, j=p]
            #   If d<=p<=n-1: -h[p-d]  [from (1-h[j+d]) term, j=p-d]
            j_max = n - 1 - d
            for p in range(0, j_max + 1):
                grads[k, p] += (1.0 - h[p + d])
            for p in range(d, n):
                grads[k, p] -= h[p - d]
        elif d < 0:

            # c_k = sum_{j=dp}^{n-1} h[j] * (1-h[j-dp])
            # d/dh[p]:
            #   If dp<=p<=n-1: (1-h[p-dp]) [from h[j] term, j=p]
            #   If 0<=p<=n-1-dp: -h[p+dp] [from (1-h[j-dp]) term, j=p+dp]
            for p in range(dp, n):
                grads[k, p] += (1.0 - h[p - dp])
            for p in range(0, n - dp):
                grads[k, p] -= h[p + dp]
        else:

            # d/dh[p] = (1-h[p]) - h[p] = 1 - 2*h[p]
            grads[k, :] = 1.0 - 2.0 * h

def project_feasible(x, target_sum):
    n = len(x)
    x = np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0)
    lo, hi = -2.0, 3.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h = np.clip(x - lam, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        cnt = mask.sum()
        if cnt > 0:
            h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = corr.argmax()
    return corr[idx] * dx, corr, idx


def compute_c5_val(h):
    c, _, _ = compute_c5(h)
    return c


def grad_corr_vec(h):
    n = len(h)
    m = 2 * n - 1
    G = np.zeros((m, n))
    for d in range(-(n - 1), n):
        k = d + n - 1
        if d > 0:
            L = n - d
            G[k, :L] += 1.0 - h[d:]
            G[k, d:] -= h[:L]
        elif d < 0:
            L = n + d
            G[k, -d:] += 1.0 - h[:L]
            G[k, :L] -= h[-d:]
        else:
            G[k, :] = 1.0 - 2.0 * h
    return G


def slsqp_solve(h_init, target_sum, maxiter=300, ftol=1e-12, r_margin=0.01):
    n = len(h_init)
    dx = 2.0 / n
    m = 2 * n - 1
    c_init = compute_c5_val(h_init)
    r0 = c_init + r_margin
    x0 = np.concatenate(([r0], h_init))
    
    def obj(x):
        return x[0]
    
    def obj_jac(x):
        g = np.zeros(n + 1)
        g[0] = 1.0
        return g
    
    def eq_fun(x):
        return x[1:].sum() - target_sum
    
    def eq_jac(x):
        g = np.zeros(n + 1)
        g[1:] = 1.0
        return g
    
    def ineq_fun(x):
        h = x[1:]
        corr = np.correlate(h, 1.0 - h, mode='full')
        return x[0] - dx * corr
    
    def ineq_jac(x):
        h = x[1:]
        G = grad_corr_vec(h)
        J = np.empty((m, n + 1))
        J[:, 0] = 1.0
        J[:, 1:] = -dx * G
        return J
    
    constraints = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    bounds = [(0.0, 0.5)] + [(0.0, 1.0)] * n
    
    res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                   bounds=bounds, constraints=constraints,
                   options={'maxiter': maxiter, 'ftol': ftol})
    
    h_out = project_feasible(res.x[1:], target_sum)
    c_out = compute_c5_val(h_out)
    return h_out, c_out, res


def line_polish(h, target_sum, n_directions=20, tol=1e-10):
    n = len(h)
    best_c5 = compute_c5_val(h)
    best_h = h.copy()
    rng = np.random.RandomState(12345)
    
    for _ in range(n_directions):
        # Random zero-mean direction
        v = rng.randn(n)
        v -= v.mean()
        # Compute feasible interval for alpha
        alphas = []
        for i in range(n):
            if abs(v[i]) > 1e-12:
                a1 = -h[i] / v[i]
                a2 = (1.0 - h[i]) / v[i]
                alphas.append((min(a1, a2), max(a1, a2)))
        if alphas:
            lo = max(a[0] for a in alphas)
            hi = min(a[1] for a in alphas)
            lo = max(lo, -1.0)
            hi = min(hi, 1.0)
            if hi - lo < 1e-8:
                continue
            # Golden section search
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            a, b = lo, hi
            c_ = b - gr * (b - a)
            d_ = a + gr * (b - a)
            fc = compute_c5_val(project_feasible(h + c_ * v, target_sum))
            fd = compute_c5_val(project_feasible(h + d_ * v, target_sum))
            for _it in range(40):
                if fc < fd:
                    b = d_
                    d_ = c_
                    fd = fc

                    c_ = d_
                    fc = fd
                    d_ = a + gr * (b - a)
                    fd = compute_c5_val(project_feasible(h + d_ * v, target_sum))
            alpha_best = (a + b) / 2.0

            c5_new = compute_c5_val(h_new)
            if c5_new < best_c5 - tol:
                best_c5 = c5_new
                best_h = h_new
                h = h_new
    
    return best_h, best_c5


def perturb_h(h, target_sum, sigma, rng):
    h_pert = h + sigma * rng.randn(len(h))
    return project_feasible(h_pert, target_sum)

import numpy as np
from scipy.optimize import minimize
import time


def project_feasible(x, target_sum):
    n = len(x)
    x = np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0)
    lo, hi = -2.0, 3.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h = np.clip(x - lam, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        cnt = mask.sum()
        if cnt > 0:
            h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = int(corr.argmax())
    return corr[idx] * dx, corr, idx


def compute_c5_val(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max()) * dx


def grad_corr_vec(h):
    n = len(h)
    m = 2 * n - 1
    G = np.zeros((m, n))
    for d in range(-(n - 1), n):
        k = d + n - 1
        if d > 0:
            L = n - d
            G[k, :L] += 1.0 - h[d:]
            G[k, d:] -= h[:L]
        elif d < 0:
            L = n + d
            G[k, -d:] += 1.0 - h[:L]
            G[k, :L] -= h[-d:]
        else:
            G[k, :] = 1.0 - 2.0 * h
    return G


def slsqp_full(h_init, target_sum, maxiter=300, ftol=1e-12, r_margin=0.01):
    n = len(h_init)
    dx = 2.0 / n
    m = 2 * n - 1
    c_init = compute_c5_val(h_init)
    r0 = c_init + r_margin
    x0 = np.concatenate(([r0], h_init))

    def obj(x):
        return x[0]

    def obj_jac(x):
        g = np.zeros(n + 1)
        g[0] = 1.0
        return g

    def eq_fun(x):
        return x[1:].sum() - target_sum

    def eq_jac(x):
        g = np.zeros(n + 1)
        g[1:] = 1.0
        return g

    def ineq_fun(x):
        h = x[1:]
        corr = np.correlate(h, 1.0 - h, mode='full')
        return x[0] - dx * corr

    def ineq_jac(x):
        h = x[1:]
        G = grad_corr_vec(h)
        J = np.empty((m, n + 1))
        J[:, 0] = 1.0
        J[:, 1:] = -dx * G
        return J

    constraints = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    bounds = [(0.0, 0.5)] + [(0.0, 1.0)] * n

    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': ftol})
    except Exception:
        return h_init, c_init, None

    h_out = project_feasible(res.x[1:], target_sum)
    c_out = compute_c5_val(h_out)
    return h_out, c_out, res


def slsqp_active(h_init, target_sum, maxiter=300, ftol=1e-12,
                 r_margin=0.01, n_active=40):

    dx = 2.0 / n

    r0 = c_init + r_margin
    x0 = np.concatenate(([r0], h_init))
    active_idx = np.argsort(corr_init)[-n_active:]

    def obj(x):
        return x[0]

    def obj_jac(x):
        g = np.zeros(n + 1)
        g[0] = 1.0
        return g

        return g

    def ineq_fun(x):
        h = x[1:]
        corr = np.correlate(h, 1.0 - h, mode='full')
        return x[0] - dx * corr[active_idx]

    def ineq_jac(x):
        h = x[1:]
        G = grad_corr_vec(h)
        Ga = G[active_idx]
        J = np.empty((len(active_idx), n + 1))
        J[:, 0] = 1.0
        J[:, 1:] = -dx * Ga
        return J

                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': ftol})

        return h_init, c_init, None

    h_out = project_feasible(res.x[1:], target_sum)
    c_out = compute_c5_val(h_out)
    return h_out, c_out, res


def line_polish(h, target_sum, n_dirs=30, polar_seed=42):
    n = len(h)
    best_c5 = compute_c5_val(h)
    best_h = h.copy()
    rng = np.random.RandomState(polar_seed)

    for _ in range(n_dirs):
        v = rng.randn(n)
        v -= v.mean()
        alphas_lo = []
        alphas_hi = []
        for i in range(n):
            if abs(v[i]) > 1e-12:
                a1 = -h[i] / v[i]
                a2 = (1.0 - h[i]) / v[i]
                alphas_lo.append(min(a1, a2))
                alphas_hi.append(max(a1, a2))
        if not alphas_lo:
            continue
        lo = max(max(alphas_lo), -1.0)
        hi = min(min(alphas_hi), 1.0)
        if hi - lo < 1e-9:
            continue
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        a, b = lo, hi
        c_pt = b - gr * (b - a)
        d_pt = a + gr * (b - a)
        fc = compute_c5_val(project_feasible(h + c_pt * v, target_sum))
        fd = compute_c5_val(project_feasible(h + d_pt * v, target_sum))
        for _it in range(45):
            if fc < fd:
                b = d_pt
                d_pt = c_pt
                fd = fc
                c_pt = b - gr * (b - a)
                fc = compute_c5_val(project_feasible(h + c_pt * v, target_sum))
            else:
                a = c_pt
                c_pt = d_pt
                fc = fd
                d_pt = a + gr * (b - a)
                fd = compute_c5_val(project_feasible(h + d_pt * v, target_sum))
        alpha_best = (a + b) / 2.0
        h_new = project_feasible(h + alpha_best * v, target_sum)
        c5_new = compute_c5_val(h_new)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new

    return best_h, best_c5


def perturb_h(h, target_sum, sigma, rng):
    h_pert = h + sigma * rng.randn(len(h))
    return project_feasible(h_pert, target_sum)
