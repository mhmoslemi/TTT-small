**KKT-based active set method:**

At a local minimum h*, for the active set A:
- 0 must be in the convex hull of {G[d,:] projected onto sum=0 : d ∈ A} (when no variables are at bounds)
- More precisely, there exist multipliers λ_d ≥ 0 for d ∈ A such that sum_d λ_d * G[d,:] = 0 (projected onto the feasible directions)

The projected gradient (onto sum=0 subspace) of the active constraints gives us:
- If we can find λ ≥ 0, sum λ = 1 such that G_A^T @ λ (projected) = 0, we're at a stationary point
- Otherwise, the descent direction is -(G_A^T @ λ_opt) where λ_opt minimizes ||G_A^T @ λ||² subject to λ ≥ 0, sum λ = 1

This is a QP! And it gives us both the optimality check AND the descent direction.

The QP is:
minimize ||G_A^T @ λ||² subject to λ ≥ 0, sum(λ) = 1

Where G_A is the matrix of active gradient rows, projected onto the sum-zero subspace.

Let me denote G̃ = G_A - mean(G_A, axis=1, keepdims=True) (each row projected to sum-zero).

Then the QP is: minimize ||G̃^T @ λ||² s.t. λ ≥ 0, sum(λ) = 1

This is equivalent to: minimize λ^T (G̃ G̃^T) λ s.t. λ ≥ 0, sum(λ) = 1

This is a convex QP! We can solve it with scipy.optimize.minimize or even with a simple projected gradient method.

    from scipy.optimize import minimize
    
    # Objective: ||G_tilde^T @ lambda||^2 = lambda^T (G_tilde @ G_tilde.T) lambda
    A_mat = G_tilde @ G_tilde.T
    c = np.zeros(n_active)
    
    from scipy.optimize import linprog  # Actually this is a QP, not LP
    
    # Use scipy minimize
    def obj(lam):
        G_lam = G_tilde.T @ lam
        return np.dot(G_lam, G_lam)
    
    def obj_jac(lam):
        G_lam = G_tilde.T @ lam
        return 2 * G_tilde @ G_lam

import time

def compute_profile(h):

    dx = 2.0 / n

def repair_feasible(h):

            rem = min(remaining, h[idx])

        else:
            idx = int(np.argmax(h))

def build_gradient(h):

    dx = 2.0 / n
    M = 2 * n - 1
    G = np.zeros((M, n))
    center = n - 1
    for d in range(0, n):

        if d == 0:
            G[k, :] = dx * (1.0 - 2.0 * h)
        else:
            G[k, :n-d] += dx * (1.0 - h[d:])
            G[k, d:] += -dx * h[:n-d]
    for d in range(1, n):
        k = center - d
        G[k, d:] += dx * (1.0 - h[:n-d])
        G[k, :n-d] += -dx * h[d:]
    return G

def project_sum_zero(v):
    return v - v.mean()

def kkt_descent_direction(h, active_set):
    """Compute KKT-based descent direction using QP."""
    n = len(h)
    G = build_gradient(h)
    G_A = G[active_set, :]
    # Project to sum-zero
    G_A = G_A - G_A.mean(axis=1, keepdims=True)
    m = len(active_set)
    if m == 1:
        v = project_sum_zero(-G_A[0])
        norm = np.linalg.norm(v)
        if norm < 1e-15:
            return None, 0.0
        return v / norm, norm
    # Solve QP: min ||G_A^T @ lambda||^2 s.t. lambda >= 0, sum(lambda)=1
    # Equivalent: min lambda^T (G_A G_A^T) lambda
    A_mat = G_A @ G_A.T
    def qp_obj(lam):
        return float(lam @ A_mat @ lam)
    def qp_jac(lam):
        return 2.0 * (A_mat @ lam)
    bounds = [(0.0, 1.0)] * m
    cons = [{'type': 'eq', 'fun': lambda lam: lam.sum() - 1.0,
             'jac': lambda lam: np.ones(m)}]
    x0 = np.ones(m) / m
    try:
        res = minimize(qp_obj, x0, jac=qp_jac, bounds=bounds,
                       constraints=cons, method='SLSQP',
                       options={'maxiter': 200, 'ftol': 1e-14})
        if res.success:
            lam = res.x
        else:
            lam = x0
    except Exception:
        lam = x0
    v = -G_A.T @ lam
    v = project_sum_zero(v)
    norm = np.linalg.norm(v)
    if norm < 1e-15:
        return None, 0.0
    return v / norm, norm

def box_line_search(h, direction, c_current, max_iter=50):
    """Line search with box constraints. Returns (h_new, c_new)."""
    n = len(h)
    # Find max step before hitting box
    t_max = 1.0  # default
    neg_mask = direction < -1e-12
    pos_mask = direction > 1e-12
    if np.any(neg_mask):
        t_max = min(t_max, np.min(h[neg_mask] / (-direction[neg_mask])))
    if np.any(pos_mask):
        t_max = min(t_max, np.min((1.0 - h[pos_mask]) / direction[pos_mask]))
    t_max = max(t_max, 1e-8)
    
    # Golden section search on [0, t_max]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = 0.0, t_max
    c_pt = b - gr * (b - a)
    d_pt = a + gr * (b - a)
    f_c = exact_score(repair_feasible(h + c_pt * direction))
    f_d = exact_score(repair_feasible(h + d_pt * direction))
    
    best_t = 0.0
    best_f = c_current
    h_best = h.copy()
    if f_c < best_f:
        best_f, best_t, h_best = f_c, c_pt, repair_feasible(h + c_pt * direction)
    if f_d < best_f:
        best_f, best_t, h_best = f_d, d_pt, repair_feasible(h + d_pt * direction)
    
    for _ in range(max_iter):
        if b - a < 1e-10 * max(1.0, t_max):
            break
        if f_c < f_d:
            b = d_pt
            d_pt = c_pt
            f_d = f_c
            c_pt = b - gr * (b - a)
            f_c = exact_score(repair_feasible(h + c_pt * direction))
        else:
            a = c_pt
            c_pt = d_pt
            f_c = f_d
            d_pt = a + gr * (b - a)
            f_d = exact_score(repair_feasible(h + d_pt * direction))
        if f_c < best_f:
            best_f, best_t, h_best = f_c, c_pt, repair_feasible(h + c_pt * direction)
        if f_d < best_f:
            best_f, best_t, h_best = f_d, d_pt, repair_feasible(h + d_pt * direction)
    
    return h_best, best_f

def pair_transfer_search(h, c_current, n_pairs=50, rng=None):
    """Try random pair transfers."""
    n = len(h)
    profile = compute_profile(h)
    # Find active lags
    T = np.max(profile)
    active = np.where(profile >= T - 0.01 * T)[0]
    
    best_h = h.copy()
    best_c = c_current
    
    if rng is None:
        rng = np.random.RandomState(0)
    
    # Build gradient for active set
    G = build_gradient(h)
    G_A = G[active, :]
    
    # For each pair (p0, p1), compute the direction e[p1] - e[p0]
    # and check if it helps
    candidates = []
    for _ in range(n_pairs * 3):
        p0 = rng.randint(n)
        p1 = rng.randint(n)
        if p0 == p1:
            continue
        # Direction: decrease h[p0], increase h[p1]
        # Effect on profile: G_A[:, p1] - G_A[:, p0]
        effect = G_A[:, p1] - G_A[:, p0]
        # We want to decrease the max, so we want effect to be negative on active lags
        score = -np.max(effect)  # Higher is better (more negative effect on max active)
        # Also check capacity
        if h[p0] < 1e-6 or h[p1] > 1 - 1e-6:
            continue
        candidates.append((score, p0, p1))
    
    candidates.sort(key=lambda x: -x[0])
    
    for score, p0, p1 in candidates[:n_pairs]:
        if score <= 0:
            break
        # Try transferring mass from p0 to p1
        mu = min(h[p0], 1.0 - h[p1])

            h_try[p0] -= t
            h_try[p1] += t
            h_try = repair_feasible(h_try)
            c_try = exact_score(h_try)
            if c_try < best_c - 1e-14:
                best_c = c_try
                best_h = h_try
        # After updating, check again
        if best_c < c_current - 0.001:
            # Renew candidates might be too expensive; just continue
            pass
    
    return best_h, best_c

def get_active_set(profile, tol=0.0, top_k=20):
    T = np.max(profile)
    thresh = T - max(tol, 1e-6)
    active = np.where(profile >= thresh)[0]
    if len(active) == 0:
        active = np.array([np.argmax(profile)])
    if len(active) > top_k:
        # Keep top_k by value
        order = np.argsort(profile[active])[::-1]
        active = active[order[:top_k]]
    return active

        if budget_s - elapsed < 5:
            break

            best_c = c
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
