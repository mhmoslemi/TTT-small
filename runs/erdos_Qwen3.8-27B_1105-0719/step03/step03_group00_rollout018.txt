```python
def optimal_transfer(h, i, j, dx):
    n = len(h)
    # Find optimal a in [0, min(h[i], 1-h[j])] such that
    # max_k (corr_k + a * delta_k) is minimized
    # where delta_k = (h[i-off_k] - h[j-off_k]) - (h[i+off_k] - h[j+off_k])
    
    # First compute all correlations and all delta_k
    corrs = np.correlate(h, 1.0 - h, mode='full')
    n_corr = len(corrs)
    
    deltas = np.zeros(n_corr)
    for k in range(n_corr):
        off = k - (n-1)
        d = 0.0
        # h[i-off] if valid
        if 0 <= i-off < n:
            d += h[i-off]
        # -h[j-off] if valid
        if 0 <= j-off < n:
            d -= h[j-off]
        # -h[i+off] if valid
        if 0 <= i+off < n:
            d -= h[i+off]
        # +h[j+off] if valid
        if 0 <= j+off < n:
            d += h[j+off]
        deltas[k] = d
    
    a_lo = 0.0
    a_hi = min(h[i], 1.0 - h[j])
    if a_hi <= 0:
        return 0.0
    
    # We want to minimize max_k (corrs[k] + a * deltas[k]) over a in [a_lo, a_hi]
    # This is a 1D convex piecewise-linear minimization
    
    # The minimum of max of linear functions occurs where two (or more) functions are equal
    # and it's a local minimum of the max function
    
    # Simple approach: ternary search or golden section
    # Or: the minimum is at a point where for some pair (p,q):
    # corrs[p] + a*delta_p = corrs[q] + a*delta_q
    # and this is a local minimum
    
    # Even simpler: since we're in 1D, use golden section search
    # Or just try a grid + refinement
    
    # Actually for piecewise linear, the minimum is at a "kink" point
    # Kink points are where corrs[p] + a*delta_p = corrs[q] + a*delta_q
    # i.e., a = (corrs[q] - corrs[p]) / (delta_p - delta_q)
    
    # But there are O(n^2) such points, which is too many.
    # Instead, use bisection on the subgradient.
    
    # The subgradient of f(a) = max_k (c_k + a*d_k) is the convex hull of {d_k : k in argmax}
    # f is decreasing where all active d_k < 0, increasing where all active d_k > 0
    # The minimum is where 0 is in the convex hull of active d_k's
    
    # Simple approach: binary search for where the sign of the "slope" changes
    
    lo, hi = a_lo, a_hi
    for _ in range(50):
        mid = (lo + hi) / 2
        vals = corrs + mid * deltas
        max_val = vals.max()
        active = np.where(vals >= max_val - 1e-12)[0]
        # If all active deltas are negative, we can go higher
        # If all active deltas are positive, we can go lower
        # If mixed, we're at the minimum
        if deltas[active].max() < 0:
            lo = mid
        elif deltas[active].min() > 0:
            hi = mid
        else:
            lo = hi = mid
            break
    
    a_opt = (lo + hi) / 2
    return a_opt
```

```python
    corrs = compute_corrs(h)
    max_c = corrs.max()
    exps = np.exp((corrs - max_c) / tau)

    w_threshold = 1e-8
    active = w > w_threshold
```

    best_h = cur.copy()
    
    corrs = compute_corrs(cur)
    # Try all (i,j) pairs where h[i] > 0 and h[j] < 1
    active_i = np.where(cur > 1e-8)[0]
    active_j = np.where(cur < 1.0 - 1e-8)[0]
    
    improved = False
    for i in active_i:
        if budget_s - (time.time() - t0) < 5.0:
            break
        for j in active_j:
            if j == i:
                continue
            a = pair_optimal_a(cur, i, j)
            if a > 1e-10:
                h_new = apply_transfer(cur, i, j, a)
                c5_new = c5_of_h(h_new, dx)
                if c5_new < cur_c5:
                    cur = h_new
                    cur = project_feasible(cur, target_sum)
                    cur_c5 = c5_new
                    improved = True
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = cur.copy()
                    break
        if improved:
            improved = False
            # Restart the scan with updated corrs
            continue
        # If no improvement found for this i, continue
    return best_h, best_c5

def softmax_grad(h, dx, tau):
    n = len(h)
    corrs = compute_corrs(h)
    max_c = corrs.max()
    exps = np.exp((corrs - max_c) / tau)
    w = exps / exps.sum()
    
    grad = np.zeros(n)
    for k in range(len(corrs)):
        wi = w[k]
        if wi < 1e-8:
            continue
        offset = k - (n - 1)
        # First term: (1-h[i+offset]) for valid i
        i_start = max(0, -offset)
        i_end = min(n - 1, n - 1 - offset)
        if i_start <= i_end:
            idx = np.arange(i_start, i_end + 1)
            grad[idx] += wi * (1.0 - h[idx + offset])
        # Second term: -h[i-offset] for valid i
        i_start2 = max(0, offset)
        i_end2 = min(n - 1, offset + n - 1)
        if i_start2 <= i_end2:
            idx2 = np.arange(i_start2, i_end2 + 1)
            grad[idx2] += wi * (-h[idx2 - offset])
    
    return dx * grad

def softmax_descent(h0, dx, target_sum, budget_s, t0):
    n = len(h0)
    cur = h0.copy()
    cur_c5 = c5_of_h(cur, dx)
    best_h = cur.copy()
    best_c5 = cur_c5
    
    # Continuation: tau from large to small
    tau_values = [2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02]
    
    for tau in tau_values:
        if budget_s - (time.time() - t0) < 5.0:
            break
        n_iter = 50
        for it in range(n_iter):
            if budget_s - (time.time() - t0) < 3.0:
                break
            grad = softmax_grad(cur, dx, tau)
            # Gradient step
            lr = 0.002
            h_new = project_feasible(cur - lr * grad, target_sum)
            # Line search
            improved = False
            for t in [1.0, 0.5, 0.25, 0.1]:
                h_try = project_feasible(cur + t * (h_new - cur), target_sum)
                c5_try = c5_of_h(h_try, dx)
                if c5_try < cur_c5 - 1e-15:
                    cur = h_try
                    cur_c5 = c5_try
                    improved = True
                    if c5_try < best_c5:
                        best_c5 = c5_try
                        best_h = cur.copy()
                    break
            if not improved:
                break
    
    return best_h, best_c5

def binary_round(h, n_target):
    """Round h to binary with exactly n_target/2 ones."""
    n = len(h)
    # Sort by value, take top n/2 as 1
    order = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[order[:n//2]] = 1.0
    return h_bin

def binary_local_search(h_bin, dx, target_sum, budget_s, t0, max_passes=5):
    n = len(h_bin)
    cur = h_bin.copy()
    cur_c5 = c5_of_h(cur, dx)
    best_h = cur.copy()
    best_c5 = cur_c5
    
    for pose in range(max_passes):
        if budget_s - (time.time() - t0) < 5.0:
            break
        
        ones = np.where(cur == 1.0)[0]
        zeros = np.where(cur == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        improved = False
        # Try all swaps: move 1 from i to j
        best_delta = 0
        best_swap = None
        corrs = compute_corrs(cur)
        max_val = corrs.max()
        
        for i in ones:
            if budget_s - (time.time() - t0) < 3.0:
                break
            for j in zeros:
                # Compute change in max correlation
                # After swap: cur[i]=0, cur[j]=1
                # Delta for corr_k: 
                #   -(1-cur[i+off]) + cur[i-off] + (1-cur[j+off]) - cur[j-off]
                # For binary: these are just 0 or 1 values
                # Quick check: does this swap help the max?
                deltas = pair_delta(cur, i, j)
                new_max = (corrs + deltas).max()
                if new_max < max_val:
                    if max_val - new_max > best_delta:
                        best_delta = max_val - new_max
                        best_swap = (i, j)
        
        if best_swap is not None:
            i, j = best_swap
            cur[i] = 0.0
            cur[j] = 1.0
            cur_c5 = c5_of_h(cur, dx)
            improved = True
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = cur.copy()
        
        if not improved:
            break
    
    return best_h, best_c5

def transfer_grid(h_src, n_target):
    n_src = len(h_src)
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_tgt = (np.arange(n_target) + 0.5) * (2.0 / n_target)
    return np.interp(x_tgt, x_src, h_src)
