def slp_step(h, R, method='linprog'):
    """Trust-region SLP step using linear program."""
    n = len(h)
    M = 2 * n - 1
    p = compute_profile(h)
    dx = 2.0 / n
    
    # Build gradient matrix
    G = np.zeros((M, n))
    half = n - 1

        if d == 0:

        elif d > 0:
            G[m, d:] += dx * (1.0 - h[:n-d])
            G[m, :n-d] -= dx * h[d:]

            G[m, pi:] -= dx * h[:n-pi]

    #      sum(d) = 0     (1 equality)

    #      t >= p.min() - R (loose lower bound for t)

    bounds.append((tlo, None))

            t = res.x[-1]

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(-(1.0 - h))
        remaining = diff
        for i in range(n):
            idx = order[i]
            cap = 1.0 - h[idx]
            if cap <= 1e-15:
                continue
            add = min(remaining, cap)
            h[idx] += add
            remaining -= add
            if remaining <= 1e-14:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            if h[idx] <= 1e-15:
                continue
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-14:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] = float(np.clip(h[idx] + diff2, 0.0, 1.0))
    return h


def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))


def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx


def grad_row(h, m, n):
    dx = 2.0 / n
    d = m - (n - 1)
    if d == 0:
        return dx * (1.0 - 2.0 * h)
    elif d > 0:
        g = np.zeros(n)
        g[d:] += dx * (1.0 - h[:n - d])
        g[:n - d] -= dx * h[d:]
        return g
    else:
        p = -d
        g = np.zeros(n)
        g[:n - p] += dx * (1.0 - h[p:])
        g[p:] -= dx * h[:n - p]
        return g


def lse_grad(h, beta):
    n = len(h)
    p = compute_profile(h)
    M = len(p)
    c = p.max()
    z = beta * (p - c)
    z = np.clip(z, -50.0, 0.0)
    q = np.exp(z)
    qsum = q.sum()
    w = q / qsum
    dx = 2.0 / n
    half = n - 1
    grad = np.zeros(n)
    for m in range(M):
        wm = w[m]
        if wm < 1e-14:
            continue
        d = m - half
        if d == 0:
            grad += wm * dx * (1.0 - 2.0 * h)
        elif d > 0:
            grad[d:] += wm * dx * (1.0 - h[:n - d])
            grad[:n - d] -= wm * dx * h[d:]
        else:
            pi = -d
            grad[:n - pi] += wm * dx * (1.0 - h[pi:])
            grad[pi:] -= wm * dx * h[:n - pi]
    lse_val = (1.0 / beta) * (c + np.log(qsum))
    return grad, lse_val


def build_grad_matrix(h):
    n = len(h)
    M = 2 * n - 1
    G = np.zeros((M, n))
    dx = 2.0 / n
    half = n - 1
    for m in range(M):
        d = m - half
        if d == 0:
            G[m] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            G[m, d:] += dx * (1.0 - h[:n - d])
            G[m, :n - d] -= dx * h[d:]
        else:
            pi = -d
            G[m, :n - pi] += dx * (1.0 - h[pi:])
            G[m, pi:] -= dx * h[:n - pi]
    return G


def slp_step(h, R):
    n = len(h)
    M = 2 * n - 1
    p = compute_profile(h)
    G = build_grad_matrix(h)
    x_len = n + 1
    c_obj = np.zeros(x_len)
    c_obj[-1] = 1.0
    A_ub = np.zeros((M, x_len))
    A_ub[:, :n] = G
    A_ub[:, -1] = -1.0
    b_ub = -p.copy()
    A_eq = np.zeros((1, x_len))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    lo = np.maximum(-R, -h)
    hi = np.minimum(R, 1.0 - h)
    bounds = list(zip(lo, hi))
    bounds.append((0.0, None))
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0 and res.x is not None:
            return res.x[:n]
    except Exception:
        pass
    return None


def slp_descent(h, max_iters=80, R_init=0.08):
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    R = R_init
    stall = 0
    for it in range(max_iters):
        d = slp_step(h, R)
        if d is None:
            R *= 0.5
            stall += 1
            if stall > 6:
                break
            continue
        alphas = [1.0, 0.75, 0.5, 0.35, 0.25, 0.15, 0.1, 0.05]
        improved = False
        for alpha in alphas:
            h_try = repair_feasible(h + alpha * d)
            c_try = exact_score(h_try)
            if c_try < best_c - 1e-13:
                h = h_try
                best_c = c_try
                improved = True
                stall = 0
                R = min(R * 1.15, 0.25)
                break
        if not improved:
            R *= 0.55
            stall += 1
            if stall > 10 or R < 0.002:
                break
    return h, best_c


def lse_line_search(h, d, beta, alpha_max):
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    _, lse0 = lse_grad(h, beta)
    best_alpha = 0.0
    best_lse = lse0
    best_h_local = h.copy()
    a_ls, b_ls = 0.0, alpha_max
    for _ in range(18):
        c1 = b_ls - gr * (b_ls - a_ls)
        c2 = a_ls + gr * (b_ls - a_ls)
        h1 = repair_feasible(h + c1 * d)
        h2 = repair_feasible(h + c2 * d)
        l1, _ = lse_grad(h1, beta)
        l2, _ = lse_grad(h2, beta)
        if l1 < best_lse:
            best_lse = l1
            best_alpha = c1
            best_h_local = h1.copy()
        if l2 < best_lse:
            best_lse = l2
            best_alpha = c2
            best_h_local = h2.copy()
        if l1 < l2:
            b_ls = c2
        else:
            a_ls = c1
        if b_ls - a_ls < 1e-14:
            break
    return best_h_local, best_alpha


def lse_descent_schedule(h, betas, steps_per_beta=8):
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    for beta in betas:
        for _ in range(steps_per_beta):
            grad, lse_val = lse_grad(h, beta)
            grad -= grad.mean()
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break
            d = -grad / gnorm
            pos_m = d > 1e-12
            neg_m = d < -1e-12
            alpha_max = 0.5
            if pos_m.any():
                a1 = ((1.0 - h[pos_m]) / d[pos_m]).min() * 0.99
                alpha_max = min(alpha_max, float(a1))
            if neg_m.any():
                a2 = (h[neg_m] / (-d[neg_m])).min() * 0.99
                alpha_max = min(alpha_max, float(a2))
            alpha_max = max(alpha_max, 1e-10)
            h_new, alpha_used = lse_line_search(h, d, beta, alpha_max)
            if alpha_used > 1e-12:
                h = h_new
                c = exact_score(h)
                if c < best_c:
                    best_c = c
    return h, best_c


def interp_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    edges_old = np.linspace(0.0, 2.0, n + 1)
    edges_new = np.linspace(0.0, 2.0, new_n + 1)
    h_new = np.zeros(new_n)
    for k in range(new_n):
        a, b = edges_new[k], edges_new[k + 1]
        total = 0.0
        for j in range(n):
            lo = max(a, edges_old[j])
            hi = min(b, edges_old[j + 1])
            if hi > lo:
                total += h[j] * (hi - lo)
        h_new[k] = total / (b - a)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)


def perturb_h(h, sigma, rng):
    n = len(h)
    noise = rng.normal(0.0, sigma, n)
    noise -= noise.mean()
    h_new = repair_feasible(h + noise)
    return h_new


def active_descent_h(h, max_iters=200, active_tol=0.003, max_active=15):
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for it in range(max_iters):
        profile = compute_profile(h)
        c = float(profile.max())
        thresh = c - active_tol
        active = np.where(profile >= thresh)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(profile))])
        if len(active) > max_active:
            top = np.argsort(profile[active])[-max_active:]
            active = active[top]
        g = np.zeros(n)
        for m in active:
            g += grad_row(h, m, n)
        g -= g.mean()
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-14:
            break
        d = -g / gnorm
        pos_m = d > 1e-12
        neg_m = d < -1e-12
        alpha_max = 0.5
        if pos_m.any():
            alpha_max = min(alpha_max, float(((1.0 - h[pos_m]) / d[pos_m]).min()) * 0.99)
        if neg_m.any():
            alpha_max = min(alpha_max, float((h[neg_m] / (-d[neg_m])).min()) * 0.99)
        alpha_max = max(alpha_max, 1e-10)
        best_alpha = 0.0
        best_val = c
        best_h_local = h.copy()
        a_ls, b_ls = 0.0, alpha_max
        for _ in range(18):
            c1 = b_ls - gr * (b_ls - a_ls)
            c2 = a_ls + gr * (b_ls - a_ls)
            h1 = repair_feasible(h + c1 * d)
            h2 = repair_feasible(h + c2 * d)
            f1 = exact_score(h1)
            f2 = exact_score(h2)
            if f1 < best_val:
                best_val = f1
                best_alpha = c1
                best_h_local = h1.copy()
            if f2 < best_val:
                best_val = f2
                best_alpha = c2
                best_h_local = h2.copy()
            if f1 < f2:
                b_ls = c2
            else:
                a_ls = c1
            if b_ls - a_ls < 1e-14:
                break
        if best_val >= c - 1e-14:
            break
        h = best_h_local
        if best_val < best_c:
            best_c = best_val
    return h, best_c


def slsqp_epigraph(h_start, maxiter=150):
    n = len(h_start)
    dx = 2.0 / n
    M = 2 * n - 1
    c0 = exact_score(h_start)
    t0 = c0 + 1e-4

    def obj_func(x):
        return x[-1]

    def obj_grad(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g

    def eq_func(x):
        return x[:-1].sum() - n / 2.0

    def eq_grad(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g

    def ineq_func(x):
        h = x[:-1]
        t = x[-1]
        profile = np.correlate(h, 1.0 - h, mode="full") * dx
        return t - profile

    def ineq_grad(x):
        h = x[:-1]
        G = build_grad_matrix(h)
        J = np.zeros((M, n + 1))
        J[:, :n] = -G
        J[:, -1] = 1.0
        return J

    x0 = np.concatenate([h_start, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 2.0)]
    cons = [
        {'type': 'eq', 'fun': eq_func, 'jac': eq_grad},
        {'type': 'ineq', 'fun': ineq_func, 'jac': ineq_grad}
    ]
    try:
        res = minimize(obj_func, x0, method='SLSQP', jac=obj_grad,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None and np.isfinite(res.x).all():
            h_new = repair_feasible(res.x[:-1])
            c_new = exact_score(h_new)
            if c_new <= c0 + 1e-8:
                return h_new
    except Exception:
        pass
    return None


def binary_seed(n, pattern_type, rng):
    half = n // 2
    h = np.zeros(n)
    if pattern_type == 0:
        vals = (np.arange(n) * 0.618033988749895) % 1.0
        top = np.argsort(vals)[-half:]
        h[top] = 1.0
    elif pattern_type == 1:
        vals = (np.arange(n) * 0.7548776662466928) % 1.0
        top = np.argsort(vals)[-half:]
        h[top] = 1.0
    elif pattern_type == 2:
        for i in range(n):
            if (i * 3 + 1) % 4 < 2:
                h[i] = 1.0
        ones = int(h.sum())
        if ones > half:
            one_idx = np.where(h == 1.0)[0]
            to_flip = rng.choice(one_idx, ones - half, replace=False)
            h[to_flip] = 0.0
        elif ones < half:
            zero_idx = np.where(h == 0.0)[0]
            to_flip = rng.choice(zero_idx, half - ones, replace=False)
            h[to_flip] = 1.0
    else:
        perm = rng.permutation(n)
        h[perm[:half]] = 1.0
    return repair_feasible(h)


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s - (time.perf_counter() - t_start)

    try:
        parent = np.array(initial_h_values, dtype=np.float64)
    except Exception:
        parent = np.full(70, 0.5)
    if parent is None or len(parent) < 10:
        parent = np.full(70, 0.5)
    parent = np.clip(parent, 0.0, 1.0)
    parent = repair_feasible(parent)

    best_h = parent.copy()
    best_c = exact_score(parent)

    # Phase 1: LSE descent with increasing beta
    if time_left() > 250:
        betas1 = [3.0, 5.0, 8.0, 12.0, 20.0, 35.0, 60.0, 100.0]
        h_lse, c_lse = lse_descent_schedule(best_h, betas1, steps_per_beta=8)
        if c_lse < best_c:
            best_h, best_c = h_lse, c_lse

    # Phase 2: SLP trust region
    if time_left() > 200:
        h_slp, c_slp = slp_descent(best_h, max_iters=60, R_init=0.1)
        if c_slp < best_c:
            best_h, best_c = h_slp, c_slp

    # Phase 3: Higher beta LSE
    if time_left() > 150:
        betas2 = [150.0, 250.0, 400.0, 600.0, 1000.0, 1500.0]
        h_lse, c_lse = lse_descent_schedule(best_h, betas2, steps_per_beta=6)
        if c_lse < best_c:
            best_h, best_c = h_lse, c_lse

    # Phase 4: SLP refinement with smaller trust
    if time_left() > 100:
        h_slp, c_slp = slp_descent(best_h, max_iters=50, R_init=0.04)
        if c_slp < best_c:
            best_h, best_c = h_slp, c_slp

    # Phase 5: Active descent with tight tolerance
    if time_left() > 80:
        h_ad, c_ad = active_descent_h(best_h, max_iters=200, active_tol=0.001, max_active=12)
        if c_ad < best_c:
            best_h, best_c = h_ad, c_ad

    # Phase 6: Perturb and recover
    for trial in range(5):
        if time_left() < 60:
            break
        sigma = 0.025 + 0.015 * trial
        h_pert = perturb_h(best_h, sigma, rng)
        betas_p = [20.0, 50.0, 100.0, 200.0]
        h_lse, c_lse = lse_descent_schedule(h_pert, betas_p, steps_per_beta=5)
        h_slp, c_slp = slp_descent(h_lse, max_iters=30, R_init=0.06)
        if c_slp < best_c:
            best_h, best_c = h_slp, c_slp

    # Phase 7: Dimension exploration
    n_b = len(best_h)
    dim_list = []
    for dn in [-4, 2, 4, -2, 6, 8, -6, 10, -8, 12]:
        d_new = n_b + dn
        if 40 <= d_new <= 150 and d_new % 2 == 0:
            dim_list.append(d_new)

    for new_n in dim_list[:8]:
        if time_left() < 40:
            break
        h_interp = interp_to_n(best_h, new_n)
        betas_d = [10.0, 30.0, 80.0, 200.0]
        h_lse, c_lse = lse_descent_schedule(h_interp, betas_d, steps_per_beta=6)
        h_slp, c_slp = slp_descent(h_lse, max_iters=40, R_init=0.07)
        if c_slp < best_c:
            best_h, best_c = h_slp, c_slp

    # Phase 8: SLSQP on best
    if time_left() > 50:
        h_sol = slsqp_epigraph(best_h, maxiter=120)
        if h_sol is not None:
            c_sol = exact_score(h_sol)
            if c_sol < best_c:
                best_h, best_c = h_sol, c_sol
                if time_left() > 30:
                    h_slp, c_slp = slp_descent(best_h, max_iters=20, R_init=0.03)
                    if c_slp < best_c:
                        best_h, best_c = h_slp, c_slp

    # Phase 9: Ultra-high beta LSE (near-hard-max refinement)
    if time_left() > 40:
        betas9 = [3000.0, 6000.0, 12000.0]
        h_lse, c_lse = lse_descent_schedule(best_h, betas9, steps_per_beta=4)
        if c_lse < best_c:
            best_h, best_c = h_lse, c_lse

    # Phase 10: Final SLP polish
    if time_left() > 25:
        h_slp, c_slp = slp_descent(best_h, max_iters=30, R_init=0.015)
        if c_slp < best_c:
            best_h, best_c = h_slp, c_slp

    # Phase 11: Final active descent
    if time_left() > 15:
        h_ad, c_ad = active_descent_h(best_h, max_iters=100, active_tol=0.0005, max_active=8)
        if c_ad < best_c:
            best_h, best_c = h_ad, c_ad

    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)

    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            final_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 2:
            final_c = float(eval_result[0])
    except Exception:
        pass

    return best_h, final_c, len(best_h)
