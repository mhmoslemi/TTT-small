Now let me think about the subgradient optimization more carefully. The key is to find a good descent direction.

At a point h, the active set is A = {m : P_m(h) ≈ max}. The subgradient is any convex combination of {∇P_m : m ∈ A}. The steepest descent direction in the subdifferential is the one with minimum norm:
min ||sum λ_m ∇P_m|| subject to sum λ_m = 1, λ_m ≥ 0.

This is a QP with |A| variables. For |A| ≤ 20, it's fast.

But a simpler approach: use the average gradient of the active set. This is a valid subgradient and often works well.

Let me implement a more careful subgradient method:

```python
def subgradient_optimize(h, max_iters=300):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c = exact_score(h)
    
    # Adaptive parameters
    step = 0.02
    active_width = 0.004
    top_k = 15
    momentum = 0.8
    vel = np.zeros(n)
    
    for it in range(max_iters):
        profile = compute_profile(h)
        c = float(profile.max())
        
        if c < best_c:
            best_c = c
            best_h = h.copy()
        
        # Active set
        thresh = c - active_width
        active = np.where(profile >= thresh)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(profile))])
        if len(active) > top_k:
            # Keep the most active
            idx = np.argsort(profile[active])[-top_k:]
            active = active[idx]
        
        # Weighted gradient (weight by activity)
        weights = np.exp((profile[active] - c) / (active_width * 0.5))
        weights = weights / weights.sum()
        
        g = np.zeros(n)
        for i, m in enumerate(active):
            g += weights[i] * grad_row(h, m, n)
        
        # Project to sum=0
        g = g - g.mean()
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-14:
            break
        g = g / gnorm
        
        # Momentum
        vel = momentum * vel + g
        vnorm = np.linalg.norm(vel)
        if vnorm < 1e-14:
            break
        d = -vel / vnorm
        
        # Line search
        best_alpha = 0.0
        best_val = c
        best_h_new = h.copy()
        
        # Backtracking
        alpha = step
        for _ in range(15):
            h_try = repair_feasible(h + alpha * d)
            val = exact_score(h_try)
            if val < best_val:
                best_val = val
                best_alpha = alpha
                best_h_new = h_try
            if val < c - 0.01 * alpha * gnorm:  # Armijo-like condition
                break
            alpha *= 0.5
        
        if best_alpha == 0.0:
            # Stuck, reduce step
            step *= 0.6
            if step < 1e-7:
                break
            continue
        
        h = best_h_new
        # Adapt step
        if best_val < c - 1e-12:
            step = min(step * 1.1, 0.1)
        else:
            step *= 0.7
    
    return best_h, best_c
```

Actually, let me think about this differently. The constraint is that exactly n/2 of the n values are "high" (close to 1) and n/2 are "low" (close to 0), in the binary limit. The arrangement that minimizes the maximum overlap is related to the "Golomb ruler" or "B-sequence" concept.

```python
def simulated_annealing(h, max_iters=2000, T_start=0.05, T_end=0.001):

    best_h = h.copy()
    best_c = exact_score(h)
    T = T_start
    
    for it in range(max_iters):
        T = T_start * (T_end / T_start) ** (it / max_iters)
        
        # Random perturbation
        h_new = h.copy()
        # Perturb 2-5 random coordinates
        n_perturb = rng.randint(2, 6)
        idx = rng.choice(n, n_perturb, replace=False)
        delta = rng.randn(n_perturb) * 0.1
        h_new[idx] += delta
        h_new = repair_feasible(h_new)
        
        c_new = exact_score(h_new)
        delta_c = c_new - exact_score(h)
        
        if delta_c < 0 or rng.random() < np.exp(-delta_c / max(T, 1e-10)):
            h = h_new
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
    
    return best_h, best_c
```

```python
def coord_descent(h, max_sweeps=30, tol=1e-10, max_time_s=None, t_start=None):
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    gr = (np.sqrt(5.0) - 1.0) / 2.0

        if max_time_s is not None and t_start is not None:
            if time.perf_counter() - t_start > max_time_s:
                break

        for j in range(n):
            if h[j] > 0.5:
                k = int(np.argmin(h))
            else:
                k = int(np.argmax(h))
            if k == j:
                k = (j + n // 2) % n
            
            c = h[j] + h[k]
            t_lo = max(0.0, c - 1.0)
            t_hi = min(1.0, c)
            if t_hi - t_lo < 1e-12:
                continue
            
            # Golden section
            a, b = t_lo, t_hi
            c1 = b - gr * (b - a)
            c2 = a + gr * (b - a)
            h[j], h[k] = c1, c - c1
            f1 = exact_score(h)
            h[j], h[k] = c2, c - c2
            f2 = exact_score(h)
            
            for _ in range(20):
                if f1 < f2:
                    b = c2; c2 = c1; f2 = f1
                    c1 = b - gr * (b - a)
                    h[j], h[k] = c1, c - c1
                    f1 = exact_score(h)
                else:
                    a = c1; c1 = c2; f1 = f2
                    c2 = a + gr * (b - a)
                    h[j], h[k] = c2, c - c2
                    f2 = exact_score(h)
                if b - a < tol:
                    break

                h = repair_feasible(h)

                total_improvement += best_c - c_new
                best_c = c_new
```
