```python
def smooth_max_and_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    center = n - 1
    profile = np.correlate(h, 1 - h, mode="full") * dx
    
    # Softmax weights
    shifted = (profile - profile.max()) / tau
    weights = np.exp(shifted)
    weights /= weights.sum()
    
    smax = tau * np.log((weights.sum()) ) + profile.max()  
    # Actually: tau * log(sum(exp(profile/tau)))
    # = tau * log(sum(exp((profile - profile.max())/tau) * exp(profile.max()/tau)))
    # = tau * (log(sum(exp((profile - profile.max())/tau))) + profile.max()/tau)
    # = tau * log(sum(exp(...))) + profile.max()
    # But weights = exp(shifted) / sum(exp(shifted)), so sum(exp(shifted)) = 1/min_weight_ratio
    # Actually, let me just compute it directly:
    smax = tau * (np.log(np.sum(np.exp(shifted))) + (profile.max()) / tau)
    
    # Or more simply:
    # smax = tau * log(sum(exp(profile/tau)))
    # For numerical stability:
    m = profile.max()
    smax = m + tau * np.log(np.sum(np.exp((profile - m) / tau)))
    
    # Gradient
    grad = np.zeros(n)
    for idx in range(M):
        w = weights[idx]
        if w < 1e-12:
            continue
        k = idx - center
        if k >= 0:
            # result = sum_{i=0}^{n-1-k} h[i+k]*(1-h[i])
            # grad[i+k] += w*dx*(1-h[i])  for i in 0..n-1-k
            # grad[i] -= w*dx*h[i+k]      for i in 0..n-1-k
            i_range = np.arange(n - k)
            grad[i_range + k] += w * dx * (1 - h[i_range])
            grad[i_range] -= w * dx * h[i_range + k]
        else:
            k2 = -k
            # result = sum_{i=k2}^{n-1} h[i]*(1-h[i-k2])
            # grad[i] += w*dx*(1-h[i-k2])   for i in k2..n-1
            # grad[i-k2] -= w*dx*h[i]       for i in k2..n-1
            i_range = np.arange(k2, n)
            grad[i_range] += w * dx * (1 - h[i_range - k2])
            grad[i_range - k2] -= w * dx * h[i_range]
    
    return smax, grad
```

This is O(M * n) in the worst case (when all weights are significant), but in practice with small tau, only a few weights are significant.

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:

        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx] + 1e-15)
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:

    else:

        remaining = -diff
