        c_cur = np.max(profile)
        active = get_active(profile, top_k=8, beam=0.002)
        G = build_gradient(best_h, n_f, 2.0 / n_f)
        G_a = G[active, :]
        # Try small perturbations in the null space direction
        for trial in range(20):
            if time_left() < 5:
                break
            # Random direction orthogonal to sum constraint
            d = rng.randn(n_f)
            d -= d.mean()
            d /= np.linalg.norm(d)
            for eps in [0.001, 0.0005, 0.0002, 0.0001]:
                h_try = project_feasible(best_h + eps * d)
                c_try = exact_score(h_try)
                if c_try < best_c - 1e-14:
                    best_h = h_try
                    best_c = c_try
                    break

    # Final validation
    best_h = project_feasible(best_h)
    final_c = exact_score(best_h)

    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            final_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 2:
            final_c = float(eval_result[0])
    except Exception:
        pass

    return best_h, final_c, len(best_h)
