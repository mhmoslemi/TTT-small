def grad_corr_k(h, k, n):
    """Gradient of corr[k] with respect to h."""
    offset = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -offset)
    i_hi = min(n - 1, n - 1 - offset)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    # h[idx] * (1-h[idx+offset]): derivative wrt h is (1-h[idx+offset]) for h[idx]
    grad[idx] += (1.0 - h[idx + offset])
    # For the (1-h) factor: -(h[idx-offset]) where the (1-h) term has index idx
    # i.e., (1-h[j]) where j = i+offset, so i = j-offset
    # derivative of h[i]*(1-h[i+off]) wrt h[i+off] = -h[i]
    grad[idx + offset] += (-h[idx])
    return grad

    return (logsumexp(beta * corrs) / beta) * dx

def smooth_max_grad(h, dx, beta):

    h_new[j] = np.clip(h[j] + eps_opt, 0.0, 1.0)
    h_new = project_feasible(h_new, target_sum)
    return h_new, c5_of_h(h_new, dx)

def pair_descent(h0, dx, target_sum, n_iter=200, n_pairs_per_iter=40, rng=None):
    """Iterative pair line search."""
    if rng is None:
        rng = np.random.RandomState(42)
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    
    for it in range(n_iter):
        improved = False
        # Pick random pairs
        pairs = []
        for _ in range(n_pairs_per_iter):
            i = rng.randint(n)
            j = rng.randint(n)
            if i != j:
                pairs.append((i, j))
        
        for i, j in pairs:
            h_new, c5_new = exact_pair_line_search(h, i, j, dx, target_sum)
            if c5_new < best_c5 - 1e-14:
                h = h_new
                best_c5 = c5_new
                best_h = h.copy()
                improved = True
        
        if not improved:
            # Try all pairs (expensive but thorough)
            improved2 = False
            for i in range(n):
                for j in range(i+1, n):
                    h_new, c5_new = exact_pair_line_search(h, i, j, dx, target_sum)
                    if c5_new < best_c5 - 1e-14:
                        h = h_new
                        best_c5 = c5_new
                        best_h = h.copy()
                        improved2 = True
            if not improved2:
                break
    
    return best_h, best_c5

def active_set_qp(h, dx, target_sum, n_active=30, step_size=0.1):
    """LP-based direction with exact line search."""
    n = len(h)
    corrs = compute_corrs(h)
    top_idx = np.argsort(corrs)[-n_active:]
    
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_rows = []
    b_rows = []
    for k in top_idx:
        row = np.zeros(n + 1)
        row[:n] = dx * grad_corr_k(h, k, n)
        row[n] = -1.0
        A_rows.append(row)
        b_rows.append(-corrs[k] * dx - 1e-13)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    bounds = []
    for i in range(n):
        bounds.append((max(-h[i], -step_size), min(1.0 - h[i], step_size)))
    bounds.append((0.0, None))
    
    try:
        res = linprog(c_obj, A_ub=np.array(A_rows), b_ub=np.array(b_rows),
                      A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    except Exception:
        return h, c5_of_h(h, dx)
    if res.status != 0:
        return h, c5_of_h(h, dx)
    
    delta = res.x[:n]
    if np.max(np.abs(delta)) < 1e-14:
        return h, c5_of_h(h, dx)
    
    # Exact line search along delta
    h_dir = project_feasible(h + delta, target_sum)  # direction normalized to be feasible at step 1
    # Actually, let's do line search: h(t) = h + t * delta, projected
    def f_line(t):
        h_t = h + t * delta
        h_t = project_feasible(h_t, target_sum)
        return c5_of_h(h_t, dx)
    
    # Search t in [0, 1] (or slightly beyond)
    res_ls = minimize_scalar(f_line, bounds=(0, 1.5), method='bounded',
                             options={'xatol': 1e-10})
    t_opt = res_ls.x
    h_new = project_feasible(h + t_opt * delta, target_sum)
    return h_new, c5_of_h(h_new, dx)

def block_l_bfgs(h0, dx, target_sum, block_size=4, n_iter=100, beta=500):
    """Optimize blocks of coordinates using L-BFGS."""
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    rng = np.random.RandomState(123)
    
    for it in range(n_iter):
        # Pick a random block
        start = rng.randint(n - block_size + 1)
        idx = list(range(start, start + block_size))
        
        h_block = h[idx].copy()
        
        def sub_obj(block_vals):
            h_trial = h.copy()
            h_trial[idx] = block_vals
            h_trial = np.clip(h_trial, 0, 1)
            # Adjust for sum constraint: need sum(h_trial) = target_sum
            h_trial = project_feasible(h_trial, target_sum)
            return smooth_max_obj(h_trial, dx, beta)
        
        def sub_grad(block_vals):
            h_trial = h.copy()
            h_trial[idx] = block_vals
            h_trial = np.clip(h_trial, 0, 1)
            h_trial = project_feasible(h_trial, target_sum)
            g = smooth_max_grad(h_trial, dx, beta)
            return g[idx]
        
        try:
            res = minimize(sub_obj, h_block, jac=sub_grad, method='L-BFGS-B',
                          bounds=[(0,1)]*block_size, options={'maxiter': 50})
            h_new = h.copy()
            h_new[idx] = res.x
            h_new = project_feasible(h_new, target_sum)
            c5_new = c5_of_h(h_new, dx)
            if c5_new < best_c5 - 1e-14:
                h = h_new
                best_c5 = c5_new
                best_h = h.copy()
        except Exception:
            pass
    
    return best_h, best_c5

min_{v ∈ [max(0, target_sum - sum(h excluding i)), min(1, same)]} max_k (A_k + B_k * v)

The minimum of the max of linear functions is at a point where two (or more) of the lines cross at their maximum. We can find this by:
1. Computing all the (A_k, B_k) pairs
2. Finding the v that minimizes max_k (A_k + B_k * v)

This is a 1D linear programming problem! We want:
min t
s.t. A_k + B_k * v <= t for all k
     lower <= v <= upper

This can be solved in O(n log n) using a convex hull / rotating calipers approach, or O(n²) by checking all pairs.

- Check if v* is in the valid range
- At v*, compute max_k (A_k + B_k * v*)
- The answer is the minimum over all such v* and the endpoints

import numpy as np
from scipy.optimize import minimize, linprog, minimize_scalar
from scipy.special import logsumexp
import time


def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -2.0, 3.0
    for _ in range(120):
        mid = (lo + hi) / 2.0
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            adj = diff / mask.sum()
            if h[mask].min() + adj >= 0 and h[mask].max() + adj <= 1:
                h[mask] += adj
    return np.clip(h, 0.0, 1.0)


def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def c5_of_h(h, dx):
    return compute_corrs(h).max() * dx


def grad_corr_k(h, k, n):
    offset = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -offset)
    i_hi = min(n - 1, n - 1 - offset)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx] += (1.0 - h[idx + offset])
    grad[idx + offset] += (-h[idx])
    return grad


def smooth_max_obj(h, dx, beta):
    corrs = compute_corrs(h)
    return (logsumexp(beta * corrs) / beta) * dx


def smooth_max_grad(h, dx, beta):
    n = len(h)
    corrs = compute_corrs(h)
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones_like(corrs) / len(corrs)
    else:
        w /= w_sum
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] > 1e-13:
            grad += w[k] * grad_corr_k(h, k, n)
    return grad * dx


def slsqp_solve(h0, dx, target_sum, beta, maxiter=2000):
    n = len(h0)

    def obj(x):
        return smooth_max_obj(x, dx, beta)

    def grad_obj(x):
        return smooth_max_grad(x, dx, beta)

    def c_fun(x):
        return x.sum() - target_sum

    def c_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': c_fun, 'jac': c_jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0, jac=grad_obj, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        h_out = project_feasible(res.x, target_sum)
    except Exception:
        h_out = project_feasible(h0, target_sum)
    return h_out, c5_of_h(h_out, dx)


def pair_line_search(h, i, j, dx, target_sum):
    eps_lo = max(-h[j], -(1.0 - h[i]))
    eps_hi = min(h[i], 1.0 - h[j])
    if eps_hi - eps_lo < 1e-15:
        return h, c5_of_h(h, dx)

    def f1d(eps):
        h_tmp = h.copy()
        h_tmp[i] = np.clip(h[i] + eps, 0.0, 1.0)
        h_tmp[j] = np.clip(h[j] - eps, 0.0, 1.0)
        return compute_corrs(h_tmp).max() * dx

    # Grid search
    grid = np.linspace(eps_lo, eps_hi, 25)
    vals = np.array([f1d(e) for e in grid])
    best_idx = np.argmin(vals)

    if best_idx > 0 and best_idx < len(grid) - 1:
        a = grid[best_idx - 1]
        b = grid[best_idx + 1]
    else:
        a, b = eps_lo, eps_hi

    # Golden section
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c_pt = b - gr * (b - a)
    d_pt = a + gr * (b - a)
    fc, fd = f1d(c_pt), f1d(d_pt)
    for _ in range(35):
        if b - a < 1e-13:
            break
        if fc < fd:
            b = d_pt
            d_pt = c_pt
            fd = fc
            c_pt = b - gr * (b - a)
            fc = f1d(c_pt)
        else:
            a = c_pt
            c_pt = d_pt
            fc = fd
            d_pt = a + gr * (b - a)
            fd = f1d(d_pt)

    eps_opt = (a + b) / 2.0
    h_new = h.copy()
    h_new[i] = np.clip(h[i] + eps_opt, 0.0, 1.0)
    h_new[j] = np.clip(h[j] - eps_opt, 0.0, 1.0)
    h_new = project_feasible(h_new, target_sum)
    return h_new, c5_of_h(h_new, dx)


def gradient_guided_pair_descent(h0, dx, target_sum, n_iter=200, rng=None, beta_guide=200.0):
    if rng is None:
        rng = np.random.RandomState(0)
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    stall = 0

    for it in range(n_iter):
        # Compute gradient to guide pair selection
        grad = smooth_max_grad(h, dx, beta_guide)

        # Most promising pairs: i with high grad (want to decrease) and j with low grad (want to increase)
        # We transfer from i to j: h[i] += eps, h[j] -= eps
        # So we want grad[i] < 0 (decreasing h[i] helps) and grad[j] > 0 (increasing h[j] helps)
        # Actually: if grad[i] > 0, increasing h[i] increases objective, so we want to decrease h[i]
        # Transfer from i to j means h[i] decreases, h[j] increases
        # So we want grad[i] > 0 and grad[j] < 0

        dec_candidates = np.where(grad > 0.01 * np.max(np.abs(grad)))[0]
        inc_candidates = np.where(grad < -0.01 * np.max(np.abs(grad)))[0]

        if len(dec_candidates) == 0 or len(inc_candidates) == 0:
            dec_candidates = np.argsort(grad)[-n // 4:]
            inc_candidates = np.argsort(grad)[:n // 4]

        improved = False
        n_tries = min(60, len(dec_candidates) * len(inc_candidates))
        for _ in range(n_tries):
            i = dec_candidates[rng.randint(len(dec_candidates))]
            j = inc_candidates[rng.randint(len(inc_candidates))]
            if i == j:
                continue
            h_new, c5_new = pair_line_search(h, i, j, dx, target_sum)
            if c5_new < best_c5 - 1e-14:
                h = h_new
                best_c5 = c5_new
                best_h = h.copy()
                improved = True

        if not improved:
            stall += 1
            if stall > 20:
                break
        else:
            stall = 0

    return best_h, best_c5


def random_pair_descent(h0, dx, target_sum, n_iter=150, rng=None):
    if rng is None:
        rng = np.random.RandomState(1)
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    stall = 0

    for it in range(n_iter):
        improved = False
        for _ in range(40):
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            h_new, c5_new = pair_line_search(h, i, j, dx, target_sum)
            if c5_new < best_c5 - 1e-14:
                h = h_new
                best_c5 = c5_new
                best_h = h.copy()
                improved = True
        if not improved:
            stall += 1
            if stall > 25:
                break
        else:
            stall = 0

    return best_h, best_c5


def active_set_lp_line_search(h, dx, target_sum, n_active=40, step=0.1):
    n = len(h)
    corrs = compute_corrs(h)
    top_idx = np.argsort(corrs)[-n_active:]

    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_rows = []
    b_rows = []
    for k in top_idx:
        row = np.zeros(n + 1)
        row[:n] = dx * grad_corr_k(h, k, n)
        row[n] = -1.0
        A_rows.append(row)
        b_rows.append(-corrs[k] * dx - 1e-13)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)

    bounds = []
    for i in range(n):
        bounds.append((max(-h[i], -step), min(1.0 - h[i], step)))
    bounds.append((0.0, None))

    try:
        res = linprog(c_obj, A_ub=np.array(A_rows), b_ub=np.array(b_rows),
                      A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    except Exception:
        return h, c5_of_h(h, dx)
    if res.status != 0:
        return h, c5_of_h(h, dx)

    delta = res.x[:n]
    if np.max(np.abs(delta)) < 1e-14:
        return h, c5_of_h(h, dx)

    # Exact line search
    def f_line(t):
        h_t = h + t * delta
        h_t = project_feasible(h_t, target_sum)
        return c5_of_h(h_t, dx)

    try:
        res_ls = minimize_scalar(f_line, bounds=(0.0, 2.0), method='bounded',
                                 options={'xatol': 1e-10})
        t_opt = res_ls.x
    except Exception:
        t_opt = 1.0

    h_new = project_feasible(h + t_opt * delta, target_sum)
    return h_new, c5_of_h(h_new, dx)


def iterative_lp_polish(h0, dx, target_sum, max_rounds=50):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    step = 0.15
    stall = 0

    for r in range(max_rounds):
        n_active = min(50, max(20, 2 * len(h) // 3))
        h_new, c5_new = active_set_lp_line_search(h, dx, target_sum, n_active, step)
        if c5_new < best_c5 - 1e-14:
            h = h_new
            best_c5 = c5_new
            best_h = h.copy()
            stall = 0
            step = min(step * 1.1, 0.15)
        else:
            stall += 1
            step *= 0.7
            if stall > 15 or step < 1e-6:
                break

    return best_h, best_c5


def momentum_subgrad(h0, dx, target_sum, max_iter=8000, beta_smooth=50.0):
    n = len(h0)
    h = h0.copy()
    momentum = np.zeros(n)
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    no_improve = 0
    lr = 0.5 / np.sqrt(n)
    beta_mom = 0.85

    for it in range(max_iter):
        corrs = compute_corrs(h)
        c5_val = corrs.max() * dx

        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 3000:
            break

        shifted = corrs - corrs.max()
        w = np.exp(beta_smooth * shifted)
        w /= w.sum()

        grad = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] > 1e-12:
                grad += w[k] * grad_corr_k(h, k, n)
        grad *= dx

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        momentum = beta_mom * momentum - lr * grad / (gnorm + 1e-10)
        h_new = project_feasible(h + momentum, target_sum)
        new_c5 = c5_of_h(h_new, dx)

        tries = 0
        while new_c5 > c5_val + 1e-14 and tries < 5:
            lr *= 0.5
            momentum = beta_mom * momentum - lr * grad / (gnorm + 1e-10)
            h_new = project_feasible(h + momentum, target_sum)
            new_c5 = c5_of_h(h_new, dx)
            tries += 1

        h = h_new

    return best_h, best_c5


def full_optimize(h0, dx, target_sum, time_left_fn, rng):
    n = len(h0)
    best_h = h0.copy()
    best_c5 = c5_of_h(h0, dx)

    for beta in [100, 300, 800, 2000, 5000, 12000]:

        if c5_val < best_c5:

    # LP polish
    if time_left_fn() > 20:
        h_lp, c5_lp = iterative_lp_polish(best_h, dx, target_sum, max_rounds=40)
        if c5_lp < best_c5:
            best_c5 = c5_lp
            best_h = h_lp

    # Gradient-guided pair descent
    if time_left_fn() > 30:
        h_pg, c5_pg = gradient_guided_pair_descent(best_h, dx, target_sum,
                                                    n_iter=150, rng=rng)
        if c5_pg < best_c5:
            best_c5 = c5_pg
            best_h = h_pg

    # Random pair descent
    if time_left_fn() > 25:
        h_rp, c5_rp = random_pair_descent(best_h, dx, target_sum,
                                           n_iter=100, rng=rng)
        if c5_rp < best_c5:
            best_c5 = c5_rp
            best_h = h_rp

    # More SLSQP
    if time_left_fn() > 20:
        for beta in [5000, 15000]:
            if time_left_fn() < 15:

            h_opt, c5_val = slsqp_solve(best_h, dx, target_sum, beta, 2000)

    if time_left_fn() > 15:
        h_pg2, c5_pg2 = gradient_guided_pair_descent(best_h, dx, target_sum,

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s - (time.time() - t0)

    # Start from initial_h_values
    h0 = np.array(initial_h_values, dtype=float)
    n = len(h0)
    dx = 2.0 / n
    target_sum = n / 2.0
    h0 = project_feasible(h0, target_sum)

    best_h = h0.copy()
    best_c5 = c5_of_h(h0, dx)

    # Phase 1: Main optimization from initial construction
    h_opt, c5_val = full_optimize(h0, dx, target_sum, time_left, rng)
    if c5_val < best_c5:
        best_c5 = c5_val
        best_h = h_opt

    # Phase 2: Perturbation restarts
    for trial in range(20):
        if time_left() < 25:
            break
        if trial % 4 == 0:
            perturb = 0.03 * (rng.random(n) - 0.5)
        elif trial % 4 == 1:
            perturb = 0.1 * (rng.random(n) - 0.5)
        elif trial % 4 == 2:
            idx = rng.choice(n, size=max(3, n // 8), replace=False)
            perturb = np.zeros(n)
            perturb[idx] = rng.choice([-1, 1], size=len(idx)) * 0.08
        else:
            # Block perturbation
            block = n // 6
            start = rng.randint(0, n - block)
            perturb = np.zeros(n)
            perturb[start:start + block] = (rng.random(block) - 0.5) * 0.2

        h_try = project_feasible(best_h + perturb, target_sum)
        h_opt2, c5_val2 = full_optimize(h_try, dx, target_sum, time_left, rng)
        if c5_val2 < best_c5:
            best_c5 = c5_val2
            best_h = h_opt2

    # Phase 3: Try different n values
    n_current = len(best_h)
    n_tries = []
    for offset in [-20, -15, -10, -5, -3, -2, -1, 1, 2, 3, 5, 10, 15, 20, 30, 40, 50]:
        nn = n_current + offset
        if nn >= 40 and nn % 2 == 0 and nn != n_current:
            n_tries.append(nn)
    n_tries = sorted(set(n_tries))[:12]

    for n_try in n_tries:
        if time_left() < 35:
            break
        dx_try = 2.0 / n_try
        ts_try = n_try / 2.0
        x_orig = np.linspace(0.0, 2.0, n_current, endpoint=False) + 1.0 / n_current
        x_new = np.linspace(0.0, 2.0, n_try, endpoint=False) + 1.0 / n_try
        h_interp = np.interp(x_new, x_orig, best_h)
        h_interp = project_feasible(h_interp, ts_try)

        h_opt3, c5_val3 = full_optimize(h_interp, dx_try, ts_try, time_left, rng)
        if c5_val3 < best_c5:
            best_c5 = c5_val3
            best_h = h_opt3

    # Phase 4: Final intensive polish
    if time_left() > 25:
        n_b = len(best_h)
        dx_b = 2.0 / n_b
        ts_b = n_b / 2.0

        # Very high beta SLSQP
        if time_left() > 15:
            h_f, c5_f = slsqp_solve(best_h, dx_b, ts_b, 20000, 3000)
            if c5_f < best_c5:
                best_c5 = c5_f
                best_h = h_f

        # Fine LP
        if time_left() > 10:
            h_lp_f, c5_lp_f = iterative_lp_polish(best_h, dx_b, ts_b, max_rounds=40)
            if c5_lp_f < best_c5:
                best_c5 = c5_lp_f
                best_h = h_lp_f

        # Pair descent
        if time_left() > 10:
            h_p_f, c5_p_f = gradient_guided_pair_descent(best_h, dx_b, ts_b,
                                                          n_iter=100, rng=rng)
            if c5_p_f < best_c5:
                best_c5 = c5_p_f
                best_h = h_p_f

    # Final validation
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final / 2.0)
    best_c5 = c5_of_h(best_h, 2.0 / n_final)

    return best_h, best_c5, n_final
