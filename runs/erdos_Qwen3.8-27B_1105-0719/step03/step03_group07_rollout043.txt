def build_gradient_matrix(h, rows):
    n=len(h)
    m=len(rows)
    G=np.zeros((m,n),dtype=np.float64)
    dx=2.0/n
    for r, row in enumerate(rows):
       s = n - 1 - int(row)
       if s >= 0:
          for i in range(n - s):
             G[r, i] += dx * (1.0 - h[i+s])
             G[r, i+s] -= dx * h[i]
       else:
          rshift = -s
          for j in range(n - rshift):
             G[r, j + rshift] += dx * (1.0 - h[j])
             G[r, j] -= dx * h[j + rshift]
    return G

import time
import numpy as np
from scipy.optimize import minimize, linprog

_SL_N = 0
_SL_DX = 0.0
_SL_TARGET = 0.0
_SL_ACTIVE = np.empty(0, dtype=int)


def time_ok(t_start, budget_s, margin=5.0):
    return (time.perf_counter() - t_start) < float(budget_s) - float(margin)


def profile(h):
    n = len(h)
    if n <= 0:
        return np.empty(0, dtype=np.float64)
    dx = 2.0 / float(n)
    return np.correlate(h, 1.0 - h, mode="full") * dx


def exact_score(h):
    if len(h) == 0:
        return float("inf")
    return float(np.max(profile(h)))


def repair_feasible(vec):
    h = np.asarray(vec, dtype=np.float64).copy()
    n = len(h)
    if n == 0:
        return h
    if not np.all(np.isfinite(h)):
        h = np.full(n, 0.5, dtype=np.float64)
    np.clip(h, 0.0, 1.0, out=h)

    target = 0.5 * n
    diff = target - h.sum()

    if diff > 1e-14:
        caps = 1.0 - h
        order = np.argsort(-caps)
        rem = diff
        for idx in order:
            if rem <= 1e-15:
                break
            add = min(rem, caps[idx])
            if add > 0.0:
                h[idx] += add
                rem -= add
    elif diff < -1e-14:
        order = np.argsort(h)
        rem = -diff
        for idx in order:
            if rem <= 1e-15:
                break
            sub = min(rem, h[idx])
            if sub > 0.0:
                h[idx] -= sub
                rem -= sub

    for _ in range(4):
        diff = target - h.sum()
        if abs(diff) <= 1e-15:
            break
        if diff > 0.0:
            idx = int(np.argmin(h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff
        np.clip(h, 0.0, 1.0, out=h)

    return h


def symmetrize(h):
    return (h + h[::-1].copy()) * 0.5


def interpolate_to_n(h, new_n):
    n = len(h)
    new_n = int(new_n)
    if new_n < 2:
        new_n = 2
    if n == new_n:
        return repair_feasible(h)
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_x = np.empty(n + 2, dtype=np.float64)
    full_h = np.empty(n + 2, dtype=np.float64)
    full_x[0] = 0.0
    full_x[1:-1] = xs
    full_x[-1] = 2.0
    full_h[0] = h[0]
    full_h[1:-1] = h
    full_h[-1] = h[-1]
    h_new = np.interp(xt, full_x, full_h)
    return repair_feasible(np.clip(h_new, 0.0, 1.0))


def build_gradient_matrix(h, rows):
    n = len(h)
    rcount = len(rows)
    G = np.zeros((rcount, n), dtype=np.float64)
    dx = 2.0 / float(n)

    for r, row in enumerate(rows):
        s = n - 1 - int(row)
        if s >= 0:
            for i in range(n - s):
                G[r, i] += dx * (1.0 - h[i + s])
                G[r, i + s] -= dx * h[i]
        else:
            rshift = -s
            for j in range(n - rshift):
                G[r, j + rshift] += dx * (1.0 - h[j])
                G[r, j] -= dx * h[j + rshift]
    return G


def select_active(P, K, near_tol=1e-7):
    M = len(P)
    K = int(K)
    if K <= 0:
        return np.empty(0, dtype=int)
    if K >= M:
        return np.arange(M, dtype=int)

    c = float(P.max())
    cands = np.where(P >= c - near_tol)[0]
    if cands.size >= K:
        order = cands[np.argsort(P[cands])]
        return order[-K:]
    return np.argsort(P)[-K:]


def direction_line_search(h, d, max_alpha, best_c):
    d = np.asarray(d, dtype=np.float64).copy()
    n = len(h)
    d -= np.mean(d)

    if max_alpha is None or not np.isfinite(max_alpha) or max_alpha <= 0.0:
        return h, best_c
    if np.max(np.abs(d)) <= 1e-15:
        return h, best_c

    alpha_max = float(max_alpha)
    for i in range(n):
        if d[i] > 1e-14:
            a = (1.0 - h[i]) / d[i]
            if a < alpha_max:
                alpha_max = a
        elif d[i] < -1e-14:
            a = (-h[i]) / d[i]
            if a < alpha_max:
                alpha_max = a

    if not np.isfinite(alpha_max) or alpha_max <= 1e-15:
        return h, best_c
    if alpha_max > 5.0:
        alpha_max = 5.0

    best_h = h
    bc = float(best_c)
    fractions = (1.0, 0.7, 0.5, 0.35, 0.25, 0.15, 0.1, 0.05, 0.025, 0.0125)

    for frac in fractions:
        a = frac * alpha_max
        if a <= 0.0:
            break
        trial = np.clip(h + a * d, 0.0, 1.0)
        trial = repair_feasible(trial)
        c = exact_score(trial)
        if c < bc - 1e-14:
            bc = c
            best_h = trial

    tiny = 1e-5 * alpha_max
    if tiny > 0.0:
        trial = np.clip(h + tiny * d, 0.0, 1.0)
        trial = repair_feasible(trial)
        c = exact_score(trial)
        if c < bc - 1e-14:
            bc = c
            best_h = trial

    return best_h, bc


def lp_step(h, K, D, best_c=None):
    n = len(h)
    P = profile(h)
    c = float(P.max())
    bc = float(best_c) if best_c is not None else c
    idx = select_active(P, K)
    if idx.size == 0:
        return h, bc

    G = build_gradient_matrix(h, idx)

    cvec = np.zeros(n + 1, dtype=np.float64)
    cvec[-1] = 1.0

    A_ub = np.empty((idx.size, n + 1), dtype=np.float64)
    for i in range(idx.size):
        A_ub[i, :n] = G[i]
        A_ub[i, -1] = -1.0
    b_ub = np.zeros(idx.size, dtype=np.float64)

    A_eq = np.zeros((1, n + 1), dtype=np.float64)
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1, dtype=np.float64)

    bounds = []
    for i in range(n):
        hi = 1.0 - h[i]
        if hi < 0.0:
            hi = 0.0
        lo = h[i]
        if lo < 0.0:
            lo = 0.0
        bounds.append((float(-min(D, lo)), float(min(D, hi))))
    bounds.append((-5.0, 5.0))

    res = None
    try:
        res = linprog(cvec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method="highs")
    except Exception:
        try:
            res = linprog(cvec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method="simplex")
        except Exception:
            res = None

    if res is None or not getattr(res, "success", False) or res.x is None:
        return h, bc
    if not np.all(np.isfinite(res.x)):
        return h, bc

    t = float(res.x[-1])
    if t >= -1e-8:
        return h, bc

    d = res.x[:n].copy()
    d -= np.mean(d)

    for i in range(n):
        if h[i] <= 1e-12 and d[i] < -1e-12:
            d[i] = 0.0
        if h[i] >= 1.0 - 1e-12 and d[i] > 1e-12:
            d[i] = 0.0
    d -= np.mean(d)

    return direction_line_search(h, d, 1.0, bc)


def softmax_step(h, K, beta, step, best_c=None):
    n = len(h)
    P = profile(h)
    c = float(P.max())
    bc = float(best_c) if best_c is not None else c
    idx = select_active(P, K)
    if idx.size == 0:
        return h, bc

    q = P[idx]
    shift = float(q.max())
    w = np.exp(beta * (q - shift))
    s = float(w.sum())
    if (not np.isfinite(s)) or s <= 1e-300:
        w = np.full(idx.size, 1.0 / float(idx.size), dtype=np.float64)
    else:
        w = w / s

    G = build_gradient_matrix(h, idx)
    grad = G.T.dot(w)
    grad -= np.mean(grad)

    norm = float(np.max(np.abs(grad)))
    if norm <= 1e-15:
        return h, bc

    d = -grad / norm
    return direction_line_search(h, d, step, bc)

        return np.empty(0, dtype=np.float64)

    P = profile(h)
    return x[-1] - P[_SL_ACTIVE]

    J = np.zeros((len(_SL_ACTIVE), n + 1), dtype=np.float64)
    if len(_SL_ACTIVE) == 0:
        return J
    G = build_gradient_matrix(x[:-1], _SL_ACTIVE)
    J[:, :-1] = -G
    J[:, -1] = 1.0
    return J


def slsqp_active(h0, K, maxiter=200):
    global _SL_N, _SL_DX, _SL_TARGET, _SL_ACTIVE

    n = len(h0)
    dx = 2.0 / float(n)
    target = 0.5 * float(n)
    P = profile(h0)
    c0 = float(P.max())
    idx = select_active(P, K)
    if idx.size == 0:
        return None

    _SL_N = n
    _SL_DX = dx
    _SL_TARGET = target
    _SL_ACTIVE = idx

    x0 = np.empty(n + 1, dtype=np.float64)
    x0[:n] = h0
    x0[-1] = c0 + 1e-5

    bounds = []
    for _ in range(n):
        bounds.append((0.0, 1.0))
    bounds.append((0.0, 1.0))

    cons = [
        {"type": "eq", "fun": slsqp_eq, "jac": slsqp_eq_jac},
        {"type": "ineq", "fun": slsqp_ineq, "jac": slsqp_ineq_jac},
    ]

    try:
        res = minimize(
            slsqp_obj,
            x0,
            method="SLSQP",
            jac=slsqp_obj_jac,
            bounds=bounds,
            constraints=cons,
            options={"maxiter": int(maxiter), "ftol": 1e-12},
        )
    except Exception:
        return None

    if res is None or res.x is None or not np.all(np.isfinite(res.x)):
        return None

    h_new = repair_feasible(res.x[:n])
    c_new = exact_score(h_new)
    if c_new <= c0 + 1e-8:
        return h_new
    return None


def try_pair_move(h, i, j, best_c):
    s = h[i] + h[j]
    lo = max(0.0, s - 1.0)
    hi = min(1.0, s)
    if hi - lo < 1e-12:
        return h, best_c, False

    max_delta = min(0.1, 0.25 * (hi - lo))
    deltas = (max_delta, 0.5 * max_delta, 0.25 * max_delta, 0.125 * max_delta,
              0.05, 0.02, 0.01, 0.005, 0.001)

    for sign in (1.0, -1.0):
        for dlt in deltas:
            new_i = h[i] + sign * dlt
            if new_i < lo - 1e-12 or new_i > hi + 1e-12:
                continue
            trial = h.copy()
            trial[i] = new_i
            trial[j] = s - new_i
            trial = repair_feasible(trial)
            c = exact_score(trial)
            if c < best_c - 1e-14:
                return trial, c, True
    return h, best_c, False

def balance_binary(h):
    h = np.asarray(h, dtype=np.float64).copy()
    n = len(h)
    target = n // 2
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h <= 0.5)[0]
    cnt = int(len(ones))
    if cnt > target:
        for k in range(cnt - target):
            h[ones[k]] = 0.0
    elif cnt < target:
        for k in range(target - cnt):
            h[zeros[k]] = 1.0
    return repair_feasible(h)

        for delta in (30, 40):
            nn = base_n - delta
            if nn >= 30:
                recs.append((delta, nn, 1))
            nn = base_n + delta
            if nn <= 140:
                recs.append((delta, nn, 2))

        seen = set()
        unique = []
        for rec in recs:
            nn = rec[1]
            if (nn % 2 == 0) and nn not in seen:
                seen.add(nn)
                unique.append(rec)
        unique.sort()

        taken = 0.0
        for rec in unique:
            nn = rec[1]
            if not time_ok(t_start, budget_s, 100):
                break
            if taken >= dim_budget:
                break
            start = time.perf_counter()
            limit = min(30.0, dim_budget - taken, budget_s - (time.perf_counter() - t_start) - 120.0)
            if limit < 8.0:
                break
            h_interp = interpolate_to_n(best_h, nn)
            maxit = 80 if limit < 20.0 else 120
            h_opt, c_opt = optimize_at_n(h_interp, t_start, budget_s, limit,
                                         use_slsqp=(limit > 15.0),
                                         slsqp_maxiter=maxit,
                                         seed=seed + 500 + nn)
            taken += time.perf_counter() - start
            if c_opt < best_c - 1e-12:
                best_h = h_opt
                best_c = c_opt

    pert_budget = min(0.10 * budget_s, max(0.0, budget_s - (time.perf_counter() - t_start) - 120.0))
    if pert_budget > 30.0:
        taken = 0.0
        n = len(best_h)
        for trial in range(12):
            if not time_ok(t_start, budget_s, 100):
                break
            if taken >= pert_budget:
                break
            eps = 0.005 + 0.008 * (trial % 3)
            z = rng.randn(n)
            z -= z.mean()
            hp = repair_feasible(best_h + eps * z)
            cp = exact_score(hp)
            if cp <= best_c + 0.02:
                start = time.perf_counter()
                limit = min(30.0, pert_budget - taken, budget_s - (time.perf_counter() - t_start) - 100.0)
                if limit < 8.0:
                    break
                h_opt, c_opt = optimize_at_n(hp, t_start, budget_s, limit,
                                             use_slsqp=False, slsqp_maxiter=80,
                                             seed=seed + 800 + trial)
                taken += time.perf_counter() - start
                if c_opt < best_c - 1e-12:
                    best_h = h_opt
                    best_c = c_opt

    left = budget_s - (time.perf_counter() - t_start) - 20.0
    if left > 25.0:
        h_opt, c_opt = optimize_at_n(best_h, t_start, budget_s, left,
                                     use_slsqp=True, slsqp_maxiter=250,
                                     seed=seed + 999)
        if c_opt < best_c - 1e-13:
            best_h = h_opt
            best_c = c_opt

    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)

    try:
        ev = evaluate_erdos_solution(best_h)
        val = None
        if isinstance(ev, (float, int)):
            val = float(ev)
        elif isinstance(ev, tuple):
            for item in ev:
                if isinstance(item, (float, int)):
                    val = float(item)
                    break
        if val is not None and np.isfinite(val) and val < final_c:
            final_c = val
    except Exception:
        pass

    return best_h, float(final_c), int(len(best_h))
