```python
import numpy as np
from scipy.optimize import linprog, minimize
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    prof = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(prof))

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx])
            h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def profile_h(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def build_gradient_for_lag(h, d, n, dx):
    """Build gradient of P(d) w.r.t. h for lag d (in correlative output index).
    
    The output array from np.correlate(h, 1-h, 'full') has index m,
    and the corresponding lag is d = m - (n-1).
    
    For lag d >= 0: P(d) = dx * sum_{i=0}^{n-d-1} h[i+d]*(1-h[i])
    For lag d < 0:  P(d) = dx * sum_{i=0}^{n-|d|-1} h[i]*(1-h[i+|d|])
    """
    g = np.zeros(n)
    abs_d = abs(d)
    if abs_d == 0:
        # P(0) = dx * sum h[i]*(1-h[i]) = dx * (sum h - sum h^2)
        # dP/dh[i] = dx * (1 - 2*h[i])
        g[:] = dx * (1.0 - 2.0 * h)
    elif d > 0:
        # P(d) = dx * sum_{i=0}^{n-d-1} h[i+d]*(1-h[i])
        # = dx * sum h[i+d] - dx * sum h[i+d]*h[i]
        # dP/dh[j]:
        #   from first sum: dx if j >= d and j < n (i.e., j-d < n-d, always true for j in [d, n-1])
        #   Wait: first sum is over i=0..n-d-1 of h[i+d], so h[j] appears when j = i+d, i.e., i = j-d
        #   This requires 0 <= j-d <= n-d-1, i.e., d <= j <= n-1
        #   So contribution: dx for j in [d, n-1]
        #   from second sum: -dx * (h[i] if j = i+d, i.e., i=j-d, valid if d<=j<=n-1: contrib -dx*h[j-d])
        #                    -dx * (h[i+d] if j = i, i.e., i=j, valid if 0<=j<=n-d-1: contrib -dx*h[j+d])
        for j in range(d, n):
            g[j] += dx  # from linear part
            g[j] -= dx * h[j - d]  # from quadratic, h[j] as h[i+d]
        for j in range(0, n - d):
            g[j] -= dx * h[j + d]  # from quadratic, h[j] as h[i]
    else:
        # d < 0, let d' = -d > 0
        # P(d) = dx * sum_{i=0}^{n-d'-1} h[i]*(1-h[i+d'])
        # = dx * sum h[i] - dx * sum h[i]*h[i+d']
        # dP/dh[j]:
        #   from first sum: dx if 0 <= j <= n-d'-1
        #   from second sum: -dx * (h[j+d'] if 0<=j<=n-d'-1) - dx * (h[j-d'] if d'<=j<=n-1)
        dp = -d
        for j in range(0, n - dp):
            g[j] += dx  # from linear part
            g[j] -= dx * h[j + dp]  # from quadratic, h[j] as h[i]
        for j in range(dp, n):
            g[j] -= dx * h[j - dp]  # from quadratic, h[j] as h[i+d']
    return g

def active_lp_direction(h, active_lags, n):
    """Solve LP to find descent direction for active constraints.
    
    min t
    s.t. G_j · v <= t  for each active lag j
         sum(v) = 0
         -1 <= v_i <= 1
    """
    dx = 2.0 / n
    K = len(active_lags)
    
    # Build constraint matrix
    # Variables: v_0, ..., v_{n-1}, t  (total n+1)
    # Inequality: G_j · v - t <= 0 for each j  =>  [G_j | -1] · [v; t] <= 0
    A_ub = np.zeros((K, n + 1))
    for idx, lag in enumerate(active_lags):
        g = build_gradient_for_lag(h, lag, n, dx)
        A_ub[idx, :n] = g
        A_ub[idx, n] = -1.0
    b_ub = np.zeros(K)
    
    # Equality: sum(v) = 0  =>  [1,1,...,1, 0] · [v; t] = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Objective: minimize t  =>  c = [0, 0, ..., 0, 1]
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    # Bounds: -1 <= v_i <= 1, t is free (but we can bound it)
    bounds = [(-1.0, 1.0)] * n + [(-1.0, None)]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.success and res.x is not None:
        t_val = res.x[n]
        if t_val < -1e-8:  # meaningful descent
            return res.x[:n], t_val
    return None, 0.0

def line_search(h, direction, current_c, n, max_alpha=1.0, min_alpha=1e-6):
    """Binary line search along direction, maintaining feasibility."""
    lo = 0.0
    hi = max_alpha
    best_h = h.copy()
    best_c = current_c
    
    for _ in range(30):
        mid = (lo + hi) / 2.0
        h_try = repair_feasible(h + mid * direction / max(1.0, np.max(np.abs(direction))))
        c_try = exact_score(h_try)
        if c_try < best_c:
            best_c = c_try
            best_h = h_try.copy()
            lo = mid
        else:
            hi = mid
        if hi - lo < min_alpha:
            break
    
    return best_h, best_c

def active_set_optimize(h0, max_iters=2000, active_k=12, t_start=None, budget_s=1100):
    """Main optimization loop using active-set LP directions."""
    n = len(h0)
    h = h0.copy()
    current_c = exact_score(h)
    best_h = h.copy()
    best_c = current_c
    stuck_count = 0
    
    for it in range(max_iters):
        if t_start is not None:
            elapsed = time.perf_counter() - t_start
            if elapsed > budget_s - 30:
                break
        
        prof = profile_h(h)
        current_c = float(np.max(prof))
        
        if current_c < best_c:
            best_c = current_c
            best_h = h.copy()
            stuck_count = 0
        else:
            stuck_count += 1
        
        # Select active lags
        sorted_idx = np.argsort(prof)[::-1]
        # Pick lags within a threshold of the max
        thresh = current_c - max(0.002, 0.03 * (current_c - prof[sorted_idx[min(active_k, len(prof)-1)]]))
        active = sorted_idx[prof[sorted_idx] >= thresh]
        if len(active) < 3:
            active = sorted_idx[:min(active_k, len(prof))]
        if len(active) > active_k:
            active = active[:active_k]
        
        # Get LP direction
        direction, t_val = active_lp_direction(h, active, n)
        if direction is None:
            # Fallback: use negative sum of active gradients
            dx = 2.0 / n
            grad_sum = np.zeros(n)
            for lag in active:
                grad_sum += build_gradient_for_lag(h, lag, n, dx)
            direction = -grad_sum
            direction -= direction.mean()  # ensure sum = 0
        
        # Normalize direction
        dnorm = np.max(np.abs(direction))
        if dnorm > 1e-10:
            direction /= dnorm
        
        # Line search
        h_new, c_new = line_search(h, direction, current_c, n)
        
        if c_new < current_c - 1e-14:
            h = h_new
            stuck_count = 0
        else:
            # Try smaller step
            h_try = repair_feasible(h + 0.01 * direction)
            c_try = exact_score(h_try)
            if c_try < current_c:
                h = h_try
            else:
                stuck_count += 1
                if stuck_count > 50:
                    break
    
    return best_h, best_c
```

```python
def find_lp_direction(h, active_indices, n):
    """Find a descent direction using LP.
    
    active_indices: indices into the profile array (0 to 2n-2)
    Returns direction vector (sum = 0) or None.
    """
    dx = 2.0 / n
    K = len(active_indices)
    
    # Build gradient matrix: G[k, j] = grad_lag(h, active_indices[k], n, dx)[j]
    G = np.zeros((K, n))
    for k, idx in enumerate(active_indices):
        G[k, :] = grad_lag(h, idx, n, dx)
    
    # LP: min t s.t. G @ v - t <= 0 (for each row), sum(v) = 0, -1 <= v <= 1
    # Variables: v (n), t (1) => n+1 total
    # c = [0,...,0, 1] (minimize t)
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    A_ub = np.zeros((K, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = np.zeros(K)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    bounds = [(-1.0, 1.0)] * n + [(-10.0, 10.0)]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.success:
        t_val = res.x[-1]
        if t_val < -1e-6:  # meaningful descent
            v = res.x[:n]
            v -= v.mean()  # ensure exact zero sum
            return v, t_val
    return None, None
```

The feasible range of α is:
- For each i with v[i] > 0: α <= (1 - h[i]) / v[i]
- For each i with v[i] < 0: α <= h[i] / (-v[i])
- α >= 0

So α_max = min over valid i of the above bounds.

Within [0, α_max], the objective is the max of quadratic functions (each profile value is quadratic in α). So it's a convex function (max of convex functions), and we can find the minimum using ternary search or by noting that the minimum occurs where two profile values are equal (or at a boundary).

For simplicity, let me use a modified golden section search or just evaluate at several points.

Actually, since each P_m(α) = P_m(h + αv) is a quadratic in α (parabola), the max of these quadratics is a convex piecewise-quadratic function. The minimum of this function can be found by:
1. Evaluating at α = 0 and α = α_max
2. If the function is decreasing at α=0, try α_max/2, etc.
3. Use ternary search on [0, α_max]

Since the function is convex, ternary search works perfectly.

Let me implement a clean version:

```python
def line_search(h, v, n, max_evals=25):
    """Find best step size for h + alpha*v."""
    dx = 2.0 / n
    
    # Find alpha_max
    alpha_max = 1e10
    for i in range(n):
        if v[i] > 1e-12:
            alpha_max = min(alpha_max, (1.0 - h[i]) / v[i])
        elif v[i] < -1e-12:
            alpha_max = min(alpha_max, h[i] / (-v[i]))
    alpha_max = max(alpha_max, 0.0)
    alpha_max = min(alpha_max, 2.0)  # cap it
    
    if alpha_max < 1e-12:
        return h, exact_score(h)
    
    # Ternary search
    lo = 0.0
    hi = alpha_max
    c0 = exact_score(h)
    
    for _ in range(max_evals):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = exact_score(repair_feasible(h + m1 * v))
        f2 = exact_score(repair_feasible(h + m2 * v))
        if f1 < f2:
            hi = m2
        else:
            lo = m1
        if hi - lo < 1e-8:
            break
    
    alpha = (lo + hi) / 2.0
    h_new = repair_feasible(h + alpha * v)
    c_new = exact_score(h_new)
    
    if c_new >= c0:
        return h, c0
    return h_new, c_new
```

```python
def active_set_optimize(h0, max_iters=1000, t_start=None, budget_s=1100):
    n = len(h0)
    h = h0.copy()
    current_c = exact_score(h)
    best_h = h.copy()
    best_c = current_c
    stuck = 0
    active_k = 8  # start with fewer active constraints
    
    for it in range(max_iters):
        if t_start is not None:
            if time.perf_counter() - t_start > budget_s - 40:
                break
        
        prof = profile_h(h)
        current_c = float(np.max(prof))
        
        if current_c < best_c - 1e-13:
            best_c = current_c
            best_h = h.copy()
            stuck = 0
        else:
            stuck += 1
        
        # Adaptive active set size
        if stuck > 30:
            active_k = min(active_k + 2, 20)
        elif stuck < 5:
            active_k = max(active_k - 1, 5)
        
        # Find active indices (top K profile values)
        sorted_idx = np.argsort(prof)[::-1]
        # Use a threshold to include lags close to the max
        gap = current_c - prof[sorted_idx[min(active_k, 2*n-1)]]
        thresh = current_c - max(gap * 0.5, 0.001)
        active = sorted_idx[prof[sorted_idx] >= thresh]
        if len(active) > active_k:
            active = active[:active_k]
        
        # Find LP direction
        v, t_val = find_lp_direction(h, active, n)
        if v is None:
            # Fallback: negative average gradient of active lags
            dx = 2.0 / n
            g_avg = np.zeros(n)
            for idx in active:
                g_avg += grad_lag(h, idx, n, dx)
            g_avg /= len(active)
            v = -g_avg
            v -= v.mean()
        
        # Line search
        h, c = line_search(h, v, n)
        
        if stuck > 100:
            break
    
    return best_h, best_c
```

```python
def frank_wolfe_warmup(h0, n_iters=200, t_start=None, budget_s=1100):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c = exact_score(h)
    dx = 2.0 / n
    
    for it in range(n_iters):
        if t_start is not None and time.perf_counter() - t_start > budget_s - 60:
            break
        
        prof = profile_h(h)
        current_c = float(np.max(prof))
        
        if current_c < best_c:
            best_c = current_c
            best_h = h.copy()
        
        # Pick the worst lag
        max_idx = int(np.argmax(prof))
        g = grad_lag(h, max_idx, n, dx)
        
        # Frank-Wolfe: solve min g·v s.t. 0<=v<=1, sum(v)=n/2
        # Solution: v_i = 1 for the n/2 smallest g_i
        sorted_g = np.sort(g)
        threshold = sorted_g[n // 2 - 1]  # 0-indexed, so (n/2)-1-th smallest
        v = (g <= threshold).astype(float)
        
        # Handle the case where the threshold appears multiple times
        n_ones = int(v.sum())
        if n_ones != n // 2:
            # Need to adjust
            if n_ones > n // 2:
                # Set some of the threshold-valued entries to 0
                excess = n_ones - n // 2
                thresh_idx = np.where(np.isclose(g, threshold))[0]
                # Randomly pick which to set to 0
                for i in range(excess):
                    v[thresh_idx[i]] = 0.0
            else:
                deficit = n // 2 - n_ones
                # Set some entries just above threshold to 1
                above = np.where(g > threshold)[0]
                above_sorted = above[np.argsort(g[above])]
                for i in range(deficit):
                    v[above_sorted[i]] = 1.0
        
        # Step size: use backtracking or the standard FW step size
        # h_new = (1 - t) * h + t * v, t = 2/(it+2)
        t = 2.0 / (it + 2)
        h_new = (1 - t) * h + t * v
        h_new = repair_feasible(h_new)
        
        c_new = exact_score(h_new)
        if c_new < current_c:
            h = h_new
        else:
            # Try smaller step
            for t2 in [t * 0.5, t * 0.25, t * 0.1]:
                h_try = repair_feasible((1 - t2) * h + t2 * v)
                c_try = exact_score(h_try)
                if c_try < current_c:
                    h = h_try
                    break
    
    return best_h, best_c
```
