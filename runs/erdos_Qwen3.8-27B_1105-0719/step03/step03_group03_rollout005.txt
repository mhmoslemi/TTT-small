def repair_feasible(vec):

def lp_trust_region(h0, profile, active_set, n, dx, trust, rho=0.01):
    """Solve trust-region linearized subproblem."""
    if not HAS_CVXPY:
        return None
    target = n / 2.0
    G = build_grad(h0, n, dx)
    G_a = G[active_set, :]
    P_a = profile[active_set]
    # Linearized: P_a + G_a @ (h - h0) <= t
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    cons = [
        h_var >= 0,
        h_var <= 1,
        cp.sum(h_var) == target,
        h_var >= h0 - trust,
        h_var <= h0 + trust,
        G_a @ h_var - t_var <= P_a - G_a @ h0
    ]
    obj = cp.Minimize(t_var + rho * cp.sum((h_var - h0) ** 2))
    prob = cp.Problem(obj, cons)
    for solver in [cp.ECOS, cp.SCS, cp.CBC]:
        try:
            prob.solve(solver=solver, max_iters=5000, abstol=1e-10)
            break
        except Exception:
            continue
    if h_var.value is None:
        return None
    h_sol = np.array(h_var.value).flatten()
    if not np.isfinite(h_sol).all():
        return None
    return repair_feasible(h_sol)

import numpy as np
from scipy.optimize import minimize
import time

try:
    import cvxpy as cp
    HAS_CVXPY = True

    HAS_CVXPY = False


def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))


def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx


def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx])
            h[idx] += add
            remaining -= add
            if remaining <= 1e-15:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-15:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h


def build_grad(h, n, dx):
    M = 2 * n - 1
    G = np.zeros((M, n))
    center = n - 1
    G[center, :] = dx * (1.0 - 2.0 * h)
    for d in range(1, n):
        m_pos = center + d
        idx1 = np.arange(n - d)
        G[m_pos, idx1] = -dx * h[idx1 + d]
        G[m_pos, idx1 + d] = dx * (1.0 - h[idx1])
        m_neg = center - d
        G[m_neg, idx1] = dx * (1.0 - h[idx1 + d])
        G[m_neg, idx1 + d] = -dx * h[idx1]
    return G


def lse_grad(h, profile, eta):
    n = len(h)
    dx = 2.0 / n
    c_max = np.max(profile)
    z = (profile - c_max) / eta
    z = z - np.max(z)
    w = np.exp(z)
    w /= w.sum()
    G = build_grad(h, n, dx)
    return G.T @ w


def lse_refine(h0, t_abs, deadline, reserve=5.0):
    n = len(h0)
    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    best_c = exact_score(h)
    eta = max(0.004, best_c * 0.04)
    step = 0.015
    stall = 0
    mom = np.zeros(n)
    beta = 0.6

    while time.perf_counter() < deadline - reserve:
        profile = compute_profile(h)
        c_cur = np.max(profile)
        if c_cur < best_c - 1e-14:
            best_c = c_cur
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

        grad = lse_grad(h, profile, eta)
        grad -= grad.mean()
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        grad /= gn

        accepted = False
        for s in [step, step * 0.6, step * 0.4, step * 0.2, step * 0.1]:
            mom = beta * mom - s * grad
            h_try = repair_feasible(h - s * grad + 0.3 * mom)
            c_try = exact_score(h_try)
            if c_try < c_cur - 1e-11:
                h = h_try
                accepted = True
                break
        if not accepted:
            for s in [step * 0.3, step * 0.1, step * 0.05]:
                h_try = repair_feasible(h - s * grad)
                c_try = exact_score(h_try)
                if c_try < c_cur - 1e-11:
                    h = h_try
                    mom *= 0.3
                    accepted = True
                    break
        if not accepted:
            step *= 0.65
            eta *= 0.85
            mom *= 0.3
            if step < 5e-6 or stall > 60:
                break
        else:
            step = min(step * 1.08, 0.04)
            eta = max(0.002, eta * 0.998)
    return h, best_h, best_c


def lp_step(h0, profile, n, dx, trust, max_constr=35):
    if not HAS_CVXPY:
        return None
    c_max = np.max(profile)
    spread = c_max - np.median(profile)
    beam = max(0.004, 0.12 * spread)
    active = np.where(profile >= c_max - beam)[0]
    if len(active) == 0:
        active = np.array([np.argmax(profile)])
    if len(active) > max_constr:
        active = active[np.argsort(profile[active])[-max_constr:]]
    G = build_grad(h0, n, dx)
    G_a = G[active, :]
    P_a = profile[active]
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    cons = [
        h_var >= 0,
        h_var <= 1,
        cp.sum(h_var) == n / 2.0,
        h_var >= h0 - trust,
        h_var <= h0 + trust,
        G_a @ h_var - t_var <= P_a - G_a @ h0
    ]
    obj = cp.Minimize(t_var + 0.02 * cp.sum((h_var - h0) ** 2))
    prob = cp.Problem(obj, cons)
    for solver in [cp.ECOS, cp.SCS, cp.CBC]:
        try:
            prob.solve(solver=solver, max_iters=8000, abstol=1e-10)
            break
        except Exception:
            continue
    if h_var.value is None:
        return None
    h_sol = np.array(h_var.value).flatten()
    if not np.isfinite(h_sol).all():
        return None
    return repair_feasible(h_sol)


def combined_loop(h0, t_abs, deadline, reserve=8.0, max_iters=600):
    n = len(h0)
    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    best_c = exact_score(h)
    trust = 0.012
    it = 0
    while it < max_iters and time.perf_counter() < deadline - reserve:
        it += 1
        if it % 4 == 0:
            profile = compute_profile(h)
            eta = max(0.003, 0.025 * (np.max(profile) - np.median(profile)))
            grad = lse_grad(h, profile, eta)
            grad -= grad.mean()
            gn = np.linalg.norm(grad)
            if gn > 1e-12:
                grad /= gn
                for s in [0.012, 0.006, 0.003, 0.001, 0.0005]:
                    h_try = repair_feasible(h - s * grad)
                    c_try = exact_score(h_try)
                    if c_try < best_c - 1e-12:
                        h = h_try
                        best_c = c_try
                        best_h = h.copy()
                        break
        else:
            profile = compute_profile(h)
            h_lp = lp_step(h, profile, n, dx, trust)
            if h_lp is not None:
                c_lp = exact_score(h_lp)
                if c_lp < best_c - 1e-12:
                    h = h_lp
                    best_c = c_lp
                    best_h = h.copy()

                else:
                    trust *= 0.75
            else:
                trust *= 0.5
    return h, best_h, best_c


def interp_to(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    fx = np.concatenate([[0.0], xs, [2.0]])
    fh = np.concatenate([[h[0]], h, [h[-1]]])
    hn = np.interp(xt, fx, fh)
    return repair_feasible(np.clip(hn, 0, 1))


def slsqp_polish(h0, maxiter=500, tol=1e-13):
    n = len(h0)
    dx = 2.0 / n
    M = 2 * n - 1

    def obj(x):
        return x[-1]

    def eq_fun(x):
        return x[:-1].sum() - n / 2.0

    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g

    def ineq_fun(x):
        h = x[:-1]
        return x[-1] - np.correlate(h, 1 - h, mode="full") * dx

    def ineq_jac(x):
        h = x[:-1]
        G = build_grad(h, n, dx)
        J = np.zeros((M, n + 1))
        J[:, :-1] = -G
        J[:, -1] = 1.0
        return J

    t_init = exact_score(h0) + 1e-3
    x0 = np.concatenate([h0, [t_init]])
    bounds = [(0.0, 1.0)] * n + [(0.0, t_init + 0.5)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    try:
        res = minimize(obj, x0, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': tol})
        if res.x is not None and np.isfinite(res.x).all():
            hs = repair_feasible(res.x[:-1])
            if np.isfinite(exact_score(hs)):
                return hs
    except Exception:
        pass
    return None


def quantize_try(h, levels, target):
    """Try quantizing to discrete levels and repair."""
    n = len(h)
    hq = np.zeros_like(h)
    for i in range(n):
        hq[i] = min(levels, key=lambda l: abs(h[i] - l))
    hq = repair_feasible(hq)
    return hq
