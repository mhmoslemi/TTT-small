    # G[m,:]·d - S <= -P[m]

The second order terms: for the profile P_m = sum over appropriate indices of h[a]*(1-h[b]) * dx, changing h_i by t and h_j by -t:
- If i appears as h: contributes t*(1-h[b])*dx - t²*dx (the -t² from h_i*(1-h_i) type terms... wait no)

Actually let me think again. P_m = sum_{(a,b) pairs for lag m} h[a] * (1 - h[b]) * dx

If we change h[i] += t and h[j] -= t:
- For each pair (a,b) where a = i: h[i]*(1-h[b]) changes to (h[i]+t)*(1-h[b]), delta = t*(1-h[b])
- For each pair (a,b) where b = i: h[a]*(1-h[i]) changes to h[a]*(1-h[i]-t), delta = -t*h[a]
- For each pair (a,b) where a = j: h[j]*(1-h[b]) changes to (h[j]-t)*(1-h[b]), delta = -t*(1-h[b])
- For each pair (a,b) where b = j: h[a]*(1-h[j]) changes to h[a]*(1-h[j]+t), delta = t*h[a]

So the total change in P_m is:
ΔP_m = t * [sum_{b: (i,b) in pairs} (1-h[b]) - sum_{a: (a,i) in pairs} h[a] - sum_{b: (j,b) in pairs} (1-h[b]) + sum_{a: (a,j) in pairs} h[a]] * dx

This is exactly t * (G[m,i] - G[m,j]) * ... wait, let me check.

G[m,i] is the derivative of P_m with respect to h[i]. So:
∂P_m/∂h_i = G[m,i]

For a pair move h_i += t, h_j -= t:
ΔP_m ≈ t * (G[m,i] - G[m,j])

This is first-order. The second-order term comes from the fact that P_m is bilinear in h:
P_m = sum h[a]*(1-h[b])*dx = sum (h[a] - h[a]*h[b])*dx

So P_m is quadratic in h. The second-order term for a pair move would be:
- From h[i]*h[i] type: if i appears as both a and b in some pair, we get -t²*dx
- From h[i]*h[j] type: if (i,b) and (a,j) or (j,b) and (a,i) pairs exist, we get cross terms

import numpy as np
from scipy.optimize import linprog
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx])
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-14:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmin(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def build_gradient_matrix(h, n, dx):
    M = 2 * n - 1
    G = np.zeros((M, n))
    for m in range(M):
        d = m - (n - 1)
        if d == 0:
            G[m, :] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            G[m, d:] += dx * (1.0 - h[0:n-d])
            G[m, 0:n-d] += -dx * h[d:n]
        else:
            dp = -d
            G[m, 0:n-dp] += dx * (1.0 - h[dp:n])
            G[m, dp:n] += -dx * h[0:n-dp]
    return G

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    # Area-preserving interpolation
    xs = np.linspace(0, 2, n + 1)
    h_interp = np.zeros(new_n)
    for k in range(new_n):
        lo_k = k * 2.0 / new_n
        hi_k = (k + 1) * 2.0 / new_n
        total_mass = 0.0
        total_width = 0.0
        for j in range(n):
            lo_j = j * 2.0 / n
            hi_j = (j + 1) * 2.0 / n
            ov = min(hi_k, hi_j) - max(lo_k, lo_j)
            if ov > 0:
                total_mass += h[j] * ov
                total_width += ov
        h_interp[k] = total_mass / total_width if total_width > 1e-15 else 0.5
    h_interp = np.clip(h_interp, 0.0, 1.0)
    return repair_feasible(h_interp)

    return np.clip((h + h[::-1]) / 2.0, 0.0, 1.0)

def lp_minimax_direction(h, D=0.5):
    """Solve the linearized minimax LP to find best descent direction."""
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    P = compute_profile(h)
    c = float(np.max(P))
    G = build_gradient_matrix(h, n, dx)
    
    # LP: min S s.t. G@m d - S <= -P[m], sum(d)=0, |d_i|<=D
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    A_ub = np.zeros((M, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -P
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    bounds = [(-D, D)] * n + [(-c, c + 1.0)]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x is not None:
            d = res.x[:n].copy()
            S_star = float(res.x[n])
            # Remove mean to ensure exact mass preservation
            d = d - d.mean()
            return d, S_star, c
    except Exception:
        pass
    return None, None, c

def line_search(h, direction, n_samples=30):
    """Line search along a direction with mass preservation."""
    n = len(h)
    direction = direction - direction.mean()
    dnorm = np.linalg.norm(direction)
    if dnorm < 1e-14:
        return h.copy(), exact_score(h)
    direction /= dnorm  # normalize
    
    # Safe bounds
    pos_mask = direction > 1e-10
    neg_mask = direction < -1e-10
    lam_max = 2.0
    if np.any(pos_mask):
        lam_max = min(lam_max, np.min((1.0 - h[pos_mask]) / direction[pos_mask]))
    if np.any(neg_mask):
        lam_max = min(lam_max, np.min(h[neg_mask] / (-direction[neg_mask])))
    lam_max = max(lam_max, 1e-10)
    lam_min = -lam_max
    
    best_h = h.copy()
    best_c = exact_score(h)
    
    # Initial sampling
    t_values = np.concatenate([np.linspace(0, lam_max, n_samples),
                               np.linspace(lam_min, 0, n_samples)])
    
    for t in t_values:
        h_try = repair_feasible(h + t * direction)
        c_try = exact_score(h_try)
        if c_try < best_c:
            best_c = c_try
            best_h = h_try.copy()
    
    return best_h, best_c

def pair_line_search(h, i, j, n_samples=20):
    """Line search for pair move: h[i] += t, h[j] -= t."""
    # Safe range for t
    t_max = min(1.0 - h[i], h[j])
    t_min = -min(h[i], 1.0 - h[j])
    
    if t_max - t_min < 1e-14:
        return h.copy(), exact_score(h)
    
    best_h = h.copy()
    best_c = exact_score(h)
    
    t_values = np.linspace(t_min, t_max, n_samples)
    for t in t_values:
        h_try = h.copy()
        h_try[i] += t
        h_try[j] -= t
        if h_try[i] < -1e-12 or h_try[i] > 1 + 1e-12 or h_try[j] < -1e-12 or h_try[j] > 1 + 1e-12:
            continue
        h_try = np.clip(h_try, 0.0, 1.0)
        c_try = exact_score(h_try)
        if c_try < best_c:
            best_c = c_try
            best_h = h_try.copy()
    
    return best_h, best_c

def active_set_descent(h, max_iters=200, n_active=12, alpha_init=0.005):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best_c = exact_score(h)
    alpha = alpha_init
    stuck = 0
    for it in range(max_iters):
        profile = compute_profile(h)
        c = float(np.max(profile))
        if c + 1e-14 < best_c:
            best_c = c
            stuck = 0
            alpha = min(alpha * 1.08, 0.025)
        margin = max(0.002, 4.0 * (c - float(np.median(profile))))
        active = np.where(profile >= c - margin)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(profile))])
        if len(active) > n_active:
            top_idx = np.argsort(profile[active])[-n_active:]
            active = active[top_idx]
        G = build_gradient_matrix(h, n, dx)
        G_a = G[active, :]
        w = np.ones(len(active)) / len(active)
        grad = G_a.T @ w
        grad -= grad.mean()
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad /= gnorm
        improved = False
        for a in [alpha, alpha*0.5, alpha*0.25, alpha*0.1, alpha*0.05]:
            h_try = repair_feasible(h - a * grad)
            s = exact_score(h_try)
            if s < best_c - 1e-14:
                h = h_try
                best_c = s
                improved = True
                break
            elif s < c - 1e-14:
                h = h_try
                improved = True
                break
        if not improved:
            alpha *= 0.55
            stuck += 1
            if stuck > 40 or alpha < 1e-9:
                break
    return h, best_c

    alpha = alpha_init

def lp_descent(h, max_iters=100, D_init=0.3):
    """LP-based minimax descent."""
    n = len(h)

            if stuck > 5:

def structured_pair_pass(h, profile, n_candidates=20, n_samples=15):
    """Try pair moves based on gradient structure."""
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best_c = exact_score(h)
    G = build_gradient_matrix(h, n, dx)
    
    # Active rows
    c = float(np.max(profile))
    margin = max(0.003, 2.0 * (c - float(np.median(profile))))
    active = np.where(profile >= c - margin)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(profile))])
    
    # For each coordinate, compute "impulse" from active rows
    G_a = G[active, :]
    w = np.maximum(0, profile[active] - (c - margin))
    if w.sum() < 1e-12:
        w = np.ones(len(active))
    w = w / w.sum()
    
    impulses = (G_a.T @ w) - (G_a.T @ w).mean()
    
    # Best pair: max impulse paired with min impulse
    order = np.argsort(impulses)
    pairs = []
    for k in range(min(n_candidates, n//2)):
        i = order[k]  # most negative
        j = order[n-1-k]  # most positive
        if i != j:
            pairs.append((j, i))  # move mass from i (low) to j (high)? 
            # Actually: h[j] += t, h[i] -= t means moving from i to j
            # We want to decrease the profile, so move from high-impulse to low
            # impulses[j] > 0 means increasing h[j] increases profile
            # So we want to decrease h[j] and increase h[i]
            pairs.append((i, j))  # h[i] += t, h[j] -= t
    
    # Try unique pairs
    seen = set()
    for (i, j) in pairs:
        key = (min(i,j), max(i,j))
        if key in seen:
            continue
        seen.add(key)
        h_try, c_try = pair_line_search(h, i, j, n_samples=n_samples)
        if c_try < best_c - 1e-13:
            h = h_try
            best_c = c_try
    
    return h, best_c

    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx])
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-14:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmin(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def build_gradient_matrix(h, n, dx):
    M = 2 * n - 1
    G = np.zeros((M, n))
    for m in range(M):
        d = m - (n - 1)
        if d == 0:
            G[m, :] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            G[m, d:] += dx * (1.0 - h[0:n-d])
            G[m, 0:n-d] += -dx * h[d:n]
        else:
            dp = -d
            G[m, 0:n-dp] += dx * (1.0 - h[dp:n])
            G[m, dp:n] += -dx * h[0:n-dp]
    return G

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    h_interp = np.zeros(new_n)
    for k in range(new_n):
        lo_k = k * 2.0 / new_n
        hi_k = (k + 1) * 2.0 / new_n
        total_mass = 0.0
        total_width = 0.0
        for j in range(n):
            lo_j = j * 2.0 / n
            hi_j = (j + 1) * 2.0 / n
            ov = min(hi_k, hi_j) - max(lo_k, lo_j)
            if ov > 0:
                total_mass += h[j] * ov
                total_width += ov
        h_interp[k] = total_mass / total_width if total_width > 1e-15 else 0.5
    h_interp = np.clip(h_interp, 0.0, 1.0)
    return repair_feasible(h_interp)

def symmetrize(h):
    return np.clip((h + h[::-1]) / 2.0, 0.0, 1.0)

def antisymmetrize(h):
    n = len(h)
    h_out = np.array([(h[i] + (1.0 - h[n-1-i])) / 2.0 for i in range(n)])
    return repair_feasible(h_out)

def lp_minimax_direction(h, D=0.5):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    P = compute_profile(h)
    c = float(np.max(P))
    G = build_gradient_matrix(h, n, dx)
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_ub = np.zeros((M, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -P
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    bounds = [(-D, D)] * n + [(-c - 0.5, c + 1.0)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x is not None:
            d = res.x[:n].copy()
            S_star = float(res.x[n])
            d = d - d.mean()
            if np.linalg.norm(d) > 1e-12:
                return d, S_star, c
    except Exception:
        pass
    return None, None, c

def line_search(h, direction, n_samples=25):
    n = len(h)
    direction = direction - direction.mean()
    dnorm = np.linalg.norm(direction)
    if dnorm < 1e-14:
        return h.copy(), exact_score(h)
    direction /= dnorm
    pos_mask = direction > 1e-10
    neg_mask = direction < -1e-10
    lam_max = 2.0
    if np.any(pos_mask):
        lam_max = min(lam_max, np.min((1.0 - h[pos_mask]) / direction[pos_mask]))
    if np.any(neg_mask):
        lam_max = min(lam_max, np.min(h[neg_mask] / (-direction[neg_mask])))
    lam_max = max(lam_max, 1e-10)
    lam_min = -lam_max
    best_h = h.copy()
    best_c = exact_score(h)
    t_values = np.concatenate([np.linspace(0, lam_max, n_samples),
                               np.linspace(lam_min, 0, n_samples)])
    for t in t_values:
        h_try = repair_feasible(h + t * direction)
        c_try = exact_score(h_try)
        if c_try < best_c:
            best_c = c_try
            best_h = h_try.copy()
    return best_h, best_c

def pair_line_search(h, i, j, n_samples=15):
    t_max = min(1.0 - h[i], h[j])
    t_min = -min(h[i], 1.0 - h[j])
    if t_max - t_min < 1e-14:
        return h.copy(), exact_score(h)
    best_h = h.copy()
    best_c = exact_score(h)
    t_values = np.linspace(t_min, t_max, n_samples)
    for t in t_values:
        h_try = h.copy()
        h_try[i] = np.clip(h[i] + t, 0.0, 1.0)
        h_try[j] = np.clip(h[j] - t, 0.0, 1.0)
        c_try = exact_score(h_try)
        if c_try < best_c:
            best_c = c_try
            best_h = h_try.copy()
    return best_h, best_c

def lp_descent(h, max_iters=80, D_init=0.4):
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    D = D_init
    stuck = 0
    for it in range(max_iters):
        d, S_star, c = lp_minimax_direction(h, D=D)
        if d is None:
            break
        if S_star >= c - 1e-9:
            D *= 0.6
            stuck += 1
            if stuck > 8 or D < 0.005:
                break
            continue
        h_try, c_try = line_search(h, d, n_samples=25)
        if c_try < best_c - 1e-13:
            h = h_try
            best_c = c_try
            stuck = 0
            D = min(D * 1.05, 1.5)
        else:
            stuck += 1
            D *= 0.75
            if stuck > 12 or D < 0.005:
                break
    return h, best_c

def active_set_descent(h, max_iters=200, n_active=12, alpha_init=0.005):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best_c = exact_score(h)
    alpha = alpha_init
    stuck = 0
    for it in range(max_iters):
        profile = compute_profile(h)
        c = float(np.max(profile))
        if c + 1e-14 < best_c:
            best_c = c
            stuck = 0
            alpha = min(alpha * 1.08, 0.025)
        margin = max(0.002, 4.0 * (c - float(np.median(profile))))
        active = np.where(profile >= c - margin)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(profile))])
        if len(active) > n_active:
            top_idx = np.argsort(profile[active])[-n_active:]
            active = active[top_idx]
        G = build_gradient_matrix(h, n, dx)
        G_a = G[active, :]
        w = np.ones(len(active)) / len(active)
        grad = G_a.T @ w
        grad -= grad.mean()
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad /= gnorm
        improved = False
        for a in [alpha, alpha*0.5, alpha*0.25, alpha*0.1, alpha*0.05]:
            h_try = repair_feasible(h - a * grad)
            s = exact_score(h_try)
            if s < best_c - 1e-14:
                h = h_try
                best_c = s
                improved = True
                break
            elif s < c - 1e-14:
                h = h_try
                improved = True
                break
        if not improved:
            alpha *= 0.55
            stuck += 1
            if stuck > 40 or alpha < 1e-9:
                break
    return h, best_c

def lse_descent(h, beta=80.0, max_iters=150, alpha_init=0.004):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best_c = exact_score(h)
    alpha = alpha_init
    stuck = 0
    for it in range(max_iters):
        profile = compute_profile(h)
        scaled = beta * profile
        max_s = np.max(scaled)
        shifted = scaled - max_s
        weights = np.exp(shifted)
        wsum = weights.sum()
        if wsum < 1e-300:
            break
        weights /= wsum
        G = build_gradient_matrix(h, n, dx)
        grad = G.T @ weights
        grad -= grad.mean()
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad /= gnorm
        improved = False
        for a in [alpha, alpha*0.5, alpha*0.25, alpha*0.1]:
            h_try = repair_feasible(h - a * grad)
            s = exact_score(h_try)
            if s < best_c - 1e-14:
                h = h_try
                best_c = s
                improved = True
                alpha = min(alpha * 1.1, 0.02)
                stuck = 0
                break
            elif s < float(np.max(compute_profile(h))) - 1e-14:
                h = h_try
                improved = True
                break
        if not improved:
            alpha *= 0.5
            stuck += 1
            if stuck > 30 or alpha < 1e-9:
                break
    return h, best_c

def structured_pair_pass(h, n_candidates=30, n_samples=20):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best_c = exact_score(h)
    profile = compute_profile(h)
    c = float(np.max(profile))
    margin = max(0.003, 2.5 * (c - float(np.median(profile))))
    active = np.where(profile >= c - margin)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(profile))])
    G = build_gradient_matrix(h, n, dx)
    G_a = G[active, :]
    w = np.maximum(0, profile[active] - (c - margin))
    if w.sum() < 1e-12:
        w = np.ones(len(active))
    w = w / w.sum()
    impulses = (G_a.T @ w)
    impulses = impulses - impulses.mean()
    order = np.argsort(impulses)
    seen = set()
    tried = 0
    for k in range(min(n_candidates, n // 2)):
        i = order[k]
        j = order[n - 1 - k]
        if i == j:
            continue
        key = (min(i, j), max(i, j))
        if key in seen:
            continue
        seen.add(key)
        h_try, c_try = pair_line_search(h, i, j, n_samples=n_samples)
        if c_try < best_c - 1e-13:
            h = h_try
            best_c = c_try
        tried += 1
        if tried >= n_candidates:
            break
    return h, best_c

def random_pair_pass(h, n_pairs=25, rng=None):
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    if rng is None:
        rng = np.random.RandomState(0)
    for _ in range(n_pairs):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        h_try, c_try = pair_line_search(h, i, j, n_samples=12)
        if c_try < best_c - 1e-13:
            h = h_try
            best_c = c_try
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()
    rng = np.random.RandomState(seed)
    np.random.seed(seed)

    def time_left():
        return budget_s - (time.perf_counter() - t_start)

    parent = None
    try:
        parent = np.array(initial_h_values, dtype=np.float64)
    except NameError:
        pass
    except Exception:
        pass

    if parent is None or len(parent) < 4:
        parent = np.full(70, 0.5)

    parent = np.clip(parent, 0.0, 1.0)
    parent = repair_feasible(parent)

    best_h = parent.copy()
    best_c = exact_score(parent)

    # Symmetry transforms
    h_sym = repair_feasible(symmetrize(parent))
    c_sym = exact_score(h_sym)
    if c_sym < best_c - 1e-12:
        best_h = h_sym
        best_c = c_sym

    h_asym = antisymmetrize(parent)
    c_asym = exact_score(h_asym)
    if c_asym < best_c - 1e-12:
        best_h = h_asym
        best_c = c_asym

    # Stage 1: LP-based descent (main new method)
    if time_left() > 120:
        h_lp, c_lp = lp_descent(best_h, max_iters=60, D_init=0.4)
        if c_lp < best_c - 1e-12:
            best_h = h_lp
            best_c = c_lp

    # Stage 2: Active set refinement
    if time_left() > 100:
        h_ase, c_ase = active_set_descent(best_h, max_iters=150, n_active=10)
        if c_ase < best_c - 1e-12:
            best_h = h_ase
            best_c = c_ase

    # Stage 3: LSE refinement
    if time_left() > 80:
        h_lse, c_lse = lse_descent(best_h, beta=60, max_iters=100)
        if c_lse < best_c - 1e-12:
            best_h = h_lse
            best_c = c_lse

    # Stage 4: Structured pair moves
    if time_left() > 60:
        h_sp, c_sp = structured_pair_pass(best_h, n_candidates=35, n_samples=20)
        if c_sp < best_c - 1e-13:
            best_h = h_sp
            best_c = c_sp

    # Stage 5: Resolution exploration
    n_parent = len(best_h)
    if time_left() > 180:
        dims = set()
        for dn in [-8, -6, -4, -2, 2, 4, 6, 8, 10, 12]:
            d = n_parent + dn
            if d % 2 != 0:
                d += 1
            if 32 <= d <= 130:
                dims.add(d)
        dims.discard(n_parent)
        dims_list = sorted(dims, key=lambda x: abs(x - n_parent))

        for new_n in dims_list[:6]:
            if time_left() < 50:
                break
            h_interp = interpolate_to_n(best_h, new_n)
            h_lp, c_lp = lp_descent(h_interp, max_iters=35, D_init=0.35)
            if c_lp < best_c - 1e-12:
                best_h = h_lp
                best_c = c_lp
            h_ase, c_ase = active_set_descent(best_h, max_iters=80, n_active=10)
            if c_ase < best_c - 1e-12:
                best_h = h_ase
                best_c = c_ase
            h_sp, c_sp = structured_pair_pass(best_h, n_candidates=25, n_samples=15)
            if c_sp < best_c - 1e-13:
                best_h = h_sp
                best_c = c_sp

    # Stage 6: Iterative LP + pairs cycle
    for iteration in range(8):
        if time_left() < 60:
            break
        h_lp, c_lp = lp_descent(best_h, max_iters=35, D_init=0.25)
        if c_lp < best_c - 1e-13:
            best_h = h_lp
            best_c = c_lp
        h_sp, c_sp = structured_pair_pass(best_h, n_candidates=30, n_samples=18)
        if c_sp < best_c - 1e-13:
            best_h = h_sp
            best_c = c_sp
        h_rp, c_rp = random_pair_pass(best_h, n_pairs=15, rng=rng)
        if c_rp < best_c - 1e-13:
            best_h = h_rp
            best_c = c_rp
        h_lse, c_lse = lse_descent(best_h, beta=80, max_iters=40, alpha_init=0.003)
        if c_lse < best_c - 1e-13:
            best_h = h_lse
            best_c = c_lse

    # Stage 7: Multi-start perturbation
    if time_left() > 120:
        n_b = len(best_h)
        starts = [best_h.copy()]
        for eps in [0.004, 0.008, 0.015, 0.03, 0.05, 0.08]:
            z = rng.randn(n_b)
            z -= z.mean()
            starts.append(repair_feasible(best_h + eps * z))
        z = rng.randn(n_b)
        z_sym = (z + z[::-1]) / 2.0
        z_sym -= z_sym.mean()
        starts.append(repair_feasible(best_h + 0.012 * z_sym))
        z2 = rng.randn(n_b)
        z_as = antisymmetrize(z2 * 0.3 + 0.5)
        z_as = z_as - z_as.mean()
        starts.append(repair_feasible(best_h + 0.01 * z_as))

        for h0 in starts:
            if time_left() < 35:
                break
            h_lp, c_lp = lp_descent(h0, max_iters=30, D_init=0.25)
            if c_lp < best_c - 1e-12:
                best_h = h_lp
                best_c = c_lp
                h_sp, c_sp = structured_pair_pass(best_h, n_candidates=25, n_samples=15)
                if c_sp < best_c - 1e-13:
                    best_h = h_sp
                    best_c = c_sp
            h_lse, c_lse = lse_descent(h0, beta=100, max_iters=50)
            if c_lse < best_c - 1e-12:
                best_h = h_lse
                best_c = c_lse

    # Stage 8: Fine LP with small steps
    if time_left() > 50:
        h_lp, c_lp = lp_descent(best_h, max_iters=60, D_init=0.12)
        if c_lp < best_c - 1e-13:
            best_h = h_lp
            best_c = c_lp

    # Stage 9: Final structured pairs with more candidates
    if time_left() > 30:
        h_sp, c_sp = structured_pair_pass(best_h, n_candidates=45, n_samples=25)
        if c_sp < best_c - 1e-13:
            best_h = h_sp
            best_c = c_sp

    # Stage 10: Final active set + LSE polish
    if time_left() > 35:
        h_ase, c_ase = active_set_descent(best_h, max_iters=100, n_active=8, alpha_init=0.002)
        if c_ase < best_c - 1e-13:
            best_h = h_ase
            best_c = c_ase
        h_lse, c_lse = lse_descent(best_h, beta=150, max_iters=60, alpha_init=0.002)
        if c_lse < best_c - 1e-13:
            best_h = h_lse
            best_c = c_lse

    # Stage 11: One more LP pass
    if time_left() > 20:
        h_lp, c_lp = lp_descent(best_h, max_iters=40, D_init=0.1)
        if c_lp < best_c - 1e-13:
            best_h = h_lp
            best_c = c_lp

    # Stage 12: Final pair polish
    if time_left() > 15:
        h_sp, c_sp = structured_pair_pass(best_h, n_candidates=30, n_samples=20)
        if c_sp < best_c - 1e-14:
            best_h = h_sp
            best_c = c_sp
        h_rp, c_rp = random_pair_pass(best_h, n_pairs=15, rng=rng)
        if c_rp < best_c - 1e-14:
            best_h = h_rp
            best_c = c_rp

    # Final validation
    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)

    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            final_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 2:
            final_c = float(eval_result[0])
    except Exception:
        pass

    return best_h, final_c, len(best_h)
