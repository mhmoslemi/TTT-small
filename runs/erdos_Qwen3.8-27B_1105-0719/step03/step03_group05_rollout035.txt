def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(profile))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx] + 1e-15)
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx] + 1e-15)
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-14:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmin(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = np.linspace(0, 2, n + 1)
    xt = np.linspace(0, 2, new_n + 1)
    h_interp = np.zeros(new_n)
    for k in range(new_n):
        lo_k, hi_k = xt[k], xt[k+1]
        total_mass = 0.0
        total_width = 0.0
        for j in range(n):
            lo_j, hi_j = xs[j], xs[j+1]
            overlap = min(hi_k, hi_j) - max(lo_k, lo_j)
            if overlap > 0:
                total_mass += h[j] * overlap
                total_width += overlap
        h_interp[k] = total_mass / total_width if total_width > 1e-15 else 0.5
    h_interp = np.clip(h_interp, 0.0, 1.0)
    return repair_feasible(h_interp)

    b = h[j]

    current_c = exact_score(h)

        c = exact_score(h_try)

        lo = max(t_min, best_t - step)

subject to F_k(h) ≤ t for all k
0 ≤ h ≤ 1
sum(h) = n/2

This is a linear program in (h, t) if we expand F_k(h)! Wait, F_k(h) = sum_x h[x](1-h[x+k]) * dx = sum_x h[x]*dx - sum_x h[x]*h[x+k]*dx

The second term h[x]*h[x+k] is quadratic (bilinear), so this is NOT a linear program. It's a quadratically constrained optimization problem.

However, if h is binary (0 or 1), then h[x]*h[x+k] = h[x] AND h[x+k], and the problem becomes:
F_k(h) = (number of 1s in positions 0..n-k-1) - (number of adjacent 1 pairs) * dx

This is related to the Erdős problem! For binary h, F_k is the number of "1 followed by 0" patterns at shift k.

But we're not restricted to binary. The continuous relaxation allows fractional values.

Let me think about another approach: **alternating minimization**. 

Fix all but one coordinate, optimize that coordinate. This is coordinate descent. But for a single coordinate h[i], the objective C(h) = max_k F_k(h) as a function of h[i] is the max of linear functions (since each F_k is linear in h[i] when all other h's are fixed). So it's convex, and we can find the minimum exactly.

Wait, but changing h[i] breaks the sum constraint. So we need to maintain sum(h) = n/2. If we change h[i] by δ, we need to change some other h[j] by -δ. That's exactly the pair move!

So coordinate descent with a pair move is the right approach. And we've already established that the objective along a pair line is convex, so we can find the minimum exactly.

import numpy as np
from scipy.optimize import minimize
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx] + 1e-15)
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx] + 1e-15)
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-14:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmin(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = np.linspace(0, 2.0, n + 1)
    xt = np.linspace(0, 2.0, new_n + 1)
    h_interp = np.zeros(new_n)
    for k in range(new_n):
        total_mass = 0.0
        total_width = 0.0
        for j in range(n):
            overlap = min(xt[k+1], xs[j+1]) - max(xt[k], xs[j])
            if overlap > 0:
                total_mass += h[j] * overlap
                total_width += overlap
        h_interp[k] = total_mass / total_width if total_width > 1e-15 else 0.5
    h_interp = np.clip(h_interp, 0.0, 1.0)
    return repair_feasible(h_interp)

def antisymmetrize(h):
    n = len(h)
    return repair_feasible((h + (1.0 - h[::-1])) / 2.0)

def _score_pair(h, i, j, t):
    """Evaluate score after h[i]+=t, h[j]-=t (in-place, with restoration)"""
    oi, oj = h[i], h[j]
    h[i] = oi + t
    h[j] = oj - t
    c = float(np.max(np.correlate(h, 1.0 - h, mode="full") * (2.0/len(h))))
    h[i] = oi
    h[j] = oj
    return c

def _score_anti_pair(h, i, j, t):
    """Evaluate score after h[i]+=t, h[n-1-i]-=t, h[j]-=t, h[n-1-j]+=t"""
    n = len(h)
    ip, jp = n-1-i, n-1-j
    oi, op, oj, ojp = h[i], h[ip], h[j], h[jp]
    h[i] = oi + t
    h[ip] = op - t
    h[j] = oj - t
    h[jp] = ojp + t
    c = float(np.max(np.correlate(h, 1.0 - h, mode="full") * (2.0/len(h))))
    h[i] = oi
    h[ip] = op
    h[j] = oj
    h[jp] = ojp
    return c

def pair_line_search_single(h, i, j, n_grid=31):
    """Line search for pair (i,j): h[i]+=t, h[j]-=t"""
    a, b = h[i], h[j]
    t_lo = max(-a, b - 1.0) + 1e-12
    t_hi = min(1.0 - a, b) - 1e-12
    if t_hi - t_lo < 1e-11:
        return h.copy()
    
    # Grid search
    ts = np.linspace(t_lo, t_hi, n_grid)
    best_t = 0.0
    best_c = _score_pair(h, i, j, 0.0)
    for t in ts:
        if abs(t) < 1e-13:
            continue
        c = _score_pair(h, i, j, t)
        if c < best_c - 1e-14:
            best_c = c
            best_t = t
    
    if best_t == 0.0:
        return h.copy()
    
    # Golden section refinement
    step = (t_hi - t_lo) / n_grid
    lo = max(t_lo, best_t - 2*step)
    hi = min(t_hi, best_t + 2*step)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for _ in range(25):
        if hi - lo < 1e-11:
            break
        m1 = hi - gr * (hi - lo)
        m2 = lo + gr * (hi - lo)
        c1 = _score_pair(h, i, j, m1)
        c2 = _score_pair(h, i, j, m2)
        if c1 < c2:
            hi = m2
        else:
            lo = m1
    best_t = (lo + hi) / 2.0
    
    h_new = h.copy()
    h_new[i] = a + best_t
    h_new[j] = b - best_t
    # Verify
    if exact_score(h_new) < best_c - 1e-13:
        return h_new
    return h.copy()

def anti_pair_line_search_single(h, i, j, n_grid=31):

        if c < best_c - 1e-14:

    step = (t_hi - t_lo) / n_grid
    lo = max(t_lo, best_t - 2*step)
    hi = min(t_hi, best_t + 2*step)

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(-(1.0 - h))
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, 1.0 - h[idx] + 1e-15)
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx] + 1e-15)
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-14:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmin(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = np.linspace(0.0, 2.0, n + 1)
    xt = np.linspace(0.0, 2.0, new_n + 1)
    h_interp = np.zeros(new_n)
    for k in range(new_n):
        total_mass = 0.0
        total_width = 0.0
        for j in range(n):
            overlap = min(xt[k+1], xs[j+1]) - max(xt[k], xs[j])
            if overlap > 0:
                total_mass += h[j] * overlap
                total_width += overlap
        h_interp[k] = total_mass / total_width if total_width > 1e-15 else 0.5
    return repair_feasible(np.clip(h_interp, 0.0, 1.0))

def _score_pair(h, i, j, t):

def build_gradient_matrix(h, n, dx):
    M = 2 * n - 1
    G = np.zeros((M, n))
    for m in range(M):
        d = m - (n - 1)
        if d == 0:
            G[m, :] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            G[m, :n-d] += dx * (1.0 - h[d:n])
            G[m, d:] += -dx * h[0:n-d]
        else:
            dp = -d
            G[m, :n-dp] += dx * (1.0 - h[dp:n])
            G[m, dp:] += -dx * h[0:n-dp]
    return G

        if wsum < 1e-300:
            break

        G = build_gradient_matrix(h, n, dx)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad /= gnorm
        improved = False

def slsqp_solve(h0, maxiter=200):
    n = len(h0)
    dx = 2.0 / n
    M = 2 * n - 1
    c0 = exact_score(h0)

    def obj(x):
        return x[-1]

    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g

    def eq_fun(x):
        return x[:-1].sum() - n / 2.0

    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g

    def ineq_fun(x):
        h = np.clip(x[:-1], 0, 1)
        t = x[-1]
        profile = np.correlate(h, 1 - h, mode="full") * dx
        return t - profile

    def ineq_jac(x):
        h = np.clip(x[:-1], 0, 1)
        G = build_gradient_matrix(h, n, dx)
        J = np.zeros((M, n + 1))
        J[:, :-1] = -G
        J[:, -1] = 1.0
        return J

    x0 = np.concatenate([h0, [c0 + 1e-3]])
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 0.5)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None and np.isfinite(res.x).all():
            h_new = repair_feasible(res.x[:-1])
            c_new = exact_score(h_new)
            if c_new <= c0 + 1e-9:
                return h_new, c_new
    except Exception:
        pass
    return None, None

            best_h, best_c = h_pd, c_pd

    # Stage 2: Anti-symmetric pair descent
    if time_left() > 150:
        h_ap, c_ap = anti_pair_descent(best_h, max_sweeps=200, n_grid=15, time_fn=time_left)
        if c_ap < best_c:
            best_h, best_c = h_ap, c_ap

    # Stage 3: LSE + active set refinement
    if time_left() > 120:
        h_lse, c_lse = lse_descent(best_h, beta=80, max_iters=100)
        if c_lse < best_c:
            best_h, best_c = h_lse, c_lse
        h_ase, c_ase = active_set_descent(best_h, max_iters=150, n_active=10)
        if c_ase < best_c:
            best_h, best_c = h_ase, c_ase

    # Stage 4: More pair descent after gradient refinement
    if time_left() > 150:
        h_pd, c_pd = pair_descent(best_h, max_sweeps=200, n_grid=15, time_fn=time_left)
        if c_pd < best_c:
            best_h, best_c = h_pd, c_pd
        h_ap, c_ap = anti_pair_descent(best_h, max_sweeps=150, n_grid=18, time_fn=time_left)
        if c_ap < best_c:
            best_h, best_c = h_ap, c_ap

    # Stage 5: Multi-resolution exploration
    if time_left() > 200:
        n_best = len(best_h)
        resol_list = []
        for factor in [0.85, 0.9, 0.95, 1.05, 1.1, 1.2, 1.3]:
            d = int(round(n_best * factor / 2)) * 2
            d = max(24, min(d, 120))
            if d != n_best:
                resol_list.append(d)
        resol_list = sorted(set(resol_list), key=lambda x: abs(x - n_best))

        for new_n in resol_list[:5]:
            if time_left() < 60:
                break
            h_interp = interpolate_to_n(best_h, new_n)
            h_a = antisymmetrize(h_interp)
            h_pd, c_pd = pair_descent(h_a, max_sweeps=150, n_grid=13, time_fn=time_left)
            if c_pd < best_c:
                best_h, best_c = h_pd, c_pd
            h_ase, c_ase = active_set_descent(best_h, max_iters=80, n_active=10)
            if c_ase < best_c:
                best_h, best_c = h_ase, c_ase

    # Stage 6: SLSQP polish
    if time_left() > 80:
        h_sol, c_sol = slsqp_solve(best_h, maxiter=250)
        if h_sol is not None and c_sol < best_c:
            best_h, best_c = h_sol, c_sol
            h_pd, c_pd = pair_descent(best_h, max_sweeps=100, n_grid=15, time_fn=time_left)
            if c_pd < best_c:
                best_h, best_c = h_pd, c_pd

    # Stage 7: Perturbation + re-optimize (escape local optima)
    if time_left() > 150:
        n_b = len(best_h)
        for eps in [0.008, 0.02, 0.04]:
            if time_left() < 50:
                break
            z = rng.randn(n_b)
            z -= z.mean()
            h_pert = repair_feasible(best_h + eps * z)
            h_pd, c_pd = pair_descent(h_pert, max_sweeps=150, n_grid=13, time_fn=time_left)
            if c_pd < best_c:
                best_h, best_c = h_pd, c_pd
            h_ase, c_ase = active_set_descent(best_h, max_iters=100, n_active=10)
            if c_ase < best_c:
                best_h, best_c = h_ase, c_ase

    # Stage 8: Anti-symmetric perturbation
    if time_left() > 100:
        n_b = len(best_h)
        if n_b % 2 == 0:
            z = rng.randn(n_b)
            h_pert = antisymmetrize(best_h + 0.015 * z)
            h_ap, c_ap = anti_pair_descent(h_pert, max_sweeps=150, n_grid=15, time_fn=time_left)
            if c_ap < best_c:
                best_h, best_c = h_ap, c_pd if hasattr(c_ap, '__class__') and False else c_ap
            h_pd, c_pd = pair_descent(best_h, max_sweeps=100, n_grid=13, time_fn=time_left)
            if c_pd < best_c:
                best_h, best_c = h_pd, c_pd

    # Stage 9: Fine pair descent with more grid points
    if time_left() > 80:
        h_pd, c_pd = pair_descent(best_h, max_sweeps=200, n_grid=21, time_fn=time_left)
        if c_pd < best_c:
            best_h, best_c = h_pd, c_pd

    # Stage 10: Final LSE + pair
    if time_left() > 50:
        h_lse, c_lse = lse_descent(best_h, beta=120, max_iters=100)
        if c_lse < best_c:
            best_h, best_c = h_lse, c_lse
        h_pd, c_pd = pair_descent(best_h, max_sweeps=100, n_grid=18, time_fn=time_left)
        if c_pd < best_c:
            best_h, best_c = h_pd, c_pd

    # Stage 11: SLSQP final
    if time_left() > 40:
        h_sol, c_sol = slsqp_solve(best_h, maxiter=300)
        if h_sol is not None and c_sol < best_c:
            best_h, best_c = h_sol, c_sol

    # Stage 12: Final pair polish
    if time_left() > 20:
        h_pd, c_pd = pair_descent(best_h, max_sweeps=200, n_grid=21, time_fn=time_left)
        if c_pd < best_c:
            best_h, best_c = h_pd, c_pd
        h_ap, c_ap = anti_pair_descent(best_h, max_sweeps=100, n_grid=21, time_fn=time_left)
        if c_ap < best_c:
            best_h, best_c = h_ap, c_ap

    # Final validation
    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)

    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            final_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 2:
            final_c = float(eval_result[0])
    except Exception:
        pass

    return best_h, final_c, len(best_h)
