import numpy as np
from scipy.optimize import linprog
import time

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(-(1.0 - h))
        remaining = diff
        for i in range(n):
            idx = order[i]
            cap = 1.0 - h[idx]
            if cap <= 1e-15:
                continue
            add = min(remaining, cap)
            h[idx] += add
            remaining -= add
            if remaining <= 1e-14:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            if h[idx] <= 1e-15:
                continue
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-14:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] = float(np.clip(h[idx] + diff2, 0.0, 1.0))
    return h

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def compute_gradient(h, m):
    """Compute gradient of profile row m w.r.t. h."""
    n = len(h)
    dx = 2.0 / n
    d = m - (n - 1)
    g = np.zeros(n)
    if d == 0:
        g = dx * (1.0 - 2.0 * h)
    elif d > 0:
        # g[i] = dx * ((1-h[i+d]) * (i <= n-1-d) - h[i-d] * (i >= d))
        # First term: i in [0, n-1-d]
        g[:n-d] += dx * (1.0 - h[d:])
        # Second term: i in [d, n-1]
        g[d:] -= dx * h[:n-d]
    else:
        p = -d
        # g[i] = dx * ((1-h[i-p]) * (i >= p) - h[i+p] * (i <= n-1-p))
        # First term: i in [p, n-1]
        g[p:] += dx * (1.0 - h[:n-p])
        # Second term: i in [0, n-1-p]
        g[:n-p] -= dx * h[p:]
    return g

def compute_all_gradients(h):
    """Compute gradient matrix G of shape (2n-1, n)."""
    n = len(h)
    M = 2 * n - 1
    G = np.zeros((M, n))
    for m in range(M):
        G[m, :] = compute_gradient(h, m)
    return G

def verify_gradients(h):
    """Verify gradient formulas against finite differences."""
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    P = compute_profile(h)
    G = compute_all_gradients(h)
    eps = 1e-7
    max_err = 0.0
    for m in range(0, M, max(1, M // 5)):  # Check subset
        for i in range(0, n, max(1, n // 5)):
            h_plus = h.copy()
            h_plus[i] = min(1.0, h_plus[i] + eps)
            h_minus = h.copy()
            h_minus[i] = max(0.0, h_minus[i] - eps)
            P_plus = compute_profile(h_plus)
            P_minus = compute_profile(h_minus)
            fd = (P_plus[m] - P_minus[m]) / (h_plus[i] - h_minus[i])
            err = abs(fd - G[m, i])
            max_err = max(max_err, err)
    return max_err

def interp_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    edges_old = np.linspace(0.0, 2.0, n + 1)
    edges_new = np.linspace(0.0, 2.0, new_n + 1)
    h_new = np.zeros(new_n)
    for k in range(new_n):
        a, b = edges_new[k], edges_new[k + 1]
        total = 0.0
        for j in range(n):
            lo = max(a, edges_old[j])
            hi = min(b, edges_old[j + 1])
            if hi > lo:
                total += h[j] * (hi - lo)
        h_new[k] = total / (b - a)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

def lp_trust_descent(h, max_iters=50, delta_init=0.04, deadline_fn=None):
    """LP-based trust region descent."""
    n = len(h)
    M = 2 * n - 1
    h = h.copy()
    best_c = exact_score(h)
    delta = delta_init
    
    for it in range(max_iters):
        if deadline_fn is not None and deadline_fn() < 5:
            break
        
        P = compute_profile(h)
        c = float(P.max())
        if c < best_c:
            best_c = c
        
        G = compute_all_gradients(h)
        
        # LP: min tau
        # s.t. G[m] @ d - tau <= -P[m] for all m
        #      sum(d) = 0
        #      bounds on d
        
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        
        A_ub = np.zeros((M, n + 1))
        A_ub[:, :n] = G
        A_ub[:, -1] = -1.0
        b_ub = -P
        
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)
        
        lo = np.maximum(-delta, -h)
        hi = np.minimum(delta, 1.0 - h)
        bounds = list(zip(lo, hi)) + [(best_c * 0.5, best_c + 2.0)]
        
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs')
        except Exception:
            break
        
        if res.status != 0 or res.x is None:
            if it > 5:
                delta *= 0.7
                continue
            break
        
        d_vec = res.x[:n]
        tau = res.x[-1]
        
        # Try to accept
        alpha_max = 1.0
        pos = d_vec > 1e-12
        neg = d_vec < -1e-12
        if pos.any():
            alpha_max = min(alpha_max, float(((1.0 - h[pos]) / d_vec[pos]).min()))
        if neg.any():
            alpha_max = min(alpha_max, float((h[neg] / (-d_vec[neg])).min()))
        alpha_max = max(alpha_max, 1e-12)
        
        # Line search: try fractions
        best_alpha = 0.0
        best_val = c
        best_h = h.copy()
        
        for alpha in [alpha_max, 0.75 * alpha_max, 0.5 * alpha_max, 0.25 * alpha_max]:
            h_try = repair_feasible(h + alpha * d_vec)
            val = exact_score(h_try)
            if val < best_val:
                best_val = val
                best_alpha = alpha
                best_h = h_try
        
        if best_alpha < 1e-12 or best_val >= c - 1e-14:
            delta *= 0.6
            if delta < 1e-4:
                break
            continue
        
        h = best_h
        if best_val < best_c:
            best_c = best_val
        delta = min(delta * 1.2, 0.3)
    
    return h, best_c

def active_descent(h, max_iters=300, active_tol=0.003):
    """Active set descent with momentum."""
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    momentum_vec = np.zeros(n)
    momentum = 0.3
    
    for it in range(max_iters):
        P = compute_profile(h)
        c = float(P.max())
        if c < best_c:
            best_c = c
        thresh = c - active_tol
        active = np.where(P >= thresh)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(P))])
        if len(active) > 25:
            top = np.argsort(P[active])[-25:]
            active = active[top]
        
        g = np.zeros(n)
        for m in active:
            g += compute_gradient(h, m)
        g -= g.mean()
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            break
        
        momentum_vec = momentum * momentum_vec + g
        mv_norm = np.linalg.norm(momentum_vec)
        if mv_norm < 1e-15:
            break
        d = -momentum_vec / mv_norm
        
        pos_mask = d > 1e-12
        neg_mask = d < -1e-12
        alpha_max = 1.0
        if pos_mask.any():
            alpha_max = min(alpha_max, float(((1.0 - h[pos_mask]) / d[pos_mask]).min()))
        if neg_mask.any():
            alpha_max = min(alpha_max, float((h[neg_mask] / (-d[neg_mask])).min()))
        alpha_max = max(alpha_max, 1e-10)
        alpha_max = min(alpha_max, 0.5)
        
        best_alpha = 0.0
        best_val = c
        best_h = h
        a_ls, b_ls = 0.0, alpha_max
        for _ in range(20):
            c1 = b_ls - gr * (b_ls - a_ls)
            c2 = a_ls + gr * (b_ls - a_ls)
            h1 = repair_feasible(h + c1 * d)
            h2 = repair_feasible(h + c2 * d)
            f1 = exact_score(h1)
            f2 = exact_score(h2)
            if f1 < best_val:
                best_val = f1
                best_alpha = c1
                best_h = h1.copy()
            if f2 < best_val:
                best_val = f2
                best_alpha = c2
                best_h = h2.copy()
            if f1 < f2:
                b_ls = c2
            else:
                a_ls = c1
            if b_ls - a_ls < 1e-14:
                break
        
        if best_val >= c - 1e-15:
            break
        h = best_h
    
    return h, best_c

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(-(1.0 - h))
        remaining = diff
        for i in range(n):
            idx = order[i]
            cap = 1.0 - h[idx]
            if cap <= 1e-15:
                continue
            add = min(remaining, cap)
            h[idx] += add
            remaining -= add
            if remaining <= 1e-14:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            if h[idx] <= 1e-15:
                continue
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-14:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] = float(np.clip(h[idx] + diff2, 0.0, 1.0))
    return h

    return g

def compute_all_grad(h):
    n = len(h)
    M = 2 * n - 1
    G = np.zeros((M, n))
    for m in range(M):
        G[m, :] = grad_row(h, m, n)
    return G

def interp_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    edges_old = np.linspace(0.0, 2.0, n + 1)
    edges_new = np.linspace(0.0, 2.0, new_n + 1)
    h_new = np.zeros(new_n)
    for k in range(new_n):
        a, b = edges_new[k], edges_new[k + 1]
        total = 0.0
        for j in range(n):
            lo = max(a, edges_old[j])
            hi = min(b, edges_old[j + 1])
            if hi > lo:
                total += h[j] * (hi - lo)
        h_new[k] = total / (b - a)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

def lp_trust_descent(h, max_iters=60, delta_init=0.035, frac_active=1.0, deadline_fn=None):
    n = len(h)
    M = 2 * n - 1
    h = h.copy()
    best_c = exact_score(h)
    delta = delta_init
    stall = 0

    for it in range(max_iters):
        if deadline_fn is not None and deadline_fn() < 5:
            break
        P = compute_profile(h)
        c = float(P.max())
        if c < best_c:
            best_c = c

        if frac_active < 1.0:
            n_active = max(10, int(M * frac_active))
            top_idx = np.argsort(P)[-n_active:]
        else:
            top_idx = np.arange(M)

        G_sub = compute_all_grad(h)[top_idx, :]
        P_sub = P[top_idx]
        M_sub = len(top_idx)

        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        A_ub = np.zeros((M_sub, n + 1))
        A_ub[:, :n] = G_sub
        A_ub[:, -1] = -1.0
        b_ub = -P_sub
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)
        lo = np.maximum(-delta, -h)
        hi = np.minimum(delta, 1.0 - h)
        bounds = list(zip(lo, hi)) + [(0.0, c + 3.0)]

        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
        except Exception:
            stall += 1
            if stall > 5:
                break
            delta *= 0.7
            continue

        if res.status != 0 or res.x is None:
            stall += 1
            if stall > 5:
                break
            delta *= 0.7
            continue

        d_vec = res.x[:n]
        norm_d = np.linalg.norm(d_vec)
        if norm_d < 1e-12:
            break

        pos = d_vec > 1e-12
        neg = d_vec < -1e-12
        alpha_max = 1.0
        if pos.any():
            alpha_max = min(alpha_max, float(((1.0 - h[pos]) / d_vec[pos]).min()))
        if neg.any():
            alpha_max = min(alpha_max, float((h[neg] / (-d_vec[neg])).min()))
        alpha_max = max(alpha_max, 1e-12)

        best_alpha = 0.0
        best_val = c
        best_h = h.copy()

        for frac in [1.0, 0.8, 0.6, 0.4, 0.25, 0.15]:
            alpha = frac * alpha_max
            if alpha < 1e-13:
                continue
            h_try = repair_feasible(h + alpha * d_vec)
            val = exact_score(h_try)
            if val < best_val:
                best_val = val
                best_alpha = alpha
                best_h = h_try

        if best_alpha < 1e-12 or best_val >= c - 1e-14:
            stall += 1
            delta *= 0.65
            if delta < 1e-5:
                break
            continue

        h = best_h
        if best_val < best_c:
            best_c = best_val
        stall = 0
        if best_alpha >= 0.8 * alpha_max:
            delta = min(delta * 1.25, 0.35)
        else:
            delta = min(delta * 1.05, 0.25)

    return h, best_c

def active_descent(h, max_iters=400, active_tol=0.003, deadline_fn=None):
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    momentum_vec = np.zeros(n)
    momentum = 0.35

    for it in range(max_iters):
        if deadline_fn is not None and deadline_fn() < 5:
            break
        P = compute_profile(h)
        c = float(P.max())
        if c < best_c:
            best_c = c
        thresh = c - active_tol
        active = np.where(P >= thresh)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(P))])
        if len(active) > 30:
            top = np.argsort(P[active])[-30:]
            active = active[top]

        g = np.zeros(n)
        for m in active:
            g += grad_row(h, m, n)
        g -= g.mean()
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            break

        momentum_vec = momentum * momentum_vec + g
        mv_norm = np.linalg.norm(momentum_vec)
        if mv_norm < 1e-15:
            break
        d = -momentum_vec / mv_norm

        pos_mask = d > 1e-12
        neg_mask = d < -1e-12
        alpha_max = 1.0
        if pos_mask.any():
            alpha_max = min(alpha_max, float(((1.0 - h[pos_mask]) / d[pos_mask]).min()))
        if neg_mask.any():
            alpha_max = min(alpha_max, float((h[neg_mask] / (-d[neg_mask])).min()))
        alpha_max = max(alpha_max, 1e-10)
        alpha_max = min(alpha_max, 0.5)

        best_alpha = 0.0
        best_val = c
        best_h = h
        a_ls, b_ls = 0.0, alpha_max
        for _ in range(22):
            c1 = b_ls - gr * (b_ls - a_ls)
            c2 = a_ls + gr * (b_ls - a_ls)
            h1 = repair_feasible(h + c1 * d)
            h2 = repair_feasible(h + c2 * d)
            f1 = exact_score(h1)
            f2 = exact_score(h2)
            if f1 < best_val:
                best_val = f1
                best_alpha = c1
                best_h = h1.copy()
            if f2 < best_val:
                best_val = f2
                best_alpha = c2
                best_h = h2.copy()
            if f1 < f2:
                b_ls = c2
            else:
                a_ls = c1
            if b_ls - a_ls < 1e-14:
                break

        if best_val >= c - 1e-15:
            break
        h = best_h

    return h, best_c

def dual_transform(h):
    return repair_feasible(1.0 - h[::-1])

def golden_seed(n, freq=0.618033988749895):
    idx = np.arange(n)
    vals = (idx * freq) % 1.0
    half = n // 2
    top = np.argsort(vals)[-half:]
    h = np.zeros(n)
    h[top] = 1.0
    return repair_feasible(h)

def sine_blend_seed(n, k):
    x = np.linspace(0.0, 2.0 * k * np.pi, n)
    h = 0.5 * (1.0 + np.sin(x))
    return repair_feasible(h)

def smooth_seed(h, width=2):
    n = len(h)
    h_s = np.zeros(n)
    for i in range(n):
        lo = max(0, i - width)
        hi = min(n, i + width + 1)
        h_s[i] = h[lo:hi].mean()
    return repair_feasible(h_s)

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()
    rng = np.random.RandomState(seed)

    def tl():
        return budget_s - (time.perf_counter() - t_start)

    try:
        parent = np.array(initial_h_values, dtype=np.float64)
    except Exception:
        parent = None
    if parent is None or len(parent) < 4:
        parent = np.full(64, 0.5)
    parent = np.clip(parent, 0.0, 1.0)
    parent = repair_feasible(parent)
    n_p = len(parent)

    best_h = parent.copy()
    best_c = exact_score(parent)

    # Phase 1: LP trust descent on parent (all constraints)
    if tl() > 80:
        h1, c1 = lp_trust_descent(best_h, max_iters=50, delta_init=0.04, frac_active=1.0, deadline_fn=tl)
        if c1 < best_c:
            best_h, best_c = h1, c1

    # Phase 2: Active descent
    if tl() > 60:
        h2, c2 = active_descent(best_h, max_iters=350, active_tol=0.003, deadline_fn=tl)
        if c2 < best_c:
            best_h, best_c = h2, c2

    # Phase 3: LP with partial active set (more focused)
    if tl() > 50:
        h3, c3 = lp_trust_descent(best_h, max_iters=40, delta_init=0.025, frac_active=0.6, deadline_fn=tl)
        if c3 < best_c:
            best_h, best_c = h3, c1 if False else h3, c3

    # Phase 4: Active descent fine
    if tl() > 40:
        h4, c4 = active_descent(best_h, max_iters=300, active_tol=0.001, deadline_fn=tl)
        if c4 < best_c:
            best_h, best_c = h4, c4

    # Phase 5: Dimension exploration
    n_best = len(best_h)
    dims = []
    for dn in [-8, -6, -4, -2, 2, 4, 6, 8, 10, 12, 14, 16]:
        d = n_best + dn
        if 36 <= d <= 150:
            dims.append(d)
    dims.sort(key=lambda x: abs(x - n_best))

    for new_n in dims[:8]:
        if tl() < 25:
            break
        h_int = interp_to_n(best_h, new_n)
        h_lp, c_lp = lp_trust_descent(h_int, max_iters=30, delta_init=0.03, frac_active=0.8, deadline_fn=tl)
        h_ad, c_ad = active_descent(h_lp, max_iters=200, active_tol=0.002, deadline_fn=tl)
        if c_ad < best_c:
            best_h, best_c = h_ad, c_ad

    # Phase 6: Dual + crossover exploration
    if tl() > 80:
        n_exp = len(best_h)
        h_dual = dual_transform(best_h)
        c_dual = exact_score(h_dual)

        candidates = []
        if c_dual < best_c:
            candidates.append((c_dual, h_dual))

        # Crossover with smoothed versions
        h_smooth = smooth_seed(best_h, 2)
        c_sm = exact_score(h_smooth)
        candidates.append((c_sm, h_smooth))

        # Blend with golden seed
        h_gold = golden_seed(n_exp)
        h_blend1 = repair_feasible(0.6 * best_h + 0.4 * h_gold)
        candidates.append((exact_score(h_blend1), h_blend1))

        # Sine blend
        h_sin = sine_blend_seed(n_exp, 3)
        h_blend2 = repair_feasible(0.7 * best_h + 0.3 * h_sin)
        candidates.append((exact_score(h_blend2), h_blend2))

        # Dual-blend
        h_blend3 = repair_feasible(0.5 * best_h + 0.5 * h_dual)
        candidates.append((exact_score(h_blend3), h_blend3))

        candidates.sort(key=lambda x: x[0])

        for c_cand, h_cand in candidates[:4]:
            if tl() < 20:
                break
            h_lp, c_lp = lp_trust_descent(h_cand, max_iters=25, delta_init=0.03, frac_active=0.7, deadline_fn=tl)
            h_ad, c_ad = active_descent(h_lp, max_iters=200, active_tol=0.002, deadline_fn=tl)
            if c_ad < best_c:
                best_h, best_c = h_ad, c_ad

    # Phase 7: More dimension search from improved solution
    if tl() > 60:
        n_cur = len(best_h)
        for new_n in [n_cur - 4, n_cur + 4, n_cur + 8, n_cur - 8, n_cur + 12]:
            if tl() < 20:
                break
            if new_n < 36 or new_n > 150:
                continue
            h_int = interp_to_n(best_h, new_n)
            h_lp, c_lp = lp_trust_descent(h_int, max_iters=35, delta_init=0.03, frac_active=0.7, deadline_fn=tl)
            h_ad, c_ad = active_descent(h_lp, max_iters=250, active_tol=0.001, deadline_fn=tl)
            if c_ad < best_c:
                best_h, best_c = h_ad, c_ad

    # Phase 8: Fine LP polish
    if tl() > 40:
        h_lp, c_lp = lp_trust_descent(best_h, max_iters=50, delta_init=0.015, frac_active=0.5, deadline_fn=tl)
        if c_lp < best_c:
            best_h, best_c = h_lp, c_lp

    # Phase 9: Ultra-fine active descent
    if tl() > 25:
        h_ad, c_ad = active_descent(best_h, max_iters=300, active_tol=0.0005, deadline_fn=tl)
        if c_ad < best_c:
            best_h, best_c = h_ad, c_ad

    # Phase 10: Final LP micro-steps
    if tl() > 15:
        h_lp, c_lp = lp_trust_descent(best_h, max_iters=30, delta_init=0.008, frac_active=0.4, deadline_fn=tl)
        if c_lp < best_c:
            best_h, best_c = h_lp, c_lp

    # Phase 11: One more active descent pass
    if tl() > 10:
        h_ad, c_ad = active_descent(best_h, max_iters=150, active_tol=0.0003, deadline_fn=tl)
        if c_ad < best_c:
            best_h, best_c = h_ad, c_ad

    # Phase 12: Try reducing dimension for equivalence
    if tl() > 8:
        n_cur = len(best_h)
        for new_n in [n_cur - 2, n_cur - 4, n_cur - 6]:
            if tl() < 5:
                break
            if new_n < 20:
                continue
            h_int = interp_to_n(best_h, new_n)
            h_ad2, c_ad2 = active_descent(h_int, max_iters=150, active_tol=0.001, deadline_fn=tl)
            if c_ad2 < best_c:
                best_h, best_c = h_ad2, c_ad2

    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)

    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            final_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 2:
            final_c = float(eval_result[0])
    except Exception:
        pass

    return best_h, final_c, len(best_h)
