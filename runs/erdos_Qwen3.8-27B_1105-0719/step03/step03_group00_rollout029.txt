import numpy as np
from scipy.optimize import minimize
import time

def project_feasible(x, target_sum):
    n = len(x)
    x = np.where(np.isfinite(x), x, 0.5)
    if target_sum <= 0:
        return np.zeros(n)
    if target_sum >= n:
        return np.ones(n)
    lo = np.min(x) - abs(target_sum) - 2.0
    hi = np.max(x) + abs(target_sum) + 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    err = target_sum - h.sum()
    if abs(err) > 1e-14:
        if err > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if mask.sum() > 0:
            h[mask] += err / mask.sum()
    h = np.clip(h, 0.0, 1.0)
    # Final check
    final_err = target_sum - h.sum()
    if abs(final_err) > 1e-8:
        # Degenerate fallback
        f = int(np.floor(target_sum))
        frac = target_sum - f
        h2 = np.zeros(n)
        h2[:f] = 1.0
        if frac > 1e-12:
            h2[f] = frac
        h2 = project_feasible(h2, target_sum)
        return h2
    return h

def c5_of_h(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * dx

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def weighted_grad(h, w, dx):
    n = len(h)
    grad = np.zeros(n)
    # d = 0
    w0 = w[n-1]
    grad += w0 * (1.0 - 2.0 * h)
    for d in range(1, n):
        wp = w[n-1+d]
        wn = w[n-1-d]
        s = wp + wn
        if s < 1e-15:
            continue
        n1 = n - d
        grad[0:n1] += wp * (1.0 - h[d:]) - wn * h[d:]
        grad[d:] += -wp * h[0:n1] + wn * (1.0 - h[0:n1])
    return grad * dx

def softmax_c5_grad(h, alpha, dx):
    """Compute gradient of the soft-max surrogate dx * alpha * logsumexp(corr/alpha)"""
    n = len(h)
    corr = get_corr(h)
    scaled = corr / alpha
    m = scaled.max()
    exp_s = np.exp(scaled - m)
    w = exp_s / exp_s.sum()
    grad = weighted_grad(h, w, dx)
    return grad, w, corr

def softmax_descent_step(h, alpha, dx, target_sum, step_size):
    """One step of gradient descent on the soft-max surrogate"""
    n = len(h)
    grad, w, corr = softmax_c5_grad(h, alpha, dx)
    # Remove mean to maintain sum constraint approximately
    grad -= grad.mean()
    h_new = h - step_size * grad
    h_new = project_feasible(h_new, target_sum)
    return h_new

def fast_local_search(h, dx, target_sum, n_iterations=50, rng=None):
    """Exact coordinate and swap-based local search"""
    n = len(h)
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    corr = get_corr(h)
    
    if rng is None:
        rng = np.random.RandomState(0)
    
    for it in range(n_iterations):
        improved = False
        # Try index order
        order = rng.permutation(n)
        for i in order:
            # Try pushing h[i] toward 0 or 1
            for target in [0.0, 1.0]:
                new_h = best_h.copy()
                delta = target - new_h[i]
                if abs(delta) < 1e-10:
                    continue
                # Distribute -delta to other coordinates
                others = np.array([j for j in range(n) if j != i])
                # Find coordinates that can absorb
                can_absorb = (others >= 0) & (others < n)
                # For simplicity, distribute uniformly to non-i coordinates
                # that won't violate bounds
                if delta > 0:
                    # h[i] increases, need to decrease others
                    mask = best_h[others] > 1e-12
                else:
                    # h[i] decreases, need to increase others
                    mask = best_h[others] < 1.0 - 1e-12
                
                if mask.sum() == 0:
                    continue
                
                # Distribute
                idx = others[mask]
                new_h[i] = target
                new_h[idx] -= delta / mask.sum()
                new_h = np.clip(new_h, 0.0, 1.0)
                new_h = project_feasible(new_h, target_sum)
                
                c5_new = c5_of_h(new_h, dx)
                if c5_new < best_c5 - 1e-14:
                    best_h = new_h
                    best_c5 = c5_new
                    improved = True
        
        if not improved:
            break
    
    return best_h, best_c5

def swap_local_search(h, dx, target_sum, n_iter=100, rng=None):
    """Try swapping values between pairs of coordinates"""
    n = len(h)
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    
    if rng is None:
        rng = np.random.RandomState(0)
    
    for it in range(n_iter):
        improved = False
        for _ in range(200):
            i, j = rng.choice(n, 2, replace=False)
            if i == j:
                continue
            # Swap
            new_h = best_h.copy()
            new_h[i], new_h[j] = best_h[j], best_h[i]
            c5_new = c5_of_h(new_h, dx)
            if c5_new < best_c5 - 1e-14:
                best_h = new_h
                best_c5 = c5_new
                improved = True
        if not improved:
            break
    return best_h, best_c5

def transfer_grid(h_src, n_target):
    n_src = len(h_src)
    if n_src == n_target:
        return h_src.copy()
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_tgt = (np.arange(n_target) + 0.5) * (2.0 / n_target)
    return np.interp(x_tgt, x_src, h_src)
