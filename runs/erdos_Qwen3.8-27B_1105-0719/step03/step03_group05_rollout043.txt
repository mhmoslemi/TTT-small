    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx] + 1e-15)
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx] + 1e-15)
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    return h

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = np.linspace(0, 2, n + 1)
    xt = np.linspace(0, 2, new_n + 1)
    h_interp = np.zeros(new_n)
    for k in range(new_n):
        lo_k, hi_k = xt[k], xt[k + 1]
        total_mass = 0.0
        total_width = 0.0
        for j in range(n):
            lo_j, hi_j = xs[j], xs[j + 1]
            overlap = min(hi_k, hi_j) - max(lo_k, lo_j)
            if overlap > 0:
                total_mass += h[j] * overlap
                total_width += overlap
        h_interp[k] = total_mass / total_width if total_width > 1e-15 else 0.5
    h_interp = np.clip(h_interp, 0.0, 1.0)
    return repair_feasible(h_interp)

def antisym_reflect(h):
    n = len(h)
    m = n // 2
    y = np.array([0.5 * (h[j] + 1.0 - h[n - 1 - j]) for j in range(m)])
    return np.clip(y, 0.0, 1.0)

def build_reflect(y, n):
    m = len(y)
    h = np.zeros(n)
    h[:m] = y
    h[m:] = 1.0 - y[::-1]
    if n % 2 == 1:
        h[n // 2] = 0.5
    return h

def antisym_shift(h):
    n = len(h)
    m = n // 2
    y = np.array([0.5 * (h[j] + 1.0 - h[(j + m) % n]) for j in range(m)])
    return np.clip(y, 0.0, 1.0)

def build_shift(y, n):
    m = len(y)
    h = np.zeros(n)
    h[:m] = y
    h[m:] = 1.0 - y
    return h

def coord_descent_1d(y, j, n, build_fn):
    old = y[j]
    n_samples = 13
    points = np.linspace(0.0, 1.0, n_samples)
    best_val = old
    best_score = exact_score(build_fn(y, n))
    for p in points:
        if abs(p - old) < 1e-6:
            continue
        y[j] = p
        s = exact_score(build_fn(y, n))
        if s < best_score:
            best_score = s
            best_val = p
    y[j] = old
    lo = max(0.0, best_val - 0.08)
    hi = min(1.0, best_val + 0.08)
    if hi - lo < 1e-5:
        y[j] = best_val
        final = exact_score(build_fn(y, n))
        return min(best_score, final)
    for _ in range(10):
        mid1 = lo + (hi - lo) / 3.0
        mid2 = hi - (hi - lo) / 3.0
        y[j] = mid1
        s1 = exact_score(build_fn(y, n))
        y[j] = mid2
        s2 = exact_score(build_fn(y, n))
        if s1 < s2:
            hi = mid2
        else:
            lo = mid1
    best_val = (lo + hi) / 2.0
    y[j] = best_val
    final = exact_score(build_fn(y, n))
    return min(best_score, final)

def coord_descent(y, n, build_fn, max_sweeps=40):
    m = len(y)
    y = y.copy()
    best_c = exact_score(build_fn(y, n))
    for sweep in range(max_sweeps):
        improved = False
        for j in range(m):
            new_c = coord_descent_1d(y, j, n, build_fn)
            if new_c < best_c - 1e-10:
                best_c = new_c
                improved = True
        full_c = exact_score(build_fn(y, n))
        if full_c < best_c:
            best_c = full_c
        if not improved and sweep > 3:
            break
    return y, best_c

def active_set_grad(y, n, build_fn, K=8, max_iters=400, lr_init=0.008):
    m = len(y)
    y = y.copy()
    best_c = exact_score(build_fn(y, n))
    lr = lr_init
    eps = 2e-5
    for it in range(max_iters):
        h = build_fn(y, n)
        profile = compute_profile(h)
        top_idx = np.argsort(profile)[-K:]
        grad = np.zeros(m)
        for j in range(m):
            y[j] += eps
            hp = build_fn(y, n)
            pp = compute_profile(hp)
            y[j] -= 2.0 * eps
            hm = build_fn(y, n)
            pm = compute_profile(hm)
            y[j] += eps
            grad[j] = (pp[top_idx].mean() - pm[top_idx].mean()) / (2.0 * eps)
        y_new = np.clip(y - lr * grad, 0.0, 1.0)
        c_new = exact_score(build_fn(y_new, n))
        if c_new < best_c - 1e-11:
            y = y_new
            best_c = c_new
        else:
            lr *= 0.85
            if lr < 1e-8:
                break
        if it > 50 and it % 50 == 0:
            K = min(20, K + 2)
    return y, best_c

def slsqp_reduced(y0, n, build_fn, maxiter=200):
    m = len(y0)
    c0 = exact_score(build_fn(y0, n))

    def obj(x):
        y_t = np.clip(x, 0, 1)
        h_t = build_fn(y_t, n)
        return exact_score(h_t)

    bounds = [(0.0, 1.0)] * m
    try:
        res = minimize(obj, y0.copy(), method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-6})
        if res.x is not None and np.isfinite(res.x).all():
            y_new = np.clip(res.x, 0, 1)
            c_new = exact_score(build_fn(y_new, n))
            if c_new <= c0 + 1e-9:
                return y_new, c_new
    except Exception:
        pass
    return None, None

def full_space_active(h0, K=10, max_iters=200, lr_init=0.004):

1. `antisym_reflect(h)`: For n=2m, y[j] = 0.5*(h[j] + 1 - h[2m-1-j]). This is the projection of h onto the subspace where h[i] + h[2m-1-i] = 1.

2. `build_reflect(y, n)`: h[:m] = y, h[m:] = 1 - y[::-1]. Let me verify: h[n-1-j] should be 1-y[j]. n-1-j = 2m-1-j. For j=0: h[2m-1] = 1-y[0]. In h[m:], the element at position 2m-1 is h[m:][m-1] = (1-y[::-1])[m-1] = 1-y[m-1-m+1] = 1-y[0]. ✓. For j=m-1: h[2m-1-(m-1)] = h[m] = (1-y[::-1])[0] = 1-y[m-1]. ✓.

3. `active_set_grad`: Computes numerical gradient of the mean of top-K profile values w.r.t. each reduced variable, then does gradient descent. The key is that we're in the reduced space, so no feasibility repair is needed (just clipping to [0,1]).

4. `full_space_active`: Same idea but in full space, with mass constraint handled by centering the gradient and using repair_feasible.

- Working in the anti-symmetric subspace as the primary optimization space
- Using coordinate descent with exact 1D search (different from gradient-based methods)
- Using active-set gradient in reduced space (simpler and more direct than the full gradient matrix approach)
