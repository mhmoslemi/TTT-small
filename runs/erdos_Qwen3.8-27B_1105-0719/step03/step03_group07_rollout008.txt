import numpy as np
from scipy.optimize import minimize, linprog
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    prof = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(prof.max())

def profile_h(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        remaining = diff
        for idx in order:
            add = min(remaining, 1.0 - h[idx])
            h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for idx in order:
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    # Final fix
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def build_gradient(h, d, n, dx):
    """Gradient of P_d with respect to h, where d is the lag offset."""
    g = np.zeros(n)
    if d == 0:
        g[:] = dx * (1.0 - 2.0 * h)
    elif d > 0:
        for i in range(n - d):
            g[i] += dx * (1.0 - h[i + d])
            g[i + d] -= dx * h[i]
    else:
        dp = -d
        for i in range(n - dp):
            g[i + dp] += dx * (1.0 - h[i])
            g[i] -= dx * h[i + dp]
    return g

def lp_direction(h, active_lags, n):
    """Find LP direction to decrease all active lags simultaneously."""
    dx = 2.0 / n
    A_eq = np.ones((1, n))
    b_eq = [0.0]
    bounds = [(-1.0, 1.0)] * n
    
    A_ub = None
    b_ub = None
    
    # We want: G v - t * 1 <= 0 for all active lags
    # Which is: G v <= t
    # Variables: [v_0, ..., v_{n-1}, t]
    # Constraints: G v - t <= 0 for each row of G
    #              sum(v) = 0
    #              -1 <= v_i <= 1
    # Objective: minimize t
    
    G = np.array([build_gradient(h, d, n, dx) for d in active_lags])
    # G v - t <= 0  =>  [G, -1] [v; t] <= 0
    A_ub = np.hstack([G, -np.ones((len(G), 1))])
    b_ub = np.zeros(len(G))
    
    # Objective: minimize t (last variable)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds + [(None, None)], method='highs')
        if res.success:
            v = res.x[:n]
            t_val = res.x[-1]
            return v, t_val
    except Exception:
        pass
    return None, None

def binary_sa(h0, n, t_start, budget_s, margin=5, rng_seed=42):
    """Simulated annealing on binary states to find good binary patterns."""
    rng = np.random.RandomState(rng_seed)
    h = np.zeros(n)
    ones = rng.choice(n, n//2, replace=False)
    h[ones] = 1.0
    best_h = h.copy()
    best_c = exact_score(h)
    prof = profile_h(h)
    
    temperature = 0.01
    cool_rate = 0.999
    
    while time.perf_counter() - t_start < budget_s - margin:
        # Pick a random 1 and a random 0
        ones_mask = h == 1.0
        zeros_mask = h == 0.0
        a = rng.choice(np.where(ones_mask)[0])
        b = rng.choice(np.where(zeros_mask)[0])
        
        # Swap
        h[a], h[b] = h[b], h[a]
        new_prof = profile_h(h)
        new_c = new_prof.max()
        
        delta = new_c - best_c
        if delta < 0 or rng.random() < np.exp(-delta / temperature):
            best_h = h.copy()
            best_c = new_c
            prof = new_prof
        else:
            h[a], h[b] = h[b], h[a]  # revert
        
        temperature = max(temperature * cool_rate, 0.001)
        
        # Periodic reset
        if temperature <= 0.001:
            temperature = 0.01
    
    return best_h, best_c

def symmetrize(h):
    """Make h symmetric: h[i] = h[n-1-i]."""
    return (h + h[::-1]) / 2.0

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    return repair_feasible(np.clip(h_new, 0.0, 1.0))

def coordinate_descent(h, max_sweeps=30, t_start=None, budget_s=1100, margin=10):
    """Pair coordinate descent with exact evaluation."""
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    
    for sweep in range(max_sweeps):
        if t_start is not None and time.perf_counter() - t_start > budget_s - margin:
            break
        improved = False
        rng = np.random.RandomState(sweep)
        # Random pair order
        pairs = []
        for _ in range(n // 2):
            a = rng.randint(n)
            b = rng.randint(n)
            if a != b:
                pairs.append((a, b))
        
        for (a, b) in pairs:
            if t_start is not None and time.perf_counter() - t_start > budget_s - margin:
                break
            s = h[a] + h[b]
            lo = max(0.0, s - 1.0)
            hi = min(1.0, s)
            if lo >= hi:
                continue
            # Try several values of h[a] in [lo, hi]
            best_t = h[a]
            best_s = best_c
            for t in np.linspace(lo, hi, 7):
                if abs(t - h[a]) < 1e-10:
                    continue
                h[a] = t
                h[b] = s - t
                sc = exact_score(h)
                if sc < best_s:
                    best_s = sc
                    best_t = t
            if best_t != h[a]:
                h[a] = best_t
                h[b] = s - best_t
                if best_s < best_c:
                    best_c = best_s
                    improved = True
                h = repair_feasible(h)
    
    return h, best_c

import numpy as np
from scipy.optimize import minimize, linprog
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    prof = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(prof.max())

def profile_h(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        remaining = diff
        for idx in order:
            add = min(remaining, 1.0 - h[idx])
            h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for idx in order:
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def build_gradient(h, d, n, dx):
    g = np.zeros(n)
    if d == 0:
        g[:] = dx * (1.0 - 2.0 * h)
    elif d > 0:
        for i in range(n - d):
            g[i] += dx * (1.0 - h[i + d])
            g[i + d] -= dx * h[i]
    else:
        dp = -d
        for i in range(n - dp):
            g[i + dp] += dx * (1.0 - h[i])
            g[i] -= dx * h[i + dp]
    return g

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    return repair_feasible(np.clip(h_new, 0.0, 1.0))

def symmetrize(h):
    return (h + h[::-1]) / 2.0

def lp_minimax_direction(h, active_indices, n, max_active=20):
    """LP to find direction that decreases all active lags.
    active_indices: indices into the profile array (0 to 2n-2)
    Returns: (direction, t_value) or (None, None)
    """
    if len(active_indices) == 0:
        return None, None
    if len(active_indices) > max_active:
        # Take the worst ones
        prof = profile_h(h)
        order = np.argsort(prof[active_indices])[::-1][:max_active]
        active_indices = active_indices[order]
    
    dx = 2.0 / n
    G_list = []
    for m in active_indices:
        d = int(m - (n - 1))
        g = build_gradient(h, d, n, dx)
        # Normalize for numerical stability
        gmax = np.abs(g).max()
        if gmax > 1e-15:
            g = g / gmax
        G_list.append(g)
    G = np.array(G_list)
    
    # LP: min t
    # s.t. G v - t <= 0  (i.e., G v <= t for each row)
    #      1^T v = 0
    #      -1 <= v_i <= 1
    A_ub = np.hstack([G, -np.ones((len(G), 1))])
    b_ub = np.zeros(len(G))
    A_eq = np.ones((1, n))
    b_eq = np.array([0.0])
    c = np.zeros(n + 1)
    c[-1] = 1.0
    bounds = [(-1.0, 1.0)] * n + [(None, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
        if res.success:
            v = res.x[:n]
            # Check that sum(v) ≈ 0
            v = v - v.mean()  # Re-center for numerical safety
            return v, res.x[-1]
    except Exception:
        pass
    return None, None

        n_pairs = min(50, n * n // 4)
        for _ in range(n_pairs):

                break
            a = rng.randint(n)
            b = rng.randint(n)
            if a == b:

            s = h[a] + h[b]
            lo = max(0.0, s - 1.0)
            hi = min(1.0, s)
