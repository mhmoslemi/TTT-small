    best_c5 = c5_value(h0)

    # Phase 1: Projected gradient descent with multiple betas (15% budget)
    t_phase1 = budget_s * 0.12
    betas = [10.0, 30.0, 80.0, 200.0, 500.0, 1000.0]
    for beta in betas:
        if time.time() - t0 > t_phase1:
            break
        h_opt, c5_opt = proj_grad_descent(best_h, dx, target_sum, beta,
                                          max_iters=2000, t0=t0, max_time=t_phase1)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 2: Fast SA rounds (55% budget)
    t_phase2 = budget_s * 0.65
    sa_configs = [
        (0.005, 0.999999, 0.03, seed + 1),
        (0.003, 0.9999995, 0.02, seed + 2),
        (0.004, 0.999999, 0.04, seed + 3),
        (0.002, 0.999999, 0.025, seed + 4),
        (0.006, 0.999998, 0.05, seed + 5),
    ]
    for idx, (ts, cool, db, s) in enumerate(sa_configs):
        if time.time() - t0 > t_phase2:
            break
        remaining = t_phase2 - (time.time() - t0)
        if remaining < 10:
            break
        h_opt, c5_opt = sa_fast2(best_h, dx, target_sum, t0,
                                 time.time() - t0 + remaining, s,
                                 t_start=ts, cooling=cool, delta_base=db)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 3: Gradient-guided SA (15% budget)
    t_phase3 = budget_s * 0.82
    for beta in [30.0, 100.0, 300.0]:
        if time.time() - t0 > t_phase3:
            break
        remaining = t_phase3 - (time.time() - t0)
        if remaining < 10:
            break
        h_opt, c5_opt = sa_gradient_guided(best_h, dx, target_sum, t0,
                                           time.time() - t0 + remaining,
                                           seed + 10 + int(beta), beta=beta,
                                           t_start=0.002, cooling=0.999999)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 4: More gradient descent polish (8% budget)
    t_phase4 = budget_s * 0.92
    for beta in [100.0, 300.0, 1000.0, 3000.0]:
        if time.time() - t0 > t_phase4:
            break
        h_opt, c5_opt = proj_grad_descent(best_h, dx, target_sum, beta,
                                          max_iters=3000, t0=t0, max_time=t_phase4)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 5: Final low-T SA polish (8% budget)
    if time.time() - t0 < budget_s * 0.96:
        remaining = budget_s * 0.96 - (time.time() - t0)
        if remaining > 5:
            h_opt, c5_opt = sa_fast2(best_h, dx, target_sum, t0,
                                     time.time() - t0 + remaining,
                                     seed + 100, t_start=0.0005,
                                     cooling=0.999999, delta_base=0.01)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    n_final = len(best_h)
    ts_final = n_final / 2.0
    best_h = project_feasible(best_h, ts_final)
    best_c5 = c5_value(best_h)

    return best_h, best_c5, n_final
