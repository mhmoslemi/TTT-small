def line_search(h, v, n, coarse_steps=20):
    """Minimize f(α) = exact_score(h + αv) over feasible α."""
    lo, hi = alpha_bounds(h, v)
    if lo >= hi:
        return 0.0, exact_score(h)
    
    # Coarse grid
    alphas = np.linspace(lo, hi, coarse_steps)
    scores = np.array([exact_score(np.clip(h + a*v, 0, 1)) for a in alphas])
    best_idx = np.argmin(scores)
    
    # Golden section around best
    gl = alphas[max(0, best_idx-1)]
    gh = alphas[min(len(alphas)-1, best_idx+1)]
    
    gr = (np.sqrt(5) - 1) / 2
    c1 = gh - gr * (gh - gl)
    c2 = gl + gr * (gh - gl)
    f1 = exact_score(np.clip(h + c1*v, 0, 1))
    f2 = exact_score(np.clip(h + c2*v, 0, 1))
    
    for _ in range(30):
        if f1 < f2:
            gh = c2; c2 = c1; f2 = f1
            c1 = gh - gr * (gh - gl)
            f1 = exact_score(np.clip(h + c1*v, 0, 1))
        else:
            gl = c1; c1 = c2; f1 = f2
            c2 = gl + gr * (gh - gl)
            f2 = exact_score(np.clip(h + c2*v, 0, 1))
    
    best_alpha = (gl + gh) / 2
    best_score = exact_score(np.clip(h + best_alpha*v, 0, 1))
    return best_alpha, best_score

    """Find best α for direction v, minimizing exact_score(h + αv)."""
    lo, hi = alpha_bounds(h, v)
    if hi - lo < 1e-12:
        c0 = exact_score(h)
        return 0.0, c0
    
    # Coarse grid
    alphas = np.linspace(lo, hi, n_coarse)
    best_score = exact_score(h)
    best_alpha = 0.0
    scores = np.zeros(n_coarse)
    for i, a in enumerate(alphas):
        h_try = h + a * v
        if np.any(h_try < -1e-12) or np.any(h_try > 1 + 1e-12):
            scores[i] = 999.0
        else:
            h_try = np.clip(h_try, 0, 1)
            scores[i] = exact_score(h_try)
            if scores[i] < best_score:
                best_score = scores[i]
                best_alpha = a
    
    if best_alpha == 0.0:
        return 0.0, best_score
    
    # Golden section refinement around best
    lo_g = alphas[max(0, np.argmin(scores) - 1)]
    hi_g = alphas[min(n_coarse-1, np.argmin(scores) + 1)]
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi_g - gr * (hi_g - lo_g)
    c2 = lo_g + gr * (hi_g - lo_g)
    f1 = exact_score(np.clip(h + c1*v, 0, 1))
    f2 = exact_score(np.clip(h + c2*v, 0, 1))
    
    for _ in range(n_golden):
        if hi_g - lo_g < 1e-10:
            break
        if f1 < f2:
            hi_g = c2
            c2 = c1
            f2 = f1
            c1 = hi_g - gr * (hi_g - lo_g)
            f1 = exact_score(np.clip(h + c1*v, 0, 1))
        else:
            lo_g = c1
            c1 = c2
            f1 = f2
            c2 = lo_g + gr * (hi_g - lo_g)
            f2 = exact_score(np.clip(h + c2*v, 0, 1))
    
    best_a = (lo_g + hi_g) / 2.0
    best_s = exact_score(np.clip(h + best_a * v, 0, 1))
    
    if best_s < best_score:
        best_score = best_s
        best_alpha = best_a
    
    return best_alpha, best_score

def soft_max_gradient(h, n, temperature=0.01):
    """Compute gradient direction using soft-max weights on profile."""
    profile = compute_profile(h)
    # Soft-max weights
    shifted = profile - profile.max()
    weights = np.exp(shifted / temperature)
    weights /= weights.sum()
    
    # Compute gradient
    dx = 2.0 / n
    M = 2 * n - 1
    center = n - 1
    grad = np.zeros(n)
    
    for k in range(M):
        if weights[k] < 1e-10:
            continue
        s = k - center
        w = weights[k] * dx
        if s >= 0:
            overlap = n - s
            a = h[:overlap]
            b = h[s:]
            grad[:overlap] += w * (1.0 - b)
            grad[s:s+overlap] += w * (-a)
        else:
            t = -s
            overlap = n - t
            a = h[t:]
            b = h[:overlap]
            grad[t:t+overlap] += w * (1.0 - b)
            grad[:overlap] += w * (-a)
    
    # Project to sum=0
    grad -= grad.mean()
    return grad

def coordinate_descent_step(h, n, max_pairs=50):
    """Try coordinate pair moves: transfer from i to j."""
    best_h = h.copy()
    best_c = exact_score(h)

    profile = compute_profile(h)
    active_lags = np.where(profile >= profile.max() - 0.01)[0]

    n_high = np.where(h > 0.1)[0]

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

    return np.correlate(h, 1.0 - h, mode='full') * dx

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx])
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:

        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)

    if not np.isfinite(lo):
        lo = -1.0
    margin = 1e-9 * max(1.0, abs(lo), abs(hi))
    lo += margin
    hi -= margin
    if lo >= hi:
        return 0.0, 0.0
    return lo, hi

def line_search_fast(h, v, n, n_coarse=20, n_golden=30):
    lo, hi = alpha_bounds(h, v)
    if hi - lo < 1e-13:
        return 0.0, exact_score(h)
    A, B, C = precompute_line(h, v, n)

    def f(a):
        return float(np.max(A + a * B - a * a * C))

    alphas = np.linspace(lo, hi, n_coarse)
    scores = np.array([f(a) for a in alphas])
    best_idx = int(np.argmin(scores))
    best_score = float(scores[best_idx])
    best_alpha = float(alphas[best_idx])

    gl = float(alphas[max(0, best_idx - 1)])
    gh = float(alphas[min(n_coarse - 1, best_idx + 1)])
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = gh - gr * (gh - gl)
    c2 = gl + gr * (gh - gl)
    f1, f2 = f(c1), f(c2)
    for _ in range(n_golden):
        if gh - gl < 1e-11:
            break
        if f1 < f2:
            gh, c2, f2 = c2, c1, f1
            c1 = gh - gr * (gh - gl)
            f1 = f(c1)
        else:
            gl, c1, f1 = c1, c2, f2
            c2 = gl + gr * (gh - gl)
            f2 = f(c2)
    mid = (gl + gh) / 2.0
    fm = f(mid)
    if fm < best_score:
        best_score, best_alpha = fm, mid
    return best_alpha, best_score

def soft_max_gradient(h, n, temperature=0.01):
    profile = compute_profile(h)
    shifted = profile - profile.max()
    weights = np.exp(shifted / temperature)
    wsum = weights.sum()
    if wsum < 1e-30:
        return np.zeros(n)
    weights /= wsum
    dx = 2.0 / n
    M = 2 * n - 1
    center = n - 1
    grad = np.zeros(n)
    for k in range(M):
        if weights[k] < 1e-12:
            continue
        s = k - center
        w = weights[k] * dx
        if s >= 0:
            ov = n - s
            a = h[:ov]
            b = h[s:]
            grad[:ov] += w * (1.0 - b)
            grad[s:s + ov] += w * (-a)
        else:
            t = -s
            ov = n - t
            a = h[t:]
            b = h[:ov]
            grad[t:t + ov] += w * (1.0 - b)
            grad[:ov] += w * (-a)
    grad -= grad.mean()
    return grad

def weighted_gradient_top(h, n, top_k=5):
    profile = compute_profile(h)
    idx = np.argsort(profile)[::-1][:top_k]
    dx = 2.0 / n
    M = 2 * n - 1
    center = n - 1
    grad = np.zeros(n)
    for k in idx:
        s = k - center
        w = dx
        if s >= 0:
            ov = n - s
            a = h[:ov]
            b = h[s:]
            grad[:ov] += w * (1.0 - b)
            grad[s:s + ov] += w * (-a)
        else:
            t = -s
            ov = n - t
            a = h[t:]
            b = h[:ov]
            grad[t:t + ov] += w * (1.0 - b)
            grad[:ov] += w * (-a)
    grad -= grad.mean()
    return grad

def binary_local_search(h_init, n, budget_check, max_iters=5000):
    h_bin = (h_init > 0.5).astype(np.float64)
    n_ones = int(h_bin.sum())
    target = n // 2
    if n_ones != target:
        if n_ones > target:
            order = np.argsort(-h_bin)
            for idx in order:
                if n_ones == target:
                    break
                h_bin[idx] = 0
                n_ones -= 1
        else:
            order = np.argsort(h_bin)
            for idx in order:
                if n_ones == target:
                    break
                h_bin[idx] = 1
                n_ones += 1
    best_c = exact_score(h_bin)
    best_h = h_bin.copy()
    for it in range(max_iters):
        if budget_check() < 5:
            break
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j or h_bin[i] == h_bin[j]:
            continue
        h_bin[i], h_bin[j] = h_bin[j], h_bin[i]
        c = exact_score(h_bin)
        if c < best_c or (c - best_c) < 0.001 * np.random.rand():
            best_c, best_h = c, h_bin.copy()
        else:
            h_bin[i], h_bin[j] = h_bin[j], h_bin[i]
    return best_h, best_c

def interp_to_dim(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    fx = np.concatenate([[0.0], xs, [2.0]])
    fh = np.concatenate([[h[0]], h, [h[-1]]])
    hn = np.interp(xt, fx, fh)
    return repair_feasible(np.clip(hn, 0, 1))

def optimize_at_dim(h0, n, budget_check, max_iters=300):
    h = h0.copy()
    best_c = exact_score(h)
    temps = [0.003, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.5]
    stagnant = 0
    for it in range(max_iters):
        if budget_check() < 10:
            break
        improved = False
        for T in temps:
            if budget_check() < 5:
                break
            v = soft_max_gradient(h, n, temperature=T)
            vn = np.linalg.norm(v)
            if vn < 1e-14:
                continue
            for scale in [1.0, 0.5, 0.2, 0.1, 0.05, 0.02]:
                v_s = v * (scale / vn)
                alpha, score = line_search_fast(h, v_s, n)
                if score < best_c - 1e-13:
                    h_new = np.clip(h + alpha * v_s, 0, 1)
                    h_new = repair_feasible(h_new)
                    sc = exact_score(h_new)
                    if sc < best_c - 1e-13:
                        h, best_c = h_new, sc
                        stagnant = 0
                        improved = True
                        break
            if improved:
                break
        if not improved:
            v = weighted_gradient_top(h, n, top_k=3)
            vn = np.linalg.norm(v)
            if vn > 1e-14:
                for scale in [0.1, 0.05, 0.02, 0.01]:
                    v_s = v * (scale / vn)
                    alpha, score = line_search_fast(h, v_s, n)
                    if score < best_c - 1e-13:
                        h_new = repair_feasible(np.clip(h + alpha * v_s, 0, 1))
                        sc = exact_score(h_new)
                        if sc < best_c - 1e-13:
                            h, best_c = h_new, sc
                            stagnant = 0
                            improved = True
                            break
        if not improved:
            stagnant += 1
            if stagnant > 30:
                break
    return h, best_c

def coordinate_refine(h, n, budget_check, max_steps=200):
    best_h = h.copy()
    best_c = exact_score(h)
    for it in range(max_steps):
        if budget_check() < 5:
            break
        improved = False
        for _ in range(20):
            if budget_check() < 3:
                break
            i = np.random.randint(n)
            j = np.random.randint(n)
            if i == j or abs(best_h[i] - best_h[j]) < 1e-6:
                continue
            v = np.zeros(n)
            if best_h[i] > best_h[j]:
                v[i] = -1.0
                v[j] = 1.0
            else:
                v[i] = 1.0
                v[j] = -1.0
            alpha, score = line_search_fast(best_h, v, n, n_coarse=10, n_golden=20)
            if alpha != 0 and score < best_c - 1e-13:
                best_h = repair_feasible(np.clip(best_h + alpha * v, 0, 1))
                best_c = exact_score(best_h)
                improved = True
        if not improved:
            break
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()
    rng = np.random.RandomState(seed)

    def tl():
        return budget_s - (time.perf_counter() - t_start)

    try:
        parent = np.array(initial_h_values, dtype=np.float64).copy()
    except (NameError, Exception):
        parent = np.full(70, 0.5)

    parent = repair_feasible(np.clip(parent, 0.0, 1.0))
    best_h = parent.copy()
    best_c = exact_score(parent)
    n = len(best_h)

    h_sym = repair_feasible((best_h + best_h[::-1]) / 2.0)
    c_sym = exact_score(h_sym)
    if c_sym < best_c:
        best_h, best_c = h_sym, c_sym
        n = len(best_h)

    # Phase 1: Main optimization loop at parent dimension
    if tl() > 60:
        bh, bc = optimize_at_dim(best_h, n, tl, max_iters=300)
        if bc < best_c:
            best_h, best_c, n = bh, bc, len(bh)

    # Phase 2: Coordinate refinement
    if tl() > 40:
        bh, bc = coordinate_refine(best_h, n, tl, max_steps=100)
        if bc < best_c:
            best_h, best_c, n = bh, bc, len(bh)

    # Phase 3: Binary local search
    if tl() > 80:
        hb, cb = binary_local_search(best_h, n, tl, max_iters=3000)
        if cb < best_c:
            best_h, best_c, n = hb, cb, n
        # Also try from the continuous solution's binary projection
        hp = (best_h > 0.5).astype(np.float64)
        if hp.sum() != n / 2:
            hp = repair_feasible(hp)
        if tl() > 40:
            hb2, cb2 = binary_local_search(hp, n, tl, max_iters=2000)
            if cb2 < best_c:
                best_h, best_c, n = hb2, cb2, n
            # Refine the binary solution continuously
            if cb2 < best_c + 0.01 and tl() > 30:
                bh, bc = optimize_at_dim(hb2, n, tl, max_iters=200)
                if bc < best_c:
                    best_h, best_c, n = bh, bc, len(bh)

    # Phase 4: Dimension exploration
    if tl() > 120:
        n0 = len(best_h)
        dims_to_try = []
        for d in range(30, 130, 2):
            if d != n0 and d >= 30:
                dims_to_try.append(d)
        dims_to_try.sort(key=lambda d: abs(d - n0))
        for nd in dims_to_try[:8]:
            if tl() < 40:
                break
            hi = interp_to_dim(best_h, nd)
            ci = exact_score(hi)
            if ci < best_c + 0.03:
                bh, bc = optimize_at_dim(hi, nd, tl, max_iters=150)
                if bc < best_c:
                    best_h, best_c, n = bh, bc, nd
                    if tl() > 30:
                        bh2, bc2 = coordinate_refine(best_h, n, tl, max_steps=80)
                        if bc2 < best_c:
                            best_h, best_c, n = bh2, bc2, n

    # Phase 5: Random restarts from perturbed versions
    if tl() > 80:
        for attempt in range(8):
            if tl() < 20:
                break
            eps = 0.03 + 0.05 * attempt
            noise = rng.randn(n) * eps
            noise -= noise.mean()
            h_try = repair_feasible(best_h + noise)
            c_try = exact_score(h_try)
            if c_try < best_c + 0.02:
                bh, bc = optimize_at_dim(h_try, n, tl, max_iters=150)
                if bc < best_c:
                    best_h, best_c = bh, bc
                    n = len(best_h)
                if tl() > 20:
                    bh, bc = coordinate_refine(best_h, n, tl, max_steps=60)
                    if bc < best_c:
                        best_h, best_c, n = bh, bc, len(bh)

    # Phase 6: Final coordinate polish
    if tl() > 20:
        bh, bc = coordinate_refine(best_h, len(best_h), tl, max_steps=150)
        if bc < best_c:
            best_h, best_c = bh, bc
            n = len(best_h)

    # Phase 7: Very fine gradient steps
    if tl() > 15:
        h_f = best_h.copy()
        c_f = best_c
        n_f = len(h_f)
        for _ in range(50):
            if tl() < 5:
                break
            imp = False
            for T in [0.001, 0.002, 0.005]:
                v = soft_max_gradient(h_f, n_f, temperature=T)
                vn = np.linalg.norm(v)
                if vn < 1e-14:
                    continue
                for scale in [0.01, 0.005, 0.002, 0.001]:
                    v_s = v * (scale / vn)
                    alpha, score = line_search_fast(h_f, v_s, n_f, n_coarse=10, n_golden=20)
                    if score < c_f - 1e-14:
                        h_new = repair_feasible(np.clip(h_f + alpha * v_s, 0, 1))
                        sc = exact_score(h_new)
                        if sc < c_f - 1e-14:
                            h_f, c_f, imp = h_new, sc, True
                        break
                if imp:
                    break
            if not imp:
                break
        if c_f < best_c:
            best_h, best_c = h_f, c_f
            n = len(best_h)

    # Final validation
    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)

    try:
        er = evaluate_erdos_solution(best_h)
        if isinstance(er, (float, int)):
            final_c = min(final_c, float(er))
        elif isinstance(er, tuple) and len(er) >= 2:
            final_c = min(final_c, float(er[0]))
    except Exception:
        pass

    return best_h, final_c, len(best_h)
