        corrs = corr_arr(h)
        c_max = corrs.max()
        
        # Adaptive active set: all constraints within threshold of max
        threshold = c_max - threshold_frac * c_max
        active_mask = corrs >= threshold
        active_k = np.where(active_mask)[0]
        
        # Cap the active set size
        if len(active_k) > n_total:
            active_k = np.argsort(corrs)[-n_active_start:]
        elif len(active_k) < 5:
            active_k = np.argsort(corrs)[-max(5, n_active_start):]
        
        h_new, c5_new, improved = active_set_step(h, dx, target_sum, 
                                                   active_k, alpha, n_total)
        
        if improved and c5_new < best_c5 - 1e-14:
            h = h_new
            best_c5 = c5_new
            best_h = h_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            alpha *= alpha_decay
            if no_improve % 20 == 0:
                # Adjust active set size
                n_active_start = min(n_total, n_active_start + 10)
                threshold_frac = max(0.003, threshold_frac * 0.9)
            if no_improve > max_no_improve:
                break
    
    return best_h, best_c5

def pair_exchange(h, dx, target_sum, n, t0, max_time, max_pairs=200):
    """Try pairwise exchanges with exact 1D optimization."""
    best_h = h.copy()
    best_c5 = c5_of(h, dx)
    n_total = 2 * n - 1
    
    pairs_done = 0
    no_improve = 0
    
    while time.time() - t0 < max_time and no_improve < 30:
        # Pick a random pair
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j:
            continue
        if h[i] <= 0 or h[j] <= 0:
            continue
        
        # Transfer t from j to i
        t_lo = max(-h[i] * 0.99, -0.49)
        t_hi = min((1.0 - h[i]) * 0.99, h[j] * 0.99)
        if t_lo >= t_hi:
            continue
        
        def obj_1d(t):
            h_new = best_h.copy()
            h_new[i] += t
            h_new[j] -= t
            h_new = np.clip(h_new, 0, 1)
            # Fix sum
            diff = target_sum - h_new.sum()
            if abs(diff) > 1e-10:
                h_new[i] += diff
                h_new = project_feasible(h_new, target_sum)
            return c5_of(h_new, dx)
        
        res = minimize_scalar(obj_1d, bounds=(t_lo, t_hi), method='bounded',
                              options={'xatol': 1e-8})
        
        if res.fun < best_c5 - 1e-12:
            best_h[i] += res.x
            best_h[j] -= res.x
            best_h = project_feasible(best_h, target_sum)
            best_c5 = c5_of(best_h, dx)
            no_improve = 0
        else:
            no_improve += 1
        
        pairs_done += 1
    
    return best_h, best_c5

def spectral_seed(n, dx, freqs, amplitudes, phase=0.0):
    """Create a spectral seed."""
    x = np.arange(n) / n
    h = np.full(n, 0.5)
    for f, a in zip(freqs, amplitudes):
        h += a * np.sin(2 * np.pi * f * x + phase)
    return project_feasible(h, n / 2.0)
