from scipy.optimize import linprog, minimize, minimize_scalar
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx])
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            if rem > 0:

            remaining -= rem

        else:
            idx = int(np.argmin(h))

        h = np.clip(h, 0.0, 1.0)

    G = np.zeros((M, n))
    for m in range(M):
        d = m - (n - 1)
        if d == 0:
            G[m, :] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            G[m, d:] += dx * (1.0 - h[0:n-d])
            G[m, 0:n-d] += -dx * h[d:n]
        else:
            dp = -d
            G[m, 0:n-dp] += dx * (1.0 - h[dp:n])
            G[m, dp:n] += -dx * h[0:n-dp]
    return G

def lp_minimax_direction(h, n, dx, D=1.0):
    """Solve LP to find direction minimizing linearized max profile."""
    G = build_gradient_matrix(h, n, dx)
    P = compute_profile(h)
    M = 2 * n - 1
    c = max(P)
    
    # Variables: [d_0, ..., d_{n-1}, S]
    # Minimize S
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    # Inequality: G[m] @ d - S <= -P[m]  for all m
    # i.e., G[m] @ d + [0,...,0,-1] @ S <= -P[m]
    A_ub = np.zeros((M, n + 1))
    A_ub[:, :n] = G
    A_ub[:, -1] = -1.0
    b_ub = -P
    
    # Equality: sum(d_i) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = [0.0]
    
    # Bounds: -D <= d_i <= D, S in [-10, 10]
    bounds = [(-D, D)] * n + [(-10.0, 10.0)]
    
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
    
    if result.success:
        d = result.x[:n]
        S_pred = result.x[-1]
        return d, S_pred
    return None, None

def line_search(h, d, alpha_max=None, n_samples=25):
    """Find best alpha in [0, alpha_max] minimizing exact_score(h + alpha*d)."""
    if alpha_max is None:
        alpha_max = 1.0
        for i in range(len(h)):
            if d[i] > 1e-12:
                alpha_max = min(alpha_max, (1.0 - h[i]) / d[i])
            elif d[i] < -1e-12:
                alpha_max = min(alpha_max, h[i] / (-d[i]))
        alpha_max = max(alpha_max, 0.0)
    
    if alpha_max < 1e-14:
        return 0.0, exact_score(h)
    
    # Sample
    alphas = np.linspace(0, alpha_max, n_samples)
    scores = []
    for a in alphas:
        h_try = h + a * d
        if h_try.min() < -1e-10 or h_try.max() > 1 + 1e-10:
            scores.append(1e10)
        else:
            scores.append(exact_score(h_try))
    scores = np.array(scores)
    best_idx = np.argmin(scores)
    best_alpha = alphas[best_idx]
    best_score = scores[best_idx]
    
    # Refine around best
    if 0 < best_idx < n_samples - 1:
        lo = alphas[max(0, best_idx - 1)]
        hi = alphas[min(n_samples - 1, best_idx + 1)]
    elif best_idx == 0:
        lo = 0.0
        hi = alphas[1]
    else:
        lo = alphas[-2]
        hi = alphas[-1]
    
    def obj(a):
        h_try = h + a * d
        if h_try.min() < -1e-10 or h_try.max() > 1 + 1e-10:
            return 1e10
        return exact_score(h_try)
    
    res = minimize_scalar(obj, bounds=(lo, hi), method='bounded',
                          options={'maxiter': 20})
    if res.success and res.fun < best_score:
        best_alpha = res.x
        best_score = res.fun
    
    return best_alpha, best_score

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = np.linspace(0, 2, n + 1)
    xt = np.linspace(0, 2, new_n + 1)
    h_interp = np.zeros(new_n)

import numpy as np
from scipy.optimize import linprog, minimize, minimize_scalar
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx])
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-14:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmin(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def build_gradient_matrix(h, n, dx):
    M = 2 * n - 1
    G = np.zeros((M, n))
    for m in range(M):
        d = m - (n - 1)
        if d == 0:
            G[m, :] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            G[m, d:] += dx * (1.0 - h[0:n-d])
            G[m, 0:n-d] += -dx * h[d:n]
        else:
            dp = -d
            G[m, 0:n-dp] += dx * (1.0 - h[dp:n])
            G[m, dp:n] += -dx * h[0:n-dp]
    return G

def lp_minimax_direction(h, D=1.0):
    n = len(h)
    dx = 2.0 / n
    G = build_gradient_matrix(h, n, dx)
    P = compute_profile(h)
    M = 2 * n - 1
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_ub = np.zeros((M, n + 1))
    A_ub[:, :n] = G
    A_ub[:, -1] = -1.0
    b_ub = -P
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = [0.0]
    bounds = [(-D, D)] * n + [(-10.0, 10.0)]
    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs')
        if result.success:
            d = result.x[:n]
            S_pred = float(result.x[-1])
            return d, S_pred
    except Exception:
        pass
    return None, None

def line_search(h, d, alpha_max=None):
    if alpha_max is None:
        alpha_max = 1.0
        for i in range(len(h)):
            if d[i] > 1e-12:
                alpha_max = min(alpha_max, (1.0 - h[i]) / d[i])
            elif d[i] < -1e-12:
                alpha_max = min(alpha_max, h[i] / (-d[i]))
        alpha_max = min(alpha_max, 2.0)
        alpha_max = max(alpha_max, 0.0)
    if alpha_max < 1e-14:
        return 0.0, exact_score(h)
    n_samples = 20
    alphas = np.linspace(0, alpha_max, n_samples)
    base_c = exact_score(h)
    best_alpha = 0.0
    best_score = base_c
    for a in alphas:
        h_try = h + a * d
        if h_try.min() < -5e-6 or h_try.max() > 1.0 + 5e-6:
            continue
        s = exact_score(h_try)
        if s < best_score:
            best_score = s
            best_alpha = a
    if best_alpha > 0 and best_alpha < alpha_max:
        margin = alpha_max * 0.3
        lo = max(0.0, best_alpha - margin)
        hi = min(alpha_max, best_alpha + margin)
        def obj_ls(a):
            h_try = h + a * d
            if h_try.min() < -5e-6 or h_try.max() > 1.0 + 5e-6:
                return 1e10
            return exact_score(h_try)
        try:
            res = minimize_scalar(obj_ls, bounds=(lo, hi), method='bounded',
                                  options={'maxiter': 30})
            if res.fun < best_score:
                best_alpha = res.x
                best_score = res.fun
        except Exception:
            pass
    return best_alpha, best_score

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = np.linspace(0, 2, n + 1)
    xt = np.linspace(0, 2, new_n + 1)
    h_interp = np.zeros(new_n)
    for k in range(new_n):
        lo_k, hi_k = xt[k], xt[k + 1]
        total_mass = 0.0
        total_width = 0.0
        for j in range(n):
            lo_j, hi_j = xs[j], xs[j + 1]
            overlap = min(hi_k, hi_j) - max(lo_k, lo_j)
            if overlap > 0:
                total_mass += h[j] * overlap
                total_width += overlap
        h_interp[k] = total_mass / total_width if total_width > 1e-15 else 0.5
    h_interp = np.clip(h_interp, 0.0, 1.0)
    return repair_feasible(h_interp)

def symmetrize(h):
    return (h + h[::-1]) / 2.0

def antisymmetrize(h):
    n = len(h)
    h_out = np.zeros(n)
    for i in range(n):
        h_out[i] = (h[i] + (1.0 - h[n - 1 - i])) / 2.0
    return repair_feasible(h_out)

def lp_descent(h, max_iters=300, D=1.0, time_limit=30.0):
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    t0 = time.perf_counter()
    stuck = 0
    for it in range(max_iters):
        if time.perf_counter() - t0 > time_limit:
            break
        d, S_pred = lp_minimax_direction(h, D=D)
        if d is None:
            break
        d_norm = np.linalg.norm(d)
        if d_norm < 1e-10:
            break
        d = d / d_norm
        alpha, s_new = line_search(h, d)
        h_try = h + alpha * d
        h_try = repair_feasible(h_try)
        s_actual = exact_score(h_try)
        if s_actual < best_c - 1e-13:
            h = h_try
            best_c = s_actual
            stuck = 0
        elif s_actual < best_c:
            h = h_try
            best_c = s_actual
            stuck += 1
        else:
            stuck += 1
            if stuck > 25:
                break
    return h, best_c

def pair_move_descent(h, max_iters=500, time_limit=30.0, rng=None):
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    t0 = time.perf_counter()
    stuck = 0
    for it in range(max_iters):
        if time.perf_counter() - t0 > time_limit:
            break
        if rng is not None:
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
        else:
            i = (it * 7 + 3) % n
            j = (it * 13 + 5) % n
            if i == j:
                continue
        if h[i] <= h[j]:
            i, j = j, i
        delta_max = h[i] - h[j]
        if delta_max < 1e-10:
            delta_max = min(h[i], 1.0 - h[j])
        if delta_max < 1e-10:
            continue
        d = np.zeros(n)
        d[j] = 1.0
        d[i] = -1.0
        alpha, s_new = line_search(h, d, alpha_max=delta_max)
        if s_new < best_c - 1e-13:
            h = h + alpha * d
            h = repair_feasible(h)
            best_c = exact_score(h)
            stuck = 0
        else:
            stuck += 1
            if stuck > 100:
                break
    return h, best_c

def lse_descent(h, beta=80.0, max_iters=200, alpha_init=0.004, time_limit=20.0):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best_c = exact_score(h)
    alpha = alpha_init
    stuck = 0
    t0 = time.perf_counter()
    for it in range(max_iters):
        if time.perf_counter() - t0 > time_limit:
            break
        profile = compute_profile(h)
        scaled = beta * profile
        max_s = np.max(scaled)
        shifted = scaled - max_s
        weights = np.exp(shifted)
        wsum = weights.sum()
        if wsum < 1e-300:
            break
        weights /= wsum
        G = build_gradient_matrix(h, n, dx)
        grad = G.T @ weights
        grad -= grad.mean()
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad /= gnorm
        improved = False
        for a in [alpha, alpha * 0.5, alpha * 0.25]:
            h_try = repair_feasible(h - a * grad)
            s = exact_score(h_try)
            if s < best_c - 1e-13:
                h = h_try
                best_c = s
                improved = True
                alpha = min(alpha * 1.1, 0.02)
                stuck = 0
                break
            elif s < exact_score(h):
                h = h_try
                improved = True
                break
        if not improved:
            alpha *= 0.5
            stuck += 1
            if stuck > 30 or alpha < 1e-9:
                break
    return h, best_c

def active_set_descent(h, max_iters=200, n_active=12, alpha_init=0.005, time_limit=20.0):

            break
        profile = compute_profile(h)
        c = float(np.max(profile))

def slsqp_solve(h0, maxiter=200):
    n = len(h0)
    dx = 2.0 / n
    M = 2 * n - 1
    c0 = exact_score(h0)
    def obj(x):
        return x[-1]
    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    def eq_fun(x):
        return x[:-1].sum() - n / 2.0
    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g
    def ineq_fun(x):
        h = np.clip(x[:-1], 0, 1)
        t = x[-1]
        profile = np.correlate(h, 1 - h, mode="full") * dx
        return t - profile
    def ineq_jac(x):
        h = np.clip(x[:-1], 0, 1)
        G = build_gradient_matrix(h, n, dx)
        J = np.zeros((M, n + 1))
        J[:, :-1] = -G
        J[:, -1] = 1.0
        return J
    x0 = np.concatenate([h0, [c0 + 1e-3]])
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 0.5)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None and np.isfinite(res.x).all():
            h_new = repair_feasible(res.x[:-1])
            c_new = exact_score(h_new)
            if c_new <= c0 + 1e-9:
                return h_new, c_new
    except Exception:
        pass
    return None, None

            if c_lp < best_c - 1e-13:

                best_c = c_lp
            h_ase, c_ase = active_set_descent(best_h, max_iters=80, n_active=10, time_limit=10.0)
            if c_ase < best_c - 1e-13:
                best_h = h_ase.copy()
                best_c = c_ase

    # Stage 11: Final SLSQP
    if time_left() > 30:
        h_sol, c_sol = slsqp_solve(best_h, maxiter=300)
        if h_sol is not None and c_sol < best_c - 1e-14:
            best_h = h_sol.copy()
            best_c = c_sol

    # Stage 12: Final polish
    if time_left() > 15:
        h_lp, c_lp = lp_descent(best_h, max_iters=100, D=0.5, time_limit=8.0)
        if c_lp < best_c - 1e-14:
            best_h = h_lp.copy()
            best_c = c_lp
        h_ase, c_ase = active_set_descent(best_h, max_iters=150, n_active=8, time_limit=5.0)
        if c_ase < best_c - 1e-14:
            best_h = h_ase.copy()
            best_c = c_ase
        h_lse, c_lse = lse_descent(best_h, beta=120, max_iters=100, alpha_init=0.002, time_limit=5.0)
        if c_lse < best_c - 1e-14:
            best_h = h_lse.copy()
            best_c = c_lse

    # Final validation
    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)

    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            final_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 2:
            final_c = float(eval_result[0])
    except Exception:
        pass

    return best_h, final_c, len(best_h)
