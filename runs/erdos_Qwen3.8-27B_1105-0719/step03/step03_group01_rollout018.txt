import time

def project_feasible(x, target_sum):
    """Project x to [0,1]^n with sum = target_sum."""
    n = len(x)
    x = np.clip(x, -5.0, 5.0)
    lo, hi = -10.0, 10.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        cnt = mask.sum()
        if cnt > 0:
            h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def c5_of_h(h, dx):
    return compute_corrs(h).max() * dx

def grad_corr_d(h, d, n):

    if d >= 0:

        # d/dh[j] = (1-h[j+d]) if j+d < n AND j < n-d (first factor)
        #         + (-h[j-d]) if j >= d AND j-d < n-d (second factor, i=j-d)

def lse_direction(h, n, T):
    """Compute LSE-smoothed gradient direction."""
    r = compute_corrs(h)  # shape (2n-1,)
    r_max = r.max()
    # Stable log-sum-exp weights
    d = (r - r_max) / T
    d = np.clip(d, -50.0, 0.0)  # numerical safety
    w = np.exp(d)
    w /= w.sum()
    # Add small floor
    w = w + 1e-6 / len(w)
    w /= w.sum()
    
    # Direction indices
    # r[p] corresponds to shift d = p - (n-1)
    G = np.zeros(n)
    for p in range(len(r)):
        if w[p] < 1e-10:
            continue
        d = p - (n - 1)
        G += w[p] * grad_corr_d(h, d, n)
    
    # Remove mean to maintain sum constraint
    G = G - G.mean()
    
    # Normalize
    norm = np.max(np.abs(G))
    if norm < 1e-15:
        return None, r.max() * (2.0/n)  # no direction
    G /= norm
    return G, r.max() * (2.0/n)

def lse_step(h, dx, target_sum, T, max_alpha=1.0):
    """One LSE-smoothed gradient step with line search."""
    n = len(h)
    G, cur_c5 = lse_direction(h, n, T)
    if G is None:
        return h, cur_c5, False
    
    # Find valid alpha range: h + alpha*G in [0,1]
    # G[i] > 0: alpha <= (1 - h[i]) / G[i]
    # G[i] < 0: alpha >= -(h[i]) / G[i] = h[i] / |G[i]|
    pos = G > 1e-12
    neg = G < -1e-12
    
    alpha_max = max_alpha
    if pos.any():
        alpha_max = min(alpha_max, np.min((1.0 - h[pos]) / G[pos]))
    if neg.any():
        # alpha >= -(-h[i]/G[i]) = h[i]/|G[i]|, but alpha should be positive for descent
        # Actually we want alpha > 0 such that h + alpha*G stays in [0,1]
        # For G[i] < 0: h[i] + alpha*G[i] >= 0 => alpha <= h[i]/|G[i]|
        alpha_max = min(alpha_max, np.min(h[neg] / np.abs(G[neg])))
    
    if alpha_max <= 0:
        return h, cur_c5, False
    
    # Line search
    best_h = h.copy()
    best_c5 = cur_c5
    alphas = [alpha_max, alpha_max*0.7, alpha_max*0.5, alpha_max*0.3, alpha_max*0.1, alpha_max*0.05]
    for a in alphas:
        if a <= 0:
            continue
        h_new = project_feasible(h + a * G, target_sum)
        c5_new = c5_of_h(h_new, dx)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new
    
    return best_h, best_c5, best_c5 < cur_c5

def lse_optimize(h, dx, target_sum, t0, max_time, T_start=0.5, T_end=0.01, 
                 n_steps=500, alpha_scale=1.0):
    """Run LSE optimization with cooling schedule."""
    n = len(h)
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    cur_h = h.copy()
    cur_c5 = best_c5
    
    n_steps_actual = 0
    no_improve = 0
    
    for step in range(n_steps):
        if time.time() - t0 > max_time:
            break
        # Temperature schedule: exponential cooling
        frac = step / max(1, n_steps)
        T = T_start * (T_end / T_start) ** frac
        
        # Occasionally restart from best
        if no_improve > 50:
            cur_h = best_h + 0.02 * np.random.randn(n)
            cur_h = project_feasible(cur_h, target_sum)
            cur_c5 = c5_of_h(cur_h, dx)
            no_improve = 0
        
        h_new, c5_new, improved = lse_step(cur_h, dx, target_sum, T, 
                                            max_alpha=alpha_scale)
        n_steps_actual += 1
        
        if improved:
            cur_h = h_new
            cur_c5 = c5_new
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
    
    return best_h, best_c5
