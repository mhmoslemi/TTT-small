import numpy as np
from scipy.optimize import minimize, linprog
from scipy.special import logsumexp
import time


def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)


def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def c5_of_h(h, dx):
    return compute_corrs(h).max() * dx


def grad_corr_k(h, k, n):
    offset = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -offset)
    i_hi = min(n - 1, n - 1 - offset)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx] += (1.0 - h[idx + offset])
    grad[idx + offset] += (-h[idx])
    return grad

    if w_sum < 1e-30:

def pair_transfer(h0, dx, target_sum, n_top=30, n_iter=5):
    n = len(h0)
    n_c = 2 * n - 1
    h = h0.copy()
    best_h = h.copy()

    for iteration in range(n_iter):
        corrs = compute_corrs(h)
        
        # Compute weighted gradient
        beta = 500.0
        m = corrs.max()
        w = np.exp(beta * (corrs - m))
        w /= w.sum()
        grad = np.zeros(n)
        for k in range(n_c):
            if w[k] > 1e-13:
                grad += w[k] * grad_corr_k(h, k, n)
        
        dec_idx = np.argsort(grad)[-n_top:]
        inc_idx = np.argsort(grad)[:n_top]
        
        improved = False
        for i in dec_idx:
            for j in inc_idx:
                if i == j:
                    continue
                eps_lo = max(-h[j], -1e-15)
                eps_hi = min(h[i], 1.0 - h[j])
                if eps_hi - eps_lo < 1e-14:
                    continue
                
                d = np.zeros(n_c)
                for k in range(n_c):
                    gk = grad_corr_k(h, k, n)
                    d[k] = gk[j] - gk[i]
                
                # Minimize max_k (corrs[k] + eps * d[k]) over eps in [eps_lo, eps_hi]
                # Golden section search
                gr = (np.sqrt(5.0) - 1.0) / 2.0
                a, b = eps_lo, eps_hi
                c_pt = b - gr * (b - a)
                d_pt = a + gr * (b - a)
                fc = (corrs + c_pt * d).max()
                fd = (corrs + d_pt * d).max()
                for _ in range(40):
                    if b - a < 1e-15:
                        break
                    if fc < fd:
                        b = d_pt
                        d_pt = c_pt
                        fd = fc
                        c_pt = b - gr * (b - a)
                        fc = (corrs + c_pt * d).max()
                    else:
                        a = c_pt
                        c_pt = d_pt
                        fc = fd
                        d_pt = a + gr * (b - a)
                        fd = (corrs + d_pt * d).max()
                
                eps_opt = (a + b) / 2.0
                if eps_opt <= eps_lo + 1e-14 or eps_opt >= eps_hi - 1e-14:
                    continue
                
                new_h = h.copy()
                new_h[i] -= eps_opt
                new_h[j] += eps_opt
                new_h = np.clip(new_h, 0.0, 1.0)
                new_h = project_feasible(new_h, target_sum)
                new_c5 = c5_of_h(new_h, dx)
                
                if new_c5 < best_c5 - 1e-14:
                    h = new_h
                    best_c5 = new_c5
                    best_h = h.copy()
                    corrs = compute_corrs(h)

            break

def lmo_seed(h0, dx, target_sum, n_active=32):
    """Generate a binary seed using LMO on aggregated gradient."""
    n = len(h0)
    n_c = 2 * n - 1
    corrs = compute_corrs(h0)
    
    # Get top active
    top_k = np.argsort(corrs)[-n_active:]
    
    # Aggregated gradient (uniform weights over active)
    G = np.zeros(n)
    w = 1.0 / len(top_k)
    for k in top_k:
        G += w * dx * grad_corr_k(h0, k, n)
    
    # LMO: min G@v s.t. 0<=v<=1, sum(v)=n/2
    # Solution: v[i]=1 for the n/2 positions with smallest G[i], 0 otherwise
    n_ones = int(target_sum)
    sorted_idx = np.argsort(G)
    v = np.zeros(n)
    v[sorted_idx[:n_ones]] = 1.0
    
    return v


def lmo_line_search(h0, v, dx, target_sum, n_gammas=10):
    """Try (1-gamma)*h + gamma*v for various gamma."""
    best_h = h0.copy()
    best_c5 = c5_of_h(h0, dx)
    
    gammas = np.linspace(0.1, 0.9, n_gammas)
    for g in gammas:
        h_try = (1 - g) * h0 + g * v
        h_try = project_feasible(h_try, target_sum)
        c5_try = c5_of_h(h_try, dx)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try
    
    return best_h, best_c5


def grad_descent_steps(h0, dx, target_sum, n_steps=500, init_lr=0.05, beta=100.0):
    """Projected gradient descent with smooth max."""
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    lr = init_lr
    
    for it in range(n_steps):
        g = smooth_max_grad_all(h, dx, beta, n)
        h_new = h - lr * g
        h_new = project_feasible(h_new, target_sum)
        new_c5 = c5_of_h(h_new, dx)
        
        if new_c5 < best_c5 - 1e-15:
            best_c5 = new_c5
            best_h = h_new.copy()
            h = h_new
            # Decay lr
            if it % 50 == 0:
                lr *= 0.9
        else:
            h = h_new
            lr *= 0.9
            if lr < 1e-6:
                break
    
    return best_h, best_c5


def make_binary_start(n, rng, target_sum):
    """Create a random balanced binary vector."""
    idx = rng.choice(n, size=int(target_sum), replace=False)
    h = np.zeros(n)
    h[idx] = 1.0
    return h


def optimize_single(h0, dx, target_sum, time_fn, label=""):
    """Full optimization pipeline for a single starting point."""
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    
    # Step 1: Quick gradient descent
    if time_fn() > 30:
        h_gd, c5_gd = grad_descent_steps(h, dx, target_sum, n_steps=300, init_lr=0.03, beta=50)
        if c5_gd < best_c5:
            best_c5 = c5_gd
            best_h = h_gd
    
    # Step 2: All-active LP polish
    if time_fn() > 20:
        h_lp, c5_lp = all_active_lp_polish(best_h, dx, target_sum, max_iters=200, init_step=0.08)
        if c5_lp < best_c5:
            best_c5 = c5_lp
            best_h = h_lp
    
    # Step 3: Pair transfer
    if time_fn() > 20:
        h_pt, c5_pt = pair_transfer(best_h, dx, target_sum, n_top=25, n_iter=5)
        if c5_pt < best_c5:
            best_c5 = c5_pt
            best_h = h_pt
    
    # Step 4: Another LP pass
    if time_fn() > 15:
        h_lp2, c5_lp2 = all_active_lp_polish(best_h, dx, target_sum, max_iters=200, init_step=0.05)
        if c5_lp2 < best_c5:
            best_c5 = c5_lp2
            best_h = h_lp2
    
    # Step 5: Final pair transfer
    if time_fn() > 10:
        h_pt2, c5_pt2 = pair_transfer(best_h, dx, target_sum, n_top=20, n_iter=3)
        if c5_pt2 < best_c5:
            best_c5 = c5_pt2
            best_h = h_pt2
    
    return best_h, best_c5
