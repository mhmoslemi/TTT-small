```python
import numpy as np
from scipy.optimize import minimize
import time

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(-(1.0 - h))
        remaining = diff
        for i in range(n):
            idx = order[i]
            cap = 1.0 - h[idx]
            if cap <= 1e-15:
                continue
            add = min(remaining, cap)
            h[idx] += add
            remaining -= add
            if remaining <= 1e-14:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            if h[idx] <= 1e-15:
                continue
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-14:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] = float(np.clip(h[idx] + diff2, 0.0, 1.0))
    return h

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def grad_profile_row(h, m, n):
    dx = 2.0 / n
    d = m - (n - 1)
    if d == 0:
        return dx * (1.0 - 2.0 * h)
    elif d > 0:
        g = np.zeros(n)
        g[d:] += dx * (1.0 - h[:-d])
        g[:-d] -= dx * h[d:]
        return g
    else:
        p = -d
        g = np.zeros(n)
        g[:-p] += dx * (1.0 - h[p:])
        g[p:] -= dx * h[:-p]
        return g

def lse_obj_and_grad(h, beta):
    n = len(h)
    profile = compute_profile(h)
    max_p = profile.max()
    shifted = beta * (profile - max_p)
    # Clip to avoid overflow (shouldn't happen since shifted <= 0)
    shifted = np.minimum(shifted, 0.0)
    exp_terms = np.exp(shifted)
    sum_exp = exp_terms.sum()
    if sum_exp < 1e-300:
        # Fallback: only max is active
        lse_val = max_p
        weights = np.zeros(len(profile))
        weights[int(np.argmax(profile))] = 1.0
    else:
        lse_val = max_p + np.log(sum_exp) / beta
        weights = exp_terms / sum_exp
    grad = np.zeros(n)
    # Only use non-negligible weights
    active = np.where(weights > 1e-10)[0]
    for m in active:
        grad += weights[m] * grad_profile_row(h, m, n)
    # Project out the mean (to respect the equality constraint)
    grad -= grad.mean()
    return lse_val, grad

def lse_lbfgs(h_init, beta, maxiter=200, time_limit=30):
    t0 = time.perf_counter()
    n = len(h_init)
    
    def fun_and_grad(x):
        h = repair_feasible(x)
        val, grad = lse_obj_and_grad(h, beta)
        return val, grad
    
    x0 = h_init.copy()
    try:

def subgradient_averaging(h_init, max_iters=500, time_limit=60):
    t0 = time.perf_counter()
    n = len(h_init)
    h = h_init.copy()
    best_c = exact_score(h)
    best_h = h.copy()
    # Polyak-type averaging

    sum_steps = 0
    c0 = exact_score(h)
    # Estimate lower bound (use the average of the profile)

    lb = profile.mean()  # crude lower bound

        if time.perf_counter() - t0 > time_limit:

        profile = compute_profile(h)
        c = float(profile.max())
        if c < best_c:

            best_h = h.copy()
        
        # Subgradient: pick the active constraint

        g = grad_profile_row(h, m_active, n)

        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:

        # Use alpha = c / (g_norm^2 * sqrt(it + 1))

        if c_avg < best_c:
            best_c = c_avg
            best_h = h_avg_repaired.copy()
        
        h = h_new
    
    return best_h, best_c

    n = len(h)
    h = h.copy()
    best_c = exact_score(h)

            g += grad_profile_row(h, m, n)

        return h.copy()

    edges_new = np.linspace(0.0, 2.0, new_n + 1)
    h_new = np.zeros(new_n)
    for k in range(new_n):
        a, b = edges_new[k], edges_new[k + 1]
        total = 0.0
        for j in range(n):
            lo = max(a, edges_old[j])
            hi = min(b, edges_old[j + 1])
            if hi > lo:
                total += h[j] * (hi - lo)
        h_new[k] = total / (b - a)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

def binary_seed(n, rng, mode=0):

        # Golden ratio
        idx = np.arange(n)
        vals = (idx * 0.618033988749895) % 1.0
        top = np.argsort(vals)[-half:]
        h[top] = 1.0
    elif mode == 1:
        # Different irration
        idx = np.arange(n)
        vals = (idx * 0.4142135623730951) % 1.0
        top = np.argsort(vals)[-half:]
        h[top] = 1.0
    elif mode == 2:
        # Periodic
        period = max(3, n // 7)
        for i in range(n):
            if (i // period) % 2 == 0:
                h[i] = 1.0
        ones = int(h.sum())
        if ones > half:
            excess = ones - half
            one_idx = np.where(h == 1.0)[0]
            to_flip = rng.choice(one_idx, excess, replace=False)
            h[to_flip] = 0.0
        elif ones < half:
            deficit = half - ones
            zero_idx = np.where(h == 0.0)[0]
            to_flip = rng.choice(zero_idx, deficit, replace=False)
            h[to_flip] = 1.0
    else:
        perm = rng.permutation(n)
        h[perm[:half]] = 1.0
    return repair_feasible(h)
```
