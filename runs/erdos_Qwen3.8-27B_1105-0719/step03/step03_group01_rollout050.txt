Wait, let me be more careful. The term in c_d that involves both h[j] and h[l] (when they appear in the same product):
- Original: h[l]*(1-h[j]) if d = l-j (j is second factor, l is first)
- After swap: (h[l]-δ)*(1-(h[j]+δ)) = (h[l]-δ)*(1-h[j]-δ) = h[l]*(1-h[j]) - h[l]*δ - δ*(1-h[j]) + δ²
  = h[l]*(1-h[j]) - δ*(h[l] + 1 - h[j]) + δ²
- Change: -δ*(h[l] + 1 - h[j]) + δ²

So the quadratic coefficient is +1 in this case.

Similarly if d = j-l (l is second factor, j is first):
- Original: h[j]*(1-h[l])
- After: (h[j]+δ)*(1-(h[l]-δ)) = (h[j]+δ)*(1-h[l]+δ) = h[j]*(1-h[l]) + h[j]*δ + δ*(1-h[l]) + δ²
  Change: δ*(h[j] + 1 - h[l]) + δ²

So in both cases where |j-l| = |d|, there's a +δ² term (the coefficient of δ² is +1).

This means for pairs where |j-l| equals some lag d, the objective f(δ) has a quadratic component, making it not purely piecewise linear. But it's still convex (the δ² coefficient is positive), so the minimum is still well-defined.

For a given h and lag d:
grad[j] = (1 - h[j-d]) * I[0<=j-d<n] + (-h[j+d]) * I[0<=j+d<n]

This can be computed as:
- Part 1: (1 - h[j-d]) where j-d >= 0 and j-d < n → j ranges from max(0,d) to min(n-1, n-1+d)
  Actually, 0 <= j-d < n means d <= j < n+d, but also 0 <= j < n
  So j ranges from max(0, d) to min(n-1, n-1+d) = min(n-1, n+d-1)
  For d >= 0: j from d to n-1 (since n+d-1 >= n-1 when d>=0)
  For d < 0: j from 0 to n-1+d (since max(0,d)=0)
  
- Part 2: -h[j+d] where 0 <= j+d < n → -d <= j < n-d, and 0 <= j < n
  For d >= 0: j from 0 to n-1-d
  For d < 0: j from -d to n-1

                        h = h_new
                        c5_cur = c5_new
                        best_c5 = c5_new
                        best_h = h_new.copy()
                        improved = True
                        no_improve = 0
            except Exception:
                pass

        if not improved:
            no_improve += 1

    return best_h, best_c5


def subgradient_descent(h, dx, target_sum, t0, deadline, rng, n_iters=5000):
    """Subgradient method for minimax with mirror descent steps"""
    n = len(h)
    c5_cur = c5_of_h(h, dx)
    best_h = h.copy()
    best_c5 = c5_cur

    for it in range(n_iters):
        if time.time() > deadline:
            break
        if it % 200 == 0 and best_c5 < c5_cur * 0.999:
            h = best_h.copy()
            c5_cur = best_c5

        corrs = compute_corrs(h)
        c5_cur = corrs.max() * dx

        # Find active lags (within 5% of max)
        max_val = corrs.max()
        active = np.where(corrs >= max_val * 0.97)[0]
        n_active = len(active)

        # Subgradient: average gradient over active lags
        G_sum = np.zeros(n)
        for k in active:
            d = k - (n - 1)
            G_sum += grad_corr_d(h, d)
        G_avg = dx * G_sum / n_active

        # Mirror descent step (exponential weights / negative entropy)
        lr = 0.5 / (it + 10) * max(0.01, 1.0 - 0.9 * it / n_iters)
        # Update in log-space (for multiplicative weights)
        h_log = np.where(h > 1e-8, np.log(h / (1.0 - h + 1e-8) + 1e-8), -5.0)
        h_log -= lr * G_avg / (0.01 + np.abs(G_avg).max())
        h_new = 1.0 / (1.0 + np.exp(-np.clip(h_log, -20, 20)))
        h_new = project_feasible(h_new, target_sum)
        c5_new = c5_of_h(h_new, dx)

        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new.copy()

    return best_h, best_c5


def try_n_value(n_try, h_best, c5_best, t0, deadline, rx):
    n_orig = len(h_best)
    x_orig = (np.arange(n_orig) + 0.5) / n_orig
    x_new = (np.arange(n_try) + 0.5) / n_try
    h_interp = np.interp(x_new, x_orig, h_best)
    target_sum = n_try / 2.0
    h_interp = project_feasible(h_interp, target_sum)
    dx = 2.0 / n_try
    c5_init = c5_of_h(h_interp, dx)

    if c5_init < c5_best:
        # Try SLSQP on this
        if time.time() < deadline - 10:
            h_opt, c5_opt = slsqp_solve(h_interp, dx, target_sum, t0, deadline - 5, max_iter=50)
            if h_opt is not None and c5_opt < c5_best:
                return h_opt, c5_opt, n_try
        return h_interp, c5_init, n_try
    return None, c5_best, n_try


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    # Load parent
    if 'initial_h_values' in dir() or True:
        try:
            h0 = np.array(initial_h_values, dtype=float)
        except NameError:
            h0 = None

    if h0 is None or len(h0) == 0:
        # Fallback: construct a simple feasible vector
        n = 126
        h0 = np.full(n, 0.5)
        h0[:n // 4] = 1.0
        h0[n - n // 4:] = 0.0
        h0 = project_feasible(h0, n / 2.0)

    n = len(h0)
    dx = 2.0 / n
    target_sum = n / 2.0
    h0 = project_feasible(h0, target_sum)
    best_h = h0.copy()
    best_c5 = c5_of_h(h0, dx)

    # Phase 1: SLSQP on parent (most impactful)
    deadline1 = t0 + budget_s * 0.20
    h_s, c5_s = slsqp_solve(best_h, dx, target_sum, t0, deadline1, max_iter=150)
    if h_s is not None and c5_s < best_c5:
        best_c5 = c5_s
        best_h = h_s

    # Phase 2: Swap descent (main local optimizer)
    deadline2 = t0 + budget_s * 0.50
    best_h, best_c5 = swap_descent(best_h, dx, target_sum, t0, deadline2, rng)

    # Phase 3: SLSQP again on improved point
    deadline3 = t0 + budget_s * 0.62
    if time.time() < deadline3 - 5:
        h_s2, c5_s2 = slsqp_solve(best_h, dx, target_sum, t0, deadline3, max_iter=100, ftol=1e-12)
        if h_s2 is not None and c5_s2 < best_c5:
            best_c5 = c5_s2
            best_h = h_s2

    # Phase 4: More swap descent with different random seed
    deadline4 = t0 + budget_s * 0.75
    if time.time() < deadline4 - 5:
        rng2 = np.random.RandomState(seed + 999)
        best_h, best_c5 = swap_descent(best_h, dx, target_sum, t0, deadline4, rng2)

    # Phase 5: Subgradient for global mixing
    deadline5 = t0 + budget_s * 0.85
    if time.time() < deadline5 - 10:
        h_sub, c5_sub = subgradient_descent(best_h, dx, target_sum, t0, deadline5, rng, n_iters=3000)
        if c5_sub < best_c5:
            best_c5 = c5_sub
            best_h = h_sub
        # Polish subgradient result
        if time.time() < deadline5:
            h_pol, c5_pol = swap_descent(h_sub, dx, target_sum, t0, deadline5, rng)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol

    # Phase 6: Try different n values
    deadline6 = t0 + budget_s * 0.94
    n_candidates = [100, 108, 112, 120, 128, 132, 140, 144, 148]
    for n_try in n_candidates:
        if time.time() > deadline6:
            break
        if n_try == n:
            continue
        dt = deadline6 - (time.time() - t0)
        if dt < 8:
            break
        h_new, c5_new, _ = try_n_value(n_try, best_h, best_c5, t0, deadline6, rng)
        if h_new is not None and c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

    # Phase 7: Final intensive polish
    final_deadline = t0 + budget_s - 3.0
    n_f = len(best_h)
    dx_f = 2.0 / n_f
    ts_f = n_f / 2.0

    # Tiny SLSQP
    if time.time() < final_deadline - 8:
        h_f, c5_f = slsqp_solve(best_h, dx_f, ts_f, t0, final_deadline, max_iter=80, ftol=1e-13)
        if h_f is not None and c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f

    # Final swap polish
    if time.time() < final_deadline:
        best_h, best_c5 = swap_descent(best_h, dx_f, ts_f, t0, final_deadline, rng, max_sweeps=50)

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    n_final = len(best_h)
    ts_final = n_final / 2.0
    best_h = project_feasible(best_h, ts_final)
    dx_final = 2.0 / n_final
    best_c5 = c5_of_h(best_h, dx_final)

    return best_h, best_c5, n_final
