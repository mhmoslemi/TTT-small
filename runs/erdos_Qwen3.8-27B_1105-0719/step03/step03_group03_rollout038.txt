def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    # Water-filling to adjust sum
    if diff > 0:

        # Add to entries with most room (closest to 1 from below)

            add = min(remaining, room[idx])

At the optimum, the subgradient of f at h is the convex hull of the gradients of the active constraints. For a descent direction, we need a direction v such that the directional derivative f'(h; v) < 0.

```python
def compute_smooth_gradient(h, w):
    n = len(h)
    M = 2 * n - 1
    grad = np.zeros(n)
    # For each m, compute the contribution
    # First term: (1-h[i-m]) for i in [m, n-1] (when m < n)
    #              (1-h[i-m]) for i in [0, 2n-1-m-1] (when m >= n)
    # Second term: -h[i+m] for i in [0, n-1-m] (when m < n)
    #              -h[i+m] for i in [0, 2n-1-m-1] (when m >= n)
    
    for m in range(M):
        wm = w[m]
        if abs(wm) < 1e-20:
            continue
        # First term: indices i where 0 <= i-m < n
        i_lo1 = max(0, m)
        i_hi1 = min(n - 1, m + n - 1)
        if i_lo1 <= i_hi1:
            k_idx = np.arange(i_lo1, i_hi1 + 1) - m  # these are valid indices into h
            grad[i_lo1:i_hi1+1] += wm * (1.0 - h[k_idx])
        # Second term: indices i where 0 <= i+m < n
        i_lo2 = 0
        i_hi2 = min(n - 1, n - 1 - m)
        if i_lo2 <= i_hi2:
            k_idx2 = np.arange(i_lo2, i_hi2 + 1) + m
            grad[i_lo2:i_hi2+1] -= wm * h[k_idx2]
    return grad
```

This is vectorized over i for each m, so it's O(M * n) but with vectorized inner loops. For n=70, M=139, this is about 10k floating point ops per call. Very fast.

import numpy as np
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full")) * dx)

def compute_profile_raw(h):
    return np.correlate(h, 1 - h, mode="full")

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        order = np.argsort(-room)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, room[idx] + 1e-15)
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx] + 1e-15)
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    # Final fix
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] = np.clip(h[idx] + diff2, 0, 1)
    return h

def line_search_exact(h, v, c_current, n_grid=40):
    """Exact 1D line search along h + alpha*v where sum(v)=0."""
    n = len(h)
    P_raw = compute_profile_raw(h)
    A = np.correlate(v, 1 - h, mode='full') - np.correlate(h, v, mode='full')
    W = np.correlate(v, v, mode='full')
    dx = 2.0 / n
    
    # Valid alpha range
    alpha_lo = 0.0
    alpha_hi = 0.3
    pos_v = v > 1e-12
    neg_v = v < -1e-12
    if np.any(pos_v):
        bounds_lo = -h[pos_v] / v[pos_v]
        alpha_lo = max(alpha_lo, bounds_lo.min())
    if np.any(neg_v):
        bounds_hi = -h[neg_v] / v[neg_v]
        alpha_hi = min(alpha_hi, bounds_hi.min())
    
    if alpha_hi - alpha_lo < 1e-14:
        return h.copy(), c_current
    
    def f(alpha):
        return float(np.max(P_raw + alpha * A - alpha*alpha * W) * dx)
    
    # Coarse grid
    alphas = np.linspace(alpha_lo, alpha_hi, n_grid)
    vals = np.array([f(a) for a in alphas])
    idx = np.argmin(vals)
    best_alpha = alphas[idx]
    best_val = vals[idx]
    
    # Golden section refinement
    if idx > 0 and idx < n_grid - 1:
        a, b = alphas[idx-1], alphas[idx+1]
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c_pt = b - gr * (b - a)
        d_pt = a + gr * (b - a)
        fc = f(c_pt)
        fd = f(d_pt)
        for _ in range(50):
            if b - a < 1e-14:
                break
            if fc < fd:
                b, d_pt, fd = d_pt, c_pt, fc
                c_pt = b - gr * (b - a)
                fc = f(c_pt)
            else:
                a, c_pt, fc = c_pt, d_pt, fd
                d_pt = a + gr * (b - a)
                fd = f(d_pt)
        if fc < fd:
            best_alpha, best_val = c_pt, fc
        else:
            best_alpha, best_val = d_pt, fd
    
    if best_val < c_current - 1e-13:
        h_new = np.clip(h + best_alpha * v, 0, 1)
        h_new = repair_feasible(h_new)
        c_new = exact_score(h_new)
        if c_new < c_current - 1e-13:
            return h_new, c_new
    return h.copy(), c_current

def smooth_gradient(h, w):
    """Compute the smooth gradient direction for minimizing weighted sum of profile values."""
    n = len(h)
    M = 2 * n - 1
    grad = np.zeros(n)
    for m in range(M):
        wm = w[m]
        if abs(wm) < 1e-18:
            continue
        # First term: (1-h[i-m]) for i where 0<=i-m<n
        i_lo = max(0, m)
        i_hi = min(n - 1, m + n - 1)
        if i_lo <= i_hi:
            k = np.arange(i_lo, i_hi + 1) - m
            grad[i_lo:i_hi+1] += wm * (1.0 - h[k])
        # Second term: -h[i+m] for i where 0<=i+m<n
        i_hi2 = min(n - 1 - m, n - 1)
        if i_hi2 >= 0:
            k2 = np.arange(0, i_hi2 + 1) + m
            grad[0:i_hi2+1] -= wm * h[k2]
    return grad

def soft_max_weights(P_raw, gamma):
    """Compute soft-max weights from profile."""
    scale = np.max(P_raw)
    if scale < 1e-15:
        return np.ones(len(P_raw)) / len(P_raw)
    w = np.exp(gamma * (P_raw - scale) / scale)
    w /= w.sum()
    return w

def interp_to(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    fx = np.concatenate([[0.0], xs, [2.0]])
    fh = np.concatenate([[h[0]], h, [h[-1]]])
    hn = np.interp(xt, fx, fh)
    return repair_feasible(np.clip(hn, 0, 1))

def optimize_smooth(h, budget_fn, min_time=10):
    """Main optimization using smooth gradient + exact line search."""
    n = len(h)
    c_curr = exact_score(h)
    best_c = c_curr
