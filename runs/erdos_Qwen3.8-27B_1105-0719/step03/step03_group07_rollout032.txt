def pairwise_sweep(h, n_sweeps=1, n_pairs_per_sweep=None, t_start=None, budget_s=1100, margin=10):
    """Multiple sweeps of pairwise minimization."""
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    rng = np.random.RandomState(12345)
    
    if n_pairs_per_sweep is None:
        n_pairs_per_sweep = n * 3
    
    for sweep in range(n_sweeps):
        if t_start is not None and time.perf_counter() - t_start > budget_s - margin:
            break
        # Random pairs
        for _ in range(n_pairs_per_sweep):
            if t_start is not None and time.perf_counter() - t_start > budget_s - margin:
                break
            p = rng.randint(n)
            q = rng.randint(n)
            if p == q:
                continue
            h_before = exact_score(h)
            h = pairwise_minimize(h, p, q)
            c = exact_score(h)
            if c < best_c:
                best_c = c
    
    return h, best_c

def pairwise_sweep_ordered(h, t_start=None, budget_s=1100, margin=10):
    """Ordered pairwise minimization (all pairs in a fixed order)."""
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    
    for p in range(n):
        if t_start is not None and time.perf_counter() - t_start > budget_s - margin:
            break
        for q in range(p + 1, n):
            if t_start is not None and time.perf_counter() - t_start > budget_s - margin:
                break
            h = pairwise_minimize(h, p, q)
            # Only recompute score every few pairs for efficiency
            if (q - p) % 5 == 0:
                c = exact_score(h)
                if c < best_c:
                    best_c = c
    
    c = exact_score(h)
    if c < best_c:
        best_c = c
    return h, best_c

def slsqp_active(h0, maxiter=300, active_k=12, tol=1e-11):
    """SLSQP with top-K active constraints."""
    n = len(h0)
    dx = 2.0 / n
    M = 2 * n - 1
    P = profile_h(h0)
    c0 = P.max()
    
    thresh = c0 - max(0.003, 0.05 * (c0 - np.median(P)))
    active = np.where(P >= thresh)[0]
    if len(active) == 0:
        active = np.array([np.argmax(P)])
    if len(active) > active_k:
        top_idx = np.argsort(P[active])[-active_k:]
        active = active[top_idx]
    
    def build_J(active_lags):
        G = np.zeros((len(active_lags), n))
        for idx, m in enumerate(active_lags):
            d = m - (n - 1)
            if d == 0:
                G[idx, :] = dx * (1.0 - 2.0 * h0)
            elif d > 0:
                for i in range(n - d):
                    G[idx, i + d] += dx * (1.0 - h0[i])
                    G[idx, i] -= dx * h0[i + d]
            else:
                dp = -d
                for j in range(n - dp):
                    G[idx, j + dp] += dx * (1.0 - h0[j])
                    G[idx, j] -= dx * h0[j + dp]
        return G
    
    # Use current h0 for Jacobian (fixed Jacobian approximation)
    G_fixed = build_J(active)
    
    def obj(x):
        return x[-1]
    
    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    
    def eq_fun(x):
        return x[:-1].sum() - n / 2.0
    
    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g
    
    def ineq_fun(x):
        h = x[:-1]
        t = x[-1]
        prof = profile_h(h)
        return t - prof[list(active)]
    
    def ineq_jac(x):
        h = x[:-1]
        G = np.zeros((len(active), n))
        mm = active
        for idx, m in enumerate(mm):
            d = m - (n - 1)
            if d == 0:
                G[idx, :] = dx * (1.0 - 2.0 * h)
            elif d > 0:
                for i in range(n - d):
                    G[idx, i + d] += dx * (1.0 - h[i])
                    G[idx, i] -= dx * h[i + d]
            else:
                dp = -d
                for j in range(n - dp):
                    G[idx, j + dp] += dx * (1.0 - h[j])
                    G[idx, j] -= dx * h[j + dp]
        J = np.zeros((len(active), n + 1))
        J[:, :-1] = -G
        J[:, -1] = 1.0
        return J
    
    x0 = np.concatenate([h0, [c0 + 1e-4]])
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 1.0)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': tol})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = repair_feasible(res.x[:-1])
            c_new = exact_score(h_new)
            if c_new <= c0 + 1e-10:
                return h_new, c_new
    except Exception:
        pass
    return None, None

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    return repair_feasible(np.clip(h_new, 0.0, 1.0))

def symmetrize(h):
    return (h + h[::-1].copy()) / 2.0

import numpy as np
from scipy.optimize import minimize
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full"))) * dx

def profile_h(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx])
            h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    return h

def _pairwise_minimize(h, p, q):
    n = len(h)
    s = h[p] + h[q]
    t_min = max(0.0, s - 1.0)
    t_max = min(1.0, s)
    if t_max - t_min < 1e-14:
        return
    t0 = h[p]
    dx = 2.0 / n
    M = 2 * n - 1
    P0 = profile_h(h)
    d0 = n - 1
    # Quadratic term for d=0
    def g_quad(t):
        return s - s * s + 2.0 * s * t - 2.0 * t * t
    g0 = g_quad(t0)
    # Linear slopes for d != 0
    slopes = np.zeros(M)
    for m in range(M):
        if m == d0:
            continue
        d = m - d0
        sl = 0.0
        if 0 <= p + d < n:
            sl += (1.0 - h[p + d])
        if 0 <= p - d < n:
            sl -= h[p - d]
        if 0 <= q + d < n:
            sl -= (1.0 - h[q + d])
        if 0 <= q - d < n:
            sl += h[q - d]
        slopes[m] = dx * sl
    intercepts = P0 - slopes * t0
    intercepts[d0] = P0[d0]
    quad_offset = P0[d0] - dx * g0

    def eval_at(t):
        Pv = intercepts + slopes * t
        Pv[d0] = quad_offset + dx * g_quad(t)
        return Pv.max()

    # Build candidates
    candidates = [t_min, t_max, t_min + 0.25 * (t_max - t_min),
                  t_min + 0.5 * (t_max - t_min), t_min + 0.75 * (t_max - t_mid if (t_max - t_min) < 1e-14 else (t_max - t_min))]
    # Fix that line
    candidates = [t_min, t_max,
                  t_min + 0.25 * (t_max - t_min),
                  t_min + 0.5 * (t_max - t_min),
                  t_min + 0.75 * (t_max - t_min)]

    # Find top lags for crossing computation (excluding d=0)
    t_mid = (t_min + t_max) / 2.0
    P_mid = intercepts + slopes * t_mid
    P_mid[d0] = quad_offset + dx * g_quad(t_mid)
    P_min = intercepts + slopes * t_min
    P_min[d0] = quad_offset + dx * g_quad(t_min)
    P_mx = intercepts + slopes * t_max
    P_mx[d0] = quad_offset + dx * g_quad(t_max)

    P_max_val = max(P_mid.max(), P_min.max(), P_mx.max())
    thresh = P_max_val - max(0.003, 0.08 * max(P_max_val - np.median(P_mid), 1e-10))
    top_mask = (P_mid >= thresh) | (P_min >= thresh) | (P_mx >= thresh)
    top_lags = np.where(top_mask)[0]
    top_lags = top_lags[top_lags != d0]
    if len(top_lags) == 0:
        tmp = P_mid.copy()
        tmp[d0] = -np.inf
        top_lags = np.argsort(tmp)[-15:]
    elif len(top_lags) > 35:
        vals = np.maximum(np.maximum(P_mid[top_lags], P_min[top_lags]), P_mx[top_lags])
        idx_s = np.argsort(vals)[-35:]
        top_lags = top_lags[idx_s]

    # Linear crossings
    tl = top_lags
    for i in range(len(tl)):
        for j in range(i + 1, len(tl)):
            di, dj = tl[i], tl[j]
            ds = slopes[di] - slopes[dj]
            if abs(ds) < 1e-15:
                continue
            t_cross = (intercepts[dj] - intercepts[di]) / ds
            if t_min - 1e-10 <= t_cross <= t_max + 1e-10:
                candidates.append(np.clip(t_cross, t_min, t_max))

    # Evaluate all candidates
    best_t = t0
    best_val = eval_at(t0)
    for t_c in candidates:
        t_c = np.clip(t_c, t_min, t_max)
        val = eval_at(t_c)
        if val < best_val - 1e-13:
            best_val = val
            best_t = t_c

    if abs(best_t - t0) > 1e-14:
        h[p] = best_t
        h[q] = s - best_t
        np.clip(h, 0.0, 1.0, out=h)

def pairwise_sweep_ordered(h, t_start=None, budget_s=1100, margin=10):
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    for p in range(n):
        if t_start is not None and time.perf_counter() - t_start > budget_s - margin:
            break
        for q in range(p + 1, n):
            if t_start is not None and time.perf_counter() - t_start > budget_s - margin:
                break
            _pairwise_minimize(h, p, q)
    c = exact_score(h)
    if c < best_c:
        best_c = c
    return h, best_c

def pairwise_sweep_random(h, n_pairs, seed, t_start=None, budget_s=1100, margin=10):
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    rng = np.random.RandomState(seed)
    for _ in range(n_pairs):
        if t_start is not None and time.perf_counter() - t_start > budget_s - margin:
            break
        p = rng.randint(n)
        q = rng.randint(n)
        if p == q:
            continue
        _pairwise_minimize(h, p, q)
    c = exact_score(h)
    if c < best_c:
        best_c = c
    return h, best_c

def slsqp_active(h0, maxiter=300, active_k=12, tol=1e-11):
    n = len(h0)
    dx = 2.0 / n
    M = 2 * n - 1
    P = profile_h(h0)
    c0 = P.max()
    thresh = c0 - max(0.004, 0.06 * (c0 - np.median(P)))
    active = np.where(P >= thresh)[0]
    if len(active) == 0:
        active = np.array([np.argmax(P)])
    if len(active) > active_k:
        top_idx = np.argsort(P[active])[-active_k:]
        active = active[top_idx]

    def build_G(h, lags):
        G = np.zeros((len(lags), n))
        for idx, m in enumerate(lags):
            d = m - (n - 1)
            if d == 0:
                G[idx, :] = dx * (1.0 - 2.0 * h)
            elif d > 0:
                for i in range(n - d):
                    G[idx, i + d] += dx * (1.0 - h[i])
                    G[idx, i] -= dx * h[i + d]
            else:
                dp = -d
                for j in range(n - dp):
                    G[idx, j + dp] += dx * (1.0 - h[j])
                    G[idx, j] -= dx * h[j + dp]
        return G

    def obj(x):
        return x[-1]

    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g

    def eq_fun(x):
        return x[:-1].sum() - n / 2.0

    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g

    def ineq_fun(x):
        h = x[:-1]
        t = x[-1]
        prof = profile_h(h)
        return t - prof[active]

    def ineq_jac(x):
        h = x[:-1]
        G = build_G(h, active)
        J = np.zeros((len(active), n + 1))
        J[:, :-1] = -G
        J[:, -1] = 1.0
        return J

    x0 = np.concatenate([h0, [c0 + 1e-4]])
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 1.0)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': tol})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = repair_feasible(res.x[:-1])
            c_new = exact_score(h_new)
            if c_new <= c0 + 1e-10:
                return h_new, c_new
    except Exception:
        pass
    return None, None

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    return repair_feasible(np.clip(h_new, 0.0, 1.0))

def symmetrize(h):
    return (h + h[::-1].copy()) / 2.0
