from scipy.optimize import minimize, linprog
from scipy.special import logsumexp
import time


def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)

def c5_of_h(h, dx):
    return np.correlate(h, 1.0 - h, mode='full').max() * dx


def grad_corr_k(h, k, n):
    offset = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -offset)
    i_hi = min(n - 1, n - 1 - offset)

    grad[idx] += (-h[idx + offset])
    return grad


def smooth_max_grad(h, dx, beta):
    n = len(h)
    corrs = np.correlate(h, 1.0 - h, mode='full')
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones_like(corrs) / len(corrs)
    else:
        w /= w_sum
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] > 1e-13:
            grad += w[k] * grad_corr_k(h, k, n)
    return grad * dx


def slsqp_smooth(h0, dx, target_sum, beta, maxiter):
    n = len(h0)

    def obj(x):
        corrs = np.correlate(x, 1.0 - x, mode='full')
        return logsumexp(beta * corrs) / beta * dx

    def grad_obj(x):
        return smooth_max_grad(x, dx, beta)

    def c_fun(x):
        return x.sum() - target_sum

    def c_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': c_fun, 'jac': c_jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0, jac=grad_obj, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        h_out = project_feasible(res.x, target_sum)
    except Exception:
        h_out = project_feasible(h0, target_sum)
    return h_out, c5_of_h(h_out, dx)


def lp_direction(h, dx, target_sum, active_k, step_bound):
    """Solve LP to get direction d minimizing max adjusted active corrs."""
    n = len(h)
    corrs = np.correlate(h, 1.0 - h, mode='full')

    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_rows = []
    b_rows = []
    for k in active_k:
        row = np.zeros(n + 1)
        row[:n] = dx * grad_corr_k(h, k, n)
        row[n] = -1.0
        A_rows.append(row)
        b_rows.append(-corrs[k] * dx - 1e-15)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)

    bounds = []
    for i in range(n):
        bounds.append((-step_bound, step_bound))
    bounds.append((0.0, None))

    try:
        res = linprog(c_obj, A_ub=np.array(A_rows), b_ub=np.array(b_rows),
                      A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    except Exception:
        return None
    if res.status != 0:
        return None
    return res.x[:n]

                alpha, f_val = exact_line_search(h_curr, d, dx, n_iter=50)
                if f_val * dx < c5_val - 1e-14:
                    h_new = project_feasible(h_curr + alpha * d, target_sum)
                    new_c5 = c5_of_h(h_new, dx)
                    if new_c5 < best_c5 - 1e-14:
                        h_curr = h_new
                        best_h = h_curr.copy()
                        best_c5 = new_c5
                        corrs = np.correlate(h_curr, 1.0 - h_curr, mode='full')
                        c5_val = corrs.max() * dx
                        improved = True

        if not improved:
            break

    return best_h, best_c5


def make_structured_start(n, target_sum, mode=0, rng=None):
    """Create structured starting vectors."""
    if rng is None:
        rng = np.random.RandomState(0)

    if mode == 0:
        # Smooth cosine-like
        x = np.arange(n)
        h = 0.5 + 0.3 * np.cos(2 * np.pi * x / n * 3 + rng.uniform(0, 2 * np.pi))
        h += 0.1 * np.cos(2 * np.pi * x / n * 7 + rng.uniform(0, 2 * np.pi))
    elif mode == 1:
        # Block pattern
        block = max(2, n // 8)
        h = np.zeros(n)
        for start in range(0, n, block * 2):
            h[start:start + block] = 1.0
    elif mode == 2:
        # Random smooth
        h = rng.beta(2, 2, size=n)
        # Smooth it
        kernel = np.ones(5) / 5.0
        h = np.convolve(h, kernel, mode='same')
    elif mode == 3:
        # Sine mixture
        x = np.arange(n)
        h = 0.5
        for freq in [2, 5, 11]:
            h += 0.15 * np.sin(2 * np.pi * freq * x / n + rng.uniform(0, 2 * np.pi))
    else:
        h = rng.beta(1.5, 1.5, size=n)

    return project_feasible(h, target_sum)


def full_pipeline(h0, dx, target_sum, time_left, label=""):
    """Complete optimization pipeline for a given starting point."""
    n = len(h0)
    best_h = h0.copy()
    best_c5 = c5_of_h(h0, dx)

    # Phase A: SLSQP smoothing
    for beta in [100, 300, 800, 2000, 5000]:
        if time_left() < 20:
            break
        h_s, c5_s = slsqp_smooth(best_h, dx, target_sum, beta, 2000)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s

    # Phase B: LP with line search (primary method)
    if time_left() > 30:
        h_lp, c5_lp = lp_with_linesearch(best_h, dx, target_sum,
                                         max_iters=400, n_active_max=80, init_step=0.12)
        if c5_lp < best_c5:
            best_c5 = c5_lp
            best_h = h_lp

    # Phase C: Multi-direction line search
    if time_left() > 20:
        h_md, c5_md = multi_dir_linesearch(best_h, dx, target_sum, n_dirs=3, n_iter=150)
        if c5_md < best_c5:
            best_c5 = c5_md
            best_h = h_md

    # Phase D: LP again with tighter params
    if time_left() > 20:
        h_lp2, c5_lp2 = lp_with_linesearch(best_h, dx, target_sum,
                                           max_iters=300, n_active_max=50, init_step=0.06)
        if c5_lp2 < best_c5:
            best_c5 = c5_lp2
            best_h = h_lp2

    # Phase E: Pair exact search
    if time_left() > 15:
        h_pt, c5_pt = pair_exact_search(best_h, dx, target_sum, n_top=25, n_rounds=8)
        if c5_pt < best_c5:
            best_c5 = c5_pt
            best_h = h_pt

    # Phase F: Final SLSQP + LP
    if time_left() > 15:
        h_f, c5_f = slsqp_smooth(best_h, dx, target_sum, 8000, 1500)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f
    if time_left() > 10:
        h_lpf, c5_lpf = lp_with_linesearch(best_h, dx, target_sum,
                                           max_iters=200, n_active_max=40, init_step=0.04)
        if c5_lpf < best_c5:
            best_c5 = c5_lpf
            best_h = h_lpf
    if time_left() > 8:
        h_pt2, c5_pt2 = pair_exact_search(best_h, dx, target_sum, n_top=20, n_rounds=5)
        if c5_pt2 < best_c5:
            best_c5 = c5_pt2
            best_h = h_pt2

    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s - (time.time() - t0)

    best_h_global = None
    best_c5_global = float('inf')

    # Use initial construction
    h0 = np.array(initial_h_values, dtype=float)
    n_main = len(h0)
    dx_main = 2.0 / n_main
    ts_main = n_main / 2.0
    h0 = project_feasible(h0, ts_main)

    # Main optimization on initial_h_values
    h_main, c5_main = full_pipeline(h0, dx_main, ts_main, time_left, "main")
    best_h_global = h_main
    best_c5_global = c5_main

    # Try perturbations of the best solution
    n_perturb = min(8, max(3, int(time_left() / 60)))
    for trial in range(n_perturb):
        if time_left() < 30:
            break
        scale = 0.03 + 0.02 * trial
        perturb = scale * (rng.random(n_main) - 0.5)
        h_try = project_feasible(best_h_global + perturb, ts_main)
        h_opt, c5_opt = full_pipeline(h_try, dx_main, ts_main, time_left, f"pert{trial}")
        if c5_opt < best_c5_global:
            best_c5_global = c5_opt
            best_h_global = h_opt

    # Try other n values
    n_candidates = []
    for nn in range(60, 300):
        if nn % 2 == 0:
            n_candidates.append(nn)
    n_candidates.sort(key=lambda x: abs(x - n_main))
    n_candidates = n_candidates[:20]

    for n_try in n_candidates:
        if time_left() < 40:
            break
        if n_try == n_main:
            continue
        dx_try = 2.0 / n_try
        ts_try = n_try / 2.0

        # Interpolate from best solution
        x_orig = np.linspace(0.0, 2.0, len(best_h_global), endpoint=False) + 1.0 / len(best_h_global)
        x_new = np.linspace(0.0, 2.0, n_try, endpoint=False) + 1.0 / n_try
        h_interp = np.interp(x_new, x_orig, best_h_global)
        h_interp = project_feasible(h_interp, ts_try)

        h_opt, c5_opt = full_pipeline(h_interp, dx_try, ts_try, time_left, f"n{n_try}")
        if c5_opt < best_c5_global:
            best_c5_global = c5_opt
            best_h_global = h_opt

    # Try structured starts at the best n
    n_best = len(best_h_global)
    dx_best = 2.0 / n_best
    ts_best = n_best / 2.0

    for mode in range(4):
        if time_left() < 30:
            break
        h_struct = make_structured_start(n_best, ts_best, mode=mode, rng=rng)
        h_opt, c5_opt = full_pipeline(h_struct, dx_best, ts_best, time_left, f"struct{mode}")
        if c5_opt < best_c5_global:
            best_c5_global = c5_opt
            best_h_global = h_opt

    # Final intensive polish
    if time_left() > 20:
        n_f = len(best_h_global)
        dx_f = 2.0 / n_f
        ts_f = n_f / 2.0
        h_f, c5_f = full_pipeline(best_h_global, dx_f, ts_f, time_left, "final")
        if c5_f < best_c5_global:
            best_c5_global = c5_f
            best_h_global = h_f

    # Ultra-fine final step
    if time_left() > 10 and best_h_global is not None:
        n_f = len(best_h_global)
        dx_f = 2.0 / n_f
        ts_f = n_f / 2.0
        h_pt, c5_pt = pair_exact_search(best_h_global, dx_f, ts_f, n_top=40, n_rounds=10)
        if c5_pt < best_c5_global:
            best_c5_global = c5_pt
            best_h_global = h_pt
        h_lp_final, c5_lp_final = lp_with_linesearch(best_h_global, dx_f, ts_f,
                                                      max_iters=100, n_active_max=30, init_step=0.02)
        if c5_lp_final < best_c5_global:
            best_c5_global = c5_lp_final
            best_h_global = h_lp_final

    # Ensure feasibility
    if best_h_global is None:
        best_h_global = h0
        best_c5_global = c5_of_h(h0, 2.0 / len(h0))
    else:
        n_final = len(best_h_global)
        best_h_global = project_feasible(best_h_global, n_final / 2.0)
        best_c5_global = c5_of_h(best_h_global, 2.0 / n_final)

    return best_h_global, best_c5_global, len(best_h_global)
