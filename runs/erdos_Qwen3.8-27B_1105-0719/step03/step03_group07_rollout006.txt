import numpy as np
from scipy.optimize import minimize
import time


def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full"))) * dx


def profile_of(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx


def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx])
            h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h


def grad_for_lag(h, d):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    if d == 0:
        g[:] = (1.0 - 2.0 * h) * dx
    elif d > 0:
        for i in range(n - d):
            g[i] -= h[i + d] * dx
            g[i + d] += (1.0 - h[i]) * dx
    else:
        dp = -d
        for i in range(n - dp):
            g[i] += (1.0 - h[i + dp]) * dx
            g[i + dp] -= h[i] * dx
    return g


def adaptive_subgradient(h, K_min=8, K_max=20):
    n = len(h)
    P = profile_of(h)
    c = P.max()
    idx_sorted = np.argsort(-P)
    w = np.zeros(len(P))
    for k in range(K_min, min(K_max, len(P))):
        m = idx_sorted[k]
        d = m - (n - 1)
        w[m] = np.exp(-0.5 * (P[m] - c) / max(c * 0.01, 1e-6))
    w_sum = w.sum()
    if w_sum < 1e-30:
        w[idx_sorted[:K_min]] = 1.0 / K_min
        w_sum = 1.0
    else:
        w /= w_sum
    g = np.zeros(n)
    active = idx_sorted[:K_max]
    for m in active:
        if w[m] > 1e-6:
            d = m - (n - 1)
            g += w[m] * grad_for_lag(h, d)
    return g, c, P


def project_direction(h, g):
    n = len(h)
    d = -g.copy()
    d -= d.mean()
    for i in range(n):
        if h[i] <= 1e-14 and d[i] < 0:
            d[i] = 0.0
        elif h[i] >= 1.0 - 1e-14 and d[i] > 0:
            d[i] = 0.0
    d -= d.mean()
    norm = np.linalg.norm(d)
    if norm < 1e-15:
        return d
    return d / norm


def line_search_exact(h, direction, c_curr, max_alpha=0.3):
    n = len(h)
    alphas = []
    for i in range(n):
        if abs(direction[i]) > 1e-14:
            if direction[i] > 0:
                alphas.append((1.0 - h[i] + 1e-8) / direction[i])
            else:
                alphas.append((h[i] - 1e-8) / (-direction[i]))
    alpha_max = min(min(alphas) * 0.9, max_alpha) if alphas else max_alpha
    if alpha_max < 1e-8:
        return h, c_curr, 0.0
    best_alpha = 0.0
    best_c = c_curr
    for t in np.linspace(0.02 * alpha_max, alpha_max, 16):
        h_try = repair_feasible(h + t * direction)
        c_try = exact_score(h_try)
        if c_try < best_c:
            best_c = c_try
            best_alpha = t
    if best_alpha < 0.1 * alpha_max:
        for t in np.linspace(1e-6, 0.02 * alpha_max, 8):
            h_try = repair_feasible(h + t * direction)
            c_try = exact_score(h_try)
            if c_try < best_c:
                best_c = c_try
                best_alpha = t
    if best_alpha == 0.0:
        return h, c_curr, 0.0
    return repair_feasible(h + best_alpha * direction), best_c, best_alpha

def slsqp_polish(h0, maxiter=300, active_k=12):
    n = len(h0)
    dx = 2.0 / n
    P = profile_of(h0)
    c0 = P.max()
    thresh = c0 - max(0.002, 0.08 * (c0 - np.median(P)))
    active = np.where(P >= thresh)[0]
    if len(active) == 0:
        active = np.array([np.argmax(P)])
    if len(active) > active_k:
        top_idx = np.argsort(P[active])[-active_k:]
        active = active[top_idx]

    def obj_fun(x):
        return x[-1]

    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g

    def eq_fun(x):
        return x[:-1].sum() - n / 2.0

    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g

    def ineq_fun(x):
        h = x[:-1]
        t = x[-1]
        prof = profile_of(h)
        return t - prof[active]

    def ineq_jac(x):
        h = x[:-1]
        J = np.zeros((len(active), n + 1))
        for idx, m in enumerate(active):
            d = m - (n - 1)
            g = grad_for_lag(h, d)
            J[idx, :-1] = -g
            J[idx, -1] = 1.0
        return J

    x0 = np.concatenate([h0, [c0 + 1e-4]])
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 1.0)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    try:
        res = minimize(obj_fun, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = repair_feasible(res.x[:-1])
            c_new = exact_score(h_new)
            if c_new <= c0 + 1e-10:
                return h_new
    except Exception:
        pass
    return None


def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    return repair_feasible(np.clip(h_new, 0.0, 1.0))

                        best_c = c_sol

    if time.perf_counter() - t_start < budget_s - 60:
        for mode in range(4):
            if elapsed_check():
                break
            h_struct = make_structured_start(n, rng, mode)
            for mix in [0.3, 0.5]:
                if elapsed_check():
                    break
                h_mix = repair_feasible(mix * h_struct + (1 - mix) * best_h)
                h_ph, c_ph = projected_gradient_phase(h_mix, t_start, budget_s, margin=50)
                if c_ph < best_c - 1e-10:
                    best_h = h_ph
                    best_c = c_ph

    if time.perf_counter() - t_start < budget_s - 40:
        sol = slsqp_polish(best_h, maxiter=250, active_k=15)
        if sol is not None:
            c_sol = exact_score(sol)
            if c_sol < best_c - 1e-13:
                best_h = sol
                best_c = c_sol

    if time.perf_counter() - t_start < budget_s - 20:
        best_h, best_c = coordinate_polish(best_h, max_sweeps=6, t_start=t_start, budget_s=budget_s)

    if time.perf_counter() - t_start < budget_s - 15:
        sol = slsqp_polish(best_h, maxiter=200, active_k=10)
        if sol is not None:
            c_sol = exact_score(sol)
            if c_sol < best_c - 1e-14:
                best_h = sol
                best_c = c_sol

    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)

    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            final_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 1:
            if np.isfinite(eval_result[0]):
                final_c = float(eval_result[0])
    except Exception:
        pass

    return best_h, final_c, len(best_h)
