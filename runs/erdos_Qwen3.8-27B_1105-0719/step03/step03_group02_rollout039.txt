    current_c5 = corrs.max() * dx

    n = len(h0)

def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)


def c5_of_h(h, dx):
    return np.correlate(h, 1.0 - h, mode='full').max() * dx


def compute_effects(h, n):
    corrs_len = 2 * n - 1
    effects = np.zeros((n, corrs_len))
    one_minus_h = 1.0 - h
    rev_h = h[::-1]
    for a in range(n):
        effects[a, n - 1 - a:2 * n - 1 - a] += one_minus_h
        effects[a, a:a + n] -= rev_h
    return effects


def binary_best_swap(h_bin, dx, max_iters=5000, time_fn=None, time_limit=None):

    effects = compute_effects(h, n)

    zeros = np.where(h == 0.0)[0]

def epigraph_optimize(h0, dx, target_sum, maxiter=300, n_active=60):
    n = len(h0)
    corrs_init = np.correlate(h0, 1.0 - h0, mode='full')
    max_c = corrs_init.max()
    active_k = np.argsort(corrs_init)[-n_active:]

    def objective(z):
        return z[n]

    def grad_objective(z):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g

    def ineq_fun(z):
        h = z[:n]
        tau = z[n]
        c_h = np.correlate(h, 1.0 - h, mode='full')
        return tau - c_h[active_k] * dx

    def ineq_jac(z):
        h = z[:n]
        rows = []
        for k in active_k:
            s = k - (n - 1)
            i_lo = max(0, -s)
            i_hi = min(n - 1, n - 1 - s)
            row = np.zeros(n + 1)
            row[n] = 1.0
            if i_lo <= i_hi:
                idx = np.arange(i_lo, i_hi + 1)
                idx2 = idx + s
                row[idx] += -(1.0 - h[idx2]) * dx
                row[idx2] += -h[idx] * dx
            rows.append(row)
        return np.array(rows)

    def eq_fun(z):
        return z[:n].sum() - target_sum

    def eq_jac(z):
        v = np.zeros(n + 1)
        v[:n] = 1.0
        return v

    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    z0 = np.concatenate([h0, [max_c * dx + 1e-4]])

    try:
        res = minimize(objective, z0, jac=grad_objective, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        h_out = project_feasible(np.clip(res.x[:n], 0, 1), target_sum)
    except Exception:
        h_out = project_feasible(h0, target_sum)
    return h_out, c5_of_h(h_out, dx)

    # Phase 2: Iterated local search with annealing
    for cycle in range(8):
        if time_left() < 30:
            break
        # Anneal from best
        h_an, c5_an = binary_anneal(best_h, dx, t0=0.3, t1=0.003,
                                     max_iters=4000, rng=rng,
                                     time_fn=time_left, time_limit=15)
        if c5_an < best_c5:
            best_c5 = c5_an
            best_h = h_an
        # Best-swap from result
        h_sw, c5_sw = binary_best_swap(best_h, dx, max_iters=5000,
                                        time_fn=time_left, time_limit=8)
        if c5_sw < best_c5:
            best_c5 = c5_sw
            best_h = h_sw

    # Phase 3: Multi-swap for deeper local search
    if time_left() > 20:
        h_ms, c5_ms = multi_swap_descent(best_h, dx, max_sweeps=30,
                                          time_fn=time_left, time_limit=10)
        if c5_ms < best_c5:
            best_c5 = c5_ms
            best_h = h_ms

    # Phase 4: Try different n values
    n_candidates = []
    for offset in [0, 2, -2, 4, -4, 6, -6, 8, -8]:
        nn = n + offset
        if 80 <= nn <= 200 and nn % 2 == 0:
            n_candidates.append(nn)
    n_candidates = sorted(set(n_candidates))

    for n_try in n_candidates:
        if time_left() < 25:
            break
        if n_try == n:
            continue
        dx_try = 2.0 / n_try
        ts_try = n_try / 2.0
        x_orig = np.linspace(0.0, 2.0, len(best_h), endpoint=False) + 1.0 / len(best_h)
        x_new = np.linspace(0.0, 2.0, n_try, endpoint=False) + 1.0 / n_try
        h_interp = np.interp(x_new, x_orig, best_h)
        h_bin_try = binarize_smart(h_interp, n_try)
        h_sw, c5_sw = binary_best_swap(h_bin_try, dx_try, max_iters=5000,
                                        time_fn=time_left, time_limit=10)
        if c5_sw < best_c5:
            best_c5 = c5_sw
            best_h = h_sw
            n = n_try
            dx = dx_try
            target_sum = ts_try
            h_an2, c5_an2 = binary_anneal(best_h, dx, t0=0.2, t1=0.002,
                                           max_iters=2000, rng=rng,
                                           time_fn=time_left, time_limit=8)
            if c5_an2 < best_c5:
                best_c5 = c5_an2
                best_h = h_an2
            h_sw2, c5_sw2 = binary_best_swap(best_h, dx, max_iters=3000,
                                              time_fn=time_left, time_limit=5)
            if c5_sw2 < best_c5:
                best_c5 = c5_sw2
                best_h = h_sw2

    # Phase 5: Continuous epigraph optimization
    if time_left() > 30:
        h_cont = best_h.astype(float)
        h_opt, c5_opt = epigraph_optimize(h_cont, dx, target_sum, maxiter=400, n_active=80)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 6: Final rounds of binary search from best
    for cycle in range(4):
        if time_left() < 10:
            break
        h_b = binarize_smart(best_h, len(best_h))
        h_sw, c5_sw = binary_best_swap(h_b, dx, max_iters=5000,
                                        time_fn=time_left, time_limit=5)
        if c5_sw < best_c5:
            best_c5 = c5_sw
            best_h = h_sw
        h_an, c5_an = binary_anneal(best_h, dx, t0=0.15, t1=0.001,
                                     max_iters=2000, rng=rng,
                                     time_fn=time_left, time_limit=5)
        if c5_an < best_c5:
            best_c5 = c5_an
            best_h = h_an
        h_sw2, c5_sw2 = binary_best_swap(best_h, dx, max_iters=3000,
                                          time_fn=time_left, time_limit=3)
        if c5_sw2 < best_c5:
            best_c5 = c5_sw2
            best_h = h_sw2

    # Final polish: multi-swap
    if time_left() > 5:
        h_ms, c5_ms = multi_swap_descent(best_h, dx, max_sweeps=10,
                                          time_fn=time_left, time_limit=3)
        if c5_ms < best_c5:
            best_c5 = c5_ms
            best_h = h_ms

    # Ensure feasibility
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final / 2.0)
    best_c5 = c5_of_h(best_h, 2.0 / n_final)

    return best_h, best_c5, n_final
