def project_feasible(x, target_sum):

def get_pair_A(h, p, q, n):
    """Compute coefficient vector A such that new_corr = old_corr + A*t for pair move h[p]+=t, h[q]-=t"""
    A = np.zeros(2 * n - 1)
    hp = h[p]
    hq = h[q]
    A[n - 1] = 2.0 * (hq - hp)
    
    d = np.arange(1, n)
    # corr[n-1-d] indices: 0 to n-2
    idx_neg = n - 1 - d  # = 0, 1, ..., n-2
    
    # p as first index in corr[n-1-d]: need p + d < n
    mp = d < (n - p)
    if mp.any():
        A[idx_neg[mp]] += (1.0 - h[p + d[mp]])
    # p as second index: need p - d >= 0
    mp2 = d <= p
    if mp2.any():
        A[idx_neg[mp2]] -= h[p - d[mp2]]
    # q as first index: need q + d < n
    mq = d < (n - q)
    if mq.any():
        A[idx_neg[mq]] -= (1.0 - h[q + d[mq]])
    # q as second index: need q - d >= 0
    mq2 = d <= q
    if mq2.any():
        A[idx_neg[mq2]] += h[q - d[mq2]]
    
    # corr[n-1+d] indices: n to 2n-2
    idx_pos = n - 1 + d  # = n, n+1, ..., 2n-2
    
    # p as second index: need p-d >= 0 and p-d < n-d (i.e., p < n, always true)
    mp3 = d <= p
    if mp3.any():
        A[idx_pos[mp3]] += (1.0 - h[p - d[mp3]])
    # p as first index: need p < n-d (i.e., d < n-p)
    mp4 = d < (n - p)
    if mp4.any():
        A[idx_pos[mp4]] -= h[p + d[mp4]]
    # q as second index: need q-d >= 0
    mq3 = d <= q
    if mq3.any():
        A[idx_pos[mq3]] -= (1.0 - h[q - d[mq3]])
    # q as first index: need q < n-d
    mq4 = d < (n - q)
    if mq4.any():
        A[idx_pos[mq4]] += h[q + d[mq4]]
    
    return A

def solve_1d_lp(A, corr, L, U, dx):

        t = (L + U) / 2.0
        return t, dx * np.max(A * t + corr)
    
    # Bisection on subgradient

        else:
            break

    mid = (lo + hi) / 2.0

def pair_sweep(h, dx, target_sum, n, rng, max_pairs=-1):
    """One sweep of pair moves. Returns (h_new, c5_new, n_improvements)."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    cur_c5 = corr.max() * dx
    best_c5 = cur_c5
    best_h = h.copy()
    n_imp = 0
    
    # Get indices of non-binary coordinates
    non_bin = np.where((h > 1e-6) & (h < 1.0 - 1e-6))[0]
    
    # Order: try pairs involving non-binary coords first
    if len(non_bin) > 0 and max_pairs > 0:
        # Create pair list prioritizing non-bin pairs
        pairs = []
        for i in non_bin:
            for j in range(n):
                if j != i:
                    pairs.append((i, j))
        # Shuffle
        rng.shuffle(pairs)
        if max_pairs > 0:
            pairs = pairs[:max_pairs]
    else:
        pairs = [(i, j) for i in range(n) for j in range(n) if j != i]
        rng.shuffle(pairs)
        if max_pairs > 0:
            pairs = pairs[:max_pairs]
    
    for p, q in pairs:
        hp = best_h[p]
        hq = best_h[q]
        # Feasible interval for t
        L = max(-hp, hq - 1.0)
        U = min(1.0 - hp, hq)
        if U - L < 1e-12:
            continue
        
        A = get_pair_A(best_h, p, q, n)
        t_opt, f_opt = solve_1d_lp(A, corr, L, U, dx)
        
        if f_opt < cur_c5 - 1e-14:
            # Apply the move
            best_h[p] += t_opt
            best_h[q] -= t_opt
            # Update corr
            corr += A * t_opt
            # Fix numerical drift
            if best_h[p] < 0: best_h[p] = 0
            if best_h[p] > 1: best_h[p] = 1
            if best_h[q] < 0: best_h[q] = 0
            if best_h[q] > 1: best_h[q] = 1
            cur_c5 = corr.max() * dx

            n_imp += 1

    best_h = project_feasible(best_h, target_sum)
    final_c5 = c5_of_h(best_h, dx)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    return best_h, best_c5, n_imp

def smooth_gradient_step(h, q, dx, target_sum, n, step):
    """One gradient step on the smooth-sum surrogate sum(corr^q)."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    maxc = max(corr.max(), 1e-30)
    # Stable weights: w_k = (corr_k / maxc)^(q-1)
    ratio = np.maximum(corr / maxc, 0.0)
    w = ratio ** (q - 1)
    
    # Compute gradient of sum(corr_k * w_k) w.r.t. h
    # (This is the gradient of the smooth sum, ignoring the q*corr_k^(q-1) factor
    #  since w already contains it up to a constant)
    g = np.zeros(n)
    
    # d=0: corr[n-1] = sum h[i]*(1-h[i])
    # d/dh[i] = 1 - 2*h[i]
    w0 = w[n-1]
    if w0 > 0:
        g += w0 * (1.0 - 2.0 * h)
    
    # d >= 1
    d = np.arange(1, n)
    K = n - d
    for idx in range(len(d)):
        di = d[idx]
        Ki = K[idx]
        wp = w[n-1-di]  # weight for corr[n-1-d]
        wm = w[n-1+di]  # weight for corr[n-1+d]
        if abs(wp) < 1e-15 and abs(wm) < 1e-15:
            continue
        u = h[:Ki]
        v = h[di:n]
        if abs(wp) > 1e-15:
            g[:Ki] += wp * (1.0 - v)
            g[di:] -= wp * u
        if abs(wm) > 1e-15:
            g[:Ki] -= wm * v
            g[di:] += wm * (1.0 - u)
    
    # Remove mean (to preserve sum constraint to first order)
    g -= g.mean()
    
    # Project onto feasible set
    h_new = project_feasible(h - step * g, target_sum)
    return h_new
