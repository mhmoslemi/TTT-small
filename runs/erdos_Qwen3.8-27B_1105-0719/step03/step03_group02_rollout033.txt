def c5_of_h(h, dx):
    return float(compute_corrs(h).max() * dx)

def grad_corr_k(h, k, n):
    """Gradient of corr_k w.r.t. h, where k is the index into the full correlation array"""
    offset = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -offset)
    i_hi = min(n - 1, n - 1 - offset)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx + offset] += (1.0 - h[idx])
    grad[idx] += (-h[idx + offset])
    return grad

def lp_direction(h, dx, target_sum, n_active, step_bound):
    """Find a direction using LP linearization of the active constraints"""
    n = len(h)
    corrs = compute_corrs(h)
    
    # Select active constraints (top n_active correlations)
    top_idx = np.argsort(corrs)[-n_active:]
    
    # Build LP: min t s.t. corr_k + grad_k . d <= t, sum(d) = 0, |d_i| <= step_bound
    # Variables: d[0..n-1], t
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    A_rows = []
    b_rows = []
    for k in top_idx:
        row = np.zeros(n + 1)
        row[:n] = dx * grad_corr_k(h, k, n)
        row[n] = -1.0
        A_rows.append(row)
        b_rows.append(-corrs[k] * dx - 1e-12)

    bounds.append((0.0, None))

import numpy as np
from scipy.optimize import linprog
import time


def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)


def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def c5_of_h(h, dx):
    return float(compute_corrs(h).max() * dx)


def compute_A(h):
    """Compute A matrix: A[k, i] = coefficient of h[i] in corr_k"""
    n = len(h)
    ncorr = 2 * n - 1
    A = np.zeros((ncorr, n))
    for k in range(ncorr):
        offset = k - (n - 1)
        i_lo = max(0, -offset)
        i_hi = min(n - 1, n - 1 - offset)
        if i_lo > i_hi:
            continue
        idx = np.arange(i_lo, i_hi + 1)
        A[k, idx + offset] += (1.0 - h[idx])
        A[k, idx] -= h[idx + offset]
    return A


def paired_cd(h0, dx, n_sweeps=200, rng=None):
    """Paired coordinate descent with exact 1D subgradient search."""
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    corrs = compute_corrs(h)
    
    if rng is None:
        rng = np.random.RandomState(123)
    
    for sweep in range(n_sweeps):
        A = compute_A(h)
        improved = False
        n_improvements = 0
        
        # Randomize pair order each sweep
        pairs = []
        for i in range(n):
            for j in range(i + 1, n):
                pairs.append((i, j))
        rng.shuffle(pairs)
        
        for (i, j) in pairs:
            a_vec = A[:, i] - A[:, j]
            max_abs_a = np.max(np.abs(a_vec))
            if max_abs_a < 1e-15:
                continue
            t_lo = max(-h[i], h[j] - 1.0)
            t_hi = min(1.0 - h[i], h[j])
            if t_hi - t_lo < 1e-12:
                continue
            
            # Subgradient binary search for min of max_k(corrs + t*a_vec)
            lo, hi = t_lo, t_hi
            best_t = 0.0
            best_f = corrs.max()
            
            for _ in range(28):
                mid = (lo + hi) * 0.5
                f_mid = (corrs + mid * a_vec).max()
                if f_mid < best_f:
                    best_f = f_mid
                    best_t = mid
                max_val = (corrs + mid * a_vec).max()
                active_mask = np.abs(corrs + mid * a_vec - max_val) < 1e-10
                if active_mask.any():
                    a_act = a_vec[active_mask]
                    if a_act.min() > 1e-12:
                        hi = mid
                    elif a_act.max() < -1e-12:
                        lo = mid
                    else:
                        best_t = mid
                        best_f = f_mid
                        break
                else:
                    break
            
            if best_f < corrs.max() - 1e-13:
                h[i] += best_t
                h[j] -= best_t
                corrs += best_t * a_vec
                improved = True
                n_improvements += 1
                if corrs.max() * dx < best_c5:
                    best_c5 = corrs.max() * dx
                    best_h = h.copy()
        
        if not improved:
            break
    
    # Final projection and verification
    best_h = project_feasible(best_h, n / 2.0)
    best_c5 = c5_of_h(best_h, dx)
    return best_h, best_c5


def lp_direction(h, dx, n_active, step_bound):
    """Find LP direction for active constraints."""
    n = len(h)
    corrs = compute_corrs(h)
    A = compute_A(h)
    
    top_idx = np.argsort(corrs)[-n_active:]
    
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_rows = []
    b_rows = []
    for k in top_idx:
        row = np.zeros(n + 1)
        row[:n] = dx * A[k, :]
        row[n] = -1.0
        A_rows.append(row)
        b_rows.append(-corrs[k] * dx - 1e-12)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    bounds = []
    for i in range(n):
        bounds.append((max(-h[i], -step_bound), min(1.0 - h[i], step_bound)))
    bounds.append((0.0, None))
    
    try:
        res = linprog(c_obj, A_ub=np.array(A_rows), b_ub=np.array(b_rows),
                      A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    except Exception:
        return None, None
    if res.status != 0:
        return None, None
    return res.x[:n], res.x[-1] - corrs.max() * dx


def line_search_alpha(h, d, dx, alpha_max=1.0):
    """Find best alpha in [0, alpha_max] for h + alpha*d."""
    n_samples = 25
    alphas = np.linspace(0, alpha_max, n_samples)
    vals = np.empty(n_samples)
    for idx, a in enumerate(alphas):
        vals[idx] = c5_of_h(h + a * d, dx)
    best_idx = np.argmin(vals)
    if best_idx == 0:
        return 0.0, vals[0]
    
    a_lo = alphas[max(0, best_idx - 1)]
    a_hi = alphas[min(n_samples - 1, best_idx + 1)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c_pt = a_hi - gr * (a_hi - a_lo)
    d_pt = a_lo + gr * (a_hi - a_lo)
    fc = c5_of_h(h + c_pt * d, dx)
    fd = c5_of_h(h + d_pt * d, dx)
    for _ in range(35):
        if a_hi - a_lo < 1e-11:
            break
        if fc <= fd:
            a_hi = d_pt
            d_pt = c_pt
            fd = fc
            c_pt = a_hi - gr * (a_hi - a_lo)
            fc = c5_of_h(h + c_pt * d, dx)
        else:
            a_lo = c_pt
            c_pt = d_pt
            fc = fd
            d_pt = a_lo + gr * (a_hi - a_lo)
            fd = c5_of_h(h + d_pt * d, dx)
    alpha_opt = (a_lo + a_hi) / 2.0
    return alpha_opt, c5_of_h(h + alpha_opt * d, dx)
