import numpy as np
from scipy.optimize import linprog
import time


def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)


def compute_c5(h, dx):
    corrs = np.correlate(h, 1.0 - h, mode='full')
    return corrs.max() * dx


def min_max_affine(A, B, t_lo=0.0, t_hi=1.0):
    """Find t in [t_lo, t_hi] minimizing max_k (A[k] + B[k]*t)."""
    if t_hi - t_lo < 1e-15:
        return 0.5 * (t_lo + t_hi)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = t_lo, t_hi
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = np.max(A + B * c)
    fd = np.max(A + B * d)
    for _ in range(80):
        if b - a < 1e-15:
            break
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = np.max(A + B * c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = np.max(A + B * d)
    return 0.5 * (a + b)


def coord_descent_sweep(h, n, order=None):
    """One full sweep of exact coordinate descent. Returns improved h."""
    h = h.copy()
    if order is None:
        order = np.arange(n)
    for p in order:
        h_saved = h[p]
        h[p] = 0.0
        corrs = np.correlate(h, 1.0 - h, mode='full')
        h[p] = h_saved
        # Compute B_k
        B = np.zeros(2 * n - 1)
        for k in range(2 * n - 1):
            off = k - (n - 1)
            b_val = 0.0
            ip_off = p + off
            if 0 <= ip_off < n:
                b_val += (1.0 - h[ip_off])
            im_off = p - off
            if 0 <= im_off < n:
                b_val -= h[im_off]
            B[k] = b_val
        t_star = min_max_affine(corrs, B, 0.0, 1.0)
        h[p] = t_star
    return h


def pair_transfer_sweep(h, n, max_pairs=5000, rng=None):
    """Exact pair transfer: move mass between pairs of coordinates."""
    h = h.copy()
    best_c5 = compute_c5(h, 1.0)
    n_pairs_total = n * (n - 1) // 2
    if rng is not None:
        pairs = rng.randint(0, n_pairs_total, size=min(max_pairs, n_pairs_total))
    else:
        pairs = np.arange(min(max_pairs, n_pairs_total))
    improved_any = False
    for pi in pairs:
        # Convert linear index to (i, j) with i < j
        # Number of pairs with first index < i is i*(i-1)/2 + (n-1-i)
        # Use inverse: for pair index pi, find i,j
        # i(i-1)/2 + (n-1-i) <= pi
        # Approximate: i ≈ sqrt(2*pi)
        i = int(np.floor((1 + np.sqrt(1 + 8.0 * pi)) / 2.0))
        while i * (i - 1) // 2 + (n - 1 - i) > pi and i > 0:
            i -= 1
        while i * (i - 1) // 2 + (n - 1 - i) < pi and i < n - 1:
            i += 1
        offset_in_row = pi - i * (i - 1) // 2
        j = i + 1 + offset_in_row
        if j >= n:
            continue
        # h[i] += t, h[j] -= t
        # t range: h[i]+t in [0,1], h[j]-t in [0,1]
        t_lo = max(-h[i], h[j] - 1.0)
        t_hi = min(1.0 - h[i], h[j])
        if t_hi - t_lo < 1e-13:
            continue
        # Compute A and B for the pair direction
        # Set h[i]=0, h[j]=0 temporarily to get base
        hi_saved, hj_saved = h[i], h[j]
        h[i] = 0.0
        h[j] = 0.0
        corrs = np.correlate(h, 1.0 - h, mode='full')
        h[i] = hi_saved
        h[j] = hj_saved
        # B_k = contribution from h[i] increasing by t + contribution from h[j] decreasing by t
        B = np.zeros(2 * n - 1)
        for k in range(2 * n - 1):
            off = k - (n - 1)
            b_val = 0.0
            # h[i] += t
            ip_off = i + off
            if 0 <= ip_off < n:
                b_val += (1.0 - h[ip_off])
            im_off = i - off
            if 0 <= im_off < n:
                b_val -= h[im_off]
            # h[j] -= t, so effective sign is -1
            jp_off = j + off
            if 0 <= jp_off < n:
                b_val -= (1.0 - h[jp_off])
            jm_off = j - off
            if 0 <= jm_off < n:
                b_val += h[jm_off]
            B[k] = b_val
        t_star = min_max_affine(corrs, B, t_lo, t_hi)
        h[i] += t_star
        h[j] -= t_star
        h[i] = np.clip(h[i], 0.0, 1.0)
        h[j] = np.clip(h[j], 0.0, 1.0)
        new_c5 = compute_c5(h, 1.0)
        if new_c5 < best_c5 - 1e-15:
            best_c5 = new_c5
            improved_any = True
    return h, best_c5, improved_any


def lp_polish(h0, dx, target_sum, max_iters=100, init_step=0.12):
    """Active-set LP polish."""
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    step = init_step
    stall = 0
    for it in range(max_iters):
        corrs = np.correlate(h, 1.0 - h, mode='full')
        c5_val = corrs.max() * dx
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 30:
            step *= 0.6
            stall = 0
            if step < 1e-7:
                break
        max_c = corrs.max()
        tol_val = max_c * 0.95
        active_k = np.where(corrs >= tol_val)[0]
        if len(active_k) > 60:
            active_k = np.argsort(corrs)[-60:]
        # Build LP
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        A_rows = []
        b_rows = []
        for k in active_k:
            off = k - (n - 1)
            i_lo = max(0, -off)
            i_hi = min(n - 1, n - 1 - off)
            idx = np.arange(i_lo, i_hi + 1)
            row = np.zeros(n + 1)
            # grad_corr_k: d/delta[i] of corr_k = (1-h[i+off]) - h[i-off]
            # For i in valid range: d[i] appears as:
            #   as h[i]*(1-h[i+off]): derivative w.r.t. delta[i] is (1-h[i+off])
            #   as h[i-off]*(1-h[i]): derivative w.r.t. delta[i] is -h[i-off]
            for ii in idx:
                dval = 0.0
                if 0 <= ii + off < n:
                    dval += (1.0 - h[ii + off])
                if 0 <= ii - off < n:
                    dval -= h[ii - off]
                row[ii] = dx * dval
            row[n] = -1.0
            A_rows.append(row)
            b_rows.append(-corrs[k] * dx - 1e-14)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        bounds = [(max(-h[i2], -step), min(1.0 - h[i2], step)) for i2 in range(n)]
        bounds.append((0.0, None))
        try:
            res = linprog(c_obj, A_ub=np.array(A_rows), b_ub=np.array(b_rows),
                          A_eq=A_eq, b_eq=np.zeros(1), bounds=bounds, method='highs')
        except Exception:
            break
        if res.status != 0:
            break
        delta = res.x[:n]
        if np.max(np.abs(delta)) < 1e-14:
            break
        h_new = project_feasible(h + delta, target_sum)
        new_c5 = compute_c5(h_new, dx)
        if new_c5 < c5_val - 1e-14:
            h = h_new
            step = min(step * 1.15, init_step)
        else:

            for frac in [0.5, 0.25, 0.1]:

                if compute_c5(h_try, dx) < c5_val - 1e-14:
                    h = h_try
                    improved = True
                    break
            if not improved:
                step *= 0.5
                if step < 1e-8:
                    break
    return best_h, best_c5

                if not improved:
                    break

    # Final validation
    if best_h is None:
        best_h = h0
        best_c5 = compute_c5(h0, 2.0 / len(h0))
    else:
        n_final = len(best_h)
        best_h = project_feasible(best_h, n_final / 2.0)
        best_c5 = compute_c5(best_h, 2.0 / n_final)

    return best_h, best_c5, len(best_h)
