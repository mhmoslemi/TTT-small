    try:

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.special import logsumexp
import time


def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)


def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def c5_of_h(h, dx):
    return compute_corrs(h).max() * dx


def grad_corr_k(h, k, n):
    offset = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -offset)
    i_hi = min(n - 1, n - 1 - offset)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx + offset] += (1.0 - h[idx])
    grad[idx] += (-h[idx + offset])
    return grad


def smooth_max_obj(h, dx, beta):
    corrs = compute_corrs(h)
    return logsumexp(beta * corrs) / beta * dx


def smooth_max_grad(h, dx, beta):
    n = len(h)
    corrs = compute_corrs(h)
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones_like(corrs) / len(corrs)
    else:
        w /= w_sum
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] > 1e-13:
            grad += w[k] * grad_corr_k(h, k, n)
    return grad * dx


def target_penalty_obj(h, dx, beta, theta):
    corrs = compute_corrs(h) * dx
    excess = beta * (corrs - theta)
    sp = np.where(excess > 0, excess + np.log1p(np.exp(-excess)), np.log1p(np.exp(excess)))
    return sp.sum()


def target_penalty_grad(h, dx, beta, theta):
    n = len(h)
    corrs = compute_corrs(h) * dx
    excess = beta * (corrs - theta)
    sig = 1.0 / (1.0 + np.exp(-np.clip(excess, -500, 500)))
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if sig[k] > 1e-14:
            grad += sig[k] * grad_corr_k(h, k, n)
    return grad * dx


def trust_region_lp(h0, dx, target_sum, radius, n_active=50):
    n = len(h0)
    corrs = compute_corrs(h0)
    top_idx = np.argsort(corrs)[-n_active:]
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_rows = []
    b_rows = []
    for k in top_idx:
        row = np.zeros(n + 1)
        row[:n] = dx * grad_corr_k(h0, k, n)
        row[n] = -1.0
        A_rows.append(row)
        b_rows.append(-corrs[k] * dx)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(-radius, radius)] * n + [(0.0, None)]
    try:
        res = linprog(c_obj, A_ub=np.array(A_rows), b_ub=np.array(b_rows),
                      A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    except Exception:
        return None, None
    if res.status != 0:
        return None, None
    delta = res.x[:n]
    h_new = project_feasible(h0 + delta, target_sum)
    return h_new, c5_of_h(h_new, dx)


def lp_polish(h0, dx, target_sum, max_iters=100, init_radius=0.08):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    radius = init_radius
    stall = 0
    for it in range(max_iters):
        h_new, new_c5 = trust_region_lp(h, dx, target_sum, radius, n_active=50)
        if h_new is None or new_c5 is None:
            radius *= 0.5
            stall += 1
            if stall > 20 or radius < 1e-8:
                break
            continue
        if new_c5 < best_c5 - 1e-15:
            best_c5 = new_c5
            best_h = h_new.copy()
            h = h_new
            stall = 0
        else:
            improved = False
            for frac in [0.5, 0.25, 0.1]:
                h_frac = project_feasible(h + frac * (h_new - h), target_sum)
                c5_frac = c5_of_h(h_frac, dx)
                if c5_frac < best_c5 - 1e-15:
                    best_c5 = c5_frac
                    best_h = h_frac.copy()
                    h = h_frac
                    improved = True
                    break
            if not improved:
                radius *= 0.6
                stall += 1
                if stall > 25 or radius < 1e-8:
                    break
    return best_h, best_c5

    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)

                if i == j:
                    continue
                eps_hi = min(h[i], 1.0 - h[j])
                if eps_hi < 1e-12:
                    continue
                d = np.zeros(2 * n - 1)

            break
    return best_h, best_c5


def slsqp_smooth(h0, dx, target_sum, beta, maxiter=800):
    n = len(h0)

    def obj(x):
        return smooth_max_obj(x, dx, beta)

    def jac(x):
        return smooth_max_grad(x, dx, beta)

    def c_fun(x):
        return x.sum() - target_sum

    def c_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': c_fun, 'jac': c_jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0, jac=jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        h_out = project_feasible(res.x, target_sum)
    except Exception:
        h_out = project_feasible(h0, target_sum)
    return h_out, c5_of_h(h_out, dx)


def slsqp_target(h0, dx, target_sum, beta, theta, maxiter=600):
    n = len(h0)

    def obj(x):
        return target_penalty_obj(x, dx, beta, theta)

    def jac(x):
        return target_penalty_grad(x, dx, beta, theta)

    def c_fun(x):
        return x.sum() - target_sum

    def c_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': c_fun, 'jac': c_jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0, jac=jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        h_out = project_feasible(res.x, target_sum)
    except Exception:
        h_out = project_feasible(h0, target_sum)
    return h_out, c5_of_h(h_out, dx)

def mirror_symmetrize(h):
    n = len(h)
    rev_comp = 1.0 - h[::-1]
    return (h + rev_comp) / 2.0


def make_fourier_seed(n, target_sum, rng, n_harmonics=10):
    x = np.linspace(0.0, 2.0, n, endpoint=False) + 1.0 / n
    y = np.zeros(n)
    for k in range(1, n_harmonics + 1):
        amp = rng.uniform(0.1, 1.0) / k
        phase = rng.uniform(0, 2 * np.pi)
        y += amp * np.cos(2 * np.pi * k * x / 2.0 + phase)
    y = y - y.mean()
    h = 0.5 + y * rng.uniform(0.3, 0.8)
    return project_feasible(h, target_sum)


def make_periodic_seed(n, target_sum, rng):

    for i in range(n):
        h[i] = 1.0 if (i % period) < fill else 0.0
    h += rng.uniform(-0.1, 0.1, size=n)
    return project_feasible(h, target_sum)

            h_interp = project_feasible(h_interp, ts_try)
            h_opt3, c5_val3 = local_optimize(h_interp, dx_try, ts_try, time_left)
            if c5_val3 < best_c5:
                best_c5 = c5_val3
                best_h = h_opt3

    # Step 7: Final intensive polish on best solution
    if time_left() > 20:
        n_b = len(best_h)
        dx_b = 2.0 / n_b
        ts_b = n_b / 2.0
        for round_idx in range(4):
            if time_left() < 10:
                break
            h_lp, c5_lp = lp_polish(best_h, dx_b, ts_b, max_iters=40, init_radius=0.03)
            if c5_lp < best_c5:
                best_c5 = c5_lp
                best_h = h_lp
            h_pt, c5_pt = pair_transfer(best_h, dx_b, ts_b, n_top=20, n_iter=2)
            if c5_pt < best_c5:
                best_c5 = c5_pt
                best_h = h_pt
            # Target descent on final
            if time_left() > 12:
                h_td, c5_td = iterative_target_descent(best_h, dx_b, ts_b, time_left, max_rounds=10)
                if c5_td < best_c5:
                    best_c5 = c5_td
                    best_h = h_td

    # Final validation
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final / 2.0)
    best_c5 = c5_of_h(best_h, 2.0 / n_final)

    return best_h, best_c5, n_final
