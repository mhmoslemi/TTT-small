4. At a_next, update active set, repeat

    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def c5_of_h(h, dx):
    return compute_corrs(h).max() * dx

def corr_quad(h, d):
    """Return A, B, C such that corr_k(h+a*d) ≈ A[k]+B[k]*a+C[k]*a²"""
    A = np.correlate(h, 1.0 - h, mode='full')
    B = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = -np.correlate(d, d, mode='full')
    return A, B, C

def max_of_quads(A, B, C, alphas):
    """Evaluate max_k(A[k]+B[k]*a+C[k]*a²) for each a in alphas."""
    # Vectorized: alphas has shape (m,), A,B,C have shape (p,)
    # Result: (m,)
    return (A[None, :] + B[None, :] * alphas[:, None] + C[None, :] * alphas[:, None]**2).max(axis=1)

def feasil_alpha_range(h, d):
    """Find alpha range where h + alpha*d stays in [0,1] for alpha >= 0."""
    alpha_max = 100.0
    for i in range(len(h)):
        if d[i] > 1e-15:
            alpha_max = min(alpha_max, (1.0 - h[i]) / d[i])
        elif d[i] < -1e-15:
            alpha_max = min(alpha_max, h[i] / (-d[i]))
    return max(0.0, alpha_max)

def exact_1d_search(h, d, dx, n_grid=2000):
    """Find alpha >= 0 minimizing c5(h + alpha*d). Returns (best_alpha, best_c5)."""
    alpha_max = feasil_alpha_range(h, d)
    if alpha_max < 1e-12:
        return 0.0, c5_of_h(h, dx)
    
    A, B, C = corr_quad(h, d)
    
    # Dense grid
    alphas = np.linspace(0, alpha_max, n_grid)
    vals = max_of_quads(A, B, C, alphas) * dx
    idx = np.argmin(vals)
    best_alpha = alphas[idx]
    best_val = vals[idx]
    
    # Refine around the best point
    if 0 < idx < n_grid - 1:
        a_lo = alphas[idx-1]
        a_hi = alphas[idx+1]
        # Golden section refinement
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        for _ in range(50):
            if a_hi - a_lo < 1e-13:
                break
            a1 = a_hi - gr * (a_hi - a_lo)
            a2 = a_lo + gr * (a_hi - a_lo)
            f1 = max_of_quads(A, B, C, np.array([a1]))[0]
            f2 = max_of_quads(A, B, C, np.array([a2]))[0]
            if f1 < f2:
                a_hi = a2
            else:
                a_lo = a1
        best_alpha = (a_lo + a_hi) / 2.0
        best_val = max_of_quads(A, B, C, np.array([best_alpha]))[0] * dx
    
    return best_alpha, best_val

def grad_corr_k(h, k, n):
    offset = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -offset)
    i_hi = min(n - 1, n - 1 - offset)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx + offset] += (1.0 - h[idx])
    grad[idx] += (-h[idx + offset])
    return grad

def lp_direction(h, dx, target_sum, n_active, step_bound):
    """Solve LP to find best direction. Returns direction d with sum(d)=0."""
    n = len(h)
    corrs = compute_corrs(h)
    
    # Select active constraints
    if n_active >= len(corrs):
        active_k = np.arange(len(corrs))
    else:
        active_k = np.argsort(corrs)[-n_active:]

    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_rows = []
    b_rows = []
    for k in active_k:
        row = np.zeros(n + 1)
        row[:n] = dx * grad_corr_k(h, k, n)
        row[n] = -1.0
        A_rows.append(row)
        b_rows.append(-corrs[k] * dx - 1e-13)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)

    bounds = []
    for i in range(n):
        bounds.append((max(-h[i], -step_bound), min(1.0 - h[i], step_bound)))
    bounds.append((0.0, None))

    try:
        res = linprog(c_obj, A_ub=np.array(A_rows), b_ub=np.array(b_rows),
                      A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    except Exception:
        return None
    if res.status != 0:
        return None

    delta = res.x[:n]
    if np.max(np.abs(delta)) < 1e-14:
        return None
    
    # Ensure sum is exactly 0
    delta -= delta.sum() / n
    
    return delta

def lp_polish_exact(h0, dx, target_sum, max_iters=200, init_step=0.15, n_active_frac=0.4):
    """LP polish with exact 1D line search."""
    n = len(h0)

            else:
                stall += 1
        else:
            stall += 1
            step *= 0.7
            if step < 1e-8:
                break
    
    return best_h, best_c5

        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)


def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def c5_of_h(h, dx):
    return compute_corrs(h).max() * dx


def grad_corr_k(h, k, n):
    offset = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -offset)
    i_hi = min(n - 1, n - 1 - offset)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx + offset] += (1.0 - h[idx])
    grad[idx] += (-h[idx + offset])
    return grad


def corr_quad_coeffs(h, d):
    A = np.correlate(h, 1.0 - h, mode='full')
    B = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    C = -np.correlate(d, d, mode='full')
    return A, B, C


def max_quads_eval(A, B, C, a):
    return (A + B * a + C * a * a).max()


def feasible_alpha_max(h, d):
    alpha_max = 100.0
    pos = d > 1e-15
    neg = d < -1e-15
    if pos.any():
        alpha_max = min(alpha_max, np.min((1.0 - h[pos]) / d[pos]))
    if neg.any():
        alpha_max = min(alpha_max, np.min(h[neg] / (-d[neg])))
    return max(0.0, alpha_max)


def exact_1d_search(h, d, dx, n_grid=1500):
    alpha_max = feasible_alpha_max(h, d)
    if alpha_max < 1e-12:
        return 0.0, c5_of_h(h, dx)
    A, B, C = corr_quad_coeffs(h, d)
    alphas = np.linspace(0.0, alpha_max, n_grid)
    vals = (A[None, :] + B[None, :] * alphas[:, None] + C[None, :] * alphas[:, None] ** 2).max(axis=1) * dx
    idx = np.argmin(vals)
    best_alpha = alphas[idx]
    best_val = vals[idx]
    lo_b, hi_b = 0.0, alpha_max
    if idx > 0 and idx < n_grid - 1:
        lo_b = alphas[idx - 1]
        hi_b = alphas[idx + 1]
    elif idx == 0:
        hi_b = alphas[1]
    else:
        lo_b = alphas[idx - 1]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a1 = hi_b - gr * (hi_b - lo_b)
    a2 = lo_b + gr * (hi_b - lo_b)
    f1 = max_quads_eval(A, B, C, a1)
    f2 = max_quads_eval(A, B, C, a2)
    for _ in range(60):
        if hi_b - lo_b < 1e-13:
            break
        if f1 < f2:
            hi_b = a2
            a2 = a1
            f2 = f1
            a1 = hi_b - gr * (hi_b - lo_b)
            f1 = max_quads_eval(A, B, C, a1)
        else:
            lo_b = a1
            a1 = a2
            f1 = f2
            a2 = lo_b + gr * (hi_b - lo_b)
            f2 = max_quads_eval(A, B, C, a2)
    best_alpha = (lo_b + hi_b) / 2.0
    best_val = max_quads_eval(A, B, C, best_alpha) * dx
    return best_alpha, best_val


def lp_direction(h, dx, n_active, step_bound):
    n = len(h)
    corrs = compute_corrs(h)
    n_corr = len(corrs)
    if n_active >= n_corr:
        active_k = np.arange(n_corr)
    else:
        active_k = np.argsort(corrs)[-n_active:]
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_rows = []
    b_rows = []
    for k in active_k:
        row = np.zeros(n + 1)
        row[:n] = dx * grad_corr_k(h, k, n)
        row[n] = -1.0
        A_rows.append(row)
        b_rows.append(-corrs[k] * dx - 1e-13)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for i in range(n):
        bounds.append((max(-h[i], -step_bound), min(1.0 - h[i], step_bound)))
    bounds.append((0.0, None))
    try:
        res = linprog(c_obj, A_ub=np.array(A_rows), b_ub=np.array(b_rows),
                      A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    except Exception:
        return None
    if res.status != 0:
        return None
    delta = res.x[:n].copy()
    delta -= delta.sum() / n
    if np.max(np.abs(delta)) < 1e-14:
        return None
    return delta


def lp_polish_exact(h0, dx, target_sum, max_iters=100, init_step=0.2, n_active_frac=0.5, n_grid=1200):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    step = init_step
    stall = 0
    n_active = max(15, int(n_active_frac * (2 * n - 1)))
    for it in range(max_iters):
        d = lp_direction(h, dx, n_active, step)
        if d is None:
            step *= 0.5
            stall += 1
            if stall > 40 or step < 1e-9:
                break
            continue
        best_alpha, pred_val = exact_1d_search(h, d, dx, n_grid=n_grid)
        if best_alpha > 1e-12 and pred_val < best_c5 - 1e-13:
            h_new = project_feasible(h + best_alpha * d, target_sum)
            new_c5 = c5_of_h(h_new, dx)
            if new_c5 < best_c5 - 1e-13:
                h = h_new
                best_h = h.copy()
                best_c5 = new_c5
                stall = 0
                step = min(step * 1.25, init_step)
            else:
                stall += 1
                step *= 0.7
        else:
            for frac in [0.5, 0.25, 0.1]:
                h_new = project_feasible(h + frac * best_alpha * d, target_sum)
                new_c5 = c5_of_h(h_new, dx)
                if new_c5 < best_c5 - 1e-13:
                    h = h_new
                    best_h = h.copy()
                    best_c5 = new_c5
                    stall = 0
                    break
            else:
                stall += 1
                step *= 0.6
            if step < 1e-9:
                break
    return best_h, best_c5


def subgrad_line_search(h0, dx, target_sum, n_iter=800, beta=80.0, n_grid=800):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    no_improve = 0
    for it in range(n_iter):
        corrs = compute_corrs(h)
        shifted = corrs - corrs.max()
        w = np.exp(beta * shifted)
        wsum = w.sum()
        if wsum < 1e-30:
            w = np.ones_like(corrs) / len(corrs)
        else:
            w /= wsum
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] > 1e-11:
                grad += w[k] * grad_corr_k(h, k, n)
        grad *= dx
        d = -grad
        d -= d.sum() / n
        dnorm = np.linalg.norm(d)
        if dnorm < 1e-15:
            break
        d = d / dnorm
        best_alpha, pred_val = exact_1d_search(h, d, dx, n_grid=n_grid)
        if pred_val < best_c5 - 1e-13:
            h_new = project_feasible(h + best_alpha * d, target_sum)
            new_c5 = c5_of_h(h_new, dx)
            if new_c5 < best_c5 - 1e-13:
                h = h_new
                best_h = h.copy()
                best_c5 = new_c5
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        if no_improve > 200:
            break
    return best_h, best_c5


def pair_transfer_exact(h, dx, target_sum, n_top=25, n_iter=4, n_grid=400):
    n = len(h)
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    for iteration in range(n_iter):
        corrs = compute_corrs(best_h)
        grad_total = np.zeros(n)
        for k in range(2 * n - 1):
            grad_total += corrs[k] * grad_corr_k(best_h, k, n)
        dec_idx = np.argsort(grad_total)[-n_top:]
        inc_idx = np.argsort(grad_total)[:n_top]
        improved = False
        for i in dec_idx:
            for j in inc_idx:
                if i == j:
                    continue
                d = np.zeros(n)
                d[i] = -1.0
                d[j] = 1.0
                ba, bv = exact_1d_search(best_h, d, dx, n_grid=n_grid)
                if ba > 1e-10 and bv < best_c5 - 1e-11:
                    h_new = project_feasible(best_h + ba * d, target_sum)
                    nc = c5_of_h(h_new, dx)
                    if nc < best_c5 - 1e-11:
                        best_h = h_new
                        best_c5 = nc
                        improved = True
        if not improved:
            break
    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def tl():
        return budget_s - (time.time() - t0)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    dx = 2.0 / n
    ts = n / 2.0
    h = project_feasible(h_init, ts)

    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)

    # Phase 1: Main LP polish with exact line search
    while tl() > 80:
        h, c5 = lp_polish_exact(h, dx, ts, max_iters=60, init_step=0.2, n_active_frac=0.5, n_grid=1200)
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_h = h.copy()
        else:
            break

    # Phase 1b: Subgradient polish
    if tl() > 60:
        h_sg, c5_sg = subgrad_line_search(best_h, dx, ts, n_iter=500, beta=100.0)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()

    # Phase 2: Perturbations
    for trial in range(20):
        if tl() < 50:
            break
        n_cur = len(best_h)
        ts_cur = n_cur / 2.0
        dx_cur = 2.0 / n_cur
        m = trial % 5
        if m == 0:
            p = 0.07 * (rng.random(n_cur) - 0.5)
        elif m == 1:
            p = 0.03 * (rng.random(n_cur) - 0.5)
        elif m == 2:
            idx = rng.choice(n_cur, size=max(5, n_cur // 10), replace=False)
            p = np.zeros(n_cur)
            p[idx] = rng.choice([-1, 1], size=len(idx)) * 0.12
        elif m == 3:
            freq = rng.randint(1, max(2, n_cur // 4))
            p = 0.08 * np.sin(2.0 * np.pi * freq * np.arange(n_cur) / n_cur + rng.random() * 2 * np.pi)
        else:
            p = rng.normal(0, 0.04, size=n_cur)
        h_try = project_feasible(best_h + p, ts_cur)
        h_opt, c5_opt = lp_polish_exact(h_try, dx_cur, ts_cur, max_iters=40, init_step=0.12, n_active_frac=0.4, n_grid=800)
        c5_opt = c5_of_h(h_opt, dx_cur)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 3: Different n values
    n_cur = len(best_h)
    n_tries = list(range(70, 200, 4))
    n_tries.sort(key=lambda x: abs(x - n_cur))
    n_tries = [nt for nt in n_tries if nt != n_cur][:12]
    for n_try in n_tries:
        if tl() < 60:
            break
        dx_t = 2.0 / n_try
        ts_t = n_try / 2.0
        x_old = np.linspace(0.0, 2.0, n_cur, endpoint=False) + 1.0 / n_cur
        x_new = np.linspace(0.0, 2.0, n_try, endpoint=False) + 1.0 / n_try
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, ts_t)
        h_opt, c5_opt = lp_polish_exact(h_interp, dx_t, ts_t, max_iters=40, init_step=0.15, n_active_frac=0.5, n_grid=1000)
        c5_opt = c5_of_h(h_opt, dx_t)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
            n_cur = n_try

    # Phase 4: Final intensive polish on best
    n_cur = len(best_h)
    dx_cur = 2.0 / n_cur
    ts_cur = n_cur / 2.0
    if tl() > 40:
        h_f, c5_f = lp_polish_exact(best_h, dx_cur, ts_cur, max_iters=80, init_step=0.1, n_active_frac=0.6, n_grid=1500)
        c5_f = c5_of_h(h_f, dx_cur)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f

    # Phase 5: Pair transfer fine polish
    if tl() > 25:
        h_pt, c5_pt = pair_transfer_exact(best_h, dx_cur, ts_cur, n_top=20, n_iter=3, n_grid=300)
        c5_pt = c5_of_h(h_pt, dx_cur)
        if c5_pt < best_c5:
            best_c5 = c5_pt
            best_h = h_pt

    # Phase 6: One more LP + subgrad if time
    if tl() > 20:
        h_f2, c5_f2 = lp_polish_exact(best_h, dx_cur, ts_cur, max_iters=50, init_step=0.08, n_active_frac=0.5, n_grid=1200)
        c5_f2 = c5_of_h(h_f2, dx_cur)
        if c5_f2 < best_c5:
            best_c5 = c5_f2
            best_h = h_f2
        if tl() > 15:
            h_s2, c5_s2 = subgrad_line_search(best_h, dx_cur, ts_cur, n_iter=300, beta=150.0)
            c5_s2 = c5_of_h(h_s2, dx_cur)
            if c5_s2 < best_c5:
                best_c5 = c5_s2
                best_h = h_s2

    # Final validation
    nf = len(best_h)
    best_h = project_feasible(best_h, nf / 2.0)
    best_c5 = c5_of_h(best_h, 2.0 / nf)
    return best_h, best_c5, nf
