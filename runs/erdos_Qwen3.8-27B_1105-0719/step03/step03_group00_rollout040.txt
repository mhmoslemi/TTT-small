import numpy as np
from scipy.optimize import linprog
import time

def project_feasible(z, target):
    n = len(z)
    lo = np.min(z) - 2.0
    hi = np.max(z) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.clip(z - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def c5_of_h(h, dx):
    return np.correlate(h, 1.0 - h, mode='full').max() * dx

def corr_of_h(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_grad_matrix(h):
    """Compute the (2n-1) x n gradient matrix for all correlations."""
    n = len(h)
    n_corr = 2 * n - 1
    grads = np.zeros((n_corr, n))
    for k in range(n_corr):
        d = k - (n - 1)
        i_lo = max(0, -d)
        i_hi = min(n - 1, n - 1 - d)
        if i_lo > i_hi:
            continue
        i = np.arange(i_lo, i_hi + 1)
        j = i + d
        grads[k, i] += (1.0 - h[j])
        grads[k, j] += (-h[i])
    return grads

def pair_swap_min(corr, a, L, U, n_iter=25):
    """
    Find t in [L, U] that minimizes f(t) = max_k(corr[k] + t * a[k]).
    f is convex piecewise linear.
    Returns (t_opt, f_opt).
    """
    t_lo = L
    t_hi = U
    for _ in range(n_iter):
        t_mid = (t_lo + t_hi) / 2
        vals = corr + t_mid * a
        f_mid = vals.max()
        # Active set: values within 1e-10 of the max
        active = vals >= f_mid - 1e-10
        a_active = a[active]
        if a_active.min() >= -1e-12:
            # f is non-decreasing at t_mid, minimum is to the left
            t_hi = t_mid
        elif a_active.max() <= 1e-12:
            # f is non-increasing at t_mid, minimum is to the right
            t_lo = t_mid
        else:
            # 0 is in subgradient, t_mid is optimal
            return t_mid, f_mid
    t_star = (t_lo + t_hi) / 2
    return t_star, (corr + t_star * a).max()

def full_pair_pass(h, dx, target, corr, grads, best_c5):
    """
    Try all pairs (p, q) and apply the best improvement.
    Returns improved h and c5.
    """
    n = len(h)
    best_pair = None
    best_t = None
    best_f = best_c5 / dx  # work in corr space
    
    # Current max corr
    curr_max = corr.max()
    
    if curr_max < best_f:
        best_f = curr_max
    
    for p in range(n):
        for q in range(p + 1, n):
            L = max(-h[p], h[q] - 1.0)
            U = min(1.0 - h[p], h[q])
            if U - L < 1e-14:
                continue
            a = grads[:, p] - grads[:, q]
            if np.abs(a).max() < 1e-15:
                continue
            t_opt, f_opt = pair_swap_min(corr, a, L, U)
            if f_opt < best_f - 1e-12:
                best_f = f_opt
                best_pair = (p, q)
                best_t = t_opt
    
    if best_pair is None:
        return h, best_c5, False
    
    p, q = best_pair
    h_new = h.copy()
    h_new[p] += best_t
    h_new[q] -= best_t
    h_new = project_feasible(h_new, target)
    new_c5 = c5_of_h(h_new, dx)
    return h_new, new_c5, True

def project_feasible(z, target):
    n = len(z)
    lo = np.min(z) - 2.0
    hi = np.max(z) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.clip(z - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def c5_of_h(h, dx):
    return np.correlate(h, 1.0 - h, mode='full').max() * dx

def corr_of_h(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_grads_for_lags(h, lags):
    """Compute gradients for specified lags. Returns (len(lags), n) array."""
    n = len(h)
    K = len(lags)
    grads = np.zeros((K, n))
    for idx in range(K):
        k = lags[idx]
        d = k - (n - 1)
        i_lo = max(0, -d)
        i_hi = min(n - 1, n - 1 - d)
        if i_lo > i_hi:
            continue
        i = np.arange(i_lo, i_hi + 1)
        j = i + d
        grads[idx, i] += (1.0 - h[j])
        grads[idx, j] += (-h[i])
    return grads

def solve_soft_lp(h, dx, corr, active_lags, alpha, rho_s=5.0, rho_q=0.3):
    """
    Solve soft-slack LP to find descent direction.
    
    min rho_q * t + rho_s * sum(s_j)
    s.t. dx * grad_j . delta + s_j - t <= -corr[j]  for j in active_lags
         sum(delta) = 0
         -h[i]*alpha <= delta[i] <= (1-h[i])*alpha
         s_j >= 0
         t >= 0 (not needed, follows from constraints)
    """
    n = len(h)
    K = len(active_lags)
    # Variables: delta (n), s (K), t (1) => total n + K + 1
    m = n + K + 1
    
    # Objective: minimize rho_q * t + rho_s * sum(s)
    c_obj = np.zeros(m)
    c_obj[n:n+K] = rho_s
    c_obj[n+K] = rho_q
    
    # Inequality constraints: A_ub @ x <= b_ub
    # For each active lag j: dx * grad_j . delta - t + s_j <= -corr[lag_j]
    # => dx * grad_j . delta + [0...0, 1, 0]_j - [0...0, 1]_t <= -corr[lag_j]
    A_ub = np.zeros((K, m))
    b_ub = np.zeros(K)
    for j in range(K):
        A_ub[j, :n] = dx * compute_grads_for_lags(h, [active_lags[j]])[0]
        A_ub[j, n+j] = 1.0     # s_j coefficient
        A_ub[j, n+K] = -1.0    # t coefficient
        b_ub[j] = -corr[active_lags[j]]
    
    # Equality: sum(delta) = 0
    A_eq = np.zeros((1, m))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds
    bounds = [(-h[i]*alpha, (1.0-h[i])*alpha) for i in range(n)]
    bounds += [(0.0, None)] * K  # s_j >= 0
    bounds.append((0.0, None))   # t >= 0
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            return res.x[:n]
    except Exception:
        pass
    return None

def line_search(h, cur_c5, direction, dx, target):
    """Try multiple step sizes, return best feasible (h_new, c5_new)."""
    n = len(h)
    best_h = h.copy()
    best_c5 = cur_c5

    """

        if U - L < 1e-14:

        a = np.zeros(2*n - 1)

            a_k = 0.0

                i_hi = min(n-1, n-1-d)
                if i_lo <= p <= i_hi:
                    a_k += (1.0 - h[p + d])
            i2 = p - d

                i_hi = min(n-1, n-1-d)
                if i_lo <= i2 <= i_hi:

            # q contributions (negated)
            if 0 <= q + d < n:
                i_lo = max(0, -d)
                i_hi = min(n-1, n-1-d)
                if i_lo <= q <= i_hi:

            i3 = q - d

                i_lo = max(0, -d)
                i_hi = min(n-1, n-1-d)
                if i_lo <= i3 <= i_hi:
                    a_k += h[i3]
            a[k] = a_k
        
        if np.abs(a).max() < 1e-15:
            continue
        
        # Binary search for optimal t
        t_lo, t_hi = L, U
        for _ in range(n_iter):
            t_mid = (t_lo + t_hi) / 2
            vals = corr + t_mid * a
            f_mid = vals.max()
            active = vals >= f_mid - 1e-10
            a_act = a[active]
            if a_act.min() >= -1e-12:
                t_hi = t_mid
            elif a_act.max() <= 1e-12:
                t_lo = t_mid
            else:
                break
        t_opt = (t_lo + t_hi) / 2
        f_opt = (corr + t_opt * a).max()

            h[q] -= t_opt
            h = project_feasible(h, target)

def project_feasible(z, target):
    n = len(z)
    lo = float(np.min(z)) - 2.0
    hi = float(np.max(z)) + 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = float(np.clip(z - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - float(h.sum())
    if abs(diff) > 1e-14:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = int(mask.sum())
        if cnt > 0:
            h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)


def c5_of_h(h, dx):
    return float(np.correlate(h, 1.0 - h, mode='full').max()) * dx

def compute_grads_for_lags(h, lags):
    n = len(h)
    K = len(lags)
    grads = np.zeros((K, n))
    for idx in range(K):
        k = int(lags[idx])
        d = k - (n - 1)
        i_lo = max(0, -d)
        i_hi = min(n - 1, n - 1 - d)
        if i_lo > i_hi:
            continue
        i = np.arange(i_lo, i_hi + 1)
        j = i + d
        grads[idx, i] += (1.0 - h[j])
        grads[idx, j] += (-h[i])
    return grads

def binary_project(h, target):
    """Project to binary, fixing sum."""
    n = len(h)
    h_bin = (h > 0.5).astype(float)
    diff = target - h_bin.sum()
    if diff > 0:
        # Need more 1s: change smallest h values that are 0 to 1
        zeros = np.where(h_bin == 0)[0]
        order = np.argsort(-h[zeros])[:int(np.ceil(diff))]
        h_bin[zeros[order]] = 1.0
    elif diff < 0:
        ones = np.where(h_bin == 1)[0]
        order = np.argsort(h[ones])[:int(np.ceil(-diff))]
        h_bin[ones[order]] = 0.0
    return h_bin


def discrete_local_search(h_bin, dx, target, n_swaps, t0, budget_s, rng, min_time=10.0):

        q = ones[rng.randint(len(ones))]
        old_c5 = c5_of_h(cur, dx)

        cur[q] = 0.0

        else:

    elif r < 0.6:
        # Sparse flips
        n_flip = max(2, int(n * 0.05))
        indices = rng.choice(n, size=n_flip, replace=False)
        pert = np.zeros(n)
        pert[indices] = rng.choice([-1, 1], size=n_flip) * scale * 3
        pert -= pert.mean()
    else:
        # Block perturbation
        block_size = max(4, n // 10)
        start = rng.randint(0, n - block_size)
        pert = np.zeros(n)
        pert[start:start + block_size] = rng.standard_normal(block_size) * scale * 2
        pert -= pert.mean()
    return h + pert


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def tl():
        return budget_s - (time.time() - t0)

    try:
        h0 = np.array(initial_h_values, dtype=float)
    except Exception:
        h0 = np.full(128, 0.5)

    n = len(h0)
    dx = 2.0 / n
    target = n / 2.0
    h0 = project_feasible(h0, target)
    best_h = h0.copy()
    best_c5 = c5_of_h(h0, dx)

    # Phase 1: Intensive LP polish with multiple strategies
    for use_soft in [False, True, False, True]:
        if tl() < 30:
            break
        for alpha in [0.35, 0.2, 0.12, 0.06]:
            if tl() < 20:
                break
            h_lp, c5_lp = lp_polish(best_h, dx, target, alpha, 120, t0, budget_s, use_soft)
            if c5_lp < best_c5:
                best_c5 = c5_lp
                best_h = h_lp

    # Phase 2: Pair-swap refinement
    if tl() > 30:
        h_ps, c5_ps = pair_swap_refine(best_h, dx, target, 500, t0, budget_s, rng)
        if c5_ps < best_c5:
            best_c5 = c5_ps
            best_h = h_ps

    # Phase 3: Alternate LP and pair-swap
    for round_idx in range(4):
        if tl() < 60:
            break
        h_lp, c5_lp = lp_polish(best_h, dx, target, 0.15, 100, t0, budget_s, False)
        if c5_lp < best_c5:
            best_c5 = c5_lp
            best_h = h_lp
        if tl() < 30:
            break
        h_ps, c5_ps = pair_swap_refine(best_h, dx, target, 300, t0, budget_s, rng)
        if c5_ps < best_c5:
            best_c5 = c5_ps
            best_h = h_ps

    # Phase 4: Binary projection + discrete search + LP re-relaxation
    if tl() > 60:
        h_bin = binary_project(best_h, target)
        h_bs, c5_bs = discrete_local_search(h_bin, dx, target, 2000, t0, budget_s, rng)
        # LP relaxation from best binary
        h_rel = h_bs + 0.05 * rng.standard_normal(len(h_bs))
        h_rel = project_feasible(h_rel, target)
        h_rel, c5_rel = lp_polish(h_rel, dx, target, 0.2, 150, t0, budget_s, False)
        if c5_rel < best_c5:
            best_c5 = c5_rel
            best_h = h_rel
        h_ps, c5_ps = pair_swap_refine(best_h, dx, target, 300, t0, budget_s, rng)
        if c5_ps < best_c5:
            best_c5 = c5_ps
            best_h = h_ps

    # Phase 5: Perturbation restarts
    n_restarts = 25
    for r in range(n_restarts):
        if tl() < 40:
            break
        scale = max(0.03, 0.15 / (1 + r * 0.3))
        h_pert = make_perturbation(best_h, scale, rng)
        h_pert = project_feasible(h_pert, target)
        h_opt, c5_opt = lp_polish(h_pert, dx, target, 0.2, 80, t0, budget_s, (r % 3 == 0))
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
        if tl() < 20:
            break
        h_ps, c5_ps = pair_swap_refine(best_h, dx, target, 200, t0, budget_s, rng)
        if c5_ps < best_c5:
            best_c5 = c5_ps
            best_h = h_ps

    # Phase 6: Grid size exploration
    n_base = len(best_h)
    n_candidates = set()
    for dn in [-16, -12, -8, -6, -4, -2, 2, 4, 6, 8, 12, 16, 20, 24]:
        nc = n_base + dn
        if 64 <= nc <= 240 and nc % 2 == 0:
            n_candidates.add(nc)
    for nc in [80, 96, 104, 112, 120, 128, 176, 192, 200, 224]:
        if 64 <= nc <= 240 and nc % 2 == 0:
            n_candidates.add(nc)

    for n_try in sorted(n_candidates):
        if tl() < 30:
            break
        if n_try == n_base:
            continue
        dx_t = 2.0 / float(n_try)
        ts_t = float(n_try) / 2.0
        h_interp = transfer_grid(best_h, n_try)
        h_interp = project_feasible(h_interp, ts_t)
        h_opt, c5_opt = lp_polish(h_interp, dx_t, ts_t, 0.2, 100, t0, budget_s, False)
        h_ps, c5_ps = pair_swap_refine(h_opt, dx_t, ts_t, 200, t0, budget_s, rng)
        if c5_ps < c5_opt:
            h_opt, c5_opt = h_ps, c5_ps
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
            n_base = n_try

    # Phase 7: Final intensive polish
    n_b = len(best_h)
    dx_b = 2.0 / float(n_b)
    ts_b = float(n_b) / 2.0

    if tl() > 30:
        for alpha in [0.15, 0.1, 0.06, 0.04, 0.02]:
            if tl() < 15:
                break
            h_f, c5_f = lp_polish(best_h, dx_b, ts_b, alpha, 150, t0, budget_s, (alpha < 0.05))
            if c5_f < best_c5:
                best_c5 = c5_f
                best_h = h_f

    if tl() > 20:
        h_ps, c5_ps = pair_swap_refine(best_h, dx_b, ts_b, 400, t0, budget_s, rng)
        if c5_ps < best_c5:
            best_c5 = c5_ps
            best_h = h_ps

    # Phase 8: More perturbation + refine
    if tl() > 30:
        for r in range(15):
            if tl() < 15:
                break
            scale = max(0.02, 0.08 / (1 + r * 0.5))
            h_pert = make_perturbation(best_h, scale, rng)
            h_pert = project_feasible(h_pert, ts_b)
            h_opt, c5_opt = lp_polish(h_pert, dx_b, ts_b, 0.1, 60, t0, budget_s, False)
            h_ps2, c5_ps2 = pair_swap_refine(h_opt, dx_b, ts_b, 150, t0, budget_s, rng)
            if c5_ps2 < c5_opt:
                h_opt, c5_opt = h_ps2, c5_ps2
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt

    # Phase 9: Final micro-polish
    if tl() > 10:
        h_final, c5_final = lp_polish(best_h, dx_b, ts_b, 0.03, 200, t0, budget_s, True)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final
        h_ps3, c5_ps3 = pair_swap_refine(best_h, dx_b, ts_b, 300, t0, budget_s, rng, min_time=5.0)
        if c5_ps3 < best_c5:
            best_c5 = c5_ps3
            best_h = h_ps3

    # Final validation
    n_final = len(best_h)
    dx_final = 2.0 / float(n_final)
    target_final = float(n_final) / 2.0
    best_h = project_feasible(best_h, target_final)
    best_c5 = c5_of_h(best_h, dx_final)

    return best_h, best_c5, n_final
