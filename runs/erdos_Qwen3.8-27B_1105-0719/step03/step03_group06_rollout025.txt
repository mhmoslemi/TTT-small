        g += wts[k] * grad_row(h, k, n)
    return val, g

def lbfgs_optimize(h0, n, beta, penalty=100.0, maxiter=300):
    """L-BFGS-B optimization with smooth max + penalty for sum constraint"""
    target = n / 2.0
    
    def obj(x):
        prof = compute_profile(x)
        val, _ = smooth_max_vp(prof, beta)
        s = x.sum() - target
        return val + 0.5 * penalty * s * s
    
    def obj_grad(x):
        prof = compute_profile(x)
        _, wts = smooth_max_vp(prof, beta)
        g = np.zeros(n)
        nz = np.where(wts > 1e-12)[0]
        for k in nz:
            g += wts[k] * grad_row(x, k, n)
        s = x.sum() - target
        g += penalty * s * np.ones(n)
        return g
    
    res = minimize(obj, h0, method='L-BFGS-B', jac=obj_grad,
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10})
    if res success and finite:
        h_new = project_feasible(res.x, n)
        return h_new
    return None

def active_descent(h, max_iters=300, time_limit=60, t_start=None):
    """Active set descent with adaptive tolerance"""
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    h_best = h.copy()
    momentum_vec = np.zeros(n)
    
    for it in range(max_iters):
        if t_start is not None and time.perf_counter() - t_start > time_limit:
            break
        profile = compute_profile(h)
        c = float(profile.max())
        # Adaptive threshold: wider early, tighter later
        progress = min(1.0, it / max_iters)
        thresh = c - 0.005 * (1.0 - progress) - 0.0005
        active = np.where(profile >= thresh)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(profile))])
        if len(active) > 30:
            top = np.argsort(profile[active])[-30:]
            active = active[top]
        
        g = np.zeros(n)
        for m in active:
            g += grad_row(h, m, n)
        g -= g.mean()
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-14:
            break
        
        # Adaptive momentum
        momentum = 0.5 - 0.3 * progress
        momentum_vec = momentum * momentum_vec + g
        mv_norm = np.linalg.norm(momentum_vec)
        if mv_norm < 1e-14:
            break
        d = -momentum_vec / mv_norm
        
        # Line search
        pos_mask = d > 1e-12
        neg_mask = d < -1e-12
        alpha_max = 1.0
        if pos_mask.any():
            alpha_max = min(alpha_max, float(((1.0 - h[pos_mask]) / d[pos_mask]).min()))
        if neg_mask.any():
            alpha_max = min(alpha_max, float((h[neg_mask] / (-d[neg_mask])).min()))
        alpha_max = max(alpha_max, 1e-10)
        alpha_max = min(alpha_max, 1.0)
        
        # Backtracking line search
        alpha = alpha_max
        best_alpha = 0.0
        best_val = c
        best_h = h
        for ls in range(30):
            h_try = h + alpha * d
            h_try = project_feasible(h_try, n)
            val = exact_score(h_try)
            if val < best_val:
                best_val = val
                best_alpha = alpha
                best_h = h_try.copy()
            if val < c - alpha * gnorm * 0.01:  # Armijo-like condition
                break
            alpha *= 0.8
        
        if best_val >= c - 1e-14:
            # Try smaller fixed steps
            for trial_alpha in [0.01, 0.005, 0.001]:
                h_try = project_feasible(h + trial_alpha * d, n)
                val = exact_score(h_try)
                if val < c:
                    best_val = val
                    best_alpha = trial_alpha
                    best_h = h_try.copy()
                    break
            if best_val >= c - 1e-14:
                break
        
        h = best_h
        if best_val < best_c:
            best_c = best_val
            h_best = h.copy()
    
    return h_best, best_c

def interp_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    x_old = np.linspace(0, 2, n + 1)
    x_new = np.linspace(0, 2, new_n + 1)
    h_new = np.zeros(new_n)
    for k in range(new_n):
        a, b = x_new[k], x_new[k+1]
        total = 0.0
        for j in range(n):
            lo = max(a, x_old[j])
            hi = min(b, x_old[j+1])
            if hi > lo:
                total += h[j] * (hi - lo)
        h_new[k] = total / (b - a)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_feasible(h_new, new_n)

def binary_seed_golden(n, rng):
    """Binary seed using golden ratio distribution"""
    half = n // 2
    # Place ones at positions closest to i * n / half for i = 0, ..., half-1
    ideal = np.arange(half) * n / half
    positions = np.round(ideal).astype(int)
    positions = np.sort(np.unique(positions))
    # If we have duplicates, adjust
    if len(positions) < half:
        used = set(positions.tolist())
        candidates = np.array([i for i in range(n) if i not in used])
        for _ in range(half - len(positions)):
            if len(candidates) == 0:
                break
            idx = rng.randint(len(candidates))
            positions = np.append(positions, candidates[idx])
            used.add(int(candidates[idx]))
            candidates = np.array([i for i in candidates if i != candidates[idx]])
        positions = np.sort(positions)
    h = np.zeros(n)
    h[positions[:half]] = 1.0
    return h

def binary_local_search(h, n, max_iters=5000, time_limit=20):
    """Local search on binary arrangements via swaps"""
    t0 = time.perf_counter()
    h = h.copy()
    best_c = exact_score(h)
    best_h = h.copy()
    no_improve = 0
    
    for it in range(max_iters):
        if time.perf_counter() - t0 > time_limit:
            break
        # Pick random 1 and 0 to swap
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        h[i] = 0.0
        h[j] = 1.0
        c_new = exact_score(h)
        if c_new < best_c:
            best_c = c_new
            best_h = h.copy()
            no_improve = 0
        else:
            h[i] = 1.0
            h[j] = 0.0
            no_improve += 1
            if no_improve > 200:
                # Restart with new random binary
                h = binary_seed_golden(n, np.random.RandomState(it))
                if exact_score(h) < best_c:
                    best_c = exact_score(h)
                    best_h = h.copy()
                no_improve = 0
    
    return best_h, best_c
