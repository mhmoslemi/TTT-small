The Erdős minimum overlap problem asks: given a function h: [0,2] → [0,1] with ∫h = 1, what is the minimum possible value of max_k ∫ h(x)(1-h(x+k)) dx?

The current best is about 0.38085. The trivial bound is 0.5 (achieved by h = 1/2 everywhere).

The structure of the optimal solution is not well understood. It's believed to be close to a binary function (taking values 0 and 1) with a specific structure related to "Furstenberg sets" or similar constructions.

For a binary h with h = 1 on set A (measure 1) and h = 0 on set B (measure 1), the overlap at shift k is |A ∩ (B-k)| = |A| - |A ∩ (A+k)| = 1 - |A ∩ (A+k)|.

So C5 = max_k (1 - |A ∩ (A+k)|) = 1 - min_k |A ∩ (A+k)|.

We want to maximize min_k |A ∩ (A+k)|, which is the minimum autocorrelation of the set A.

This is related to the "Salem-Spencer" type problems and additive combinatorics.

For the discrete analog (h as a 0-1 vector of length n with n/2 ones), the overlap at shift d is:
corr_d = (n/2 - d) - |{i: h[i]=1 and h[i+d]=1}| for d > 0 (approximately)

import numpy as np
import time

def project_box_sum(x, target):
    n = len(x)
    x = np.where(np.isfinite(x), x, 0.5)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(x.min()) - target - 2.0
    hi = float(x.max()) + target + 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    err = target - h.sum()
    if abs(err) > 1e-14:
        if err > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        nm = mask.sum()
        if nm > 0:
            h[mask] += err / nm
    return np.clip(h, 0.0, 1.0)

def c5_of_h(h, dx):
    return np.correlate(h, 1.0 - h, mode='full').max() * dx

def softmax_gradient(h, dx, alpha, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = corr / alpha
    scaled -= scaled.max()
    w = np.exp(scaled)
    w /= w.sum()
    grad = np.zeros(n)
    grad += w[n - 1] * (1.0 - 2.0 * h)
    for d in range(1, n):
        a_d = w[n - 1 + d]
        b_d = w[n - 1 - d]
        s_d = a_d + b_d
        if n - d > 0:
            grad[:n - d] += a_d - s_d * h[d:]
        if d < n:
            grad[d:] += b_d - s_d * h[:n - d]
    grad *= dx
    return grad

def smooth_descent(h0, dx, target, budget_s, t0, max_time=None):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    if max_time is None:
        max_time = budget_s - (time.time() - t0) - 10.0
    momentum = 0.8
    alphas = [0.4, 0.25, 0.15, 0.1, 0.07, 0.05, 0.035, 0.025, 0.018, 0.012, 0.008, 0.005, 0.003, 0.002]
    for alpha in alphas:
        if budget_s - (time.time() - t0) < 12.0:
            break
        v = np.zeros(n)
        lr = 0.3 / n
        h = best_h.copy()
        for it in range(3000):
            if budget_s - (time.time() - t0) < 12.0:
                break
            g = softmax_gradient(h, dx, alpha, n)
            g -= g.mean()
            v = momentum * v + (1.0 - momentum) * g
            h_new = h - lr * v
            h_new = project_box_sum(h_new, target)
            c5_new = c5_of_h(h_new, dx)
            if c5_new <= c5_of_h(h, dx) + 1e-15:
                h = h_new
                lr = min(lr * 1.03, 3.0 / n)
            else:
                lr *= 0.6
                v *= 0.5
            if c5_new < best_c5 - 1e-15:
                best_h = h_new.copy()
                best_c5 = c5_new
            if lr < 1e-7:
                lr = 1.0 / n
                v = np.zeros(n)
                h = best_h.copy()
    return best_h, best_c5

def build_modes(n, D, freq_type='cosine'):
    D = min(D, n - 2)
    B = np.zeros((n, D))
    t = (np.arange(n) + 0.5) / n
    for r in range(1, D + 1):
        if freq_type == 'cosine':
            B[:, r - 1] = np.cos(np.pi * r * t)
        elif freq_type == 'sine':
            B[:, r - 1] = np.sin(np.pi * r * t)
        else:
            B[:, r - 1] = np.cos(2.0 * np.pi * r * t)
        B[:, r - 1] -= B[:, r - 1].mean()
    Q, _ = np.linalg.qr(B)
    return Q[:, :D]

def subspace_search(h0, dx, target, budget_s, t0, rng):

    best_h = h0.copy()
    best_c5 = c5_of_h(h0, dx)

            best_c5 = c5_new

            no_improve = 0

                    best_c5 = c5_jump

                    no_improve = 0

    return best_h, best_c5

def binary_round_refine(h0, dx, target, budget_s, t0, rng):
