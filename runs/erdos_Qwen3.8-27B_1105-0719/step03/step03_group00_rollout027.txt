    scaled = corr / alpha
    m = scaled.max()
    w = np.exp(scaled - m)
    w /= w.sum()

    # d = 0 term
    w0 = w[n-1]
    if w0 > 1e-15:
        grad += dx * w0 * (1.0 - 2.0 * h)
    
    for d in range(1, n):
        wp = w[n-1+d]  # weight for offset +d
        wn = w[n-1-d]  # weight for offset -d
        
        if wp > 1e-15:
            # Positive offset d: grad[j] += (1-h[j+d]) for j < n-d

            grad[:n-d] += dx * wp * (1.0 - h[d:])

            # Negative offset d: grad[j] += (1-h[j-d]) for j >= d
            #                    grad[j] += (-h[j+d]) for j < n-d
            grad[d:] += dx * wn * (1.0 - h[:n-d])
            grad[:n-d] += -dx * wn * h[d:]
    
    return grad, w

def project_feasible(x, target_sum):
    """Project x onto {0<=h<=1, sum(h)=target_sum}"""
    n = len(x)
    x = np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-13:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def c5_of_h(h, dx):
    """Compute C5 = max of full correlation * dx"""
    return np.correlate(h, 1.0 - h, mode='full').max() * dx

def compute_corrs(h, n):
    return np.correlate(h, 1.0 - h, mode='full')

def grad_corr_offset(h, d, n):
    """Gradient of corr at offset d with respect to h. Returns array of length n."""
    grad = np.zeros(n)
    if d == 0:
        grad[:] = 1.0 - 2.0 * h
        return grad
    if d > 0:
        # corr(d) = sum_{a=0}^{n-1-d} h[a]*(1-h[a+d])
        # d/dh[j] = (1-h[j+d]) for j < n-d, minus h[j-d] for j >= d
        grad[:n-d] += (1.0 - h[d:])
        grad[d:] += -h[:n-d]
    else:
        p = -d
        # corr(-p) = sum_{a=0}^{n-1-p} h[a+p]*(1-h[a])
        # d/dh[j] = (1-h[j-p]) for j >= p, minus h[j+p] for j < n-p
        grad[p:] += (1.0 - h[:n-p])
        grad[:n-p] += -h[p:]
    return grad

def weighted_gradient(h, n, weights):
    """Compute dx * sum_k weights[k] * grad_corr_k(h)"""
    # weights has length 2*n-1, indexed by full correlate index
    # offset d corresponds to index n-1+d
    dx = 2.0 / n
    grad = np.zeros(n)
    w_mid = weights[n-1]
    if abs(w_mid) > 1e-15:
        grad += w_mid * (1.0 - 2.0 * h)
    for d in range(1, n):
        wp = weights[n-1+d]
        wn = weights[n-1-d]
        if abs(wp) > 1e-15:
            grad[:n-d] += wp * (1.0 - h[d:])
            grad[d:] += -wp * h[:n-d]
        if abs(wn) > 1e-15:
            grad[d:] += wn * (1.0 - h[:n-d])
            grad[:n-d] += -wn * h[d:]
    return dx * grad

def softmax_weights(corr, alpha):
    """Compute softmax(corr/alpha) weights"""
    scaled = corr / alpha
    m = scaled.max()
    w = np.exp(scaled - m)
    total = w.sum()
    if total < 1e-300:
        w = np.ones_like(corr) / len(corr)
    else:
        w /= total
    return w

            alpha = min(0.35, alpha * 1.08)

    return best_h, best_c5


def softmax_descent(h, dx, target_sum, alpha_start, alpha_end, max_outer,
                    inner_iters, budget_s, t0, min_time=8.0):
    n = len(h)
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    cur = h.copy()
    cur_c5 = best_c5

    n_alphas = max_outer
    alphas = np.geomspace(alpha_start, alpha_end, n_alphas)

    for ai, alpha in enumerate(alphas):
        if budget_s - (time.time() - t0) < min_time:
            break
        for it in range(inner_iters):
            if budget_s - (time.time() - t0) < min_time:
                break
            corrs = compute_corrs(cur, n)
            w = softmax_weights(corrs, alpha)
            g = weighted_gradient(cur, n, w)
            g = g - g.mean()

            step = 0.5 / (np.linalg.norm(g) + 1e-10)
            improved = False
            for mult in [1.0, 0.7, 0.5, 0.35, 0.25, 0.15, 0.1, 0.05]:
                s = step * mult
                h_try = cur - s * g
                h_try = project_feasible(h_try, target_sum)
                new_c5 = c5_of_h(h_try, dx)
                if new_c5 < cur_c5:
                    cur = h_try
                    cur_c5 = new_c5
                    improved = True
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = h_try.copy()
                    break
            if not improved:
                break

    return best_h, best_c5


def binary_local_search(h, dx, target_sum, n_iter, budget_s, t0, rng, min_time=10.0):
    n = len(h)
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    cur = h.copy()
    cur_c5 = best_c5
    temp = 0.01

    ones = np.where(cur > 0.5)[0]
    zeros = np.where(cur <= 0.5)[0]

    for it in range(n_iter):
        if budget_s - (time.time() - t0) < min_time:
            break
        for _ in range(50):
            if budget_s - (time.time() - t0) < min_time:
                break
            if len(ones) < 1 or len(zeros) < 1:
                break
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            if i == j:
                continue
            old_i, old_j = cur[i], cur[j]
            cur[i], cur[j] = old_j, old_i
            cur = project_feasible(cur, target_sum)
            new_c5 = c5_of_h(cur, dx)
            if new_c5 < cur_c5 or rng.random() < np.exp((cur_c5 - new_c5) / temp):
                cur_c5 = new_c5
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = cur.copy()
                    ones = np.where(cur > 0.5)[0]
                    zeros = np.where(cur <= 0.5)[0]
            else:
                cur[i], cur[j] = old_i, old_j
                cur = project_feasible(cur, target_sum)
        temp = max(0.001, temp * 0.85)

    return best_h, best_c5


def perturbation_reopt(h, dx, target_sum, rng, n_rounds, budget_s, t0, min_time=15.0):
    n = len(h)
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)

    for r in range(n_rounds):
        if budget_s - (time.time() - t0) < min_time:
            break
        scale = max(0.02, 0.12 / (r + 1))
        corrs = compute_corrs(h, n)
        w = softmax_weights(corrs, 0.05)
        g = weighted_gradient(h, n, w)
        g = g - g.mean()
        gn = np.linalg.norm(g)
        if gn > 1e-10:
            g = g / gn

        for attempt in range(3):
            noise = rng.standard_normal(n)
            noise -= noise.mean()
            nn = np.linalg.norm(noise)
            if nn > 1e-10:
                noise = noise / nn
            dir_vec = 0.7 * g + 0.3 * noise
            dir_vec = dir_vec - dir_vec.mean()
            h_p = h + scale * dir_vec
            h_p = project_feasible(h_p, target_sum)
            h_opt, c5_opt = lp_local(h_p, dx, target_sum, 0.2, 80, budget_s, t0, 10.0)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                h = h_opt.copy()
                break

    return best_h, best_c5


def transfer_grid(h_src, n_target):
    n_src = len(h_src)
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_tgt = (np.arange(n_target) + 0.5) * (2.0 / n_target)
    return np.interp(x_tgt, x_src, h_src)


def subgradient_local(h, dx, target_sum, n_iter, budget_s, t0, min_time=8.0):
    n = len(h)
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    cur = h.copy()
    cur_c5 = best_c5
    step_init = 0.08

    for it in range(n_iter):
        if budget_s - (time.time() - t0) < min_time:
            break
        corrs = compute_corrs(cur, n)
        max_c = corrs.max()
        active = np.where(corrs >= max_c - 0.05)[0]
        if len(active) == 0:
            active = np.array([np.argmax(corrs)])
        g = np.zeros(n)
        for k in active:
            g += grad_corr_k(cur, k, n)
        g /= len(active)
        g = g - g.mean()
        gn = np.linalg.norm(g)
        if gn < 1e-12:
            break
        g = g / gn
        step = step_init / (1.0 + 0.05 * it)
        improved = False
        for mult in [1.0, 0.7, 0.5, 0.3, 0.2, 0.1]:
            s = step * mult
            h_try = cur - s * g
            h_try = project_feasible(h_try, target_sum)
            new_c5 = c5_of_h(h_try, dx)
            if new_c5 < cur_c5:
                cur = h_try
                cur_c5 = new_c5
                improved = True
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = cur.copy()
                break
        if not improved:
            step_init *= 0.7
            if step_init < 0.002:
                break

    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def tl():
        return budget_s - (time.time() - t0)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = np.full(128, 0.5)

    n = len(h0)
    dx = 2.0 / n
    target_sum = n / 2.0
    h0 = project_feasible(h0, target_sum)
    best_h = h0.copy()
    best_c5 = c5_of_h(h0, dx)

    # Phase 1: Soft-max continuation from parent
    if tl() > 60:
        corrs_init = compute_corrs(best_h, n)
        alpha_s = max(0.05, (corrs_init.max() - corrs_init.min()) * 0.3)
        h_sm, c5_sm = softmax_descent(best_h, dx, target_sum, alpha_s, 0.005,
                                      12, 40, budget_s, t0)
        if c5_sm < best_c5:
            best_c5 = c5_sm
            best_h = h_sm

    # Phase 2: Subgradient method
    if tl() > 40:
        h_sg, c5_sg = subgradient_local(best_h, dx, target_sum, 100, budget_s, t0)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg

    # Phase 3: Intensive LP polish - multiple alpha starts
    for alpha_s in [0.25, 0.15, 0.1, 0.06, 0.04]:
        if tl() < 30:
            break
        h_lp, c5_lp = lp_local(best_h, dx, target_sum, alpha_s, 200, budget_s, t0)
        if c5_lp < best_c5:
            best_c5 = c5_lp
            best_h = h_lp

    # Phase 4: Binary rounding and local search
    if tl() > 50:
        h_bin = (best_h > 0.5).astype(float)
        # Adjust to exact target
        n_ones = int(h_bin.sum())
        n_zeros = n - n_ones
        needed = int(target_sum)
        if n_ones > needed:
            idx = np.where(h_bin == 1)[0]
            # Remove from positions most "interior" to top correlations
            corrs_b = compute_corrs(h_bin, n)
            for _ in range(n_ones - needed):
                scores = np.zeros(n)
                for j in idx:
                    if h_bin[j] == 1:
                        scores[j] = np.sum(np.abs(np.gradient(best_h, axis=0)) * 0)
                # Simply remove random ones
                pick = idx[rng.randint(len(idx))]
                h_bin[pick] = 0
                idx = np.where(h_bin == 1)[0]
        elif n_ones < needed:
            idx = np.where(h_bin == 0)[0]
            for _ in range(needed - n_ones):
                pick = idx[rng.randint(len(idx))]
                h_bin[pick] = 1
                idx = np.where(h_bin == 0)[0]
        h_bin = project_feasible(h_bin, target_sum)
        c5_bin = c5_of_h(h_bin, dx)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()

        # Binary local search
        h_bl, c5_bl = binary_local_search(best_h, dx, target_sum, 400,
                                          budget_s, t0, rng)
        if c5_bl < best_c5:
            best_c5 = c5_bl
            best_h = h_bl

        # Re-refine fractionally after binary
        if tl() > 25:
            h_rf, c5_rf = lp_local(best_h, dx, target_sum, 0.15, 100, budget_s, t0)
            if c5_rf < best_c5:
                best_c5 = c5_rf
                best_h = h_rf

    # Phase 5: Structured perturbation + re-optimize
    if tl() > 50:
        h_p, c5_p = perturbation_reopt(best_h, dx, target_sum, rng, 12,
                                       budget_s, t0)
        if c5_p < best_c5:
            best_c5 = c5_p
            best_h = h_p

    # Phase 6: Multi-resolution
    n_base = len(best_h)
    n_candidates = []
    for dn in [-6, -4, -2, 2, 4, 6, 8, -10, 10]:
        nt = n_base + dn
        if 48 <= nt <= 220 and nt % 2 == 0:
            n_candidates.append(nt)
    for extra in [64, 72, 80, 96, 104, 112, 120, 132, 136, 144, 152, 160, 168, 176, 184]:
        if extra != n_base and 48 <= extra <= 220:
            n_candidates.append(extra)
    n_candidates = sorted(set(n_candidates))

    for n_try in n_candidates:
        if tl() < 30:
            break
        if n_try == n_base:
            continue
        dx_t = 2.0 / n_try
        ts_t = n_try / 2.0
        h_interp = transfer_grid(best_h, n_try)
        h_interp = project_feasible(h_interp, ts_t)
        h_opt, c5_opt = lp_local(h_interp, dx_t, ts_t, 0.2, 80, budget_s, t0, 8.0)
        c5_check = c5_of_h(h_opt, dx_t)
        if c5_check < best_c5:
            best_c5 = c5_check
            best_h = h_opt.copy()
            n_base = n_try

    # Phase 7: Final intensive polish
    n_b = len(best_h)
    dx_b = 2.0 / n_b
    ts_b = n_b / 2.0

    if tl() > 30:
        h_f, c5_f = lp_local(best_h, dx_b, ts_b, 0.12, 200, budget_s, t0)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f

    if tl() > 20:
        h_f2, c5_f2 = lp_local(best_h, dx_b, ts_b, 0.03, 200, budget_s, t0)
        if c5_f2 < best_c5:
            best_c5 = c5_f2
            best_h = h_f2

    if tl() > 30:
        h_p2, c5_p2 = perturbation_reopt(best_h, dx_b, ts_b, rng, 8, budget_s, t0)
        if c5_p2 < best_c5:
            best_c5 = c5_p2
            best_h = h_p2
            if tl() > 15:
                h_f3, c5_f3 = lp_local(best_h, dx_b, ts_b, 0.05, 150, budget_s, t0)
                if c5_f3 < best_c5:
                    best_c5 = c5_f3
                    best_h = h_f3

    # Phase 8: Very fine final steps
    if tl() > 12:
        h_fin, c5_fin = lp_local(best_h, dx_b, ts_b, 0.015, 150, budget_s, t0, 5.0)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin

    # Final validation
    n_final = len(best_h)
    dx_final = 2.0 / n_final
    ts_final = n_final / 2.0
    best_h = project_feasible(best_h, ts_final)
    best_c5 = c5_of_h(best_h, dx_final)

    return best_h, best_c5, n_final
