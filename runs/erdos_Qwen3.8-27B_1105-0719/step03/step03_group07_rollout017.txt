def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full"))) * dx

def profile_h(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx])
            h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
            h[idx] = min(1.0, h[idx] + diff2)
        else:
            idx = int(np.argmax(h))
            h[idx] = max(0.0, h[idx] + diff2)
    return h

def grad_for_lag(h, m, n, dx):
    """Compute gradient of P_m with respect to h.
    m is the lag offset (0 to n-1), where m=0 is the central correlation.
    In the full correlation array, index = n-1+m for positive m, n-1 for m=0, n-1-m for negative m.
    """
    g = np.zeros(n)
    if m == 0:
        g[:] = dx * (1.0 - 2.0 * h)
    elif m > 0:
        # P_m = dx * sum_{i=0}^{n-m-1} h[i] * (1 - h[i+m])
        # dP_m/dh[j] = dx * ((1-h[j+m]) if j+m<n) - dx * (h[j-m]) if j-m>=0
        # Actually: dP_m/dh[j] = dx * (1-h[j+m]) * 1[j+m <= n-1] - dx * h[j-m] * 1[j-m >= 0]
        # Wait, let me be more careful.
        # P_m = dx * sum_{i=0}^{n-m-1} h[i](1-h[i+m])
        # = dx * sum_{i=0}^{n-m-1} h[i] - dx * sum_{i=0}^{n-m-1} h[i]h[i+m]
        # dP_m/dh[j]:
        #   From first sum: dx * 1[j <= n-m-1]
        #   From second sum: -dx * sum_i h[i]h[i+m] * d/dh[j]
        #     If j = i: -dx * h[j+m] for j <= n-m-1
        #     If j = i+m: -dx * h[j-m] for j >= m
        # So: dP_m/dh[j] = dx * 1[j <= n-m-1] - dx * h[j+m]*1[j<=n-m-1] - dx * h[j-m]*1[j>=m]
        # = dx * (1 - h[j+m]) * 1[j <= n-m-1] - dx * h[j-m] * 1[j >= m]
        for j in range(n - m):
            g[j] += dx * (1.0 - h[j + m])
        for j in range(m, n):
            g[j] -= dx * h[j - m]
    else:
        # m < 0, let dp = -m
        # P_{-dp} = dx * sum_{i=0}^{n-dp-1} h[i] * (1 - h[i+dp])  -- no wait
        # Actually for negative lag, the correlation is:
        # np.correlate(h, 1-h, mode='full')[n-1-m] for m<0 means lag -m = |m|
        # The correlation at offset d (where d = index - (n-1)) is:
        # sum_{i} h[i] * (1-h[i+d]) for valid i
        # So for d = m (negative), it's sum_{i} h[i] * (1-h[i+m]) = sum_{i} h[i] * (1-h[i-dp])
        # where dp = -m > 0
        # = dx * sum_{i=dp}^{n-1} h[i] * (1 - h[i-dp])
        # dP/dh[j]:
        #   From first factor h[i]: dx * (1-h[j-dp]) * 1[j >= dp]
        #   From second factor (1-h[i-dp]): -dx * h[j+dp] * 1[j+dp < n]  [when i-dp = j, i = j+dp]
        # Wait, let me redo.
        # P = dx * sum_{i=dp}^{n-1} h[i] * (1 - h[i-dp])
        # dP/dh[j]:
        #   If j appears as i (first h): dx * (1-h[j-dp]) for j >= dp
        #   If j appears as i-dp (second h): -dx * h[j+dp] for j+dp >= dp, i.e., j >= 0, and j+dp <= n-1, i.e., j <= n-1-dp
        dp = -m
        for j in range(dp, n):
            g[j] += dx * (1.0 - h[j - dp])
        for j in range(0, n - dp):
            g[j] -= dx * h[j + dp]
    return g

def lp_leveling(h, active_lags, n, dx, max_time=0.5):
    """Find LP direction that simultaneously decreases active constraints.
    Returns (direction, predicted_improvement) or (None, 0).
    """
    t0 = time.perf_counter()
    
    # Build gradient matrix for active lags
    G = np.array([grad_for_lag(h, m, n, dx) for m in active_lags])
    # G has shape (len(active_lags), n)
    
    # We want to minimize t subject to:
    #   G @ d <= -t  (for each active constraint, want negative derivative)
    #   sum(d) = 0
    #   -1 <= d <= 1
    #   
    # Variables: d (n vars), t (1 var), total n+1 vars
    # 
    # min t
    # s.t. G@dh - t <= 0  → for each row i: G[i,:]@d - t <= 0
    #      sum(d) = 0
    #      -1 <= d <= 1
    #      t >= 0
    #
    # For linprog: min c @ x, s.t. A_ub @ x <= b_ub, A_eq @ x = b_eq, bounds
    
    m_act = len(active_lags)
    # c: minimize t (last variable)
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    # A_ub: G[i,:] @ d - t <= 0 → [G[i,:], -1] @ [d; t] <= 0
    A_ub = np.zeros((m_act, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = np.zeros(m_act)
    
    # A_eq: sum(d) = 0 → [ones, 0] @ [d; t] = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds: -1 <= d <= 1, 0 <= t <= inf
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    # Wait, d can be negative. Let me use -1 to 1.
    bounds = [(-1.0, 1.0)] * n + [(0.0, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0 and res.fun > 1e-10:
            return res.x[:n], float(res.fun)
    except Exception:
        pass
    
    return None, 0.0

def lp_leveling_direction(h, active_lags, n, dx):
    """Simpler LP: find direction that minimizes max derivative over active set.
    min max_i (G[i,:] @ d)  s.t. sum(d)=0, ||d||_inf <= 1
    This is equivalent to: min t s.t. G[i,:]@d <= t for all i, sum(d)=0, -1<=d<=1
    """
    m_act = len(active_lags)
    G = np.array([grad_for_lag(h, m, n, dx) for m in active_lags])
    
    # min t s.t. G@dh - t <= 0, -G@dh - t <= 0 (to handle both signs)
    # Actually we just want min max, so: min t s.t. G_i @ d <= t for all i
    # And we want t < 0 (negative = decrease)
    # But to find the best direction, we want to minimize t (most negative)
    # So: min t s.t. G_i @ d <= t, sum(d) = 0, -1 <= d <= 1
    # t is unbounded below in general, so we need to bound it.
    # Actually, since d is bounded and G is bounded, t is automatically bounded.
    
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize t
    
    # G_i @ d - t <= 0 for each i
    A_ub = np.zeros((m_act, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = np.zeros(m_act)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    bounds = [(-1.0, 1.0)] * n + [(None, None)]

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    return repair_feasible(np.clip(h_new, 0.0, 1.0))

def select_active(h, bandwidth=0.005, max_k=25):
    """Select lags that are within bandwidth of the max."""
    P = profile_h(h)
    Pmax = P.max()
    thresh = Pmax - bandwidth
    active = np.where(P >= thresh)[0]
    # Convert to lag offsets
    n = len(h)
    lags = active - (n - 1)
    # Sort by P value, keep top max_k
    if len(active) > max_k:
        order = np.argsort(P[active])[::-1][:max_k]
        lags = lags[order]
    return lags.tolist()

def slsqp_active(h0, maxiter=300, active_k=10, tol=1e-12):
    """SLSQP with top-K active constraints."""
    n = len(h0)
    dx = 2.0 / n
    M = 2 * n - 1
    P = profile_h(h0)
    c0 = P.max()
    
    thresh = c0 - max(0.003, 0.05 * (c0 - np.median(P)))
    active = np.where(P >= thresh)[0]
    if len(active) == 0:
        active = np.array([np.argmax(P)])
    if len(active) > active_k:
        top_idx = np.argsort(P[active])[-active_k:]
        active = active[top_idx]
    
    def obj(x):
        return x[-1]
    
    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    
    def eq_fun(x):
        return x[:-1].sum() - n / 2.0
    
    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g
    
    def ineq_fun(x):
        h = x[:-1]
        t = x[-1]
        prof = profile_h(h)
        return t - prof[active]
    
    def ineq_jac(x):
        h = x[:-1]
        lags = active - (n - 1)
        G = np.array([grad_for_lag(h, m, n, dx) for m in lags])
        J = np.zeros((len(active), n + 1))
        J[:, :-1] = -G
        J[:, -1] = 1.0
        return J
    
    x0 = np.concatenate([h0, [c0 + 1e-4]])
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 1.0)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    try:

def coordinate_descent(h, max_sweeps=20, t_start=None, budget_s=1100):
    """Coordinate-wise line search."""
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    
    for sweep in range(max_sweeps):
        if t_start is not None and time.perf_counter() - t_start > budget_s - 10:
            break
        improved = False
        for i in range(n):
            if t_start is not None and time.perf_counter() - t_start > budget_s - 10:
                break
            h_save = h[i]
            h_sum_save = h.sum()
            for direction in [1, -1]:
                for frac in [0.01, 0.005, 0.002, 0.001]:
                    delta = direction * frac
                    h_new_val = h_save + delta
                    if h_new_val < -1e-12 or h_new_val > 1.0 + 1e-12:
                        continue
                    h[i] = np.clip(h_new_val, 0.0, 1.0)
                    # Repair sum
                    h = repair_feasible(h)
                    s = exact_score(h)
                    if s < best_c - 1e-15:
                        best_c = s
                        improved = True
                        break
                    else:
                        h[i] = h_save
                        h = repair_feasible(h)
                        break
                if improved:
                    break
            if t_start is not None and time.perf_counter() - t_start > budget_s - 10:
                break
    return h, best_c

def make_structured_start(n, rng, method='golden'):
    """Create structured starting points."""
    if method == 'golden':
        h = np.zeros(n)
        golden = 0.6180339887
        for i in range(n // 2):
            pos = int((i * golden * n) % n)
            h[pos] = 1.0
        return repair_feasible(h)
    elif method == 'vandermonde':
        h = np.zeros(n)
        # Use modular arithmetic pattern
        for i in range(n // 2):
            pos = (i * (n // 3 + 1)) % n
            h[pos] = 1.0
        return repair_feasible(h)
    elif method == 'random':
        h = np.zeros(n)
        one_positions = rng.choice(n, n // 2, replace=False)
        h[one_positions] = 1.0
        return repair_feasible(h)
    return np.full(n, 0.5)

import numpy as np

def exact_score(h):

def profile_h(h):

    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx


def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx])
            h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
            h[idx] = min(1.0, h[idx] + diff2)
        else:
            idx = int(np.argmax(h))
            h[idx] = max(0.0, h[idx] + diff2)
    return h


def grad_for_lag(h, m, n, dx):
    g = np.zeros(n)
    if m == 0:
        g[:] = dx * (1.0 - 2.0 * h)
    elif m > 0:
        g[:n - m] += dx * (1.0 - h[m:])
        g[m:] -= dx * h[:n - m]
    else:
        dp = -m
        g[dp:] += dx * (1.0 - h[:n - dp])
        g[:n - dp] -= dx * h[dp:]
    return g


def grad_matrix(h, lags, n, dx):
    return np.array([grad_for_lag(h, m, n, dx) for m in lags])


def select_active(h, bandwidth=0.005, max_k=25):

    Pmax = P.max()
    thresh = Pmax - bandwidth
    active = np.where(P >= thresh)[0]
    n = len(h)
    lags = active - (n - 1)
    if len(active) > max_k:

        lags = lags[order]

def lp_direction(h, active_lags, n, dx):
    m_act = len(active_lags)
    if m_act == 0:
        return None
    G = grad_matrix(h, active_lags, n, dx)
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_ub = np.zeros((m_act, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = np.zeros(m_act)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(-1.0, 1.0)] * n + [(None, None)]
    try:

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    return repair_feasible(np.clip(h_new, 0.0, 1.0))

def coordinate_descent(h, max_sweeps=10, t_start=None, budget_s=1100):
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    for sweep in range(max_sweeps):
        if t_start is not None and time.perf_counter() - t_start > budget_s - 10:
            break
        any_improved = False
        for i in range(n):
            if t_start is not None and time.perf_counter() - t_start > budget_s - 10:
                break
            h_save = h[i]
            for direction in [1, -1]:
                for frac in [0.005, 0.002, 0.001, 0.0005]:
                    delta = direction * frac
                    h_new_val = h_save + delta
                    if h_new_val < -1e-12 or h_new_val > 1.0 + 1e-12:
                        continue
                    h[i] = np.clip(h_new_val, 0.0, 1.0)
                    h = repair_feasible(h)
                    s = exact_score(h)
                    if s < best_c - 1e-16:
                        best_c = s
                        any_improved = True
                        break
                    h[i] = h_save
                if h.sum() != n / 2.0:
                    h = repair_feasible(h)
            if time.perf_counter() - t_start > budget_s - 10:
                break
        if not any_improved:
            break
    return h, best_c

    if method == 'golden':
        h = np.zeros(n)
        golden = 0.6180339887
        for i in range(n // 2):
            pos = int((i * golden * n) % n)
            h[pos] = 1.0
        return repair_feasible(h)
    elif method == 'mod':
        h = np.zeros(n)
        step = n // 3 + 1
        for i in range(n // 2):
            pos = (i * step) % n
            h[pos] = 1.0
        return repair_feasible(h)
    else:
        h = np.zeros(n)
        one_positions = rng.choice(n, n // 2, replace=False)
        h[one_positions] = 1.0
        return repair_feasible(h)


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()
    rng = np.random.RandomState(seed)

    def time_ok(margin=15):
        return time.perf_counter() - t_start < budget_s - margin

    parent = None
    try:
        parent = np.array(initial_h_values, dtype=np.float64)
    except NameError:
        pass
    except Exception:
        pass

    if parent is None or len(parent) < 4:
        parent = np.full(70, 0.5)

    parent = np.clip(parent, 0.0, 1.0)
    parent = repair_feasible(parent)

    best_h = parent.copy()
    best_c = exact_score(parent)

    # Symmetrization
    if time_ok():
        h_sym = repair_feasible(symmetrize(parent))
        c_sym = exact_score(h_sym)
        if c_sym < best_c - 1e-13:
            best_h = h_sym
            best_c = c_sym

    n = len(best_h)

    # Phase 1: LP-based leveling (primary method)
    if time_ok(120):
        h_lp, c_lp = lp_optimize(best_h, max_iters=200, t_start=t_start,
                                  budget_s=budget_s,
                                  bw_schedule=[0.01, 0.007, 0.005, 0.004, 0.003, 0.002, 0.0015, 0.001, 0.0007, 0.0005, 0.0003],
                                  step_schedule=[0.04, 0.035, 0.03, 0.025, 0.02, 0.015, 0.012, 0.01, 0.008, 0.006, 0.004])
        if c_lp < best_c - 1e-14:
            best_h = h_lp
            best_c = c_lp

    # Phase 2: SLSQP polish
    if time_ok(80):
        for ak in [12, 15, 18]:
            if not time_ok(25):
                break
            sol = slsqp_active(best_h, maxiter=250, active_k=ak, tol=1e-12)
            if sol is not None:
                c = exact_score(sol)
                if c < best_c - 1e-13:
                    best_h = sol
                    best_c = c

    # Phase 3: Second LP round (narrower bandwidth)
    if time_ok(80):
        h_lp, c_lp = lp_optimize(best_h, max_iters=150, t_start=t_start,
                                  budget_s=budget_s,
                                  bw_schedule=[0.004, 0.003, 0.002, 0.0015, 0.001, 0.0007, 0.0005, 0.0003, 0.0002, 0.0001],
                                  step_schedule=[0.015, 0.012, 0.01, 0.008, 0.006, 0.005, 0.004, 0.003, 0.002, 0.0015])
        if c_lp < best_c - 1e-14:
            best_h = h_lp
            best_c = c_lp

    # Phase 4: SLSQP with perturbed starts
    if time_ok(60):
        for eps in [0.004, 0.008, 0.015]:
            if not time_ok(20):
                break
            for trial in range(2):
                if not time_ok(10):
                    break
                z = rng.randn(n)
                z -= z.mean()
                h_p = repair_feasible(best_h + eps * z)
                sol = slsqp_active(h_p, maxiter=200, active_k=12, tol=1e-11)
                if sol is not None:
                    c = exact_score(sol)
                    if c < best_c - 1e-12:
                        best_h = sol
                        best_c = c
                        # LP from new best
                        if time_ok(20):
                            h_lp2, c_lp2 = lp_optimize(best_h, max_iters=50, t_start=t_start,
                                                        budget_s=budget_s,
                                                        bw_schedule=[0.003, 0.002, 0.001, 0.0005, 0.0003],
                                                        step_schedule=[0.01, 0.008, 0.006, 0.004, 0.003])
                            if c_lp2 < best_c - 1e-14:
                                best_h = h_lp2
                                best_c = c_lp2

    # Phase 5: Dimension exploration
    if time_ok(80):
        n_best = len(best_h)
        cand_dims = []
        for d in range(max(20, n_best - 12), min(140, n_best + 20)):
            if d % 2 == 0 and d != n_best:
                cand_dims.append(d)
        cand_dims.sort(key=lambda d: abs(d - n_best))
        found_better_dim = False
        for new_n in cand_dims[:12]:
            if not time_ok(30) or found_better_dim:
                break
            h_interp = interpolate_to_n(best_h, new_n)
            h_new, c_new = lp_optimize(h_interp, max_iters=60, t_start=t_start,
                                        budget_s=budget_s,
                                        bw_schedule=[0.005, 0.003, 0.002, 0.001, 0.0005, 0.0003],
                                        step_schedule=[0.03, 0.02, 0.015, 0.01, 0.006, 0.004])
            if c_new < best_c - 5e-5:
                best_h = h_new
                best_c = c_new
                found_better_dim = True
                # Polish at new dimension
                if time_ok(30):
                    sol = slsqp_active(best_h, maxiter=200, active_k=12, tol=1e-12)
                    if sol is not None:
                        c2 = exact_score(sol)
                        if c2 < best_c - 1e-13:
                            best_h = sol
                            best_c = c2
                    if time_ok(20):
                        h_lp3, c_lp3 = lp_optimize(best_h, max_iters=40, t_start=t_start,
                                                    budget_s=budget_s,
                                                    bw_schedule=[0.002, 0.001, 0.0005, 0.0003],
                                                    step_schedule=[0.01, 0.006, 0.004, 0.003])
                        if c_lp3 < best_c - 1e-14:
                            best_h = h_lp3
                            best_c = c_lp3

    # Phase 6: Structured restarts
    if time_ok(50):
        n_curr = len(best_h)
        for method in ['golden', 'mod']:
            if not time_ok(15):
                break
            h_struct = make_structured_start(n_curr, rng, method)
            for mix in [0.4, 0.6]:
                if not time_ok(10):
                    break
                h_mix = repair_feasible(mix * h_struct + (1 - mix) * best_h)
                h_new, c_new = lp_optimize(h_mix, max_iters=40, t_start=t_start,
                                            budget_s=budget_s,
                                            bw_schedule=[0.005, 0.003, 0.002, 0.001, 0.0005],
                                            step_schedule=[0.03, 0.02, 0.015, 0.01, 0.006])
                if c_new < best_c - 1e-10:
                    best_h = h_new
                    best_c = c_new

    # Phase 7: Coordinate descent fine-tune
    if time_ok(25):
        best_h, best_c = coordinate_descent(best_h, max_sweeps=5,
                                            t_start=t_start, budget_s=budget_s)

    # Phase 8: Final LP polish with very tight parameters
    if time_ok(30):
        h_lp, c_lp = lp_optimize(best_h, max_iters=80, t_start=t_start,
                                  budget_s=budget_s,
                                  bw_schedule=[0.001, 0.0007, 0.0005, 0.0003, 0.0002, 0.0001],
                                  step_schedule=[0.005, 0.004, 0.003, 0.002, 0.0015, 0.001])
        if c_lp < best_c - 1e-15:
            best_h = h_lp
            best_c = c_lp

    # Phase 9: Final SLSQP
    if time_ok(20):
        sol = slsqp_active(best_h, maxiter=200, active_k=10, tol=1e-13)
        if sol is not None:
            c = exact_score(sol)
            if c < best_c - 1e-15:
                best_h = sol
                best_c = c

    # Final validation
    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)

    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            final_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 1:
            if np.isfinite(eval_result[0]):
                final_c = float(eval_result[0])
    except Exception:
        pass

    return best_h, final_c, len(best_h)
