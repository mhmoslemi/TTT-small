def lse_gradient(h, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    n1 = n - 1
    
    # Compute weights
    corr_shifted = corr - corr.max()  # for numerical stability
    weights = np.exp(beta * corr_shifted)
    weights /= weights.sum()
    
    # Compute gradient
    grad = np.zeros(n)
    for i in range(n):
        # Effect 1: dc[p]/dh[i] = (1-h[n1+i-p]) for i <= p <= n-1+i
        p1 = np.arange(i, min(n-1+i, 2*n-2) + 1)
        s1 = n1 + i - p1
        grad[i] += np.sum(weights[p1] * (1.0 - h[s1]))
        # Effect 2: dc[p]/dh[i] = -h[i+p-n1] for n1-i <= p <= n1+i
        p2 = np.arange(max(0, n1-i), min(n1+i, 2*n-2) + 1)
        t2 = i + p2 - n1
        grad[i] -= np.sum(weights[p2] * h[t2])
    
    return grad

import numpy as np
from scipy.optimize import linprog
import time


def project_feasible(x, target_sum):
    n = len(x)
    x = np.clip(x, -10.0, 10.0)
    lo, hi = -10.0, 10.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        cnt = mask.sum()
        if cnt > 0:
            h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)


def c5_of_h(h, dx):
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)


def transfer_corr_delta(n, i, j, delta, h):
    n1 = n - 1
    dc = np.zeros(2 * n - 1)
    p1_lo = i
    p1_hi = min(n - 1 + i, 2 * n - 2)
    if p1_lo <= p1_hi:
        p1 = np.arange(p1_lo, p1_hi + 1)
        s1 = n1 + i - p1
        dc[p1] += delta * (1.0 - h[s1])
    p2_lo = max(0, n1 - i)
    p2_hi = min(n1 + i, 2 * n - 2)
    if p2_lo <= p2_hi:
        p2 = np.arange(p2_lo, p2_hi + 1)
        t2 = i + p2 - n1
        dc[p2] -= delta * h[t2]
    p3_lo = j
    p3_hi = min(n - 1 + j, 2 * n - 2)
    if p3_lo <= p3_hi:
        p3 = np.arange(p3_lo, p3_hi + 1)
        s3 = n1 + j - p3
        dc[p3] -= delta * (1.0 - h[s3])
    p4_lo = max(0, n1 - j)
    p4_hi = min(n1 + j, 2 * n - 2)
    if p4_lo <= p4_hi:
        p4 = np.arange(p4_lo, p4_hi + 1)
        t4 = j + p4 - n1
        dc[p4] += delta * h[t4]
    return dc


def run_fast_sa(h, dx, target_sum, t0, max_time, seed, T_start=0.3, T_end=0.005,
                cool=0.99995, max_no_improve=8000, delta_mode='cont'):
    rng = np.random.RandomState(seed)
    n = len(h)
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    cur_c5 = float(corr.max() * dx)
    best_c5 = cur_c5
    best_h = h.copy()
    no_improve = 0
    restarts = 0
    T = T_start
    dx_scale = dx

    for it in range(20000000):
        if time.time() - t0 > max_time:
            break
        if no_improve > max_no_improve and T < 0.02:
            if restarts < 10:
                h = project_feasible(best_h + 0.05 * rng.randn(n), target_sum)
                corr = np.correlate(h, 1.0 - h, mode='full')
                cur_c5 = float(corr.max() * dx)
                T = T_start * 0.7
                no_improve = 0
                restarts += 1
            else:
                break

        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue

        if delta_mode == 'quant':
            M = 200
            max_d = min(int(h[j] * M), int((1.0 - h[i]) * M))
            if max_d <= 0:
                continue
            d = rng.randint(1, max_d + 1) / M
        else:
            max_d = min(h[j], 1.0 - h[i])
            if max_d < 1e-12:
                continue
            if rng.random() < 0.7:
                d = 0.005 + 0.045 * rng.random()
            else:
                d = 0.05 + 0.45 * rng.random()
            d = min(d, max_d)

        dc = transfer_corr_delta(n, i, j, d, h)
        corr_new = corr + dc
        new_max = float(corr_new.max())
        new_c5 = new_max * dx_scale
        delta_obj = new_c5 - cur_c5

        if delta_obj < -1e-14 or (rng.random() < np.exp(-delta_obj / (T * dx_scale))):
            corr = corr_new
            h[i] += d
            h[j] -= d
            cur_c5 = new_c5
            if cur_c5 < best_c5 - 1e-15:
                best_c5 = cur_c5
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if it % 5000 == 0:
            T = max(T_end, T * cool)
            if no_improve == 3000:
                T = min(T_start, T * 1.5)

    best_h = project_feasible(best_h, target_sum)
    best_c5 = c5_of_h(best_h, dx)
    return best_h, best_c5


def lse_obj_grad(h, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    n1 = n - 1
    cs = corr - corr.max()
    w = np.exp(beta * cs)
    wsum = w.sum()
    if wsum < 1e-300:
        w = np.ones_like(corr) / len(corr)
    else:
        w /= wsum
    obj = float(np.logsumexp(beta * cs) / beta)
    grad = np.zeros(n)
    for i in range(n):
        g = 0.0
        p_lo = i
        p_hi = min(n - 1 + i, 2 * n - 2)
        for p in range(p_lo, p_hi + 1):
            s = n1 + i - p
            g += w[p] * (1.0 - h[s])
        p_lo2 = max(0, n1 - i)
        p_hi2 = min(n1 + i, 2 * n - 2)
        for p in range(p_lo2, p_hi2 + 1):
            t = i + p - n1
            g -= w[p] * h[t]
        grad[i] = g
    return obj, grad


def run_lse_opt(h, dx, target_sum, t0, max_time, beta_start=50, beta_end=500,
                ls_steps=10, lr_start=0.01):
    n = len(h)
    h = h.copy()
    best_c5 = c5_of_h(h, dx)
    best_h = h.copy()
    lr = lr_start
    total_betas = int(np.log(beta_end / beta_start) / np.log(1.5))
    for bi in range(total_betas):
        if time.time() - t0 > max_time:
            break
        beta = beta_start * (1.5 ** bi)
        for sub in range(ls_steps):
            if time.time() - t0 > max_time:
                break
            obj, grad = lse_obj_grad(h, beta)
            norm_g = np.linalg.norm(grad)
            if norm_g < 1e-10:
                break
            grad_n = grad / norm_g
            for s in [lr * 10, lr * 5, lr * 2, lr, lr * 0.5, lr * 0.25, lr * 0.1]:
                h_try = project_feasible(h - s * grad_n, target_sum)
                c5_try = c5_of_h(h_try, dx)
                if c5_try < best_c5 - 1e-14:
                    h = h_try
                    best_c5 = c5_try
                    best_h = h.copy()
                    break
                elif c5_try < c5_of_h(h, dx):
                    h = h_try
                    break
            lr = max(1e-5, lr * 0.9)
    best_h = project_feasible(best_h, target_sum)
    best_c5 = c5_of_h(best_h, dx)
    return best_h, best_c5


def grad_corr_k(h, k, n):
    offset = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -offset)
    i_hi = min(n - 1, n - 1 - offset)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    j1 = idx + offset
    grad[j1] += (1.0 - h[idx])
    grad[idx] += (-h[idx + offset])
    return grad


def lp_step(h, dx, target_sum, n_active, alpha):
    n = len(h)
    corrs = np.correlate(h, 1.0 - h, mode='full')
    n_total = 2 * n - 1
    current_c5 = float(corrs.max() * dx)
    if n_active >= n_total:
        active_idx = np.arange(n_total)
    else:
        active_idx = np.argsort(corrs)[-n_active:]
    G = np.zeros((len(active_idx), n))
    for row, k in enumerate(active_idx):
        G[row] = grad_corr_k(h, k, n)
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_ub = np.empty((len(active_idx), n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -corrs[active_idx]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(-h[i] * alpha, (1.0 - h[i]) * alpha) for i in range(n)]
    bounds.append((-np.inf, np.inf))
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status != 0:
            return h, current_c5, False
        delta = res.x[:n]
    except Exception:
        return h, current_c5, False
    if np.abs(delta).max() < 1e-15:
        return h, current_c5, False
    for s in [1.0, 0.7, 0.5, 0.3, 0.15, 0.05]:
        h_new = project_feasible(h + s * delta, target_sum)
        c5_new = c5_of_h(h_new, dx)
        if c5_new < current_c5 - 1e-14:
            return h_new, c5_new, True
    return h, current_c5, False


def run_lp(h, dx, target_sum, t0, max_time, alpha_start=0.2, n_active_frac=0.4,
           alpha_decay=0.99, max_no_improve=60):
    n = len(h)
    n_total = 2 * n - 1
    n_active = max(20, int(n_active_frac * n_total))
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    alpha = alpha_start
    no_improve = 0
    while time.time() - t0 < max_time:
        h_new, c5_new, improved = lp_step(h, dx, target_sum, n_active, alpha)
        if improved and c5_new < best_c5 - 1e-14:
            h = h_new
            best_c5 = c5_new
            best_h = h_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            alpha *= alpha_decay
            if no_improve == 25:
                n_active = min(n_total, n_active + 20)
                alpha = max(alpha, 0.15)
            if no_improve > max_no_improve:
                break
    return best_h, best_c5


def quantize_sa(h, dx, target_sum, t0, max_time, seed, M=100):

    target_a = int(round(n * M / 2))

        if cur_sum == target_a:
            break
        if cur_sum > target_a:
            i = rng.randint(n)
            if a[i] > 0:
                a[i] -= 1
                cur_sum -= 1
        else:
            i = rng.randint(n)
            if a[i] < M:
                a[i] += 1
                cur_sum += 1
    h_q = a.astype(float) / M
    corr = np.correlate(h_q, 1.0 - h_q, mode='full')
    cur_c5 = float(corr.max() * dx)
    best_c5 = cur_c5
    best_a = a.copy()
    T = 2.0
    no_imp = 0
    for it in range(500000):

        i = rng.randint(n)
        j = rng.randint(n)

        if a[i] == 0 or a[j] == M:

        a[i] += 1
        a[j] -= 1
        h_new = a.astype(float) / M
        corr_new = corr + transfer_corr_delta(n, i, j, 1.0 / M, h_q)
        new_c5 = float(corr_new.max() * dx)

                best_c5 = c5_lse
                best_h = h_lse

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    n_final = len(best_h)
    ts_final = n_final / 2.0
    best_h = project_feasible(best_h, ts_final)
    dx_final = 2.0 / n_final
    best_c5 = c5_of_h(best_h, dx_final)

    return best_h, best_c5, n_final
