def project_feasible(x, target_sum):
    """Project x onto {h : 0<=h<=1, sum(h)=target_sum}"""
    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)

    if abs(diff) > 1e-14:

        if mask.sum() > 0:

def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

    offset = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -offset)

        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx] += (1.0 - h[idx + offset])
    grad[idx + offset] += (-h[idx])
    return grad

        return g

import numpy as np
from scipy.optimize import minimize
import time


def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)


def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def c5_of_h(h, dx):
    return compute_corrs(h).max() * dx


def grad_corr_k(h, k, n):
    offset = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -offset)
    i_hi = min(n - 1, n - 1 - offset)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx] += (1.0 - h[idx + offset])
    grad[idx + offset] += (-h[idx])
    return grad


def corr_single(h, k, n):
    offset = k - (n - 1)
    i_lo = max(0, -offset)
    i_hi = min(n - 1, n - 1 - offset)
    if i_lo > i_hi:
        return 0.0
    idx = np.arange(i_lo, i_hi + 1)
    return float(np.sum(h[idx] * (1.0 - h[idx + offset])))


def lmo_vertex(G, n, S):
    """Find binary vertex minimizing G^T v over {v: 0<=v<=1, sum v = S}"""
    order = np.argsort(G)
    v = np.zeros(n)
    v[order[:S]] = 1.0
    return v


def epigraph_solve(h0, dx, target_sum, active_k, maxiter=400):
    n = len(h0)
    n_a = len(active_k)

    def obj(z):
        return z[-1]

    def obj_grad(z):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g

    def ineq_fun(z):
        h = z[:n]
        out = np.empty(n_a)
        for j in range(n_a):
            out[j] = z[-1] - dx * corr_single(h, active_k[j], n)
        return out

    def ineq_jac(z):
        h = z[:n]
        J = np.zeros((n_a, n + 1))
        for j in range(n_a):
            J[j, :n] = -dx * grad_corr_k(h, active_k[j], n)
            J[j, n] = 1.0
        return J

    def eq_fun(z):
        return z[:n].sum() - target_sum

    def eq_jac(z):
        row = np.zeros(n + 1)
        row[:n] = 1.0
        return row

    cons = [
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac},
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}
    ]
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    corrs0 = compute_corrs(h0)
    tau0 = dx * corrs0.max() + 1e-10
    z0 = np.append(h0.copy(), tau0)

    try:
        res = minimize(obj, z0, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        h_out = project_feasible(res.x[:n], target_sum)
    except Exception:
        h_out = h0.copy()
    return h_out, c5_of_h(h_out, dx)


def active_set_optimize(h0, dx, target_sum, n_active_start=20, n_active_end=60,
                        max_rounds=8, maxiter_per=300):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)

    for rnd in range(max_rounds):
        corrs = compute_corrs(h)
        if dx * corrs.max() < best_c5:
            best_c5 = dx * corrs.max()
            best_h = h.copy()

        n_act = min(n_active_start + rnd * 5, n_active_end, 2 * n - 1)
        active_k = np.argsort(corrs)[-n_act:]
        active_k = np.sort(active_k)

        h_new, c5_new = epigraph_solve(h, dx, target_sum, active_k, maxiter_per)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
        if c5_new < c5_of_h(h, dx) - 1e-15:
            h = h_new
        else:
            h = h_new

    return best_h, best_c5


def lmo_jump_search(h0, dx, target_sum, n_active=40, n_jumps=10):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)

    for _ in range(n_jumps):
        corrs = compute_corrs(h)
        top_idx = np.argsort(corrs)[-n_active:]
        G = np.zeros(n)
        for k in top_idx:
            G += grad_corr_k(h, k, n)
        G *= dx

        v = lmo_vertex(G, n, int(round(target_sum)))
        # Line search on (1-g)*h + g*v
        best_g = 0.0
        best_c5_jump = best_c5
        for g in np.linspace(0.05, 1.0, 20):
            h_try = (1.0 - g) * h + g * v
            h_try = project_feasible(h_try, target_sum)
            c5_try = c5_of_h(h_try, dx)
            if c5_try < best_c5_jump:
                best_c5_jump = c5_try
                best_g = g
        if best_g > 0:
            h_new = (1.0 - best_g) * h + best_g * v
            h_new = project_feasible(h_new, target_sum)
            h = h_new
            if best_c5_jump < best_c5:
                best_c5 = best_c5_jump
                best_h = h.copy()
        else:
            break
    return best_h, best_c5


def binary_swap_polish(h0, dx, target_sum, max_rounds=200):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()

            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 50:
            break

        # Find positions to try swapping
        # Try swapping pairs where one is high and one is low

        n_try = min(30, n // 4)

        high_idx = sorted_idx[-n_try:]

        for i in high_idx:

                if h[i] - h[j] < 0.1:
                    continue
                # Swap: h[i] -> h[i] - d, h[j] -> h[j] + d where d = h[i] - h[j] (full swap)

                    improved = True

def smooth_max_solve(h0, dx, target_sum, beta, maxiter=2000):
    n = len(h0)

    def obj_fn(x):
        corrs = compute_corrs(x)
        shifted = beta * (corrs - corrs.max())
        return np.log(np.exp(shifted).sum()) / beta * dx

    def obj_grad_fn(x):
        corrs = compute_corrs(x)
        shifted = corrs - corrs.max()
        w = np.exp(beta * shifted)
        w /= w.sum()
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] > 1e-12:
                grad += w[k] * grad_corr_k(x, k, n)
        return grad * dx

    def c_fun(x):
        return x.sum() - target_sum

    def c_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': c_fun, 'jac': c_jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_fn, h0, jac=obj_grad_fn, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        h_out = project_feasible(res.x, target_sum)
    except Exception:
        h_out = h0.copy()
    return h_out, c5_of_h(h_out, dx)

    def time_left():
        return budget_s - (time.time() - t0)

    # Start from initial construction
    h0 = np.array(initial_h_values, dtype=float)
    n = len(h0)
    dx = 2.0 / n
    target_sum = n / 2.0
    h0 = project_feasible(h0, target_sum)

    best_h = h0.copy()
    best_c5 = c5_of_h(h0, dx)

    # Main optimization from initial
    h_opt, c5_val = full_optimize(h0, dx, target_sum, time_left, "main")
    if c5_val < best_c5:
        best_c5 = c5_val
        best_h = h_opt

    # Multi-start: perturbations
    n_perturb = 12
    for trial in range(n_perturb):
        if time_left() < 35:
            break
        if trial % 4 == 0:
            perturb = 0.08 * (rng.random(n) - 0.5)
        elif trial % 4 == 1:
            perturb = 0.2 * (rng.random(n) - 0.5)
        elif trial % 4 == 2:
            # LMO-based perturbation
            corrs = compute_corrs(best_h)
            top_k = np.argsort(corrs)[-30:]
            G = np.zeros(n)
            for k in top_k:
                G += grad_corr_k(best_h, k, n)
            v = lmo_vertex(G, n, int(target_sum))
            perturb = 0.3 * (v - best_h)
        else:
            # Random binary perturbation
            idx = rng.choice(n, size=n // 5, replace=False)
            perturb = np.zeros(n)
            perturb[idx] = rng.choice([-0.3, 0.3], size=len(idx))

        h_try = project_feasible(best_h + perturb, target_sum)
        h_opt2, c5_val2 = full_optimize(h_try, dx, target_sum, time_left, "pert")
        if c5_val2 < best_c5:
            best_c5 = c5_val2
            best_h = h_opt2

    # Try different n values
    n_candidates = [140, 142, 144, 146, 148, 150, 152, 154, 156, 158, 160]
    n_candidates = [nn for nn in n_candidates if nn != n]
    n_candidates.sort(key=lambda x: abs(x - n))

    for n_try in n_candidates[:6]:
        if time_left() < 40:
            break
        dx_try = 2.0 / n_try
        ts_try = n_try / 2.0
        x_orig = np.linspace(0.0, 2.0, n, endpoint=False) + dx / 2
        x_new = np.linspace(0.0, 2.0, n_try, endpoint=False) + dx_try / 2
        h_interp = np.interp(x_new, x_orig, best_h)
        h_interp = project_feasible(h_interp, ts_try)

        h_opt3, c5_val3 = full_optimize(h_interp, dx_try, ts_try, time_left, "n%d" % n_try)
        if c5_val3 < best_c5:
            best_c5 = c5_val3
            best_h = h_opt3

    # Final intensive polish on best
    if time_left() > 25 and best_h is not None:
        n_b = len(best_h)
        dx_b = 2.0 / n_b
        ts_b = n_b / 2.0
        h_final, c5_final = full_optimize(best_h, dx_b, ts_b, time_left, "final")
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final

    # Ultra-fine polish
    if time_left() > 15 and best_h is not None:
        n_b = len(best_h)
        dx_b = 2.0 / n_b
        ts_b = n_b / 2.0
        # Multiple short LP steps
        for _ in range(3):
            if time_left() < 5:
                break
            h_p, c5_p = lp_step_polish(best_h, dx_b, ts_b, max_iters=80, init_step=0.05)
            if c5_p < best_c5:
                best_c5 = c5_p
                best_h = h_p
        # Binary swap
        if time_left() > 10:
            h_s, c5_s = binary_swap_polish(best_h, dx_b, ts_b, max_rounds=100)
            if c5_s < best_c5:
                best_c5 = c5_s
                best_h = h_s
        # Coordinate descent
        if time_left() > 10:
            h_cd, c5_cd = coordinate_descent_polish(best_h, dx_b, ts_b, max_rounds=30)
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd

    # Final validation
    if best_h is None:
        best_h = h0
        best_c5 = c5_of_h(h0, 2.0 / len(h0))
    else:
        n_final = len(best_h)
        best_h = project_feasible(best_h, n_final / 2.0)
        best_c5 = c5_of_h(best_h, 2.0 / n_final)

    return best_h, best_c5, len(best_h)
