        bounds.append((0.0, None))

        try:
            res = linprog(c_obj, A_ub=A_rows, b_ub=b_rows,
                          A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
        except Exception:
            break
        if res.status != 0:
            break

        delta = res.x[:n]
        if np.max(np.abs(delta)) < 1e-14:
            break

        h_new = project_feasible(h + delta, target_sum)
        new_c5 = c5_of_h(h_new, dx)

        if new_c5 < c5_val - 1e-14:
            h = h_new
            step = min(step * 1.1, step_init)
        else:
            improved = False
            for frac in [0.5, 0.25, 0.1, 0.05, 0.02]:
                h_new = project_feasible(h + frac * delta, target_sum)
                new_c5 = c5_of_h(h_new, dx)
                if new_c5 < c5_val - 1e-14:
                    h = h_new
                    improved = True
                    step = min(step * 1.05, step_init)
                    break
            if not improved:
                step *= 0.6
                if step < 1e-10:
                    break

    return best_h, best_c5


def systematic_pair_sweep(h, dx, target_sum, n_rounds=20, max_pairs_per_round=500, tol_frac=0.04):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)

    for rnd in range(n_rounds):
        corrs = compute_corrs(h)
        max_c = corrs.max()
        active_k = np.where(corrs >= max_c * (1.0 - tol_frac))[0]
        if len(active_k) > 30:
            active_k = np.argsort(corrs)[-30:]

        grad = np.zeros(n)
        for k in active_k:
            grad += grad_corr_k(h, k, n)

        # Rank pairs by gradient difference
        i_sorted = np.argsort(grad)
        j_sorted = np.argsort(grad)[::-1]

        tried = 0
        improved = False
        for ii in range(n):
            if tried >= max_pairs_per_round:
                break
            i = j_sorted[ii]
            for jj in range(n):
                if tried >= max_pairs_per_round:
                    break
                j = i_sorted[jj]
                if i == j:
                    continue
                tried += 1

                eps_hi = min(h[i], 1.0 - h[j])
                if eps_hi < 1e-10:
                    continue

                d = np.zeros(2 * n - 1)
                for k in active_k:
                    gk = grad_corr_k(h, k, n)
                    d[k] = gk[j] - gk[i]

                # Golden section on [0, eps_hi]
                gr = (np.sqrt(5.0) - 1.0) / 2.0
                a, b = 0.0, eps_hi
                c_pt = b - gr * (b - a)
                d_pt = a + gr * (b - a)
                fc = (corrs + c_pt * d).max()
                fd = (corrs + d_pt * d).max()
                for _ in range(35):
                    if b - a < 1e-14:
                        break
                    if fc < fd:
                        b, d_pt, fd = d_pt, c_pt, fc
                        c_pt = b - gr * (b - a)
                        fc = (corrs + c_pt * d).max()
                    else:
                        a, c_pt, fc = c_pt, d_pt, fd
                        d_pt = a + gr * (b - a)
                        fd = (corrs + d_pt * d).max()

                eps_opt = (a + b) / 2.0
                if eps_opt < 1e-10:
                    continue

                new_h = h.copy()
                new_h[i] -= eps_opt
                new_h[j] += eps_opt
                new_h = np.clip(new_h, 0.0, 1.0)
                new_h = project_feasible(new_h, target_sum)
                new_c5 = c5_of_h(new_h, dx)
                if new_c5 < best_c5 - 1e-14:
                    h = new_h
                    best_c5 = new_c5
                    best_h = h.copy()
                    corrs = compute_corrs(h)
                    max_c = corrs.max()
                    active_k = np.where(corrs >= max_c * (1.0 - tol_frac))[0]
                    if len(active_k) > 30:
                        active_k = np.argsort(corrs)[-30:]
                    grad = np.zeros(n)
                    for k in active_k:
                        grad += grad_corr_k(h, k, n)
                    j_sorted = np.argsort(grad)[::-1]
                    i_sorted = np.argsort(grad)
                    improved = True

        if not improved:
            break

    return best_h, best_c5


def smooth_max_slsqp(h0, dx, target_sum, beta, maxiter=1200):
    n = len(h0)

    def obj(x):
        corrs = compute_corrs(x)
        return logsumexp(beta * corrs) / beta * dx

    def grad_obj(x):
        corrs = compute_corrs(x)
        shifted = corrs - corrs.max()
        w = np.exp(beta * shifted)
        w /= w.sum()
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] > 1e-12:
                grad += w[k] * grad_corr_k(x, k, n)
        return grad * dx

    def c_fun(x):
        return x.sum() - target_sum

    def c_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': c_fun, 'jac': c_jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0, jac=grad_obj, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        h_out = project_feasible(res.x, target_sum)
    except Exception:
        h_out = project_feasible(h0, target_sum)
    return h_out, c5_of_h(h_out, dx)


def fine_pair_descent(h, dx, target_sum, max_rounds=30, max_pairs=300, tol_frac=0.03):
    """Very fine pair descent with small steps for final polishing."""
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)

    for rnd in range(max_rounds):
        corrs = compute_corrs(h)
        max_c = corrs.max()
        active_k = np.where(corrs >= max_c * (1.0 - tol_frac))[0]
        if len(active_k) > 40:
            active_k = np.argsort(corrs)[-40:]

        grad = np.zeros(n)
        for k in active_k:
            grad += grad_corr_k(h, k, n)

        # Only try pairs with large gradient difference
        sorted_idx = np.argsort(np.abs(grad))
        candidates = sorted_idx[::-1]  # highest gradient magnitude first

        dec = np.where(grad > 0)[0]
        inc = np.where(grad < 0)[0]
        dec = dec[np.argsort(grad[dec])[::-1]][:max_pairs // 2]
        inc = inc[np.argsort(grad[inc])][:max_pairs // 2]

        tried = 0
        improved = False
        for i in dec:
            if tried >= max_pairs:
                break
            for j in inc:
                if tried >= max_pairs:
                    break
                if i == j:
                    continue
                tried += 1
                eps_hi = min(h[i], 1.0 - h[j])
                if eps_hi < 1e-10:
                    continue
                # Limit step to be small
                eps_hi = min(eps_hi, 0.05)

                d = np.zeros(2 * n - 1)
                for k in active_k:
                    gk = grad_corr_k(h, k, n)
                    d[k] = gk[j] - gk[i]

                gr = (np.sqrt(5.0) - 1.0) / 2.0
                a, b = 0.0, eps_hi
                c_pt = b - gr * (b - a)
                d_pt = a + gr * (b - a)
                fc = (corrs + c_pt * d).max()
                fd = (corrs + d_pt * d).max()
                for _ in range(30):
                    if b - a < 1e-14:
                        break
                    if fc < fd:
                        b, d_pt, fd = d_pt, c_pt, fc
                        c_pt = b - gr * (b - a)
                        fc = (corrs + c_pt * d).max()
                    else:
                        a, c_pt, fc = c_pt, d_pt, fd
                        d_pt = a + gr * (b - a)
                        fd = (corrs + d_pt * d).max()

                eps_opt = (a + b) / 2.0
                if eps_opt < 1e-12:
                    continue

                new_h = h.copy()
                new_h[i] -= eps_opt
                new_h[j] += eps_opt
                new_h = np.clip(new_h, 0.0, 1.0)
                new_h = project_feasible(new_h, target_sum)
                new_c5 = c5_of_h(new_h, dx)
                if new_c5 < best_c5 - 1e-14:
                    h = new_h
                    best_c5 = new_c5
                    best_h = h.copy()
                    corrs = compute_corrs(h)
                    max_c = corrs.max()
                    active_k = np.where(corrs >= max_c * (1.0 - tol_frac))[0]
                    if len(active_k) > 40:
                        active_k = np.argsort(corrs)[-40:]
                    grad = np.zeros(n)
                    for k in active_k:
                        grad += grad_corr_k(h, k, n)
                    dec = np.where(grad > 0)[0]
                    inc = np.where(grad < 0)[0]
                    dec = dec[np.argsort(grad[dec])[::-1]][:max_pairs // 2]
                    inc = inc[np.argsort(grad[inc])][:max_pairs // 2]
                    improved = True

        if not improved:
            break

    return best_h, best_c5


def momentum_descent(h0, dx, target_sum, max_iter=5000, beta_smooth=50.0, lr_init=0.02):
    n = len(h0)
    h = h0.copy()
    momentum = np.zeros(n)
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    no_improve = 0
    lr = lr_init
    beta_mom = 0.85

    for it in range(max_iter):
        corrs = compute_corrs(h)
        c5_val = corrs.max() * dx
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 2000:
            break

        shifted = corrs - corrs.max()
        w = np.exp(beta_smooth * shifted)
        w /= w.sum()

        grad = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] > 1e-12:
                grad += w[k] * grad_corr_k(h, k, n)
        grad *= dx

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        momentum = beta_mom * momentum - lr * grad / (gnorm + 1e-10)
        h_new = h + momentum
        h_new = project_feasible(h_new, target_sum)
        new_c5 = c5_of_h(h_new, dx)

        tries = 0
        while new_c5 > c5_val + 1e-12 and tries < 5:
            lr *= 0.5
            momentum = beta_mom * momentum - lr * grad / (gnorm + 1e-10)
            h_new = h + momentum
            h_new = project_feasible(h_new, target_sum)
            new_c5 = c5_of_h(h_new, dx)
            tries += 1

        h = h_new

    return best_h, best_c5


def full_optimize(h0, dx, target_sum, time_left_fn, label=""):
    n = len(h0)
    best_h = h0.copy()
    best_c5 = c5_of_h(h0, dx)

    # SLSQP with increasing beta
    for beta in [100, 300, 800, 2000, 4000, 8000]:
        if time_left_fn() < 20:
            break
        h_opt, c5_val = smooth_max_slsqp(best_h, dx, target_sum, beta, 1200)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_opt

    # LP trust region
    if time_left_fn() > 30:
        h_lp, c5_lp = lp_trust_region(best_h, dx, target_sum, max_iters=500, tol_frac=0.02, max_active=60, step_init=0.08)
        if c5_lp < best_c5:
            best_c5 = c5_lp
            best_h = h_lp

    # Systematic pair sweep
    if time_left_fn() > 30:
        h_ps, c5_ps = systematic_pair_sweep(best_h, dx, target_sum, n_rounds=15, max_pairs_per_round=400, tol_frac=0.04)
        if c5_ps < best_c5:
            best_c5 = c5_ps
            best_h = h_ps

    # More LP
    if time_left_fn() > 25:
        h_lp2, c5_lp2 = lp_trust_region(best_h, dx, target_sum, max_iters=400, tol_frac=0.015, max_active=70, step_init=0.05)
        if c5_lp2 < best_c5:
            best_c5 = c5_lp2
            best_h = h_lp2

    # Fine pair descent
    if time_left_fn() > 25:
        h_fd, c5_fd = fine_pair_descent(best_h, dx, target_sum, max_rounds=20, max_pairs=250, tol_frac=0.03)
        if c5_fd < best_c5:
            best_c5 = c5_fd
            best_h = h_fd

    # Momentum
    if time_left_fn() > 40:
        h_mom, c5_mom = momentum_descent(best_h, dx, target_sum, max_iter=3000, beta_smooth=40.0, lr_init=0.015)
        if c5_mom < best_c5:
            best_c5 = c5_mom
            best_h = h_mom

    # Final LP + pair
    if time_left_fn() > 20:
        h_lp3, c5_lp3 = lp_trust_region(best_h, dx, target_sum, max_iters=300, tol_frac=0.01, max_active=80, step_init=0.03)
        if c5_lp3 < best_c5:
            best_c5 = c5_lp3
            best_h = h_lp3
    if time_left_fn() > 15:
        h_fd2, c5_fd2 = fine_pair_descent(best_h, dx, target_sum, max_rounds=15, max_pairs=200, tol_frac=0.02)
        if c5_fd2 < best_c5:
            best_c5 = c5_fd2
            best_h = h_fd2

    return best_h, best_c5


def interpolate_to_n(h_src, n_new):
    n_src = len(h_src)
    x_src = np.linspace(0.0, 2.0, n_src, endpoint=False) + 1.0 / n_src
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False) + 1.0 / n_new
    h_new = np.interp(x_new, x_src, h_src)
    return project_feasible(h_new, n_new / 2.0)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s - (time.time() - t0)

    # Start from parent
    h0 = np.array(initial_h_values, dtype=float)
    n = len(h0)
    dx = 2.0 / n
    target_sum = n / 2.0
    h0 = project_feasible(h0, target_sum)

    best_h = h0.copy()
    best_c5 = c5_of_h(h0, dx)

    # Main optimization on parent
    h_opt, c5_val = full_optimize(h0, dx, target_sum, time_left, "main")
    if c5_val < best_c5:
        best_c5 = c5_val
        best_h = h_opt

    # Try perturbed versions
    n_perturbs = 10
    for trial in range(n_perturbs):
        if time_left() < 30:
            break
        if trial % 3 == 0:
            perturb = 0.04 * (rng.random(n) - 0.5)
        elif trial % 3 == 1:
            perturb = 0.12 * (rng.random(n) - 0.5)
        else:
            idx = rng.choice(n, size=max(5, n // 12), replace=False)
            perturb = np.zeros(n)
            perturb[idx] = rng.choice([-1, 1], size=len(idx)) * 0.08
        h_try = project_feasible(best_h + perturb, target_sum)
        h_opt2, c5_val2 = full_optimize(h_try, dx, target_sum, time_left, "perturb")
        if c5_val2 < best_c5:
            best_c5 = c5_val2
            best_h = h_opt2

    # Try different n values
    n_candidates = [120, 128, 136, 144, 148, 152, 160, 168]
    n_candidates.sort(key=lambda x: abs(x - len(best_h)))
    for n_try in n_candidates:
        if time_left() < 35:
            break
        if n_try == len(best_h):
            continue
        dx_try = 2.0 / n_try
        ts_try = n_try / 2.0
        h_interp = interpolate_to_n(best_h, n_try)
        h_opt3, c5_val3 = full_optimize(h_interp, dx_try, ts_try, time_left, f"n{n_try}")
        if c5_val3 < best_c5:
            best_c5 = c5_val3
            best_h = h_opt3

    # Final intensive polish
    if time_left() > 25 and best_h is not None:
        n_b = len(best_h)
        dx_b = 2.0 / n_b
        ts_b = n_b / 2.0
        h_final, c5_final = full_optimize(best_h, dx_b, ts_b, time_left, "final")
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final

    # Ultra-fine pair transfer
    if time_left() > 15 and best_h is not None:
        n_b = len(best_h)
        dx_b = 2.0 / n_b
        ts_b = n_b / 2.0
        h_pt, c5_pt = fine_pair_descent(best_h, dx_b, ts_b, max_rounds=25, max_pairs=300, tol_frac=0.02)
        if c5_pt < best_c5:
            best_c5 = c5_pt
            best_h = h_pt

    # Final LP with very tight parameters
    if time_left() > 10 and best_h is not None:
        n_b = len(best_h)
        dx_b = 2.0 / n_b
        ts_b = n_b / 2.0
        h_lp_final, c5_lp_final = lp_trust_region(best_h, dx_b, ts_b, max_iters=300, tol_frac=0.005, max_active=100, step_init=0.02)
        if c5_lp_final < best_c5:
            best_c5 = c5_lp_final
            best_h = h_lp_final

    # Final validation
    if best_h is None:
        best_h = h0
        best_c5 = c5_of_h(h0, dx)
    else:
        n_final = len(best_h)
        best_h = project_feasible(best_h, n_final / 2.0)
        best_c5 = c5_of_h(best_h, 2.0 / n_final)

    return best_h, best_c5, len(best_h)
