def soft_objective(h, dx, temperature):
    corrs = np.correlate(h, 1-h, mode='full') * dx
    # Use log-sum-exp for numerical stability
    max_val = corrs.max()
    if temperature * max_val > 700:
        # Scale to avoid overflow
        return corrs.max() + np.log(np.exp(corrs - max_val).sum()) / temperature
    return (1/temperature) * np.log(np.exp(temperature * corrs).sum())

def soft_gradient(h, dx, temperature):

    # Softmax weights
    max_val = corrs.max()

    weights /= weights.sum()
    # Gradient is dx * sum_k weights[k] * dC[k]/dh
    grad = np.zeros(n)
    for k in range(len(corrs)):

            if p-d >= 0 and p-d < n:

def build_lp_direction(h, active_idx, dx, alpha, n):
    """Solve LP to find direction in trust region that decreases active correlations"""
    K = len(active_idx)
    if K == 0:
        return None
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0  # minimize alpha (step size)
    
    A_ub = np.zeros((K, n + 1))
    b_ub = np.zeros(K)
    
    for j, k in enumerate(active_idx):
        d = k - (n - 1)
        g = grad_at_lag(h, d, n)
        A_ub[j, :n] = dx * g
        A_ub[j, -1] = -1.0
        # We want: dx * g . delta <= 0 (decrease the correlation)
        # Actually we want the new value to be <= current: dx*(C[k] + g.delta) <= C[k]*dx
        # So: dx * g.delta <= 0
        b_ub[j] = 0.0
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    bounds = [(-h[i] * alpha, (1.0 - h[i]) * alpha) for i in range(n)]
    bounds.append((0.0, alpha))
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True, 'time_limit': 10})
        if res.status == 0:
            return res.x[:n]
    except Exception:
        pass
    return None

def lp_polish(h, dx, target_sum, alpha, n_iter, budget_s, t0):

import numpy as np
from scipy.optimize import linprog
import time

def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -3.0, 3.0
    for _ in range(70):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def c5_of_h(h, dx):
    return np.correlate(h, 1.0 - h, mode='full').max() * dx

def full_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def grad_at_lag(h, d, n):
    grad = np.zeros(n)
    p_lo = max(0, -d)
    p_hi = min(n, n - d)
    if p_lo < p_hi:
        grad[p_lo:p_hi] += (1.0 - h[p_lo + d: p_hi + d])
    p_lo2 = max(0, d)
    p_hi2 = min(n, n + d)
    if p_lo2 < p_hi2:
        grad[p_lo2:p_hi2] += (-h[p_lo2 - d: p_hi2 - d])
    return grad

def soft_grad(h, dx, temp):
    n = len(h)
    corrs = full_corrs(h) * dx
    max_c = corrs.max()
    exps = np.exp(temp * (corrs - max_c))
    s = exps.sum()
    if s < 1e-300:
        return np.zeros(n)
    weights = exps / s
    grad = np.zeros(n)
    for k in range(len(corrs)):
        if weights[k] < 1e-12:
            continue
        d = k - (n - 1)
        grad += weights[k] * dx * grad_at_lag(h, d, n)
    return grad

def soft_descent(h, dx, target_sum, budget_s, t0):
    n = len(h)
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    cur = h.copy()
    temps = [15, 12, 10, 8, 6, 5, 4, 3, 2.5, 2, 1.5, 1.0, 0.8]
    for temp in temps:
        if budget_s - (time.time() - t0) < 15:
            break
        cur_c5 = c5_of_h(cur, dx)
        step = 0.015
        stalled = 0
        for it in range(80):
            if budget_s - (time.time() - t0) < 10:
                break
            g = soft_grad(cur, dx, temp)
            g = g - g.mean()
            g[cur >= 1.0 - 1e-10] = 0.0
            g[cur <= 1e-10] = 0.0
            g_norm = np.linalg.norm(g)
            if g_norm < 1e-12:
                break
            g = g / g_norm * 0.02
            cur_new = project_feasible(cur + g, target_sum)
            c5_new = c5_of_h(cur_new, dx)
            if c5_new < cur_c5 - 1e-12:
                cur = cur_new
                cur_c5 = c5_new
                stalled = 0
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = cur.copy()
            else:
                stalled += 1
                if stalled > 20:
                    break
    return best_h, best_c5

def lp_min_alpha(h, active_idx, corrs, dx, alpha, n):
    K = len(active_idx)
    if K == 0:
        return None
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_ub = np.zeros((K, n + 1))
    b_ub = np.zeros(K)
    for j, k in enumerate(active_idx):
        d = k - (n - 1)
        g = grad_at_lag(h, d, n)
        A_ub[j, :n] = dx * g
        A_ub[j, -1] = -1.0
        b_ub[j] = -1e-8
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(-h[i] * alpha, (1.0 - h[i]) * alpha) for i in range(n)]
    bounds.append((0.0, alpha))
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 5})
        if res.status == 0:
            return res.x[:n]
    except Exception:
        pass
    return None

def lp_min_norm(h, active_idx, corrs, dx, step, n):
    K = len(active_idx)
    if K == 0:
        return None
    nv = 2 * n + 1
    c_obj = np.zeros(nv)
    c_obj[:n] = 1.0
    c_obj[n:2*n] = 1.0
    A_ub = np.zeros((K, nv))
    b_ub = np.zeros(K)
    for j, k in enumerate(active_idx):
        d = k - (n - 1)
        g = grad_at_lag(h, d, n)
        A_ub[j, :n] = dx * g
        A_ub[j, n:2*n] = -dx * g
        b_ub[j] = -1e-8
    A_eq = np.zeros((1, nv))
    A_eq[0, :n] = 1.0
    A_eq[0, n:2*n] = -1.0
    b_eq = np.zeros(1)
    bounds = []
    for i in range(n):
        bounds.append((0.0, (1.0 - h[i]) * step))
        bounds.append((0.0, h[i] * step))
    bounds.append((0.0, None))
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 5})
        if res.status == 0:
            delta = res.x[:n] - res.x[n:2*n]
            return delta
    except Exception:
        pass
    return None

def lp_polish(h, dx, target_sum, alpha, n_iter, budget_s, t0):
    n = len(h)
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    cur = h.copy()
    cur_c5 = best_c5
    cur_alpha = alpha
    no_improve = 0
    for it in range(n_iter):
        if budget_s - (time.time() - t0) < 3:
            break
        corrs = full_corrs(cur)
        max_c = corrs.max()
        margin = max(0.5, 0.06 * max_c)
        active_mask = corrs >= (max_c - margin)
        n_top = max(10, min(50, int(0.2 * (2 * n - 1))))
        top_idx = np.argsort(corrs)[-n_top:]
        active_mask[top_idx] = True
        active_idx = np.where(active_mask)[0]
        if len(active_idx) < 5:
            active_idx = top_idx
        delta = lp_min_alpha(cur, active_idx, corrs, dx, cur_alpha, n)
        if delta is None:
            delta = lp_min_norm(cur, active_idx, corrs, dx, cur_alpha, n)
        if delta is None:
            cur_alpha *= 0.5
            no_improve += 1
            if cur_alpha < 0.003 or no_improve > 15:
                break
            continue
        improved = False
        for t in [1.0, 0.75, 0.5, 0.4, 0.3, 0.2, 0.1, 0.05, 0.02]:
            h_c = cur + t * delta
            h_c = project_feasible(h_c, target_sum)
            new_c5 = c5_of_h(h_c, dx)
            if new_c5 < cur_c5 - 1e-13:
                cur = h_c
                cur_c5 = new_c5
                improved = True
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h_c.copy()
                no_improve = 0
                break
        if not improved:
            cur_alpha *= 0.65
            no_improve += 1
            if cur_alpha < 0.003 or no_improve > 15:
                break
        else:
            cur_alpha = min(0.4, cur_alpha * 1.08)
    return best_h, best_c5

def pair_moves(h, dx, target_sum, budget_s, t0, n_rounds=5, seed=7):
    n = len(h)
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    rng = np.random.RandomState(seed)
    for rnd in range(n_rounds):
        if budget_s - (time.time() - t0) < 8:
            break
        cur = best_h.copy()
        cur_c5 = best_c5
        n_pairs = 300 + 100 * rnd
        for _ in range(n_pairs):
            if budget_s - (time.time() - t0) < 4:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            direction = rng.choice([1, -1])
            old_i, old_j = cur[i], cur[j]
            found = False
            for delta in [0.02, 0.01, 0.05, 0.005]:
                ni = old_i + direction * delta
                nj = old_j - direction * delta
                if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                    continue
                cur[i], cur[j] = ni, nj
                c5_try = c5_of_h(cur, dx)
                if c5_try < cur_c5 - 1e-13:
                    cur_c5 = c5_try
                    found = True
                    if c5_try < best_c5:
                        best_c5 = c5_try
                        best_h = cur.copy()
                    break
            if not found:
                cur[i], cur[j] = old_i, old_j
    return best_h, best_c5

def anneal_step(h, dx, target_sum, T, rng, n_steps=200):
    n = len(h)
    cur = h.copy()
    cur_c5 = c5_of_h(cur, dx)
    best_cur = cur.copy()
    best_cur_c5 = cur_c5
    for _ in range(n_steps):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        delta = rng.uniform(-0.05, 0.05)
        ni = np.clip(cur[i] + delta, 0, 1)
        nj = np.clip(cur[j] - delta, 0, 1)
        actual_delta = ni - cur[i]
        if abs(actual_delta) < 1e-10:
            continue
        cur[i] = ni
        cur[j] = np.clip(cur[j] - actual_delta, 0, 1)
        cur = project_feasible(cur, target_sum)
        new_c5 = c5_of_h(cur, dx)
        diff = new_c5 - cur_c5
        if diff < 0 or rng.random() < np.exp(-diff / max(T, 1e-10)):
            cur_c5 = new_c5
            if new_c5 < best_cur_c5:
                best_cur_c5 = new_c5
                best_cur = cur.copy()
        else:
            cur = best_cur.copy()
            cur_c5 = best_cur_c5
    return best_cur, best_cur_c5

def transfer_grid(h_src, n_target):
    n_src = len(h_src)
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_tgt = (np.arange(n_target) + 0.5) * (2.0 / n_target)
    return np.interp(x_tgt, x_src, h_src)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def tl():
        return budget_s - (time.time() - t0)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = np.full(120, 0.5)

    n = len(h0)
    dx = 2.0 / n
    target_sum = n / 2.0
    h0 = project_feasible(h0, target_sum)
    best_h = h0.copy()
    best_c5 = c5_of_h(h0, dx)

    # Phase 1: Soft gradient descent for landscape exploration
    if tl() > 120:
        h_soft, c5_soft = soft_descent(best_h, dx, target_sum, budget_s, t0)
        if c5_soft < best_c5:
            best_c5 = c5_soft
            best_h = h_soft

    # Phase 2: LP polish - multiple alpha values
    for alpha_try in [0.25, 0.15, 0.1, 0.07, 0.05, 0.03]:
        if tl() < 40:
            break
        h_lp, c5_lp = lp_polish(best_h, dx, target_sum, alpha_try, 100, budget_s, t0)
        if c5_lp < best_c5:
            best_c5 = c5_lp
            best_h = h_lp

    # Phase 3: Pair coordinate moves
    if tl() > 50:
        h_pm, c5_pm = pair_moves(best_h, dx, target_sum, budget_s, t0, n_rounds=6, seed=seed)
        if c5_pm < best_c5:
            best_c5 = c5_pm
            best_h = h_pm

    # Phase 4: Short annealing to escape local optima
    if tl() > 80:
        n_b4 = len(best_h)
        dx_b4 = 2.0 / n_b4
        ts_b4 = n_b4 / 2.0
        T = 0.005
        for _ in range(8):
            if tl() < 20:
                break
            h_an, c5_an = anneal_step(best_h, dx_b4, ts_b4, T, rng, n_steps=300)
            if c5_an < best_c5:
                best_c5 = c5_an
                best_h = h_an
                # Polish immediately
                h_lp, c5_lp = lp_polish(best_h, dx_b4, ts_b4, 0.08, 40, budget_s, t0)
                if c5_lp < best_c5:
                    best_c5 = c5_lp
                    best_h = h_lp
            T *= 0.7

    # Phase 5: Multi-grid exploration
    n_base = len(best_h)
    n_candidates = []
    for dn in range(-24, 25, 4):
        nc = n_base + dn
        if 60 <= nc <= 200 and nc % 2 == 0:
            n_candidates.append(nc)
    for nc in [72, 80, 88, 96, 104, 112, 120, 128, 132, 136, 140,
               144, 148, 152, 156, 160, 164, 168, 172, 176, 180, 184, 192]:
        if nc not in n_candidates and 60 <= nc <= 200:
            n_candidates.append(nc)
    n_candidates = sorted(set(n_candidates))

    for n_try in n_candidates:
        if tl() < 25:
            break
        if n_try == n_base:
            continue
        dx_t = 2.0 / n_try
        ts_t = n_try / 2.0
        h_interp = transfer_grid(best_h, n_try)
        h_interp = project_feasible(h_interp, ts_t)
        h_opt, c5_opt = lp_polish(h_interp, dx_t, ts_t, 0.12, 60, budget_s, t0)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
            n_base = n_try
            if tl() > 30:
                h_opt2, c5_opt2 = lp_polish(best_h, dx_t, ts_t, 0.05, 40, budget_s, t0)
                if c5_opt2 < best_c5:
                    best_c5 = c5_opt2
                    best_h = h_opt2

    # Phase 6: Final intensive polish on best grid
    n_b = len(best_h)
    dx_b = 2.0 / n_b
    ts_b = n_b / 2.0

    for alpha_final in [0.1, 0.06, 0.04, 0.02, 0.01]:
        if tl() < 15:
            break
        h_f, c5_f = lp_polish(best_h, dx_b, ts_b, alpha_final, 120, budget_s, t0)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f

    # Phase 7: More pair moves after final polish
    if tl() > 20:
        h_pm2, c5_pm2 = pair_moves(best_h, dx_b, ts_b, budget_s, t0, n_rounds=4, seed=seed+1)
        if c5_pm2 < best_c5:
            best_c5 = c5_pm2
            best_h = h_pm2
            if tl() > 15:
                h_f2, c5_f2 = lp_polish(best_h, dx_b, ts_b, 0.03, 60, budget_s, t0)
                if c5_f2 < best_c5:
                    best_c5 = c5_f2
                    best_h = h_f2

    # Phase 8: Perturbation restarts
    if tl() > 40:
        for pscale in [0.04, 0.025, 0.015, 0.01, 0.005]:
            if tl() < 20:
                break
            pert = rng.standard_normal(n_b)
            pert -= pert.mean()
            h_p = project_feasible(best_h + pscale * pert, ts_b)
            h_opt, c5_opt = lp_polish(h_p, dx_b, ts_b, 0.06, 50, budget_s, t0)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt

    # Phase 9: Ultimate fine polish
    if tl() > 15:
        h_ult, c5_ult = lp_polish(best_h, dx_b, ts_b, 0.015, 100, budget_s, t0)
        if c5_ult < best_c5:
            best_c5 = c5_ult
            best_h = h_ult
        if tl() > 10:
            h_ult2, c5_ult2 = lp_polish(best_h, dx_b, ts_b, 0.008, 80, budget_s, t0)
            if c5_ult2 < best_c5:
                best_c5 = c5_ult2
                best_h = h_ult2

    # Final validation
    n_final = len(best_h)
    dx_final = 2.0 / n_final
    ts_final = n_final / 2.0
    best_h = project_feasible(best_h, ts_final)
    best_c5 = c5_of_h(best_h, dx_final)

    return best_h, best_c5, n_final
