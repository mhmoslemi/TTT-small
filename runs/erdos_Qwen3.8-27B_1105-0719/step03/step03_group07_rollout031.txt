                    h[j] = hj

        if not improved:
            break

    return h, best_c

def slsqp_optimize(h0, maxiter=300, active_k=15, tol=1e-12):
    n = len(h0)
    dx = 2.0 / n
    M = 2 * n - 1
    P = profile_h(h0)
    c0 = P.max()

    thresh = c0 - max(0.005, 0.08 * (c0 - np.median(P)))
    active = np.where(P >= thresh)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(P))])
    if len(active) > active_k:
        top_idx = np.argsort(P[active])[-active_k:]
        active = active[top_idx]
    active = np.sort(active)

    def obj(x):
        return x[-1]

    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g

    def eq_fun(x):
        return x[:-1].sum() - n / 2.0

    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g

    def ineq_fun(x):
        h = x[:-1]
        t = x[-1]
        prof = profile_h(h)
        return t - prof[active]

    def ineq_jac(x):
        h = x[:-1]
        prof = profile_h(h)
        dP = np.zeros((M, n))
        for idx, m in enumerate(active):
            d = m - (n - 1)
            if d == 0:
                dP[idx, :] = dx * (1.0 - 2.0 * h)
            elif d > 0:
                for i in range(n - d):
                    dP[idx, i + d] += dx * (1.0 - h[i])
                    dP[idx, i] += -dx * h[i + d]
            else:
                dp = -d
                for j in range(n - dp):
                    dP[idx, j + dp] += dx * (1.0 - h[j])
                    dP[idx, j] += -dx * h[j + dp]
        J = np.zeros((len(active), n + 1))
        J[:, :-1] = -dP[active, :]
        J[:, -1] = 1.0
        return J

    x0 = np.concatenate([h0, [c0 + 1e-4]])
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 1.0)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': tol})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = repair_feasible(res.x[:-1])
            c_new = exact_score(h_new)
            if c_new <= c0 + 1e-10:
                return h_new
    except Exception:
        pass
    return None

def softmax_descent(h, max_iters=100, beta=100.0, momentum=0.3,
                    t_start=None, budget_s=1100, margin=25):
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    vel = np.zeros(n)

    for it in range(max_iters):
        if time.perf_counter() - t_start > budget_s - margin:
            break
        P = profile_h(h)
        Pmax = P.max()
        exp_arg = beta * (P - Pmax)
        exp_arg[exp_arg < -50] = -50
        weights = np.exp(exp_arg)
        S = weights.sum()
        if S < 1e-300:
            weights = np.zeros(len(P))
            weights[int(np.argmax(P))] = 1.0
        else:
            weights /= S

        active = np.where(weights > 0.02)[0]
        if len(active) == 0:
            active = np.argsort(weights)[-8:]

        dP = np.zeros((len(active), n))
        for idx, m in enumerate(active):
            d = m - (n - 1)
            dx = 2.0 / n
            if d == 0:
                dP[idx, :] = dx * (1.0 - 2.0 * h)
            elif d > 0:
                for i in range(n - d):
                    dP[idx, i + d] += dx * (1.0 - h[i])
                    dP[idx, i] += -dx * h[i + d]
            else:
                dp = -d
                for j in range(n - dp):
                    dP[idx, j + dp] += dx * (1.0 - h[j])
                    dP[idx, j] += -dx * h[j + dp]

        w_norm = weights[active]
        w_norm = w_norm / w_norm.sum()
        grad = dP.T @ w_norm
        sum_err = h.sum() - n / 2.0
        grad += 200.0 * sum_err / n
        grad -= grad.mean()

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad /= gnorm
        vel = momentum * vel + grad
        vnorm = np.linalg.norm(vel)
        if vnorm > 0:
            vel /= vnorm

        alpha = 0.05
        improved = False
        for a in [alpha, alpha * 0.5, alpha * 0.25, alpha * 0.1]:
            h_try = repair_feasible(h - a * vel)
            s = exact_score(h_try)
            if s < best_c - 1e-14:
                h = h_try
                best_c = s
                improved = True
                break
        if not improved:
            alpha *= 0.5

    return h, best_c

def binary_local_search(h_cont, max_flips=500, t_start=None, budget_s=1100, margin=25):
    n = len(h_cont)
    h_bin = (h_cont > 0.5).astype(float)
    h_bin = repair_feasible(h_bin)
    c_bin = exact_score(h_bin)

    for _ in range(max_flips):
        if time.perf_counter() - t_start > budget_s - margin:
            break
        improved = False
        ones = np.where(h_bin == 1.0)[0]
        zeros = np.where(h_bin == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        for a in ones[:min(20, len(ones))]:
            if time.perf_counter() - t_start > budget_s - margin:
                break
            for b in zeros[:min(20, len(zeros))]:
                if a == b:
                    continue
                h_bin[a] = 0.0
                h_bin[b] = 1.0
                c_new = exact_score(h_bin)
                if c_new < c_bin - 1e-14:
                    c_bin = c_new
                    improved = True
                    break
                else:
                    h_bin[a] = 1.0
                    h_bin[b] = 0.0
            if improved:
                break
        if not improved:
            break

    return h_bin, c_bin

def symmetrize_shift(h):
    n = len(h)
    half = n // 2
    h_shift = 1.0 - np.roll(h, -half)
    return repair_feasible((h + h_shift) / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()
    rng = np.random.RandomState(seed)

    def time_ok(margin=20):
        return time.perf_counter() - t_start < budget_s - margin

    parent = None
    try:
        parent = np.array(initial_h_values, dtype=np.float64)
    except NameError:
        pass
    except Exception:
        pass

    if parent is None or len(parent) < 4:
        parent = np.full(70, 0.5)

    parent = np.clip(parent, 0.0, 1.0)
    parent = repair_feasible(parent)
    best_h = parent.copy()
    best_c = exact_score(parent)

    # Symmetrize
    if time_ok():
        h_sym = symmetrize_shift(best_h)
        c_sym = exact_score(h_sym)
        if c_sym < best_c - 1e-12:
            best_h = h_sym
            best_c = c_sym

    n = len(best_h)

    # Phase 1: Main pairwise descent
    if time_ok(100):
        h_pd, c_pd = pairwise_descent(best_h, max_sweeps=400, t_start=t_start,
                                       budget_s=budget_s, margin=80)
        if c_pd < best_c - 1e-13:
            best_h = h_pd
            best_c = c_pd

    # Phase 2: SLSQP polish (multiple active set sizes)
    if time_ok(120):
        for ak in [10, 15, 20]:
            if not time_ok(40):
                break
            sol = slsqp_optimize(best_h, maxiter=200, active_k=ak, tol=1e-11)
            if sol is not None:
                c = exact_score(sol)
                if c < best_c - 1e-12:
                    best_h = sol
                    best_c = c
                    # Follow up with pairwise
                    h_pd, c_pd = pairwise_descent(best_h, max_sweeps=200,
                                                   t_start=t_start, budget_s=budget_s, margin=40)
                    if c_pd < best_c - 1e-13:
                        best_h = h_pd
                        best_c = c_pd

    # Phase 3: Softmax gradient descent
    if time_ok(80):
        for beta in [50.0, 100.0, 200.0]:
            if not time_ok(30):
                break
            h_sm, c_sm = softmax_descent(best_h, max_iters=60, beta=beta,
                                          momentum=0.4, t_start=t_start,
                                          budget_s=budget_s, margin=30)
            if c_sm < best_c - 1e-13:
                best_h = h_sm
                best_c = c_sm
                # Polish with pairwise
                h_pd, c_pd = pairwise_descent(best_h, max_sweeps=150,
                                               t_start=t_start, budget_s=budget_s, margin=30)
                if c_pd < best_c - 1e-13:
                    best_h = h_pd
                    best_c = c_pd

    # Phase 4: Dimension exploration
    if time_ok(150):
        n_best = len(best_h)
        cand_dims = []
        for d in range(max(30, n_best - 12), min(120, n_best + 14)):
            if d % 2 == 0 and d != n_best:
                cand_dims.append(d)
        cand_dims.sort(key=lambda d: abs(d - n_best))

        for new_n in cand_dims[:6]:
            if not time_ok(50):
                break
            h_interp = interpolate_to_n(best_h, new_n)
            h_pd, c_pd = pairwise_descent(h_interp, max_sweeps=200,
                                           t_start=t_start, budget_s=budget_s, margin=40)
            if c_pd < best_c - 1e-10:
                best_h = h_pd
                best_c = c_pd
                # SLSQP at this dimension
                sol = slsqp_optimize(best_h, maxiter=200, active_k=12, tol=1e-11)
                if sol is not None:
                    c2 = exact_score(sol)
                    if c2 < best_c - 1e-12:
                        best_h = sol
                        best_c = c2
                        h_pd2, c_pd2 = pairwise_descent(best_h, max_sweeps=150,
                                                         t_start=t_start, budget_s=budget_s, margin=30)
                        if c_pd2 < best_c - 1e-13:
                            best_h = h_pd2
                            best_c = c_pd2

    # Phase 5: Binary-inspired restart
    if time_ok(80):
        n_b = len(best_h)
        for trial in range(3):
            if not time_ok(25):
                break
            h_bin, c_bin = binary_local_search(best_h, max_flips=200,
                                               t_start=t_start, budget_s=budget_s, margin=20)
            # Blend binary with continuous best
            for mix in [0.0, 0.2, 0.5]:
                if not time_ok(15):
                    break
                h_mix = repair_feasible(mix * h_bin + (1.0 - mix) * best_h)
                h_pd, c_pd = pairwise_descent(h_mix, max_sweeps=150,
                                               t_start=t_start, budget_s=budget_s, margin=20)
                if c_pd < best_c - 1e-10:
                    best_h = h_pd
                    best_c = c_pd

    # Phase 6: Final SLSQP + pairwise
    if time_ok(60):
        sol = slsqp_optimize(best_h, maxiter=300, active_k=15, tol=1e-12)
        if sol is not None:
            c = exact_score(sol)
            if c < best_c - 1e-13:
                best_h = sol
                best_c = c
        if time_ok(30):
            h_pd, c_pd = pairwise_descent(best_h, max_sweeps=300,
                                           t_start=t_start, budget_s=budget_s, margin=15)
            if c_pd < best_c - 1e-14:
                best_h = h_pd
                best_c = c_pd

    # Phase 7: One more SLSQP with wider active set
    if time_ok(30):
        sol = slsqp_optimize(best_h, maxiter=200, active_k=25, tol=1e-12)
        if sol is not None:
            c = exact_score(sol)
            if c < best_c - 1e-13:
                best_h = sol
                best_c = c
                h_pd, c_pd = pairwise_descent(best_h, max_sweeps=100,
                                               t_start=t_start, budget_s=budget_s, margin=5)
                if c_pd < best_c - 1e-14:
                    best_h = h_pd
                    best_c = c_pd

    # Final validation
    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)

    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            final_c = min(final_c, float(eval_result))
        elif isinstance(eval_result, tuple) and len(eval_result) >= 1:
            if np.isfinite(eval_result[0]):
                final_c = min(final_c, float(eval_result[0]))
    except Exception:
        pass

    return best_h, final_c, len(best_h)
