import numpy as np
from scipy.optimize import linprog
import time

def project_feasible(x, target_sum):
    n = len(x)
    lo = float(np.min(x)) - 2.0
    hi = float(np.max(x)) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) / 2, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def c5_of_h(h, dx):
    return np.correlate(h, 1.0 - h, mode='full').max() * dx

def grad_lag(h, k, n):
    off = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -off)
    i_hi = min(n - 1, n - 1 - off)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx + off] += (1.0 - h[idx])
    grad[idx] += (-h[idx + off])
    return grad

def solve_soft_lp(h, active_idx, corrs, dx, alpha, rho_q, rho_s, n):
    K = len(active_idx)
    m_vars = n + K + 1  # delta (n), s (K), q (1)
    
    # Gradients for active lags
    grads = np.zeros((K, n))
    for j, k in enumerate(active_idx):
        grads[j] = grad_lag(h, k, n)
    
    # Objective: minimize rho_q * q + rho_s * sum(s_j)
    c_obj = np.zeros(m_vars)
    c_obj[-1] = rho_q  # q variable
    c_obj[n:n+K] = rho_s  # s variables
    
    # Constraints: dx * grad_j . delta + s_j - q <= -r_{a_j}
    # Rearranged: dx * grad_j . delta + s_j - q + r_{a_j} <= 0
    A_ub = np.zeros((K, m_vars))
    b_ub = np.zeros(K)
    for j in range(K):
        A_ub[j, :n] = dx * grads[j]
        A_ub[j, n+j] = 1.0  # s_j
        A_ub[j, -1] = -1.0  # -q
        b_ub[j] = -corrs[active_idx[j]]
    
    # Equality: sum(delta) = 0
    A_eq = np.zeros((1, m_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds
    bounds = [(-h[i] * alpha, (1.0 - h[i]) * alpha) for i in range(n)]
    bounds += [(0.0, None)] * K  # s_j >= 0
    bounds.append((0.0, None))  # q >= 0
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.status != 0:
            return None
        return res.x[:n]
    except Exception:
        return None

def solve_linear_lp(h, active_idx, corrs, dx, alpha, n):
    """Simple linear sum LP: minimize sum dx*grad_j . delta subject to box + sum=0"""
    K = len(active_idx)
    grads = np.zeros((K, n))
    for j, k in enumerate(active_idx):
        grads[j] = grad_lag(h, k, n)
    
    c_obj = np.zeros(n)
    for j in range(K):
        w = max(0.1, corrs[active_idx[j]] - corrs.max() + 0.05)  # weight by proximity to max
        c_obj += dx * w * grads[j]
    
    A_eq = np.zeros((1, n))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    bounds = [(-h[i] * alpha, (1.0 - h[i]) * alpha) for i in range(n)]
    
    try:
        res = linprog(c_obj, A_eq=A_eq, b_eq=b_eq, bounds=bounds,
                      method='highs', options={'presolve': True})
        if res.status != 0:
            return None
        return res.x[:n]
    except Exception:
        return None

def line_search(cur, cur_c5, delta, dx, target_sum):
    """Try to accept direction delta with line search"""
    for t in [1.0, 0.8, 0.6, 0.5, 0.4, 0.3, 0.2, 0.15, 0.1, 0.05, 0.02]:
        h_c = cur + t * delta
        h_c = project_feasible(h_c, target_sum)
        new_c5 = c5_of_h(h_c, dx)
        if new_c5 < cur_c5 - 1e-15:
            return h_c, new_c5
    return None, None

def pair_refine(h, dx, target_sum, n_pairs, rng, budget_s, t0):
    """Try random pair moves with exact evaluation"""
    n = len(h)
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    cur = h.copy()
    cur_c5 = best_c5
    
    for _ in range(n_pairs):
        if budget_s - (time.time() - t0) < 5.0:
            break
        p = rng.randint(n)
        q = rng.randint(n)
        if p == q:
            continue
        a = cur[p]
        b = cur[q]
        # t range: h[p]+t in [0,1], h[q]-t in [0,1]
        t_lo = max(-a, b - 1.0)
        t_hi = min(1.0 - a, b)
        if t_hi - t_lo < 1e-12:
            continue
        
        # Sample t values
        best_t = 0.0
        best_val = cur_c5
        n_samples = 15
        t_samples = np.linspace(t_lo, t_hi, n_samples)
        for t in t_samples:
            if abs(t) < 1e-12:
                continue
            cur[p] = a + t
            cur[q] = b - t
            val = c5_of_h(cur, dx)
            if val < best_val:
                best_val = val
                best_t = t
        cur[p] = a
        cur[q] = b
        
        if abs(best_t) > 1e-12 and best_val < cur_c5 - 1e-15:
            cur[p] = a + best_t
            cur[q] = b - best_t
            cur_c5 = best_val
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = cur.copy()
    
    return best_h, best_c5

def coord_sweep(h, dx, target_sum, n_sweeps, budget_s, t0, rng):
    """Coordinate descent: try pushing each coordinate toward 0 or 1"""
    n = len(h)
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    cur = h.copy()
    cur_c5 = best_c5
    
    for sweep in range(n_sweeps):
        if budget_s - (time.time() - t0) < 10.0:
            break
        order = rng.permutation(n)
        for i in order:

                break
            old_val = cur[i]
            # Try pushing toward 0 or 1
            if old_val > 0.5:
                new_target = 1.0
            else:
                new_target = 0.0

            cur[i] = new_target
            cur = project_feasible(cur, target_sum)
            new_c5 = c5_of_h(cur, dx)
            if new_c5 < cur_c5 - 1e-14:
                cur_c5 = new_c5
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = cur.copy()
            else:
                cur[i] = old_val
                cur = project_feasible(cur, target_sum)
    
    return best_h, best_c5

    n = len(h)

        active_mask[top_idx] = True

        if len(active_idx) < 5:

        if delta is None:
            alpha *= 0.5
            if alpha < 0.005:
                break
            continue

            alpha = min(0.4, alpha * 1.05)

    return best_h, best_c5

import numpy as np
from scipy.optimize import linprog
import time


def project_feasible(x, target_sum):
    n = len(x)
    lo = float(np.min(x)) - 2.0
    hi = float(np.max(x)) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) / 2, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)


def c5_of_h(h, dx):
    return np.correlate(h, 1.0 - h, mode='full').max() * dx


def grad_lag(h, k, n):
    off = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -off)
    i_hi = min(n - 1, n - 1 - off)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx + off] += (1.0 - h[idx])
    grad[idx] += (-h[idx + off])
    return grad


def solve_soft_lp(h, active_idx, corrs, dx, alpha, rho_q, rho_s, n):
    K = len(active_idx)
    m_vars = n + K + 1
    grads = np.zeros((K, n))
    for j, k in enumerate(active_idx):
        grads[j] = grad_lag(h, k, n)
    c_obj = np.zeros(m_vars)
    c_obj[-1] = rho_q
    c_obj[n:n + K] = rho_s
    A_ub = np.zeros((K, m_vars))
    b_ub = np.zeros(K)
    for j in range(K):
        A_ub[j, :n] = dx * grads[j]
        A_ub[j, n + j] = 1.0
        A_ub[j, -1] = -1.0
        b_ub[j] = -corrs[active_idx[j]]
    A_eq = np.zeros((1, m_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(-h[i] * alpha, (1.0 - h[i]) * alpha) for i in range(n)]
    bounds += [(0.0, None)] * K
    bounds.append((0.0, None))
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.status != 0:
            return None
        return res.x[:n]
    except Exception:
        return None


def solve_weighted_lp(h, active_idx, corrs, dx, alpha, n, decay=0.8):
    K = len(active_idx)
    grads = np.zeros((K, n))
    for j, k in enumerate(active_idx):
        grads[j] = grad_lag(h, k, n)
    max_c = corrs.max()
    c_obj = np.zeros(n)
    for j in range(K):
        w = decay ** (max_c - corrs[active_idx[j]])
        c_obj += dx * w * grads[j]
    A_eq = np.zeros((1, n))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(-h[i] * alpha, (1.0 - h[i]) * alpha) for i in range(n)]
    try:
        res = linprog(c_obj, A_eq=A_eq, b_eq=b_eq, bounds=bounds,
                      method='highs', options={'presolve': True})
        if res.status != 0:
            return None
        return res.x[:n]
    except Exception:
        return None

def lp_loop(h, dx, target_sum, alpha, n_iter, budget_s, t0, mode='soft'):

        if c5_opt > best_c5 - 0.002:
            h_opt2, c5_opt2 = pair_refine(h_opt, dx_t, ts_t, 10000, rng, budget_s, t0)
            if c5_opt2 < c5_opt:
                c5_opt = c5_opt2
                h_opt = h_opt2
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
            n_base = n_try

    # Phase 8: Final intensive polish
    n_b = len(best_h)
    dx_b = 2.0 / n_b
    ts_b = n_b / 2.0

    if tl() > 25:
        h_f, c5_f = lp_loop(best_h, dx_b, ts_b, 0.04, 80, budget_s, t0, mode='soft')
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f

    if tl() > 20:
        h_pr2, c5_pr2 = pair_refine(best_h, dx_b, ts_b, 100000, rng, budget_s, t0)
        if c5_pr2 < best_c5:
            best_c5 = c5_pr2
            best_h = h_pr2

    if tl() > 15:
        h_f2, c5_f2 = lp_loop(best_h, dx_b, ts_b, 0.02, 60, budget_s, t0, mode='soft')
        if c5_f2 < best_c5:
            best_c5 = c5_f2
            best_h = h_f2

    if tl() > 10:
        h_pr3, c5_pr3 = pair_refine(best_h, dx_b, ts_b, 50000, rng, budget_s, t0)
        if c5_pr3 < best_c5:
            best_c5 = c5_pr3
            best_h = h_pr3

    # Final validation
    n_final = len(best_h)
    dx_final = 2.0 / n_final
    best_h = project_feasible(best_h, n_final / 2.0)
    best_c5 = c5_of_h(best_h, dx_final)
    return best_h, best_c5, n_final
