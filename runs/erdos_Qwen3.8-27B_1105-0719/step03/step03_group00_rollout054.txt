import numpy as np
from scipy.optimize import linprog
import time

def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        h_val = np.clip(x - mid, 0.0, 1.0)
        if h_val.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h_val = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h_val.sum()
    if abs(diff) > 1e-14:
        mask = (h_val > 1e-12) & (h_val < 1.0 - 1e-12)
        if mask.sum() > 0:
            h_val[mask] += diff / mask.sum()
    return np.clip(h_val, 0.0, 1.0)

def c5_of_h(h, dx):
    return np.correlate(h, 1.0 - h, mode='full').max() * dx

def compute_A(h, n):
    """Compute A[k, p] = partial corr_k / partial h_p for all k, p."""
    n_lags = 2 * n - 1
    A = np.zeros((n_lags, n))
    for k in range(n_lags):
        off = k - (n - 1)
        i_lo = max(0, -off)
        i_hi = min(n - 1, n - 1 - off)
        if i_lo > i_hi:
            continue
        idx = np.arange(i_lo, i_hi + 1)
        A[k, idx] += (1.0 - h[idx + off])
        A[k, idx + off] += (-h[idx])
    return A

def solve_1d(corr, slopes, L, U, n_top=20):
    """Minimize max_k(corr[k] + t*slopes[k]) for t in [L,U]."""
    if U - L < 1e-12:
        return 0.0
    top_idx = np.argsort(corr)[-n_top:]
    c = corr[top_idx]
    s = slopes[top_idx]
    K = len(top_idx)
    
    cands = [L, U, 0.0]
    for i in range(K):
        for j in range(i + 1, K):
            ds = s[i] - s[j]
            if abs(ds) > 1e-14:
                tb = (c[j] - c[i]) / ds
                if L <= tb <= U:
                    cands.append(tb)
    
    cands = np.array(cands)
    cands = np.clip(cands, L, U)
    
    best_t = 0.0
    best_v = np.inf
    for t in cands:
        v = (c + t * s).max()
        if v < best_v:
            best_v = v
            best_t = t
    return best_t

def pair_refine(h, dx, target, n_sweeps=20, n_top=25, q_candidates=15, time_left_fn=None):
    """Exhaustive pair move refinement."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = corr.max() * dx
    
    for sweep in range(n_sweeps):
        if time_left_fn is not None and time_left_fn() < 10:
            break
        
        A = compute_A(h, n)
        # Weighted score for q selection
        top_lags = np.argsort(corr)[-n_top:]
        w = np.maximum(corr[top_lags] - corr.max() + 0.01, 0.001)
        w = w / w.sum()
        wscore = A[top_lags, :].T @ w
        
        # Try pairs in order of potential
        # For each p, best q has high wscore
        q_order = np.argsort(-wscore)
        
        any_improved = False
        for p in range(n):
            if time_left_fn is not None and time_left_fn() < 5:
                break
            tried = 0
            for q in q_order:
                if q == p:
                    continue
                tried += 1
                if tried > q_candidates:
                    break
                
                slopes = A[:, p] - A[:, q]
                L = max(-h[p], h[q] - 1.0)
                U = min(1.0 - h[p], h[q])
                if U - L < 1e-10:
                    continue
                
                t_opt = solve_1d(corr, slopes, L, U, n_top)
                if abs(t_opt) < 1e-12:
                    continue
                
                h[p] += t_opt
                h[q] -= t_opt
                h = project_feasible(h, target)
                
                corr = np.correlate(h, 1.0 - h, mode='full')
                new_max = corr.max()
                if new_max < best_c5:
                    best_c5 = new_max
                any_improved = True
                # Recompute A (expensive but necessary for correctness)
                # Actually, let's not recompute A every time - just every few moves
                # For now, recompute (it's fast for n=160)
                A = compute_A(h, n)
                # Update wscore
                top_lags = np.argsort(corr)[-n_top:]
                w = np.maximum(corr[top_lags] - corr.max() + 0.01, 0.001)
                w = w / w.sum()
                wscore = A[top_lags, :].T @ w
                q_order = np.argsort(-wscore)
                break  # After one acceptance, restart q search for this p
    
        if not any_improved:
            break
    
    return h, best_c5

def lp_soft_slack(h, dx, target, alpha, n_iters, time_left_fn):
    """LP with soft slack variables for balanced descent."""
    n = len(h)
    n_lags = 2 * n - 1
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = corr.max() * dx
    
    for it in range(n_iters):
        if time_left_fn() < 5:
            break
        
        # Select active lags
        max_c = corr.max()
        margin = 0.05 * max_c
        active = np.where(corr >= max_c - margin)[0]
        if len(active) < 10:
            active = np.argsort(corr)[-30:]
        
        K = len(active)
        # Build LP: min rho_q * q + sum rho_j * s_j
        # s.t. dx * grad_j . delta + s_j - q <= -corr[j] for j in active
        #      sum(delta) = 0
        #      -h[i]*alpha <= delta[i] <= (1-h[i])*alpha
        #      s_j >= 0
        
        # m_vars = n (delta) + K (s) + 1 (q)
        m_vars = n + K + 1
        
        # Build gradient matrix for active lags
        G = np.zeros((K, n))
        for j, k in enumerate(active):
            off = k - (n - 1)
            i_lo = max(0, -off)
            i_hi = min(n - 1, n - 1 - off)
            if i_lo <= i_hi:
                idx = np.arange(i_lo, i_hi + 1)
                G[j, idx] += (1.0 - h[idx + off])
                G[j, idx + off] += (-h[idx])
        
        # Objective: [0...0, rho_0..rho_{K-1}, rho_q]
        rho_q = 0.35
        rho_s = np.ones(K) * 0.05
        c_obj = np.zeros(m_vars)
        c_obj[n:n+K] = rho_s
        c_obj[n+K] = rho_q
        
        # Inequality constraints: dx*G_j . delta + s_j - q <= -corr[j]
        A_ub = np.zeros((K, m_vars))
        b_ub = np.zeros(K)
        for j in range(K):
            A_ub[j, :n] = dx * G[j]
            A_ub[j, n+j] = 1.0
            A_ub[j, n+K] = -1.0
            b_ub[j] = -corr[j]
        
        # Equality: sum(delta) = 0
        A_eq = np.zeros((1, m_vars))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)
        
        # Bounds
        bounds = []
        for i in range(n):
            bounds.append((-h[i] * alpha, (1.0 - h[i]) * alpha))
        for j in range(K):
            bounds.append((0.0, None))  # s_j >= 0
        bounds.append((0.0, None))  # q >= 0
        
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if res.status != 0:
                continue
            delta = res.x[:n]
        except:
            continue
        
        # Line search
        improved = False
        for t in [1.0, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05]:
            h_new = project_feasible(h + t * delta, target)
            c5_new = c5_of_h(h_new, dx)
            if c5_new < best_c5 - 1e-13:
                h = h_new
                best_c5 = c5_new
                corr = np.correlate(h, 1.0 - h, mode='full')
                improved = True
                break
        if not improved:
            alpha *= 0.7
            if alpha < 0.01:
                break
    
    return h, best_c5

import numpy as np
from scipy.optimize import linprog
import time

def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        h_val = np.clip(x - mid, 0.0, 1.0)
        if h_val.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h_val = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h_val.sum()
    if abs(diff) > 1e-14:
        mask = (h_val > 1e-12) & (h_val < 1.0 - 1e-12)
        if mask.sum() > 0:
            h_val[mask] += diff / mask.sum()
    return np.clip(h_val, 0.0, 1.0)

def c5_of_h(h, dx):
    return np.correlate(h, 1.0 - h, mode='full').max() * dx

def compute_A(h, n):
    n_lags = 2 * n - 1
    A = np.zeros((n_lags, n))
    for k in range(n_lags):
        off = k - (n - 1)
        i_lo = max(0, -off)
        i_hi = min(n - 1, n - 1 - off)
        if i_lo > i_hi:
            continue
        idx = np.arange(i_lo, i_hi + 1)
        A[k, idx] += (1.0 - h[idx + off])
        A[k, idx + off] += (-h[idx])
    return A

def solve_1d_minimax(corr, slopes, L, U, n_top=25):
    if U - L < 1e-12:
        return 0.0
    top_idx = np.argsort(corr)[-n_top:]
    c = corr[top_idx]
    s = slopes[top_idx]
    K = len(top_idx)
    cands = [L, U, 0.0]
    for i in range(K):
        for j in range(i + 1, K):
            ds = s[i] - s[j]
            if abs(ds) > 1e-14:
                tb = (c[j] - c[i]) / ds
                if L - 1e-12 <= tb <= U + 1e-12:
                    cands.append(np.clip(tb, L, U))
    cands = np.array(cands)
    cands = np.clip(cands, L, U)
    best_t = 0.0
    best_v = np.inf
    for t in cands:
        v = (c + t * s).max()
        if v < best_v - 1e-15:
            best_v = v
            best_t = t
    return best_t

def pair_refine(h, dx, target, n_sweeps, n_top, q_cand, time_left_fn):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = corr.max() * dx
    n_lags = 2 * n - 1
    for sweep in range(n_sweeps):
        if time_left_fn() < 8:
            break
        A = compute_A(h, n)
        top_lags = np.argsort(corr)[-n_top:]
        w = np.maximum(corr[top_lags] - corr.max() + 0.02, 0.001)
        w = w / w.sum()
        wscore = A[top_lags, :].T @ w
        q_order = np.argsort(-wscore)
        any_imp = False
        for p in range(n):
            if time_left_fn() < 5:
                break
            best_q = -1
            best_t = 0.0
            best_new_max = np.inf
            tried = 0
            for q in q_order:
                if q == p:
                    continue
                tried += 1
                if tried > q_cand:
                    break
                slopes = A[:, p] - A[:, q]
                L = max(-h[p], h[q] - 1.0)
                U = min(1.0 - h[p], h[q])
                if U - L < 1e-10:
                    continue
                t_opt = solve_1d_minimax(corr, slopes, L, U, n_top)
                if abs(t_opt) < 1e-12:
                    continue
                h_trial = h.copy()
                h_trial[p] += t_opt
                h_trial[q] -= t_opt
                h_trial = project_feasible(h_trial, target)
                new_corr = np.correlate(h_trial, 1.0 - h_trial, mode='full')
                nm = new_corr.max()
                if nm < best_new_max:
                    best_new_max = nm
                    best_q = q
                    best_t = t_opt
            if best_q >= 0 and best_new_max < corr.max() - 1e-12:
                h[best_q] -= best_t
                h[p] += best_t
                h = project_feasible(h, target)
                corr = np.correlate(h, 1.0 - h, mode='full')
                if corr.max() < best_c5:
                    best_c5 = corr.max() * dx
                any_imp = True
                A = compute_A(h, n)
                top_lags = np.argsort(corr)[-n_top:]
                w = np.maximum(corr[top_lags] - corr.max() + 0.02, 0.001)
                w = w / w.sum()
                wscore = A[top_lags, :].T @ w
                q_order = np.argsort(-wscore)
        if not any_imp:

    return h, best_c5

def binary_swap_refine(h, dx, target, max_iter, time_left_fn):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = corr.max() * dx
    for it in range(max_iter):
        if time_left_fn() < 5:
            break
        A = compute_A(h, n)
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        best_nm = np.inf
        best_p = -1
        best_q = -1
        for p in ones:
            for q in zeros:
                delta = A[:, q] - A[:, p]
                nm = (corr + delta).max()
                if nm < best_nm:
                    best_nm = nm
                    best_p = p
                    best_q = q
        if best_p < 0 or best_nm >= corr.max() - 1e-12:

        h[best_p] = 0.0
        h[best_q] = 1.0
        h = project_feasible(h, target)

            best_c5 = corr.max() * dx

def lp_soft_slack(h, dx, target, alpha, n_iters, time_left_fn):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = corr.max() * dx
    cur_alpha = alpha
    for it in range(n_iters):
        if time_left_fn() < 5:

        max_c = corr.max()
        margin = max(0.03 * max_c, 0.005)
        active = np.where(corr >= max_c - margin)[0]
        if len(active) < 10:

        K = len(active)
        G = np.zeros((K, n))
        for j, k in enumerate(active):
            off = k - (n - 1)
            i_lo = max(0, -off)
            i_hi = min(n - 1, n - 1 - off)
            if i_lo <= i_hi:

                G[j, idx] += (1.0 - h[idx + off])
                G[j, idx + off] += (-h[idx])
        m_vars = n + K + 1
        rho_q = 0.35
        c_obj = np.zeros(m_vars)
        c_obj[n:n+K] = 0.05
        c_obj[n+K] = rho_q
        A_ub = np.zeros((K, m_vars))
        b_ub = np.zeros(K)
        for j in range(K):
            A_ub[j, :n] = dx * G[j]
            A_ub[j, n+j] = 1.0
            A_ub[j, n+K] = -1.0
            b_ub[j] = -corr[j]
        A_eq = np.zeros((1, m_vars))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)
        bounds = [(-h[i]*cur_alpha, (1.0-h[i])*cur_alpha) for i in range(n)]
        bounds += [(0.0, None)] * K

                continue
            delta = res.x[:n]

            cur_alpha *= 0.7

        h_final, c5_final = pair_refine(best_h, 2.0/len(best_h), len(best_h)/2.0,
                                         30, 35, 35, tl)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final
        if tl() > 5:
            h_fin2, c5_fin2 = lp_soft_slack(best_h, 2.0/len(best_h), len(best_h)/2.0,
                                             0.02, 80, tl)
            if c5_fin2 < best_c5:
                best_c5 = c5_fin2
                best_h = h_fin2
            if tl() > 3:
                h_fin3, c5_fin3 = pair_refine(best_h, 2.0/len(best_h), len(best_h)/2.0,
                                               20, 30, 25, tl)
                if c5_fin3 < best_c5:
                    best_c5 = c5_fin3
                    best_h = h_fin3

    # Final validation
    n_final = len(best_h)
    dx_final = 2.0 / n_final
    best_h = project_feasible(best_h, n_final / 2.0)
    best_c5 = c5_of_h(best_h, dx_final)
    return best_h, best_c5, n_final
