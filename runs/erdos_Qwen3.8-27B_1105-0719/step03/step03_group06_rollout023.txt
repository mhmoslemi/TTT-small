def repair_feasible(vec):

    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-15:
        return h
    if diff > 0:

        slack = 1.0 - h
        order = np.argsort(-slack)
        remaining = diff
        for i in range(n):
            idx = order[i]
            if slack[idx] <= 0:
                continue
            add = min(remaining, slack[idx])
            h[idx] += add
            slack[idx] -= add

def repair_feasible(vec):
    n = len(vec)
    h = np.nan_to_num(np.asarray(vec, dtype=np.float64), nan=0.5)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-15:

    if diff > 0:

        order = np.argsort(-slack)
        remaining = diff
        for idx in order:
            if remaining <= 1e-15:

            remaining -= a
    else:

            h[idx] -= a
            remaining -= a

- At current h, compute P = profile(h) and G = grad_rows(h)
- Linearize: P_m(h + d) ≈ P_m(h) + G[m] · d
- Minimize tau subject to:

  - |d_i| ≤ delta (trust region)
  - 0 ≤ h_i + d_i ≤ 1 (box)

- Constraints: M inequalities (from linearization), 1 equality (mass), 2n inequalities (box)

```python
def lp_trust_step(h, delta, target_max):
    """Solve LP trust-region step to reduce the max profile value."""
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    P = profile(h)
    G = grad_rows(h, n)
    
    # Variables: [d_0, ..., d_{n-1}, tau]
    # Minimize tau
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    # Inequality constraints: G[m]·d - tau <= -P[m]
    # i.e., G[m]·d - tau + P[m] <= 0
    # In linprog form: A_ub @ x <= b_ub
    A_ub = np.zeros((M, n + 1))
    A_ub[:, :n] = G
    A_ub[:, -1] = -1.0
    b_ub = -P
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    # Bounds on d_i: max(-delta, -h_i) <= d_i <= min(delta, 1-h_i)
    bounds = []
    for i in range(n):
        lo = max(-delta, -h[i])
        hi = min(delta, 1.0 - h[i])
        bounds.append((lo, hi))
    # tau is unbounded below (but effectively bounded by the constraints)
    bounds.append((target_max - 1.0, target_max + 1.0))
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.success and res.x is not None:
        d = res.x[:n]
        tau_pred = res.x[-1]
        return d, tau_pred
    return None, None
```

def repair_feasible(vec):
    n = len(vec)
    h = np.nan_to_num(np.asarray(vec, dtype=np.float64), nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        slack = 1.0 - h
        order = np.argsort(-slack)
        remaining = diff
        for idx in order:
            if remaining <= 1e-15:
                break
            a = min(remaining, slack[idx])
            h[idx] += a
            remaining -= a
    else:
        order = np.argsort(h)
        remaining = -diff

            if remaining <= 1e-15:
                break
            a = min(remaining, h[idx])
            h[idx] -= a
            remaining -= a

    if abs(diff2) > 1e-14:
        idx = int(np.argmax(1.0 - h)) if diff2 > 0 else int(np.argmax(h))
        h[idx] = float(np.clip(h[idx] + diff2, 0.0, 1.0))
    return h

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def profile_of(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def grad_matrix(h):
    """Build full gradient matrix G[m,i] = d(P_m)/dh_i."""
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    G = np.zeros((M, n))
    G[n-1] = dx * (1.0 - 2.0 * h)
    for d in range(1, n):
        m = n - 1 + d
        G[m, :n-d] = dx * (1.0 - h[d:])
        G[m, d:] -= dx * h[:n-d]
        m2 = n - 1 - d
        G[m2, d:] = dx * (1.0 - h[:n-d])
        G[m2, :n-d] -= dx * h[d:]
    return G

    A_ub[:, -1] = -1.0
    b_ub = -P.copy()
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    bounds = []

        lo = max(-delta, -h[i] + 1e-12)
