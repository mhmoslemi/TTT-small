Key ideas to implement:
1. **Epigraph optimization**: Use scipy.optimize.minimize with SLSQP on (h, tau) where we minimize tau with constraints tau >= dx*corrs[k] for a set of active lags. This is different from the smooth-max approach.
2. **LMO vertex seeds**: At current h, compute gradient of active correlations, solve the linear projection to get a binary vertex v, then line-search (1-γ)h + γ*v.
3. **Active set management**: Dynamically update which lags are "active" (near-maximum).

So grad of corrs[k] w.r.t. h[i]:
- For i in the overlap: contribution from h[i] being the first factor: (1 - h[i+offset])
- For i in the overlap: contribution from h[i] being the second factor (i.e., i = j + offset for some j in overlap): -h[j] = -h[i-offset]

Let me code this up properly.

For the LMO vertex jump:
- Compute G = sum_k w_k * dx * J_k (gradient of weighted sum of active correlations w.r.t. h)
- Minimize G^T v subject to 0 <= v <= 1, sum(v) = n/2
- This is solved by sorting G and setting v[i] = 1 where G[i] is smallest, v[i] = 0 where largest, with exactly n/2 ones.
- Then line-search along (1-γ)h + γ*v for γ ∈ [0,1]

This is a nice approach because:
1. The epigraph directly targets the true max (not a smoothed approximation)
2. The LMO gives structurally different starting points (binary vertices)
3. The line search along h→v is exact and cheap to evaluate

- Parent uses SLSQP on h directly

My approach:
- Direct epigraph with true max constraints (not smoothed)
- LMO vertex jumps (global binary moves)
- Multi-start with different n values

Let me also add some additional ideas:
- **Projected subgradient with restart**: Use the exact subgradient (which is the set of convex combinations of active correlation gradients) with momentum, but with periodic restarts to the LMO vertex.

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.special import logsumexp
import time

def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def c5_of_h(h, dx):
    return compute_corrs(h).max() * dx

def grad_corr_k(h, k, n):
    offset = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -offset)
    i_hi = min(n - 1, n - 1 - offset)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx] += (1.0 - h[idx + offset])
    grad[idx + offset] += (-h[idx])
    return grad

def smooth_max_obj(h, dx, beta):
    corrs = compute_corrs(h)
    return logsumexp(beta * corrs) / beta * dx

def smooth_max_grad(h, dx, beta):
    n = len(h)
    corrs = compute_corrs(h)
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones_like(corrs) / len(corrs)
    else:
        w /= w_sum
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] > 1e-13:
            grad += w[k] * grad_corr_k(h, k, n)
    return grad * dx

def epigraph_solve(h0, dx, target_sum, active_k, maxiter=500):
    """Minimize tau s.t. tau >= dx*corrs[k] for k in active_k, sum(h)=target, 0<=h<=1"""
    n = len(h0)
    
    def objective(x):
        return x[n]  # tau
    
    def objective_grad(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g
    
    cons = []
    for k in active_k:
        def c_fun(z, k=k):
            h_z = z[:n]
            corrs = compute_corrs(h_z)
            return z[n] - dx * corrs[k]
        def c_jac(z, k=k):
            h_z = z[:n]
            row = np.zeros(n + 1)
            row[:n] = -dx * grad_corr_k(h_z, k, n)
            row[n] = 1.0
            return row
        cons.append({'type': 'ineq', 'fun': c_fun, 'jac': c_jac})
    
    def eq_fun(x):
        return x[:n].sum() - target_sum
    def eq_jac(x):
        j = np.zeros(n + 1)
        j[:n] = 1.0
        return j
    cons.append({'type': 'eq', 'fun': eq_fun, 'jac': eq_jac})
    
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    x0 = np.concatenate([h0, [c5_of_h(h0, dx) + 0.05]])
    
    try:
        res = minimize(objective, x0, jac=objective_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.success or res.fun < c5_of_h(h0, dx):
            h_out = project_feasible(res.x[:n], target_sum)
            return h_out, c5_of_h(h_out, dx)
    except Exception:
        pass
    return h0, c5_of_h(h0, dx)

def lmo_vertex_jump(h0, dx, target_sum, n_active=40, n_trials=10):
    """Use LMO to find binary vertex, then line search along h->v"""
    n = len(h0)
    corrs = compute_corrs(h0)
    c5_val = c5_of_h(h0, dx)
    
    top_k = np.argsort(corrs)[-n_active:]
    G = np.zeros(n)
    for k in top_k:
        G += grad_corr_k(h0, k, n)
    G *= dx / n_active
    
    # LMO: min G^T v s.t. 0<=v<=1, sum(v)=n/2
    # Solution: v[i]=1 where G[i] is smallest, exactly n/2 of them
    sorted_idx = np.argsort(G)
    v = np.zeros(n)
    v[sorted_idx[:n // 2]] = 1.0
    
    # Line search: (1-gamma)*h0 + gamma*v
    best_h = h0.copy()
    best_c5 = c5_val
    
    gammas = np.linspace(0.05, 0.95, 40)
    for g in gammas:
        h_try = (1.0 - g) * h0 + g * v
        h_try = project_feasible(h_try, target_sum)
        c5_try = c5_of_h(h_try, dx)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try
    
    return best_h, best_c5

def slsqp_direct(h0, dx, target_sum, beta, maxiter):
    n = len(h0)
    
    def obj(x):
        corrs = compute_corrs(x)
        return logsumexp(beta * corrs) / beta * dx
    
    def grad_obj(x):
        return smooth_max_grad(x, dx, beta)
    
    def c_fun(x):
        return x.sum() - target_sum
    def c_jac(x):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': c_fun, 'jac': c_jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0, jac=grad_obj, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        h_out = project_feasible(res.x, target_sum)
    except Exception:
        h_out = project_feasible(h0, target_sum)
    return h_out, c5_of_h(h_out, dx)

def iterative_lp_polish(h0, dx, target_sum, max_iters=200, init_step=0.12):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    step = init_step
    stall_count = 0

    for it in range(max_iters):
        corrs = compute_corrs(h)
        c5_val = corrs.max() * dx
        max_c = corrs.max()

        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            stall_count = 0
        else:
            stall_count += 1
        if stall_count > 40:
            step *= 0.7
            stall_count = 0
            if step < 1e-7:
                break

        tol_frac = 0.05
        tol_val = max_c * (1.0 - tol_frac)
        active_mask = corrs >= tol_val
        active_k = np.where(active_mask)[0]

        if len(active_k) > 80:
            active_k = np.argsort(corrs)[-80:]

        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        A_rows = []
        b_rows = []
        for k in active_k:
            row = np.zeros(n + 1)
            row[:n] = dx * grad_corr_k(h, k, n)
            row[n] = -1.0
            A_rows.append(row)
            b_rows.append(-corrs[k] * dx - 1e-14)

        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)

        bounds = []
        for i in range(n):
            bounds.append((max(-h[i], -step), min(1.0 - h[i], step)))
        bounds.append((0.0, None))

        try:
            res = linprog(c_obj, A_ub=np.array(A_rows), b_ub=np.array(b_rows),
                          A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
        except Exception:
            break
        if res.status != 0:
            break

        delta = res.x[:n]
        if np.max(np.abs(delta)) < 1e-14:
            break

        h_new = project_feasible(h + delta, target_sum)
        new_c5 = c5_of_h(h_new, dx)

        if new_c5 < c5_val - 1e-14:
            h = h_new
            step = min(step * 1.2, init_step)
        else:
            improved = False
            for frac in [0.5, 0.25, 0.1, 0.05]:
                h_new = project_feasible(h + frac * delta, target_sum)
                new_c5 = c5_of_h(h_new, dx)
                if new_c5 < c5_val - 1e-14:
                    h = h_new
                    improved = True
                    break
            if not improved:
                step *= 0.5
                if step < 1e-8:
                    break

    return best_h, best_c5

def pair_transfer_polish(h0, dx, target_sum, n_top=25, n_iter=3):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)

    for iteration in range(n_iter):
        corrs = compute_corrs(h)
        grad = smooth_max_grad(h, dx, 600.0)

        dec_idx = np.argsort(grad)[-n_top:]
        inc_idx = np.argsort(grad)[:n_top]

        improved = False
        for i in dec_idx:
            for j in inc_idx:
                if i == j:
                    continue
                eps_lo = max(-h[i], h[j] - 1.0)  
                # Actually: h[i] - eps >= 0 => eps <= h[i], h[j] + eps <= 1 => eps <= 1-h[j]
                # h[i] >= 0 always, h[j] <= 1 always
                # So: eps >= 0 (we only consider moving from i to j if grad[i] > grad[j])
                # Wait, dec_idx has highest grad (want to decrease), inc_idx has lowest grad (want to increase)
                # Moving eps from i to j: h[i]-eps, h[j]+eps
                # Need: h[i]-eps >= 0 => eps <= h[i]
                #       h[j]+eps <= 1 => eps <= 1-h[j]
                #       eps >= 0
                eps_lo = 0.0
                eps_hi = min(h[i], 1.0 - h[j])
                if eps_hi - eps_lo < 1e-12:
                    continue

                d = np.zeros(2 * n - 1)
                for k in range(2 * n - 1):
                    gk = grad_corr_k(h, k, n)
                    d[k] = gk[j] - gk[i]

                # Minimize max(corrs + eps*d) over eps in [eps_lo, eps_hi]
                # Use golden section search
                gr = (np.sqrt(5.0) - 1.0) / 2.0
                a, b = eps_lo, eps_hi
                c_pt = b - gr * (b - a)
                d_pt = a + gr * (b - a)
                
                def f_ps(eps):
                    return (corrs + eps * d).max()
                
                fc, fd = f_ps(c_pt), f_ps(d_pt)
                for _ in range(50):
                    if b - a < 1e-14:
                        break
                    if fc < fd:
                        b = d_pt
                        d_pt = c_pt
                        fd = fc
                        c_pt = b - gr * (b - a)
                        fc = f_ps(c_pt)
                    else:
                        a = c_pt
                        c_pt = d_pt
                        fc = fd
                        d_pt = a + gr * (b - a)
                        fd = f_ps(d_pt)

                eps_opt = (a + b) / 2.0
                new_c5_rough = (corrs + eps_opt * d).max() * dx
                if new_c5_rough < best_c5 - 1e-14:
                    new_h = h.copy()
                    new_h[i] -= eps_opt
                    new_h[j] += eps_opt
                    new_h = np.clip(new_h, 0.0, 1.0)
                    new_h = project_feasible(new_h, target_sum)
                    new_c5 = c5_of_h(new_h, dx)
                    if new_c5 < best_c5 - 1e-14:
                        h = new_h
                        best_c5 = new_c5
                        best_h = h.copy()
                        corrs = compute_corrs(h)
                        improved = True

        if not improved:
            break

    return best_h, best_c5

def momentum_subgrad(h0, dx, target_sum, max_iter=10000, beta_smooth=30.0):
    n = len(h0)
    h = h0.copy()
    momentum = np.zeros(n)
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    no_improve = 0
    lr = 1.0 / np.sqrt(n)
    beta_mom = 0.9

    for it in range(max_iter):
        corrs = compute_corrs(h)
        c5_val = corrs.max() * dx

        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 5000:
            break

        shifted = corrs - corrs.max()
        w = np.exp(beta_smooth * shifted)
        w /= w.sum()

        grad = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] > 1e-12:
                grad += w[k] * grad_corr_k(h, k, n)
        grad *= dx

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        momentum = beta_mom * momentum - lr * grad / (gnorm + 1e-10)
        h_new = h + momentum
        h_new = project_feasible(h_new, target_sum)
        new_c5 = c5_of_h(h_new, dx)

        tries = 0
        while new_c5 > c5_val + 1e-14 and tries < 4:
            lr *= 0.5
            momentum = beta_mom * momentum - lr * grad / (gnorm + 1e-10)
            h_new = h + momentum
            h_new = project_feasible(h_new, target_sum)
            new_c5 = c5_of_h(h_new, dx)
            tries += 1

        h = h_new

    return best_h, best_c5

def full_optimize(h0, dx, target_sum, time_left_fn, label=""):
    n = len(h0)
    best_h = h0.copy()
    best_c5 = c5_of_h(h0, dx)

    # SLSQP with increasing beta
    for beta in [80, 200, 500, 1200, 2500, 5000]:
        if time_left_fn() < 15:
            break
        h_opt, c5_val = slsqp_direct(best_h, dx, target_sum, beta, 1500)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_opt

    # Epigraph with active set
    if time_left_fn() > 20:
        corrs = compute_corrs(best_h)
        top_k = np.argsort(corrs)[-40:]
        h_ep, c5_ep = epigraph_solve(best_h, dx, target_sum, top_k, maxiter=300)
        if c5_ep < best_c5:
            best_c5 = c5_ep
            best_h = h_ep

    # LMO vertex jump
    if time_left_fn() > 15:
        h_lmo, c5_lmo = lmo_vertex_jump(best_h, dx, target_sum, n_active=40, n_trials=10)
        if c5_lmo < best_c5:
            best_c5 = c5_lmo
            best_h = h_lmo

    # Iterative LP polish
    if time_left_fn() > 20:
        h_lp, c5_lp = iterative_lp_polish(best_h, dx, target_sum, max_iters=300, init_step=0.12)
        if c5_lp < best_c5:
            best_c5 = c5_lp
            best_h = h_lp

    # Pair transfer
    if time_left_fn() > 15:
        h_pt, c5_pt = pair_transfer_polish(best_h, dx, target_sum, n_top=20, n_iter=3)
        if c5_pt < best_c5:
            best_c5 = c5_pt
            best_h = h_pt

    # Momentum subgradient
    if time_left_fn() > 40:
        h_sg, c5_sg = momentum_subgrad(best_h, dx, target_sum, max_iter=10000)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg

    # Second round of epigraph + LMO
    if time_left_fn() > 20:
        corrs = compute_corrs(best_h)
        top_k = np.argsort(corrs)[-50:]
        h_ep2, c5_ep2 = epigraph_solve(best_h, dx, target_sum, top_k, maxiter=500)
        if c5_ep2 < best_c5:
            best_c5 = c5_ep2
            best_h = h_ep2

    if time_left_fn() > 15:
        h_lmo2, c5_lmo2 = lmo_vertex_jump(best_h, dx, target_sum, n_active=30, n_trials=10)
        if c5_lmo2 < best_c5:
            best_c5 = c5_lmo2
            best_h = h_lmo2

    # Final SLSQP + LP
    if time_left_fn() > 20:
        h_f, c5_f = slsqp_direct(best_h, dx, target_sum, 4000, 1500)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f
    if time_left_fn() > 15:
        h_lp2, c5_lp2 = iterative_lp_polish(best_h, dx, target_sum, max_iters=150, init_step=0.08)
        if c5_lp2 < best_c5:
            best_c5 = c5_lp2
            best_h = h_lp2
    if time_left_fn() > 10:
        h_pt2, c5_pt2 = pair_transfer_polish(best_h, dx, target_sum, n_top=15, n_iter=2)
        if c5_pt2 < best_c5:
            best_c5 = c5_pt2
            best_h = h_pt2

    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s - (time.time() - t0)

    best_h = None
    best_c5 = float('inf')

    # Phase 1: Optimize from the given construction
    h0 = np.array(initial_h_values, dtype=float)
    n = len(h0)
    dx = 2.0 / n
    target_sum = n / 2.0
    h0 = project_feasible(h0, target_sum)

    h_opt, c5_val = full_optimize(h0, dx, target_sum, time_left, "main")
    best_h = h_opt
    best_c5 = c5_val

    # Phase 2: LMO-based multi-start
    for trial in range(10):
        if time_left() < 30:
            break
        # Generate LMO vertex from current best
        corrs = compute_corrs(best_h)
        n_act = max(10, min(60, (2*n-1)//3))
        top_k = np.argsort(corrs)[-n_act:]
        G = np.zeros(n)
        for k in top_k:
            G += grad_corr_k(best_h, k, n)
        G *= dx / n_act
        
        # Add random perturbation to G for diversity
        G_pert = G + 0.1 * rng.randn(n)
        sorted_idx = np.argsort(G_pert)
        v = np.zeros(n)
        v[sorted_idx[:n // 2]] = 1.0
        
        # Line search from best_h to v
        for gamma in np.linspace(0.1, 0.9, 8):
            h_try = (1.0 - gamma) * best_h + gamma * v
            h_try = project_feasible(h_try, target_sum)
            c5_try = c5_of_h(h_try, dx)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try
        
        # Also try full optimize from this seed if it was good
        if best_c5 < c5_val - 1e-4:
            h_opt2, c5_val2 = full_optimize(best_h, dx, target_sum, time_left, "lmo")
            if c5_val2 < best_c5:
                best_c5 = c5_val2
                best_h = h_opt2

    # Phase 3: Perturbed versions
    for trial in range(10):
        if time_left() < 25:
            break
        if trial % 3 == 0:
            perturb = 0.05 * (rng.random(n) - 0.5)
        elif trial % 3 == 1:
            perturb = 0.15 * (rng.random(n) - 0.5)
        else:

            perturb = np.zeros(n)
            perturb[idx] = rng.choice([-1, 1], size=len(idx)) * 0.1
        h_try = project_feasible(best_h + perturb, target_sum)
        h_opt2, c5_val2 = full_optimize(h_try, dx, target_sum, time_left, "perturb")
        if c5_val2 < best_c5:
            best_c5 = c5_val2
            best_h = h_opt2

def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(80):

        else:

    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)


def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def c5_of_h(h, dx):
    return compute_corrs(h).max() * dx


def grad_corr_k(h, k, n):
    offset = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -offset)
    i_hi = min(n - 1, n - 1 - offset)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx] += (1.0 - h[idx + offset])
    grad[idx + offset] += (-h[idx])
    return grad


def smooth_max_grad(h, dx, beta):
    n = len(h)
    corrs = compute_corrs(h)
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:

    else:

    grad = np.zeros(n)

        if w[k] > 1e-13:
            grad += w[k] * grad_corr_k(h, k, n)
    return grad * dx


def slsqp_direct(h0, dx, target_sum, beta, maxiter):
    n = len(h0)

    def obj(x):
        corrs = compute_corrs(x)
        return logsumexp(beta * corrs) / beta * dx

    def grad_obj(x):
        return smooth_max_grad(x, dx, beta)

    def c_fun(x):
        return x.sum() - target_sum

    def c_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': c_fun, 'jac': c_jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0, jac=grad_obj, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})

        h_out = project_feasible(h0, target_sum)

        g[n] = 1.0

        inc_idx = np.argsort(grad)[:n_top]

                if i == j:
                    continue
                eps_hi = min(h[i], 1.0 - h[j])
                if eps_hi < 1e-12:
                    continue

                def f_ps(eps):
                    return (corrs + eps * d).max()

                        b = d_pt
                        d_pt = c_pt

                        fc = f_ps(c_pt)
                    else:
                        a = c_pt

                        fc = fd

                        fd = f_ps(d_pt)

    n = len(h0)
    h = h0.copy()
    momentum = np.zeros(n)
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    no_improve = 0
    lr = 1.0 / np.sqrt(n)
    beta_mom = 0.85

    for it in range(max_iter):
        corrs = compute_corrs(h)
        c5_val = corrs.max() * dx

        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 6000:
            break

        shifted = corrs - corrs.max()
        w = np.exp(beta_smooth * shifted)
        w /= w.sum()

        grad = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] > 1e-12:
                grad += w[k] * grad_corr_k(h, k, n)
        grad *= dx

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        momentum = beta_mom * momentum - lr * grad / (gnorm + 1e-10)
        h_new = h + momentum
        h_new = project_feasible(h_new, target_sum)
        new_c5 = c5_of_h(h_new, dx)

        tries = 0
        while new_c5 > c5_val + 1e-14 and tries < 5:
            lr *= 0.5
            momentum = beta_mom * momentum - lr * grad / (gnorm + 1e-10)
            h_new = h + momentum
            h_new = project_feasible(h_new, target_sum)
            new_c5 = c5_of_h(h_new, dx)
            tries += 1

        h = h_new

    return best_h, best_c5


def penalty_epigraph(h0, dx, target_sum, n_active=30, max_iter=2000):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    mu = 10.0

    for it in range(max_iter):
        corrs = compute_corrs(h)
        top_k = np.argsort(corrs)[-n_active:]
        c5_val = corrs.max() * dx

        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()

        grad = np.zeros(n)
        for k in top_k:
            excess = dx * corrs[k] - c5_val
            if excess > 0:
                grad += mu * excess * dx * grad_corr_k(h, k, n)
            else:
                grad += 0.1 * dx * grad_corr_k(h, k, n)
        grad *= dx

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        lr = 1.0 / (mu * gnorm + 1e-10)
        h_new = h - lr * grad
        h_new = project_feasible(h_new, target_sum)
        new_c5 = c5_of_h(h_new, dx)

        if new_c5 < c5_val - 1e-15:
            h = h_new
        else:
            lr *= 0.5
            h_new = h - lr * grad
            h_new = project_feasible(h_new, target_sum)
            if c5_of_h(h_new, dx) < c5_val:
                h = h_new
            else:
                mu *= 0.8
                if mu < 0.1:
                    break

    return best_h, best_c5


def full_optimize(h0, dx, target_sum, time_left_fn, label=""):
    n = len(h0)
    best_h = h0.copy()
    best_c5 = c5_of_h(h0, dx)

    for beta in [100, 300, 800, 2000, 5000]:
        if time_left_fn() < 12:
            break
        h_opt, c5_val = slsqp_direct(best_h, dx, target_sum, beta, 2000)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_opt

    if time_left_fn() > 18:
        corrs = compute_corrs(best_h)
        top_k = np.argsort(corrs)[-30:]
        h_ep, c5_ep = epigraph_solve(best_h, dx, target_sum, top_k, maxiter=300)
        if c5_ep < best_c5:
            best_c5 = c5_ep
            best_h = h_ep

    if time_left_fn() > 12:
        h_lp, c5_lp = iterative_lp_polish(best_h, dx, target_sum, max_iters=250, init_step=0.10)
        if c5_lp < best_c5:
            best_c5 = c5_lp
            best_h = h_lp

    if time_left_fn() > 12:
        h_pt, c5_pt = pair_transfer_polish(best_h, dx, target_sum, n_top=20, n_iter=4)
        if c5_pt < best_c5:
            best_c5 = c5_pt
            best_h = h_pt

    if time_left_fn() > 30:
        h_sg, c5_sg = momentum_subgrad(best_h, dx, target_sum, max_iter=8000, beta_smooth=40.0)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg

    if time_left_fn() > 15:
        h_pen, c5_pen = penalty_epigraph(best_h, dx, target_sum, n_active=25, max_iter=1500)
        if c5_pen < best_c5:
            best_c5 = c5_pen
            best_h = h_pen

    if time_left_fn() > 15:
        h_f, c5_f = slsqp_direct(best_h, dx, target_sum, 6000, 2000)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f

    if time_left_fn() > 12:
        h_lp2, c5_lp2 = iterative_lp_polish(best_h, dx, target_sum, max_iters=200, init_step=0.06)
        if c5_lp2 < best_c5:
            best_c5 = c5_lp2
            best_h = h_lp2

    if time_left_fn() > 10:
        h_pt2, c5_pt2 = pair_transfer_polish(best_h, dx, target_sum, n_top=15, n_iter=3)
        if c5_pt2 < best_c5:
            best_c5 = c5_pt2
            best_h = h_pt2

    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s - (time.time() - t0)

    best_h = None
    best_c5 = float('inf')

    h0 = np.array(initial_h_values, dtype=float)
    n = len(h0)
    dx = 2.0 / n
    target_sum = n / 2.0
    h0 = project_feasible(h0, target_sum)

    h_opt, c5_val = full_optimize(h0, dx, target_sum, time_left, "main")
    best_h = h_opt.copy()
    best_c5 = c5_val

    # LMO-based multi-start
    for trial in range(8):
        if time_left() < 35:
            break
        h_lmo, c5_lmo = lmo_vertex_jump(best_h, dx, target_sum,
                                        n_active=30 + 5 * trial, rng=rng)
        if c5_lmo < best_c5:
            best_c5 = c5_lmo
            best_h = h_lmo

        if time_left() > 40:
            h_opt2, c5_val2 = full_optimize(best_h, dx, target_sum, time_left, "lmo")
            if c5_val2 < best_c5:
                best_c5 = c5_val2
                best_h = h_opt2

    # Perturbed restarts
    for trial in range(12):
        if time_left() < 28:
            break
        if trial % 4 == 0:
            perturb = 0.04 * (rng.random(n) - 0.5)
        elif trial % 4 == 1:
            perturb = 0.12 * (rng.random(n) - 0.5)
        elif trial % 4 == 2:
            idx = rng.choice(n, size=max(8, n // 8), replace=False)
            perturb = np.zeros(n)
            perturb[idx] = rng.choice([-1, 1], size=len(idx)) * 0.08
        else:
            # Structured: flip a contiguous block
            start = rng.randint(0, n - 10)
            length = rng.randint(5, 15)
            perturb = np.zeros(n)
            perturb[start:start + length] = 0.1 * (rng.random(length) - 0.5)

        h_try = project_feasible(best_h + perturb, target_sum)
        h_opt2, c5_val2 = full_optimize(h_try, dx, target_sum, time_left, "perturb")
        if c5_val2 < best_c5:
            best_c5 = c5_val2
            best_h = h_opt2

    # Try different n values
    n_candidates = [60, 80, 100, 110, 120, 130, 140, 150, 160, 170, 180]
    n_candidates.sort(key=lambda x: abs(x - len(best_h)))
    n_candidates = n_candidates[:8]

    for n_try in n_candidates:
        if time_left() < 35:
            break
        if n_try == len(best_h):
            continue
        dx_try = 2.0 / n_try
        ts_try = n_try / 2.0
        x_orig = np.linspace(0.0, 2.0, len(best_h), endpoint=False) + 1.0 / len(best_h)
        x_new = np.linspace(0.0, 2.0, n_try, endpoint=False) + 1.0 / n_try
        h_interp = np.interp(x_new, x_orig, best_h)
        h_interp = project_feasible(h_interp, ts_try)

        h_opt3, c5_val3 = full_optimize(h_interp, dx_try, ts_try, time_left, "n_val")
        if c5_val3 < best_c5:
            best_c5 = c5_val3
            best_h = h_opt3

    # Final intensive polish
    if time_left() > 25 and best_h is not None:
        n_b = len(best_h)
        dx_b = 2.0 / n_b
        ts_b = n_b / 2.0
        h_final, c5_final = full_optimize(best_h, dx_b, ts_b, time_left, "final")
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final

    # Ultra-fine pair transfer
    if time_left() > 12 and best_h is not None:
        n_b = len(best_h)
        dx_b = 2.0 / n_b
        ts_b = n_b / 2.0
        h_pt, c5_pt = pair_transfer_polish(best_h, dx_b, ts_b, n_top=30, n_iter=6)
        if c5_pt < best_c5:
            best_c5 = c5_pt
            best_h = h_pt

    # Last LP polish
    if time_left() > 10 and best_h is not None:
        n_b = len(best_h)
        dx_b = 2.0 / n_b
        ts_b = n_b / 2.0
        h_last, c5_last = iterative_lp_polish(best_h, dx_b, ts_b, max_iters=200, init_step=0.04)
        if c5_last < best_c5:
            best_c5 = c5_last
            best_h = h_last

    if best_h is None:
        best_h = h0
        best_c5 = c5_of_h(h0, 2.0 / len(h0))
    else:
        n_final = len(best_h)
        best_h = project_feasible(best_h, n_final / 2.0)
        best_c5 = c5_of_h(best_h, 2.0 / n_final)

    return best_h, best_c5, len(best_h)
