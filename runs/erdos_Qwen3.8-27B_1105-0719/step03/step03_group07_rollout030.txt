Let me think more carefully about the LP formulation:

At point h, the active lags are those where P_m ≈ max(P). The gradient of P_m with respect to h is g_m (computed via the formulas). We want to find a direction v such that:
- g_m · v < 0 for all active m (reduces the active overlaps)
- sum(v) = 0 (approximately preserves the sum constraint)
- v is bounded (for step size control)

LP formulation:
- Variables: v_0, ..., v_{n-1}, η
- Maximize η
- Subject to: g_m · v + η ≤ 0 for each active m
- sum(v) = 0
- -1 ≤ v_i ≤ 1

This is a standard LP that scipy.optimize.linprog can solve.

After finding v, we do a line search: h_new = h + α·v, clip to [0,1], repair sum, evaluate.

Let me also think about the pair coordinate descent more carefully:

For pair (i,j), with s = h[i] + h[j], the feasible range for t = h[i] is [max(0, s-1), min(1, s)]. The score as a function of t is:
score(t) = max_m P_m(h with h[i]=t, h[j]=s-t)

This is a max of linear functions of t (since each P_m is linear in each h_k), so score(t) is a piecewise-linear convex function. We can find its minimum by checking the breakpoints.

import numpy as np
from scipy.optimize import minimize, linprog
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    prof = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(prof)) * dx

def profile_h(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx])
            h[idx] += add
            remaining -= add
            if remaining <= 1e-14:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-14:
                break
    # Final fix
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def build_grad_for_offset(h, d):
    """Gradient of P_{d} with respect to h, where d is the lag offset.
    d > 0: P = dx * sum_{i=0}^{n-d-1} h[i](1-h[i+d])
    d < 0: P = dx * sum_{i=0}^{n-|d|-1} h[i+|d|](1-h[i])
    d = 0: P = dx * sum h[i](1-h[i])
    """
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    if d == 0:
        g[:] = dx * (1.0 - 2.0 * h)
    elif d > 0:
        for i in range(n - d):
            g[i] += dx * (1.0 - h[i + d])
            g[i + d] -= dx * h[i]
    else:
        dp = -d
        for j in range(n - dp):
            g[j + dp] += dx * (1.0 - h[j])
            g[j] -= dx * h[j + dp]
    return g

def build_grads_for_offsets(h, offsets):
    """Build gradients for multiple offsets. Returns array of shape (len(offsets), n)."""
    n = len(h)
    G = np.zeros((len(offsets), n))
    for k, d in enumerate(offsets):
        G[k] = build_grad_for_offset(h, d)
    return G

def lp_direction(h, active_offsets, max_norm=1.0):
    """Find direction v that reduces all active constraints simultaneously.
    LP: max eta s.t. g_j . v + eta <= 0, sum(v)=0, -max_norm <= v_i <= max_norm
    """
    n = len(h)
    G = build_grads_for_offsets(h, active_offsets)
    m = len(active_offsets)
    
    # Variables: v_0, ..., v_{n-1}, eta (total n+1 vars)
    # c = [0, ..., 0, -1] (maximize eta = minimize -eta)
    c = np.zeros(n + 1)
    c[-1] = -1.0
    
    # Inequality constraints: G @ v + eta <= 0 => G @ v + [0,...,0,1] @ eta <= 0
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = G
    A_ub[:, -1] = 1.0
    b_ub = np.zeros(m)
    
    # Equality: sum(v) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    bounds = [(-max_norm, max_norm)] * n + [(-1.0, 1.0)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success and res.x is not None:
        v = res.x[:n]
        eta = res.x[-1]
        if eta > 1e-6:
            return v
    return None

def line_search_alpha(h, v, h0_score, n_alpha=20):
    """Find best alpha for h + alpha * v."""
    n = len(h)
    best_alpha = 0.0
    best_score = h0_score
    # Try a range of alphas
    max_a = min(1.0, 0.5 / max(np.max(np.abs(v)), 1e-10))
    alphas = np.linspace(0, max_a, n_alpha)
    for a in alphas:
        h_try = repair_feasible(h + a * v)
        s = exact_score(h_try)
        if s < best_score - 1e-14:
            best_score = s
            best_alpha = a
    return best_alpha, best_score

def lp_descent(h, max_iters=200, t_start=None, budget_s=1100):
    """LP-based active constraint descent."""
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    
    for it in range(max_iters):
        if t_start is not None and time.perf_counter() - t_start > budget_s - 30:
            break
        prof = profile_h(h)
        c = prof.max()
        if c < best_c - 1e-14:
            best_c = c
        
        # Find active offsets
        thresh = c - 0.005
        active_mask = prof >= thresh
        active_lags = np.where(active_mask)[0]
        if len(active_lags) == 0:
            active_lags = np.array([np.argmax(prof)])
        
        # Convert lag indices to offsets
        # In full mode, index k corresponds to offset k - (n-1)
        offsets = active_lags - (n - 1)
        
        # Try different active set sizes
        found_direction = False
        for k_top in [len(offsets), 10, 6, 4]:
            if k_top > len(offsets):
                continue
            # Take top k by profile value
            top_idx = np.argsort(prof[active_lags])[-k_top:]
            active_offsets = offsets[top_idx]
            
            v = lp_direction(h, active_offsets, max_norm=1.0)
            if v is not None:
                alpha, new_c = line_search_alpha(h, v, best_c, n_alpha=30)
                if alpha > 0:
                    h = repair_feasible(h + alpha * v)
                    if new_c < best_c:
                        best_c = new_c
                found_direction = True
                break
        
        if not found_direction:
            # Fallback: negative gradient of max
            max_lag = np.argmax(prof)
            offset = max_lag - (n - 1)
            g = build_grad_for_offset(h, offset)
            g -= g.mean()
            v = -g
            alpha, new_c = line_search_alpha(h, v, best_c, n_alpha=20)
            if alpha > 0 and new_c < best_c:
                h = repair_feasible(h + alpha * v)
                best_c = new_c
            else:
                break
    
    h = repair_feasible(h)
    final_c = exact_score(h)
    if final_c < best_c:
        best_c = final_c
    return h, best_c

def pair_descent(h, max_sweeps=50, t_start=None, budget_s=1100):
    """Pair coordinate descent with exact evaluation."""
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    
    for sweep in range(max_sweeps):
        if t_start is not None and time.perf_counter() - t_start > budget_s - 20:
            break
        improved = False
        # Randomize pair order each sweep
        pairs = [(i, j) for i in range(n) for j in range(i+1, n)]
        np.random.shuffle(pairs)
        
        for i, j in pairs:
            if t_start is not None and time.perf_counter() - t_start > budget_s - 15:
                break
            s = h[i] + h[j]
            lo = max(0.0, s - 1.0)
            hi = min(1.0, s)
            if lo >= hi - 1e-12:
                continue
            
            # Grid search over t = h[i], with h[j] = s - t
            n_grid = 11
            ts = np.linspace(lo, hi, n_grid)
            best_t = h[i]
            best_s = best_c
            
            for t in ts:
                h[i] = t
                h[j] = s - t
                sc = exact_score(h)
                if sc < best_s - 1e-14:
                    best_s = sc
                    best_t = t
            
            if best_t != h[i] or best_s < best_c - 1e-14:
                h[i] = best_t
                h[j] = s - best_t
                if best_s < best_c:
                    best_c = best_s
                    improved = True
        
        if not improved:
            break
    
    h = repair_feasible(h)
    final_c = exact_score(h)
    if final_c < best_c:
        best_c = final_c
    return h, best_c

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    return repair_feasible(np.clip(h_new, 0.0, 1.0))

def symmetrize(h):
    return (h + h[::-1].copy()) / 2.0

def make_structured_start(n, rng, strategy='golden'):
    """Create structured starting points."""
    h = np.zeros(n)
    half = n // 2
    if strategy == 'golden':
        golden = 0.6180339887
        positions = set()
        for i in range(half):
            pos = int((i * golden + rng.uniform(0, 0.1)) % 1.0 * n)
            positions.add(pos % n)
            if len(positions) >= half:
                break
        # Fill remaining
        if len(positions) < half:
            remaining = [p for p in range(n) if p not in positions]
            rng.shuffle(remaining)
            for p in remaining:
                positions.add(p)
                if len(positions) >= half:
                    break
        for p in positions:
            h[p] = 1.0
    elif strategy == 'block':
        # Alternating blocks
        block_size = max(1, n // (2 * rng.randint(2, 5)))
        for i in range(n):
            if (i // block_size) % 2 == 0:
                h[i] = 1.0
    elif strategy == 'random':
        indices = rng.choice(n, size=half, replace=False)
        h[indices] = 1.0
    
    return repair_feasible(h)

def slsqp_active(h0, maxiter=200, active_k=10, tol=1e-12):
    """SLSQP with top-K active constraints."""
    n = len(h0)
    dx = 2.0 / n
    M = 2 * n - 1
    P = profile_h(h0)
    c0 = P.max()
    
    thresh = c0 - max(0.003, 0.05 * (c0 - np.median(P)))
    active = np.where(P >= thresh)[0]
    if len(active) == 0:
        active = np.array([np.argmax(P)])
    if len(active) > active_k:
        top_idx = np.argsort(P[active])[-active_k:]
        active = active[top_idx]
    
    def obj(x):
        return x[-1]
    
    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    
    def eq_fun(x):
        return x[:-1].sum() - n / 2.0
    
    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g
    
    def ineq_fun(x):
        h = x[:-1]
        t = x[-1]
        prof = profile_h(h)
        return t - prof[active]
    
    def ineq_jac(x):
        h = x[:-1]
        offsets = active - (n - 1)
        G = build_grads_for_offsets(h, offsets)
        J = np.zeros((len(active), n + 1))
        J[:, :-1] = -G
        J[:, -1] = 1.0
        return J
    
    x0 = np.concatenate([h0, [c0 + 1e-4]])
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 1.0)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': tol})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = repair_feasible(res.x[:-1])
            c_new = exact_score(h_new)
            if c_new <= c0 + 1e-10:
                return h_new
    except Exception:
        pass
    return None

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    prof = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(prof)) * dx

def profile_h(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for idx in order:
            if remaining <= 1e-14:
                break
            add = min(remaining, capacity[idx])
            h[idx] += add
            remaining -= add
    else:
        order = np.argsort(h)
        remaining = -diff
        for idx in order:
            if remaining <= 1e-14:
                break
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

            max_lag = np.argmax(prof)
            offset = max_lag - (n - 1)
            g = build_grad_for_offset(h, offset)
            g -= g.mean()
            gnorm = np.linalg.norm(g)
            if gnorm < 1e-14:
                break
            v = -g / gnorm
            alpha, new_c = line_search_alpha(h, v, best_c, n_alpha=25)
            if alpha > 1e-10 and new_c < best_c:
                h = repair_feasible(h + alpha * v)
                best_c = new_c
            else:
                break
    h = repair_feasible(h)
    final_c = exact_score(h)
    if final_c < best_c:
        best_c = final_c
    return h, best_c

def pair_descent(h, max_sweeps=50, t_start=None, budget_s=1100, rng=None):
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    if rng is None:
        rng = np.random.RandomState(0)
    pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
    for sweep in range(max_sweeps):
        if t_start is not None and time.perf_counter() - t_start > budget_s - 20:
            break
        improved = False
        rng.shuffle(pairs)
        for i, j in pairs:
            if t_start is not None and time.perf_counter() - t_start > budget_s - 15:
                break
            s = h[i] + h[j]
            lo = max(0.0, s - 1.0)
            hi = min(1.0, s)
            if hi - lo < 1e-12:
                continue
            best_t = h[i]
            best_s = best_c
            n_grid = 13
            ts = np.linspace(lo, hi, n_grid)
            for t in ts:
                h[i] = t
                h[j] = s - t
                sc = exact_score(h)
                if sc < best_s - 1e-14:
                    best_s = sc
                    best_t = t
            h[i] = best_t
            h[j] = s - best_t
            if best_s < best_c - 1e-14:
                best_c = best_s
                improved = True
        if not improved:
            break
    h = repair_feasible(h)
    final_c = exact_score(h)
    if final_c < best_c:
        best_c = final_c
    return h, best_c

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    return repair_feasible(np.clip(h_new, 0.0, 1.0))

def symmetrize(h):
    return (h + h[::-1].copy()) / 2.0

def make_structured_start(n, seed, strategy='golden'):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    half = n // 2
    if strategy == 'golden':
        golden = 0.6180339887
        positions = []
        used = set()
        for i in range(n * 2):
            pos = int((i * golden + rng.uniform(0, 0.05)) % 1.0 * n) % n
            if pos not in used:
                used.add(pos)
                positions.append(pos)
                if len(positions) >= half:
                    break
        for p in positions:
            h[p] = 1.0
    elif strategy == 'block':
        nb = rng.randint(2, 5)
        bs = max(1, n // nb)
        for i in range(n):
            if (i // bs) % 2 == 0:
                h[i] = 1.0
    elif strategy == 'random':
        indices = rng.choice(n, size=half, replace=False)
        h[indices] = 1.0
    return repair_feasible(h)

def slsqp_active(h0, maxiter=250, active_k=10, tol=1e-1２):
    n = len(h0)
    dx = 2.0 / n
    M = 2 * n - 1
    P = profile_h(h0)
    c0 = P.max()
    thresh = c0 - max(0.003, 0.05 * (c0 - np.median(P)))
    active = np.where(P >= thresh)[0]
    if len(active) == 0:
        active = np.array([np.argmax(P)])
    if len(active) > active_k:
        top_idx = np.argsort(P[active])[-active_k:]
        active = active[top_idx]

    def obj(x):
        return x[-1]

    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g

    def eq_fun(x):
        return x[:-1].sum() - n / 2.0

    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g

    def ineq_fun(x):
        h = x[:-1]
        t = x[-1]
        prof = profile_h(h)
        return t - prof[active]

    def ineq_jac(x):
        h = x[:-1]
        offsets = active - (n - 1)
        G = build_grads_for_offsets(h, offsets)
        J = np.zeros((len(active), n + 1))
        J[:, :-1] = -G
        J[:, -1] = 1.0
        return J

    x0 = np.concatenate([h0, [c0 + 1e-4]])
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 1.0)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': tol})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = repair_feasible(res.x[:-1])
            c_new = exact_score(h_new)
            if c_new <= c0 + 1e-10:
                return h_new
    except Exception:
        pass
    return None

def gradient_flow(h, max_iters=300, t_start=None, budget_s=1100, beta=80.0):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best_c = exact_score(h)
    alpha = 0.02
    for it in range(max_iters):
        if t_start is not None and time.perf_counter() - t_start > budget_s - 30:
            break
        prof = np.correlate(h, 1.0 - h, mode="full") * dx
        pmax = prof.max()
        exp_arg = beta * (prof - pmax)
        weights = np.exp(exp_arg)
        wsum = weights.sum()
        if wsum < 1e-300:
            weights = np.zeros(len(prof))
            weights[np.argmax(prof)] = 1.0
        else:
            weights /= wsum
        active = np.where(weights > 0.01 / len(prof))[0]
        if len(active) == 0:
            active = np.argsort(weights)[-5:]
        offsets = active - (n - 1)
        G = build_grads_for_offsets(h, offsets)
        wa = weights[active]
        wa = wa / wa.sum()
        grad = G.T @ wa
        sum_err = h.sum() - n / 2.0
        grad += 2.0 * 50.0 * sum_err / n
        grad -= grad.mean()
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad /= gnorm
        improved = False
        for a in [alpha, alpha * 0.5, alpha * 0.25, alpha * 0.1]:
            h_try = repair_feasible(h - a * grad)
            s = exact_score(h_try)
            if s < best_c - 1e-14:
                h = h_try
                best_c = s
                improved = True
                break
            elif s < exact_score(h):
                h = h_try
                if exact_score(h) < best_c:
                    best_c = exact_score(h)
                    improved = True
        if improved:
            alpha = min(alpha * 1.1, 0.06)
        else:
            alpha *= 0.5
            if alpha < 1e-6:
                break
    h = repair_feasible(h)
    fc = exact_score(h)
    if fc < best_c:
        best_c = fc
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()
    rng = np.random.RandomState(seed)

    def time_ok(margin=25):
        return time.perf_counter() - t_start < budget_s - margin

    parent = None
    try:
        parent = np.array(initial_h_values, dtype=np.float64)
    except NameError:
        pass
    except Exception:
        pass

    if parent is None or len(parent) < 4:
        parent = np.full(70, 0.5)

    parent = np.clip(parent, 0.0, 1.0)
    parent = repair_feasible(parent)

    best_h = parent.copy()
    best_c = exact_score(parent)

    if time_ok():
        h_sym = repair_feasible(symmetrize(parent))
        c_sym = exact_score(h_sym)
        if c_sym < best_c - 1e-13:
            best_h = h_sym
            best_c = c_sym

    # Phase 1: LP-based active constraint descent
    if time_ok(120):
        h_lp, c_lp = lp_descent(best_h, max_iters=120, t_start=t_start, budget_s=budget_s)
        if c_lp < best_c - 1e-13:
            best_h = h_lp
            best_c = c_lp

    # Phase 2: Pair coordinate descent
    if time_ok(120):
        best_h, best_c = pair_descent(best_h, max_sweeps=25, t_start=t_start,
                                      budget_s=budget_s, rng=rng)

    # Phase 3: Gradient flow for broader exploration
    if time_ok(100):
        h_gf, c_gf = gradient_flow(best_h, max_iters=150, t_start=t_start,
                                   budget_s=budget_s, beta=100.0)
        if c_gf < best_c - 1e-13:
            best_h = h_gf
            best_c = c_gf

    # Phase 4: LP descent again
    if time_ok(100):
        h_lp, c_lp = lp_descent(best_h, max_iters=100, t_start=t_start, budget_s=budget_s)
        if c_lp < best_c - 1e-13:
            best_h = h_lp
            best_c = c_lp

    # Phase 5: Pair descent
    if time_ok(100):
        best_h, best_c = pair_descent(best_h, max_sweeps=30, t_start=t_start,
                                      budget_s=budget_s, rng=rng)

    # Phase 6: SLSQP polish
    if time_ok(80):
        sol = slsqp_active(best_h, maxiter=250, active_k=12, tol=1e-12)
        if sol is not None:
            c = exact_score(sol)
            if c < best_c - 1e-13:
                best_h = sol
                best_c = c

    # Phase 7: Another LP + pair round
    if time_ok(100):
        h_lp, c_lp = lp_descent(best_h, max_iters=80, t_start=t_start, budget_s=budget_s)
        if c_lp < best_c - 1e-13:
            best_h = h_lp
            best_c = c_lp
        if time_ok(60):
            best_h, best_c = pair_descent(best_h, max_sweeps=25, t_start=t_start,
                                          budget_s=budget_s, rng=rng)

    # Phase 8: Dimension exploration
    if time_ok(150):
        n_curr = len(best_h)
        cand_dims = []
        for d in range(max(40, n_curr - 12), min(120, n_curr + 12)):
            if d % 2 == 0 and d != n_curr:
                cand_dims.append(d)
        cand_dims.sort(key=lambda d: abs(d - n_curr))
        for new_n in cand_dims[:8]:
            if not time_ok(60):
                break
            h_interp = interpolate_to_n(best_h, new_n)
            h_try, c_try = lp_descent(h_interp, max_iters=60, t_start=t_start,
                                      budget_s=budget_s)
            h_try, c_try = pair_descent(h_try, max_sweeps=20, t_start=t_start,
                                        budget_s=budget_s, rng=rng)
            if c_try < best_c - 1e-10:
                best_h = h_try
                best_c = c_try

    # Phase 9: Structured restarts
    if time_ok(120):
        n_best = len(best_h)
        for trial in range(5):
            if not time_ok(30):
                break
            for strat in ['golden', 'block']:
                if not time_ok(15):
                    break
                h_bin = make_structured_start(n_best, seed + 200 + trial, strat)
                for mix in [0.3, 0.55]:
                    if not time_ok(10):
                        break
                    h_mix = repair_feasible(mix * h_bin + (1.0 - mix) * best_h)
                    h_try, c_try = lp_descent(h_mix, max_iters=40, t_start=t_start,
                                              budget_s=budget_s)
                    h_try, c_try = pair_descent(h_try, max_sweeps=10, t_start=t_start,
                                                budget_s=budget_s, rng=rng)
                    if c_try < best_c - 1e-10:
                        best_h = h_try
                        best_c = c_try

    # Phase 10: Final polish
    if time_ok(40):
        h_lp, c_lp = lp_descent(best_h, max_iters=60, t_start=t_start, budget_s=budget_s)
        if c_lp < best_c - 1e-13:
            best_h = h_lp
            best_c = c_lp
    if time_ok(25):
        best_h, best_c = pair_descent(best_h, max_sweeps=15, t_start=t_start,
                                      budget_s=budget_s, rng=rng)
    if time_ok(15):
        sol = slsqp_active(best_h, maxiter=150, active_k=8, tol=1e-13)
        if sol is not None:
            c = exact_score(sol)
            if c < best_c - 1e-14:
                best_h = sol
                best_c = c

    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)

    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            final_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 1:
            if np.isfinite(eval_result[0]):
                final_c = float(eval_result[0])
    except Exception:
        pass

    return best_h, final_c, len(best_h)
