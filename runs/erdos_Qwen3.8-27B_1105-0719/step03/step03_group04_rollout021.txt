def repair_feasible(h):
    n = len(h)
    h = np.clip(np.asarray(h, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, 1.0 - h[idx])
            h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)[::-1]
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    return h

def compute_G(h, n):
    """Compute G[i, col] = ∂R_d/∂h[i] / dx where col = d + (n-1)."""
    center = n - 1

        d_lo = max(-(n-1), i-(n-1))

For our purpose, we want to minimize the max. If the discrete max is always ≥ the continuous max (or vice versa), then higher n gives a tighter bound. But in general, the relationship is complex.

I think the safest approach is to optimize at the parent's n=70, but also try nearby n values.

import numpy as np
from scipy.optimize import linprog
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full")) * dx)

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def repair_feasible(h):
    n = len(h)
    h = np.clip(np.asarray(h, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, 1.0 - h[idx])
            h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)[::-1]
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    return h

def compute_G(h, n):
    center = n - 1
    G = np.zeros((n, 2*n-1))
    for i in range(n):
        d_lo = max(-(n-1), i-(n-1))
        d_hi = min(n-1, i)
        if d_lo <= d_hi:
            idx = np.arange(d_lo, d_hi+1)
            G[i, idx + center] += (1.0 - h[i - idx])
        d_lo2 = max(-(n-1), -i)
        d_hi2 = min(n-1, n-1-i)
        if d_lo2 <= d_hi2:
            idx = np.arange(d_lo2, d_hi2+1)
            G[i, idx + center] -= h[i + idx]
    return G

def find_best_exchange(h, profile, G, c, n):
    """Find the best single pairwise exchange.
    Returns (p, q, alpha, new_c) or None.
    """
    dx = 2.0 / n
    best_imp = 0.0
    best_p, best_q, best_a = -1, -1, 0.0
    
    # Get active set (top lags)
    active = np.argsort(profile)[::-1][:12]
    
    for p in range(n):
        for q in range(p+1, n):
            alpha_max = min(h[p], 1.0 - h[q])
            if alpha_max < 1e-10:
                continue
            
            s = dx * (G[q] - G[p])  # ΔR_d per unit alpha
            
            # Quick check: is there any chance of improvement?
            # The best case is if all active lags decrease
            s_active = s[active]
            if np.all(s_active >= 0):
                continue  # All active lags going up or flat, no improvement
            
            # Find candidate alphas
            candidates = []
            candidates.append(alpha_max)
            candidates.append(0.5 * alpha_max)
            candidates.append(0.25 * alpha_max)
            
            # Crossing points
            downs = active[np.where(s[active] < 0)[0]]
            ups = active[np.where(s[active] > 0)[0]]
            if len(downs) > 0 and len(ups) > 0:
                d_dn = downs[0]  # most negative slope
                d_up = ups[0]  # most positive slope
                ds = s[d_up] - s[d_dn]
                if ds > 1e-15:
                    a_cross = (profile[d_dn] - profile[d_up]) / ds
                    if 0 < a_cross < alpha_max:
                        candidates.append(a_cross)
                        candidates.append(0.8 * a_cross)
                        candidates.append(1.2 * a_cross)
            
            # Evaluate candidates
            for a in candidates:
                if a < 0 or a > alpha_max:
                    continue
                new_c = np.max(profile + a * s)
                imp = c - new_c
                if imp > best_imp:
                    best_imp = imp
                    best_p, best_q, best_a = p, q, a
    
    if best_p < 0:
        return None
    return best_p, best_q, best_a, c - best_imp

def lp_polish(h, t_start, budget_s, max_iters=100):
    """LP-based polish using the active set."""
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best_c = exact_score(h)
    best_h = h.copy()
    radius = 0.02
    
    for it in range(max_iters):
        elapsed = time.perf_counter() - t_start
        if budget_s - elapsed < 5:
            break
        
        profile = compute_profile(h)
        c = np.max(profile)
        G = compute_G(h, n)
        
        # Active set
        active = np.argsort(profile)[::-1][:10]
        
        # LP: minimize t s.t. G[a,:] @ (h-h0) + profile[a] <= t, sum(h)=n/2, 0<=h<=1
        # Variables: delta (n) + t (1)
        M = len(active)
        G_a = G[active, :]
        p_a = profile[active]
        
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        
        A_ub = np.zeros((M, n + 1))
        A_ub[:, :n] = G_a
        A_ub[:, -1] = -1.0
        b_ub = -p_a
        
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)
        
        lower = np.maximum(-h, -radius)
        upper = np.minimum(1.0 - h, radius)
        bounds = list(zip(lower, upper))
        bounds.append((0.0, None))
        
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs', options={'maxiter': 2000})
            if res.success and res.x is not None:
                h_new = repair_feasible(h + res.x[:n])
                c_new = exact_score(h_new)
                if c_new < best_c - 1e-14:
                    h = h_new
                    best_c = c_new
                    best_h = h_new.copy()
                    radius = min(radius * 1.1, 0.08)
                else:
                    radius *= 0.8
                    if radius < 0.001:
                        break
        except Exception:
            break
    
    return best_h, best_c

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)
