import numpy as np
from scipy.optimize import minimize, linprog
from scipy.special import logsumexp
import time


def project_feasible(x, target_sum):
    """Water-filling projection onto {0<=x<=1, sum(x)=target_sum}"""
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-15) & (h < 1.0 - 1e-15)
        nm = mask.sum()
        if nm > 0:
            adj = diff / nm
            # Check if adjustment keeps in bounds
            if h[mask].min() + adj >= -1e-15 and h[mask].max() + adj <= 1.0 + 1e-15:
                h[mask] += adj
            else:
                # Fallback: adjust all non-boundary elements
                h[mask] = np.clip(h[mask] + adj, 0.0, 1.0)
    return np.clip(h, 0.0, 1.0)


def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def c5_of_h(h, dx):
    return compute_corrs(h).max() * dx


def grad_corr_k(h, k, n):
    """Gradient of correlation at lag k with respect to h."""
    s = k - (n - 1)
    g = np.zeros(n)
    if s >= 0:
        # c_s = sum_{t=s}^{n-1} h[t]*(1-h[t-s])
        # d/dh[i]: (1-h[i-s]) if i>=s, plus -h[i+s] if i<=n-1-s
        i_lo_fwd = s
        i_hi_fwd = n - 1
        idx_fwd = np.arange(i_lo_fwd, i_hi_fwd + 1)
        g[idx_fwd] += (1.0 - h[idx_fwd - s])
        i_lo_bwd = 0
        i_hi_bwd = n - 1 - s
        if i_lo_bwd <= i_hi_bwd:
            idx_bwd = np.arange(i_lo_bwd, i_hi_bwd + 1)
            g[idx_bwd] += (-h[idx_bwd + s])
    else:
        sp = -s
        # c_s = sum_{t=0}^{n-1-sp} h[t]*(1-h[t+sp])
        # d/dh[i]: (1-h[i+sp]) if i<=n-1-sp, plus -h[i-sp] if i>=sp
        i_lo_fwd = 0
        i_hi_fwd = n - 1 - sp
        if i_lo_fwd <= i_hi_fwd:
            idx_fwd = np.arange(i_lo_fwd, i_hi_fwd + 1)
            g[idx_fwd] += (1.0 - h[idx_fwd + sp])
        i_lo_bwd = sp
        i_hi_bwd = n - 1
        idx_bwd = np.arange(i_lo_bwd, i_hi_bwd + 1)
        g[idx_bwd] += (-h[idx_bwd - sp])
    return g


def smooth_max_obj_and_grad(h, dx, beta):
    """Compute smooth max objective and gradient."""
    corrs = compute_corrs(h)
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    ws = w.sum()
    if ws < 1e-30:
        w = np.ones_like(corrs) / len(corrs)
    else:
        w /= ws
    obj = logsumexp(beta * corrs) / beta * dx
    n = len(h)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] > 1e-12:
            grad += w[k] * grad_corr_k(h, k, n)
    grad *= dx
    return obj, grad


def slsqp_smooth(h0, dx, target_sum, beta, maxiter=2000):
    n = len(h0)
    
    def obj(x):
        o, _ = smooth_max_obj_and_grad(x, dx, beta)
        return o
    
    def grad_obj(x):
        _, g = smooth_max_obj_and_grad(x, dx, beta)
        return g
    
    def c_fun(x):
        return x.sum() - target_sum
    
    def c_jac(x):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': c_fun, 'jac': c_jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0.copy(), jac=grad_obj, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        h_out = project_feasible(res.x, target_sum)
    except Exception:
        h_out = project_feasible(h0, target_sum)
    return h_out, c5_of_h(h_out, dx)


def lp_trust_region(h0, dx, target_sum, max_iters=1000, init_step=0.04, n_active_max=60):
    """LP-based trust region method for the minimax problem."""
    n = len(h0)

    step = init_step
    stall = 0

        c5_val = corrs.max() * dx
        
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 50:
                step *= 0.7
                stall = 0
                if step < 1e-8:
                    break
        
        # Active set: top correlations
        max_c = corrs.max()
        n_active = min(n_active_max, max(10, int(0.3 * (2 * n - 1))))
        active_k = np.argsort(corrs)[-n_active:]

        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        A_rows = []
        b_rows = []

            gk = grad_corr_k(h, k, n)
            row[:n] = dx * gk
            row[n] = -1.0
            A_rows.append(row)
            b_rows.append(-corrs[k] * dx + 1e-12)

            bounds.append((max(-h[i] - 1e-12, -step), min(1.0 - h[i] - 1e-12, step)))
        bounds.append((0.0, None))
        
        try:
            res = linprog(c_obj, A_ub=np.array(A_rows), b_ub=np.array(b_rows),
                          A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
        except Exception:
            break
        if res.status != 0:
            break
        
        delta = res.x[:n]
        t_pred = res.x[-1]

        new_c5 = c5_of_h(h_new, dx)

def exact_pair_transfer(h0, dx, target_sum, n_top=25, n_passes=8):
    """Pair transfer with exact evaluation of max correlation change."""
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    
    for p in range(n_passes):
        corrs = compute_corrs(h)
        # Compute gradient of smooth max for ranking
        beta = 500.0
        shifted = corrs - corrs.max()
        w = np.exp(beta * shifted)
        w /= w.sum()
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] > 1e-10:
                grad += w[k] * grad_corr_k(h, k, n)
        
        dec_idx = np.argsort(grad)[-n_top:]
        inc_idx = np.argsort(grad)[:n_top]
        
        improved = False
        for i in dec_idx:
            for j in inc_idx:
                if i == j:
                    continue
                eps_lo = max(0.0, h[j] - 1.0 + 1e-15)  # h[j]+eps <= 1
                eps_hi = min(h[i], 1.0 - h[j] - 1e-15)  # h[i]-eps >= 0
                if eps_hi - eps_lo < 1e-12:
                    continue
                
                # Compute d_s = grad_corr_k[j] - grad_corr_k[i] for all k
                # Actually, compute exact change in each correlation
                d = np.zeros(2 * n - 1)
                for k in range(2 * n - 1):
                    gk = grad_corr_k(h, k, n)
                    d[k] = gk[j] - gk[i]
                
                # Golden section on f(eps) = max(corrs + eps*d)
                gr = (np.sqrt(5.0) - 1.0) / 2.0
                a, b = eps_lo, eps_hi
                c_pt = b - gr * (b - a)
                d_pt = a + gr * (b - a)
                fc = (corrs + c_pt * d).max()
                fd = (corrs + d_pt * d).max()
                for _ in range(40):
                    if b - a < 1e-14:
                        break
                    if fc < fd:
                        b = d_pt; d_pt = c_pt; fd = fc
                        c_pt = b - gr * (b - a)
                        fc = (corrs + c_pt * d).max()
                    else:
                        a = c_pt; c_pt = d_pt; fc = fd
                        d_pt = a + gr * (b - a)
                        fd = (corrs + d_pt * d).max()
                
                eps_opt = (a + b) / 2.0
                new_c5_est = (corrs + eps_opt * d).max() * dx
                if new_c5_est < best_c5 * (1.0 - 1e-10):
                    new_h = h.copy()

                        corrs = compute_corrs(h)

    return best_h, best_c5


def cyclic_active_descent(h0, dx, target_sum, max_iters=2000, n_conj=5):
    """Cyclic descent focusing on the worst constraint."""
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    
    for it in range(max_iters):

        if c5_val < best_c5 - 1e-15:

        # Find worst lag
        k_worst = np.argmax(corrs)
        g = grad_corr_k(h, k_worst, n)
        
        # Also consider a few nearby lags

        g_avg = np.zeros(n)
        for k in sorted_k:
