def trust_region_lp(h, R=0.05, max_backtracks=10):

    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1

    # For each m: p_m + G_m · d <= t
    # G_m · d - t <= -p_m
    A = np.zeros((M, n + 1))
    b = np.zeros(M)

        A[m, n] = -1.0
        b[m] = -profile[m]

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds
    lower = np.maximum(-R, -h)
    upper = np.minimum(R, 1.0 - h)
    bounds = list(zip(lower, upper)) + [(None, None)]

    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=b_eq,
