Let me also think about what makes the softmax approach different from the LP approach in a way that could find better solutions:

The LP approach: pick the top-K active constraints, solve a trust-region LP to get a direction, then line search. This is like a cutting-plane / active-set method.

The softmax approach: compute a smooth approximation of the max, get a gradient that's a weighted combination of ALL constraint gradients (not just active ones), then line search. This is more like a generalized gradient method.

The key difference: the softmax gradient includes contributions from non-peak constraints (with exponentially decaying weights), which can help "anticipate" which constraints will become active and steer the solution away from them. This is analogous to the idea in interior-point methods where barrier terms keep you away from boundaries.

Another advantage: the softmax parameter β controls the trade-off between "sharp" (β → ∞, equivalent to just tracking the max) and "smooth" (β → 0, equivalent to minimizing the mean). By adapting β during optimization, we can first use a moderate β to get a good global direction, then increase β to polish the peaks.

import numpy as np
import time

def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def c5_of_h(h, dx):
    return np.correlate(h, 1.0 - h, mode='full').max() * dx

def full_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def softmax_gradient(h, beta, dx):
    """Compute the gradient of the log-sum-exp approximation of max correlation."""
    n = len(h)
    corrs = full_corr(h)
    # Softmax weights
    max_c = corrs.max()
    exp_vals = np.exp(beta * (corrs - max_c))
    weights = exp_vals / exp_vals.sum()
    # Gradient
    grad = np.zeros(n)
    for k in range(len(corrs)):
        w = weights[k]
        if w < 1e-12:
            continue
        s = k - (n - 1)
        i_lo = max(0, -s)
        i_hi = min(n - 1, n - 1 - s)
        if i_lo > i_hi:
            continue
        idx = np.arange(i_lo, i_hi + 1)
        grad[idx + s] += w * (1.0 - h[idx])
        grad[idx] += w * (-h[idx + s])
    grad *= dx
    return grad

def softmax_line_search(h, grad, dx, target_sum, base_step=0.1):
    """Backtracking line search in the -grad direction."""
    cur_c5 = c5_of_h(h, dx)
    n = len(h)
    for scale in [1.0, 0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005]:
        t = base_step * scale
        h_new = h - t * grad
        h_new = project_feasible(h_new, target_sum)
        new_c5 = c5_of_h(h_new, dx)
        if new_c5 < cur_c5 - 1e-15:
            return h_new, new_c5, t
    return h, cur_c5, 0.0

def softmax_descent(h, dx, target_sum, beta, n_iter, budget_s, t0):
    """Iterative softmax descent with adaptive beta."""
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    cur_h = h.copy()
    cur_c5 = best_c5
    base_step = 0.05
    
    for it in range(n_iter):
        if budget_s - (time.time() - t0) < 5.0:
            break
        grad = softmax_gradient(cur_h, beta, dx)
        h_new, new_c5, t = softmax_line_search(cur_h, grad, dx, target_sum, base_step)
        if t > 0:
            cur_h = h_new
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = cur_h.copy()
            base_step = min(0.1, base_step * 1.05)
        else:
            base_step *= 0.7
            if base_step < 0.001:

    return best_h, best_c5

def pair_move_search(h, i, j, dx, target_sum):
    """Find the best t for pair move h[i]+=t, h[j]-=t."""
    # Feasible range for t
    t_min = max(-h[i], h[j] - 1.0)
    t_max = min(1.0 - h[i], h[j])
    if t_min >= t_max - 1e-12:
        return h, c5_of_h(h, dx)
    # Evaluate at several points
    cur_c5 = c5_of_h(h, dx)
    best_t = 0.0
    best_new_c5 = cur_c5
    # Try a grid of t values

        h_new = h.copy()
        h_new[i] += t
        h_new[j] -= t
        h_new = project_feasible(h_new, target_sum)
        new_c5 = c5_of_h(h_new, dx)

import numpy as np
import time

def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def c5_of_h(h, dx):
    return np.correlate(h, 1.0 - h, mode='full').max() * dx

def full_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def softmax_gradient(h, beta, dx):
    n = len(h)
    corrs = full_corr(h)
    max_c = corrs.max()
    exp_vals = np.exp(beta * (corrs - max_c))
    exp_sum = exp_vals.sum()
    if exp_sum < 1e-300:
        weights = np.zeros(len(corrs))
        weights[np.argmax(corrs)] = 1.0
    else:
        weights = exp_vals / exp_sum
    grad = np.zeros(n)
    for k in range(len(corrs)):
        w = weights[k]
        if w < 1e-12:
            continue
        s = k - (n - 1)
        i_lo = max(0, -s)
        i_hi = min(n - 1, n - 1 - s)
        if i_lo > i_hi:
            continue
        idx = np.arange(i_lo, i_hi + 1)
        grad[idx + s] += w * (1.0 - h[idx])
        grad[idx] += w * (-h[idx + s])
    grad *= dx
    return grad

def softmax_descent_momentum(h, dx, target_sum, beta, n_iter, budget_s, t0, momentum=0.9):
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    cur_h = h.copy()
    cur_c5 = best_c5
    vel = np.zeros(len(h))
    base_lr = 0.02

    for it in range(n_iter):
        if budget_s - (time.time() - t0) < 5.0:
            break
        grad = softmax_gradient(cur_h, beta, dx)
        vel = momentum * vel - base_lr * grad
        h_proposed = cur_h + vel
        h_proposed = project_feasible(h_proposed, target_sum)
        new_c5 = c5_of_h(h_proposed, dx)
        if new_c5 < cur_c5 - 1e-15:
            cur_h = h_proposed
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = cur_h.copy()
            base_lr = min(0.05, base_lr * 1.02)
        else:
            base_lr *= 0.8
            vel *= 0.5
            if base_lr < 1e-5:
                break
    return best_h, best_c5

def pair_move_eval(h, i, j, t):
    h_new = h.copy()
    h_new[i] += t
    h_new[j] -= t
    return h_new

def pair_moves_optimize(h, dx, target_sum, budget_s, t0, max_rounds=8):

def binary_round_and_eval(h, target_sum):
    n = len(h)
    h_b = (h > 0.5).astype(float)
    n_ones = h_b.sum()
    target = int(target_sum)
    if n_ones > target:
        excess = int(n_ones) - target
        flat = (h_b > 0) & (h < 0.99)
        flat_frac = np.sort(h[flat])[:excess] if excess > 0 else np.array([])
        if len(flat_frac) < excess:

        h_interp = project_feasible(h_interp, ts_t)
        h_opt, c5_opt = softmax_descent_momentum(
            h_interp, dx_t, ts_t, 40.0, 120, budget_s, t0, momentum=0.9)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            n_base = n_try
        h_pm4, c5_pm4 = pair_moves_optimize(best_h, 2.0/len(best_h), len(best_h)/2.0, budget_s, t0, max_rounds=5)
        if c5_pm4 < best_c5:
            best_c5 = c5_pm4
            best_h = h_pm4.copy()

    # Phase 8: Final intensive polish on best grid
    n_final = len(best_h)
    dx_final = 2.0 / n_final
    ts_final = n_final / 2.0

    if remaining() > 20:
        h_f, c5_f = softmax_descent_momentum(
            best_h, dx_final, ts_final, 200.0, 150, budget_s, t0, momentum=0.8)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    if remaining() > 15:
        h_f2, c5_f2 = pair_moves_optimize(best_h, dx_final, ts_final, budget_s, t0, max_rounds=6)
        if c5_f2 < best_c5:
            best_c5 = c5_f2
            best_h = h_f2.copy()

    if remaining() > 15:
        h_f3, c5_f3 = softmax_descent_momentum(
            best_h, dx_final, ts_final, 500.0, 100, budget_s, t0, momentum=0.7)
        if c5_f3 < best_c5:
            best_c5 = c5_f3
            best_h = h_f3.copy()

    # Final validation
    best_h = project_feasible(best_h, ts_final)
    best_c5 = c5_of_h(best_h, dx_final)

    return best_h, best_c5, n_final
