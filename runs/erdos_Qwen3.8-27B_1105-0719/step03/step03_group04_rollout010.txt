    if mode == 0:
        # Alternating blocks
        block = max(2, n // 20)
        pattern = []
        for _ in range(n // block + 1):
            pattern.extend([1.0] * (block // 2))
            pattern.extend([0.0] * (block - block // 2))
        h[:] = pattern[:n]
    elif mode == 1:
        # Fibonacci-like spacing
        phi = (1 + 5**0.5) / 2
        positions = []
        for i in range(n // 2):
            pos = int((i * phi) % n)
            if pos not in positions:
                positions.append(pos)
        for p in positions[:n//2]:
            h[p] = 1.0
    elif mode == 2:
        # Random binary
        positions = rng.choice(n, n//2, replace=False)
        h[positions] = 1.0
    else:
        # Smooth perturbation of 0.5
        h[:] = 0.5 + 0.2 * np.sin(2 * np.pi * np.arange(n) / (n * 0.3))
    return repair_feasible(h)

def slsqp_polish(h, t_start, budget_s):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    c0 = exact_score(h)
    
    def obj_fn(x):
        return x[-1]
    
    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    
    def eq_fun(x):
        return np.array([x[:-1].sum() - n / 2.0])
    
    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g
    
    def ineq_fun(x):
        hv = np.clip(x[:-1], 0.0, 1.0)
        prof = np.correlate(hv, 1.0 - hv, mode="full") * dx
        return x[-1] - prof
    
    def ineq_jac(x):
        hv = np.clip(x[:-1], 0.0, 1.0)
        G = build_gradient(hv)
        J = np.zeros((M, n + 1))
        J[:, :-1] = -G
        J[:, -1] = 1.0
        return J
    
    x0 = np.concatenate([h, [c0 + 1e-3]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    
    try:
        res = minimize(obj_fn, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': 300, 'ftol': 1e-12})
        if res.success or res.x is not None:
            h_sol = repair_feasible(res.x[:-1])
            c_sol = exact_score(h_sol)
            return h_sol, c_sol
    except Exception:
        pass
    return h, c0

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()
    rng = np.random.RandomState(seed)
    
    def time_left():
        return budget_s - (time.perf_counter() - t_start)
    
    # Load parent
    parent = None
    try:
        parent = np.array(initial_h_values, dtype=np.float64)
    except (NameError, Exception):
        pass
    
    if parent is None or len(parent) < 4 or not np.isfinite(parent).all():
        parent = np.full(70, 0.5)
    
    parent = np.clip(parent, 0.0, 1.0)
    parent = repair_feasible(parent)
    c_parent = exact_score(parent)
    
    best_h = parent.copy()
    best_c = c_parent
    
    # Phase 1: Intensive pair descent at parent dimension
    if time_left() > 250:
        h_opt, c_opt = pair_descent(best_h, t_start, budget_s, max_iters=8000)
        if c_opt < best_c - 1e-14:
            best_h = h_opt
            best_c = c_opt
    
    # Phase 2: Try nearby dimensions
    n0 = len(best_h)
    dims = []
    for d in range(max(30, n0 - 10), min(160, n0 + 20), 2):
        if d != n0:
            dims.append(d)
    dims.sort(key=lambda d: abs(d - n0))
    
    for new_n in dims[:8]:
        if time_left() < 80:
            break
        h_interp = interpolate_to_n(best_h, new_n)
        c_interp = exact_score(h_interp)
        h_opt, c_opt = pair_descent(h_interp, t_start, budget_s, max_iters=4000)
        if c_opt < best_c - 1e-14:
            best_h = h_opt
            best_c = c_opt
    
    # Phase 3: Multi-start at best dimension
    n_b = len(best_h)
    if time_left() > 150:
        starts = []
        # Symmetrized
        starts.append(repair_feasible((best_h + best_h[::-1]) / 2.0))
        # Binary starts
        for mode in range(4):
            starts.append(make_binary_start(n_b, rng, mode))
        # Blends
        for b in [0.3, 0.6, 0.8]:
            smooth = 0.5 + 0.25 * np.sin(np.pi * np.arange(n_b) / n_b + 0.5)
            starts.append(repair_feasible((1-b) * best_h + b * smooth))
        
        for h_start in starts:
            if time_left() < 50:
                break
            c_s = exact_score(h_start)
            if c_s < best_c + 0.03:
                h_opt, c_opt = pair_descent(h_start, t_start, budget_s, max_iters=3000)
                if c_opt < best_c - 1e-14:
                    best_h = h_opt
                    best_c = c_opt
    
    # Phase 4: Re-try best dimension with fresh optimization
    if time_left() > 100:
        h_opt, c_opt = pair_descent(best_h, t_start, budget_s, max_iters=6000)
        if c_opt < best_c - 1e-14:
            best_h = h_opt
            best_c = c_opt
    
    # Phase 5: SLSQP polish
    if time_left() > 80:
        h_pol, c_pol = slsqp_polish(best_h, t_start, budget_s)
        if c_pol < best_c - 1e-14:
            best_h = h_pol
            best_c = c_pol
    
    # Phase 6: Final micro pair descent
    if time_left() > 40:
        h_opt, c_opt = pair_descent(best_h, t_start, budget_s, max_iters=5000)
        if c_opt < best_c - 1e-14:
            best_h = h_opt
            best_c = c_opt
    
    # Final validation
    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)
    
    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            final_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 2:
            final_c = float(eval_result[0])
    except Exception:
        pass
    
    return best_h, final_c, len(best_h)
