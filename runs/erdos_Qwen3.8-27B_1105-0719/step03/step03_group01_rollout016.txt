import numpy as np

def project_feasible(x, target_sum):
    n = len(x)
    x = np.clip(x, -5.0, 5.0)
    lo, hi = -5.0, 5.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        cnt = mask.sum()
        if cnt > 0:
            h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)


def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)


def lse_gradient(h, dx, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = beta * corr
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        w /= w_sum
    grad = np.zeros(n)
    for p in range(2 * n - 1):
        if w[p] < 1e-14:
            continue
        off = p - (n - 1)
        j_lo = max(0, -off)
        j_hi = min(n - 1, n - 1 - off)
        if j_lo <= j_hi:
            grad[j_lo:j_hi + 1] += w[p] * (1.0 - h[j_lo + off:j_hi + 1 + off])
        j_lo2 = max(0, off)
        j_hi2 = min(n - 1, n - 1 + off)
        if j_lo2 <= j_hi2:
            grad[j_lo2:j_hi2 + 1] -= w[p] * h[j_lo2 - off:j_hi2 + 1 - off]
    grad *= dx
    lse_val = -(1.0 / beta) * np.log(w_sum / (2 * n - 1)) + corr.max() * 0
    lse_val = (corr.max() - (1.0 / beta) * np.log((2 * n - 1) / w_sum)) * dx if w_sum > 0 else corr.max() * dx
    return lse_val, grad, w

def grad_corr_k(h, k, n):
    offset = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -offset)
    i_hi = min(n - 1, n - 1 - offset)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    j1 = idx + offset
    grad[j1] += (1.0 - h[idx])
    grad[idx] += (-h[idx + offset])
    return grad

        improved = False
        for s in [1.0, 0.7, 0.5, 0.3, 0.15, 0.05]:
            h_new = project_feasible(h_curr + s * delta, target_sum)
            c5_new = exact_c5(h_new)
            if c5_new < exact_c5(h_curr) - 1e-14:
                h_curr = h_new
                improved = True
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                break
        if improved:
            no_improve = 0
        else:
            no_improve += 1
            alpha *= alpha_decay
            if no_improve == 25:
                n_active = min(n_total, n_active + 20)
                alpha = max(alpha, 0.15)
            if no_improve > max_no_improve:
                break
    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float)
    n = len(h0)
    dx = 2.0 / n
    target_sum = n / 2.0
    h0 = project_feasible(h0, target_sum)
    best_h = h0.copy()
    best_c5 = exact_c5(h0)

    # Phase 1: LSE-smoothed optimization (main new approach)
    t1_end = t0 + budget_s * 0.35
    h_lse, c5_lse = lse_optimize(best_h, dx, target_sum, t0, t1_end,
                                  beta_start=15.0, beta_end=2000.0,
                                  n_betas=12, step_init=0.06)
    if c5_lse < best_c5:
        best_c5 = c5_lse
        best_h = h_lse

    # Phase 2: More LSE with different params from the best so far
    t2_end = t0 + budget_s * 0.45
    if time.time() < t2_end:
        h_lse2, c5_lse2 = lse_optimize(best_h, dx, target_sum, time.time(),
                                        time.time() + (t2_end - time.time()),
                                        beta_start=50.0, beta_end=5000.0,
                                        n_betas=8, step_init=0.04)
        if c5_lse2 < best_c5:
            best_c5 = c5_lse2
            best_h = h_lse2

    # Phase 3: SA exploration from multiple starting points
    t3_end = t0 + budget_s * 0.7
    sa_starts = [best_h]
    # Add perturbed versions
    for sigma in [0.03, 0.05, 0.01]:
        h_pert = project_feasible(best_h + sigma * rng.randn(n), target_sum)
        sa_starts.append(h_pert)
    # Complement
    h_comp = project_feasible(1.0 - best_h, target_sum)
    sa_starts.append(h_comp)

    sa_params = [
        (40000, 0.003, 1e-6),
        (30000, 0.004, 1e-6),
        (50000, 0.002, 1e-7),
        (25000, 0.005, 1e-6),
    ]
    for idx, start_h in enumerate(sa_starts):
        if time.time() > t3_end:
            break
        if idx >= len(sa_params):
            break
        ni, ts, te = sa_params[idx]
        dt = t3_end - time.time()
        if dt < 5:
            break
        h_sa, c5_sa = sa_optimize(start_h, dx, target_sum, ni, ts, te,
                                   seed + 100 + idx * 13, time.time(), time.time() + dt)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa

    # Phase 4: LP polishing
    t4_end = t0 + budget_s * 0.88
    lp_configs = [
        (0.25, 0.4, 0.985, 80),
        (0.15, 0.3, 0.99, 100),
        (0.2, 0.5, 0.985, 60),
        (0.1, 0.35, 0.99, 120),
    ]
    for alpha_s, frac, decay, max_ni in lp_configs:
        if time.time() > t4_end:
            break
        dt = t4_end - time.time()
        if dt < 5:
            break
        h_lp, c5_lp = lp_optimize(best_h, dx, target_sum, time.time(),
                                   time.time() + dt,
                                   alpha_start=alpha_s, n_active_frac=frac,
                                   alpha_decay=decay, max_no_improve=max_ni)
        if c5_lp < best_c5:
            best_c5 = c5_lp
            best_h = h_lp

    # Phase 5: Final SA + LP polish
    t5_end = t0 + budget_s * 0.95
    if time.time() < t5_end:
        dt_sa = min(30.0, (t5_end - time.time()) * 0.4)
        if dt_sa > 3:
            h_sa2, c5_sa2 = sa_optimize(best_h, dx, target_sum, 30000, 0.001, 1e-7,
                                         seed + 999, time.time(), time.time() + dt_sa)
            if c5_sa2 < best_c5:
                best_c5 = c5_sa2
                best_h = h_sa2
        dt_lp = t5_end - time.time()
        if dt_lp > 5:
            h_lp2, c5_lp2 = lp_optimize(best_h, dx, target_sum, time.time(),
                                         time.time() + dt_lp,
                                         alpha_start=0.1, n_active_frac=0.5,
                                         alpha_decay=0.99, max_no_improve=150)
            if c5_lp2 < best_c5:
                best_c5 = c5_lp2
                best_h = h_lp2

    # Final: LSE polish from very high beta
    if time.time() < t0 + budget_s - 3:
        dt_final = t0 + budget_s - 3 - time.time()
        if dt_final > 8:
            h_fin, c5_fin = lse_optimize(best_h, dx, target_sum, time.time(),
                                          time.time() + dt_final * 0.7,
                                          beta_start=2000.0, beta_end=10000.0,
                                          n_betas=4, step_init=0.02)
            if c5_fin < best_c5:
                best_c5 = c5_fin
                best_h = h_fin

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    n_final = len(best_h)
    ts_final = n_final / 2.0
    best_h = project_feasible(best_h, ts_final)
    dx_final = 2.0 / n_final
    best_c5 = exact_c5(best_h)
    # Ensure sum is exact
    diff = ts_final - best_h.sum()
    if abs(diff) > 1e-10:
        best_h = project_feasible(best_h, ts_final)
        best_c5 = exact_c5(best_h)
    return best_h, best_c5, n_final
