- h[i] appears in h[j+k] when j+k=i, i.e., j=i-k. Valid if max(0,-k) ≤ i-k ≤ min(n-1,n-1-k).
  Contribution: (1-h[i-k])
- h[i] appears in (1-h[j]) when j=i. Valid if max(0,-k) ≤ i ≤ min(n-1,n-1-k).
  Contribution: -h[i+k] (since d/dh[i] of (1-h[i]) = -1, and h[j+k]=h[i+k])

So: ∂c[k]/∂h[i] = (1-h[i-k]) * [valid(i-k)] + (-h[i+k]) * [valid(i)]

where valid(m) means 0≤m≤n-1, and the conditions are:
- For the first term: max(0,-k) ≤ i-k ≤ min(n-1,n-1-k), which simplifies to 0≤i-k≤n-1 AND 0≤k≤n-1 (the second is about k being in range, which it is)
  Actually: j = i-k must satisfy max(0,-k) ≤ j ≤ min(n-1, n-1-k)
  This means: j ≥ 0 (always for valid i-k), j ≤ n-1 (i-k ≤ n-1, i.e., i ≤ n-1+k), j ≥ -k (i-k ≥ -k, i.e., i ≥ 0), j+k ≤ n-1 (i ≤ n-1)
  So the condition is simply: 0 ≤ i-k ≤ n-1 (i.e., i-k is a valid index)

- For the second term: j = i must satisfy max(0,-k) ≤ i ≤ min(n-1, n-1-k)
  This means: i ≥ 0 (always), i ≤ n-1 (always), i ≥ -k (i.e., i+k ≥ 0), i ≤ n-1-k (i.e., i+k ≤ n-1)
  So the condition is: 0 ≤ i+k ≤ n-1 (i+k is a valid index)

Great, so:
∂c[k]/∂h[i] = (1-h[i-k]) * [0≤i-k≤n-1] - h[i+k] * [0≤i+k≤n-1]

This is clean. Let me implement this vectorized:

```python
def grad_all(h, n):
    """Compute gradient of all c[k] with respect to h. Returns array of shape (2n-1, n)."""
    grads = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        # First term: (1-h[i-k]) for valid i-k
        # i-k valid means 0 <= i-k <= n-1, i.e., k <= i <= k+n-1
        # Also 0 <= i <= n-1
        i_lo = max(0, k)
        i_hi = min(n-1, n-1+k)
        # For i in [i_lo, i_hi]: contribution is (1-h[i-k])
        idx = np.arange(i_lo, i_hi+1)
        grads[k, idx] += (1.0 - h[idx - k])
        
        # Second term: -h[i+k] for valid i+k
        # i+k valid means 0 <= i+k <= n-1, i.e., -k <= i <= n-1-k
        # Also 0 <= i <= n-1
        j_lo = max(0, -k)
        j_hi = min(n-1, n-1-k)
        # For i in [j_lo, j_hi]: contribution is -h[i+k]
        idx2 = np.arange(j_lo, j_hi+1)
        grads[k, idx2] += -h[idx2 + k]
    
    return grads
```

```python
def lp_step(h, dx, target, n_active, alpha, t0, deadline):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    current_c5 = corr.max() * dx
    
    # Pick active set
    active = np.argsort(corr)[-n_active:]
    
    # Compute gradients for active constraints
    G = np.zeros((len(active), n))
    for row, k in enumerate(active):
        i_lo = max(0, k)
        i_hi = min(n-1, n-1+k)
        idx = np.arange(i_lo, i_hi+1)
        G[row, idx] += (1.0 - h[idx - k])
        j_lo = max(0, -k)
        j_hi = min(n-1, n-1-k)
        idx2 = np.arange(j_lo, j_hi+1)
        G[row, idx2] += -h[idx2 + k]
    
    # LP: minimize t s.t. G @ delta - t <= -corr[active], sum(delta) = 0, |delta| <= alpha
    # Variables: delta (n), t (1)
    # Objective: minimize t
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    A_ub = np.hstack([G, -np.ones((len(active), 1))])
    b_ub = -corr[active]
    
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    bounds = [(-h[i]*alpha, (1-h[i])*alpha) for i in range(n)] + [(-np.inf, np.inf)]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    
    if res.status != 0:
        return h, current_c5
    
    delta = res.x[:n]
    
    # Line search
    best_c5 = current_c5
    best_h = h
    for s in [1.0, 0.5, 0.3, 0.2, 0.1]:
        h_new = project_feasible(h + s * delta, target)
        c5_new = c5_eval(h_new, dx)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new
    return best_h, best_c5
```

```python
import numpy as np
from scipy.optimize import linprog
import time


def project_feasible(x, target_sum):
    x = np.asarray(x, dtype=float).copy()
    lo, hi = -3.0, 3.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = h < 1.0 - 1e-9
        else:
            mask = h > 1e-9
        cnt = mask.sum()
        if cnt > 0:
            h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)


def c5_eval(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)


def grad_for_hop(h, k, n):
    g = np.zeros(n)
    i_lo = max(0, k)
    i_hi = min(n - 1, n - 1 + k)
    if i_lo <= i_hi:
        idx = np.arange(i_lo, i_hi + 1)
        g[idx] += (1.0 - h[idx - k])
    j_lo = max(0, -k)
    j_hi = min(n - 1, n - 1 - k)
    if j_lo <= j_hi:
        idx2 = np.arange(j_lo, j_hi + 1)
        g[idx2] += -h[idx2 + k]
    return g


def coord_descend(h, dx, target, n_delta=25, max_passes=200, t0=0.0, deadline=float('inf')):
    n = len(h)
    cur_c5 = c5_eval(h, dx)
    best_c5 = cur_c5
    best_h = h.copy()
    for p in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            for j in range(i + 1, n):
                d_lo = max(-h[i], h[j] - 1.0)
                d_hi = min(1.0 - h[i], h[j])
                if d_hi - d_lo < 1e-12:
                    continue
                deltas = np.linspace(d_lo + 1e-10, d_hi - 1e-10, n_delta)
                best_local = cur_c5
                best_d = 0.0
                for d in deltas:
                    h[i] += d
                    h[j] -= d
                    c = c5_eval(h, dx)
                    if c < best_local:
                        best_local = c
                        best_d = d
                    h[i] -= d
                    h[j] += d
                if best_d != 0.0 and best_local < cur_c5:
                    h[i] += best_d
                    h[j] -= best_d
                    cur_c5 = best_local
                    improved = True
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
        if not improved:
            break
    return best_h, best_c5


def sa_run(h, dx, target, seed, t0, deadline, n_initial_restarts=5):
    n = len(h)
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    best_c5 = c5_eval(h, dx)
    
    t_start = 0.02
    t_end = 1e-7
    cooling = 0.9992
    n_iter_per_run = 200000
    
    for restart in range(n_initial_restarts + 1):
        if time.time() > deadline:
            break
        if restart == 0:
            h_cur = h.copy()
        else:
            sigma = 0.03 * (1 + 0.1 * restart)
            h_cur = project_feasible(h_cur + sigma * rng.randn(n), target)
        
        cur_c5 = c5_eval(h_cur, dx)
        temp = t_start
        no_improve = 0
        it = 0
        while it < n_iter_per_run:
            if time.time() > deadline:
                break
            it += 1
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d_hi = min(1.0 - h_cur[i], h_cur[j])
            d_lo = max(-h_cur[i], h_cur[j] - 1.0)
            if d_hi - d_lo < 1e-12:
                continue
            d = rng.uniform(d_lo, d_hi)
            h_cur[i] += d
            h_cur[j] -= d
            new_c5 = c5_eval(h_cur, dx)
            delta = new_c5 - cur_c5
            if delta < 0 or (temp > t_end and rng.random() < np.exp(-delta / temp)):
                cur_c5 = new_c5
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h_cur.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                h_cur[i] -= d
                h_cur[j] += d
                no_improve += 1
            temp = max(t_end, temp * cooling)
            if no_improve > 50000 and temp < t_start * 0.3:
                break
    
    return best_h, best_c5


def lp_polish(h, dx, target, t0, deadline, n_active_frac=0.4, alpha_start=0.2, max_steps=200):
    n = len(h)
    n_total = 2 * n - 1
    n_active = max(10, int(n_active_frac * n_total))
    best_c5 = c5_eval(h, dx)
    best_h = h.copy()
    alpha = alpha_start
    no_improve = 0
    for step in range(max_steps):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        if n_active >= n_total:
            active = np.arange(n_total)
        else:
            active = np.argsort(corr)[-n_active:]
        
        G = np.zeros((len(active), n))
        for row, k in enumerate(active):
            G[row] = grad_for_hop(h, k, n)
        
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        A_ub = np.empty((len(active), n + 1))
        A_ub[:, :n] = G
        A_ub[:, n] = -1.0
        b_ub = -corr[active]
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)
        bounds = [(-h[i] * alpha, (1.0 - h[i]) * alpha) for i in range(n)]
        bounds.append((-np.inf, np.inf))
        
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
        except Exception:
            break
        if res.status != 0:
            no_improve += 1
            if no_improve > 30:
                break
            alpha *= 0.95
            continue
        
        delta = res.x[:n]
        if np.abs(delta).max() < 1e-14:
            no_improve += 1
            if no_improve > 30:
                break
            alpha *= 0.95
            continue
        
        improved = False
        for s in [1.0, 0.7, 0.5, 0.3, 0.2, 0.1]:
            h_new = project_feasible(h + s * delta, target)
            c5_new = c5_eval(h_new, dx)
            if c5_new < best_c5 - 1e-14:
                h = h_new
                best_c5 = c5_new
                best_h = h_new.copy()
                improved = True
                no_improve = 0
                break
        if not improved:
            no_improve += 1
            alpha *= 0.95
            if no_improve == 20:
                alpha = max(alpha, 0.3)
                n_active = min(n_total, n_active + 10)
            elif no_improve > 50:
                break
    
    return best_h, best_c5

            cur = c5_eval(h, dx)
```
