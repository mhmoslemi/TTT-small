import numpy as np
from scipy.optimize import linprog
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full"))) * dx

def profile_h(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity, kind='stable')
        remaining = diff
        for i in range(len(order)):
            idx = order[i]
            add = min(remaining, capacity[idx])
            h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h, kind='stable')
        remaining = -diff
        for i in range(len(order)):
            idx = order[i]
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    return h

def build_gradient_all(h, n, dx):
    """Build gradient matrix for all lags. Returns (2n-1, n) array."""
    M = 2 * n - 1
    G = np.zeros((M, n))
    for m in range(M):
        d = m - (n - 1)
        if d == 0:
            G[m, :] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            for i in range(n - d):
                G[m, i + d] += dx * (1.0 - h[i])
                G[m, i] -= dx * h[i + d]
        else:
            dp = -d
            for j in range(n - dp):
                G[m, j + dp] += dx * (1.0 - h[j])
                G[m, j] -= dx * h[j + dp]
    return G

def build_gradient_subset(h, lags, n, dx):
    """Build gradient matrix for subset of lags."""
    G = np.zeros((len(lags), n))
    for idx, m in enumerate(lags):
        d = m - (n - 1)
        if d == 0:
            G[idx, :] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            for i in range(n - d):
                G[idx, i + d] += dx * (1.0 - h[i])
                G[idx, i] -= dx * h[i + d]
        else:
            dp = -d
            for j in range(n - dp):
                G[idx, j + dp] += dx * (1.0 - h[j])
                G[idx, j] -= dx * h[j + dp]
    return G

def lp_direction(h, active_lags, n, dx):
    """Solve LP to find direction that decreases all active lags.
    
    Maximize t
    Subject to: G_a @ v + t <= 0 for each active lag a
                sum(v) = 0
                -1 <= v_i <= 1
    
    Returns direction v or None.
    """
    G = build_gradient_subset(h, active_lags, n, dx)
    k = len(active_lags)
    
    # Variables: v_0, ..., v_{n-1}, t
    # Objective: maximize t → minimize -t
    c = np.zeros(n + 1)
    c[-1] = -1.0
    
    # Inequality constraints: G_a @ v + t <= 0 → G_a @ v + [0,...,0,1] @ t <= 0
    # For linprog: A_ub @ x <= b_ub
    A_ub = np.zeros((k, n + 1))
    A_ub[:, :n] = G
    A_ub[:, -1] = 1.0
    b_ub = np.zeros(k)
    
    # Equality: sum(v) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    # Bounds: -1 <= v_i <= 1, t <= some bound (say 1)
    bounds = [(-1.0, 1.0)] * n + [(0.0, 1.0)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x is not None:
            v = res.x[:n]
            t_val = res.x[-1]
            if t_val > 1e-6:
                # Direction is good
                v = v - v.mean()  # Ensure sum zero (numerical)
                return v, t_val
    except Exception:
        pass
    return None, 0.0

    best_c = exact_score(h)
    prof = profile_h(h)
    
    # Precompute which pairs to try

    sweep = 0
    while sweep < max_sweeps:
        if t_start is not None and time.perf_counter() - t_start > budget_s - 10:
            break
        
        # Generate pairs: nearby and far pairs
        pairs = []
        rng_local = np.random.RandomState(sweep * 1000 + int(np.floor(best_c * 100000)))
        for _ in range(n_pairs_per_sweep):
            i = rng_local.randint(n)
            j = rng_local.randint(n)
            if i != j:
                pairs.append((i, j))
        
        improved_sweep = False
        for (i, j) in pairs:
            if t_start is not None and time.perf_counter() - t_start > budget_s - 10:
                break
            
            s = h[i] + h[j]
            if s < 1e-10 or s > 2.0 - 1e-10:
                continue
            
            lo = max(0.0, s - 1.0)
            hi = min(1.0, s)
            
            # Sample candidate values for h[i] (h[j] = s - h[i])
            # Use a mix of uniform and local samples
            if hi - lo < 1e-10:
                continue
            
            # Try several values
            candidates = np.linspace(lo, hi, 8)
            # Add current value
            candidates = np.concatenate([candidates, [h[i]]])
            
            best_local = best_c
            best_t = h[i]
            
            for t in candidates:
                h[i] = t
                h[j] = s - t
                c_new = exact_score(h)
                if c_new < best_local - 1e-14:
                    best_local = c_new
                    best_t = t
            
            if best_t != h[i]:
                h[i] = best_t
                h[j] = s - best_t
                best_c = best_local
                improved_sweep = True
        
        if not improved_sweep:
            break
        sweep += 1
    
    return h, best_c

def subgradient_step(h, active_lags, step_size, n, dx):
    """Take a subgradient step using weighted average of active gradients."""
    G = build_gradient_subset(h, active_lags, n, dx)
    # Weight by how active each constraint is
    prof = profile_h(h)
    c_max = prof.max()
    # Exponential weighting
    weights = np.exp(50 * (prof[active_lags] - c_max))
    weights = weights / weights.sum()
    
    v = G.T @ weights
    v = v - v.mean()  # Preserve sum
    
    # Normalize
    norm = np.linalg.norm(np.abs(v), 1)
    if norm < 1e-10:
        return h, exact_score(h)
    v = v / norm
    
    h_new = repair_feasible(h - step_size * v)
    return h_new, exact_score(h_new)

def fibonacci_pattern(n):
    """Create a binary pattern using golden ratio placement."""
    h = np.zeros(n)
    golden = (1 + np.sqrt(5)) / 2.0
    for i in range(n // 2):
        pos = int(((i + 0.5) / n * golden) % 1.0 * n)
        h[pos] = 1.0
    return repair_feasible(h)

def quadratic_residue_pattern(n):
    """Create a binary pattern using quadratic residues mod p."""
    h = np.zeros(n)
    # Use the first n positions, mark quadratic residues
    p = 1
    while p < n:
        p *= 2
    # Actually, let's use a simpler approach: mark positions i where i*i mod p < p/2
    threshold = p // 2
    for i in range(n):
        if (i * i) % p < threshold:
            h[i] = 1.0
    return repair_feasible(h)

def random_binary(n, seed):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    positions = rng.choice(n, size=n//2, replace=False)
    h[positions] = 1.0
    return h

def binary_anneal(h, n_iter=500, t_start=None, budget_s=1100, seed=42):
    """Simulated annealing in binary subspace."""
    n = len(h)
    h = np.round(h).clip(0, 1)
    # Ensure exactly n/2 ones
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h <= 0.5)[0]
    if len(ones) != n // 2:
        if len(ones) > n // 2:
            extra = len(ones) - n // 2
            h[np.random.RandomState(seed).choice(ones, extra, replace=False)] = 0
        else:
            need = n // 2 - len(ones)
            h[np.random.RandomState(seed).choice(zeros, need, replace=False)] = 1
    
    rng = np.random.RandomState(seed)
    current_c = exact_score(h)
    best_h = h.copy()
    best_c = current_c
    
    T_start = 0.5
    T_end = 0.01
    
    for it in range(n_iter):
        if t_start is not None and time.perf_counter() - t_start > budget_s - 15:
            break
        
        T = T_start * (T_end / T_start) ** (it / max(n_iter - 1, 1))
        
        # Random swap: pick a 1 and a 0, swap them
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h <= 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        h[i] = 0
        h[j] = 1
        new_c = exact_score(h)
        
        delta = new_c - current_c
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            current_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            h[i] = 1
            h[j] = 0
    
    return best_h, best_c

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    return repair_feasible(np.clip(h_new, 0.0, 1.0))

def active_set_slsqp(h0, maxiter=200, active_k=10, tol=1e-11):
    """SLSQP with adaptive active constraints."""
    n = len(h0)
    dx = 2.0 / n
    P = profile_h(h0)
    c0 = P.max()
    
    # Select active constraints
    margin = max(0.01, 0.1 * (c0 - P.min()))
    active = np.where(P >= c0 - margin)[0]
    if len(active) > active_k:
        active = np.argsort(P[active])[-active_k:]
    
    def obj(x):
        return x[-1]
    
    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    
    def eq_fun(x):
        return x[:-1].sum() - n / 2.0
    
    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g
    
    def ineq_fun(x):
        h = x[:-1]
        t = x[-1]
        prof = profile_h(h)
        return t - prof[active]
    
    def ineq_jac(x):
        h = x[:-1]
        G = build_gradient_subset(h, active, n, dx)
        J = np.zeros((len(active), n + 1))
        J[:, :-1] = -G
        J[:, -1] = 1.0
        return J
    
    x0 = np.concatenate([h0, [c0 + 1e-4]])
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 1.0)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': tol})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = repair_feasible(res.x[:-1])
            c_new = exact_score(h_new)
            if c_new <= c0 + 1e-10:
                return h_new
    except Exception:
        pass
    return None
