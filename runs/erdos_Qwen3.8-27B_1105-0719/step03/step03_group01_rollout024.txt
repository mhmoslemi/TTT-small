Let me think about the LSE approach more carefully:

For h with n points, dx = 2/n, target_sum = n/2.
The correlations are c_k = sum_i h[i] * (1-h[i+k]) for valid i.
C₅ = max_k c_k * dx.

The LSE smooth approximation: L_T(h) = T * log(sum_k exp(c_k / T))
As T → 0, this approaches max_k c_k.

The gradient of L_T with respect to h is:
∂L_T/∂h = (1/T) * sum_k (exp(c_k/T)) * ∂c_k/∂h
= sum_k u_k * ∂c_k/∂h where u_k = exp(c_k/T) / sum_j exp(c_j/T)

For a shift d (where c_d = sum_i h[i] * (1-h[i+d])):
∂c_d/∂h[j] = sum over valid i where j is involved:
- If j = i: (1 - h[i+d])
- If j = i+d: -h[i]

So the gradient is a weighted sum of these shift-specific gradients.

The key insight: we can compute this gradient efficiently, then:
1. Center it (subtract mean) to maintain the sum constraint
2. Find the maximum step size that keeps h in [0,1]
3. Do a line search (or just try a few step sizes)
4. Evaluate the exact objective

        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=[0.0],
                      bounds=bounds, method='highs')
        if res.status != 0:
            return h, current_c5, False
        delta = res.x[:n]
    except Exception:
        return h, current_c5, False

    if np.abs(delta).max() < 1e-15:
        return h, current_c5, False

    best_h = h
    best_c5 = current_c5
    for s in [1.0, 0.7, 0.5, 0.3, 0.15, 0.05]:
        h_new = project_feasible(h + s * delta, target_sum)
        c5_new = c5_of_h(h_new, dx)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

    return best_h, best_c5, best_c5 < current_c5 - 1e-14


def active_set_optimize(h, dx, target_sum, t0, max_time, alpha_start=0.25,
                        n_active_frac=0.35, alpha_decay=0.98, max_no_improve=60):
    n = len(h)
    n_total = 2 * n - 1
    n_active = max(10, int(n_active_frac * n_total))
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    alpha = alpha_start
    no_improve = 0

    while time.time() - t0 < max_time and alpha > 0.002 and no_improve < max_no_improve:
        h_new, c5_new, improved = active_set_lp_step(h, dx, target_sum, n_active, alpha)
        if improved:
            h = h_new
            best_h = h_new.copy()
            best_c5 = c5_new
            no_improve = 0
        else:
            no_improve += 1
            alpha *= alpha_decay
            if no_improve == 25:
                n_active = min(n_total, n_active + 10)
                alpha = max(alpha, 0.15)
            elif no_improve == 50:
                n_active = max(20, n_active - 15)
                alpha = max(alpha, 0.2)

    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n = len(h0)
    dx = 2.0 / n
    target_sum = n / 2.0
    h0 = project_feasible(h0, target_sum)

    best_h = h0.copy()
    best_c5 = c5_of_h(h0, dx)

    def get_remaining():
        return budget_s - (time.time() - t0)

    # Phase 1: Full LP (all constraints) - most principled
    dt = get_remaining() * 0.25
    t_end = time.time() + dt
    h_opt, c5_opt = main_lp_optimize(best_h, dx, target_sum, t0, t_end,
                                     alpha_start=0.35, alpha_decay=0.975, max_no_improve=100)
    if c5_opt < best_c5:
        best_c5 = c5_opt
        best_h = h_opt

    # Phase 2: LSE optimization
    dt = get_remaining() * 0.15
    t_end = time.time() + dt
    h_opt, c5_opt = lse_optimize(best_h, dx, target_sum, t0, t_end,
                                  T_start=4.0, T_min=0.02, T_decay=0.90,
                                  tau_start=0.4, max_no_improve=60)
    if c5_opt < best_c5:
        best_c5 = c5_opt
        best_h = h_opt

    # Phase 3: Active-set LP with varying active set size
    dt = get_remaining() * 0.15
    t_end = time.time() + dt
    h_opt, c5_opt = active_set_optimize(best_h, dx, target_sum, t0, t_end,
                                        alpha_start=0.3, n_active_frac=0.3,
                                        alpha_decay=0.975, max_no_improve=80)
    if c5_opt < best_c5:
        best_c5 = c5_opt
        best_h = h_opt

    # Phase 4: Multiple perturbations with local optimization
    n_perturb = 15
    sigmas = [0.02, 0.04, 0.06, 0.03, 0.05, 0.01, 0.08, 0.025,
              0.035, 0.015, 0.055, 0.045, 0.07, 0.012, 0.038]
    for p in range(n_perturb):
        if get_remaining() < 25:
            break
        sigma = sigmas[p % len(sigmas)]
        h_pert = perturb_feasible(best_h, target_sum, sigma, rng)

        # Quick LSE warmup
        dt_warm = min(get_remaining() * 0.04, 25)
        t_end = time.time() + dt_warm
        h_warm, c5_warm = lse_optimize(h_pert, dx, target_sum, t0, t_end,
                                        T_start=2.0, T_min=0.05, T_decay=0.88,
                                        tau_start=0.3, max_no_improve=30)

        # LP polish
        dt_pol = min(get_remaining() * 0.04, 35)
        t_end = time.time() + dt_pol
        h_pol, c5_pol = main_lp_optimize(h_warm, dx, target_sum, t0, t_end,
                                          alpha_start=0.2, alpha_decay=0.96,
                                          max_no_improve=50)

        c5_cand = c5_pol if c5_pol < c5_warm else c5_warm
        h_cand = h_pol if c5_pol < c5_warm else h_warm
        if c5_cand < best_c5:
            best_c5 = c5_cand
            best_h = h_cand

    # Phase 5: Try different resolutions
    n_candidates = [n - 6, n - 4, n - 2, n + 2, n + 4, n + 8, n + 12]
    for n_try in n_candidates:
        if get_remaining() < 30:
            break
        if n_try < 40 or n_try == n:
            continue
        try:
            x_orig = np.linspace(0.0, 2.0, n)
            x_new = np.linspace(0.0, 2.0, n_try)
            h_interp = np.interp(x_new, x_orig, best_h)
            ts_try = n_try / 2.0
            h_interp = project_feasible(h_interp, ts_try)
            dx_try = 2.0 / n_try
            c5_init = c5_of_h(h_interp, dx_try)
            if c5_init >= best_c5:
                continue
            dt_try = min(get_remaining() * 0.08, 50)
            t_end = time.time() + dt_try
            h_opt, c5_opt = main_lp_optimize(h_interp, dx_try, ts_try, t0, t_end,
                                              alpha_start=0.25, alpha_decay=0.96,
                                              max_no_improve=50)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
        except Exception:
            continue

    # Phase 6: Final intensive alternating polish
    if get_remaining() > 15:
        n_b = len(best_h)
        dx_b = 2.0 / n_b
        ts_b = n_b / 2.0

        for round_i in range(6):
            if get_remaining() < 12:
                break
            # LSE pass
            dt1 = min(get_remaining() * 0.25, 25)
            t_end = time.time() + dt1
            h_opt, c5_opt = lse_optimize(best_h, dx_b, ts_b, t0, t_end,
                                          T_start=1.5, T_min=0.02, T_decay=0.85,
                                          tau_start=0.2, max_no_improve=40)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt

            # Full LP pass
            dt2 = min(get_remaining() * 0.25, 25)
            t_end = time.time() + dt2
            h_opt, c5_opt = main_lp_optimize(best_h, dx_b, ts_b, t0, t_end,
                                              alpha_start=0.15, alpha_decay=0.95,
                                              max_no_improve=50)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt

            # Tiny perturbation + LSE
            if get_remaining() > 15:
                h_tiny = perturb_feasible(best_h, ts_b, 0.008, rng)
                dt3 = min(get_remaining() * 0.15, 15)
                t_end = time.time() + dt3
                h_opt, c5_opt = lse_optimize(h_tiny, dx_b, ts_b, t0, t_end,
                                              T_start=1.0, T_min=0.03, T_decay=0.82,
                                              tau_start=0.15, max_no_improve=30)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    n_final = len(best_h)
    ts_final = n_final / 2.0
    best_h = project_feasible(best_h, ts_final)
    dx_final = 2.0 / n_final
    best_c5 = c5_of_h(best_h, dx_final)

    return best_h, best_c5, n_final
