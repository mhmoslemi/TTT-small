Let me think about the LMO projection:

- This is a linear program on the capped simplex
- The optimal solution is a vertex: set v_i = 1 for the n/2 positions where G is smallest, v_i = 0 elsewhere

For the line search between h and v:

    n = len(h)
    corrs = compute_corrs(h)
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    w /= w_sum
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] > 1e-12:
            grad += w[k] * grad_corr_k(h, k, n)
    return grad * dx

def lmo_vertex(h, dx, n_active=64):
    """Compute LMO vertex: binary vector minimizing aggregated active gradient"""
    n = len(h)
    corrs = compute_corrs(h)
    max_c = corrs.max()
    
    # Select active lags (top correlations)
    n_act = min(n_active, len(corrs))
    active_k = np.argsort(corrs)[-n_act:]
    
    G = np.zeros(n)
    for k in active_k:
        G += grad_corr_k(h, k, n)
    G *= dx
    
    # Vertex: set 1 where G is smallest, 0 elsewhere
    n_ones = n // 2
    v = np.zeros(n)
    v[np.argsort(G)[:n_ones]] = 1.0
    v = project_feasible(v, n / 2.0)
    return v

def frank_wolfe_step(h, dx, target_sum, n_active=64, n_line_search=50):
    """One Frank-Wolfe iteration with exact line search"""
    n = len(h)
    v = lmo_vertex(h, dx, n_active)
    
    # Line search: minimize c5_of_h((1-t)*h + t*v)
    best_t = 0.0
    best_val = c5_of_h(h, dx)
    
    # Use golden section or fine grid
    for t in np.linspace(0, 1, n_line_search):
        h_try = (1 - t) * h + t * v
        h_try = project_feasible(h_try, target_sum)
        val = c5_of_h(h_try, dx)
        if val < best_val:
            best_val = val
            best_t = t
    
    if best_t > 0:
        h_out = project_feasible((1 - best_t) * h + best_t * v, target_sum)
    else:
        h_out = h.copy()
    
    return h_out, best_val

def pair_transfer_polish(h0, dx, target_sum, n_top=25, n_iter=3):

                    if fc < fd:

                        fc = f_ps(c_pt)
                    else:
                        a = c_pt
                        c_pt = d_pt
                        fc = fd
                        d_pt = a + gr * (b - a)
                        fd = f_ps(d_pt)

                eps_opt = (a + b) / 2.0
                if eps_opt < 1e-12:

                new_c5_rough = (corrs + eps_opt * d).max() * dx

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.special import logsumexp
import time


def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)


def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def c5_of_h(h, dx):
    return compute_corrs(h).max() * dx


def grad_corr_k(h, k, n):
    offset = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -offset)
    i_hi = min(n - 1, n - 1 - offset)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx] += (1.0 - h[idx + offset])
    grad[idx + offset] += (-h[idx])
    return grad


def smooth_max_grad(h, dx, beta):
    n = len(h)
    corrs = compute_corrs(h)
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    w /= w_sum
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] > 1e-12:
            grad += w[k] * grad_corr_k(h, k, n)
    return grad * dx


def slsqp_direct(h0, dx, target_sum, beta, maxiter):
    n = len(h0)

    def obj(x):
        corrs = compute_corrs(x)
        return logsumexp(beta * corrs) / beta * dx

    def grad_obj(x):
        return smooth_max_grad(x, dx, beta)

    def c_fun(x):
        return x.sum() - target_sum

    def c_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': c_fun, 'jac': c_jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0, jac=grad_obj, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        h_out = project_feasible(res.x, target_sum)
    except Exception:
        h_out = project_feasible(h0, target_sum)
    return h_out, c5_of_h(h_out, dx)


def slsqp_z(h0, dx, target_sum, beta, maxiter):
    n = len(h0)

    def sigmoid_2z(z):
        return 1.0 / (1.0 + np.exp(-np.clip(2.0 * z, -500, 500)))

    def obj(z):
        h = sigmoid_2z(z)
        corrs = compute_corrs(h)
        return logsumexp(beta * corrs) / beta * dx

    def grad_obj(z):
        h = sigmoid_2z(z)
        grad_h = smooth_max_grad(h, dx, beta)
        return 2.0 * h * (1.0 - h) * grad_h

    def c_fun(z):
        return sigmoid_2z(z).sum() - target_sum

    def c_jac(z):
        h = sigmoid_2z(z)
        return 2.0 * h * (1.0 - h)

    cons = [{'type': 'eq', 'fun': c_fun, 'jac': c_jac}]
    h0_clip = np.clip(h0, 1e-8, 1.0 - 1e-8)
    z0 = -0.5 * np.log((1.0 - h0_clip) / h0_clip)
    try:
        res = minimize(obj, z0, jac=grad_obj, method='SLSQP',
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        h_out = project_feasible(sigmoid_2z(res.x), target_sum)
    except Exception:
        h_out = project_feasible(h0_clip, target_sum)
    return h_out, c5_of_h(h_out, dx)


def epigraph_inq_fun(z, args):
    n, dx, k = args
    h = z[:n]
    return z[n] - dx * compute_corrs(h)[k]


def epigraph_inq_jac(z, args):
    n, dx, k = args
    h = z[:n]
    g = np.zeros(n + 1)
    g[:-1] = -dx * grad_corr_k(h, k, n)
    g[n] = 1.0
    return g


def epigraph_eq_fun(z, args):
    n, target_sum = args
    return z[:n].sum() - target_sum


def epigraph_eq_jac(z, args):
    n, target_sum = args
    g = np.zeros(n + 1)
    g[:n] = 1.0
    return g


def epigraph_obj(z, args):
    return z[-1]


def epigraph_obj_grad(z, args):
    n = args[0]
    g = np.zeros(n + 1)
    g[-1] = 1.0
    return g


def epigraph_solve(h0, dx, target_sum, active_k, maxiter=300):
    n = len(h0)
    n_act = len(active_k)
    nobj = (n,)
    neq = (n, target_sum)
    constraints = []
    for k in active_k:
        constraints.append({'type': 'ineq', 'fun': epigraph_inq_fun,
                            'jac': epigraph_inq_jac, 'args': (n, dx, k)})
    constraints.append({'type': 'eq', 'fun': epigraph_eq_fun,
                        'jac': epigraph_eq_jac, 'args': neq})
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    corrs = compute_corrs(h0)
    z0 = np.zeros(n + 1)
    z0[:n] = h0
    z0[n] = corrs[active_k].max() * dx + 0.05
    try:
        res = minimize(epigraph_obj, z0, jac=epigraph_obj_grad, args=nobj,
                       method='SLSQP', bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        h_out = project_feasible(res.x[:n], target_sum)
    except Exception:
        h_out = project_feasible(h0, target_sum)
    return h_out, c5_of_h(h_out, dx)


def epigraph_iterative(h0, dx, target_sum, n_act=12, max_rounds=15, maxiter_per=250):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    for rnd in range(max_rounds):
        corrs = compute_corrs(h)
        active_k = np.argsort(corrs)[-n_act:]
        h_new, c5_new = epigraph_solve(h, dx, target_sum, active_k, maxiter_per)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
        else:
            h_mix = project_feasible(0.7 * h + 0.3 * h_new, target_sum)
            c5_mix = c5_of_h(h_mix, dx)
            if c5_mix < best_c5:
                best_c5 = c5_mix
                best_h = h_mix.copy()
                h = h_mix
            else:
                h = h_new
    return best_h, best_c5


def lmo_vertex(h, dx, n_active=64):
    n = len(h)
    corrs = compute_corrs(h)
    n_act = min(n_active, len(corrs))
    active_k = np.argsort(corrs)[-n_act:]
    G = np.zeros(n)
    for k in active_k:
        G += grad_corr_k(h, k, n)
    G *= dx
    n_ones = n // 2
    v = np.zeros(n)
    v[np.argsort(G)[:n_ones]] = 1.0
    v = project_feasible(v, n / 2.0)
    return v


def frank_wolfe_full(h0, dx, target_sum, max_iter=150, n_active=64):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    for it in range(max_iter):
        v = lmo_vertex(h, dx, n_active)
        best_t = 0.0
        best_val = best_c5
        for t in np.linspace(0, 1, 25):
            h_try = project_feasible((1 - t) * h + t * v, target_sum)
            val = c5_of_h(h_try, dx)
            if val < best_val:
                best_val = val
                best_t = t
        if best_t == 0:
            break
        t_lo = max(0.0, best_t - 0.04)
        t_hi = min(1.0, best_t + 0.04)
        for t in np.linspace(t_lo, t_hi, 30):
            h_try = project_feasible((1 - t) * h + t * v, target_sum)
            val = c5_of_h(h_try, dx)
            if val < best_val:
                best_val = val
                best_t = t
        h = project_feasible((1 - best_t) * h + best_t * v, target_sum)
        if best_val < best_c5:
            best_c5 = best_val
            best_h = h.copy()
        else:
            break
    return best_h, best_c5


def lp_polish(h0, dx, target_sum, max_iters=200, init_step=0.15):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    step = init_step
    stall = 0
    for it in range(max_iters):
        corrs = compute_corrs(h)
        c5_val = corrs.max() * dx
        max_c = corrs.max()
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 50:
            step *= 0.7
            stall = 0
            if step < 1e-7:
                break
        tol_val = max_c * 0.95
        active_mask = corrs >= tol_val
        active_k = np.where(active_mask)[0]
        if len(active_k) > 60:
            active_k = np.argsort(corrs)[-60:]
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        A_rows = []
        b_rows = []
        for k in active_k:
            row = np.zeros(n + 1)
            row[:n] = dx * grad_corr_k(h, k, n)
            row[n] = -1.0
            A_rows.append(row)
            b_rows.append(-corrs[k] * dx - 1e-14)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)
        bounds = []
        for i in range(n):
            bounds.append((max(-h[i], -step), min(1.0 - h[i], step)))
        bounds.append((0.0, None))
        try:
            res = linprog(c_obj, A_ub=np.array(A_rows), b_ub=np.array(b_rows),
                          A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
        except Exception:
            break
        if res.status != 0:
            break
        delta = res.x[:n]
        if np.max(np.abs(delta)) < 1e-14:
            break
        h_new = project_feasible(h + delta, target_sum)
        new_c5 = c5_of_h(h_new, dx)
        if new_c5 < c5_val - 1e-14:
            h = h_new
            step = min(step * 1.15, init_step)
        else:
            improved = False
            for frac in [0.5, 0.25, 0.1]:
                h_new = project_feasible(h + frac * delta, target_sum)
                new_c5 = c5_of_h(h_new, dx)
                if new_c5 < c5_val - 1e-14:
                    h = h_new
                    improved = True
                    break
            if not improved:
                step *= 0.5
                if step < 1e-8:
                    break
    return best_h, best_c5


def pair_transfer_polish(h0, dx, target_sum, n_top=25, n_iter=4):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    for iteration in range(n_iter):
        corrs = compute_corrs(h)
        grad = smooth_max_grad(h, dx, 800.0)
        dec_idx = np.argsort(grad)[-n_top:]
        inc_idx = np.argsort(grad)[:n_top]
        improved = False
        for i in dec_idx:
            for j in inc_idx:
                if i == j:
                    continue
                eps_hi = min(h[i], 1.0 - h[j])
                if eps_hi < 1e-12:
                    continue
                d = np.zeros(2 * n - 1)
                for k in range(2 * n - 1):
                    gk = grad_corr_k(h, k, n)
                    d[k] = gk[j] - gk[i]
                gr = (np.sqrt(5.0) - 1.0) / 2.0
                a, b = 0.0, eps_hi
                c_pt = b - gr * (b - a)
                d_pt = a + gr * (b - a)
                fc = (corrs + c_pt * d).max()
                fd = (corrs + d_pt * d).max()
                for _ in range(45):
                    if b - a < 1e-14:
                        break
                    if fc < fd:
                        b = d_pt
                        d_pt = c_pt
                        fd = fc
                        c_pt = b - gr * (b - a)
                        fc = (corrs + c_pt * d).max()
                    else:
                        a = c_pt
                        c_pt = d_pt
                        fc = fd
                        d_pt = a + gr * (b - a)
                        fd = (corrs + d_pt * d).max()
                eps_opt = (a + b) / 2.0
                if eps_opt < 1e-12:
                    continue
                rough = (corrs + eps_opt * d).max() * dx
                if rough < best_c5 - 1e-14:
                    new_h = h.copy()
                    new_h[i] -= eps_opt
                    new_h[j] += eps_opt
                    new_h = np.clip(new_h, 0.0, 1.0)
                    new_h = project_feasible(new_h, target_sum)
                    new_c5 = c5_of_h(new_h, dx)
                    if new_c5 < best_c5 - 1e-14:
                        h = new_h
                        best_c5 = new_c5
                        best_h = h.copy()
                        corrs = compute_corrs(h)
                        improved = True
        if not improved:
            break
    return best_h, best_c5


def binary_sa(h0, dx, target_sum, n_steps=8000, T0=0.03, cooldown=0.9997):
    n = len(h0)
    n_ones = n // 2
    h_bin = (h0 > 0.5).astype(float)
    ones_count = int(h_bin.sum())
    if ones_count != n_ones:
        if ones_count > n_ones:
            one_positions = np.where(h_bin == 1.0)[0]
            to_zero = rng_local.choice(one_positions, ones_count - n_ones, replace=False)
            h_bin[to_zero] = 0.0
        else:
            zero_positions = np.where(h_bin == 0.0)[0]
            to_one = rng_local.choice(zero_positions, n_ones - ones_count, replace=False)
            h_bin[to_one] = 1.0
    h = h_bin.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    T = T0
    for _ in range(n_steps):
        one_pos = np.where(h == 1.0)[0]
        zero_pos = np.where(h == 0.0)[0]
        if len(one_pos) == 0 or len(zero_pos) == 0:
            break
        i = one_pos[rng_local.randint(len(one_pos))]
        j = zero_pos[rng_local.randint(len(zero_pos))]
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = c5_of_h(h, dx)
        delta = new_c5 - best_c5
        if delta < 0 or rng_local.random() < np.exp(-delta / max(T, 1e-10)):
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        T *= cooldown
    return best_h, best_c5


rng_local = np.random.RandomState(12345)


def multi_n_optimize(h_best, n_try, dx_try, ts_try, time_left_fn, budget_min=30):
    x_orig = np.linspace(0.0, 2.0, len(h_best), endpoint=False) + 1.0 / len(h_best)
    x_new = np.linspace(0.0, 2.0, n_try, endpoint=False) + 1.0 / n_try
    h_interp = np.interp(x_new, x_orig, h_best)
    h_interp = project_feasible(h_interp, ts_try)
    best_h = h_interp.copy()
    best_c5 = c5_of_h(h_interp, dx_try)
    if time_left_fn() < 15:
        return best_h, best_c5
    h_opt, c5_val = slsqp_direct(h_interp, dx_try, ts_try, 800, 400)
    if c5_val < best_c5:
        best_c5 = c5_val
        best_h = h_opt
    if time_left_fn() < 10:
        return best_h, best_c5
    h_lp, c5_lp = lp_polish(best_h, dx_try, ts_try, max_iters=80, init_step=0.1)
    if c5_lp < best_c5:
        best_c5 = c5_lp
        best_h = h_lp
    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    global rng_local
    t0 = time.time()
    rng = np.random.RandomState(seed)
    rng_local = np.random.RandomState(seed + 777)

    def time_left():
        return budget_s - (time.time() - t0)

    best_h = None
    best_c5 = float('inf')

    h0 = np.array(initial_h_values, dtype=float)
    n = len(h0)
    dx = 2.0 / n
    target_sum = n / 2.0
    h0 = project_feasible(h0, target_sum)
    best_h = h0.copy()
    best_c5 = c5_of_h(h0, dx)

    # Phase 1: SLSQP smooth max with increasing beta
    for beta in [150, 400, 1200, 3500]:
        if time_left() < 25:
            break
        h_opt, c5_val = slsqp_direct(best_h, dx, target_sum, beta, 1200)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_opt

    # Phase 2: Z-space SLSQP
    if time_left() > 50:
        for beta in [800, 2500]:
            if time_left() < 20:
                break
            h_opt, c5_val = slsqp_z(best_h, dx, target_sum, beta, 800)
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h_opt

    # Phase 3: LP polish
    if time_left() > 30:
        h_lp, c5_lp = lp_polish(best_h, dx, target_sum, max_iters=250, init_step=0.12)
        if c5_lp < best_c5:
            best_c5 = c5_lp
            best_h = h_lp

    # Phase 4: Frank-Wolfe with LMO vertices
    if time_left() > 50:
        h_fw, c5_fw = frank_wolfe_full(best_h, dx, target_sum, max_iter=120, n_active=48)
        if c5_fw < best_c5:
            best_c5 = c5_fw
            best_h = h_fw
        if time_left() > 30:
            h_fw2, c5_fw2 = frank_wolfe_full(best_h, dx, target_sum, max_iter=80, n_active=96)
            if c5_fw2 < best_c5:
                best_c5 = c5_fw2
                best_h = h_fw2

    # Phase 5: Epigraph optimization
    if time_left() > 50:
        h_ep, c5_ep = epigraph_iterative(best_h, dx, target_sum, n_act=10, max_rounds=10, maxiter_per=250)
        if c5_ep < best_c5:
            best_c5 = c5_ep
            best_h = h_ep
        if time_left() > 30:
            h_ep2, c5_ep2 = epigraph_iterative(best_h, dx, target_sum, n_act=15, max_rounds=8, maxiter_per=200)
            if c5_ep2 < best_c5:
                best_c5 = c5_ep2
                best_h = h_ep2

    # Phase 6: Pair transfer
    if time_left() > 20:
        h_pt, c5_pt = pair_transfer_polish(best_h, dx, target_sum, n_top=25, n_iter=4)
        if c5_pt < best_c5:
            best_c5 = c5_pt
            best_h = h_pt

    # Phase 7: Binary SA for exploration
    if time_left() > 60:
        h_sa, c5_sa = binary_sa(best_h, dx, target_sum, n_steps=10000, T0=0.04)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa
        # Polish the SA result
        if time_left() > 30:
            h_sa_pol, c5_sa_pol = lp_polish(h_sa, dx, target_sum, max_iters=150, init_step=0.1)
            if c5_sa_pol < best_c5:
                best_c5 = c5_sa_pol
                best_h = h_sa_pol

    # Phase 8: Try different n values
    n_candidates = [n - 10, n - 6, n - 4, n - 2, n + 2, n + 4, n + 6, n + 10,
                    n + 14, n - 14, 120, 160]
    n_candidates = sorted(set([nn for nn in n_candidates if nn >= 60 and nn % 2 == 0 and nn != n]))
    n_candidates = n_candidates[:10]
    for n_try in n_candidates:
        if time_left() < 35:
            break
        dx_try = 2.0 / n_try
        ts_try = n_try / 2.0
        h_try, c5_try = multi_n_optimize(best_h, n_try, dx_try, ts_try, time_left)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try

    # Phase 9: Perturbation multi-start
    for trial in range(12):
        if time_left() < 30:
            break
        n_b = len(best_h)
        dx_b = 2.0 / n_b
        ts_b = n_b / 2.0
        if trial % 4 == 0:
            perturb = 0.06 * (rng.random(n_b) - 0.5)
        elif trial % 4 == 1:
            idx = rng.choice(n_b, size=max(8, n_b // 6), replace=False)
            perturb = np.zeros(n_b)
            perturb[idx] = rng.choice([-1, 1], size=len(idx)) * 0.12
        elif trial % 4 == 2:
            v = lmo_vertex(best_h, dx_b, 40)
            perturb = 0.4 * (v - best_h)
        else:
            # Frequency-based perturbation
            freq = rng.randint(3, 10)
            perturb = 0.05 * np.sin(2 * np.pi * freq * np.arange(n_b) / n_b + rng.random() * 2 * np.pi)

        h_try = project_feasible(best_h + perturb, ts_b)
        h_opt, c5_val = slsqp_direct(h_try, dx_b, ts_b, 2000, 600)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_opt
        if time_left() > 20:
            h_lp3, c5_lp3 = lp_polish(h_opt, dx_b, ts_b, max_iters=100, init_step=0.08)
            if c5_lp3 < best_c5:
                best_c5 = c5_lp3
                best_h = h_lp3

    # Phase 10: Final intensive polish
    if time_left() > 25 and best_h is not None:
        n_b = len(best_h)
        dx_b = 2.0 / n_b
        ts_b = n_b / 2.0
        h_f, c5_f = slsqp_direct(best_h, dx_b, ts_b, 6000, 2000)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f
        if time_left() > 20:
            h_lp4, c5_lp4 = lp_polish(best_h, dx_b, ts_b, max_iters=300, init_step=0.05)
            if c5_lp4 < best_c5:
                best_c5 = c5_lp4
                best_h = h_lp4
        if time_left() > 15:
            h_pt2, c5_pt2 = pair_transfer_polish(best_h, dx_b, ts_b, n_top=35, n_iter=6)
            if c5_pt2 < best_c5:
                best_c5 = c5_pt2
                best_h = h_pt2
        if time_left() > 30:
            h_ep3, c5_ep3 = epigraph_iterative(best_h, dx_b, ts_b, n_act=8, max_rounds=10, maxiter_per=300)
            if c5_ep3 < best_c5:
                best_c5 = c5_ep3
                best_h = h_ep3
        if time_left() > 15:
            h_fw3, c5_fw3 = frank_wolfe_full(best_h, dx_b, ts_b, max_iter=60, n_active=32)
            if c5_fw3 < best_c5:
                best_c5 = c5_fw3
                best_h = h_fw3
            if time_left() > 10:
                h_lp5, c5_lp5 = lp_polish(best_h, dx_b, ts_b, max_iters=200, init_step=0.04)
                if c5_lp5 < best_c5:
                    best_c5 = c5_lp5
                    best_h = h_lp5

    # Final validation
    if best_h is None:
        best_h = h0
        best_c5 = c5_of_h(h0, 2.0 / len(h0))
    else:
        n_final = len(best_h)
        best_h = project_feasible(best_h, n_final / 2.0)
        best_c5 = c5_of_h(best_h, 2.0 / n_final)

    return best_h, best_c5, len(best_h)
