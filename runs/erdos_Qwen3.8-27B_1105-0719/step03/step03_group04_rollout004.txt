def gradient_vector(h, n, dx):
    """Compute gradient of all profile components w.r.t. h."""
    M = 2 * n - 1
    G = np.zeros((M, n))
    center = n - 1
    G[center, :] = dx * (1.0 - 2.0 * h)
    for d in range(1, n):

        G[center + d, d:n] += dx * (1.0 - h[0:n-d])
        G[center - d, 0:n-d] += dx * (1.0 - h[d:n])
        G[center - d, d:n] += -dx * h[0:n-d]
    return G

def logsumexp_score(h, tau):
    """Compute log-sum-exp of the profile as smooth surrogate for max."""
    profile = compute_profile(h)
    pmax = profile.max()
    shifted = (profile - pmax) / tau
    # For numerical stability
    m = shifted.max()
    lse = pmax + tau * (m + np.log(np.sum(np.exp(shifted - m))))
    return lse

def logsumexp_grad(h, tau):

    n = len(h)
    dx = 2.0 / n

    return grad

import numpy as np
from scipy.optimize import linprog, minimize
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def repair_feasible(h):
    n = len(h)
    h = np.clip(np.asarray(h, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, 1.0 - h[idx])
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)[::-1]
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def gradient_matrix(h, n, dx):
    M = 2 * n - 1
    G = np.zeros((M, n))
    center = n - 1
    G[center, :] = dx * (1.0 - 2.0 * h)
    for d in range(1, n):
        G[center + d, 0:n-d] += -dx * h[d:n]
        G[center + d, d:n] += dx * (1.0 - h[0:n-d])
        G[center - d, 0:n-d] += dx * (1.0 - h[d:n])
        G[center - d, d:n] += -dx * h[0:n-d]
    return G

def lp_minmax_step(h, beam, radius, max_constr=30):
    n = len(h)
    dx = 2.0 / n
    G = gradient_matrix(h, n, dx)
    profile = compute_profile(h)
    T = profile.max()
    thresh = T - beam
    active = np.where(profile >= thresh)[0]
    if len(active) == 0:
        active = np.array([np.argmax(profile)])
    if len(active) > max_constr:
        top_idx = np.argsort(profile[active])[-max_constr:]
        active = active[top_idx]
    M = len(active)
    G_a = G[active, :]
    p_a = profile[active]
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_ub = np.zeros((M, n + 1))
    A_ub[:, :n] = G_a
    A_ub[:, -1] = -1.0
    b_ub = -p_a
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    lower = np.maximum(-h, -radius)
    upper = np.minimum(1.0 - h, radius)
    bounds = list(zip(lower, upper))
    bounds.append((0.0, None))
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'maxiter': 5000})
        if res.success and res.x is not None and np.isfinite(res.x).all():
            h_new = repair_feasible(h + res.x[:n])
            return h_new, float(res.x[-1])
    except Exception:
        pass
    return None, None

def slsqp_polish(h, maxiter, all_constr=False):
    n = len(h)
    dx = 2.0 / n
    M_full = 2 * n - 1
    c0 = exact_score(h)
    profile = compute_profile(h)
    T = profile.max()
    if all_constr:
        active = np.arange(M_full)
    else:
        beam = 0.015
        active = np.where(profile >= T - beam)[0]
        if len(active) == 0:
            active = np.array([np.argmax(profile)])
    M = len(active)

    def obj_fn(x):
        return x[-1]

    def obj_jac_fn(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g

    def eq_fn(x):
        return x[:-1].sum() - n / 2.0

    def eq_jac_fn(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g

    def ineq_fn(x):
        hv = x[:-1]
        t = x[-1]
        prof = np.correlate(hv, 1 - hv, mode="full") * dx
        return t - prof[active]

    def ineq_jac_fn(x):
        hv = x[:-1]
        G = gradient_matrix(hv, n, dx)
        J = np.zeros((M, n + 1))
        J[:, :-1] = -G[active, :]
        J[:, -1] = 1.0
        return J

    x0 = np.concatenate([h, [c0 + 1e-3]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    cons = [
        {'type': 'eq', 'fun': eq_fn, 'jac': eq_jac_fn},
        {'type': 'ineq', 'fun': ineq_fn, 'jac': ineq_jac_fn}
    ]
    try:
        res = minimize(obj_fn, x0, method='SLSQP', jac=obj_jac_fn,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        if res.x is not None and np.isfinite(res.x).all():
            h_sol = repair_feasible(res.x[:-1])
            c_sol = exact_score(h_sol)
            if c_sol <= c0 + 1e-10:
                return h_sol, c_sol
    except Exception:
        pass
    return h.copy(), c0

def lse_gradient(h, tau):
    n = len(h)
    dx = 2.0 / n
    profile = compute_profile(h)
    pmax = profile.max()
    shifted = (profile - pmax) / tau
    m = shifted.max()
    exps = np.exp(shifted - m)
    w = exps / exps.sum()
    G = gradient_matrix(h, n, dx)
    grad = G.T @ w
    grad -= grad.mean()
    return grad

def symmetrize(h):
    return (h + h[::-1]) / 2.0

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

def make_start(n, rng, mode):
    h = np.zeros(n)
    if mode == 0:
        h[:] = 0.5
        phase = rng.uniform(0, 2 * np.pi)
        freq = rng.uniform(1.0, 3.0)
        for i in range(n):
            h[i] += 0.3 * np.sin(2 * np.pi * freq * i / n + phase)
    elif mode == 1:
        nb = rng.randint(3, 8)
        block = n // nb
        for b in range(nb):
            h[b * block:(b + 1) * block] = rng.uniform(0.15, 0.85)
        if nb * block < n:
            h[nb * block:] = rng.uniform(0.15, 0.85, n - nb * block)
    elif mode == 2:
        phi = 0.6180339887498949
        amp = rng.uniform(0.2, 0.5)
        for i in range(n):
            h[i] = 0.5 + amp * np.sin(2 * np.pi * ((i * phi) % 1.0))
    elif mode == 3:
        half = n // 2
        h[:half] = rng.uniform(0.1, 0.9, half)
        h[half:] = 1.0 - h[:half][::-1]
        if n % 2 == 1:
            h[half] = 0.5
    elif mode == 4:
        h[:] = 0.5 + 0.25 * np.sin(np.pi * np.arange(n) / n * rng.randint(2, 5))
    else:
        h[:] = rng.uniform(0.25, 0.75, n)
    return repair_feasible(h)

def optimize_main(h, t_start, budget_s, max_iters=300):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c = exact_score(h)
    radius = 0.08
    beam = 0.008
    no_improve = 0
    lse_tau = 0.03

    for it in range(max_iters):
        elapsed = time.perf_counter() - t_start
        if budget_s - elapsed < 8:
            break
        c = exact_score(h)
        if c < best_c - 1e-14:
            best_c = c
            best_h = h.copy()
            no_improve = 0
            radius = min(radius * 1.1, 0.12)
        else:
            no_improve += 1
        if no_improve > 60:
            break
        # LP trust region step
        h_lp, t_lp = lp_minmax_step(h, beam, radius, max_constr=25)
        improved = False
        if h_lp is not None:
            c_lp = exact_score(h_lp)
            if c_lp < best_c - 1e-12:
                h = h_lp
                improved = True
                continue
        # LSE gradient step
        grad = lse_gradient(h, lse_tau)
        gnorm = np.linalg.norm(grad)
        if gnorm > 1e-14:
            grad = grad / gnorm
            for alpha in [0.015, 0.008, 0.004, 0.002, 0.001]:
                h_try = repair_feasible(h - alpha * grad)
                s = exact_score(h_try)
                if s < best_c - 1e-12:
                    h = h_try
                    improved = True
                    break
        if not improved:
            # Try wider beam LP
            if no_improve > 10:
                h_lp2, _ = lp_minmax_step(h, 0.02, radius * 0.7, max_constr=15)
                if h_lp2 is not None:
                    c_lp2 = exact_score(h_lp2)
                    if c_lp2 < best_c - 1e-12:
                        h = h_lp2
                        no_improve = 0
                        continue
            # Try SLSQP periodically
            if no_improve > 20 and no_improve % 15 == 0:
                h_sl, c_sl = slsqp_polish(h, maxiter=150)
                if c_sl < best_c - 1e-12:
                    h = h_sl
                    best_h = h_sl.copy()
                    best_c = c_sl
                    no_improve = 0
                    continue
            # Shrink radius and beam
            radius *= 0.85
            beam *= 0.9
            lse_tau = max(lse_tau * 0.9, 0.003)
            if radius < 0.002:
                break
    return best_h, best_c
