```python
    b = 0.0

            a += 1.0
            b += (h[p] + 1.0 - h[q])
        else:

            a += 1.0
            b += (h[p] + 1.0 - h[q])
        else:
```

- $-h_i \alpha \leq \delta_i \leq (1-h_i) \alpha$

```python
def pair_move_optimize(h, p, q, corrs, n):
    """Find optimal t for pair (p,q) and return (best_t, best_c5)."""
    L = max(-h[p], h[q] - 1.0)

    # and d_hi = max(d_hi_p, d_hi_p2, d_hi_q, d_hi_q2)

    # Actually, it's easier to just evaluate all lags in the affected range
    # and use the original corrs for unaffected ones.

    # f(t) = max over all k of C_k(t)
    # where C_k(t) = C_k + a_k*t^2 + b_k*t for affected k, else C_k

    # We need the max of corrs[k] for k not in affected_ks.
```

- Compute $C_d + A \cdot t_j^2 + B \cdot t_j$ for all $j, i$ → shape $(S, M)$

```python
import numpy as np
from scipy.optimize import linprog
import time

def project_feasible(x, target_sum):
    n = len(x)
    lo = float(x.min()) - 2.0
    hi = float(x.max()) + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.any():
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def c5_of_h(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full').max() * dx

def corr_of_h(h):
    return np.correlate(h, 1.0 - h, mode='full')

def grad_lag(h, d, n):
    """Gradient of correlation at lag d (offset d = k - (n-1))."""
    grad = np.zeros(n)
    i_lo = max(0, -d)
    i_hi = min(n - 1, n - 1 - d)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx] += (1.0 - h[idx + d])
    grad[idx + d] += (-h[idx])
    return grad

def pair_coeff(h, p, q, d, n):
    """Compute (a, b) such that C_d changes by a*t^2 + b*t for pair move (p+ t, q - t)."""
    a = 0.0
    b = 0.0
    hp = h[p]
    hq = h[q]
    
    if 0 <= p + d < n:
        if p + d == q:
            a += 1.0
            b += (hp + 1.0 - hq)
        else:
            b += (1.0 - h[p + d])
    
    if 0 <= q + d < n:
        if q + d == p:
            a += 1.0
            b += -(hq + 1.0 - hp)
        else:
            b += -(1.0 - h[q + d])
    
    if 0 <= p - d < n:
        if p - d == q:
            a += 1.0
            b += -(hq + 1.0 - hp)
        else:
            b += -h[p - d]
    
    if 0 <= q - d < n:
        if q - d == p:
            a += 1.0
            b += (hp + 1.0 - hq)
        else:
            b += h[q - d]
    
    return (a, b)

def pair_move_evaluate(h, p, q, corrs, n):
    """Evaluate the best t for pair (p,q). Returns (best_c5, best_t)."""
    L = max(-h[p], h[q] - 1.0)
    U = min(1.0 - h[p], h[q])
    if U - L < 1e-14:
        return (c5_of_h(h), 0.0)
    
    # Determine affected lag range
    d_lo = min(-p, p - n + 1, -q, q - n + 1)
    d_hi = max(n - 1 - p, p, n - 1 - q, q)
    d_lo = max(d_lo, -(n - 1))
    d_hi = min(d_hi, n - 1)
    
    ds = np.arange(d_lo, d_hi + 1)
    k_arr = ds + (n - 1)
    # Filter valid correlation indices
    valid = (k_arr >= 0) & (k_arr <= 2 * n - 2)
    ds = ds[valid]
    k_arr = k_arr[valid]
    
    if len(ds) == 0:
        return (c5_of_h(h), 0.0)
    
    # Compute coefficients
    a_arr = np.zeros(len(ds))
    b_arr = np.zeros(len(ds))
    for i, d in enumerate(ds):
        a_arr[i], b_arr[i] = pair_coeff(h, p, q, int(d), n)
    
    aff_mask = (a_arr != 0.0) | (b_arr != 0.0)
    if not aff_mask.any():
        return (c5_of_h(h), 0.0)
    
    a_aff = a_arr[aff_mask]
    b_aff = b_arr[aff_mask]
    k_aff = k_arr[aff_mask]
    c_aff = corrs[k_aff]
    
    # Unaffected max
    all_mask = np.ones(2 * n - 1, dtype=bool)
    all_mask[k_aff] = False
    if all_mask.any():
        max_unaff = float(corrs[all_mask].max())
    else:
        max_unaff = 0.0
    
    # Sample and evaluate
    S = 80
    t_samples = np.linspace(L, U, S)
    M = len(a_aff)
    
    # Vectorized: (S, M)
    t2 = t_samples[:, None] ** 2
    t1 = t_samples[:, None]
    vals = c_aff[None, :] + a_aff[None, :] * t2 + b_aff[None, :] * t1
    f_vals = vals.max(axis=1)
    f_vals = np.maximum(f_vals, max_unaff)
    
    best_idx = int(f_vals.argmin())
    t_best = float(t_samples[best_idx])
    
    # Refine
    if 0 < best_idx < S - 1:
        t_lo = float(t_samples[best_idx - 1])
        t_hi = float(t_samples[best_idx + 1])
        t_fine = np.linspace(t_lo, t_hi, 40)
        t2f = t_fine[:, None] ** 2
        t1f = t_fine[:, None]
        vals_f = c_aff[None, :] + a_aff[None, :] * t2f + b_aff[None, :] * t1f
        f_fine = vals_f.max(axis=1)
        f_fine = np.maximum(f_fine, max_unaff)
        t_best = float(t_fine[f_fine.argmin()])
    
    # Compute actual C5
    h_new = h.copy()
    h_new[p] += t_best
    h_new[q] -= t_best
    actual = c5_of_h(h_new)
```

```python
def lp_soft_slack(h, dx, target_sum, alpha, n, top_n_lags=40, rho_q=0.3, rho_s=5.0):
    """Solve soft minmax LP and return direction delta."""
    corrs = corr_of_h(h)
    max_c = corrs.max()
    
    # Select active lags: top by value + some near-top
    k_sorted = np.argsort(corrs)[-top_n_lags:]
    k_set = set(k_sorted.tolist())
    
    # Add near-top lags
    margin = 0.05 * max_c
    near = np.where(corrs >= max_c - margin)[0]
    k_set = k_set.union(near.tolist())
    k_arr = np.array(sorted(k_set))
    M = len(k_arr)
    
    if M < 3:
        k_arr = np.argsort(corrs)[-10:]
        M = len(k_arr)
    
    # Build gradient matrix for selected lags
    G = np.zeros((M, n))  # G[j, i] = grad_k[i] for lag at index k_arr[j]
    for j, k in enumerate(k_arr):
        d = k - (n - 1)
        G[j] = grad_lag(h, d, n)
    
    # LP variables: delta (n), s (M), q (1) → total n + M + 1
    nvar = n + M + 1
    q_idx = n + M  # index of q variable
    s_start = n    # s variables are at indices n..n+M-1
    
    # Objective: rho_q * q + rho_s * sum(s_j)
    c_obj = np.zeros(nvar)
    c_obj[q_idx] = rho_q
    c_obj[s_start:s_start + M] = rho_s
    
    # Constraints: for each j: dx * G[j] @ delta + s_j - q <= -corrs[k_arr[j]]
    # Rearranged: dx * G[j] @ delta + s_j - q + corrs[k_arr[j]] <= 0
    A_ub = np.zeros((M, nvar))
    for j in range(M):
        A_ub[j, :n] = dx * G[j]
        A_ub[j, s_start + j] = 1.0
        A_ub[j, q_idx] = -1.0
    b_ub = -corrs[k_arr]
    
    # Equality: sum(delta) = 0
    A_eq = np.zeros((1, nvar))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds
    bounds = []
    for i in range(n):
        bounds.append((-h[i] * alpha, (1.0 - h[i]) * alpha))
    for j in range(M):
        bounds.append((0.0, None))  # s_j >= 0
    bounds.append((None, None))  # q unbounded
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status != 0:
            return None
        return res.x[:n]
    except Exception:
        return None
```
