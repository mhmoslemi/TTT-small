def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx])
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-14:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-14:
                break

    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h


def feasible_interval(h, v):
    """Compute the feasible alpha interval for h + alpha*v."""
    n = len(h)

def line_search(h, v, n_samples=200):
    """Exact line search along direction v from h."""
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    
    v = np.asarray(v, dtype=np.float64)
    v_sum = v.sum()
    if abs(v_sum) > 1e-10:
        v = v - v_sum / n
    
    max_abs = np.max(np.abs(v))
    if max_abs < 1e-10:
        return None, None
    
    v = v / max_abs  # normalize so max |v_i| = 1
    
    interval = feasible_interval(h, v)
    if interval is None:
        return None, None
    
    lo, hi = interval
    if hi - lo < 1e-10:
        return None, None
    
    # Compute quadratic coefficients A, B, C for each lag
    # R_k(alpha) = A_k + alpha * B_k + alpha^2 * C_k
    # A_k = raw correlation of h at lag k
    # B_k = sum_i [v_i*(1-h_{i+k}) - h_i*v_{i+k}]
    # C_k = -sum_i v_i * v_{i+k}
    
    A = np.correlate(h, 1.0 - h, mode='full')
    B = np.correlate(v, 1.0 - h, mode='full') - np.correlate(h, v, mode='full')
    C = -np.correlate(v, v, mode='full')
    
    # Sample the interval
    alphas = np.linspace(lo, hi, n_samples)
    
    # Evaluate max at each alpha
    # Shape: (n_samples, M)
    vals = A[np.newaxis, :] + alphas[:, np.newaxis] * B[np.newaxis, :] + (alphas[:, np.newaxis])**2 * C[np.newaxis, :]
    max_vals = np.max(vals, axis=1)
    
    best_idx = np.argmin(max_vals)
    best_alpha = alphas[best_idx]
    best_val = max_vals[best_idx] * dx
    
    # Refine with ternary search in a small window
    # Actually, let's do a finer grid around the best point
    if best_idx > 0 and best_idx < n_samples - 1:
        step = alphas[1] - alphas[0]
        fine_lo = alphas[max(0, best_idx - 3)]
        fine_hi = alphas[min(n_samples - 1, best_idx + 3)]
        fine_alphas = np.linspace(fine_lo, fine_hi, 100)
        fine_vals = A[np.newaxis, :] + fine_alphas[:, np.newaxis] * B[np.newaxis, :] + (fine_alphas[:, np.newaxis])**2 * C[np.newaxis, :]
        fine_max = np.max(fine_vals, axis=1)
        fine_best_idx = np.argmin(fine_max)
        best_alpha = fine_alphas[fine_best_idx]
        best_val = fine_max[fine_best_idx] * dx
    
    # Also check endpoints
    for ep in [lo, hi]:
        vals_ep = A + ep * B + ep**2 * C
        max_ep = np.max(vals_ep) * dx
        if max_ep < best_val:
            best_alpha = ep
            best_val = max_ep
    
    h_new = h + best_alpha * v
    h_new = repair_feasible(h_new)
    actual_score = exact_score(h_new)
    
    return h_new, actual_score


def gen_direction_gradient(h, profile, n, dx, rng):
    """Generate a direction based on the subgradient of the max."""
    c_max = np.max(profile)
    active = np.where(profile >= c_max - 0.01 * (c_max - np.min(profile)))[0]
    if len(active) == 0:
        active = np.array([np.argmax(profile)])
    
    M = 2 * n - 1
    center = n - 1
    
    # Weight active lags
    w = (profile[active] - profile[active].min() + 1e-8)
    w = w / w.sum()
    
    # Compute gradient of the weighted sum of active lags
    v = np.zeros(n)
    for idx, k in enumerate(active):
        d = k - center
        if d == 0:
            v += w[idx] * dx * (1.0 - 2.0 * h)
        else:
            idx1 = np.arange(n - abs(d))
            if d > 0:
                v[idx1] += -w[idx] * dx * h[idx1 + d]
                v[idx1 + d] += w[idx] * dx * (1.0 - h[idx1])
            else:
                v[idx1] += w[idx] * dx * (1.0 - h[idx1 + abs(d)])
                v[idx1 + abs(d)] += -w[idx] * dx * h[idx1]
    
    v -= v.mean()
    max_abs = np.max(np.abs(v))
    if max_abs < 1e-12:
        return None
    return v / max_abs


def gen_direction_random(n, rng):
    """Random zero-sum direction."""
    v = rng.randn(n)
    v -= v.mean()
    max_abs = np.max(np.abs(v))
    if max_abs < 1e-10:
        return None
    return v / max_abs


def gen_direction_smooth(n, rng, freq_range=(1, 5)):
    """Smooth periodic direction."""
    freq = rng.uniform(*freq_range)
    phase = rng.uniform(0, 2 * np.pi)
    x = np.arange(n) / n * 2 * np.pi
    v = np.sin(freq * x + phase) + 0.5 * np.sin(2 * freq * x + 2 * phase)
    v -= v.mean()
    max_abs = np.max(np.abs(v))
    if max_abs < 1e-10:
        return None
    return v / max_abs


def gen_direction_sparse(h, n, rng, num_pairs=3):
    """Sparse transfer direction: move mass from high to low."""
    v = np.zeros(n)
    for _ in range(num_pairs):
        # Pick source from high values, destination from low values
        source_probs = h ** 2 + 1e-6
        source_probs /= source_probs.sum()
        dest_probs = (1 - h) ** 2 + 1e-6
        dest_probs /= dest_probs.sum()
        
        i = rng.choice(n, p=source_probs)
        j = rng.choice(n, p=dest_probs)
        if i != j:
            v[i] = -1.0
            v[j] = 1.0
    
    v -= v.mean()
    max_abs = np.max(np.abs(v))
    if max_abs < 1e-10:
        return None
    return v / max_abs

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full")) * dx)


def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx


def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx])
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-14:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-14:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-14:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h


def feasible_interval(h, v):
    v_pos = v > 1e-15
    v_neg = v < -1e-15
    lo = 0.0
    hi = 0.0
    any_pos = False
    any_neg = False
    if np.any(v_pos):
        lo = max(lo, np.max(-h[v_pos] / v[v_pos]))
        hi = min(hi if any_pos else np.inf, np.min((1.0 - h[v_pos]) / v[v_pos]))
        any_pos = True
    if np.any(v_neg):
        lo = max(lo, np.max((1.0 - h[v_neg]) / v[v_neg]))
        hi = min(hi if any_neg else np.inf, np.min(-h[v_neg] / v[v_neg]))
        any_neg = True
    if not any_pos and not any_neg:
        return None
    lo = max(0.0, lo)
    if lo < hi - 1e-12:
        return lo, hi
    return None


def line_search(h, v, n_coarse=200, n_fine=100):
    n = len(h)
    dx = 2.0 / n
    v = np.asarray(v, dtype=np.float64).copy()
    v -= v.mean()
    norm = np.linalg.norm(v)
    if norm < 1e-12:
        return None, None
    v = v / norm

    interval = feasible_interval(h, v)
    if interval is None:
        return None, None
    lo, hi = interval

    A = np.correlate(h, 1.0 - h, mode='full')
    B = np.correlate(v, 1.0 - h, mode='full') - np.correlate(h, v, mode='full')
    C = -np.correlate(v, v, mode='full')

    alphas = np.linspace(lo, hi, n_coarse)
    vals = A[np.newaxis, :] + alphas[:, np.newaxis] * B[np.newaxis, :] + (alphas[:, np.newaxis] ** 2) * C[np.newaxis, :]
    max_vals = np.max(vals, axis=1)
    best_idx = np.argmin(max_vals)
    best_alpha = alphas[best_idx]
    best_val = max_vals[best_idx]

    if best_idx > 2 and best_idx < n_coarse - 3:
        step = alphas[1] - alphas[0]
        f_lo = alphas[max(0, best_idx - 4)]
        f_hi = alphas[min(n_coarse - 1, best_idx + 4)]
        fine_alphas = np.linspace(f_lo, f_hi, n_fine)
        fvals = A[np.newaxis, :] + fine_alphas[:, np.newaxis] * B[np.newaxis, :] + (fine_alphas[:, np.newaxis] ** 2) * C[np.newaxis, :]
        fmax = np.max(fvals, axis=1)
        fidx = np.argmin(fmax)
        best_alpha = fine_alphas[fidx]
        best_val = fmax[fidx]

    for ep in [lo, hi]:
        ev = np.max(A + ep * B + ep ** 2 * C)
        if ev < best_val:
            best_alpha = ep
            best_val = ev

    h_new = repair_feasible(h + best_alpha * v)
    actual_score = exact_score(h_new)
    return h_new, actual_score


def gen_dir_gradient(h, profile, n):
    c_max = np.max(profile)
    c_min = np.min(profile)
    spread = c_max - c_min
    if spread < 1e-10:
        return None
    beam = 0.05 * spread
    active = np.where(profile >= c_max - beam)[0]
    if len(active) == 0:
        active = np.array([np.argmax(profile)])
    w = profile[active] - profile[active].min() + 1e-8
    w = w / w.sum()
    M = 2 * n - 1
    center = n - 1
    v = np.zeros(n)
    for idx, k in enumerate(active):
        d = int(k) - center
        if d == 0:
            v += w[idx] * (1.0 - 2.0 * h)
        else:
            ad = abs(d)
            idx1 = np.arange(n - ad)
            if d > 0:
                v[idx1] += -w[idx] * h[idx1 + ad]
                v[idx1 + ad] += w[idx] * (1.0 - h[idx1])
            else:
                v[idx1] += w[idx] * (1.0 - h[idx1 + ad])
                v[idx1 + ad] += -w[idx] * h[idx1]
    v -= v.mean()
    norm = np.linalg.norm(v)
    if norm < 1e-12:
        return None
    return v / norm

def gen_dir_sparse(h, n, rng):

    n_pairs = rng.randint(2, 6)

        j = rng.choice(n, p=dp)

            v[i] -= 1.0
            v[j] += 1.0
    v -= v.mean()
    norm = np.linalg.norm(v)

def gen_dir_antisym(n, rng):

def gen_dir_block(h, n, rng):
    """Direction that shifts mass between contiguous blocks."""
    n_blocks = rng.randint(2, 5)
    block_sizes = np.ones(n_blocks) / n_blocks
    starts = np.concatenate([[0], np.cumsum(block_sizes[:-1])]).astype(int)
    v = np.zeros(n)
    for i in range(n_blocks - 1):
        si, sj = starts[i], starts[i + 1]
        di, dj = starts[i + 1], starts[i + 2] if i + 2 < n_blocks else 0
        if i + 1 < n_blocks:
            di = starts[i + 1]
            dj = starts[i + 1] + int(n_blocks * block_sizes[i + 1])
            mass = (h[si:di].mean() - h[dj:min(dj + (di - si), n)].mean())
            if abs(mass) > 1e-6:
                v[si:di] -= np.sign(mass) * 0.5
                v[dj:min(dj + (di - si), n)] += np.sign(mass) * 0.5
    v -= v.mean()
    norm = np.linalg.norm(v)
    if norm < 1e-12:
        return None
    return v / norm

def interp_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()

    fx = np.concatenate([[0.0], xs, [2.0]])
    fh = np.concatenate([[h[0]], h, [h[-1]]])
    hn = np.interp(xt, fx, fh)
    return repair_feasible(np.clip(hn, 0, 1))


def slsqp_refine(h0, maxiter, tol):
    n = len(h0)
    dx = 2.0 / n
    M = 2 * n - 1

    def obj_func(x):
        return x[-1]

    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g

    def eq_fun(x):
        return x[:-1].sum() - n / 2.0

    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g

    def ineq_fun(x):
        h = x[:-1]
        return x[-1] - np.correlate(h, 1 - h, mode="full") * dx

    def ineq_jac(x):
        h = x[:-1]
        G = np.zeros((M, n))
        center = n - 1
        G[center, :] = dx * (1.0 - 2.0 * h)
        for d in range(1, n):
            idx1 = np.arange(n - d)
            G[center + d, idx1] += -dx * h[idx1 + d]
            G[center + d, idx1 + d] += dx * (1.0 - h[idx1])
            G[center - d, idx1] += dx * (1.0 - h[idx1 + d])
            G[center - d, idx1 + d] += -dx * h[idx1]
        J = np.zeros((M, n + 1))
        J[:, :-1] = -G
        J[:, -1] = 1.0
        return J

    t0 = exact_score(h0) + 1e-3
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, t0 + 2.0)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    try:
        res = minimize(obj_func, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': tol})
        if res.x is not None and np.isfinite(res.x).all():
            hs = repair_feasible(res.x[:-1])
            if np.isfinite(exact_score(hs)):
                return hs
    except Exception:
        pass
    return None

    if tl() > 40:
        n0 = len(best_h)
        for nd in [n0 - 2, n0 + 2, n0 - 4, n0 + 4]:
            if nd < 30 or tl() < 15:
                continue
            hi = interp_to_n(best_h, nd)
            ci = exact_score(hi)
            if ci < best_c + 0.005:
                bh, bc = improve_local(hi, tl, rng, max_rounds=100)
                if bc < best_c - 1e-14:
                    best_h, best_c = bh, bc

    # Phase 9: Final intensive push
    if tl() > 30:
        bh, bc = improve_local(best_h, tl, rng, max_rounds=200)
        if bc < best_c - 1e-14:
            best_h, best_c = bh, bc

    # Final SLSQP polish
    if tl() > 20:
        hs = slsqp_refine(best_h, maxiter=300, tol=1e-13)
        if hs is not None:
            cs = exact_score(hs)
            if cs < best_c - 1e-14:
                best_h, best_c = hs, cs

    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)

    try:
        er = evaluate_erdos_solution(best_h)
        if isinstance(er, (float, int)):
            final_c = min(final_c, float(er))
        elif isinstance(er, tuple) and len(er) >= 2:
            final_c = min(final_c, float(er[0]))
    except Exception:
        pass

    return best_h, final_c, len(best_h)
