def repair_feasible(h):
    n = len(h)
    h = np.clip(h.astype(np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        # Need to add mass
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for idx in order:
            add = min(remaining, capacity[idx])
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-13:
                break
    else:
        # Need to remove mass
        order = np.argsort(h)
        remaining = -diff
        for idx in order:
            rem = min(remaining, h[idx])
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-13:
                break
    # Final fix
    diff2 = target - h.sum()
    if abs(diff2) > 1e-14:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmin(h))
        h[idx] = np.clip(h[idx] + diff2, 0.0, 1.0)
    return h

    xs = np.linspace(0, 2, n, endpoint=False) + 1.0/n

def full_pair_descent(h, max_sweeps=200, n_grid=25, max_pairs_per_sweep=80,
                      t_start=None, time_limit=None, rng=None, use_guided=True):
    n = len(h)
    h_cur = h.copy()
    best_c = exact_score(h)
    if rng is None:
        rng = np.random.RandomState(0)
    no_improve = 0
    for sweep in range(max_sweeps):
        if t_start is not None and time_limit is not None:
            if time.perf_counter() - t_start > time_limit:
                break
        profile = get_profile(h_cur)
        c_cur = float(np.max(profile))
        pairs_to_try = []
        if use_guided:
            margin = max(0.001, 2.0 * (c_cur - float(np.median(profile))))
            active = np.where(profile >= c_cur - margin)[0]
            n_active = len(active)
            grad_approx = np.zeros(n)
            for am in active[:min(n_active, 40)]:
                d = am - (n - 1)
                if d >= 0:
                    for ii in range(n - d):
                        grad_approx[ii] += (1.0 - h_cur[ii + d])
                        grad_approx[ii + d] -= h_cur[ii]
                else:
                    dp = -d
                    for ii in range(n - dp):
                        grad_approx[ii] += (1.0 - h_cur[ii + dp])
                        grad_approx[ii + dp] -= h_cur[ii]
            n_pos = int(np.ceil(n * 0.4))
            n_neg = int(np.ceil(n * 0.4))
            pos_idx = np.argsort(-grad_approx)[:n_pos]
            neg_idx = np.argsort(grad_approx)[:n_neg]
            seen = set()
            count = 0
            max_attempts = max_pairs_per_sweep * 3
            attempts = 0
            while count < max_pairs_per_sweep and attempts < max_attempts:
                attempts += 1
                pi = pos_idx[rng.randint(n_pos)]
                ni = neg_idx[rng.randint(n_neg)]
                if pi == ni:
                    continue
                key = (min(pi, ni), max(pi, ni))
                if key in seen:
                    continue
                seen.add(key)
                pairs_to_try.append((int(pi), int(ni)))
                count += 1
            for _ in range(max(1, max_pairs_per_sweep // 4)):
                pi = rng.randint(n)
                ni = rng.randint(n)
                if pi != ni:
                    key = (min(pi, ni), max(pi, ni))
                    if key not in seen:
                        seen.add(key)
                        pairs_to_try.append((int(pi), int(ni)))
        else:
            all_pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
            rng.shuffle(all_pairs)
            pairs_to_try = all_pairs[:max_pairs_per_sweep]
        improved = False
        for i, j in pairs_to_try:
            if t_start is not None and time_limit is not None:
                if time.perf_counter() - t_start > time_limit:
                    return h_cur, best_c
            h_new, c_new = pair_line_search_full(h_cur, i, j, n_grid=n_grid)
            if c_new < best_c - 1e-14:
                best_c = c_new
                h_cur = h_new
                improved = True
        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0
    return h_cur, best_c

def slsqp_solve(h0, maxiter=150):
    n = len(h0)
    dx = 2.0 / n
    M = 2 * n - 1
    c0 = exact_score(h0)
    def obj(x):
        return x[-1]
    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    def eq_fun(x):
        return x[:-1].sum() - n / 2.0
    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g
    def ineq_fun(x):
        h = np.clip(x[:-1], 0, 1)
        t = x[-1]
        prof = np.correlate(h, 1.0 - h, mode="full") * dx
        return t - prof
    def ineq_jac(x):
        h = np.clip(x[:-1], 0, 1)
        J = np.zeros((M, n + 1))
        for m_idx in range(M):
            d = m_idx - (n - 1)
            if d >= 0:
                for i in range(n - d):
                    J[m_idx, i] += dx * (1.0 - h[i + d])
                    J[m_idx, i + d] -= dx * h[i]
            else:
                dp = -d
                for i in range(n - dp):
                    J[m_idx, i] += dx * (1.0 - h[i + dp])
                    J[m_idx, i + dp] -= dx * h[i]
        J[:, -1] = 1.0
        return J
    x0 = np.concatenate([h0, [c0 + 1e-3]])
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 0.5)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None and np.isfinite(res.x).all():
            h_new = repair_feasible(res.x[:-1])
            c_new = exact_score(h_new)
            if c_new <= c0 + 1e-9:
                return h_new, c_new
    except Exception:
        pass
    return None, None

def lse_descent(h, beta=80.0, max_iters=100, alpha_init=0.003, t_start=None, time_limit=None):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best_c = exact_score(h)
    alpha = alpha_init
    stuck = 0
    for it in range(max_iters):
        if t_start is not None and time_limit is not None:
            if time.perf_counter() - t_start > time_limit:
                break
        profile = get_profile(h)
        c = float(np.max(profile))
        scaled = beta * (profile - c)
        weights = np.exp(scaled)
        wsum = weights.sum()
        if wsum < 1e-300:
            break
        weights /= wsum
        grad = np.zeros(n)
        for m_idx in range(len(profile)):
            wm = weights[m_idx]
            if wm < 1e-6:
                continue
            d = m_idx - (n - 1)
            if d >= 0:
                for i in range(n - d):
                    grad[i] += wm * (1.0 - h[i + d])
                    grad[i + d] -= wm * h[i]
            else:
                dp = -d
                for i in range(n - dp):
                    grad[i] += wm * (1.0 - h[i + dp])
                    grad[i + dp] -= wm * h[i]
        grad *= dx
        grad -= grad.mean()
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad /= gnorm
        improved = False
        for a in [alpha, alpha * 0.5, alpha * 0.25, alpha * 0.1]:
            h_try = repair_feasible(h - a * grad)
            s = exact_score(h_try)
            if s < best_c - 1e-14:
                h = h_try
                best_c = s
                improved = True
                alpha = min(alpha * 1.1, 0.02)
                stuck = 0
                break
            elif s < c - 1e-14:
                h = h_try
                improved = True
                break
        if not improved:
            alpha *= 0.5
            stuck += 1
            if stuck > 25 or alpha < 1e-10:
                break
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t_start_global = time.perf_counter()
    rng = np.random.RandomState(seed)
    np.random.seed(seed)

    def time_left():
        return budget_s + t_start_global - time.perf_counter()

    parent = None
    try:
        parent = np.array(initial_h_values, dtype=np.float64)
    except NameError:
        pass
    except Exception:
        pass

    if parent is None or len(parent) < 4:
        parent = np.full(70, 0.5)

    parent = np.clip(parent, 0.0, 1.0)
    parent = repair_feasible(parent)
    c_parent = exact_score(parent)

    best_h = parent.copy()
    best_c = c_parent
    n_parent = len(best_h)

    # Phase 1: Full pair coordinate descent from initial
    if time_left() > 100:
        ph1_limit = min(0.25 * budget_s, time_left() - 10)
        h_new, c_new = full_pair_descent(
            best_h, max_sweeps=300, n_grid=25, max_pairs_per_sweep=100,
            t_start=t_start_global, time_limit=ph1_limit, rng=rng, use_guided=True)
        if c_new < best_c - 1e-14:
            best_h = h_new
            best_c = c_new

    # Phase 2: LSE descent for smoothing
    if time_left() > 80:
        h_new, c_new = lse_descent(best_h, beta=60, max_iters=80,
                                   t_start=t_start_global,
                                   time_limit=min(0.1 * budget_s, time_left() - 10))
        if c_new < best_c - 1e-14:
            best_h = h_new
            best_c = c_new

    # Phase 3: Anti-symmetric optimization
    if time_left() > 100:
        m = n_parent // 2
        y0 = antisym_from_h(best_h)
        y_opt, c_asym = anti_sym_descent(
            y0, max_sweeps=300, n_grid=25, max_pairs_per_sweep=60,
            t_start=t_start_global, time_limit=min(0.2 * budget_s, time_left() - 10), rng=rng)
        h_asym = make_anti_sym(y_opt)
        h_asym = repair_feasible(h_asym)
        c_asym2 = exact_score(h_asym)
        if c_asym2 < best_c - 1e-14:
            best_h = h_asym
            best_c = c_asym2
            # Continue optimizing in anti-sym space
            y_opt2 = antisym_from_h(best_h)
            y_opt2, c_asym3 = anti_sym_descent(
                y_opt2, max_sweeps=200, n_grid=30, max_pairs_per_sweep=80,
                t_start=t_start_global, time_limit=min(0.1 * budget_s, time_left() - 10), rng=rng)
            h_asym2 = repair_feasible(make_anti_sym(y_opt2))
            c_asym3 = exact_score(h_asym2)
            if c_asym3 < best_c - 1e-14:
                best_h = h_asym2
                best_c = c_asym3

    # Phase 4: Multi-resolution exploration
    if time_left() > 80:
        for dn in [-4, -2, 2, 4, 6, -6]:
            if time_left() < 30:
                break
            new_n = n_parent + dn
            if new_n < 24 or new_n > 140 or new_n % 2 != 0:
                continue
            h_interp = interpolate_to_n(best_h, new_n)
            h_lse, c_lse = lse_descent(h_interp, beta=80, max_iters=50,
                                       t_start=t_start_global,
                                       time_limit=min(0.06 * budget_s, time_left() - 10))
            if c_lse < best_c - 1e-14:
                best_h = h_lse
                best_c = c_lse
            h_pd, c_pd = full_pair_descent(
                best_h, max_sweeps=100, n_grid=25, max_pairs_per_sweep=80,
                t_start=t_start_global, time_limit=min(0.05 * budget_s, time_left() - 10),
                rng=rng, use_guided=True)
            if c_pd < best_c - 1e-14:
                best_h = h_pd
                best_c = c_pd

    # Phase 5: SLSQP on best
    if time_left() > 60:
        h_sol, c_sol = slsqp_solve(best_h, maxiter=200)
        if h_sol is not None and c_sol < best_c - 1e-14:
            best_h = h_sol
            best_c = c_sol
            h_pd, c_pd = full_pair_descent(
                best_h, max_sweeps=100, n_grid=25, max_pairs_per_sweep=80,
                t_start=t_start_global, time_limit=min(0.05 * budget_s, time_left() - 10),
                rng=rng, use_guided=True)
            if c_pd < best_c - 1e-14:
                best_h = h_pd
                best_c = c_pd

    # Phase 6: Multi-start SLSQP
    if time_left() > 80:
        n_b = len(best_h)
        starts = [best_h.copy()]
        for eps in [0.005, 0.015, 0.03]:
            z = rng.randn(n_b)
            z -= z.mean()
            starts.append(repair_feasible(best_h + eps * z))
        for h0 in starts:
            if time_left() < 30:
                break
            h_sol, c_sol = slsqp_solve(h0, maxiter=100)
            if h_sol is not None and c_sol < best_c - 1e-14:
                best_h = h_sol
                best_c = c_sol
                h_pd, c_pd = full_pair_descent(
                    best_h, max_sweeps=80, n_grid=25, max_pairs_per_sweep=60,
                    t_start=t_start_global, time_limit=min(0.04 * budget_s, time_left() - 10),
                    rng=rng, use_guided=False)
                if c_pd < best_c - 1e-14:
                    best_h = h_pd
                    best_c = c_pd

    # Phase 7: Intensive pair descent with both guided and random
    if time_left() > 50:
        h_pd, c_pd = full_pair_descent(
            best_h, max_sweeps=400, n_grid=30, max_pairs_per_sweep=120,
            t_start=t_start_global, time_limit=min(0.15 * budget_s, time_left() - 10),
            rng=rng, use_guided=True)
        if c_pd < best_c - 1e-14:
            best_h = h_pd
            best_c = c_pd

    # Phase 8: LSE with different betas
    if time_left() > 40:
        for beta in [40, 120, 200]:
            if time_left() < 15:
                break
            h_lse, c_lse = lse_descent(best_h, beta=beta, max_iters=60,
                                       t_start=t_start_global,
                                       time_limit=min(0.04 * budget_s, time_left() - 5))
            if c_lse < best_c - 1e-14:
                best_h = h_lse
                best_c = c_lse

    # Phase 9: Anti-sym again on best
    if time_left() > 40:
        m = len(best_h) // 2
        y0 = antisym_from_h(best_h)
        y_opt, c_asym = anti_sym_descent(
            y0, max_sweeps=300, n_grid=30, max_pairs_per_sweep=80,
            t_start=t_start_global, time_limit=min(0.1 * budget_s, time_left() - 10), rng=rng)
        h_asym = repair_feasible(make_anti_sym(y_opt))
        c_asym = exact_score(h_asym)
        if c_asym < best_c - 1e-14:
            best_h = h_asym
            best_c = c_asym

    # Phase 10: Final resolution fine-tuning
    if time_left() > 30:
        n_final = len(best_h)
        for dn in [-2, 2, -4, 4]:
            if time_left() < 15:
                break
            new_n = n_final + dn
            if new_n < 24 or new_n > 140 or new_n % 2 != 0:
                continue
            h_interp = interpolate_to_n(best_h, new_n)
            h_pd, c_pd = full_pair_descent(
                h_interp, max_sweeps=150, n_grid=25, max_pairs_per_sweep=80,
                t_start=t_start_global, time_limit=min(0.04 * budget_s, time_left() - 5),
                rng=rng, use_guided=True)
            if c_pd < best_c - 1e-14:
                best_h = h_pd
                best_c = c_pd

    # Phase 11: Final polish - pair descent + LSE
    if time_left() > 20:
        h_pd, c_pd = full_pair_descent(
            best_h, max_sweeps=300, n_grid=35, max_pairs_per_sweep=150,
            t_start=t_start_global, time_limit=min(0.08 * budget_s, time_left() - 5),
            rng=rng, use_guided=True)
        if c_pd < best_c - 1e-14:
            best_h = h_pd
            best_c = c_pd
        if time_left() > 10:
            h_lse, c_lse = lse_descent(best_h, beta=100, max_iters=80,
                                       t_start=t_start_global,
                                       time_limit=min(0.03 * budget_s, time_left() - 3))
            if c_lse < best_c - 1e-14:
                best_h = h_lse
                best_c = c_lse

    # Final SLSQP if time permits
    if time_left() > 25:
        h_sol, c_sol = slsqp_solve(best_h, maxiter=200)
        if h_sol is not None and c_sol < best_c - 1e-14:
            best_h = h_sol
            best_c = c_sol

    # Final validation
    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)

    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            final_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 2:
            final_c = float(eval_result[0])
    except Exception:
        pass

    return best_h, final_c, len(best_h)
