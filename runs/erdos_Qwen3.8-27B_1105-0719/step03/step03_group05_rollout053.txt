Let me analyze the problem and the strategy carefully.

The key insight from the strategy is to work in the "complement symmetric" (anti-reflective) subspace where h(x) + h(2-x) = 1. This means:
- For even n=2m: h_j = y_j for j=0..m-1, and h_{n-1-j} = 1 - y_j
- The mass constraint sum(h) = n/2 is automatically satisfied
- We only need to optimize m variables instead of n

The previous program achieved C₅ ≤ 0.380917, and we need to beat 0.38085 (target 0.3808).

Key ideas to implement:
1. Work in the reduced anti-symmetric subspace where h_i + h_{n-1-i} = 1
2. Exact 1D line searches in reduced coordinates (single coordinate moves, pair moves)
3. LSE (log-sum-exp) descent in reduced coordinates
4. Active set methods in reduced coordinates
5. SLSQP in reduced coordinates (fewer variables, no equality constraint needed)
6. Multiple resolution attempts
7. Perturbation and restart strategies

Let me think about the mathematical structure:
- If h_i + h_{n-1-i} = 1 for all i, then the profile P_s = dx * sum h_i*(1-h_{i+s}) has some structure.
- Let me think about what the overlap integral looks like in this subspace.

For the anti-symmetric case: h_i = y_i, h_{n-1-i} = 1-y_i
The overlap at lag s: P_s = dx * sum_{i} h_i * (1 - h_{i+s})

When s = 0: P_0 = dx * sum h_i * (1-h_i) = dx * sum h_i (since h_i ∈ [0,1])... wait no, P_0 = dx * sum h_i * (1-h_i)

Actually let me reconsider. The profile is:
P_s = dx * sum_{i=0}^{n-1-s} h_i * (1-h_{i+s}) for s >= 0

For the anti-symmetric constraint h_i + h_{n-1-i} = 1:
- P_0 = dx * sum_i h_i(1-h_i) = dx * sum h_i - dx * sum h_i^2 = dx*(n/2) - dx*sum h_i^2
  (since sum h_i = n/2)
  So P_0 = dx*n/2 - dx*sum h_i^2 = 1 - dx*sum h_i^2

For other lags, the structure is more complex but the point is that in the reduced space we have m = n/2 variables.

Let me now write the code. I'll focus on:

1. `run()` function that:
   - Gets initial_h_values
   - Projects into anti-symmetric subspace
   - Runs exact line searches (single and pair moves) in reduced coordinates
   - Runs LSE descent in reduced coordinates
   - Runs SLSQP in reduced coordinates (no mass constraint needed!)
   - Tries different resolutions
   - Returns best solution

The key advantage: in the reduced space, SLSQP doesn't need the equality constraint sum(h)=n/2, making it much faster and more reliable.

import numpy as np
from scipy.optimize import minimize
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx] + 1e-15)
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx] + 1e-15)
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-14:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmin(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def to_reduced(h):
    """Convert full h to reduced anti-symmetric variables y."""
    n = len(h)
    m = n // 2
    y = np.zeros(m)
    for j in range(m):
        y[j] = 0.5 * (h[j] + 1.0 - h[n-1-j])
    return np.clip(y, 0.0, 1.0)

def from_reduced(y):
    """Convert reduced variables y back to full h (anti-symmetric)."""
    m = len(y)
    n = 2 * m
    h = np.zeros(n)
    h[:m] = y
    h[m:] = 1.0 - y[::-1]
    return h

def reduced_score(y):
    """Compute exact C5 for reduced variables."""
    h = from_reduced(y)
    return exact_score(h)

def reduced_profile(y):
    """Compute exact profile for reduced variables."""
    h = from_reduced(y)
    return compute_profile(h)

def line_search_single(y, j, n_eval=32):
    """Exact line search along single reduced coordinate y[j]."""
    m = len(y)
    t_min = -y[j]
    t_max = 1.0 - y[j]
    # Fine grid search
    ts = np.linspace(t_min, t_max, n_eval)
    scores = np.zeros(len(ts))
    for k, t in enumerate(ts):
        y_new = y.copy()
        y_new[j] = y[j] + t
        scores[k] = reduced_score(y_new)
    best_k = int(np.argmin(scores))
    # Refine around best
    if 0 < best_k < len(ts) - 1:
        lo, hi = ts[best_k - 1], ts[best_k + 1]
        for _ in range(10):
            mid1 = lo + (hi - lo) / 3
            mid2 = hi - (hi - lo) / 3
            y1 = y.copy(); y1[j] = y[j] + mid1
            y2 = y.copy(); y2[j] = y[j] + mid2
            s1 = reduced_score(y1)
            s2 = reduced_score(y2)
            if s1 < s2:
                hi = mid2
            else:
                lo = mid1
        best_t = (lo + hi) / 2
    else:
        best_t = ts[best_k]
    y_new = y.copy()
    y_new[j] = y[j] + best_t
    y_new = np.clip(y_new, 0.0, 1.0)
    return y_new, reduced_score(y_new)

def line_search_pair(y, i, j, n_eval=32):
    """Exact line search along pair move y[i]+t, y[j]-t."""
    t_min = max(-y[i], y[j] - 1.0)
    t_max = min(1.0 - y[i], y[j])
    if t_max - t_min < 1e-14:
        return y, reduced_score(y)
    ts = np.linspace(t_min, t_max, n_eval)
    scores = np.zeros(len(ts))
    for k, t in enumerate(ts):
        y_new = y.copy()
        y_new[i] = y[i] + t
        y_new[j] = y[j] - t
        scores[k] = reduced_score(y_new)
    best_k = int(np.argmin(scores))
    if 0 < best_k < len(ts) - 1:
        lo, hi = ts[best_k - 1], ts[best_k + 1]
        for _ in range(10):
            mid1 = lo + (hi - lo) / 3
            mid2 = hi - (hi - lo) / 3
            y1 = y.copy(); y1[i] = y[i] + mid1; y1[j] = y[j] - mid1
            y2 = y.copy(); y2[i] = y[i] + mid2; y2[j] = y[j] - mid2
            s1 = reduced_score(y2)  # careful with assignment
            # Actually let me redo this
            y1 = y.copy(); y1[i] = y[i] + mid1; y1[j] = y[j] - mid1
            y2 = y.copy(); y2[i] = y[i] + mid2; y2[j] = y[j] - mid2
            s1 = reduced_score(y1)
            s2 = reduced_score(y2)
            if s1 < s2:
                hi = mid2
            else:
                lo = mid1
        best_t = (lo + hi) / 2
    else:
        best_t = ts[best_k]
    y_new = y.copy()
    y_new[i] = y[i] + best_t
    y_new[j] = y[j] - best_t
    y_new = np.clip(y_new, 0.0, 1.0)
    return y_new, reduced_score(y_new)

def lse_reduced(y, beta=80.0, max_iters=150, alpha_init=0.003):
    """LSE descent in reduced coordinates."""
    m = len(y)
    y = y.copy()
    best_c = reduced_score(y)
    alpha = alpha_init
    stuck = 0
    for it in range(max_iters):
        profile = reduced_profile(y)
        c = float(np.max(profile))
        if c + 1e-14 < best_c:
            best_c = c
            stuck = 0
            alpha = min(alpha * 1.05, 0.02)
        
        # Compute reduced gradient of LSE objective
        h = from_reduced(y)
        n = len(h)
        dx = 2.0 / n
        
        # Full gradient
        G = np.zeros((2*n-1, n))
        for q in range(2*n-1):
            s = q - (n-1)
            if s == 0:
                for i in range(n):
                    G[q, i] = dx * (1.0 - 2.0*h[i])
            elif s > 0:
                for i in range(n - s):
                    G[q, i] += dx * (1.0 - h[i+s])
                    G[q, i+s] += dx * (-h[i])
            else:
                r = -s
                for i in range(n - r):
                    G[q, i+r] += dx * (1.0 - h[i])
                    G[q, i] += dx * (-h[i+r])
        
        # LSE weights
        scaled = beta * profile
        max_s = np.max(scaled)
        weights = np.exp(scaled - max_s)
        wsum = weights.sum()
        if wsum < 1e-300:
            break
        weights /= wsum
        
        # Full gradient weighted
        g_full = G.T @ weights  # shape (n,)
        
        # Reduced gradient: g_var[j] = g_full[j] - g_full[n-1-j]
        g_var = np.zeros(m)
        for j in range(m):
            g_var[j] = g_full[j] - g_full[n-1-j]
        
        gnorm = np.linalg.norm(g_var)
        if gnorm < 1e-14:
            break
        g_var /= gnorm
        
        improved = False
        for a in [alpha, alpha*0.5, alpha*0.25, alpha*0.1]:
            y_try = np.clip(y - a * g_var, 0.0, 1.0)
            s = reduced_score(y_try)
            if s < best_c - 1e-14:
                y = y_try
                best_c = s
                improved = True
                break
            elif s < c - 1e-14:
                y = y_try
                improved = True
                break
        if not improved:
            alpha *= 0.5
            stuck += 1
            if stuck > 30 or alpha < 1e-10:
                break
    return y, best_c

def active_set_reduced(y, max_iters=200, n_active=12, alpha_init=0.004):
    """Active set descent in reduced coordinates."""
    m = len(y)
    y = y.copy()
    best_c = reduced_score(y)
    alpha = alpha_init
    stuck = 0
    for it in range(max_iters):
        profile = reduced_profile(y)
        c = float(np.max(profile))
        if c + 1e-14 < best_c:
            best_c = c
            stuck = 0
            alpha = min(alpha * 1.06, 0.02)
        
        h = from_reduced(y)
        n = len(h)
        dx = 2.0 / n
        
        # Build gradient matrix efficiently
        G = np.zeros((2*n-1, n))
        for q in range(2*n-1):
            s = q - (n-1)
            if s == 0:
                G[q, :] = dx * (1.0 - 2.0*h)
            elif s > 0:
                for i in range(n - s):
                    G[q, i] += dx * (1.0 - h[i+s])
                    G[q, i+s] += dx * (-h[i])
            else:
                r = -s
                for i in range(n - r):
                    G[q, i+r] += dx * (1.0 - h[i])
                    G[q, i] += dx * (-h[i+r])
        
        # Active set
        margin = max(0.002, 3.0 * (c - float(np.median(profile))))
        active = np.where(profile >= c - margin)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(profile))])
        if len(active) > n_active:
            top_idx = np.argsort(profile[active])[-n_active:]
            active = active[top_idx]
        
        w = np.ones(len(active)) / len(active)
        g_full = G[active, :].T @ w
        g_var = np.zeros(m)
        for j in range(m):
            g_var[j] = g_full[j] - g_full[n-1-j]
        
        gnorm = np.linalg.norm(g_var)
        if gnorm < 1e-14:
            break
        g_var /= gnorm
        
        improved = False
        for a in [alpha, alpha*0.5, alpha*0.25, alpha*0.1]:
            y_try = np.clip(y - a * g_var, 0.0, 1.0)
            s = reduced_score(y_try)
            if s < best_c - 1e-14:
                y = y_try
                best_c = s
                improved = True
                break
            elif s < c - 1e-14:
                y = y_try
                improved = True
                break
        if not improved:
            alpha *= 0.55
            stuck += 1
            if stuck > 40 or alpha < 1e-10:
                break
    return y, best_c

def coordinate_descent(y, max_rounds=50, rng=None):
    """Cyclic coordinate descent with exact line search in reduced space."""
    m = len(y)
    y = y.copy()
    best_c = reduced_score(y)
    if rng is None:
        rng = np.random.RandomState(0)
    
    for rnd in range(max_rounds):
        improved = False
        order = rng.permutation(m)
        for j in order:
            y_new, s = line_search_single(y, j, n_eval=24)
            if s < best_c - 1e-12:
                y = y_new
                best_c = s
                improved = True
        if not improved and rnd > 0:
            break
    return y, best_c

def pair_descent(y, max_rounds=30, n_pairs=20, rng=None):
    """Randomized pair descent in reduced space."""
    m = len(y)
    y = y.copy()
    best_c = reduced_score(y)
    if rng is None:
        rng = np.random.RandomState(1)
    
    for rnd in range(max_rounds):
        improved = False
        for _ in range(n_pairs):
            i, j = rng.choice(m, 2, replace=False)
            y_new, s = line_search_pair(y, i, j, n_eval=24)
            if s < best_c - 1e-12:
                y = y_new
                best_c = s
                improved = True
        if not improved and rnd > 0:
            break
    return y, best_c

def slsqp_reduced(y0, maxiter=200):
    """SLSQP in reduced coordinates (no equality constraint!)."""
    m = len(y0)
    c0 = reduced_score(y0)
    
    def obj(y):
        h = from_reduced(y)
        return exact_score(h)
    
    def obj_grad(y):
        h = from_reduced(y)
        n = len(h)
        dx = 2.0 / n
        profile = compute_profile(h)
        # For max, gradient is at the argmax
        q_max = int(np.argmax(profile))
        s = q_max - (n-1)
        g_full = np.zeros(n)
        if s == 0:
            g_full = dx * (1.0 - 2.0*h)
        elif s > 0:
            for i in range(n - s):
                g_full[i] += dx * (1.0 - h[i+s])
                g_full[i+s] += dx * (-h[i])
        else:
            r = -s
            for i in range(n - r):
                g_full[i+r] += dx * (1.0 - h[i])
                g_full[i] += dx * (-h[i+r])
        g_var = np.zeros(m)
        for j in range(m):
            g_var[j] = g_full[j] - g_full[n-1-j]
        return g_var
    
    def ineq(y, t_target):
        h = from_reduced(y)
        profile = compute_profile(h)
        return t_target - profile
        # Wait, we need this as a single scalar or vector?
        # SLSQP can handle vector 'fun' for 'ineq'
        # Actually for SLSQP, 'ineq' constraints must have fun returning array
        # where each element >= 0
        # But we have 2n-1 inequalities... that might be too many.
        # Let's use a smooth max approximation instead.
    
    # Actually, let's use a different approach: minimize a smooth approximation
    # or just use the simple approach without explicit constraints on the profile
    # since SLSQP will handle the bounds and we just minimize the max
    
    # Better: use the trick of adding a variable t and minimizing t
    # subject to profile <= t
    
    n = 2 * m
    h0 = from_reduced(y0)
    dx = 2.0 / n
    
    def obj_full(x):
        return x[m]  # t
    
    def obj_grad_full(x):
        g = np.zeros(m + 1)
        g[m] = 1.0
        return g
    
    def con_fun(x):
        h = x[:m]
        h_full = from_reduced(h)
        profile = compute_profile(h_full)
        return x[m] - profile  # each element >= 0 means t >= profile[j]
    
    def con_jac(x):
        h = x[:m]
        h_full = from_reduced(h)
        nn = len(h_full)
        ddx = 2.0 / nn
        M = 2*nn - 1
        G = np.zeros((M, nn))
        for q in range(M):
            s = q - (nn-1)
            if s == 0:
                G[q, :] = ddx * (1.0 - 2.0*h_full)
            elif s > 0:
                for i in range(nn - s):
                    G[q, i] += ddx * (1.0 - h_full[i+s])
                    G[q, i+s] += ddx * (-h_full[i])
            else:
                r = -s
                for i in range(nn - r):
                    G[q, i+r] += ddx * (1.0 - h_full[i])
                    G[q, i] += ddx * (-h_full[i+r])
        # Reduce gradient
        G_var = np.zeros((M, m))
        for j in range(m):
            G_var[:, j] = G[:, j] - G[:, nn-1-j]
        J = np.zeros((M, m+1))
        J[:, :m] = -G_var
        J[:, m] = 1.0
        return J
    
    x0 = np.concatenate([y0, [c0 + 1e-4]])
    bounds = [(0.0, 1.0)] * m + [(0.0, c0 + 0.5)]
    cons = [{'type': 'ineq', 'fun': con_fun, 'jac': con_jac}]
    
    try:
        res = minimize(obj_full, x0, method='SLSQP', jac=obj_grad_full,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.x is not None and np.isfinite(res.x).all():
            y_new = np.clip(res.x[:m], 0.0, 1.0)
            c_new = reduced_score(y_new)
            if c_new <= c0 + 1e-10:
                return y_new, c_new
    except Exception:
        pass
    return None, None

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    # Simple area-preserving interpolation
    h_interp = np.zeros(new_n)
    for k in range(new_n):
        lo_k = 2.0 * k / new_n
        hi_k = 2.0 * (k+1) / new_n
        total = 0.0
        for j in range(n):
            lo_j = 2.0 * j / n
            hi_j = 2.0 * (j+1) / n
            overlap = min(hi_k, hi_j) - max(lo_k, lo_j)
            if overlap > 0:
                total += h[j] * overlap
        width = hi_k - lo_k
        h_interp[k] = total / width if width > 1e-15 else 0.5
    h_interp = np.clip(h_interp, 0.0, 1.0)
    return repair_feasible(h_interp)

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()
    rng = np.random.RandomState(seed)
    
    def time_left():
        return budget_s + t_start - time.perf_counter()
    
    # Get initial values
    parent = None
    try:
        parent = np.array(initial_h_values, dtype=np.float64)
    except (NameError, Exception):
        pass
    
    if parent is None or len(parent) < 4:
        parent = np.full(70, 0.5)
    
    parent = np.clip(parent, 0.0, 1.0)
    parent = repair_feasible(parent)
    
    # Best full vector (always maintained)
    best_h = parent.copy()
    best_c = exact_score(parent)
    
    # Try both as-is and anti-symmetric projection
    n = len(parent)
    m = n // 2
    
    # Anti-symmetric projection
    y_initial = to_reduced(parent)
    h_anti = from_reduced(y_initial)
    c_anti = exact_score(h_anti)
    
    if c_anti < best_c:
        best_h = h_anti
        best_c = c_anti
    
    # Symmetric projection
    h_sym = (parent + parent[::-1]) / 2.0
    h_sym = repair_feasible(h_sym)
    c_sym = exact_score(h_sym)
    if c_sym < best_c:
        best_h = h_sym
        best_c = c_sym
    
    # Now work primarily in reduced space
    # Start from best of anti-symmetric versions
    y_start = to_reduced(best_h)
    
    # Stage 1: Coordinate descent
    if time_left() > 100:
        y_cd, c_cd = coordinate_descent(y_start, max_rounds=30, rng=rng)
        if c_cd < best_c - 1e-12:
            best_h = from_reduced(y_cd)
            best_c = c_cd
    
    # Stage 2: Pair descent
    if time_left() > 80:
        y_pd, c_pd = pair_descent(y_start, max_rounds=20, n_pairs=15, rng=rng)
        if c_pd < best_c - 1e-12:
            best_h = from_reduced(y_pd)
            best_c = c_pd
    
    # Stage 3: LSE descent in reduced space
    if time_left() > 100:
        y_lse, c_lse = lse_reduced(y_start, beta=80, max_iters=150)
        if c_lse < best_c - 1e-12:
            best_h = from_reduced(y_lse)
            best_c = c_lse
    
    # Stage 4: Active set in reduced space
    if time_left() > 80:
        y_as, c_as = active_set_reduced(y_start, max_iters=150, n_active=10)
        if c_as < best_c - 1e-12:
            best_h = from_reduced(y_as)
            best_c = c_as
    
    # Stage 5: SLSQP in reduced space
    if time_left() > 60:
        y0 = to_reduced(best_h)
        y_sol, c_sol = slsqp_reduced(y0, maxiter=200)
        if y_sol is not None and c_sol < best_c - 1e-12:
            best_h = from_reduced(y_sol)
            best_c = c_sol
    
    # Stage 6: Resolution changes
    if time_left() > 150:
        n_best = len(best_h)
        dims_to_try = []
        for factor in [0.7, 0.8, 0.9, 1.0, 1.1, 1.2, 1.4]:
            d = int(round(n_best * factor / 2)) * 2
            d = max(20, min(d, 160))
            if d != n_best:
                dims_to_try.append(d)
        dims_to_try = sorted(set(dims_to_try), key=lambda x: abs(x - n_best))
        
        for new_n in dims_to_try[:8]:
            if time_left() < 40:
                break
            h_interp = interpolate_to_n(best_h, new_n)
            y_new = to_reduced(h_interp)
            
            # Quick coordinate descent
            y_cd, c_cd = coordinate_descent(y_new, max_rounds=15, rng=rng)
            if c_cd < best_c - 1e-12:
                best_h = from_reduced(y_cd)
                best_c = c_cd
            
            # LSE
            if time_left() > 30:
                y_lse, c_lse = lse_reduced(to_reduced(best_h), beta=60, max_iters=60)

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    # Vectorized area-preserving interpolation
    x_old = (np.arange(n) + 0.5) * 2.0 / n  # centers of old bins
    x_new = (np.arange(new_n) + 0.5) * 2.0 / new_n  # centers of new bins
    # For each new bin, find overlapping old bins
    # Old bin j covers [j*2/n, (j+1)*2/n)
    # New bin k covers [k*2/new_n, (k+1)*2/new_n)
    edges_old = np.arange(n+1) * 2.0 / n
    edges_new = np.arange(new_n+1) * 2.0 / new_n
    
    h_interp = np.zeros(new_n)
    for k in range(new_n):
        lo, hi = edges_new[k], edges_new[k+1]
        # Find old bins that overlap
        j_start = max(0, int(np.floor(lo * n / 2.0)))
        j_end = min(n, int(np.ceil(hi * n / 2.0)))
        total = 0.0
        for j in range(j_start, j_end):
            olo, ohi = edges_old[j], edges_old[j+1]
            overlap = min(hi, ohi) - max(lo, olo)
            if overlap > 0:
                total += h[j] * overlap
        h_interp[k] = total / (hi - lo)
    
    h_interp = np.clip(h_interp, 0.0, 1.0)
    return repair_feasible(h_interp)

    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def compute_profile(h):

    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx] + 1e-15)
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx] + 1e-15)
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-14:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmin(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)

    return G_var

def line_search_single(y, j, n_eval=24):
    m = len(y)
    t_min = -y[j]
    t_max = 1.0 - y[j]
    ts = np.linspace(t_min, t_max, n_eval)
    scores = np.zeros(len(ts))
    for k in range(len(ts)):
        y[k] + ts[k]
        y_new = y.copy()
        y_new[j] = y[j] + ts[k]
        scores[k] = reduced_score(y_new)
    best_k = int(np.argmin(scores))
    if 0 < best_k < len(ts) - 1:
        lo, hi = ts[best_k - 1], ts[best_k + 1]
        for _ in range(8):
            mid1 = lo + (hi - lo) / 3.0
            mid2 = hi - (hi - lo) / 3.0
            y1 = y.copy(); y1[j] = y[j] + mid1
            y2 = y.copy(); y2[j] = y[j] + mid2
            s1 = reduced_score(y1)
            s2 = reduced_score(y2)
            if s1 < s2:
                hi = mid2
            else:
                lo = mid1
        best_t = (lo + hi) / 2.0
    else:
        best_t = ts[best_k]
    y_new = y.copy()
    y_new[j] = y[j] + best_t
    y_new = np.clip(y_new, 0.0, 1.0)
    return y_new, reduced_score(y_new)

def line_search_pair(y, i, j, n_eval=24):
    t_min = max(-y[i], y[j] - 1.0)
    t_max = min(1.0 - y[i], y[j])
    if t_max - t_min < 1e-14:
        return y.copy(), reduced_score(y)
    ts = np.linspace(t_min, t_max, n_eval)
    scores = np.zeros(len(ts))
    for k in range(len(ts)):
        y_new = y.copy()
        y_new[i] = y[i] + ts[k]
        y_new[j] = y[j] - ts[k]
        scores[k] = reduced_score(y_new)
    best_k = int(np.argmin(scores))
    if 0 < best_k < len(ts) - 1:
        lo, hi = ts[best_k - 1], ts[best_k + 1]
        for _ in range(8):
            mid1 = lo + (hi - lo) / 3.0
            mid2 = hi - (hi - lo) / 3.0
            y1 = y.copy(); y1[i] = y[i] + mid1; y1[j] = y[j] - mid1
            y2 = y.copy(); y2[i] = y[i] + mid2; y2[j] = y[j] - mid2
            s1 = reduced_score(y1)
            s2 = reduced_score(y2)
            if s1 < s2:
                hi = mid2
            else:
                lo = mid1
        best_t = (lo + hi) / 2.0
    else:
        best_t = ts[best_k]
    y_new = y.copy()
    y_new[i] = y[i] + best_t
    y_new[j] = y[j] - best_t
    y_new = np.clip(y_new, 0.0, 1.0)
    return y_new, reduced_score(y_new)

def coordinate_descent(y, max_rounds=40, rng=None):
    m = len(y)
    y = y.copy()
    best_c = reduced_score(y)
    if rng is None:
        rng = np.random.RandomState(0)
    for rnd in range(max_rounds):
        improved = False
        order = rng.permutation(m)
        for j in order:
            y_new, s = line_search_single(y, j)
            if s < best_c - 1e-13:
                y = y_new
                best_c = s
                improved = True
        if not improved and rnd > 1:
            break
    return y, best_c

def pair_descent(y, max_rounds=25, n_pairs=15, rng=None):
    m = len(y)
    y = y.copy()
    best_c = reduced_score(y)
    if rng is None:
        rng = np.random.RandomState(1)
    for rnd in range(max_rounds):
        improved = False
        for _ in range(n_pairs):
            i, j = rng.choice(m, 2, replace=False)
            y_new, s = line_search_pair(y, i, j)
            if s < best_c - 1e-13:
                y = y_new
                best_c = s
                improved = True
        if not improved and rnd > 1:
            break
    return y, best_c

def lse_reduced(y, beta=80.0, max_iters=150, alpha_init=0.004):
    m = len(y)
    y = y.copy()
    best_c = reduced_score(y)
    alpha = alpha_init
    stuck = 0
    for it in range(max_iters):
        h = from_reduced(y)
        profile = reduced_profile(y)
        c = float(np.max(profile))
        if c + 1e-14 < best_c:
            best_c = c
            stuck = 0
            alpha = min(alpha * 1.04, 0.018)
        G_var = build_reduced_grad(h, m)
        scaled = beta * profile
        max_s = np.max(scaled)
        weights = np.exp(scaled - max_s)
        wsum = weights.sum()
        if wsum < 1e-300:
            break
        weights /= wsum
        g_var = G_var.T @ weights
        gnorm = np.linalg.norm(g_var)
        if gnorm < 1e-14:
            break
        g_var /= gnorm
        improved = False
        for a in [alpha, alpha*0.5, alpha*0.25, alpha*0.1]:
            y_try = np.clip(y - a * g_var, 0.0, 1.0)
            s = reduced_score(y_try)
            if s < best_c - 1e-14:
                y = y_try
                best_c = s
                improved = True
                break
            elif s < c - 1e-14:
                y = y_try
                improved = True
                break
        if not improved:
            alpha *= 0.5
            stuck += 1
            if stuck > 35 or alpha < 1e-10:
                break
    return y, best_c

def active_set_reduced(y, max_iters=200, n_active=12, alpha_init=0.004):

def slsqp_reduced(y0, maxiter=200):
    m = len(y0)
    n = 2 * m
    dx = 2.0 / n
    M = 2 * n - 1
    c0 = reduced_score(y0)

    def obj_full(x):
        return x[m]

    def obj_grad_full(x):
        g = np.zeros(m + 1)
        g[m] = 1.0
        return g

    def con_fun(x):
        h = from_reduced(x[:m])
        profile = compute_profile(h)
        return x[m] - profile

    def con_jac(x):
        h = from_reduced(x[:m])
        G = build_grad(h)
        G_var = np.zeros((M, m))
        for j in range(m):
            G_var[:, j] = G[:, j] - G[:, n-1-j]
        J = np.zeros((M, m + 1))
        J[:, :m] = -G_var
        J[:, m] = 1.0
        return J

    x0 = np.concatenate([y0, [c0 + 1e-4]])
    bounds = [(0.0, 1.0)] * m + [(0.0, c0 + 0.5)]
    cons = [{'type': 'ineq', 'fun': con_fun, 'jac': con_jac}]

    try:
        res = minimize(obj_full, x0, method='SLSQP', jac=obj_grad_full,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.x is not None and np.isfinite(res.x).all():
            y_new = np.clip(res.x[:m], 0.0, 1.0)
            c_new = reduced_score(y_new)
            if c_new <= c0 + 1e-10:
                return y_new, c_new
    except Exception:
        pass
    return None, None

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    edges_old = np.arange(n + 1) * 2.0 / n
    edges_new = np.arange(new_n + 1) * 2.0 / new_n
    h_interp = np.zeros(new_n)
    for k in range(new_n):
        lo, hi = edges_new[k], edges_new[k + 1]
        j_start = max(0, int(np.floor(lo * n / 2.0)))
        j_end = min(n, int(np.ceil(hi * n / 2.0)) + 1)
        total = 0.0
        for j in range(j_start, j_end):
            olo, ohi = edges_old[j], edges_old[j + 1]
            overlap = min(hi, ohi) - max(lo, olo)
            if overlap > 0:
                total += h[j] * overlap
        h_interp[k] = total / (hi - lo)
    h_interp = np.clip(h_interp, 0.0, 1.0)
    return repair_feasible(h_interp)

def lse_full(h, beta=80.0, max_iters=100, alpha_init=0.003):
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    alpha = alpha_init
    stuck = 0
    for it in range(max_iters):
        profile = compute_profile(h)
        c = float(np.max(profile))
        if c + 1e-14 < best_c:
            best_c = c
            stuck = 0
            alpha = min(alpha * 1.04, 0.015)
        G = build_grad(h)
        scaled = beta * profile
        max_s = np.max(scaled)
        weights = np.exp(scaled - max_s)
        wsum = weights.sum()
        if wsum < 1e-300:
            break
        weights /= wsum
        grad = G.T @ weights
        grad -= grad.mean()
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad /= gnorm
        improved = False
        for a in [alpha, alpha*0.5, alpha*0.25, alpha*0.1]:
            h_try = repair_feasible(h - a * grad)
            s = exact_score(h_try)
            if s < best_c - 1e-14:
                h = h_try
                best_c = s
                improved = True
                break
            elif s < c - 1e-14:
                h = h_try
                improved = True
                break
        if not improved:
            alpha *= 0.5
            stuck += 1
            if stuck > 30 or alpha < 1e-10:
                break
    return h, best_c

def slsqp_full(h0, maxiter=200):
    n = len(h0)
    dx = 2.0 / n
    M = 2 * n - 1
    c0 = exact_score(h0)

    def obj(x):
        return x[n]

    def obj_jac(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g

    def eq_fun(x):
        return x[:n].sum() - n / 2.0

    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:n] = 1.0
        return g

    def ineq_fun(x):
        h = np.clip(x[:n], 0, 1)
        profile = compute_profile(h)
        return x[n] - profile

    def ineq_jac(x):
        h = np.clip(x[:n], 0, 1)
        G = build_grad(h)
        J = np.zeros((M, n + 1))
        G_red = G - G * 0  # just use G
        J[:, :n] = -G
        # But we need to account for clipping... use raw gradient
        J[:, n] = 1.0
        return J

    x0 = np.concatenate([h0, [c0 + 1e-4]])
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 0.5)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.x is not None and np.isfinite(res.x).all():
            h_new = repair_feasible(res.x[:n])
            c_new = exact_score(h_new)
            if c_new <= c0 + 1e-10:
                return h_new, c_new
    except Exception:
        pass
    return None, None

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s + t_start - time.perf_counter()

    parent = None
    try:
        parent = np.array(initial_h_values, dtype=np.float64)
    except (NameError, Exception):
        pass

    if parent is None or len(parent) < 4:
        parent = np.full(70, 0.5)

    parent = np.clip(parent, 0.0, 1.0)
    parent = repair_feasible(parent)

    best_h = parent.copy()
    best_c = exact_score(parent)

    # Anti-symmetric projection
    y_initial = to_reduced(parent)
    h_anti = from_reduced(y_initial)
    c_anti = exact_score(h_anti)
    if c_anti < best_c:
        best_h = h_anti.copy()
        best_c = c_anti

    # Symmetric projection
    h_sym = repair_feasible((parent + parent[::-1]) / 2.0)
    c_sym = exact_score(h_sym)
    if c_sym < best_c:
        best_h = h_sym
        best_c = c_sym

    # Stage 1: Coordinate descent in reduced space
    if time_left() > 200:
        y_cn = len(best_h) // 2
        y0 = to_reduced(best_h)
        y_cd, c_cd = coordinate_descent(y0, max_rounds=30, rng=rng)
        if c_cd < best_c - 1e-13:
            best_h = from_reduced(y_cd)
            best_c = c_cd

    # Stage 2: Pair descent in reduced space
    if time_left() > 180:
        y0 = to_reduced(best_h)
        y_pd, c_pd = pair_descent(y0, max_rounds=20, n_pairs=12, rng=rng)
        if c_pd < best_c - 1e-13:
            best_h = from_reduced(y_pd)
            best_c = c_pd

    # Stage 3: LSE in reduced space
    if time_left() > 150:
        y0 = to_reduced(best_h)
        y_lse, c_lse = lse_reduced(y0, beta=80, max_iters=150)
        if c_lse < best_c - 1e-13:
            best_h = from_reduced(y_lse)
            best_c = c_lse

    # Stage 4: Active set in reduced space
    if time_left() > 120:
        y0 = to_reduced(best_h)
        y_as, c_as = active_set_reduced(y0, max_iters=200, n_active=10)
        if c_as < best_c - 1e-13:
            best_h = from_reduced(y_as)
            best_c = c_as

    # Stage 5: SLSQP in reduced space (no mass constraint!)
    if time_left() > 80:
        y0 = to_reduced(best_h)
        y_sol, c_sol = slsqp_reduced(y0, maxiter=250)
        if y_sol is not None and c_sol < best_c - 1e-13:
            best_h = from_reduced(y_sol)
            best_c = c_sol
            # Polish
            y_pol = to_reduced(best_h)
            y_cd2, c_cd2 = coordinate_descent(y_pol, max_rounds=15, rng=rng)
            if c_cd2 < best_c - 1e-13:
                best_h = from_reduced(y_cd2)
                best_c = c_cd2

    # Stage 6: Iterative refinement loop
    if time_left() > 200:
        y_cur = to_reduced(best_h)
        for rnd in range(6):
            if time_left() < 50:
                break
            y_cd, c_cd = coordinate_descent(y_cur, max_rounds=15, rng=rng)
            if c_cd < best_c - 1e-13:
                best_c = c_cd
            y_cur = y_cd
            y_as, c_as = active_set_reduced(y_cur, max_iters=100, n_active=12)
            if c_as < best_c - 1e-13:
                best_c = c_as
            y_cur = y_as
            y_lse, c_lse = lse_reduced(y_cur, beta=100, max_iters=80)
            if c_lse < best_c - 1e-13:
                best_c = c_lse
            y_cur = y_lse
        best_h = from_reduced(y_cur)

    # Stage 7: Resolution exploration
    if time_left() > 200:
        n_best = len(best_h)
        dims_to_try = []
        for factor in [0.6, 0.7, 0.8, 0.9, 1.05, 1.1, 1.2, 1.4, 1.6]:
            d = int(round(n_best * factor / 2)) * 2
            d = max(16, min(d, 200))
            if d != n_best:
                dims_to_try.append(d)
        dims_to_try = sorted(set(dims_to_try), key=lambda x: abs(x - n_best))

        for new_n in dims_to_try[:10]:
            if time_left() < 40:
                break
            h_interp = interpolate_to_n(best_h, new_n)
            y_new = to_reduced(h_interp)
            y_cd, c_cd = coordinate_descent(y_new, max_rounds=10, rng=rng)
            if c_cd < best_c - 1e-13:
                best_h = from_reduced(y_cd)
                best_c = c_cd
            if time_left() > 30:
                y_lse, c_lse = lse_reduced(to_reduced(best_h), beta=80, max_iters=50)
                if c_lse < best_c - 1e-13:
                    best_h = from_reduced(y_lse)
                    best_c = c_lse

    # Stage 8: Full-space polish (break anti-symmetry)
    if time_left() > 100:
        h_cur = best_h.copy()
        h_lse, c_lse = lse_full(h_cur, beta=80, max_iters=100)
        if c_lse < best_c - 1e-13:
            best_h = h_lse
            best_c = c_lse
        h_as, c_as = active_set_full(best_h, max_iters=100, n_active=10)
        if c_as < best_c - 1e-13:
            best_h = h_as
            best_c = c_as

    # Stage 9: SLSQP full space
    if time_left() > 80:
        h_sol, c_sol = slsqp_full(best_h, maxiter=200)
        if h_sol is not None and c_sol < best_c - 1e-13:
            best_h = h_sol
            best_c = c_sol

    # Stage 10: Multi-start in reduced space
    if time_left() > 120:
        n_b = len(best_h)
        if n_b % 2 == 0:
            m_b = n_b // 2
            y_base = to_reduced(best_h)
            starts = [y_base.copy()]
            for eps in [0.01, 0.02, 0.05]:
                z = rng.randn(m_b) * eps
                starts.append(np.clip(y_base + z, 0.0, 1.0))
            # Smoothing perturbation
            kernel = np.array([1.0, 2.0, 3.0, 2.0, 1.0]) / 9.0
            y_smooth = np.convolve(y_base, kernel, mode='same')
            starts.append(np.clip(y_smooth, 0.0, 1.0))

            for ys in starts[:4]:
                if time_left() < 30:
                    break
                y_cd, c_cd = coordinate_descent(ys, max_rounds=10, rng=rng)
                y_lse, c_lse = lse_reduced(y_cd, beta=100, max_iters=50)
                c_final = min(c_cd, c_lse)
                y_best = y_cd if c_cd < c_lse else y_lse
                if c_final < best_c - 1e-13:
                    best_h = from_reduced(y_best)
                    best_c = c_final

    # Stage 11: Final SLSQP reduced polish
    if time_left() > 50:
        n_f = len(best_h)
        if n_f % 2 == 0:
            y0 = to_reduced(best_h)
            y_sol, c_sol = slsqp_reduced(y0, maxiter=300)
            if y_sol is not None and c_sol < best_c - 1e-14:
                best_h = from_reduced(y_sol)
                best_c = c_sol

    # Stage 12: Final coordinate + pair polish
    if time_left() > 25:
        n_f = len(best_h)
        if n_f % 2 == 0:
            y_cur = to_reduced(best_h)
            y_cd, c_cd = coordinate_descent(y_cur, max_rounds=20, rng=rng)
            if c_cd < best_c - 1e-14:
                best_c = c_cd
            y_cur = y_cd
            y_pd, c_pd = pair_descent(y_cur, max_rounds=10, n_pairs=10, rng=rng)
            if c_pd < best_c - 1e-14:
                best_c = c_pd
            y_best = y_cd if c_cd < c_pd else y_pd
            best_h = from_reduced(y_best)

    # Final validation
    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)

    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            final_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 2:
            final_c = float(eval_result[0])
    except Exception:
        pass

    return best_h, final_c, len(best_h)
