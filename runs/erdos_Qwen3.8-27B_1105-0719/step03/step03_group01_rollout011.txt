import numpy as np
from scipy.optimize import minimize

def project_feasible(x, target_sum):
    n = len(x)
    if n == 0:
        return x
    x = np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0).astype(float).copy()
    lo, hi = -2.0, 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            mask = h < 1.0 - 1e-12
            cnt = mask.sum()
            if cnt > 0:
                h[mask] += diff / cnt
        else:
            mask = h > 1e-12
            cnt = mask.sum()
            if cnt > 0:
                h[mask] += diff / cnt
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - h.sum()
    if abs(residual) > 1e-14:
        if residual > 0:
            idx = np.argmin(1.0 - h)
            add = min(residual, 1.0 - h[idx])
            h[idx] += add
        else:
            idx = np.argmax(h)
            sub = min(-residual, h[idx])
            h[idx] -= sub
    return h


def c5_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)


def compute_corr_vector(h):
    return np.correlate(h, 1.0 - h, mode='full')


def grad_corr_k(h, k, n):
    d = k - (n - 1)
    g = np.zeros(n)
    i_start = max(0, -d)
    i_end = min(n, n - d)
    idx = np.arange(i_start, i_end)
    if d == 0:
        g[idx] = 1.0 - 2.0 * h[idx]
    else:
        g[idx] += 1.0 - h[idx + d]
        g[idx + d] -= h[idx]
    return g


def nlp_solve(h0, target_sum, n, dx, top_k=30, maxiter=60, maxfun=1500):
    corrs = compute_corr_vector(h)
    t0_score = c5_score(h0)
    active_k = np.argsort(corrs)[-top_k:]
    m = len(active_k)

    def obj(x):
        return x[0]

    def obj_grad(x):
        g = np.zeros(n + 1)
        g[0] = 1.0
        return g

    def con_fun(x):
        h = x[1:]
        t = x[0]
        vals = np.zeros(m)
        for i, k in enumerate(active_k):
            d = k - (n - 1)
            i_start = max(0, -d)
            i_end = min(n, n - d)
            idx = np.arange(i_start, i_end)
            if d == 0:
                vals[i] = t - dx * np.sum(h[idx] * (1.0 - h[idx]))
            else:
                vals[i] = t - dx * np.sum(h[idx] * (1.0 - h[idx + d]))
        return vals

    def con_jac(x):
        h = x[1:]
        J = np.zeros((m, n + 1))
        J[:, 0] = 1.0
        for i, k in enumerate(active_k):
            d = k - (n - 1)
            i_start = max(0, -d)
            i_end = min(n, n - d)
            idx = np.arange(i_start, i_end)
            if d == 0:
                J[i, 1 + idx] = -dx * (1.0 - 2.0 * h[idx])
            else:
                J[i, 1 + idx] += -dx * (1.0 - h[idx + d])
                J[i, 1 + idx + d] += -dx * h[idx]
        return J

    def eq_fun(x):
        return np.array([x[1:].sum() - target_sum])

    def eq_jac(x):
        J = np.zeros((1, n + 1))
        J[0, 1:] = 1.0
        return J

    x0 = np.zeros(n + 1)
    x0[0] = t0_score + 0.005
    x0[1:] = h0

    bounds = [(0.0, 0.55)] + [(0.0, 1.0)] * n
    cons = [
        {'type': 'ineq', 'fun': con_fun, 'jac': con_jac},
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
    ]

    try:
        res = minimize(obj, x0, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'maxfun': maxfun, 'ftol': 1e-10})
        h_new = project_feasible(res.x[1:], target_sum)
        score = c5_score(h_new)
        return h_new, score, True
    except Exception:
        return h0.copy(), c5_score(h0), False

def greedy_pair_swap(h, target_sum, n, dx, max_sweeps=200, t_end=None):
    h = h.copy()
    best_score = c5_score(h)
    best_h = h.copy()
    no_improve = 0

    for sweep in range(max_sweeps):
        if t_end is not None and time.time() > t_end:
            break
        improved = False
        corrs = compute_corr_vector(h)
        top_k = np.argsort(corrs)[-10:]
        for p in range(n):
            if t_end is not None and time.time() > t_end:
                break
            old_p = h[p]
            for q in range(p + 1, n):
                if t_end is not None and time.time() > t_end + 5:
                    break
                if h[p] >= 1.0 or h[q] <= 0.0:
                    continue
                delta = min(1.0 - h[p], h[q])
                if delta < 1e-10:
                    continue
                h[p] += delta
                h[q] -= delta
                s = c5_score(h)
                h[p] -= delta
                h[q] += delta
                if s < best_score - 1e-12:
                    h[p] += delta
                    h[q] -= delta
                    best_score = s
                    best_h = h.copy()
                    improved = True
                    break
        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0

    return best_h, best_score


def subgradient_refine(h, target_sum, n, dx, alpha_start=0.002, n_iters=500,
                       top_k=5, t_end=None):
    h = h.copy()
    best_score = c5_score(h)
    best_h = h.copy()
    alpha = alpha_start
    no_improve = 0

    for it in range(n_iters):
        if t_end is not None and time.time() > t_end:
            break
        corrs = compute_corr_vector(h)
        max_c = corrs.max()
        threshold = max_c - 0.002 * max_c
        active = np.where(corrs >= threshold)[0]
        top_active = active[np.argsort(corrs[active])[-top_k:]]

        g = np.zeros(n)
        for k in top_active:
            g += grad_corr_k(h, k, n)
        g *= dx / len(top_active)

        h_new = project_feasible(h - alpha * g, target_sum)
        s_new = c5_score(h_new)
        if s_new < best_score - 1e-14:
            h = h_new
            best_score = s_new
            best_h = h_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 100:
                alpha *= 0.9
                no_improve = 0
                if alpha < 1e-6:
                    break
        alpha *= 0.999

    return best_h, best_score
