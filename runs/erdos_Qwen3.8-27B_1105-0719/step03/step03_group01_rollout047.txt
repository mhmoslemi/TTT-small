def repair(x, target_sum):
    n = len(x)
    x = np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0)
    lo, hi = -5.0, 5.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0 - 1e-10
            if mask.any():
                h[mask] += diff / mask.sum()
        else:
            mask = h > 1e-10
            if mask.any():
                h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)


def score(h):
    dx = 2.0 / len(h)
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)


def grad_corr_k(h, k, n):
    d = k - (n - 1)
    g = np.zeros(n)
    if d >= 0:
        i_max = n - d
        if i_max > 0:
            g[:i_max] += (1.0 - h[d:])
            g[d:] += (-h[:i_max])
    else:
        ad = -d
        g[ad:] += (1.0 - h[:n - ad])
        g[:n - ad] += (-h[ad:])
    return g


def lp_step(h, dx, target_sum, alpha, use_all=True):

    c_obj[-1] = 1.0
    A_ub = np.zeros((m, n + 1))

    A_ub[:, -1] = -1.0
    b_ub = np.zeros(m)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    bounds = [(-alpha * max(h[i], 0.01), alpha * max(1.0 - h[i], 0.01))] * n + [(-np.inf, np.inf)]
    
    try:

def slsqp_polish(h0, n, target_sum, n_active=25, max_iter=60):
    dx = 2.0 / n
    m_total = 2 * n - 1
    corrs0 = np.correlate(h0, 1.0 - h0, mode='full')
    cur_max = corrs0.max() * dx
    active = np.argsort(corrs0)[-n_active:]
    
    def obj(v):
        return v[0]
    
    def obj_grad(v):
        g = np.zeros(n + 1)
        g[0] = 1.0
        return g
    
    def constr(v):
        h = v[1:]
        h = np.clip(h, 0, 1)
        c = np.correlate(h, 1.0 - h, mode='full')
        return v[0] - c[active] * dx
    
    def constr_jac(v):
        h = v[1:]
        h = np.clip(h, 0, 1)
        J = np.zeros((len(active), n + 1))
        J[:, 0] = 1.0
        for i, k in enumerate(active):
            J[i, 1:] = -dx * grad_corr_k(h, k, n)
        return J
    
    def eq_con(v):
        return v[1:].sum() - target_sum
    
    bounds = [(cur_max - 0.05, 0.6)] + [(0.0, 1.0)] * n
    x0 = np.zeros(n + 1)
    x0[0] = cur_max + 1e-4
    x0[1:] = h0
    
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_grad,
                       constraints=[{'type': 'ineq', 'fun': constr, 'jac': constr_jac},
                                    {'type': 'eq', 'fun': eq_con}],
                       bounds=bounds,
                       options={'maxiter': max_iter, 'ftol': 1e-12})
        if res.x is not None:

        for alpha_f in [0.1, 0.05, 0.02]:
            if dt_final < 5:
                break
            h_opt, c5_opt = lp_optimize_loop(best_h, dx_b, ts_b, time.time(),
                                              min(dt_final * 0.4, 60) + (time.time() - t0),
                                              alpha_start=alpha_f, n_active_frac=0.0, min_alpha=0.0005, max_no_improve=100)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
            dt_final = budget_s - 3 - (time.time() - t0)
        
        # Final SLSQP
        if dt_final > 10:
            h_res, c5_res = slsqp_polish(best_h, n_b, ts_b, n_active=30, max_iter=80)
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = h_res.copy()
    
    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    n_final = len(best_h)
    ts_final = n_final / 2.0
    best_h = repair(best_h, ts_final)
    best_c5 = score(best_h)
    
    return best_h, best_c5, n_final
