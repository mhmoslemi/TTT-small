def random_transfer(h, rng, n_pairs=1):
    """Random coordinate transfer moves."""
    n = len(h)
    h_new = h.copy()
    for _ in range(n_pairs):
        i, j = rng.choice(n, size=2, replace=False)
        max_delta = min(h[j], 1.0 - h[i])
        if max_delta <= 1e-12:
            # Try reverse
            max_delta = min(h[i], 1.0 - h[j])
            if max_delta > 1e-12:
                delta = rng.uniform(0.01, max_delta)
                h_new[j] += delta
                h_new[i] -= delta
            continue
        delta = rng.uniform(0.01, max_delta)
        h_new[i] += delta
        h_new[j] -= delta
    return repair_feasible(h_new)


def block_swap(h, rng):
    """Swap two blocks of equal length."""
    n = len(h)
    if n < 4:
        return h.copy()
    blen = rng.choice([1, 2, max(1, n // 8), max(1, n // 4)])
    blen = min(blen, n // 2)
    i = rng.randint(0, n - blen)
    # Find a non-overlapping position for j
    valid = list(range(0, n - blen))
    valid = [v for v in valid if abs(v - i) >= blen]
    if not valid:
        return h.copy()
    j = rng.choice(valid)
    h_new = h.copy()
    h_new[i:i+blen], h_new[j:j+blen] = h[j:j+blen].copy(), h[i:i+blen].copy()
    return repair_feasible(h_new)

import numpy as np
from scipy.optimize import minimize
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))


def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx


def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx])
            h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-14:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] = np.clip(h[idx] + diff2, 0.0, 1.0)
    return h


def interp_h(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    fx = np.concatenate([[0.0], xs, [2.0]])
    fh = np.concatenate([[h[0]], h, [h[-1]]])
    hn = np.interp(xt, fx, fh)
    return repair_feasible(np.clip(hn, 0, 1))


def profile_grad_smoothed(h, t):
    """Gradient of the log-sum-exp smoothed max of the profile."""
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    center = n - 1
    profile = np.correlate(h, 1.0 - h, mode="full") * dx
    pmax = np.max(profile)
    exps = np.exp(t * (profile - pmax))
    s = exps.sum()
    if s < 1e-300:
        return np.zeros(n)
    weights = exps / s

    grad = np.zeros(n)
    for k in range(M):
        if weights[k] < 1e-12:
            continue
        w = weights[k]
        d = k - center
        if d == 0:
            grad += w * dx * (1.0 - 2.0 * h)
        elif d > 0:
            grad[d:n] += w * dx * (1.0 - h[0:n - d])
            grad[0:n - d] += -w * dx * h[d:n]
        else:
            dd = -d
            grad[0:n - dd] += w * dx * (1.0 - h[dd:n])
            grad[dd:n] += -w * dx * h[0:n - dd]
    return grad


def smooth_max_value(h, t):
    n = len(h)
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode="full") * dx
    pmax = np.max(profile)
    return pmax + np.log(np.sum(np.exp(t * (profile - pmax)))) / t


def projected_step(h, t, step_sizes):
    """Take a projected gradient step on the smoothed max."""
    grad = profile_grad_smoothed(h, t)
    grad -= grad.mean()
    gn = np.linalg.norm(grad)
    if gn < 1e-15:
        return None, None
    grad /= gn
    curr = exact_score(h)
    for s in step_sizes:
        h_try = repair_feasible(h - s * grad)
        sc = exact_score(h_try)
        if sc < curr - 1e-14:
            return h_try, sc
    return None, None


def random_transfer(h, rng, n_pairs=2):
    n = len(h)
    h_new = h.copy()
    for _ in range(n_pairs):
        i, j = rng.choice(n, size=2, replace=False)
        md = min(h[j], 1.0 - h[i])
        if md > 1e-12:
            delta = rng.uniform(0.005, md)
            h_new[i] += delta
            h_new[j] -= delta
        else:
            md = min(h[i], 1.0 - h[j])
            if md > 1e-12:
                delta = rng.uniform(0.005, md)
                h_new[j] += delta
                h_new[i] -= delta
    return repair_feasible(h_new)


def block_transfer(h, rng):
    """Transfer mass between two blocks."""
    n = len(h)
    if n < 6:
        return h.copy()
    blen = min(rng.choice([1, 2, 3, max(1, n // 10)]), n // 2)
    i = rng.randint(0, n - blen)
    valid_js = [j for j in range(n - blen) if abs(j - i) >= blen]
    if not valid_js:
        return h.copy()
    j = int(rng.choice(valid_js))
    h_new = h.copy()
    # Transfer half the mass from block j to block i
    for k in range(blen):
        amt = (h_new[j + k] - h_new[i + k]) * 0.3
        if amt > 0:
            dt = min(amt, h_new[j + k], 1.0 - h_new[i + k])
            h_new[i + k] += dt
            h_new[j + k] -= dt
        else:
            dt = min(-amt, h_new[i + k], 1.0 - h_new[j + k])
            h_new[j + k] += dt
            h_new[i + k] -= dt
    return repair_feasible(h_new)


def active_set_grad_step(h, profile, beam, step_sizes):
    """Gradient step targeting active (near-max) constraints."""
    n = len(h)
    dx = 2.0 / n
    center = n - 1
    c_max = np.max(profile)
    active = np.where(profile >= c_max - beam)[0]
    if len(active) > 20:
        active = active[np.argsort(profile[active])[-20:]]
    if len(active) == 0:
        active = np.array([np.argmax(profile)])

    grad = np.zeros(n)
    for k in active:
        d = k - center
        if d == 0:
            grad += dx * (1.0 - 2.0 * h)
        elif d > 0:
            grad[d:n] += dx * (1.0 - h[0:n - d])
            grad[0:n - d] += -dx * h[d:n]
        else:
            dd = -d
            grad[0:n - dd] += dx * (1.0 - h[dd:n])
            grad[dd:n] += -dx * h[0:n - dd]

    grad /= len(active)
    grad -= grad.mean()
    gn = np.linalg.norm(grad)
    if gn < 1e-15:
        return None, None
    grad /= gn

    curr = c_max
    for s in step_sizes:
        h_try = repair_feasible(h - s * grad)
        sc = exact_score(h_try)
        if sc < curr - 1e-14:
            return h_try, sc
    return None, None


def local_optimize(h0, budget_fn, rng, max_iters=3000):
    """Main local optimization: continuation + gradient + random moves."""
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c = exact_score(h)

    # Continuation: start with moderate smoothing, increase
    t_schedule = [20, 40, 80, 160, 320, 640]
    temp = 0.003

    it = 0
    for t in t_schedule:
        if budget_fn() < 20:
            break
        # Do several steps at this t value
        steps_per_t = max(50, 2000 // len(t_schedule))
        for _ in range(steps_per_t):
            if budget_fn() < 15:
                break
            if it % 300 == 0 and it > 0:
                temp *= 0.6
            it += 1

            profile = compute_profile(h)
            c_max = np.max(profile)
            if c_max < best_c - 1e-14:
                best_c = c_max
                best_h = h.copy()

            # Choose move type
            r = rng.random()
            if r < 0.45:
                # Smoothed gradient step
                hg, cg = projected_step(
                    h, t,
                    [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
                )
                if hg is not None:
                    h = hg
                    continue
            elif r < 0.7:
                # Active-set gradient
                beam = max(0.001, 0.005 * (c_max - np.median(profile)))
                hg, cg = active_set_grad_step(
                    h, profile, beam,
                    [0.01, 0.005, 0.002, 0.001, 0.0005]
                )
                if hg is not None:
                    h = hg
                    continue
            elif r < 0.85:
                # Random transfer
                h_try = random_transfer(h, rng, n_pairs=rng.choice([1, 2, 3]))
                c_try = exact_score(h_try)
                if c_try < c_max or rng.random() < np.exp(-(c_try - c_max) / max(temp, 1e-8)):
                    h = h_try
                    continue
            else:
                # Block transfer
                h_try = block_transfer(h, rng)
                c_try = exact_score(h_try)
                if c_try < c_max or rng.random() < np.exp(-(c_try - c_max) / max(temp, 1e-8)):
                    h = h_try
                    continue

    # Final fine-tuning with very small steps
    for _ in range(500):
        if budget_fn() < 10:
            break
        profile = compute_profile(h)
        c_max = np.max(profile)
        if c_max < best_c - 1e-14:
            best_c = c_max
            best_h = h.copy()

        beam = max(0.0005, 0.002 * (c_max - np.median(profile)))
        hg, cg = active_set_grad_step(
            h, profile, beam,
            [0.002, 0.001, 0.0005, 0.0002, 0.0001]
        )
        if hg is not None:
            h = hg
        else:
            h_try = random_transfer(h, rng, n_pairs=1)
            c_try = exact_score(h_try)
            if c_try < c_max:
                h = h_try

    return best_h, best_c


def _slsqp_profile(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx


def _slsqp_ineq_fun(x):
    n = len(x) - 1
    h = x[:-1]
    return x[-1] - _slsqp_profile(h, n)


def _slsqp_ineq_jac(x):
    n = len(x) - 1
    h = x[:-1]
    dx = 2.0 / n
    M = 2 * n - 1
    center = n - 1
    J = np.zeros((M, n + 1))
    for k in range(M):
        d = k - center
        if d == 0:
            J[k, :n] = -dx * (1.0 - 2.0 * h)
        elif d > 0:
            J[k, d:n] += -dx * (1.0 - h[0:n - d])
            J[k, 0:n - d] += dx * h[d:n]
        else:
            dd = -d
            J[k, 0:n - dd] += -dx * (1.0 - h[dd:n])
            J[k, dd:n] += dx * h[0:n - dd]
        J[k, n] = 1.0
    return J


def _slsqp_eq_fun(x):
    n = len(x) - 1
    return x[:-1].sum() - n / 2.0


def _slsqp_eq_jac(x):
    n = len(x) - 1
    g = np.zeros(n + 1)
    g[:-1] = 1.0
    return g


def _slsqp_obj(x):
    return x[-1]


def _slsqp_obj_jac(x):
    g = np.zeros(len(x))
    g[-1] = 1.0
    return g


def run_slsqp(h0, maxiter=300, ftol=1e-14):
    n = len(h0)
    t_init = exact_score(h0) + 1e-3
    x0 = np.concatenate([h0, [t_init]])
    bounds = [(0.0, 1.0)] * n + [(0.0, t_init + 2.0)]
    cons = [
        {'type': 'eq', 'fun': _slsqp_eq_fun, 'jac': _slsqp_eq_jac},
        {'type': 'ineq', 'fun': _slsqp_ineq_fun, 'jac': _slsqp_ineq_jac}
    ]
    try:
        res = minimize(_slsqp_obj, x0, method='SLSQP', jac=_slsqp_obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        if res.x is not None and np.isfinite(res.x).all():
            hs = repair_feasible(res.x[:-1])
            if np.isfinite(exact_score(hs)):
                return hs
    except Exception:
        pass
    return None


def make_sinusoidal(n, freq, amp):
    x = np.arange(n)
    h = 0.5 + amp * np.sin(2.0 * np.pi * freq * x / n)
    return repair_feasible(h)


def make_block_pattern(n, num_blocks, duty):
    h = np.zeros(n)
    block_size = n // num_blocks
    for b in range(num_blocks):
        start = b * block_size
        length = int(duty * block_size)
        h[start:start + length] = min(1.0, 0.5 / duty)
    return repair_feasible(h)


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()
    rng = np.random.RandomState(seed)

    def tl():
        return budget_s + t_start - time.perf_counter()

    # Load parent
    try:
        parent = np.array(initial_h_values, dtype=np.float64)
    except (NameError, Exception):
        parent = np.full(70, 0.5)

    if len(parent) < 4:
        parent = np.full(70, 0.5)

    parent = repair_feasible(np.clip(parent, 0.0, 1.0))
    n0 = len(parent)

    best_h = parent.copy()
    best_c = exact_score(parent)

    # Symmetrize
    h_sym = repair_feasible((parent + parent[::-1]) / 2.0)
    c_sym = exact_score(h_sym)
    if c_sym < best_c:
        best_h, best_c = h_sym, c_sym

    # Phase 1: Local optimization from parent
    if tl() > 250:
        bh, bc = local_optimize(best_h, tl, rng, max_iters=4000)
        if bc < best_c:
            best_h, best_c = bh, bc

    # Phase 2: Try alternative initializations at same resolution
    if tl() > 200:
        alternatives = []
        # Sinusoidal patterns
        for freq in [1, 2, 3, 4, 5, 7]:
            for amp in [0.15, 0.25, 0.35, 0.45]:
                ha = make_sinusoidal(n0, freq, amp)
                ca = exact_score(ha)
                alternatives.append((ca, ha))
        # Block patterns
        for nb in [3, 4, 5, 6, 8]:
            for duty in [0.3, 0.4, 0.5, 0.6]:
                hp = make_block_pattern(n0, nb, duty)
                cp_ = exact_score(hp)
                alternatives.append((cp_, hp))

        alternatives.sort(key=lambda x: x[0])
        # Try the best few alternative initializations
        for score, halt in alternatives[:5]:
            if tl() < 80:
                break
            if score < best_c + 0.08:
                bh, bc = local_optimize(halt, tl, rng, max_iters=1500)
                if bc < best_c:
                    best_h, best_c = bh, bc

    # Phase 3: Multi-resolution exploration
    if tl() > 180:
        cur_n = len(best_h)
        dims = []
        for d in range(24, 220, 2):
            if d != cur_n and abs(d - cur_n) <= 50:
                dims.append(d)
        dims.sort(key=lambda d: abs(d - cur_n))

        for nd in dims[:6]:
            if tl() < 80:
                break
            hi = interp_h(best_h, nd)
            ci = exact_score(hi)
            if ci < best_c + 0.04:
                bh, bc = local_optimize(hi, tl, rng, max_iters=2000)
                if bc < best_c:
                    best_h, best_c = bh, bc

    # Phase 4: SLSQP polish
    if tl() > 100:
        hs = run_slsqp(best_h, maxiter=400, ftol=1e-14)
        if hs is not None:
            cs = exact_score(hs)
            if cs < best_c:
                best_h, best_c = hs, cs

    # Phase 5: Another round of local optimization after SLSQP
    if tl() > 80:
        bh, bc = local_optimize(best_h, tl, rng, max_iters=3000)
        if bc < best_c:
            best_h, best_c = bh, bc

    # Phase 6: Try shifted variants
    if tl() > 60:
        n_f = len(best_h)
        for shift in [1, -1, 2, -2, 3]:
            if tl() < 30:
                break
            hs = repair_feasible(np.roll(best_h, shift))
            cs = exact_score(hs)
            if cs < best_c + 0.005:
                bh, bc = local_optimize(hs, tl, rng, max_iters=800)
                if bc < best_c:
                    best_h, best_c = bh, bc

    # Phase 7: Final SLSQP
    if tl() > 50:
        hs = run_slsqp(best_h, maxiter=500, ftol=1e-15)
        if hs is not None:
            cs = exact_score(hs)
            if cs < best_c:
                best_h, best_c = hs, cs

    # Phase 8: Ultra-fine local steps
    if tl() > 30:
        n_f = len(best_h)
        h_f = best_h.copy()
        c_f = best_c
        for _ in range(300):
            if tl() < 10:
                break
            profile = compute_profile(h_f)
            cm = np.max(profile)
            if cm < c_f:
                c_f = cm
            beam = max(0.0003, 0.001 * (cm - np.median(profile)))
            hg, cg = active_set_grad_step(
                h_f, profile, beam,
                [0.001, 0.0005, 0.0002, 0.0001, 5e-5, 2e-5]
            )
            if hg is not None:
                h_f = hg
            else:
                h_try = random_transfer(h_f, rng, n_pairs=1)
                c_try = exact_score(h_try)
                if c_try < cm - 1e-14:
                    h_f = h_try
                else:
                    break
        if c_f < best_c:
            best_h, best_c = h_f, c_f

    # Final validation
    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)

    try:
        er = evaluate_erdos_solution(best_h)
        if isinstance(er, (float, int)):
            final_c = min(final_c, float(er))
        elif isinstance(er, tuple) and len(er) >= 2:
            final_c = min(final_c, float(er[0]))
    except Exception:
        pass

    return best_h, final_c, len(best_h)
