import numpy as np
from scipy.optimize import minimize
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))

def profile_full(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def repair_feasible(h):
    h = np.clip(h.astype(np.float64), 0.0, 1.0)
    target = len(h) / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = int(np.argmax(1.0 - h))
        add = min(diff, 1.0 - h[idx])
        h[idx] += add
        diff -= add
        if diff > 1e-14:
            order = np.argsort(1.0 - h)
            for i in order:
                add = min(diff, 1.0 - h[i])
                if add <= 0: break
                h[i] += add
                diff -= add
                if diff <= 1e-14: break
    else:
        idx = int(np.argmin(h))
        rem = min(-diff, h[idx])
        h[idx] -= rem
        diff += rem
        if diff < -1e-14:
            order = np.argsort(h)
            for i in order:
                rem = min(-diff, h[i])
                if rem <= 0: break
                h[i] -= rem
                diff += rem
                if diff >= -1e-14: break
    return h

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = np.linspace(0, 2, n + 1)
    xt = np.linspace(0, 2, new_n + 1)
    h_new = np.zeros(new_n)
    for k in range(new_n):
        total_mass = 0.0
        total_width = 0.0
        for j in range(n):
            overlap = min(xt[k+1], xs[j+1]) - max(xt[k], xs[j])
            if overlap > 0:
                total_mass += h[j] * overlap
                total_width += overlap
        h_new[k] = total_mass / total_width if total_width > 1e-15 else 0.5
    return repair_feasible(np.clip(h_new, 0, 1))

def pair_line_search(h, i, j, n_grid=41):
    a, b = h[i], h[j]
    t_lo = max(-a, b - 1.0)
    t_hi = min(1.0 - a, b)
    if t_hi - t_lo < 1e-12:
        return h.copy(), exact_score(h)
    
    best_t = 0.0
    best_s = exact_score(h)
    
    ts = np.linspace(t_lo, t_hi, n_grid)
    for t in ts:
        h[i] = a + t
        h[j] = b - t
        s = exact_score(h)
        if s < best_s:
            best_s = s
            best_t = t

        span = t_hi - t_lo

        lo = ts[max(0, best_idx - 1)] if best_idx > 0 else t_lo
        hi = ts[min(n_grid-1, best_idx + 1)] if best_idx < n_grid - 1 else t_hi
        ts2 = np.linspace(lo, hi, 21)
        for t in ts2:

            h[j] = b - t

            if s < best_s:

    h[j] = b - best_t

def compute_descent_scores(h, n_active=10):
    """Compute per-coordinate descent priority scores."""
    n = len(h)
    prof = profile_full(h)
    c = prof.max()
    # Weight by proximity to max
    margin = max(0.005, 5.0 * (c - np.median(prof)))
    weights = np.exp(-30.0 * (c - prof) / margin)  # smooth max weights
    wsum = weights.sum()
    if wsum < 1e-30:
        weights = np.zeros_like(prof)
        weights[np.argmax(prof)] = 1.0
    else:
        weights /= wsum
    
    # Gradient for each coordinate
    scores = np.zeros(n)
    dx = 2.0 / n
    for m in range(len(prof)):
        d = m - (n - 1)
        w = weights[m]
        if w < 1e-15:
            continue
        if d == 0:
            scores += w * dx * (1.0 - 2.0 * h)
        elif d > 0:
            # P_d = sum_{k=0}^{n-1-d} h[k+d]*(1-h[k]) * dx
            # dP_d/dh[k] for k in [0, n-1]:
            #   if k >= d: contributes (1-h[k-d]) * dx
            #   if k <= n-1-d: contributes -h[k+d] * dx
            idx_first = np.arange(d, n)  # k where k>=d
            idx_second = np.arange(0, n - d)  # k where k <= n-1-d
            scores[idx_first] += w * dx * (1.0 - h[idx_first - d])
            scores[idx_second] += w * dx * (-h[idx_second + d])
        else:
            dp = -d
            # P_d = sum_{k=0}^{n-1-dp} h[k]*(1-h[k+dp]) * dx
            idx_first = np.arange(0, n - dp)
            idx_second = np.arange(dp, n)
            scores[idx_first] += w * dx * (1.0 - h[idx_first + dp])
            scores[idx_second] += w * dx * (-h[idx_second - dp])
    
    return scores

def pair_descent_sweep(h, max_rounds=200, n_pairs_per_round=200, n_grid=33):
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    no_improve = 0
    
    for rnd in range(max_rounds):
        scores = compute_descent_scores(h, n_active=15)
        # We want to increase h where scores < 0 (decreasing the max)
        # and decrease h where scores > 0
        # Actually, to decrease the max profile, we want to move in the direction -gradient
        # So increase h[k] where score[k] < 0, decrease where score[k] > 0
        
        inc_idx = np.where(scores < -1e-10)[0]
        dec_idx = np.where(scores > 1e-10)[0]
        
        if len(inc_idx) == 0 or len(dec_idx) == 0:
            no_improve += 1
            if no_improve > 10:
                break
            # Try all pairs
            inc_idx = np.where(scores < scores.mean())[0]
            dec_idx = np.where(scores > scores.mean())[0]
            if len(inc_idx) == 0 or len(dec_idx) == 0:
                break
        
        # Sort by priority
        inc_idx = inc_idx[np.argsort(scores[inc_idx])]  # most negative first
        dec_idx = dec_idx[np.argsort(-scores[dec_idx])]  # most positive first
        
        improved = False
        n_improve = 0
        
        # Try top pairs
        ni = min(len(inc_idx), 15)
        nd = min(len(dec_idx), 15)
        
        for ii in range(ni):
            for jj in range(nd):
                i, j = inc_idx[ii], dec_idx[jj]
                if i == j:
                    continue
                a, b = h[i], h[j]
                t_lo = max(-a, b - 1.0)
                t_hi = min(1.0 - a, b)
                if t_hi - t_lo < 1e-12:
                    continue
                
                ts = np.linspace(t_lo, t_hi, n_grid)
                best_t = 0.0
                best_s = exact_score(h)
                
                for t in ts:
                    h[i] = a + t
                    h[j] = b - t
                    s = exact_score(h)
                    if s < best_s:
                        best_s = s
                        best_t = t
                
                if best_t != 0.0 and best_s < best_c - 1e-13:
                    h[i] = a + best_t
                    h[j] = b - best_t
                    best_c = best_s
                    improved = True
                    n_improve += 1
                    if n_improve >= n_pairs_per_round:
                        break
                else:
                    h[i] = a
                    h[j] = b
            
            if n_improve >= n_pairs_per_round:
                break
        
        if not improved:
            no_improve += 1
            if no_improve > 5:
                break
        else:
            no_improve = 0
    
    return h, best_c

def lse_descent(h, beta=80.0, max_iters=200, alpha_init=0.004):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best_c = exact_score(h)
    alpha = alpha_init
    stuck = 0
    for it in range(max_iters):
        prof = profile_full(h)
        scaled = beta * prof
        max_s = np.max(scaled)
        weights = np.exp(scaled - max_s)
        wsum = weights.sum()
        if wsum < 1e-300:
            break
        weights /= wsum
        
        # Gradient
        grad = np.zeros(n)
        for m in range(len(prof)):
            d = m - (n - 1)
            w = weights[m] * dx
            if w < 1e-18:
                continue
            if d == 0:
                grad += w * (1.0 - 2.0 * h)
            elif d > 0:
                idx1 = np.arange(d, n)
                idx2 = np.arange(0, n - d)
                grad[idx1] += w * (1.0 - h[idx1 - d])
                grad[idx2] += w * (-h[idx2 + d])
            else:
                dp = -d
                idx1 = np.arange(0, n - dp)
                idx2 = np.arange(dp, n)
                grad[idx1] += w * (1.0 - h[idx1 + dp])
                grad[idx2] += w * (-h[idx2 - dp])
        
        grad -= grad.mean()
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad /= gnorm
        
        improved = False
        for a in [alpha, alpha*0.5, alpha*0.25, alpha*0.1]:
            h_try = repair_feasible(h - a * grad)
            s = exact_score(h_try)
            if s < best_c - 1e-13:
                h = h_try
                best_c = s
                improved = True
                alpha = min(alpha * 1.1, 0.02)
                stuck = 0
                break
        if not improved:
            alpha *= 0.5
            stuck += 1
            if stuck > 30 or alpha < 1e-9:
                break
    return h, best_c

                dp = -d
                i1 = np.arange(0, n - dp)
                i2 = np.arange(dp, n)
                grad[i1] += w * (1.0 - h[i1 + dp])
                grad[i2] += w * (-h[i2 - dp])
        grad -= grad.mean()
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        grad /= gn
        improved = False
        for a in [alpha, alpha * 0.5, alpha * 0.25, alpha * 0.1]:
            h_try = repair_feasible(h - a * grad)
            s = exact_score(h_try)
            if s < best_c - 1e-13:
                h = h_try
                best_c = s
                improved = True
                alpha = min(alpha * 1.08, 0.02)
                stuck = 0
                break
        if not improved:
            alpha *= 0.5
            stuck += 1
            if stuck > 30 or alpha < 1e-9:
                break
    return h, best_c

def active_set_descent(h, max_iters=200, n_active=12, alpha_init=0.004):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best_c = exact_score(h)
    alpha = alpha_init
    stuck = 0
    for _ in range(max_iters):
        prof = profile_full(h)
        c = float(np.max(prof))
        if c + 1e-14 < best_c:
            best_c = c
            stuck = 0
            alpha = min(alpha * 1.06, 0.02)
        margin = max(0.002, 4.0 * (c - float(np.median(prof))))
        active = np.where(prof >= c - margin)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(prof))])
        if len(active) > n_active:
            ti = np.argsort(prof[active])[-n_active:]
            active = active[ti]
        grad = np.zeros(n)
        for m in active:
            d = m - (n - 1)
            w = dx / len(active)
            if d == 0:
                grad += w * (1.0 - 2.0 * h)
            elif d > 0:
                i1 = np.arange(d, n)
                i2 = np.arange(0, n - d)
                grad[i1] += w * (1.0 - h[i1 - d])
                grad[i2] += w * (-h[i2 + d])
            else:
                dp = -d
                i1 = np.arange(0, n - dp)
                i2 = np.arange(dp, n)
                grad[i1] += w * (1.0 - h[i1 + dp])
                grad[i2] += w * (-h[i2 - dp])
        grad -= grad.mean()
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        grad /= gn
        improved = False
        for a in [alpha, alpha * 0.5, alpha * 0.25, alpha * 0.1]:
            h_try = repair_feasible(h - a * grad)
            s = exact_score(h_try)
            if s < best_c - 1e-13:
                h = h_try
                best_c = s
                improved = True
                break
        if not improved:
            alpha *= 0.55
            stuck += 1
            if stuck > 40 or alpha < 1e-9:
                break
    return h, best_c

def slsqp_solve(h0, maxiter=150):
    n = len(h0)
    dx = 2.0 / n
    M = 2 * n - 1
    c0 = exact_score(h0)

    def obj(x):
        return x[-1]

    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g

    def eq_fun(x):
        return x[:-1].sum() - n / 2.0

    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g

    def ineq_fun(x):
        hh = np.clip(x[:-1], 0, 1)
        return x[-1] - np.correlate(hh, 1 - hh, mode="full") * dx

    def ineq_jac(x):
        hh = np.clip(x[:-1], 0, 1)
        J = np.zeros((M, n + 1))
        for m in range(M):
            d = m - (n - 1)
            if d == 0:
                J[m, :n] = dx * (1.0 - 2.0 * hh)
            elif d > 0:
                i1 = np.arange(d, n)
                i2 = np.arange(0, n - d)
                J[m, i1] += dx * (1.0 - hh[i1 - d])
                J[m, i2] += dx * (-hh[i2 + d])
            else:
                dp = -d
                i1 = np.arange(0, n - dp)
                i2 = np.arange(dp, n)
                J[m, i1] += dx * (1.0 - hh[i1 + dp])
                J[m, i2] += dx * (-hh[i2 - dp])
            J[m, -1] = 1.0
        return J

    x0 = np.concatenate([h0, [c0 + 1e-3]])
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 0.5)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None and np.isfinite(res.x).all():
            h_new = repair_feasible(res.x[:-1])
            c_new = exact_score(h_new)
            if c_new <= c0 + 1e-9:
                return h_new, c_new
    except Exception:
        pass
    return None, None

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s + t_start - time.perf_counter()

    # Get parent
    parent = None
    try:
        parent = np.array(initial_h_values, dtype=np.float64)
    except NameError:
        pass
    except Exception:
        pass

    if parent is None or len(parent) < 4:
        parent = np.full(70, 0.5)

    parent = repair_feasible(parent)
    best_h = parent.copy()
    best_c = exact_score(parent)
    n_base = len(best_h)

    # Stage 1: Initial gradient polish
    if time_left() > 950:
        h_new, c_new = lse_descent(best_h, beta=60, max_iters=60)
        if c_new < best_c - 1e-13:
            best_h, best_c = h_new, c_new
        h_new, c_new = active_set_descent(best_h, max_iters=60, n_active=10)
        if c_new < best_c - 1e-13:
            best_h, best_c = h_new, c_new

    # Stage 2: Multi-resolution exploration
    if time_left() > 850:
        for factor in [0.8, 0.9, 1.0, 1.1, 1.2]:
            if time_left() < 780:
                break
            new_n = int(round(n_base * factor / 2)) * 2
            new_n = max(30, min(new_n, 130))
            if new_n == n_base:
                continue
            h_interp = interpolate_to_n(best_h, new_n)
            h_new, c_new = lse_descent(h_interp, beta=80, max_iters=50)
            h_new, c_new = pair_descent_sweep(h_new, max_rounds=30, n_pairs=80, n_grid=31)
            h_new, c_new = active_set_descent(h_new, max_iters=40, n_active=10)
            if c_new < best_c - 1e-12:
                # Interpolate back to base resolution
                best_h = interpolate_to_n(h_new, n_base)
                best_c = exact_score(best_h)
            elif c_new < best_c + 1e-6:
                # Keep at this resolution for further work
                h_alt = interpolate_to_n(h_new, n_base)
                hc = exact_score(h_alt)
                if hc < best_c - 1e-12:
                    best_h, best_c = h_alt, hc

    # Stage 3: Main loop - alternate pair descent and gradient polish
    n = len(best_h)
    cycle = 0
    while time_left() > 180:
        cycle += 1
        # Pair descent
        if time_left() > 160:
            max_r = 40 if cycle % 3 == 0 else 25
            h_new, c_new = pair_descent_sweep(best_h, max_rounds=max_r, n_pairs=120, n_grid=35)
            if c_new < best_c - 1e-13:
                best_h, best_c = h_new, c_new
        # Gradient polish
        if time_left() > 140:
            h_new, c_new = lse_descent(best_h, beta=80 + 20 * cycle, max_iters=40)
            if c_new < best_c - 1e-13:
                best_h, best_c = h_new, c_new
        if time_left() > 120:
            h_new, c_new = active_set_descent(best_h, max_iters=40, n_active=10 + (cycle % 5) * 2)
            if c_new < best_c - 1e-13:
                best_h, best_c = h_new, c_new
        # Random pair exploration every few cycles
        if cycle % 4 == 0 and time_left() > 100:
            h_new, c_new = random_pair_descent(best_h, n_iter=150, n_grid=21, rng=rng)
            if c_new < best_c - 1e-13:
                best_h, best_c = h_new, c_new

    # Stage 4: Random pair exploration (longer)
    if time_left() > 100:
        h_new, c_new = random_pair_descent(best_h, n_iter=400, n_grid=25, rng=rng)
        if c_new < best_c - 1e-13:
            best_h, best_c = h_new, c_new
            h_new, c_new = lse_descent(best_h, beta=100, max_iters=40)
            if c_new < best_c - 1e-13:
                best_h, best_c = h_new, c_new

    # Stage 5: SLSQP
    if time_left() > 70:
        h_sol, c_sol = slsqp_solve(best_h, maxiter=200)
        if h_sol is not None and c_sol < best_c - 1e-12:
            best_h, best_c = h_sol, c_sol
            h_new, c_new = pair_descent_sweep(best_h, max_rounds=30, n_pairs=100, n_grid=35)
            if c_new < best_c - 1e-13:
                best_h, best_c = h_new, c_new

    # Stage 6: Final intensive pair descent
    if time_left() > 50:
        h_new, c_new = pair_descent_sweep(best_h, max_rounds=50, n_pairs=150, n_grid=41)
        if c_new < best_c - 1e-13:
            best_h, best_c = h_new, c_new

    # Stage 7: Final gradient polish
    if time_left() > 25:
        h_new, c_new = lse_descent(best_h, beta=120, max_iters=60)
        if c_new < best_c - 1e-13:
            best_h, best_c = h_new, c_new
        h_new, c_new = active_set_descent(best_h, max_iters=80, n_active=12)
        if c_new < best_c - 1e-13:
            best_h, best_c = h_new, c_new

    # Stage 8: Very final pair polish
    if time_left() > 15:
        h_new, c_new = pair_descent_sweep(best_h, max_rounds=20, n_pairs=100, n_grid=45)
        if c_new < best_c - 1e-14:
            best_h, best_c = h_new, c_new

    # Final validation
    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)

    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            final_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 2:
            final_c = float(eval_result[0])
    except Exception:
        pass

    return best_h, final_c, len(best_h)
