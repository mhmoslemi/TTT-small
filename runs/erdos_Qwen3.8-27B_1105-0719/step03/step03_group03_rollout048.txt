    n = len(vec)

    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(-h)  # ones first (least capacity)
        # Actually, we want to add to entries with most capacity
        order = np.argsort(1.0 - h)  # most capacity first
        remaining = diff

            idx = order[i]
            add = min(remaining, 1.0 - h[idx])
            h[idx] += add
            remaining -= add
            if remaining <= 1e-12:

    else:

def frac_opt_x(A, B, tol=1e-10, max_iter=60):
    """Find x in [0,1] minimizing C(x) = max_m(A[m]*x + B[m])."""
    lo = 0.0
    hi = 1.0
    for _ in range(max_iter):
        if hi - lo < tol:
            break
        mid = 0.5 * (lo + hi)
        vals = A * mid + B
        c = vals.max()
        active = np.where(np.abs(vals - c) < 1e-12)[0]
        max_slope = A[active].max()
        min_slope = A[active].min()
        if min_slope > 1e-12:
            # C is increasing at mid, go left
            hi = mid
        elif max_slope < -1e-12:
            # C is decreasing at mid, go right
            lo = mid
        else:
            # 0 is in the subgradient
            break
    return 0.5 * (lo + hi)

import numpy as np
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, 1.0 - h[idx])
            h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            sub = min(remaining, h[idx])
            h[idx] -= sub
            remaining -= sub
            if remaining <= 1e-12:
                break
    h = np.clip(h, 0.0, 1.0)
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def thue_morse_seq(n_pow):
    n2 = 1 << n_pow
    t = np.zeros(n2, dtype=np.float64)
    for i in range(1, n2):
        t[i] = t[i >> 1] ^ (i & 1)
    return t

def compute_delta_P_all(h):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    offsets = np.arange(M) - (n - 1)
    idx1 = np.arange(n)[:, None] + offsets[None, :]
    mask1 = (idx1 >= 0) & (idx1 < n)
    idx2 = np.arange(n)[:, None] - offsets[None, :]
    mask2 = (idx2 >= 0) & (idx2 < n)
    h_idx1 = np.where(mask1, h[np.clip(idx1, 0, n - 1)], 0.0)
    h_idx2 = np.where(mask2, h[np.clip(idx2, 0, n - 1)], 0.0)
    dP = dx * ((1.0 - h_idx1) * mask1 - h_idx2 * mask2)
    return dP

def binary_swap_descent(h, max_sweeps=500, tl_fn=None):
    n = len(h)
    h = h.copy()
    P = compute_profile(h)
    current_best = P.max()
    for sweep in range(max_sweeps):
        if tl_fn is not None and tl_fn() < 15:
            break
        P = compute_profile(h)
        current_max = P.max()
        if current_max < current_best - 1e-13:
            current_best = current_max
        ones = np.where(h > 0.99)[0]
        zeros = np.where(h < 0.01)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        dP = compute_delta_P_all(h)
        dP_o = dP[ones]
        dP_z = dP[zeros]
        best_new_max = current_max
        best_j, best_k = -1, -1
        for ii in range(len(ones)):
            base = P - dP_o[ii]
            new_maxs = np.max(base[None, :] + dP_z, axis=1)
            min_val = new_maxs.min()
            if min_val < best_new_max - 1e-12:
                best_new_max = min_val
                best_k = zeros[new_maxs.argmin()]
                best_j = ones[ii]
        if best_j == -1:
            break
        h[best_j] = 0.0
        h[best_k] = 1.0
    return h, current_best

def frac_opt_x(A, B, tol=1e-11, max_iter=60):
    lo = 0.0
    hi = 1.0
    for _ in range(max_iter):
        if hi - lo < tol:
            break
        mid = 0.5 * (lo + hi)
        vals = A * mid + B
        c = vals.max()
        active = np.where(np.abs(vals - c) < 1e-13)[0]
        max_slope = A[active].max()
        min_slope = A[active].min()
        if min_slope > 1e-14:
            hi = mid
        elif max_slope < -1e-14:
            lo = mid
        else:
            break
    return 0.5 * (lo + hi)

def fractional_refine(h, max_sweeps=40, tl_fn=None):
    n = len(h)
    h = h.copy()
    for sweep in range(max_sweeps):
        if tl_fn is not None and tl_fn() < 10:
            break
        P = compute_profile(h)
        dP = compute_delta_P_all(h)
        current_max = P.max()
        improved = False
        for j in range(n):
            old_val = h[j]
            B = P - dP[j] * old_val
            A = dP[j]
            opt_x = frac_opt_x(A, B)
            if abs(opt_x - old_val) > 1e-9:
                new_c = np.max(A * opt_x + B)
                if new_c < current_max - 1e-10:
                    h[j] = opt_x
                    improved = True
        if not improved:
            break
        h = repair_feasible(h)
    return h

def interp_rescale(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    fx = np.concatenate([[0.0], xs, [2.0]])
    fh = np.concatenate([[h[0]], h, [h[-1]]])
    hn = np.interp(xt, fx, fh)
    return repair_feasible(np.clip(hn, 0, 1))

def subgradient_polish(h, max_iters=200, tl_fn=None):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    h = h.copy()
    best_c = exact_score(h)
    step = 0.02
    for it in range(max_iters):
        if tl_fn is not None and tl_fn() < 10:
            break
        P = compute_profile(h)
        c_max = P.max()
        active = np.where(P >= c_max - 0.005)[0]
        if len(active) > 15:
            active = active[np.argsort(P[active])[-15:]]
        w = P[active] - P[active].min() + 1e-8
        w /= w.sum()
        dP = compute_delta_P_all(h)
        dP_a = dP[:, active]
        grad = dP_a @ w
        grad -= grad.mean()
        gn = np.linalg.norm(grad)
        if gn < 1e-15:

        grad /= gn
        h_try = repair_feasible(h - step * grad)

        if c_try < best_c - 1e-13:
            h = h_try
            best_c = c_try
            step = min(step * 1.1, 0.05)
        else:
            step *= 0.6
            if step < 0.0005:

    return h, best_c
