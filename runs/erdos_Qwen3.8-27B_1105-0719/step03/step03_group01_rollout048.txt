def project_feasible(x, target_sum):
    """Project x onto {h : 0<=h<=1, sum(h)=target_sum}"""
    n = len(x)
    x = np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        cnt = mask.sum()
        if cnt > 0:
            h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def c5_of_h(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def corr_and_grad(h, dx):
    """Compute all correlations and their gradients.
    Returns: corrs (array of length 2n-1), grad_corrs (array of shape (2n-1, n))
    """
    n = len(h)
    m = 2 * n - 1
    corrs = np.correlate(h, 1.0 - h, mode='full')
    return corrs

def grad_single_corr(h, k, n):
    """Gradient of the k-th correlation (lag) with respect to h."""
    offset = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -offset)
    i_hi = min(n - 1, n - 1 - offset)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    j1 = idx + offset
    # d/dh[i] of h[i]*(1-h[i+offset]) = (1-h[i+offset])
    grad[idx] += (1.0 - h[j1])
    # d/dh[i+offset] of h[i]*(1-h[i+offset]) = -h[i]
    grad[j1] += (-h[idx])
    return grad

def slsqp_solve(h, dx, target_sum, max_iter=200, ftol=1e-10, maxfun=5000):
    """Solve the epigraph problem using SLSQP.
    Variables: x = [t, h_0, ..., h_{n-1}]
    """
    n = len(h)
    m = 2 * n - 1
    x0 = np.zeros(n + 1)
    x0[0] = c5_of_h(h, dx) + 0.005  # slight slack
    x0[1:] = h
    
    def objective(x):
        return x[0]
    
    def obj_jac(x):
        g = np.zeros(n + 1)
        g[0] = 1.0
        return g
    
    # Inequality constraints: t - dx * corr_k(h) >= 0 for all k
    # We'll use only the top active constraints for efficiency
    
    def active_constraints(x):
        t_val = x[0]
        h_val = x[1:]
        corrs = np.correlate(h_val, 1.0 - h_val, mode='full')
        # Return t - dx*corr_k for all k (should be >= 0)
        slacks = t_val - dx * corrs
        return slacks
    
    def active_jac(x):
        h_val = x[1:]
        n_loc = len(h_val)
        m_loc = 2 * n_loc - 1
        J = np.zeros((m_loc, n_loc + 1))
        J[:, 0] = 1.0  # d/dt = 1
        for k in range(m_loc):
            J[k, 1:] = -dx * grad_single_corr(h_val, k, n_loc)
        return J
    
    # Equality constraint: sum(h) - target_sum = 0
    def eq_constraint(x):
        return np.sum(x[1:]) - target_sum
    
    def eq_jac(x):
        g = np.zeros(n + 1)
        g[1:] = 1.0
        return g
    
    bounds = [(0.0, 0.6)] + [(0.0, 1.0)] * n
    
    constraints = [
        {'type': 'ineq', 'fun': active_constraints, 'jac': active_jac},
        {'type': 'eq', 'fun': eq_constraint, 'jac': eq_jac},
    ]
    
    try:
        res = minimize(objective, x0, method='SLSQP', jac=obj_jac,
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': max_iter, 'ftol': ftol, 'maxfun': maxfun})
    except Exception:
        return h, c5_of_h(h, dx)
    
    if res.x is not None and np.all(np.isfinite(res.x)):
        h_new = project_feasible(res.x[1:], target_sum)
        c5_new = c5_of_h(h_new, dx)
        return h_new, c5_new
    
    return h, c5_of_h(h, dx)

    x0[0] = c5_init + t_slack
    x0[1:] = h_init

        return x[0]
    
    def ineq_fun(x):
        corrs = np.correlate(x[1:], 1.0 - x[1:], mode='full')
        return x[0] - dx * corrs
    
    def ineq_jac(x):
        h = x[1:]
        J = np.zeros((m, n + 1))
        J[:, 0] = 1.0
        for k in range(m):
            d = k - (n - 1)
            J[k, 1:] = -dx * grad_single(h, d, n)
        return J
    
    def eq_fun(x):
        return np.array([np.sum(x[1:]) - target_sum])
    
    def eq_jac(x):
        J = np.zeros((1, n + 1))
        J[0, 1:] = 1.0
        return J
    
    bounds = [(0.0, 0.55)] + [(0.0, 1.0)] * n
    constraints = [
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac},
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
    ]
    
    try:
        res = minimize(obj, x0, method='SLSQP', bounds=bounds,
                      constraints=constraints,
                      options={'maxiter': max_iter, 'ftol': ftol, 'maxfun': maxfun})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = project_feasible(res.x[1:], target_sum)
            return h_new, c5_of_h(h_new, dx)
    except Exception:
        pass
    return h_init, c5_init
