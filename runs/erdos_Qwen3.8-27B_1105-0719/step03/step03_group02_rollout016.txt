```python
def swap_delta(h, i, j, n):
    # i goes 1->0, j goes 0->1
    deltas = np.zeros(2*n-1)
    s = np.arange(-(n-1), n)  # all offsets
    r = s + (n-1)  # indices into correlation array
    
    # i goes 1->0
    # Condition: 0 <= i+s <= n-1, i.e., -i <= s <= n-1-i
    mask1 = (s >= -i) & (s <= n-1-i)
    # h[i+s] for valid s
    valid_idx = i + s[mask1]
    deltas[r[mask1]] += h[valid_idx]
    
    # Condition: 0 <= i-s <= n-1, i.e., i-(n-1) <= s <= i
    mask2 = (s >= i-(n-1)) & (s <= i)
    valid_idx2 = i - s[mask2]
    deltas[r[mask2]] -= (1 - h[valid_idx2])
    
    # j goes 0->1
    mask3 = (s >= -j) & (s <= n-1-j)
    valid_idx3 = j + s[mask3]
    deltas[r[mask3]] -= h[valid_idx3]
    
    mask4 = (s >= j-(n-1)) & (s <= j)
    valid_idx4 = j - s[mask4]
    deltas[r[mask4]] += (1 - h[valid_idx4])
    
    return deltas
```
