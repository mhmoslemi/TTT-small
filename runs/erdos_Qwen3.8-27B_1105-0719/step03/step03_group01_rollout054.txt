import numpy as np

def project_feasible(x, target_sum):
    n = len(x)
    x = np.nan_to_num(np.asarray(x, dtype=float), nan=0.5, posinf=1.0, neginf=0.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = h < 1.0 - 1e-9
        else:
            mask = h > 1e-9
        cnt = mask.sum()
        if cnt > 0:
            h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def c5_of_h(h, dx):
    return float(np.max(compute_corrs(h)) * dx)

def corr_grad_at_k(h, k, n):
    """Gradient of corr[k] w.r.t. h."""
    d = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, d)
    i_hi = min(n - 1, n - 1 + d)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    j = idx - d
    grad[idx] += (1.0 - h[j])
    grad[j] += (-h[idx])
    return grad

def lse_value(h, dx, T):
    """Compute log-sum-exp(T * dx * corrs) / T."""
    corrs = compute_corrs(h)
    a = T * dx * corrs
    a_max = np.max(a)
    return a_max / T + np.log(np.sum(np.exp(a - a_max))) / T

def lse_grad(h, dx, T):
    """Gradient of log-sum-exp(T * dx * corrs) / T w.r.t. h."""
    n = len(h)
    corrs = compute_corrs(h)
    a = T * dx * corrs
    a_max = np.max(a)
    weights = np.exp(a - a_max)
    weights /= np.sum(weights)
    # grad = dx * sum_k weights[k] * grad_corr_k
    grad = np.zeros(n)
    for k in range(len(corrs)):
        if weights[k] < 1e-12:
            continue
        grad += weights[k] * corr_grad_at_k(h, k, n)
    return dx * grad

                h_ep = epigraph_optimize(h_opt, dx, 50, target_sum, maxiter=40)
                c5_ep = c5_of_h(h_ep, dx)
                if c5_ep < best_c5:
                    best_h = h_ep
                    best_c5 = c5_ep

    # Phase 4: Resolution scanning
    if time.time() < t0 + budget_s * 0.82:
        n_candidates = [96, 108, 112, 120, 128, 132, 136, 140, 144, 150]
        for n_try in n_candidates:
            if time.time() > t0 + budget_s * 0.82:
                break
            if n_try == n:
                continue
            h_new, c5_new = try_resolution(n_try, best_h, best_c5, t0,
                                           t0 + budget_s * 0.82, seed + n_try)
            if h_new is not None and c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new

    # Phase 5: Final polish - pair swaps and small LSE
    if time.time() < hard_deadline:
        n_b = len(best_h)
        dx_b = 2.0 / n_b
        target_b = n_b / 2.0
        remaining = hard_deadline - time.time()

        # Pair swap polish
        n_swaps = int(min(400, remaining * 80))
        h_pol, c5_pol = pair_swap_polish(best_h, dx_b, n_swaps, rng, t0, hard_deadline - 10)
        if c5_pol < best_c5:
            best_h = h_pol
            best_c5 = c5_pol

        # Final LSE + epigraph
        if time.time() < hard_deadline - 20:
            h_final = lse_optimize(best_h, dx_b, 200.0, target_b, maxiter=40)
            c5_f = c5_of_h(h_final, dx_b)
            if c5_f < best_c5:
                best_h = h_final
                best_c5 = c5_f

        if time.time() < hard_deadline - 10:
            h_ep = epigraph_optimize(best_h, dx_b, 60, target_b, maxiter=40)
            c5_e = c5_of_h(h_ep, dx_b)
            if c5_e < best_c5:
                best_h = h_ep
                best_c5 = c5_e

        # Tiny pair swaps at the very end
        if time.time() < hard_deadline - 2:
            h_min, c5_min = pair_swap_polish(best_h, dx_b, 50, rng, t0, hard_deadline - 1)
            if c5_min < best_c5:
                best_h = h_min
                best_c5 = c5_min

    # Final validation
    n_final = len(best_h)
    dx_final = 2.0 / n_final
    target_final = n_final / 2.0
    best_h = project_feasible(best_h, target_final)
    best_c5 = c5_of_h(best_h, dx_final)

    return best_h, best_c5, n_final
