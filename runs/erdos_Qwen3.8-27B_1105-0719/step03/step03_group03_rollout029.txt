def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx])
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:

    else:
        order = np.argsort(h)
        remaining = -diff

        else:
            idx = int(np.argmin(h))

    return h


def feasible_interval(h, v):
    a_lo = -np.inf
    a_hi = np.inf
    pos = v > 1e-14
    neg = v < -1e-14
    if np.any(pos):
        a_hi = np.min((1.0 - h[pos]) / v[pos])
    if np.any(neg):
        a_lo = np.max(-h[neg] / v[neg])
    if not (a_lo < a_hi):
        return None
    # Cap to reasonable range
    a_lo = max(a_lo, -2.0)
    a_hi = min(a_hi, 2.0)
    if a_lo >= a_hi:
        return None
    return (a_lo, a_hi)


def ray_search(h, v, n_grid=200):
    n = len(h)
    R0 = np.correlate(h, 1.0 - h, mode='full')
    R1 = np.correlate(v, 1.0 - h, mode='full') - np.correlate(h, v, mode='full')
    R2 = -np.correlate(v, v, mode='full')
    
    fi = feasible_interval(h, v)
    if fi is None:
        return None
    a_lo, a_hi = fi
    current_score = np.max(R0)
    
    # Quick check: at alpha=0, score is current_score
    # Check endpoints and a few interior points first
    candidates = [a_lo + 1e-10, a_hi - 1e-10, 0.0]
    # Add grid
    alphas = np.linspace(a_lo, a_hi, n_grid)
    
    best_val = current_score
    best_alpha = None
    for a in alphas:
        val = np.max(R0 + a * R1 + a * a * R2)
        if val < best_val:
            best_val = val
            best_alpha = a
    
    if best_alpha is None:
        return None
    
    # Golden section refinement
    # Bracket around best_alpha
    idx = int(np.argmin([np.max(R0 + a * R1 + a * a * R2) for a in alphas]))
    lo_i = max(0, idx - 2)
    hi_i = min(len(alphas) - 1, idx + 2)
    lo, hi = float(alphas[lo_i]), float(alphas[hi_i])
    
    for _ in range(60):
        if hi - lo < 1e-13:
            break
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        a1 = hi - gr * (hi - lo)
        a2 = lo + gr * (hi - lo)
        f1 = np.max(R0 + a1 * R1 + a1 * a1 * R2)
        f2 = np.max(R0 + a2 * R1 + a2 * a2 * R2)
        if f1 < f2:
            hi = a2
        else:
            lo = a1

    final_val = np.max(R0 + best_alpha * R1 + best_alpha * best_alpha * R2)
    
    if final_val >= current_score - 1e-14:
        return None
    
    h_new = h + best_alpha * v
    h_new = repair_feasible(h_new)
    c_new = exact_score(h_new)
    return h_new, c_new

    """Generate a batch of diverse directions."""
    n = len(h)
    directions = []
    
    # Random full directions

        mv = np.max(np.abs(v))

def interp_to_dim(h, new_n):
    n = len(h)
    if n == new_n:

    xs = (np.arange(n) + 0.5) * (2.0 / n)

def slsqp_refine(h0, maxiter, tol):

    dx = 2.0 / n
    M = 2 * n - 1
    t0 = exact_score(h0) + 1e-4
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, t0 + 2.0)]
    
    def obj(x):
        return x[-1]
    
    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    
    def eq_fun(x):
        return x[:-1].sum() - n / 2.0
    
    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g
    
    def ineq_fun(x):
        h = x[:-1]
        return x[-1] - np.correlate(h, 1 - h, mode="full") * dx
    
    def build_G(h, n, dx):
        M_local = 2 * n - 1
        G = np.zeros((M_local, n))
        center = n - 1
        G[center, :] = dx * (1.0 - 2.0 * h)
        for d in range(1, n):
            m_pos = center + d
            idx1 = np.arange(n - d)
            G[m_pos, idx1] += -dx * h[idx1 + d]
            G[m_pos, idx1 + d] += dx * (1.0 - h[idx1])
            m_neg = center - d
            G[m_neg, idx1] += dx * (1.0 - h[idx1 + d])
            G[m_neg, idx1 + d] += -dx * h[idx1]
        return G
    
    def ineq_jac(x):
        h = x[:-1]
        G = build_G(h, n, dx)
        J = np.zeros((M, n + 1))
        J[:, :-1] = -G
        J[:, -1] = 1.0
        return J
    
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': tol})
        if res.x is not None and np.isfinite(res.x).all():
            hs = repair_feasible(res.x[:-1])
            if np.isfinite(exact_score(hs)):
                return hs
    except Exception:
        pass
    return None

`np.correlate(a, b, mode='full')` at index `m` (0 to 2n-2) computes:
`sum_{i} a[i] * b[i + m - (n_b - 1)]`

For our case, a = h, b = 1-h, both of length n:
At lag m: `sum_i h[i] * (1-h[i + m - (n-1)])`

The valid range of i is such that both i and i+m-(n-1) are in [0, n-1].

Let d = m - (n-1). Then the lag is d (can be negative, zero, or positive).
At lag d: `sum_{i} h[i] * (1-h[i+d])` where i ranges such that 0 ≤ i < n and 0 ≤ i+d < n.

If d ≥ 0: i ranges from 0 to n-1-d
If d < 0: i ranges from -d to n-1

So P[d] = sum_{i=max(0,-d)}^{min(n-1, n-1-d)} h[i] * (1-h[i+d])

The gradient of P[d] with respect to h[j]:
- h[j] appears in the term where i = j: contributes (1-h[j+d]) if j+d is valid
- h[j] appears in the term where i+d = j, i.e., i = j-d: contributes -h[j-d] if j-d is valid

So dP[d]/dh[j] = (1-h[j+d]) * [j+d valid] - h[j-d] * [j-d valid]

where [x] is 1 if x is a valid index, 0 otherwise.

def compute_subgradient(h, n, delta=0.01):
    """Compute a subgradient of the objective at h."""
    profile = np.correlate(h, 1.0 - h, mode='full')
    p_max = np.max(profile)
    active = np.where(profile >= p_max - delta)[0]
    
    # Build gradient matrix
    G = np.zeros((2*n-1, n))
    center = n - 1
    G[center, :] = 1.0 - 2.0 * h
    for d in range(1, n):
        m_pos = center + d
        idx1 = np.arange(n - d)
        G[m_pos, idx1] += -h[idx1 + d]
        G[m_pos, idx1 + d] += (1.0 - h[idx1])
        m_neg = center - d
        G[m_neg, idx1] += (1.0 - h[idx1 + d])
        G[m_neg, idx1 + d] += -h[idx1]
    
    # Weighted average of active gradients
    # Weight proportional to (profile[k] - (p_max - delta))
    weights = profile[active] - (p_max - delta)
    weights = weights / weights.sum()
    
    sg = G[active].T @ weights
    
    # Project onto zero-sum
    sg -= sg.mean()
    return sg

            idx = int(np.argmin(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h


def feasible_interval(h, v):
    a_lo = -np.inf
    a_hi = np.inf
    pos = v > 1e-14
    neg = v < -1e-14
    if np.any(pos):
        a_hi = np.min((1.0 - h[pos]) / v[pos])
    if np.any(neg):
        a_lo = np.max(-h[neg] / v[neg])
    if not (a_lo < a_hi):
        return None
    a_lo = max(a_lo, -3.0)
    a_hi = min(a_hi, 3.0)
    if a_lo >= a_hi:
        return None
    return (a_lo, a_hi)


def ray_search(h, v, n_grid=250):
    n = len(h)
    R0 = np.correlate(h, 1.0 - h, mode='full')
    R1 = np.correlate(v, 1.0 - h, mode='full') - np.correlate(h, v, mode='full')
    R2 = -np.correlate(v, v, mode='full')

    fi = feasible_interval(h, v)
    if fi is None:
        return None
    a_lo, a_hi = fi
    current_score = np.max(R0)

    alphas = np.linspace(a_lo, a_hi, n_grid)
    best_val = current_score
    best_alpha = None
    best_idx = -1

    for i_a, a in enumerate(alphas):
        val = np.max(R0 + a * R1 + a * a * R2)
        if val < best_val:
            best_val = val
            best_alpha = a
            best_idx = i_a

    if best_alpha is None:
        return None

    lo_i = max(0, best_idx - 3)
    hi_i = min(len(alphas) - 1, best_idx + 3)
    lo, hi = float(alphas[lo_i]), float(alphas[hi_i])

    for _ in range(70):
        if hi - lo < 1e-14:
            break
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        a1 = hi - gr * (hi - lo)
        a2 = lo + gr * (hi - lo)
        f1 = np.max(R0 + a1 * R1 + a1 * a1 * R2)
        f2 = np.max(R0 + a2 * R1 + a2 * a2 * R2)
        if f1 < f2:
            hi = a2
        else:
            lo = a1

    best_alpha = (lo + hi) / 2.0
    final_val = np.max(R0 + best_alpha * R1 + best_alpha * best_alpha * R2)

    if final_val >= current_score - 1e-14:
        return None

    h_new = h + best_alpha * v
    h_new = repair_feasible(h_new)
    c_new = exact_score(h_new)
    return h_new, c_new


def build_grad_matrix(h, n):
    M = 2 * n - 1
    G = np.zeros((M, n))
    center = n - 1
    G[center, :] = 1.0 - 2.0 * h
    for d in range(1, n):
        m_pos = center + d
        idx1 = np.arange(n - d)
        G[m_pos, idx1] += -h[idx1 + d]
        G[m_pos, idx1 + d] += (1.0 - h[idx1])
        m_neg = center - d
        G[m_neg, idx1] += (1.0 - h[idx1 + d])
        G[m_neg, idx1 + d] += -h[idx1]
    return G


def compute_subgradient_direction(h, delta=0.005):
    n = len(h)
    profile = np.correlate(h, 1.0 - h, mode='full')
    p_max = np.max(profile)
    active = np.where(profile >= p_max - delta)[0]
    if len(active) == 0:
        active = np.array([np.argmax(profile)])

    G = build_grad_matrix(h, n)
    weights = profile[active] - (p_max - delta)
    if weights.sum() < 1e-15:
        weights = np.ones(len(active))
    weights = weights / weights.sum()

    sg = G[active].T @ weights
    sg -= sg.mean()
    nv = np.linalg.norm(sg)
    if nv < 1e-12:
        return None
    return -sg / nv


def gen_random_direction(n, rng):
    v = rng.randn(n)
    v -= v.mean()
    mv = np.max(np.abs(v))
    if mv < 1e-10:
        return None
    return v / mv


def gen_structured_directions(h, rng):
    n = len(h)
    dirs = []

    # Sinusoidal
    for k in [1, 2, 3, 4, 5, 7, 10, 14]:
        if k >= n // 2:
            break
        phase = rng.uniform(0, 2 * np.pi)
        v = np.sin(2 * np.pi * k * np.arange(n) / n + phase)
        v -= v.mean()
        mv = np.max(np.abs(v))
        if mv > 1e-10:
            dirs.append(v / mv)

    # Linear ramp
    v = np.arange(n, dtype=float)
    v -= v.mean()
    dirs.append(v / np.max(np.abs(v)))
    dirs.append(-v / np.max(np.abs(v)))

    # Cosine ramp
    v = np.cos(2 * np.pi * np.arange(n) / n)
    v -= v.mean()
    mv = np.max(np.abs(v))
    if mv > 1e-10:
        dirs.append(v / mv)

    # Block transfer high to low
    quarter = max(1, n // 4)
    high_idx = np.argsort(-h)[:quarter]
    low_idx = np.argsort(h)[:quarter]
    v = np.zeros(n)
    v[high_idx] = -1.0
    v[low_idx] = 1.0
    v -= v.mean()
    mv = np.max(np.abs(v))
    if mv > 1e-10:
        dirs.append(v / mv)

    # Pair transfers
    for _ in range(6):
        i = int(rng.randint(n))
        j = int(rng.randint(n))
        if i != j:
            v = np.zeros(n)
            v[i] = -1.0
            v[j] = 1.0
            dirs.append(v)

    # Anti-diagonal
    v = np.zeros(n)
    for i in range(n):
        v[i] = 1.0 if i < n // 2 else -1.0
    v -= v.mean()
    mv = np.max(np.abs(v))
    if mv > 1e-10:
        dirs.append(v / mv)

    return dirs


def interp_to_dim(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    fx = np.concatenate([[0.0], xs, [2.0]])
    fh = np.concatenate([[h[0]], h, [h[-1]]])
    hn = np.interp(xt, fx, fh)
    return repair_feasible(np.clip(hn, 0, 1))


def slsqp_refine(h0, maxiter, tol):
    n = len(h0)
    dx = 2.0 / n
    M = 2 * n - 1
    t0 = exact_score(h0) + 1e-4
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, t0 + 2.0)]

    def obj(x):
        return x[-1]

    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g

    def eq_fun(x):
        return x[:-1].sum() - n / 2.0

    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g

    def ineq_fun(x):
        h = x[:-1]
        return x[-1] - np.correlate(h, 1 - h, mode="full") * dx

    def ineq_jac(x):
        h = x[:-1]
        G = build_grad_matrix(h, n) * dx
        J = np.zeros((M, n + 1))
        J[:, :-1] = -G
        J[:, -1] = 1.0
        return J

    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': tol})
        if res.x is not None and np.isfinite(res.x).all():
            hs = repair_feasible(res.x[:-1])
            if np.isfinite(exact_score(hs)):
                return hs
    except Exception:
        pass
    return None


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()
    rng = np.random.RandomState(seed)

    def tl():
        return budget_s + t_start - time.perf_counter()

    parent = None
    try:
        parent = np.array(initial_h_values, dtype=np.float64)
    except (NameError, Exception):
        pass
    if parent is None or len(parent) < 4:
        parent = np.full(70, 0.5)

    parent = repair_feasible(np.clip(parent, 0.0, 1.0))
    best_h = parent.copy()
    best_c = exact_score(parent)

    # Symmetrize
    h_sym = repair_feasible((parent + parent[::-1]) / 2.0)
    c_sym = exact_score(h_sym)
    if c_sym < best_c - 1e-12:
        best_h, best_c = h_sym, c_sym

    def do_ray_phase(h, c, n_rounds, time_budget_fn):
        h = h.copy()
        c = c
        n = len(h)
        for r in range(n_rounds):
            if time_budget_fn() < 12:
                break
            # Collect directions for this round
            dirs = []

            # Subgradient-based direction (most targeted)
            sg_dir = compute_subgradient_direction(h, delta=0.005)
            if sg_dir is not None:
                dirs.append(sg_dir)
                # Also try with different deltas
                sg_dir2 = compute_subgradient_direction(h, delta=0.002)
                if sg_dir2 is not None:
                    dirs.append(sg_dir2)
                sg_dir3 = compute_subgradient_direction(h, delta=0.01)
                if sg_dir3 is not None:
                    dirs.append(sg_dir3)

            # Random directions
            for _ in range(6):
                v = gen_random_direction(n, rng)
                if v is not None:
                    dirs.append(v)

            # Structured directions
            struct_dirs = gen_structured_directions(h, rng)
            dirs.extend(struct_dirs[:8])

            # Try each direction
            any_accept = False
            for v in dirs:
                if time_budget_fn() < 12:
                    break
                mv = np.max(np.abs(v))
                if mv < 1e-12:
                    continue
                vn = v / mv
                result = ray_search(h, vn, n_grid=250)
                if result is not None:
                    h_new, c_new = result
                    if c_new < c - 1e-14:
                        h = h_new
                        c = c_new
                        any_accept = True
        return h, c

    # Phase 1: Intensive ray search with subgradient guidance
    if tl() > 200:
        best_h, best_c = do_ray_phase(best_h, best_c, n_rounds=50, time_budget_fn=tl)

    # Phase 2: SLSQP refinement
    if tl() > 130:
        hs = slsqp_refine(best_h, maxiter=1000, tol=1e-12)
        if hs is not None:
            cs = exact_score(hs)
            if cs < best_c - 1e-14:
                best_h, best_c = hs, cs

    # Phase 3: More ray search
    if tl() > 180:
        best_h, best_c = do_ray_phase(best_h, best_c, n_rounds=60, time_budget_fn=tl)

    # Phase 4: Dimension exploration
    if tl() > 120:
        n0 = len(best_h)
        dims = []
        for d in range(max(40, n0 - 16), min(110, n0 + 17), 2):
            if d != n0 and d >= 40:
                dims.append(d)
        dims.sort(key=lambda d: abs(d - n0))
        for nd in dims[:8]:
            if tl() < 40:
                break
            hi = interp_to_dim(best_h, nd)
            ci = exact_score(hi)
            if ci < best_c + 0.025:
                bh, bc = do_ray_phase(hi, ci, n_rounds=30, time_budget_fn=tl)
                if bc < best_c - 1e-14:
                    best_h, best_c = bh, bc

    # Phase 5: SLSQP again
    if tl() > 100:
        hs = slsqp_refine(best_h, maxiter=1200, tol=1e-13)
        if hs is not None:
            cs = exact_score(hs)
            if cs < best_c - 1e-14:
                best_h, best_c = hs, cs

    # Phase 6: Final intensive ray search
    if tl() > 80:
        best_h, best_c = do_ray_phase(best_h, best_c, n_rounds=40, time_budget_fn=tl)

    # Phase 7: Perturbation jumps + ray search
    if tl() > 80:
        for _ in range(8):
            if tl() < 20:
                break
            eps = rng.uniform(0.002, 0.015)
            z = rng.randn(len(best_h))
            z -= z.mean()
            z /= (np.linalg.norm(z) / len(best_h) + 1e-15)
            hp = repair_feasible(best_h + eps * z)
            cp_ = exact_score(hp)
            if cp_ < best_c + 0.004:
                bh, bc = do_ray_phase(hp, cp_, n_rounds=20, time_budget_fn=tl)
                if bc < best_c - 1e-14:
                    best_h, best_c = bh, bc

    # Phase 8: Final SLSQP polish
    if tl() > 50:
        hs = slsqp_refine(best_h, maxiter=1500, tol=1e-14)
        if hs is not None:
            cs = exact_score(hs)
            if cs < best_c - 1e-14:
                best_h, best_c = hs, cs

    # Phase 9: Final ray micro-steps
    if tl() > 20:
        best_h, best_c = do_ray_phase(best_h, best_c, n_rounds=20, time_budget_fn=tl)

    # Final validation
    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)

    try:
        er = evaluate_erdos_solution(best_h)
        if isinstance(er, (float, int)):
            final_c = min(final_c, float(er))
        elif isinstance(er, tuple) and len(er) >= 2:
            final_c = min(final_c, float(er[0]))
    except Exception:
        pass

    return best_h, final_c, len(best_h)
