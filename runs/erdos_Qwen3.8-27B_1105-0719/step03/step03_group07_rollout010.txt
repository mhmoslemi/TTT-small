import numpy as np
from scipy.optimize import minimize, linprog
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    prof = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(prof)) * dx

def profile_full(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for idx in order:
            add = min(remaining, capacity[idx])
            h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for idx in order:
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_lag(h, d):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    if d == 0:
        g[:] = dx * (1.0 - 2.0 * h)
    elif d > 0:
        for i in range(n - d):
            g[i] += dx * (1.0 - h[i + d])
            g[i + d] -= dx * h[i]
    else:
        dp = -d
        for j in range(n - dp):
            g[j + dp] += dx * (1.0 - h[j])
            g[j] -= dx * h[j + dp]
    return g

def interp_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    return repair_feasible(np.clip(h_new, 0.0, 1.0))

def lp_leveling_direction(h, active_indices, v_bound=0.5):
    n = len(h)
    M = 2 * n - 1
    A_rows = []
    for idx in active_indices:
        d = idx - (n - 1)
        g = grad_lag(h, d)
        row = np.append(g, 1.0)
        A_rows.append(row)
    A_ineq = np.array(A_rows)
    b_ineq = np.zeros(len(active_indices))
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    c = np.zeros(n + 1)
    c[-1] = -1.0
    bounds = [(-v_bound, v_bound)] * n + [(-1.0, None)]
    try:
        res = linprog(c, A_ub=A_ineq, b_ub=b_ineq, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            v = res.x[:n].copy()
            eta = res.x[-1]
            if eta > 1e-10:

    dx = 2.0 / n
    P = profile_full(h0)
    c0 = P.max()
    
    def obj(x):
        return x[-1]
    
    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g

        return x[:-1].sum() - n / 2.0

        g[:-1] = 1.0
        return g

        h = x[:-1]
        t = x[-1]

def pair_line_search(h, i, j, n_samples=25):
    s = h[i] + h[j]
    lo = max(0.0, s - 1.0)
    hi = min(1.0, s)
    if hi - lo < 1e-10:
        return h, exact_score(h)
    best_c = exact_score(h)
    best_val = h[i]
    t = np.linspace(lo, hi, n_samples)
    for tv in t:
        h[i] = tv
        h[j] = s - tv
        c = exact_score(h)
        if c < best_c - 1e-14:
            best_c = c
            best_val = tv
    h[i] = best_val
    h[j] = s - best_val
    return h, best_c
