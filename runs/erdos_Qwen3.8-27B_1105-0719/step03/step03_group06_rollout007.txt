def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(-(1.0 - h))
        remaining = diff
        for i in range(n):
            idx = order[i]
            cap = 1.0 - h[idx]
            if cap <= 1e-15:
                continue
            add = min(remaining, cap)
            h[idx] += add
            remaining -= add
            if remaining <= 1e-14:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            if h[idx] <= 1e-15:
                continue
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-14:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] = float(np.clip(h[idx] + diff2, 0.0, 1.0))
    return h

def grad_row(h, m, n):

    d = m - (n - 1)
    if d == 0:
        return dx * (1.0 - 2.0 * h)
    elif d > 0:
        g = np.zeros(n)
        g[d:] += dx * (1.0 - h[:-d])
        g[:-d] -= dx * h[d:]
        return g
    else:
        p = -d
        g = np.zeros(n)
        g[:-p] += dx * (1.0 - h[p:])
        g[p:] -= dx * h[:-p]
        return g

    obj = (profile.max() + np.log(Z) / mu)

            break
        h = best_h_try
        if best_val < best_c:
            best_c = best_val
            best_h = h.copy()
    return best_h, best_c

def binary_local_search(h_cont, max_iters=2000, temp0=0.01, cooldown=0.997):
    n = len(h_cont)
    h = (h_cont > 0.5).astype(float)
    ones = int(h.sum())
    target = n // 2
    if ones > target:
        one_idx = np.where(h == 1.0)[0]
        excess = ones - target
        flip = one_idx[:excess]
        h[flip] = 0.0
    elif ones < target:
        zero_idx = np.where(h == 0.0)[0]
        deficit = target - ones
        flip = zero_idx[:deficit]
        h[flip] = 1.0
    h = repair_feasible(h)
    best_c = exact_score(h)
    best_h = h.copy()
    temp = temp0
    M = 2 * n - 1
    profile = overlap_profile(h)
    for it in range(max_iters):
        c = float(profile.max())
        thresh = c - 0.003
        active = np.where(profile >= thresh)[0]
        if len(active) > 10:
            active = active[np.argsort(profile[active])[-10:]]
        zero_idx = np.where(h < 0.5)[0]
        one_idx = np.where(h > 0.5)[0]
        if len(zero_idx) == 0 or len(one_idx) == 0:
            break
        best_move_c = c
        best_move = None
        n_tries = min(50, len(zero_idx) * len(one_idx))
        rng_local = np.random.RandomState(it * 7 + 3)
        for _ in range(n_tries):
            i = zero_idx[rng_local.randint(len(zero_idx))]
            j = one_idx[rng_local.randint(len(one_idx))]
            h_try = h.copy()
            h_try[i] = 1.0
            h_try[j] = 0.0
            s = 0.0
            for m in active:
                d = m - (n - 1)
                if d == 0:
                    s += h_try[i] * (1.0 - h_try[i])
                elif d > 0:
                    if i >= d and i - d < n:
                        s += h_try[i - d] * (1.0 - h_try[i])
                    if j >= d and j - d < n:
                        s += h_try[j - d] * (1.0 - h_try[j])
                else:
                    p = -d
                    if i < n - p:
                        s += h_try[i] * (1.0 - h_try[i + p])
                    if j < n - p:
                        s += h_try[j] * (1.0 - h_try[j + p])
            est_c = c - s / len(active) * 0.5
            if best_move is None or est_c < best_move_c:
                best_move_c = est_c
                best_move = (i, j)
        if best_move is None:
            break
        i, j = best_move
        h_try = h.copy()
        h_try[i] = 1.0
        h_try[j] = 0.0
        h_try = repair_feasible(h_try)
        s_try = exact_score(h_try)
        delta = s_try - c
        if delta < 0 or np.random.random() < np.exp(-delta / max(temp, 1e-10)):
            h = h_try
            profile = overlap_profile(h)
            if s_try < best_c:
                best_c = s_try
                best_h = h.copy()
        temp *= cooldown
    return best_h, best_c

def interp_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    edges_old = np.linspace(0.0, 2.0, n + 1)
    edges_new = np.linspace(0.0, 2.0, new_n + 1)
    h_new = np.zeros(new_n)
    for k in range(new_n):
        a, b = edges_new[k], edges_new[k + 1]
        total = 0.0
        for j in range(n):
            lo = max(a, edges_old[j])
            hi = min(b, edges_old[j + 1])
            if hi > lo:
                total += h[j] * (hi - lo)
        h_new[k] = total / (b - a)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

def beatty_seed(n, alpha, rng):
    half = n // 2
    h = np.zeros(n)
    idx = np.arange(n)
    vals = (idx * alpha) % 1.0
    top = np.argsort(vals)[-half:]
    h[top] = 1.0
    return repair_feasible(h)

def quadratic_residue_seed(n, p, rng):
    half = n // 2
    h = np.zeros(n)
    idx = np.arange(n)
    residues = set()
    for x in range(p):
        residues.add((x * x) % p)
    vals = np.array([0.5 if i % p in residues else 0.0 for i in idx])
    ones = int((vals > 0.3).sum())
    if ones > half:
        to_flip = np.where(vals > 0.3)[0]
        to_flip = to_flip[rng.permutation(len(to_flip))[:ones - half]]
        vals[to_flip] = 0.0
    elif ones < half:
        to_flip = np.where(vals <= 0.3)[0]
        to_flip = to_flip[rng.permutation(len(to_flip))[:half - ones]]
        vals[to_flip] = 1.0
    h = (vals > 0.3).astype(float)
    return repair_feasible(h)

def slsqp_optimize(h_start, maxiter=150):
    n = len(h_start)
    dx = 2.0 / n
    M = 2 * n - 1
    c0 = exact_score(h_start)
    t0 = c0 + 1e-3
    def obj(x):
        return x[-1]
    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    def eq_fun(x):
        return x[:-1].sum() - n / 2.0
    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g
    def ineq_fun(x):
        h = x[:-1]
        t = x[-1]
        profile = np.correlate(h, 1 - h, mode="full") * dx
        return t - profile
    def ineq_jac(x):
        h = x[:-1]
        J = np.zeros((M, n + 1))
        for m in range(M):
            J[m, :n] = -grad_row(h, m, n)
        J[:, -1] = 1.0
        return J
    x0 = np.concatenate([h_start, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 2.0)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None and np.isfinite(res.x).all():
            h_new = repair_feasible(res.x[:-1])
            c_new = exact_score(h_new)
            if c_new <= c0 + 1e-8:
                return h_new
    except Exception:
        pass
    return None

def coordinate_polish(h, max_iters=60):
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    for it in range(max_iters):
        profile = overlap_profile(h)
        c = float(profile.max())
        thresh = c - 0.001
        active = np.where(profile >= thresh)[0]
        if len(active) > 8:
            active = active[np.argsort(profile[active])[-8:]]
        improved = False
        idx_desc = np.argsort(-h)
        idx_asc = np.argsort(h)
        for attempt in range(min(25, n)):
            i = int(idx_desc[attempt % n])
            j = int(idx_asc[attempt % n])
            if i == j:
                continue
            delta = 0.03 * (h[i] - h[j])
            if abs(delta) < 1e-6:
                continue
            for sign in [1.0, -1.0]:
                h_try = h.copy()
                h_try[i] += sign * delta
                h_try[j] -= sign * delta
                if h_try.min() < -1e-10 or h_try.max() > 1 + 1e-10:
                    continue
                h_try = repair_feasible(h_try)
                s = exact_score(h_try)
                if s < best_c - 1e-14:
                    h = h_try
                    best_c = s
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s - (time.perf_counter() - t_start)

    try:
        parent = np.array(initial_h_values, dtype=np.float64)
    except Exception:
        parent = None
    if parent is None or len(parent) < 4:
        parent = np.full(70, 0.5)
    parent = np.clip(parent, 0.0, 1.0)
    parent = repair_feasible(parent)
    n_parent = len(parent)

    best_h = parent.copy()
    best_c = exact_score(parent)

    # Phase 1: Nesterov projected gradient on parent
    if time_left() > 200:
        for mu in [60.0, 100.0, 200.0]:
            if time_left() < 60:
                break
            h_new, c_new = nesterov_projected(best_h, max_iters=200, mu=mu, alpha0=0.03)
            if c_new < best_c - 1e-6:
                best_h, best_c = h_new, c_new

    # Phase 2: Adaptive active descent
    if time_left() > 100:
        h_ad, c_ad = active_descent_adaptive(best_h, max_iters=500, tol0=0.005, tol_min=0.0003)
        if c_ad < best_c:
            best_h, best_c = h_ad, c_ad

    # Phase 3: Binary local search at parent dimension
    if time_left() > 150:
        h_bin, c_bin = binary_local_search(best_h, max_iters=3000, temp0=0.008, cooldown=0.996)
        if c_bin < best_c:
            best_h, best_c = h_bin, c_bin
            # Refine continuously
            if time_left() > 80:
                h_ref, c_ref = active_descent_adaptive(best_h, max_iters=300, tol0=0.003, tol_min=0.0003)
                if c_ref < best_c:
                    best_h, best_c = h_ref, c_ref

    # Phase 4: Multi-start with diverse seeds
    n_ref = len(best_h)
    seeds_to_try = []
    # Beatty sequences with golden ratio and related constants
    seeds_to_try.append(beatty_seed(n_ref, 0.6180339887, rng))
    seeds_to_try.append(beatty_seed(n_ref, 0.4142135624, rng))
    seeds_to_try.append(beatty_seed(n_ref, 0.3819660113, rng))
    # Quadratic residue based
    seeds_to_try.append(quadratic_residue_seed(n_ref, 71, rng))
    # Mixed: 50% parent + perturbation
    h_pert = best_h + 0.1 * (rng.randn(n_ref) - 0.5)
    seeds_to_try.append(repair_feasible(h_pert))
    # Another perturbation
    h_pert2 = best_h + 0.2 * (rng.randn(n_ref) - 0.5)
    seeds_to_try.append(repair_feasible(h_pert2))

    for h_seed in seeds_to_try:
        if time_left() < 50:
            break
        h_try, c_try = active_descent_adaptive(h_seed, max_iters=200, tol0=0.005, tol_min=0.001)
        if c_try < best_c:
            best_h, best_c = h_try, c_try
            if time_left() > 60:
                h_n, c_n = nesterov_projected(best_h, max_iters=150, mu=100.0, alpha0=0.02)
                if c_n < best_c:
                    best_h, best_c = h_n, c_n

    # Phase 5: Dimension exploration
    n_cur = len(best_h)
    dims_to_try = []
    for dn in [-4, -2, 2, 4, 6, 8, 10, 12, -8, -6]:
        d = n_cur + dn
        if 40 <= d <= 140 and d % 2 == 0:
            dims_to_try.append(d)
    # Remove duplicates and sort by proximity
    dims_to_try = sorted(set(dims_to_try), key=lambda x: abs(x - n_cur))

    for new_n in dims_to_try[:6]:
        if time_left() < 40:
            break
        h_interp = interp_to_n(best_h, new_n)
        h_opt, c_opt = active_descent_adaptive(h_interp, max_iters=250, tol0=0.004, tol_min=0.0005)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt
            if time_left() > 50:
                h_n, c_n = nesterov_projected(best_h, max_iters=150, mu=120.0, alpha0=0.02)
                if c_n < best_c:
                    best_h, best_c = h_n, c_n

    # Phase 6: SLSQP refinement
    if time_left() > 60:
        h_slsqp = slsqp_optimize(best_h, maxiter=200)
        if h_slsqp is not None:
            c_slsqp = exact_score(h_slsqp)
            if c_slsqp < best_c:
                best_h, best_c = h_slsqp, c_slsqp
                if time_left() > 40:
                    h_ad2, c_ad2 = active_descent_adaptive(best_h, max_iters=200, tol0=0.002, tol_min=0.0002)
                    if c_ad2 < best_c:
                        best_h, best_c = h_ad2, c_ad2

    # Phase 7: Nesterov with higher mu for fine-tuning
    if time_left() > 50:
        h_fin, c_fin = nesterov_projected(best_h, max_iters=300, mu=300.0, alpha0=0.01)
        if c_fin < best_c:
            best_h, best_c = h_fin, c_fin

    # Phase 8: Coordinate polish
    if time_left() > 30:
        h_cp, c_cp = coordinate_polish(best_h, max_iters=50)
        if c_cp < best_c:
            best_h, best_c = h_cp, c_cp

    # Phase 9: Final tight active descent
    if time_left() > 25:
        h_tight, c_tight = active_descent_adaptive(best_h, max_iters=200, tol0=0.001, tol_min=0.0001)
        if c_tight < best_c:
            best_h, best_c = h_tight, c_tight

    # Phase 10: One more SLSQP if time allows
    if time_left() > 30:
        h_s2 = slsqp_optimize(best_h, maxiter=150)
        if h_s2 is not None:
            c_s2 = exact_score(h_s2)
            if c_s2 < best_c:
                best_h, best_c = h_s2, c_s2

    # Final: tightest active descent
    if time_left() > 15:
        h_last, c_last = active_descent_adaptive(best_h, max_iters=150, tol0=0.0005, tol_min=0.0001)
        if c_last < best_c:
            best_h, best_c = h_last, c_last

    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)

    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            final_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 2:
            final_c = float(eval_result[0])
    except Exception:
        pass

    return best_h, final_c, len(best_h)
