def logsumexp_objective(h, beta):
    n = len(h)
    P = profile_h(h)
    Pmax = P.max()
    if Pmax < 1e-30:
        return 0.0, np.zeros(n)
    log_sum = Pmax + np.log(np.sum(np.exp(beta * (P - Pmax))))
    f = log_sum / beta
    # Gradient
    weights = np.exp(beta * (P - Pmax))
    weights /= weights.sum()
    active = np.where(weights > 0.01 / len(P))[0]
    if len(active) == 0:
        active = np.argsort(weights)[-5:]
    dx = 2.0 / n
    G = build_gradient_rows(h, active, n, dx)
    w = weights[active]
    w = w / w.sum()
    grad = G.T @ w
    return f, grad
