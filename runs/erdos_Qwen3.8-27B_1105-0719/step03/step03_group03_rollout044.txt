def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))


def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx


def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx])
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    return h


def build_G(h, dx):
    n = len(h)
    M = 2 * n - 1
    G = np.zeros((M, n))
    center = n - 1
    G[center, :] = dx * (1.0 - 2.0 * h)
    for d in range(1, n):
        idx = np.arange(n - d)
        m_pos = center + d
        G[m_pos, idx + d] += dx * (1.0 - h[idx])
        G[m_pos, idx] += dx * (-h[idx + d])
        m_neg = center - d
        G[m_neg, idx] += dx * (1.0 - h[idx + d])
        G[m_neg, idx + d] += dx * (-h[idx])
    return G


def coord_min(G_col, b):
    """Minimize max_j(G_col[j]*v + b[j]) over v in [0,1]"""
    g0 = np.max(b)
    g1 = np.max(G_col + b)
    
    if g0 <= g1:
        active0 = G_col[b >= g0 - 1e-14]
        if len(active0) > 0 and active0.min() >= -1e-14:
            return 0.0
    else:
        active1 = G_col[G_col + b >= g1 - 1e-14]
        if len(active1) > 0 and active1.max() <= 1e-14:
            return 1.0
    
    # Bisection on subgradient
    lo, hi = 0.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        gm = np.max(G_col * mid + b)
        active = G_col[np.abs(G_col * mid + b - gm) <= 1e-12]
        if active.size == 0:
            # numerical issue, use tolerance
            max_val = np.max(G_col * mid + b)
            active = G_col[np.abs(G_col * mid + b - max_val) <= 1e-10]
        if active.size == 0:
            return mid
        if active.min() > 1e-14:
            hi = mid
        elif active.max() < -1e-14:
            lo = mid
        else:
            return mid
    return (lo + hi) / 2.0


def cd_sweep(h, dx, n, profile, G, order):
    """One sweep of coordinate descent, updating profile in place."""
    improved = False
    for i in order:
        G_col = G[:, i]
        b = profile - G_col * h[i]
        v = coord_min(G_col, b)
        if abs(v - h[i]) > 1e-12:
            h[i] = v
            profile = compute_profile(h)
            improved = True
    return improved
