def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        # Need to add mass: add to positions with most room
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for idx in order:
            add = min(remaining, capacity[idx])
            h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        # Need to remove mass: remove from positions with most mass
        order = np.argsort(h)
        remaining = -diff
        for idx in order:
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    # Final correction
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))

Let me think about one more thing: the golden section search. The function we're minimizing is:
f(t) = max_d P_d(t)

where P_d(t) is either linear or a convex quadratic in t. The maximum of convex functions is convex, so f is convex on the feasible interval. def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0).copy()
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for idx in order:
            add = min(remaining, capacity[idx])
            h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for idx in order:
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
    return np.clip(h, 0.0, 1.0)

def _pair_eval(h, i, j, a, b, t):
    """Evaluate max profile with h[i]=a+t, h[j]=b-t."""
    n = len(h)

    h[i] = a + t
    h[j] = b - t

    h[j] = b
    return val

def pair_minimize(h, i, j, max_iter=45):
    """Minimize max profile over t: h[i]=a+t, h[j]=b-t."""
    a, b = h[i], h[j]
    t_lo = max(-a, b - 1.0)
    t_hi = min(1.0 - a, b)
    if t_hi - t_lo < 1e-14:
        return h, exact_score(h)
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = t_lo, t_hi
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = _pair_eval(h, i, j, a, b, c)
    fd = _pair_eval(h, i, j, a, b, d)
    
    for _ in range(max_iter):
        if fc < fd:
            hi = d; d = c; fd = fc
            c = hi - gr * (hi - lo)
            fc = _pair_eval(h, i, j, a, b, c)
        else:
            lo = c; c = d; fc = fd
            d = lo + gr * (hi - lo)
            fd = _pair_eval(h, i, j, a, b, d)
    
    t_opt = (lo + hi) / 2.0
    val_opt = _pair_eval(h, i, j, a, b, t_opt)
    current = exact_score(h)
    if val_opt < current - 1e-15:
        h[i] = a + t_opt
        h[j] = b - t_opt
        return h, val_opt
    return h, current

        return x[-1]

        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g

        return x[:-1].sum() - n / 2.0

        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g

        t = x[-1]
        prof = profile_h(h)
        return t - prof[active]

        dx_local = 2.0 / len(h)
        # Build Jacobian for active lags
        J_prof = np.zeros((len(active), n))

            d = m - (n - 1)
            if d == 0:
                J_prof[idx, :] = dx_local * (1.0 - 2.0 * h)
            elif d > 0:

        J[:, -1] = 1.0
        return J
    
    x0 = np.concatenate([h0, [c0 + 1e-6]])

            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
    return np.clip(h, 0.0, 1.0)

def _pair_eval(h, i, j, a, b, t):
    n = len(h)
    dx = 2.0 / n
    h[i] = a + t
    h[j] = b - t
    val = float(np.max(np.correlate(h, 1 - h, mode="full"))) * dx
    h[i] = a
    h[j] = b
    return val

def pair_minimize(h, i, j, max_iter=45):
    a, b = h[i], h[j]
    t_lo = max(-a, b - 1.0)
    t_hi = min(1.0 - a, b)
    if t_hi - t_lo < 1e-14:
        return h, exact_score(h)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = t_lo, t_hi
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = _pair_eval(h, i, j, a, b, c)
    fd = _pair_eval(h, i, j, a, b, d)
    for _ in range(max_iter):
        if fc < fd:
            hi = d; d = c; fd = fc
            c = hi - gr * (hi - lo)
            fc = _pair_eval(h, i, j, a, b, c)
        else:
            lo = c; c = d; fc = fd
            d = lo + gr * (hi - lo)
            fd = _pair_eval(h, i, j, a, b, d)
    t_opt = (lo + hi) / 2.0
    val_opt = _pair_eval(h, i, j, a, b, t_opt)
    current = exact_score(h)
    if val_opt < current - 1e-15:
        h[i] = a + t_opt
        h[j] = b - t_opt
        return h, val_opt
    return h, current

def pairwise_optimize(h, n_sweeps=30, t_start=None, budget_s=1100, margin=30, rng=None):
    n = len(h)
    best_c = exact_score(h)
    for sweep in range(n_sweeps):
        if t_start is not None and time.perf_counter() - t_start > budget_s - margin:
            break
        improved = False
        if rng is not None and sweep > 0:
            order_i = list(range(n))
            rng.shuffle(order_i)
        else:
            order_i = list(range(n))
        for ii in order_i:
            i = order_i[ii]
            if t_start is not None and time.perf_counter() - t_start > budget_s - margin:
                break
            for j in range(i + 1, n):
                h, c = pair_minimize(h, i, j)
                if c < best_c - 1e-15:
                    best_c = c
                    improved = True
                if t_start is not None and time.perf_counter() - t_start > budget_s - margin:
                    break
        if not improved:
            break
    return h, best_c

def build_grad_rows(h, rows, n, dx):
    G = np.zeros((len(rows), n))
    for idx, m in enumerate(rows):
        d = m - (n - 1)
        if d == 0:
            G[idx, :] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            for i in range(n - d):
                G[idx, i + d] += dx * (1.0 - h[i])
                G[idx, i] -= dx * h[i + d]
        else:
            dp = -d
            for j in range(n - dp):
                G[idx, j + dp] += dx * (1.0 - h[j])
                G[idx, j] -= dx * h[j + dp]
    return G

def slsqp_active(h0, maxiter=300, active_k=12, tol=1e-12):
    n = len(h0)
    dx = 2.0 / n
    P = profile_h(h0)
    c0 = P.max()
    thresh = c0 - max(0.005, 0.1 * (c0 - np.median(P)))
    active = np.where(P >= thresh)[0]
    if len(active) == 0:
        active = np.array([np.argmax(P)])
    if len(active) > active_k:
        top_idx = np.argsort(P[active])[-active_k:]
        active = active[top_idx]

    def obj(x):
        return x[-1]

    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g

    def eq_fun(x):
        return x[:-1].sum() - n / 2.0

    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g

    def ineq_fun(x):
        h = x[:-1]
        t = x[-1]
        prof = profile_h(h)
        return t - prof[active]

    def ineq_jac(x):
        h = x[:-1]
        nloc = len(h)
        dxloc = 2.0 / nloc
        G = build_grad_rows(h, active, nloc, dxloc)
        J = np.zeros((len(active), nloc + 1))
        J[:, :-1] = -G
        J[:, -1] = 1.0
        return J

    x0 = np.concatenate([h0, [c0 + 1e-6]])
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 1.0)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': tol})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = repair_feasible(res.x[:-1])
            c_new = exact_score(h_new)
            if c_new <= c0 + 1e-10:
                return h_new
    except Exception:
        pass
    return None

def slsqp_all(h0, maxiter=500, tol=1e-12):
    n = len(h0)
    M = 2 * n - 1
    P = profile_h(h0)
    c0 = P.max()

    def obj(x):
        return x[-1]

    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g

    def eq_fun(x):
        return x[:-1].sum() - n / 2.0

    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g

    def ineq_fun(x):
        h = x[:-1]
        t = x[-1]
        prof = profile_h(h)
        return t - prof

    def ineq_jac(x):
        h = x[:-1]
        nloc = len(h)
        dxloc = 2.0 / nloc
        Mloc = 2 * nloc - 1
        G = build_grad_rows(h, np.arange(Mloc), nloc, dxloc)
        J = np.zeros((Mloc, nloc + 1))
        J[:, :-1] = -G
        J[:, -1] = 1.0
        return J

    x0 = np.concatenate([h0, [c0 + 1e-6]])
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 1.0)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': tol})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = repair_feasible(res.x[:-1])
            c_new = exact_score(h_new)
            if c_new <= c0 + 1e-10:
                return h_new
    except Exception:
        pass
    return None

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    return repair_feasible(np.clip(h_new, 0.0, 1.0))

def symmetrize(h):
    return (h + h[::-1].copy()) / 2.0

def softmax_descent(h, max_iters=80, t_start=None, budget_s=1100, margin=30):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best_c = exact_score(h)
    vel = np.zeros(n)
    momentum = 0.4
    betas = [30, 60, 120, 250, 500]
    iters_per_beta = max(8, max_iters // len(betas))
    for beta in betas:
        alpha = 0.015
        for it in range(iters_per_beta):
            if t_start is not None and time.perf_counter() - t_start > budget_s - margin:
                return h, best_c
            P = profile_h(h)
            Pmax = P.max()
            exp_arg = beta * (P - Pmax)
            exp_arg -= exp_arg.max()
            weights = np.exp(exp_arg)
            S = weights.sum()
            if S < 1e-300:
                weights = np.zeros(len(P))
                weights[np.argmax(P)] = 1.0
            else:
                weights /= S
            active = np.where(weights > 0.02)[0]
            if len(active) == 0:
                active = np.argsort(weights)[-5:]
            G = build_grad_rows(h, active, n, dx)
            w_a = weights[active]
            w_a = w_a / w_a.sum()
            grad = G.T @ w_a
            sum_err = h.sum() - n / 2.0
            grad += 2.0 * 40.0 * sum_err / n
            grad -= grad.mean()
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break
            grad /= gnorm
            vel = momentum * vel + grad
            vnorm = np.linalg.norm(vel)
            if vnorm > 0:
                vel /= vnorm
            improved = False
            base_c = exact_score(h)
            for a in [alpha, alpha * 0.5, alpha * 0.25, alpha * 0.1]:
                h_try = repair_feasible(h - a * vel)
                s = exact_score(h_try)
                if s < best_c - 1e-14:
                    h = h_try
                    best_c = s
                    improved = True
                    break
                elif s < base_c:
                    h = h_try
                    if exact_score(h) < best_c:
                        best_c = exact_score(h)
                    improved = True
            if improved:
                alpha = min(alpha * 1.05, 0.06)
            else:
                alpha *= 0.6
                if alpha < 1e-8:
                    break
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()
    rng = np.random.RandomState(seed)

    def time_ok(margin=20):
        return time.perf_counter() - t_start < budget_s - margin

    # Get parent
    parent = None
    try:
        parent = np.array(initial_h_values, dtype=np.float64)
    except (NameError, Exception):
        pass

    if parent is None or len(parent) < 4:
        parent = np.full(70, 0.5)

    parent = np.clip(parent, 0.0, 1.0)
    parent = repair_feasible(parent)

    best_h = parent.copy()
    best_c = exact_score(parent)

    # Symmetrization
    if time_ok():
        h_sym = repair_feasible(symmetrize(parent))
        c_sym = exact_score(h_sym)
        if c_sym < best_c - 1e-13:
            best_h = h_sym
            best_c = c_sym

    # Phase 1: Intensive pairwise coordinate descent
    if time_ok(80):
        best_h, best_c = pairwise_optimize(best_h, n_sweeps=50,
                                           t_start=t_start, budget_s=budget_s, margin=80, rng=rng)

    # Phase 2: SLSQP active-set polish
    if time_ok(40):
        sol = slsqp_active(best_h, maxiter=350, active_k=14, tol=1e-12)
        if sol is not None:
            c = exact_score(sol)
            if c < best_c - 1e-14:
                best_h = sol
                best_c = c
    if time_ok(30):
        sol = slsqp_active(best_h, maxiter=250, active_k=10, tol=1e-12)
        if sol is not None:
            c = exact_score(sol)
            if c < best_c - 1e-14:
                best_h = sol
                best_c = c

    # Phase 3: More pairwise after SLSQP
    if time_ok(120):
        best_h, best_c = pairwise_optimize(best_h, n_sweeps=40,
                                           t_start=t_start, budget_s=budget_s, margin=60, rng=rng)

    # Phase 4: Softmax descent for global moves
    if time_ok(80):
        h_md, c_md = softmax_descent(best_h, max_iters=60,
                                     t_start=t_start, budget_s=budget_s, margin=40)
        if c_md < best_c - 1e-14:
            best_h = h_md
            best_c = c_md
        if time_ok(40):
            best_h, best_c = pairwise_optimize(best_h, n_sweeps=20,
                                               t_start=t_start, budget_s=budget_s, margin=40, rng=rng)

    # Phase 5: Dimension exploration
    if time_ok(150):
        n_best = len(best_h)
        cand_dims = []
        for d in range(max(24, n_best - 15), min(140, n_best + 20)):
            if d != n_best:
                cand_dims.append(d)
        for new_n in cand_dims[:14]:
            if not time_ok(25):
                break
            h_interp = interpolate_to_n(best_h, new_n)
            h_opt, c_opt = pairwise_optimize(h_interp, n_sweeps=12,
                                             t_start=t_start, budget_s=budget_s, margin=25, rng=rng)
            if c_opt < best_c - 1e-14:
                best_h = h_opt
                best_c = c_opt
            elif time_ok(15):
                sol = slsqp_active(h_interp, maxiter=200, active_k=12, tol=1e-11)
                if sol is not None:
                    c = exact_score(sol)
                    if c < best_c - 1e-14:
                        best_h = sol
                        best_c = c

    # Phase 6: Random restarts
    if time_ok(200):
        n0 = len(best_h)
        for trial in range(6):
            if not time_ok(30):
                break
            z = rng.randn(n0)
            z -= z.mean()
            z /= (np.linalg.norm(z) + 1e-10)
            for eps in [0.03, 0.07, 0.12]:
                if not time_ok(15):
                    break
                h_start = repair_feasible(best_h + eps * z)
                h_sym2 = repair_feasible(symmetrize(h_start))
                h_opt, c_opt = pairwise_optimize(h_sym2, n_sweeps=15,
                                                 t_start=t_start, budget_s=budget_s, margin=15, rng=rng)
                if c_opt < best_c - 1e-14:
                    best_h = h_opt
                    best_c = c_opt
                    break

    # Phase 7: Final polish
    if time_ok(60):
        sol = slsqp_active(best_h, maxiter=400, active_k=16, tol=1e-13)
        if sol is not None:
            c = exact_score(sol)
            if c < best_c - 1e-14:
                best_h = sol
                best_c = c
    if time_ok(40):
        best_h, best_c = pairwise_optimize(best_h, n_sweeps=30,
                                           t_start=t_start, budget_s=budget_s, margin=20, rng=rng)
    if time_ok(25):
        sol = slsqp_all(best_h, maxiter=300, tol=1e-13)
        if sol is not None:
            c = exact_score(sol)
            if c < best_c - 1e-14:
                best_h = sol
                best_c = c
    if time_ok(15):
        best_h, best_c = pairwise_optimize(best_h, n_sweeps=15,
                                           t_start=t_start, budget_s=budget_s, margin=10, rng=rng)

    # Final validation
    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)

    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            final_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 1:
            if np.isfinite(eval_result[0]):
                final_c = float(eval_result[0])
    except Exception:
        pass

    return best_h, final_c, len(best_h)
