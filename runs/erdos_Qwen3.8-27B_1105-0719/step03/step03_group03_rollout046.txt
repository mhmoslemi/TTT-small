        tol = max(feasibility_tol, 0.15 * (c_max - med))
        active = np.where(profile >= c_max - tol)[0]
        if len(active) > 20:
            active = active[np.argsort(profile[active])[-20:]]
        grad = subgradient_step(h, active, n)
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        step = step0 / (1.0 + 0.05 * it)
        h_new = project_polytope(h - step * grad / gn * np.max(profile), S)
        c_new = exact_score(h_new)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new
        profile = compute_profile(h)
    return best_h, best_c

def coordinate_search(h0, n_rounds, n_trials, seed=99):
    n = len(h0)
    S = n / 2.0
    h = h0.copy()
    best_c = exact_score(h)
    rng = np.random.RandomState(seed)
    deltas = [0.01, 0.02, 0.03, 0.05, 0.08, 0.12, -0.01, -0.02, -0.03, -0.05, -0.08, -0.12]
    for r in range(n_rounds):
        improved = False
        for _ in range(n_trials):
            i, j = rng.choice(n, 2, replace=False)
            for delta in deltas:
                if h[i] + delta < -1e-10 or h[i] + delta > 1.0 + 1e-10:
                    continue
                if h[j] - delta < -1e-10 or h[j] - delta > 1.0 + 1e-10:
                    continue
                h[i] += delta
                h[j] -= delta
                c_try = exact_score(h)
                if c_try < best_c - 1e-14:
                    best_c = c_try
                    improved = True
                    break
                h[i] -= delta
                h[j] += delta
        if not improved:
            if r >= 3:
                break
    return h, best_c

def nesterov_projected(h0, beta, n_iter, eta0, budget_fn):
    n = len(h0)
    S = n / 2.0
    h = h0.copy()
    h_prev = h0.copy()
    best_h = h0.copy()
    best_c = exact_score(h0)
    t = 1.0
    for it in range(n_iter):
        if budget_fn() < 15:
            break
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = h + (t - 1.0) / t_next * (h - h_prev)
        loss, grad = smoothed_loss_grad(y, beta)
        eta = eta0
        for _ in range(15):
            x_new = project_polytope(y - eta * grad, S)
            loss_new, _ = smoothed_loss_grad(x_new, beta)
            if loss_new <= loss - 0.5 * eta * eta * np.linalg.norm(grad) ** 2:
                break
            eta *= 0.5
        x_new = project_polytope(y - eta * grad, S)
        c = exact_score(x_new)
        if c < best_c:
            best_c = c
            best_h = x_new.copy()
        h_prev = h
        h = x_new
        t = t_next
    return best_h, best_c

def interp_h(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    fx = np.concatenate([[0.0], xs, [2.0]])
    fh = np.concatenate([[h[0]], h, [h[-1]]])
    hn = np.interp(xt, fx, fh)
    return project_polytope(np.clip(hn, 0.0, 1.0), new_n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()
    rng = np.random.RandomState(seed)

    def tl():
        return budget_s + t_start - time.perf_counter()

    parent = None
    try:
        parent = np.array(initial_h_values, dtype=np.float64)
    except (NameError, Exception):
        pass
    if parent is None or len(parent) < 4:
        parent = np.full(60, 0.5)
    parent = project_polytope(np.clip(parent, 0.0, 1.0), len(parent) / 2.0)

    best_h = parent.copy()
    best_c = exact_score(parent)

    h_sym = project_polytope((parent + parent[::-1]) / 2.0, len(parent) / 2.0)
    c_sym = exact_score(h_sym)
    if c_sym < best_c - 1e-14:
        best_h, best_c = h_sym, c_sym

    h_comp = project_polytope(1.0 - best_h, len(best_h) / 2.0)
    c_comp = exact_score(h_comp)
    if c_comp < best_c - 1e-14:
        best_h, best_c = h_comp, c_comp

    # Phase 1: Multi-resolution SLSQP with smoothing continuation
    if tl() > 200:
        n0 = len(best_h)
        dims = sorted(set([n0 - 4, n0, n0 + 4, n0 + 8, n0 - 8]),
                     key=lambda d: abs(d - n0))
        dims = [d for d in dims if d >= 30]
        beta_schedule = [15, 30, 60, 120, 250, 500, 1000]
        for nd in dims[:5]:
            if tl() < 40:
                break
            hi = interp_h(best_h, nd)
            ci = exact_score(hi)
            for beta in beta_schedule:
                if tl() < 30:
                    break
                maxit = max(200, min(800, int(120000 / nd)))
                hs, cs = slsqp_smoothed(hi, beta, maxit, 1e-14)
                if hs is not None and cs < ci - 1e-14:
                    hi, ci = hs, cs
            if ci < best_c - 1e-14:
                best_h, best_c = hi, ci

    # Phase 2: Nesterov projected gradient with continuation
    if tl() > 250:
        n0 = len(best_h)
        h_cur = best_h.copy()
        c_cur = best_c
        for beta in [50, 100, 200, 400, 800, 1500, 3000]:
            if tl() < 40:
                break
            n_iter = min(300, max(50, int(20000 / n0)))
            hs, cs = nesterov_projected(h_cur, beta, n_iter, 1.0 / beta, tl)
            if cs < c_cur - 1e-14:
                h_cur, c_cur = hs, cs
        if c_cur < best_c - 1e-14:
            best_h, best_c = h_cur, c_cur

    # Phase 3: SLSQP refinement with high beta
    if tl() > 200:
        n0 = len(best_h)
        for beta in [500, 1000, 2000, 4000]:
            if tl() < 30:
                break
            maxit = max(300, min(1000, int(100000 / n0)))
            hs, cs = slsqp_smoothed(best_h, beta, maxit, 1e-15)
            if hs is not None and cs < best_c - 1e-14:
                best_h, best_c = hs, cs

    # Phase 4: Subgradient refinement
    if tl() > 150:
        n0 = len(best_h)
        h_sub = best_h.copy()
        c_sub = best_c
        step0 = 0.02
        for _ in range(3):
            if tl() < 30:
                break
            hs, cs = subgradient_refine(h_sub, 500, step0, feasibility_tol=0.003)
            if cs < c_sub - 1e-14:
                h_sub, c_sub = hs, cs
            step0 *= 0.7
        if c_sub < best_c - 1e-14:
            best_h, best_c = h_sub, c_sub

    # Phase 5: Coordinate local search
    if tl() > 120:
        n0 = len(best_h)
        n_trials = max(200, min(2000, int(500000 / (n0 * n0))))
        h_coord, c_coord = coordinate_search(best_h, n_rounds=20, n_trials=n_trials, seed=seed)
        if c_coord < best_c - 1e-14:
            best_h, best_c = h_coord, c_coord

    # Phase 6: Try nearby resolutions with full pipeline
    if tl() > 150:
        n0 = len(best_h)
        for nd in [n0 - 2, n0 + 2, n0 - 6, n0 + 6]:
            if tl() < 30:
                break
            if nd < 30:
                continue
            hi = interp_h(best_h, nd)
            for beta in [100, 300, 1000, 3000]:
                if tl() < 20:
                    break
                hs, cs = slsqp_smoothed(hi, beta, 400, 1e-14)
                if hs is not None and cs < exact_score(hi):
                    hi = hs
            ci = exact_score(hi)
            if ci < best_c - 1e-14:
                best_h, best_c = hi, ci

    # Phase 7: Final SLSQP polish
    if tl() > 60:
        n0 = len(best_h)
        for beta in [2000, 5000]:
            if tl() < 30:
                break
            hs, cs = slsqp_smoothed(best_h, beta, 600, 1e-15)
            if hs is not None and cs < best_c - 1e-14:
                best_h, best_c = hs, cs

    # Phase 8: Final subgradient micro-steps
    if tl() > 40:
        hs, cs = subgradient_refine(best_h, 300, 0.005, feasibility_tol=0.002)
        if cs < best_c - 1e-14:
            best_h, best_c = hs, cs

    # Phase 9: Final coordinate search
    if tl() > 30:
        n0 = len(best_h)
        n_trials = max(100, min(500, int(200000 / (n0 * n0))))
        hc, cc = coordinate_search(best_h, n_rounds=10, n_trials=n_trials, seed=seed + 1)
        if cc < best_c - 1e-14:
            best_h, best_c = hc, cc

    # Final validation
    best_h = project_polytope(best_h, len(best_h) / 2.0)
    final_c = exact_score(best_h)

    try:
        er = evaluate_erdos_solution(best_h)
        if isinstance(er, (float, int)):
            final_c = min(final_c, float(er))
        elif isinstance(er, tuple) and len(er) >= 2:
            final_c = min(final_c, float(er[0]))
    except Exception:
        pass

    return best_h, final_c, len(best_h)
