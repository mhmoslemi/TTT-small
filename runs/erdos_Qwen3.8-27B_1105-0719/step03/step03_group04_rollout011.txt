def qp_trust_region(h, active, profile, G, radius, gamma):
    """Solve QP trust region subproblem using cvxpy."""
    import cvxpy as cp
    n = len(h)
    M = len(active)
    G_a = G[active, :]
    p_a = profile[active]
    
    delta = cp.Variable(n)
    r = cp.Variable()
    
    # Objective: minimize r + gamma * ||delta||^2
    # Constraint: p_a + G_a @ delta <= r
    # sum(delta) == 0
    # -h <= delta <= 1-h (box constraints)
    # Also |delta_i| <= radius (trust region)
    
    objective = cp.Minimize(cp.sum(r) / M + gamma * cp.sum_squares(delta))
    # Wait, I want to minimize the maximum constraint violation
    # Better: minimize r + gamma * ||delta||^2 s.t. p_a + G_a*delta <= r
    
    objective = cp.Minimize(r + gamma * cp.sum_squares(delta))
    constraints = [
        G_a @ delta <= r - p_a,
        cp.sum(delta) == 0,
        delta >= -np.minimum(h, radius),
        delta <= np.minimum(1.0 - h, radius),
        r >= 0
    ]
    
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.SCS, max_iters=10000, eps=1e-8)
        if prob.status == 'optimal' and delta.value is not None:
            return np.array(delta.value).flatten(), r.value
    except Exception:
        pass
    
    # Fallback to ECOS
    try:
        prob.solve(solver=cp.ECOS, max_iters=10000, abstol=1e-9, reltol=1e-9)
        if prob.status == 'optimal' and delta.value is not None:
            return np.array(delta.value).flatten(), r.value
    except Exception:
        pass
    
    # Fallback to CLARABEL
    try:
        prob.solve(solver=cp.CLARABEL)
        if prob.status in ('optimal', 'optimal_inaccurate') and delta.value is not None:
            return np.array(delta.value).flatten(), r.value
    except Exception:
        pass
    
    return None, None

def lp_fallback(h, active, profile, G, radius):
    """LP fallback when QP fails."""
    n = len(h)
    dx = 2.0 / n
    M = len(active)
    G_a = G[active, :]
    p_a = profile[active]
    
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    A_ub = np.zeros((M, n + 1))
    A_ub[:, :n] = G_a
    A_ub[:, -1] = -1.0
    b_ub = -p_a
    
    A_eq = np.ones((1, n + 1))
    b_eq = np.zeros(1)
    
    lower = -np.minimum(h, radius)
    upper = np.minimum(1.0 - h, radius)
    bounds = list(zip(lower, upper))
    bounds.append((0.0, None))
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'maxiter': 5000})
        if res.success and res.x is not None and np.isfinite(res.x).all():
            return res.x[:n], res.x[-1]
    except Exception:
        pass
    return None, None

def gradient_step(h, active, profile, G, radius):
    """Compute weighted gradient direction."""
    n = len(h)
    G_a = G[active, :]
    p_a = profile[active]
    T = np.max(p_a)
    # Weight by how close to max
    weights = np.maximum(p_a - (T - 0.1 * (T - p_a.min()) + 1e-12), 0.0)
    if weights.sum() < 1e-15:
        weights = np.ones(len(active)) / len(active)
    else:
        weights /= weights.sum()
    
    grad = G_a.T @ weights
    grad -= grad.mean() / n  # Project to sum=0 subspace
    norm = np.linalg.norm(grad)
    if norm < 1e-15:
        return None
    return grad / norm

def line_search(h0, direction, current_score, max_step=0.1, n_steps=10):
    n = len(h0)
    best_h = h0.copy()
    best_s = current_score
    # Geometric sequence of step sizes
    for i in range(n_steps):
        a = max_step * (0.7 ** i)
        h_try = repair_feasible(h0 - a * direction)
        s = exact_score(h_try)
        if s < best_s - 1e-14:
            best_h = h_try
            best_s = s
    return best_h, best_s

def symmetrize(h):

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    h_new = np.clip(h_new, 0.0, 1.0)

        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h
