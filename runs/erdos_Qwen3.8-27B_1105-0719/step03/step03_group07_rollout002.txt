import numpy as np
from scipy.optimize import minimize, linprog
import cvxpy as cp
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full"))) * dx

def profile_h(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx])
            h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def build_gradient_rows(h, rows, n, dx):
    G = np.zeros((len(rows), n))
    for idx, m in enumerate(rows):
        d = m - (n - 1)
        if d == 0:
            G[idx, :] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            for i in range(n - d):
                G[idx, i + d] += dx * (1.0 - h[i])
                G[idx, i] -= dx * h[i + d]
        else:
            dp = -d
            for j in range(n - dp):
                G[idx, j + dp] += dx * (1.0 - h[j])
                G[idx, j] -= dx * h[j + dp]
    return G

def lp_descent_step(h, active, step_size=0.05, solver='cvxpy'):
    """Find LP direction to minimize max linearized active constraint values."""
    n = len(h)
    dx = 2.0 / n
    P = profile_h(h)
    G = build_gradient_rows(h, active, n, dx)
    current_vals = P[active]
    c_max = P.max()
    
    if solver == 'cvxpy':
        d = cp.Variable(n)
        t = cp.Variable()
        constraints = [
            G @ d + current_vals <= t,
            np.ones(n) @ d == 0,
            d >= -step_size,
            d <= step_size
        ]
        prob = cp.Problem(cp.Minimize(t), constraints)
        try:
            prob.solve(solver=cp.ECOS, max_iter=10000)
            if d.value is not None and t.value < c_max - 1e-10:
                return d.value, t.value
        except Exception:
            pass
    elif solver == 'scipy':
        # Minimize t subject to G@w <= t - current_vals, sum(d)=0, -s <= d <= s
        # Variables: d[0..n-1], t
        # c = [0, ..., 0, 1]
        # A_ub: rows for G@w <= t - current_vals => G@w - t <= -current_vals
        # A_eq: [ones(n), 0] @ [d, t] = 0
        # bounds: d in [-s, s], t in [0, inf)
        m = len(active)
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((m, n + 1))
        A_ub[:, :n] = G
        A_ub[:, -1] = -1.0
        b_ub = -current_vals
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = [0.0]
        bounds = [(-step_size, step_size)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.fun < c_max - 1e-10:
            return res.x[:n], res.x[-1]
    return None, c_max

def active_set_leverage(h, k=20, band=0.005):
    """Find active constraints: top k by value or within band of max."""
    P = profile_h(h)
    c_max = P.max()
    # Include all within band of max
    active = np.where(P >= c_max - band)[0]
    # Also include top k
    top_k = np.argsort(P)[-k:]
    active = np.unique(np.concatenate([active, top_k]))
    return active

def lp_descent_loop(h, max_iters=50, step_size=0.05, band=0.005, k=20,
                    t_start=None, budget_s=1100, margin=30):
    """Iterative LP-based descent."""
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    step = step_size

            break
        P = profile_h(h)
        c_max = P.max()

            best_c = c_max if c_max < best_c else best_c
            if c_max > best_c:
                h = repair_feasible(h)
                best_c = exact_score(h)
        active = active_set_leverage(h, k=k, band=band)
        d, t_val = lp_descent_step(h, active, step_size=step)
        if d is None:
            step *= 0.7
            if step < 0.001:
                break
            continue
        # Line search
        improved = False
        for frac in [1.0, 0.7, 0.5, 0.3, 0.1]:

                break

def slsqp_active(h0, maxiter=300, active_k=15, tol=1e-12):
    n = len(h0)
    dx = 2.0 / n
    M = 2 * n - 1
    P = profile_h(h0)
    c0 = P.max()
    
    thresh = c0 - max(0.003, 0.05 * (c0 - np.median(P)))

    if len(active) == 0:
        active = np.array([np.argmax(P)])
    if len(active) > active_k:
        top_idx = np.argsort(P[active])[-active_k:]
        active = active[top_idx]
    
    def obj(x):
        return x[-1]
    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    def eq_fun(x):
        return x[:-1].sum() - n / 2.0
    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g
    def ineq_fun(x):
        h = x[:-1]
        t = x[-1]
        prof = profile_h(h)
        return t - prof[active]
    def ineq_jac(x):
        h = x[:-1]

        J = np.zeros((len(active), n + 1))
        J[:, :-1] = -G
        J[:, -1] = 1.0
        return J
    
    x0 = np.concatenate([h0, [c0 + 1e-4]])
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 1.0)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': tol})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = repair_feasible(res.x[:-1])
            c_new = exact_score(h_new)

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    return repair_feasible(np.clip(h_new, 0.0, 1.0))

def symmetrize(h):
    return (h + h[::-1].copy()) / 2.0
