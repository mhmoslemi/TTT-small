def repair_feasible(vec):

    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h

        order = np.argsort(-capacity)
        remaining = diff

            idx = order[i]
            add = min(remaining, capacity[idx])
            if add > 0:

            remaining -= add
            if remaining <= 1e-12:

    else:

        remaining = -diff
        for i in range(n):

            rem = min(remaining, h[idx])
            if rem > 0:

            remaining -= rem
            if remaining <= 1e-12:

    if abs(diff2) > 1e-14:

        else:

    m = len(h)

    full[m:] = 1.0 - h[::-1]

    """Project h_old to complement symmetric subspace of dimension n_new."""
    m = n_new // 2

    else:

    # Project to anti-symmetric

    for j in range(m):
        y[j] = 0.5 * (h_interp[j] + 1.0 - h_interp[n_new-1-j])
    y = np.clip(y, 0.0, 1.0)

    n = len(h)
    if n == new_n:
        return h.copy()
    xs = np.linspace(0, 2, n + 1)
    xt = np.linspace(0, 2, new_n + 1)
    h_interp = np.zeros(new_n)
    for k in range(new_n):

        total_mass = 0.0

            overlap = min(hi_k, hi_j) - max(lo_k, lo_j)
            if overlap > 0:

                total_width += overlap

def line_search_c5(h, D, t_min, t_max, n_grid=200):

    P = c[None, :] + ts[:, None] * b[None, :] + (ts[None, :] ** 2) * a[None, :]
    scores = P.max(axis=1)
    best_idx = int(np.argmin(scores))
    best_t = ts[best_idx]
    best_score = float(scores[best_idx])

    if 0 < best_idx < n_grid - 1:
        ts2 = np.linspace(ts[best_idx-1], ts[best_idx+1], n_grid)

        scores2 = P2.max(axis=1)
        best_idx2 = int(np.argmin(scores2))
        best_t = ts2[best_idx2]
        best_score = float(scores2[best_idx2])
    
    return best_t, best_score

    t_max = 0.0
    if np.any(pos):

        t_max = min(t_max, min((1-y[i]) / d[i] for i in np.where(pos)[0]))
    if np.any(neg):

        t_max = min(t_max, min(-y[i] / d[i] for i in np.where(neg)[0]))

    t_min += 1e-12
    t_max -= 1e-12
    if t_max <= t_min:
        return 0.0, exact_score(build_full(y)), y.copy()

    h = build_full(y)
    D = np.zeros(2 * m)
    D[:m] = d
    D[m:] = -d[::-1]
    
    best_t, best_score = line_search_c5(h, D, t_min, t_max, n_grid)

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx])
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:

    else:

        remaining = -diff

    for sweep in range(200):
        if elapsed() > max_time - 30:
            break
        improved = False
        for j in range(m):
            d = np.zeros(m)
            d[j] = 1.0
            _, score, y_new = reduced_ls(y, d, n_grid)
            if score < best_score - 1e-13:
                y = y_new
                best_score = score
                improved = True
        if not improved:
            break

    # Phase 2: Pair descent
    pairs = [(i, j) for i in range(m) for j in range(i + 1, m)]
    for sweep in range(100):
        if elapsed() > max_time - 40:
            break
        improved = False
        rng.shuffle(pairs)
        for i, j in pairs:
            d = np.zeros(m)
            d[i] = 1.0
            d[j] = -1.0
            _, score, y_new = reduced_ls(y, d, n_grid)
            if score < best_score - 1e-13:
                y = y_new
                best_score = score
                improved = True
        if not improved:
            break

    # Phase 3: Gradient-based directions
    for it in range(200):
        if elapsed() > max_time - 50:
            break
        h = build_full(y)
        profile = compute_profile(h)
        g = reduced_gradient(h, profile, beta=80.0)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-14:
            break
        d = -g / gnorm
        _, score, y_new = reduced_ls(y, d, n_grid)
        if score < best_score - 1e-13:
            y = y_new
            best_score = score
        else:
            d = -g / gnorm + rng.randn(m) * 0.1
            d /= (np.linalg.norm(d) + 1e-15)
            _, score, y_new = reduced_ls(y, d, n_grid)
            if score < best_score - 1e-13:
                y = y_new
                best_score = score

    # Phase 4: More coordinate + pair (refined)
    for sweep in range(100):
        if elapsed() > max_time - 20:
            break
        improved = False
        for j in range(m):
            d = np.zeros(m)
            d[j] = 1.0
            _, score, y_new = reduced_ls(y, d, 500)
            if score < best_score - 1e-13:
                y = y_new
                best_score = score
                improved = True
        for i, j in pairs:
            if elapsed() > max_time - 20:
                break
            d = np.zeros(m)
            d[i] = 1.0
            d[j] = -1.0
            _, score, y_new = reduced_ls(y, d, 500)
            if score < best_score - 1e-13:
                y = y_new
                best_score = score
                improved = True
        if not improved:
            break

    return y, best_score

def slsqp_reduced(y0, maxiter=300):
    m = len(y0)
    n = 2 * m
    dx = 2.0 / n
    M = 2 * n - 1
    c0 = exact_score(build_full(y0))

    def obj(z):
        t = z[-1]
        return t

    def obj_jac(z):
        g = np.zeros(m + 1)
        g[-1] = 1.0
        return g

    def ineq_fun(z):
        y = np.clip(z[:m], 0, 1)
        h = build_full(y)
        t = z[-1]
        prof = compute_profile(h)
        return t - prof

    def ineq_jac(z):
        y = np.clip(z[:m], 0, 1)
        h = build_full(y)
        G = compute_grad_matrix(h)
        J = np.zeros((M, m + 1))
        J[:, :m] = -G[:, :m] + G[:, m:][::-1]
        J[:, m] = 1.0
        return J

    x0 = np.concatenate([y0, [c0 + 0.01]])
    bounds = [(0.0, 1.0)] * m + [(0.0, c0 + 0.5)]
    cons = [{'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}]
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None and np.isfinite(res.x).all():
            y_new = np.clip(res.x[:m], 0.0, 1.0)
            c_new = exact_score(build_full(y_new))
            if c_new <= c0 + 1e-9:
                return y_new, c_new
    except Exception:
        pass
    return None, None

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()
    rng = np.random.RandomState(seed)
    np.random.seed(seed)

    def time_left():
        return budget_s - (time.perf_counter() - t_start)

    parent = None
    try:
        parent = np.array(initial_h_values, dtype=np.float64)
    except NameError:
        pass
    except Exception:
        pass

    if parent is None or len(parent) < 4:
        parent = np.full(70, 0.5)

    parent = np.clip(parent, 0.0, 1.0)
    parent = repair_feasible(parent)

    best_h = parent.copy()
    best_c = exact_score(parent)

    n_parent = len(parent)

    # Try resolutions
    resolutions = sorted(set([
        n_parent,
        n_parent - 4, n_parent - 2, n_parent + 2, n_parent + 4,
        n_parent + 8, n_parent + 12, n_parent + 20,
    ]), key=lambda x: abs(x - n_parent))
    resolutions = [r for r in resolutions if r >= 40 and r <= 120 and r % 2 == 0]

    for n_res in resolutions:
        if time_left() < 60:
            break
        # Interpolate and project
        if n_res == n_parent:
            h_interp = parent.copy()
        else:
            h_interp = interpolate_to_n(parent, n_res)
        h_interp = repair_feasible(h_interp)
        y = project_to_anti(h_interp)

        # Allocate time based on resolution closeness to parent
        frac = 1.0 / (1.0 + 0.1 * abs(n_res - n_parent))
        alloc = min(0.35 * budget_s * frac, time_left() - 30)
        alloc = max(alloc, 40)

        y_opt, c_opt = optimize_anti(y, max_time=alloc, seed=seed + n_res)
        h_opt = build_full(y_opt)
        c_verify = exact_score(h_opt)
        if c_verify < best_c - 1e-14:
            best_c = c_verify
            best_h = h_opt.copy()

        # SLSQP polish
        if time_left() > 50:
            y_slq, c_slq = slsqp_reduced(y_opt, maxiter=200)
            if y_slq is not None and c_slq < best_c - 1e-14:
                h_slq = build_full(y_slq)
                best_c = c_slq
                best_h = h_slq.copy()
                y_opt = y_slq

        # Update interpolation source
        if best_h.shape[0] != n_parent:
            parent = interpolate_to_n(best_h, n_parent)
            parent = repair_feasible(parent)

    # Final: try symmetric and asymmetric variants of best
    n_best = len(best_h)
    h_sym = 0.5 * (best_h + best_h[::-1])
    h_sym = repair_feasible(h_sym)
    c_sym = exact_score(h_sym)
    if c_sym < best_c - 1e-14:
        best_c = c_sym
        best_h = h_sym

    # Final coordinate descent on the anti projection of best
    if time_left() > 40:
        y_best = project_to_anti(best_h)
        alloc = min(60, time_left() - 10)
        y_opt, c_opt = optimize_anti(y_best, max_time=alloc, seed=seed + 999)
        h_opt = build_full(y_opt)
        c_v = exact_score(h_opt)
        if c_v < best_c - 1e-14:
            best_c = c_v
            best_h = h_opt

    # SLSQP final polish
    if time_left() > 30:
        y_best = project_to_anti(best_h)
        y_slq, c_slq = slsqp_reduced(y_best, maxiter=300)
        if y_slq is not None and c_slq < best_c - 1e-14:
            h_slq = build_full(y_slq)
            best_c = c_slq
            best_h = h_slq

    # Final validation
    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)

    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            final_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 2:
            final_c = float(eval_result[0])
    except Exception:
        pass

    return best_h, final_c, len(best_h)
