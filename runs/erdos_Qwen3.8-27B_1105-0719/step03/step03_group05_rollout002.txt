import numpy as np
from scipy.optimize import minimize, linprog
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        # Need to add mass. Add to positions with most room.
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx])
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        # Need to remove mass. Remove from positions with most mass.
        order = np.argsort(-h)  # descending, so largest first
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    h = np.clip(h, 0.0, 1.0)
    return h

def build_gradient_matrix(h):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    G = np.zeros((M, n))
    for m in range(M):
        d = m - (n - 1)
        if d == 0:
            G[m, :] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            # P = dx * sum_{i=d}^{n-1} h[i]*(1-h[i-d])
            # dP/dh[i] for i>=d: dx*(1-h[i-d])
            # dP/dh[i] for i<d: -dx*h[i+d]
            G[m, d:] += dx * (1.0 - h[0:n-d])
            G[m, 0:d] += -dx * h[d:n]
        else:
            r = -d
            # P = dx * sum_{i=0}^{n-1-r} h[i+r]*(1-h[i])
            # dP/dh[i] for i>=r: dx*(1-h[i-r])
            # dP/dh[i] for i<r: -dx*h[i+r]
            G[m, r:] += dx * (1.0 - h[0:n-r])
            G[m, 0:r] += -dx * h[r:n]
    return G

def lp_minimax_direction(h, D=0.5, active_margin=0.0, use_all_rows=True):
    """Solve LP to find direction minimizing linearized max profile."""
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    P = compute_profile(h)
    G = build_gradient_matrix(h)
    c = float(np.max(P))
    
    # Minimize S
    # Subject to: G[m,:] @ d + P[m] <= S for all m (or active subset)
    #             sum(d) = 0
    #             -D <= d_i <= D
    
    if not use_all_rows:
        active = np.where(P >= c - active_margin)[0]
        if len(active) == 0:
            active = np.array([np.argmax(P)])
        G_sub = G[active, :]
        P_sub = P[active]
    else:
        G_sub = G
        P_sub = P
    
    n_rows = len(G_sub)
    
    # Variables: d_0, ..., d_{n-1}, S
    # c vector: [0, ..., 0, 1] (minimize S)
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    # Inequality: G_sub @ d - S <= -P_sub
    # i.e., G_sub concat (-1 in S column) <= -P_sub
    A_ub = np.zeros((n_rows, n + 1))
    A_ub[:, :n] = G_sub
    A_ub[:, -1] = -1.0
    b_ub = -P_sub
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    # Bounds: -D <= d_i <= D, S free (but bounded for solver)
    bounds = [(-D, D)] * n + [(-c, c + 1.0)]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.success:
        d = res.x[:n]
        S = res.x[-1]
        # Ensure sum(d) is exactly 0
        d -= d.mean()
        return d, S, c
    return None, None, c

def pair_move_search(h, max_pairs=100, rng=None):
    """Try pairwise coordinate adjustments."""
    n = len(h)
    best_h = h.copy()
    best_c = exact_score(h)
    
    if rng is None:
        rng = np.random.RandomState(0)
    
    # Try moving mass from one coordinate to another
    for _ in range(max_pairs):
        # Choose source (high value) and target (low value)
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        # Try both directions of mass transfer
        for sign in [1, -1]:
            # Move small amount from i to j (or vice versa)

                h_try[i] += delta
                h_try[j] -= delta

                c_try = exact_score(h_try)

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = np.linspace(0, 2, n + 1)
    xt = np.linspace(0, 2, new_n + 1)
    h_interp = np.zeros(new_n)
    for k in range(new_n):
        lo_k, hi_k = xt[k], xt[k+1]
        total_mass = 0.0
        total_width = 0.0
        for j in range(n):
            lo_j, hi_j = xs[j], xs[j+1]
            overlap = min(hi_k, hi_j) - max(lo_k, lo_j)
            if overlap > 0:
                total_mass += h[j] * overlap
                total_width += overlap
        h_interp[k] = total_mass / total_width if total_width > 1e-15 else 0.5
    h_interp = np.clip(h_interp, 0.0, 1.0)
    return repair_feasible(h_interp)

def lse_descent(h, beta=80.0, max_iters=150, alpha_init=0.004):
    """Log-sum-exp smoothed descent."""
    n = len(h)

    best_c = exact_score(h)
    alpha = alpha_init
    stuck = 0
    
    for it in range(max_iters):

        scaled = beta * profile
        max_s = np.max(scaled)
        shifted = scaled - max_s
        weights = np.exp(shifted)
        wsum = weights.sum()
        if wsum < 1e-300:
            break
        weights /= wsum
        
        G = build_gradient_matrix(h)
        grad = G.T @ weights
        grad -= grad.mean()
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad /= gnorm
        
        improved = False
        for a in [alpha, alpha*0.5, alpha*0.25, alpha*0.1]:
            h_try = repair_feasible(h - a * grad)
            s = exact_score(h_try)
            if s < best_c - 1e-14:
                h = h_try
                best_c = s
                improved = True
                alpha = min(alpha * 1.1, 0.02)
                stuck = 0
                break
            elif s < float(np.max(compute_profile(h))) - 1e-14:
                h = h_try
                improved = True
                break
        if not improved:
            alpha *= 0.5
            stuck += 1
            if stuck > 30 or alpha < 1e-9:
                break
    
    return h, best_c

```python
import numpy as np
from scipy.optimize import linprog
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx])
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(-h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    return np.clip(h, 0.0, 1.0)

def build_gradient_matrix(h):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    G = np.zeros((M, n))
    for m in range(M):
        d = m - (n - 1)
        if d == 0:
            G[m, :] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            G[m, d:] += dx * (1.0 - h[0:n-d])
            G[m, 0:d] += -dx * h[d:n]
        else:
            r = -d
            G[m, r:] += dx * (1.0 - h[0:n-r])
            G[m, 0:r] += -dx * h[r:n]
    return G

def lp_minimax_dir(h, D=0.5, use_all=True, margin=0.0):
    n = len(h)
    M = 2 * n - 1
    P = compute_profile(h)
    c = float(np.max(P))
    
    if use_all:
        rows = np.arange(M)
    else:
        rows = np.where(P >= c - margin)[0]
        if len(rows) == 0:
            rows = np.array([np.argmax(P)])
    
    G = build_gradient_matrix(h)
    Gs = G[rows, :]
    Ps = P[rows]
    nr = len(rows)
    
    # LP: min S s.t. Gs@v - S <= -Ps, sum(v)=0, -D<=v<=D
    c_obj = np.zeros(n+1)
    c_obj[-1] = 1.0
    
    A_ub = np.zeros((nr, n+1))
    A_ub[:, :n] = Gs
    A_ub[:, n] = -1.0
    b_ub = -Ps
    
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    bounds = [(-D, D)]*n + [(-c-0.5, c+0.5)]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            v = res.x[:n]
            v -= v.mean()
            S = float(res.x[n])
            return v, S, c
    except Exception:
        pass
    return None, None, c

def line_search(h, d, t_max=None, n_samples=40):
    n = len(h)
    if t_max is None:
        pos = d > 1e-14
        neg = d < -1e-14
        t_max = 1.0
        if np.any(pos):
            t_max = min(t_max, ((1.0 - h[pos]) / d[pos]).min())
        if np.any(neg):
            t_max = min(t_max, (h[neg] / (-d[neg])).min())
        t_max *= 0.95
        if t_max < 1e-8:
            return h, exact_score(h)
    
    best_t = 0.0
    best_c = exact_score(h)
    
    # Coarse search
    ts = np.linspace(0.005*t_max, t_max, n_samples)
    for t in ts:
        h_try = repair_feasible(h + t*d)
        c_try = exact_score(h_try)
        if c_try < best_c:
            best_c = c_try
            best_t = t
    
    # Fine search around best
    if best_t > 0:
        lo = max(0, best_t - t_max/n_samples)
        hi = best_t + t_max/n_samples
        ts2 = np.linspace(lo, hi, 30)
        for t in ts2:
            if t <= 0 or t > t_max:
                continue
            h_try = repair_feasible(h + t*d)
            c_try = exact_score(h_try)
            if c_try < best_c:
                best_c = c_try
                best_t = t
    
    if best_t > 0:
        return repair_feasible(h + best_t*d), best_c
    return h, best_c

def lse_descent(h, beta=80.0, max_iters=100, alpha_init=0.004):

    h = h.copy()
    best_c = exact_score(h)
    alpha = alpha_init
    stuck = 0
    for it in range(max_iters):
        P = compute_profile(h)
        scaled = beta * P
        shifted = scaled - scaled.max()
        w = np.exp(shifted)
        ws = w.sum()
        if ws < 1e-300:
            break
        w /= ws
        G = build_gradient_matrix(h)
        grad = G.T @ w
        grad -= grad.mean()
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        grad /= gn
        improved = False
        for a in [alpha, alpha*0.5, alpha*0.25, alpha*0.1]:

            s = exact_score(h_try)

                alpha = min(alpha * 1.1, 0.02)

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = np.linspace(0, 2, n+1)
    xt = np.linspace(0, 2, new_n+1)
    h_interp = np.zeros(new_n)
    for k in range(new_n):
        lo_k, hi_k = xt[k], xt[k+1]
        tm = 0.0
        tw = 0.0
        for j in range(n):
            lo_j, hi_j = xs[j], xs[j+1]
            ov = min(hi_k, hi_j) - max(lo_k, lo_j)
            if ov > 0:
                tm += h[j] * ov
                tw += ov
        h_interp[k] = tm / tw if tw > 1e-15 else 0.5
    return repair_feasible(np.clip(h_interp, 0.0, 1.0))

def coord_descent(h, max_iter=30, top_k=20):
    """Coordinate descent: for each i, find best j to transfer mass to/from."""
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    alphas = [0.005, 0.01, 0.02, 0.04, 0.07]
    
    for it in range(max_iter):
        improved = False
        # Focus on coordinates with extreme values
        order_hi = np.argsort(-h)[:top_k]  # most likely sources
        order_lo = np.argsort(h)[:top_k]   # most likely targets
        
        for i in order_hi:
            for j in order_lo:
                if i == j:
                    continue
                for a in alphas:
                    if h[i] - a >= 0 and h[j] + a <= 1:
                        h_try = h.copy()
                        h_try[i] -= a
                        h_try[j] += a
                        c_try = exact_score(h_try)

                            h = h_try
                            best_c = c_try
                            improved = True
                            break
                if improved:
                    break
            if improved:
                break
        if not improved:
            break
    return h, best_c
