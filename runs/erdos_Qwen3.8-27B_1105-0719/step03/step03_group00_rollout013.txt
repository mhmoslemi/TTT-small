            lo = mid
        else:
            hi = mid

def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-9) & (h < 1.0 - 1e-9)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()

```python
def compute_grad_matrix(h):
    n = len(h)
    nk = 2*n - 1
    G = np.zeros((nk, n))
    for k in range(nk):
        n_lo = max(0, k - (n-1))
        n_hi = min(n-1, k)
        # For j in [n_lo, n_hi]: j is in valid range
        # For j such that k-j is in [n_lo, n_hi]: k-j in valid, i.e., j in [k-n_hi, k-n_lo]
        valid_j = np.arange(n_lo, n_hi + 1)
        valid_kj = np.arange(k - n_hi, k - n_lo + 1)
        valid_kj = valid_kj[(valid_kj >= 0) & (valid_kj <= n-1)]
        
        # Term 1: (1-h[k-j]) for j in valid_j
        for j in valid_j:
            kj = k - j
            if 0 <= kj <= n-1:
                G[k, j] += 1.0 - h[kj]
            else:
                G[k, j] += 1.0
        
        # Term 2: -h[k-j] for j in valid_kj
        for j in valid_kj:
            kj = k - j
            if 0 <= kj <= n-1:
                G[k, j] -= h[kj]
    
    return G
```

This is O(n^2) which is fine for n=160.

def compute_grad_matrix(h):
    n = len(h)
    nk = 2*n - 1
    G = np.zeros((nk, n))
    for k in range(nk):
        n_lo = max(0, k - (n-1))
        n_hi = min(n-1, k)
        # valid_j range
        j_start, j_end = n_lo, n_hi
        # For these j, k-j ranges from k-j_end to k-j_start
        kj_start = k - j_end
        kj_end = k - j_start
        # Clip to [0, n-1]
        kj_start_clipped = max(0, kj_start)
        kj_end_clipped = min(n-1, kj_end)
        
        # Term 1: G[k, j] += 1 - h[k-j] for j in [j_start, j_end] where k-j in [0, n-1]
        # j where 0 ≤ k-j ≤ n-1: j in [k-(n-1), k]
        j_t1_lo = max(j_start, k-(n-1))
