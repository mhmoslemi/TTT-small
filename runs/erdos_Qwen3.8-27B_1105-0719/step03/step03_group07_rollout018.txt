        dp = -d
        for i in range(n - dp):
            g[i + dp] += dx * (1.0 - h[i])
            g[i] -= dx * h[i + dp]
    return g

def slsqp_active_set(h0, maxiter=300, active_k=12, tol=1e-12):
    """SLSQP with adaptive active set."""
    n = len(h0)
    dx = 2.0 / n
    M = 2 * n - 1
    
    def current_active_set(h, k):
        P = profile_h(h)
        c = P.max()
        # Select constraints within a threshold of the max
        thresh = c - max(0.002, 0.04 * (c - np.median(P)))
        active = np.where(P >= thresh)[0]
        if len(active) == 0:
            active = np.array([np.argmax(P)])
        if len(active) > k:
            top_idx = np.argsort(P[active])[::-1][:k]
            active = active[top_idx]
        return active
    
    active = current_active_set(h0, active_k)
    
    def obj(x):
        return x[-1]
    
    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    
    def eq_fun(x):
        return x[:-1].sum() - n / 2.0
    
    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g
    
    def ineq_fun(x):
        h = x[:-1]
        t = x[-1]
        prof = profile_h(h)
        return t - prof[active]
    
    def ineq_jac(x):
        h = x[:-1]
        J = np.zeros((len(active), n + 1))
        for i, lag_idx in enumerate(active):
            J[i, :-1] = -build_gradient_for_lag(h, lag_idx, n, dx)
            J[i, -1] = 1.0
        return J
    
    c0 = exact_score(h0)
    x0 = np.concatenate([h0, [c0 + 1e-4]])
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 1.0)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': tol})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = repair_feasible(res.x[:-1])
            c_new = exact_score(h_new)
            if c_new <= c0 + 1e-10:
                return h_new
    except Exception:
        pass
    return None

def pairwise_cd(h, max_sweeps=30, pairs_per_sweep=200, t_start=None, budget_s=1100):
    """Pairwise coordinate descent with exact 1D line search."""
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    rng = np.random.RandomState(12345)
    
    # Precompute all possible pairs
    all_pairs = [(p, q) for p in range(n) for q in range(p + 1, n)]
    
    for sweep in range(max_sweeps):
        if t_start is not None and time.perf_counter() - t_start > budget_s - 15:
            break
        improved = False
        # Random subset of pairs
        if len(all_pairs) > pairs_per_sweep:
            idx = rng.choice(len(all_pairs), pairs_per_sweep, replace=False)
            pairs = [all_pairs[i] for i in idx]
        else:
            pairs = all_pairs
        
        for p, q in pairs:
            if t_start is not None and time.perf_counter() - t_start > budget_s - 15:
                break
            s = h[p] + h[q]
            lo = max(0.0, s - 1.0)
            hi = min(1.0, s)
            if hi - lo < 1e-10:
                continue
            
            def line_obj(t):
                old_p, old_q = h[p], h[q]
                h[p] = t
                h[q] = s - t
                val = exact_score(h)
                h[p] = old_p
                h[q] = old_q
                return val
            
            result = minimize_scalar(line_obj, bounds=(lo, hi), method='bounded',
                                    options={'xatol': 1e-9, 'maxiter': 30})
            if result.success and result.fun < best_c - 1e-14:
                h[p] = result.x
                h[q] = s - result.x
                best_c = result.fun
                improved = True
    
        if not improved:
            break
    return h, best_c

def symmetrize_h(h):
    """Enforce h[i] + h[n-1-i] = 1 (palindromic complement)."""
    n = len(h)
    h_sym = np.zeros(n)
    for i in range(n):
        h_sym[i] = (h[i] + 1.0 - h[n - 1 - i]) / 2.0
    return repair_feasible(h_sym)

def symmetric_slsqp(h0, maxiter=200, tol=1e-11):
    """SLSQP on the symmetric subspace h[i] + h[n-1-i] = 1."""
    n = len(h0)
    if n % 2 != 0:
        return None  # Only works for even n
    half = n // 2
    dx = 2.0 / n
    
    # Free variables: x[0], ..., x[half-1] where h[i] = x[i] for i < half
    # and h[half+i] = x[half-1-i] for i < half (to maintain symmetry h[i]+h[n-1-i]=1)
    # Wait: h[i] + h[n-1-i] = 1 means h[n-1-i] = 1 - h[i]
    # So if i < half: h[i] = x[i], h[n-1-i] = 1-x[i]
    # n-1-i ranges from n-1 down to half when i ranges from 0 to half-1
    # So the free variables are x[0..half-1] = h[0..half-1]
    # and h[half..n-1] = 1 - h[n-1-half..0] = 1 - x[half-1..0]
    
    def x_to_h(x):
        h = np.zeros(n)
        for i in range(half):
            h[i] = x[i]
            h[n - 1 - i] = 1.0 - x[i]
        return h
    
    # Check: sum(h) = sum_{i=0}^{half-1} (x[i] + 1-x[i]) = half = n/2. ✓
    
    h_sym0 = symmetrize_h(h0)
    x0 = h_sym0[:half].copy()
    
    def obj(x):
        h = x_to_h(x)
        t = exact_score(h)
        return t
    
    def obj_grad(x):
        h = x_to_h(x)
        P = profile_h(h)
        kmax = np.argmax(P)
        g_full = build_gradient_for_lag(h, kmax, n, dx)
        g = np.zeros(half)
        for i in range(half):
            g[i] = g_full[i] - g_full[n - 1 - i]
        return g
    
    bounds = [(0.0, 1.0)] * half
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_grad,
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': tol})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = x_to_h(np.clip(res.x, 0, 1))
            h_new = repair_feasible(h_new)
            c_new = exact_score(h_new)
            c0 = exact_score(h_sym0)
            if c_new <= c0 + 1e-10:
                return h_new
    except Exception:
        pass
    return None

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    return repair_feasible(np.clip(h_new, 0.0, 1.0))

def softmax_projected_descent(h, max_iters=80, beta_start=50, beta_end=500,
                               momentum=0.4, t_start=None, budget_s=1100):
    """Softmax-projected gradient descent with momentum."""
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    vel = np.zeros(n)
    betas = np.geomspace(beta_start, beta_end, max_iters)
    
    for it in range(max_iters):
        if t_start is not None and time.perf_counter() - t_start > budget_s - 20:
            break
        beta = betas[it]
        P = profile_h(h)
        Pmax = P.max()
        exp_arg = beta * (P - Pmax)
        weights = np.exp(exp_arg)
        S = weights.sum()
        if S < 1e-300:
            weights = np.zeros(len(P))
            weights[np.argmax(P)] = 1.0
        else:
            weights /= S
        
        # Compute gradient as weighted sum of active lag gradients
        active = np.where(weights > 0.005)[0]
        if len(active) == 0:
            active = np.argsort(weights)[-5:]
        
        grad = np.zeros(n)
        dx = 2.0 / n
        for idx in active:
            grad += weights[idx] * build_gradient_for_lag(h, idx, n, dx)
        
        # Sum constraint: project gradient
        grad -= grad.mean()
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad /= gnorm
        
        vel = momentum * vel + grad
        vnorm = np.linalg.norm(vel)
        if vnorm > 0:
            vel /= vnorm
        
        # Backtracking line search
        alpha = 0.05
        for _ in range(10):
            h_try = repair_feasible(h - alpha * vel)
            s = exact_score(h_try)
            if s <= best_c:
                h = h_try
                if s < best_c:
                    best_c = s
                break
            alpha *= 0.5
        else:
            break
    
    return h, best_c

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx])
            h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break

            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))

                t_start=None, budget_s=1100):
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    rng = np.random.RandomState(seed)
    all_pairs = [(p, q) for p in range(n) for q in range(p + 1, n)]
    m = len(all_pairs)

    for sweep in range(max_sweeps):
        if t_start is not None and time.perf_counter() - t_start > budget_s - 15:
            break
        improved = False
        k = min(pairs_per_sweep, m)
        if k < m:
            idx = rng.choice(m, k, replace=False)
            pairs = [all_pairs[i] for i in idx]
        else:
            pairs = all_pairs

        for p, q in pairs:
            if t_start is not None and time.perf_counter() - t_start > budget_s - 15:
                break
            s = h[p] + h[q]
            lo = max(0.0, s - 1.0)
            hi = min(1.0, s)
            if hi - lo < 1e-10:
                continue

            old_p = h[p]
            old_q = h[q]

            def line_obj(t, _h=h, _p=p, _q=q, _s=s, _op=old_p, _oq=old_q):
                _h[_p] = t
                _h[_q] = _s - t
                val = exact_score(_h)
                _h[_p] = _op
                _h[_q] = _oq
                return val

            res = minimize_scalar(line_obj, bounds=(lo, hi), method='bounded',
                                  options={'xatol': 1e-9, 'maxiter': 25})
            if res.success and res.fun < best_c - 1e-14:
                h[p] = res.x
                h[q] = s - res.x
                best_c = res.fun
                improved = True

        if not improved:
            break
    return h, best_c

def symmetrize_h(h):
    n = len(h)
    h_sym = np.zeros(n)
    for i in range(n):
        h_sym[i] = (h[i] + 1.0 - h[n - 1 - i]) / 2.0
    return repair_feasible(h_sym)

def symmetric_slsqp(h0, maxiter=200, tol=1e-11):
    n = len(h0)
    if n % 2 != 0:
        return None
    half = n // 2
    dx = 2.0 / n

    def x_to_h(x):
        h = np.zeros(n)
        for i in range(half):
            h[i] = x[i]
            h[n - 1 - i] = 1.0 - x[i]
        return h

    h_sym0 = symmetrize_h(h0)
    x0 = h_sym0[:half].copy()

    def obj(x):
        h = x_to_h(x)
        return exact_score(h)

    def obj_grad(x):
        h = x_to_h(x)
        P = profile_h(h)
        kmax = np.argmax(P)
        g_full = build_gradient_for_lag(h, kmax, n, dx)
        g = np.zeros(half)
        for i in range(half):
            g[i] = g_full[i] - g_full[n - 1 - i]
        return g

    bounds = [(0.0, 1.0)] * half
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_grad,
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': tol})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = x_to_h(np.clip(res.x, 0.0, 1.0))
            h_new = repair_feasible(h_new)
            c_new = exact_score(h_new)
            c0 = exact_score(h_sym0)
            if c_new <= c0 + 1e-10:
                return h_new
    except Exception:
        pass
    return None

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    return repair_feasible(np.clip(h_new, 0.0, 1.0))

def softmax_projected_descent(h, max_iters=80, beta_start=50, beta_end=500,
                              momentum=0.4, t_start=None, budget_s=1100):
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    vel = np.zeros(n)
    betas = np.geomspace(beta_start, beta_end, max_iters)
    dx = 2.0 / n

    for it in range(max_iters):
        if t_start is not None and time.perf_counter() - t_start > budget_s - 20:
            break
        beta = betas[it]
        P = profile_h(h)
        Pmax = P.max()
        exp_arg = beta * (P - Pmax)
        weights = np.exp(exp_arg)
        S = weights.sum()
        if S < 1e-300:
            weights = np.zeros(len(P))
            weights[np.argmax(P)] = 1.0
        else:
            weights /= S

        active = np.where(weights > 0.005)[0]
        if len(active) == 0:
            active = np.argsort(weights)[-5:]

        grad = np.zeros(n)
        for idx in active:
            grad += weights[idx] * build_gradient_for_lag(h, idx, n, dx)

        grad -= grad.mean()
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad /= gnorm

        vel = momentum * vel + grad
        vnorm = np.linalg.norm(vel)
        if vnorm > 0:
            vel /= vnorm

        alpha = 0.05
        for _ in range(12):
            h_try = repair_feasible(h - alpha * vel)
            s = exact_score(h_try)
            if s <= best_c:
                h = h_try
                if s < best_c:
                    best_c = s
                break
            alpha *= 0.5
        else:
            break

    return h, best_c

def multi_start_slsqp(h_best, n, rng, t_start, budget_s, margin=30):
    best_c = exact_score(h_best)
    starts = [h_best.copy()]
    for eps in [0.004, 0.008, 0.015, 0.025, 0.04]:
        if time.perf_counter() - t_start > budget_s - 40:
            break
        z = rng.randn(n)
        z -= z.mean()
        h_p = repair_feasible(h_best + eps * z)
        starts.append(h_p)
    for eps in [0.01, 0.02]:
        if time.perf_counter() - t_start > budget_s - 30:
            break
        z = rng.randn(n)
        h_p = repair_feasible(symmetrize_h(h_best + eps * z))
        starts.append(h_p)

    for h0 in starts:
        if time.perf_counter() - t_start > budget_s - margin:
            break
        for ak in [10, 15, 20]:
            if time.perf_counter() - t_start > budget_s - 15:
                break
            sol = slsqp_active_set(h0, maxiter=250, active_k=ak, tol=1e-11)
            if sol is not None:
                c = exact_score(sol)
                if c < best_c - 1e-12:
                    h_best = sol
                    best_c = c
    return h_best, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()
    rng = np.random.RandomState(seed)

    parent = None
    try:
        parent = np.array(initial_h_values, dtype=np.float64)
    except NameError:
        pass
    except Exception:
        pass

    if parent is None or len(parent) < 4:
        parent = np.full(70, 0.5)

    parent = np.clip(parent, 0.0, 1.0)
    parent = repair_feasible(parent)
    n = len(parent)

    best_h = parent.copy()
    best_c = exact_score(parent)

    # Symmetrization
    h_sym = symmetrize_h(parent)
    c_sym = exact_score(h_sym)
    if c_sym < best_c - 1e-13:
        best_h = h_sym
        best_c = c_sym

    # Phase 1: Multi-start SLSQP
    if time.perf_counter() - t_start < budget_s - 120:
        best_h, best_c = multi_start_slsqp(best_h, n, rng, t_start, budget_s, margin=60)

    # Phase 2: Pairwise coordinate descent
    if time.perf_counter() - t_start < budget_s - 200:
        h_p, c_p = pairwise_cd(best_h, max_sweeps=25, pairs_per_sweep=300,
                               seed=seed + 1, t_start=t_start, budget_s=budget_s)
        if c_p < best_c - 1e-14:
            best_h = h_p
            best_c = c_p

    # Phase 3: SLSQP after pairwise
    if time.perf_counter() - t_start < budget_s - 100:
        sol = slsqp_active_set(best_h, maxiter=300, active_k=15, tol=1e-12)
        if sol is not None:
            c = exact_score(sol)
            if c < best_c - 1e-13:
                best_h = sol
                best_c = c

    # Phase 4: Projected gradient descent
    if time.perf_counter() - t_start < budget_s - 150:
        h_md, c_md = softmax_projected_descent(best_h, max_iters=60,
            beta_start=50, beta_end=400, momentum=0.4,
            t_start=t_start, budget_s=budget_s)
        if c_md < best_c - 1e-14:
            best_h = h_md
            best_c = c_md
        if time.perf_counter() - t_start < budget_s - 50:
            sol = slsqp_active_set(best_h, maxiter=200, active_k=12, tol=1e-12)
            if sol is not None:
                c = exact_score(sol)
                if c < best_c - 1e-13:
                    best_h = sol
                    best_c = c

    # Phase 5: More pairwise
    if time.perf_counter() - t_start < budget_s - 120:
        h_p, c_p = pairwise_cd(best_h, max_sweeps=20, pairs_per_sweep=250,
                               seed=seed + 100, t_start=t_start, budget_s=budget_s)
        if c_p < best_c - 1e-14:
            best_h = h_p
            best_c = c_p

    # Phase 6: Symmetric subspace
    if time.perf_counter() - t_start < budget_s - 80 and len(best_h) % 2 == 0:
        sol = symmetric_slsqp(best_h, maxiter=200, tol=1e-11)
        if sol is not None:
            c = exact_score(sol)
            if c < best_c - 1e-12:
                best_h = sol
                best_c = c
            if time.perf_counter() - t_start < budget_s - 40:
                h_p, c_p = pairwise_cd(best_h, max_sweeps=10, pairs_per_sweep=200,
                                       seed=seed+200, t_start=t_start, budget_s=budget_s)
                if c_p < best_c - 1e-14:
                    best_h = h_p
                    best_c = c_p

    # Phase 7: Dimension exploration
    if time.perf_counter() - t_start < budget_s - 150:
        n_cur = len(best_h)
        cand_dims = []
        for d in range(max(30, n_cur - 12), min(110, n_cur + 14), 2):
            if d != n_cur:
                cand_dims.append(d)
        for new_n in cand_dims[:6]:
            if time.perf_counter() - t_start > budget_s - 30:
                break
            h_interp = interpolate_to_n(best_h, new_n)
            sol = slsqp_active_set(h_interp, maxiter=200, active_k=12, tol=1e-11)
            if sol is not None:
                c = exact_score(sol)
                if c < best_c - 1e-11:
                    best_h = sol
                    best_c = c
                    if time.perf_counter() - t_start < budget_s - 20:
                        h_p, c_p = pairwise_cd(sol, max_sweeps=8, pairs_per_sweep=150,
                                               seed=seed+new_n, t_start=t_start,
                                               budget_s=budget_s)
                        if c_p < best_c - 1e-14:
                            best_h = h_p
                            best_c = c_p

    # Phase 8: Final pairwise + SLSQP polish
    if time.perf_counter() - t_start < budget_s - 50:
        h_p, c_p = pairwise_cd(best_h, max_sweeps=15, pairs_per_sweep=250,
                               seed=seed+500, t_start=t_start, budget_s=budget_s)
        if c_p < best_c - 1e-14:
            best_h = h_p
            best_c = c_p
    if time.perf_counter() - t_start < budget_s - 25:
        sol = slsqp_active_set(best_h, maxiter=250, active_k=10, tol=1e-13)
        if sol is not None:
            c = exact_score(sol)
            if c < best_c - 1e-14:
                best_h = sol
                best_c = c

    # Phase 9: Final micro-polish
    if time.perf_counter() - t_start < budget_s - 10:
        h_p, c_p = pairwise_cd(best_h, max_sweeps=5, pairs_per_sweep=200,
                               seed=seed+999, t_start=t_start, budget_s=budget_s)
        if c_p < best_c - 1e-15:
            best_h = h_p
            best_c = c_p

    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)

    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            final_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 1:
            if np.isfinite(eval_result[0]):
                final_c = float(eval_result[0])
    except Exception:
        pass

    return best_h, final_c, len(best_h)
