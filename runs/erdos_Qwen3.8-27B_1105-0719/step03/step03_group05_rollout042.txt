    t_max = 0.0

def pair_optimize(y, i, j, n_samples=25):

def reduced_gradient_active(y, active_lags, n):
    """Compute reduced gradient weighted by active lags."""
    h = make_anti(y)
    dx = 2.0 / n
    m = len(y)
    g = np.zeros(m)
    for q in active_lags:
        s = q - (n - 1)
        for j in range(m):
            i1 = j
            i2 = n - 1 - j
            gk1 = 0.0
            gk2 = 0.0
            if s > 0:
                if i1 <= n - 1 - s:
                    gk1 += (1.0 - h[i1 + s])
                if i1 - s >= 0:
                    gk1 -= h[i1 - s]
                if i2 <= n - 1 - s:
                    gk2 += (1.0 - h[i2 + s])
                if i2 - s >= 0:
                    gk2 -= h[i2 - s]
            elif s < 0:

                    gk1 += (1.0 - h[i1 - r])

                    gk1 -= h[i1 + r]

def interpolate_to_n(h, new_n):

        return h.copy()

    for k in range(new_n):

        total_mass = 0.0
        total_width = 0.0
        for j in range(n):
            lo_j, hi_j = xs[j], xs[j + 1]
            overlap = min(hi_k, hi_j) - max(lo_k, lo_j)
            if overlap > 0:
                total_mass += h[j] * overlap
                total_width += overlap
        h_interp[k] = total_mass / total_width if total_width > 1e-15 else 0.5
    h_interp = np.clip(h_interp, 0.0, 1.0)
    # Repair sum
    target = new_n / 2.0
    diff = target - h_interp.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            cap = 1.0 - h_interp
            order = np.argsort(-cap)
            rem = diff
            for i in range(new_n):
                add = min(rem, cap[order[i]] + 1e-15)
                h_interp[order[i]] += add
                rem -= add
                if rem <= 1e-12:
                    break
        else:
            order = np.argsort(h_interp)
            rem = -diff
            for i in range(new_n):
                sub = min(rem, h_interp[order[i]] + 1e-15)
                h_interp[order[i]] -= sub
                rem -= sub
                if rem <= 1e-12:
                    break
    h_interp = np.clip(h_interp, 0.0, 1.0)
    return h_interp

        h_pol, c_pol = full_space_polish(best_h, max_iters=30, rng=rng)
        if c_pol < best_c - 1e-12:
            best_h = h_pol.copy()
            best_c = c_pol
        # Re-project and re-optimize in anti-space
        h_ap = project_to_anti(best_h)
        if len(h_ap) % 2 == 0:
            y_ap = extract_anti(h_ap)
            y_opt2, c_opt2 = optimize_anti(y_ap, max_sweeps=60, rng=rng, budget_check=budget_check)
            h_opt2 = make_anti(y_opt2)
            if c_opt2 < best_c - 1e-12:
                best_h = h_opt2.copy()
                best_c = c_opt2

    # Stage 3: Resolution exploration
    if time_left() > 100:
        n_cur = len(best_h)
        candidates = set()
        for f in [0.7, 0.8, 0.9, 1.0, 1.1, 1.2, 1.3]:
            d = int(round(n_cur * f / 2)) * 2
            d = max(28, min(d, 140))
            candidates.add(d)
        candidates.discard(n_cur)
        cands_sorted = sorted(candidates, key=lambda x: abs(x - n_cur))

        for new_n in cands_sorted[:6]:
            if time_left() < 50:
                break
            h_interp = interpolate_to_n(best_h, new_n) if new_n != n_cur else best_h.copy()
            if new_n % 2 != 0:
                new_n += 1
                h_interp = interpolate_to_n(h_interp, new_n)
            h_ap = project_to_anti(h_interp)
            y_new = extract_anti(h_ap)
            y_optn, c_optn = optimize_anti(y_new, max_sweeps=60, rng=rng, budget_check=budget_check)
            h_res = make_anti(y_optn)
            if c_optn < best_c - 1e-12:
                best_h = h_res.copy()
                best_c = c_optn
                # Polish
                h_pol, c_pol = full_space_polish(best_h, max_iters=20, rng=rng)
                if c_pol < best_c - 1e-12:
                    best_h = h_pol.copy()
                    best_c = c_pol

    # Stage 4: Multiple anti-space restarts at best resolution
    if time_left() > 80:
        n_best = len(best_h)
        if n_best % 2 != 0:
            n_best += 1
            best_h = interpolate_to_n(best_h, n_best)
            best_h = repair_feasible(best_h)
            best_c = exact_score(best_h)

        h_ap = project_to_anti(best_h)
        y_base = extract_anti(h_ap)
        m = len(y_base)

        for restart in range(5):
            if time_left() < 30:
                break
            if restart == 0:
                y_start = y_base.copy()
            else:
                eps = 0.03 * (restart + 1)
                d = rng.randn(m) * eps
                t_min, t_max = feasible_t_range(y_base, d)
                t = rng.uniform(t_min, t_max)
                y_start = y_base + t * d

            y_r, c_r = optimize_anti(y_start, max_sweeps=50, rng=rng, budget_check=budget_check)
            h_r = make_anti(y_r)
            if c_r < best_c - 1e-12:
                best_h = h_r.copy()
                best_c = c_r

    # Stage 5: Intensive coordinate optimization at final resolution
    if time_left() > 60:
        n_fin = len(best_h)
        if n_fin % 2 != 0:
            n_fin += 1
            best_h = interpolate_to_n(best_h, n_fin)
            best_h = repair_feasible(best_h)
            best_c = exact_score(best_h)

        h_ap = project_to_anti(best_h)
        y_fin = extract_anti(h_ap)
        m = len(y_fin)

        # Many coordinate sweeps with refinement
        for sweep in range(30):
            if time_left() < 15:
                break
            improved = False
            for j in range(m):
                c = coord_optimize_refine(y_fin, j, 20, 20)
                if c < best_c - 1e-14:
                    best_c = c
                    improved = True
            # Pair moves
            if not improved:
                for _ in range(40):
                    i, j = rng.randint(m, 2)
                    if i == j:
                        continue
                    c = pair_optimize(y_fin, i, j, 25)
                    if c < best_c - 1e-14:
                        best_c = c
                        improved = True
            if not improved:
                # Gradient
                h_tmp = make_anti(y_fin)
                profile = compute_profile(h_tmp)
                c_max = float(np.max(profile))
                margin = max(0.0005, 0.06 * c_max)
                active = np.where(profile >= c_max - margin)[0]
                if len(active) > 0:
                    y_try, c_try = gradient_line_search(y_fin.copy(), active, len(h_tmp), 50)
                    if c_try < best_c - 1e-14:
                        y_fin[:] = y_try
                        best_c = c_try
                        improved = True
            if not improved and sweep > 10:
                break

        h_final_anti = make_anti(y_fin)
        if best_c < exact_score(best_h) - 1e-12:
            best_h = h_final_anti.copy()

    # Stage 6: Final full-space polish
    if time_left() > 40:
        h_pol, c_pol = full_space_polish(best_h, max_iters=40, rng=rng)
        if c_pol < best_c - 1e-12:
            best_h = h_pol.copy()
            best_c = c_pol

    # Stage 7: One more anti optimization from final
    if time_left() > 30:
        n_f = len(best_h)
        if n_f % 2 != 0:
            n_f += 1
            best_h = interpolate_to_n(best_h, n_f)
            best_h = repair_feasible(best_h)
            best_c = exact_score(best_h)
        h_ap = project_to_anti(best_h)
        y_f = extract_anti(h_ap)
        y_f, c_f = optimize_anti(y_f, max_sweeps=40, rng=rng, budget_check=lambda: time_left() < 10)
        h_f = make_anti(y_f)
        if c_f < best_c - 1e-12:
            best_h = h_f.copy()
            best_c = c_f

    # Final validation
    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)

    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            final_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 2:
            final_c = float(eval_result[0])
    except Exception:
        pass

    return best_h, final_c, len(best_h)
