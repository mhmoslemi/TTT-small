    G[0] = 1.0 - 2.0 * h

import numpy as np
from scipy.optimize import linprog, minimize_scalar
import time


def project_feasible(x, target_sum):
    n = len(x)
    x = np.clip(x, -3.0, 3.0)
    lo, hi = -3.0, 3.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = mask.sum()
        if cnt > 0:
            h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)


def get_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def c5_of(h, dx):
    return get_corrs(h).max() * dx


def grad_lag(h, lag, n):
    if lag == 0:
        return 1.0 - 2.0 * h
    g = np.zeros(n)
    g[lag:] += (1.0 - h[:n - lag])
    g[:n - lag] += -h[lag:]
    return g


def lp_step(h, dx, target_sum, active_idx, alpha, n):
    corrs = get_corrs(h)
    current_c5 = corrs.max() * dx
    m = len(active_idx)
    G = np.zeros((m, n))
    for row, idx in enumerate(active_idx):
        lag = abs(int(idx - (n - 1)))
        G[row] = grad_lag(h, lag, n)
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -corrs[active_idx]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(-h[i] * alpha, (1.0 - h[i]) * alpha) for i in range(n)]
    bounds.append((0.0, np.inf))
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status != 0:
            return h, current_c5, False
        delta = res.x[:n]
    except Exception:
        return h, current_c5, False
    if np.abs(delta).max() < 1e-15:
        return h, current_c5, False
    best_s = 0.0
    best_c5 = current_c5
    best_h = h
    for s in [1.0, 0.75, 0.55, 0.4, 0.3, 0.2, 0.12, 0.06, 0.03]:
        h_new = project_feasible(h + s * delta, target_sum)
        c5_new = c5_of(h_new, dx)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_s = s
            best_h = h_new
    return best_h, best_c5, best_s > 0


def lp_optimize(h, dx, target_sum, t0, time_limit,
                alpha_start=0.25, n_active_frac=0.4, alpha_decay=0.988,
                min_alpha=0.003, max_no_improve=120):
    n = len(h)
    n_total = 2 * n - 1
    n_active = max(20, int(n_active_frac * n_total))
    best_h = h.copy()
    best_c5 = c5_of(h, dx)
    alpha = alpha_start
    no_improve = 0
    stuck_counter = 0
    while time.time() - t0 < time_limit:
        if alpha < min_alpha:
            break
        corrs = get_corrs(h)
        active_idx = np.argsort(corrs)[-n_active:]
        h_new, c5_new, improved = lp_step(h, dx, target_sum, active_idx, alpha, n)
        if improved and c5_new < best_c5 - 1e-14:
            h = h_new
            best_c5 = c5_new
            best_h = h_new.copy()
            no_improve = 0
            stuck_counter = 0
        else:
            no_improve += 1
            stuck_counter += 1
            alpha *= alpha_decay
            if stuck_counter % 25 == 0:
                n_active = min(n_total, n_active + 10)
                alpha = max(alpha, 0.12)
            if stuck_counter % 50 == 0:
                n_active = max(20, n_active - 15)
                alpha = max(alpha, 0.18)
            if no_improve > max_no_improve:
                break
    return best_h, best_c5


def pairwise_optimize(h, dx, target_sum, t0, time_limit, pairs_per_round=120):

        h_new, c5_new = try_n_value(n_try, best_h, best_c5, t0, dt, rng)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new
            n = len(h_new)
            dx = 2.0 / n
            target_sum = n / 2.0

    # Phase 5: Final intensive polish (10% budget)
    t5 = t0 + budget_s - 3.0
    if time.time() < t5:
        n_b = len(best_h)
        dx_b = 2.0 / n_b
        ts_b = n_b / 2.0
        # Intensive LP with large active set
        dt = t5 - time.time()
        if dt > 15:
            h_opt, c5_opt = lp_optimize(best_h, dx_b, ts_b, t0, dt,
                                         alpha_start=0.15, n_active_frac=0.6,
                                         alpha_decay=0.995, min_alpha=0.001,
                                         max_no_improve=250)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
        # Pairwise final
        if time.time() < t5 - 5:
            dt2 = t5 - time.time()
            if dt2 > 10:
                h_opt, c5_opt = pairwise_optimize(best_h, dx_b, ts_b, t0, dt2,
                                                   pairs_per_round=200)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt
        # Tiny perturbation + LP
        if time.time() < t5 - 8:
            h_tiny = perturb_h(best_h, ts_b, 0.008, rng)
            dt3 = t5 - time.time()
            if dt3 > 8:
                h_opt, c5_opt = lp_optimize(h_tiny, dx_b, ts_b, t0, dt3,
                                             alpha_start=0.1, n_active_frac=0.5,
                                             alpha_decay=0.997, min_alpha=0.001,
                                             max_no_improve=150)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    n_final = len(best_h)
    ts_final = n_final / 2.0
    best_h = project_feasible(best_h, ts_final)
    dx_final = 2.0 / n_final
    best_c5 = c5_of(best_h, dx_final)
    return best_h, best_c5, n_final
