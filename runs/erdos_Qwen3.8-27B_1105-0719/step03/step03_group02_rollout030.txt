```python
        # We want G[k*, i] > G[k*, j] (transfer from high gradient to low gradient at k*)
        
        g_max = G[max_k]  # gradient of the max correlation
        # Sort by g_max: high g means "remove from here", low g means "add here"
        sorted_idx = np.argsort(g_max)
        
        best_improvement = 0
        best_move = None
        
        # Try pairs from top-decrease to bottom-increase
        n_try = min(n_pairs, n//2)
        candidates = []
        for a in range(n_try):
            for b in range(n - n_try, n):
                if a != b:
                    candidates.append((a, b))
        
        # Also try some random pairs
        rng_local = np.random.RandomState(it)
        for _ in range(n_try):
            a = rng_local.randint(n)
            b = rng_local.randint(n)
            if a != b:
                candidates.append((a, b))
        
        for (i, j) in candidates:
            eps_lo = max(h[i] - 1.0, -h[j])
            eps_hi = min(h[i], 1.0 - h[j])
            if eps_hi - eps_lo < 1e-12:
                continue
            
            d = G[:, j] - G[:, i]  # shape (2n-1,)
            # f(eps) = max(corrs + eps * d)
            # Evaluate at several points
            # Use a grid + golden section refinement
            
            # Quick evaluation at boundary and midpoint
            eps_mid = (eps_lo + eps_hi) / 2
            vals = np.array([
                np.max(corrs + eps_lo * d),
                np.max(corrs + eps_mid * d),
                np.max(corrs + eps_hi * d),
                np.max(corrs + 0.25*(eps_lo + eps_hi) * d),
                np.max(corrs + 0.75*(eps_lo + eps_hi) * d),
            ])
            best_local = np.max(corrs) * dx  # current value (eps=0 gives current)
            # Actually eps=0 might not be in [eps_lo, eps_hi]
            # Let's just find the min of the grid
            best_eps_idx = np.argmin(vals)
            best_eps = np.array([eps_lo, eps_mid, eps_hi, 0.25*(eps_lo+eps_hi), 0.75*(eps_lo+eps_hi)])[best_eps_idx]
            best_val = vals[best_eps_idx]
            
            # Refine with golden section
            a, b = eps_lo, eps_hi
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            c_pt = b - gr*(b-a)
            d_pt = a + gr*(b-a)
            fc = np.max(corrs + c_pt * d)
            fd = np.max(corrs + d_pt * d)
            for _ in range(25):
                if b - a < 1e-12:
                    break
                if fc < fd:
                    b = d_pt
                    d_pt = c_pt
                    fd = fc
                    c_pt = b - gr*(b-a)
                    fc = np.max(corrs + c_pt * d)
                else:
                    a = c_pt
                    c_pt = d_pt
                    fc = fd
                    d_pt = a + gr*(b-a)
                    fd = np.max(corrs + d_pt * d)
            
            eps_opt = (a + b) / 2.0
            val_opt = np.max(corrs + eps_opt * d)
            
            improvement = c5_val - val_opt * dx
            if improvement > best_improvement:
                best_improvement = improvement
                best_move = (i, j, eps_opt)
        
        if best_move is None or best_improvement < 1e-14:
            break
        
        i, j, eps_opt = best_move
        h[i] -= eps_opt
        h[j] += eps_opt
        h = np.clip(h, 0.0, 1.0)
        h = project_feasible(h, target_sum)
    
    return best_h, best_c5
```

```python
import numpy as np
from scipy.optimize import minimize, linprog
from scipy.special import logsumexp
import time

def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def c5_of_h(h, dx):
    return float(np.max(compute_corrs(h)) * dx)

def grad_all_k(h):
    n = len(h)
    G = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        offset = k - (n-1)
        i_lo = max(0, -offset)
        i_hi = min(n-1, n-1-offset)
        if i_lo > i_hi:
            continue
        idx = np.arange(i_lo, i_hi+1)
        G[k, idx+offset] += (1.0 - h[idx])
        G[k, idx] -= h[idx+offset]
    return G

def slsqp_smooth(h0, dx, target_sum, beta, maxiter=1000):
    n = len(h0)
    
    def obj_and_grad(x):
        h = np.clip(x, 0, 1)  # not strictly needed with bounds
        corrs = compute_corrs(h)
        objective = logsumexp(beta * corrs) / beta * dx
        shifted = corrs - corrs.max()
        w = np.exp(beta * shifted)
        w /= w.sum()
        G = grad_all_k(h)
        gradient = (w @ G) * dx
        return objective, gradient
    
    def c_eq(x):
        return x.sum() - target_sum
    
    def c_eq_jac(x):
        return np.ones(len(x))
    
    cons = [{'type': 'eq', 'fun': c_eq, 'jac': c_eq_jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_and_grad, h0, jac=True, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        h_out = project_feasible(res.x, target_sum)
    except Exception:
        h_out = project_feasible(h0.copy(), target_sum)
    return h_out, c5_of_h(h_out, dx)

def pair_descent(h0, dx, target_sum, max_iters=2000, n_top=20, n_rand=20, seed=0):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    rng = np.random.RandomState(seed)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    
    for it in range(max_iters):
        corrs = compute_corrs(h)
        c5_val = corrs.max() * dx
        G = grad_all_k(h)
        
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
        
        # Build candidate pair list
        # Top pairs: those that most decrease the max correlation
        max_k = np.argmax(corrs)
        g_max = G[max_k]
        sorted_idx = np.argsort(g_max)
        
        n_dec = min(n_top, n // 3)
        n_inc = min(n_top, n // 3)
        dec_set = set(sorted_idx[:n_dec].tolist())
        inc_set = set(sorted_idx[-n_inc:].tolist())
        
        candidates = []
        for i in sorted_idx[:n_dec]:
            for j in sorted_idx[-n_inc:]:
                if i != j:
                    candidates.append((i, j))
        
        # Add random pairs
        for _ in range(n_rand):
            a = rng.randint(n)
            b = rng.randint(n)
            if a != b:
                candidates.append((a, b))
        
        # Deduplicate
        candidates = list(set(candidates))
        
        best_improvement = 0
        best_move = None
        
        for (i, j) in candidates:
            eps_lo = max(h[i] - 1.0, -h[j])
            eps_hi = min(h[i], 1.0 - h[j])
            if eps_hi - eps_lo < 1e-12:
                continue
            
            d = G[:, j] - G[:, i]
            
            # Golden section search on f(eps) = max(corrs + eps*d)
            a, b = eps_lo, eps_hi
            c_pt = b - gr*(b-a)
            d_pt = a + gr*(b-a)
            fc = np.max(corrs + c_pt * d)
            fd = np.max(corrs + d_pt * d)
            for _ in range(30):
                if b - a < 1e-13:
                    break
                if fc < fd:
                    b = d_pt
                    d_pt = c_pt
                    fd = fc
                    c_pt = b - gr*(b-a)
                    fc = np.max(corrs + c_pt * d)
                else:
                    a = c_pt
                    c_pt = d_pt
                    fc = fd
                    d_pt = a + gr*(b-a)
                    fd = np.max(corrs + d_pt * d)
            
            eps_opt = (a + b) / 2.0
            val_opt = np.max(corrs + eps_opt * d)
            improvement = c5_val - val_opt * dx
            
            if improvement > best_improvement:
                best_improvement = improvement
                best_move = (i, j, eps_opt)
        
        if best_move is None or best_improvement < 1e-14:
            break
        
        i, j, eps_opt = best_move
        h[i] -= eps_opt
        h[j] += eps_opt
        h = np.clip(h, 0.0, 1.0)
        h = project_feasible(h, target_sum)
    
    return best_h, best_c5

def slsqp_z(h0, dx, target_sum, beta, maxiter=1000):
    """SLSQP in z-space using sigmoid transform."""
    n = len(h0)
    
    def sigmoid(z):
        return 1.0 / (1.0 + np.exp(-np.clip(2.0 * z, -500, 500)))
    
    h0_clip = np.clip(h0, 1e-8, 1.0 - 1e-8)
    z0 = -0.5 * np.log((1.0 - h0_clip) / h0_clip)
    
    def obj(z):
        h = sigmoid(z)
        return logsumexp(beta * compute_corrs(h)) / beta * dx
    
    def grad_obj(z):
        h = sigmoid(z)
        corrs = compute_corrs(h)
        shifted = corrs - corrs.max()
        w = np.exp(beta * shifted)
        w /= w.sum()
        G = grad_all_k(h)
        grad_h = (w @ G) * dx
        return 2.0 * h * (1.0 - h) * grad_h
    
    def c_eq(z):
        return sigmoid(z).sum() - target_sum
    
    def c_eq_jac(z):
        h = sigmoid(z)
        return 2.0 * h * (1.0 - h)
    
    cons = [{'type': 'eq', 'fun': c_eq, 'jac': c_eq_jac}]
    try:
        res = minimize(obj, z0, jac=grad_obj, method='SLSQP',
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        h_out = project_feasible(sigmoid(res.x), target_sum)
    except Exception:
        h_out = project_feasible(h0_clip, target_sum)
    return h_out, c5_of_h(h_out, dx)

def lp_polish(h0, dx, target_sum, max_iters=200, init_step=0.1):
    """LP-based polish using linearization of top correlations."""
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    step = init_step
    
    for it in range(max_iters):
        corrs = compute_corrs(h)
        c5_val = corrs.max() * dx
        max_c = corrs.max()
        
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
        
        # Active set: top correlations
        n_active = max(10, int(0.3 * (2*n-1)))
        top_idx = np.argsort(corrs)[-n_active:]
        
        G = grad_all_k(h)
        
        # LP: min t s.t. corrs[k] + dx*sum_i G[k,i]*delta_i <= t*dx, sum delta_i = 0, |delta_i| <= step
        # Variables: delta[0..n-1], t
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        
        A_rows = []
        b_rows = []
        for k in top_idx:
            row = np.zeros(n + 1)
            row[:n] = dx * G[k]
            row[n] = -1.0
            A_rows.append(row)
            b_rows.append(-(corrs[k] + 1e-12) * dx)
        
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)
        
        bounds = []
        for i in range(n):
            bounds.append((max(-h[i], -step), min(1.0 - h[i], step)))
        bounds.append((0.0, None))
        
        try:
            res = linprog(c_obj, A_ub=np.array(A_rows), b_ub=np.array(b_rows),
                          A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
        except Exception:
            break
        if res.status != 0:
            break
        
        delta = res.x[:n]
        if np.max(np.abs(delta)) < 1e-14:
            break
        
        h_new = project_feasible(h + delta, target_sum)
        new_c5 = c5_of_h(h_new, dx)
        
        if new_c5 < c5_val - 1e-14:
            h = h_new
            step = min(step * 1.3, init_step)
        else:
            # Try smaller step
            improved = False
            for frac in [0.5, 0.25, 0.1]:
                h_try = project_feasible(h + frac * delta, target_sum)
                if c5_of_h(h_try, dx) < c5_val - 1e-14:
                    h = h_try
                    improved = True
                    break
            if not improved:
                step *= 0.5
                if step < 1e-8:
                    break
    
    return best_h, best_c5

def block_descent(h0, dx, target_sum, max_iters=500, block_size=3, n_blocks=20, seed=0):
    """Block coordinate descent: optimize groups of 3 coordinates."""
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    rng = np.random.RandomState(seed)
    
    for it in range(max_iters):
        corrs = compute_corrs(h)
        c5_val = corrs.max() * dx
        G = grad_all_k(h)
        
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
        
        best_improvement = 0
        best_move = None
        
        # Try random blocks of 3
        for _ in range(n_blocks):
            block = rng.choice(n, size=block_size, replace=False)
            i, j, k = int(block[0]), int(block[1]), int(block[2])
            
            # Parametrize: h_i - eps1, h_j + eps1 - eps2, h_k + eps2
            # d1 = G[:,j] - G[:,i], d2 = G[:,k] - G[:,j]
            d1 = G[:, j] - G[:, i]
            d2 = G[:, k] - G[:, j]
            
            # f(eps1, eps2) = max(corrs + eps1*d1 + eps2*d2)
            # Constraints:
            # 0 <= h_i - eps1 <= 1  =>  h_i-1 <= eps1 <= h_i
            # 0 <= h_j + eps1 - eps2 <= 1  =>  1 - h_j <= eps1 - eps2 <= h_j (hmm need to be careful)
            #   => (1-h_j) - eps1 <= -eps2 <= h_j - eps1... let me just use box constraints
            # 0 <= h_k + eps2 <= 1  =>  -h_k <= eps2 <= 1 - h_k
            
            # Simplify: eps1 ∈ [h_i-1, h_i], eps2 ∈ [-h_k, 1-h_k]
            # And h_j + eps1 - eps2 ∈ [0, 1] => eps1 - eps2 ∈ [-h_j, 1-h_j]
            
            # For simplicity, use a grid + refinement
            # Grid search over eps1, eps2
            e1_lo, e1_hi = max(h[i]-1, -h[j]), min(h[i], 1-h[j])  # approximate bounds for eps1
            e2_lo, e2_hi = -h[k], 1-h[k]
            
            if e1_hi - e1_lo < 1e-10 or e2_hi - e2_lo < 1e-10:
                continue
            
            # 5x5 grid
            e1_grid = np.linspace(e1_lo, e1_hi, 5)
            e2_grid = np.linspace(e2_lo, e2_hi, 5)
            
            best_local = c5_val / dx
            best_e1, best_e2 = 0, 0
            
            for e1 in e1_grid:
                for e2 in e2_grid:
                    # Check feasibility
                    hi_v = h[i] - e1
                    hj_v = h[j] + e1 - e2
                    hk_v = h[k] + e2
                    if hi_v < -1e-10 or hi_v > 1+1e-10:
                        continue
                    if hj_v < -1e-10 or hj_v > 1+1e-10:
                        continue
                    if hk_v < -1e-10 or hk_v > 1+1e-10:
                        continue
                    
                    val = np.max(corrs + e1*d1 + e2*d2)
                    if val < best_local:
                        best_local = val
                        best_e1, best_e2 = e1, e2
            
            improvement = c5_val - best_local * dx
            if improvement > best_improvement:
                best_improvement = improvement
                best_move = (i, j, k, best_e1, best_e2)
        
        if best_move is None or best_improvement < 1e-14:
            break
        
        i, j, k, e1, e2 = best_move
        h[i] -= e1
        h[j] += e1 - e2
        h[k] += e2
        h = np.clip(h, 0.0, 1.0)
        h = project_feasible(h, target_sum)
    
    return best_h, best_c5
```
