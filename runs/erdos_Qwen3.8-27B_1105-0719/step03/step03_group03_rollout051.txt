def coord_descent(h, max_passes=20, n_values=11, budget_fn=None, min_time=5.0):

    vals = np.linspace(0.0, 1.0, n_values)

def subgradient_step(h, profile, n_active=3):
    n = len(h)
    dx = 2.0 / n
    idx = np.argsort(profile)[-n_active:]
    grad = np.zeros(n)
    for m in idx:
        k = abs(m - (n - 1))
        if k == 0:
            g = 1.0 - 2.0 * h
        else:
            g = np.zeros(n)
            lo = k
            hi = n
            g[lo:hi] += (1.0 - h[:hi - lo])
            lo2 = 0
            hi2 = n - k
            g[lo2:hi2] -= h[k:hi2 + k]
        grad += g
    grad /= len(idx)
    grad -= grad.mean()
    return grad * dx

def interp_to(h, new_n):

        hr, cr = coord_descent(best_h, max_passes=20, n_values=13, budget_fn=tl, min_time=30)
        if cr < best_c - 1e-12:
            best_c = cr
            best_h = hr

    # Phase 7: Multi-resolution search
    if tl() > 100:
        n_best = len(best_h)
        dims = sorted(set([d for d in [n_best - 10, n_best - 6, n_best - 4, n_best - 2,
                                       n_best + 2, n_best + 4, n_best + 6, n_best + 10]
                           if d >= 30 and d <= 150 and d % 2 == 0]))
        for nd in dims[:8]:
            if tl() < 50:
                break
            hi = interp_to(best_h, nd)
            ci = exact_score(hi)
            if ci < best_c + 0.03:
                hb = binarize(hi)
                bh2, bc2 = swap_optimize(hb, max_sweeps=100, budget_fn=tl, min_time=30)
                if bc2 < best_c - 1e-12:
                    best_c = bc2
                    best_h = bh2
                hr2, cr2 = coord_descent(np.array(best_h, dtype=np.float64), max_passes=10,
                                         n_values=9, budget_fn=tl, min_time=25)
                if cr2 < best_c - 1e-12:
                    best_c = cr2
                    best_h = hr2

    # Phase 8: Simulated annealing on binary
    if tl() > 60:
        hb = binarize(best_h)
        bh, bc = swap_optimize(hb, max_sweeps=100, budget_fn=tl, min_time=30,
                               allow_worse=True, temperature=0.5)
        if bc < best_c - 1e-12:
            best_c = bc
            best_h = bh

    # Phase 9: Final subgradient polish
    if tl() > 40:
        hs, cs = subgradient_method(best_h, max_iters=200, budget_fn=tl, min_time=15)
        if cs < best_c - 1e-12:
            best_c = cs
            best_h = hs

    # Phase 10: Final coordinate descent with fine grid
    if tl() > 30:
        hr, cr = coord_descent(best_h, max_passes=10, n_values=21, budget_fn=tl, min_time=10)
        if cr < best_c - 1e-12:
            best_c = cr
            best_h = hr

    # Phase 11: Last binary swap pass
    if tl() > 20:
        hb = binarize(best_h)
        bh, bc = swap_optimize(hb, max_sweeps=100, budget_fn=tl, min_time=8)
        if bc < best_c - 1e-12:
            best_c = bc
            best_h = bh

    # Final validation
    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)

    try:
        er = evaluate_erdos_solution(best_h)
        if isinstance(er, (float, int)):
            final_c = min(final_c, float(er))
        elif isinstance(er, tuple) and len(er) >= 2:
            final_c = min(final_c, float(er[0]))
    except Exception:
        pass

    return best_h, final_c, len(best_h)
