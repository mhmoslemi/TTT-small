from scipy.optimize import linprog

def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(80):

        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:

        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def c5_of_h(h, dx):
    return compute_corrs(h).max() * dx

def build_grad_matrix(h, n):
    """G[i,k] = d(corr_k)/d(h_i)"""
    m = 2 * n - 1
    G = np.zeros((n, m))
    offsets = np.arange(0, m) - (n - 1)  # offset for each k
    for i in range(n):

        im = i - offsets  # i - offset_k for all k
        valid_p = (ip >= 0) & (ip < n)
        valid_m = (im >= 0) & (im < n)
        G[i, valid_p] += (1.0 - h[ip[valid_p]])
        G[i, valid_m] += (-h[im[valid_m]])
    return G

def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)


def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def build_grad_matrix(h, n):
    m = 2 * n - 1
    G = np.zeros((n, m))
    offsets = np.arange(0, m) - (n - 1)
    for i in range(n):
        ip = i + offsets
        im = i - offsets
        vp = (ip >= 0) & (ip < n)
        vm = (im >= 0) & (im < n)
        G[i, vp] += (1.0 - h[ip[vp]])
        G[i, vm] += (-h[im[vm]])
    return G

def grad_corr_k(h, k, n):
    offset = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -offset)
    i_hi = min(n - 1, n - 1 - offset)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx + offset] += (1.0 - h[idx])
    grad[idx] += (-h[idx + offset])
    return grad

def smooth_max_obj(h, dx, beta):
    corrs = compute_corrs(h)
    return logsumexp(beta * corrs) / beta * dx


def smooth_max_grad(h, dx, beta):
    n = len(h)
    corrs = compute_corrs(h)
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones_like(corrs) / len(corrs)
    else:
        w /= w_sum
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] > 1e-13:
            grad += w[k] * grad_corr_k(h, k, n)
    return grad * dx


def slsqp_polish(h0, dx, target_sum, beta, maxiter=800):
    n = len(h0)

    def obj(x):
        corrs = compute_corrs(x)
        return logsumexp(beta * corrs) / beta * dx

    def grad_obj(x):
        return smooth_max_grad(x, dx, beta)

    def c_fun(x):
        return x.sum() - target_sum

    def c_jac(x):
        return np.ones(n)

    from scipy.optimize import minimize
    cons = [{'type': 'eq', 'fun': c_fun, 'jac': c_jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0, jac=grad_obj, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        h_out = project_feasible(res.x, target_sum)
    except Exception:
        h_out = project_feasible(h0, target_sum)
    return h_out, c5_of_h(h_out, dx)

    # Phase 7: SLSQP on best

        h_s, c5_s = slsqp_polish(best_h, 2.0 / len(best_h), len(best_h) / 2.0, 3000, 600)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s

    # Phase 8: Final coordinate descent + LP
    if time_left() > 30:
        n_b = len(best_h)
        dx_b = 2.0 / n_b
        ts_b = n_b / 2.0
        h_f, c5_f = exact_coordinate_descent(best_h, dx_b, ts_b, max_sweeps=80, time_fn=time_left)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f
    if time_left() > 20:
        n_b = len(best_h)
        dx_b = 2.0 / n_b
        ts_b = n_b / 2.0
        h_lpf, c5_lpf = lp_polish(best_h, dx_b, ts_b, n_steps=40, lam=0.1, time_fn=time_left)
        if c5_lpf < best_c5:
            best_c5 = c5_lpf
            best_h = h_lpf

    # Final validation
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final / 2.0)
    best_c5 = c5_of_h(best_h, 2.0 / n_final)

    return best_h, best_c5, n_final
