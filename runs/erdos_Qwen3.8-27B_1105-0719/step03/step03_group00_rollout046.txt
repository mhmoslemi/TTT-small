    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx + off] += (1.0 - h[idx])
    grad[idx] -= h[idx + off]
    return grad

def solve_soft_lp(h, dx, active_idx, corrs, alpha, rho_q, rho_s_arr):
    n = len(h)
    K = len(active_idx)
    m = n + K + 1
    c_obj = np.zeros(m)
    c_obj[-1] = rho_q
    c_obj[n:n+K] = rho_s_arr
    A_ub = np.zeros((K, m))
    b_ub = np.zeros(K)
    for j, k in enumerate(active_idx):
        g = grad_lag(h, k, n)
        A_ub[j, :n] = dx * g
        A_ub[j, n+j] = 1.0
        A_ub[j, -1] = -1.0
        b_ub[j] = -corrs[k]
    A_eq = np.zeros((1, m))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(-h[i]*alpha, (1.0-h[i])*alpha) for i in range(n)]
    for j in range(K):
        bounds.append((0.0, None))
    bounds.append((0.0, None))
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status != 0:
            return None
        return res.x[:n]
    except Exception:
        return None

def solve_sum_lp(h, dx, active_idx, weights, alpha):
    n = len(h)
    K = len(active_idx)
    c_vec = np.zeros(n)
    for j, k in enumerate(active_idx):
        c_vec += weights[j] * dx * grad_lag(h, k, n)
    A_eq = np.zeros((1, n))
    A_eq[0, :] = 1.0
    bounds = [(-h[i]*alpha, (1.0-h[i])*alpha) for i in range(n)]
    try:
        res = linprog(c_vec, A_eq=A_eq, b_eq=np.zeros(1), bounds=bounds, method='highs')
        if res.status != 0:
            return None
        return res.x
    except Exception:
        return None

def do_line_search(h, cur_c5, direction, dx, target_sum):
    scales = [1.0, 0.8, 0.6, 0.5, 0.4, 0.3, 0.2, 0.15, 0.1, 0.06, 0.04, 0.02]
    best_h = h.copy()
    best_c5 = cur_c5
    best_t = 0.0
    for t in scales:
        h_new = project_feasible(h + t * direction, target_sum)
        c5_new = c5_of_h(h_new, dx)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new
            best_t = t
    return best_h, best_c5, best_t

def pair_refine(h, dx, target_sum, n_passes, time_limit, t0):
    n = len(h)
    best_c5 = c5_of_h(h, dx)
    ones = np.where(h > 0.01)[0]
    zeros = np.where(h < 0.99)[0]
    for p in range(n_passes):
        if time.time() - t0 > time_limit:
            break
        improved = False
        for i in range(n):
            if time.time() - t0 > time_limit:
                break
            for j in range(i+1, n):
                if abs(h[i] - h[j]) < 0.001:
                    continue
                lo_t = max(-h[i], h[j] - 1.0)
                hi_t = min(1.0 - h[i], h[j])
                if hi_t - lo_t < 1e-4:
                    continue
                ts = np.linspace(lo_t + 0.01*(hi_t-lo_t), hi_t - 0.01*(hi_t-lo_t), 7)
                for t in ts:
                    h[i] += t
                    h[j] -= t
                    c5_new = c5_of_h(h, dx)
                    if c5_new < best_c5 - 1e-14:
                        best_c5 = c5_new
                        improved = True
                    else:
                        h[i] -= t
                        h[j] += t
                        break
                else:
                    h[i] -= ts[-1]
                    h[j] += ts[-1]
        if not improved:
            break
    return h, best_c5

def binary_sa(h, dx, target_sum, rng, n_iter, time_limit, t0):
    n = len(h)
    h_bin = np.where(h > 0.5, 1.0, 0.0)
    n_ones = int(round(target_sum))
    if h_bin.sum() > n_ones:
        idx = np.where(h_bin == 1.0)[0]
        perm = rng.permutation(len(idx))
        for i in range(int(h_bin.sum()) - n_ones):
            h_bin[idx[perm[i]]] = 0.0
    elif h_bin.sum() < n_ones:
        idx = np.where(h_bin == 0.0)[0]
        perm = rng.permutation(len(idx))
        for i in range(n_ones - int(h_bin.sum())):
            h_bin[idx[perm[i]]] = 1.0
    cur_c5 = c5_of_h(h_bin, dx)
    best_h = h_bin.copy()
    best_c5 = cur_c5
    T = 0.02
    T_end = 0.001
    decay = (T_end / T) ** (1.0 / max(1, n_iter))
    for it in range(n_iter):
        if time.time() - t0 > time_limit:
            break
        T *= decay
        i1 = rng.randint(n)
        if h_bin[i1] != 1.0:
            continue
        i2 = rng.randint(n)
        if h_bin[i2] != 0.0:
            continue
        h_bin[i1] = 0.0
        h_bin[i2] = 1.0
        new_c5 = c5_of_h(h_bin, dx)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_bin.copy()
        else:
            h_bin[i1] = 1.0
            h_bin[i2] = 0.0
    return best_h, best_c5

def lp_optimize(h, dx, target_sum, t0, time_limit, n_iters=200):
    n = len(h)
    cur = h.copy()
    cur_c5 = c5_of_h(h, dx)
    best_h = cur.copy()
    best_c5 = cur_c5
    alpha = 0.25
    stall = 0
    for it in range(n_iters):
        if time.time() - t0 > time_limit:
            break
        corrs = corr_of_h(cur)
        max_c = corrs.max()
        margin = max(0.15, 0.05 * max_c)
        n_top = max(20, min(60, int(0.4 * (2*n - 1))))
        top_idx = np.argsort(corrs)[-n_top:]
        active_mask = corrs >= (max_c - margin)
        active_mask[top_idx] = True
        active_idx = np.where(active_mask)[0]
        if len(active_idx) < 5:
            active_idx = top_idx
        # Soft slack LP
        rho_q = 0.3 + 0.3 * (it % 3) / 2.0
        rho_s = np.ones(len(active_idx)) * (3.0 + 2.0 * (it % 2))
        direction = solve_soft_lp(cur, dx, active_idx, corrs, alpha, rho_q, rho_s)
        if direction is None:
            # Try linear sum
            wts = np.maximum(0, corrs[active_idx] - (max_c - margin))
            if wts.sum() > 0:
                wts = wts / wts.sum()
            else:
                wts = np.ones(len(active_idx)) / len(active_idx)
            direction = solve_sum_lp(cur, dx, active_idx, wts, alpha)
        if direction is None:
            alpha *= 0.7
            if alpha < 0.01:
                break
            continue
        new_h, new_c5, t_used = do_line_search(cur, cur_c5, direction, dx, target_sum)
        if t_used > 0:
            cur = new_h
            cur_c5 = new_c5
            stall = 0
            alpha = min(0.4, alpha * 1.1)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = cur.copy()
        else:
            stall += 1
            alpha *= 0.75
            if alpha < 0.01:
                break
    return best_h, best_c5

def transfer_grid(h_src, n_target):
    n_src = len(h_src)
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_tgt = (np.arange(n_target) + 0.5) * (2.0 / n_target)
    return np.interp(x_tgt, x_src, h_src)

def full_optimize(h_init, n, t0, time_budget):
    dx = 2.0 / n
    target_sum = n / 2.0
    h = project_feasible(h_init, target_sum)
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    t_limit = t0 + time_budget
    # LP phase
    h, c5 = lp_optimize(h, dx, target_sum, t0, t_limit - 30, n_iters=300)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h
    # Pair refine
    h, c5 = pair_refine(h, dx, target_sum, 3, t_limit - 20, t0)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h
    # LP after pairs
    if time.time() < t_limit - 15:
        h, c5 = lp_optimize(h, dx, target_sum, t0, t_limit - 10, n_iters=150)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h
    return best_h, best_c5, n

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = np.full(126, 0.5)
    n0 = len(h0)
    # Phase 1: Intensive optimization on parent grid
    h1, c5_1, n1 = full_optimize(h0, n0, t0, budget_s * 0.30)
    best_h, best_c5, best_n = h1, c5_1, n1
    # Phase 2: Binary SA on parent
    if time.time() - t0 < budget_s * 0.75:
        dx0 = 2.0 / n0
        tb, cb = binary_sa(best_h, dx0, n0/2.0, rng, 20000, t0 + budget_s*0.15, t0)
        if cb < best_c5:
            h2, c5_2, _ = full_optimize(tb, n0, t0, budget_s * 0.60)
            if c5_2 < best_c5:
                best_h, best_c5, best_n = h2, c5_2, n0
    # Phase 3: Grid sweep
    n_candidates = [best_n - 4, best_n - 2, best_n + 2, best_n + 4, 128, 144, 152, 156, 164, 172, 176, 184, 192]
    n_candidates = sorted(set([nc for nc in n_candidates if 64 <= nc <= 220 and nc % 2 == 0]))
    remaining = budget_s - (time.time() - t0)
    per_grid = remaining / max(1, len(n_candidates)) * 0.7
    for nc in n_candidates:
        if time.time() - t0 > budget_s * 0.90:
            break
        if nc == best_n:
            continue
        h_interp = transfer_grid(best_h, nc)
        h3, c5_3, _ = full_optimize(h_interp, nc, t0, max(10, per_grid))
        if c5_3 < best_c5:
            best_h, best_c5, best_n = h3, c5_3, nc
    # Phase 4: Final intensive polish on best
    remaining = budget_s - (time.time() - t0)
    if remaining > 20:
        dx_b = 2.0 / best_n
        ts_b = best_n / 2.0
        # Multiple LP rounds with varying params
        for r in range(3):
            if time.time() - t0 > budget_s - 10:
                break
            h4, c5_4, _ = full_optimize(best_h, best_n, t0, min(remaining/3.0, budget_s - (time.time()-t0) - 5))
            if c5_4 < best_c5:
                best_h, best_c5 = h4, c5_4
        # Pair refinement
        if time.time() - t0 < budget_s - 10:
            h5, c5_5 = pair_refine(best_h, 2.0/best_n, best_n/2.0, 5, budget_s - (time.time()-t0) - 5, t0)
            if c5_5 < best_c5:
                best_h, best_c5 = h5, c5_5
    # Final validation
    dx_f = 2.0 / best_n
    ts_f = best_n / 2.0
    best_h = project_feasible(best_h, ts_f)
    best_c5 = c5_of_h(best_h, dx_f)
    return best_h, best_c5, best_n
