def project_feasible(g, n):
    """Project g onto {0 <= h <= 1, sum h = n/2}."""
    target = n / 2.0
    
    def sum_at_lambda(lam):
        h = np.clip(g - lam, 0.0, 1.0)
        return h.sum()
    
    # Bisection
    lo = g.min() - 1.0
    hi = g.max() + 1.0
    
    # We want sum_at_lambda(lam) = target
    # sum_at_lambda is decreasing in lam
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = sum_at_lambda(mid)
        if s > target:
            lo = mid
        else:
            hi = mid
    
    lam = (lo + hi) / 2.0
    h = np.clip(g - lam, 0.0, 1.0)
    # Fix any rounding
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        # Distribute the difference
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        count = mask.sum()
        if count > 0:
            h[mask] += diff / count
            h = np.clip(h, 0.0, 1.0)
    return h

import numpy as np
from scipy.optimize import minimize
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def repair_feasible(h):
    n = len(h)
    h = np.clip(np.asarray(h, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in range(n):
            idx = order[i]
            add = min(diff, 1.0 - h[idx])
            h[idx] += add
            diff -= add
            if diff <= 1e-12:
                break
    else:
        order = np.argsort(h)[::-1]
        diff = -diff
        for i in range(n):
            idx = order[i]
            rem = min(diff, h[idx])
            h[idx] -= rem
            diff -= rem
            if diff <= 1e-12:
                break
    return np.clip(h, 0.0, 1.0)

def project_feasible(g):
    """Project onto {0<=h<=1, sum h = n/2} via bisection on Lagrange multiplier."""
    n = len(g)
    target = n / 2.0
    
    def s_lam(lam):
        return np.clip(g - lam, 0.0, 1.0).sum()
    
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if s_lam(mid) > target:
            lo = mid
        else:
            hi = mid
    
    lam = (lo + hi) / 2.0
    h = np.clip(g - lam, 0.0, 1.0)

        else:

        cnt = mask.sum()

def profile_gradient(h):
    """Compute M x n matrix of partial derivatives of profile wrt h."""
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    G = np.zeros((M, n))
    center = n - 1
    G[center, :] = dx * (1.0 - 2.0 * h)
    for d in range(1, n):
        G[center + d, 0:n-d] += dx * (1.0 - h[d:n])
        G[center + d, d:n] += -dx * h[0:n-d]
        G[center - d, 0:n-d] += -dx * h[d:n]
        G[center - d, d:n] += dx * (1.0 - h[0:n-d])
    return G

def smooth_gradient(h, tau):
    """Compute gradient of log-mean-exp of profile."""
    n = len(h)
    profile = compute_profile(h)
    G = profile_gradient(h)
    # Weights: softmax of profile/tau
    shifted = (profile - np.max(profile)) / tau
    weights = np.exp(shifted)
    weights /= weights.sum()
    return G.T @ weights

    best_h = h.copy()
    best_c = exact_score(h)

            elapsed = time.perf_counter() - t_start
            if budget_s - elapsed < 10:

            grad = smooth_gradient(h, tau)
            # Center the gradient (subtract mean to preserve sum constraint approximately)
            grad -= grad.mean()

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def repair_feasible(h):
    n = len(h)
    h = np.clip(np.asarray(h, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in range(n):
            idx = order[i]
            add = min(diff, 1.0 - h[idx])
            h[idx] += add
            diff -= add
            if diff <= 1e-12:
                break
    else:
        order = np.argsort(h)[::-1]
        diff = -diff
        for i in range(n):
            idx = order[i]
            rem = min(diff, h[idx])
            h[idx] -= rem
            diff -= rem
            if diff <= 1e-12:
                break
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-15:
        if diff > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def project_feasible(g):
    n = len(g)
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = np.clip(g - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h = np.clip(g - lam, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt > 0:
            h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def profile_gradient(h):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    G = np.zeros((M, n))
    center = n - 1
    G[center, :] = dx * (1.0 - 2.0 * h)
    for d in range(1, n):
        G[center + d, 0:n-d] += dx * (1.0 - h[d:n])
        G[center + d, d:n] += -dx * h[0:n-d]
        G[center - d, 0:n-d] += -dx * h[d:n]
        G[center - d, d:n] += dx * (1.0 - h[0:n-d])
    return G

def smooth_grad(h, tau):
    profile = compute_profile(h)
    G = profile_gradient(h)
    shifted = (profile - np.max(profile)) / tau
    w = np.exp(shifted)
    w /= w.sum()
    return G.T @ w

def lag_grad(h, G, lag_idx):
    g = G[lag_idx, :].copy()
    g -= g.mean()
    return g

def proj_step(h, grad, alpha0):
    c0 = exact_score(h)
    alpha = alpha0
    for _ in range(35):
        h_try = project_feasible(h - alpha * grad)
        c_try = exact_score(h_try)
        if c_try < c0 - 1e-15:
            return h_try, c_try
        alpha *= 0.5
        if alpha < 1e-8:
            break
    return None, c0

def interp_h(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    return repair_feasible(h_new)

def slsqp_topk(h, k, maxiter):
    n = len(h)
    dx = 2.0 / n
    prof = compute_profile(h)
    idx = np.argsort(prof)[::-1][:k]
    c0 = exact_score(h)

    def obj(x):
        return x[-1]

    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g

    def eq_c(x):
        return x[:-1].sum() - n / 2.0

    def eq_j(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g

    def ineq_c(x):
        h = np.clip(x[:-1], 0, 1)
        t = x[-1]
        p = np.correlate(h, 1 - h, mode="full") * dx
        return t - p[idx]

    def ineq_j(x):
        h = np.clip(x[:-1], 0, 1)
        G = profile_gradient(h)
        J = np.zeros((k, n + 1))
        J[:, :-1] = -G[idx, :]
        J[:, -1] = 1.0
        return J

    x0 = np.concatenate([h, [c0 + 0.0005]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    cons = [
        {'type': 'eq', 'fun': eq_c, 'jac': eq_j},
        {'type': 'ineq', 'fun': ineq_c, 'jac': ineq_j}
    ]
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.x is not None and np.isfinite(res.x).all():
            h_sol = repair_feasible(res.x[:-1])
            return h_sol, exact_score(h_sol)
    except Exception:
        pass
    return h.copy(), c0

def slsqp_full(h, maxiter):
    n = len(h)
    M = 2 * n - 1
    dx = 2.0 / n
    c0 = exact_score(h)

    def obj(x):
        return x[-1]

    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g

    def eq_c(x):
        return x[:-1].sum() - n / 2.0

    def eq_j(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g

    def ineq_c(x):
        h = np.clip(x[:-1], 0, 1)
        t = x[-1]
        p = np.correlate(h, 1 - h, mode="full") * dx
        return t - p

    def ineq_j(x):
        h = np.clip(x[:-1], 0, 1)
        G = profile_gradient(h)
        J = np.zeros((M, n + 1))
        J[:, :-1] = -G
        J[:, -1] = 1.0
        return J

    x0 = np.concatenate([h, [c0 + 0.0005]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    cons = [
        {'type': 'eq', 'fun': eq_c, 'jac': eq_j},
        {'type': 'ineq', 'fun': ineq_c, 'jac': ineq_j}
    ]
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.x is not None and np.isfinite(res.x).all():
            h_sol = repair_feasible(res.x[:-1])
            return h_sol, exact_score(h_sol)
    except Exception:
        pass
    return h.copy(), c0

def lp_active(h, active, radius):
    n = len(h)
    G = profile_gradient(h)
    prof = compute_profile(h)
    G_a = G[active, :]
    p_a = prof[active]
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_ub = np.zeros((len(active), n + 1))
    A_ub[:, :n] = G_a
    A_ub[:, -1] = -1.0
    b_ub = -p_a
    A_eq = np.array([[np.ones(n), 0.0]])
    b_eq = np.zeros(1)
    lower = np.maximum(-h, -radius)
    upper = np.minimum(1.0 - h, radius)
    bounds = list(zip(lower, upper)) + [(0.0, None)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'maxiter': 3000})
        if res.success and res.x is not None and np.isfinite(res.x).all():
            return repair_feasible(h + res.x[:n])
    except Exception:
        pass
    return None

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()
    rng = np.random.RandomState(seed)

    try:
        parent = np.array(initial_h_values, dtype=np.float64)
    except Exception:
        parent = np.full(70, 0.5)
    if parent is None or len(parent) < 4 or not np.isfinite(parent).all():
        parent = np.full(70, 0.5)
    parent = repair_feasible(parent)

    best_h = parent.copy()
    best_c = exact_score(parent)

    # Try symmetrization
    h_sym = repair_feasible((parent + parent[::-1]) / 2.0)
    c_sym = exact_score(h_sym)
    if c_sym < best_c - 1e-12:
        best_h = h_sym
        best_c = c_sym

    # Phase 1: Annealed smoothed gradient
    h_cur = best_h.copy()
    n = len(h_cur)
    tau_list = [0.04, 0.025, 0.015, 0.01, 0.007, 0.004, 0.0025, 0.0015, 0.001, 0.0006, 0.0003]
    for tau in tau_list:
        elapsed = time.perf_counter() - t_start
        if budget_s - elapsed < 30:
            break
        n_steps = 100 if tau > 0.01 else 150
        for _ in range(n_steps):
            if time.perf_counter() - t_start > budget_s - 20:
                break
            grad = smooth_grad(h_cur, tau)
            grad -= grad.mean()
            h_new, c_new = proj_step(h_cur, grad, 3 * tau)
            if h_new is not None:
                h_cur = h_new
                if c_new < best_c - 1e-14:
                    best_c = c_new
                    best_h = h_cur.copy()

    # Phase 2: Mixed active-set LP + gradient refinement
    h_cur = best_h.copy()
    n = len(h_cur)
    stuck = 0
    for it in range(800):
        if time.perf_counter() - t_start > budget_s - 50:
            break
        prof = compute_profile(h_cur)
        c_cur = float(np.max(prof))
        if c_cur < best_c - 1e-14:
            best_c = c_cur
            best_h = h_cur.copy()
            stuck = 0
        else:
            stuck += 1
        if stuck > 60:
            break

        T = np.max(prof)
        thresh = T - max(0.003, 0.08 * (T - np.median(prof)))
        active = np.where(prof >= thresh)[0]
        if len(active) == 0:
            active = np.array([np.argmax(prof)])
        if len(active) > 12:
            top_idx = np.argsort(prof[active])[-12:]
            active = active[top_idx]

        improved = False
        # LP step
        h_lp = lp_active(h_cur, active, 0.025)
        if h_lp is not None:
            c_lp = exact_score(h_lp)
            if c_lp < c_cur - 1e-14:
                h_cur = h_lp
                improved = True

        # Top lag gradient
        if not improved:
            G = profile_gradient(h_cur)
            max_lag = int(np.argmax(prof))
            g = lag_grad(h_cur, G, max_lag)
            h_new, c_new = proj_step(h_cur, g, 0.04)
            if h_new is not None:
                h_cur = h_new
                improved = True

        # Top-3 lags
        if not improved:
            G = profile_gradient(h_cur)
            top3 = np.argsort(prof)[::-1][:3]
            for lag in top3:
                g = lag_grad(h_cur, G, int(lag))
                h_new, c_new = proj_step(h_cur, g, 0.03)
                if h_new is not None:
                    h_cur = h_new
                    improved = True
                    break

        if not improved:
            # Smooth gradient fallback
            grad = smooth_grad(h_cur, 0.002)
            grad -= grad.mean()
            h_new, c_new = proj_step(h_cur, grad, 0.01)
            if h_new is not None:
                h_cur = h_new
            else:
                break

    if exact_score(h_cur) < best_c - 1e-14:
        best_c = exact_score(h_cur)
        best_h = h_cur.copy()

    # Phase 3: Progressive SLSQP
    if time.perf_counter() - t_start < budget_s - 40:
        n = len(best_h)
        for k in [8, 15, 25, 40, 2 * n - 1]:
            if time.perf_counter() - t_start > budget_s - 20:
                break
            h_sol, c_sol = slsqp_topk(best_h, min(k, 2*n-1), 200)
            if c_sol < best_c - 1e-14:
                best_c = c_sol
                best_h = h_sol

    # Phase 4: Fine micro-steps
    h_cur = best_h.copy()
    n = len(h_cur)
    stuck = 0
    for it in range(1000):
        if time.perf_counter() - t_start > budget_s - 15:
            break
        prof = compute_profile(h_cur)
        c_cur = float(np.max(prof))
        if c_cur < best_c - 1e-14:
            best_c = c_cur
            best_h = h_cur.copy()
            stuck = 0
        else:
            stuck += 1
        if stuck > 80:
            break
        G = profile_gradient(h_cur)
        top_lags = np.argsort(prof)[::-1][:5]
        improved = False
        for lag in top_lags:
            g = lag_grad(h_cur, G, int(lag))
            h_new, c_new = proj_step(h_cur, g, 0.008)
            if h_new is not None:
                h_cur = h_new
                improved = True
                break
        if not improved:
            grad = smooth_grad(h_cur, 0.0005)
            grad -= grad.mean()
            h_new, c_new = proj_step(h_cur, grad, 0.004)
            if h_new is not None:
                h_cur = h_new
            else:
                break

    if exact_score(h_cur) < best_c - 1e-14:
        best_c = exact_score(h_cur)
        best_h = h_cur.copy()

    # Phase 5: Try nearby dimensions
    n0 = len(best_h)
    for new_n in [n0 - 2, n0 + 2, n0 - 4, n0 + 4]:
        if time.perf_counter() - t_start > budget_s - 30:
            break
        if new_n < 20 or new_n > 200:
            continue
        h_interp = interp_h(best_h, new_n)
        c_interp = exact_score(h_interp)
        if c_interp > best_c + 0.003:
            continue
        h_cur2 = h_interp.copy()
        for it2 in range(300):
            if time.perf_counter() - t_start > budget_s - 15:
                break
            prof2 = compute_profile(h_cur2)
            c2 = float(np.max(prof2))
            if c2 < best_c - 1e-14:
                best_c = c2
                best_h = h_cur2.copy()
            G2 = profile_gradient(h_cur2)
            ml = int(np.argmax(prof2))
            g2 = lag_grad(h_cur2, G2, ml)
            h_new2, c_new2 = proj_step(h_cur2, g2, 0.03)
            if h_new2 is None:
                break
            h_cur2 = h_new2
        # SLSQP at this dim
        if time.perf_counter() - t_start < budget_s - 10:
            h_sol2, c_sol2 = slsqp_topk(h_cur2, min(20, 2*new_n-1), 150)
            if c_sol2 < best_c - 1e-14:
                best_c = c_sol2
                best_h = h_sol2

    # Phase 6: Final SLSQP full
    if time.perf_counter() - t_start < budget_s - 5:
        h_sol, c_sol = slsqp_full(best_h, 200)
        if c_sol < best_c - 1e-14:
            best_c = c_sol
            best_h = h_sol

    # Final validation
    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)

    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            final_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 2:
            final_c = float(eval_result[0])
    except Exception:
        pass

    return best_h, final_c, len(best_h)
