def coordinate_descent(h, max_sweeps=15, t_start=None, budget_s=1100):

    h = h.copy()
    best_c = exact_score(h)
    
    for sweep in range(max_sweeps):

            break
        for i in range(n):
            if t_start is not None and time.perf_counter() - t_start > budget_s - 15:
                break
            h_save = h[i]
            for delta in [0.005, -0.005, 0.002, -0.002, 0.001, -0.001]:

                    best_c = s

def interpolate_to_n(h, new_n):
    n = len(h)

import numpy as np
from scipy.optimize import linprog, minimize
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full"))) * dx

def profile_h(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx])
            h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def build_gradient_matrix(h, n):
    dx = 2.0 / n
    M = 2 * n - 1
    G = np.zeros((M, n))
    for i_out in range(M):
        k = i_out - (n - 1)
        if k == 0:
            G[i_out, :] = dx * (1.0 - 2.0 * h)
        elif k > 0:
            for i in range(k, n):
                G[i_out, i] += dx * (1.0 - h[i - k])
            for i in range(n - k):
                G[i_out, i] -= dx * h[i + k]
        else:
            dl = -k
            for i in range(n - dl):
                G[i_out, i] += dx * (1.0 - h[i + dl])
            for i in range(dl, n):
                G[i_out, i] -= dx * h[i - dl]

def lp_direction(h, active, max_step):
    n = len(h)
    M = 2 * n - 1
    G = build_gradient_matrix(h, n)
    Ga = G[active, :]
    c = np.concatenate([np.zeros(n), [1.0]])
    A_ub = np.zeros((len(active), n + 1))
    A_ub[:, :n] = Ga
    A_ub[:, n] = -1.0
    b_ub = np.zeros(len(active))
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(-max_step, max_step)] * n + [(None, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0 and res.x is not None:
        d = res.x[:n]
        if np.all(np.isfinite(d)):
            return d, float(res.x[n])
    return None, 0.0

def lp_optimize(h, t_start, budget_s, max_iters=300, margin=25):
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    band_width = 0.003
    max_step = 0.035
    stall_count = 0

    for it in range(max_iters):
        if time.perf_counter() - t_start > budget_s - margin:
            break
        P = profile_h(h)
        C = P.max()
        thresh = C - band_width
        active = np.where(P >= thresh)[0]
        if len(active) == 0:
            active = np.array([np.argmax(P)])
        if len(active) > 60:
            idx = np.argsort(P[active])[-60:]
            active = active[idx]

        d, slope = lp_direction(h, active, max_step)
        if d is None:
            break
        if slope >= -1e-10:
            band_width *= 1.5
            max_step *= 0.85
            stall_count += 1
            if stall_count > 8 or band_width > 0.2:
                break
            continue

        improved = False
        for frac in [1.0, 0.75, 0.55, 0.4, 0.25, 0.15, 0.08]:
            h_new = repair_feasible(h + frac * d)

                h = h_new
                best_c = c_new
                improved = True
                break

            max_step = min(max_step * 1.03, 0.1)
        else:
            band_width *= 1.2
            max_step *= 0.7
            stall_count += 1

    return h, best_c

def slsqp_optimize(h0, maxiter=300, active_k=18, tol=1e-12):
    n = len(h0)
    dx = 2.0 / n
    P = profile_h(h0)
    c0 = P.max()
    thresh = c0 - max(0.004, 0.05 * (c0 - np.median(P)))
    active = np.where(P >= thresh)[0]
    if len(active) == 0:
        active = np.array([np.argmax(P)])
    if len(active) > active_k:
        top_idx = np.argsort(P[active])[-active_k:]
        active = active[top_idx]

    def obj(x):
        return x[-1]
    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    def eq_fun(x):
        return x[:-1].sum() - n / 2.0
    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g
    def ineq_fun(x):
        prof = profile_h(x[:-1])
        return x[-1] - prof[active]
    def ineq_jac(x):

        G = build_gradient_matrix(h, n)

        J[:, n] = 1.0
        return J

    x0 = np.concatenate([h0, [c0 + 1e-4]])
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 1.0)]

def coordinate_descent(h, max_sweeps=12, t_start=None, budget_s=1100):

                    h = h_rep

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    return repair_feasible(np.clip(h_new, 0.0, 1.0))

                    h_lp, c_lp = lp_optimize(h_mix, t_start, budget_s, max_iters=100)
                    if c_lp < best_c:
                        best_h, best_c = h_lp, c_lp

    # Phase 7: Second round LP + SLSQP
    if time_ok(50):
        h3, c3 = lp_optimize(best_h, t_start, budget_s, max_iters=200)
        if c3 < best_c:
            best_h, best_c = h3, c3
        if time_ok(30):
            sol = slsqp_optimize(best_h, maxiter=300, active_k=20, tol=1e-13)
            if sol is not None:
                c = exact_score(sol)
                if c < best_c:
                    best_h, best_c = sol, c

    # Phase 8: Symmetric refinement
    if time_ok(40):
        h_sym = repair_feasible(symmetrize(best_h))
        c_sym = exact_score(h_sym)
        if c_sym < best_c + 0.0005:
            h_lp, c_lp = lp_optimize(h_sym, t_start, budget_s, max_iters=150)
            if c_lp < best_c:
                best_h, best_c = h_lp, c_lp

    # Phase 9: Coordinate descent polish
    if time_ok(30):
        best_h, best_c = coordinate_descent(best_h, max_sweeps=10,
                                            t_start=t_start, budget_s=budget_s)

    # Phase 10: Final LP
    if time_ok(15):
        h_final, c_final = lp_optimize(best_h, t_start, budget_s, max_iters=80)
        if c_final < best_c:
            best_h, best_c = h_final, c_final

    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)

    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            final_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 1:
            if np.isfinite(eval_result[0]):
                final_c = float(eval_result[0])
    except Exception:
        pass

    return best_h, final_c, len(best_h)
