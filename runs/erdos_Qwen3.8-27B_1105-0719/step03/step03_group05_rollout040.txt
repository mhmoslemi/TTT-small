    #   h[j] appears as "(1-h)" factor in pairs where i+d=j: contributes -h[i]*dx

Now, the question is: **which pairs (j, k) to update?**

Options:
1. Cyclic: for each j, update with k = (j+1) mod n.

3. Random: pick random pairs.
4. Greedy with limited search: for each j, try a few k values and pick the best.

The greedy approach (option 2) is O(n^2 * M^2) per iteration, which is 70^2 * 139^2 ≈ 95 million. That might be too slow for many iterations.

Actually, let me think about this more carefully. The pair update method is essentially a form of **coordinate descent** for the non-convex problem. It's guaranteed to converge to a local minimum (under some conditions). The question is: will it find a better local minimum than the gradient-based methods?

One advantage of coordinate descent is that each step is exact (for that coordinate pair), so there's no step-size tuning. Another advantage is that it naturally maintains feasibility (the sum constraint is maintained, and we clip to [0,1]).

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx] + 1e-15)
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx] + 1e-15)
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-14:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmin(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_A_for_pair(h, j, k):
    """Compute A[m] = d(g_m)/d(h[j]) - d(g_m)/d(h[k]) for all m."""
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    A = np.zeros(M)
    for m in range(M):
        d = m - (n - 1)
        dj = 0.0
        dk = 0.0
        if d >= 0:
            # pairs (i, i+d) for i=0..n-d-1
            if 0 <= j < n - d:
                dj += dx * (1.0 - h[j + d])
            if d <= j < n:
                dj += -dx * h[j - d]
            if 0 <= k < n - d:
                dk += dx * (1.0 - h[k + d])
            if d <= k < n:
                dk += -dx * h[k - d]
        else:
            d_abs = -d
            if d_abs <= j < n:
                dj += dx * (1.0 - h[j - d_abs])
            if 0 <= j < n - d_abs:
                dj += -dx * h[j + d_abs]
            if d_abs <= k < n:
                dk += dx * (1.0 - h[k - d_abs])
            if 0 <= k < n - d_abs:
                dk += -dx * h[k + d_abs]
        A[m] = dj - dk
    return A

def find_best_delta(A, profile, lo, hi):
    """Find delta in [lo,hi] minimizing max_m (A[m]*delta + profile[m])."""
    M = len(A)
    candidates = np.array([lo, hi])
    
    # Add intersection points (vectorized)
    i_idx, j_idx = np.triu_indices(M, k=1)
    denom = A[i_idx] - A[j_idx]
    valid = np.abs(denom) > 1e-15
    if np.any(valid):
        d_vals = (profile[j_idx[valid]] - profile[i_idx[valid]]) / denom[valid]
        in_range = (d_vals >= lo - 1e-10) & (d_vals <= hi + 1e-10)
        if np.any(in_range):
            candidates = np.concatenate([candidates, d_vals[in_range]])
    
    candidates = np.clip(candidates, lo, hi)
    # Evaluate
    vals = np.max(A[None, :] * candidates[:, None] + profile[None, :], axis=1)
    best_i = np.argmin(vals)
    return float(candidates[best_i]), float(vals[best_i])

def pair_cd_sweep(h, n_k=3, max_sweeps=50):
    """Cyclic coordinate descent with pair updates."""
    n = len(h)
    best_c = exact_score(h)
    h = h.copy()
    
    for sweep in range(max_sweeps):
        improved = False
        for j in range(n):
            # Try a few k values
            best_delta = 0.0
            best_val = best_c
            profile = compute_profile(h)
            for k in [(j+1) % n, (j+2) % n, (j+3) % n]:
                if k == j:
                    continue
                lo = max(-h[j], h[k] - 1.0)
                hi = min(1.0 - h[j], h[k])
                if hi - lo < 1e-14:
                    continue
                A = compute_A_for_pair(h, j, k)
                delta, val = find_best_delta(A, profile, lo, hi)
                if val < best_val - 1e-14:

                    best_delta = delta
                    best_k = k
            if best_val < best_c - 1e-14:
                h[j] += best_delta
                h[best_k] -= best_delta
                h = np.clip(h, 0.0, 1.0)
                best_c = max(best_c, best_val)  # Actually, we need to recompute
                best_c = exact_score(h)
                improved = True
        if not improved:
            break
    return h, best_c

                    A[m] = dj - dk
                # Add quadratic terms for shifts where j,k are in same pair
                delta_jk = k - j
                if 0 < delta_jk < n:
                    m_idx = (n - 1) + delta_jk
                    if m_idx < M:
                        A[m_idx] = A[m_idx]  # linear part already computed
                        # The quadratic term is +dx*delta^2, handled separately
                if -n < delta_jk < 0:
                    m_idx = (n - 1) + delta_jk
                    if 0 <= m_idx < M:
                        pass
                # Find best delta considering linear + quadratic
                # For simplicity, treat as linear (the quadratic correction is small)
                candidates = np.array([lo, hi])
                i_idx, j_idx = np.triu_indices(M, k=1)
                denom = A[i_idx] - A[j_idx]
                valid = np.abs(denom) > 1e-15
                if np.any(valid):
                    d_vals = (profile[j_idx[valid]] - profile[i_idx[valid]]) / denom[valid]
                    in_range = (d_vals >= lo - 1e-10) & (d_vals <= hi + 1e-10)
                    if np.any(in_range):
                        candidates = np.concatenate([candidates, d_vals[in_range]])
                candidates = np.clip(candidates, lo, hi)
                if len(candidates) > 500:
                    # Subsample for speed
                    perm = np.random.permutation(len(candidates))[:500]
                    candidates = candidates[perm]
                vals = np.max(A[None, :] * candidates[:, None] + profile[None, :], axis=1)
                bi = np.argmin(vals)
                if float(vals[bi]) < best_val - 1e-14:
                    best_val = float(vals[bi])
                    best_delta = float(candidates[bi])
                    best_k = k
            if best_k >= 0:
                h[j] += best_delta
                h[best_k] -= best_delta
                h = np.clip(h, 0.0, 1.0)
                new_c = exact_score(h)
                if new_c < best_c:
                    best_c = new_c
                sweep_improved = True
        if not sweep_improved:
            break
    return h, best_c

def binary_local_search(h, max_iters=3000, rng=None):
    n = len(h)
    dx = 2.0 / n
    half = n // 2
    idx = np.argsort(h)[::-1][:half]
    bits = np.zeros(n, dtype=np.int64)
    bits[idx] = 1
    ones = set(idx.tolist())
    zeros = set(range(n)) - ones
    
    def bscore():
        a = bits.astype(np.float64)
        return float(np.max(np.correlate(a, 1 - a, mode="full") * dx))
    
    cur = bscore()
    best_bits = bits.copy()
    best_s = cur
    ones_list = list(ones)
    zeros_list = list(zeros)
    
    for it in range(max_iters):
        found = False
        for attempt in range(8):
            j = ones_list[rng.randint(len(ones_list))]
            k = zeros_list[rng.randint(len(zeros_list))]
            bits[j], bits[k] = 0, 1
            s = bscore()
            if s < cur - 1e-15:
                cur = s
                ones.add(j); ones.remove(k)
                zeros.add(k); zeros.remove(j)
                ones_list = list(ones); zeros_list = list(zeros)
                found = True
                if cur < best_s:
                    best_s = cur
                    best_bits = bits.copy()
                break
            else:
                bits[j], bits[k] = 1, 0
        if not found:
            if rng.rand() < 0.05:
                j = ones_list[rng.randint(len(ones_list))]
                k = zeros_list[rng.randint(len(zeros_list))]
                bits[j], bits[k] = 0, 1
                cur = bscore()
                ones.add(j); ones.remove(k)
                zeros.add(k); zeros.remove(j)
                ones_list = list(ones); zeros_list = list(zeros)
                if cur < best_s:
                    best_s = cur
                    best_bits = bits.copy()
            else:
                break
    return best_bits, best_s

def slsqp_solve(h0, maxiter=200):
    n = len(h0)
    dx = 2.0 / n
    c0 = exact_score(h0)
    def obj(x):
        return x[-1]
    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    def eq_fun(x):
        return x[:-1].sum() - n / 2.0
    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g
    def ineq_fun(x):
        h = np.clip(x[:-1], 0, 1)
        t = x[-1]
        profile = np.correlate(h, 1 - h, mode="full") * dx
        return t - profile
    def ineq_jac(x):
        h = np.clip(x[:-1], 0, 1)
        G = build_grad_matrix(h)
        J = np.zeros((2*n - 1, n + 1))
        J[:, :-1] = -G
        J[:, -1] = 1.0
        return J
    x0 = np.concatenate([h0, [c0 + 1e-3]])
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 0.5)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None and np.isfinite(res.x).all():
            h_new = repair_feasible(res.x[:-1])
            c_new = exact_score(h_new)
            if c_new <= c0 + 1e-9:
                return h_new, c_new
    except Exception:
        pass
    return None, None

def symmetrize(h):
    return repair_feasible((h + h[::-1]) / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()
    rng = np.random.RandomState(seed)
    
    def time_left():
        return budget_s - (time.perf_counter() - t_start)
    
    parent = None
    try:
        parent = np.array(initial_h_values, dtype=np.float64)
    except NameError:
        pass
    except Exception:
        pass
    
    if parent is None or len(parent) < 4:
        parent = np.full(70, 0.5)
    
    parent = np.clip(parent, 0.0, 1.0)
    parent = repair_feasible(parent)
    best_h = parent.copy()
    best_c = exact_score(parent)
    
    # Try symmetrization
    h_sym = symmetrize(parent)
    c_sym = exact_score(h_sym)
    if c_sym < best_c - 1e-14:
        best_h = h_sym
        best_c = c_sym
    
    # Stage 1: Pair CD (exact coordinate descent)
    if time_left() > 300:
        t0 = time.perf_counter()
        h_cd, c_cd = pair_cd_sweep(best_h, max_sweeps=100, k_range=3,
                                   time_limit=min(200, time_left()-50), t_start=t0)
        if c_cd < best_c - 1e-14:
            best_h = h_cd.copy()
            best_c = c_cd
    
    # Stage 2: LSE descent
    if time_left() > 200:
        h_lse, c_lse = lse_descent(best_h, beta=80, max_iters=100, alpha_init=0.003)
        if c_lse < best_c - 1e-14:
            best_h = h_lse.copy()
            best_c = c_lse
    
    # Stage 3: More pair CD after LSE
    if time_left() > 150:
        h_cd, c_cd = pair_cd_sweep(best_h, max_sweeps=100, k_range=4,
                                   time_limit=min(100, time_left()-30), t_start=time.perf_counter())
        if c_cd < best_c - 1e-14:
            best_h = h_cd.copy()
            best_c = c_cd
    
    # Stage 4: SLSQP
    if time_left() > 100:
        h_sol, c_sol = slsqp_solve(best_h, maxiter=200)
        if h_sol is not None and c_sol < best_c - 1e-13:
            best_h = h_sol.copy()
            best_c = c_sol
    
    # Stage 5: Binary local search + continuous refinement
    if time_left() > 200:
        n_b = len(best_h)
        bits, bscore = binary_local_search(best_h, max_iters=3000, rng=rng)
        # Continuous relaxation from binary
        h_bin = bits.astype(np.float64)
        h_bin = repair_feasible(h_bin)
        # Add small perturbation and optimize
        for pert in [0.01, 0.03, 0.05]:
            if time_left() < 30:
                break
            h_pert = repair_feasible(h_bin + pert * (rng.rand(n_b) - 0.5))
            h_lse, c_lse = lse_descent(h_pert, beta=100, max_iters=80)
            if c_lse < best_c - 1e-14:
                best_h = h_lse.copy()
                best_c = c_lse
            h_cd, c_cd = pair_cd_sweep(best_h, max_sweeps=50, k_range=3,
                                       time_limit=min(40, time_left()-20), t_start=time.perf_counter())
            if c_cd < best_c - 1e-14:
                best_h = h_cd.copy()
                best_c = c_cd
    
    # Stage 6: Try different resolutions
    if time_left() > 150:
        n_cur = len(best_h)
        for dn in [-2, 2, -4, 4, 6, -6]:
            if time_left() < 30:
                break
            new_n = n_cur + dn
            if new_n < 30 or new_n > 120 or new_n % 2 != 0:
                continue
            h_interp = interp_to_n(best_h, new_n)
            h_lse, c_lse = lse_descent(h_interp, beta=80, max_iters=60, alpha_init=0.003)
            if c_lse < best_c - 1e-14:
                best_h = h_lse.copy()
                best_c = c_lse
            h_cd, c_cd = pair_cd_sweep(best_h, max_sweeps=50, k_range=3,
                                       time_limit=min(30, time_left()-20), t_start=time.perf_counter())
            if c_cd < best_c - 1e-14:
                best_h = h_cd.copy()
                best_c = c_cd
    
    # Stage 7: Multi-start with perturbations
    if time_left() > 100:
        n_b = len(best_h)
        for eps in [0.005, 0.015, 0.03]:
            if time_left() < 30:
                break
            z = rng.randn(n_b)
            z -= z.mean()
            h_start = repair_feasible(best_h + eps * z)
            h_lse, c_lse = lse_descent(h_start, beta=100, max_iters=60, alpha_init=0.004)
            if c_lse < best_c - 1e-14:
                best_h = h_lse.copy()
                best_c = c_lse
    
    # Stage 8: Final polish - alternating pair CD and LSE
    if time_left() > 60:
        for _ in range(3):
            if time_left() < 20:
                break
            h_cd, c_cd = pair_cd_sweep(best_h, max_sweeps=80, k_range=5,
                                       time_limit=min(30, time_left()-15), t_start=time.perf_counter())
            if c_cd < best_c - 1e-15:
                best_h = h_cd.copy()
                best_c = c_cd
            h_lse, c_lse = lse_descent(best_h, beta=150, max_iters=80, alpha_init=0.002)
            if c_lse < best_c - 1e-15:
                best_h = h_lse.copy()
                best_c = c_lse
    
    # Stage 9: Final SLSQP
    if time_left() > 40:
        h_sol, c_sol = slsqp_solve(best_h, maxiter=300)
        if h_sol is not None and c_sol < best_c - 1e-14:
            best_h = h_sol.copy()
            best_c = c_sol
    
    # Final validation
    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)
    
    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            final_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 2:
            final_c = float(eval_result[0])
    except Exception:
        pass
    
    return best_h, final_c, len(best_h)
