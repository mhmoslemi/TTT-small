def c5_of_h(h, dx):
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def project_feasible(z, target):
    n = len(z)
    z = np.where(np.isfinite(z), z, 0.5)
    lo = z.min() - 1.0
    hi = z.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.clip(z - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        mask = (h > 1e-8) & (h < 1.0 - 1e-8)
        cnt = mask.sum()
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_grad(h, q):

    corr = np.correlate(h, 1.0 - h, mode='full')
