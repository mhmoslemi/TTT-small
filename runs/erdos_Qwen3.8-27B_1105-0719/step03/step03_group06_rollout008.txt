        return g
    else:
        p = -d
        g = np.zeros(n)
        g[:-p] += dx * (1.0 - h[p:])
        g[p:] -= dx * h[:-p]
        return g

def interp_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    edges_old = np.linspace(0.0, 2.0, n + 1)
    edges_new = np.linspace(0.0, 2.0, new_n + 1)
    h_new = np.zeros(new_n)
    for k in range(new_n):
        a, b = edges_new[k], edges_new[k + 1]
        total = 0.0
        for j in range(n):
            lo = max(a, edges_old[j])
            hi = min(b, edges_old[j + 1])
            if hi > lo:
                total += h[j] * (hi - lo)
        h_new[k] = total / (b - a)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

def lp_trust_descent(h, delta, max_iters=200, tol=1e-12):
    """LP-based trust region descent for the max-overlap problem."""
    n = len(h)
    M = 2 * n - 1
    h = h.copy()
    best_c = exact_score(h)
    
    for it in range(max_iters):
        profile = compute_profile(h)
        c = float(profile.max())
        
        # Solve LP: min tau s.t. G_m . d - tau <= -P_m, sum(d)=0, bounds on d
        # Variables: d_0, ..., d_{n-1}, tau (total n+1)
        A_ub = np.zeros((M, n + 1))
        b_ub = np.zeros(M)
        for m in range(M):
            A_ub[m, :n] = grad_row(h, m, n)
            A_ub[m, n] = -1.0
            b_ub[m] = -profile[m]
        
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([0.0])
        
        lo = np.maximum(-delta, -h + 1e-12)
        hi = np.minimum(delta, 1.0 - h - 1e-12)
        bounds = list(zip(lo, hi))
        bounds.append((0.0, c + 10.0))  # tau is free (lower bounded by 0)
        
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0
        
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs')
        except Exception:
            try:
                res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                             bounds=bounds, method='simplex')
            except Exception:
                break
        
        if res.status != 0:
            delta *= 0.7
            if delta < 1e-5:
                break
            continue
        
        d_step = res.x[:n]
        d_norm = np.linalg.norm(d_step)
        if d_norm < 1e-10:
            break
        
        # Line search along d_step
        # Find max alpha such that h + alpha*d stays in [0,1]
        max_alpha = 1.0
        pos = d_step > 1e-12
        neg = d_step < -1e-12
        if pos.any():
            max_alpha = min(max_alpha, float(((1.0 - h[pos]) / d_step[pos]).min()))
        if neg.any():
            max_alpha = min(max_alpha, float((h[neg] / (-d_step[neg])).min()))
        max_alpha = max(max_alpha, 1e-10)
        max_alpha = min(max_alpha, 1.0)
        
        # Try several alphas
        best_alpha = 0.0
        best_val = c
        best_h = h.copy()
        alphas = np.linspace(0.1, max_alpha, 10)
        for alpha in alphas:
            h_try = repair_feasible(h + alpha * d_step)
            val = exact_score(h_try)
            if val < best_val - 1e-14:
                best_val = val
                best_alpha = alpha
                best_h = h_try
        
        if best_val >= c - 1e-14:
            # Try smaller alpha
            alphas_small = np.linspace(0.01, 0.1, 5)
            for alpha in alphas_small:
                h_try = repair_feasible(h + alpha * d_step)
                val = exact_score(h_try)
                if val < best_val - 1e-14:
                    best_val = val
                    best_alpha = alpha
                    best_h = h_try
        
        if best_val >= c - 1e-14:
            delta *= 0.6
            if delta < 1e-5:
                break
            continue
        
        h = best_h
        if best_val < best_c:
            best_c = best_val
        # Adapt delta
        if best_alpha >= 0.8 * max_alpha:
            delta = min(2.0 * delta, 0.3)
        else:
            delta = min(1.2 * delta, 0.25)
    
    return h, best_c

def sturmian_seed(n, slope_idx=156, rng=None):
    """Generate a Sturmian word (mechanical word) of length n."""
    h = np.zeros(n)
    # Use golden ratio as slope
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    slope = 1.0 / phi  # ≈ 0.618...
    # Actually for density 1/2, use a different approach
    # Use Beatty sequence: h[i] = 1 if {i * alpha} < beta
    # For density 1/2, choose alpha irrational, beta = 1/2
    alpha = slope  # irrational
    for i in range(n):
        val = (i * alpha) % 1.0
        h[i] = 1.0 if val < 0.5 else 0.0
    return repair_feasible(h)

def golden_ruler_seed(n, rng=None):
    """Generate using golden ratio ruler."""
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    h = np.zeros(n)
    for i in range(n):
        val = (i * (1.0 / phi)) % 1.0
        h[i] = 1.0 if val < 0.5 else 0.0
    return repair_feasible(h)

def morphic_seed(n, rng=None):
    """Generate a Thue-Morse-like sequence."""
    h = np.zeros(n)
    # Thue-Morse: t(i) = popcount(i) mod 2
    for i in range(n):
        bc = bin(i).count('1')
        h[i] = 1.0 if bc % 2 == 0 else 0.0
    return repair_feasible(h)

def active_descent(h, max_iters=300, active_tol=0.003):
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    momentum_vec = np.zeros(n)
    momentum = 0.3
    for it in range(max_iters):
        profile = compute_profile(h)
        c = float(profile.max())
        thresh = c - active_tol
        active = np.where(profile >= thresh)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(profile))])
        if len(active) > 25:
            top = np.argsort(profile[active])[-25:]
            active = active[top]
        g = np.zeros(n)
        for m in active:
            g += grad_row(h, m, n)
        g -= g.mean()
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            break
        momentum_vec = momentum * momentum_vec + g
        mv_norm = np.linalg.norm(momentum_vec)
        if mv_norm < 1e-15:
            break
        d = -momentum_vec / mv_norm
        pos_mask = d > 1e-12
        neg_mask = d < -1e-12
        alpha_max = 1.0
        if pos_mask.any():
            alpha_max = min(alpha_max, float(((1.0 - h[pos_mask]) / d[pos_mask]).min()))
        if neg_mask.any():
            alpha_max = min(alpha_max, float((h[neg_mask] / (-d[neg_mask])).min()))
        alpha_max = max(alpha_max, 1e-10)
        alpha_max = min(alpha_max, 0.5)
        best_alpha = 0.0
        best_val = c
        best_h = h
        a_ls, b_ls = 0.0, alpha_max
        for _ in range(20):
            c1 = b_ls - gr * (b_ls - a_ls)
            c2 = a_ls + gr * (b_ls - a_ls)
            h1 = repair_feasible(h + c1 * d)
            h2 = repair_feasible(h + c2 * d)
            f1 = exact_score(h1)
            f2 = exact_score(h2)
            if f1 < best_val:
                best_val = f1
                best_alpha = c1
                best_h = h1.copy()
            if f2 < best_val:
                best_val = f2
                best_alpha = c2
                best_h = h2.copy()
            if f1 < f2:
                b_ls = c2
            else:
                a_ls = c1
            if b_ls - a_ls < 1e-13:
                break
        if best_val >= c - 1e-15:
            break
        h = best_h
        if best_val < best_c:
            best_c = best_val
    return h, best_c

def coordinate_polish(h, max_iters=80):
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    for it in range(max_iters):
        profile = compute_profile(h)
        c = float(profile.max())
        thresh = c - 0.002
        active = np.where(profile >= thresh)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(profile))])
        improved = False
        idx_desc = np.argsort(-h)
        idx_asc = np.argsort(h)
        for attempt in range(40):
            i = int(idx_desc[attempt % n])
            j = int(idx_asc[attempt % n])
            if i == j:
                continue
            for delta_frac in [0.01, 0.03, 0.05]:
                delta = delta_frac * (h[i] - h[j])
                if abs(delta) < 1e-7:
                    continue
                h_try = h.copy()
                h_try[i] += delta
                h_try[j] -= delta
                if h_try.min() < -1e-10 or h_try.max() > 1 + 1e-10:
                    continue
                h_try = repair_feasible(h_try)
                s = exact_score(h_try)
                if s < best_c - 1e-14:
                    h = h_try
                    best_c = s
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()
    rng = np.random.RandomState(seed)
    
    def time_left():
        return budget_s - (time.perf_counter() - t_start)
    
    # Get parent
    try:
        parent = np.array(initial_h_values, dtype=np.float64)
    except Exception:
        parent = None
    if parent is None or len(parent) < 4:
        parent = golden_ruler_seed(70)
    parent = np.clip(parent, 0.0, 1.0)
    parent = repair_feasible(parent)
    n_parent = len(parent)
    
    best_h = parent.copy()
    best_c = exact_score(parent)
    
    # Stage 1: LP descent on parent
    if time_left() > 100:
        h_lp, c_lp = lp_trust_descent(best_h, delta=0.05, max_iters=150)
        if c_lp < best_c:
            best_h, best_c = h_lp, c_lp
        h_ad, c_ad = active_descent(best_h, max_iters=200, active_tol=0.003)
        if c_ad < best_c:
            best_h, best_c = h_ad, c_ad
    
    # Stage 2: Dimension exploration
    n_best = len(best_h)
    dims_to_try = []
    for dn in range(-12, 24, 2):
        d = n_best + dn
        if d >= 30 and d <= 130:
            dims_to_try.append(d)
    dims_to_try.sort(key=lambda x: abs(x - n_best))
    
    for new_n in dims_to_try[:10]:
        if time_left() < 40:
            break
        h_interp = interp_to_n(best_h, new_n)
        h_lp, c_lp = lp_trust_descent(h_interp, delta=0.04, max_iters=100)
        if c_lp < best_c:
            best_h, best_c = h_lp, c_lp
            n_best = new_n
        h_ad, c_ad = active_descent(h_interp, max_iters=100, active_tol=0.003)
        if c_ad < best_c:
            best_h, best_c = h_ad, c_ad
            n_best = new_n
    
    # Stage 3: LP refinement on best
    for _ in range(3):
        if time_left() < 50:
            break
        h_lp, c_lp = lp_trust_descent(best_h, delta=0.03, max_iters=150)
        if c_lp < best_c:
            best_h, best_c = h_lp, c_lp
        h_ad, c_ad = active_descent(best_h, max_iters=100, active_tol=0.001)
        if c_ad < best_c:
            best_h, best_c = h_ad, c_ad
    
    # Stage 4: Try different initializations at best dimension
    if time_left() > 120:
        n_exp = len(best_h)
        seeds = [
            golden_ruler_seed(n_exp),
            morphic_seed(n_exp),
        ]
        # Add some random perturbations of the best
        for _ in range(3):
            pert = best_h + rng.normal(0, 0.05, n_exp)
            seeds.append(repair_feasible(pert))
        
        for seed_h in seeds:
            if time_left() < 30:
                break
            h_lp, c_lp = lp_trust_descent(seed_h, delta=0.05, max_iters=80)
            if c_lp < best_c:
                best_h, best_c = h_lp, c_lp
                n_exp = len(best_h)
            h_ad, c_ad = active_descent(seed_h, max_iters=80, active_tol=0.004)
            if c_ad < best_c:
                best_h, best_c = h_ad, c_ad
    
    # Stage 5: Crossover-based exploration
    if time_left()>100:
        n_cur = len(best_h)
        # Create diverse candidates
        pool = [best_h.copy()]
        for _ in range(4):
            if time_left() < 30:
                break
            # Random feasible vector of same length
            idx = np.arange(n_cur)
            vals = (idx * 0.618033988749895 + rng.uniform(0, 1)) % 1.0
            h_rand = (vals > 0.5).astype(float)
            h_rand = repair_feasible(h_rand)
            # Mix with best
            t = rng.uniform(0.2, 0.7)
            h_mix = (1 - t) * best_h + t * h_rand
            h_mix = repair_feasible(h_mix)
            pool.append(h_mix)
        
        for pool_h in pool:
            if time_left() < 25:
                break
            if pool_h is pool[0]:
                continue  # already optimized
            h_lp, c_lp = lp_trust_descent(pool_h, delta=0.04, max_iters=60)
            if c_lp < best_c:
                best_h, best_c = h_lp, c_lp
    
    # Stage 6: Fine-tune
    if time_left() > 30:
        h_lp, c_lp = lp_trust_descent(best_h, delta=0.02, max_iters=200)
        if c_lp < best_c:
            best_h, best_c = h_lp, c_lp
        h_ad, c_ad = active_descent(best_h, max_iters=200, active_tol=0.001)
        if c_ad < best_c:
            best_h, best_c = h_ad, c_ad
    
    # Stage 7: Coordinate polish
    if time_left() > 20:
        h_cp, c_cp = coordinate_polish(best_h, max_iters=60)
        if c_cp < best_c:
            best_h, best_c = h_cp, c_cp
    
    # Final
    if time_left() > 15:
        h_lp, c_lp = lp_trust_descent(best_h, delta=0.01, max_iters=100)
        if c_lp < best_c:
            best_h, best_c = h_lp, c_lp
        h_ad, c_ad = active_descent(best_h, max_iters=100, active_tol=0.0005)
        if c_ad < best_c:
