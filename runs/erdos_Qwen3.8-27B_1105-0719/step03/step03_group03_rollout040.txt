def exact_score(h):

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx])
            h[idx] += add
            remaining -= add
            if remaining <= 1e-10:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-10:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-10:
        idx = int(np.argmax(1.0 - h)) if diff2 > 0 else int(np.argmax(h))
        h[idx] = np.clip(h[idx] + diff2, 0.0, 1.0)
    return h

def grad_dir(h, profile, beam_frac=0.04, beam_min=0.003, max_active=10):
    n = len(h)
    dx = 2.0 / n
    c_max = np.max(profile)
    spread = c_max - np.min(profile)
    beam = max(beam_min, beam_frac * spread)
    active = np.where(profile >= c_max - beam)[0]
    if len(active) == 0:
        active = np.array([np.argmax(profile)])
    if len(active) > max_active:
        active = active[np.argsort(profile[active])[-max_active:]]
    w = profile[active] - (c_max - beam)
    w = w / (w.sum() + 1e-15)
    center = n - 1
    grad = np.zeros(n)
    for idx, k in enumerate(active):
        offset = k - center
        j_lo1 = max(0, -offset)
        j_hi1 = min(n - 1, n - 1 - offset)
        if j_lo1 <= j_hi1:
            grad[j_lo1:j_hi1 + 1] += w[idx] * dx * (1.0 - h[j_lo1:j_hi1 + 1 + offset])
        j_lo2 = max(0, offset)
        j_hi2 = min(n - 1, n - 1 + offset)
        if j_lo2 <= j_hi2:
            grad[j_lo2:j_hi2 + 1] -= w[idx] * dx * h[j_lo2:j_hi2 + 1 - offset]
    grad -= grad.mean()
    gn = np.linalg.norm(grad)
    if gn < 1e-15:
        return None
    return -grad / gn

        return x[-1]

        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g

        return x[:-1].sum() - n / 2.0

    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g

    def ineq_fun(x):

        return x[-1] - np.correlate(hv, 1 - hv, mode='full') * dx

        G = np.zeros((M, n))
        center = n - 1

            G[center + d, idx1 + d] += dx * hv[idx1]
            G[center - d, idx1] += dx * omh[idx1 + d]
            G[center - d, idx1 + d] += -dx * hv[idx1]
        J = np.zeros((M, n + 1))
        J[:, :-1] = -G
        J[:, -1] = 1.0
        return J

            if hs is not None and cs < best_c:
                best_h, best_c = hs, cs

    # Phase 7: Final micro-improvements
    if remaining() > 15:
        t7 = remaining() - 5
        if t7 > 10:
            h, c = improve(best_h, t_start, budget_s, max_time=t7, seed=seed + 5000)
            if c < best_c:
                best_h, best_c = h, c

    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)

    try:
        er = evaluate_erdos_solution(best_h)
        if isinstance(er, (float, int)):
            final_c = min(final_c, float(er))
        elif isinstance(er, tuple) and len(er) >= 2:
            final_c = min(final_c, float(er[0]))
    except Exception:
        pass

    return best_h, final_c, len(best_h)
