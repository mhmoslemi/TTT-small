import numpy as np
from scipy.optimize import linprog
import time

def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = min(x) - 2.0, max(x) + 2.0

            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)

    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:

def c5_of_h(h):
    return np.correlate(h, 1.0 - h, mode='full').max() * (2.0 / len(h))

def corr_of_h(h):
    return np.correlate(h, 1.0 - h, mode='full')

def grad_lag(h, k, n):
    off = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -off)
    i_hi = min(n - 1, n - 1 - off)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx + off] += (1.0 - h[idx])
    grad[idx] += (-h[idx + off])
    return grad

def solve_soft_slack_lp(h, dx, top_idx, corrs, alpha, rho_q, rho_s):
    n = len(h)
    K = len(top_idx)
    m_vars = n + K + 1
    c_obj = np.zeros(m_vars)
    c_obj[n + K] = rho_q
    for j in range(K):
        c_obj[n + j] = rho_s
    A_ub = np.zeros((K, m_vars))
    b_ub = np.zeros(K)
    for j, k in enumerate(top_idx):
        g = grad_lag(h, k, n)
        A_ub[j, :n] = dx * g
        A_ub[j, n + j] = 1.0
        A_ub[j, n + K] = -1.0
        b_ub[j] = -corrs[k]
    A_eq = np.zeros((1, m_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(-h[i] * alpha, (1.0 - h[i]) * alpha) for i in range(n)]
    bounds += [(0.0, None)] * K
    bounds.append((0.0, None))
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.status != 0:
            return None
        return res.x[:n]
    except Exception:
        return None

def solve_weighted_sum_lp(h, dx, top_idx, corrs, alpha, weights):
    n = len(h)
    c_obj = np.zeros(n)
    for j, k in enumerate(top_idx):
        g = grad_lag(h, k, n)
        c_obj += weights[j] * dx * g
    A_eq = np.ones((1, n))
    b_eq = np.zeros(1)
    bounds = [(-h[i] * alpha, (1.0 - h[i]) * alpha) for i in range(n)]
    try:
        res = linprog(c_obj, A_eq=A_eq, b_eq=b_eq, bounds=bounds,

def lp_polish(h, dx, target_sum, n_iter, alpha, budget_s, t0, min_time=3.0):

        top_idx = np.argsort(corrs)[-n_top:]

                h_p3, c5_p3 = pair_optimize(best_h, 2.0/len(best_h), len(best_h)/2.0,
                                             150, rng, budget_s, t0, min_time=5.0)
                if c5_p3 < best_c5:
                    best_c5 = c5_p3
                    best_h = h_p3

    # Phase 6: Random restarts on best grid
    n_cur = len(best_h)
    dx_cur = 2.0 / n_cur
    ts_cur = n_cur / 2.0

    n_restarts = 5
    for r in range(n_restarts):
        if remaining() < 40:
            break
        scale = 0.08 / (r + 1)
        pert = rng.standard_normal(n_cur)
        pert -= pert.mean()
        h_p = best_h + scale * pert
        h_p = project_feasible(h_p, ts_cur)
        h_opt, c5_opt = lp_polish(h_p, dx_cur, ts_cur, 80, 0.15, budget_s, t0, min_time=5.0)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 7: Alternating LP and pair on best
    if remaining() > 30:
        for _ in range(3):
            if remaining() < 20:
                break
            h_a, c5_a = lp_polish(best_h, 2.0/len(best_h), len(best_h)/2.0,
                                   100, 0.1, budget_s, t0, min_time=5.0)
            if c5_a < best_c5:
                best_c5 = c5_a
                best_h = h_a
            if remaining() < 20:
                break
            h_b, c5_b = pair_optimize(best_h, 2.0/len(best_h), len(best_h)/2.0,
                                       150, rng, budget_s, t0, min_time=5.0)
            if c5_b < best_c5:
                best_c5 = c5_b
                best_h = h_b

    # Phase 8: Very fine polish
    n_fin = len(best_h)
    dx_fin = 2.0 / n_fin
    ts_fin = n_fin / 2.0
    if remaining() > 15:
        h_f, c5_f = lp_polish(best_h, dx_fin, ts_fin, 200, 0.05, budget_s, t0, min_time=3.0)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f

    # Phase 9: Final pair polish
    if remaining() > 10:
        h_fp, c5_fp = pair_optimize(best_h, dx_fin, ts_fin, 200, rng, budget_s, t0, min_time=3.0)
        if c5_fp < best_c5:
            best_c5 = c5_fp
            best_h = h_fp

    # Final validation
    n_final = len(best_h)
    dx_final = 2.0 / n_final
    ts_final = n_final / 2.0
    best_h = project_feasible(best_h, ts_final)
    best_c5 = c5_of_h(best_h)

    return best_h, best_c5, n_final
