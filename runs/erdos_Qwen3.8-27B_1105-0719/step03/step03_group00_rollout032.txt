def project_feasible(x, target_sum):
    n = len(x)
    if not np.all(np.isfinite(x)):
        x = np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0)
    lo = x.min() - 1.0
    hi = x.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = mask.sum()
        if cnt > 0:
            h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full').max() * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def power_gradient(h, q, corr):
    n = len(h)
    maxc = max(corr.max(), 1e-30)
    a = corr / maxc
    w = np.power(a, q - 1.0)
    g = np.zeros(n)
    w0 = w[n - 1]
    if w0 > 1e-15:
        g += w0 * (1.0 - 2.0 * h)
    for d in range(1, n):
        K = n - d
        wp = w[n - 1 + d]
        wm = w[n - 1 - d]
        if wp > 1e-15:
            g[:K] += wp * (1.0 - h[d:])
            g[d:] -= wp * h[:K]
        if wm > 1e-15:
            g[d:] += wm * (1.0 - h[:K])
            g[:K] -= wm * h[d:]
    g -= g.mean()
    return g

import numpy as np

def project_feasible(x, target_sum):
    n = len(x)
    if not np.all(np.isfinite(x)):
        x = np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0)
    lo = x.min() - 1.0
    hi = x.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = mask.sum()
        if cnt > 0:
            h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full').max() * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def power_gradient(h, q, corr):
    n = len(h)
    maxc = max(corr.max(), 1e-30)
    a = corr / maxc
    w = np.power(a, q - 1.0)
    g = np.zeros(n)
    w0 = w[n - 1]
    if w0 > 1e-15:
        g += w0 * (1.0 - 2.0 * h)
    for d in range(1, n):
        K = n - d
        wp = w[n - 1 + d]
        wm = w[n - 1 - d]
        if wp > 1e-15:
            g[:K] += wp * (1.0 - h[d:])
            g[d:] -= wp * h[:K]
        if wm > 1e-15:
            g[d:] += wm * (1.0 - h[:K])
            g[:K] -= wm * h[d:]
    g -= g.mean()
    return g

def smooth_phase(h, target_sum, q, t0, budget_s, max_iter=300, step_init=0.04):
    n = len(h)
    cur = h.copy()
    cur_c5 = compute_c5(cur)
    best = cur.copy()
    best_c5 = cur_c5
    step = step_init
    stall = 0
    for it in range(max_iter):
        if budget_s - (time.time() - t0) < 5.0:
            break
        corr = compute_corr(cur)
        g = power_gradient(cur, q, corr)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        improved = False
        for scale in [1.0, 0.7, 0.4, 0.25, 0.15, 0.08, 0.04, 0.02, 0.01]:
            alpha = step * scale / gn

            if c5v < cur_c5 - 1e-14:
                cur = cand
                cur_c5 = c5v
                improved = True
                if c5v < best_c5:
                    best_c5 = c5v
                    best = cur.copy()
                break
        if improved:
            step = min(step * 1.1, 0.4)
            stall = 0
        else:
            step *= 0.6

                break
