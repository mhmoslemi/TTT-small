def lse_slsqp(h_start, beta=50.0, maxiter=300):
    n = len(h_start)
    dx = 2.0 / n
    M = 2 * n - 1
    c0 = exact_score(h_start)
    
    def obj(x):
        h = x
        profile = np.correlate(h, 1 - h, mode="full") * dx
        max_p = profile.max()
        return float(max_p + np.log(np.sum(np.exp(beta * (profile - max_p)))) / beta)
    
    def obj_grad(x):
        h = x
        profile = np.correlate(h, 1 - h, mode="full") * dx
        max_p = profile.max()
        shifted = beta * (profile - max_p)
        exp_vals = np.exp(shifted)
        weights = exp_vals / exp_vals.sum()
        g = np.zeros(n)
        for m in range(M):
            if weights[m] > 1e-10:
                g += weights[m] * grad_row(h, m, n)
        return g
    
    def eq_fun(x):
        return x.sum() - n / 2.0
    
    def eq_jac(x):
        return np.ones(n)
    
    x0 = h_start.copy()
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    
    res = minimize(obj, x0, method='SLSQP', jac=obj_grad,
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    
    if res.x is not None and np.isfinite(res.x).all():

        c_new = exact_score(h_new)
        if c_new <= c0 + 1e-8:

    return None, c0

        if k < n - k:  # Only fill positive frequencies (Hermitian symmetry)
            H[k] = np.exp(1j * phases[k])
    h = np.real(np.fft.ifft(H))
    # Scale and clip to [0,1]
    # This won't automatically satisfy 0 ≤ h ≤ 1, so we need to project
    h = repair_feasible(h)
    return h

import numpy as np
from scipy.optimize import minimize
import time

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(-(1.0 - h))
        remaining = diff
        for i in range(n):
            idx = order[i]
            cap = 1.0 - h[idx]
            if cap <= 1e-15:
                continue
            add = min(remaining, cap)
            h[idx] += add
            remaining -= add
            if remaining <= 1e-14:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            if h[idx] <= 1e-15:
                continue
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-14:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmin(h))
        h[idx] = float(np.clip(h[idx] + diff2, 0.0, 1.0))
    return h

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def grad_row(h, m, n):
    dx = 2.0 / n
    d = m - (n - 1)
    if d == 0:
        return dx * (1.0 - 2.0 * h)
    elif d > 0:
        g = np.zeros(n)
        g[d:] += dx * (1.0 - h[:-d])
        g[:-d] -= dx * h[d:]
        return g
    else:
        p = -d
        g = np.zeros(n)
        g[:-p] += dx * (1.0 - h[p:])
        g[p:] -= dx * h[:-p]
        return g

def interp_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    edges_old = np.linspace(0.0, 2.0, n + 1)
    edges_new = np.linspace(0.0, 2.0, new_n + 1)
    h_new = np.zeros(new_n)
    for k in range(new_n):
        a, b = edges_new[k], edges_new[k + 1]
        total = 0.0
        for j in range(n):
            lo = max(a, edges_old[j])
            hi = min(b, edges_old[j + 1])
            if hi > lo:
                total += h[j] * (hi - lo)
        h_new[k] = total / (b - a)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

def active_descent(h, max_iters=400, active_tol=0.003):
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    momentum_vec = np.zeros(n)
    momentum = 0.3
    for it in range(max_iters):
        profile = compute_profile(h)
        c = float(profile.max())
        thresh = c - active_tol
        active = np.where(profile >= thresh)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(profile))])
        if len(active) > 25:
            top = np.argsort(profile[active])[-25:]
            active = active[top]
        g = np.zeros(n)
        for m in active:
            g += grad_row(h, m, n)
        g -= g.mean()
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            break
        momentum_vec = momentum * momentum_vec + g
        mv_norm = np.linalg.norm(momentum_vec)
        if mv_norm < 1e-15:
            break
        d = -momentum_vec / mv_norm
        pos_mask = d > 1e-12
        neg_mask = d < -1e-12
        alpha_max = 1.0
        if pos_mask.any():
            alpha_max = min(alpha_max, float(((1.0 - h[pos_mask]) / d[pos_mask]).min()))
        if neg_mask.any():
            alpha_max = min(alpha_max, float((h[neg_mask] / (-d[neg_mask])).min()))
        alpha_max = max(alpha_max, 1e-10)
        alpha_max = min(alpha_max, 0.5)
        best_alpha = 0.0
        best_val = c
        best_h = h
        a_ls, b_ls = 0.0, alpha_max
        for _ in range(24):
            c1 = b_ls - gr * (b_ls - a_ls)
            c2 = a_ls + gr * (b_ls - a_ls)
            h1 = repair_feasible(h + c1 * d)
            h2 = repair_feasible(h + c2 * d)
            f1 = exact_score(h1)
            f2 = exact_score(h2)
            if f1 < best_val:
                best_val = f1
                best_alpha = c1
                best_h = h1.copy()
            if f2 < best_val:
                best_val = f2
                best_alpha = c2
                best_h = h2.copy()
            if f1 < f2:
                b_ls = c2
            else:
                a_ls = c1
            if b_ls - a_ls < 1e-13:
                break
        if best_val >= c - 1e-15:
            break
        h = best_h
        if best_val < best_c:
            best_c = best_val
    return h, best_c

def lse_slsqp(h_start, beta=50.0, maxiter=250):
    n = len(h_start)
    dx = 2.0 / n
    M = 2 * n - 1
    c0 = exact_score(h_start)

    def obj(x):
        h = x
        profile = np.correlate(h, 1 - h, mode="full") * dx
        max_p = profile.max()
        return float(max_p + np.log(np.sum(np.exp(beta * (profile - max_p)))) / beta)

    def obj_grad(x):
        h = x
        profile = np.correlate(h, 1 - h, mode="full") * dx
        max_p = profile.max()
        shifted = beta * (profile - max_p)
        exp_vals = np.exp(shifted)
        s = exp_vals.sum()
        if s < 1e-300:
            return np.zeros(n)
        weights = exp_vals / s
        g = np.zeros(n)
        for m in range(M):
            if weights[m] > 1e-10:
                g += weights[m] * grad_row(h, m, n)
        return g

    def eq_fun(x):
        return x.sum() - n / 2.0

    def eq_jac(x):
        return np.ones(n)

    x0 = h_start.copy()
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_grad,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        if res.x is not None and np.isfinite(res.x).all():
            h_new = repair_feasible(res.x)
            c_new = exact_score(h_new)
            if c_new <= c0 + 1e-8:
                return h_new, c_new
    except Exception:
        pass
    return None, c0

def epigraph_slsqp(h_start, maxiter=250):
    n = len(h_start)
    dx = 2.0 / n
    M = 2 * n - 1
    c0 = exact_score(h_start)
    t0 = c0 + 1e-4

    def obj(x):
        return x[-1]

    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g

    def eq_fun(x):
        return x[:-1].sum() - n / 2.0

    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g

    def ineq_fun(x):
        h = x[:-1]
        t = x[-1]
        profile = np.correlate(h, 1 - h, mode="full") * dx
        return t - profile

    def ineq_jac(x):
        h = x[:-1]
        J = np.zeros((M, n + 1))
        for m in range(M):
            J[m, :n] = -grad_row(h, m, n)
        J[:, -1] = 1.0
        return J

    x0 = np.concatenate([h_start, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 2.0)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.x is not None and np.isfinite(res.x).all():
            h_new = repair_feasible(res.x[:-1])
            c_new = exact_score(h_new)
            if c_new <= c0 + 1e-8:
                return h_new, c_new
    except Exception:
        pass
    return None, c0

def lse_descent(h0, max_iters=300, beta_start=10.0, beta_end=200.0):
    n = len(h0)
    h = h0.copy()
    v = np.zeros(n)
    best_h = h0.copy()
    best_c = exact_score(h0)
    for it in range(max_iters):
        frac = (it + 1) / max_iters
        beta = beta_start * (beta_end / beta_start) ** frac
        profile = compute_profile(h)
        max_p = profile.max()
        shifted = beta * (profile - max_p)
        exp_vals = np.exp(shifted)
        s = exp_vals.sum()
        if s < 1e-300:
            weights = np.zeros(len(profile))
            weights[int(np.argmax(profile))] = 1.0
        else:
            weights = exp_vals / s
        g = np.zeros(n)
        active_idx = np.where(weights > 1e-8)[0]
        for m in active_idx:
            g += weights[m] * grad_row(h, m, n)
        g -= g.mean()
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-14:
            break
        lr = 0.08 / (1.0 + 0.005 * it)
        v = 0.5 * v - lr * g
        h_new = repair_feasible(h + v)
        c = exact_score(h_new)
        if c < best_c:
            best_c = c
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c

def quasi_periodic_seed(n, rng, alpha_idx=0):
    alphas = [0.618033988749895, 0.4142135623730951, 0.3819660112501051,
              1.0 / 3.0, 0.5, 1.0 / 4.0, 0.7071067811865476]
    alpha = alphas[alpha_idx % len(alphas)]
    idx = np.arange(n)
    vals = (idx * alpha + rng.uniform(0, 0.5)) % 1.0
    half = n // 2
    top = np.argsort(vals)[-half:]
    h = np.zeros(n)
    h[top] = 1.0
    return repair_feasible(h)

def smooth_binary_seed(n, rng, alpha_idx=0):
    h_bin = quasi_periodic_seed(n, rng, alpha_idx)
    # Blend with uniform to create interior values
    t = rng.uniform(0.2, 0.5)
    h = t * h_bin + (1 - t) * 0.5
    return repair_feasible(h)
