        candidates = np.array(candidates)
        
        # Evaluate max at each candidate
        P0_cand = -dx*candidates**2 + dx*candidates + C0
        # For linear terms: P(k) = a[k]*x + b[k]
        # Shape: (n_cand, M-1)
        P_lin_cand = np.outer(candidates, a) + b  # includes k0 but a[k0]=0 there
        # Actually a[k0] = 0 (we didn't set it), so P_lin_cand[:, k0] = b[k0] = P[k0] - 0 = P[k0]
        # But the true P(k0) at candidate x is P0_cand, not b[k0]
        # So we need to exclude k0 from the linear evaluation
        P_lin_excl = np.outer(candidates, np.delete(a, k0)) + np.delete(b, k0)
        max_at_cand = np.maximum(P0_cand, P_lin_excl.max(axis=1))
        
        best_idx = np.argmin(max_at_cand)
        h[i] = candidates[best_idx]
    
    return h
```

Wait, this has O(n) iterations in the inner loop (for computing the gradient), making the per-coordinate cost O(n²) due to the M=2n-1 iterations. For n=70, that's 70*140 = 9800 per coordinate, 70*9800 = 686,000 per sweep. With 50 sweeps, that's 34 million ops. Should be fine.

But I can optimize the gradient computation. Instead of looping over all M lags, I can compute it directly:

For coordinate i, the gradient at lag d is:
- d=0: dx*(1-2*h[i]) [but we handle this as quadratic, so we skip]
- d>0: dx*[(1-h[i-d])*1_{i≥d} - h[i+d]*1_{i+d<n}]
- d<0: let dp=-d: dx*[(1-h[i+dp])*1_{i+dp<n} - h[i-dp]*1_{i≥dp}]

I can compute this for all d at once:
- For d>0: the first term is nonzero for d≤i, and equals dx*(1-h[i-d]) for d=1,...,min(i, n-1)
- The second term is nonzero for i+d<n, i.e., d<n-i, and equals -dx*h[i+d] for d=1,...,min(n-1-i, n-1)

So I can construct the gradient array in O(n) time per coordinate.

Let me vectorize this:

```python
def grad_col(i, h, dx, n):
    M = 2*n - 1
    g = np.zeros(M)
    k0 = n - 1
    
    # d > 0
    max_d_pos = min(i, n-1)  # first term valid for d=1..max_d_pos
    if max_d_pos > 0:
        d_range = np.arange(1, max_d_pos+1)
        g[n-1+d_range] += dx * (1 - h[i-d_range])
    
    max_d_pos2 = min(n-1-i, n-1)  # second term valid for d=1..max_d_pos2
    if max_d_pos2 > 0:
        d_range2 = np.arange(1, max_d_pos2+1)
        g[n-1+d_range2] -= dx * h[i+d_range2]
    
    # d < 0: dp = -d
    max_dp = min(n-1-i, n-1)  # first term: i+dp < n, dp=1..max_dp
    if max_dp > 0:
        dp_range = np.arange(1, max_dp+1)
        g[n-1-dp_range] += dx * (1 - h[i+dp_range])
    
    max_dp2 = min(i, n-1)  # second term: i >= dp, dp=1..max_dp2
    if max_dp2 > 0:
        dp_range2 = np.arange(1, max_dp2+1)
        g[n-1-dp_range2] -= dx * h[i-dp_range2]
    
    return g
```

This is O(n) per coordinate with numpy vectorization. Good.

Now let me put it all together. I'll write the complete code.

One more consideration: the coordinate minimax might be slow if I recompute the full profile at each coordinate. Instead, I can update the profile incrementally. When I change h[i] from x to x', the profile changes predictably:
- P(0) changes by: dx*(x'-x) - dx*(x'²-x²) = dx*(x'-x)(1-x'-x)
- P(k) for k≠0 changes by: a[k]*(x'-x) (linear change)

So I can update the profile in O(M) = O(n) per coordinate step, rather than O(n) for the full correlate.

Actually, np.correlate is already O(n²) but very fast in C. For n=70, it's negligible. Let me not over-optimize and just recompute.

Let me now write the final code. I'll aim for a balance between the coordinate minimax (new), Nesterov subgradient (new), LSE (improved from before), and some of the other techniques from before.

Let me also add a "plateau detection" mechanism: if the objective hasn't improved in many iterations, try a different strategy (e.g., random perturbation, different resolution).

Here's my final code structure:

```python
import numpy as np
from scipy.optimize import minimize
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1-h, mode="full") * dx))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1-h, mode="full") * dx

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx])
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    return h

def grad_col(i, h, dx, n):
    """Compute dP(k)/dh[i] for all k=0,...,2n-2."""
    M = 2 * n - 1
    g = np.zeros(M)
    k0 = n - 1
    # d > 0: first term (1-h[i-d]) for d=1..min(i, n-1)
    m1 = min(i, n-1)
    if m1 > 0:
        d = np.arange(1, m1+1)
        g[n-1+d] += dx * (1.0 - h[i-d])
    # d > 0: second term -h[i+d] for d=1..min(n-1-i, n-1)
    m2 = min(n-1-i, n-1)
    if m2 > 0:
        d = np.arange(1, m2+1)
        g[n-1+d] -= dx * h[i+d]
    # d < 0 (dp=-d): first term (1-h[i+dp]) for dp=1..min(n-1-i, n-1)
    m3 = min(n-1-i, n-1)
    if m3 > 0:
        dp = np.arange(1, m3+1)
        g[n-1-dp] += dx * (1.0 - h[i+dp])
    # d < 0: second term -h[i-dp] for dp=1..min(i, n-1)
    m4 = min(i, n-1)
    if m4 > 0:
        dp = np.arange(1, m4+1)
        g[n-1-dp] -= dx * h[i-dp]
    return g

def coord_minimax_one_sweep(h, dx, n):
    """One sweep of exact 1D minimax coordinate descent."""
    M = 2 * n - 1
    k0 = n - 1  # profile index for lag 0
    
    for i in range(n):
        P = compute_profile(h)
        C0 = P[k0] + dx*h[i]**2 - dx*h[i]
        a = grad_col(i, h, dx, n)
        a[k0] = 0.0  # exclude lag 0 (handled as quadratic)
        b = P.copy()
        for k in range(M):
            if k != k0:
                b[k] = P[k] - a[k]*h[i]
        
        # Candidates
        cands = [0.0, 1.0, 0.5]
        # Intersections with parabola
        for k in range(M):
            if k == k0:
                continue
            A = -dx
            B = dx - a[k]
            C = C0 - b[k]
            disc = B*B - 4.0*A*C
            if disc >= 0:
                sd = np.sqrt(disc)
                for x_c in [(-B+sd)/(2*A), (-B-sd)/(2*A)]:
                    if -1e-10 <= x_c <= 1.0+1e-10:
                        cands.append(np.clip(x_c, 0.0, 1.0))
        
        cands = np.array(cands)
        # Evaluate
        P0_c = -dx*cands**2 + dx*cands + C0
        # Linear terms (exclude k0)
        mask = np.ones(M, dtype=bool)
        mask[k0] = False
        P_lin = cands[:,None] * a[None,:] + b[None,:]
        P_lin = P_lin[:, mask]
        max_c = np.maximum(P0_c, P_lin.max(axis=1))
        
        best = np.argmin(max_c)
        h[i] = cands[best]
    
    return h
