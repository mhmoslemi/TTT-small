def project_feasible(x, target_sum):
    """Project x onto [0,1]^n ∩ {sum = target_sum}"""
    n = len(x)
    x = np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0)
    x = np.clip(x, -10.0, 10.0)
    
    lo, hi = -10.0, 10.0
    # Bracket
    if np.clip(x - lo, 0, 1).sum() < target_sum:
        lo = x.min() - 1.0
    if np.clip(x - hi, 0, 1).sum() > target_sum:
        hi = x.max() + 1.0
    
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    
    h = np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        cnt = mask.sum()
        if cnt > 0:
            h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def grad_correlation(h, d, N):
    """Compute gradient of correlation at offset d with respect to h."""
    # d is the offset: c = sum_n h[n+d]*(1-h[n]) for valid n
    grad = np.zeros(N)

            if ad <= j <= N - 1:

def grad_all_lags(h):
    """Compute gradients of all lag correlations. Returns array of shape (2N-1, N)."""
    N = len(h)
    grads = np.zeros((2*N - 1, N))
    for m in range(2*N - 1):
        d = m - (N - 1)
        ad = abs(d)
        if d >= 0:
            if N - 1 - d >= 0:
                # First term: indices d to N-1 get (1-h[j-d])
                grads[m, d:] = (1.0 - h[:N-d]) + grads[m, d:]
                # Second term: indices 0 to N-1-d get (-h[j+d])
                grads[m, :N-d] += (-h[d:])
        else:
            # d < 0
            if N - 1 - ad >= 0:
                # First term: indices 0 to N-1-ad get (1-h[j+ad])
                grads[m, :N-ad] = (1.0 - h[ad:]) + grads[m, :N-ad]
                # Second term: indices ad to N-1 get (-h[j-ad])
                grads[m, ad:] += (-h[:N-ad])
