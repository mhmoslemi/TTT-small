        active = active[top_idx]
    return active, T

def lp_trust_region(h, active, radius):
    n = len(h)
    dx = 2.0 / n
    G = build_gradient(h, rows=active)
    profile = compute_profile(h)
    p_a = profile[active]
    M = len(active)
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_ub = np.zeros((M, n + 1))
    A_ub[:, :n] = G
    A_ub[:, -1] = -1.0
    b_ub = -p_a
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    lower = np.maximum(-h, -radius)
    upper = np.minimum(1.0 - h, radius)
    bounds = list(zip(lower, upper))
    bounds.append((0.0, None))
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'maxiter': 3000})
        if res.success and res.x is not None and np.isfinite(res.x).all():
            h_new = repair(h + res.x[:n])
            return h_new, float(res.x[-1])
    except Exception:
        pass
    return None, None

def kkt_descent_direction(h, active):
    n = len(h)
    dx = 2.0 / n
    G = build_gradient(h, rows=active)
    profile = compute_profile(h)
    p_a = profile[active]
    T = p_a.max()
    spread = T - p_a.min()
    if spread < 1e-12:
        weights = np.ones(len(active)) / len(active)
    else:
        w = 0.1 + 4.0 * ((p_a - p_a.min()) / (spread + 1e-15)) ** 2
        weights = w / w.sum()
    grad = G.T @ weights
    grad -= grad.mean()
    norm = np.linalg.norm(grad)
    if norm < 1e-15:
        return None
    return grad / norm

def pair_transfer_score(h, p, q, t):
    h_new = h.copy()
    h_new[p] -= t
    h_new[q] += t
    if h_new[p] < -1e-12 or h_new[q] > 1 + 1e-12:
        return None
    h_new = np.clip(h_new, 0, 1)
    return exact_score(h_new)

def pair_beam_search(h, active, max_pairs=60, t_samples=6):
    n = len(h)
    profile = compute_profile(h)
    T = profile.max()
    G = build_gradient(h, rows=active)
    best_c = T
    best_h = h.copy()
    grad = G.T @ np.ones(len(active)) / len(active)
    grad -= grad.mean()
    grad_abs = np.abs(grad)
    indices = np.argsort(grad_abs)[::-1]
    pool = indices[:min(20, n)]
    pairs = []
    seen = set()
    for i in range(len(pool)):
        for j in range(len(pool)):
            p, q = pool[i], pool[j]
            if p == q:
                continue
            key = (min(p, q), max(p, q))
            if key in seen:
                continue
            seen.add(key)
            mu = min(h[p], 1.0 - h[q])
            if mu < 1e-8:
                continue
            v = G[:, q] - G[:, p]
            t_pred = 0.0
            if v.max() > 1e-15:
                pass
            else:
                t_pred = min(mu, max(0.001, -T * 0.1 / (abs(v.max()) + 1e-15)))
            pred_drop = -v @ np.ones(len(active)) / len(active)
            pairs.append((pred_drop * min(mu, 0.1), p, q, mu, t_pred))
    pairs.sort(key=lambda x: -x[0])
    tested = 0
    for pred_drop, p, q, mu, t_pred in pairs[:max_pairs]:
        if mu < 1e-8:
            continue
        t_candidates = [0.1 * mu, 0.25 * mu, 0.5 * mu, 0.75 * mu, 0.9 * mu]
        if 0 < t_pred < mu:
            t_candidates.append(t_pred)
        t_candidates = [max(0, min(mu, t)) for t in t_candidates]
        t_candidates = sorted(set([round(t, 10) for t in t_candidates]))
        for t in t_candidates[:t_samples]:
            h_new = h.copy()
            h_new[p] -= t
            h_new[q] += t
            h_new = repair(h_new)
            c = exact_score(h_new)
            if c < best_c - 1e-14:
                best_c = c
                best_h = h_new.copy()
        tested += 1
    return best_h, best_c

def interpolate_h(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    return repair(np.clip(h_new, 0, 1))

def coherent_steps(h, active, deadline, max_iters=200, radius0=0.08):
    n = len(h)
    best_h = h.copy()
    best_c = exact_score(h)
    radius = radius0
    stall = 0
    for it in range(max_iters):
        if time.perf_counter() > deadline:
            break
        profile = compute_profile(h)
        c = profile.max()
        if c < best_c - 1e-13:
            best_c = c
            best_h = h.copy()
            stall = 0
            radius = min(radius * 1.3, 0.2)
        else:
            stall += 1
        if stall > 50:
            break
        active, T = get_active(profile, beam_frac=0.01, min_beam=0.003, max_count=30)
        h_lp, t_lp = lp_trust_region(h, active, radius)
        improved = False
        if h_lp is not None and t_lp > 1e-10:
            for alpha in [1.0, 0.7, 0.5, 0.3, 0.15]:
                h_try = repair(h + alpha * (h_lp - h))
                c_try = exact_score(h_try)
                if c_try < best_c - 1e-13:
                    h = h_try
                    improved = True
                    break
        if not improved:
            grad = kkt_descent_direction(h, active)
            if grad is not None:
                scale = np.max(np.abs(grad))
                for a in [0.02, 0.01, 0.005, 0.002, 0.001]:
                    step = a * grad / (scale + 1e-15)
                    h_try = repair(h - step)
                    c_try = exact_score(h_try)
                    if c_try < best_c - 1e-13:
                        h = h_try
                        improved = True
                        stall = 0
                        break
        if not improved:
            h_pb, c_pb = pair_beam_search(h, active, max_pairs=30, t_samples=5)
            if c_pb < best_c - 1e-13:
                h = h_pb
                improved = True
                stall = 0
        if not improved:
            radius *= 0.75
            if radius < 0.002:
                break
    return best_h, best_c

def fine_polish(h, deadline, max_iters=150):
    best_h = h.copy()
    best_c = exact_score(h)
    n = len(h)
    for it in range(max_iters):
        if time.perf_counter() > deadline:
            break
        profile = compute_profile(h)
        c = profile.max()
        if c < best_c - 1e-13:
            best_c = c
            best_h = h.copy()
        active, T = get_active(profile, beam_frac=0.005, min_beam=0.001, max_count=15)
        improved = False
        h_lp, _ = lp_trust_region(h, active, 0.005)
        if h_lp is not None:
            for alpha in [1.0, 0.5, 0.25, 0.1]:
                h_try = repair(h + alpha * (h_lp - h))
                c_try = exact_score(h_try)
                if c_try < best_c - 1e-13:
                    h = h_try
                    improved = True
                    break
        if not improved:
            grad = kkt_descent_direction(h, active)
            if grad is not None:
                for a in [0.003, 0.001, 0.0005, 0.0002]:
                    scale = np.max(np.abs(grad)) + 1e-15
                    h_try = repair(h - (a / scale) * grad * 0.1)
                    c_try = exact_score(h_try)
                    if c_try < best_c - 1e-13:
                        h = h_try
                        improved = True
                        break
        if not improved:
            h_pb, c_pb = pair_beam_search(h, active, max_pairs=20, t_samples=4)
            if c_pb < best_c - 1e-13:
                h = h_pb
                improved = True
        if not improved:
            break
    return best_h, best_c

def slsqp_polish(h, deadline):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    c0 = exact_score(h)
    def obj(x):
        return x[-1]
    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    def eq(x):
        return x[:-1].sum() - n / 2.0
    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g
    def ineq(x):
        h = x[:-1]
        t = x[-1]
        prof = np.correlate(h, 1 - h, mode="full") * dx
        return t - prof
    G = build_gradient(h, rows=None)
    def ineq_jac(x):
        h = x[:-1]
        Gc = build_gradient(h, rows=None)
        J = np.zeros((M, n + 1))
        J[:, :n] = -Gc
        J[:, n] = 1.0
        return J
    x0 = np.concatenate([h, [c0 + 1e-4]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    cons = [
        {'type': 'eq', 'fun': eq, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq, 'jac': ineq_jac}
    ]
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': 200, 'ftol': 1e-14})
        if res.x is not None and np.isfinite(res.x).all():
            h_sol = repair(res.x[:-1])
            c_sol = exact_score(h_sol)
            return h_sol, c_sol
    except Exception:
        pass
    return h, c0

def make_blend(h_a, h_b, alpha):
    n = len(h_a)
    if len(h_b) != n:
        h_b = interpolate_h(h_b, n)
    h = alpha * h_a + (1 - alpha) * h_b
    return repair(h)

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()
    deadline = t_start + budget_s - 8
    rng = np.random.RandomState(seed)

    parent = None
    try:
        parent = np.array(initial_h_values, dtype=np.float64)
    except Exception:
        pass
    if parent is None or len(parent) < 10 or not np.isfinite(parent).all():
        parent = np.full(70, 0.5)

    parent = np.clip(parent, 0, 1)
    parent = repair(parent)
    best_h = parent.copy()
    best_c = exact_score(parent)

    h_sym = repair((parent + parent[::-1]) / 2.0)
    c_sym = exact_score(h_sym)
    if c_sym < best_c:
        best_h, best_c = h_sym, c_sym

    h_opt, c_opt = coherent_steps(best_h, None, deadline, max_iters=300, radius0=0.1)
    if c_opt < best_c:
        best_h, best_c = h_opt, c_opt

    h_pol, c_pol = fine_polish(best_h, deadline, max_iters=100)
    if c_pol < best_c:
        best_h, best_c = h_pol, c_pol

    n0 = len(best_h)
    dims = set()
    for d in [n0 - 4, n0 - 2, n0 + 2, n0 + 4, n0 + 6, n0 + 10, n0 + 20]:
        if 20 <= d <= 200 and d != n0 and d % 2 == 0:
            dims.add(d)
    dims.add(n0 * 2)
    if len(dims[d]) <= 200 if False else n0 * 2 <= 200:
        dims.add(n0 * 2)
    dims = sorted(dims, key=lambda d: abs(d - n0))

    for new_n in dims[:8]:
        if time.perf_counter() > deadline - 60:
            break
        h_interp = interpolate_h(best_h, new_n)
        c_interp = exact_score(h_interp)
        if c_interp < best_c + 0.02:
            h_opt, c_opt = coherent_steps(h_interp, None, deadline - 30, max_iters=150, radius0=0.08)
            if c_opt < best_c:
                best_h, best_c = h_opt, c_opt
                if time.perf_counter() < deadline - 40:
                    h_pol, c_pol = fine_polish(best_h, deadline - 20, max_iters=60)
                    if c_pol < best_c:
                        best_h, best_c = h_pol, c_pol

    if time.perf_counter() < deadline - 80:
        for n_try in [n0, n0 + 4, n0 - 4]:
            if time.perf_counter() > deadline - 40:
                break
            if n_try < 20 or n_try > 200:
                continue
            for mode in range(3):
                if time.perf_counter() > deadline - 30:
                    break
                if mode == 0:
                    h_init = np.zeros(n_try)
                    freq = rng.uniform(1, 4)
                    ph = rng.uniform(0, 2 * np.pi)
                    for i in range(n_try):
                        h_init[i] = 0.5 + 0.35 * np.sin(2 * np.pi * freq * i / n_try + ph)
                elif mode == 1:
                    nb = rng.randint(3, 6)
                    h_init = np.zeros(n_try)
                    block = n_try // nb
                    for b in range(nb):
                        h_init[b*block:(b+1)*block] = rng.uniform(0.1, 0.9)
                    if nb * block < n_try:
                        h_init[nb*block:] = rng.uniform(0.1, 0.9)
                else:
                    phi = 0.6180339887498949
                    h_init = np.zeros(n_try)
                    for i in range(n_try):
                        h_init[i] = 0.5 + 0.4 * np.cos(2 * np.pi * ((i * phi + rng.uniform(0, 0.1)) % 1.0))
                h_init = repair(np.clip(h_init, 0, 1))
                h_blend = repair(0.4 * best_h[:n_try] + 0.6 * h_init if n_try == n0 else h_init)
                c_blend = exact_score(h_blend)
                if c_blend < best_c + 0.03:
                    h_opt, c_opt = coherent_steps(h_blend, None, deadline - 30, max_iters=100, radius0=0.06)
                    if c_opt < best_c:
                        best_h, best_c = h_opt, c_opt

    if time.perf_counter() < deadline - 50:
        h_pol, c_pol = fine_polish(best_h, deadline - 20, max_iters=100)
        if c_pol < best_c:
            best_h, best_c = h_pol, c_pol

    if time.perf_counter() < deadline - 30:
        h_sl, c_sl = slsqp_polish(best_h, deadline - 10)
        if c_sl < best_c:
            best_h, best_c = h_sl, c_sl

    if time.perf_counter() < deadline - 10:
        h_pol, c_pol = fine_polish(best_h, deadline, max_iters=80)
        if c_pol < best_c:
            best_h, best_c = h_pol, c_pol

    best_h = repair(best_h)
    final_c = exact_score(best_h)
    if final_c > best_c:
        best_c = final_c

    try:
        ev = evaluate_erdos_solution(best_h)
        if isinstance(ev, (float, int)):
            final_c = float(ev)
        elif isinstance(ev, tuple) and len(ev) >= 2:
            final_c = float(ev[0])
    except Exception:
        pass

    return best_h, final_c, len(best_h)
