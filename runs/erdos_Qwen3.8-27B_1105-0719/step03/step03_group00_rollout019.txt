import numpy as np
import time

def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def c5_of_h(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full').max() * (2.0 / n)

def grad_lq(h, q):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = max(corr.max(), 1e-30)
    a = np.maximum(corr, 0.0) / max_c
    with np.errstate(divide='ignore'):
        log_w = (q - 1) * np.log(np.maximum(a, 1e-30))
    log_w[np.isinf(log_w)] = -1e10
    max_log = log_w.max()
    w = np.exp(log_w - max_log)
    
    g = np.zeros(n)
    g += w[n-1] * (1.0 - 2.0 * h)
    for d in range(1, n):
        wm = w[n-1-d]
        wp = w[n-1+d]

        g[:K] -= wp * h[d:]

            g = grad_lq(cur, q)
            gn = np.linalg.norm(g)
            if gn < 1e-12:

def pair_coefficients(h, p, q):
    """Compute a[i] = dc[i]/dh[p] - dc[i]/dh[q] for all i."""
    n = len(h)
    a = np.zeros(2*n - 1)
    
    # d=0: position n-1
    a[n-1] = (1 - 2*h[p]) - (1 - 2*h[q])
    # = 2*(h[q] - h[p])
    
    for d in range(1, n):
        wm = 1.0  # weight (we just need the unweighted derivative for the pair move)
        K = n - d
        
        # Contribution from p:
        if p < K:
            a[n-1-d] += (1 - h[p+d])
            a[n-1+d] += (-h[p+d])
        if p >= d:
            a[n-1-d] += (-h[p-d])
            a[n-1+d] += (1 - h[p-d])
        
        # Contribution from q (subtracted):
        if q < K:
            a[n-1-d] -= (1 - h[q+d])
            a[n-1+d] -= (-h[q+d])
        if q >= d:
            a[n-1-d] -= (-h[q-d])
            a[n-1+d] -= (1 - h[q-d])
    
    return a

def compute_jacobian(h):
    n = len(h)
    J = np.zeros((2*n-1, n))
    J[n-1] = 1.0 - 2.0 * h
    for d in range(1, n):
        K = n - d
        J[n-1-d, :K] += (1.0 - h[d:])
        J[n-1-d, d:] += (-h[:K])
        J[n-1+d, d:] += (1.0 - h[:K])
        J[n-1+d, :K] += (-h[d:])
    return J

Now for the pair descent:
- Precompute J (n × (2n-1) matrix, J[i][p] = dc[i]/dh[p])
- For pair (p, q): a = J[:, p] - J[:, q]
- Feasible t: L = max(-h[p], h[q]-1), U = min(1-h[p], h[q])
- Solve: min_t max_i(corr[i] + a[i]*t) for t in [L, U]

For the 1D minimization, since f(t) is convex PL:
- Evaluate at L, U, and the "kink points"
- The kink points are where two lines cross: (c[i] + a[i]*t) = (c[j] + a[j]*t)
  → t = (c[j] - c[i]) / (a[i] - a[j]) (when a[i] ≠ a[j])

2. Find the best
3. Do 10 golden section iterations in the neighborhood

Actually, even simpler: since f is convex, I can use the fact that the minimum is at a point where the subgradient contains 0. I can find this by:
1. Compute f'(t) (the subgradient) at several points
2. Binary search for where f' crosses 0

The subgradient of f at t is [min_{i active} a[i], max_{i active} a[i]] where active are the indices achieving the max.

If at some t, min active a[i] <= 0 <= max active a[i], then t is optimal.

Practical approach:
1. Try t=0. If subgradient contains 0, done.
2. Otherwise, the optimal t is in the direction of the sign of the subgradient.
3. Use bisection: at each step, evaluate the subgradient and narrow the interval.
