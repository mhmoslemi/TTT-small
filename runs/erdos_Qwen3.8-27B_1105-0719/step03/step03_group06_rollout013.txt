            h[idx] += add
            remaining -= add
            if remaining <= 1e-14:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            if h[idx] <= 1e-15:
                continue
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-14:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] = float(np.clip(h[idx] + diff2, 0.0, 1.0))
    return h

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def grad_row(h, m, n):
    dx = 2.0 / n
    d = m - (n - 1)
    if d == 0:
        return dx * (1.0 - 2.0 * h)
    elif d > 0:
        g = np.zeros(n)
        g[d:] += dx * (1.0 - h[:-d])
        g[:-d] -= dx * h[d:]
        return g
    else:
        p = -d
        g = np.zeros(n)
        g[:-p] += dx * (1.0 - h[p:])
        g[p:] -= dx * h[:-p]
        return g

def active_descent_v2(h, max_iters=600, start_tol=0.002, end_tol=0.0005,
                      start_mom=0.3, end_mom=0.6, max_active=35):
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    mom_vec = np.zeros(n)
    stall_count = 0

    for it in range(max_iters):
        frac = it / max(max_iters - 1, 1)
        tol = start_tol + (end_tol - start_tol) * frac
        mom = start_mom + (end_mom - start_mom) * frac

        profile = compute_profile(h)
        c = float(profile.max())
        thresh = c - tol
        active = np.where(profile >= thresh)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(profile))])
        if len(active) > max_active:
            top_idx = np.argsort(profile[active])[-max_active:]
            active = active[np.sort(top_idx)]

        g = np.zeros(n)
        for m in active:
            g += grad_row(h, m, n)
        g -= g.mean()
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-14:
            break

        mom_vec = mom * mom_vec + g
        d = -mom_vec
        d_norm = np.linalg.norm(d)
        if d_norm < 1e-14:
            break
        d = d / d_norm

        # Armijo backtracking line search
        pos_mask = d > 1e-12
        neg_mask = d < -1e-12
        alpha_max = 1.0
        if pos_mask.any():
            alpha_max = min(alpha_max, float(((1.0 - h[pos_mask]) / d[pos_mask]).min()))
        if neg_mask.any():
            alpha_max = min(alpha_max, float((h[neg_mask] / (-d[neg_mask])).min()))
        alpha_max = max(alpha_max, 1e-10)

        alpha = min(0.15, alpha_max)
        best_alpha = 0.0
        best_val = c + 1e10
        best_h = h.copy()

        for _ in range(25):
            h_try = repair_feasible(h + alpha * d)
            v = float(np.max(np.correlate(h_try, 1 - h_try, mode="full") * (2.0 / n)))
            if v < best_val:
                best_val = v
                best_alpha = alpha
                best_h = h_try.copy()
            if v <= c - 1e-4 * alpha * gnorm / max(d_norm, 1e-10):
                break
            alpha *= 0.6

        if best_val >= c - 1e-14:
            stall_count += 1
            if stall_count > 15:
                break
            # Try much smaller step
            small_alpha = min(0.01, alpha_max)
            h_try = repair_feasible(h + small_alpha * d)
            v = exact_score(h_try)
            if v < c - 1e-14:
                h = h_try
                if v < best_c:
                    best_c = v
            else:
                break
        else:
            stall_count = 0
            h = best_h
            if best_val < best_c:
                best_c = best_val

    return h, best_c

def slsqp_epigraph(h_start, maxiter=200):
    n = len(h_start)
    dx = 2.0 / n
    M = 2 * n - 1
    c0 = exact_score(h_start)
    t0 = c0 + 1e-4

    def obj(x):
        return x[-1]

    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g

    def eq_fun(x):
        return x[:-1].sum() - n / 2.0

    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g

    def ineq_fun(x):
        h = x[:-1]
        t = x[-1]
        profile = np.correlate(h, 1 - h, mode="full") * dx
        return t - profile

    def ineq_jac(x):
        h = x[:-1]
        J = np.zeros((M, n + 1))
        for m in range(M):
            J[m, :n] = -grad_row(h, m, n)
        J[:, -1] = 1.0
        return J

    x0 = np.concatenate([h_start, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 2.0)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-13})
        if res.x is not None and np.isfinite(res.x).all():
            h_new = repair_feasible(res.x[:-1])
            c_new = exact_score(h_new)
            if c_new <= c0 + 1e-8:
                return h_new
    except Exception:
        pass
    return None

def interp_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    edges_old = np.linspace(0.0, 2.0, n + 1)
    edges_new = np.linspace(0.0, 2.0, new_n + 1)
    h_new = np.zeros(new_n)
    for k in range(new_n):
        a, b = edges_new[k], edges_new[k + 1]
        total = 0.0
        for j in range(n):
            lo = max(a, edges_old[j])
            hi = min(b, edges_old[j + 1])
            if hi > lo:
                total += h[j] * (hi - lo)
        h_new[k] = total / (b - a)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

def coordinate_polish(h, max_iters=150, tol=0.001):
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    dx = 2.0 / n
    for it in range(max_iters):
        profile = compute_profile(h)
        c = float(profile.max())
        thresh = c - tol
        active = np.where(profile >= thresh)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(profile))])
        G = np.array([grad_row(h, m, n) for m in active])

        improved = False
        idx_desc = np.argsort(-h)
        idx_asc = np.argsort(h)
        for attempt in range(min(40, 2 * n)):
            i = int(idx_desc[attempt % n])
            j = int(idx_asc[attempt % n])
            if i == j:
                continue
            base_delta = 0.03 * (h[i] - h[j])
            if abs(base_delta) < 1e-6:
                continue
            for sign in [1.0, -1.0]:
                for scale in [1.0, 0.5, 0.25]:
                    delta = sign * base_delta * scale
                    if abs(delta) < 1e-7:
                        continue
                    h_try = h.copy()
                    h_try[i] += delta
                    h_try[j] -= delta
                    if h_try.min() < -1e-10 or h_try.max() > 1 + 1e-10:
                        continue
                    h_try = repair_feasible(h_try)
                    s = float(np.max(np.correlate(h_try, 1 - h_try, mode="full") * dx))
                    if s < best_c - 1e-13:
                        h = h_try
                        best_c = s
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        if not improved:
            break
    return h, best_c

def structured_seed(n, rng, style):
    h = np.zeros(n)
    if style == 0:
        # Golden ratio low discrepancy
        pts = (np.arange(n) * 0.618033988749895) % 1.0
        cutoff = 0.5
        h[pts < cutoff] = 1.0
        h[pts >= cutoff] = 0.0
    elif style == 1:
        # Van der Corput (binary)
        vdc = np.zeros(n)
        for i in range(n):
            v = 0.0
            base = 0.5
            j = i
            while j > 0:
                if j % 2 == 1:
                    v += base
                j //= 2
                base *= 0.5
            vdc[i] = v
        h[vdc <= 0.5] = 1.0
    elif style == 2:
        # Multi-scale blocks
        block = max(3, n // 12)
        phase = 0
        for i in range(n):
            pos_in_block = i % block
            if pos_in_block < block // 2:
                h[i] = float(block // 2) / block
            else:
                h[i] = 0.0
    elif style == 3:
        # Smooth sinusoidal
        h = 0.5 + 0.4 * np.sin(2 * np.pi * 3 * np.linspace(0, 1, n))
        h = np.clip(h, 0, 1)
    else:
        # Random
        perm = rng.permutation(n)
        h[perm[:n // 2]] = 1.0
    h = np.clip(h, 0.0, 1.0)
    return repair_feasible(h)

def blurred_seed(h_raw, width):
    """Smooth a raw array with a box filter of given width"""
    n = len(h_raw)
    if width <= 1:
        return repair_feasible(h_raw)
    kernel = np.ones(width) / width
    h_pad = np.concatenate([h_raw[-width + 1:], h_raw, h_raw[:width - 1]])
    h_blur = np.array([np.dot(h_pad[i:i + width], kernel) for i in range(n)])
    return repair_feasible(np.clip(h_blur, 0, 1))

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s - (time.perf_counter() - t_start)

    try:
        parent = np.array(initial_h_values, dtype=np.float64)
    except Exception:
        parent = None
    if parent is None or len(parent) < 4:
        parent = np.full(80, 0.5)
    parent = np.clip(parent, 0.0, 1.0)
    parent = repair_feasible(parent)

    best_h = parent.copy()
    best_c = exact_score(parent)

    # Stage 1: Thorough active descent on parent
    if time_left() > 120:
        h_ad, c_ad = active_descent_v2(best_h, max_iters=500, start_tol=0.004,
                                        end_tol=0.0005, start_mom=0.3, end_mom=0.65)
        if c_ad < best_c:
            best_h, best_c = h_ad, c_ad

    # Stage 2: Try different dimensions
    n_best = len(best_h)
    dim_list = []
    for dn in range(-10, 21, 2):
        d = n_best + dn
        if 48 <= d <= 120 and d % 2 == 0:
            dim_list.append(d)
    dim_list.sort(key=lambda x: abs(x - n_best))

    for new_n in dim_list[:10]:
        if time_left() < 40:
            break
        h_interp = interp_to_n(best_h, new_n)
        h_ad, c_ad = active_descent_v2(h_interp, max_iters=350, start_tol=0.003,
                                        end_tol=0.0005, start_mom=0.35, end_mom=0.6)
        if c_ad < best_c:
            best_h, best_c = h_ad, c_ad

    # Stage 3: SLSQP refinement
    for _ in range(4):
        if time_left() < 35:
            break
        h_sol = slsqp_epigraph(best_h, maxiter=180)
        if h_sol is not None:
            c_sol = exact_score(h_sol)
            if c_sol < best_c - 1e-12:
                best_h, best_c = h_sol, c_sol
            if time_left() > 30:
                h_ad, c_ad = active_descent_v2(best_h, max_iters=200, start_tol=0.002,
                                                end_tol=0.0003, start_mom=0.4, end_mom=0.6)
                if c_ad < best_c:
                    best_h, best_c = h_ad, c_ad
        else:
            break

    # Stage 4: Structured seeds at multiple dimensions
    if time_left() > 150:
        for new_n in [n_best, n_best + 4, n_best - 4, n_best + 8]:
            if time_left() < 30:
                break
            if new_n < 48 or new_n > 140 or new_n % 2 != 0:
                continue
            for style in range(5):
                if time_left() < 20:
                    break
                h_s = structured_seed(new_n, rng, style)
                h_blur = blurred_seed(h_s, 3)
                h_mix = 0.3 * h_blur + 0.7 * best_h[:len(best_h)]
                if len(h_mix) != len(h_blur):
                    h_mix = 0.5 * h_blur + 0.5 * interp_to_n(best_h, new_n)
                h_mix = repair_feasible(h_mix)
                h_ad, c_ad = active_descent_v2(h_mix, max_iters=250, start_tol=0.004,
                                                end_tol=0.001, start_mom=0.3, end_mom=0.5)
                if c_ad < best_c:
                    best_h, best_c = h_ad, c_ad

    # Stage 5: Random perturbations of current best
    if time_left() > 100:
        n_b = len(best_h)
        for trial in range(6):
            if time_left() < 20:
                break
            noise_scale = 0.05 + 0.03 * trial
            h_pert = best_h + noise_scale * (rng.randn(n_b) - 0.5)
            h_pert = repair_feasible(h_pert)
            h_ad, c_ad = active_descent_v2(h_pert, max_iters=200, start_tol=0.003,
                                            end_tol=0.0005, start_mom=0.3, end_mom=0.55)
            if c_ad < best_c:
                best_h, best_c = h_ad, c_ad

    # Stage 6: More dimension exploration from improved best
    if time_left() > 80:
        n_b = len(best_h)
        for new_n in [n_b + 2, n_b - 2, n_b + 6, n_b - 6, n_b + 10, n_b - 10]:
            if time_left() < 25:
                break
            if new_n < 48 or new_n > 140 or new_n % 2 != 0:
                continue
            h_interp = interp_to_n(best_h, new_n)
            h_ad, c_ad = active_descent_v2(h_interp, max_iters=300, start_tol=0.002,
                                            end_tol=0.0003, start_mom=0.35, end_mom=0.6)
            if c_ad < best_c:
                best_h, best_c = h_ad, c_ad
                if time_left() > 30:
                    h_sol = slsqp_epigraph(best_h, maxiter=120)
                    if h_sol is not None:
                        c_sol = exact_score(h_sol)
                        if c_sol < best_c:
                            best_h, best_c = h_sol, c_sol

    # Stage 7: Fine-tuning active descent
    if time_left() > 40:
        h_ad, c_ad = active_descent_v2(best_h, max_iters=400, start_tol=0.001,
                                        end_tol=0.0002, start_mom=0.4, end_mom=0.65)
        if c_ad < best_c:
            best_h, best_c = h_ad, c_ad

    # Stage 8: Coordinate polish
    if time_left() > 30:
        h_cp, c_cp = coordinate_polish(best_h, max_iters=120, tol=0.0005)
        if c_cp < best_c:
            best_h, best_c = h_cp, c_ad

    # Stage 9: Final SLSQP
    if time_left() > 20:
        h_sol = slsqp_epigraph(best_h, maxiter=250)
        if h_sol is not None:
            c_sol = exact_score(h_sol)
            if c_sol < best_c:
                best_h = h_sol
                best_c = c_sol
                if time_left() > 10:
                    h_ad, c_ad = active_descent_v2(best_h, max_iters=150, start_tol=0.0005,
                                                    end_tol=0.0002, start_mom=0.5, end_mom=0.7)
                    if c_ad < best_c:
                        best_h, best_c = h_ad, c_ad

    # Stage 10: Ultra-fine active descent
    if time_left() > 10:
        h_ad, c_ad = active_descent_v2(best_h, max_iters=200, start_tol=0.0003,
                                        end_tol=0.0001, start_mom=0.5, end_mom=0.7)
        if c_ad < best_c:
            best_h, best_c = h_ad, c_ad

    # Final fix
    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)

    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            final_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 2:
            final_c = float(eval_result[0])
    except Exception:
        pass

    return best_h, final_c, len(best_h)
