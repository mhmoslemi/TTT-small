```python
        d[m] = dx * (coeff_j - coeff_i)

    # So δ in [h[i]-1, min(h[i], 1-h[j])]
    
    lo = h[i] - 1.0
    hi = min(h[i], 1.0 - h[j])

    # Or just evaluate at several points

    # The breakpoints are where two lines cross: profile[m1] + δ*d[m1] = profile[m2] + δ*d[m2]
    # δ = (profile[m2] - profile[m1]) / (d[m1] - d[m2])

        if val < best_val:

        for _ in range(3):

            candidates = candidates[(candidates >= lo) & (candidates <= hi)]
            for δ in candidates:

                if val < best_val:
                    best_val = val
                    best_delta = δ
            step *= 0.3
    
    if best_delta != 0:
        h[i] -= best_delta
        h[j] += best_delta
        h = repair_feasible(h)
    
    return h, compute_profile(h).max()
```

```python
def build_gradient_matrix(h, n, dx):
    """G[m, i] = ∂(overlap(m)*dx)/∂h[i]"""
    M = 2 * n - 1
    G = np.zeros((M, n))
    center = n - 1
    
    for s in range(n):  # s >= 0
        m_pos = center + s
        # overlap(s) = sum_{a=0}^{n-1-s} h[a]*(1-h[a+s])
        # d/dh[a] = (1-h[a+s]) for a in [0, n-1-s]
        # d/dh[a+s] = -h[a] for a in [0, n-1-s], i.e., d/dh[b] = -h[b-s] for b in [s, n-1]
        a_range = np.arange(n - s)
        G[m_pos, a_range] += dx * (1.0 - h[a_range + s])
        G[m_pos, a_range + s] += -dx * h[a_range]
        
        if s > 0:
            m_neg = center - s
            # overlap(-s) = sum_{a=0}^{n-1-s} h[a+s]*(1-h[a])
            # d/dh[a+s] = (1-h[a]) for a in [0, n-1-s]
            # d/dh[a] = -h[a+s] for a in [0, n-1-s]
            G[m_neg, a_range + s] += dx * (1.0 - h[a_range])
            G[m_neg, a_range] += -dx * h[a_range + s]
    
    return G
```

```python
                G = build_G(h, n, dx)
                n_accepts += 1

        if n_accepts > 0 and c_curr < c_sweep_start - 1e-10:
            no_improve = 0
        else:
            no_improve += 1

    return h, c_curr


def subgradient_step(h, alpha):
    n = len(h)
    dx = 2.0 / n
    profile = compute_profile(h)
    c_max = profile.max()
    thresh = c_max - 0.01
    active = np.where(profile >= thresh)[0]
    if len(active) == 0:
        active = np.array([np.argmax(profile)])
    w = np.ones(len(active)) / len(active)
    G = build_G(h, n, dx)
    grad = G[active, :].T @ w
    grad -= grad.mean()
    h_new = h - alpha * grad
    return repair_feasible(h_new)


def random_01_init(n, rng):
    h = np.zeros(n)
    half = n // 2
    idx = rng.choice(n, half, replace=False)
    h[idx] = 1.0
    if n % 2 == 1:
        h[idx[0]] = h.mean() * 2.0 / (n / 2.0)
    return repair_feasible(h)


def smooth_init(n, rng, freq=1.0, noise=0.05):
    x = np.linspace(0, 2, n)
    h = 0.5 + 0.4 * np.sin(2 * np.pi * freq * x / 2.0)
    h += noise * rng.randn(n)
    return repair_feasible(np.clip(h, 0, 1))


def block_optimize(h, block_size, rng, time_fn, min_time=10.0, n_iter=50):
    """Optimize blocks of consecutive coordinates together."""
    n = len(h)
    dx = 2.0 / n
    c_curr = compute_profile(h).max()

    for it in range(n_iter):
        if time_fn() < min_time:
            break
        start = int(rng.randint(0, max(1, n - block_size)))
        block = np.arange(start, start + block_size)
        block_sum = h[block].sum()
        c_before = c_curr

        for _ in range(20):
            G = build_G(h, n, dx)
            profile = compute_profile(h)
            # Gradient w.r.t. block coordinates
            gb = G[:, block].sum(axis=1)
            # Direction: move block coords in negative gradient direction
            # while preserving their sum
            delta = -gb @ profile  # scalar
            # Actually use: for each block coord, the gradient component
            grad_block = G[profile.argmax(), block]
            # Simple: shift mass within block from high-gradient to low-gradient
            order = np.argsort(grad_block)
            lo_idx = block[order[0]]
            hi_idx = block[order[-1]]
            # Try small transfer
            transfer = 0.02
            h_try = h.copy()
            h_try[lo_idx] += transfer
            h_try[hi_idx] -= transfer
            h_try = repair_feasible(h_try)
            c_try = exact_score(h_try)
            if c_try < c_curr - 1e-10:
                h = h_try
                c_curr = c_try
            else:
                transfer = -transfer
                h_try = h.copy()
                h_try[lo_idx] += transfer
                h_try[hi_idx] -= transfer
                h_try = repair_feasible(h_try)
                c_try = exact_score(h_try)
                if c_try < c_curr - 1e-10:
                    h = h_try
                    c_curr = c_try
                else:
                    break

        if c_curr >= c_before - 1e-12:
            break

    return h, c_curr


def annealed_pairs(h, rng, time_fn, min_time=30.0, T0=0.01, T1=0.0001, n_steps=2000):
    """Simulated annealing with pair transfers."""
    n = len(h)
    dx = 2.0 / n
    c_curr = compute_profile(h).max()
    ii, jj = make_pair_indices(n)
    n_pairs = len(ii)
    best_h = h.copy()
    best_c = c_curr

    for step in range(n_steps):
        if time_fn() < min_time:
            break
        T = T0 * (T1 / T0) ** (step / max(1, n_steps))
        i = int(ii[rng.randint(n_pairs)])
        j = int(jj[rng.randint(n_pairs)])
        delta = (rng.rand() - 0.5) * 0.1
        if h[i] + delta < 0 or h[i] + delta > 1:
            continue
        if h[j] - delta < 0 or h[j] - delta > 1:
            continue
        h[i] += delta
        h[j] -= delta
        c_new = exact_score(h)
        delta_c = c_new - c_curr
        if delta_c < 0 or rng.rand() < np.exp(-delta_c / max(T, 1e-8)):
            c_curr = c_new
            if c_curr < best_c:
                best_c = c_curr
                best_h = h.copy()
        else:
            h[i] -= delta
            h[j] += delta

    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()
    rng = np.random.RandomState(seed)

    def time_fn():
        return budget_s - (time.perf_counter() - t_start)

    # Load parent
    try:
        parent = np.array(initial_h_values, dtype=np.float64)
    except (NameError, Exception):
        parent = np.full(70, 0.5)

    parent = repair_feasible(np.clip(parent, 0.0, 1.0))
    best_h = parent.copy()
    best_c = exact_score(parent)

    # Symmetrize
    h_sym = repair_feasible((parent + parent[::-1]) / 2.0)
    c_sym = exact_score(h_sym)
    if c_sym < best_c - 1e-12:
        best_h, best_c = h_sym, c_sym

    # Phase 1: Intensive pair sweep on parent
    if time_fn() > 120:
        h_try = best_h.copy()
        h_try, c_try = pair_sweep(h_try, rng, time_fn, min_time=100, n_grid=31)
        if c_try < best_c - 1e-12:
            best_h, best_c = h_try, c_try

    # Phase 2: Multi-dimensional exploration
    if time_fn() > 200:
        n0 = len(best_h)
        dims = [32, 36, 40, 44, 48, 50, 52, 54, 56, 58, 60, 62, 64, 66, 68, 70, 72, 74, 76, 78, 80]
        dims = [d for d in dims if d != n0]
        dims.sort(key=lambda d: abs(d - n0))

        for nd in dims[:10]:
            if time_fn() < 60:
                break
            h_try = interp_h(best_h, nd)
            c_try = exact_score(h_try)
            if c_try < best_c + 0.03:
                h_try, c_try = pair_sweep(h_try, rng, time_fn, min_time=40, n_grid=21)
                if c_try < best_c - 1e-12:
                    best_h, best_c = h_try, c_try

    # Phase 3: Subgradient escape + pair sweep
    if time_fn() > 100:
        for alpha in [0.03, 0.02, 0.01, 0.005, 0.003, 0.001]:
            if time_fn() < 40:
                break
            h_try = subgradient_step(best_h, alpha)
            c_try = exact_score(h_try)
            if c_try < best_c + 0.02:
                h_try, c_try = pair_sweep(h_try, rng, time_fn, min_time=30, n_grid=21)
                if c_try < best_c - 1e-12:
                    best_h, best_c = h_try, c_try
                    break

    # Phase 4: Random perturbation + pair sweep
    if time_fn() > 80:
        for eps in [0.03, 0.05, 0.08, 0.12, 0.18]:
            if time_fn() < 30:
                break
            z = rng.randn(len(best_h))
            z -= z.mean()
            z /= (np.linalg.norm(z) / len(best_h) + 1e-15)
            h_try = repair_feasible(best_h + eps * z)
            c_try = exact_score(h_try)
            if c_try < best_c + 0.02:
                h_try, c_try = pair_sweep(h_try, rng, time_fn, min_time=25, n_grid=21)
                if c_try < best_c - 1e-12:
                    best_h, best_c = h_try, c_try

    # Phase 5: Different structured initializations
    if time_fn() > 120:
        n0 = len(best_h)
        inits = []
        # Checkerboard
        h_cb = np.zeros(n0)
        h_cb[::2] = 1.0
        h_cb = repair_feasible(h_cb)
        inits.append(h_cb)
        # Sine
        x = np.linspace(0, 2, n0)
        h_sin = 0.5 + 0.45 * np.sin(3 * np.pi * x)
        inits.append(repair_feasible(np.clip(h_sin, 0, 1)))
        h_sin2 = 0.5 + 0.45 * np.sin(5 * np.pi * x + np.pi / 4)
        inits.append(repair_feasible(np.clip(h_sin2, 0, 1)))
        # Random 01
        inits.append(random_01_init(n0, rng))
        # Ternary
        h_ter = np.zeros(n0)
        h_ter[::3] = 1.0
        h_ter[1::3] = 0.5
        inits.append(repair_feasible(h_ter))

        for h_init in inits:
            if time_fn() < 50:
                break
            c_init = exact_score(h_init)
            if c_init < best_c + 0.04:
                h_try, c_try = pair_sweep(h_init.copy(), rng, time_fn, min_time=35, n_grid=21)
                if c_try < best_c - 1e-12:
                    best_h, best_c = h_try, c_try

    # Phase 6: Annealed pairs on best
    if time_fn() > 60:
        h_try, c_try = annealed_pairs(best_h.copy(), rng, time_fn, min_time=30,
                                      T0=0.005, T1=5e-5, n_steps=3000)
        if c_try < best_c - 1e-12:
            best_h, best_c = h_try, c_try
            # Follow up with pair sweep
            h_try, c_try = pair_sweep(best_h.copy(), rng, time_fn, min_time=20, n_grid=25)
            if c_try < best_c - 1e-12:
                best_h, best_c = h_try, c_try

    # Phase 7: Multi-dimension with structured inits
    if time_fn() > 100:
        for nd in [n0 - 4, n0 + 4, n0 - 8, n0 + 8]:
            if time_fn() < 40 or nd < 20:
                continue
            h_try = interp_h(best_h, nd)
            c_try = exact_score(h_try)
            if c_try < best_c + 0.02:
                h_try, c_try = pair_sweep(h_try, rng, time_fn, min_time=30, n_grid=25)
                if c_try < best_c - 1e-12:
                    best_h, best_c = h_try, c_try

    # Phase 8: Final intensive pair sweep with fine grid
    if time_fn() > 40:
        h_try, c_try = pair_sweep(best_h.copy(), rng, time_fn, min_time=10, n_grid=41)
        if c_try < best_c - 1e-12:
            best_h, best_c = h_try, c_try

    # Phase 9: Very fine local search - try tiny perturbations
    if time_fn() > 20:
        n_f = len(best_h)
        c_f = best_c
        for _ in range(200):
            if time_fn() < 8:
                break
            i = int(rng.randint(n_f))
            j = int(rng.randint(n_f))
            if i == j:
                continue
            for delta in [0.005, -0.005, 0.002, -0.002, 0.001, -0.001]:
                h_try = best_h.copy()
                h_try[i] += delta
                h_try[j] -= delta
                if h_try[i] < 0 or h_try[i] > 1 or h_try[j] < 0 or h_try[j] > 1:
                    continue
                h_try = repair_feasible(h_try)
                c_try = exact_score(h_try)
                if c_try < c_f - 1e-12:
                    best_h = h_try
                    c_f = c_try
                    break
        if c_f < best_c - 1e-12:
            best_h, best_c = best_h, c_f
            # One more pair sweep to consolidate
            h_try, c_try = pair_sweep(best_h.copy(), rng, time_fn, min_time=5, n_grid=31)
            if c_try < best_c - 1e-12:
                best_h, best_c = h_try, c_try

    # Final validation
    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)

    try:
        er = evaluate_erdos_solution(best_h)
        if isinstance(er, (float, int)):
            final_c = min(final_c, float(er))
        elif isinstance(er, tuple) and len(er) >= 2:
            final_c = min(final_c, float(er[0]))
    except Exception:
        pass

    return best_h, final_c, len(best_h)
```
