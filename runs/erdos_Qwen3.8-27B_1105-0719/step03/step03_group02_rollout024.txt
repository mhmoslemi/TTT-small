import numpy as np
from scipy.optimize import minimize, linprog
from scipy.special import logsumexp
import time


def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)


def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def c5_of_h(h, dx):
    return compute_corrs(h).max() * dx


def compute_sensitivity(h, n):
    S = np.zeros((2 * n - 1, n))
    for L in range(2 * n - 1):
        s = L - (n - 1)
        i_lo1 = max(0, -s)
        i_hi1 = min(n - 1, n - 1 - s)
        if i_lo1 <= i_hi1:
            idx = np.arange(i_lo1, i_hi1 + 1)
            S[L, idx] += (1.0 - h[idx + s])
        i_lo2 = max(0, s)
        i_hi2 = min(n - 1, n - 1 + s)
        if i_lo2 <= i_hi2:
            idx = np.arange(i_lo2, i_hi2 + 1)
            S[L, idx] += (-h[idx - s])
    return S


def exact_1d_min(corrs, d, eps_lo, eps_hi, n_iters=55):
    if eps_hi - eps_lo < 1e-14:
        return eps_lo, corrs.max()
    lo, hi = eps_lo, eps_hi
    for _ in range(n_iters):
        if hi - lo < 1e-15:
            break
        mid = (lo + hi) / 2
        vals = corrs + mid * d
        f_val = vals.max()
        active = vals >= f_val - 1e-12
        d_min = d[active].min()
        d_max = d[active].max()
        if d_min > 1e-10:
            hi = mid
        elif d_max < -1e-10:
            lo = mid
        else:
            break
    eps_opt = max(eps_lo, min(eps_hi, (lo + hi) / 2))
    f_opt = (corrs + eps_opt * d).max()
    return eps_opt, f_opt


def selective_pair_transfer(h0, dx, target_sum, max_sweeps=100, top_k=25, inner_iters=8):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    beta = 150.0
    for sweep in range(max_sweeps):
        corrs = compute_corrs(h)
        S = compute_sensitivity(h, n)
        shifted = corrs - corrs.max()
        w = np.exp(beta * shifted)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(2 * n - 1) / (2 * n - 1)
        else:
            w /= w_sum
        grad = S.T @ w
        dec_idx = np.argsort(grad)[-top_k:]
        inc_idx = np.argsort(grad)[:top_k]
        sweep_improved = False
        for _ in range(inner_iters):
            best_improv = 0.0
            best_eps = 0.0
            best_i, best_j = -1, -1
            for i in dec_idx:
                for j in inc_idx:
                    if i == j:
                        continue
                    d = S[:, i] - S[:, j]
                    eps_hi = min(h[i], 1.0 - h[j])
                    if eps_hi < 1e-13:
                        continue
                    eps_opt, f_opt = exact_1d_min(corrs, d, 0.0, eps_hi)
                    improv = corrs.max() - f_opt
                    if improv > best_improv:
                        best_improv = improv
                        best_eps = eps_opt
                        best_i, best_j = i, j
            if best_i < 0 or best_improv < 1e-13:
                break
            h[best_i] -= best_eps
            h[best_j] += best_eps
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target_sum)

            S = compute_sensitivity(h, n)

def anneal_pair_search(h0, dx, target_sum, t_start=0.05, t_end=0.001, n_iters=3000, top_k=20, rng=None):

def lp_polish(h0, dx, target_sum, max_iters=200, init_step=0.12):

    best_c5 = c5_of_h(h, dx)
    step = init_step
    stall = 0
    for it in range(max_iters):
        corrs = compute_corrs(h)
        c5_val = corrs.max() * dx
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 50:
            step *= 0.6
            stall = 0
            if step < 1e-8:
                break
        max_c = corrs.max()
        n_active = max(20, min(80, int(0.3 * (2 * n - 1))))
        active_k = np.argsort(corrs)[-n_active:]
        S = compute_sensitivity(h, n)
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        A_rows = []
        b_rows = []
        for k in active_k:
            row = np.zeros(n + 1)
            row[:n] = dx * S[k, :]
            row[n] = -1.0
            A_rows.append(row)
            b_rows.append(-corrs[k] * dx - 1e-13)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)

            res = linprog(c_obj, A_ub=np.array(A_rows), b_ub=np.array(b_rows),
                          A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
        except Exception:
            break
        if res.status != 0:
            break

        if np.max(np.abs(delta)) < 1e-14:
            break
        h_new = project_feasible(h + delta, target_sum)
        new_c5 = c5_of_h(h_new, dx)
        if new_c5 < c5_val - 1e-14:
            h = h_new
            step = min(step * 1.3, init_step)
        else:
            improved = False

                h_new = project_feasible(h + frac * delta, target_sum)
                new_c5 = c5_of_h(h_new, dx)

def slsqp_refine(h0, dx, target_sum, beta=800, maxiter=800):
    n = len(h0)

    def obj(x):
        corrs = compute_corrs(x)
        return logsumexp(beta * corrs) / beta * dx

    def grad_obj(x):
        corrs = compute_corrs(x)
        shifted = corrs - corrs.max()
        w = np.exp(beta * shifted)
        w /= w.sum()
        S = compute_sensitivity(x, n)
        return (S.T @ w) * dx

    def c_eq_fun(x):
        return x.sum() - target_sum

    def c_eq_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': c_eq_fun, 'jac': c_eq_jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0, jac=grad_obj, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        h_out = project_feasible(res.x, target_sum)
    except Exception:
        h_out = project_feasible(h0, target_sum)
    return h_out, c5_of_h(h_out, dx)

                best_c5 = c5_s
                best_h = h_s
    # Phase C: LP polish
    if time_left_fn() > 20:
        h_lp, c5_lp = lp_polish(best_h, dx, target_sum, max_iters=150, init_step=0.1)
        if c5_lp < best_c5:
            best_c5 = c5_lp
            best_h = h_lp
    # Phase D: More pair transfer
    if time_left_fn() > 20:
        h_pt2, c5_pt2 = selective_pair_transfer(best_h, dx, target_sum,
                                                max_sweeps=40, top_k=35, inner_iters=5)
        if c5_pt2 < best_c5:
            best_c5 = c5_pt2
            best_h = h_pt2
    # Phase E: SLSQP again
    if time_left_fn() > 20:
        h_s2, c5_s2 = slsqp_refine(best_h, dx, target_sum, beta=4000, maxiter=800)
        if c5_s2 < best_c5:
            best_c5 = c5_s2
            best_h = h_s2
    # Phase F: Final LP
    if time_left_fn() > 15:
        h_lp2, c5_lp2 = lp_polish(best_h, dx, target_sum, max_iters=100, init_step=0.06)
        if c5_lp2 < best_c5:
            best_c5 = c5_lp2
            best_h = h_lp2
    # Phase G: Final pair transfer
    if time_left_fn() > 10:
        h_pt3, c5_pt3 = selective_pair_transfer(best_h, dx, target_sum,
                                                max_sweeps=30, top_k=40, inner_iters=4)
        if c5_pt3 < best_c5:
            best_c5 = c5_pt3
            best_h = h_pt3
    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s - (time.time() - t0)

    # Start from initial construction
    h0 = np.array(initial_h_values, dtype=float)
    n = len(h0)
    dx = 2.0 / n
    target_sum = n / 2.0
    h0 = project_feasible(h0, target_sum)

    best_h = h0.copy()
    best_c5 = c5_of_h(h0, dx)

    # Main optimization
    h_opt, c5_opt = full_pipeline(h0, dx, target_sum, time_left, "main")
    if c5_opt < best_c5:
        best_c5 = c5_opt
        best_h = h_opt

    # Annealing phase
    if time_left() > 40:
        h_an, c5_an = anneal_pair_search(best_h, dx, target_sum,
                                         t_start=0.03, t_end=0.001,
                                         n_iters=2000, top_k=20, rng=rng)
        if c5_an < best_c5:
            best_c5 = c5_an
            best_h = h_an
        # Polish the annealed result
        if time_left() > 20:
            h_ap, c5_ap = full_pipeline(h_an, dx, target_sum, time_left, "anneal_polish")
            if c5_ap < best_c5:
                best_c5 = c5_ap
                best_h = h_ap

    # Perturbation restarts
    n_perturb_tries = 12
    for trial in range(n_perturb_tries):
        if time_left() < 35:
            break
        if trial % 3 == 0:
            pert = 0.04 * (rng.random(n) - 0.5)
        elif trial % 3 == 1:
            pert = 0.12 * (rng.random(n) - 0.5)
        else:
            idx = rng.choice(n, size=max(5, n // 8), replace=False)
            pert = np.zeros(n)
            pert[idx] = rng.choice([-1, 1], size=len(idx)) * 0.08
        h_try = project_feasible(best_h + pert, target_sum)
        h_p, c5_p = full_pipeline(h_try, dx, target_sum, time_left, f"pert{trial}")
        if c5_p < best_c5:
            best_c5 = c5_p
            best_h = h_p

    # Try different n values
    n_candidates = [60, 80, 100, 120, 140, 146, 150, 160, 170, 180]
    n_candidates = [x for x in n_candidates if x != n]
    n_candidates.sort(key=lambda x: abs(x - n))
    n_candidates = n_candidates[:8]

    for n_try in n_candidates:
        if time_left() < 40:
            break
        dx_try = 2.0 / n_try
        ts_try = n_try / 2.0
        x_old = np.linspace(0.0, 2.0, len(best_h), endpoint=False) + 1.0 / len(best_h)
        x_new = np.linspace(0.0, 2.0, n_try, endpoint=False) + 1.0 / n_try
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, ts_try)
        h_n, c5_n = full_pipeline(h_interp, dx_try, ts_try, time_left, f"n{n_try}")
        if c5_n < best_c5:
            best_c5 = c5_n
            best_h = h_n

    # Final intensive polish
    if time_left() > 20 and best_h is not None:
        n_f = len(best_h)
        dx_f = 2.0 / n_f
        ts_f = n_f / 2.0
        h_f, c5_f = full_pipeline(best_h, dx_f, ts_f, time_left, "final")
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f

    # Ultra-fine pair transfer
    if time_left() > 15 and best_h is not None:
        n_f = len(best_h)
        dx_f = 2.0 / n_f
        ts_f = n_f / 2.0
        h_pt, c5_pt = selective_pair_transfer(best_h, dx_f, ts_f,
                                              max_sweeps=40, top_k=45, inner_iters=5)
        if c5_pt < best_c5:
            best_c5 = c5_pt
            best_h = h_pt

    # Ensure feasibility
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final / 2.0)
    best_c5 = c5_of_h(best_h, 2.0 / n_final)

    return best_h, best_c5, n_final
