def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))


def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx


def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        # Add to elements with most room
        order = np.argsort(1.0 - h)
        remaining = diff
        for idx in order:
            add = min(remaining, 1.0 - h[idx])
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for idx in order:
            rem = min(remaining, h[idx])
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    # Final correction
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h


def interp_h(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    fx = np.concatenate([[0.0], xs, [2.0]])
    fh = np.concatenate([[h[0]], h, [h[-1]]])
    hn = np.interp(xt, fx, fh)
    return repair_feasible(np.clip(hn, 0, 1))


def make_sawtooth(n, freq=1):
    x = np.arange(n) / n
    h = 0.5 + 0.5 * np.sign(np.sin(2 * np.pi * freq * x))
    # Make it have mean exactly 0.5
    h = np.where(h > 0.5, 1.0, 0.0)
    # Adjust to get sum = n/2
    return repair_feasible(h)

def block_swap(h, rng, block_size):
    """Swap two contiguous blocks of h."""
    n = len(h)
    b = min(block_size, n // 2)
    i = rng.randint(0, n - b)
    j = rng.randint(0, n - b)
    if abs(i - j) < 2 * b:
        return h.copy()
    h_new = h.copy()
    h_new[i:i+b], h_new[j:j+b] = h[j:j+b].copy(), h[i:i+b].copy()
    return h_new


def single_perturb(h, rng, magnitude):
    """Perturb a few elements while maintaining feasibility."""
    n = len(h)
    h_new = h.copy()
    k = rng.randint(1, min(4, n // 3) + 1)
    idx = rng.choice(n, size=k, replace=False)
    h_new[idx] += magnitude * (rng.rand(k) - 0.5)
    return repair_feasible(h_new)


def block_perturb(h, rng, block_size, magnitude):
    """Perturb a block of elements."""
    n = len(h)
    b = min(block_size, n // 4)
    i = rng.randint(0, n - b)
    h_new = h.copy()
    h_new[i:i+b] += magnitude * (rng.rand(b) - 0.5)
    return repair_feasible(h_new)


def coord_descent(h, max_iters=500, tol=1e-14):
    """Coordinate descent: try adjusting each element slightly."""
    n = len(h)
    best = h.copy()
    best_c = exact_score(best)
    
    for it in range(max_iters):
        improved = False
        for i in range(n):
            for delta in [0.01, -0.01, 0.05, -0.05, 0.02, -0.02]:
                h_try = best.copy()
                # Move mass from i to another element to maintain sum
                # Actually, let's just clip and repair
                h_try[i] = max(0.0, min(1.0, h_try[i] + delta))
                h_try = repair_feasible(h_try)
                c_try = exact_score(h_try)
                if c_try < best_c - tol:
                    best = h_try
                    best_c = c_try
                    improved = True
                    break
        if not improved:
            break
    
    return best, best_c

    if not HAS_CVXPY:
        return h0, exact_score(h0)

    stagnation = 0

            break

            stagnation = 0
        else:
            stagnation += 1

        spread = c_max - np.median(profile)
        beam = max(0.003, min(0.02, 0.2 * spread))
        active = np.where(profile >= c_max - beam)[0]
        if len(active) == 0:
            active = np.array([np.argmax(profile)])
        if len(active) > 15:
            active = active[np.argsort(profile[active])[-15:]]

        M = len(active)
        G = np.zeros((M, n))
        center = n - 1

        # For general lag d:
        # c[d] = sum_i h[i+d] * (1-h[i]) for d < n
        # dc[d]/dh[j] = dx * (1-h[j-d]) if j >= d  (from h[j] in first factor)
        #            + dx * (-h[j+d]) if j <= n-1-d  (from (1-h[j]) in second factor)
        
        for ai, a in enumerate(active):
            if a < n:
                d = a
                if d == 0:
                    G[ai, :] = dx * (1.0 - 2.0 * h)
                else:
                    for j in range(d, n):
                        G[ai, j] += dx * (1.0 - h[j - d])
                    for j in range(0, n - d):
                        G[ai, j] += -dx * h[j + d]
            else:
                d = a - (n - 1)  # This is the negative lag
                for j in range(0, n - d):
                    G[ai, j] += dx * (1.0 - h[j + d])
                for j in range(d, n):
                    G[ai, j] += -dx * h[j - d]

        # We want: profile[active] + G[active] @ (h_new - h) <= t
        # i.e., G[active] @ h_new <= t - profile[active] + G[active] @ h

        t_var = cp.Variable()

            h_var <= 1,
            cp.sum(h_var) == n / 2.0,
            h_var >= h - trust,
            h_var <= h + trust,
            G @ h_var - t_var <= b_vec

        prob = cp.Problem(obj, cons)

                solved = True
                break
            except Exception:
                continue

            trust *= 0.7
            if trust < 0.005:
                break
            continue

        h_sol = repair_feasible(h_sol)
        c_sol = exact_score(h_sol)

            h = h_sol
            best_c = c_sol

def simulated_annealing(h0, n_steps, t_start, t_end, budget_fn, rng):
    """Simulated annealing with block and single moves."""
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c = exact_score(h)
    curr_c = best_c
    
    for step in range(n_steps):
        if budget_fn() < 30:
            break
        
        # Temperature
        progress = step / max(1, n_steps - 1)
        T = t_start * (t_end / t_start) ** progress
        
        # Choose move type
        r = rng.random()
        if r < 0.3:
            h_try = block_swap(h, rng, rng.randint(2, max(3, n // 8)))
        elif r < 0.6:
            h_try = block_perturb(h, rng, rng.randint(2, max(3, n // 5)), 0.05 + 0.1 * T / t_start)
        else:
            h_try = single_perturb(h, rng, 0.03 + 0.07 * T / t_start)
        
        c_try = exact_score(h_try)
        
        # Metropolis acceptance
        delta = c_try - curr_c
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            h = h_try
            curr_c = c_try
            if curr_c < best_c:
                best_c = curr_c
                best_h = h.copy()
    
    return best_h, best_c


def slsqp_refine(h0, maxiter=300, tol=1e-12):
    n = len(h0)
    dx = 2.0 / n
    M = 2 * n - 1
    
    def obj(x):
        return x[-1]
    
    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    
    def eq_fun(x):
        return x[:-1].sum() - n / 2.0
    
    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g
    
    def ineq_fun(x):
        h = x[:-1]
        return x[-1] - compute_profile(h)
    
    def ineq_jac(x):
        h = x[:-1]
        G = np.zeros((M, n))
        center = n - 1
        for a in range(M):
            if a < n:
                d = a
                if d == 0:
                    G[a, :] = dx * (1.0 - 2.0 * h)
                else:
                    for j in range(d, n):
                        G[a, j] += dx * (1.0 - h[j - d])
                    for j in range(0, n - d):
                        G[a, j] += -dx * h[j + d]
            else:
                d = a - (n - 1)
                for j in range(0, n - d):
                    G[a, j] += dx * (1.0 - h[j + d])
                for j in range(d, n):
                    G[a, j] += -dx * h[j - d]
        J = np.zeros((M, n + 1))
        J[:, :-1] = -G
        J[:, -1] = 1.0
        return J
    
    t0 = exact_score(h0) + 1e-4
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, t0 + 2.0)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': tol})
        if res.x is not None and np.isfinite(res.x).all():
            hs = repair_feasible(res.x[:-1])
            cs = exact_score(hs)
            if np.isfinite(cs):

    except Exception:
        pass
    return None

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))


def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx


def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        remaining = diff
        for idx in order:
            add = min(remaining, 1.0 - h[idx])
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for idx in order:
            rem = min(remaining, h[idx])
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h


def interp_h(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    fx = np.concatenate([[0.0], xs, [2.0]])
    fh = np.concatenate([[h[0]], h, [h[-1]]])
    hn = np.interp(xt, fx, fh)
    return repair_feasible(np.clip(hn, 0, 1))


def build_gradient_matrix(h, n, dx, indices):
    M = len(indices)
    G = np.zeros((M, n))
    center = n - 1
    for ai, a in enumerate(indices):
        d = a - center
        if d >= 0:
            for j in range(d, n):
                G[ai, j] += dx * (1.0 - h[j - d])
            for j in range(0, n - d):
                G[ai, j] += -dx * h[j + d]
        else:
            dd = -d
            for j in range(0, n - dd):
                G[ai, j] += dx * (1.0 - h[j + dd])
            for j in range(dd, n):
                G[ai, j] += -dx * h[j - dd]
    return G

def pair_swap(h, rng):
    n = len(h)
    i, j = rng.choice(n, size=2, replace=False)
    h_new = h.copy()
    h_new[i], h_new[j] = h[j], h[i]
    return h_new


def block_swap(h, rng, max_block):
    n = len(h)
    b = rng.randint(1, max(max_block, 2))
    b = min(b, n // 3)
    i = rng.randint(0, n - b)
    j = rng.randint(0, n - b)
    if abs(i - j) <= b:
        return h.copy()
    h_new = h.copy()
    h_new[i:i+b], h_new[j:j+b] = h[j:j+b].copy(), h[i:i+b].copy()
    return h_new


def block_perturb(h, rng, max_block, mag):
    n = len(h)
    b = rng.randint(1, max(max_block, 2))
    b = min(b, n // 4)
    i = rng.randint(0, n - b)
    h_new = h.copy()
    h_new[i:i+b] += mag * (rng.rand(b) - 0.5)
    return repair_feasible(h_new)


def small_perturb(h, rng, mag):
    n = len(h)
    k = rng.randint(1, min(5, n // 5) + 1)
    idx = rng.choice(n, size=k, replace=False)
    h_new = h.copy()
    h_new[idx] += mag * (rng.rand(k) - 0.5)
    return repair_feasible(h_new)


def simulated_annealing(h0, n_steps, t_start, t_end, budget_fn, rng, max_block=8):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c = exact_score(h)
    curr_c = best_c
    max_block_eff = min(max_block, n // 4)

    for step in range(n_steps):
        if budget_fn() < 25:
            break
        progress = step / max(1, n_steps - 1)
        T = t_start * (t_end / t_start) ** progress

        r = rng.random()
        if r < 0.25:
            h_try = pair_swap(h, rng)
        elif r < 0.5:
            h_try = block_swap(h, rng, max(max_block_eff, 2))
        elif r < 0.75:
            h_try = block_perturb(h, rng, max(max_block_eff, 2), 0.04 + 0.08 * T / t_start)
        else:
            h_try = small_perturb(h, rng, 0.02 + 0.06 * T / t_start)

        c_try = exact_score(h_try)
        delta = c_try - curr_c
        if delta < 0 or (T > 1e-12 and rng.random() < np.exp(-delta / T)):
            h = h_try
            curr_c = c_try
            if curr_c < best_c:
                best_c = curr_c
                best_h = h.copy()
    return best_h, best_c


def slsqp_refine(h0, maxiter=300, tol=1e-12):

        J[:, :-1] = -G

        return J

    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, t0 + 2.0)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,

def gradient_descent_step(h, profile, active, n, dx):
    G = build_gradient_matrix(h, n, dx, active)
    w = profile[active] - profile[active].min() + 1e-6
    w = w / w.sum()
    grad = G.T @ w
    grad -= grad.mean()
    gn = np.linalg.norm(grad)
    if gn < 1e-15:
        return None, None
    grad /= gn
    curr_c = float(np.max(profile))
    for s in [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]:
        h_try = repair_feasible(h - s * grad)
        sc = exact_score(h_try)
        if sc < curr_c - 1e-14:
            return h_try, sc
    return None, None

                bh, bc = active_set_refine(hp, max_iters=50, budget_fn=tl)
                if bc < best_c - 1e-14:
                    best_h, best_c = bh, bc

    # Phase 9: Final SA with very low temperature
    if tl() > 50:
        n_sa = int(min(200, max(50, tl() * 0.3)))
        bh, bc = simulated_annealing(best_h, n_sa, 0.003, 0.0001, tl, rng,
                                     max_block=2)
        if bc < best_c - 1e-14:
            best_h, best_c = bh, bc

    # Phase 10: Final SLSQP + active-set
    if tl() > 40:
        hs = slsqp_refine(best_h, maxiter=400, tol=1e-13)
        if hs is not None:
            cs = exact_score(hs)
            if cs < best_c - 1e-14:
                best_h, best_c = hs, cs
    if tl() > 20:
        bh, bc = active_set_refine(best_h, max_iters=80, budget_fn=tl)
        if bc < best_c - 1e-14:
            best_h, best_c = bh, bc

    # Final validation
    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)

    try:
        er = evaluate_erdos_solution(best_h)
        if isinstance(er, (float, int)):
            final_c = min(final_c, float(er))
        elif isinstance(er, tuple) and len(er) >= 2:
            final_c = min(final_c, float(er[0]))
    except Exception:
        pass

    return best_h, final_c, len(best_h)
