Let me design the algorithm:

**Stage 1: LSE-homotopy descent on parent**
- Start with `initial_h_values`
- For beta in an ascending schedule (e.g., 10, 20, 50, 100, 200, 500):
  - Compute LSE objective and its gradient
  - Use projected gradient descent / momentum to minimize
  - This pushes down all near-active lags simultaneously

**Stage 2: Trust-region SLP**
- At each iteration:
  - Compute full gradient matrix G (M rows, n cols)
  - Solve LP: minimize t s.t. G @ d - t*ones <= -p, sum(d)=0, bounds on d
  - Backtrack and accept if exact score improves
  - Adjust trust radius

**Stage 3: Epigraph SLSQP**
- Standard scipy SLSQP with epigraph formulation
- Good for final polishing

**Stage 4: Multi-start exploration**
- Try different dimensions
- Try different initial patterns
- Use LSE to smooth and refine

Let me think about the LSE gradient more carefully:

The LSE objective is:
s_beta(h) = (1/beta) * log(sum_m exp(beta * p_m(h)))

The gradient is:
∇s_beta(h) = sum_m w_m * ∇p_m(h)

where w_m = exp(beta * (p_m - c)) / sum_n exp(beta * (p_n - c)), c = max(p)

This is a weighted average of the gradient rows, with weights concentrated on the active/near-active rows.

For the trust-region SLP, I need to solve:
minimize t
subject to: p_m + G[m,:] @ d <= t, for all m
           sum(d) = 0
           -R <= d_i <= R (or tighter bounds based on current h)

This is a linear program with M+1 equality/equality constraints and 2n bound constraints. For n=70, M=139, this is a moderate LP that scipy's linprog should handle quickly.

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(-(1.0 - h))
        remaining = diff
        for i in range(n):
            idx = order[i]
            cap = 1.0 - h[idx]
            if cap <= 1e-15:
                continue
            add = min(remaining, cap)
            h[idx] += add
            remaining -= add
            if remaining <= 1e-14:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            if h[idx] <= 1e-15:
                continue
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem

def exact_score(h):

def compute_profile(h):

def grad_row(h, m, n):
    dx = 2.0 / n
    d = m - (n - 1)
    if d == 0:
        return dx * (1.0 - 2.0 * h)
    elif d > 0:
        g = np.zeros(n)
        g[d:] += dx * (1.0 - h[:n-d])
        g[:n-d] -= dx * h[d:]
        return g
    else:
        p = -d
        g = np.zeros(n)
        g[:n-p] += dx * (1.0 - h[p:])
        g[p:] -= dx * h[:n-p]
        return g

def compute_full_gradient(h):

    M = 2 * n - 1
    G = np.zeros((M, n))
    for m in range(M):
        G[m] = grad_row(h, m, n)
    return G

def lse_objective_and_grad(h, beta):

    z = beta * (p - c)
    z = z - z.max()  # numerical stability

    g = np.zeros(n)

        g += weights[m] * grad_row(h, m, n)
    val = c + np.log(s) / beta
    return val, g

def trust_region_slp(h, max_iters=100, init_radius=0.05, radius_min=0.001):
    n = len(h)
    M = 2 * n - 1
    h = h.copy()
    best_h = h.copy()
    best_c = exact_score(h)
    radius = init_radius
    
    for it in range(max_iters):
        p = compute_profile(h)
        G = compute_full_gradient(h)
        
        # LP: minimize t
        # s.t. p_m + G[m,:] @ d <= t  =>  G[m,:] @ d - t <= -p_m
        #      sum(d) = 0
        #      lower_i <= d_i <= upper_i
        #
        # Variables: [d_0, ..., d_{n-1}, t]  (total n+1)
        # Objective: minimize t => c_obj = [0, 0, ..., 0, 1]
        
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        
        # Inequality constraints: G @ d - t <= -p
        # A_ub shape: (M, n+1)
        A_ub = np.zeros((M, n + 1))
        A_ub[:, :n] = G
        A_ub[:, -1] = -1.0
        b_ub = -p
        
        # Equality: sum(d) = 0
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = [0.0]

        lower = np.maximum(-radius, -h)
        upper = np.minimum(radius, 1.0 - h)
        bounds = list(zip(lower, upper))
        bounds.append((0.0, best_c + 0.01))  # t bounds
        
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs')
            if res.status == 0 and res.x is not None:
                d = res.x[:n]
                # Backtrack
                best_alpha = 0.0
                best_val = best_c

def slsqp_epigraph(h_start, maxiter=200):

    dx = 2.0 / n
    M = 2 * n - 1
    c0 = exact_score(h_start)
    t0 = c0 + 1e-4
    
    def obj(x):
        return x[-1]
    
    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    
    def eq_fun(x):
        return x[:-1].sum() - n / 2.0
    
    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g
    
    def ineq_fun(x):
        h = x[:-1]
        t = x[-1]
        profile = np.correlate(h, 1 - h, mode="full") * dx
        return t - profile
    
    def ineq_jac(x):
        h = x[:-1]
        J = np.zeros((M, n + 1))
        for m in range(M):
            J[m, :n] = -grad_row(h, m, n)
        J[:, -1] = 1.0
        return J
    
    x0 = np.concatenate([h_start, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 2.0)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None and np.isfinite(res.x).all():
            h_new = repair_feasible(res.x[:-1])
            c_new = exact_score(h_new)
            if c_new <= c0 + 1e-8:
                return h_new
    except Exception:
        pass
    return None

def interp_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    edges_old = np.linspace(0.0, 2.0, n + 1)
    edges_new = np.linspace(0.0, 2.0, new_n + 1)
    h_new = np.zeros(new_n)
    for k in range(new_n):
        a, b = edges_new[k], edges_new[k + 1]
        total = 0.0
        for j in range(n):
            lo = max(a, edges_old[j])
            hi = min(b, edges_old[j + 1])
            if hi > lo:
                total += h[j] * (hi - lo)
        h_new[k] = total / (b - a)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

def binary_seed(n, pattern_type, rng):
    half = n // 2
    h = np.zeros(n)
    if pattern_type == 0:
        idx = np.arange(n)
        vals = (idx * 0.618033988749895) % 1.0
        top = np.argsort(vals)[-half:]
        h[top] = 1.0
    elif pattern_type == 1:
        idx = np.arange(n)
        vals = (idx * 0.4142135623730951) % 1.0
        top = np.argsort(vals)[-half:]
        h[top] = 1.0
    elif pattern_type == 2:
        period = max(3, n // 8)
        for i in range(n):
            if (i // period) % 2 == 0:
                h[i] = 1.0
        ones = int(h.sum())
        if ones > half:
            excess = ones - half
            one_idx = np.where(h == 1.0)[0]
            to_flip = rng.choice(one_idx, excess, replace=False)
            h[to_flip] = 0.0
        elif ones < half:
            deficit = half - ones
            zero_idx = np.where(h == 0.0)[0]
            to_flip = rng.choice(zero_idx, deficit, replace=False)
            h[to_flip] = 1.0
    else:
        perm = rng.permutation(n)
        h[perm[:half]] = 1.0
    return repair_feasible(h)

import numpy as np
from scipy.optimize import minimize, linprog
import time


def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(-(1.0 - h))
        remaining = diff
        for i in range(n):
            idx = order[i]
            cap = 1.0 - h[idx]
            if cap <= 1e-15:
                continue
            add = min(remaining, cap)
            h[idx] += add
            remaining -= add
            if remaining <= 1e-14:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            if h[idx] <= 1e-15:
                continue
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-14:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] = float(np.clip(h[idx] + diff2, 0.0, 1.0))
    return h


def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))


def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx


def grad_row(h, m, n):
    dx = 2.0 / n
    d = m - (n - 1)
    if d == 0:
        return dx * (1.0 - 2.0 * h)
    elif d > 0:
        g = np.zeros(n)
        g[d:] += dx * (1.0 - h[:n - d])
        g[:n - d] -= dx * h[d:]
        return g
    else:
        p = -d
        g = np.zeros(n)
        g[:n - p] += dx * (1.0 - h[p:])
        g[p:] -= dx * h[:n - p]
        return g


def compute_full_gradient(h):
    n = len(h)
    M = 2 * n - 1
    G = np.zeros((M, n))
    for m in range(M):
        G[m] = grad_row(h, m, n)
    return G


def lse_grad(h, beta):
    p = compute_profile(h)
    n = len(h)
    M = 2 * n - 1
    c = p.max()
    z = beta * (p - c)
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    s = exp_z.sum()
    if s < 1e-300:
        return np.zeros(n)
    weights = exp_z / s
    g = np.zeros(n)
    for m in range(M):
        if weights[m] > 1e-12:
            g += weights[m] * grad_row(h, m, n)
    g -= g.mean()
    return g


def lse_descent(h, betas, max_iters_per_beta=30):
    n = len(h)
    h_work = h.copy()
    best_h = h.copy()
    best_c = exact_score(h)
    momentum_vec = np.zeros(n)
    for beta in betas:
        for it in range(max_iters_per_beta):
            g = lse_grad(h_work, beta)
            gnorm = np.linalg.norm(g)
            if gnorm < 1e-14:
                break
            momentum_vec = 0.5 * momentum_vec + g
            mvn = np.linalg.norm(momentum_vec)
            if mvn < 1e-14:
                break
            d = -momentum_vec / mvn
            pos_mask = d > 1e-12
            neg_mask = d < -1e-12
            alpha_max = 0.3
            if pos_mask.any():
                alpha_max = min(alpha_max, float(((1.0 - h_work[pos_mask]) / d[pos_mask]).min()))
            if neg_mask.any():
                alpha_max = min(alpha_max, float((h_work[neg_mask] / (-d[neg_mask])).min()))
            alpha_max = max(alpha_max, 1e-10)
            best_alpha = 0.0
            best_val = exact_score(h_work)
            best_h_local = h_work.copy()
            alpha = alpha_max
            for ls in range(12):
                h_try = repair_feasible(h_work + alpha * d)
                c_try = exact_score(h_try)
                if c_try < best_val:
                    best_val = c_try
                    best_alpha = alpha
                    best_h_local = h_try
                alpha *= 0.6
                if alpha < 1e-9:
                    break
            if best_alpha <= 0:
                break
            h_work = best_h_local
            if best_val < best_c:
                best_c = best_val
                best_h = h_work.copy()
    return best_h, best_c


def trust_region_slp(h, max_iters=100, init_radius=0.04, radius_min=0.001):
    n = len(h)
    M = 2 * n - 1
    h_work = h.copy()
    best_h = h.copy()
    best_c = exact_score(h)
    radius = init_radius
    for it in range(max_iters):
        p = compute_profile(h_work)
        G = compute_full_gradient(h_work)
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        A_ub = np.zeros((M, n + 1))
        A_ub[:, :n] = G
        A_ub[:, -1] = -1.0
        b_ub = -p
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = [0.0]
        lower = np.maximum(-radius, -h_work)
        upper = np.minimum(radius, 1.0 - h_work)
        bounds = list(zip(lower, upper))
        bounds.append((0.0, best_c + 0.02))
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if res.status == 0 and res.x is not None:
                d = res.x[:n]
                best_alpha = 0.0
                best_val = exact_score(h_work)
                best_h_local = h_work.copy()
                for frac in [1.0, 0.75, 0.5, 0.3, 0.15, 0.075]:
                    h_try = repair_feasible(h_work + frac * d)
                    c_try = exact_score(h_try)
                    if c_try < best_val - 1e-12:
                        best_val = c_try
                        best_alpha = frac
                        best_h_local = h_try
                        break
                    if c_try < best_val:
                        best_val = c_try
                        best_alpha = frac
                        best_h_local = h_try
                if best_alpha > 0 and best_val < best_c - 1e-12:
                    h_work = best_h_local
                    best_c = best_val
                    best_h = h_work.copy()
                    radius = min(radius * 1.4, 0.12)
                elif best_alpha > 0:
                    h_work = best_h_local
                    radius = min(radius * 1.2, 0.12)
                else:
                    radius *= 0.5
            else:
                radius *= 0.5
        except Exception:
            radius *= 0.5
        if radius < radius_min:
            break
    return best_h, best_c


def slsqp_epigraph(h_start, maxiter=200):
    n = len(h_start)
    dx = 2.0 / n
    M = 2 * n - 1
    c0 = exact_score(h_start)
    t0 = c0 + 1e-4

    def obj(x):
        return x[-1]

    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g

    def eq_fun(x):
        return x[:-1].sum() - n / 2.0

    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g

    def ineq_fun(x):
        h_arr = x[:-1]
        t = x[-1]
        profile = np.correlate(h_arr, 1 - h_arr, mode="full") * dx
        return t - profile

    def ineq_jac(x):
        h_arr = x[:-1]
        J = np.zeros((M, n + 1))
        for m in range(M):
            J[m, :n] = -grad_row(h_arr, m, n)
        J[:, -1] = 1.0
        return J

    x0 = np.concatenate([h_start, [t0]])
    bnds = [(0.0, 1.0)] * n + [(0.0, c0 + 2.0)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-13})
        if res.x is not None and np.isfinite(res.x).all():
            h_new = repair_feasible(res.x[:-1])
            c_new = exact_score(h_new)
            if c_new <= c0 + 1e-9:
                return h_new
    except Exception:
        pass
    return None


def interp_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    edges_old = np.linspace(0.0, 2.0, n + 1)
    edges_new = np.linspace(0.0, 2.0, new_n + 1)
    h_new = np.zeros(new_n)
    for k in range(new_n):
        a, b = edges_new[k], edges_new[k + 1]
        total = 0.0
        for j in range(n):
            lo = max(a, edges_old[j])
            hi = min(b, edges_old[j + 1])
            if hi > lo:
                total += h[j] * (hi - lo)
        h_new[k] = total / (b - a)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)


def active_descent(h, max_iters=400, active_tol=0.003):
    n = len(h)
    h_work = h.copy()
    best_c = exact_score(h)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    momentum_vec = np.zeros(n)
    momentum = 0.3
    for it in range(max_iters):
        profile = compute_profile(h_work)
        c = float(profile.max())
        thresh = c - active_tol
        active = np.where(profile >= thresh)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(profile))])
        if len(active) > 25:
            top = np.argsort(profile[active])[-25:]
            active = active[top]
        g = np.zeros(n)
        for m in active:
            g += grad_row(h_work, m, n)
        g -= g.mean()
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            break
        momentum_vec = momentum * momentum_vec + g
        mv_norm = np.linalg.norm(momentum_vec)
        if mv_norm < 1e-15:
            break
        d = -momentum_vec / mv_norm
        pos_mask = d > 1e-12
        neg_mask = d < -1e-12
        alpha_max = 0.5
        if pos_mask.any():
            alpha_max = min(alpha_max, float(((1.0 - h_work[pos_mask]) / d[pos_mask]).min()))
        if neg_mask.any():
            alpha_max = min(alpha_max, float((h_work[neg_mask] / (-d[neg_mask])).min()))
        alpha_max = max(alpha_max, 1e-10)
        best_alpha = 0.0
        best_val = c
        best_h = h_work
        a_ls, b_ls = 0.0, alpha_max
        for _ in range(20):
            c1 = b_ls - gr * (b_ls - a_ls)
            c2 = a_ls + gr * (b_ls - a_ls)
            h1 = repair_feasible(h_work + c1 * d)
            h2 = repair_feasible(h_work + c2 * d)
            f1 = exact_score(h1)
            f2 = exact_score(h2)
            if f1 < best_val:
                best_val = f1
                best_alpha = c1
                best_h = h1.copy()
            if f2 < best_val:
                best_val = f2
                best_alpha = c2
                best_h = h2.copy()
            if f1 < f2:
                b_ls = c2
            else:
                a_ls = c1
            if b_ls - a_ls < 1e-13:
                break
        if best_val >= c - 1e-15:
            break
        h_work = best_h
        if best_val < best_c:
            best_c = best_val
    return h_work, best_c


def binary_seed(n, pattern_type, rng):
    half = n // 2
    h = np.zeros(n)
    if pattern_type == 0:
        idx = np.arange(n)
        vals = (idx * 0.618033988749895) % 1.0
        top = np.argsort(vals)[-half:]
        h[top] = 1.0
    elif pattern_type == 1:
        idx = np.arange(n)
        vals = (idx * 0.4142135623730951) % 1.0
        top = np.argsort(vals)[-half:]
        h[top] = 1.0
    elif pattern_type == 2:
        period = max(3, n // 8)
        for i in range(n):
            if (i // period) % 2 == 0:
                h[i] = 1.0
        ones = int(h.sum())
        if ones > half:
            excess = ones - half
            one_idx = np.where(h == 1.0)[0]
            to_flip = rng.choice(one_idx, size=excess, replace=False)
            h[to_flip] = 0.0
        elif ones < half:
            deficit = half - ones
            zero_idx = np.where(h == 0.0)[0]
            to_flip = rng.choice(zero_idx, size=deficit, replace=False)
            h[to_flip] = 1.0
    else:
        perm = rng.permutation(n)
        h[perm[:half]] = 1.0
    return repair_feasible(h)
