import numpy as np
from scipy.optimize import minimize

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(-(1.0 - h))
        remaining = diff
        for i in range(n):
            idx = order[i]
            cap = 1.0 - h[idx]
            if cap <= 1e-15:
                continue
            add = min(remaining, cap)
            h[idx] += add
            remaining -= add
            if remaining <= 1e-14:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            if h[idx] <= 1e-15:
                continue
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-14:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] = float(np.clip(h[idx] + diff2, 0.0, 1.0))
    return h

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def smooth_max(profile, alpha):
    """Log-sum-exp approximation of max."""
    m = profile.max()
    if m < -1e300:
        return 0.0
    shifted = profile - m
    return m + np.log(np.sum(np.exp(alpha * shifted))) / alpha

def smooth_max_grad(profile, alpha):
    """Gradient of log-sum-exp max with respect to profile."""
    m = profile.max()
    shifted = alpha * (profile - m)
    weights = np.exp(shifted)
    weights /= weights.sum()
    return weights

def grad_overlap(h, m, n):
    """Gradient of overlap at lag m with respect to h."""
    dx = 2.0 / n
    d = m - (n - 1)
    if d == 0:
        return dx * (1.0 - 2.0 * h)
    elif d > 0:
        g = np.zeros(n)
        g[d:] += dx * (1.0 - h[:-d])
        g[:-d] -= dx * h[d:]
        return g
    else:
        p = -d
        g = np.zeros(n)
        g[:-p] += dx * (1.0 - h[p:])
        g[p:] -= dx * h[:-p]
        return g

def smooth_obj(h, alpha):
    """Smooth approximation of max overlap."""
    profile = compute_profile(h)
    return smooth_max(profile, alpha)

def smooth_obj_grad(h, alpha):
    """Gradient of smooth max overlap objective."""
    n = len(h)
    profile = compute_profile(h)
    weights = smooth_max_grad(profile, alpha)
    g = np.zeros(n)
    for m in range(len(profile)):
        if weights[m] > 1e-12:
            g += weights[m] * grad_overlap(h, m, n)
    return g

def lsqs_smooth(h, alpha, maxiter=200):
    """Optimize smooth max overlap using L-BFGS-B with projection."""
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    
    # Use a parameterization that handles the constraint
    # We'll work with free variables and project
    
    def obj_and_grad(x):
        h_try = repair_feasible(x)
        val = smooth_obj(h_try, alpha)
        grad = smooth_obj_grad(h_try, alpha)
        # Remove mean to respect the equality constraint
        grad = grad - grad.mean()
        return val, grad
    
    x0 = h.copy()
    bounds = [(0.0, 1.0)] * n
    
    res = minimize(obj_and_grad, x0, jac=True, method='L-BFGS-B',
                   bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10})
    
    h_new = repair_feasible(res.x)
    c_new = exact_score(h_new)
    if c_new < best_c:
        return h_new, c_new
    return h, best_c

def interp_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    edges_old = np.linspace(0.0, 2.0, n + 1)
    edges_new = np.linspace(0.0, 2.0, new_n + 1)
    h_new = np.zeros(new_n)
    for k in range(new_n):
        a, b = edges_new[k], edges_new[k + 1]
        total = 0.0
        for j in range(n):
            lo = max(a, edges_old[j])
            hi = min(b, edges_old[j + 1])
            if hi > lo:
                total += h[j] * (hi - lo)
        h_new[k] = total / (b - a)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

import numpy as np
from scipy.optimize import minimize
import time

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(-(1.0 - h))
        remaining = diff
        for i in range(n):
            idx = order[i]
            cap = 1.0 - h[idx]
            if cap <= 1e-15:
                continue
            add = min(remaining, cap)
            h[idx] += add
            remaining -= add
            if remaining <= 1e-14:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            if h[idx] <= 1e-15:
                continue
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-14:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] = float(np.clip(h[idx] + diff2, 0.0, 1.0))
    return h

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def grad_overlap(h, m, n):
    dx = 2.0 / n
    d = m - (n - 1)
    if d == 0:
        return dx * (1.0 - 2.0 * h)
    elif d > 0:
        g = np.zeros(n)
        g[d:] += dx * (1.0 - h[:-d])
        g[:-d] -= dx * h[d:]
        return g
    else:
        p = -d
        g = np.zeros(n)
        g[:-p] += dx * (1.0 - h[p:])
        g[p:] -= dx * h[:-p]
        return g

def smooth_max_val(profile, alpha):
    m = profile.max()
    shifted = alpha * (profile - m)
    return m + np.log(np.sum(np.exp(shifted))) / alpha

def smooth_max_weights(profile, alpha):
    m = profile.max()
    shifted = alpha * (profile - m)
    w = np.exp(shifted)
    w /= w.sum()
    return w

def smooth_grad(h, alpha):
    n = len(h)
    profile = compute_profile(h)
    w = smooth_max_weights(profile, alpha)
    g = np.zeros(n)
    M = len(profile)
    for m in range(M):
        wm = w[m]
        if wm > 1e-10:
            g += wm * grad_overlap(h, m, n)
    return g - g.mean()

def lsqs_smooth(h, alpha, maxiter=150):
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)

    def obj_grad(x):
        h_try = repair_feasible(x)
        val = smooth_max_val(compute_profile(h_try), alpha)
        grad = smooth_grad(h_try, alpha)
        return val, grad

    try:
        res = minimize(obj_grad, h, jac=True, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-12})
        h_new = repair_feasible(res.x)
        c_new = exact_score(h_new)
        if c_new < best_c:
            return h_new, c_new
    except Exception:
        pass
    return h, best_c

def active_descent(h, max_iters=300, active_tol=0.004, top_k=12, momentum_rate=0.25):
    n = len(h)
    h = h.copy()

        aw /= aw.sum()

        g -= g.mean()
        gn = np.linalg.norm(g)
        if gn < 1e-15:

def interp_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()

        for j in range(n):

            hi = min(b, eo[j + 1])
            if hi > lo:
                total += h[j] * (hi - lo)
        h_new[k] = total / (b - a)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)
