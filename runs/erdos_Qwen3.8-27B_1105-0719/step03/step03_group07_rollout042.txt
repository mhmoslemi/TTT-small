    target = n / 2.0

    if abs(diff) < 1e-14:

        order = np.argsort(-capacity)

            add = min(remaining, capacity[idx])

    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx])
            h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    return repair_feasible(np.clip(h_new, 0.0, 1.0))

def pair_sweep(h, max_sweeps=20, t_start=None, budget_s=1100, rng=None):
    """Full sweep of pairwise minimization."""
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    
    for sweep in range(max_sweeps):
        if t_start is not None and time.perf_counter() - t_start > budget_s - 30:
            break
        
        improved = False
        # Process pairs in a shuffled order for diversity
        indices = np.arange(n)
        rng.shuffle(indices)
        
        sweep_improvement = 0.0
        for ii in range(n):
            i = indices[ii]
            for jj in range(ii + 1, n):
                j = indices[jj]
                # Skipped if both are near 0 or near 1 (can't move)
                s = h[i] + h[j]
                t_lo = max(0.0, s - 1.0)
                t_hi = min(1.0, s)
                if t_hi - t_lo < 1e-12:
                    continue
                
                h_new, c_new = pair_minimize(h, i, j)
                if c_new < best_c - 1e-13:
                    h = h_new
                    best_c = c_new
                    improved = True
                    sweep_improvement += 1
        
        if not improved or sweep_improvement == 0:
            break
    
    return h, best_c

def pair_minimize(h, i, j):
    """Minimize max profile over h[i]=t, h[j]=s-t."""
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    s = h[i] + h[j]
    t_lo = max(0.0, s - 1.0)
    t_hi = min(1.0, s)
    
    if t_hi - t_lo < 1e-12:
        return h, exact_score(h)
    
    P = np.correlate(h, 1 - h, mode="full") * dx
    
    # Compute A and B coefficient vectors
    A = np.zeros(M)
    B = np.zeros(M)
    
    for k in range(M):
        d = k - (n - 1)
        if 0 <= i + d < n:
            A[k] += 1.0 - h[i + d]
        if 0 <= i - d < n:
            A[k] -= h[i - d]
        if 0 <= j + d < n:
            B[k] += 1.0 - h[j + d]
        if 0 <= j - d < n:
            B[k] -= h[j - d]
    
    C = (s - h[j]) * B - h[i] * A
    AB = A - B
    
    # Ternary search
    a, b = t_lo, t_hi
    for _ in range(40):
        m1 = a + (b - a) / 3.0
        m2 = b - (b - a) / 3.0
        f1 = np.max(P + dx * (m1 * AB + C))
        f2 = np.max(P + dx * (m2 * AB + C))
        if f1 <= f2:
            b = m2
        else:
            a = m1
    
    t_opt = (a + b) / 2.0
    t_opt = np.clip(t_opt, t_lo, t_hi)
    
    new_score = np.max(P + dx * (t_opt * AB + C))
    current_score = P.max()
    
    if new_score < current_score - 1e-14:
        h[i] = np.clip(t_opt, 0.0, 1.0)
        h[j] = np.clip(s - t_opt, 0.0, 1.0)
        return h, new_score
    
    return h, current_score

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0

            h[idx] += add

        if fc < fd:
            hi = d; d = c; fd = fc
            c = hi - gr * (hi - lo); fc = f(c)
        else:
            lo = c; c = d; fc = fd
            d = lo + gr * (hi - lo); fd = f(d)
    
    t_opt = (lo + hi) / 2.0
    final_val = f(t_opt)
    
    if final_val < current_score - 1e-14:
        h[i] = np.clip(t_opt, 0.0, 1.0)
        h[j] = np.clip(s - t_opt, 0.0, 1.0)
        return h, final_val
    else:
        h[i] = hi_save
        h[j] = hj_save
        return h, current_score


def pair_sweep(h, rng, max_sweeps=15, t_start=None, budget_s=1100):
    """Multiple sweeps of pairwise minimization."""
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    
    for sweep in range(max_sweeps):
        if t_start is not None and time.perf_counter() - t_start > budget_s - 40:
            break
        
        # Shuffle pair order
        perm = rng.permutation(n)
        n_improvements = 0
        
        for a_idx in range(n):
            i = perm[a_idx]
            for b_idx in range(a_idx + 1, n):
                j = perm[b_idx]
                
                s = h[i] + h[j]
                t_lo = max(0.0, s - 1.0)
                t_hi = min(1.0, s)
                if t_hi - t_lo < 1e-12:
                    continue
                
                h, c = pair_minimize(h, i, j)
                if c < best_c - 1e-13:
                    best_c = c
                    n_improvements += 1
        
        if n_improvements == 0:
            break
    
    return h, best_c

def softmax_objective(h, beta):
    n = len(h)
    dx = 2.0 / n
    P = np.correlate(h, 1 - h, mode="full") * dx
    P_max = P.max()
    # Shift for numerical stability
    shifted = P - P_max
    log_sum = np.log(np.sum(np.exp(beta * shifted))) / beta
    return P_max + log_sum
