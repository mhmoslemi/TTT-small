import time
import cvxpy as cp

    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full"))) * dx

def profile_h(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def repair_feasible(vec):

    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)

            add = min(remaining, capacity[idx])
            h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem

    if abs(diff2) > 1e-15:

        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    return repair_feasible(np.clip(h_new, 0.0, 1.0))

def slp_optimize(h0, max_iters=500, step_max=0.05, tol=1e-10, 
                 active_frac=0.3, t_start=None, budget_s=1100):
    """SLP minimax optimization with active set selection."""
    n = len(h0)
    h = h0.copy()
    best_c = exact_score(h)
    M = 2 * n - 1
    lags = np.arange(-(n-1), n)  # lag values
    
    for it in range(max_iters):
        if t_start is not None:
            if time.perf_counter() - t_start > budget_s - 10:
                break
        
        P = profile_h(h)
        c_max = P.max()
        
        # Select active constraints
        thresh = c_max - active_frac * (c_max - P.min())
        active_idx = np.where(P >= thresh)[0]
        if len(active_idx) < 5:
            active_idx = np.argsort(P)[-max(5, int(0.1*M)):]
        active_lags = lags[active_idx]
        active_P = P[active_idx]
        
        # Compute gradients for active lags
        G = np.zeros((len(active_lags), n))
        dx = 2.0 / n
        for idx, k in enumerate(active_lags):
            if k == 0:
                G[idx, :] = dx * (1.0 - 2.0 * h)
            elif k > 0:
                for i in range(n - k):
                    G[idx, i + k] += dx * (1.0 - h[i])
                    G[idx, i] -= dx * h[i + k]
            else:
                p = -k
                for i in range(n - p):
                    G[idx, i] += dx * (1.0 - h[i + p])
                    G[idx, i + p] -= dx * h[i]

        d = cp.Variable(n)
        t = cp.Variable()
        constraints = [
            G @ d + active_P <= t,
            d >= -step_max,
            d <= step_max,
            cp.sum(d) == 0,
            h + d >= 0,
            h + d <= 1
        ]
        obj = cp.Minimize(t)
        prob = cp.Problem(obj, constraints)
        
        try:
            prob.solve(solver=cp.CLARABEL)
        except Exception:
            try:
                prob.solve(solver=cp.SCS)
            except Exception:
                break
        
        if d.value is None or not np.all(np.isfinite(d.value)):
            break
        
        d_val = d.value
        # Check if LP predicts improvement
        if t.value >= best_c - 1e-12:
            # Try smaller step
            d_val = d_val * 0.5
            if t.value * 0.5 + min(active_P) >= best_c:
                break

        improved = False
        for alpha in [1.0, 0.7, 0.5, 0.35, 0.25, 0.15, 0.1]:

            h_new = repair_feasible(h_new)
            c_new = exact_score(h_new)
            if c_new < best_c - 1e-14:
                h = h_new
                best_c = c_new
                improved = True
                break

        prev_c = best_c

    return h, best_c

def pair_exchange_polish(h0, max_sweeps=5, t_start=None, budget_s=1100):
    """Polish using pair-exchange moves that preserve the sum constraint."""
    n = len(h0)
    h = h0.copy()
    best_c = exact_score(h)
    
    # Precompute which pairs are "interior" (both values in (0,1))
    for sweep in range(max_sweeps):
        if t_start is not None:
            if time.perf_counter() - t_start > budget_s - 10:
                break
        
        improved_any = False
        # Sample a random subset of pairs for speed
        interior = np.where((h > 0.01) & (h < 0.99))[0]
        if len(interior) < 2:
            break
        
        # Try pairs in a fixed order (use i stride for diversity)
        rng = np.random.RandomState(42 + sweep)
        pairs = []
        for _ in range(min(500, len(interior) * (len(interior) - 1) // 2)):
            i, j = rng.choice(len(interior), 2, replace=False)
            pairs.append((interior[i], interior[j]))
        
        for (i, j) in pairs:
            if t_start is not None and time.perf_counter() - t_start > budget_s - 10:
                break
            h_i = h[i]
            h_j = h[j]
            # Bounds for delta
            delta_min = max(-h_i, -(1.0 - h_j), -0.1)
            delta_max = min(1.0 - h_i, h_j, 0.1)
            if delta_max - delta_min < 1e-6:
                continue
            
            # Try a few delta values
            best_delta = 0.0
            best_local = best_c
            deltas = np.linspace(delta_min, delta_max, 9)
            for delta in deltas:
                if abs(delta) < 1e-8:
                    continue
                h[i] = h_i + delta
                h[j] = h_j - delta
                c_new = exact_score(h)
                if c_new < best_local - 1e-15:
                    best_local = c_new
                    best_delta = delta
                h[i] = h_i
                h[j] = h_j
            
            if best_delta != 0.0:
                h[i] = h_i + best_delta
                h[j] = h_j - best_delta
                h = repair_feasible(h)
                best_c = exact_score(h)
                improved_any = True
        
        if not improved_any:
            break
    
    return h, best_c

def symmetrize(h):
    return (h + h[::-1].copy()) / 2.0

def make_diverse_starts(h_base, n, rng, n_starts=5):
    """Create diverse starting points."""
    starts = [h_base.copy()]
    
    # Symmetric version
    h_sym = repair_feasible(symmetrize(h_base))
    starts.append(h_sym)
    
    # Perturbed versions
    for eps in [0.01, 0.03, 0.05]:
        z = rng.randn(n)
        z -= z.mean()
        starts.append(repair_feasible(h_base + eps * z))
    
    # Symmetric perturbation
    z = rng.randn(n)
    z_sym = symmetrize(z)
    starts.append(repair_feasible(symmetrize(h_base + 0.02 * z_sym)))
    
    return starts[:n_starts]
