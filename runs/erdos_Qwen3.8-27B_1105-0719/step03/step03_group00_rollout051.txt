def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -3.0, 3.0
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        nm = mask.sum()
        if nm > 0:
            h[mask] += diff / nm
            h = np.clip(h, 0.0, 1.0)
            # Fix any remaining error
            diff2 = target_sum - h.sum()
            if abs(diff2) > 1e-12:
                mask2 = (h > 1e-10) & (h < 1.0 - 1e-10)
                if mask2.sum() > 0:
                    h[mask2] += diff2 / mask2.sum()
    return np.clip(h, 0.0, 1.0)

def c5_of_h(h, dx):
    return np.correlate(h, 1.0 - h, mode='full').max() * dx

def corr_of_h(h):
    return np.correlate(h, 1.0 - h, mode='full')

def grad_corr_k(h, k, n):
    off = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -off)
    i_hi = min(n - 1, n - 1 - off)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx] += (1.0 - h[idx + off])
    grad[idx + off] += (-h[idx])
    return grad

def solve_slack_lp(h, dx, n, active_idx, slack_penalty, q_weight, max_trust):
    """Solve the soft-slack LP for a descent direction."""
    corr = corr_of_h(h)
    K = len(active_idx)
    # Variables: [delta (n), slacks (K), q (1)]
    m_vars = n + K + 1
    
    c_obj = np.zeros(m_vars)
    c_obj[n:n+K] = slack_penalty
    c_obj[-1] = q_weight
    
    # Constraints: dx*grad_k·delta + s_k - q <= -corr[k_k] for each active k
    A_ub = np.zeros((K, m_vars))
    b_ub = np.zeros(K)
    for j, k in enumerate(active_idx):
        g_k = grad_corr_k(h, k, n)
        A_ub[j, :n] = dx * g_k
        A_ub[j, n+j] = 1.0
        A_ub[j, -1] = -1.0
        b_ub[j] = -corr[k]
    
    # Equality: sum(delta) = 0
    A_eq = np.zeros((1, m_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds
    bounds = []
    for i in range(n):
        lo = -h[i] * max_trust
        hi = (1.0 - h[i]) * max_trust
        bounds.append((lo, hi))
    for j in range(K):
        bounds.append((0.0, None))  # slacks >= 0
    bounds.append((None, None))  # q free
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs', options={'presolve': True})
        if res.status != 0:
            return None
        return res.x[:n]
    except Exception:
        return None

For a binary pattern, corr_k = number of i where h[i]=1 and h[i+off]=0 (for the valid range). This is related to the autocorrelation of the binary sequence.

Actually, for binary h with exactly n/2 ones:
corr_k = sum_{i: valid} h[i]*(1-h[i+off])

If h is binary with n/2 ones and n/2 zeros:
- When h[i]=1: contributes (1-h[i+off]) = 1 if h[i+off]=0, else 0
- When h[i]=0: contributes 0*(1-h[i+off]) = 0

So corr_k = |{i: h[i]=1, h[i+off]=0, both in range}|

This is the number of 1s in positions that shift to 0 positions.

For the "worst" lag, we want to minimize this count. This is related to the "minimum cross-correlation" of a binary sequence with its complement, which is a known problem in coding theory.

The optimal binary sequences for this problem are related to m-sequences and other low-autocorrelation sequences. But our problem is slightly different because we're looking at cross-correlation with the complement, not autocorrelation.

For a binary sequence of length n with n/2 ones:
corr_k (with offset off) = #{i in valid range: h[i]=1, h[i+off]=0}

Note that #{i: h[i]=1, h[i+off]=0} + #{i: h[i]=0, h[i+off]=1} = #{i: h[i] ≠ h[i+off]} (in the valid range)

import numpy as np
from scipy.optimize import linprog
import time

def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -3.0, 3.0
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        nm = mask.sum()
        if nm > 0:
            h[mask] += diff / nm
    return np.clip(h, 0.0, 1.0)

def c5_of_h(h, dx):
    return np.correlate(h, 1.0 - h, mode='full').max() * dx

def corr_of_h(h):
    return np.correlate(h, 1.0 - h, mode='full')

def grad_corr_k(h, k, n):
    off = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -off)
    i_hi = min(n - 1, n - 1 - off)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx] += (1.0 - h[idx + off])
    grad[idx + off] += (-h[idx])
    return grad

def solve_lp_slack(h, dx, n, active_idx, slack_pen, q_w, trust):
    K = len(active_idx)
    corr = corr_of_h(h)
    m = n + K + 1
    c = np.zeros(m)
    c[n:n+K] = slack_pen
    c[-1] = q_w
    A_ub = np.zeros((K, m))
    b_ub = np.zeros(K)
    for j, k in enumerate(active_idx):
        g = grad_corr_k(h, k, n)
        A_ub[j, :n] = dx * g
        A_ub[j, n+j] = 1.0
        A_ub[j, -1] = -1.0
        b_ub[j] = -corr[k]
    A_eq = np.zeros((1, m))
    A_eq[0, :n] = 1.0
    bounds = [(-h[i]*trust, (1.0-h[i])*trust) for i in range(n)]
    bounds += [(0.0, None)] * K
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.zeros(1),
                     bounds=bounds, method='highs')
        if res.status != 0:
            return None
        d = res.x[:n]
        if np.linalg.norm(d) < 1e-15:
            return None
        return d
    except Exception:
        return None

def solve_lp_hard(h, dx, n, active_idx, trust):
    K = len(active_idx)
    corr = corr_of_h(h)
    m = n + 1
    c = np.zeros(m)
    c[-1] = 1.0
    A_ub = np.zeros((K, m))
    b_ub = np.zeros(K)
    for j, k in enumerate(active_idx):
        g = grad_corr_k(h, k, n)
        A_ub[j, :n] = dx * g
        A_ub[j, -1] = -1.0
        b_ub[j] = -corr[k]
    A_eq = np.zeros((1, m))
    A_eq[0, :n] = 1.0
    bounds = [(-h[i]*trust, (1.0-h[i])*trust) for i in range(n)]
    bounds.append((0.0, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.zeros(1),
                     bounds=bounds, method='highs')
        if res.status != 0:
            return None
        d = res.x[:n]
        if np.linalg.norm(d) < 1e-15:
            return None
        return d
    except Exception:
        return None

def line_search(cur, delta, dx, target_sum, cur_c5, max_t=1.0):
    best_h = None
    best_c5 = cur_c5
    for t in [max_t, max_t*0.75, max_t*0.55, max_t*0.4, max_t*0.3, max_t*0.2, max_t*0.12, max_t*0.06, max_t*0.03]:
        h_c = project_feasible(cur + t * delta, target_sum)
        c5_val = c5_of_h(h_c, dx)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h_c
    return best_h, best_c5

def select_active(corr, n):
    max_c = corr.max()
    n_full = 2 * n - 1
    n_top = max(20, min(60, int(0.30 * n_full)))
    top_idx = np.argsort(corr)[-n_top:]
    margin = max(0.12, 0.05 * max_c)
    am = corr >= (max_c - margin)
    am[top_idx] = True
    active = np.where(am)[0]
    if len(active) < 10:
        active = top_idx
    return active

def lp_phase(h, dx, ts, budget_s, t0, n_rounds=100, slack=True, spen=4.0, qw=0.25):
    n = len(h)
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    cur = h.copy()
    cur_c5 = best_c5
    alpha = 0.25
    stall = 0
    for it in range(n_rounds):
        if budget_s - (time.time() - t0) < 5.0:
            break
        corr = corr_of_h(cur)
        active = select_active(corr, n)
        if slack:
            delta = solve_lp_slack(cur, dx, n, active, spen, qw, alpha)
        else:
            delta = solve_lp_hard(cur, dx, n, active, alpha)
        if delta is None:
            alpha *= 0.5
            stall += 1
            if stall > 15 or alpha < 0.003:
                break
            continue
        h_new, c5_new = line_search(cur, delta, dx, ts, cur_c5)
        if h_new is not None:
            cur = h_new
            cur_c5 = c5_new
            alpha = min(0.35, alpha * 1.04)
            stall = 0
            if cur_c5 < best_c5 - 1e-15:
                best_c5 = cur_c5
                best_h = cur.copy()
        else:
            alpha *= 0.6
            stall += 1
            if stall > 20 or alpha < 0.003:
                break
    return best_h, best_c5

def subgrad_phase(h, dx, ts, budget_s, t0, n_rounds=150):
    n = len(h)
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    cur = h.copy()
    cur_c5 = best_c5
    mom = np.zeros(n)
    beta = 0.85
    step = 0.12
    stall = 0
    for it in range(n_rounds):
        if budget_s - (time.time() - t0) < 5.0:
            break
        corr = corr_of_h(cur)
        max_c = corr.max()
        temp = max(0.004, 0.012 * max_c)
        w = np.exp((corr - max_c) / temp)
        w /= w.sum()
        grad = np.zeros(n)
        sig = np.where(w > 0.004 * w.max())[0]
        for k in sig:
            grad += w[k] * grad_corr_k(cur, k, n)
        mom = beta * mom + (1.0 - beta) * grad
        mom -= mom.mean()
        h_new, c5_new = line_search(cur, -mom, dx, ts, cur_c5, max_t=step)
        if h_new is not None:
            cur = h_new
            cur_c5 = c5_new
            step = min(0.4, step * 1.03)
            stall = 0
            if cur_c5 < best_c5 - 1e-15:
                best_c5 = cur_c5
                best_h = cur.copy()
        else:
            step *= 0.65
            stall += 1
            if stall > 25 or step < 1e-5:
                break
    return best_h, best_c5

def pair_phase(h, dx, ts, rng, budget_s, t0, n_rounds=300):
    n = len(h)
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    cur = h.copy()
    cur_c5 = best_c5
    for it in range(n_rounds):
        if budget_s - (time.time() - t0) < 5.0:
            break
        corr = corr_of_h(cur)
        k_top = np.argsort(corr)[-5:]
        g = np.zeros(n)
        for k in k_top:
            g += grad_corr_k(cur, k, n)
        med = np.median(g)
        pos = np.where(g > med + 0.01)[0]
        neg = np.where(g < med - 0.01)[0]
        if len(pos) > 0 and len(neg) > 0:
            i = pos[rng.randint(len(pos))]
            j = neg[rng.randint(len(neg))]
        else:
            i, j = rng.choice(n, 2, replace=False)
        a, b = cur[i], cur[j]
        t_lo = max(-a, b - 1.0)
        t_hi = min(1.0 - a, b)
        if t_hi - t_lo < 1e-8:
            continue
        tvals = np.linspace(t_lo, t_hi, 13)
        best_t = 0.0
        best_obj = cur_c5
        for t in tvals:
            cur[i] = a + t
            cur[j] = b - t
            obj = c5_of_h(cur, dx)
            if obj < best_obj:
                best_obj = obj
                best_t = t
        if best_t != 0.0:
            dt = (t_hi - t_lo) / 13
            fv = np.linspace(max(t_lo, best_t - 2*dt), min(t_hi, best_t + 2*dt), 9)
            for t in fv:
                cur[i] = a + t
                cur[j] = b - t
                obj = c5_of_h(cur, dx)
                if obj < best_obj:
                    best_obj = obj
                    best_t = t
            cur[i] = a + best_t
            cur[j] = b - best_t
            if best_obj < cur_c5 - 1e-12:
                cur_c5 = best_obj
                if cur_c5 < best_c5 - 1e-14:
                    best_c5 = cur_c5
                    best_h = cur.copy()
        else:
            cur[i] = a
            cur[j] = b
    return best_h, best_c5

def transfer_grid(h_src, n_tgt):
    n_s = len(h_src)
    xs = (np.arange(n_s) + 0.5) * (2.0 / n_s)
    xt = (np.arange(n_tgt) + 0.5) * (2.0 / n_tgt)
    return np.interp(xt, xs, h_src)

def perturb_binary(h, rng, ts):
    n = len(h)
    h_bin = np.where(h > 0.5, 1.0, 0.0)
    s = h_bin.sum()
    if abs(s - ts) < 0.5:
        if s < ts:
            idx = np.where(h_bin == 0.0)[0]
            if len(idx) > 0:
                h_bin[idx[rng.randint(len(idx))]] = 1.0
        elif s > ts:
            idx = np.where(h_bin == 1.0)[0]
            if len(idx) > 0:
                h_bin[idx[rng.randint(len(idx))]] = 0.0
    return project_feasible(h_bin, ts)
