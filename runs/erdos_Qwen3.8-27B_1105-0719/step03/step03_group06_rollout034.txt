def grad_row(h, m, n):
    dx = 2.0 / n
    d = m - (n - 1)
    if d == 0:
        return dx * (1.0 - 2.0 * h)
    elif d > 0:
        g = np.zeros(n)
        g[:n-d] += dx * (1.0 - h[d:])
        g[d:] -= dx * h[:n-d]
        return g
    else:
        p = -d
        g = np.zeros(n)
        g[:n-p] -= dx * h[p:]
        g[p:] += dx * (1.0 - h[:n-p])
        return g

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_candidates = np.where(u * np.arange(1, n+1) > cssv)[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def optimal_weights(G, max_iter=200):
    """Find w >= 0, sum(w)=1 minimizing ||G @ w||^2"""
    k = G.shape[0]
    if k == 1:
        return np.array([1.0])
    H = G.T @ G  # k x k
    w = np.ones(k) / k
    Hw = H @ w
    for it in range(max_iter):
        grad = 2.0 * Hw  # gradient of w' H w
        # Backtracking line search
        step = 1.0 / (1.0 + it * 0.1)
        for _ in range(20):

            nw = H @ w_try
            if w_try @ nw <= w @ Hw:
                break
            step *= 0.5
        w = w_try
        Hw = nw
    return w

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(-(1.0 - h))
        remaining = diff
        for i in range(n):
            idx = order[i]
            cap = 1.0 - h[idx]
            if cap <= 1e-15:
                continue
            add = min(remaining, cap)
            h[idx] += add
            remaining -= add
            if remaining <= 1e-14:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            if h[idx] <= 1e-15:
                continue
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-14:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] = float(np.clip(h[idx] + diff2, 0.0, 1.0))
    return h

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def grad_row(h, m, n):
    dx = 2.0 / n
    d = m - (n - 1)
    if d == 0:
        return dx * (1.0 - 2.0 * h)
    elif d > 0:
        g = np.zeros(n)
        g[:n-d] += dx * (1.0 - h[d:])
        g[d:] -= dx * h[:n-d]
        return g
    else:
        p = -d
        g = np.zeros(n)
        g[:n-p] -= dx * h[p:]
        g[p:] += dx * (1.0 - h[:n-p])
        return g

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_candidates = np.where(u * np.arange(1, n + 1) > cssv)[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def optimal_weights(G):
    k = G.shape[0]
    if k == 1:
        return np.array([1.0])
    H = G.T @ G
    try:
        def qp_obj(w):
            return float(w @ H @ w)
        def qp_grad(w):
            return 2.0 * (H @ w)
        x0 = np.ones(k) / k
        bounds = [(0.0, 1.0)] * k
        cons = [{'type': 'eq', 'fun': lambda w: w.sum() - 1.0,
                 'jac': lambda w: np.ones(k)}]
        res = minimize(qp_obj, x0, method='SLSQP', jac=qp_grad,
                      bounds=bounds, constraints=cons,
                      options={'maxiter': 200, 'ftol': 1e-16})
        if res.x is not None and np.isfinite(res.x).all():
            w = np.clip(res.x, 0.0, None)
            s = w.sum()
            if s > 1e-15:
                w = w / s
                return w
    except Exception:
        pass
    # Fallback: projected gradient
    w = np.ones(k) / k
    Hw = H @ w
    for it in range(300):
        grad = 2.0 * Hw
        step = 1.0 / (1.0 + 0.05 * it)
        w_try = project_simplex(w - step * grad)
        Hw_try = H @ w_try
        if w_try @ Hw_try <= w @ Hw + 1e-18:
            w = w_try
            Hw = Hw_try
        else:
            step *= 0.5
            w_try = project_simplex(w - step * grad)
            Hw_try = H @ w_try
            w = w_try
            Hw = Hw_try
    return w
