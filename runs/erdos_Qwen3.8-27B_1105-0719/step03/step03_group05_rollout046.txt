def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx] + 1e-15)
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx] + 1e-15)
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()

        else:
            idx = int(np.argmin(h))

        h = np.clip(h, 0.0, 1.0)

def build_gradient_matrix(h, n, dx):
    """Build exact gradient matrix of correlation profile.
    G[m, i] = dP_m/dh_i where P_m = correlate(h, 1-h, 'full')[m] * dx
    """
    M = 2 * n - 1
    G = np.zeros((M, n))
    for m in range(M):
        d = m - (n - 1)
        if d == 0:
            G[m, :] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            # P_m = dx * sum_{i=d}^{n-1} h[i] * (1 - h[i-d])
            # dP_m/dh[i] for i >= d: dx * (1 - h[i-d])
            # dP_m/dh[i] for i < n-d: -dx * h[i+d] (from the (1-h) term)
            G[m, d:] += dx * (1.0 - h[0:n-d])
            G[m, 0:n-d] += -dx * h[d:n]
        else:
            dp = -d
            # P_m = dx * sum_{i=0}^{n-1-dp} h[i] * (1 - h[i+dp])
            G[m, 0:n-dp] += dx * (1.0 - h[dp:n])
            G[m, dp:n] += -dx * h[0:n-dp]
    return G

def solve_lp_direction(h, G, profile, D):
    """Solve LP: min S s.t. G_m·d - S <= -P_m, sum(d)=0, |d_i|<=D"""
    n = len(h)
    M = len(profile)
    # Variables: [d_0,...,d_{n-1}, S]
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize S
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    A_ub = np.zeros((M, n + 1))
    A_ub[:, :n] = G  # G_m · d
    A_ub[:, n] = -1.0  # -S
    b_ub = -profile  # G_m·d - S <= -P_m
    
    bounds_list = [(-D, D)] * n + [(-np.inf, np.inf)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds_list, method='highs')
        if res.success and res.x is not None:
            return res.x[:n]
    except Exception:
        pass
    return None

- Similar for h[j] with -t

So the change is linear + quadratic only if both i and j appear in the same product. Specifically, if there's a term h[i]*(1-h[j]) or h[j]*(1-h[i]), then we get a -t² or +t² term.

Let me be more precise. For lag d > 0:
P_m = dx * sum_{k=0}^{n-1-d} h[k+d] * (1 - h[k])

If I change h[i] → h[i] + δ_i and h[j] → h[j] + δ_j where δ_i + δ_j = 0:
∂P_m/∂t = dx * sum_k [(∂h[k+d]/∂t)(1-h[k]) + h[k+d] * (∂(1-h[k])/∂t)]
= dx * sum_k [(δ_{k+d}/t * 1)(1-h[k]) + h[k+d] * (-δ_k/t * 1)]

where δ_k is 1 if k=i, -1 if k=j, 0 otherwise (for direction e_i - e_j).

Second derivative: only nonzero if both i and j contribute to the same term, i.e., if k+d = i and k = j (or k+d = j and k = i). In that case, the term h[i]*(1-h[j]) becomes (h[i]+t)*(1-h[j]+t) = h[i]*(1-h[j]) + t*(1-h[j]) - t*h[i] - t², so the second derivative contribution is -dx (for this specific pair in the sum).

So the total second derivative term q_m(i,j) = dx * (contribution from cross terms).

import numpy as np
from scipy.optimize import linprog
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx] + 1e-15)
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx] + 1e-15)
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-14:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmin(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def build_gradient_matrix(h, n, dx):
    M = 2 * n - 1
    G = np.zeros((M, n))
    for m in range(M):
        d = m - (n - 1)
        if d == 0:
            G[m, :] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            G[m, d:] += dx * (1.0 - h[0:n - d])
            G[m, 0:n - d] += -dx * h[d:n]
        else:
            dp = -d
            G[m, 0:n - dp] += dx * (1.0 - h[dp:n])
            G[m, dp:n] += -dx * h[0:n - dp]
    return G

def solve_lp_direction(h, G, profile, D):
    n = len(h)
    M = len(profile)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    A_ub = np.zeros((M, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -profile
    bounds_list = [(-D, D)] * n + [(-10.0, 10.0)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds_list, method='highs')
        if res.success and res.x is not None:
            return res.x[:n]
    except Exception:
        pass
    return None

def line_search(h, direction, n_samples=50):
    direction = direction - direction.mean()
    norm = np.linalg.norm(direction)
    if norm < 1e-14:
        return h.copy(), exact_score(h)
    dir_n = direction / norm
    t_max = 2.0
    pos_mask = dir_n > 1e-14
    neg_mask = dir_n < -1e-14
    if pos_mask.any():
        t_max = min(t_max, np.min((1.0 - h[pos_mask]) / dir_n[pos_mask]) * 0.99)
    if neg_mask.any():
        t_max = min(t_max, np.min(h[neg_mask] / (-dir_n[neg_mask])) * 0.99)
    t_max = max(t_max, 0.0)
    if t_max < 1e-12:
        return h.copy(), exact_score(h)
    t_vals = np.linspace(0.0, t_max, n_samples)
    scores = np.zeros(n_samples)
    for i in range(n_samples):
        if t_vals[i] < 1e-12:
            scores[i] = exact_score(h)
        else:
            scores[i] = exact_score(repair_feasible(h + t_vals[i] * dir_n))
    best_idx = int(np.argmin(scores))
    best_t = t_vals[best_idx]
    best_c = scores[best_idx]
    if 0 < best_idx < n_samples - 1:
        lo, hi = t_vals[best_idx - 1], t_vals[best_idx + 1]
    elif best_idx == 0:
        lo, hi = 0.0, t_vals[1]
    else:
        lo, hi = t_vals[-2], t_vals[-1]
    for _ in range(25):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        s1 = exact_score(repair_feasible(h + m1 * dir_n))
        s2 = exact_score(repair_feasible(h + m2 * dir_n))
        if s1 < s2:
            hi = m2
            if s1 < best_c:
                best_c, best_t = s1, m1
        else:
            lo = m1
            if s2 < best_c:
                best_c, best_t = s2, m2
        if hi - lo < 1e-12:
            break
    best_h = repair_feasible(h + best_t * dir_n) if best_t > 1e-12 else h.copy()
    return best_h, exact_score(best_h)

def pair_line_search(h, i, j, n_samples=30):
    if i == j:
        return h.copy(), exact_score(h)
    direction = np.zeros(len(h))
    direction[i] = 1.0
    direction[j] = -1.0
    t_max = 1.0
    if h[i] < h[j]:
        t_max = min(t_max, h[i] * 0.99)
    else:
        t_max = min(t_max, (1.0 - h[j]) * 0.99)
    if t_max < 1e-12:
        return h.copy(), exact_score(h)
    t_vals = np.linspace(0.0, t_max, n_samples)
    scores = np.zeros(n_samples)
    for k in range(n_samples):
        if t_vals[k] < 1e-12:
            scores[k] = exact_score(h)
        else:
            h_try = h.copy()
            h_try[i] = h[i] + t_vals[k]
            h_try[j] = h[j] - t_vals[k]
            scores[k] = exact_score(h_try)
    best_k = int(np.argmin(scores))
    best_t = t_vals[best_k]
    if best_t < 1e-12:
        return h.copy(), exact_score(h)
    if 0 < best_k < n_samples - 1:
        lo, hi = t_vals[best_k - 1], t_vals[best_k + 1]
    elif best_k == 0:
        lo, hi = 0.0, t_vals[1]
    else:
        lo, hi = t_vals[-2], t_vals[-1]
    best_c = scores[best_k]
    for _ in range(20):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        h1 = h.copy()
        h1[i], h1[j] = h[i] + m1, h[j] - m1
        h2 = h.copy()
        h2[i], h2[j] = h[i] + m2, h[j] - m2
        s1, s2 = exact_score(h1), exact_score(h2)
        if s1 < s2:
            hi = m2
            if s1 < best_c:
                best_c, best_t = s1, m1
        else:
            lo = m1
            if s2 < best_c:
                best_c, best_t = s2, m2
        if hi - lo < 1e-12:
            break
    best_h = h.copy()
    best_h[i] = h[i] + best_t
    best_h[j] = h[j] - best_t
    return best_h, exact_score(best_h)

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = np.linspace(0, 2, n + 1)
    xt = np.linspace(0, 2, new_n + 1)
    h_interp = np.zeros(new_n)
    for k in range(new_n):
        lo_k, hi_k = xt[k], xt[k + 1]
        total_mass = 0.0
        total_width = 0.0
        for j in range(n):
            lo_j, hi_j = xs[j], xs[j + 1]
            overlap = min(hi_k, hi_j) - max(lo_k, lo_j)
            if overlap > 0:
                total_mass += h[j] * overlap
                total_width += overlap
        h_interp[k] = total_mass / total_width if total_width > 1e-15 else 0.5
    h_interp = np.clip(h_interp, 0.0, 1.0)
    return repair_feasible(h_interp)

def symmetrize(h):
    return repair_feasible((h + h[::-1]) / 2.0)

def lse_descent(h, beta=80.0, max_iters=100, alpha_init=0.004):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best_c = exact_score(h)
    alpha = alpha_init
    stuck = 0
    for _ in range(max_iters):
        profile = compute_profile(h)
        scaled = beta * profile
        shifted = scaled - np.max(scaled)
        weights = np.exp(shifted)
        wsum = weights.sum()
        if wsum < 1e-300:
            break
        weights /= wsum
        G = build_gradient_matrix(h, n, dx)
        grad = G.T @ weights
        grad -= grad.mean()
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad /= gnorm
        improved = False
        for a in [alpha, alpha * 0.5, alpha * 0.25, alpha * 0.1]:
            h_try = repair_feasible(h - a * grad)
            s = exact_score(h_try)
            if s < best_c - 1e-14:
                h, best_c = h_try, s
                improved = True
                alpha = min(alpha * 1.1, 0.02)
                stuck = 0
                break
        if not improved:
            alpha *= 0.5
            stuck += 1
            if stuck > 25 or alpha < 1e-9:
                break
    return h, best_c

def active_set_descent(h, max_iters=200, n_active=12, alpha_init=0.005):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best_c = exact_score(h)
    alpha = alpha_init
    stuck = 0
    for _ in range(max_iters):
        profile = compute_profile(h)
        c = float(np.max(profile))
        if c + 1e-14 < best_c:
            best_c = c
            stuck = 0
            alpha = min(alpha * 1.08, 0.025)
        margin = max(0.002, 4.0 * (c - float(np.median(profile))))
        active = np.where(profile >= c - margin)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(profile))])
        if len(active) > n_active:
            top_idx = np.argsort(profile[active])[-n_active:]
            active = active[top_idx]
        G = build_gradient_matrix(h, n, dx)
        grad = G[active, :].mean(axis=0)
        grad -= grad.mean()
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad /= gnorm
        improved = False
        for a in [alpha, alpha * 0.5, alpha * 0.25, alpha * 0.1, alpha * 0.05]:
            h_try = repair_feasible(h - a * grad)
            s = exact_score(h_try)
            if s < best_c - 1e-14:
                h, best_c = h_try, s
                improved = True
                break
        if not improved:
            alpha *= 0.55
            stuck += 1
            if stuck > 35 or alpha < 1e-9:
                break
    return h, best_c
