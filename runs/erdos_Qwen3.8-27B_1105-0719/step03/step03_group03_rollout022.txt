The max of quadratics is convex in t (well, not necessarily convex, but piecewise quadratic). We can find the minimum of the max by:

    if t_hi <= t_lo:
        return 0.0, exact_score(h)
    
    # Coarse grid

    best_t = 0.0

def subgrad_nesterov(h, beta_ls, n_iter, budget_fn):
    """Nesterov accelerated subgradient with exact line search."""
    h_curr = h.copy()
    h_next = h.copy()
    c_curr = exact_score(h_curr)
    best_h = h_curr.copy()
    best_c = c_curr
    t_k = 1.0
    
    for k in range(n_iter):
        if budget_fn() < 10:
            break
        
        # Nesterov update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        theta = (t_k - 1.0) / t_new
        h_next = (1.0 + theta) * h_curr - theta * h_next
        
        # Compute gradient at h_next
        g = softmax_grad(h_next, beta_ls)
        g = g - g.mean()  # zero sum
        
        # Normalize
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        g = g / gn
        
        # Line search
        t_best, c_try = exact_line_search(h_curr, -g, n_grid=80)
        
        if c_try < c_curr - 1e-14:
            h_curr = h_curr + t_best * (-g)
            c_curr = c_try
            if c_curr < best_c:
                best_c = c_curr
                best_h = h_curr.copy()
        
        t_k = t_new
    
    return best_h, best_c

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            add = min(remaining, capacity[order[i]])
            if add > 0:
                h[order[i]] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            rem = min(remaining, h[order[i]])
            if rem > 0:
                h[order[i]] -= rem
            remaining -= rem
            if remaining <= 1e-12:

    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            h[int(np.argmax(1.0 - h))] += diff2
        else:
            h[int(np.argmax(h))] += diff2

def softmax_grad(h, beta):

    dx = 2.0 / n
    profile = compute_profile(h)
    s = profile - profile.max()
    ew = np.exp(beta * s)

    omh = 1.0 - h

            g[ks2] += wm * dx * (-h[ks2 - d])
    return g


def lag_gradient(h, lag_idx):
    """Gradient of P_{lag_idx} with respect to h."""
    n = len(h)
    dx = 2.0 / n
    center = n - 1
    d = lag_idx - center
    g = np.zeros(n)
    omh = 1.0 - h
    if d >= 0:
        ks = np.arange(0, n - d)
        g[ks] += dx * omh[ks + d]
        if d > 0:
            ks2 = np.arange(d, n)
            g[ks2] += dx * (-h[ks2 - d])
    else:
        ks = np.arange(-d, n)
        g[ks] += dx * omh[ks + d]
        ks2 = np.arange(0, n + d)
        g[ks2] += dx * (-h[ks2 - d])
    return g


def try_direction(h, v, best_c):
    """Try moving along direction v (zero-sum), return (h_new, c_new) or (None, best_c)."""
    v = v - v.mean()
    vn = np.linalg.norm(v)
    if vn < 1e-14:
        return None, best_c
    v = v / vn
    t_lo, t_hi = feasible_range(h, v)
    if t_hi - t_lo < 1e-10:
        return None, best_c
    A, B, C = compute_abc(h, v)
    t_opt, F_opt = quadratic_line_search(A, B, C, t_lo, t_hi)
    if F_opt < best_c - 1e-12:
        h_new = np.clip(h + t_opt * v, 0.0, 1.0)
        h_new = repair_feasible(h_new)
        c_actual = exact_score(h_new)
        if c_actual < best_c - 1e-12:
            return h_new, c_actual
    return None, best_c


def interp_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    fx = np.concatenate([[0.0], xs, [2.0]])
    fh = np.concatenate([[h[0]], h, [h[-1]]])
    hn = np.interp(xt, fx, fh)
    return repair_feasible(np.clip(hn, 0, 1))


def binary_to_balanced(h_bin, n_target_ones=None):
    """Ensure binary vector has exactly n/2 ones."""
    n = len(h_bin)
    target = n // 2
    n_ones = int((h_bin > 0.5).sum())
    if n_ones == target:
        return h_bin.copy()
    h = h_bin.copy()
    if n_ones > target:
        one_idx = np.where(h_bin > 0.5)[0]
        order = one_idx[np.argsort(h_bin[one_idx])]
        for i in range(n_ones - target):
            h[order[i]] = 0.0
    else:
        zero_idx = np.where(h_bin < 0.5)[0]
        order = zero_idx[np.argsort(-h_bin[zero_idx])]
        for i in range(target - n_ones):
            h[order[i]] = 1.0
    return h


def binary_local_search(h_bin, max_rounds=50):
    """Local search via swaps on binary patterns."""
    h = h_bin.copy()
    n = len(h)
    best_c = exact_score(h)
    for _ in range(max_rounds):
        improved = False
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        for i_idx in range(len(ones)):
            for j_idx in range(len(zeros)):
                i, j = ones[i_idx], zeros[j_idx]
                h[i], h[j] = 0.0, 1.0
                c = exact_score(h)
                if c < best_c - 1e-14:
                    best_c = c
                    improved = True
                    break
                else:
                    h[i], h[j] = 1.0, 0.0
            if improved:
                break
        if not improved:
            break
    return h, best_c


def coordinate_swap_search(h, max_swaps=200, budget_fn=None):
    """Try coordinate transfer moves h[i]+=eps, h[j]-=eps."""
    n = len(h)
    best_c = exact_score(h)
    eps_vals = [0.01, 0.02, 0.05, 0.1]
    order_i = np.argsort(-h)
    order_j = np.argsort(h)
    count = 0
    for ei in range(min(30, n)):
        for ej in range(min(30, n)):
            if count > max_swaps:
                break
            if budget_fn is not None and budget_fn() < 15:
                return h, best_c
            i, j = order_i[ei], order_j[ej]
            if i == j or h[i] < 1e-10 or h[j] > 1.0 - 1e-10:
                continue
            for eps in eps_vals:
                if h[i] - eps < 0 or h[j] + eps > 1:
                    continue
                v = np.zeros(n)
                v[i] = -eps
                v[j] = eps
                h_try = h + v - v.mean()
                h_try = np.clip(h_try, 0, 1)
                h_try = repair_feasible(h_try)
                c_try = exact_score(h_try)
                if c_try < best_c - 1e-14:
                    h = h_try
                    best_c = c_try
                    count += 1
                    break
    return h, best_c


def nesterov_loop(h, beta, n_iter, budget_fn):
    """Nesterov-accelerated subgradient with exact quadratic line search."""
    h_curr = h.copy()
    h_prev = h.copy()
    c_curr = exact_score(h_curr)
    best_h = h_curr.copy()
    best_c = c_curr
    t_k = 1.0

    for k in range(n_iter):
        if budget_fn() < 8:
            break
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        theta = (t_k - 1.0) / t_new
        h_y = (1.0 + theta) * h_curr - theta * h_prev

        g = softmax_grad(h_y, beta)
        h_new, c_new = try_direction(h_curr, -g, c_curr)
        if h_new is not None:
            h_prev = h_curr
            h_curr = h_new
            c_curr = c_new
            if c_curr < best_c:
                best_c = c_curr
                best_h = h_curr.copy()
        else:
            # Try individual lag gradients
            profile = compute_profile(h_curr)
            active = np.argsort(profile)[-5:]
            found = False
            for m in active:
                g_m = lag_gradient(h_curr, m)
                h_new, c_new = try_direction(h_curr, -g_m, c_curr)
                if h_new is not None:
                    h_prev = h_curr
                    h_curr = h_new
                    c_curr = c_new
                    if c_curr < best_c:
                        best_c = c_curr
                        best_h = h_curr.copy()
                    found = True
                    break
            if not found:
                h_prev = h_curr
        t_k = t_new
    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()
    rng = np.random.RandomState(seed)

    def tl():
        return budget_s + t_start - time.perf_counter()

    try:
        parent = np.array(initial_h_values, dtype=np.float64)
    except (NameError, Exception):
        parent = np.full(70, 0.5)

    if parent is None or len(parent) < 4:
        parent = np.full(70, 0.5)

    parent = repair_feasible(np.clip(parent, 0.0, 1.0))
    best_h = parent.copy()
    best_c = exact_score(parent)

    # Symmetrize
    h_sym = repair_feasible((parent + parent[::-1]) / 2.0)
    if exact_score(h_sym) < best_c - 1e-12:
        best_h, best_c = h_sym, exact_score(h_sym)

    # Phase 1: Nesterov with various beta
    if tl() > 200:
        for beta in [30.0, 80.0, 150.0, 300.0]:
            if tl() < 60:
                break
            hn, cn = nesterov_loop(best_h, beta, n_iter=200, budget_fn=tl)
            if cn < best_c - 1e-14:
                best_h, best_c = hn, cn

    # Phase 2: Binary exploration from parent
    if tl() > 150:
        n_par = len(best_h)
        h_bin = binary_to_balanced((best_h > 0.5).astype(np.float64))
        hb, cb = binary_local_search(h_bin, max_rounds=80)
        if cb < best_c - 1e-14:
            best_h, best_c = hb, cb
        # Nesterov from binary
        hn, cn = nesterov_loop(hb, beta=100.0, n_iter=150, budget_fn=tl)
        if cn < best_c - 1e-14:
            best_h, best_c = hn, cn

    # Phase 3: Random binary restarts
    if tl() > 150:
        n_r = len(best_h)
        for attempt in range(30):
            if tl() < 40:
                break
            h_rand = np.zeros(n_r)
            ones_pos = rng.choice(n_r, size=n_r // 2, replace=False)
            h_rand[ones_pos] = 1.0
            cr = exact_score(h_rand)
            if cr < best_c + 0.08:
                hb, cb = binary_local_search(h_rand, max_rounds=20)
                if cb < best_c - 1e-14:
                    best_h, best_c = hb, cb
                hn, cn = nesterov_loop(hb, beta=100.0, n_iter=60, budget_fn=tl)
                if cn < best_c - 1e-14:
                    best_h, best_c = hn, cn

    # Phase 4: Multi-resolution
    if tl() > 120:
        n0 = len(best_h)
        dims = sorted([d for d in range(max(20, n0 - 16), min(140, n0 + 17), 2)
                       if d != n0 and d >= 20], key=lambda d: abs(d - n0))
        for nd in dims[:8]:
            if tl() < 50:
                break
            hi = interp_to_n(best_h, nd)
            ci = exact_score(hi)
            if ci < best_c + 0.03:
                hn, cn = nesterov_loop(hi, beta=100.0, n_iter=100, budget_fn=tl)
                if cn < best_c - 1e-14:
                    best_h, best_c = hn, cn

    # Phase 5: Coordinate swap search
    if tl() > 100:
        best_h, best_c = coordinate_swap_search(best_h, max_swaps=300, budget_fn=tl)
        hn, cn = nesterov_loop(best_h, beta=200.0, n_iter=100, budget_fn=tl)
        if cn < best_c - 1e-14:
            best_h, best_c = hn, cn

    # Phase 6: SLSQP polish
    if tl() > 80:
        h_s = best_h.copy()
        n_s = len(h_s)
        dx_s = 2.0 / n_s
        M = 2 * n_s - 1

        def obj_fn(x):
            return x[-1]

        def obj_jac_fn(x):
            gg = np.zeros(n_s + 1)
            gg[-1] = 1.0
            return gg

        def eq_fn(x):
            return x[:-1].sum() - n_s / 2.0

        def eq_jac_fn(x):
            gg = np.zeros(n_s + 1)
            gg[:-1] = 1.0
            return gg

        def ineq_fn(x):
            hh = x[:-1]
            return x[-1] - np.correlate(hh, 1.0 - hh, mode="full") * dx_s

        def ineq_jac_fn(x):
            hh = x[:-1]
            G = np.zeros((M, n_s))
            center = n_s - 1
            omhh = 1.0 - hh
            for m in range(M):
                d = m - center
                if d >= 0:
                    ks = np.arange(0, n_s - d)
                    G[m, ks] += dx_s * omhh[ks + d]
                    if d > 0:
                        ks2 = np.arange(d, n_s)
                        G[m, ks2] += -dx_s * hh[ks2 - d]
                else:
                    ks = np.arange(-d, n_s)
                    G[m, ks] += dx_s * omhh[ks + d]
                    ks2 = np.arange(0, n_s + d)
                    G[m, ks2] += -dx_s * hh[ks2 - d]
            J = np.zeros((M, n_s + 1))
            J[:, :-1] = -G
            J[:, -1] = 1.0
            return J

        t0 = exact_score(h_s) + 1e-3
        x0 = np.concatenate([h_s, [t0]])
        bounds = [(0.0, 1.0)] * n_s + [(0.0, t0 + 1.0)]
        cons = [
            {'type': 'eq', 'fun': eq_fn, 'jac': eq_jac_fn},
            {'type': 'ineq', 'fun': ineq_fn, 'jac': ineq_jac_fn}
        ]
        try:
            res = minimize(obj_fn, x0, method='SLSQP', jac=obj_jac_fn,
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 400, 'ftol': 1e-14})
            if res.x is not None and np.isfinite(res.x).all():
                hs = repair_feasible(res.x[:-1])
                cs = exact_score(hs)
                if cs < best_c - 1e-14:
                    best_h, best_c = hs, cs
        except Exception:
            pass

    # Phase 7: Final Nesterov polish with high beta
    if tl() > 30:
        for beta in [200.0, 500.0]:
            if tl() < 15:
                break
            hn, cn = nesterov_loop(best_h, beta, n_iter=200, budget_fn=tl)
            if cn < best_c - 1e-14:
                best_h, best_c = hn, cn

    # Phase 8: Micro coordinate adjustments
    if tl() > 15:
        best_h, best_c = coordinate_swap_search(best_h, max_swaps=100, budget_fn=tl)

    # Final validation
    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)

    try:
        er = evaluate_erdos_solution(best_h)
        if isinstance(er, (float, int)):
            final_c = min(final_c, float(er))
        elif isinstance(er, tuple) and len(er) >= 2:
            final_c = min(final_c, float(er[0]))
    except Exception:
        pass

    return best_h, final_c, len(best_h)
