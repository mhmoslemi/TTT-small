```python
def coord_descent_exact(h, dx, target_sum, n_sweeps=50):
    n = len(h)
    h = h.copy()
    best_c5 = c5_of_h(h, dx)
    best_h = h.copy()
    
    for sweep in range(n_sweeps):
        improved = False
        for i in range(n):
            # Compute a_k and b_k for each correlation k
            # For correlation k with offset m = k - (n-1):
            # a_k = 1 - h[i+m] - h[i-m] (out of range = 0)
            # b_k = corr_k - h[i]*(1-h[i+m]) + h[i]*h[i-m]
            
            corrs = np.correlate(h, 1.0 - h, mode='full')
            
            a = np.zeros(2*n - 1)
            b = np.zeros(2*n - 1)
            
            for k in range(2*n - 1):
                m = k - (n - 1)
                # h[i+m] contribution
                ipm = i + m
                h_ipm = h[ipm] if 0 <= ipm < n else 0.0
                # h[i-m] contribution
                imm = i - m
                h_imm = h[imm] if 0 <= imm < n else 0.0
                
                a[k] = 1.0 - h_ipm - h_imm
                b[k] = corrs[k] - h[i]*(1.0 - h_ipm) + h[i]*h_imm
            
            # Minimize f(t) = max_k (a[k]*t + b[k]) over t in [0,1]
            # Use golden section search
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            lo, hi = 0.0, 1.0
            c_pt = hi - gr * (hi - lo)
            d_pt = lo + gr * (hi - lo)
            fc = (a * c_pt + b).max()
            fd = (a * d_pt + b).max()
            
            for _ in range(80):
                if hi - lo < 1e-14:
                    break
                if fc < fd:
                    hi = d_pt
                    d_pt = c_pt
                    fd = fc
                    c_pt = hi - gr * (hi - lo)
                    fc = (a * c_pt + b).max()
                else:
                    lo = c_pt
                    c_pt = d_pt
                    fc = fd
                    d_pt = lo + gr * (hi - lo)
                    fd = (a * d_pt + b).max()
            
            t_opt = (lo + hi) / 2.0
            
            # Also check boundaries
            f_lo = (a * 0.0 + b).max()
            f_hi = (a * 1.0 + b).max()
            f_mid = (a * t_opt + b).max()
            
            if f_lo <= f_mid and f_lo <= f_hi:
                t_opt = 0.0
            elif f_hi <= f_mid and f_hi <= f_lo:
                t_opt = 1.0
            
            if t_opt != h[i]:
                h[i] = t_opt
                improved = True
        
        c5_val = c5_of_h(h, dx)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h.copy()
        
        if not improved or c5_val >= best_c5 - 1e-15:
            # Check if we made progress this sweep
            if c5_val >= best_c5 - 1e-15 and sweep > 2:
                break
    
    return best_h, best_c5
```

```python
def directional_line_search(h0, dx, target_sum, d, alpha_max=1.0):
    """Find best alpha in [0, alpha_max] to move along direction d (sum(d)=0).
    h_new = project_feasible(h0 + alpha * d, target_sum)"""
    n = len(h0)
    
    # Find valid alpha range where 0 <= h0 + alpha*d <= 1
    pos_mask = d > 0
    neg_mask = d < 0
    a_lo = 0.0
    a_hi = alpha_max
    if pos_mask.any():
        a_hi = min(a_hi, (1.0 - h0[pos_mask]) / d[pos_mask]).min()
    if neg_mask.any():
        a_lo = max(a_lo, -h0[neg_mask] / d[neg_mask]).min()
    
    if a_hi - a_lo < 1e-12:
        return h0, c5_of_h(h0, dx)
    
    # Scan 200 points
    alphas = np.linspace(a_lo, a_hi, 200)
    best_alpha = a_lo
    best_val = float('inf')
    
    for a in alphas:
        h_try = h0 + a * d
        # Check feasibility (should be by construction)
        if h_try.min() < -1e-10 or h_try.max() > 1.0 + 1e-10:
            continue
        h_try = np.clip(h_try, 0, 1)
        c5 = compute_corrs(h_try).max() * dx
        if c5 < best_val:
            best_val = c5
            best_alpha = a
    
    # Refine around best_alpha with golden section (but function may not be convex)
    # Use fine scan instead
    span = (a_hi - a_lo) / 200.0
    alphas_fine = np.linspace(best_alpha - span, best_alpha + span, 200)
    alphas_fine = alphas_fine[(alphas_fine >= a_lo) & (alphas_fine <= a_hi)]
    
    for a in alphas_fine:
        h_try = np.clip(h0 + a * d, 0, 1)
        c5 = compute_corrs(h_try).max() * dx
        if c5 < best_val:
            best_val = c5
            best_alpha = a
    
    h_new = np.clip(h0 + best_alpha * d, 0, 1)
    h_new = project_feasible(h_new, target_sum)
    return h_new, c5_of_h(h_new, dx)
```

```python
import numpy as np
from scipy.optimize import minimize, linprog
from scipy.special import logsumexp
import time


def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)


def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def c5_of_h(h, dx):
    return compute_corrs(h).max() * dx


def golden_min_convex_pl(A, B, lo, hi, n_iter=55):
    """Minimize f(t)=max_k(A[k]*t+B[k]) over [lo,hi]. f is convex PL."""
    if hi - lo < 1e-15:
        return (lo + hi) / 2.0
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = (A * c + B).max()
    fd = (A * d + B).max()
    for _ in range(n_iter):
        if hi - lo < 1e-15:
            break
        if fc <= fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = (A * c + B).max()
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = (A * d + B).max()
    t_mid = (lo + hi) / 2.0
    f_mid = (A * t_mid + B).max()
    f_lo = (A * lo + B).max()
    f_hi = (A * hi + B).max()
    if f_lo <= f_mid and f_lo <= f_hi:
        return lo
    if f_hi <= f_mid and f_hi <= f_lo:
        return hi
    return t_mid


def paired_cd_update(h, i, j, corrs, m_vals, n):
    """Compute optimal pair update for coordinates i,j. Returns (t_opt, new_h_i, new_h_j)."""
    h_i, h_j = h[i], h[j]
    t_lo = max(0.0, h_i + h_j - 1.0)
    t_hi = min(1.0, h_i + h_j)
    if t_hi - t_lo < 1e-12:
        return t_lo, h_i, h_j

    ipm = i + m_vals
    imm = i - m_vals
    jpm = j + m_vals
    jmm = j - m_vals

    alpha_i = ((ipm >= 0) & (ipm < n)).astype(np.float64)
    beta_i = ((imm >= 0) & (imm < n)).astype(np.float64)
    alpha_j = ((jpm >= 0) & (jpm < n)).astype(np.float64)
    beta_j = ((jmm >= 0) & (jmm < n)).astype(np.float64)

    h_ipm = np.where(alpha_i > 0, h[np.clip(ipm, 0, n - 1).astype(int)], 0.0)
    h_imm = np.where(beta_i > 0, h[np.clip(imm, 0, n - 1).astype(int)], 0.0)
    h_jpm = np.where(alpha_j > 0, h[np.clip(jpm, 0, n - 1).astype(int)], 0.0)
    h_jmm = np.where(beta_j > 0, h[np.clip(jmm, 0, n - 1).astype(int)], 0.0)

    old_i = alpha_i * h_i * (1.0 - h_ipm) + beta_i * h_imm * (1.0 - h_i)
    old_j = alpha_j * h_j * (1.0 - h_jpm) + beta_j * h_jmm * (1.0 - h_j)

    A = alpha_i * (1.0 - h_ipm) - beta_i * h_imm - alpha_j * (1.0 - h_jpm) + beta_j * h_jmm
    C_k = beta_i * h_imm + alpha_j * (h_i + h_j) * (1.0 - h_jpm) + beta_j * h_jmm * (1.0 - h_i - h_j)
    D = corrs - old_i - old_j + C_k

    t_opt = golden_min_convex_pl(A, D, t_lo, t_hi, 50)
    return t_opt, t_opt, h_i + h_j - t_opt


def paired_coord_descent(h0, dx, target_sum, n_sweeps=300, rng=None, mode='sequential'):
    """Exact paired coordinate descent with sum constraint."""
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    if rng is None:
        rng = np.random.RandomState(42)

    nn = 2 * n - 1
    m_vals = np.arange(nn, dtype=np.float64) - (n - 1)
    stall = 0

    for sweep in range(n_sweeps):
        any_improved = False
        corrs = compute_corrs(h)

        if mode == 'sequential':
            order = rng.permutation(n)
            pairs = [(order[i], order[(i + 1) % n]) for i in range(n)]
        elif mode == 'random':
            pairs = [(rng.randint(n), rng.randint(n)) for _ in range(3 * n)]
        else:  # 'shifted'
            shift = sweep % max(1, n // 2)
            order = rng.permutation(n)
            pairs = [(order[i], order[(i + 1 + shift) % n]) for i in range(n)]

        for i, j in pairs:
            if i == j:
                continue
            t_opt, new_hi, new_hj = paired_cd_update(h, i, j, corrs, m_vals, n)
            if abs(t_opt - h[i]) > 1e-13:
                h[i] = new_hi
                h[j] = new_hj
                any_improved = True
                corrs = compute_corrs(h)

        c5_val = c5_of_h(h, dx)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

        if not any_improved or stall > 30:
            break

    return best_h, best_c5


def grad_corr_k(h, k, n):
    offset = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -offset)
    i_hi = min(n - 1, n - 1 - offset)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx + offset] += (1.0 - h[idx])
    grad[idx] += (-h[idx + offset])
    return grad


def smooth_max_grad(h, dx, beta):
    n = len(h)
    corrs = compute_corrs(h)
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones_like(corrs) / len(corrs)
    else:
        w /= w_sum
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] > 1e-12:
            grad += w[k] * grad_corr_k(h, k, n)
    return grad * dx


def directional_search(h0, dx, target_sum, d, n_scan=300):
    """Exact line search along direction d (sum(d)=0) on the true objective."""
    n = len(h0)
    d = d - d.mean()  # ensure sum = 0
    if np.abs(d).max() < 1e-15:
        return h0, c5_of_h(h0, dx)

    pos_mask = d > 1e-15
    neg_mask = d < -1e-15
    a_lo = 0.0
    a_hi = 2.0
    if pos_mask.any():
        a_hi = min(a_hi, ((1.0 - h0[pos_mask]) / d[pos_mask]).min() * 0.99)
    if neg_mask.any():
        a_lo = max(a_lo, (-h0[neg_mask] / d[neg_mask]).min() * 0.99)

    if a_hi <= a_lo:
        return h0, c5_of_h(h0, dx)

    alphas = np.linspace(a_lo, a_hi, n_scan)
    best_alpha = a_lo
    best_val = float('inf')
    corrs0 = compute_corrs(h0)

    for a in alphas:
        h_try = h0 + a * d
        if h_try.min() < -1e-10 or h_try.max() > 1.0 + 1e-10:
            continue
        c5 = compute_corrs(h_try).max() * dx
        if c5 < best_val:
            best_val = c5
            best_alpha = a

    # Refine
    span = (a_hi - a_lo) / n_scan
    alphas_fine = np.linspace(max(a_lo, best_alpha - 2 * span), min(a_hi, best_alpha + 2 * span), 200)
    for a in alphas_fine:
        h_try = h0 + a * d
        if h_try.min() < -1e-10 or h_try.max() > 1.0 + 1e-10:
            continue
        c5 = compute_corrs(h_try).max() * dx
        if c5 < best_val:
            best_val = c5
            best_alpha = a

    h_new = np.clip(h0 + best_alpha * d, 0, 1)
    h_new = project_feasible(h_new, target_sum)
    return h_new, c5_of_h(h_new, dx)


def slsqp_optimize(h0, dx, target_sum, beta, maxiter=1000):
    n = len(h0)

    def obj(x):
        corrs = compute_corrs(x)
        return logsumexp(beta * corrs) / beta * dx

    def grad_obj(x):
        return smooth_max_grad(x, dx, beta)

    def c_fun(x):
        return x.sum() - target_sum

    cons = [{'type': 'eq', 'fun': c_fun, 'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0, jac=grad_obj, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        h_out = project_feasible(res.x, target_sum)
    except Exception:
        h_out = project_feasible(h0, target_sum)
    return h_out, c5_of_h(h_out, dx)


def lp_polish(h0, dx, target_sum, n_steps=50, lam=0.15):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)

    for s in range(n_steps):
        corrs = compute_corrs(h)
        max_c = corrs.max()
        c5_val = max_c * dx

        n_active = max(10, int(0.3 * (2 * n - 1)))
        top_idx = np.argsort(corrs)[-n_active:]

        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        A_rows = []
        b_rows = []
        for k in top_idx:
            row = np.zeros(n + 1)
            row[:n] = dx * grad_corr_k(h, k, n)
            row[n] = -1.0
            A_rows.append(row)
            b_rows.append(-corrs[k] * dx - 1e-14)

        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)

        bounds = []
        for i in range(n):
            bounds.append((max(-h[i], -lam), min(1.0 - h[i], lam)))
        bounds.append((0.0, None))

        try:
            res = linprog(c_obj, A_ub=np.array(A_rows), b_ub=np.array(b_rows),
                          A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
        except Exception:
            break
        if res.status != 0:
            break

        delta = res.x[:n]
        if np.max(np.abs(delta)) < 1e-14:
            break

        h_new = project_feasible(h + delta, target_sum)
        new_c5 = c5_of_h(h_new, dx)

        if new_c5 < c5_val - 1e-14:
            h = h_new
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_new.copy()
        else:
            lam *= 0.5
            if lam < 1e-7:
                break

    return best_h, best_c5


def make_random_start(n, target_sum, seed):
```
