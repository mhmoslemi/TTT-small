def tangent_descent_dir(grad, h):
    """Project -grad onto the tangent cone of the feasible set at h."""
    n = len(h)
    # Free variables: 0 < h_i < 1
    free = (h > 1e-10) & (h < 1 - 1e-10)
    # Bounded below: h_i ≈ 0, can only increase
    below = h <= 1e-10
    # Bounded above: h_i ≈ 1, can only decrease
    above = h >= 1 - 1e-10
    
    # d_i = -grad_i - lambda for free
    # d_i = max(0, -grad_i - lambda) for below
    # d_i = min(0, -grad_i - lambda) for above
    # Sum constraint: sum(d) = 0
    
    # Binary search on lambda
    def sum_d(lam):
        s = 0.0
        for i in range(n):
            v = -grad[i] - lam
            if free[i]:
                s += v
            elif below[i]:
                s += max(0, v)
            else:
                s += min(0, v)
        return s

    lo, hi = -100, 100

    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    
    step_sizes = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    step_idx = 0
    stall = 0
    
    max_iters = 2000
    for it in range(max_iters):
        if time.time() - t0 > time_budget:  # need to pass t0
            break
        
        corrs = compute_corrs(h)
        c5_val = corrs.max() * dx
        
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > 200:
            break
        
        # Active set: top correlations
        if stall < 50:
            n_active = 60
        elif stall < 150:
            n_active = 40
        else:
            n_active = 25
        
        top_idx = np.argsort(corrs)[-n_active:]
        
        # LP
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        A_rows = []
        b_rows = []
        step = step_sizes[min(step_idx, len(step_sizes)-1)]
        
        for k in top_idx:
            gk = grad_corr_k(h, k, n)
            row = np.zeros(n + 1)
            row[:n] = dx * gk
            row[n] = -1.0
            A_rows.append(row)
            b_rows.append(-corrs[k] * dx)
        
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        bounds = []
        for i in range(n):
            bounds.append((max(-h[i], -step), min(1.0 - h[i], step)))
        bounds.append((0.0, None))
        
        try:
            res = linprog(c_obj, A_ub=np.array(A_rows), b_ub=np.array(b_rows),
                          A_eq=A_eq, b_eq=np.zeros(1), bounds=bounds, method='highs')
        except:
            break
        
        if res.status != 0:
            break
        
        delta = res.x[:n]
        if np.max(np.abs(delta)) < 1e-15:
            # Try smaller step
            step_idx = min(step_idx + 1, len(step_sizes) - 1)
            if step_sizes[step_idx] < 1e-6:
                break
            continue

        new_c5 = c5_of_h(h_new, dx)
        
        if new_c5 < c5_val - 1e-15:
            h = h_new
            step_idx = 0  # Reset step size on success

                    break

                if step_sizes[step_idx] < 1e-6:
                    break

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.special import logsumexp
import time


def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h


def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def c5_of_h(h, dx):
    return compute_corrs(h).max() * dx


def grad_corr_k(h, k, n):
    offset = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -offset)
    i_hi = min(n - 1, n - 1 - offset)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx + offset] += (1.0 - h[idx])
    grad[idx] += (-h[idx + offset])
    return grad


def smooth_max_grad(h, dx, beta):
    n = len(h)
    corrs = compute_corrs(h)
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones_like(corrs) / len(corrs)
    else:
        w /= w_sum
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] > 1e-12:
            grad += w[k] * grad_corr_k(h, k, n)
    return grad * dx


def slsqp_opt(h0, dx, target_sum, beta, maxiter, t0, deadline):
    n = len(h0)

    def obj(x):
        corrs = compute_corrs(x)
        return logsumexp(beta * corrs) / beta * dx

    def grad_obj(x):
        return smooth_max_grad(x, dx, beta)

    def c_fun(x):
        return x.sum() - target_sum

    def c_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': c_fun, 'jac': c_jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0, jac=grad_obj, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        h_out = project_feasible(res.x, target_sum)
    except Exception:
        h_out = project_feasible(h0, target_sum)
    return h_out, c5_of_h(h_out, dx)


def lp_polish(h0, dx, target_sum, t0, deadline, n_active_base=50, max_iters=2000):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)

    step_cycle = [0.08, 0.04, 0.02, 0.01, 0.005, 0.003, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    step_idx = 0
    stall = 0
    stall_threshold = 100

    for it in range(max_iters):
        if time.time() > deadline:
            break

        corrs = compute_corrs(h)
        c5_val = corrs.max() * dx

        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

        if stall > stall_threshold:
            step_idx = min(step_idx + 1, len(step_cycle) - 1)
            if step_cycle[step_idx] < 1e-7:
                break
            stall = 0

        # Adaptive active set size
        if stall < 30:
            n_act = n_active_base
        elif stall < 80:
            n_act = max(20, n_active_base - 20)
        else:
            n_act = max(12, n_active_base - 35)

        top_idx = np.argsort(corrs)[-n_act:]
        step = step_cycle[step_idx]

        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        A_rows = []
        b_rows = []
        for k in top_idx:
            gk = grad_corr_k(h, k, n)
            row = np.zeros(n + 1)
            row[:n] = dx * gk
            row[n] = -1.0
            A_rows.append(row)
            b_rows.append(-corrs[k] * dx - 1e-14)

        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        bounds = []
        for i in range(n):
            lo = max(-h[i], -step)
            hi = min(1.0 - h[i], step)
            bounds.append((lo, hi))
        bounds.append((0.0, None))

        try:
            res = linprog(c_obj, A_ub=np.array(A_rows), b_ub=np.array(b_rows),
                          A_eq=A_eq, b_eq=np.zeros(1), bounds=bounds, method='highs')
        except Exception:
            break

        if res.status != 0:
            step_idx = min(step_idx + 1, len(step_cycle) - 1)
            if step_cycle[step_idx] < 1e-7:
                break
            continue

        delta = res.x[:n]
        if np.max(np.abs(delta)) < 1e-15:
            step_idx = min(step_idx + 1, len(step_cycle) - 1)
            if step_cycle[step_idx] < 1e-7:
                break
            continue

        h_new = project_feasible(h + delta, target_sum)
        new_c5 = c5_of_h(h_new, dx)

        if new_c5 < c5_val - 1e-15:
            h = h_new
            step_idx = max(0, step_idx - 1)
        else:
            improved = False
            for frac in [0.5, 0.25, 0.1, 0.05, 0.02]:
                h_new = project_feasible(h + frac * delta, target_sum)
                new_c5 = c5_of_h(h_new, dx)
                if new_c5 < c5_val - 1e-15:
                    h = h_new
                    improved = True
                    break
            if not improved:
                step_idx = min(step_idx + 1, len(step_cycle) - 1)
                if step_cycle[step_idx] < 1e-7:
                    break

    return best_h, best_c5


def pair_transfer(h0, dx, target_sum, t0, deadline, n_top=30, n_rounds=5):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)

    for rnd in range(n_rounds):
        if time.time() > deadline:
            break

        corrs = compute_corrs(h)
        grad = smooth_max_grad(h, dx, 400.0)

        dec_idx = np.argsort(grad)[-n_top:]
        inc_idx = np.argsort(grad)[:n_top]

        improved_round = False
        for i in dec_idx:
            if time.time() > deadline:
                break
            for j in inc_idx:
                if i == j:
                    continue
                eps_lo = max(h[i] - 1.0, -h[j])
                eps_hi = min(h[i], 1.0 - h[j])
                if eps_hi - eps_lo < 1e-12:
                    continue

                d = np.zeros(2 * n - 1)
                for k in range(2 * n - 1):
                    gk = grad_corr_k(h, k, n)
                    d[k] = gk[j] - gk[i]

                corrs_base = corrs.copy()

                def f_eval(eps):
                    return (corrs_base + eps * d).max()

                gr = (np.sqrt(5.0) - 1.0) / 2.0
                a, b = eps_lo, eps_hi
                c_pt = b - gr * (b - a)
                d_pt = a + gr * (b - a)
                fc, fd = f_eval(c_pt), f_eval(d_pt)
                for _ in range(40):
                    if b - a < 1e-13:
                        break
                    if fc < fd:
                        b = d_pt
                        d_pt = c_pt
                        fd = fc
                        c_pt = b - gr * (b - a)
                        fc = f_eval(c_pt)
                    else:
                        a = c_pt
                        c_pt = d_pt
                        fc = fd
                        d_pt = a + gr * (b - a)
                        fd = f_eval(d_pt)

                eps_opt = (a + b) / 2.0
                new_h = h.copy()
                new_h[i] -= eps_opt
                new_h[j] += eps_opt
                new_h = project_feasible(new_h, target_sum)
                new_c5 = c5_of_h(new_h, dx)

                if new_c5 < best_c5 - 1e-14:
                    h = new_h
                    best_h = h.copy()
                    best_c5 = new_c5
                    corrs = compute_corrs(h)
                    improved_round = True

        if not improved_round:
            break

    return best_h, best_c5


def tangent_descent(h0, dx, target_sum, t0, deadline, max_iter=5000, beta=100.0):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    no_improve = 0

    for it in range(max_iter):
        if time.time() > deadline:
            break

        corrs = compute_corrs(h)
        c5_val = corrs.max() * dx

        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 1000:
            break

        shifted = corrs - corrs.max()
        w = np.exp(beta * shifted)
        w /= w.sum()
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] > 1e-10:
                grad += w[k] * grad_corr_k(h, k, n)
        grad *= dx

        # T
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Project negative gradient onto tangent cone
        d = -grad / gnorm
        d_mean = d - d.mean()

        # L
        alpha = 1.0 / (it + 1) * 0.3
        h_new = h + alpha * d_mean
        h_new = project_feasible(h_new, target_sum)
        new_c5 = c5_of_h(h_new, dx)

        if new_c5 > c5_val:
            alpha *= 0.25
            h_new = project_feasible(h + alpha * d_mean, target_sum)
            new_c5 = c5_of_h(h_new, dx)
            if new_c5 > c5_val:
                alpha *= 0.25
                h_new = project_feasible(h + alpha * d_mean, target_sum)
                new_c5 = c5_of_h(h_new, dx)
                if new_c5 > c5_val:
                    continue

        h = h_new

    return best_h, best_c5
