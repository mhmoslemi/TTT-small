    # Using bisection on the Lagrange multiplier
    n = len(x)
    x = np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0)

        if h.sum() > target_sum:

        else:

def grad_all_corrs(h, n):
    """Compute gradients of all correlation values w.r.t. h. Returns (2n-1, n) array."""

import numpy as np
from scipy.optimize import linprog, minimize_scalar, minimize
import time


def project_feasible(x, target_sum):
    n = len(x)
    x = np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0)
    lo, hi = -1.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = h < 1.0 - 1e-9
        else:
            mask = h > 1e-9
        cnt = mask.sum()
        if cnt > 0:
            h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)


def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def c5_of_h(h, dx):
    return float(compute_corrs(h).max() * dx)


def grad_corr_k(h, k, n):
    offset = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -offset)
    i_hi = min(n - 1, n - 1 - offset)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    j1 = idx + offset
    grad[j1] += (1.0 - h[idx])
    grad[idx] += (-h[idx + offset])
    return grad


def compute_all_grads(h, n):
    m = 2 * n - 1
    G = np.zeros((m, n))
    for k in range(m):
        G[k] = grad_corr_k(h, k, n)
    return G


def lp_step_exact(h, dx, target_sum, n_active, alpha):
    n = len(h)
    m_total = 2 * n - 1
    corrs = compute_corrs(h)
    current_c5 = corrs.max() * dx

    if n_active >= m_total:
        active_idx = np.arange(m_total)
    else:

    G = np.array([grad_corr_k(h, k, n) for k in active_idx])
    m = len(active_idx)

    c_obj[-1] = 1.0
    A_ub = np.empty((m, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -corrs[active_idx]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)

    bounds = []
    for i in range(n):
        bounds.append((-h[i] * alpha, (1.0 - h[i]) * alpha))
    bounds.append((-np.inf, np.inf))

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status != 0:
            return h, current_c5, False
        delta = res.x[:n]
    except Exception:
        return h, current_c5, False

    if np.abs(delta).max() < 1e-15:
        return h, current_c5, False

    # Find max step s such that h + s*delta stays in [0,1]
    s_max = 1.0
    pos = delta > 1e-14
    neg = delta < -1e-14
    if pos.any():

    if neg.any():
        s_max = min(s_max, np.min(-h[neg] / delta[neg]))
    s_max = max(s_max, 1e-12)

        return c5_of_h(h_try, dx)

    best_h = h.copy()

    alpha = alpha_start
    no_improve = 0
    it = 0

        pass
    return h, c5_init


def make_con_fun(ki, n, dx):
    def fun(xv):
        hv = xv[1:]
        offset = ki - (n - 1)
        i_lo = max(0, -offset)
        i_hi = min(n - 1, n - 1 - offset)
        if i_lo > i_hi:
            return xv[0]
        idx = np.arange(i_lo, i_hi + 1)
        cv = float(np.sum(hv[idx] * (1.0 - hv[idx + offset])))
        return xv[0] - dx * cv
    return fun


def make_con_jac(ki, n, dx):
    def jac(xv):
        hv = xv[1:]
        g = np.zeros(n + 1)
        g[0] = 1.0
        g[1:] = -dx * grad_corr_k(hv, ki, n)
        return g
    return jac


def make_eq_fun(n):
    def fun(xv):
        return float(np.sum(xv[1:])) - n / 2.0
    return fun


def make_eq_jac(n):
    def jac(xv):
        g = np.zeros(n + 1)
        g[1:] = 1.0
        return g
    return jac


def fw_step(h, dx, target_sum):
    n = len(h)
    corrs = compute_corrs(h)
    c5_init = corrs.max() * dx
    k_max = int(np.argmax(corrs))
    grad = grad_corr_k(h, k_max, n) * dx
    # FW subproblem: min grad . v s.t. sum(v)=n/2, 0<=v<=1
    # Solution: v_i = 1 for n/2 smallest grad components
    sorted_idx = np.argsort(grad)
    v = np.zeros(n)
    v[sorted_idx[:n // 2]] = 1.0
    direction = v - h
    # Line search
    s_max = 1.0
    pos = direction > 1e-14
    neg = direction < -1e-14
    if pos.any():
        s_max = min(s_max, np.min((1.0 - h[pos]) / direction[pos]))
    if neg.any():
        s_max = min(s_max, np.min(-h[neg] / direction[neg]))
    s_max = max(s_max, 0.0)

    def c5_at_s(s):
        return c5_of_h(h + s * direction, dx)

    try:
        result = minimize_scalar(c5_at_s, bounds=(0.0, s_max), method='bounded',
                                 options={'maxiter': 30})
        s_opt = float(np.clip(result.x, 0.0, s_max))
    except Exception:
        s_opt = 2.0 / 3.0 * s_max
    h_new = project_feasible(h + s_opt * direction, target_sum)
    c5_new = c5_of_h(h_new, dx)
    return h_new, c5_new


def perturb_h(h, target_sum, sigma, rng):
    h_pert = h + sigma * rng.randn(len(h))
    return project_feasible(h_pert, target_sum)


def resize_h(h_from, n_from, n_to, target_to):
    x_from = (np.arange(n_from) + 0.5) / n_from
    x_to = (np.arange(n_to) + 0.5) / n_to
    h_interp = np.interp(x_to, x_from, h_from)
    return project_feasible(h_interp, target_to)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s

    # Load parent
    try:
        h0 = np.array(initial_h_values, dtype=float).flatten()
    except Exception:
        h0 = np.full(126, 0.5)
    n = len(h0)
    dx = 2.0 / n
    target_sum = n / 2.0
    h0 = project_feasible(h0, target_sum)

    best_h = h0.copy()
    best_c5 = c5_of_h(h0, dx)

    # Phase 1: LP optimization with various configs (40% of time)
    t_phase1 = t0 + budget_s * 0.40
    configs = [
        (0.30, 0.50, 0.985, 120),
        (0.20, 0.60, 0.99, 150),
        (0.15, 0.55, 0.98, 100),
        (0.25, 0.65, 0.985, 130),
        (0.10, 0.70, 0.99, 160),
        (0.20, 0.50, 0.988, 140),
    ]
    for ci, (alpha_s, frac, decay, max_ni) in enumerate(configs):
        if time.time() > t_phase1 - 5:
            break
        dt = t_phase1 - 5 - time.time()
        if dt < 10:
            break
        h_opt, c5_opt = lp_optimize(best_h, dx, target_sum, t0,
                                    time.time() + dt,
                                    alpha_start=alpha_s, n_active_frac=frac,
                                    alpha_decay=decay, min_alpha=0.001,
                                    max_no_improve=max_ni)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 2: SLSQP from best (15% of time)
    t_phase2 = t0 + budget_s * 0.55
    slsqp_configs = [(40, 100), (50, 80), (30, 120), (60, 60)]
    for n_c, mi in slsqp_configs:
        if time.time() > t_phase2 - 3:
            break
        dt = t_phase2 - 3 - time.time()
        if dt < 8:
            break
        h_slq, c5_slq = slsqp_step(best_h, dx, target_sum, n_cons=n_c, maxiter=mi)
        if c5_slq < best_c5:
            best_c5 = c5_slq
            best_h = h_slq.copy()
        # Follow up with LP polish
        if time.time() < t_phase2 - 5:
            h_opt, c5_opt = lp_optimize(best_h, dx, target_sum, t0,
                                        min(time.time() + 20, t_phase2 - 5),
                                        alpha_start=0.2, n_active_frac=0.6,
                                        alpha_decay=0.99, min_alpha=0.002,
                                        max_no_improve=60)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()

    # Phase 3: Perturb and re-optimize (15% of time)
    t_phase3 = t0 + budget_s * 0.70
    sigmas = [0.05, 0.03, 0.08, 0.02, 0.04, 0.06]
    for s_val in sigmas:
        if time.time() > t_phase3 - 10:
            break
        dt = t_phase3 - 10 - time.time()
        if dt < 8:
            break
        h_pert = perturb_h(best_h, target_sum, s_val, rng)
        h_opt, c5_opt = lp_optimize(h_pert, dx, target_sum, t0,
                                    time.time() + dt,
                                    alpha_start=0.2, n_active_frac=0.55,
                                    alpha_decay=0.985, min_alpha=0.002,
                                    max_no_improve=80)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 4: FW + LP hybrid (10% of time)
    t_phase4 = t0 + budget_s * 0.80
    h_cur = best_h.copy()
    c5_cur = best_c5
    for fw_it in range(200):
        if time.time() > t_phase4 - 5:
            break
        h_fw, c5_fw = fw_step(h_cur, dx, target_sum)
        if c5_fw < c5_cur - 1e-13:
            h_cur = h_fw
            c5_cur = c5_fw
        else:
            # Short LP polish
            if time.time() < t_phase4 - 3:
                h_opt, c5_opt = lp_optimize(h_cur, dx, target_sum, t0,
                                            time.time() + 3,
                                            alpha_start=0.15, n_active_frac=0.7,
                                            alpha_decay=0.99, min_alpha=0.003,
                                            max_no_improve=30)
                if c5_opt < c5_cur:
                    h_cur = h_opt
                    c5_cur = c5_opt
                else:
                    break
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h_cur.copy()

    # Phase 5: Resolution scan (10% of time)
    t_phase5 = t0 + budget_s * 0.90
    n_candidates = [n - 4, n - 2, n + 2, n + 4, n - 8, n + 8, 128, 132, 120]
    for n_try in n_candidates:
        if time.time() > t_phase5 - 15:
            break
        if n_try <= 0 or n_try == n or n_try > 200:
            continue
        dt = t_phase5 - 15 - time.time()
        if dt < 12:
            break
        dx_try = 2.0 / n_try
        ts_try = n_try / 2.0
        h_interp = resize_h(best_h, n, n_try, ts_try)
        c5_interp = c5_of_h(h_interp, dx_try)
        if c5_interp > best_c5 + 0.005:
            continue
        h_opt, c5_opt = lp_optimize(h_interp, dx_try, ts_try, t0,
                                    time.time() + dt,
                                    alpha_start=0.2, n_active_frac=0.55,
                                    alpha_decay=0.985, min_alpha=0.002,
                                    max_no_improve=80)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            n = n_try
            dx = dx_try
            target_sum = ts_try

    # Phase 6: Final intensive polish (10% of time)
    t_final = deadline - 3
    n = len(best_h)
    dx = 2.0 / n
    target_sum = n / 2.0
    m_total = 2 * n - 1

    # Large active set LP
    final_configs = [(0.1, 0.8, 0.995, 100), (0.05, 0.9, 0.997, 150), (0.08, 0.85, 0.993, 120)]
    for alpha_f, frac_f, decay_f, ni_f in final_configs:
        if time.time() > t_final - 5:
            break
        dt = t_final - 5 - time.time()
        if dt < 5:
            break
        h_opt, c5_opt = lp_optimize(best_h, dx, target_sum, t0,
                                    time.time() + dt,
                                    alpha_start=alpha_f, n_active_frac=frac_f,
                                    alpha_decay=decay_f, min_alpha=0.0005,
                                    max_no_improve=ni_f)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # All-constraints final LP
    if time.time() < t_final - 3:
        h_opt, c5_opt = lp_optimize(best_h, dx, target_sum, t0,
                                    t_final - 2,
                                    alpha_start=0.05, n_active_frac=1.0,
                                    alpha_decay=0.995, min_alpha=0.001,
                                    max_no_improve=60)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Tiny perturbation + polish
    if time.time() < t_final - 3:
        h_tiny = perturb_h(best_h, target_sum, 0.005, rng)
        h_opt, c5_opt = lp_optimize(h_tiny, dx, target_sum, t0,
                                    t_final - 1,
                                    alpha_start=0.1, n_active_frac=0.7,
                                    alpha_decay=0.99, min_alpha=0.002,
                                    max_no_improve=40)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Final validation and repair
    best_h = np.clip(best_h, 0.0, 1.0)
    n_final = len(best_h)
    ts_final = n_final / 2.0
    best_h = project_feasible(best_h, ts_final)
    dx_final = 2.0 / n_final
    best_c5 = c5_of_h(best_h, dx_final)

    return best_h, best_c5, n_final
