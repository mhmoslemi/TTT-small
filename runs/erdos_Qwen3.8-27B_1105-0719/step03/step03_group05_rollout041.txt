import numpy as np
from scipy.optimize import minimize
import time

def score_h(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def build_h_from_y(y):
    m = len(y)
    n = 2 * m
    h = np.empty(n)
    h[:m] = y
    h[m:] = 1.0 - y[::-1]
    return h

def y_from_h(h):
    n = len(h)
    m = n // 2
    y = 0.5 * (h[:m] + 1.0 - h[n-1:m-1:-1])
    return np.clip(y, 0.0, 1.0)

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx])
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-14:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmin(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = np.linspace(0, 2, n + 1)
    xt = np.linspace(0, 2, new_n + 1)
    h_interp = np.zeros(new_n)
    for k in range(new_n):
        lo_k, hi_k = xt[k], xt[k+1]
        total_mass = 0.0
        total_width = 0.0
        for j in range(n):
            lo_j, hi_j = xs[j], xs[j+1]
            overlap = min(hi_k, hi_j) - max(lo_k, lo_j)
            if overlap > 0:
                total_mass += h[j] * overlap
                total_width += overlap
        h_interp[k] = total_mass / total_width if total_width > 1e-15 else 0.5
    h_interp = np.clip(h_interp, 0.0, 1.0)
    return repair_feasible(h_interp)

def fine_line_search_single(y, j, n_coarse=35, n_fine=18):
    c_cur = score_h(build_h_from_y(y))
    best_c = c_cur
    best_t = y[j]
    y_trial = y.copy()
    t_vals = np.linspace(0.0, 1.0, n_coarse)
    for t in t_vals:
        y_trial[j] = t
        c = score_h(build_h_from_y(y_trial))
        if c < best_c:
            best_c = c
            best_t = t
    step = 1.0 / (n_coarse - 1) if n_coarse > 1 else 0.05
    lo = max(0.0, best_t - 2.5 * step)
    hi = min(1.0, best_t + 2.5 * step)
    if hi - lo > 1e-10:
        t_fine = np.linspace(lo, hi, n_fine)
        for t in t_fine:
            y_trial[j] = t
            c = score_h(build_h_from_y(y_trial))
            if c < best_c:
                best_c = c
                best_t = t
    y[j] = best_t
    return y, best_c

def fine_line_search_pair(y, i, j, n_coarse=25, n_fine=15):
    t_min = max(-y[i], y[j] - 1.0)
    t_max = min(1.0 - y[i], y[j])
    if t_max - t_min < 1e-12:
        return y, score_h(build_h_from_y(y))
    c_cur = score_h(build_h_from_y(y))
    best_c = c_cur
    best_t = 0.0
    y_trial = y.copy()
    t_vals = np.linspace(t_min, t_max, n_coarse)
    for t in t_vals:
        y_trial[i] = y[i] + t
        y_trial[j] = y[j] - t
        c = score_h(build_h_from_y(y_trial))
        if c < best_c:
            best_c = c
            best_t = t
    span = t_max - t_min
    margin = 2.5 * span / n_coarse
    lo = max(t_min, best_t - margin)
    hi = min(t_max, best_t + margin)
    if hi - lo > 1e-10:
        t_fine = np.linspace(lo, hi, n_fine)
        for t in t_fine:
            y_trial[i] = y[i] + t

            c = score_h(build_h_from_y(y_trial))
            if c < best_c:
                best_c = c
                best_t = t
    y[i] = y[i] + best_t
    y[j] = y[j] - best_t
    return y, best_c

def coord_descent(y, max_sweeps=50, n_coarse=35, n_fine=18, time_fn=None):

    best_c = score_h(build_h_from_y(y))
    for sweep in range(max_sweeps):

        improved = False

                best_c = c_new

def pair_descent(y, n_pairs=2000, n_coarse=25, n_fine=15, rng=None, time_fn=None):
    m = len(y)
    if rng is None:
        rng = np.random.RandomState(0)

    return y, best_c

def compute_gradient_h(h, weights):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    M = 2 * n - 1
    for q in range(M):
        w = weights[q]
        if w < 1e-10:
            continue
        d = q - (n - 1)
        if d == 0:
            g += w * dx * (1.0 - 2.0 * h)
        elif d > 0:
            r = d
            g[r:] += w * dx * (1.0 - h[:n-r])
            g[:n-r] += w * dx * (-h[r:])
        else:
            r = -d
            g[:n-r] += w * dx * (1.0 - h[r:])
            g[r:] += w * dx * (-h[:n-r])
    return g

def lse_descent_full(h, beta=80.0, max_iters=100, alpha_init=0.005, time_fn=None):
    n = len(h)
    h = h.copy()
    best_c = score_h(h)
    alpha = alpha_init
    stuck = 0
    for it in range(max_iters):
        if time_fn is not None and time_fn() < 15:
            break
        profile = np.correlate(h, 1 - h, mode="full") * (2.0 / n)
        scaled = beta * profile
        shifted = scaled - np.max(scaled)
        weights = np.exp(shifted)
        wsum = weights.sum()
        if wsum < 1e-300:
            break
        weights /= wsum
        g = compute_gradient_h(h, weights)
        g -= g.mean()
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-14:
            break
        g /= gnorm
        improved = False
        for a in [alpha, alpha*0.5, alpha*0.25, alpha*0.1]:
            h_try = repair_feasible(h - a * g)
            s = score_h(h_try)
            if s < best_c - 1e-14:
                h = h_try
                best_c = s
                improved = True
                alpha = min(alpha * 1.08, 0.025)
                stuck = 0
                break
        if not improved:
            alpha *= 0.5
            stuck += 1
            if stuck > 25 or alpha < 1e-10:
                break
    return h, best_c

        n_b = len(best_h)
        # Multiple perturbation scales
        for eps in [0.003, 0.008, 0.015, 0.03, 0.06]:
            if time_left() < 25:
                break
            z = rng.randn(n_b)
            z -= z.mean()
            h_pert = repair_feasible(best_h + eps * z)
            # Quick LSE
            h_opt, c_opt = lse_descent_full(h_pert, beta=60, max_iters=50, time_fn=time_left)
            if c_opt < best_c - 1e-14:
                best_h = h_opt
                best_c = c_opt
            # Active set
            h_as2, c_as2 = active_set_descent_full(best_h, max_iters=60, n_active=8, time_fn=time_left)
            if c_as2 < best_c - 1e-14:
                best_h = h_as2
                best_c = c_as2

    # Phase 6: Re-project to anti-symmetric if best is there, do more search
    if time_left() > 80:
        y = y_from_h(best_h)
        y, c_anti2 = anti_search(y, time_left, rng, n_coarse=45, n_fine=22, n_pairs=3000)
        h_cand = build_h_from_y(y)
        c_cand = score_h(h_cand)
        if c_cand < best_c - 1e-14:
            best_h = h_cand
            best_c = c_cand

    # Phase 7: Targeted LSE with different betas
    if time_left() > 50:
        for beta in [40, 120, 200]:
            if time_left() < 15:
                break
            h_lse, c_lse = lse_descent_full(best_h, beta=beta, max_iters=50, alpha_init=0.003, time_fn=time_left)
            if c_lse < best_c - 1e-14:
                best_h = h_lse
                best_c = c_lse

    # Phase 8: Active set with different active set sizes
    if time_left() > 40:
        for na in [6, 15, 20]:
            if time_left() < 15:
                break
            h_as, c_as = active_set_descent_full(best_h, max_iters=80, n_active=na, time_fn=time_left)
            if c_as < best_c - 1e-14:
                best_h = h_as
                best_c = c_as

    # Phase 9: Final resolution fine-tuning
    if time_left() > 50:
        n_fin = len(best_h)
        for dn in [-2, -4, 2, 4]:
            if time_left() < 15:
                break
            nn = n_fin + dn
            if nn < 24 or nn > 140 or nn % 2 != 0:
                continue
            h_i = interpolate_to_n(best_h, nn)
            y_i = y_from_h(h_i)
            y_i, c_i = coord_descent(y_i, max_sweeps=30, n_coarse=40, n_fine=20, time_fn=time_left)
            h_cand = build_h_from_y(y_i)
            c_cand = score_h(h_cand)
            if time_left() > 10:
                h_lse, c_lse = lse_descent_full(h_cand, beta=80, max_iters=30, time_fn=time_left)
                if c_lse < c_cand:
                    h_cand = h_lse
                    c_cand = c_lse
            if c_cand < best_c - 1e-14:
                best_h = h_cand
                best_c = c_cand

    # Phase 10: Final polish
    if time_left() > 20:
        h_lse, c_lse = lse_descent_full(best_h, beta=100, max_iters=60, alpha_init=0.002, time_fn=time_left)
        if c_lse < best_c - 1e-14:
            best_h = h_lse
            best_c = c_lse
        h_as, c_as = active_set_descent_full(best_h, max_iters=60, n_active=10, alpha_init=0.002, time_fn=time_left)
        if c_as < best_c - 1e-14:
            best_h = h_as
            best_c = c_as

    # Final validation
    best_h = repair_feasible(best_h)
    final_c = score_h(best_h)

    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            final_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 2:
            final_c = float(eval_result[0])
    except Exception:
        pass

    return best_h, final_c, len(best_h)
