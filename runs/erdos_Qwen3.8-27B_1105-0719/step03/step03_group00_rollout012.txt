import time

def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h = np.clip(h + diff / mask.sum(), 0.0, 1.0)
            # Re-project if needed
            h2 = project_feasible_simple(h, target_sum)
            if h2 is not None:
                h = h2
    return np.clip(h, 0.0, 1.0)

def project_feasible_simple(x, target_sum):
    n = len(x)
    # Simple: if sum is close, distribute difference
    diff = target_sum - x.sum()
    if abs(diff) < 1e-12:
        return np.clip(x, 0, 1)
    mask = (x > 1e-10) & (x < 1.0 - 1e-10)
    if mask.sum() == 0:
        return None
    x_new = x.copy()
    x_new[mask] = np.clip(x_new[mask] + diff / mask.sum(), 0, 1)
    return x_new

    return np.correlate(h, 1.0 - h, mode='full').max() * dx

    n = len(h)
    t_old = h[i]
    corr = np.correlate(h, 1.0 - h, mode='full')
    offsets = np.arange(2*n-1) - (n-1)

    i_minus = i - offsets
    valid_plus = (i_plus >= 0) & (i_plus < n)
    valid_minus = (i_minus >= 0) & (i_minus < n)
    
    slopes = np.zeros(2*n-1)
    slopes[valid_plus] += 1.0 - h[i_plus[valid_plus]]
    slopes[valid_minus] -= h[i_minus[valid_minus]]
    
    # f(t) = max_k (corr[k] + (t - t_old) * slopes[k])
    # Evaluate at boundaries
    f_0 = np.max(corr - t_old * slopes)
    f_1 = np.max(corr + (1.0 - t_old) * slopes)

    grid_offsets = [-20, -16, -12, -8, -4, 4, 8, 12, 16, 20, 24, -24]
    for dn in grid_offsets:
        if elapsed() > budget_s * 0.82:
            break
        n_try = n_base + dn
        if n_try < 40 or n_try > 256:
            continue
        if n_try % 2 != 0:
            n_try += 1
        if n_try < 40 or n_try > 256:
            continue
        h_t = transfer_grid(best_h, n_try)
        h_t = coord_sweep(h_t, n_try / 2.0, n_top=40, max_sweeps=30)
        c5_t = c5_of_h(h_t)
        if c5_t < best_c5 - 1e-14:
            best_c5 = c5_t
            best_h = h_t
            n_base = n_try
    
    # Phase 6: Intensive final polish on best grid
    n_final = len(best_h)
    target_final = n_final / 2.0
    dx_final = 2.0 / n_final
    
    for r in range(200):
        if elapsed() > budget_s * 0.95:
            break
        order = rng.permutation(n_final)
        h_new = coord_sweep(best_h, target_final, n_top=50, order=order, max_sweeps=3)
        c5_new = c5_of_h(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new
    
    # Phase 7: One more subgradient + polish cycle
    if elapsed() < budget_s * 0.97:
        step = 0.03
        h_sg = subgradient_step(best_h, target_final, step, dx_final)
        h_sg = coord_sweep(h_sg, target_final, n_top=50, max_sweeps=20)
        c5_sg = c5_of_h(h_sg)
        if c5_sg < best_c5 - 1e-14:
            best_c5 = c5_sg
            best_h = h_sg
    
    # Final validation
    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_c5 = c5_of_h(best_h)
    
    return best_h, best_c5, len(best_h)
