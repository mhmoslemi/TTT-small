def repair_feasible(vec):
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0).copy()
    n = h.size
    target = 0.5 * n
    for _ in range(50):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        rem = abs(diff)
        for idx in order:
            if diff > 0:
                add = min(rem, 1.0 - h[idx])
                if add > 0: h[idx] += add
                rem -= add
            else:
                sub = min(rem, h[idx])
                if sub > 0: h[idx] -= sub
                rem -= sub
            if rem <= 1e-15:
                break
        h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-15:
        # last resort put remaining on one element if possible
        if diff > 0:
            idx = int(np.argmax(1.0 - h))
            h[idx] = min(1.0, h[idx] + diff)
        else:
            idx = int(np.argmax(h))
            h[idx] = max(0.0, h[idx] + diff)
        h = np.clip(h, 0.0, 1.0)
    return h

def solve_linprog(c, A_ub, b_ub, A_eq, b_eq, bounds):

def repair_feasible(vec):
    h = np.asarray(vec, dtype=np.float64).copy()

    n = h.size
    target = 0.5 * n
    for _ in range(60):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0.0:
            order = np.argsort(1.0 - h)

def interp(h, new_n):
    n = len(h)
    if new_n == n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    hn = np.interp(xt, xs, h)
    return repair_feasible(np.clip(hn, 0.0, 1.0))


def get_precomp(n):
    if n in _PRECACHE:
        return _PRECACHE[n]
    idx = np.arange(n)
    diff_mat = (idx[:, None] - idx[None, :]) + (n - 1)
    diff_flat = diff_mat.ravel()

    edges = []

        if s >= 0:

            u = v + s
        else:

            u = v + s
        edges.append((u, v))
    obj = (diff_flat, ia, ja, edges)
    _PRECACHE[n] = obj

def compute_grad_matrix(h, n, edges):
    M = 2 * n - 1
    G = np.zeros((M, n), dtype=np.float64)
    for m, (u, v) in enumerate(edges):
        if u.size:
            G[m, u] += 1.0 - h[v]
            G[m, v] -= h[u]
    return G


def normalize_direction(q):

    if maxabs < 1e-12:
        return None
    q = q / maxabs
    return q


def direction_from_gradient(g):
    if g is None:
        return None
    g = np.asarray(g, dtype=np.float64)
    if not np.isfinite(g).all():
        return None
    return normalize_direction(-g)


def softmax_weights(prof, tau):
    m = float(prof.max())
    if tau <= 1e-15:
        w = np.zeros_like(prof)
        top = np.where(prof >= m - 1e-12)[0]
        if top.size == 0:
            top = np.array([int(np.argmax(prof))], dtype=int)
        w[top] = 1.0 / top.size
        return w
    z = np.exp((prof - m) / tau)
    s = float(z.sum())
    if s <= 0.0 or not np.isfinite(s):
        w = np.zeros_like(prof)
        top = np.where(prof >= m - 1e-12)[0]
        if top.size == 0:
            top = np.array([int(np.argmax(prof))], dtype=int)
        w[top] = 1.0 / top.size
        return w
    return z / s


def margin_weights(prof, margin):
    m = float(prof.max())
    w = np.where(prof >= m - margin, 1.0, 0.0)
    s = float(w.sum())
    if s <= 0.0:
        w = np.zeros_like(prof)
        w[int(np.argmax(prof))] = 1.0
        return w
    return w / s


def cheby_direction(grads, n):
    grads = np.asarray(grads, dtype=np.float64)
    K = grads.shape[0]
    if K == 0:
        return None
    m = K + 2
    c = np.zeros(m)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n, m), dtype=np.float64)
    b_ub = np.zeros(2 * n, dtype=np.float64)
    row = 0
    for i in range(n):
        A_ub[row, :K] = grads[:, i]
        A_ub[row, K] = -1.0
        A_ub[row, K + 1] = -1.0
        row += 1
        A_ub[row, :K] = -grads[:, i]
        A_ub[row, K] = 1.0
        A_ub[row, K + 1] = -1.0
        row += 1
    A_eq = np.zeros((1, m), dtype=np.float64)
    A_eq[0, :K] = -grads.sum(axis=1)
    A_eq[0, K] = float(n)
    bounds = [(0.0, 1.0)] * K + [(-1e6, 1e6), (0.0, 1e6)]
    x = solve_linprog(c, A_ub, b_ub, A_eq, [0.0], bounds)
    if x is None:
        return None
    w = x[:K]
    if float(w.sum()) <= 0.0:
        return None
    g = grads.T @ w
    return direction_from_gradient(g)


def active_q_lp_direction(GA, n):
    GA = np.asarray(GA, dtype=np.float64)
    K = GA.shape[0]
    if K == 0:
        return None
    m = n + 1
    c = np.zeros(m)
    c[-1] = 1.0
    A_ub = np.zeros((K, m), dtype=np.float64)
    A_ub[:, :n] = GA
    A_ub[:, -1] = -1.0
    b_ub = np.zeros(K, dtype=np.float64)
    A_eq = np.zeros((1, m), dtype=np.float64)
    A_eq[0, :n] = 1.0
    b_eq = [0.0]
    bounds = [(-1.0, 1.0)] * n + [(-100.0, 1000.0)]
    x = solve_linprog(c, A_ub, b_ub, A_eq, b_eq, bounds)
    if x is None:
        return None
    if float(x[-1]) > -1e-7:
        return None
    return normalize_direction(x[:n])


def trust_lp_direction(h, raw_active, GA, trust, n):
    GA = np.asarray(GA, dtype=np.float64)
    raw_active = np.asarray(raw_active, dtype=np.float64)
    K = GA.shape[0]
    if K == 0:
        return None
    m = n + 1
    c = np.zeros(m)
    c[-1] = 1.0
    A_ub = np.zeros((K, m), dtype=np.float64)
    A_ub[:, :n] = GA
    A_ub[:, -1] = -1.0
    b_ub = GA @ h - raw_active
    A_eq = np.zeros((1, m), dtype=np.float64)
    A_eq[0, :n] = 1.0
    b_eq = [0.5 * n]
    lower = np.maximum(0.0, h - trust)
    upper = np.minimum(1.0, h + trust)
    bounds = []
    for i in range(n):
        bounds.append((float(lower[i]), float(upper[i])))
    bounds.append((-100.0, 1000.0))
    x = solve_linprog(c, A_ub, b_ub, A_eq, b_eq, bounds)
    if x is None:
        return None
    q = x[:n] - h
    return normalize_direction(q)


def exact_line_search(h, q, profile_raw, n, diff_flat, ia, ja):
    q = np.asarray(q, dtype=np.float64)
    if q.size != n or not np.isfinite(q).all():
        return 0.0
    q = q - np.mean(q)
    maxabs = float(np.max(np.abs(q)))
    if maxabs < 1e-14:
        return 0.0
    q = q / maxabs

    lo = -np.inf
    hi = np.inf
    pos = q > 1e-15
    if np.any(pos):
        hi = float(np.min((1.0 - h[pos]) / q[pos]))
    neg = q < -1e-15
    if np.any(neg):
        lo = float(np.max(h[neg] / (-q[neg])))
    if not np.isfinite(hi):
        hi = 10.0
    if not np.isfinite(lo):
        lo = -10.0
    hi = min(hi, 10.0)
    lo = max(lo, -10.0)
    lo = min(lo, 0.0)
    hi = max(hi, 0.0)
    if hi - lo < 1e-12:
        return 0.0
    eps = 1e-12
    lo = max(lo, -10.0 + eps)
    hi = min(hi, 10.0 - eps)

    M = 2 * n - 1
    Lmat = q[:, None] * (1.0 - h[None, :]) - h[:, None] * q[None, :]
    Qmat = -q[:, None] * q[None, :]
    L = np.bincount(diff_flat, weights=Lmat.ravel(), minlength=M)
    Q = np.bincount(diff_flat, weights=Qmat.ravel(), minlength=M)
    A = Q
    B = L
    C = np.asarray(profile_raw, dtype=np.float64)

    cands = [np.array([lo, hi, 0.0], dtype=np.float64)]

    vm = A > 1e-12
    if np.any(vm):
        vv = -B[vm] / (2.0 * A[vm])
        mask = np.isfinite(vv) & (vv >= lo - 1e-10) & (vv <= hi + 1e-10)
        if np.any(mask):
            cands.append(vv[mask])

    if ia.size > 0:
        Da = A[ia] - A[ja]
        Db = B[ia] - B[ja]
        Dc = C[ia] - C[ja]
        roots = []
        quad = np.abs(Da) > 1e-14
        if np.any(quad):
            Dqa = Da[quad]
            Dqb = Db[quad]
            Dqc = Dc[quad]
            disc = Dqb * Dqb - 4.0 * Dqa * Dqc
            valid = disc >= -1e-10
            if np.any(valid):
                discv = np.maximum(disc[valid], 0.0)
                sv = np.sqrt(discv)
                den = 2.0 * Dqa[valid]
                r1 = (-Dqb[valid] + sv) / den
                r2 = (-Dqb[valid] - sv) / den
                roots.append(r1)
                roots.append(r2)
        lin = (~quad) & (np.abs(Db) > 1e-14)
        if np.any(lin):
            roots.append(-Dc[lin] / Db[lin])
        if roots:
            r = np.concatenate(roots)
            mask = np.isfinite(r) & (r >= lo - 1e-10) & (r <= hi + 1e-10)
            r = r[mask]
            if r.size > 0:
                cands.append(r)

    c = np.concatenate(cands)
    mask = np.isfinite(c) & (c >= lo - 1e-9) & (c <= hi + 1e-9)
    c = c[mask]
    if c.size == 0:
        return 0.0
    c = np.clip(c, lo, hi)
    if c.size > 5000:
        c = np.unique(c)
    max_cand = 40000 if n <= 85 else 25000
    if c.size > max_cand:
        sel = np.linspace(0, c.size - 1, max_cand).astype(np.int64)
        c = c[sel]

    best_alpha = 0.0
    best_val = float("inf")
    chunk = 4096
    for start in range(0, c.size, chunk):
        cc = c[start:start + chunk]
        P = A[:, None] * (cc[None, :] ** 2) + B[:, None] * cc[None, :] + C[:, None]
        vals = P.max(axis=0)
        mi = int(np.argmin(vals))
        if vals[mi] < best_val:
            best_val = float(vals[mi])
            best_alpha = float(cc[mi])
    return best_alpha


def random_direction(n, rng):
    z = rng.randn(n)
    return normalize_direction(z)


def fourier_direction(n, p, use_cos):
    x = (np.arange(n) + 0.5) / float(n)
    if use_cos:
        q = np.cos(2.0 * np.pi * p * x)
    else:
        q = np.sin(2.0 * np.pi * p * x)
    return normalize_direction(q)


def transform_best(h):
    h = repair_feasible(h)
    best = h
    best_c = exact_score(h)
    hr = h[::-1].copy()
    cr = exact_score(hr)
    if cr < best_c - 1e-14:
        best = hr
        best_c = cr
    n = len(h)
    for s in range(1, n // 2 + 1):
        ht = np.roll(h, s)
        ct = exact_score(ht)
        if ct < best_c - 1e-14:
            best = ht
            best_c = ct
    return best


def better_solution(c, n, bc, bn):
    tol = 1e-12
    if c < bc - tol:
        return True
    if abs(c - bc) <= tol and n < bn:
        return True
    return False


def random_balanced_binary(n, rng):
    h = np.zeros(n, dtype=np.float64)
    if n > 0:
        idx = np.arange(n)
        rng.shuffle(idx)
        h[idx[:n // 2]] = 1.0
    return h


def binary_scout(h0, limit_s, t_start, budget_s, rng, min_left=10.0):
    h = repair_feasible(h0)
    n = len(h)
    target = 0.5 * n
    if abs(target - round(target)) > 1e-9:
        return h, exact_score(h)
    h = np.zeros(n, dtype=np.float64)
    idx = np.argsort(-h0)
    h[idx[:n // 2]] = 1.0
    h = repair_feasible(h)
    if int(np.count_nonzero(h > 0.5)) != n // 2:
        h = repair_feasible(random_balanced_binary(n, rng))
    ones = list(np.where(h > 0.5)[0])
    zeros = list(np.where(h < 0.5)[0])
    if len(ones) == 0 or len(zeros) == 0:
        return h, exact_score(h)
    best_c = exact_score(h)
    start = time.perf_counter()
    while (time.perf_counter() - start) < limit_s and time_left(t_start, budget_s) > min_left:
        if len(ones) == 0 or len(zeros) == 0:
            break
        ai = int(rng.randint(len(ones)))
        aj = int(rng.randint(len(zeros)))
        i = ones[ai]
        j = zeros[aj]
        ones[ai] = ones[-1]
        ones.pop()
        zeros[aj] = zeros[-1]
        zeros.pop()
        h2 = h.copy()
        h2[i] = 0.0
        h2[j] = 1.0
        sc = exact_score(h2)
        if sc < best_c - 1e-10:
            h = h2
            best_c = sc
            zeros.append(i)
            ones.append(j)
        else:
            ones.append(i)
            zeros.append(j)
    return h, best_c


def local_improve(h0, limit_s, t_start, budget_s, rng, min_left=20.0, rich=True):
    h = transform_best(h0)
    best_c = exact_score(h)
    n = len(h)
    dx = 2.0 / n
    diff_flat, ia, ja, edges = get_precomp(n)
    raw = raw_profile(h)
    prof = raw * dx
    start = time.perf_counter()
    it = 0
    noimp = 0

    while (time.perf_counter() - start) < limit_s and time_left(t_start, budget_s) > min_left:
        it += 1
        G = compute_grad_matrix(h, n, edges)
        dirs = []
        order = np.argsort(-prof)

        if rich:
            k_list = (1, 2, 3, 5, 8, 12, 16)
            taus = (1e-4, 1e-3, 5e-3, 2e-2, 0.1)
        else:
            k_list = (1, 2, 4, 8)
            taus = (1e-3, 2e-2)

        active_sets = []
        for K in k_list:
            if K < prof.size:
                active_sets.append(order[:K])

        if rich and (it % 2 == 1):
            for K in (2, 4, 8):
                if K < prof.size:
                    q = active_q_lp_direction(G[order[:K]], n)
                    if q is not None:
                        dirs.append(q)

        for active in active_sets[:4]:
            q = cheby_direction(G[active], n)
            if q is not None:
                dirs.append(q)

        if rich and (it % 3 == 2) and n <= 95:
            for K in (4, 8):
                if K < prof.size:
                    q = trust_lp_direction(h, raw[order[:K]], G[order[:K]], 0.03, n)
                    if q is not None:
                        dirs.append(q)
            q = trust_lp_direction(h, raw[order], G[order], 0.02, n)
            if q is not None:
                dirs.append(q)

        for tau in taus:
            w = softmax_weights(prof, tau)
            g = (G * w[:, None]).sum(axis=0)
            q = direction_from_gradient(g)
            if q is not None:
                dirs.append(q)

        for margin in (0.002, 0.01):
            w = margin_weights(prof, margin)
            g = (G * w[:, None]).sum(axis=0)
            q = direction_from_gradient(g)
            if q is not None:
                dirs.append(q)

        w = softmax_weights(prof, 0.005)
        g = (G * w[:, None]).sum(axis=0)
        high = np.argsort(g)[-6:][::-1]
        low = np.argsort(g)[:6]
        for hi_i in high[:4]:
            for lo_j in low[:4]:
                if int(hi_i) != int(lo_j):
                    q = np.zeros(n)
                    q[hi_i] = -1.0
                    q[lo_j] = 1.0
                    q = normalize_direction(q)
                    if q is not None:
                        dirs.append(q)

        q = normalize_direction(0.5 - h)
        if q is not None:
            dirs.append(q)
        for th in (0.2, 0.5, 0.8):
            q = normalize_direction((h >= th).astype(np.float64) - h)
            if q is not None:
                dirs.append(q)
        for s in (1, 2, 3):
            q = normalize_direction(np.roll(h, s) - h)
            if q is not None:
                dirs.append(q)

        if it % 2 == 0:
            for p in (1, 2, 3, 5):
                q = fourier_direction(n, p, False)
                if q is not None:
                    dirs.append(q)
        for _ in range(2):
            q = random_direction(n, rng)
            if q is not None:
                dirs.append(q)

        max_dirs = 36 if (rich and n <= 90) else 22
        eval_count = 0
        best_trial = None
        best_trial_c = best_c

        for q in dirs:
            if eval_count >= max_dirs:
                break
            if time.perf_counter() - start >= limit_s:
                break
            if time_left(t_start, budget_s) <= min_left:
                break
            q = normalize_direction(q)
            if q is None:
                continue
            a = exact_line_search(h, q, raw, n, diff_flat, ia, ja)
            if a is None or abs(a) < 1e-12:
                continue
            eval_count += 1
            h_try = repair_feasible(h + a * q)
            sc = exact_score(h_try)
            if sc < best_trial_c - 1e-10:
                best_trial_c = sc
                best_trial = h_try

        if best_trial is not None:
            h = best_trial
            best_c = best_trial_c
            noimp = 0
        else:
            improved = False
            for eps in (0.02, 0.05, 0.10, 0.20):
                z = rng.randn(n)
                z = z - z.mean()
                z = z / (np.linalg.norm(z) / n + 1e-15)
                hp = repair_feasible(h + eps * z)
                cp = exact_score(hp)
                if cp < best_c - 1e-9:
                    h = hp
                    best_c = cp
                    improved = True
                    noimp = 0
                    break
            if not improved:
                ht = transform_best(h)
                ct = exact_score(ht)
                if ct < best_c - 1e-9:
                    h = ht
                    best_c = ct
                    improved = True
                    noimp = 0
            if not improved:
                noimp += 1
                if noimp >= (4 if rich else 2):
                    break

    return h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()
    budget_s = float(budget_s)
    rng = np.random.RandomState(seed)

    parent = None
    try:
        parent = np.asarray(initial_h_values, dtype=np.float64)
    except Exception:
        parent = None
    if parent is None or parent.size < 4:
        parent = np.full(70, 0.5, dtype=np.float64)

    best_h = repair_feasible(parent.copy())
    best_c = exact_score(best_h)
    best_n = int(len(best_h))

    ht = transform_best(best_h)
    ct = exact_score(ht)
    if better_solution(ct, int(len(ht)), best_c, best_n):
        best_h, best_c, best_n = ht, ct, int(len(ht))

    if time_left(t_start, budget_s) > 180:
        parent_limit = min(280.0, max(30.0, time_left(t_start, budget_s) - 700.0))
        h1, c1 = local_improve(best_h, parent_limit, t_start, budget_s, rng, min_left=20.0, rich=True)
        if better_solution(c1, int(len(h1)), best_c, best_n):
            best_h, best_c, best_n = h1, c1, int(len(h1))

    if time_left(t_start, budget_s) > 350:
        hbin = random_balanced_binary(best_n, rng)
        h1, c1 = local_improve(hbin, 30.0, t_start, budget_s, rng, min_left=10.0, rich=False)
        if better_solution(c1, int(len(h1)), best_c, best_n):
            best_h, best_c, best_n = h1, c1, int(len(h1))

    dims = list(range(24, 113, 2))
    if best_n not in dims:
        dims.append(best_n)
    dim_keys = []
    for d in dims:
        dim_keys.append((abs(int(d) - int(best_n)), int(d)))
    dim_keys.sort()

    for _, d in dim_keys:
        if int(d) == int(best_n):
            continue
        if time_left(t_start, budget_s) < 80:
            break
        if time_left(t_start, budget_s) > 500:
            alloc = 25.0
        else:
            alloc = 12.0
        if time_left(t_start, budget_s) < alloc + 40:
            alloc = max(10.0, time_left(t_start, budget_s) - 40)
        h0 = interp(best_h, int(d))
        h1, c1 = local_improve(h0, alloc, t_start, budget_s, rng, min_left=8.0, rich=(alloc > 15.0))
        if better_solution(c1, int(len(h1)), best_c, best_n):
            best_h, best_c, best_n = h1, c1, int(len(h1))

        if int(d) in (40, 60, 80, 100) and time_left(t_start, budget_s) > 160:
            hbin = random_balanced_binary(int(d), rng)
            hb, cb = binary_scout(hbin, 8.0, t_start, budget_s, rng, min_left=6.0)
            if better_solution(cb, int(len(hb)), best_c, best_n):
                best_h, best_c, best_n = hb, cb, int(len(hb))
            if time_left(t_start, budget_s) > 120:
                h1, c1 = local_improve(hb, 25.0, t_start, budget_s, rng, min_left=8.0, rich=False)
                if better_solution(c1, int(len(h1)), best_c, best_n):
                    best_h, best_c, best_n = h1, c1, int(len(h1))

    for eps in (0.02, 0.05, 0.10):
        if time_left(t_start, budget_s) < 90:
            break
        z = rng.randn(best_h.size)
        z = z - z.mean()
        z = z / (np.linalg.norm(z) / best_h.size + 1e-15)
        hp = repair_feasible(best_h + eps * z)
        p_limit = min(40.0, max(10.0, time_left(t_start, budget_s) - 60.0))
        h1, c1 = local_improve(hp, p_limit, t_start, budget_s, rng, min_left=15.0, rich=True)
        if better_solution(c1, int(len(h1)), best_c, best_n):
            best_h, best_c, best_n = h1, c1, int(len(h1))

    if time_left(t_start, budget_s) > 50:
        final_limit = min(150.0, max(10.0, time_left(t_start, budget_s) - 20.0))
        h1, c1 = local_improve(best_h, final_limit, t_start, budget_s, rng, min_left=8.0, rich=True)
        if better_solution(c1, int(len(h1)), best_c, best_n):
            best_h, best_c, best_n = h1, c1, int(len(h1))

    ht = transform_best(best_h)
    ct = exact_score(ht)
    if better_solution(ct, int(len(ht)), best_c, best_n):
        best_h, best_c, best_n = ht, ct, int(len(ht))

    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)
    return best_h, float(final_c), int(len(best_h))
