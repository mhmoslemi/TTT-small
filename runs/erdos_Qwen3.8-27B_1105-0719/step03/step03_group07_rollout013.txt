```python
def lp_leveling_direction(h, active_offsets, n, dx):
    """Find a direction v that decreases all active overlaps simultaneously."""
    from scipy.optimize import linprog
    
    # Build gradient matrix for active offsets
    G = np.zeros((len(active_offsets), n))
    for idx, d in enumerate(active_offsets):
        G[idx] = build_gradient_row(h, d, n, dx)
    
    # LP variables: v[0..n-1], t
    # Maximize t
    # Subject to: G @ v + t <= 0  (i.e., G @ v <= -t)
    #             sum(v) = 0
    #             -1 <= v_i <= 1
    #             t >= 0
    
    c_obj = np.zeros(n + 1)
    c_obj[-1] = -1.0  # minimize -t = maximize t
    
    A_ub = np.zeros((len(active_offsets), n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = 1.0  # G @ v + t <= 0
    b_ub = np.zeros(len(active_offsets))
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    bounds = [(-1.0, 1.0)] * n + [(0.0, 10.0)]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.status == 0 and res.x is not None:
        t_val = res.x[-1]
        if t_val > 0.01:  # meaningful direction
            return res.x[:n], t_val
    return None, 0.0
```

```python
def pair_line_search(h, i, j):
    """Optimize the split between h[i] and h[j] keeping their sum fixed."""
    s = h[i] + h[j]
    lo = max(0.0, s - 1.0)
    hi = min(1.0, s)
    if hi - lo < 1e-15:
        return h
    
    # Sample many points + use scipy minimize
    best_t = (lo + hi) / 2.0
    best_score = exact_score(h)
    
    n_samples = 20
    candidates = np.linspace(lo, hi, n_samples)
    # Also add points near current values
    candidates = np.concatenate([candidates, [h[i], h[i] + 0.001, h[i] - 0.001,
                                                h[i] + 0.01, h[i] - 0.01]])
    candidates = np.clip(candidates, lo, hi)
    candidates = np.unique(candidates)
    
    for t in candidates:
        h_temp = h.copy()
        h_temp[i] = t
        h_temp[j] = s - t
        score = exact_score(h_temp)
        if score < best_score:
            best_score = score
            best_t = t
    
    # Refine with golden section
    a, b = lo, hi
    gr = (np.sqrt(5) - 1) / 2
    for _ in range(30):
        c = b - gr * (b - a)
        d = a + gr * (b - a)
        h_c = h.copy(); h_c[i] = c; h_c[j] = s - c
        h_d = h.copy(); h_d[i] = d; h_d[j] = s - d
        if exact_score(h_c) < exact_score(h_d):
            b = d
        else:
            a = c
    t = (a + b) / 2
    h_temp = h.copy()
    h_temp[i] = t
    h_temp[j] = s - t
    if exact_score(h_temp) < best_score:
        best_score = exact_score(h_temp)
        best_t = t
    
    h[i] = best_t
    h[j] = s - best_t
    return h, best_score
```

Let me also think about the structure of the optimal solution. For the Erdős problem, the optimal h is expected to be "as uniform as possible" while maintaining the constraint. The constraint sum(h) = n/2 means the average value is 0.5. So the optimal solution is close to h = 0.5 everywhere, but with some structure that reduces the maximum overlap.

Actually, at h = 0.5 everywhere:
P_d = dx * sum (0.5)(0.5) = dx * 0.25 * (n - |d|) for lag d
The max is at d=0: P_0 = dx * 0.25 * n = (2/n) * 0.25 * n = 0.5

So the uniform solution gives C_5 = 0.5, which is much worse than 0.38. The structure must be quite different.

For a binary solution (h ∈ {0,1}), with exactly n/2 ones:
P_d = dx * sum h[i+d](1-h[i]) = dx * (number of i where h[i]=1 and h[i+d]=0) for d>0
    = dx * (number of "1→0" transitions at lag d)

For a random binary pattern with n/2 ones:
E[P_d] ≈ dx * (n/2) * (n/2) / n = dx * n/4 = (2/n) * n/4 = 0.5

But we can do much better with structured patterns. The key is to minimize the maximum over all lags.

For a regular pattern (e.g., alternating 1,0,1,0,...):
- d=0: P_0 = dx * 0 (no self-overlap since h(1-h) = 0 for binary)
  Wait, P_0 = sum h[i](1-h[i]) = 0 for binary!
- d=1: P_1 = sum h[i+1](1-h[i]). For alternating, h[i+1] = 1-h[i], so P_1 = sum (1-h[i])^2... no.
  For alternating 1,0,1,0,...: h[i+1](1-h[i]) = (1-h[i])(1-h[i]) = (1-h[i])^2. For binary, this is 1 when h[i]=0.
  So P_1 = number of zeros * dx = (n/2) * (2/n) = 1.
  
  That's even worse! So binary alternating is bad.

What about a block pattern? E.g., 1,1,...,1,0,0,...,0 (n/2 ones then n/2 zeros):
- d=0: P_0 = 0 (binary)
- d=1: P_1 = sum h[i+1](1-h[i]). Only non-zero at the boundary: h[n/2-1]=1, h[n/2]=0, so one term = 1. P_1 = dx * 1 = 2/n.
- d=n/2: P_{n/2} = sum h[i+n/2](1-h[i]). h[i+n/2] is 0 for i < n/2 and 1 for i ≥ n/2.
  So h[i+n/2](1-h[i]) = 1*(1-0) = 1 for i < n/2, and 0 for i ≥ n/2.
  P_{n/2} = (n/2) * dx = 1.

So the block pattern gives max P = 1, which is terrible.

The key insight is that we need a pattern where for every lag d, the number of "1→0" transitions is small AND the number of "0→1" transitions is small. This is essentially a problem of finding a binary (or gray-scale) sequence with low autocorrelation.

This is related to the theory of low-discrepancy sequences and balancing sequences!

For the continuous case (h not restricted to {0,1}), the optimal solution will have h values between 0 and 1, effectively "smoothing" the binary pattern.

```python
import numpy as np
from scipy.optimize import linprog
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    prof = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(prof))

def profile_h(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def repair_feasible(v):
    n = len(v)
    h = np.clip(np.asarray(v, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx])
            h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def build_grad_row(h, d, n, dx):
    g = np.zeros(n)
    if d == 0:
        g[:] = dx * (1.0 - 2.0 * h)
    elif d > 0:
        for j in range(d, n):
            g[j] += dx * (1.0 - h[j - d])
        for j in range(n - d):
            g[j] -= dx * h[j + d]
    else:
        m = -d
        for j in range(n - m):
            g[j] += dx * (1.0 - h[j + m])
        for j in range(m, n):
            g[j] -= dx * h[j - m]
    return g

def lp_leveling_direction(h, active_offsets, n, dx):
    G = np.zeros((len(active_offsets), n))
    for idx, d in enumerate(active_offsets):
        G[idx] = build_grad_row(h, d, n, dx)
    c_obj = np.zeros(n + 1)
    c_obj[-1] = -1.0
    A_ub = np.zeros((len(active_offsets), n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = 1.0
    b_ub = np.zeros(len(active_offsets))
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    bounds = [(-1.0, 1.0)] * n + [(0.0, 50.0)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0 and res.x is not None:
            t_val = res.x[-1]
            if t_val > 0.005:
                return res.x[:n], t_val
    except Exception:
        pass
    return None, 0.0

def line_search_direction(h, direction, n, best_c):
    alphas = [0.3, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002]
    best_h = h.copy()
    best_sc = best_c
    for a in alphas:
        h_try = h + a * direction
        h_try = np.clip(h_try, 0.0, 1.0)
        h_try = repair_feasible(h_try)
        sc = exact_score(h_try)
        if sc < best_sc - 1e-15:
            best_sc = sc
            best_h = h_try.copy()
    return best_h, best_sc

def pair_line_search(h, i, j):
    s = h[i] + h[j]
    lo = max(0.0, s - 1.0)
    hi = min(1.0, s)
    if hi - lo < 1e-14:
        return h, exact_score(h)
    a, b = lo, hi
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    current = h[i]
    fa = exact_score(h)
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    hc = h.copy(); hc[i] = c; hc[j] = s - c
    hd = h.copy(); hd[i] = d; hd[j] = s - d
    fc = exact_score(hc)
    fd = exact_score(hd)
    for _ in range(40):
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            hc = h.copy(); hc[i] = c; hc[j] = s - c
            fc = exact_score(hc)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            hd = h.copy(); hd[i] = d; hd[j] = s - d
            fd = exact_score(hd)
    t = (a + b) / 2.0
    h_best = h.copy()
    h_best[i] = t
    h_best[j] = s - t
    sc_best = exact_score(h_best)
    if sc_best < fa - 1e-15:

    return h, fa

    best_c = exact_score(h)
    for sweep in range(max_sweeps):

            break

                if sc < best_c - 1e-14:
                    h = h_new
                    best_c = sc
                    improved = True
        if not improved:
            break
    return h, best_c

def lp_ascent_phase(h, t_start=None, budget_s=1100, max_iters=200):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    best_c = exact_score(h)
    for it in range(max_iters):
        if t_start is not None and time.perf_counter() - t_start > budget_s - 30:
            break
        P = profile_h(h)
        c_max = P.max()
        median_P = np.median(P)
        spread = c_max - median_P
        for k_active in [3, 5, 8, 12]:
            thresh = c_max - max(0.002, 0.1 * spread)
            active = np.where(P >= thresh)[0]
            if len(active) > k_active:
                top_indices = np.argsort(P[active])[-k_active:]
                active = active[top_indices]
            if len(active) < 2:

            offsets = active - (n - 1)
            direction, t_val = lp_leveling_direction(h, offsets, n, dx)

                if sc_new < best_c - 1e-14:
                    h = h_new
                    best_c = sc_new
                    break

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    return repair_feasible(np.clip(h_new, 0.0, 1.0))

def random_perturb(h, eps, rng):
    n = len(h)
    z = rng.randn(n)
    z -= z.mean()
    h_new = np.clip(h + eps * z, 0.0, 1.0)
    return repair_feasible(h_new)

def triple_line_search(h, i, j, k):
    s = h[i] + h[j] + h[k]
```
