def bit_energy(bits,n,dx,soft_tau,soft_weight):
    full=(1<<n)-1; zeros=full^bits; m=0; aux=0; counts = np.empty(n-1)
    for p in range(1,n):
        pc = (bits & (zeros >> p)).bit_count()
        nc = (zeros & (bits >> p)).bit_count()
        c = pc if pc>=nc else nc
        counts[p-1]=c
    if len(counts)>0:
        mx = counts.max(); m=mx
        z = (counts - mx)/soft_tau
        hard = np.exp(z); aux=hard.sum()
        energy = m + soft_weight*aux
    else: energy=0
