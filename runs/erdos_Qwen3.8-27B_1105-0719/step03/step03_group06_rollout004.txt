        remaining = diff

            idx = order[i]

            add = min(remaining, cap)
            h[idx] += add
            remaining -= add

        a_ls = 0.0
        b_ls = alpha_max
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        for _ in range(20):
            c1 = b_ls - gr * (b_ls - a_ls)
            c2 = a_ls + gr * (b_ls - a_ls)
            h1 = repair_feasible(h + c1 * direction)
            h2 = repair_feasible(h + c2 * direction)
            f1 = exact_score(h1)
            f2 = exact_score(h2)
            if f1 < best_val:
                best_val = f1
                best_alpha = c1
                best_h_new = h1.copy()
            if f2 < best_val:
                best_val = f2
                best_alpha = c2
                best_h_new = h2.copy()
            if f1 <= f2:
                b_ls = c2
            else:
                a_ls = c1
            if b_ls - a_ls < 1e-14:
                break
        if best_val >= c - 1e-15:
            break
        h = best_h_new
        c = best_val
        if c < best_c:
            best_c = c
            best_h = h.copy()
    return best_h, best_c

def interp_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    edges_old = np.linspace(0.0, 2.0, n + 1)
    edges_new = np.linspace(0.0, 2.0, new_n + 1)
    h_new = np.zeros(new_n)
    for k in range(new_n):
        a, b = edges_new[k], edges_new[k + 1]
        total = 0.0
        for j in range(n):
            lo = max(a, edges_old[j])
            hi = min(b, edges_old[j + 1])
            if hi > lo:
                total += h[j] * (hi - lo)
        h_new[k] = total / (b - a)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

def golden_seed(n, rng=None):
    half = n // 2
    h = np.zeros(n)
    idx = np.arange(n)
    vals = (idx * 0.6180339887498949) % 1.0
    top = np.argsort(vals)[-half:]
    h[top] = 1.0
    return repair_feasible(h)

def balanced_seed(n, rng=None):
    half = n // 2
    h = np.zeros(n)
    period = max(3, n // 7)
    for i in range(n):
        if (i // period) % 2 == 0:
            h[i] = 1.0
    ones = int(h.sum())
    if ones > half:
        excess = ones - half
        one_idx = np.where(h == 1.0)[0]
        if rng is not None:
            to_flip = rng.choice(one_idx, excess, replace=False)
        else:
            to_flip = one_idx[:excess]
        h[to_flip] = 0.0
    elif ones < half:
        deficit = half - ones
        zero_idx = np.where(h == 0.0)[0]
        if rng is not None:
            to_flip = rng.choice(zero_idx, deficit, replace=False)
        else:
            to_flip = zero_idx[:deficit]
        h[to_flip] = 1.0
    return repair_feasible(h)

def fourier_seed(n, rng=None):
    half = n // 2
    x = np.linspace(0, 2.0 * np.pi, n, endpoint=False)
    vals = np.sin(3 * x) * 0.5 + np.sin(7 * x) * 0.3 + np.sin(11 * x) * 0.2
    threshold = np.sort(vals)[half]
    h = (vals >= threshold).astype(float)
    ones = int(h.sum())
    if ones > half:
        idx = np.where(h == 1.0)[0]
        order = np.argsort(vals[idx])
        h[idx[order[:ones - half]]] = 0.0
    elif ones < half:
        idx = np.where(h == 0.0)[0]
        order = np.argsort(-vals[idx])
        h[idx[order[:half - ones]]] = 1.0
    return repair_feasible(h)

def slsqp_epigraph(h_start, maxiter=150):
    n = len(h_start)
    dx = 2.0 / n
    M = 2 * n - 1
    c0 = exact_score(h_start)
    t0 = c0 + 1e-4
    def obj(x):
        return x[-1]
    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    def eq_fun(x):
        return x[:-1].sum() - n / 2.0
    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g
    def ineq_fun(x):
        h = x[:-1]
        t = x[-1]
        profile = np.correlate(h, 1.0 - h, mode="full") * dx
        return t - profile
    def ineq_jac(x):
        h = x[:-1]
        J = np.zeros((M, n + 1))
        for m in range(M):
            J[m, :n] = -grad_row(h, m, n)
        J[:, -1] = 1.0
        return J
    x0 = np.concatenate([h_start, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 2.0)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None and np.isfinite(res.x).all():
            h_new = repair_feasible(res.x[:-1])
            c_new = exact_score(h_new)
            if c_new <= c0 + 1e-8:
                return h_new
    except Exception:
        pass
    return None

def swap_polish(h, max_rounds=60):
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    for rnd in range(max_rounds):
        profile = compute_profile(h)
        c = float(profile.max())
        active = np.where(profile >= c - 0.003)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(profile))])
        ones_idx = np.where(h > 0.5)[0]
        zeros_idx = np.where(h < 0.5)[0]
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            high_idx = np.argsort(-h)
            low_idx = np.argsort(h)
            ones_idx = high_idx[:n // 4]
            zeros_idx = low_idx[:n // 4]
        improved = False
        rng_local = np.random.RandomState(int(c * 1e8) + rnd)
        for attempt in range(30):
            i = int(ones_idx[rng_local.randint(len(ones_idx))])
            j = int(zeros_idx[rng_local.randint(len(zeros_idx))])
            if i == j:
                continue
            delta = 0.1 * (h[i] - h[j])
            if abs(delta) < 1e-7:
                continue
            h_try = h.copy()
            h_try[i] -= delta
            h_try[j] += delta
            if h_try.min() < -1e-10 or h_try.max() > 1 + 1e-10:
                continue
            h_try = repair_feasible(h_try)
            s = exact_score(h_try)
            if s < best_c - 1e-14:
                h = h_try
                best_c = s
                improved = True
                break
        if not improved:
            break
    return h, best_c

def hard_active_descent(h, max_iters=300, active_tol=0.002):
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    v = np.zeros(n)
    momentum = 0.35
    for it in range(max_iters):
        profile = compute_profile(h)
        c = float(profile.max())
        thresh = c - active_tol
        active = np.where(profile >= thresh)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(profile))])
        if len(active) > 25:
            top = np.argsort(profile[active])[-25:]
            active = active[top]
        g = np.zeros(n)
        for m in active:
            g += grad_row(h, m, n)
        g -= g.mean()
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            break
        v = momentum * v + g
        vnorm = np.linalg.norm(v)
        if vnorm < 1e-15:
            break
        d = -v / vnorm
        pos_mask = d > 1e-12
        neg_mask = d < -1e-12
        alpha_max = 1.0
        if pos_mask.any():
            alpha_max = min(alpha_max, float(((1.0 - h[pos_mask]) / d[pos_mask]).min()))
        if neg_mask.any():
            alpha_max = min(alpha_max, float((h[neg_mask] / (-d[neg_mask])).min()))
        alpha_max = max(alpha_max, 1e-10)
        alpha_max = min(alpha_max, 0.35)
        best_alpha = 0.0
        best_val = c
        best_h = h
        a_ls, b_ls = 0.0, alpha_max
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        for _ in range(20):
            c1 = b_ls - gr * (b_ls - a_ls)
            c2 = a_ls + gr * (b_ls - a_ls)
            h1 = repair_feasible(h + c1 * d)
            h2 = repair_feasible(h + c2 * d)
            f1 = exact_score(h1)
            f2 = exact_score(h2)
            if f1 < best_val:
                best_val = f1
                best_alpha = c1
                best_h = h1.copy()
            if f2 < best_val:
                best_val = f2
                best_alpha = c2
                best_h = h2.copy()
            if f1 <= f2:
                b_ls = c2
            else:
                a_ls = c1
            if b_ls - a_ls < 1e-14:
                break
        if best_val >= c - 1e-15:
            break
        h = best_h
        if best_val < best_c:
            best_c = best_val
    return h, best_c

def random_jitter(h, sigma, rng):
    n = len(h)
    noise = rng.normal(0, sigma, n)
    h_j = h + noise
    return repair_feasible(h_j)

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s - (time.perf_counter() - t_start)

    try:
        parent = np.array(initial_h_values, dtype=np.float64)
    except Exception:
        parent = None
    if parent is None or len(parent) < 10:
        parent = golden_seed(70)
    parent = repair_feasible(parent)
    n_parent = len(parent)

    best_h = parent.copy()
    best_c = exact_score(parent)

    # Phase 1: LSE active descent on parent
    if time_left() > 120:
        h_ad, c_ad = lse_active_descent(best_h, max_iters=500,
                                         beta_start=15.0, beta_end=400.0)
        if c_ad < best_c:
            best_h, best_c = h_ad, c_ad

    # Phase 2: Hard active descent with tight tolerance
    if time_left() > 80:
        h_ad, c_ad = hard_active_descent(best_h, max_iters=400, active_tol=0.001)
        if c_ad < best_c:
            best_h, best_c = h_ad, c_ad

    # Phase 3: Dimension exploration
    n_best = len(best_h)
    dims_to_try = []
    for dn in [-4, -2, 2, 4, 6, 8, -6, -8, 10, 12]:
        d = n_best + dn
        if 36 <= d <= 140 and d % 2 == 0:
            if d not in dims_to_try:
                dims_to_try.append(d)
    dims_to_try.sort(key=lambda x: abs(x - n_best))

    for new_n in dims_to_try[:6]:
        if time_left() < 40:
            break
        h_interp = interp_to_n(best_h, new_n)
        h_ad, c_ad = lse_active_descent(h_interp, max_iters=350,
                                         beta_start=20.0, beta_end=300.0)
        if c_ad < best_c:
            best_h, best_c = h_ad, c_ad
            n_best = new_n
        h_h, c_h = hard_active_descent(h_ad, max_iters=200, active_tol=0.001)
        if c_h < best_c:
            best_h, best_c = h_h, c_h
            n_best = new_n

    # Phase 4: SLSQP polish
    for _ in range(2):
        if time_left() < 30:
            break
        h_sol = slsqp_epigraph(best_h, maxiter=120)
        if h_sol is not None:
            c_sol = exact_score(h_sol)
            if c_sol < best_c - 1e-12:
                best_h, best_c = h_sol, c_sol
            h_ad, c_ad = hard_active_descent(best_h, max_iters=200, active_tol=0.001)
            if c_ad < best_c:
                best_h, best_c = h_ad, c_ad

    # Phase 5: Multi-restart from diverse seeds
    if time_left() > 150:
        n_explore = len(best_h)
        seed_types = [
            (lambda: golden_seed(n_explore, rng), 0.3),
            (lambda: balanced_seed(n_explore, rng), 0.3),
            (lambda: fourier_seed(n_explore, rng), 0.3),
            (lambda: repair_feasible(np.full(n_explore, 0.5) +
                rng.normal(0, 0.1, n_explore)), 0.3),
        ]
        for seed_fn, mix_w in seed_types:
            if time_left() < 30:
                break
            h_seed = seed_fn()
            # Mix with current best for warm-starting
            h_mix = (1.0 - mix_w) * best_h + mix_w * h_seed
            h_mix = repair_feasible(h_mix)
            h_ad, c_ad = lse_active_descent(h_mix, max_iters=300,
                                             beta_start=15.0, beta_end=350.0)
            if c_ad < best_c:
                best_h, best_c = h_ad, c_ad
                if time_left() > 30:
                    h_h, c_h = hard_active_descent(best_h, max_iters=200, active_tol=0.001)
                    if c_h < best_c:
                        best_h, best_c = h_h, c_h

    # Phase 6: Swap polish
    if time_left() > 40:
        h_sw, c_sw = swap_polish(best_h, max_rounds=80)
        if c_sw < best_c:
            best_h, best_c = h_sw, c_sw

    # Phase 7: Perturbation and re-optimize (escape local minima)
    if time_left() > 80:
        for sigma in [0.05, 0.03, 0.02]:
            if time_left() < 25:
                break
            h_jit = random_jitter(best_h, sigma, rng)
            h_ad, c_ad = hard_active_descent(h_jit, max_iters=250, active_tol=0.001)
            if c_ad < best_c:
                best_h, best_c = h_ad, c_ad
            h_ad2, c_ad2 = lse_active_descent(h_ad, max_iters=200,
                                               beta_start=30.0, beta_end=500.0)
            if c_ad2 < best_c:
                best_h, best_c = h_ad2, c_ad2

    # Phase 8: Final tight optimization
    if time_left() > 30:
        h_ad, c_ad = lse_active_descent(best_h, max_iters=400,
                                         beta_start=50.0, beta_end=800.0)
        if c_ad < best_c:
            best_h, best_c = h_ad, c_ad

    if time_left() > 20:
        h_ad, c_ad = hard_active_descent(best_h, max_iters=300, active_tol=0.0005)
        if c_ad < best_c:
            best_h, best_c = h_ad, c_ad

    # Phase 9: Final SLSQP
    if time_left() > 15:
        h_sol = slsqp_epigraph(best_h, maxiter=150)
        if h_sol is not None:
            c_sol = exact_score(h_sol)
            if c_sol < best_c:
                best_h, best_c = h_sol, c_sol

    # Final repair and evaluation
    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)

    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            final_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 2:
            final_c = float(eval_result[0])
    except Exception:
        pass

    return best_h, final_c, len(best_h)
