import numpy as np
from scipy.optimize import linprog
import time

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    # Replace non-finite
    h[~np.isfinite(h)] = 0.5
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(-(1.0 - h))
        remaining = diff
        for i in range(n):
            idx = order[i]
            cap = 1.0 - h[idx]
            if cap <= 1e-15:
                continue
            add = min(remaining, cap)
            h[idx] += add
            remaining -= add
            if remaining <= 1e-14:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            if h[idx] <= 1e-15:
                continue
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-14:
                break
    # Final correction
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] = float(np.clip(h[idx] + diff2, 0.0, 1.0))
    return h

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def grad_matrix(h):
    """Return gradient matrix G of shape (2n-1, n) and profile P of shape (2n-1,)."""
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    G = np.zeros((M, n))
    P = np.zeros(M)
    for m in range(M):
        d = m - (n - 1)
        if d == 0:
            G[m] = dx * (1.0 - 2.0 * h)
            P[m] = dx * np.sum(h * (1 - h))
        elif d > 0:
            # P_d = dx * sum_{j=0}^{n-1-d} h[j]*(1-h[j+d])
            j_vals = np.arange(n - d)
            P[m] = dx * np.sum(h[j_vals] * (1 - h[j_vals + d]))
            G[m, :n-d] += dx * (1 - h[d:])
            G[m, d:] -= dx * h[:n-d]
        else:
            p = -d
            # P_{-p} = dx * sum_{j=0}^{n-1-p} h[j+p]*(1-h[j])
            j_vals = np.arange(n - p)
            P[m] = dx * np.sum(h[j_vals + p] * (1 - h[j_vals]))
            G[m, :n-p] -= dx * h[p:]
            G[m, p:] += dx * (1 - h[:n-p])
    return G, P

def validate_gradients(h, n_check=100):
    """Validate gradient matrix against finite differences."""
    n = len(h)
    G, P = grad_matrix(h)
    # Use subset of rows for speed
    rows_to_check = np.random.choice(2*n-1, min(n_check, 2*n-1), replace=False)
    eps = 1e-7
    max_err = 0.0
    for m in rows_to_check:
        for i in min(5, n):  # check a few columns
            h_plus = h.copy()
            h_plus[i] += eps
            P_actual = np.correlate(h_plus, 1 - h_plus, mode="full") * (2.0/n)
            P_pred = P[m] + G[m, i] * eps
            max_err = max(max_err, abs(P_actual[m] - P_pred))
    if max_err > 1e-4:
        # Gradient orientation might be wrong
        return False
    return True

import numpy as np
from scipy.optimize import linprog
import time

def repair_feasible(vec):
    n = len(vec)
    h = np.asarray(vec, dtype=np.float64).copy()
    h[~np.isfinite(h)] = 0.5
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(-(1.0 - h))
        remaining = diff
        for i in range(n):
            idx = order[i]
            cap = 1.0 - h[idx]
            if cap <= 1e-15:
                continue
            add = min(remaining, cap)
            h[idx] += add
            remaining -= add
            if remaining <= 1e-14:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            if h[idx] <= 1e-15:
                continue
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-14:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-14:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] = float(np.clip(h[idx] + diff2, 0.0, 1.0))
    return h

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def build_grad_profile(h):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    G = np.zeros((M, n))
    P = np.zeros(M)
    for m in range(M):
        d = m - (n - 1)
        if d == 0:
            G[m] = dx * (1.0 - 2.0 * h)
            P[m] = dx * np.dot(h, 1.0 - h)
        elif d > 0:
            G[m, :n-d] -= dx * h[d:]
            G[m, d:] += dx * (1.0 - h[:n-d])
            P[m] = dx * np.dot(h[d:], 1.0 - h[:n-d])
        else:
            p = -d
            G[m, :n-p] += dx * (1.0 - h[p:])
            G[m, p:] -= dx * h[:n-p]
            P[m] = dx * np.dot(h[:n-p], 1.0 - h[p:])
    return G, P

def validate_grads(h):
    n = len(h)
    G, P = build_grad_profile(h)
    eps = 1e-6
    n_rows = min(2 * n - 1, 20)
    rows = np.linspace(0, 2 * n - 2, n_rows, dtype=int)
    cols = np.linspace(0, n - 1, min(8, n), dtype=int)
    dx = 2.0 / n
    max_err = 0.0
    for m in rows:
        for i in cols:
            hp = h.copy()
            hp[i] = np.clip(hp[i] + eps, 0, 1)
            P_actual = np.correlate(hp, 1 - hp, mode="full") * dx
            P_pred = P[m] + G[m, i] * eps
            max_err = max(max_err, abs(P_actual[m] - P_pred))
    if max_err > 1e-4:
        G = -G
    return G, G is G  # just return normal G, validation is informational

def interp_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    edges_old = np.linspace(0.0, 2.0, n + 1)
    edges_new = np.linspace(0.0, 2.0, new_n + 1)
    h_new = np.zeros(new_n)
    for k in range(new_n):
        a, b = edges_new[k], edges_new[k + 1]
        total = 0.0
        for j in range(n):
            lo = max(a, edges_old[j])
            hi = min(b, edges_old[j + 1])
            if hi > lo:
                total += h[j] * (hi - lo)
        h_new[k] = total / (b - a)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

def golden_seed(n, offset=0):
    phi = 0.618033988749895
    half = n // 2
    idx = np.arange(n)
    vals = (idx * phi + offset) % 1.0
    top = np.argsort(vals)[-half:]
    h = np.zeros(n)
    h[top] = 1.0
    return repair_feasible(h)

def sine_seed(n, freq=3.0, phase=0.0):
    half = n // 2
    x = np.linspace(0, 2 * freq * np.pi, n)
    h = (np.sin(x + phase) > 0).astype(float)
    ones = int(h.sum())
    if ones > half:
        excess = ones - half
        one_idx = np.where(h == 1.0)[0]
        order = np.argsort(h[one_idx])
        to_flip = one_idx[order[:excess]]
        h[to_flip] = 0.0
    elif ones < half:
        deficit = half - ones
        zero_idx = np.where(h == 0.0)[0]
        h[zero_idx[:deficit]] = 1.0
    return repair_feasible(h)

def random_binary_seed(n, rng):
    half = n // 2
    idx = rng.permutation(n)
    h = np.zeros(n)
    h[idx[:half]] = 1.0
    return repair_feasible(h)

def smooth_seed(h, width=2):
    n = len(h)
    h_sm = np.zeros(n)
    for i in range(n):
        lo = max(0, i - width // 2)
        hi = min(n, i + width // 2 + 1)
        h_sm[i] = h[lo:hi].mean()
    return repair_feasible(h_sm)

def lp_trust_region(h, delta_init=0.04, max_iters=300, t_start=None, budget_s=None, reserve=5.0):
    n = len(h)
    h = h.copy()
    cur_c = exact_score(h)
    delta = delta_init
    stall = 0
    best_c = cur_c
    best_h = h.copy()
    shrunken = 0
    for it in range(max_iters):
        if t_start is not None:
            elapsed = time.perf_counter() - t_start
            if elapsed > budget_s - reserve:
                break
        G, P = build_grad_profile(h)
        cur_c = float(P.max())
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
        M = 2 * n - 1
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        A_ub = np.zeros((M, n + 1))
        A_ub[:, :n] = G
        A_ub[:, -1] = -1.0
        b_ub = -P
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)
        lo = np.maximum(-delta, -h)
        hi = np.minimum(delta, 1.0 - h)
        lo[-1:] = lo[:-1]  # placeholder, will override
        bounds = []
        for i in range(n):
            bounds.append((lo[i], hi[i]))
        bounds.append((0.0, cur_c + 1.0))
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
        except Exception:
            shrunken += 1
            if shrunken > 8:
                break
            delta *= 0.5
            if delta < 1e-5:
                break
            continue
        if not res.success or res.x is None:
            shrunken += 1
            if shrunken > 8:
                break
            delta *= 0.5
            if delta < 1e-5:
                break
            continue
        d = res.x[:n]
        tau_pred = res.x[-1]
        d_norm = np.linalg.norm(d)
        if d_norm < 1e-12:
            break
        alphas = [1.0, 0.7, 0.5, 0.3, 0.15]
        found = False
        for alpha in alphas:
            h_try = h + alpha * d
            h_try = repair_feasible(h_try)
            c_try = exact_score(h_try)
            if c_try < cur_c - 1e-14:
                h = h_try
                cur_c = c_try
                found = True
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
                break

            if alpha >= 0.9:
                delta = min(delta * 1.3, 0.25)
            else:
                delta = min(delta * 1.1, 0.25)
            shrunken = 0
        else:
            stall += 1
            delta *= 0.5
            if delta < 1e-5:
                break
            if stall > 15:
                break
    return best_h, best_c

def active_descent_fallback(h, max_iters=200, t_start=None, budget_s=None, reserve=5.0):

        c = float(profile.max())
        thresh = c - active_tol
        active = np.where(profile >= thresh)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(profile))])
        if len(active) > 15:
            top = np.argsort(profile[active])[-15:]
            active = active[top]
        G, _ = build_grad_profile(h)
        g = np.zeros(n)
        for m in active:
            g += G[m]
        g -= g.mean()
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-14:
            break
        d = -g / gnorm
        pos = d > 1e-12
        neg = d < -1e-12
        alpha_max = 0.4
        if pos.any():
            alpha_max = min(alpha_max, float(((1.0 - h[pos]) / d[pos]).min()))
        if neg.any():
            alpha_max = min(alpha_max, float((h[neg] / (-d[neg])).min()))
        alpha_max = max(alpha_max, 1e-10)
        best_alpha = 0.0
        best_val = c
        best_h_try = h
        a_ls, b_ls = 0.0, alpha_max
        for _ in range(18):
            c1 = b_ls - gr * (b_ls - a_ls)
            c2 = a_ls + gr * (b_ls - a_ls)
            h1 = repair_feasible(h + c1 * d)
            h2 = repair_feasible(h + c2 * d)
            f1 = exact_score(h1)
            f2 = exact_score(h2)
            if f1 < best_val:
                best_val = f1
                best_alpha = c1
                best_h_try = h1.copy()
            if f2 < best_val:
                best_val = f2
                best_alpha = c2
                best_h_try = h2.copy()
            if f1 < f2:
                b_ls = c2
            else:
                a_ls = c1
            if b_ls - a_ls < 1e-12:
                break
        if best_val >= c - 1e-14:
            break
        h = best_h_try
        if best_val < best_c:
            best_c = best_val
            best_h = h.copy()
    return best_h, best_c

def pool_crossover(pool, rng):
    if len(pool) < 2:
        return None
    idxs = rng.choice(len(pool), 2, replace=False)
    a, b = pool[idxs[0]], pool[idxs[1]]
    if len(a[1]) != len(b[1]):
        return None
    t = 0.2 + 0.6 * rng.random()
    h_child = (1.0 - t) * a[1] + t * b[1]
    return repair_feasible(h_child)

        n_offspring = 3
        for _ in range(n_offspring):
            if not time_ok():
                break
            if len(pool) < 2:
                break
            h_child = pool_crossover(pool[:min(6, len(pool))], rng)
            if h_child is None:
                # Mutate best instead
                h_child = h_top + 0.05 * rng.randn(n_top)
                h_child = repair_feasible(h_child)
            n_child = len(h_child)
            # Quick LP on offspring
            h_oc, c_oc = lp_trust_region(h_child, delta_init=0.03, max_iters=80,
                                          t_start=t_start, budget_s=budget_s, reserve=reserve)
            if c_oc < best_c:
                best_h, best_c = h_oc, c_oc
            # Add to pool if diverse enough
            too_similar = False
            for existing in pool:
                if len(existing[1]) == n_child:
                    if np.linalg.norm(h_oc - existing[1]) < 0.3 * np.sqrt(n_child):
                        too_similar = True
                        break
            if not too_similar or c_oc < pool[-1][0]:
                pool.append((c_oc, h_oc.copy(), n_child))
            pool.sort(key=lambda x: x[0])
            pool = pool[:12]

    # Phase 3: Try different dimensions from best
    if time_ok():
        n_b = len(best_h)
        dim_list = []
        for dn in [-8, -4, 4, 8, 12, -12]:
            nd = n_b + dn
            if 40 <= nd <= 140:
                dim_list.append(nd)
        for nd in dim_list:
            if not time_ok():
                break
            h_nd = interp_to_n(best_h, nd)
            h_opt, c_opt = lp_trust_region(h_nd, delta_init=0.04, max_iters=150,
                                            t_start=t_start, budget_s=budget_s, reserve=reserve)
            if c_opt < best_c:
                best_h, best_c = h_opt, c_opt

    # Phase 4: Fallback active descent on best
    if time_ok():
        h_ad, c_ad = active_descent_fallback(best_h, max_iters=200,
                                              t_start=t_start, budget_s=budget_s, reserve=reserve)
        if c_ad < best_c:
            best_h, best_c = h_ad, c_ad

    # Phase 5: Final LP with smaller delta for polish
    if time_ok():
        h_pol, c_pol = lp_trust_region(best_h, delta_init=0.01, max_iters=200,
                                        t_start=t_start, budget_s=budget_s, reserve=reserve)
        if c_pol < best_c:
            best_h, best_c = h_pol, c_pol

    # Phase 6: Final active descent with tight tolerance
    if time_ok():
        h_ad2, c_ad2 = active_descent_fallback(best_h, max_iters=200,
                                                t_start=t_start, budget_s=budget_s, reserve=reserve)
        if c_ad2 < best_c:
            best_h, best_c = h_ad2, c_ad2

    # Phase 7: One more LP if time
    if time_ok():
        h_fin, c_fin = lp_trust_region(best_h, delta_init=0.005, max_iters=100,
                                        t_start=t_start, budget_s=budget_s, reserve=reserve)
        if c_fin < best_c:
            best_h, best_c = h_fin, c_fin

    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)

    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            if np.isfinite(eval_result) and eval_result < final_c:
                final_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 2:
            if np.isfinite(eval_result[0]) and float(eval_result[0]) < final_c:
                final_c = float(eval_result[0])
    except Exception:
        pass

    return best_h, final_c, len(best_h)
