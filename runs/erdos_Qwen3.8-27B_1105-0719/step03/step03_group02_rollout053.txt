    #     where δ[i]=1, δ[j]=-1
    #     = (1-h[i+o]) - h[i-o] - (1-h[j+o]) + h[j-o]
    #     (with appropriate validity checks)
    #
    # c_k = sum over valid l of δ[l]*δ[l+o]
    #     = δ[i]*δ[i+o] + δ[j]*δ[j+o]
    #     = δ[i+o] - δ[j+o]
    #     (with validity checks)

        l_lo = max(0, -o)
        l_hi = min(n-1, n-1-o)

        # From l=i (if valid): δ[i]*(1-h[i+o]) = (1-h[i+o]) if i valid
        # From l+o=j (i.e., l=j-o): h[l]*(-δ[l+o]) = -h[j-o]*(-1) = h[j-o] if l valid
        # From l=j (if valid): δ[j]*(1-h[j+o]) = -(1-h[j+o]) if j valid
        # From l+o=i (i.e., l=i-o): h[l]*(-δ[l+o]) = -h[i-o]*1 = -h[i-o] if l valid
        
        bval = 0.0
        if l_lo <= i <= l_hi:  # i is valid left index
            bval += (1.0 - h[i + o])
        if l_lo <= (i - o) <= l_hi:  # i-o is valid left index, so l+o=i
            bval -= h[i - o]
        if l_lo <= j <= l_hi:  # j is valid left index
            bval -= (1.0 - h[j + o])
        if l_lo <= (j - o) <= l_hi:  # j-o is valid left index, so l+o=j
            bval += h[j - o]
        b[k] = bval
        
        # c contribution:
        # c = δ[i]*δ[i+o] + δ[j]*δ[j+o] (if i and j are valid left indices)
        # δ[i]=1, δ[j]=-1
        # δ[i+o] = 1 if o=0 and i+o=i, or -1 if i+o=j (i.e., o=j-i), else 0
        # δ[j+o] = 1 if o=0 and j+o=j, or -1 if j+o=i (i.e., o=i-j), else 0
        
        cval = 0.0
        # δ[i]*δ[i+o]: δ[i]=1, so just δ[i+o]
        # i+o = i means o=0: δ[i]=1, so c += 1
        # i+o = j means o=j-i: δ[j]=-1, so c += -1
        # i+o = something else: δ=0, c += 0
        if l_lo <= i <= l_hi:  # i must be valid for this term to exist
            if o == 0:
                cval += 1  # δ[i]*δ[i] = 1*1
            elif o == j - i:
                cval += -1  # δ[i]*δ[j] = 1*(-1)
        
        # δ[j]*δ[j+o]: δ[j]=-1
        if l_lo <= j <= l_hi:
            if o == 0:
                cval += 1  # δ[j]*δ[j] = (-1)*(-1) = 1
            elif o == i - j:
                cval += -1  # δ[j]*δ[i] = (-1)*(1) = -1
        c[k] = cval
    
    return b, c
