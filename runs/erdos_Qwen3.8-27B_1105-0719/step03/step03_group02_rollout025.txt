    grad = np.zeros(n)
    i_lo = max(0, -offset)
    i_hi = min(n - 1, n - 1 - offset)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx + offset] += (1.0 - h[idx])
    grad[idx] += (-h[idx + offset])
    return grad


def smooth_max_grad(h, dx, beta):
    n = len(h)
    corrs = compute_corrs(h)
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones_like(corrs) / len(corrs)
    else:
        w /= w_sum
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] > 1e-13:
            grad += w[k] * grad_corr_k(h, k, n)
    return grad * dx


def slsqp_optimize(h0, dx, target_sum, beta, maxiter):
    n = len(h0)
    
    def obj(x):
        corrs = compute_corrs(x)
        return logsumexp(beta * corrs) / beta * dx
    
    def grad_obj(x):
        return smooth_max_grad(x, dx, beta)
    
    def c_fun(x):
        return x.sum() - target_sum
    
    def c_jac(x):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': c_fun, 'jac': c_jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0, jac=grad_obj, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        h_out = project_feasible(res.x, target_sum)
    except Exception:
        h_out = project_feasible(h0, target_sum)
    return h_out, c5_of_h(h_out, dx)


def lp_polish(h0, dx, target_sum, max_iters=200, init_step=0.15, n_active_max=80):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    step = init_step
    stall_count = 0
    
    for it in range(max_iters):
        corrs = compute_corrs(h)
        c5_val = corrs.max() * dx
        max_c = corrs.max()
        
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            stall_count = 0
        else:
            stall_count += 1
        if stall_count > 50:
            step *= 0.7
            stall_count = 0
            if step < 1e-7:
                break
        
        tol_frac = 0.03
        tol_val = max_c * (1.0 - tol_frac)
        active_mask = corrs >= tol_val
        active_k = np.where(active_mask)[0]
        
        if len(active_k) > n_active_max:
            active_k = np.argsort(corrs)[-n_active_max:]
        
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        A_rows = []
        b_rows = []
        for k in active_k:
            row = np.zeros(n + 1)
            row[:n] = dx * grad_corr_k(h, k, n)
            row[n] = -1.0
            A_rows.append(row)
            b_rows.append(-corrs[k] * dx - 1e-13)
        
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)
        
        bounds = []
        for i in range(n):
            bounds.append((max(-h[i], -step), min(1.0 - h[i], step)))
        bounds.append((0.0, None))
        
        try:
            res = linprog(c_obj, A_ub=np.array(A_rows), b_ub=np.array(b_rows),
                          A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
        except Exception:
            break
        if res.status != 0:
            break
        
        delta = res.x[:n]
        if np.max(np.abs(delta)) < 1e-14:
            break
        
        h_new = project_feasible(h + delta, target_sum)
        new_c5 = c5_of_h(h_new, dx)
        
        if new_c5 < c5_val - 1e-14:
            h = h_new
            step = min(step * 1.3, init_step * 1.5)
        else:
            improved = False
            for frac in [0.5, 0.25, 0.1, 0.05]:
                h_new = project_feasible(h + frac * delta, target_sum)
                new_c5 = c5_of_h(h_new, dx)
                if new_c5 < c5_val - 1e-14:
                    h = h_new
                    improved = True
                    break
            if not improved:
                step *= 0.5
                if step < 1e-8:
                    break
    
    return best_h, best_c5


def pairwise_2d_optimize(h0, dx, target_sum, n_top=8, n_iter=5, grid_size=15):
    """Optimize pairs of variables using 2D grid search."""
    n = len(h0)
    best_h = h0.copy()
    best_c5 = c5_of_h(h, dx)  # wait, h is not defined
    
    best_h = h0.copy()
    best_c5 = c5_of_h(best_h, dx)
    
    for iteration in range(n_iter):
        corrs = compute_corrs(best_h)
        grad = smooth_max_grad(best_h, dx, 300.0)
        
        # Find top gradient variables
        top_vars = np.argsort(np.abs(grad))[-n_top * 2:]
        
        improved = False
        moves = 0
        for ii in range(len(top_vars)):
            for jj in range(ii + 1, len(top_vars)):
                p, q = top_vars[ii], top_vars[jj]
                # Only try conflicting directions
                if grad[p] * grad[q] >= 0:
                    continue
                
                best_local = None
                best_local_val = best_c5
                
                lo_p = max(0.0, best_h[p] - 0.25)
                hi_p = min(1.0, best_h[p] + 0.25)
                lo_q = max(0.0, best_h[q] - 0.25)
                hi_q = min(1.0, best_h[q] + 0.25)
                
                hp_grid = np.linspace(lo_p, hi_p, grid_size)
                hq_grid = np.linspace(lo_q, hi_q, grid_size)
                
                for hp in hp_grid:
                    for hq in hq_grid:
                        h_try = best_h.copy()
                        h_try[p] = hp
                        h_try[q] = hq
                        h_try = project_feasible(h_try, target_sum)
                        c5_try = c5_of_h(h_try, dx)
                        if c5_try < best_local_val - 1e-14:
                            best_local_val = c5_try
                            best_local = h_try.copy()
                
                if best_local is not None:
                    best_h = best_local
                    best_c5 = best_local_val
                    improved = True
                    moves += 1
                    if moves >= 3:
                        break
            
            if moves >= 3:
                break
        
        if not improved:
            break
    
    return best_h, best_c5


def pair_transfer_polish(h0, dx, target_sum, n_top=20, n_iter=3):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    
    for iteration in range(n_iter):
        corrs = compute_corrs(h)
        grad = smooth_max_grad(h, dx, 500.0)
        
        dec_idx = np.argsort(grad)[-n_top:]
        inc_idx = np.argsort(grad)[:n_top]
        
        improved = False
        for i in dec_idx:
            for j in inc_idx:
                if i == j:
                    continue
                eps_lo = max(-h[j], -1.0 + 1.0)  # h[i] - eps >= 0 → eps <= h[i]; h[j] + eps <= 1 → eps <= 1-h[j]
                eps_lo = max(0.0, -(1.0 - h[j]))  # wait, let me think
                # h[i] - eps >= 0 → eps <= h[i]
                # h[j] + eps <= 1 → eps <= 1 - h[j]
                # h[i] - eps >= 0 → eps >= -h[i] but we want eps > 0 for "decreasing i"
                # Actually, let's allow negative eps too (swap direction)
                # Range of eps: h[i] - eps in [0,1] and h[j] + eps in [0,1]
                # eps in [max(-h[i], h[j]-1), min(h[i], 1-h[j])]
                eps_lo = max(-h[i], h[j] - 1.0)
                eps_hi = min(h[i], 1.0 - h[j])
                if eps_hi - eps_lo < 1e-10:
                    continue
                
                # Compute direction of change in correlations
                d = np.zeros(2 * n - 1)
                for k in range(2 * n - 1):
                    gk = grad_corr_k(h, k, n)
                    d[k] = gk[j] - gk[i]
                
                # Maximize the minimum margin: find eps that minimizes max(corrs + eps*d)
                # Use golden section search
                gr = (np.sqrt(5.0) - 1.0) / 2.0
                a, b = eps_lo + 1e-12, eps_hi - 1e-12
                if b <= a:
                    continue
                c_pt = b - gr * (b - a)
                d_pt = a + gr * (b - a)
                
                def f_eval(eps):
                    return (corrs + eps * d).max()
                
                fc, fd = f_eval(c_pt), f_eval(d_pt)
                for _ in range(40):
                    if b - a < 1e-13:
                        break
                    if fc < fd:
                        b = d_pt
                        d_pt = c_pt
                        fd = fc
                        c_pt = b - gr * (b - a)
                        fc = f_eval(c_pt)
                    else:
                        a = c_pt
                        c_pt = d_pt
                        fc = fd
                        d_pt = a + gr * (b - a)
                        fd = f_eval(d_pt)
                
                eps_opt = (a + b) / 2.0
                new_c5_rough = (corrs + eps_opt * d).max() * dx
                if new_c5_rough < best_c5 - 1e-13:
                    new_h = h.copy()
                    new_h[i] -= eps_opt
                    new_h[j] += eps_opt
                    new_h = np.clip(new_h, 0.0, 1.0)
                    new_h = project_feasible(new_h, target_sum)
                    new_c5 = c5_of_h(new_h, dx)
