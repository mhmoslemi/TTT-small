Let me analyze the problem and strategy carefully.

The task is to minimize C₅ = max_k ∫ h(x)(1-h(x+k)) dx over step functions h: [0,2] → [0,1] with ∫₀² h(x) dx = 1.

The current best is 0.38085, and the previous program achieved 0.380917. The target is 0.3808.

The strategy proposes working in the "complement symmetric" (anti-reflective) subspace where h(x) + h(2-x) = 1. This means:
- For even n = 2m: h_j = y_j for j=0..m-1, h_{n-1-j} = 1 - y_j
- The mass constraint is automatically satisfied: sum(h) = m = n/2

This is a clever reduction because:
1. It halves the dimension (m instead of n)
2. The mass constraint is automatically satisfied
3. Single coordinate moves are automatically feasible (preserving both bounds and mass)

Let me think about the key algorithms to implement:

1. **Reduced coordinate optimization**: Work with y of length m, where h is fully determined.
2. **Exact line search in reduced coordinates**: For a direction d in reduced space, find the best t along y + t*d.
3. **Active set descent in reduced space**: Focus on the most constraining lags.
4. **LSE (log-sum-exp) smoothing**: Use a smooth approximation to the max.
5. **Multiple resolutions**: Try different values of n.

Let me think about the gradient computation in reduced space.

For the complement symmetric case with even n = 2m:
- h[0:m] = y
- h[m:2m] = 1 - y_reversed, i.e., h[m+j] = 1 - y[m-1-j] for j=0..m-1

Wait, let me be more careful. If n = 2m:
- h[j] = y[j] for j = 0, ..., m-1
- h[n-1-j] = 1 - y[j] for j = 0, ..., m-1

So h[n-1] = 1 - y[0], h[n-2] = 1 - y[1], etc.
That means h[m-1] = y[m-1] and h[m] = 1 - y[m-1], h[m+1] = 1 - y[m-2], ..., h[2m-1] = 1 - y[0].

Wait, n-1-j for j=0..m-1 gives indices 2m-1, 2m-2, ..., m. So:
- h[0] = y[0], h[1] = y[1], ..., h[m-1] = y[m-1]
- h[m] = 1 - y[m-1], h[m+1] = 1 - y[m-2], ..., h[2m-1] = 1 - y[0]

So the second half is the reverse complement of the first half.

The profile is P_k = dx * sum_i h[i] * (1 - h[i+k]) for the appropriate range.

For the gradient with respect to y[j]:
- dh/dy[j] has +1 at position j and -1 at position n-1-j = 2m-1-j.

So dP_k/dy[j] = dP_k/dh[j] * 1 + dP_k/dh[2m-1-j] * (-1)

The key insight is that in the reduced space, we have m variables, and each variable move automatically preserves feasibility. This makes line searches much more efficient.

import numpy as np
from scipy.optimize import minimize
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx])
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-14:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmin(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def to_reduced(h):
    """Extract reduced variables from a feasible h of even length."""
    n = len(h)
    assert n % 2 == 0
    m = n // 2
    y = (h[:m] + (1.0 - h[m:][::-1])) / 2.0
    y = np.clip(y, 0.0, 1.0)
    return y

def from_reduced(y):
    """Build full h from reduced variables (even n)."""
    m = len(y)
    n = 2 * m
    h = np.zeros(n)
    h[:m] = y
    h[m:] = 1.0 - y[::-1]
    return h

def reduced_profile(y):
    """Compute the full profile from reduced variables."""
    h = from_reduced(y)
    return compute_profile(h)

def reduced_score(y):
    """Compute C5 from reduced variables."""
    h = from_reduced(y)
    return exact_score(h)

def build_gradient_matrix(h, n, dx):
    """Build gradient matrix G where G[k] = dP_k/dh for profile row k."""
    M = 2 * n - 1
    G = np.zeros((M, n))
    for m_idx in range(M):
        d = m_idx - (n - 1)
        if d == 0:
            G[m_idx, :] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            G[m_idx, :n-d] += dx * (1.0 - h[d:])
            G[m_idx, d:] += -dx * h[:n-d]
        else:
            dp = -d
            G[m_idx, dp:] += dx * (1.0 - h[:n-dp])
            G[m_idx, :n-dp] += -dx * h[dp:]
    return G

def reduced_gradient(y, h, profile, weights):
    """Compute gradient of weighted profile in reduced coordinates."""
    n = len(h)
    m = len(y)
    dx = 2.0 / n
    G = build_gradient_matrix(h, n, dx)
    g_full = G.T @ weights  # shape (n,)
    # For reduced variable j: dh[workspace_j]/dy[j] = +1 (at position j)
    #                         dh[n-1-j]/dy[j] = -1 (at position n-1-j)
    g_reduced = np.zeros(m)
    for j in range(m):
        g_reduced[j] = g_full[j] - g_full[n-1-j]
    return g_reduced

def exact_line_search_reduced(y, direction, best_c):
    """Find best t along y + t*direction, respecting [0,1] bounds on y."""
    m = len(y)
    # Find feasible interval for t
    t_min = 0.0
    t_max = 0.0
    for j in range(m):
        if direction[j] > 1e-15:
            t_min = max(t_min, -y[j] / direction[j])
            t_max = min(t_max, (1.0 - y[j]) / direction[j])
        elif direction[j] < -1e-15:
            t_min = max(t_min, (1.0 - y[j]) / direction[j])
            t_max = min(t_max, -y[j] / direction[j])
    
    if t_max - t_min < 1e-14:
        return 0.0, best_c
    
    # Bisection/golden search for minimum of score along the line
    # Use a few sample points
    best_t = 0.0
    best_val = best_c
    n_samples = 20
    for i in range(n_samples + 1):
        t = t_min + i * (t_max - t_min) / n_samples
        y_try = y + t * direction
        y_try = np.clip(y_try, 0.0, 1.0)
        c_try = reduced_score(y_try)
        if c_try < best_val:
            best_val = c_try
            best_t = t
    
    # Refine around best_t
    if best_t > 0 and best_t < 1.0:
        delta = (t_max - t_min) / n_samples
        for i in range(-5, 6):
            t = best_t + i * delta / 5.0
            if t < t_min or t > t_max:
                continue
            y_try = np.clip(y + t * direction, 0.0, 1.0)
            c_try = reduced_score(y_try)
            if c_try < best_val:
                best_val = c_try
                best_t = t
    
    return best_t, best_val

def lse_descent_reduced(y, beta=80.0, max_iters=200, alpha_init=0.005):
    """LSE-smoothed descent in reduced coordinates."""
    m = len(y)
    best_c = reduced_score(y)
    y_best = y.copy()
    alpha = alpha_init
    stuck = 0
    
    for it in range(max_iters):
        h = from_reduced(y)
        n = len(h)
        dx = 2.0 / n
        profile = compute_profile(h)
        scaled = beta * profile
        max_s = np.max(scaled)
        shifted = scaled - max_s
        weights = np.exp(shifted)
        wsum = weights.sum()
        if wsum < 1e-300:
            break
        weights /= wsum
        
        grad = reduced_gradient(y, h, profile, weights)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad /= gnorm
        
        # Line search
        best_t, best_val = exact_line_search_reduced(y, -grad, best_c)
        
        if best_val < best_c - 1e-14:
            y = np.clip(y + best_t * (-grad), 0.0, 1.0)
            y_best = y.copy()
            best_c = best_val
            stuck = 0
            alpha = min(alpha * 1.05, 0.05)
        else:
            stuck += 1
            alpha *= 0.7
            if stuck > 40 or alpha < 1e-10:
                break
    
    return y_best, best_c

def active_set_descent_reduced(y, max_iters=300, n_active=10):
    """Active set descent in reduced coordinates."""
    m = len(y)
    y = y.copy()
    best_c = reduced_score(y)
    stuck = 0
    
    for it in range(max_iters):
        h = from_reduced(y)
        n = len(h)
        dx = 2.0 / n
        profile = compute_profile(h)
        c = float(np.max(profile))
        if c + 1e-14 < best_c:
            best_c = c
            stuck = 0
        
        margin = max(0.002, 3.0 * (c - float(np.median(profile))))
        active = np.where(profile >= c - margin)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(profile))])
        if len(active) > n_active:
            top_idx = np.argsort(profile[active])[-n_active:]
            active = active[top_idx]
        
        weights = np.ones(len(active)) / len(active)
        grad = reduced_gradient(y, h, profile, weights)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad /= gnorm
        
        best_t, best_val = exact_line_search_reduced(y, -grad, best_c)
        
        if best_val < best_c - 1e-14:
            y = np.clip(y + best_t * (-grad), 0.0, 1.0)
            best_c = best_val
            stuck = 0
        else:
            stuck += 1
            if stuck > 50:
                break
    
    return y, best_c

def coordinate_descent_reduced(y, max_iters=500):
    """Cyclic coordinate descent in reduced coordinates."""
    m = len(y)
    y = y.copy()
    best_c = reduced_score(y)
    stuck = 0
    
    for it in range(max_iters):
        improved = False
        for j in range(m):
            direction = np.zeros(m)
            direction[j] = 1.0
            # Try both +1 and -1
            for sign in [1.0, -1.0]:
                d = sign * direction
                best_t, best_val = exact_line_search_reduced(y, d, best_c)
                if best_val < best_c - 1e-14:
                    y = np.clip(y + best_t * d, 0.0, 1.0)
                    best_c = best_val
                    improved = True
        if not improved:
            stuck += 1
            if stuck > 10:
                break
        else:
            stuck = 0
    
    return y, best_c

def pair_descent_reduced(y, max_iters=200, rng=None):
    """Randomized pair moves in reduced coordinates."""
    m = len(y)
    y = y.copy()
    best_c = reduced_score(y)
    stuck = 0
    
    for it in range(max_iters):
        i = rng.randint(m)
        j = rng.randint(m)
        if i == j:
            continue
        # Pair move: y[i] + t, y[j] - t
        direction = np.zeros(m)
        direction[i] = 1.0
        direction[j] = -1.0
        
        best_t, best_val = exact_line_search_reduced(y, direction, best_c)
        if best_val < best_c - 1e-14:
            y = np.clip(y + best_t * direction, 0.0, 1.0)
            best_c = best_val
            stuck = 0
        else:
            stuck += 1
            if stuck > 50:
                break
    
    return y, best_c

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = np.linspace(0, 2, n + 1)
    xt = np.linspace(0, 2, new_n + 1)
    h_interp = np.zeros(new_n)
    for k in range(new_n):
        lo_k, hi_k = xt[k], xt[k+1]
        total_mass = 0.0
        total_width = 0.0
        for j in range(n):
            lo_j, hi_j = xs[j], xs[j+1]
            overlap = min(hi_k, hi_j) - max(lo_k, lo_j)
            if overlap > 0:
                total_mass += h[j] * overlap
                total_width += overlap
        h_interp[k] = total_mass / total_width if total_width > 1e-15 else 0.5
    h_interp = np.clip(h_interp, 0.0, 1.0)
    return repair_feasible(h_interp)

def slsqp_reduced(y0, maxiter=200):
    """SLSQP in reduced coordinates with epigraph variable."""
    m = len(y0)
    y0 = y0.copy()
    c0 = reduced_score(y0)
    
    def obj(v):
        return v[-1]
    
    def obj_jac(v):
        g = np.zeros(m + 1)
        g[-1] = 1.0
        return g
    
    def ineq_fun(v):
        y_v = np.clip(v[:m], 0, 1)
        t = v[-1]
        h_v = from_reduced(y_v)
        profile = compute_profile(h_v)
        return t - profile
    
    def ineq_jac(v):
        y_v = np.clip(v[:m], 0, 1)
        h_v = from_reduced(y_v)
        n = len(h_v)
        dx = 2.0 / n
        G = build_gradient_matrix(h_v, n, dx)
        M = len(G)
        J = np.zeros((M, m + 1))
        # g_full = G.T @ weights, then g_reduced[j] = g_full[j] - g_full[n-1-j]
        gf = G.T  # shape (n, M), gf[i,k] = dP_k/dh_i
        # J[:, j] = -dP_k/dy_j = -(gf[k,j] - gf[k, n-1-j]) for each row k
        for j in range(m):
            J[:, j] = -(G[:, j] - G[:, n-1-j])
        J[:, -1] = 1.0
        return J
    
    x0 = np.concatenate([y0, [c0 + 1e-3]])
    bounds = [(0.0, 1.0)] * m + [(0.0, c0 + 0.5)]
    cons = [{'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}]
    
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None and np.isfinite(res.x).all():
            y_new = np.clip(res.x[:m], 0, 1)
            h_new = from_reduced(y_new)
            h_new = repair_feasible(h_new)
            c_new = exact_score(h_new)
            if c_new <= c0 + 1e-9:
                return y_new, h_new, c_new
    except Exception:
        pass
    return None, None, None
