def pair_line_evaluate(h, p, q, mu, n_samples=30):
    """Evaluate exact score along pair line, return (best_t, best_c)."""
    best_t = 0.0
    best_c = np.inf
    ts = np.linspace(0, mu, n_samples + 1)
    hp = h[p]
    hq = h[q]
    for t in ts:
        h[p] = hp - t
        h[q] = hq + t
        c = exact_score(h)
        if c < best_c:
            best_c = c
            best_t = t
    h[p] = hp
    h[q] = hq
    return best_t, best_c

def pair_refine(h, p, q, t_lo, t_hi, n_samples=20):
    """Refine pair line search in a sub-interval."""
    best_t = t_lo
    best_c = np.inf
    ts = np.linspace(t_lo, t_hi, n_samples + 1)
    hp = h[p]
    hq = h[q]
    for t in ts:
        h[p] = hp - t
        h[q] = hq + t
        c = exact_score(h)
        if c < best_c:
            best_c = c
            best_t = t
    h[p] = hp
    h[q] = hq
    return best_t, best_c

def do_pair_search(h, G, prof, active, max_pairs=80, n_coarse=25, n_fine=20):
    """Main pair search: generate candidates, score them, do exact line search on top.
    Returns (best_h, best_c) or (None, None) if no improvement."""
    n = len(h)
    current_c = prof.max()
    A = G[active, :]
    best_c = current_c
    best_h = None
    
    # Generate candidate pairs
    candidates = set()
    
    # Weight active constraints
    spread = prof[active].max() - prof[active].min()
    if spread > 1e-12:
        w = (prof[active] - prof[active].min()) / spread
        w = w ** 2 + 0.1
    else:
        w = np.ones(len(active))
    w = w / w.sum()
    
    # Gradient
    g = A.T @ w
    g -= g.mean()
    
    # Top/bottom gradient splits
    nq = max(5, n // 4)
    highs = set(np.argsort(g)[-nq:].tolist())
    lows = set(np.argsort(g)[:nq].tolist())
    for p in highs:
        for q in lows:
            if p != q:
                candidates.add((p, q))
    # Also reverse
    for p in lows:
        for q in highs:
            if p != q:
                candidates.add((p, q))
    
    # Per-constraint extremes
    for i in range(min(6, len(active))):
        row = A[i, :]
        p = int(np.argmax(row))
        q = int(np.argmin(row))
        if p != q:
            candidates.add((p, q))
            candidates.add((q, p))
    
    # Limit candidates
    if len(candidates) > max_pairs * 2:
        # Score and select
        scored = []
        for (p, q) in candidates:
            mu = min(h[p], 1.0 - h[q])
            if mu < 1e-8:
                continue
            V = A[:, q] - A[:, p]
            s = float(np.dot(w, V)) * mu
            if s < 0:  # descent direction
                scored.append((s, p, q, mu))
        scored.sort()
        candidates_list = scored[:max_pairs]
    else:
        candidates_list = []
        for (p, q) in candidates:
            mu = min(h[p], 1.0 - h[q])
            if mu >= 1e-8:
                V = A[:, q] - A[:, p]
                s = float(np.dot(w, V)) * mu
                if s < 0:
                    candidates_list.append((s, p, q, mu))
        candidates_list.sort()
        candidates_list = candidates_list[:max_pairs]
    
    if not candidates_list:
        return None, None
    
    # Exact line search on each candidate
    for (sc, p, q, mu) in candidates_list:
        # Coarse search
        t_est, c_est = pair_line_evaluate(h, p, q, mu, n_coarse)
        if t_est > 1e-10 and c_est < best_c - 1e-13:
            # Fine search around best
            step = mu / n_coarse
            t_fin, c_fin = pair_refine(h, p, q, max(0, t_est - step), min(mu, t_est + step), n_fine)
            if c_fin < best_c - 1e-13:
                best_c = c_fin
                h[p] -= t_fin
                h[q] += t_fin
                best_h = h.copy()
            else:
                # Restore (shouldn't need since we restore in pair_line_evaluate)
                pass
        # If best_c improved, some candidates might now be non-descent. Skip check for efficiency.
    
    if best_h is not None:
        return best_h, best_c
    return None, None

def grad_direction_step(h, G, prof, active, max_step=0.03):
    """Full-gradient direction with exact line search."""
    n = len(h)
    current_c = prof.max()
    A = G[active, :]
    
    # Weighted gradient
    spread = prof[active].max() - prof[active].min()
    if spread > 1e-12:
        w = (prof[active] - prof[active].min()) / spread
        w = w ** 2 + 0.1
    else:
        w = np.ones(len(active))
    w /= w.sum()
    
    g = A.T @ w
    g -= g.mean()
    mx = np.max(np.abs(g))
    if mx < 1e-12:
        return None, None
    d = -g / mx
    
    best_c = current_c
    best_h = None
    for alpha in [0.0005, 0.001, 0.003, 0.01, 0.02, max_step, max_step * 2, max_step * 5]:
        h_new = repair(h + alpha * d)
        c = exact_score(h_new)
        if c < best_c - 1e-13:
            best_c = c
            best_h = h_new
    return best_h, best_c

def select_active(prof, c, min_count=5, max_count=50):
    """Select active constraint indices."""
    M = len(prof)
    if M <= max_count:
        return np.arange(M)
    spread = c - prof.min()
    if spread < 1e-10:
        return np.argsort(prof)[-min_count:]
    for frac in [0.05, 0.1, 0.15, 0.25, 0.4, 0.6, 0.8]:
        thr = c - frac * spread
        idx = np.where(prof >= thr)[0]
        if len(idx) >= min_count and len(idx) <= max_count:
            return idx
    return np.argsort(prof)[-max_count:]

def interpolate_to(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) / n * 2.0
    xt = (np.arange(new_n) + 0.5) / new_n * 2.0
    h_new = np.interp(xt, xs, h)
    return repair(h_new)
