def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx] + 1e-15)
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx] + 1e-15)
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    return h

    m = n // 2

    for j in range(m):
        y[j] = 0.5 * (h[j] + (1.0 - h[n-1-j]))
    y = np.clip(y, 0.0, 1.0)
    h_anti = build_anti_full(y, n)
    return h_anti

    m = n // 2

        h[j] = y[j]
        h[n-1-j] = 1.0 - y[j]

def interpolate_anti(h_old, n_old, n_new):
    """Interpolate h_old to n_new points, preserving area."""
    if n_old == n_new:
        return h_old.copy()
    xs_old = np.linspace(0, 2.0, n_old + 1)
    xs_new = np.linspace(0, 2.0, n_new + 1)
    h_new = np.zeros(n_new)
    for k in range(n_new):
        lo, hi = xs_new[k], xs_new[k+1]
        mass = 0.0
        width = 0.0
        for j in range(n_old):
            lo_j, hi_j = xs_old[j], xs_old[j+1]
            overlap = min(hi, hi_j) - max(lo, lo_j)
            if overlap > 0:
                mass += h_old[j] * overlap
                width += overlap
        h_new[k] = mass / width if width > 1e-15 else 0.5
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

    # t ranges over [-y[j], 1-y[j]]
    t_min = -y[j]
    t_max = 1.0 - y[j]
    if t_max - t_min < 1e-14:
        return 0.0, exact_score(build_anti_full(y, n))
    
    # Sample on a grid
    t_vals = np.linspace(t_min, t_max, t_grid_size)
    best_t = 0.0
    best_c = exact_score(build_anti_full(y, n))

    m = n // 2
    mirror = n - 1 - j

        h_try[j] = y[j] + t
        h_try[mirror] = 1.0 - y[j] - t
        if h_try[j] < -1e-14 or h_try[j] > 1+1e-14 or h_try[mirror] < -1e-14 or h_try[mirror] > 1+1e-14:

        h_try = np.clip(h_try, 0, 1)

            best_t = t

    m = n // 2
    c = exact_score(build_anti_full(y, n))

        improved = False
        for j in range(m):

                if new_c < c - 1e-12:
                    c = new_c
                    improved = True
        if not improved:

import numpy as np
from scipy.optimize import minimize
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx] + 1e-15)
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx] + 1e-15)
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    return h

def build_anti_full(y, n):
    m = n // 2
    h = np.zeros(n)
    h[:m] = y
    h[m:] = 1.0 - y[::-1]
    return h

def project_anti(h, n):
    m = n // 2
    y = np.zeros(m)
    for j in range(m):
        y[j] = 0.5 * (h[j] + (1.0 - h[n - 1 - j]))
    y = np.clip(y, 0.0, 1.0)
    return y

def interpolate_area(h_old, n_old, n_new):
    if n_old == n_new:
        return h_old.copy()
    xs_old = np.linspace(0.0, 2.0, n_old + 1)
    xs_new = np.linspace(0.0, 2.0, n_new + 1)
    h_new = np.zeros(n_new)
    for k in range(n_new):
        lo, hi = xs_new[k], xs_new[k + 1]
        mass = 0.0
        width = 0.0
        for j in range(n_old):
            lo_j, hi_j = xs_old[j], xs_old[j + 1]
            ov = min(hi, hi_j) - max(lo, lo_j)
            if ov > 0:
                mass += h_old[j] * ov
                width += ov
        h_new[k] = mass / width if width > 1e-15 else 0.5
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

def grad_profile_lags(h, n, lags):
    """Compute dP_s/dh[j] for given lags. Returns array of shape (len(lags), n)."""
    m = len(lags)
    dx = 2.0 / n
    G = np.zeros((m, n))
    for idx, s in enumerate(lags):
        if s == 0:
            G[idx, :] = dx * (1.0 - 2.0 * h)
        elif s > 0:
            valid1 = np.arange(0, n - s)
            for j in range(n - s + 1):
                if j <= n - 1 - s:
                    G[idx, j] += dx * (1.0 - h[j + s])
                if j - s >= 0 and j - s <= n - 1 - s:
                    G[idx, j] -= dx * h[j - s]
        else:
            r = -s
            for j in range(n - r + 1):
                if r <= j <= n - 1:
                    G[idx, j] += dx * (1.0 - h[j - r])
                if 0 <= j <= n - 1 - r:
                    G[idx, j] -= dx * h[j + r]
    return G

def grad_reduced(h, n, lags, weights):
    """Gradient of weighted profile w.r.t. reduced variables y."""
    m = n // 2
    G = grad_profile_lags(h, n, lags)
    w = np.asarray(weights, dtype=np.float64)
    w = w / w.sum()
    g_full = G.T @ w
    g_red = np.zeros(m)
    for j in range(m):
        g_red[j] = g_full[j] - g_full[n - 1 - j]
    return g_red

def soft_max_eps(h, n, beta):
    """Smooth max of profile using log-sum-exp."""
    prof = compute_profile(h)
    scaled = beta * prof
    mx = np.max(scaled)
    return (mx + np.log(np.sum(np.exp(scaled - mx)))) / beta

def soft_max_grad(h, n, beta):
    """Gradient of smooth max w.r.t. h."""
    prof = compute_profile(h)
    scaled = beta * prof
    mx = np.max(scaled)
    w = np.exp(scaled - mx)
    w /= w.sum()
    M = 2 * n - 1
    lags = np.arange(M) - (n - 1)
    G = grad_profile_lags(h, n, lags)
    return G.T @ w

def line_search_single(y, j, n, grid=50):

    t_min = -y[j]
    t_max = 1.0 - y[j]
    if t_max - t_min < 1e-14:

    h_base = build_anti_full(y, n)

    best_t = 0.0

        h_try = np.clip(h_try, 0.0, 1.0)
        c = exact_score(h_try)
        if c < best_c - 1e-15:
            best_c = c
            best_t = t
    if best_t > t_min + 1e-14 and best_t < t_max - 1e-14:
        spread = (t_max - t_min) / grid
        for _ in range(6):
            t_lo = max(t_min, best_t - spread)
            t_hi = min(t_max, best_t + spread)
            t_fine = np.linspace(t_lo, t_hi, 20)

                h_try = h_base.copy()
                h_try[j] = y[j] + t

                c = exact_score(h_try)

                    best_t = t
            spread *= 0.4
    return best_t, best_c

def line_search_pair(y, i, j, n, grid=30):
    """Exact line search on pair y[i]+=t, y[j]-=t."""
    if i == j:
        return 0.0, exact_score(build_anti_full(y, n))
    t_min = max(-y[i], y[j] - 1.0)
    t_max = min(1.0 - y[i], y[j])
    if t_max - t_min < 1e-14:
        return 0.0, exact_score(build_anti_full(y, n))
    h_base = build_anti_full(y, n)
    mi = n - 1 - i
    mj = n - 1 - j
    best_t = 0.0
    best_c = exact_score(h_base)
    for t in np.linspace(t_min, t_max, grid):
        h_try = h_base.copy()
        h_try[i] = y[i] + t

        h_try[mj] = 1.0 - y[j] + t

def slsqp_reduced(y0, n, maxiter=200):

    m = n // 2
    h0 = build_anti_full(y0, n)
    c0 = exact_score(h0)
    dx = 2.0 / n
    M = 2 * n - 1
    
    def obj(yv):
        return exact_score(build_anti_full(yv, n))
    
    res = minimize(
        lambda yv: exact_score(build_anti_full(yv, n)),
        y0,
        method='L-BFGS-B',
        bounds=[(0.0, 1.0)] * m,
        options={'maxiter': maxiter, 'ftol': 1e-12}
    )
    c_lbfgs = exact_score(build_anti_full(res.x, n)) if res.x is not None else c0
    best_y = y0.copy()
    best_c = c0
    if c_lbfgs < best_c:
        best_y = np.clip(res.x, 0.0, 1.0)
        best_c = c_lbfgs

    h_best = build_anti_full(best_y, n)
    prof = compute_profile(h_best)
    top_k = np.argsort(prof)[::-1][:min(20, M)]

        t = x[-1]
        h = build_anti_full(yv, n)
        p = compute_profile(h)
        viol = np.max(p[top_k] - t)
        return t + 0.01 * max(0.0, viol) ** 2
    
    x0 = np.concatenate([best_y, [best_c + 0.001]])
    res2 = minimize(slsqp_obj, x0, method='SLSQP',
                    bounds=[(0.0, 1.0)] * m + [(0.0, best_c + 0.5)],
                    options={'maxiter': maxiter, 'ftol': 1e-12})
    if res2.x is not None:
        c_sol = exact_score(build_anti_full(np.clip(res2.x[:-1], 0, 1), n))
        if c_sol < best_c:
            best_y = np.clip(res2.x[:-1], 0.0, 1.0)
            best_c = c_sol
    
    return best_y, best_c

def perturb_anti(y, n, rng, eps):
    """Random perturbation in reduced space."""
    y_new = y + eps * rng.randn(len(y))
    return np.clip(y_new, 0.0, 1.0)

def break_symmetry(h_anti, n, rng, eps):
    """Break anti-symmetry slightly."""
    h = h_anti.copy()
    m = n // 2
    perturb = eps * rng.randn(m)
    for j in range(m):
        h[j] += perturb[j]
        h[n - 1 - j] -= perturb[j]
    return repair_feasible(np.clip(h, 0.0, 1.0))

def full_gradient_descent(h, n, max_iters=200, beta=80.0):
    """Gradient descent in full space with mass constraint."""
    m = n
    best_c = exact_score(h)
    best_h = h.copy()
    alpha = 0.003
    stuck = 0
    for it in range(max_iters):
        g = soft_max_grad(h, n, beta)
        g -= g.mean()
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn
        found = False
        for a in [alpha, alpha * 0.5, alpha * 0.25, alpha * 0.1]:
            h_try = repair_feasible(h - a * g)
            c_try = exact_score(h_try)
            if c_try < best_c - 1e-14:
                h = h_try
                best_c = c_try
                best_h = h.copy()
                found = True
                alpha = min(alpha * 1.05, 0.02)
                stuck = 0
                break
            elif c_try < exact_score(h) - 1e-14:
                h = h_try
                found = True
                break
        if not found:
            alpha *= 0.5
            stuck += 1
            if stuck > 40 or alpha < 1e-10:
                break
    return best_h, best_c

def pair_move_full(h, i, j, n, grid=30):
    """Mass-preserving pair move in full space."""
    t_min = max(-h[i], h[j] - 1.0)
    t_max = min(1.0 - h[i], h[j])
    if t_max - t_min < 1e-14:
        return h, exact_score(h)
    best_t = 0.0
    best_c = exact_score(h)
    for t in np.linspace(t_min, t_max, grid):
        h_try = h.copy()
        h_try[i] += t
        h_try[j] -= t
        h_try = np.clip(h_try, 0.0, 1.0)
        c = exact_score(h_try)
        if c < best_c - 1e-15:
            best_c = c
            best_t = t
    h_new = h.copy()
    h_new[i] += best_t
    h_new[j] -= best_t
    return h_new, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()
    rng = np.random.RandomState(seed)
    
    def time_left():
        return budget_s + t_start - time.perf_counter()
    
    # Get parent
    parent = None
    try:
        parent = np.array(initial_h_values, dtype=np.float64)
    except NameError:
        pass
    except Exception:
        pass
    
    if parent is None or len(parent) < 4:
        parent = np.full(70, 0.5)
    
    parent = np.clip(parent, 0.0, 1.0)
    parent = repair_feasible(parent)
    n_parent = len(parent)
    
    # Ensure even length
    if n_parent % 2 != 0:
        parent = parent[:n_parent - 1]
        parent = repair_feasible(parent)
        n_parent = len(parent)
    
    best_h = parent.copy()
    best_c = exact_score(parent)
    
    # Project to anti-symmetric
    n = n_parent
    m = n // 2
    y = project_anti(parent, n)
    h_anti = build_anti_full(y, n)
    c_anti = exact_score(h_anti)
    
    # Stage 1: Coordinate descent in anti-symmetric space
    if time_left() > 100:
        y, c_cd = coord_descent(y, n, max_sweeps=30, grid=50)
        if c_cd < c_anti:
            c_anti = c_cd
            h_anti = build_anti_full(y, n)
    
    # Stage 2: Gradient descent in anti-symmetric space
    if time_left() > 80:
        y, c_gd = gradient_descent_anti(y, n, max_iters=300, beta=100.0)
        if c_gd < c_anti:
            c_anti = c_gd
            h_anti = build_anti_full(y, n)
    
    # Stage 3: Active set descent
    if time_left() > 80:
        for na in [6, 10, 15]:
            if time_left() < 30:
                break
            y, c_as = descent_active_set(y, n, max_iters=200, n_active=na)
            if c_as < c_anti:
                c_anti = c_as
                h_anti = build_anti_full(y, n)
    
    # Stage 4: Pair descent
    if time_left() > 80:
        y, c_pd = pair_descent(y, n, max_sweeps=15, grid=25)
        if c_pd < c_anti:
            c_anti = c_pd
            h_anti = build_anti_full(y, n)
    
    # Stage 5: L-BFGS-B + SLSQP polish
    if time_left() > 60:
        y, c_opt = slsqp_reduced(y, n, maxiter=200)
        if c_opt < c_anti:
            c_anti = c_opt
            h_anti = build_anti_full(y, n)
    
    # Update global best
    if c_anti < best_c:
        best_c = c_anti
        best_h = h_anti.copy()
    
    # Stage 6: Resolution exploration
    if time_left() > 100:
        n_best = len(best_h)
        for factor in [0.85, 0.9, 0.95, 1.05, 1.1, 1.15, 1.25, 1.4]:
            if time_left() < 40:
                break
            new_n = int(round(n_best * factor / 2)) * 2
            new_n = max(28, min(new_n, 140))
            if new_n == n_best:
                continue
            h_interp = interpolate_area(best_h, n_best, new_n)
            y_new = project_anti(h_interp, new_n)
            y_new, c_new = coord_descent(y_new, new_n, max_sweeps=15, grid=40)
            y_new, c_new = gradient_descent_anti(y_new, new_n, max_iters=200, beta=80.0)
            if c_new < best_c - 1e-12:
                best_c = c_new
                best_h = build_anti_full(y_new, new_n)
            y_new, c_new = descent_active_set(y_new, new_n, max_iters=150, n_active=8)
            if c_new < best_c - 1e-12:
                best_c = c_new
                best_h = build_anti_full(y_new, new_n)
    
    # Stage 7: Break symmetry and optimize in full space
    if time_left() > 100:
        n_b = len(best_h)
        for eps in [0.01, 0.02, 0.05]:
            if time_left() < 30:
                break
            h_broken = break_symmetry(best_h, n_b, rng, eps)
            h_full, c_full = full_gradient_descent(h_broken, n_b, max_iters=200, beta=80.0)
            if c_full < best_c - 1e-12:
                best_c = c_full
                best_h = h_full.copy()
    
    # Stage 8: Full-space pair moves
    if time_left() > 80:
        n_b = len(best_h)
        h_work = best_h.copy()
        for iteration in range(5):
            if time_left() < 20:
                break
            improved = False
            order = list(range(n_b))
            rng.shuffle(order)
            for i in order[:min(100, n_b)]:
                for j in order[:min(100, n_b)]:
                    if i == j:
                        continue
                    h_work, c_try = pair_move_full(h_work, i, j, n_b, grid=20)
                    if c_try < best_c - 1e-12:
                        best_c = c_try
                        best_h = h_work.copy()
                        improved = True
                        break
                if improved:
                    break
            if not improved:
                break
    
    # Stage 9: Multi-start in anti-symmetric space
    if time_left() > 100:
        n_b = len(best_h)
        if n_b % 2 == 0:
            y_base = project_anti(best_h, n_b)
            for trial in range(5):
                if time_left() < 30:
                    break
                y_pert = perturb_anti(y_base, n_b, rng, 0.03 + 0.02 * trial)
                y_pert, c_pert = coord_descent(y_pert, n_b, max_sweeps=10, grid=40)
                y_pert, c_pert = gradient_descent_anti(y_pert, n_b, max_iters=150, beta=100.0)
                y_pert, c_pert = slsqp_reduced(y_pert, n_b, maxiter=100)
                if c_pert < best_c - 1e-12:
                    best_c = c_pert
                    best_h = build_anti_full(y_pert, n_b)
    
    # Stage 10: Intensive coordinate descent on best
    if time_left() > 60:
        n_b = len(best_h)
        if n_b % 2 == 0:
            y_b = project_anti(best_h, n_b)
            for sweep_round in range(3):
                if time_left() < 20:
                    break
                y_b, c_cd = coord_descent(y_b, n_b, max_sweeps=30, grid=60)
                if c_cd < best_c - 1e-13:
                    best_c = c_cd
                    best_h = build_anti_full(y_b, n_b)
                y_b, c_pd = pair_descent(y_b, n_b, max_sweeps=8, grid=30)
                if c_pd < best_c - 1e-13:
                    best_c = c_pd
                    best_h = build_anti_full(y_b, n_b)
    
    # Stage 11: Final SLSQP/L-BFGS polish
    if time_left() > 40:
        n_b = len(best_h)
        if n_b % 2 == 0:
            y_b = project_anti(best_h, n_b)
            y_b, c_opt = slsqp_reduced(y_b, n_b, maxiter=300)
            if c_opt < best_c - 1e-13:
                best_c = c_opt
                best_h = build_anti_full(y_b, n_b)
    
    # Stage 12: Final gradient descent polish (full space if not anti)
    if time_left() > 30:
        n_b = len(best_h)
        h_pol, c_pol = full_gradient_descent(best_h, n_b, max_iters=100, beta=200.0)
        if c_pol < best_c - 1e-13:
            best_c = c_pol
            best_h = h_pol.copy()
    
    # Stage 13: Very fine coordinate descent
    if time_left() > 25:
        n_b = len(best_h)
        if n_b % 2 == 0:
            y_b = project_anti(best_h, n_b)
            for j in range(n_b // 2):
                if time_left() < 5:
                    break
                t, nc = line_search_single(y_b, j, n_b, grid=80)
                if abs(t) > 1e-12:
                    y_b[j] = np.clip(y_b[j] + t, 0.0, 1.0)
            c_final_cd = exact_score(build_anti_full(y_b, n_b))
            if c_final_cd < best_c - 1e-14:
                best_c = c_final_cd
                best_h = build_anti_full(y_b, n_b)
    
    # Final validation
    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)
    
    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            final_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 2:
            final_c = float(eval_result[0])
    except Exception:
        pass
    
    return best_h, final_c, len(best_h)
