```python
import numpy as np
from scipy.optimize import minimize, linprog
import time
import cvxpy as cp

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full"))) * dx

def profile_h(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def compute_gradient_matrix(h, dx):
    """Compute G where G[m, k] = dP[m]/dh[k] for all m, k."""
    n = len(h)
    M = 2 * n - 1
    G = np.zeros((M, n))
    for m in range(M):
        d = m - (n - 1)
        if d == 0:
            G[m, :] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            # P[m] = dx * sum_{j=0}^{n-1-d} h[j+d]*(1-h[j])
            for j in range(n - d):
                G[m, j + d] += dx * (1.0 - h[j])
                G[m, j] -= dx * h[j + d]
        else:
            p = -d
            # P[m] = dx * sum_{j=0}^{n-1-p} h[j]*(1-h[j+p])
            for j in range(n - p):
                G[m, j] += dx * (1.0 - h[j + p])
                G[m, j + p] -= dx * h[j]
    return G

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx])
            h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def slp_iteration(h_old, use_cvxpy=True):
    """One SLP iteration: solve LP to get direction, then line search."""
    n = len(h_old)
    dx = 2.0 / n
    M = 2 * n - 1
    G = compute_gradient_matrix(h_old, dx)
    P_old = profile_h(h_old)
    
    # LP: min t
    # s.t. G @ h - t <= -P_old + G @ h_old  (for each m)
    #      0 <= h <= 1
    #      sum(h) = n/2
    
    if use_cvxpy:
        return _slp_cvxpy(h_old, G, P_old, n, dx)
    else:
        return _slp_linprog(h_old, G, P_old, n, dx)

def _slp_cvxpy(h_old, G, P_old, n, dx):
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    
    # Linearized constraint: P_old + G @ (h_var - h_old) <= t_var
    # i.e., G @ h_var - t_var <= -P_old + G @ h_old
    rhs = -P_old + G @ h_old
    constraints = [
        G @ h_var - t_var <= rhs,
        h_var >= 0,
        h_var <= 1,
        cp.sum(h_var) == n / 2.0
    ]
    obj = cp.Minimize(t_var)
    prob = cp.Problem(obj, constraints)
    
    try:
        prob.solve(solver=cp.ECOS, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.CVXOPT, verbose=False)
        except Exception:
            try:
                prob.solve(verbose=False)
            except Exception:
                return None
    
    if h_var.value is None:
        return None
    h_lp = h_var.value
    h_lp = repair_feasible(h_lp)
    return h_lp

def _slp_linprog(h_old, G, P_old, n, dx):
    """Use scipy linprog as alternative."""
    # Variables: [h_0, ..., h_{n-1}, t]
    # Minimize: [0, ..., 0, 1] @ x
    # Subject to: G @ h - t <= -P_old + G @ h_old
    #             -1 <= -h <= 0  (i.e., 0 <= h <= 1)
    #             sum(h) = n/2
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Inequality constraints: A_ub @ x <= b_ub
    # G @ h - t <= rhs  ->  [G, -I] @ [h; t] <= rhs
    rhs = -P_old + G @ h_old
    A_ub = np.hstack([G, -np.eye(len(rhs))])
    b_ub = rhs
    
    # h >= 0: -h <= 0 -> [-I, 0] @ [h; t] <= 0
    A_neg = np.hstack([-np.eye(n), np.zeros((n, 1))])
    b_neg = np.zeros(n)
    
    # h <= 1: h <= 1 -> [I, 0] @ [h; t] <= 1
    A_pos = np.hstack([np.eye(n), np.zeros((n, 1))])
    b_pos = np.ones(n)
    
    A_ub = np.vstack([A_ub, A_neg, A_pos])
    b_ub = np.concatenate([b_ub, b_neg, b_pos])
    
    # Equality: sum(h) = n/2
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      method='highs', options={'presolve': True})
        if res.success:
            h_lp = res.x[:n]
            h_lp = repair_feasible(h_lp)
            return h_lp
    except Exception:
        pass
    return None

def slp_line_search(h_old, h_dir, max_alpha=1.0, min_alpha=1e-4):
    """Line search between h_old and h_dir."""
```

```python
import numpy as np
from scipy.optimize import linprog
import time

# --- Helper functions ---

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full"))) * dx

def profile_h(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def compute_G(h, dx):
    n = len(h)
    M = 2 * n - 1
    G = np.zeros((M, n))
    for m in range(M):
        d = m - (n - 1)
        if d == 0:
            G[m, :] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            for j in range(n - d):
                G[m, j + d] += dx * (1.0 - h[j])
                G[m, j] -= dx * h[j + d]
        else:
            p = -d
            for j in range(n - p):
                G[m, j] += dx * (1.0 - h[j + p])
                G[m, j + p] -= dx * h[j]
    return G

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx])
            h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    h_new = np.interp(xt, xs, h)
    return repair_feasible(h_new)

# --- SLP ---

def solve_lp_step(h_old):
    n = len(h_old)
    dx = 2.0 / n
    M = 2 * n - 1
    G = compute_G(h_old, dx)
    P_old = profile_h(h_old)
    
    # Variables: [h (n), t (1)] => n+1 variables
    # Minimize: t
    # Subject to: G @ h - t <= -P_old + G @ h_old
    #             0 <= h <= 1
    #             sum(h) = n/2
    
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize t (last variable)
    
    # Inequality: G @ h - t <= rhs
    rhs = -P_old + G @ h_old
    A_ub1 = np.hstack([G, -np.eye(M)])
    b_ub1 = rhs
    
    # h >= 0: -h <= 0
    A_ub2 = np.hstack([-np.eye(n), np.zeros((n, 1))])
    b_ub2 = np.zeros(n)
    
    # h <= 1: h <= 1
    A_ub3 = np.hstack([np.eye(n), np.zeros((n, 1))])
    b_ub3 = np.ones(n)
    
    A_ub = np.vstack([A_ub1, A_ub2, A_ub3])
    b_ub = np.concatenate([b_ub1, b_ub2, b_ub3])
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      method='highs')
        if res.success and res.x is not None:
            h_lp = repair_feasible(res.x[:n])
            return h_lp
    except Exception:
        pass
    
    # Fallback: try interior-point
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      method='interior-point')
        if res.success and res.x is not None:
            h_lp = repair_feasible(res.x[:n])
            return h_lp
    except Exception:
        pass
    
    return None

def slp_optimize(h0, max_iters=500, t_start=None, budget_s=1100):

    best_h = h.copy()
    best_c = exact_score(h)
    stagnant = 0

                break
        
        h_lp = solve_lp_step(h)
        if h_lp is None:

            if stagnant > 5:
                break
            continue
        
        # Line search
        improved = False
        for a in [1.0, 0.7, 0.5, 0.35, 0.2, 0.1, 0.05, 0.02, 0.01]:
            h_try = repair_feasible((1.0 - a) * h + a * h_lp)
            c_try = exact_score(h_try)
            if c_try < best_c - 1e-14:
                h = h_try
                best_h = h_try.copy()
                best_c = c_try
                improved = True
                break
            elif c_try < exact_score(h) - 1e-14:
                h = h_try
                if c_try < best_c:
                    best_h = h_try.copy()
                    best_c = c_try
                improved = True
        
        if improved:
            stagnant = 0
        else:
            stagnant += 1
            if stagnant > 20:
                # Try perturbation

                if stagnant > 5:
                    break
    
    return best_h, best_c

def pair_swap_optimize(h, max_sweeps=30, t_start=None, budget_s=1100):
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    
    for sweep in range(max_sweeps):
        if t_start is not None and time.perf_counter() - t_start > budget_s - 15:
            break
        improved = False
        for i in range(n):
            if t_start is not None and time.perf_counter() - t_start > budget_s - 15:
                break
            for j in range(i + 1, n):
                for sign in [1, -1]:
                    for frac in [0.005, 0.002, 0.001]:
                        if sign > 0:
                            # Move from i to j
                            if h[i] < frac or h[j] > 1.0:
                                continue
                            old_i, old_j = h[i], h[j]
                            h[i] = old_i - frac
                            h[j] = old_j + frac
                        else:
                            # Move from j to i
                            if h[j] < frac or h[i] > 1.0:
                                continue
                            old_i, old_j = h[i], h[j]
                            h[j] = old_j - frac
                            h[i] = old_i + frac
                        
                        c = exact_score(h)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                            break
                        else:
                            h[i], h[j] = old_i, old_j
                    else:
                        continue
                    break
        if not improved:
            break
    
    return h, best_c
```

```python
from scipy.optimize import linprog

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full"))) * dx

def profile_h(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def compute_G(h, dx):
    n = len(h)
    M = 2 * n - 1
    G = np.zeros((M, n))
    for m in range(M):
        d = m - (n - 1)
        if d == 0:
            G[m, :] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            for j in range(n - d):
                G[m, j + d] += dx * (1.0 - h[j])
                G[m, j] -= dx * h[j + d]
        else:
            p = -d
            for j in range(n - p):
                G[m, j] += dx * (1.0 - h[j + p])
                G[m, j + p] -= dx * h[j]
    return G

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx])
            h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))

def solve_lp_step(h_old):

    dx = 2.0 / n
    M = 2 * n - 1
    G = compute_G(h_old, dx)

    c[n] = 1.0
    rhs = -P_old + G @ h_old
    A_ub1 = np.hstack([G, -np.eye(M)])
    b_ub1 = rhs
    A_ub2 = np.hstack([-np.eye(n), np.zeros((n, 1))])
    b_ub2 = np.zeros(n)
    A_ub3 = np.hstack([np.eye(n), np.zeros((n, 1))])
    b_ub3 = np.ones(n)
    A_ub = np.vstack([A_ub1, A_ub2, A_ub3])
    b_ub = np.concatenate([b_ub1, b_ub2, b_ub3])
    A_eq = np.zeros((1, n + 1))
```
