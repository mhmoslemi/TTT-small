        if time_left() < 35:
            break
        h_interp = interp_to_n(best_h, new_n)
        h_ad, c_ad = active_descent_v2(h_interp, max_iters=120, active_tol=0.004)
        if c_ad < best_c:
            h_sol = slsqp_smooth(h_ad, 0.005, maxiter=150)
            if h_sol is not None:
                c_sol = exact_score(h_sol)
                if c_sol < c_ad:
                    h_ad, c_ad = h_sol, c_sol
            h_ad2, c_ad2 = active_descent_v2(h_ad, max_iters=150, active_tol=0.002)
            if c_ad2 < c_ad:
                h_ad, c_ad = h_ad2, c_ad2
            if c_ad < best_c:
                best_h, best_c = h_ad, c_ad

    # Phase 4: Perturbation restarts
    for trial in range(15):
        if time_left() < 25:
            break
        scale = 0.015 * (1.0 + 0.3 * trial)
        h_pert = perturb_h(best_h, scale, rng)
        h_ad, c_ad = active_descent_v2(h_pert, max_iters=120, active_tol=0.003)
        if c_ad < best_c:
            best_h, best_c = h_ad, c_ad
            if time_left() > 40:
                h_sol = slsqp_smooth(best_h, 0.003, maxiter=100)
                if h_sol is not None:
                    c_sol = exact_score(h_sol)
                    if c_sol < best_c:
                        best_h, best_c = h_sol, c_sol

    # Phase 5: SLSQP at very small tau
    if time_left() > 40:
        h_sol = slsqp_smooth(best_h, 0.0008, maxiter=300)
        if h_sol is not None:
            c_sol = exact_score(h_sol)
            if c_sol < best_c:
                best_h, best_c = h_sol, c_sol

    # Phase 6: Final active descent with tight tolerance
    if time_left() > 30:
        h_ad, c_ad = active_descent_v2(best_h, max_iters=400, active_tol=0.0008, momentum=0.3)
        if c_ad < best_c:
            best_h, best_c = h_ad, c_ad

    # Phase 7: Coordinate descent polish
    if time_left() > 20:
        h_cd, c_cd = coordinate_descent(best_h, max_iters=40)
        if c_cd < best_c:
            best_h, best_c = h_cd, c_cd

    # Phase 8: Final SLSQP
    if time_left() > 15:
        h_sol = slsqp_smooth(best_h, 0.0005, maxiter=200)
        if h_sol is not None:
            c_sol = exact_score(h_sol)
            if c_sol < best_c:
                best_h, best_c = h_sol, c_sol

    # Phase 9: Ultra-fine active descent
    if time_left() > 10:
        h_ad, c_ad = active_descent_v2(best_h, max_iters=200, active_tol=0.0003, momentum=0.2)
        if c_ad < best_c:
            best_h, best_c = h_ad, c_ad

    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)

    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            final_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 2:
            final_c = float(eval_result[0])
    except Exception:
        pass

    return best_h, final_c, len(best_h)
