For improvement, I should try:

2. **Trust-region SLP** - use a linear programming approach to find multi-coordinate steps

4. **Adaptive beta schedule** for LSE that starts smooth and sharpens

Let me think about the LSE approach more carefully:

The objective is f(h) = max_m p_m(h) where p_m(h) = ∑_j h_j(1-h_{j+m}) * dx (after appropriate indexing).

The LSE approximation: f_β(h) = (1/β) log(Σ_m exp(β * p_m(h)))

Gradient: ∇f_β(h) = (1/Σ exp(β p_m)) * Σ_m β * exp(β p_m) * ∇p_m(h)
              = Σ_m w_m * ∇p_m(h) where w_m = exp(β p_m) / Σ exp(β p_m)

This gives a weighted combination of all row gradients, weighted by how close each row is to the maximum.

For the trust-region SLP approach:
- At point h, linearize: p_m(h + d) ≈ p_m(h) + ∇p_m(h) · d
- Minimize t subject to: p_m(h) + ∇p_m(h) · d ≤ t for all m
- Subject to: Σ d = 0, -R ≤ d_j ≤ R (with box constraints from [0,1])
- This is a linear program!

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    # Use water-filling via bisection on lambda
    # h_j = clip(y_j - lambda, 0, 1), find lambda so sum = target
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        z = np.clip(h - mid, 0.0, 1.0)
        s = z.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    mid = (lo + hi) / 2.0
    z = np.clip(h - mid, 0.0, 1.0)
    r = target - z.sum()
    if abs(r) > 1e-13:

            remaining = r
            for i in range(n):
                idx = order[i]
                cap = 1.0 - z[idx]
                if cap <= 1e-15:
                    continue
                add = min(remaining, cap)
                z[idx] += add
                remaining -= add

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def grad_row(h, m, n):
    dx = 2.0 / n
    d = m - (n - 1)
    if d == 0:
        return dx * (1.0 - 2.0 * h)
    elif d > 0:
        g = np.zeros(n)
        g[d:] += dx * (1.0 - h[:n-d])
        g[:n-d] -= dx * h[d:]
        return g
    else:
        p = -d
        g = np.zeros(n)
        g[:n-p] += dx * (1.0 - h[p:])
        g[p:] -= dx * h[:n-p]
        return g

def lse_gradient(h, beta):
    """Compute gradient of LSE approximation to max of profile."""
    n = len(h)
    profile = compute_profile(h)
    c = profile.max()
    z = beta * (profile - c)
    # Clip for numerical stability
    z = np.clip(z, -500, 0)
    exp_z = np.exp(z)
    w = exp_z / exp_z.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    M = len(profile)
    for m in range(M):
        if w[m] > 1e-10:
            g += w[m] * grad_row(h, m, n)
    return g, c

def lse_descent(h, max_iters=200, beta_start=5.0, beta_end=200.0):
    """LSE-based descent with adaptive beta."""
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    n_iters = max(10, max_iters)
    for it in range(n_iters):
        frac = it / n_iters
        beta = beta_start * (beta_end / beta_start) ** (frac ** 0.7)
        # Compute LSE gradient
        g, c_max = lse_gradient(h, beta)
        g -= g.mean()  # Project onto sum=0
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            break

        # Line search
        pos_mask = d > 1e-12
        neg_mask = d < -1e-12
        alpha_max = 0.5
        if pos_mask.any():
            alpha_max = min(alpha_max, float(((1.0 - h[pos_mask]) / d[pos_mask]).min()) * 0.9)
        if neg_mask.any():
            alpha_max = min(alpha_max, float((h[neg_mask] / (-d[neg_mask])).min()) * 0.9)
        alpha_max = max(alpha_max, 1e-8)
        # Backtracking line search
        c_current = exact_score(h)
        best_alpha = 0.0
        best_val = c_current
        best_h = h.copy()
        for ls_it in range(20):
            alpha = alpha_max * (0.5 ** ls_it)

                break

            f_try = exact_score(h_try)
            if f_try < best_val:
                best_val = f_try

                    best_alpha = alpha
                    best_h = h_try.copy()
            if best_alpha == 0.0:
                break
        h = best_h
        if best_val < best_c:
            best_c = best_val
    return h, best_c

def trust_region_slp(h, R=0.05, max_iters=100):
    """Trust-region sequential linear programming."""
    n = len(h)
    M = 2 * n - 1
    h = h.copy()
    best_c = exact_score(h)
    R_current = R
    for it in range(max_iters):
        profile = compute_profile(h)
        c = float(profile.max())
        # Build constraint matrix for LP
        # Variables: d_0, ..., d_{n-1}, t
        # Minimize t
        # Constraints: for each m: sum_j G[m,j]*d_j - t <= -profile[m]
        #              sum_j d_j = 0
        # Bounds: max(-R, -h[j]) <= d_j <= min(R, 1-h[j])
        
        # Only include rows that are "active" (near max) to keep LP small
        thresh = c - 0.01
        active = np.where(profile >= thresh)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(profile))])
        if len(active) > 30:
            top = np.argsort(profile[active])[-30:]
            active = active[top]
        
        M_a = len(active)
        # A_ub shape: (M_a, n+1)
        A_ub = np.zeros((M_a, n + 1))
        b_ub = np.zeros(M_a)
        for i, m in enumerate(active):
            A_ub[i, :n] = grad_row(h, m, n)
            A_ub[i, n] = -1.0
            b_ub[i] = -profile[m]
        
        # c vector for objective: minimize t => c = [0,...,0,1]
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0
        
        # A_eq: sum d_j = 0
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)
        
        # Bounds
        bounds = []
        for j in range(n):
            lo = max(-R_current, -h[j] + 1e-10)
            hi = min(R_current, 1.0 - h[j] - 1e-10)
            if lo > hi:
                lo = hi = 0.0
            bounds.append((lo, hi))
        bounds.append((0.0, c + 1.0))
        
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs')
            if res.success:
                d = res.x[:n]
                h_try = repair_feasible(h + d)
                c_try = exact_score(h_try)
                if c_try < best_c:
                    h = h_try
                    best_c = c_try
                    R_current = min(R_current * 1.3, 0.2)
                else:
                    R_current *= 0.6
            else:
                R_current *= 0.5
        except Exception:
            R_current *= 0.5
        
        if R_current < 0.001:
            break
    
    return h, best_c

def active_descent(h, max_iters=400, active_tol=0.003):

            h1 = repair_feasible(h + c1 * d)

            f1 = exact_score(h1)
            f2 = exact_score(h2)
            if f1 < best_val:
                best_val = f1
                best_alpha = c1
                best_h = h1.copy()
            if f2 < best_val:
                best_val = f2
                best_alpha = c2
                best_h = h2.copy()
            if f1 < f2:
                b_ls = c2
            else:
                a_ls = c1
            if b_ls - a_ls < 1e-13:
                break
        if best_val >= c - 1e-15:
            break
        h = best_h
        if best_val < best_c:
            best_c = best_val

def slsqp_epigraph(h_start, maxiter=200):
    n = len(h_start)
    dx = 2.0 / n
    M = 2 * n - 1
    c0 = exact_score(h_start)
    t0 = c0 + 1e-4
    
    def obj(x):
        return x[-1]
    
    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    
    def eq_fun(x):
        return x[:-1].sum() - n / 2.0
    
    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g
    
    def ineq_fun(x):
        h = x[:-1]
        t = x[-1]
        profile = np.correlate(h, 1 - h, mode="full") * dx
        return t - profile
    
    def ineq_jac(x):
        h = x[:-1]
        J = np.zeros((M, n + 1))
        for m in range(M):
            J[m, :n] = -grad_row(h, m, n)
        J[:, -1] = 1.0
        return J
    
    x0 = np.concatenate([h_start, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 2.0)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None and np.isfinite(res.x).all():
            h_new = repair_feasible(res.x[:-1])
            c_new = exact_score(h_new)
            if c_new <= c0 + 1e-8:
                return h_new
    except Exception:
        pass
    return None

def interp_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    edges_old = np.linspace(0.0, 2.0, n + 1)
    edges_new = np.linspace(0.0, 2.0, new_n + 1)
    h_new = np.zeros(new_n)
    for k in range(new_n):
        a, b = edges_new[k], edges_new[k + 1]
        total = 0.0
        for j in range(n):
            lo = max(a, edges_old[j])
            hi = min(b, edges_old[j + 1])
            if hi > lo:
                total += h[j] * (hi - lo)
        h_new[k] = total / (b - a)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

def binary_seed(n, pattern_type, rng):
    half = n // 2
    h = np.zeros(n)
    if pattern_type == 0:
        idx = np.arange(n)
        vals = (idx * 0.618033988749895) % 1.0
        top = np.argsort(vals)[-half:]
        h[top] = 1.0
    elif pattern_type == 1:
        idx = np.arange(n)
        vals = (idx * 0.4142135623730951) % 1.0
        top = np.argsort(vals)[-half:]
        h[top] = 1.0
    elif pattern_type == 2:
        period = max(3, n // 8)
        for i in range(n):
            if (i // period) % 2 == 0:
                h[i] = 1.0
        ones = int(h.sum())
        if ones > half:
            excess = ones - half
            one_idx = np.where(h == 1.0)[0]
            to_flip = rng.choice(one_idx, excess, replace=False)
            h[to_flip] = 0.0
        elif ones < half:
            deficit = half - ones
            zero_idx = np.where(h == 0.0)[0]
            to_flip = rng.choice(zero_idx, deficit, replace=False)
            h[to_flip] = 1.0
    elif pattern_type == 3:
        h = (np.sin(np.linspace(0, 6 * np.pi, n)) > 0).astype(float)
        ones = int(h.sum())
        if ones > half:
            excess = ones - half
            one_idx = np.where(h == 1.0)[0]
            to_flip = rng.choice(one_idx, excess, replace=False)
            h[to_flip] = 0.0
        elif ones < half:
            deficit = half - ones
            zero_idx = np.where(h == 0.0)[0]
            to_flip = rng.choice(zero_idx, deficit, replace=False)
            h[to_flip] = 1.0
    elif pattern_type == 4:
        # Triangular wave
        x = np.linspace(0, 2, n)
        h = 0.5 + 0.5 * np.sin(2 * np.pi * x)
        h = np.clip(h, 0, 1)
    else:
        perm = rng.permutation(n)
        h[perm[:half]] = 1.0
    return repair_feasible(h)

def coordinate_polish(h, max_iters=100):
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    for it in range(max_iters):
        profile = compute_profile(h)
        c = float(profile.max())
        thresh = c - 0.002
        active = np.where(profile >= thresh)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(profile))])
        improved = False
        idx_sorted_desc = np.argsort(-h)
        idx_sorted_asc = np.argsort(h)
        for attempt in range(30):
            i = int(idx_sorted_desc[attempt % n])
            j = int(idx_sorted_asc[attempt % n])
            if i == j:
                continue
            delta = 0.01 * (h[i] - h[j])
            if abs(delta) < 1e-6:
                continue
            for sign in [1.0, -1.0]:
                h_try = h.copy()
                h_try[i] += sign * delta
                h_try[j] -= sign * delta
                if h_try.min() < -1e-10 or h_try.max() > 1 + 1e-10:
                    continue
                h_try = repair_feasible(h_try)
                s = exact_score(h_try)
                if s < best_c - 1e-14:
                    h = h_try
                    best_c = s
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
    return h, best_c

def grad_row(h, m, n):

    d = m - (n - 1)

def slsqp_epigraph(h_start, maxiter=200):

    dx = 2.0 / n
    M = 2 * n - 1
    c0 = exact_score(h_start)
    t0 = c0 + 1e-4
    def obj(x):
        return x[-1]
    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    def eq_fun(x):
        return x[:-1].sum() - n / 2.0
    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g
    def ineq_fun(x):
        h = x[:-1]
        t = x[-1]
        profile = np.correlate(h, 1 - h, mode="full") * dx
        return t - profile
    def ineq_jac(x):
        h = x[:-1]
        J = np.zeros((M, n + 1))
        for m in range(M):
            J[m, :n] = -grad_row(h, m, n)
        J[:, -1] = 1.0
        return J
    x0 = np.concatenate([h_start, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 2.0)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None and np.isfinite(res.x).all():
            h_new = repair_feasible(res.x[:-1])
            c_new = exact_score(h_new)

    return None

def interp_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    edges_old = np.linspace(0.0, 2.0, n + 1)
    edges_new = np.linspace(0.0, 2.0, new_n + 1)
    h_new = np.zeros(new_n)
    for k in range(new_n):
        a, b = edges_new[k], edges_new[k + 1]
        total = 0.0
        for j in range(n):
            lo = max(a, edges_old[j])
            hi = min(b, edges_old[j + 1])
            if hi > lo:
                total += h[j] * (hi - lo)
        h_new[k] = total / (b - a)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

    if time_left() > 20:
        h_sol = slsqp_epigraph(best_h, maxiter=150)
        if h_sol is not None:
            c_sol = exact_score(h_sol)
            if c_sol < best_c:
                best_h, best_c = h_sol, c_sol

    # Phase 14: Final LSE polish
    if time_left() > 15:
        h_lse, c_lse = lse_descent(best_h, max_iters=60, beta_start=80.0, beta_end=800.0)
        if c_lse < best_c:
            best_h, best_c = h_lse, c_lse

    # Phase 15: Final trust-region
    if time_left() > 15:
        h_tr, c_tr = trust_region_slp(best_h, R=0.01, max_iters=30)
        if c_tr < best_c:
            best_h, best_c = h_tr, c_tr

    # Final polish
    if time_left() > 10:
        h_ad, c_ad = active_descent(best_h, max_iters=80, active_tol=0.0005)
        if c_ad < best_c:
            best_h, best_c = h_ad, c_ad

    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)

    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            final_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 2:
            final_c = float(eval_result[0])
    except Exception:
        pass

    return best_h, final_c, len(best_h)
