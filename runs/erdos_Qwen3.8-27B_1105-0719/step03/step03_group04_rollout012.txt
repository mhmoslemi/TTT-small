def pair_swap_optimize(h, p, q, profile, G, dx):
    """Find optimal t for decreasing h[p] by t and increasing h[q] by t."""
    mu = min(h[p], 1.0 - h[q])
    if mu <= 1e-15:
        return 0.0, None
    
    delta_G = G[:, q] - G[:, p]
    # Quadratic terms at lags j1 = center + (q-p) and j2 = center - (q-p)
    center = len(h) - 1
    diff = q - p
    j1 = center + diff
    j2 = center - diff
    # Q is nonzero at j1 and j2
    quad = np.zeros(len(profile))
    quad[j1] = dx
    quad[j2] = dx
    
    def f(t):
        return np.max(profile + t * delta_G + t * t * quad)
    
    # Golden section search on [0, mu]
    gr = (np.sqrt(5.0) - 1.0) / 2.0  # golden ratio ≈ 0.618
    a, b = 0.0, mu
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = f(c)
    fd = f(d)
    for _ in range(60):
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = f(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = f(d)
    t_opt = (a + b) / 2.0
