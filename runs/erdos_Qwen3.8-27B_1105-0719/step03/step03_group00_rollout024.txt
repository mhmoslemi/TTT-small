import numpy as np
from scipy.optimize import linprog
import time

def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def c5_of_h(h, dx):
    return np.correlate(h, 1.0 - h, mode='full').max() * dx

def corr_of_h(h):
    return np.correlate(h, 1.0 - h, mode='full')

def grad_lag(h, k, n):
    """Gradient of corr_k with respect to h."""
    off = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -off)
    i_hi = min(n - 1, n - 1 - off)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx + off] += (1.0 - h[idx])
    grad[idx] += (-h[idx + off])
    return grad

def solve_soft_slack_lp(h, dx, target_sum, alpha, top_idx, n):
    """Solve a soft-slack LP to find a descent direction."""
    corrs = corr_of_h(h)
    max_c = corrs.max()
    
    # Select active set: top lags + near-top
    margin = max(0.01, 0.05 * max_c)
    active_mask = corrs >= (max_c - margin)
    active_mask[top_idx] = True
    active_idx = np.where(active_mask)[0]
    
    K = len(active_idx)
    if K < 3:
        return None
    
    # Build gradients for active lags
    grads = np.zeros((K, n))
    for j, k in enumerate(active_idx):
        grads[j] = grad_lag(h, k, n)
    
    # Soft slack LP:
    # Variables: delta (n), slack (K), q (1)
    # min: rho_q * q + sum(rho_s_j * s_j)
    # s.t.: dx * grad_j . delta + s_j - q <= -r_j  for j in active
    #       sum(delta) = 0
    #       -h_i * alpha <= delta_i <= (1-h_i) * alpha
    #       s_j >= 0
    
    rho_q = 0.5
    rho_s = np.ones(K) * 3.0  # uniform slack penalty
    
    m_vars = n + K + 1
    c_obj = np.zeros(m_vars)
    c_obj[n:n+K] = rho_s
    c_obj[n+K] = rho_q
    
    # Inequality constraints: dx * grad_j . delta + s_j - q <= -r_j
    A_ub = np.zeros((K, m_vars))
    b_ub = np.zeros(K)
    for j in range(K):
        A_ub[j, :n] = dx * grads[j]
        A_ub[j, n+j] = 1.0
        A_ub[j, n+K] = -1.0
        b_ub[j] = -corrs[active_idx[j]]
    
    # Equality: sum(delta) = 0
    A_eq = np.zeros((1, m_vars))
    A_eq[0, :n] = 1.0
    
    bounds = [(-h[i]*alpha, (1.0-h[i])*alpha) for i in range(n)]
    bounds += [(0.0, None)] * K  # slacks >= 0
    bounds.append((None, None))  # q free
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.zeros(1),
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.status != 0:
            return None
        return res.x[:n]
    except Exception:
        return None

def solve_hard_lp(h, dx, target_sum, alpha, top_idx, n):

    corrs = corr_of_h(h)
    max_c = corrs.max()

    try:

def line_search(h_cur, c5_cur, direction, dx, target_sum, n):

    steps = [1.0, 0.8, 0.6, 0.5, 0.4, 0.3, 0.25, 0.2, 0.15, 0.1, 0.07, 0.05, 0.03, 0.02, 0.01]

def coordinate_pair_optimize(h, dx, target_sum, n, budget_s, t0, n_rounds=20):
    """Exact coordinate pair optimization.
    For pairs (p,q), move mass from one to the other and find the best t
    that minimizes C5 exactly.
    """
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    rng = np.random.RandomState(12345)
    
    for rnd in range(n_rounds):

            break

        n_sample = min(n, 200)

            if budget_s - (time.time() - t0) < 5.0:
                break
            # Try pairing p with a few random other indices
            for _ in range(5):
                if budget_s - (time.time() - t0) < 3.0:

                q = rng.randint(n)

                    continue

                b = best_h[q]

                t_lo = max(-a, b-1.0)
                t_hi = min(1.0-a, b)
                if t_hi - t_lo < 1e-10:
                    continue
                
                # Grid search over t
                n_grid = 30
                ts = np.linspace(t_lo, t_hi, n_grid)
                best_t = 0.0
                best_pair_c5 = best_c5

                    best_h[q] = b - best_t
                    best_c5 = best_pair_c5
                    improved = True

    return best_h, best_c5

def binary_local_search(h, dx, target_sum, n, budget_s, t0, n_rounds=10):

import numpy as np
from scipy.optimize import linprog
import time

def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def c5_of_h(h, dx):
    return np.correlate(h, 1.0 - h, mode='full').max() * dx

def corr_of_h(h):
    return np.correlate(h, 1.0 - h, mode='full')

def grad_lag(h, k, n):
    off = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -off)
    i_hi = min(n - 1, n - 1 - off)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx + off] += (1.0 - h[idx])
    grad[idx] += (-h[idx + off])
    return grad

def solve_soft_lp(h, dx, alpha, top_idx, n, rho_q=0.5, rho_s_base=3.0, margin_frac=0.05):
    corrs = corr_of_h(h)
    max_c = corrs.max()
    margin = max(0.01, margin_frac * max_c)
    active_mask = corrs >= (max_c - margin)
    active_mask[top_idx] = True
    active_idx = np.where(active_mask)[0]
    K = len(active_idx)
    if K < 3:
        return None
    
    grads = np.zeros((K, n))
    for j, k in enumerate(active_idx):
        grads[j] = grad_lag(h, k, n)
    
    # Weight slacks by correlation importance
    w = corrs[active_idx] / (max_c + 1e-15)
    rho_s = rho_s_base * w
    
    m_vars = n + K + 1
    c_obj = np.zeros(m_vars)
    c_obj[n:n+K] = rho_s
    c_obj[n+K] = rho_q
    
    A_ub = np.zeros((K, m_vars))
    b_ub = np.zeros(K)
    for j in range(K):
        A_ub[j, :n] = dx * grads[j]
        A_ub[j, n+j] = 1.0
        A_ub[j, n+K] = -1.0
        b_ub[j] = -corrs[active_idx[j]]
    
    A_eq = np.zeros((1, m_vars))
    A_eq[0, :n] = 1.0
    
    bounds = [(-h[i]*alpha, (1.0-h[i])*alpha) for i in range(n)]
    bounds += [(0.0, None)] * K
    bounds.append((None, None))
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.zeros(1),
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.status != 0:
            return None
        return res.x[:n]
    except Exception:
        return None

def solve_hard_lp(h, dx, alpha, top_idx, n):
    corrs = corr_of_h(h)
    max_c = corrs.max()
    margin = max(0.01, 0.03 * max_c)
    active_mask = corrs >= (max_c - margin)
    active_mask[top_idx] = True
    active_idx = np.where(active_mask)[0]
    K = len(active_idx)
    if K < 3:
        return None
    
    grads = np.zeros((K, n))
    for j, k in enumerate(active_idx):
        grads[j] = grad_lag(h, k, n)
    
    m_vars = n + 1
    c_obj = np.zeros(m_vars)
    c_obj[-1] = 1.0
    
    A_ub = np.zeros((K, m_vars))
    b_ub = np.zeros(K)
    for j in range(K):
        A_ub[j, :n] = dx * grads[j]
        A_ub[j, -1] = -1.0
        b_ub[j] = -corrs[active_idx[j]]
    
    A_eq = np.zeros((1, m_vars))
    A_eq[0, :n] = 1.0
    
    bounds = [(-h[i]*alpha, (1.0-h[i])*alpha) for i in range(n)]
    bounds.append((0.0, None))
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.zeros(1),
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.status != 0:
            return None
        return res.x[:n]
    except Exception:
        return None

def line_search(h_cur, c5_cur, direction, dx, target_sum):
    steps = [1.0, 0.7, 0.5, 0.4, 0.3, 0.25, 0.2, 0.15, 0.1, 0.06, 0.04, 0.02, 0.01]
    for t in steps:
        h_new = h_cur + t * direction
        h_new = project_feasible(h_new, target_sum)
        c5_new = c5_of_h(h_new, dx)
        if c5_new < c5_cur - 1e-14:
            return h_new, c5_new
    return None, c5_cur

def lp_polish(h, dx, target_sum, alpha, n_inner, budget_s, t0, soft=True, margin_frac=0.05):
    n = len(h)
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    cur = h.copy()
    cur_c5 = best_c5
    n_top = max(20, min(50, n // 3))
    
    for it in range(n_inner):
        if budget_s - (time.time() - t0) < 3.0:
            break
        corrs = corr_of_h(cur)
        top_idx = np.argsort(corrs)[-n_top:]
        
        delta = None
        if soft:
            delta = solve_soft_lp(cur, dx, alpha, top_idx, n, margin_frac=margin_frac)
        if delta is None:
            delta = solve_hard_lp(cur, dx, alpha, top_idx, n)
        if delta is None:
            alpha *= 0.5
            if alpha < 0.003:
                break
            continue
        
        h_new, c5_new = line_search(cur, cur_c5, delta, dx, target_sum)
        if h_new is not None:
            cur = h_new
            cur_c5 = c5_new
            alpha = min(0.4, alpha * 1.08)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = cur.copy()
        else:
            alpha *= 0.6
            if alpha < 0.003:
                break
    
    return best_h, best_c5

def pair_optimize(h, dx, target_sum, budget_s, t0, rng, n_rounds=20, max_pairs_per_round=50):
    """Exact coordinate pair optimization with random pair selection."""
    n = len(h)
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    
    # Precompute which coordinates are "useful" (not exactly 0 or 1)
    for rnd in range(n_rounds):
        if budget_s - (time.time() - t0) < 8.0:
            break
        
        # Select candidate pairs
        # Prioritize pairs where one is "high" (near 1) and other is "low" (near 0)
        highs = np.where(best_h > 0.3)[0]
        lows = np.where(best_h < 0.7)[0]

        n_lows = min(len(lows), 30)
        
        if n_highs < 1 or n_lows < 1:
            break
        
        h_sample = rng.choice(highs, size=n_highs, replace=False)
        l_sample = rng.choice(lows, size=n_lows, replace=False)

        for p in h_sample[:10]:
            for q in l_sample[:10]:
                if p != q:
                    pairs.append((p, q))
        # Also add some random pairs
        for _ in range(10):
            p = rng.randint(n)
            q = rng.randint(n)
            if p != q:
                pairs.append((p, q))
        
        rng.shuffle(pairs)
        pairs = pairs[:max_pairs_per_round]
        
        improved = False
        for (p, q) in pairs:
            if budget_s - (time.time() - t0) < 3.0:
                break
            a = best_h[p]
            b = best_h[q]
            t_lo = max(-a, b - 1.0)
            t_hi = min(1.0 - a, b)
            if t_hi - t_lo < 1e-10:
                continue
            
            # Quick check: sample a few points
            n_grid = 15
            ts = np.linspace(t_lo, t_hi, n_grid)
            best_t = 0.0
            best_pair_c5 = best_c5
            
            for t in ts:
                h_test = best_h.copy()
                h_test[p] = a + t
                h_test[q] = b - t
                c5_test = c5_of_h(h_test, dx)
                if c5_test < best_pair_c5:
                    best_pair_c5 = c5_test
                    best_t = t
            
            if best_t != 0.0 and best_pair_c5 < best_c5 - 1e-12:
                # Refine with finer grid around best_t
                t_ref_lo = max(t_lo, best_t - (t_hi-t_lo)/n_grid)
                t_ref_hi = min(t_hi, best_t + (t_hi-t_lo)/n_grid)

                    h_test[p] = a + t
                    h_test[q] = b - t
                    c5_test = c5_of_h(h_test, dx)
                    if c5_test < best_pair_c5:
                        best_pair_c5 = c5_test
                        best_t = t
                
                best_h[p] = a + best_t
                best_h[q] = b - best_t
                best_c5 = best_pair_c5
                improved = True
        
        if not improved and rnd > 5:
            break
    
    return best_h, best_c5

def binary_search_local(h, dx, target_sum, budget_s, t0, rng):
    """Systematic binary (0/1) local search with pair swaps."""
    n = len(h)
    target = n // 2
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    
    # Start from the nearest binary vector
    cur = (best_h > 0.5).astype(float)
    n_ones = int(cur.sum())
    if n_ones > target:
        ones = np.where(cur == 1.0)[0]
        rng.shuffle(ones)
        for i in range(n_ones - target):
            cur[ones[i]] = 0.0
    elif n_ones < target:
        zeros = np.where(cur == 0.0)[0]
        rng.shuffle(zeros)
        for i in range(target - n_ones):
            cur[zeros[i]] = 1.0
    cur_c5 = c5_of_h(cur, dx)
    
    # Iterative pair swap search
    ones = list(np.where(cur == 1.0)[0])
    zeros = list(np.where(cur == 0.0)[0])
    
    for sweep in range(5):
        if budget_s - (time.time() - t0) < 10.0:
            break
        improved = False
        # Try swapping each 1 with some 0s
        rng.shuffle(ones)
        rng.shuffle(zeros)
        n_try = min(len(ones), 40)
        n_pair = min(len(zeros), 40)
        
        for i in range(n_try):
            if budget_s - (time.time() - t0) < 3.0:
                break
            p = ones[i]
            for j in range(n_pair):
                q = zeros[j]
                cur[p] = 0.0
                cur[q] = 1.0
                c5_new = c5_of_h(cur, dx)
                if c5_new < best_c5 - 1e-12:
                    best_c5 = c5_new
                    best_h = cur.copy()
                    ones.remove(p)
                    ones.append(q)
                    zeros.remove(q)
                    zeros.append(p)
                    improved = True
                else:
                    cur[p] = 1.0
                    cur[q] = 0.0
        
        if not improved:
            break
    
    return best_h, best_c5

def transfer_grid(h_src, n_target):
    n_src = len(h_src)
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_tgt = (np.arange(n_target) + 0.5) * (2.0 / n_target)
    return np.interp(x_tgt, x_src, h_src)

def perturb_reopt(h, dx, target_sum, rng, n_rounds, budget_s, t0):
    n = len(h)
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    
    for r in range(n_rounds):
        if budget_s - (time.time() - t0) < 15.0:

        scale = max(0.01, 0.06 / (r + 1))
        pert = rng.standard_normal(n)
        pert -= pert.mean()
        h_p = h + scale * pert
        h_p = project_feasible(h_p, target_sum)

        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
            h = h_opt.copy()
    
    return best_h, best_c5

def smart_pairs(h, n, top_k_lags):
    """Find pairs (p,q) where p is 1, q is 0, and they're separated by a top lag."""
    pairs = []
    for k in top_k_lags:
        off = k - (n - 1)
        if off == 0:
            continue
        i_lo = max(0, -off)

import numpy as np
from scipy.optimize import linprog
import time

def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def c5_of_h(h, dx):
    return np.correlate(h, 1.0 - h, mode='full').max() * dx

def corr_of_h(h):
    return np.correlate(h, 1.0 - h, mode='full')

def grad_lag(h, k, n):
    off = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -off)
    i_hi = min(n - 1, n - 1 - off)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx + off] += (1.0 - h[idx])
    grad[idx] += (-h[idx + off])
    return grad

def solve_soft_lp(h, dx, alpha, top_idx, n, rho_q=0.5, rho_s_base=3.0, margin_frac=0.05):
    corrs = corr_of_h(h)
    max_c = corrs.max()
    margin = max(0.01, margin_frac * max_c)
    active_mask = corrs >= (max_c - margin)
    active_mask[top_idx] = True
    active_idx = np.where(active_mask)[0]
    K = len(active_idx)
    if K < 3:
        return None
    grads = np.zeros((K, n))
    for j, k in enumerate(active_idx):
        grads[j] = grad_lag(h, k, n)
    w = corrs[active_idx] / (max_c + 1e-15)
    rho_s = rho_s_base * w
    m_vars = n + K + 1
    c_obj = np.zeros(m_vars)
    c_obj[n:n+K] = rho_s
    c_obj[n+K] = rho_q
    A_ub = np.zeros((K, m_vars))
    b_ub = np.zeros(K)
    for j in range(K):
        A_ub[j, :n] = dx * grads[j]
        A_ub[j, n+j] = 1.0
        A_ub[j, n+K] = -1.0
        b_ub[j] = -corrs[active_idx[j]]
    A_eq = np.zeros((1, m_vars))
    A_eq[0, :n] = 1.0
    bounds = [(-h[i]*alpha, (1.0-h[i])*alpha) for i in range(n)]
    bounds += [(0.0, None)] * K
    bounds.append((None, None))
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.zeros(1),
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.status != 0:
            return None
        return res.x[:n]
    except Exception:
        return None

def transfer_grid(h_src, n_target):
