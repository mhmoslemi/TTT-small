import numpy as np
from scipy.optimize import minimize, linprog

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(-(1.0 - h))
        remaining = diff
        for i in range(n):
            idx = order[i]
            cap = 1.0 - h[idx]
            if cap <= 1e-15:
                continue
            add = min(remaining, cap)
            h[idx] += add
            remaining -= add
            if remaining <= 1e-14:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            if h[idx] <= 1e-15:
                continue
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-14:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] = float(np.clip(h[idx] + diff2, 0.0, 1.0))
    return h

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def grad_row(h, m, n):
    dx = 2.0 / n
    d = m - (n - 1)
    if d == 0:
        return dx * (1.0 - 2.0 * h)
    elif d > 0:
        g = np.zeros(n)
        g[d:] += dx * (1.0 - h[:-d])
        g[:-d] -= dx * h[d:]
        return g
    else:
        p = -d
        g = np.zeros(n)
        g[:-p] += dx * (1.0 - h[p:])
        g[p:] -= dx * h[:-p]
        return g

def build_grad_matrix(h):
    n = len(h)
    M = 2 * n - 1
    G = np.zeros((M, n))
    for m in range(M):
        G[m] = grad_row(h, m, n)
    return G

def interp_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    edges_old = np.linspace(0.0, 2.0, n + 1)
    edges_new = np.linspace(0.0, 2.0, new_n + 1)
    h_new = np.zeros(new_n)
    for k in range(new_n):
        a, b = edges_new[k], edges_new[k + 1]
        total = 0.0
        for j in range(n):
            lo = max(a, edges_old[j])
            hi = min(b, edges_old[j + 1])
            if hi > lo:
                total += h[j] * (hi - lo)
        h_new[k] = total / (b - a)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

def lp_trust_region(h, delta=0.05, max_decline_check=False):
    """
    Solve the linearized epigraph LP trust region problem.
    Returns direction d and predicted tau, or None on failure.
    """
    n = len(h)
    M = 2 * n - 1
    dx = 2.0 / n
    P = compute_profile(h)  # length M, already scaled by dx
    G = build_grad_matrix(h)  # shape (M, n), already scaled by dx
    
    # LP: minimize tau
    # subject to: G[m] · d - tau <= -P[m]  for all m
    #              sum(d) = 0
    #              h[i] - delta <= h[i] + d[i] <= h[i] + delta  (i.e., -delta+... bounds on d)
    
    # Variables: [d_0, ..., d_{n-1}, tau]
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    # Inequality: G[m] · d - tau <= -P[m]
    # A_ub shape: (M, n+1)
    A_ub = np.zeros((M, n + 1))
    A_ub[:, :n] = G
    A_ub[:, -1] = -1.0
    b_ub = -P
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds: for d[i]: max(-delta, -h[i]) <= d[i] <= min(delta, 1-h[i])
    # For tau: no meaningful bound, use large range
    lo = np.maximum(-delta, -h) - 1e-10
    hi = np.minimum(delta, 1.0 - h) + 1e-10
    bounds = list(zip(lo, hi))
    bounds.append((0.0, 1.0))  # tau
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x is not None:
            d = res.x[:n]
            tau = res.x[-1]
            return d, tau
    except Exception:
        pass
    return None, None

def active_descent(h, max_iters=300, active_tol=0.004):

    h = h.copy()
    best_c = exact_score(h)
    gr = (np.sqrt(5.0) - 1.0) / 2.0

        c = float(profile.max())
        thresh = c - active_tol
        active = np.where(profile >= thresh)[0]
        if len(active) == 0:

        if len(active) > 25:

            active = active[top]

            g += grad_row(h, m, n)
        g -= g.mean()
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            break
        momentum_vec = momentum * momentum_vec + g
        mv_norm = np.linalg.norm(momentum_vec)
        if mv_norm < 1e-15:
            break
        d = -momentum_vec / mv_norm
        pos_mask = d > 1e-12
        neg_mask = d < -1e-12
        alpha_max = 0.5
        if pos_mask.any():
            alpha_max = min(alpha_max, float(((1.0 - h[pos_mask]) / d[pos_mask]).min()))

        alpha_max = max(alpha_max, 1e-10)
        best_alpha = 0.0

        best_h = h
        a_ls, b_ls = 0.0, alpha_max
        for _ in range(20):
            c1 = b_ls - gr * (b_ls - a_ls)
            c2 = a_ls + gr * (b_ls - a_ls)
            h1 = repair_feasible(h + c1 * d)
            h2 = repair_feasible(h + c2 * d)
            f1 = exact_score(h1)
            f2 = exact_score(h2)
            if f1 < best_val:

                best_alpha = c1
                best_h = h1.copy()
            if f2 < best_val:
                best_val = f2

                break
        if best_val >= c - 1e-15:
            break
        h = best_h
        if best_val < best_c:

def population_search(h_pool, max_generations=10, time_limit=120.0, t_start=None, rng=None):
    """Population-based search using convex combinations and mutation."""
    if t_start is None:
        t_start = time.perf_counter()
    if rng is None:
        rng = np.random.RandomState(42)
    
    best_h = h_pool[0][1]
    best_c = h_pool[0][0]
    
    pool = list(h_pool)  # list of (c, h)
    
    for gen in range(max_generations):
        if time.perf_counter() - t_start > time_limit:
            break
        
        # Generate offspring
        offspring = []
        n_parents = len(pool)
        for i in range(n_parents):
            if time.perf_counter() - t_start > time_limit:
                break
            for j in range(n_parents):
                if i == j:
                    continue
                # Convex combination
                t = rng.uniform(0.2, 0.8)
                h_mix = t * pool[i][1] + (1 - t) * pool[j][1]
                # Ensure same length
                if len(h_mix) != len(pool[i][1]):
                    continue
                h_mix = repair_feasible(h_mix)
                
                # Add small mutation
                if rng.random() < 0.5:
                    noise = rng.normal(0, 0.02, len(h_mix))
                    h_mix = repair_feasible(h_mix + noise)
                
                offspring.append(h_mix)
        
        # Evaluate offspring (quick local refine)
        for h_off in offspring:
            if time.perf_counter() - t_start > time_limit:
                break
            c_off = exact_score(h_off)
            pool.append((c_off, h_off))
        
        # Select top candidates
        pool.sort(key=lambda x: x[0])
        pool = pool[:8]
        
        # Update best
        if pool[0][0] < best_c:
            best_c = pool[0][0]
            best_h = pool[0][1]
    
    return best_h, best_c

def golden_seed(n, variant=0):
    """Generate a golden-ratio-based seed."""
    half = n // 2
    h = np.zeros(n)
    if variant == 0:
        idx = np.arange(n)
        vals = (idx * 0.618033988749895) % 1.0
    elif variant == 1:
        idx = np.arange(n)
        vals = (idx * 0.4142135623730951) % 1.0
    elif variant == 2:
        idx = np.arange(n)
        vals = (idx * 0.7548776662466927) % 1.0
    else:
        idx = np.arange(n)
        vals = (np.sin(idx * 1.618033988749895 * np.pi) ** 2)
    top = np.argsort(vals)[-half:]
    h[top] = 1.0
    return repair_feasible(h)

        return h

        return g

            if time_left() < 25:
                break
            d = n_cur + dn
            if 36 <= d <= 140 and d % 2 == 0:
                h_new = interp_to_n(best_h, d)
                h_lp, c_lp = lp_refine(h_new, max_iters=60, init_delta=0.05,
                                        time_limit=min(25.0, time_left() * 0.3),
                                        t_start=time.perf_counter())
                if c_lp < best_c:
                    best_h, best_c = h_lp, c_lp

    # Phase 7: Final LP pass
    if time_left() > 40:
        h_lp, c_lp = lp_refine(best_h, max_iters=100, init_delta=0.03,
                                time_limit=time_left() * 0.4,
                                t_start=time.perf_counter())
        if c_lp < best_c:
            best_h, best_c = h_lp, c_lp

    # Phase 8: Tight active descent
    if time_left() > 30:
        h_ad, c_ad = active_descent(best_h, max_iters=400, active_tol=0.001)
        if c_ad < best_c:
            best_h, best_c = h_ad, c_ad

    # Phase 9: Coordinate polish
    if time_left() > 20:
        h_cp, c_cp = coordinate_polish(best_h, max_iters=50)
        if c_cp < best_c:
            best_h, best_c = h_cp, c_cp

    # Phase 10: Final tiny LP
    if time_left() > 10:
        h_lp, c_lp = lp_refine(best_h, max_iters=50, init_delta=0.02,
                                time_limit=time_left() * 0.6,
                                t_start=time.perf_counter())
        if c_lp < best_c:
            best_h, best_c = h_lp, c_lp

    # Final
    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)

    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            final_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 2:
            final_c = float(eval_result[0])
    except Exception:
        pass

    return best_h, final_c, len(best_h)
