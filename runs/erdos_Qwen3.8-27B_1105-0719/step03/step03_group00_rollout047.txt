def project_feasible(x, target_sum):
    n = len(x)
    lo = -2.0
    hi = 2.0
    for _ in range(60):

        h = np.clip(x - mid, 0.0, 1.0)
        s = h.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-13:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def c5_of_h(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * dx

def corr_of_h(h):
    return np.correlate(h, 1.0 - h, mode='full')

def grad_lag(h, k, n):
    """Gradient of corr_k with respect to h."""
    off = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -off)
    i_hi = min(n - 1, n - 1 - off)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    # h[i] appears as first factor: derivative is (1 - h[i+off])
    grad[idx] += (1.0 - h[idx + off])
    # h[i+off] appears as second factor: derivative is -h[i]
    grad[idx + off] += (-h[idx])
    return grad

def pair_move_deltas(h, p, q, n):
    """For pair move h[p]+=t, h[q]-=t, compute d_k for all lags k.
    corr_k(t) = corr_k + t * d_k"""
    corr = corr_of_h(h)
    d = np.zeros(2*n - 1)
    for k in range(2*n - 1):
        off = k - (n - 1)
        i_lo = max(0, -off)
        i_hi = min(n - 1, n - 1 - off)
        if i_lo > i_hi:
            continue
        dk = 0.0
        # p as first factor (i = p, need p in [i_lo, i_hi] and p+off in [0,n-1])
        if p >= i_lo and p <= i_hi:
            j = p + off
            if 0 <= j < n:
                dk += (1.0 - h[j])  # for h[p]
                dk -= (1.0 - h[j])  # for h[q]... wait no
                # h[p] changes by +t: contribution is t*(1-h[p+off])
                # h[q] changes by -t: contribution is -t*(1-h[q+off])
        # Let me redo this more carefully
        # corr_k = sum_{i=i_lo}^{i_hi} h[i] * (1 - h[i+off])
        # After change: h[p] -> h[p]+t, h[q] -> h[q]-t
        # Change in corr_k:
        #   t * (1 - h[p+off]) if p is in [i_lo, i_hi]
        #  -t * h[p-off] if p-off is in [i_lo, i_hi] (i.e., p = i+off for some valid i)
        #  -t * (1 - h[q+off]) if q is in [i_lo, i_hi]
        #  +t * h[q-off] if q-off is in [i_lo, i_hi]
        
        # So d_k = [indicator(p in range) * (1-h[p+off]) - indicator(p-off in range) * h[p-off]] 
        #         - [indicator(q in range) * (1-h[q+off]) - indicator(q-off in range) * h[q-off]]
        
        term_p = 0.0
        if i_lo <= p <= i_hi:
            term_p += (1.0 - h[p + off])
        if i_lo <= p - off <= i_hi:
            term_p -= h[p - off]

        if i_lo <= q <= i_hi:
            term_q += (1.0 - h[q + off])
        if i_lo <= q - off <= i_hi:
            term_q -= h[q - off]
        
        d[k] = term_p - term_q
    
    return corr, d

def find_best_t(corr, d, L, U, n_ribbon=50):
    """Find t in [L,U] that minimizes max_k(corr_k + t*d_k).
    This is a 1D convex piecewise-linear minimization."""
    # Coarse grid
    ts = np.linspace(L, U, n_ribbon)
    vals = np.zeros(n_ribbon)
    for i, t in enumerate(ts):
        vals[i] = (corr + t * d).max()
    
    # Find the best coarse point
    best_i = np.argmin(vals)
    
    # Refine with golden section around the best point
    if best_i == 0:
        lo, hi = L, ts[1]
    elif best_i == n_ribbon - 1:
        lo, hi = ts[-2], U
    else:
        lo, hi = ts[best_i - 1], ts[best_i + 1]
    
    # Golden section search
    gr = (np.sqrt(5) - 1) / 2  # ~0.618
    a, b = lo, hi
    c = b - gr * (b - a)
    dpt = a + gr * (b - a)
    fc = (corr + c * d).max()
    fd = (corr + dpt * d).max()
    
    for _ in range(40):
        if fc < fd:
            b = dpt
            dpt = c
            fd = fc
            c = b - gr * (b - a)
            fc = (corr + c * d).max()
        else:
            a = c
            c = dpt
            fc = fd
            dpt = a + gr * (b - a)
            fd = (corr + dpt * d).max()
    
    t_opt = (a + b) / 2
    return t_opt, (corr + t_opt * d).max()

def lp_soft_slack(h, dx, target_sum, alpha, n, top_lags, rho_q, rho_s):
    """Solve soft-slack LP for direction."""
    n_lags = len(top_lags)
    corr = corr_of_h(h)
    
    # Build gradient matrix for selected lags
    M = np.zeros((n_lags, n))
    for j, k in enumerate(top_lags):
        M[j] = grad_lag(h, k, n) * dx
    
    # Variables: delta (n), slack (n_lags), q (1)
    m_vars = n + n_lags + 1
    q_idx = n + n_lags
    
    # Objective: rho_q * q + sum rho_s * s_j
    c_obj = np.zeros(m_vars)
    c_obj[q_idx] = rho_q
    for j in range(n_lags):
        c_obj[n + j] = rho_s[j]
    
    # Constraints: M[j] @ delta + s_j - q <= -corr[top_lags[j]]
    A_ub = np.zeros((n_lags, m_vars))
    for j in range(n_lags):
        A_ub[j, :n] = M[j]
        A_ub[j, n + j] = 1.0
        A_ub[j, q_idx] = -1.0
    b_ub = -corr[top_lags]
    
    # Equality: sum delta = 0
    A_eq = np.zeros((1, m_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    bounds = [(None, None)] * m_vars
    for i in range(n):
        bounds[i] = (-alpha * h[i], alpha * (1.0 - h[i]))
    for j in range(n_lags):
        bounds[n + j] = (0.0, None)
    bounds[q_idx] = (None, None)
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.status == 0:
            return res.x[:n]
    except Exception:
        pass
    return None

def lp_hard_minmax(h, dx, target_sum, alpha, n, top_lags):

    corr = corr_of_h(h)
    n_lags = len(top_lags)
    
    M = np.zeros((n_lags, n))
    for j, k in enumerate(top_lags):
        M[j] = grad_lag(h, k, n) * dx
    
    m_vars = n + 1
    q_idx = n
    
    c_obj = np.zeros(m_vars)
    c_obj[q_idx] = 1.0

        A_ub[j, :n] = M[j]
        A_ub[j, q_idx] = -1.0

def line_search(h, c5_cur, direction, dx, target_sum, n):

def do_pair_moves(h, dx, target_sum, n, rng, max_pairs, budget_s, t0):

def softmax_subgradient_step(h, dx, target_sum, n, beta, learning_rate):
    """One step of projected subgradient on smoothed max."""
    corr = corr_of_h(h)
    # Softmax weights
    scaled = beta * corr
    scaled -= scaled.max()
    weights = np.exp(scaled)
    weights /= weights.sum()
    
    # Weighted gradient
    grad = np.zeros(n)
    for k in range(n):  # Only need to sum over relevant lags
        if weights[k] < 1e-10:
            continue
        grad += weights[k] * grad_lag(h, k, n)
    grad *= dx
    
    # Projected gradient step
    h_new = h - learning_rate * grad
    h_new = project_feasible(h_new, target_sum)
    return h_new

def grad_lag(h, k, n):
    off = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -off)
    i_hi = n - 1 - abs(off) if abs(off) <= n - 1 else -1
    i_hi = min(n - 1, n - 1 - off)
    i_lo = max(0, -off)

        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx] += (1.0 - h[idx + off])
    grad[idx + off] += (-h[idx])
    return grad

                top_k = max(15, min(40, int(0.3 * (2 * n_try - 1))))
                top_idx = np.argsort(corr)[-top_k:]
                direction = soft_slack_direction(h_opt, dx_t, ts_t, alpha, n_try, top_idx, 0.3, 4.0)
                if direction is not None:
                    h_new, c5_new = line_search(h_opt, c5_opt, direction, ts_t, dx_t)
                    if h_new is not None:
                        h_opt = h_new
                        c5_opt = c5_new
        # Quick pair moves on new grid
        if remaining() > 15 and c5_opt < best_c5 + 0.005:
            h_opt, c5_opt, _ = do_pair_moves(h_opt, dx_t, ts_t, n_try, rng, 2000, budget_s, t0)
        if c5_opt < best_c5 - 1e-6:
            best_c5 = c5_opt
            best_h = h_opt
            n_base = n_try

    # Phase 8: Final intensive polish on best grid
    n_f = len(best_h)
    dx_f = 2.0 / n_f
    ts_f = n_f / 2.0

    if remaining() > 30:
        for _ in range(2):
            if remaining() < 10:
                break
            best_h, best_c5, _ = do_pair_moves(best_h, dx_f, ts_f, n_f, rng, 3000, budget_s, t0)
            if remaining() < 10:
                break
            for _ in range(80):
                if remaining() < 3:
                    break
                corr = corr_of_h(best_h)
                top_k = max(20, min(50, int(0.3 * (2 * n_f - 1))))
                top_idx = np.argsort(corr)[-top_k:]
                direction = soft_slack_direction(best_h, dx_f, ts_f, 0.03, n_f, top_idx, 0.5, 5.0)
                if direction is not None:
                    h_new, c5_new = line_search(best_h, best_c5, direction, ts_f, dx_f)
                    if h_new is not None:
                        best_h = h_new
                        best_c5 = c5_new

    # Phase 9: Very fine pair polish
    if remaining() > 20:
        best_h, best_c5, _ = do_directed_pair_moves(best_h, dx_f, ts_f, n_f, rng, 4000, budget_s, t0)

    # Final validation
    n_final = len(best_h)
    dx_final = 2.0 / n_final
    best_h = project_feasible(best_h, n_final / 2.0)
    best_c5 = c5_of_h(best_h)

    return best_h, best_c5, n_final
