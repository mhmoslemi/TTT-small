So ΔP[d] = (t-h[p]) * [(1-h[p+d])*I1 - h[p-d]*I2 - (1-h[q+d])*I3 + h[q-d]*I4]

where I1, I2, I3, I4 are validity indicators. BUT this formula assumes the four terms are distinct. When they're not (collision cases), I need to adjust.

  New: t*(1-(s-t)) = t*(t-s+1) = t*(t - (h[p]+h[q]) + 1)

  For the combined term (i = p = q-d):

  New value: t*(1-(s-t)) = t*(1-s+t)
  Δ = t*(1-s+t) - h[p]*(1-s+h[p])
  
  This is quadratic in t (has a t^2 term).

For a given pair (p, q) and a trial t:

import numpy as np

import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full"))) * dx

def profile_h(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, 1.0 - h[idx])
            h[idx] += add
            remaining -= add
            if remaining <= 1e-14:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-14:
                break
    return np.clip(h, 0.0, 1.0)

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    return repair_feasible(np.clip(h_new, 0.0, 1.0))

def pairwise_sweep_v2(h, n_gsl=30, rng=None, max_passes=1):
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
    n_pairs = len(pairs)
    
    for pass_num in range(max_passes):
        if rng is not None:
            perm = rng.permutation(n_pairs)
        else:
            perm = np.arange(n_pairs)
        
        improved = False
        for k in perm:
            p, q = pairs[k]
            s = h[p] + h[q]
            t_min = max(0.0, s - 1.0)
            t_max = min(1.0, s)
            if t_max - t_min < 1e-12:
                continue
            
            # Golden section search on g(t) = max(np.correlate(h_mod, 1-h_mod)) * dx
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            a, b = t_min, t_max
            c_pt = a + (1.0 - gr) * (b - a)
            d_pt = a + gr * (b - a)
            
            # Save original values
            hp0, hq0 = h[p], h[q]
            
            def eval_g(t):
                h[p] = t
                h[q] = s - t
                val = float(np.max(np.correlate(h, 1 - h, mode="full")))
                h[p] = hp0
                h[q] = hq0
                return val
            
            fc = eval_g(c_pt)
            fd = eval_g(d_pt)
            
            for _ in range(n_gsl):
                if fc <= fd:
                    b, d_pt, fd = d_pt, c_pt, fc
                    c_pt = a + (1.0 - gr) * (b - a)
                    fc = eval_g(c_pt)
                else:
                    a, c_pt, fc = c_pt, d_pt, fd
                    d_pt = a + gr * (b - a)
                    fd = eval_g(d_pt)
            
            t_opt = (a + b) * 0.5
            # Evaluate at optimal
            h[p] = t_opt
            h[q] = s - t_opt
            new_c = float(np.max(np.correlate(h, 1 - h, mode="full"))) * (2.0 / n)

                best_c = new_c
                improved = True
            else:
                h[p] = hp0
                h[q] = hq0
        
        if not improved:
            break
    
    return h, best_c

def build_gradient_rows(h, rows, n, dx):
    M = 2 * n - 1
    G = np.zeros((len(rows), n))
    for idx, m in enumerate(rows):
        d = m - (n - 1)
        if d == 0:
            G[idx, :] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            for i in range(n - d):
                G[idx, i + d] += dx * (1.0 - h[i])
                G[idx, i] -= dx * h[i + d]
        else:
            dp = -d
            for j in range(n - dp):
                G[idx, j + dp] += dx * (1.0 - h[j])
                G[idx, j] -= dx * h[j + dp]
    return G

def slsqp_active(h0, maxiter=300, active_k=12, tol=1e-12):
    n = len(h0)
    dx = 2.0 / n
    M = 2 * n - 1
    P = profile_h(h0)
    c0 = float(P.max())
    
    thresh = c0 - max(0.003, 0.05 * (c0 - np.median(P)))
    active = np.where(P >= thresh)[0]
    if len(active) == 0:
        active = np.array([np.argmax(P)])
    if len(active) > active_k:
        top_idx = np.argsort(P[active])[-active_k:]
        active = active[top_idx]
    
    def obj(x):
        return x[-1]
    
    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    
    def eq_fun(x):
        return x[:-1].sum() - n / 2.0
    
    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g
    
    def ineq_fun(x):
        h = x[:-1]
        t = x[-1]
        prof = profile_h(h)
        return t - prof[active]
    
    def ineq_jac(x):
        h = x[:-1]
        G = build_gradient_rows(h, active, n, dx)
        J = np.zeros((len(active), n + 1))
        J[:, :-1] = -G
        J[:, -1] = 1.0
        return J
    
    x0 = np.concatenate([h0, [c0 + 1e-4]])
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 1.0)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': tol})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = repair_feasible(res.x[:-1])
            c_new = exact_score(h_new)
            if c_new <= c0 + 1e-10:
                return h_new
    except Exception:
        pass
    return None

def projected_descent(h, max_iters=80, momentum=0.4, t_start=None, budget_s=1100):
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    vel = np.zeros(n)
    
    betas = [50, 100, 200, 400, 800]
    iters_per_beta = max(8, max_iters // len(betas))
    
    for beta in betas:
        alpha = 0.02
        for it in range(iters_per_beta):
            if t_start is not None and time.perf_counter() - t_start > budget_s - 25:
                return h, best_c
            P = profile_h(h)
            Pmax = P.max()
            exp_arg = beta * (P - Pmax)
            exp_arg -= exp_arg.max()
            weights = np.exp(exp_arg)
            S = weights.sum()
            if S < 1e-300:
                weights = np.zeros(len(P))
                weights[np.argmax(P)] = 1.0
            else:
                weights /= S
            active = np.where(weights > 0.01 / len(P))[0]
            if len(active) == 0:
                active = np.argsort(weights)[-5:]
            G = build_gradient_rows(h, active, n, 2.0 / n)
            w_active = weights[active]
            w_active = w_active / w_active.sum()
            grad = G.T @ w_active
            sum_err = h.sum() - n / 2.0
            grad += 2.0 * 50.0 * sum_err / n
            grad -= grad.mean()
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break
            grad /= gnorm
            vel = momentum * vel + grad
            vnorm = np.linalg.norm(vel)
            if vnorm > 0:
                vel /= vnorm
            improved = False
            for a in [alpha, alpha * 0.5, alpha * 0.25, alpha * 0.1]:
                h_try = repair_feasible(h - a * vel)
                s = exact_score(h_try)
                if s < best_c - 1e-14:
                    h = h_try
                    best_c = s
                    improved = True
                    break
            if improved:
                alpha = min(alpha * 1.05, 0.08)
            else:
                alpha *= 0.6
                if alpha < 1e-8:
                    break
    return h, best_c
