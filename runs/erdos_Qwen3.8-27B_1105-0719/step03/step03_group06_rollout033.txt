import numpy as np
from scipy.optimize import linprog
import time

def repair_feasible(vec):
    """Repair a vector to satisfy 0<=h<=1 and sum(h)=n/2."""
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        # Need to add mass
        order = np.argsort(-(1.0 - h))  # most slack first
        for i in order:
            cap = 1.0 - h[i]
            if cap <= 1e-15:
                continue
            add = min(diff, cap)
            h[i] += add
            diff -= add
            if diff <= 1e-14:
                break
    else:
        # Need to remove mass
        order = np.argsort(h)  # least value first
        for i in order:
            if h[i] <= 1e-15:
                continue
            rem = min(-diff, h[i])
            h[i] -= rem
            diff += rem
            if diff >= -1e-14:
                break
    # Final correction
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] = float(np.clip(h[idx] + diff2, 0.0, 1.0))
    return h

def compute_profile(h):
    """Compute the full correlation profile."""
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def exact_score(h):
    """Compute C5 = max of profile."""
    return float(compute_profile(h).max())

def grad_row(h, m, n):
    """Compute gradient of profile row m with respect to h."""
    dx = 2.0 / n
    d = m - (n - 1)
    if d == 0:
        return dx * (1.0 - 2.0 * h)
    elif d > 0:
        g = np.zeros(n)
        g[d:] += dx * (1.0 - h[:-d])
        g[:-d] -= dx * h[d:]
        return g
    else:
        p = -d
        g = np.zeros(n)
        g[:-p] += dx * (1.0 - h[p:])
        g[p:] -= dx * h[:-p]
        return g

def full_gradient_matrix(h):
    """Compute full gradient matrix (M, n) where M = 2n-1."""
    n = len(h)
    M = 2 * n - 1
    G = np.zeros((M, n))
    for m in range(M):
        G[m] = grad_row(h, m, n)
    return G

def lp_trust_step(h, P, G, active_idx, delta):
    """
    Solve LP: min tau
    s.t. G[m,:] @ d - tau <= -P[m] for m in active_idx
         sum(d) = 0
         -delta <= d_i <= delta
         h_i - delta <= h_i + d_i - h_i <= 1 - h_i + delta (redundant with delta)
    
    Actually: 0 <= h_i + d_i <= 1 means -h_i <= d_i <= 1-h_i
    Combined with |d_i| <= delta: max(-delta, -h_i) <= d_i <= min(delta, 1-h_i)
    """
    n = len(h)
    M_act = len(active_idx)
    
    # Objective: min tau (last variable)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Inequality constraints: G[m,:] @ d - tau <= -P[m]
    # => G[m,:n] @ d + [0,...,0,-1] @ [d; tau] <= -P[m]
    A_ub = np.zeros((M_act, n + 1))
    for i, m in enumerate(active_idx):
        A_ub[i, :n] = G[m, :]
        A_ub[i, -1] = -1.0
    b_ub = -P[active_idx]
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds on d_i: max(-delta, -h_i) <= d_i <= min(delta, 1-h_i)
    lo = np.maximum(-delta, -h)
    hi = np.minimum(delta, 1.0 - h)
    bounds = list(zip(lo, hi))
    bounds.append((0.0, np.inf))  # tau >= 0 (loose upper bound)
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and hasattr(res, 'x') and res.x is not None:
            d = res.x[:n]
            tau = res.x[-1]
            if np.isfinite(d).all() and np.isfinite(tau):
                return d, tau
    except Exception:
        pass
    return None, None

def interp_to_n(h, new_n):
    """Interpolate h from its current length to new_n."""
    n = len(h)
    if n == new_n:
        return h.copy()
    edges_old = np.linspace(0.0, 2.0, n + 1)
    edges_new = np.linspace(0.0, 2.0, new_n + 1)
    h_new = np.zeros(new_n)
    for k in range(new_n):
        a, b = edges_new[k], edges_new[k + 1]
        total = 0.0
        for j in range(n):
            lo = max(a, edges_old[j])
            hi = min(b, edges_old[j + 1])
            if hi > lo:
                total += h[j] * (hi - lo)
        h_new[k] = total / (b - a) if (b - a) > 0 else 0.0
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

def grad_row(h, m, n):

        return g
    else:
        p = -d

        g[:-p] += dx * (1.0 - h[p:])

        return g

        c_init = exact_score(h_interp)
        if c_init > best_c + 0.02:
            continue
        if time_left() > 30:
            deadline = t_start + (budget_s - 30)
            h_lp, c_lp = lp_descent(h_interp, max_iters=60, delta_init=0.04,
                                    delta_min=0.002, tol_init=0.01,
                                    tol_min=0.002, deadline=deadline, max_active=20)
            if c_lp < best_c:
                best_h, best_c = h_lp, c_lp
        if time_left() > 20:
            h_ag, c_ag = active_gradient_descent(best_h, max_iters=100,
                                                 active_tol=0.003, max_active=15,
                                                 deadline=t_start + (budget_s - 10))
            if c_ag < best_c:
                best_h, best_c = h_ag, c_ag

    # Phase 5: Population exploration
    if time_left() > 100:
        n_pop = len(best_h)
        exploration_h_list = []

        h_g = golden_seed(n_pop)
        h_g_smooth = 0.3 * h_g + 0.7 * np.full(n_pop, 0.5)
        exploration_h_list.append(repair_feasible(h_g_smooth))

        h_s1 = sine_seed(n_pop, 2)
        h_s1_smooth = 0.4 * h_s1 + 0.6 * np.full(n_pop, 0.5)
        exploration_h_list.append(repair_feasible(h_s1_smooth))

        h_s2 = sine_seed(n_pop, 3)
        h_s2_smooth = 0.4 * h_s2 + 0.6 * np.full(n_pop, 0.5)
        exploration_h_list.append(repair_feasible(h_s2_smooth))

        for t_cross in [0.2, 0.4]:
            h_cr = repair_feasible((1 - t_cross) * best_h + t_cross * h_g)
            exploration_h_list.append(h_cr)

        h_pert = repair_feasible(best_h + 0.03 * rng.randn(n_pop))
        exploration_h_list.append(h_pert)

        for exp_h in exploration_h_list:
            if time_left() < 25:
                break
            c_exp = exact_score(exp_h)
            if c_exp < best_c + 0.04:
                if time_left() > 20:
                    deadline = t_start + (budget_s - 20)
                    h_lp, c_lp = lp_descent(exp_h, max_iters=40, delta_init=0.03,
                                            delta_min=0.002, tol_init=0.012,
                                            tol_min=0.002, deadline=deadline,
                                            max_active=20)
                    if c_lp < best_c:
                        best_h, best_c = h_lp, c_lp
                if time_left() > 15:
                    h_ag, c_ag = active_gradient_descent(best_h, max_iters=80,
                                                         active_tol=0.003,
                                                         max_active=15,
                                                         deadline=t_start + (budget_s - 10))
                    if c_ag < best_c:
                        best_h, best_c = h_ag, c_ag

    # Phase 6: Fine LP
    if time_left() > 40:
        deadline = t_start + (budget_s - 20)
        h_lp, c_lp = lp_descent(best_h, max_iters=100, delta_init=0.015,
                                delta_min=0.0005, tol_init=0.004,
                                tol_min=0.0005, deadline=deadline, max_active=30)
        if c_lp < best_c:
            best_h, best_c = h_lp, c_lp

    # Phase 7: Coordinate polish
    if time_left() > 25:
        deadline = t_start + (budget_s - 15)
        h_cp, c_cp = coordinate_polish(best_h, max_iters=80, deadline=deadline)
        if c_cp < best_c:
            best_h, best_c = h_cp, c_cp

    # Phase 8: Ultra-fine LP
    if time_left() > 20:
        deadline = t_start + (budget_s - 10)
        h_lp, c_lp = lp_descent(best_h, max_iters=80, delta_init=0.008,
                                delta_min=0.0002, tol_init=0.002,
                                tol_min=0.0002, deadline=deadline, max_active=35)
        if c_lp < best_c:
            best_h, best_c = h_lp, c_lp

    # Phase 9: Final active gradient polish
    if time_left() > 15:
        h_ag, c_ag = active_gradient_descent(best_h, max_iters=200,
                                             active_tol=0.001, max_active=12,
                                             deadline=t_start + (budget_s - 5))
        if c_ag < best_c:
            best_h, best_c = h_ag, c_ag

    # Phase 10: Final ultra-fine LP
    if time_left() > 12:
        h_lp, c_lp = lp_descent(best_h, max_iters=50, delta_init=0.005,
                                delta_min=0.0001, tol_init=0.001,
                                tol_min=0.0001, deadline=t_start + (budget_s - 3),
                                max_active=40)
        if c_lp < best_c:
            best_h, best_c = h_lp, c_lp

    # Final repair and evaluation
    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)

    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            final_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 2:
            final_c = float(eval_result[0])
    except Exception:
        pass

    return best_h, final_c, len(best_h)
