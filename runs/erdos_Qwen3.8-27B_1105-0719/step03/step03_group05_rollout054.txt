```python
def compute_profile(h):

    return np.correlate(h, 1 - h, mode="full") * dx

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx] + 1e-15)
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx] + 1e-15)
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-14:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmin(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def gradient_of_profile(h, n, dx):
    """Compute gradient matrix G where G[k, j] = d(profile[k])/d(h[j])"""
    M = 2 * n - 1
    G = np.zeros((M, n))
    for m in range(M):
        d = m - (n - 1)
        if d == 0:
            G[m, :] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            # profile[m] = sum_{i=0}^{n-1-d} h[i]*(1-h[i+d])*dx
            # d/dh[j] = dx*(1-h[j+d]) for j < n-d (from h[j] term)
            #           - dx*h[j-d] for j >= d (from (1-h[j]) term where i=j-d)
            for j in range(n - d):
                G[m, j] += dx * (1.0 - h[j + d])
            for j in range(d, n):
                G[m, j] += -dx * h[j - d]
        else:
            dp = -d

            for j in range(0, n - dp):
                G[m, j] += -dx * h[j + dp]
```

```python
def fista_minmax(h0, beta=100, max_iters=500, alpha=None):
    """FISTA (accelerated proximal gradient) for min max_k f_k(h)"""
    n = len(h0)
    dx = 2.0 / n
    h = h0.copy()
    h_prev = h0.copy()
    best_h = h0.copy()
    best_c = exact_score(h0)
    
    if alpha is None:
        # Estimate Lipschitz constant
        # L ≈ β * max_k ||∇f_k||^2 (for smooth approximation)
        # or for the non-smooth case, use a backtracking approach
        alpha = 0.01
    
    t = 1.0
    for it in range(max_iters):
        # FISTA extrapolation
        theta = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        h_new_temp = h + (t - 1.0) / theta * (h - h_prev)
        h_new_temp = np.clip(h_new_temp, 0.0, 1.0)
        
        # Compute smoothed objective and gradient
        profile = compute_profile(h_new_temp)
        scaled = beta * profile
        max_s = np.max(scaled)
        weights = np.exp(scaled - max_s)
        wsum = weights.sum()
        weights /= wsum
        
        # Gradient
        G = gradient_of_profile(h_new_temp, n, dx)
        grad = G.T @ weights
        grad -= grad.mean()  # maintain sum constraint
        
        # Step
        h_candidate = h_new_temp - alpha * grad
        h_candidate = repair_feasible(h_candidate)
        
        c = exact_score(h_candidate)
        h_prev = h.copy()
        h = h_candidate
        t = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        
        if c < best_c:
            best_c = c
            best_h = h.copy()
    
    return best_h, best_c
```

```python
        # From h[j] term: j = 0 to n-1-d
        G[n-1+d, :n-d] += dx * (1.0 - h[d:n])
        # From (1-h[j]) term: j = d to n-1, contribution is -h[j-d]*dx
        G[n-1+d, d:] += -dx * h[:n-d]
        
        # d < 0 (lag = -d, index = n-1-d)
        # h[j]*(1-h[j-d]) for j = d to n-1
        # (1-h[j])*h[j+d] for j = 0 to n-1-d
        G[n-1-d, d:] += dx * (1.0 - h[:n-d])
        G[n-1-d, :n-d] += -dx * h[d:]
    return G
```

Let me also think about the **backtracking line search** for the non-convex smooth case:
- At point h with step direction d (negative gradient), we want to find α such that f(h + αd) < f(h) - c*α*||d||^2 (Armijo condition)
- Start with α = α_max, decrease by factor ρ until condition is met

For our problem, the feasible set is [0,1]^n ∩ {sum = n/2}. Along a direction d with sum(d) = 0, the feasible set is a line segment. We can find the maximum step before leaving [0,1]^n, and do the line search within that range.

```python
def backtracking_step(h, direction, alpha_init=0.1, c1=1e-4, rho=0.5, max_bt=20):
    """Find step size using backtracking line search"""
    n = len(h)
    # Find max feasible step
    pos = direction > 0
    neg = direction < 0
    if pos.any():
        alpha_max_pos = np.min((1.0 - h[pos]) / direction[pos])
    else:
        alpha_max_pos = np.inf
    if neg.any():
        alpha_max_neg = np.min((-h[neg]) / direction[neg])
    else:
        alpha_max_neg = np.inf

        f_new = exact_score(h_new)
        if f_new < f0 - c1 * alpha * np.dot(direction, direction):
            return h_new, f_new, alpha
        alpha *= rho
    return h, f0, 0.0
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx] + 1e-15)
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx] + 1e-15)
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-14:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmin(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_weighted(h, weights, n, dx):
    """Compute sum_k weights[k] * d f_k(h) / dh"""
    result = np.zeros(n)
    result += weights[n-1] * dx * (1.0 - 2.0 * h)
    for d in range(1, n):
        wd = weights[n-1+d]
        wn = weights[n-1-d]
        result[:n-d] += wd * dx * (1.0 - h[d:n])
        result[d:] += -wd * dx * h[:n-d]
        result[d:] += wn * dx * (1.0 - h[:n-d])
        result[:n-d] += -wn * dx * h[d:]
    return result

def fista_continuation(h0, betas=None, max_iters_per_beta=300, t_start_val=0.5):
    """FISTA with continuation in beta for the smoothed min-max problem"""
    n = len(h0)
    dx = 2.0 / n
    if betas is None:
        betas = [20, 40, 80, 160, 320, 640]
    
    h = h0.copy()
    best_h = h0.copy()
    best_c = exact_score(h0)
    
    for beta in betas:
        h_prev = h.copy()
        t = 1.0
        alpha = 0.5 / (beta * dx * n)  # Initial step size estimate
        
        for it in range(max_iters_per_beta):
            # FISTA extrapolation
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
            theta = (t - 1.0) / t_new
            h_temp = h + theta * (h - h_prev)
            h_temp = np.clip(h_temp, 0.0, 1.0)
            h_temp = repair_feasible(h_temp)
            
            # Compute smoothed gradient
            profile = compute_profile(h_temp)
            scaled = beta * profile
            max_s = np.max(scaled)
            exp_s = np.exp(scaled - max_s)
            weights = exp_s / exp_s.sum()
            
            grad = grad_weighted(h_temp, weights, n, dx)
            grad -= grad.mean()  # maintain sum constraint
            
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-15:
                break
            
            # Backtracking line search
            h_try = h_temp
            f_try = None
            for a in [alpha, alpha*0.5, alpha*0.25, alpha*0.1, alpha*0.05]:
                h_candidate = repair_feasible(h_temp - a * grad)
                # Evaluate smoothed objective
                prof_c = compute_profile(h_candidate)
                f_val = np.log(np.sum(np.exp(beta * prof_c - max_s))) / beta + max_s / beta
                if f_val < f_try - 1e-10 if f_try is not None else True:
                    h_try = h_candidate
                    f_try = f_val
                    if a == alpha:
                        alpha = min(alpha * 1.05, 2.0 / (beta * dx))
                    break
            
            h_current_score = exact_score(h_try)
            
            h_prev = h.copy()
            h = h_try
            t = t_new
            
            if h_current_score < best_c - 1e-14:
                best_c = h_current_score
                best_h = h.copy()
    
    return best_h, best_c

def active_set_exact(h0, max_iters=500, n_active=15, alpha_init=0.003):
    """Active set method with exact line search for the non-smooth max objective"""
    n = len(h0)
    dx = 2.0 / n
    h = h0.copy()
    best_c = exact_score(h)
    alpha = alpha_init
    stuck = 0
    
    for it in range(max_iters):
        profile = compute_profile(h)
        c = float(np.max(profile))
        if c + 1e-14 < best_c:
            best_c = c
            stuck = 0
        
        # Find active set (constraints near the max)
        margin = max(0.001, 3.0 * (c - float(np.median(profile))))
        active = np.where(profile >= c - margin)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(profile))])
        if len(active) > n_active:
            top_idx = np.argsort(profile[active])[-n_active:]
            active = active[top_idx]
        
        # Equal weights for active constraints
        w = np.zeros(2 * n - 1)
        w[active] = 1.0 / len(active)
        
        grad = grad_weighted(h, w, n, dx)
        grad -= grad.mean()
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad /= gnorm
        
        # Line search with backtracking
        improved = False
        for a in [alpha, alpha*0.5, alpha*0.25, alpha*0.1, alpha*0.05, alpha*0.02]:
            h_try = repair_feasible(h - a * grad)
            s = exact_score(h_try)
            if s < c - 1e-12:
                h = h_try
                if s < best_c:
                    best_c = s
                improved = True
                alpha = min(a * 1.1, 0.02)
                break
        
        if not improved:
            alpha *= 0.5
            stuck += 1
            if stuck > 50 or alpha < 1e-10:
                break
    
    return h, best_c

    n = len(h)

    # Use linear interpolation on the midpoint values

    h_interp = np.clip(h_interp, 0.0, 1.0)

def binary_sa(bits, max_iters=300, T_init=0.3, T_final=0.003, rng=None):
    """Simulated annealing for binary patterns"""
    n = len(bits)
    dx = 2.0 / n
    bits = bits.copy()
    current_c = exact_score(bits)
    best_bits = bits.copy()
    best_c = current_c
    T = T_init
    cooling = (T_final / T_init) ** (1.0 / max(max_iters, 1))
    
    for _ in range(max_iters):
        T *= cooling
        i = rng.randint(n)
        j = rng.randint(n)
        if bits[i] == bits[j]:
            continue
        bits[i], bits[j] = bits[j], bits[i]
        new_c = exact_score(bits)
        delta = new_c - current_c
        if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-10)):
            current_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_bits = bits.copy()
        else:
            bits[i], bits[j] = bits[j], bits[i]
    
    return best_bits, best_c

def symmetrize(h):
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx] + 1e-15)
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx] + 1e-15)
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-14:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmin(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_weighted(h, weights, n, dx):
    result = np.zeros(n)
    result += weights[n - 1] * dx * (1.0 - 2.0 * h)
    for d in range(1, n):
        wd = weights[n - 1 + d]
        wn = weights[n - 1 - d]
        result[:n - d] += wd * dx * (1.0 - h[d:n])
        result[d:] += -wd * dx * h[:n - d]
        result[d:] += wn * dx * (1.0 - h[:n - d])
        result[:n - d] += -wn * dx * h[d:]
    return result

def fista_continue(h0, betas, max_iters_per_beta):
    n = len(h0)
    dx = 2.0 / n
    h = h0.copy()
    best_h = h0.copy()
    best_c = exact_score(h0)
    for beta in betas:
        h_prev = h.copy()
        t = 1.0
        alpha = 0.8 / (beta * dx * n * 1.5)
        for it in range(max_iters_per_beta):
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
            theta = (t - 1.0) / t_new
            h_temp = h + theta * (h - h_prev)
            h_temp = repair_feasible(np.clip(h_temp, 0.0, 1.0))
            profile = compute_profile(h_temp)
            scaled = beta * profile
            max_s = np.max(scaled)
            exp_s = np.exp(scaled - max_s)
            wsum = exp_s.sum()
            if wsum < 1e-300:
                break
            weights = exp_s / wsum
            grad = grad_weighted(h_temp, weights, n, dx)
            grad -= grad.mean()
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-15:
                break
            improved = False
            for a in [alpha, alpha * 0.5, alpha * 0.25, alpha * 0.1]:
                h_try = repair_feasible(h_temp - a * grad)
                prof_try = compute_profile(h_try)
                f_try = np.log(np.sum(np.exp(beta * prof_try - max_s))) / beta + max_s / beta
                prof_cur = compute_profile(h_temp)
                f_cur = np.log(np.sum(np.exp(beta * prof_cur - max_s))) / beta + max_s / beta
                if f_try < f_cur - 1e-12:
                    h = h_try
                    improved = True
                    if a == alpha:
                        alpha = min(alpha * 1.06, 3.0 / (beta * dx * 2.0))
                    break
            h_prev = h.copy()
            t = t_new
            c_now = exact_score(h)
            if c_now < best_c - 1e-14:
                best_c = c_now
                best_h = h.copy()
    return best_h, best_c

def active_set_method(h0, max_iters, n_active, alpha_init):
    n = len(h0)
    dx = 2.0 / n
    h = h0.copy()
    best_c = exact_score(h)
    alpha = alpha_init
    stuck = 0
    for it in range(max_iters):
        profile = compute_profile(h)
        c = float(np.max(profile))
        if c + 1e-14 < best_c:

            stuck = 0
        margin = max(0.001, 3.5 * (c - float(np.median(profile))))
        active = np.where(profile >= c - margin)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(profile))])

            top_idx = np.argsort(profile[active])[-n_active:]
            active = active[top_idx]

        w[active] = 1.0 / len(active)
        grad = grad_weighted(h, w, n, dx)
        grad -= grad.mean()
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad /= gnorm
        improved = False
        for a in [alpha, alpha * 0.5, alpha * 0.25, alpha * 0.1, alpha * 0.05, alpha * 0.02]:

                break

                break
    return h, best_c

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    x_old = np.linspace(0.0, 2.0, n, endpoint=False) + 1.0 / n
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False) + 1.0 / new_n
    h_interp = np.interp(x_new, x_old, h)
    h_interp = np.clip(h_interp, 0.0, 1.0)
    return repair_feasible(h_interp)

def binary_sa(bits, max_iters, T_init, T_final, rng):
    n = len(bits)
    bits = bits.copy()
    current_c = exact_score(bits)
    best_bits = bits.copy()
    best_c = current_c
    T = T_init
    cooling = (T_final / T_init) ** (1.0 / max(max_iters, 1))
    for _ in range(max_iters):
        T *= cooling
        i = rng.randint(n)
        j = rng.randint(n)
        if bits[i] == bits[j]:
            continue
        bits[i], bits[j] = bits[j], bits[i]
        new_c = exact_score(bits)
        delta = new_c - current_c
        if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-10)):
            current_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_bits = bits.copy()
        else:
            bits[i], bits[j] = bits[j], bits[i]
    return best_bits, best_c

def symmetrize(h):
    return (h + h[::-1]) / 2.0
```
