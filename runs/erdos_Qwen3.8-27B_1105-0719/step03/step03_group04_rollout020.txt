    G[center, :] = dx * (1.0 - 2.0 * h)

        G[center - d, 0:n-d] += dx * (1.0 - h[d:n])

            # Try gradient direction
            grad, _, _ = softmax_gradient(h, 100.0)
            gnorm = np.linalg.norm(grad)
            if gnorm > 1e-14:
                grad_n = grad / gnorm
                for alpha in [0.01, 0.005, 0.002, 0.001]:
                    h_try = repair_feasible(h - alpha * grad_n)
                    c_try = exact_score(h_try)
                    if c_try < best_c - 1e-15:
                        h = h_try
                        best_c = c_try
                        best_h = h_try.copy()
                        improved = True
                        break
        if not improved:
            # Try expanded active set
            beam2 = beam * 3
            active2 = np.where(profile >= T - beam2)[0]
            if len(active2) > 40:
                top2 = np.argsort(profile[active2])[-40:]
                active2 = active2[top2]
            h_new2 = lp_step(h, active2, 0.01)
            if h_new2 is not None:
                c_new2 = exact_score(h_new2)
                if c_new2 < best_c - 1e-15:
                    h = h_new2
                    best_c = c_new2
                    best_h = h_new2.copy()
                    improved = True
        if not improved:
            break
    return best_h, best_c

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    h_new = np.interp(xt, xs, h)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

def pair_transfer(h, n_trials, seed_offset=0):
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    best_h = h.copy()
    rng = np.random.RandomState(99 + seed_offset)
    for _ in range(n_trials):
        # Weighted sampling: sources from high h, sinks from low h
        p_weights = np.clip(h, 0.005, None)
        p_weights /= p_weights.sum()
        q_weights = np.clip(1.0 - h, 0.005, None)
        q_weights /= q_weights.sum()
        p = int(rng.choice(n, p=p_weights))
        q = int(rng.choice(n, p=q_weights))
        if p == q:
            continue
        max_trans = min(h[p], 1.0 - h[q])
        if max_trans < 1e-8:
            continue
        for frac in [0.2, 0.5, 0.8]:
            eps = frac * max_trans
            h_try = h.copy()
            h_try[p] -= eps
            h_try[q] += eps
            h_try = repair_feasible(h_try)
            c_try = exact_score(h_try)
            if c_try < best_c - 1e-15:
                best_c = c_try
                best_h = h_try.copy()
                h = h_try.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()
    t_limit = t_start + budget_s - 10

    def time_ok():
        return time.perf_counter() < t_limit

    # Load parent
    parent = None
    try:
        parent = np.array(initial_h_values, dtype=np.float64)
    except (NameError, Exception):
        pass
    if parent is None or len(parent) < 4 or not np.isfinite(parent).all():
        parent = np.full(70, 0.5)
    parent = np.clip(parent, 0.0, 1.0)
    parent = repair_feasible(parent)

    best_h = parent.copy()
    best_c = exact_score(parent)

    # Phase 1: Homotopy soft-max gradient descent
    betas = [5, 10, 20, 50, 100, 200, 500, 1000]
    for beta in betas:
        if not time_ok():
            break
        t_sub = min(t_limit, time.perf_counter() + 80)
        h_opt, c_opt = gradient_step_ds(best_h, beta, max_iters=200,
                                         t_start=t_start, t_limit=t_sub)
        if c_opt < best_c - 1e-14:
            best_h = h_opt
            best_c = c_opt

    # Phase 2: Active-set LP optimization
    if time_ok():
        t_sub = min(t_limit, time.perf_counter() + 250)
        h_opt, c_opt = lp_optimize(best_h, t_start, t_sub, max_iters=300)
        if c_opt < best_c - 1e-14:
            best_h = h_opt
            best_c = c_opt

    # Phase 3: More soft-max at high beta + LP interleaved
    if time_ok():
        for beta in [200, 500, 1000, 2000]:
            if not time_ok():
                break
            t_sub = min(t_limit, time.perf_counter() + 40)
            h_opt, c_opt = gradient_step_ds(best_h, beta, max_iters=100,
                                             t_start=t_start, t_limit=t_sub)
            if c_opt < best_c - 1e-14:
                best_h = h_opt
                best_c = c_opt
            if not time_ok():
                break
            t_sub2 = min(t_limit, time.perf_counter() + 60)
            h_opt2, c_opt2 = lp_optimize(best_h, t_start, t_sub2, max_iters=100)
            if c_opt2 < best_c - 1e-14:
                best_h = h_opt2
                best_c = c_opt2

    # Phase 4: Multi-dimension search
    n0 = len(best_h)
    dims_to_try = []
    for d in [n0 - 6, n0 - 4, n0 - 2, n0 + 2, n0 + 4, n0 + 6, n0 + 8,
              n0 + 10, n0 - 8, n0 + 12]:
        if 16 <= d <= 200 and d != n0 and d % 2 == 0:
            dims_to_try.append(d)
    dims_to_try = list(set(dims_to_try))
    dims_to_try.sort(key=lambda d: abs(d - n0))

    for new_n in dims_to_try[:8]:
        if not time_ok():
            break
        h_interp = interpolate_to_n(best_h, new_n)
        c_interp = exact_score(h_interp)
        if c_interp > best_c + 0.005:
            continue
        # Optimize at new dimension
        t_sub = min(t_limit, time.perf_counter() + 60)
        h_opt, c_opt = lp_optimize(h_interp, t_start, t_sub, max_iters=80)
        if c_opt < best_c - 1e-14:
            best_h = h_opt
            best_c = c_opt
        if time_ok():
            t_sub2 = min(t_limit, time.perf_counter() + 30)
            h_opt2, c_opt2 = gradient_step_ds(best_h, 200, max_iters=80,
                                               t_start=t_start, t_limit=t_sub2)
            if c_opt2 < best_c - 1e-14:
                best_h = h_opt2
                best_c = c_opt2
            if time_ok():
                t_sub3 = min(t_limit, time.perf_counter() + 30)
                h_opt3, c_opt3 = lp_optimize(best_h, t_start, t_sub3, max_iters=50)
                if c_opt3 < best_c - 1e-14:
                    best_h = h_opt3
                    best_c = c_opt3

    # Phase 5: Pair transfer fine-tuning
    if time_ok():
        t_sub = min(t_limit, time.perf_counter() + 80)
        h_opt, c_opt = pair_transfer(best_h, n_trials=2000, seed_offset=0)
        if c_opt < best_c - 1e-14:
            best_h = h_opt
            best_c = c_opt

    # Phase 6: Final LP polish with very small steps
    if time_ok():
        t_sub = min(t_limit, time.perf_counter() + 120)
        h_opt, c_opt = lp_optimize(best_h, t_start, t_sub, max_iters=200)
        if c_opt < best_c - 1e-14:
            best_h = h_opt
            best_c = c_opt

    # Phase 7: More pair transfers with different seed
    if time_ok():
        t_sub = min(t_limit, time.perf_counter() + 60)
        h_opt, c_opt = pair_transfer(best_h, n_trials=1500, seed_offset=100)
        if c_opt < best_c - 1e-14:
            best_h = h_opt
            best_c = c_opt

    # Phase 8: Final micro-LP
    if time_ok():
        n = len(best_h)
        profile = compute_profile(best_h)
        T = np.max(profile)
        spread = T - np.min(profile)
        for beam_frac in [0.005, 0.01, 0.02, 0.05]:
            if not time_ok():
                break
            beam = beam_frac * spread
            active = np.where(profile >= T - beam)[0]
            if len(active) > 15:
                top = np.argsort(profile[active])[-15:]
                active = active[top]
            for radius in [0.005, 0.002, 0.001, 0.0005]:
                h_new = lp_step(best_h, active, radius)
                if h_new is not None:
                    c_new = exact_score(h_new)
                    if c_new < best_c - 1e-15:
                        best_h = h_new
                        best_c = c_new
                        break

    # Phase 9: Final pair transfer
    if time_ok():
        h_opt, c_opt = pair_transfer(best_h, n_trials=1000, seed_offset=200)
        if c_opt < best_c - 1e-14:
            best_h = h_opt
            best_c = c_opt

    # Final validation
    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)

    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            final_c = min(final_c, float(eval_result))
        elif isinstance(eval_result, tuple) and len(eval_result) >= 2:
            final_c = min(final_c, float(eval_result[0]))
    except Exception:
        pass

    return best_h, final_c, len(best_h)
