            h[idx] -= sub

            if rem <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-14:
        if diff2 > 0:
            h[int(np.argmax(1.0 - h))] += diff2
        else:
            h[int(np.argmax(h))] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def build_grad(h, n, dx):
    M = 2 * n - 1
    G = np.zeros((M, n))
    center = n - 1
    G[center, :] = dx * (1.0 - 2.0 * h)
    for d in range(1, n):
        idx1 = np.arange(n - d)
        G[center + d, idx1] += -dx * h[idx1 + d]
        G[center + d, idx1 + d] += dx * (1.0 - h[idx1])
        G[center - d, idx1] += dx * (1.0 - h[idx1 + d])
        G[center - d, idx1 + d] += -dx * h[idx1]
    return G

def line_search(h, v, n_grid=300):
    n = len(h)
    dx = 2.0 / n
    v = v - v.mean()
    vmax = np.max(np.abs(v))
    if vmax < 1e-12:
        return h, exact_score(h)
    v = v / vmax
    lo = -1e10
    hi = 1e10
    pos = v > 1e-14
    neg = v < -1e-14
    if np.any(pos):
        lo = max(lo, np.max(-h[pos] / v[pos]))
        hi = min(hi, np.min((1.0 - h[pos]) / v[pos]))
    if np.any(neg):
        lo = max(lo, np.min((1.0 - h[neg]) / v[neg]))
        hi = min(hi, np.max(-h[neg] / v[neg]))
    lo = max(lo, -1.5)
    hi = min(hi, 1.5)
    if lo >= hi - 1e-10:
        return h, exact_score(h)
    P0 = np.correlate(h, 1.0 - h, mode="full")
    B = np.correlate(v, 1.0 - h, mode="full") - np.correlate(h, v, mode="full")
    C = -np.correlate(v, v, mode="full")
    alphas = np.linspace(lo, hi, n_grid)
    best_val = np.inf
    best_a = 0.0
    for a in alphas:
        val = np.max(P0 + B * a + C * a * a)
        if val < best_val:
            best_val = val
            best_a = a
    i_best = int((best_a - lo) / (hi - lo) * (n_grid - 1))
    span = max(2, n_grid // 30)
    a_left = alphas[max(0, i_best - span)]
    a_right = alphas[min(n_grid - 1, i_best + span)]
    alphas2 = np.linspace(a_left, a_right, 200)
    for a in alphas2:
        val = np.max(P0 + B * a + C * a * a)
        if val < best_val:
            best_val = val
            best_a = a
    h_new = repair_feasible(h + best_a * v)
    return h_new, exact_score(h_new)

def gen_directions(h, n, profile, dx, rng, G):
    dirs = []
    c_max = np.max(profile)
    c_med = np.median(profile)
    beam = max(0.003, 0.15 * (c_max - c_med))
    active = np.where(profile >= c_max - beam)[0]
    if len(active) == 0:
        active = np.array([np.argmax(profile)])
    for k in active[:8]:
        g = -G[k, :].copy()
        g -= g.mean()
        gn = np.linalg.norm(g)
        if gn > 1e-10:
            dirs.append(g / gn)
    w = (profile[active] - (c_max - beam)) / max(beam, 1e-10) + 0.1
    gw = -(G[active, :] * w[:, None]).sum(axis=0)
    gw -= gw.mean()
    gn = np.linalg.norm(gw)
    if gn > 1e-10:
        dirs.append(gw / gn)
    x = np.arange(n)
    for m in range(1, 6):
        for ph in [0.0, 0.5 * np.pi, np.pi, 1.5 * np.pi]:
            v = np.cos(2.0 * np.pi * m * x / n + ph)
            v -= v.mean()
            gn = np.linalg.norm(v)
            if gn > 1e-10:
                dirs.append(v / gn)
        for ph in [0.0, 0.5 * np.pi]:
            v = np.sin(2.0 * np.pi * m * x / n + ph)
            v -= v.mean()
            gn = np.linalg.norm(v)
            if gn > 1e-10:
                dirs.append(v / gn)
    for _ in range(8):
        v = rng.randn(n)
        v -= v.mean()
        gn = np.linalg.norm(v)
        if gn > 1e-10:
            dirs.append(v / gn)
    v1 = (0.5 - h).copy()
    v1 -= v1.mean()
    gn = np.linalg.norm(v1)
    if gn > 1e-10:
        dirs.append(v1 / gn)
    v2 = (h - 0.5).copy()
    v2 -= v2.mean()
    gn = np.linalg.norm(v2)
    if gn > 1e-10:
        dirs.append(v2 / gn)
    v3 = (h * (1.0 - h) - 0.25).copy()
    v3 -= v3.mean()
    gn = np.linalg.norm(v3)
    if gn > 1e-10:
        dirs.append(v3 / gn)
    for _ in range(5):
        i = rng.randint(0, max(1, n - 4))
        j = rng.randint(0, max(1, n - 4))
        if abs(i - j) < 3:
            j = (i + n // 2) % n
        v = np.zeros(n)
        v[i:i+4] = 1.0
        v[j:j+4] = -1.0
        v -= v.mean()
        gn = np.linalg.norm(v)
        if gn > 1e-10:
            dirs.append(v / gn)
    return dirs

def optimize_local(h0, tl, rng, max_rounds=500):
    n = len(h0)
    dx = 2.0 / n
    h = h0.copy()
    best_c = exact_score(h)
    stagnation = 0
    for rnd in range(max_rounds):
        if tl() < 25:
            break
        profile = np.correlate(h, 1.0 - h, mode="full") * dx
        c_max = np.max(profile)
        if c_max < best_c - 1e-14:
            best_c = c_max
            stagnation = 0
        else:
            stagnation += 1
        G = build_grad(h, n, dx)
        dirs = gen_directions(h, n, profile, dx, rng, G)
        rng.shuffle(dirs)
        accepted = False
        for v in dirs:
            if tl() < 10:
                break
            h_try, c_try = line_search(h, v, n_grid=250)
            if c_try < best_c - 1e-14:
                h = h_try
                best_c = c_try
                accepted = True
                stagnation = 0
                break
        if not accepted:
            if stagnation > 20:
                scales = [0.005, 0.01, 0.02, 0.04]
                found = False
                for sc in scales:
                    if tl() < 10:
                        break
                    z = rng.randn(n)
                    z -= z.mean()
                    z /= (np.linalg.norm(z) + 1e-15)
                    h_try, c_try = line_search(h, z, n_grid=150)
                    if c_try < best_c - 1e-14:
                        h = h_try
                        best_c = c_try
                        found = True
                        stagnation = 0
                        break
                if not found:
                    h = repair_feasible(h + 0.01 * z)
                    stagnation = 0
            elif stagnation > 50:
                break
    return h, best_c

def interp_to(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    fx = np.concatenate([[-(2.0 / n)], xs, [2.0 + (2.0 / n)]])
    fh = np.concatenate([[h[0]], h, [h[-1]]])
    hn = np.interp(xt, fx, fh)
    return repair_feasible(np.clip(hn, 0, 1))

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.perf_counter()

    def tl():
        return budget_s - (time.perf_counter() - t0)

    rng = np.random.RandomState(seed)

    parent = None
    try:
        parent = np.array(initial_h_values, dtype=np.float64)
    except (NameError, Exception):
        pass
    if parent is None or len(parent) < 4:
        parent = np.full(70, 0.5)
    parent = repair_feasible(np.clip(parent, 0.0, 1.0))
    n0 = len(parent)

    best_h = parent.copy()
    best_c = exact_score(parent)

    h_sym = repair_feasible((parent + parent[::-1]) / 2.0)
    c_sym = exact_score(h_sym)
    if c_sym < best_c - 1e-12:
        best_h, best_c = h_sym, c_sym

    # Phase 1: Main optimization on parent dimension
    if tl() > 100:
        bh, bc = optimize_local(best_h, tl, rng, max_rounds=300)
        if bc < best_c - 1e-14:
            best_h, best_c = bh, bc

    # Phase 2: Symmetrize and optimize
    if tl() > 80:
        hs = repair_feasible((best_h + best_h[::-1]) / 2.0)
        cs = exact_score(hs)
        if cs < best_c - 1e-14:
            best_h, best_c = hs, cs
            bh, bc = optimize_local(best_h, tl, rng, max_rounds=150)
            if bc < best_c - 1e-14:
                best_h, best_c = bh, bc

    # Phase 3: Try different dimensions
    if tl() > 150:
        candidate_dims = []
        for d in [max(30, n0 - 20), max(30, n0 - 10), n0 + 10, n0 + 20, n0 + 40, n0 + 60]:
            if d >= 30 and d <= 200 and d != n0:
                candidate_dims.append(d)
        # Add some specific dims
        for d in [48, 60, 80, 96, 100, 120, 140, 160]:
            if d >= 30 and d <= 200 and d != n0 and d not in candidate_dims:
                candidate_dims.append(d)
        candidate_dims.sort(key=lambda d: abs(d - n0))
        for nd in candidate_dims[:8]:
            if tl() < 40:
                break
            hi = interp_to(best_h, nd)
            ci = exact_score(hi)
            if ci < best_c + 0.03:
                bh, bc = optimize_local(hi, tl, rng, max_rounds=100)
                if bc < best_c - 1e-14:
                    best_h, best_c = bh, bc

    # Phase 4: Second pass on best dimension
    if tl() > 100:
        bh, bc = optimize_local(best_h, tl, rng, max_rounds=300)
        if bc < best_c - 1e-14:
            best_h, best_c = bh, bc

    # Phase 5: More dimension exploration around best
    if tl() > 120:
        n_best = len(best_h)
        for offset in [-6, -4, -2, 2, 4, 6, 12, -12]:
            if tl() < 30:
                break
            nd = n_best + offset
            if nd < 30 or nd > 250:
                continue
            hi = interp_to(best_h, nd)
            ci = exact_score(hi)
            if ci < best_c + 0.01:
                bh, bc = optimize_local(hi, tl, rng, max_rounds=80)
                if bc < best_c - 1e-14:
                    best_h, best_c = bh, bc

    # Phase 6: Random restarts with line search
    if tl() > 100:
        n_best = len(best_h)
        for eps in [0.03, 0.05, 0.08, 0.12, 0.15]:
            if tl() < 30:
                break
            z = rng.randn(n_best)
            z -= z.mean()
            z /= (np.linalg.norm(z) + 1e-15)
            hp = repair_feasible(best_h + eps * z)
            cp = exact_score(hp)
            if cp < best_c + 0.02:
                bh, bc = optimize_local(hp, tl, rng, max_rounds=80)
                if bc < best_c - 1e-14:
                    best_h, best_c = bh, bc

    # Phase 7: Final refinement
    if tl() > 60:
        bh, bc = optimize_local(best_h, tl, rng, max_rounds=400)
        if bc < best_c - 1e-14:
            best_h, best_c = bh, bc

    # Phase 8: Symmetrize final
    if tl() > 20:
        hs = repair_feasible((best_h + best_h[::-1]) / 2.0)
        cs = exact_score(hs)
        if cs < best_c - 1e-14:
            best_h, best_c = hs, cs

    # Final validation
    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)

    try:
        er = evaluate_erdos_solution(best_h)
        if isinstance(er, (float, int)):
            final_c = min(final_c, float(er))
        elif isinstance(er, tuple) and len(er) >= 2:
            final_c = min(final_c, float(er[0]))
    except Exception:
        pass

    return best_h, final_c, len(best_h)
