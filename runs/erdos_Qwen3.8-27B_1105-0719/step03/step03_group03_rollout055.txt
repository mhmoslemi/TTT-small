    n = len(h)

    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    fx = np.concatenate([[0.0], xs, [2.0]])
    fh = np.concatenate([[h[0]], h, [h[-1]]])
    hn = np.interp(xt, fx, fh)
    return project_box_sum(hn, new_n / 2.0)


def cd_once(h, m, profile, G):
    """One coordinate descent step: find optimal h[m]."""
    n = len(h)
    M = len(profile)
    A = G[:, m]
    B = profile - A * h[m]
    
    # Find x in [0,1] minimizing max_k (A_k*x + B_k)
    # Use ternary-search-like approach for convex PL function
    # Or: find all breakpoints and check
    
    # Efficient approach: the minimum of a convex PL function on [0,1]
    # is at 0, 1, or at a breakpoint where the derivative changes sign
    
    # Compute the "upper envelope" by finding which lines are on top
    # For each x candidate, max_k (A_k*x + B_k)
    
    # Simple approach: check 0, 1, and the O(M) breakpoints that are relevant
    # A breakpoint between lines i and j is at x = (B_j - B_i)/(A_i - A_j)
    
    # But checking all C(M,2) breakpoints is expensive. Instead:
    # The minimum must be at a point where the subdifferential contains 0,
    # which means there exist i,j with A_i < 0 < A_j (or A_i = 0)
    # and A_i*x + B_i = A_j*x + B_j = max_k (A_k*x + B_k)
    
    # Faster: just do a golden section search (works for unimodal functions)
    # But the function is convex, so ternary search works!
    
    # Actually for convex PL, bisection on the derivative works:
    # f'(x) = max active k of A_k (the slope of the upper envelope)
    
    def f_of_x(x):
        return np.max(A * x + B)
    
    # Binary search for where f' changes sign
    # f'(x) >= 0 means the rightmost active line has A >= 0
    # f'(x-) <= 0 means the leftmost active line has A <= 0
    
    lo, hi = 0.0, 1.0
    for _ in range(50):
        mid = (lo + hi) / 2
        vals = A * mid + B
        idx_max = np.argmax(vals)
        # Check if mid is a breakpoint (multiple lines tied at max)
        # If A[idx_max] > 0, function is increasing at mid, go left
        # If A[idx_max] < 0, function is decreasing at mid, go right
        # If A[idx_max] == 0 or multiple tied, we're at the minimum
        # But we need to be more careful at breakpoints
        
        # Use subgradient: if max A among active > 0, go left
        # if min A among active < 0, go right
        active = np.where(vals >= vals[idx_max] - 1e-12)[0]
        if np.max(A[active]) > 1e-10:
            hi = mid
        elif np.min(A[active]) < -1e-10:
            lo = mid
        else:
            break
    
    x_opt = (lo + hi) / 2
    return x_opt

import numpy as np
from scipy.optimize import minimize
import time

try:
    import cvxpy as cp
    HAS_CVXPY = True
except ImportError:
    HAS_CVXPY = False


def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))


def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx


def build_G(h, dx):
    n = len(h)
    M = 2 * n - 1
    center = n - 1
    m_arr = np.arange(n)
    G = np.zeros((M, n))
    for k in range(M):
        d = k - center
        as_i = (m_arr + d >= 0) & (m_arr + d <= n - 1)
        as_j = (m_arr - d >= 0) & (m_arr - d <= n - 1)
        if np.any(as_i):
            G[k, as_i] += (1.0 - h[m_arr[as_i] + d])
        if np.any(as_j):
            G[k, as_j] += -h[m_arr[as_j] - d]
    G *= dx
    return G


def project_box_sum(v, target):
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    lo, hi = -10.0, 10.0
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2
    h = np.clip(v - lam, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.argsort(-(1.0 - h))
        else:
            idx = np.argsort(h)
        remaining = abs(diff)
        for i in idx:
            if diff > 0:
                add = min(remaining, 1.0 - h[i])
                h[i] += add
            else:
                add = min(remaining, h[i])
                h[i] -= add
            remaining -= add
            if remaining < 1e-14:
                break
    return h


def interp_h(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    fx = np.concatenate([[0.0], xs, [2.0]])
    fh = np.concatenate([[h[0]], h, [h[-1]]])
    hn = np.interp(xt, fx, fh)
    return project_box_sum(hn, new_n / 2.0)


def cd_once(h, m, profile, G):
    n = len(h)
    M = len(profile)
    A = G[:, m]
    B = profile - A * h[m]

    def f_of_x(x):
        return np.max(A * x + B)

    lo, hi = 0.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        vals = A * mid + B
        v_max = np.max(vals)
        active = np.where(vals >= v_max - 1e-14)[0]
        a_max = np.max(A[active])
        a_min = np.min(A[active])
        if a_max > 1e-12:
            hi = mid
        elif a_min < -1e-12:
            lo = mid
        else:
            lo = hi = mid
            break
    x_opt = (lo + hi) / 2
    return float(np.clip(x_opt, 0.0, 1.0))


def cd_sweep(h, max_sweeps=200, budget_fn=None):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_c = exact_score(h)
    best_h = h.copy()

    for sweep in range(max_sweeps):
        if budget_fn is not None and budget_fn() < 5:
            break
        improved = False
        G = build_G(h, dx)
        profile = compute_profile(h)

        order = list(range(n))
        if sweep % 2 == 1:
            order = order[::-1]

        for m in order:
            x_opt = cd_once(h, m, profile, G)
            if abs(x_opt - h[m]) > 1e-10:
                h_save = h[m]
                h[m] = x_opt
                h = project_box_sum(h, target)
                profile = compute_profile(h)
                c_new = np.max(profile)
                if c_new < best_c - 1e-14:
                    best_c = c_new
                    best_h = h.copy()
                    improved = True
                    G = build_G(h, dx)
                else:
                    h[m] = h_save
                    h = project_box_sum(h, target)
                    profile = compute_profile(h)
                    G = build_G(h, dx)

        if not improved:
            break

    return best_h, best_c


def momentum_subgrad(h0, max_iters=3000, budget_fn=None):
    n = len(h0)
    dx = 2.0 / n
    target = n / 2.0
    h = h0.copy()
    v = np.zeros(n)
    beta = 0.85
    best_c = exact_score(h)
    best_h = h.copy()
    step = 0.008

    for it in range(max_iters):
        if budget_fn is not None and budget_fn() < 3:
            break
        profile = compute_profile(h)
        G = build_G(h, dx)

        c_max = np.max(profile)
        spread = c_max - np.median(profile)
        tol = max(0.001, 0.03 * spread)
        active = np.where(profile >= c_max - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(profile)])

        w = np.zeros(2 * n - 1)
        w[active] = 1.0 / len(active)
        grad = G.T @ w
        grad -= grad.mean()

        v = beta * v - step * grad
        h_try = project_box_sum(h + v, target)
        c_try = exact_score(h_try)

        if c_try < best_c - 1e-14:
            h = h_try
            best_c = c_try
            best_h = h.copy()
        else:
            v = 0.5 * v
            h_try = project_box_sum(h + v, target)
            c_try = exact_score(h_try)
            if c_try < best_c - 1e-14:
                h = h_try
                best_c = c_try
                best_h = h.copy()
            else:
                step *= 0.9
                if step < 0.0001:
                    step = 0.003
                    beta = 0.8

    return best_h, best_c


def swap_improve(h, max_passes=20, budget_fn=None):
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)

    for p in range(max_passes):
        if budget_fn is not None and budget_fn() < 5:
            break
        improved = False
        indices = np.random.permutation(n)
        for i in range(n):
            m = indices[i]
            for j in range(i + 1, min(i + 15, n)):
                jj = indices[j]
                if abs(h[m] - h[jj]) < 1e-8:
                    continue
                h[m], h[jj] = h[jj], h[m]
                c_new = exact_score(h)
                if c_new < best_c - 1e-14:
                    best_c = c_new
                    improved = True
                else:
                    h[m], h[jj] = h[jj], h[m]
        if not improved:
            break

    return h, best_c


def lp_local(h0, active, dx, rho, trust):
    if not HAS_CVXPY:
        return None
    n = len(h0)
    target = n / 2.0
    G = build_G(h0, dx)
    G_a = G[active, :]
    profile = compute_profile(h0)
    b_a = profile[active] - G_a @ h0

    h_var = cp.Variable(n)
    t_var = cp.Variable()
    cons = [
        h_var >= 0,
        h_var <= 1,
        cp.sum(h_var) == target,
        h_var >= h0 - trust,
        h_var <= h0 + trust,
        G_a @ h_var - t_var <= -b_a,
    ]
    obj = cp.Minimize(t_var + rho * cp.norm1(h_var - h0))
    prob = cp.Problem(obj, cons)
    for solver in [cp.ECOS, cp.SCS, cp.CBC]:
        try:
            prob.solve(solver=solver, max_iters=10000, abstol=1e-12)
            break
        except Exception:
            continue
    if h_var.value is None:
        return None
    h_sol = np.array(h_var.value).flatten()
    if not np.isfinite(h_sol).all():
        return None
    return project_box_sum(h_sol, target)
