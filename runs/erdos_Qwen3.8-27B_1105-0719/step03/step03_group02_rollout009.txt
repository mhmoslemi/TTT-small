from scipy.special import logsumexp
import time

# Helper functions (all top-level)

def project_feasible(x, target_sum):
    """Water-filling projection to [0,1]^n with sum = target_sum"""
    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def c5_of_h(h, dx):
    return np.correlate(h, 1.0 - h, mode='full').max() * dx

def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def grad_corr_k(h, k, n):
    offset = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -offset)
    i_hi = min(n - 1, n - 1 - offset)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx + offset] += (1.0 - h[idx])
    grad[idx] += (-h[idx + offset])
    return grad

def smooth_max(h, dx, beta):
    corrs = compute_corrs(h)
    return logsumexp(beta * corrs) / beta * dx

def smooth_max_grad(h, dx, beta):
    n = len(h)
    corrs = compute_corrs(h)
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones_like(corrs) / len(corrs)
    else:
        w /= w_sum
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] > 1e-13:
            grad += w[k] * grad_corr_k(h, k, n)
    return grad * dx

def target_penalty(h, dx, target):
    corrs = compute_corrs(h) * dx
    excess = np.maximum(0, corrs - target)
    return (excess ** 2).sum()

def target_penalty_grad(h, dx, target):
    n = len(h)
    corrs = compute_corrs(h)
    q = corrs * dx
    excess = np.maximum(0, q - target)
    mask = excess > 1e-12
    if not mask.any():
        return np.zeros(n)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if mask[k]:
            grad += 2 * (q[k] - target) * dx * grad_corr_k(h, k, n)
    return grad

def slsqp_optimize(h0, dx, target_sum, beta, maxiter=2000):
    n = len(h0)
    
    def obj(x):
        return smooth_max(x, dx, beta)
    
    def grad_obj(x):
        return smooth_max_grad(x, dx, beta)
    
    def c_fun(x):
        return x.sum() - target_sum
    
    def c_jac(x):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': c_fun, 'jac': c_jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0, jac=grad_obj, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        h_out = project_feasible(res.x, target_sum)
    except Exception:
        h_out = project_feasible(h0, target_sum)
    return h_out, c5_of_h(h_out, dx)

def lp_polish(h0, dx, target_sum, max_iters=200, init_step=0.15, n_active_max=80):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    step = init_step
    stall_count = 0

    for it in range(max_iters):
        corrs = compute_corrs(h)
        c5_val = corrs.max() * dx
        max_c = corrs.max()

        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            stall_count = 0
        else:
            stall_count += 1
        if stall_count > 50:
            step *= 0.7
            stall_count = 0
            if step < 1e-8:
                break

        tol_frac = 0.05
        tol_val = max_c * (1.0 - tol_frac)
        active_k = np.where(corrs >= tol_val)[0]
        if len(active_k) > n_active_max:
            active_k = np.argsort(corrs)[-n_active_max:]

        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        A_rows = []
        b_rows = []
        for k in active_k:
            row = np.zeros(n + 1)
            row[:n] = dx * grad_corr_k(h, k, n)
            row[n] = -1.0
            A_rows.append(row)
            b_rows.append(-corrs[k] * dx - 1e-14)

        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)

        bounds = []
        for i in range(n):
            bounds.append((max(-h[i], -step), min(1.0 - h[i], step)))
        bounds.append((0.0, None))

        try:
            res = linprog(c_obj, A_ub=np.array(A_rows), b_ub=np.array(b_rows),
                          A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
        except Exception:
            break
        if res.status != 0:
            break

        delta = res.x[:n]
        if np.max(np.abs(delta)) < 1e-14:
            break

        h_new = project_feasible(h + delta, target_sum)
        new_c5 = c5_of_h(h_new, dx)

        if new_c5 < c5_val - 1e-14:
            h = h_new
            step = min(step * 1.1, init_step)
        else:
            improved = False
            for frac in [0.5, 0.25, 0.1, 0.05]:
                h_new = project_feasible(h + frac * delta, target_sum)
                new_c5 = c5_of_h(h_new, dx)
                if new_c5 < c5_val - 1e-14:
                    h = h_new
                    improved = True
                    break
            if not improved:
                step *= 0.5
                if step < 1e-8:
                    break

    return best_h, best_c5

def binary_sa(h0, dx, target_sum, n_iter=50000, t_start=1.0, t_end=0.01, rng=None):
    """Simulated annealing on binary strings"""
    n = len(h0)
    if rng is None:
        rng = np.random.RandomState(42)
    
    # Threshold to binary
    sorted_idx = np.argsort(h0)[::-1]
    b = np.zeros(n)
    b[sorted_idx[:n//2]] = 1.0
    
    best_b = b.copy()
    best_c5 = c5_of_h(b, dx)
    current_c5 = best_c5
    current_corrs = compute_corrs(b)
    
    ones = np.where(b == 1.0)[0]
    zeros = np.where(b == 0.0)[0]
    
    for it in range(n_iter):
        t = t_start * (t_end / t_start) ** (it / max(1, n_iter))
        
        # Random swap
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        b[i] = 0.0
        b[j] = 1.0
        new_corrs = compute_corrs(b)
        new_c5 = new_corrs.max() * dx
        
        delta_c = new_c5 - current_c5
        if delta_c < 0 or rng.random() < np.exp(-delta_c / (t * dx + 1e-15)):
            # Accept
            current_c5 = new_c5
            current_corrs = new_corrs
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_b = b.copy()
            # Update ones/zeros
            ones = np.where(b == 1.0)[0]
            zeros = np.where(b == 0.0)[0]
        else:
            # Reject
            b[i] = 1.0
            b[j] = 0.0
    
    return best_b, best_c5

def binary_local_search(h0, dx, target_sum, n_iter=10000, rng=None):
    """Greedy local search on binary strings with random restarts"""
    n = len(h0)
    if rng is None:
        rng = np.random.RandomState(42)
    
    sorted_idx = np.argsort(h0)[::-1]
    b = np.zeros(n)
    b[sorted_idx[:n//2]] = 1.0
    
    best_b = b.copy()
    best_c5 = c5_of_h(b, dx)
    current_c5 = best_c5
    
    for it in range(n_iter):
        # Try random swap
        ones = np.where(b == 1.0)[0]
        zeros = np.where(b == 0.0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        b[i] = 0.0
        b[j] = 1.0
        new_c5 = c5_of_h(b, dx)
        
        if new_c5 <= current_c5:
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_b = b.copy()
        else:
            b[i] = 1.0
            b[j] = 0.0
    
    return best_b, best_c5

def target_descent(h0, dx, target_sum, initial_target_frac=0.99, n_iter=5000, lr=0.01):
    """Target-based penalty descent"""
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    
    target = best_c5 * initial_target_frac
    lr_schedule = np.ones(n_iter) * lr
    for i in range(n_iter // 2):
        lr_schedule[n_iter // 2 + i] = lr * (1 - i / (n_iter // 2))
    
    momentum = np.zeros(n)
    beta_mom = 0.9
    
    for it in range(n_iter):
        pen = target_penalty(h, dx, target)
        if pen < 1e-18:
            # All correlations below target, try to lower target
            target *= 0.999
            continue
        
        grad = target_penalty_grad(h, dx, target)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        momentum = beta_mom * momentum - lr_schedule[it] * grad / gnorm
        h_new = h + momentum
        h_new = project_feasible(h_new, target_sum)
        
        new_c5 = c5_of_h(h_new, dx)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
        
        h = h_new
    
    return best_h, best_c5

def make_barker_start(n, rng):
    """Create a Barker-like starting vector"""
    # Use a quadratic residue pattern
    h = np.zeros(n)
    for i in range(n):
        # Quadratic residue test (mod a prime near n)
        p = n
        if i == 0:
            h[i] = 1.0
        else:
            h[i] = 1.0 if (pow(i, p - 1, 2 * p) == 1) else 0.0
    return project_feasible(h, n / 2.0)

def make_chirp_start(n, rng):
    """Create a chirp-like starting vector using sinusoidal pattern"""
    x = np.arange(n)
    # Frequency increases over time
    phase = 2 * np.pi * 0.3 * (x / n) ** 2
    h = 0.5 + 0.5 * np.sin(phase)
    return project_feasible(h, n / 2.0)

def make_random_binary(n, rng):
    """Random balanced binary vector"""
    idx = rng.permutation(n)
    h = np.zeros(n)
    h[idx[:n//2]] = 1.0
    return h

def full_pipeline(h0, dx, target_sum, time_left_fn, label=""):
    """Run the full optimization pipeline on a starting point"""
    n = len(h0)
    best_h = h0.copy()
    best_c5 = c5_of_h(h0, dx)
    
    # Step 1: SLSQP with increasing beta
    for beta in [100, 300, 800, 2000, 5000]:
        if time_left_fn() < 10:
            break
        h_new, c5_new = slsqp_optimize(best_h, dx, target_sum, beta, 2000)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new
    
    # Step 2: LP polish
    if time_left_fn() > 15:
        h_new, c5_new = lp_polish(best_h, dx, target_sum, max_iters=300, init_step=0.12)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new
    
    # Step 3: Target descent
    if time_left_fn() > 20:
        h_new, c5_new = target_descent(best_h, dx, target_sum, n_iter=3000, lr=0.02)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new
    
    # Step 4: Binary SA (threshold current solution)
    if time_left_fn() > 30:
        b, c5_b = binary_sa(best_h, dx, target_sum, n_iter=20000, rng=np.random.RandomState(hash(label) % 2**31))
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = b
    
    # Step 5: Back to continuous from binary
    if time_left_fn() > 20:
        h_new, c5_new = slsqp_optimize(best_h, dx, target_sum, 1000, 2000)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new
        if time_left_fn() > 10:
            h_new, c5_new = lp_polish(best_h, dx, target_sum, max_iters=200, init_step=0.08)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new
    
    return best_h, best_c5
