Let me think about the LP formulation more carefully:

At point h, we have:
- Profile P = correlate(h, 1-h, mode="full") * dx
- Gradient matrix G where G[m,i] = ∂P[m]/∂h[i]
- Current max c = max(P)

The LP:
- Variables: d ∈ R^n (direction), S ∈ R (predicted peak)
- Minimize S
- Subject to:
  - G[m,:] · d <= S - P[m] for all m = 0, ..., 2n-2
  - sum(d) = 0
  - -D <= d_i <= D for all i

If the optimal S* < c, then moving in direction d should reduce the max.

Then we do a line search along d to find the best actual step.

This is fundamentally different from:
- LSE: uses exp(beta * profile) weighting, which is an approximation
- Active-set: uses a fixed subset of constraints
- SLSQP: uses interior point method on the full nonlinear problem

The LP approach directly solves the first-order approximation of the minimax problem, which should give better directions when the profile is nearly piecewise-linear (which it is, since each P[m] is quadratic in h but we're only considering first-order changes).

import numpy as np
from scipy.optimize import linprog, minimize
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx] + 1e-15)
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx] + 1e-15)
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-14:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmin(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def build_gradient_matrix(h, n, dx):
    M = 2 * n - 1
    G = np.zeros((M, n))
    for m in range(M):
        d = m - (n - 1)
        if d == 0:
            G[m, :] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            G[m, d:] += dx * (1.0 - h[0:n-d])
            G[m, 0:n-d] += -dx * h[d:n]
        else:
            dp = -d
            G[m, 0:n-dp] += dx * (1.0 - h[dp:n])
            G[m, dp:n] += -dx * h[0:n-dp]
    return G

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = np.linspace(0, 2, n + 1)
    xt = np.linspace(0, 2, new_n + 1)
    h_interp = np.zeros(new_n)
    for k in range(new_n):
        lo_k, hi_k = xt[k], xt[k+1]
        total_mass = 0.0
        total_width = 0.0
        for j in range(n):
            lo_j, hi_j = xs[j], xs[j+1]
            overlap = min(hi_k, hi_j) - max(lo_k, lo_j)
            if overlap > 0:
                total_mass += h[j] * overlap
                total_width += overlap
        h_interp[k] = total_mass / total_width if total_width > 1e-15 else 0.5
    h_interp = np.clip(h_interp, 0.0, 1.0)
    return repair_feasible(h_interp)

def symmetrize(h):
    return (h + h[::-1]) / 2.0

def antisymmetrize(h):
    n = len(h)
    h_out = np.zeros(n)
    for i in range(n):
        h_out[i] = (h[i] + (1.0 - h[n-1-i])) / 2.0
    return repair_feasible(h_out)

def safe_step_bound(h, direction):
    """Find maximum step size along direction staying in [0,1]^n"""
    n = len(h)
    max_step = 1e10
    for i in range(n):
        if direction[i] > 1e-15:
            max_step = min(max_step, (1.0 - h[i]) / direction[i])
        elif direction[i] < -1e-15:
            max_step = min(max_step, h[i] / (-direction[i]))
    return max_step

def line_search(h, direction, max_iters=40):
    """Line search along direction, finding best feasible point"""
    n = len(h)
    dx = 2.0 / n
    best_score = exact_score(h)
    best_h = h.copy()
    
    # Normalize direction
    d_norm = np.linalg.norm(direction)
    if d_norm < 1e-15:
        return best_h, best_score
    direction = direction / d_norm
    
    # Remove mean to preserve mass (approximately)
    # Actually we'll use repair_feasible after each step
    
    bound_fwd = safe_step_bound(h, direction)
    bound_bwd = safe_step_bound(h, -direction)
    
    # Try forward and backward
    for sign in [1.0, -1.0]:
        bound = bound_fwd if sign > 0 else bound_bwd
        if bound < 1e-10:
            continue
        # Sample the line
        # Use logarithmic + uniform sampling
        steps = []
        for i in range(8):
            steps.append(bound * (0.1 + 0.8 * i / 7.0))
        # Add some fine steps near the best
        steps = np.array(sorted(steps))
        
        local_best = best_score
        local_best_h = best_h.copy()
        
        for t in steps:
            h_try = repair_feasible(h + sign * t * direction)
            s = exact_score(h_try)
            if s < local_best - 1e-15:
                local_best = s
                local_best_h = h_try.copy()
        
        # Refine around best
        if local_best < best_score:
            # Find where the best was
            # Do a finer search
            best_t = 0
            for i in range(len(steps)):
                h_try = repair_feasible(h + sign * steps[i] * direction)
                if exact_score(h_try) == local_best:
                    best_t = steps[i]
                    break
            # Fine search around best_t
            for i in range(20):
                t_fine = best_t * (0.7 + 0.6 * i / 19.0)
                if t_fine > bound:
                    t_fine = bound * 0.99
                h_try = repair_feasible(h + sign * t_fine * direction)
                s = exact_score(h_try)
                if s < local_best - 1e-15:
                    local_best = s
                    local_best_h = h_try.copy()
            
            if local_best < best_score:
                best_score = local_best
                best_h = local_best_h
    
    return best_h, best_score

def lp_minimax_direction(h, D=0.5, active_margin=0.05):
    """
    Solve LP to find direction d that minimizes the predicted max profile.
    min S
    s.t. G[m,:] @ d <= S - P[m] for active m
         sum(d) = 0
         -D <= d <= D
    """
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    profile = compute_profile(h)
    c = float(np.max(profile))
    
    # Select active constraints (profile close to max)
    margin = max(active_margin, 3.0 * (c - float(np.median(profile))))
    active = np.where(profile >= c - margin)[0]
    if len(active) < 3:
        active = np.argsort(profile)[-min(10, M):]
    
    G = build_gradient_matrix(h, n, dx)
    G_a = G[active, :]
    P_a = profile[active]
    
    # Variables: [d_0, ..., d_{n-1}, S]
    # Objective: minimize S -> c_obj = [0,...,0, 1]
    nvar = n + 1
    c_obj = np.zeros(nvar)
    c_obj[-1] = 1.0
    
    # Inequality constraints: G_a[:,i] * d_i - S <= -P_a[m]
    # i.e., G_a @ d - S <= -P_a
    # A_ub @ x <= b_ub
    A_ub = np.zeros((len(active), nvar))
    A_ub[:, :n] = G_a
    A_ub[:, n] = -1.0
    b_ub = -P_a
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, nvar))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds: -D <= d_i <= D, S free (but bounded)
    bounds = [(-D, D)] * n + [(-c, c)]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
        if res.status == 0:
            d = res.x[:n]
            S = res.x[n]
            return d, S
        else:
            # Try with fewer active constraints
            if len(active) > 5:
                top_k = active[np.argsort(profile[active])[-5:]]
                G_a2 = G[top_k, :]
                P_a2 = profile[top_k]
                A_ub2 = np.zeros((len(top_k), nvar))
                A_ub2[:, :n] = G_a2
                A_ub2[:, n] = -1.0
                b_ub2 = -P_a2
                res2 = linprog(c_obj, A_ub=A_ub2, b_ub=b_ub2, A_eq=A_eq, b_eq=b_eq,
                             bounds=bounds, method='highs')
                if res2.status == 0:
                    return res2.x[:n], res2.x[n]
    except Exception:
        pass
    return None, None

def lp_descent(h, max_iters=200, D_list=[0.3, 0.5, 0.8], active_margin=0.05):
    """LP-based minimax descent"""
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    stuck = 0
    
    for it in range(max_iters):
        d, S_pred = lp_minimax_direction(h, D=D_list[it % len(D_list)], 
                                         active_margin=active_margin)
        if d is None:
            break
        
        # Line search along d
        h_new, c_new = line_search(h, d)
        
        if c_new < best_c - 1e-14:
            h = h_new
            best_c = c_new
            stuck = 0
        else:
            stuck += 1
            if stuck > 20:
                break
    
    return h, best_c

def lse_descent(h, beta=80.0, max_iters=150, alpha_init=0.005):
    """Log-sum-exp descent"""
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best_c = exact_score(h)
    alpha = alpha_init
    stuck = 0
    
    for it in range(max_iters):
        profile = compute_profile(h)
        scaled = beta * profile
        max_s = np.max(scaled)
        shifted = scaled - max_s
        weights = np.exp(shifted)
        wsum = weights.sum()
        if wsum < 1e-300:
            break
        weights /= wsum
        G = build_gradient_matrix(h, n, dx)
        grad = G.T @ weights
        grad -= grad.mean()
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad /= gnorm
        
        improved = False
        for a in [alpha, alpha*0.5, alpha*0.25, alpha*0.1]:
            h_try = repair_feasible(h - a * grad)
            s = exact_score(h_try)
            if s < best_c - 1e-14:
                h = h_try
                best_c = s
                improved = True
                alpha = min(alpha * 1.1, 0.03)
                stuck = 0
                break
        if not improved:
            alpha *= 0.5
            stuck += 1
            if stuck > 30 or alpha < 1e-9:
                break
    
    return h, best_c

def active_set_descent(h, max_iters=200, n_active=12, alpha_init=0.005):
    """Active-set descent"""
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best_c = exact_score(h)
    alpha = alpha_init
    stuck = 0
    
    for it in range(max_iters):
        profile = compute_profile(h)
        c = float(np.max(profile))
        if c + 1e-14 < best_c:
            best_c = c
            stuck = 0
            alpha = min(alpha * 1.08, 0.025)
        margin = max(0.003, 4.0 * (c - float(np.median(profile))))
        active = np.where(profile >= c - margin)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(profile))])
        if len(active) > n_active:
            top_idx = np.argsort(profile[active])[-n_active:]
            active = active[top_idx]
        G = build_gradient_matrix(h, n, dx)
        G_a = G[active, :]
        w = np.ones(len(active)) / len(active)
        grad = G_a.T @ w
        grad -= grad.mean()
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad /= gnorm
        
        improved = False
        for a in [alpha, alpha*0.5, alpha*0.25, alpha*0.1, alpha*0.04]:
            h_try = repair_feasible(h - a * grad)
            s = exact_score(h_try)
            if s < best_c - 1e-14:
                h = h_try
                best_c = s
                improved = True
                break
            elif s < c - 1e-14:
                h = h_try
                improved = True
                break
        if not improved:
            alpha *= 0.55
            stuck += 1
            if stuck > 40 or alpha < 1e-9:
                break
    
    return h, best_c

def pair_moves(h, max_pairs=50, max_iters=100):
    """Systematic pair moves: transfer mass between coordinates"""
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    stuck = 0
    
    for it in range(max_iters):
        improved = False
        profile = compute_profile(h)
        c = float(np.max(profile))
        margin = max(0.002, 3.0 * (c - float(np.median(profile))))
        active = np.where(profile >= c - margin)[0]
        G = build_gradient_matrix(h, n, 2.0/n)
        
        # Compute gradient for active set
        w = np.ones(len(active)) / len(active)
        grad = G[active, :].T @ w
        
        # Find best pair: (i, j) where grad[i] > 0 and grad[j] < 0
        pos = np.where(grad > 0.01)[0]
        neg = np.where(grad < -0.01)[0]
        
        if len(pos) == 0 or len(neg) == 0:
            stuck += 5
        else:
            # Try top few pairs
            pos_sorted = pos[np.argsort(-grad[pos])][:5]
            neg_sorted = neg[np.argsort(grad[neg])][:5]
            
            for i in pos_sorted:
                for j in neg_sorted:
                    # Transfer small amount from i to j
                    # direction: -e_i + e_j
                    direction = np.zeros(n)
                    direction[j] = 1.0
                    direction[i] = -1.0
                    h_new, c_new = line_search(h, direction, max_iters=15)
                    if c_new < best_c - 1e-14:
                        h = h_new
                        best_c = c_new
                        improved = True
                        break
                if improved:
                    break
        
        if not improved:
            stuck += 1
            if stuck > 20:
                break
    
    return h, best_c

def slsqp_solve(h0, maxiter=200):
    n = len(h0)
    dx = 2.0 / n
    M = 2 * n - 1
    c0 = exact_score(h0)
    def obj(x):
        return x[-1]
    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    def eq_fun(x):
        return x[:-1].sum() - n / 2.0
    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g
    def ineq_fun(x):
        h = np.clip(x[:-1], 0, 1)
        t = x[-1]
        profile = np.correlate(h, 1 - h, mode="full") * dx
        return t - profile
    def ineq_jac(x):
        h = np.clip(x[:-1], 0, 1)
        G = build_gradient_matrix(h, n, dx)
        J = np.zeros((M, n + 1))
        J[:, :-1] = -G
        J[:, -1] = 1.0
        return J
    x0 = np.concatenate([h0, [c0 + 1e-3]])
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 0.5)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None and np.isfinite(res.x).all():
            h_new = repair_feasible(res.x[:-1])
            c_new = exact_score(h_new)
            if c_new <= c0 + 1e-9:
                return h_new, c_new
    except Exception:
        pass
    return None, None

def random_direction_descent(h, rng, n_dirs=10, max_iters=30):
    """Try random directions with line search"""
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    stuck = 0
    
    for it in range(max_iters):
        improved = False
        for _ in range(n_dirs):
            d = rng.randn(n)
            d -= d.mean()
            dnorm = np.linalg.norm(d)
            if dnorm < 1e-10:
                continue
            d /= dnorm
            h_new, c_new = line_search(h, d, max_iters=20)
            if c_new < best_c - 1e-14:
                h = h_new
                best_c = c_new
                improved = True
                break
        if not improved:
            stuck += 1
            if stuck > 10:
                break
    
    return h, best_c

        J[:, -1] = 1.0
        return J
    x0 = np.concatenate([h0, [c0 + 1e-3]])
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 0.5)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None and np.isfinite(res.x).all():
            h_new = repair_feasible(res.x[:-1])
            c_new = exact_score(h_new)
            if c_new <= c0 + 1e-9:
                return h_new, c_new
    except Exception:
        pass
    return None, None

def random_descent(h, rng, n_dirs=6, max_iters=20):
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    stuck = 0
    for it in range(max_iters):
        improved = False
        for _ in range(n_dirs):
            d = rng.randn(n)
            d -= d.mean()
            dn = np.linalg.norm(d)
            if dn < 1e-10:
                continue
            d /= dn
            h_new, c_new = line_search(h, d, n_samples=15)
            if c_new < best_c - 1e-14:
                h = h_new
                best_c = c_new
                improved = True
                break
        if not improved:
            stuck += 1
            if stuck > 8:
                break
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()
    rng = np.random.RandomState(seed)
    np.random.seed(seed)

    def time_left():
        return budget_s + t_start - time.perf_counter()

    parent = None
    try:
        parent = np.array(initial_h_values, dtype=np.float64)
    except NameError:
        pass
    except Exception:
        pass

    if parent is None or len(parent) < 4:
        parent = np.full(70, 0.5)

    parent = np.clip(parent, 0.0, 1.0)
    parent = repair_feasible(parent)
    best_h = parent.copy()
    best_c = exact_score(parent)

    # Symmetry improvements
    h_sym = repair_feasible(symmetrize(parent))
    c_sym = exact_score(h_sym)
    if c_sym < best_c - 1e-12:
        best_h = h_sym
        best_c = c_sym
    h_asym = antisymmetrize(parent)
    c_asym = exact_score(h_asym)
    if c_asym < best_c - 1e-12:
        best_h = h_asym
        best_c = c_asym

    n_parent = len(best_h)

    # Stage 1: LP-based minimax (main novel component)
    if time_left() > 250:
        h_lp, c_lp = lp_descent(best_h, max_iters=80, D_list=[0.4, 0.6, 0.3], margin=0.04)
        if c_lp < best_c - 1e-13:
            best_h = h_lp.copy()
            best_c = c_lp
    if time_left() > 200:
        h_lp, c_lp = lp_descent(best_h, max_iters=80, D_list=[0.2, 0.35, 0.5], margin=0.025)
        if c_lp < best_c - 1e-13:
            best_h = h_lp.copy()
            best_c = c_lp

    # Stage 2: LSE descent
    if time_left() > 180:
        h_lse, c_lse = lse_descent(best_h, beta=80, max_iters=80)
        if c_lse < best_c - 1e-13:
            best_h = h_lse.copy()
            best_c = c_lse
        h_lse2, c_lse2 = lse_descent(best_h, beta=120, max_iters=60)
        if c_lse2 < best_c - 1e-13:
            best_h = h_lse2.copy()
            best_c = c_lse2

    # Stage 3: Active set
    if time_left() > 150:
        h_ase, c_ase = active_set_descent(best_h, max_iters=100, n_active=10, alpha_init=0.004)
        if c_ase < best_c - 1e-13:
            best_h = h_ase.copy()
            best_c = c_ase
        h_ase2, c_ase2 = active_set_descent(best_h, max_iters=80, n_active=15, alpha_init=0.004)
        if c_ase2 < best_c - 1e-13:
            best_h = h_ase2.copy()
            best_c = c_ase2

    # Stage 4: Pair moves
    if time_left() > 120:
        h_pm, c_pm = pair_moves(best_h, max_iters=60)
        if c_pm < best_c - 1e-13:
            best_h = h_pm.copy()
            best_c = c_pm

    # Stage 5: Resolution exploration
    if time_left() > 180:
        n_cur = len(best_h)
        dims = set()
        for factor in [0.65, 0.75, 0.85, 0.92, 1.0, 1.08, 1.15, 1.25, 1.4]:
            d = int(round(n_cur * factor / 2)) * 2
            d = max(24, min(d, 130))
            dims.add(d)
        dims.discard(n_cur)
        dims_list = sorted(dims, key=lambda x: abs(x - n_cur))
        for new_n in dims_list[:7]:
            if time_left() < 50:
                break
            h_interp = interpolate_to_n(best_h, new_n)
            h_lp, c_lp = lp_descent(h_interp, max_iters=50, D_list=[0.3, 0.5], margin=0.04)
            if c_lp < best_c - 1e-13:
                best_h = h_lp.copy()
                best_c = c_lp
            h_lse, c_lse = lse_descent(best_h, beta=100, max_iters=50)
            if c_lse < best_c - 1e-13:
                best_h = h_lse.copy()
                best_c = c_lse
            h_ase, c_ase = active_set_descent(best_h, max_iters=50, n_active=10)
            if c_ase < best_c - 1e-13:
                best_h = h_ase.copy()
                best_c = c_ase

    # Stage 6: SLSQP
    if time_left() > 60:
        h_sol, c_sol = slsqp_solve(best_h, maxiter=150)
        if h_sol is not None and c_sol < best_c - 1e-12:
            best_h = h_sol.copy()
            best_c = c_sol
            h_lp, c_lp = lp_descent(best_h, max_iters=40, D_list=[0.2, 0.3], margin=0.02)
            if c_lp < best_c - 1e-13:
                best_h = h_lp.copy()
                best_c = c_lp

    # Stage 7: Multi-start perturbations
    if time_left() > 100:
        n_b = len(best_h)
        starts = [best_h.copy()]
        for eps in [0.003, 0.007, 0.014, 0.028]:
            z = rng.randn(n_b)
            z -= z.mean()
            starts.append(repair_feasible(best_h + eps * z))
        z_sym = symmetrize(rng.randn(n_b))
        z_sym -= z_sym.mean()
        starts.append(repair_feasible(best_h + 0.01 * z_sym))
        h_as = antisymmetrize(0.5 + 0.3 * rng.randn(n_b))
        starts.append(repair_feasible(best_h + 0.008 * (h_as - best_h.mean())))
        for h0 in starts[:6]:
            if time_left() < 25:
                break
            h_lp, c_lp = lp_descent(h0, max_iters=40, D_list=[0.3, 0.4], margin=0.04)
            h_lse, c_lse = lse_descent(h_lp, beta=100, max_iters=40)
            h_ase, c_ase = active_set_descent(h_lse, max_iters=40, n_active=10)
            if c_ase < best_c - 1e-12:
                best_h = h_ase.copy()
                best_c = c_ase

    # Stage 8: Random directions
    if time_left() > 50:
        h_rd, c_rd = random_descent(best_h, rng, n_dirs=6, max_iters=15)
        if c_rd < best_c - 1e-13:
            best_h = h_rd.copy()
            best_c = c_rd

    # Stage 9: LP with varying margins
    for m in [0.015, 0.02, 0.03, 0.06, 0.1]:
        if time_left() < 15:
            break
        h_lp, c_lp = lp_descent(best_h, max_iters=60, D_list=[0.2, 0.35, 0.5], margin=m)
        if c_lp < best_c - 1e-14:
            best_h = h_lp.copy()
            best_c = c_lp

    # Stage 10: Final polish
    if time_left() > 25:
        h_lp, c_lp = lp_descent(best_h, max_iters=80, D_list=[0.1, 0.2, 0.3], margin=0.015)
        if c_lp < best_c - 1e-14:
            best_h = h_lp.copy()
            best_c = c_lp
        h_ase, c_ase = active_set_descent(best_h, max_iters=80, n_active=8, alpha_init=0.002)
        if c_ase < best_c - 1e-14:
            best_h = h_ase.copy()
            best_c = c_ase
        h_lse, c_lse = lse_descent(best_h, beta=150, max_iters=60, alpha_init=0.002)
        if c_lse < best_c - 1e-14:
            best_h = h_lse.copy()
            best_c = c_lse

    # Stage 11: Final pair moves
    if time_left() > 15:
        h_pm, c_pm = pair_moves(best_h, max_iters=40)
        if c_pm < best_c - 1e-14:
            best_h = h_pm.copy()
            best_c = c_pm

    # Stage 12: Final LP
    if time_left() > 15:
        h_lp, c_lp = lp_descent(best_h, max_iters=60, D_list=[0.05, 0.1, 0.15], margin=0.01)
        if c_lp < best_c - 1e-14:
            best_h = h_lp.copy()
            best_c = c_lp

    # Final validation
    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)

    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            final_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 2:
            final_c = float(eval_result[0])
    except Exception:
        pass

    return best_h, final_c, len(best_h)
