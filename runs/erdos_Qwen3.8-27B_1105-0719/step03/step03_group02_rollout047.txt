        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx + offset] += (1.0 - h[idx])
    grad[idx] += (-h[idx + offset])
    return grad


def smooth_max_grad(h, dx, beta):
    n = len(h)
    corrs = compute_corrs(h)
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones_like(corrs) / len(corrs)
    else:
        w /= w_sum
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] > 1e-13:
            grad += w[k] * grad_corr_k(h, k, n)
    return grad * dx


def slsqp_direct(h0, dx, target_sum, beta, maxiter):
    n = len(h0)

    def obj(x):
        corrs = compute_corrs(x)
        return logsumexp(beta * corrs) / beta * dx

    def grad_obj(x):
        return smooth_max_grad(x, dx, beta)

    def c_fun(x):
        return x.sum() - target_sum

    def c_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': c_fun, 'jac': c_jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0, jac=grad_obj, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        h_out = project_feasible(res.x, target_sum)
    except Exception:
        h_out = project_feasible(h0, target_sum)
    return h_out, c5_of_h(h_out, dx)


def exact_line_search(h, d, dx, target_sum, max_alpha=2.0):
    n = len(h)
    pos_mask = d > 1e-15
    neg_mask = d < -1e-15
    alpha_hi = max_alpha
    if pos_mask.any():
        alpha_hi = min(alpha_hi, ((1.0 - h[pos_mask]) / d[pos_mask]).min())
    if neg_mask.any():
        alpha_hi = min(alpha_hi, (h[neg_mask] / (-d[neg_mask])).min())
    alpha_hi = max(0.0, alpha_hi)
    if alpha_hi < 1e-10:
        return h, c5_of_h(h, dx)

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = 0.0, alpha_hi
    c_pt = b - gr * (b - a)
    d_pt = a + gr * (b - a)

    def f_eval(alpha):
        hv = project_feasible(h + alpha * d, target_sum)
        return c5_of_h(hv, dx)

    fc, fd = f_eval(c_pt), f_eval(d_pt)
    for _ in range(50):
        if b - a < 1e-12 * max(1.0, alpha_hi):
            break
        if fc < fd:
            b = d_pt
            d_pt = c_pt
            fd = fc
            c_pt = b - gr * (b - a)
            fc = f_eval(c_pt)
        else:
            a = c_pt
            c_pt = d_pt
            fc = fd
            d_pt = a + gr * (b - a)
            fd = f_eval(d_pt)
    best_alpha = (a + b) / 2.0
    h_opt = project_feasible(h + best_alpha * d, target_sum)
    return h_opt, c5_of_h(h_opt, dx)


def lp_polish_ls(h0, dx, target_sum, max_iters=1000, init_step=0.1):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    step = init_step
    stall = 0

    for it in range(max_iters):
        corrs = compute_corrs(h)
        max_c = corrs.max()
        c5_val = max_c * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 100:
                step *= 0.6
                stall = 0
                if step < 1e-8:
                    break

        tol_frac = 0.03
        active_mask = corrs >= max_c * (1.0 - tol_frac)
        active_k = np.where(active_mask)[0]
        if len(active_k) > 120:
            active_k = np.argsort(corrs)[-120:]
        if len(active_k) < 5:
            active_k = np.argsort(corrs)[-20:]

        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        A_rows = []
        b_rows = []
        for k in active_k:
            row = np.zeros(n + 1)
            row[:n] = dx * grad_corr_k(h, k, n)
            row[n] = -1.0
            A_rows.append(row)
            b_rows.append(-corrs[k] * dx - 1e-13)

        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)

        bounds = []
        for i in range(n):
            bounds.append((max(-h[i], -step), min(1.0 - h[i], step)))
        bounds.append((0.0, None))

        try:
            res = linprog(c_obj, A_ub=np.array(A_rows), b_ub=np.array(b_rows),
                          A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
        except Exception:
            break
        if res.status != 0:
            break

        delta = res.x[:n]
        if np.max(np.abs(delta)) < 1e-14:
            break

        h_new, new_c5 = exact_line_search(h, delta, dx, target_sum)

        if new_c5 < c5_val - 1e-14:
            h = h_new
            step = min(step * 1.05, init_step)
        else:
            h_half, c5_half = exact_line_search(h, 0.5 * delta, dx, target_sum)
            if c5_half < c5_val - 1e-14:
                h = h_half
            else:
                step *= 0.5
                if step < 1e-9:
                    break

    return best_h, best_c5


def lp_polish_wide(h0, dx, target_sum, n_steps=100, lam=0.15):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)

    for s in range(n_steps):
        corrs = compute_corrs(h)
        max_c = corrs.max()
        c5_val = max_c * dx

        n_active = max(20, int(0.5 * (2 * n - 1)))
        top_idx = np.argsort(corrs)[-n_active:]

        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        A_rows = []
        b_rows = []
        for k in top_idx:
            row = np.zeros(n + 1)
            row[:n] = dx * grad_corr_k(h, k, n)
            row[n] = -1.0
            A_rows.append(row)
            b_rows.append(-corrs[k] * dx - 1e-13)

        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)

        bounds = []
        for i in range(n):
            bounds.append((max(-h[i], -lam), min(1.0 - h[i], lam)))
        bounds.append((0.0, None))

        try:
            res = linprog(c_obj, A_ub=np.array(A_rows), b_ub=np.array(b_rows),
                          A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
        except Exception:
            break
        if res.status != 0:
            break

        delta = res.x[:n]
        if np.max(np.abs(delta)) < 1e-14:
            break

        h_new, new_c5 = exact_line_search(h, delta, dx, target_sum)

        if new_c5 < c5_val - 1e-14:
            h = h_new
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_new.copy()
        else:
            lam *= 0.7
            if lam < 1e-8:
                break

    return best_h, best_c5


def pair_transfer_polish(h0, dx, target_sum, n_top=25, n_iter=5):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)

    for iteration in range(n_iter):
        corrs = compute_corrs(h)
        grad = smooth_max_grad(h, dx, 800.0)

        dec_idx = np.argsort(grad)[-n_top:]
        inc_idx = np.argsort(grad)[:n_top]

        improved = False
        for i in dec_idx:
            for j in inc_idx:
                if i == j:
                    continue
                d = np.zeros(2 * n - 1)
                for k in range(2 * n - 1):
                    gk = grad_corr_k(h, k, n)
                    d[k] = gk[j] - gk[i]

                gr_phi = (np.sqrt(5.0) - 1.0) / 2.0
                eps_lo = 0.0
                eps_hi = min(h[i], 1.0 - h[j])
                if eps_hi - eps_lo < 1e-12:
                    continue

                a, b = eps_lo, eps_hi
                c_pt = b - gr_phi * (b - a)
                d_pt = a + gr_phi * (b - a)

                def f_ps(eps):
                    return (corrs + eps * d).max()

                fc, fd = f_ps(c_pt), f_ps(d_pt)
                for _ in range(40):
                    if b - a < 1e-14:
                        break
                    if fc < fd:
                        b = d_pt
                        d_pt = c_pt
                        fd = fc
                        c_pt = b - gr_phi * (b - a)
                        fc = f_ps(c_pt)
                    else:
                        a = c_pt
                        c_pt = d_pt
                        fc = fd
                        d_pt = a + gr_phi * (b - a)
                        fd = f_ps(d_pt)

                eps_opt = (a + b) / 2.0
                new_c5_rough = (corrs + eps_opt * d).max() * dx
                if new_c5_rough < best_c5 - 1e-12:
                    new_h = h.copy()
                    new_h[i] -= eps_opt
                    new_h[j] += eps_opt
                    new_h = np.clip(new_h, 0.0, 1.0)
                    new_h = project_feasible(new_h, target_sum)
                    new_c5 = c5_of_h(new_h, dx)
                    if new_c5 < best_c5 - 1e-14:
                        h = new_h
                        best_c5 = new_c5
                        best_h = h.copy()
                        corrs = compute_corrs(h)
                        improved = True

        if not improved:
            break

    return best_h, best_c5


def block_cd(h0, dx, target_sum, block_size=8, n_sweeps=5):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)

    for sweep in range(n_sweps):
        improved_any = False
        for start in range(0, n, block_size):
            end = min(start + block_size, n)
            idx = np.arange(start, end)
            b_len = len(idx)
            h_block = h[idx].copy()
            h_fixed = h.copy()

            def block_obj(x_block):
                h_try = h_fixed.copy()
                h_try[idx] = x_block
                h_try = project_feasible(h_try, target_sum)
                return c5_of_h(h_try, dx)

            try:
                res = minimize(block_obj, h_block, method='Nelder-Mead',
                               options={'maxiter': 500, 'xatol': 1e-10, 'fatol': 1e-12})
                if res.fun < block_obj(h_block) - 1e-12:
                    h_new = h_fixed.copy()
                    h_new[idx] = res.x
                    h_new = project_feasible(h_new, target_sum)
                    new_c5 = c5_of_h(h_new, dx)
                    if new_c5 < best_c5 - 1e-14:
                        h = h_new
                        best_c5 = new_c5
                        best_h = h.copy()
                        improved_any = True
            except Exception:
                pass

        if not improved_any:
            break

    return best_h, best_c5


def momentum_subgrad(h0, dx, target_sum, max_iter=8000, beta_smooth=40.0):
    n = len(h0)
    h = h0.copy()
    momentum = np.zeros(n)
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    no_improve = 0
    lr = 1.0 / np.sqrt(n)
    beta_mom = 0.9

    for it in range(max_iter):
        corrs = compute_corrs(h)
        c5_val = corrs.max() * dx

        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 3000:
            break

        shifted = corrs - corrs.max()
        w = np.exp(beta_smooth * shifted)
        w /= w.sum()

        grad = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] > 1e-12:
                grad += w[k] * grad_corr_k(h, k, n)
        grad *= dx

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        momentum = beta_mom * momentum - lr * grad / (gnorm + 1e-10)
        h_new = project_feasible(h + momentum, target_sum)
        new_c5 = c5_of_h(h_new, dx)

        tries = 0
        while new_c5 > c5_val + 1e-14 and tries < 5:
            lr *= 0.5
            momentum = beta_mom * momentum - lr * grad / (gnorm + 1e-10)
            h_new = project_feasible(h + momentum, target_sum)
            new_c5 = c5_of_h(h_new, dx)
            tries += 1

        h = h_new

    return best_h, best_c5


def full_optimize(h0, dx, target_sum, time_left_fn, label=""):
    n = len(h0)
    best_h = h0.copy()
    best_c5 = c5_of_h(h0, dx)

    for beta in [100, 300, 800, 2000, 5000]:
        if time_left_fn() < 20:
            break
        h_opt, c5_val = slsqp_direct(best_h, dx, target_sum, beta, 1000)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_opt

    if time_left_fn() > 30:
        h_lp, c5_lp = lp_polish_ls(best_h, dx, target_sum, max_iters=800, init_step=0.08)
        if c5_lp < best_c5:
            best_c5 = c5_lp
            best_h = h_lp

    if time_left_fn() > 20:
        h_w, c5_w = lp_polish_wide(best_h, dx, target_sum, n_steps=80, lam=0.12)
        if c5_w < best_c5:
            best_c5 = c5_w
            best_h = h_w

    if time_left_fn() > 20:
        h_pt, c5_pt = pair_transfer_polish(best_h, dx, target_sum, n_top=20, n_iter=4)
        if c5_pt < best_c5:
            best_c5 = c5_pt
            best_h = h_pt

    if time_left_fn() > 25:
        h_cd, c5_cd = block_cd(best_h, dx, target_sum, block_size=6, n_sweeps=3)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd

    if time_left_fn() > 40:
        h_sg, c5_sg = momentum_subgrad(best_h, dx, target_sum, max_iter=6000)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg

    if time_left_fn() > 20:
        h_f, c5_f = slsqp_direct(best_h, dx, target_sum, 4000, 1500)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f

    if time_left_fn() > 15:
        h_lp2, c5_lp2 = lp_polish_ls(best_h, dx, target_sum, max_iters=500, init_step=0.05)
        if c5_lp2 < best_c5:
            best_c5 = c5_lp2
            best_h = h_lp2

    if time_left_fn() > 10:
        h_pt2, c5_pt2 = pair_transfer_polish(best_h, dx, target_sum, n_top=15, n_iter=3)
        if c5_pt2 < best_c5:
            best_c5 = c5_pt2
            best_h = h_pt2

    return best_h, best_c5


def make_periodic_start(n, target_sum, period):
    h = np.zeros(n)
    fill = max(1, int(round(target_sum / n * period)))
    fill = min(fill, period)
    for i in range(n):
        h[i] = 1.0 if (i % period) < fill else 0.0
    return project_feasible(h, target_sum)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s - (time.time() - t0)

    best_h = None
    best_c5 = float('inf')

    # Phase 1: Optimize from the given construction
    h0 = np.array(initial_h_values, dtype=float)
    n = len(h0)
    dx = 2.0 / n
    target_sum = n / 2.0
    h0 = project_feasible(h0, target_sum)

    h_opt, c5_val = full_optimize(h0, dx, target_sum, time_left, "main")
    best_h = h_opt
    best_c5 = c5_val

    # Phase 2: Perturbed restarts
    n_perturb_tries = 12
    for trial in range(n_perturb_tries):
        if time_left() < 30:
            break
        if trial % 4 == 0:
            perturb = 0.04 * (rng.random(n) - 0.5)
        elif trial % 4 == 1:
            perturb = 0.12 * (rng.random(n) - 0.5)
        elif trial % 4 == 2:
            idx = rng.choice(n, size=max(5, n // 12), replace=False)
            perturb = np.zeros(n)
            perturb[idx] = rng.choice([-1, 1], size=len(idx)) * 0.08
        else:
            center = rng.randint(0, n)
            width = rng.randint(3, 10)
            perturb = np.zeros(n)
            for i in range(n):
                dist = min(abs(i - center), n - abs(i - center))
                perturb[i] = 0.1 * np.cos(np.pi * dist / width) if dist < width else 0.0
        h_try = project_feasible(best_h + perturb, target_sum)
        h_opt2, c5_val2 = full_optimize(h_try, dx, target_sum, time_left, "perturb")
        if c5_val2 < best_c5:
            best_c5 = c5_val2
            best_h = h_opt2

    # Phase 3: Try periodic starts
    periods_to_try = [3, 4, 5, 6, 7, 8, 9, 10, 12, 14, 15]
    for period in periods_to_try:
        if time_left() < 35:
            break
        h_per = make_periodic_start(n, target_sum, period)
        h_opt3, c5_val3 = full_optimize(h_per, dx, target_sum, time_left, "period")
        if c5_val3 < best_c5:
            best_c5 = c5_val3
            best_h = h_opt3

    # Phase 4: Try different n values
    n_candidates = []
    for nn in range(60, 250):
        if nn % 2 == 0:
            n_candidates.append(nn)
    n_candidates.sort(key=lambda x: abs(x - len(best_h)))
    n_candidates = n_candidates[:20]

    for n_try in n_candidates:
        if time_left() < 35:
            break
        if n_try == len(best_h):
            continue
        dx_try = 2.0 / n_try
        ts_try = n_try / 2.0
        x_orig = np.linspace(0.0, 2.0, len(best_h), endpoint=False) + 1.0 / len(best_h)
        x_new = np.linspace(0.0, 2.0, n_try, endpoint=False) + 1.0 / n_try
        h_interp = np.interp(x_new, x_orig, best_h)
        h_interp = project_feasible(h_interp, ts_try)
        h_opt4, c5_val4 = full_optimize(h_interp, dx_try, ts_try, time_left, f"n{n_try}")
        if c5_val4 < best_c5:
            best_c5 = c5_val4
            best_h = h_opt4

    # Phase 5: Intensive final polish
    if time_left() > 30 and best_h is not None:
        n_b = len(best_h)
        dx_b = 2.0 / n_b
        ts_b = n_b / 2.0

        h_lp_final, c5_final = lp_polish_ls(best_h, dx_b, ts_b, max_iters=2000, init_step=0.03)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_lp_final

        if time_left() > 20:
            h_pt_final, c5_ptf = pair_transfer_polish(best_h, dx_b, ts_b, n_top=30, n_iter=6)
            if c5_ptf < best_c5:
                best_c5 = c5_ptf
                best_h = h_pt_final

        if time_left() > 15:
            h_lp_ultra, c5_ultra = lp_polish_ls(best_h, dx_b, ts_b, max_iters=1500, init_step=0.01)
            if c5_ultra < best_c5:
                best_c5 = c5_ultra
                best_h = h_lp_ultra

        if time_left() > 10:
            h_pt_ultra, c5_ptu = pair_transfer_polish(best_h, dx_b, ts_b, n_top=25, n_iter=4)
            if c5_ptu < best_c5:
                best_c5 = c5_ptu
                best_h = h_pt_ultra

    # Final validation
    if best_h is None:
        best_h = h0
        best_c5 = c5_of_h(h0, 2.0 / len(h0))
    else:
        n_final = len(best_h)
        best_h = project_feasible(best_h, n_final / 2.0)
        best_c5 = c5_of_h(best_h, 2.0 / n_final)

    return best_h, best_c5, len(best_h)
