        return val

    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

def stochastic_improve(h, rng, n_trials, scale_base=0.03):
    n = len(h)
    best_h = h.copy()
    best_c = exact_score(h)
    for t in range(n_trials):
        scale = scale_base / (1.0 + 0.3 * t / max(1, n_trials))
        noise = rng.normal(0, scale, n)
        h_try = repair_feasible(best_h + noise)
        c_try = exact_score(h_try)
        if c_try < best_c - 1e-12:
            best_h = h_try
            best_c = c_try
    return best_h, best_c
