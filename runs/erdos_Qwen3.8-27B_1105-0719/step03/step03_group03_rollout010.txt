def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)


def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx


def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(-(1.0 - h))
        remaining = diff
        for i in range(n):
            idx = order[i]
            cap = 1.0 - h[idx]
            add = min(remaining, cap)
            if add > 0:
                h[idx] += add
                remaining -= add
            if remaining <= 1e-13:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            if rem > 0:
                h[idx] -= rem
                remaining -= rem
            if remaining <= 1e-13:
                break
    # Final fix
    diff2 = target - h.sum()
    if abs(diff2) > 1e-14:
        if diff2 > 0:
            mask = h < 1.0
            if mask.any():
                h[mask] += diff2 / mask.sum()
        else:
            mask = h > 0.0
            if mask.any():
                h[mask] += diff2 / mask.sum()
    h = np.clip(h, 0.0, 1.0)
    return h


def feasible_alpha(h, v):
    """Return (lo, hi) such that h + a*v is in [0,1] for a in (lo, hi)."""
    lo = 0.0
    hi = 0.0
    for i in range(len(h)):
        vi = v[i]
        if vi > 1e-14:
            bound = -h[i] / vi
            if bound > lo:
                lo = bound
        elif vi < -1e-14:
            bound = (1.0 - h[i]) / (-vi)
            if bound < hi:
                hi = bound
    lo += 1e-10
    hi -= 1e-10
    if lo >= hi:
        return None
    return (lo, hi)


def ray_search(h, v, n_grid=200):
    """Minimize max(profile) along h + alpha*v. Returns (h_new, score) or (None, None)."""
    rng_a = feasible_alpha(h, v)
    if rng_a is None:
        return None, None
    lo, hi = rng_a
    if hi - lo < 1e-12:
        return None, None
    
    n = len(h)
    A = np.correlate(h, 1 - h, mode='full')
    B = -np.correlate(h, v, mode='full') + np.correlate(v, 1 - h, mode='full')
    C = -np.correlate(v, v, mode='full')
    
    # Grid search
    alphas = np.linspace(lo, hi, n_grid)
    # Compute scores: shape (n_grid,)
    scores = np.empty(n_grid)
    for i, a in enumerate(alphas):
        scores[i] = np.max(A + B * a + C * a * a)
    
    best_i = np.argmin(scores)
    
    # Refine with golden section
    if best_i > 0:
        a_left = alphas[best_i - 1]
    else:
        a_left = lo
    if best_i < n_grid - 1:
        a_right = alphas[best_i + 1]
    else:
        a_right = hi
    
    invphi = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = a_left, a_right
    for _ in range(40):
        c = b - invphi * (b - a)
        d = a + invphi * (b - a)
        fc = np.max(A + B * c + C * c * c)
        fd = np.max(A + B * d + C * d * d)
        if fc <= fd:

        else:
            a = c

def build_gradient(h, n):
    """Build the gradient matrix G where G[k,i] = d profile[k] / d h[i]."""
    dx = 2.0 / n
    M = 2 * n - 1
    center = n - 1
    G = np.zeros((M, n))
    G[center, :] = dx * (1.0 - 2.0 * h)
    for d in range(1, n):
        idx = np.arange(n - d)
        G[center + d, idx] += -dx * h[idx + d]
        G[center + d, idx + d] += dx * (1.0 - h[idx])
        G[center - d, idx] += dx * (1.0 - h[idx + d])
        G[center - d, idx + d] += -dx * h[idx]
    return G


def grad_direction(h, n, profile):
    """Compute a direction that decreases the weighted max of the profile."""
    G = build_gradient(h, n)
    # Weight active lags
    p = profile / (2.0 / n)  # raw profile (undo dx scaling)
    p_max = p.max()
    spread = p_max - p.mean()
    if spread < 1e-10:
        spread = 0.01
    # Temperature: control how "focused" on the max
    T = 0.05 * spread
    w = np.exp(-(p_max - p) / T)
    w = w / (w.sum() + 1e-15)
    grad = G.T @ w
    grad -= grad.mean()
    gn = np.linalg.norm(grad)
    if gn < 1e-15:
        return None
    return -grad / gn

    vals = np.interp(xt, xs, h)
