        elif best_alpha < 0.25:
            delta = max(0.7 * delta, 1e-4)
    return h, best_c

def interp_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    edges_old = np.linspace(0.0, 2.0, n + 1)
    edges_new = np.linspace(0.0, 2.0, new_n + 1)
    h_new = np.zeros(new_n)
    for k in range(new_n):
        a, b = edges_new[k], edges_new[k + 1]
        total = 0.0
        for j in range(n):
            lo = max(a, edges_old[j])
            hi = min(b, edges_old[j + 1])
            if hi > lo:
                total += h[j] * (hi - lo)
        h_new[k] = total / (b - a)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

def make_seed(n, seed_type, rng):
    half = n // 2
    h = np.zeros(n)
    if seed_type == 0:
        idx = np.arange(n)
        vals = (idx * 0.618033988749895) % 1.0
        top = np.argsort(vals)[-half:]
        h[top] = 1.0
    elif seed_type == 1:
        idx = np.arange(n)
        vals = (idx * 0.4142135623730951) % 1.0
        top = np.argsort(vals)[-half:]
        h[top] = 1.0
    elif seed_type == 2:
        period = max(3, n // 7)
        for i in range(n):
            if (i // period) % 2 == 0:
                h[i] = 1.0
        ones = int(h.sum())
        if ones > half:
            excess = ones - half
            one_idx = np.where(h == 1.0)[0]
            to_flip = rng.choice(one_idx, excess, replace=False)
            h[to_flip] = 0.0
        elif ones < half:
            deficit = half - ones
            zero_idx = np.where(h == 0.0)[0]
            to_flip = rng.choice(zero_idx, deficit, replace=False)
            h[to_flip] = 1.0
    elif seed_type == 3:
        h = (np.sin(np.linspace(0, 4 * np.pi, n)) > 0).astype(float)
        ones = int(h.sum())
        if ones > half:
            excess = ones - half
            one_idx = np.where(h == 1.0)[0]
            to_flip = rng.choice(one_idx, excess, replace=False)
            h[to_flip] = 0.0
        elif ones < half:
            deficit = half - ones
            zero_idx = np.where(h == 0.0)[0]
            to_flip = rng.choice(zero_idx, deficit, replace=False)
            h[to_flip] = 1.0
    elif seed_type == 4:
        x = np.linspace(0, 1, n)
        h = (0.5 + 0.3 * np.sin(2.0 * np.pi * x) + 0.2 * np.cos(4.0 * np.pi * x)).clip(0, 1)
        h = repair_feasible(h)
    elif seed_type == 5:
        x = np.linspace(0, 1, n)
        h = (0.5 + 0.4 * np.sin(2.0 * np.pi * 1.5 * x)).clip(0, 1)
        h = repair_feasible(h)
    else:
        perm = rng.permutation(n)
        h[perm[:half]] = 1.0
    return repair_feasible(h)

def crossover(ha, hb, t):
    return repair_feasible((1.0 - t) * ha + t * hb)

def fine_polish(h, t_start, time_budget):
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    delta = 0.01
    for frac in [1.0, 0.8, 0.6, 0.4, 0.3, 0.2, 0.1, 0.05, 0.03, 0.02, 0.01, 0.005]:
        if t_start is not None and time.perf_counter() - t_start > time_budget - 5:
            break
        P = compute_profile(h)
        c = float(P.max())
        G = build_grad_matrix(h, n)
        M = 2 * n - 1
        # Use only top active constraints for speed
        active = np.argsort(-P)[:min(40, M)]
        A_ub = np.zeros((len(active), n + 1))
        A_ub[:, :n] = G[active, :]
        A_ub[:, -1] = -1.0
        b_ub = -P[active]
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        lo = np.maximum(-delta, -h)
        hi = np.minimum(delta, 1.0 - h)
        bounds = list(zip(lo, hi))
        bounds.append((0.0, c + 1.0))
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.zeros(1),
                         bounds=bounds, method='highs')
        except Exception:
            break
        if not res.success:
            delta *= 0.5
            continue
        d = res.x[:n]
        if np.linalg.norm(d) < 1e-13:
            delta *= 0.5
            continue
        h_try = repair_feasible(h + d)
        c_try = exact_score(h_try)
        if c_try < best_c - 1e-14:
            h = h_try
            best_c = c_try
            delta = min(delta * 1.2, 0.05)
        else:
            delta *= 0.6
            if delta < 1e-6:
                break
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()
    rng = np.random.RandomState(seed)

    def elapsed():
        return time.perf_counter() - t_start

    def time_left():
        return budget_s - elapsed()

    try:
        parent = np.array(initial_h_values, dtype=np.float64)
    except Exception:
        parent = None
    if parent is None or len(parent) < 4:
        parent = np.full(64, 0.5)
    parent = np.clip(parent, 0.0, 1.0)
    parent = repair_feasible(parent)

    best_h = parent.copy()
    best_c = exact_score(parent)
    n_best = len(best_h)

    # Stage 1: LP trust descent on parent
    if time_left() > 60:
        h_ad, c_ad = lp_trust_descent(best_h, delta=0.04, max_iters=500, t_start=t_start, time_budget=budget_s - 60)
        if c_ad < best_c:
            best_h, best_c = h_ad, c_ad
            n_best = len(best_h)

    # Stage 2: Fine polish
    if time_left() > 40:
        h_fp, c_fp = fine_polish(best_h, t_start, budget_s - 40)
        if c_fp < best_c:
            best_h, best_c = h_fp, c_fp

    # Stage 3: Dimension exploration
    candidate_pool = [(best_c, best_h.copy(), n_best)]
    dims_to_try = []
    for dn in range(-12, 17, 2):
        d = n_best + dn
        if 36 <= d <= 140:
            dims_to_try.append(d)
    dims_to_try.sort(key=lambda x: abs(x - n_best))

    for new_n in dims_to_try[:10]:
        if time_left() < 30:
            break
        if new_n == n_best:
            continue
        h_interp = interp_to_n(best_h, new_n)
        h_lp, c_lp = lp_trust_descent(h_interp, delta=0.035, max_iters=300, t_start=t_start, time_budget=budget_s - 30)
        if c_lp < best_c:
            best_h, best_c = h_lp, c_lp
            n_best = len(best_h)
        candidate_pool.append((c_lp, h_lp.copy(), new_n))

    # Stage 4: Diverse seeds
    n_explore = n_best
    for st in range(6):
        if time_left() < 25:
            break
        h_seed = make_seed(n_explore, st, rng)
        # Blur to smooth binary patterns
        if st <= 3:
            h_blur = np.zeros(n_explore)
            for i in range(n_explore):
                lo = max(0, i - 1)
                hi = min(n_explore - 1, i + 1)
                h_blur[i] = h_seed[lo:hi + 1].mean()
            h_seed = repair_feasible(h_blur)
        h_lp, c_lp = lp_trust_descent(h_seed, delta=0.04, max_iters=250, t_start=t_start, time_budget=budget_s - 25)
        if c_lp < best_c:
            best_h, best_c = h_lp, c_lp
            n_best = len(best_h)
        candidate_pool.append((c_lp, h_lp.copy(), n_explore))

    # Stage 5: Crossover of top candidates
    candidate_pool.sort(key=lambda x: x[0])
    top_k = min(5, len(candidate_pool))
    crossover_done = 0
    for i in range(top_k):
        for j in range(i + 1, top_k):
            if time_left() < 20:
                break
            if len(candidate_pool[i][1]) != len(candidate_pool[j][1]):
                continue
            for t_mix in [0.3, 0.5, 0.7]:
                if time_left() < 15:
                    break
                h_mix = crossover(candidate_pool[i][1], candidate_pool[j][1], t_mix)
                h_lp, c_lp = lp_trust_descent(h_mix, delta=0.03, max_iters=200, t_start=t_start, time_budget=budget_s - 15)
                crossover_done += 1
                if c_lp < best_c:
                    best_h, best_c = h_lp, c_lp
                    n_best = len(best_h)
                candidate_pool.append((c_lp, h_lp.copy(), len(h_lp)))

    # Stage 6: Final fine polish with tighter settings
    if time_left() > 20:
        h_fp, c_fp = fine_polish(best_h, t_start, budget_s - 10)
        if c_fp < best_c:
            best_h, best_c = h_lp if False else h_fp, c_fp

    # Stage 7: One more LP pass with small delta
    if time_left() > 15:
        h_final, c_final = lp_trust_descent(best_h, delta=0.01, max_iters=200, t_start=t_start, time_budget=budget_s - 5)
        if c_final < best_c:
            best_h, best_c = h_final, c_final

    # Final repair and scoring
    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)

    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            final_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 2:
            final_c = float(eval_result[0])
    except Exception:
        pass

    return best_h, final_c, len(best_h)
