    shifted = (corrs - rmax) / T
    shifted = np.clip(shifted, -100, 0)
    w = np.exp(shifted)
    wsum = w.sum()
    if wsum < 1e-30:

import numpy as np

def project_feasible(x, target_sum):
    n = len(x)
    x = np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0)
    x = np.clip(x, -5.0, 5.0)
    lo, hi = -5.0, 5.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.clip(x - mid, 0.0, 1.0).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        cnt = mask.sum()
        if cnt > 0:
            h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corrs = np.correlate(h, 1.0 - h, mode='full')
    return corrs.max() * dx, corrs


def lse_grad(h, corrs, T):
    n = len(h)
    rmax = corrs.max()
    s = np.clip((corrs - rmax) / T, -100.0, 0.0)
    w = np.exp(s)
    ws = w.sum()
    if ws < 1e-30:
        w = np.full(len(corrs), 1.0 / len(corrs))
    else:
        w /= ws
    grad = np.zeros(n)
    for p in range(2 * n - 1):
        k = p - (n - 1)
        wp = w[p]
        if wp < 1e-14:
            continue
        if k == 0:
            grad += wp * (1.0 - 2.0 * h)
        elif k > 0:
            grad[k:] += wp * (1.0 - h[:n - k])
            grad[:n - k] += -wp * h[k:]
        else:
            kk = -k
            grad[:n - kk] += wp * (1.0 - h[kk:])
            grad[kk:] += -wp * h[:n - kk]
    return grad


def do_line_search(h, v, target_sum, best_c5, n_evals=15):

    pos = v > 1e-12
    neg = v < -1e-12

    if neg.any():
        amax = min(amax, (h[neg] / (-v[neg])).min() * 0.99)
    if amax < 1e-10:
        return h, best_c5, 0.0
    als = np.unique(np.concatenate([
        np.linspace(0.01, min(amax, 0.4), 9),
        np.linspace(0.4, amax, 6)
    ]))
    als = als[(als > 0) & (als <= amax)]
    best_a, best_v = 0.0, best_c5
    for a in als:
        ht = project_feasible(h + a * v, target_sum)
        cv, _ = compute_c5(ht)
        if cv < best_v:
            best_v = cv
            best_a = a
    if best_a > 0:
        return project_feasible(h + best_a * v, target_sum), best_v, best_a
    return h, best_c5, 0.0


def lse_optimize(h, target_sum, t0, deadline, T0=0.4, Tend=0.003, decay=0.996):
    n = len(h)
    best_h = h.copy()
    best_c5, best_corrs = compute_c5(h)
    T = T0
    stagnant = 0
    improved_count = 0

    while time.time() < deadline:
        grad = lse_grad(h, best_corrs, T)
        grad -= grad.mean()
        gm = np.abs(grad).max()
        if gm < 1e-12:
            T *= decay
            stagnant += 1

            continue

    return best_h, best_c5
