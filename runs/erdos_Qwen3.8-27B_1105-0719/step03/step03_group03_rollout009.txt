            j_range2 = np.arange(dp, n)
            G[m, j_range2] += -dx * h[j_range2 - dp]

def lse_gradient_step(h, profile, G, eta, step_sizes):
    """One step of log-sum-exp gradient descent."""
    n = len(h)
    c_max = profile.max()
    z = (profile - c_max) / eta
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    w = exp_z / exp_z.sum()
    grad = G.T @ w
    # Remove mean component (since sum(h) is constrained)
    grad -= grad.mean()
    gn = np.linalg.norm(grad)
    if gn < 1e-15:
        return None, None
    curr_c = c_max
    for s in step_sizes:
        h_try = repair_feasible(h - s * grad / gn)
        sc = exact_score(h_try)
        if sc < curr_c - 1e-14:
            return h_try, sc
    return None, None

def trust_qp(h, profile, active, G, trust, mu):
    """Solve a trust-region QP subproblem using CVXPY."""
    if not HAS_CVXPY:
        return None
    n = len(h)
    M = len(profile)
    G_a = G[active, :].copy()
    P_a = profile[active].copy()
    
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    
    cons = [
        h_var >= 0,
        h_var <= 1,
        cp.sum(h_var) == n / 2.0,
        h_var >= h - trust,
        h_var <= h + trust,
    ]
    # Linearized constraints: t >= P_a + G_a @ (h_var - h)
    # i.e., G_a @ h_var - t <= P_a - G_a @ h
    rhs = P_a - G_a @ h
    cons.append(G_a @ h_var - t_var <= rhs)
    
    obj = cp.Minimize(t_var + mu * cp.sum(cp.square(h_var - h)))
    prob = cp.Problem(obj, cons)
    
    for solver in [cp.ECOS, cp.SCS, cp.CLARABEL]:
        try:
            prob.solve(solver=solver)
            break
        except Exception:
            continue
    
    if h_var.value is None:
        return None
    h_sol = np.array(h_var.value).flatten()
    if not np.isfinite(h_sol).all():
        return None
    return h_sol

def interp_resample(h, new_n):
    """Interpolate h to new resolution."""
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    fn = np.interp(xt, xs, h)
    return repair_feasible(np.clip(fn, 0, 1))

def slsqp_polish(h0, maxiter=300, tol=1e-12):

    M = 2 * n - 1

        return x[-1]

        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g

        return x[:-1].sum() - n / 2.0

        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g

        return x[-1] - np.correlate(h, 1 - h, mode="full") * dx

        h = x[:-1]

        J = np.zeros((M, n + 1))
        J[:, :-1] = -Gh
        J[:, -1] = 1.0
        return J

    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, t0 + 1.0)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}

                       bounds=bounds, constraints=cons,

            hs = repair_feasible(res.x[:-1])
            cs = exact_score(hs)
            if np.isfinite(cs):

        pass
    return None, None

def build_gradient(h):

            G[m, j1] += dx * (1.0 - h[j1 - d])

            G[m, j2] += -dx * h[j2 + d]

            dp = -d

            G[m, j1] += dx * (1.0 - h[j1 + dp])
            j2 = np.arange(dp, n)
            G[m, j2] += -dx * h[j2 - dp]
    return G

                alloc = min(rem() * 0.08, 30)
                bh, bc = optimize_loop(hs, alloc, rng)
                if bc < best_c - 1e-14:
                    best_h, best_c = bh, bc

    # Phase 7: Final SLSQP
    if rem() > 50:
        hs, cs = slsqp_polish(best_h, maxiter=600, tol=1e-13)
        if hs is not None and cs < best_c - 1e-14:
            best_h, best_c = hs, cs

    # Phase 8: Final micro LSE steps
    if rem() > 25:
        n_f = len(best_h)
        h_f = best_h.copy()
        c_f = best_c
        G_f = build_gradient(h_f)
        no_improve = 0
        for _ in range(300):
            if rem() < 8:
                break
            prof = compute_profile(h_f)
            cm = prof.max()
            eta = 0.0008
            z_arr = (prof - cm) / eta
            zm = z_arr.max()
            ez = np.exp(z_arr - zm)
            w = ez / ez.sum()
            grad = G_f.T @ w
            grad -= grad.mean()
            gn = np.linalg.norm(grad)
            found = False
            if gn > 1e-15:
                for s in [8e-4, 4e-4, 2e-4, 1e-4, 5e-5, 2e-5]:
                    h_try = repair_feasible(h_f - s * grad / gn)
                    sc = exact_score(h_try)
                    if sc < c_f - 1e-14:
                        h_f = h_try
                        c_f = sc
                        G_f = build_gradient(h_f)
                        no_improve = 0
                        found = True
                        if c_f < best_c - 1e-14:
                            best_h, best_c = h_f.copy(), c_f
                        break
            if not found:
                no_improve += 1
                if no_improve > 15:
                    break

    # Final validation
    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)

    try:
        er = evaluate_erdos_solution(best_h)
        if isinstance(er, (float, int)):
            final_c = min(final_c, float(er))
        elif isinstance(er, tuple) and len(er) >= 2:
            final_c = min(final_c, float(er[0]))
    except Exception:
        pass

    return best_h, final_c, len(best_h)
