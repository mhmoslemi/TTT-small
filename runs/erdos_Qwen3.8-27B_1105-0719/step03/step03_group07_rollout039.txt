import numpy as np
from scipy.optimize import linprog, minimize
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def profile_h(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx])
            h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    # Final correction
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmin(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    return repair_feasible(np.clip(h_new, 0.0, 1.0))

def compute_A(h, i, j):
    """Compute A_m for all offsets m for pair (i,j).
    A_m = (1-h[i-m])*I(0<=i-m<n) - h[i+m]*I(0<=i+m<n) 
          + h[j+m]*I(0<=j+m<n) - (1-h[j-m])*I(0<=j-m<n)
    """
    n = len(h)
    dx = 2.0 / n
    # m ranges from -(n-1) to (n-1), indexed by m + (n-1)
    A = np.zeros(2 * n - 1)
    for m_idx in range(2 * n - 1):
        m = m_idx - (n - 1)
        val = 0.0
        # Term from i as second factor: (1-h[i-m]) if 0<=i-m<n
        im = i - m
        if 0 <= im < n:
            val += (1.0 - h[im])
        # Term from i as first factor: -h[i+m] if 0<=i+m<n
        ip = i + m
        if 0 <= ip < n:
            val -= h[ip]
        # Term from j as first factor: +h[j+m] if 0<=j+m<n
        jp = j + m
        if 0 <= jp < n:
            val += h[jp]
        # Term from j as second factor: -(1-h[j-m]) if 0<=j-m<n
        jm = j - m
        if 0 <= jm < n:
            val -= (1.0 - h[jm])
        A[m_idx] = val
    return A * dx

def pair_coordinate_search(h, max_sweeps=5, t_start=None, budget_s=1100):
    """Pair-coordinate descent with exact convex subproblem."""
    n = len(h)
    h = h.copy()
    P = profile_h(h)
    best_c = P.max()
    
    for sweep in range(max_sweeps):
        if t_start is not None and time.perf_counter() - t_start > budget_s - 15:
            break
        improved = False
        # Shuffle pair order for diversity
        indices = np.random.permutation(n)
        for ii in range(n):
            if t_start is not None and time.perf_counter() - t_start > budget_s - 15:
                break
            for jj in range(ii + 1, n):
                i = indices[ii]
                j = indices[jj]
                s = h[i] + h[j]
                # δ range: h[i]+δ ∈ [0,1], h[j]-δ ∈ [0,1]
                dmin = max(-h[i], h[j] - 1.0)
                dmax = min(1.0 - h[i], h[j])
                if dmax - dmin < 1e-12:
                    continue
                # Compute A
                A = compute_A(h, i, j)
                # Minimize max_m (P[m] + δ*A[m]) over δ ∈ [dmin, dmax]
                # Use ternary search (function is convex)
                lo, hi = dmin, dmax
                for _ in range(30):
                    m1 = lo + (hi - lo) / 3.0
                    m2 = hi - (hi - lo) / 3.0
                    f1 = np.max(P + m1 * A)
                    f2 = np.max(P + m2 * A)
                    if f1 < f2:
                        hi = m2
                    else:
                        lo = m1
                delta = (lo + hi) / 2.0
                # Apply
                new_c = np.max(P + delta * A)
                if new_c < best_c - 1e-10:
                    h[i] += delta
                    h[j] -= delta
                    P = P + delta * A
                    best_c = new_c
                    improved = True
        if not improved:
            break
    return repair_feasible(h), exact_score(repair_feasible(h))

def lp_direction(h, P, active_thresh=0.005):
    """Find a feasible descent direction using LP.
    Returns direction v (sum=0) that decreases all active P_m, or None.
    """
    n = len(h)
    dx = 2.0 / n
    c = P.max()
    active = np.where(P >= c - active_thresh)[0]
    if len(active) == 0:
        active = np.array([np.argmax(P)])
    
    # Build constraint matrix G (active x n) where G[m, i] = ∂P[m]/∂h_i / dx
    # (We divide by dx since it's a constant scaling)
    G = np.zeros((len(active), n))
    for idx, m_idx in enumerate(active):
        m = m_idx - (n - 1)
        for i in range(n):
            val = 0.0
            im = i - m
            if 0 <= im < n:
                val += (1.0 - h[im])
            ip = i + m
            if 0 <= ip < n:
                val -= h[ip]

def softmax_gradient_descent(h, max_iters=100, beta=200, t_start=None, budget_s=1100):
    """Softmax-based gradient descent as backup."""
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best_c = exact_score(h)
    vel = np.zeros(n)
    
    for it in range(max_iters):
        if t_start is not None and time.perf_counter() - t_start > budget_s - 15:
            break
        P = profile_h(h)
        Pmax = P.max()
        exp_arg = beta * (P - Pmax)
        weights = np.exp(exp_arg)
        S = weights.sum()
        if S < 1e-300:
            weights = np.zeros(len(P))
            weights[np.argmax(P)] = 1.0
        else:
            weights /= S
        
        # Gradient: weighted sum of individual gradients
        grad = np.zeros(n)
        for m_idx in range(len(P)):
            if weights[m_idx] < 1e-4:
                continue
            m = m_idx - (n - 1)
            for i in range(n):
                val = 0.0
                im = i - m
                if 0 <= im < n:
                    val += (1.0 - h[im])
                ip = i + m
                if 0 <= ip < n:
                    val -= h[ip]
                grad[i] += weights[m_idx] * dx * val
        
        # Sum constraint
        sum_err = h.sum() - n / 2.0
        grad += 2.0 * 50.0 * sum_err / n
        
        # Normalize and step
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad /= gnorm
        vel = 0.4 * vel + grad
        vnorm = np.linalg.norm(vel)
        if vnorm > 0:
            vel /= vnorm
        
        alpha = 0.03
        for a in [alpha, alpha*0.5, alpha*0.25, alpha*0.1]:
            h_try = repair_feasible(h - a * vel)
            s = exact_score(h_try)
            if s < best_c - 1e-12:
                h = h_try
                best_c = s
                alpha = min(alpha * 1.1, 0.08)
                break
            elif s < exact_score(h):
                h = h_try
                best_c = min(best_c, exact_score(h))
            else:
                h_try = h  # revert
        else:
            alpha *= 0.5
            if alpha < 1e-8:
                break
    return h, best_c

def slsqp_active(h0, maxiter=300, active_k=10, tol=1e-12):
    """SLSQP with top-K active constraints."""
    n = len(h0)
    dx = 2.0 / n
    M = 2 * n - 1
    P = profile_h(h0)
    c0 = P.max()
    
    thresh = c0 - max(0.003, 0.05 * (c0 - np.median(P)))
    active = np.where(P >= thresh)[0]
    if len(active) == 0:
        active = np.array([np.argmax(P)])
    if len(active) > active_k:
        top_idx = np.argsort(P[active])[-active_k:]
        active = active[top_idx]
    
    def build_G(active_set):
        G = np.zeros((len(active_set), n))
        for idx, m_idx in enumerate(active_set):
            m = m_idx - (n - 1)
            for i in range(n):
                val = 0.0
                im = i - m
                if 0 <= im < n:
                    val += (1.0 - h0[im])
                ip = i + m
                if 0 <= ip < n:
                    val -= h0[ip]
                G[idx, i] = val * dx
        return G
    
    G = build_G(active)
    
    def obj(x):
        return x[-1]
    
    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    
    def eq_fun(x):
        return x[:-1].sum() - n / 2.0
    
    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g
    
    def ineq_fun(x):
        prof = profile_h(x[:-1])
        return x[-1] - prof[active]
    
    def ineq_jac(x):
        h = x[:-1]
        # Recompute gradient with current h
        Gk = np.zeros((len(active), n))
        for idx, m_idx in enumerate(active):
            m = m_idx - (n - 1)
            for i in range(n):
                val = 0.0
                im = i - m
                if 0 <= im < n:
                    val += (1.0 - h[im])
                ip = i + m
                if 0 <= ip < n:
                    val -= h[ip]
                Gk[idx, i] = val * dx
        J = np.zeros((len(active), n + 1))
        J[:, :-1] = -Gk
        J[:, -1] = 1.0
        return J
    
    x0 = np.concatenate([h0, [c0 + 1e-4]])
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 1.0)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': tol})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = repair_feasible(res.x[:-1])
            c_new = exact_score(h_new)
            if c_new <= c0 + 1e-10:
                return h_new
    except Exception:
        pass
    return None

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def profile_h(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx])
            h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmin(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    return repair_feasible(np.clip(h_new, 0.0, 1.0))

def compute_A(h, i, j):
    n = len(h)
    dx = 2.0 / n
    A = np.zeros(2 * n - 1)
    for m_idx in range(2 * n - 1):
        m = m_idx - (n - 1)
        val = 0.0
        im = i - m
        if 0 <= im < n:
            val += (1.0 - h[im])
        ip = i + m
        if 0 <= ip < n:
            val -= h[ip]
        jp = j + m
        if 0 <= jp < n:
            val += h[jp]
        jm = j - m
        if 0 <= jm < n:
            val -= (1.0 - h[jm])
        A[m_idx] = val * dx
    return A

def build_gradient_matrix(h, active_offsets):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((len(active_offsets), n))
    for idx, m_idx in enumerate(active_offsets):
        m = m_idx - (n - 1)
        for i in range(n):
            val = 0.0
            im = i - m
            if 0 <= im < n:
                val += (1.0 - h[im])
            ip = i + m
            if 0 <= ip < n:
                val -= h[ip]
            G[idx, i] = val * dx
    return G

def lp_direction(h, P, active_thresh=0.004):
    n = len(h)
    c = P.max()
    active = np.where(P >= c - active_thresh)[0]
    if len(active) == 0:
        active = np.array([np.argmax(P)])
    if len(active) > 20:
        top_idx = np.argsort(P[active])[-20:]
        active = active[top_idx]
    G = build_gradient_matrix(h, active)
    A_ub = np.hstack([G, np.eye(len(active))])
    b_ub = np.zeros(len(active))
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(-1.0, 1.0)] * n + [(0.0, c + 1.0)]
    c_obj = np.zeros(n + 1)
    c_obj[-1] = -1.0
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x is not None:
            eta = res.x[-1]
            if eta > 1e-6:
                v = res.x[:n]
                nv = np.linalg.norm(v)
                if nv > 1e-12:
                    v /= nv
                    return v, eta
    except Exception:
        pass
    return None, 0.0

def softmax_gradient_descent(h, max_iters=80, beta=300, t_start=None, budget_s=1100):

                    sol, c_sol = slsqp_active(best_h, maxiter=250, active_k=12, tol=1e-12)
                    if sol is not None and c_sol < best_c - 1e-12:
                        best_h = sol
                        best_c = c_sol
                    if time_ok(10):
                        best_h, best_c = pair_coordinate_search(
                            best_h, max_sweeps=5,
                            t_start=t_start, budget_s=budget_s, rng=rng)

    # Phase 6: Structured restarts
    if time_ok(60):
        for s in range(3):
            if not time_ok(20):
                break
            n_try = len(best_h)
            h_struct = structured_start(n_try, seed + 200 + s)
            # Blend with current best
            for mix in [0.3, 0.5]:
                if not time_ok(10):
                    break
                h_blend = repair_feasible(mix * h_struct + (1 - mix) * best_h)
                h_blend, c_blend = pair_coordinate_search(h_blend, max_sweeps=5,
                                                           t_start=t_start, budget_s=budget_s, rng=rng)
                if c_blend < best_c - 1e-10:
                    sol, c_sol = slsqp_active(h_blend, maxiter=200, active_k=12, tol=1e-12)
                    if sol is not None and c_sol < c_blend - 1e-12:
                        h_blend = sol
                        c_blend = c_sol
                    best_h = h_blend
                    best_c = c_blend

    # Phase 7: Extended pair-coordinate with multiple LP cycles
    if time_ok(60):
        for cycle in range(3):
            if not time_ok(15):
                break
            h_lp, c_lp = active_set_descent(best_h, max_iters=30,
                                            t_start=t_start, budget_s=budget_s)
            if c_lp < best_c - 1e-12:
                best_h = h_lp
                best_c = c_lp
            if time_ok(10):
                best_h, best_c = pair_coordinate_search(best_h, max_sweeps=8,
                                                         t_start=t_start, budget_s=budget_s, rng=rng)

    # Phase 8: Final SLSQP polish
    if time_ok(25):
        sol, c_sol = slsqp_active(best_h, maxiter=500, active_k=15, tol=1e-13)
        if sol is not None and c_sol < best_c - 1e-14:
            best_h = sol
            best_c = c_sol

    # Phase 9: Final pair-coordinate micro-polish
    if time_ok(15):
        best_h, best_c = pair_coordinate_search(best_h, max_sweeps=20,
                                                 t_start=t_start, budget_s=budget_s, rng=rng)

    # Final validation
    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)

    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            final_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 1:
            if np.isfinite(eval_result[0]):
                final_c = float(eval_result[0])
    except Exception:
        pass

    return best_h, final_c, len(best_h)
