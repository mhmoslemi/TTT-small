import numpy as np

def score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def project(h, n):
    h = np.clip(np.asarray(h, dtype=np.float64), 0.0, 1.0).copy()
    target = 0.5 * n
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            cap = 1.0 - h
            s = cap.sum()
            if s > 1e-15:
                h += diff * cap / s
            h = np.clip(h, 0.0, 1.0)
            diff = target - h.sum()
            if abs(diff) > 1e-13:
                idx = np.argsort(1.0 - h)
                for i in idx[::-1]:
                    a = min(diff, 1.0 - h[i])
                    h[i] += a
                    diff -= a
                    if diff < 1e-13:
                        break
        else:
            s = h.sum()
            if s > 1e-15:
                h += diff * h / s
            h = np.clip(h, 0.0, 1.0)
            diff = target - h.sum()
            if abs(diff) > 1e-13:
                idx = np.argsort(h)
                for i in idx:
                    a = min(-diff, h[i])
                    h[i] -= a
                    diff += a
                    if diff > -1e-13:
                        break
    return h

def softmax_weights(p, temp):
    m = p.max()
    e = np.exp((p - m) / temp)
    return e / e.sum()

def smooth_grad(h, n, temp):
    dx = 2.0 / n
    p = np.correlate(h, 1 - h, mode="full") * dx
    w = softmax_weights(p, temp)
    c = n - 1
    g = np.zeros(n)
    g += w[c] * (1.0 - 2.0 * h)
    for d in range(1, n):
        g[d:] += w[c + d] * (1.0 - h[:n - d])
        g[:n - d] -= w[c + d] * h[d:]
        g[:n - d] += w[c - d] * (1.0 - h[d:])
        g[d:] -= w[c - d] * h[:n - d]
    return g

def binary_sa(h0, n, rng, budget_fn, max_iter=100000, temp_start=0.04):

def random_feasible_dir(n, rng):
    d = rng.randn(n)
    d -= d.mean()
    return d

def random_directions_opt(h0, n, budget_fn, rng, n_dirs=50):

def interp_to(h, nn):
    n = len(h)
    if n == nn:
        return h.copy()
    xs = (np.arange(n) + 0.5) * 2.0 / n
    xt = (np.arange(nn) + 0.5) * 2.0 / nn
    return project(np.interp(xt, np.concatenate([[0.0], xs, [2.0]]),
                             np.concatenate([[h[0]], h, [h[-1]]])), nn)

def slsqp_refine(h0, n, budget_fn, maxiter=250):
    dx = 2.0 / n
    M = 2 * n - 1

    def mis_func(x):
        h = x[:n]
        t = x[-1]
        p = np.correlate(h, 1 - h, mode="full") * dx
        return t - p

    def mis_func_sum(x):
        return np.sum(np.maximum(0, np.correlate(x[:n], 1 - x[:n], mode="full") * dx - x[-1])) + x[-1]

    x0 = np.concatenate([h0, [score(h0) + 0.005]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 0.8)]
    cons = [{'type': 'eq', 'fun': lambda x: x[:n].sum() - n / 2.0,
             'jac': lambda x: np.concatenate([np.ones(n), [0.0]])}]
    try:
        res = minimize(lambda x: x[-1] + 1e6 * np.sum(np.maximum(0, mis_func(x))),
                       x0, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.x is not None and np.isfinite(res.x).all():
            h = project(res.x[:n], n)
            c = score(h)
            if np.isfinite(c) and c < 1.0:
                return h
    except Exception:
        pass
    return None
