def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full"))) * dx

def profile_h(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx])
            h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def build_gradient_all(h, d):
    """Gradient of P_d with respect to h, scaled by dx. d is the offset (m = k - (n-1))."""
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    if d == 0:
        g[:] = dx * (1.0 - 2.0 * h)
    elif d > 0:
        for i in range(n - d):
            g[i + d] += dx * (1.0 - h[i])
            g[i] -= dx * h[i + d]
    else:
        p = -d
        for i in range(n - p):
            g[i] += dx * (1.0 - h[i + p])
            g[i + p] -= dx * h[i]
    return g

def lp_direction(h, active_offsets, max_eta=10.0):
    """Find LP minimax direction. Returns (direction, eta) or (None, 0)."""
    n = len(h)
    # Build constraint matrix
    # Variables: d_0,...,d_{n-1}, eta (total n+1)
    # Minimize -eta
    # subject to: G_j @ d - eta <= 0 for each active j
    #              sum(d) = 0
    #              -1 <= d_i <= 1, -max_eta <= eta <= max_eta
    
    m = len(active_offsets)
    if m == 0:
        return None, 0.0
    
    A_ub = np.zeros((m, n + 1))
    b_ub = np.zeros(m)
    
    for idx, off in enumerate(active_offsets):
        g = build_gradient_all(h, off)
        A_ub[idx, :n] = g
        A_ub[idx, n] = -1.0  # G_j @ d - eta <= 0
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    c = np.zeros(n + 1)
    c[n] = -1.0  # minimize -eta
    
    bounds = [(-1.0, 1.0)] * n + [(-max_eta, max_eta)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x is not None:
            d = res.x[:n]
            eta = res.x[n]
            if eta > 1e-8:
                # Normalize: scale so max |d_i| = 1
                dmax = np.max(np.abs(d))
                if dmax > 1e-12:
                    d = d / dmax
                return d, eta / dmax if dmax > 0 else 0.0
    except Exception:
        pass
    return None, 0.0

def binary_sa(n, rng, t_start, budget_s, margin=30, n_iters=200000):

    h = np.zeros(n)

    h[positions] = 1.0
    current_score = exact_score(h)
    best_h = h.copy()
    best_score = current_score
    
    T = 0.05  # initial temperature
    cool_rate = 0.9999
    
    one_pos = positions.copy()
    zero_pos = np.setdiff1d(np.arange(n), positions)
    
    for it in range(n_iters):
        if time.perf_counter() - t_start > budget_s - margin:
            break
        
        # Pick random swap
        ia = rng.randint(len(one_pos))
        ib = rng.randint(len(zero_pos))
        a = one_pos[ia]
        b = zero_pos[ib]
        
        # Swap
        h[a] = 0
        h[b] = 1
        
        new_score = exact_score(h)
        delta = new_score - current_score
        
        if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-10)):
            current_score = new_score
            # Update position arrays
            one_pos[ia] = b
            zero_pos[ib] = a
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            # Revert
            h[b] = 0
            h[a] = 1
        
        T *= cool_rate
    
    return best_h, best_score

def pair_coord_descent(h, t_start, budget_s, margin=20, max_sweeps=30):
    """Pair coordinate descent with sum preservation."""
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    
    rng = np.random.RandomState(123)
    
    for sweep in range(max_sweeps):
        if time.perf_counter() - t_start > budget_s - margin:
            break
        improved = False
        # Random pairs
        for _ in range(3 * n):
            if time.perf_counter() - t_start > budget_s - margin:
                break
            i, j = rng.choice(n, 2, replace=False)
            s = h[i] + h[j]
            if s < 1e-12 or s > 2.0 - 1e-12:
                continue
            lo = max(0.0, s - 1.0)
            hi = min(1.0, s)
            if hi - lo < 1e-12:
                continue
            
            # Try several values of t = h[i], h[j] = s - t
            best_t = h[i]
            best_s_val = best_c
            for frac in [0.0, 0.1, 0.25, 0.5, 0.75, 0.9, 1.0]:
                t = lo + frac * (hi - lo)
                if abs(t - h[i]) < 1e-10:
                    continue
                h_new = h.copy()
                h_new[i] = t
                h_new[j] = s - t
                s_val = exact_score(h_new)
                if s_val < best_s_val:
                    best_s_val = s_val
                    best_t = t
            
            if best_t != h[i]:
                h[i] = best_t
                h[j] = s - best_t
                best_c = best_s_val
                improved = True
        
        if not improved:
            break
    
    return h, best_c

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    return repair_feasible(np.clip(h_new, 0.0, 1.0))
