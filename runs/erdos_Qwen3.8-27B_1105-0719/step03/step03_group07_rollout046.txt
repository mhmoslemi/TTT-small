def pairwise_sweep(h, max_sweeps=20, t_start=None, budget_s=1100):
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    
    for sweep in range(max_sweeps):
        if t_start is not None and time.perf_counter() - t_start > budget_s - 30:
            break
        improved = False
        for i in range(n):
            for j in range(i+1, n):
                s = h[i] + h[j]
                t_min = max(0.0, s - 1.0)
                t_max = min(1.0, s)
