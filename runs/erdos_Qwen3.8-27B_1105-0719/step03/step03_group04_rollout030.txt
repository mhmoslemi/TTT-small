    """Score the pair transfer e_q - e_p based on active lags."""
    # active_G has shape (num_active, n), rows are gradients for active lags
    # active_profile has shape (num_active,)

    # Effect on active lag k: active_G[k, q] - active_G[k, p]

    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        idx = int(np.argmax(1.0 - h)) if diff2 > 0 else int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def get_active_set(profile, top_k=15, beam=0.004):
    T = np.max(profile)
    thresh = T - beam
    active = np.where(profile >= thresh)[0]
    if len(active) == 0:
        active = np.array([np.argmax(profile)])
    if len(active) > top_k:
        top_idx = np.argsort(profile[active])[-top_k:]
        active = active[top_idx]
    return active, T

def compute_active_gradient(h, active, dx, n):
    M = len(active)
    G = np.zeros((M, n))
    center = n - 1
    for ai, k in enumerate(active):
        d = k - center
        if d == 0:
            G[ai, :] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            G[ai, d:n] += dx * (1.0 - h[0:n-d])
            G[ai, 0:n-d] += -dx * h[d:n]
        else:
            r = -d
            G[ai, 0:n-r] += dx * (1.0 - h[r:n])
            G[ai, r:n] += -dx * h[0:n-r]
    return G

def predict_pair_step(p_act, v, mu):
    """Find t in [0,mu] minimizing max_k(p_act[k] + t*v[k])."""
    k = len(v)
    if k == 0 or mu < 1e-15:
        return 0.0
    if k == 1:
        if v[0] >= 0:
            return 0.0
        return mu
    candidates = [0.0, mu]
    for i in range(k):
        for j in range(i+1, k):
            dv = v[i] - v[j]
            if abs(dv) > 1e-14:
                bt = (p_act[j] - p_act[i]) / dv
                if eps_ok := (0.0 < bt < mu):
                    candidates.append(bt)
    best_t = 0.0
    best_f = np.max(p_act)
    for t_c in candidates:
        f_c = np.max(p_act + t_c * v)
        if f_c < best_f:
            best_f = f_c
            best_t = t_c
    return best_t

def pair_transfer_optimize(h, t_start, budget_s, max_iters=500, top_pairs=40):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c = exact_score(h)
    no_improve = 0
    
    for it in range(max_iters):
        elapsed = time.perf_counter() - t_start
        if budget_s - elapsed < 8:
            break
        if no_improve > 40:
            break
        
        profile = compute_profile(h)
        c = np.max(profile)
        if c < best_c - 1e-14:
            best_c = c
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        beam = 0.003 if no_improve > 20 else 0.005
        active, T = get_active_set(profile, top_k=12, beam=beam)
        p_act = profile[active]
        G_act = compute_active_gradient(h, active, dx, n)
        
        # Generate candidate pairs
        high_idx = np.where(h > 0.5)[0]
        low_idx = np.where(h < 0.5)[0]
        
        if len(high_idx) == 0 or len(low_idx) == 0:
            break
        
        # Score candidates: predict improvement for each pair
        scores = []
        for p in high_idx:
            for q in low_idx:
                if p == q:
                    continue
                mu = min(h[p], 1.0 - h[q])
                if mu < 1e-12:
                    continue
                v = G_act[:, q] - G_act[:, p]
                t_star = predict_pair_step(p_act, v, mu)
                if t_star < 1e-12:
                    continue
                pred_max = np.max(p_act + t_star * v)
                pred_improve = T - pred_max
                if pred_improve > 1e-10:
                    scores.append((pred_improve, p, q, mu, t_star))
        
        if not scores:
            # Try reversed direction
            for p in low_idx:
                for q in high_idx:
                    if p == q:
                        continue
                    mu = min(1.0 - h[p], h[q])
                    if mu < 1e-12:
                        continue
                    v = G_act[:, q] - G_act[:, p]
                    # Direction: increase h[p], decrease h[q] => e_p - e_q
                    # Effect: -v on active set
                    v_rev = -v
                    t_star = predict_pair_step(p_act, v_rev, mu)
                    if t_star < 1e-12:
                        continue
                    pred_max = np.max(p_act + t_star * v_rev)
                    pred_improve = T - pred_max
                    if pred_improve > 1e-10:
                        scores.append((pred_improve, q, p, mu, t_star))
            scores.sort(key=lambda x: -x[0])
        
        if not scores:
            no_improve += 5
            continue
        
        scores.sort(key=lambda x: -x[0])
        
        # Try top candidates with multiple step sizes
        found_improvement = False
        for pred_imp, p, q, mu, t_star in scores[:top_pairs]:
            if time.perf_counter() - t_start > budget_s - 8:
                break
            # Try multiple step sizes
            step_candidates = [t_star, t_star * 0.7, t_star * 0.4, t_star * 0.2,
                              mu * 0.5, mu * 0.3, mu * 0.1]
            step_candidates = sorted(set(max(0.0, min(s, mu)) for s in step_candidates), reverse=True)
            
            best_step_h = None
            best_step_c = c
            
            for s in step_candidates[:8]:
                if s < 1e-10:
                    continue
                h_try = h.copy()
                h_try[p] -= s
                h_try[q] += s
                h_try = repair_feasible(h_try)
                c_try = exact_score(h_try)
                if c_try < best_step_c - 1e-13:
                    best_step_c = c_try
                    best_step_h = h_try
            
            if best_step_h is not None:
                h = best_step_h
                found_improvement = True
                break
        
        if not found_improvement:
            # Try LP step as fallback
            h_lp = lp_trust_step(h, active, profile, dx, n, radius=0.03)
            if h_lp is not None:
                c_lp = exact_score(h_lp)
                if c_lp < c - 1e-13:
                    h = h_lp
                    found_improvement = True
    
    return best_h, best_c

def lp_trust_step(h, active, profile, dx, n, radius=0.03):
    M = len(active)
    G_act = compute_active_gradient(h, active, dx, n)
    p_act = profile[active]
    
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_ub = np.zeros((M, n + 1))
    A_ub[:, :n] = G_act
    A_ub[:, -1] = -1.0
    b_ub = -p_act
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    lower = np.maximum(-h, -radius)
    upper = np.minimum(1.0 - h, radius)
    bounds = list(zip(lower, upper))
    bounds.append((0.0, None))
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A
_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'maxiter': 1000})
        if res.success and res.x is not None and np.isfinite(res.x).all():
            h_new = repair_feasible(h + res.x[:n])
            return h_new
    except Exception:
        pass
    return None

def kkt_directed_step(h, active, profile, dx, n, step_sizes=None):
    """Use KKT conditions to find a good direction."""
    G_act = compute_active_gradient(h, active, dx, n)
    p_act = profile[active]
    k = len(active)
    
    # Solve for lambda: minimize ||G_act.T @ lambda||_2 subject to lambda >= 0, sum(lambda) = 1
    # This is equivalent to: find the point in the convex hull of {G_act[i,:]} closest to the mean
    # Use a simple approach: try various weight patterns
    
    best_dir = None
    best_score = np.inf
    
    if step_sizes is None:
        step_sizes = [0.001, 0.003, 0.01, 0.03]
    
    # Uniform weights
    weights = np.ones(k) / k
    g = G_act.T @ weights
    g -= g.mean()
    if np.linalg.norm(g) > 1e-14:
        g /= np.linalg.norm(g)
        for a in step_sizes:
            h_try = repair_feasible(h - a * g)
            s = exact_score(h_try)
            if s < best_score:
                best_score = s
                best_dir = h_try
    
    # Weighted by profile values (emphasize highest)
    p_min = p_act.min()
    p_range = p_act.max() - p_min
    if p_range > 1e-12:
        w2 = 0.5 + 2.0 * (p_act - p_min) / p_range
        w2 /= w2.sum()
        g2 = G_act.T @ w2
        g2 -= g2.mean()
        if np.linalg.norm(g2) > 1e-14:
            g2 /= np.linalg.norm(g2)
            for a in step_sizes:
                h_try = repair_feasible(h - a * g2)
                s = exact_score(h_try)
                if s < best_score:
                    best_score = s
                    best_dir = h_try
    
    # Exponential weights
    if p_range > 1e-12:
        w3 = np.exp(3.0 * (p_act - p_min) / p_range)
        w3 /= w3.sum()
        g3 = G_act.T @ w3
        g3 -= g3.mean()
        if np.linalg.norm(g3) > 1e-14:
            g3 /= np.linalg.norm(g3)
            for a in step_sizes:
                h_try = repair_feasible(h - a * g3)
                s = exact_score(h_try)
                if s < best_score:
                    best_score = s
                    best_dir = h_try
    
    return best_dir

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

def fine_polish(h, t_start, budget_s, max_iters=200):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best_c = exact_score(h)
    no_improve = 0
    
    for it in range(max_iters):
        elapsed = time.perf_counter() - t_start
        if budget_s - elapsed < 5:
            break
        if no_improve > 50:
            break
        
        profile = compute_profile(h)
        c = np.max(profile)
        if c < best_c - 1e-14:
            best_c = c
            no_improve = 0
        else:
            no_improve += 1
        
        active, T = get_active_set(profile, top_k=10, beam=0.002)
        
        # KKT-directed step
        h_new = kkt_directed_step(h, active, profile, dx, n,
                                   step_sizes=[0.0005, 0.001, 0.002, 0.005])
        if h_new is not None:
            c_new = exact_score(h_new)
            if c_new < c - 1e-14:
                h = h_new
                continue
        
        # Very small pair transfers
        high_idx = np.where(h > 0.5)[0]
        low_idx = np.where(h < 0.5)[0]
        G_act = compute_active_gradient(h, active, dx, n)
        p_act = profile[active]
        
        found = False
        # Sample a subset of pairs for speed
        n_try = min(50, len(high_idx) * len(low_idx))
        pairs = []
        rng_local = np.random.RandomState(it)
        for _ in range(n_try):
            p = high_idx[rng_local.randint(len(high_idx))]
            q = low_idx[rng_local.randint(len(low_idx))]
            if p != q:
                pairs.append((p, q))
        
        for p, q in pairs[:30]:
            mu = min(h[p], 1.0 - h[q])
            if mu < 1e-10:
                continue
            v = G_act[:, q] - G_act[:, p]
            t_star = predict_pair_step(p_act, v, mu)
            for s in [t_star, t_star * 0.5, mu * 0.1, 0.001, 0.0005]:
                if s <= 0 or s > mu:
                    continue
                h_try = h.copy()
                h_try[p] -= s
                h_try[q] += s
                h_try = repair_feasible(h_try)
                c_try = exact_score(h_try)
                if c_try < best_c - 1e-14:
                    h = h_try
                    best_c = c_try
                    found = True
                    break
            if found:
                break
        
        if not found:
            no_improve += 1
    
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()
    rng = np.random.RandomState(seed)
    
    def time_left():
        return budget_s - (time.perf_counter() - t_start)
    
    # Load parent
    try:
        parent = np.array(initial_h_values, dtype=np.float64)
    except (NameError, Exception):
        parent = np.full(70, 0.5)
    
    if parent is None or len(parent) < 4 or not np.isfinite(parent).all():
        parent = np.full(70, 0.5)
    
    parent = np.clip(parent, 0.0, 1.0)
    parent = repair_feasible(parent)
    
    best_h = parent.copy()
    best_c = exact_score(parent)
    best_n = len(best_h)
    
    # Try symmetrization
    h_sym = repair_feasible((parent + parent[::-1]) / 2.0)
    c_sym = exact_score(h_sym)
    if c_sym < best_c - 1e-12:
        best_h = h_sym
        best_c = c_sym
    
    # Phase 1: Main pair-transfer optimization at parent dimension
    if time_left() > 100:
        h_opt, c_opt = pair_transfer_optimize(best_h, t_start, budget_s,
                                               max_iters=300, top_pairs=30)
        if c_opt < best_c - 1e-14:
            best_h = h_opt
            best_c = c_opt
    
    # Phase 2: KKT-directed polish
    if time_left() > 60:
        h_pol, c_pol = fine_polish(best_h, t_start, budget_s, max_iters=150)
        if c_pol < best_c - 1e-14:
            best_h = h_pol
            best_c = c_pol
    
    # Phase 3: Try nearby dimensions
    n0 = len(best_h)
    dims_to_try = []
    for delta in [-4, -3, -2, -1, 1, 2, 3, 4, 5, 6, 7, 8]:
        d = n0 + delta
        if 40 <= d <= 150:
            dims_to_try.append(d)
    
    for new_n in dims_to_try:
        if time_left() < 80:
            break
        h_interp = interpolate_to_n(best_h, new_n)
        h_opt, c_opt = pair_transfer_optimize(h_interp, t_start, budget_s,
                                               max_iters=150, top_pairs=25)
        if c_opt < best_c - 1e-14:
            best_h = h_opt
            best_c = c_opt
            best_n = new_n
            # Quick polish
            if time_left() > 50:
                hp, cp = fine_pelish(h_opt, t_start, budget_s, max_iters=80)
                if cp < best_c - 1e-14:
                    best_h = hp
                    best_c = cp
    
    # Phase 4: Return to original dimension with refined solution
    if best_n != n0 and time_left() > 80:
        h_back = interpolate_to_n(best_h, n0)
        h_opt, c_opt = pair_transfer_optimize(h_back, t_start, budget_s,
                                               max_iters=100, top_pairs=25
)
        if c_opt < best_c - 1e-14:
            best_h = h_opt
            best_c = c_opt
            best_n = n0
    
    # Phase 5: Final intensive pair transfer at best dimension
    if time_left() > 60:
        h_opt, c_opt = pair_transfer_optimize(best_h, t_start, budget_s,
                                               max_iters=200, top_pairs=20)
        if c_opt < best_c - 1e-14:
            best_h = h_opt
            best_c = c_opt
    
    # Phase 6: Final fine polish
    if time_left() > 30:
        h_pol, c_pol = fine_polish(best_h, t_start, budget_s, max_iters=100)
        if c_pol < best_c - 1e-14:
            best_h = h_pol
            best_c = c_pol
    
    # Phase 7: LP-based final refinement
    if time_left() > 40:
        n_f = len(best_h)
        dx_f = 2.0 / n_f
        profile_f = compute_profile(best_h)
        active_f, _ = get_active_set(profile_f, top_k=10, beam=0.003)
        
        # Multiple small LP steps
        for _ in range(10):
            if time_left() < 5:
                break
            h_lp = lp_trust_step(best_h, active_f, profile_f, dx_f, n_f, radius=0.01)
            if h_lp is not None:
                c_lp = exact_score(h_lp)
                if c_lp < best_c - 1e-14:
                    best_h = h_lp
                    best_c = c_lp
                    profile_f = compute_profile(best_h)
                    active_f, _ = get_active_set(profile_f, top_k=10, beam=0.003)
                else:
                    break
    
    # Phase 8: SLSQP epigraph attempt
    if time_left() > 45:
        n_s = len(best_h)
        dx_s = 2.0 / n_s
        M_s = 2 * n_s - 1
        c0_s = exact_score(best_h)
        
        def sl_eq(x):
            return x[:-1].sum() - n_s / 2.0
        
        def sl_eq_jac(x):
            g = np.zeros(n_s + 1)
            g[:-1] = 1.0
            return g
        
        def sl_ineq(x):
            h_v = x[:-1]
            t_v = x[-1]
            prof = np.correlate(h_v, 1.0 - h_v, mode="full") * dx_s
            return t_v - prof
        
        def sl_ineq_jac(x):
            h_v = x[:-1]
            n_v = len(h_v)
            G = compute_active_gradient(h_v, np.arange(2*n_v-1), dx_s, n_v)
            J = np.zeros((M_s, n_s + 1))
            J[:, :-1] = -G
            J[:, -1] = 1.0
            return J
        
        x0 = np.concatenate([best_h, [c0_s + 1e-5]])
        bounds_s = [(0.0, 1.0)] * n_s + [(0.0, None)]
        cons_s = [
            {'type': 'eq', 'fun': sl_eq, 'jac': sl_eq_jac},
            {'type': 'ineq', 'fun': sl_ineq, 'jac': sl_ineq_jac}
        ]
        try:
            res_s = minimize(lambda x: x[-1], x0, method='SLSQP',
                            bounds=bounds_s, constraints=cons_s,
                            options={'maxiter': 200, 'ftol': 1e-14})
            if res_s.success:
                h_s = repair_feasible(res_s.x[:-1])
                c_s = exact_score(h_s)
                if c_s < best_c - 1e-14:
                    best_h = h_s
                    best_c = c_s
        except Exception:
            pass
    
    # Phase 9: Final micro-polish
    if time_left() > 15:
        h_pol, c_pol = fine_polish(best_h, t_start, budget_s, max_iters=80)
        if c_pol < best_c - 1e-14:
            best_h = h_pol
            best_c = c_pol
    
    # Final validation
    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)
    
    return best_h, final_c, len(best_h)
