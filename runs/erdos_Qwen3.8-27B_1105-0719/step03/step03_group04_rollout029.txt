    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def grad_matrix(h):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    G = np.zeros((M, n))
    center = n - 1
    # d = 0: G[center, i] = dx * (1 - 2*h[i])

        G[center - d, :n-d] += -dx * h[d:n]

    n = len(h)
    dx = 2.0 / n

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n

def grad_matrix(h):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    G = np.zeros((M, n))
    center = n - 1
    G[center, :] = dx * (1.0 - 2.0 * h)
    for d in range(1, n):
        G[center + d, :n-d] += dx * (1.0 - h[d:n])
        G[center + d, d:n] += -dx * h[:n-d]
        G[center - d, d:n] += dx * (1.0 - h[:n-d])
        G[center - d, :n-d] += -dx * h[d:n]
    return G

import numpy as np
from scipy.optimize import minimize
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def grad_matrix(h):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    G = np.zeros((M, n))
    center = n - 1
    G[center, :] = dx * (1.0 - 2.0 * h)
    for d in range(1, n):
        G[center + d, :n-d] += dx * (1.0 - h[d:n])
        G[center + d, d:n] += -dx * h[:n-d]
        G[center - d, d:n] += dx * (1.0 - h[:n-d])
        G[center - d, :n-d] += -dx * h[d:n]
    return G

def repair_feasible(h):
    n = len(h)
    h = np.clip(np.asarray(h, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        remaining = diff
        for idx in order:
            add = min(remaining, 1.0 - h[idx])
            h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)[::-1]
        remaining = -diff
        for idx in order:
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def lbs_lowse(h, beta, penalty, t_start, budget_s, max_outer=8):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c = exact_score(h)
    for _ in range(max_outer):
        if time.perf_counter() - t_start > budget_s - 30:
            break
        def obj(x):
            prof = compute_profile(x)
            m = prof.max()
            lse = m + (1.0/beta) * np.log(np.sum(np.exp(beta * (prof - m))))
            return lse + penalty * (x.sum() - n/2.0)**2
        def grad(x):
            prof = compute_profile(x)
            G = grad_matrix(x)
            m = prof.max()
            w = np.exp(beta * (prof - m))
            w /= w.sum()
            return G.T @ w + 2.0 * penalty * (x.sum() - n/2.0)
        try:
            res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                          bounds=[(0.0, 1.0)]*n,
                          options={'maxiter': 100, 'ftol': 1e-15, 'gtol': 1e-10})
        except Exception:
            break
        h = repair_feasible(res.x)
        c = exact_score(h)
        if c < best_c - 1e-15:
            best_c = c
            best_h = h.copy()
        if res.nit <= 1:
            break
    return best_h, best_c

def qp_region(h, t_start, budget_s, max_iter=100):
    try:
        import cvxpy as cp
    except ImportError:
        return h, exact_score(h)
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c = exact_score(h)
    rho = 0.04
    gamma = 8.0
    for it in range(max_iter):
        if time.perf_counter() - t_start > budget_s - 20:
            break
        profile = compute_profile(h)
        c = profile.max()
        if c < best_c - 1e-15:
            best_c = c
            best_h = h.copy()
        thresh = c - 0.004
        active = np.where(profile >= thresh)[0]
        if len(active) == 0:
            active = np.array([np.argmax(profile)])
        if len(active) > 25:
            active = active[np.argsort(profile[active])[-25:]]
        G = grad_matrix(h)
        G_a = G[active, :]
        P_a = profile[active]
        delta_v = cp.Variable(n)
        t_v = cp.Variable()
        cons = [delta_v >= -rho, delta_v <= rho,
                h + delta_v >= 0, h + delta_v <= 1,
                sum(delta_v) == 0]
        for k in range(len(active)):
            cons.append(P_a[k] + G_a[k] @ delta_v <= t_v)
        prob = cp.Problem(cp.Minimize(t_v + gamma * cp.sum_squares(delta_v)), cons)
        solved = False
        for solver in [cp.SCS, cp.CLARABEL, cp.ECOS]:
            try:
                prob.solve(solver=solver, max_iters=5000)
                if delta_v.value is not None:
                    delta = np.array(delta_v.value).flatten()
                    if np.isfinite(delta).all():
                        solved = True
                        break
            except Exception:
                continue
        if not solved:
            rho *= 0.6
            if rho < 0.002:
                break
            continue
        improved = False
        for alpha in [1.0, 0.7, 0.5, 0.3, 0.15, 0.07]:
            h_try = repair_feasible(h + alpha * delta)
            s = exact_score(h_try)

                h = h_try

                break
    c_fin = exact_score(h)

def coord_descend(h, t_start, budget_s, max_sweeps=60, n_grid=51):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c = exact_score(h)
    prev_c = best_c
    for sweep in range(max_sweeps):
        if time.perf_counter() - t_start > budget_s - 15:
            break
        profile = compute_profile(h)
        G = grad_matrix(h)
        any_improve = False
        for i in range(n):
            if time.perf_counter() - t_start > budget_s - 15:
                break
            best_t = 0.0
            best_val = profile.max()
            best_j = -1
            d_all = G[:, i] - G  # shape (M, n), d_all[k,j] = G[k,i]-G[k,j]
            t_lo = np.maximum(-h[i], h[:-1] if i>0 else h)  # not quite right, do per-j
            for j in range(n):
                if j == i:
                    continue
                d = d_all[:, j]
                t_min = max(-h[i], h[j] - 1.0)
                t_max = min(1.0 - h[i], h[j])
                if t_max - t_min < 1e-12:
                    continue
                ts = np.linspace(t_min, t_max, n_grid)
                vals = np.max(profile[None, :] + ts[:, None] * d[None, :], axis=1)
                idx = np.argmin(vals)
                if vals[idx] < best_val - 1e-14:
                    best_val = vals[idx]
                    best_t = ts[idx]
                    best_j = j
            if best_j >= 0 and best_t != 0.0:
                h[i] += best_t
                h[best_j] -= best_t
                h = repair_feasible(h)
                profile = compute_profile(h)
                any_improve = True
        c = exact_score(h)
        if c < best_c - 1e-15:
            best_c = c
            best_h = h.copy()
        if c >= prev_c - 1e-15 and not any_improve:
            break
        prev_c = c
    return best_h, best_c

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    fx = np.concatenate([[0.0], xs, [2.0]])
    fh = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, fx, fh)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)
