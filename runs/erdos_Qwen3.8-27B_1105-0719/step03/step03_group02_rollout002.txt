import numpy as np
from scipy.optimize import minimize, linprog
from scipy.special import logsumexp
import time


def project_feasible(x, target_sum):
    """Project x onto {sum = target_sum, 0 <= x <= 1} using water-filling."""
    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)


def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def c5_of_h(h, dx):
    return compute_corrs(h).max() * dx


def grad_corr_k(h, k, n):
    """Gradient of corr_k with respect to h."""
    offset = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -offset)
    i_hi = min(n - 1, n - 1 - offset)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx + offset] += (1.0 - h[idx])
    grad[idx] += (-h[idx + offset])
    return grad


def smooth_max_grad(h, dx, beta):
    """Gradient of smooth max objective."""
    n = len(h)
    corrs = compute_corrs(h)
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones_like(corrs) / len(corrs)
    else:
        w /= w_sum
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] > 1e-13:
            grad += w[k] * grad_corr_k(h, k, n)
    return grad * dx

    """SLSQP with smooth max objective."""
    n = len(h0)
    
    def obj_fn(x):
        corrs = compute_corrs(x)
        return logsumexp(beta * corrs) / beta * dx
    
    def grad_fn(x):
        return smooth_max_grad(x, dx, beta)
    
    def c_fun(x):
        return x.sum() - target_sum
    
    def c_jac(x):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': c_fun, 'jac': c_jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_fn, h0, jac=grad_fn, method='SLSQP',
                       bounds=bounds, constraints=cons,

        h_out = project_feasible(h0, target_sum)

                    break
    
    return best_h, best_c5


def target_descent(h0, dx, target_sum, max_iters=300):
    """Try to drive all top correlations below a target level."""
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    
    for it in range(max_iters):
        corrs = compute_corrs(h)
        c5_val = corrs.max() * dx
        current_max = corrs.max()
        
        # Target: slightly below current max
        target = current_max * (1.0 - 0.02)
        
        # Get top active correlations
        n_active = min(40, len(corrs))
        top_idx = np.argsort(corrs)[-n_active:]
        
        # LP: minimize step size (approximated by max |delta|)
        # min t s.t. dx * grad_k . delta + corr_k * dx <= target, |delta_i| <= t
        # Actually let's just use the standard LP with tighter bound
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        A_rows = []
        b_rows = []
        for k in top_idx:
            row = np.zeros(n + 1)
            row[:n] = dx * grad_corr_k(h, k, n)
            row[n] = -1.0
            A_rows.append(row)
            b_rows.append(-target * dx - 1e-13)
        
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)
        
        step_bound = 0.2
        bounds = []
        for i in range(n):
            bounds.append((max(-h[i], -step_bound), min(1.0 - h[i], step_bound)))
        bounds.append((0.0, None))
        
        try:
            res = linprog(c_obj, A_ub=np.array(A_rows), b_ub=np.array(b_rows),
                          A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
        except Exception:
            break
        
        if res.status != 0:
            break
        
        delta = res.x[:n]
        if np.max(np.abs(delta)) < 1e-14:
            break
        
        h_new = project_feasible(h + delta, target_sum)
        new_c5 = c5_of_h(h_new, dx)
        
        if new_c5 < best_c5 - 1e-14:
            h = h_new
            best_c5 = new_c5
            best_h = h_new.copy()
        elif new_c5 < c5_val + 1e-10:
            h = h_new
        else:
            # Try smaller step
            h_new = project_feasible(h + 0.5 * delta, target_sum)
            new_c5 = c5_of_h(h_new, dx)
            if new_c5 < best_c5 - 1e-14:
                h = h_new
                best_c5 = new_c5
                best_h = h_new.copy()
            else:
                break
    
    return best_h, best_c5


def pair_refine(h0, dx, target_sum, n_top=30, n_iter=5):
    """Pairwise transfer refinement with exact line search."""
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    
    for iteration in range(n_iter):
        corrs = compute_corrs(h)
        # Gradient of smooth max (high beta ≈ exact max gradient)
        grad = smooth_max_grad(h, dx, 500.0)
        
        # Top increasing and decreasing coordinates
        inc_idx = np.argsort(grad)[:n_top]
        dec_idx = np.argsort(grad)[-n_top:]
        
        improved = False
        for i in dec_idx:
            for j in inc_idx:
                if i == j:
                    continue
                eps_hi = min(h[i], 1.0 - h[j])
                if eps_hi < 1e-10:
                    continue
                
                # Compute d vector: change in correlations when moving eps from i to j
                d = np.zeros(2 * n - 1)
                for k in range(2 * n - 1):
                    gk = grad_corr_k(h, k, n)
                    d[k] = gk[j] - gk[i]
                
                # Golden section search for optimal eps
                gr = (np.sqrt(5.0) - 1.0) / 2.0
                a, b = 0.0, eps_hi
                c_pt = b - gr * (b - a)
                d_pt = a + gr * (b - a)
                
                def f_eps(eps):
                    new_corrs = corrs + eps * d
                    return new_corrs.max() * dx
                
                fc, fd = f_eps(c_pt), f_eps(d_pt)
                for _ in range(40):
                    if b - a < 1e-12:
                        break
                    if fc < fd:
                        b = d_pt
                        d_pt = c_pt
                        fd = fc
                        c_pt = b - gr * (b - a)
                        fc = f_eps(c_pt)
                    else:
                        a = c_pt
                        c_pt = d_pt
                        fc = fd
                        d_pt = a + gr * (b - a)
                        fd = f_eps(d_pt)
                
                eps_opt = (a + b) / 2.0
                new_h = h.copy()
                new_h[i] -= eps_opt
                new_h[j] += eps_opt
                new_h = np.clip(new_h, 0.0, 1.0)
                new_h = project_feasible(new_h, target_sum)
                new_c5 = c5_of_h(new_h, dx)
                
                if new_c5 < best_c5 - 1e-14:
                    h = new_h
                    best_c5 = new_c5
                    best_h = h.copy()
                    corrs = compute_corrs(h)
                    improved = True
        
        if not improved:
            break
    
    return best_h, best_c5


def momentum_descent(h0, dx, target_sum, max_iter=20000, beta_smooth=50.0):
    """Momentum subgradient descent with adaptive learning rate."""
    n = len(h0)
    h = h0.copy()
    momentum = np.zeros(n)
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    no_improve = 0
    lr = 0.5 / np.sqrt(n)
    beta_mom = 0.85
    
    for it in range(max_iter):
        corrs = compute_corrs(h)
        c5_val = corrs.max() * dx
        
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
            lr = min(lr * 1.05, 2.0 / np.sqrt(n))
        else:
            no_improve += 1
            if no_improve % 500 == 0:
                lr *= 0.8
        if no_improve > 8000 or lr < 1e-6:
            break
        
        shifted = corrs - corrs.max()
        w = np.exp(beta_smooth * shifted)
        w /= w.sum()
        
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] > 1e-12:
                grad += w[k] * grad_corr_k(h, k, n)
        grad *= dx
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        momentum = beta_mom * momentum - lr * grad / (gnorm + 1e-10)
        h_new = project_feasible(h + momentum, target_sum)
        new_c5 = c5_of_h(h_new, dx)
        
        tries = 0
        while new_c5 > c5_val + 1e-10 and tries < 3:
            lr *= 0.5
            momentum = beta_mom * momentum - lr * grad / (gnorm + 1e-10)
            h_new = project_feasible(h + momentum, target_sum)
            new_c5 = c5_of_h(h_new, dx)
            tries += 1
        
        h = h_new
    
    return best_h, best_c5


def make_structured_seed(n, target_sum, rng, seed_type=0):
    """Create various structured initial seeds."""
    if seed_type == 0:
        # Smooth random
        h = rng.beta(2.0, 2.0, size=n)
    elif seed_type == 1:
        # Sinusoidal
        x = np.linspace(0, 2 * np.pi * 3, n)
        h = 0.5 + 0.4 * np.sin(x + rng.random() * 2 * np.pi)
    elif seed_type == 2:
        # Multi-scale
        h = np.zeros(n)
        for freq in [1, 2, 3, 5]:
            amp = 1.0 / freq
            h += amp * np.sin(2 * np.pi * freq * np.arange(n) / n + rng.random() * 2)
        h = h - h.min()
        h = h / (h.max() - h.min() + 1e-10)
    elif seed_type == 3:
        # Block pattern
        block = max(2, n // 20)
        h = np.zeros(n)
        positions = rng.choice(n, size=n//2, replace=False)
        h[positions] = 1.0
    else:
        h = np.ones(n) * 0.5
    
    return project_feasible(h, target_sum)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    
    def time_left():
        return budget_s - (time.time() - t0)
    
    # Get initial
    h0 = np.array(initial_h_values, dtype=float)
    n = len(h0)
    dx = 2.0 / n
    target_sum = n / 2.0
    h0 = project_feasible(h0, target_sum)
    
    best_h = h0.copy()
    best_c5 = c5_of_h(h0, dx)
    
    # Phase 1: SLSQP with increasing beta (main optimization)
    for beta in [200, 500, 1500, 5000, 15000]:
        if time_left() < 25:
            break
        h_opt, c5_val = slsqp_optimize(best_h, dx, target_sum, beta, 2000)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_opt
    
    # Phase 2: LP polish (main workhorse)
    if time_left() > 60:
        h_lp, c5_lp = lp_polish(best_h, dx, target_sum, max_iters=500, init_step=0.12)
        if c5_lp < best_c5:
            best_c5 = c5_lp
            best_h = h_lp
    
    # Phase 3: Target descent
    if time_left() > 40:
        h_td, c5_td = target_descent(best_h, dx, target_sum, max_iters=300)
        if c5_td < best_c5:
            best_c5 = c5_td
            best_h = h_td
    
    # Phase 4: More LP
    if time_left() > 40:
        h_lp2, c5_lp2 = lp_polish(best_h, dx, target_sum, max_iters=400, init_step=0.08)
        if c5_lp2 < best_c5:
            best_c5 = c5_lp2
            best_h = h_lp2
    
    # Phase 5: Pair refinement
    if time_left() > 30:
        h_pr, c5_pr = pair_refine(best_h, dx, target_sum, n_top=25, n_iter=4)
        if c5_pr < best_c5:
            best_c5 = c5_pr
            best_h = h_pr
    
    # Phase 6: Momentum descent
    if time_left() > 60:
        h_md, c5_md = momentum_descent(best_h, dx, target_sum, max_iter=15000)
        if c5_md < best_c5:
            best_c5 = c5_md
            best_h = h_md
    
    # Phase 7: Final SLSQP + LP
