```python
def lse_gradient(h, n, tau):
    """Compute gradient of log-sum-exp smoothed objective."""
    dx = 2.0 / n
    profile = compute_profile(h)
    M = len(profile)
    center = n - 1
    
    # Softmax weights
    shifted = (profile - profile.max()) / tau
    weights = np.exp(shifted)
    weights /= weights.sum()
    
    # Gradient = weighted sum of individual profile gradients
    # G[k, j] = d profile_k / d h_j
    # grad[j] = sum_k weights[k] * G[k, j]

            valid = n - s
            grad[valid:] += w * dx * (1.0 - h[0:n-valid])
```

```python
def annealed_lse_optimize(h, n, t_start, budget_s, tau_start=0.05, tau_end=0.001, 
                          lr_start=0.01, momentum=0.9, max_iters=1000):
    """Annealed log-sum-exp projected gradient descent."""
    h = h.copy()
    best_h = h.copy()
    best_c = exact_score(h)
    v = np.zeros(n)  # momentum velocity
    
    for it in range(max_iters):
        if time.perf_counter() - t_start > budget_s - 30:
            break
        
        # Schedule
        frac = it / max_iters
        tau = tau_start * (tau_end / tau_start) ** frac
        lr = lr_start * (1.0 - 0.5 * frac)  # slight decay
        
        # Compute gradient
        profile = compute_profile(h)
        shifted = (profile - profile.max()) / tau
        weights = np.exp(shifted)
        weights /= weights.sum()
        
        G = build_gradient(h, n, 2.0/n)
        grad = G.T @ weights
        
        # Remove mean component (tangent to sum constraint)
        grad -= grad.mean()
        
        # Momentum
        v = momentum * v - lr * grad
        h_new = project_feasible(h + v, n)
        
        # Evaluate
        c = exact_score(h_new)
        if c < best_c - 1e-14:
            best_c = c
            best_h = h_new.copy()
        
        h = h_new
    
    return best_h, best_c
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def build_gradient(h, n):
    dx = 2.0 / n
    M = 2 * n - 1
    G = np.zeros((M, n))
    center = n - 1
    G[center, :] = dx * (1.0 - 2.0 * h)
    for d in range(1, n):
        G[center + d, 0:n-d] += -dx * h[d:n]
        G[center + d, d:n] += dx * (1.0 - h[0:n-d])
        G[center - d, 0:n-d] += dx * (1.0 - h[d:n])
        G[center - d, d:n] += -dx * h[0:n-d]
    return G

def project_feasible(y, n):
    h = y.copy()
    target = n / 2.0
    lo = h.min() - 2.0
    hi = h.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) / 2, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt > 0:
            h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def annealed_lse_optimize(h0, n, t_start, deadline, tau_start=0.1, tau_end=0.0005,
                          lr_start=0.02, lr_end=0.001, momentum=0.85, max_iters=5000):
    h = h0.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c = exact_score(h)
    no_improve = 0
    
    for it in range(max_iters):
        if time.perf_counter() > deadline:
            break
        
        frac = it / max_iters
        # Cosine annealing for tau
        tau = tau_end + 0.5 * (tau_start - tau_end) * (1 + np.cos(np.pi * frac))
        lr = lr_end + 0.5 * (lr_start - lr_end) * (1 + np.cos(np.pi * frac))
        
        # Compute LSE gradient
        profile = compute_profile(h)
        M = len(profile)
        pmax = profile.max()
        shifted = (profile - pmax) / tau
        # Numerical stability
        shifted -= shifted.max()
        weights = np.exp(shifted)
        weights /= weights.sum()
        
        G = build_gradient(h, n)
        grad = G.T @ weights
        grad -= grad.mean()  # Tangent to sum constraint
        
        # Momentum
        v = momentum * v - lr * grad
        h_new = project_feasible(h + v, n)
        
        c = exact_score(h_new)
        if c < best_c - 1e-15:
            best_c = c
            best_h = h_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        h = h_new
        
        # Jitter periodically
        if no_improve > 200 and it % 50 == 0:
            h = project_feasible(h + 0.01 * np.random.randn(n), n)
            no_improve = 0
        
        if no_improve > 2000:
            break
    
    return best_h, best_c

def slsqp_polish(h0, n, t_start, deadline):
    """SLSQP epigraph formulation."""
    dx = 2.0 / n
    M = 2 * n - 1
    c0 = exact_score(h0)
    
    def obj(x):
        return x[-1]
    
    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    
    def eq_con(x):
        return x[:-1].sum() - n / 2.0
    
    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g
    
    def ineq_con(x):
        h = x[:-1]
        t = x[-1]
        prof = np.correlate(h, 1 - h, mode="full") * dx
        return t - prof
    
    def ineq_jac(x):
        h = x[:-1]
        G = build_gradient(h, n)
        J = np.zeros((M, n + 1))
        J[:, :-1] = -G
        J[:, -1] = 1.0
        return J
    
    x0 = np.concatenate([h0, [c0 + 1e-3]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    cons = [
        {'type': 'eq', 'fun': eq_con, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_con, 'jac': ineq_jac}
    ]
    
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': 500, 'ftol': 1e-14})
        if res.x is not None and np.isfinite(res.x).all():
            h_sol = project_feasible(res.x[:-1], n)
            c_sol = exact_score(h_sol)
            if c_sol < c0:
                return h_sol, c_sol
    except Exception:
        pass
    return h0, c0

def subgradient_step(h, n, top_k=10):
    """Exact subgradient step using top-k active lags."""
    profile = compute_profile(h)
    M = len(profile)
    T = profile.max()
    
    # Select active set
    thresh = T - 0.01 * (T - profile.min())
    active = np.where(profile >= thresh)[0]
    if len(active) > top_k:
        active = active[np.argsort(profile[active])[-top_k:]]
    elif len(active) == 0:
        active = np.array([np.argmax(profile)])
    
    G = build_gradient(h, n)
    G_a = G[active, :]
    # Average gradient of active set
    grad = G_a.mean(axis=0)
    grad -= grad.mean()
    
    return grad

def subgradient_optimize(h0, n, t_start, deadline, max_iters=5000):
    h = h0.copy()
    best_h = h.copy()
    best_c = exact_score(h)
    lr = 0.005
    no_improve = 0
    
    for it in range(max_iters):
        if time.perf_counter() > deadline:
            break
        
        grad = subgradient_step(h, n)
        norm = np.linalg.norm(grad)
        if norm < 1e-12:
            break
        
        # Step size (diminishing for convergence)
        step = lr / (1 + 0.001 * it)
        
        h_new = project_feasible(h - step * grad, n)
        c = exact_score(h_new)
        
        if c < best_c - 1e-15:
            best_c = c
            best_h = h_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        h = h_new
        
        if no_improve > 500:
            break
    
    return best_h, best_c

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    return project_feasible(h_new, new_n)
```

```python
def active_set_lp(h, n, top_k=15):
    """LP step that reduces the max over the top-k active lags."""
    dx = 2.0 / n
    profile = compute_profile(h)
    T = profile.max()
    
    # Select active set
    thresh = T - 0.02 * (T - profile.min())
    active = np.where(profile >= thresh)[0]
    if len(active) > top_k:
        active = active[np.argsort(profile[active])[-top_k:]]
    if len(active) == 0:
        active = np.array([np.argmax(profile)])
    
    G = build_gradient(h, n)
    G_a = G[active, :]
    p_a = profile[active]
    
    # LP: min t s.t. G_a @ delta <= -p_a + t, sum delta = 0, |delta_i| <= radius
    # i.e., min t s.t. G_a @ delta - t <= -p_a, sum delta = 0, -radius <= delta_i <= radius
    radius = 0.05
    from scipy.optimize import linprog
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    A_ub = np.zeros((len(active), n + 1))
    A_ub[:, :n] = G_a
    A_ub[:, -1] = -1.0
    b_ub = -p_a
    
    A_eq = np.ones((1, n + 1))
    A_eq[0, -1] = 0.0
    b_eq = np.array([0.0])
    
    bounds = [(-radius, radius)] * n + [(0.0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        delta = res.x[:n]
        h_new = project_feasible(h + delta, n)
        return h_new
    
    return None
```
