def soft_max_gradient(profile, tau):
    """Compute softmax weights for the profile."""
    shifted = (profile - profile.max()) / tau
    exp_vals = np.exp(shifted)
    weights = exp_vals / exp_vals.sum()
    return weights

def log_sum_exp(profile, tau):
    """Compute log-sum-exp of profile with temperature tau."""
    shifted = (profile - profile.max()) / tau
    return profile.max() + tau * np.log(np.exp(shifted).sum())

def soft_max_value_and_grad(h, tau):
    """Compute the smooth max value and its gradient w.r.t. h."""
    n = len(h)
    dx = 2.0 / n
    profile = compute_profile(h)
    M = len(profile)
    G = build_gradient_matrix(h, n, dx)
    
    shifted = (profile - profile.max()) / tau
    exp_vals = np.exp(shifted)
    weights = exp_vals / exp_vals.sum()
    
    value = profile.max() + tau * np.log(exp_vals.sum())
    grad = G.T @ weights  # shape (n,)
    return value, grad, profile, weights

def anneal_optimize(h, t_start, budget_s, tau_lo=0.001, tau_hi=1.0,

        elapsed = time.perf_counter() - t_start
        if budget_s - elapsed < 5:
            break
        
        # Current temperature (exponential decay)
        progress = it / n_actual
        tau = tau_hi * (tau_lo / tau_hi) ** progress

def slsqp_polish(h, t_start, budget_s):
    """Final SLSQP epigraph polishing."""
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    c0 = exact_score(h)
    
    def obj_fn(x):
        return x[-1]
    
    def eq_fn(x):
        return x[:-1].sum() - n / 2.0
    
    def ineq_fn(x):
        h_vec = x[:-1]
        t = x[-1]
        prof = np.correlate(h_vec, 1 - h_vec, mode="full") * dx
        return t - prof
    
    x0 = np.concatenate([h, [c0 + 1e-4]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    cons = [
        {'type': 'eq', 'fun': eq_fn},
        {'type': 'ineq', 'fun': ineq_fn}
    ]
    
    try:
        res = minimize(obj_fn, x0, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': 500, 'ftol': 1e-14})
        if res.success or res.x is not None:
            h_sol = np.clip(res.x[:-1], 0, 1)
            h_sol = project_feasible(h_sol, n / 2.0)
            c_sol = exact_score(h_sol)
            if c_sol < c0 - 1e-14:
                return h_sol, c_sol
    except Exception:
        pass
    return h, c0

def active_set_lp_polish(h, t_start, budget_s, max_iters=200):

            break

        c = profile.max()
        T = c
        
        # Get active set (profile values close to max)
        beam = max(0.001, 0.02 * (T - profile.min()))
        active = np.where(profile >= T - beam)[0]
        
        if len(active) == 0:
            active = np.array([np.argmax(profile)])
        if len(active) > 20:
            top_idx = np.argsort(profile[active])[-20:]
            active = active[top_idx]

        G_a = G[active, :]

        c_obj[-1] = 1.0

        A_ub[:, -1] = -1.0
        b_ub = -p_a
        A_eq = np.zeros((1, n + 1))

        upper = np.minimum(1.0 - h, radius)
        bounds = list(zip(lower, upper))

            if res.success and res.x is not None:

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    h_new = np.clip(h_new, 0, 1)
    return project_feasible(h_new, new_n / 2.0)

        else:

    diff = target_sum - h_proj.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            order = np.argsort(1.0 - h_proj)
        else:
            order = np.argsort(h_proj)
        remaining = diff
        for idx in order:
            if abs(remaining) < 1e-15:
                break
            if remaining > 0:
                add = min(remaining, 1.0 - h_proj[idx])
                if add > 0:
                    h_proj[idx] += add
                    remaining -= add
            else:
                rem = min(-remaining, h_proj[idx])
                if rem > 0:
                    h_proj[idx] -= rem
                    remaining += rem
    return h_proj

def build_gradient_matrix(h, n, dx):
    M = 2 * n - 1
    G = np.zeros((M, n))
    center = n - 1
    G[center, :] = dx * (1.0 - 2.0 * h)
    for d in range(1, n):
        G[center + d, 0:n-d] += -dx * h[d:n]
        G[center + d, d:n] += dx * (1.0 - h[0:n-d])
        G[center - d, 0:n-d] += dx * (1.0 - h[d:n])
        G[center - d, d:n] += -dx * h[0:n-d]
    return G

def soft_max_grad(h, tau):

    weights = exp_vals / exp_vals.sum()
