import time

def evaluate_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def project_to_feasible(h, target_sum):
    """Project h to [0,1]^n with sum = target_sum using bisection on threshold."""
    n = len(h)
    lo, hi = -1.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        h_proj = np.clip(h - mid, 0.0, 1.0)
        if h_proj.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix any tiny numerical error
    diff = target_sum - h_proj.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h_proj < 1.0 - 1e-10
        else:
            mask = h_proj > 1e-10
        cnt = mask.sum()
        if cnt > 0:
            h_proj[mask] += diff / cnt
    return np.clip(h_proj, 0.0, 1.0)

def coord_descent_pass(h, n, dx, grid_size=500, order=None):
    """One pass of coordinate descent. Returns new h and whether improvement was made."""
    if order is None:
        order = np.random.permutation(n)
    
    corrs = np.correlate(h, 1.0 - h, mode='full')
    old_c5 = corrs.max() * dx
    
    for i0 in order:
        # Compute A and d for the 1D subproblem
        A = np.empty(2*n - 1)
        d = np.empty(2*n - 1)
        
        for m in range(2*n - 1):
            o = m - (n - 1)
            if o == 0:
                orig_contrib = h[i0] * (1.0 - h[i0])
                A[m] = corrs[m] - orig_contrib
                d[m] = -1.0  # flag
            else:
                contrib = 0.0
                coeff = 0.0
                if 0 <= i0 - o < n:
                    contrib += h[i0] * (1.0 - h[i0 - o])
                    coeff += (1.0 - h[i0 - o])
                if 0 <= i0 + o < n:
                    contrib += h[i0 + o] * (1.0 - h[i0])
                    coeff += -h[i0 + o]
                A[m] = corrs[m] - contrib
                d[m] = coeff
        
        # Solve 1D: minimize max_m C[m](t) over t in [0,1]
        ts = np.linspace(0.0, 1.0, grid_size)
        vals = A[None, :] + d[None, :] * ts[:, None]
        quad = A[n-1] + ts - ts**2
        vals[:, n-1] = quad
        f_vals = vals.max(axis=1)
        
        # Boundary check
        best_idx = np.argmin(f_vals)
        t_star = float(ts[best_idx])
        
        # Refine
        if 0 < best_idx < grid_size - 1:
            t_lo = float(ts[best_idx - 1])
            t_hi = float(ts[best_idx + 1])
            ts_r = np.linspace(t_lo, t_hi, 100)
            vals_r = A[None, :] + d[None, :] * ts_r[:, None]
            quad_r = A[n-1] + ts_r - ts_r**2
            vals_r[:, n-1] = quad_r
            f_vals_r = vals_r.max(axis=1)
            best_idx_r = np.argmin(f_vals_r)
            t_star = float(ts_r[best_idx_r])
        
        if t_star != h[i0]:
            h[i0] = t_star
            # Update corrs (expensive but necessary for correctness)
            # Actually, we can update corrs incrementally...
            # For now, recompute
            corrs = np.correlate(h, 1.0 - h, mode='full')
    
    new_c5 = corrs.max() * dx
    return h, new_c5 < old_c5 - 1e-12

    # For m != n-1 (o != 0): C[m](t) = A[m] + d[m] * t (linear)
    # For m == n-1 (o == 0): C[m](t) = A0 + t - t² (concave quadratic)
