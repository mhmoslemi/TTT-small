    d = np.zeros(nx)

    # Term 2: if 0 <= p - off < n and off != 0: c[k] += -h[p-off], d[k] += h[p-off]

def pair_optimize(h, i, j, target_sum, n):
    """Optimize h[i] and h[j] given all others fixed, with h[i]+h[j] fixed."""
    s = target_sum - (np.sum(h) - h[i] - h[j])
    # t range: max(0, s-1) <= t <= min(1, s)
    t_lo = max(0.0, s - 1.0)
    t_hi = min(1.0, s)
    if t_lo >= t_hi:
        return h[i], h[j]  # no freedom
    
    # For each lag k, compute g_k as linear in t (for k ≠ n-1)
    # h[i] = t, h[j] = s - t
    # g_k = sum over valid terms
    
    nx = 2 * n - 1
    offs = np.arange(nx) - (n - 1)
    
    # For coordinate i with value t:
    # Term 1: if 0 <= i + off < n and off != 0: contributes t * (1 - h[i+off])
    # Term 2: if 0 <= i - off < n and off != 0: contributes h[i-off] * (1 - t)
    # For coordinate j with value s-t:
    # Term 1: if 0 <= j + off < n and off != 0: contributes (s-t) * (1 - h[j+off])
    # Term 2: if 0 <= j - off < n and off != 0: contributes h[j-off] * (1 - (s-t))
    
    # Plus for k = n-1: the quadratic terms for i and j
    
    # Linear coefficients for t in each g_k (k ≠ n-1):
    c = np.zeros(nx)
    d = np.zeros(nx)
    
    off = offs
    
    # Contributions from h[i] = t:
    m1i = (off >= -i) & (off < n - i) & (off != 0)
    if m1i.any():
        c[m1i] += (1.0 - h[i + off[m1i]])  # coeff of t
    
    m2i = (off >= i - (n - 1)) & (off <= i) & (off != 0)
    if m2i.any():
        idx = i - off[m2i]
        c[m2i] += (-1.0)  # from -t in h[i-off]*(1-t)
        d[m2i] += h[idx]  # constant part h[i-off]
    
    # Contributions from h[j] = s - t:
    # h[j] = s - t, so terms involving h[j] become (s-t) or (1-(s-t)) = (1-s+t)
    # Term 1: (s-t) * (1 - h[j+off])
    #   = (s-t) * (1-h[j+off]) = s*(1-h[j+off]) - t*(1-h[j+off])
    #   coeff of t: -(1-h[j+off]), const: s*(1-h[j+off])
    m1j = (off >= -j) & (off < n - j) & (off != 0)
    if m1j.any():
        c[m1j] += (-(1.0 - h[j + off[m1j]]))
        d[m1j] += s * (1.0 - h[j + off[m1j]])
    
    # Term 2: h[j-off] * (1 - (s-t)) = h[j-off] * (1-s+t)
    #   = h[j-off]*(1-s) + h[j-off]*t
    #   coeff of t: h[j-off], const: h[j-off]*(1-s)
    m2j = (off >= j - (n - 1)) & (off <= j) & (off != 0)
    if m2j.any():
        idx = j - off[m2j]
        c[m2j] += h[idx]
        d[m2j] += h[idx] * (1.0 - s)
    
    # Quadratic part for k = n-1:
    # g_{n-1}(t) = -2t² + 2st + (s - s²) + const
    # where const = sum of h[k]*(1-h[k]) for k ≠ i, j
    const_quad = np.sum(h * (1.0 - h)) - h[i]*(1.0-h[i]) - h[j]*(1.0-h[j])
    
    # Grid search
    G = 300
    T = np.linspace(t_lo, t_hi, G)
    
    mask_lin = np.ones(nx, dtype=bool)
    mask_lin[n - 1] = False
    c_lin = c[mask_lin]
    d_lin = d[mask_lin]
    
    g_lin = c_lin[:, None] * T[None, :] + d_lin[None, :]
    max_lin = g_lin.max(axis=0)
    
    g_quad = -2.0*T**2 + 2.0*s*T + (s - s**2) + const_quad
    f = np.maximum(max_lin, g_quad)
    
    best_idx = f.argmin()
    best_t = T[best_idx]
    
    # Refine
    if 0 < best_idx < G - 1:
        lo2 = T[best_idx - 1]
        hi2 = T[best_idx + 1]
        T2 = np.linspace(lo2, hi2, 100)
        g_lin2 = c_lin[:, None] * T2[None, :] + d_lin[None, :]
        max_lin2 = g_lin2.max(axis=0)
        g_quad2 = -2.0*T2**2 + 2.0*s*T2 + (s - s**2) + const_quad
        f2 = np.maximum(max_lin2, g_quad2)
        best_t = T2[f2.argmin()]
    
    return best_t, s - best_t

def ensure_feasible(y, target_sum):
    n = len(y)
    y = np.clip(y, 0.0, 1.0)
    diff = target_sum - y.sum()
    if abs(diff) < 1e-10:
        return y
    if diff > 0:
        mask = y < 1.0 - 1e-10
        cnt = mask.sum()
        if cnt > 0:

    else:
