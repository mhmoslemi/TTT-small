    n = len(h)
    dx = 2.0 / n
    c = h[p] + h[q]
    t_lo = max(0.0, c - 1.0)
    t_hi = min(1.0, c)
    if t_hi - t_lo < 1e-12:
        return None
    # Save original values
    hp, hq = h[p], h[q]
    # Zero out
    h[p] = 0.0
    h[q] = 0.0
    h_omp[p] = 1.0
    h_omp[q] = 1.0

        h[p] = t
        h[q] = c - t
        h_omp[p] = 1.0 - t
        h_omp[q] = 1.0 - (c - t)
        return np.max(np.correlate(h, h_omp, mode="full")) * dx

    fd = f(d_)

        if fc < fd:

            fc = f(c_)

            d_ = a + gr * (b - a)
            fd = f(d_)

    h[p], h[q] = hp, hq
    h_omp[p], h_omp[q] = 1.0 - hp, 1.0 - hq

def exact_score(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full"))) * (2.0 / n)

def profile_h(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx])
            h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    return h

def build_grad_rows(h, rows, n, dx):
    G = np.zeros((len(rows), n))
    for idx, m in enumerate(rows):
        d = m - (n - 1)
        if d == 0:
            G[idx, :] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            for i in range(n - d):
                G[idx, i + d] += dx * (1.0 - h[i])
                G[idx, i] -= dx * h[i + d]
        else:
            dp = -d
            for j in range(n - dp):
                G[idx, j + dp] += dx * (1.0 - h[j])
                G[idx, j] -= dx * h[j + dp]
    return G

def slsqp_active(h0, maxiter=300, active_k=15, tol=1e-12):
    n = len(h0)
    dx = 2.0 / n
    M = 2 * n - 1
    P = profile_h(h0)

    thresh = c0 - max(0.002, 0.04 * (c0 - np.median(P)))
    active = np.where(P >= thresh)[0]
    if len(active) == 0:
        active = np.array([np.argmax(P)])

        return x[-1]

        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g

        return x[:-1].sum() - n / 2.0

        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g

        return J

        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_grad}

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    return repair_feasible(np.clip(h_new, 0.0, 1.0))
