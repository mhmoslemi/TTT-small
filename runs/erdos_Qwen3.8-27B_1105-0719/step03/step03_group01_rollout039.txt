def pair_swap_optimize(h, dx, target_sum, rng, n_iters=2000):

                    best_h = cur_h.copy()

                cur_h[j] -= delta

import numpy as np
from scipy.optimize import minimize
import time


def project_feasible(x, target_sum):
    n = len(x)
    x = np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        cnt = mask.sum()
        if cnt > 0:
            h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)


def c5_of_h(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)


def corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')


def grad_corr_comp(h, d, n):
    grad = np.zeros(n)
    if d >= 0 and d < n:
        grad[d:] += (1.0 - h[:n - d])
        grad[:n - d] -= h[d:]
    elif d < 0 and -d < n:
        da = -d
        grad[:n - da] += (1.0 - h[da:])
        grad[da:] -= h[:n - da]
    return grad


def slsqp_obj(x, n, dx, active_idx, target_sum):
    return x[0]


def slsqp_obj_jac(x, n, dx, active_idx, target_sum):
    g = np.zeros(n + 1)
    g[0] = 1.0
    return g


def slsqp_ineq(x, n, dx, active_idx, target_sum):
    t = x[0]
    h = x[1:]
    c = corr_full(h)
    return t - dx * c[active_idx]


def slsqp_ineq_jac(x, n, dx, active_idx, target_sum):
    h = x[1:]
    m = len(active_idx)
    J = np.zeros((m, n + 1))
    J[:, 0] = 1.0
    for i in range(m):
        d = active_idx[i] - (n - 1)
        J[i, 1:] = -dx * grad_corr_comp(h, d, n)
    return J


def slsqp_eq(x, n, dx, active_idx, target_sum):
    return np.array([x[1:].sum() - target_sum])


def slsqp_eq_jac(x, n, dx, active_idx, target_sum):
    J = np.zeros((1, n + 1))
    J[0, 1:] = 1.0
    return J


def do_slsqp(h, dx, target_sum, maxiter=150, maxfun=5000, n_active=None):
    n = len(h)
    m_total = 2 * n - 1
    corr = corr_full(h)
    n_lp = float(corr.max() * dx)

    if n_active is None or n_active >= m_total:
        active_idx = np.arange(m_total)
    else:
        active_idx = np.argsort(corr)[-n_active:]

    x0 = np.zeros(n + 1)
    x0[0] = n_lp + 1e-5
    x0[1:] = h

    args = (n, dx, active_idx, target_sum)
    bounds = [(0.0, 0.55)] + [(0.0, 1.0)] * n

    try:
        res = minimize(
            slsqp_obj, x0, args=args,
            jac=slsqp_obj_jac,
            method='SLSQP',
            bounds=bounds,
            constraints=[
                {'type': 'ineq', 'fun': slsqp_ineq, 'jac': slsqp_ineq_jac, 'args': args},
                {'type': 'eq', 'fun': slsqp_eq, 'jac': slsqp_eq_jac, 'args': args}
            ],
            options={'maxiter': maxiter, 'ftol': 1e-12, 'maxfun': maxfun}
        )
    except Exception:
        return h, n_lp

        c5_new = c5_of_h(h_new, dx)

def subgrad_step(h, dx, target_sum, alpha, momentum_vec, beta):
    n = len(h)
    corr = corr_full(h)
    k_max = int(np.argmax(corr))
    d = k_max - (n - 1)
    grad = dx * grad_corr_comp(h, d, n)
    momentum_vec = beta * momentum_vec - alpha * grad
    h_new = project_feasible(h + momentum_vec, target_sum)
    return h_new, momentum_vec


def subgrad_optimize(h, dx, target_sum, n_steps=500, alpha0=0.005, beta=0.85, alpha_decay=0.999):
    n = len(h)
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    cur_h = h.copy()
    cur_c5 = best_c5
    mom = np.zeros(n)
    alpha = alpha0
    stagnation = 0

    for _ in range(n_steps):
        cur_h, mom = subgrad_step(cur_h, dx, target_sum, alpha, mom, beta)
        c5_new = c5_of_h(cur_h, dx)
        if c5_new < best_c5 - 1e-13:
            best_c5 = c5_new
            best_h = cur_h.copy()
            stagnation = 0
        elif c5_new < cur_c5 - 1e-13:
            stagnation = 0
        else:

            cur_h = project_feasible(best_h, target_sum)
            cur_c5 = best_c5
            stagnation += 1
            if stagnation > 100:
                break
        alpha = max(1e-5, alpha * alpha_decay)

    return best_h, best_c5

    return best_h, best_c5


def perturb_and_solve(h, dx, target_sum, rng, sigma=0.03, maxiter=100, maxfun=3000):
    n = len(h)
    h_pert = project_feasible(h + sigma * rng.randn(n), target_sum)
    h_opt, c5_opt = do_slsqp(h_pert, dx, target_sum, maxiter=maxiter, maxfun=maxfun)
    return h_opt, c5_opt


def try_resolution(h_best, c5_best, n_new, t0, time_budget):
    n_old = len(h_best)
    dx_old = 2.0 / n_old
    target_old = n_old / 2.0

    if n_new < 40 or n_new > 250:
        return None, c5_best

    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h_best)
    target_new = n_new / 2.0
    dx_new = 2.0 / n_new
    h_new = project_feasible(h_new, target_new)
    c5_new = c5_of_h(h_new, dx_new)

    if c5_new >= c5_best:
        return None, c5_best

    if time.time() - t0 < time_budget - 15:
        h_opt, c5_opt = do_slsqp(h_new, dx_new, target_new, maxiter=80, maxfun=2500)
        if c5_opt < c5_best:
            return h_opt, c5_opt

    if c5_new < c5_best:
        return h_new, c5_new
    return None, c5_best


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float)
    except NameError:
        h0 = np.full(126, 0.5)

    n = len(h0)
    dx = 2.0 / n
    target_sum = n / 2.0
    h0 = project_feasible(h0, target_sum)

    best_h = h0.copy()
    best_c5 = c5_of_h(h0, dx)

    def elapsed():
        return time.time() - t0

    def remaining():
        return budget_s - elapsed()

    # Phase 1: Primary SLSQP from parent
    if remaining() > 30:
        best_h, best_c5 = do_slsqp(best_h, dx, target_sum, maxiter=200, maxfun=8000)

    # Phase 2: SLSQP with expanding active set
    if remaining() > 30:
        best_h, best_c5 = do_slsqp_active_expand(best_h, dx, target_sum,
                                                  maxiter=100, maxfun=3000,
                                                  start_active=60, expand_to=2 * n - 1)

    # Phase 3: More SLSQP passes with full constraints
    for _ in range(3):
        if remaining() < 20:
            break
        h_opt, c5_opt = do_slsqp(best_h, dx, target_sum, maxiter=150, maxfun=5000)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
        else:
            h_opt, c5_opt = do_slsqp(best_h, dx, target_sum, maxiter=100, maxfun=3000, n_active=50)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt

    # Phase 4: Subgradient method
    if remaining() > 60:
        h_sub, c5_sub = subgrad_optimize(best_h, dx, target_sum, n_steps=300, alpha0=0.003)
        if c5_sub < best_c5:
            best_c5 = c5_sub
            best_h = h_sub
            h_opt, c5_opt = do_slsqp(best_h, dx, target_sum, maxiter=100, maxfun=3000)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt

    # Phase 5: Try variants
    variants = [project_feasible(1.0 - best_h, target_sum),
                project_feasible(best_h[::-1].copy(), target_sum)]
    for v in variants:
        if remaining() < 20:
            break
        c5_v = c5_of_h(v, dx)
        if c5_v < best_c5:
            best_c5 = c5_v
            best_h = v.copy()
        h_opt, c5_opt = do_slsqp(v, dx, target_sum, maxiter=100, maxfun=3000)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 6: Perturbed SLSQP
    sigmas = [0.02, 0.04, 0.01, 0.06, 0.03]
    for s in sigmas:
        if remaining() < 20:
            break
        h_opt, c5_opt = perturb_and_solve(best_h, dx, target_sum, rng, sigma=s,
                                          maxiter=100, maxfun=3000)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 7: Line search polish
    if remaining() > 30:
        n_dirs = min(60, max(20, int(remaining() / 8)))
        best_h, best_c5 = line_polish(best_h, dx, target_sum, rng, n_dirs=n_dirs)

    # Phase 8: Pair swap
    if remaining() > 40:
        n_iters = min(5000, max(1000, int(remaining() / 3)))
        best_h, best_c5 = pair_swap_optimize(best_h, dx, target_sum, rng, n_iters=n_iters)

    # Phase 9: Try different resolutions
    if remaining() > 80:
        n_candidates = [n - 2, n + 2, n - 4, n + 4, n - 6, n + 6, 120, 128, 124, 130]
        for n_try in n_candidates:
            if remaining() < 40:
                break
            if n_try == n or n_try < 60 or n_try > 200:
                continue
            h_new, c5_new = try_resolution(best_h, best_c5, n_try, t0, remaining())
            if h_new is not None and c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new

    # Phase 10: Final intensive polish with SLSQP
    if remaining() > 20:
        for mi, mf in [(200, 8000), (300, 10000)]:
            if remaining() < 20:
                break
            h_opt, c5_opt = do_slsqp(best_h, dx, target_sum, maxiter=mi, maxfun=mf)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt

    # Phase 11: Tiny line search at the end
    if remaining() > 10:
        best_h, best_c5 = line_polish(best_h, dx, target_sum, rng, n_dirs=20)

    # Phase 12: Final SLSQP if time permits
    if remaining() > 15:
        h_opt, c5_opt = do_slsqp(best_h, dx, target_sum, maxiter=100, maxfun=4000)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Final validation
    n_final = len(best_h)
    dx_final = 2.0 / n_final
    target_final = n_final / 2.0
    best_h = project_feasible(best_h, target_final)
    best_h = np.clip(best_h, 0.0, 1.0)
    # Ensure exact sum
    s = best_h.sum()
    if abs(s - target_final) > 1e-10:
        mask = (best_h > 1e-8) & (best_h < 1.0 - 1e-8)
        if mask.sum() > 0:
            best_h[mask] += (target_final - s) / mask.sum()
        best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = c5_of_h(best_h, dx_final)

    return best_h, best_c5, n_final
