import numpy as np
from scipy.optimize import linprog, minimize
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full"))) * dx

def profile_h(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx])
            h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    return h

def build_gradient_matrix(h, lag_indices, n, dx):
    """Build gradient matrix. G[m, j] = dP[m] / dh[j] for m in lag_indices."""
    M = len(lag_indices)
    G = np.zeros((M, n))
    for idx, m in enumerate(lag_indices):
        d = m - (n - 1)
        if d == 0:
            G[idx, :] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            for i in range(n - d):
                G[idx, i + d] += dx * (1.0 - h[i])
                G[idx, i] -= dx * h[i + d]
        else:
            dp = -d
            for j in range(n - dp):
                G[idx, j + dp] += dx * (1.0 - h[j])
                G[idx, j] -= dx * h[j + dp]
    return G

def lp_direction(h, active_lags):
    """Solve LP to find direction that minimizes max of G@d over active lags."""
    n = len(h)
    dx = 2.0 / n
    G = build_gradient_matrix(h, active_lags, n, dx)  # shape (M, n)
    M = len(active_lags)
    
    # Variables: d[0..n-1], t
    # c = [0, ..., 0, 1] (minimize t)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Inequality constraints: G @ d - t ≤ 0 for each active lag
    # A_ub shape: (M, n+1)
    A_ub = np.zeros((M, n + 1))
    A_ub[:, :n] = G
    A_ub[:, -1] = -1.0
    b_ub = np.zeros(M)
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = [0.0]
    
    # Bounds: -1 ≤ d[i] ≤ 1, t unbounded below
    bounds = [(-1.0, 1.0)] * n + [(None, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            d = res.x[:n]
            t_val = res.x[-1]
            return d, t_val
    except Exception:
        pass
    return None, 0.0

def lp_step(h, max_lags=30, max_iters=200, t_start=None, budget_s=1100):
    """Iterative LP-based active set stepping."""
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    
    for it in range(max_iters):
        if t_start is not None and time.perf_counter() - t_start > budget_s - 30:
            break
        
        P = profile_h(h)
        c_max = P.max()
        if abs(c_max - best_c) > 1e-10:
            best_c = c_max
        
        # Find active lags
        threshold = c_max - max(0.005, 0.1 * (c_max - P.min()))
        active = np.where(P >= threshold)[0]
        if len(active) > max_lags:
            active = active[np.argsort(P[active])[-max_lags:]]
        
        # Solve LP for direction
        d, t_val = lp_direction(h, active)
        if d is None:
            break
        
        if t_val >= -1e-10:
            # No improving direction found
            # Try expanding active set
            threshold2 = c_max - 0.001
            active2 = np.where(P >= threshold2)[0]
            if len(active2) <= max_lags:
                d, t_val = lp_direction(h, active2)
                if d is None or t_val >= -1e-10:
                    break
            else:
                break
        
        # Line search
        improved = False
        for alpha in [0.01, 0.005, 0.002, 0.001, 0.0005]:
            h_try = repair_feasible(h + alpha * d)
            s = exact_score(h_try)
            if s < best_c - 1e-14:
                h = h_try
                best_c = s
                improved = True
                break
        
        if not improved:
            # Try smaller steps
            for alpha in [0.0002, 0.0001, 0.00005]:
                h_try = repair_feasible(h + alpha * d)
                s = exact_score(h_try)
                if s < best_c - 1e-14:
                    h = h_try
                    best_c = s
                    improved = True
                    break
        if not improved:
            # Stuck - try perturbation
            break
    
    return h, best_c

def coordinate_optimal(h, i, lam):
    """Find optimal h[i] given all other coordinates fixed and Lagrange multiplier lam."""
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    
    # For each lag m, compute A_m and B_m (the constant and linear coefficient of h[i])
    # P[m] = A_m + B_m * h[i]
    # B_m = dx * (1-h[i+d_m]) [if valid] - dx * h[i-d_m] [if valid]
    # A_m = P[m] - B_m * h[i] (we can compute this from current P)
    
    P = profile_h(h)
    B = np.zeros(M)
    for m in range(M):
        d = m - (n - 1)
        coeff = 0.0
        if 0 <= i + d < n:
            coeff += dx * (1.0 - h[i + d])
        if 0 <= i - d < n:
            coeff -= dx * h[i - d]
        B[m] = coeff
    
    A = P - B * h[i]
    
    # Now minimize max_m (A[m] + B[m] * x) + lam * x over x in [0, 1]
    # with the constraint that sum(h) stays at n/2 (handled by lam)
    # Actually the lam*x term handles it

    # Wait, the Lagrangian is max_m P_m + lam*(sum - n/2)
    # = max_m (A_m + B_m * x) + lam * (sum_others + x - n/2)
    # = max_m (A_m + B_m * x) + lam * x + const
    
    # So we're minimizing: max_m (A_m + B_m * x) + lam * x
    # Let C_m = B_m, D_m = A_m
    # f(x) = max_m (D_m + C_m * x) + lam * x
    
    # This is convex. Find minimum by checking where f'(x) = 0 (if it exists)
    # f'(x) = C_{argmax} * x_dir + lam (subgradient)

    # The optimal x is where 0 is in the subdifferential
    
    # Let's find candidate x values:
    # The function changes slope at intersection points of pairs of lines

        else:

def local_search(h, t_start, budget_s):

    h = h.copy()
    best_c = exact_score(h)
    P = profile_h(h)
    
    while time.perf_counter() - t_start < budget_s - 30:

        # For each pair (i, j) with i < j, try h[i] += eps, h[j] -= eps
        # and h[i] -= eps, h[j] += eps
        best_move = None
        best_score = best_c

            for j in range(i+1, n):
                if h[i] <= 0 or h[j] <= 0:
                    continue  # Can't decrease if already 0

                for eps in [0.01, 0.02, 0.05]:

                    h[j] -= eps

                    if s < best_score:

                        best_move = (i, j, eps)
                    # Restore

                    h[j] += eps

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full"))) * dx

def profile_h(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx])
            h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def build_G(h):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    G = np.zeros((M, n))
    for j in range(n):
        for m in range(M):
            d = m - (n - 1)
            c = 0.0
            if 0 <= j + d < n:
                c += dx * (1.0 - h[j + d])
            if 0 <= j - d < n:
                c -= dx * h[j - d]
            G[m, j] = c
    return G

def vectorized_local_search(h, t_start, budget_s, margin=40, eps_list=None):

            new_max_minus = new_P_minus.max(axis=0)

            feas_minus = (h[i_idx] >= eps - 1e-12) & (h[j_idx] + eps <= 1.0 + 1e-12)

def lp_direction(h, active_lags):
    n = len(h)
    dx = 2.0 / n
    G = build_G(h)
    G_act = G[active_lags, :]  # (K, n)
    K = len(active_lags)
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    A_ub = np.zeros((K, n + 1))
    A_ub[:, :n] = G_act
    A_ub[:, -1] = -1.0
    b_ub = np.zeros(K)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    bounds = [(-1.0, 1.0)] * n + [(None, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            return res.x[:n], res.x[-1]
    except Exception:
        pass
    return None, 0.0

def coordinate_descent_fast(h, t_start, budget_s, margin=50):
    h = h.copy()
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    best_c = exact_score(h)
    P = profile_h(h)
    
    for sweep in range(50):
        if time.perf_counter() - t_start > budget_s - margin:
            break
        improved_any = False
        for i in range(n):
            if time.perf_counter() - t_start > budget_s - margin:
                break
            # Compute B_m for this coordinate
            B = np.zeros(M)
            for m in range(M):
                d = m - (n - 1)
                c = 0.0
                if 0 <= i + d < n:
                    c += dx * (1.0 - h[i + d])
                if 0 <= i - d < n:
                    c -= dx * h[i - d]
                B[m] = c
            
            A = P - B * h[i]
            
            # Find optimal x: minimize max_m(A[m] + B[m]*x) over x in [0,1]
            # subject to sum constraint (handled by redistribution)
            # Use bisection on subgradient
            lo, hi = 0.0, 1.0
            for _ in range(40):
                mid = (lo + hi) / 2.0
                vals = A + B * mid
                mx = vals.max()
                act = np.where(np.abs(vals - mx) < 1e-10)[0]
                sg_lo = B[act].min()
                sg_hi = B[act].max()
                if sg_lo <= 1e-10 <= sg_hi:
                    break
                elif sg_hi < 1e-10:
                    lo = mid
                else:
                    hi = mid
            
            x_new = (lo + hi) / 2.0
            if abs(x_new - h[i]) > 1e-8:
                delta = x_new - h[i]
                # Redistribute: reduce other coordinates proportionally
                others = np.delete(h, i)
                if delta > 0:
                    # Need to subtract delta from others
                    room = others
                    total_room = room.sum()
                    if total_room > 1e-12:
                        frac = np.minimum(room, delta / total_room * room)
                        h_new = h.copy()
                        h_new[i] = x_new
                        h_new[[j for j in range(n) if j != i]] -= frac
                        h_new = repair_feasible(h_new)
                        s = exact_score(h_new)
                        if s < best_c:
                            h = h_new
                            P = profile_h(h)
                            best_c = s
                            improved_any = True
                else:
                    # Need to add -delta to others
                    room = 1.0 - others
                    total_room = room.sum()
                    if total_room > 1e-12:
                        frac = np.minimum(room, (-delta) / total_room * room)
                        h_new = h.copy()
                        h_new[i] = x_new
                        h_new[[j for j in range(n) if j != i]] += frac
                        h_new = repair_feasible(h_new)
                        s = exact_score(h_new)
                        if s < best_c:
                            h = h_new
                            P = profile_h(h)
                            best_c = s
                            improved_any = True
        if not improved_any:
            break
    return h, best_c

def slsqp_polish(h0, maxiter=300, active_k=15, tol=1e-11):
    n = len(h0)
    dx = 2.0 / n
    M = 2 * n - 1
    P = profile_h(h0)
    c0 = P.max()
    
    thresh = c0 - max(0.004, 0.06 * (c0 - np.median(P)))
    active = np.where(P >= thresh)[0]
    if len(active) == 0:
        active = np.array([np.argmax(P)])
    if len(active) > active_k:
        top_idx = np.argsort(P[active])[-active_k:]
        active = active[top_idx]
    K = len(active)
    
    def obj(x):
        return x[-1]
    
    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    
    def eq_fun(x):
        return x[:-1].sum() - n / 2.0
    
    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g
    
    def ineq_fun(x):
        h = x[:-1]
        t = x[-1]
        prof = profile_h(h)
        return t - prof[active]
    
    def ineq_jac(x):
        h = x[:-1]
        G = build_G(h)
        J = np.zeros((K, n + 1))
        J[:, :-1] = -G[active, :]
        J[:, -1] = 1.0
        return J
    
    x0 = np.concatenate([h0, [c0 + 1e-4]])
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 1.0)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': tol})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = repair_feasible(res.x[:-1])
            c_new = exact_score(h_new)
            if c_new <= c0 + 1e-9:
                return h_new
    except Exception:
        pass
    return None

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    return repair_feasible(np.clip(h_new, 0.0, 1.0))

def symmetrize(h):
    return (h + h[::-1].copy()) / 2.0
