            idx = int(np.argmin(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def raw_profile(h):
    return np.correlate(h, 1 - h, mode='full')

def compute_gradients(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        p = n - 1 + d
        # For all j: ∂P[p]/∂h[j] = (1-h[j-d])*1_{valid} - h[j+d]*1_{valid}
        # Vectorized:
        # First term: (1-h[j-d]) for valid j-d
        # Second term: -h[j+d] for valid j+d
        # First term: j ranges from max(0,d) to min(n-1, n-1+d)
        if d >= 0:
            # j-d valid means j >= d, so j from d to n-1
            first = np.zeros(n)
            first[d:] = 1.0 - h[:n-d]  # h[j-d] for j=d..n-1
            # j+d valid means j <= n-1-d, so j from 0 to n-1-d
            second = np.zeros(n)
            second[:n-d] = -h[d:]  # -h[j+d] for j=0..n-1-d
        else:
            ad = -d
            # j-d = j+ad valid means j+ad < n, so j from 0 to n-1-ad
            first = np.zeros(n)
            first[:n-ad] = 1.0 - h[ad:]  # h[j-d] = h[j+ad] for j=0..n-1-ad
            # j+d = j-ad valid means j >= ad, so j from ad to n-1
            second = np.zeros(n)
            second[ad:] = -h[:n-ad]  # -h[j+d] = -h[j-ad] for j=ad..n-1
        G[p, :] = first + second
    return G

def line_search_quad(R, B, C, h, u):
    """Exact line search using quadratic model.
    R[p] = current profile values
    B[p] = directional derivatives
    C[p] = second-order coefficients
    Returns (best_alpha, new_score).
    """
    n = len(h)
    dx = 2.0 / n
    
    # Feasible alpha range
    pos_limits = [(1.0 - h[i]) / u[i] for i in range(n) if u[i] > 1e-15]
    alpha_max_pos = min(pos_limits) * 0.99 if pos_limits else 1.0
    alpha_max_pos = max(0.0, alpha_max_pos)
    
    neg_limits = [h[i] / (-u[i]) for i in range(n) if u[i] < -1e-15]
    alpha_max_neg = min(neg_limits) * 0.99 if neg_limits else 1.0
    alpha_max_neg = max(0.0, alpha_max_neg)
    
    c0 = float(np.max(R) * dx)
    best_score = c0
    best_alpha = 0.0
    
    n_samples = 200
    
    # Positive direction
    if alpha_max_pos > 1e-12:
        alphas = np.linspace(0.0, alpha_max_pos, n_samples)
        # scores[ai] = max_p(R[p] + alphas[ai]*B[p] + alphas[ai]^2*C[p]) * dx
        M = R[None, :] + alphas[:, None] * B[None, :] + (alphas[:, None])**2 * C[None, :]
        scores = M.max(axis=1) * dx
        idx = int(np.argmin(scores))
        if scores[idx] < best_score - 1e-15:
            # Refine
            lo = alphas[max(0, idx-1)]
            hi = alphas[min(n_samples-1, idx+1)]
            fine_alphas = np.linspace(lo, hi, 100)
            M2 = R[None, :] + fine_alphas[:, None] * B[None, :] + (fine_alphas[:, None])**2 * C[None, :]
            scores2 = M2.max(axis=1) * dx
            idx2 = int(np.argmin(scores2))
            best_score = float(scores2[idx2])
            best_alpha = float(fine_alphas[idx2])
    
    # Negative direction
    if alpha_max_neg > 1e-12:
        alphas = np.linspace(-alpha_max_neg, 0.0, n_samples)
        M = R[None, :] + alphas[:, None] * B[None, :] + (alphas[:, None])**2 * C[None, :]
        scores = M.max(axis=1) * dx
        idx = int(np.argmin(scores))
        if scores[idx] < best_score - 1e-15:
            lo = alphas[max(0, idx-1)]
            hi = alphas[min(n_samples-1, idx+1)]
            fine_alphas = np.linspace(lo, hi, 100)
            M2 = R[None, :] + fine_alphas[:, None] * B[None, :] + (fine_alphas[:, None])**2 * C[None, :]
            scores2 = M2.max(axis=1) * dx
            idx2 = int(np.argmin(scores2))
            if scores2[idx2] < best_score:
                best_score = float(scores2[idx2])
                best_alpha = float(fine_alphas[idx2])
    
    return best_alpha, best_score

def compute_C_from_u(u):
    """Compute C[p] = ½ uᵀ H_p u for all p."""
    n = len(u)
    C = np.zeros(2 * n - 1)
    for ad in range(n):
        if ad == 0:
            c_val = -np.sum(u * u)
        else:
            c_val = -np.sum(u[:n-ad] * u[ad:])
        C[n-1-ad] = c_val
        if ad > 0:
            C[n-1+ad] = c_val
    return C

def subgradient_direction(h, G, R, tau=0.005):
    """Compute smoothed subgradient of max(R) using log-sum-exp."""
    P_max = R.max()
    weights = np.exp((R - P_max) / tau)
    weights /= weights.sum()
    g = weights @ G  # shape (n,)
    # Project onto sum=0
    g -= g.mean()
    return g

def interpolate_h(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    fx = np.concatenate([[0.0], xs, [2.0]])
    fh = np.concatenate([[h[0]], h, [h[-1]]])
    hn = np.interp(xt, fx, fh)
    return repair_feasible(np.clip(hn, 0, 1))

        if elapsed > max_time:
            break
        
        # Compute subgradient direction
        g = subgradient_direction(h, G, R, tau)
        
        # Apply momentum
        momentum = mu * momentum + (1 - mu) * g
        mu = min(mu * 1.001, 0.95)  # slowly increase momentum

            break
