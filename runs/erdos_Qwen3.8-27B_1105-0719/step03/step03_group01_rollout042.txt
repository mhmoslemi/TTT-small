def project_feasible(x, target_sum):

import numpy as np
import time


def c5_of_h(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = mask.sum()
        if cnt > 0:
            h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)


def compute_AB(i, j, h, n):
    offset = n - 1
    A = np.zeros(2 * n - 1)
    B = np.zeros(2 * n - 1)

    s1lo = max(-(n - 1), -i)
    s1hi = min(n - 1, n - 1 - i)
    if s1lo <= s1hi:
        A[s1lo + offset:s1hi + 1 + offset] += (1.0 - h[i+s1lo:i+s1hi+1])

    s2lo = max(-(n - 1), i - n + 1)
    s2hi = min(n - 1, i)
    if s2lo <= s2hi:
        A[s2lo + offset:s2hi + 1 + offset] -= h[i-s2hi:i-s2lo+1][::-1]

    s3lo = max(-(n - 1), -j)
    s3hi = min(n - 1, n - 1 - j)
    if s3lo <= s3hi:
        A[s3lo + offset:s3hi + 1 + offset] -= (1.0 - h[j+s3lo:j+s3hi+1])

    s4lo = max(-(n - 1), j - n + 1)
    s4hi = min(n - 1, j)
    if s4lo <= s4hi:
        A[s4lo + offset:s4hi + 1 + offset] += h[j-s4hi:j-s4lo+1][::-1]

    s1 = j - i
    if -(n-1) <= s1 <= n-1 and 0 <= i + s1 < n:
        B[s1 + offset] += 1.0
    s2 = i - j
    if -(n-1) <= s2 <= n-1 and 0 <= j + s2 < n:
        B[s2 + offset] += 1.0

    return A, B


def f_eval(corr, t, A, B):
    if B.any():
        return np.max(corr + t * A + t * t * B)
    return np.max(corr + t * A)


def golden_search(corr, A, B, lo, hi, max_iter=30):
    if hi - lo < 1e-14:
        return 0.0
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = lo, hi
    c_pt = b - gr * (b - a)
    d_pt = a + gr * (b - a)
    fc = f_eval(corr, c_pt, A, B)
    fd = f_eval(corr, d_pt, A, B)
    for _ in range(max_iter):
        if b - a < 1e-13:
            break
        if fc < fd:
            b = d_pt
            d_pt = c_pt
            fd = fc
            c_pt = b - gr * (b - a)
            fc = f_eval(corr, c_pt, A, B)
        else:
            a = c_pt
            c_pt = d_pt
            fc = fd
            d_pt = a + gr * (b - a)
            fd = f_eval(corr, d_pt, A, B)
    return (a + b) / 2.0


def try_pair(i, j, h, corr, n, tol):
    lo = max(-h[i], h[j] - 1.0)
    hi = min(1.0 - h[i], h[j])
    if hi - lo < 1e-13:
        return 0.0, False

    A, B = compute_AB(i, j, h, n)
    f0 = np.max(corr)

    f_lo = f_eval(corr, lo, A, B)
    f_hi = f_eval(corr, hi, A, B)

    if f_lo >= f0 - tol and f_hi >= f0 - tol:
        mid = (lo + hi) * 0.5
        f_mid = f_eval(corr, mid, A, B)
        if f_mid >= f0 - tol:
            return 0.0, False

    if f_lo < f0 - tol and f_hi >= f0 - tol:
        t_opt = golden_search(corr, A, B, lo, 0.0)
    elif f_hi < f0 - tol and f_lo >= f0 - tol:
        t_opt = golden_search(corr, A, B, 0.0, hi)
    else:
        t_opt = golden_search(corr, A, B, lo, hi)

    if abs(t_opt) < 1e-13:
        return 0.0, False

    f_new = f_eval(corr, t_opt, A, B)
    if f_new < f0 - tol:
        corr += t_opt * A + t_opt * t_opt * B
        h[i] += t_opt
        h[j] -= t_opt
        return t_opt, True
    return 0.0, False


def cd_pass(h, corr, n, pairs, rng, deadline, tol=1e-11):
    improved_count = 0
    for idx, (i, j) in enumerate(pairs):
        if time.time() > deadline:
            break
        if idx % 500 == 0 and idx > 0:
            if time.time() > deadline:
                break
        _, improved = try_pair(i, j, h, corr, n, tol)
        if improved:
            improved_count += 1
    return improved_count
