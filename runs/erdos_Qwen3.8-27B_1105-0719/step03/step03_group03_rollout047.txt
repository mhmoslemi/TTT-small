        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h


def interp_h(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    fx = np.concatenate([[0.0], xs, [2.0]])
    fh = np.concatenate([[h[0]], h, [h[-1]]])
    hn = np.interp(xt, fx, fh)
    return repair_feasible(np.clip(hn, 0, 1))


def compute_alpha(h, j, k, n):
    """Compute alpha[m] = dc[m]/dh[j] - dc[m]/dh[k] for pair (j,k)."""
    center = n - 1
    m_all = np.arange(2 * n - 1)
    alpha = np.zeros(2 * n - 1)

    qj = j + m_all - center
    mask_fj = (qj >= 0) & (qj < n)
    pj = j - m_all + center
    mask_sj = (pj >= 0) & (pj < n)

    qk = k + m_all - center
    mask_fk = (qk >= 0) & (qk < n)
    pk = k - m_all + center
    mask_sk = (pk >= 0) & (pk < n)

    alpha[mask_fj] += (1.0 - h[qj[mask_fj]])
    alpha[mask_sj] += (-h[pj[mask_sj]])
    alpha[mask_fk] -= (1.0 - h[qk[mask_fk]])
    alpha[mask_sk] -= (-h[pk[mask_sk]])

    alpha[center] = 0.0
    return alpha


def pair_transfer_eval(h, profile, alpha, j, k, n):
    """Find optimal delta for pair (j,k). Returns (delta, f_value)."""
    center = n - 1
    delta_lo = max(-h[j], h[k] - 1.0)
    delta_hi = min(1.0 - h[j], h[k])

    if delta_hi - delta_lo < 1e-13:
        return 0.0, float(np.max(profile))

    beta = 2.0 * (h[k] - h[j])
    c_center = profile[center]

    cands = [delta_lo, delta_hi, 0.0]
    dv = beta / 4.0
    if delta_lo <= dv <= delta_hi:
        cands.append(dv)

    # Quadratic-line intersections
    for m in range(2 * n - 1):
        if m == center:
            continue
        b2 = alpha[m] - beta
        c2 = profile[m] - c_center
        disc = b2 * b2 - 8.0 * c2
        if disc >= 0:
            sq = np.sqrt(disc)
            d1 = (-b2 + sq) / 4.0
            d2 = (-b2 - sq) / 4.0
            if delta_lo <= d1 <= delta_hi:
                cands.append(d1)
            if delta_lo <= d2 <= delta_hi:
                cands.append(d2)

    # Grid
    span = delta_hi - delta_lo
    for i in range(1, 20):
        cands.append(delta_lo + span * i / 20.0)

    cands = np.array(cands)
    cands = cands[(cands >= delta_lo - 1e-12) & (cands <= delta_hi + 1e-12)]
    if len(cands) == 0:
        return 0.0, float(np.max(profile))
    cands = np.clip(cands, delta_lo, delta_hi)

    C = len(cands)
    g_vals = profile[None, :] + alpha[None, :] * cands[:, None]
    g_vals[:, center] = -np.inf
    g_max = np.max(g_vals, axis=1)
    q_vals = c_center + beta * cands - 2.0 * cands ** 2
    f_vals = np.maximum(g_max, q_vals)

    best_idx = np.argmin(f_vals)
    return float(cands[best_idx]), float(f_vals[best_idx])


def pair_descent(h0, max_rounds=3000, tol=1e-11, budget_fn=None, rng=None):
    """Exact pair-transfer coordinate descent."""
    n = len(h0)
    h = h0.copy()
    profile = compute_profile(h)
    best_c = float(np.max(profile))
    n_pairs_total = n * (n - 1) // 2

    for rnd in range(max_rounds):
        if budget_fn is not None and budget_fn() < 15:
            break

        best_gain = 0.0
        best_pair = None
        best_delta = 0.0

        if n <= 120:
            for j in range(n):
                for k in range(j + 1, n):
                    if h[k] < 1e-14 and h[j] < 1e-14:
                        continue
                    alpha = compute_alpha(h, j, k, n)
                    delta, f_val = pair_transfer_eval(h, profile, alpha, j, k, n)
                    gain = best_c - f_val
                    if gain > best_gain:
                        best_gain = gain
                        best_pair = (j, k)
                        best_delta = delta
        else:
            # Random subset for large n
            n_try = min(n_pairs_total, 3000)
            js = np.random.randint(0, n, size=n_try)
            ks = np.random.randint(0, n, size=n_try)
            for idx in range(n_try):
                j, k = int(js[idx]), int(ks[idx])
                if j == k:
                    continue
                alpha = compute_alpha(h, j, k, n)
                delta, f_val = pair_transfer_eval(h, profile, alpha, j, k, n)
                gain = best_c - f_val
                if gain > best_gain:
                    best_gain = gain
                    best_pair = (j, k)
                    best_delta = delta

        if best_pair is None or best_gain < tol:
            break

        j, k = best_pair
        h[j] += best_delta
        h[k] -= best_delta
        h = np.clip(h, 0.0, 1.0)
        profile = compute_profile(h)
        new_c = float(np.max(profile))

        if new_c < best_c - 1e-16:
            best_c = new_c
        else:
            break

    return h, best_c


def smooth_min_correction(h, profile, n, steps=50):
    """Subgradient-style descent on the max, focusing on active constraints."""
    center = n - 1
    h = h.copy()
    best_c = float(np.max(profile))

    for _ in range(steps):
        active = np.where(profile >= best_c - 0.005)[0]
        if len(active) == 0:
            active = np.array([np.argmax(profile)])
        if len(active) > 20:
            active = active[np.argsort(profile[active])[-20:]]

        # Subgradient of max: uniform over active
        G = np.zeros(n)
        for m in active:
            adj = 1.0 / len(active)
            t = np.arange(n)
            q = t + m - center
            mask_q = (q >= 0) & (q < n)
            p = t - m + center
            mask_p = (p >= 0) & (p < n)
            dc = np.zeros(n)
            dc[mask_q] += (1.0 - h[q[mask_q]]) * adj
            dc[mask_p] += (-h[p[mask_p]]) * adj
            G += dc

        G -= G.mean()
        gn = np.linalg.norm(G)
        if gn < 1e-14:
            break

        # Line search
        for s in [0.005, 0.002, 0.001, 0.0005, 0.0001]:
            h_try = repair_feasible(h - s * G / gn)
            c_try = exact_score(h_try)
            if c_try < best_c - 1e-13:
                h = h_try
                best_c = c_try
                profile = compute_profile(h)
                break

    return h, best_c
