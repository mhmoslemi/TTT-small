**Key idea: Use a "soft max" approximation of the max, optimize with projected gradient, then refine with active set.**

The soft max: f_β(h) = (1/β) * log(sum_m exp(β * g_m(h)))

where g_m(h) is the m-th entry of the profile. This is smooth and we can use gradient-based optimization.

The gradient of f_β is: sum_m [exp(β*g_m) / sum_exp] * grad(g_m) = weighted average of active gradients.

This is actually what the active set method is approximating! But the soft max gives a smooth objective that we can optimize more efficiently.

Let me implement this:

```python
def soft_max_objective(h, beta):
    n = len(h)
    profile = compute_profile(h)
    # Shift for numerical stability
    shifted = beta * (profile - profile.max())
    exp_vals = np.exp(shifted)
    log_sum = np.log(exp_vals.sum())
    return (profile.max() + log_sum / beta)

def soft_max_gradient(h, beta):
    n = len(h)
    profile = compute_profile(h)
    shifted = beta * (profile - profile.max())
    exp_vals = np.exp(shifted)
    weights = exp_vals / exp_vals.sum()
    g = np.zeros(n)
    for m in range(len(profile)):
        if weights[m] > 1e-10:
            g += weights[m] * grad_row(h, m, n)
    return g

def project_feasible(x):
    """Project x onto {0 <= h <= 1, sum(h) = n/2}"""
    n = len(x)
    target = n / 2.0
    # Bisection on lambda
    lo = x.min() - 1.0
    hi = x.max() + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        h = np.clip(x - mid, 0.0, 1.0)
        s = h.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
        if abs(s - target) < 1e-15:
            break
    h = np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)
    return h
```

    h = project_feasible(h.copy())
    best_c = exact_score(h)
    best_h = h.copy()

    for it in range(max_iters):

def active_set_refine(h, max_iters=200, tol=0.002):
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    momentum_vec = np.zeros(n)
    for it in range(max_iters):
        profile = compute_profile(h)
        c = float(profile.max())
        thresh = c - tol
        active = np.where(profile >= thresh)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(profile))])
        if len(active) > 30:
            top = np.argsort(profile[active])[-30:]
            active = active[top]
        g = np.zeros(n)
        for m in active:
            g += grad_row(h, m, n)
        g /= len(active)
        g -= g.mean()  # Remove mean to preserve sum constraint
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-14:
            break
        # BB step size
        direction = -g / gnorm
        # Line search
        best_alpha = 0.0
        best_val = c
        best_h = h
        a, b = 0.0, min(1.0, 0.5 / gnorm)
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        for _ in range(20):
            c1 = b - gr * (b - a)
            c2 = a + gr * (b - a)
            h1 = project_feasible(h + c1 * direction)
            h2 = project_feasible(h + c2 * direction)
            f1 = exact_score(h1)
            f2 = exact_score(h2)
            if f1 < best_val:
                best_val = f1
                best_alpha = c1
                best_h = h1.copy()
            if f2 < best_val:
                best_val = f2

import numpy as np
from scipy.optimize import minimize
import time

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(-(1.0 - h))
        remaining = diff
        for i in range(n):
            idx = order[i]
            cap = 1.0 - h[idx]
            if cap <= 1e-15:
                continue
            add = min(remaining, cap)
            h[idx] += add
            remaining -= add
            if remaining <= 1e-14:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            if h[idx] <= 1e-15:
                continue
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-14:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] = float(np.clip(h[idx] + diff2, 0.0, 1.0))
    return h

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def grad_row(h, m, n):
    dx = 2.0 / n
    d = m - (n - 1)
    if d == 0:
        return dx * (1.0 - 2.0 * h)
    elif d > 0:
        g = np.zeros(n)
        g[d:] += dx * (1.0 - h[:-d])
        g[:-d] -= dx * h[d:]
        return g
    else:
        p = -d
        g = np.zeros(n)
        g[:-p] += dx * (1.0 - h[p:])
        g[p:] -= dx * h[:-p]
        return g

def interp_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    edges_old = np.linspace(0.0, 2.0, n + 1)
    edges_new = np.linspace(0.0, 2.0, new_n + 1)
    h_new = np.zeros(new_n)
    for k in range(new_n):
        a, b = edges_new[k], edges_new[k + 1]
        total = 0.0
        for j in range(n):
            lo = max(a, edges_old[j])
            hi = min(b, edges_old[j + 1])
            if hi > lo:
                total += h[j] * (hi - lo)
        h_new[k] = total / (b - a)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

    n = len(x)
    target = n / 2.0
    lo = x.min() - 1.0
    hi = x.max() + 1.0
    for _ in range(80):

        s = h.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
        if abs(s - target) < 1e-15:
            break
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    return h

def soft_max_val(profile, beta):
    shifted = beta * profile
    shifted -= shifted.max()
    return profile.max() + np.log(np.exp(shifted).sum()) / beta

def soft_max_grad(h, beta):
    n = len(h)
    profile = compute_profile(h)
    shifted = beta * profile
    shifted -= shifted.max()
    exp_vals = np.exp(shifted)
    weights = exp_vals / exp_vals.sum()
    g = np.zeros(n)
    for m in range(len(profile)):
        w = weights[m]
        if w > 1e-10:
            g += w * grad_row(h, m, n)
    return g

    n = len(h)
    h = h.copy()

        tl = min(20.0, time_left() - 5)
        h_new, c_new = active_set_refine(best_h, max_iters=500, tol=0.0003)
        if c_new < best_c:
            best_h, best_c = h_new, c_new

    # Phase 8: Try 1-h symmetry as new start
    if time_left() > 40:
        h_flip = 1.0 - best_h
        h_flip = repair_feasible(h_flip)
        tl = min(20.0, time_left() - 15)
        h_new, c_new = fista_optimize(h_flip, 100, max_iters=3000, time_limit=tl)
        if time_left() > 20:
            h_new, c_new = active_set_refine(h_new, max_iters=200, tol=0.001)
        if c_new < best_c:
            best_h, best_c = h_new, c_new

    # Phase 9: Final dimension check - try smaller
    if time_left() > 50:
        n_b = len(best_h)
        for small_n in [n_b - 6, n_b - 10, n_b + 6, n_b + 10]:
            if time_left() < 25:
                break
            if small_n < 40 or small_n > 130 or small_n % 2 != 0:
                continue
            h_interp = interp_to_n(best_h, small_n)
            tl = min(20.0, time_left() - 15)
            h_new, c_new = fista_optimize(h_interp, 80, max_iters=3000, time_limit=tl)
            if time_left() > 15:
                h_new, c_new = active_set_refine(h_new, max_iters=150, tol=0.002)
            if c_new < best_c:
                best_h, best_c = h_new, c_new

    # Final polish
    if time_left() > 10:
        h_new, c_new = active_set_refine(best_h, max_iters=200, tol=0.0002)
        if c_new < best_c:
            best_h, best_c = h_new, c_new

    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)

    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            final_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 2:
            final_c = float(eval_result[0])
    except Exception:
        pass

    return best_h, final_c, len(best_h)
