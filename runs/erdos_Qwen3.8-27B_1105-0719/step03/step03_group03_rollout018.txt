import numpy as np
from scipy.optimize import minimize

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))


def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx


def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx] + 1e-15)
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx] + 1e-15)
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] = np.clip(h[idx] + diff2, 0.0, 1.0)
    return h


def build_grad(h):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    G = np.zeros((M, n))
    center = n - 1
    G[center, :] = dx * (1.0 - 2.0 * h)
    for d in range(1, n):
        m_pos = center + d
        idx1 = np.arange(n - d)
        G[m_pos, idx1] += -dx * h[idx1 + d]
        G[m_pos, idx1 + d] += dx * (1.0 - h[idx1])
        m_neg = center - d
        G[m_neg, idx1] += dx * (1.0 - h[idx1 + d])
        G[m_neg, idx1 + d] += -dx * h[idx1]
    return G


def lse_direction(h, profile, eta, G):
    """Compute LSE soft gradient direction."""
    c_max = np.max(profile)
    z = (profile - c_max) / eta
    z_max = np.max(z)
    exp_z = np.exp(z - z_max)
    w = exp_z / np.sum(exp_z)
    grad = G.T @ w
    # Remove mean to maintain feasibility (sum constraint)
    grad -= grad.mean()
    return grad, w


def backtracking_line_search(h, grad, profile, alpha0, max_iter=50):
    """Backtracking line search for LSE objective."""
    n = len(h)
    c_curr = np.max(profile)
    for i in range(max_iter):
        alpha = alpha0 * (0.5 ** i)
        if alpha < 1e-8:
            break
        h_try = repair_feasible(h - alpha * grad)
        c_try = exact_score(h_try)
        if c_try < c_curr - 1e-12:
            return h_try, c_try, alpha
    return None, None, 0


def trust_region_qp(h, profile, G, active, tau, mu):
    """Solve trust-region QP for minimax problem."""
    if not HAS_CVXPY:
        return None
    n = len(h)
    target = n / 2.0
    G_a = G[active, :]
    
    # Linearize: P_a(h_new) ≈ P_a(h) + G_a @ (h_new - h)
    # Want: P_a(h) + G_a @ (h_new - h) <= t for all a in active
    # i.e., G_a @ h_new - t <= -P_a(h) + G_a @ h
    
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    cons = [
        h_var >= 0,
        h_var <= 1,
        cp.sum(h_var) == target,
        h_var >= h - tau,
        h_var <= h + tau,
        G_a @ h_var - t_var <= -profile[active] + G_a @ h
    ]
    obj = cp.Minimize(t_var + mu * cp.sum(cp.square(h_var - h)))
    prob = cp.Problem(obj, cons)
    
    for solver in [cp.ECOS, cp.SCS, cp.CBC]:
        try:
            prob.solve(solver=solver, max_iters=10000, abstol=1e-10)
            break
        except Exception:
            continue
    
    if h_var.value is None:
        return None
    h_sol = np.array(h_var.value).flatten()
    if not np.isfinite(h_sol).all():
        return None
    return h_sol


def interp(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    fx = np.concatenate([[0.0], xs, [2.0]])
    fh = np.concatenate([[h[0]], h, [h[-1]]])
    hn = np.interp(xt, fx, fh)
    return repair_feasible(np.clip(hn, 0, 1))


def slsqp_refine(h0, maxiter, tol):
    n = len(h0)
    dx = 2.0 / n
    M = 2 * n - 1
    
    def obj(x):
        return x[-1]
    
    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    
    def eq_fun(x):
        return x[:-1].sum() - n / 2.0
    
    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g
    
    def ineq_fun(x):
        h = x[:-1]
        return x[-1] - np.correlate(h, 1 - h, mode="full") * dx
    
    def ineq_jac(x):
        h = x[:-1]
        G = build_grad(h)
        J = np.zeros((M, n + 1))
        J[:, :-1] = -G
        J[:, -1] = 1.0
        return J
    
    t0 = exact_score(h0) + 1e-4
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, t0 + 2.0)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': tol})
        if res.x is not None and np.isfinite(res.x).all():
            hs = repair_feasible(res.x[:-1])
            if np.isfinite(exact_score(hs)):
                return hs
    except Exception:
        pass
    return None

def build_grad(h):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    G = np.zeros((M, n))
    center = n - 1
    G[center, :] = dx * (1.0 - 2.0 * h)

        idx1 = np.arange(n - d)

        G[m_pos, idx1 + d] += dx * (1.0 - h[idx1])
        m_neg = center - d
        G[m_neg, idx1] += dx * (1.0 - h[idx1 + d])
        G[m_neg, idx1 + d] += -dx * h[idx1]
    return G

    n = len(h)
    if n == new_n:
        return h.copy()

        xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
        fx = np.concatenate([[0.0], xs, [2.0]])
        fh = np.concatenate([[h[0]], h, [h[-1]]])
        hn = np.interp(xt, fx, fh)
        return repair_feasible(np.clip(hn, 0, 1))
    else:
        groups = np.array_split(np.arange(n), new_n)
        hn = np.array([h[g].mean() for g in groups])

def slsqp_refine(h0, maxiter, tol):
    n = len(h0)
    dx = 2.0 / n
    M = 2 * n - 1

    def slp_obj(x):
        return x[-1]

    def slp_obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g

    def slp_eq_fun(x):
        return x[:-1].sum() - n / 2.0

    def slp_eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g

    def slp_ineq_fun(x):
        hh = x[:-1]
        return x[-1] - np.correlate(hh, 1 - hh, mode="full") * dx

    def slp_ineq_jac(x):
        hh = x[:-1]
        G = build_grad(hh)
        J = np.zeros((M, n + 1))
        J[:, :-1] = -G
        J[:, -1] = 1.0
        return J

    t0 = exact_score(h0) + 1e-4
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, t0 + 2.0)]
    cons = [
        {'type': 'eq', 'fun': slp_eq_fun, 'jac': slp_eq_jac},
        {'type': 'ineq', 'fun': slp_ineq_fun, 'jac': slp_ineq_jac}
    ]
    try:
        res = minimize(slp_obj, x0, method='SLSQP', jac=slp_obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': tol})
        if res.x is not None and np.isfinite(res.x).all():
            hs = repair_feasible(res.x[:-1])
            if np.isfinite(exact_score(hs)):
                return hs
    except Exception:
        pass
    return None

        if bc < best_c - 1e-14:
            best_h, best_c = bh, bc

    # Phase 10: Final SLSQP
    if time_left() > 30:
        hs = slsqp_refine(best_h, maxiter=400, tol=1e-14)
        if hs is not None:
            cs = exact_score(hs)
            if cs < best_c - 1e-14:
                best_h, best_c = hs, cs

    # Final validation
    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)

    try:
        er = evaluate_erdos_solution(best_h)
        if isinstance(er, (float, int)):
            final_c = min(final_c, float(er))
        elif isinstance(er, tuple) and len(er) >= 2:
            final_c = min(final_c, float(er[0]))
    except Exception:
        pass

    return best_h, final_c, len(best_h)
