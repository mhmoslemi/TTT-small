    n = len(h)
    dx = 2.0 / n

    h = np.clip(h.astype(np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, 1.0 - h[idx])
            h[idx] += add
            remaining -= add

            rem = min(remaining, h[idx])

        res = minimize(obj_fn, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-14})
        if res.success or res.x is not None:
            h_sol = repair(res.x[:-1])
            c_sol = exact_score(h_sol)
            if c_sol < c0 - 1e-14:
                return h_sol, c_sol
    except Exception:
        pass
    return h, c0


def make_smooth_start(n, rng, freq_scale=1.0):
    x = np.linspace(0, 2, n)
    h = np.zeros(n)
    for k in range(1, 5):
        amp = 0.15 / k
        phase = rng.uniform(0, 2 * np.pi)
        h += amp * np.sin(2 * np.pi * k * freq_scale * x / 2.0 + phase)
    h = 0.5 + h
    return repair(h)


def make_binary_start(n, rng):
    h = np.zeros(n)
    ones = n // 2
    # Random subset
    idx = rng.choice(n, ones, replace=False)
    h[idx] = 1.0
    # Add small noise to break symmetry
    h += rng.uniform(-0.05, 0.05, n)
    return repair(h)


def make_periodic_start(n, rng):
    x = np.linspace(0, 2, n, endpoint=False)
    period = 2.0 / rng.uniform(1.5, 4.0)
    phase = rng.uniform(0, 2 * np.pi)
    h = 0.5 + 0.3 * np.sin(2 * np.pi * (x + phase) / period)
    h = np.clip(h, 0, 1)
    return repair(h)


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s - (time.perf_counter() - t_start)

    # Load parent
    parent = None
    try:
        parent = np.array(initial_h_values, dtype=np.float64)
    except (NameError, Exception):
        pass

    if parent is None or len(parent) < 4 or not np.isfinite(parent).all():
        parent = np.full(70, 0.5)

    parent = np.clip(parent, 0.0, 1.0)
    parent = repair(parent)
    c_parent = exact_score(parent)

    best_h = parent.copy()
    best_c = c_parent
    n_best = len(best_h)

    # Quick symmetry check
    h_sym = repair(symmetrize(parent))
    c_sym = exact_score(h_sym)
    if c_sym < best_c - 1e-12:
        best_h = h_sym
        best_c = c_sym

    # Phase 1: SQP at parent dimension
    if time_left() > 120:
        h_opt, c_opt = sqp_optimize(best_h, t_start, budget_s, max_iters=250)
        if c_opt < best_c - 1e-14:
            best_h, best_c = h_opt, c_opt
        # Symmetrize check
        h_s = repair(symmetrize(best_h))
        c_s = exact_score(h_s)
        if c_s < best_c - 1e-12:
            best_h, best_c = h_s, c_s

    # Phase 2: SLSQP polish
    if time_left() > 80:
        h_pol, c_pol = slsqp_polish(best_h, t_start, budget_s, max_iter=150)
        if c_pol < best_c - 1e-14:
            best_h, best_c = h_pol, c_pol
        h_s = repair(symmetrize(best_h))
        c_s = exact_score(h_s)
        if c_s < best_c - 1e-12:
            best_h, best_c = h_s, c_s
        # More SQP after SLSQP
        if time_left() > 50:
            h_opt2, c_opt2 = sqp_optimize(best_h, t_start, budget_s, max_iters=150)
            if c_opt2 < best_c - 1e-14:
                best_h, best_c = h_opt2, c_opt2

    # Phase 3: Try nearby dimensions
    n_base = len(best_h)
    dims = []
    for d in [n_base - 4, n_base - 2, n_base + 2, n_base + 4,
              n_base + 6, n_base - 6, n_base + 8, n_base - 8]:
        if 30 <= d <= 200 and d % 2 == 0 and d != n_base:
            dims.append(d)
    dims = sorted(set(dims))

    for new_n in dims[:8]:
        if time_left() < 60:
            break
        h_interp = interpolate_to_n(best_h, new_n)
        h_opt, c_opt = sqp_optimize(h_interp, t_start, budget_s, max_iters=120)
        if c_opt < best_c - 1e-14:
            best_h, best_c = h_opt, c_opt
            # SLSQP on new best
            if time_left() > 40:
                h_pol, c_pol = slsqp_polish(best_h, t_start, budget_s, max_iter=100)
                if c_pol < best_c - 1e-14:
                    best_h, best_c = h_pol, c_pol
        h_s = repair(symmetrize(h_opt))
        c_s = exact_score(h_s)
        if c_s < best_c - 1e-12:
            best_h, best_c = h_s, c_s

    # Phase 4: Diverse restarts at best dimension
    n_cur = len(best_h)
    starts = []
    for i in range(6):
        starts.append(make_smooth_start(n_cur, rng, freq_scale=0.5 + i * 0.5))
    for i in range(3):
        starts.append(make_binary_start(n_cur, rng))
    for i in range(3):
        starts.append(make_periodic_start(n_cur, rng))
    # Also blend with best
    for i in range(3):
        s = make_smooth_start(n_cur, rng, freq_scale=1.0 + i)
        starts.append(repair(0.6 * best_h + 0.4 * s))

    for s in starts:
        if time_left() < 40:
            break
        c_s = exact_score(s)
        if c_s > best_c + 0.05:
            continue
        h_opt, c_opt = sqp_optimize(s, t_start, budget_s, max_iters=80)
        if c_opt < best_c - 1e-14:
            best_h, best_c = h_opt, c_opt
            if time_left() > 30:
                h_pol, c_pol = slsqp_polish(best_h, t_start, budget_s, max_iter=80)
                if c_pol < best_c - 1e-14:
                    best_h, best_c = h_pol, c_pol

    # Phase 5: Final intensive optimization
    if time_left() > 30:
        h_opt, c_opt = sqp_optimize(best_h, t_start, budget_s, max_iters=200)
        if c_opt < best_c - 1e-14:
            best_h, best_c = h_opt, c_opt
        h_s = repair(symmetrize(best_h))
        c_s = exact_score(h_s)
        if c_s < best_c - 1e-12:
            best_h, best_c = h_s, c_s

    # Phase 6: Ultimate SLSQP
    if time_left() > 25:
        h_pol, c_pol = slsqp_polish(best_h, t_start, budget_s, max_iter=300)
        if c_pol < best_c - 1e-14:
            best_h, best_c = h_pol, c_pol
        h_s = repair(symmetrize(best_h))
        c_s = exact_score(h_s)
        if c_s < best_c - 1e-12:
            best_h, best_c = h_s, c_s

    # Final validation
    best_h = repair(best_h)
    final_c = exact_score(best_h)

    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            final_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 2:
            final_c = float(eval_result[0])
    except Exception:
        pass

    return best_h, final_c, len(best_h)
