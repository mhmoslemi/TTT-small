import numpy as np
from scipy.optimize import linprog, minimize
import time


def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full"))) * dx


def profile_h(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx


def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx])
            h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h


def build_gradient_matrix(h, n, dx):
    """Build full gradient matrix G of shape (2n-1, n)."""
    M = 2 * n - 1
    G = np.zeros((M, n))
    for m in range(M):
        d = m - (n - 1)
        if d == 0:
            G[m, :] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            for i in range(n - d):
                G[m, i + d] += dx * (1.0 - h[i])
                G[m, i] -= dx * h[i + d]
        else:
            dp = -d
            for j in range(n - dp):
                G[m, j + dp] += dx * (1.0 - h[j])
                G[m, j] -= dx * h[j + dp]
    return G


def slp_clip(h, max_iters=200, thresh_frac=0.0, t_start=None, budget_s=1100, margin=20):
    """SLP minimax with optional constraint clipping."""
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    h = h.copy()
    best_c = exact_score(h)

    for it in range(max_iters):
        if t_start is not None and time.perf_counter() - t_start > budget_s - margin:
            break

        P = profile_h(h)
        c_max = P.max()

        # Select constraints: all if thresh_frac==0, else near-active
        if thresh_frac > 0.0:
            thresh = c_max - thresh_frac * (c_max - P.min() + 1e-10)
            active = np.where(P >= thresh)[0]
            if len(active) < 5:
                active = np.arange(M)
        else:
            active = np.arange(M)

        n_active = len(active)
        G_full = build_gradient_matrix(h, n, dx)
        G = G_full[active, :]
        P_act = P[active]

        # LP: min t
        # s.t. G[i]·h' - t <= G[i]·h - P[i]  for i in active
        #      0 <= h' <= 1, sum(h') = n/2
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0

        A_ub = np.zeros((n_active, n + 1))
        A_ub[:, :-1] = G
        A_ub[:, -1] = -1.0
        b_ub = G @ h - P_act

        A_eq = np.zeros((1, n + 1))
        A_eq[0, :-1] = 1.0
        b_eq = np.array([n / 2.0])

        lb = np.concatenate([np.zeros(n), [0.0]])
        ub = np.concatenate([np.ones(n), [best_c + 0.05]])

        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=list(zip(lb, ub)), method='highs')
            if res.status == 0:
                h_prop = res.x[:-1]
                d = h_prop - h
                dn = np.linalg.norm(d)
                if dn < 1e-14:
                    break
                best_alpha = 0.0
                improved = False
                for alpha in [1.0, 0.8, 0.6, 0.5, 0.4, 0.3, 0.2, 0.15, 0.1, 0.05, 0.02, 0.01]:
                    h_try = repair_feasible(h + alpha * d)
                    s = exact_score(h_try)
                    if s < best_c - 1e-13:
                        best_c = s
                        h = h_try
                        best_alpha = alpha
                        improved = True
                        break
                if not improved:
                    break
            else:
                break
        except Exception:
            break

    return h, best_c


def slsqp_active(h0, maxiter=300, active_k=15, tol=1e-12):
    n = len(h0)
    dx = 2.0 / n
    M = 2 * n - 1
    P = profile_h(h0)
    c0 = P.max()

    thresh = c0 - max(0.002, 0.04 * (c0 - np.median(P) + 1e-10))
    active = np.where(P >= thresh)[0]
    if len(active) == 0:
        active = np.array([np.argmax(P)])
    if len(active) > active_k:
        top_idx = np.argsort(P[active])[-active_k:]
        active = active[top_idx]

    def obj(x):
        return x[-1]

    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g

    def eq_fun(x):
        return x[:-1].sum() - n / 2.0

    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g

    def ineq_fun(x):
        prof = profile_h(x[:-1])
        return x[-1] - prof[active]

    def ineq_jac(x):
        h = x[:-1]
        G = build_gradient_matrix(h, n, dx)
        J = np.zeros((len(active), n + 1))
        J[:, :-1] = -G[active, :]
        J[:, -1] = 1.0
        return J

    x0 = np.concatenate([h0, [c0 + 1e-5]])
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 1.0)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': tol})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = repair_feasible(res.x[:-1])
            c_new = exact_score(h_new)
            if c_new <= c0 + 1e-9:
                return h_new
    except Exception:
        pass
    return None


def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    return repair_feasible(np.clip(h_new, 0.0, 1.0))


def symmetrize(h):
    return (h + h[::-1].copy()) / 2.0


def coord_fine_tune(h, max_sweeps=8, t_start=None, budget_s=1100, margin=10):
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    for sweep in range(max_sweeps):
        if t_start is not None and time.perf_counter() - t_start > budget_s - margin:
            break
        improved = False
        for i in range(n):
            if t_start is not None and time.perf_counter() - t_start > budget_s - margin:
                break
            h_save = h[i]
            for direction in [1, -1]:
                found = False
                for frac in [0.008, 0.004, 0.002, 0.001, 0.0005]:
                    delta = direction * frac
                    h_new_val = h_save + delta
                    if h_new_val < -1e-12 or h_new_val > 1.0 + 1e-12:
                        continue
                    h[i] = h_new_val
                    h_rep = repair_feasible(h)
                    s = exact_score(h_rep)
                    if s < best_c - 1e-14:
                        h = h_rep
                        best_c = s
                        improved = True
                        found = True
                        break
                    else:
                        h[i] = h_save
                if found:
                    break
        if not improved:
            break
    return h, best_c


def make_binary_n(n, seed):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    golden = 0.6180339887
    for i in range(n // 2):
        pos = int(np.floor(((i * golden + 0.3) % 1.0) * n))
        h[pos] = 1.0
    return repair_feasible(h)
