def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx] + 1e-15)
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx] + 1e-15)
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    # Final fix
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h


def build_grad(h):
    n = len(h)
    M = 2 * n - 1
    center = n - 1
    G = np.zeros((M, n))
    # d = 0
    G[center, :] = 1.0 - 2.0 * h
    # d > 0
    for d in range(1, n):
        m_pos = center + d
        idx1 = np.arange(n - d)
        G[m_pos, idx1] += (1.0 - h[idx1 + d])
        idx2 = np.arange(d, n)
        G[m_pos, idx2] += -h[idx2 - d]
    # d < 0
    for d in range(1, n):
        m_neg = center - d
        idx1 = np.arange(n - d)
        G[m_neg, idx1] += -h[idx1 + d]
        idx2 = np.arange(d, n)
        G[m_neg, idx2] += (1.0 - h[idx2 - d])
    return G


def lse_grad_direction(h, eta):
    """Compute the LSE gradient direction."""
    n = len(h)
    dx = 2.0 / n
    profile = np.correlate(h, 1 - h, mode="full") * dx  # f values
    c_max = np.max(profile)
    # LSE weights
    z = (profile - c_max) / eta
    z_max = np.max(z)
    exp_z = np.exp(z - z_max)
    w = exp_z / np.sum(exp_z)
    # Gradient
    G = build_grad(h)
    grad = G.T @ w
    # Project onto sum=0 subspace
    grad -= grad.mean()
    return grad, c_max, w

    h = h0.copy()
    best_h = h.copy()
    best_c = exact_score(h)
    
    for eta in eta_schedule:
        if budget_fn() < reserved_time:
            break
        alpha = 0.01
        for iteration in range(max_iter_per_eta):
            if budget_fn() < reserved_time:

            grad, c_max, w = lse_grad_direction(h, eta)
            gn = np.linalg.norm(grad)
            if gn < 1e-15:
                break
            
            # Backtracking line search
            accepted = False
            for _ in range(10):
                h_try = repair_feasible(h - alpha * grad)
                c_try = exact_score(h_try)
                if c_try < best_c - 1e-14:
                    h = h_try
                    best_h = h_try.copy()
                    best_c = c_try
                    accepted = True
                    break
                elif c_try < c_max - 1e-14:
                    # Accept if better than current (even if not best)
                    h = h_try
                    accepted = True
                    break
                alpha *= 0.5
                if alpha < 1e-8:
                    break
            
            if not accepted:
                # Try smaller alpha
                for a in [0.001, 0.0005, 0.0001]:
                    h_try = repair_feasible(h - a * grad)
                    c_try = exact_score(h_try)
                    if c_try < c_max - 1e-14:
                        h = h_try
                        if c_try < best_c:
                            best_h = h_try.copy()
                            best_c = c_try
                        break
        
    return best_h, best_c

Hmm, wait. I think there's an issue with my LSE descent. The gradient of the LSE points in the direction that DECREASES the LSE. But we want to DECREASE the max (i.e., the score). Since the LSE is an upper bound on the max (L_η ≥ max for all η > 0), decreasing the LSE should generally decrease the max, but it's not guaranteed at every step.

Actually, L_η ≥ max always, and as η→0, L_η → max. So for small η, the LSE gradient is a good approximation of the steepest descent direction for the max.

But for large η, the LSE gradient might not point in a direction that decreases the max. So I should:
1. Start with moderately large η for global exploration
2. Decrease η for local refinement
3. Always use exact score for acceptance

Let me also think about the step size more carefully. The LSE gradient has magnitude roughly O(1) (since the weights sum to 1 and each row of G has entries of order 1). The function values are in [0, 1] (since h ∈ [0,1] and we're computing products). So a reasonable initial step size is O(0.01) to O(0.1).

The gradient of the LSE has components of order... let's see. Each row of G has entries that are (1-h) or -h or (1-2h), all in [-1, 1]. The weights sum to 1. So the gradient is a convex combination of rows of G, each of which has norm at most sqrt(n). So |grad| ≤ sqrt(n) ≈ 8.4 for n=70.
