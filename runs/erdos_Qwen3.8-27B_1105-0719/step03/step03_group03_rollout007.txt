def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full")) * dx)


def compute_raw_profile(h):
    return np.correlate(h, 1 - h, mode="full")


def repair_feasible(h):
    n = len(h)
    h = np.clip(np.asarray(h, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx])
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h


def build_gradient(h, n):
    M = 2 * n - 1
    dx = 2.0 / n
    G = np.zeros((M, n))
    center = n - 1
    G[center, :] = dx * (1.0 - 2.0 * h)
    for d in range(1, n):
        idx1 = np.arange(n - d)
        G[center + d, idx1] = -dx * h[idx1 + d]
        G[center + d, idx1 + d] = dx * (1.0 - h[idx1])
        G[center - d, idx1] = dx * (1.0 - h[idx1 + d])
        G[center - d, idx1 + d] = -dx * h[idx1]
    return G


def subgradient_direction(h, n):
    profile = compute_raw_profile(h)
    cm = np.max(profile)
    spread = cm - np.percentile(profile, 50)
    threshold = cm - 0.15 * (spread + 1e-10)
    active = np.where(profile >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(profile)])
    G = build_gradient(h, n)
    w = profile[active] - profile[active].min() + 1e-8
    w = w / w.sum()
    grad = G[active, :].T @ w
    grad -= grad.mean()
    gn = np.linalg.norm(grad)
    if gn > 1e-14:
        grad /= gn
    return grad


def zero_sum_random(n, rng):
    v = rng.randn(n)
    v -= v.mean()
    vn = np.max(np.abs(v))
    if vn > 1e-12:
        v /= vn
    return v


def smooth_direction(n, freq, rng):
    x = np.arange(n)
    phase = rng.rand() * 2.0 * np.pi
    v = np.sin(2.0 * np.pi * freq * x / n + phase)
    v += 0.4 * np.sin(2.0 * np.pi * (2 * freq) * x / n + 1.7 * phase)
    v += 0.2 * np.sin(2.0 * np.pi * (3 * freq) * x / n + 2.3 * phase)
    v -= v.mean()
    vn = np.max(np.abs(v))
    if vn > 1e-12:
        v /= vn
    return v


def sparse_direction(n, rng):
    k = rng.randint(2, max(3, n // 4))
    v = np.zeros(n)
    idx = rng.choice(n, size=k, replace=False)
    v[idx] = rng.randn(k)
    v -= v.mean()
    vn = np.max(np.abs(v))
    if vn > 1e-12:
        v /= vn
    return v


def antisym_direction(n, rng):
    v = rng.randn(n)
    v = v - v[::-1]
    v -= v.mean()
    vn = np.max(np.abs(v))
    if vn > 1e-12:
        v /= vn
    else:
        v = zero_sum_random(n, rng)
    return v


def quadratic_minimax(A, B, C, lo, hi, n_samples=500):
    alphas = np.linspace(lo, hi, n_samples)
    vals = A[None, :] + alphas[:, None] * B[None, :] + alphas[:, None] ** 2 * C[None, :]
    max_vals = vals.max(axis=1)
    best_idx = np.argmin(max_vals)
    best_alpha = alphas[best_idx]
    for _ in range(6):
        step = (hi - lo) / n_samples
        local_lo = max(best_alpha - step, lo)
        local_hi = min(best_alpha + step, hi)
        if local_hi - local_lo < 1e-10:
            break
        fine_alphas = np.linspace(local_lo, local_hi, 200)
        fine_vals = A[None, :] + fine_alphas[:, None] * B[None, :] + fine_alphas[:, None] ** 2 * C[None, :]
        fine_max = fine_vals.max(axis=1)
        fi = np.argmin(fine_max)
        new_alpha = fine_alphas[fi]
        if fine_max[fi] < max_vals[best_idx] - 1e-15:
            best_alpha = new_alpha
        step *= 0.2
    return best_alpha


def line_search(h, v, n_samples=500):
    n = len(h)
    v = np.asarray(v, dtype=np.float64).copy()
    vsum = v.sum()
    if abs(vsum) > 1e-10:
        v = v - vsum / n
    vmax = np.max(np.abs(v))
    if vmax < 1e-12:
        return None, None

    pos_mask = v > 1e-15
    neg_mask = v < -1e-15
    hi = np.inf
    lo = -np.inf
    if np.any(pos_mask):
        hi = np.min((1.0 - h[pos_mask]) / v[pos_mask])
    if np.any(neg_mask):
        lo = np.max(-h[neg_mask] / v[neg_mask])

    hi = min(hi, 3.0 / vmax)
    lo = max(lo, -3.0 / vmax)
    if lo >= hi - 1e-12:
        return None, None

    A = compute_raw_profile(h)
    B = np.correlate(v, 1 - h, mode="full") - np.correlate(h, v, mode="full")
    C = -np.correlate(v, v, mode="full")

    best_alpha = quadratic_minimax(A, B, C, lo, hi, n_samples)

    h_new = repair_feasible(h + best_alpha * v)
    score_new = exact_score(h_new)

    candidates = [(h_new, score_new)]
    h_half = repair_feasible(h + 0.5 * best_alpha * v)
    candidates.append((h_half, exact_score(h_half)))
    h_quarter = repair_feasible(h + 0.25 * best_alpha * v)
    candidates.append((h_quarter, exact_score(h_quarter)))

    best_i = np.argmin([c[1] for c in candidates])
    return candidates[best_i][0], candidates[best_i][1]


def coord_descent(h, n, max_sweeps=5):
    best_h = h.copy()
    best_c = exact_score(h)
    for sweep in range(max_sweeps):
        improved = False
        for i in range(n):
            v = np.zeros(n)
            v[i] = 1.0
            v[j] = -1.0
            # Actually do proper coordinate descent: fix all except i
            # h[i] is the only free variable, sum constraint means nothing changes
            # We just find best h[i] in [0,1]
            h_orig = best_h[i]
            # Try values
            for hv in [0.0, 1.0, h_orig * 0.5, h_orig * 1.5, h_orig - 0.1, h_orig + 0.1]:
                if hv < 0 or hv > 1:
                    continue
                # Adjust another coordinate to maintain sum
                # Actually for coordinate descent with equality constraint,
                # we need to move two coordinates. Let's do pairwise.
                pass
            # Pairwise: transfer from i to j
            for j in [0, n - 1, i // 2]:
                if j == i:
                    continue
                v2 = np.zeros(n)
                v2[i] = -1.0
                v2[j] = 1.0
                h_new, c_new = line_search(best_h, v2, n_samples=100)
                if h_new is not None and c_new < best_c - 1e-10:
                    best_h = h_new
                    best_c = c_new
                    improved = True
        if not improved:
            break
    return best_h, best_c


def interp_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    fx = np.concatenate([[0.0], xs, [2.0]])
    fh = np.concatenate([[h[0]], h, [h[-1]]])
    hn = np.interp(xt, fx, fh)
    return repair_feasible(np.clip(hn, 0, 1))


def local_optimize(h, n, rng, time_limit, max_no_improve=80):

    h_curr = h.copy()

    no_improve = 0
    iter_count = 0

    while time.perf_counter() - t0 < time_limit:
        iter_count += 1

        # Direction selection
        r = iter_count % 7
        if r == 0:
            v = subgradient_direction(h_curr, n)
        elif r == 1:
            v = -subgradient_direction(h_curr, n)
        elif r == 2:
            v = zero_sum_random(n, rng)
        elif r == 3:

            v = sparse_direction(n, rng)
        elif r == 5:
            v = antisym_direction(n, rng)
        else:
            # Blend of subgradient and random
            v1 = subgradient_direction(h_curr, n)
            v2 = zero_sum_random(n, rng)
            v = 0.7 * v1 + 0.3 * v2
            v -= v.mean()
            vn = np.linalg.norm(v)
            if vn > 1e-12:
                v /= vn

        if v is None or np.max(np.abs(v)) < 1e-12:
            v = zero_sum_random(n, rng)

        h_new, c_new = line_search(h_curr, v, n_samples=400)

        if h_new is not None and c_new < best_c - 1e-10:
            h_curr = h_new
            best_h = h_new
            best_c = c_new
            no_improve = 0
        else:

                no_improve = 0

    return best_h, best_c
