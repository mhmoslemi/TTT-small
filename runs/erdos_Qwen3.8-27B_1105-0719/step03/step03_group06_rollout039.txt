def repair_feasible(vec):
    """Project onto {0≤h≤1, sum(h) = n/2}"""
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        # Need to add mass, add to entries with most room
        order = np.argsort(-(1.0 - h))
        remaining = diff
        for i in range(n):
            idx = order[i]
            cap = 1.0 - h[idx]
            if cap <= 1e-15:
                continue
            add = min(remaining, cap)
            h[idx] += add
            remaining -= add
            if remaining <= 1e-14:
                break
    else:
        # Need to remove mass, remove from entries with least value
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            if h[idx] <= 1e-15:
                continue
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-14:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:

        else:
            idx = int(np.argmax(h))

    return h

        profile = compute_profile(y)
        c = float(profile.max())
        thresh = c - tol
        active = np.where(profile >= thresh)[0]

            active = np.array([int(np.argmax(profile))])

        g = np.zeros(n)
        for m in active:

        g = np.zeros(n)
        for m in active:

        profile = compute_profile(h)
        c = float(profile.max())
        thresh = c - active_tol
        active = np.where(profile >= thresh)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(profile))])
        if not use_all and len(active) > 30:
            top = np.argsort(profile[active])[-30:]
            active = active[top]
        g = np.zeros(n)
        for m in active:
            g += grad_row(h, m, n)
        g -= g.mean()
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            break
        momentum_vec = momentum * momentum_vec + g
        mv_norm = np.linalg.norm(momentum_vec)
        if mv_norm < 1e-15:
            break
        d = -momentum_vec / mv_norm
        pos_mask = d > 1e-12
        neg_mask = d < -1e-12
        alpha_max = 1.0
        if pos_mask.any():
            a = float(((1.0 - h[pos_mask]) / d[pos_mask]).min())
            if a < alpha_max:
                alpha_max = a
        if neg_mask.any():
            a = float((h[neg_mask] / (-d[neg_mask])).min())
            if a < alpha_max:
                alpha_max = a
        alpha_max = max(alpha_max, 1e-10)
        alpha_max = min(alpha_max, 0.8)
        best_alpha = 0.0
        best_val = c
        best_h_ls = h
        a_ls, b_ls = 0.0, alpha_max
        for _ in range(25):
            c1 = b_ls - gr * (b_ls - a_ls)
            c2 = a_ls + gr * (b_ls - a_ls)
            h1 = repair_feasible(h + c1 * d)
            h2 = repair_feasible(h + c2 * d)
            f1 = exact_score(h1)
            f2 = exact_score(h2)
            if f1 < best_val:
                best_val = f1
                best_alpha = c1
                best_h_ls = h1.copy()
            if f2 < best_val:
                best_val = f2
                best_alpha = c2
                best_h_ls = h2.copy()
            if f1 < f2:
                b_ls = c2
            else:
                a_ls = c1
            if b_ls - a_ls < 1e-14:
                break
        if best_val >= c - 1e-15:
            stagnation += 1
            if stagnation > 20:
                break
        else:
            stagnation = 0
        h = best_h_ls
        if best_val < best_c:
            best_c = best_val
            best_h = h.copy()
    return best_h, best_c

def frank_wolfe(h_start, max_iters=1500, tol=0.003):
    n = len(h_start)
    h = h_start.copy()
    best_h = h.copy()
    best_c = exact_score(h)
    for it in range(max_iters):
        profile = compute_profile(h)
        c = float(profile.max())
        thresh = c - tol
        active = np.where(profile >= thresh)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(profile))])
        g = np.zeros(n)
        for m in active:
            g += grad_row(h, m, n)
        g -= g.mean()
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-14:
            break
        k = n // 2
        fw_idx = np.argsort(g)[:k]
        h_fw = np.zeros(n)
        h_fw[fw_idx] = 1.0
        alpha = 2.0 / (it + 2.0)
        h_new = (1.0 - alpha) * h + alpha * h_fw
        h_new = repair_feasible(h_new)
        c_new = exact_score(h_new)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c

def interp_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xc_old = np.linspace(1.0 / n, 2.0 - 1.0 / n, n)
    xc_new = np.linspace(1.0 / new_n, 2.0 - 1.0 / new_n, new_n)
    h_new = np.interp(xc_new, xc_old, h)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

def spectral_init(n, seed_val=99):
    rng = np.random.RandomState(seed_val)
    h = np.full(n, 0.5)
    phases = rng.uniform(0, 2 * np.pi, n)
    freqs = np.arange(n)
    pert = np.zeros(n)
    for k in range(n // 6):
        amp = 1.0 / np.sqrt(k + 1)
        pert += amp * np.cos(2 * np.pi * k * freqs / n + phases[k])
    h += 0.25 * pert / (np.max(np.abs(pert)) + 1e-10)
    h = np.clip(h, 0, 1)
    return repair_feasible(h)

def slsqp_epigraph(h_start, maxiter=200):
    n = len(h_start)
    dx = 2.0 / n
    M = 2 * n - 1
    c0 = exact_score(h_start)
    t0 = c0 + 1e-3
    def obj(x):
        return x[-1]
    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    def eq_fun(x):
        return x[:-1].sum() - n / 2.0
    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g
    def ineq_fun(x):
        h = x[:-1]
        t = x[-1]
        profile = np.correlate(h, 1 - h, mode="full") * dx
        return t - profile
    def ineq_jac(x):
        h = x[:-1]
        J = np.zeros((M, n + 1))
        for m in range(M):
            J[m, :n] = -grad_row(h, m, n)
        J[:, -1] = 1.0
        return J
    x0 = np.concatenate([h_start, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 2.0)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-13})
        if res.x is not None and np.isfinite(res.x).all():
            h_new = repair_feasible(res.x[:-1])
            c_new = exact_score(h_new)
            if c_new <= c0 + 1e-8:
                return h_new
    except Exception:
        pass
    return None

def pairwise_polish(h, max_rounds=50):
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    idx_desc = np.argsort(-h)
    idx_asc = np.argsort(h)
    for rnd in range(max_rounds):
        improved = False
        for attempt in range(40):
            i = int(idx_desc[attempt % n])
            j = int(idx_asc[(attempt * 7) % n])
            if i == j:
                continue
            val = 0.01 * (h[i] - h[j])
            if abs(val) < 1e-7:
                continue
            for sign in [1.0, -1.0]:
                h_try = h.copy()
                h_try[i] += sign * val
                h_try[j] -= sign * val
                if h_try.min() < -1e-10 or h_try.max() > 1 + 1e-10:
                    continue
                h_try = repair_feasible(h_try)
                s = exact_score(h_try)
                if s < best_c - 1e-13:
                    h = h_try
                    best_c = s
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s - (time.perf_counter() - t_start)

    try:
        parent = np.array(initial_h_values, dtype=np.float64)
    except Exception:
        parent = None
    if parent is None or len(parent) < 4:
        parent = np.full(70, 0.5)
    parent = np.clip(parent, 0.0, 1.0)
    parent = repair_feasible(parent)

    best_h = parent.copy()
    best_c = exact_score(parent)

    # Phase 1: Active descent on parent with all active constraints
    if time_left() > 80:
        h_ad, c_ad = active_descent_v2(best_h, max_iters=500, active_tol=0.003, use_all=True)
        if c_ad < best_c:
            best_h, best_c = h_ad, c_ad

    # Phase 2: Frank-Wolfe for diverse exploration
    if time_left() > 60:
        h_fw, c_fw = frank_wolfe(best_h, max_iters=1000, tol=0.004)
        if c_fw < best_c:
            h_polish, c_polish = active_descent_v2(h_fw, max_iters=200, active_tol=0.002, use_all=True)
            if c_polish < best_c:
                best_h, best_c = h_polish, c_polish
            else:
                best_h, best_c = h_fw, c_fw

    # Phase 3: Dimension exploration
    n_best = len(best_h)
    dims = sorted(set([n_best + d for d in [-8, -6, -4, -2, 2, 4, 6, 8, 10, 12, 14, 16, 20]
                      if 44 <= n_best + d <= 160 and (n_best + d) % 2 == 0]))
    for new_n in dims[:10]:
        if time_left() < 45:
            break
        h_interp = interp_to_n(best_h, new_n)
        h_ad, c_ad = active_descent_v2(h_interp, max_iters=300, active_tol=0.003, use_all=True)
        if c_ad < best_c:
            best_h, best_c = h_ad, c_ad
            if time_left() > 40:
                h_fw2, c_fw2 = frank_wolfe(best_h, max_iters=500, tol=0.003)
                if c_fw2 < best_c:
                    h_p2, c_p2 = active_descent_v2(h_fw2, max_iters=150, active_tol=0.002, use_all=True)
                    if c_p2 < best_c:
                        best_h, best_c = h_p2, c_p2
                    else:
                        best_h, best_c = h_fw2, c_fw2

    # Phase 4: Multi-start
    n_cur = len(best_h)
    inits = []
    if time_left() > 30:
        inits.append(spectral_init(n_cur, 77))
    if time_left() > 30:
        for s in range(3):
            h_r = rng.uniform(0, 1, n_cur)
            h_r = np.convolve(h_r, np.ones(7) / 7, mode='same')
            inits.append(repair_feasible(h_r))
    if time_left() > 30:
        h_b = (rng.uniform(0, 1, n_cur) > 0.5).astype(float)
        h_b = 0.3 * h_b + 0.7 * np.full(n_cur, 0.5)
        inits.append(repair_feasible(h_b))

    for h_init in inits:
        if time_left() < 25:
            break
        h_ad, c_ad = active_descent_v2(h_init, max_iters=250, active_tol=0.004, use_all=True)
        if c_ad < best_c:
            best_h, best_c = h_ad, c_ad
            if time_left() > 30:
                h_fw3, c_fw3 = frank_wolfe(best_h, max_iters=400, tol=0.003)
                if c_fw3 < best_c:
                    h_p3, c_p3 = active_descent_v2(h_fw3, max_iters=100, active_tol=0.002, use_all=True)
                    best_h, best_c = (h_p3, c_p3) if c_p3 < best_c else (h_fw3, c_fw3)

    # Phase 5: SLSQP refinement (multiple passes)
    for _ in range(4):
        if time_left() < 25:
            break
        h_sol = slsqp_epigraph(best_h, maxiter=150)
        if h_sol is not None:
            c_sol = exact_score(h_sol)
            if c_sol < best_c - 1e-12:
                best_h, best_c = h_sol, c_sol
            if time_left() > 20:
                h_ad, c_ad = active_descent_v2(best_h, max_iters=150, active_tol=0.001, use_all=True)
                if c_ad < best_c:
                    best_h, best_c = h_ad, c_ad

    # Phase 6: Pairwise polish
    if time_left() > 20:
        h_pw, c_pw = pairwise_polish(best_h, max_rounds=60)
        if c_pw < best_c:
            best_h, best_c = h_pw, c_pw

    # Phase 7: Final tight active descent
    if time_left() > 15:
        h_ad, c_ad = active_descent_v2(best_h, max_iters=300, active_tol=0.0005, use_all=True)
        if c_ad < best_c:
            best_h, best_c = h_ad, c_ad

    # Phase 8: Final SLSQP
    if time_left() > 15:
        h_sol = slsqp_epigraph(best_h, maxiter=250)
        if h_sol is not None:
            c_sol = exact_score(h_sol)
            if c_sol < best_c:
                best_h, best_c = h_sol, c_sol
                if time_left() > 10:
                    h_ad, c_ad = active_descent_v2(best_h, max_iters=100, active_tol=0.0003, use_all=True)
                    if c_ad < best_c:
                        best_h, best_c = h_ad, c_ad

    # Phase 9: Final pairwise
    if time_left() > 10:
        h_pw, c_pw = pairwise_polish(best_h, max_rounds=40)
        if c_pw < best_c:
            best_h, best_c = h_pw, c_pw

    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)

    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            final_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 2:
            final_c = float(eval_result[0])
    except Exception:
        pass

    return best_h, final_c, len(best_h)
