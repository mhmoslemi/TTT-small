        qd = 0.0
        
        # Case 1: i=p, p+d valid
        pp = p + d
        if 0 <= pp < n:
            if pp == q:
                kd += (1.0 - b)
                qd += 1.0
            else:
                kd += (1.0 - hd[pp])
        
        # Case 2: i=p-d valid
        pm = p - d
        if 0 <= pm < n:
            if pm == q:
                kd += (a - b - 1.0)
                qd += 1.0
            else:
                kd += -hd[pm]
        
        # Case 3: i=q, q+d valid
        qp = q + d
        if 0 <= qp < n:
            if qp == p:
                kd += (a - b - 1.0)
                qd += 1.0
            else:
                kd += -(1.0 - hd[qp])
        
        # Case 4: i=q-d valid
        qm = q - d
        if 0 <= qm < n:
            if qm == p:
                kd += (1.0 - b)
                qd += 1.0
            else:
                kd += -hd[qm]
        
        K[m] = kd
        Q[m] = qd
    
    P0 = profile_h(h)
    
    def max_prof(t_val):
        ut = t_val - a
        return np.max(P0 + dx * (ut * K + ut * ut * Q))
    
    # Golden section
    phi = (np.sqrt(5.0) - 1.0) / 2.0
    ga = t_lo
    gb = t_hi
    gc = gb - phi * (gb - ga)
    gd = ga + phi * (gb - ga)
    fc = max_prof(gc)
    fd = max_prof(gd)
    
    for _ in range(50):
        if gb - ga < 1e-11:
            break
        if fc < fd:
            gb = gd
            gd = gc
            fd = fc
            gc = gb - phi * (gb - ga)
            fc = max_prof(gc)
        else:
            ga = gc
            gc = gd
            fc = fd
            gd = ga + phi * (gb - ga)
            fd = max_prof(gd)
    
    return (ga + gb) / 2.0

import numpy as np

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full"))) * dx

def profile_h(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = int(order[i])
            add = min(remaining, capacity[idx])
            h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = int(order[i])
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:

        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))

        h = np.clip(h, 0.0, 1.0)

def pairwise_minimax(h, p, q):
    n = len(h)
    dx = 2.0 / n
    a, b = h[p], h[q]
    s = a + b
    t_lo = max(0.0, s - 1.0)
    t_hi = min(1.0, s)
    if t_hi - t_lo < 1e-14:
        return None
    K, Q = compute_KQ(h, p, q)
    P0 = profile_h(h)

    C = dx * Q

    ga, gb = t_lo, t_hi
    gc = gb - PHI * (gb - ga)
    gd = ga + PHI * (gb - ga)

def pairwise_sweep(h, max_sweeps, rng, t_start, budget_s):

def slsqp_active(h0, maxiter=200, active_k=12, tol=1e-11):
    n = len(h0)
    dx = 2.0 / n
    M = 2 * n - 1
    P = profile_h(h0)
    c0 = P.max()
    thresh = c0 - max(0.002, 0.06 * (c0 - np.median(P)))
    active = np.where(P >= thresh)[0]
    if len(active) == 0:
        active = np.array([np.argmax(P)])
    if len(active) > active_k:
        top_idx = np.argsort(P[active])[-active_k:]
        active = active[top_idx]

    def obj(x):
        return x[-1]

    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g

    def eq_fun(x):
        return x[:-1].sum() - n / 2.0

    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g

    def ineq_fun(x):
        h = x[:-1]
        t = x[-1]
        prof = profile_h(h)
        return t - prof[active]

    def ineq_jac(x):
        h = x[:-1]
        K, _ = compute_KQ(h, 0, 1)
        # Build gradient matrix for active lags
        G = np.zeros((len(active), n))
        for idx, m in enumerate(active):
            d = m - (n - 1)
            if d == 0:
                G[idx, :] = dx * (1.0 - 2.0 * h)
            elif d > 0:
                for i in range(n - d):
                    G[idx, i + d] += dx * (1.0 - h[i])
                    G[idx, i] -= dx * h[i + d]
            else:
                dp = -d
                for j in range(n - dp):
                    G[idx, j + dp] += dx * (1.0 - h[j])
                    G[idx, j] -= dx * h[j + dp]
        J = np.zeros((len(active), n + 1))
        J[:, :-1] = -G
        J[:, -1] = 1.0
        return J

    x0 = np.concatenate([h0, [c0 + 1e-4]])
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 1.0)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': tol})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = repair_feasible(res.x[:-1])
            c_new = exact_score(h_new)
            if c_new <= c0 + 1e-8:
                return h_new
    except Exception:
        pass
    return None

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    return repair_feasible(np.clip(h_new, 0.0, 1.0))

        cand_dims = []
        for d in range(max(24, n_best - 12), min(120, n_best + 14)):
            if d % 2 == 0 and d != n_best:
                cand_dims.append(d)
        cand_dims.sort(key=lambda d: abs(d - n_best))
        for new_n in cand_dims[:6]:
            if time.perf_counter() - t_start > budget_s - 30:
                break
            h_interp = interpolate_to_n(best_h, new_n)
            # Quick pairwise optimization at new dimension
            h5, c5 = pairwise_sweep(h_interp, 8, rng, t_start, budget_s)
            if c5 < best_c - 1e-10:
                best_h, best_c = h5, c5
                # More sweeps at winning dimension
                h6, c6 = pairwise_sweep(best_h, 12, rng, t_start, budget_s)
                if c6 < best_c - 1e-14:
                    best_h, best_c = h6, c6
                # SLSQP at winning dimension
                if time.perf_counter() - t_start < budget_s - 40:
                    sol = slsqp_active(best_h, maxiter=150, active_k=12, tol=1e-11)
                    if sol is not None:
                        c_s = exact_score(sol)
                        if c_s < best_c - 1e-12:
                            best_h, best_c = sol, c_s
                            h7, c7 = pairwise_sweep(best_h, 8, rng, t_start, budget_s)
                            if c7 < best_c - 1e-14:
                                best_h, best_c = h7, c7

    # Phase 6: Final intensive pairwise polishing
    if time.perf_counter() - t_start < budget_s - 20:
        h8, c8 = pairwise_sweep(best_h, 20, rng, t_start, budget_s)
        if c8 < best_c - 1e-15:
            best_h, best_c = h8, c8

    # Phase 7: Symmetrize and re-optimize
    if time.perf_counter() - t_start < budget_s - 20:
        h_sym2 = repair_feasible(symmetrize(best_h))
        c_sym2 = exact_score(h_sym2)
        if c_sym2 < best_c - 1e-14:
            best_h, best_c = h_sym2, c_sym2
            h9, c9 = pairwise_sweep(best_h, 10, rng, t_start, budget_s)
            if c9 < best_c - 1e-15:
                best_h, best_c = h9, c9

    # Phase 8: Final SLSQP polish
    if time.perf_counter() - t_start < budget_s - 20:
        sol = slsqp_active(best_h, maxiter=300, active_k=10, tol=1e-13)
        if sol is not None:
            c = exact_score(sol)
            if c < best_c - 1e-14:
                best_h, best_c = sol, c

    # Final validation
    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)

    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            final_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 1:
            if np.isfinite(eval_result[0]):
                final_c = float(eval_result[0])
    except Exception:
        pass

    return best_h, final_c, len(best_h)
