    mask1 = (idx1 >= 0) & (idx1 < n)

def pair_line_search(h, i, j, profile, dx):
    n = len(h)
    a, b = h[i], h[j]
    t_low = max(-a, b - 1.0)
    t_high = min(1.0 - a, b)
    if t_high - t_low < 1e-12:
        return None, None
    
    g = pair_gradient(h, i, j)
    
    # Find minimum of max(profile + t*g) over t in [t_low, t_high]
    # This is a convex PWL function, so we can use golden section or just grid search
    
    # Fine grid
    n_grid = 50
    t_grid = np.linspace(t_low, t_high, n_grid)
    # profile shape (M,), g shape (M,)
    # For each t, new_profile = profile + t * g
    # We want max over lag for each t
    vals = profile + np.outer(t_grid, g)  # shape (n_grid, M)
    f_vals = vals.max(axis=1)  # shape (n_grid,)
    
    best_idx = np.argmin(f_vals)
    t_best = t_grid[best_idx]
    f_best = f_vals[best_idx]
    
    # Refine around best
    if 0 < best_idx < n_grid - 1:
        t_lo = t_grid[best_idx - 1]
        t_hi = t_grid[best_idx + 1]
        t_fine = np.linspace(t_lo, t_hi, 20)
        vals_f = profile + np.outer(t_fine, g)
        f_f = vals_f.max(axis=1)
        best_fi = np.argmin(f_f)
        t_best = t_fine[best_fi]
        f_best = f_f[best_fi]
    
    return t_best, f_best

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx] + 1e-14)
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx] + 1e-14)
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    h = np.clip(h, 0.0, 1.0)
    return h

def interpolate_to_n(h, new_n):

        width = 0.0

            lo_j, hi_j = xs[j], xs[j+1]
            overlap = min(hi_k, hi_j) - max(lo_k, lo_j)

                width += overlap

def build_gradient_matrix(h, n, dx):
    M = 2 * n - 1
    G = np.zeros((M, n))
    for m in range(M):
        d = m - (n - 1)
        if d == 0:
            G[m, :] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            G[m, d:] += dx * (1.0 - h[0:n-d])
            G[m, 0:n-d] += -dx * h[d:n]
        else:
            dp = -d
            G[m, 0:n-dp] += dx * (1.0 - h[dp:n])
            G[m, dp:n] += -dx * h[0:n-dp]
    return G

        if wsum < 1e-300:
            break
        weights /= wsum

        return t - profile

        J = np.zeros((M, n + 1))

        J[:, -1] = 1.0

    x0 = np.concatenate([h0, [c0 + 1e-3]])
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 0.5)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
