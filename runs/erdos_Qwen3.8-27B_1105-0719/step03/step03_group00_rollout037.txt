    off = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -off)

        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx + off] += (1.0 - h[idx])
    grad[idx] += (-h[idx + off])
    return grad

    return (c_max + np.log(np.sum(np.exp(lam * (corr - c_max)))) / lam)

    c_max = corr.max()
    weights = np.exp(lam * (corr - c_max))
    weights /= weights.sum()
    return weights

    K = len(active_idx)
    m_vars = n + 1
    grads = np.zeros((K, n))
    for j, k in enumerate(active_idx):
        grads[j] = grad_corr_k(h, k, n)
    c_obj = np.zeros(m_vars)
    c_obj[-1] = 1.0
    A_ub = np.zeros((K, m_vars))
    for j in range(K):
        A_ub[j, :n] = dx * grads[j]
        A_ub[j, -1] = -1.0
    b_ub = -dx * corrs[active_idx]
    A_eq = np.zeros((1, m_vars))
    A_eq[0, :n] = 1.0

    """Compute weighted sum of gradients, where weights[k] = w_k"""
    grad = np.zeros(n)
    for k_idx in range(2*n-1):
        w = weights[k_idx]
        if w < 1e-8:
            continue
        m = k_idx - (n-1)

        grad[valid2] += -w * cur[j1[valid2] - m]

    off = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -off)
    i_hi = min(n - 1, n - 1 - off)
    if i_lo > i_hi:

    idx = np.arange(i_lo, i_hi + 1)

    grad[idx] += (-h[idx + off])
    return grad

    m_vars = n + 1
    c_obj = np.zeros(m_vars)
    c_obj[-1] = 1.0
    A_ub = np.zeros((K, m_vars))
    for j, k in enumerate(active_idx):
        g = grad_corr_k(h, k, n)
        A_ub[j, :n] = dx * g
        A_ub[j, -1] = -1.0
    b_ub = -dx * corrs[active_idx]
    A_eq = np.zeros((1, m_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-h[i]*alpha, -1.0), min((1.0-h[i])*alpha, 1.0)) for i in range(n)]

import numpy as np
from scipy.optimize import linprog
import time

def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def c5_of_h(h, dx):
    return np.correlate(h, 1.0 - h, mode='full').max() * dx

def corr_all(h):
    return np.correlate(h, 1.0 - h, mode='full')

def grad_corr_k(h, k, n):
    off

=k - (n - 1)
    i_lo = max(0, -off)
    i_hi = min(n - 1, n - 1 - off)
    if i_lo > i_hi:
        return np.zeros(n)
    grad = np.zeros(n)
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx + off] += (1.0 - h[idx])
    grad[idx] -= h[idx + off]
    return grad

def lp_solve(h, active_idx, corrs, dx, alpha, n):
    K = len(active_idx)
    if K == 0:
        return None
    m_vars = n + 1
    c_obj = np.zeros(m_vars)
    c_obj[-1] = 1.0
    A_ub = np.zeros((K, m_vars))
    for j, k in enumerate(active_idx):
        g = grad_corr_k(h, k, n)
        A_ub[j, :n] = dx * g
        A_ub[j, -1] = -1.0
    b_ub = -dx * corrs[active_idx] - 1e-10  # small slack
    A_eq = np.zeros((1, m_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for i in range(n):
        lo

 = max(-h[i] * alpha, -1.0)
        hi = min((1.0 - h[i]) * alpha, 1.0)
        bounds.append((lo, hi))
    bounds.append((0.0, None))
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status != 0:
            return None
        return res.x[:n]
    except:
        return None

def lp_polish(h, dx, target_sum, budget_s, t0, n_iters=200, alpha_init=0.2):

    best_c5 = c5_of_h(h, dx)
    cur = h.copy()
    cur_c5 = best_c5
    alpha = alpha_init

        if budget_s - (time.time() - t0) < 3.0:
            break
        corrs = corr_all(cur)
        max_c = corrs.max()
        margin

 = max(0.03, 0.02 * max_c)
        n_top = max(8, min(40, int(0.2 * (2 * n - 1))))
        top_idx = np.argsort(corrs)[-n_top:]
        active_mask = corrs >= (max_c - margin)
        active_mask[top_idx] = True
        active_idx = np.where(active_mask)[0]
        if len(active_idx) < 3:

            keep = np.argsort(corrs[active_idx])[-80:]
            active_idx = active_idx[keep]
        
        delta = lp_solve(cur, active_idx, corrs, dx, alpha, n)
        if delta is None:
            alpha *= 0.5
            if alpha < 0.001:

            continue
        
        improved = False

def smooth_continuation(h, dx, target_sum, budget_s, t0):
    n = len(h)
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    cur = h.copy()
    cur_c5 = best_c5
    
    lam_schedule = [5, 15, 40, 80, 150, 300, 600, 1500]
    
    for lam in lam_schedule:
        if budget_s - (time.time() - t0) < 15:
            break
        step = 0.02
        for it in range(200):
            if budget_s - (time.time() - t0) < 8:
                break
            corrs = corr_all(cur)
            c_max = corrs.max()
            exponents = lam * (corrs - c_max)
            exponents = np.clip(exponents, -300, 300)
            weights = np.exp(exponents)
            sw = weights.sum()
            if sw < 1e-10:
                break
            weights /= sw
            
            # Weighted gradient
            grad = np.zeros(n)
            active = np.where(weights > 1e-6)[0]
            for ki in active:
                w = weights[ki]
                m = ki - (n - 1)
                j = np.arange(n)
                v1 = (j + m >= 0) & (j + m < n)
                grad[v1] += w * (1.0 - cur[j[v1] + m])
                v2 = (j - m >= 0) & (j - m < n)
                grad[v2] -= w * cur[j[v2] - m]
            grad *= dx
            grad -= np.mean(grad)
            
            # Line search
            improved = False
            for s in [step, step*0.5, step*0.25, step*0.1]:
                h_new = project_feasible(cur - s * grad, target_sum)
                c5_new = c5_of_h(h_new, dx)
                if c5_new < cur_c5 - 1e-15:
                    cur = h_new
                    cur_c5 = c5_new
                    improved = True
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = cur.copy()
                    break
            if not improved:
                step *= 0.5
                if step < 1e-7:
                    break
            else:
                step = min(step * 1.2, 0.06)
    
    return best_h, best_c5

def binary_sa(h, dx, target_sum, rng, budget_s, t0, n_iters=5000):

    h_bin = h.copy()

    thresh = np.sort(h_bin)[::-1][n//2]
    h_bin = (h_bin > thresh).astype(float)
    # Fix if needed
    n_ones

 = int(h_bin.sum())
    if n_ones != n // 2:
        # Adjust
        zeros = np.where(h_bin == 0)[0]
        ones = np.where(h_bin == 1)[0]
        if n_ones > n // 2:
            for idx in zeros[:n_ones - n//2]:
                # Flip a 1 to 0
                h_bin[ones[0]] = 0
        elif n_ones < n // 2:
