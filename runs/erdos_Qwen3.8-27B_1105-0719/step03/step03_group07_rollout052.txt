    """Find optimal t for h[i]+=t, h[j]-=t to minimize max of profile."""
    n = len(h)
    dx = 2.0 / n
    P = np.correlate(h, 1 - h, mode="full") * dx
    
    # Compute slope for each lag
    s = np.zeros(2*n - 1)
    for m in range(2*n - 1):
        d = m - (n - 1)
        sm = 0.0
        if 0 <= i+d < n:
            sm += 1.0 - h[i+d]
        if 0 <= i-d < n:
            sm -= h[i-d]
        if 0 <= j+d < n:
            sm -= (1.0 - h[j+d])
        if 0 <= j-d < n:
            sm += h[j-d]
        s[m] = dx * sm
    
    t_min = max(-h[i], 0.0)  # h[i] + t >= 0 => t >= -h[i]; but we want t>=0 for increase
    # Actually t can be negative (decrease h[i], increase h[j])
    t_lo = -h[i]  # h[i] + t >= 0
    t_hi = 1.0 - h[i]  # h[i] + t <= 1
    # Also h[j] - t >= 0 => t <= h[j]
    # And h[j] - t <= 1 => t >= h[j] - 1

    t_hi = min(t_hi, h[j])
    
    if t_hi <= t_lo:
        return h, exact_score(h)

    # This is convex piecewise linear. Find minimum in [t_lo, t_hi].

        vals = P + t * s
        return vals.max()

    # Subgradient at t is s[argmax(P + t*s)]

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full"))) * dx

def profile_h(h):

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx])
            h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    return repair_feasible(np.clip(h_new, 0.0, 1.0))

def pair_line_search(h, i, j):
    """Exact 1D line search for pair (i, j). Returns new h and new score."""
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    P = np.correlate(h, 1 - h, mode="full") * dx
    
    # Compute slope vector
    s = np.zeros(M)
    d_range = np.arange(-(n-1), n)
    for idx in range(M):
        d = d_range[idx]
        sm = 0.0
        ip = i + d
        im = i - d
        jp = j + d
        jm = j - d
        if 0 <= ip < n:
            sm += 1.0 - h[ip]
        if 0 <= im < n:
            sm -= h[im]
        if 0 <= jp < n:
            sm -= (1.0 - h[jp])
        if 0 <= jm < n:
            sm += h[jm]
        s[idx] = dx * sm
    
    # Feasible range
    t_lo = max(-h[i], h[j] - 1.0)
    t_hi = min(1.0 - h[i], h[j])
    
    if t_hi - t_lo < 1e-15:
        return h, P.max()
    
    # Bisection to find minimum of f(t) = max(P + t*s)
    lo, hi = t_lo, t_hi
    for _ in range(60):
        mid = (lo + hi) * 0.5
        vals = P + mid * s
        m_star = vals.argmax()
        if s[m_star] > 1e-15:
            hi = mid
        elif s[m_star] < -1e-15:
            lo = mid
        else:
            break
    
    # Evaluate at candidates
    candidates = [t_lo, t_hi, (lo + hi) * 0.5, 0.0]
    best_t = 0.0
    best_val = P.max()
    for t in candidates:
        if t < t_lo - 1e-15 or t > t_hi + 1e-15:
            continue
        t = np.clip(t, t_lo, t_hi)
        val = (P + t * s).max()
        if val < best_val - 1e-14:
            best_val = val
            best_t = t
    
    if abs(best_t) < 1e-15:
        return h, P.max()
    
    h_new = h.copy()
    h_new[i] += best_t
    h_new[j] -= best_t
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = repair_feasible(h_new)
    return h_new, exact_score(h_new)

def pairwise_cd(h, max_sweeps=5, t_start=None, budget_s=1100, margin=30):
    """Multiple sweeps of pairwise coordinate descent."""
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    P = profile_h(h)
    
    for sweep in range(max_sweeps):
        if t_start is not None and time.perf_counter() - t_start > budget_s - margin:
            break
        improved = False
        for i in range(n):
            for j in range(i+1, n):
                if t_start is not None and time.perf_counter() - t_start > budget_s - margin:
                    break
                h_new, new_c = pair_line_search(h, i, j)
                if new_c < best_c - 1e-13:
                    h = h_new
                    best_c = new_c
                    improved = True
        if not improved:
            break
    
    return h, best_c

def pair_slopes_vec(h, i, j):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    d_range = np.arange(-(n-1), n)
    
    # For index i:
    # (1-h[i+d]) valid when 0 <= i+d < n => -i <= d < n-i
    # -h[i-d] valid when 0 <= i-d < n => i-n < d <= i
    # For index j:
    # -(1-h[j+d]) valid when 0 <= j+d < n
    # +h[j-d] valid when 0 <= j-d < n
    
    s = np.zeros(M)
    # Term 1: (1-h[i+d]) where 0 <= i+d < n
    mask1 = (d_range >= -i) & (d_range < n - i)
    s[mask1] += 1.0 - h[i + d_range[mask1]]
    # Term 2: -h[i-d] where 0 <= i-d < n
    mask2 = (d_range > i - n) & (d_range <= i)
    s[mask2] -= h[i - d_range[mask2]]
    # Term 3: -(1-h[j+d]) where 0 <= j+d < n
    mask3 = (d_range >= -j) & (d_range < n - j)
    s[mask3] -= 1.0 - h[j + d_range[mask3]]
    # Term 4: +h[j-d] where 0 <= j-d < n
    mask4 = (d_range > j - n) & (d_range <= j)
    s[mask4] += h[j - d_range[mask4]]
    
    s *= dx
    return s

def slsqp_optimize(h0, maxiter=300, active_k=15, tol=1e-12):
    n = len(h0)
    dx = 2.0 / n
    M = 2 * n - 1
    
    def get_active(P, k):
        c0 = P.max()
        # Include all lags within a small margin of the max
        thresh = c0 - max(0.001, 0.02 * c0)
        active = np.where(P >= thresh)[0]
        if len(active) > k:
            top_idx = np.argsort(P[active])[-k:]
            active = active[top_idx]
        return active
    
    P = profile_h(h0)
    active = get_active(P, active_k)
    
    def obj(x):
        return x[-1]
    
    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    
    def eq_fun(x):
        return x[:-1].sum() - n / 2.0
    
    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g
    
    def ineq_fun(x):
        h = x[:-1]
        t = x[-1]
        prof = np.correlate(h, 1 - h, mode="full") * dx
        return t - prof[active]
    
    def ineq_jac(x):
        h = x[:-1]
        M = 2 * n - 1
        G = np.zeros((len(active), n))
        for idx, m in enumerate(active):
            d = m - (n - 1)
            if d == 0:
                G[idx, :] = dx * (1.0 - 2.0 * h)
            elif d > 0:
                for i in range(n - d):
                    G[idx, i + d] += dx * (1.0 - h[i])
                    G[idx, i] -= dx * h[i + d]
            else:
                dp = -d
                for j in range(n - dp):
                    G[idx, j + dp] += dx * (1.0 - h[j])
                    G[idx, j] -= dx * h[j + dp]
        J = np.zeros((len(active), n + 1))
        J[:, :-1] = -G
        J[:, -1] = 1.0
        return J
    
    c0 = P.max()
    x0 = np.concatenate([h0, [c0 + 1e-4]])
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 1.0)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': tol})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = repair_feasible(res.x[:-1])
            c_new = exact_score(h_new)
            if c_new <= c0 + 1e-10:
                return h_new
    except Exception:
        pass
    return None

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx])
            h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    return repair_feasible(np.clip(h_new, 0.0, 1.0))

def pair_slopes_vec(h, i, j):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    d_range = np.arange(-(n - 1), n, dtype=np.int64)
    s = np.zeros(M)
    mask1 = (d_range >= -i) & (d_range <= n - 1 - i)
    if np.any(mask1):
        s[mask1] += 1.0 - h[i + d_range[mask1]]
    mask2 = (d_range >= i - n + 1) & (d_range <= i)
    if np.any(mask2):
        s[mask2] -= h[i - d_range[mask2]]
    mask3 = (d_range >= -j) & (d_range <= n - 1 - j)
    if np.any(mask3):
        s[mask3] -= 1.0 - h[j + d_range[mask3]]
    mask4 = (d_range >= j - n + 1) & (d_range <= j)
    if np.any(mask4):
        s[mask4] += h[j - d_range[mask4]]
    s *= dx
    return s

def pairwise_cd(h, max_sweeps, t_start, budget_s, margin=30):
    n = len(h)
    h = h.copy()
    P = profile_h(h)
    best_c = P.max()
    for sweep in range(max_sweeps):
        if time.perf_counter() - t_start > budget_s - margin:
            break
        any_improved = False
        for i in range(n):

                if time.perf_counter() - t_start > budget_s - margin:

                h, c = pair_line_search(h, i, j, P)
                if c < best_c - 1e-14:
                    best_c = c
                    P = profile_h(h)
                    any_improved = True

def build_grad_matrix(h, active, n, dx):
    M = 2 * n - 1
    G = np.zeros((len(active), n))
    for idx, m in enumerate(active):
        d = m - (n - 1)
        if d == 0:
            G[idx, :] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            for k in range(n - d):
                G[idx, k + d] += dx * (1.0 - h[k])
                G[idx, k] -= dx * h[k + d]
        else:
            dp = -d
            for k in range(n - dp):
                G[idx, k + dp] += dx * (1.0 - h[k])
                G[idx, k] -= dx * h[k + dp]
    return G

def slsqp_optimize(h0, maxiter=300, active_k=15, tol=1e-12):
    n = len(h0)
    dx = 2.0 / n
    P = profile_h(h0)
    c0 = P.max()
    thresh = c0 - max(0.002, 0.03 * c0)
    active = np.where(P >= thresh)[0]
    if len(active) > active_k:
        top_idx = np.argsort(P[active])[-active_k:]
        active = active[top_idx]
    if len(active) == 0:
        active = np.array([np.argmax(P)])

    def obj(x):
        return x[-1]

    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g

    def eq_fun(x):
        return x[:-1].sum() - n / 2.0

    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g

    def ineq_fun(x):
        h = x[:-1]
        t = x[-1]
        prof = np.correlate(h, 1 - h, mode="full") * dx
        return t - prof[active]

    def ineq_jac(x):
        h = x[:-1]
        G = build_grad_matrix(h, active, n, dx)
        J = np.zeros((len(active), n + 1))
        J[:, :-1] = -G
        J[:, -1] = 1.0
        return J

    x0 = np.concatenate([h0, [c0 + 1e-4]])
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 1.0)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': tol})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = repair_feasible(res.x[:-1])
            c_new = exact_score(h_new)
            if c_new <= c0 + 1e-10:
                return h_new
    except Exception:
        pass
    return None

def symmetrize(h):

        n_best = len(best_h)
        cand_dims = []
        for d in range(max(24, n_best - 14), min(100, n_best + 14)):
            if d % 2 == 0 and d != n_best:
                cand_dims.append(d)
        cand_dims.sort(key=lambda d: abs(d - n_best))

        for new_n in cand_dims[:8]:
            if not time_ok(50):
                break
            h_interp = interpolate_to_n(best_h, new_n)
            h_cd, c_cd = pairwise_cd(h_interp, max_sweeps=3, t_start=t_start, budget_s=budget_s, margin=40)
            if c_cd < best_c - 1e-10:
                best_h = h_cd
                best_c = c_cd
                if time_ok(40):
                    sol = slsqp_optimize(best_h, maxiter=250, active_k=14, tol=1e-11)
                    if sol is not None:
                        c = exact_score(sol)
                        if c < best_c - 1e-10:
                            best_h = sol
                            best_c = c
                        h_cd2, c_cd2 = pairwise_cd(best_h, max_sweeps=3, t_start=t_start, budget_s=budget_s, margin=20)
                        if c_cd2 < best_c - 1e-12:
                            best_h = h_cd2
                            best_c = c_cd2
            elif time_ok(40):
                sol = slsqp_optimize(h_interp, maxiter=200, active_k=12, tol=1e-11)
                if sol is not None:
                    c = exact_score(sol)
                    if c < best_c - 1e-10:
                        best_h = sol
                        best_c = c

    # Phase 6: SLSQP multiple rounds with active set updates
    if time_ok(120):
        for r in range(4):
            if not time_ok(30):
                break
            P = profile_h(best_h)
            c0 = P.max()
            sol = slsqp_optimize(best_h, maxiter=250, active_k=15, tol=1e-12)
            if sol is not None:
                c = exact_score(sol)
                if c < best_c - 1e-12:
                    best_h = sol
                    best_c = c
                else:
                    break
            else:
                break

    # Phase 7: Final pairwise CD
    if time_ok(50):
        h_cd, c_cd = pairwise_cd(best_h, max_sweeps=8, t_start=t_start, budget_s=budget_s, margin=15)
        if c_cd < best_c - 1e-13:
            best_h = h_cd
            best_c = c_cd

    # Phase 8: Final SLSQP polish
    if time_ok(25):
        sol = slsqp_optimize(best_h, maxiter=200, active_k=10, tol=1e-13)
        if sol is not None:
            c = exact_score(sol)
            if c < best_c - 1e-14:
                best_h = sol
                best_c = c

    # Final validation
    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)

    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            final_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 1:
            if np.isfinite(eval_result[0]):
                final_c = float(eval_result[0])
    except Exception:
        pass

    return best_h, final_c, len(best_h)
