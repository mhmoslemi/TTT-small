```python
import numpy as np
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def repair_feasible(vec):
    # Clip to [0,1] and adjust sum to n/2
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        # Need to add mass - add to entries with most room
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx])
            h[idx] += add
            remaining -= add
            if remaining <= 1e-15:
                break
    else:
        # Need to remove mass - remove from entries with most value
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-15:
                break
    # Final correction
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        for idx in range(n):
            if diff2 > 0 and h[idx] < 1.0:
                h[idx] = min(1.0, h[idx] + diff2)
                break
            elif diff2 < 0 and h[idx] > 0.0:
                h[idx] = max(0.0, h[idx] + diff2)
                break
    return h

def soft_gradient(h, beta):
    """Compute the gradient of the soft-max objective."""
    n = len(h)
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode="full")
    # Shift profile to center
    # profile[k] corresponds to lag m = k - (n-1)
    # Soft weights
    max_p = np.max(profile)
    exp_p = np.exp(beta * (profile - max_p))  # shift to avoid overflow
    w = exp_p / np.sum(exp_p)
    
    # Gradient: g[j] = sum_k w[k] * dR[k]/dh[j]
    # R[k] = sum_i h[i] * (1-h[i+m]) where m = k-(n-1)
    # dR[k]/dh[j] = (1-h[j+m])*[0<=j+m<n] - h[j-m]*[0<=j-m<n]
    g = np.zeros(n)
    center = n - 1
    for m in range(-(n-1), n):
        km = m + center  # = k
        wk = w[km] * dx  # include dx in the gradient
        # First term: (1-h[j+m]) for valid j
        if m >= 0:
            j_range = range(0, n - m)
            for j in j_range:
                g[j] += wk * (1.0 - h[j + m])
        else:
            j_range = range(0, n + m)  # m is negative, so n+m < n
            for j in j_range:
                g[j] += wk * (1.0 - h[j + m])
        # Second term: -h[j-m] for valid j
        if m <= 0:
            j_range = range(0, n + m)  # j-m >= 0 means j >= -m, but we need j-m < n
            # j-m in [0, n-1] means j in [m, m+n-1], intersect [0, n-1]
            # If m <= 0: j in [0, n-1] and j >= -m, so j in [-m, n-1]... wait
            # j-m >= 0 => j >= m. Since m <= 0, j >= 0 is already satisfied.
            # j-m < n => j < n+m. So j in [0, n+m-1]
            for j in range(0, n + m):
                g[j] -= wk * h[j - m]
        else:
            # j-m >= 0 => j >= m. j-m < n => j < m+n. So j in [m, n-1] (since m > 0)
            for j in range(m, n):
                g[j] -= wk * h[j - m]
    
    # Project onto sum=0
    g -= np.mean(g)
    return g
```

```python
import numpy as np
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx])
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-15:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-15:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmin(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def build_jacobian(n):
    """Build the Jacobian matrix J of shape (2n-1, n) where J[k,j] = dR[k]/dh[j]."""
    M = 2 * n - 1
    J = np.zeros((M, n))
    center = n - 1
    for m in range(-(n - 1), n):
        k = m + center
        if m >= 0:
            J[k, 0:n - m] += 1.0   # (1-h[j+m]) gives +1 for h[j]
            if n - m > 0:
                J[k, 0:n - m] -= 0.0  # no second term for this j range from first factor
        else:
            J[k, 0:n + m] += 1.0
        if m <= 0:
            J[k, 0:n + m] -= 1.0   # -h[j-m] gives -1 for h[j] when j in [0, n+m-1]
        else:
            J[k, m:n] -= 1.0       # -h[j-m] gives -1 for h[j] when j in [m, n-1]
    return J

def soft_gradient(h, beta):
    """Compute gradient of soft-max objective, projected to sum=0."""
    n = len(h)
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode="full")
    max_p = np.max(profile)
    exp_p = np.exp(beta * (profile - max_p))
    w = exp_p / np.sum(exp_p)
    J = build_jacobian(n)
    g = J.T @ w
    g *= dx
    g -= np.mean(g)
    return g

def line_search_alpha(h, v, curr_c, max_alpha=1.0):
    """Find optimal alpha in [0, max_alpha] minimizing exact_score(h + alpha*v)."""
    n = len(h)
    # Compute feasible upper bound
    alpha_hi = max_alpha
    pos = v > 1e-15
    neg = v < -1e-15
    if np.any(pos):
        alpha_hi = min(alpha_hi, np.min((1.0 - h[pos]) / v[pos]))
    if np.any(neg):
        alpha_hi = min(alpha_hi, np.min(h[neg] / (-v[neg])))
    alpha_hi = max(0.0, min(alpha_hi, 1.0))
    if alpha_hi < 1e-15:
        return 0.0, curr_c
    # Coarse grid
    best_a = 0.0
    best_c = curr_c
    n_grid = 20
    alphas = np.linspace(0, alpha_hi, n_grid + 1)
    for a in alphas:
        if a == 0:
            continue
        h_try = h + a * v
        if np.any(h_try < -1e-12) or np.any(h_try > 1 + 1e-12):
            continue
        h_try = np.clip(h_try, 0, 1)
        c_try = exact_score(h_try)
        if c_try < best_c - 1e-15:
            best_c = c_try
            best_a = a
    # Golden section refinement if we found improvement
    if best_a > 1e-10:
        gr = (np.sqrt(5.0) + 1.0) / 2.0
        a_lo = best_a - (alphas[1] - alphas[0])
        a_hi = best_a + (alphas[1] - alphas[0])
        a_lo = max(a_lo, 0.0)
        a_hi = min(a_hi, alpha_hi)
        if a_hi - a_lo < 1e-12:
            return best_a, best_c
        c = a_hi - (a_hi - a_lo) / gr
        b = a_lo + (a_hi - a_lo) / gr
        hc = np.clip(h + c * v, 0, 1)
        hb = np.clip(h + b * v, 0, 1)
        fc = exact_score(hc)
        fb = exact_score(hb)
        for _ in range(40):
            if fc < fb:
                a_hi = b
                b = c
                hb = hc
                fb = fc
                c = a_hi - (a_hi - a_lo) / gr
                hc = np.clip(h + c * v, 0, 1)
                fc = exact_score(hc)
            else:
                a_lo = c
                c = b
                hc = hb
                fc = fb
                b = a_lo + (a_hi - a_lo) / gr
                hb = np.clip(h + b * v, 0, 1)
                fb = exact_score(hb)
        a_opt = (a_lo + a_hi) / 2.0
        h_opt = np.clip(h + a_opt * v, 0, 1)
        c_opt = exact_score(h_opt)
        if c_opt < best_c:
            best_a = a_opt
            best_c = c_opt
    return best_a, best_c

def coordinate_pair_sweep(h, max_swaps=200):
    """Try all coordinate pairs (i,j) with h[i]<h[j], swap/transfer to minimize score."""
    n = len(h)
    curr_c = exact_score(h)
    swaps_done = 0
    for _ in range(max_swaps):
        improved = False
        # Try all pairs: transfer from j to i
        for j in range(n):
            for i in range(n):
                if i == j:
                    continue
                if h[j] <= h[i] + 1e-10:
                    continue
                # Transfer alpha from j to i
                alpha_max = min(1.0 - h[i], h[j])
                if alpha_max < 1e-10:
                    continue
                v = np.zeros(n)
                v[i] = 1.0
                v[j] = -1.0
                best_a, best_c = line_search_alpha(h, v, curr_c, max_alpha=alpha_max)
                if best_c < curr_c - 1e-14 and best_a > 1e-10:
                    h = np.clip(h + best_a * v, 0, 1)
                    h = repair_feasible(h)
                    curr_c = best_c
                    swaps_done += 1
                    improved = True
                    break
            if improved or swaps_done >= max_swaps:
                break
        if not improved:
            break
    return h, curr_c, swaps_done

def soft_gradient_descent(h, max_iters=5000, beta_start=30.0, beta_end=200.0, time_budget_fn=None):
    """Iterative soft-gradient with exact line search."""
    n = len(h)
    curr_c = exact_score(h)
    no_improve = 0
    for it in range(max_iters):
        if time_budget_fn is not None and time_budget_fn() < 30:
            break
        beta = beta_start + (beta_end - beta_start) * min(1.0, it / (max_iters * 0.8))
        v = soft_gradient(h, beta)
        vn = np.linalg.norm(v)
        if vn < 1e-15:
            break
        v /= vn
        best_a, best_c = line_search_alpha(h, v, curr_c, max_alpha=0.1)
        if best_c < curr_c - 1e-14 and best_a > 1e-10:
            h = np.clip(h + best_a * v, 0, 1)
            h = repair_feasible(h)
            curr_c = exact_score(h)
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 50:
                # Try smaller beta
                v2 = soft_gradient(h, beta * 0.5)
                vn2 = np.linalg.norm(v2)
                if vn2 > 1e-15:
                    v2 /= vn2
                    best_a2, best_c2 = line_search_alpha(h, v2, curr_c, max_alpha=0.1)
                    if best_c2 < curr_c - 1e-14 and best_a2 > 1e-10:
                        h = np.clip(h + best_a2 * v2, 0, 1)
                        h = repair_feasible(h)
                        curr_c = exact_score(h)
                        no_improve = 0
            if no_improve > 200:
                break
    return h, curr_c

def interp_to_n(h, new_n):
    """Interpolate h to new_n points and repair feasibility."""
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    fx = np.concatenate([[0.0], xs, [2.0]])
    fh = np.concatenate([[h[0]], h, [h[-1]]])
    hn = np.interp(xt, fx, fh)
    return repair_feasible(hn)

def binary_local_search(h_bin, max_swaps=500):
    """Local search on binary vectors via coordinate swaps."""
    n = len(h_bin)
    h = h_bin.copy()
    curr_c = exact_score(h)
    for _ in range(max_swaps):
        improved = False
        # Try all 0->1 and 1->0 swaps
        zeros = np.where(h < 0.5)[0]
        ones = np.where(h >= 0.5)[0]
        for i in zeros:
            for j in ones:
                h_try = h.copy()
                h_try[i] = 1.0
                h_try[j] = 0.0
                c_try = exact_score(h_try)
                if c_try < curr_c - 1e-12:
                    h = h_try
                    curr_c = c_try
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
    return h, curr_c
```
