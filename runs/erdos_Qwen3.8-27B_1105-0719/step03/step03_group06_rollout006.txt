c[k] = sum_j a[j+k] * conj(v[j]) for valid k

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(-(1.0 - h))
        remaining = diff
        for i in range(n):
            idx = order[i]
            cap = 1.0 - h[idx]
            if cap <= 1e-15:
                continue
            add = min(remaining, cap)
            h[idx] += add
            remaining -= add
            if remaining <= 1e-14:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            if h[idx] <= 1e-15:
                continue
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-14:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] = float(np.clip(h[idx] + diff2, 0.0, 1.0))
    return h

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def grad_row(h, m, n):
    dx = 2.0 / n
    d = m - (n - 1)
    if d == 0:
        return dx * (1.0 - 2.0 * h)
    elif d > 0:
        g = np.zeros(n)
        g[d:] += dx * (1.0 - h[:-d])
        g[:-d] -= dx * h[d:]
        return g
    else:
        p = -d
        g = np.zeros(n)
        g[:-p] += dx * (1.0 - h[p:])
        g[p:] -= dx * h[:-p]
        return g

def interp_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    edges_old = np.linspace(0.0, 2.0, n + 1)
    edges_new = np.linspace(0.0, 2.0, new_n + 1)
    h_new = np.zeros(new_n)
    for k in range(new_n):
        a, b = edges_new[k], edges_new[k + 1]
        total = 0.0
        for j in range(n):
            lo = max(a, edges_old[j])
            hi = min(b, edges_old[j + 1])
            if hi > lo:
                total += h[j] * (hi - lo)
        h_new[k] = total / (b - a)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

def active_descent_nesterov(h, max_iters=300, active_tol=0.003):
    """Nesterov-accelerated active descent."""
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    y = h.copy()  # Nesterov momentum variable
    beta = 0.9
    
    for it in range(max_iters):
        profile = compute_profile(h)
        c = float(profile.max())
        thresh = c - active_tol
        active = np.where(profile >= thresh)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(profile))])
        if len(active) > 25:
            top = np.argsort(profile[active])[-25:]
            active = active[top]
        
        # Compute subgradient as sum of active row gradients
        g = np.zeros(n)
        for m in active:
            g += grad_row(h, m, n)
        g -= g.mean()  # project onto sum=0 subspace
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-14:
            break
        
        # Nesterov step
        y = h + beta * (h - y + 1.0/(it+2) * (h - y))
        # Actually standard Nesterov: y = h + (t-1)/t * (h - y_prev)
        # Let me use a simpler momentum approach
        
        # Simple momentum
        y = (1 - beta) * h + beta * y
        g_y = np.zeros(n)
        for m in active:
            g_y += grad_row(y, m, n)
        g_y -= g_y.mean()
        
        # Step size (Polyak-like or decreasing)
        step = 0.01 / (1 + 0.05 * it)
        d = -g_y / max(gnorm, 1e-10)
        
        # Projected step with line search
        # Find max alpha that keeps in [0,1]
        pos_mask = d > 1e-12
        neg_mask = d < -1e-12
        alpha_max = 1.0
        if pos_mask.any():
            alpha_max = min(alpha_max, float(((1.0 - h[pos_mask]) / d[pos_mask]).min()))
        if neg_mask.any():
            alpha_max = min(alpha_max, float((h[neg_mask] / (-d[neg_mask])).min()))
        alpha_max = max(alpha_max, 1e-10)
        alpha = min(step, alpha_max)
        
        h_new = repair_feasible(h + alpha * d)
        c_new = exact_score(h_new)
        
        if c_new < best_c:
            best_c = c_new
        
        if c_new < c - 1e-15:
            h = h_new
        else:
            # Try smaller step
            alpha = min(step * 0.3, alpha_max)
            h_new = repair_feasible(h + alpha * d)
            c_new = exact_score(h_new)
            if c_new < c - 1e-15:
                h = h_new
                if c_new < best_c:
                    best_c = c_new
            else:
                # Try even smaller or break
                alpha = min(step * 0.1, alpha_max)
                h_new = repair_feasible(h + alpha * d)
                c_new = exact_score(h_new)
                if c_new < c - 1e-15:
                    h = h_new
                    if c_new < best_c:
                        best_c = c_new
                else:
                    break
    
    return h, best_c

    n = len(h)

        c = float(profile.max())

            top = np.argsort(profile[active])[-20:]
            active = active[top]
        g = np.zeros(n)
        for m in active:
            g += grad_row(h, m, n)
        g -= g.mean()
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            break
        momentum_vec = momentum * momentum_vec + g
        mv_norm = np.linalg.norm(momentum_vec)
        if mv_norm < 1e-15:
            break
        d = -momentum_vec / mv_norm
        pos_mask = d > 1e-12
        neg_mask = d < -1e-12
        alpha_max = 1.0

def slsqp_epigraph(h_start, maxiter=200):
    n = len(h_start)
    dx = 2.0 / n
    M = 2 * n - 1
    c0 = exact_score(h_start)
    t0 = c0 + 1e-4
    
    def obj(x):
        return x[-1]
    
    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    
    def eq_fun(x):
        return x[:-1].sum() - n / 2.0
    
    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g
    
    def ineq_fun(x):
        h = x[:-1]
        t = x[-1]
        profile = np.correlate(h, 1 - h, mode="full") * dx
        return t - profile
    
    def ineq_jac(x):
        h = x[:-1]
        J = np.zeros((M, n + 1))
        for m in range(M):
            J[m, :n] = -grad_row(h, m, n)
        J[:, -1] = 1.0
        return J
    
    x0 = np.concatenate([h_start, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 2.0)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None and np.isfinite(res.x).all():
            h_new = repair_feasible(res.x[:-1])
            c_new = exact_score(h_new)
            if c_new <= c0 + 1e-8:
                return h_new
    except Exception:
        pass
    return None

def binarode_seed(n, freq, rng):
    """Generate structured binary seed with given frequency."""
    x = np.linspace(0, 2, n, endpoint=False)
    # Use a sum of sinusoids at different frequencies
    val = np.zeros(n)
    for k in range(1, 6):
        val += np.sin(2 * np.pi * freq * k * x / 1.0) * (1.0 / k)
    # Threshold at median to get exactly n/2 ones
    thresh = np.median(val)
    h = (val >= thresh).astype(float)
    ones = int(h.sum())
    if ones > n // 2:
        # Flip some 1s to 0s
        one_idx = np.where(h == 1.0)[0]
        excess = ones - n // 2
        to_flip = rng.choice(one_idx, excess, replace=False)
        h[to_flip] = 0.0
    elif ones < n // 2:
        zero_idx = np.where(h == 0.0)[0]
        deficit = n // 2 - ones
        to_flip = rng.choice(zero_idx, deficit, replace=False)
        h[to_flip] = 1.0
    else:
        pass
    return repair_feasible(h)

def golden_seed(n, rng):
    """Golden ratio based seed."""
    idx = np.arange(n)
    vals = (idx * 0.618033988749895) % 1.0
    # Add small random perturbation
    vals += rng.normal(0, 0.05, n)
    top = np.argsort(vals)[-n//2:]
    h = np.zeros(n)
    h[top] = 1.0
    return repair_feasible(h)
