import numpy as np
from scipy.optimize import minimize, linprog
from scipy.special import logsumexp
import time

def project_feasible(x, target_sum):
    """Project x onto {h: 0<=h<=1, sum(h)=target_sum}"""
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    # Fix any numerical issues
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            adj = diff / mask.sum()
            h[mask] = np.clip(h[mask] + adj, 0.0, 1.0)
    return h

def compute_corrs(h):
    """Compute all correlation values (before multiplying by dx)"""
    return np.correlate(h, 1.0 - h, mode='full')

def c5_of_h(h, dx):
    return np.correlate(h, 1.0 - h, mode='full').max() * dx

def grad_corr_k(h, k, n):
    """Gradient of correlation k with respect to h"""
    # corr_k = sum_i h[i] * (1 - h[i+k]) for valid i
    # d/dh[j] = sum over all pairs where j appears
    offset = k - (n - 1)  # shift index
    grad = np.zeros(n)
    i_lo = max(0, -offset)
    i_hi = min(n - 1, n - 1 - offset)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    # h[idx] appears as the first factor: derivative is (1-h[idx+offset])
    grad[idx] += (1.0 - h[idx + offset])
    # h[idx+offset] appears as the second factor: derivative is -h[idx]
    grad[idx + offset] += (-h[idx])
    return grad

def smooth_max_grad(h, dx, beta):
    """Gradient of logsumexp(beta * corrs) / beta * dx"""
    n = len(h)
    corrs = compute_corrs(h)
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones_like(corrs) / len(corrs)
    else:
        w /= w_sum
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] > 1e-12:
            grad += w[k] * grad_corr_k(h, k, n)
    return grad * dx

def exact_line_search(h, d, target_sum, dx, alpha_lo, alpha_hi, n_eval=50):
    """Find best alpha in [alpha_lo, alpha_hi] for direction d using exact evaluation"""
    best_alpha = 0.0
    best_c5 = c5_of_h(h, dx)
    
    # Golden section search combined with grid
    # First do a coarse grid
    alphas = np.linspace(alpha_lo, alpha_hi, n_eval)
    c5_vals = np.array([c5_of_h(project_feasible(h + a * d, target_sum), dx) for a in alphas])
    best_idx = np.argmin(c5_vals)
    best_c5 = c5_vals[best_idx]
    best_alpha = alphas[best_idx]
    
    # Refine around best
    if best_idx > 0 and best_idx < len(alphas) - 1:
        alphas2 = np.linspace(alphas[best_idx-1], alphas[best_idx+1], 30)
        c5_vals2 = np.array([c5_of_h(project_feasible(h + a * d, target_sum), dx) for a in alphas2])
        best_idx2 = np.argmin(c5_vals2)
        if c5_vals2[best_idx2] < best_c5:
            best_c5 = c5_vals2[best_idx2]
            best_alpha = alphas2[best_idx2]
    
    return best_alpha, best_c5

def lp_direction(h, dx, target_sum, n_active, step_size):
    """Find a descent direction via LP"""
    n = len(h)
    corrs = compute_corrs(h)
    max_c = corrs.max()
    
    # Get top active constraints
    n_act = min(n_active, 2*n - 1)
    active_idx = np.argsort(corrs)[-n_act:]
    
    # LP: min t s.t. grad_k^T d + c_k*dx <= t for active k, sum(d)=0, |d_i| <= step
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_rows = []
    b_rows = []
    for k in active_idx:
        row = np.zeros(n + 1)
        row[:n] = dx * grad_corr_k(h, k, n)
        row[n] = -1.0
        A_rows.append(row)
        b_rows.append(-corrs[k] * dx - 1e-12)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    bounds = [(max(-h[i], -step_size), min(1.0 - h[i], step_size)) for i in range(n)]
    bounds.append((0.0, None))
    
    try:
        res = linprog(c_obj, A_ub=np.array(A_rows), b_ub=np.array(b_rows),
                      A_eq=A_eq, b_eq=np.zeros(1), bounds=bounds, method='highs')
        if res.status == 0:
            return res.x[:n]
    except:
        pass
    return None

def slsqp_solve(h0, dx, target_sum, beta, maxiter=1000):
    """SLSQP with smooth max objective"""
    n = len(h0)
    
    def obj(x):
        corrs = compute_corrs(x)
        return logsumexp(beta * corrs) / beta * dx
    
    def grad_obj(x):
        return smooth_max_grad(x, dx, beta)
    
    def c_fun(x):
        return x.sum() - target_sum
    
    def c_jac(x):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': c_fun, 'jac': c_jac}]
    bounds = [(0.0, 1.0)] * n
    
    try:
        res = minimize(obj, h0, jac=grad_obj, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        h_out = project_feasible(res.x, target_sum)
    except:
        h_out = project_feasible(h0, target_sum)
    return h_out, c5_of_h(h_out, dx)

def frank_wolfe(h0, dx, target_sum, max_iter=5000, beta=100.0):
    """Frank-Wolfe / conditional gradient method"""
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    no_improve = 0
    
    for it in range(max_iter):
        c5_val = c5_of_h(h, dx)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 500:
            break
        
        # Compute subgradient (weighted gradient of active constraints)
        corrs = compute_corrs(h)
        shifted = corrs - corrs.max()
        w = np.exp(beta * shifted)
        w /= w.sum()
        
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] > 1e-12:
                grad += w[k] * grad_corr_k(h, k, n)
        grad *= dx
        
        # FW step: find y in {0<=y<=1, sum(y)=target_sum} that minimizes grad^T y
        # This is: pick the target_sum positions with smallest grad, set them to 1, rest to 0
        # But we need sum(y) = target_sum exactly
        # Optimal: sort grad, set top (n - target_sum) to 0, rest to 1
        # Actually: minimize grad^T y s.t. 0<=y<=1, sum(y)=target_sum
        # Optimal: y[i] = 1 if grad[i] is among smallest target_sum values, 0 otherwise
        # (with possible fractional value at the boundary)
        sorted_idx = np.argsort(grad)
        y = np.zeros(n)
        # We need sum(y) = target_sum
        # Greedy: fill the smallest grad positions with 1
        n_fill = int(target_sum)
        frac = target_sum - n_fill
        y[sorted_idx[:n_fill]] = 1.0
        if frac > 1e-10:
            y[sorted_idx[n_fill]] = frac
        
        # Step
        gamma = 2.0 / (it + 2)  # standard FW step size
        h_new = (1 - gamma) * h + gamma * y
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_feasible(h_new, target_sum)
        
        new_c5 = c5_of_h(h_new, dx)
        if new_c5 < c5_val:
            h = h_new
        else:
            # Try smaller step
            for g in [gamma/2, gamma/4, gamma/10]:
                h_try = (1 - g) * h + g * y
                h_try = project_feasible(np.clip(h_try, 0, 1), target_sum)
                if c5_of_h(h_try, dx) < c5_val:
                    h = h_try
                    break
            else:
                if gamma < 1e-4:
                    break
                gamma *= 0.5
                continue
    return best_h, best_c5

def lp_polish_with_line_search(h0, dx, target_sum, max_iters=200, init_step=0.1, n_active=40):
    """LP-based direction finding + exact line search"""
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    step = init_step
    stall = 0
    
    for it in range(max_iters):
        c5_val = c5_of_h(h, dx)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 50:
            step *= 0.7
            stall = 0
            if step < 1e-6:
                break
        
        d = lp_direction(h, dx, target_sum, n_active, step)
        if d is None or np.max(np.abs(d)) < 1e-13:
            break
        
        # Exact line search
        alpha_lo, alpha_hi = -1.0, 1.0
        alpha_opt, c5_ls = exact_line_search(h, d, target_sum, dx, alpha_lo, alpha_hi, n_eval=40)
        
        if c5_ls < c5_val - 1e-13:
            h = project_feasible(h + alpha_opt * d, target_sum)
        else:
            # Try half step
            h_try = project_feasible(h + 0.5 * alpha_opt * d, target_sum)
            c5_try = c5_of_h(h_try, dx)
            if c5_try < c5_val - 1e-13:
                h = h_try
            else:
                step *= 0.5
                if step < 1e-7:
                    break
    return best_h, best_c5

def block_descent(h0, dx, target_sum, block_size=4, max_iters=500):
    """Block coordinate descent with 1D/low-dim optimization per block"""
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    
    for it in range(max_iters):
        improved = False
        # Process blocks
        for start in range(0, n, block_size):
            end = min(start + block_size, n)
            idx = np.arange(start, end)
            b = end - start
            
            # For this block, try small perturbations
            # The change in corr_k from changing h[idx] by delta is:
            # For each k, delta_c_k = sum over affected terms
            # This is complex, so let's just try grid of perturbations
            for trial in range(20):
                delta = (np.random.randn(b) * 0.05)
                h_try = h.copy()
                h_try[idx] = np.clip(h_try[idx] + delta, 0, 1)
                h_try = project_feasible(h_try, target_sum)
                c5_try = c5_of_h(h_try, dx)
                if c5_try < best_c5 - 1e-13:
                    h = h_try

def momentum_method(h0, dx, target_sum, max_iter=10000, beta_smooth=50.0):
    """Momentum-based subgradient method"""
    n = len(h0)
    h = h0.copy()

    best_h = h.copy()

            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 2000:
            break
        
        corrs = compute_corrs(h)
        shifted = corrs - corrs.max()
        w = np.exp(beta_smooth * shifted)
        w /= w.sum()
        
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] > 1e-12:
                grad += w[k] * grad_corr_k(h, k, n)
        grad *= dx
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        v = beta_mom * v - lr * grad / (gnorm + 1e-10)

        new_c5 = c5_of_h(h_new, dx)

        else:
            lr *= 0.8
            v = beta_mom * v - lr * grad / (gnorm + 1e-10)
            h_new = project_feasible(h + v, target_sum)
            h = h_new
    
    return best_h, best_c5

def pair_swap_optimal(h0, dx, target_sum, n_top=30, passes=5):
    """For each pair, find optimal transfer amount via golden section"""
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    
    for _pass in range(passes):
        improved = False
        corrs = compute_corrs(h)
        
        # Find indices to move from and to
        grad = smooth_max_grad(h, dx, 200.0)
        dec_idx = np.argsort(grad)[-n_top:]
        inc_idx = np.argsort(grad)[:n_top]
        
        for i in dec_idx:
            for j in inc_idx:
                if i == j:
                    continue
                # Max transfer: min(h[i], 1-h[j])
                eps_max = min(h[i], 1.0 - h[j])
                if eps_max < 1e-10:
                    continue
                
                # Golden section search for optimal eps
                gr = (np.sqrt(5.0) - 1.0) / 2.0
                a, b = 0.0, eps_max
                c_pt = b - gr * (b - a)
                d_pt = a + gr * (b - a)
                
                def f_eps(eps):
                    h_try = h.copy()
                    h_try[i] = h[i] - eps
                    h_try[j] = h[j] + eps
                    return c5_of_h(h_try, dx)
                
                fc, fd = f_eps(c_pt), f_eps(d_pt)
                for _ in range(60):
                    if b - a < 1e-10:
                        break
                    if fc < fd:
                        b = d_pt
                        d_pt = c_pt
                        fd = fc
                        c_pt = b - gr * (b - a)
                        fc = f_eps(c_pt)
                    else:
                        a = c_pt
                        c_pt = d_pt
                        fc = fd
                        d_pt = a + gr * (b - a)
                        fd = f_eps(d_pt)
                
                eps_opt = (a + b) / 2.0
                if eps_opt > 1e-8:
                    h_try = h.copy()
                    h_try[i] -= eps_opt
                    h_try[j] += eps_opt
                    h_try = np.clip(h_try, 0, 1)
                    h_try = project_feasible(h_try, target_sum)
                    c5_try = c5_of_h(h_try, dx)
                    if c5_try < best_c5 - 1e-13:
                        h = h_try
                        best_c5 = c5_try
                        best_h = h.copy()
                        improved = True
                        corrs = compute_corrs(h)
                        break
            if improved:
                break
        if not improved:
            break
    return best_h, best_c5

def diverse_starts(n, target_sum, rng):
    """Generate diverse starting points"""
    starts = []
    
    # Random
    h = rng.rand(n)
    starts.append(project_feasible(h, target_sum))
    
    # Periodic patterns
    for period in [3, 4, 5, 7, 11, 13]:
        h = np.zeros(n)
        fill = max(1, int(target_sum / n * period))
        for i in range(n):
            h[i] = 1.0 if (i % period) < fill else 0.0
        starts.append(project_feasible(h, target_sum))
    
    # Cosine-based
    x = np.linspace(0, 2, n)
    for freq in [1, 2, 3, 5]:
        h = 0.5 + 0.5 * np.cos(2 * np.pi * freq * x / 2.0)
        starts.append(project_feasible(h, target_sum))
    
    # Two-level: random assignment of 0 and 1
    n_ones = int(target_sum)
    idx = rng.choice(n, size=n_ones, replace=False)
    h = np.zeros(n)
    h[idx] = 1.0
    starts.append(project_feasible(h, target_sum))
    
    return starts

def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def c5_of_h(h, dx):
    return compute_corrs(h).max() * dx


def grad_corr_k(h, k, n):
    offset = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -offset)
    i_hi = min(n - 1, n - 1 - offset)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx] += (1.0 - h[idx + offset])
    grad[idx + offset] += (-h[idx])
    return grad

        if gnorm < 1e-14:
            break
        v = beta_mom * v - lr * grad / (gnorm + 1e-10)
        h_new = project_feasible(h + v, target_sum)
        new_c5 = c5_of_h(h_new, dx)
        if new_c5 <= c5_val + 1e-13:
            h = h_new
        else:
            lr *= 0.8
            v = beta_mom * v - lr * grad / (gnorm + 1e-10)
            h = project_feasible(h + v, target_sum)
    return best_h, best_c5


def diverse_starts(n, target_sum, rng):
    starts = []
    h = rng.rand(n)
    starts.append(project_feasible(h, target_sum))
    for period in [3, 4, 5, 7]:
        h = np.zeros(n)
        fill = max(1, int(target_sum / n * period))
        for i in range(n):
            h[i] = 1.0 if (i % period) < fill else 0.0
        starts.append(project_feasible(h, target_sum))
    x = np.linspace(0, 2, n)
    for freq in [1, 2, 3]:
        h = 0.5 + 0.3 * np.cos(2 * np.pi * freq * x)
        starts.append(project_feasible(h, target_sum))
    n_ones = int(target_sum)
    idx = rng.choice(n, size=n_ones, replace=False)
    h = np.zeros(n)
    h[idx] = 1.0
    starts.append(project_feasible(h, target_sum))
    return starts


def full_optimize(h0, dx, target_sum, time_left_fn, label=""):
    best_h = h0.copy()
    best_c5 = c5_of_h(h0, dx)
    for beta in [100, 300, 800, 2000, 5000]:
        if time_left_fn() < 8:
            break
        h_opt, c5_val = slsqp_solve(best_h, dx, target_sum, beta, 600)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_opt
    if time_left_fn() > 25:
        h_lp, c5_lp = lp_polish_ls(best_h, dx, target_sum, max_iters=200, init_step=0.08, n_active=50)
        if c5_lp < best_c5:
            best_c5 = c5_lp
            best_h = h_lp
    if time_left_fn() > 25:
        h_fw, c5_fw = frank_wolfe(best_h, dx, target_sum, max_iter=2500, beta=80.0)
        if c5_fw < best_c5:
            best_c5 = c5_fw
            best_h = h_fw
    if time_left_fn() > 15:
        h_ps, c5_ps = pair_swap_optimal(best_h, dx, target_sum, n_top=25, passes=4)
        if c5_ps < best_c5:
            best_c5 = c5_ps
            best_h = h_ps
    if time_left_fn() > 25:
        h_mom, c5_mom = momentum_method(best_h, dx, target_sum, max_iter=6000, beta_sm=40.0)
        if c5_mom < best_c5:
            best_c5 = c5_mom
            best_h = h_mom
    if time_left_fn() > 15:
        h_f, c5_f = slsqp_solve(best_h, dx, target_sum, 3000, 800)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f
    if time_left_fn() > 12:
        h_lp2, c5_lp2 = lp_polish_ls(best_h, dx, target_sum, max_iters=120, init_step=0.05, n_active=35)
        if c5_lp2 < best_c5:
            best_c5 = c5_lp2
            best_h = h_lp2
    if time_left_fn() > 8:
        h_ps2, c5_ps2 = pair_swap_optimal(best_h, dx, target_sum, n_top=20, passes=3)
        if c5_ps2 < best_c5:
            best_c5 = c5_ps2
            best_h = h_ps2
    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s - (time.time() - t0)

    best_h = None
    best_c5 = float('inf')

    h0 = np.array(initial_h_values, dtype=float)
    n = len(h0)
    dx = 2.0 / n
    target_sum = n / 2.0
    h0 = project_feasible(h0, target_sum)

    h_opt, c5_val = full_optimize(h0, dx, target_sum, time_left, "main")
    best_h = h_opt.copy()
    best_c5 = c5_val

    for trial in range(10):
        if time_left() < 20:
            break
        if trial % 5 == 0:
            pert = 0.04 * (rng.random(n) - 0.5)
        elif trial % 5 == 1:
            pert = 0.12 * (rng.random(n) - 0.5)
        elif trial % 5 == 2:
            idx = rng.choice(n, size=max(5, n // 10), replace=False)
            pert = np.zeros(n)
            pert[idx] = rng.choice([-1, 1], size=len(idx)) * 0.06
        elif trial % 5 == 3:
            pert = 0.2 * (rng.random(n) - 0.5)
        else:
            idx = rng.choice(n, size=n // 15, replace=False)
            pert = np.zeros(n)
            for i in idx:
                pert[i] = (1.0 - best_h[i]) - best_h[i]
        h_try = project_feasible(best_h + pert, target_sum)
        h_opt2, c5_val2 = full_optimize(h_try, dx, target_sum, time_left, f"pert{trial}")
        if c5_val2 < best_c5:
            best_c5 = c5_val2
            best_h = h_opt2.copy()

    n_current = len(best_h)
    test_ns = []
    for delta in [-8, -6, -4, -2, 2, 4, 6, 8, -12, 12]:
        nn = n_current + delta
        if nn >= 40 and nn <= 260 and nn % 2 == 0:
            test_ns.append(nn)
    test_ns = sorted(set(test_ns), key=lambda x: abs(x - n_current))

    for n_try in test_ns[:8]:
        if time_left() < 25:
            break
        if n_try == n_current:
            continue
        dx_try = 2.0 / n_try
        ts_try = n_try / 2.0
        x_orig = np.linspace(0, 2, len(best_h), endpoint=False) + 1.0 / len(best_h)
        x_new = np.linspace(0, 2, n_try, endpoint=False) + 1.0 / n_try
        h_interp = np.interp(x_new, x_orig, best_h)
        h_interp = project_feasible(h_interp, ts_try)
        h_opt3, c5_val3 = full_optimize(h_interp, dx_try, ts_try, time_left, f"n{n_try}")
        if c5_val3 < best_c5:
            best_c5 = c5_val3
            best_h = h_opt3.copy()

    n_best = len(best_h)
    if time_left() > 50:
        dx_best = 2.0 / n_best
        ts_best = n_best / 2.0
        starts = diverse_starts(n_best, ts_best, rng)
        for s_idx, h_start in enumerate(starts):
            if time_left() < 20:
                break
            h_opt4, c5_val4 = full_optimize(h_start, dx_best, ts_best, time_left, f"div{s_idx}")
            if c5_val4 < best_c5:
                best_c5 = c5_val4
                best_h = h_opt4.copy()

    if time_left() > 25 and best_h is not None:
        n_b = len(best_h)
        dx_b = 2.0 / n_b
        ts_b = n_b / 2.0
        for rnd in range(4):
            if time_left() < 8:
                break
            h_lp, c5_lp = lp_polish_ls(best_h, dx_b, ts_b, max_iters=100, init_step=0.04, n_active=40)
            if c5_lp < best_c5:
                best_c5 = c5_lp
                best_h = h_lp
            h_ps, c5_ps = pair_swap_optimal(best_h, dx_b, ts_b, n_top=20, passes=3)
            if c5_ps < best_c5:
                best_c5 = c5_ps
                best_h = h_ps

    if best_h is None:
        best_h = h0
        best_c5 = c5_of_h(h0, 2.0 / len(h0))
    else:
        n_final = len(best_h)
        best_h = project_feasible(best_h, n_final / 2.0)
        best_c5 = c5_of_h(best_h, 2.0 / n_final)

    return best_h, best_c5, len(best_h)
