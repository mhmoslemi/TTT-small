import numpy as np
from scipy.optimize import linprog
import time

def c5_of_h(h, dx):
    """Compute C5 = dx * max corr_k"""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * dx

def corr_of_h(h):
    """Compute all correlations without dx"""
    return np.correlate(h, 1.0 - h, mode='full')

def grad_lag(h, k, n):
    """Gradient of corr_k with respect to h"""
    offset = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -offset)
    i_hi = min(n - 1, n - 1 - offset)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx + offset] += (1.0 - h[idx])
    grad[idx] += (-h[idx + offset])
    return grad

def project_feasible(z, target):
    """Project z onto {h: 0<=h<=1, sum(h)=target}"""
    n = len(z)
    lo = -2.0
    hi = 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        h = np.clip(z - mid, 0.0, 1.0)
        if h.sum() > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def solve_soft_slack_lp(h, dx, active_idx, corrs, top_n, alpha, n):
    """
    Solve LP with soft slacks to find a direction that reduces the max correlation.
    Variables: delta (n), s (K), q (1)
    min: rho_q * q + sum rho_j * s_j
    s.t.: dx * grad_j . delta + s_j - q <= -r_j  for j in active
         sum delta = 0
         -h_i * alpha <= delta_i <= (1-h_i) * alpha
    """
    K = len(active_idx)
    m_vars = n + K + 1
    
    # Objective
    c_obj = np.zeros(m_vars)
    c_obj[-1] = 0.35  # rho_q
    # Slack penalties: higher for top lags
    for j in range(K):
        c_obj[n + j] = 5.0 if j < top_n else 1.5
    
    # Inequality constraints: dx * grad_j . delta + s_j - q <= -r_j
    A_ub = np.zeros((K, m_vars))
    b_ub = np.zeros(K)
    for j, k in enumerate(active_idx):
        g = grad_lag(h, k, n)
        A_ub[j, :n] = dx * g
        A_ub[j, n + j] = 1.0
        A_ub[j, -1] = -1.0
        b_ub[j] = -corrs[k]
    
    # Equality: sum delta = 0
    A_eq = np.zeros((1, m_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds
    bounds = [(None, None)] * m_vars
    for i in range(n):
        bounds[i] = (-h[i] * alpha, (1.0 - h[i]) * alpha)
    for j in range(K):
        bounds[n + j] = (0.0, None)
    bounds[-1] = (None, None)
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.status != 0:
            return None
        return res.x[:n]
    except Exception:
        return None

def solve_linear_sum_lp(h, dx, active_idx, corrs, alpha, n):
    """
    Solve LP that minimizes weighted sum of correlations for active lags.
    Variables: delta (n), q (1) -- actually just delta
    min: sum_j w_j * dx * grad_j . delta
    s.t.: sum delta = 0
         -h_i * alpha <= delta_i <= (1-h_i) * alpha
    """

    m_vars = n
    
    # Objective: weighted sum of gradients
    # Weights: exponential decay from top
    weights = np.exp(-np.arange(K) * 0.1)
    weights /= weights.sum()
    
    c_obj = np.zeros(n)
    for j, k in enumerate(active_idx):
        g = grad_lag(h, k, n)
        c_obj += weights[j] * dx * g
    
    # Equality: sum delta = 0
    A_eq = np.zeros((1, m_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds
    bounds = [(-h[i] * alpha, (1.0 - h[i]) * alpha) for i in range(n)]
    
    try:

                      bounds=bounds, method='highs', options={'presolve': True})
        if res.status != 0:
            return None
        return res.x[:n]
    except Exception:
        return None

    diff = target - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def solve_soft_slack_lp(h, dx, active_idx, corrs, top_n, alpha, n):
    K = len(active_idx)
    if K == 0:
        return None
    m_vars = n + K + 1
    c_obj = np.zeros(m_vars)
    c_obj[-1] = 0.35
    for j in range(K):
        c_obj[n + j] = 5.0 if j < top_n else 1.5
    A_ub = np.zeros((K, m_vars))
    b_ub = np.zeros(K)
    for j, k in enumerate(active_idx):
        g = grad_lag(h, k, n)
        A_ub[j, :n] = dx * g
        A_ub[j, n + j] = 1.0
        A_ub[j, -1] = -1.0
        b_ub[j] = -corrs[k]
    A_eq = np.zeros((1, m_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for i in range(n):
        bounds.append((-h[i] * alpha, (1.0 - h[i]) * alpha))
    for j in range(K):
        bounds.append((0.0, None))
    bounds.append((None, None))
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.status != 0:
            return None
        return res.x[:n]
    except Exception:
        return None

def solve_linear_sum_lp(h, dx, active_idx, alpha, n):
    K = len(active_idx)
    if K == 0:
        return None
    m_vars = n
    weights = np.exp(-np.arange(K) * 0.08)
    weights /= weights.sum()
    c_obj = np.zeros(n)
    for j, k in enumerate(active_idx):
        g = grad_lag(h, k, n)
        c_obj += weights[j] * dx * g
    A_eq = np.zeros((1, m_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(-h[i] * alpha, (1.0 - h[i]) * alpha) for i in range(n)]
    try:
        res = linprog(c_obj, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.status != 0:
            return None
        return res.x[:n]
    except Exception:
        return None

def line_search(h_cur, c5_cur, direction, dx, target):
    best_h = h_cur
    best_c5 = c5_cur
    for t in [1.0, 0.8, 0.6, 0.5, 0.4, 0.3, 0.2, 0.15, 0.1, 0.07, 0.05, 0.03, 0.01]:
        h_try = h_cur + t * direction
        h_try = project_feasible(h_try, target)
        c5_try = c5_of_h(h_try, dx)
        if c5_try < best_c5 - 1e-15:
            best_h = h_try
            best_c5 = c5_try
    return best_h, best_c5

def lp_optimize(h, dx, target, alpha, n_inner, budget_s, t0):
    n = len(h)
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    cur = h.copy()
    cur_c5 = best_c5
    for _ in range(n_inner):
        if budget_s - (time.time() - t0) < 3.0:
            break
        corrs = corr_of_h(cur)
        max_c = corrs.max()
        n_total = 2 * n - 1
        n_top = max(20, min(50, n_total // 4))
        top_idx = np.argsort(corrs)[-n_top:]
        margin = max(0.05, 0.03 * max_c)
        active_mask = corrs >= (max_c - margin)
        active_mask[top_idx[:n_top // 2]] = True
        active_idx = np.where(active_mask)[0]
        if len(active_idx) < 5:
            active_idx = top_idx
        delta = solve_soft_slack_lp(cur, dx, active_idx, corrs, n_top, alpha, n)
        if delta is None:
            delta = solve_linear_sum_lp(cur, dx, active_idx, alpha, n)
        if delta is None:
            alpha *= 0.5
            if alpha < 0.003:
                break
            continue
        h_new, c5_new = line_search(cur, cur_c5, delta, dx, target)
        if c5_new < cur_c5 - 1e-15:
            cur = h_new
            cur_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            alpha = min(0.25, alpha * 1.08)
        else:
            alpha *= 0.6
            if alpha < 0.003:
                break
    return best_h, best_c5

def pair_swap_refine(h, dx, target, n, rng, max_outer, budget_s, t0):
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    for outer in range(max_outer):
        if budget_s - (time.time() - t0) < 5.0:
            break
        improved = False
        n_pairs = 300
        for _ in range(n_pairs):
            if budget_s - (time.time() - t0) < 3.0:
                break
            p = rng.randint(0, n)
            r = rng.random()
            if r < 0.4:
                q = (p + rng.randint(1, max(2, n // 4))) % n
            elif r < 0.8:
                q = (p + rng.randint(1, max(2, n // 2))) % n
            else:
                q = rng.randint(0, n)
            if p == q:
                continue
            a = best_h[p]
            b = best_h[q]
            L = max(-a, b - 1.0)
            U = min(1.0 - a, b)
            if U - L < 1e-12:
                continue
            ts = np.linspace(L, U, 11)
            best_t = 0.0
            best_c5_local = best_c5
            for t in ts:
                if abs(t) < 1e-12:
                    continue
                h_try = best_h.copy()
                h_try[p] = a + t
                h_try[q] = b - t
                c5_try = c5_of_h(h_try, dx)
                if c5_try < best_c5_local - 1e-14:
                    best_c5_local = c5_try
                    best_t = t
            if best_t != 0.0:
                span = (U - L) / 10.0
                fine_ts = np.linspace(best_t - span, best_t + span, 17)
                fine_ts = fine_ts[(fine_ts >= L) & (fine_ts <= U)]
                for t in fine_ts:
                    if abs(t) < 1e-12:
                        continue
                    h_try = best_h.copy()
                    h_try[p] = a + t
                    h_try[q] = b - t
                    c5_try = c5_of_h(h_try, dx)
                    if c5_try < best_c5_local - 1e-14:
                        best_c5_local = c5_try
                        best_t = t
                best_h[p] = a + best_t
                best_h[q] = b - best_t
                best_c5 = best_c5_local
                improved = True
        if not improved:
            break
    return best_h, best_c5

def perturb_and_optimize(h, dx, target, rng, n_rounds, budget_s, t0):
    n = len(h)
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    for r in range(n_rounds):
        if budget_s - (time.time() - t0) < 15.0:
            break
        scale = max(0.03, 0.12 / (r + 1))
        pert = rng.standard_normal(n)
        pert -= pert.mean()
        h_p = h + scale * pert
        h_p = project_feasible(h_p, target)
        h_opt, c5_opt = lp_optimize(h_p, dx, target, 0.12, 40, budget_s, t0)
        h_opt, c5_opt = pair_swap_refine(h_opt, dx, target, n, rng, 5, budget_s, t0)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
            h = h_opt.copy()
    return best_h, best_c5

def transfer_grid(h_src, n_target):
    n_src = len(h_src)
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_tgt = (np.arange(n_target) + 0.5) * (2.0 / n_target)
    return np.interp(x_tgt, x_src, h_src)

def multi_coord_move(h, dx, target, n, rng, n_moves, budget_s, t0):
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    for _ in range(n_moves):
        if budget_s - (time.time() - t0) < 5.0:
            break
        improved = False
        for attempt in range(200):
            k_coords = rng.randint(2, 6)
            idx = rng.choice(n, size=k_coords, replace=False)
            signs = rng.choice([-1, 1], size=k_coords)
            mags = rng.uniform(0.1, 0.6, size=k_coords)
            h_try = best_h.copy()
            for i in range(k_coords):
                h_try[idx[i]] += signs[i] * mags[i]
            h_try = project_feasible(h_try, target)
            c5_try = c5_of_h(h_try, dx)
            if c5_try < best_c5 - 1e-14:
                best_h = h_try
                best_c5 = c5_try
                improved = True
                break
        if not improved:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = np.full(160, 0.5)

    n = len(h0)
    dx = 2.0 / n
    target = n / 2.0
    h0 = project_feasible(h0, target)
    best_h = h0.copy()
    best_c5 = c5_of_h(h0, dx)

    # Phase 1: LP optimization with multiple alphas
    for alpha in [0.25, 0.15, 0.08, 0.04, 0.02]:
        if budget_s - (time.time() - t0) < 15:
            break
        h_opt, c5_opt = lp_optimize(best_h, dx, target, alpha, 120, budget_s, t0)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 2: Pair swap refinement
    if budget_s - (time.time() - t0) > 20:
        h_ps, c5_ps = pair_swap_refine(best_h, dx, target, n, rng, 30, budget_s, t0)
        if c5_ps < best_c5:
            best_c5 = c5_ps
            best_h = h_ps

    # Phase 3: Multi-coordinate moves
    if budget_s - (time.time() - t0) > 20:
        h_mc, c5_mc = multi_coord_move(best_h, dx, target, n, rng, 10, budget_s, t0)
        if c5_mc < best_c5:
            best_c5 = c5_mc
            best_h = h_mc
        if c5_mc < best_c5:
            h_opt, c5_opt = lp_optimize(best_h, dx, target, 0.05, 60, budget_s, t0)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt

    # Phase 4: Perturbation + re-optimization
    if budget_s - (time.time() - t0) > 30:
        h_p, c5_p = perturb_and_optimize(best_h, dx, target, rng, 10, budget_s, t0)
        if c5_p < best_c5:
            best_c5 = c5_p
            best_h = h_p

    # Phase 5: Second round of pair swaps
    if budget_s - (time.time() - t0) > 15:
        h_ps2, c5_ps2 = pair_swap_refine(best_h, dx, target, n, rng, 30, budget_s, t0)
        if c5_ps2 < best_c5:
            best_c5 = c5_ps2
            best_h = h_ps2

    # Phase 6: Try different grid sizes
    n_base = len(best_h)
    n_candidates = []
    for dn in range(-16, 17, 2):
        nc = n_base + dn
        if 64 <= nc <= 200 and nc % 2 == 0 and nc != n_base:
            n_candidates.append(nc)
    for extra in [80, 100, 128, 144, 176]:
        if extra not in n_candidates and 64 <= extra <= 200:
            n_candidates.append(extra)
    n_candidates = sorted(set(n_candidates))

    for n_try in n_candidates:
        if budget_s - (time.time() - t0) < 20:
            break
        dx_t = 2.0 / n_try
        ts_t = n_try / 2.0
        h_interp = transfer_grid(best_h, n_try)
        h_interp = project_feasible(h_interp, ts_t)
        h_opt, c5_opt = lp_optimize(h_interp, dx_t, ts_t, 0.1, 50, budget_s, t0)
        h_opt, c5_opt = pair_swap_refine(h_opt, dx_t, ts_t, n_try, rng, 8, budget_s, t0)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
            n_base = n_try

    # Phase 7: Final intensive polish
    n_b = len(best_h)
    dx_b = 2.0 / n_b
    ts_b = n_b / 2.0

    if budget_s - (time.time() - t0) > 20:
        h_f, c5_f = lp_optimize(best_h, dx_b, ts_b, 0.03, 200, budget_s, t0)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f

    if budget_s - (time.time() - t0) > 15:
        h_ps3, c5_ps3 = pair_swap_refine(best_h, dx_b, ts_b, n_b, rng, 40, budget_s, t0)
        if c5_ps3 < best_c5:
            best_c5 = c5_ps3
            best_h = h_ps3

    if budget_s - (time.time() - t0) > 15:
        h_mc2, c5_mc2 = multi_coord_move(best_h, dx_b, ts_b, n_b, rng, 10, budget_s, t0)
        if c5_mc2 < best_c5:
            best_c5 = c5_mc2
            best_h = h_mc2
        if c5_mc2 < best_c5:
            h_f2, c5_f2 = lp_optimize(best_h, dx_b, ts_b, 0.03, 100, budget_s, t0)
            if c5_f2 < best_c5:
                best_c5 = c5_f2
                best_h = h_f2

    # Final validation
    n_final = len(best_h)
    dx_final = 2.0 / n_final
    ts_final = n_final / 2.0
    best_h = project_feasible(best_h, ts_final)
    best_c5 = c5_of_h(best_h, dx_final)

    return best_h, best_c5, n_final
