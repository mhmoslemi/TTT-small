import numpy as np
from scipy.optimize import linprog, minimize
import time

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(-(1.0 - h))
        remaining = diff
        for i in range(n):
            idx = order[i]
            cap = 1.0 - h[idx]
            if cap <= 1e-15:
                continue
            add = min(remaining, cap)
            h[idx] += add
            remaining -= add
            if remaining <= 1e-14:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            if h[idx] <= 1e-15:
                continue
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-14:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] = float(np.clip(h[idx] + diff2, 0.0, 1.0))
    return h

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def grad_rows(h):
    """Compute gradient matrix G of shape (M, n) where G[m, i] = dP_m/dh_i"""
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    G = np.zeros((M, n))
    for m in range(M):
        d = m - (n - 1)
        if d == 0:
            G[m, :] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            G[m, d:] += dx * (1.0 - h[:-d])
            G[m, :-d] -= dx * h[d:]
        else:
            p = -d
            G[m, p:] += dx * (1.0 - h[:-p])
            G[m, :-p] -= dx * h[p:]
    return G

def trlp_step(h, delta, max_row=None):
    """Solve trust-region LP. Returns (direction, tau_predicted) or None."""
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    P = compute_profile(h)  # length M
    G = grad_rows(h)  # shape (M, n)
    
    # Variables: d (length n) + tau (length 1) = n+1
    # Objective: minimize tau
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    # Constraints: G[m] @ d - tau <= -P[m] for all m
    # i.e., G[m] @ d - tau + P[m] <= 0
    # A_ub = [G | -1], b_ub = -P
    A_ub = np.zeros((M, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -P
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds on d: max(-delta, -h_i) <= d_i <= min(delta, 1-h_i)
    lo_d = np.maximum(-delta, -h)
    hi_d = np.minimum(delta, 1.0 - h)
    bounds = list(zip(lo_d, hi_d)) + [(0.0, None)]  # tau >= 0 (loose)
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs-ds')
        if res.success:
            d = res.x[:n]
            tau = res.x[n]
            return d, tau
    except Exception:
        pass
    return None

    best_c = exact_score(h)
    delta = delta_init

    best_delta = delta
    
    for it in range(max_iters):

        if result is None:

            continue
        d, tau_pred = result
        if not np.all(np.isfinite(d)):
            delta *= 0.5
            continue
        dn = np.linalg.norm(d)
        if dn < 1e-12:
            break
        
        # Line search
        best_alpha = 0.0
        best_val = exact_score(h)
        best_h_try = h.copy()
        
        for frac in step_fracs:

            h_try = repair_feasible(h + alpha * d)
            c_try = exact_score(h_try)
            if c_try < best_val - 1e-14:

                best_alpha = alpha
                best_h_try = h_try.copy()
        
        if best_alpha > 0 and best_val < best_c - 1e-14:

            # Increase delta if step was accepted at full scale
            if best_alpha >= 0.7 * delta / max(dn, 1e-15):

            else:

        else:
            stall += 1
            delta *= 0.5
            if stall > 15 or delta < 1e-6:
                break
    
    return h, best_c

    n = len(h)

    thresh = c_max - 0.01  # include constraints within 0.01 of max

        top_idx = np.argsort(P[active])[-top_k:]

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(-(1.0 - h))
        remaining = diff
        for i in range(n):
            idx = order[i]
            cap = 1.0 - h[idx]
            if cap <= 1e-15:
                continue
            add = min(remaining, cap)
            h[idx] += add
            remaining -= add
            if remaining <= 1e-14:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            if h[idx] <= 1e-15:
                continue
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-14:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] = float(np.clip(h[idx] + diff2, 0.0, 1.0))
    return h

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def grad_row_single(h, m, n):
    dx = 2.0 / n
    d = m - (n - 1)
    g = np.zeros(n)
    if d == 0:
        g[:] = dx * (1.0 - 2.0 * h)
    elif d > 0:
        g[d:] += dx * (1.0 - h[:-d])
        g[:-d] -= dx * h[d:]
    else:
        p = -d
        g[p:] += dx * (1.0 - h[:-p])
        g[:-p] -= dx * h[p:]
    return g

def interp_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    edges_old = np.linspace(0.0, 2.0, n + 1)
    edges_new = np.linspace(0.0, 2.0, new_n + 1)
    h_new = np.zeros(new_n)
    for k in range(new_n):
        a, b = edges_new[k], edges_new[k + 1]
        total = 0.0
        for j in range(n):
            lo = max(a, edges_old[j])
            hi = min(b, edges_old[j + 1])
            if hi > lo:
                total += h[j] * (hi - lo)
        h_new[k] = total / (b - a)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

        best_h = h.copy()
        a_ls, b_ls = 0.0, alpha_max
        for _ in range(18):
            c1 = b_ls - gr * (b_ls - a_ls)
            c2 = a_ls + gr * (b_ls - a_ls)
            h1 = repair_feasible(h + c1 * d)
            h2 = repair_feasible(h + c2 * d)
            f1 = exact_score(h1)
            f2 = exact_score(h2)
            if f1 < best_val:
                best_val = f1
                best_alpha = c1
                best_h = h1.copy()
            if f2 < best_val:
                best_val = f2
                best_alpha = c2
                best_h = h2.copy()
            if f1 < f2:
                b_ls = c2
            else:
                a_ls = c1
            if b_ls - a_ls < 1e-13:
                break
        
        if best_val >= c - 1e-15:
            break
        h = best_h
        if best_val < best_c:
            best_c = best_val
    
    return h, best_c

def snap_and_reoptimize(h, rng, deadline):
    h_new = h.copy()
    interior = (h_new > 0.1) & (h_new < 0.9)
    if interior.any():
        h_new[interior] = 0.5
    h_new = repair_feasible(h_new)
    noise = rng.randn(len(h_new)) * 0.03
    h_new = repair_feasible(h_new + noise)
    h_out, c_out = trlp_descent_active(h_new, delta_init=0.04, max_iters=100,
                                        active_tol=0.006, deadline=deadline)
    return h_out, c_out

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()
    rng = np.random.RandomState(seed)
    
    def remaining():
        return budget_s - (time.perf_counter() - t_start)
    
    try:
        parent = np.array(initial_h_values, dtype=np.float64)
    except Exception:
        parent = None
    if parent is None or len(parent) < 10:
        parent = golden_ratio_seed(70)
    parent = repair_feasible(parent)
    
    best_h = parent.copy()
    best_c = exact_score(parent)
    n_parent = len(best_h)
    
    # Phase 1: TRLP full constraints on parent
    if remaining() > 80:
        dl = t_start + remaining() - 10
        h, c = trlp_descent(best_h, delta_init=0.08, max_iters=200, deadline=dl)
        if c < best_c:
            best_h, best_c = h, c
    
    # Phase 2: TRLP active set (faster, smaller LP)
    if remaining() > 60:
        dl = t_start + remaining() - 5
        h, c = trlp_descent_active(best_h, delta_init=0.05, max_iters=200,
                                    active_tol=0.005, deadline=dl)
        if c < best_c:
            best_h, best_c = h, c
    
    # Phase 3: Dimension sweep
    n_best = len(best_h)
    dim_offsets = [-8, -6, -4, -2, 2, 4, 6, 8, 10, 12, 14, 16]
    dims = []
    for dn in dim_offsets:
        d = n_best + dn
        if 40 <= d <= 140 and d % 2 == 0:
            dims.append(d)
    dims.sort(key=lambda x: abs(x - n_best))
    
    for new_n in dims[:10]:
        if remaining() < 45:
            break
        h_interp = interp_to_n(best_h, new_n)
        dl = t_start + remaining() - 5
        h, c = trlp_descent_active(h_interp, delta_init=0.05, max_iters=150,
                                    active_tol=0.005, deadline=dl)
        if c < best_c - 1e-12:
            best_h, best_c = h, c
            n_best = new_n
    
    # Phase 4: Gradient active descent polish
    if remaining() > 40:
        dl = t_start + remaining() - 5
        h, c = gradient_active_descent(best_h, max_iters=150, active_tol=0.003, deadline=dl)
        if c < best_c:
            best_h, best_c = h, c
    
    # Phase 5: Another TRLP round after gradient
    if remaining() > 35:
        dl = t_start + remaining() - 5
        h, c = trlp_descent_active(best_h, delta_init=0.03, max_iters=200,
                                    active_tol=0.003, deadline=dl)
        if c < best_c:
            best_h, best_c = h, c
    
    # Phase 6: Random perturbations + reoptimize
    if remaining() > 100:
        for k in range(12):
            if remaining() < 25:
                break
            scale = 0.04 + 0.03 * (k % 3)
            noise = rng.randn(len(best_h)) * scale
            h_try = repair_feasible(best_h + noise)
            dl = t_start + remaining() - 5
            h, c = trlp_descent_active(h_try, delta_init=0.04, max_iters=100,
                                        active_tol=0.006, deadline=dl)
            if c < best_c - 1e-12:
                best_h, best_c = h, c
    
    # Phase 7: Binary seed exploration at best dimension
    n_exp = len(best_h)
    if remaining() > 120:
        for pt in range(4):
            if remaining() < 30:
                break
            h_bin = binary_seed_varied(n_exp, rng, variant=pt)
            h_mix = 0.35 * h_bin + 0.65 * best_h
            h_mix = repair_feasible(h_mix)
            dl = t_start + remaining() - 5
            h, c = trlp_descent_active(h_mix, delta_init=0.05, max_iters=120,
                                        active_tol=0.006, deadline=dl)
            if c < best_c - 1e-12:
                best_h, best_c = h, c
    
    # Phase 8: Snap to binary-ish and reoptimize
    if remaining() > 50:
        dl = t_start + remaining() - 5
        h_snapped, c_snapped = snap_and_reoptimize(best_h, rng, dl)
        if c_snapped < best_c - 1e-12:
            best_h, best_c = h_snapped, c_snapped
    
    # Phase 9: Dimension re-sweep from new best
    n_cur = len(best_h)
    if remaining() > 60:
        for dn in [-4, -2, 2, 4]:
            if remaining() < 20:
                break
            new_n = n_cur + dn
            if new_n < 40 or new_n > 140 or new_n % 2 != 0:
                continue
            h_interp = interp_to_n(best_h, new_n)
            dl = t_start + remaining() - 3
            h, c = trlp_descent_active(h_interp, delta_init=0.04, max_iters=120,
                                        active_tol=0.004, deadline=dl)
            if c < best_c - 1e-12:
                best_h, best_c = h, c
                n_cur = new_n
    
    # Phase 10: Final fine polish with very small trust region
    if remaining() > 25:
        dl = t_start + remaining() - 3
        h, c = trlp_descent_active(best_h, delta_init=0.015, max_iters=400,
                                    active_tol=0.002, deadline=dl)
        if c < best_c:
            best_h, best_c = h, c
    
    # Phase 11: Tiny gradient polish
    if remaining() > 15:
        dl = t_start + remaining() - 2
        h, c = gradient_active_descent(best_h, max_iters=100, active_tol=0.001, deadline=dl)
        if c < best_c:
            best_h, best_c = h, c
    
    # Phase 12: Final TRLP
    if remaining() > 10:
        dl = t_start + remaining() - 2
        h, c = trlp_descent(best_h, delta_init=0.01, max_iters=300, deadline=dl)
        if c < best_c:
            best_h, best_c = h, c
    
    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)
    
    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            if float(eval_result) < final_c + 1e-9:
                final_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 2:
            val = float(eval_result[0])
            if val < final_c + 1e-9:
                final_c = val
    except Exception:
        pass
    
    return best_h, final_c, len(best_h)
