    return float(np.max(prof))


def compute_profile(h):

    dx = 2.0 / n

def repair_feasible(h):
    n = len(h)
    h = np.clip(np.asarray(h, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, 1.0 - h[idx])
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)[::-1]
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-14:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h


def build_gradient(h, n, dx):
    M = 2 * n - 1
    G = np.zeros((M, n))
    center = n - 1
    for d in range(-n + 1, n):

        if d >= 0:
            i = np.arange(0, n - d)
            j = i + d
            G[row, i] += dx * (1.0 - h[j])
            G[row, j] += -dx * h[i]
        else:
            e = -d
            i = np.arange(e, n)
            j = i - e
            G[row, i] += dx * (1.0 - h[j])
            G[row, j] += -dx * h[i]
    return G

    n = len(h)

    # Phase 7: Try symmetry as final trick
    h_sym = repair_feasible(symmetrize(best_h))
    c_sym = exact_score(h_sym)
    if c_sym < best_c - 1e-13:
        best_h = h_sym
        best_c = c_sym

    # Phase 8: Very final micro-polish
    if time_left() > 10:
        h_pol, c_pol = fine_polish(best_h, t_start, budget_s, max_iters=50)
        if c_pol < best_c - 1e-14:
            best_h = h_pol
            best_c = c_pol

    # Final validation
    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)

    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            final_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 2:
            final_c = float(eval_result[0])
    except Exception:
        pass

    return best_h, final_c, len(best_h)
