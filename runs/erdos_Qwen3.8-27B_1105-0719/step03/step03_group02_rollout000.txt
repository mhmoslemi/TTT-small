import numpy as np
from scipy.optimize import minimize, linprog
from scipy.special import logsumexp
import time


def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        interior = (h > 1e-12) & (h < 1.0 - 1e-12)
        if interior.sum() > 0:
            h[interior] += diff / interior.sum()
    return np.clip(h, 0.0, 1.0)


def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def c5_of_h(h, dx):
    return compute_corrs(h).max() * dx


def grad_corr_k(h, k, n):
    offset = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -offset)
    i_hi = min(n - 1, n - 1 - offset)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx + offset] += (1.0 - h[idx])
    grad[idx] += (-h[idx + offset])
    return grad


def smooth_max_val(h, dx, beta):
    corrs = compute_corrs(h)
    return logsumexp(beta * corrs) / beta * dx


def smooth_max_grad(h, dx, beta):
    n = len(h)
    corrs = compute_corrs(h)
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones_like(corrs) / len(corrs)
    else:
        w /= w_sum
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] > 1e-13:
            grad += w[k] * grad_corr_k(h, k, n)
    return grad * dx


def binary_eval(b, dx):
    corrs = np.correlate(b.astype(float), 1.0 - b.astype(float), mode='full')
    return corrs.max() * dx, corrs


def binary_local_swap(b, dx, n_swaps=2000, rng=None):
    n = len(b)
    b = b.copy()
    cur_val, cur_corrs = binary_eval(b, dx)
    dx1, dx2 = 2.0 / n
    if rng is None:
        rng = np.random.RandomState(0)
    for _ in range(n_swaps):
        ones = np.where(b == 1)[0]
        zeros = np.where(b == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]

        new_val, _ = binary_eval(b, dx)
        if new_val < cur_val - 1e-12:
            cur_val = new_val
            cur_corrs = compute_corrs(b.astype(float))
        else:

    return b, cur_val


def make_random_binary(n, rng):
    idx = rng.choice(n, size=n // 2, replace=False)
    b = np.zeros(n, dtype=int)
    b[idx] = 1
    return b


def make_periodic_binary(n, period, fill, rng=None):
    b = np.zeros(n, dtype=int)
    for i in range(n):
        if (i % period) < fill:
            b[i] = 1
    if b.sum() != n // 2:
        if b.sum() > n // 2:
            ones = np.where(b == 1)[0]
            remove = rng.choice(ones, size=b.sum() - n // 2, replace=False)
            b[remove] = 0
        else:
            zeros = np.where(b == 0)[0]
            add = rng.choice(zeros, size=n // 2 - b.sum(), replace=False)
            b[add] = 1
    return b

def slsqp_optimize(h0, dx, target_sum, beta, maxiter=2000):
    n = len(h0)

    def obj(x):
        return smooth_max_val(x, dx, beta)

    def jac(x):
        return smooth_max_grad(x, dx, beta)

    def c_fun(x):
        return x.sum() - target_sum

    def c_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': c_fun, 'jac': c_jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0, jac=jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        h_out = project_feasible(res.x, target_sum)
    except Exception:
        h_out = project_feasible(h0, target_sum)
    return h_out, c5_of_h(h_out, dx)


def pair_transfer(h0, dx, target_sum, n_top=20, n_iter=5):

    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)

    for iteration in range(n_iter):
        corrs = compute_corrs(h)
        grad_full = np.zeros(n)
        shifted = corrs - corrs.max()
        w = np.exp(500.0 * shifted)
        w /= w.sum()
        for k in range(2 * n - 1):
            if w[k] > 1e-12:
                grad_full += w[k] * grad_corr_k(h, k, n)
        grad_full *= dx

        dec_idx = np.argsort(grad_full)[-n_top:]
        inc_idx = np.argsort(grad_full)[:n_top]

        improved = False
        for i in dec_idx:
            for j in inc_idx:
                if i == j:
                    continue
                eps_hi = min(h[i], 1.0 - h[j])
                if eps_hi < 1e-12:
                    continue
                d = np.zeros(2 * n - 1)
                for k in range(2 * n - 1):
                    gk = grad_corr_k(h, k, n)
                    d[k] = gk[j] - gk[i]
                f_arr = corrs + eps_hi * d
                if f_arr.max() * dx > best_c5:
                    lo_id, hi_id = 0.0, eps_hi
                    gr = (np.sqrt(5.0) - 1.0) / 2.0

                        if hi_id - lo_id < 1e-13:
                            break
                        fc = (corrs + c_pt * d).max()

                        if fc < fd:
                            hi_id, d_pt = d_pt, c_pt
                            c_pt = hi_id - gr * (hi_id - lo_id)
                        else:
                            lo_id, c_pt = c_pt, d_pt
                            d_pt = lo_id + gr * (hi_id - lo_id)
                    eps_opt = (lo_id + hi_id) / 2.0
                    if (corrs + eps_opt * d).max() * dx < best_c5 - 1e-14:

                        h[j] += eps_opt
                        h = np.clip(h, 0.0, 1.0)

                            best_h = h.copy()
                            corrs = compute_corrs(h)

def momentum_subgrad(h0, dx, target_sum, max_iter=8000, beta_smooth=50.0):

        h_sg, c5_sg = momentum_subgrad(best_h, dx, target_sum, max_iter=6000)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg

    if time_left() > 30:
        h_f, c5_f = slsqp_optimize(best_h, dx, target_sum, 3000, 1500)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f

    if time_left() > 20:
        h_lp3, c5_lp3 = aggressive_lp_polish(best_h, dx, target_sum,
                                             max_iters=300, init_step=0.08,
                                             active_frac=0.7)
        if c5_lp3 < best_c5:
            best_c5 = c5_lp3
            best_h = h_lp3

    if time_left() > 10:
        h_pt2, c5_pt2 = pair_transfer(best_h, dx, target_sum, n_top=15, n_iter=3)
        if c5_pt2 < best_c5:
            best_c5 = c5_pt2
            best_h = h_pt2

    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s - (time.time() - t0)

    best_h = None
    best_c5 = float('inf')

    # Phase 1: Start from initial_h_values
    h0 = np.array(initial_h_values, dtype=float)
    n = len(h0)
    dx = 2.0 / n
    target_sum = n / 2.0
    h0 = project_feasible(h0, target_sum)

    h_opt, c5_val = full_pipeline(h0, dx, target_sum, time_left, "main")
    best_h = h_opt
    best_c5 = c5_val

    # Phase 2: Binary exploration
    if time_left() > 60:
        n_bin = 60
        for trial in range(n_bin):
            if time_left() < 30:
                break
            if trial < 10:
                b = make_random_binary(n, rng)
            elif trial < 30:
                period = rng.randint(3, 15)
                fill = rng.randint(1, period)
                b = make_periodic_binary(n, period, fill, rng)
            else:
                b = make_random_binary(n, rng)
                b, _ = binary_local_swap(b, dx, n_swaps=3000, rng=rng)

            b_float = b.astype(float)
            b_val = binary_eval(b_float, dx)

            if b_val < best_c5 - 1e-6:
                h_bin = project_feasible(b_float, target_sum)
                h_opt2, c5_opt2 = full_pipeline(h_bin, dx, target_sum, time_left, "bin")
                if c5_opt2 < best_c5:
                    best_c5 = c5_opt2
                    best_h = h_opt2

    # Phase 3: Perturbation from best
    for trial in range(20):
        if time_left() < 30:
            break
        scale = 0.03 + 0.08 * (trial % 5) / 5.0
        if trial % 4 == 0:
            perturb = scale * (rng.random(n) - 0.5)
        elif trial % 4 == 1:
            idx = rng.choice(n, size=max(5, n // 8), replace=False)
            perturb = np.zeros(n)
            perturb[idx] = scale * rng.choice([-1, 1], size=len(idx))
        else:
            perturb = scale * rng.standard_normal(n) * 0.3

        h_try = project_feasible(best_h + perturb, target_sum)
        h_opt3, c5_opt3 = full_pipeline(h_try, dx, target_sum, time_left, "pert")
        if c5_opt3 < best_c5:
            best_c5 = c5_opt3
            best_h = h_opt3

    # Phase 4: Try different n values
    n_candidates = list(range(max(40, n - 20), min(250, n + 40) + 1, 2))
    n_candidates = [nn for nn in n_candidates if nn != n and nn >= 40]
    n_candidates.sort(key=lambda x: abs(x - n))
    n_candidates = n_candidates[:15]

    for n_try in n_candidates:
        if time_left() < 35:
            break
        dx_try = 2.0 / n_try
        ts_try = n_try / 2.0
        x_orig = (np.arange(n) + 0.5) * (2.0 / n)
        x_new = (np.arange(n_try) + 0.5) * (2.0 / n_try)
        h_interp = np.interp(x_new, x_orig, best_h)
        h_interp = project_feasible(h_interp, ts_try)

        h_opt4, c5_opt4 = full_pipeline(h_interp, dx_try, ts_try, time_left, "n%d" % n_try)
        if c5_opt4 < best_c5:
            best_c5 = c5_opt4
            best_h = h_opt4

    # Phase 5: Final intensive polish on the very best
    if time_left() > 20 and best_h is not None:
        n_b = len(best_h)
        dx_b = 2.0 / n_b
        ts_b = n_b / 2.0
        h_fin, c5_fin = full_pipeline(best_h, dx_b, ts_b, time_left, "final")
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin

    # Phase 6: Ultra-fine pair transfer + LP
    if time_left() > 10 and best_h is not None:
        n_b = len(best_h)
        dx_b = 2.0 / n_b
        ts_b = n_b / 2.0
        h_pt, c5_pt = pair_transfer(best_h, dx_b, ts_b, n_top=30, n_iter=6)
        if c5_pt < best_c5:
            best_c5 = c5_pt
            best_h = h_pt
        if time_left() > 8:
            h_lp, c5_lp = aggressive_lp_polish(best_h, dx_b, ts_b,
                                               max_iters=200, init_step=0.05,
                                               active_frac=0.8)
            if c5_lp < best_c5:
                best_c5 = c5_lp
                best_h = h_lp

    # Final validation
    if best_h is None:
        best_h = h0
        best_c5 = c5_of_h(h0, dx)
    else:
        n_final = len(best_h)
        best_h = project_feasible(best_h, n_final / 2.0)
        best_c5 = c5_of_h(best_h, 2.0 / n_final)

    return best_h, best_c5, len(best_h)
