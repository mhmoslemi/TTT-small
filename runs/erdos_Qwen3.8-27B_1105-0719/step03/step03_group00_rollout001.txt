def project_feasible(z, target):
    n = len(z)
    lo = -2.0
    hi = 2.0
    for _ in range(50):
        mid = (lo + hi) / 2
        h = np.clip(z - mid, 0.0, 1.0)
        if h.sum() > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def c5_of_h(h, dx):
    return np.correlate(h, 1.0 - h, mode='full').max() * dx

def corr_of_h(h):
    return np.correlate(h, 1.0 - h, mode='full')

def grad_lag(h, k, n):
    """Gradient of corr_k with respect to h."""
    off = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -off)
    i_hi = min(n - 1, n - 1 - off)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx + off] += (1.0 - h[idx])
    grad[idx] += (-h[idx + off])
    return grad

def solve_soft_slack_lp(h, dx, corr, top_idx, alpha, rho_q=0.35, rho_s=5.0):
    """Solve soft slack LP to find descent direction."""
    n = len(h)
    K = len(top_idx)
    m_vars = n + K + 1  # delta, slacks, q
    
    c_obj = np.zeros(m_vars)
    c_obj[-1] = rho_q  # cost on q
    c_obj[n:n+K] = rho_s  # cost on slacks
    
    A_ub = np.zeros((K, m_vars))
    b_ub = np.zeros(K)
    
    for j, k in enumerate(top_idx):
        g = grad_lag(h, k, n)
        A_ub[j, :n] = dx * g
        A_ub[j, n + j] = 1.0  # slack variable
        # Constraint: dx*grad_k·δ + s_j - q <= -corr[k]
        # i.e., A_ub[j,:n]·δ + A_ub[j,n+j]·s_j + (-1)·q <= -corr[k]
        A_ub[j, -1] = -1.0
        b_ub[j] = -corr[k]
    
    A_eq = np.zeros((1, m_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    bounds = [(-h[i] * alpha, (1.0 - h[i]) * alpha) for i in range(n)]
    bounds += [(0.0, None)] * K  # slacks >= 0
    bounds.append((-0.01, None))  # q free (can be slightly negative)
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.status != 0:
            return None
        return res.x[:n]
    except Exception:
        return None

def solve_weighted_sum_lp(h, dx, corr, top_idx, alpha, weights=None):
    """Solve weighted sum LP to find descent direction."""
    n = len(h)
    K = len(top_idx)
    
    if weights is None:
        weights = np.ones(K)
    
    # Objective: min dx * sum_j w_j * grad_j · δ

    A_eq = np.zeros((1, n))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    bounds = [(-h[i] * alpha, (1.0 - h[i]) * alpha) for i in range(n)]

            hi = mid

        if mask.sum() > 0:

    grad = np.zeros(n)
    i_lo = max(0, -off)
    i_hi = min(n - 1, n - 1 - off)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx + off] += (1.0 - h[idx])
    grad[idx] += (-h[idx + off])
    return grad


def solve_soft_slack_lp(h, dx, corr, top_idx, alpha, rho_q, rho_s):
    n = len(h)
    K = len(top_idx)
    m_vars = n + K + 1

    c_obj = np.zeros(m_vars)
    c_obj[-1] = rho_q
    c_obj[n:n + K] = rho_s

    A_ub = np.zeros((K, m_vars))
    b_ub = np.zeros(K)

    for j, k in enumerate(top_idx):
        g = grad_lag(h, k, n)
        A_ub[j, :n] = dx * g
        A_ub[j, n + j] = 1.0
        A_ub[j, -1] = -1.0
        b_ub[j] = -corr[k]

    A_eq = np.zeros((1, m_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)

    bounds = [(-h[i] * alpha, (1.0 - h[i]) * alpha) for i in range(n)]
    bounds += [(0.0, None)] * K
    bounds.append((-0.01, None))

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.status != 0:
            return None
        return res.x[:n]
    except Exception:
        return None


def solve_weighted_sum_lp(h, dx, corr, top_idx, alpha, weights):
    n = len(h)
    K = len(top_idx)
    g_sum = np.zeros(n)
    for j, k in enumerate(top_idx):
        g_sum += weights[j] * grad_lag(h, k, n)
    c_obj = dx * g_sum

    A_eq = np.zeros((1, n))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)

    bounds = [(-h[i] * alpha, (1.0 - h[i]) * alpha) for i in range(n)]

    try:
        res = linprog(c_obj, A_ub=None, b_ub=None, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.status != 0:
            return None
        return res.x[:n]
    except Exception:
        return None


def solve_hard_minmax_lp(h, dx, corr, top_idx, alpha):
    n = len(h)
    K = len(top_idx)
    m_vars = n + 1
    c_obj = np.zeros(m_vars)
    c_obj[-1] = 1.0
    A_ub = np.zeros((K, m_vars))
    b_ub = np.zeros(K)
    for j, k in enumerate(top_idx):
        A_ub[j, :n] = dx * grad_lag(h, k, n)
        A_ub[j, -1] = -1.0
        b_ub[j] = -corr[k]
    A_eq = np.zeros((1, m_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(-h[i] * alpha, (1.0 - h[i]) * alpha) for i in range(n)]
    bounds.append((0.0, None))
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.status != 0:
            return None
        return res.x[:n]
    except Exception:
        return None


def line_search(h_cur, c5_cur, direction, dx, target, alpha):
    best_t = 0.0
    best_c5 = c5_cur
    best_h = h_cur.copy()
    max_up = np.min(np.where(direction > 1e-12, alpha * (1.0 - h_cur) / direction, np.inf))
    max_dn = np.min(np.where(direction < -1e-12, -alpha * h_cur / direction, np.inf))
    max_t = min(max_up, max_dn, 1.5)
    if not np.isfinite(max_t) or max_t < 1e-12:
        return best_h, best_c5, best_t
    for frac in [1.0, 0.8, 0.6, 0.5, 0.4, 0.3, 0.2, 0.15, 0.1, 0.07, 0.05, 0.03, 0.02, 0.01]:
        t = max_t * frac
        h_new = project_feasible(h_cur + t * direction, target)
        c5_new = c5_of_h(h_new, dx)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new
            best_t = t
    return best_h, best_c5, best_t


def pair_move(h, idx1, idx2, dx):
    a = h[idx1]
    b = h[idx2]
    L = max(-a, b - 1.0)
    U = min(1.0 - a, b)
    if U - L < 1e-10:
        return h.copy(), c5_of_h(h, dx)
    base_c5 = c5_of_h(h, dx)
    best_t = 0.0
    best_c5 = base_c5
    best_h = h.copy()
    ts = np.linspace(L, U, 17)
    for t in ts:
        h_new = h.copy()
        h_new[idx1] = a + t
        h_new[idx2] = b - t
        c5_val = c5_of_h(h_new, dx)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_t = t
            best_h = h_new
    if best_t != 0.0 and (U - L) > 1e-8:
        step = (U - L) / 17.0
        lo = max(L, best_t - 2.5 * step)
        hi = min(U, best_t + 2.5 * step)
        for t in np.linspace(lo, hi, 13):
            h_new = h.copy()
            h_new[idx1] = a + t
            h_new[idx2] = b - t
            c5_val = c5_of_h(h_new, dx)
            if c5_val < best_c5:
                best_c5 = c5_val
                best_t = t
                best_h = h_new
    return best_h, best_c5


def lp_refine(h, dx, target, alpha, n_iter, budget_s, t0, mode='soft'):
    n = len(h)
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    cur = h.copy()
    cur_c5 = best_c5
    stall = 0
    for it in range(n_iter):
        if budget_s - (time.time() - t0) < 3.0:
            break
        corr = corr_of_h(cur)
        max_c = corr.max()
        margin = max(0.12, 0.04 * max_c)
        n_top = max(20, min(55, int(0.25 * (2 * n - 1))))
        threshold = max_c - margin
        active_mask = corr >= threshold
        top_idx = np.argsort(corr)[-n_top:]
        active_mask[top_idx] = True
        active_idx = np.where(active_mask)[0]
        if len(active_idx) < 4:
            active_idx = top_idx

        if mode == 'soft':
            delta = solve_soft_slack_lp(cur, dx, corr, active_idx, alpha, 0.35, 5.0)
        elif mode == 'weighted':
            weights = np.exp(-15.0 * (max_c - corr[active_idx]))
            weights = weights / (weights.sum() + 1e-15)
            delta = solve_weighted_sum_lp(cur, dx, corr, active_idx, alpha, weights)
        else:
            delta = solve_hard_minmax_lp(cur, dx, corr, active_idx, alpha)

        if delta is None:
            alpha *= 0.5
            if alpha < 0.003:
                break
            continue

        h_new, c5_new, _ = line_search(cur, cur_c5, delta, dx, target, alpha)
        if c5_new < cur_c5 - 1e-14:
            cur = h_new
            cur_c5 = c5_new
            stall = 0
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            alpha = min(0.4, alpha * 1.04)
        else:
            stall += 1
            alpha *= 0.65
            if stall > 10 or alpha < 0.003:
                break
    return best_h, best_c5


def pair_polish(h, dx, budget_s, t0, rng, n_rounds=5):
    n = len(h)
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    corr = corr_of_h(best_h)
    top_lags = np.argsort(corr)[-35:]
    important = set()
    for k in top_lags:
        off = k - (n - 1)
        i_lo = max(0, -off)
        i_hi = min(n - 1, n - 1 - off)
        for i in range(i_lo, i_hi + 1):
            important.add(i)
            if off != 0:
                important.add(i + off)
    imp_list = list(important)
    for rnd in range(n_rounds):
        if budget_s - (time.time() - t0) < 8.0:
            break
        improved = False
        order = rng.permutation(min(len(imp_list), 120))
        for pi in order:
            if budget_s - (time.time() - t0) < 4.0:
                break
            i = imp_list[pi]
            j_candidates = [x for x in imp_list if x != i]
            if not j_candidates:
                j_candidates = list(range(n))
            j = j_candidates[rng.randint(len(j_candidates))]
            h_new, c5_new = pair_move(best_h, i, j, dx)
            if c5_new < best_c5 - 1e-13:
                best_h = h_new
                best_c5 = c5_new
                improved = True
        if not improved:
            for _ in range(40):
                if budget_s - (time.time() - t0) < 4.0:
                    break
                i = rng.randint(n)
                j = rng.randint(n)
                if i == j:
                    continue
                h_new, c5_new = pair_move(best_h, i, j, dx)
                if c5_new < best_c5 - 1e-13:
                    best_h = h_new
                    best_c5 = c5_new
    return best_h, best_c5


def transfer_grid(h_src, n_target):
    n_src = len(h_src)
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_tgt = (np.arange(n_target) + 0.5) * (2.0 / n_target)
    return np.interp(x_tgt, x_src, h_src)


def perturb_reopt(h, dx, target, rng, n_rounds, budget_s, t0):
    n = len(h)
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    for r in range(n_rounds):
        if budget_s - (time.time() - t0) < 15.0:
            break
        scale = max(0.04, 0.2 / (r + 1))
        pert = rng.standard_normal(n)
        pert -= pert.mean()
        h_p = project_feasible(h + scale * pert, target)
        h_opt, c5_opt = lp_refine(h_p, dx, target, 0.2, 70, budget_s, t0, mode='soft')
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
            h = h_opt.copy()
    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def tl():
        return budget_s - (time.time() - t0)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = np.full(160, 0.5)

    n = len(h0)
    dx = 2.0 / n
    target_sum = n / 2.0
    h0 = project_feasible(h0, target_sum)
    best_h = h0.copy()
    best_c5 = c5_of_h(h0, dx)

    # Phase 1: Soft slack LP - primary improvement method
    for alpha_start in [0.3, 0.2, 0.15, 0.1, 0.07, 0.05]:
        if tl() < 15:
            break
        h_lp, c5_lp = lp_refine(best_h, dx, target_sum, alpha_start, 200, budget_s, t0, mode='soft')
        if c5_lp < best_c5:
            best_c5 = c5_lp
            best_h = h_lp

    # Phase 2: Weighted sum variant
    if tl() > 30:
        h_w, c5_w = lp_refine(best_h, dx, target_sum, 0.2, 150, budget_s, t0, mode='weighted')
        if c5_w < best_c5:
            best_c5 = c5_w
            best_h = h_w

    # Phase 3: Hard minmax for comparison
    if tl() > 30:
        h_h, c5_h = lp_refine(best_h, dx, target_sum, 0.15, 100, budget_s, t0, mode='hard')
        if c5_h < best_c5:
            best_c5 = c5_h
            best_h = h_h

    # Phase 4: Pair coordinate polish
    if tl() > 30:
        h_p, c5_p = pair_polish(best_h, dx, budget_s, t0, rng, n_rounds=5)
        if c5_p < best_c5:
            best_c5 = c5_p
            best_h = h_p

    # Phase 5: More soft slack with smaller alpha
    if tl() > 25:
        h_s, c5_s = lp_refine(best_h, dx, target_sum, 0.04, 200, budget_s, t0, mode='soft')
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s

    # Phase 6: Perturbation + re-optimization
    if tl() > 40:
        h_pr, c5_pr = perturb_reopt(best_h, dx, target_sum, rng, 8, budget_s, t0)
        if c5_pr < best_c5:
            best_c5 = c5_pr
            best_h = h_pr

    # Phase 7: Try nearby grid sizes
    n_base = len(best_h)
    n_candidates = sorted(set(
        [n_base + dn for dn in [-4, -2, 2, 4, 8]
         if 64 <= n_base + dn <= 200 and (n_base + dn) % 2 == 0 and n_base + dn != n_base]
    ))
    for n_try in n_candidates:
        if tl() < 20:
            break
        dx_t = 2.0 / n_try
        ts_t = n_try / 2.0
        h_interp = project_feasible(transfer_grid(best_h, n_try), ts_t)
        h_opt, c5_opt = lp_refine(h_interp, dx_t, ts_t, 0.15, 100, budget_s, t0, mode='soft')
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
            h_opt2, c5_opt2 = pair_polish(h_opt, dx_t, budget_s, t0, rng, n_rounds=3)
            if c5_opt2 < best_c5:
                best_c5 = c5_opt2
                best_h = h_opt2

    # Phase 8: Final intensive soft slack
    n_b = len(best_h)
    dx_b = 2.0 / n_b
    ts_b = n_b / 2.0
    if tl() > 20:
        h_f, c5_f = lp_refine(best_h, dx_b, ts_b, 0.08, 200, budget_s, t0, mode='soft')
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f

    # Phase 9: Final pair polish
    if tl() > 20:
        h_pf, c5_pf = pair_polish(best_h, dx_b, budget_s, t0, rng, n_rounds=6)
        if c5_pf < best_c5:
            best_c5 = c5_pf
            best_h = h_pf

    # Phase 10: Very fine final LP
    if tl() > 12:
        h_fin, c5_fin = lp_refine(best_h, dx_b, ts_b, 0.02, 150, budget_s, t0, mode='soft')
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin

    # Phase 11: One more pair pass
    if tl() > 10:
        h_fp, c5_fp = pair_polish(best_h, dx_b, budget_s, t0, rng, n_rounds=4)
        if c5_fp < best_c5:
            best_c5 = c5_fp
            best_h = h_fp

    # Final validation
    n_final = len(best_h)
    dx_final = 2.0 / n_final
    ts_final = n_final / 2.0
    best_h = project_feasible(best_h, ts_final)
    best_c5 = c5_of_h(best_h, dx_final)
    return best_h, best_c5, n_final
