import numpy as np
from scipy.optimize import linprog
import time


def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return (corr * dx).max()


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, target):
    n = len(h)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    mu = (lo + hi) / 2.0
    h_new = np.clip(h - mu, 0.0, 1.0)
    diff = target - h_new.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            mask = h_new < 1.0 - 1e-12
        else:
            mask = h_new > 1e-12
        cnt = mask.sum()
        if cnt > 0:
            h_new[mask] += diff / cnt
    return np.clip(h_new, 0.0, 1.0)


def coord_optimal_t(c, h, i, n, top_m=40):
    center = n - 1
    lags = np.arange(-(n - 1), n).astype(float)
    vp = ((i + lags >= 0) & (i + lags < n)).astype(float)
    vm = ((i - lags >= 0) & (i - lags < n)).astype(float)
    vm[center] = 0.0
    idx_p = np.clip(i + lags.astype(int), 0, n - 1)
    idx_m = np.clip(i - lags.astype(int), 0, n - 1)
    h_plus = h[idx_p] * vp
    h_minus = h[idx_m] * vm
    old = h[i] * (1.0 - h_plus) * vp + h_minus * (1.0 - h[i]) * vm
    C = c.copy() - old + h_minus * vm
    D = (1.0 - h_plus) * vp - h_minus * vm
    C0 = c[center] - h[i] * (1.0 - h[i])
    cands = [0.0, 1.0, 0.5, h[i]]
    active = np.argsort(c)[-top_m:]
    for a in active:
        if a == center:
            continue
        Bq = D[a] - 1.0
        Cq = C[a] - C0
        disc = Bq * Bq - 4.0 * Cq
        if disc >= 0.0:
            sd = np.sqrt(disc)
            t1 = (-Bq + sd) / 2.0
            t2 = (-Bq - sd) / 2.0
            if 0.0 <= t1 <= 1.0:
                cands.append(t1)
            if 0.0 <= t2 <= 1.0:
                cands.append(t2)
    am = active[:12]
    for ii in range(len(am)):
        for jj in range(ii + 1, len(am)):
            a, b = am[ii], am[jj]
            if a == center or b == center:
                continue
            dd = D[a] - D[b]
            if abs(dd) > 1e-12:
                tl = (C[b] - C[a]) / dd
                if 0.0 <= tl <= 1.0:
                    cands.append(tl)
    cands.extend(list(np.linspace(0.0, 1.0, 16)))
    cands = np.array(cands)
    cands = np.clip(cands, 0.0, 1.0)
    cands = np.unique(cands)
    vals_lin = C + np.outer(cands, D)
    vals_lin[:, center] = C0 + cands - cands * cands
    g_vals = vals_lin.max(axis=1)
    best_idx = g_vals.argmin()
    return cands[best_idx], g_vals[best_idx]


def coord_descent_sweep(h, c, n, order, top_m=40, max_sweeps=80, t0=None, deadline=None):
    for sweep in range(max_sweeps):
        if t0 is not None and deadline is not None:
            if time.time() > deadline:
                break
        improved = False
        for i in order:
            t_new, _ = coord_optimal_t(c, h, i, n, top_m)
            if abs(t_new - h[i]) > 1e-12:
                delta = t_new - h[i]
                idx_p_all = np.clip(i + np.arange(-(n-1), n), 0, n-1)
                idx_m_all = np.clip(i - np.arange(-(n-1), n), 0, n-1)
                vp = ((i + np.arange(!(n-1)) & 0,))  # placeholder, use precomputed
                # Update c directly
                lags_int = np.arange(-(n - 1), n)
                vp = ((i + lags_int >= 0) & (i + lags_int < n)).astype(float)
                vm = ((i - lags_int >= 0) & (i - lags_int < n)).astype(float)
                vm[n - 1] = 0.0
                ip = np.clip(i + lags_int, 0, n - 1)
                im = np.clip(i - lags_int, 0, n - 1)
                hp = h[ip] * vp
                hm = h[im] * vm
                old_contrib = h[i] * (1.0 - hp) * vp + hm * (1.0 - h[i]) * vm
                new_contrib = t_new * (1.0 - hp) * vp + hm * (1.0 - t_new) * vm
                c = c - old_contrib + new_contrib
                c[n - 1] = c[n - 1] - h[i] * (1.0 - h[i]) + t_new * (1.0 - t_new)
                h[i] = t_new
                improved = True
        if not improved:
            break
    return h, c


def lp_polish(h, dx, target_sum, t0, deadline, alpha_start=0.15, n_active_frac=0.4,
              alpha_decay=0.988, min_alpha=0.002, max_no_improve=100):
    n = len(h)
    n_total = 2 * n - 1
    n_active = max(12, int(n_active_frac * n_total))
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    alpha = alpha_start
    no_improve = 0
    while time.time() < deadline and alpha >= min_alpha:
        corr = compute_corr(h)
        active_idx = np.argsort(corr)[-n_active:]
        m = len(active_idx)
        G = np.zeros((m, n))
        for row, k_idx in enumerate(active_idx):
            k_offset = k_idx - (n - 1)
            i_lo = max(0, -k_offset)
            i_hi = min(n - 1, n - 1 - k_offset)
            if i_lo > i_hi:
                continue
            idx = np.arange(i_lo, i_hi + 1)
            j1 = idx + k_offset
            G[row, j1] += (1.0 - h[idx])
            G[row, idx] += (-h[idx + k_offset])
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        A_ub = np.empty((m, n + 1))
        A_ub[:, :n] = G
        A_ub[:, n] = -1.0
        b_ub = -corr[active_idx]
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)
        bounds = [(-h[i] * alpha, (1.0 - h[i]) * alpha) for i in range(n)]
        bounds.append((-np.inf, np.inf))
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if res.status != 0:
                no_improve += 1
                alpha *= alpha_decay
                continue
            delta = res.x[:n]
        except Exception:
            no_improve += 1
            alpha *= alpha_decay
            continue
        if np.abs(delta).max() < 1e-15:
            no_improve += 1
            alpha *= alpha_decay
            continue
        best_s = 0.0
        best_c5_local = best_c5
        best_h_local = h.copy()
        for s in [1.0, 0.7, 0.5, 0.35, 0.2, 0.1, 0.05]:
            h_new = project_feasible(h + s * delta, target_sum)
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5_local - 1e-14:
                best_c5_local = c5_new
                best_s = s
                best_h_local = h_new
                break
        if best_s > 0 and best_c5_local < best_c5 - 1e-14:
            h = best_h_local
            best_c5 = best_c5_local
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            alpha *= alpha_decay
            if no_improve == 25:
                n_active = min(n_total, n_active + 20)
                alpha = max(alpha, 0.12)
            elif no_improve == 50:
                n_active = max(15, n_active - 15)
                alpha = max(alpha, 0.18)
            if no_improve > max_no_improve:
                break
    return best_h, best_c5

                                  alpha_decay=0.992, min_alpha=0.001, max_no_improve=150)

            best_c5 = c5_lp2
            best_h = h_lp2

    n_restarts = 6

                                  alpha_decay=0.995, min_alpha=0.001, max_no_improve=200)

    # Final validation

    best_h = project_feasible(best_h, target_sum)
    n_final = len(best_h)
    dx_final = 2.0 / n_final
    c_final = compute_corr(best_h)
    best_c5 = (c_final * dx_final).max()

    return best_h, best_c5, n_final
