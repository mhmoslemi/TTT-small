from scipy.special import logsumexp
import time

def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid

    if abs(diff) > 1e-14:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def c5_of_h(h, dx):
    return compute_corrs(h).max() * dx

def grad_corr_k(h, k, n):
    """Gradient of corr_k with respect to h."""
    offset = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -offset)
    i_hi = min(n - 1, n - 1 - offset)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx + offset] += (1.0 - h[idx])
    grad[idx] += (-h[idx + offset])
    return grad

def make_feasible_direction(h, d, target_sum):
    """Project d to be a feasible direction: sum(d)=0 and h+alpha*d in [0,1] for some alpha>0."""
    d = d.copy()
    # Ensure sum(d) = 0
    d -= d.mean()
    # Find max positive step
    pos_mask = d > 1e-15
    neg_mask = d < -1e-15
    if pos_mask.any():
        alpha_pos = ((1.0 - h[pos_mask]) / d[pos_mask]).min()
    else:
        alpha_pos = 1.0
    if neg_mask.any():
        alpha_neg = (h[neg_mask] / (-d[neg_mask])).min()
    else:
        alpha_neg = 1.0
    # Scale d so that the feasible interval is [0, 1]
    alpha_max = min(alpha_pos, alpha_neg)
    if alpha_max < 1e-15:
        return None, 0.0
    d = d / alpha_max
    return d, alpha_max  # d scaled so alpha in [0, 1] is feasible

def exact_line_search(h, d, dx, target_sum, n_iter=60):
    """Minimize c5(h + alpha*d) over feasible alpha using golden section search."""
    # Find feasible interval
    alpha_lo = 0.0
    alpha_hi = 0.0
    pos_mask = d > 1e-15
    neg_mask = d < -1e-15
    if pos_mask.any():
        alpha_hi = ((1.0 - h[pos_mask]) / d[pos_mask]).min()
    else:
        alpha_hi = 1.0
    if neg_mask.any():
        alpha_lo = (-h[neg_mask] / d[neg_mask]).min()  # = h[i]/(-d[i]) for neg d
    else:
        alpha_lo = 0.0
    
    if alpha_hi - alpha_lo < 1e-14:
        return h, c5_of_h(h, dx)
    
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a = alpha_lo
    b = alpha_hi
    c = b - gr * (b - a)
    d_pt = a + gr * (b - a)
    fc = c5_of_h(np.clip(h + c * d, 0, 1), dx)
    fd = c5_of_h(np.clip(h + d_pt * d, 0, 1), dx)
    
    best_alpha = a
    best_val = min(fc, fd)
    if fd < fc:
        best_alpha = d_pt
    
    for _ in range(n_iter):
        if b - a < 1e-12:
            break
        if fc < fd:
            b = d_pt
            d_pt = c
            fd = fc
            c = b - gr * (b - a)
            h_new = np.clip(h + c * d, 0, 1)
            fc = c5_of_h(h_new, dx)
        else:
            a = c
            c = d_pt
            fc = fd
            d_pt = a + gr * (b - a)
            h_new = np.clip(h + d_pt * d, 0, 1)
            fd = c5_of_h(h_new, dx)
        if fc < best_val:
            best_val = fc
            best_alpha = c
        if fd < best_val:
            best_val = fd
            best_alpha = d_pt
    
    h_best = np.clip(h + best_alpha * d, 0, 1)
    # Project to ensure exact feasibility
    h_best = project_feasible(h_best, target_sum)
    return h_best, c5_of_h(h_best, dx)
