def project_feasible(x, target_sum):
    """Project x onto {0<=h<=1, sum(h)=target_sum} using bisection on shift."""
    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def c5_of_h(h, dx):
    return compute_corrs(h).max() * dx

def grad_corr_k(h, k, n):
    """Gradient of corr_k with respect to h."""
    offset = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -offset)
    i_hi = min(n - 1, n - 1 - offset)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx + offset] += (1.0 - h[idx])
    grad[idx] += (-h[idx + offset])
    return grad

def smooth_max_value(corrs, beta):
    return logsumexp(beta * corrs) / beta

def smooth_max_grad(h, dx, beta):
    n = len(h)
    corrs = compute_corrs(h)
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones_like(corrs) / len(corrs)
    else:
        w /= w_sum
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] > 1e-13:

        val = c5_of_h(h_new, dx)

def slsqp_optimize(h0, dx, target_sum, beta, maxiter=800):
    n = len(h0)
    
    def obj(x):
        corrs = compute_corrs(x)
        return logsumexp(beta * corrs) / beta * dx
    
    def grad_obj(x):
        return smooth_max_grad(x, dx, beta)
    
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0, jac=grad_obj, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})

    except Exception:
        h_out = project_feasible(h0, target_sum)
    return h_out, c5_of_h(h_out, dx)

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.special import logsumexp
import time


def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)


def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def c5_of_h(h, dx):
    return compute_corrs(h).max() * dx


def grad_corr_k(h, k, n):
    offset = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -offset)
    i_hi = min(n - 1, n - 1 - offset)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx + offset] += (1.0 - h[idx])
    grad[idx] += (-h[idx + offset])
    return grad


def smooth_max_grad(h, dx, beta):
    n = len(h)
    corrs = compute_corrs(h)
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones_like(corrs) / len(corrs)
    else:
        w /= w_sum
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] > 1e-13:
            grad += w[k] * grad_corr_k(h, k, n)
    return grad * dx


def exact_line_search(h, d, dx, n_grid=200):
    n = len(h)
    a_max = float('inf')
    for i in range(n):
        if d[i] > 1e-15:
            a_max = min(a_max, (1.0 - h[i]) / d[i])
        elif d[i] < -1e-15:
            a_max = min(a_max, h[i] / (-d[i]))
    if a_max <= 0 or a_max == float('inf'):
        return h, c5_of_h(h, dx)
    best_val = c5_of_h(h, dx)
    best_alpha = 0.0
    grid = np.linspace(0.0, a_max, n_grid)
    for alpha in grid:
        h_new = h + alpha * d
        val = c5_of_h(h_new, dx)
        if val < best_val - 1e-15:
            best_val = val
            best_alpha = alpha
    if best_alpha == 0.0:
        return h, best_val
    idx = int(best_alpha / a_max * (n_grid - 1))
    if idx <= 0:
        lo, hi = 0.0, grid[min(2, n_grid - 1)]
    elif idx >= n_grid - 1:
        lo, hi = grid[max(0, n_grid - 3)], a_max
    else:
        lo, hi = grid[idx - 1], grid[idx + 1]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for _ in range(35):
        c_pt = hi - gr * (hi - lo)
        d_pt = lo + gr * (hi - lo)
        f_c = c5_of_h(h + c_pt * d, dx)
        f_d = c5_of_h(h + d_pt * d, dx)
        if f_c < f_d:
            hi = d_pt
        else:
            lo = c_pt
        if hi - lo < 1e-14:
            break
    alpha_opt = (lo + hi) / 2.0
    h_opt = h + alpha_opt * d
    val_opt = c5_of_h(h_opt, dx)
    if val_opt < best_val:
        return h_opt, val_opt
    return h + best_alpha * d, best_val


def slsqp_optimize(h0, dx, target_sum, beta, maxiter=600):
    n = len(h0)

    def obj(x):
        return logsumexp(beta * compute_corrs(x)) / beta * dx

    def grad_obj(x):
        return smooth_max_grad(x, dx, beta)

    def c_fun(x):
        return x.sum() - target_sum

    def c_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': c_fun, 'jac': c_jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0, jac=grad_obj, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        h_out = project_feasible(res.x, target_sum)
    except Exception:
        h_out = project_feasible(h0, target_sum)
    return h_out, c5_of_h(h_out, dx)

def slsqp_z(h0, dx, target_sum, beta, maxiter=500):
    n = len(h0)

    def sigmoid_2z(z):
        return 1.0 / (1.0 + np.exp(-np.clip(2.0 * z, -500, 500)))

    def sigmoid_2z_grad(z):
        h = sigmoid_2z(z)
        return 2.0 * h * (1.0 - h)

    def obj(z):
        return logsumexp(beta * compute_corrs(sigmoid_2z(z))) / beta * dx

    def grad_obj(z):
        sz = sigmoid_2z_grad(z)
        h = sigmoid_2z(z)
        gh = smooth_max_grad(h, dx, beta)
        return sz * gh

    def c_fun(z):
        return sigmoid_2z(z).sum() - target_sum

    def c_jac(z):
        return sigmoid_2z_grad(z)

    cons = [{'type': 'eq', 'fun': c_fun, 'jac': c_jac}]
    h0_clip = np.clip(h0, 1e-8, 1.0 - 1e-8)
    z0 = -0.5 * np.log((1.0 - h0_clip) / h0_clip)
    try:
        res = minimize(obj, z0, jac=grad_obj, method='SLSQP',
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        h_out = project_feasible(sigmoid_2z(res.x), target_sum)
    except Exception:
        h_out = project_feasible(h0_clip, target_sum)
    return h_out, c5_of_h(h_out, dx)

        if time_left_fn() < 25:
            break
        h_opt, c5_val = slsqp_optimize(best_h, dx, target_sum, beta, 500)
        if c5_val < best_c5 - 1e-16:
            best_c5 = c5_val
            best_h = h_opt

    # Z-space SLSQP
    if time_left_fn() > 40:
        for beta in [400, 1200]:
            if time_left_fn() < 20:
                break
            h_opt, c5_val = slsqp_z(best_h, dx, target_sum, beta, 400)
            if c5_val < best_c5 - 1e-16:
                best_c5 = c5_val
                best_h = h_opt

    # LP + exact line search
    if time_left_fn() > 30:
        h_lp, c5_lp = lp_exact_polish(best_h, dx, target_sum, max_iters=120, init_step=0.1)
        if c5_lp < best_c5:
            best_c5 = c5_lp
            best_h = h_lp

    # Pair transfers
    if time_left_fn() > 25:
        h_pt, c5_pt = pair_transfer_exact(best_h, dx, target_sum, n_top=12, n_iter=5)
        if c5_pt < best_c5:
            best_c5 = c5_pt
            best_h = h_pt

    # Random direction line search
    if time_left_fn() > 30:
        h_rd, c5_rd = random_dir_line_search(best_h, dx, target_sum, n_trials=30, n_grid=80)
        if c5_rd < best_c5:
            best_c5 = c5_rd
            best_h = h_rd

    # Second round: LP + pair
    if time_left_fn() > 30:
        h_lp2, c5_lp2 = lp_exact_polish(best_h, dx, target_sum, max_iters=80, init_step=0.06)
        if c5_lp2 < best_c5:
            best_c5 = c5_lp2
            best_h = h_lp2
    if time_left_fn() > 20:
        h_pt2, c5_pt2 = pair_transfer_exact(best_h, dx, target_sum, n_top=10, n_iter=4)
        if c5_pt2 < best_c5:
            best_c5 = c5_pt2
            best_h = h_pt2

    # Final SLSQP
    if time_left_fn() > 15:
        h_f, c5_f = slsqp_optimize(best_h, dx, target_sum, 5000, 400)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f

    # Final pair polish
    if time_left_fn() > 10:
        h_pf, c5_pf = pair_transfer_exact(best_h, dx, target_sum, n_top=8, n_iter=3)
        if c5_pf < best_c5:
            best_c5 = c5_pf
            best_h = h_pf

    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s - (time.time() - t0)

    h0 = np.array(initial_h_values, dtype=float)
    n = len(h0)
    dx = 2.0 / n
    target_sum = n / 2.0
    h0 = project_feasible(h0, target_sum)

    best_h = h0.copy()
    best_c5 = c5_of_h(h0, dx)

    # Main optimization from initial construction
    h_opt, c5_val = full_optimize(h0, dx, target_sum, time_left)
    if c5_val < best_c5:
        best_c5 = c5_val
        best_h = h_opt

    # Perturbation restarts - small perturbations
    for trial in range(12):
        if time_left() < 40:
            break
        scale = 0.04 + 0.02 * trial
        if trial % 3 == 0:
            perturb = scale * (rng.random(n) - 0.5)
        elif trial % 3 == 1:
            idx = rng.choice(n, size=max(5, n // 8), replace=False)
            perturb = np.zeros(n)
            perturb[idx] = rng.choice([-1, 1], size=len(idx)) * scale
        else:
            # Block perturb
            block_size = max(3, n // 20)
            start = rng.randint(0, max(1, n - block_size))
            perturb = np.zeros(n)
            perturb[start:start + block_size] = scale * (rng.random(block_size) - 0.5)
            perturb[start:n] -= perturb[start:start + block_size].sum() / (n - start)

        h_try = project_feasible(best_h + perturb, target_sum)
        h_opt2, c5_val2 = full_optimize(h_try, dx, target_sum, time_left)
        if c5_val2 < best_c5:
            best_c5 = c5_val2
            best_h = h_opt2

    # Try different n values
    if time_left() > 80:
        for n_offset in [-6, -4, -2, 2, 4, 6]:
            if time_left() < 50:
                break
            n_try = n + n_offset
            if n_try < 40 or n_try % 2 != 0:
                continue
            dx_try = 2.0 / n_try
            ts_try = n_try / 2.0
            x_orig = np.linspace(0.0, 2.0, len(best_h), endpoint=False) + 1.0 / len(best_h)
            x_new = np.linspace(0.0, 2.0, n_try, endpoint=False) + 1.0 / n_try
            h_interp = np.interp(x_new, x_orig, best_h)
            h_interp = project_feasible(h_interp, ts_try)
            h_opt3, c5_val3 = full_optimize(h_interp, dx_try, ts_try, time_left)
            if c5_val3 < best_c5:
                best_c5 = c5_val3
                best_h = h_opt3

    # Final intensive polish
    if time_left() > 25 and best_h is not None:
        n_b = len(best_h)
        dx_b = 2.0 / n_b
        ts_b = n_b / 2.0
        # Multiple rounds of LP + pair
        for rnd in range(3):
            if time_left() < 15:
                break
            h_lp, c5_lp = lp_exact_polish(best_h, dx_b, ts_b, max_iters=60, init_step=0.05)
            if c5_lp < best_c5:
                best_c5 = c5_lp
                best_h = h_lp
            h_pt, c5_pt = pair_transfer_exact(best_h, dx_b, ts_b, n_top=10, n_iter=3)
            if c5_pt < best_c5:
                best_c5 = c5_pt
                best_h = h_pt
            h_rd, c5_rd = random_dir_line_search(best_h, dx_b, ts_b, n_trials=20, n_grid=60)
            if c5_rd < best_c5:
                best_c5 = c5_rd
                best_h = h_rd

    # Final validation
    if best_h is None:
        best_h = h0
        best_c5 = c5_of_h(h0, 2.0 / n)
    else:
        n_final = len(best_h)
        best_h = project_feasible(best_h, n_final / 2.0)
        best_c5 = c5_of_h(best_h, 2.0 / n_final)

    return best_h, best_c5, len(best_h)
