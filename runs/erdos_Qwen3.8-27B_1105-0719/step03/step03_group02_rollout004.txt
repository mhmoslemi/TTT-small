Let me reconsider the approach:

For exact line search in direction d:
f(α) = max_k corr_k(h + α*d) * dx

Each corr_k(h + α*d) = sum_j (h[j] + α*d[j]) * (1 - h[j+k] - α*d[j+k])
= sum_j h[j](1-h[j+k]) - α*sum_j(h[j]*d[j+k] + d[j]*(1-h[j+k])) + α²*sum_j d[j]*d[j+k]

So each corr_k is quadratic in α, and f(α) = max of quadratics * dx.

The max of quadratics is convex (max of convex functions is convex... wait, quadratics aren't necessarily convex). Actually, each corr_k as a function of α is:
corr_k(α) = c0_k + c1_k * α + c2_k * α²

where c2_k = sum_j d[j]*d[j+k]. This could be positive, negative, or zero.

If c2_k > 0, it's convex in α. If c2_k < 0, it's concave. The max of a mix could be non-convex.

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.special import logsumexp
import time

def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def c5_of_h(h, dx):
    return compute_corrs(h).max() * dx

def grad_corr_k(h, k, n):
    offset = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -offset)
    i_hi = min(n - 1, n - 1 - offset)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx + offset] += (1.0 - h[idx])
    grad[idx] += (-h[idx + offset])
    return grad

def full_grad(h, n):
    """Compute gradient vector for the max objective (subgradient at max)."""
    corrs = compute_corrs(h)
    max_c = corrs.max()
    active = np.where(corrs >= max_c - 1e-12)[0]
    g = np.zeros(n)
    for k in active:
        g += grad_corr_k(h, k, n)
    g /= len(active)
    return g

def smooth_max_obj(h, dx, beta):
    corrs = compute_corrs(h)
    return logsumexp(beta * corrs) / beta * dx

def smooth_max_grad(h, dx, beta):
    n = len(h)
    corrs = compute_corrs(h)
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones_like(corrs) / len(corrs)
    else:
        w /= w_sum
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] > 1e-13:
            grad += w[k] * grad_corr_k(h, k, n)
    return grad * dx

def exact_line_search(h, d, dx, target_sum, alpha_max=1.0, n_evals=200):
    """
    Exact line search along direction d.
    d should satisfy sum(d) = 0.
    We find feasible alpha range, then do dense evaluation + golden section refinement.
    """
    n = len(h)
    # Find feasible alpha range
    # h + alpha*d must be in [0,1]
    pos_d = d > 1e-15
    neg_d = d < -1e-15
    alpha_lo = 0.0
    alpha_hi = alpha_max
    
    if pos_d.any():
        alpha_hi = min(alpha_hi, ((1.0 - h)[pos_d] / d[pos_d]).min())
    if neg_d.any():
        alpha_lo = max(alpha_lo, (-h[neg_d] / d[neg_d]).min())
    
    if alpha_hi <= alpha_lo:
        return h, c5_of_h(h, dx), 0.0
    
    # Dense evaluation
    alphas = np.linspace(alpha_lo, alpha_hi, n_evals)
    best_val = float('inf')
    best_alpha = 0.0
    
    for a in alphas:
        h_try = h + a * d
        h_try = np.clip(h_try, 0.0, 1.0)
        h_try = project_feasible(h_try, target_sum)
        val = c5_of_h(h_try, dx)
        if val < best_val:
            best_val = val
            best_alpha = a
    
    # Golden section refinement around best_alpha
    if best_alpha > alpha_lo and best_alpha < alpha_hi:
        idx = np.argmin(alphas - best_alpha)  # not great, let's just use neighborhood
        a_lo = max(alpha_lo, best_alpha - (alpha_hi - alpha_lo) / n_evals * 2)
        a_hi = min(alpha_hi, best_alpha + (alpha_hi - alpha_lo) / n_evals * 2)
        
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c_pt = a_hi - gr * (a_hi - a_lo)
        d_pt = a_lo + gr * (a_hi - a_lo)
        
        def eval_alpha(a):
            h_try = h + a * d
            h_try = np.clip(h_try, 0.0, 1.0)
            h_try = project_feasible(h_try, target_sum)
            return c5_of_h(h_try, dx)
        
        fc = eval_alpha(c_pt)
        fd = eval_alpha(d_pt)
        for _ in range(30):
            if a_hi - a_lo < 1e-10:
                break
            if fc < fd:
                a_hi = d_pt
                d_pt = c_pt
                fd = fc
                c_pt = a_hi - gr * (a_hi - a_lo)
                fc = eval_alpha(c_pt)
            else:
                a_lo = c_pt
                c_pt = d_pt
                fc = fd
                d_pt = a_lo + gr * (a_hi - a_lo)
                fd = eval_alpha(d_pt)
        
        a_opt = (a_lo + a_hi) / 2.0
        h_opt = h + a_opt * d
        h_opt = np.clip(h_opt, 0.0, 1.0)
        h_opt = project_feasible(h_opt, target_sum)
        val_opt = c5_of_h(h_opt, dx)
        if val_opt < best_val:
            return h_opt, val_opt, a_opt
    
    h_opt = h + best_alpha * d
    h_opt = np.clip(h_opt, 0.0, 1.0)
    h_opt = project_feasible(h_opt, target_sum)
    return h_opt, c5_of_h(h_opt, dx), best_alpha

def pair_exact_search(h, i, j, dx, target_sum):
    """
    For coordinates i and j, find optimal h[i], h[j] with h[i]+h[j] = const.
    """
    n = len(h)
    s = h[i] + h[j]
    # t = h[i], h[j] = s - t
    # t must be in [max(0, s-1), min(1, s)]
    t_lo = max(0.0, s - 1.0)
    t_hi = min(1.0, s)
    
    if t_hi - t_lo < 1e-14:
        return h, c5_of_h(h, dx)
    
    def eval_t(t):
        h_try = h.copy()
        h_try[i] = t
        h_try[j] = s - t
        return c5_of_h(h_try, dx)
    
    # Dense scan
    n_scan = 100
    ts = np.linspace(t_lo, t_hi, n_scan)
    vals = np.array([eval_t(t) for t in ts])
    best_idx = np.argmin(vals)
    best_val = vals[best_idx]
    best_t = ts[best_idx]
    
    # Golden section
    lo = max(t_lo, ts[max(0, best_idx-1)])
    hi = min(t_hi, ts[min(n_scan-1, best_idx+1)])
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c_pt = hi - gr * (hi - lo)
    d_pt = lo + gr * (hi - lo)
    fc, fd = eval_t(c_pt), eval_t(d_pt)
    for _ in range(40):
        if hi - lo < 1e-12:
            break
        if fc < fd:
            hi = d_pt
            d_pt = c_pt
            fd = fc
            c_pt = hi - gr * (hi - lo)
            fc = eval_t(c_pt)
        else:
            lo = c_pt
            c_pt = d_pt
            fc = fd
            d_pt = lo + gr * (hi - lo)
            fd = eval_t(d_pt)
    
    t_opt = (lo + hi) / 2.0
    h_opt = h.copy()
    h_opt[i] = t_opt
    h_opt[j] = s - t_opt
    h_opt = np.clip(h_opt, 0.0, 1.0)
    h_opt = project_feasible(h_opt, target_sum)
    return h_opt, c5_of_h(h_opt, dx)

def lp_direction(h, dx, target_sum, n_active_top=50):
    """
    Generate a direction using LP linearization.
    Returns direction d with sum(d)=0, or None.
    """
    n = len(h)
    corrs = compute_corrs(h)
    max_c = corrs.max()
    
    top_k = np.argsort(corrs)[-n_active_top:]
    
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_rows = []
    b_rows = []
    for k in top_k:
        row = np.zeros(n + 1)
        row[:n] = dx * grad_corr_k(h, k, n)
        row[n] = -1.0
        A_rows.append(row)
        b_rows.append(-corrs[k] * dx - 1e-14)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    bounds = []
    for i in range(n):
        bounds.append((max(-h[i], -0.3), min(1.0 - h[i], 0.3)))
    bounds.append((0.0, None))
    
    try:
        res = linprog(c_obj, A_ub=np.array(A_rows), b_ub=np.array(b_rows),
                      A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    except Exception:
        return None
    if res.status != 0:
        return None
    
    delta = res.x[:n]
    if np.max(np.abs(delta)) < 1e-12:
        return None
    return delta

def slsqp_smooth(h0, dx, target_sum, beta, maxiter=1000):
    n = len(h0)
    
    def obj(x):
        return smooth_max_obj(x, dx, beta)
    
    def grad_obj(x):
        return smooth_max_grad(x, dx, beta)
    
    def c_fun(x):
        return x.sum() - target_sum
    
    def c_jac(x):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': c_fun, 'jac': c_jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0, jac=grad_obj, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        h_out = project_feasible(res.x, target_sum)
    except Exception:
        h_out = project_feasible(h0, target_sum)
    return h_out, c5_of_h(h_out, dx)

def subgrad_lsqr(h0, dx, target_sum, n_iter=5000, lr_init=0.05):
    """Subgradient method with exact line search on the step."""
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    lr = lr_init
    no_improve = 0
    
    for it in range(n_iter):
        corrs = compute_corrs(h)
        c5_val = corrs.max() * dx
        
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 2000:
                break
        
        # Compute subgradient (average of active)
        max_c = corrs.max()
        tol = max_c - 0.05 * max_c  # relatively active
        active = np.where(corrs >= tol)[0]
        if len(active) > 30:
            active = np.argsort(corrs)[-30:]
        
        g = np.zeros(n)
        for k in active:
            g += grad_corr_k(h, k, n)
        g /= len(active)
        g *= dx
        
        # Project gradient to sum(d)=0
        g_mean = g.mean()
        d = g - g_mean
        
        # Compute step size
        gnorm = np.linalg.norm(d)
        if gnorm < 1e-14:
            break
        
        d_norm = d / gnorm
        step = lr * (best_c5 - c5_val + 0.01) / gnorm  # heuristic step
        step = min(step, 0.2)
        
        # Try a few step sizes
        best_step = 0
        best_val = c5_val
        for s in [step, step*0.5, step*0.25, step*2, step*0.75]:
            if s <= 0 or s > 0.5:
                continue
            h_try = h - s * d_norm
            h_try = project_feasible(h_try, target_sum)
            val = c5_of_h(h_try, dx)
            if val < best_val:
                best_val = val
                best_step = s
        
        if best_step > 0:
            h = h - best_step * d_norm
            h = project_feasible(h, target_sum)
            lr = min(lr * 1.1, 0.3)
        else:
            lr *= 0.7
            if lr < 1e-6:
                break
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.special import logsumexp
import time


def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)


def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def c5_of_h(h, dx):
    return compute_corrs(h).max() * dx


def grad_corr_k(h, k, n):
    offset = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -offset)
    i_hi = min(n - 1, n - 1 - offset)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx + offset] += (1.0 - h[idx])
    grad[idx] += (-h[idx + offset])
    return grad


def smooth_max_obj(h, dx, beta):
    corrs = compute_corrs(h)
    return logsumexp(beta * corrs) / beta * dx


def smooth_max_grad(h, dx, beta):
    n = len(h)
    corrs = compute_corrs(h)
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones_like(corrs) / len(corrs)
    else:
        w /= w_sum
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] > 1e-13:
            grad += w[k] * grad_corr_k(h, k, n)
    return grad * dx


def exact_line_search(h, d, dx, target_sum, alpha_max=1.0, n_scan=150):
    """Exact line search along direction d (must have sum(d)=0)."""
    n = len(h)
    d_sum = d.sum()
    if abs(d_sum) > 1e-10:
        d = d - d.sum() / n

    pos_d = d > 1e-15
    neg_d = d < -1e-15
    alpha_lo = 0.0
    alpha_hi = alpha_max

    if pos_d.any():
        alpha_hi = min(alpha_hi, ((1.0 - h)[pos_d] / d[pos_d]).min())
    if neg_d.any():
        alpha_lo = max(alpha_lo, (-h[neg_d] / d[neg_d]).min())

    if alpha_hi - alpha_lo < 1e-14:
        return h, c5_of_h(h, dx), 0.0

    def eval_alpha(a):
        h_try = h + a * d
        if (h_try.min() < -1e-12) or (h_try.max() > 1.0 + 1e-12):
            return float('inf')
        return c5_of_h(h_try, dx)

    alphas = np.linspace(alpha_lo, alpha_hi, n_scan)
    best_val = float('inf')
    best_alpha = 0.0
    vals = np.empty(n_scan)
    for i_a, a in enumerate(alphas):
        v = eval_alpha(a)
        vals[i_a] = v
        if v < best_val:
            best_val = v
            best_alpha = a

    if best_val == float('inf'):
        return h, c5_of_h(h, dx), 0.0

    idx = np.argmin(vals)
    spread = (alpha_hi - alpha_lo) / n_scan * 3
    a_lo = max(alpha_lo, best_alpha - spread)
    a_hi = min(alpha_hi, best_alpha + spread)

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c_pt = a_hi - gr * (a_hi - a_lo)
    d_pt = a_lo + gr * (a_hi - a_lo)
    fc = eval_alpha(c_pt)
    fd = eval_alpha(d_pt)
    for _ in range(35):
        if a_hi - a_lo < 1e-11:
            break
        if fc < fd:
            a_hi = d_pt
            d_pt = c_pt
            fd = fc
            c_pt = a_hi - gr * (a_hi - a_lo)
            fc = eval_alpha(c_pt)
        else:
            a_lo = c_pt
            c_pt = d_pt
            fc = fd
            d_pt = a_lo + gr * (a_hi - a_lo)
            fd = eval_alpha(d_pt)

    a_opt = (a_lo + a_hi) / 2.0
    h_opt = h + a_opt * d
    h_opt = project_feasible(h_opt, target_sum)
    val_opt = c5_of_h(h_opt, dx)
    if val_opt < best_val:
        best_val = val_opt
        best_alpha = a_opt

    h_out = h + best_alpha * d
    h_out = project_feasible(h_out, target_sum)
    return h_out, c5_of_h(h_out, dx), best_alpha


def pair_exact_search(h, i, j, dx, target_sum):
    n = len(h)
    s = h[i] + h[j]
    t_lo = max(0.0, s - 1.0)
    t_hi = min(1.0, s)
    if t_hi - t_lo < 1e-14:
        return h, c5_of_h(h, dx)

    def eval_t(t):
        h_try = h.copy()
        h_try[i] = t
        h_try[j] = s - t
        return c5_of_h(h_try, dx)

    n_scan = 80
    ts = np.linspace(t_lo, t_hi, n_scan)
    best_val = float('inf')
    best_t = t_lo
    for t in ts:
        v = eval_t(t)
        if v < best_val:
            best_val = v
            best_t = t

    idx = np.argmin([eval_t(t) for t in [ts[max(0, np.argmin(ts - best_t) - 1)],
                                          ts[min(n_scan-1, np.argmin(ts - best_t) + 1)]] + [best_t])
    spread = (t_hi - t_lo) / n_scan * 3
    lo = max(t_lo, best_t - spread)
    hi = min(t_hi, best_t + spread)

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c_pt = hi - gr * (hi - lo)
    d_pt = lo + gr * (hi - lo)
    fc = eval_t(c_pt)
    fd = eval_t(d_pt)
    for _ in range(40):
        if hi - lo < 1e-12:
            break
        if fc < fd:
            hi = d_pt
            d_pt = c_pt
            fd = fc
            c_pt = hi - gr * (hi - lo)
            fc = eval_t(c_pt)
        else:
            lo = c_pt
            c_pt = d_pt
            fc = fd
            d_pt = lo + gr * (hi - lo)
            fd = eval_t(d_pt)

    t_opt = (lo + hi) / 2.0
    h_opt = h.copy()
    h_opt[i] = t_opt
    h_opt[j] = s - t_opt
    h_opt = project_feasible(h_opt, target_sum)
    return h_opt, c5_of_h(h_opt, dx)


def lp_direction(h, dx, target_sum, n_active=50, box_size=0.25):
    n = len(h)
    corrs = compute_corrs(h)
    top_k = np.argsort(corrs)[-n_active:]

    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_rows = []
    b_rows = []
    for k in top_k:
        row = np.zeros(n + 1)
        row[:n] = dx * grad_corr_k(h, k, n)
        row[n] = -1.0
        A_rows.append(row)
        b_rows.append(-corrs[k] * dx - 1e-14)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)

    bounds = []
    for i in range(n):
        bounds.append((max(-h[i], -box_size), min(1.0 - h[i], box_size)))
    bounds.append((0.0, None))

    try:
        res = linprog(c_obj, A_ub=np.array(A_rows), b_ub=np.array(b_rows),
                      A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    except Exception:
        return None
    if res.status != 0:
        return None

    delta = res.x[:n]
    if np.max(np.abs(delta)) < 1e-12:
        return None
    return delta


def slsqp_smooth(h0, dx, target_sum, beta, maxiter=800):
    n = len(h0)

    def obj(x):
        return smooth_max_obj(x, dx, beta)

    def grad_obj(x):
        return smooth_max_grad(x, dx, beta)

    def c_fun(x):
        return x.sum() - target_sum

    def c_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': c_fun, 'jac': c_jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0, jac=grad_obj, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        h_out = project_feasible(res.x, target_sum)
    except Exception:
        h_out = project_feasible(h0, target_sum)
    return h_out, c5_of_h(h_out, dx)


def do_lp_line_search_loop(h, dx, target_sum, max_iters=200, n_active=50):
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    stall = 0

    for it in range(max_iters):
        d = lp_direction(best_h, dx, target_sum, n_active=n_active, box_size=0.3)
        if d is None:
            break
        h_new, c5_new, _ = exact_line_search(best_h, d, dx, target_sum,
                                             alpha_max=1.0, n_scan=120)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new
            stall = 0
        else:
            stall += 1
            if stall > 30:
                break
    return best_h, best_c5


def do_pair_search(h, dx, target_sum, n_crit=25, n_passes=5, n_random=30):
    n = len(h)
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)

    for p in range(n_passes):
        improved = False
        corrs = compute_corrs(best_h)
        max_c = corrs.max()
        active = np.where(corrs >= max_c * 0.88)[0]
        g = np.zeros(n)
        for k in active:
            g += grad_corr_k(best_h, k, n)

        abs_g = np.abs(g)
        threshold = np.percentile(abs_g, 100 * (1 - n_crit / n))
        crit_idx = np.where(abs_g >= threshold)[0]
        if len(crit_idx) < n_crit:
            crit_idx = np.argsort(abs_g)[-n_crit:]

        for i in range(len(crit_idx)):
            for j in range(i + 1, len(crit_idx)):
                ii, jj = crit_idx[i], crit_idx[j]
                h_new, c5_new = pair_exact_search(best_h, ii, jj, dx, target_sum)
                if c5_new < best_c5 - 1e-13:
                    best_c5 = c5_new
                    best_h = h_new
                    improved = True
                    break
            if improved:
                break

        for _ in range(n_random):
            ii, jj = np.random.choice(n, 2, replace=False)
            h_new, c5_new = pair_exact_search(best_h, ii, jj, dx, target_sum)
            if c5_new < best_c5 - 1e-13:
                best_c5 = c5_new
                best_h = h_new
                improved = True

        if not improved:
            break
    return best_h, best_c5

        best_c5 = c5_of_h(best_h, 2.0 / n_final)

    return best_h, best_c5, len(best_h)
