So all entries are ≤ 0. The matrix is symmetric (since if j = i+s_k then i = j-s_k, and w is the same). 

Is it negative semi-definite? A symmetric matrix with all non-positive entries is NOT necessarily negative semi-definite. For example, [[0, -1], [-1, 0]] has eigenvalues 1 and -1, so it's indefinite.

So the QP might be non-convex. That's fine, SLSQP can handle it (it finds a local optimum).

import numpy as np
from scipy.optimize import minimize, linprog
import time


def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def c5_of_h(h, dx):
    return compute_corrs(h).max() * dx


def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)


def grad_corr_k(h, k, n):
    offset = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -offset)
    i_hi = min(n - 1, n - 1 - offset)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx + offset] += (1.0 - h[idx])
    grad[idx] += (-h[idx + offset])
    return grad


def binary_swap_delta(b, i, j, n):
    """Compute delta in correlations when b[i] goes 1->0 and b[j] goes 0->1."""
    delta = np.zeros(2 * n - 1)
    s_range = np.arange(-(n - 1), n)
    for s in s_range:
        d = 0.0
        # i as first factor: n+s=i -> n=i-s
        if 0 <= i - s < n:
            d -= (1.0 - b[i - s])
        # i as second factor: n=i
        if 0 <= i + s < n:
            d += b[i + s]
        # j as first factor: n+s=j -> n=j-s
        if 0 <= j - s < n:
            d += (1.0 - b[j - s])
        # j as second factor: n=j
        if 0 <= j + s < n:
            d -= b[j + s]
        delta[s + (n - 1)] = d
    return delta


def binary_local_search(b0, n, max_sweeps=200, top_lags=8, time_limit=None, t0=None):
    b = b0.astype(float).copy()
    corrs = compute_corrs(b)
    max_c = corrs.max()
    best_b = b.copy()
    best_c5_val = max_c
    ones_idx = np.where(b == 1.0)[0]
    zeros_idx = np.where(b == 0.0)[0]

    for sweep in range(max_sweeps):
        if time_limit is not None and t0 is not None:
            if time.time() - t0 > time_limit:
                break
        improved = False
        n_ones = len(ones_idx)
        n_zeros = len(zeros_idx)
        for a in range(n_ones):
            for bidx in range(n_zeros):
                i = ones_idx[a]
                j = zeros_idx[bidx]
                if i == j:
                    continue
                delta = binary_swap_delta(b, i, j, n)
                new_corrs = corrs + delta
                new_max = new_corrs.max()
                if new_max < max_c - 1e-12:
                    b[i] = 0.0
                    b[j] = 1.0
                    corrs = new_corrs
                    max_c = new_max
                    if max_c < best_c5_val:
                        best_c5_val = max_c
                        best_b = b.copy()
                    ones_idx = np.where(b == 1.0)[0]
                    zeros_idx = np.where(b == 0.0)[0]
                    improved = True
        if not improved:
            break
    return best_b, best_c5_val

    """Optimized: only check top lags for the max."""
    b = b0.astype(float).copy()

                        b[j] = 0.0
        if not improved:

def qp_obj_and_grad(h, w, dx):
    n = len(h)
    corrs = compute_corrs(h)
    obj = np.dot(w, corrs) * dx
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] > 1e-10:
            grad += w[k] * grad_corr_k(h, k, n)
    grad *= dx
    return obj, grad


def qp_best_response(h0, w, dx, target_sum, maxiter=300):
    n = len(h0)

    def obj_fun(h):
        val, _ = qp_obj_and_grad(h, w, dx)
        return val

    def grad_fun(h):
        _, g = qp_obj_and_grad(h, w, dx)
        return g

    def c_fun(h):
        return h.sum() - target_sum

    def c_jac(h):
        return np.ones(n)

    res = minimize(obj_fun, h0, jac=grad_fun, method='SLSQP',
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': c_fun, 'jac': c_jac}],
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    h_out = project_feasible(res.x, target_sum)
    return h_out


def fictitious_play(h0, dx, target_sum, max_rounds=400, time_limit=None, t0=None):
    n = len(h0)
    h = h0.copy()
    w = np.ones(2 * n - 1) / (2 * n - 1)
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    hist_w = np.zeros(2 * n - 1)
    hist_count = 0

    for rnd in range(max_rounds):
        if time_limit is not None and t0 is not None:
            if time.time() - t0 > time_limit:
                break
        # Best response
        h = qp_best_response(h, w, dx, target_sum, maxiter=200)
        corrs = compute_corrs(h) * dx
        c5_val = corrs.max()
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()

        # Update weights: uniform over history of worst lags
        worst_k = np.argmax(corrs)
        hist_w[worst_k] += 1.0
        hist_count += 1
        # Use recent uniform over top lags
        top_frac = 0.15
        threshold = np.sort(corrs)[int((1 - top_frac) * len(corrs)):]
        threshold = threshold[0] if len(threshold) > 0 else corrs.max()
        top_mask = corrs >= threshold - 1e-10
        w_new = top_mask.astype(float)
        w_new /= w_new.sum()
        # Blend with history
        hist_frac = min(1.0, 20.0 / hist_count)
        w = (1 - hist_frac) * w_new + hist_frac * (hist_w / (hist_w.sum() + 1e-10))

    return best_h, best_c5


def pair_transfer_line_search(h, i, j, corrs, dx, target_sum):
    """Find optimal transfer from j to i (h[i]+=t, h[j]-=t) using exact max evaluation."""
    n = len(h)
    t_lo = max(0.0, h[j] - 1.0 + h[i])  # ensure h[j]-t >= 0
    t_hi = min(1.0 - h[i], h[j])  # ensure h[i]+t <= 1
    t_lo = max(t_lo, 0.0)
    t_hi = min(t_hi, 1.0)
    if t_hi - t_lo < 1e-14:
        return 0.0

    # delta_k = grad effect of moving t from j to i
    # corr_k changes linearly: corr_k(t) = corr_k(0) + t * d_k
    # d_k = (component at i) - (component at j) for the grad of corr_k
    d = np.zeros(2 * n - 1)
    for k in range(2 * n - 1):
        gk = grad_corr_k(h, k, n)
        d[k] = gk[i] - gk[j]

    # Minimize max_k (corrs[k] + t*d[k]) over t in [t_lo, t_hi]
    # This is minimizing a convex piecewise linear function
    # Use ternary search (works for convex functions)
    a, b = t_lo, t_hi
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for _ in range(60):
        if b - a < 1e-13:
            break
        m1 = b - gr * (b - a)
        m2 = a + gr * (b - a)
        f1 = (corrs + m1 * d).max()
        f2 = (corrs + m2 * d).max()
        if f1 < f2:
            b = m2
        else:
            a = m1
    return (a + b) / 2.0


def pair_transfer_polish(h0, dx, target_sum, n_top=30, n_iter=10, time_limit=None, t0=None):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()

    for iteration in range(n_iter):

        corrs = compute_corrs(h)
        max_c = corrs.max()
        improved = False
        top_idx = np.argsort(grad_corr_k(h, int(np.argmax(corrs)), n))
        dec_idx = top_idx[-n_top:]
        inc_idx = top_idx[:n_top]

        for i in dec_idx:
            for j in inc_idx:
                if i == j:
                    continue
                # Try moving from high-grad to low-grad
                t_opt = pair_transfer_line_search(h, j, i, corrs, dx, target_sum)
                if abs(t_opt) < 1e-14:
                    continue
                new_h = h.copy()
                new_h[i] += t_opt
                new_h[j] -= t_opt
                new_h = np.clip(new_h, 0.0, 1.0)
                new_h = project_feasible(new_h, target_sum)
                new_c5 = c5_of_h(new_h, dx)
                if new_c5 < best_c5 - 1e-14:
                    h = new_h
                    best_c5 = new_c5
                    best_h = h.copy()
                    corrs = compute_corrs(h)
                    improved = True
                # Also try reverse
                t_opt2 = pair_transfer_line_search(h, i, j, corrs, dx, target_sum)
                if abs(t_opt2) < 1e-14:

                new_h2 = h.copy()
                new_h2[j] += t_opt2
                new_h2[i] -= t_opt2

def smooth_max_obj(h, dx, beta):
    corrs = compute_corrs(h)
    shifted = corrs - corrs.max()
    return (np.exp(beta * shifted).sum() / (beta * len(corrs))) * dx + corrs.max() * dx


def smooth_max_grad(h, dx, beta):
    n = len(h)
    corrs = compute_corrs(h)
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    w /= w.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] > 1e-10:
            grad += w[k] * grad_corr_k(h, k, n)
    return grad * dx


def slsqp_beta(h0, dx, target_sum, beta, maxiter=2000):
    n = len(h0)

    def obj(x):
        corrs = compute_corrs(x)
        shifted = corrs - corrs.max()
        return (np.exp(beta * shifted).max() / beta + corrs.max()) * dx

    def grad_obj(x):
        return smooth_max_grad(x, dx, beta)

    def c_fun(x):
        return x.sum() - target_sum

    def c_jac(x):
        return np.ones(n)

    try:
        res = minimize(obj, h0, jac=grad_obj, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': c_fun, 'jac': c_jac}],
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        h_out = project_feasible(res.x, target_sum)
    except Exception:
        h_out = project_feasible(h0, target_sum)
    return h_out, c5_of_h(h_out, dx)


def lp_polish(h0, dx, target_sum, max_iters=200, init_step=0.15, time_limit=None, t0=None):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    step = init_step
    stall = 0

    for it in range(max_iters):
        if time_limit is not None and t0 is not None:
            if time.time() - t0 > time_limit:
                break
        corrs = compute_corrs(h)
        max_c = corrs.max()
        c5_val = max_c * dx

        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 50:
            step *= 0.6
            stall = 0
            if step < 1e-8:
                break

        n_active = max(10, min(60, int(0.3 * (2 * n - 1))))
        top_idx = np.argsort(corrs)[-n_active:]

        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        A_rows = []
        b_rows = []
        for k in top_idx:
            row = np.zeros(n + 1)
            row[:n] = dx * grad_corr_k(h, k, n)
            row[n] = -1.0
            A_rows.append(row)
            b_rows.append(-corrs[k] * dx - 1e-13)

        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)

        bounds = []
        for i in range(n):
            bounds.append((max(-h[i], -step), min(1.0 - h[i], step)))
        bounds.append((0.0, None))

        try:
            res = linprog(c_obj, A_ub=np.array(A_rows), b_ub=np.array(b_rows),
                          A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
        except Exception:
            break
        if res.status != 0:
            break
        delta = res.x[:n]
        if np.max(np.abs(delta)) < 1e-14:
            break

        h_new = project_feasible(h + delta, target_sum)
        new_c5 = c5_of_h(h_new, dx)
        if new_c5 < c5_val - 1e-14:

        else:
            improved = False
            for frac in [0.5, 0.25, 0.1]:

                if new_c5 < c5_val - 1e-14:
                    h = h_new
                    improved = True

def make_binary_random(n, rng):
    pos = rng.choice(n, size=n // 2, replace=False)
    b = np.zeros(n)
    b[pos] = 1.0
    return b


def make_periodic_binary(n, rng, period):
    b = np.zeros(n)
    fill = max(1, int(n / 2.0 / n * period))
    for i in range(n):
        if (i % period) < fill:
            b[i] = 1.0
    # Fix count
    ones = np.sum(b == 1.0)
    if ones > n // 2:
        extra = np.where(b == 1.0)[0]
        rng.shuffle(extra)
        b[extra[:ones - n // 2]] = 0.0
    elif ones < n // 2:
        missing = np.where(b == 0.0)[0]
        rng.shuffle(missing)
        b[missing[:n // 2 - ones]] = 1.0
    return b

    if time_left() > 60:
        h_opt, c5_val = full_local_optimize(h0, dx0, target0, time_left, t0)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_opt
            best_n = n0

    # Phase 2: Fictitious play on parent size
    if time_left() > 60:
        h_fp, c5_fp = fictitious_play(best_h, dx0, target0, max_rounds=300, t0=t0)
        if c5_fp < best_c5:
            best_c5 = c5_fp
            best_h = h_fp
            best_n = n0
        # Polish the result
        if time_left() > 30:
            h_pol, c5_pol = full_local_optimize(best_h, dx0, target0, time_left, t0)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol

    # Phase 3: Binary search at parent size
    if time_left() > 40:
        b_best = None
        b_best_c = float('inf')
        for trial in range(30):
            if time_left() < 5:
                break
            b = make_binary_random(n0, rng)
            b_opt, b_c = fast_binary_local_search(b, n0, max_sweeps=100, t0=t0)
            if b_c < b_best_c:
                b_best_c = b_c
                b_best = b_opt
        if b_best is not None:
            b_c5_continuous = b_best_c * dx0
            if b_c5_continuous < best_c5:
                # Convert to continuous and polish
                h_from_bin = b_best.copy()
                h_from_bin = project_feasible(h_from_bin, target0)
                h_pol, c5_pol = full_local_optimize(h_from_bin, dx0, target0, time_left, t0)
                if c5_pol < best_c5:
                    best_c5 = c5_pol
                    best_h = h_pol
                    best_n = n0

    # Phase 4: Try different n values
    n_candidates = []
    for nn in range(60, 180, 2):
        if nn != n0:
            n_candidates.append(nn)
    # Sort by proximity to current best n
    n_candidates.sort(key=lambda x: abs(x - best_n))
    n_candidates = n_candidates[:20]

    for n_try in n_candidates:
        if time_left() < 40:
            break
        dx_try = 2.0 / n_try
        ts_try = n_try / 2.0
        # Interpolate best solution to new size
        x_orig = np.linspace(0.0, 2.0, best_n, endpoint=False) + 1.0 / best_n
        x_new = np.linspace(0.0, 2.0, n_try, endpoint=False) + 1.0 / n_try
        h_interp = np.interp(x_new, x_orig, best_h)
        h_interp = project_feasible(h_interp, ts_try)

        h_opt, c5_val = full_local_optimize(h_interp, dx_try, ts_try, time_left, t0)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_opt
            best_n = n_try

        # Also try binary at this size
        if time_left() > 30:
            b = make_binary_random(n_try, rng)
            b_opt, b_c = fast_binary_local_search(b, n_try, max_sweeps=80, t0=t0)
            b_c5 = b_c * dx_try
            if b_c5 < best_c5 + 0.002:
                h_from_b = b_opt.astype(float)
                h_from_b = project_feasible(h_from_b, ts_try)
                h_pol2, c5_pol2 = full_local_optimize(h_from_b, dx_try, ts_try, time_left, t0)
                if c5_pol2 < best_c5:
                    best_c5 = c5_pol2
                    best_h = h_pol2
                    best_n = n_try

    # Phase 5: Multi-start perturbations on best
    for trial in range(20):
        if time_left() < 20:
            break
        if trial % 3 == 0:
            perturb = 0.03 * (rng.random(best_n) - 0.5)
        elif trial % 3 == 1:
            perturb = 0.1 * (rng.random(best_n) - 0.5)
        else:
            idx = rng.choice(best_n, size=max(3, best_n // 15), replace=False)
            perturb = np.zeros(best_n)
            perturb[idx] = rng.choice([-1, 1], size=len(idx)) * 0.08
        h_try = project_feasible(best_h + perturb, best_n / 2.0)
        h_opt2, c5_val2 = full_local_optimize(h_try, 2.0 / best_n, best_n / 2.0, time_left, t0)
        if c5_val2 < best_c5:
            best_c5 = c5_val2
            best_h = h_opt2

    # Phase 6: Final intensive polish
    if time_left() > 15:
        n_b = len(best_h)
        dx_b = 2.0 / n_b
        ts_b = n_b / 2.0
        # Very high beta SLSQP
        for beta in [5000, 10000, 20000]:
            if time_left() < 10:
                break
            h_f, c5_f = slsqp_beta(best_h, dx_b, ts_b, beta, 3000)
            if c5_f < best_c5:
                best_c5 = c5_f
                best_h = h_f
        # LP
        if time_left() > 10:
            h_lp, c5_lp = lp_polish(best_h, dx_b, ts_b, max_iters=150, init_step=0.05)
            if c5_lp < best_c5:
                best_c5 = c5_lp
                best_h = h_lp
        # Pair transfer final
        if time_left() > 10:
            h_pt, c5_pt = pair_transfer_polish(best_h, dx_b, ts_b, n_top=35, n_iter=8, t0=t0)
            if c5_pt < best_c5:
                best_c5 = c5_pt
                best_h = h_pt

    # Final validation
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final / 2.0)
    best_c5 = c5_of_h(best_h, 2.0 / n_final)

    return best_h, best_c5, n_final
