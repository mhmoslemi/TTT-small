import numpy as np
import time

def project_feasible(x, target_sum):
    """Project x onto {h: 0<=h<=1, sum(h)=target_sum}"""
    n = len(x)
    x = np.clip(np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0), 0.0, 1.0)
    
    # Bisection on the shift
    lo, hi = -5.0, 5.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    
    # Fine correction
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    
    return np.clip(h, 0.0, 1.0)

def c5_of_h(h, dx):
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def smooth_power_gradient(h, q, corr=None):
    """Compute gradient of sum(corr^q) with respect to h, projected to sum-zero."""
    n = len(h)
    if corr is None:
        corr = np.correlate(h, 1.0 - h, mode='full')
    
    # Stabilize: normalize by max
    max_c = max(float(corr.max()), 1e-30)
    a = corr / max_c
    
    # Compute weights w[m] proportional to corr[m]^(q-1)
    eps = 1e-30
    a_clipped = np.maximum(a, eps)
    log_w = (q - 1) * np.log(a_clipped)
    max_log = log_w.max()
    w = np.exp(log_w - max_log)  # bounded in [0, 1]
    
    g = np.zeros(n)
    # d=0
    g += w[n-1] * (1.0 - 2.0 * h)
    
    for d in range(1, n):
        K = n - d
        wp = w[n-1-d]
        wm = w[n-1+d]
        g[:K] += wp * (1.0 - h[d:]) - wm * h[d:]
        g[d:] += wm * (1.0 - h[:K]) - wp * h[:K]
    
    g -= g.mean()
    return g

def smooth_power_phase(h, q_values, dx, target_sum, time_budget, t0, min_time=5.0):
    """Run smooth-power gradient descent with multiple q values."""
    n = len(h)
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    cur = h.copy()
    
    for q in q_values:
        if time_budget - (time.time() - t0) < min_time:
            break
        
        step_size = 0.02
        n_no_improve = 0
        max_no_improve = 30
        
        while n_no_improve < max_no_improve:
            if time_budget - (time.time() - t0) < min_time:
                break
            
            corr = np.correlate(cur, 1.0 - cur, mode='full')
            grad = smooth_power_gradient(cur, q, corr)
            grad_norm = np.linalg.norm(grad)
            
            if grad_norm < 1e-12:
                break
            
            # Try step
            improved = False
            for attempt in range(20):
                ss = step_size / (attempt + 1) * (1 + 0.5 * (attempt % 4))
                candidate = cur - ss * grad / grad_norm * np.linalg.norm(cur - 0.5)
                candidate = project_feasible(candidate, target_sum)
                c5_new = c5_of_h(candidate, dx)

                cur = project_feasible(cur, target_sum)
                best_c5 = c5_of_h(cur, dx)
                best_h = cur.copy()
    
    return best_h, best_c5

def project_feasible(x, target_sum):
    x = np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0)
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def c5_of_h(h, dx):
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def smooth_power_gradient(h, q, corr=None):
    n = len(h)
    if corr is None:
        corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = max(float(corr.max()), 1e-30)
    a = np.maximum(corr / max_c, 1e-30)
    log_w = (q - 1) * np.log(a)
    max_log = log_w.max()
    w = np.exp(log_w - max_log)
    
    g = np.zeros(n)
    g += w[n-1] * (1.0 - 2.0 * h)
    for d in range(1, n):
        K = n - d
        wp = w[n-1-d]
        wm = w[n-1+d]
        g[:K] += wp * (1.0 - h[d:]) - wm * h[d:]
        g[d:] += wm * (1.0 - h[:K]) - wp * h[:K]
    g -= g.mean()
    return g
