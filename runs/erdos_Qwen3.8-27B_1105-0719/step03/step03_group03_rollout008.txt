    """Project onto {0≤h≤1, sum(h)=n/2}"""
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        # Need to add mass; add to entries with most room
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            add = min(remaining, capacity[order[i]])
            h[order[i]] += add
            remaining -= add
            if remaining <= 1e-14:
                break
    else:
        # Need to remove mass; remove from entries with most value
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            rem = min(remaining, h[order[i]])
            h[order[i]] -= rem
            remaining -= rem

        else:
