def lp_minimax_direction(h, n, dx, D=0.5):
    """Solve LP to find direction minimizing worst-case linearized increase."""
    M = 2 * n - 1
    P = exact_profile(h)
    G = build_gradient_matrix(h, n, dx)
    
    # Variables: d (length n), t (scalar)
    # Minimize: t
    # Subject to:
    #   G_m @ d - t <= -P_m  for all m  (i.e., G_m @ d + P_m - t <= 0)
    #   sum(d) = 0
    #   -D <= d_i <= D
    
    # linprog minimizes c @ x subject to A_ub @ x <= b_ub, A_eq @ x = b_eq
    # x = [d_0, ..., d_{n-1}, t]
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize t
    
    # Inequality: G_m @ d - t <= -P_m  =>  G_m @ d + 0*t - t <= -P_m
    # i.e., [G_m, -1] @ [d, t] <= -P_m
    A_ub = np.zeros((M, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -P
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    bounds = [(-D, D)] * n + [(-4.0, 4.0)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x is not None:
            d = res.x[:n]
            t = res.x[n]
            return d, t
    except Exception:
        pass
    return None, None

    # Coarse search
    alphas = np.linspace(-safe, safe, n_coarse)
    best_a = 0.0
    best_c = exact_score(h)
    best_h = h.copy()
    
    scores = np.zeros(n_coarse)
    for i, a in enumerate(alphas):
        h_try = repair_feasible(h + a * d)
        scores[i] = exact_score(h_try)
        if scores[i] < best_c - 1e-14:
            best_c = scores[i]
            best_a = a
            best_h = h_try
    
    # Fine search around best
    if best_a != 0:
        idx = int(np.argmin(scores))
        lo = alphas[max(0, idx-1)]
        hi = alphas[min(n_coarse-1, idx+1)]
        alphas_fine = np.linspace(lo, hi, n_fine)
        for a in alphas_fine:
            h_try = repair_feasible(h + a * d)
            s = exact_score(h_try)
            if s < best_c - 1e-14:
                best_c = s
                best_a = a
                best_h = h_try
    
    return best_h, best_c

def lp_descent(h, max_iters=300, D=0.5, active_margin=0.0):
    """LP-based minimax descent."""
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    no_improve = 0
    
    for it in range(max_iters):
        d, t_pred = lp_minimax_direction(h, D=D, active_margin=active_margin)
        if d is None:
            break
        if t_pred >= best_c - 1e-8:
            # LP predicts no improvement; try active set focus
            if active_margin == 0:
                d, t_pred = lp_minimax_direction(h, D=D, active_margin=0.01)
                if d is None or t_pred >= best_c - 1e-8:
                    no_improve += 1
                    if no_improve > 10:
                        break
                    continue
        h_new, c_new = line_search(h, d, n_coarse=30, n_fine=15)
        if c_new < best_c - 1e-13:
            h = h_new
            best_c = c_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 30:
                break
    
    return h, best_c

def slsqp_solve(h0, maxiter=200):
    n = len(h0)
    dx = 2.0 / n
    M = 2 * n - 1
    c0 = exact_score(h0)
    
    def obj(x):
        return x[-1]
    
    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    
    def eq_fun(x):
        return x[:-1].sum() - n / 2.0
    
    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g
    
    def ineq_fun(x):
        h = np.clip(x[:-1], 0, 1)
        t = x[-1]
        profile = np.correlate(h, 1 - h, mode="full") * dx
        return t - profile
    
    def ineq_jac(x):
        h = np.clip(x[:-1], 0, 1)
        G = build_gradient_matrix(h, n, dx)
        J = np.zeros((M, n + 1))
        J[:, :-1] = -G
        J[:, -1] = 1.0
        return J
    
    x0 = np.concatenate([h0, [c0 + 1e-3]])
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 0.5)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.x is not None and np.isfinite(res.x).all():
            h_new = repair_feasible(res.x[:-1])
            c_new = exact_score(h_new)
            if c_new <= c0 + 1e-9:
                return h_new, c_new
    except Exception:
        pass
    return None, None

def lse_descent(h, beta=80.0, max_iters=200, alpha_init=0.004):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best_c = exact_score(h)
    alpha = alpha_init
    stuck = 0
    for it in range(max_iters):
        profile = exact_profile(h)
        scaled = beta * profile
        max_s = np.max(scaled)
        shifted = scaled - max_s
        weights = np.exp(shifted)
        wsum = weights.sum()
        if wsum < 1e-300:
            break
        weights /= wsum
        G = build_gradient_matrix(h, n, dx)
        grad = G.T @ weights
        grad -= grad.mean()
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad /= gnorm
        improved = False
        for a in [alpha, alpha*0.5, alpha*0.25, alpha*0.1]:
            h_try = repair_feasible(h - a * grad)
            s = exact_score(h_try)
            if s < best_c - 1e-14:
                h = h_try
                best_c = s
                improved = True
                alpha = min(alpha * 1.1, 0.02)
                stuck = 0
                break
            elif s < float(np.max(exact_profile(h))) - 1e-14:
                h = h_try
                improved = True
                break
        if not improved:
            alpha *= 0.5
            stuck += 1
            if stuck > 30 or alpha < 1e-9:
                break
    return h, best_c

def active_set_descent(h, max_iters=300, n_active=12, alpha_init=0.005):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best_c = exact_score(h)
    alpha = alpha_init
    stuck = 0
    for it in range(max_iters):
        profile = exact_profile(h)
        c = float(np.max(profile))
        if c + 1e-14 < best_c:
            best_c = c
            stuck = 0
            alpha = min(alpha * 1.08, 0.025)
        margin = max(0.002, 4.0 * (c - float(np.median(profile))))
        active = np.where(profile >= c - margin)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(profile))])
        if len(active) > n_active:
            top_idx = np.argsort(profile[active])[-n_active:]
            active = active[top_idx]
        G = build_gradient_matrix(h, n, dx)
        G_a = G[active, :]
        w = np.ones(len(active)) / len(active)
        grad = G_a.T @ w
        grad -= grad.mean()
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad /= gnorm
        improved = False
        for a in [alpha, alpha*0.5, alpha*0.25, alpha*0.1, alpha*0.05]:
            h_try = repair_feasible(h - a * grad)
            s = exact_score(h_try)
            if s < best_c - 1e-14:
                h = h_try
                best_c = s
                improved = True
                break
            elif s < c - 1e-14:
                h = h_try
                improved = True
                break
        if not improved:
            alpha *= 0.55
            stuck += 1
            if stuck > 40 or alpha < 1e-9:
                break
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()
    rng = np.random.RandomState(seed)
    
    def time_left():
        return budget_s + t_start - time.perf_counter()
    
    # Get parent
    parent = None
    try:
        parent = np.array(initial_h_values, dtype=np.float64)
    except NameError:
        pass
    except Exception:
        pass
    
    if parent is None or len(parent) < 4:
        parent = np.full(70, 0.5)
    
    parent = np.clip(parent, 0.0, 1.0)
    parent = repair_feasible(parent)
    c_parent = exact_score(parent)
    
    best_h = parent.copy()
    best_c = c_parent
    
    # Try symmetries
    h_sym = repair_feasible(symmetrize(parent))
    c_sym = exact_score(h_sym)
    if c_sym < best_c - 1e-12:
        best_h = h_sym.copy()
        best_c = c_sym
    
    n_parent = len(best_h)
    
    # Phase 1: LP descent from initial
    if time_left() > 200:
        h_lp, c_lp = lp_descent(best_h, max_iters=200, D=0.5, active_margin=0.0)
        if c_lp < best_c - 1e-13:
            best_h = h_lp.copy()
            best_c = c_lp
        h_ase, c_ase = active_set_descent(best_h, max_iters=150, n_active=10)
        if c_ase < best_c - 1e-13:
            best_h = h_ase.copy()
            best_c = c_ase
        h_lse, c_lse = lse_descent(best_h, beta=100, max_iters=100)
        if c_lse < best_c - 1e-13:
            best_h = h_lse.copy()
            best_c = c_lse
    
    # Phase 2: Resolution exploration
    if time_left() > 200:
        dims = set()
        for factor in [0.85, 0.9, 0.95, 1.0, 1.05, 1.1, 1.15, 1.2]:
            d = int(round(n_parent * factor / 2)) * 2
            d = max(30, min(d, 120))
            dims.add(d)
        dims.discard(n_parent)
        dims_list = sorted(dims, key=lambda x: abs(x - n_parent))
