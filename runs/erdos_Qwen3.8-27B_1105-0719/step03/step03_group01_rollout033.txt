    t_lo = max(-h[I], h[J] - 1.0)
    t_hi = min(1.0 - h[I], h[J])

        return 0.0
    
    # Sample t values
    n_samples = 100
    t_vals = np.linspace(t_lo, t_hi, n_samples)
    best_t = 0.0
    best_obj = c5_of_h(h, dx)  # current value

    h_copy = h.copy()
    for t in t_vals:
        h_copy[I] = h[I] + t
        h_copy[J] = h[J] - t
        obj = np.max(np.correlate(h_copy, 1.0 - h_copy, mode='full'))
        if obj < best_obj:

    t_range = (t_hi - t_lo) / n_samples

```python
def pair_line_search_fast(h, corr, I, J, n):
    """Fast 1D line search using precomputed correlation and pair coefficients."""
    t_lo = max(-h[I], h[J] - 1.0) + 1e-14
    t_hi = min(1.0 - h[I], h[J]) - 1e-14
    if t_hi - t_lo < 1e-10:
        return 0.0
    
    # Compute a_d and c_d for all lags
    # a_d = change in f_d per unit t (linear coefficient)
    # c_d = coefficient of -t² (1 if I and J interact at lag d, else 0)
    
    a_coeff = np.zeros(2*n - 1)
    c_coeff = np.zeros(2*n - 1)
    
    for d in range(-(n-1), n):
        idx = d + (n-1)  # index in corr array
        a_d = 0.0
        c_d = 0.0
        
        # I as first index: valid if 0<=I+d<n
        if 0 <= I+d < n:
            a_d += (1.0 - h[I+d])
            if I+d == J:
                c_d += 1.0  # cross term: -h[I]*h[J] → -(h[I]+t)(h[J]-t)
        
        # I as second index: valid if 0<=I-d<n
        if 0 <= I-d < n:
            a_d += -h[I-d]
            if I-d == J:
                c_d += 1.0
        
        # J as first index: valid if 0<=J+d<n
        if 0 <= J+d < n:
            a_d += -(1.0 - h[J+d])
            if J+d == I:
                c_d += 1.0
        
        # J as second index: valid if 0<=J-d<n
        if 0 <= J-d < n:
            a_d += h[J-d]
            if J-d == I:
                c_d += 1.0
        
        a_coeff[idx] = a_d
        c_coeff[idx] = c_d
    
    # Now minimize max_d (corr[d] + a_coeff[d]*t - c_coeff[d]*t²) over [t_lo, t_hi]
    # Sample on a grid
    n_samples = 80
    t_vals = np.linspace(t_lo, t_hi, n_samples)
    
    best_t = 0.0
    best_obj = np.inf
    
    for t in t_vals:
        vals = corr + a_coeff * t - c_coeff * t * t
        obj = vals.max()
        if obj < best_obj:
            best_obj = obj
            best_t = t
    
    # Refine
    step = (t_hi - t_lo) / n_samples
    fl = max(t_lo, best_t - 2*step)
    fh = min(t_hi, best_t + 2*step)
    t_fine = np.linspace(fl, fh, 60)
    
    for t in t_fine:
        vals = corr + a_coeff * t - c_coeff * t * t
        obj = vals.max()
        if obj < best_obj:
            best_obj = obj
            best_t = t
    
    return best_t
```
