def repair_feasible(vec):

    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()

        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)

            if remaining <= 1e-13:
                break
            add = min(remaining, capacity[idx])
            h[idx] += add
            remaining -= add
    else:

        remaining = -diff

            if remaining <= 1e-13:

            remaining -= rem

            d = m - (n - 1)
            row = np.zeros(n)
            for i in range(n):
                g_val = 0.0
                j_idx = i - d
                if 0 <= j_idx < n:
                    g_val += (1.0 - h_local[j_idx])
                if 0 <= i + d < n:
                    g_val += (-h_local[i + d])
                row[i] = -dx * g_val
            J[idx, :-1] = row
            J[idx, -1] = 1.0
        return J

    x0 = np.concatenate([h0.copy(), [c0 + 1e-3]])
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 1.0)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]

    try:
        if t_start is not None and time.perf_counter() - t_start > budget_s - margin:
            return None
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': tol})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = repair_feasible(res.x[:-1])
            c_new = exact_score(h_new)
            if c_new <= c0 + 1e-10:
                return h_new
    except Exception:
        pass
    return None


def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    h_new = np.interp(xt, xs, h)
    return repair_feasible(np.clip(h_new, 0.0, 1.0))


def soft_max_gradient(h, beta, n_active=10):
    n = len(h)
    dx = 2.0 / n
    P = np.correlate(h, 1 - h, mode="full") * dx
    Pmax = P.max()
    exp_arg = beta * (P - Pmax)
    weights = np.exp(exp_arg)
    S = weights.sum()
    if S < 1e-300:
        return np.zeros(n)
    weights /= S
    top_idx = np.argsort(weights)[-n_active:]
    grad = np.zeros(n)
    for m in top_idx:
        w = weights[m]
        if w < 1e-10:
            continue
        d = m - (n - 1)
        for i in range(n):
            g_val = 0.0
            j_idx = i - d
            if 0 <= j_idx < n:
                g_val += (1.0 - h[j_idx])
            if 0 <= i + d < n:
                g_val += (-h[i + d])
            grad[i] += w * dx * g_val
    return grad


def projected_soft_max_descent(h, max_iters=100, t_start=None, budget_s=1100, margin=30):
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    vel = np.zeros(n)
    betas = [50, 100, 200, 500]

    for beta in betas:
        alpha = 0.02
        for it in range(max_iters // len(betas)):
            if t_start is not None and time.perf_counter() - t_start > budget_s - margin:
                return h, best_c
            grad = soft_max_gradient(h, beta, n_active=8)
            grad -= grad.mean()
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break
            grad /= gnorm
            vel = 0.4 * vel + grad
            vnorm = np.linalg.norm(vel)
            if vnorm > 0:
                vel /= vnorm
            improved = False
            for a in [alpha, alpha * 0.5, alpha * 0.25, alpha * 0.1]:
                h_try = repair_feasible(h - a * vel)
                s = exact_score(h_try)
                if s < best_c - 1e-13:
                    h = h_try
                    best_c = s
                    improved = True
                    break
            if improved:
                alpha = min(alpha * 1.05, 0.08)
            else:
                alpha *= 0.5
                if alpha < 1e-7:
                    break
    return h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()
    rng = np.random.RandomState(seed)

    def time_ok(margin=20):
        return time.perf_counter() - t_start < budget_s - margin

    parent = None
    try:
        parent = np.array(initial_h_values, dtype=np.float64)
    except (NameError, Exception):
        pass

    if parent is None or len(parent) < 4:
        n_init = 70
        h_init = np.zeros(n_init)
        for i in range(n_init // 2):
            pos = int(i * (n_init - 1) / max(1, n_init // 2 - 1))
            h_init[pos] = 1.0
        parent = repair_feasible(h_init)

    parent = np.clip(parent, 0.0, 1.0)
    parent = repair_feasible(parent)

    best_h = parent.copy()
    best_c = exact_score(parent)

    # Phase 1: Coarse pairwise (fast convergence)
    if time_ok(80):
        best_h, best_c = pairwise_sweep(best_h, n_sweeps=10, t_start=t_start,
                                        budget_s=budget_s, margin=60, xatol=1e-7)

    # Phase 2: Fine pairwise
    if time_ok(120):
        best_h, best_c = pairwise_sweep(best_h, n_sweeps=25, t_start=t_start,
                                        budget_s=budget_s, margin=60, xatol=1e-9)

    # Phase 3: SLSQP polish
    if time_ok(80):
        sol = slsqp_optimize(best_h, maxiter=300, active_k=15, tol=1e-12,
                             t_start=t_start, budget_s=budget_s, margin=40)
        if sol is not None:
            c = exact_score(sol)
            if c < best_c - 1e-12:
                best_h = sol
                best_c = c

    # Phase 4: More fine pairwise after SLSQP
    if time_ok(100):
        best_h, best_c = pairwise_sweep(best_h, n_sweeps=20, t_start=t_start,
                                        budget_s=budget_s, margin=60, xatol=1e-10)

    # Phase 5: Soft-max gradient descent for global moves
    if time_ok(60):
        h_md, c_md = projected_soft_max_descent(best_h, max_iters=80,
                                                t_start=t_start, budget_s=budget_s, margin=40)
        if c_md < best_c - 1e-12:
            best_h = h_md
            best_c = c_md
            # Polish with pairwise
            if time_ok(40):
                best_h, best_c = pairwise_sweep(best_h, n_sweeps=10, t_start=t_start,
                                                budget_s=budget_s, margin=40, xatol=1e-9)

    # Phase 6: Dimension exploration
    if time_ok(120):
        n_cur = len(best_h)
        dims_to_try = []
        for delta in [2, -2, 4, -4, 6]:
            new_n = n_cur + delta
            if 20 <= new_n <= 150:
                dims_to_try.append(new_n)

        for new_n in dims_to_try:
            if not time_ok(30):
                break
            h_new = interpolate_to_n(best_h, new_n)
            h_new, c_new = pairwise_sweep(h_new, n_sweeps=12, t_start=t_start,
                                           budget_s=budget_s, margin=50, xatol=1e-9)
            if c_new < best_c - 1e-10:
                best_h = h_new
                best_c = c_new
                if time_ok(30):
                    sol = slsqp_optimize(best_h, maxiter=200, active_k=12, tol=1e-12,
                                         t_start=t_start, budget_s=budget_s, margin=40)
                    if sol is not None:
                        c2 = exact_score(sol)
                        if c2 < best_c - 1e-12:
                            best_h = sol
                            best_c = c2
                            h_new2, c_new2 = pairwise_sweep(best_h, n_sweeps=10,
                                                            t_start=t_start, budget_s=budget_s,
                                                            margin=40, xatol=1e-9)
                            if c_new2 < best_c - 1e-13:
                                best_h = h_new2
                                best_c = c_new2

    # Phase 7: Multi-start perturbations
    if time_ok(80):
        n = len(best_h)
        for trial in range(4):
            if not time_ok(20):
                break
            eps = 0.005 * (trial + 1)
            z = rng.randn(n)
            z -= z.mean()
            h_pert = repair_feasible(best_h + eps * z)
            h_pert, c_pert = pairwise_sweep(h_pert, n_sweeps=12, t_start=t_start,
                                             budget_s=budget_s, margin=40, xatol=1e-9)
            if c_pert < best_c - 5e-5:
                best_h = h_pert
                best_c = c_pert

    # Phase 8: Final polish
    if time_ok(40):
        best_h, best_c = pairwise_sweep(best_h, n_sweeps=15, t_start=t_start,
                                        budget_s=budget_s, margin=25, xatol=1e-10)
    if time_ok(30):
        sol = slsqp_optimize(best_h, maxiter=400, active_k=20, tol=1e-13,
                             t_start=t_start, budget_s=budget_s, margin=15)
        if sol is not None:
            c = exact_score(sol)
            if c < best_c - 1e-13:
                best_h = sol
                best_c = c
    if time_ok(20):
        best_h, best_c = pairwise_sweep(best_h, n_sweeps=8, t_start=t_start,
                                        budget_s=budget_s, margin=10, xatol=1e-10)

    # Final validation
    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)

    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            final_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 1:
            if np.isfinite(eval_result[0]):
                final_c = float(eval_result[0])
    except Exception:
        pass

    return best_h, final_c, len(best_h)
