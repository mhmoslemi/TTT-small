import numpy as np
from scipy.optimize import minimize, linprog
import time

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(-(1.0 - h))
        remaining = diff
        for i in range(n):
            idx = int(order[i])
            cap = 1.0 - h[idx]
            if cap <= 1e-15:
                continue
            add = min(remaining, cap)
            h[idx] += add
            remaining -= add
            if remaining <= 1e-14:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = int(order[i])
            if h[idx] <= 1e-15:
                continue
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-14:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] = float(np.clip(h[idx] + diff2, 0.0, 1.0))
    return h

def grad_row(h, m, n):
    dx = 2.0 / n
    d = m - (n - 1)
    if d == 0:
        return dx * (1.0 - 2.0 * h)
    elif d > 0:
        g = np.zeros(n)
        g[d:] += dx * (1.0 - h[:-d])
        g[:-d] -= dx * h[d:]
        return g
    else:
        p = -d
        g = np.zeros(n)
        g[:-p] += dx * (1.0 - h[p:])
        g[p:] -= dx * h[:-p]
        return g

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def exact_score(h):
    return float(np.max(compute_profile(h)))

def lse_descent(h, beta_schedule, max_iters_per_beta=50, tol=1e-10):
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    for beta in beta_schedule:
        for it in range(max_iters_per_beta):
            profile = compute_profile(h)
            c = float(profile.max())
            if c <= best_c - 1e-12:
                best_c = c
            # Compute LSE gradient
            z = beta * (profile - c)
            z = z - z.max()  # for numerical stability (z.max() = 0)
            w = np.exp(z)
            w_sum = w.sum()
            if w_sum < 1e-30:
                break
            w = w / w_sum
            # Gradient is weighted sum of grad rows
            g = np.zeros(n)
            for m in range(len(profile)):
                if w[m] > 1e-10:
                    g += w[m] * grad_row(h, m, n)
            g -= g.mean()  # ensure sum(g)=0 direction
            gnorm = np.linalg.norm(g)
            if gnorm < 1e-14:
                break
            d = -g / gnorm
            # Line search
            pos_mask = d > 1e-12
            neg_mask = d < -1e-12
            alpha_max = 1.0
            if pos_mask.any():
                alpha_max = min(alpha_max, float(((1.0 - h[pos_mask]) / d[pos_mask]).min()))
            if neg_mask.any():
                alpha_max = min(alpha_max, float((h[neg_mask] / (-d[neg_mask])).min()))
            alpha_max = max(alpha_max, 1e-10)
            alpha_max = min(alpha_max, 0.3)
            # Golden section search
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            a, b = 0.0, alpha_max
            best_alpha = 0.0
            best_val = c
            for _ in range(15):
                c1 = b - gr * (b - a)
                c2 = a + gr * (b - a)
                h1 = repair_feasible(h + c1 * d)
                h2 = repair_feasible(h + c2 * d)
                f1 = exact_score(h1)
                f2 = exact_score(h2)
                if f1 < best_val:
                    best_val = f1
                    best_alpha = c1
                if f2 < best_val:
                    best_val = f2
                    best_alpha = c2
                if f1 < f2:
                    b = c2
                else:
                    a = c1
                if b - a < 1e-12:
                    break
            if best_val >= c - 1e-13:
                break
            h = repair_feasible(h + best_alpha * d)
            if best_val < best_c:
                best_c = best_val
    return h, best_c

import numpy as np
from scipy.optimize import linprog, minimize
import time

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(-(1.0 - h))
        rem = diff
        for i in range(n):
            idx = int(order[i])
            cap = 1.0 - h[idx]
            if cap <= 1e-15:
                continue
            add = min(rem, cap)
            h[idx] += add
            rem -= add
            if rem <= 1e-14:
                break
    else:
        order = np.argsort(h)
        rem = -diff
        for i in range(n):
            idx = int(order[i])
            if h[idx] <= 1e-15:
                continue
            r = min(rem, h[idx])
            h[idx] -= r
            rem -= r
            if rem <= 1e-14:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] = float(np.clip(h[idx] + diff2, 0.0, 1.0))
    return h

def grad_row(h, m, n):
    dx = 2.0 / n
    d = m - (n - 1)
    if d == 0:
        return dx * (1.0 - 2.0 * h)
    elif d > 0:
        g = np.zeros(n)
        g[d:] += dx * (1.0 - h[:n - d])
        g[:n - d] -= dx * h[d:]
        return g
    else:
        p = -d
        g = np.zeros(n)
        g[:n - p] += dx * (1.0 - h[p:])
        g[p:] -= dx * h[:n - p]
        return g

    n = len(h)
    M = 2 * n - 1
    G = np.zeros((M, n))
    for m in range(M):
        G[m] = grad_row(h, m, n)
    return G

    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

    return float(np.max(compute_profile(h)))

def lse_descent(h, betas, iters=30, alpha_max=0.3):
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    inv_golden = (np.sqrt(5.0) + 1.0) / 2.0
    for beta in betas:
        for it in range(iters):
            prof = compute_profile(h)
            c = float(prof.max())
            z = beta * (prof - c)
            z -= z.max()
            w = np.exp(z)
            ws = w.sum()
            if ws < 1e-30:
                break
            w /= ws
            G = build_grad_matrix(h)
            g = G.T @ w
            g -= g.mean()
            gn = np.linalg.norm(g)
            if gn < 1e-14:
                break
            d = -g / gn
            pm = d > 1e-12
            nm = d < -1e-12
            amax = alpha_max
            if pm.any():
                amax = min(amax, float(((1.0 - h[pm]) / d[pm]).min()))
            if nm.any():
                amax = min(amax, float((h[nm] / (-d[nm])).min()))
            amax = max(amax, 1e-12)
            # Golden search
            a, b = 0.0, amax
            ba, bv = 0.0, c
            for _ in range(14):

                c2 = a + (b - a) / inv_golden

                h2 = repair_feasible(h + c2 * d)
                f1 = exact_score(h1)

                if f2 < bv:

                else:
                    a = c1

def interp_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    eo = np.linspace(0.0, 2.0, n + 1)
    en = np.linspace(0.0, 2.0, new_n + 1)
    h_new = np.zeros(new_n)
    for k in range(new_n):
        a, b = en[k], en[k + 1]
        total = 0.0
        for j in range(n):
            lo = max(a, eo[j])
            hi = min(b, eo[j + 1])
            if hi > lo:
                total += h[j] * (hi - lo)
        h_new[k] = total / (b - a)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

    K = min(M, 30)
