import numpy as np
from scipy.optimize import minimize, linprog
from scipy.special import logsumexp
import time


def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)


def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def c5_of_h(h, dx):
    return compute_corrs(h).max() * dx


def grad_corr_k(h, k, n):
    offset = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -offset)
    i_hi = min(n - 1, n - 1 - offset)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx + offset] += (1.0 - h[idx])
    grad[idx] += (-h[idx + offset])
    return grad


def smooth_max_grad(h, dx, beta):
    n = len(h)
    corrs = compute_corrs(h)
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones_like(corrs) / len(corrs)
    else:
        w /= w_sum
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] > 1e-13:
            grad += w[k] * grad_corr_k(h, k, n)
    return grad * dx


def slsqp_direct(h0, dx, target_sum, beta, maxiter):
    n = len(h0)

    def obj(x):
        corrs = compute_corrs(x)
        return logsumexp(beta * corrs) / beta * dx

    def grad_obj(x):
        return smooth_max_grad(x, dx, beta)

    def c_fun(x):
        return x.sum() - target_sum

    def c_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': c_fun, 'jac': c_jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0, jac=grad_obj, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        h_out = project_feasible(res.x, target_sum)
    except Exception:
        h_out = project_feasible(h0, target_sum)
    return h_out, c5_of_h(h_out, dx)

def exact_line_search(h, d, dx, target_sum, n_samples=80, n_refine=35):
    n = len(h)
    alpha_min = 0.0
    alpha_max = 1e10

    for i in range(n):
        if d[i] > 1e-15:
            amax = (1.0 - h[i]) / d[i]
            if amax < alpha_max:
                alpha_max = amax
        elif d[i] < -1e-15:
            amin = -h[i] / d[i]
            if amin > alpha_min:
                alpha_min = amin

    if alpha_max <= alpha_min + 1e-14:
        return h.copy(), c5_of_h(h, dx)

    alpha_min = max(alpha_min, 0.0)
    alpha_max = min(alpha_max, 2.0)

    if alpha_max - alpha_min < 1e-12:
        return h.copy(), c5_of_h(h, dx)

    alphas = np.linspace(alpha_min + 1e-12, alpha_max - 1e-12, n_samples)
    best_c5 = c5_of_h(h, dx)
    best_alpha = 0.0

    for a in alphas:
        h_try = project_feasible(h + a * d, target_sum)
        c5_try = c5_of_h(h_try, dx)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_alpha = a

    if best_alpha < 1e-12:
        return h.copy(), best_c5

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    span = (alpha_max - alpha_min) / max(n_samples - 1, 1) * 2.0
    a_lo = max(alpha_min, best_alpha - span)
    a_hi = min(alpha_max, best_alpha + span)

    if a_hi - a_lo < 1e-14:
        return project_feasible(h + best_alpha * d, target_sum), best_c5

    c1 = a_hi - gr * (a_hi - a_lo)
    c2 = a_lo + gr * (a_hi - a_lo)
    f1 = c5_of_h(project_feasible(h + c1 * d, target_sum), dx)
    f2 = c5_of_h(project_feasible(h + c2 * d, target_sum), dx)

    for _ in range(n_refine):
        if a_hi - a_lo < 1e-11:
            break
        if f1 < f2:
            a_hi = c2
            c2 = c1
            f2 = f1
            c1 = a_hi - gr * (a_hi - a_lo)
            f1 = c5_of_h(project_feasible(h + c1 * d, target_sum), dx)
        else:
            a_lo = c1
            c1 = c2
            f1 = f2
            c2 = a_lo + gr * (a_hi - a_lo)
            f2 = c5_of_h(project_feasible(h + c2 * d, target_sum), dx)

    a_opt = (a_lo + a_hi) / 2.0
    h_opt = project_feasible(h + a_opt * d, target_sum)
    c5_opt = c5_of_h(h_opt, dx)
    return h_opt, c5_opt

def gradient_line_search(h0, dx, target_sum, max_iters=200, n_ls_samples=60):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    stall = 0

    for it in range(max_iters):
        c5_val = c5_of_h(h, dx)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 30:
                break

        beta = 200.0 + 100.0 * it
        grad = smooth_max_grad(h, dx, beta)

        # Project gradient to feasible direction (sum=0, box constraints)
        d = -grad
        d -= d.mean()  # ensure sum(d) = 0

        # Scale d to fit in box
        dnorm = np.max(np.abs(d))
        if dnorm < 1e-14:
            break
        scale = 0.1 / dnorm
        d *= scale

        h_new, c5_new = exact_line_search(h, d, dx, target_sum, n_ls_samples, 35)

        if c5_new < c5_val - 1e-14:
            h = h_new
        else:
            # Try smaller step
            d_small = d * 0.3
            h_new2, c5_new2 = exact_line_search(h, d_small, dx, target_sum, 40, 25)
            if c5_new2 < c5_val - 1e-14:
                h = h_new2
            else:
                break

    return best_h, best_c5


def pair_transfer_polish(h0, dx, target_sum, n_top=25, n_iter=3):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)

    for iteration in range(n_iter):
        corrs = compute_corrs(h)
        grad = smooth_max_grad(h, dx, 600.0)

        dec_idx = np.argsort(grad)[-n_top:]
        inc_idx = np.argsort(grad)[:n_top]

        improved = False
        for i in dec_idx:
            for j in inc_idx:
                if i == j:
                    continue
                eps_hi = min(h[i], 1.0 - h[j])
                if eps_hi < 1e-12:
                    continue

                d_vec = np.zeros(n)
                for k in range(2 * n - 1):
                    gk = grad_corr_k(h, k, n)
                    d_vec[k] = gk[j] - gk[i]

                gr = (np.sqrt(5.0) - 1.0) / 2.0
                a, b = 0.0, eps_hi
                c_pt = b - gr * (b - a)
                d_pt = a + gr * (b - a)
                fc = (corrs + c_pt * d_vec).max()
                fd = (corrs + d_pt * d_vec).max()
                for _ in range(45):
                    if b - a < 1e-13:
                        break
                    if fc < fd:
                        b = d_pt
                        d_pt = c_pt
                        fd = fc
                        c_pt = b - gr * (b - a)
                        fc = (corrs + c_pt * d_vec).max()
                    else:
                        a = c_pt
                        c_pt = d_pt
                        fc = fd
                        d_pt = a + gr * (b - a)
                        fd = (corrs + d_pt * d_vec).max()

                eps_opt = (a + b) / 2.0
                rough_c5 = (corrs + eps_opt * d_vec).max() * dx
                if rough_c5 < best_c5 - 1e-13:
                    new_h = h.copy()
                    new_h[i] -= eps_opt
                    new_h[j] += eps_opt
                    new_h = np.clip(new_h, 0.0, 1.0)
                    new_h = project_feasible(new_h, target_sum)
                    new_c5 = c5_of_h(new_h, dx)
                    if new_c5 < best_c5 - 1e-14:
                        h = new_h
                        best_c5 = new_c5
                        best_h = h.copy()
                        corrs = compute_corrs(h)
                        improved = True

        if not improved:
            break

    return best_h, best_c5


def balanced_active_lp(h0, dx, target_sum, max_iters=100):
    """LP that tries to equalize the top correlations by constraining each to be <= current max - epsilon."""
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)

    for it in range(max_iters):
        corrs = compute_corrs(h)
        max_c = corrs.max()
        c5_val = max_c * dx
        second_max = np.sort(corrs)[-2]

        # Target: reduce max to be closer to second_max
        target_gap = max_c - second_max
        if target_gap < 1e-6:
            break

        # Constrain the max correlation to be reduced
        k_max = np.argmax(corrs)
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        row = np.zeros(n + 1)
        row[:n] = dx * grad_corr_k(h, k_max, n)
        row[n] = -1.0
        A_ub = np.array([row])
        b_ub = np.array([-(max_c - target_gap * 0.3) * dx])

        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)
        bounds = [(max(-h[i], -0.08), min(1.0 - h[i], 0.08)) for i in range(n)]
        bounds.append((0.0, None))

        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
        except Exception:
            break
        if res.status != 0:
            break

        delta = res.x[:n]
        if np.max(np.abs(delta)) < 1e-14:
            break

        h_new = project_feasible(h + delta, target_sum)
        new_c5 = c5_of_h(h_new, dx)

        if new_c5 < c5_val - 1e-14:
            h = h_new
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()

        if c5_f3 < best_c5:
            best_c5 = c5_f3
            best_h = h_f3

    # Final validation
    if best_h is None:
        best_h = h0
        best_c5 = c5_of_h(h0, 2.0 / len(h0))
    else:
        n_final = len(best_h)
        best_h = project_feasible(best_h, n_final / 2.0)
        best_c5 = c5_of_h(best_h, 2.0 / n_final)

    return best_h, best_c5, len(best_h)
