        if c < best_c - 1e-14:
            best_c = c
            best_h = h.copy()
            no_improve = 0
            radius = min(radius * 1.15, 0.12)
        else:
            no_improve += 1

        if no_improve > 50:
            break

        active, T = get_active_set(profile, thresh, max_count)

        if it % 3 == 0:
            direction = qp_step(h, active, radius, gamma)
        else:
            direction = None

        if direction is None:
            direction = lp_step_fallback(h, active, radius)

        if direction is not None:
            h_new, s_new = line_search_improve(h, direction, c)
            if s_new < best_c - 1e-14:
                h = h_new
                continue
            h_new2, s_new2 = line_search_improve(h, -direction, c)
            if s_new2 < best_c - 1e-14:
                h = h_new2
                continue

        # Gradient-based fallback
        G = build_gradient(h)
        G_a = G[active, :]
        p_a = profile[active]
        weights = np.maximum(p_a - (T - 0.1 * (T - np.min(p_a))) + 1e-6, 0.01)
        weights /= weights.sum()
        grad = G_a.T @ weights
        grad -= grad.mean()
        norm = np.linalg.norm(grad)
        if norm > 1e-15:
            grad_dir = -grad / norm
            h_new, s_new = line_search_improve(h, grad_dir, c)
            if s_new < best_c - 1e-14:
                h = h_new
                no_improve = 0
                continue

        # Adapt parameters
        if no_improve % 10 == 5:
            radius *= 0.7
            thresh *= 1.3
            max_count = min(max_count + 5, 2 * n - 1)
        if no_improve % 15 == 10:
            gamma *= 1.5
        if no_improve % 20 == 15:
            gamma /= 2.0

        if radius < 0.002:
            break

    return best_h, best_c

def fine_polish_v2(h, t_start, budget_s, max_iters=200):
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    radius = 0.01
    gamma = 50.0
    thresh = 0.003
    max_count = 20
    no_improve = 0

    for it in range(max_iters):
        elapsed = time.perf_counter() - t_start
        if budget_s - elapsed < 4:
            break

        profile = compute_profile(h)
        c = float(np.max(profile))
        active, T = get_active_set(profile, thresh, max_count)
        direction = qp_step(h, active, radius, gamma)
        if direction is None:
            direction = lp_step_fallback(h, active, radius)

        if direction is not None:
            h_new, s_new = line_search_improve(h, direction, c)
            if s_new < best_c - 1e-14:
                h = h_new
                best_c = s_new
                no_improve = 0
                continue
            h_new2, s_new2 = line_search_improve(h, -direction, c)
            if s_new2 < best_c - 1e-14:
                h = h_new2
                best_c = s_new2
                no_improve = 0
                continue

        no_improve += 1
        if no_improve % 8 == 4:
            radius *= 0.7
        if no_improve > 30:
            break

    return h, best_c

def make_structure(n, seed, mode):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    x = np.linspace(0, 2, n)
    if mode == 0:
        h = 0.5 + 0.25 * np.sin(np.pi * x + rng.uniform(0, 2 * np.pi))
    elif mode == 1:
        nb = rng.randint(3, 6)
        block = n // nb
        for b in range(nb):
            h[b * block:(b + 1) * block] = rng.uniform(0.15, 0.85)
        if nb * block < n:
            h[nb * block:] = 0.5
    elif mode == 2:
        phi = (1.0 + np.sqrt(5.0)) / 2.0
        h = 0.5 + 0.3 * np.sin(2 * np.pi * (n * phi * x / 2.0) % 1.0)
    elif mode == 3:
        h = 0.5 + 0.3 * np.cos(np.pi * x) + 0.1 * np.sin(2 * np.pi * x)
    elif mode == 4:
        half = n // 2
        h[:half] = 0.5 + 0.3 * np.sin(np.pi * np.arange(half) / half)
        h[half:] = 1.0 - h[:half][::-1]
    else:
        h = 0.5 + 0.2 * rng.randn(n)
    return repair_feasible(h)

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s - (time.perf_counter() - t_start)

    # Load parent
    parent = None
    try:
        parent = np.array(initial_h_values, dtype=np.float64)
    except (NameError, Exception):
        pass

    if parent is None or len(parent) < 4 or not np.isfinite(parent).all():
        parent = np.full(70, 0.5)

    parent = np.clip(parent, 0.0, 1.0)
    parent = repair_feasible(parent)
    best_h = parent.copy()
    best_c = exact_score(parent)
    n_best = len(best_h)

    # Try basic transformations on parent
    transforms = [
        repair_feasible(symmetrize(best_h)),
        repair_feasible(best_h[::-1]),
        repair_feasible(1.0 - best_h),
        repair_feasible(0.7 * best_h + 0.3 * symmetrize(best_h)),
    ]
    for t_h in transforms:
        c = exact_score(t_h)
        if c < best_c - 1e-12:
            best_h = t_h
            best_c = c

    # Phase 1: Main local optimization at parent dimension
    if time_left() > 200:
        h_opt, c_opt = optimize_local_v2(best_h, t_start, budget_s, max_iters=250)
        if c_opt < best_c - 1e-14:
            best_h = h_opt
            best_c = c_opt
            n_best = len(best_h)

    # Phase 2: Fine polish
    if time_left() > 80:
        h_pol, c_pol = fine_polish_v2(best_h, t_start, budget_s, max_iters=150)
        if c_pol < best_c - 1e-14:
            best_h = h_pol
            best_c = c_pol
            n_best = len(best_h)

    # Phase 3: Try nearby dimensions
    n0 = len(best_h)
    dims_to_try = []
    for d in range(max(20, n0 - 10), min(200, n0 + 15), 2):
        if d != n0:
            dims_to_try.append(d)
    dims_to_try.sort(key=lambda d: abs(d - n0))

    for new_n in dims_to_try[:8]:
        if time_left() < 80:
            break
        h_interp = interpolate_to_n(best_h, new_n)
        h_opt, c_opt = optimize_local_v2(h_interp, t_start, budget_s, max_iters=150)
        if c_opt < best_c - 1e-14:
            best_h = h_opt
            best_c = c_opt
            n_best = new_n
            if time_left() > 40:
                h_pol, c_pol = fine_polish_v2(best_h, t_start, budget_s, max_iters=60)
                if c_pol < best_c - 1e-14:
                    best_h = h_pol
                    best_c = c_pol

    # Phase 4: Try structured starts at best dimension
    n_b = len(best_h)
    if time_left() > 100:
        for mode in range(5):
            if time_left() < 40:
                break
            h_struct = make_structure(n_b, seed + mode * 100, mode)
            h_blend = 0.4 * best_h + 0.6 * h_struct
            h_blend = repair_feasible(h_blend)
            c_blend = exact_score(h_blend)
            if c_blend < best_c + 0.02:
                h_opt, c_opt = optimize_local_v2(h_blend, t_start, budget_s, max_iters=120)
                if c_opt < best_c - 1e-14:
                    best_h = h_opt
                    best_c = c_opt
                    n_b = len(best_h)

    # Phase 5: Finer resolution
    if time_left() > 80:
        new_n = min(n_best * 2, 300)
        if new_n > n_best:
            h_fine = interpolate_to_n(best_h, new_n)
            h_opt, c_opt = optimize_local_v2(h_fine, t_start, budget_s, max_iters=120)
            if c_opt < best_c - 1e-14:
                best_h = h_opt
                best_c = c_opt

    # Phase 6: SLSQP epigraph refinement
    if time_left() > 50:
        n_f = len(best_h)
        dx_f = 2.0 / n_f
        M_f = 2 * n_f - 1
        c0_f = exact_score(best_h)

        def slsqp_obj(x):
            return x[-1]

        def slsqp_jac(x):
            g = np.zeros(n_f + 1)
            g[-1] = 1.0
            return g

        def slsqp_eq(x):
            return x[:-1].sum() - n_f / 2.0

        def slsqp_eq_jac(x):
            g = np.zeros(n_f + 1)
            g[:-1] = 1.0
            return g

        def slsqp_ineq(x):
            h = x[:-1]
            t = x[-1]
            prof = np.correlate(h, 1.0 - h, mode="full") * dx_f
            return t - prof

        def slsqp_ineq_jac(x):
            h = x[:-1]
            G = build_gradient(h)
            J = np.zeros((M_f, n_f + 1))
            J[:, :-1] = -G
            J[:, -1] = 1.0
            return J

        x0 = np.concatenate([best_h, [c0_f + 1e-4]])
        bounds = [(0.0, 1.0)] * n_f + [(0.0, None)]
        cons = [
            {'type': 'eq', 'fun': slsqp_eq, 'jac': slsqp_eq_jac},
            {'type': 'ineq', 'fun': slsqp_ineq, 'jac': slsqp_ineq_jac}
        ]
        try:
            res = minimize(slsqp_obj, x0, method='SLSQP', jac=slsqp_jac,
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 500, 'ftol': 1e-14})
            if res.x is not None and np.isfinite(res.x).all():
                h_sol = repair_feasible(res.x[:-1])
                c_sol = exact_score(h_sol)
                if c_sol < best_c - 1e-14:
                    best_h = h_sol
                    best_c = c_sol
        except Exception:
            pass

    # Phase 7: Final intensive polish
    if time_left() > 25:
        h_pol, c_pol = fine_polish_v2(best_h, t_start, budget_s, max_iters=100)
        if c_pol < best_c - 1e-14:
            best_h = h_pol
            best_c = c_pol

    # Phase 8: Symmetry + micro adjustments
    if time_left() > 15:
        h_sym = repair_feasible(symmetrize(best_h))
        c_sym = exact_score(h_sym)
        if c_sym < best_c - 1e-14:
            best_h = h_sym
            best_c = c_sym

    # Phase 9: Tiny perturbations
    if time_left() > 10:
        for trial in range(5):
            if time_left() < 5:
                break
            eps = 1e-4 * (trial + 1)
            h_pert = repair_feasible(best_h + eps * (2.0 * rng.rand(len(best_h)) - 1.0))
            h_pol, c_pol = fine_polish_v2(h_pert, t_start, budget_s, max_iters=40)
            if c_pol < best_c - 1e-14:
                best_h = h_pol
                best_c = c_pol

    # Final validation
    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)

    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            final_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 1:
            val = float(eval_result[0])
            if np.isfinite(val) and val >= 0:
                final_c = val
    except Exception:
        pass

    return best_h, final_c, len(best_h)
