    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

    """Compute phi(i, k) for all i and k.
    phi(i,k) = (1-h[i-k])*[0<=i-k<n] - h[i+k]*[0<=i+k<n]
    Shape: (n, 2n-1) where k is offset by n-1
    """
    n = len(h)
    M = 2 * n - 1
    phi = np.zeros((n, M))
    for i in range(n):
        # k ranges from -(n-1) to n-1, stored as index k + (n-1)
        # phi(i,k) = (1-h[i-k]) if 0 <= i-k < n else 0
        #          - h[i+k] if 0 <= i+k < n else 0
        for k in range(1 - n, n):
            idx = k + n - 1
            if 0 <= i - k < n:
                phi[i, idx] += (1.0 - h[i - k])
            if 0 <= i + k < n:
                phi[i, idx] -= h[i + k]
    return phi

import numpy as np

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def compute_phi(h):
    n = len(h)
    M = 2 * n - 1
    phi = np.zeros((n, M))
    # Vectorized computation
    # phi[i, k+n-1] = (1-h[i-k])*[0<=i-k<n] - h[i+k]*[0<=i+k<n]
    for k in range(1 - n, n):
        idx = k + n - 1
        # First term: (1-h[i-k]) for valid i-k
        lo = max(0, k)  # i-k >= 0 → i >= k
        hi = min(n, n + k)  # i-k < n → i < n+k
        if lo < hi:
            i_range = np.arange(lo, hi)
            phi[i_range, idx] += (1.0 - h[i_range - k])
        # Second term: -h[i+k] for valid i+k
        lo2 = max(0, -k)  # i+k >= 0 → i >= -k
        hi2 = min(n, n - k)  # i+k < n → i < n-k
        if lo2 < hi2:
            i_range2 = np.arange(lo2, hi2)
            phi[i_range2, idx] -= h[i_range2 + k]
    return phi

def repair_feasible(h):
    n = len(h)
    h = np.clip(np.asarray(h, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h.copy()
    if diff > 0:
        order = np.argsort(1.0 - h)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, 1.0 - h[idx])
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    return h

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

def pair_move_step(h, profile, phi):
    """Find the best single pair move. Returns (s, t, alpha) or None."""
    n = len(h)
    T = np.max(profile)
    
    # Weighted scores emphasizing active lags
    # Use exponential weighting
    spread = T - np.min(profile)
    if spread < 1e-10:
        spread = 0.01
    weights = np.exp(-(profile - T) / (spread * 0.1))
    weights = np.maximum(weights, 1e-6)
    weights /= weights.sum()
    
    # Score for each position
    # As source (decrease): want high phi[i,k] for active k
    source_score = phi @ weights  # shape (n,)
    # As sink (increase): want low phi[i,k] for active k
    sink_score = phi @ weights  # same! We want low values
    
    # Actually, for a pair (s,t), the change in profile is alpha * (phi[t,:] - phi[s,:])
    # We want this to reduce the max, so we want phi[t,k] < phi[s,k] for active k
    # Good sources: high source_score (high phi values)
    # Good sinks: low sink_score (low phi values)
    
    n_cand = min(25, n)
    top_sources = np.argsort(source_score)[-n_cand:][::-1]
    top_sinks = np.argsort(sink_score)[:n_cand]
    
    best_new_T = T
    best_move = None
    
    for s in top_sources:
        for t in top_sinks:
            if s == t:
                continue
            alpha_max = min(h[s], 1.0 - h[t])
            if alpha_max <= 1e-10:
                continue
            # Try a few alpha values
            for frac in [0.3, 0.5, 0.7, 1.0]:
                alpha = alpha_max * frac
                new_profile = profile + alpha * (phi[t, :] - phi[s, :])
                new_T = np.max(new_profile)
                if new_T < best_new_T - 1e-12:
                    best_new_T = new_T
                    best_move = (s, t, alpha)
    
    if best_move is None:
        # Broader search: try all pairs with alpha = 0.5 * alpha_max
        for s in range(n):
            alpha_s_max = h[s]
            if alpha_s_max < 0.01:

            for t in range(n):
                if s == t:

                alpha_max = min(h[s], 1.0 - h[t])

                    continue
                alpha = 0.5 * alpha_max
                new_profile = profile + alpha * (phi[t, :] - phi[s, :])
                new_T = np.max(new_profile)
                if new_T < best_new_T - 1e-12:
                    best_new_T = new_T
                    best_move = (s, t, alpha)
    
    return best_move, best_new_T

def compute_phi(h):
    """Compute phi(i, k) for all i, k.
    phi[i, k+(n-1)] = (1-h[i-k])*[0<=i-k<n] - h[i+k]*[0<=i+k<n]
    """
    n = len(h)
    M = 2 * n - 1
    phi = np.zeros((n, M))
    for k in range(1 - n, n):
        idx = k + n - 1
        # (1-h[i-k]) for valid i
        i_start = max(0, k)
        i_end = min(n, n + k)
        if i_start < i_end:
            i = np.arange(i_start, i_end)
            phi[i, idx] += (1.0 - h[i - k])
        # -h[i+k] for valid i

            phi[i2, idx] -= h[i2 + k]
    return phi

import numpy as np
from scipy.optimize import minimize
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def compute_phi(h):
    n = len(h)
    M = 2 * n - 1
    phi = np.zeros((n, M))
    for k in range(1 - n, n):
        idx = k + n - 1
        i_start = max(0, k)
        i_end = min(n, n + k)
        if i_start < i_end:
            i = np.arange(i_start, i_end)
            phi[i, idx] += (1.0 - h[i - k])
        i_start2 = max(0, -k)
        i_end2 = min(n, n - k)
        if i_start2 < i_end2:
            i2 = np.arange(i_start2, i_end2)
            phi[i2, idx] -= h[i2 + k]
    return phi

def repair_feasible(h):
    n = len(h)
    h = np.clip(np.asarray(h, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, 1.0 - h[idx])
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    return h

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

def find_best_pair(h, profile, phi, n_cand=80, n_alpha=12):
    n = len(h)
    T = np.max(profile)
    delta = phi[np.newaxis, :, :] - phi[:, np.newaxis, :]
    alpha_max = np.minimum(h[:, np.newaxis], 1.0 - h[np.newaxis, :])
    frac = 0.5
    screen = profile[None, None, :] + (frac * alpha_max)[:, :, None] * delta
    new_max = np.max(screen, axis=2)
    np.fill_diagonal(new_max, 1e10)
    flat = new_max.ravel()
    top = np.argsort(flat)[:n_cand]
    best_new_T = T
    best_move = None
    for fi in top:
        s, t = np.unravel_index(fi, (n, n))
        if s == t:
            continue
        amax = alpha_max[s, t]
        if amax <= 1e-12:
            continue
        d = delta[s, t]
        alphas = np.linspace(0.0, amax, n_alpha)
        vals = np.maximum.reduce([profile + a * d for a in alphas[1:]])
        for a in alphas[1:]:
            nt = np.max(profile + a * d)
            if nt < best_new_T - 1e-13:
                best_new_T = nt
                best_move = (s, t, a)
    return best_move

        if move is None and no_improve > 10:
            move = find_best_pair_extended(h, profile, phi)
        if move is not None:

            h[t] += alpha
            h = np.clip(h, 0.0, 1.0)

            no_improve += 1
        if no_improve > 30:
            sr = rng_local.randint(n)

                ar = 0.005 * min(h[sr], 1.0 - h[tr])

                    h[sr] -= ar
                    h[tr] += ar
                    h = np.clip(h, 0.0, 1.0)

def slsqp_polish(h, t_start, time_limit):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1

    def obj(x):
        return x[-1]

    def obj_grad(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g

    def eq_fun(x):
        return np.array([x[:-1].sum() - n / 2.0])

    def eq_jac(x):
        J = np.zeros((1, n + 1))
        J[0, :n] = 1.0
        return J

    def ineq_fun(x):
        hx = x[:-1]
        t = x[-1]
        prof = np.correlate(hx, 1.0 - hx, mode="full") * dx
        return t - prof

    def ineq_jac(x):
        hx = x[:-1]
        phi = compute_phi(hx)
        J = np.zeros((M, n + 1))
        J[:, :n] = -dx * phi
        J[:, -1] = 1.0
        return J

    c0 = exact_score(h)
    x0 = np.concatenate([h, [c0 + 1e-4]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac},
    ]
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_grad,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': 300, 'ftol': 1e-15})
        if res.success and np.isfinite(res.x).all():
            h_sol = repair_feasible(res.x[:-1])
            c_sol = exact_score(h_sol)
            return h_sol, c_sol
    except Exception:
        pass
    return None, None
