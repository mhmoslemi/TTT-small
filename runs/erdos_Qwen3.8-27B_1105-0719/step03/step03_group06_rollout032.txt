def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(-(1.0 - h))
        remaining = diff
        for i in range(n):
            idx = order[i]
            cap = 1.0 - h[idx]
            if cap <= 1e-15:
                continue
            add = min(remaining, cap)
            h[idx] += add
            remaining -= add
            if remaining <= 1e-14:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            if h[idx] <= 1e-15:
                continue
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem

import numpy as np

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(-(1.0 - h))
        remaining = diff
        for i in range(n):
            idx = order[i]
            cap = 1.0 - h[idx]
            if cap <= 1e-15:
                continue
            add = min(remaining, cap)
            h[idx] += add
            remaining -= add
            if remaining <= 1e-14:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            if h[idx] <= 1e-15:
                continue
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-14:
                break
    return h

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def full_grad_fast(h):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    G = np.zeros((M, n))
    G[n-1] = dx * (1.0 - 2.0 * h)
    for d in range(1, n):
        m = n - 1 + d
        G[m, :n-d] += dx * (1.0 - h[d:])
        G[m, d:] -= dx * h[:n-d]
    for p in range(1, n):
        m = n - 1 - p
        G[m, p:] += dx * (1.0 - h[:n-p])
        G[m, :n-p] -= dx * h[p:]
    return G

def interp_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    edges_old = np.linspace(0.0, 2.0, n + 1)
    edges_new = np.linspace(0.0, 2.0, new_n + 1)
    h_new = np.zeros(new_n)
    for k in range(new_n):
        a, b = edges_new[k], edges_new[k + 1]
        total = 0.0
        for j in range(n):
            lo = max(a, edges_old[j])
            hi = min(b, edges_old[j + 1])
            if hi > lo:
                total += h[j] * (hi - lo)
        h_new[k] = total / (b - a)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

def momentum_softmax(h, max_iters, beta_start=15.0, beta_end=300.0,
                     lr_start=0.08, lr_end=0.002, mu=0.88, t_start=None, t_budget=None):
    n = len(h)
    h = h.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c = exact_score(h)
    for it in range(max_iters):
        if t_start is not None and t_budget is not None:
            if time.perf_counter() - t_start > t_budget - 15:
                break
        t = (it + 1) / max(max_iters, 1)
        beta = beta_start * (beta_end / beta_start) ** t
        lr = lr_start * (lr_end / lr_start) ** t
        profile = compute_profile(h)
        shifted = profile - profile.max()
        w = np.exp(beta * shifted)
        wsum = w.sum()
        if wsum < 1e-300:
            w = np.zeros(len(profile))
            w[np.argmax(profile)] = 1.0
        else:
            w /= wsum
        G = full_grad_fast(h)
        g = G.T @ w
        g -= g.mean()
        v = mu * v - lr * g
        h_new = h + v
        h_new = np.clip(h_new, 0.0, 1.0)

def active_descent_all(h, max_iters=400, tol=0.002, t_start=None, t_budget=None):
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    v = np.zeros(n)
    mu = 0.3
    for it in range(max_iters):
        if t_start is not None and t_budget is not None:
            if time.perf_counter() - t_start > t_budget - 15:
                break
        profile = compute_profile(h)
        c = float(profile.max())
        thresh = c - tol
        active = np.where(profile >= thresh)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(profile))])
        G = full_grad_fast(h)
        g = G[active].mean(axis=0)
        g -= g.mean()
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            break
        d = -g / gnorm
        v = mu * v + d
        vnorm = np.linalg.norm(v)
        if vnorm < 1e-15:
            break
        d = v / vnorm
        pos_mask = d > 1e-12
        neg_mask = d < -1e-12
        alpha_max = 0.5
        if pos_mask.any():
            alpha_max = min(alpha_max, float(((1.0 - h[pos_mask]) / d[pos_mask]).min()))
        if neg_mask.any():
            alpha_max = min(alpha_max, float((h[neg_mask] / (-d[neg_mask])).min()))
        alpha_max = max(alpha_max, 1e-10)
        best_alpha = 0.0
        best_val = c
        best_h_try = h.copy()
        fr = (np.sqrt(5.0) - 1.0) / 2.0
        a_ls, b_ls = 0.0, alpha_max
        for _ in range(20):
            c1 = b_ls - fr * (b_ls - a_ls)
            h_try = repair_feasible(h + c1 * d)
            f1 = exact_score(h_try)
            if f1 < best_val:
                best_val = f1
                best_alpha = c1
                best_h_try = h_try.copy()
            if f1 < exact_score(repair_feasible(h + a_ls * d)):
                b_ls = c1
            else:
                a_ls = c1
            if b_ls - a_ls < 1e-12:
                break
        if best_val >= c - 1e-14:
            h_try2 = repair_feasible(h + 0.01 * d)
            f_try2 = exact_score(h_try2)
            if f_try2 < c:
                best_val = f_try2
                best_alpha = 0.01
                best_h_try = h_try2.copy()
            else:
                break
        h = best_h_try
        if best_val < best_c:
            best_c = best_val
    return h, best_c

def slsqp_epigraph(h_start, maxiter=200):
    n = len(h_start)
    dx = 2.0 / n
    M = 2 * n - 1
    c0 = exact_score(h_start)
    t0 = c0 + 1e-4
    def obj(x):
        return x[-1]
    def ineq_fun(x):
        h = x[:-1]
        t = x[-1]
        profile = np.correlate(h, 1 - h, mode="full") * dx
        return t - profile
    def ineq_jac(x):
        h = x[:-1]
        G = full_grad_fast(h)
        J = np.zeros((M, n + 1))
        J[:, :n] = -G
        J[:, -1] = 1.0
        return J
    eq_fun = lambda x: x[:-1].sum() - n / 2.0
    eq_jac = lambda x: np.concatenate([np.ones(n), [0.0]])
    x0 = np.concatenate([h_start, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 2.0)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    try:
        res = minimize(obj, x0, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None and np.isfinite(res.x).all():
            h_new = repair_feasible(res.x[:-1])
            c_new = exact_score(h_new)
            if c_new <= c0 + 1e-8:
                return h_new
    except Exception:
        pass
    return None

        h4, c4 = active_descent_all(h3, max_iters=200, tol=0.002,
                                    t_start=t_start, t_budget=budget_s)
        if c4 < best_c:
            best_h, best_c = h4, c4

    # Stage 4: SLSQP on best
    if time_left() > 80:
        h_sol = slsqp_epigraph(best_h, maxiter=200)
        if h_sol is not None:
            c_sol = exact_score(h_sol)
            if c_sol < best_c:
                best_h, best_c = h_sol, c_sol
                if time_left() > 40:
                    h5, c5 = active_descent_all(best_h, max_iters=200, tol=0.001,
                                                t_start=t_start, t_budget=budget_s)
                    if c5 < best_c:
                        best_h, best_c = h5, c5

    # Stage 5: Structured seed exploration
    if time_left() > 120:
        n_exp = len(best_h)
        for pt in range(8):
            if time_left() < 30:
                break
            h_bin = structured_seed(n_exp, rng)
            h_mix = 0.3 * h_bin + 0.7 * np.full(n_exp, 0.5)
            h_mix = repair_feasible(h_mix)
            h6, c6 = momentum_softmax(h_mix, max_iters=300, beta_start=8, beta_end=100,
                                      lr_start=0.05, lr_end=0.001, mu=0.85,
                                      t_start=t_start, t_budget=budget_s)
            h7, c7 = active_descent_all(h6, max_iters=150, tol=0.003,
                                        t_start=t_start, t_budget=budget_s)
            if c7 < best_c:
                best_h, best_c = h7, c5 if c7 < best_c else (h7, c7)

    # Stage 6: Pair perturbation on best
    if time_left() > 60:
        h8, c8 = pair_perturb(best_h, n_trials=3000, t_start=t_start, t_budget=budget_s)
        if c8 < best_c:
            best_h, best_c = h8, c8

    # Stage 7: Tight active descent
    if time_left() > 40:
        h9, c9 = active_descent_all(best_h, max_iters=300, tol=0.0005,
                                    t_start=t_start, t_budget=budget_s)
        if c9 < best_c:
            best_h, best_c = h9, c9

    # Stage 8: Another SLSQP attempt
    if time_left() > 50:
        h_sol2 = slsqp_epigraph(best_h, maxiter=250)
        if h_sol2 is not None:
            c_sol2 = exact_score(h_sol2)
            if c_sol2 < best_c:
                best_h, best_c = h_sol2, c_sol2

    # Stage 9: Final momentum polish with high beta
    if time_left() > 30:
        h10, c10 = momentum_softmax(best_h, max_iters=300, beta_start=100, beta_end=500,
                                    lr_start=0.01, lr_end=0.0001, mu=0.9,
                                    t_start=t_start, t_budget=budget_s)
        if c10 < best_c:
            best_h, best_c = h10, c10

    # Stage 10: Ultra-fine active descent
    if time_left() > 20:
        h11, c11 = active_descent_all(best_h, max_iters=200, tol=0.0002,
                                      t_start=t_start, t_budget=budget_s)
        if c11 < best_c:
            best_h, best_c = h11, c11

    # Stage 11: Final pair perturb
    if time_left() > 15:
        h12, c12 = pair_perturb(best_h, n_trials=2000, t_start=t_start, t_budget=budget_s)
        if c12 < best_c:
            best_h, best_c = h12, c12

    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)

    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            final_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 2:
            final_c = float(eval_result[0])
    except Exception:
        pass

    return best_h, final_c, len(best_h)
