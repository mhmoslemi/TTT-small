```python
import numpy as np
from scipy.optimize import minimize, linprog
from scipy.special import logsumexp
import time


def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)


def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def c5_of_h(h, dx):
    return compute_corrs(h).max() * dx


def grad_corr_k(h, k, n):
    offset = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -offset)
    i_hi = min(n - 1, n - 1 - offset)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx + offset] += (1.0 - h[idx])
    grad[idx] += (-h[idx + offset])
    return grad


def smooth_max_grad_fn(h, dx, beta):
    n = len(h)
    corrs = compute_corrs(h)
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones_like(corrs) / len(corrs)
    else:
        w /= w_sum
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] > 1e-13:
            grad += w[k] * grad_corr_k(h, k, n)
    return grad * dx


def slsqp_optimize(h0, dx, target_sum, beta, maxiter):
    n = len(h0)

    def obj(x):
        corrs = compute_corrs(x)
        return logsumexp(beta * corrs) / beta * dx

    def grad_obj(x):
        return smooth_max_grad_fn(x, dx, beta)

    def c_fun(x):
        return x.sum() - target_sum

    def c_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': c_fun, 'jac': c_jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0, jac=grad_obj, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        h_out = project_feasible(res.x, target_sum)
    except Exception:
        h_out = project_feasible(h0, target_sum)
    return h_out, c5_of_h(h_out, dx)


def slsqp_spectrum(h0, dx, target_sum, betas, maxiter_each=2000):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    for beta in betas:
        h_opt, c5_opt = slsqp_optimize(h, dx, target_sum, beta, maxiter_each)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
        h = h_opt
    return best_h, best_c5


def lp_polish(h0, dx, target_sum, max_iters=500, init_step=0.08, tol_frac=0.15, max_active=80):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    step = init_step
    stall = 0

    for it in range(max_iters):
        corrs = compute_corrs(h)
        max_c = corrs.max()
        c5_val = max_c * dx

        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 40:
            step *= 0.6
            stall = 0
            if step < 1e-9:
                break

        tol_val = max_c * (1.0 - tol_frac)
        active_mask = corrs >= tol_val
        active_k = np.where(active_mask)[0]
        if len(active_k) > max_active:
            active_k = np.argsort(corrs)[-max_active:]
        if len(active_k) == 0:
            active_k = np.array([np.argmax(corrs)])

        m = len(active_k)
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        A_ub = np.zeros((m, n + 1))
        b_ub = np.zeros(m)
        for j, k in enumerate(active_k):
            gk = grad_corr_k(h, k, n)
            A_ub[j, :n] = dx * gk
            A_ub[j, n] = -1.0
            b_ub[j] = -corrs[k] * dx

        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)

        bounds = []
        for i in range(n):
            lo_i = max(-h[i], -step)
            hi_i = min(1.0 - h[i], step)
            bounds.append((lo_i, hi_i))
        bounds.append((0.0, None))

        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub,
                          A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
        except Exception:
            break
        if res.status != 0:
            break

        delta = res.x[:n]
        if np.max(np.abs(delta)) < 1e-14:
            break

        h_new = project_feasible(h + delta, target_sum)
        new_c5 = c5_of_h(h_new, dx)

        if new_c5 < c5_val - 1e-14:
            h = h_new
        else:
            improved = False
            for frac in [0.7, 0.5, 0.3, 0.15, 0.08]:

                if c5_of_h(h_try, dx) < c5_val - 1e-14:
                    h = h_try
                    improved = True
                    break
            if not improved:
                step *= 0.5
                if step < 1e-9:
                    break

    return best_h, best_c5


def pair_transfer_polish(h0, dx, target_sum, n_top=25, n_iter=10):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    for iteration in range(n_iter):
        corrs = compute_corrs(h)
        n_active = min(25, 2 * n - 1)
        top_k = np.argsort(corrs)[-n_active:]

        active_grad = np.zeros(n)
        for k in top_k:
            active_grad += grad_corr_k(h, k, n)

        sources = np.argsort(active_grad)[-n_top:]
        sinks = np.argsort(active_grad)[:n_top]

        improved = False
        for i in sources:
            for j in sinks:
                if i == j:
                    continue
                eps_hi = min(h[i], 1.0 - h[j])
                if eps_hi < 1e-12:
                    continue

                d = np.zeros(2 * n - 1)
                for k in range(2 * n - 1):
                    gk = grad_corr_k(h, k, n)
                    d[k] = gk[j] - gk[i]

                a, b = 0.0, eps_hi
                c_pt = b - gr * (b - a)
                d_pt = a + gr * (b - a)
                fc = (corrs + c_pt * d).max()
                fd = (corrs + d_pt * d).max()
                for _ in range(55):
                    if b - a < 1e-13:
                        break
                    if fc < fd:
                        b = d_pt
                        d_pt = c_pt
                        fd = fc
                        c_pt = b - gr * (b - a)
                        fc = (corrs + c_pt * d).max()
                    else:
                        a = c_pt
                        c_pt = d_pt
                        fc = fd
                        d_pt = a + gr * (b - a)
                        fd = (corrs + d_pt * d).max()

                eps_opt = (a + b) / 2.0
                if eps_opt < 1e-12:
                    continue
                new_max_rough = (corrs + eps_opt * d).max()
                if new_max_rough < corrs.max() - 1e-13:
                    new_h = h.copy()
                    new_h[i] -= eps_opt
                    new_h[j] += eps_opt
                    new_h = np.clip(new_h, 0.0, 1.0)
                    new_h = project_feasible(new_h, target_sum)
                    new_c5 = c5_of_h(new_h, dx)
                    if new_c5 < best_c5 - 1e-14:
                        h = new_h
                        best_c5 = new_c5
                        best_h = h.copy()
                        corrs = compute_corrs(h)
                        improved = True

        if not improved:
            break

    return best_h, best_c5

def make_structured_seed(n, target_sum, rng, mode):
    """Create structured seeds for multi-start."""
    if mode == 0:
        h = 0.5 + 0.2 * np.sin(2 * np.pi * rng.randint(1, 5) * np.arange(n) / n)
    elif mode == 1:
        period = rng.randint(3, 10)
        fill = max(1, int(n / 2.0 / n * period * rng.uniform(0.3, 0.7)))
        h = np.zeros(n)
        for i in range(n):
            h[i] = 1.0 if (i % period) < fill else 0.0
    elif mode == 2:
        h = rng.beta(2.0, 2.0, size=n)
    elif mode == 3:
        block_size = rng.randint(4, 12)
        n_blocks = n // block_size
        vals = rng.uniform(0, 1, size=n_blocks)
        h = np.repeat(vals, block_size)[:n]
        if len(h) < n:
            h = np.concatenate([h, np.zeros(n - len(h))])
    else:
        h = 0.5 + 0.3 * np.cos(2 * np.pi * rng.uniform(0.5, 3.0) * np.arange(n) / n + rng.uniform(0, 2 * np.pi))
    return project_feasible(h, target_sum)

            h_try = project_feasible(best_h + 0.08 * (rng.random(n) - 0.5), target_sum)
        elif trial < 5:
            freq = rng.randint(2, 6)
            pert = 0.06 * np.sin(2 * np.pi * freq * np.arange(n) / n + rng.uniform(0, 6.28))
            h_try = project_feasible(best_h + pert, target_sum)
        elif trial < 7:
            idx = rng.choice(n, size=n // 4, replace=False)
            pert = np.zeros(n)
            pert[idx] = rng.uniform(-0.15, 0.15, size=len(idx))
            h_try = project_feasible(best_h + pert, target_sum)
        else:
            h_try = make_structured_seed(n, target_sum, rng, mode=trial - 7)

        h_opt, c5_opt = full_optimize(h_try, dx, target_sum, time_left, time_budget=80)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 3: Try different n values
    n_base = len(best_h)
    n_candidates = []
    for delta in [-6, -4, -2, 2, 4, 6, 8]:
        nt = n_base + delta
        if nt >= 40 and nt % 2 == 0:
            n_candidates.append(nt)

    for n_try in n_candidates:
        if time_left() < 35:
            break
        dx_try = 2.0 / n_try
        ts_try = n_try / 2.0
        x_old = np.linspace(0, 2, n_base, endpoint=False) + 1.0 / n_base
        x_new = np.linspace(0, 2, n_try, endpoint=False) + 1.0 / n_try
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, ts_try)

        h_opt, c5_opt = full_optimize(h_interp, dx_try, ts_try, time_left, time_budget=60)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
            n_base = n_try

    # Phase 4: Final intensive polish
    n_b = len(best_h)
    dx_b = 2.0 / n_b
    ts_b = n_b / 2.0

    if time_left() > 25:
        h_opt, c5_opt = slsqp_spectrum(best_h, dx_b, ts_b, [1000, 2500, 5000, 8000], 1200)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    if time_left() > 20:
        h_lp, c5_lp = lp_polish(best_h, dx_b, ts_b, max_iters=400, init_step=0.03, tol_frac=0.05)
        if c5_lp < best_c5:
            best_c5 = c5_lp
            best_h = h_lp

    if time_left() > 15:
        h_pt, c5_pt = pair_transfer_polish(best_h, dx_b, ts_b, n_top=40, n_iter=12)
        if c5_pt < best_c5:
            best_c5 = c5_pt
            best_h = h_pt

    if time_left() > 12:
        h_osc, c5_osc = slow_oscillation_polish(best_h, dx_b, ts_b, n_freqs=15, n_iters=25)
        if c5_osc < best_c5:
            best_c5 = c5_osc
            best_h = h_osc

    if time_left() > 10:
        h_lp, c5_lp = lp_polish(best_h, dx_b, ts_b, max_iters=200, init_step=0.02, tol_frac=0.03)
        if c5_lp < best_c5:
            best_c5 = c5_lp
            best_h = h_lp

    if time_left() > 8:
        h_pt, c5_pt = pair_transfer_polish(best_h, dx_b, ts_b, n_top=50, n_iter=8)
        if c5_pt < best_c5:
            best_c5 = c5_pt
            best_h = h_pt

    # Final validation
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final / 2.0)
    best_c5 = c5_of_h(best_h, 2.0 / n_final)

    return best_h, best_c5, n_final
```
