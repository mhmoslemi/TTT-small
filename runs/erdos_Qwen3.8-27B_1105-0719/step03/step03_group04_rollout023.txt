def lp_active_step(h, profile, active_set, radius):
    """LP step to reduce active profile values."""
    n = len(h)
    dx = 2.0 / n
    G = build_gradient(h)
    G_a = G[active_set, :]
    p_a = profile[active_set]
    
    # Minimize t s.t. G_a @ delta + p_a <= t, |delta| <= radius, sum(delta) = 0
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_ub = np.zeros((len(active_set), n + 1))
    A_ub[:, :n] = G_a
    A_ub[:, -1] = -1.0
    b_ub = -p_a
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    lower = np.maximum(-h, -radius)
    upper = np.minimum(1.0 - h, radius)
    bounds = list(zip(lower, upper))
    bounds.append((0.0, None))
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'maxiter': 5000})
        if res.success and res.x is not None:
            return waterfill_project(h + res.x[:n], n / 2.0)
    except Exception:
        pass
    return None

def slsqp_polish(h, t_start, budget_s, max_iter=200):
    """SLSQP on exact epigraph formulation."""
    n = len(h)
    dx = 2.0 / n
    c0 = exact_score(h)
    
    def obj(x):
        return x[-1]
    
    def obj_grad(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    
    def eq_fun(x):
        return np.array([x[:-1].sum() - n / 2.0])
    
    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g
    
    def ineq_fun(x):
        h = x[:-1]
        prof = np.correlate(h, 1 - h, mode="full") * dx
        return x[-1] - prof  # should be >= 0
    
    def ineq_jac(x):
        h = x[:-1]
        G = build_gradient(h)
        J = np.zeros((2*n-1, n+1))
        J[:, :-1] = -G
        J[:, -1] = 1.0
        return J
    
    x0 = np.concatenate([h, [c0 + 1e-3]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_grad,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-14})
        if res.x is not None and np.isfinite(res.x).all():
            h_sol = waterfill_project(res.x[:-1], n / 2.0)
            return h_sol, exact_score(h_sol)
    except Exception:
        pass
    return h, c0

import numpy as np
from scipy.optimize import minimize, linprog
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def build_gradient(h):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    G = np.zeros((M, n))
    center = n - 1
    G[center, :] = dx * (1.0 - 2.0 * h)
    for d in range(1, n):
        G[center + d, 0:n-d] += dx * (1.0 - h[d:n])
        G[center + d, d:n] += -dx * h[0:n-d]
        G[center - d, 0:n-d] += -dx * h[d:n]
        G[center - d, d:n] += dx * (1.0 - h[0:n-d])
    return G

def waterfill_project(v, target):
    n = len(v)
    lo = v.min() - 1.5
    hi = v.max() + 1.5
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(v - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h = np.clip(v - lam, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            idx = int(np.argmax(1.0 - h))
            h[idx] = min(1.0, h[idx] + diff)
        else:
            idx = int(np.argmax(h))
            h[idx] = max(0.0, h[idx] + diff)
    return h

def smooth_logexp_weights(profile, tau):
    pmax = np.max(profile)
    if tau < 1e-10:
        w = np.zeros(len(profile))
        w[np.argmax(profile)] = 1.0
        return w
    shifted = np.clip((profile - pmax) / tau, -50, 50)
    e = np.exp(shifted)
    return e / e.sum()

    w = smooth_logexp_weights(profile, tau)
    G = build_gradient(h)
    return G.T @ w

    c_obj[-1] = 1.0
    A_ub = np.zeros((len(active_set), n + 1))
    A_ub[:, :n] = G_a
    A_ub[:, -1] = -1.0
    b_ub = -p_a
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    lower = np.maximum(-h, -radius)
    upper = np.minimum(1.0 - h, radius)
    bounds = list(zip(lower, upper))
    bounds.append((0.0, None))
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'maxiter': 5000})
        if res.success and res.x is not None and np.isfinite(res.x).all():
            return waterfill_project(h + res.x[:n], n / 2.0)
    except Exception:
        pass
    return None

def lp_refine_loop(h, t_start, budget_s, max_iters=80):
    n = len(h)
    best_h = h.copy()
    best_c = exact_score(h)
    for it in range(max_iters):
        if time.perf_counter() - t_start > budget_s - 10:
            break
        profile = compute_profile(h)
        T = np.max(profile)
        spread = T - np.min(profile)
        thresh = T - max(0.003, 0.08 * spread)
        active = np.where(profile >= thresh)[0]
        if len(active) > 15:
            top_idx = np.argsort(profile[active])[-15:]
            active = active[top_idx]
        if len(active) == 0:
            active = np.array([np.argmax(profile)])
        radius = 0.03
        h_lp = lp_active_step(h, profile, active, radius)
        if h_lp is not None:
            c_lp = exact_score(h_lp)
            if c_lp < best_c - 1e-11:
                best_c = c_lp
                best_h = h_lp.copy()
                h = h_lp
            else:
                break
        else:
            break
    return best_h, best_c

def slsqp_polish(h, t_start, budget_s, max_iter=200):
    n = len(h)
    dx = 2.0 / n
    c0 = exact_score(h)

    def obj(x):
        return x[-1]

    def obj_grad(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g

    def eq_fun(x):
        return np.array([x[:-1].sum() - n / 2.0])

    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g

    def ineq_fun(x):
        h = x[:-1]
        prof = np.correlate(h, 1 - h, mode="full") * dx
        return x[-1] - prof

    def ineq_jac(x):
        h = x[:-1]
        G = build_gradient(h)
        J = np.zeros((2 * n - 1, n + 1))
        J[:, :-1] = -G
        J[:, -1] = 1.0
        return J

    x0 = np.concatenate([h, [c0 + 1e-3]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_grad,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-15})
        if res.x is not None and np.isfinite(res.x).all():
            h_sol = waterfill_project(res.x[:-1], n / 2.0)
            return h_sol, exact_score(h_sol)
    except Exception:
        pass
    return h, c0

def symmetrize(h):
    return (h + h[::-1]) / 2.0

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    return waterfill_project(h_new, new_n / 2.0)

def fine_polish(h, t_start, budget_s, max_iters=60):
    n = len(h)
    best_h = h.copy()
    best_c = exact_score(h)
    for it in range(max_iters):
        if time.perf_counter() - t_start > budget_s - 5:
            break
        profile = compute_profile(h)
        T = np.max(profile)
        thresh = T - max(0.001, 0.03 * (T - np.min(profile)))
        active = np.where(profile >= thresh)[0]
        if len(active) > 10:
            top_idx = np.argsort(profile[active])[-10:]
            active = active[top_idx]
        h_lp = lp_active_step(h, profile, active, 0.01)
        if h_lp is not None:
            c_lp = exact_score(h_lp)
            if c_lp < best_c - 1e-12:
                best_c = c_lp
                best_h = h_lp.copy()
                h = h_lp
            else:
                break
        else:
            grad = smooth_obj_grad(h, 0.002)
            grad -= grad.mean()
            step = 0.005
            improved = False
            for ls in range(20):
                h_try = waterfill_project(h - step * grad, n / 2.0)
                c = exact_score(h_try)
                if c < exact_score(h) - 1e-13:
                    h = h_try
                    improved = True
                    if c < best_c:
                        best_c = c
                        best_h = h.copy()
                    break
                step *= 0.5
            if not improved:
                break
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s - (time.perf_counter() - t_start)

    try:
        parent = np.array(initial_h_values, dtype=np.float64)
    except (NameError, Exception):
        parent = np.full(70, 0.5)

    parent = np.clip(parent, 0.0, 1.0)
    parent = waterfill_project(parent, len(parent) / 2.0)

    best_h = parent.copy()
    best_c = exact_score(parent)

    h_sym = waterfill_project(symmetrize(parent), len(parent) / 2.0)
    c_sym = exact_score(h_sym)
    if c_sym < best_c:
        best_h, best_c = h_sym, c_sym

    n0 = len(best_h)

    # Phase 1: Annealed smooth descent
    if time_left() > 150:
        h_opt, c_opt = annealed_descent(best_h, n_stages=6, iters_per_stage=40,
                                        t_start=t_start, budget_s=budget_s)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt

    # Phase 2: LP active set refinement
    if time_left() > 100:
        h_lp, c_lp = lp_refine_loop(best_h, t_start, budget_s, max_iters=60)
        if c_lp < best_c:
            best_h, best_c = h_lp, c_lp

    # Phase 3: Second round of annealing (finer)
    if time_left() > 100:
        h_opt, c_opt = annealed_descent(best_h, n_stages=5, iters_per_stage=30,
                                        t_start=t_start, budget_s=budget_s)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt
        if time_left() > 60:
            h_lp, c_lp = lp_refine_loop(best_h, t_start, budget_s, max_iters=40)
            if c_lp < best_c:
                best_h, best_c = h_lp, c_lp

    # Phase 4: SLSQP polishing
    if time_left() > 50:
        h_sol, c_sol = slsqp_polish(best_h, t_start, budget_s, max_iter=150)
        if c_sol < best_c:
            best_h, best_c = h_sol, c_sol

    # Phase 5: Dimension transfers
    n_b = len(best_h)
    dims = sorted(set([d for d in [n_b - 2, n_b - 4, n_b + 2, n_b + 4,
                                    n_b + 6, n_b + 8, n_b - 6, n_b + 10]
                       if 28 <= d <= 160 and d != n_b and d % 2 == 0]),
                  key=lambda d: abs(d - n_b))
    for new_n in dims[:4]:
        if time_left() < 80:
            break
        h_interp = interpolate_to_n(best_h, new_n)
        h_opt, c_opt = annealed_descent(h_interp, n_stages=5, iters_per_stage=30,
                                        t_start=t_start, budget_s=budget_s)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt
            if time_left() > 50:
                h_lp, c_lp = lp_refine_loop(best_h, t_start, budget_s, max_iters=30)
                if c_lp < best_c:
                    best_h, best_c = h_lp, c_lp

    # Phase 6: Random restarts
    n_f = len(best_h)
    for restart in range(8):
        if time_left() < 40:
            break
        scale = 0.05 + 0.05 * (restart / 8.0)
        h_rand = np.clip(best_h + scale * rng.randn(n_f), 0, 1)
        h_rand = waterfill_project(h_rand, n_f / 2.0)
        h_opt, c_opt = annealed_descent(h_rand, n_stages=4, iters_per_stage=25,
                                        t_start=t_start, budget_s=budget_s)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt

    # Phase 7: Structured starts
    for mode in range(3):
        if time_left() < 30:
            break
        n_s = n_f
        h_struct = np.zeros(n_s)
        if mode == 0:
            for i in range(n_s):
                h_struct[i] = 0.5 + 0.3 * np.sin(np.pi * (i + 0.5) / n_s * 3 + rng.uniform(0, 1))
        elif mode == 1:
            nb = 4
            block = n_s // nb
            for b in range(nb):
                h_struct[b * block:(b + 1) * block] = 0.5 + 0.25 * (-1) ** b
        else:
            phi = 0.6180339887498949
            for i in range(n_s):
                h_struct[i] = 0.5 + 0.35 * np.sin(2 * np.pi * ((i * phi + 0.3) % 1.0))
        h_struct = waterfill_project(h_struct, n_s / 2.0)
        h_blend = 0.6 * best_h + 0.4 * h_struct
        h_blend = waterfill_project(h_blend, n_f / 2.0)
        h_opt, c_opt = annealed_descent(h_blend, n_stages=4, iters_per_stage=25,
                                        t_start=t_start, budget_s=budget_s)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt

    # Phase 8: Final fine polish
    if time_left() > 30:
        h_pol, c_pol = fine_polish(best_h, t_start, budget_s, max_iters=50)
        if c_pol < best_c:
            best_h, best_c = h_pol, c_pol

    # Phase 9: Final SLSQP
    if time_left() > 20:
        h_sol, c_sol = slsqp_polish(best_h, t_start, budget_s, max_iter=100)
        if c_sol < best_c:
            best_h, best_c = h_sol, c_sol

    # Phase 10: Micro-polish
    if time_left() > 10:
        h_pol, c_pol = fine_polish(best_h, t_start, budget_s, max_iters=30)
        if c_pol < best_c:
            best_h, best_c = h_pol, c_pol

    best_h = waterfill_project(best_h, len(best_h) / 2.0)
    final_c = exact_score(best_h)

    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            final_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 2:
            final_c = float(eval_result[0])
    except Exception:
        pass

    return best_h, final_c, len(best_h)
