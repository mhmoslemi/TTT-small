import numpy as np

def repair_feasible(vec):
    """Repair vector to satisfy 0<=h<=1 and sum(h)=n/2."""
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        # Need to add mass, prefer positions with most room
        order = np.argsort(-(1.0 - h))
        remaining = diff
        for i in range(n):
            idx = order[i]
            cap = 1.0 - h[idx]
            if cap <= 1e-15:
                continue
            add = min(remaining, cap)
            h[idx] += add
            remaining -= add
            if remaining <= 1e-14:
                break
    else:
        # Need to remove mass, prefer positions with most value
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            if h[idx] <= 1e-15:
                continue
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-14:
                break
    # Final correction
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] = float(np.clip(h[idx] + diff2, 0.0, 1.0))
    return h

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def grad_row(h, m, n):
    """Gradient of profile entry m with respect to h."""
    dx = 2.0 / n
    d = m - (n - 1)
    g = np.zeros(n)
    if d == 0:
        g += dx * (1.0 - 2.0 * h)
    elif d > 0:
        # P[m] = dx * sum_{i=0}^{n-1-d} h[i+d]*(1-h[i])
        # dP/dh[j] = dx*(1-h[j-d]) for j=d..n-1 (from h[j] term)
        #           - dx*h[j+d] for j=0..n-1-d (from (1-h[j]) term)
        g[d:] += dx * (1.0 - h[:n-d])
        g[:n-d] -= dx * h[d:]
    else:
        p = -d
        # P[m] = dx * sum_{i=0}^{n-1-p} h[i]*(1-h[i+p])
        # dP/dh[j] = dx*(1-h[j+p]) for j=0..n-1-p
        #           - dx*h[j-p] for j=p..n-1
        g[:n-p] += dx * (1.0 - h[p:])
        g[p:] -= dx * h[:n-p]
    return g

def lse_objective(profile, beta):
    """Compute LSE of profile."""
    shift = profile.max()
    if not np.isfinite(shift):
        return 0.0
    scaled = beta * (profile - shift)
    # Clip to avoid overflow
    scaled = np.clip(scaled, -500, 500)
    return shift + np.log(np.sum(np.exp(scaled))) / beta

def lse_gradient(h, beta, n=None):
    """Compute gradient of LSE objective."""
    if n is None:
        n = len(h)
    profile = compute_profile(h)
    shift = profile.max()
    if not np.isfinite(shift):
        return np.zeros(n)
    scaled = beta * (profile - shift)
    scaled = np.clip(scaled, -500, 500)
    weights = np.exp(scaled) / np.sum(np.exp(scaled))
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if weights[m] > 1e-10:
            grad += weights[m] * grad_row(h, m, n)
    return grad

def pairwise_polish(h, max_rounds=200):

    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    
    for rnd in range(max_rounds):

        c = float(profile.max())
        thresh = c - 0.003

        if len(active) == 0:
            active = np.array([int(np.argmax(profile))])
        
        # Compute the aggregate gradient of active lags
        g = np.zeros(n)
        for m in active:
            g += grad_row(h, m, n)
        
        # Try to find the best pairwise move
        # For each pair (i,j), moving delta from i to j (or vice versa)
        # The change in the max is approximately proportional to (g[i] - g[j]) * delta
        
        improved = False
        # Sort by gradient: want to decrease high-g, increase low-g
        idx = np.argsort(g)
        
        for attempt in range(min(50, n)):
            i = int(idx[attempt % n])           # low gradient -> increase
            j = int(idx[n - 1 - attempt % n])    # high gradient -> decrease
            if i == j:
                continue
            
            # Try moving mass from j to i
            delta_candidates = [0.05, 0.025, 0.01, 0.05 * (h[j] - h[i])]
            for delta in delta_candidates:
                if delta <= 1e-8 or delta > h[j] or delta > (1.0 - h[i]):
                    continue
                h_try = h.copy()
                h_try[i] += delta
                h_try[j] -= delta
                h_try = repair_feasible(h_try)
                s = exact_score(h_try)
                if s < best_c - 1e-15:
                    h = h_try
                    best_c = s
                    improved = True
                    # Recompute gradient
                    profile = compute_profile(h)
                    c = float(profile.max())
                    thresh = c - 0.003
                    active = np.where(profile >= thresh)[0]
                    if len(active) == 0:

                    g = np.zeros(n)
                    for m in active:
                        g += grad_row(h, m, n)
                    idx = np.argsort(g)

            if improved:
                break
        
        if not improved:

    return h, best_c

def interp_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    edges_old = np.linspace(0.0, 2.0, n + 1)
    edges_new = np.linspace(0.0, 2.0, new_n + 1)
    h_new = np.zeros(new_n)
    for k in range(new_n):
        a, b = edges_new[k], edges_new[k + 1]
        total = 0.0
        for j in range(n):
            lo = max(a, edges_old[j])
            hi = min(b, edges_old[j + 1])
            if hi > lo:
                total += h[j] * (hi - lo)
        h_new[k] = total / (b - a)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g
    def ineq_fun(x):
        h = x[:-1]
        t = x[-1]
        profile = np.correlate(h, 1 - h, mode="full") * dx
        return t - profile
    def ineq_jac(x):
        h = x[:-1]
        J = np.zeros((M, n + 1))
        for m in range(M):
            J[m, :n] = -grad_row(h, m, n)
        J[:, -1] = 1.0
        return J
    
    x0 = np.concatenate([h_start, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 2.0)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None and np.isfinite(res.x).all():
            h_new = repair_feasible(res.x[:-1])
            c_new = exact_score(h_new)
            if c_new <= c0 + 1e-8:
                return h_new
    except Exception:
        pass
    return None

def golden_binary_seed(n, rng=None):
    half = n // 2
    h = np.zeros(n)
    phi = (1 + np.sqrt(5)) / 2
    vals = np.array([(i * (1.0/phi)) % 1.0 for i in range(n)])
    top = np.argsort(vals)[-half:]
    h[top] = 1.0
    return repair_feasible(h)

def structured_seed(n, rng, variant=0):
    half = n // 2
    if variant == 0:
        # Golden ratio
        return golden_binary_seed(n)
    elif variant == 1:
        # Alternating blocks
        block = max(2, n // 16)
        h = np.zeros(n)
        for i in range(n):
            if ((i // block) % 2 == 0):
                h[i] = 1.0
        # Fix count
        ones = h.sum()
        if ones > half:
            order = np.argsort(h)  # zeros first
            excess = int(ones - half + 0.5)
            h[order[:excess]] = 0.0
        elif ones < half:
            order = np.argsort(-h)  # ones first
            deficit = int(half - ones + 0.5)
            h[order[:deficit]] = 1.0
        return repair_feasible(h)
    elif variant == 2:
        # Random
        h = np.zeros(n)
        h[rng.choice(n, half, replace=False)] = 1.0
        return repair_feasible(h)
    else:
        # Sine-based
        x = np.linspace(0, 4*np.pi, n)
        h = ((np.sin(x) + 1) / 2).astype(float)
        # Binary threshold
        h = (h > 0.5).astype(float)
        ones = h.sum()
        if ones > half:
            order = np.argsort(h)
            excess = int(ones - half)
            h[order[:excess]] = 0.0
        elif ones < half:
            order = np.argsort(-h)
            deficit = int(half - ones)
            h[order[:deficit]] = 1.0
        return repair_feasible(h)

def smooth_binary(h, alpha=0.3):
    """Slightly smooth a binary vector."""
    n = len(h)
    h_smooth = np.zeros(n)
    for i in range(n):
        window = h[max(0,i-1):min(n,i+2)]
        h_smooth[i] = window.mean()
    h_mix = alpha * h + (1 - alpha) * h_smooth
    return repair_feasible(h_mix)
