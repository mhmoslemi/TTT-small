def interpolate_to_n(h, new_n):

def build_gradient_rows(h, rows, n, dx):
    G = np.zeros((len(rows), n))
    for idx, m in enumerate(rows):
        d = m - (n - 1)
        if d == 0:
            G[idx, :] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            for i in range(n - d):
                G[idx, i + d] += dx * (1.0 - h[i])
                G[idx, i] -= dx * h[i + d]
        else:
            dp = -d
            for j in range(n - dp):
                G[idx, j + dp] += dx * (1.0 - h[j])
                G[idx, j] -= dx * h[j + dp]
    return G

def slsqp_active(h0, maxiter=300, active_k=12, tol=1e-12):
    n = len(h0)
    dx = 2.0 / n
    M = 2 * n - 1
    P = profile_h(h0)
    c0 = P.max()
    thresh = c0 - max(0.003, 0.05 * (c0 - np.median(P)))
    active = np.where(P >= thresh)[0]
    if len(active) == 0:
        active = np.array([np.argmax(P)])
    if len(active) > active_k:
        top_idx = np.argsort(P[active])[-active_k:]
        active = active[top_idx]
    
    def obj(x):
        return x[-1]
    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    def eq_fun(x):
        return x[:-1].sum() - n / 2.0
    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g
    def ineq_fun(x):
        h = x[:-1]
        t = x[-1]
        prof = profile_h(h)
        return t - prof[active]
    def ineq_jac(x):
        h = x[:-1]
        G = build_gradient_rows(h, active, n, dx)
        J = np.zeros((len(active), n + 1))
        J[:, :-1] = -G
        J[:, -1] = 1.0
        return J
    
    x0 = np.concatenate([h0, [c0 + 1e-4]])
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 1.0)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    try:

                       bounds=bounds, constraints=cons,

            h_new = repair_feasible(res.x[:-1])
            c_new = exact_score(h_new)

def slsqp_random_active(h0, maxiter=200, n_active=10, tol=1e-11, seed=0):

def softmax_gradient(h, beta, lam_sum=50.0):
    n = len(h)
    dx = 2.0 / n
    P = np.correlate(h, 1 - h, mode="full") * dx
    Pmax = P.max()
    exp_arg = beta * (P - Pmax)
    weights = np.exp(exp_arg)
    S = weights.sum()
    if S < 1e-300:
        weights = np.zeros(len(P))
        weights[np.argmax(P)] = 1.0
    else:
        weights /= S
    active = np.where(weights > 0.01 / len(P))[0]
    if len(active) == 0:
        active = np.argsort(weights)[-5:]
    G = build_gradient_rows(h, active, n, dx)
    w_active = weights[active]
    w_active = w_active / w_active.sum()
    grad = G.T @ w_active
    sum_err = h.sum() - n / 2.0
    grad += 2.0 * lam_sum * sum_err / n
    return grad

            if t_start is not None and time.perf_counter() - t_start > budget_s - 40:
                return h, best_c

            grad -= grad.mean()
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break

            vel = momentum * vel + grad
            vnorm = np.linalg.norm(vel)

                    h = h_try

                    if c_now < best_c:
                        best_c = c_now
            if improved:
                alpha = min(alpha * 1.05, 0.08)
            else:
                alpha *= 0.6
                if alpha < 1e-7:
                    break
    return h, best_c

def make_binary_start(n, rng):
    h = np.zeros(n)
    golden = 0.6180339887
    positions = set()
    for i in range(n // 2):
        pos = int((i * golden) % 1.0 * n)
        if pos not in positions:
            h[pos] = 1.0
            positions.add(pos)
        else:
            pos2 = (pos + 1) % n
            h[pos2] = 1.0
    return repair_feasible(h)

def symmetrize(h):
    return (h + h[::-1].copy()) / 2.0

def anneal_pair_cd(h, t_start, budget_s, n_sweeps=5, cool_rate=0.7, seed=0):
    """Annealed pairwise CD: occasionally try random perturbations."""
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    rng = np.random.RandomState(seed)
    temp = 0.02
    for phase in range(n_sweeps):
        if t_start is not None and time.perf_counter() - t_start > budget_s - 45:
            break
        # Perturb
        if phase > 0:
            z = rng.randn(n) * temp
            z -= z.mean()
            h_try = repair_feasible(h + z)
            c_try = exact_score(h_try)
            if c_try < best_c + 0.001:
                h = h_try
        # Pairwise descent
        h, c = pair_cd(h, max_sweeps=2, n_grid=32, t_start=t_start, budget_s=budget_s, seed=seed + phase)
        if c < best_c:
            best_c = c
        temp *= cool_rate
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()
    rng = np.random.RandomState(seed)
    
    def time_ok(margin=20):
        return time.perf_counter() - t_start < budget_s - margin
    
    parent = None
    try:
        parent = np.array(initial_h_values, dtype=np.float64)
    except NameError:
        pass
    except Exception:
        pass
    
    if parent is None or len(parent) < 4:
        parent = np.full(70, 0.5)
    
    parent = np.clip(parent, 0.0, 1.0)
    parent = repair_feasible(parent)
    
    best_h = parent.copy()
    best_c = exact_score(parent)
    
    # Symmetrize
    if time_ok():
        h_sym = repair_feasible(symmetrize(parent))
        c_sym = exact_score(h_sym)
        if c_sym < best_c - 1e-13:
            best_h = h_sym
            best_c = c_sym
    
    # Phase 1: Pairwise coordinate descent from parent
    if time_ok(80):
        h, c = pair_cd(best_h, max_sweeps=8, n_grid=40, t_start=t_start, budget_s=budget_s, seed=seed)
        if c < best_c - 1e-13:
            best_h = h
            best_c = c
    
    # Phase 2: SLSQP with active constraints (multiple tries)
    if time_ok(60):
        for trial in range(4):
            if not time_ok(15):
                break
            sol = slsqp_active(best_h, maxiter=250, active_k=10 + trial * 2, tol=1e-11)
            if sol is not None:
                c = exact_score(sol)
                if c < best_c - 1e-12:
                    best_h = sol
                    best_c = c
                # Follow with pair CD
                if time_ok(20):
                    h2, c2 = pair_cd(sol, max_sweeps=4, n_grid=36, t_start=t_start, budget_s=budget_s, seed=seed+trial)
                    if c2 < best_c - 1e-13:
                        best_h = h2
                        best_c = c2
    
    # Phase 3: Projected descent
    if time_ok(60):
        h_md, c_md = projected_descent(best_h, max_iters=80, momentum=0.45, t_start=t_start, budget_s=budget_s)
        if c_md < best_c - 1e-13:
            best_h = h_md
            best_c = c_md
    
    # Phase 4: More pair CD after gradient descent
    if time_ok(50):
        h, c = pair_cd(best_h, max_sweeps=10, n_grid=44, t_start=t_start, budget_s=budget_s, seed=seed+10)
        if c < best_c - 1e-13:
            best_h = h
            best_c = c
    
    # Phase 5: Dimension exploration
    n_orig = len(best_h)
    dims_to_try = []
    for d in range(max(24, n_orig - 14), min(100, n_orig + 14)):
        if d % 2 == 0 and d != n_orig:
            dims_to_try.append(d)
    dims_to_try.sort(key=lambda d: abs(d - n_orig))
    
    for new_n in dims_to_try[:6]:
        if not time_ok(40):
            break
        h_interp = interpolate_to_n(best_h, new_n)
        # Quick pair CD at new dimension
        h_new, c_new = pair_cd(h_interp, max_sweeps=4, n_grid=36, t_start=t_start, budget_s=budget_s, seed=seed+new_n)
        if c_new < best_c - 1e-10:
            # SLSQP polish
            if time_ok(20):
                sol = slsqp_active(h_new, maxiter=200, active_k=12, tol=1e-11)
                if sol is not None:
                    c_s = exact_score(sol)
                    if c_s < c_new - 1e-13:
                        h_new = sol
                        c_new = c_s
            # More pair CD
            if time_ok(20):
                h_new, c_new = pair_cd(h_new, max_sweeps=5, n_grid=44, t_start=t_start, budget_s=budget_s, seed=seed+new_n+1)
            best_h = h_new
            best_c = c_new
    
    # Phase 6: Annealed search from current best
    if time_ok(60):
        h_ann, c_ann = anneal_pair_cd(best_h, t_start, budget_s, n_sweeps=4, cool_rate=0.6, seed=seed+100)
        if c_ann < best_c - 1e-13:
            best_h = h_ann
            best_c = c_ann
    
    # Phase 7: Random restarts with structured perturbations
    if time_ok(50):
        n = len(best_h)
        for trial in range(5):
            if not time_ok(15):
                break
            # Different perturbation strategies
            eps = 0.01 * (trial + 1)
            z = rng.randn(n)
            z -= z.mean()
            h_p = repair_feasible(best_h + eps * z)
            # Try to use binary structure
            if trial % 2 == 0:
                h_bin = make_binary_start(n, rng)
                h_p = repair_feasible(0.5 * h_p + 0.5 * h_bin)
            # Quick optimize
            h_opt, c_opt = pair_cd(h_p, max_sweeps=3, n_grid=32, t_start=t_start, budget_s=budget_s, seed=seed+200+trial)
            if time_ok(10):
                sol = slsqp_active(h_opt, maxiter=150, active_k=10, tol=1e-10)
                if sol is not None:
                    c_s = exact_score(sol)
                    if c_s < c_opt:
                        h_opt = sol
                        c_opt = c_s
            if c_opt < best_c - 1e-10:
                best_h = h_opt
                best_c = c_opt
    
    # Phase 8: Final intensive polish
    if time_ok(40):
        # SLSQP
        sol = slsqp_active(best_h, maxiter=300, active_k=14, tol=1e-12)
        if sol is not None:
            c = exact_score(sol)
            if c < best_c - 1e-14:
                best_h = sol
                best_c = c
        # Pair CD final
        h, c = pair_cd(best_h, max_sweeps=8, n_grid=56, t_start=t_start, budget_s=budget_s, seed=seed+500)
        if c < best_c - 1e-14:
            best_h = h
            best_c = c
    
    # Phase 9: One more SLSQP + pair CD cycle
    if time_ok(25):
        sol = slsqp_active(best_h, maxiter=200, active_k=12, tol=1e-13)
        if sol is not None:
            c = exact_score(sol)
            if c < best_c - 1e-14:
                best_h = sol
                best_c = c
        h, c = pair_cd(best_h, max_sweeps=4, n_grid=56, t_start=t_start, budget_s=budget_s, seed=seed+600)
        if c < best_c - 1e-14:
            best_h = h
            best_c = c
    
    # Final validation
    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)
    
    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            final_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 1:
            if np.isfinite(eval_result[0]):
                final_c = float(eval_result[0])
    except Exception:
        pass
    
    return best_h, final_c, len(best_h)
