        if len(active_idx) > top_K:
            active_idx = np.argsort(corrs)[-top_K:]

        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        A_rows = []
        b_rows = []
        for k in active_idx:
            row = np.zeros(n + 1)
            row[:n] = dx * grad_corr_k(h, k, n)
            row[n] = -1.0
            A_rows.append(row)
            b_rows.append(-corrs[k] * dx - 1e-14)

        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)

        bounds = []
        for i in range(n):
            bounds.append((max(-h[i], -step), min(1.0 - h[i], step)))
        bounds.append((0.0, None))

        try:
            res = linprog(c_obj, A_ub=np.array(A_rows), b_ub=np.array(b_rows),
                          A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
        except Exception:
            break
        if res.status != 0:
            break

        delta = res.x[:n]
        if np.max(np.abs(delta)) < 1e-14:
            break

        h_new = project_feasible(h + delta, target_sum)
        new_c5 = c5_of_h(h_new, dx)

        if new_c5 < c5_curr - 1e-15:
            h = h_new
            step = min(step * 1.15, init_step)
        else:
            improved = False
            for frac in [0.6, 0.35, 0.15, 0.06]:
                h_try = project_feasible(h + frac * delta, target_sum)
                c5_try = c5_of_h(h_try, dx)
                if c5_try < c5_curr - 1e-15:
                    h = h_try
                    improved = True
                    break
            if not improved:
                step *= 0.45
                if step < 1e-9:
                    break

    return best_h, best_c5


def target_lp_polish(h0, dx, target_sum, max_iters=400, top_K=25):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)

    for it in range(max_iters):
        corrs = compute_corrs(h)
        c5_curr = corrs.max() * dx
        if c5_curr < best_c5 - 1e-15:
            best_c5 = c5_curr
            best_h = h.copy()

        target = c5_curr * 0.9995
        max_c = corrs.max()
        active_idx = np.argsort(corrs)[-top_K:]

        A_rows = []
        b_rows = []
        for k in active_idx:
            row = np.zeros(n)
            row[:n] = dx * grad_corr_k(h, k, n)
            A_rows.append(row)
            b_rows.append(target - corrs[k] * dx - 1e-13)

        A_eq = np.zeros((1, n))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)

        bounds = [(max(-h[i], -0.2), min(1.0 - h[i], 0.2))] * n

        try:
            res = linprog(c_obj=np.zeros(n), A_ub=np.array(A_rows), b_ub=np.array(b_rows),
                          A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
        except Exception:
            break
        if res.status != 0:
            break

        delta = res.x
        if np.max(np.abs(delta)) < 1e-14:
            break

        h_new = project_feasible(h + delta, target_sum)
        new_c5 = c5_of_h(h_new, dx)
        if new_c5 < c5_curr - 1e-15:
            h = h_new
        else:
            h_try = project_feasible(h + 0.3 * delta, target_sum)
            c5_try = c5_of_h(h_try, dx)
            if c5_try < c5_curr - 1e-15:
                h = h_try
            else:
                break

    return best_h, best_c5


def pair_transfer_polish(h0, dx, target_sum, n_top=30, n_iter=5):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)

    for iteration in range(n_iter):
        corrs = compute_corrs(h)
        grad = smooth_max_grad(h, dx, 800.0)

        dec_idx = np.argsort(grad)[-n_top:]
        inc_idx = np.argsort(grad)[:n_top]

        improved = False
        for i in dec_idx:
            for j in inc_idx:
                if i == j:
                    continue
                eps_lo = max(0.0, h[i] - (1.0 - h[j]))
                eps_hi = min(h[i], 1.0 - h[j])
                if eps_hi - eps_lo < 1e-12:
                    continue

                d = np.zeros(2 * n - 1)
                for k in range(2 * n - 1):
                    gk = grad_corr_k(h, k, n)
                    d[k] = gk[j] - gk[i]

                gr = (np.sqrt(5.0) - 1.0) / 2.0
                a, b = eps_lo, eps_hi
                c_pt = b - gr * (b - a)
                d_pt = a + gr * (b - a)
                fc = (corrs + c_pt * d).max()
                fd = (corrs + d_pt * d).max()
                for _ in range(40):
                    if b - a < 1e-15:
                        break
                    if fc < fd:
                        b = d_pt
                        d_pt = c_pt
                        fd = fc
                        c_pt = b - gr * (b - a)
                        fc = (corrs + c_pt * d).max()
                    else:
                        a = c_pt
                        c_pt = d_pt
                        fc = fd
                        d_pt = a + gr * (b - a)
                        fd = (corrs + d_pt * d).max()

                eps_opt = (a + b) / 2.0
                if abs(eps_opt) < 1e-13:
                    continue
                new_h = h.copy()
                new_h[i] -= eps_opt
                new_h[j] += eps_opt
                new_h = np.clip(new_h, 0.0, 1.0)
                new_h = project_feasible(new_h, target_sum)
                new_c5 = c5_of_h(new_h, dx)
                if new_c5 < best_c5 - 1e-14:
                    h = new_h
                    best_c5 = new_c5
                    best_h = h.copy()
                    corrs = compute_corrs(h)
                    improved = True

        if not improved:
            break

    return best_h, best_c5


def momentum_subgrad(h0, dx, target_sum, max_iter=20000, beta_s=40.0):
    n = len(h0)
    h = h0.copy()
    momentum = np.zeros(n)
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    no_improve = 0
    lr = 1.0 / np.sqrt(n)
    beta_m = 0.9

    for it in range(max_iter):
        corrs = compute_corrs(h)
        c5_val = corrs.max() * dx
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 8000:
            break

        shifted = corrs - corrs.max()
        w = np.exp(beta_s * shifted)
        w /= w.sum()
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] > 1e-12:
                grad += w[k] * grad_corr_k(h, k, n)
        grad *= dx

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break

        momentum = beta_m * momentum - lr * grad / (gnorm + 1e-10)
        h_new = h + momentum
        h_new = project_feasible(h_new, target_sum)
        new_c5 = c5_of_h(h_new, dx)

        tries = 0
        while new_c5 > c5_val + 1e-14 and tries < 5:
            lr *= 0.5
            momentum = beta_m * momentum - lr * grad / (gnorm + 1e-10)
            h_new = h + momentum
            h_new = project_feasible(h_new, target_sum)
            new_c5 = c5_of_h(h_new, dx)
            tries += 1
        h = h_new

    return best_h, best_c5


def mirror_symmetry(h):
    n = len(h)
    complement = 1.0 - h[::-1]
    return 0.5 * (h + complement)


def full_optimize(h0, dx, target_sum, time_left_fn, label=""):
    n = len(h0)
    best_h = h0.copy()
    best_c5 = c5_of_h(h0, dx)

    for beta in [100, 300, 800, 2000, 5000]:
        if time_left_fn() < 20:
            break
        h_opt, c5_val = slsqp_smooth(best_h, dx, target_sum, beta, 2000)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_opt

    if time_left_fn() > 40:
        for beta in [800, 2000, 5000]:
            if time_left_fn() < 20:
                break
            h_opt, c5_val = slsqp_zspace(best_h, dx, target_sum, beta, 1500)
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h_opt

    if time_left_fn() > 25:
        h_lp, c5_lp = adaptive_lp_polish(best_h, dx, target_sum, max_iters=400, init_step=0.12, top_K=25)
        if c5_lp < best_c5:
            best_c5 = c5_lp
            best_h = h_lp

    if time_left_fn() > 20:
        h_tl, c5_tl = target_lp_polish(best_h, dx, target_sum, max_iters=300, top_K=30)
        if c5_tl < best_c5:
            best_c5 = c5_tl
            best_h = h_tl

    if time_left_fn() > 15:
        h_pt, c5_pt = pair_transfer_polish(best_h, dx, target_sum, n_top=25, n_iter=4)
        if c5_pt < best_c5:
            best_c5 = c5_pt
            best_h = h_pt

    if time_left_fn() > 50:
        h_sg, c5_sg = momentum_subgrad(best_h, dx, target_sum, max_iter=15000, beta_s=50.0)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg

    if time_left_fn() > 25:
        h_f, c5_f = slsqp_smooth(best_h, dx, target_sum, 8000, 2500)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f

    if time_left_fn() > 20:
        h_lp2, c5_lp2 = adaptive_lp_polish(best_h, dx, target_sum, max_iters=300, init_step=0.06, top_K=40)
        if c5_lp2 < best_c5:
            best_c5 = c5_lp2
            best_h = h_lp2

    if time_left_fn() > 15:
        h_tl2, c5_tl2 = target_lp_polish(best_h, dx, target_sum, max_iters=250, top_K=50)
        if c5_tl2 < best_c5:
            best_c5 = c5_tl2
            best_h = h_tl2

    if time_left_fn() > 12:
        h_pt2, c5_pt2 = pair_transfer_polish(best_h, dx, target_sum, n_top=35, n_iter=3)
        if c5_pt2 < best_c5:
            best_c5 = c5_pt2
            best_h = h_pt2

    return best_h, best_c5


def make_periodic_seed(n, rng, period):
    h = np.zeros(n)
    frac = 0.5
    fill = max(1, int(frac * period))
    for i in range(n):
        if (i % period) < fill:
            h[i] = 1.0
    return project_feasible(h, n / 2.0)


def make_sturmian_seed(n, rng):
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    alpha = rng.uniform(0, 1)
    vals = np.floor(np.arange(n) * phi / n + alpha) - np.floor((np.arange(n) + 1) * phi / n + alpha)
    h = vals.astype(float)
    return project_feasible(h, n / 2.0)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s - (time.time() - t0)

    best_h = None
    best_c5 = float('inf')

    # Phase 1: Polish the given construction
    if 'initial_h_values' in dir() or True:
        try:
            h0 = np.array(initial_h_values, dtype=float)
            n0 = len(h0)
            dx0 = 2.0 / n0
            ts0 = n0 / 2.0
            h0 = project_feasible(h0, ts0)
            initial_c5 = c5_of_h(h0, dx0)

            # Try mirror symmetry
            h_mirror = mirror_symmetry(h0)
            h_mirror = project_feasible(h_mirror, ts0)
            mirror_c5 = c5_of_h(h_mirror, dx0)
            if mirror_c5 < initial_c5:
                h0 = h_mirror
                initial_c5 = mirror_c5

            h_opt, c5_val = full_optimize(h0, dx0, ts0, time_left, "main")
            best_h = h_opt
            best_c5 = c5_val
        except NameError:
            n0 = 146
            h0 = make_sturmian_seed(n0, rng)
            dx0 = 2.0 / n0
            ts0 = n0 / 2.0
            h_opt, c5_val = full_optimize(h0, dx0, ts0, time_left, "main")
            best_h = h_opt
            best_c5 = c5_val

    # Phase 2: Perturbation-based restarts
    n_best = len(best_h)
    dx_best = 2.0 / n_best
    ts_best = n_best / 2.0

    for trial in range(20):
        if time_left() < 30:
            break
        if trial % 4 == 0:
            perturb = 0.04 * (rng.random(n_best) - 0.5)
        elif trial % 4 == 1:
            perturb = 0.12 * (rng.random(n_best) - 0.5)
        elif trial % 4 == 2:
            idx = rng.choice(n_best, size=max(5, n_best // 8), replace=False)
            perturb = np.zeros(n_best)
            perturb[idx] = rng.choice([-1, 1], size=len(idx)) * 0.08
        else:
            h_mirror = mirror_symmetry(best_h)
            h_mirror = project_feasible(h_mirror, ts_best)
            perturb = h_mirror - best_h

        h_try = project_feasible(best_h + perturb, ts_best)
        h_opt2, c5_val2 = full_optimize(h_try, dx_best, ts_best, time_left, "perturb")
        if c5_val2 < best_c5:
            best_c5 = c5_val2
            best_h = h_opt2

    # Phase 3: Try different n values
    n_candidates = []
    for nn in range(int(n_best * 0.6), int(n_best * 1.5) + 1):
        if nn % 2 == 0 and nn >= 40:
            n_candidates.append(nn)
    n_candidates.sort(key=lambda x: abs(x - len(best_h)))
    n_candidates = [x for x in n_candidates if x != len(best_h)][:20]

    for n_try in n_candidates:
        if time_left() < 35:
            break
        dx_try = 2.0 / n_try
        ts_try = n_try / 2.0
        x_orig = np.linspace(0.0, 2.0, len(best_h), endpoint=False) + 1.0 / len(best_h)
        x_new = np.linspace(0.0, 2.0, n_try, endpoint=False) + 1.0 / n_try
        h_interp = np.interp(x_new, x_orig, best_h)
        h_interp = project_feasible(h_interp, ts_try)
        h_opt3, c5_val3 = full_optimize(h_interp, dx_try, ts_try, time_left, "ntry")
        if c5_val3 < best_c5:
            best_c5 = c5_val3
            best_h = h_opt3

    # Phase 4: Structured seeds at different sizes
    for n_struct in [100, 120, 146, 160, 180]:
        if time_left() < 40:
            break
        dx_s = 2.0 / n_struct
        ts_s = n_struct / 2.0
        for mk in range(3):
            if time_left() < 40:
                break
            if mk == 0:
                h_s = make_sturmian_seed(n_struct, rng)
            elif mk == 1:
                h_s = make_periodic_seed(n_struct, rng, period=7 + mk * 3)
            else:
                h_s = make_periodic_seed(n_struct, rng, period=11 + mk * 2)
            h_opt4, c5_val4 = full_optimize(h_s, dx_s, ts_s, time_left, "struct")
            if c5_val4 < best_c5:
                best_c5 = c5_val4
                best_h = h_opt4

    # Phase 5: Final intensive polish
    if time_left() > 25 and best_h is not None:
        n_b = len(best_h)
        dx_b = 2.0 / n_b
        ts_b = n_b / 2.0

        h_f1, c5_f1 = adaptive_lp_polish(best_h, dx_b, ts_b, max_iters=500, init_step=0.05, top_K=60)
        if c5_f1 < best_c5:
            best_c5 = c5_f1
            best_h = h_f1

        if time_left() > 20:
            h_f2, c5_f2 = target_lp_polish(best_h, dx_b, ts_b, max_iters=400, top_K=80)
            if c5_f2 < best_c5:
                best_c5 = c5_f2
                best_h = h_f2

        if time_left() > 15:
            h_f3, c5_f3 = pair_transfer_polish(best_h, dx_b, ts_b, n_top=50, n_iter=5)
            if c5_f3 < best_c5:
                best_c5 = c5_f3
                best_h = h_f3

        if time_left() > 15:
            h_f4, c5_f4 = slsqp_smooth(best_h, dx_b, ts_b, 10000, 3000)
            if c5_f4 < best_c5:
                best_c5 = c5_f4
                best_h = h_f4

        if time_left() > 12:
            h_f5, c5_f5 = adaptive_lp_polish(best_h, dx_b, ts_b, max_iters=300, init_step=0.03, top_K=100)
            if c5_f5 < best_c5:
                best_c5 = c5_f5
                best_h = h_f5

        if time_left() > 10:
            h_f6, c5_f6 = pair_transfer_polish(best_h, dx_b, ts_b, n_top=60, n_iter=4)
            if c5_f6 < best_c5:
                best_c5 = c5_f6
                best_h = h_f6

    # Final validation
    if best_h is None:
        n_final = 146
        best_h = make_sturmian_seed(n_final, rng)
        best_c5 = c5_of_h(best_h, 2.0 / n_final)
    else:
        n_final = len(best_h)
        best_h = project_feasible(best_h, n_final / 2.0)
        best_c5 = c5_of_h(best_h, 2.0 / n_final)

    return best_h, best_c5, len(best_h)
