import numpy as np
import time

def score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full")) * dx)

def prof_vec(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    d = target - h.sum()
    if abs(d) < 1e-14:
        return h
    if d > 0:
        order = np.argsort(h)
        rem = d
        for i in order:
            a = min(rem, 1.0 - h[i])
            h[i] += a
            rem -= a
            if rem <= 1e-14:
                break
    else:
        order = np.argsort(-h)
        rem = -d
        for i in order:
            a = min(rem, h[i])
            h[i] -= a
            rem -= a
            if rem <= 1e-14:
                break
    d2 = target - h.sum()
    if abs(d2) > 1e-14:
        if d2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += d2
        h = np.clip(h, 0, 1)
    return h

def build_G(h, n):
    """Jacobian: G[m,j] = d(P[m]*dx)/dh[j]"""
    dx = 2.0 / n
    M = 2 * n - 1
    G = np.zeros((M, n))
    for d in range(-(n - 1), n):
        m = n - 1 + d
        lo = max(0, -d)
        hi = min(n - 1, n - 1 - d)
        idx = np.arange(lo, hi + 1)
        G[m, idx] += (1.0 - h[idx + d])
        G[m, idx + d] += (-h[idx])
    return G * dx

def smooth_weights(p, tau):
    s = p - p.max()
    w = np.exp(s / tau)
    w /= w.sum()
    return w

def grad_direction(h, n, tau):
    p = prof_vec(h)
    G = build_G(h, n)
    w = smooth_weights(p, tau)
    g = G.T @ w
    g -= g.mean()
    gn = np.linalg.norm(g)
    if gn < 1e-14:
        return None
    return -g / gn

def golden_line(h, v, n, max_a=0.4):
    """Minimize score(h + alpha*v) via golden section, respecting box constraints."""
    pos = v > 1e-14
    neg = v < -1e-14
    a_lo = 0.0
    a_hi = max_a
    if pos.any():
        a_hi = min(a_hi, float(np.min((1.0 - h)[pos] / v[pos])))
    if neg.any():
        a_lo = max(a_lo, float(np.min(-h[neg] / v[neg])))
    if a_hi - a_lo < 1e-13:
        return h.copy(), score(h)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = a_lo, a_hi
    c_pt = b - gr * (b - a)
    d_pt = a + gr * (b - a)
    fc = score(h + c_pt * v)
    fd = score(h + d_pt * v)
    best_a = c_pt
    best_f = fc
    for _ in range(55):
        if b - a < 1e-11:
            break
        if fc < fd:
            b = d_pt
            d_pt = c_pt
            fd = fc
            c_pt = b - gr * (b - a)
            fc = score(h + c_pt * v)
        else:
            a = c_pt
            c_pt = d_pt
            fc = fd
            d_pt = a + gr * (b - a)
            fd = score(h + d_pt * v)
        if fc < best_f:
            best_f = fc
            best_a = c_pt
        if fd < best_f:
            best_f = fd
            best_a = d_pt
    return h + best_a * v, best_f

def interp_to(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    hn = np.interp(xt, xs, h)
    return feasible(hn, new_n)

def optimize_local(h0, n, tl_func, max_time=60, taus=None):
    """Local optimization with exact line search. Returns (h, c)."""
    if taus is None:
        taus = [0.001, 0.003, 0.008, 0.02, 0.05, 0.1]
    h = h0.copy()
    c = score(h)
    rng_local = np.random.RandomState(12345)
    stag = 0
    while tl_func() > max_time:
        improved = False
        # Multi-temperature gradient

            if v is None:

                stag = 0
                break
        if improved:

        # Coordinate moves

            if tl_func() < max_time:

            if h[i] < 0.02 and h[j] > 0.98:

            if h[i] < 0.02 and h[j] > 0.98:

            h_new, c_new = golden_line(h, v, n, max_a=0.25)

        if improved:

        v -= v.mean()
        vn = np.linalg.norm(v)

            v /= vn

        v -= v.mean()
        v /= np.linalg.norm(v)

            h = h_new
            c = c_new
            improved = True
            stag = 0

                break
    return h, c

def binary_search_local(h0, n, tl_func, max_swaps=200):
    """Local search on binary solutions via swaps."""
    h = (h0 > 0.5).astype(np.float64)
    n1 = int(h.sum())
    t1 = n // 2
    if n1 > t1:
        ones = np.where(h == 1)[0]
        order = np.argsort(h0[ones])[:n1 - t1]
        h[ones[order]] = 0
    elif n1 < t1:
        zeros = np.where(h == 0)[0]
        order = np.argsort(-h0[zeros])[:t1 - n1]
        h[zeros[order]] = 1
    c = score(h)
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    for _ in range(max_swaps):
        if tl_func() < 30:
            break
        best_swap_c = c
        best_pair = None
        for _ in range(30):
            i = ones[rng_global.randint(len(ones))]
            j = zeros[rng_global.randint(len(zeros))]
            h[i] = 0
            h[j] = 1
            c_try = score(h)
            h[i] = 1
            h[j] = 0
            if c_try < best_swap_c - 1e-14:
                best_swap_c = c_try
                best_pair = (i, j)
        if best_pair is None:
            break
        i, j = best_pair
        h[i] = 0
        h[j] = 1
        c = best_swap_c
        ones = np.where(h == 1)[0]
