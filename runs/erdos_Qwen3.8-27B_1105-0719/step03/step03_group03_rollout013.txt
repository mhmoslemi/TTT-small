def compute_profile(h):

def smooth_objective(h, alpha):
    """Log-sum-exp smoothed max of profile."""
    profile = compute_profile(h)
    m = profile.max()
    if alpha * m > 700:  # overflow protection
        return m + np.log(profile.size) / alpha
    s = np.exp(alpha * (profile - m))
    return m + np.log(s.sum()) / alpha

def smooth_gradient(h, alpha):

    n = len(h)
    dx = 2.0 / n
    profile = compute_profile(h)
    m = profile.max()

    shifted = np.clip(shifted, -50, 50)
    w = np.exp(shifted)
    w_sum = w.sum()
    w = w / w_sum
    w = w / alpha  # This is the actual softmax gradient weight / alpha
    
    grad = np.zeros(n)
    center = n - 1
    for d in range(-(n-1), n):

    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    # Redistribute
    if diff > 0:
        # Need to add 'diff' total, spread to elements with most room
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx])
            h[idx] += add
            remaining -= add

            rem = min(remaining, h[idx])

import numpy as np
from scipy.optimize import minimize
import time

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def exact_score(h):
    return float(np.max(compute_profile(h)))

def smooth_obj(h, alpha):
    p = compute_profile(h)
    m = p.max()
    if alpha * (p.max() - p.min()) > 500:
        return m
    s = np.exp(alpha * (p - m))
    return m + np.log(s.sum()) / alpha

def smooth_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    p = compute_profile(h)
    m = p.max()
    shifted = np.clip(alpha * (p - m), -50, 50)
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        return np.zeros(n)
    w = w / ws / alpha
    grad = np.zeros(n)
    center = n - 1
    for d in range(-(n-1), n):
        idx = center + d
        wm = w[idx]
        if abs(wm) < 1e-18:
            continue
        if d == 0:
            grad += wm * dx * (1.0 - 2.0 * h)
        elif d > 0:
            i1 = np.arange(n - d)
            grad[i1] += wm * dx * (1.0 - h[i1 + d])
            i2 = np.arange(d, n)
            grad[i2] += wm * dx * (-h[i2 - d])
        else:
            dp = -d
            i1 = np.arange(dp, n)
            grad[i1] += wm * dx * (1.0 - h[i1 - dp])
            i2 = np.arange(n - dp)
            grad[i2] += wm * dx * (-h[i2 + dp])
    return grad

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        cap = 1.0 - h
        order = np.argsort(-cap)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, cap[idx])
            h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    return h

def slsqp_smooth(h0, alpha, maxiter=300):
    n = len(h0)
    def obj(x):
        return smooth_obj(x, alpha)
    def grad(x):
        return smooth_grad(x, alpha)
    def eq_fun(x):
        return x.sum() - n / 2.0
    def eq_jac(x):
        return np.ones(n)
    x0 = repair_feasible(h0)
    try:
        res = minimize(obj, x0, method='SLSQP', jac=grad,
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}],
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None and np.isfinite(res.x).all():
            return repair_feasible(res.x)
    except Exception:
        pass
    return h0

def multi_alpha_optimize(h0, alpha_start=20, alpha_end=8000, n_levels=8, maxiter_per=200):
    h = h0.copy()
    best_h = h.copy()
    best_c = exact_score(h)
    n = len(h)
    for i in range(n_levels):
        alpha = alpha_start * (alpha_end / alpha_start) ** (i / max(n_levels - 1, 1))
        alpha = float(alpha)
        h = slsqp_smooth(h, alpha, maxiter=maxiter_per)
        c = exact_score(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    return best_h, best_c

def interp_h(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    fx = np.concatenate([[0.0], xs, [2.0]])
    fh = np.concatenate([[h[0]], h, [h[-1]]])
    hn = np.interp(xt, fx, fh)
    return repair_feasible(np.clip(hn, 0, 1))

def grad_descent_step(h, alpha, lr, n_steps=5):
    n = len(h)
    h_curr = h.copy()
    for _ in range(n_steps):
        g = smooth_grad(h_curr, alpha)
        h_try = repair_feasible(h_curr - lr * g)
        if exact_score(h_try) <= exact_score(h_curr) + 1e-10:
            h_curr = h_try
        else:
            break
    return h_curr

            else:

                i2 = np.arange(n - dp)
