import numpy as np
from scipy.optimize import minimize, linprog
from scipy.special import logsumexp
import time


def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)


def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def c5_of_h(h, dx):
    return compute_corrs(h).max() * dx


def smooth_max_grad_fast(h, dx, beta):
    n = len(h)
    corrs = compute_corrs(h)
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones_like(corrs) / len(corrs)
    else:
        w /= w_sum
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        nk = k - (n - 1)
        # First term: (1-h[j-nk]) for j where 0<=j-nk<n
        j_lo = max(0, nk)
        j_hi = min(n, nk + n)
        if j_lo < j_hi:
            grad[j_lo:j_hi] += w[k] * (1.0 - h[j_lo - nk:j_hi - nk])
        # Second term: -h[j+nk] for j where 0<=j+nk<n
        j_lo2 = max(0, -nk)
        j_hi2 = min(n, n - nk)
        if j_lo2 < j_hi2:
            grad[j_lo2:j_hi2] -= w[k] * h[j_lo2 + nk:j_hi2 + nk]
    return grad * dx


def smooth_max_val(h, dx, beta):
    corrs = compute_corrs(h)
    return logsumexp(beta * corrs) / beta * dx


def slsqp_direct(h0, dx, target_sum, beta, maxiter):
    n = len(h0)

    def obj(x):
        return smooth_max_val(x, dx, beta)

    def grad_obj(x):
        return smooth_max_grad_fast(x, dx, beta)

    def c_fun(x):
        return x.sum() - target_sum

    def c_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': c_fun, 'jac': c_jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0, jac=grad_obj, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        h_out = project_feasible(res.x, target_sum)
    except Exception:
        h_out = project_feasible(h0, target_sum)
    return h_out, c5_of_h(h_out, dx)


def binary_to_continuous(b, dx, target_sum):
    return project_feasible(b.astype(float), target_sum)


def binary_local_search(h0, dx, target_sum, max_passes=500, time_limit=60):

            break
        best_delta = 0
        best_swap = None
        base_corr = compute_corrs(b)
        base_max = base_corr.max()
        for i in one_list:
            for j in zero_list:
                b[i] = 0.0
                b[j] = 1.0
                new_corr = compute_corrs(b)
                new_max = new_corr.max()
                if new_max < best_delta - 1e-12:
                    best_delta = new_max
                    best_swap = (i, j)
                b[i] = 1.0
                b[j] = 0.0
            if time.time() - t_start > time_limit:
                break
        if best_swap is not None:
            i, j = best_swap
            b[i] = 0.0
            b[j] = 1.0
            one_set.remove(i)
            zero_set.remove(j)
            one_set.add(j)
            zero_set.add(i)
            if best_delta < best_corr_max:
                best_corr_max = best_delta
                best_b = b.copy()
            improved = True
        if not improved:
            break
    return project_feasible(best_b, target_sum), c5_of_h(project_feasible(best_b, target_sum), dx)


def make_balanced_binary(n, rng):
    h = np.zeros(n)
    positions = rng.permutation(n)
    h[positions[:n // 2]] = 1.0
    return h


def make_structured_binary(n, rng, seed_offset=0):
    h = np.zeros(n)
    p = rng.randint(3, max(4, n // 10) + 1) + seed_offset
    fill = max(1, int(n / 2.0 / n * p))
    for i in range(n):
        if (i % p) < fill:
            h[i] = 1.0
    # Fix balance
    n_ones = int(h.sum())
    n_needed = n // 2
    if n_ones > n_needed:
        one_idx = np.where(h == 1)[0]
        for idx in one_idx[:n_ones - n_needed]:
            h[idx] = 0.0
    elif n_ones < n_needed:
        zero_idx = np.where(h == 0)[0]
        for idx in zero_idx[:n_needed - n_ones]:
            h[idx] = 1.0
    return h


def iterative_lp_polish(h0, dx, target_sum, max_iters=200, init_step=0.12, time_limit=30):
    n = len(h0)
    t_start = time.time()
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    step = init_step
    stall = 0

    for it in range(max_iters):
        if time.time() - t_start > time_limit:
            break
        corrs = compute_corrs(h)
        max_c = corrs.max()
        c5_val = max_c * dx

        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 30:
            step *= 0.7
            stall = 0
            if step < 1e-7:
                break

        tol_frac = 0.05
        tol_val = max_c * (1.0 - tol_frac)
        active_mask = corrs >= tol_val
        active_k = np.where(active_mask)[0]
        if len(active_k) > 60:
            active_k = np.argsort(corrs)[-60:]

        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        A_rows = []
        b_rows = []
        for k in active_k:
            nk = k - (n - 1)
            row = np.zeros(n + 1)
            # Gradient of correlation k w.r.t. h
            j_lo = max(0, nk)
            j_hi = min(n, nk + n)
            if j_lo < j_hi:
                row[j_lo:j_hi] += (1.0 - h[j_lo - nk:j_hi - nk])
            j_lo2 = max(0, -nk)
            j_hi2 = min(n, n - nk)
            if j_lo2 < j_hi2:
                row[j_lo2:j_hi2] -= h[j_lo2 + nk:j_hi2 + nk]
            row *= dx
            row[n] = -1.0
            A_rows.append(row)
            b_rows.append(-corrs[k] * dx - 1e-14)

        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)

        bounds = []
        for i in range(n):
            bounds.append((max(-h[i], -step), min(1.0 - h[i], step)))
        bounds.append((0.0, None))

        try:
            res = linprog(c_obj, A_ub=np.array(A_rows), b_ub=np.array(b_rows),
                          A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
        except Exception:
            break
        if res.status != 0:
            break

        delta = res.x[:n]
        if np.max(np.abs(delta)) < 1e-14:
            break

        h_new = project_feasible(h + delta, target_sum)
        new_c5 = c5_of_h(h_new, dx)

        if new_c5 < c5_val - 1e-14:
            h = h_new
            step = min(step * 1.15, init_step)
        else:
            improved = False
            for frac in [0.5, 0.25, 0.1]:
                h_new = project_feasible(h + frac * delta, target_sum)
                new_c5 = c5_of_h(h_new, dx)
                if new_c5 < c5_val - 1e-14:
                    h = h_new
                    improved = True
                    break
            if not improved:
                step *= 0.5
                if step < 1e-8:
                    break

    return best_h, best_c5


def pair_transfer_polish(h0, dx, target_sum, n_top=20, n_iter=3, time_limit=20):
    n = len(h0)
    t_start = time.time()
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)

    for iteration in range(n_iter):
        if time.time() - t_start > time_limit:
            break
        corrs = compute_corrs(h)
        grad = smooth_max_grad_fast(h, dx, 500.0)

        dec_idx = np.argsort(grad)[-n_top:]
        inc_idx = np.argsort(grad)[:n_top]

        improved = False
        for i in dec_idx:
            if time.time() - t_start > time_limit:
                break
            for j in inc_idx:
                if i == j:
                    continue
                eps_lo = -h[i]
                eps_hi = min(h[i], 1.0 - h[j])
                if eps_hi - eps_lo < 1e-12:
                    continue

                d = np.zeros(2 * n - 1)
                for k in range(2 * n - 1):
                    nk = k - (n - 1)
                    gi = 0.0
                    if 0 <= i - nk < n:
                        gi += (1.0 - h[i - nk])
                    if 0 <= i + nk < n:
                        gi -= h[i + nk]
                    gj = 0.0
                    if 0 <= j - nk < n:
                        gj += (1.0 - h[j - nk])
                    if 0 <= j + nk < n:
                        gj -= h[j + nk]
                    d[k] = gj - gi

                # Golden section search for best eps
                gr = (np.sqrt(5.0) - 1.0) / 2.0
                a, b = eps_lo, eps_hi
                c_pt = b - gr * (b - a)
                d_pt = a + gr * (b - a)

                def f_ps(eps):
                    return (corrs + eps * d).max()

                fc, fd = f_ps(c_pt), f_ps(d_pt)
                for _ in range(40):
                    if b - a < 1e-14:
                        break
                    if fc < fd:
                        b = d_pt
                        d_pt = c_pt
                        fd = fc
                        c_pt = b - gr * (b - a)
                        fc = f_ps(c_pt)
                    else:
                        a = c_pt
                        c_pt = d_pt
                        fc = fd
                        d_pt = a + gr * (b - a)
                        fd = f_ps(d_pt)

                eps_opt = (a + b) / 2.0
                new_c5_rough = (corrs + eps_opt * d).max() * dx
                if new_c5_rough < best_c5 - 1e-13:
                    new_h = h.copy()
                    new_h[i] -= eps_opt
                    new_h[j] += eps_opt
                    new_h = np.clip(new_h, 0.0, 1.0)
                    new_h = project_feasible(new_h, target_sum)
                    new_c5 = c5_of_h(new_h, dx)
                    if new_c5 < best_c5 - 1e-14:
                        h = new_h
                        best_c5 = new_c5
                        best_h = h.copy()
                        corrs = compute_corrs(h)
                        improved = True

        if not improved:
            break

    return best_h, best_c5


def penalty_descent(h0, dx, target_sum, max_iter=3000, time_limit=30):
    n = len(h0)
    t_start = time.time()
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    target = best_c5 * 0.995
    lr = 0.02 / np.sqrt(n)

    for it in range(max_iter):
        if time.time() - t_start > time_limit:
            break
        corrs = compute_corrs(h)
        c5_val = corrs.max() * dx

        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()

        # Penalty gradient: 2*dx * sum_{k: corrs[k]*dx > target} (corrs[k]*dx - target) * grad_k
        penalty_mask = corrs * dx > target
        if not penalty_mask.any():
            break
        weights = (corrs * dx - target) * penalty_mask
        w_sum = weights.sum()
        if w_sum < 1e-20:
            break

        grad = np.zeros(n)
        for k in np.where(penalty_mask)[0]:
            wk = weights[k]
            nk = k - (n - 1)
            j_lo = max(0, nk)
            j_hi = min(n, nk + n)
            if j_lo < j_hi:
                grad[j_lo:j_hi] += wk * (1.0 - h[j_lo - nk:j_hi - nk])
            j_lo2 = max(0, -nk)
            j_hi2 = min(n, n - nk)
            if j_lo2 < j_hi2:
                grad[j_lo2:j_hi2] -= wk * h[j_lo2 + nk:j_hi2 + nk]
        grad *= 2.0 * dx / w_sum

        h_new = project_feasible(h - lr * grad, target_sum)
        new_c5 = c5_of_h(h_new, dx)
        if new_c5 < c5_val:
            h = h_new
        else:
            lr *= 0.5
            if lr < 1e-10:
                break
        # Slowly tighten target
        target = max(best_c5 * 0.98, c5_val * 0.999)

    return best_h, best_c5


def full_optimize(h0, dx, target_sum, time_left_fn, budget_phase=100):
    n = len(h0)
    best_h = h0.copy()
    best_c5 = c5_of_h(h0, dx)

    # SLSQP with increasing beta
    for beta in [100, 300, 800, 2000, 5000]:
        if time_left_fn() < 10:
            break
        h_opt, c5_val = slsqp_direct(best_h, dx, target_sum, beta, 2000)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_opt

    # LP polish
    if time_left_fn() > 15:
        t_lim = min(30, time_left_fn() * 0.3)
        h_lp, c5_lp = iterative_lp_polish(best_h, dx, target_sum, max_iters=300,
                                          init_step=0.1, time_limit=t_lim)
        if c5_lp < best_c5:
            best_c5 = c5_lp
            best_h = h_lp

    # Pair transfer
    if time_left_fn() > 10:
        h_pt, c5_pt = pair_transfer_polish(best_h, dx, target_sum, n_top=20, n_iter=3,
                                           time_limit=min(15, time_left_fn() * 0.3))
        if c5_pt < best_c5:
            best_c5 = c5_pt
            best_h = h_pt

    # Penalty descent
    if time_left_fn() > 15:
        h_pen, c5_pen = penalty_descent(best_h, dx, target_sum, max_iter=2000,
                                        time_limit=min(20, time_left_fn() * 0.3))
        if c5_pen < best_c5:
            best_c5 = c5_pen
            best_h = h_pen

    # Final SLSQP + LP
    if time_left_fn() > 10:
        h_f, c5_f = slsqp_direct(best_h, dx, target_sum, 3000, 1500)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f
    if time_left_fn() > 10:
        h_lp2, c5_lp2 = iterative_lp_polish(best_h, dx, target_sum, max_iters=200,
                                            init_step=0.06, time_limit=min(15, time_left_fn() * 0.4))
        if c5_lp2 < best_c5:
            best_c5 = c5_lp2
            best_h = h_lp2
    if time_left_fn() > 8:
        h_pt2, c5_pt2 = pair_transfer_polish(best_h, dx, target_sum, n_top=15, n_iter=2,
                                             time_limit=min(10, time_left_fn() * 0.4))
        if c5_pt2 < best_c5:
            best_c5 = c5_pt2
            best_h = h_pt2

    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s - (time.time() - t0)

    best_h = None
    best_c5 = float('inf')

    # Phase 1: Use initial construction and optimize
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
        n = len(h0)
        dx = 2.0 / n
        target_sum = n / 2.0
        h0 = project_feasible(h0, target_sum)
        init_c5 = c5_of_h(h0, dx)
        best_h = h0.copy()
        best_c5 = init_c5

        # Binary search from the initial construction
        if time_left() > 200:
            b_init = (h0 > 0.5).astype(float)
            n_ones = int(b_init.sum())
            if n_ones != n // 2:
                if n_ones > n // 2:
                    one_idx = np.where(b_init == 1)[0]
                    for idx in one_idx[:n_ones - n // 2]:
                        b_init[idx] = 0.0
                else:
                    zero_idx = np.where(b_init == 0)[0]
                    for idx in zero_idx[:n // 2 - n_ones]:
                        b_init[idx] = 1.0
            h_bin, c5_bin = binary_local_search(b_init, dx, target_sum,
                                                max_passes=200, time_limit=40)
            if c5_bin < best_c5:
                best_c5 = c5_bin
                best_h = h_bin.copy()

        # Continuous optimization from best
        if time_left() > 100:
            h_opt, c5_opt = full_optimize(best_h, dx, target_sum, time_left, budget_phase=100)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
    else:
        # Fallback: create initial from structured pattern
        n = 146
        dx = 2.0 / n
        target_sum = n / 2.0
        h0 = make_structured_binary(n, rng, seed_offset=0)
        best_h = project_feasible(h0, target_sum)
        best_c5 = c5_of_h(best_h, dx)

    # Phase 2: Multi-n binary exploration
    n_base = len(best_h)
    n_candidates = []
    for nn in range(max(60, n_base - 40), min(220, n_base + 40), 2):
        n_candidates.append(nn)
    rng.shuffle(n_candidates)
    # Prioritize n near current best
    n_candidates.sort(key=lambda x: abs(x - n_base))
    n_candidates = n_candidates[:30]

    for n_try in n_candidates:
        if time_left() < 40:
            break
        if n_try == n_base:
            continue
        dx_try = 2.0 / n_try
        ts_try = n_try / 2.0

        # Try multiple starting points for binary search
        best_n_c5 = float('inf')
        best_n_h = None
        n_starts = 3 if time_left() > 60 else 1

        for st in range(n_starts):
            if time_left() < 15:
                break
            if st == 0:
                # Interpolate from current best
                x_orig = np.linspace(0, 2, n_base, endpoint=False) + 1.0 / n_base
                x_new = np.linspace(0, 2, n_try, endpoint=False) + 1.0 / n_try
                h_interp = np.interp(x_new, x_orig, best_h)
                b_start = (h_interp > 0.5).astype(float)
                # Fix balance
                n_ones = int(b_start.sum())
                if n_ones > n_try // 2:
                    one_idx = np.where(b_start == 1)[0]
                    for idx in one_idx[:n_ones - n_try // 2]:
                        b_start[idx] = 0.0
                elif n_ones < n_try // 2:
                    zero_idx = np.where(b_start == 0)[0]
                    for idx in zero_idx[:n_try // 2 - n_ones]:
                        b_start[idx] = 1.0
            elif st == 1:
                b_start = make_balanced_binary(n_try, rng)
            else:
                b_start = make_structured_binary(n_try, rng, seed_offset=st + 1)

            h_b, c5_b = binary_local_search(b_start, dx_try, ts_try,
                                             max_passes=100, time_limit=max(10, time_left() * 0.15))
            if c5_b < best_n_c5:
                best_n_c5 = c5_b
                best_n_h = h_b

        # If binary result is good, try continuous refinement
        if best_n_h is not None and best_n_c5 < best_c5 + 0.01:
            if time_left() > 30:
                h_cont, c5_cont = full_optimize(best_n_h, dx_try, ts_try, time_left, budget_phase=50)
                if c5_cont < best_n_c5:
                    best_n_c5 = c5_cont
                    best_n_h = h_cont
            if best_n_c5 < best_c5:
                best_c5 = best_n_c5
                best_h = best_n_h.copy()
                n_base = n_try

    # Phase 3: Perturbation-based multi-start on best n
    n_final = len(best_h)
    dx_final = 2.0 / n_final
    ts_final = n_final / 2.0

    n_perturb = max(1, int((time_left() - 20) / 25))
    n_perturb = min(n_perturb, 10)
    for trial in range(n_perturb):
        if time_left() < 25:
            break
        if trial % 3 == 0:
            perturb = 0.05 * (rng.random(n_final) - 0.5)
        elif trial % 3 == 1:
            perturb = 0.12 * (rng.random(n_final) - 0.5)
        else:
            idx = rng.choice(n_final, size=max(5, n_final // 10), replace=False)
            perturb = np.zeros(n_final)
            perturb[idx] = rng.choice([-1, 1], size=len(idx)) * 0.08
        h_try = project_feasible(best_h + perturb, ts_final)
        h_opt2, c5_val2 = full_optimize(h_try, dx_final, ts_final, time_left, budget_phase=40)
        if c5_val2 < best_c5:
            best_c5 = c5_val2
            best_h = h_opt2

    # Phase 4: Final intensive polish
    if time_left() > 15:
        n_b = len(best_h)
        dx_b = 2.0 / n_b
        ts_b = n_b / 2.0
        h_final, c5_final = full_optimize(best_h, dx_b, ts_b, time_left, budget_phase=50)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final

    # Phase 5: Ultra-fine pair transfer
    if time_left() > 10:
        n_b = len(best_h)
        dx_b = 2.0 / n_b
        ts_b = n_b / 2.0
        h_pt, c5_pt = pair_transfer_polish(best_h, dx_b, ts_b, n_top=25, n_iter=4,
                                           time_limit=min(10, time_left() * 0.5))
        if c5_pt < best_c5:
            best_c5 = c5_pt
            best_h = h_pt

    # Final validation
    if best_h is None:
        n = 146
        h0 = make_structured_binary(n, rng)
        best_h = project_feasible(h0, n / 2.0)
        best_c5 = c5_of_h(best_h, 2.0 / n)
    else:
        n_final = len(best_h)
        best_h = project_feasible(best_h, n_final / 2.0)
        best_c5 = c5_of_h(best_h, 2.0 / n_final)

    return best_h, best_c5, len(best_h)
