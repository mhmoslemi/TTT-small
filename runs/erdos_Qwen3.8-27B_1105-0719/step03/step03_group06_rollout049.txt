def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(-(1.0 - h))
        remaining = diff
        for i in range(n):
            idx = order[i]
            cap = 1.0 - h[idx]
            if cap <= 1e-15:
                continue
            add = min(remaining, cap)
            h[idx] += add
            remaining -= add
            if remaining <= 1e-14:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            if h[idx] <= 1e-15:
                continue
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-14:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:

            idx = int(np.argmax(h))

def coordinate_polish(h, max_iters=200, time_limit=None):

    n = len(h)
    h = h.copy()
    best_c = exact_score(h)

        g -= g.mean()

        improved = False

        order_desc = np.argsort(-g)

                        improved = True
                        break

                    j = int(order_asc[(attempt + n // 3) % n])

import numpy as np
from scipy.optimize import linprog
import time


def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(-(1.0 - h))
        remaining = diff
        for i in range(n):
            idx = order[i]
            cap = 1.0 - h[idx]
            if cap <= 1e-15:
                continue
            add = min(remaining, cap)
            h[idx] += add
            remaining -= add
            if remaining <= 1e-14:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            if h[idx] <= 1e-15:
                continue
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-14:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] = float(np.clip(h[idx] + diff2, 0.0, 1.0))
    return h


def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))


def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx


def grad_row(h, m, n):
    dx = 2.0 / n
    d = m - (n - 1)
    if d == 0:
        return dx * (1.0 - 2.0 * h)
    elif d > 0:
        g = np.zeros(n)
        g[d:] += dx * (1.0 - h[:-d])
        g[:-d] -= dx * h[d:]
        return g
    else:
        p = -d
        g = np.zeros(n)
        g[:-p] += dx * (1.0 - h[p:])
        g[p:] -= dx * h[:-p]
        return g


def interp_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    edges_old = np.linspace(0.0, 2.0, n + 1)
    edges_new = np.linspace(0.0, 2.0, new_n + 1)
    h_new = np.zeros(new_n)
    for k in range(new_n):
        a, b = edges_new[k], edges_new[k + 1]
        total = 0.0
        for j in range(n):
            lo = max(a, edges_old[j])
            hi = min(b, edges_old[j + 1])
            if hi > lo:
                total += h[j] * (hi - lo)
        h_new[k] = total / (b - a)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)


def slp_optimize(h, max_iters=400, delta0=0.06, active_tol=0.005, time_limit=None):
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    delta = delta0
    stall = 0

    for it in range(max_iters):
        if time_limit is not None and time.perf_counter() > time_limit:
            break

        P = compute_profile(h)
        c = float(P.max())
        thresh = c - active_tol
        active = np.where(P >= thresh)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(P))])
        if len(active) > 35:
            active = active[np.argsort(P[active])[-35:]]

        G = np.array([grad_row(h, m, n) for m in active])

        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0

        A_ub = np.zeros((len(active), n + 1))
        b_ub = -P[active].copy()
        for i in range(len(active)):
            A_ub[i, :n] = G[i]
            A_ub[i, n] = -1.0

        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0

        lo = np.maximum(-delta, -h)
        hi = np.minimum(delta, 1.0 - h)
        bounds = list(zip(lo, hi)) + [(None, None)]

        except Exception:

        else:

        c = float(profile.max())
        thresh = c - 0.004
        active = np.where(profile >= thresh)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(profile))])

        g = np.zeros(n)
        for m in active:
            g += grad_row(h, m, n)
        g -= g.mean()

        order_desc = np.argsort(-g)
        order_asc = np.argsort(g)

        improved = False
        fracs = [0.015, 0.008, 0.004, 0.002, 0.001]
        for frac in fracs:
            for attempt in range(min(15, n // 2)):
                i = int(order_desc[attempt % n])
                j = int(order_asc[(attempt + n // 3) % n])
                if i == j:
                    continue
                delta = frac * max(abs(h[i] - h[j]), 0.02)
                for sign in [1.0, -1.0]:
                    h_try = h.copy()
                    h_try[i] += sign * delta
                    h_try[j] -= sign * delta
                    if h_try.min() < -1e-10 or h_try.max() > 1 + 1e-10:
                        continue
                    h_try = repair_feasible(h_try)
                    s = exact_score(h_try)
                    if s < best_c - 1e-16:
                        h = h_try
                        best_c = s
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            break

    return h, best_c


def binary_seed(n, pattern_type, rng):
    half = n // 2
    h = np.zeros(n)
    if pattern_type == 0:
        idx = np.arange(n)
        vals = (idx * 0.618033988749895) % 1.0
        top = np.argsort(vals)[-half:]
        h[top] = 1.0
    elif pattern_type == 1:
        idx = np.arange(n)
        vals = (idx * 0.4142135623730951) % 1.0
        top = np.argsort(vals)[-half:]
        h[top] = 1.0
    elif pattern_type == 2:
        period = max(3, n // 8)
        for i in range(n):
            if (i // period) % 2 == 0:
                h[i] = 1.0
        ones = int(h.sum())
        if ones > half:
            excess = ones - half
            one_idx = np.where(h == 1.0)[0]
            to_flip = rng.choice(one_idx, excess, replace=False)
            h[to_flip] = 0.0
        elif ones < half:
            deficit = half - ones
            zero_idx = np.where(h == 0.0)[0]
            to_flip = rng.choice(zero_idx, deficit, replace=False)
            h[to_flip] = 1.0
    elif pattern_type == 3:
        h = (np.sin(np.linspace(0, 6 * np.pi, n)) > 0).astype(float)
        ones = int(h.sum())
        if ones > half:
            excess = ones - half
            one_idx = np.where(h == 1.0)[0]
            to_flip = rng.choice(one_idx, excess, replace=False)
            h[to_flip] = 0.0
        elif ones < half:
            deficit = half - ones
            zero_idx = np.where(h == 0.0)[0]
            to_flip = rng.choice(zero_idx, deficit, replace=False)
            h[to_flip] = 1.0
    else:
        perm = rng.permutation(n)
        h[perm[:half]] = 1.0
    return repair_feasible(h)


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s - (time.perf_counter() - t_start)

    try:
        parent = np.array(initial_h_values, dtype=np.float64)
    except Exception:
        parent = None
    if parent is None or len(parent) < 4:
        parent = np.full(70, 0.5)
    parent = np.clip(parent, 0.0, 1.0)
    parent = repair_feasible(parent)

    best_h = parent.copy()
    best_c = exact_score(parent)
    n_best = len(best_h)

    # Phase 1: SLP with moderate trust region
    deadline = t_start + budget_s * 0.18
    h_slp, c_slp = slp_optimize(best_h, max_iters=500, delta0=0.07, active_tol=0.006, time_limit=deadline)
    if c_slp < best_c:
        best_h, best_c = h_slp, c_slp

    # Phase 2: Active descent refinement
    if time_left() > 40:
        deadline = t_start + budget_s * 0.28
        h_ad, c_ad = active_descent(best_h, max_iters=300, active_tol=0.003, time_limit=deadline)
        if c_ad < best_c:
            best_h, best_c = h_ad, c_ad

    # Phase 3: SLP with smaller trust region (refinement)
    if time_left() > 60:
        deadline = t_start + budget_s * 0.42
        h_slp, c_slp = slp_optimize(best_h, max_iters=500, delta0=0.025, active_tol=0.004, time_limit=deadline)
        if c_slp < best_c:
            best_h, best_c = h_slp, c_slp

    # Phase 4: Dimension exploration
    if time_left() > 80:
        n_b = len(best_h)
        dim_list = [n_b - 4, n_b - 2, n_b + 2, n_b + 4, n_b + 6, n_b - 6, n_b + 8, n_b - 8]
        dim_list = [d for d in dim_list if 36 <= d <= 150]
        dim_list.sort(key=lambda x: abs(x - n_b))
        for new_n in dim_list[:6]:
            if time_left() < 25:
                break
            h_interp = interp_to_n(best_h, new_n)
            deadline = t_start + (budget_s - time_left() + 20)
            h_slp, c_slp = slp_optimize(h_interp, max_iters=300, delta0=0.05, active_tol=0.005, time_limit=deadline)
            if c_slp < best_c:
                best_h, best_c = h_slp, c_slp
                n_best = len(best_h)

    # Phase 5: Alternating SLP and active descent
    for rnd in range(5):
        if time_left() < 25:
            break
        deadline = t_start + (budget_s - time_left() + 22)
        h_slp, c_slp = slp_optimize(best_h, max_iters=300, delta0=0.015, active_tol=0.003, time_limit=deadline)
        if c_slp < best_c:
            best_h, best_c = h_slp, c_slp
        if time_left() > 20:
            deadline = t_start + (budget_s - time_left() + 18)
            h_ad, c_ad = active_descent(best_h, max_iters=200, active_tol=0.002, time_limit=deadline)
            if c_ad < best_c:
                best_h, best_c = h_ad, c_ad

    # Phase 6: Coordinate polish
    if time_left() > 30:
        deadline = t_start + (budget_s - time_left() + 25)
        h_cp, c_cp = coordinate_polish(best_h, max_iters=300, time_limit=deadline)
        if c_cp < best_c:
            best_h, best_c = h_cp, c_cp

    # Phase 7: SLP with very small trust region
    if time_left() > 20:
        deadline = t_start + (budget_s - time_left() + 15)
        h_slp, c_slp = slp_optimize(best_h, max_iters=400, delta0=0.004, active_tol=0.002, time_limit=deadline)
        if c_slp < best_c:
            best_h, best_c = h_slp, c_slp

    # Phase 8: Final active descent
    if time_left() > 15:
        deadline = t_start + (budget_s - time_left() + 12)
        h_ad, c_ad = active_descent(best_h, max_iters=300, active_tol=0.001, time_limit=deadline)
        if c_ad < best_c:
            best_h, best_c = h_ad, c_ad

    # Phase 9: Diversity - try blurred binary seeds
    if time_left() > 40:
        n_exp = len(best_h)
        for pt in range(4):
            if time_left() < 20:
                break
            h_bin = binary_seed(n_exp, pt, rng)
            h_smooth = 0.35 * h_bin + 0.65 * np.full(n_exp, 0.5)
            h_smooth = repair_feasible(h_smooth)
            deadline = t_start + (budget_s - time_left() + 18)
            h_slp, c_slp = slp_optimize(h_smooth, max_iters=250, delta0=0.08, active_tol=0.006, time_limit=deadline)
            if c_slp < best_c:
                best_h, best_c = h_slp, c_slp

    # Phase 10: Final fine SLP
    if time_left() > 15:
        deadline = t_start + (budget_s - time_left() + 10)
        h_slp, c_slp = slp_optimize(best_h, max_iters=400, delta0=0.003, active_tol=0.0015, time_limit=deadline)
        if c_slp < best_c:
            best_h, best_c = h_slp, c_slp

    # Phase 11: Final coordinate polish
    if time_left() > 10:
        deadline = t_start + (budget_s - time_left() + 8)
        h_cp, c_cp = coordinate_polish(best_h, max_iters=200, time_limit=deadline)
        if c_cp < best_c:
            best_h, best_c = h_cp, c_cp

    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)

    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            if float(eval_result) < final_c + 1e-12:
                final_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 2:
            if float(eval_result[0]) < final_c + 1e-12:
                final_c = float(eval_result[0])
    except Exception:
        pass

    return best_h, final_c, len(best_h)
