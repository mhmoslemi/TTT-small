    c_obj[-1] = 1.0

    A_ub[:, -1] = -1.0

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        remaining = diff
        for i in range(n):
            idx = order[i]
            if remaining > 0:
                add = min(remaining, 1.0 - h[idx])
                h[idx] += add
                remaining -= add
            else:
                rem = min(-remaining, h[idx])
                h[idx] -= rem
                remaining += rem
            if abs(remaining) < 1e-13:
                break
    return np.clip(h, 0.0, 1.0)

def build_gradient_matrix(h, n, dx):
    M = 2 * n - 1
    G = np.zeros((M, n))
    for m in range(M):
        d = m - (n - 1)
        if d == 0:
            G[m, :] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            G[m, d:] += dx * (1.0 - h[0:n-d])
            G[m, 0:n-d] += -dx * h[d:n]
        else:
            r = -d
            G[m, 0:n-r] += dx * (1.0 - h[r:n])
            G[m, r:n] += -dx * h[0:n-r]
    return G

def lp_direction(h, n, dx, D=0.3, top_k=40):
    M = 2 * n - 1
    profile = compute_profile(h)
    c_cur = float(np.max(profile))
    G = build_gradient_matrix(h, n, dx)
    
    margin = max(0.005, 0.03 * (c_cur - np.median(profile)))
    active = np.where(profile >= c_cur - margin)[0]

        active = active[idx]

    c_obj[-1] = 1.0

    A_ub[:, :n] = A_sub
    A_ub[:, -1] = -1.0
    b_ub = -P_sub
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)

    except Exception:

    return None
