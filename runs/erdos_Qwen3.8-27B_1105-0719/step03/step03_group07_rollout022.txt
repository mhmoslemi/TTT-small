import numpy as np
from scipy.optimize import linprog, minimize
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full"))) * dx

def profile_h(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def compute_gradients(h):
    """Compute gradients of all P_m with respect to h."""
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    G = np.zeros((M, n))
    
    # Lag 0
    G[n-1, :] = dx * (1.0 - 2.0 * h)
    
    # Positive lags d = 1, ..., n-1
    for d in range(1, n):
        # P_d = dx * sum_{i=0}^{n-1-d} h[i+d] * (1-h[i])
        # ∂P_d/∂h[j] = dx*(1-h[j-d]) if j >= d, minus dx*h[j+d] if j <= n-1-d
        j_range = np.arange(n)
        mask_pos = j_range >= d  # h[j] appears as h[i+d]
        mask_neg = j_range <= (n-1-d)  # h[j] appears as (1-h[i])
        
        G[n-1+d, mask_pos] += dx * (1.0 - h[np.arange(n)[mask_pos] - d])
        if np.any(mask_neg):
            G[n-1+d, mask_neg] -= dx * h[np.arange(n)[mask_neg] + d]
        
        # P_{-d} = dx * sum_{i=0}^{n-1-d} h[i] * (1-h[i+d])
        # ∂P_{-d}/∂h[j] = dx*(1-h[j+d]) if j <= n-1-d, minus dx*h[j-d] if j >= d
        G[n-1-d, mask_neg] += dx * (1.0 - h[np.arange(n)[mask_neg] + d])
        if np.any(mask_pos):
            G[n-1-d, mask_pos] -= dx * h[np.arange(n)[mask_pos] - d]
    
    return G

def slp_step(h, max_iter_lp=50):
    """One SLP step: solve LP to find direction, then line search."""
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    
    P = profile_h(h)
    G = compute_gradients(h)
    
    # LP: min t
    # s.t. G @ d + P <= t  (M constraints)
    #      -h <= d <= 1-h  (bound constraints)
    #      sum(d) = 0
    #
    # Variables: [d_0, ..., d_{n-1}, t]
    # A_ub rows: G - I_M (for G@d + P <= t => G@d - t <= -P)
    # A_eq: [ones(1,n), 0] @ [d, t] = 0
    
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize t
    
    A_ub = np.zeros((M, n + 1))
    A_ub[:, :n] = G
    A_ub[:, -1] = -1.0
    b_ub = -P
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    bounds = [(-h[i], 1.0 - h[i]) for i in range(n)] + [(-np.inf, np.inf)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.status != 0:
        return None, None
    
    d = res.x[:n]
    # Check if direction is meaningful
    if np.max(np.abs(d)) < 1e-12:
        return None, None
    
    # Line search
    best_h = h.copy()
    best_c = exact_score(h)
    
    for alpha in [1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125, 0.015625]:
        h_try = h + alpha * d
        h_try = _repair(h_try)
        c_try = exact_score(h_try)
        if c_try < best_c:
            best_c = c_try
            best_h = h_try
        else:
            break
    
    return best_h, best_c

def _repair(vec):
    n = len(vec)
    h = np.clip(vec, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx])
            h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def slp_loop(h, max_iters=30, t_start=None, budget_s=1100):
    """Iterate SLP until convergence or budget."""
    n = len(h)
    best_c = exact_score(h)
    
    for it in range(max_iters):
        if t_start is not None and time.perf_counter() - t_start > budget_s - 20:
            break
        h_new, c_new = slp_step(h)
        if h_new is None:
            break
        if c_new < best_c - 1e-13:
            h = h_new
            best_c = c_new
        elif c_new < best_c:
            h = h_new
            best_c = c_new
        else:
            # No improvement, try with smaller effective step
            # or break
            break
    return h, best_c

def make_binary_n(n, method='golden', seed=42):
    """Create binary sequence of length n with n/2 ones."""
    h = np.zeros(n)
    if method == 'golden':
        golden = (1 + np.sqrt(5)) / 2
        frac = 1.0 / golden
        positions = set()
        for i in range(n // 2):
            pos = int((i * frac) % 1.0 * n)
            positions.add(pos)
        h[list(positions)] = 1.0
    elif method == 'alternating':
        h[::2] = 1.0 if n % 2 == 0 else 1.0
        # Adjust
        h = h[:n]
        # Ensure exactly n/2 ones
        count = h.sum()
        if count != n/2:
            # Fix
            if count > n/2:
                idxs = np.where(h == 1)[0]
                for i in idxs:
                    h[i] = 0
                    if h.sum() == n/2:
                        break
            else:
                idxs = np.where(h == 0)[0]

import time

    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full"))) * dx

def profile_h(h):
    n = len(h)

def compute_gradients(h):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    G = np.zeros((M, n))
    G[n-1, :] = dx * (1.0 - 2.0 * h)
    for d in range(1, n):
        G[n-1+d, d:] += dx * (1.0 - h[:n-d])
        G[n-1+d, :n-d] -= dx * h[d:]
        G[n-1-d, :n-d] += dx * (1.0 - h[d:])
        G[n-1-d, d:] -= dx * h[:n-d]
    return G

def repair(vec):
    n = len(vec)
    h = np.clip(vec.astype(np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx])
            h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def slp_optimize(h0, max_iters=50, step_bound=0.05, t_start=None, budget_s=1100):
    """SLP minimax optimization."""
    n = len(h0)
    h = h0.copy()
    best_c = exact_score(h)
    no_improve_count = 0
    
    for it in range(max_iters):
        if t_start is not None and time.perf_counter() - t_start > budget_s - 30:
            break
        
        P = profile_h(h)
        G = compute_gradients(h)
        M = 2 * n - 1
        
        # LP: min t s.t. G@d - t <= -P, -h <= d <= 1-h (clipped to step_bound), sum(d) = 0
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        
        A_ub = np.zeros((M, n + 1))
        A_ub[:, :n] = G
        A_ub[:, -1] = -1.0
        b_ub = -P
        
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)
        
        lb = np.maximum(-h, -step_bound)
        ub = np.minimum(1.0 - h, step_bound)
        bounds = list(zip(lb, ub)) + [(-np.inf, np.inf)]
        
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        
        if res.status != 0:
            break
        
        d = res.x[:n]
        d_norm = np.max(np.abs(d))
        if d_norm < 1e-10:
            break
        
        # Line search
        improved = False
        for alpha in [1.0, 0.8, 0.6, 0.4, 0.2, 0.1, 0.05]:
            h_try = repair(h + alpha * d)
            c_try = exact_score(h_try)
            if c_try < best_c - 1e-13:
                h = h_try
                best_c = c_try
                improved = True
                break
            elif c_try < best_c:
                h = h_try
                best_c = c_try
                improved = True
                break
        
        if improved:
            no_improve_count = 0
        else:
            no_improve_count += 1
            if no_improve_count >= 3:
                break
    
    return h, best_c

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    return repair(np.clip(h_new, 0.0, 1.0))

def make_binary_n(n, seed=42):
    """Create binary sequence using golden ratio."""
    h = np.zeros(n)
    golden_frac = (np.sqrt(5) - 1) / 2
    positions = set()
    for i in range(n // 2):
        pos = int((i * golden_frac) % 1.0 * n)
        while pos in positions:
            pos = (pos + 1) % n
        positions.add(pos)
    h[list(positions)] = 1.0
    return repair(h)

def coord_descent(h, sweeps=5, t_start=None, budget_s=1100):
    """Simple coordinate descent with sum-preserving pair moves."""
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    
    for sweep in range(sweeps):
        if t_start is not None and time.perf_counter() - t_start > budget_s - 15:
            break
        for i in range(0, n, 2):
            if t_start is not None and time.perf_counter() - t_start > budget_s - 15:

            j = (i + 1) % n
            if i == j:

            h_i, h_j = h[i], h[j]

                        best_c = c_try
                        h_i, h_j = new_i, new_j
                        break
            h[i], h[j] = h_i, h_j
    
    return h, best_c

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full"))) * dx

def profile_h(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def compute_gradients(h):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    G = np.zeros((M, n))
    G[n - 1, :] = dx * (1.0 - 2.0 * h)
    for d in range(1, n):
        G[n - 1 + d, d:] += dx * (1.0 - h[:n - d])
        G[n - 1 + d, :n - d] -= dx * h[d:]
        G[n - 1 - d, :n - d] += dx * (1.0 - h[d:])
        G[n - 1 - d, d:] -= dx * h[:n - d]
    return G

def repair(vec):
    n = len(vec)
    h = np.clip(vec.astype(np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx])
            h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def slp_optimize(h0, max_iters=50, step_bound=0.05, t_start=None, budget_s=1100):
    n = len(h0)
    h = h0.copy()
    best_c = exact_score(h)
    stall = 0

    for it in range(max_iters):
        if t_start is not None and time.perf_counter() - t_start > budget_s - 30:
            break
        P = profile_h(h)
        G = compute_gradients(h)
        M = 2 * n - 1

        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        A_ub = np.zeros((M, n + 1))
        A_ub[:, :n] = G
        A_ub[:, -1] = -1.0
        b_ub = -P
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)

        lb = np.maximum(-h, -step_bound)
        ub = np.minimum(1.0 - h, step_bound)
        bounds = list(zip(lb, ub)) + [(-np.inf, np.inf)]

        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
        except Exception:
            break

        if res.status != 0:
            break

        d = res.x[:n]
        if np.max(np.abs(d)) < 1e-11:

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    return repair(np.clip(h_new, 0.0, 1.0))
