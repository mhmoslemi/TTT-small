        active = np.where(profile >= c_max - beam)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(profile))])
        if len(active) > 16:
            active = active[np.argsort(profile[active])[-16:]]

        G = build_grad(h, n, dx)
        G_a = G[active, :]
        b_a = profile[active] - G_a @ h

        h_var = cp.Variable(n)
        t_var = cp.Variable()
        target = n / 2.0
        cons = [
            h_var >= 0, h_var <= 1,
            cp.sum(h_var) == target,
            h_var >= h - trust,
            h_var <= h + trust,
            G_a @ h_var - t_var <= -b_a
        ]
        obj = cp.Minimize(t_var + 0.008 * cp.norm1(h_var - h))
        prob = cp.Problem(obj, cons)
        solved = False
        for solver in [cp.ECOS, cp.SCS, cp.CBC]:
            try:
                prob.solve(solver=solver, max_iters=8000, abstol=1e-11)
                if h_var.value is not None:
                    solved = True
                break
            except Exception:
                continue

        if solved and h_var.value is not None:
            h_sol = repair_feasible(np.array(h_var.value).flatten())
            c_sol = exact_score(h_sol)
            if c_sol < best_c - 1e-14:
                h = h_sol
                best_c = c_sol
                trust = min(trust * 1.2, 0.07)
            else:
                improved = False
                for alpha in [0.6, 0.4, 0.25, 0.15]:
                    hb = repair_feasible(alpha * h_sol + (1.0 - alpha) * h)
                    cb = exact_score(hb)
                    if cb < best_c - 1e-14:
                        h = hb
                        best_c = cb
                        trust = min(trust * 1.1, 0.07)
                        improved = True
                        break
                if not improved:
                    trust *= 0.5
                    if trust < 0.003:
                        break
        else:
            trust *= 0.45
            if trust < 0.003:
                break

    return h, best_c


def sa_optimize(h0, rng, budget_fn, time_limit, T0, T_end, max_iters):
    n = len(h0)
    h = h0.copy()
    best_h = h0.copy()
    best_c = exact_score(h0)
    curr_c = best_c
    T = T0
    t0 = time.perf_counter()

    for it in range(max_iters):
        if budget_fn() < time_limit:
            break
        if it > 0 and it % 3000 == 0:
            frac = float(it) / float(max_iters)
            T = T0 * (T_end / T0) ** (0.3 * frac + 0.7)

        mt = rng.randint(6)
        if mt == 0:
            i = rng.randint(n)
            d = rng.uniform(-0.15, 0.15)
            h_new = h.copy()
            h_new[i] = float(np.clip(h_new[i] + d, 0.0, 1.0))
            h_new = repair_feasible(h_new)
        elif mt == 1:
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            h_new = h.copy()
            h_new[i], h_new[j] = h_new[j], h_new[i]
        elif mt == 2:
            blen = rng.randint(2, min(10, n // 3))
            start = rng.randint(0, max(1, n - blen))
            h_new = h.copy()
            h_new[start:start + blen] = np.roll(h[start:start + blen], rng.randint(1, blen))
            h_new = repair_feasible(h_new)
        elif mt == 3:
            k = rng.randint(2, min(8, n // 3))
            idx = rng.choice(n, k, replace=False)
            h_new = h.copy()
            h_new[idx] = rng.uniform(0, 1, k)
            h_new = repair_feasible(h_new)
        elif mt == 4:
            noise = rng.randn(n) * 0.04
            noise -= noise.mean()
            h_new = repair_feasible(h + noise)
        else:
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            h_new = h.copy()
            d = rng.uniform(-0.08, 0.08)
            h_new[i] = float(np.clip(h_new[i] + d, 0.0, 1.0))
            h_new[j] = float(np.clip(h_new[j] - d, 0.0, 1.0))
            h_new = repair_feasible(h_new)

        if np.allclose(h_new, h, atol=1e-12):
            continue

        c_new = exact_score(h_new)
        delta = c_new - curr_c
        if delta < 0 or rng.uniform() < np.exp(-delta / max(T, 1e-10)):
            h = h_new
            curr_c = c_new
            if c_new < best_c - 1e-14:
                best_h = h.copy()
                best_c = c_new
                T = min(T * 1.08, 0.03)

    return best_h, best_c


def hybrid_refine(h0, rng, budget_fn, time_limit):
    n = len(h0)
    h = h0.copy()
    best_h = h0.copy()
    best_c = exact_score(h0)
    iterate = 0

    while budget_fn() > time_limit:
        if iterate % 5 == 0:
            hh, cc = sa_optimize(h, rng, budget_fn, time_limit, 0.02, 0.002, 5000)
            if cc < best_c - 1e-14:
                h, best_h, best_c = hh, hh, cc
        elif iterate % 5 == 1:
            hh, cc = lp_solve(h, budget_fn, time_limit, max_iters=80)
            if cc < best_c - 1e-14:
                h, best_h, best_c = hh, hh, cc
        elif iterate % 5 == 2:
            profile = compute_profile(h)
            dx = 2.0 / n
            scales = [0.005, 0.002, 0.001, 0.0005, 0.0002]
            hg, cg = grad_descent_step(h, profile, n, dx, scales)
            if hg is not None and cg < best_c - 1e-14:
                h, best_h, best_c = hg, hg, cg
        elif iterate % 5 == 3:
            for _ in range(5):
                if budget_fn() < time_limit:
                    break
                i = rng.randint(n)
                j = rng.randint(n)
                if i == j:
                    continue
                for d in [0.03, -0.03, 0.06, -0.06, 0.01, -0.01]:
                    h_try = h.copy()
                    h_try[i] = float(np.clip(h_try[i] + d, 0.0, 1.0))
                    h_try[j] = float(np.clip(h_try[j] - d, 0.0, 1.0))
                    h_try = repair_feasible(h_try)
                    c_try = exact_score(h_try)
                    if c_try < best_c - 1e-14:
                        h, best_h, best_c = h_try, h_try, c_try
                        break
                    if c_try < exact_score(h) - 1e-14:
                        h = h_try
                        break
        else:
            if budget_fn() < time_limit + 20:
                break
            eps = rng.choice([0.02, 0.04, 0.07, 0.12])
            noise = rng.randn(n) * eps
            noise -= noise.mean()
            hp = repair_feasible(h + noise)
            cp_ = exact_score(hp)
            if cp_ < best_c + 0.005:
                hh, cc = lp_solve(hp, budget_fn, time_limit, max_iters=40)
                if cc < best_c - 1e-14:
                    h, best_h, best_c = hh, hh, cc
        iterate += 1

    return best_h, best_c


def slsqp_polish(h0, maxiter, tol):
    n = len(h0)
    dx = 2.0 / n
    M = 2 * n - 1

    def obj_fn(x):
        return x[-1]

    def obj_jac_fn(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g

    def eq_fun_fn(x):
        return x[:-1].sum() - n / 2.0

    def eq_jac_fn(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g

    def ineq_fun_fn(x):
        h = x[:-1]
        return x[-1] - np.correlate(h, 1.0 - h, mode="full") * dx

    def ineq_jac_fn(x):
        h = x[:-1]
        G = build_grad(h, n, dx)
        J = np.zeros((M, n + 1))
        J[:, :-1] = -G
        J[:, -1] = 1.0
        return J

    t0 = exact_score(h0) + 1e-5
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, t0 + 1.5)]
    cons = [
        {'type': 'eq', 'fun': eq_fun_fn, 'jac': eq_jac_fn},
        {'type': 'ineq', 'fun': ineq_fun_fn, 'jac': ineq_jac_fn}
    ]
    try:
        res = minimize(obj_fn, x0, method='SLSQP', jac=obj_jac_fn,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': tol})
        if res.x is not None and np.isfinite(res.x).all():
            hs = repair_feasible(res.x[:-1])
            if np.isfinite(exact_score(hs)):
                return hs
    except Exception:
        pass
    return None


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()
    rng = np.random.RandomState(seed)

    def tl():
        return budget_s + t_start - time.perf_counter()

    parent = None
    try:
        parent = np.array(initial_h_values, dtype=np.float64)
    except (NameError, Exception):
        pass
    if parent is None or len(parent) < 20:
        parent = np.full(70, 0.5)

    parent = repair_feasible(np.clip(parent, 0.0, 1.0))
    n0 = len(parent)
    best_h = parent.copy()
    best_c = exact_score(parent)

    h_sym = repair_feasible((parent + parent[::-1]) / 2.0)
    c_sym = exact_score(h_sym)
    if c_sym < best_c - 1e-12:
        best_h, best_c = h_sym, c_sym

    # Phase 1: Dimension sweep with SA + LP
    if tl() > 200:
        dims = []
        for d in range(max(40, n0 - 12), min(120, n0 + 13), 2):
            if d % 2 == 0:
                dims.append(d)
        dims.sort(key=lambda d: abs(d - n0))

        for nd in dims[:12]:
            if tl() < 80:
                break
            hi = interp_to(best_h, nd)
            ci = exact_score(hi)
            if ci < best_c + 0.04:
                # Quick SA to explore
                hs, cs = sa_optimize(hi, rng, tl, 30, 0.04, 0.005, 8000)
                if cs < best_c - 1e-14:
                    best_h, best_c = hs, cs
                # LP refinement
                hl, cl = lp_solve(best_h if cs < ci else hi, tl, 30, max_iters=100)
                if cl < best_c - 1e-14:
                    best_h, best_c = hl, cl

    # Phase 2: Hybrid refinement on best
    if tl() > 180:
        bh, bc = hybrid_refine(best_h, rng, tl, 40)
        if bc < best_c - 1e-14:
            best_h, best_c = bh, bc

    # Phase 3: SLSQP polish
    if tl() > 60:
        hs = slsqp_polish(best_h, maxiter=600, tol=1e-13)
        if hs is not None:
            cs = exact_score(hs)
            if cs < best_c - 1e-14:
                best_h, best_c = hs, cs

    # Phase 4: Post-SLSQP LP + gradient
    if tl() > 50:
        hh, cc = lp_solve(best_h, tl, 20, max_iters=150)
        if cc < best_c - 1e-14:
            best_h, best_c = hh, cc

    # Phase 5: Try symmetric variants
    if tl() > 40:
        h_s = repair_feasible((best_h + best_h[::-1]) / 2.0)
        c_s = exact_score(h_s)
        if c_s < best_c - 1e-14:
            best_h, best_c = h_s, c_s
            hh, cc = lp_solve(best_h, tl, 15, max_iters=100)
            if cc < best_c - 1e-14:
                best_h, best_c = hh, cc

    # Phase 6: More SA at multiple temperatures
    if tl() > 80:
        for T0_val in [0.01, 0.005, 0.002]:
            if tl() < 30:
                break
            hs, cs = sa_optimize(best_h, rng, tl, 20, T0_val, T0_val * 0.1, 15000)
            if cs < best_c - 1e-14:
                best_h, best_c = hs, cs
                hh, cc = lp_solve(best_h, tl, 15, max_iters=80)
                if cc < best_c - 1e-14:
                    best_h, best_c = hh, cc

    # Phase 7: Final SLSQP
    if tl() > 50:
        hs = slsqp_polish(best_h, maxiter=800, tol=1e-14)
        if hs is not None:
            cs = exact_score(hs)
            if cs < best_c - 1e-14:
                best_h, best_c = hs, cs

    # Phase 8: Micro gradient steps
    if tl() > 15:
        n_f = len(best_h)
        dx_f = 2.0 / n_f
        h_f = best_h.copy()
        c_f = best_c
        for _ in range(200):
            if tl() < 5:
                break
            prof = compute_profile(h_f)
            cm = float(np.max(prof))
            scales = [0.001, 0.0005, 0.0002, 0.0001, 5e-5, 2e-5]
            hg, cg = grad_descent_step(h_f, prof, n_f, dx_f, scales)
            if hg is not None:
                h_f = hg
                c_f = cg
            else:
                # Try coordinate micro-adjustments
                found = False
                for _inner in range(20):
                    if tl() < 5:
                        break
                    i = rng.randint(n_f)
                    for d in [0.005, -0.005, 0.002, -0.002, 0.001, -0.001]:
                        h_try = h_f.copy()
                        h_try[i] = float(np.clip(h_try[i] + d, 0.0, 1.0))
                        h_try = repair_feasible(h_try)
                        c_try = exact_score(h_try)
                        if c_try < c_f - 1e-14:
                            h_f = h_try
                            c_f = c_try
                            found = True
                            break
                    if found:
                        break
                if not found:
                    break
        if c_f < best_c - 1e-14:
            best_h, best_c = h_f, c_f

    # Final validation
    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)

    try:
        er = evaluate_erdos_solution(best_h)
        if isinstance(er, (float, int)):
            final_c = min(final_c, float(er))
        elif isinstance(er, tuple) and len(er) >= 2:
            final_c = min(final_c, float(er[0]))
    except Exception:
        pass

    return best_h, final_c, len(best_h)
