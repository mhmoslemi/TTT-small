    best_h = h.copy()
    best_c = c

    eta = eta_init
    step_scale = 1.0

        if elapsed > budget_s - 30:

        c = float(profile.max())

            best_c = c
            best_h = h.copy()

        # Focus on constraints near the maximum
        margin = 0.01
        active_mask = profile >= c - margin
        active_idx = np.where(active_mask)[0]
        if len(active_idx) == 0:
            active_idx = np.array([np.argmax(profile)])
        
        # Update weights for active constraints
        # w_k proportional to exp(eta * P_k)
        logits = eta * profile[active_idx]

        g -= g.mean()
        
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-12:
            break
        
        # Step size: decrease as we go

        best_val = c

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(-(1.0 - h))
        remaining = diff
        for i in range(n):
            idx = order[i]
            cap = 1.0 - h[idx]
            if cap <= 1e-15:
                continue
            add = min(remaining, cap)
            h[idx] += add
            remaining -= add
            if remaining <= 1e-14:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            if h[idx] <= 1e-15:
                continue
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-14:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] = float(np.clip(h[idx] + diff2, 0.0, 1.0))
    return h

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def grad_row(h, m, n):
    dx = 2.0 / n
    d = m - (n - 1)
    if d == 0:
        return dx * (1.0 - 2.0 * h)
    elif d > 0:
        g = np.zeros(n)
        g[d:] += dx * (1.0 - h[:-d])
        g[:n-d] -= dx * h[d:]
        return g
    else:
        p = -d
        g = np.zeros(n)
        g[:n-p] += dx * (1.0 - h[p:])
        g[p:] -= dx * h[:-p]
        return g

def interp_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    edges_old = np.linspace(0.0, 2.0, n + 1)
    edges_new = np.linspace(0.0, 2.0, new_n + 1)
    h_new = np.zeros(new_n)
    for k in range(new_n):
        a, b = edges_new[k], edges_new[k + 1]
        total = 0.0
        for j in range(n):
            lo = max(a, edges_old[j])
            hi = min(b, edges_old[j + 1])
            if hi > lo:
                total += h[j] * (hi - lo)
        h_new[k] = total / (b - a)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

def active_descent(h, max_iters=300, active_tol=0.004, t_start=None, budget_s=None):
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    momentum_vec = np.zeros(n)
    momentum = 0.3
    for it in range(max_iters):
        if t_start is not None and budget_s is not None:
            if time.perf_counter() - t_start > budget_s - 20:
                break
        profile = compute_profile(h)
        c = float(profile.max())
        if c < best_c - 1e-12:
            best_c = c
        thresh = c - active_tol
        active = np.where(profile >= thresh)[0]
        if len(active) == 0:
            active = np.array([np.argmax(profile)])
        if len(active) > 25:
            top = np.argsort(profile[active])[-25:]
            active = active[top]
        g = np.zeros(n)
        for m in active:
            g += grad_row(h, m, n)
        g -= g.mean()
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-14:
            break
        momentum_vec = momentum * momentum_vec + g
        mv_norm = np.linalg.norm(momentum_vec)
        if mv_norm < 1e-15:
            break
        d = -momentum_vec / mv_norm
        pos_mask = d > 1e-12
        neg_mask = d < -1e-12
        alpha_max = 1.0
        if pos_mask.any():
            amax_pos = float(((1.0 - h[pos_mask]) / d[pos_mask]).min())
            alpha_max = min(alpha_max, amax_pos)
        if neg_mask.any():
            amax_neg = float((h[neg_mask] / (-d[neg_mask])).min())
            alpha_max = min(alpha_max, amax_neg)
        alpha_max = max(alpha_max, 1e-10)
        alpha_max = min(alpha_max, 0.5)
        best_alpha = 0.0
        best_val = c
        best_h = h
        a_ls, b_ls = 0.0, alpha_max
        for _ in range(20):
            c1 = b_ls - gr * (b_ls - a_ls)
            c2 = a_ls + gr * (b_ls - a_ls)
            h1 = repair_feasible(h + c1 * d)
            h2 = repair_feasible(h + c2 * d)
            f1 = exact_score(h1)
            f2 = exact_score(h2)
            if f1 < best_val:
                best_val = f1
                best_alpha = c1
                best_h = h1.copy()
            if f2 < best_val:
                best_val = f2
                best_alpha = c2
                best_h = h2.copy()
            if f1 < f2:
                b_ls = c2
            else:
                a_ls = c1
            if b_ls - a_ls < 1e-14:
                break
        if best_val >= c - 1e-15:
            break
        h = best_h
        if best_val < best_c:

def _slsqp_obj(x):
    return x[-1]

def _slsqp_obj_jac(x):
    n = len(x) - 1
    g = np.zeros(n + 1)
    g[-1] = 1.0
    return g

def _slsqp_eq_fun(x):
    n = len(x) - 1
    return x[:-1].sum() - n / 2.0

def _slsqp_eq_jac(x):
    n = len(x) - 1
    g = np.zeros(n + 1)
    g[:-1] = 1.0
    return g

def _slsqp_ineq_fun(x):
    h = x[:-1]
    t = x[-1]
    n = len(h)
    dx = 2.0 / n
    profile = np.correlate(h, 1 - h, mode="full") * dx
    return t - profile

def _slsqp_ineq_jac(x):
    h = x[:-1]
    n = len(h)
    M = 2 * n - 1
    J = np.zeros((M, n + 1))
    for m in range(M):
        J[m, :n] = -grad_row(h, m, n)
    J[:, -1] = 1.0
    return J

def slsqp_refine(h_start, maxiter=150):
    n = len(h_start)
    c0 = exact_score(h_start)
    x0 = np.concatenate([h_start, [c0 + 1e-4]])
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 2.0)]
    cons = [
        {'type': 'eq', 'fun': _slsqp_eq_fun, 'jac': _slsqp_eq_jac},
        {'type': 'ineq', 'fun': _slsqp_ineq_fun, 'jac': _slsqp_ineq_jac}
    ]
    try:
        res = minimize(_slsqp_obj, x0, method='SLSQP', jac=_slsqp_obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None and np.isfinite(res.x).all():
            h_new = repair_feasible(res.x[:-1])
            c_new = exact_score(h_new)
            if c_new <= c0 + 1e-8:
                return h_new
    except Exception:
        pass
    return None

def binary_seed(n, pattern_type, rng):
    half = n // 2
    h = np.zeros(n)
    if pattern_type == 0:
        idx = np.arange(n)
        vals = (idx * 0.618033988749895) % 1.0
        top = np.argsort(vals)[-half:]
        h[top] = 1.0
    elif pattern_type == 1:
        idx = np.arange(n)
        vals = (idx * 0.4142135623730951) % 1.0
        top = np.argsort(vals)[-half:]
        h[top] = 1.0
    elif pattern_type == 2:
        period = max(3, n // 8)
        for i in range(n):
            if (i // period) % 2 == 0:
                h[i] = 1.0
        ones = int(h.sum())
        if ones > half:
            excess = ones - half
            one_idx = np.where(h == 1.0)[0]
            to_flip = rng.choice(one_idx, excess, replace=False)
            h[to_flip] = 0.0
        elif ones < half:
            deficit = half - ones
            zero_idx = np.where(h == 0.0)[0]
            to_flip = rng.choice(zero_idx, deficit, replace=False)
            h[to_flip] = 1.0
    elif pattern_type == 3:
        h = (np.sin(np.linspace(0, 6 * np.pi, n)) > 0).astype(float)
        ones = int(h.sum())
        if ones > half:
            excess = ones - half
            one_idx = np.where(h == 1.0)[0]
            to_flip = rng.choice(one_idx, excess, replace=False)
            h[to_flip] = 0.0
        elif ones < half:
            deficit = half - ones
            zero_idx = np.where(h == 0.0)[0]
            to_flip = rng.choice(zero_idx, deficit, replace=False)
            h[to_flip] = 1.0
    elif pattern_type == 4:
        idx = np.arange(n)
        vals = (idx * 0.25 + 0.1 * np.sin(idx * 0.7)) % 1.0
        top = np.argsort(vals)[-half:]
        h[top] = 1.0
    else:
        perm = rng.permutation(n)
        h[perm[:half]] = 1.0
    return repair_feasible(h)

def lm_newton_step(h, active, weights, n, lam=0.01):
    dx = 2.0 / n
    g = np.zeros(n)
    H = np.zeros((n, n))
    for i in range(len(active)):
        m = int(active[i])
        w = weights[i]
        g += w * grad_row(h, m, n)
        d = m - (n - 1)
        if d == 0:
            H[np.diag_indices(n)] += -2.0 * dx * w
        else:
            p = abs(d)
            for j in range(n - p):
                if d > 0:
                    H[j, j + p] += -dx * w
                    H[j + p, j] += -dx * w
                else:
                    H[j, j + p] += -dx * w
                    H[j + p, j] += -dx * w
    g -= g.mean()
    A = H + lam * np.eye(n)
    A -= np.outer(np.ones(n), A.mean(axis=1)) / n
    A -= np.outer(A.mean(axis=0), np.ones(n)) / n
    A += np.eye(n) * 1e-10
    try:
        delta = np.linalg.solve(A, -g)
    except np.linalg.LinAlgError:
        delta = -g / max(np.linalg.norm(g), 1e-15)
    delta -= delta.mean()
    return delta, g

def newton_polish(h_start, max_iters=50, t_start=None, budget_s=None):

        pos_mask = delta > 1e-12

def coordinate_mix(h, other_h, t_start=None, budget_s=None):
    n = len(h)
    if len(other_h) != n:
        return h
    best_c = exact_score(h)
    best_h = h.copy()
    for alpha in [0.1, 0.2, 0.3, 0.5, 0.7]:

        val = exact_score(h_try)
        if val < best_c - 1e-10:

            best_h = h_try
    return best_h
