import numpy as np
from scipy.optimize import minimize, linprog
from scipy.spatial.distance import cdist
import time

def repair_feasible(vec, target=None):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    if target is None:
        target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        order = np.argsort(-(1.0 - h))
        remaining = diff
        for i in range(n):
            idx = order[i]
            cap = 1.0 - h[idx]
            if cap <= 1e-15:
                continue
            add = min(remaining, cap)
            h[idx] += add
            remaining -= add
            if remaining <= 1e-13:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            if h[idx] <= 1e-15:
                continue
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-13:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-14:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] = float(np.clip(h[idx] + diff2, 0.0, 1.0))
    return h

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def build_gradient_matrix(h):
    """Build full M x N gradient matrix."""
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    G = np.zeros((M, n))
    for m in range(M):
        d = m - (n - 1)
        if d == 0:
            G[m, :] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            G[m, d:] += dx * (1.0 - h[:-d])
            G[m, :-d] -= dx * h[d:]
        else:
            p = -d
            G[m, :-p] += dx * (1.0 - h[p:])
            G[m, p:] -= dx * h[:-p]
    return G

def lp_trust_step(h, delta=0.03):
    """Solve LP for trust region step. Returns (d, tau) or None."""
    n = len(h)
    M = 2 * n - 1
    P = compute_profile(h)
    G = build_gradient_matrix(h)
    
    # Variables: d[0..n-1], tau
    # Minimize tau
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    # Constraints: G[m]*d - tau <= -P[m], i.e., G[m]*d + (-1)*tau <= -P[m]
    A_ub = np.zeros((M, n + 1))
    A_ub[:, :n] = G
    A_ub[:, -1] = -1.0
    b_ub = -P
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds on d[i]: max(-delta, -h[i]) <= d[i] <= min(delta, 1-h[i])
    lo = np.maximum(-delta, -h)
    hi = np.minimum(delta, 1.0 - h)
    bounds = list(zip(lo, hi))
    # tau: unconstrained below, upper bounded by current max + delta
    bounds.append((-delta, P.max() + delta))
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x is not None:
            return res.x[:n], float(res.x[-1])
    except Exception:
        pass
    return None

def lp_trust_region(h, max_iters=200, delta_init=0.03, delta_min=0.002,
                    time_limit=None, t_start=None):
    """LP-based trust region optimizer."""
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    delta = delta_init
    stall = 0
    
    for it in range(max_iters):
        if time_limit is not None and t_start is not None:
            if time.perf_counter() - t_start > time_limit:
                break
        
        result = lp_trust_step(h, delta)
        if result is None:
            delta *= 0.5
            stall += 1
            if delta < delta_min or stall > 15:
                break
            continue
        
        d, tau_pred = result
        
        # Check if step is meaningful
        norm_d = np.linalg.norm(d)
        if norm_d < 1e-12:
            delta *= 0.5
            stall += 1
            if delta < delta_min or stall > 15:
                break
            continue
        
        # Line search with backtracking
        best_alpha = 0.0
        best_h_try = h.copy()
        best_val = exact_score(h)
        
        for alpha in [1.0, 0.7, 0.5, 0.35, 0.2, 0.1]:
            h_try = repair_feasible(h + alpha * d)
            c_try = exact_score(h_try)
            if c_try < best_val - 1e-14:
                best_val = c_try
                best_alpha = alpha
                best_h_try = h_try
                if alpha >= 0.5:

        if best_alpha > 0 and best_val < best_c - 1e-13:
            h = best_h_try
            best_c = best_val
            stall = 0
            # Increase delta if full step worked
            if best_alpha >= 0.7:
                delta = min(delta * 1.3, 0.15)
        else:
            stall += 1
            delta *= 0.7
            if delta < delta_min or stall > 20:
                break
    
    return h, best_c

import numpy as np

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        order = np.argsort(-(1.0 - h))
        remaining = diff
        for i in range(n):
            idx = int(order[i])
            cap = 1.0 - h[idx]
            if cap <= 1e-15:
                continue
            add = min(remaining, cap)
            h[idx] += add
            remaining -= add
            if remaining <= 1e-13:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = int(order[i])
            if h[idx] <= 1e-15:
                continue
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-13:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-14:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] = float(np.clip(h[idx] + diff2, 0.0, 1.0))
    return h

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def grad_row(h, m):
    n = len(h)
    dx = 2.0 / n
    d = m - (n - 1)
    g = np.zeros(n)
    if d == 0:
        g[:] = dx * (1.0 - 2.0 * h)
    elif d > 0:
        g[d:] += dx * (1.0 - h[:-d])
        g[:-d] -= dx * h[d:]
    else:
        p = -d
        g[:-p] += dx * (1.0 - h[p:])
        g[p:] -= dx * h[:-p]
    return g

def build_grad_matrix(h):
    n = len(h)
    M = 2 * n - 1
    G = np.zeros((M, n))
    for m in range(M):
        G[m] = grad_row(h, m)
    return G

def interp_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    edges_old = np.linspace(0.0, 2.0, n + 1)
    edges_new = np.linspace(0.0, 2.0, new_n + 1)
    h_new = np.zeros(new_n)
    for k in range(new_n):
        a, b = edges_new[k], edges_new[k + 1]
        total = 0.0
        for j in range(n):
            lo = max(a, edges_old[j])
            hi = min(b, edges_old[j + 1])
            if hi > lo:
                total += h[j] * (hi - lo)
        h_new[k] = total / (b - a)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

def lp_step(h, delta=0.05, all_constraints=True):
    """Solve LP for trust region direction."""
    n = len(h)
    M = 2 * n - 1
    P = compute_profile(h)
    G = build_grad_matrix(h)
    
    if not all_constraints:
        C = P.max()
        active = np.where(P >= C - 0.01)[0]
        if len(active) < 3:
            active = np.argsort(-P)[:20]
        G_use = G[active, :]
        P_use = P[active]
    else:
        G_use = G
        P_use = P
    
    M_use = G_use.shape[0]
    
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    A_ub = np.zeros((M_use, n + 1))
    A_ub[:, :n] = G_use
    A_ub[:, -1] = -1.0
    b_ub = -P_use
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    lo = np.maximum(-delta, -h)
    hi = np.minimum(delta, 1.0 - h)
    bounds = list(zip(lo.tolist(), hi.tolist()))
    bounds.append((-(P.max() + delta), P.max() + delta))
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x is not None:
            return res.x[:n], float(res.x[-1])
    except Exception:
        pass
    return None, None

def active_descent(h, max_iters=200, active_tol=0.003, t_start=None, time_limit=None):
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    momentum_vec = np.zeros(n)
    momentum = 0.4
    
    for it in range(max_iters):

            if time.perf_counter() - t_start > time_limit:
                break
        
        profile = compute_profile(h)
        c = float(profile.max())
        thresh = c - active_tol

        if len(active) == 0:
            active = np.array([int(np.argmax(profile))])
        if len(active) > 25:
            top = np.argsort(profile[active])[-25:]
            active = active[top]

            g += grad_row(h, int(m))
        g -= g.mean()
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            break
        
        momentum_vec = momentum * momentum_vec + g
        mv_norm = np.linalg.norm(momentum_vec)
        if mv_norm < 1e-15:

        neg_mask = d < -1e-12
        alpha_max = 0.5

            alpha_max = min(alpha_max, float(((1.0 - h[pos_mask]) / d[pos_mask]).min()))
        if neg_mask.any():

        alpha_max = max(alpha_max, 1e-10)

        best_alpha = 0.0
        best_val = c
        best_h = h
        
        for _ in range(20):

            c2 = a_ls + gr * (b_ls - a_ls)
            h1 = repair_feasible(h + c1 * d)
            h2 = repair_feasible(h + c2 * d)
            f1 = exact_score(h1)
            f2 = exact_score(h2)
            if f1 < best_val:

            if f2 < best_val:

                a_ls = c1

    """Generate a golden-ratio based binary seed."""
    half = n // 2
    idx = np.arange(n)
    vals = (idx * 0.618033988749895) % 1.0
    top = np.argsort(vals)[-half:]
    h = np.zeros(n)
    h[top] = 1.0
    h[~np.isin(idx, top)] = 0.0

def sine_seed(n, freq, phase, rng):
    """Generate a sine-wave threshold seed."""
    x = np.linspace(0, freq * 2 * np.pi, n) + phase
    h = (np.sin(x) > 0).astype(float)
    ones = int(h.sum())
    half = n // 2
    if ones > half:
        excess = ones - half
        one_idx = np.where(h == 1.0)[0]
        to_flip = rng.choice(one_idx, excess, replace=False)
        h[to_flip] = 0.0
    elif ones < half:
        deficit = half - ones
        zero_idx = np.where(h == 0.0)[0]
        to_flip = rng.choice(zero_idx, deficit, replace=False)
        h[to_flip] = 1.0
    return repair_feasible(h)
