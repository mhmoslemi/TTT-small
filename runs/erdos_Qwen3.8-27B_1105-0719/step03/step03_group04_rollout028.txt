    p_a = profile[active_set]
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_ub = np.zeros((M, n + 1))
    A_ub[:, :n] = G_a
    A_ub[:, -1] = -1.0
    b_ub = -p_a
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    lower = np.maximum(-h, -radius)
    upper = np.minimum(1.0 - h, radius)
    bounds = list(zip(lower, upper))
    bounds.append((0.0, None))
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'maxiter': 5000})
        if res.success and res.x is not None and np.isfinite(res.x).all():
            step = res.x[:n].copy()
            step[step < -1e-15] = 0.0
            step[step > 1e-15] = step[step > 1e-15]
            norm = np.linalg.norm(step, np.inf)
            if norm < 1e-12:
                return None, res.x[-1]
            direction = step / norm
            return direction, res.x[-1]
    except Exception:
        pass
    return None, None

def adaptive_line_search(h, direction, current_score, max_step=1.0):
    n = len(h)
    best_h = h.copy()
    best_s = current_score
    steps = [max_step, max_step*0.7, max_step*0.5, max_step*0.3,
             max_step*0.15, max_step*0.08, max_step*0.04, max_step*0.02]
    for a in steps:
        h_try = h + a * direction
        h_try = repair_feasible(h_try)
        s = exact_score(h_try)
        if s < best_s - 1e-14:
            best_s = s
            best_h = h_try
    return best_h, best_s

def gradient_direction(h, active_set):
    n = len(h)
    dx = 2.0 / n
    G = build_gradient(h, n, dx)
    profile = compute_profile(h)
    T = np.max(profile)
    G_a = G[active_set, :]
    p_a = profile[active_set]
    weights = np.ones(len(active_set)) / len(active_set)
    grad = G_a.T @ weights
    grad -= grad.mean()
    norm = np.linalg.norm(grad)
    if norm < 1e-15:
        return None
    return grad / norm

def kkt_check(h, active_set):
    n = len(h)
    dx = 2.0 / n
    G = build_gradient(h, n, dx)
    G_a = G[active_set, :]
    G_proj = G_a - G_a.mean(axis=1, keepdims=True)
    m = len(active_set)
    if m <= 1:
        return float(np.max(np.abs(G_proj.T))) if m == 1 else 0.0
    lam = np.ones(m) / m
    r = G_proj.T @ lam
    return float(np.linalg.norm(r))

def optimize_lp(h, t_start, budget_s, max_iters=1000):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c = exact_score(h)
    radius = 0.1
    no_improve = 0
    for it in range(max_iters):
        if time.perf_counter() - t_start > budget_s - 15:
            break
        profile = compute_profile(h)
        c = np.max(profile)
        if c < best_c - 1e-14:
            best_c = c
            best_h = h.copy()
            no_improve = 0
            radius = min(radius * 1.3, 0.2)
        else:
            no_improve += 1
        if no_improve > 80:
            break
        active = get_active_set(profile)
        improved = False
        for _attempt in range(3):
            direction, t_lp = lp_direction(h, active, radius)
            if direction is None:
                break
            h_new, s_new = adaptive_line_search(h, direction, c, max_step=radius)
            if s_new < best_c - 1e-14:
                h = h_new
                improved = True
                break
            if _attempt < 2:
                radius *= 0.5
                active = get_active_set(compute_profile(h), threshold_frac=0.05, max_count=10)
        if improved:
            continue
        grad = gradient_direction(h, active)
        if grad is not None:
            h_new, s_new = adaptive_line_search(h, grad, c, max_step=0.03)
            if s_new < best_c - 1e-14:
                h = h_new
                improved = True
        if improved:
            continue
        if no_improve > 10:
            active2 = get_active_set(profile, threshold_frac=0.08, max_count=24)
            direction2, _ = lp_direction(h, active2, radius * 0.5)
            if direction2 is not None:
                h_new, s_new = adaptive_line_search(h, direction2, c, max_step=radius*0.5)
                if s_new < best_c - 1e-14:
                    h = h_new
                    improved = True
        if not improved:
            radius *= 0.7
            if radius < 0.002:
                break
    return best_h, best_c

def slsqp_polish(h, t_start, budget_s):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    c0 = exact_score(h)
    def obj_fn(x):
        return x[-1]
    def obj_jac_fn(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    def eq_fn(x):
        return np.array([x[:-1].sum() - n / 2.0])
    def eq_jac_fn(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g[:, np.newaxis]
    def ineq_fn(x):
        h_val = x[:-1]
        t_val = x[-1]
        prof = np.correlate(h_val, 1 - h_val, mode="full") * dx
        return t_val - prof
    def ineq_jac_fn(x):
        h_val = x[:-1]
        G = build_gradient(h_val, n, dx)
        J = np.zeros((M, n + 1))
        J[:, :-1] = -G
        J[:, -1] = 1.0
        return J
    x0 = np.concatenate([h, [c0 + 1e-6]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    cons = [
        {'type': 'eq', 'fun': eq_fn, 'jac': eq_jac_fn},
        {'type': 'ineq', 'fun': ineq_fn, 'jac': ineq_jac_fn}
    ]
    max_iter = 800 if n <= 80 else 400
    try:
        res = minimize(obj_fn, x0, method='SLSQP', jac=obj_jac_fn,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-16, 'disp': False})
        if res.x is not None and np.isfinite(res.x).all():
            h_sol = repair_feasible(res.x[:-1])
            c_sol = exact_score(h_sol)
            return h_sol, c_sol
    except Exception:
        pass
    return h, c0

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

def pair_polish(h, t_start, budget_s, max_rounds=50):
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    for rnd in range(max_rounds):
        if time.perf_counter() - t_start > budget_s - 5:
            break
        profile = compute_profile(h)
        c = np.max(profile)
        dx = 2.0 / n
        improved = False
        hi_idx = np.argsort(h)[-15:]
        lo_idx = np.argsort(h)[:15]
        best_move_score = best_c
        best_move = None
        for i in hi_idx:
            for j in lo_idx:
                if i == j:
                    continue
                if h[i] - h[j] < 0.05:
                    continue
                alpha_max_v = h[i] * 0.4
                for frac in [0.3, 0.5, 0.7, 1.0]:
                    alpha = alpha_max_v * frac
                    h_try = h.copy()
                    h_try[i] -= alpha
                    h_try[j] += alpha
                    h_try = repair_feasible(h_try)
                    s = exact_score(h_try)
                    if s < best_move_score - 1e-12:
                        best_move_score = s
                        best_move = h_try.copy()
        if best_move is not None:
            h = best_move
            best_c = best_move_score
            improved = True
        if not improved:
            break
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s - (time.perf_counter() - t_start)

    parent = None
    try:
        parent = np.array(initial_h_values, dtype=np.float64)
    except (NameError, Exception):
        pass
    if parent is None or len(parent) < 4 or not np.isfinite(parent).all():
        parent = np.full(70, 0.5)
    parent = np.clip(parent, 0.0, 1.0)
    parent = repair_feasible(parent)
    best_h = parent.copy()
    best_c = exact_score(parent)

    h_sym = repair_feasible((parent + parent[::-1]) / 2.0)
    c_sym = exact_score(h_sym)
    if c_sym < best_c - 1e-12:
        best_h = h_sym
        best_c = c_sym

    n0 = len(best_h)

    # Phase 1: LP optimization at parent dimension
    if time_left() > 120:
        h_opt, c_opt = optimize_lp(best_h, t_start, t_start + min(180, time_left() - 10), max_iters=800)
        if c_opt < best_c - 1e-14:
            best_h = h_opt
            best_c = c_opt

    # Phase 2: SLSQP at parent dimension
    if time_left() > 100:
        budget_for_slsqp = t_start + min(80, time_left() - 10)
        h_pol, c_pol = slsqp_polish(best_h, t_start, budget_for_slsqp)
        if c_pol < best_c - 1e-14:
            best_h = h_pol
            best_c = c_pol

    # Phase 3: Pair polish
    if time_left() > 60:
        budget_pp = t_start + min(40, time_left() - 10)
        h_pp, c_pp = pair_polish(best_h, t_start, budget_pp, max_rounds=40)
        if c_pp < best_c - 1e-14:
            best_h = h_pp
            best_c = c_pp

    # Phase 4: Try different dimensions
    n_b = len(best_h)
    dims = []
    for d in [n_b-10, n_b-5, n_b+5, n_b+10, n_b+15, n_b+20, n_b+30, n_b*2]:
        if 24 <= d <= 250 and d != n_b and d % 2 == 0:
            dims.append(d)
    for d in range(30, 200, 10):
        if d not in dims and d != n_b:
            dims.append(d)
    dims = list(set(dims))
    dims.sort(key=lambda d: abs(d - n_b))

    for new_n in dims[:10]:
        if time_left() < 80:
            break
        h_interp = interpolate_to_n(best_h, new_n)
        phase_budget = t_start + min(time_left()() if callable(time_left()) else time_left(), 90) - 5
        h_opt2, c_opt2 = optimize_lp(h_interp, t_start, phase_budget, max_iters=400)
        if c_opt2 < best_c - 1e-14:
            best_h = h_opt2
            best_c = c_opt2
            if time_left() > 60:
                slsqp_budget = t_start + min(50, time_left() - 5)
                h_sl, c_sl = slsqp_polish(best_h, t_start, slsqp_budget)
                if c_sl < best_c - 1e-14:
                    best_h = h_sl
                    best_c = c_sl

    # Phase 5: Re-optimize at best dimension with fresh LP
    if time_left() > 80:
        n_final = len(best_h)
        h_fresh = best_h + 0.02 * rng.randn(n_final)
        h_fresh = repair_feasible(h_fresh)
        phase_budget = t_start + min(60, time_left() - 10)
        h_opt3, c_opt3 = optimize_lp(h_fresh, t_start, phase_budget, max_iters=300)
        if c_opt3 < best_c - 1e-14:
            best_h = h_opt3
            best_c = c_opt3

    # Phase 6: Final SLSQP
    if time_left() > 50:
        slsqp_budget = t_start + min(40, time_left() - 5)
        h_final, c_final = slsqp_polish(best_h, t_start, slsqp_budget)
        if c_final < best_c - 1e-14:
            best_h = h_final
            best_c = c_final

    # Phase 7: Final pair polish
    if time_left() > 20:
        pp_budget = t_start + min(15, time_left() - 2)
        h_fp, c_fp = pair_polish(best_h, t_start, pp_budget, max_rounds=20)
        if c_fp < best_c - 1e-14:
            best_h = h_fp
            best_c = c_fp

    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)

    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            final_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 2:
            final_c = float(eval_result[0])
    except Exception:
        pass

    return best_h, final_c, len(best_h)
