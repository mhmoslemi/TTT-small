        else:
            hi = mid

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

    dx = 2.0 / n

    m = 2 * n - 1
    G = np.zeros((m, n))
    G[n - 1, :] = 1.0 - 2.0 * h
    for d in range(1, n):
        G[n - 1 + d, d:] += 1.0 - h[:n - d]
        G[n - 1 + d, :n - d] += -h[d:]
        G[n - 1 - d, :n - d] += 1.0 - h[d:]
        G[n - 1 - d, d:] += -h[:n - d]
    return G

    return x[0]

def _slsqp_obj_jac(x):
    return np.concatenate(([1.0], np.zeros(len(x) - 1)))

    h = x[1:]
    return x[0] - _G_DX * compute_corr(h)

    h = x[1:]
    n = _G_N
    G = all_corr_grads(h, n)
    J = np.empty((2 * n - 1, n + 1))
    J[:, 0] = 1.0
    J[:, 1:] = -_G_DX * G
    return J

    return x[1:].sum() - _G_TS

    global _G_N, _G_DX, _G_TS
    n = len(h0)
    _G_N = n
    _G_DX = dx
    _G_TS = target_sum
    
    x0 = np.concatenate(([t_init], h0))
    bounds = [(0.001, 0.6)] + [(0.0, 1.0)] * n
    
    cons = [
        {'type': 'ineq', 'fun': _slsqp_ineq, 'jac': _slsqp_ineq_jac},
        {'type': 'eq', 'fun': _slsqp_eq, 'jac': _slsqp_eq_jac},
    ]
    
    try:
        res = minimize(_slsqp_obj, x0, method='SLSQP', jac=_slsqp_obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol, 'maxfun': maxfun})

        h_new = project_feasible(h_new, target_sum)
        return h_new
