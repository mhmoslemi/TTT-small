import numpy as np
from scipy.optimize import minimize, linprog

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx] + 1e-15)
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx] + 1e-15)
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-14:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmin(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def build_gradient_matrix(h, n, dx):
    M = 2 * n - 1
    G = np.zeros((M, n))
    for m in range(M):
        d = m - (n - 1)
        if d == 0:
            G[m, :] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            G[m, d:] += dx * (1.0 - h[0:n-d])
            G[m, 0:n-d] += -dx * h[d:n]
        else:
            dp = -d
            G[m, 0:n-dp] += dx * (1.0 - h[dp:n])
            G[m, dp:n] += -dx * h[0:n-dp]
    return G

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = np.linspace(0, 2, n + 1)
    xt = np.linspace(0, 2, new_n + 1)
    h_interp = np.zeros(new_n)
    for k in range(new_n):
        lo_k, hi_k = xt[k], xt[k+1]
        total_mass = 0.0
        total_width = 0.0
        for j in range(n):
            lo_j, hi_j = xs[j], xs[j+1]
            overlap = min(hi_k, hi_j) - max(lo_k, lo_j)
            if overlap > 0:
                total_mass += h[j] * overlap
                total_width += overlap
        h_interp[k] = total_mass / total_width if total_width > 1e-15 else 0.5
    h_interp = np.clip(h_interp, 0.0, 1.0)
    return repair_feasible(h_interp)

def symmetrize(h):
    return (h + h[::-1]) / 2.0

def lse_descent(h, beta=80.0, max_iters=200, alpha_init=0.004):
    """Log-sum-exp smoothing descent"""
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best_c = exact_score(h)
    alpha = alpha_init
    stuck = 0
    for it in range(max_iters):
        profile = compute_profile(h)
        scaled = beta * profile
        max_s = np.max(scaled)
        shifted = scaled - max_s
        weights = np.exp(shifted)
        wsum = weights.sum()
        if wsum < 1e-300:
            break
        weights /= wsum
        G = build_gradient_matrix(h, n, dx)
        grad = G.T @ weights
        grad -= grad.mean()
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad /= gnorm
        improved = False
        for a in [alpha, alpha*0.5, alpha*0.25, alpha*0.1]:
            h_try = repair_feasible(h - a * grad)
            s = exact_score(h_try)
            if s < best_c - 1e-14:
                h = h_try
                best_c = s
                improved = True
                alpha = min(alpha * 1.1, 0.02)
                stuck = 0
                break
            elif s < float(np.max(compute_profile(h))) - 1e-14:
                h = h_try
                improved = True
                break
        if not improved:
            alpha *= 0.5
            stuck += 1
            if stuck > 30 or alpha < 1e-9:
                break
    return h, best_c

def active_set_descent(h, max_iters=300, n_active=12, alpha_init=0.005):

        gnorm = np.linalg.norm(grad)

            break

def lp_minimax_direction(h, D=0.5):
    """Solve the linearized minimax LP to find best descent direction.
    
    min S
    s.t. P_m + G_m @ d <= S  for all m
         sum(d) = 0
         -D <= d_i <= D
    """
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    P = compute_profile(h)
    G = build_gradient_matrix(h, n, dx)
    
    # Variables: d (n vars) + S (1 var)
    # Minimize: S
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    # Inequalities: G @ d - S <= -P  =>  G @ d + (-1)*S <= -P
    # In standard form A_ub @ x <= b_ub
    A_ub = np.zeros((M, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -P
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds: -D <= d_i <= D, S free (or bounded)
    bounds = [(-D, D)] * n + [(-10.0, 10.0)]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            d = res.x[:n]
            S = res.x[n]
            return d, float(S)
    except Exception:
        pass
    
    # Fallback: try with active constraints only
    c = float(np.max(P))
    margin = max(0.005, 0.1 * (c - float(np.min(P))))
    active = np.where(P >= c - margin)[0]
    if len(active) < 3:
        active = np.argsort(P)[-10:]
    
    A_ub2 = np.zeros((len(active), n + 1))
    A_ub2[:, :n] = G[active, :]
    A_ub2[:, n] = -1.0
    b_ub2 = -P[active]
    
    try:
        res = linprog(c_obj, A_ub=A_ub2, b_ub=b_ub2, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            d = res.x[:n]
            S = res.x[n]
            return d, float(S)
    except Exception:
        pass
    
    return None, None

def line_search_pair(h, i, j, n_samples=40):
    """Find best pair move: shift mass from j to i.
    h_i -> h_i + delta, h_j -> h_j - delta
    Valid range: delta in [-h_j, 1 - h_i]
    """
    n = len(h)
    best_c = exact_score(h)
    delta_lo = -h[j]
    delta_hi = 1.0 - h[i]
    
    if delta_hi - delta_lo < 1e-12:
        return h.copy(), best_c
    
    # Sample the interval
    deltas = np.linspace(delta_lo, delta_hi, n_samples)
    best_delta = 0.0
    best_local_c = best_c
    
    for d in deltas:
        h_try = h.copy()
        h_try[i] += d
        h_try[j] -= d
        if h_try[i] < -1e-12 or h_try[i] > 1 + 1e-12 or h_try[j] < -1e-12 or h_try[j] > 1 + 1e-12:
            continue
        h_try = np.clip(h_try, 0, 1)
        c_try = exact_score(h_try)
        if c_try < best_local_c:
            best_local_c = c_try
            best_delta = d
    
    if best_delta != 0.0:
        # Refine with golden section
        # Find bracket around best_delta
        step = (delta_hi - delta_lo) / n_samples
        lo = max(delta_lo, best_delta - step)
        hi = min(delta_hi, best_delta + step)
        
        for _ in range(20):
            m1 = lo + 0.382 * (hi - lo)
            m2 = lo + 0.618 * (hi - lo)
            h1 = h.copy(); h1[i] += m1; h1[j] -= m1
            h2 = h.copy(); h2[i] += m2; h2[j] -= m2
            h1 = np.clip(h1, 0, 1); h2 = np.clip(h2, 0, 1)
            c1 = exact_score(h1)
            c2 = exact_score(h2)
            if c1 < c2:
                hi = m2
            else:
                lo = m1
        
        best_delta = (lo + hi) / 2
        h_try = h.copy()
        h_try[i] += best_delta
        h_try[j] -= best_delta
        h_try = np.clip(h_try, 0, 1)
        h_try = repair_feasible(h_try)
        best_local_c = exact_score(h_try)
        if best_local_c < best_c:
            return h_try, best_local_c
    
    return h.copy(), best_c

def stochastic_kick(h, rng, amplitude=0.02):
    """Random perturbation preserving feasibility."""
    n = len(h)
    z = rng.randn(n)
    z -= z.mean()
    z /= np.linalg.norm(z) + 1e-15
    h_new = repair_feasible(h + amplitude * z)
    return h_new

                        best_local = c_try
                        best_delta = d
                if best_delta != 0.0 and best_local < best_c - 1e-14:
                    step = (d_hi - d_lo) / n_samples
                    lo = max(d_lo, best_delta - step)
                    hi = min(d_hi, best_delta + step)
                    for _ in range(25):
                        m1 = lo + 0.382 * (hi - lo)
                        m2 = lo + 0.618 * (hi - lo)
                        ht1 = h.copy(); ht1[i] += m1; ht1[j] -= m1
                        ht2 = h.copy(); ht2[i] += m2; ht2[j] -= m2
                        c1 = exact_score(ht1)
                        c2 = exact_score(ht2)
                        if c1 < c2:
                            hi = m2
                        else:
                            lo = m1
                    best_delta = (lo + hi) / 2.0
                    h_new = h.copy()
                    h_new[i] += best_delta
                    h_new[j] -= best_delta
                    h_new = repair_feasible(h_new)
                    c_new = exact_score(h_new)
                    if c_new < best_c:
                        h = h_new
                        best_c = c_new
                        improved = True
                        break
            if improved:
                break
        if not improved:
            break
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()
    rng = np.random.RandomState(seed)
    np.random.seed(seed)

    def time_left():
        return budget_s + t_start - time.perf_counter()

    parent = None
    try:
        parent = np.array(initial_h_values, dtype=np.float64)
    except NameError:
        pass
    except Exception:
        pass
    if parent is None or len(parent) < 4:
        parent = np.full(70, 0.5)
    parent = np.clip(parent, 0.0, 1.0)
    parent = repair_feasible(parent)

    best_h = parent.copy()
    best_c = exact_score(parent)

    h_sym = repair_feasible(symmetrize(parent))
    c_sym = exact_score(h_sym)
    if c_sym < best_c - 1e-12:
        best_h = h_sym.copy()
        best_c = c_sym

    # Phase 1: Quick polish
    if time_left() > 200:
        h_lse, c_lse = lse_descent(best_h, beta=80, max_iters=120)
        if c_lse < best_c:
            best_h = h_lse.copy()
            best_c = c_lse
        h_ase, c_ase = active_set_descent(best_h, max_iters=120, n_active=14)
        if c_ase < best_c:
            best_h = h_ase.copy()
            best_c = c_ase

    # Phase 2: Main loop - LP direction + pair moves
    n = len(best_h)
    stagnation = 0
    use_all_lp = True

    while time_left() > 150:
        # LP direction
        D_val = 0.6 if use_all_lp else 0.4
        d_dir, S_pred = lp_minimax_direction(best_h, D=D_val, use_all=use_all_lp)
        lp_improved = False
        if d_dir is not None:
            h_new, c_new = line_search_direction(best_h, d_dir)
            if c_new < best_c - 1e-14:
                best_h = h_new
                best_c = c_new
                lp_improved = True
                stagnation = 0
                use_all_lp = True
            else:
                # Try smaller D or inactive-only
                if use_all_lp:
                    use_all_lp = False
                else:
                    d_dir2, S2 = lp_minimax_direction(best_h, D=0.3, use_all=False)
                    if d_dir2 is not None:
                        h_new2, c_new2 = line_search_direction(best_h, d_dir2)
                        if c_new2 < best_c - 1e-14:
                            best_h = h_new2
                            best_c = c_new2
                            lp_improved = True
                            stagnation = 0

        # Pair move step
        if time_left() > 60:
            h_pm, c_pm = pair_move_descent(best_h, max_rounds=3, n_samples=30, rng=rng)
            if c_pm < best_c - 1e-14:
                best_h = h_pm
                best_c = c_pm
                lp_improved = True
                stagnation = 0

        if not lp_improved:
            stagnation += 1
            if stagnation > 6:
                # Stochastic kick to escape
                amplitude = 0.01 * stagnation
                z = rng.randn(n)
                z -= z.mean()
                h_kick = repair_feasible(best_h + amplitude * z)
                c_kick = exact_score(h_kick)
                if c_kick < best_c:
                    best_h = h_kick
                    best_c = c_kick
                    stagnation = 0
                    use_all_lp = True
                elif stagnation > 15:
                    break
            continue

    # Phase 3: Resolution exploration
    n_best = len(best_h)
    if time_left() > 150:
        candidate_ns = []
        for dn in [-4, -2, 2, 4, 6]:
            nn = n_best + dn
            if nn >= 30 and nn <= 110 and nn % 2 == 0:
                candidate_ns.append(nn)
        for new_n in sorted(candidate_ns, key=lambda x: abs(x - n_best)):
            if time_left() < 50:
                break
            h_interp = interpolate_to_n(best_h, new_n)
            c_interp = exact_score(h_interp)
            h_opt, c_opt = lse_descent(h_interp, beta=100, max_iters=60)
            if c_opt < c_interp:
                c_interp = c_opt
                h_interp = h_opt
            h_opt2, c_opt2 = active_set_descent(h_interp, max_iters=60, n_active=15)
            if c_opt2 < c_interp:
                c_interp = c_opt2
                h_interp = h_opt2
            # Quick LP round
            for _ in range(5):
                if time_left() < 30:
                    break
                dd, Sp = lp_minimax_direction(h_interp, D=0.5, use_all=True)
                if dd is None:
                    break
                h_try, c_try = line_search_direction(h_interp, dd)
                if c_try < c_interp - 1e-14:
                    h_interp = h_try
                    c_interp = c_try
                else:
                    break
            # Pair moves at new resolution
            h_pm, c_pm = pair_move_descent(h_interp, max_rounds=2, n_samples=25)
            if c_pm < c_interp:
                c_interp = c_pm
                h_interp = h_pm
            if c_interp < best_c - 1e-14:
                best_h = h_interp.copy()
                best_c = c_interp

    # Phase 4: Multi-start exploration
    if time_left() > 150:
        n_b = len(best_h)
        starts = []
        starts.append(best_h.copy())
        starts.append(repair_feasible(symmetrize(best_h)))
        z1 = rng.randn(n_b); z1 -= z1.mean()
        starts.append(repair_feasible(best_h + 0.04 * z1))
        z2 = rng.randn(n_b); z2 -= z2.mean()
        starts.append(repair_feasible(best_h + 0.10 * z2))
        # Binary projection blend
        thresh = np.percentile(best_h, 50)
        bin_proj = (best_h > thresh + 0.03).astype(float)
        bin_proj = repair_feasible(bin_proj)
        starts.append(repair_feasible(0.5 * best_h + 0.5 * bin_proj))
        # Fourier perturbation
        freqs = rng.choice(np.arange(2, n_b//2), size=min(4, n_b//2-2), replace=False)
        zf = np.zeros(n_b)
        for f in freqs:
            zf += rng.rand() * np.cos(2*np.pi*f*np.arange(n_b)/n_b) + 0.5*rng.rand()*np.sin(2*np.pi*f*np.arange(n_b)/n_b)
        zf -= zf.mean()
        starts.append(repair_feasible(best_h + 0.03 * zf))
        # Large kick
        z3 = rng.randn(n_b); z3 -= z3.mean()
        starts.append(repair_feasible(best_h + 0.2 * z3))
        # Antisymmetric
        za = rng.randn(n_b)
        za_as = 0.5*(za + (1.0 - za[::-1]) - 1.0)
        za_as -= za_as.mean()
        starts.append(repair_feasible(best_h + 0.025 * za_as))

        for h0 in starts:
            if time_left() < 40:
                break
            c0 = exact_score(h0)
            h_opt, c_opt = lse_descent(h0, beta=100, max_iters=50)
            h_opt2, c_opt2 = active_set_descent(h_opt, max_iters=50, n_active=12)
            if c_opt2 < c_opt:
                h_opt = h_opt2
                c_opt = c_opt2
            # LP round
            for _ in range(4):
                if time_left() < 20:
                    break
                dd, Sp = lp_minimax_direction(h_opt, D=0.4, use_all=True)
                if dd is None:
                    break
                h_try, c_try = line_search_direction(h_opt, dd)
                if c_try < c_opt - 1e-14:
                    h_opt = h_try
                    c_opt = c_try
                else:
                    break
            # Pair moves
            if time_left() > 30:
                h_pm, c_pm = pair_move_descent(h_opt, max_rounds=2, n_samples=25)
                if c_pm < c_opt:
                    c_opt = c_pm
                    h_opt = h_pm
            if c_opt < best_c - 1e-14:
                best_h = h_opt.copy()
                best_c = c_opt

    # Phase 5: Final intensive polish
    if time_left() > 40:
        for rnd in range(15):
            if time_left() < 10:
                break
            h_a, c_a = active_set_descent(best_h, max_iters=30, n_active=10, alpha_init=0.003)
            if c_a < best_c:
                best_h = h_a.copy()
                best_c = c_a
            dd, Sp = lp_minimax_direction(best_h, D=0.35, use_all=True)
            if dd is not None:
                h_l, c_l = line_search_direction(best_h, dd)
                if c_l < best_c:
                    best_h = h_l
                    best_c = c_l

    # Phase 6: Final LSE + active set
    if time_left() > 20:
        h_lse, c_lse = lse_descent(best_h, beta=150, max_iters=80, alpha_init=0.002)
        if c_lse < best_c:
            best_h = h_lse.copy()
            best_c = c_lse

    if time_left() > 10:
        h_ase, c_ase = active_set_descent(best_h, max_iters=150, n_active=8, alpha_init=0.001)
        if c_ase < best_c:
            best_h = h_ase.copy()
            best_c = c_ase

    # Final validation
    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)

    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            final_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 2:
            final_c = float(eval_result[0])
    except Exception:
        pass

    return best_h, final_c, len(best_h)
