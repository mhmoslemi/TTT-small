```python
import numpy as np
from scipy.optimize import minimize
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def repair_feasible(h):
    n = len(h)
    h = np.clip(np.asarray(h, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, 1.0 - h[idx])
            h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)[::-1]
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        idx = int(np.argmax(1.0 - h)) if diff2 > 0 else int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def build_gradient_matrix(h):
    """Build the M×n gradient matrix for the profile."""
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    G = np.zeros((M, n))
    center = n - 1
    
    # For d >= 0 (k = center + d):
    # P[k] = sum_{i=d}^{n-1} h[i] * (1 - h[i-d])
    # dP[k]/dh[p] = (1 - h[p-d]) [if p >= d] + (-h[p+d]) [if p+d <= n-1]
    for d in range(0, n):
        k = center + d
        # p as first element: p >= d
        if d == 0:
            # p as both first and second: (1-h[p]) + (-h[p]) = 1 - 2*h[p]
            G[k, :] = dx * (1.0 - 2.0 * h)
        else:
            # p as first element (i=p in the sum, need p >= d)
            G[k, d:] += dx * (1.0 - h[:n-d])
            # p as second element (i-d=p, i=p+d, need p+d <= n-1, so p <= n-1-d)
            G[k, :n-d] += -dx * h[d:]
    
    # For d < 0 (k = center + d, let r = -d):
    # P[k] = sum_{i=0}^{n-1-r} h[i] * (1 - h[i+r])
    # dP[k]/dh[p] = (1 - h[p+r]) [if p <= n-1-r] + (-h[p-r]) [if p >= r]
    for r in range(1, n):
        k = center - r
        # p as first element (i=p, need p <= n-1-r)
        G[k, :n-r] += dx * (1.0 - h[r:n])
        # p as second element (i+r=p, i=p-r, need p >= r)
        G[k, r:] += -dx * h[:n-r]
    
    return G
```
