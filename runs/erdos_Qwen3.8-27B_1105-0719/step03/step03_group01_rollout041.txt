```python
def compute_smoothed_grad(h, weights, n):
    """Compute gradient of weighted sum of correlations."""
    grad = np.zeros(n)
    for p in range(2*n - 1):
        w = weights[p]
        if w < 1e-10:
            continue
        d = p - (n-1)
        if d >= 0:
            # c[p] = sum_{i=0}^{n-d-1} h[i+d] * (1-h[i])
            for i in range(n-d):
                grad[i+d] += w * (1 - h[i])   # derivative w.r.t. h[i+d]
                grad[i] += w * (-h[i+d])       # derivative w.r.t. h[i]
        else:
            # d < 0, let k = -d
            k = -d
            # c[p] = sum_{i=k}^{n-1} h[i-k] * (1-h[i])
            for i in range(k, n):
                grad[i-k] += w * (1 - h[i])   # derivative w.r.t. h[i-k]
                grad[i] += w * (-h[i-k])       # derivative w.r.t. h[i]
    return grad
```
