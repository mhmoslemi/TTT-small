```python
def project_feasible(x, target_sum):

    n = len(x)

def grad_corr_k(h, k, n):
```

corr_k(eps) = corr_k + eps * linear_k + eps² * quad_k

Where quad_k is:
- 0 for most k
- -1 for the k where i+off = j or j+off = i (because of the -h[i]*h[j] cross term)

So corr_k(eps) = a + b*eps - c*eps² where c ≥ 0 (at most 2 values of k have c > 0).

For the 2 values with c > 0, the function is concave (opens downward).
For all other values, the function is linear.

The max of {linear functions, a few concave quadratic functions} is NOT necessarily convex or concave in general. It's a piecewise function.

However, since the concave parts are only at specific k values, the max will typically be determined by the linear parts (which form a convex V-shape) with possible modifications at the points where the concave parts become active.

For practical purposes, the golden section search should still work well because:
1. The function is unimodal in practice (for small eps_max)
2. Even if not strictly unimodal, the golden section will find a good local minimum

```python
import numpy as np
from scipy.optimize import minimize, linprog
from scipy.special import logsumexp
import time


def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)


def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def c5_of_h(h, dx):
    return compute_corrs(h).max() * dx


def grad_corr_k(h, k, n):
    offset = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -offset)
    i_hi = min(n - 1, n - 1 - offset)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx + offset] += (1.0 - h[idx])
    grad[idx] += (-h[idx + offset])
    return grad


def build_grad_matrix(h, n):
    n_corr = 2 * n - 1
    G = np.zeros((n_corr, n))
    for k in range(n_corr):
        G[k, :] = grad_corr_k(h, k, n)
    return G


def eval_pair_c5(h, i, j, eps, dx):
    eh = h.copy()
    eh[i] -= eps
    eh[j] += eps
    return compute_corrs(eh).max() * dx


def _obj_smooth(x, dx, beta):
    corrs = compute_corrs(x)
    return logsumexp(beta * corrs) / beta * dx


def _grad_smooth(x, dx, beta):
    n = len(x)
    n_corr = 2 * n - 1
    corrs = compute_corrs(x)
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(n_corr) / n_corr
    else:
        w /= w_sum
    G = build_grad_matrix(x, n)
    return (w @ G) * dx


def _c_eq(x, target_sum):
    return x.sum() - target_sum


def _c_eq_jac(x, target_sum):
    return np.ones(len(x))


def slsqp_optimize(h0, dx, target_sum, beta, maxiter, t0, time_limit):
    n = len(h0)
    if time.time() - t0 > time_limit:
        return h0, c5_of_h(h0, dx)
    cons = [{'type': 'eq', 'fun': _c_eq, 'jac': _c_eq_jac, 'args': (target_sum,)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_obj_smooth, h0, args=(dx, beta), jac=_grad_smooth,
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        h_out = project_feasible(res.x, target_sum)
    except Exception:
        h_out = project_feasible(h0, target_sum)
    return h_out, c5_of_h(h_out, dx)


def lp_polish(h0, dx, target_sum, max_iters, init_step, t0, time_limit):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    step = init_step
    stall = 0

    for it in range(max_iters):
        if time.time() - t0 > time_limit:
            break

        corrs = compute_corrs(h)
        n_corr = len(corrs)
        max_c = corrs.max()

        if max_c * dx < best_c5 - 1e-15:
            best_c5 = max_c * dx
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 60:
            step *= 0.65
            stall = 0
            if step < 1e-9:
                break

        tol_frac = 0.10
        tol_val = max_c * (1.0 - tol_frac)
        active_k = np.where(corrs >= tol_val)[0]
        if len(active_k) > 50:
            active_k = np.argsort(corrs)[-50:]

        G = build_grad_matrix(h, n)

        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        A_rows = []
        b_rows = []
        for k in active_k:
            row = np.zeros(n + 1)
            row[:n] = dx * G[k, :]
            row[n] = -1.0
            A_rows.append(row)
            b_rows.append(-corrs[k] * dx - 1e-13)

        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)

        bounds = []
        for i in range(n):
            bounds.append((max(-h[i], -step), min(1.0 - h[i], step)))
        bounds.append((0.0, None))

        try:
            res = linprog(c_obj, A_ub=np.array(A_rows), b_ub=np.array(b_rows),
                          A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
        except Exception:
            break
        if res.status != 0:
            break

        delta = res.x[:n]
        if np.max(np.abs(delta)) < 1e-14:
            break

        h_new = project_feasible(h + delta, target_sum)
        new_c5 = c5_of_h(h_new, dx)

        if new_c5 < max_c * dx - 1e-14:
            h = h_new
            step = min(step * 1.25, init_step)
        else:
            improved = False
            for frac in [0.5, 0.25, 0.1, 0.05]:
                h_new = project_feasible(h + frac * delta, target_sum)
                new_c5 = c5_of_h(h_new, dx)
                if new_c5 < max_c * dx - 1e-14:
                    h = h_new
                    improved = True
                    break
            if not improved:
                step *= 0.5
                if step < 1e-9:
                    break

    return best_h, best_c5


def pair_transfer(h0, dx, target_sum, n_top, max_rounds, t0, time_limit):
    n = len(h0)
    best_h = h0.copy()
    best_c5 = c5_of_h(h0, dx)
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    for rnd in range(max_rounds):
        if time.time() - t0 > time_limit:
            break

        corrs = compute_corrs(best_h)
        n_corr = 2 * n - 1

        beta = 400.0
        shifted = corrs - corrs.max()
        w = np.exp(beta * shifted)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(n_corr) / n_corr
        else:
            w /= w_sum

        G = build_grad_matrix(best_h, n)
        grad = (w @ G) * dx

        dec_idx = np.argsort(grad)[-n_top:]
        inc_idx = np.argsort(grad)[:n_top]

        improved = False
        for i in dec_idx:
            if time.time() - t0 > time_limit:
                break
            for j in inc_idx:
                if i == j:
                    continue
                eps_max = min(best_h[i], 1.0 - best_h[j])
                if eps_max < 1e-10:
                    continue

                a, b = 0.0, eps_max
                c_pt = b - gr * (b - a)
                d_pt = a + gr * (b - a)
                fc = eval_pair_c5(best_h, i, j, c_pt, dx)
                fd = eval_pair_c5(best_h, i, j, d_pt, dx)

                for _ in range(22):
                    if b - a < 1e-13:
                        break
                    if fc < fd:
                        b = d_pt
                        d_pt = c_pt
                        fd = fc
                        c_pt = b - gr * (b - a)
                        fc = eval_pair_c5(best_h, i, j, c_pt, dx)
                    else:
                        a = c_pt
                        c_pt = d_pt
                        fc = fd
                        d_pt = a + gr * (b - a)
                        fd = eval_pair_c5(best_h, i, j, d_pt, dx)

                eps_opt = (a + b) / 2.0
                if eps_opt < 1e-12:
                    continue

                new_h = best_h.copy()
                new_h[i] -= eps_opt
                new_h[j] += eps_opt
                new_h = np.clip(new_h, 0.0, 1.0)
                new_c5 = c5_of_h(new_h, dx)

                if new_c5 < best_c5 - 1e-15:
                    best_h = new_h
                    best_c5 = new_c5
                    improved = True

        if not improved:
            break

    return best_h, best_c5


def momentum_subgrad(h0, dx, target_sum, max_iter, beta_smooth, t0, time_limit):
    n = len(h0)
    h = h0.copy()
    momentum = np.zeros(n)
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    no_improve = 0
    lr = 1.0 / np.sqrt(n)
    beta_mom = 0.9

    for it in range(max_iter):
        if time.time() - t0 > time_limit:
            break

        corrs = compute_corrs(h)
        c5_val = corrs.max() * dx

        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 3000:
            break

        shifted = corrs - corrs.max()
        w = np.exp(beta_smooth * shifted)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(len(corrs)) / len(corrs)
        else:
            w /= w_sum

        G = build_grad_matrix(h, n)
        grad = (w @ G) * dx

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        momentum = beta_mom * momentum - lr * grad / (gnorm + 1e-10)
        h_new = h + momentum
        h_new = project_feasible(h_new, target_sum)
        new_c5 = c5_of_h(h_new, dx)

        tries = 0
        while new_c5 > c5_val + 1e-14 and tries < 5:
            lr *= 0.5
            momentum = beta_mom * momentum - lr * grad / (gnorm + 1e-10)
            h_new = h + momentum
            h_new = project_feasible(h_new, target_sum)
            new_c5 = c5_of_h(h_new, dx)
            tries += 1

        h = h_new

    return best_h, best_c5


def make_periodic_seed(n, target_sum, period, fill_ratio):
    h = np.zeros(n)
    fill = max(1, int(fill_ratio * period))
    for i in range(n):
        h[i] = 1.0 if (i % period) < fill else 0.0
    return project_feasible(h, target_sum)
```
