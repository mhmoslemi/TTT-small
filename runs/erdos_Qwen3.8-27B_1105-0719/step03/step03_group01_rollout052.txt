def project_feasible(x, target_sum):
    n = len(x)
    x = np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0)
    x = np.clip(x, -3.0, 3.0)
    lo, hi = -3.0, 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = mask.sum()
        if cnt > 0:
            h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corrs = np.correlate(h, 1.0 - h, mode='full')
    return corrs.max() * dx


def compute_c5_and_corrs(h):
    n = len(h)
    dx = 2.0 / n
    corrs = np.correlate(h, 1.0 - h, mode='full')
    return corrs.max() * dx, corrs


def slsqp_solve(h0, maxiter=250, ftol=1e-12, t_margin=1e-4):
    n = len(h0)
    dx = 2.0 / n
    target_sum = n / 2.0
    m = 2 * n - 1
    c5_init = compute_c5(h0)

    def obj(z):
        return z[0]

    def obj_grad(z):
        g = np.zeros(n + 1)
        g[0] = 1.0
        return g

    def constr(z):
        h = z[1:]
        corrs = np.correlate(h, 1.0 - h, mode='full')
        return z[0] - dx * corrs

    def constr_jac(z):
        h = z[1:]
        J = np.zeros((m, n + 1))
        J[:, 0] = 1.0
        for k in range(m):
            d = k - (n - 1)
            if d >= 0:
                J[k, d + 1:n + 1] -= dx * (1.0 - h[0:n - d])
                J[k, 0:n - d] += dx * h[d:n]
            else:
                dp = -d
                J[k, 0:n - dp] -= dx * (1.0 - h[dp:n])
                J[k, dp:n + 1] += dx * h[0:n - dp]
        return J

    def eq_fun(z):
        return z[1:].sum() - target_sum

    def eq_jac(z):
        g = np.zeros(n + 1)
        g[1:] = 1.0
        return g

    z0 = np.concatenate(([c5_init + t_margin], h0))
    bounds = [(0.0, 0.6)] + [(0.0, 1.0)] * n
    constraints = [
        {'type': 'ineq', 'fun': constr, 'jac': constr_jac},
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}
    ]

    try:
        res = minimize(obj, z0, method='SLSQP', jac=obj_grad,
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': ftol})
    except Exception:
        return h0, c5_init

    h_new = project_feasible(res.x[1:], target_sum)
    c5_new = compute_c5(h_new)
    if c5_new < c5_init - 1e-12:
        return h_new, c5_new
    return h0, c5_init

def line_search_optimize(h0, v):
    n = len(h0)
    dx = 2.0 / n

    lo_cands = []
    hi_cands = []
    for i in range(n):
        if v[i] > 1e-14:
            lo_cands.append(-h0[i] / v[i])
            hi_cands.append((1.0 - h0[i]) / v[i])
        elif v[i] < -1e-14:
            lo_cands.append((1.0 - h0[i]) / v[i])
            hi_cands.append(-h0[i] / v[i])

    alpha_lo = max(lo_cands) if lo_cands else -0.5
    alpha_hi = min(hi_cands) if hi_cands else 0.5
    alpha_lo = max(alpha_lo, -0.4)
    alpha_hi = min(alpha_hi, 0.4)
    alpha_lo = min(alpha_lo, -1e-6)
    alpha_hi = max(alpha_hi, 1e-6)

    if alpha_hi - alpha_lo < 1e-8:
        return h0, compute_c5(h0)

    c5_init = compute_c5(h0)

    def f(alpha):
        h_new = h0 + alpha * v
        return compute_c5(np.clip(h_new, 0.0, 1.0))

                              method='bounded', options={'maxiter': 60, 'xatol': 1e-10})

            c5_new = compute_c5(h_new)
            if c5_new < c5_init:

        pass
    return h0, c5_init


def gen_zero_sum_dir(n, rng, mode=0):
    if mode == 0:
        v = rng.randn(n)
    elif mode == 1:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            j = (i + 1) % n
        v = np.zeros(n)
        v[i] = 1.0
        v[j] = -1.0
    elif mode == 2:
        v = np.sin(np.pi * np.arange(n) * (1 + rng.randint(3)))
    else:
        v = np.zeros(n)
        idx = rng.choice(n, size=4, replace=False)
        sig = np.array([1, -1, 1, -1])
        for k in range(4):
            v[idx[k]] = sig[k]
    v -= v.mean()
    nv = np.linalg.norm(v)
    if nv > 1e-14:
        v /= nv
    return v
