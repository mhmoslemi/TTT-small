    n = len(h)
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(profile))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx])
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-10:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmin(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def build_grad(h, active):
    """Compute gradient g[i] = sum_{m in active} dP_m/dh[i] (without dx factor, we'll add it)."""
    n = len(h)
    g = np.zeros(n)
    for m in active:

        if d == 0:
            g += (1.0 - 2.0 * h)
        elif d > 0:
            if d <= n - 1 - d:  # d <= (n-1)/2
                # [0, d-1]: -h[i+d]
                g[0:d] += -h[d:2*d]
                # [d, n-1-d]: (1-h[i-d]) - h[i+d]
                g[d:n-d] += (1.0 - h[0:n-2*d]) - h[2*d:n]
                # [n-d, n-1]: (1-h[i-d])
                g[n-d:] += (1.0 - h[n-2*d:n-d])
            else:  # d > (n-1)/2
                # [0, n-d-1]: -h[i+d]
                g[0:n-d] += -h[d:n]
                # [n-d, d-1]: 0
                # [d, n-1]: (1-h[i-d])
                g[d:] += (1.0 - h[0:n-d])
        else:
            dp = -d
            if dp <= n - 1 - dp:  # dp <= (n-1)/2
                # [0, dp-1]: (1-h[i+dp])

                # [n-dp, dp-1]: 0
                # [dp, n-1]: -h[i-dp]
