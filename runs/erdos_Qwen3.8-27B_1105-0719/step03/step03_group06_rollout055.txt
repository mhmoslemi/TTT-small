    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    G = np.zeros((M, n))
    # d = 0 (m = n-1)
    G[n-1, :] = dx * (1.0 - 2.0 * h)
    # d > 0 (m = n-1+1, ..., 2n-2, i.e., d = 1, ..., n-1)
    for d in range(1, n):
        m = n - 1 + d
        G[m, d:] += dx * (1.0 - h[:n-d])
        G[m, :n-d] -= dx * h[d:]
    # d < 0 (m = 0, ..., n-2, i.e., p = -d = 1, ..., n-1, so p = n-1-m)
    for p in range(1, n):
        m = n - 1 - p
        G[m, :n-p] += dx * (1.0 - h[p:])
        G[m, p:] -= dx * h[:n-p]
    return G

        # Variables: d[0..n-1], tau
        # Constraints:
        #   G[m,:] @ d - tau <= -profile[m]  for all m
        #   sum(d) = 0
        #   -min(h[i], delta) <= d[i] <= min(1-h[i], delta)

        c_obj[-1] = 1.0  # minimize tau

        A_ub[:, -1] = -1.0
        b_ub = -profile

            hi = min(1.0 - h[i], delta)

import numpy as np
from scipy.optimize import linprog

def repair_feasible(vec):
    n = len(vec)
    h = np.asarray(vec, dtype=np.float64).copy()
    h[~np.isfinite(h)] = 0.5
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(-(1.0 - h))
        remaining = diff
        for i in range(n):
            idx = order[i]
            cap = 1.0 - h[idx]
            if cap <= 1e-15:
                continue
            add = min(remaining, cap)
            h[idx] += add
            remaining -= add
            if remaining <= 1e-14:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            if h[idx] <= 1e-15:
                continue
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-14:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] = float(np.clip(h[idx] + diff2, 0.0, 1.0))
    return h
