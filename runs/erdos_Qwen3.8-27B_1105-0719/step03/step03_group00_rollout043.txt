```python
import numpy as np
from scipy.optimize import minimize, linprog
import time

def project_feasible(x, target_sum):
    """Project x onto {0<=h<=1, sum(h)=target_sum}"""
    n = len(x)
    # Binary search for the threshold
    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    # Fix any small discrepancy
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            adj = diff / mask.sum()
            if abs(adj) < 1e-10:
                h[mask] += adj
            else:
                # Clamp
                h = np.clip(h + adj * mask / max(1, mask.sum()), 0, 1)
                # Re-project if needed
                h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    return h

def c5_eval(h, dx):
    """Compute C5 = max_k correlate(h, 1-h) * dx"""
    return np.correlate(h, 1.0 - h, mode='full').max() * dx

def correlation_vector(h, dx):
    """Compute all correlation values scaled by dx"""
    return np.correlate(h, 1.0 - h, mode='full') * dx

def softmax_objective(h, dx, T):
    """Soft-max approximation of max correlation"""
    corrs = np.correlate(h, 1.0 - h, mode='full') * dx
    if T < 1e-10:
        return corrs.max()
    m = corrs.max()
    # For numerical stability
    exps = np.exp((corrs - m) / T)
    return m + T * np.log(exps.sum())

def softmax_gradient(h, dx, T):
    """Gradient of soft-max objective"""
    n = len(h)
    corrs = np.correlate(h, 1.0 - h, mode='full') * dx
    if T < 1e-10:
        # Hard max: gradient only from active set
        max_c = corrs.max()
        active = (corrs >= max_c - 1e-10)
        weights = active / active.sum()
    else:
        m = corrs.max()
        exps = np.exp((corrs - m) / T)
        weights = exps / exps.sum()
    
    # Gradient of correlation k w.r.t. h[i]:
    # corr_k = sum_i h[i] * (1-h[i+k-(n-1)])
    # d(corr_k)/d(h[i]) = (1-h[i+k-(n-1)]) - h[i-k-(n-1)]
    # (where indices are valid)
    
    grad = np.zeros(n)
    # For each lag, add its contribution weighted by its weight
    full_corr_idx = np.arange(2*n-1)
    for k in range(2*n-1):
        if weights[k] < 1e-10:
            continue
        off = k - (n - 1)
        i_lo = max(0, -off)
        i_hi = min(n-1, n-1-off)
        if i_lo > i_hi:
            continue
        idx = np.arange(i_lo, i_hi+1)
        # d(c_k)/d(h[i]) = (1-h[i+off]) - h[i-off] for valid i
        # h[i+off] is valid when 0 <= i+off < n, i.e., i >= -off and i <= n-1-off
        # h[i-off] is valid when 0 <= i-off < n, i.e., i >= off and i <= n-1+off
        term1 = np.zeros(n)
        term2 = np.zeros(n)
        valid1_lo = max(0, -off)
        valid1_hi = min(n-1, n-1-off)
        if valid1_lo <= valid1_hi:
            term1[valid1_lo:valid1_hi+1] = 1.0 - h[valid1_lo+off:valid1_hi+off]
        valid2_lo = max(0, off)
        valid2_hi = min(n-1, n-1+off)
        if valid2_lo <= valid2_hi:
            term2[valid2_lo:valid2_hi+1] = h[valid2_lo-off:valid2_hi-off]
        grad += weights[k] * dx * (term1 - term2)
    
    return grad

def softmax_descent(h_init, dx, target_sum, T_start, T_end, n_steps, lr, budget_s, t0):
    """Gradient descent on soft-max objective with projection"""
    n = len(h_init)
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = c5_eval(h, dx)
    T = T_start
    
    for step in range(n_steps):
        if budget_s - (time.time() - t0) < 3.0:
            break
        
        # Anneal temperature
        frac = step / max(1, n_steps - 1)
        T = T_start * (T_end / T_start) ** frac
        
        grad = softmax_gradient(h, dx, T)
        
        # Line search with a few trial steps
        best_t = 0.0
        best_obj = c5_eval(h, dx)
        for t in [lr, lr*0.5, lr*0.25, lr*0.1, lr*0.05]:
            h_trial = h - t * grad
            h_trial = project_feasible(h_trial, target_sum)
            obj = c5_eval(h_trial, dx)
            if obj < best_obj:
                best_obj = obj
                best_t = t
        
        if best_t > 0:
            h = h - best_t * grad
            h = project_feasible(h, target_sum)
            if c5_eval(h, dx) < best_c5:
                best_c5 = c5_eval(h, dx)
                best_h = h.copy()
    
    return best_h, best_c5

def pair_descent(h, dx, target_sum, n_pairs, budget_s, t0):
    """Pair coordinate descent"""
    n = len(h)
    best_h = h.copy()
    best_c5 = c5_eval(h, dx)
    
    for iteration in range(n_pairs):
        if budget_s - (time.time() - t0) < 5.0:
            break
        improved = False
        # Try random pairs
        for _ in range(20):
            if budget_s - (time.time() - t0) < 3.0:
                break
            i, j = np.random.randint(0, n, 2)
            if i == j:
                continue
            # Try moving h[i] up and h[j] down (or vice versa)
            for direction in [1, -1]:
                best_t = 0.0
                best_obj = best_c5
                for t in [0.05, 0.1, 0.2, 0.3, 0.5]:
                    h_trial = best_h.copy()
                    h_trial[i] += direction * t
                    h_trial[j] -= direction * t
                    h_trial = project_feasible(h_trial, target_sum)
                    obj = c5_eval(h_trial, dx)
                    if obj < best_obj:
                        best_obj = obj
                        best_t = t
                if best_t > 0:
                    best_h[i] += direction * best_t
                    best_h[j] -= direction * best_t
                    best_h = project_feasible(best_h, target_sum)
                    best_c5 = c5_eval(best_h, dx)
                    improved = True
        if not improved:
            break
    
    return best_h, best_c5

def binary_sa(h_init, dx, target_sum, n_iter, T_start, T_end, budget_s, t0):
    """Simulated annealing on binary configurations"""
    n = len(h_init)
    target_sum = int(round(target_sum))
    
    # Start from binary initialization
    h = (h_init > 0.5).astype(float)
    if h.sum() != target_sum:
        # Adjust to get exactly target_sum ones
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if h.sum() > target_sum:
            # Remove some ones
            excess = h.sum() - target_sum
            idx = rng_perm(ones, int(excess))
            h[idx] = 0.0
        elif h.sum() < target_sum:
            # Add some ones
            deficit = target_sum - h.sum()
            idx = rng_perm(zeros, int(deficit))
            h[idx] = 1.0
    
    best_h = h.copy()
    best_c5 = c5_eval(h, dx)
    cur_c5 = best_c5
    
    T = T_start
    n_total = 2*n - 1
    
    for it in range(n_iter):
        if budget_s - (time.time() - t0) < 3.0:
            break
        
        # Anneal
        frac = it / max(1, n_iter - 1)
        T = T_start * (T_end / T_start) ** frac
        
        # Propose: flip a random bit (and swap to maintain sum)
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = c5_eval(h, dx)
        
        if new_c5 <= cur_c5 or np.random.random() < np.exp((cur_c5 - new_c5) / max(T, 1e-10)):
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
    
    return best_h, best_c5

def rng_perm(arr, k):
    """Return k random elements from arr"""
    idx = np.random.choice(len(arr), size=min(k, len(arr)), replace=False)
    return arr[idx]

def active_set_lp(h, dx, target_sum, n_iters, budget_s, t0, band_frac=0.05, top_n=40):
    """Active set LP refinement"""
    n = len(h)
    best_h = h.copy()
    best_c5 = c5_eval(h, dx)
    alpha = 0.2
    
    for it in range(n_iters):
        if budget_s - (time.time() - t0) < 3.0:
            break
        
        corrs = np.correlate(best_h, 1.0 - best_h, mode='full')
        max_c = corrs.max()
        band = max(0.01, band_frac * max_c)
        
        # Select active set: top correlations within band
        active_mask = corrs >= (max_c - band)
        # Also include top_n
        top_idx = np.argsort(corrs)[-top_n:]
        active_mask[top_idx] = True
        active_idx = np.where(active_mask)[0]
        
        if len(active_idx) < 3:
            active_idx = top_idx
        
        # Build LP: find delta that minimizes t subject to
        # corrs[k] + dx * (grad_k · delta) <= 0 for k in active
        # sum(delta) = 0
        # |delta[i]| <= alpha
        
        K = len(active_idx)
        m_vars = n + 1
        c_obj = np.zeros(m_vars)
        c_obj[-1] = 1.0
        
        A_ub = np.zeros((K, m_vars))
        b_ub = np.zeros(K)
        
        for j, k in enumerate(active_idx):
            off = k - (n - 1)
            grad = np.zeros(n)
            i_lo = max(0, -off)
            i_hi = min(n-1, n-1-off)
            if i_lo <= i_hi:
                idx = np.arange(i_lo, i_hi+1)
                grad[idx + off] += (1.0 - best_h[idx])
                grad[idx] += (-best_h[idx + off])
            A_ub[j, :n] = dx * grad
            A_ub[j, -1] = -1.0
            b_ub[j] = -dx * corrs[k]
        
        A_eq = np.ones((1, m_vars))
        b_eq = np.zeros(1)
        
        bounds = [(-best_h[i]*alpha, (1.0-best_h[i])*alpha) for i in range(n)]
        bounds.append((0.0, None))
        
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs', options={'presolve': True})
            if res.status != 0:
                alpha *= 0.5
                if alpha < 0.003:
                    break
                continue
            delta = res.x[:n]
        except Exception:
            alpha *= 0.5
            if alpha < 0.003:
                break
            continue
        
        # Line search
        improved = False
        for t in [1.0, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05]:
            h_trial = project_feasible(best_h + t * delta, target_sum)
            new_c5 = c5_eval(h_trial, dx)
            if new_c5 < best_c5 - 1e-14:
                best_h = h_trial
                best_c5 = new_c5
                improved = True
                break
        
        if improved:
            alpha = min(0.4, alpha * 1.03)
        else:
            alpha *= 0.7
            if alpha < 0.003:
                break
    
    return best_h, best_c5
```

```python
    """Find v in [lo, hi] minimizing max_k (a[k]*v + b[k])"""
    K = len(a)
    # Compute all breakpoints
    bp = set()
    bp.add(lo)
    bp.add(hi)
    for k in range(K):
        for l in range(k+1, K):
            da = a[k] - a[l]
            if abs(da) > 1e-15:
                v = (b[l] - b[k]) / da
                if lo <= v <= hi:
                    bp.add(v)

    best_v = lo
    best_val = max(a * lo + b)
    for v in bp:
        val = max(a * v + b)
        if val < best_val:
            best_val = val
            best_v = v
    
    return best_v, best_val
```

```python
import numpy as np
from scipy.optimize import linprog
import time

def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def c5_eval(h, dx):
    return np.correlate(h, 1.0 - h, mode='full').max() * dx

def pair_cd(h, dx, target_sum, n_sweeps, n_evals, budget_s, t0, rng):
    """Pair cyclic coordinate descent"""
    n = len(h)
    best_h = h.copy()
    best_c5 = c5_eval(h, dx)
    prev_c5 = best_c5
    
    for sweep in range(n_sweeps):
        if budget_s - (time.time() - t0) < 5.0:
            break
        improved = False
        order = rng.permutation(n)
        
        for i in order:
            j = (i + 1) % n
            if i == j:
                continue
            s = best_h[i] + best_h[j]
            lo_v = max(0.0, s - 1.0)
            hi_v = min(1.0, s)
            if hi_v - lo_v < 1e-10:
                continue
            
            # Try n_evals points
            if n_evals <= 1:
                v_candidates = np.array([best_h[i]])
            else:
                v_candidates = np.linspace(lo_v, hi_v, n_evals)
            
            trial_h = best_h.copy()
            best_v = best_h[i]
            local_best = c5_eval(best_h, dx)
            
            for v in v_candidates:
                dv = v - best_h[i]
                if abs(dv) < 1e-12:
                    continue
                trial_h[i] = v
                trial_h[j] = s - v
                obj = c5_eval(trial_h, dx)
                if obj < local_best:
                    local_best = obj
                    best_v = v
            
            if abs(best_v - best_h[i]) > 1e-10:
                best_h[i] = best_v
                best_h[j] = s - best_v
                improved = True
                if local_best < best_c5:
                    best_c5 = local_best
        
        if not improved:
            break
    
    return best_h, best_c5

def binary_sa(h_init, dx, target_sum, n_iter, T_start, T_end, budget_s, t0, rng):
    """Simulated annealing on binary configs (swap one 1 with one 0)"""
    n = len(h_init)
    n_ones = int(round(target_sum))
    
    # Initialize binary
    h = (h_init > 0.5).astype(float)
    if int(h.sum()) != n_ones:
        diff = n_ones - int(h.sum())
        if diff > 0:
            zeros = np.where(h == 0.0)[0]
            idx = rng.choice(zeros, size=diff, replace=False)
            h[idx] = 1.0
        elif diff < 0:
            ones = np.where(h == 1.0)[0]
            idx = rng.choice(ones, size=-diff, replace=False)
            h[idx] = 0.0
    
    best_h = h.copy()
    best_c5 = c5_eval(h, dx)
    cur_c5 = best_c5
    
    for it in range(n_iter):
        if budget_s - (time.time() - t0) < 3.0:
            break
        
        frac = it / max(1, n_iter - 1)
        T = T_start * (T_end / T_start) ** frac
        
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = c5_eval(h, dx)
        
        if new_c5 <= cur_c5 or (T > 1e-8 and rng.random() < np.exp(-(new_c5 - cur_c5) / T)):
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
    
    return best_h, best_c5

def active_set_lp(h, dx, target_sum, n_iters, budget_s, t0, band_frac=0.05, top_n=40, alpha_init=0.2):
    """Active set LP-based refinement"""
    n = len(h)
    best_h = h.copy()
    best_c5 = c5_eval(h, dx)
    alpha = alpha_init
    
    for it in range(n_iters):
        if budget_s - (time.time() - t0) < 3.0:
            break
        
        corrs = np.correlate(best_h, 1.0 - best_h, mode='full')
        max_c = corrs.max()
        band = max(0.005, band_frac * max_c)
        
        active_mask = corrs >= (max_c - band)
        top_idx = np.argsort(corrs)[-top_n:]
        active_mask[top_idx] = True
        active_idx = np.where(active_mask)[0]
        
        if len(active_idx) < 3:
            active_idx = top_idx
        
        K = len(active_idx)
        m_vars = n + 1
        c_obj = np.zeros(m_vars)
        c_obj[-1] = 1.0
        A_ub = np.zeros((K, m_vars))
        b_ub = np.zeros(K)
        
        for j, k in enumerate(active_idx):
            off = k - (n - 1)
            grad = np.zeros(n)
            i_lo = max(0, -off)
            i_hi = min(n-1, n-1-off)
            if i_lo <= i_hi:
                idx = np.arange(i_lo, i_hi+1)
                grad[idx + off] += (1.0 - best_h[idx])
                grad[idx] += (-best_h[idx + off])
            A_ub[j, :n] = dx * grad
            A_ub[j, -1] = -1.0
            b_ub[j] = -dx * corrs[k]
        
        A_eq = np.ones((1, m_vars))
        b_eq = np.zeros(1)
        bounds = [(max(-best_h[i]*alpha, -0.5), min((1.0-best_h[i])*alpha, 0.5)) for i in range(n)]
        bounds.append((0.0, None))
        
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs', options={'presolve': True})
            if res.status != 0:
                alpha *= 0.5
                if alpha < 0.002:
                    break
                continue
            delta = res.x[:n]
        except Exception:
            alpha *= 0.5
            if alpha < 0.002:
                break
            continue
        
        improved = False
        for t in [1.0, 0.8, 0.6, 0.4, 0.3, 0.2, 0.1, 0.05]:

            if new_c5 < best_c5 - 1e-15:
                best_h = h_trial
                best_c5 = new_c5
                improved = True
                break
        
        if improved:
            alpha = min(0.4, alpha * 1.05)
        else:
            alpha *= 0.6
            if alpha < 0.002:
                break
    
    return best_h, best_c5

def softmax_grad(h, dx, T):
    """Gradient of soft-max of correlations"""
    n = len(h)
    corrs = np.correlate(h, 1.0 - h, mode='full') * dx
    m = corrs.max()
    exps = np.exp((corrs - m) / T)
    w = exps / exps.sum()
    
    grad = np.zeros(n)
    for k in range(2*n-1):
        if w[k] < 1e-12:

            break
        frac = step / max(1, n_steps-1)
        T = T_start * (T_end / T_start) ** frac
        lr_cur = lr * (1.0 - 0.5*frac)
        
        grad = softmax_grad(h, dx, T)

            new_c5 = c5_eval(h_trial, dx)

                    best_h = h.copy()
```

```python
def min_1d_max_linear(a, c, lo, hi):
    """Minimize max_k(a[k]*v + c[k]) over v in [lo, hi]"""
    # f(v) = max_k(a_k*v + c_k) is convex piecewise linear
    # Find minimum
    
    # Check if f is monotone
    f_lo = np.max(a * lo + c)
    f_hi = np.max(a * hi + c)
    
    # Subgradient at lo: all a_k that achieve the max at lo
    vals_lo = a * lo + c
    max_lo = vals_lo.max()
    sg_lo_min = a[vals_lo >= max_lo - 1e-12].min()
    
    # Subgradient at hi
    vals_hi = a * hi + c
    max_hi = vals_hi.max()
    sg_hi_max = a[vals_hi >= max_hi - 1e-12].max()
    
    if sg_lo_min >= 0:
        return lo, f_lo
    if sg_hi_max <= 0:
        return hi, f_hi
    
    # Bisection
    for _ in range(40):
        mid = (lo + hi) / 2
        vals = a * mid + c
        m = vals.max()
        active = vals >= m - 1e-12
        sg_min = a[active].min()
        sg_max = a[active].max()
        if sg_min >= 0:
            lo = mid
        elif sg_max <= 0:
            hi = mid
        else:
            return mid, m
    
    mid = (lo + hi) / 2
    return mid, np.max(a * mid + c)
```

```python
import numpy as np
from scipy.optimize import linprog
import time

def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def c5_eval(h, dx):
    return np.correlate(h, 1.0 - h, mode='full').max() * dx

def min_1d_max_linear(a, c, lo, hi):
    f_lo = np.max(a * lo + c)
    if lo >= hi - 1e-14:
        return lo, f_lo
    f_hi = np.max(a * hi + c)
    vals_lo = a * lo + c
    m_lo = vals_lo.max()
    sg_lo_min = a[vals_lo >= m_lo - 1e-10].min()
    if sg_lo_min >= -1e-12:
        return lo, f_lo
    vals_hi = a * hi + c
    m_hi = vals_hi.max()
    sg_hi_max = a[vals_hi >= m_hi - 1e-10].max()
    if sg_hi_max <= 1e-12:
        return hi, f_hi
    for _ in range(50):
        mid = (lo + hi) / 2.0
        vals = a * mid + c
        m = vals.max()
        active = vals >= m - 1e-10
        sg_min = a[active].min()
        sg_max = a[active].max()
        if sg_min >= -1e-12:
            lo = mid
        elif sg_max <= 1e-12:
            hi = mid
        else:
            return mid, m
    mid = (lo + hi) / 2.0
    return mid, np.max(a * mid + c)

def pair_cd_exact(h, dx, target_sum, n_sweeps, budget_s, t0, rng):
    n = len(h)
    best_h = h.copy()
    best_c5 = c5_eval(h, dx)
    two_n_minus_1 = 2 * n - 1
    for sweep in range(n_sweeps):
        if budget_s - (time.time() - t0) < 5.0:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            j = (i + 1) % n
            s = best_h[i] + best_h[j]
            lo_v = max(0.0, s - 1.0)
            hi_v = min(1.0, s)
            if hi_v - lo_v < 1e-10:
                continue
            corrs = np.correlate(best_h, 1.0 - best_h, mode='full') * dx
            a = np.zeros(two_n_minus_1)
            for k in range(two_n_minus_1):
                off = k - (n - 1)
                t1 = (1.0 - best_h[i + off]) if (0 <= i + off < n) else 0.0
                t2 = (1.0 - best_h[j + off]) if (0 <= j + off < n) else 0.0
                t3 = best_h[i - off] if (0 <= i - off < n) else 0.0
                t4 = best_h[j - off] if (0 <= j - off < n) else 0.0
                a[k] = dx * (t1 - t2 - t3 + t4)
            c_vec = corrs - a * best_h[i]
            v_new, _ = min_1d_max_linear(a, c_vec, lo_v, hi_v)
            if abs(v_new - best_h[i]) > 1e-10:
                best_h[i] = v_new
                best_h[j] = s - v_new
                new_c5 = c5_eval(best_h, dx)
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    improved = True
        if not improved:
            break
    return best_h, best_c5

def pair_cd_multi(h, dx, target_sum, n_sweeps, budget_s, t0, rng, pair_stride=1):
    n = len(h)
    best_h = h.copy()
    best_c5 = c5_eval(h, dx)
    two_n_minus_1 = 2 * n - 1
    for sweep in range(n_sweeps):
        if budget_s - (time.time() - t0) < 5.0:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            j = (i + pair_stride) % n
            s = best_h[i] + best_h[j]
            lo_v = max(0.0, s - 1.0)
            hi_v = min(1.0, s)
            if hi_v - lo_v < 1e-10:
                continue
            corrs = np.correlate(best_h, 1.0 - best_h, mode='full') * dx
            a = np.zeros(two_n_minus_1)
            for k in range(two_n_minus_1):
                off = k - (n - 1)
                t1 = (1.0 - best_h[i + off]) if (0 <= i + off < n) else 0.0
                t2 = (1.0 - best_h[j + off]) if (0 <= j + off < n) else 0.0
                t3 = best_h[i - off] if (0 <= i - off < n) else 0.0
                t4 = best_h[j - off] if (0 <= j - off < n) else 0.0
                a[k] = dx * (t1 - t2 - t3 + t4)
            c_vec = corrs - a * best_h[i]
            v_new, _ = min_1d_max_linear(a, c_vec, lo_v, hi_v)
            if abs(v_new - best_h[i]) > 1e-10:
                best_h[i] = v_new
                best_h[j] = s - v_new
                new_c5 = c5_eval(best_h, dx)
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    improved = True
        if not improved:
            break
    return best_h, best_c5

def binary_sa(h_init, dx, target_sum, n_iter, T_start, T_end, budget_s, t0, rng):
    n = len(h_init)
    n_ones = int(round(target_sum))
    h = (h_init > 0.5).astype(float)
    diff = n_ones - int(h.sum())
    if diff > 0:
        zeros = np.where(h == 0.0)[0]
        h[rng.choice(zeros, size=diff, replace=False)] = 1.0
    elif diff < 0:
        ones = np.where(h == 1.0)[0]
        h[rng.choice(ones, size=-diff, replace=False)] = 0.0
    best_h = h.copy()
    best_c5 = c5_eval(h, dx)
    cur_c5 = best_c5
    for it in range(n_iter):
        if budget_s - (time.time() - t0) < 3.0:
            break
        frac = it / max(1, n_iter - 1)
        T = T_start * (T_end / T_start) ** frac
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = c5_eval(h, dx)
        if new_c5 <= cur_c5 or (T > 1e-8 and rng.random() < np.exp(-(new_c5 - cur_c5) / T)):
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_c5

def active_set_lp(h, dx, target_sum, n_iters, budget_s, t0, band_frac, top_n, alpha_init):
    n = len(h)
    best_h = h.copy()
    best_c5 = c5_eval(h, dx)
    alpha = alpha_init
    for it in range(n_iters):
        if budget_s - (time.time() - t0) < 3.0:
            break
        corrs = np.correlate(best_h, 1.0 - best_h, mode='full')
        max_c = corrs.max()
        band = max(0.005, band_frac * max_c)
        active_mask = corrs >= (max_c - band)
        top_idx = np.argsort(corrs)[-top_n:]
        active_mask[top_idx] = True
        active_idx = np.where(active_mask)[0]
        if len(active_idx) < 3:
            active_idx = top_idx
        K = len(active_idx)
        m_vars = n + 1
        c_obj = np.zeros(m_vars)
        c_obj[-1] = 1.0
        A_ub = np.zeros((K, m_vars))
        b_ub = np.zeros(K)
        for jj, k in enumerate(active_idx):
            off = k - (n - 1)
            grad = np.zeros(n)
            i_lo = max(0, -off)
            i_hi = min(n - 1, n - 1 - off)
            if i_lo <= i_hi:
                idx = np.arange(i_lo, i_hi + 1)
                grad[idx + off] += (1.0 - best_h[idx])
                grad[idx] += (-best_h[idx + off])
            A_ub[jj, :n] = dx * grad
            A_ub[jj, -1] = -1.0
            b_ub[jj] = -dx * corrs[k]
        A_eq = np.ones((1, m_vars))
        b_eq = np.zeros(1)
        bounds = [(max(-best_h[i] * alpha, -0.5), min((1.0 - best_h[i]) * alpha, 0.5)) for i in range(n)]
        bounds.append((0.0, None))
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs', options={'presolve': True})
            if res.status != 0:
                alpha *= 0.5
                if alpha < 0.002:
                    break
                continue
            delta = res.x[:n]
        except Exception:
            alpha *= 0.5
            if alpha < 0.002:
                break
            continue
        improved = False
        for t in [1.0, 0.8, 0.6, 0.4, 0.3, 0.2, 0.1, 0.05]:
            h_trial = project_feasible(best_h + t * delta, target_sum)
            new_c5 = c5_eval(h_trial, dx)
            if new_c5 < best_c5 - 1e-15:
                best_h = h_trial
                best_c5 = new_c5
                improved = True
                break
        if improved:
            alpha = min(0.4, alpha * 1.05)
        else:
            alpha *= 0.6
            if alpha < 0.002:
                break
    return best_h, best_c5

def softmax_grad(h, dx, T):
    n = len(h)
    corrs = np.correlate(h, 1.0 - h, mode='full') * dx
    m = corrs.max()
    exps = np.exp((corrs - m) / T)
    w = exps / exps.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        off = k - (n - 1)
        i_lo = max(0, -off)
        i_hi = min(n - 1, n - 1 - off)
        if i_lo <= i_hi:
            idx = np.arange(i_lo, i_hi + 1)
            grad[idx + off] += w[k] * dx * (1.0 - h[idx])
            grad[idx] += w[k] * dx * (-h[idx + off])
    return grad

def softmax_descent(h, dx, target_sum, T_start, T_end, n_steps, lr, budget_s, t0):
    best_h = h.copy()
    best_c5 = c5_eval(h, dx)
    for step in range(n_steps):
        if budget_s - (time.time() - t0) < 3.0:
            break
        frac = step / max(1, n_steps - 1)
        T = T_start * (T_end / T_start) ** frac
        lr_cur = lr * (1.0 - 0.5 * frac)
        grad = softmax_grad(h, dx, T)
        cur_c5 = c5_eval(h, dx)
        improved = False
        for t_mult in [1.0, 0.5, 0.25, 0.1, 0.05]:
            t = lr_cur * t_mult
            h_trial = project_feasible(h - t * grad, target_sum)
            new_c5 = c5_eval(h_trial, dx)
            if new_c5 < cur_c5 - 1e-15:
                h = h_trial
                improved = True
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
                break
        if not improved:
            continue
    return best_h, best_c5

def perturb_reopt(h, dx, target_sum, n_rounds, budget_s, t0, rng):
    best_h = h.copy()
    best_c5 = c5_eval(h, dx)
    for r in range(n_rounds):
        if budget_s - (time.time() - t0) < 15.0:
            break
        scale = max(0.02, 0.08 / (r + 1))
        pert = rng.standard_normal(len(h))
        pert -= pert.mean()
        h_p = project_feasible(best_h + scale * pert, target_sum)
        h_opt, c5_opt = pair_cd_exact(h_p, dx, target_sum, 30, budget_s, t0, rng)
        h_opt2, c5_opt2 = active_set_lp(h_opt, dx, target_sum, 50, budget_s, t0, 0.03, 40, 0.2)
        if c5_opt2 < best_c5:
            best_c5 = c5_opt2
            best_h = h_opt2
    return best_h, best_c5
```
