Also, I should think about what makes the Erdős overlap problem special. The key insight is:

1. The function h determines a set of measure 1 in [0,2]

This is related to the "cross-correlation" of the set with its complement.

For a binary function (h ∈ {0,1}), the overlap at shift k is the measure of A \ (A-k) where A is the set where h=1. So we want to minimize max_k |A \ (A-k)|.

This is a well-studied problem. The optimal structure for small measures tends to be a "block" - a single interval. But for measure 1 in [0,2], it's more interesting.

Actually, for a single interval of length 1 in [0,2], say [0,1]:
- At shift k>0: A \ (A-k) = [0,1] \ [-k, 1-k] = [1-k, 1] for k<1, so measure k
- At shift k<0: Similar by symmetry
- Maximum is at k→1: measure → 1

That's terrible. A single interval gives C₅ ≈ 1.

import numpy as np
from scipy.optimize import minimize
import time

try:
    import cvxpy as cp
    HAS_CVXPY = True
except ImportError:
    HAS_CVXPY = False


def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))


def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx


def repair_feasible(h):
    n = len(h)
    h = np.clip(np.asarray(h, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx] + 1e-15)
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx] + 1e-15)
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h


def interp_to_res(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    fx = np.concatenate([[0.0], xs, [2.0]])
    fh = np.concatenate([[h[0]], h, [h[-1]]])
    hn = np.interp(xt, fx, fh)
    return repair_feasible(np.clip(hn, 0, 1))


def conflict_coords(h, top_lags, n):
    """Find coordinates most involved in top lags."""
    center = n - 1
    coords = np.zeros(n)
    for lag_idx in top_lags:
        shift = abs(lag_idx - center)
        if shift == 0:
            coords[:] += 1.0
        elif shift < n:
            m = n - shift
            coords[:m] += 1.0
            coords[shift:] += 1.0
    return coords


def simulated_annealing(h0, t_check, rng, max_iters=10000, t_start=0.01, t_end=0.00005):
    n = len(h0)
    h = h0.copy()
    c = exact_score(h)
    best_h = h.copy()
    best_c = c
    profile = compute_profile(h)

    for it in range(max_iters):
        if t_check() < 50:
            break
        T = t_start * (t_end / t_start) ** (float(it) / float(max_iters))

        # Bias selection towards conflict coordinates
        top5 = np.argsort(profile)[-5:]
        w = conflict_coords(h, top5, n) + 0.1
        w = w / w.sum()
        i = rng.choice(n, p=w)
        j = rng.randint(n)
        if i == j:
            continue

        delta = rng.uniform(0.001, 0.05) * rng.choice([1, -1])
        hi = h[i] - delta
        hj = h[j] + delta
        if hi < 0 or hj > 1:
            continue

        h_new = h.copy()
        h_new[i] = hi
        h_new[j] = hj
        c_new = exact_score(h_new)
        dc = c_new - c
        if dc < 0 or rng.random() < np.exp(-dc / max(T, 1e-12)):
            h = h_new
            c = c_new
            if c < best_c - 1e-14:
                best_c = c
                best_h = h.copy()
            if it % 50 == 0:
                profile = compute_profile(h)

    return best_h, best_c


def targeted_hill_climb(h0, t_check, rng, max_rounds=500):
    n = len(h0)
    h = h0.copy()
    c = exact_score(h)
    profile = compute_profile(h)
    best_c = c
    best_h = h.copy()

    scales_coarse = [0.04, 0.03, 0.02, 0.015, 0.01, 0.007, 0.005]
    scales_fine = [0.003, 0.002, 0.001, 0.0005, 0.0003, 0.0001]

    for rnd in range(max_rounds):
        if t_check() < 30:
            break

        # Update profile periodically
        if rnd % 20 == 0:
            profile = compute_profile(h)
            cm = np.max(profile)
            if cm < best_c - 1e-14:
                best_c = cm
                best_h = h.copy()

        top_lags = np.argsort(profile)[-8:]
        conflict_w = conflict_coords(h, top_lags, n) + 0.05
        conflict_w = conflict_w / conflict_w.sum()

        improved = False
        use_fine = rnd > max_rounds // 2
        scales = scales_fine if use_fine else scales_coarse

        for scale in scales:
            for _ in range(20):
                if t_check() < 30:
                    break
                i = rng.choice(n, p=conflict_w)
                j = rng.choice(n, p=conflict_w)
                if i == j:
                    continue
                for sign in [1, -1]:
                    delta = sign * scale
                    hi = h[i] - delta
                    hj = h[j] + delta
                    if hi < 0 or hj > 1:
                        continue
                    h_new = h.copy()
                    h_new[i] = hi
                    h_new[j] = hj
                    c_new = exact_score(h_new)
                    if c_new < c - 1e-15:
                        h = h_new
                        c = c_new
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            # Try block reversal
            for _ in range(3):
                i = rng.randint(n)
                j = rng.randint(i + 3, n)
                h_new = h.copy()
                h_new[i:j + 1] = h[i:j + 1][::-1]
                c_new = exact_score(h_new)
                if c_new < c - 1e-15:
                    h = h_new
                    c = c_new
                    improved = True
                    break
            if not improved:
                break

    return best_h, best_c


def triple_move_search(h0, t_check, rng, max_iters=3000):
    """Try adjusting triples of coordinates simultaneously."""
    n = len(h0)
    h = h0.copy()
    c = exact_score(h)
    profile = compute_profile(h)

    for it in range(max_iters):
        if t_check() < 25:
            break
        if it % 40 == 0:
            profile = compute_profile(h)
            cm = np.max(profile)
            if cm < c:
                c = cm

        top_lags = np.argsort(profile)[-5:]
        w = conflict_coords(h, top_lags, n) + 0.05
        w = w / w.sum()

        idx = rng.choice(n, size=3, replace=False, p=w)
        i, j, k = idx
        # Move delta from i to j and k (or reverse)
        delta = rng.uniform(0.002, 0.02)
        for pattern in [
            (delta, -delta / 2.0, -delta / 2.0),
            (-delta, delta / 2.0, delta / 2.0),
            (delta, -delta, 0.0),
            (-delta, delta, 0.0),
            (0.0, delta, -delta),
            (0.0, -delta, delta),
        ]:
            di, dj, dk = pattern
            hi = h[i] + di
            hj = h[j] + dj
            hk = h[k] + dk
            if hi < 0 or hi > 1 or hj < 0 or hj > 1 or hk < 0 or hk > 1:
                continue
            h_new = h.copy()
            h_new[i] = hi
            h_new[j] = hj
            h_new[k] = hk
            s = h_new.sum()
            target = n / 2.0
            if abs(s - target) > 1e-12:
                continue
            c_new = exact_score(h_new)
            if c_new < c - 1e-15:
                h = h_new
                c = c_new
                break

    return h, c


def slsqp_refine(h0, maxiter=300):
    n = len(h0)
    dx = 2.0 / n
    M = 2 * n - 1

    def obj(x):
        return x[-1]

    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g

    def eq_fun(x):
        return x[:-1].sum() - n / 2.0

    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g

    def ineq_fun(x):
        hx = x[:-1]
        return x[-1] - np.correlate(hx, 1 - hx, mode="full") * dx

    def ineq_jac(x):
        hx = x[:-1]
        G = np.zeros((M, n))
        center = n - 1
        G[center, :] = dx * (1.0 - 2.0 * hx)
        for d in range(1, n):
            m_pos = center + d
            idx1 = np.arange(n - d)
            G[m_pos, idx1] += -dx * hx[idx1 + d]
            G[m_pos, idx1 + d] += dx * (1.0 - hx[idx1])
            m_neg = center - d
            G[m_neg, idx1] += dx * (1.0 - hx[idx1 + d])
            G[m_neg, idx1 + d] += -dx * hx[idx1]
        J = np.zeros((M, n + 1))
        J[:, :-1] = -G
        J[:, -1] = 1.0
        return J

    t0 = exact_score(h0) + 2e-3
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, t0 + 0.5)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac},
    ]
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-13})
        if res.x is not None and np.isfinite(res.x).all():
            hs = repair_feasible(res.x[:-1])
            if np.isfinite(exact_score(hs)):
                return hs
    except Exception:
        pass
    return None


def lp_subproblem(h0, top_lags, trust=0.03):
    """Solve a small LP on active constraints to get a descent direction."""
    if not HAS_CVXPY:
        return None
    n = len(h0)
    dx = 2.0 / n
    M = 2 * n - 1
    center = n - 1

    # Build gradient for active constraints only
    active = list(top_lags)
    if len(active) == 0:
        return None
    G_a = np.zeros((len(active), n))
    for row, lag_idx in enumerate(active):
        shift = lag_idx - center
        if shift == 0:
            G_a[row, :] = dx * (1.0 - 2.0 * h0)
        elif 0 < abs(shift) < n:
            s = abs(shift)
            m = n - s
            if m <= 0:
                continue
            if shift > 0:
                G_a[row, :m] += -dx * h0[s:]
                G_a[row, s:] += dx * (1.0 - h0[:m])
            else:
                G_a[row, s:] += dx * (1.0 - h0[s:])
                G_a[row, :m] += -dx * h0[:m]

    profile = compute_profile(h0)
    b_a = -profile[active] + G_a @ h0

    h_var = cp.Variable(n)
    t_var = cp.Variable()
    cons = [
        h_var >= 0,
        h_var <= 1,
        cp.sum(h_var) == n / 2.0,
        h_var >= h0 - trust,
        h_var <= h0 + trust,
        G_a @ h_var - t_var <= b_a,
    ]
    obj = cp.Minimize(t_var + 0.01 * cp.norm1(h_var - h0))
    prob = cp.Problem(obj, cons)
    for solver in [cp.ECOS, cp.SCS, cp.CBC]:
        try:
            prob.solve(solver=solver, max_iters=5000, abstol=1e-10)
            break
        except Exception:
            continue
    if h_var.value is None:
        return None
    h_sol = np.array(h_var.value).flatten()
    if not np.isfinite(h_sol).all():
        return None
    return repair_feasible(h_sol)
