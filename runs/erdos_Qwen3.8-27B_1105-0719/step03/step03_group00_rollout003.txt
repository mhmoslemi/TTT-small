def pair_swap_best_t(h, i, j, dx, corr_base, n):
    """Find best t for pair swap h[i]+=t, h[j]-=t using golden section."""
    a_val = h[i]
    b_val = h[j]
    t_lo = max(-a_val, b_val - 1.0)
    t_hi = min(1.0 - a_val, b_val)
    if t_hi - t_lo < 1e-14:
        return None, None
    
    n_full = 2 * n - 1
    # Precompute which k are affected
    # p as first: k in [n-1-p, 2*(n-1)-p]
    # p as second: k in [p, n-1+p]
    k_lo_i = min(i, n - 1 - i)
    k_hi_i = n - 1 + max(i, n - 1 - i)
    k_lo_j = min(j, n - 1 - j)
    k_hi_j = n - 1 + max(j, n - 1 - j)
    
    k_start = max(0, min(k_lo_i, k_lo_j))
    k_end = min(n_full - 1, max(k_hi_i, k_hi_j))
    
    # For unaffected k, the max is:
    unaffected_max = 0.0
    if k_start > 0:
        unaffected_max = max(unaffected_max, corr_base[:k_start].max())
    if k_end < n_full - 1:
        unaffected_max = max(unaffected_max, corr_base[k_end+1:].max())
    
    # For affected k, compute c_k and d_k
    # corr_k(t) = corr_k(0) + c_k * t + d_k * t^2
    affected_k = np.arange(k_start, k_end + 1)
    c_vals = np.zeros(len(affected_k))
    d_vals = np.zeros(len(affected_k))
    
    for idx, k in enumerate(affected_k):
        offset = k - (n - 1)
        lo = max(0, -offset)
        hi = min(n - 1, n - 1 - offset)
        if lo > hi:
            continue
        # h[i] appears as first if i in [lo, hi]
        # h[j] appears as first if j in [lo, hi]
        # h[i] appears as second if i-offset in [lo, hi] (i.e., i = a+offset for a in [lo,hi])
        # h[j] appears as second if j-offset in [lo, hi]
        
        c = 0.0
        d = 0.0
        
        # Contribution from h[i] changing to h[i]+t:
        if i >= lo and i <= hi:
            # i is "first": term h[i]*(1-h[i+offset]) → (h_i+t)*(1-h[i+offset])
            # Change: t * (1-h[i+offset])
            other = h[i + offset]
            c += (1.0 - other)
            if i + offset == j:
                # Second factor also changes: (h_i+t)*(1-h_j+t)
                # Already counted t*(1-h_j) from h_i change
                # Now h_j changes too: need to add t^2 term
                # (h_i+t)*(1-h_j+t) = h_i(1-h_j) + h_i*t + (1-h_j)*t + t^2
                # vs h_i(1-h_j), so change = (h_i + 1-h_j)*t + t^2
                # But we already added (1-h_j)*t from i being first.
                # The j change as second: h[i]*(1-(h_j-t)) = h[i]*(1-h_j+t) → change h[i]*t
                # Wait, I'm double counting. Let me think more carefully.
                
                # The term is h[i]*(1-h[i+offset]) where i+offset = j
                # After swap: (h_i+t)*(1-(h_j-t)) = (h_i+t)*(1-h_j+t)
                # = h_i(1-h_j) + h_i*t + (1-h_j)*t + t^2
                # Change = (h_i + 1-h_j)*t + t^2
                # So c += (h_i + 1 - h_j), d += 1
                # But I already added (1-h_j) to c for i being first.
                # So I should add h_i to c and 1 to d.
                c += a_val  # h_i
                d += 1.0
            # If i+offset != j, just c += (1-h[i+offset]) which is already done
            # Wait, I need to be more careful. The term h[i]*(1-h[i+offset]):
            # If i+offset == j: after swap both factors change.
            #   (h_i+t)*(1-(h_j-t)) = (h_i+t)*(1-h_j+t)
            #   = h_i*(1-h_j) + [h_i + (1-h_j)]*t + t^2
            #   Δcorr = [h_i + (1-h_j)]*t + t^2
            #   So c contribution = h_i + (1-h_j), d contribution = 1
            #   But wait, I already did c += (1-h[i+offset]) = (1-h_j) above.
            #   So I need to additionally add h_i to c and 1 to d.
            #   That's what I did above. ✓
            # If i+offset != j: only first factor changes.
            #   (h_i+t)*(1-h[i+offset]) = h_i*(1-h[i+offset]) + t*(1-h[i+offset])
            #   Δcorr = t*(1-h[i+offset])
            #   c contribution = (1-h[i+offset]) ✓ (already done)
        elif i - offset >= lo and i - offset <= hi and i - offset >= 0 and i - offset <= n - 1:
            # i is "second": term h[i-offset]*(1-h[i]) → h[i-offset]*(1-(h_i+t))
            # Change: -h[i-offset]*t
            c += -h[i - offset]
            if i - offset == j:
                # Both change: (h_j-t)*(1-(h_i+t)) = (h_j-t)*(1-h_i-t)
                # = h_j*(1-h_i) - h_j*t - t + h_i*t + t^2
                # = h_j*(1-h_i) + (h_i - h_j - 1)*t + t^2
                # Δcorr = (h_i - h_j - 1)*t + t^2
                # I already added -h[i-offset] = -h_j to c.
                # Need to add (h_i - 1) to c and 1 to d.
                # Wait: total c should be (h_i - h_j - 1), I have -h_j so need h_i - 1 more.
                c += a_val - 1.0
                d += 1.0
            # If i-offset != j: only second factor changes.
            # h[i-offset]*(1-(h_i+t)) = h[i-offset]*(1-h_i) - h[i-offset]*t
            # Δcorr = -h[i-offset]*t ✓ (already done)
        
        # Contribution from h[j] changing to h[j]-t:
        if j >= lo and j <= hi:
            # j is "first": term h[j]*(1-h[j+offset]) → (h_j-t)*(1-h[j+offset])
            # Change: -t*(1-h[j+offset])
            other = h[j + offset]
            c += -(1.0 - other)
            if j + offset == i:
                # Cross term: (h_j-t)*(1-(h_i+t)) = (h_j-t)*(1-h_i-t)
                # Already handled in the "i as second" case above.
                # So we should NOT double count. Let me reconsider.
                pass  # Skip, already handled
        elif j - offset >= lo and j - offset <= hi and j - offset >= 0 and j - offset <= n - 1:
            # j is "second": term h[j-offset]*(1-h[j]) → h[j-offset]*(1-(h_j-t))
            # Change: +h[j-offset]*t
            c += h[j - offset]
            if j - offset == i:
                # Cross term: already handled in "i as first" case.
                pass
        
        c_vals[idx] = c
        d_vals[idx] = d
    
    # Now minimize f(t) = max(unaffected_max, max_affected(corr_base[k] + c_k*t + d_k*t^2)) over [t_lo, t_hi]
    # This is convex in t (max of affine and convex quadratic functions).
    
    best_t = None
    best_val = float('inf')
    
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a_g = t_lo
    b_g = t_hi
    
    def eval_f(t):
        vals = corr_base[affected_k] + c_vals * t + d_vals * t * t
        return max(unaffected_max, vals.max() if len(vals) > 0 else 0.0)
    
    # Sample a few points first
    n_samples = 21
    t_samples = np.linspace(t_lo, t_hi, n_samples)
    f_samples = np.array([eval_f(t) for t in t_samples])
    best_idx = f_samples.argmin()
    
    # Refine with golden section around the best sample
    lo_g = t_samples[max(0, best_idx - 1)]
    hi_g = t_samples[min(n_samples - 1, best_idx + 1)]
    
    # Golden section
    c_g = hi_g - gr * (hi_g - lo_g)
    d_g = lo_g + gr * (hi_g - lo_g)
    fc = eval_f(c_g)
    fd = eval_f(d_g)
    
    for _ in range(40):
        if fc < fd:
            hi_g = d_g
            d_g = c_g
            fd = fc
            c_g = hi_g - gr * (hi_g - lo_g)
            fc = eval_f(c_g)
        else:
            lo_g = c_g
            c_g = d_g
            fc = fd
            d_g = lo_g + gr * (hi_g - lo_g)
            fd = eval_f(d_g)
    
    t_mid = (lo_g + hi_g) / 2.0
    f_mid = eval_f(t_mid)
    
    if f_mid < best_val:
        best_t = t_mid
        best_val = f_mid
    
    # Also check endpoints
    for t_check in [t_lo, t_hi, t_mid]:
        fc = eval_f(t_check)
        if fc < best_val:
            best_val = fc
            best_t = t_check
    
    if best_t is None or best_val >= corr_base.max() - 1e-14:
        return None, None
    
    return best_t, best_val * dx

def soft_slack_lp(h, dx, top_idx, active_idx, alpha, n):
    """Solve soft-slack LP for direction finding."""
    K = len(active_idx)
    if K == 0:
        return None
    
    n_vars = n + K + 1  # delta (n), slacks (K), q (1)
    c_obj = np.zeros(n_vars)
    c_obj[-1] = 0.35  # coefficient on q
    for j in range(K):
        c_obj[n + j] = 5.0  # slack penalty
    
    # Constraints: dx*grad_j . delta + s_j - q <= -corr_j
    # Rearranged: dx*grad_j . delta + s_j - q + corr_j <= 0
    A_ub = np.zeros((K, n_vars))
    b_ub = np.zeros(K)
    
    corrs = corr_of_h(h)
    
    for j, k in enumerate(active_idx):
        g = grad_corr_k(h, k, n)
        A_ub[j, :n] = dx * g
        A_ub[j, n + j] = 1.0
        A_ub[j, -1] = -1.0
        b_ub[j] = -corrs[k]
    
    # Equality: sum(delta) = 0
    A_eq = np.zeros((1, n_vars))
    A_eq[0, :n] = 1.0
    
    bounds = []
    for i in range(n):
        bounds.append((-h[i] * alpha, (1.0 - h[i]) * alpha))
    for j in range(K):
        bounds.append((0.0, None))
    bounds.append((0.0, None))
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.zeros(1),
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.status != 0:
            return None
        return res.x[:n]
    except Exception:
        return None

def line_search(h, direction, dx, target_sum, cur_c5, n):
    """Exact line search along direction."""
    best_h = h.copy()
    best_val = cur_c5
    for t in [1.0, 0.8, 0.6, 0.5, 0.4, 0.3, 0.2, 0.15, 0.1, 0.07, 0.05, 0.03, 0.02, 0.01]:
        h_c = h + t * direction
        h_c = project_feasible(h_c, target_sum)
        c5_val = c5_of_h(h_c, dx)
        if c5_val < best_val - 1e-15:
            best_val = c5_val
            best_h = h_c
    return best_h, best_val

def transfer_grid(h_src, n_target):
    n_src = len(h_src)
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_tgt = (np.arange(n_target) + 0.5) * (2.0 / n_target)
    h_interp = np.interp(x_tgt, x_src, h_src)
    return h_interp

def pair_swap_phase(h, dx, target_sum, corr_base, n, rng, max_pairs):
    """Try pair swaps to improve C5."""
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    improved = False
    
    for _ in range(max_pairs):
        # Choose pair: mix of adjacent and random
        if rng.random() < 0.4:
            i = rng.randint(0, n - 1)
            j = i + 1
        else:
            i = rng.randint(0, n)
            j = rng.randint(0, n)
            if i == j:
                continue
        
        # Skip if both are at boundaries
        if (h[i] < 1e-10 and h[j] < 1e-10) or (h[i] > 1-1e-10 and h[j] > 1-1e-10):
            continue
        # Skip if one is 0 and other is 1 (or vice versa) - no room to move
        if abs(h[i] - h[j]) < 1e-10:
            continue
        
        t_best, c5_est = pair_swap_best_t(h, i, j, dx, corr_base, n)
        if t_best is None:
            continue
        
        # Apply the swap and verify
        h_new = h.copy()
        h_new[i] += t_best
        h_new[j] -= t_best
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_feasible(h_new, target_sum)
        c5_new = c5_of_h(h_new, dx)
        
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            h = h_new
            best_h = h_new.copy()
            corr_base = corr_of_h(h)
            improved = True
    
    return best_h, best_c5, improved

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    
    def remaining():
        return budget_s - (time.time() - t0)
    
    best_h = None
    best_c5 = float('inf')
    
    # Load parent
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = np.full(126, 0.5)
    
    n = len(h0)
    dx = 2.0 / n
    target_sum = n / 2.0
    h0 = project_feasible(h0, target_sum)
    best_h = h0.copy()
    best_c5 = c5_of_h(h0, dx)
    corr_base = corr_of_h(h0)
    
    # Phase 1: LP directions with soft slack
    alpha = 0.3
    for it in range(300):
        if remaining() < 30:
            break
        
        # Select active set
        max_corr = corr_base.max()
        n_top = max(20, min(50, int(0.25 * (2 * n - 1))))
        top_idx = np.argsort(corr_base)[-n_top:]
        margin = max(0.05, 0.02 * max_corr)
        active_mask = corr_base >= (max_corr - margin)
        active_mask[top_idx] = True
        active_idx = np.where(active_mask)[0]
        if len(active_idx) < 5:
            active_idx = top_idx
        
        # Solve LP
        direction = soft_slack_lp(best_h, dx, top_idx, active_idx, alpha, n)
        if direction is None:
            alpha *= 0.7
            if alpha < 0.01:
                alpha = 0.2
                rng.random()  # advance rng
            continue
        
        # Line search
        h_new, c5_new = line_search(best_h, direction, dx, target_sum, best_c5, n)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new
            corr_base = corr_of_h(best_h)
            alpha = min(0.4, alpha * 1.05)
        else:
            alpha *= 0.6
    
    # Phase 2: Pair swap local search
    if remaining() > 40:
        n_it = 0
        while remaining() > 60 and n_it < 50:
            h_new, c5_new, improved = pair_swap_phase(best_h, dx, target_sum, corr_base, n, rng, 200)
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                best_h = h_new
                corr_base = corr_of_h(best_h)
            n_it += 1
    
    # Phase 3: Interleave LP and pair swaps
    if remaining() > 60:
        alpha = 0.2
        for it in range(400):
            if remaining() < 30:
                break
            if it % 5 == 0:
                # Pair swap round
                h_new, c5_new, _ = pair_swap_phase(best_h, dx, target_sum, corr_base, n, rng, 100)
                if c5_new < best_c5 - 1e-14:
                    best_c5 = c5_new
                    best_h = h_new
                    corr_base = corr_of_h(best_h)
            else:
                # LP round
                max_corr = corr_base.max()
                n_top = max(15, min(40, int(0.2 * (2 * n - 1))))
                top_idx = np.argsort(corr_base)[-n_top:]
                margin = max(0.05, 0.02 * max_corr)
                active_mask = corr_base >= (max_corr - margin)
                active_mask[top_idx] = True
                active_idx = np.where(active_mask)[0]
                if len(active_idx) < 5:
                    active_idx = top_idx
                
                direction = soft_slack_lp(best_h, dx, top_idx, active_idx, alpha, n)
                if direction is not None:
                    h_new, c5_new = line_search(best_h, direction, dx, target_sum, best_c5, n)
                    if c5_new < best_c5 - 1e-14:
                        best_c5 = c5_new
                        best_h = h_new
                        corr_base = corr_of_h(best_h)
                        alpha = min(0.4, alpha * 1.05)
                    else:
                        alpha *= 0.7
    
    # Phase 4: Perturbation + re-optimization
    if remaining() > 50:
        for r in range(8):
            if remaining() < 20:
                break
            scale = max(0.03, 0.15 / (r + 1))
            pert = rng.standard_normal(n)
            pert -= pert.mean()
            h_p = best_h + scale * pert
            h_p = project_feasible(h_p, target_sum)
            corr_p = corr_of_h(h_p)
            
            # Quick LP polish
            for it in range(30):
                if remaining() < 15:
                    break
                max_corr = corr_p.max()
