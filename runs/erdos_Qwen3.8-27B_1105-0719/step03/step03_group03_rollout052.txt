def simulated_annealing(h0, max_iters, time_fn, rng):
    n = len(h0)
    h = h0.copy()

    best_h = h.copy()

    T_min = 0.00005
    cooling = 0.997

    for it in range(max_iters):
        if time_fn() < 5:
            break
        T = max(T_min, T * cooling)
        i = rng.randint(n)
        j = rng.randint(n - 1)
        if j >= i:
            j += 1
        if i == j:
            continue
        max_fwd = min(h[i], 1.0 - h[j])
        max_bwd = min(h[j], 1.0 - h[i])
        if max_fwd < 1e-6 and max_bwd < 1e-6:
            continue
        rng_val = rng.uniform(-1, 1)
        if rng_val > 0 and max_fwd > 1e-6:
            delta = rng_val * max_fwd
            h[i] -= delta
            h[j] += delta
        elif max_bwd > 1e-6:

            h[i] -= delta
            h[j] += delta
        else:

                cur_c = c_new
            else:

                J[m, :] = 0
                J[m, -1] = 1.0
                J[m, :] += -dx * (1.0 - 2.0 * hh)
            else:
                j_lo = max(0, -d)
                j_hi = min(n - 1, n - 1 - d)
                J[m, j_lo:j_hi + 1] += -dx * (1.0 - hh[j_lo + d:j_hi + 1 + d])
                if d > 0:
                    J[m, d:n] += dx * hh[0:n - d]
                else:
                    J[m, 0:n + d] += dx * hh[-d:n]
        return J

    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, t0 + 1.0)]
    cons = [
        {'type': 'eq', 'fun': eq_fn, 'jac': eq_jac_fn},
        {'type': 'ineq', 'fun': ineq_fn, 'jac': ineq_jac_fn},
    ]
    try:
        res = minimize(obj_fn, x0, method='SLSQP', jac=obj_jac_fn,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': tol})
        if res.success and res.x is not None and np.isfinite(res.x).all():
            hs = repair_feasible(res.x[:-1])
            cs = exact_score(hs)
            if np.isfinite(cs):
                return hs, cs
    except Exception:
        pass
    return None, None


def multi_resolution(bh, bc, time_fn, rng):
    n0 = len(bh)
    candidates = []
    for delta in [-10, -8, -6, -4, -2, 2, 4, 6, 8, 10, 14, 18]:
        nd = n0 + delta
        if 20 <= nd <= 160:
            candidates.append(nd)
    candidates = sorted(set(candidates), key=lambda d: abs(d - n0))
    for nd in candidates[:8]:
        if time_fn() < 30:
            break
        hi = interp_resample(bh, nd)
        ci = exact_score(hi)
        if ci < bc + 0.03:
            bh2, bc2 = coord_descent(hi, max_sweeps=50, time_fn=time_fn)
            if bc2 < bc - 1e-12:
                bh, bc = bh2, bc2
                bh2, bc2 = subgradient_opt(bh, max_iters=100, time_fn=time_fn)
                if bc2 < bc - 1e-12:
                    bh, bc = bh2, bc2
    return bh, bc


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()
    rng = np.random.RandomState(seed)

    def tl():
        return budget_s + t_start - time.perf_counter()

    parent = None
    try:
        parent = np.array(initial_h_values, dtype=np.float64)
    except (NameError, Exception):
        pass
    if parent is None or len(parent) < 4:
        parent = np.full(70, 0.5)

    parent = repair_feasible(np.clip(parent, 0.0, 1.0))
    best_h = parent.copy()
    best_c = exact_score(parent)

    # Symmetrize
    h_sym = repair_feasible((parent + parent[::-1]) / 2.0)
    c_sym = exact_score(h_sym)
    if c_sym < best_c - 1e-12:
        best_h, best_c = h_sym, c_sym

    # Phase 1: Coordinate descent (fast, exact per-coordinate)
    if tl() > 60:
        bh, bc = coord_descent(best_h, max_sweeps=200, time_fn=tl)
        if bc < best_c - 1e-14:
            best_h, best_c = bh, bc

    # Phase 2: Subgradient refinement
    if tl() > 80:
        bh, bc = subgradient_opt(best_h, max_iters=200, time_fn=tl)
        if bc < best_c - 1e-14:
            best_h, best_c = bh, bc
        # Another coord descent after subgradient
        bh, bc = coord_descent(best_h, max_sweeps=100, time_fn=tl)
        if bc < best_c - 1e-14:
            best_h, best_c = bh, bc

    # Phase 3: Multi-resolution exploration
    if tl() > 100:
        bh, bc = multi_resolution(best_h, best_c, tl, rng)
        if bc < best_c - 1e-12:
            best_h, best_c = bh, bc

    # Phase 4: Simulated annealing for global search
    if tl() > 120:
        bh, bc = simulated_annealing(best_h, max_iters=5000, time_fn=tl, rng=rng)
        if bc < best_c - 1e-14:
            best_h, best_c = bh, bc
        # Polish after SA
        bh, bc = coord_descent(best_h, max_sweeps=100, time_fn=tl)
        if bc < best_c - 1e-14:
            best_h, best_c = bh, bc

    # Phase 5: Try different starting points
    if tl() > 150:
        n_cur = len(best_h)
        # Periodic constructions
        for period in [7, 9, 11, 13, 15, 17, 5, 19, 21, 23]:
            if tl() < 20:
                break
            x = np.linspace(0, 2.0, n_cur, endpoint=False)
            phase = (x % (2.0 / period)) / (2.0 / period)
            for duty in [0.3, 0.35, 0.4, 0.45, 0.5]:
                h_p = (phase < duty).astype(float)
                h_p = repair_feasible(h_p)
                c_p = exact_score(h_p)
                if c_p < best_c + 0.05:
                    bh, bc = coord_descent(h_p, max_sweeps=80, time_fn=tl)
                    if bc < best_c - 1e-14:
                        best_h, best_c = bh, bc
                    bh, bc = subgradient_opt(best_h, max_iters=80, time_fn=tl)
                    if bc < best_c - 1e-14:
                        best_h, best_c = bh, bc

    # Phase 6: SLSQP polish
    if tl() > 80:
        hs, cs = slsqp_refine(best_h, maxiter=300, tol=1e-11)
        if hs is not None and cs < best_c - 1e-14:
            best_h, best_c = hs, cs
        # Coord descent after SLSQP
        bh, bc = coord_descent(best_h, max_sweeps=100, time_fn=tl)
        if bc < best_c - 1e-14:
            best_h, best_c = bh, bc

    # Phase 7: More subgradient rounds with different beams
    if tl() > 60:
        for beam_val in [0.001, 0.003, 0.01, 0.02]:
            if tl() < 15:
                break
            bh, bc = subgradient_opt(best_h, max_iters=100, time_fn=tl, beam=beam_val)
            if bc < best_c - 1e-14:
                best_h, best_c = bh, bc
            bh, bc = coord_descent(best_h, max_sweeps=50, time_fn=tl)
            if bc < best_c - 1e-14:
                best_h, best_c = bh, bc

    # Phase 8: Random restarts from perturbed best
    if tl() > 100:
        for eps in [0.02, 0.04, 0.06, 0.08, 0.12]:
            if tl() < 20:
                break
            z = rng.randn(len(best_h))
            z -= z.mean()
            z /= (np.linalg.norm(z) + 1e-15)
            hp = repair_feasible(best_h + eps * z)
            cp_ = exact_score(hp)
            if cp_ < best_c + 0.02:
                bh, bc = coord_descent(hp, max_sweeps=80, time_fn=tl)
                if bc < best_c - 1e-14:
                    best_h, best_c = bh, bc
                bh, bc = subgradient_opt(best_h, max_iters=80, time_fn=tl)
                if bc < best_c - 1e-14:
                    best_h, best_c = bh, bc

    # Phase 9: Final multi-resolution
    if tl() > 80:
        bh, bc = multi_resolution(best_h, best_c, tl, rng)
        if bc < best_c - 1e-12:
            best_h, best_c = bh, bc

    # Phase 10: Final polish
    if tl() > 40:
        bh, bc = coord_descent(best_h, max_sweeps=100, time_fn=tl)
        if bc < best_c - 1e-14:
            best_h, best_c = bh, bc
        hs, cs = slsqp_refine(best_h, maxiter=200, tol=1e-12)
        if hs is not None and cs < best_c - 1e-14:
            best_h, best_c = hs, cs
        bh, bc = coord_descent(best_h, max_sweeps=50, time_fn=tl)
        if bc < best_c - 1e-14:
            best_h, best_c = bh, bc

    # Final validation
    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)

    try:
        er = evaluate_erdos_solution(best_h)
        if isinstance(er, (float, int)):
            final_c = min(final_c, float(er))
        elif isinstance(er, tuple) and len(er) >= 2:
            final_c = min(final_c, float(er[0]))
    except Exception:
        pass

    return best_h, final_c, len(best_h)
