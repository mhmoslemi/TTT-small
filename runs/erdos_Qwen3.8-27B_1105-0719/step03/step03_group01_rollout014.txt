def lse_gradient(h, T):
    """Compute the LSE-smoothed gradient direction (zero-mean)."""
    n = len(h)
    raw = np.correlate(h, 1.0 - h, mode='full')
    m = len(raw)  # 2n-1
    
    # Softmax weights
    rmax = raw.max()
    w = np.exp((raw - rmax) / T)
    w /= w.sum()
    
    # Compute weighted gradient
    # G[p, j] = d(raw[p])/dh[j]
    # We compute grad = w @ G = sum_p w[p] * G[p, :]
    grad = np.zeros(n)
    
    k = np.arange(m) - (n - 1)  # lags: -(n-1), ..., 0, ..., n-1
    
    for p in range(m):
        if w[p] < 1e-15:
            continue
        kk = k[p]
        # Condition 1: j in [max(0, p-n+1), min(n-1, p)]
        j1_lo = max(0, p - n + 1)
        j1_hi = min(n - 1, p)
        if j1_lo <= j1_hi:
            js = np.arange(j1_lo, j1_hi + 1)
            grad[js] += w[p] * (1.0 - h[js - kk])

            grad[js] += w[p] * (-h[js + kk])

    grad -= grad.mean()

    n = len(h)

    pos = grad > 1e-14

    if neg.any():

    alpha_max = max(alpha_max, 0.0)

    best_h = h
    best_c5 = c5_of_h(h, dx)

    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    n_total_steps = n_temps * steps_per_temp
    
    for t_idx in range(n_temps):

            break
        frac = t_idx / max(n_temps - 1, 1)
        T = T_start * (T_end / T_start) ** frac

            h_new, c5_new, improved = lse_line_search(h, grad, target_sum, dx)

                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
            else:
                break  # Can't improve at this temperature, move to next
