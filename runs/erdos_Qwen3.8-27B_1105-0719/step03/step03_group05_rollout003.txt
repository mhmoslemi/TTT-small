**Another new idea: Use the "method of conditional gradients" (Frank-Wolfe)**

The feasible set is a polytope (box constraint + sum constraint). The Frank-Wolfe algorithm:
1. Compute gradient of objective
2. Find the vertex of the polytope that minimizes the linear approximation (this is the "linear oracle")
3. Take a step towards that vertex

For our problem, the linear oracle is:

subject to: sum(v) = n/2, 0 ≤ v[i] ≤ 1

This is solved by sorting grad and greedily assigning 1 to the smallest grad entries and 0 to the largest, with one fractional value.

```python
def frank_wolfe(h, max_iters=1000, beta=50):
    n = len(h)
    dx = 2.0 / n
    M = 2*n - 1
    h = h.copy()
    best_c = exact_score(h)
    
    for it in range(max_iters):
        profile = compute_profile(h)
        # LSE weights
        scaled = beta * profile
        shifted = scaled - scaled.max()
        weights = np.exp(shifted)
        weights /= weights.sum()
        
        # Gradient
        G = build_gradient_matrix(h, n, dx)
        grad = G.T @ weights
        grad -= grad.mean()  # Project onto sum constraint
        
        # Frank-Wolfe linear oracle
        # minimize sum_i grad[i] * v[i] s.t. sum(v) = n/2, 0 <= v <= 1
        # Solution: sort grad, assign 1 to n/2 smallest
        sorted_idx = np.argsort(grad)
        v = np.zeros(n)
        v[sorted_idx[:n//2]] = 1.0
        # (Since n is even and sum must be n/2, we get exactly n/2 ones)
        
        # Line search
        gamma = 2.0 / (it + 2)  # Standard FW step size
        h_new = (1 - gamma) * h + gamma * v
        h_new = np.clip(h_new, 0, 1)
        h_new = repair_feasible(h_new)
        
        new_c = exact_score(h_new)
        if new_c < best_c:
            best_c = new_c
        h = h_new
        
        # Convergence check
        if it % 50 == 0:
            fw_gap = np.dot(grad, h - v)
            if fw_gap < 1e-10:
                break
    
    return h, best_c
```

```python
        for j in range(n):
            lo_j, hi_j = xs[j], xs[j+1]
            overlap = min(hi_k, hi_j) - max(lo_k, lo_j)
            if overlap > 0:
                total_mass += h[j] * overlap
                total_width += overlap
        h_interp[k] = total_mass / total_width if total_width > 1e-15 else 0.5
    h_interp = np.clip(h_interp, 0.0, 1.0)
    return repair_feasible(h_interp)

def frank_wolfe_step(h, grad):
    n = len(h)
    sorted_idx = np.argsort(grad)
    v = np.zeros(n)
    v[sorted_idx[:n // 2]] = 1.0
    return v

def fw_descent(h, beta=50.0, max_iters=2000):
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    best_h = h.copy()
    for it in range(max_iters):
        grad, profile = compute_gradient(h, beta)
        v = frank_wolfe_step(h, grad)
        # FW gap
        fw_gap = np.dot(grad, h - v)
        gamma = 2.0 / (it + 2)
        h_new = (1.0 - gamma) * h + gamma * v
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = repair_feasible(h_new)
        new_c = exact_score(h_new)
        if new_c <= best_c + 1e-15:
            h = h_new
            if new_c < best_c - 1e-15:
                best_c = new_c
                best_h = h.copy()
        if it % 200 == 0 and fw_gap < 1e-8:
            break
    return best_h, best_c

def pairwise_descent(h, beta=80.0, max_iters=500, n_candidates=10):
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    for it in range(max_iters):
        grad, profile = compute_gradient(h, beta)
        # Find best pairs to move mass
        order_high = np.argsort(-grad)[:n_candidates]
        order_low = np.argsort(grad)[:n_candidates]
        best_delta_c = best_c
        best_move = None
        for i in order_high:
            for j in order_low:
                if i == j:
                    continue
                # Move mass from i to j: h[i] - delta, h[j] + delta
                max_delta = min(h[i], 1.0 - h[j])
                # Try several deltas
                for frac in [0.1, 0.2, 0.35, 0.5, 0.7, 1.0]:
                    delta = max_delta * frac
                    if delta < 1e-10:
                        continue
                    h_try = h.copy()
                    h_try[i] -= delta
                    h_try[j] += delta
                    h_try = np.clip(h_try, 0.0, 1.0)
                    h_try = repair_feasible(h_try)
                    c_try = exact_score(h_try)
                    if c_try < best_delta_c - 1e-14:
                        best_delta_c = c_try
                        best_move = (i, j, delta)
        if best_move is not None:
            i, j, delta = best_move
            h[i] -= delta
            h[j] += delta
            h = np.clip(h, 0.0, 1.0)
            h = repair_feasible(h)
            best_c = best_delta_c
        else:
            break
    return h, best_c

def lse_descent(h, beta=80.0, max_iters=300, alpha_init=0.004):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best_c = exact_score(h)
    alpha = alpha_init
    stuck = 0
    for it in range(max_iters):
        grad, profile = compute_gradient(h, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad_dir = grad / gnorm
        improved = False
        for a in [alpha, alpha*0.5, alpha*0.25, alpha*0.1, alpha*0.04]:
            h_try = repair_feasible(h - a * grad_dir)
            s = exact_score(h_try)
            if s < best_c - 1e-14:
                h = h_try
                best_c = s
                improved = True
                alpha = min(alpha * 1.12, 0.03)
                stuck = 0
                break
            elif s < float(np.max(compute_profile(h))) - 1e-14:
                h = h_try
                improved = True
                break
        if not improved:
            alpha *= 0.55
            stuck += 1
            if stuck > 30 or alpha < 1e-10:
                break
    return h, best_c

def active_set_descent(h, max_iters=300, n_active=12, alpha_init=0.005):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best_c = exact_score(h)
    alpha = alpha_init
    stuck = 0
    for it in range(max_iters):
        profile = compute_profile(h)
        c = float(np.max(profile))
        if c + 1e-14 < best_c:
            best_c = c
            stuck = 0
            alpha = min(alpha * 1.1, 0.03)
        margin = max(0.001, 3.0 * (c - float(np.median(profile))))
        active = np.where(profile >= c - margin)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(profile))])
        if len(active) > n_active:
            top_idx = np.argsort(profile[active])[-n_active:]
            active = active[top_idx]
        G = build_gradient_matrix(h, n, dx)
        G_a = G[active, :]
        w = np.ones(len(active)) / len(active)
        grad = G_a.T @ w
        grad -= grad.mean()
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad /= gnorm
        improved = False
        for a in [alpha, alpha*0.5, alpha*0.25, alpha*0.1, alpha*0.04]:
            h_try = repair_feasible(h - a * grad)
            s = exact_score(h_try)
            if s < best_c - 1e-14:
                h = h_try
                best_c = s
                improved = True
                break
            elif s < c - 1e-14:
                h = h_try
                improved = True
                break
        if not improved:
            alpha *= 0.5
            stuck += 1
            if stuck > 40 or alpha < 1e-10:
                break
    return h, best_c

def slsqp_solve(h0, maxiter=200):
    n = len(h0)
    dx = 2.0 / n
    M = 2 * n - 1
    c0 = exact_score(h0)
    def obj(x):
        return x[-1]
    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    def eq_fun(x):
        return x[:-1].sum() - n / 2.0
    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g
    def ineq_fun(x):
        h = np.clip(x[:-1], 0, 1)
        t = x[-1]
        profile = np.correlate(h, 1 - h, mode="full") * dx
        return t - profile
    def ineq_jac(x):
        h = np.clip(x[:-1], 0, 1)
        G = build_gradient_matrix(h, n, dx)
        J = np.zeros((M, n + 1))
        J[:, :-1] = -G
        J[:, -1] = 1.0
        return J
    x0 = np.concatenate([h0, [c0 + 1e-3]])
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 0.5)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None and np.isfinite(res.x).all():
            h_new = repair_feasible(res.x[:-1])
            c_new = exact_score(h_new)
            if c_new <= c0 + 1e-9:
                return h_new, c_new
    except Exception:
        pass
    return None, None

def greedy_binary(n, rng):
    h = np.zeros(n)
    target = n // 2
    placed = 0
    available = list(range(n))
    for _ in range(target):
        best_idx = -1
        best_c = float('inf')
        # Try each available position, pick the one that minimizes max profile
        # For speed, sample a subset
        candidates = available if len(available) <= 30 else rng.choice(available, 30, replace=False)
        for idx in candidates:
            h[idx] = 1.0
            c = exact_score(h)
            if c < best_c:
                best_c = c
                best_idx = idx
            h[idx] = 0.0
        h[best_idx] = 1.0
        available.remove(best_idx)
        placed += 1
    return repair_feasible(h)

def random_binary(n, rng):
    idx = rng.permutation(n)[:n // 2]
    h = np.zeros(n)
    h[idx] = 1.0
    return h

def hybrid_descent(h, beta=60.0, max_iters=500, rng=None):
    """Alternate between FW and LSE gradient steps."""
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    best_h = h.copy()
    alpha = 0.005
    for it in range(max_iters):
        if it % 3 == 2:
            # FW step
            grad, profile = compute_gradient(h, beta)
            v = frank_wolfe_step(h, grad)
            gamma = 2.0 / (it + 2)
            h_new = (1.0 - gamma) * h + gamma * v
            h_new = np.clip(h_new, 0.0, 1.0)
            h_new = repair_feasible(h_new)
            c_new = exact_score(h_new)
            if c_new <= best_c + 1e-12:
                h = h_new
                if c_new < best_c - 1e-14:
                    best_c = c_new
                    best_h = h.copy()
        else:
            # LSE gradient step
            grad, profile = compute_gradient(h, beta)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break
            grad_dir = grad / gnorm
            improved = False
            for a in [alpha, alpha*0.5, alpha*0.25, alpha*0.1]:
                h_try = repair_feasible(h - a * grad_dir)
                s = exact_score(h_try)
                if s < best_c - 1e-14:
                    h = h_try
                    best_c = s
                    best_h = h.copy()
                    improved = True
                    alpha = min(alpha * 1.1, 0.025)
                    break
                elif s < float(np.max(compute_profile(h))) - 1e-14:
                    h = h_try
                    improved = True
                    break
            if not improved:
                alpha *= 0.5
                if alpha < 1e-10:
                    break
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()
    rng = np.random.RandomState(seed)
    np.random.seed(seed)

    def time_left():
        return budget_s + t_start - time.perf_counter()

    # Get parent
    parent = None
    try:
        parent = np.array(initial_h_values, dtype=np.float64)
    except NameError:
        pass
    except Exception:
        pass

    if parent is None or len(parent) < 4:
        parent = np.full(70, 0.5)

    parent = np.clip(parent, 0.0, 1.0)
    parent = repair_feasible(parent)
    c_parent = exact_score(parent)

    best_h = parent.copy()
    best_c = c_parent

    # Phase 1: Hybrid FW+LSE descent on parent
    if time_left() > 200:
        h1, c1 = hybrid_descent(best_h, beta=50, max_iters=800, rng=rng)
        if c1 < best_c - 1e-13:
            best_h = h1
            best_c = c1

    # Phase 2: Pairwise descent
    if time_left() > 150:
        h2, c2 = pairwise_descent(best_h, beta=80, max_iters=200)
        if c2 < best_c - 1e-13:
            best_h = h2
            best_c = c2

    # Phase 3: LSE descent with multiple betas
    for beta in [40, 80, 120, 200]:
        if time_left() < 40:
            break
        h3, c3 = lse_descent(best_h, beta=beta, max_iters=200, alpha_init=0.004)
        if c3 < best_c - 1e-13:
            best_h = h3
            best_c = c3

    # Phase 4: Active set descent with various active set sizes
    for na in [6, 10, 15, 20, 30]:
        if time_left() < 30:
            break
        h4, c4 = active_set_descent(best_h, max_iters=200, n_active=na, alpha_init=0.004)
        if c4 < best_c - 1e-13:
            best_h = h4
            best_c = c4

    # Phase 5: Multi-resolution exploration
    n_parent = len(best_h)
    if time_left() > 120:
        dims_to_try = sorted(set(
            [n_parent - 4, n_parent - 2, n_parent + 2, n_parent + 4,
             n_parent + 6, n_parent + 8, n_parent - 6,
             int(n_parent * 0.85) // 2 * 2, int(n_parent * 1.15) // 2 * 2,
             int(n_parent * 1.3) // 2 * 2]
        ), key=lambda x: abs(x - n_parent))
        for new_n in dims_to_try:
            if time_left() < 50:
                break
            if new_n < 20 or new_n > 200 or new_n % 2 != 0:
                continue
            h_interp = interpolate_to_n(best_h, new_n)
            # Quick hybrid descent
            h_r, c_r = hybrid_descent(h_interp, beta=80, max_iters=400, rng=rng)
            if c_r < best_c - 1e-13:
                best_h = h_r
                best_c = c_r
            # Follow up with pairwise
            h_p, c_p = pairwise_descent(best_h, beta=100, max_iters=100)
            if c_p < best_c - 1e-13:
                best_h = h_p
                best_c = c_p

    # Phase 6: SLSQP on best
    if time_left() > 60:
        h6, c6 = slsqp_solve(best_h, maxiter=300)
        if h6 is not None and c6 < best_c - 1e-12:
            best_h = h6
            best_c = c6
            # Follow-up
            h_as, c_as = active_set_descent(best_h, max_iters=100, n_active=10)
            if c_as < best_c - 1e-13:
                best_h = h_as
                best_c = c_as

    # Phase 7: Try diverse starting points
    if time_left() > 150:
        n_best = len(best_h)
        # Random binary starts
        for trial in range(4):
            if time_left() < 40:
                break
            h_rand = random_binary(n_best, rng)
            h_r, c_r = hybrid_descent(h_rand, beta=60, max_iters=300, rng=rng)
            if c_r < best_c - 1e-12:
                best_h = h_r
                best_c = c_r
                h_p, c_p = pairwise_descent(best_h, beta=80, max_iters=100)
                if c_p < best_c - 1e-12:
                    best_h = h_p
                    best_c = c_p

        # Greedy binary
        if time_left() > 60:
            h_g = greedy_binary(n_best, rng)
            h_r, c_r = hybrid_descent(h_g, beta=60, max_iters=300, rng=rng)
            if c_r < best_c - 1e-12:
                best_h = h_r
                best_c = c_r
                h_p, c_p = pairwise_descent(best_h, beta=80, max_iters=100)
                if c_p < best_c - 1e-12:
                    best_h = h_p
                    best_c = c_p

        # Symmetric/antisymmetric perturbations
        if time_left() > 60:
            for pert_scale in [0.01, 0.03, 0.05]:
                if time_left() < 30:
                    break
                z = rng.randn(len(best_h))
                z_sym = (z + z[::-1]) / 2.0
                z_sym -= z_sym.mean()
                h_pert = repair_feasible(best_h + pert_scale * z_sym)
                h_r, c_r = hybrid_descent(h_pert, beta=80, max_iters=200, rng=rng)
                if c_r < best_c - 1e-12:
                    best_h = h_r
                    best_c = c_r

    # Phase 8: Aggressive local refinement
    if time_left() > 60:
        # Multiple pairwise rounds
        for _ in range(3):
            if time_left() < 20:
                break
            h_p, c_p = pairwise_descent(best_h, beta=120, max_iters=150)
            if c_p < best_c - 1e-14:
                best_h = h_p
                best_c = c_p

    # Phase 9: Final LSE with high beta
    if time_left() > 30:
        h9, c9 = lse_descent(best_h, beta=200, max_iters=200, alpha_init=0.002)
        if c9 < best_c - 1e-14:
            best_h = h9
            best_c = c9

    # Phase 10: Final active set polish
    if time_left() > 20:
        h10, c10 = active_set_descent(best_h, max_iters=200, n_active=8, alpha_init=0.002)
        if c10 < best_c - 1e-14:
            best_h = h10
            best_c = c10

    # Phase 11: Final SLSQP
    if time_left() > 35:
        h_sol, c_sol = slsqp_solve(best_h, maxiter=300)
        if h_sol is not None and c_sol < best_c - 1e-14:
            best_h = h_sol
            best_c = c_sol

    # Phase 12: One more round of everything
    if time_left() > 25:
        h_p, c_p = pairwise_descent(best_h, beta=150, max_iters=80)
        if c_p < best_c - 1e-14:
            best_h = h_p
            best_c = c_p
        h_as, c_as = active_set_descent(best_h, max_iters=100, n_active=10, alpha_init=0.001)
        if c_as < best_c - 1e-14:
            best_h = h_as
            best_c = c_as

    # Final validation
    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)

    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            final_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 2:
            final_c = float(eval_result[0])
    except Exception:
        pass

    return best_h, final_c, len(best_h)
```
