def project_feasible(x, target_sum, tol=1e-11):
    n = len(x)
    lo = -2.0
    hi = 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > tol:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def c5_of_h(h, dx):
    return np.correlate(h, 1.0 - h, mode='full').max() * dx

def grad_corr_k(h, k, n):
    """Gradient of corr_k with respect to h."""
    off = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -off)
    i_hi = min(n - 1, n - 1 - off)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx + off] += (1.0 - h[idx])
    grad[idx] += (-h[idx + off])
    return grad

def solve_soft_lp(h, dx, top_idx, alpha, n, rho_q=0.5, rho_s=1.0):
    """Soft slack LP: minimize rho_q * q + rho_s * sum(s_k)
    subject to: dx*grad_k · delta + s_k - q <= -dx*corr_k for k in top_idx
                sum(delta) = 0
                -h_i*alpha <= delta_i <= (1-h_i)*alpha
    """
    K = len(top_idx)
    m_vars = n + K + 1  # delta (n), s (K), q (1)
    # Objective: minimize rho_q * q + rho_s * sum(s)
    c_obj = np.zeros(m_vars)
    c_obj[n:n+K] = rho_s
    c_obj[n+K] = rho_q  # q is the last variable
    
    # Constraints: dx*grad_k · delta + s_k - q <= -dx*corr_k
    corrs = np.correlate(h, 1.0 - h, mode='full')
    A_ub = np.zeros((K, m_vars))
    b_ub = np.zeros(K)
    for j, k in enumerate(top_idx):
        g = grad_corr_k(h, k, n)
        A_ub[j, :n] = dx * g
        A_ub[j, n+j] = 1.0   # s_j
        A_ub[j, n+K] = -1.0  # q
        b_ub[j] = -dx * corrs[k]
    
    # Equality: sum(delta) = 0
    A_eq = np.zeros((1, m_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds
    bounds = [(-h[i]*alpha, (1.0-h[i])*alpha) for i in range(n)]
    bounds += [(0.0, None)] * K  # s_k >= 0
    bounds.append((None, None))  # q unbounded
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.status != 0:
            return None
        return res.x[:n]
    except Exception:
        return None

def solve_sum_lp(h, dx, top_idx, alpha, n, weights=None):

    """

def project_feasible(x, target_sum, tol=1e-11):
    n = len(x)
    lo = -3.0
    hi = 3.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > tol:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            adj = diff / mask.sum()
            h[mask] = np.clip(h[mask] + adj, 0.0, 1.0)
    return np.clip(h, 0.0, 1.0)

def c5_of_h(h, dx):
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def grad_corr_k(h, k, n):
    off = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -off)
    i_hi = min(n - 1, n - 1 - off)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx + off] += (1.0 - h[idx])
    grad[idx] += (-h[idx + off])
    return grad

def solve_hard_lp(h, dx, top_idx, alpha, n):
    K = len(top_idx)
    m_vars = n + 1
    corrs = np.correlate(h, 1.0 - h, mode='full')
    c_obj = np.zeros(m_vars)
    c_obj[-1] = 1.0
    A_ub = np.zeros((K, m_vars))
    b_ub = np.zeros(K)
    for j, k in enumerate(top_idx):
        g = grad_corr_k(h, k, n)
        A_ub[j, :n] = dx * g
        A_ub[j, -1] = -1.0
        b_ub[j] = -dx * corrs[k]
    A_eq = np.ones((1, n))
    A_eq = np.hstack([A_eq, np.zeros((1, 1))])
    b_eq = np.zeros(1)
    bounds = [(-h[i] * alpha, (1.0 - h[i]) * alpha) for i in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.status != 0:
            return None
        return res.x[:n]
    except Exception:
        return None

def solve_soft_lp(h, dx, top_idx, alpha, n, rho_q=0.3, rho_s=1.0):

    corrs = np.correlate(h, 1.0 - h, mode='full')
    c_obj = np.zeros(m_vars)
    c_obj[n:n + K] = rho_s
    c_obj[n + K] = rho_q

    for j, k in enumerate(top_idx):

        A_ub[j, n + K] = -1.0
        b_ub[j] = -dx * corrs[k]
    A_eq = np.zeros((1, m_vars))

def solve_sum_lp(h, dx, top_idx, alpha, n, weights=None):

        weights = np.ones(K)
