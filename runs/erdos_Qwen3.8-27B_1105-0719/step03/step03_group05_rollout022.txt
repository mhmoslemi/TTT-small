    h_new = h.copy()

def pair_line_search(h, i, j, n_evals=30):
    """Find best t for pair move (i,j) using ternary search."""
    lo = max(-h[i], h[j]-1.0)
    hi = min(1.0-h[i], h[j])
    if lo >= hi:
        return 0.0, eval_profile_at_t(h, i, j, 0.0)
    
    # Add small margin
    lo += 1e-12
    hi -= 1e-12
    if lo >= hi:
        return 0.0, eval_profile_at_t(h, i, j, 0.0)
    
    # Ternary search
    a, b = lo, hi
    for _ in range(n_evals):
        m1 = a + (b-a)/3
        m2 = b - (b-a)/3
        f1 = eval_profile_at_t(h, i, j, m1)
        f2 = eval_profile_at_t(h, i, j, m2)
        if f1 < f2:
            b = m2
        else:
            a = m1
    t_opt = (a+b)/2
    f_opt = eval_profile_at_t(h, i, j, t_opt)
    return t_opt, f_opt

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)

        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx] + 1e-15)
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff

def lse_descent(h, beta=80.0, max_iters=100, alpha_init=0.003):

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = np.linspace(0, 2, n + 1)
    xt = np.linspace(0, 2, new_n + 1)
    h_interp = np.zeros(new_n)
    for k in range(new_n):
        total_mass = 0.0
        total_width = 0.0
        lo_k, hi_k = xt[k], xt[k+1]
        for j in range(n):
            lo_j, hi_j = xs[j], xs[j+1]
            overlap = min(hi_k, hi_j) - max(lo_k, lo_j)
            if overlap > 0:
                total_mass += h[j] * overlap
                total_width += overlap
        h_interp[k] = total_mass / total_width if total_width > 1e-15 else 0.5
    h_interp = np.clip(h_interp, 0.0, 1.0)
    return repair_feasible(h_interp)

    n = len(h)

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx] + 1e-15)
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx] + 1e-15)
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-14:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmin(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def build_gradient(h, n):
    """Build gradient of profile w.r.t. h for all lags. Returns (M, n) matrix."""
    dx = 2.0 / n
    M = 2 * n - 1
    G = np.zeros((M, n))
    for d_idx in range(M):
        d = d_idx - (n - 1)
        if d == 0:
            G[d_idx, :] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            G[d_idx, :n-d] += dx * (1.0 - h[d:])
            G[d_idx, d:] += -dx * h[:n-d]
        else:
            dp = -d
            G[d_idx, dp:] += dx * (1.0 - h[:n-dp])
            G[d_idx, :n-dp] += -dx * h[dp:]
    return G

def lse_descent(h, beta=80.0, max_iters=80, alpha_init=0.003):
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    alpha = alpha_init
    stuck = 0
    for it in range(max_iters):
        profile = compute_profile(h)
        scaled = beta * (profile - np.max(profile))
        weights = np.exp(scaled)
        wsum = weights.sum()
        if wsum < 1e-300:
            break
        weights /= wsum
        G = build_gradient(h, n)
        grad = G.T @ weights
        grad -= grad.mean()
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad /= gnorm
        improved = False
        for a_try in [alpha, alpha * 0.5, alpha * 0.25, alpha * 0.1]:
            h_try = repair_feasible(h - a_try * grad)
            s = exact_score(h_try)
            if s < best_c - 1e-14:
                h = h_try
                best_c = s
                improved = True
                alpha = min(alpha * 1.06, 0.018)
                stuck = 0
                break
        if not improved:
            alpha *= 0.5
            stuck += 1
            if stuck > 20 or alpha < 1e-9:
                break
    return h, best_c

def active_set_descent(h, max_iters=80, n_active=10, alpha_init=0.004):
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    alpha = alpha_init
    stuck = 0
    for it in range(max_iters):
        profile = compute_profile(h)
        c = float(np.max(profile))
        if c + 1e-14 < best_c:
            best_c = c
            stuck = 0
        margin = max(0.001, 3.0 * (c - float(np.median(profile))))
        active = np.where(profile >= c - margin)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(profile))])
        if len(active) > n_active:
            top_idx = np.argsort(profile[active])[-n_active:]
            active = active[top_idx]
        G = build_gradient(h, n)
        grad = G[active, :].mean(axis=0)
        grad -= grad.mean()
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad /= gnorm
        improved = False
        for a_try in [alpha, alpha * 0.5, alpha * 0.25, alpha * 0.1]:
            h_try = repair_feasible(h - a_try * grad)
            s = exact_score(h_try)
            if s < best_c - 1e-14:
                h = h_try
                best_c = s
                improved = True
                break
        if not improved:
            alpha *= 0.5
            stuck += 1
            if stuck > 25 or alpha < 1e-9:
                break
    return h, best_c

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = np.linspace(0, 2, n + 1)
    xt = np.linspace(0, 2, new_n + 1)
    h_interp = np.zeros(new_n)
    for k in range(new_n):
        total_mass = 0.0
        total_width = 0.0
        lo_k, hi_k = xt[k], xt[k + 1]
        for j in range(n):
            lo_j, hi_j = xs[j], xs[j + 1]
            overlap = min(hi_k, hi_j) - max(lo_k, lo_j)
            if overlap > 0:
                total_mass += h[j] * overlap
                total_width += overlap
        h_interp[k] = total_mass / total_width if total_width > 1e-15 else 0.5
    h_interp = np.clip(h_interp, 0.0, 1.0)
    return repair_feasible(h_interp)

def pair_line_search(h, i, j, n_grid=35):
    """Exact 1D minimization along pair move h[i]+t, h[j]-t using grid + refinement."""
    lo = max(-h[i] + 1e-12, h[j] - 1.0 + 1e-12)
    hi = min(1.0 - h[i] - 1e-12, h[j] - 1e-12)
    if lo >= hi:
        return 0.0
    ts = np.linspace(lo, hi, n_grid)
    best_t = 0.0
    best_f = exact_score(h)
    for t in ts:
        h_try = h.copy()
        h_try[i] += t
        h_try[j] -= t
        f = exact_score(h_try)
        if f < best_f:
            best_f = f
            best_t = t
    if best_t != 0.0 and abs(best_t) < (hi - lo) * 0.4:
        step = (hi - lo) / n_grid
        local_ts = np.linspace(best_t - step, best_t + step, 15)
        local_ts = local_ts[(local_ts >= lo) & (local_ts <= hi)]
        for t in local_ts:
            h_try = h.copy()
            h_try[i] += t
            h_try[j] -= t
            f = exact_score(h_try)
            if f < best_f:
                best_f = f
                best_t = t
    return best_t

def select_pairs(profile, h, n, n_pairs=40, rng=None):
    """Select promising (i,j) pairs based on active lags and value extremes."""
    c = float(np.max(profile))
    pairs = []
    seen = set()
    # From active lags
    active_lags = np.where(profile >= c - 0.004)[0]
    for d_idx in active_lags[:15]:
        d = int(d_idx) - (n - 1)
        if d == 0:
            continue
        if d > 0:
            for i in range(n - d):
                if h[i] > h[i + d] + 0.05:
                    key = (i, i + d)
                    if key not in seen:
                        pairs.append(key)
                        seen.add(key)
        else:
            dp = -d
            for i in range(dp, n):
                if h[i] > h[i - dp] + 0.05:
                    key = (i, i - dp)
                    if key not in seen:
                        pairs.append(key)
                        seen.add(key)
    # Add random high-low pairs
    sorted_idx = np.argsort(h)
    low_set = sorted_idx[:n // 3]
    high_set = sorted_idx[2 * n // 3:]
    n_rand = max(10, n_pairs // 4)
    for _ in range(n_rand * 3):
        if len(pairs) >= n_pairs:
            break
        i = int(rng.choice(high_set))
        j = int(rng.choice(low_set))
        if i != j and (i, j) not in seen:
            pairs.append((i, j))
            seen.add((i, j))
    # Shuffle
    if len(pairs) > n_pairs:
        idx = rng.permutation(len(pairs))[:n_pairs]
        pairs = [pairs[k] for k in idx]
    return pairs

def slsqp_solve(h0, maxiter=150):
    n = len(h0)
    c0 = exact_score(h0)
    def obj(x):
        return x[-1]
    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    def eq_fun(x):
        return x[:-1].sum() - n / 2.0
    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g
    def ineq_fun(x):
        h = np.clip(x[:-1], 0, 1)
        t = x[-1]
        profile = np.correlate(h, 1 - h, mode="full") * (2.0 / n)
        return t - profile
    def ineq_jac(x):
        h = np.clip(x[:-1], 0, 1)
        G = build_gradient(h, n)
        J = np.zeros((2 * n - 1, n + 1))
        J[:, :-1] = -G
        J[:, -1] = 1.0
        return J
    x0 = np.concatenate([h0, [c0 + 1e-3]])
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 0.5)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None and np.isfinite(res.x).all():
            h_new = repair_feasible(res.x[:-1])
            c_new = exact_score(h_new)
            if c_new <= c0 + 1e-9:
                return h_new, c_new
    except Exception:
        pass
    return None, None

def smooth_step(h, window=3):
    """Apply mild smoothing while preserving feasibility."""
    n = len(h)
    w = np.ones(window) / window
    h_smooth = np.convolve(h, w, mode='same')
    return repair_feasible(h_smooth)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.perf_counter()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - (time.perf_counter() - t0)

    # Get parent
    try:
        parent = np.array(initial_h_values, dtype=np.float64)
    except (NameError, Exception):
        parent = np.full(70, 0.5)

    parent = np.clip(parent, 0.0, 1.0)
    parent = repair_feasible(parent)
    best_h = parent.copy()
    best_c = exact_score(parent)

    # Try quick symmetrizations
    h_sym = repair_feasible((parent + parent[::-1]) / 2.0)
    c_sym = exact_score(h_sym)
    if c_sym < best_c - 1e-12:
        best_h = h_sym
        best_c = c_sym

    # Antisymmetric: h[i] + h[n-1-i] = 1
    n_p = len(best_h)
    h_anti = np.zeros(n_p)
    for i in range(n_p):
        h_anti[i] = (best_h[i] + 1.0 - best_h[n_p - 1 - i]) / 2.0
    h_anti = repair_feasible(h_anti)
    c_anti = exact_score(h_anti)
    if c_anti < best_c - 1e-12:
        best_h = h_anti
        best_c = c_anti

    # Phase 1: Quick LSE + active set
    if tleft() > 100:
        best_h, best_c = lse_descent(best_h, beta=60, max_iters=40)
        best_h, best_c = active_set_descent(best_h, max_iters=40, n_active=8)

    # Phase 2: Resolution sweep
    if tleft() > 200:
        n_cur = len(best_h)
        candidates_n = []
        for f in [0.85, 0.9, 0.95, 1.0, 1.05, 1.1, 1.15, 1.2]:
            d = int(round(n_cur * f / 2)) * 2
            d = max(24, min(d, 120))
            if d not in candidates_n:
                candidates_n.append(d)
        candidates_n = sorted(set(candidates_n), key=lambda x: abs(x - n_cur))
        for new_n in candidates_n[:5]:
            if tleft() < 30:
                break
            h_interp = interpolate_to_n(best_h, new_n)
            h_pol, c_pol = lse_descent(h_interp, beta=80, max_iters=30)
            if c_pol < best_c - 1e-13:
                best_h = h_pol
                best_c = c_pol
            h_ase, c_ase = active_set_descent(best_h, max_iters=30, n_active=8)
            if c_ase < best_c - 1e-13:
                best_h = h_ase
                best_c = c_ase

    # Phase 3: Main pair-line search loop
    n = len(best_h)
    no_improve_rounds = 0
    round_num = 0
    while tleft() > 120:
        round_num += 1
        profile = compute_profile(best_h)
        pairs = select_pairs(profile, best_h, n, n_pairs=35, rng=rng)
        improved_this_round = False

        for (i, j) in pairs:
            if tleft() < 30:
                break
            t = pair_line_search(best_h, i, j, n_grid=30)
            if abs(t) > 1e-8:
                h_try = best_h.copy()
                h_try[i] += t
                h_try[j] -= t
                h_try = np.clip(h_try, 0.0, 1.0)
                if abs(h_try.sum() - n / 2.0) > 1e-10:
                    h_try = repair_feasible(h_try)
                s = exact_score(h_try)
                if s < best_c - 1e-12:
                    best_h = h_try
                    best_c = s
                    improved_this_round = True

        if improved_this_round:
            no_improve_rounds = 0
        else:
            no_improve_rounds += 1

        # Every 15 rounds without improvement, do a quick LSE step
        if no_improve_rounds % 15 == 0 and no_improve_rounds > 0 and tleft() > 60:
            h_lse, c_lse = lse_descent(best_h, beta=100, max_iters=15)
            if c_lse < best_c - 1e-13:
                best_h = h_lse
                best_c = c_lse
            h_ase, c_ase = active_set_descent(best_h, max_iters=15, n_active=10)
            if c_ase < best_c - 1e-13:
                best_h = h_ase
                best_c = c_ase

        # Every 40 rounds, try a small resolution change
        if round_num % 40 == 0 and tleft() > 60:
            for dn in [2, -2, 4, -4]:
                new_n = n + dn
                if new_n < 24 or new_n > 140 or new_n % 2 != 0:
                    continue
                h_interp = interpolate_to_n(best_h, new_n)
                h_pol, c_pol = lse_descent(h_interp, beta=100, max_iters=15)
                if c_pol < best_c - 1e-13:
                    best_h = h_pol
                    best_c = c_pol
                    n = new_n
                break

        # Every 60 rounds, try smoothing
        if round_num % 60 == 0 and tleft() > 40:
            h_sm = smooth_step(best_h, window=3)
            c_sm = exact_score(h_sm)
            if c_sm < best_c - 1e-13:
                best_h = h_sm
                best_c = c_sm

    # Phase 4: Final polish
    if tleft() > 60:
        h_sol, c_sol = slsqp_solve(best_h, maxiter=150)
        if h_sol is not None and c_sol < best_c - 1e-12:
            best_h = h_sol
            best_c = c_sol

    if tleft() > 30:
        best_h, best_c = lse_descent(best_h, beta=120, max_iters=50)
        best_h, best_c = active_set_descent(best_h, max_iters=50, n_active=10)

    # Final pair-line polish
    n = len(best_h)
    for _ in range(30):
        if tleft() < 5:
            break
        profile = compute_profile(best_h)
        pairs = select_pairs(profile, best_h, n, n_pairs=20, rng=rng)
        for (i, j) in pairs[:10]:
            if tleft() < 3:
                break
            t = pair_line_search(best_h, i, j, n_grid=25)
            if abs(t) > 1e-8:
                h_try = best_h.copy()
                h_try[i] += t
                h_try[j] -= t
                h_try = np.clip(h_try, 0.0, 1.0)
                if abs(h_try.sum() - n / 2.0) > 1e-10:
                    h_try = repair_feasible(h_try)
                s = exact_score(h_try)
                if s < best_c - 1e-12:
                    best_h = h_try
                    best_c = s

    # Final SLSQP if time allows
    if tleft() > 40:
        h_sol, c_sol = slsqp_solve(best_h, maxiter=200)
        if h_sol is not None and c_sol < best_c - 1e-12:
            best_h = h_sol
            best_c = c_sol

    # Final LSE polish
    if tleft() > 15:
        best_h, best_c = lse_descent(best_h, beta=150, max_iters=30, alpha_init=0.002)
        best_h, best_c = active_set_descent(best_h, max_iters=30, n_active=8, alpha_init=0.002)

    # Final validation
    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)

    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            final_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 2:
            final_c = float(eval_result[0])
    except Exception:
        pass

    return best_h, final_c, len(best_h)
