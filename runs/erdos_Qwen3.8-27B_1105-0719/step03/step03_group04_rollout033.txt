import time

def compute_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def compute_profile(h):

    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

    n = len(h)

    center = n - 1

        C[d:, k] -= h[:n-d]
        k = center - d
        C[d:, k] += (1.0 - h[:n-d])
        C[:n-d, k] -= h[d:n]
    return C

def pair_move_optimize(h, t_start, budget_s, min_improve=1e-12):
    n = len(h)
    M = 2 * n - 1
    h = np.array(h, dtype=np.float64)
    best_c = compute_score(h)
    best_h = h.copy()
    
    # Precompute pair indices
    idx = np.arange(n)
    ii, jj = np.meshgrid(idx, idx, indexing='ij')
    mask = ii != jj
    pairs_i = ii[mask]
    pairs_j = jj[mask]
    num_pairs = len(pairs_i)
    
    # Feasible ranges (will be updated as h changes)
    alpha_lo = -h[pairs_j]
    alpha_hi = 1.0 - h[pairs_i]
    
    no_improve = 0
    iter_count = 0
    
    while True:
        elapsed = time.perf_counter() - t_start
        if budget_s - elapsed < 5.0:
            break
        if no_improve > 10:
            break
        
        P = compute_profile(h)
        C = compute_grad_matrix(h)
        
        # Update feasible ranges
        alpha_lo = -h[pairs_j]
        alpha_hi = 1.0 - h[pairs_i]
        
        # Compute ΔC for all pairs
        dC = C[pairs_i] - C[pairs_j]  # (num_pairs, M)
        
        # Ternary search for optimal α for each pair
        lo = alpha_lo.copy()
        hi = alpha_hi.copy()
        
        # Skip pairs where lo >= hi (infeasible)
        feasible = lo < hi - 1e-15
        if not feasible.any():
            break
        
        for _ in range(35):
            mid1 = lo + (hi - lo) / 3.0
            mid2 = hi - (hi - lo) / 3.0
            
            # f(α) = max(P + α * dC) for all pairs
            f1 = np.maximum.reduce(P[None, :] + mid1[:, None] * dC)
            f2 = np.maximum.reduce(P[None, :] + mid2[:, None] * dC)
            
            # For convex f: if f(mid1) <= f(mid2), min is in [lo, mid2]
            mask_improve = f1 <= f2
            hi[mask_improve] = mid1[mask_improve]
            lo[~mask_improve] = mid2[~mask_improve]
        
        alpha_opt = (lo + hi) / 2.0
        f_opt = np.maximum.reduce(P[None, :] + alpha_opt[:, None] * dC)
        
        # Find best improving pair
        current_max = np.max(P)
        improving = f_opt < current_max - min_improve
        if not improving.any():
            no_improve += 1
            continue
        
        # Pick the best improving pair
        best_candidates = np.where(improving)[0]
        best_p = best_candidates[np.argmin(f_opt[best_candidates])]
        
        i = pairs_i[best_p]
        j = pairs_j[best_p]
        alpha = alpha_opt[best_p]
        
        # Apply the move
        h[i] += alpha
        h[j] -= alpha
        
        # Ensure feasibility (should be exact, but clip for safety)
        h = np.clip(h, 0.0, 1.0)
        
        # Verify sum constraint (pair move preserves sum, but clipping might break it)
        target = n / 2.0
        diff = target - h.sum()
        if abs(diff) > 1e-14:
            # Find the element to adjust
            if diff > 0:
                adj_idx = np.argmax(1.0 - h)
            else:
                adj_idx = np.argmax(h)
            h[adj_idx] += diff
            h = np.clip(h, 0.0, 1.0)
        
        no_improve = 0
        iter_count += 1
        
        c = compute_score(h)
        if c < best_c - 1e-15:
            best_c = c
            best_h = h.copy()
    
    return best_h, best_c
