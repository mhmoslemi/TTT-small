import numpy as np
import time

def compute_raw(h):
    return np.correlate(h, 1.0 - h, mode='full')

def c5_of_h(h, dx):
    return compute_raw(h).max() * dx

def project_feasible(x, target_sum):
    n = len(x)
    x = np.asarray(x, dtype=float).copy()
    x[~np.isfinite(x)] = 0.5
    x_max = np.max(np.abs(x))
    lo, hi = -3.0 - x_max, 3.0 + x_max
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            cap = 1.0 - h
            cs = cap.sum()
            if cs > 1e-15:
                h += cap * (diff / cs)
        else:
            ps = h.sum()
            if ps > 1e-15:
                h -= h * (-diff / ps)
    return np.clip(h, 0.0, 1.0)

def lse_weights(raw, T):
    rmax = raw.max()
    w = np.exp((raw - rmax) / T)
    s = w.sum()
    if s < 1e-300:
        mask = raw >= rmax - 1e-10
        u = np.zeros_like(raw)
        u[mask] = 1.0 / mask.sum()
        return u
    return w / s

def lse_grad(h, raw, T):
    n = len(h)
    u = lse_weights(raw, T)
    g = np.zeros(n)
    for p in range(len(raw)):
        wp = u[p]
        if wp < 1e-14:
            continue
        d = p - (n - 1)
        if d >= 0:
            nd = n - d
            if nd > 0:
                g[:nd] += wp * (1.0 - h[d:])
            g[d:] += wp * (-h[:nd])
        else:
            dp = -d
            g[dp:] += wp * (1.0 - h[:n - dp])
            ndp = n - dp
            g[:ndp] += wp * (-h[dp:])
    return g

def alpha_bounds(h, v):
    """Find max alpha such that h + alpha*v stays in [0,1]."""
    pos = v > 1e-12
    neg = v < -1e-12
    a_max = 1e10
    if pos.any():
        a_max = np.min((1.0 - h[pos]) / v[pos])
    if neg.any():
        a_min = np.max(-h[neg] / v[neg])
    else:
        a_min = 0.0
    return max(0.0, a_min), min(a_max, 1e10)

def lse_step(h, dx, target_sum, T, max_time_check=None, t0=None):

    g = lse_grad(h, raw, T)
    v = g - g.mean()
    nrm = np.max(np.abs(v))
    if nrm < 1e-15:
        return h, c5_of_h(h, dx), False
    
    a_lo, a_hi = alpha_bounds(h, v)
    if a_hi - a_lo < 1e-12:
        return h, c5_of_h(h, dx), False
    
    cur_c5 = c5_of_h(h, dx)
    best_h = h
    best_c5 = cur_c5
    improved = False
    
    # Try fractions of the max feasible step
    for frac in [1.0, 0.5, 0.25, 0.125, 0.06, 0.03, 0.01]:
        if t0 is not None and time.time() - t0 > max_time_check:
            break
        a = a_lo + (a_hi - a_lo) * frac
        if a <= a_lo + 1e-14:
            continue
        h_new = project_feasible(h + a * v, target_sum)
        c5_new = c5_of_h(h_new, dx)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new
            improved = True
            break
    
    return best_h, best_c5, improved

def lse_anneal(h, dx, target_sum, T_start, T_end, n_steps, t0, max_time):

                best_h = h_new.copy()
            stuck = 0
        else:

    n = len(h)
    raw = compute_raw(h)
    u = lse_weights(raw, T)
    m_total = len(raw)
    
    # Select active set by LSE weight
    order = np.argsort(u)[::-1]
    active_idx = order[:min(n_active, m_total)]
    
    current_c5 = raw.max() * dx
    
    # Build LP
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    A_ub = np.zeros((len(active_idx), n + 1))
    b_ub = np.zeros(len(active_idx))
    
    for row, p in enumerate(active_idx):
        d = p - (n - 1)
        grad = np.zeros(n)
        if d >= 0:
            nd = n - d
            if nd > 0:
                grad[:nd] += (1.0 - h[d:])
            grad[d:] += (-h[:nd])
        else:
            dp = -d
            grad[dp:] += (1.0 - h[:n - dp])
            ndp = n - dp
            grad[:ndp] += (-h[dp:])
        A_ub[row, :n] = grad
        A_ub[row, n] = -1.0
        b_ub[row] = -raw[p]
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    bounds = [(max(-h[i]*alpha_limit, -0.5), min((1-h[i])*alpha_limit, 0.5)) for i in range(n)]
    bounds.append((-np.inf, np.inf))
    
    try:
        from scipy.optimize import linprog
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
        if res.status != 0:
            return h, current_c5, False
        delta = res.x[:n]
    except Exception:
        return h, current_c5, False
    
    if np.abs(delta).max() < 1e-15:
        return h, current_c5, False
    
    best_h = h
    best_c5 = current_c5
    improved = False
    
    for s in [1.0, 0.7, 0.5, 0.3, 0.15, 0.05]:

            best_c5 = c5_new
            best_h = h_new
            improved = True
            break
