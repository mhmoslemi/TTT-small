The key insight from the strategy is to work in the **anti-symmetric (complement) subspace** where:
- h[j] + h[n-1-j] = 1 (for all j)
- This automatically satisfies the mass constraint sum(h) = n/2

This means we only need to optimize m = n/2 variables (the left half), and the right half is determined. This halves the dimensionality and eliminates the need for constraint repair during the search.

Let me design the algorithm:

1. **Setup**: Get the parent from `initial_h_values`, project it into the anti-symmetric subspace.

2. **Core search in reduced coordinates**:
   - Single coordinate line search on reduced variables
   - Pair coordinate moves (mass-preserving in reduced space)
   - Gradient-based direction search
   - LSE (log-sum-exp) surrogate optimization

3. **Resolution exploration**: Try different n values, interpolate, project to anti-space, optimize.

4. **Full-space escape moves**: Occasionally break out of anti-symmetric constraint with pair moves.

Let me think about the implementation more carefully:

For n = 2m (even):
- y[0..m-1] are the free variables
- h[j] = y[j] for j = 0..m-1
- h[n-1-j] = 1 - y[j] for j = 0..m-1

The objective is C5 = max_k sum_i h[i]*(1-h[i+k]) * dx (with appropriate indexing for the full correlation).

Let me think about the gradient in reduced coordinates:
- dC5/dy[j] = dC5/dh[j] - dC5/dh[n-1-j]

For the line search, given a direction d in reduced space, I can:
- Parametrize y(t) = y + t*d
- Find the valid t range where all y[i] stay in [0,1]
- Do a golden section or ternary search on C5(y(t))

This is much cleaner than the full-space approach because:
1. No constraint repair needed
2. The line search is exact (we're on the feasible manifold)

def build_gradient_matrix(h, n, dx):
    """Build the (2n-1) x n gradient matrix for the profile."""
    M = 2 * n - 1
    G = np.zeros((M, n))

    stuck = 0

        max_s = np.max(scaled)

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = np.linspace(0, 2, n + 1)
    xt = np.linspace(0, 2, new_n + 1)
    h_interp = np.zeros(new_n)
    for k in range(new_n):
        lo_k, hi_k = xt[k], xt[k+1]
        total_mass = 0.0
        total_width = 0.0
        for j in range(n):
            lo_j, hi_j = xs[j], xs[j+1]
            overlap = min(hi_k, hi_j) - max(lo_k, lo_j)
            if overlap > 0:
                total_mass += h[j] * overlap
                total_width += overlap
        h_interp[k] = total_mass / total_width if total_width > 1e-15 else 0.5
    h_interp = np.clip(h_interp, 0.0, 1.0)
    return repair_feasible(h_interp)

def anti_reduced_grad(y, n, weights):
    m = len(y)
    h = h_from_anti(y, n)
    dx = 2.0 / n
    G = build Gradient_matrix(h, n, dx) if False else build_gradient_matrix(h, n, dx)
    full_grad = G.T @ weights
    red_grad = np.zeros(m)
    for j in range(m):
        red_grad[j] = full_grad[j] - full_grad[n - 1 - j]
    return red_grad

def build_gradient_matrix(h, n, dx):
    M = 2 * n - 1
    G = np.zeros((M, n))
    for m in range(M):
        d = m - (n - 1)
        if d == 0:
            G[m, :] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            G[m, d:] += dx * (1.0 - h[0:n-d])
            G[m, 0:n-d] += -dx * h[d:n]
        else:
            dp = -d
            G[m, 0:n-dp] += dx * (1.0 - h[dp:n])
            G[m, dp:n] += -dx * h[0:n-dp]
    return G

def line_search_anti(y, direction, n, fine_nsamp=25):
    m = len(y)
    t_lo, t_hi = 0.0, 0.0
    for i in range(m):
        di = direction[i]
        if di > 1e-15:
            t_lo = max(t_lo, -y[i] / di)
            t_hi = min(t_hi, (1.0 - y[i]) / di)
        elif di < -1e-15:
            t_lo = max(t_lo, (1.0 - y[i]) / di)
            t_hi = min(t_hi, -y[i] / di)
    if t_hi - t_lo < 1e-13:
        return y.copy(), 0.0, anti_score(y, n)
    t_lo += 1e-10
    t_hi -= 1e-10
    if t_hi < t_lo:
        return y.copy(), 0.0, anti_score(y, n)
    best_t = 0.0
    best_f = anti_score(y, n)
    ts = np.linspace(t_lo, t_hi, fine_nsamp)
    for t in ts:
        f = anti_score(y + t * direction, n)
        if f < best_f:
            best_f = f
            best_t = t
    if best_t != 0.0:
        span = t_hi - t_lo
        lo = max(t_lo, best_t - 0.3 * span)
        hi = min(t_hi, best_t + 0.3 * span)
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        for _ in range(25):
            t1 = hi - gr * (hi - lo)
            t2 = lo + gr * (hi - lo)
            f1 = anti_score(y + t1 * direction, n)
            f2 = anti_score(y + t2 * direction, n)
            if f1 < f2:
                hi = t2
            else:
                lo = t1
            if hi - lo < 1e-12:
                break
        best_t = (lo + hi) * 0.5
        best_f = anti_score(y + best_t * direction, n)
    return y + best_t * direction, best_t, best_f

def lse_anti(y, n, beta=80.0, max_iters=200, alpha_init=0.005):
    m = len(y)
    y = y.copy()
    best_c = anti_score(y, n)
    stuck = 0
    for it in range(max_iters):
        h = h_from_anti(y, n)
        profile = compute_profile(h)
        c = float(np.max(profile))
        scaled = beta * profile
        mx = np.max(scaled)
        weights = np.exp(scaled - mx)
        wsum = weights.sum()
        if wsum < 1e-300:
            break
        weights /= wsum
        dx = 2.0 / n
        G = build_gradient_matrix(h, n, dx)
        full_grad = G.T @ weights
        red_grad = np.zeros(m)
        for j in range(m):
            red_grad[j] = full_grad[j] - full_grad[n - 1 - j]
        gnorm = np.linalg.norm(red_grad)
        if gnorm < 1e-14:
            break
        direction = -red_grad / gnorm
        y_new, t, f = line_search_anti(y, direction, n)
        if f < best_c - 1e-14:
            y = y_new
            best_c = f
            stuck = 0
        elif f < c - 1e-14:
            y = y_new
            stuck = 0
        else:
            stuck += 1
            if stuck > 40:
                break
    return y, best_c

def active_set_anti(y, n, max_iters=200, n_active=10):
    m = len(y)
    y = y.copy()
    best_c = anti_score(y, n)
    stuck = 0
    for it in range(max_iters):
        h = h_from_anti(y, n)
        profile = compute_profile(h)
        c = float(np.max(profile))
        margin = max(0.003, 3.0 * (c - float(np.median(profile))))
        active = np.where(profile >= c - margin)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(profile))])
        if len(active) > n_active:
            top_idx = np.argsort(profile[active])[-n_active:]
            active = active[top_idx]
        dx = 2.0 / n
        G = build_gradient_matrix(h, n, dx)
        G_a = G[active, :]
        w = np.ones(len(active)) / len(active)
        full_grad = G_a.T @ w
        red_grad = np.zeros(m)
        for j in range(m):
            red_grad[j] = full_grad[j] - full_grad[n - 1 - j]
        gnorm = np.linalg.norm(red_grad)
        if gnorm < 1e-14:
            break
        direction = -red_grad / gnorm
        y_new, t, f = line_search_anti(y, direction, n)
        if f < best_c - 1e-14:
            y = y_new
            best_c = f
            stuck = 0
        elif f < c - 1e-14:
            y = y_new
            stuck = 0
        else:
            d2 = np.random.randn(m)
            d2 /= (np.linalg.norm(d2) + 1e-14)
            y_new2, t2, f2 = line_search_anti(y, d2, n)
            if f2 < best_c - 1e-14:
                y = y_new2
                best_c = f2
                stuck = 0
            else:
                stuck += 1
                if stuck > 50:
                    break
    return y, best_c

def random_dir_anti(y, n, rng, n_trials=80):
    m = len(y)
    y = y.copy()
    best_c = anti_score(y, n)
    for _ in range(n_trials):
        d = rng.randn(m)
        dn = np.linalg.norm(d)
        if dn < 1e-14:
            continue
        d /= dn
        y_new, t, f = line_search_anti(y, d, n)
        if f < best_c - 1e-14:
            y = y_new
            best_c = f
    return y, best_c

def coord_descent_anti(y, n, max_rounds=15, rng=None):
    m = len(y)
    y = y.copy()
    best_c = anti_score(y, n)
    if rng is None:
        rng = np.random.RandomState(0)
    for rnd in range(max_rounds):
        improved = False
        order = rng.permutation(m)
        for j in order:
            direction = np.zeros(m)
            direction[j] = 1.0
            y_new, t, f = line_search_anti(y, direction, n)
            if f < best_c - 1e-14:
                y = y_new
                best_c = f
                improved = True
        if not improved:
            break
    return y, best_c

def pair_descent_anti(y, n, max_iters=200, rng=None):
    m = len(y)
    y = y.copy()
    best_c = anti_score(y, n)
    if rng is None:
        rng = np.random.RandomState(0)
    stuck = 0
    for it in range(max_iters):
        i = rng.randint(m)
        j = rng.randint(m)
        if i == j:
            continue
        direction = np.zeros(m)
        direction[i] = 1.0
        direction[j] = -1.0
        y_new, t, f = line_search_anti(y, direction, n)
        if f < best_c - 1e-14:
            y = y_new
            best_c = f
            stuck = 0
        else:
            stuck += 1
            if stuck > 80:
                break
    return y, best_c

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = np.linspace(0, 2, n + 1)
    xt = np.linspace(0, 2, new_n + 1)
    h_interp = np.zeros(new_n)
    for k in range(new_n):
        lo_k, hi_k = xt[k], xt[k + 1]
        total_mass = 0.0
        total_width = 0.0
        for j in range(n):
            lo_j, hi_j = xs[j], xs[j + 1]
            overlap = min(hi_k, hi_j) - max(lo_k, lo_j)
            if overlap > 0:
                total_mass += h[j] * overlap
                total_width += overlap
        h_interp[k] = total_mass / total_width if total_width > 1e-15 else 0.5
    h_interp = np.clip(h_interp, 0.0, 1.0)
    return repair_feasible(h_interp)

def full_pair_descent(h, n, max_iters=200, rng=None):
    h = h.copy()
    best_c = exact_score(h)
    if rng is None:
        rng = np.random.RandomState(0)
    stuck = 0
    for it in range(max_iters):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        t_lo = max(-h[i], h[j] - 1.0)
        t_hi = min(1.0 - h[i], h[j])
        if t_hi - t_lo < 1e-13:
            continue
        t_lo += 1e-10
        t_hi -= 1e-10
        if t_hi < t_lo:
            continue
        best_t = 0.0
        best_f = best_c
        for t in np.linspace(t_lo, t_hi, 20):
            h_try = h.copy()
            h_try[i] += t
            h_try[j] -= t
            f = exact_score(h_try)
            if f < best_f:
                best_f = f
                best_t = t
        if best_t != 0.0:
            h[i] += best_t
            h[j] -= best_t
            best_c = best_f
            stuck = 0
        else:
            stuck += 1
            if stuck > 60:
                break
    return h, best_c

def sa_anti(y, n, rng, max_iters=500, T_init=0.3, T_final=0.01):
    m = len(y)
    y = y.copy()
    cur_c = anti_score(y, n)
    best_y = y.copy()
    best_c = cur_c
    T = T_init
    cooling = (T_final / T_init) ** (1.0 / max(max_iters, 1))
    for it in range(max_iters):
        T *= cooling
        i = rng.randint(m)
        delta = rng.uniform(-0.15, 0.15)
        y_try = y.copy()
        y_try[i] = np.clip(y_try[i] + delta, 0.0, 1.0)
        new_c = anti_score(y_try, n)
        delta_c = new_c - cur_c
        if delta_c < 0 or rng.rand() < np.exp(-delta_c / max(T, 1e-10)):
            y = y_try
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_y = y.copy()
    return best_y, best_c

def slsqp_solve(h0, maxiter=150):
    n = len(h0)
    dx = 2.0 / n
    M = 2 * n - 1
    c0 = exact_score(h0)
    def obj(x):
        return x[-1]
    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    def eq_fun(x):
        return x[:-1].sum() - n / 2.0
    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g
    def ineq_fun(x):
        h = np.clip(x[:-1], 0, 1)
        t = x[-1]
        profile = np.correlate(h, 1 - h, mode="full") * dx
        return t - profile
    def ineq_jac(x):
        h = np.clip(x[:-1], 0, 1)
        G = build_gradient_matrix(h, n, dx)
        J = np.zeros((M, n + 1))
        J[:, :-1] = -G
        J[:, -1] = 1.0
        return J
    x0 = np.concatenate([h0, [c0 + 1e-3]])
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 0.5)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None and np.isfinite(res.x).all():
            h_new = repair_feasible(res.x[:-1])
            c_new = exact_score(h_new)
            if c_new <= c0 + 1e-9:
                return h_new, c_new
    except Exception:
        pass
    return None, None

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s + t_start - time.perf_counter()

    parent = None
    try:
        parent = np.array(initial_h_values, dtype=np.float64)
    except NameError:
        pass
    except Exception:
        pass

    if parent is None or len(parent) < 4:
        parent = np.full(70, 0.5)

    parent = np.clip(parent, 0.0, 1.0)
    parent = repair_feasible(parent)
    n_parent = len(parent)
    c_parent = exact_score(parent)

    best_h = parent.copy()
    best_c = c_parent

    # Project parent to anti-space and optimize
    y_anti = anti_from_h(best_h)
    n_cur = n_parent

    # LSE descent in anti-space
    if time_left() > 150:
        y_anti, c_a = lse_anti(y_anti, n_cur, beta=60, max_iters=120)
        h_a = h_from_anti(y_anti, n_cur)
        c_a = exact_score(repair_feasible(h_a))
        if c_a < best_c - 1e-13:
            best_h = repair_feasible(h_a)
            best_c = c_a
            y_anti = anti_from_h(best_h)

    # Active set in anti-space
    if time_left() > 150:
        y_anti, c_a = active_set_anti(y_anti, n_cur, max_iters=150, n_active=10)
        h_a = h_from_anti(y_anti, n_cur)
        c_a = exact_score(repair_feasible(h_a))
        if c_a < best_c - 1e-13:
            best_h = repair_feasible(h_a)
            best_c = c_a
            y_anti = anti_from_h(best_h)

    # Pair descent in anti-space
    if time_left() > 120:
        y_anti, c_a = pair_descent_anti(y_anti, n_cur, max_iters=300, rng=rng)
        h_a = h_from_anti(y_anti, n_cur)
        c_a = exact_score(repair_feasible(h_a))
        if c_a < best_c - 1e-13:
            best_h = repair_feasible(h_a)
            best_c = c_a
            y_anti = anti_from_h(best_h)

    # Coordinate descent in anti-space
    if time_left() > 100:
        y_anti, c_a = coord_descent_anti(y_anti, n_cur, max_rounds=20, rng=rng)
        h_a = h_from_anti(y_anti, n_cur)
        c_a = exact_score(repair_feasible(h_a))
        if c_a < best_c - 1e-13:
            best_h = repair_feasible(h_a)
            best_c = c_a
            y_anti = anti_from_h(best_h)

    # Random direction search in anti-space
    if time_left() > 80:
        y_anti, c_a = random_dir_anti(y_anti, n_cur, rng, n_trials=100)
        h_a = h_from_anti(y_anti, n_cur)
        c_a = exact_score(repair_feasible(h_a))
        if c_a < best_c - 1e-13:
            best_h = repair_feasible(h_a)
            best_c = c_a
            y_anti = anti_from_h(best_h)

    # SA in anti-space for escape
    if time_left() > 100:
        y_anti, c_a = sa_anti(y_anti, n_cur, rng, max_iters=400, T_init=0.2, T_final=0.005)
        h_a = h_from_anti(y_anti, n_cur)
        c_a = exact_score(repair_feasible(h_a))
        if c_a < best_c - 1e-13:
            best_h = repair_feasible(h_a)
            best_c = c_a
            y_anti = anti_from_h(best_h)

    # Resolution exploration in anti-space
    if time_left() > 150:
        current_n = len(best_h)
        dims = []
        for factor in [0.8, 0.85, 0.9, 0.95, 1.05, 1.1, 1.15, 1.2, 1.3, 1.4]:
            d = int(round(current_n * factor / 2)) * 2
            d = max(24, min(d, 160))
            if d != current_n and d not in dims:
                dims.append(d)
        dims.sort(key=lambda x: abs(x - current_n))
        for new_n in dims[:8]:
            if time_left() < 30:
                break
            h_new = interpolate_to_n(best_h, new_n)
            y_new = anti_from_h(h_new)
            y_new, c_new = lse_anti(y_new, new_n, beta=80, max_iters=80)
            h_new = repair_feasible(h_from_anti(y_new, new_n))
            c_new = exact_score(h_new)
            if c_new < best_c - 1e-13:
                best_h = h_new
                best_c = c_new
                current_n = new_n
            y_new, c_new = active_set_anti(anti_from_h(best_h if len(best_h) == new_n else interpolate_to_n(best_h, new_n)), new_n, max_iters=60, n_active=10)
            h_test = repair_feasible(h_from_anti(y_new, new_n))
            c_test = exact_score(h_test)
            if c_test < best_c - 1e-13:
                best_h = h_test
                best_c = c_test
                current_n = new_n

    # Full-space pair descent (escape from anti constraint)
    if time_left() > 80:
        h_esc, c_esc = full_pair_descent(best_h, len(best_h), max_iters=300, rng=rng)
        if c_esc < best_c - 1e-13:
            best_h = h_esc
            best_c = c_esc
            y_anti = anti_from_h(best_h)

    # SLSQP polish
    if time_left() > 60:
        h_sol, c_sol = slsqp_solve(best_h, maxiter=200)
        if h_sol is not None and c_sol < best_c - 1e-12:
            best_h = h_sol
            best_c = c_sol
            y_anti = anti_from_h(best_h)

    # More LSE with different betas
    for beta in [40, 120, 200]:
        if time_left() < 25:
            break
        y_tmp = anti_from_h(best_h)
        y_tmp, c_tmp = lse_anti(y_tmp, len(best_h), beta=beta, max_iters=100)
        h_tmp = repair_feasible(h_from_anti(y_tmp, len(best_h)))
        c_tmp = exact_score(h_tmp)
        if c_tmp < best_c - 1e-13:
            best_h = h_tmp
            best_c = c_tmp
            y_anti = anti_from_h(best_h)

    # More active set with varying n_active
    for n_act in [6, 8, 12, 20, 30]:
        if time_left() < 20:
            break
        y_tmp = anti_from_h(best_h)
        y_tmp, c_tmp = active_set_anti(y_tmp, len(best_h), max_iters=100, n_active=n_act)
        h_tmp = repair_feasible(h_from_anti(y_tmp, len(best_h)))
        c_tmp = exact_score(h_tmp)
        if c_tmp < best_c - 1e-13:
            best_h = h_tmp
            best_c = c_tmp
            y_anti = anti_from_h(best_h)

    # Multi-start: perturb and re-optimize
    if time_left() > 100:
        n_b = len(best_h)
        for eps in [0.005, 0.01, 0.02, 0.05]:
            if time_left() < 30:
                break
            z = rng.randn(n_b)
            z -= z.mean()
            h_start = repair_feasible(best_h + eps * z)
            y_start = anti_from_h(h_start)
            y_start, c_tmp = lse_anti(y_start, n_b, beta=80, max_iters=60)
            h_tmp = repair_feasible(h_from_anti(y_start, n_b))
            c_tmp = exact_score(h_tmp)
            if c_tmp < best_c - 1e-12:
                best_h = h_tmp
                best_c = c_tmp
                y_anti = anti_from_h(best_h)

    # Final resolution sweep
    if time_left() > 80:
        n_f = len(best_h)
        for dn in [-4, -2, 2, 4, 6, 8]:
            if time_left() < 20:
                break
            new_n = n_f + dn
            if new_n < 24 or new_n > 160 or new_n % 2 != 0:
                continue
            h_int = interpolate_to_n(best_h, new_n)
            y_int = anti_from_h(h_int)
            y_int, c_int = lse_anti(y_int, new_n, beta=100, max_iters=60)
            h_tmp = repair_feasible(h_from_anti(y_int, new_n))
            c_tmp = exact_score(h_tmp)
            if c_tmp < best_c - 1e-13:
                best_h = h_tmp
                best_c = c_tmp
                n_f = new_n
                y_anti = anti_from_h(best_h)

    # Final pair descent + active set
    if time_left() > 30:
        y_tmp = anti_from_h(best_h)
        y_tmp, c_tmp = pair_descent_anti(y_tmp, len(best_h), max_iters=200, rng=rng)
        h_tmp = repair_feasible(h_from_anti(y_tmp, len(best_h)))
        c_tmp = exact_score(h_tmp)
        if c_tmp < best_c - 1e-13:
            best_h = h_tmp
            best_c = c_tmp

    if time_left() > 20:
        y_tmp = anti_from_h(best_h)
        y_tmp, c_tmp = active_set_anti(y_tmp, len(best_h), max_iters=150, n_active=10)
        h_tmp = repair_feasible(h_from_anti(y_tmp, len(best_h)))
        c_tmp = exact_score(h_tmp)
        if c_tmp < best_c - 1e-13:
            best_h = h_tmp
            best_c = c_tmp

    # Final SLSQP
    if time_left() > 30:
        h_sol, c_sol = slsqp_solve(best_h, maxiter=250)
        if h_sol is not None and c_sol < best_c - 1e-13:
            best_h = h_sol
            best_c = c_sol

    # Final LSE polish
    if time_left() > 15:
        y_tmp = anti_from_h(best_h)
        y_tmp, c_tmp = lse_anti(y_tmp, len(best_h), beta=150, max_iters=80)
        h_tmp = repair_feasible(h_from_anti(y_tmp, len(best_h)))
        c_tmp = exact_score(h_tmp)
        if c_tmp < best_c - 1e-13:
            best_h = h_tmp
            best_c = c_tmp

    # Final validation
    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)

    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            final_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 2:
            final_c = float(eval_result[0])
    except Exception:
        pass

    return best_h, final_c, len(best_h)
