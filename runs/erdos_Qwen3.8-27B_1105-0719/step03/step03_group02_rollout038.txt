def lmo_vertex(h, n, top_q=64):
    """Generate a binary vertex that minimizes the aggregated gradient."""
    corrs = compute_corrs(h)
    top_idx = np.argsort(corrs)[-top_q:]

    v = np.zeros(n)

For coordinate i, the offset for correlation k is `k - (n-1)`.
- i + offset < n means i + k - (n-1) < n, i.e., k < 2n-1-i
- i + offset >= 0 means i + k - (n-1) >= 0, i.e., k >= n-1-i
- Similarly for i - offset: 0 <= i - (k-(n-1)) < n

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.special import logsumexp
import time


def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)


def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def c5_of_h(h, dx):
    return compute_corrs(h).max() * dx


def grad_corr_k(h, k, n):
    offset = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -offset)
    i_hi = min(n - 1, n - 1 - offset)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx + offset] += (1.0 - h[idx])
    grad[idx] += (-h[idx + offset])
    return grad


def coord_optimal_i(h, i, n, corrs):
    """Find optimal h[i] given all other coordinates fixed, using exact 1D minimization."""
    # Compute linear coefficients: corr_k = base_k + h[i] * coeff_k
    m = 2 * n - 1
    coeffs = np.zeros(m)
    for k in range(m):
        offset = k - (n - 1)
        c = 0.0
        j1 = i + offset
        if 0 <= j1 < n:
            c += (1.0 - h[j1])
        j2 = i - offset
        if 0 <= j2 < n:
            c += (-h[j2])
        coeffs[k] = c
    
    base = corrs - h[i] * coeffs
    
    # f(t) = max_k (base[k] + t * coeffs[k]), t in [0,1]
    # Find minimum of this convex piecewise linear function
    
    def f_at(t):
        return np.max(base + t * coeffs)
    
    # Check if minimum is at boundary
    f0 = f_at(0.0)
    f1 = f_at(1.0)
    
    # Slope at t: right derivative = max{coeffs[k] : base[k] + t*coeffs[k] = f(t)}
    # Left derivative = min{coeffs[k] : base[k] + t*coeffs[k] = f(t)}
    
    def right_slope(t):
        vals = base + t * coeffs
        mx = vals.max()
        active = np.where(np.abs(vals - mx) < 1e-12)[0]
        return coeffs[active].max()
    
    def left_slope(t):
        vals = base + t * coeffs
        mx = vals.max()
        active = np.where(np.abs(vals - mx) < 1e-12)[0]
        return coeffs[active].min()
    
    rs0 = right_slope(0.0)
    ls1 = left_slope(1.0)
    
    if rs0 >= 0:
        return 0.0, f0
    if ls1 <= 0:
        return 1.0, f1
    
    # Binary search for zero crossing of slope
    lo, hi = 0.0, 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        ls = left_slope(mid)
        rs = right_slope(mid)
        if ls > 0:
            hi = mid
        elif rs < 0:
            lo = mid
        else:
            break
    
    t_opt = (lo + hi) / 2.0
    f_opt = f_at(t_opt)
    return t_opt, f_opt


def coord_descent(h0, dx, n, max_sweeps=100, tol=1e-12):
    """Exact coordinate descent for the Erdos overlap problem."""
    h = h0.copy()
    target_sum = n / 2.0
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    
    for sweep in range(max_sweeps):
        corrs = compute_corrs(h)
        max_change = 0.0
        
        for i in range(n):
            t_opt, f_opt = coord_optimal_i(h, i, n, corrs)
            if abs(t_opt - h[i]) > 1e-14:
                h[i] = t_opt
                corrs = compute_corrs(h)  # full recompute for safety
                max_change = max(max_change, abs(t_opt - h[i]))
        
        c5_val = c5_of_h(h, dx)
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
        
        if max_change < tol:
            break
    
    return best_h, best_c5


def lmo_vertex(h, n, top_q=64):
    """Generate a balanced binary vertex minimizing aggregated gradient."""
    corrs = compute_corrs(h)
    top_idx = np.argsort(corrs)[-top_q:]
    G = np.zeros(n)
    for k in top_idx:
        G += grad_corr_k(h, k, n)
    G *= 2.0 / n
    
    n_ones = n // 2
    sorted_idx = np.argsort(G)
    v = np.zeros(n)
    v[sorted_idx[:n_ones]] = 1.0
    return v


def lmo_line_search(h, v, dx, n):
    """Line search along (1-t)*h + t*v."""
    best_t = 0.0
    best_val = c5_of_h(h, dx)
    
    # Grid search
    for t in np.linspace(0.05, 1.0, 20):
        h_t = (1.0 - t) * h + t * v
        val = c5_of_h(h_t, dx)
        if val < best_val:
            best_val = val
            best_t = t
    
    # Refine

        hi = min(1.0, best_t + 0.05)

            val = c5_of_h(h_t, dx)
            if val < best_val:
                best_val = val
                best_t = t
    
    return best_t, best_val


def epigraph_solve(h0, dx, n, active_k, maxiter=300):
    """Solve min tau s.t. tau >= dx*corr_k for k in active_k, sum(h)=n/2, 0<=h<=1."""
    target_sum = n / 2.0
    m = 2 * n - 1
    
    tau0 = max(compute_corrs(h0)) * dx + 0.01
    z0 = np.concatenate([h0, [tau0]])
    
    def objective(z):
        return z[n]
    
    def obj_grad(z):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g
    
    cons = []
    for k in active_k:
        def make_ineq(k_val):
            def fun(z):
                return z[n] - dx * compute_corrs(z[:n])[k_val]
            def jac(z):
                g = np.zeros(n + 1)
                g[n] = 1.0
                g[:n] = -dx * grad_corr_k(z[:n], k_val, n)
                return g
            return fun, jac
        
        f, j = make_ineq(k)
        cons.append({'type': 'ineq', 'fun': f, 'jac': j})
    
    def eq_fun(z):
        return z[:n].sum() - target_sum
    def eq_jac(z):
        g = np.zeros(n + 1)
        g[:n] = 1.0
        return g
    cons.append({'type': 'eq', 'fun': eq_fun, 'jac': eq_jac})
    
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    
    try:
        res = minimize(objective, z0, jac=obj_grad, method='SLSQP',
                      bounds=bounds, constraints=cons,

            h_out = res.x[:n]
            h_out = project_feasible(h_out, target_sum)
            return h_out, c5_of_h(h_out, dx)
    except Exception:
        pass
    return h0, c5_of_h(h0, dx)

def make_structured_start(n, seed):

delta_corrs[k] = (new_val - old_val) * coeffs_k

where coeffs_k is the coefficient I already computed! So I can update corrs in O(n) without recomputing from scratch.

Wait, there's a subtlety. When I update h[i], the `coeffs` I computed were based on the current h (before the update). But the coefficient of h[i] in corr_k doesn't depend on h[i] itself (it depends on h[j] for j≠i). So the coeffs are still valid after the update! Let me verify:

coeff_k = (1 - h[i+offset]) if valid + (-h[i-offset]) if valid

This depends on h[i+offset] and h[i-offset], NOT on h[i]. So yes, the coeffs are unchanged by the update to h[i]. 

So the incremental update is:
corrs_new = corrs_old + (t_opt - h[i]) * coeffs

This is correct and O(n) per coordinate step.

- O(n^2) for the initial compute_corrs

Let me also think about the 2D block coordinate descent. For a pair (i,j), the problem is:
min_{t1,t2 ∈ [0,1]} max_k (base_k + t1*a_k + t2*b_k)

where a_k is the coefficient of h[i] and b_k is the coefficient of h[j] in corr_k.

This is a 2D problem. The function is convex (max of linear functions), so the minimum is unique (or on a face). We can solve it by:
1. For a fixed t1, the optimal t2 is found by the same 1D method
2. Then do a 1D search over t1

Or we can use the fact that the minimum of a convex piecewise linear function over a polytope occurs at a vertex. The vertices of the "upper envelope" arrangement in 2D are O(n^2) in the worst case, but in practice much fewer.

A practical approach: use a grid of 10×10 = 100 points, then refine the best with a finer grid or Nelder-Mead.

Actually, for the 2D case, since f(t1,t2) = max_k (c_k + a_k*t1 + b_k*t2) is convex, the minimum over [0,1]^2 is where the subgradient contains (0,0). The subgradient at a point is the convex hull of {(a_k, b_k) : k ∈ argmax}. So we need 0 ∈ conv({(a_k, b_k) : k ∈ argmax}).

This means the minimum is at a point where at least 3 lines are tied (in general position) or where a single line has (a_k, b_k) = (0,0) (unlikely).

A practical algorithm for 2D:
1. Evaluate f at a grid of points
2. Find the best grid point

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.special import logsumexp
import time


def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)


def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def c5_of_h(h, dx):
    return compute_corrs(h).max() * dx


def grad_corr_k(h, k, n):
    offset = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -offset)
    i_hi = min(n - 1, n - 1 - offset)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx + offset] += (1.0 - h[idx])
    grad[idx] += (-h[idx + offset])
    return grad


def compute_coeffs_for_i(h, i, n):
    m = 2 * n - 1
    coeffs = np.zeros(m)
    for k in range(m):
        offset = k - (n - 1)
        c = 0.0
        j1 = i + offset
        if 0 <= j1 < n:
            c += (1.0 - h[j1])
        j2 = i - offset
        if 0 <= j2 < n:
            c += (-h[j2])
        coeffs[k] = c
    return coeffs


def coord_optimal_i(h, i, n, corrs):

    base = corrs - h[i] * coeffs

    def f_at(t):
        return np.max(base + t * coeffs)

    vals_0 = base
    mx_0 = vals_0.max()
    active_0 = np.where(np.abs(vals_0 - mx_0) < 1e-12)[0]
    rs0 = coeffs[active_0].max()
    if rs0 >= 0:
        return 0.0, coeffs

    vals_1 = base + coeffs
    mx_1 = vals_1.max()
    active_1 = np.where(np.abs(vals_1 - mx_1) < 1e-12)[0]
    ls1 = coeffs[active_1].min()
    if ls1 <= 0:

        ls = coeffs[active].min()
        rs = coeffs[active].max()
        if ls > 0:
            hi = mid
        elif rs < 0:
            lo = mid
        else:

    t_opt = (lo + hi) / 2.0
    return t_opt, coeffs


def coord_descent(h0, dx, n, max_sweeps=80, tol=1e-13):
    h = h0.copy()
    target_sum = n / 2.0
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)

    for sweep in range(max_sweeps):
        corrs = compute_corrs(h)
        max_change = 0.0

        for i in range(n):
            t_opt, coeffs = coord_optimal_i(h, i, n, corrs)
            if abs(t_opt - h[i]) > 1e-14:
                delta = t_opt - h[i]
                corrs = corrs + delta * coeffs
                h[i] = t_opt
                max_change = max(max_change, abs(delta))

        if sweep % 5 == 4:
            corrs = compute_corrs(h)

        c5_val = corrs.max() * dx
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()

        if max_change < tol:
            break

    final_c5 = c5_of_h(h, dx)
    if final_c5 < best_c5:
        best_c5 = final_c5
        best_h = h.copy()
    else:
        best_c5 = c5_of_h(best_h, dx)
    return best_h, best_c5


def lmo_vertex(h, n, top_q=64, alpha=0.0):
    corrs = compute_corrs(h)
    top_idx = np.argsort(corrs)[-top_q:]
    if alpha > 0:
        weights = np.exp(alpha * (corrs[top_idx] - corrs.max()))
        weights /= weights.sum()
    else:
        weights = np.ones(len(top_idx)) / len(top_idx)

    G = np.zeros(n)
    for wi, k in zip(weights, top_idx):
        G += wi * grad_corr_k(h, k, n)
    G *= 2.0 / n

    n_ones = n // 2
    sorted_idx = np.argsort(G)
    v = np.zeros(n)
    v[sorted_idx[:n_ones]] = 1.0
    return v


def line_search_hv(h, v, dx, n, n_grid=40):
    d = v - h
    best_t = 0.0
    best_val = c5_of_h(h, dx)

    t_grid = np.linspace(0.0, 1.0, n_grid + 1)
    vals = np.zeros(len(t_grid))
    for idx, t in enumerate(t_grid):
        h_t = h + t * d
        vals[idx] = np.correlate(h_t, 1.0 - h_t, mode='full').max() * dx

    best_idx = np.argmin(vals)
    best_val = vals[best_idx]
    best_t = t_grid[best_idx]

    if 0 < best_idx < len(t_grid) - 1:
        lo_t = t_grid[best_idx - 1]
        hi_t = t_grid[best_idx + 1]
        t_refine = np.linspace(lo_t, hi_t, 30)
        for t in t_refine:

            val = np.correlate(h_t, 1.0 - h_t, mode='full').max() * dx
            if val < best_val:

                best_t = t

    return best_t, best_val


def epigraph_solve(h0, dx, n, active_k, maxiter=300):
    target_sum = n / 2.0
    tau0 = compute_corrs(h0).max() * dx + 0.02
    z0 = np.concatenate([h0, [tau0]])

    def objective(z):
        return z[n]

    def obj_grad(z):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g

    cons = []
    for k in active_k:
        def make_ineq(k_val):
            def fun(z):
                return z[n] - dx * compute_corrs(z[:n])[k_val]

            def jac(z):
                g = np.zeros(n + 1)
                g[n] = 1.0
                g[:n] = -dx * grad_corr_k(z[:n], k_val, n)
                return g
            return fun, jac

        f, j = make_ineq(k)
        cons.append({'type': 'ineq', 'fun': f, 'jac': j})

    def eq_fun(z):
        return z[:n].sum() - target_sum

    def eq_jac(z):
        g = np.zeros(n + 1)
        g[:n] = 1.0
        return g
    cons.append({'type': 'eq', 'fun': eq_fun, 'jac': eq_jac})

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    try:
        res = minimize(objective, z0, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.x is not None:
            h_out = project_feasible(res.x[:n], target_sum)
            return h_out, c5_of_h(h_out, dx)
    except Exception:
        pass
    return h0, c5_of_h(h0, dx)


def epigraph_iterative(h0, dx, n, max_rounds=5, n_active_init=30, maxiter_per=200):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    target_sum = n / 2.0

    for rnd in range(max_rounds):
        corrs = compute_corrs(h)
        n_act = min(n_active_init + rnd * 10, 2 * n - 1)
        active_k = np.argsort(corrs)[-n_act:]

        h_new, c5_new = epigraph_solve(h, dx, n, active_k, maxiter=maxiter_per)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new

    return best_h, best_c5


def lp_polish(h0, dx, target_sum, n_steps=40, lam=0.2):

            best_h = h_cd

    return best_h, best_c5


def make_binary_start(n, seed):
    rng = np.random.RandomState(seed)
    idx = rng.choice(n, size=n // 2, replace=False)
    h = np.zeros(n)
    h[idx] = 1.0
    return h


def make_half_periodic(n, seed):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    period = max(4, n // 20)
    for i in range(n):
        h[i] = 1.0 if (i % (2 * period)) < period else 0.0
    offsets = rng.randint(0, period, size=5)
    for off in offsets:
        h = np.roll(h, off)
    return project_feasible(h, n / 2.0)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s - (time.time() - t0)

    best_h = None
    best_c5 = float('inf')

    # Phase 1: Start from initial construction
    try:
        h0 = np.array(initial_h_values, dtype=float)
    except NameError:
        h0 = make_binary_start(146, seed)
    n = len(h0)
    dx = 2.0 / n
    target_sum = n / 2.0
    h0 = project_feasible(h0, target_sum)

    h_opt, c5_val = full_optimize(h0, dx, n, time_left, "main")
    best_h = h_opt
    best_c5 = c5_val

    # Phase 2: Iterative LMO exploration from best
    if time_left() > 30:
        n_b = len(best_h)
        dx_b = 2.0 / n_b
        ts_b = n_b / 2.0
        for iteration in range(12):
            if time_left() < 10:
                break
            for q in [48, 80, 120]:
                if time_left() < 5:
                    break
                v = lmo_vertex(best_h, n_b, top_q=q, alpha=2.0)
                t_opt, val = line_search_hv(best_h, v, dx_b, n_b)
                if t_opt > 0.1:
                    h_try = (1.0 - t_opt) * best_h + t_opt * v
                    h_try = project_feasible(h_try, ts_b)
                    h_cd, c5_cd = coord_descent(h_try, dx_b, n_b, max_sweeps=30)
                    if c5_cd < best_c5:
                        best_c5 = c5_cd
                        best_h = h_cd

    # Phase 3: Try different n values
    n_current = len(best_h)
    n_candidates = [120, 130, 136, 138, 140, 142, 144, 146, 148, 150, 152, 154, 156, 158, 160, 164, 168, 172, 176, 180, 184, 190, 196, 200, 204, 210, 220, 232]
    n_candidates = [x for x in n_candidates if x % 2 == 0]
    n_candidates.sort(key=lambda x: abs(x - n_current))
    n_candidates = n_candidates[:12]

    for n_try in n_candidates:
        if time_left() < 25:
            break
        if n_try == n_current:
            continue
        dx_try = 2.0 / n_try
        ts_try = n_try / 2.0

        x_old = np.linspace(0.0, 2.0, len(best_h), endpoint=False) + 1.0 / len(best_h)
        x_new = np.linspace(0.0, 2.0, n_try, endpoint=False) + 1.0 / n_try
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, ts_try)

        h_opt, c5_val = full_optimize(h_interp, dx_try, n_try, time_left, f"n{n_try}")
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_opt
            n_current = n_try

    # Phase 4: Structured and random binary starts
    n_try = len(best_h)
    dx_try = 2.0 / n_try
    ts_try = n_try / 2.0

    starts = []
    for s in range(4):
        starts.append(make_half_periodic(n_try, seed + 200 + s))
    for s in range(4):
        starts.append(make_binary_start(n_try, seed + 300 + s))

    for h_start in starts:
        if time_left() < 15:
            break
        h_cd, c5_cd = coord_descent(h_start, dx_try, n_try, max_sweeps=40)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd
            h_opt, c5_val = full_optimize(h_cd, dx_try, n_try, time_left, "structured")
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h_opt

    # Phase 5: Intensive final optimization
    if time_left() > 20:
        n_b = len(best_h)
        dx_b = 2.0 / n_b
        ts_b = n_b / 2.0

        h_cd, c5_cd = coord_descent(best_h, dx_b, n_b, max_sweeps=80)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd

        if time_left() > 15:
            corrs = compute_corrs(best_h)
            n_act = min(50, 2 * n_b - 1)
            active_k = np.argsort(corrs)[-n_act:]
            h_ep, c5_ep = epigraph_solve(best_h, dx_b, n_b, active_k, maxiter=300)
            if c5_ep < best_c5:
                best_c5 = c5_ep
                best_h = h_ep
                h_cd, c5_cd = coord_descent(best_h, dx_b, n_b, max_sweeps=40)
                if c5_cd < best_c5:
                    best_c5 = c5_cd
                    best_h = h_cd

        if time_left() > 10:
            h_lp, c5_lp = lp_polish(best_h, dx_b, ts_b, n_steps=30, lam=0.1)
            if c5_lp < best_c5:
                best_c5 = c5_lp
                best_h = h_lp
                h_cd, c5_cd = coord_descent(best_h, dx_b, n_b, max_sweeps=20)
                if c5_cd < best_c5:
                    best_c5 = c5_cd
                    best_h = h_cd

        if time_left() > 8:
            h_pt, c5_pt = pair_transfer_polish(best_h, dx_b, ts_b, n_top=25, n_iter=4)
            if c5_pt < best_c5:
                best_c5 = c5_pt
                best_h = h_pt

    # Phase 6: Ultra-fine final coordinate descent
    if time_left() > 5 and best_h is not None:
        n_b = len(best_h)
        dx_b = 2.0 / n_b
        h_cd, c5_cd = coord_descent(best_h, dx_b, n_b, max_sweeps=50)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd
        h_pt, c5_pt = pair_transfer_polish(best_h, dx_b, n_b / 2.0, n_top=30, n_iter=3)
        if c5_pt < best_c5:
            best_c5 = c5_pt
            best_h = h_pt

    # Ensure feasibility
    if best_h is None:
        best_h = make_binary_start(146, seed)
        best_h = project_feasible(best_h, 73.0)
        best_c5 = c5_of_h(best_h, 2.0 / 146)
    else:
        n_final = len(best_h)
        best_h = project_feasible(best_h, n_final / 2.0)
        best_c5 = c5_of_h(best_h, 2.0 / n_final)

    return best_h, best_c5, len(best_h)
