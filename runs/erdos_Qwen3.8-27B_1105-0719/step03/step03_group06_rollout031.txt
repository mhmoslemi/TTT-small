    best_c = exact_score(h)

        alpha_max = max(alpha_max, 1e-12)
        alpha_max = min(alpha_max, 0.5)
        
        # Golden section line search
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        a_ls, b_ls = 0.0, alpha_max
        best_alpha = 0.0
        best_val = c
        best_h_try = h.copy()
        for _ in range(20):
            c1 = b_ls - gr * (b_ls - a_ls)
            c2 = a_ls + gr * (b_ls - a_ls)
            h1 = repair_feasible(h + c1 * d)
            h2 = repair_feasible(h + c2 * d)
            f1 = exact_score(h1)
            f2 = exact_score(h2)

                best_alpha = c1

            if f2 < best_val:

def interp_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()

    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False) + 1.0/new_n
    h_new = np.interp(new_x, old_x, h)
    return repair_feasible(h_new)

def golden_seed(n, rng=None):
    """Generate a near-binary seed using golden ratio."""
    half = n // 2
    idx = np.arange(n)
    phi = 0.618033988749895
    vals = (idx * phi) % 1.0
    top = np.argsort(vals)[-half:]
    h = np.zeros(n)
    h[top] = 1.0
    return repair_feasible(h)

def structured_seed(n, pattern, rng=None):
    """Generate various structured seeds."""
    half = n // 2
    if pattern == 0:
        return golden_seed(n, rng)
    elif pattern == 1:
        # Alternating blocks
        block = max(1, n // (2 * half))
        h = np.zeros(n)
        for i in range(n):
            if (i // block) % 2 == 0:
                h[i] = 1.0
        # Adjust to exactly n/2 ones
        ones = int(h.sum())
        if ones > half:
            excess = ones - half
            one_idx = np.where(h == 1.0)[0]
            if rng is not None:
                to_flip = rng.choice(one_idx, min(excess, len(one_idx)), replace=False)
            else:
                to_flip = one_idx[:excess]
            h[to_flip] = 0.0
        elif ones < half:
            deficit = half - ones
            zero_idx = np.where(h == 0.0)[0]
            if rng is not None:
                to_flip = rng.choice(zero_idx, min(deficit, len(zero_idx)), replace=False)
            else:
                to_flip = zero_idx[:deficit]
            h[to_flip] = 1.0
        return repair_feasible(h)
    elif pattern == 2:
        # Cosine-based
        x = np.linspace(0, 2 * np.pi * 3, n)
        h = (np.cos(x) > 0).astype(float)
        ones = int(h.sum())
        if ones > half:
            excess = ones - half
            one_idx = np.where(h == 1.0)[0]
            h[one_idx[:excess]] = 0.0
        elif ones < half:
            deficit = half - ones
            zero_idx = np.where(h == 0.0)[0]
            h[zero_idx[:deficit]] = 1.0
        return repair_feasible(h)
    elif pattern == 3:
        # Squaring pattern (Thue-Morse like)
        h = np.zeros(n)
        for i in range(n):
            # Count trailing zeros
            v = i
            cnt = 0
            while v % 2 == 0 and v > 0:
                cnt += 1
                v //= 2
            h[i] = 1.0 if cnt % 2 == 0 else 0.0
        ones = int(h.sum())
        if ones > half:
            excess = ones - half
            one_idx = np.where(h == 1.0)[0]
            h[one_idx[:excess]] = 0.0
        elif ones < half:
            deficit = half - ones
            zero_idx = np.where(h == 0.0)[0]
            h[zero_idx[:deficit]] = 1.0
        return repair_feasible(h)
    else:
        # Random
        h = np.zeros(n)
        if rng is not None:
            perm = rng.permutation(n)
            h[perm[:half]] = 1.0
        else:
            h[:half] = 1.0
        return repair_feasible(h)

        pos_mask = d > 1e-12
        neg_mask = d < -1e-12
        alpha_max = 1.0
        if pos_mask.any():
            alpha_max = min(alpha_max, float(((1.0 - h[pos_mask]) / d[pos_mask]).min()))
        if neg_mask.any():
            alpha_max = min(alpha_max, float((h[neg_mask] / (-d[neg_mask])).min()))
        alpha_max = max(alpha_max, 1e-12)
        alpha_max = min(alpha_max, 0.4)
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        a_ls, b_ls = 0.0, alpha_max
        best_alpha = 0.0
        best_val = c
        best_h_try = h.copy()
        for _ in range(22):
            c1 = b_ls - gr * (b_ls - a_ls)
            c2 = a_ls + gr * (b_ls - a_ls)
            h1 = repair_feasible(h + c1 * d)
            h2 = repair_feasible(h + c2 * d)
            f1 = exact_score(h1)
            f2 = exact_score(h2)
            if f1 < best_val:
                best_val = f1
                best_alpha = c1
                best_h_try = h1.copy()
            if f2 < best_val:
                best_val = f2
                best_alpha = c2
                best_h_try = h2.copy()
            if f1 < f2:
                b_ls = c2
            else:
                a_ls = c1
            if b_ls - a_ls < 1e-12:
                break
        if best_val >= c - 1e-14:
            break
        h = best_h_try
    final_c = exact_score(h)
    if final_c < best_c:
        best_c = final_c
        best_h = h.copy()
    return best_h, best_c

def interp_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, n, endpoint=False) + 1.0 / n
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False) + 1.0 / new_n
    h_new = np.interp(new_x, old_x, h)
    return repair_feasible(h_new)

def make_seed(n, pattern, rng):
    half = n // 2
    if pattern == 0:
        idx = np.arange(n)
        phi = 0.618033988749895
        vals = (idx * phi) % 1.0
        top = np.argsort(vals)[-half:]
        h = np.zeros(n)
        h[top] = 1.0
    elif pattern == 1:
        idx = np.arange(n)
        vals = (idx * 0.4142135623730951) % 1.0
        top = np.argsort(vals)[-half:]
        h = np.zeros(n)
        h[top] = 1.0
    elif pattern == 2:
        h = np.zeros(n)
        for i in range(n):
            cnt = 0
            v = i if i > 0 else 1
            while v % 2 == 0:
                cnt += 1
                v //= 2
            h[i] = 1.0 if cnt % 2 == 0 else 0.0
        ones = int(h.sum())
        if ones > half:
            excess = ones - half
            one_idx = np.where(h == 1.0)[0]
            if excess <= len(one_idx):
                picked = rng.choice(one_idx, excess, replace=False)
                h[picked] = 0.0
        elif ones < half:
            deficit = half - ones
            zero_idx = np.where(h == 0.0)[0]
            if deficit <= len(zero_idx):
                picked = rng.choice(zero_idx, deficit, replace=False)
                h[picked] = 1.0
    elif pattern == 3:
        h = (np.sin(np.linspace(0, 4 * np.pi, n)) > 0).astype(float)
        ones = int(h.sum())
        if ones > half:
            excess = ones - half
            one_idx = np.where(h == 1.0)[0]
            if excess <= len(one_idx):
                picked = rng.choice(one_idx, excess, replace=False)
                h[picked] = 0.0
        elif ones < half:
            deficit = half - ones
            zero_idx = np.where(h == 0.0)[0]
            if deficit <= len(zero_idx):
                picked = rng.choice(zero_idx, deficit, replace=False)
                h[picked] = 1.0
    elif pattern == 4:
        perm = rng.permutation(n)
        h = np.zeros(n)
        h[perm[:half]] = 1.0
    else:
        idx = np.arange(n)
        vals = (idx * 0.7548776662466928) % 1.0
        top = np.argsort(vals)[-half:]
        h = np.zeros(n)
        h[top] = 1.0
    return repair_feasible(h)

def smooth_binary(h_bin, alpha):
    n = len(h_bin)
    h = alpha * h_bin + (1.0 - alpha) * 0.5
    return repair_feasible(h)

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()
    rng = np.random.RandomState(seed)

    try:
        parent = np.array(initial_h_values, dtype=np.float64)
    except Exception:
        parent = None
    if parent is None or len(parent) < 4:
        parent = np.full(70, 0.5)
    parent = np.clip(parent, 0.0, 1.0)
    parent = repair_feasible(parent)

    best_h = parent.copy()
    best_c = exact_score(parent)

    # Phase 1: LSE descent on parent
    if budget_s - (time.perf_counter() - t_start) > 80:
        alloc = min((budget_s - (time.perf_counter() - t_start)) * 0.22, 220)
        h1, c1 = lse_descent(best_h, t_start, alloc, max_iters=4000,
                            beta_start=15.0, beta_end=8000.0, use_momentum=True)
        if c1 < best_c:
            best_h, best_c = h1, c1

    # Phase 2: Active descent polish
    if budget_s - (time.perf_counter() - t_start) > 50:
        alloc = min((budget_s - (time.perf_counter() - t_start)) * 0.12, 100)
        h2, c2 = active_descent(best_h, max_iters=600, active_tol=0.003,
                                t_start=t_start, time_limit=alloc)
        if c2 < best_c:
            best_h, best_c = h2, c2

    # Phase 3: Dimension search
    n_best = len(best_h)
    dims_to_try = []
    for dn in range(-24, 28, 2):
        d = n_best + dn
        if 40 <= d <= 130 and d % 2 == 0 and d != n_best:
            dims_to_try.append(d)
    dims_to_try.sort(key=lambda x: abs(x - n_best))

    for new_n in dims_to_try[:14]:
        if budget_s - (time.perf_counter() - t_start) < 15:
            break
        alloc = min((budget_s - (time.perf_counter() - t_start)) * 0.12, 50)
        h_interp = interp_to_n(best_h, new_n)
        h3, c3 = lse_descent(h_interp, t_start, alloc, max_iters=2000,
                            beta_start=30.0, beta_end=8000.0, use_momentum=True)
        if c3 < best_c:
            best_h, best_c = h3, c3
        if budget_s - (time.perf_counter() - t_start) > 12 and c3 < best_c + 0.003:
            h4, c4 = active_descent(h3, max_iters=250, active_tol=0.002,
                                    t_start=t_start,
                                    time_limit=min((budget_s - (time.perf_counter() - t_start)) * 0.06, 30))
            if c4 < best_c:
                best_h, best_c = h4, c4

    # Phase 4: Structured seeds
    if budget_s - (time.perf_counter() - t_start) > 60:
        dims_seeds = [n_best - 4, n_best, n_best + 4, n_best + 8]
        dims_seeds = [d for d in dims_seeds if 40 <= d <= 130 and d % 2 == 0]
        for new_n in dims_seeds:
            for pat in range(5):
                if budget_s - (time.perf_counter() - t_start) < 8:
                    break
                h_bin = make_seed(new_n, pat, rng)
                h_mix = smooth_binary(h_bin, 0.5)
                alloc = min((budget_s - (time.perf_counter() - t_start)) * 0.06, 20)
                h5, c5 = lse_descent(h_mix, t_start, alloc, max_iters=1000,
                                    beta_start=50.0, beta_end=6000.0, use_momentum=True)
                if c5 < best_c:
                    best_h, best_c = h5, c4 if False else c5
                    best_h, best_c = h5, c5

    # Phase 5: Dense LSE refinement (alternating LSE and active)
    if budget_s - (time.perf_counter() - t_start) > 60:
        for round_idx in range(3):
            if budget_s - (time.perf_counter() - t_start) < 20:
                break
            alloc = min((budget_s - (time.perf_counter() - t_start)) * 0.2, 150)
            h6, c6 = lse_descent(best_h, t_start, alloc, max_iters=3000,
                                beta_start=50 + round_idx * 100,
                                beta_end=10000 + round_idx * 20000,
                                use_momentum=True)
            if c6 < best_c:
                best_h, best_c = h6, c6
            if budget_s - (time.perf_counter() - t_start) > 15:
                alloc2 = min((budget_s - (time.perf_counter() - t_start)) * 0.1, 60)
                h7, c7 = active_descent(best_h, max_iters=400, active_tol=0.001,
                                        t_start=t_start, time_limit=alloc2)
                if c7 < best_c:
                    best_h, best_c = h7, c7

    # Phase 6: Final tight active descent
    if budget_s - (time.perf_counter() - t_start) > 20:
        alloc = min((budget_s - (time.perf_counter() - t_start)) * 0.4, 100)
        h8, c8 = active_descent(best_h, max_iters=1500, active_tol=0.0003,
                                t_start=t_start, time_limit=alloc)
        if c8 < best_c:
            best_h, best_c = h8, c8

    # Phase 7: Very high beta LSE (essentially exact max subgradient)
    if budget_s - (time.perf_counter() - t_start) > 15:
        alloc = min((budget_s - (time.perf_counter() - t_start)) * 0.5, 60)
        h9, c9 = lse_descent(best_h, t_start, alloc, max_iters=2000,
                            beta_start=2000.0, beta_end=200000.0, use_momentum=False)
        if c9 < best_c:
            best_h, best_c = h9, c9

    # Phase 8: Final micro active descent
    if budget_s - (time.perf_counter() - t_start) > 5:
        alloc = min((budget_s - (time.perf_counter() - t_start)) * 0.7, 30)
        h10, c10 = active_descent(best_h, max_iters=500, active_tol=0.0001,
                                  t_start=t_start, time_limit=alloc)
        if c10 < best_c:
            best_h, best_c = h10, c10

    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)

    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            final_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 2:
            final_c = float(eval_result[0])
    except Exception:
        pass

    return best_h, final_c, len(best_h)
