import numpy as np
from scipy.optimize import minimize
import time

try:
    import cvxpy as cp
    HAS_CVXPY = True
except ImportError:
    HAS_CVXPY = False


def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))


def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx


def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx] + 1e-15)
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx] + 1e-15)
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h


def build_gradient(h):
    """Build gradient matrix G where G[m, p] = d(profile[m])/d(h[p])."""
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    G = np.zeros((M, n))
    
    center = n - 1
    # Lag d = 0 (center):
    # c[center] = sum_i h[i]*(1-h[i])
    # d/dh[p] = 1 - 2*h[p]
    G[center, :] = dx * (1.0 - 2.0 * h)
    
    for d in range(1, n):
        # d > 0: c[center+d] = sum_{i=0}^{n-1-d} h[i+d]*(1-h[i])
        # d/dh[p]: (1-h[p-d]) if p>=d, minus h[p+d] if p < n-d
        m_pos = center + d
        # Positive contribution: p in [d, n-1], deriv = (1-h[p-d])
        p_range = np.arange(d, n)
        G[m_pos, p_range] += dx * (1.0 - h[p_range - d])
        # Negative contribution: p in [0, n-1-d], deriv = -h[p+d]
        p_range2 = np.arange(0, n - d)
        G[m_pos, p_range2] += -dx * h[p_range2 + d]
        
        # d < 0: c[center-d] = sum_{i=0}^{n-1-d} h[i]*(1-h[i+d])
        # d/dh[p]: (1-h[p+d]) if p < n-d, minus h[p-d] if p>=d
        m_neg = center - d
        p_range3 = np.arange(0, n - d)
        G[m_neg, p_range3] += dx * (1.0 - h[p_range3 + d])
        p_range4 = np.arange(d, n)
        G[m_neg, p_range4] += -dx * h[p_range4 - d]
    
    return G


def verify_gradient(h):
    """Verify gradient with finite differences on a small example."""
    n = len(h)
    G = build_gradient(h)
    profile = compute_profile(h)
    eps = 1e-7
    error = 0
    for p in range(min(n, 5)):
        h_plus = h.copy()
        h_plus[p] += eps
        h_minus = h.copy()
        h_minus[p] -= eps
        # Need to maintain feasibility for FD check - just check raw derivative
        # Actually the gradient is of the raw profile, not the feasible one
        fp = np.correlate(h_plus, 1 - h_plus, mode="full") * (2.0/n)
        fm = np.correlate(h_minus, 1 - h_minus, mode="full") * (2.0/n)
        fd = (fp - fm) / (2 * eps)
        error = max(error, np.max(np.abs(fd - G[:, p])))
    return error


def lse_gradient(h, profile, eta):
    """Compute log-sum-exp weighted gradient direction."""
    n = len(h)
    G = build_gradient(h)
    # Soft weights
    beta = 1.0 / eta
    z = beta * (profile - profile.max())
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    w = exp_z / exp_z.sum()
    # Gradient of LSE
    grad = G.T @ w
    # Project out the mean (to maintain sum constraint direction)
    grad -= grad.mean()
    return grad


def lse_optimize(h0, eta_schedule, max_iter, budget_fn, reserved_time=10.0):
    """Log-sum-exp based optimization."""
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c = exact_score(h)
    
    for eta in eta_schedule:
        if budget_fn() < reserved_time:
            break
        profile = compute_profile(h)
        grad = lse_gradient(h, profile, eta)
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            continue
        grad /= gn
        
        # Adaptive step size with backtracking
        alpha = min(0.05, eta * 5)
        for _ in range(max_iter):
            if budget_fn() < reserved_time:
                break
            h_try = repair_feasible(h - alpha * grad)
            c_try = exact_score(h_try)
            if c_try < best_c - 1e-14:
                h = h_try
                best_c = c_try
                best_h = h.copy()
                alpha = min(alpha * 1.1, 0.05)
            elif c_try < exact_score(h) - 1e-14:
                h = h_try
                alpha = min(alpha * 1.05, 0.05)
            else:
                alpha *= 0.6
                if alpha < 1e-5:
                    break
            # Recompute gradient periodically
            profile = compute_profile(h)
            grad = lse_gradient(h, profile, eta)
            gn = np.linalg.norm(grad)
            if gn < 1e-15:
                break
            grad /= gn
    
    return h, best_c


def trust_qp(h0, profile, K, trust_radius, mu, budget_fn, reserved_time=10.0):
    """Solve trust-region QP with top K active constraints."""
    if not HAS_CVXPY:
        return None
    
    n = len(h0)
    dx = 2.0 / n
    M = 2 * n - 1
    c_max = np.max(profile)
    
    # Select top K lags
    if K < M:
        active = np.argsort(profile)[-K:]
    else:
        active = np.arange(M)
    
    G = build_gradient(h0)
    G_a = G[active, :]
    b_a = profile[active] - G_a @ h0  # P[a] = G_a @ h0 + b_a (linearized)
    
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    
    cons = [
        h_var >= 0,
        h_var <= 1,
        cp.sum(h_var) == n / 2.0,
        h_var >= h0 - trust_radius,
        h_var <= h0 + trust_radius,
        G_a @ h_var - t_var <= -b_a
    ]
    
    obj = cp.Minimize(t_var + mu * cp.sum(cp.square(h_var - h0)))
    prob = cp.Problem(obj, cons)
    
    for solver in [cp.ECOS, cp.SCS, cp.CLARABEAU]:
        try:
            prob.solve(solver=solver, max_iters=20000, abstol=1e-9, reltol=1e-9)
            break
        except Exception:
            continue
    
    if h_var.value is None:
        return None
    h_sol = np.array(h_var.value).flatten()
    if not np.isfinite(h_sol).all():
        return None
    return repair_feasible(h_sol)


def slsqp_refine(h0, maxiter, tol, budget_fn, reserved_time=10.0):
    n = len(h0)
    if n > 200:
        return None
    dx = 2.0 / n
    M = 2 * n - 1
    
    def obj(x):
        return x[-1]
    
    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    
    def eq_fun(x):
        return x[:-1].sum() - n / 2.0
    
    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g
    
    def ineq_fun(x):
        h = x[:-1]
        return x[-1] - np.correlate(h, 1 - h, mode="full") * dx
    
    def ineq_jac(x):
        h = x[:-1]
        G = build_gradient(h)
        J = np.zeros((M, n + 1))
        J[:, :-1] = -G
        J[:, -1] = 1.0
        return J
    
    t0 = exact_score(h0) + 1e-4
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, t0 + 2.0)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': tol})
        if res.x is not None and np.isfinite(res.x).all():
            hs = repair_feasible(res.x[:-1])
            cs = exact_score(hs)
            if np.isfinite(cs):
                return hs
    except Exception:
        pass
    return None


def interp(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    fx = np.concatenate([[0.0], xs, [2.0]])
    fh = np.concatenate([[h[0]], h, [h[-1]]])
    hn = np.interp(xt, fx, fh)
    return repair_feasible(np.clip(hn, 0, 1))


def downsample(h, new_n):
    """Downsample by averaging groups."""
    n = len(h)
    if new_n >= n:
        return h.copy()
    # Simple box filter approach
    ratio = n / new_n
    result = np.zeros(new_n)
    for i in range(new_n):
        start = int(i * ratio)
        end = int((i + 1) * ratio)
        if end > n:
            end = n
        result[i] = h[start:end].mean()
    return repair_feasible(np.clip(result, 0, 1))


def upsample(h, factor):
    """Upsample by repeating each element."""
    result = np.repeat(h, factor)
    return repair_feasible(result)

        bh, bc = combined_optimize(best_h, budget_fn=tl, reserved_time=120,
                                   max_rounds=100, use_momentum=True)
        if bc < best_c - 1e-14:
            best_h, best_c = bh, bc
    
    # Phase 2: Dimension exploration
    if tl() > 150:
        n0 = len(best_h)
        dims = [d for d in range(max(28, n0 - 20), min(110, n0 + 21)) if d != n0]
        dims.sort(key=lambda d: abs(d - n0))
        for nd in dims[:12]:
            if tl() < 80:
                break
            hi = interp(best_h, nd)
            ci = exact_score(hi)
            if ci < best_c + 0.012:
                bh, bc = combined_optimize(hi, budget_fn=tl, reserved_time=60,
                                           max_rounds=40, use_momentum=True)
                if bc < best_c - 1e-14:
                    best_h, best_c = bh, bc
    
    # Phase 3: SLSQP
    if tl() > 120:
        hs = slsqp_refine(best_h, maxiter=400, tol=1e-12)
        if hs is not None:
            cs = exact_score(hs)
            if cs < best_c - 1e-14:
                best_h, best_c = hs, cs
    
    # Phase 4: Post-SLSQP combined
    if tl() > 80:
        bh, bc = combined_optimize(best_h, budget_fn=tl, reserved_time=60,
                                   max_rounds=50, use_momentum=False)
        if bc < best_c - 1e-14:
            best_h, best_c = bh, bc
    
    # Phase 5: Fine dimension search
    if tl() > 70:
        n0 = len(best_h)
        for nd in [n0 - 3, n0 - 2, n0 - 1, n0 + 1, n0 + 2, n0 + 3,
                   n0 - 5, n0 + 5]:
            if tl() < 35 or nd < 24 or nd > 160:
                continue
            hi = interp(best_h, nd)
            ci = exact_score(hi)
            if ci < best_c + 0.004:
                bh, bc = combined_optimize(hi, budget_fn=tl, reserved_time=35,
                                           max_rounds=30, use_momentum=True)
                if bc < best_c - 1e-14:
                    best_h, best_c = bh, bc
    
    # Phase 6: Random perturbations
    if tl() > 90:
        for trial in range(8):
            if tl() < 40:
                break
            eps = 0.004 + 0.01 * trial
            z = rng.randn(len(best_h))
            z -= z.mean()
            z /= (np.linalg.norm(z) / len(best_h) + 1e-15)
            hp = repair_feasible(best_h + eps * z)
            cp_ = exact_score(hp)
            if cp_ < best_c + 0.008:
                bh, bc = combined_optimize(hp, budget_fn=tl, reserved_time=30,
                                           max_rounds=25, use_momentum=True)
                if bc < best_c - 1e-14:
                    best_h, best_c = bh, bc
    
    # Phase 7: Final SLSQP
    if tl() > 60:
        hs = slsqp_refine(best_h, maxiter=500, tol=1e-13)
        if hs is not None:
            cs = exact_score(hs)
            if cs < best_c - 1e-14:
                best_h, best_c = hs, cs
    
    # Phase 8: Final tight pass
    if tl() > 30:
        bh, bc = combined_optimize(best_h, budget_fn=tl, reserved_time=15,
                                   max_rounds=30, use_momentum=False)
        if bc < best_c - 1e-14:
            best_h, best_c = bh, bc
    
    # Final validation
    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)
    
    try:
        er = evaluate_erdos_solution(best_h)
        if isinstance(er, (float, int)):
            final_c = min(final_c, float(er))
        elif isinstance(er, tuple) and len(er) >= 2:
            final_c = min(final_c, float(er[0]))
    except Exception:
        pass
    
    return best_h, final_c, len(best_h)
