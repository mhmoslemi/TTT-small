def slsqp_target(h0, dx, target_sum, beta, theta, maxiter=500):
    """SLSQP with softplus target penalty."""
    n = len(h0)
    
    def obj(x):
        corrs = compute_corrs(x) * dx
        # softplus penalty: sum of log(1 + exp(beta*(corr_k - theta)))
        val = beta * (corrs - theta)
        # stable softplus
        sp = np.where(val > 0, val + np.log1p(np.exp(-val)), np.log1p(np.exp(val)))
        return sp.sum()
    
    def grad_obj(x):
        corrs = compute_corrs(x) * dx
        val = beta * (corrs - theta)
        # sigmoid
        sig = np.where(val > 0, 1.0 / (1.0 + np.exp(-val)), np.exp(val) / (1.0 + np.exp(val)))
        G = grad_all_corrs(x, n) * dx  # (2n-1, n)
        return (sig[:, None] * G).sum(axis=0)
    
    def c_fun(x):
        return x.sum() - target_sum
    
    def c_jac(x):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': c_fun, 'jac': c_jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0, jac=grad_obj, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        h_out = project_feasible(res.x, target_sum)
    except Exception:
        h_out = project_feasible(h0, target_sum)
    return h_out, c5_of_h(h_out, dx)


def slsqp_smooth(h0, dx, target_sum, beta, maxiter=500):
    """SLSQP with logsumexp smoothing."""
    n = len(h0)
    
    def obj(x):
        corrs = compute_corrs(x)
        return logsumexp(beta * corrs) / beta * dx
    
    def grad_obj(x):
        return smooth_max_grad(x, dx, beta)
    
    def c_fun(x):
        return x.sum() - target_sum
    
    def c_jac(x):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': c_fun, 'jac': c_jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0, jac=grad_obj, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        h_out = project_feasible(res.x, target_sum)
    except Exception:
        h_out = project_feasible(h0, target_sum)
    return h_out, c5_of_h(h_out, dx)


def pair_transfer_polish(h0, dx, target_sum, n_top=30, n_iter=5):
    """Pair transfer with exact line search."""
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    
    for iteration in range(n_iter):
        corrs = compute_corrs(h)
        grad = smooth_max_grad(h, dx, 500.0)
        
        dec_idx = np.argsort(grad)[-n_top:]
        inc_idx = np.argsort(grad)[:n_top]
        
        improved = False
        for i in dec_idx:
            for j in inc_idx:
                if i == j:
                    continue
                # Move eps from i to j
                eps_hi = min(h[i], 1.0 - h[j])
                if eps_hi < 1e-12:
                    continue
                
                # d(corr[k])/d(eps) for moving from i to j
                d = np.zeros(2 * n - 1)
                for k in range(2 * n - 1):
                    gk = grad_corr_k(h, k, n)
                    d[k] = gk[j] - gk[i]
                
                # Golden section search on [0, eps_hi]
                gr = (np.sqrt(5.0) - 1.0) / 2.0
                a, b = 0.0, eps_hi
                c_pt = b - gr * (b - a)
                d_pt = a + gr * (b - a)
                fc = (corrs + c_pt * d).max()
                fd = (corrs + d_pt * d).max()
                for _ in range(40):
                    if b - a < 1e-14:
                        break
                    if fc < fd:
                        b = d_pt
                        d_pt = c_pt
                        fd = fc
                        c_pt = b - gr * (b - a)
                        fc = (corrs + c_pt * d).max()
                    else:
                        a = c_pt
                        c_pt = d_pt
                        fc = fd
                        d_pt = a + gr * (b - a)
                        fd = (corrs + d_pt * d).max()
                
                eps_opt = (a + b) / 2.0
                if eps_opt > 1e-10:
                    new_h = h.copy()
                    new_h[i] -= eps_opt
                    new_h[j] += eps_opt
                    new_h = np.clip(new_h, 0.0, 1.0)
                    new_h = project_feasible(new_h, target_sum)
                    new_c5 = c5_of_h(new_h, dx)
                    if new_c5 < best_c5 - 1e-14:
                        h = new_h
                        best_c5 = new_c5
                        best_h = h.copy()
                        corrs = compute_corrs(h)
                        improved = True
        
        if not improved:
            break
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.special import logsumexp
import time


def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)


def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def c5_of_h(h, dx):
    return compute_corrs(h).max() * dx


def grad_corr_k(h, k, n):
    offset = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -offset)
    i_hi = min(n - 1, n - 1 - offset)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx + offset] += (1.0 - h[idx])
    grad[idx] += (-h[idx + offset])
    return grad


def grad_vector_for_i(h, i, n):
    g = np.zeros(2 * n - 1)
    k_lo1 = n - 1 - i
    k_hi1 = n - 1 + i
    idx1 = np.arange(k_lo1, k_hi1 + 1)
    offsets1 = idx1 - (n - 1)
    g[idx1] += (1.0 - h[i + offsets1])
    k_lo2 = i
    k_hi2 = i + n - 1
    idx2 = np.arange(k_lo2, k_hi2 + 1)
    offsets2 = idx2 - (n - 1)
    g[idx2] += (-h[i - offsets2])
    return g


def smooth_max_grad(h, dx, beta):
    n = len(h)
    corrs = compute_corrs(h)
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones_like(corrs) / len(corrs)
    else:
        w /= w_sum
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] > 1e-13:
            grad += w[k] * grad_corr_k(h, k, n)
    return grad * dx


def slsqp_smooth_obj(x, dx, beta):
    corrs = compute_corrs(x)
    return logsumexp(beta * corrs) / beta * dx


def slsqp_smooth_grad(x, dx, beta):
    return smooth_max_grad(x, dx, beta)


def slsqp_target_obj(x, dx, beta, theta):
    corrs = compute_corrs(x) * dx
    val = beta * (corrs - theta)
    sp = np.where(val > 0, val + np.log1p(np.exp(-val)), np.log1p(np.exp(val)))
    return sp.sum()


def slsqp_target_grad(x, dx, beta, theta):
    n = len(x)
    corrs = compute_corrs(x) * dx
    val = beta * (corrs - theta)
    sig = np.where(val > 0, 1.0 / (1.0 + np.exp(-val)), np.exp(val) / (1.0 + np.exp(val)))
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if sig[k] > 1e-15:
            grad += sig[k] * grad_corr_k(x, k, n) * dx
    return grad


def run_slsqp_smooth(h0, dx, target_sum, beta, maxiter):
    n = len(h0)
    try:
        res = minimize(slsqp_smooth_obj, h0, jac=slsqp_smooth_grad,
                       args=(dx, beta), method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq',
                                     'fun': lambda x: x.sum() - target_sum,
                                     'jac': lambda x: np.ones(n)}],
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        h_out = project_feasible(res.x, target_sum)
    except Exception:
        h_out = project_feasible(h0, target_sum)
    return h_out, c5_of_h(h_out, dx)

                if actual_c5 < best_c5 - 1e-15:
                    best_c5 = actual_c5
                    best_h = h.copy()
                improved = True

        if not improved:
            break

    return best_h, best_c5


def pair_transfer(h0, dx, target_sum, n_top=30, n_iter=5, time_fn=None):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)

    for iteration in range(n_iter):
        if time_fn is not None and time_fn() < 20:
            break
        corrs = compute_corrs(h)
        grad = smooth_max_grad(h, dx, 400.0)

        dec_idx = np.argsort(grad)[-n_top:]
        inc_idx = np.argsort(grad)[:n_top]

        improved = False
        for i in dec_idx:
            for j in inc_idx:
                if i == j:
                    continue
                eps_hi = min(h[i], 1.0 - h[j])
                if eps_hi < 1e-12:
                    continue

                d = np.zeros(2 * n - 1)
                gv_i = grad_vector_for_i(h, i, n)
                gv_j = grad_vector_for_i(h, j, n)
                d = gv_j - gv_i

                gr = (np.sqrt(5.0) - 1.0) / 2.0
                a_p, b_p = 0.0, eps_hi
                c_pt = b_p - gr * (b_p - a_p)
                d_pt = a_p + gr * (b_p - a_p)
                fc = (corrs + c_pt * d).max()
                fd = (corrs + d_pt * d).max()
                for _ in range(40):
                    if b_p - a_p < 1e-14:
                        break
                    if fc < fd:
                        b_p = d_pt
                        d_pt = c_pt
                        fd = fc
                        c_pt = b_p - gr * (b_p - a_p)
                        fc = (corrs + c_pt * d).max()
                    else:
                        a_p = c_pt
                        c_pt = d_pt
                        fc = fd
                        d_pt = a_p + gr * (b_p - a_p)
                        fd = (corrs + d_pt * d).max()

                eps_opt = (a_p + b_p) / 2.0
                if eps_opt > 1e-10:
                    new_h = h.copy()
                    new_h[i] -= eps_opt
                    new_h[j] += eps_opt
                    new_h = np.clip(new_h, 0.0, 1.0)
                    new_h = project_feasible(new_h, target_sum)
                    new_c5 = c5_of_h(new_h, dx)
                    if new_c5 < best_c5 - 1e-14:
                        h = new_h
                        best_c5 = new_c5
                        best_h = h.copy()
                        corrs = compute_corrs(h)
                        improved = True

        if not improved:
            break

    return best_h, best_c5


def multi_lane_optimize(h0, dx, target_sum, time_fn):
    n = len(h0)
    best_h = h0.copy()
    best_c5 = c5_of_h(h0, dx)

    for beta in [100, 300, 800, 2000, 5000, 10000]:
        if time_fn() < 30:
            break
        h_opt, c5_val = run_slsqp_smooth(best_h, dx, target_sum, beta, 1500)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_opt

    if time_fn() > 40:
        h_lp, c5_lp = lp_polish_full(best_h, dx, target_sum, n_steps=150, lam=0.08, time_fn=time_fn)
        if c5_lp < best_c5:
            best_c5 = c5_lp
            best_h = h_lp

    if time_fn() > 40:
        h_cd, c5_cd = coordinate_descent(best_h, dx, target_sum, n_sweeps=20, time_fn=time_fn)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd

    if time_fn() > 30:
        h_lp2, c5_lp2 = lp_polish_active(best_h, dx, target_sum, n_steps=100, lam=0.06, n_active=80, time_fn=time_fn)
        if c5_lp2 < best_c5:
            best_c5 = c5_lp2
            best_h = h_lp2

    if time_fn() > 25:
        h_pt, c5_pt = pair_transfer(best_h, dx, target_sum, n_top=25, n_iter=4, time_fn=time_fn)
        if c5_pt < best_c5:
            best_c5 = c5_pt
            best_h = h_pt

    return best_h, best_c5

    t0 = time.time()

        return budget_s - (time.time() - t0)

    best_h = None
    best_c5 = float('inf')

    h0 = np.array(initial_h_values, dtype=float)
    n = len(h0)
    dx = 2.0 / n
    target_sum = n / 2.0
    h0 = project_feasible(h0, target_sum)
    best_h = h0.copy()
    best_c5 = c5_of_h(h0, dx)

    # Phase 1: Main multi-lane optimization
    h_opt, c5_val = multi_lane_optimize(best_h, dx, target_sum, time_left)
    if c5_val < best_c5:
        best_c5 = c5_val
        best_h = h_opt

    # Phase 2: Target penalty SLSQP
    if time_left() > 40:
        theta = best_c5 * (1.0 - 0.02)
        for beta_tp in [50, 150, 400]:
            if time_left() < 15:
                break
            h_tp, c5_tp = run_slsqp_target(best_h, dx, target_sum, beta_tp, theta, 800)
            if c5_tp < best_c5:
                best_c5 = c5_tp
                best_h = h_tp

    # Phase 3: Intensive LP + CD cycle
    for cycle in range(4):
        if time_left() < 30:

            best_h = h_cd
        if time_left() < 15:

        h_sls, c5_sls = run_slsqp_smooth(best_h, dx, target_sum, 8000, 1000)
        if c5_sls < best_c5:

            best_h = h_sls
