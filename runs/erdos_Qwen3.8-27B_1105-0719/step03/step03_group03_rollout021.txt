def repair_feasible(vec):

    P1 = np.correlate(v, 1.0 - h, mode="full") - np.correlate(h, v, mode="full")

def gen_direction_structured(n, mode, rng):
    """Generate structured directions."""
    v = np.zeros(n)
    if mode == 0:
        # Sinusoidal
        freq = rng.uniform(0.5, 5.0)
        phase = rng.uniform(0, 2 * np.pi)
        v = np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
    elif mode == 1:
        # Block pattern
        block = rng.randint(n // 4, n // 2)
        start = rng.randint(0, n - block)
        v[start:start+block] = 1.0
        v[:block] -= 1.0  # to make zero-sum approximately
    elif mode == 2:
        # Ramp
        v = np.linspace(-1, 1, n)
        if rng.rand() > 0.5:
            v = -v
    elif mode == 3:
        # Alternating blocks
        block = max(2, n // 8)
        for i in range(0, n, block):
            v[i:i+block] = 1.0 if (i // block) % 2 == 0 else -1.0
    elif mode == 4:
        # Random permutation of zeros and ones (balanced)
        half = n // 2
        perm = rng.permutation(n)
        v[perm[:half]] = 1.0
        v[perm[half:]] = -1.0
    else:
        v = rng.randn(n)
    
    v -= v.mean()
    nv = np.linalg.norm(v)
    if nv < 1e-10:
        return None
    return v / nv

                if score is not None and score < c_best - 1e-12:

def interp_to_n(h, new_n):
    """Interpolate h to a new resolution."""
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    fx = np.concatenate([[0.0], xs, [2.0]])
    fh = np.concatenate([[h[0]], h, [h[-1]]])
    hn = np.interp(xt, fx, fh)
    return repair_feasible(np.clip(hn, 0, 1))


def slsqp_refine(h0, maxiter=300, tol=1e-12):

        return x[-1]

        g[-1] = 1.0
        return g
    
    def eq_fun(x):
        return x[:-1].sum() - n / 2.0
    
    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g

        return x[-1] - np.correlate(h, 1 - h, mode="full") * dx

        # Build Jacobian
        J = np.zeros((M, n + 1))
        center = n - 1

    n = len(h)

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx])
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-10:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-10:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-14:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h


def feasible_interval(h, v):
    n = len(h)
    alpha_min = 0.0
    alpha_max = 1.0
    for i in range(n):
        if v[i] > 1e-14:
            alpha_min = max(alpha_min, -h[i] / v[i])
            alpha_max = min(alpha_max, (1.0 - h[i]) / v[i])
        elif v[i] < -1e-14:
            alpha_max = min(alpha_max, -h[i] / v[i])
            alpha_min = max(alpha_min, (1.0 - h[i]) / v[i])
    if alpha_min >= alpha_max - 1e-10:
        return None
    return alpha_min, alpha_max


def exact_line_search(h, v, n_alpha=250):
    n = len(h)
    fi = feasible_interval(h, v)
    if fi is None:
        return None
    a_min, a_max = fi

        return None
    P0 = np.correlate(h, 1.0 - h, mode="full")
    P1 = np.correlate(v, 1.0 - h, mode="full") - np.correlate(h, v, mode="full")
    P2 = np.correlate(v, v, mode="full")
    best_alpha = a_min
    best_val = np.inf
    alphas = np.linspace(a_min, a_max, n_alpha)
    for a in alphas:
        c = P0 + a * P1 - a * a * P2
        m = float(np.max(c))
        if m < best_val:
            best_val = m
            best_alpha = a
    if 1e-8 < best_alpha < a_max - 1e-8:
        span = max((a_max - a_min) / n_alpha * 4, 1e-5)
        a_lo = max(a_min, best_alpha - span)
        a_hi = min(a_max, best_alpha + span)
        alphas2 = np.linspace(a_lo, a_hi, 80)

            c = P0 + a * P1 - a * a * P2
            m = float(np.max(c))
            if m < best_val:
                best_val = m
                best_alpha = a
    h_new = h + best_alpha * v
    h_new = repair_feasible(h_new)

    v = rng.randn(n)

    nv = np.linalg.norm(v)

def gen_dir_gradient(h, profile, temperature=0.008):
    n = len(h)
    c_max = np.max(profile)
    active = np.where(profile >= c_max - temperature * 3)[0]
    if len(active) == 0:
        active = np.array([np.argmax(profile)])
    weights = np.exp((profile[active] - c_max) / temperature)
    weights /= weights.sum()
    center = n - 1
    v = np.zeros(n)
    for idx_k, k in enumerate(active):
        w_k = weights[idx_k]
        d = k - center
        for i in range(max(0, -d), min(n, n - d)):
            v[i] -= w_k * (1.0 - h[i + d])
            v[i + d] += w_k * h[i]
    v -= v.mean()
    nv = np.linalg.norm(v)
    if nv < 1e-10:
        return None
    return v / nv

def interp_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    fx = np.concatenate([[0.0], xs, [2.0]])
    fh = np.concatenate([[h[0]], h, [h[-1]]])
    hn = np.interp(xt, fx, fh)
    return repair_feasible(np.clip(hn, 0, 1))


def slsqp_refine(h0, maxiter=400, tol=1e-12):
    n = len(h0)
    dx = 2.0 / n
    M = 2 * n - 1
    center = n - 1

    def build_ineq_jac(h):
        J = np.zeros((M, n + 1))
        for k in range(M):
            d = k - center
            i_lo = max(0, -d)
            i_hi = min(n, n - d)
            for i in range(i_lo, i_hi):
                J[k, i] -= dx * (1.0 - h[i + d])
                J[k, i + d] += dx * h[i]
        J[:, -1] = 1.0
        return J

    def obj(x):
        return x[-1]

    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g

    def eq_fun(x):
        return x[:-1].sum() - n / 2.0

    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g

    def ineq_fun(x):
        h = x[:-1]
        return x[-1] - np.correlate(h, 1 - h, mode="full") * dx

    def ineq_jac(x):
        return build_ineq_jac(x[:-1])

    t0 = exact_score(h0) + 1e-3
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, t0 + 2.0)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': tol})
        if res.x is not None and np.isfinite(res.x).all():
            hs = repair_feasible(res.x[:-1])
            if np.isfinite(exact_score(hs)):
                return hs
    except Exception:
        pass
    return None


def lp_step(h_curr, profile, dx, trust, budget_fn):
    if not HAS_CVXPY or budget_fn() < 15:
        return None
    n = len(h_curr)
    c_max = np.max(profile)
    beam = max(0.003, 0.12 * (c_max - np.median(profile)))
    active = np.where(profile >= c_max - beam)[0]
    if len(active) == 0:
        active = np.array([np.argmax(profile)])
    if len(active) > 15:
        active = active[np.argsort(profile[active])[-15:]]
    target = n / 2.0
    center = n - 1
    G = np.zeros((len(active), n))
    for row, k in enumerate(active):
        d = k - center
        for i in range(max(0, -d), min(n, n - d)):
            G[row, i] += dx * (1.0 - h_curr[i + d])
            G[row, i + d] += -dx * h_curr[i]
    b = profile[active] - G @ h_curr
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    cons = [
        h_var >= 0, h_var <= 1,
        cp.sum(h_var) == target,
        h_var >= h_curr - trust,
        h_var <= h_curr + trust,
        G @ h_var - t_var <= -b
    ]
    obj = cp.Minimize(t_var + 0.005 * cp.norm1(h_var - h_curr))
    prob = cp.Problem(obj, cons)
    for solver in [cp.ECOS, cp.SCS, cp.CBC]:
        try:
            prob.solve(solver=solver, max_iters=10000, abstol=1e-10)
            break
        except Exception:
            continue
    if h_var.value is None:
        return None
    h_sol = np.array(h_var.value).flatten()
    if not np.isfinite(h_sol).all():
        return None
    return repair_feasible(h_sol)

    # Phase 4: SLSQP
    if tl() > 80:
        hs = slsqp_refine(best_h, maxiter=500, tol=1e-13)
        if hs is not None:
            cs = exact_score(hs)
            if cs < best_c - 1e-14:
                best_h, best_c = hs, cs

    # Phase 5: Line search after SLSQP
    if tl() > 60:
        h1, c1 = line_search_loop(best_h, max_iters=400, budget_fn=lambda: tl() - 15, rng=rng, n_alpha=300)
        if c1 < best_c - 1e-14:
            best_h, best_c = h1, c1

    # Phase 6: More dimension tries
    if tl() > 70:
        n0 = len(best_h)
        dims = sorted(set([d for d in range(max(44, n0 - 12), min(120, n0 + 13), 2) if d != n0]))
        for nd in dims[:8]:
            if tl() < 25:
                break
            hi = interp_to_n(best_h, nd)
            ci = exact_score(hi)
            if ci < best_c + 0.01:
                h_new, c_new = line_search_loop(hi, max_iters=100, budget_fn=lambda: tl() - 10, rng=rng, n_alpha=200)
                if c_new < best_c - 1e-14:
                    best_h, best_c = h_new, c_new

    # Phase 7: Final SLSQP + line search
    if tl() > 50:
        hs = slsqp_refine(best_h, maxiter=600, tol=1e-14)
        if hs is not None:
            cs = exact_score(hs)
            if cs < best_c - 1e-14:
                best_h, best_c = hs, cs
    if tl() > 20:
        h1, c1 = line_search_loop(best_h, max_iters=200, budget_fn=lambda: tl() - 5, rng=rng, n_alpha=300)
        if c1 < best_c - 1e-14:
            best_h, best_c = h1, c1

    # Final validation
    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)

    try:
        er = evaluate_erdos_solution(best_h)
        if isinstance(er, (float, int)):
            final_c = min(final_c, float(er))
        elif isinstance(er, tuple) and len(er) >= 2:
            final_c = min(final_c, float(er[0]))
    except Exception:
        pass

    return best_h, final_c, len(best_h)
