def lp_refine(h, dx, target_sum, n, n_iter, t0, budget_s):
    """LP-based direction finding + line search."""
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    cur = h.copy()
    cur_c5 = best_c5
    alpha = 0.3
    
    for it in range(n_iter):
        if budget_s - (time.time() - t0) < 10:
            break
        
        corrs = np.correlate(cur, 1.0 - cur, mode='full')
        max_c = corrs.max()
        top_n = min(40, 2 * n - 1)
        top_idx = np.argsort(corrs)[-top_n:]
        
        K = len(top_idx)
        m_vars = n + 1
        grads = np.zeros((K, n))
        for jj, kk in enumerate(top_idx):
            grads[jj] = grad_corr_k(cur, kk, n)
        
        c_obj = np.zeros(m_vars)
        c_obj[-1] = 1.0
        A_ub = np.zeros((K, m_vars))
        for jj in range(K):
            A_ub[jj, :n] = dx * grads[jj]
            A_ub[jj, -1] = -1.0
        b_ub = -dx * corrs[top_idx]
        A_eq = np.ones((1, n))
        b_eq = np.zeros(1)
        bounds = [(-cur[i] * alpha, (1.0 - cur[i]) * alpha) for i in range(n)]
        bounds.append((0.0, None))
        
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs', options={'presolve': True})
            if res.status != 0:
                alpha *= 0.5
                if alpha < 0.01:
                    break
                continue
            delta = res.x[:n]
        except Exception:
            alpha *= 0.5
            if alpha < 0.01:
                break
            continue
        
        improved = False
        for t in [1.0, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05]:
            h_new = cur + t * delta
            h_new = project_feasible(h_new, target_sum)
            new_c5 = c5_of_h(h_new, dx)
            if new_c5 < cur_c5 - 1e-14:
                cur = h_new
                cur_c5 = new_c5
                improved = True
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h_new.copy()
                break
        
        if not improved:
            alpha *= 0.6
            if alpha < 0.01:
                break
        else:
            alpha = min(0.5, alpha * 1.1)
    
    return best_h, best_c5

def transfer_grid(h_src, n_target):
    n_src = len(h_src)
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_tgt = (np.arange(n_target) + 0.5) * (2.0 / n_target)
    return np.interp(x_tgt, x_src, h_src)

    # Phase 1: Pair optimization sweeps (main workhorse)
    pairs_i, pairs_j = np.triu_indices(n, k=1)
    n_pairs = len(pairs_i)
    
    while budget_s - (time.time() - t0) > 250:
        order = rng.permutation(n_pairs)
        sweep_improved = False
