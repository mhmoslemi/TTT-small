import time
import cvxpy as cp

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full"))) * dx

def profile_h(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx])
            h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_grad_matrix(h):
    """G[k, m] = ∂c_k/∂h[m] where c_k is correlation at index k."""
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    G = np.zeros((M, n))
    
    for lag_idx in range(M):
        d = lag_idx - (n - 1)
        if d == 0:
            G[lag_idx, :] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            # c_d = dx * sum_{j=d}^{n-1} h[j]*(1-h[j-d])
            for m in range(d, n):
                G[lag_idx, m] += dx * (1.0 - h[m - d])
            for m in range(0, n - d):
                G[lag_idx, m] -= dx * h[m + d]
        else:
            dp = -d
            for m in range(0, n - dp):
                G[lag_idx, m] += dx * (1.0 - h[m + dp])
            for m in range(dp, n):
                G[lag_idx, m] -= dx * h[m - dp]
    
    return G

def slp_iteration(h0, max_z=None):
    """One SLP iteration. Returns (h_lp, z_lp) or (None, None)."""
    n = len(h0)
    c_vals = profile_h(h0)  # unscaled? No, profile_h already includes dx
    G = compute_grad_matrix(h0)
    
    h_var = cp.Variable(n)
    z_var = cp.Variable()
    
    # Linearized constraint: c_k(h0) + G[k,:]@(h-h0) <= z
    # => G[k,:]@h - z <= z_bound_k where z_bound_k = -c_k(h0) + G[k,:]@h0
    b = -c_vals + G @ h0
    
    constraints = [
        G @ h_var - z_var <= b,
        h_var >= 0,
        h_var <= 1,
        cp.sum(h_var) == n / 2.0
    ]

    prob = cp.Problem(cp.Minimize(z_var), constraints)

        except Exception:
            try:
                prob.solve(solver=solver)
            except Exception:
                continue
    
    return None, None

def slp_line_search(h0, h_lp, c0, max_alphas=10):
    """Line search from h0 toward h_lp. Returns (h_best, c_best)."""
    d = h_lp - h0
    best_h = h0.copy()
    best_c = c0
    
    alphas = [1.0]
    for i in range(max_alphas - 1):
        alphas.append(alphas[-1] * 0.5)
    
    for a in alphas:
        h_try = repair_feasible(h0 + a * d)
        c_try = exact_score(h_try)
        if c_try < best_c:
            best_h = h_try
            best_c = c_try
    
    return best_h, best_c

def run_slp(h0, max_iters=50, t_start=None, budget_s=1100, margin=30):
    """Run SLP until convergence or budget exhausted."""
    h = h0.copy()
    c = exact_score(h)
    max_z = c  # initial upper bound on z
    
    for it in range(max_iters):
        if t_start is not None and time.perf_counter() - t_start > budget_s - margin:
            break
        
        h_lp, z_lp = slp_iteration(h, max_z=max_z)
        if h_lp is None:

            c = c_new
            max_z = c  # tighten the z bound

            # Check if h_lp ≈ h (converged)

                else:

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    return repair_feasible(np.clip(h_new, 0.0, 1.0))
