def smooth_power_grad(h, q, corr=None):
    n = len(h)
    if corr is None:
        corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    if max_c < 1e-30:
        return np.zeros(n)
    a = corr / max_c  # in [0, 1] since corr >= 0
    if q == 1:
        w = a.copy()
    else:
        w = np.power(a, q-1)
        w[a < 1e-30] = 0.0
    
    g = np.zeros(n)
    # Lag 0
    g += w[n-1] * (1.0 - 2.0 * h)
    
    for d in range(1, n):
        K = n - d  # valid range for "i+d < n" is i in [0, K-1]
        wp = w[n-1-d]
        wm = w[n-1+d]
        wsum = wp + wm
        wdiff = wp - wm
        
        # Middle region: i in [d, K-1] (both indicators active)
        # Need d <= K-1, i.e., d <= n-d-1, i.e., 2d <= n-1
        if d < K:  # d <= K-1
            lo = d
            hi = K  # exclusive
            g[lo:hi] += wsum * (1.0 - h[lo+d:hi+d] - h[lo-d:hi-d])
        
        # Left region: i in [0, min(d, K)-1] (only first-indicator from both)
        left_hi = min(d, K)
        if left_hi > 0:
            g[:left_hi] += wp * (1.0 - h[d:left_hi+d]) - wm * h[d:left_hi+d]
        
        # Right region: i in [K, n-1] (only second-indicator from both)
        if K < n:
            g[K:] += wm * (1.0 - h[K-d:n-d]) - wp * h[K-d:n-d]
    
    g -= g.mean()
    return g

def project_feasible(x, target_sum):
    n = len(x)
    x = np.where(np.isfinite(x), x, 0.5)
    lo = x.min() - 2.0
    hi = x.max() + 2.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            adj = diff / mask.sum()
            adj = max(adj, -h[mask].min() + 1e-13)
            adj = min(adj, (1.0 - h[mask]).max() - 1e-13)
            h[mask] += adj
    return np.clip(h, 0.0, 1.0)

def c5_of_h(h, dx):

        w[a < 1e-20] = 0.0

            if c5_cur < c5_best:
                c5_best = c5_cur
                h_best = h.copy()
            beta = min(0.3, beta * 1.05)
        else:
            stall += 1
            beta *= 0.7
            if beta < 0.005 or stall > 40:
                break
    return h_best, c5_best

def compute_pair_slopes(h, p, q, n, corr):
    """Compute slope s_k for each lag k when h[p]->h[p]+t, h[q]->h[q]-t."""
    s = np.zeros(2 * n - 1)
    for d in range(0, n):
        K = n - d
        if d == 0:
            s[n - 1] += (1.0 - 2.0 * h[p]) - (1.0 - 2.0 * h[q])
        else:
            # corr[n-1-d] = sum_{j=0}^{K-1} h[j]*(1-h[j+d])
            # d/dh[i] = (1-h[i+d])*1{i<K} - h[i-d]*1{i>=d}
            sp_p = 0.0
            sp_q = 0.0
            if p < K:
                sp_p += (1.0 - h[p + d])
            if p >= d:
                sp_p -= h[p - d]
            if q < K:
                sp_q += (1.0 - h[q + d])
            if q >= d:
                sp_q -= h[q - d]
            s[n - 1 - d] += sp_p - sp_q
            # corr[n-1+d] = sum_{j=0}^{K-1} h[j+d]*(1-h[j])
            # d/dh[i] = (1-h[i-d])*1{i>=d} - h[i+d]*1{i<K}
            sm_p = 0.0
            sm_q = 0.0
            if p >= d:
                sm_p += (1.0 - h[p - d])
            if p < K:
                sm_p -= h[p + d]
            if q >= d:
                sm_q += (1.0 - h[q - d])
            if q < K:
                sm_q -= h[q + d]
            s[n - 1 + d] += sm_p - sm_q
    return s

def pair_descent(h_in, dx, target, n_sweeps, budget_s, t0, rng, min_time=15.0):
    n = len(h_in)
    h = h_in.copy()
    c5_cur = c5_of_h(h, dx)
    c5_best = c5_cur
    h_best = h.copy()
    eps = 1e-8
    n_evals=25
    for sweep in range(n_sweeps):
        if budget_s - (time.time() - t0) < min_time:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.where((h > eps) & (h < 1.0 - eps))[0]
        if len(idx) < 2:
            break
        perm = rng.permutation(len(idx))
        improved_total = 0
        for ii in range(len(idx)):
            if budget_s - (time.time() - t0) < min_time:
                break
            p = idx[perm[ii]]
            if h[p] <= eps or h[p] >= 1.0 - eps:
                continue
            for jj in range(ii + 1, min(ii + 20, len(idx))):
                if budget_s - (time.time() - t0) < min_time:
                    break
                q = idx[perm[jj]]
                if h[q] <= eps or h[q] >= 1.0 - eps:
                    continue
                L = max(-h[p], h[q] - 1.0)
                U = min(1.0 - h[p], h[q])
                if U - L < 1e-12:
                    continue
                s = compute_pair_slopes(h, p, q, n, corr)
                # f(t) = max(corr + s * t), minimize over [L, U]
                ts = np.linspace(L, U, n_evals)
                vals = np.max(corr[None, :] + s[None, :] * ts[:, None], axis=1)
                best_i = np.argmin(vals)
                best_t = ts[best_i]
                best_val = vals[best_i]
                # Refine around best
                if 0 < best_i < n_evals - 1:
                    tl = ts[best_i - 1]
                    tr = ts[best_i + 1]
                    for _ in range(8):
                        tm1 = tl + (tr - tl) / 3.0
                        tm2 = tr - (tr - tl) / 3.0
                        v1 = np.max(corr + s * tm1)
                        v2 = np.max(corr + s * tm2)
                        if v1 < v2:
                            tr = tm2
                        else:
                            tl = tm1
                    best_t = (tl + tr) * 0.5
                    best_val = np.max(corr + s * best_t)
                # Check boundaries too
                vL = np.max(corr + s * L)
                vU = np.max(corr + s * U)
                if vL < best_val:
                    best_t, best_val = L, vL
                if vU < best_val:
                    best_t, best_val = U, vU
                if best_val * dx < c5_cur - 1e-14:
                    h[p] += best_t
                    h[q] -= best_t
                    h = project_feasible(h, target)
                    c5_new = c5_of_h(h, dx)
                    if c5_new < c5_cur - 1e-15:
                        c5_cur = c5_new
                        improved_total += 1
                        corr = np.correlate(h, 1.0 - h, mode='full')
                        if c5_new < c5_best:
                            c5_best = c5_new
                            h_best = h.copy()
                else:
                    # Recompute corr since pair move changed things
                    pass
        if improved_total == 0:
            break
    return h_best, c5_best

def transfer_grid(h_src, n_target):
    n_src = len(h_src)
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_tgt = (np.arange(n_target) + 0.5) * (2.0 / n_target)
    h_t = np.interp(x_tgt, x_src, h_src)
    return h_t

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def tl():
        return budget_s - (time.time() - t0)

    # Load parent
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = np.full(160, 0.5)

    n = len(h0)
    dx = 2.0 / n
    target = n / 2.0
    h0 = project_feasible(h0, target)
    best_h = h0.copy()
    best_c5 = c5_of_h(best_h, dx)

    # Phase 1: Smooth gradient with q=8
    if tl() > 200:
        h1, c5_1 = smooth_phase(best_h, dx, target, 8, 200, budget_s, t0, min_time=20)
        if c5_1 < best_c5:
            best_c5 = c5_1
            best_h = h1

    # Phase 2: Pair descent
    if tl() > 250:
        h2, c5_2 = pair_descent(best_h, dx, target, 4, budget_s, t0, rng, min_time=20)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2

    # Phase 3: Smooth gradient with q=16
    if tl() > 180:
        h3, c5_3 = smooth_phase(best_h, dx, target, 16, 150, budget_s, t0, min_time=15)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3

    # Phase 4: Pair descent again
    if tl() > 200:
        h4, c5_4 = pair_descent(best_h, dx, target, 4, budget_s, t0, rng, min_time=20)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4

    # Phase 5: Smooth with q=32 (very peaked, focuses on top correlations)
    if tl() > 100:
        h5, c5_5 = smooth_phase(best_h, dx, target, 32, 100, budget_s, t0, min_time=10)
        if c5_5 < best_c5:
            best_c5 = c5_5
            best_h = h5

    # Phase 6: Try nearby grid sizes
    n_base = len(best_h)
    n_candidates = sorted(set([n_base + dn for dn in [-8, -4, -2, 2, 4, 8]
                                if 80 <= n_base + dn <= 240 and (n_base + dn) % 2 == 0]))
    for n_try in n_candidates:
        if tl() < 60:
            break
        dx_t = 2.0 / n_try
        ts_t = n_try / 2.0
        h_t = transfer_grid(best_h, n_try)
        h_t = project_feasible(h_t, ts_t)
        # Quick smooth + pair
        h_t, c5_t = smooth_phase(h_t, dx_t, ts_t, 8, 80, budget_s, t0, min_time=15)
        if c5_t < best_c5 - 1e-8:
            h_t, c5_t = pair_descent(h_t, dx_t, ts_t, 2, budget_s, t0, rng, min_time=15)
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h_t
                n_base = n_try

    # Phase 7: Final pair descent on best
    n_b = len(best_h)
    dx_b = 2.0 / n_b
    ts_b = n_b / 2.0
    if tl() > 80:
        h7, c5_7 = pair_descent(best_h, dx_b, ts_b, 5, budget_s, t0, rng, min_time=10)
        if c5_7 < best_c5:
            best_c5 = c5_7
            best_h = h7

    # Phase 8: Final fine smooth
    if tl() > 40:
        h8, c5_8 = smooth_phase(best_h, 2.0/len(best_h), len(best_h)/2.0, 64, 80, budget_s, t0, min_time=8)
        if c5_8 < best_c5:
            best_c5 = c5_8
            best_h = h8

    # Final validation
    n_final = len(best_h)
    dx_final = 2.0 / n_final
    ts_final = n_final / 2.0
    best_h = project_feasible(best_h, ts_final)
    best_c5 = c5_of_h(best_h, dx_final)

    return best_h, best_c5, n_final
