import numpy as np
from scipy.optimize import minimize

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(-(1.0 - h))
        remaining = diff
        for i in range(n):
            idx = order[i]
            cap = 1.0 - h[idx]
            if cap <= 1e-15:
                continue
            add = min(remaining, cap)
            h[idx] += add
            remaining -= add
            if remaining <= 1e-14:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            if h[idx] <= 1e-15:
                continue
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-14:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] = float(np.clip(h[idx] + diff2, 0.0, 1.0))
    return h

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def grad_row(h, m, n):
    dx = 2.0 / n
    d = m - (n - 1)
    if d == 0:
        return dx * (1.0 - 2.0 * h)
    elif d > 0:
        g = np.zeros(n)
        g[:n - d] += dx * (1.0 - h[d:])
        g[d:] -= dx * h[:n - d]
        return g
    else:
        p = -d
        g = np.zeros(n)
        g[:n - p] -= dx * h[p:]
        g[p:] += dx * (1.0 - h[:n - p])
        return g

def interp_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    edges_old = np.linspace(0.0, 2.0, n + 1)
    edges_new = np.linspace(0.0, 2.0, new_n + 1)
    h_new = np.zeros(new_n)
    for k in range(new_n):
        a, b = edges_new[k], edges_new[k + 1]
        total = 0.0
        for j in range(n):
            lo = max(a, edges_old[j])
            hi = min(b, edges_old[j + 1])
            if hi > lo:
                total += h[j] * (hi - lo)
        h_new[k] = total / (b - a)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

        c = float(profile.max())

        # Weight by how close to max

            active = active[top_idx]
            weights = weights[top_idx]

            w_norm = 1.0
        g = np.zeros(n)
        for m, w in zip(active, weights):
            g += (w / w_norm) * grad_row(h, m, n)
        g -= g.mean()
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-16:

def flatten_profile(h, n_iters=200):
    """Try to flatten the top of the profile to reduce the max."""
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    for it in range(n_iters):

        c = float(profile.max())
        thresh = c - 0.005
        active = np.where(profile >= thresh)[0]
        if len(active) < 2:
            break

        # Direction: reduce the spread
        g = np.zeros(n)
        prof_active = profile[active]
        mean_p = prof_active.mean()
        # Gradient of (max - mean_active)^2 approximately
        for m in active:

        g -= g.mean()
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break

        pos_mask = d_dir > 1e-12
        neg_mask = d_dir < -1e-12

        alpha_max = max(alpha_max, 1e-12)
        # Line search
        best_a = 0.0
        best_v = exact_score(repair_feasible(h))

def slsqp_epigraph(h_start, maxiter=100):
    n = len(h_start)
    dx = 2.0 / n
    M = 2 * n - 1
    c0 = exact_score(h_start)
    t0 = c0 + 1e-5

    def obj(x):
        return x[-1]

    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g

    def eq_fun(x):
        return x[:-1].sum() - n / 2.0

    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g

    def ineq_fun(x):
        h = x[:-1]
        t = x[-1]
        profile = np.correlate(h, 1 - h, mode="full") * dx
        return t - profile

    def ineq_jac(x):
        h = x[:-1]
        J = np.zeros((M, n + 1))
        for m in range(M):
            J[m, :n] = -grad_row(h, m, n)
        J[:, -1] = 1.0
        return J

    x0 = np.concatenate([h_start, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 1.0)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-13})
        if res.x is not None and np.isfinite(res.x).all():
            h_new = repair_feasible(res.x[:-1])
            c_new = exact_score(h_new)
            if c_new <= c0 + 1e-10:
                return h_new
    except Exception:
        pass
    return None

    if time_left() > 30:
        h_r, c_r = active_descent_h(best_h, max_iters=200, tol=0.0002, n_active_max=6)
        if c_r < best_c:
            best_h, best_c = h_r, c_r

    # Stage 7: Coordinate polish
    if time_left() > 25:
        h_cp, c_cp = coordinate_polish(best_h, max_iters=100)
        if c_cp < best_c:
            best_h, best_c = h_cp, c_r

    # Stage 8: Final SLSQP + polish
    if time_left() > 20:
        h_sol = slsqp_epigraph(best_h, maxiter=150)
        if h_sol is not None:
            c_sol = exact_score(h_sol)
            if c_sol < best_c:
                best_h, best_c = h_sol, c_sol

    if time_left() > 15:
        h_r, c_r = active_descent_h(best_h, max_iters=150, tol=0.0001, n_active_max=5)
        if c_r < best_c:
            best_h, best_c = h_r, c_r

    # Stage 9: Ultra-fine final polish
    if time_left() > 10:
        h_r, c_r = active_descent_h(best_h, max_iters=300, tol=0.00005, n_active_max=4)
        if c_r < best_c:
            best_h, best_c = h_r, c_r

    if time_left() > 5:
        h_cp, c_cp = coordinate_polish(best_h, max_iters=50)
        if c_cp < best_c:
            best_h, best_c = h_cp, c_cp

    # Final: try 1-h one more time
    if time_left() > 10:
        h_sym = 1.0 - best_h
        h_sym = repair_feasible(h_sym)
        h_r, c_r = active_descent_h(h_sym, max_iters=100, tol=0.0001, n_active_max=4)
        if c_r < best_c:
            best_h, best_c = h_r, c_r

    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)

    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            final_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 2:
            final_c = float(eval_result[0])
    except Exception:
        pass

    return best_h, final_c, len(best_h)
