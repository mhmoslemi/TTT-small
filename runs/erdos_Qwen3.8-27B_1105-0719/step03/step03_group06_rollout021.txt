    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def grad_row(h, m, n):
    """Gradient of P[m] with respect to h."""
    dx = 2.0 / n
    d = m - (n - 1)
    g = np.zeros(n)
    if d == 0:
        g = dx * (1.0 - 2.0 * h)
    elif d > 0:
        # P[m] = dx * sum_i h[i]*(1-h[i+d]) for i=0..n-1-d
        # dP/dh[j] = dx*(1-h[j+d]) for j=0..n-1-d
        #           - dx*h[j-d] for j=d..n-1
        g[:n-d] += dx * (1.0 - h[d:])
        g[d:] -= dx * h[:n-d]
    else:
        p = -d
        # P[m] = dx * sum_i h[i]*(1-h[i-p]) for i=p..n-1

        g[p:] += dx * (1.0 - h[:n-p])
        g[:n-p] -= dx * h[p:]
    return g

    n = len(h)
    profile = compute_profile(h)
    M = len(profile)
    # Stable softmax
    shifted = beta * (profile - profile.max())
    weights = np.exp(shifted)
    wsum = weights.sum()
    if wsum < 1e-300:

    weights /= wsum

        if weights[m] > 1e-12:
            g += weights[m] * grad_row(h, m, n)
    # Project gradient onto tangent space of sum constraint
    g -= g.sum() / n
    return g

    best_c = exact_score(h)
    t_start = time.perf_counter()
    
    betas = np.geomspace(beta_start, beta_end, beta_steps)
    
    for beta in betas:
        iters_this_beta = max(1, max_iters // beta_steps)
        prev_c = exact_score(h)
        for it in range(iters_this_beta):
            if time.perf_counter() - t_start > time_limit:

            g = lse_gradient(h, beta)

                    if c_try < best_c:

                        best_h = h.copy()
                    improved = True
                    break
                alpha *= 0.5
