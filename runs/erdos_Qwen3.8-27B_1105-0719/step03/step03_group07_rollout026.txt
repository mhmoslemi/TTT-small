def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0

            add = min(remaining, capacity[idx])
            h[idx] += add
            remaining -= add

    #      sum(d) = 0
    #      -h_i <= d_i <= 1 - h_i

    A_eq[0, :n] = 1.0

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full")) * dx)

def profile_h(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx])
            h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def build_gradient_matrix(h, lags, n, dx):
    M = len(lags)
    G = np.zeros((M, n))
    for idx, m in enumerate(lags):
        d = m - (n - 1)
        if d == 0:
            G[idx, :] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            for i in range(n - d):
                G[idx, i + d] += dx * (1.0 - h[i])
                G[idx, i] -= dx * h[i + d]
        else:
            dp = -d
            for j in range(n - dp):
                G[idx, j + dp] += dx * (1.0 - h[j])
                G[idx, j] -= dx * h[j + dp]
    return G

def lp_direction(h, active_lags):
    n = len(h)
    dx = 2.0 / n
    G = build_gradient_matrix(h, active_lags, n, dx)
    A_row = G.shape[0]
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_ub = np.hstack([G, -np.ones((A_row, 1))])
    b_ub = np.zeros(A_row)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(-h[i], 1.0 - h[i]) for i in range(n)] + [(-1.0, 1.0)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            return res.x[:n], res.x[-1]
    except Exception:
        pass
    return None, None

def lp_step(h, active_tol=0.002):
    n = len(h)
    P = profile_h(h)
    c_max = P.max()
    thresh = c_max - active_tol
    active = np.where(P >= thresh)[0]
    if len(active) == 0:
        active = np.array([np.argmax(P)])
    if len(active) > 30:
        active = np.argsort(P[active])[-30:]
    d, t_val = lp_direction(h, active)
    if d is None:
        return None, None
    if t_val >= -1e-12:
        return None, None
    best_h = h.copy()
    best_c = exact_score(h)
    alphas = [1.0, 0.9, 0.8, 0.7, 0.6, 0.5, 0.4, 0.3, 0.2, 0.15, 0.1, 0.05, 0.02]

def slsqp_active(h0, maxiter=300, active_k=10, tol=1e-12):

def interpolate_to_n(h, new_n):

def coordinate_descent(h, max_sweeps=20, t_start=None, budget_s=1100):
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    for sweep in range(max_sweeps):
        if t_start is not None and time.perf_counter() - t_start > budget_s - 15:
            break
        improved = False
        for i in range(n):
            if t_start is not None and time.perf_counter() - t_start > budget_s - 15:
                break
            h_save = h[i]
            for direction in [1, -1]:
                for frac in [0.02, 0.01, 0.005, 0.002]:
                    delta = direction * frac
                    h_new_val = h_save + delta
                    if h_new_val < -1e-12 or h_new_val > 1.0 + 1e-12:
                        continue
                    h[i] = h_new_val
                    h = repair_feasible(h)
                    s = exact_score(h)
                    if s < best_c - 1e-14:
                        best_c = s
                        improved = True
                        break
                    else:
                        h[i] = h_save
                        h = repair_feasible(h)
                if improved:
                    break
        if not improved:
            break
    return h, best_c

def pair_transfer_optimize(h, rng, n_pairs=1000, t_start=None, budget_s=1100):
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    for _ in range(n_pairs):
        if time.perf_counter() - t_start > budget_s - 15:
            break
        i, j = rng.randint(0, n, 2)
        if i == j:
            continue
        for delta in [0.015, 0.008, 0.004, 0.002]:
            if h[i] < delta:
                break
            h[i] -= delta
            h[j] += delta
            h = repair_feasible(h)
            c_try = exact_score(h)
            if c_try < best_c:
                best_c = c_try
            else:
                h[i] += delta
                h[j] -= delta
                h = repair_feasible(h)
    return h, best_c

        for i in range(n):
            h[i] = 0.5 + 0.25 * np.sin(3 * np.pi * i / n) + 0.15 * np.cos(5 * np.pi * i / n)
    return repair_feasible(np.clip(h, 0.0, 1.0))

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()
    rng = np.random.RandomState(seed)

    def time_ok(margin=15):
        return time.perf_counter() - t_start < budget_s - margin

    parent = None
    try:
        parent = np.array(initial_h_values, dtype=np.float64)
    except NameError:
        pass
    except Exception:
        pass

    if parent is None or len(parent) < 4:
        parent = np.full(70, 0.5)

    parent = np.clip(parent, 0.0, 1.0)
    parent = repair_feasible(parent)
    best_h = parent.copy()
    best_c = exact_score(parent)

    # Phase 1: LP-based active constraint leveling (primary method)
    tol_schedules = [
        [0.003, 0.005, 0.008, 0.012, 0.002, 0.004],
        [0.002, 0.004, 0.006, 0.001, 0.003],
        [0.005, 0.008, 0.01, 0.015, 0.003, 0.005],
    ]
    for sched in tol_schedules:
        if not time_ok(100):
            break
        h_new, c_new = lp_loop(best_h, max_iters=200, tol_schedule=sched,
                               t_start=t_start, budget_s=budget_s)
        if c_new < best_c - 1e-13:
            best_h = h_new
            best_c = c_new

    # Phase 2: SLSQP polishing with different active sets
    if time_ok(80):
        for ak in [10, 15, 20]:
            if not time_ok(25):
                break
            sol = slsqp_active(best_h, maxiter=400, active_k=ak, tol=1e-12)
            if sol is not None:
                c = exact_score(sol)
                if c < best_c - 1e-13:
                    best_h = sol
                    best_c = c

    # Phase 3: More LP iterations with tighter tolerances
    if time_ok(80):
        h_new, c_new = lp_loop(best_h, max_iters=200,
                               tol_schedule=[0.001, 0.002, 0.003, 0.0015],
                               t_start=t_start, budget_s=budget_s)
        if c_new < best_c - 1e-13:
            best_h = h_new
            best_c = c_new

    # Phase 4: Wide SLSQP
    if time_ok(60):
        sol = slsqp_wide(best_h, maxiter=500, tol=1e-12)
        if sol is not None:
            c = exact_score(sol)
            if c < best_c - 1e-13:
                best_h = sol
                best_c = c

    # Phase 5: Another LP round
    if time_ok(50):
        h_new, c_new = lp_loop(best_h, max_iters=150,
                               tol_schedule=[0.002, 0.003, 0.005],
                               t_start=t_start, budget_s=budget_s)
        if c_new < best_c - 1e-13:
            best_h = h_new
            best_c = c_new

    # Phase 6: Dimension exploration
    if time_ok(120):
        n_best = len(best_h)
        cand_dims = []
        for d in range(max(40, n_best - 12), min(130, n_best + 20)):
            if d % 2 == 0 and d != n_best:
                cand_dims.append(d)
        cand_dims.sort(key=lambda x: abs(x - n_best))
        for new_n in cand_dims[:8]:
            if not time_ok(25):
                break
            h_interp = interpolate_to_n(best_h, new_n)
            h_interp, c_new = lp_loop(h_interp, max_iters=80,
                                      tol_schedule=[0.003, 0.005, 0.008],
                                      t_start=t_start, budget_s=budget_s)
            if c_new < best_c - 1e-10:
                sol = slsqp_active(h_interp, maxiter=200, active_k=12, tol=1e-12)
                if sol is not None:
                    c2 = exact_score(sol)
                    if c2 < c_new:
                        h_interp = sol
                        c_new = c2
                best_h = h_interp
                best_c = c_new

    # Phase 7: Coordinated descent for fine tuning
    if time_ok(40):
        best_h, best_c = coordinate_descent(best_h, max_sweeps=15,
                                            t_start=t_start, budget_s=budget_s)

    # Phase 8: Pair transfers
    if time_ok(30):
        best_h, best_c = pair_transfer_optimize(best_h, rng, n_pairs=2000,
                                                 t_start=t_start, budget_s=budget_s)

    # Phase 9: Final LP + SLSQP
    if time_ok(25):
        h_new, c_new = lp_loop(best_h, max_iters=100,
                               tol_schedule=[0.001, 0.002],
                               t_start=t_start, budget_s=budget_s)
        if c_new < best_c - 1e-14:
            best_h = h_new
            best_c = c_new

    if time_ok(20):
        sol = slsqp_active(best_h, maxiter=500, active_k=12, tol=1e-13)
        if sol is not None:
            c = exact_score(sol)
            if c < best_c - 1e-14:
                best_h = sol
                best_c = c

    # Phase 10: Try alternative starting structures if still above target
    if best_c > 0.3808 and time_ok(60):
        n_cur = len(best_h)
        for mode in [1, 2, 4]:
            if not time_ok(20):
                break
            h_struct = make_structured_start(n_cur, mode)
            # Blend with best
            for mix in [0.3, 0.5, 0.7]:
                if not time_ok(10):
                    break
                h_blend = repair_feasible(mix * h_struct + (1 - mix) * best_h)
                h_new, c_new = lp_loop(h_blend, max_iters=50,
                                       tol_schedule=[0.005, 0.01],
                                       t_start=t_start, budget_s=budget_s)
                if c_new < best_c - 1e-10:
                    sol = slsqp_active(h_new, maxiter=200, active_k=10, tol=1e-12)
                    if sol is not None:
                        c2 = exact_score(sol)
                        if c2 < c_new:
                            h_new = sol
                            c_new = c2
                    best_h = h_new
                    best_c = c_new

    # Final validation
    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)

    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            final_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 1:
            if np.isfinite(eval_result[0]):
                final_c = float(eval_result[0])
    except Exception:
        pass

    return best_h, final_c, len(best_h)
