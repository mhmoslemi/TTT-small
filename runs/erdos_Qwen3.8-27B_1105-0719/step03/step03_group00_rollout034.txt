def compute_gradient(h, weights, dx):
    n = len(h)
    K = 2*n - 1
    grad = np.zeros(n)
    # k=0 corresponds to offset -(n-1)
    # k=n-1 corresponds to offset 0
    # k=2n-2 corresponds to offset n-1
    for k in range(K):
        if weights[k] == 0:
            continue
        d = k - (n-1)  # offset
        if d == 0:
            grad += weights[k] * (1 - 2*h)
        elif d > 0:
            # corr_+(d)
            idx1 = np.arange(n - d)
            grad[idx1] += weights[k] * (1 - h[idx1 + d])
            idx2 = np.arange(d, n)
            grad[idx2] -= weights[k] * h[idx2 - d]
        else:
            p = -d
            # corr_-(p)
            idx1 = np.arange(p, n)
            grad[idx1] += weights[k] * (1 - h[idx1 - p])
            idx2 = np.arange(n - p)
            grad[idx2] -= weights[k] * h[idx2 + p]
    return dx * grad

Let me think about the pairwise optimization more carefully. If I change h[i] by +ε and h[j] by -ε (keeping sum constant), how does each correlation change?

For corr[d] where d > 0:

The change is:
Δcorr[d] = ε * [∂corr[d]/∂h[i] - ∂corr[d]/∂h[j]]

Where:
∂corr[d]/∂h[i] = (1-h[i+d])*1{i∈[0,n-1-d]} - h[i-d]*1{i∈[d,n-1]}

So the total change in the maximum correlation is approximately:
ΔC5 ≈ dx * max_k (corr[k] + Δcorr[k]) - dx * max_k corr[k]

For small ε, this is:
ΔC5 ≈ dx * max_k (∂corr[k]/∂h[i] - ∂corr[k]/∂h[j]) * ε

This is a linear approximation in ε. For the exact change, since corr[k] is bilinear in h, the change is:
Δcorr[k] = ε * (∂_i corr[k] - ∂_j corr[k]) + ε^2 * (terms from the product)

import numpy as np
import time

def project_box_sum(x, target):
    n = len(x)
    x = np.where(np.isfinite(x), x, 0.5)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = x.min() - abs(target) - 2.0
    hi = x.max() + abs(target) + 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target:
            lo = mid
        else:
            hi = mid
    s = (lo + hi) * 0.5
    h = np.clip(x - s, 0.0, 1.0)
    err = target - h.sum()
    if abs(err) > 1e-10:
        if err > 0:
            mask = h < 1.0 - 1e-10
            cnt = mask.sum()
            if cnt > 0:
                h[mask] += err / cnt
        else:
            mask = h > 1e-10
            cnt = mask.sum()
            if cnt > 0:
                h[mask] += err / cnt
    return np.clip(h, 0.0, 1.0)

def c5_eval(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * dx, corr

def softmax_weights(corr, alpha):
    scaled = corr / alpha
    m = scaled.max()
    w = np.exp(scaled - m)
    s = w.sum()
    if s < 1e-300:
        return np.full(len(corr), 1.0 / len(corr))
    return w / s

def grad_weighted(h, weights, dx):
    n = len(h)
    grad = np.zeros(n)
    grad += weights[n-1] * (1.0 - 2.0 * h)
    for d in range(1, n):
        wp = weights[n-1+d]
        wn = weights[n-1-d]
        if wp == 0.0 and wn == 0.0:
            continue
        if wp != 0.0:
            idx = np.arange(n - d)
            grad[idx] += wp * (1.0 - h[idx + d])
            idx2 = np.arange(d, n)
            grad[idx2] -= wp * h[idx2 - d]
        if wn != 0.0:
            idx = np.arange(d, n)
            grad[idx] += wn * (1.0 - h[idx - d])
            idx2 = np.arange(n - d)
            grad[idx2] -= wn * h[idx2 + d]
    return dx * grad

def softmax_descend(h, dx, target, alpha_seq, budget_s, t0, min_time=5.0):
    n = len(h)
    best_h = h.copy()
    best_c5, _ = c5_eval(h, dx)
    cur = h.copy()
    cur_c5 = best_c5
    
    for alpha in alpha_seq:
        if budget_s - (time.time() - t0) < min_time:
            break
        # Number of iterations at this alpha
        n_iter = int(max(50, min(500, 2000.0 / (alpha * 100))))
        lr = 0.05
        
        for it in range(n_iter):
            if budget_s - (time.time() - t0) < min_time:
                break
            _, corr = c5_eval(cur, dx)
            w = softmax_weights(corr, alpha)
            g = grad_weighted(cur, w, dx)
            g -= g.mean()  # enforce sum-zero direction
            gnorm = np.linalg.norm(g)
            if gnorm < 1e-12:
                break
            direction = -g / gnorm
            
            # Line search
            improved = False
            for lr_try in [lr, lr*0.5, lr*0.25, lr*0.1, lr*0.05, lr*0.01]:
                h_new = cur + lr_try * direction
                h_new = project_box_sum(h_new, target)
                c5_new, _ = c5_eval(h_new, dx)
                if c5_new < cur_c5 - 1e-15:
                    cur = h_new
                    cur_c5 = c5_new
                    improved = True
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = cur.copy()
                    break
            
        if not improved:
            lr = max(0.005, lr * 0.7)
    
    return best_h, best_c5

def build_modes(n, L):
    j = (np.arange(n) + 0.5) / n
    B = np.zeros((n, L))
    for r in range(L):
        B[:, r] = np.cos(np.pi * (r+1) * j)
        B[:, r] -= B[:, r].mean()
        norm = np.linalg.norm(B[:, r])
        if norm > 1e-10:
            B[:, r] /= norm
    return B

def pairwise_moves(h, dx, target, budget_s, t0, rng, n_trials=500, min_time=5.0):

- The sum constraint means h[i] = target - sum(others), which is fixed if I don't change others
- Wait, that's not right. The sum constraint is a single linear constraint. If I change h[i], I need to compensate by changing something else.

A better approach: **2-coordinate optimization**. Pick two coordinates i and j. The constraint is h[i] + h[j] = c (constant, to maintain the sum). So h[i] = t, h[j] = c - t. The objective as a function of t is:
f(t) = max_k corr_k(t)

Each corr_k(t) is a quadratic in t (at most, since corr_k is bilinear in h). So f(t) is the max of K quadratics.

The minimum of f(t) can be found by:
1. Finding the upper envelope of the K quadratics
2. Finding the minimum of this envelope

This is a 1D problem that can be solved by ternary search (if f is unimodal) or by finding the critical points.

In practice, a simpler approach: use bisection or golden section search on t, evaluating f(t) at each step.

But this requires evaluating f(t) = max_k corr_k(t) at each step, which is O(K) = O(n). And the search takes O(log(1/ε)) steps. So each 2-coordinate optimization takes O(n log(1/ε)) ≈ O(n) operations. For n=160, that's very fast.

The question is: how many pairs to try? If I try all O(n^2) pairs, that's O(n^3) total, which for n=160 is about 4M operations per sweep. That's fast enough to do multiple sweeps.

def coordinate_2d_optimize(h, dx, target, n_sweeps, budget_s, t0, min_time=5.0):
    n = len(h)
    best_h = h.copy()
    best_c5, _ = c5_eval(h, dx)
    
    for sweep in range(n_sweeps):
        if budget_s - (time.time() - t0) < min_time:
            break
        improved = False
        for i in range(n):
            for j in range(i+1, n):
                c = best_h[i] + best_h[j]
                # Binary search for optimal t
                # f(t) = C5 with h[i]=t, h[j]=c-t (then projected)
                lo, hi = max(0, c-1), min(1, c)
                if lo >= hi:
                    continue
                # Golden section search
                gr = (np.sqrt(5) - 1) / 2
                a, b = lo, hi
                c_pt = b - gr * (b - a)
                d_pt = a + gr * (b - a)
                
                def eval_t(t):
                    h_t = best_h.copy()
                    h_t[i] = t
                    h_t[j] = c - t
                    h_t = project_box_sum(h_t, target)
                    val, _ = c5_eval(h_t, dx)
                    return val
                
                fc = eval_t(c_pt)
                fd = eval_t(d_pt)
                for _ in range(40):
                    if fc < fd:
                        b = d_pt
                        d_pt = c_pt
                        fd = fc
                        c_pt = b - gr * (b - a)
                        fc = eval_t(c_pt)
                    else:
                        a = c_pt
                        c_pt = d_pt
                        fc = fd
                        d_pt = a + gr * (b - a)
                        fd = eval_t(d_pt)
                
                t_opt = (a + b) / 2
                h_test = best_h.copy()
                h_test[i] = t_opt
                h_test[j] = c - t_opt
                h_test = project_box_sum(h_test, target)
                c5_test, _ = c5_eval(h_test, dx)
                
                if c5_test < best_c5 - 1e-14:
                    best_c5 = c5_test
                    best_h = h_test
                    improved = True
        
        if not improved:
            break
    
    return best_h, best_c5

import numpy as np
import time

def project_box_sum(x, target):
    n = len(x)
    x = np.where(np.isfinite(x), x, 0.5)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = x.min() - abs(target) - 2.0
    hi = x.max() + abs(target) + 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target:
            lo = mid
        else:
            hi = mid
    s = (lo + hi) * 0.5
    h = np.clip(x - s, 0.0, 1.0)
    err = target - h.sum()
    if abs(err) > 1e-10:
        if err > 0:
            mask = h < 1.0 - 1e-10
            cnt = mask.sum()
            if cnt > 0:
                h[mask] += err / cnt
        else:
            mask = h > 1e-10
            cnt = mask.sum()
            if cnt > 0:
                h[mask] += err / cnt
    return np.clip(h, 0.0, 1.0)

def c5_eval(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * dx, corr

def softmax_weights(corr, alpha):
    scaled = corr / alpha
    m = scaled.max()
    w = np.exp(scaled - m)
    s = w.sum()
    if s < 1e-300:
        return np.full(len(corr), 1.0 / len(corr))
    return w / s

def grad_weighted(h, weights, dx):
    n = len(h)
    grad = np.zeros(n)
    grad += weights[n-1] * (1.0 - 2.0 * h)
    for d in range(1, n):
        wp = weights[n-1+d]
        wn = weights[n-1-d]
        if wp == 0.0 and wn == 0.0:
            continue
        if wp != 0.0:
            idx = np.arange(n - d)
            grad[idx] += wp * (1.0 - h[idx + d])
            idx2 = np.arange(d, n)
            grad[idx2] -= wp * h[idx2 - d]
        if wn != 0.0:
            idx = np.arange(d, n)
            grad[idx] += wn * (1.0 - h[idx - d])
            idx2 = np.arange(n - d)
            grad[idx2] -= wn * h[idx2 + d]
    return dx * grad

def softmax_descend(h, dx, target, alphas, n_iters_per_alpha, budget_s, t0, min_time=5.0, lr_init=0.03):
    n = len(h)
    best_h = h.copy()
    best_c5, _ = c5_eval(h, dx)
    cur = h.copy()
    cur_c5 = best_c5
    lr = lr_init
    
    for alpha in alphas:
        if budget_s - (time.time() - t0) < min_time:
            break
        stagnant = 0
        for it in range(n_iters_per_alpha):
            if budget_s - (time.time() - t0) < min_time:
                break
            _, corr = c5_eval(cur, dx)
            w = softmax_weights(corr, alpha)
            g = grad_weighted(cur, w, dx)
            g -= g.mean()
            gnorm = np.linalg.norm(g)
            if gnorm < 1e-14:
                stagnant = n_iters_per_alpha
                break
            direction = -g / gnorm
            
            improved = False
            for t_factor in [lr, lr*0.7, lr*0.5, lr*0.3, lr*0.2, lr*0.1, lr*0.05]:
                h_new = cur + t_factor * direction
                h_new = project_box_sum(h_new, target)
                c5_new, _ = c5_eval(h_new, dx)
                if c5_new < cur_c5 - 1e-15:
                    cur = h_new
                    cur_c5 = c5_new
                    improved = True
                    stagnant = 0
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = cur.copy()
                    break
            
            if not improved:
                stagnant += 1
                if stagnant > 30:
                    lr = max(0.002, lr * 0.8)
                    stagnant = 0
    
    return best_h, best_c5

def build_cos_modes(n, L):
    j = (np.arange(n) + 0.5) / n
    B = np.zeros((n, L))
    for r in range(L):
        B[:, r] = np.cos(np.pi * (r+1) * j)
        B[:, r] -= B[:, r].mean()
        nm = np.linalg.norm(B[:, r])
        if nm > 1e-10:
            B[:, r] /= nm
    return B
