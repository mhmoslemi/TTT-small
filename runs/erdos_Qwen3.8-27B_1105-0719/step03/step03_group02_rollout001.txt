The key ideas from the strategy:
1. **Epigraph formulation** - Instead of using log-sum-exp smoothing, directly minimize `tau` subject to `tau >= dx * corrs(k, h)` for active constraints.
2. **LMO-based vertex seeds** - Use linearized gradient information to generate balanced binary starting points.

Let me think about what's actually different and potentially better:

1. **Epigraph with scipy SLSQP**: Instead of `logsumexp(beta * corrs) / beta` as objective, use `tau` as objective with inequality constraints `tau - dx * corrs[k] >= 0` for a subset of active k values. This is a different optimization landscape.

2. **LMO vertex generation**: Linearize the problem, find the optimal binary vector by solving a linear assignment, then line-search between current solution and that vertex.

3. **Coordinate descent with exact max**: Directly optimize individual coordinates while tracking the exact maximum correlation.

Let me think about implementation details:

For the epigraph approach with SLSQP:
- Variables: h[0..n-1], tau
- Objective: minimize tau
- Constraints: tau - dx * corrs[k] >= 0 for active k values
- Equality: sum(h) = n/2
- Bounds: 0 <= h[i] <= 1, tau >= 0

The issue is that corrs(k) = sum_i h[i] * (1 - h[i+offset]) which is bilinear (nonlinear). So the constraints are nonlinear inequality constraints, which SLSQP can handle.

For the LMO approach:
- At current h, compute aggregated gradient G = sum_k w_k * grad_k
- Solve: min G^T v subject to 0 <= v <= 1, sum(v) = n/2
- The solution is: set v_i = 1 for the n/2 indices with smallest G_i, v_i = 0 otherwise
- Then line search: find best gamma in [0,1] for (1-gamma)*h + gamma*v

This is essentially the Frank-Wolfe / conditional gradient approach adapted to our problem.

        h = np.clip(x - mid, 0.0, 1.0)

            lo = mid
        else:

        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def c5_of_h(h, dx):
    return compute_corrs(h).max() * dx

def grad_corr_k(h, k, n):
    offset = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -offset)
    i_hi = min(n - 1, n - 1 - offset)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    grad[idx] += (1.0 - h[idx + offset])
    grad[idx + offset] += (-h[idx])
    return grad

def all_grads(h, n, k_indices):
    """Compute gradients for a set of k indices."""
    result = np.zeros((len(k_indices), n))
    for ii, k in enumerate(k_indices):
        result[ii] = grad_corr_k(h, k, n)
    return result

def epigraph_solve(h0, dx, target_sum, n_active=64, maxiter=500):
    """Solve min tau s.t. tau >= dx*corrs[k] for active k, sum(h)=n/2, 0<=h<=1."""
    n = len(h0)
    
    def obj(z):
        return z[n]  # tau
    
    def grad_obj(z):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g
    
    # We need to handle active set dynamically
    # For SLSQP, we provide constraints as a fixed set
    # Let's use a moderate active set
    corrs0 = compute_corrs(h0)
    top_idx = np.argsort(corrs0)[-n_active:]
    
    # Build constraint functions
    def make_constraint(k):
        def fun(z):
            h = z[:n]
            tau = z[n]
            offset = k - (n - 1)
            i_lo = max(0, -offset)
            i_hi = min(n - 1, n - 1 - offset)
            if i_lo > i_hi:
                return tau  # constraint always satisfied
            idx = np.arange(i_lo, i_hi + 1)
            corr_val = np.sum(h[idx] * (1.0 - h[idx + offset]))
            return tau - dx * corr_val
        def jac(z):
            h = z[:n]
            offset = k - (n - 1)
            i_lo = max(0, -offset)
            i_hi = min(n - 1, n - 1 - offset)
            j = np.zeros(n + 1)
            j[n] = 1.0
            if i_lo <= i_hi:
                idx = np.arange(i_lo, i_hi + 1)
                j[idx] += -(1.0 - h[idx + offset])
                j[idx + offset] += -h[idx]
            j[:n] *= dx
            return j
        return fun, jac
    
    constraints = []
    for k in top_idx:
        fun, jac = make_constraint(k)
        constraints.append({'type': 'ineq', 'fun': fun, 'jac': jac})

        return z[:n].sum() - target_sum
    def eq_jac(z):
        j = np.zeros(n + 1)
        j[:n] = 1.0
        return j

    """Iteratively update active set and solve epigraph."""
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)

        if c5_val < best_c5:

            best_h = h_new.copy()
        h = h_new
    
    return best_h, best_c5

def lmo_vertex(h, dx, target_sum, n_top=32):
    """Generate a balanced binary vertex using linearized gradient info."""
    n = len(h)
    corrs = compute_corrs(h)
    top_idx = np.argsort(corrs)[-n_top:]
    
    # Aggregated gradient
    G = np.zeros(n)
    for k in top_idx:
        G += dx * grad_corr_k(h, k, n)
    
    # LMO: min G^T v s.t. 0<=v<=1, sum(v)=n/2
    # Solution: v_i = 1 for n/2 indices with smallest G_i
    n_ones = n // 2
    v = np.zeros(n)
    smallest_idx = np.argsort(G)[:n_ones]
    v[smallest_idx] = 1.0
    
    return v

def lmo_line_search(h, v, dx, target_sum):
    """Line search along (1-gamma)*h + gamma*v."""
    n = len(h)
    best_gamma = 0.0
    best_c5 = c5_of_h(h, dx)
    
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = 0.0, 1.0
    c_pt = b - gr * (b - a)
    d_pt = a + gr * (b - a)
    
    def f(gamma):
        h_try = (1.0 - gamma) * h + gamma * v
        h_try = project_feasible(h_try, target_sum)
        return c5_of_h(h_try, dx)
    
    fc, fd = f(c_pt), f(d_pt)
    for _ in range(40):
        if b - a < 1e-12:
            break
        if fc < fd:
            b = d_pt
            d_pt = c_pt
            fd = fc
            c_pt = b - gr * (b - a)
            fc = f(c_pt)
        else:
            a = c_pt
            c_pt = d_pt
            fc = fd
            d_pt = a + gr * (b - a)
            fd = f(d_pt)
    
    best_gamma = (a + b) / 2.0
    h_new = project_feasible((1.0 - best_gamma) * h + best_gamma * v, target_sum)
    c5_new = c5_of_h(h_new, dx)
    return h_new, c5_new

def conditional_gradient_step(h, dx, target_sum, n_top=32):
    """One conditional gradient (Frank-Wolfe) step."""
    n = len(h)
    v = lmo_vertex(h, dx, target_sum, n_top)
    h_new, c5_new = lmo_line_search(h, v, dx, target_sum)
    return h_new, c5_new

def cond_grad_main(h0, dx, target_sum, n_steps=50):
    """Main conditional gradient optimization."""
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    gamma_lr = 2.0 / 3.0  # Standard FW step size
    
    for step in range(n_steps):
        # FW step
        v = lmo_vertex(h, dx, target_sum, n_top=32)
        h_fw = (1.0 - gamma_lr / (step + 2)) * h + (gamma_lr / (step + 2)) * v
        h_fw = project_feasible(h_fw, target_sum)
        c5_fw = c5_of_h(h_fw, dx)
        
        if c5_fw < best_c5:
            best_c5 = c5_fw
            best_h = h_fw.copy()
        h = h_fw
        
        # Also try line search
        h_ls, c5_ls = lmo_line_search(h, v, dx, target_sum)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()
            h = h_ls
    
    return best_h, best_c5

        grad = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] > 1e-12:
                grad += w[k] * grad_corr_k(h, k, n)
        grad *= dx

        dec_idx = np.argsort(grad)[-n_top:]
        inc_idx = np.argsort(grad)[:n_top]
        improved = False
        for i in dec_idx:
            for j in inc_idx:
                if i == j:
                    continue
                eps_hi = min(h[i], 1.0 - h[j])
                if eps_hi < 1e-12:
                    continue
                d = np.zeros(2 * n - 1)
                for k in range(2 * n - 1):
                    gk = grad_corr_k(h, k, n)
                    d[k] = gk[j] - gk[i]
                gr = (np.sqrt(5.0) - 1.0) / 2.0
                a, b = 0.0, eps_hi

                def f_ps(e):
                    return (corrs + e * d).max()

                c_pt = b - gr * (b - a)
                d_pt = a + gr * (b - a)
                fc, fd = f_ps(c_pt), f_ps(d_pt)
                for _ in range(40):
                    if b - a < 1e-14:
                        break
                    if fc < fd:
                        b = d_pt
                        d_pt = c_pt
                        fd = fc
                        c_pt = b - gr * (b - a)
                        fc = f_ps(c_pt)
                    else:
                        a = c_pt
                        c_pt = d_pt
                        fc = fd
                        d_pt = a + gr * (b - a)
                        fd = f_ps(d_pt)
                eps_opt = (a + b) / 2.0
                if eps_opt < 1e-13:
                    continue
                new_h = h.copy()
                new_h[i] -= eps_opt
                new_h[j] += eps_opt
                new_h = np.clip(new_h, 0.0, 1.0)
                new_h = project_feasible(new_h, target_sum)
                new_c5 = c5_of_h(new_h, dx)
                if new_c5 < best_c5 - 1e-15:
                    h = new_h
                    best_c5 = new_c5
                    best_h = h.copy()
                    corrs = compute_corrs(h)
                    improved = True
        if not improved:
            break
    return best_h, best_c5


def slsqp_smooth(h0, dx, target_sum, beta, maxiter=800):
    n = len(h0)

    def obj(x):
        corrs = compute_corrs(x)
        return logsumexp(beta * corrs) / beta * dx

    def grad_obj(x):
        corrs = compute_corrs(x)
        shifted = corrs - corrs.max()
        w = np.exp(beta * shifted)
        w /= w.sum()
        g = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] > 1e-13:
                g += w[k] * grad_corr_k(h, k, n) if False else w[k] * grad_corr_k(x, k, n)
        return g * dx

    def c_fun(x):
        return x.sum() - target_sum

    def c_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': c_fun, 'jac': c_jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0, jac=grad_obj, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        h_out = project_feasible(res.x, target_sum)
    except Exception:
        h_out = project_feasible(h0, target_sum)
    return h_out, c5_of_h(h_out, dx)


def full_optimize(h0, dx, target_sum, time_left, label=""):
    n = len(h0)
    best_h = h0.copy()
    best_c5 = c5_of_h(h0, dx)

    # Phase A: Epigraph with growing active set
    for na in [40, 64, 100, 160]:
        if time_left() < 20:
            break
        h_e, c5_e = epigraph_iterative(best_h, dx, target_sum, max_rounds=8, n_active=na)
        if c5_e < best_c5:
            best_c5 = c5_e
            best_h = h_e

    # Phase B: Conditional gradient
    if time_left() > 30:
        h_cg, c5_cg = conditional_gradient(best_h, dx, target_sum, n_steps=40)
        if c5_cg < best_c5:
            best_c5 = c5_cg
            best_h = h_cg
        # Polish after CG
        if time_left() > 20:
            h_lp1, c5_lp1 = lp_polish(best_h, dx, target_sum, max_iters=150, n_active=50)
            if c5_lp1 < best_c5:
                best_c5 = c5_lp1
                best_h = h_lp1

    # Phase C: Smoothed SLSQP
    if time_left() > 25:
        for beta in [500, 1500, 4000]:
            if time_left() < 15:
                break
            h_s, c5_s = slsqp_smooth(best_h, dx, target_sum, beta, 600)
            if c5_s < best_c5:
                best_c5 = c5_s
                best_h = h_s

    # Phase D: Full active set epigraph
    if time_left() > 25:
        h_full, c5_full = epigraph_iterative(best_h, dx, target_sum, max_rounds=6, n_active=200, n_all=True)
        if c5_full < best_c5:
            best_c5 = c5_full
            best_h = h_full

    # Phase E: LP polish
    if time_left() > 15:
        h_lp, c5_lp = lp_polish(best_h, dx, target_sum, max_iters=200, n_active=70, init_step=0.08)
        if c5_lp < best_c5:
            best_c5 = c5_lp
            best_h = h_lp

    # Phase F: Pair transfer
    if time_left() > 15:
        h_pt, c5_pt = pair_polish(best_h, dx, target_sum, n_top=20, n_iter=3)
        if c5_pt < best_c5:
            best_c5 = c5_pt
            best_h = h_pt

    # Phase G: Final epigraph + LP
    if time_left() > 20:
        h_fe, c5_fe = epigraph_iterative(best_h, dx, target_sum, max_rounds=5, n_active=80)
        if c5_fe < best_c5:
            best_c5 = c5_fe
            best_h = h_fe
    if time_left() > 15:
        h_fl, c5_fl = lp_polish(best_h, dx, target_sum, max_iters=100, n_active=40, init_step=0.05)
        if c5_fl < best_c5:
            best_c5 = c5_fl
            best_h = h_fl
    if time_left() > 10:
        h_fp, c5_fp = pair_polish(best_h, dx, target_sum, n_top=30, n_iter=3)
        if c5_fp < best_c5:
            best_c5 = c5_fp
            best_h = h_fp

    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s - (time.time() - t0)

    h0 = np.array(initial_h_values, dtype=float)
    n = len(h0)
    dx = 2.0 / n
    target_sum = n / 2.0
    h0 = project_feasible(h0, target_sum)

    best_h = h0.copy()
    best_c5 = c5_of_h(h0, dx)

    # Main optimization
    h_opt, c5_opt = full_optimize(h0, dx, target_sum, time_left, "main")
    if c5_opt < best_c5:
        best_c5 = c5_opt
        best_h = h_opt

    # Perturbation attempts
    for trial in range(12):
        if time_left() < 30:
            break
        if trial % 3 == 0:
            perturb = 0.04 * (rng.random(n) - 0.5)
        elif trial % 3 == 1:
            perturb = 0.12 * (rng.random(n) - 0.5)
        else:
            idx = rng.choice(n, size=max(5, n // 8), replace=False)
            perturb = np.zeros(n)
            perturb[idx] = rng.choice([-1, 1], size=len(idx)) * 0.08
        h_try = project_feasible(best_h + perturb, target_sum)
        h_p, c5_p = full_optimize(h_try, dx, target_sum, time_left, "pert")
        if c5_p < best_c5:
            best_c5 = c5_p
            best_h = h_p

    # Try different n values
    n_candidates = sorted(range(n - 10, n + 12, 2))
    for n_try in n_candidates:
        if time_left() < 35:
            break
        if n_try == n or n_try < 40:
            continue
        dx_try = 2.0 / n_try
        ts_try = n_try / 2.0
        x_orig = np.linspace(0.0, 2.0, n, endpoint=False) + 1.0 / n
        x_new = np.linspace(0.0, 2.0, n_try, endpoint=False) + 1.0 / n_try
        h_interp = np.interp(x_new, x_orig, best_h)
        h_interp = project_feasible(h_interp, ts_try)
        h_n, c5_n = full_optimize(h_interp, dx_try, ts_try, time_left, f"n{n_try}")
        if c5_n < best_c5:
            best_c5 = c5_n
            best_h = h_n

    # Final intensive polish
    if time_left() > 25 and best_h is not None:
        n_b = len(best_h)
        dx_b = 2.0 / n_b
        ts_b = n_b / 2.0
        h_f, c5_f = full_optimize(best_h, dx_b, ts_b, time_left, "final")
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f

    # Ultra-fine pair polish
    if time_left() > 10 and best_h is not None:
        n_b = len(best_h)
        dx_b = 2.0 / n_b
        ts_b = n_b / 2.0
        h_pp, c5_pp = pair_polish(best_h, dx_b, ts_b, n_top=40, n_iter=5)
        if c5_pp < best_c5:
            best_c5 = c5_pp
            best_h = h_pp

    # Final validation
    if best_h is None:
        best_h = h0
        best_c5 = c5_of_h(h0, dx)
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final / 2.0)
    best_c5 = c5_of_h(best_h, 2.0 / n_final)

    return best_h, best_c5, n_final
