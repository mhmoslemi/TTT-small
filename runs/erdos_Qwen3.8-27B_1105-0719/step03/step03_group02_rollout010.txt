    corrs = compute_corrs(h)
    return logsumexp(beta * corrs) / beta * dx


def smooth_max_grad(h, dx, beta):
    n = len(h)
    corrs = compute_corrs(h)
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones_like(corrs) / len(corrs)
    else:
        w /= w_sum
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] > 1e-13:
            grad += w[k] * grad_corr_k(h, k, n)
    return grad * dx


def slsqp_direct(h0, dx, target_sum, beta, maxiter=2000):
    n = len(h0)

    def obj(x):
        return smooth_max_obj(x, dx, beta)

    def grad_obj(x):
        return smooth_max_grad(x, dx, beta)

    def c_fun(x):
        return x.sum() - target_sum

    def c_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': c_fun, 'jac': c_jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0, jac=grad_obj, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        h_out = project_feasible(res.x, target_sum)
    except Exception:
        h_out = project_feasible(h0, target_sum)
    return h_out, c5_of_h(h_out, dx)


def epigraph_solve(h0, dx, target_sum, n, n_active=50, maxiter=3000):
    corrs0 = compute_corrs(h0)
    tau0 = corrs0.max() * dx + 1e-5
    top_idx = np.argsort(corrs0)[-n_active:]
    n_vars = n + 1

    def obj(z):
        return z[n]

    def obj_grad(z):
        g = np.zeros(n_vars)
        g[n] = 1.0
        return g

    constraints = []
    for k in top_idx:
        kk = int(k)

        def inq_fun(z, kk=kk):
            return z[n] - dx * compute_corrs(z[:n])[kk]

        def inq_jac(z, kk=kk):
            h = z[:n]
            g = np.zeros(n_vars)
            g[n] = 1.0
            g[:n] = -dx * grad_corr_k(h, kk, n)
            return g

        constraints.append({'type': 'ineq', 'fun': inq_fun, 'jac': inq_jac})

    def eq_fun(z):
        return z[:n].sum() - target_sum

    def eq_jac(z):
        g = np.zeros(n_vars)
        g[:n] = 1.0
        return g

    constraints.append({'type': 'eq', 'fun': eq_fun, 'jac': eq_jac})
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    h0c = np.clip(h0, 1e-10, 1.0 - 1e-10)
    z0 = np.concatenate([h0c, [tau0]])
    try:
        res = minimize(obj, z0, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        h_out = project_feasible(res.x[:n], target_sum)
        return h_out, c5_of_h(h_out, dx)
    except Exception:
        return h0, c5_of_h(h0, dx)


def lmo_binary_seed(h, n, target_sum, n_active=64, beta_w=5.0):
    dx = 2.0 / n
    corrs = compute_corrs(h)
    shifted = corrs - corrs.max()
    w = np.exp(beta_w * shifted)
    w /= w.sum()
    top_idx = np.argsort(corrs)[-n_active:]
    G = np.zeros(n)
    for k in top_idx:
        G += w[k] * grad_corr_k(h, k, n)
    n_ones = int(round(target_sum))
    sorted_idx = np.argsort(G)
    v = np.zeros(n)
    v[sorted_idx[:n_ones]] = 1.0
    return v


def lmo_multi_step(h, dx, target_sum, n, n_trials=8, n_active=64):
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    rng_local = np.random.RandomState(sum(h * 1e6) % 100000)
    for t in range(n_trials):
        if t == 0:
            n_act = n_active
        else:
            n_act = max(16, n_active + rng_local.randint(-32, 32))
        v = lmo_binary_seed(h, n, target_sum, n_active=n_act)
        for gamma in [0.15, 0.3, 0.5, 0.7, 0.9]:
            h_try = (1.0 - gamma) * h + gamma * v
            h_try = project_feasible(h_try, target_sum)
            c5_try = c5_of_h(h_try, dx)
            if c5_try < best_c5 - 1e-14:
                best_c5 = c5_try
                best_h = h_try.copy()
                h = h_try
    return best_h, best_c5


def lp_polish(h0, dx, target_sum, max_iters=500, init_step=0.15, tol_frac=0.08):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    step = init_step
    stall = 0

    for it in range(max_iters):
        corrs = compute_corrs(h)
        max_c = corrs.max()
        c5_val = max_c * dx

        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 60:
            step *= 0.65
            stall = 0
            if step < 5e-8:
                break

        tol_val = max_c * (1.0 - tol_frac)
        active_k = np.where(corrs >= tol_val)[0]
        if len(active_k) > 100:
            active_k = np.argsort(corrs)[-100:]

        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        A_rows = []
        b_rows = []
        for k in active_k:
            row = np.zeros(n + 1)
            row[:n] = dx * grad_corr_k(h, k, n)
            row[n] = -1.0
            A_rows.append(row)
            b_rows.append(-corrs[k] * dx - 1e-14)

        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)

        bounds = []
        for i in range(n):
            bounds.append((max(-h[i], -step), min(1.0 - h[i], step)))
        bounds.append((0.0, None))

        try:
            res = linprog(c_obj, A_ub=np.array(A_rows), b_ub=np.array(b_rows),
                          A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
        except Exception:
            break
        if res.status != 0:
            break

        delta = res.x[:n]
        if np.max(np.abs(delta)) < 1e-14:
            break

        h_new = project_feasible(h + delta, target_sum)
        new_c5 = c5_of_h(h_new, dx)

        if new_c5 < c5_val - 1e-14:
            h = h_new
            step = min(step * 1.3, init_step * 1.5)
        else:
            improved = False
            for frac in [0.5, 0.25, 0.1, 0.05]:
                h_new = project_feasible(h + frac * delta, target_sum)
                new_c5 = c5_of_h(h_new, dx)
                if new_c5 < c5_val - 1e-14:
                    h = h_new
                    improved = True
                    break
            if not improved:
                step *= 0.4
                if step < 1e-9:
                    break

    return best_h, best_c5


def pair_transfer(h0, dx, target_sum, n_top=25, n_iter=5):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)

    for iteration in range(n_iter):
        corrs = compute_corrs(h)
        grad = smooth_max_grad(h, dx, 800.0)

        dec_idx = np.argsort(grad)[-n_top:]
        inc_idx = np.argsort(grad)[:n_top]

        improved = False
        for i in dec_idx:
            for j in inc_idx:
                if i == j:
                    continue
                d = np.zeros(2 * n - 1)
                for k in range(2 * n - 1):
                    gk = grad_corr_k(h, k, n)
                    d[k] = gk[j] - gk[i]

                eps_hi = min(h[i], 1.0 - h[j])
                if eps_hi < 1e-12:
                    continue

                gr = (np.sqrt(5.0) - 1.0) / 2.0
                a, b = 0.0, eps_hi
                c_pt = b - gr * (b - a)
                d_pt = a + gr * (b - a)
                fc = (corrs + c_pt * d).max()
                fd = (corrs + d_pt * d).max()
                for _ in range(40):
                    if b - a < 1e-13:
                        break
                    if fc < fd:
                        b = d_pt
                        d_pt = c_pt
                        fd = fc
                        c_pt = b - gr * (b - a)
                        fc = (corrs + c_pt * d).max()
                    else:
                        a = c_pt
                        c_pt = d_pt
                        fc = fd
                        d_pt = a + gr * (b - a)
                        fd = (corrs + d_pt * d).max()

                eps_opt = (a + b) / 2.0
                if eps_opt < 1e-13:
                    continue
                new_c5_r = (corrs + eps_opt * d).max() * dx
                if new_c5_r < best_c5 - 1e-14:
                    new_h = h.copy()
                    new_h[i] -= eps_opt
                    new_h[j] += eps_opt
                    new_h = project_feasible(new_h, target_sum)
                    new_c5 = c5_of_h(new_h, dx)
                    if new_c5 < best_c5 - 1e-14:
                        h = new_h
                        best_c5 = new_c5
                        best_h = h.copy()
                        corrs = compute_corrs(h)
                        improved = True

        if not improved:
            break

    return best_h, best_c5


def momentum_subgrad(h0, dx, target_sum, max_iter=12000, beta_smooth=50.0):
    n = len(h0)
    h = h0.copy()
    momentum = np.zeros(n)
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    no_improve = 0
    lr = 1.0 / np.sqrt(n)
    beta_mom = 0.9

    for it in range(max_iter):
        corrs = compute_corrs(h)
        c5_val = corrs.max() * dx

        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 4000:
            break

        shifted = corrs - corrs.max()
        w = np.exp(beta_smooth * shifted)
        w /= w.sum()

        grad = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] > 1e-12:
                grad += w[k] * grad_corr_k(h, k, n)
        grad *= dx

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        momentum = beta_mom * momentum - lr * grad / (gnorm + 1e-10)
        h_new = project_feasible(h + momentum, target_sum)
        new_c5 = c5_of_h(h_new, dx)

        tries = 0
        while new_c5 > c5_val + 1e-14 and tries < 4:
            lr *= 0.5
            momentum = beta_mom * momentum - lr * grad / (gnorm + 1e-10)
            h_new = project_feasible(h + momentum, target_sum)
            new_c5 = c5_of_h(h_new, dx)
            tries += 1
        h = h_new

    return best_h, best_c5


def full_optimize(h0, dx, target_sum, time_left):
    n = len(h0)
    best_h = h0.copy()
    best_c5 = c5_of_h(h0, dx)

    for beta in [100, 300, 800, 2000, 5000]:
        if time_left() < 20:
            break
        ho, cv = slsqp_direct(best_h, dx, target_sum, beta, 1500)
        if cv < best_c5:
            best_c5 = cv
            best_h = ho

    if time_left() > 40:
        ho, cv = epigraph_solve(best_h, dx, target_sum, n, n_active=50, maxiter=2000)
        if cv < best_c5:
            best_c5 = cv
            best_h = ho

    if time_left() > 30:
        ho, cv = lp_polish(best_h, dx, target_sum, max_iters=400, init_step=0.12, tol_frac=0.08)
        if cv < best_c5:
            best_c5 = cv
            best_h = ho

    if time_left() > 25:
        ho, cv = pair_transfer(best_h, dx, target_sum, n_top=20, n_iter=4)
        if cv < best_c5:
            best_c5 = cv
            best_h = ho

    if time_left() > 50:
        ho, cv = momentum_subgrad(best_h, dx, target_sum, max_iter=8000)
        if cv < best_c5:
            best_c5 = cv
            best_h = ho

    if time_left() > 25:
        ho, cv = slsqp_direct(best_h, dx, target_sum, 4000, 1500)
        if cv < best_c5:
            best_c5 = cv
            best_h = ho

    if time_left() > 20:
        ho, cv = epigraph_solve(best_h, dx, target_sum, n, n_active=80, maxiter=2000)
        if cv < best_c5:
            best_c5 = cv
            best_h = ho

    if time_left() > 15:
        ho, cv = lp_polish(best_h, dx, target_sum, max_iters=200, init_step=0.06, tol_frac=0.12)
        if cv < best_c5:
            best_c5 = cv
            best_h = ho

    if time_left() > 10:
        ho, cv = pair_transfer(best_h, dx, target_sum, n_top=15, n_iter=3)
        if cv < best_c5:
            best_c5 = cv
            best_h = ho

    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s - (time.time() - t0)

    h0 = np.array(initial_h_values, dtype=float)
    n = len(h0)
    dx = 2.0 / n
    target_sum = n / 2.0
    h0 = project_feasible(h0, target_sum)

    best_h = h0.copy()
    best_c5 = c5_of_h(h0, dx)

    # Phase 1: Main optimization from parent
    if time_left() > 60:
        best_h, best_c5 = full_optimize(h0, dx, target_sum, time_left)

    # Phase 2: LMO-based multi-start
    if time_left() > 40:
        for trial in range(12):
            if time_left() < 35:
                break
            v = lmo_binary_seed(best_h, n, target_sum, n_active=48, beta_w=3.0)
            for gamma in [0.3, 0.5, 0.7]:
                h_try = (1.0 - gamma) * best_h + gamma * v
                h_try = project_feasible(h_try, target_sum)
                c5_try = c5_of_h(h_try, dx)
                if c5_try < best_c5:
                    if time_left() > 40:
                        ho, cv = full_optimize(h_try, dx, target_sum, time_left)
                        if cv < best_c5:
                            best_c5 = cv
                            best_h = ho

    # Phase 3: Random perturbations
    for trial in range(20):
        if time_left() < 30:
            break
        if trial % 4 == 0:
            perturb = 0.03 * (rng.random(n) - 0.5)
        elif trial % 4 == 1:
            perturb = 0.1 * (rng.random(n) - 0.5)
        elif trial % 4 == 2:
            idx = rng.choice(n, size=max(5, n // 12), replace=False)
            perturb = np.zeros(n)
            perturb[idx] = rng.choice([-1, 1], size=len(idx)) * 0.08
        else:
            # Flip-based: threshold around 0.5
            thresh = 0.5 + 0.1 * (rng.random() - 0.5)
            perturb = np.where(best_h > thresh, -0.15, 0.15)
        h_try = project_feasible(best_h + perturb, target_sum)
        if time_left() > 40:
            ho, cv = full_optimize(h_try, dx, target_sum, time_left)
            if cv < best_c5:
                best_c5 = cv
                best_h = ho

    # Phase 4: Try different n values
    n_try_list = [n - 10, n - 6, n - 4, n + 4, n + 6, n + 10, n + 14, 100, 120, 140, 160, 180]
    n_try_list = sorted(set([x for x in n_try_list if x >= 40 and x % 2 == 0]))
    n_try_list.sort(key=lambda x: abs(x - n))
    n_try_list = n_try_list[:12]

    for n_try in n_try_list:
        if time_left() < 35:
            break
        if n_try == n:
            continue
        dx_try = 2.0 / n_try
        ts_try = float(n_try) / 2.0
        x_orig = np.linspace(0.0, 2.0, n, endpoint=False) + 1.0 / n
        x_new = np.linspace(0.0, 2.0, n_try, endpoint=False) + 1.0 / n_try
        h_interp = np.interp(x_new, x_orig, best_h)
        h_interp = project_feasible(h_interp, ts_try)
        ho, cv = full_optimize(h_interp, dx_try, ts_try, time_left)
        if cv < best_c5:
            best_c5 = cv
            best_h = ho

    # Phase 5: Final intensive polish
    if time_left() > 15 and best_h is not None:
        n_b = len(best_h)
        dx_b = 2.0 / n_b
        ts_b = float(n_b) / 2.0
        # Aggressive LP with wide tolerance
        ho, cv = lp_polish(best_h, dx_b, ts_b, max_iters=300, init_step=0.1, tol_frac=0.15)
        if cv < best_c5:
            best_c5 = cv
            best_h = ho
        # Pair transfer
        if time_left() > 10:
            ho, cv = pair_transfer(best_h, dx_b, ts_b, n_top=30, n_iter=5)
            if cv < best_c5:
                best_c5 = cv
                best_h = ho
        # Epigraph with many constraints
        if time_left() > 15:
            ho, cv = epigraph_solve(best_h, dx_b, ts_b, n_b, n_active=120, maxiter=2000)
            if cv < best_c5:
                best_c5 = cv
                best_h = ho
        # Final LP
        if time_left() > 10:
            ho, cv = lp_polish(best_h, dx_b, ts_b, max_iters=200, init_step=0.04, tol_frac=0.2)
            if cv < best_c5:
                best_c5 = cv
                best_h = ho
        # Final pair transfer
        if time_left() > 8:
            ho, cv = pair_transfer(best_h, dx_b, ts_b, n_top=20, n_iter=3)
            if cv < best_c5:
                best_c5 = cv
                best_h = ho

    # Final validation
    if best_h is None:
        best_h = h0
        best_c5 = c5_of_h(h0, 2.0 / len(h0))
    else:
        n_final = len(best_h)
        best_h = project_feasible(best_h, float(n_final) / 2.0)
        best_c5 = c5_of_h(best_h, 2.0 / n_final)

    return best_h, best_c5, len(best_h)
