def exact_score(h):

def profile_h(h):

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx])
            h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def gradient_of_lag(h, d_lag):
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    if d_lag == 0:
        grad[:] = dx * (1.0 - 2.0 * h)
    elif d_lag > 0:
        for i in range(n - d_lag):
            grad[i] += dx * (1.0 - h[i + d_lag])
            grad[i + d_lag] -= dx * h[i]
    else:
        dp = -d_lag
        for j in range(n - dp):
            grad[j + dp] += dx * (1.0 - h[j])
            grad[j] -= dx * h[j + dp]
    return grad

def lp_direction_simultaneous(h, active_indices):
    n = len(h)
    if len(active_indices) == 0:
        return None, 0.0
    c = np.zeros(n + 1)
    c[-1] = -1.0
    A_ub = np.zeros((len(active_indices), n + 1))
    b_ub = np.zeros(len(active_indices))
    for idx, k in enumerate(active_indices):
        d_lag = k - (n - 1)
        g = gradient_of_lag(h, d_lag)
        A_ub[idx, :n] = g
        A_ub[idx, n] = 1.0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(-1.0, 1.0)] * n + [(-5.0, 5.0)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            d = res.x[:n].copy()
            t_val = -res.fun
            if t_val > 1e-13:
                return d, t_val
    except Exception:
        pass
    return None, 0.0

def lp_direction_weighted(h, active_indices, weights):
    n = len(h)
    if len(active_indices) == 0:
        return None, 0.0
    c = np.zeros(n + 1)
    c[-1] = -1.0
    g_weighted = np.zeros(n)
    for idx, k in enumerate(active_indices):
        d_lag = k - (n - 1)
        g_weighted += weights[idx] * gradient_of_lag(h, d_lag)
    A_ub = np.zeros((1, n + 1))
    A_ub[0, :n] = g_weighted
    A_ub[0, n] = 1.0
    b_ub = np.zeros(1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(-1.0, 1.0)] * n + [(-5.0, 5.0)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            d = res.x[:n].copy()
            t_val = -res.fun
            if t_val > 1e-13:
                return d, t_val
    except Exception:
        pass
    return None, 0.0

def take_step(h, d, score_before):
    n = len(h)
    alpha_max = 1.0
    for i in range(n):
        if d[i] > 1e-15:
            alpha_max = min(alpha_max, (1.0 - h[i]) / d[i])
        elif d[i] < -1e-15:
            alpha_max = min(alpha_max, h[i] / (-d[i]))
    alpha_max = min(alpha_max, 0.5)
    if alpha_max < 1e-15:
        return h, score_before
    best_h = h.copy()
    best_s = score_before
    for frac in [1.0, 0.8, 0.6, 0.4, 0.25, 0.125, 0.0625]:
        alpha = alpha_max * frac
        if alpha < 1e-15:
            break
        h_try = repair_feasible(h + alpha * d)
        s = exact_score(h_try)
        if s < best_s - 1e-15:
            best_h = h_try
            best_s = s
            break
    return best_h, best_s

def lp_leveling(h, max_rounds=100, t_start=None, budget_s=1100):
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    eps_values = [0.0005, 0.001, 0.002, 0.004, 0.008, 0.015, 0.03, 0.06]
    stall_count = 0
    for rnd in range(max_rounds):
        if t_start is not None and time.perf_counter() - t_start > budget_s - 25:
            break
        P = profile_h(h)
        Pmax = P.max()
        improved_this_round = False
        for eps in eps_values:
            active = np.where(P >= Pmax - eps)[0]
            if len(active) == 0:
                active = np.array([int(np.argmax(P))])
            d, t_val = lp_direction_simultaneous(h, active)
            if d is not None and t_val > 1e-12:
                h_new, s_new = take_step(h, d, best_c)
                if s_new < best_c - 1e-15:
                    h = h_new
                    best_c = s_new
                    improved_this_round = True
                    stall_count = 0
                    P = profile_h(h)
                    Pmax = P.max()
                    break
        if not improved_this_round:
            # Try weighted LP with broader active set
            active = np.where(P >= Pmax - 0.02)[0]
            if len(active) > 50:
                top_idx = np.argsort(P[active])[-50:]
                active = active[top_idx]
            if len(active) == 0:
                active = np.array([int(np.argmax(P))])
            gap = Pmax - P[active] + 0.0005
            weights = gap / gap.sum() * len(active)
            d, t_val = lp_direction_weighted(h, active, weights)
            if d is not None and t_val > 1e-12:
                h_new, s_new = take_step(h, d, best_c)
                if s_new < best_c - 1e-15:
                    h = h_new
                    best_c = s_new
                    stall_count = 0
                else:
                    stall_count += 1
            else:
                stall_count += 1
        if stall_count > 15:
            break
    return h, best_c

def slsqp_active(h0, maxiter=300, active_k=12, tol=1e-12):
    n = len(h0)
    P = profile_h(h0)
    c0 = P.max()
    thresh = c0 - max(0.002, 0.04 * (c0 - np.median(P)))
    active = np.where(P >= thresh)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(P))])
    if len(active) > active_k:
        top_idx = np.argsort(P[active])[-active_k:]
        active = active[top_idx]
    def obj(x):
        return x[-1]
    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    def eq_fun(x):
        return x[:-1].sum() - n / 2.0
    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g
    def ineq_fun(x):
        prof = profile_h(x[:-1])
        return x[-1] - prof[active]
    def ineq_jac(x):
        h = x[:-1]
        J = np.zeros((len(active), n + 1))
        for idx, k in enumerate(active):
            J[idx, :n] = -gradient_of_lag(h, k - (n - 1))
            J[idx, n] = 1.0
        return J
    x0 = np.concatenate([h0, [c0 + 1e-4]])
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 1.0)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': tol})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = repair_feasible(res.x[:-1])
            c_new = exact_score(h_new)
            if c_new <= c0 + 1e-9:
                return h_new
    except Exception:
        pass
    return None

def slsqp_full(h0, maxiter=500, tol=1e-12):
    n = len(h0)
    M = 2 * n - 1
    P = profile_h(h0)
    c0 = P.max()
    def obj(x):
        return x[-1]
    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    def eq_fun(x):
        return x[:-1].sum() - n / 2.0
    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g
    def ineq_fun(x):
        prof = profile_h(x[:-1])
        return x[-1] - prof
    def ineq_jac(x):
        h = x[:-1]
        J = np.zeros((M, n + 1))
        for k in range(M):
            J[k, :n] = -gradient_of_lag(h, k - (n - 1))
            J[k, n] = 1.0
        return J
    x0 = np.concatenate([h0, [c0 + 1e-4]])
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 1.0)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': tol})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = repair_feasible(res.x[:-1])
            c_new = exact_score(h_new)
            if c_new <= c0 + 1e-9:
                return h_new
    except Exception:
        pass
    return None

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    return repair_feasible(np.clip(h_new, 0.0, 1.0))

def symmetrize(h):
    return (h + h[::-1].copy()) / 2.0

def perturb(h, rng, eps=0.01):
    z = rng.randn(len(h))
    z -= z.mean()
    return repair_feasible(h + eps * z)

def gradient_projection_h(h, max_iters=200, t_start=None, budget_s=1100):
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    alpha = 0.005
    for it in range(max_iters):
        if t_start is not None and time.perf_counter() - t_start > budget_s - 25:
            break
        P = profile_h(h)
        Pmax = P.max()
        active = np.where(P >= Pmax - 0.005)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(P))])
        grad = np.zeros(n)
        for k in active:
            grad -= gradient_of_lag(h, k - (n - 1))
        grad -= grad.sum() / n
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad /= gnorm
        h_try = repair_feasible(h - alpha * grad)
        s = exact_score(h_try)
        if s < best_c - 1e-15:
            h = h_try
            best_c = s
            alpha = min(alpha * 1.1, 0.02)
        else:
            alpha *= 0.7
            if alpha < 1e-8:
                break
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()
    rng = np.random.RandomState(seed)
    
    def time_ok(margin=20):
        return time.perf_counter() - t_start < budget_s - margin
    
    parent = None
    try:
        parent = np.array(initial_h_values, dtype=np.float64)
    except NameError:
        pass
    except Exception:
        pass
    
    if parent is None or len(parent) < 4:
        parent = np.full(70, 0.5)
    
    parent = np.clip(parent, 0.0, 1.0)
    parent = repair_feasible(parent)
    
    best_h = parent.copy()
    best_c = exact_score(parent)
    
    # Symmetrize if helpful
    if time_ok():
        h_sym = repair_feasible(symmetrize(parent))
        c_sym = exact_score(h_sym)
        if c_sym < best_c - 1e-14:
            best_h = h_sym
            best_c = c_sym
    
    # Phase 1: LP leveling (primary new method)
    if time_ok(80):
        h_lp, c_lp = lp_leveling(best_h, max_rounds=120, t_start=t_start, budget_s=budget_s)
        if c_lp < best_c - 1e-15:
            best_h = h_lp
            best_c = c_lp
    
    # Phase 2: SLSQP active polish
    if time_ok(80):
        sol = slsqp_active(best_h, maxiter=300, active_k=12, tol=1e-12)
        if sol is not None:
            c = exact_score(sol)
            if c < best_c - 1e-14:
                best_h = sol
                best_c = c
    
    # Phase 3: More LP with perturbations
    if time_ok(100):
        for eps in [0.002, 0.005, 0.01, 0.02, 0.035]:
            if not time_ok(30):
                break
            h_pert = perturb(best_h, rng, eps)
            h_lp, c_lp = lp_leveling(h_pert, max_rounds=60, t_start=t_start, budget_s=budget_s)
            if c_lp < best_c - 1e-14:
                best_h = h_lp
                best_c = c_lp
                if time_ok(25):
                    sol = slsqp_active(best_h, maxiter=200, active_k=12, tol=1e-12)
                    if sol is not None:
                        c = exact_score(sol)
                        if c < best_c - 1e-14:
                            best_h = sol
                            best_c = c
    
    # Phase 4: Gradient projection for intermediate refinement
    if time_ok(60):
        h_gp, c_gp = gradient_projection_h(best_h, max_iters=150, t_start=t_start, budget_s=budget_s)
        if c_gp < best_c - 1e-14:
            best_h = h_gp
            best_c = c_gp
        if time_ok(30):
            h_lp, c_lp = lp_leveling(best_h, max_rounds=50, t_start=t_start, budget_s=budget_s)
            if c_lp < best_c - 1e-15:
                best_h = h_lp
                best_c = c_lp
    
    # Phase 5: Full SLSQP
    if time_ok(120):
        sol = slsqp_full(best_h, maxiter=400, tol=1e-13)
        if sol is not None:
            c = exact_score(sol)
            if c < best_c - 1e-14:
                best_h = sol
                best_c = c
    
    # Phase 6: Another LP + SLSQP cycle
    if time_ok(80):
        h_lp, c_lp = lp_leveling(best_h, max_rounds=80, t_start=t_start, budget_s=budget_s)
        if c_lp < best_c - 1e-15:
            best_h = h_lp
            best_c = c_lp
        if time_ok(40):
            sol = slsqp_active(best_h, maxiter=300, active_k=15, tol=1e-13)
            if sol is not None:
                c = exact_score(sol)
                if c < best_c - 1e-14:
                    best_h = sol
                    best_c = c
    
    # Phase 7: Dimension exploration
    if time_ok(100):
        n_best = len(best_h)
        cand_dims = [d for d in range(max(40, n_best - 10), min(120, n_best + 15))
                     if d % 2 == 0 and d != n_best]
        cand_dims.sort(key=lambda d: abs(d - n_best))
        for new_n in cand_dims[:6]:
            if not time_ok(40):
                break
            h_interp = interpolate_to_n(best_h, new_n)
            h_lp, c_lp = lp_leveling(h_interp, max_rounds=40, t_start=t_start, budget_s=budget_s)
            if c_lp < best_c - 1e-12:
                best_h = h_lp
                best_c = c_lp
                if time_ok(20):
                    sol = slsqp_active(best_h, maxiter=200, active_k=12, tol=1e-12)
                    if sol is not None:
                        c = exact_score(sol)
                        if c < best_c - 1e-14:
                            best_h = sol
                            best_c = c
            elif time_ok(30):
                sol = slsqp_active(h_interp, maxiter=250, active_k=12, tol=1e-12)
                if sol is not None:
                    c = exact_score(sol)
                    if c < best_c - 1e-12:
                        best_h = sol
                        best_c = c
    
    # Phase 8: Final LP polish
    if time_ok(40):
        h_lp, c_lp = lp_leveling(best_h, max_rounds=60, t_start=t_start, budget_s=budget_s)
        if c_lp < best_c - 1e-15:
            best_h = h_lp
            best_c = c_lp
    
    # Phase 9: Final SLSQP with tightest tolerance
    if time_ok(25):
        sol = slsqp_active(best_h, maxiter=300, active_k=10, tol=1e-14)
        if sol is not None:
            c = exact_score(sol)
            if c < best_c - 1e-15:
                best_h = sol
                best_c = c
    
    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)
    
    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            final_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 1:
            if np.isfinite(eval_result[0]):
                final_c = float(eval_result[0])
    except Exception:
        pass
    
    return best_h, final_c, len(best_h)
