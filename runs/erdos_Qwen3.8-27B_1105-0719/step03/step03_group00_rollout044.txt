import numpy as np
from scipy.optimize import linprog
import time


def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)


def c5_of_h(h, dx):
    return np.correlate(h, 1.0 - h, mode='full').max() * dx


def corr_of_h(h):
    return np.correlate(h, 1.0 - h, mode='full')


def grad_lag(h, k, n):
    off = k - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -off)
    i_hi = min(n - 1, n - 1 - off)
    if i_lo > i_hi:
        return grad
    idx = np.arange(i_lo, i_hi + 1)
    if off == 0:
        grad[idx] += (1.0 - 2.0 * h[idx])
    else:
        grad[idx + off] += (1.0 - h[idx])
        grad[idx] += (-h[idx + off])
    return grad


def solve_lp_soft(h, dx, target_sum, active_idx, corrs, alpha, rho_q, rho_s, n):
    K = len(active_idx)
    m_vars = n + K + 1
    c_obj = np.zeros(m_vars)
    for j in range(K):
        c_obj[n + j] = rho_s
    c_obj[-1] = rho_q
    A_ub = np.zeros((K, m_vars))
    for j, k in enumerate(active_idx):
        g = grad_lag(h, k, n)
        A_ub[j, :n] = dx * g
        A_ub[j, n + j] = 1.0
        A_ub[j, -1] = -1.0
    b_ub = -corrs[active_idx]
    A_eq = np.zeros((1, m_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(-h[i] * alpha, (1.0 - h[i]) * alpha) for i in range(n)]
    for _ in range(K):
        bounds.append((0.0, None))
    bounds.append((None, None))
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 5})
        if res.status != 0:
            return None
        return res.x[:n]
    except Exception:
        return None


def solve_lp_hard(h, dx, target_sum, active_idx, corrs, alpha, n):
    K = len(active_idx)
    m_vars = n + 1
    c_obj = np.zeros(m_vars)
    c_obj[-1] = 1.0
    A_ub = np.zeros((K, m_vars))
    for j, k in enumerate(active_idx):
        g = grad_lag(h, k, n)
        A_ub[j, :n] = dx * g
        A_ub[j, -1] = -1.0
    b_ub = -corrs[active_idx]
    A_eq = np.zeros((1, m_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(-h[i] * alpha, (1.0 - h[i]) * alpha) for i in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 5})
        if res.status != 0:
            return None
        return res.x[:n]
    except Exception:
        return None


def solve_lp_linear(h, dx, target_sum, active_idx, corrs, alpha, weights, n):
    m_vars = n
    g = np.zeros(n)
    for j, k in enumerate(active_idx):
        g += weights[j] * grad_lag(h, k, n)
    c_obj = dx * g
    A_eq = np.zeros((1, m_vars))
    A_eq[0, :] = 1.0
    b_eq = np.zeros(1)
    bounds = [(-h[i] * alpha, (1.0 - h[i]) * alpha) for i in range(n)]
    try:
        res = linprog(c_obj, A_eq=A_eq, b_eq=b_eq, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 3})
        if res.status != 0:
            return None
        return res.x[:n]
    except Exception:
        return None


def line_search(h, direction, dx, target_sum, cur_c5, n_max=12):
    best_h = h.copy()
    best_c5 = cur_c5
    t_vals = [1.0, 0.8, 0.6, 0.5, 0.4, 0.3, 0.2, 0.15, 0.1, 0.06, 0.03, 0.01]
    for t in t_vals[:n_max]:
        h_c = h + t * direction
        h_c = project_feasible(h_c, target_sum)
        c5_val = c5_of_h(h_c, dx)
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h_c
    return best_h, best_c5

def binary_round(h, target_sum, rng):
    n = len(h)
    h_bin = (h > 0.5).astype(float)
    target = int(target_sum)
    num_ones = int(h_bin.sum())
    if num_ones < target:
        zeros = np.where(h_bin == 0)[0]
        if len(zeros) > 0:
            order = rng.permutation(len(zeros))
            for idx in order[:target - num_ones]:
                h_bin[zeros[idx]] = 1.0
    elif num_ones > target:
        ones = np.where(h_bin == 1)[0]
        if len(ones) > 0:
            order = rng.permutation(len(ones))
            for idx in order[:num_ones - target]:
                h_bin[ones[idx]] = 0.0
    return project_feasible(h_bin, target_sum)


def transfer_grid(h_src, n_target):
    n_src = len(h_src)
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_tgt = (np.arange(n_target) + 0.5) * (2.0 / n_target)
    return np.interp(x_tgt, x_src, h_src)


def lp_round(h, dx, target_sum, n_iters, alpha, budget_s, t0, mode='soft'):
    n = len(h)
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    cur = h.copy()
    cur_c5 = best_c5
    for it in range(n_iters):
        if time.time() - t0 > budget_s - 15:
            break
        corrs = corr_of_h(cur)
        max_c = corrs.max()
        margin = max(0.02, 0.04 * max_c)
        n_top = max(20, min(60, int(0.3 * (2 * n - 1))))
        top_idx = np.argsort(corrs)[-n_top:]
        active_mask = corrs >= (max_c - margin)
        active_mask[top_idx] = True
        active_idx = np.where(active_mask)[0]
        if len(active_idx) < 5:
            active_idx = top_idx
        K = len(active_idx)
        if K > 80:
            weights = corrs[active_idx] - corrs[active_idx].min() + 0.1
            weights /= weights.max()
            sort_w = np.argsort(weights)[-60:]
            active_idx = active_idx[sort_w]
        direction = None
        if mode == 'soft':
            direction = solve_lp_soft(cur, dx, target_sum, active_idx, corrs, alpha, 0.3, 5.0, n)
        elif mode == 'hard':
            direction = solve_lp_hard(cur, dx, target_sum, active_idx, corrs, alpha, n)
        else:
            weights = corrs[active_idx]
            weights = (weights - weights.min() + 0.1) / max(weights.max() - weights.min(), 1e-10)
            direction = solve_lp_linear(cur, dx, target_sum, active_idx, corrs, alpha, weights, n)
        if direction is None:
            alpha *= 0.7
            if alpha < 0.005:
                break
            continue
        new_h, new_c5 = line_search(cur, direction, dx, target_sum, cur_c5)
        if new_c5 < cur_c5 - 1e-15:
            cur = new_h
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = cur.copy()
            alpha = min(0.35, alpha * 1.03)
        else:
            alpha *= 0.65
            if alpha < 0.005:
                break
    return best_h, best_c5


def perturb_reopt(h, dx, target_sum, rng, n_rounds, budget_s, t0):
    n = len(h)
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    for r in range(n_rounds):
        if time.time() - t0 > budget_s - 20:
            break
        scale = max(0.03, 0.15 / (r + 1))
        if r % 3 == 0:
            pert = rng.standard_normal(n)
        elif r % 3 == 1:
            pert = np.zeros(n)
            for _ in range(n // 10):
                idx = rng.randint(n)
                pert[idx] = rng.standard_normal()
        else:
            pert = np.sin(np.linspace(0, 3 * np.pi, n)) * rng.uniform(0.5, 1.5)
        pert -= pert.mean()
        h_p = project_feasible(h + scale * pert, target_sum)
        h_lp, c5_lp = lp_round(h_p, dx, target_sum, 80, 0.2, budget_s, t0, mode='soft')
        h_pr, c5_pr = pair_refine(h_lp, dx, target_sum, 200, rng, budget_s, t0)
        if c5_pr < best_c5:
            best_c5 = c5_pr
            best_h = h_pr.copy()
    return best_h, best_c5

    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = np.full(126, 0.5)

    n = len(h0)
    dx = 2.0 / n
    target_sum = n / 2.0
    h0 = project_feasible(h0, target_sum)
    best_h = h0.copy()
    best_c5 = c5_of_h(h0, dx)

    def remaining():
        return budget_s - (time.time() - t0)

    # Phase 1: LP optimization (soft mode)
    for alpha_start in [0.25, 0.15, 0.1, 0.06]:
        if remaining() < 30:
            break
        h_lp, c5_lp = lp_round(best_h, dx, target_sum, 120, alpha_start, budget_s, t0, mode='soft')
        if c5_lp < best_c5:
            best_c5 = c5_lp
            best_h = h_lp

    # Phase 2: Pair refinement
    if remaining() > 30:
        h_pr, c5_pr = pair_refine(best_h, dx, target_sum, 300, rng, budget_s, t0)
        if c5_pr < best_c5:
            best_c5 = c5_pr
            best_h = h_pr

    # Phase 3: Hard LP + pair
    if remaining() > 30:
        h_lp2, c5_lp2 = lp_round(best_h, dx, target_sum, 80, 0.15, budget_s, t0, mode='hard')
        if c5_lp2 < best_c5:
            best_c5 = c5_lp2
            best_h = h_lp2
        if remaining() > 20:
            h_pr2, c5_pr2 = pair_refine(best_h, dx, target_sum, 200, rng, budget_s, t0)
            if c5_pr2 < best_c5:
                best_c5 = c5_pr2
                best_h = h_pr2

    # Phase 4: Linear-sum LP + pair
    if remaining() > 30:
        h_lp3, c5_lp3 = lp_round(best_h, dx, target_sum, 80, 0.2, budget_s, t0, mode='linear')
        if c5_lp3 < best_c5:
            best_c5 = c5_lp3
            best_h = h_lp3
        if remaining() > 20:
            h_pr3, c5_pr3 = pair_refine(best_h, dx, target_sum, 150, rng, budget_s, t0)
            if c5_pr3 < best_c5:
                best_c5 = c5_pr3
                best_h = h_pr3

    # Phase 5: Binary rounding + re-optimize
    if remaining() > 60:
        for _ in range(3):
            if remaining() < 30:
                break
            h_bin = binary_round(best_h, target_sum, rng)
            h_bp, c5_bp = lp_round(h_bin, dx, target_sum, 100, 0.2, budget_s, t0, mode='soft')
            h_bp2, c5_bp2 = pair_refine(h_bp, dx, target_sum, 200, rng, budget_s, t0)
            if c5_bp2 < best_c5:
                best_c5 = c5_bp2
                best_h = h_bp2

    # Phase 6: Perturbation + re-optimization
    if remaining() > 80:
        h_pt, c5_pt = perturb_reopt(best_h, dx, target_sum, rng, 8, budget_s, t0)
        if c5_pt < best_c5:
            best_c5 = c5_pt
            best_h = h_pt

    # Phase 7: Interleaved LP + pair (multiple cycles)
    for cycle in range(6):
        if remaining() < 40:
            break
        mode = ['soft', 'hard', 'linear', 'soft', 'hard', 'soft'][cycle]
        alpha_c = [0.2, 0.15, 0.2, 0.1, 0.08, 0.05][cycle]
        h_c1, c5_c1 = lp_round(best_h, dx, target_sum, 60, alpha_c, budget_s, t0, mode=mode)
        h_c2, c5_c2 = pair_refine(h_c1, dx, target_sum, 150, rng, budget_s, t0)
        if c5_c2 < best_c5:
            best_c5 = c5_c2
            best_h = h_c2

    # Phase 8: Try different grid sizes
    n_base = len(best_h)
    n_candidates = []
    for dn in [-8, -4, -2, 2, 4, 8, 12, 16, 20, 24, 32]:
        nt = n_base + dn
        if 48 <= nt <= 220 and nt % 2 == 0:
            n_candidates.append(nt)
    for nt in [60, 64, 72, 80, 88, 96, 104, 112, 120, 128, 140, 144, 152, 168, 176, 184, 192, 200]:
        if nt not in n_candidates and 48 <= nt <= 220 and nt % 2 == 0:
            n_candidates.append(nt)
    n_candidates = sorted(set(n_candidates))

    for n_try in n_candidates:
        if remaining() < 35:
            break
        if n_try == n_base:
            continue
        dx_t = 2.0 / n_try
        ts_t = n_try / 2.0
        h_interp = transfer_grid(best_h, n_try)
        h_interp = project_feasible(h_interp, ts_t)
        h_opt, c5_opt = lp_round(h_interp, dx_t, ts_t, 60, 0.15, budget_s, t0, mode='soft')
        h_opt2, c5_opt2 = pair_refine(h_opt, dx_t, ts_t, 100, rng, budget_s, t0)
        if c5_opt2 < best_c5:
            best_c5 = c5_opt2
            best_h = h_opt2
            n_base = n_try

    # Phase 9: Final intensive polish
    n_f = len(best_h)
    dx_f = 2.0 / n_f
    ts_f = n_f / 2.0

    if remaining() > 25:
        h_f1, c5_f1 = lp_round(best_h, dx_f, ts_f, 100, 0.08, budget_s, t0, mode='soft')
        if c5_f1 < best_c5:
            best_c5 = c5_f1
            best_h = h_f1

    if remaining() > 15:
        h_f2, c5_f2 = pair_refine(best_h, dx_f, ts_f, 300, rng, budget_s, t0)
        if c5_f2 < best_c5:
            best_c5 = c5_f2
            best_h = h_f2

    if remaining() > 15:
        h_f3, c5_f3 = lp_round(best_h, dx_f, ts_f, 80, 0.04, budget_s, t0, mode='soft')
        if c5_f3 < best_c5:
            best_c5 = c5_f3
            best_h = h_f3

    if remaining() > 10:
        h_f4, c5_f4 = pair_refine(best_h, dx_f, ts_f, 200, rng, budget_s, t0)
        if c5_f4 < best_c5:
            best_c5 = c5_f4
            best_h = h_f4

    # Final validation
    n_final = len(best_h)
    dx_final = 2.0 / n_final
    ts_final = n_final / 2.0
    best_h = project_feasible(best_h, ts_final)
    best_c5 = c5_of_h(best_h, dx_final)

    return best_h, best_c5, n_final
