import numpy as np
from scipy.optimize import minimize
import time

def project_simplex_box(y, target):
    n = len(y)
    lo, hi = -3.0, 3.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        h = np.clip(y - mid, 0.0, 1.0)
        s = h.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] = float(np.clip(h[idx] + diff, 0.0, 1.0))
    return h

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def exact_score(h):
    return float(compute_profile(h).max())

def grad_row(h, m, n):
    dx = 2.0 / n
    d = m - (n - 1)
    if d == 0:
        return dx * (1.0 - 2.0 * h)
    elif d > 0:
        g = np.zeros(n)
        g[d:] += dx * (1.0 - h[:-d])
        g[:-d] -= dx * h[d:]
        return g
    else:
        p = -d
        g = np.zeros(n)
        g[:-p] += dx * (1.0 - h[p:])
        g[p:] -= dx * h[:-p]
        return g

def interp_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    edges_old = np.linspace(0.0, 2.0, n + 1)
    edges_new = np.linspace(0.0, 2.0, new_n + 1)
    h_new = np.zeros(new_n)
    for k in range(new_n):
        a, b = edges_new[k], edges_new[k + 1]
        total = 0.0
        for j in range(n):
            lo = max(a, edges_old[j])
            hi = min(b, edges_old[j + 1])
            if hi > lo:
                total += h[j] * (hi - lo)
        h_new[k] = total / (b - a)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_simplex_box(h_new, new_n / 2.0)

def active_descent(h, max_iters=300, active_tol=0.003, max_active=25):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_c = exact_score(h)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    momentum_vec = np.zeros(n)
    momentum = 0.35
    for it in range(max_iters):
        profile = compute_profile(h)
        c = float(profile.max())
        thresh = c - active_tol
        active = np.where(profile >= thresh)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(profile))])
        if len(active) > max_active:
            top = np.argsort(-profile[active])[:max_active]
            active = active[top]
        g = np.zeros(n)
        for m in active:
            g += grad_row(h, m, n)
        g -= g.mean()
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-14:
            break
        momentum_vec = momentum * momentum_vec + g
        mv_norm = np.linalg.norm(momentum_vec)
        if mv_norm < 1e-14:
            break
        d = -momentum_vec / mv_norm
        pos_mask = d > 1e-12
        neg_mask = d < -1e-12
        alpha_max = 1.0
        if pos_mask.any():
            alpha_max = min(alpha_max, float(((1.0 - h[pos_mask]) / d[pos_mask]).min()))
        if neg_mask.any():
            alpha_max = min(alpha_max, float((h[neg_mask] / (-d[neg_mask])).min()))
        alpha_max = max(alpha_max, 1e-10)
        alpha_max = min(alpha_max, 0.6)
        best_alpha = 0.0
        best_val = c
        best_h = h.copy()
        a_ls, b_ls = 0.0, alpha_max
        for _ in range(20):
            c1 = b_ls - gr * (b_ls - a_ls)
            c2 = a_ls + gr * (b_ls - a_ls)
            h1 = project_simplex_box(h + c1 * d, target)
            h2 = project_simplex_box(h + c2 * d, target)
            f1 = exact_score(h1)
            f2 = exact_score(h2)
            if f1 < best_val:
                best_val = f1
                best_alpha = c1
                best_h = h1.copy()
            if f2 < best_val:
                best_val = f2
                best_alpha = c2
                best_h = h2.copy()
            if f1 < f2:
                b_ls = c2
            else:
                a_ls = c1
            if b_ls - a_ls < 1e-12:
                break
        if best_val >= c - 1e-14:
            break
        h = best_h
        if best_val < best_c:
            best_c = best_val
    return h, best_c

def perturb_solution(h, rng, strength=0.08, n_swaps=6):
    n = len(h)
    target = n / 2.0
    h_new = h.copy()
    for _ in range(n_swaps):
        i, j = int(rng.integers(0, n)), int(rng.integers(0, n))
        if i == j:
            continue
        delta = strength * rng.uniform(0.5, 1.5)
        if rng.rand() < 0.5:
            move = min(delta, h_new[i], 1.0 - h_new[j])
            if move > 0:
                h_new[i] -= move
                h_new[j] += move
        else:
            move = min(delta, h_new[j], 1.0 - h_new[i])
            if move > 0:
                h_new[j] -= move
                h_new[i] += move
    return project_simplex_box(h_new, target)

def gaussian_perturb(h, rng, sigma=0.05):

def slsqp_epigraph(h_start, maxiter=150):
    n = len(h_start)
    dx = 2.0 / n
    M = 2 * n - 1
    c0 = exact_score(h_start)
    t0 = c0 + 1e-4
    x0 = np.concatenate([h_start, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 2.0)]
    
    def _obj(x):
        return x[-1]
    
    def _obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    
    def _eq_fun(x):
        return x[:-1].sum() - n / 2.0
    
    def _eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g
    
    def _ineq_fun(x):
        h = x[:-1]
        t = x[-1]
        profile = np.correlate(h, 1.0 - h, mode="full") * dx
        return t - profile
    
    def _ineq_jac(x):
        h = x[:-1]
        J = np.zeros((M, n + 1))
        for m in range(M):
            J[m, :n] = -grad_row(h, m, n)
        J[:, -1] = 1.0
        return J
    
    cons = [
        {'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac},
        {'type': 'ineq', 'fun': _ineq_fun, 'jac': _ineq_jac}
    ]
    try:
        res = minimize(_obj, x0, method='SLSQP', jac=_obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None and np.isfinite(res.x).all():
            h_new = project_simplex_box(res.x[:-1], n / 2.0)
            c_new = exact_score(h_new)
            if c_new <= c0 + 1e-8:
                return h_new
    except Exception:
        pass
    return None

                best_h, best_c = h_sol, c_sol
            h_ad, c_ad = active_descent(best_h, max_iters=100, active_tol=0.002)
            if c_ad < best_c:
                best_h, best_c = h_ad, c_ad

    # Phase 8: More ILS rounds if time permits
    for il in range(6):
        if time_left() < 30:
            break
        strength = 0.05 + 0.01 * il
        h_pert = perturb_solution(best_h, rng, strength=strength, n_swaps=5)
        h_ad, c_ad = active_descent(h_pert, max_iters=120, active_tol=0.003)
        if c_ad < best_c:
            best_h, best_c = h_ad, c_ad

    # Phase 9: Final aggressive polish
    if time_left() > 20:
        h_ad, c_ad = active_descent(best_h, max_iters=300, active_tol=0.001, max_active=30)
        if c_ad < best_c:
            best_h, best_c = h_ad, c_ad

    # Phase 10: Ultra-fine active descent
    if time_left() > 10:
        h_ad, c_ad = active_descent(best_h, max_iters=200, active_tol=0.0005, max_active=40)
        if c_ad < best_c:
            best_h, best_c = h_ad, c_ad

    best_h = project_simplex_box(best_h, len(best_h) / 2.0)
    final_c = exact_score(best_h)

    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            final_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 2:
            final_c = float(eval_result[0])
    except Exception:
        pass

    return best_h, final_c, len(best_h)
