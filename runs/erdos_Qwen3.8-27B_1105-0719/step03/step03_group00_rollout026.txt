import time

def project_feasible(x, target_sum):
    n = len(x)
    if not np.all(np.isfinite(x)):
        x = np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)

            h[mask] += diff / mask.sum()
        else:

            h += diff / n
            h = np.clip(h, 0.0, 1.0)

def c5_of_h(h, dx):
    return np.correlate(h, 1.0 - h, mode='full').max() * dx

def compute_pair_coefficients(h, p, q, n):
    """Compute linear coefficients for pair move (p,q).
    Returns (a, c_minus, c_plus) where:
    - new corr at lag 0: base + a*t - 2*t^2
    - new corr at lag -d: base + c_minus[d-1]*t
    - new corr at lag +d: base + c_plus[d-1]*t

    ds = np.arange(1, n)
    c_minus = np.zeros(n - 1)
    c_plus = np.zeros(n - 1)

    # = (1-h[p+d])_{p+d<n} - h[p-d]_{p-d>=0} - (1-h[q+d])_{q+d<n} + h[q-d]_{q-d>=0}
    m = ds + p < n
    c_minus[m] += (1.0 - h[p + ds[m]])
    m = p - ds >= 0
    c_minus[m] -= h[p - ds[m]]
    m = ds + q < n
    c_minus[m] -= (1.0 - h[q + ds[m]])
    m = q - ds >= 0
    c_minus[m] += h[q - ds[m]]

    # = (1-h[p-d])_{p-d>=0} - h[p+d]_{p+d<n} - (1-h[q-d])_{q-d>=0} + h[q+d]_{q+d<n}
    m = p - ds >= 0
    c_plus[m] += (1.0 - h[p - ds[m]])
    m = ds + p < n
    c_plus[m] -= h[p + ds[m]]
    m = q - ds >= 0
    c_plus[m] -= (1.0 - h[q - ds[m]])
    m = ds + q < n
    c_plus[m] += h[q + ds[m]]

    #   i.e., base[n-2-d+1]... 
    # Actually: base index n-1-d is lag -d, so:
    #   base[n-2] is lag -1, base[n-3] is lag -2, ..., base[0] is lag -(n-1)
    # base[n:] are lags +1, +2, ..., +(n-1)
    #   base[n] is lag +1, base[n+1] is lag +2, ..., base[2n-2] is lag +(n-1)

    # base[n-1-d] for d=1..n-1 are base[n-2], base[n-3], ..., base[0]
    # That's base[:n-1] in REVERSE order
    
    # For c_plus[d-1] (d=1..n-1), it modifies base[n-1+d]
    # base[n-1+d] for d=1..n-1 are base[n], base[n+1], ..., base[2n-2]
    # That's base[n:] in forward order
    
    left_base = base[:n-1][::-1]  # lags -1, -2, ..., -(n-1)
    right_base = base[n:]         # lags +1, +2, ..., +(n-1)

        left_vals = left_base + c_minus * t
        right_vals = right_base + c_plus * t
        results[i] = max(mid_val, left_vals.max(), right_vals.max()) * dx
    
    return results

        if sweep == 0:

        else:

            rng.shuffle(indices)
            pairs = [(int(i), int(j)) for i in indices for j in indices if i < j]
        
        base = np.correlate(best_h, 1.0 - best_h, mode='full')
        improved = False
        
        for p, q in pairs:
            hp, hq = best_h[p], best_h[q]
            L = max(-hp, hq - 1.0)
            U = min(1.0 - hp, hq)
            if L >= U:
                continue
            
            a, c_minus, c_plus = compute_pair_coefficients(best_h, p, q, n)
            
            # Sample t values
            t_vals = np.linspace(L, U, t_samples)
            # Also include a few random points
            t_rand = L + (U - L) * rng.uniform(size=4)
            t_vals = np.concatenate([t_vals, t_rand])
            
            scores = eval_pair_move(base, a, c_minus, c_plus, t_vals, n, dx)
            best_idx = scores.argmin()
            best_t = t_vals[best_idx]
            best_score = scores[best_idx]
            
            if best_score < best_c5 - 1e-12:
                # Apply the move
                best_h[p] += best_t
                best_h[q] -= best_t
                best_c5 = best_score
                improved = True
                # Recompute base
                base = np.correlate(best_h, 1.0 - best_h, mode='full')

            break

    target = n / 2.0
    best_h = project_feasible(best_h, target)
    best_c5 = c5_of_h(best_h, dx)
    
    return best_h, best_c5
