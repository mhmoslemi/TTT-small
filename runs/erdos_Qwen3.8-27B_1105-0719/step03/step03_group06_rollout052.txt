import numpy as np
from scipy.optimize import linprog
import time

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    # Replace non-finite
    h[~np.isfinite(h)] = 0.5
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(-(1.0 - h))
        remaining = diff
        for idx in order:
            if remaining <= 1e-14:
                break
            cap = 1.0 - h[idx]
            if cap <= 1e-15:
                continue
            add = min(remaining, cap)
            h[idx] += add
            remaining -= add
    else:
        order = np.argsort(h)
        remaining = -diff
        for idx in order:
            if remaining <= 1e-14:
                break
            if h[idx] <= 1e-15:
                continue
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
    # Final correction
    diff2 = target - h.sum()
    if abs(diff2) > 1e-13:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] = float(np.clip(h[idx] + diff2, 0.0, 1.0))
    return h

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))

def row_grad(h, m, n):
    """Gradient of profile row m with respect to h."""
    dx = 2.0 / n
    k = m - (n - 1)
    g = np.zeros(n)
    if k >= 0:

        # dP/dh[j] = (1-h[j-k]) for j>=k, minus h[j+k] for j<=n-1-k

            g[k:] += dx * (1.0 - h[:n-k])

        g[:n-d] += dx * (1.0 - h[d:])
        g[d:] -= dx * h[:n-d]
    return g

    if n == new_n:

    edges_new = np.linspace(0.0, 2.0, new_n + 1)

import numpy as np

def repair_feasible(vec):
    n = len(vec)
    h = np.asarray(vec, dtype=np.float64).copy()
    h[~np.isfinite(h)] = 0.5
    np.clip(h, 0.0, 1.0, out=h)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(-(1.0 - h))
        remaining = diff
        for idx in order:
            if remaining <= 1e-14:
                break
            cap = 1.0 - h[idx]
            if cap <= 1e-15:
                continue
            add = min(remaining, cap)
            h[idx] += add
            remaining -= add
    else:
        order = np.argsort(h)
        remaining = -diff
        for idx in order:
            if remaining <= 1e-14:
                break
            if h[idx] <= 1e-15:
                continue
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
    diff2 = target - h.sum()
    if abs(diff2) > 1e-13:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] = float(np.clip(h[idx] + diff2, 0.0, 1.0))
    return h

def row_grad(h, m, n):
    dx = 2.0 / n
    k = m - (n - 1)
    g = np.zeros(n)
    if k >= 0:
        if k == 0:
            g[:] = dx * (1.0 - 2.0 * h)
        else:
            g[k:] += dx * (1.0 - h[:n - k])
            g[:n - k] -= dx * h[k:]
    else:
        d = -k
        g[:n - d] += dx * (1.0 - h[d:])
        g[d:] -= dx * h[:n - d]
    return g

            hi = min(b, edges_old[j + 1])
            if hi > lo:

    elif pattern_type == 3:
        for i in range(n):
            if np.sin(2 * np.pi * 3.0 * i / n) > 0:
                h[i] = 1.0
        ones = int(h.sum())
        if ones > half:
            excess = ones - half
            one_idx = np.where(h == 1.0)[0]
            to_flip = rng.choice(one_idx, excess, replace=False)
            h[to_flip] = 0.0
        elif ones < half:
            deficit = half - ones
            zero_idx = np.where(h == 0.0)[0]
            to_flip = rng.choice(zero_idx, deficit, replace=False)
            h[to_flip] = 1.0
    else:
        perm = rng.permutation(n)
        h[perm[:half]] = 1.0
    return repair_feasible(h)

def smooth_mix(h_bin, blend=0.5):
    n = len(h_bin)
    h_smooth = np.zeros(n)
    for i in range(n):
        h_smooth[i] = (h_bin[i - 1] + 2.0 * h_bin[i] + h_bin[(i + 1) % n]) / 4.0
    return repair_feasible(blend * h_bin + (1.0 - blend) * h_smooth)

def crossover(ha, hb, rng):
    n = len(ha)
    assert len(hb) == n
    t = rng.uniform(0.2, 0.8)
    h_child = t * ha + (1.0 - t) * hb
    return repair_feasible(h_child)

def mutate(h, rng, radius=0.08):
    n = len(h)
    d = rng.uniform(-1.0, 1.0, size=n)
    d -= d.mean()
    h_new = h + radius * d
    return repair_feasible(h_new)

def coordinate_polish(h, max_iters=60, t_start=0.0, budget_s=1100.0, margin=5.0):
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    for it in range(max_iters):
        if not check_time(t_start, budget_s, margin):
            break
        P = full_profile(h)
        c = float(np.max(P))
        thresh = c - 0.003
        active = np.where(P >= thresh)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(P))])
        if len(active) > 15:
            top = np.argsort(P[active])[-15:]
            active = active[top]
        g = np.zeros(n)
        for m in active:
            g += row_grad(h, m, n)
        g -= g.mean()
        improved = False
        idx_desc = np.argsort(-h)
        idx_asc = np.argsort(h)
        for attempt in range(min(40, n)):
            i = int(idx_desc[attempt % n])
            j = int(idx_asc[attempt % n])
            if i == j:
                continue
            for delta_val in [0.03, 0.015, 0.005]:
                for sign in [1.0, -1.0]:
                    h_try = h.copy()
                    h_try[i] += sign * delta_val
                    h_try[j] -= sign * delta_val
                    if h_try.min() < -1e-10 or h_try.max() > 1 + 1e-10:
                        continue
                    h_try = repair_feasible(h_try)
                    s = exact_score(h_try)
                    if s < best_c - 1e-14:
                        h = h_try
                        best_c = s
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        if not improved:
            break
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()
    rng = np.random.RandomState(seed)

    try:
        parent = np.array(initial_h_values, dtype=np.float64)
    except Exception:
        parent = None
    if parent is None or len(parent) < 4:
        parent = np.full(70, 0.5)
    parent = repair_feasible(np.clip(parent, 0.0, 1.0))
    n_parent = len(parent)

    best_h = parent.copy()
    best_c = exact_score(parent)

    # Phase 1: Deep LP optimization of parent
    if check_time(t_start, budget_s, 200):
        h1, c1 = lp_trust_region(best_h, delta=0.08, max_iters=400,
                                  t_start=t_start, budget_s=budget_s, margin=180)
        if c1 < best_c:
            best_h, best_c = h1, c1
        h2, c2 = lp_trust_region(best_h, delta=0.03, max_iters=300,
                                  t_start=t_start, budget_s=budget_s, margin=160)
        if c2 < best_c:
            best_h, best_c = h2, c1 if c2 < best_c else c2

    # Phase 2: Dimension exploration
    n_best = len(best_h)
    dim_list = []
    for dn in [-8, -6, -4, -2, 2, 4, 6, 8, 10, 12, 14]:
        d = n_best + dn
        if 36 <= d <= 150:
            dim_list.append(d)
    dim_list.sort(key=lambda x: abs(x - n_best))

    for new_n in dim_list[:6]:
        if not check_time(t_start, budget_s, 120):
            break
        h_interp = interp_to_n(best_h, new_n)
        h_opt, c_opt = lp_trust_region(h_interp, delta=0.06, max_iters=250,
                                        t_start=t_start, budget_s=budget_s, margin=100)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt
            n_best = len(best_h)

    # Phase 3: Population-based search
    n_pop = len(best_h)
    pool = [(best_c, best_h.copy())]
    
    # Add diverse seeds
    for pt in range(4):
        if not check_time(t_start, budget_s, 80):
            break
        h_bin = binary_seed(n_pop, pt, rng)
        h_smooth = smooth_mix(h_bin, blend=0.6)
        h_opt, c_opt = lp_trust_region(h_smooth, delta=0.06, max_iters=150,
                                        t_start=t_start, budget_s=budget_s, margin=60)
        pool.append((c_opt, h_opt.copy()))
    
    # Add interpolated variants
    for pt in range(2):
        if not check_time(t_start, budget_s, 60):
            break
        src_n = n_pop + (4 if pt == 0 else -4)
        if src_n < 36 or src_n > 150:
            continue
        h_var = interp_to_n(best_h, src_n)
        h_var = mutate(h_var, rng, radius=0.1)
        h_opt, c_opt = lp_trust_region(h_var, delta=0.06, max_iters=150,
                                        t_start=t_start, budget_s=budget_s, margin=50)
        if len(h_opt) == n_pop:
            pool.append((c_opt, h_opt.copy()))
    
    # Sort pool
    pool.sort(key=lambda x: x[0])
    
    # Generations
    max_gens = 30
    for gen in range(max_gens):
        if not check_time(t_start, budget_s, 60):
            break
        # Pick parents
        if len(pool) >= 2:
            p1 = pool[0][1]
            # Pick second parent with some randomness
            pick_idx = min(1 + int(rng.exponential(1.5)), len(pool) - 1)
            p2 = pool[pick_idx][1]
            if len(p1) == len(p2):
                child = crossover(p1, p2, rng)
            else:
                child = interp_to_n(p2, len(p1))
                child = crossover(p1, child, rng)
        else:
            child = mutate(pool[0][1], rng, radius=0.1)
        
        # Mutate child
        child = mutate(child, rng, radius=0.05)
        
        # Optimize
        h_opt, c_opt = lp_trust_region(child, delta=0.05, max_iters=200,
                                        t_start=t_start, budget_s=budget_s, margin=50)
        pool.append((c_opt, h_opt.copy()))
        
        # Keep top-K diverse
        pool.sort(key=lambda x: x[0])
        max_pool = 12
        if len(pool) > max_pool:
            # Remove farthest from best in score, but keep diversity
            kept = [pool[0]]
            for item in pool[1:]:
                if len(kept) >= max_pool:
                    break
                # Check diversity
                diverse = True
                for ki in kept:
                    if len(item[1]) == len(ki[1]):
                        dist = np.linalg.norm(item[1] - ki[1])
                        if dist < 0.1 * np.sqrt(len(item[1])):
                            diverse = False
                            break
                if diverse or item[0] < kept[-1][0] - 0.005:
                    kept.append(item)
            pool = kept
        
        if pool[0][0] < best_c:
            best_c = pool[0][0]
            best_h = pool[0][1].copy()
    
    # Phase 4: Final fine-tuning
    if check_time(t_start, budget_s, 40):
        h_fine, c_fine = lp_trust_region(best_h, delta=0.02, max_iters=400,
                                          t_start=t_start, budget_s=budget_s, margin=30)
        if c_fine < best_c:
            best_h, best_c = h_fine, c_fine
    
    if check_time(t_start, budget_s, 30):
        h_fine, c_fine = lp_trust_region(best_h, delta=0.008, max_iters=300,
                                          t_start=t_start, budget_s=budget_s, margin=20)
        if c_fine < best_c:
            best_h, best_c = h_fine, c_fine
    
    if check_time(t_start, budget_s, 20):
        h_cp, c_cp = coordinate_polish(best_h, max_iters=80,
                                        t_start=t_start, budget_s=budget_s, margin=10)
        if c_cp < best_c:
            best_h, best_c = h_cp, c_cp
    
    if check_time(t_start, budget_s, 10):
        h_final, c_final = lp_trust_region(best_h, delta=0.003, max_iters=200,
                                            t_start=t_start, budget_s=budget_s, margin=5)
        if c_final < best_c:
            best_h, best_c = h_final, c_final

    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)

    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            final_c = min(final_c, float(eval_result))
        elif isinstance(eval_result, tuple) and len(eval_result) >= 2:
            final_c = min(final_c, float(eval_result[0]))
    except Exception:
        pass

    return best_h, final_c, len(best_h)
