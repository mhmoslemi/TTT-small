        else:

    dx = 2.0 / n
    corrs = np.correlate(h, 1.0 - h, mode='full')
    return corrs.max() * dx, corrs, dx

def grad_corr(h, d, n):
    grad = np.zeros(n)
    if d >= 0:
        m = n - d
        idx = np.arange(m)
        grad[idx] += (1.0 - h[idx + d])
        grad[idx + d] += (-h[idx])
    else:
        kk = -d
        idx = np.arange(kk, n)
        grad[idx] += (1.0 - h[idx - kk])
        grad[idx - kk] += (-h[idx])
    return grad

def solve_slsqp(h, dx, n, target_sum, t0, time_limit, n_active=40, maxiter=200):
    """Solve the epigraph SLSQP problem."""
    # Variable: z = [t, h_0, ..., h_{n-1}]
    nvar = n + 1
    
    # Get current corrs and active set
    c5, corrs, _ = c5_eval(h)
    # Active set: top n_active correlations
    m_total = 2 * n - 1
    n_act = min(n_active, m_total)
    top_idx = np.argsort(corrs)[-n_act:]
    active_offsets = top_idx - (n - 1)
    
    def objective(z):
        return z[0]
    
    def objective_grad(z):
        g = np.zeros(nvar)
        g[0] = 1.0
        return g
    
    def constraint_fun(z):
        t = z[0]
        hv = z[1:]
        # For each active offset, compute t - dx*corr
        vals = np.zeros(len(active_offsets))
        for i, d in enumerate(active_offsets):
            if d >= 0:
                vals[i] = t - dx * np.sum(hv[:n-d] * (1.0 - hv[d:]))
            else:
                kk = -d
                vals[i] = t - dx * np.sum(hv[kk:] * (1.0 - hv[:n-kk]))
        return vals
    
    def constraint_jac(z):
        t = z[0]
        hv = z[1:]
        J = np.zeros((len(active_offsets), nvar))
        J[:, 0] = 1.0
        for i, d in enumerate(active_offsets):
            J[i, 1:] = -dx * grad_corr(hv, d, n)
        return J

    def eq_fun(z):
        return np.array([z[1:].sum() - target_sum])
    
    def eq_fun_jac(z):
        J = np.zeros((1, nvar))
        J[0, 1:] = 1.0
        return J
    
    # Bounds
    bounds = [(0.0, min(1.0, c5 + 0.05))]  # t bound
    bounds += [(0.0, 1.0)] * n
    
    # Initial point
    z0 = np.zeros(nvar)
    z0[0] = c5 + 1e-6
    z0[1:] = h
    
    # Solve
    res = minimize(
        objective, z0, method='SLSQP',
        jac=objective_grad,
        constraints=[
            {'type': 'ineq', 'fun': constraint_fun, 'jac': constraint_jac},
            {'type': 'eq', 'fun': eq_fun, 'jac': eq_fun_jac}
        ],
        bounds=bounds,
        options={'maxiter': maxiter, 'ftol': 1e-10}
    )
    
    # Extract result

        h_new = project_feasible(res.x[1:], target_sum)
        c5_new = c5_eval(h_new)[0]
        if c5_new < c5 - 1e-10:

```python
import numpy as np
from scipy.optimize import minimize
import time

def project_feasible(x, target_sum):
    n = len(x)
    x = np.nan_to_num(x, nan=0.5)
    lo, hi = -3.0, 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        cnt = mask.sum()
        if cnt > 0:
            h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def c5_eval(h):
    n = len(h)
    dx = 2.0 / n
    corrs = np.correlate(h, 1.0 - h, mode='full')
    return corrs.max() * dx, corrs

def grad_corr(h, d, n):
    grad = np.zeros(n)
    if d >= 0:
        m = n - d
        idx = np.arange(m)
        grad[idx] += (1.0 - h[idx + d])
        grad[idx + d] += (-h[idx])
    else:
        kk = -d
        idx = np.arange(kk, n)
        grad[idx] += (1.0 - h[idx - kk])
        grad[idx - kk] += (-h[idx])
    return grad

def all_corrs_grads(h, n):
    """Compute gradients for all lags. Returns array of shape (2n-1, n)."""
    corrs = np.correlate(h, 1.0 - h, mode='full')
    m = 2 * n - 1
    G = np.zeros((m, n))
    for k in range(m):
        d = k - (n - 1)
        G[k] = grad_corr(h, d, n)
    return corrs, G

def smooth_max_obj_and_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corrs, G = all_corrs_grads(h, n)
    scaled = T * corrs
    m = scaled.max()
    exps = np.exp(scaled - m)
    w = exps / exps.sum()
    obj = (dx / T) * (m + np.log(exps.sum()))
    grad = dx * (G.T @ w)  # shape (n,)
    return obj, grad

def proj_gradient(h, target_sum, dx, T, n_max_iter=50000, step0=1.0, t0_time=None, time_limit=None):
    """Projected gradient method on smooth max."""
    n = len(h)
    h = h.copy()
    obj, grad = smooth_max_obj_and_grad(h, T)
    step = step0
    last_time = time.time()
    
    for it in range(n_max_iter):
        if time_limit is not None and (time.time() - t0_time > time_limit):
            break
        # Projected gradient step
        h_new = project_feasible(h - step * grad, target_sum)
        obj_new, grad_new = smooth_max_obj_and_grad(h_new, T)
        
        if obj_new < obj - 1e-15:
            h = h_new
            obj = obj_new
            grad = grad_new
            step = min(step * 1.05, 1.0)
        else:
            step *= 0.5
            if step < 1e-10:
                break
    
    return h, obj

def slsqp_optimize(h, dx, n, target_sum, n_active=50, maxiter=200):
    """SLSQP optimization with epigraph formulation."""
    c5_old, corrs = c5_eval(h)
    m_total = 2 * n - 1
    n_act = min(n_active, m_total)
    top_idx = np.argsort(corrs)[-n_act:]
    active_offsets = np.array([top_idx[i] - (n - 1) for i in range(n_act)])
    
    nvar = n + 1
    
    def obj(z):
        return z[0]
    
    def obj_grad(z):
        g = np.zeros(nvar)
        g[0] = 1.0
        return g
    
    def ineq_fun(z):
        t = z[0]
        hv = z[1:]
        vals = np.zeros(n_act)
        for i, d in enumerate(active_offsets):
            if d >= 0:
                vals[i] = t - dx * np.sum(hv[:n-d] * (1.0 - hv[d:]))
            else:
                kk = -d
                vals[i] = t - dx * np.sum(hv[kk:] * (1.0 - hv[:n-kk]))
        return vals
    
    def ineq_jac(z):
        hv = z[1:]
        J = np.zeros((n_act, nvar))
        J[:, 0] = 1.0
        for i, d in enumerate(active_offsets):
            J[i, 1:] = -dx * grad_corr(hv, d, n)
        return J
    
    def eq_fun(z):
        return np.array([z[1:].sum() - target_sum])
    
    def eq_jac(z):
        J = np.zeros((1, nvar))
        J[0, 1:] = 1.0
        return J
    
    bounds = [(0.0, c5_old + 0.02)] + [(0.0, 1.0)] * n
    z0 = np.concatenate(([c5_old + 1e-8], h))
    
    try:
        res = minimize(obj, z0, method='SLSQP', jac=obj_grad,
                      constraints=[
                          {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac},
                          {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}
                      ],
                      bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None:
            h_new = project_feasible(res.x[1:], target_sum)
            c5_new = c5_eval(h_new)[0]
            if c5_new < c5_old - 1e-12:
                return h_new, c5_new, True
    except Exception:
        pass
    return h, c5_old, False

def line_search_polish(h, target_sum, dx, n, n_dirs=20):
    """Custom line search along zero-sum directions."""
    rng = np.random.RandomState(42)
    c5_cur, corrs = c5_eval(h)
    
    for _ in range(n_dirs):
        # Random zero-mean direction
        v = rng.randn(n)
        v -= v.mean()
        # Scale to respect bounds

        max_neg = np.max(-v[v < 0]) if np.any(v < 0) else 1
        alpha_max = min(np.min(h[v > 0]) / max_pos if np.any(v > 0) else 1e6,
                       np.min(1 - h[v < 0]) / max_neg if np.any(v < 0) else 1e6)
        alpha_max = min(alpha_max, 0.1)
        if alpha_max < 1e-8:
            continue
        
        # Try several alpha values
        for alpha in [alpha_max * f for f in [1.0, 0.7, 0.5, 0.3, 0.1, 0.05]]:

            if c5_try < c5_cur - 1e-12:
