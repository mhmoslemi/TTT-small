import numpy as np
from scipy.optimize import minimize, linprog

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx] + 1e-15)
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx] + 1e-15)
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-14:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmin(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def build_gradient_matrix(h, n, dx):
    M = 2 * n - 1
    G = np.zeros((M, n))
    for m in range(M):
        d = m - (n - 1)
        if d == 0:
            G[m, :] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            G[m, d:] += dx * (1.0 - h[0:n-d])
            G[m, 0:n-d] += -dx * h[d:n]
        else:
            dp = -d
            G[m, 0:n-dp] += dx * (1.0 - h[dp:n])
            G[m, dp:n] += -dx * h[0:n-dp]
    return G

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = np.linspace(0, 2, n + 1)
    xt = np.linspace(0, 2, new_n + 1)
    h_interp = np.zeros(new_n)
    for k in range(new_n):
        lo_k, hi_k = xt[k], xt[k+1]
        total_mass = 0.0
        total_width = 0.0
        for j in range(n):
            lo_j, hi_j = xs[j], xs[j+1]
            overlap = min(hi_k, hi_j) - max(lo_k, lo_j)
            if overlap > 0:
                total_mass += h[j] * overlap
                total_width += overlap
        h_interp[k] = total_mass / total_width if total_width > 1e-15 else 0.5
    h_interp = np.clip(h_interp, 0.0, 1.0)
    return repair_feasible(h_interp)

def symmetrize(h):
    return (h + h[::-1]) / 2.0

def antisymmetrize(h):
    n = len(h)
    h_out = np.zeros(n)
    for i in range(n):
        h_out[i] = (h[i] + (1.0 - h[n-1-i])) / 2.0
    return repair_feasible(h_out)

def newton_lse(h, beta=80.0, max_iters=100, alpha_init=0.01, lambda_reg=1e-4):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best_c = exact_score(h)
    alpha = alpha_init
    stuck = 0
    for it in range(max_iters):
        profile = compute_profile(h)
        c = float(np.max(profile))
        if c + 1e-15 < best_c:
            best_c = c
            stuck = 0
        scaled = beta * profile
        max_s = np.max(scaled)
        shifted = scaled - max_s
        weights = np.exp(shifted)
        wsum = weights.sum()
        if wsum < 1e-300:
            break
        weights /= wsum
        G = build_gradient_matrix(h, n, dx)
        g = G.T @ weights
        g -= g.mean()
        W_minus = np.diag(weights) - np.outer(weights, weights)
        H = G.T @ W_minus @ G + lambda_reg * np.eye(n)
        H[0, 0] += lambda_reg
        try:
            s = solve(H, -g)
        except Exception:
            gnorm = np.linalg.norm(g)
            if gnorm < 1e-14:
                break
            s = -g / gnorm
        snorm = np.linalg.norm(s)
        if snorm < 1e-15:
            break
        improved = False
        for a in [alpha, alpha*0.5, alpha*0.25, alpha*0.1, alpha*0.05]:
            step = a * s
            if np.max(np.abs(step)) > 0.5:
                step = 0.5 * step / np.max(np.abs(step))
            h_try = repair_feasible(h + step)
            s_score = exact_score(h_try)
            if s_score < best_c - 1e-15:
                h = h_try
                best_c = s_score
                improved = True
                alpha = min(alpha * 1.05, 0.05)
                break
            elif s_score < c - 1e-15:
                h = h_try
                improved = True
                break
        if not improved:
            alpha *= 0.5
            stuck += 1
            if stuck > 30 or alpha < 1e-10:
                break
    return h, best_c

        return J
    x0 = np.concatenate([h0, [c0 + 1e-3]])
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 0.5)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None and np.isfinite(res.x).all():
            h_new = repair_feasible(res.x[:-1])
            c_new = exact_score(h_new)
            if c_new <= c0 + 1e-9:
                return h_new, c_new
    except Exception:
        pass
    return None, None

def random_perturb(h, rng, scale=0.02):
    n = len(h)
    z = rng.randn(n)
    z -= z.mean()
    return repair_feasible(h + scale * z)

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s - (time.perf_counter() - t_start)

    parent = None
    try:
        parent = np.array(initial_h_values, dtype=np.float64)
    except NameError:
        pass
    except Exception:
        pass

    if parent is None or len(parent) < 4:
        parent = np.full(60, 0.5)

    parent = np.clip(parent, 0.0, 1.0)
    parent = repair_feasible(parent)
    best_h = parent.copy()
    best_c = exact_score(parent)

    h_sym = repair_feasible(symmetrize(parent))
    c_sym = exact_score(h_sym)
    if c_sym < best_c - 1e-12:
        best_h = h_sym
        best_c = c_sym

    h_asym = antisymmetrize(parent)
    c_asym = exact_score(h_asym)
    if c_asym < best_c - 1e-12:
        best_h = h_asym
        best_c = c_asym

    n_parent = len(best_h)

    # Phase 1: Newton LSE at parent resolution
    if time_left() > 150:
        h_new, c_new = newton_lse(best_h, beta=80, max_iters=80, alpha_init=0.01)
        if c_new < best_c - 1e-14:
            best_h = h_new
            best_c = c_new
        h_new, c_new = active_set_descent(best_h, max_iters=60, n_active=10)
        if c_new < best_c - 1e-14:
            best_h = h_new
            best_c = c_new

    # Phase 2: Resolution scan
    if time_left() > 200:
        scan_results = []
        for factor in [0.7, 0.75, 0.8, 0.85, 0.9, 0.95, 1.0, 1.05, 1.1, 1.15, 1.2, 1.3, 1.4]:
            d = int(round(n_parent * factor / 2)) * 2
            d = max(24, min(d, 160))
            scan_results.append(d)
        scanResults = list(set(scan_results))
        scanResults.sort(key=lambda x: abs(x - n_parent))

        for new_n in scanResults:
            if time_left() < 80:
                break
            h_interp = interpolate_to_n(best_h, new_n)
            h_new, c_new = newton_lse(h_interp, beta=100, max_iters=50, alpha_init=0.008)
            if c_new < best_c - 1e-14:
                best_h = h_new
                best_c = c_new
            h_new, c_new = active_set_descent(best_h, max_iters=40, n_active=10)
            if c_new < best_c - 1e-14:
                best_h = h_new
                best_c = c_new

    # Phase 3: LP direction at best resolution
    if time_left() > 150:
        h_new, c_new = lp_direction(best_h, max_iters=25, n_active=8)
        if c_new < best_c - 1e-14:
            best_h = h_new
            best_c = c_new
        h_new, c_new = newton_lse(best_h, beta=100, max_iters=50, alpha_init=0.005)
        if c_new < best_c - 1e-14:
            best_h = h_new
            best_c = c_new

    # Phase 4: SLSQP
    if time_left() > 100:
        h_sol, c_sol = slsqp_solve(best_h, maxiter=150)
        if h_sol is not None and c_sol < best_c - 1e-12:
            best_h = h_sol
            best_c = c_sol
            h_new, c_new = newton_lse(best_h, beta=120, max_iters=40, alpha_init=0.003)
            if c_new < best_c - 1e-14:
                best_h = h_new
                best_c = c_new

    # Phase 5: Multi-start with perturbations at best resolution
    if time_left() > 200:
        n_b = len(best_h)
        starts = [best_h.copy()]
        for eps in [0.005, 0.01, 0.015, 0.025, 0.04, 0.06]:
            starts.append(random_perturb(best_h, rng, eps))
        z = rng.randn(n_b)
        z_sym = symmetrize(z)
        z_sym -= z_sym.mean()
        starts.append(repair_feasible(best_h + 0.015 * z_sym))
        z_as = rng.randn(n_b)
        z_as[n_b//2:] = -z_as[:n_b//2]
        z_as -= z_as.mean()
        starts.append(repair_feasible(best_h + 0.01 * z_as))

        for h0 in starts:
            if time_left() < 50:
                break
            h_new, c_new = newton_lse(h0, beta=100, max_iters=40, alpha_init=0.008)
            if c_new < best_c - 1e-13:
                best_h = h_new
                best_c = c_new
            h_new, c_new = lp_direction(best_h, max_iters=15, n_active=6)
            if c_new < best_c - 1e-13:
                best_h = h_new
                best_c = c_new

    # Phase 6: Resolution fine-tuning near best
    if time_left() > 150:
        n_best = len(best_h)
        for dn in [-4, -3, -2, -1, 1, 2, 3, 4, 5, 6, 8]:
            if time_left() < 30:
                break
            new_n = n_best + dn
            if new_n < 20 or new_n > 200 or new_n % 2 != 0:
                continue
            h_interp = interpolate_to_n(best_h, new_n)
            h_new, c_new = newton_lse(h_interp, beta=120, max_iters=40, alpha_init=0.006)
            if c_new < best_c - 1e-14:
                best_h = h_new
                best_c = c_new
            h_new, c_new = active_set_descent(best_h, max_iters=30, n_active=8)
            if c_new < best_c - 1e-14:
                best_h = h_new
                best_c = c_new

    # Phase 7: Extended Newton with varying beta
    for beta in [50, 100, 150, 250]:
        if time_left() < 30:
            break
        h_new, c_new = newton_lse(best_h, beta=beta, max_iters=60, alpha_init=0.004)
        if c_new < best_c - 1e-14:
            best_h = h_new
            best_c = c_new

    # Phase 8: Active set with varying sizes
    for n_act in [5, 7, 10, 14, 20]:
        if time_left() < 25:
            break
        h_new, c_new = active_set_descent(best_h, max_iters=80, n_active=n_act, alpha_init=0.003)
        if c_new < best_c - 1e-14:
            best_h = h_new
            best_c = c_new

    # Phase 9: LP + Newton cycling
    if time_left() > 60:
        for cycle in range(5):
            if time_left() < 20:
                break
            h_new, c_new = lp_direction(best_h, max_iters=15, n_active=6)
            if c_new < best_c - 1e-14:
                best_h = h_new
                best_c = c_new
            h_new, c_new = newton_lse(best_h, beta=150, max_iters=30, alpha_init=0.003)
            if c_new < best_c - 1e-14:
                best_h = h_new
                best_c = c_new

    # Phase 10: Final SLSQP polish
    if time_left() > 50:
        h_sol, c_sol = slsqp_solve(best_h, maxiter=200)
        if h_sol is not None and c_sol < best_c - 1e-14:
            best_h = h_sol
            best_c = c_sol

    # Phase 11: Final active set + Newton
    if time_left() > 20:
        h_new, c_new = active_set_descent(best_h, max_iters=100, n_active=8, alpha_init=0.002)
        if c_new < best_c - 1e-15:
            best_h = h_new
            best_c = c_new
        h_new, c_new = newton_lse(best_h, beta=200, max_iters=80, alpha_init=0.002, lambda_reg=1e-5)
        if c_new < best_c - 1e-15:
            best_h = h_new
            best_c = c_new

    # Final validation
    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)

    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            final_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 2:
            final_c = float(eval_result[0])
    except Exception:
        pass

    return best_h, final_c, len(best_h)
