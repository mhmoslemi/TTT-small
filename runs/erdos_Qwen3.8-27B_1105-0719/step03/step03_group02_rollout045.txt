For reducing corr_k: corr_k = sum_i h[i]*(1-h[i+k]). To reduce this, we want to decrease h[i] where h[i+k] is small, or increase h[i+k] where h[i] is large. The gradient of corr_k w.r.t. h is:
- For coordinate i: (1-h[i+k]) - h[i-k] (with boundary handling)

So the direction that reduces corr_k is -grad_corr_k, projected to have sum=0.

def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)


def c5_of_h(h, dx):
    return np.correlate(h, 1.0 - h, mode='full').max() * dx


def corr_max_1d(h, d, a):
    h_a = h + a * d
    return np.correlate(h_a, 1.0 - h_a, mode='full').max()


def exact_line_search(h, d, dx, target_sum, n_grid=60, n_golden=35):
    n = len(h)
    pos = d > 1e-15
    neg = d < -1e-15
    if not pos.any() and not neg.any():
        return h, c5_of_h(h, dx)

    lo = 0.0
    hi = np.inf
    if pos.any():
        hi = min(hi, np.min((1.0 - h[pos]) / d[pos]))
    if neg.any():
        hi = min(hi, np.min(h[neg] / (-d[neg])))
        lo = max(lo, np.max((h[neg] - 1.0) / (-d[neg])))

    lo = max(lo, 0.0)
    if hi <= lo or hi < 1e-15:
        return h, c5_of_h(h, dx)

    base_c5 = c5_of_h(h, dx)
    alphas = np.linspace(0.0, hi, n_grid)
    best_a = 0.0
    best_v = base_c5
    for i in range(1, n_grid):
        a = alphas[i]
        v = corr_max_1d(h, d, a)
        if v < best_v:
            best_v = v
            best_a = a

    if best_a < 1e-15:
        return h, base_c5

    margin = max(hi / n_grid, 1e-8)
    a_lo = max(0.0, best_a - margin)
    a_hi = min(hi, best_a + margin)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    gc = a_hi - gr * (a_hi - a_lo)
    gd = a_lo + gr * (a_hi - a_lo)
    fc = corr_max_1d(h, d, gc)
    fd = corr_max_1d(h, d, gd)

    for _ in range(n_golden):
        if a_hi - a_lo < 1e-12:
            break
        if fc < fd:
            a_hi = gd
            gd = gc
            fd = fc
            gc = a_hi - gr * (a_hi - a_lo)
            fc = corr_max_1d(h, d, gc)
        else:
            a_lo = gc
            gc = gd
            fc = fd
            gd = a_lo + gr * (a_hi - a_lo)
            fd = corr_max_1d(h, d, gd)

    best_a = (a_lo + a_hi) / 2
    h_new = project_feasible(h + best_a * d, target_sum)
    c5_new = c5_of_h(h_new, dx)
    if c5_new < base_c5 - 1e-15:
        return h_new, c5_new
    return h, base_c5


def grad_smooth_max(h, dx, beta):
    n = len(h)
    corrs = np.correlate(h, 1.0 - h, mode='full')
    shifted = corrs - corrs.max()
    w = np.exp(beta * shifted)
    ws = w.sum()
    if ws < 1e-30:
        w = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        w /= ws
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] > 1e-14:
            offset = k - (n - 1)
            i_lo = max(0, -offset)
            i_hi = min(n - 1, n - 1 - offset)
            if i_lo <= i_hi:
                idx = np.arange(i_lo, i_hi + 1)
                wk = w[k]
                grad[idx + offset] += wk * (1.0 - h[idx])
                grad[idx] -= wk * h[idx + offset]
    return grad * dx


def zero_sum(d, n):
    d = d - d.mean()
    return d


def make_lp_direction(h, dx, target_sum, n_active=40):
    n = len(h)
    corrs = np.correlate(h, 1.0 - h, mode='full')
    max_c = corrs.max()
    tol = max_c * 0.95
    active = np.where(corrs >= tol)[0]
    if len(active) > n_active:
        active = np.argsort(corrs)[-n_active:]

    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_rows = []
    b_rows = []
    for k in active:
        offset = k - (n - 1)
        i_lo = max(0, -offset)
        i_hi = min(n - 1, n - 1 - offset)
        row = np.zeros(n + 1)
        if i_lo <= i_hi:
            idx = np.arange(i_lo, i_hi + 1)
            row[idx + offset] += (1.0 - h[idx])
            row[idx] -= h[idx + offset]
        row[:n] *= dx
        row[n] = -1.0
        A_rows.append(row)
        b_rows.append(-corrs[k] * dx - 1e-13)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    bounds = [(max(-h[i], -0.3), min(1.0 - h[i], 0.3)) for i in range(n)]
    bounds.append((0.0, None))

    try:
        res = linprog(c_obj, A_ub=np.array(A_rows), b_ub=np.array(b_rows),
                      A_eq=A_eq, b_eq=np.zeros(1), bounds=bounds, method='highs')
        if res.status == 0:
            return res.x[:n]
    except Exception:
        pass
    return None


def pair_exact_search(h, paper_i, paper_j, dx, target_sum):
    n = len(h)
    d = np.zeros(n)
    d[paper_i] = -1.0
    d[paper_j] = 1.0
    return exact_line_search(h, d, dx, target_sum, n_grid=50, n_golden=30)


def block_direction(h, n, block_size, rng):
    d = np.zeros(n)
    start1 = rng.randint(0, n - block_size)
    start2 = rng.randint(0, n - block_size)
    if start1 == start2:
        return None
    d[start1:start1 + block_size] = -1.0 / block_size
    d[start2:start2 + block_size] = 1.0 / block_size
    return zero_sum(d, n)


def gradient_descent_exact(h0, dx, target_sum, max_iter=2000, beta=50.0):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    no_improve = 0
    lr = 0.8

    for it in range(max_iter):
        grad = grad_smooth_max(h, dx, beta)
        grad = zero_sum(grad, n)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:

def structured_pair_polish(h, dx, target_sum, max_rounds=5):
    n = len(h)
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)

    for rnd in range(max_rounds):
        corrs = np.correlate(h, 1.0 - h, mode='full')
        max_k = np.argmax(corrs)
        offset = max_k - (n - 1)
        i_lo = max(0, -offset)
        i_hi = min(n - 1, n - 1 - offset)
        idx = np.arange(i_lo, i_hi + 1)
        contribution = h[idx] * (1.0 - h[idx + offset])
        top_pairs = idx[np.argsort(contribution)[-15:]]

        improved = False
        for i in top_pairs:

            if 0 <= j < n:
                h_new, c5_new = pair_exact_search(h, i, j, dx, target_sum)
                if c5_new < best_c5 - 1e-14:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h = h_new
                    improved = True
        if not improved:

    return best_h, best_c5

def try_resample(h_best, n_new, time_left_fn):
    n_old = len(h_best)
    dx_new = 2.0 / n_new
    ts_new = n_new / 2.0
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False) + 1.0 / n_old
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False) + 1.0 / n_new
    h_interp = np.interp(x_new, x_old, h_best, left=0.0, right=0.0)
    h_interp = project_feasible(h_interp, ts_new)
    h_opt, c5_opt = full_polish(h_interp, dx_new, ts_new, time_left_fn, f"n{n_new}")
    return h_opt, c5_opt, n_new
