= (1-h[m-k]) * [indicator that m-k is a valid n] + (-h[m+k]) * [indicator that m is a valid n]

Where "m-k is valid" means 0 <= m-k < n AND 0 <= (m-k)+k < n, i.e., 0 <= m-k < n AND 0 <= m < n.
And "m is valid" means 0 <= m < n AND 0 <= m+k < n.

So:
∂C_p/∂h[m] = (1-h[m-k]) * I[0 <= m-k < n] * I[0 <= m < n] + (-h[m+k]) * I[0 <= m < n] * I[0 <= m+k < n]

Since we only compute for 0 <= m < n (the length of h), the second I[0<=m<n] is always true:
∂C_p/∂h[m] = (1-h[m-k]) * I[0 <= m-k < n] + (-h[m+k]) * I[0 <= m+k < n]

For k = 0:
∂C_p/∂h[m] = (1-h[m]) + (-h[m]) = 1 - 2h[m]

For k > 0:
- First term: (1-h[m-k]) when 0 <= m-k < n, i.e., k <= m <= n-1
- Second term: -h[m+k] when 0 <= m+k < n, i.e., 0 <= m <= n-1-k

For k < 0 (let d = -k > 0):
- First term: (1-h[m+d]) when 0 <= m+d < n, i.e., 0 <= m <= n-1-d
- Second term: -h[m-d] when 0 <= m-d < n, i.e., d <= m <= n-1

def lse_gradient(h, T):
    n = len(h)
    corrs = np.correlate(h, 1.0 - h, mode='full')
    m = corrs.max()
    w = np.exp((corrs - m) / T)
    w /= w.sum()
    
    grad = np.zeros(n)
    num_lags = 2 * n - 1
    for p in range(num_lags):
        wp = w[p]
        if wp < 1e-15:
            continue
        k = p - (n - 1)
        if k == 0:
            grad += wp * (1.0 - 2.0 * h)
        elif k > 0:
            # First term: (1-h[m-k]) for m in [k, n-1]
            grad[k:] += wp * (1.0 - h[:n-k])
            # Second term: -h[m+k] for m in [0, n-1-k]
            grad[:n-k] += -wp * h[k:]
        else:
            d = -k
            # First term: (1-h[m+d]) for m in [0, n-1-d]
            grad[:n-d] += wp * (1.0 - h[d:])
            # Second term: -h[m-d] for m in [d, n-1]
            grad[d:] += -wp * h[:n-d]
    return grad

            continue
        h_trial = project_feasible(h + a * d, target_sum)
        c5_trial = c5_exact(h_trial, dx)
        if c5_trial < best_c5 - 1e-15:
            best_c5 = c5_trial
            best_h = h_trial
    
    return best_h, best_c5

def lse_anneal(h, dx, target_sum, t0, max_time, T_start=3.0, T_end=0.03, momentum=0.8, init_step=0.05):
    """Main LSE annealing optimizer."""
    n = len(h)
    best_h = h.copy()
    best_c5 = c5_exact(h, dx)
    v = np.zeros(n)
    step = init_step
    no_improve = 0
    
    step_id = 0
    total_steps = 50000  # will be limited by time
    
    while time.time() - t0 < max_time:
        # Temperature schedule: geometric from T_start to T_end
        frac = min(step_id / total_steps, 1.0)
        T = T_start * (T_end / T_start) ** frac
        
        d = lse_direction(h, T)
        if d is None:
            break
        
        # Momentum
        v = momentum * v + d
        v_norm = np.linalg.norm(v)
        if v_norm > 1e-15:
            d = v / v_norm
        
        h_new, c5_new = line_search(h, d, target_sum, dx, step)
        
        if c5_new < c5_exact(h, dx) - 1e-15:
            h = h_new
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
            step = min(step * 1.01, 0.2)
        else:
            no_improve += 1
            v *= 0.5  # reduce momentum
            step *= 0.95
            if step < 0.002:
                step = 0.01
                no_improve = 0  # reset
        
        if no_improve > 500 and T < 0.1:
            break
        
        step_id += 1
    
    return best_h, best_c5

                h[j] += amt
                no_improve += 1

        if no_improve > 3000 and temp < 1e-5:
