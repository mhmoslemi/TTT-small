import numpy as np
from scipy.optimize import linprog
import time


def exact_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx


def exact_score(h):
    return float(np.max(exact_profile(h)))


def repair_feasible(h):
    n = len(h)
    h = np.clip(np.asarray(h, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-13:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        val = np.clip(h + mid, 0.0, 1.0).sum()
        if val < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for i in range(n):
            idx = order[i]
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
            else:
                add = min(diff, -h[idx])
            h[idx] += add
            diff -= add
            if abs(diff) < 1e-13:
                break
    return np.clip(h, 0.0, 1.0)


def build_gradient(h):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    G = np.zeros((M, n))
    for m in range(M):
        d = m - (n - 1)
        if d == 0:
            G[m, :] = dx * (1.0 - 2.0 * h)
        else:
            r = abs(d)
            for j in range(n):
                if j >= r and j + r < n:
                    G[m, j] = dx * (1.0 - h[j - r]) - dx * h[j + r]
                elif j >= r and j + r >= n:
                    G[m, j] = dx * (1.0 - h[j - r])
                elif j < r and j + r < n:
                    G[m, j] = -dx * h[j + r]
    return G


def solve_lp_direction(h, D):
    n = len(h)
    M = 2 * n - 1
    P = exact_profile(h)
    G = build_gradient(h)
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_ub = np.hstack([G, -np.ones((M, 1))])
    b_ub = -P
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(-D, D)] * n + [(None, None)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and np.isfinite(res.x).all():
            return res.x[:n], float(res.x[n])
    except Exception:
        pass
    return None, None


def line_search_direction(h, direction, n_grid=80):
    n = len(h)
    lam_pos = 1.0
    lam_neg = 1.0
    if np.any(direction > 1e-15):
        lam_pos = float(np.min((1.0 - h)[direction > 1e-15] / direction[direction > 1e-15]))
    if np.any(direction < -1e-15):
        lam_neg = float(np.min(h[direction < -1e-15] / (-direction[direction < -1e-15])))
    lam_pos = min(lam_pos, 2.0 / max(np.max(np.abs(direction)), 1e-15))
    lam_neg = min(lam_neg, 2.0 / max(np.max(np.abs(direction)), 1e-15))
    best_lam = 0.0
    best_s = exact_score(h)
    lams = np.linspace(-lam_neg, lam_pos, n_grid)
    for lam in lams:
        h_new = h + lam * direction
        if np.min(h_new) < -1e-9 or np.max(h_new) > 1.0 + 1e-9:
            continue
        s = exact_score(h_new)
        if s < best_s:
            best_s = s
            best_lam = lam
    if best_lam != 0.0:
        margin = (lam_pos - lam_neg) / n_grid
        lo = max(-lam_neg, best_lam - 3 * margin)
        hi = min(lam_pos, best_lam + 3 * margin)
        lams2 = np.linspace(lo, hi, 40)

                continue
            s = exact_score(h_new)
            if s < best_s:
                best_s = s
                best_lam = lam
    h_best = repair_feasible(h + best_lam * direction)
    return h_best, best_s


def pair_optimize(h, n_trials, rng):
    n = len(h)
    best_h = h.copy()
    best_c = exact_score(h)
    for _ in range(n_trials):

        j = rng.randint(n)

            continue

                s = exact_score(h_try)
                if s < best_c - 1e-13:
                    best_c = s
                    best_h = h_try.copy()
    return best_h, best_c


def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    dx_old = 2.0 / n
    dx_new = 2.0 / new_n
    h_new = np.zeros(new_n)
    for k in range(new_n):
        lo = k * dx_new
        hi = (k + 1) * dx_new
        total = 0.0
        for j in range(n):
            lo_j = j * dx_old
            hi_j = (j + 1) * dx_old
            overlap = min(hi, hi_j) - max(lo, lo_j)
            if overlap > 0:
                total += h[j] * overlap
        h_new[k] = total / dx_new
    return repair_feasible(np.clip(h_new, 0.0, 1.0))


def gradient_pair_direction(h, n_pairs, rng):
    n = len(h)
    P = exact_profile(h)
    G = build_gradient(h)
    c = float(np.max(P))
    margin = max(0.003, 0.3 * (c - float(np.median(P))))
    active = np.where(P >= c - margin)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(P))])
    grad_avg = G[active, :].mean(axis=0)
    grad_avg -= grad_avg.mean()
    best_h = h.copy()
    best_c = c
    coords = np.argsort(np.abs(grad_avg))[::-1][:min(30, n)]
    for _ in range(n_pairs):
        i = coords[rng.randint(len(coords))]
        j = coords[rng.randint(len(coords))]
        if i == j:
            continue
        if grad_avg[i] * grad_avg[j] > 0:
            continue
        for eps in [0.01, 0.03, 0.06, 0.1]:
            for sign in [1, -1]:
                h_try = best_h.copy()
                h_try[i] += sign * eps
                h_try[j] -= sign * eps
                if h_try[i] < -1e-10 or h_try[i] > 1.0 + 1e-10:
                    continue
                if h_try[j] < -1e-10 or h_try[j] > 1.0 + 1e-10:
                    continue
                h_try = repair_feasible(h_try)
                s = exact_score(h_try)
                if s < best_c - 1e-13:
                    best_c = s
                    best_h = h_try.copy()
    return best_h, best_c
