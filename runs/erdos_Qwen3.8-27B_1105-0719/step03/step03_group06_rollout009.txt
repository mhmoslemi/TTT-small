Let me think about the LSE approach more carefully:

The objective is f(h) = max_m p_m(h)

The LSE approximation is: f_beta(h) = (1/beta) * log(sum_m exp(beta * p_m(h)))

As beta → ∞, f_beta → f. The gradient of f_beta is:
∇f_beta = (1/beta) * sum_m exp(beta*p_m) * ∇p_m / sum_m exp(beta*p_m)
         = sum_m w_m * ∇p_m
where w_m = exp(beta*p_m) / sum_j exp(beta*p_j)

This is a convex combination of the gradient rows, weighted by the "soft-max" weights. This naturally handles the case where multiple rows are near-active.

    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        # Need to add mass: add to those with most room
        order = np.argsort(-(1.0 - h))
        remaining = diff
        for i in range(n):
            idx = order[i]
            cap = 1.0 - h[idx]
            if cap <= 1e-15:
                continue
            add = min(remaining, cap)
            h[idx] += add
            remaining -= add
            if remaining <= 1e-14:
                break
    else:
        # Need to remove mass: remove from those with most mass
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            if h[idx] <= 1e-15:
                continue
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-14:

    # Fix any residual

        free = np.where((h > 1e-10) & (h < 1.0 - 1e-10))[0]

def exact_score(h):

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def grad_row(h, m, n):
    dx = 2.0 / n
    d = m - (n - 1)
    if d == 0:
        return dx * (1.0 - 2.0 * h)
    elif d > 0:
        g = np.zeros(n)
        g[d:] += dx * (1.0 - h[:-d])
        g[:-d] -= dx * h[d:]
        return g
    else:
        p = -d
        g = np.zeros(n)
        g[:-p] += dx * (1.0 - h[p:])
        g[p:] -= dx * h[:-p]
        return g

def lse_gradient(h, beta):

    n = len(h)
    p = compute_profile(h)
    p_max = p.max()
    z = beta * (p - p_max)

    exp_z = np.exp(z)
    w = exp_z / exp_z.sum()

    g = np.zeros(n)
    for m in range(len(p)):
        if w[m] > 1e-10:
            g += w[m] * grad_row(h, m, n)
    return g

    n = len(h)

            hi = min(b, edges_old[j + 1])
            if hi > lo:

def trust_region_lp(h, radius=0.05):
    """Single trust-region LP step."""
    n = len(h)
    M = 2 * n - 1
    p = compute_profile(h)
    
    # Build constraint matrix: p_m + G[m,:] @ d <= t
    # Variables: [d_0, ..., d_{n-1}, t]
    # A_ub @ x <= b_ub
    A_rows = []
    b_rows = []
    for m in range(M):
        gm = grad_row(h, m, n)
        row = np.zeros(n + 1)
        row[:n] = gm  # G[m,:] @ d
        row[n] = -1.0  # -t
        A_rows.append(row)
        b_rows.append(-p[m])  # p_m + G@ d - t <= 0  => G@d - t <= -p_m
    A_ub = np.array(A_rows)
    b_ub = np.array(b_rows)
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    # Bounds: |d_j| <= radius, t free (but bounded)
    c_max = p.max()
    bounds = [(max(-radius, -h[j]), min(radius, 1.0 - h[j])) for j in range(n)]
    bounds.append((0.0, c_max + radius * 2.0))
    
    # Objective: minimize t
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x is not None:
            d = res.x[:n]
            h_new = repair_feasible(h + d)
            c_new = exact_score(h_new)
            if c_new < exact_score(h) - 1e-14:
                return h_new, c_new
    except Exception:
        pass
    return None, None

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(-(1.0 - h))
        remaining = diff
        for i in range(n):
            idx = order[i]
            cap = 1.0 - h[idx]
            if cap <= 1e-15:
                continue
            add = min(remaining, cap)
            h[idx] += add
            remaining -= add
            if remaining <= 1e-14:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            if h[idx] <= 1e-15:
                continue
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-14:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        free = np.where((h > 1e-10) & (h < 1.0 - 1e-10))[0]
        if len(free) > 0:
            h[free] += diff2 / len(free)
            h = np.clip(h, 0.0, 1.0)
        else:
            idx = int(np.argmax(1.0 - h)) if diff2 > 0 else int(np.argmax(h))
            h[idx] = float(np.clip(h[idx] + diff2, 0.0, 1.0))
    return h

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def grad_row(h, m, n):
    dx = 2.0 / n
    d = m - (n - 1)
    if d == 0:
        return dx * (1.0 - 2.0 * h)
    elif d > 0:
        g = np.zeros(n)
        g[d:] += dx * (1.0 - h[:-d])
        g[:-d] -= dx * h[d:]
        return g
    else:
        p = -d
        g = np.zeros(n)
        g[:-p] += dx * (1.0 - h[p:])
        g[p:] -= dx * h[:-p]
        return g

def lse_grad(h, beta):
    n = len(h)
    p = compute_profile(h)
    p_max = p.max()
    z = beta * (p - p_max)
    z = np.clip(z, -600.0, 0.0)
    exp_z = np.exp(z)
    w = exp_z / exp_z.sum()
    g = np.zeros(n)
    for m in range(len(p)):
        if w[m] > 1e-12:
            g += w[m] * grad_row(h, m, n)
    return g

def lse_descent_momentum(h, beta_schedule, max_iters_per_beta=40, momentum=0.4):
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    vel = np.zeros(n)
    for beta in beta_schedule:
        for it in range(max_iters_per_beta):
            g = lse_grad(h, beta)
            g = g - g.mean()
            gnorm = np.linalg.norm(g)
            if gnorm < 1e-15:
                break
            vel = momentum * vel + g
            vnorm = np.linalg.norm(vel)
            if vnorm < 1e-15:
                break
            d = -vel / vnorm
            pos_mask = d > 1e-12
            neg_mask = d < -1e-12
            alpha_max = 0.25
            if pos_mask.any():
                alpha_max = min(alpha_max, float(((1.0 - h[pos_mask]) / d[pos_mask]).min()))
            if neg_mask.any():
                alpha_max = min(alpha_max, float((h[neg_mask] / (-d[neg_mask])).min()))
            alpha_max = max(alpha_max, 1e-12)
            c_old = exact_score(h)
            found = False
            alpha = alpha_max
            for _ in range(25):
                h_try = repair_feasible(h + alpha * d)
                c_try = exact_score(h_try)
                if c_try < c_old - 1e-15:
                    h = h_try
                    best_c = min(best_c, c_try)
                    found = True
                    break
                alpha *= 0.55
                if alpha < 1e-13:
                    break
            if not found:
                vel *= 0.5
                if np.linalg.norm(vel) < 1e-14:
                    break
    return h, best_c

def interp_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    eo = np.linspace(0.0, 2.0, n + 1)
    en = np.linspace(0.0, 2.0, new_n + 1)
    h_new = np.zeros(new_n)
    for k in range(new_n):
        a, b = en[k], en[k + 1]
        total = 0.0
        for j in range(n):
            lo = max(a, eo[j])
            hi = min(b, eo[j + 1])
            if hi > lo:
                total += h[j] * (hi - lo)
        h_new[k] = total / (b - a)
    h_new = np.clip(h_new, 0.0, 1.0)
    return repair_feasible(h_new)

def topk_descent(h, max_iters=300, k_list=None):
    n = len(h)
    if k_list is None:
        k_list = [1, 3, 5, 10, 20]
    h = h.copy()
    best_c = exact_score(h)
    for it in range(max_iters):
        profile = compute_profile(h)
        c = float(profile.max())
        improved = False
        for k in k_list:
            top_k = np.argsort(profile)[-k:]
            g = np.zeros(n)
            for m in top_k:
                g += grad_row(h, m, n)
            g -= g.mean()
            gnorm = np.linalg.norm(g)
            if gnorm < 1e-15:
                continue
            d = -g / gnorm
            pos_mask = d > 1e-12
            neg_mask = d < -1e-12
            alpha_max = 0.2
            if pos_mask.any():
                alpha_max = min(alpha_max, float(((1.0 - h[pos_mask]) / d[pos_mask]).min()))
            if neg_mask.any():
                alpha_max = min(alpha_max, float((h[neg_mask] / (-d[neg_mask])).min()))
            alpha_max = max(alpha_max, 1e-12)
            alpha = alpha_max
            for _ in range(20):
                h_try = repair_feasible(h + alpha * d)
                c_try = exact_score(h_try)
                if c_try < c - 1e-15:
                    h = h_try
                    best_c = c_try
                    improved = True
                    break
                alpha *= 0.5
                if alpha < 1e-13:
                    break
            if improved:
                break
        if not improved:
            break
    return h, best_c

def trust_region_lp(h, radius=0.04):
    n = len(h)
    M = 2 * n - 1
    p = compute_profile(h)
    A_rows = []
    b_rows = []
    for m in range(M):
        gm = grad_row(h, m, n)
        row = np.zeros(n + 1)
        row[:n] = gm
        row[n] = -1.0
        A_rows.append(row)
        b_rows.append(-p[m])
    A_ub = np.array(A_rows)
    b_ub = np.array(b_rows)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    c_max = p.max()
    bounds = []
    for j in range(n):
        lo = max(-radius, -h[j])
        hi = min(radius, 1.0 - h[j])
        bounds.append((lo, hi))
    bounds.append((0.0, c_max + radius * 2.0))
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x is not None:
            d = res.x[:n]
            h_new = repair_feasible(h + d)
            c_new = exact_score(h_new)
            c_old = exact_score(h)
            if c_new < c_old - 1e-14:
                return h_new, c_new
    except Exception:
        pass
    return None, None

def slsqp_epigraph(h_start, maxiter=150):
    n = len(h_start)
    dx = 2.0 / n
    M = 2 * n - 1
    c0 = exact_score(h_start)
    t0 = c0 + 1e-3
    def obj(x):
        return x[-1]
    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    def eq_fun(x):
        return x[:-1].sum() - n / 2.0
    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g
    def ineq_fun(x):
        h = x[:-1]
        t = x[-1]
        profile = np.correlate(h, 1 - h, mode="full") * dx
        return t - profile
    def ineq_jac(x):
        h = x[:-1]
        J = np.zeros((M, n + 1))
        for m in range(M):
            J[m, :n] = -grad_row(h, m, n)
        J[:, -1] = 1.0
        return J
    x0 = np.concatenate([h_start, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 1.0)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.x is not None and np.isfinite(res.x).all():
            h_new = repair_feasible(res.x[:-1])
            c_new = exact_score(h_new)
            if c_new <= c0 + 1e-10:
                return h_new, c_new
    except Exception:
        pass
    return None, None

def coordinate_polish(h, max_iters=80):
    n = len(h)
    h = h.copy()
    best_c = exact_score(h)
    for it in range(max_iters):
        profile = compute_profile(h)
        c = float(profile.max())
        active = np.where(profile >= c - 0.002)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(profile))])
        g = np.zeros(n)
        for m in active:
            g += grad_row(h, m, n)
        g -= g.mean()
        improved = False
        idx_desc = np.argsort(-h)
        idx_asc = np.argsort(h)
        for attempt in range(min(40, n)):
            i = int(idx_desc[attempt % n])
            j = int(idx_asc[attempt % n])
            if i == j:
                continue
            for delta in [0.015, 0.008, 0.004]:
                for sign in [1.0, -1.0]:
                    h_try = h.copy()
                    h_try[i] += sign * delta
                    h_try[j] -= sign * delta
                    if h_try.min() < -1e-10 or h_try.max() > 1 + 1e-10:
                        continue
                    h_try = repair_feasible(h_try)
                    s = exact_score(h_try)
                    if s < best_c - 1e-14:
                        h = h_try
                        best_c = s
                        improved = True
                        break
                if improved:
                    break
        if not improved:
            break
    return h, best_c

def random_perturb(h, strength, rng):
    n = len(h)
    h_p = h + strength * (rng.randn(n) - 0.5)
    return repair_feasible(h_p)

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s - (time.perf_counter() - t_start)

    try:
        parent = np.array(initial_h_values, dtype=np.float64)
    except Exception:
        parent = None
    if parent is None or len(parent) < 4:
        parent = np.full(70, 0.5)
    parent = np.clip(parent, 0.0, 1.0)
    parent = repair_feasible(parent)

    best_h = parent.copy()
    best_c = exact_score(parent)

    # Stage 1: LSE homotopy with momentum (main workhorse)
    if time_left() > 300:
        h1, c1 = lse_descent_momentum(best_h, [20, 50, 100, 200], max_iters_per_beta=35, momentum=0.35)
        if c1 < best_c:
            best_h, best_c = h1, c1
    if time_left() > 200:
        h2, c2 = lse_descent_momentum(best_h, [400, 800, 1500], max_iters_per_beta=25, momentum=0.3)
        if c2 < best_c:
            best_h, best_c = h2, c2

    # Stage 2: Top-k descent refinement
    if time_left() > 150:
        h3, c3 = topk_descent(best_h, max_iters=400, k_list=[1, 2, 3, 5, 8, 15, 30])
        if c3 < best_c:
            best_h, best_c = h3, c3

    # Stage 3: Trust-region LP
    if time_left() > 100:
        for radius in [0.05, 0.03, 0.02, 0.01, 0.005, 0.003]:
            if time_left() < 30:
                break
            for _ in range(15):
                if time_left() < 20:
                    break
                h_lp, c_lp = trust_region_lp(best_h, radius=radius)
                if h_lp is not None:
                    best_h, best_c = h_lp, c_lp
                else:
                    break

    # Stage 4: SLSQP epigraph
    if time_left() > 80:
        h4, c4 = slsqp_epigraph(best_h, maxiter=200)
        if h4 is not None and c4 < best_c:
            best_h, best_c = h4, c4
        if time_left() > 50:
            h5, c5 = lse_descent_momentum(best_h, [1000, 3000], max_iters_per_beta=15, momentum=0.2)
            if c5 < best_c:
                best_h, best_c = h5, c5

    # Stage 5: Dimension exploration
    if time_left() > 180:
        n_cur = len(best_h)
        dim_list = []
        for dn in [-4, -2, 2, 4, 6, 8, -6, 10, 12]:
            d = n_cur + dn
            if 40 <= d <= 140 and d % 2 == 0:
                dim_list.append(d)
        for new_n in dim_list[:6]:
            if time_left() < 40:
                break
            h_int = interp_to_n(best_h, new_n)
            h6, c6 = lse_descent_momentum(h_int, [50, 200, 800], max_iters_per_beta=25, momentum=0.3)
            if c6 < best_c:
                best_h, best_c = h6, c6
            h7, c7 = topk_descent(best_h, max_iters=200, k_list=[1, 3, 5, 10, 20])
            if c7 < best_c:
                best_h, best_c = h7, c7

    # Stage 6: Coordinate polish
    if time_left() > 40:
        h8, c8 = coordinate_polish(best_h, max_iters=100)
        if c8 < best_c:
            best_h, best_c = h8, c8

    # Stage 7: Second round LSE at high beta
    if time_left() > 30:
        h9, c9 = lse_descent_momentum(best_h, [3000, 8000, 20000], max_iters_per_beta=12, momentum=0.15)
        if c9 < best_c:
            best_h, best_c = h9, c9

    # Stage 8: Final SLSQP
    if time_left() > 25:
        h10, c10 = slsqp_epigraph(best_h, maxiter=200)
        if h10 is not None and c10 < best_c:
            best_h, best_c = h10, c10

    # Stage 9: Micro top-1 descent
    if time_left() > 15:
        h11, c11 = topk_descent(best_h, max_iters=300, k_list=[1, 2, 3])
        if c11 < best_c:
            best_h, best_c = h11, c11

    # Stage 10: Random restart perturb + quick descent
    if time_left() > 60:
        for pert in [0.02, 0.01, 0.005]:
            if time_left() < 20:
                break
            h_p = random_perturb(best_h, pert, rng)
            h12, c12 = lse_descent_momentum(h_p, [200, 1000], max_iters_per_beta=20, momentum=0.25)
            if c12 < best_c:
                best_h, best_c = h12, c12

    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)

    try:
        eval_result = evaluate_erdos_solution(best_h)
        if isinstance(eval_result, (float, int)):
            final_c = float(eval_result)
        elif isinstance(eval_result, tuple) and len(eval_result) >= 2:
            final_c = float(eval_result[0])
    except Exception:
        pass

    return best_h, final_c, len(best_h)
