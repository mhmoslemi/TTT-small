import numpy as np
from scipy.optimize import minimize
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full"))) * dx

def profile_h(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            idx = order[i]
            add = min(remaining, capacity[idx])
            h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            idx = order[i]
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def two_opt(h, i, j, n_gs=35):
    n = len(h)
    s = h[i] + h[j]
    lo = max(0.0, s - 1.0)
    hi = min(1.0, s)
    if hi - lo < 1e-15:
        return h, exact_score(h)
    h_save = h.copy()
    c_save = exact_score(h_save)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = lo, hi
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    h[j] = c
    h[i] = s - c
    fc = exact_score(h)
    h[j] = d
    h[i] = s - d
    fd = exact_score(h)
    for _ in range(n_gs):
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            h[j] = c
            h[i] = s - c
            fc = exact_score(h)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            h[j] = d
            h[i] = s - d
            fd = exact_score(h)
    best_x = (a + b) / 2.0
    h[j] = best_x
    h[i] = s - best_x
    h = np.clip(h, 0.0, 1.0)
    c_new = exact_score(h)
    if c_new > c_save:
        return h_save, c_save
    return h, c_new

def coordinate_descent_2d(h, max_sweeps=10, t_start=None, budget_s=1100):
    n = len(h)
    h = h.copy()
    c_cur = exact_score(h)
    rng = np.random.RandomState(123)
    for sweep in range(max_sweeps):
        if t_start is not None and time.perf_counter() - t_start > budget_s - 15:
            break
        improved = False
        order = list(range(n))
        rng.shuffle(order)
        for idx in range(len(order)):
            if t_start is not None and time.perf_counter() - t_start > budget_s - 10:
                break
            i = order[idx]
            candidates = []
            for dj in [-3, -2, -1, 1, 2, 3]:
                j = i + dj
                if 0 <= j < n and j != i:
                    candidates.append(j)
            j_sym = n - 1 - i
            if j_sym != i and j_sym not in candidates:
                candidates.append(j_sym)
            for j in candidates:
                h2, c2 = two_opt(h, i, j, n_gs=30)
                if c2 < c_cur - 1e-14:
                    h = h2
                    c_cur = c2
                    improved = True
                    break
        if not improved:
            break
    return h, c_cur

def build_gradient_rows(h, rows, n, dx):
    G = np.zeros((len(rows), n))
    for idx, m in enumerate(rows):
        d = m - (n - 1)
        if d == 0:
            G[idx, :] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            for i in range(n - d):
                G[idx, i + d] += dx * (1.0 - h[i])
                G[idx, i] -= dx * h[i + d]
        else:
            dp = -d
            for j in range(n - dp):
                G[idx, j + dp] += dx * (1.0 - h[j])
                G[idx, j] -= dx * h[j + dp]
    return G

def slsqp_active(h0, maxiter=300, active_k=15, tol=1e-12):
    n = len(h0)
    dx = 2.0 / n
    M = 2 * n - 1
    P = profile_h(h0)
    c0 = P.max()
    thresh = c0 - max(0.005, 0.08 * (c0 - np.median(P)))
    active = np.where(P >= thresh)[0]
    if len(active) == 0:
        active = np.array([np.argmax(P)])
    if len(active) > active_k:
        top_idx = np.argsort(P[active])[-active_k:]
        active = active[top_idx]

    def obj(x):
        return x[-1]

    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g

    def eq_fun(x):
        return x[:-1].sum() - n / 2.0

    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:-1] = 1.0
        return g

    def ineq_fun(x):
        h = x[:-1]
        t = x[-1]
        prof = profile_h(h)
        return t - prof[active]

    def ineq_jac(x):
        h = x[:-1]
        G = build_gradient_rows(h, active, n, dx)
        J = np.zeros((len(active), n + 1))
        J[:, :-1] = -G
        J[:, -1] = 1.0
        return J

    x0 = np.concatenate([h0, [c0 + 1e-5]])
    bounds = [(0.0, 1.0)] * n + [(0.0, c0 + 1.0)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
    ]
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': tol})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = repair_feasible(res.x[:-1])
            c_new = exact_score(h_new)
            if c_new <= c0 + 1e-9:
                return h_new
    except Exception:
        pass
    return None

def nesterov_softmax(h, beta=300, max_iters=150, t_start=None, budget_s=1100):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    h_ylm1 = h.copy()
    c_best = exact_score(h)
    c_cur = c_best
    t = 1.0
    for it in range(max_iters):
        if t_start is not None and time.perf_counter() - t_start > budget_s - 15:
            break
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        theta = 1.0 - (t + 1.0) / t_new
        h_y = h + theta * (h - h_ylm1)
        h_y = np.clip(h_y, 0.0, 1.0)
        h_y = repair_feasible(h_y)
        P = profile_h(h_y)
        Pmax = P.max()
        exp_arg = beta * (P - Pmax)
        exp_arg = np.maximum(exp_arg, -500.0)
        weights = np.exp(exp_arg)
        S = weights.sum()
        if S < 1e-300:
            weights = np.zeros(len(P))
            weights[np.argmax(P)] = 1.0
        else:
            weights /= S
        active = np.where(weights > 0.005 / len(P))[0]
        if len(active) == 0:
            active = np.argsort(weights)[-8:]
        G = build_gradient_rows(h_y, active, n, dx)
        w_a = weights[active]
        w_a = w_a / w_a.sum()
        grad = G.T @ w_a
        grad -= grad.mean()
        snorm = np.linalg.norm(grad)
        if snorm < 1e-15:
            break
        grad /= snorm
        step = 2.0 / (t_new + 1.0)
        h_try = h - step * grad
        h_try = repair_feasible(h_try)
        c_try = exact_score(h_try)
        if c_try <= c_cur + 1e-14:
            h_ylm1 = h.copy()
            h = h_try
            c_cur = c_try
            if c_try < c_best:
                c_best = c_try
        else:
            h_ylm1 = h.copy()
        t = t_new
    return h, c_best

def interpolate_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    full_xs = np.concatenate([[0.0], xs, [2.0]])
    full_h = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(xt, full_xs, full_h)
    return repair_feasible(np.clip(h_new, 0.0, 1.0))

def symmetrize(h):
    return (h + h[::-1].copy()) / 2.0
