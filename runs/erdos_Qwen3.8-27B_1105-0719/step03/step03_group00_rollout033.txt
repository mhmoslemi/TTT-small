        s = w[n-1+d] + w[n-1-d]
        if abs(s) < 1e-15:
            continue
        grad[:n-d] -= s * h[d:]
        grad[d:] += s * (1.0 - h[:n-d])
    return dx * grad

def nag_smooth(h, dx, target, alpha, n_steps, lr_init, momentum, budget_s, t0, min_time=5.0):
    """Nesterov Accelerated Gradient on softmax-smoothed objective"""
    n = len(h)
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    y = h.copy()  # Nesterov lookahead
    v = np.zeros(n)
    cur = h.copy()
    lr = lr_init
    stall = 0
    
    for step in range(n_steps):
        if budget_s - (time.time() - t0) < min_time:
            break
        
        # Nesterov step
        y = cur + momentum * v
        y = np.clip(y, 0.0, 1.0)  # keep in box for gradient evaluation
        
        corr = np.correlate(y, 1.0 - y, mode='full')
        w = compute_weights(corr, alpha)
        g = weighted_gradient(y, w, dx)
        g -= g.mean()  # ensure zero-mean direction
        
        v_new = momentum * v - lr * g
        candidate = cur + v_new
        candidate = project_feasible(candidate, target)
        
        # Line search
        improved = False
        for t in [1.0, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01]:
            h_try = cur + t * v_new
            h_try = project_feasible(h_try, target)
            c5_try = c5_of_h(h_try, dx)
            if c5_try < best_c5:
                stall = 0
            if c5_try < c5_of_h(cur, dx) or c5_try < best_c5:
                cur = h_try
                v = v_new
                improved = True
                if c5_try < best_c5:
                    best_c5 = c5_try
                    best_h = cur.copy()
                break
        
        if not improved:
            lr *= 0.6
            stall += 1
            if stall > 30 or lr < 1e-10:
                break
        else:
            lr = min(0.05, lr * 1.05)
            if c5_of_h(cur, dx) < best_c5:
                best_c5 = c5_of_h(cur, dx)
                best_h = cur.copy()
    
    # Update best
    c5_cur = c5_of_h(cur, dx)
    if c5_cur < best_c5:
        best_c5 = c5_cur
        best_h = cur.copy()
    
    return best_h, best_c5

def softmax_continuation(h, dx, target, budget_s, t0, alphas=None, steps_per_alpha=50):
    """Run gradient descent with decreasing alpha"""
    n = len(h)
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    cur = h.copy()
    
    if alphas is None:
        # Start smooth, end sharp
        alphas = [0.3, 0.2, 0.15, 0.1, 0.07, 0.05, 0.03, 0.02, 0.01, 0.005]
    
    for alpha in alphas:
        if budget_s - (time.time() - t0) < 15.0:
            break
        lr = max(0.001, 0.02 * alpha)
        h_new, c5_new = nag_smooth(cur, dx, target, alpha, steps_per_alpha, lr, 0.8, budget_s, t0)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new
            cur = h_new
    
    return best_h, best_c5

def coordinate_descent(h, dx, target, n_passes, budget_s, t0, min_time=5.0):
    """Try moving each coordinate to 0, 1, or optimal value"""
    n = len(h)
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    cur = h.copy()
    
    for p in range(n_passes):
        if budget_s - (time.time() - t0) < min_time:
            break
        improved = False
        for i in range(n):
            if budget_s - (time.time() - t0) < min_time:
                    break
            old = cur[i]
            # Try moving to a few candidate values
            candidates = [0.0, 1.0, old + 0.1, old - 0.1, old + 0.05, old - 0.05]
            best_candidate = old
            best_cand_c5 = c5_of_h(cur, dx)
            for cv in candidates:
                cur[i] = cv
                cur = project_feasible(cur, target)
                c5_try = c5_of_h(cur, dx)
                if c5_try < best_cand_c5 - 1e-14:
                    best_cand_c5 = c5_try
                    best_candidate = cv
            cur[i] = best_candidate
            cur = project_feasible(cur, target)
            c5_cur = c5_of_h(cur, dx)
            if c5_cur < best_c5 - 1e-14:
                improved = True
                best_c5 = c5_cur
                best_h = cur.copy()
        if not improved:
            break
    
    return best_h, best_c5

def mode_escape(h, dx, target, rng, n_modes, n_iter, budget_s, t0, min_time=10.0):
    """Search in low-dimensional mode space"""
    n = len(h)
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    
    j = np.arange(n) + 0.5
    modes = np.zeros((n, n_modes))
    for r in range(n_modes):
        freq = r + 1
        if r % 2 == 0:
            m = np.cos(2.0 * np.pi * freq * j / (2.0 * n))
        else:
            m = np.sin(2.0 * np.pi * freq * j / (2.0 * n))
        m -= m.mean()
        for prev in range(r):
            m -= np.dot(m, modes[:, prev]) * modes[:, prev]
        norm = np.linalg.norm(m)
        if norm > 1e-10:
            m /= norm
        modes[:, r] = m
    
    D = n_modes
    pop_size = max(12, 2 * D)
    pop = rng.randn(pop_size, D) * 0.15
    pop_c5 = np.zeros(pop_size)
    for i in range(pop_size):
        h_try = project_feasible(h + pop[i] @ modes.T, target)
        pop_c5[i] = c5_of_h(h_try, dx)
    
    best_idx = pop_c5.argmin()
    best_pop_c5 = pop_c5[best_idx]
    best_pop_h = project_feasible(h + pop[best_idx] @ modes.T, target)
    
    if best_pop_c5 < best_c5:
        best_c5 = best_pop_c5
        best_h = best_pop_h.copy()
    
    F, CR = 0.6, 0.8
    for it in range(n_iter):
        if budget_s - (time.time() - t0) < min_time:
            break
        for i in range(pop_size):
            a, b, c = rng.choice(pop_size, 3, replace=False)
            donor = pop[a] + F * (pop[b] - pop[c])
            mask = rng.random(D) < CR
            if not mask.any():
                mask[rng.randint(D)] = True
            trial = np.where(mask, donor, pop[i])
            h_trial = project_feasible(h + trial @ modes.T, target)
            c5_trial = c5_of_h(h_trial, dx)
            if c5_trial <= pop_c5[i]:
                pop[i] = trial
                pop_c5[i] = c5_trial
                if c5_trial < best_pop_c5:
                    best_pop_c5 = c5_trial
                    if c5_trial < best_c5:
                        best_c5 = c5_trial
                        best_h = h_trial.copy()
    
    return best_h, best_c5

import numpy as np
import time

def project_feasible(x, target_sum):
    n = len(x)
    if target_sum <= 0:
        return np.zeros(n)
    if target_sum >= n:
        return np.ones(n)
    x = np.nan_to_num(x, nan=0.5, posinf=2.0, neginf=-2.0)
    lo, hi = -3.0, 3.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    err = target_sum - h.sum()
    if abs(err) > 1e-13:
        if err > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if mask.sum() > 0:
            h[mask] += err / mask.sum()
            h = np.clip(h, 0.0, 1.0)
            err2 = target_sum - h.sum()
            if abs(err2) > 1e-13:
                if err2 > 0:
                    mask2 = h < 1.0 - 1e-12
                else:
                    mask2 = h > 1e-12
                if mask2.sum() > 0:
                    h[mask2] += err2 / mask2.sum()
                    h = np.clip(h, 0.0, 1.0)
    return h

def c5_of_h(h, dx):
    return np.correlate(h, 1.0 - h, mode='full').max() * dx

def compute_weights(corr, alpha):
    scaled = corr / alpha
    scaled -= scaled.max()
    p = np.exp(scaled)
    s = p.sum()
    if s < 1e-300:
        idx = np.argsort(corr)[-10:]
        p = np.zeros_like(corr)
        p[idx] = 1.0
        return p / 10.0
    return p / s

def weighted_gradient(h, w, dx):
    n = len(h)
    grad = np.zeros(n)
    w0 = w[n - 1]
    grad += w0 * (1.0 - 2.0 * h)
    for d in range(1, n):
        s = w[n - 1 + d] + w[n - 1 - d]
        if abs(s) < 1e-15:
            continue
        grad[:n - d] -= s * h[d:]
        grad[d:] += s * (1.0 - h[:n - d])
    return dx * grad

def momentum_gd(h, dx, target, alpha, n_steps, lr_init, mu, budget_s, t0, min_time=5.0):
    n = len(h)
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    cur = h.copy()
    v = np.zeros(n)
    lr = lr_init
    stall = 0

    for step in range(n_steps):
        if budget_s - (time.time() - t0) < min_time:
            break
        corr = np.correlate(cur, 1.0 - cur, mode='full')
        w = compute_weights(corr, alpha)
        g = weighted_gradient(cur, w, dx)
        g -= g.mean()
        v = mu * v - lr * g
        cur_c5 = c5_of_h(cur, dx)
        improved = False
        for t in [1.0, 0.6, 0.4, 0.25, 0.15, 0.1, 0.05, 0.02, 0.01, 0.005]:
            h_try = project_feasible(cur + t * v, target)
            c5_try = c5_of_h(h_try, dx)
            if c5_try < cur_c5 - 1e-15:
                cur = h_try
                improved = True
                if c5_try < best_c5:
                    best_c5 = c5_try
                    best_h = cur.copy()
                break
        if not improved:
            lr *= 0.5
            v *= 0.3
            stall += 1
            if stall > 60 or lr < 1e-12:
                break
        else:
            lr = min(0.08, lr * 1.03)
            if stall > 0:
                stall = 0

    c5_cur = c5_of_h(cur, dx)
    if c5_cur < best_c5:
        best_c5 = c5_cur
        best_h = cur.copy()
    return best_h, best_c5

def softmax_continuation(h, dx, target, budget_s, t0, alphas, steps_per_alpha=50):
    n = len(h)
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    cur = h.copy()
    for alpha in alphas:
        if budget_s - (time.time() - t0) < 15.0:
            break
        lr = max(0.002, 0.03 * alpha)
        h_new, c5_new = momentum_gd(cur, dx, target, alpha, steps_per_alpha, lr, 0.8, budget_s, t0)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new
            cur = h_new
    return best_h, best_c5

def coordinate_descent(h, dx, target, n_passes, budget_s, t0, min_time=5.0):
    n = len(h)
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    cur = h.copy()
    for p in range(n_passes):
        if budget_s - (time.time() - t0) < min_time:
            break
        improved = False
        for i in range(n):
            if budget_s - (time.time() - t0) < min_time:
                break
            old = cur[i]
            cur_c5 = c5_of_h(cur, dx)
            best_cv = old
            best_cv_c5 = cur_c5
            for cv in [0.0, 1.0, old + 0.15, old - 0.15, old + 0.08, old - 0.08, old + 0.04, old - 0.04]:
                if cv < -0.01 or cv > 1.01:
                    continue
                cur[i] = cv
                cur = project_feasible(cur, target)
                c5_try = c5_of_h(cur, dx)
                if c5_try < best_cv_c5 - 1e-14:
                    best_cv_c5 = c5_try
                    best_cv = cv
            if best_cv != old:
                cur[i] = best_cv
                cur = project_feasible(cur, target)
                c5_cur = c5_of_h(cur, dx)
                if c5_cur < best_c5 - 1e-14:
                    improved = True
                    best_c5 = c5_cur
                    best_h = cur.copy()
            else:
                cur[i] = old
                cur = project_feasible(cur, target)
        if not improved:
            break
    return best_h, best_c5

def pairwise_moves(h, dx, target, n_trials, rng, budget_s, t0, min_time=5.0):
    n = len(h)
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    cur = h.copy()
    for trial in range(n_trials):
        if budget_s - (time.time() - t0) < min_time:
            break
        i, j = int(rng.randint(n)), int(rng.randint(n))
        if i == j:
            continue
        delta = rng.uniform(-0.4, 0.4)
        cur[i] = cur[i] + delta
        cur[j] = cur[j] - delta
        cur = project_feasible(cur, target)
        c5_try = c5_of_h(cur, dx)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = cur.copy()
        else:
            cur = best_h.copy()
    return best_h, best_c5

def build_modes(n, n_modes, seed=0):
    rng_local = np.random.RandomState(seed)
    j = np.arange(n) + 0.5
    modes = np.zeros((n, n_modes))
    for r in range(n_modes):
        freq = rng_local.randint(1, max(2, n // 3))
        if r % 2 == 0:
            m = np.cos(2.0 * np.pi * freq * j / (2.0 * n))
        else:
            m = np.sin(2.0 * np.pi * freq * j / (2.0 * n))
        m += rng_local.randn(n) * 0.1
        m -= m.mean()
        for prev in range(r):
            m -= np.dot(m, modes[:, prev]) * modes[:, prev]
        norm = np.linalg.norm(m)
        if norm > 1e-10:
            m /= norm
        modes[:, r] = m
    return modes

def mode_escape(h, dx, target, rng, n_modes, n_iter, budget_s, t0, min_time=10.0, scale=0.15):
    n = len(h)
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    modes = build_modes(n, n_modes, seed=int(rng.randint(10000)))
    D = n_modes
    pop_size = max(12, 2 * D)
    pop = rng.randn(pop_size, D) * scale
    pop_c5 = np.zeros(pop_size)
    for i in range(pop_size):
        h_try = project_feasible(h + pop[i] @ modes.T, target)
        pop_c5[i] = c5_of_h(h_try, dx)
    best_idx = int(pop_c5.argmin())
    best_pop_c5 = pop_c5[best_idx]
    if best_pop_c5 < best_c5:
        best_c5 = best_pop_c5
        best_h = project_feasible(h + pop[best_idx] @ modes.T, target)
    F, CR = 0.6, 0.8
    for it in range(n_iter):
        if budget_s - (time.time() - t0) < min_time:
            break
        for i in range(pop_size):
            a, b, c = rng.choice(pop_size, 3, replace=False)
            donor = pop[a] + F * (pop[b] - pop[c])
            mask = rng.random(D) < CR
            if not mask.any():
                mask[int(rng.randint(D))] = True
            trial = np.where(mask, donor, pop[i])
            h_trial = project_feasible(h + trial @ modes.T, target)
            c5_trial = c5_of_h(h_trial, dx)
            if c5_trial <= pop_c5[i]:
                pop[i] = trial
                pop_c5[i] = c5_trial
                if c5_trial < best_c5:
                    best_c5 = c5_trial
                    best_h = h_trial.copy()
    return best_h, best_c5

def transfer_grid(h_src, n_target):
    n_src = len(h_src)
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_tgt = (np.arange(n_target) + 0.5) * (2.0 / n_target)
    return np.interp(x_tgt, x_src, h_src)

def stochastic_restarts(h, dx, target, rng, n_restarts, budget_s, t0, min_time=15.0):
    n = len(h)
    best_h = h.copy()
    best_c5 = c5_of_h(h, dx)
    for r in range(n_restarts):
        if budget_s - (time.time() - t0) < min_time:
            break
        scale = max(0.03, 0.15 / (r + 1))
        pert = rng.standard_normal(n)
        pert -= pert.mean()
        pert /= (np.linalg.norm(pert) + 1e-10)
        h_p = project_feasible(h + scale * pert, target)
        h_opt, c5_opt = momentum_gd(h_p, dx, target, 0.02, 40, 0.005, 0.8, budget_s, t0)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
    return best_h, best_c5
