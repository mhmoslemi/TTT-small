Key insights from the strategy:

Let me think about how to implement this more effectively:

For a direction v with sum(v) = 0:
- profile(h) = R₀ (raw correlation at α=0)
- profile(h+v) = R₊
- profile(h-v) = R₋

Then profile(h + αv) = R₀ + α·B + α²·C where:
- C = (R₊ + R₋)/2 - R₀
- B = (R₊ - R₋)/2

The objective is max_lag profile(h + αv) * dx

For each lag l, profile_l(α) = C_l·α² + B_l·α + R₀_l

We need to find α that minimizes max_l (C_l·α² + B_l·α + R₀_l) * dx, subject to h_i + α·v_i ∈ [0,1] for all i.

The feasible α interval:
- For v_i > 0: α ≤ (1 - h_i)/v_i
- For v_i < 0: α ≥ -h_i/(-v_i) = h_i/v_i (since v_i < 0, this is negative)
- For v_i = 0: no constraint

So α ∈ [max_i h_i/v_i for v_i < 0, min_i (1-h_i)/v_i for v_i > 0]

Now, to find the minimum of the piecewise max of quadratics over an interval, we can:
1. Sample the interval densely
2. Find bracket around minimum
3. Refine

But actually, since we're evaluating the full profile (2n-1 values) at each point, and we need max over all lags, this is a 1D minimax problem. We can:
1. Coarse grid over feasible α
2. Find minimum
3. Golden section or ternary search around it

If we do 100 samples per direction, and 50 directions per round, that's 5000 correlate calls per round. Each correlate call for n=70 is fast (maybe 50μs). So 5000 * 50μs = 250ms per round. That's very manageable.

def repair_feasible(vec):

    if abs(diff) < 1e-14:

            add = min(remaining, capacity[order[i]])
            h[order[i]] += add
            remaining -= add

        remaining = -diff

            remaining -= rem
            if remaining <= 1e-15:

    """
    Exact line search along h + α*v.
    Uses quadratic interpolation from 3 evaluations.
    Returns best (h_new, score) or (None, None).
    """

    dx = 2.0 / n

        return None, None

        return None, None

    Rp = raw_profile(h + v)  # at α=1 (if feasible)
    Rm = raw_profile(h - v)  # at α=-1 (if feasible)

    # the polynomial. We just need to constrain α to the feasible interval.

    C = 0.5 * (Rp + Rm) - R0
    B = 0.5 * (Rp - Rm)
    
    # Sample the feasible interval
    alphas = np.linspace(lo, hi, n_samples)
    # Compute objective for each α: max_l (C_l*α² + B_l*α + R0_l) * dx
    # Shape: C,B,R0 are (2n-1,), alphas is (n_samples,)
    # Result: (n_samples, 2n-1)

           B[np.newaxis, :] * alphas[:, np.newaxis] + 
           R0[np.newaxis, :])

import numpy as np
from scipy.optimize import minimize
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full")) * dx)


def raw_profile(h):
    return np.correlate(h, 1 - h, mode="full")


def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for i in range(n):
            add = min(remaining, capacity[order[i]])
            h[order[i]] += add
            remaining -= add
            if remaining <= 1e-15:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in range(n):
            rem = min(remaining, h[order[i]])
            h[order[i]] -= rem
            remaining -= rem
            if remaining <= 1e-15:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h


def compute_gradient(h, active_lags, weights, n):
    """Compute gradient of weighted sum of active lag profiles."""
    dx = 2.0 / n
    center = n - 1
    grad = np.zeros(n)
    for m, w in zip(active_lags, weights):
        offset = m - center
        if offset == 0:
            grad += w * dx * (1.0 - 2.0 * h)
        else:
            if 0 <= offset:
                idx1 = np.arange(n - offset)
                idx2 = idx1 + offset
                grad[idx1] += w * dx * (1.0 - h[idx2])
                grad[idx2] -= w * dx * h[idx1]
            else:
                oa = -offset
                idx1 = np.arange(oa, n)
                idx2 = idx1 - oa
                grad[idx1] += w * dx * (1.0 - h[idx2])
                grad[idx2] -= w * dx * h[idx1]
    return grad - grad.mean()


def feasible_alpha(h, v):
    """Find alpha interval where h + alpha*v stays in [0,1]."""
    lo = 0.0
    hi = 0.0
    pm = v > 1e-15
    nm = v < -1e-15
    if np.any(pm):
        hi = float(np.min((1.0 - h[pm]) / v[pm]))
    if np.any(nm):
        lo = float(np.max(h[nm] / v[nm]))
    if hi <= lo:
        return None
    return (lo, hi)


def line_search(h, v, n_samples=200):
    """Exact quadratic line search along h + alpha*v."""
    interval = feasible_alpha(h, v)
    if interval is None:
        return None, None
    lo, hi = interval
    if hi - lo < 1e-10:
        return None, None

    R0 = raw_profile(h)
    Rp = raw_profile(h + v)
    Rm = raw_profile(h - v)
    C = 0.5 * (Rp + Rm) - R0
    B = 0.5 * (Rp - Rm)

    alphas = np.linspace(lo, hi, n_samples)
    obj = (C[np.newaxis, :] * alphas[:, np.newaxis] ** 2 +
           B[np.newaxis, :] * alphas[:, np.newaxis] +
           R0[np.newaxis, :])
    scores = np.max(obj, axis=1)
    i_min = int(np.argmin(scores))
    best_a = alphas[i_min]

    if 0 < i_min < n_samples - 1:
        a = alphas[i_min - 1]
        b = alphas[i_min + 1]
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        for _ in range(40):
            c = b - gr * (b - a)
            d = a + gr * (b - a)
            mc = float(np.max(C * c * c + B * c + R0))
            md = float(np.max(C * d * d + B * d + R0))
            if mc < md:
                b = d
            else:
                a = c
            if b - a < 1e-14:
                break
        best_a = 0.5 * (a + b)

    best_score = float(np.max(C * best_a ** 2 + B * best_a + R0))
    curr_score = float(np.max(R0))
    if best_score < curr_score - 1e-12:
        h_new = repair_feasible(h + best_a * v)
        return h_new, best_score
    return None, None


def gen_dir_random(rng, n):
    v = rng.randn(n)
    v -= v.mean()
    ma = np.max(np.abs(v))
    if ma < 1e-10:
        return None
    return v / ma


def gen_dir_sine(rng, n):
    f = rng.uniform(0.5, 6.0)
    ph = rng.uniform(0, 2 * np.pi)
    v = np.sin(2 * np.pi * f * np.arange(n) / n + ph)
    v -= v.mean()
    ma = np.max(np.abs(v))
    if ma < 1e-10:
        return None
    return v / ma


def gen_dir_block(rng, n):
    nblk = rng.randint(2, 7)
    v = np.zeros(n)
    edges = np.sort(rng.randint(1, n, nblk - 1))
    starts = np.concatenate([[0], edges])
    ends = np.concatenate([edges, [n]])
    for s, e in zip(starts, ends):
        v[s:e] = 1.0 if rng.rand() > 0.5 else -1.0
    v -= v.mean()
    ma = np.max(np.abs(v))
    if ma < 1e-10:
        return None
    return v / ma


def gen_dir_gradient(h, profile):
    n = len(h)
    c_max = float(np.max(profile))
    beam = max(0.003, 0.08 * (c_max - float(np.median(profile))))
    active = np.where(profile >= c_max - beam)[0]
    if len(active) == 0:
        active = np.array([np.argmax(profile)])
    if len(active) > 12:
        order = np.argsort(profile[active])
        active = active[order[-12:]]
    w = profile[active] - profile[active].min() + 1e-8
    w /= w.sum()
    grad = compute_gradient(h, active, w, n)
    ma = np.max(np.abs(grad))
    if ma < 1e-12:
        return None
    return -grad / ma


def improve_local(h0, t_start, budget_s, max_rounds=300):
    n = len(h0)
    rng = np.random.RandomState(int((time.perf_counter() * 7919) % 99991))
    h = h0.copy()
    c_cur = exact_score(h)
    profile = raw_profile(h)
    stagnation = 0

    for rnd in range(max_rounds):
        elapsed = time.perf_counter() - t_start
        if budget_s - elapsed < 8:
            break

        ndirs = 25
        accepted = False

        for di in range(ndirs):
            t = (rnd * 7 + di) % 8
            if t == 0:
                v = gen_dir_gradient(h, profile)
            elif t <= 2:
                v = gen_dir_random(rng, n)
            elif t <= 4:
                v = gen_dir_sine(rng, n)
            elif t == 5:
                v = gen_dir_block(rng, n)
            elif t == 6:
                v = h - h.mean()
                ma = np.max(np.abs(v))
                v = v / ma if ma > 1e-12 else None
            else:
                v = -(h - h.mean())
                ma = np.max(np.abs(v))
                v = v / ma if ma > 1e-12 else None

            if v is None:
                continue

            h_new, c_new = line_search(h, v, n_samples=150)
            if h_new is not None:
                h = h_new
                c_cur = c_new
                profile = raw_profile(h)
                accepted = True
                stagnation = 0
                break

        if not accepted:
            stagnation += 1
            if stagnation % 8 == 0:
                jump = 0.005 + 0.01 * rng.rand()
                v = gen_dir_random(rng, n)
                if v is not None:
                    h_try = repair_feasible(h + jump * v)
                    if exact_score(h_try) < c_cur + 0.005:
                        h = h_try
                        c_cur = exact_score(h)
                        profile = raw_profile(h)

    return h, c_cur


def interp_h(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    fx = np.concatenate([[0.0], xs, [2.0]])
    fh = np.concatenate([[h[0]], h, [h[-1]]])
    hn = np.interp(xt, fx, fh)
    return repair_feasible(np.clip(hn, 0, 1))


def slsqp_refine(h0, maxiter=250):
    n = len(h0)
    dx = 2.0 / n

    def obj(x):
        return float(np.max(np.correlate(x, 1.0 - x, mode="full")) * dx)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq',
             'fun': lambda x: x.sum() - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h0.copy(), method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        hs = repair_feasible(res.x)
        cs = exact_score(hs)
        if cs < exact_score(h0) + 1e-13:
            return hs
    except Exception:
        pass
    return None
