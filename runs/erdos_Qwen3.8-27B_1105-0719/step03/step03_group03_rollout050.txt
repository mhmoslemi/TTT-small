def lp_subproblem(h_curr, profile, active, dx, trust, rho):
    """Solve LP subproblem for active set."""
    if not HAS_CVXPY:
        return None
    n = len(h_curr)
    target = n / 2.0
    
    # Build gradient matrix for active constraints
    G = np.zeros((len(active), n))
    for idx, k in enumerate(active):
        shift = k - (n-1)
        for i in range(n):
            g = 0.0
            if 0 <= i + shift < n:
                g += 1.0 - h_curr[i + shift]
            if 0 <= i - shift < n:
                g -= h_curr[i - shift]
            G[idx, i] = g * dx
    
    b = profile[active] - G @ h_curr
    
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    cons = [
        h_var >= 0,
        h_var <= 1,
        cp.sum(h_var) == target,
        h_var >= h_curr - trust,
        h_var <= h_curr + trust,
        G @ h_var - t_var <= -b
    ]
    obj = cp.Minimize(t_var + rho * cp.norm1(h_var - h_curr))
    prob = cp.Problem(obj, cons)
    for solver in [cp.ECOS, cp.SCS, cp.CBC]:
        try:
            prob.solve(solver=solver, max_iters=20000, abstol=1e-10)
            break
        except Exception:
            continue
    if h_var.value is None:
        return None
    h_sol = np.array(h_var.value).flatten()
    if not np.isfinite(h_sol).all():
        return None
    return h_sol

        order = np.argsort(-capacity)
        remaining = diff
        for idx in order:
            add = min(remaining, capacity[idx] + 1e-15)
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for idx in order:
            rem = min(remaining, h[idx] + 1e-15)
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h


def interp_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    fx = np.concatenate([[0.0], xs, [2.0]])
    fh = np.concatenate([[h[0]], h, [h[-1]]])
    hn = np.interp(xt, fx, fh)
    return repair_feasible(np.clip(hn, 0, 1))


def active_set_loop(h0, max_iters, budget_fn):
    n = len(h0)
    dx = 2.0 / n
    h = h0.copy()
    best_c = exact_score(h)
    trust = 0.03
    stagnation = 0
    
    for _ in range(max_iters):
        if budget_fn() < 20:
            break
        profile = compute_profile(h)
        c_max = float(np.max(profile))
        if c_max < best_c - 1e-14:
            best_c = c_max
            stagnation = 0
        else:
            stagnation += 1
        
        # Active set
        spread = c_max - np.median(profile)
        beam = max(0.003, min(0.02, 0.2 * spread))
        active = np.where(profile >= c_max - beam)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(profile))])
        if len(active) > 20:
            active = active[np.argsort(profile[active])[-20:]]
        
        # Try LP
        accepted = False
        h_lp = lp_subproblem(h, profile, active, dx, trust, 0.01)
        if h_lp is not None:
            h_lp = repair_feasible(h_lp)
            c_lp = exact_score(h_lp)
            if c_lp < best_c - 1e-14:
                h = h_lp
                best_c = c_lp
                trust = min(trust * 1.3, 0.08)
                accepted = True
            else:
                for alpha in [0.7, 0.5, 0.3]:
                    hb = repair_feasible(alpha * h_lp + (1-alpha) * h)
                    cb = exact_score(hb)
                    if cb < best_c - 1e-14:
                        h = hb
                        best_c = cb
                        trust = min(trust * 1.1, 0.08)
                        accepted = True
                        break
                if not accepted:
                    trust *= 0.5
        
        if not accepted:
            # Gradient step
            hg, cg = gradient_step(h, profile, active, n, dx, trust)
            if hg is not None:
                h = hg
                best_c = cg
                accepted = True
            else:
                trust *= 0.4
                if trust < 0.001 or stagnation > 50:
                    break
    
    return h, best_c


def gradient_step(h, profile, active, n, dx, base_step):
    # Build gradient for active set
    G = np.zeros((len(active), n))
    for idx, k in enumerate(active):
        shift = k - (n-1)
        for i in range(n):
            g = 0.0
            if 0 <= i + shift < n:
                g += 1.0 - h[i + shift]
            if 0 <= i - shift < n:
                g -= h[i - shift]
            G[idx, i] = g * dx
    
    w = profile[active] - profile[active].min() + 1e-6
    w = w / w.sum()
    grad = G.T @ w
    grad -= grad.mean()
    gn = np.linalg.norm(grad)
    if gn < 1e-15:
        return None, None
    grad /= gn
    
    for s in [base_step, base_step*0.5, base_step*0.25, base_step*0.1, base_step*0.05]:
        h_try = repair_feasible(h - s * grad)
        sc = exact_score(h_try)
        if sc < float(np.max(profile)) - 1e-14:
            return h_try, sc
    return None, None


def greedy_binary_construction(n, rng):

        h[perm[i]] = 1.0

    """Project h onto {x: 0<=x<=1, sum(x)=target}."""
    h = np.clip(h, 0, 1)
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    # Water-filling approach
    if diff > 0:
        # Need to add diff total
        # Add to positions with most room (highest 1-h)
        room = 1.0 - h
        order = np.argsort(-room)
        remaining = diff
        for idx in order:
            add = min(remaining, room[idx])
            h[idx] += add
            remaining -= add
            if remaining < 1e-14:

        order = np.argsort(h)
        remaining = -diff
        for idx in order:
            rem = min(remaining, h[idx])
            h[idx] -= rem
            remaining -= rem
            if remaining < 1e-14:

    return h

import numpy as np

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))


def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx


def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        capacity = 1.0 - h
        order = np.argsort(-capacity)
        remaining = diff
        for idx in order:
            add = min(remaining, capacity[idx] + 1e-15)
            if add > 0:
                h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for idx in order:
            rem = min(remaining, h[idx] + 1e-15)
            if rem > 0:
                h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    diff2 = target - h.sum()
    if abs(diff2) > 1e-15:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff2
        h = np.clip(h, 0.0, 1.0)
    return h


def binarize(h, n):
    """Binarize h to have exactly n/2 ones."""
    h_bin = (h >= 0.5).astype(np.float64)
    n_ones = int(np.sum(h_bin > 0.5))
    target = n // 2
    if n_ones > target:
        ones_idx = np.where(h_bin > 0.5)[0]
        to_flip = ones_idx[n_ones - target:]
        h_bin[to_flip] = 0.0
    elif n_ones < target:
        zeros_idx = np.where(h_bin < 0.5)[0]
        to_flip = zeros_idx[:target - n_ones]
        h_bin[to_flip] = 1.0
    return h_bin


def binary_sa_search(h0, t0, tf, max_rounds, rng, patience=30, budget_fn=None):
    """Simulated annealing on binary vectors via random swaps."""
    n = len(h0)
    dx = 2.0 / n
    h = h0.copy()
    profile = np.correlate(h, 1 - h, mode="full") * dx
    cur_c = float(np.max(profile))
    best_c = cur_c
    best_h = h.copy()

    no_improve = 0
    for rnd in range(max_rounds):
        if budget_fn is not None and budget_fn() < 30:
            break
        T = t0 * (tf / t0) ** (rnd / max(max_rounds - 1, 1))
        improved = False
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        n_tries = 600
        for _ in range(n_tries):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i], h[j] = 0.0, 1.0
            new_prof = np.correlate(h, 1 - h, mode="full") * dx
            new_c = float(np.max(new_prof))
            delta = new_c - cur_c
            if delta < -1e-15:
                cur_c = new_c
                profile = new_prof
                improved = True
                if cur_c < best_c - 1e-15:
                    best_c = cur_c
                    best_h = h.copy()
                    no_improve = 0
                ones = np.where(h > 0.5)[0]
                zeros = np.where(h < 0.5)[0]
            else:
                if T > 1e-12 and rng.rand() < np.exp(-max(delta, 0.0) / (T + 1e-15)):
                    cur_c = new_c
                    profile = new_prof
                    ones = np.where(h > 0.5)[0]
                    zeros = np.where(h < 0.5)[0]
                else:
                    h[i], h[j] = 1.0, 0.0

        if not improved:
            no_improve += 1
            if no_improve >= patience:
                break

    return best_h, best_c


def smooth_max_and_grad(h, beta):
    """Compute smoothed max and its gradient."""
    n = len(h)
    dx = 2.0 / n
    profile = np.correlate(h, 1 - h, mode="full") * dx
    m = np.max(profile)
    exp_vals = np.exp(beta * (profile - m))
    w_sum = exp_vals.sum()
    weights = exp_vals / w_sum
    val = m + np.log(w_sum) / beta
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = weights[k]
        if wk < 1e-12:
            continue
        s = k - (n - 1)
        for i in range(n):
            g = 0.0
            if 0 <= i + s < n:
                g += 1.0 - h[i + s]
            if 0 <= i - s < n:
                g -= h[i - s]
            grad[i] += wk * g * dx
    return val, grad


def smoothed_max_optimize(h0, beta, maxiter, budget_fn):
    """Optimize smoothed max using projected gradient + L-BFGS-B."""
    n = len(h0)
    target = n / 2.0
    h = h0.copy()
    best_c = exact_score(h)

    def obj(x):
        v, _ = smooth_max_and_grad(x, beta)
        return v

    def obj_grad(x):
        _, g = smooth_max_and_grad(x, beta)
        return g

    t0 = exact_score(h0) + 0.01
    x0 = h.copy()
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, x0, jac=obj_grad, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.x is not None and np.isfinite(res.x).all():
            hs = repair_feasible(res.x)
            cs = exact_score(hs)
            if cs < best_c - 1e-14:
                best_c = cs
                h = hs
    except Exception:
        pass

    # Also try projected gradient with adaptive beta
    h_pg = h0.copy()
    for it in range(maxiter // 2):
        if budget_fn is not None and budget_fn() < 20:
            break
        val, grad = smooth_max_and_grad(h_pg, beta)
        grad -= grad.mean()
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        grad /= gn
        for step in [0.02, 0.01, 0.005, 0.002, 0.001]:
            h_try = repair_feasible(h_pg - step * grad)
            sc = exact_score(h_try)
            if sc < best_c - 1e-15:
                h_pg = h_try
                h = h_try
                best_c = sc
                break
    return h, best_c


def build_grad_matrix(h, n, dx):
    """Build the full gradient matrix G where G[k,i] = d(profile[k])/d(h[i])."""
    M = 2 * n - 1
    G = np.zeros((M, n))
    for k in range(M):
        s = k - (n - 1)
        for i in range(n):
            g = 0.0
            if 0 <= i + s < n:
                g += 1.0 - h[i + s]
            if 0 <= i - s < n:
                g -= h[i - s]
            G[k, i] = g * dx
    return G


def lp_subproblem(h_curr, profile, active, dx, trust, rho):
    if not HAS_CVXPY:
        return None
    n = len(h_curr)
    target = n / 2.0
    G = build_grad_matrix(h_curr, n, dx)
    G_a = G[active, :]
    b_a = profile[active] - G_a @ h_curr
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    cons = [
        h_var >= 0,
        h_var <= 1,
        cp.sum(h_var) == target,
        h_var >= h_curr - trust,
        h_var <= h_curr + trust,
        G_a @ h_var - t_var <= -b_a
    ]
    obj = cp.Minimize(t_var + rho * cp.norm1(h_var - h_curr))
    prob = cp.Problem(obj, cons)
    for solver in [cp.ECOS, cp.SCS, cp.CBC]:
        try:
            prob.solve(solver=solver, max_iters=20000, abstol=1e-10)
            break
        except Exception:
            continue
    if h_var.value is None:
        return None
    h_sol = np.array(h_var.value).flatten()
    if not np.isfinite(h_sol).all():
        return None
    return h_sol


def gradient_step(h, profile, active, n, dx, base_step):
    G = build_grad_matrix(h, n, dx)
    G_a = G[active, :]
    w = profile[active] - profile[active].min() + 1e-6
    w = w / w.sum()
    grad = G_a.T @ w
    grad -= grad.mean()
    gn = np.linalg.norm(grad)
    if gn < 1e-15:
        return None, None
    grad /= gn
    curr_c = np.max(profile)
    for s in [base_step, base_step * 0.5, base_step * 0.25, base_step * 0.1, base_step * 0.05]:
        h_try = repair_feasible(h - s * grad)
        sc = exact_score(h_try)
        if sc < curr_c - 1e-14:
            return h_try, sc
    return None, None


def active_set_loop(h0, max_iters, budget_fn):
    n = len(h0)
    dx = 2.0 / n
    h = h0.copy()
    best_c = exact_score(h)
    trust = 0.03
    stagnation = 0

    for _ in range(max_iters):
        if budget_fn() < 25:
            break
        profile = compute_profile(h)
        c_max = float(np.max(profile))
        if c_max < best_c - 1e-14:
            best_c = c_max
            stagnation = 0
        else:
            stagnation += 1

        spread = c_max - np.median(profile)
        beam = max(0.003, min(0.02, 0.2 * spread))
        active = np.where(profile >= c_max - beam)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(profile))])
        if len(active) > 20:
            active = active[np.argsort(profile[active])[-20:]]

        accepted = False
        h_lp = lp_subproblem(h, profile, active, dx, trust, 0.01)
        if h_lp is not None:
            h_lp = repair_feasible(h_lp)
            c_lp = exact_score(h_lp)
            if c_lp < best_c - 1e-14:
                h = h_lp
                best_c = c_lp
                trust = min(trust * 1.3, 0.08)
                accepted = True
            else:
                for alpha in [0.7, 0.5, 0.3]:
                    hb = repair_feasible(alpha * h_lp + (1 - alpha) * h)
                    cb = exact_score(hb)
                    if cb < best_c - 1e-14:
                        h = hb
                        best_c = cb
                        trust = min(trust * 1.1, 0.08)
                        accepted = True
                        break
                if not accepted:
                    trust *= 0.5

        if not accepted:
            hg, cg = gradient_step(h, profile, active, n, dx, trust)
            if hg is not None:
                h = hg
                best_c = cg
                accepted = True
            else:
                for b2 in [beam * 3, beam * 6]:
                    a2 = np.where(profile >= c_max - b2)[0]
                    if len(a2) > 15:
                        a2 = a2[np.argsort(profile[a2])[-15:]]
                    hg, cg = gradient_step(h, profile, a2, n, dx, 0.003)
                    if hg is not None:
                        h = hg
                        best_c = cg
                        accepted = True
                        break
        if not accepted:
            trust *= 0.4
            if trust < 0.001 or stagnation > 50:
                break

    return h, best_c


def interp_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    fx = np.concatenate([[0.0], xs, [2.0]])
    fh = np.concatenate([[h[0]], h, [h[-1]]])
    hn = np.interp(xt, fx, fh)
    return repair_feasible(np.clip(hn, 0, 1))

                    h_r3, c_r3 = full_refine(best_h, budget_fn=tl)
                    if c_r3 < best_c:
                        best_h, best_c = h_r3, c_r3

    # Phase 5: Multi-resolution
    if tl() > 180:
        for nd in [n_parent - 10, n_parent + 10, n_parent - 20, n_parent + 20,
                   n_parent - 30, n_parent + 30]:
            if tl() < 60 or nd < 30 or nd % 2 != 0:
                continue
            hi = interp_to_n(best_h, nd)
            h_bi = binarize(hi, nd)
            h_b5, c_b5 = binary_sa_search(h_bi, t0=0.04, tf=0.001, max_rounds=60,
                                          rng=rng, patience=15, budget_fn=tl)
            c_b5_interp = exact_score(interp_to_n(h_b5, nd))
            if c_b5 < best_c:
                h_interp = interp_to_n(h_b5, nd)
                if tl() > 80:
                    h_r4, c_r4 = full_refine(h_interp, budget_fn=tl)
                    if c_r4 < best_c:
                        best_h, best_c = h_r4, c_r4
                else:
                    best_h, best_c = h_interp, c_b5_interp

    # Phase 6: Final active set + gradient polish
    if tl() > 80:
        h_final, c_final = active_set_loop(best_h, max_iters=200, budget_fn=tl)
        if c_final < best_c:
            best_h, best_c = h_final, c_final

    # Phase 7: Micro gradient steps
    if tl() > 30:
        n_f = len(best_h)
        dx_f = 2.0 / n_f
        h_f = best_h.copy()
        c_f = best_c
        for _ in range(300):
            if tl() < 8:
                break
            prof = compute_profile(h_f)
            cm = float(np.max(prof))
            if cm < c_f - 1e-15:
                c_f = cm
            active = np.where(prof >= cm - 0.003)[0]
            if len(active) > 12:
                active = active[np.argsort(prof[active])[-12:]]
            hg, cg = gradient_step(h_f, prof, active, n_f, dx_f, 0.003)
            if hg is None:
                hg, cg = gradient_step(h_f, prof, active, n_f, dx_f, 0.0005)
            if hg is not None:
                h_f, c_f = hg, cg
            else:
                break
        if c_f < best_c:
            best_h, best_c = h_f, c_f

    # Final validation
    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)

    try:
        er = evaluate_erdos_solution(best_h)
        if isinstance(er, (float, int)):
            final_c = min(final_c, float(er))
        elif isinstance(er, tuple) and len(er) >= 2:
            final_c = min(final_c, float(er[0]))
    except Exception:
        pass

    return best_h, final_c, len(best_h)
