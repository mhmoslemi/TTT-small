def slsqp_solve(h, dx, target_sum, n_active=0, maxiter=200, ftol=1e-9):
    """Solve the epigraph NLP using SLSQP with exact bilinear constraints."""
    n = len(h)
    m = 2 * n - 1  # number of correlation lags
    
    # Variables: [t, h[0], h[1], ..., h[n-1]]
    x0 = np.zeros(n + 1)
    x0[0] = c5_score(h) + 1e-6  # initial t
    x0[1:] = h  # initial h

    def objective(x):
        return x[0]

    # scipy convention: constraint >= 0
    if n_active == 0 or n_active >= m:
        active_k = range(m)
    else:
        corrs = np.correlate(h, 1.0 - h, mode='full')
        active_k = np.argsort(corrs)[-n_active:]
    
    active_k = list(active_k)
    
    def make_constraint(k):
        offset = k - (n - 1)
        i_lo = max(0, -offset)
        i_hi = min(n - 1, n - 1 - offset)
        indices = np.arange(i_lo, i_hi + 1)
        partners = indices - offset  # h[partner] is the other factor
        
        def con(x):
            t = x[0]
            h_val = x[1:]
            # c_k = sum h[i] * (1 - h[i - offset]) for valid i
            # = sum (h_val[indices] * (1 - h_val[partners]))
            val = np.sum(h_val[indices] * (1.0 - h_val[partners]))
            return t - dx * val
        
        def jac(x):
            h_val = x[1:]
            j = np.zeros(n + 1)
            j[0] = 1.0
            # d/dh[i] of c_k: (1 - h[i-offset]) for i in valid range
            # d/dh[partner] of c_k: -h[i] for i in valid range
            for idx in range(len(indices)):

                p = partners[idx]
                j[i + 1] += (1.0 - h_val[p])
                j[p + 1] += (-h_val[i])
            j[1:] *= -dx
            return j
        
        return {'type': 'ineq', 'fun': con, 'jac': jac}
    
    constraints = [make_constraint(k) for k in active_k]
    
    # Equality constraint: sum(h) = target_sum
    def eq_fun(x):
        return x[1:].sum() - target_sum
    
    def eq_jac(x):
        j = np.zeros(n + 1)
        j[1:] = 1.0
        return j
    
    constraints.append({'type': 'eq', 'fun': eq_fun, 'jac': eq_jac})
    
    # Bounds
    bounds = [(0.0, 1.0)] + [(0.0, 1.0) for _ in range(n)]

            return h_new, c5_new
    except Exception:
        pass
    return h, c5_score(h)
