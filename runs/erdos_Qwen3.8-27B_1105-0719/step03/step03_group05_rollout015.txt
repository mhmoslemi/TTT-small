def pair_optimal_t(h, i, j):
    n = len(h)
    dx = 2.0 / n
    a, b = h[i], h[j]
    t_lo = max(-a, b - 1.0)
    t_hi = min(1.0 - a, b)
    if t_hi - t_lo < 1e-12:
        return 0.0
    
    def profile_at_t(t):
        h_new = h.copy()
        h_new[i] = a + t
        h_new[j] = b - t
        return np.max(np.correlate(h_new, 1 - h_new, mode="full") * dx)
    
    # Golden section search on [t_lo, t_hi]
    # Since it's convex, this will converge
    phi = (np.sqrt(5.0) - 1.0) / 2.0
    a_g, b_g = t_lo, t_hi
    c_g = b_g - phi * (b_g - a_g)
    d_g = a_g + phi * (b_g - a_g)
    f_c = profile_at_t(c_g)
    f_d = profile_at_t(d_g)
    
    best_t = c_g if f_c < f_d else d_g
    best_f = min(f_c, f_d)
    
    for _ in range(40):
        if f_c < f_d:
            b_g = d_g
            d_g = c_g
            f_d = f_c
            c_g = b_g - phi * (b_g - a_g)
            f_c = profile_at_t(c_g)
        else:
            a_g = c_g
            c_g = d_g
            f_c = f_d
            d_g = a_g + phi * (b_g - a_g)

        if cur_f < best_f:
            best_f = cur_f
            best_t = cur_best
        if b_g - a_g < 1e-10:

def repair_feasible(vec):

def pair_optimal_t(h, i, j):
    n = len(h)
    dx = 2.0 / n
    a, b = float(h[i]), float(h[j])
    t_lo = max(-a, b - 1.0)
    t_hi = min(1.0 - a, b)
    if t_hi - t_lo < 1e-14:
        return 0.0, exact_score(h)
    
    # Compute current profile
    P = np.correlate(h, 1.0 - h, mode="full") * dx  # length 2n-1
    
    # Compute slopes for all lags
    # Slope of profile[k] w.r.t. t (where t: h[i]+t, h[j]-t)
    # = slope_at(lag, i) - slope_at(lag, j)
    nM = 2 * n - 1
    slopes = np.zeros(nM)
    for k in range(nM):
        d = k - (n - 1)
        si = slope_val(d, i, h, n)
        sj = slope_val(d, j, h, n)
        slopes[k] = si - sj
    # Scale slopes by dx (since P already has dx factor)
    # Wait, P = correlate * dx, and slope is for the correlate (without dx)
    # So actual slope of P[k] w.r.t. t = dx * (si - sj)
    # Let me redefine: P[k] = correlate[k] * dx
    # dP[k]/dt = dx * (si - sj)
    slopes *= dx
    
    # Now minimize max_k (P[k] + t * slopes[k]) for t in [t_lo, t_hi]
    # This is a 1D convex optimization
    # Golden section search
    phi = (np.sqrt(5.0) - 1.0) / 2.0
    gl, gh = t_lo, t_hi
    c = gh - phi * (gh - gl)
    d = gl + phi * (gh - gl)
    fc = float(np.max(P + c * slopes))
    fd = float(np.max(P + d * slopes))
    best_t = 0.0
    best_f = float(np.max(P))
    
    for _ in range(60):
        if gh - gl < 1e-13:
            break
        if fc < fd:
            gh = d
            d = c
            fd = fc
            c = gh - phi * (gh - gl)
            fc = float(np.max(P + c * slopes))
        else:
            gl = c
            c = d
            fc = fd
            d = gl + phi * (gh - gl)
            fd = float(np.max(P + d * slopes))
        cur_f = min(fc, fd)
        if cur_f < best_f:
            best_f = cur_f
            best_t = c if fc < fd else d
    
    return best_t, best_f

    """Return array of length 2n-1 with slope of each profile component w.r.t. h[i]"""
    slopes = np.zeros(2*n - 1)

        if 0 <= i <= n-1-d:
            s += 1.0 - h[i+d]
        if d <= i <= n-1:

        s = 0.0
        if L <= i <= n-1:
            s += 1.0 - h[i-L]
        if 0 <= i <= n-1-L:
            s -= h[i+L]
        slopes[k] = s
    return slopes

- The current max is at some lag d*.
- To reduce profile[d*], we want to decrease the "high" contributions and increase the "low" ones.

- Compute the "gradient" of the max profile: for each coordinate h[k], how much does increasing it change the max?

- Coordinates with negative gradient (decreasing h[k] helps) are "high" candidates
- Coordinates with positive gradient (increasing h[k] helps) are "low" candidates
- Good pairs: (high, low)

The gradient can be approximated as:

    """Approximate gradient of max profile w.r.t. h."""
    profile = compute_profile(h)
    c_max = float(np.max(profile))
    beta = 50.0
    weights = np.exp(beta * (profile - c_max))
    weights /= weights.sum()
    
    grad = np.zeros(n)
    for k in range(n):
        s = 0.0
        for d_idx in range(2*n-1):
            d = d_idx - (n-1)
            sv = slope_at_lag(d, k, h, n)
            s += weights[d_idx] * sv
        grad[k] = s * (2.0/n)  # dx factor
    # Center the gradient (sum constraint means we can add any constant)
    grad -= grad.mean()
    return grad
