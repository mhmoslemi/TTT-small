        # Second position: h[i+k] for i = 0..n-1-k, i.e., h[j] for j = k..n-1
        g[k:] -= dx * h[:n-k]
    return g

            g2[k:] += dx * (1.0 - h[:n-k])

    """Take a subgradient step using the average gradient of active constraints."""
    n = len(h)
    center = n - 1

        if lag == 0:
            g = dx * (1.0 - 2.0 * h)
        elif lag > 0:
            g = np.zeros(n)
            g[:n-lag] += dx * (1.0 - h[lag:])
            g[lag:] -= dx * h[:n-lag]
        else:
            lag = -lag
            g = np.zeros(n)
            g[lag:] += dx * (1.0 - h[:n-lag])
            g[:n-lag] -= dx * h[lag:]
        grad += g

- Let F be the set of free variables (0 < h[i] < 1)
- The condition is: for all i ∈ F, sum_{k∈A} λ_k * ∂g_k(h)/∂h[i] = 0
- Plus: sum_{k∈A} λ_k = 1 (normalization)
- Plus: λ_k ≥ 0

This is a system of |F| + 1 equations in m unknowns (λ_k). If m > |F| + 1, it's overdetermined. If m = |F| + 1, it's exactly determined.

In practice, we can:
1. Guess the active set A and free set F
2. Solve the KKT system
3. Check if the solution is feasible (λ ≥ 0, 0 ≤ h ≤ 1)
4. If not, adjust the active/free sets and try again

def kkt_solve(h, active, n):
    """Try to solve the KKT conditions for the given active set."""
    dx = 2.0 / n
    center = n - 1
    m = len(active)
    if m == 0:
        return None
    
    # Free set: coordinates where 0 < h[i] < 1
    free = np.where((h > 1e-8) & (h < 1 - 1e-8))[0]
    n_free = len(free)
    
    if n_free == 0:
        return None
    
    # Build the system: for each free variable i, sum_k lambda_k * dg_k/dh[i] = 0
    # Plus: sum_k lambda_k = 1
    # Total: (n_free + 1) equations, m unknowns
    
    if m < n_free + 1:
        # Underdetermined: use least squares
        pass
    
    # Build matrix
    A_mat = np.zeros((n_free + 1, m))
    for idx, i in enumerate(free):
        for k_idx, k in enumerate(active):
            lag = k - center
            if lag == 0:
                A_mat[idx, k_idx] = dx * (1.0 - 2.0 * h[i])
            elif lag > 0:
                # dg_k/dh[i] = dx*(1-h[i+lag]) if i <= n-1-lag, minus dx*h[i-lag] if i >= lag
                val = 0.0
                if i <= n - 1 - lag:
                    val += dx * (1.0 - h[i + lag])
                if i >= lag:
                    val -= dx * h[i - lag]
                A_mat[idx, k_idx] = val
            else:
                lag = -lag
                val = 0.0
                if i >= lag:
                    val += dx * (1.0 - h[i - lag])
                if i <= n - 1 - lag:
                    val -= dx * h[i + lag]
                A_mat[idx, k_idx] = val
    
    # Normalization constraint
    A_mat[n_free, :] = 1.0
    b_vec = np.zeros(n_free + 1)
    
    # Solve least squares (works for both underdetermined and overdetermined)
    lam, residuals, rank, sv = np.linalg.lstsq(A_mat, b_vec, rcond=None)
    
    # Check non-negativity
    if np.min(lam) < -1e-10:
        return None
    
    lam = np.clip(lam, 0, None)
    lam /= lam.sum() + 1e-30
    
    return lam
