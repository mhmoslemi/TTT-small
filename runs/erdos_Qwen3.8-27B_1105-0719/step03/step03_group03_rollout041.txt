```python
def logsumexp_gradient(h, tau):
    n = len(h)
    dx = 2.0 / n
    profile = compute_profile(h)
    m = profile.max()
    weights = np.exp(tau * (profile - m))
    weights /= weights.sum()
    
    center = n - 1
    grad = np.full(n, dx)  # the "1" part: dx * sum(w_m) = dx
    
    # Subtract dx * sum_m w_m * h[j - d_m]
    # d_m = m - center
    for d in range(-(n-1), n):
        w = weights[center + d]  # weight for lag d
        if w < 1e-15:
            continue
        # h[j-d] for j in range(n): valid when 0 <= j-d < n
        if d >= 0:
            # j-d >= 0 means j >= d; j-d < n means j < n+d (always true for j < n)
            for j in range(d, n):
                grad[j] -= dx * w * h[j - d]
        else:
            # j-d >= 0 means j >= d = -|d|; always true
            # j-d < n means j < n+d = n-|d|
            for j in range(0, n + d):
                grad[j] -= dx * w * h[j - d]
    
    # Subtract dx * sum_m w_m * h[j + d_m] (same as above by symmetry)
    for d in range(-(n-1), n):
        w = weights[center + d]
        if w < 1e-15:
            continue
        if d >= 0:
            for j in range(0, n - d):
                grad[j] -= dx * w * h[j + d]
        else:
            for j in range(-d, n):
                grad[j] -= dx * w * h[j + d]
    
    # Project to tangent space
    grad -= grad.mean()
    return grad

profile[m] = dx * Σ_i h[i] * (1 - h[i-d])

∂/∂h[j]: the term h[j] appears in the sum when i=j, giving (1-h[j-d]).
Also, h[j] appears in the "1-h[i-d]" part when i-d=j, i.e., i=j+d, giving -h[j+d].

So ∂profile[m]/∂h[j] = dx * (1-h[j-d]) - dx * h[j+d] = dx * (1 - h[j-d] - h[j+d])

Yes, that's what I had before.

Now for the log-sum-exp:
f(h) = (1/τ) log(Σ_m exp(τ * profile[m]))

∂f/∂h[j] = Σ_m (exp(τ*profile[m]) / Σ_m exp(τ*profile[m])) * ∂profile[m]/∂h[j]
= Σ_m p_m * dx * (1 - h[j-d_m] - h[j+d_m])
= dx * (Σ_m p_m) - dx * Σ_m p_m * h[j-d_m] - dx * Σ_m p_m * h[j+d_m]
= dx * (1 - Σ_m p_m * h[j-d_m] - Σ_m p_m * h[j+d_m])

Now, Σ_m p_m * h[j-d_m]: let's substitute m = (j-d) + (n-1), so d_m = j-d... no that's circular.

Let me use a different notation. Let d_m = m - (n-1) for m = 0, ..., 2n-2. So d_m ranges from -(n-1) to n-1.

Σ_m p_m * h[j-d_m] = Σ_{d=-(n-1)}^{n-1} p_{d+(n-1)} * h[j-d] (where h[j-d] = 0 if j-d ∉ [0,n))

This is a cross-correlation of h with the weight sequence p. Let's define q[d] = p_{d+(n-1)} for d = -(n-1), ..., n-1.

Then Σ_m p_m * h[j-d_m] = Σ_d q[d] * h[j-d] = (q ⋆ h)[j] (cross-correlation)

Similarly, Σ_m p_m * h[j+d_m] = Σ_d q[d] * h[j+d] = Σ_d q[-d] * h[j-d'] (substituting d' = -d) = (q_flipped ⋆ h)[j]

where q_flipped[d] = q[-d].

But since the profile is symmetric (profile[m] = profile[2(n-1)-m]), we have p_m = p_{2(n-1)-m}, which means q[d] = q[-d]. So q is symmetric, and both cross-correlations are the same.

Therefore: ∂f/∂h[j] = dx * (1 - 2 * (q ⋆ h)[j])

where (q ⋆ h)[j] = Σ_d q[d] * h[j-d] (sum over d where j-d is in [0,n)).

This can be computed as a convolution: if we zero-pad h and q appropriately, it's just a single convolution.

For n=70, q has length 139 (from d=-69 to d=69). h has length 70. The cross-correlation can be computed with np.correlate or np.convolve in O(n * len(q)) = O(n^2) time, or with FFT in O(n log n).

For n=70, O(n^2) is only 4900 operations, so no need for FFT.

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))


def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx


def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    # Water-filling approach
    for _ in range(100):
        if diff > 1e-14:
            mask = h < 1.0
            if not mask.any():
                break
            add = diff / mask.sum()
            h[mask] = np.minimum(h[mask] + add, 1.0)
        elif diff < -1e-14:
            mask = h > 0.0
            if not mask.any():
                break
            sub = -diff / mask.sum()
            h[mask] = np.maximum(h[mask] - sub, 0.0)
        else:
            break
        diff = target - h.sum()
    # Final adjustment
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = np.where(h < 1.0)[0]
            if len(idx) > 0:
                h[idx[0]] += min(diff, 1.0 - h[idx[0]])
        else:
            idx = np.where(h > 0.0)[0]
            if len(idx) > 0:
                h[idx[0]] += max(diff, -h[idx[0]])
    h = np.clip(h, 0.0, 1.0)
    return h


def lse_objective(h, tau):
    profile = compute_profile(h)
    m = profile.max()
    shifted = tau * (profile - m)
    max_s = shifted.max()
    if max_s > 500:
        return m + (max_s / tau)
    return m + np.log(np.sum(np.exp(shifted))) / tau


def lse_gradient(h, tau):
    n = len(h)
    dx = 2.0 / n
    profile = compute_profile(h)
    m = profile.max()
    shifted = tau * (profile - m)
    max_s = shifted.max()
    if max_s > 500:
        # Essentially the max is attained at one point
        active = np.argmax(profile)
        d = active - (n - 1)
        grad = np.full(n, dx)
        for j in range(n):
            if 0 <= j - d < n:
                grad[j] -= dx * h[j - d]
            if 0 <= j + d < n:
                grad[j] -= dx * h[j + d]
    else:
        weights = np.exp(shifted - max_s)
        weights /= weights.sum()
        center = n - 1
        # Compute cross-correlation of h with weights
        # (q ⋆ h)[j] = sum_d q[d] * h[j-d]
        # where q[d] = weights[center + d]
        conv = np.zeros(n)
        for j in range(n):
            s = 0.0
            for d in range(-(n-1), n):
                val = j - d
                if 0 <= val < n:
                    s += weights[center + d] * h[val]
            conv[j] = s
        grad = dx * (1.0 - 2.0 * conv)
    
    grad -= grad.mean()
    return grad


def smooth_slsqp(h0, tau, maxiter=300, tol=1e-14):
    """Optimize the log-sum-exp smoothed objective using SLSQP."""
    n = len(h0)
    target = n / 2.0
    
    def obj(h):
        return lse_objective(h, tau)
    
    def obj_grad(h):
        return lse_gradient(h, tau)
    
    def eq_fun(h):
        return h.sum() - target
    
    def eq_jac(h):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    bounds = [(0.0, 1.0)] * n
    
    try:
        res = minimize(obj, h0, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': tol})
        if res.x is not None and np.isfinite(res.x).all():
            hs = repair_feasible(res.x)
            if np.isfinite(exact_score(hs)):

    except Exception:
        pass
    return None

    """Refine using exact subgradient of the max."""
    n = len(h0)
    dx = 2.0 / n
    h = h0.copy()
    best_c = exact_score(h)
    
    step_size = 0.005
    for step in range(steps):
        if budget_fn is not None and budget_fn() < 15:
            break
        profile = compute_profile(h)
        c_max = profile.max()
        if c_max < best_c - 1e-15:
            best_c = c_max
        
        # Find active set (lags within small tolerance of max)
        tol_active = max(0.001, 0.01 * (c_max - profile.min()))
        active = np.where(profile >= c_max - tol_active)[0]

        grad = np.zeros(n)

            d = idx - (n - 1)
            g = np.full(n, dx)
            # g[j] -= dx * h[j-d] (if in range)

                g[:n+d] -= dx * h[-d:]

                g[:-d] -= dx * h[d:]
            else:
                g[-d:] -= dx * h[:n+d]
            grad += g
        grad /= len(active)
        grad -= grad.mean()
        
        # Adaptive step size
        for s in [step_size, step_size*0.5, step_size*0.25, step_size*0.1]:

def binary_start(n, rng):
    """Create a random binary-like starting point."""
    # Choose n/2 positions to be 1, rest 0

    h[ones] = 1.0

def fourier_start(n, rng, max_freq=5):
    """Create a starting point using low-frequency Fourier modes."""
    h = np.full(n, 0.5)
    for k in range(1, max_freq+1):
        amp = rng.uniform(0.05, 0.2) / k
        phase = rng.uniform(0, 2*np.pi)
        x = (np.arange(n) + 0.5) * (2.0/n)
        h += amp * np.cos(2*np.pi*k*x/2.0 + phase)
    return repair_feasible(np.clip(h, 0, 1))


def interp_to(h, new_n):
    """Interpolate h from its current resolution to new_n points."""
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    # Use linear interpolation with clamped boundaries
    hn = np.interp(xt, xs, h)
    return repair_feasible(np.clip(hn, 0, 1))


def active_set_lp(h0, profile, active, trust=0.03):
    """Solve LP restricted to active constraints to reduce the max."""
    if not HAS_CVXPY:
        return None
    n = len(h0)
    dx = 2.0 / n
    target = n / 2.0
    center = n - 1
    
    # Build gradient matrix for active constraints
    M = len(active)
    G = np.zeros((M, n))
    for m_idx, orig_idx in enumerate(active):
        d = orig_idx - center
        g = np.full(n, dx)
        if d >= 0:
            g[d:] -= dx * h0[:n-d]
        else:
            g[:n+d] -= dx * h0[-d:]
        if d >= 0:
            g[:-d] -= dx * h0[d:]
        else:
            g[-d:] -= dx * h0[:n+d]
        G[m_idx, :] = g
    
    # Linearize: profile[m] + G[m,:] @ (h_var - h0) <= t
    # => G[m,:] @ h_var - t <= profile[m] - G[m,:] @ h0
    b = profile[active] - G @ h0
    c_max = profile.max()
    
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    cons = [
        h_var >= 0,
        h_var <= 1,
        cp.sum(h_var) == target,
        h_var >= h0 - trust,
        h_var <= h0 + trust,
    ]
    for m_idx in range(M):
        cons.append(G[m_idx] @ h_var - t_var <= b[m_idx])
    
    obj = cp.Minimize(t_var + 0.01 * cp.sum(cp.abs(h_var - h0)))
    prob = cp.Problem(obj, cons)
    
    for solver in [cp.ECOS, cp.SCS, cp.CBC, cp.GLPK, cp.PDLP]:
        try:
            prob.solve(solver=solver, max_iters=5000, abstol=1e-10)
            break
        except Exception:
            continue
    
    if h_var.value is None:
        return None
    h_sol = np.array(h_var.value).flatten()
    if not np.isfinite(h_sol).all():
        return None
    return repair_feasible(h_sol)

def lse_gradient(h, tau):
    n = len(h)
    dx = 2.0 / n
    profile = compute_profile(h)
    m = profile.max()
    shifted = tau * (profile - m)
    max_s = shifted.max()
    if max_s > 500:
        active = np.argmax(profile)
        d = active - (n - 1)
        grad = np.full(n, dx)
        for j in range(n):
            if 0 <= j - d < n:
                grad[j] -= dx * h[j - d]
            if 0 <= j + d < n:
                grad[j] -= dx * h[j + d]
    else:
        weights = np.exp(shifted - max_s)
        weights /= weights.sum()
        center = n - 1
        # conv[j] = sum_d weights[center+d] * h[j-d] (with h[out]=0)
        # This is: sum over valid (j-d) of weights[center+d] * h[j-d]
        # = sum over valid i of weights[center + (j-i)] * h[i]
        # where i = j-d, so d = j-i, and we need 0 <= center+j-i < 2n-1
        conv = np.zeros(n)
        for j in range(n):
            s = 0.0
            # i ranges over 0..n-1, d = j-i ranges over j-(n-1)..j
            for i in range(n):
                d = j - i
                idx = center + d  # index into weights
                if 0 <= idx < 2*n-1:
                    s += weights[idx] * h[i]
            conv[j] = s
        grad = dx * (1.0 - 2.0 * conv)
    
    grad -= grad.mean()
    return grad
```

import numpy as np
from scipy.optimize import minimize
import time

    import cvxpy as cp
    HAS_CVXPY = True

    HAS_CVXPY = False


def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))


def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx


def repair_feasible(vec):
    n = len(vec)
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(200):
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if not mask.any():
            break
        amt = diff / mask.sum()
        h[mask] += amt
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = int(np.argmax(1.0 - h))
            h[idx] = min(1.0, h[idx] + diff)
        else:
            idx = int(np.argmax(h))
            h[idx] = max(0.0, h[idx] + diff)
    return h


def lse_objective(h, tau):
    profile = compute_profile(h)
    m = profile.max()
    shifted = tau * (profile - m)
    ms = shifted.max()
    if ms > 500:
        return m + ms / tau
    return m + np.log(np.sum(np.exp(shifted - ms))) / tau + ms / tau


def lse_gradient(h, tau):
    n = len(h)
    dx = 2.0 / n
    profile = compute_profile(h)
    m = profile.max()
    shifted = tau * (profile - m)
    ms = shifted.max()
    if ms > 500:
        active = int(np.argmax(profile))
        d = active - (n - 1)
        grad = np.full(n, dx)
        for j in range(n):
            if 0 <= j - d < n:
                grad[j] -= dx * h[j - d]
            if 0 <= j + d < n:
                grad[j] -= dx * h[j + d]
    else:
        weights = np.exp(shifted - ms)
        weights /= weights.sum()
        center = n - 1
        conv = np.zeros(n)
        for j in range(n):
            s = 0.0
            for i in range(n):
                d = j - i
                idx = center + d
                if 0 <= idx < 2 * n - 1:
                    s += weights[idx] * h[i]
            conv[j] = s
        grad = dx * (1.0 - 2.0 * conv)
    grad -= grad.mean()
    return grad


def smooth_slsqp(h0, tau, maxiter=300, tol=1e-14):
    n = len(h0)
    target = n / 2.0

    def eq_fun(h):
        return h.sum() - target

    def eq_jac(h):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(lse_objective, h0, args=(tau,),
                       jac=lambda h: lse_gradient(h, tau),
                       method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': tol})
        if res.x is not None and np.isfinite(res.x).all():
            hs = repair_feasible(res.x)
            if np.isfinite(exact_score(hs)):
                return hs
    except Exception:
        pass
    return None


def subgradient_refine(h0, steps=300, budget_fn=None):
    n = len(h0)
    dx = 2.0 / n
    h = h0.copy()
    best_c = exact_score(h)
    step_size = 0.004
    stall = 0

    for step in range(steps):
        if budget_fn is not None and budget_fn() < 15:
            break
        profile = compute_profile(h)
        c_max = profile.max()
        if c_max < best_c - 1e-15:
            best_c = c_max
            stall = 0
        else:
            stall += 1
            if stall > 100 and step_size < 1e-5:
                break

        spread = c_max - profile.min()
        tol_a = max(0.0005, 0.02 * spread)
        active = np.where(profile >= c_max - tol_a)[0]

        grad = np.zeros(n)
        for idx in active:
            d = idx - (n - 1)
            g = np.full(n, dx)
            if d >= 0:
                g[d:] -= dx * h[:n - d]
                g[:-d] -= dx * h[d:]
            else:
                g[:n + d] -= dx * h[-d:]
                g[-d:] -= dx * h[:n + d]
            grad += g
        grad /= len(active)
        grad -= grad.mean()

        accepted = False
        for s in [step_size, step_size * 0.5, step_size * 0.25, step_size * 0.1]:
            h_try = repair_feasible(h - s * grad)
            c_try = exact_score(h_try)
            if c_try < best_c - 1e-15:
                h = h_try
                best_c = c_try
                step_size = min(step_size * 1.05, 0.015)
                accepted = True
                break
        if not accepted:
            step_size *= 0.7

    return h, best_c


def active_set_lp(h0, profile, active, trust=0.03):
    if not HAS_CVXPY:
        return None
    n = len(h0)
    dx = 2.0 / n
    target = n / 2.0
    center = n - 1
    M = len(active)
    G = np.zeros((M, n))
    for m_idx, orig_idx in enumerate(active):
        d = orig_idx - center
        g = np.full(n, dx)
        if d >= 0:
            g[d:] -= dx * h0[:n - d]
            g[:-d] -= dx * h0[d:]
        else:
            g[:n + d] -= dx * h0[-d:]
            g[-d:] -= dx * h0[:n + d]
        G[m_idx, :] = g

    b = profile[active] - G @ h0
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    cons = [
        h_var >= 0,
        h_var <= 1,
        cp.sum(h_var) == target,
        h_var >= h0 - trust,
        h_var <= h0 + trust,
    ]
    for m_idx in range(M):
        cons.append(G[m_idx] @ h_var - t_var <= b[m_idx])

    obj = cp.Minimize(t_var + 0.005 * cp.sum(cp.abs(h_var - h0)))
    prob = cp.Problem(obj, cons)
    for solver in [cp.ECOS, cp.SCS, cp.CBC, cp.PDLP, cp.GLPK]:
        try:
            prob.solve(solver=solver, max_iters=10000, abstol=1e-10)
            break
        except Exception:
            continue
    if h_var.value is None:
        return None
    h_sol = np.array(h_var.value).flatten()
    if not np.isfinite(h_sol).all():
        return None
    return repair_feasible(h_sol)


def pair_transfer(h0, n_pairs=2000, rng=None, budget_fn=None):
    if rng is None:
        rng = np.random.RandomState(0)
    n = len(h0)
    h = h0.copy()
    best_c = exact_score(h)
    for _ in range(n_pairs):
        if budget_fn is not None and budget_fn() < 10:
            break
        i, j = rng.choice(n, 2, replace=False)
        for delta in [0.01, 0.02, 0.03, -0.01, -0.02, -0.03]:
            h_try = h.copy()
            h_try[i] += delta
            h_try[j] -= delta
            if 0 <= h_try[i] <= 1 and 0 <= h_try[j] <= 1:
                c_try = exact_score(h_try)
                if c_try < best_c - 1e-14:
                    best_c = c_try
                    h = h_try
                    break
    return h, best_c


def interp_to(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    hn = np.interp(xt, xs, h)
    return repair_feasible(np.clip(hn, 0, 1))


def binary_start(n, rng):
    ones = rng.choice(n, size=n // 2, replace=False)
    h = np.zeros(n)
    h[ones] = 1.0
    h = 0.7 * h + 0.15
    return repair_feasible(h)


def fourier_start(n, rng, max_freq=4):
    h = np.full(n, 0.5)
    x = (np.arange(n) + 0.5) * (2.0 / n)

        phase = rng.uniform(0, 2 * np.pi)

        h_ref, c_ref = subgradient_refine(best_h, steps=500, budget_fn=tl)
        if c_ref < best_c - 1e-14:
            best_h, best_c = h_ref, c_ref

    # Phase 7: One more LP round
    if tl() > 40:
        h = best_h.copy()
        trust = 0.02
        for _ in range(100):
            if tl() < 20:
                break
            profile = compute_profile(h)
            c_max = profile.max()
            active = np.where(profile >= c_max - 0.003)[0]
            if len(active) > 10:
                active = active[np.argsort(profile[active])[-10:]]
            h_lp = active_set_lp(h, profile, active, trust)
            if h_lp is not None:
                c_lp = exact_score(h_lp)
                if c_lp < best_c - 1e-14:
                    h = h_lp
                    best_h = h.copy()
                    best_c = c_lp
                    trust = min(trust * 1.15, 0.05)
                else:
                    trust *= 0.6
                    if trust < 0.002:
                        break
            else:
                break

    # Final
    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)

    try:
        er = evaluate_erdos_solution(best_h)
        if isinstance(er, (float, int)):
            final_c = min(final_c, float(er))
        elif isinstance(er, tuple) and len(er) >= 2:
            final_c = min(final_c, float(er[0]))
    except Exception:
        pass

    return (best_h, final_c, len(best_h))
