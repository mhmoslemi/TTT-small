def project_feasible(x, target_sum):
    """Project x onto {h : 0 <= h <= 1, sum(h) = target_sum}."""
    n = len(x)
    x = np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0)
    x = np.clip(x, -10.0, 10.0)
    lo, hi = -10.0, 10.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            mask = h < 1.0 - 1e-13
        else:
            mask = h > 1e-13
        cnt = mask.sum()
        if cnt > 0:
            h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def c5_val(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * (2.0 / n)

def raw_corrs(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def grad_lag(h, d):
    """Gradient of lag-d raw correlation (before *dx)."""
    n = len(h)
    g = np.zeros(n)
    if d >= 0:
        nd = n - d
        if nd <= 0:
            return g
        i = np.arange(nd)
        g[i] += 1.0 - h[i + d]
        g[i + d] += -h[i]
    else:
        ad = -d
        i = np.arange(ad, n)
        g[i] += 1.0 - h[i + d]
        g[i + d] += -h[i]
    return g

def compute_subgrad(h, active_tol=0.005):
    """Compute subgradient of max corr (normalized by dx) at h."""
    n = len(h)
    dx = 2.0 / n
    c = raw_corrs(h)
    C = c.max()
    active = c >= C - active_tol * dx
    g = np.zeros(n)
    count = 0
    for k in range(len(c)):
        if active[k]:
            d = k - (n - 1)
            g += grad_lag(h, d)
            count += 1
    if count > 0:
        g /= count
    return g * dx, C, active.sum()

    best_h = h

            best_h = h_new

def lsq_gradient(h, lam):
    """Gradient of the log-sum-exp smoothed objective."""
    n = len(h)
    dx = 2.0 / n
    c = raw_corrs(h)
    # Softmax weights
    c_shifted = lam * c - lam * c.max()
    w = np.exp(c_shifted)
    w /= w.sum()
    # Weighted gradient
    g = np.zeros(n)
    for k in range(len(c)):
        if w[k] < 1e-15:
            continue
        d = k - (n - 1)
        g += w[k] * grad_lag(h, d)
    return g * dx

def lp_step(h, dx, cts, n_active, alpha):
    """LP step on active constraints. Returns (h_new, c5_new, improved)."""
    n = len(h)
    corrs = raw_corrs(h)
    m = 2 * n - 1
    current_c5 = corrs.max()
    
    if n_active >= m:
        active_idx = np.arange(m)
    else:
        active_idx = np.argsort(corrs)[-n_active:]
    
    # Build LP: minimize z (step in t direction)
    # Variables: delta_h (n vars) + z (1 var)
    # Constraints: grad_k . delta_h - z <= -c_k for k in active
    #              sum(delta_h) = 0
    #              -alpha*h <= delta_h <= alpha*(1-h)
    
    m_act = len(active_idx)
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    A_ub = np.empty((m_act, n + 1))
    A_ub[:, -1] = -1.0
    b_ub = np.empty(m_act)
    
    for row, k in enumerate(active_idx):
        d = k - (n - 1)
        A_ub[row, :n] = -grad_lag(h, d) * dx
        b_ub[row] = -corrs[k]
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    bounds = []
    for i in range(n):
        bounds.append((-alpha * h[i], alpha * (1.0 - h[i])))
    bounds.append((-alpha, alpha))
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status != 0:
            return h, current_c5, False
        delta = res.x[:n]
    except Exception:
        return h, current_c5, False
    
    if np.abs(delta).max() < 1e-15:
        return h, current_c5, False
    
    # Line search along delta
    h_best = h
    c5_best = current_c5
    for s in [1.0, 0.7, 0.5, 0.35, 0.2, 0.1, 0.05]:
        h_try = project_feasible(h + s * delta, cts)
        c5_try = c5_val(h_try)
        if c5_try < c5_best - 1e-14:
            c5_best = c5_try
            h_best = h_try
    return h_best, c5_best, c5_best < current_c5 - 1e-14
