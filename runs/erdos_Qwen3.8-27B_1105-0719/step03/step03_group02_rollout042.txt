As I discussed, each corrs_k is a quadratic in τ, so f(τ) is the max of quadratics, which is not necessarily convex. But in practice, for the FW step, the improvement should be monotone decreasing (since we're moving in a descent direction). Let me verify:

But is it monotonically decreasing on [0,1]? Not necessarily, since f is max of quadratics. However, in practice, the FW method works well even without monotonically decreasing steps, as long as we take the best τ.

```python
def fw_step(h, n, dx, target_sum):
    # Compute current correlations
    A = compute_corrs(h)  # A = corr_k(h)
    k_star = A.argmax()
    c5_current = A.max() * dx
    
    # Compute gradient at k_star
    g = grad_corr_k(h, k_star, n)
    
    # Project g: we want to minimize <g, v> over P = {v: sum(v)=n/2, 0≤v≤1}
    # This is: sort g, set v=1 for smallest, v=0 for largest
    sorted_idx = np.argsort(g)
    v = np.zeros(n)
    v[sorted_idx[:int(target_sum)]] = 1.0
    # Handle the fractional part
    frac = target_sum - int(target_sum)
    if frac > 0 and int(target_sum) < n:
        v[sorted_idx[int(target_sum)]] = frac
    
    # Direction d = v - h
    d = v - h
    
    # Precompute for line search
    B_arr = n - np.correlate(h, v, 'full') - np.correlate(v, h, 'full')
    C_arr = compute_corrs(v)
    
    # Line search: minimize max_k [(1-τ)² A_k + τ(1-τ) B_k + τ² C_k] for τ ∈ [0,1]
    best_tau = 0.0
    best_val = A.max()
    
    # Coarse search
    taus = np.linspace(0, 1, 21)
    for tau in taus:
        corrs_tau = (1-tau)**2 * A + tau*(1-tau) * B_arr + tau**2 * C_arr
        val = corrs_tau.max()
        if val < best_val:
            best_val = val
            best_tau = tau
    
    # Fine search around best_tau
    if 0 < best_tau < 1:
        # Golden section
        gr = (np.sqrt(5) - 1) / 2
        lo = max(0, best_tau - 0.05)
        hi = min(1, best_tau + 0.05)
        a = lo + (1-gr)*(hi-lo)
        b = lo + gr*(hi-lo)
        fa = ((1-a)**2 * A + a*(1-a)*B_arr + a**2 * C_arr).max()
        fb = ((1-b)**2 * A + b*(1-b)*B_arr + b**2 * C_arr).max()
        for _ in range(60):
            if fa < fb:
                hi = b; b = a; fb = fa
                a = lo + (1-gr)*(hi-lo)
                fa = ((1-a)**2 * A + a*(1-a)*B_arr + a**2 * C_arr).max()
            else:
                lo = a; a = b; fa = fb
                b = lo + gr*(hi-lo)
                fb = ((1-b)**2 * A + b*(1-b)*B_arr + b**2 * C_arr).max()
        best_tau = (lo + hi) / 2
        best_val = ((1-best_tau)**2 * A + best_tau*(1-best_tau)*B_arr + best_tau**2 * C_arr).max()
    
    if best_val < A.max() - 1e-15:
        h_new = (1 - best_tau) * h + best_tau * v
        return h_new, best_val * dx, True
    return h, A.max() * dx, False
```

The maximum of concave parabolas and lines is not necessarily convex or concave, but it's a well-behaved function. A golden section search should work well in practice (the function is typically unimodal over the feasible interval for the FW/pair transfer step).

```python
            continue

            best_val = v_max
            best_tau = tau

        tau_grid = np.linspace(0, 1, 21)
        # Find the best interval
        best_i = np.argmin([( (1-t)**2 * A + t*(1-t)*B_arr + t**2*C_arr).max() for t in tau_grid])
        lo = tau_grid[max(0, best_i-1)]
        hi = tau_grid[min(len(tau_grid)-1, best_i+1)]
        
        # Golden section
        gr = (np.sqrt(5.0) - 1.0) / 2.0

        b = lo + gr*(hi-lo)
        fa = ((1-a)**2 * A + a*(1-a)*B_arr + a**2*C_arr).max()
        fb = ((1-b)**2 * A + b*(1-b)*B_arr + b**2*C_arr).max()

                a = lo + (1-gr)*(hi-lo)
                fa = ((1-a)**2 * A + a*(1-a)*B_arr + a**2*C_arr).max()
            else:
                lo, a, fa = a, b, fb
                b = lo + gr*(hi-lo)
                fb = ((1-b)**2 * A + b*(1-b)*B_arr + b**2*C_arr).max()
        best_tau = (lo+hi)/2
        best_val = ((1-best_tau)**2 * A + best_tau*(1-best_tau)*B_arr + best_tau**2*C_arr).max()
    
    if best_val < c5_cur - 1e-14:
        h_new = (1-best_tau) * h + best_tau * v
        return h_new, best_val * dx, True
    return h, c5_cur * dx, False
```
