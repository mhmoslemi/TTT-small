def repair_feasible(h):
    h = np.asarray(h, dtype=np.float64).copy()
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    target = 0.5 * n
    for _ in range(8):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)[::-1]
        rem = diff
        for idx in order:
            if rem == 0: break
            if rem > 0:
                add = min(rem, 1.0 - h[idx])
                if add > 0:
                    h[idx] += add
                    rem -= add
            else:
                sub = min(-rem, h[idx])
                if sub > 0:
                    h[idx] -= sub
                    rem += sub
    h = np.clip(h,0,1)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        # spread diff among all with capacity using small?

            cap = 1.0 - h
            if cap.sum() > 0:

import time
import numpy as np
from scipy.optimize import minimize

SL_N = 0
SL_ACTIVE = None

SYM_N = 0
SYM_ACTIVE = None
SYM_R = None


def time_remaining(t_start, budget_s):
    return float(budget_s - (time.perf_counter() - t_start))


def repair_feasible(h):
    h = np.asarray(h, dtype=np.float64).reshape(-1)
    h = np.where(np.isfinite(h), h, 0.0)
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    target = 0.5 * n

    for _ in range(8):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0.0:
            order = np.argsort(1.0 - h)
            rem = diff
            for idx in order:
                if rem <= 1e-15:
                    break
                add = min(rem, 1.0 - h[idx])
                if add > 0.0:
                    h[idx] += add
                    rem -= add
            diff = target - h.sum()
        else:
            order = np.argsort(h)[::-1]
            rem = -diff
            for idx in order:
                if rem <= 1e-15:
                    break
                sub = min(rem, h[idx])
                if sub > 0.0:
                    h[idx] -= sub
                    rem -= sub
            diff = target - h.sum()

    h = np.clip(h, 0.0, 1.0)

    if abs(target - h.sum()) > 1e-14:
        diff = target - h.sum()
        if diff > 0.0:
            cap = 1.0 - h
            if cap.sum() > 0.0:
                h += (diff / cap.sum()) * cap
        else:
            cap = h
            if cap.sum() > 0.0:
                h += (diff / cap.sum()) * cap
        h = np.clip(h, 0.0, 1.0)

        diff = target - h.sum()
        if abs(diff) > 1e-14:
            if diff > 0.0:
                idx = int(np.argmax(1.0 - h))
            else:
                idx = int(np.argmax(h))
            h[idx] = float(np.clip(h[idx] + diff, 0.0, 1.0))

    return h


def compute_profile(h):
    h = np.asarray(h, dtype=np.float64).reshape(-1)
    n = len(h)
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)


def exact_score(h):
    return float(np.max(compute_profile(h)))


def build_gradient(h):
    h = np.asarray(h, dtype=np.float64).reshape(-1)
    n = len(h)
    dx = 2.0 / n
    m = 2 * n - 1
    c = n - 1
    G = np.zeros((m, n), dtype=np.float64)

    G[c, :] = dx * (1.0 - 2.0 * h)

    for d in range(1, n):
        neg = c - d
        pos = c + d

        G[neg, 0:n - d] += dx
        G[neg, 0:n - d] -= dx * h[d:n]
        G[neg, d:n] -= dx * h[0:n - d]

        G[pos, d:n] += dx
        G[pos, d:n] -= dx * h[0:n - d]
        G[pos, 0:n - d] -= dx * h[d:n]

    return G


def active_indices(P, K, beam=None):
    P = np.asarray(P, dtype=np.float64)
    T = float(np.max(P))
    K = max(0, int(K))

    if beam is None:
        spread = T - float(np.median(P))
        beam = max(0.0015, min(0.025, 0.30 * max(spread, 1e-8)))

    active = np.where(P >= T - beam)[0]

    if K > 0:
        top = np.argsort(P)[-K:]
        active = np.union1d(active, top)

    if active.size == 0:
        active = np.array([int(np.argmax(P))])

    limit = max(K + 12, 24)
    if active.size > limit:
        order = np.argsort(P[active])[-limit:]
        active = active[order]

    return np.sort(active)


def interpolate_to_n(h, new_n):
    h = np.asarray(h, dtype=np.float64).reshape(-1)
    n = len(h)
    if n == new_n:
        return repair_feasible(h.copy())

    xs = (np.arange(n) + 0.5) * (2.0 / n)
    xt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    xx = np.concatenate(([0.0], xs, [2.0]))
    hx = np.concatenate(([h[0]], h, [h[-1]]))
    out = np.interp(xt, xx, hx)
    return repair_feasible(np.clip(out, 0.0, 1.0))


def project_symmetric(h):
    h = np.asarray(h, dtype=np.float64).reshape(-1)
    hs = 0.5 * (h + h[::-1])
    hs = np.clip(hs, 0.0, 1.0)
    hs = repair_feasible(hs)
    hs = 0.5 * (hs + hs[::-1])
    return np.clip(hs, 0.0, 1.0)


def make_full_z(z, n):
    z = np.asarray(z, dtype=np.float64).reshape(-1)
    m = (n + 1) // 2
    if z.size < m:
        z = np.concatenate((z, np.zeros(m - z.size)))
    else:
        z = z[:m]

    h = np.empty(n, dtype=np.float64)
    if n % 2 == 0:
        h[:m] = z
        h[m:] = z[::-1]
    else:
        h[:m] = z
        h[m:] = z[:m - 1][::-1]
    return h


def set_sym_matrix(n):
    global SYM_R
    m = (n + 1) // 2
    R = np.zeros((n, m), dtype=np.float64)
    for r in range(m):
        R[r, r] = 1.0
        mir = n - 1 - r
        if mir != r:
            R[mir, r] = 1.0
    SYM_R = R
    return R


def slsqp_obj(x):
    return float(x[-1])


def slsqp_jac(x):
    x = np.asarray(x, dtype=np.float64)
    g = np.zeros(x.size, dtype=np.float64)
    g[-1] = 1.0
    return g


def slsqp_eq(x):
    n = SL_N
    return float(x[:n].sum() - 0.5 * n)


def slsqp_eq_jac(x):
    x = np.asarray(x, dtype=np.float64)
    g = np.zeros(x.size, dtype=np.float64)
    g[:SL_N] = 1.0
    return g


def slsqp_ineq(x):
    n = SL_N
    h = x[:n]
    return x[-1] - compute_profile(h)[SL_ACTIVE]


def slsqp_ineq_jac(x):
    n = SL_N
    h = x[:n]
    G = build_gradient(h)
    J = np.zeros((len(SL_ACTIVE), x.size), dtype=np.float64)
    J[:, :n] = -G[SL_ACTIVE, :]
    J[:, -1] = 1.0
    return J


def slsqp_optimize_asym(h_start, active_idx, maxiter=70, ftol=1e-12):

    n = len(h_start)
    P = compute_profile(h_start)
    T = float(np.max(P))

    if active_idx is None or len(active_idx) == 0:
        active = active_indices(P, 20, None)
    else:
        active = np.asarray(active_idx, dtype=int)
        top = np.argsort(P)[-len(active):]
        active = np.union1d(active, top)

        active = active[np.argsort(P[active])[-110:]]

    SL_N = n
    SL_ACTIVE = active

    x0[:n] = h_start
    x0[n] = T + 1e-6

    bounds = []

    bounds.append((0.0, None))

        {"type": "eq", "fun": slsqp_eq, "jac": slsqp_eq_jac},
        {"type": "ineq", "fun": slsqp_ineq, "jac": slsqp_ineq_jac},
    ]

    try:

        cnew = exact_score(hnew)
        return hnew, cnew

        hnew = make_full_z(res.x[:m], n)
        hnew = repair_feasible(hnew)
        cnew = exact_score(hnew)
        return hnew, cnew

    return h_start, T


def eval_surrogate(P, b, c, a):
    return float(np.max(P + b * a + c * a * a))


def golden_min_surrogate(P, b, c, lo, hi, max_iter=40):
    if hi - lo < 1e-13:
        a = 0.5 * (lo + hi)
        return float(a), eval_surrogate(P, b, c, a)

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a1 = hi - gr * (hi - lo)
    a2 = lo + gr * (hi - lo)
    f1 = eval_surrogate(P, b, c, a1)
    f2 = eval_surrogate(P, b, c, a2)

    for _ in range(int(max_iter)):
        if f1 < f2:
            hi = a2
            a2 = a1
            f2 = f1
            a1 = hi - gr * (hi - lo)
            f1 = eval_surrogate(P, b, c, a1)
        else:
            lo = a1
            a1 = a2
            f1 = f2
            a2 = lo + gr * (hi - lo)
            f2 = eval_surrogate(P, b, c, a2)

        if hi - lo < 1e-10:
            break

    a = 0.5 * (lo + hi)
    return float(a), eval_surrogate(P, b, c, a)


def coarse_surrogate(P, b, c, quad_pos, lo, hi):
    if hi - lo < 1e-13:
        a = 0.5 * (lo + hi)
        return float(a), eval_surrogate(P, b, c, a)

    w = hi - lo
    mid = 0.5 * (lo + hi)
    ngrid = 17 if w < 0.02 else 9
    cand = mid + 0.5 * w * np.linspace(-1.0, 1.0, ngrid)
    cand = np.clip(cand, lo, hi)

    vals = P[None, :] + b[None, :] * cand[:, None] + c[None, :] * cand[:, None] ** 2
    fvals = np.max(vals, axis=1)
    idx = int(np.argmin(fvals))
    best_a = float(cand[idx])
    best_f = float(fvals[idx])

    if quad_pos is not None and quad_pos.size > 0:
        for pos in quad_pos:
            cq = c[pos]
            if cq > 1e-12:
                aq = -b[pos] / (2.0 * cq)
                if lo + 1e-12 < aq < hi - 1e-12:
                    f = eval_surrogate(P, b, c, aq)
                    if f < best_f - 1e-15:
                        best_a = float(aq)
                        best_f = float(f)

    return best_a, best_f


def refine_surrogate(P, b, c, lo, hi, a0, f0, convex):
    if convex:
        a1, f1 = golden_min_surrogate(P, b, c, lo, hi, 45)
        if f1 < f0:
            return float(a1), float(f1)
        return float(a0), float(f0)

    w = hi - lo
    ngrid = 33
    cand = np.linspace(lo, hi, ngrid)
    vals = P[None, :] + b[None, :] * cand[:, None] + c[None, :] * cand[:, None] ** 2
    fvals = np.max(vals, axis=1)
    idx = int(np.argmin(fvals))
    if fvals[idx] < f0:
        a0 = float(cand[idx])
        f0 = float(fvals[idx])

    if lo < a0 < hi:
        ww = 0.25 * w
        l = max(lo, a0 - ww)
        r = min(hi, a0 + ww)
        if r - l > 1e-12:
            a1, f1 = golden_min_surrogate(P, b, c, l, r, 30)
            if f1 < f0:
                return float(a1), float(f1)

    return float(a0), float(f0)


def steepest_pair_step(h, max_step, active_k=24, beam=0.01, refine=True, improve_tol=1e-9):
    h = np.asarray(h, dtype=np.float64).reshape(-1)
    n = len(h)
    dx = 2.0 / n
    m = 2 * n - 1
    center = n - 1

    P = compute_profile(h)
    T = float(np.max(P))
    G = build_gradient(h)

    active = active_indices(P, active_k, beam)
    K = active.size

    P_act = P[active].astype(np.float64)
    G_act = G[active].astype(np.float64)

    c_templates = np.zeros((n, K), dtype=np.float64)
    quad_pos = [np.array([], dtype=int) for _ in range(n)]

    pos_center = np.flatnonzero(active == center)
    if pos_center.size > 0:
        c_templates[1:, pos_center] = -2.0 * dx

    for d in range(1, n):
        p1 = np.flatnonzero(active == center + d)
        p2 = np.flatnonzero(active == center - d)
        pos = np.unique(np.concatenate((p1, p2)))
        if pos.size > 0:
            c_templates[d, pos] += dx
            quad_pos[d] = pos

    best_full = T + improve_tol
    best_move = None

    for i in range(n - 1):
        for j in range(i + 1, n):
            d = abs(i - j)

            lo = -h[i]
            if lo > -max_step:
                lo = -max_step
            hi = 1.0 - h[j]
            if hi > max_step:
                hi = max_step

            if hi - lo < 1e-13:
                continue

            b = G_act[:, i] - G_act[:, j]
            c = c_templates[d]
            quad = quad_pos[d]

            a0, f0 = coarse_surrogate(P_act, b, c, quad, lo, hi)
            if f0 > T - 1e-7:
                continue

            if refine:
                convex = bool(np.all(c >= -1e-12))
                a1, f1 = refine_surrogate(P_act, b, c, lo, hi, a0, f0, convex)
                if f1 < f0:
                    a0, f0 = a1, f1

            if abs(a0) < 1e-9:
                continue
            if a0 < lo + 1e-12 or a0 > hi - 1e-12:
                continue

            bb = G[:, i] - G[:, j]
            aa = a0 * a0
            vals = P + bb * a0
            if aa > 1e-18:
                vals[center] -= 2.0 * dx * aa
                if d > 0:
                    vals[center + d] += dx * aa
                    vals[center - d] += dx * aa
            fs = float(np.max(vals))

            if fs < best_full - improve_tol:
                best_full = fs
                best_move = (i, j, a0)

    return best_move, best_full, T


def pair_descent(h_start, t_start, budget_s, max_secs=120, max_step=0.02,
                 active_k=24, beam=0.01):
    h = repair_feasible(np.asarray(h_start, dtype=np.float64).reshape(-1).copy())
    best_h = h.copy()
    best_c = exact_score(h)
    local_start = time.perf_counter()

    base_ak = int(active_k)
    specs = [
        (base_ak, float(beam)),
        (min(max(2 * base_ak, 40), 70), 0.02),
        (min(max(3 * base_ak, 60), 90), 0.04),
    ]

    for ak, bm in specs:
        max_step_cur = float(max_step)
        no_improve = 0

        while time_remaining(t_start, budget_s) > 30 and (time.perf_counter() - local_start) < max_secs:
            move, full_best, T = steepest_pair_step(
                h, max_step_cur, active_k=ak, beam=bm, refine=True, improve_tol=1e-10
            )

            if move is None:
                no_improve += 1
                if no_improve >= 2:
                    break
                max_step_cur = max(0.001, 0.7 * max_step_cur)
                continue

            old_h = h.copy()
            i, j, a = move
            h[i] += a
            h[j] -= a
            c = exact_score(h)

            if c < best_c - 1e-12:
                best_h = h.copy()
                best_c = c
                no_improve = 0
                if abs(a) < 0.004:
                    max_step_cur = max(0.001, 0.8 * max_step_cur)
            else:
                h = old_h
                no_improve += 1
                max_step_cur = max(0.001, 0.85 * max_step_cur)
                if no_improve >= 3:
                    break

    return best_h, best_c


def consider_candidate(cand, best_h, best_c):
    cand = np.asarray(cand, dtype=np.float64).reshape(-1)
    if cand.shape != best_h.shape:
        return best_h, best_c
    if not np.all(np.isfinite(cand)):
        return best_h, best_c

    cand = repair_feasible(cand)
    c = exact_score(cand)
    if c < best_c - 1e-11:
        return cand, c

    hs = project_symmetric(cand)
    cs = exact_score(hs)
    if cs < best_c - 1e-11:
        return hs, cs

    for a in (0.25, 0.5, 0.75):
        hb = repair_feasible((1.0 - a) * best_h + a * cand)
        cb = exact_score(hb)
        if cb < best_c - 1e-11:
            return hb, cb

        hb = repair_feasible((1.0 - a) * best_h + a * hs)
        cb = exact_score(hb)
        if cb < best_c - 1e-11:
            return hb, cb

    return best_h, best_c


def make_structured_start(n, rng, mode):
    h = np.zeros(n, dtype=np.float64)
    x = np.arange(n, dtype=np.float64)

    if mode == 0:
        freq = float(rng.randint(1, 4))
        phase = float(rng.uniform(0.0, 2.0 * np.pi))
        amp = float(rng.uniform(0.15, 0.45))
        h[:] = 0.5 + amp * np.sin(freq * np.pi * x / n + phase)
    elif mode == 1:
        nb = int(rng.randint(3, 8))
        block = max(1, n // nb)
        for b in range(nb):
            a = b * block
            c = min(n, a + block)
            if a < c:
                h[a:c] = float(rng.uniform(0.1, 0.9))
    elif mode == 2:
        phi = 0.6180339887498949
        phase = float(rng.uniform(0.0, 2.0 * np.pi))
        amp = float(rng.uniform(0.25, 0.45))
        h[:] = 0.5 + amp * np.sin(2.0 * np.pi * ((x * phi) % 1.0) + phase)
    else:
        half = n // 2
        a = float(rng.uniform(0.2, 0.8))
        h[:half] = rng.uniform(0.1, 0.9, half)
        h[half:half + 1] = a
        h[half + 1:] = 1.0 - h[:half][::-1]

    return repair_feasible(np.clip(h, 0.0, 1.0))


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()
    rng = np.random.RandomState(seed)

    h = None
    try:
        h = np.asarray(initial_h_values, dtype=np.float64).reshape(-1)
    except Exception:
        h = None

    if h is None or h.size < 10 or not np.all(np.isfinite(h)):
        h = np.full(70, 0.5, dtype=np.float64)

    h = repair_feasible(h)
    best_h = h.copy()
    best_c = exact_score(h)

    if budget_s < 60:
        best_h = repair_feasible(best_h)
        best_c = exact_score(best_h)
        return best_h, best_c, len(best_h)

    hs = project_symmetric(h)
    cs = exact_score(hs)
    if cs < best_c - 1e-12:
        best_h = hs
        best_c = cs

    if time_remaining(t_start, budget_s) > 180:
        h2, c2 = pair_descent(best_h, t_start, budget_s,
                              max_secs=55, max_step=0.02, active_k=24, beam=0.01)
        if c2 < best_c - 1e-12:
            best_h, best_c = h2, c2

    if time_remaining(t_start, budget_s) > 180:
        for K, beam, mi in [(12, 0.005, 70), (24, 0.01, 70), (42, 0.02, 50)]:
            if time_remaining(t_start, budget_s) < 130:
                break
            P = compute_profile(best_h)
            active = active_indices(P, K, beam)
            h2, c2 = slsqp_optimize_asym(best_h, active, mi, 1e-12)
            best_h, best_c = consider_candidate(h2, best_h, best_c)

            if time_remaining(t_start, budget_s) > 90:
                h2, c2 = pair_descent(best_h, t_start, budget_s,
                                      max_secs=30, max_step=0.012, active_k=28, beam=0.012)
                if c2 < best_c - 1e-12:
                    best_h, best_c = h2, c2

    if time_remaining(t_start, budget_s) > 150:
        for K, beam, mi in [(12, 0.005, 70), (28, 0.01, 55)]:
            if time_remaining(t_start, budget_s) < 110:
                break
            hs = project_symmetric(best_h)
            cs = exact_score(hs)
            if cs > best_c + 0.02:
                continue
            P = compute_profile(hs)
            active = active_indices(P, K, beam)
            h2, c2 = slsqp_optimize_sym(hs, active, mi, 1e-12)
            best_h, best_c = consider_candidate(h2, best_h, best_c)

            if time_remaining(t_start, budget_s) > 70:
                h2, c2 = pair_descent(best_h, t_start, budget_s,
                                      max_secs=25, max_step=0.01, active_k=30, beam=0.015)
                if c2 < best_c - 1e-12:
                    best_h, best_c = h2, c2

    if time_remaining(t_start, budget_s) > 260:
        h2, c2 = pair_descent(best_h, t_start, budget_s,
                              max_secs=160, max_step=0.018, active_k=34, beam=0.018)
        if c2 < best_c - 1e-12:
            best_h, best_c = h2, c2

    if time_remaining(t_start, budget_s) > 120:
        for K, beam, mi in [(24, 0.01, 60), (44, 0.025, 45)]:
            if time_remaining(t_start, budget_s) < 90:
                break
            P = compute_profile(best_h)
            active = active_indices(P, K, beam)
            h2, c2 = slsqp_optimize_asym(best_h, active, mi, 1e-12)
            best_h, best_c = consider_candidate(h2, best_h, best_c)

    base_n = len(best_h)
    dim_items = []
    for d in [base_n - 4, base_n - 2, base_n + 2, base_n + 4, base_n + 6, base_n + 8, base_n * 2]:
        if 40 <= d <= 160 and d % 2 == 0 and d != base_n:
            dim_items.append((abs(d - base_n), d))
    dim_items = list(set(dim_items))
    dim_items.sort()

    for _, new_n in dim_items[:4]:
        if time_remaining(t_start, budget_s) < 200:
            break

        h_cand = interpolate_to_n(best_h, new_n)
        c_cand = exact_score(h_cand)

        h_sym = project_symmetric(h_cand)
        c_sym = exact_score(h_sym)
        if c_sym < c_cand:
            h_cand, c_cand = h_sym, c_sym

        if c_cand < best_c + 0.02:
            for K, beam, mi in [(22, 0.01, 55), (38, 0.02, 40)]:
                if time_remaining(t_start, budget_s) < 130:
                    break
                P = compute_profile(h_cand)
                active = active_indices(P, K, beam)
                h2, c2 = slsqp_optimize_asym(h_cand, active, mi, 1e-12)
                h_cand, c_cand = consider_candidate(h2, h_cand, c_cand)

            if time_remaining(t_start, budget_s) > 90:
                h2, c2 = pair_descent(h_cand, t_start, budget_s,
                                      max_secs=60, max_step=0.012, active_k=32, beam=0.02)
                if c2 < c_cand - 1e-12:
                    h_cand, c_cand = h2, c2

            if len(h_cand) == len(best_h):
                best_h, best_c = consider_candidate(h_cand, best_h, best_c)
            elif c_cand < best_c - 1e-12:
                best_h, best_c = h_cand.copy(), c_cand

    if time_remaining(t_start, budget_s) > 160:
        n_b = len(best_h)
        for mode in range(4):
            if time_remaining(t_start, budget_s) < 70:
                break
            h_st = make_structured_start(n_b, rng, mode)
            h_blend = repair_feasible(0.78 * best_h + 0.22 * h_st)
            c_blend = exact_score(h_blend)

            if c_blend < best_c + 0.015:
                P = compute_profile(h_blend)
                active = active_indices(P, 26, 0.012)
                h2, c2 = slsqp_optimize_asym(h_blend, active, 60, 1e-12)
                h3, c3 = consider_candidate(h2, h_blend, c_blend)

                if time_remaining(t_start, budget_s) > 55:
                    h4, c4 = pair_descent(h3, t_start, budget_s,
                                          max_secs=35, max_step=0.009, active_k=32, beam=0.015)
                    h3, c3 = consider_candidate(h4, h3, c3)

                best_h, best_c = consider_candidate(h3, best_h, best_c)

    if time_remaining(t_start, budget_s) > 60:
        h2, c2 = pair_descent(best_h, t_start, budget_s,
                              max_secs=40, max_step=0.004, active_k=48, beam=0.006)
        if c2 < best_c - 1e-12:
            best_h, best_c = h2, c2

    if time_remaining(t_start, budget_s) > 50:
        P = compute_profile(best_h)
        active = active_indices(P, 32, 0.012)
        h2, c2 = slsqp_optimize_asym(best_h, active, 45, 1e-12)
        best_h, best_c = consider_candidate(h2, best_h, best_c)

    if time_remaining(t_start, budget_s) > 25:
        h2, c2 = pair_descent(best_h, t_start, budget_s,
                              max_secs=18, max_step=0.002, active_k=56, beam=0.005)
        if c2 < best_c - 1e-12:
            best_h, best_c = h2, c2

    best_h = repair_feasible(best_h)
    final_c = exact_score(best_h)

    try:
        val = evaluate_erdos_solution(best_h)
        if isinstance(val, (int, float, np.integer, np.floating)):
            val = float(val)
        elif isinstance(val, (tuple, list)) and len(val) > 0:
            val = float(val[0])
        if np.isfinite(val):
            final_c = max(final_c, val)
    except Exception:
        pass

    return best_h, float(final_c), len(best_h)
