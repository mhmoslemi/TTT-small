    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(profile))

def project_box_sum(h, target=None):
    """Project h to [0,1]^n with sum = n/2 (or target if given)."""
    n = len(h)
    h = np.array(h, dtype=np.float64)
    h = np.nan_to_num(h, nan=0.0, posinf=1.0, neginf=0.0)
    if target is None:
        target = n / 2.0
    # Water-filling projection
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = np.sum(h)
        diff = s - target
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            # Need to reduce sum
            mask = h > 0
            if not np.any(mask):
                break
            step = min(diff / np.sum(mask), np.min(h[mask]))
            h[mask] -= step
        else:
            # Need to increase sum
            mask = h < 1.0
            if not np.any(mask):
                break
            step = min(-diff / np.sum(mask), 1.0 - np.max(h[mask]))
            h[mask] += step
    # Final adjustment
    diff = np.sum(h) - target
    if abs(diff) > 1e-15:
        # Distribute remaining difference
        if diff > 0:
            mask = h > 0
        else:
            mask = h < 1.0
        if np.any(mask):
            h[mask] -= diff / np.sum(mask)
            h = np.clip(h, 0.0, 1.0)
    return h

    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full')

    Objective: dx * (max(r) + tau * log(sum(exp((r-max(r))/tau))))

import numpy as np
from scipy.optimize import minimize
import time

def exact_objective(h):
    n = len(h)
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(profile))

def project_box_sum(h, target=None):
    n = len(h)
    h = np.array(h, dtype=np.float64)
    h = np.nan_to_num(h, nan=0.0, posinf=1.0, neginf=0.0)
    if target is None:
        target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = np.sum(h)
        diff = s - target
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h > 0
            cnt = np.sum(mask)
            if cnt == 0:
                break
            step = min(diff / cnt, np.min(h[mask]))
            h[mask] -= step
        else:
            mask = h < 1.0
            cnt = np.sum(mask)
            if cnt == 0:
                break
            step = min(-diff / cnt, 1.0 - np.max(h[mask]))
            h[mask] += step
    diff = np.sum(h) - target
    if abs(diff) > 1e-14:
        if diff > 0:
            mask = h > 0
        else:
            mask = h < 1.0
        if np.any(mask):
            h[mask] -= diff / np.sum(mask)
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_profile_raw(h):
    return np.correlate(h, 1.0 - h, mode='full')

def smooth_obj_grad(x, tau):
    n = len(x)
    dx = 2.0 / n
    r = compute_profile_raw(x)
    r_max = np.max(r)
    shifted = np.clip((r - r_max) / tau, -500, 500)
    exp_s = np.exp(shifted)
    log_sum = np.log(np.sum(exp_s))
    value = dx * (r_max + tau * log_sum)
    weights = exp_s / np.sum(exp_s)
    grad = np.zeros(n)
    for i in range(2 * n - 1):
        k = i - (n - 1)
        w = weights[i]
        if w < 1e-14:
            continue
        if k == 0:
            grad += w * (1.0 - 2.0 * x)
        elif k > 0:
            grad[k:n] += w * (1.0 - x[0:n-k])
            grad[0:n-k] -= w * x[k:n]
        else:
            a = -k
            grad[0:n-a] += w * (1.0 - x[a:n])
            grad[a:n] -= w * x[0:n-a]
    grad *= dx
    return value, grad

def slsqp_optimize(h0, tau, maxiter=200):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0, 'jac': lambda x: np.ones(n)}]
    
    def fun(x):
        v, g = smooth_obj_grad(x, tau)
        # Add penalty for constraint violation (extra safety)
        sv = np.sum(x) - n/2.0
        v_pen = v + 1e4 * sv * sv
        g_pen = g + 2e4 * sv * np.ones(n)
        return v_pen, g_pen
    
    try:
        res = minimize(fun, h0, method='SLSQP', jac=True, bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-12, 'disp': False})
        if res.success or res.fun < 1e10:
            h_new = project_box_sum(res.x)
            return h_new
    except Exception:
        pass
    return None

def binary_threshold(h):
    """Threshold to binary with exactly n/2 ones (for even n)."""
    n = len(h)
    if n % 2 != 0:
        return None
    target = n // 2
    idx = np.argsort(-h)
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b

def binary_swap_improve(b, max_rounds=500):
    """Greedy binary swap local search."""
    n = len(b)
    cur = b.copy()
    cur_c = exact_objective(cur)
    ones = np.where(cur == 1.0)[0]
    zeros = np.where(cur == 0.0)[0]
    
    for round_idx in range(max_rounds):
        best_delta = 0
        best_i, best_j = -1, -1
        for i in ones:
            for j in zeros:
                cur[i] = 0.0
                cur[j] = 1.0
                c = exact_objective(cur)
                cur[i] = 1.0
                cur[j] = 0.0
                delta = cur_c - c
                if delta > best_delta:
                    best_delta = delta
                    best_i, best_j = i, j
        if best_i >= 0:
            cur[best_i] = 0.0
            cur[best_j] = 1.0
            cur_c -= best_delta
            ones = np.where(cur == 1.0)[0]
            zeros = np.where(cur == 0.0)[0]
        else:
            break
    return cur, cur_c

def continuous_pair_swap(h, n_trials=2000, scale=0.1):
    """Random pair swap moves preserving sum."""
    n = len(h)
    cur = h.copy()
    cur_c = exact_objective(cur)
    rng = np.random.RandomState(12345)
    
    for _ in range(n_trials):
        i, j = rng.choice(n, 2, replace=False)
        # Transfer amount from j to i (or vice versa)
        amt = scale * (0.5 + rng.random())
        if amt > cur[j]:
            amt = cur[j] * 0.9
        if amt > 1.0 - cur[i]:
            amt = (1.0 - cur[i]) * 0.9
        if amt < 0.01:
            continue
        cur[i] += amt
        cur[j] -= amt
        c = exact_objective(cur)
        if c < cur_c - 1e-12:
            cur_c = c
        else:
            cur[i] -= amt
            cur[j] += amt
    return cur, cur_c

def transfer_to_size(parent, target_n):
    """Transfer parent vector to a different size via interpolation."""
    n = len(parent)
    if n == target_n:
        return parent.copy()
    # Linear interpolation
    x_old = np.linspace(0, 1, n, endpoint=False) + 0.5/n
    x_new = np.linspace(0, 1, target_n, endpoint=False) + 0.5/target_n
    new_h = np.interp(x_new, x_old, parent)
    return project_box_sum(new_h)

import numpy as np

def exact_objective(h):
    n = len(h)
    if n == 0:
        return float('inf')
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(profile))

def project_box_sum(h, target=None):
    n = len(h)
    h = np.array(h, dtype=np.float64)
    h = np.nan_to_num(h, nan=0.0, posinf=1.0, neginf=0.0)
    if target is None:
        target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        s = np.sum(h)
        diff = s - target
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h > 0
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            step = min(diff / cnt, np.min(h[mask]))
            h[mask] -= step
        else:
            mask = h < 1.0
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            step = min(-diff / cnt, 1.0 - np.max(h[mask]))
            h[mask] += step
    diff = np.sum(h) - target
    if abs(diff) > 1e-14:
        mask = (h > 0) if diff > 0 else (h < 1.0)
        if np.any(mask):
            h[mask] -= diff / np.sum(mask)
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_profile_raw(h):
    return np.correlate(h, 1.0 - h, mode='full')

def smooth_obj_grad(x, tau):
    n = len(x)
    dx = 2.0 / n
    r = compute_profile_raw(x)
    r_max = np.max(r)
    shifted = np.clip((r - r_max) / tau, -500.0, 500.0)
    exp_s = np.exp(shifted)
    log_sum = np.log(np.sum(exp_s))
    value = dx * (r_max + tau * log_sum)
    weights = exp_s / np.sum(exp_s)
    grad = np.zeros(n)
    for i in range(2 * n - 1):
        k = i - (n - 1)
        w = weights[i]
        if w < 1e-14:
            continue
        if k == 0:
            grad += w * (1.0 - 2.0 * x)
        elif k > 0:
            grad[k:n] += w * (1.0 - x[0:n-k])
            grad[0:n-k] -= w * x[k:n]
        else:
            a = -k
            grad[0:n-a] += w * (1.0 - x[a:n])
            grad[a:n] -= w * x[0:n-a]
    grad *= dx
    return value, grad

def slsqp_optimize(h0, tau, maxiter=200):
    global _N, _tau
    _N = len(h0)
    _tau = tau
    n = _N

            if time.time() < deadline - 60:
                h_sw, c_sw = continuous_pair_swap(h_cur, n_trials=2000, scale=0.12, rng_seed=100+n_iter_global)
                if c_sw < c_cur - 1e-12:
                    h_cur = h_sw
                    c_cur = c_sw

            if n % 2 == 0:
                b = binary_threshold(h_cur)
                if b is not None:
                    b_c = exact_objective(b)
                    if b_c < c_cur:
                        h_cur = b
                        c_cur = b_c
                    b2, b2_c = binary_swap_improve(b, max_rounds=400, rng_seed=200+n_iter_global)
                    if b2_c < c_cur:
                        h_cur = b2.copy()
                        c_cur = b2_c

            if c_cur < best_c - 1e-12:
                best_c = c_cur
                best_h = h_cur.copy()
                best_n = n

    if time.time() < deadline - 30:
        h_final = project_box_sum(best_h, best_n / 2.0)
        c_final = exact_objective(h_final)
        if c_final < best_c - 1e-12:
            best_c = c_final
            best_h = h_final

    best_h = project_box_sum(best_h, best_n / 2.0)
    best_c = exact_objective(best_h)
    return (best_h, best_c, len(best_h))
