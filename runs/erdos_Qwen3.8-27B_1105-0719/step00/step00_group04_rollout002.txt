def softmax_obj_and_grad(h, tau):

    w = np.exp(log_w)

        m_end = min(n-1, n-1-L)

            j_start = m_start + L
            grad[m_start:m_end+1] += wk * (1 - h[j_start:j_start + (m_end-m_start+1)])

        m_end2 = min(n-1, L + n - 1)

            j_start2 = m_start2 - L
            grad[m_start2:m_end2+1] -= wk * h[j_start2:j_start2 + (m_end2-m_start2+1)]
    
    return obj, grad
