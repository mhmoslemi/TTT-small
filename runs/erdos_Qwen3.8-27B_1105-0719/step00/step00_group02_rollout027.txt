def pair_optimize(h, i, j, dx):
    """Optimize the pair (i,j) to minimize the maximum overlap."""
    n = len(h)
    S = h[i] + h[j]
    lo = max(0.0, S - 1.0)
    hi = min(1.0, S)
    if hi - lo < 1e-14:
        return h  # No room to move
    
    # For a given x (value of h[i]), h[j] = S - x
    def objective(x):
        h_copy = h.copy()
        h_copy[i] = x
        h_copy[j] = S - x
        corr = np.correlate(h_copy, 1 - h_copy, mode='full')
        return corr.max() * dx
    
    # Golden section search
    gr = (np.sqrt(5) - 1) / 2  # golden ratio conjugate
    a, b = lo, hi
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = objective(c)
    fd = objective(d)
    
    for _ in range(50):  # 50 iterations gives ~gr^-50 precision
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = objective(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = objective(d)
    
    x_opt = (a + b) / 2
    h_new = h.copy()
    h_new[i] = x_opt
    h_new[j] = S - x_opt
    return h_new
