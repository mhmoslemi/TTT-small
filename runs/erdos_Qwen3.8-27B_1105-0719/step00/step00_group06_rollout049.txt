2. With discretization: n points over [0,2], dx = 2/n, constraint sum(h)*dx = 1, so sum(h) = n/2
3. The evaluation is: C₅ = max(np.correlate(h, 1-h, mode="full") * dx)
4. Current record: C₅ ≤ 0.38085
5. We start from initial_h_values (n=70)

Key insights:
- The objective is max over all lags of the correlation between h and (1-h)
- We need 0 ≤ h[i] ≤ 1 and sum(h) = n/2
- Lower C₅ is better

Let me think about what makes a good h:
- For lag 0: ∫ h(x)(1-h(x)) dx = ∫ h(x) dx - ∫ h(x)² dx = 1 - ∫ h(x)² dx (since ∫h = 1)
  - To minimize this, we want to maximize ∫h² dx, meaning h should be more "binary" (concentrated to 0 or 1)
- For other lags: we want h(x) and h(x+k) to not overlap much where one is high and the other is low

    Gradient of Phi_lag(x) = dx * sum_i x[i+lag] * (1 - x[i]) for lag >= 0
    or dx * sum_i x[i] * (1 - x[i+|lag|]) for lag < 0

        #   if j appears as x[i] term: contributes -dx * x[i+lag] for i=j, needs j < n-lag

            grad[i] += 1.0 * (1.0 - x[i + a])
            # d/dx[i+a] = -dx * x[i]
            grad[i + a] += -1.0 * x[i]

from scipy.optimize import minimize
import time

def exact_objective(h):
    n = len(h)
    if n == 0:
        return float('inf'), np.array([])
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(profile)), profile

def project_box_sum(h, target=None):
    n = len(h)
    if target is None:
        target = 0.5 * n
    h = np.array(h, dtype=np.float64).copy()
    h = np.nan_to_num(h, nan=0.0, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
            count = np.sum(mask)
            if count == 0:
                break
            step = diff / count
            h[mask] = np.minimum(h[mask] + step, 1.0)
        else:
            mask = h > 0.0
            count = np.sum(mask)
            if count == 0:
                break
            step = diff / count
            h[mask] = np.maximum(h[mask] + step, 0.0)
    # Exact final fix
    diff = target - np.sum(h)
    if abs(diff) > 1e-16 and n > 0:
        h[0] += diff
        h[0] = np.clip(h[0], 0.0, 1.0)
    return h

def _gradient_lag(x, lag, dx):
    n = len(x)
    grad = np.zeros(n)
    if lag >= 0:
        s = n - lag
        if s > 0:
            grad[:s] -= x[lag:]  # -x[i+lag] for i in 0..s-1
            grad[lag:] += (1.0 - x[:s])  # (1-x[i]) for i in 0..s-1
    else:
        a = -lag
        s = n - a
        if s > 0:
            grad[:s] += (1.0 - x[a:])  # (1-x[i+a]) for i in 0..s-1
            grad[a:] -= x[:s]  # -x[i] for i in 0..s-1
    return grad * dx

def smooth_obj_and_grad(x, tau):
    n = len(x)
    dx = 2.0 / n
    profile = np.correlate(x, 1.0 - x, mode='full') * dx
    pmax = np.max(profile)
    
    if tau < 1e-12:
        m_idx = int(np.argmax(profile))
        lag = m_idx - (n - 1)
        grad = _gradient_lag(x, lag, dx)
        return pmax, grad
    
    shifted = (profile - pmax) / tau
    shifted = np.clip(shifted, -100, 100)
    weights = np.exp(shifted)
    wsum = np.sum(weights)
    log_sum = np.log(wsum)
    value = pmax + tau * log_sum
    wnorm = weights / wsum
    
    grad = np.zeros(n)
    # Only compute gradient for significant weights
    thresh = 1e-6
    active = wnorm > thresh
    for m in np.where(active)[0]:
        lag = m - (n - 1)
        grad += wnorm[m] * _gradient_lag(x, lag, dx)
    
    return value, grad

def binary_search_best(h, max_iters=2000, rng=None):
    """Try binary thresholding and swap moves"""
    n = len(h)
    if n % 2 != 0:
        return h, exact_objective(h)[0]
    
    # Threshold to binary
    target = n // 2
    idx = np.argsort(-h)
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    b = project_box_sum(b, target)
    c5, _ = exact_objective(b)
    
    if rng is None:
        rng = np.random.RandomState(0)
    
    improved = True
    iters = 0
    while improved and iters < max_iters:
        improved = False
        iters += 1
        # Try all single swaps (ones with zeros)
        ones = np.where(b > 0.5)[0]
        zeros = np.where(b < 0.5)[0]
        best_swap = None
        best_val = c5
        # Try random subset of swaps if too many
        if len(ones) * len(zeros) > 500:
            pairs = [(rng.randint(len(ones)), rng.randint(len(zeros))) for _ in range(500)]
        else:
            pairs = [(i, j) for i in range(len(ones)) for j in range(len(zeros))]
        
        for oi, zi in pairs:
            i, j = ones[oi], zeros[zi]
            if i == j:
                continue
            b[i], b[j] = 0.0, 1.0
            val, _ = exact_objective(b)
            if val < best_val - 1e-12:
                best_val = val
                best_swap = (i, j)
            b[i], b[j] = 1.0, 0.0
        
        if best_swap is not None:
            i, j = best_swap
            b[i], b[j] = 0.0, 1.0
            c5 = best_val
            improved = True
    
    return b, c5

def interpolate_and_score(ha, hb, n_alphas=11):
    """Try interpolations between two valid vectors of same length"""
    n = len(ha)
    if len(hb) != n:
        return ha, exact_objective(ha)[0]
    target = 0.5 * n
    best_h = project_box_sum(ha, target)
    best_c = exact_objective(best_h)[0]
    for alpha in np.linspace(0, 1, n_alphas):
        h = (1 - alpha) * ha + alpha * hb
        h = project_box_sum(h, target)
        c, _ = exact_objective(h)
        if c < best_c - 1e-12:
            best_c = c
            best_h = h
    return best_h, best_c

def make_binary_start(n, rng, mode=0):
    """Generate various binary starting vectors"""
    target = n // 2 if n % 2 == 0 else (n // 2 + (rng.rand() < 0.5))
    # Actually for odd n, target = n/2 which is not integer
    # We'll use fractional for odd n
    target = 0.5 * n
    
    if mode == 0:
        # Alternating
        h = np.zeros(n)
        h[::2] = 1.0
    elif mode == 1:
        # Block
        h = np.zeros(n)
        h[:target] = 1.0
    elif mode == 2:
        # Random binary
        idx = rng.choice(n, size=min(int(target), n), replace=False)
        h = np.zeros(n)
        h[idx] = 1.0
    elif mode == 3:
        # Sinusoidal
        x = np.linspace(0, 2*np.pi, n)
        h = 0.5 + 0.5 * np.cos(x * (1 + rng.randint(2, 5)))
    elif mode == 4:
        # Two blocks
        h = np.zeros(n)
        h[:n//4] = 1.0
        h[n//2:3*n//4] = 1.0
    else:
        h = np.full(n, 0.5)
    
    return project_box_sum(h, target)

def descent_from_start(h_start, n, tau_list, rng, time_limit, deadline):
    """Run SLSQP descent from a starting point with multiple tau stages"""
    best_h = project_box_sum(h_start, 0.5 * n)
    best_c, _ = exact_objective(best_h)
    
    bounds = [(0.0, 1.0)] * n
    constraint = {'type': 'eq', 'fun': lambda x: np.sum(x) - 0.5 * n,
                  'jac': lambda x: np.ones(n)}
    
    for tau in tau_list:
        if time.time() > deadline - 2:
            break
        
        def f_and_grad(x):
            val, grad = smooth_obj_and_grad(x, tau)
            # Add penalty for constraint violation
            sum_err = np.sum(x) - 0.5 * n
            val += 1e4 * sum_err ** 2
            grad = grad + 2e4 * sum_err * np.ones(n)
            # Box penalty
            below = np.maximum(0, -x)
            above = np.maximum(0, x - 1)
            val += 1e4 * (np.sum(below**2) + np.sum(above**2))
            grad = grad - 2e4 * below + 2e4 * above
            return val, grad
        
        try:
            res = minimize(
                f_and_grad, best_h,
                method='SLSQP',
                bounds=bounds,
                constraints=constraint,
                jac=True,
                options={'maxiter': min(300, n * 4 + 50), 'ftol': 1e-12}

            if res.success or hasattr(res, 'x'):
                candidate = project_box_sum(res.x, 0.5 * n)
                c, _ = exact_objective(candidate)
                if c < best_c - 1e-12:
                    best_c = c
                    best_h = candidate

            pass
    
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def exact_objective(h):
    n = len(h)
    if n == 0:
        return float('inf'), np.array([])
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(profile)), profile

def project_box_sum(h, target=None):
    n = len(h)
    if target is None:
        target = 0.5 * n
    h = np.array(h, dtype=np.float64).copy()
    h = np.nan_to_num(h, nan=0.0, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
            count = np.sum(mask)
            if count == 0:
                break
            step = diff / count
            h[mask] = np.minimum(h[mask] + step, 1.0)
        else:
            mask = h > 0.0
            count = np.sum(mask)
            if count == 0:
                break
            step = diff / count
            h[mask] = np.maximum(h[mask] + step, 0.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-16 and n > 0:
        h[0] = np.clip(h[0] + diff, 0.0, 1.0)
    return h

def _gradient_lag(x, lag, dx):
    n = len(x)
    grad = np.zeros(n)
    if lag >= 0:
        s = n - lag
        if s > 0:
            grad[:s] -= x[lag:]
            grad[lag:] += 1.0 - x[:s]
    else:
        a = -lag
        s = n - a
        if s > 0:
            grad[:s] += 1.0 - x[a:]
            grad[a:] -= x[:s]
    return grad * dx

def smooth_obj_and_grad(x, tau):
    n = len(x)
    dx = 2.0 / n
    profile = np.correlate(x, 1.0 - x, mode='full') * dx
    pmax = np.max(profile)
    if tau < 1e-12:
        m_idx = int(np.argmax(profile))
        lag = m_idx - (n - 1)
        grad = _gradient_lag(x, lag, dx)
        return pmax, grad
    shifted = (profile - pmax) / tau
    shifted = np.clip(shifted, -100, 100)
    weights = np.exp(shifted)
    wsum = np.sum(weights)
    value = pmax + tau * np.log(wsum)
    wnorm = weights / wsum
    grad = np.zeros(n)
    active = wnorm > 1e-7
    for m in np.where(active)[0]:
        lag = m - (n - 1)
        grad += wnorm[m] * _gradient_lag(x, lag, dx)
    return value, grad

def slsqp_refine(h_start, n, tau_list, deadline, max_iter_per_tau=200):
    best_h = project_box_sum(h_start, 0.5 * n)
    best_c, _ = exact_objective(best_h)
    bounds = [(0.0, 1.0)] * n
    target = 0.5 * n

    def eq_fun(x):
        return np.sum(x) - target

    def eq_jac(x):
        return np.ones(n)

    constraint = {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}

    for tau in tau_list:
        if time.time() > deadline - 3:
            break

        def obj_grad(x):
            val, grad = smooth_obj_and_grad(x, tau)
            se = np.sum(x) - target
            val += 1e5 * se * se
            grad = grad + 2e5 * se * np.ones(n)
            below = np.maximum(0.0, -x)
            above = np.maximum(0.0, x - 1.0)
            val += 1e5 * (np.sum(below * below) + np.sum(above * above))
            grad = grad - 2e5 * below + 2e5 * above
            return val, grad

        try:
            mi = max(50, min(max_iter_per_tau, n * 3 + 30))
            res = minimize(obj_grad, best_h, method='SLSQP',
                           bounds=bounds, constraints=constraint, jac=True,
                           options={'maxiter': mi, 'ftol': 1e-14})
            if hasattr(res, 'x') and res.x is not None and np.all(np.isfinite(res.x)):

                c, _ = exact_objective(cand)
                if c < best_c - 1e-13:
                    best_c = c
                    best_h = cand
        except Exception:
            pass
    return best_h, best_c

    if mode == 0:
        h = np.zeros(n)
        h[::2] = 1.0
    elif mode == 1:
        h = np.zeros(n)
        h[:int(target)] = 1.0
    elif mode == 2:
        h = np.zeros(n)
        cnt = int(round(target))
        idx = rng.choice(n, size=min(cnt, n), replace=False)
        h[idx] = 1.0
    elif mode == 3:
        x = np.arange(n, dtype=np.float64)
        freq = 1.0 + rng.randint(2, 6)
        phase = rng.uniform(0, 2 * np.pi)
        h = 0.5 + 0.5 * np.cos(2 * np.pi * freq * x / n + phase)
    elif mode == 4:
        h = np.zeros(n)
        q = n // 4
        h[:q] = 1.0
        h[2 * n // 3: 2 * n // 3 + q] = 1.0
    elif mode == 5:
        h = np.full(n, 0.5)
        h += rng.randn(n) * 0.15
    else:
        h = np.full(n, 0.5)
    return project_box_sum(h, target)

def downsample(h, new_n):
    n = len(h)
    if new_n == n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(new_n) + 0.5) / new_n
    vals = np.interp(x_new, x_old, h)
    return project_box_sum(vals, 0.5 * new_n)

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    deadline = start_time + budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c = float('inf')

    try:
        init = np.asarray(initial_h_values, dtype=np.float64)
        if init.size > 0 and np.all(np.isfinite(init)):
            n0 = len(init)
            init = project_box_sum(init, 0.5 * n0)
            c0, _ = exact_objective(init)
            best_h = init.copy()
            best_c = c0
    except (NameError, Exception):
        pass

    if best_h is None:
        n0 = 70
        best_h = make_start(n0, rng, mode=0)
        best_c, _ = exact_objective(best_h)

    n0 = len(best_h)
    tau_list = [0.005, 0.002, 0.0008, 0.0003, 0.0001]
    tau_tight = [0.0001, 0.00004, 0.00001]

    # Phase 1: SLSQP on parent
    if time.time() < deadline - 60:
        h_opt, c_opt = slsqp_refine(best_h, n0, tau_list, deadline)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 2: Binary refinement
    if n0 % 2 == 0 and time.time() < deadline - 40:
        b = binary_threshold(best_h)
        if b is not None:
            b, c_b = binary_swap_improve(b, max_rounds=60, rng=rng)
            if c_b < best_c:
                best_c = c_b
                best_h = b
            # Re-run SLSQP from binary
            if time.time() < deadline - 30:
                h_opt, c_opt = slsqp_refine(b, n0, tau_tight, deadline)
                if c_opt < best_c:
                    best_c = c_opt
                    best_h = h_opt

    # Phase 3: Try other sizes
    sizes = sorted(set([s for s in [32, 36, 40, 44, 48, 50, 52, 54, 56, 60, 62, 64, 66, 68, 70, 72, 74, 76, 80]
                        if 16 <= s <= 200]), key=lambda x: abs(x - n0))

    for n_try in sizes:
        if n_try == n0:
            continue
        if time.time() > deadline - 25:
            break
        n_starts = 4
        for mode in range(n_starts):
            if time.time() > deadline - 15:
                break
            h_start = make_start(n_try, rng, mode=mode)
            # Also try downsampled parent
            if best_h is not None and len(best_h) != n_try:
                h_ds = downsample(best_h, n_try)
                h_start = h_ds if exact_objective(h_ds)[0] < exact_objective(h_start)[0] else h_start

            h_opt, c_opt = slsqp_refine(h_start, n_try, tau_list[:3], deadline)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

            if n_try % 2 == 0 and c_opt < best_c + 0.02 and time.time() < deadline - 10:
                b = binary_threshold(h_opt)
                if b is not None:
                    b, c_b = binary_swap_improve(b, max_rounds=40, rng=rng)
                    if c_b < best_c:
                        best_c = c_b
                        best_h = b

    # Phase 4: Interpolation between best solutions of same size
    if time.time() < deadline - 20:
        n_best = len(best_h)
        for _ in range(4):
            if time.time() > deadline - 10:
                break
            h_pert = best_h + rng.randn(n_best) * 0.03
            h_pert = project_box_sum(h_pert, 0.5 * n_best)
            h_interp, c_interp = interpolate_score(best_h, h_pert)
            if c_interp < best_c:
                best_c = c_interp
                best_h = h_interp
                # SLSQP from interpolation
                if time.time() < deadline - 10:
                    h_opt, c_opt = slsqp_refine(h_interp, n_best, tau_tight, deadline)
                    if c_opt < best_c:
                        best_c = c_opt
                        best_h = h_opt

    # Phase 5: Final tight SLSQP + binary
    if time.time() < deadline - 10:
        h_opt, c_opt = slsqp_refine(best_h, len(best_h), tau_tight, deadline)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
        if len(best_h) % 2 == 0:
            b = binary_threshold(best_h)
            if b is not None:
                b, c_b = binary_swap_improve(b, max_rounds=30, rng=rng)
                if c_b < best_c:
                    best_c = c_b
                    best_h = b

    # Final validation
    best_h = project_box_sum(best_h, 0.5 * len(best_h))
    final_c, _ = exact_objective(best_h)
    best_c = final_c
    best_n = len(best_h)

    return (best_h, best_c, best_n)
