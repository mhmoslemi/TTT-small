import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    R = np.correlate(h, 1-h, mode='full')
    return float(R.max() * dx)

def lse_objective_and_grad(h, beta):
    n = len(h)
    R = np.correlate(h, 1-h, mode='full')
    m = R.max()
    exps = np.exp(beta * (R - m))
    lse_val = m + np.log(exps.sum()) / beta
    w = exps / exps.sum()
    
    p = np.arange(n)[:, None]
    k = np.arange(-(n-1), n)[None, :]
    
    idx1 = p - k
    mask1 = (idx1 >= 0) & (idx1 < n)
    idx1c = np.clip(idx1, 0, n-1)
    term1 = np.where(mask1, 1.0 - h[idx1c], 0.0)
    
    idx2 = p + k
    mask2 = (idx2 >= 0) & (idx2 < n)
    idx2c = np.clip(idx2, 0, n-1)
    term2 = np.where(mask2, -h[idx2c], 0.0)
    
    grad = (term1 + term2) @ w
    return lse_val, grad

def refine_continuous(h0, budget_s=60):
    """Refine a continuous solution using SLSQP with LSE objective."""
    n = len(h0)
    S = n / 2.0
    constraints = [{'type': 'eq', 'fun': lambda h: h.sum() - S,
                    'jac': lambda h: np.ones(n)}]
    bounds = [(0, 1)] * n
    
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    deadline = time.time() + budget_s
    
    for beta in [10, 20, 50, 100, 200]:
        if time.time() > deadline:
            break
        def obj(h):
            val, _ = lse_objective_and_grad(h, beta)
            return val
        def grad(h):
            _, g = lse_objective_and_grad(h, beta)
            return g
        
        try:
            result = minimize(obj, best_h, jac=grad, method='SLSQP',
                             bounds=bounds, constraints=constraints,
                             options={'maxiter': 200, 'ftol': 1e-12})
            if result.success or result.fun < obj(best_h):
                new_c5 = compute_c5(result.x)
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = result.x.copy()
        except Exception:
            pass
    
    return best_h, best_c5

def make_initial_binary(n, rng, method='golden'):
    """Create an initial binary sequence with n/2 ones."""
    S = n // 2
    h = np.zeros(n)
    if method == 'golden':
        phi = (1 + np.sqrt(5)) / 2
        positions = np.unique((np.arange(S) * n * (1.0/phi)) % n).astype(int)
        # Ensure exactly S positions
        while len(positions) < S:
            positions = np.append(positions, rng.integers(n))
            positions = np.unique(positions).astype(int)
        positions = positions[:S]
        h[positions] = 1
    elif method == 'random':
        positions = rng.permutation(n)[:S]
        h[positions] = 1
    elif method == 'alternating':
        h[::2] = 1
        if n % 2 == 1:
            h[-1] = 0
        # Adjust to get exactly S ones
        ones_count = h.sum()
        if ones_count > S:

            h[excess] = 0
        elif ones_count < S:

            h[h == 0][:deficit] = 1
