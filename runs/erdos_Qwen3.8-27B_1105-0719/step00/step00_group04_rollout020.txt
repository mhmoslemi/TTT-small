        t = x[self.n]
        return t + self.beta * np.sum(h * (1.0 - h))

    def gradient(self, x):
        h = x[:self.n]
        grad = np.zeros(self.n + 1)
        grad[:self.n] = self.beta * (1.0 - 2.0 * h)
        grad[self.n] = 1.0
        return grad

    def eq_fun(self, x):
        return np.sum(x[:self.n]) - self.n / 2.0

    def eq_jac(self, x):
        jac = np.zeros(self.n + 1)
        jac[:self.n] = 1.0
        return jac

    def ineq_fun(self, x):
        h = x[:self.n]
        t = x[self.n]
        return t - get_profile(h)

    def ineq_jac(self, x):
        h = x[:self.n]
        G = lag_gradient_matrix(h, self.offsets)
        jac = np.zeros((self.n_offsets, self.n + 1))
        jac[:, :self.n] = -G
        jac[:, self.n] = 1.0
        return jac

def solve_epigraph(h_start, offsets, beta=0.0, maxiter=200, ftol=1e-12):
    n = len(h_start)
    c_start = exact_score(h_start)
    problem = EpigraphProblem(n, offsets, beta)
    x0 = np.concatenate([h_start, [c_start + 1e-6]])

    eq_constraint = {
        'type': 'eq',
        'fun': problem.eq_fun,
        'jac': problem.eq_jac
    }
    ineq_constraint = {
        'type': 'ineq',
        'fun': problem.ineq_fun,
        'jac': problem.ineq_jac
    }
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]

    try:
        result = minimize(
            problem.objective,
            x0,
            jac=problem.gradient,
            method='SLSQP',
            bounds=bounds,
            constraints=[eq_constraint, ineq_constraint],
            options={'maxiter': maxiter, 'ftol': ftol}
        )
        h_sol = result.x[:n]
        h_sol = repair_feasible(h_sol, n / 2.0)
        c_sol = exact_score(h_sol)
        if c_sol < c_start + 1e-10:
            return h_sol, c_sol
    except Exception:
        pass
    return None, None

def local_descent(h, offsets, max_rounds=50, improve_tol=1e-14):
    n = len(h)
    best_h = h.copy()
    best_c = exact_score(h)
    dx = 2.0 / n

    for rnd in range(max_rounds):
        profile = get_profile(best_h)
        C = np.max(profile)
        active_mask = profile >= C - 0.005
        n_active = int(np.sum(active_mask))
        if n_active == 0:
            active_mask = np.ones(len(profile), dtype=bool)
        G = lag_gradient_matrix(best_h, offsets[active_mask])
        d = G.T.sum(axis=0)
        d = d - np.mean(d)
        norm_d = np.linalg.norm(d)
        if norm_d < 1e-14:
            break
        d = d / norm_d
        improved = False
        for alpha in [0.05, 0.03, 0.02, 0.01, 0.005, 0.003, 0.001]:
            h_try = best_h + alpha * d
            h_try = repair_feasible(h_try, n / 2.0)
            c_try = exact_score(h_try)
            if c_try < best_c - improve_tol:
                best_h = h_try
                best_c = c_try
                improved = True
                break
        if not improved:
            for alpha in [0.01, 0.005, 0.002, 0.001, 5e-4, 2e-4]:
                h_try = best_h + alpha * d
                h_try = repair_feasible(h_try, n / 2.0)
                c_try = exact_score(h_try)
                if c_try < best_c - improve_tol:
                    best_h = h_try
                    best_c = c_try
                    improved = True
                    break
        if not improved:
            break
    return best_h, best_c

def symmetrize(h):
    return (h + h[::-1]) / 2.0

def perturbed_starts(h, rng, n_pert=5, scale=0.01):
    n = len(h)
    starts = [h.copy()]
    target = n / 2.0
    for _ in range(n_pert):
        noise = rng.uniform(-scale, scale, size=n)
        noise -= noise.mean()
        h_p = repair_feasible(h + noise, target)
        starts.append(h_p)
    return starts

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()

    def time_left():
        return budget_s - (time.perf_counter() - t_start)

    # Get parent
    try:
        parent = np.asarray(initial_h_values, dtype=np.float64)
    except NameError:
        parent = None

    rng = np.random.RandomState(seed)
    best_h = None
    best_c = np.inf

    if parent is not None and len(parent) > 0:
        n_p = len(parent)
        target_p = n_p / 2.0
        h_par = repair_feasible(parent, target_p)
        c_par = exact_score(h_par)
        if c_par < best_c:
            best_h = h_par.copy()
            best_c = c_par

    if best_h is None:
        n_fb = 70
        h_fb = np.full(n_fb, 0.5)
        h_fb = repair_feasible(h_fb, n_fb / 2.0)
        best_h = h_fb
        best_c = exact_score(h_fb)

    target = 0.3808
    n_best = len(best_h)
    offsets = get_offsets(n_best)

    # Stage 1: SLSQP on parent dimension
    if time_left() > 60:
        h_sym = repair_feasible(symmetrize(best_h), n_best / 2.0)
        c_sym = exact_score(h_sym)
        if c_sym < best_c:
            best_h, best_c = h_sym, c_sym

        starts = perturbed_starts(best_h, rng, n_pert=4, scale=0.008)
        for s_idx, h_st in enumerate(starts):
            if time_left() < 30:
                break
            for beta in [0.0, 1e-5, 3e-5]:
                if time_left() < 15:
                    break
                maxit = min(200, max(30, int(time_left() / 0.5)))
                h_sol, c_sol = solve_epigraph(h_st, offsets, beta=beta, maxiter=maxit)
                if h_sol is not None and c_sol < best_c:
                    best_h, best_c = h_sol, c_sol
                    if best_c <= target:
                        return best_h, best_c, len(best_h)

    # Stage 2: Local active-set descent
    if time_left() > 30:
        h_desc, c_desc = local_descent(best_h, offsets, max_rounds=80)
        if c_desc < best_c:
            best_h, best_c = h_desc, c_desc
            if best_c <= target:
                return best_h, best_c, len(best_h)

    # Stage 3: Try different dimensions
    if time_left() > 60:
        n_cur = len(best_h)
        dim_candidates = []
        for dn in [-8, -4, -2, 2, 4, 8, 16]:
            nd = n_cur + dn
            if 40 <= nd <= 150:
                dim_candidates.append(nd)
        # Also try parent*2 if small
        if n_cur <= 80:
            dim_candidates.append(n_cur * 2)

        for nd in dim_candidates:
            if time_left() < 40:
                break
            h_int = interpolate_dim(best_h, nd)
            h_int = repair_feasible(h_int, nd / 2.0)
            c_int = exact_score(h_int)
            offsets_nd = get_offsets(nd)

            # Try SLSQP at new dimension
            maxit = min(100, max(20, int(time_left() / 1.0)))
            h_sol, c_sol = solve_epigraph(h_int, offsets_nd, beta=0.0, maxiter=maxit)
            if h_sol is not None and c_sol < best_c:
                best_h, best_c = h_sol, c_sol
                if best_c <= target:
                    return best_h, best_c, len(best_h)
            elif c_int < best_c:
                best_h, best_c = h_int, c_int

            # Local descent at new dimension
            if time_left() > 30:
                h_desc, c_desc = local_descent(h_binary if False else h_int, offsets_nd, max_rounds=40)
                if c_desc < best_c:
                    best_h, best_c = h_desc, c_desc
                    if best_c <= target:
                        return best_h, best_c, len(best_h)

    # Stage 4: Final polish with multiple SLSQP tries
    if time_left() > 30:
        n_fin = len(best_h)
        offsets_fin = get_offsets(n_fin)
        starts = perturbed_starts(best_h, rng, n_pert=3, scale=0.003)
        for h_st in starts:
            if time_left() < 20:
                break
            maxit = min(150, max(20, int(time_left() / 0.3)))
            h_sol, c_sol = solve_epigraph(h_st, offsets_fin, beta=0.0, maxiter=maxit)
            if h_sol is not None and c_sol < best_c:
                best_h, best_c = h_sol, c_sol
                if best_c <= target:
                    return best_h, best_c, len(best_h)

    # Stage 5: Final local descent
    if time_left() > 15:
        n_fin = len(best_h)
        offsets_fin = get_offsets(n_fin)
        h_desc, c_desc = local_descent(best_h, offsets_fin, max_rounds=50)
        if c_desc < best_c:
            best_h, best_c = h_desc, c_desc

    # Final: ensure feasibility and return
    best_h = repair_feasible(best_h, len(best_h) / 2.0)
    final_c = exact_score(best_h)
    return best_h, final_c, len(best_h)
