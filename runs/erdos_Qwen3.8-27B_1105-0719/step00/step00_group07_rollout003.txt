def project_to_mass(h, target):
    """Project h to have sum exactly target, keeping values in [0,1]."""
    h = np.clip(h, 0, 1).astype(float).copy()
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # Water-filling
    lo, hi = -1.0, 1.0
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo+hi)/2, 0, 1)
    # Fix remaining error
    rem = target - h.sum()
    if abs(rem) > 1e-14:
        # Distribute to positions with room
        if rem > 0:
            room = 1.0 - h
            order = np.argsort(-room)
        else:
            room = h
            order = np.argsort(room)
        idx = order[0]
