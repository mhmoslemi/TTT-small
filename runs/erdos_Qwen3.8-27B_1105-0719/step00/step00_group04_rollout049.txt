def repair_feasible(h):
    """Repair h to be exactly feasible: 0<=h<=1, sum(h)=n/2."""
    n = len(h)
    h = np.clip(np.asarray(h, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        # Need to add mass. Add to entries with most room.
        room = 1.0 - h
        order = np.argsort(-room)
        for i in order:
            add = min(room[i], diff)
            h[i] += add
            diff -= add
            if diff <= 1e-14:
                break
    else:
        # Need to remove mass. Remove from entries with most value.
        diff = -diff
        order = np.argsort(-h)
        for i in order:
            rem = min(h[i], diff)
            h[i] -= rem
            diff -= rem
            if diff <= 1e-14:
                break
    # Final exact fix: adjust one element
    residual = target - h.sum()
    if abs(residual) > 0:
        # Find an element we can adjust
        for i in range(n):
            if residual > 0 and h[i] < 1.0:
                adj = min(residual, 1.0 - h[i])
                h[i] += adj
                residual -= adj
            elif residual < 0 and h[i] > 0.0:
                adj = max(residual, -h[i])
                h[i] += adj
                residual -= adj
            if abs(residual) < 1e-14:
                break
    return h

                continue
            h_interp = interpolate_to_dim(best_h[:best_n], nd)
            h_interp = repair_feasible(h_interp)
            s_interp = exact_score(h_interp)
            consider(h_interp)

            # NLP on interpolated
            if time_left() > 30:
                nlp = EpigraphNLP(nd, beta=0.0)
                h_sol = nlp.solve(h_interp, s_interp + 1e-6, maxiter=150)
                if h_sol is not None:
                    consider(h_sol)

            # Local descent
            if time_left() > 10:
                h_ref, s_ref = local_descent(h_interp, max_rounds=30)
                consider(h_ref)

    # Stage 6: More perturbations from best
    if time_left() > 30:
        for s in range(5):
            if time_left() < 25:
                break
            for eps in [0.02, 0.01, 0.005]:
                d = rng.randn(best_n)
                d -= d.mean()
                std = np.std(d)
                if std < 1e-10:
                    continue
                h_pert = repair_feasible(best_h[:best_n] + eps * d / std)
                consider(h_pert)
                if time_left() > 25:
                    nlp = EpigraphNLP(best_n, beta=0.0)
                    h_sol = nlp.solve(h_pert, exact_score(h_pert) + 1e-6, maxiter=100)
                    if h_sol is not None:
                        consider(h_sol)

    # Stage 7: Final local refinement
    if time_left() > 5:
        h_ref, s_ref = local_descent(best_h[:best_n], max_rounds=50)
        consider(h_ref)

    # Stage 8: Symmetry averaging
    if time_left() > 5:
        h_sym = 0.5 * (best_h[:best_n] + best_h[:best_n][::-1])
        h_sym = repair_feasible(h_sym)
        consider(h_sym)

    # Final: ensure best_h is feasible and valid
    best_h = repair_feasible(best_h[:best_n])
    best_score = exact_score(best_h)
    return (best_h, best_score, int(best_n))
