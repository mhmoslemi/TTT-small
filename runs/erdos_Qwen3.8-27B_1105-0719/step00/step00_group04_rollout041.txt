def gradient_subset(h, offsets, dx, rows):

def repair_vector(v):
    x = np.asarray(v, dtype=np.float64)
    if x.ndim != 1:
        x = np.ravel(x)
    n = x.size
    if n == 0:
        return x
    target = 0.5 * n
    x = x.copy()
    bad = ~np.isfinite(x)
    if np.any(bad):
        x[bad] = 0.5
    x = np.clip(x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) <= 1e-14:
        return x
    if diff > 0.0:
        cap = 1.0 - x
        cap_sum = cap.sum()
        if cap_sum > 1e-15:
            x += (diff / cap_sum) * cap
        else:
            x.fill(0.5)
            x += target / n  # if n>0
    else:
        room = x.copy()
        room_sum = room.sum()
        if room_sum > 1e-15:
            x += (diff / room_sum) * room
        else:
            # impossible if target>0, set half
            x.fill(0.5)
    diff = target - x.sum()
    if abs(diff) > 1e-13:
        for _ in range(8):

def transfer_dimension(h, target_n):

        return None
    return p


def profile_rows(h, offsets, dx, rows=None):
    n = len(h)
    if rows is None:
        rows = np.arange(len(offsets), dtype=np.int64)
    m = len(rows)
    out = np.empty(m, dtype=np.float64)
    for k in range(m):
        d = int(offsets[int(rows[k])])
        if d >= 0:
            end = n - d
            if end > 0:
                out[k] = dx * float(np.dot(h[:end], 1.0 - h[d:d + end]))
            else:
                out[k] = 0.0
        else:
            lo = -d
            lo = min(lo, n)
            if n - lo > 0:
                out[k] = dx * float(np.dot(h[lo:n], 1.0 - h[:lo]))
            else:
                out[k] = 0.0
    return out


def gradient_subset(h, offsets, dx, rows):
    n = len(h)
    G = np.zeros((len(rows), n), dtype=np.float64)
    for k in range(len(rows)):
        d = int(offsets[int(rows[k])])
        if d >= 0:
            end = n - d
            if end > 0:
                G[k, :end] += dx * (1.0 - h[d:d + end])
                G[k, d:d + end] += -dx * h[:end]
        else:
            lo = -d
            end = n + d
            if end > 0 and lo < n:
                G[k, lo:n] += dx * (1.0 - h[:lo])
                G[k, :end] += -dx * h[lo:lo + end]
    return G


def infer_offsets(n):
    n = int(n)
    if n <= 0:
        return np.array([], dtype=np.int64)
    m = 2 * n - 1
    offs1 = (n - 1) - np.arange(m, dtype=np.int64)
    offs2 = np.arange(m, dtype=np.int64) - (n - 1)
    rng = np.random.default_rng(999983 + n)
    h = rng.uniform(0.0, 1.0, n)
    h = repair_vector(h)
    dx = 2.0 / float(n)
    e = exact_profile(h)
    p1 = profile_rows(h, offs1, dx)
    p2 = profile_rows(h, offs2, dx)
    e1 = float(np.max(np.abs(p1 - e)))
    e2 = float(np.max(np.abs(p2 - e)))
    if e1 <= e2:
        return offs1.copy()
    return offs2.copy()


def set_working(ctx, n):
    n = int(n)
    if n <= 0:
        return
    cache = ctx.setdefault("offset_cache", {})
    if n not in cache:
        cache[n] = infer_offsets(n)
    ctx["n"] = n
    ctx["dx"] = 2.0 / float(n)
    ctx["target"] = 0.5 * float(n)
    ctx["offsets"] = cache[n]


def time_left(ctx):
    if "deadline" not in ctx:
        return 0.0
    return max(0.0, float(ctx["deadline"]) - time.perf_counter())


def consider(ctx, h):
    if h is None:
        return False
    h = np.ravel(h)
    n = int(h.size)
    if n <= 0:
        return False
    hr = repair_vector(h)
    s = exact_score(hr)
    if not np.isfinite(s):
        return False
    best = ctx.get("best_score")
    if best is None or not np.isfinite(best):
        ctx["best_h"] = hr
        ctx["best_score"] = float(s)
        ctx["best_n"] = n
        return True
    if s < best - ACCEPT_TOL:
        ctx["best_h"] = hr
        ctx["best_score"] = float(s)
        ctx["best_n"] = n
        return True
    if s <= best + 1e-12 and n < int(ctx.get("best_n", n)):
        ctx["best_h"] = hr
        ctx["best_score"] = float(s)
        ctx["best_n"] = n
        return True
    return False


def select_active(profile, fraction=0.40, min_k=8, max_k=None, margin=0.01):
    M = len(profile)
    if M == 0:
        return np.array([], dtype=np.int64)
    if max_k is None:
        max_k = M
    mx = float(np.max(profile))
    k = int(min(M, max(min_k, int(round(fraction * M)))))
    top = np.argsort(profile)[::-1][:max(k, min_k)]
    extra = np.where(profile >= mx - margin)[0]
    idx = np.unique(np.concatenate((top, extra)))
    if idx.size > max_k:
        order = np.argsort(profile[idx])[::-1]
        idx = idx[order[:max_k]]
    am = int(np.argmax(profile))
    if not np.any(idx == am):
        idx = np.concatenate((idx, np.array([am], dtype=np.int64)))
        if idx.size > max_k:
            order = np.argsort(profile[idx])[::-1]
            idx = idx[order[:max_k]]
    return idx.astype(np.int64)


def obj_fun(x, ctx):
    val = float(x[-1])
    bet = float(ctx.get("beta", 0.0))
    if bet != 0.0:
        h = x[:-1]
        val += bet * float(ctx["dx"]) * float(np.dot(h, 1.0 - h))
    if not np.isfinite(val):
        return 1e100
    return val


def obj_jac(x, ctx):
    n = int(ctx["n"])
    g = np.zeros(n + 1, dtype=np.float64)
    g[-1] = 1.0
    bet = float(ctx.get("beta", 0.0))
    if bet != 0.0:
        h = x[:-1]
        g[:-1] = bet * float(ctx["dx"]) * (1.0 - 2.0 * h)
    return g


def eq_fun(x, ctx):
    return float(x[:-1].sum()) - float(ctx["target"])


def eq_jac(x, ctx):
    n = int(ctx["n"])
    J = np.zeros((1, n + 1), dtype=np.float64)
    J[0, :n] = 1.0
    return J


def active_rows(ctx):
    act = ctx.get("active_idx")
    if act is None:
        return np.arange(len(ctx["offsets"]), dtype=np.int64)
    return np.asarray(act, dtype=np.int64)


def ineq_fun(x, ctx):
    h = x[:-1]
    t = float(x[-1])
    rows = active_rows(ctx)
    if len(rows) == len(ctx["offsets"]):
        prof = exact_profile(h)
    else:
        prof = profile_rows(h, ctx["offsets"], ctx["dx"], rows)
    return t - prof


def ineq_jac(x, ctx):
    h = x[:-1]
    rows = active_rows(ctx)
    G = gradient_subset(h, ctx["offsets"], ctx["dx"], rows)
    J = np.zeros((G.shape[0], G.shape[1] + 1), dtype=np.float64)
    J[:, :-1] = -G
    J[:, -1] = 1.0
    return J


def minimize_once(ctx, h, t0, maxiter):
    n = int(ctx["n"])
    x0 = np.concatenate((h, np.array([float(t0)], dtype=np.float64)))
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    constraints = [
        {"type": "eq", "fun": eq_fun, "jac": eq_jac, "args": (ctx,)},
        {"type": "ineq", "fun": ineq_fun, "jac": ineq_jac, "args": (ctx,)}
    ]
    try:
        res = minimize(
            obj_fun,
            x0,
            method="SLSQP",
            jac=obj_jac,
            bounds=bounds,
            constraints=constraints,
            args=(ctx,),
            options={"maxiter": int(maxiter), "ftol": 1e-12, "disp": 0}
        )
    except Exception:
        return None
    if res is None or not hasattr(res, "x") or res.x is None:
        return None
    x = np.asarray(res.x, dtype=np.float64)
    if not np.all(np.isfinite(x)):
        return None
    return x


def choose_maxiter(n, full, rem, cap):
    if cap <= 0:
        return 0
    if rem < 45.0:
        return 0
    if n <= 90:
        base = 170 if full else 95
    elif n <= 160:
        base = 110 if full else 65
    else:
        base = 75 if full else 45
    if rem < 200.0:
        base = int(base * 0.7)
    if rem < 100.0:
        base = int(base * 0.5)
    return int(max(20, min(int(cap), int(base))))


def epigraph_improve(ctx, h0, beta=0.0, use_full=False, active_fraction=0.40,
                     max_inner=3, maxiter_cap=160):
    h = repair_vector(np.ravel(h0))
    if h.size == 0:
        return h
    set_working(ctx, h.size)
    prof = exact_profile(h)
    score = float(np.max(prof)) if prof.size else np.inf
    if not np.isfinite(score):
        return h

    full_eff = bool(use_full)
    cur_fraction = float(active_fraction)

    for _ in range(int(max_inner)):
        if time_left(ctx) < 30.0:
            break
        n = int(ctx["n"])
        if full_eff:
            act = None
        else:
            act = select_active(prof, fraction=cur_fraction)
        ctx["beta"] = float(beta)
        ctx["active_idx"] = act

        t0 = score + 1e-4
        if t0 < 1e-8:
            t0 = 1e-8
        mi = choose_maxiter(n, full_eff, time_left(ctx), maxiter_cap)
        if mi <= 0:
            break

        x = minimize_once(ctx, h, t0, mi)
        if x is None:
            break

        h_sol = repair_vector(x[:-1])
        prof_sol = exact_profile(h_sol)
        score_sol = float(np.max(prof_sol)) if prof_sol.size else np.inf
        consider(ctx, h_sol)

        if np.isfinite(score_sol) and score_sol < score - 1e-10:
            h = h_sol
            score = score_sol
            prof = prof_sol
            if not full_eff:
                cur_fraction = min(1.0, cur_fraction + 0.12)
            continue

        if not full_eff:
            full_eff = True
            continue
        break

    return h


def transfer_dimension(h, target_n):
    h = np.ravel(h)
    n = int(h.size)
    target_n = int(target_n)
    if n == 0:
        return h.copy()
    if target_n <= 0:
        return h.copy()
    if target_n % 2:
        target_n += 1
    if target_n == n:
        return repair_vector(h)

    dxs = 2.0 / float(n)
    xs = (np.arange(n, dtype=np.float64) + 0.5) * dxs
    xts = (np.arange(target_n, dtype=np.float64) + 0.5) * (2.0 / float(target_n))

    xa = np.concatenate((np.array([0.0], dtype=np.float64), xs, np.array([2.0], dtype=np.float64)))
    fa = np.concatenate((np.array([float(h[0])], dtype=np.float64), h, np.array([float(h[-1])], dtype=np.float64)))
    hn = np.interp(xts, xa, fa)
    return repair_vector(hn)


def test_direction(ctx, h_base, D, alphas, current_score):
    D = np.asarray(D, dtype=np.float64).copy()
    if D.size != h_base.size:
        return h_base, current_score
    D -= float(np.mean(D))
    norm = float(np.max(np.abs(D)))
    if not np.isfinite(norm) or norm < 1e-12:
        return h_base, current_score
    D /= norm

    best_h = h_base
    best_s = float(current_score)
    for a in alphas:
        a = float(a)
        cand = h_base + a * D
        hr = repair_vector(cand)
        s = exact_score(hr)
        if np.isfinite(s) and s < best_s - 1e-13:
            best_s = float(s)
            best_h = hr
    if best_h is not h_base:
        consider(ctx, best_h)
    return best_h, best_s


def local_gradient_search(ctx, h0, max_seconds=120.0):
    h = repair_vector(np.ravel(h0))
    if h.size == 0:
        return h
    set_working(ctx, h.size)
    current_score = exact_score(h)
    consider(ctx, h)

    now = time.perf_counter()
    local_deadline = now + float(max_seconds)
    if "deadline" in ctx:
        local_deadline = min(local_deadline, float(ctx["deadline"]))

    dx = float(ctx["dx"])
    offsets = ctx["offsets"]
    n = int(ctx["n"])
    M = 2 * n - 1
    K = int(min(M, max(8, int(round(0.35 * M)))))
    alphas = [0.20, 0.12, 0.07, -0.12, 0.05, -0.05, 0.02, -0.02]

    for _ in range(6):
        if time.perf_counter() >= local_deadline or time_left(ctx) < 25.0:
            break

        best_score = ctx.get("best_score", np.inf)
        best_n = int(ctx.get("best_n", 0))
        if best_n == n and np.isfinite(best_score) and best_score < current_score - 1e-12:
            h = repair_vector(ctx["best_h"])
            current_score = exact_score(h)

        prof = exact_profile(h)
        if prof.size == 0:
            break
        active = np.argsort(prof)[::-1][:K]
        G = gradient_subset(h, offsets, dx, active)

        old_score = current_score
        w = np.exp(-0.1 * np.arange(len(active), dtype=np.float64))
        D = -np.sum(G * w[:, None], axis=0)
        h, current_score = test_direction(ctx, h, D, alphas, current_score)
        if current_score < old_score - 1e-12:
            continue

        row_count = int(min(8, len(active)))
        improved_row = False
        for ri in range(row_count):
            if time.perf_counter() >= local_deadline or time_left(ctx) < 25.0:
                break
            old = current_score
            D = -G[ri]
            h, current_score = test_direction(ctx, h, D, alphas, current_score)
            if current_score < old - 1e-12:
                improved_row = True
                best_score = ctx.get("best_score", np.inf)
                best_n = int(ctx.get("best_n", 0))
                if best_n == n and np.isfinite(best_score) and best_score < current_score - 1e-12:
                    h = repair_vector(ctx["best_h"])
                    current_score = exact_score(h)
                break
        if improved_row:
            continue
        break

    return h


def finalize(ctx):
    h = ctx.get("best_h")
    if h is None:
        h = np.full(70, 0.5, dtype=np.float64)
    else:
        h = repair_vector(np.ravel(h))
    if h.size == 0:
        h = np.full(70, 0.5, dtype=np.float64)

    s = exact_score(h)
    try:
        ext = evaluate_erdos_solution(h)
        if isinstance(ext, (tuple, list)):
            ext = ext[0]
        ext = float(ext)
        if np.isfinite(ext):
            s = ext
    except Exception:
        pass

    if not np.isfinite(s):
        h = np.full(70, 0.5, dtype=np.float64)
        s = exact_score(h)
    return h, float(s), int(h.size)


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()
    try:
        budget_s = float(budget_s)
    except Exception:
        budget_s = 1100.0
    if not np.isfinite(budget_s) or budget_s < 5.0:
        budget_s = 1100.0

    deadline = t_start + budget_s * 0.94
    ctx = {
        "deadline": deadline,
        "start": t_start,
        "best_score": np.inf,
        "best_h": None,
        "best_n": 0,
        "offset_cache": {}
    }

    parent = get_parent()
    if parent is not None:
        consider(ctx, parent)
    if ctx.get("best_h") is None:
        consider(ctx, np.full(70, 0.5, dtype=np.float64))

    set_working(ctx, int(ctx.get("best_n", 70)))

    if time_left(ctx) > 5.0:
        h = repair_vector(ctx["best_h"])
        consider(ctx, h[::-1].copy())
        consider(ctx, 0.5 * (h + h[::-1]))

        if time_left(ctx) > 40.0:
            rng = np.random.default_rng(int(seed))
            n = int(h.size)
            for sig in (0.02, 0.05, 0.10):
                z = rng.normal(0.0, sig, n)
                z -= float(z.mean())
                consider(ctx, h + z)

    base_h = repair_vector(ctx["best_h"])
    set_working(ctx, int(base_h.size))

    if time_left(ctx) > 90.0:
        epigraph_improve(ctx, base_h, beta=0.0, use_full=False,
                         active_fraction=0.35, max_inner=3, maxiter_cap=160)

    if time_left(ctx) > 100.0:
        epigraph_improve(ctx, repair_vector(ctx["best_h"]), beta=0.0,
                         use_full=True, active_fraction=0.0,
                         max_inner=2, maxiter_cap=220)

    if time_left(ctx) > 100.0:
        base = repair_vector(ctx["best_h"])
        starts = [base[::-1].copy(), 0.5 * (base + base[::-1])]
        rng = np.random.default_rng(int(seed) + 17)
        n = int(base.size)
        z = rng.normal(0.0, 0.06, n)
        z -= float(z.mean())
        starts.append(base + z)

        for st in starts:
            if time_left(ctx) < 70.0:
                break
            set_working(ctx, int(st.size))
            epigraph_improve(ctx, st, beta=0.0, use_full=False,
                             active_fraction=0.45, max_inner=2, maxiter_cap=140)

    if time_left(ctx) > 80.0:
        base = repair_vector(ctx["best_h"])
        set_working(ctx, int(base.size))
        epigraph_improve(ctx, base, beta=1e-5, use_full=True,
                         active_fraction=0.50, max_inner=2, maxiter_cap=180)

    if time_left(ctx) > 110.0:
        base = repair_vector(ctx["best_h"])
        base_n = int(base.size)
        raw = [base_n * 2, int(1.5 * base_n)]
        for add in (20, 40, 60, 80, -20, -40, -60):
            raw.append(base_n + add)
        raw.extend([120, 140, 160, 200, 240, 280])

        n_options = []
        seen = set()
        for nn in raw:
            nn = int(nn)
            if nn % 2:
                nn += 1
            if 20 <= nn <= 400 and nn != base_n and nn not in seen:
                seen.add(nn)
                n_options.append(nn)

        for nn in n_options:
            if time_left(ctx) < 75.0:
                break
            h2 = transfer_dimension(base, nn)
            s2 = exact_score(h2)
            if not np.isfinite(s2) or s2 > ctx["best_score"] + 0.08:
                continue

            consider(ctx, h2)
            set_working(ctx, int(nn))
            epigraph_improve(ctx, h2, beta=0.0, use_full=False,
                             active_fraction=0.45, max_inner=2, maxiter_cap=150)

            if time_left(ctx) > 85.0 and int(ctx.get("best_n", 0)) == int(nn):
                epigraph_improve(ctx, repair_vector(ctx["best_h"]), beta=0.0,
                                 use_full=True, active_fraction=0.0,
                                 max_inner=1, maxiter_cap=180)

            base = repair_vector(ctx["best_h"])

    if time_left(ctx) > 70.0:
        best = repair_vector(ctx["best_h"])
        set_working(ctx, int(best.size))
        seconds = min(180.0, max(25.0, time_left(ctx) - 30.0))
        local_gradient_search(ctx, best, max_seconds=seconds)

    if time_left(ctx) > 80.0:
        best = repair_vector(ctx["best_h"])
        set_working(ctx, int(best.size))
        epigraph_improve(ctx, best, beta=0.0, use_full=True,
                         active_fraction=0.0, max_inner=2, maxiter_cap=260)

    return finalize(ctx)
