import numpy as np
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def project_feasible(h, target):
    n = len(h)
    h = np.asarray(h, dtype=np.float64).copy()
    h = np.clip(h, 0.0, 1.0)
    # Bisection for shift
    for _ in range(100):
        s = float(np.sum(h))
        if abs(s - target) < 1e-12:
            break
        diff = target - s
        if diff > 0:
            idx = np.where(h < 1.0)[0]
        else:
            idx = np.where(h > 0.0)[0]
        if len(idx) == 0:
            break
        h[idx] += diff / len(idx)
        h = np.clip(h, 0.0, 1.0)
    # Final exact fix
    residual = target - float(np.sum(h))
    if abs(residual) > 1e-14:
        if residual > 0:
            idx = np.where(h < 1.0)[0]
        else:
            idx = np.where(h > 0.0)[0]
        if len(idx) > 0:
            i = idx[0]
            h[i] = np.clip(h[i] + residual, 0.0, 1.0)
    return h

def pair_descent(h, t_start, time_limit, golden_iters=30):
    n = len(h)
    h = np.array(h, dtype=np.float64, copy=True)
    target = n / 2.0
    best_c = exact_score(h)
    
    for iteration in range(500):
        if time.perf_counter() - t_start > time_limit:
            break
        improved = False
        for p in range(n):
            for q in range(p + 1, n):
                s = h[p] + h[q]
                lo = max(0.0, s - 1.0)
                hi = min(1.0, s)
                if hi - lo < 1e-12:
                    continue
                # Golden section on [lo, hi]
                a_lo = lo
                a_hi = hi
                gr = (np.sqrt(5.0) - 1.0) / 2.0  # 0.618...
                m1 = a_hi - gr * (a_hi - a_lo)
                m2 = a_lo + gr * (a_hi - a_lo)
                c1 = _eval_pair(h, p, q, m1, s)
                c2 = _eval_pair(h, p, q, m2, s)
                for _ in range(golden_iters):
                    if c1 < c2:
                        a_hi = m2
                        m2 = m1
                        c2 = c1
                        m1 = a_hi - gr * (a_hi - a_lo)
                        c1 = _eval_pair(h, p, q, m1, s)
                    else:
                        a_lo = m1
                        m1 = m2
                        c1 = c2
                        m2 = a_lo + gr * (a_hi - a_lo)
                        c2 = _eval_pair(h, p, q, m2, s)
                a_star = (a_lo + a_hi) * 0.5
                c_star = _eval_pair(h, p, q, a_star, s)
                if c_star < best_c - 1e-12:
                    h[p] = a_star
                    h[q] = s - a_star
                    best_c = c_star
                    improved = True
        if not improved:

    h = project_feasible(h, target)
    c_final = exact_score(h)
    if c_final < best_c:

    return h, best_c
