import numpy as np

import time


def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n), corr


def project_feasible(y, target):
    y = np.asarray(y, dtype=float).copy()
    n = len(y)
    lo = y.min() - 1.5
    hi = y.max() + 1.5
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    x = np.clip(y - lam, 0.0, 1.0)
    r = target - x.sum()
    if abs(r) > 1e-13:
        if r > 0:
            slack = 1.0 - x
            order = np.argsort(-slack)
            for i in order:
                add = min(r, slack[i])
                if add <= 1e-15:
                    break
                x[i] += add
                r -= add
        else:
            order = np.argsort(-x)
            for i in order:
                sub = min(-r, x[i])
                if sub <= 1e-15:
                    break
                x[i] -= sub
                r += sub
    return x


def grad_lse(h, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    shifted = np.clip(beta * (corr - m), -500.0, 0.0)
    e = np.exp(shifted)
    w = e / e.sum()
    grad = np.zeros(n)
    for t in range(2 * n - 1):
        wt = w[t]
        if wt < 1e-16:
            continue
        lag = t - (n - 1)
        if lag >= 0:
            if lag < n:
                idx = np.arange(lag, n)
                grad[idx] += wt * (1.0 - h[idx - lag])
                idx2 = np.arange(0, n - lag)
                grad[idx2] -= wt * h[idx2 + lag]
        else:
            m2 = -lag
            if m2 < n:
                idx = np.arange(0, n - m2)
                grad[idx] += wt * (1.0 - h[idx + m2])
                idx2 = np.arange(m2, n)
                grad[idx2] -= wt * h[idx2 - m2]
    grad -= np.mean(grad)
    return grad


def gradient_descent(h, target, beta, max_iters, lr_init, lr_cap, time_limit_fn):
    h = h.copy()
    lr = lr_init
    best_local = h.copy()
    best_c5_local = compute_c5(h)[0]
    for it in range(max_iters):
        if time_limit_fn():
            break
        grad = grad_lse(h, beta)
        gn = np.linalg.norm(grad)
        if gn < 1e-12:
            break
        grad = grad / gn * min(gn, lr_cap)
        h_new = project_feasible(h - lr * grad, target)
        c5_new = compute_c5(h_new)[0]
        if c5_new < best_c5_local:
            best_c5_local = c5_new
            best_local = h_new.copy()
            h = h_new.copy()
            lr = min(lr * 1.03, 0.1)
        else:
            lr *= 0.95
            if lr < 1e-6:
                break
    return best_local, best_c5_local
