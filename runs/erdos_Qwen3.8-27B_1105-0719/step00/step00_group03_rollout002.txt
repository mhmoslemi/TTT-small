import numpy as np
import time

def c5_eval(h):
    """Compute C5 = max(np.correlate(h, 1-h, mode='full') * dx)"""
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1-h, mode='full') * dx)

def project_to_box_sum(h, target_sum):
    """Project h to [0,1]^n with sum = target_sum using bisection on Lagrange multiplier."""
    n = len(h)
    # Bisection on lambda: x_i = clip(h[i] + lambda, 0, 1)
    # We need sum(x) = target_sum
    lo, hi = -h.max() - 1, 1 - h.min() + 1
    for _ in range(100):
        mid = (lo + hi) / 2
        x = np.clip(h + mid, 0, 1)
        s = x.sum()
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    x = np.clip(h + (lo + hi) / 2, 0, 1)
    # Fix any tiny residual
    residual = target_sum - x.sum()
    if abs(residual) > 1e-12:
        # Distribute residual
        if residual > 0:
            # Need to add mass; add to elements with most room
            idx = np.argsort(1 - x)
            for i in range(n):
                add = min(residual, 1 - x[idx[i]])
                x[idx[i]] += add
                residual -= add
                if abs(residual) < 1e-15:
                    break
        else:
            idx = np.argsort(x)
            for i in range(n):
                sub = min(-residual, x[idx[i]])
                x[idx[i]] -= sub
                residual += sub
                if abs(residual) < 1e-15:
                    break
    return x

def pair_line_search(h, p, q, current_c5, dx):
    """Find best tau for transfer from q to p. Returns (best_tau, best_c5)."""
    # tau range: h[p] + tau in [0,1] and h[q] - tau in [0,1]
    # So tau in [max(-h[p], h[q]-1), min(1-h[p], h[q])]
    tau_lo = max(-h[p], h[q] - 1.0)
    tau_hi = min(1.0 - h[p], h[q])
    if tau_hi - tau_lo < 1e-12:
        return 0.0, current_c5
    
    # Grid search
    n_grid = 15
    taus = np.linspace(tau_lo, tau_hi, n_grid)
    
    best_tau = 0.0
    best_c5 = current_c5
    
    h_orig = h.copy()
    for tau in taus:
        h[p] = h_orig[p] + tau
        h[q] = h_orig[q] - tau
        c5 = c5_eval(h)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_tau = tau
    h[p] = h_orig[p]
    h[q] = h_orig[q]
    
    return best_tau, best_c5

def coordinate_descent(h, max_sweeps, time_limit, start_time, seed=42):
    """Perform coordinate descent with pair transfers."""
    n = len(h)
    dx = 2.0 / n
    current_c5 = c5_eval(h)
    rng = np.random.RandomState(seed)
    
    best_h = h.copy()
    best_c5 = current_c5
    
    for sweep in range(max_sweeps):
        if time.time() - start_time > time_limit:
            break
        
        improved = False
        # Random order for diversity
        pairs = list(range(n * (n-1) // 2))
        rng.shuffle(pairs)
        # Actually let's just iterate all pairs in random order
        indices = list(range(n))
        rng.shuffle(indices)
        
        for i in range(n):
            for j in range(i+1, n):
                p, q = indices[i], indices[j]
                tau, new_c5 = pair_line_search(h, p, q, current_c5, dx)
                if new_c5 < current_c5 - 1e-12:
                    h[p] += tau
                    h[q] -= tau
                    # Clip for safety
                    h[p] = np.clip(h[p], 0, 1)
                    h[q] = np.clip(h[q], 0, 1)
                    current_c5 = new_c5
                    improved = True
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_h = h.copy()
        
        if not improved:
            break
    
    return best_h, best_c5
