    lo, hi = -2.0, 2.0

            lo = mid
        else:
            hi = mid
    result = np.clip(h + (lo + hi) / 2, 0, 1)
    # Fix rounding
    diff = target - np.sum(result)
    if abs(diff) > 1e-14:

                diff -= add

                sub = min(-diff, result[i])
                result[i] -= sub
                diff += sub
            if abs(diff) < 1e-14:
                break
    return result

    corr = np.correlate(h, 1.0 - h, mode='full')
    c_values = corr * dx
    
    grads = np.zeros((2*n - 1, n))
    
    # Lag 0 at index n-1
    grads[n-1] = (1.0 - 2.0 * h) * dx
    
    for k in range(1, n):
        # Positive lag at index n-1+k
        g_pos = np.zeros(n)
        g_pos[:n-k] += -h[k:]
        g_pos[k:] += (1.0 - h[:n-k])
        grads[n-1+k] = g_pos * dx
        
        # Negative lag at index n-1-k
        g_neg = np.zeros(n)
        g_neg[:n-k] += (1.0 - h[k:])
        g_neg[k:] += -h[:n-k]
        grads[n-1-k] = g_neg * dx
    
    return c_values, grads

def smooth_max_and_grad(h, beta):
    n = len(h)
    c_values, grads = compute_lag_grads(h)
    
    max_c = np.max(c_values)
    shifted = beta * (c_values - max_c)
    # Clip to avoid overflow
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    total = np.sum(exp_vals)
    p = exp_vals / total
    
    smooth_val = max_c + np.log(total) / beta
    grad = np.dot(p, grads)  # (2n-1,) dot (2n-1, n) -> (n,)
    
    return smooth_val, grad

def pair_line_search(h, i, j, n_evals=40):
    """Find best eps for transferring from j to i."""
    lo_eps = max(-h[i], h[j] - 1.0)
    hi_eps = min(1.0 - h[i], h[j])
    
    if hi_eps - lo_eps < 1e-10:
        return h, evaluate_c5(h)
    
    # Sample eps values
    eps_candidates = np.linspace(lo_eps, hi_eps, n_evals)
    best_val = np.inf
    best_eps = 0.0
    
    for eps in eps_candidates:
        h_copy = h.copy()
        h_copy[i] += eps
        h_copy[j] -= eps
        val = evaluate_c5(h_copy)
        if val < best_val:
            best_val = val
            best_eps = eps
    
    # Refine with minimize_scalar
    def obj(eps):
        h_copy = h.copy()
        h_copy[i] += eps
        h_copy[j] -= eps
        return evaluate_c5(h_copy)
    
    try:
        res = minimize_scalar(obj, bounds=(lo_eps, hi_eps), method='bounded',
                             options={'maxiter': 50})
        if res.fun < best_val:
            best_val = res.fun
            best_eps = res.x
    except:
        pass
    
    h_new = h.copy()
    h_new[i] += best_eps
    h_new[j] -= best_eps
    return h_new, best_val

    failures = 0
    step_size = 0.1

        gnorm = np.linalg.norm(grad_centered)

            h_new = project_feasible(h_new, n)
            c_new = evaluate_c5(h_new)

                h = h_new
                c = c_new
                if c < best_c:
                    best_c = c
                    best_h = h.copy()
                accepted = True
                step_size = min(step_size * 1.1, 1.0)
                break

            step_size *= 0.5
            failures += 1

        if it % 100 == 99:

    beta = beta_start
    for it in range(max_iters):
        if time.time() > deadline or best_c <= 0.3808:
            break
        _, grad = smooth_max_grad(h, beta)
        grad_c = grad - np.mean(grad)
        gn = np.linalg.norm(grad_c)
        if gn < 1e-12:
            break
        accepted = False
        for factor in [step, step*0.5, step*0.25, step*0.1]:
            h_new = project_feasible(h - factor * gn * grad_c, n)
            c_new = evaluate_c5(h_new)
            if c_new < c - 1e-12 or c_new < best_c - 1e-12:
                h = h_new
                c = c_new
                if c < best_c:
                    best_c = c
                    best_h = h.copy()
                accepted = True
                step = min(step * 1.05, 0.5)
                break
        if not accepted:
            step *= 0.7
            failures += 1
            if failures > 15:
                h = project_feasible(h + 0.03*rng.randn(n), n)
                c = evaluate_c5(h)
                if c < best_c:
                    best_c = c
                    best_h = h.copy()
                failures = 0
                step = 0.1
        if it % 80 == 79:
            beta = min(beta * 1.4, 2000.0)
    return h, best_c, best_h

def run_pair_descent(h, n, deadline, best_c, best_h, max_rounds=5):
    h = project_feasible(h.copy(), n)
    c = evaluate_c5(h)
    if c < best_c:
        best_c = c
        best_h = h.copy()
    idx = np.arange(n)
    for round_idx in range(max_rounds):
        if time.time() > deadline or best_c <= 0.3808:
            break
        _, grad = smooth_max_grad(h, 500.0)
        grad_c = grad - np.mean(grad)
        scores = np.abs(grad_c)
        top = np.argsort(-scores)[:min(16, n)]
        # Build candidate pairs
        pairs = []
        for ai in top:
            for bi in top:
                if ai != bi:
                    pairs.append((int(ai), int(bi)))
        # Add random pairs
        for _ in range(min(10, n)):
            i = rng.randint(n)
            j = rng.randint(n)
            if i != j:
                pairs.append((i, j))
        # Evaluate and apply best pair
        best_pair_val = np.inf
        best_new_h = h.copy()
        for (pi, pj) in pairs:
            if time.time() > deadline:
                break
            h_new, val = pair_line_search(h, pi, pj, n_grid=25)
            if val < best_pair_val:
                best_pair_val = val
                best_new_h = h_new
        if best_pair_val < c - 1e-12:
            h = best_new_h
            c = best_pair_val
            if c < best_c:
                best_c = c
                best_h = h.copy()
        else:
            # Try a random kick
            h = project_feasible(h + 0.05*rng.randn(n), n)
            c = evaluate_c5(h)
            if c < best_c:
                best_c = c
                best_h = h.copy()
    return h, best_c, best_h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3.0
    rng = np.random.RandomState(seed)

    # Get initial
    initial = globals().get('initial_h_values', None)
    if initial is not None and len(initial) > 0 and np.all(np.isfinite(np.asarray(initial))):
        n0 = len(initial)
        h0 = project_feasible(np.asarray(initial, dtype=np.float64), n0)
    else:
        n0 = 42
        h0 = gen_binary_random(n0, rng)

    best_c = evaluate_c5(h0)
    best_h = h0.copy()

    # Dimension plan: keep focused on small n for speed
    dims = sorted(set([n0, max(16, n0-6), n0+6, n0+12]))
    dims = [d for d in dims if 14 <= d <= 70]

    # Phase 1: Multi-dimension search
    for n in dims:
        if time.time() > deadline or best_c <= 0.3808:
            break
        # Start from best_h resampled
        if len(best_h) == n:
            h_start = best_h.copy()
        else:
            h_start = interpolate_resize(best_h, n)
        # Also try random start
        hr = gen_binary_random(n, rng)
        cr = evaluate_c5(hr)
        if cr < best_c:
            best_c = cr
            best_h = hr.copy()
            h_start = hr.copy()

        # Gradient descent with multiple betas
        for beta in [40.0, 100.0, 250.0]:
            if time.time() > deadline or best_c <= 0.3808:
                break
            max_it = max(40, int(2500.0 / n))
            h_start, best_c, best_h = run_gradient_descent(
                h_start, n, beta, deadline, best_c, best_h, max_iters=max_it)

        # Pair descent
        if time.time() < deadline - 20:
            h_start, best_c, best_h = run_pair_descent(
                h_start, n, deadline, best_c, best_h, max_rounds=4)

    # Phase 2: Intensive search at best dimension
    n_best = len(best_h)
    remaining = deadline - time.time()
    if remaining > 15 and best_c > 0.3808:
        n_restarts = int(min(30, remaining / 8.0))
        for r in range(n_restarts):
            if time.time() > deadline or best_c <= 0.3808:
                break
            strategy = r % 4
            if strategy == 0:
                h_s = gen_binary_random(n_best, rng)
            elif strategy == 1:
                h_s = project_feasible(best_h + 0.04*rng.randn(n_best), n_best)
            elif strategy == 2:
                h_s = gen_smooth_random(n_best, rng)
            else:
                # Flip half the bits of best_h (if near-binary)
                h_s = best_h.copy()
                flip_idx = rng.choice(n_best, size=n_best//4, replace=False)
                h_s[flip_idx] = 1.0 - h_s[flip_idx]
                h_s = project_feasible(h_s, n_best)
            cs = evaluate_c5(h_s)
            if cs < best_c:
                best_c = cs
                best_h = h_s.copy()
            # Short descent
            beta = 100.0 + 50.0 * (r % 3)
            h_s, best_c, best_h = run_gradient_descent(
                h_s, n_best, beta, deadline, best_c, best_h, max_iters=80)
            if time.time() < deadline - 5:
                h_s, best_c, best_h = run_pair_descent(
                    h_s, n_best, deadline, best_c, best_h, max_rounds=3)

    # Phase 3: Final polish at best dimension
    if time.time() < deadline - 5 and best_c > 0.3808:
        n_f = len(best_h)
        h_f = best_h.copy()
        for beta in [200.0, 500.0, 800.0]:
            if time.time() > deadline:
                break
            h_f, best_c, best_h = run_gradient_descent(
                h_f, n_f, beta, deadline, best_c, best_h, max_iters=150)
        h_f, best_c, best_h = run_pair_descent(
            h_f, n_f, deadline, best_c, best_h, max_rounds=6)

    # Also try a few higher dimensions if time permits
    if time.time() < deadline - 30 and best_c > 0.3808:
        for n_try in [n_best + 4, n_best + 8]:
            if n_try > 70 or time.time() > deadline - 10:
                break
            h_try = interpolate_resize(best_h, n_try)
            for beta in [100.0, 300.0]:
                if time.time() > deadline:
                    break
                h_try, best_c, best_h = run_gradient_descent(
                    h_try, n_try, beta, deadline, best_c, best_h, max_iters=60)
            if time.time() < deadline - 5:
                h_try, best_c, best_h = run_pair_descent(
                    h_try, n_try, deadline, best_c, best_h, max_rounds=3)

    # Final: ensure feasibility
    best_h = project_feasible(best_h, len(best_h))
    final_c = evaluate_c5(best_h)
    n_points = len(best_h)
    return (best_h, final_c, n_points)
