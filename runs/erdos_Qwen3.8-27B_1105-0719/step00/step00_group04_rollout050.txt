import numpy as np
from scipy.optimize import minimize
import time

def exact_score(h):
    """Compute the exact C5 score using the official evaluator."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr * dx))

def repair_feasible(vec, target_tol=1e-12):
    """Repair a vector to be feasible: 0<=h<=1, sum=h_target."""
    h = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0)
    n = len(h)
    target = n / 2.0
    # Handle NaN
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < target_tol:
        return h
    if diff > 0:
        # Need to add mass
        # Add to elements closest to 0 (most capacity)
        capacities = 1.0 - h
        order = np.argsort(h)  # lowest first = most capacity
        rem = diff
        for i in order:
            if rem <= 1e-15:
                break
            add = min(rem, capacities[i])
            h[i] += add
            rem -= add
    else:
        # Need to remove mass
        order = np.argsort(-h)  # highest first = most to remove
        rem = -diff
        for i in order:
            if rem <= 1e-15:
                break
            remv = min(rem, h[i])
            h[i] -= remv
            rem -= remv
    # Final check
    if abs(h.sum() - target) > 1e-10:
        # Force exact by adjusting one element
        diff2 = target - h.sum()
        # Find an element we can adjust
        for i in range(n):
            if diff2 > 0 and h[i] < 1.0:
                h[i] += diff2
                break
            elif diff2 < 0 and h[i] > 0.0:
                h[i] += diff2
                break
    return h

def compute_profile_and_gradients(h):
    """Compute all lag values and their gradients.
    Returns: values array (2n-1,), gradients matrix (2n-1, n)
    """
    n = len(h)
    dx = 2.0 / n
    n_lags = 2 * n - 1
    values = np.zeros(n_lags)
    grads = np.zeros((n_lags, n))
    
    for delta in range(0, n):
        # Positive lag delta
        if delta == 0:
            vals = dx * np.sum(h * (1.0 - h))
            values[n - 1] = vals
            grads[n - 1, :] = dx * (1.0 - 2.0 * h)
        else:
            # Lag +delta: F = dx * sum_{i=0}^{n-1-delta} h[i]*(1-h[i+delta])
            idx = np.arange(n - delta)
            vals = dx * np.sum(h[idx] * (1.0 - h[idx + delta]))
            # Map to correlate index: n-1+delta
            values[n - 1 + delta] = vals
            grads[n - 1 + delta, idx] += dx * (1.0 - h[idx + delta])
            grads[n - 1 + delta, idx + delta] += -dx * h[idx]
            
            # Lag -delta: F = dx * sum_{i=delta}^{n-1} h[i]*(1-h[i-delta])
            idx2 = np.arange(delta, n)
            vals2 = dx * np.sum(h[idx2] * (1.0 - h[idx2 - delta]))
            values[n - 1 - delta] = vals2
            grads[n - 1 - delta, idx2] += dx * (1.0 - h[idx2 - delta])
            grads[n - 1 - delta, idx2 - delta] += -dx * h[idx2]
    
    return values, grads

def interpolate_to_dimension(vec, new_len):
    """Interpolate a step function from one resolution to another."""
    n = len(vec)
    x_src = (np.arange(n) + 0.5) * 2.0 / n
    x_tgt = (np.arange(new_len) + 0.5) * 2.0 / new_len
    # Extend with boundary values
    xs_ext = np.concatenate([[0.0], x_src, [2.0]])
    ys_ext = np.concatenate([[vec[0]], vec, [vec[-1]]])
    h_new = np.interp(x_tgt, xs_ext, ys_ext)
    return h_new

class EpigraphProblem:
    def __init__(self, n, beta=0.0):
        self.n = n
        self.dx = 2.0 / n
        self.n_lags = 2 * n - 1
        self.beta = beta
    
    def objective(self, x):
        t = x[self.n]
        h = x[:self.n]
        return t + self.beta * np.sum(h * (1.0 - h))
    
    def jacobian(self, x):
        h = x[:self.n]
        g = np.zeros(self.n + 1)
        if self.beta > 0:
            g[:self.n] = self.beta * (1.0 - 2.0 * h)
        g[self.n] = 1.0
        return g
    
    def eq_constraint(self, x):
        return np.sum(x[:self.n]) - self.n / 2.0
    
    def eq_gradient(self, x):
        g = np.zeros(self.n + 1)
        g[:self.n] = 1.0
        return g
    
    def ineq_constraint(self, x):
        h = x[:self.n]
        t = x[self.n]
        values, grads = compute_profile_and_gradients(h)
        return t - values
    
    def ineq_gradient(self, x):
        h = x[:self.n]
        values, grads = compute_profile_and_gradients(h)
        g = np.zeros((self.n_lags, self.n + 1))
        g[:, self.n] = 1.0
        g[:, :self.n] = -grads
        return g


def solve_epigraph(h_start, beta=0.0, maxiter=100):
    """Solve the epigraph NLP from a starting point."""
    n = len(h_start)
    h_start = repair_feasible(h_start)
    c_start = exact_score(h_start)
    
    t_start = c_start + 0.1
    x0 = np.concatenate([h_start, [t_start]])
    
    problem = EpigraphProblem(n, beta)
    
    bounds = [(0.0, 1.0)] * n + [(0.0, t_start + 1.0)]
    
    constraints = [
        {'type': 'eq', 'fun': problem.eq_constraint, 'jac': problem.eq_gradient},
        {'type': 'ineq', 'fun': problem.ineq_constraint, 'jac': problem.ineq_gradient}
    ]
    
    try:
        result = minimize(
            problem.objective, x0, jac=problem.jacobian,
            method='SLSQP', bounds=bounds, constraints=constraints,
            options={'maxiter': maxiter, 'ftol': 1e-12}
        )
    except Exception:
        return None, None
    
    if result.x is not None:
        h_sol = repair_feasible(result.x[:n])
        score = exact_score(h_sol)
        return h_sol, score
    return None, None


def active_gradient_descent(h, max_steps=50):
    """Local gradient descent on active constraints."""
    n = len(h)
    h = repair_feasible(h)
    best = h.copy()
    best_score = exact_score(best)
    
    for step in range(max_steps):
        values, grads = compute_profile_and_gradients(best)
        c = np.max(values)
        
        # Find active constraints (top few)
        n_active = min(20, 2 * n - 1)
        active_idx = np.argsort(values)[::-1][:n_active]
        
        # Compute weighted average gradient (weights based on how close to max)
        gaps = np.maximum(c - values[active_idx], 0)
        weights = np.exp(-gaps / (0.01 * c + 1e-10))
        weights /= np.sum(weights) + 1e-15
        
        # Direction: negative average gradient of active constraints
        direction = -np.sum(weights[:, None] * grads[active_idx, :], axis=0)
        
        # Remove mean to preserve sum constraint
        direction -= np.mean(direction)
        
        # Line search
        step_sizes = np.array([0.02, 0.01, 0.005, 0.002, 0.001, 0.0005])
        improved = False
        for alpha in step_sizes:

            h_trial = repair_feasible(h_trial)

            if score < best_score - 1e-12:
                best = h_trial.copy()
                best_score = score
                improved = True
                break
        
        if not improved:

    return best, best_score

            best_h = h_sol
            best_score = score
            best_n = len(h_sol)

    # Phase 2: SLSQP with limited active set
    if tl() > 60:
        values, _ = compute_profile_and_gradients(best_h)
        c_val = np.max(values)
        # Top 40 active
        top_idx = np.argsort(values)[::-1][:40]
        top_idx = np.sort(top_idx)
        mask = np.zeros(len(values), dtype=bool)
        mask[top_idx] = True
        h_sol, score = solve_epigraph(best_h, beta=0.0, maxiter=200, active_mask=top_idx)
        if h_sol is not None and score < best_score - 1e-13:
            best_h = h_sol
            best_score = score
            best_n = len(h_sol)

    # Phase 3: Regularized SLSQP
    if tl() > 50:
        for beta in [1e-5, 3e-5, 1e-4]:
            if tl() < 30:
                break
            h_sol, score = solve_epigraph(best_h, beta=beta, maxiter=150)
            if h_sol is not None and score < best_score - 1e-13:
                best_h = h_sol
                best_score = score
                best_n = len(h_sol)

    # Phase 4: Active descent
    if tl() > 20:
        h_ref, score_ref = active_descent(best_h, max_steps=60)
        if score_ref < best_score - 1e-13:
            best_h = h_ref
            best_score = score_ref
            best_n = len(h_ref)

    # Phase 5: Dimension transfers
    if tl() > 60:
        n_cur = len(best_h)
        dims_to_try = []
        for dn in [8, 16, 32]:
            nd = n_cur + dn
            if nd <= 150:
                dims_to_try.append(nd)
        if n_cur * 2 <= 150:
            dims_to_try.append(n_cur * 2)
        for nd in dims_to_try:
            if tl() < 30:
                break
            h_interp = repair_feasible(interpolate_to_dimension(best_h, nd))
            s_interp = exact_score(h_interp)
            if s_interp < best_score + 0.02:
                h_sol, score = solve_epigraph(h_interp, beta=0.0, maxiter=100)
                if h_sol is not None and score < best_score - 1e-13:
                    best_h = h_sol
                    best_score = score
                    best_n = len(h_sol)

    # Phase 6: Perturbed multi-start
    if tl() > 40:
        for trial in range(8):
            if tl() < 25:
                break
            eps = rng.uniform(0.002, 0.012)
            z = rng.uniform(-1.0, 1.0, size=len(best_h))
            z -= z.mean()
            h_pert = repair_feasible(best_h + eps * z)
            h_sol, score = solve_epigraph(h_pert, beta=0.0, maxiter=80)
            if h_sol is not None and score < best_score - 1e-13:
                best_h = h_sol
                best_score = score
                best_n = len(h_sol)

    # Phase 7: Final active descent
    if tl() > 15:
        h_ref, score_ref = active_descent(best_h, max_steps=40)
        if score_ref < best_score - 1e-13:
            best_h = h_ref
            best_score = score_ref
            best_n = len(h_ref)

    # Phase 8: One more SLSQP pass on best
    if tl() > 20:
        h_sol, score = solve_epigraph(best_h, beta=0.0, maxiter=200)
        if h_sol is not None and score < best_score - 1e-13:
            best_h = h_sol
            best_score = score
            best_n = len(h_sol)

    # Final validation
    best_h = repair_feasible(best_h)
    best_score = exact_score(best_h)
    return best_h, best_score, len(best_h)
