def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_c5_all(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def compute_c5_grad(h, dx):
    """Compute gradient of max C5 w.r.t. h using subgradient of active constraints."""
    c5_all = compute_c5_all(h, dx)
    max_val = np.max(c5_all)
    # Find active constraints (within tolerance)
    active = c5_all > max_val - 1e-8
    # Compute gradient of each active constraint
    n = len(h)
    grad = np.zeros(n)
    for k_idx in np.where(active)[0]:
        lag = k_idx - (n - 1)
        if lag >= 0:
            grad[lag:] += (1.0 - h[:n-lag]) * dx
            grad[:n-lag] += -h[lag:] * dx
        else:
            lag_neg = -lag
            grad[:n-lag_neg] += (1.0 - h[lag_neg:]) * dx
            grad[lag_neg:] += -h[:n-lag_neg] * dx
    return max_val, grad

    Returns a matrix of shape (2n-1, n)."""

        else:
