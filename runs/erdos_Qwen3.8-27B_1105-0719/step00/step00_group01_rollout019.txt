    return np.max(F), F

def softmax_value_and_weights(h, beta):
    """Compute Phi_beta and the softmax weights."""
    N = len(h)
    dx = 2.0 / N
    F = np.correlate(h, 1.0 - h, mode="full") * dx
    Fmax = np.max(F)
    shifted = beta * (F - Fmax)
    # Clip for numerical stability
    shifted = np.clip(shifted, -500, 0)
    exps = np.exp(shifted)
    log_sum = np.log(np.sum(exps))
    Phi = Fmax + log_sum / beta
    w = exps / np.sum(exps)
    return Phi, F, Fmax, w

def softmax_grad(h, beta, F=None, w=None):

    """Make equality-preserving transfer moves."""
    N = len(h)

import numpy as np

def compute_F(h):
    """Compute all correlation values F[k] * dx."""
    N = len(h)
    dx = 2.0 / N
    return np.correlate(h, 1.0 - h, mode="full") * dx

def F_gradient_single(h, k):
    """Compute gradient of F[k] w.r.t. h."""
    N = len(h)
    dx = 2.0 / N
    d = k - (N - 1)
    grad = np.zeros(N)
    
    # F[k] = dx * sum_n h[n] * (1 - h[n-d]) for valid n, n-d
    # dF/dh[i]:
    # Term 1: h[i] appears as h[n] when n=i: dx * (1 - h[i-d]) if 0 <= i-d < N
    # Term 2: h[i] appears as h[n-d] when n-d=i, i.e., n=i+d: -dx * h[i+d] if 0 <= i+d < N
    
    # Term 1: i-d valid means 0 <= i-d < N, i.e., d <= i < N+d
    # And i in [0, N-1]
    i_lo1 = max(0, d)
    i_hi1 = min(N - 1, N - 1 + d)
    if i_lo1 <= i_hi1:
        idx = np.arange(i_lo1, i_hi1 + 1)
        grad[idx] += dx * (1.0 - h[idx - d])
    
    # Term 2: i+d valid means 0 <= i+d < N, i.e., -d <= i < N-d
    # And i in [0, N-1]
    i_lo2 = max(0, -d)
    i_hi2 = min(N - 1, N - 1 - d)
    if i_lo2 <= i_hi2:
        idx = np.arange(i_lo2, i_hi2 + 1)
        grad[idx] += -dx * h[idx + d]
    
    return grad

def max_subgradient(h, F, tol=1e-8):

def active_set_direction(h, F, active_tol=1e-6, bounds=0.01):
    """Find direction using active-set LP."""
    N = len(h)
    M = np.max(F)
    active = np.where(F >= M - active_tol)[0]
    
    if len(active) == 0:
        return None
    
    # Build LP: minimize t
    # Variables: d[0..N-1], t
    # Constraints:
    #   grad(F[k])^T d - t ≤ 0 for k in active  (i.e., grad(F[k])^T d ≤ t)
    #   sum(d) = 0
    #   -bounds ≤ d[i] ≤ bounds
    #   t ≥ -M (or some lower bound to prevent unbounded)
    
    n_vars = N + 1  # d[N] + t[1]
    
    # Objective: minimize t = variable N
    c = np.zeros(n_vars)
    c[N] = 1.0
    
    # Inequality constraints: A_ub @ x <= b_ub
    A_ub = np.zeros((len(active), n_vars))
    for i, k in enumerate(active):
        gk = np.zeros(N)  # gradient of F[k]
        d_shift = k - (N - 1)
        dx = 2.0 / N
        # Compute grad(F[k])
        i_lo = max(0, d_shift)
        i_hi = min(N - 1, N - 1 + d_shift)
        if i_lo <= i_hi:
            idx = np.arange(i_lo, i_hi + 1)
            gk[idx] += dx * (1.0 - h[idx - d_shift])
        i_lo = max(0, -d_shift)
        i_hi = min(N - 1, N - 1 - d_shift)
        if i_lo <= i_hi:
            idx = np.arange(i_lo, i_hi + 1)
            gk[idx] -= dx * h[idx + d_shift]
        
        A_ub[i, :N] = gk
        A_ub[i, N] = -1.0  # gk^T d - t <= 0
    
    b_ub = np.zeros(len(active))
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n_vars))
    A_eq[0, :N] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds: -bounds <= d[i] <= bounds, -M-1 <= t
    lb = np.full(n_vars, -np.inf)

The improvement needed is 0.00005, which is very small. This suggests that:
1. The parent is very close to a local optimum
2. We need a very precise direction
3. Random search alone won't work - we need the right direction

The key insight: at a local optimum of the max objective, there exists a convex combination of active gradients that equals zero (KKT condition). If we're not at the optimum, there's a direction that reduces the max.

Let me think about the LP approach more carefully. At the current point h*, with active set A:
- If we can find d such that grad(F[k])^T d < 0 for ALL k in A, then moving in direction d will (to first order) reduce all active lags, hence reduce the max.
- The question is: does such a d exist with sum(d) = 0?

If the active gradients span a space that doesn't contain the all-ones vector (projected), then such a d exists. If they do contain it, then we're at a local optimum (of the linearized problem).

The LP to find such a direction:
- minimize: max_k (grad(F[k])^T d) for k in A
- subject to: sum(d) = 0, ||d||_2 ≤ 1 (or ||d||_∞ ≤ 1)

If the optimal value of this LP is negative, we have a descent direction.

def find_descent_direction(h, F, active_tol=1e-8, method='lp'):
    """Find a direction that reduces the max (active lags)."""
    N = len(h)
    dx = 2.0 / N
    M = F.max()
    active = np.where(F >= M - active_tol)[0]
    
    if len(active) == 0:
        return None
    
    # Compute gradients for active lags
    G = np.zeros((len(active), N))
    for i, k in enumerate(active):
        G[i] = grad_single(h, k, N, dx)
    
    if method == 'lp':
        # LP: minimize t subject to G[i] @ d <= t, sum(d) = 0, -1 <= d_i <= 1
        n = N + 1
        c = np.zeros(n); c[N] = 1.0
        A_ub = np.zeros((len(active), n))

        A_ub[:, N] = -1.0
        b_ub = np.zeros(len(active))

                     bounds=bounds, method='highs')

            return res.x[:N], res.fun
        return None, res.fun if res.success else 0
    
    else:  # subgradient

import numpy as np
from scipy.optimize import linprog
import time

def compute_F(h):
    N = len(h)
    dx = 2.0 / N
    return np.correlate(h, 1.0 - h, mode="full") * dx

def grad_single(h, k, N, dx):
    g = np.zeros(N)
    d = k - (N - 1)
    i_lo = max(0, d)
    i_hi = min(N - 1, N - 1 + d)
    if i_lo <= i_hi:
        idx = np.arange(i_lo, i_hi + 1)
        g[idx] += dx * (1.0 - h[idx - d])
    i_lo = max(0, -d)
    i_hi = min(N - 1, N - 1 - d)
    if i_lo <= i_hi:
        idx = np.arange(i_lo, i_hi + 1)
        g[idx] -= dx * h[idx + d]
    return g

def project_z(z, target):
    z = np.clip(np.nan_to_num(z), 0.0, 1.0)
    N = len(z)
    if abs(z.sum() - target) < 1e-14:
        return z.copy()
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    r = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - r.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            m = r < 1.0 - 1e-12
        else:
            m = r > 1e-12
        idx = np.where(m)[0]
        if len(idx) > 0:
            per = diff / len(idx)
            for i in idx:
                if diff > 0:
                    a = min(per, 1.0 - r[i])
                    r[i] += a
                    diff -= a
                else:
                    a = min(-per, r[i])
                    r[i] -= a
                    diff += a
                if abs(diff) < 1e-14:
                    break
    return r

def active_direction_lp(h, F, active_tol):
    N = len(h)
    dx = 2.0 / N
    M = F.max()
    active = np.where(F >= M - active_tol)[0]
    if len(active) == 0:
        active = np.array([np.argmax(F)])
    G = np.zeros((len(active), N))

        G[i] = grad_single(h, k, N, dx)
    n = N + 1
    c = np.zeros(n)
    c[N] = 1.0
    A_ub = np.zeros((len(active), n))
    A_ub[:, :N] = G
    A_ub[:, N] = -1.0
    b_ub = np.zeros(len(active))
    A_eq = np.zeros((1, n))
    A_eq[0, :N] = 1.0
    b_eq = np.zeros(1)
    bounds = [(-1.0, 1.0)] * N + [(-10.0, 10.0)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:N].copy(), float(res.fun)
    except Exception:
        pass
    g = G.mean(axis=0)
    g -= g.mean()
    return g, None
