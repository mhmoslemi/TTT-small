def compute_c5(h):
    """Compute the exact C5 value for a given h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def raw_shifts(h):
    """Compute all raw correlation values (before multiplying by dx)."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr

def h_from_a(a):
    """Construct h from symmetric reduced coordinates a."""
    m = len(a)
    n = 2 * m
    h = np.empty(n)
    h[:m] = a
    h[m:] = 1.0 - a[::-1]
    return h

def h_from_a_odd(a):
    """Construct h from symmetric reduced coordinates a (odd n)."""
    m = len(a)
    n = 2 * m + 1
    h = np.empty(n)
    h[:m] = a
    h[m] = 0.5
    h[m+1:] = 1.0 - a[::-1]
    return h

    n = len(h)
    m = n // 2
    return h[:m].copy()

    n = len(h)

    return h[:m].copy()

    """Apply T symmetry: h_sym = (h + (1-h[::-1]))/2"""
    T_h = 1.0 - h[::-1]
    return 0.5 * (h + T_h)

def project_box_mass(h, target):
    """Project h onto {0<=h<=1, sum(h)=target} via bisection on Lagrange multiplier."""
    x = np.clip(h.copy(), 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        val = np.clip(x - mid, 0.0, 1.0)
        s = np.sum(val)
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - 0.5*(lo+hi), 0.0, 1.0)

def grad_R_full(h, k):
    """Gradient of R_k with respect to h, where k is the shift (can be negative)."""
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        # R_k = sum_{i=0}^{n-1-k} h[i+k]*(1-h[i])
        for i in range(n - k):
            g[i + k] += (1.0 - h[i])
            g[i] += -h[i + k]
    else:
        # R_k with k<0: let q=-k
        q = -k
        # R_{-q} = sum_{i=q}^{n-1} h[i-q]*(1-h[i])
        for i in range(q, n):
            g[i - q] += (1.0 - h[i])
            g[i] += -h[i - q]
    return g

def grad_R_a(h, a, k, n_odd):

    # Reduce to a-coordinates
    ga = np.zeros(m)
    for j in range(m):
        ga[j] = g_full[j] - g_full[n - 1 - j]
    return ga

def compute_c5(h):

    dx = 2.0 / n

    return float(np.max(corr) * dx)

def shift_value(h, k):
    """Compute R_k = sum h[i+k]*(1-h[i]) for valid i."""
    n = len(h)
    if k >= 0:
        return np.dot(h[k:], 1.0 - h[:n-k])
    else:
        q = -k
        return np.dot(h[:n-q], 1.0 - h[q:])

    """Gradient of R_k w.r.t. h."""
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        g[k:] += (1.0 - h[:n-k])
        g[:n-k] += (-h[k:])
    else:
        q = -k
        g[:n-q] += (1.0 - h[q:])
        g[q:] += (-h[:n-q])

```python
import numpy as np
import time
import math


def compute_c5_full(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def all_R(h):
    return np.correlate(h, 1.0 - h, mode="full")


def grad_shift(h, k):
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        g[k:] += (1.0 - h[:n - k])
        g[:n - k] += (-h[k:])
    else:
        q = -k
        g[:n - q] += (1.0 - h[q:])
        g[q:] += (-h[:n - q])
    return g


def reduce_grad(g_full, m):
    return g_full[:m] - g_full[::-1][:m]


def h_from_a(a, n_odd):
    m = len(a)
    if n_odd:
        n = 2 * m + 1
        h = np.empty(n)
        h[:m] = a
        h[m] = 0.5
        h[m + 1:] = 1.0 - a[::-1]
    else:
        n = 2 * m
        h = np.empty(n)
        h[:m] = a
        h[m:] = 1.0 - a[::-1]
    return h


def symmetrize(h):
    return 0.5 * (h + (1.0 - h[::-1]))


def lse_weights(R, beta):
    M = np.max(R)
    e = np.exp(beta * (R - M))
    s = np.sum(e)
    if s < 1e-300:
        n_R = len(R)
        w = np.zeros(n_R)
        w[np.argmax(R)] = 1.0
        return w
    return e / s


def lse_gradient_a(a, n_odd, beta):
    m = len(a)
    n = 2 * m + (1 if n_odd else 0)
    h = h_from_a(a, n_odd)
    R = all_R(h)
    w = lse_weights(R, beta)
    ga = np.zeros(m)
    for t in range(len(R)):
        wt = w[t]
        if wt < 1e-14:
            continue
        k = t - (n - 1)
        g_full = grad_shift(h, k)
        ga += wt * reduce_grad(g_full, m)
    return -ga, R


def worst_shift_gradient_a(a, n_odd):
    m = len(a)
    n = 2 * m + (1 if n_odd else 0)
    h = h_from_a(a, n_odd)
    R = all_R(h)
    t_max = int(np.argmax(R))
    k = t_max - (n - 1)
    g_full = grad_shift(h, k)
    ga = reduce_grad(g_full, m)
    return -ga, k


def coordinate_search_impl(a_cur, n_odd, c5_cur, rng, max_moves):
    m = len(a_cur)
    a_work = a_cur.copy()
    c5_work = c5_cur
    deltas = [0.1, -0.1, 0.05, -0.05, 0.02, -0.02, 0.01, -0.01, 0.005, -0.005]
    for _ in range(max_moves):
        improved = False
        j = rng.randint(m)
        for delta in deltas:
            a_trial = a_work.copy()
            new_val = a_work[j] + delta
            if new_val < 0.0 or new_val > 1.0:
                continue
            a_trial[j] = new_val
            h_trial = h_from_a(a_trial, n_odd)
            c5_trial = compute_c5_full(h_trial)
            if c5_trial < c5_work - 1e-12:
                a_work = a_trial
                c5_work = c5_trial
                improved = True
                break
        if not improved:
            continue
    return a_work, c5_work


def make_logistic_seeds(m):
    x = np.linspace(0, 1, m)
    seeds = []
    for steep in [2.0, 4.0, 8.0, -2.0, -4.0, -8.0]:
        for center in [0.2, 0.4, 0.5, 0.6, 0.8]:
            a = 1.0 / (1.0 + np.exp(steep * (x - center)))
            seeds.append(np.clip(a, 0, 1))
    return seeds


def make_sinusoidal_seeds(m):
    x = np.linspace(0, 1, m)
    seeds = []
    for freq in [1, 2, 3, 4, 6]:
        for amp in [0.3, 0.5, 0.7]:
            for phase in [0.0, 0.5, 1.0, 1.5]:
                a = 0.5 + amp * np.sin(2.0 * math.pi * freq * x + phase)
                seeds.append(np.clip(a, 0, 1))
    return seeds


def try_cvxqcqp(a_init, n_odd, timeout_s):
    try:
        import cvxpy as cp
    except ImportError:
        return None
    m = len(a_init)
    n = 2 * m + (1 if n_odd else 0)
    try:
        a_var = cp.Variable(m)
        if n_odd:
            h_expr = cp.hstack([a_var, cp.Constant(0.5), 1 - a_var[::-1]])
        else:
            h_expr = cp.hstack([a_var, 1 - a_var[::-1]])
        t_var = cp.Variable()
        constraints = [a_var >= 0, a_var <= 1]
        for k in range(-(n - 1), n):
            if k >= 0:
                U_k = 0.5 * (cp.sum_squares(h_expr[k:]) + cp.sum_squares(1 - h_expr[:n - k]))
            else:
                q = -k
                U_k = 0.5 * (cp.sum_squares(h_expr[:n - q]) + cp.sum_squares(1 - h_expr[q:]))
            constraints.append(U_k <= t_var)
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        solvers = ['CLARABEL', 'SCS', 'ECOS', 'OSQP']
        for solver in solvers:
            try:
                t_start = time.time()
                prob.solve(solver=solver)
                if time.time() - t_start > timeout_s:
                    continue
                if prob.status == 'optimal' and a_var.value is not None:
                    a_sol = np.asarray(a_var.value).flatten().astype(float)
                    a_sol = np.clip(a_sol, 0.0, 1.0)
                    h_sol = h_from_a(a_sol, n_odd)
                    c5_sol = compute_c5_full(h_sol)
                    if np.isfinite(c5_sol):
                        return a_sol, c5_sol
            except Exception:
                continue
    except Exception:
        pass
    return None
