def lse_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    h1m = 1.0 - h
    corr_raw = np.correlate(h, h1m, mode="full")
    vals = corr_raw * dx
    m = np.max(vals)
    w = np.exp((vals - m) / tau)
    w /= w.sum()
    grad = np.zeros(n)
    for k in range(len(corr_raw)):
        if w[k] < 1e-14:
            continue
        p = k - (n - 1)
        # d corr[k]/d h[m]: term where h[m] is first factor: (1-h[m+p]) if m+p valid
        lo1 = max(0, -p)
        hi1 = min(n, n - p)
        if lo1 < hi1:
            grad[lo1:hi1] += w[k] * (1.0 - h[lo1 + p:hi1 + p])
        # term where (1-h[m]) is second: -h[m-p] if m-p valid  
        lo2 = max(0, p)
        hi2 = min(n, n + p)
        if lo2 < hi2:
            grad[lo2:hi2] -= w[k] * h[lo2 - p:hi2 - p]
    return grad * dx

        p = k - (n - 1)
        lo1 = max(0, -p)
        hi1 = min(n, n - p)
        if lo1 < hi1:
            grad[lo1:hi1] += wk * (1.0 - h[lo1 + p:hi1 + p])
        lo2 = max(0, p)
        hi2 = min(n, n + p)
        if lo2 < hi2:
            grad[lo2:hi2] -= wk * h[lo2 - p:hi2 - p]
    return grad * dx

def pair_descent(h, seed, deadline):
    rng = np.random.default_rng(seed)
    cur = h.copy()
    n = len(cur)
    cur_c = eval_c5(cur)
    no_improve = 0
    fracs = (0.5, 0.25, 0.1, 0.4, 0.15, 0.3, 0.05, 0.7)

    while time.perf_counter() < deadline and no_improve < 200:
        improved = False
        idx = np.arange(n)
        pair_list = [(i, j) for i in range(n) for j in range(i + 1, n)]
        rng.shuffle(pair_list)

        for (i, j) in pair_list:
            if time.perf_counter() >= deadline:
                break
            cap = min(cur[i], 1.0 - cur[j])
            if cap > 1e-10:
                ov = (cur[i], cur[j])
                for f in fracs:
                    d = cap * f
                    cur[i], cur[j] = ov[0] - d, ov[1] + d
                    nc = eval_c5(cur)
                    if nc < cur_c - 1e-13:
                        cur_c = nc
                        improved = True
                        ov = (cur[i], cur[j])
                    else:
                        cur[i], cur[j] = ov
                        break
            cap = min(cur[j], 1.0 - cur[i])
            if cap > 1e-10:
                ov = (cur[i], cur[j])
                for f in fracs:
                    d = cap * f
                    cur[i], cur[j] = ov[0] + d, ov[1] - d
                    nc = eval_c5(cur)
                    if nc < cur_c - 1e-13:
                        cur_c = nc
                        improved = True
                        ov = (cur[i], cur[j])
                    else:
                        cur[i], cur[j] = ov
                        break
        if improved:
            no_improve = 0
        else:
            no_improve += 1

    return cur, cur_c

def smooth_phase(h, seed, deadline):
    rng = np.random.default_rng(seed)
    cur = h.copy()
    n = len(cur)
    target_s = n / 2.0
    cur_c = eval_c5(cur)
    tau = 0.02
    alpha = 0.04
    no_imp = 0

    while time.perf_counter() < deadline and no_imp < 80:
        if tau < 0.0015:
            break
        g = lse_grad(cur, tau)
        g -= np.mean(g)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn
        best_a = 0.0
        best_c = cur_c
        for t in range(12):
            a = alpha * (0.7 ** t)
            if a < 1e-8:
                break
            cand = project_box_sum(cur - a * g, target_s)
            cc = eval_c5(cand)
            if cc < best_c:
                best_c = cc
                best_a = a
            if cc < cur_c - 1e-12:
                break
        if best_a > 0 and best_c < cur_c - 1e-13:
            cur = project_box_sum(cur - best_a * g, target_s)
            cur_c = best_c
            no_imp = 0
            alpha = min(alpha * 1.15, 0.1)
        else:
            no_imp += 1
            tau *= 0.82
            alpha *= 0.88

    return cur, cur_c

def sa_phase(h, seed, deadline):
    rng = np.random.default_rng(seed)
    cur = h.copy()
    n = len(cur)
    target_s = n / 2.0
    cur_c = eval_c5(cur)
    temp = 0.004
    no_imp = 0
    total = 0
    accepted = 0

    while time.perf_counter() < deadline and no_imp < 400:
        total += 1
        mode = rng.random()
        if mode < 0.75:
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue
            cap = min(cur[i], 1.0 - cur[j])
            if cap < 1e-10:
                continue
            d = cap * rng.uniform(0.05, 0.45)
            old = (cur[i], cur[j])
            cur[i], cur[j] = old[0] - d, old[1] + d
        else:
            noise = rng.standard_normal(n)
            noise -= noise.mean()
            cur = project_box_sum(cur + 0.008 * noise, target_s)
            old = None

        nc = eval_c5(cur)
        if nc <= cur_c:
            cur_c = nc
            accepted += 1
            no_imp = 0
        elif rng.random() < np.exp(-(nc - cur_c) / max(temp, 1e-8)):
            cur_c = nc
            accepted += 1
            no_imp += 1
        else:
            if old is not None:
                cur[i], cur[j] = old
            no_imp += 1

        if total % 150 == 0:
            temp *= 0.92
            if accepted / max(total, 1) < 0.08:
                temp = min(temp * 2.0, 0.02)
            total, accepted = 0, 0

    return cur, cur_c

def interpolate_to(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False) + 1.0 / n_old
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False) + 1.0 / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return project_box_sum(h_new, n_new / 2.0)

def top_lag_refine(h, seed, deadline):
    """Focus on reducing the top few lags specifically."""
    rng = np.random.default_rng(seed)
    cur = h.copy()
    n = len(cur)
    target_s = n / 2.0
    cur_c = eval_c5(cur)
    no_imp = 0

    while time.perf_counter() < deadline and no_imp < 100:
        dx = 2.0 / n
        corr = np.correlate(cur, 1.0 - cur, mode="full") * dx
        thresh = cur_c - 0.002
        active = np.where(corr >= thresh)[0]
        if len(active) == 0:
            break

        g = np.zeros(n)
        for k in active:
            p = k - (n - 1)
            lo1 = max(0, -p)
            hi1 = min(n, n - p)
            if lo1 < hi1:
                g[lo1:hi1] += (1.0 - cur[lo1 + p:hi1 + p])
            lo2 = max(0, p)
            hi2 = min(n, n + p)
            if lo2 < hi2:
                g[lo2:hi2] -= cur[lo2 - p:hi2 - p]

        g -= np.mean(g)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn

        best_a = 0.0
        best_c = cur_c
        for t in range(10):
            a = 0.02 * (0.7 ** t)
            if a < 1e-8:
                break
            cand = project_box_sum(cur - a * g, target_s)
            cc = eval_c5(cand)
            if cc < best_c:
                best_c = cc
                best_a = a
            if cc < cur_c - 1e-12:
                break

        if best_a > 0 and best_c < cur_c - 1e-13:
            cur = project_box_sum(cur - best_a * g, target_s)
            cur_c = best_c
            no_imp = 0
        else:
            no_imp += 1

    return cur, cur_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.perf_counter()

    try:
        parent = np.asarray(initial_h_values, dtype=np.float64).ravel().copy()
    except (NameError, AttributeError):
        parent = None

    if parent is None or len(parent) < 4:
        n = 47
        parent = project_box_sum(np.full(n, 0.5), n / 2.0)

    n = len(parent)
    parent = project_box_sum(parent, n / 2.0)
    best_h = parent.copy()
    best_c5 = eval_c5(best_h)

    def rem():
        return max(0.0, budget_s - (time.perf_counter() - t0))

    if rem() < 10:
        return best_h, best_c5, n

    # Phase 1: Initial pair descent
    if rem() > 30:
        dl = t0 + min(250.0, rem() * 0.30)
        cur, cc = pair_descent(best_h, seed, dl)
        if cc < best_c5:
            best_c5 = cc
            best_h = cur.copy()

    # Phase 2: Smooth gradient descent
    if rem() > 40:
        dl = t0 + min(150.0, rem() * 0.20)
        cur, cc = smooth_phase(best_h, seed + 1, dl)
        if cc < best_c5:
            best_c5 = cc
            best_h = cur.copy()

    # Phase 3: Top-lag focused refinement
    if rem() > 30:
        dl = t0 + min(100.0, rem() * 0.25)
        cur, cc = top_lag_refine(best_h, seed + 2, dl)
        if cc < best_c5:
            best_c5 = cc
            best_h = cur.copy()

    # Phase 4: More pair descent
    if rem() > 30:
        dl = t0 + min(200.0, rem() * 0.30)
        cur, cc = pair_descent(best_h, seed + 3, dl)
        if cc < best_c5:
            best_c5 = cc
            best_h = cur.copy()

    # Phase 5: SA for exploration
    if rem() > 40:
        dl = t0 + min(100.0, rem() * 0.35)
        cur, cc = sa_phase(best_h, seed + 4, dl)
        if cc < best_c5:
            best_c5 = cc
            best_h = cur.copy()

    # Phase 6: Final pair polish
    if rem() > 20:
        dl = t0 + min(80.0, rem() * 0.40)
        cur, cc = pair_descent(best_h, seed + 5, dl)
        if cc < best_c5:
            best_c5 = cc
            best_h = cur.copy()

    # Phase 7: Try alternative resolutions
    if rem() > 50:
        n_tries = [int(n * 0.75), int(n * 1.3), int(n * 0.55), int(n * 1.6)]
        for nt in n_tries:
            if rem() < 25:
                break
            if nt < 6 or nt > 250 or nt == n:
                continue
            h_try = interpolate_to(best_h, nt)
            c_try = eval_c5(h_try)
            if c_try < best_c5:
                best_c5 = c_try
                best_h = h_try.copy()
            if rem() > 25:
                dl = t0 + min(25.0, rem() * 0.4)
                h_try, c_try = pair_descent(h_try, seed + nt, dl)
                if c_try < best_c5:
                    best_c5 = c_try
                    best_h = h_try.copy()

    # Phase 8: Final polish with best found so far
    if rem() > 15:
        dl = t0 + min(12.0, rem() * 0.5)
        cur, cc = pair_descent(best_h, seed + 100, dl)
        if cc < best_c5:
            best_c5 = cc
            best_h = cur.copy()

    # Final validation and return
    n_final = len(best_h)
    best_h = project_box_sum(best_h, n_final / 2.0)
    final_c5 = eval_c5(best_h)

    return best_h, final_c5, n_final
