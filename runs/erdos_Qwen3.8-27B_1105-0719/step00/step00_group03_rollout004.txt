import numpy as np
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    scores = np.correlate(h, 1.0 - h, mode='full') * dx
    return scores.max(), scores

def project_balanced(y):
    """Project y onto {x: 0<=x[i]<=1, sum(x)=len(y)/2}"""
    n = len(y)
    target = n / 2.0
    # Binary search for lambda such that sum(clip(y+lambda, 0, 1)) = target
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(y + mid, 0.0, 1.0).sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y + mid, 0.0, 1.0)
    # Fix residual
    residual = target - x.sum()
    if abs(residual) > 1e-10:
        if residual > 0:
            # Need to add to some entries
            idx = np.where(x < 1.0)[0]
        else:
            idx = np.where(x > 0.0)[0]
        # Distribute residual
        per = residual / len(idx)
        x[idx] += per
        x = np.clip(x, 0.0, 1.0)
    return x

def compute_gradient(h, weights, n):
    """Compute gradient of weighted sum of correlation terms."""
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        offset = k - (n - 1)
        w = weights[k]
        if abs(w) < 1e-15:
            continue
        if offset >= 0:
            j = np.arange(n - offset)
            if len(j) == 0:
                continue
            grad[j] += dx * w * (1.0 - h[j + offset])
            grad[j + offset] -= dx * w * h[j]
        else:
            d = -offset
            j = np.arange(d, n)
            if len(j) == 0:
                continue
            grad[j] += dx * w * (1.0 - h[j - d])
            grad[j - d] -= dx * w * h[j]
    return grad

def make_weights(scores, mode='soft', beta=5.0):
    """Create weights from scores."""
    max_s = scores.max()
    if max_s < 1e-15:
        return np.ones(len(scores)) / len(scores)
    if mode == 'active':
        tol = 1e-7
        active = scores >= max_s - tol
        w = active.astype(float)
        w /= w.sum()
    elif mode == 'soft':
        normalized = scores / max_s
        w = np.exp(beta * (normalized - 1.0))
        if w.sum() < 1e-15:
            w = np.ones(len(scores)) / len(scores)
        else:
            w /= w.sum()
    else:  # 'all'
        w = np.ones(len(scores)) / len(scores)
    return w
