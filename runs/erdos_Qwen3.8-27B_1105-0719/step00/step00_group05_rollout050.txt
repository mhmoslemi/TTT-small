import time

_G = {}

def exact_C(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def project_feasible(u):
    n = len(u)
    target = n / 2.0
    lo = -1.0 - float(np.max(u))
    hi = 1.0 - float(np.min(u))
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(u + mid, 0.0, 1.0)))
        if s < target:
            lo = mid
        else:
            hi = mid
    result = np.clip(u + (lo + hi) / 2.0, 0.0, 1.0)
    delta = target - float(np.sum(result))
    if abs(delta) > 1e-13:
        if delta > 0:
            idx = np.where(result < 1.0 - 1e-12)[0]
        else:
            idx = np.where(result > 1e-12)[0]
        if len(idx) > 0:
            per = delta / len(idx)
            result[idx] = np.clip(result[idx] + per, 0.0, 1.0)
    return result

def sigmoid(t):
    return 1.0 / (1.0 + np.exp(-np.clip(t, -500, 500)))

    n = len(y)
    target = n / 2.0
    lo = -float(np.max(y)) - 80.0
    hi = -float(np.min(y)) + 80.0

        if s < target:

        else:
            hi = mid

def compute_grad_wrt_h(h, w, dx):
    """Compute G = sum_r w[r] * d(corr[r])/dh * dx
    where w is the softmax weight vector of length 2n-1.
    """
    n = len(h)
    G = np.zeros(n)
    
    for idx in range(2 * n - 1):
        wi = w[idx] * dx
        if wi < 1e-15:
            continue
        l = idx - (n - 1)  # lag
        
        if l == 0:
            G += wi * (1.0 - 2.0 * h)
        elif l > 0:
            # corr = sum_{i=0}^{n-1-l} h[i+l](1-h[i])
            # d/dh[j] = (1-h[j-l]) for j in [l, n-1]  (from h[j] in h[i+l] term)
            #         + (-h[j+l]) for j in [0, n-1-l] (from -h[j+l]*h[j] term)
            j_lo, j_hi = l, n - 1

            G[j_lo2:j_hi2+1] += wi * (-h[l:n])
        else:
            m = -l
            # corr = sum_{i=0}^{n-1-m} h[i](1-h[i+m])
            # d/dh[j] = (1-h[j+m]) for j in [0, n-1-m]  (from h[j] term)

            j_lo2, j_hi2 = m, n - 1
            G[j_lo2:j_hi2+1] += wi * (-h[0:n-m])

    lam = (lo + hi) / 2.0
    return sigmoid(y + lam)


def logit_from_h(h):
    eps = 1e-6
    h_safe = np.clip(h, eps, 1.0 - eps)
    return np.log(h_safe / (1.0 - h_safe))


def compute_grad_wrt_h(h, w, dx):
    n = len(h)
    G = np.zeros(n)
    for idx in range(2 * n - 1):
        wi = w[idx] * dx
        if wi < 1e-15:
            continue
        l = idx - (n - 1)
        if l == 0:
            G += wi * (1.0 - 2.0 * h)
        elif l > 0:
            G[l:n] += wi * (1.0 - h[0:n - l])
            G[0:n - l] += wi * (-h[l:n])
        else:
            m = -l
            G[0:n - m] += wi * (1.0 - h[m:n])
            G[m:n] += wi * (-h[0:n - m])
    return G


def smooth_obj_and_grad(y, beta):
    h = feasible_from_logit(y)
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    v = corr * dx
    vmax = float(np.max(v))
    shifted = np.clip(beta * (v - vmax), -500, 500)
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    if sum_exp <= 0 or not np.isfinite(sum_exp):
        return 1e6, np.zeros_like(y)
    F = vmax + np.log(sum_exp) / beta
    w = exp_vals / sum_exp
    G = compute_grad_wrt_h(h, w, dx)
    m = h * (1.0 - h)
    M = float(np.sum(m))
    if M < 1e-10:
        return F, np.zeros_like(y)
    B = float(np.dot(G, m)) / M
    grad_y = m * (G - B)
    if not np.all(np.isfinite(grad_y)):
        return 1e6, np.zeros_like(y)
    return F, grad_y


def record_candidate(h):
    h = np.asarray(h, dtype=np.float64)
    n = len(h)
    if n < 2:
        return
    if np.any(h < -1e-10) or np.any(h > 1.0 + 1e-10):
        h = project_feasible(h)
    target = n / 2.0
    if abs(float(np.sum(h)) - target) > 1e-8:
        h = project_feasible(h)
    if np.any(h < -1e-10) or np.any(h > 1.0 + 1e-10):
        return
    c = exact_C(h)
    if not np.isfinite(c):
        return
    if c < _STATE['best_c'] - 1e-15:
        _STATE['best_c'] = c
        _STATE['best_h'] = h.copy()


def _bfgs_obj(y):
    beta = _BFGS_BETA
    F, grad = smooth_obj_and_grad(y, beta)
    h_int = feasible_from_logit(y)
    record_candidate(h_int)
    return F, grad


def _pair_scalar_obj(t):
    h = _PAIR_H
    a = _PAIR_A
    b = _PAIR_B
    hc = h.copy()
    hc[a] += t
    hc[b] -= t
    return exact_C(hc)


def run_bfgs(h_start, beta, max_iter=80):
    global _BFGS_BETA
    h_start = project_feasible(h_start)
    y0 = logit_from_h(h_start)
    _BFGS_BETA = beta
    try:
        res = minimize(_bfgs_obj, y0, method='BFGS', jac=True,
                       options={'maxiter': max_iter, 'gtol': 1e-5, 'disp': False})
        h_end = feasible_from_logit(res.x)
    except Exception:
        h_end = feasible_from_logit(y0)
    record_candidate(h_end)
    return h_end


def pair_polish(h, max_time=30.0, seed=42):
    global _PAIR_H, _PAIR_A, _PAIR_B
    h = h.copy()
    n = len(h)
    c_cur = exact_C(h)
    start_time = time.time()

    y0 = logit_from_h(h)
    h_tmp = feasible_from_logit(y0)
    dx = 2.0 / n
    corr = np.correlate(h_tmp, 1.0 - h_tmp, mode="full")
    v = corr * dx
    vmax = float(np.max(v))
    shifted = np.clip(200.0 * (v - vmax), -500, 500)
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    if sum_exp > 0:
        w = exp_vals / sum_exp
    else:
        w = np.ones(2 * n - 1) / (2 * n - 1)
    G = compute_grad_wrt_h(h_tmp, w, dx)

    top_k = min(14, n)
    top_idx = np.argsort(-np.abs(G))[:top_k]

    pairs = []
    for i in range(len(top_idx)):
        for j in range(i + 1, len(top_idx)):
            pairs.append((int(top_idx[i]), int(top_idx[j])))

    rng = np.random.default_rng(seed)
    for _ in range(20):
        a = int(rng.integers(0, n))
        b = int(rng.integers(0, n))
        if a != b:
            pairs.append((a, b))

    seen = set()
    unique_pairs = []
    for a, b in pairs:
        key = (min(a, b), max(a, b))
        if key not in seen:
            seen.add(key)
            unique_pairs.append(key)

    no_improve = 0
    for a, b in unique_pairs:
        if time.time() - start_time > max_time:
            break

        t_lo = max(-h[a], h[b] - 1.0)
        t_hi = min(1.0 - h[a], h[b])
        if t_hi - t_lo < 1e-12:
            continue

        n_grid = 15
        t_vals = np.linspace(t_lo, t_hi, n_grid)
        best_t = 0.0
        best_c = c_cur
        for t in t_vals:
            hc = h.copy()
            hc[a] += float(t)
            hc[b] -= float(t)
            cc = exact_C(hc)
            if cc < best_c - 1e-14:
                best_c = cc
                best_t = float(t)

        if best_t != 0.0:
            grid_step = (t_hi - t_lo) / n_grid
            lo = max(t_lo, best_t - 2.0 * grid_step)
            hi = min(t_hi, best_t + 2.0 * grid_step)
            if hi > lo:
                _PAIR_H = h
                _PAIR_A = a
                _PAIR_B = b
                try:
                    res = minimize_scalar(_pair_scalar_obj, bounds=(lo, hi),
                                          method='bounded',
                                          options={'xatol': 1e-10, 'maxiter': 25})
                    if res.success and res.fun < best_c - 1e-14:
                        best_t = float(res.x)
                        best_c = float(res.fun)
                except Exception:
                    pass

        if best_c < c_cur - 1e-14:
            h[a] += best_t
            h[b] -= best_t
            h = project_feasible(h)
            c_cur = exact_C(h)
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 20:
                break

    record_candidate(h)
    return h


def coord_polish(h, max_time=20.0):
    global _PAIR_H, _PAIR_A, _PAIR_B
    h = h.copy()
    n = len(h)
    start_time = time.time()
    target = n / 2.0

    # Cyclic coordinate descent: for each coordinate, find the best value
    # that keeps the sum constraint (adjust another coordinate to compensate)
    # Simpler: just do pair transfers with all adjacent pairs
    for _ in range(3):
        if time.time() - start_time > max_time:
            break
        for a in range(n):
            if time.time() - start_time > max_time:
                break
            b = (a + 1) % n
            if a == b:
                continue
            t_lo = max(-h[a], h[b] - 1.0)
            t_hi = min(1.0 - h[a], h[b])
            if t_hi - t_lo < 1e-12:
                continue
            _PAIR_H = h
            _PAIR_A = a
            _PAIR_B = b
            try:
                res = minimize_scalar(_pair_scalar_obj, bounds=(t_lo, t_hi),
                                      method='bounded',
                                      options={'xatol': 1e-10, 'maxiter': 20})
                if res.success:
                    hc = h.copy()
                    hc[a] += float(res.x)
                    hc[b] -= float(res.x)
                    hc = project_feasible(hc)
                    cc = exact_C(hc)
                    if cc < exact_C(h) - 1e-14:
                        h = hc
            except Exception:
                pass
    record_candidate(h)
    return h


def interpolate_h(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new)


def run(seed=42, budget_s=1100, **kwargs):
    global _STATE
    start_time = time.time()
    deadline = start_time + 0.95 * budget_s

    _STATE = {'best_h': None, 'best_c': float('inf')}

    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=np.float64).copy()
    else:
        h0 = np.full(42, 0.5)

    n0 = len(h0)
    h0 = project_feasible(h0)
    record_candidate(h0)

    if _STATE['best_c'] <= 0.3808:
        return _STATE['best_h'], exact_C(_STATE['best_h']), len(_STATE['best_h'])

    # Phase 1: BFGS from parent and variants
    betas = [40.0, 80.0, 150.0, 300.0]
    starts = [h0, h0[::-1].copy()]
    for _ in range(3):
        h_pert = h0 + rng.normal(0, 0.04, n0)
        starts.append(project_feasible(h_pert))

    for h_start in starts:
        if time.time() > deadline - 200:
            break
        for beta in betas:
            if time.time() > deadline - 200:
                break
            h_end = run_bfgs(h_start, beta, max_iter=60)
            record_candidate(h_end)
            if _STATE['best_c'] <= 0.3808:
                return _STATE['best_h'], exact_C(_STATE['best_h']), len(_STATE['best_h'])

    # Phase 2: Pair polish
    if time.time() < deadline - 120:
        h_pol = pair_polish(_STATE['best_h'].copy(), max_time=40, seed=seed)
        record_candidate(h_pol)

    # Phase 3: BFGS + pair polish iteration
    for it in range(4):
        if time.time() > deadline - 120:
            break
        for beta in [150.0, 400.0]:
            if time.time() > deadline - 100:
                break
            h_end = run_bfgs(_STATE['best_h'].copy(), beta, max_iter=50)
            record_candidate(h_end)
        if time.time() > deadline - 80:
            break
        h_pol = pair_polish(_STATE['best_h'].copy(), max_time=20, seed=seed + it)
        record_candidate(h_pol)
        if _STATE['best_c'] <= 0.3808:
            break

    # Phase 4: Coordinate polish
    if time.time() < deadline - 50:
        h_cp = coord_polish(_STATE['best_h'].copy(), max_time=20)
        record_candidate(h_cp)

    # Phase 5: Try smaller dimensions
    if time.time() < deadline - 80:
        for n_try in [36, 30, 28]:
            if time.time() > deadline - 50:
                break
            h_res = interpolate_h(_STATE['best_h'], n_try)
            record_candidate(h_res)
            h_end = run_bfgs(h_res, 100.0, max_iter=40)
            record_candidate(h_end)
            h_pol = pair_polish(h_end, max_time=10, seed=seed + 10 + n_try)
            record_candidate(h_pol)
            if _STATE['best_c'] <= 0.3808:
                break

    # Phase 6: Final polish
    if time.time() < deadline - 20:
        h_pol = pair_polish(_STATE['best_h'].copy(),
                            max_time=max(5.0, deadline - time.time() - 15),
                            seed=seed + 100)
        record_candidate(h_pol)

    h_ret = _STATE['best_h']
    c_ret = exact_C(h_ret)
    n_ret = len(h_ret)
    return h_ret, c_ret, n_ret
