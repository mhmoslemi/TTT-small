    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode="full") * dx
    # log-sum-exp for numerical stability
    m = np.max(corr)
    return (1/beta) * (beta * m + np.log(np.sum(np.exp(beta * (corr - m)))))

def smooth_gradient(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode="full") * dx
    m = np.max(corr)
    p = np.exp(beta * (corr - m))
    p /= np.sum(p)
    
    grad = np.zeros(n)
    for i in range(n):
        s = 0.0
        for k in range(2 * n - 1):
            contrib = 0.0
            # (1-h[i-k]) * [0 <= i-k < n]
            if 0 <= i - k < n:
                contrib += (1 - h[i - k])
            # -h[i+k] * [0 <= i+k < n]
            if 0 <= i + k < n:
                contrib -= h[i + k]
            s += p[k] * contrib
        grad[i] = dx * s

    n = len(h)

    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(beta * (corr - m))))) / beta

def smooth_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    exps = np.exp(beta * (corr - m))
    p = exps / np.sum(exps)
    grad = np.zeros(n)

            jk = i + k
            if 0 <= jk < n:

            s += p[k] * contrib

    return grad

import numpy as np
from scipy.optimize import minimize
import time

_n = 45
_beta = 100.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def smooth_obj(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(beta * (corr - m))))) / beta


def smooth_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    exps = np.exp(beta * (corr - m))
    p = exps / np.sum(exps)
    grad = np.zeros(n)
    for i in range(n):
        s = 0.0
        for k in range(2 * n - 1):
            contrib = 0.0
            ik = i - k
            if 0 <= ik < n:
                contrib += (1.0 - h[ik])
            jk = i + k
            if 0 <= jk < n:
                contrib -= h[jk]
            s += p[k] * contrib
        grad[i] = dx * s
    return grad


def _constraint_fun(x):
    return np.sum(x) - _n / 2.0


def _constraint_jac(x):
    return np.ones(_n)


def _smooth_obj_sl(x):
    return smooth_obj(x, _beta)


def _smooth_grad_sl(x):
    return smooth_grad(x, _beta)


def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        cnt = np.sum(mask)
        if cnt > 0:
            h = h.copy()
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def pair_exchange_round(h, n, max_sweeps, time_limit, start_time):
    dx = 2.0 / n
    best_c5 = compute_c5(h)

    for sweep in range(max_sweeps):
        if time.time() - start_time > time_limit:
            break

        c = np.correlate(h, 1.0 - h, mode="full")
        cur_c5 = np.max(c) * dx

        L = np.zeros((n, 2 * n - 1))
        R = np.zeros((n, 2 * n - 1))
        for i in range(n):
            for k in range(2 * n - 1):
                ig = i - k
                if 0 <= ig < n:
                    L[i, k] = 1.0 - h[ig]
                jg = i + k
                if 0 <= jg < n:
                    R[i, k] = h[jg]

        best_improvement = 0.0
        best_pair = None
        best_tau = 0.0

        for i in range(n):
            for j in range(n):
                if i == j:
                    continue
                tau_lo = -h[i]
                tau_hi = 1.0 - h[j]
                if tau_lo >= tau_hi - 1e-12:
                    continue

                delta = L[j] - L[i] + R[i] - R[j]

                best_val = cur_c5
                best_tau_local = 0.0
                for t in np.linspace(tau_lo, tau_hi, 7):
                    val = np.max(c + delta * t) * dx
                    if val < best_val:
                        best_val = val
                        best_tau_local = t

                if best_val < cur_c5 - best_improvement:
                    best_improvement = cur_c5 - best_val
                    best_pair = (i, j)
                    best_tau = best_tau_local

        if best_pair is None or best_improvement < 1e-13:
            break

        i, j = best_pair
        h[i] -= best_tau
        h[j] += best_tau
        h = np.clip(h, 0.0, 1.0)
        h = project_feasible(h, n)

        c5_now = compute_c5(h)
        if c5_now < best_c5:
            best_c5 = c5_now

    return h, best_c5
