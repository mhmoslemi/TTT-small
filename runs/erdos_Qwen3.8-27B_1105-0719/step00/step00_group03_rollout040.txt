    # Phase 3: Try nearby n values
    for n2 in [47, 49, 51]:
        if tleft() < 120:
            break
        x_old = np.linspace(0.0, 2.0, n, endpoint=False) + 1.0 / n
        x_new = np.linspace(0.0, 2.0, n2, endpoint=False) + 1.0 / n2
        hi = np.interp(x_new, x_old, best_h)
        target2 = n2 / 2.0
        hi = _proj(np.clip(hi, 0.0, 1.0), target2)
        bounds2 = [(0.0, 1.0)] * n2
        cons2 = [{'type': 'eq', 'fun': lambda x, _t=target2: np.sum(x) - _t,
                  'jac': lambda x, _n2=n2: np.ones(_n2)}]
        cur2 = hi.copy()
        for a in [8.0, 32.0, 128.0, 512.0]:
            if tleft() < 30:
                break
            cur2, _ = _optimize(cur2, n2, target2, bounds2, cons2, a, t0, budget_s, 400)
            c5 = float(np.max(_c5_vals(cur2)))
            if c5 < best_c5:
                best_c5 = c5
                best_h = cur2.copy()
                n = n2

    # Phase 4: Final polish on best solution
    if tleft() > 20:
        n = len(best_h)
        target = n / 2.0
        best_h = _proj(np.clip(best_h, 0.0, 1.0), target)
        bounds_f = [(0.0, 1.0)] * n
        cons_f = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
                   'jac': lambda x: np.ones(n)}]
        for a in [2048.0, 4096.0]:
            if tleft() < 15:
                break
            best_h, _ = _optimize(best_h, n, target, bounds_f, cons_f, a, t0, budget_s, 600)
            c5 = float(np.max(_c5_vals(best_h)))
            if c5 < best_c5:
                best_c5 = c5

    best_h = _proj(np.clip(best_h, 0.0, 1.0), len(best_h) / 2.0)
    best_c5 = float(np.max(_c5_vals(best_h)))
    return best_h, best_c5, len(best_h)
