        if resid > 0:
            idx = np.where(result < 1.0)[0]
        else:
            idx = np.where(result > 0.0)[0]
        if len(idx) > 0:
            result[idx] = np.clip(result[idx] + resid / len(idx), 0.0, 1.0)
    return result

def lse_value_and_grad(h, beta, F_all=None):
    n = len(h)
    dx = 2.0 / n
    if F_all is None:
        F_all = np.correlate(h, 1.0 - h, mode="full") * dx
    Fmax = np.max(F_all)
    shifted = beta * (F_all - Fmax)
    max_shifted = np.max(shifted)
    exp_vals = np.exp(shifted - max_shifted)
    s = exp_vals.sum()
    phi = Fmax + (max_shifted + np.log(s)) / beta
    w = exp_vals / s

    grad = np.zeros(n)
    n_lags = len(F_all)
    for k in range(n_lags):
        wk = w[k]
        if wk < 1e-14:
            continue
        d = k - (n - 1)
        if d == 0:
            grad += dx * wk * (1.0 - 2.0 * h)
        elif d > 0:
            grad[d:n] += dx * wk * (1.0 - h[0:n-d])
            grad[0:n-d] -= dx * wk * h[d:n]
        else:
            dp = -d
            grad[0:n-dp] += dx * wk * (1.0 - h[dp:n])
            grad[dp:n] -= dx * wk * h[0:n-dp]
    return phi, grad, F_all

def run(seed=42, budget_s=1100, **kwargs):
    import time
    t0 = time.time()
    deadline = t0 + budget_s - 10.0

    # Get initial values
    h_init = None
    try:
        h_init = np.array(initial_h_values, dtype=np.float64).copy()
    except:
        pass

    if h_init is not None and len(h_init) >= 10:
        n = len(h_init)
    else:
        n = 63
        h_init = np.tile(np.array([0.6, 0.4, 0.5, 0.3, 0.7, 0.2, 0.5, 0.4, 0.6, 0.3]), n // 10 + 1)[:n]

    target = n / 2.0
    h_best = project_feasible(h_init, target)
    c5_best, _ = objective(h_best)

    rng = np.random.default_rng(seed)

    def time_ok():
        return time.time() < deadline

    # Validate gradient once
    def check_grad(h, beta):
        phi, grad, F = lse_value_and_grad(h, beta)
        eps = 1e-6
        idx = rng.integers(0, n, size=min(5, n))
        for i in idx:
            hp = h.copy(); hp[i] += eps
            hp = np.clip(hp, 0, 1)
            Fp = np.correlate(hp, 1.0 - hp, mode="full") * (2.0/n)
            phi_p = Fp.max() + np.log(np.exp(beta*(Fp - Fp.max())).sum()) / beta
            hm = h.copy(); hm[i] -= eps
            hm = np.clip(hm, 0, 1)
            Fm = np.correlate(hm, 1.0 - hm, mode="full") * (2.0/n)
            phi_m = Fm.max() + np.log(np.exp(beta*(Fm - Fm.max())).sum()) / beta
            fd = (phi_p - phi_m) / (2*eps)
            if abs(fd - grad[i]) > 0.01 * max(1, abs(fd)):
                return False
        return True

    # Phase 1: SLSQP on soft-max with increasing beta
    betas = [10.0, 20.0, 40.0, 80.0, 160.0, 320.0, 640.0, 1280.0]

    for beta in betas:
        if not time_ok():
            break

        def obj_func(x):
            phi, _, _ = lse_value_and_grad(x, beta)
            return phi

        def obj_grad(x):
            _, g, _ = lse_value_and_grad(x, beta)
            return g

        cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target, 'jac': lambda x: np.ones(n)}]
        bounds = [(0.0, 1.0)] * n

        x0 = h_best.copy()
        try:
            res = minimize(obj_func, x0, jac=obj_grad, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 300, 'ftol': 1e-10})
            if res.success or res.fun < c5_best:
                h_cand = project_feasible(res.x, target)
                c5_cand, _ = objective(h_cand)
                if c5_cand < c5_best - 1e-12:
                    h_best = h_cand
                    c5_best = c5_cand
        except:
            pass

    # Phase 2: Projected gradient with high beta
    for beta in [256.0, 512.0, 1024.0, 2048.0, 4096.0]:
        if not time_ok():
            break
        h_cur = h_best.copy()
        alpha = 0.05
        plateau = 0
        for it in range(3000):
            if not time_ok():
                break
            phi, grad, F_all = lse_value_and_grad(h_cur, beta, F_all=objective(h_cur)[1])
            grad_tan = grad - np.mean(grad)
            gnorm = np.linalg.norm(grad_tan)
            if gnorm < 1e-12:
                break
            h_trial = project_feasible(h_cur - alpha * grad_tan / gnorm, target)
            c5_t, _ = objective(h_trial)
            if c5_t < c5_best - 1e-13:
                h_best = h_trial.copy()
                c5_best = c5_t
                h_cur = h_trial
                plateau = 0
                alpha = min(alpha * 1.1, 0.3)
            else:
                h_cur = h_trial
                plateau += 1
                if plateau % 20 == 0:
                    alpha *= 0.8
                    if alpha < 0.001:
                        break
                if plateau > 500:
                    break

    # Phase 3: Random two-point transfers (accept only improvements)
    h_cur = h_best.copy()
    c5_cur = c5_best
    accepted = 0
    attempts = 0
    while time_ok() and attempts < 300000:
        attempts += 1
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        hi = h_cur[i]
        hj = h_cur[j]
        if hi < 1.0 and hj > 0.0:
            dt = rng.uniform(0.0005, 0.05)
            dt = min(dt, 1.0 - hi, hj)
            h_cur[i] = hi + dt
            h_cur[j] = hj - dt
            c5_t, _ = objective(h_cur)
            if c5_t < c5_cur - 1e-13:
                c5_cur = c5_t
                if c5_cur < c5_best - 1e-13:
                    h_best = h_cur.copy()
                    c5_best = c5_t
                accepted += 1
            else:
                h_cur[i] = hi
                h_cur[j] = hj
        elif hi > 0.0 and hj < 1.0:
            dt = rng.uniform(0.0005, 0.05)
            dt = min(dt, hi, 1.0 - hj)
            h_cur[i] = hi - dt
            h_cur[j] = hj + dt
            c5_t, _ = objective(h_cur)
            if c5_t < c5_cur - 1e-13:
                c5_cur = c5_t
                if c5_t < c5_best - 1e-13:
                    h_best = h_cur.copy()
                    c5_best = c5_t
                accepted += 1
            else:
                h_cur[i] = hi
                h_cur[j] = hj
        # Periodic restart from best
        if attempts % 5000 == 0:
            h_cur = h_best.copy()
            c5_cur = c5_best

    # Phase 4: Active-set LP trust region
    if time_ok():
        h_cur = h_best.copy()
        radius = 0.05
        for outer in range(50):
            if not time_ok():
                break
            c5_cur, F_all = objective(h_cur)
            if c5_cur >= c5_best + 1e-10:
                h_cur = h_best.copy()
                radius *= 0.5
                continue

            # Find active lags
            M = np.max(F_all)
            active = np.where(F_all >= M - 0.005)[0]
            if len(active) == 0:
                active = np.array([np.argmax(F_all)])
            if len(active) > 25:
                active = np.argsort(F_all)[-25:]

            # Build LP
            n_vars = n + 1
            c_lp = np.zeros(n_vars)
            c_lp[n] = 1.0  # minimize t

            A_ub = []
            b_ub = []
            for k_idx in active:
                wk = 1.0
                d = k_idx - (n - 1)
                g_k = np.zeros(n)
                dx = 2.0 / n
                if d == 0:
                    g_k = dx * (1.0 - 2.0 * h_cur)
                elif d > 0:
                    g_k[d:n] = dx * (1.0 - h_cur[0:n-d])
                    g_k[0:n-d] = -dx * h_cur[d:n]
                else:
                    dp = -d
                    g_k[0:n-dp] = dx * (1.0 - h_cur[dp:n])
                    g_k[dp:n] = -dx * h_cur[0:n-dp]

                f_k = F_all[k_idx]
                row = np.zeros(n_vars)
                row[:n] = g_k
                row[n] = -1.0
                rhs = -f_k + np.dot(g_k, h_cur)
                A_ub.append(row)
                b_ub.append(rhs)

            A_ub = np.array(A_ub)
            b_ub = np.array(b_ub)

            A_eq = np.zeros((1, n_vars))
            A_eq[0, :n] = 1.0
            b_eq = np.array([target])

            lb = np.maximum(h_cur - radius, 0.0)
            ub = np.minimum(h_cur + radius, 1.0)
            bounds_lp = list(zip(lb, ub)) + [(None, None)]

            try:
                res_lp = linprog(c_lp, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                                 bounds=bounds_lp, method='highs')
                if res_lp.success:
                    h_trial = project_feasible(res_lp.x[:n], target)
                    c5_t, _ = objective(h_trial)
                    if c5_t < c5_best - 1e-13:
                        h_best = h_trial.copy()
                        c5_best = c5_t
                        h_cur = h_trial.copy()
                        radius = min(radius * 1.2, 0.15)
                    elif c5_t < c5_cur + 0.002:
                        h_cur = h_trial.copy()
                        radius *= 0.7
                    else:
                        radius *= 0.5
                else:
                    radius *= 0.5
            except:
                radius *= 0.5

            if radius < 0.002:
                h_cur = h_best.copy()
                radius = 0.05

    # Phase 5: Final fine transfers
    if time_ok():
        h_cur = h_best.copy()
        c5_cur = c5_best
        for _ in range(200000):
            if not time_ok():
                break
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:
                continue
            hi = h_cur[i]
            hj = h_cur[j]
            if hi < 1.0 and hj > 0.0:
                dt = rng.uniform(0.0001, 0.01)
                dt = min(dt, 1.0 - hi, hj)
                if dt < 1e-8:
                    continue
                h_cur[i] = hi + dt
                h_cur[j] = hj - dt
                c5_t, _ = objective(h_cur)
                if c5_t < c5_cur - 1e-14:
                    c5_cur = c5_t
                    if c5_t < c5_best - 1e-14:
                        h_best = h_cur.copy()
                        c5_best = c5_t
                else:
                    h_cur[i] = hi
                    h_cur[j] = hj
            elif hi > 0.0 and hj < 1.0:
                dt = rng.uniform(0.0001, 0.01)
                dt = min(dt, hi, 1.0 - hj)
                if dt < 1e-8:
                    continue
                h_cur[i] = hi - dt
                h_cur[j] = hj + dt
                c5_t, _ = objective(h_cur)
                if c5_t < c5_cur - 1e-14:
                    c5_cur = c5_t
                    if c5_t < c5_best - 1e-14:
                        h_best = h_cur.copy()
                        c5_best = c5_t
                else:
                    h_cur[i] = hi
                    h_cur[j] = hj
            if _ % 10000 == 9999:
                h_cur = h_best.copy()
                c5_cur = c5_best

    # Final validation
    h_final = project_feasible(h_best, target)
    c5_final, _ = objective(h_final)

    # Ensure feasibility
    assert np.all(h_final >= -1e-15) and np.all(h_final <= 1.0 + 1e-15)
    assert abs(h_final.sum() - target) < 1e-10

    return h_final, c5_final, n
