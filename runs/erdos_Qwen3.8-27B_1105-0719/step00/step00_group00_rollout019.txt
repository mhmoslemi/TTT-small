def compute_correlation(h):
    g = 1.0 - h
    return np.correlate(h, g, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlation(h)
    return dx * np.max(corr)

def pair_alpha_beta(h, a, b):
    n = len(h)
    g = 1.0 - h
    L = 2 * n - 1
    alpha = np.zeros(L)
    beta = np.zeros(L)
    m_diag = n - 1
    
    # Diagonal terms (a,a) and (b,b) both at m_diag
    alpha[m_diag] += g[a] - h[a] + h[b] - g[b]
    beta[m_diag] -= 2
    
    # Cross terms
    m_ab = a + n - 1 - b
    if 0 <= m_ab < L:
        alpha[m_ab] += h[a] + g[b]
        beta[m_ab] += 1
    
    m_ba = b + n - 1 - a
    if 0 <= m_ba < L:
        alpha[m_ba] -= h[b] + g[a]
        beta[m_ba] += 1
    
    # Linear terms from (a,j) and (b,j) for j != a,b
    for j in range(n):
        if j == a or j == b:
            continue
        m1 = a + n - 1 - j
        alpha[m1] += g[j]
        m2 = b + n - 1 - j
        alpha[m2] -= g[j]
    
    # Linear terms from (i,a) and (i,b) for i != a,b
    for i in range(n):
        if i == a or i == b:
            continue
        m1 = i + n - 1 - a
        alpha[m1] -= h[i]
        m2 = i + n - 1 - b
        alpha[m2] += h[i]
    
    return alpha, beta

def optimize_pair(h, corr, a, b, n_grid=100):

    high = min(1.0 - h[a], h[b])

    alpha, beta = pair_alpha_beta(h, a, b)

    new_vals = corr[None, :] + alpha[None, :] * deltas[:, None] + beta[None, :] * deltas[:, None]**2

    best_delta = deltas[idx]
    best_max = maxes[idx]

        hi = deltas[idx+1]
        fine = np.linspace(lo, hi, 200)

        maxes_f = new_vals_f.max(axis=1)
        idx_f = maxes_f.argmin()
        best_delta = fine[idx_f]
        best_max = maxes_f[idx_f]

                delta, new_max = optimize_pair(h, corr, a, b, n_grid)

                    h[b] -= delta
                    h = np.clip(h, 0.0, 1.0)
                    corr = compute_correlation(h)
                    current_max = np.max(corr)
                    improved = True
        if not improved:
            break
    
    return h, dx * current_max

import numpy as np
import time


def compute_correlation(h):
    g = 1.0 - h
    return np.correlate(h, g, mode='full')


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlation(h)
    return dx * np.max(corr)


def pair_alpha_beta(h, a, b):
    n = len(h)
    g = 1.0 - h
    L = 2 * n - 1
    alpha = np.zeros(L)
    beta = np.zeros(L)
    m_diag = n - 1

    alpha[m_diag] += (g[a] - h[a]) + (h[b] - g[b])
    beta[m_diag] -= 2.0

    m_ab = a - b + n - 1
    if 0 <= m_ab < L:
        alpha[m_ab] += h[a] + g[b]
        beta[m_ab] += 1.0

    m_ba = b - a + n - 1
    if 0 <= m_ba < L:
        alpha[m_ba] -= (h[b] + g[a])
        beta[m_ba] += 1.0

    for j in range(n):
        if j == a or j == b:
            continue
        m1 = a - j + n - 1
        alpha[m1] += g[j]
        m2 = b - j + n - 1
        alpha[m2] -= g[j]

    for i in range(n):
        if i == a or i == b:
            continue
        m1 = i - a + n - 1
        alpha[m1] -= h[i]
        m2 = i - b + n - 1
        alpha[m2] += h[i]

    return alpha, beta


def optimize_pair(h, corr, a, b, n_grid=120):
    low = max(-h[a], h[b] - 1.0)
    high = min(1.0 - h[a], h[b])
    if high - low < 1e-14:
        return 0.0, np.max(corr)

    alpha, beta = pair_alpha_beta(h, a, b)

    deltas = np.linspace(low, high, n_grid)
    new_vals = corr[None, :] + alpha[None, :] * deltas[:, None] + beta[None, :] * deltas[:, None] ** 2
    maxes = new_vals.max(axis=1)

    idx = maxes.argmin()
    best_delta = deltas[idx]
    best_max = maxes[idx]

    if 0 < idx < n_grid - 1:
        lo = deltas[idx - 1]
        hi = deltas[idx + 1]
        fine = np.linspace(lo, hi, 256)
        nv = corr[None, :] + alpha[None, :] * fine[:, None] + beta[None, :] * fine[:, None] ** 2
        mx = nv.max(axis=1)
        idx_f = mx.argmin()
        best_delta = fine[idx_f]
        best_max = mx[idx_f]

    return best_delta, best_max


def local_search(h, max_passes=50, n_grid=120, time_limit=None, start_time=None):
    h = h.copy()
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlation(h)
    current_max = np.max(corr)
    best_c5 = dx * current_max

    for _ in range(max_passes):
        if time_limit is not None and time.time() - start_time > time_limit:
            break
        improved = False
        for a in range(n):
            if time_limit is not None and time.time() - start_time > time_limit:
                break
            for b in range(n):
                if a == b:
                    continue
                delta, new_max = optimize_pair(h, corr, a, b, n_grid)
                if new_max < current_max - 1e-11:
                    h[a] += delta
                    h[b] -= delta
                    h = np.clip(h, 0.0, 1.0)
                    corr = compute_correlation(h)
                    current_max = np.max(corr)
                    best_c5 = dx * current_max
                    improved = True
        if not improved:
            break

    return h, dx * current_max


def try_interpolate(h_src, n_tgt):
    n_src = len(h_src)
    x_src = np.arange(n_src, dtype=float)
    x_tgt = np.linspace(0, n_src - 1, n_tgt)
    h_interp = np.interp(x_tgt, x_src, h_src)
    s = h_interp.sum()
    if s > 1e-12:
        h_interp = h_interp * (n_tgt / 2.0) / s
    h_interp = np.clip(h_interp, 0.0, 1.0)
    diff = n_tgt / 2.0 - h_interp.sum()
    h_interp += diff / n_tgt
    h_interp = np.clip(h_interp, 0.0, 1.0)
    return h_interp


def make_cosine_init(n, freq=1, amp=0.3, seed=0):
    x = np.arange(n, dtype=float)
    h = 0.5 + amp * np.cos(2 * np.pi * freq * x / n)
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s > 1e-12:
        h = h * (n / 2.0) / s
    h = np.clip(h, 0.0, 1.0)
    diff = n / 2.0 - h.sum()
    h += diff / n
    h = np.clip(h, 0.0, 1.0)
    return h


def make_block_init(n, n_blocks=4):
    block_size = n // n_blocks
    h = np.zeros(n)
    for i in range(n_blocks):
        start = i * block_size
        end = start + block_size
        h[start:end] = 1.0 if i % 2 == 0 else 0.0
    if n % n_blocks != 0:
        h[-(n % n_blocks):] = 1.0
    s = h.sum()
    if s > 1e-12:
        h = h * (n / 2.0) / s
    h = np.clip(h, 0.0, 1.0)
    return h


def gradient_step(h, corr, step_size=0.02):
    n = len(h)
    dx = 2.0 / n
    L = 2 * n - 1
    max_val = np.max(corr)
    active = corr > max_val - 1e-8

    grad = np.zeros(n)
    for m in range(L):
        if not active[m]:
            continue
        for i in range(n):
            j = i + n - 1 - m
            if 0 <= j < n:
                grad[i] += (1.0 - h[j])
            j2 = m - n + 1 + i
            if 0 <= j2 < n and j2 != j:
                grad[i] -= h[j2]

    new_h = h - step_size * dx * grad
    new_h = np.clip(new_h, 0.0, 1.0)
    target = n / 2.0
    new_h = new_h * target / max(new_h.sum(), 1e-12)
    new_h = np.clip(new_h, 0.0, 1.0)
    return new_h
