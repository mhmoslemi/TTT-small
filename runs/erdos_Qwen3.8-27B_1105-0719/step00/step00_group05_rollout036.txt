import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_correlations(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_constraint(h, target_sum):
    """Project h to {h: 0<=h<=1, sum(h)=target_sum} using simple iteration."""
    h = h.copy()
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if not np.any(mask):
            # all at bounds, do best we can
            break
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def coordinate_descent_sweep(h, target_sum):
    """One sweep of coordinate descent."""
    n = len(h)
    for j in range(n):
        old_hj = h[j]
        h_new = optimize_single_coordinate(h, j)
        h[j] = h_new
    
    # Project to constraint
    h = project_to_constraint(h, target_sum)
    return h

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")

        idx1 = j + k

            B[m] -= h[j] * (1.0 - h[idx1])
        if 0 <= idx2 < n:
            A[m] -= h[idx2]
            B[m] -= h[idx2] * (1.0 - h[j])

    n = len(h)

        S = base_sum[j] - target_sum  # sum of others minus target
        # F(x) = max_k(A_k*x + B_k) + lam * (x + S)
        #      = max_k(A_k*x + B_k) + lam*x + lam*S

                a_val += (1.0 - h[idx1])

            B[m] = corr[m] - a_val * h[j]

    n1 = int(np.sum(h))
    target = n // 2
    if n1 > target:
        ones = np.where(h == 1)[0]
        if len(ones) >= n1 - target:
            h[rng.choice(ones, n1 - target, replace=False)] = 0
    elif n1 < target:
        zeros = np.where(h == 0)[0]
        if len(zeros) >= target - n1:
            h[rng.choice(zeros, target - n1, replace=False)] = 1
    return h

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()

    h0 = np.array(initial_h_values, dtype=float)
    n = len(h0)
    ts = n / 2.0

    best_h = h0.copy()
    best_c5 = evaluate_c5(best_h)

    h = h0.copy()
    h = project_to_constraint(h, ts)
    c5 = evaluate_c5(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 1: Coordinate descent with increasing precision
    ng = 501
    no_improve = 0
    phase1_time = min(300, budget_s * 0.3)

    while time.time() - t0 < t0 + phase1_time:
        for j in range(n):
            if time.time() - t0 > t0 + phase1_time:
                break
            h[j] = optimize_coord(h, j, ng)
        h = project_to_constraint(h, ts)
        c5 = evaluate_c5(h)
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve >= 15:
            npert = rng.randint(1, 4)
            if npert > 0:
                idx = rng.choice(n, size=npert, replace=False)
                h[idx] += rng.uniform(-0.2, 0.2, size=npert)
                h = project_to_constraint(h, ts)
            no_improve = 0
        if ng < 2001 and no_improve > 5:
            ng = min(2001, ng * 2)

    # Phase 2: Fine optimization from best solution
    h = best_h.copy()
    ng = 2001
    no_improve = 0
    phase2_end = t0 + budget_s * 0.7

    while time.time() - t0 < phase2_end:
        for j in range(n):
            if time.time() - t0 > phase2_end:
                break
            h[j] = optimize_coord(h, j, ng)
        h = project_to_constraint(h, ts)
        c5 = evaluate_c5(h)
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve >= 20:
            npert = rng.randint(1, 3)
            if npert > 0:
                idx = rng.choice(n, size=npert, replace=False)
                h[idx] = rng.uniform(0, 1, size=npert)
                h = project_to_constraint(h, ts)
            no_improve = 0
        if no_improve >= 200:
            h = try_binary_init(rng, n)
            h = project_to_constraint(h, ts)
            no_improve = 0

    # Phase 3: Ultra-fine polish
    h = best_h.copy()
    phase3_end = t0 + budget_s - 3
    no_improve = 0

    while time.time() - t0 < phase3_end:
        for j in range(n):
            if time.time() - t0 > phase3_end:
                break
            h[j] = optimize_coord_fine(h, j)
        h = project_to_constraint(h, ts)
        c5 = evaluate_c5(h)
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve >= 30:
            break

    return best_h, best_c5, n
