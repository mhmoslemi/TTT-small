import numpy as np
from scipy.optimize import minimize
import time

def exact_objective(h):
    """Compute C5 exactly."""
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def is_feasible(h, tol=1e-9):
    """Check if h satisfies constraints."""
    n = len(h)
    if not np.all(np.isfinite(h)):
        return False
    if np.min(h) < -tol or np.max(h) > 1.0 + tol:
        return False
    if abs(np.sum(h) - n/2.0) > tol:
        return False
    return True

def project_feasible(h_raw):
    """Project h_raw onto the feasible set [0,1]^n with sum = n/2."""
    n = len(h_raw)
    target = n / 2.0
    h = np.clip(h_raw.astype(np.float64), 0.0, 1.0)
    
    # Bisection to find lambda such that sum(clip(h_raw + lambda, 0, 1)) = target
    def sum_at_lambda(lam):
        return np.sum(np.clip(h_raw.astype(np.float64) + lam, 0.0, 1.0))
    
    lo, hi = -2.0, 2.0
    # Ensure target is in range
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = sum_at_lambda(mid)
        if s < target:
            lo = mid
        else:
            hi = mid
    
    h = np.clip(h_raw.astype(np.float64) + (lo+hi)/2.0, 0.0, 1.0)
    
    # Fine-tune to hit exact sum
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        # Distribute residual
        if residual > 0:
            # Add to entries with most room
            room = 1.0 - h
            idx = np.argsort(-room)
            for i in idx:
                add = min(residual, room[i])
                h[i] += add
                residual -= add
                if abs(residual) < 1e-14:
                    break
        else:
            # Remove from entries with most value
            idx = np.argsort(h)
            for i in idx:
                rem = min(-residual, h[i])
                h[i] -= rem
                residual += rem
                if abs(residual) < 1e-14:
                    break
    
    # Final clip for safety
    h = np.clip(h, 0.0, 1.0)
    return h

def directional_values(h):
    """Compute all directional correlation values (raw, without dx).
    Returns array of length 2n-1 matching np.correlate output order."""
    return np.correlate(h, 1.0 - h, mode='full')

def build_weights(vals, K=10, decay=0.2):
    """Build weights concentrated on top K values."""
    n = len(vals)
    sorted_idx = np.argsort(-vals)
    K = min(K, n)
    
    # Use top K with exponential decay
    top_vals = vals[sorted_idx[:K]]
    max_val = top_vals[0]
    
    # Tie tolerance
    tie_tol = max(1e-7, 0.01 * max_val) if max_val > 0 else 1e-7
    
    # Find how many are within tie_tol of max
    active_mask = vals >= max_val - tie_tol
    active_count = np.sum(active_mask)
    K_eff = max(K, int(active_count))
    K_eff = min(K_eff, n)
    
    weights = np.zeros(n)
    for rank in range(K_eff):
        idx = sorted_idx[rank]
        w = np.exp(-decay * rank)
        weights[idx] = w
    
    # Normalize
    total = np.sum(weights)
    if total > 1e-15:
        weights /= total
    else:
        weights = np.ones(n) / n
    
    return weights

def weighted_gradient(h, weights):
    """Compute gradient of sum_m weights[m] * c[m] where c = correlate(h, 1-h, 'full')."""
    n = len(h)
    g = np.zeros(n)
    
    # For L = 0 (m = n-1): contribution is w[n-1] * (1 - 2*h[j])
    w0 = weights[n-1]
    g += w0 * (1.0 - 2.0 * h)
    
    # For L > 0:
    # W[L] = weights[n-1+L] + weights[n-1-L]
    # d(c)/dh[j] = 1_{L<=j} * (1-h[j-L]) - 1_{j<=n-1-L} * h[j+L]
    for L in range(1, n):
        W = weights[n-1+L] + weights[n-1-L]
        if W < 1e-15:
            continue
        # Term 1: j from L to n-1, contribution (1-h[j-L])
        if L < n:
            j_range = np.arange(L, n)
            g[j_range] += W * (1.0 - h[j_range - L])
        # Term 2: j from 0 to n-1-L, contribution -h[j+L]
        if n-1-L >= 0:
            j_range = np.arange(0, n-L)
            g[j_range] -= W * h[j_range + L]
    
    return g

def build_mass_matrix(h, weights, edge_scale=1.0, rho=1.0):
    """Build mass matrix (graph Laplacian + rho*I)."""
    n = len(h)
    M = np.zeros((n, n))
    
    # Graph Laplacian from active edges
    for L in range(1, n):
        W = weights[n-1+L] + weights[n-1-L]
        if W < 1e-12:
            continue
        w = edge_scale * W
        for i in range(n - L):
            j = i + L
            M[i, i] += w
            M[j, j] += w
            M[i, j] -= w
            M[j, i] -= w
    
    # Add identity for strong convexity
    M += rho * np.eye(n)
    
    return M

class QPModel:
    def __init__(self, g, M):
        self.g = g
        self.M = M
        self.n = len(g)
    
    def objective(self, d):
        return self.g @ d + 0.5 * d @ self.M @ d
    
    def gradient(self, d):
        return self.g + self.M @ d

def solve_qp_subproblem(g, M, h, delta, n):
    """Solve the QP subproblem using SLSQP."""
    model = QPModel(g, M)
    
    def obj(d):
        return model.objective(d)
    
    def grad(d):
        return model.gradient(d)
    
    # Constraint: sum(d) = 0
    cons = [{'type': 'eq', 'fun': lambda d: np.sum(d), 'jac': lambda d: np.ones(n)}]
    
    # Bounds
    lower = np.maximum(-h, -delta)
    upper = np.minimum(1.0 - h, delta)
    bounds = list(zip(lower, upper))
    
    # Initial point
    d0 = np.zeros(n)
    
    result = minimize(obj, d0, method='SLSQP', jac=grad,
                      bounds=bounds, constraints=cons,
                      options={'maxiter': 50, 'ftol': 1e-10})
    
    if result.success or result.x is not None:
        d = result.x
        if np.all(np.isfinite(d)):
            return d
    return np.zeros(n)

def sqp_local(h_start, time_budget, params, start_time):
    """Local SQP descent from h_start."""
    h = project_feasible(h_start.copy())
    c_current = exact_objective(h)
    
    K = params.get('K', 10)
    decay = params.get('decay', 0.2)
    edge_scale = params.get('edge_scale', 1.0)
    rho_base = params.get('rho', 1.0)
    trust_init = params.get('trust', 0.05)
    trust_min = params.get('trust_min', 0.001)
    trust_max = params.get('trust_max', 0.15)
    
    trust = trust_init
    no_improve_count = 0
    max_no_improve = params.get('max_no_improve', 15)
    
    while time.time() - start_time < time_budget:
        # Compute directional values and weights
        vals = directional_values(h)
        weights = build_weights(vals, K=K, decay=decay)
        
        # Compute gradient
        g = weighted_gradient(h, weights)
        gnorm = np.linalg.norm(g)
        
        if gnorm < 1e-12:
            # At a stationary point of the weighted objective
            no_improve_count += 1
            if no_improve_count >= max_no_improve:
                break
            # Random perturbation
            rng = np.random.default_rng(int(time.time()*1000) % 100000)
            d = rng.normal(0, 0.01, len(h))
            d -= np.mean(d)
            h_try = project_feasible(h + d)
            c_try = exact_objective(h_try)
            if c_try < c_current:
                h = h_try
                c_current = c_try
                no_improve_count = 0
                trust = min(trust_max, trust * 1.2)
            else:
                trust = max(trust_min, trust * 0.5)
            continue
        
        # Build mass matrix
        rho = rho_base * gnorm / (2.0 * max(trust, 1e-4))
        rho = max(rho, 0.1)
        M = build_mass_matrix(h, weights, edge_scale=edge_scale, rho=rho)
        
        # Solve QP
        d = solve_qp_subproblem(g, M, h, trust, len(h))
        
        if np.linalg.norm(d) < 1e-14:
            no_improve_count += 1
            if no_improve_count >= max_no_improve:
                break
            trust = max(trust_min, trust * 0.7)
            continue
        
        # Line search
        found = False
        for alpha in [1.0, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05]:
            h_try = project_feasible(h + alpha * d)
            c_try = exact_objective(h_try)
            if c_try < c_current - 1e-12:
                h = h_try
                c_current = c_try
                found = True
                no_improve_count = 0
                trust = min(trust_max, trust * 1.15)
                break
            elif c_try <= c_current + 1e-10:
                # Accept neutral step (for exploration)
                if no_improve_count > 5:
                    h = h_try
                    c_current = c_try
                    found = True
                    no_improve_count = 0
                    break
        
        if not found:
            no_improve_count += 1
