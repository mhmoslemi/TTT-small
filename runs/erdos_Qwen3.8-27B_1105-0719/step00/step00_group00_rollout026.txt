import numpy as np
import time

def compute_corr(h):
    """Compute the correlation array using np.correlate"""
    g = 1.0 - h
    return np.correlate(h, g, mode="full")

def compute_c5(h):
    """Compute C5 value"""
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    return np.max(corr) * dx

def pair_b_matrix(h, a, b, n):
    """Compute B vector for pair transfer from b to a"""
    lag = np.arange(-(n-1), n)
    # B_l = g[a-l] - g[b-l] - h[a+l] + h[b+l] (with clipping)
    g = 1.0 - h
    B = np.zeros(n * 2 - 1)
    for idx, l in enumerate(lag):
        val = 0.0
        if 0 <= a - l < n:
            val += g[a - l]
        if 0 <= b - l < n:
            val -= g[b - l]
        if 0 <= a + l < n:
            val -= h[a + l]
        if 0 <= b + l < n:
            val += h[b + l]
        B[idx] = val
    return B

    # idx_right[a, k] = a + k - (n-1)

    valid_right = (idx_right >= 0) & (idx_right < n)

    idx_right_clipped = np.clip(idx_right, 0, n-1)

    G_left = g[idx_left_clipped] * valid_left  # (n, 2n-1)

    # B[k] = G_left[a,k] - G_left[b,k] - H_right[a,k] + H_right[b,k]
    # Q[k] = -2*I(k=n-1) + I(k=a-b+n-1) + I(k=b-a+n-1)

                Q[a - b + n - 1] += 1  # lag = a - b

                # Sample delta values

                    H_right = h[idx_right_clipped] * valid_right
                    improved_this_pass = True
        
        if not improved_this_pass:
            break

def compute_c5(h):
    """Compute C5 = max(correlate(h, 1-h, full)) * dx"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def precompute_indices(n):
    """Precompute index arrays for pair transfer operations."""
    a_idx = np.arange(n)[:, None].astype(np.int64)
    k_idx = np.arange(2*n-1)[None, :].astype(np.int64)
    idx_left = a_idx - k_idx + (n-1)
    idx_right = a_idx + k_idx - (n-1)
    valid_left = (idx_left >= 0) & (idx_left < n)
    valid_right = (idx_right >= 0) & (idx_right < n)
    idx_left_c = np.clip(idx_left, 0, n-1)
    idx_right_c = np.clip(idx_right, 0, n-1)
    return idx_left_c, idx_right_c, valid_left, valid_right

def pair_search(h, n, max_time, nd_search=80):

    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    best_c5 = np.max(corr) * dx
    dx = 2.0 / n
    
    idx_left_c, idx_right_c, valid_left, valid_right = precompute_indices(n)
    
    G_left = g[idx_left_c] * valid_left.astype(np.float64)
    H_right = h[idx_right_c] * valid_right.astype(np.float64)
    
    # Q template: Q[k] for pair (a,b) is -2 at k=n-1, +1 at k=a-b+n-1, +1 at k=b-a+n-1
    # We'll compute Q on the fly since it depends on (a,b)
    
    while time.time() - start < max_time:
        improved = False
        for a in range(n):
            for b in range(n):
                if a == b:
                    continue
                low = max(-h[a], h[b] - 1.0)
                high = min(1.0 - h[a], h[b])
                if high - low < 1e-10:
                    continue
                
                B = G_left[a] - G_left[b] - H_right[a] + H_right[b]
                
                Q = np.zeros(2*n-1)
                Q[n-1] += -2.0
                Q[a - b + n - 1] += 1.0
                Q[b - a + n - 1] += 1.0
                
                # Grid search for best delta
                deltas = np.linspace(low, high, nd_search)
                new_corr = corr[None, :] + deltas[:, None] * B[None, :] + (deltas**2)[:, None] * Q[None, :]
                c5_vals = np.max(new_corr, axis=1) * dx

                if c5_vals[best_i] < best_c5 - 1e-10:
                    # Refine around best grid point
                    if best_i > 0 and best_i < nd_search - 1:
                        fine_low = deltas[best_i-1]
                        fine_high = deltas[best_i+1]
                    else:
                        fine_low = low
                        fine_high = high
                    fine_deltas = np.linspace(fine_low, fine_high, 40)
                    new_corr_f = corr[None, :] + fine_deltas[:, None] * B[None, :] + (fine_deltas**2)[:, None] * Q[None, :]
                    c5_f = np.max(new_corr_f, axis=1) * dx
                    fi = np.argmin(c5_f)
                    best_delta = fine_deltas[fi]
                    best_val = c5_f[fi]

                        H_right = h[idx_right_c] * valid_right.astype(np.float64)
                        improved = True
        
        if not improved:
            break
    
    return h, best_c5

    """Interpolate step function from n_old to n_new samples."""
    if n_old == n_new:
        return h.copy()
    # Old midpoints
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    # New midpoints
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    # Interpolate
    h_new = np.interp(x_new, x_old, h, left=h[0], right=h[-1])
    # Adjust to satisfy sum constraint: sum(h_new) should be n_new/2
    # Current sum:
    current_sum = np.sum(h_new)
    target_sum = n_new / 2.0
    # Adjust: add/subtract uniformly (clipping if needed)
    h_new += (target_sum - current_sum) / n_new
    h_new = np.clip(h_new, 0.0, 1.0)

    for _ in range(10):

                k0 = n - 1
                k1 = a - b + n - 1
                k2 = b - a + n - 1

                deltas = np.linspace(low, high, nd_grid)
                new_corr = corr[None, :] + deltas[:, None] * B[None, :]
                d2 = (deltas ** 2).ravel()
                new_corr[:, k0] += d2 * (-2.0)
                new_corr[:, k1] += d2 * 1.0
                new_corr[:, k2] += d2 * 1.0

                c5_vals = np.max(new_corr, axis=1) * dx
                best_i = np.argmin(c5_vals)

                if c5_vals[best_i] < best_c5 - 1e-11:
                    lo_r = max(low, deltas[max(0, best_i - 1)])
                    hi_r = min(high, deltas[min(nd_grid - 1, best_i + 1)])
                    if hi_r - lo_r < 1e-14:
                        continue
                    fd = np.linspace(lo_r, hi_r, nd_fine)
                    fc = corr[None, :] + fd[:, None] * B[None, :]
                    fd2 = (fd ** 2).ravel()
                    fc[:, k0] += fd2 * (-2.0)
                    fc[:, k1] += fd2 * 1.0
                    fc[:, k2] += fd2 * 1.0
                    fc5 = np.max(fc, axis=1) * dx
                    fi = np.argmin(fc5)
                    if fc5[fi] < best_c5 - 1e-11:
                        bd = fd[fi]
                        h[a] += bd
                        h[b] -= bd
                        if h[a] < 0.0:
                            h[a] = 0.0
                        elif h[a] > 1.0:
                            h[a] = 1.0
                        if h[b] < 0.0:
                            h[b] = 0.0
                        elif h[b] > 1.0:
                            h[b] = 1.0
                        g = 1.0 - h
                        corr = np.correlate(h, g, mode="full")
                        best_c5 = np.max(corr) * dx
                        D = g[idx_left_c] * vl - h[idx_right_c] * vr
                        improved = True

        if not improved:
            break

    return h, best_c5

def interpolate_h(h, n_old, n_new):
    if n_old == n_new:
        return h.copy()
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h, left=h[0], right=h[-1])
    target = n_new / 2.0
    for _ in range(200):
        diff = target - np.sum(h_new)
        if abs(diff) < 1e-14:
            break
        mask = (h_new > 1e-10) & (h_new < 1.0 - 1e-10)
        cnt = int(np.sum(mask))
        if cnt == 0:
            h_new += diff / n_new
            h_new = np.clip(h_new, 0.0, 1.0)
            break
        h_new[mask] += diff / cnt
        h_new = np.clip(h_new, 0.0, 1.0)
    diff = target - np.sum(h_new)
    if abs(diff) > 1e-13:
        free = np.where((h_new > 1e-8) & (h_new < 1.0 - 1e-8))[0]
        if len(free) > 0:
            h_new[free] += diff / len(free)
            h_new = np.clip(h_new, 0.0, 1.0)
    return h_new

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t_start = time.time()

    h0 = np.array(initial_h_values, dtype=np.float64)
    n0 = len(h0)
    corr0 = np.correlate(h0, 1.0 - h0, mode="full")
    best_c5 = np.max(corr0) * (2.0 / n0)
    best_h = h0.copy()

    # Phase 1: Optimize at original resolution
    elapsed = time.time() - t_start
    t1 = max(60, int((budget_s - elapsed) * 0.35))
    if t1 > 20:
        h_opt, c5_opt = pair_search(h0.copy(), max_time=t1, nd_grid=80, nd_fine=40)
        if c5_opt < best_c5:
            best_h = h_opt.copy()
            best_c5 = c5_opt

    # Phase 2: Try higher resolutions
    resolutions = [64, 80, 100, 128]
    elapsed = time.time() - t_start
    remaining = budget_s - elapsed - 30
    if remaining > 60:
        t_per = remaining / (len(resolutions) + 1)
        for n_new in resolutions:
            if time.time() - t_start > budget_s - 40:
                break
            h_interp = interpolate_h(best_h, len(best_h), n_new)
            t_left = min(t_per, budget_s - (time.time() - t_start) - 30)
            if t_left < 15:
                break
            h_opt, c5_opt = pair_search(h_interp, max_time=t_left, nd_grid=60, nd_fine=30)
            if c5_opt < best_c5:
                best_h = h_opt.copy()
                best_c5 = c5_opt

    # Phase 3: Final refinement at best resolution found
    elapsed = time.time() - t_start
    if budget_s - elapsed > 40:
        t_left = budget_s - elapsed - 10
        nd_g = 100 if len(best_h) <= 80 else 70
        nd_f = 50 if len(best_h) <= 80 else 35
        h_opt, c5_opt = pair_search(best_h.copy(), max_time=t_left, nd_grid=nd_g, nd_fine=nd_f)
        if c5_opt < best_c5:
            best_h = h_opt.copy()
            best_c5 = c5_opt

    # Final verification
    n_final = len(best_h)
    dx_final = 2.0 / n_final
    corr_final = np.correlate(best_h, 1.0 - best_h, mode="full")
    final_c5 = float(np.max(corr_final) * dx_final)

    return best_h, final_c5, n_final
