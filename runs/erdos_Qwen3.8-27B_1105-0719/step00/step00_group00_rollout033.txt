    grad[i, k] = d(corr[indices[i]] * dx) / d(h[k])

```python
def coord_descent_step(h, corr, k):
    """Optimize h[k] keeping all other values fixed."""
    n = len(h)
    dx = 2.0 / n
    
    # For each correlation s, find the coefficient a_s and intercept b_s
    # such that corr[s] = a_s * h[k] + b_s
    # corr[s] = sum_i h[i]*(1-h[i-lag])
    # Terms involving h[k]:
    #   When i=k: h[k]*(1-h[k-lag])
    #   When i-lag=k (i.e., i=k+lag): h[k+lag]*(1-h[k])
    
    a = np.zeros(2*n-1)
    b = np.zeros(2*n-1)
    
    for s in range(2*n-1):
        lag = s - (n-1)
        # Term where i=k, contributing to lag: h[k]*(1-h[k-lag])
        # k-lag must be valid: 0 <= k-lag < n
        if 0 <= k-lag < n:
            a[s] += (1 - h[k-lag])
            # b[s] gets 0 from this term (it's all in the h[k] part)
        # Term where i=k+lag, contributing: h[k+lag]*(1-h[k])
        if 0 <= k+lag < n:
            a[s] += -h[k+lag]
            b[s] += h[k+lag]
    
    # Now corr[s] = a[s]*h[k] + (corr[s] - a[s]*h[k]_current)
    # Actually: corr[s]_current = a[s]*h[k]_current + b[s]_other
    # So b[s]_other = corr[s]_current - a[s]*h[k]_current
    # Wait, I already computed a[s] as the coefficient of h[k]
    # corr[s] = a[s]*h[k] + (everything else)
    # (everything else) = corr[s]_current - a[s]*h[k]_current
    
    intercept = corr - a * h[k]
    
    # Now minimize max_s (a[s]*x + intercept[s]) over x in [0,1]
    # (also respecting the box constraint on h[k])
    
    # The function f(x) = max_s (a[s]*x + intercept[s]) is convex in x
    # Minimum is where the subgradient contains 0, or at a boundary
    
    # For a 1D convex piecewise linear function, we can use bisection on the subgradient
    lo, hi = 0.0, 1.0
    
    for _ in range(50):
        mid = (lo + hi) / 2
        # Subgradient at mid: set of a[s] where a[s]*mid + intercept[s] = max
        vals = a * mid + intercept
        max_val = np.max(vals)
        # Subgradient is [min(a[s] where val=s is active), max(a[s] where val=s is active)]
        active = np.where(np.abs(vals - max_val) < 1e-10)[0]
        if len(active) == 0:
            active = np.where(vals == max_val)[0]
        sg_min = a[active].min()
        sg_max = a[active].max()
        if sg_min > 0:
            hi = mid  # function is increasing, move left
        elif sg_max < 0:
            lo = mid  # function is decreasing, move right
        else:
            lo = hi = mid  # 0 is in subgradient
            break
    
    x_opt = (lo + hi) / 2
    return x_opt
```

    corr = np.correlate(h, g, mode='full')

    n = len(h)
    dx = 2.0 / n

    for i, s in enumerate(indices):
        lag = s - (n - 1)

                val += (1.0 - h[j1])
            j2 = k + lag

                val -= h[j2]
            g[k] = val * dx
        grads[i] = g
    return grads

def active_set_lp(h, corr, top_k, eps):
    n = len(h)
    sorted_idx = np.argsort(-corr)
    active = sorted_idx[:top_k]
    
    G = gradient_matrix(h, active)
    current_vals = corr[active]
    
    lower = np.maximum(-eps, -h + 1e-12)
    upper = np.minimum(eps, 1.0 - h - 1e-12)
    
    if np.any(lower >= upper):
        # Tighten eps
        mask = lower < upper
        if not np.any(mask):
            return None
        lower = np.where(mask, lower, -0.001)
        upper = np.where(mask, upper, 0.001)
        if np.any(lower >= upper):
            return None
    
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    A_ub = np.hstack([G, -np.ones((top_k, 1))])
    b_ub = -current_vals + 1e-10  # small slack
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = [0.0]
    
    bounds = [(lower[k], upper[k]) for k in range(n)] + [(-1.0, 1.0)]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'maxiter': 500})
    except:
        return None
    
    if res.success and res.fun < -1e-12:  # negative means improvement
        delta = res.x[:n]
        if np.max(np.abs(delta)) > 1e-10:
            return delta
    return None

def grad_corr_all(h):
    """Compute gradient of all correlations w.r.t. h. Returns array of shape (2n-1, n)."""
    n = len(h)
    dx = 2.0 / n
    grads = np.zeros((2*n-1, n))
    for s in range(2*n-1):
        lag = s - (n-1)
        # Part 1: (1-h[k-lag]) for valid k
        p1 = np.zeros(n)
        if lag >= 0:
            # k-lag >= 0 means k >= lag
            p1[lag:] = (1 - h[:-lag]) if lag > 0 else (1 - h)
        else:
            # k-lag >= 0 means k >= 0 (always true for k>=0)
            # k-lag < n means k < n+lag = n-|lag|
            p1[:n+lag] = (1 - h[-lag:])
        # Part 2: -h[k+lag] for valid k
        p2 = np.zeros(n)
        if lag >= 0:
            # k+lag < n means k < n-lag
            p2[:n-lag] = -h[lag:]
        else:
            # k+lag >= 0 means k >= -lag
            p2[-lag:] = -h[:n+lag]
        grads[s] = (p1 + p2) * dx
    return grads

            if ok:
                h = h_new
                c5, corr = compute_c5(h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
                stag = 0
            else:
                stag += 1
        else:
            stag += 1
        if stag >= 6:
            h_sub = subgradient_step(h, corr, n)
            v_sub, _ = compute_c5(h_sub)
            if v_sub < best_c5:
                best_c5 = v_sub
                best_h = h_sub.copy()
                h = h_sub
                c5, corr = compute_c5(h)
            else:
                kick = rng.randn(n) * 0.012
                kick -= kick.mean()
                ht = np.clip(h + kick, 0, 1)
                ht = fix_sum(ht, n)
                v_try, _ = compute_c5(ht)
                if v_try <= c5:
                    h = ht
                    c5, corr = compute_c5(h)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h.copy()
            stag = 0
            top_k = int(rng.choice([5, 7, 8, 10, 12, 15]))
            eps = float(rng.uniform(0.015, 0.09))
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(45, 0.5)
    h0 = fix_sum(h0, len(h0))
    n0 = len(h0)
    t_phase1 = min(550, budget_s - 100)
    h1, c5_1 = optimize(h0, t_phase1, seed=seed)
    best_h = h1.copy()
    best_c5 = c5_1
    best_n = n0
    remaining = budget_s - (time.time() - start_time) - 30
    n_tries = [44, 46, 48, 50, 55, 60, 40]
    for nv in n_tries:
        if time.time() - start_time > budget_s - 30:
            break
        t_alloc = max(0, remaining / max(1, len(n_tries)))
        if t_alloc < 15:
            break
        x_src = np.linspace(0, 2, len(best_h))
        x_dst = np.linspace(0, 2, nv)
        h_init = np.interp(x_dst, x_src, best_h)
        h_init = fix_sum(h_init, nv)
        h_try, c5_try = optimize(h_init, min(t_alloc, 180), seed=seed + nv)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()
            best_n = nv
    remaining = budget_s - (time.time() - start_time) - 10
    if remaining > 40:
        h_fin, c5_fin = optimize(best_h, remaining, seed=seed + 7777)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin
    best_h = fix_sum(best_h, len(best_h))
    final_c5, _ = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
