def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * dx)

def lse_objective(z, beta, n):
    h = np.empty(n)
    h[:n-1] = z
    h[n-1] = n / 2.0 - z.sum()
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = corr.max()
    s = beta * (corr - mc)
    return mc + np.log(np.sum(np.exp(s))) / beta

def lse_gradient(z, beta, n):
    h = np.empty(n)
    h[:n-1] = z
    h[n-1] = n / 2.0 - z.sum()
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = corr.max()
    s = beta * (corr - mc)
    e = np.exp(s)
    w = e / e.sum()
    gh = np.zeros(n)
    nm1 = n - 1
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        m = k - nm1
        if m >= 0:
            a, b = 0, n - m
        else:
            a, b = -m, n
        if b > a:
            gh[a:b] += wk * (1.0 - h[a + m:b + m])
        if m >= 0:
            a2, b2 = m, n
        else:
            a2, b2 = 0, n + m
        if b2 > a2:
            gh[a2:b2] -= wk * h[a2 - m:b2 - m]
    gz = gh[:n-1] - gh[n-1]
    return gz

def project_feasible(y, n):
    target = n / 2.0
    lo = y.min() - 1.0
    hi = y.max() + 1.0

    return h

    elif s < n / 2.0 - 1.0:
        p += pw * (n / 2.0 - 1.0 - s) ** 2
    return base + p
