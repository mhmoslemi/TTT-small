import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def compute_correlations(h):
    return np.correlate(h, 1 - h, mode='full')

def lse_smooth(corr, beta):
    max_c = np.max(corr)
    if beta * max_c > 500:
        return max_c  # avoid overflow
    return max_c + np.log(np.sum(np.exp(beta * (corr - max_c)))) / beta

def lse_gradient(corr, beta):
    max_c = np.max(corr)
    if beta * max_c > 500:
        # Hard max
        grad = np.zeros_like(corr)
        grad[np.argmax(corr)] = 1.0
        return grad
    shifted = beta * (corr - max_c)
    exp_shifted = np.exp(shifted)
    weights = exp_shifted / np.sum(exp_shifted)
    return weights  # These are the weights for the gradient

def reduced_objective(z, n, beta, penalty_weight):
    h = np.zeros(n)
    h[:n-1] = z
    h[n-1] = n / 2.0 - np.sum(z)
    # Penalty for out-of-bounds
    penalty = 0.0
    if h[n-1] < 0:
        penalty += penalty_weight * h[n-1] ** 2
    elif h[n-1] > 1:
        penalty += penalty_weight * (h[n-1] - 1) ** 2
    h_clipped = np.clip(h, 0, 1)
    corr = np.correlate(h_clipped, 1 - h_clipped, mode='full')
    return lse_smooth(corr, beta) + penalty

def reduced_gradient(z, n, beta, penalty_weight):
    h = np.zeros(n)
    h[:n-1] = z
    h[n-1] = n / 2.0 - np.sum(z)
    h_clipped = np.clip(h, 0, 1)
    
    corr = np.correlate(h_clipped, 1 - h_clipped, mode='full')
    weights = lse_gradient(corr, beta)  # softmax weights over lags
    
    # Compute dc_k/dh_j for all j
    # For each lag k and position j:
    # dc_k/dh_j = (1-h[j-k]) * [valid_as_first] - h[j+k] * [valid_as_as_second]
    
    n_corr = len(corr)  # 2n-1
    # lag k = m - (n-1) for m = 0, ..., 2n-2
    
    # Gradient of LSE w.r.t. h
    grad_h = np.zeros(n)
    for m in range(n_corr):

        w = weights[m]
        if w < 1e-15:

        # For lag k: c_k = sum valid h[i+k]*(1-h[i])
        if k >= 0:

                grad_h[i] += -w * h_clipped[j]
