def project_box_sum(y, target):
    y = np.asarray(y,dtype=float)
    n=y.size
    if n==0: return y.copy()
    lo = float(np.min(y)) - 2.0
    hi = float(np.max(y)) + 2.0
    mid = 0.5*(lo+hi)
    for it in range(80):
       mid=0.5*(lo+hi)
       val=np.clip(y-mid,0.0,1.0)
       s=float(np.sum(val))
       if s > target:
          lo=mid
       else:
          hi=mid
    val=np.clip(y-0.5*(lo+hi),0.0,1.0)
    diff = target - float(np.sum(val))
    if diff > 0:
       if diff > 1e-12:
          idx = int(np.argmax(1.0-val))
          val[idx] = min(1.0, val[idx]+diff)
    elif diff < 0:
       if diff < -1e-12:
          idx = int(np.argmax(val))
          val[idx] = max(0.0, val[idx]+diff)
    return val

def repair_feasible(vec):
    x = np.asarray(vec, dtype=float).ravel().copy()
    n = x.size
    if n == 0:
        return x
    x = np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0)
    x = np.clip(x, 0.0, 1.0)
    target = 0.5 * n
    diff = target - float(np.sum(x))

    if diff > 0.0:
        order = np.argsort(-(1.0 - x))
        rem = diff
        for idx in order:
            if rem <= 1e-15:
                break
            cap = 1.0 - float(x[idx])
            if cap > 0.0:
                add = cap if cap < rem else rem
                x[idx] += add
                rem -= add
    elif diff < 0.0:
        order = np.argsort(-x)
        rem = -diff
        for idx in order:
            if rem <= 1e-15:
                break
            cap = float(x[idx])
            if cap > 0.0:
                sub = cap if cap < rem else rem
                x[idx] -= sub
                rem -= sub

    x = np.clip(x, 0.0, 1.0)
    diff2 = target - float(np.sum(x))
    if diff2 > 1e-14:
        idx = int(np.argmax(1.0 - x))
        x[idx] = min(1.0, float(x[idx]) + diff2)
    elif diff2 < -1e-14:
        idx = int(np.argmax(x))
        x[idx] = max(0.0, float(x[idx]) + diff2)

    x = np.clip(x, 0.0, 1.0)
    return x


def evaluate_exact(h):
    h = np.asarray(h, dtype=float).ravel()
    n = h.size
    if n == 0:
        return float("inf")
    if not np.all(np.isfinite(h)):
        return float("inf")
    dx = 2.0 / n
    vals = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(vals))


def all_lags(n):
    return np.arange(-(n - 1), n, dtype=int)


def profile_lags(h, lags):
    h = np.asarray(h, dtype=float).ravel()
    n = h.size
    out = np.empty(len(lags), dtype=float)
    dx = 2.0 / n
    for idx, k in enumerate(lags):
        k = int(k)
        if k >= 0:
            if k == 0:
                val = float(np.dot(h, 1.0 - h))
            else:
                val = float(np.dot(h[k:], 1.0 - h[:-k]))
        else:
            kk = -k
            val = float(np.dot(h[:-kk], 1.0 - h[kk:]))
        out[idx] = val
    out *= dx
    return out


def gradient_lags(h, lags):
    h = np.asarray(h, dtype=float).ravel()
    n = h.size
    m = len(lags)
    G = np.zeros((m, n), dtype=float)
    dx = 2.0 / n
    for idx, k in enumerate(lags):
        k = int(k)
        if k == 0:
            G[idx, :] = dx * (1.0 - 2.0 * h)
        elif k > 0:
            G[idx, k:] += dx * (1.0 - h[:-k])
            G[idx, :n - k] += -dx * h[k:]
        else:
            kk = -k
            G[idx, :n - kk] += dx * (1.0 - h[kk:])
            G[idx, kk:] += -dx * h[:-kk]
    return G


def resample_step(h, new_n):
    h = np.asarray(h, dtype=float).ravel()
    n = h.size
    new_n = int(new_n)
    if new_n == n:
        return h.copy()
    if n == 0 or new_n <= 0:
        return np.full(new_n, 0.5)

    dxs = 2.0 / n
    dxn = 2.0 / new_n
    xs = np.arange(n + 1, dtype=float) * dxs
    cum = np.empty(n + 1, dtype=float)
    cum[0] = 0.0
    cum[1:] = np.cumsum(h) * dxs

    left = np.arange(new_n, dtype=float) * dxn
    right = left + dxn
    vl = np.interp(left, xs, cum, left=cum[0], right=cum[-1])
    vr = np.interp(right, xs, cum, left=cum[0], right=cum[-1])
    out = (vr - vl) / dxn
    out = np.clip(out, 0.0, 1.0)
    return repair_feasible(out)


def sym_h(h):

def random_perturb(h, rng, eps):

    if eps == 0.0:
        return repair_feasible(h)
    n = h.size
    z = rng.standard_normal(n)
    z -= float(np.mean(z))
    return repair_feasible(h + eps * z)


def add_dimension(d, dims, min_n=20, max_n=200):
    try:
        d = int(d)
    except Exception:
        return
    if d < min_n or d > max_n:
        return
    if d not in dims:
        dims.append(d)


def consider_candidate(h_cand, best_h, best_c):
    if h_cand is None:
        return best_h, best_c, len(best_h)
    hh = repair_feasible(np.asarray(h_cand, dtype=float))
    nn = len(hh)
    if nn == 0 or not np.all(np.isfinite(hh)):
        return best_h, best_c, len(best_h)
    cc = evaluate_exact(hh)
    if not np.isfinite(cc):
        return best_h, best_c, len(best_h)
    if cc < best_c - 1e-13:
        return hh, cc, nn
    if abs(cc - best_c) <= 1e-14 and nn < len(best_h):
        return hh, cc, nn
    return best_h, best_c, len(best_h)


def blend_consider(h1, h2, best_h, best_c):
    if h1 is None or h2 is None:
        return best_h, best_c, len(best_h)
    h1 = np.asarray(h1, dtype=float).ravel()
    h2 = np.asarray(h2, dtype=float).ravel()
    if h1.size != h2.size:
        return best_h, best_c, len(best_h)
    for k in range(21):
        alpha = float(k) / 20.0
        hq = (1.0 - alpha) * h1 + alpha * h2
        best_h, best_c, _ = consider_candidate(hq, best_h, best_c)
    return best_h, best_c, len(best_h)


def make_bounds(n, t_lower, t_upper):
    bounds = []
    for _ in range(n):
        bounds.append((0.0, 1.0))
    bounds.append((t_lower, t_upper))
    return bounds


def obj_slsqp(x):
    h = x[:-1]
    return float(x[-1] + _OPT_BETA * np.dot(h, 1.0 - h))


def obj_slsqp_jac(x):
    h = x[:-1]
    n = _OPT_N
    g = np.zeros(n + 1, dtype=float)
    if _OPT_BETA != 0.0:
        g[:n] = _OPT_BETA * (1.0 - 2.0 * h)
    g[n] = 1.0
    return g


def eq_slsqp(x):
    return float(np.sum(x[:-1]) - 0.5 * _OPT_N)


def eq_slsqp_jac(x):
    n = _OPT_N
    g = np.zeros(n + 1, dtype=float)
    g[:n] = 1.0
    return g


def ineq_slsqp(x):
    h = x[:-1]
    return x[-1] - profile_lags(h, _OPT_LAGS)


def ineq_slsqp_jac(x):
    h = x[:-1]
    n = _OPT_N
    m = len(_OPT_LAGS)
    G = gradient_lags(h, _OPT_LAGS)
    J = np.empty((m, n + 1), dtype=float)
    J[:, :n] = -G
    J[:, n] = 1.0
    return J


def select_active_lags(h, max_count=0, margin=1e-4):
    h = np.asarray(h, dtype=float).ravel()
    n = h.size
    lags = all_lags(n)
    m = 2 * n - 1
    if max_count <= 0:
        max_count = min(m, max(60, int(0.45 * m)))
    prof = profile_lags(h, lags)
    score = float(np.max(prof))
    order = np.argsort(-prof)
    top = order[:max_count]
    mask = prof >= (score - margin)
    if np.any(mask):
        active = np.unique(np.concatenate((top, np.where(mask)[0])))
    else:
        active = top
    if len(active) > max_count:
        active = top
    return lags[active]


def solve_epigraph(h0, lags, beta, maxiter, ftol):
    global _OPT_N, _OPT_DX, _OPT_BETA, _OPT_LAGS

    h0 = repair_feasible(np.asarray(h0, dtype=float).ravel())
    n = h0.size
    if n == 0:
        return None
    lags = np.asarray(lags, dtype=int).ravel()
    if lags.size == 0:
        return h0

    _OPT_N = n
    _OPT_DX = 2.0 / n
    _OPT_BETA = float(beta)
    _OPT_LAGS = lags

    t0 = evaluate_exact(h0)
    if not np.isfinite(t0):
        return None

    t_upper = t0 + 0.25
    x0 = np.empty(n + 1, dtype=float)
    x0[:n] = h0
    x0[n] = t0 + 1e-7

    bounds = make_bounds(n, 0.0, t_upper)
    constraints = [
        {"type": "eq", "fun": eq_slsqp, "jac": eq_slsqp_jac},
        {"type": "ineq", "fun": ineq_slsqp, "jac": ineq_slsqp_jac},
    ]

    try:
        res = so.minimize(
            obj_slsqp,
            x0,
            jac=obj_slsqp_jac,
            bounds=bounds,
            constraints=constraints,
            method="SLSQP",
            options={"maxiter": int(maxiter), "ftol": float(ftol), "disp": False},
        )
    except Exception:
        return None

    if res is None or getattr(res, "x", None) is None:
        return None
    return np.asarray(res.x[:-1], dtype=float)


def pair_value(F, B, C, x):
    return float(np.max(F + B * x + C * x * x))


def make_pair_C(n, dx, d):
    C = np.zeros(2 * n - 1, dtype=float)
    if d > 0:
        base = n - 1
        C[base + d] = dx
        C[base - d] = dx
    return C


def pair_move_x_cached(h, i, j, F, G, C):
    lo = max(-float(h[i]), float(h[j]) - 1.0)
    hi = min(float(h[j]), 1.0 - float(h[i]))
    if hi <= lo + 1e-14:
        return 0.0

    B = G[:, i] - G[:, j]
    a = lo
    b = hi
    phi = 0.6180339887498949

    x1 = b - phi * (b - a)
    x2 = a + phi * (b - a)
    f1 = pair_value(F, B, C, x1)
    f2 = pair_value(F, B, C, x2)

    best_x = 0.0
    best_f = float(np.max(F))

    for _ in range(55):
        if f1 <= f2:
            b = x2
            x2 = x1
            f2 = f1
            x1 = b - phi * (b - a)
            f1 = pair_value(F, B, C, x1)
        else:
            a = x1
            x1 = x2
            f1 = f2
            x2 = a + phi * (b - a)
            f2 = pair_value(F, B, C, x2)

    xmid = 0.5 * (a + b)
    for x in (xmid, x1, x2, 0.0, lo, hi):
        if x < lo - 1e-12 or x > hi + 1e-12:
            continue
        xx = float(np.clip(x, lo, hi))
        f = pair_value(F, B, C, xx)
        if f < best_f - 1e-14:
            best_f = f
            best_x = xx
    return best_x


def pair_sweep(h0, active_tol=1e-3, max_sweeps=20, candidate_limit=400,
               test_all=False, deadline=0.0):
    h = repair_feasible(np.asarray(h0, dtype=float).ravel().copy())
    n = h.size
    if n < 2:
        return h, evaluate_exact(h)

    lags = all_lags(n)
    dx = 2.0 / n
    best_h = h.copy()
    best_c = evaluate_exact(h)

    C_cache = {}
    for d in range(1, n):
        C_cache[d] = make_pair_C(n, dx, d)

    for _ in range(int(max_sweeps)):
        if deadline and time.perf_counter() > deadline:
            break

        F = profile_lags(best_h, lags)
        score = float(np.max(F))
        G = gradient_lags(best_h, lags)

        active = F >= (score - active_tol)
        if not np.any(active):
            active = np.array([int(np.argmax(F))], dtype=int)
        G_act = G[active, :]

        plus = []
        minus = []
        for i in range(n - 1):
            Gi = G_act[:, i]
            for j in range(i + 1, n):
                b = Gi - G_act[:, j]
                mb = float(np.max(b))
                mnb = float(np.min(b))
                plus.append((mb, i, j))
                minus.append((-mnb, i, j))

        plus.sort()
        minus.sort()

        if test_all:
            limit_each = len(plus)
        else:
            limit_each = max(20, int(candidate_limit) // 2)

        tried = set()
        best_pair_c = best_c
        best_pair_h = best_h

        count = 0
        for _mb, i, j in plus:
            if count >= limit_each:
                break
            key = (i, j)
            if key in tried:
                continue
            tried.add(key)
            count += 1
            if deadline and time.perf_counter() > deadline:
                return best_h, best_c

            x = pair_move_x_cached(best_h, i, j, F, G, C_cache[abs(i - j)])
            if abs(x) > 1e-12:
                h_try = best_h.copy()
                h_try[i] += x
                h_try[j] -= x
                h_try = repair_feasible(h_try)
                c_try = evaluate_exact(h_try)
                if np.isfinite(c_try) and c_try < best_pair_c - 1e-13:
                    best_pair_c = c_try
                    best_pair_h = h_try

        count = 0
        for _nm, i, j in minus:
            if count >= limit_each:
                break
            key = (i, j)
            if key in tried:
                continue
            tried.add(key)
            count += 1
            if deadline and time.perf_counter() > deadline:
                return best_h, best_c

            x = pair_move_x_cached(best_h, i, j, F, G, C_cache[abs(i - j)])
            if abs(x) > 1e-12:
                h_try = best_h.copy()
                h_try[i] += x
                h_try[j] -= x
                h_try = repair_feasible(h_try)
                c_try = evaluate_exact(h_try)
                if np.isfinite(c_try) and c_try < best_pair_c - 1e-13:
                    best_pair_c = c_try
                    best_pair_h = h_try

        if best_pair_c < best_c - 1e-13:
            best_h = best_pair_h
            best_c = best_pair_c
        else:
            break

    return best_h, best_c


def project_box_sum(y, target):
    y = np.asarray(y, dtype=float).ravel()
    n = y.size
    if n == 0:
        return y.copy()

    lo = float(np.min(y)) - 2.0
    hi = float(np.max(y)) + 2.0
    mid = 0.5 * (lo + hi)

    for _ in range(80):
        mid = 0.5 * (lo + hi)
        val = np.clip(y - mid, 0.0, 1.0)
        s = float(np.sum(val))
        if s > target:
            lo = mid
        else:
            hi = mid

    val = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - float(np.sum(val))
    if diff > 1e-12:
        idx = int(np.argmax(1.0 - val))
        val[idx] = min(1.0, float(val[idx]) + diff)
    elif diff < -1e-12:
        idx = int(np.argmax(val))
        val[idx] = max(0.0, float(val[idx]) + diff)
    return val


def smooth_gradient(h, lags, beta):
    h = np.asarray(h, dtype=float).ravel()
    F = profile_lags(h, lags)
    m = float(np.max(F))
    z = np.exp(float(beta) * (F - m))
    s = float(np.sum(z))
    if not np.isfinite(s) or s <= 0.0:
        return np.zeros(h.size, dtype=float)
    w = z / s
    G = gradient_lags(h, lags)
    return float(beta) * np.dot(w, G)


def smooth_optimize(h0, lags, beta, max_iters, steps, deadline):
    h = repair_feasible(np.asarray(h0, dtype=float).ravel().copy())
    n = h.size
    if n == 0:
        return h, float("inf")
    target = 0.5 * n
    best_h = h.copy()
    best_c = evaluate_exact(h)

    for _ in range(int(max_iters)):
        if deadline and time.perf_counter() > deadline:
            break
        grad = smooth_gradient(h, lags, beta)
        if not np.all(np.isfinite(grad)):
            break
        improved = False
        for alpha in steps:
            y = h - alpha * grad
            hn = repair_feasible(project_box_sum(y, target))
            c = evaluate_exact(hn)
            if np.isfinite(c) and c < best_c - 1e-13:
                h = hn
                best_h = hn
                best_c = c
                improved = True
                break
        if not improved:
            break

    return best_h, best_c


def choose_slsqp_maxiter(n, remaining):
    if remaining < 30:
        return 5
    if remaining < 70:
        return 12
    if remaining < 150:
        return 25
    if n <= 80:
        return 70
    if n <= 110:
        return 45
    return 30


def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.perf_counter()
    total_budget = max(30, int(float(budget_s)))
    deadline = start_time + total_budget - 20
    rng = np.random.default_rng(seed)

    parent = get_parent_h()
    if parent is None:
        best_h = repair_feasible(np.full(70, 0.5))
    else:
        best_h = repair_feasible(parent)

    best_c = evaluate_exact(best_h)
    if not np.isfinite(best_c):
        best_h = repair_feasible(np.full(70, 0.5))
        best_c = evaluate_exact(best_h)

    best_h, best_c, _ = consider_candidate(sym_h(best_h), best_h, best_c)
    best_h, best_c, _ = consider_candidate(complement_h(best_h), best_h, best_c)
    best_h, best_c, _ = consider_candidate(sym_h(complement_h(best_h)), best_h, best_c)

    if best_c <= _TARGET_C5 + 1e-12:
        best_h = repair_feasible(best_h)
        best_c = evaluate_exact(best_h)
        return best_h, float(best_c), len(best_h)

    base_n = len(best_h)
    dims = []
    add_dimension(base_n, dims)
    add_dimension(base_n + 10, dims)
    add_dimension(base_n + 20, dims)
    if base_n <= 90:
        add_dimension(int(round(base_n * 2.0)), dims)
    add_dimension(base_n - 10, dims, 40, 200)
    add_dimension(base_n - 20, dims, 40, 200)
    if len(dims) > 5:
        dims = dims[:5]

    for dim_idx, n_dim in enumerate(dims):
        if time.perf_counter() > deadline:
            break
        if best_c <= _TARGET_C5 + 1e-12:
            break

        if len(best_h) == n_dim:
            h_dim = best_h.copy()
        else:
            h_dim = resample_step(best_h, n_dim)
        h_dim = repair_feasible(h_dim)
        best_h, best_c, _ = consider_candidate(h_dim, best_h, best_c)

        starts = [h_dim, sym_h(h_dim), complement_h(h_dim)]
        if dim_idx == 0:
            for eps in (0.005, 0.01, 0.02, 0.05):
                starts.append(random_perturb(h_dim, rng, eps))
        else:
            starts.append(random_perturb(h_dim, rng, 0.02))

        allowed_excess = 0.08 if dim_idx == 0 else 0.04

        for hs in starts:
            if time.perf_counter() > deadline:
                break
            hs = repair_feasible(hs)
            c_hs = evaluate_exact(hs)
            if not np.isfinite(c_hs) or c_hs > best_c + allowed_excess:
                continue

            n_hs = len(hs)
            if n_hs <= 90:
                lags_use = all_lags(n_hs)
            else:
                lags_use = select_active_lags(hs)

            remaining = deadline - time.perf_counter()
            if remaining < 35:
                break

            maxit = choose_slsqp_maxiter(n_hs, remaining)
            h_sol = solve_epigraph(hs, lags_use, 1e-8, maxit, 1e-10)

            if h_sol is not None:
                best_h, best_c, _ = consider_candidate(h_sol, best_h, best_c)
                best_h, best_c, _ = blend_consider(hs, h_sol, best_h, best_c)
                best_h, best_c, _ = consider_candidate(sym_h(h_sol), best_h, best_c)

                if evaluate_exact(h_sol) < c_hs - 1e-12:
                    if time.perf_counter() > deadline:
                        break
                    if n_hs <= 90:
                        lags2 = all_lags(n_hs)
                    else:
                        lags2 = select_active_lags(h_sol)
                    rem2 = deadline - time.perf_counter()
                    if rem2 > 50:
                        h_sol2 = solve_epigraph(h_sol, lags2, 1e-5,
                                                choose_slsqp_maxiter(n_hs, rem2), 1e-10)
                        if h_sol2 is not None:
                            best_h, best_c, _ = consider_candidate(h_sol2, best_h, best_c)
                            best_h, best_c, _ = blend_consider(h_sol, h_sol2, best_h, best_c)
                            best_h, best_c, _ = consider_candidate(sym_h(h_sol2), best_h, best_c)

    if best_c <= _TARGET_C5 + 1e-12:
        best_h = repair_feasible(best_h)
        best_c = evaluate_exact(best_h)
        return best_h, float(best_c), len(best_h)

    if time.perf_counter() < deadline:
        if len(best_h) <= 80:
            h_pair, c_pair = pair_sweep(best_h, 1e-3, 20, 500, False, deadline)
        else:
            h_pair, c_pair = pair_sweep(best_h, 1e-3, 15, 300, False, deadline)
        best_h, best_c, _ = consider_candidate(h_pair, best_h, best_c)

    if time.perf_counter() < deadline and len(best_h) <= 80 and best_c > _TARGET_C5 + 1e-12:
        h_pair2, c_pair2 = pair_sweep(best_h, 5e-4, 5, 500, True, deadline)
        best_h, best_c, _ = consider_candidate(h_pair2, best_h, best_c)

    if time.perf_counter() < deadline and len(best_h) <= 120 and best_c > _TARGET_C5 + 1e-12:
        h_pair3, c_pair3 = pair_sweep(best_h, 1e-5, 10, 300, False, deadline)
        best_h, best_c, _ = consider_candidate(h_pair3, best_h, best_c)

    if time.perf_counter() < deadline - 30 and best_c > _TARGET_C5 + 1e-12:
        for beta in (80.0, 150.0):
            if time.perf_counter() > deadline:
                break
            h_sm, c_sm = smooth_optimize(
                best_h,
                all_lags(len(best_h)),
                beta,
                80,
                (0.02, 0.01, 0.005, 0.002),
                deadline,
            )
            best_h, best_c, _ = consider_candidate(h_sm, best_h, best_c)
            if time.perf_counter() < deadline:
                h_p, c_p = pair_sweep(best_h, 1e-3, 5, 200, False, deadline)
                best_h, best_c, _ = consider_candidate(h_p, best_h, best_c)
            if best_c <= _TARGET_C5 + 1e-12:
                break

    if time.perf_counter() < deadline - 80 and best_c > _TARGET_C5 + 1e-12:
        for eps in (0.01, 0.02, 0.05):
            if time.perf_counter() > deadline:
                break
            hs = random_perturb(best_h, rng, eps)
            best_h, best_c, _ = consider_candidate(hs, best_h, best_c)
            n_hs = len(hs)
            if n_hs <= 90:
                lags_use = all_lags(n_hs)
            else:
                lags_use = select_active_lags(hs)
            remaining = deadline - time.perf_counter()
            if remaining < 50:
                continue
            h_sol = solve_epigraph(hs, lags_use, 1e-8, choose_slsqp_maxiter(n_hs, remaining), 1e-10)
            if h_sol is not None:
                best_h, best_c, _ = consider_candidate(h_sol, best_h, best_c)
                best_h, best_c, _ = blend_consider(hs, h_sol, best_h, best_c)
            if time.perf_counter() < deadline:
                h_p, c_p = pair_sweep(best_h, 1e-3, 5, 200, False, deadline)
                best_h, best_c, _ = consider_candidate(h_p, best_h, best_c)
            if best_c <= _TARGET_C5 + 1e-12:
                break

    if time.perf_counter() < deadline:
        h_p, c_p = pair_sweep(best_h, 1e-4, 8, 300, False, deadline)
        best_h, best_c, _ = consider_candidate(h_p, best_h, best_c)

    best_h = repair_feasible(best_h)
    best_c = evaluate_exact(best_h)
    return best_h, float(best_c), len(best_h)
