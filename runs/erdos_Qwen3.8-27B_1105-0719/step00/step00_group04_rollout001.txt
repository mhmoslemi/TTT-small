def project_feasible(v, target_sum):
    """Project v onto {0<=x<=1, sum(x)=target_sum} using bisection on shift."""
    n = len(v)
    v = np.array(v, dtype=np.float64).copy()
    v = np.clip(v, 0.0, 1.0)
    current_sum = float(np.sum(v))
    if abs(current_sum - target_sum) < 1e-14:
        return v
    
    # Bisection: find lambda such that sum(clip(v + lambda, 0, 1)) = target_sum
    lo = max(0.0, -min(v)) - 1.0  # guarantees sum < target (shift down a lot)
    hi = 1.0 - min(v) + 1.0  # guarantees sum > target (shift up a lot)
    
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v + mid, 0.0, 1.0)))
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    
    result = np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)
    
    # Fix any residual
    diff = target_sum - float(np.sum(result))
    if abs(diff) > 1e-13:

    return result
