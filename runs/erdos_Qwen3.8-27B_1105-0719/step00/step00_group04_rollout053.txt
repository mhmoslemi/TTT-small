        h = np.clip(h, 0.0, 1.0)
    return h

def infer_offsets(n):
    return np.arange(2 * n - 1) - (n - 1)

def profile_lag(h, delta):
    n = len(h)
    if delta >= 0:
        m = n - delta
        if m <= 0:
            return 0.0
        return float(np.dot(h[:m], 1.0 - h[delta:delta+m]))
    else:
        k = -delta
        m = n - k
        if m <= 0:
            return 0.0
        return float(np.dot(h[k:k+m], 1.0 - h[:m]))

def gradient_lag(h, delta):
    n = len(h)
    g = np.zeros(n)
    if delta >= 0:
        m = n - delta
        if m <= 0:
            return g
        g[:m] += (1.0 - h[delta:delta+m])
        g[delta:delta+m] -= h[:m]
    else:
        k = -delta
        m = n - k
        if m <= 0:
            return g
        g[k:k+m] += (1.0 - h[:m])
        g[:m] -= h[k:k+m]
    return g

def all_profiles(h):
    n = len(h)
    offsets = infer_offsets(n)
    return np.array([profile_lag(h, d) for d in offsets]) * (2.0 / n)

def all_gradients(h):
    n = len(h)
    offsets = infer_offsets(n)
    G = np.zeros((2 * n - 1, n))
    for m, d in enumerate(offsets):
        G[m] = gradient_lag(h, d)
    return G * (2.0 / n)

def symmetrize(h):
    return repair((h + h[::-1]) / 2.0)

    y_ext = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(x_tgt, x_ext, y_ext)

def active_direction(h, n_active=10):
    """Compute a descent direction based on active (max) lags."""
    n = len(h)
    profs = all_profiles(h)
    score = profs.max()
    # Select active lags (those within tolerance of max)
    threshold = score - 1e-6
    active_mask = profs >= threshold
    if active_mask.sum() == 0:
        active_mask = np.argsort(profs)[::-1][:n_active] > 0  # fallback
        active_mask = np.zeros(len(profs), dtype=bool)
        top_idx = np.argsort(profs)[::-1][:n_active]
        active_mask[top_idx] = True
    G = all_gradients(h)
    # Average gradient over active lags
    g_avg = G[active_mask].mean(axis=0)
    # Project to sum-zero direction
    g_avg -= g_avg.mean()
    # Negative direction (descent)
    d = -g_avg
    return d, score, active_mask

def line_search(h, direction, base_score):
    """Try several step sizes along direction, return best h and score."""
    n = len(h)
    best_h = h.copy()
    best_s = base_score
    for alpha in [0.01, 0.02, 0.05, 0.1, 0.2, 0.005, 0.002]:
        h_try = repair(h + alpha * direction)
        s = exact_score(h_try)
        if s < best_s - 1e-12:
            best_s = s
            best_h = h_try.copy()
    return best_h, best_s

def coordinate_search(h, max_rounds=50):

class SLSQPSolver:
    def __init__(self, n):
        self.n = n
        self.dx = 2.0 / n
        self.offsets = infer_offsets(n)
        self.M = 2 * n - 1
        self.target = n / 2.0

    def obj(self, x):
        return x[self.n]

    def obj_jac(self, x):
        j = np.zeros(self.n + 1)
        j[-1] = 1.0
        return j

    def ineq_fun(self, x):
        h = x[:self.n]
        t = x[self.n]
        profs = np.array([profile_lag(h, d) for d in self.offsets]) * self.dx
        return t - profs

    def ineq_jac(self, x):
        h = x[:self.n]
        G = np.zeros((self.M, self.n))
        for m, d in enumerate(self.offsets):
            G[m] = gradient_lag(h, d)
        G *= self.dx
        return np.hstack([-G, np.ones((self.M, 1))])

    def eq_fun(self, x):
        return x[:self.n].sum() - self.target

    def eq_jac(self, x):
        j = np.zeros(self.n + 1)
        j[:self.n] = 1.0
        return j

    def solve(self, h0, t_start, maxiter=200):
        x0 = np.zeros(self.n + 1)
        x0[:self.n] = h0
        x0[self.n] = t_start
        bounds = [(0.0, 1.0)] * self.n + [(0.0, t_start + 1.0)]
        cons = [
            {'type': 'ineq', 'fun': self.ineq_fun, 'jac': self.ineq_jac},
            {'type': 'eq', 'fun': self.eq_fun, 'jac': self.eq_jac},
        ]
        result = minimize(self.obj, x0, method='SLSQP', jac=self.obj_jac,
                         bounds=bounds, constraints=cons,
                         options={'maxiter': maxiter, 'ftol': 1e-14})
        return result

def repair(h):

    best_s = exact_score(h)
    h_cur = h.copy()
    step_sizes = [0.02, 0.05, 0.01, 0.1, 0.005]
    for rnd in range(max_rounds):
        if time_limit is not None and t0 is not None:
            if time.perf_counter() - t0 > time_limit:
                break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            for ss in step_sizes:
                for sign in [1, -1]:
                    h_try = h_cur.copy()
                    h_try[i] += sign * ss
                    h_try = repair(h_try)
                    s = exact_score(h_try)
                    if s < best_s - 1e-14:
                        best_s = s
                        h_cur = h_try
                        improved = True
                        break
                if improved:
                    break
            if not improved:
                for ss in [0.001, 0.002]:
                    for sign in [1, -1]:
                        h_try = h_cur.copy()
                        h_try[i] += sign * ss
                        h_try = repair(h_try)
                        s = exact_score(h_try)
                        if s < best_s - 1e-14:
                            best_s = s
                            h_cur = h_try
                            improved = True
                            break
                    if improved:
                        break
        if not improved:
            break
    return h_cur, best_s

class SLSQPSolver:
    def __init__(self, n):
        self.n = n
        self.dx = 2.0 / n
        self.offsets = infer_offsets(n)
        self.M = 2 * n - 1
        self.target = n / 2.0

    def obj(self, x):
        return x[self.n]

    def obj_jac(self, x):
        j = np.zeros(self.n + 1)
        j[-1] = 1.0
        return j

    def ineq_fun(self, x):
        h = x[:self.n]
        t = x[self.n]
        profs = np.zeros(self.M)
        for m, d in enumerate(self.offsets):
            profs[m] = profile_lag(h, d)
        return t - profs * self.dx

    def ineq_jac(self, x):
        h = x[:self.n]
        G = np.zeros((self.M, self.n))
        for m, d in enumerate(self.offsets):
            G[m] = gradient_lag(h, d)
        G *= self.dx
        col = np.ones((self.M, 1))
        return np.hstack([-G, col])

    def eq_fun(self, x):
        return x[:self.n].sum() - self.target

    def eq_jac(self, x):
        j = np.zeros(self.n + 1)
        j[:self.n] = 1.0
        return j

    def solve(self, h0, t_start, maxiter=200):
        x0 = np.zeros(self.n + 1)
        x0[:self.n] = h0
        x0[self.n] = t_start
        bounds = [(0.0, 1.0)] * self.n + [(0.0, t_start + 0.5)]
        cons = [
            {'type': 'ineq', 'fun': self.ineq_fun, 'jac': self.ineq_jac},
            {'type': 'eq', 'fun': self.eq_fun, 'jac': self.eq_jac},
        ]
        try:
            result = minimize(self.obj, x0, method='SLSQP', jac=self.obj_jac,
                              bounds=bounds, constraints=cons,
                              options={'maxiter': maxiter, 'ftol': 1e-14})
        except Exception:
            result = None
        return result


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.perf_counter()
    rng = np.random.RandomState(seed)
    target_c5 = 0.3808

    # Get initial values
    try:
        h_parent = np.array(initial_h_values, dtype=float)
    except (NameError, TypeError):
        h_parent = repair(np.full(70, 0.5))

    h_parent = repair(h_parent)
    n = len(h_parent)
    best_h = h_parent.copy()
    best_score = exact_score(best_h)

    def rem():
        return budget_s - (time.perf_counter() - t0)

    # Phase 1: SLSQP from parent
    if rem() > 60:
        solver = SLSQPSolver(n)
        result = solver.solve(h_parent, best_score, maxiter=400)
        if result is not None and result.x is not None:
            h_try = repair(result.x[:n])
            s_try = exact_score(h_try)
            if s_try < best_score:
                best_h = h_try
                best_score = s_try

    if best_score <= target_c5:
        return (best_h, best_score, len(best_h))

    # Phase 2: Active gradient descent
    if rem() > 60:
        h_cur, s_cur = active_gradient_descent(best_h, max_iters=300, n_active=20,
                                              time_limit=rem() - 30, t0=t0)
        if s_cur < best_score:
            best_h = h_cur
            best_score = s_cur

    if best_score <= target_c5:
        return (best_h, best_score, len(best_h))

    # Phase 3: Symmetrize and solve
    if rem() > 60:
        h_sym = symmetrize(best_h)
        s_sym = exact_score(h_sym)
        if s_sym < best_score:
            best_h = h_sym
            best_score = s_sym
        if rem() > 40:
            solver = SLSQPSolver(n)
            result = solver.solve(best_h, best_score, maxiter=300)
            if result is not None and result.x is not None:
                h_try = repair(result.x[:n])
                s_try = exact_score(h_try)
                if s_try < best_score:
                    best_h = h_try
                    best_score = s_try

    if best_score <= target_c5:
        return (best_h, best_score, len(best_h))

    # Phase 4: Binary push + SLSQP
    if rem() > 60:
        h_bin = binary_push(best_h, strength=0.1)
        s_bin = exact_score(h_bin)
        h_start = h_bin if s_bin < best_score else best_h
        if s_bin < best_score:
            best_h = h_bin
            best_score = s_bin
        if rem() > 40:
            solver = SLSQPSolver(n)
            result = solver.solve(h_start, best_score, maxiter=300)
            if result is not None and result.x is not None:
                h_try = repair(result.x[:n])
                s_try = exact_score(h_try)
                if s_try < best_score:
                    best_h = h_try
                    best_score = s_try

    if best_score <= target_c5:
        return (best_h, best_score, len(best_h))

    # Phase 5: SLSQP from perturbed starts
    if rem() > 90:
        for trial in range(8):
            if rem() < 30:
                break
            noise_scale = 0.005 + 0.005 * trial
            noise = rng.randn(n) * noise_scale
            h_pert = repair(best_h + noise)
            s_pert = exact_score(h_pert)
            solver = SLSQPSolver(n)
            result = solver.solve(h_pert, s_pert, maxiter=200)
            if result is not None and result.x is not None:
                h_try = repair(result.x[:n])
                s_try = exact_score(h_try)
                if s_try < best_score:
                    best_h = h_try
                    best_score = s_try
                    if best_score <= target_c5:
                        break

    if best_score <= target_c5:
        return (best_h, best_score, len(best_h))

    # Phase 6: Resolution transfer
    if rem() > 90:
        for dn in [4, -4, 8, -8, 16, -16]:
            if rem() < 40:
                break
            new_n = n + dn
            if new_n < 30 or new_n > 200:
                continue
            h_new = interpolate_to(best_h, new_n)
            s_new = exact_score(h_new)
            if s_new < best_score:
                best_h = h_new
                best_score = s_new
            if rem() < 30:
                break
            solver = SLSQPSolver(new_n)
            result = solver.solve(h_new, s_new, maxiter=150)
            if result is not None and result.x is not None:
                h_try = repair(result.x[:new_n])
                s_try = exact_score(h_try)
                if s_try < best_score:
                    best_h = h_try
                    best_score = s_try
                    n = new_n
            if best_score <= target_c5:
                break

    if best_score <= target_c5:
        return (best_h, best_score, len(best_h))

    # Phase 7: More active gradient descent on best
    if rem() > 40:
        h_cur, s_cur = active_gradient_descent(best_h, max_iters=200, n_active=12,
                                              time_limit=rem() - 20, t0=t0)
        if s_cur < best_score:
            best_h = h_cur
            best_score = s_cur

    # Phase 8: Final SLSQP polish
    if rem() > 30:
        n_now = len(best_h)
        solver = SLSQPSolver(n_now)
        result = solver.solve(best_h, best_score, maxiter=300)
        if result is not None and result.x is not None:
            h_try = repair(result.x[:n_now])
            s_try = exact_score(h_try)
            if s_try < best_score:
                best_h = h_try
                best_score = s_try

    # Phase 9: Final active gradient + coordinate
    if rem() > 30:
        h_cur, s_cur = active_gradient_descent(best_h, max_iters=100, n_active=10,
                                              time_limit=rem() - 15, t0=t0)
        if s_cur < best_score:
            best_h = h_cur
            best_score = s_cur
        if rem() > 15:
            h_cur, s_cur = coordinate_search(best_h, max_rounds=20,
                                            time_limit=rem() - 10, t0=t0)
            if s_cur < best_score:
                best_h = h_cur
                best_score = s_cur

    # Phase 10: Final SLSQP
    if rem() > 20:
        n_now = len(best_h)
        solver = SLSQPSolver(n_now)
        result = solver.solve(best_h, best_score, maxiter=200)
        if result is not None and result.x is not None:
            h_try = repair(result.x[:n_now])
            s_try = exact_score(h_try)
            if s_try < best_score:
                best_h = h_try
                best_score = s_try

    # Final verification
    best_h = repair(best_h)
    best_score = exact_score(best_h)

    return (best_h, best_score, len(best_h))
