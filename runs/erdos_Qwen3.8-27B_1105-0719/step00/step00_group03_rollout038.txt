from scipy.optimize import minimize
import time

def evaluate(h):
    """Compute C5 for a given h array."""
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1-h, mode='full') * dx
    return np.max(corr)

def coordinate_search(h, max_iterations=1000, tol=1e-10):
    """Coordinate descent to minimize C5."""
    n = len(h)
    dx = 2.0 / n
    C = evaluate(h)
    
    for it in range(max_iterations):
        improved = False
        for j in range(n):
            # Compute contribution of h[j] to each lag
            # result[k] = sum_i h[i+k]*(1-h[i]) for the full correlate
            # We need to find the marginal effect of changing h[j]
            
            # Get current correlations
            corr = np.correlate(h, 1-h, mode='full')

            # lag = k - (n-1)
            # If lag >= 0: result = sum_{i=0}^{n-1-lag} h[i+lag]*(1-h[i])
            # If lag < 0 (lag = -m): result = sum_{i=m}^{n-1} h[i-m]*(1-h[i])

            # k_full = 0 corresponds to lag = -(n-1)
            # k_full = n-1 corresponds to lag = 0
            # k_full = 2n-2 corresponds to lag = n-1
            
            # Let me just compute the gradient numerically for simplicity

            # element at index p corresponds to lag = p - (n-1)

            b = np.zeros(2*n - 1)
            for p in range(2*n - 1):
                lag = p - (n - 1)
                contrib = 0
                if lag >= 0:
                    # sum_{i=0}^{n-1-lag} h[i+lag]*(1-h[i])
                    # h[j] as h[i]: i=j, valid if 0<=j<=n-1-lag, i.e., j+lag <= n-1
                    if j + lag <= n - 1:
                        contrib -= h[j + lag]

                    if lag <= j:
                        contrib += (1 - h[j - lag])
                else:
                    m = -lag
                    # sum_{i=m}^{n-1} h[i-m]*(1-h[i])
                    # h[j] as h[i-m]: i-m=j, i=j+m, valid if m<=j+m<=n-1, i.e., j<=n-1-m
                    if j <= n - 1 - m:

                    if j >= m:
                        contrib -= h[j - m]

            lo = -h[j]
            hi = 1 - h[j]

            # This is a piecewise linear function, minimum at intersection or boundary

            candidates = [lo, hi]

            sorted_idx = np.argsort(corr)[::-1]
            top_count = min(20, 2*n - 1)

                    if b[p1] == b[p2]:

                    # corr[p1] + delta*b[p1] = corr[p2] + delta*b[p2]
                    delta = (corr[p2] - corr[p1]) / (b[p1] - b[p2])

                new_C = np.max(np.correlate(h, 1-h, mode='full') * dx)

                    best_delta = delta
                h[j] -= delta
            
            if best_delta != 0:
                h[j] += best_delta
                C = best_C
                improved = True
        
        if not improved:

    return h, C

def coord_descent(h, n_iters=200):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    C = np.max(corr) * dx
    
    for it in range(n_iters):
        improved = False
        for j in range(n):
            # Compute b[p] = d(corr[p])/d(h[j])
            b = np.zeros(2*n - 1)
            for p in range(2*n - 1):
                lag = p - (n - 1)
                if lag >= 0:
                    if j + lag <= n - 1:
                        b[p] -= h[j + lag]
                    if lag <= j:
                        b[p] += (1 - h[j - lag])
                else:
                    m = -lag
                    if j <= n - 1 - m:
                        b[p] += (1 - h[j + m])
                    if j >= m:
                        b[p] -= h[j - m]
            
            # Minimize max_p (corr[p] + delta * b[p]) over delta in [lo, hi]
            lo = -h[j]
            hi = 1 - h[j]
            
            # The function f(delta) = max_p (corr[p] + delta*b[p])
            # is convex piecewise linear
            # Minimum is at a breakpoint or boundary
            
            # Get breakpoints where pairs of lines intersect
            candidates = [lo, hi]
            
            # For efficiency, consider only the top ~30 lags
            sorted_idx = np.argsort(corr)[::-1][:30]
            for i in range(len(sorted_idx)):
                for k2 in range(i+1, len(sorted_idx)):
                    p1, p2 = sorted_idx[i], sorted_idx[k2]
                    denom = b[p1] - b[p2]
                    if abs(denom) < 1e-12:
                        continue
                    delta = (corr[p2] - corr[p1]) / denom
                    if lo - 1e-9 <= delta <= hi + 1e-9:
                        candidates.append(max(lo, min(hi, delta)))
            
            best_val = C / dx  # current max corr
            best_delta = 0.0
            for delta in candidates:
                val = np.max(corr + delta * b)
                if val < best_val - 1e-12:
                    best_val = val
                    best_delta = delta
            
            if best_delta != 0:
                h[j] += best_delta
                # Update corr
                corr += best_delta * b
                C = np.max(corr) * dx
                improved = True
        
        if not improved:
            break
    
    return h, C
