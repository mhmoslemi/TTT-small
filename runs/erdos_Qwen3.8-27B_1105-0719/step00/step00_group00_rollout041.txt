def lse_objective_and_grad(h, beta):

def optimize_lse(h0, n, beta, maxiter=1000, tol=1e-10):
    """Optimize LSE objective using L-BFGS-B with sum constraint handled by reduced coords."""
    # Use reduced coordinates: z has n-1 components, h[n-1] = n/2 - sum(z)
    T = n / 2.0
    
    def reduced_obj_grad(z):
        h = np.empty(n)
        h[:n-1] = z
        h[n-1] = T - np.sum(z)
        # Check bounds on h[n-1]
        if h[n-1] < -1e-10 or h[n-1] > 1+1e-10:
            return 1e10, np.zeros(n-1)
        obj, grad_full = lse_objective_and_grad(h, beta)
        # Project gradient to reduced space
        grad_z = grad_full[:n-1] - grad_full[n-1]  # because h[n-1] = T - sum(z)
        # dh[n-1]/dz[i] = -1, so grad_z[i] = grad_full[i] + grad_full[n-1]*(-1) = grad_full[i] - grad_full[n-1]
        return obj, grad_z
    
    # Initial point
    h = h0.copy()
    z0 = h.copy()
    # Project to feasible
    z0 = project_to_feasible(h0)
    z0_red = z0[:n-1]
    
    # Bounds for z: 0 <= z_i <= 1 and 0 <= T - sum(z) <= 1
    # The last constraint is tricky for L-BFGS-B. Let's just use box constraints and hope
    # the optimizer stays in the feasible region, or use a penalty.

                      bounds=bounds, options={'maxiter': maxiter, 'ftol': tol, 'gtol': tol})

    h_opt[n-1] = T - np.sum(result.x)

    h_opt = project_to_feasible(h_opt)

import numpy as np
from scipy.optimize import minimize
import time


def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_to_feasible(y):
    n = len(y)
    T = n / 2.0
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y)) + 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.sum(np.clip(y - mid, 0.0, 1.0)) > T:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) * 0.5
    h = np.clip(y - t, 0.0, 1.0)
    residual = T - np.sum(h)
    if abs(residual) > 1e-12:
        eligible = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(eligible):
            h[eligible] += residual / np.sum(eligible)
            h = np.clip(h, 0.0, 1.0)
    return h


def lse_obj_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    q = 1.0 - h
    corr = np.correlate(h, q, mode="full")
    scaled = beta * corr
    max_s = np.max(scaled)
    exp_s = np.exp(scaled - max_s)
    sum_exp = np.sum(exp_s)
    obj = (max_s + np.log(sum_exp)) / beta * dx
    w = exp_s / sum_exp
    first = np.correlate(w, q, mode="full")[:n][::-1]
    second = np.convolve(w, h, mode="full")[n - 1:2 * n - 1]
    grad = dx * (first - second)
    return obj, grad


def lse_slsqp(h0, beta, maxiter=200):
    n = len(h0)
    T = n / 2.0

    def obj(h_flat):
        return lse_obj_grad(h_flat, beta)[0]

    def grad(h_flat):
        return lse_obj_grad(h_flat, beta)[1]

    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - T,
                   'jac': lambda h: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        return project_to_feasible(res.x)
    except Exception:
        return project_to_feasible(h0)


def exact_subgradient(h):
    n = len(h)
    q = 1.0 - h
    corr = np.correlate(h, q, mode="full")
    max_val = np.max(corr)
    active = np.where(np.abs(corr - max_val) < 1e-10)[0]
    grad = np.zeros(n)
    for m in active:
        k = m - (n - 1)
        for j in range(n):

            if 0 <= j + k < n:
                g += (1.0 - h[j + k])
            if 0 <= j - k < n:
                g -= h[j - k]
            grad[j] += g
    grad /= len(active)
    return grad, max_val


def projected_gradient(h0, max_iter=5000, lr_init=0.05, use_momentum=True):

        else:

def pair_search(h, max_rounds=30, time_limit=15.0):
    n = len(h)
    h = h.copy()
    current_c5 = evaluate_c5(h)
    start = time.time()

    for rnd in range(max_rounds):
        if time.time() - start > time_limit:
            break
        improved = False
        order = np.random.permutation(n)
        for ii in range(n):
            for jj in range(ii + 1, n):
                i, j = int(order[ii]), int(order[jj])
                for step in steps:
                    if h[i] + step <= 1.0 and h[j] - step >= 0.0:
                        trial = h.copy()
                        trial[i] += step
                        trial[j] -= step
                        c5 = evaluate_c5(trial)

                    if h[i] - step >= 0.0 and h[j] + step <= 1.0:

                        trial[i] -= step

                        if c5 < current_c5 - 1e-12:
                            h = trial
                            current_c5 = c5
                            improved = True

                if improved:

            if improved:
                break

    return h, current_c5

    h_new = np.interp(x_new, x_src, h_src, period=2.0)
