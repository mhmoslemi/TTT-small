def project_balanced(y):

    target = n / 2.0
    y = np.clip(y, 0.0, 1.0)
    # Binary search for lambda such that sum(clip(y + lambda, 0, 1)) = target
    lo, hi = -2.0, 2.0
    for _ in range(100):

        s = np.sum(np.clip(y + mid, 0.0, 1.0))
        if s < target:

        else:

    residual = target - np.sum(x)

                add = min(room, residual)
                x[i] += add
                residual -= add

        else:

                sub = min(room, -residual)
                x[i] -= sub
                residual += sub

def compute_weights(scores, mode='active', beta=5.0, tol_frac=0.01):
    n_scores = len(scores)
    max_s = np.max(scores)
    if max_s < 1e-15:
        return np.ones(n_scores) / n_scores
    if mode == 'active':
        threshold = max_s * (1.0 - tol_frac)
        active = scores >= threshold
        w = np.zeros(n_scores)
        w[active] = 1.0 / np.sum(active)
        return w
    elif mode == 'soft':
        shifted = (max_s - scores) / max(max_s, 1e-15)
        w = np.exp(-beta * shifted)
        w /= np.sum(w)
        return w
    else:
        return np.ones(n_scores) / n_scores

def weighted_gradient(h, scores, w):
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        if w[idx] < 1e-10:
            continue
        k = idx - (n - 1)
        wi = w[idx]
        if k >= 0:
            # grad[k:n] += wi * (1-h[0:n-k])
            # grad[0:n-k] -= wi * h[k:n]
            grad[k:] += wi * (1.0 - h[:n-k])
            grad[:n-k] -= wi * h[k:]
        else:
            d = -k
            # grad[0:n-d] += wi * (1-h[d:n])
            # grad[d:n] -= wi * h[0:n-d]
            grad[:n-d] += wi * (1.0 - h[d:])
            grad[d:] -= wi * h[:n-d]
    grad *= dx
    return grad

def random_transfer(h, rng, max_delta=0.1):
    n = len(h)
    h = h.copy()
    i = rng.integers(n)
    j = rng.integers(n)
    while j == i:
        j = rng.integers(n)
    delta = rng.uniform(-max_delta, max_delta)
    # Ensure feasibility
    if delta > 0:
        # move from j to i
        max_d = min(h[j], 1.0 - h[i])
        delta = min(delta, max_d)
        if delta <= 0:
            return None
        h[i] += delta
        h[j] -= delta
    else:
        # move from i to j
        max_d = min(h[i], 1.0 - h[j])
        delta = max(delta, -max_d)
        if delta >= 0:
            return None
        h[i] += delta
        h[j] -= delta
    return h

def active_transfer(h, scores, rng, n_top=3, max_delta=0.05):
    n = len(h)
    h = h.copy()
    # Pick a top-scoring lag
    max_s = np.max(scores)

    active_idx = np.where(scores >= threshold)[0]
    if len(active_idx) == 0:
        active_idx = np.array([np.argmax(scores)])
    idx = active_idx[rng.integers(len(active_idx))]
    k = idx - (n - 1)
    if k >= 0:
        # Pairs: (i, i+k) for i in [0, n-1-k]
        if n - 1 - k < 0:
            return None
        i = rng.integers(n - k)
        j = i + k
    else:
        d = -k
        # Pairs: (i, i-d) for i in [d, n-1]
        # i.e., pairs (i-d, i) for i in [d, n-1]
        i = rng.integers(d, n)
        j = i - d
    # Transfer from one to the other
    direction = rng.integers(2)
    if direction == 0:
        src, tgt = i, j
    else:
        src, tgt = j, i
    max_d = min(h[src], 1.0 - h[tgt])
    if max_d <= 0:
        return None
    delta = rng.uniform(0.01, min(max_delta, max_d))
    h[src] -= delta
    h[tgt] += delta
    return h

def resample(h_old, n_new):
    n_old = len(h_old)
    dx_old = 2.0 / n_old
    dx_new = 2.0 / n_new
    # For each new cell [i*dx_new, (i+1)*dx_new], compute the integral of h_old
    h_new = np.zeros(n_new)

        x_end = (i + 1) * dx_new
        # Overlap with old cells
        for j in range(n_old):
            xs = max(x_start, j * dx_old)
            xe = min(x_end, (j + 1) * dx_old)
            if xe > xs:
                h_new[i] += h_old[j] * (xe - xs)
        h_new[i] /= dx_new
    return project_balanced(h_new)

def exact_scores(h):
    n = len(h)
    dx = 2.0 / n
    full = np.correlate(h, 1.0 - h, mode='full') * dx
    return full, float(np.max(full))

def project_balanced(y):
    n = len(y)
    target = n / 2.0
    y = np.asarray(y, dtype=np.float64)
    y = np.clip(y, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(y + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    x = np.clip(y + lam, 0.0, 1.0)
    residual = target - np.sum(x)
    if abs(residual) > 1e-12:
        if residual > 0:
            for i in range(n):
                room = 1.0 - x[i]
                add = min(room, residual)
                x[i] += add
                residual -= add
                if abs(residual) < 1e-15:
                    break
        else:
            for i in range(n):
                room = x[i]
                sub = min(room, -residual)
                x[i] -= sub
                residual += sub
                if abs(residual) < 1e-15:
                    break
    return x

def compute_weights(scores, mode='active', beta=5.0, tol_frac=0.01):
    n_s = len(scores)
    max_s = float(np.max(scores))
    if max_s < 1e-15:
        return np.ones(n_s) / n_s
    if mode == 'active':
        threshold = max_s * (1.0 - tol_frac)
        active = scores >= threshold
        cnt = np.sum(active)
        if cnt == 0:
            active[np.argmax(scores)] = True
            cnt = 1
        w = np.zeros(n_s)
        w[active] = 1.0 / cnt
        return w
    else:
        shifted = (max_s - scores) / max(max_s, 1e-15)
        w = np.exp(-beta * shifted)
        w = np.maximum(w, 1e-15)
        w /= np.sum(w)
        return w

def weighted_gradient(h, scores, w):
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        wi = w[idx]
        if wi < 1e-12:
            continue
        k = idx - (n - 1)
        if k >= 0:
            m = n - k
            grad[k:] += wi * (1.0 - h[:m])
            grad[:m] -= wi * h[k:]
        else:
            d = -k
            m = n - d
            grad[:m] += wi * (1.0 - h[d:])
            grad[d:] -= wi * h[:m]
    grad *= dx
    return grad

def random_transfer(h, rng, max_delta=0.08):
    n = len(h)
    i = int(rng.integers(n))
    j = int(rng.integers(n))
    if i == j:
        return None
    delta = float(rng.uniform(0.005, max_delta))
    direction = int(rng.integers(2))
    if direction == 0:
        src, tgt = i, j
    else:
        src, tgt = j, i
    max_d = min(h[src], 1.0 - h[tgt])
    if max_d < 1e-10:
        return None
    delta = min(delta, max_d)
    h_new = h.copy()
    h_new[src] -= delta
    h_new[tgt] += delta
    return h_new

def resample(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return project_balanced(h_old)
    dx_old = 2.0 / n_old
    dx_new = 2.0 / n_new
    h_new = np.zeros(n_new)
    for i in range(n_new):
        x_start = i * dx_new
        x_end = (i + 1) * dx_new
        # Find overlapping old cells
        j_start = max(0, int(np.floor(x_start / dx_old)))
        j_end = min(n_old, int(np.ceil(x_end / dx_old)))
        for j in range(j_start, j_end):
            xs = max(x_start, j * dx_old)
            xe = min(x_end, (j + 1) * dx_old)
            if xe > xs:
                h_new[i] += h_old[j] * (xe - xs)
        h_new[i] /= dx_new
    return project_balanced(h_new)
