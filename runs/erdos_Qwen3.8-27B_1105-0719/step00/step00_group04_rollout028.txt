    N = len(h)
    dx = 2.0 / N
    current_c5 = evaluate_c5(h)
    h = h.copy()
    
    for it in range(max_iter):
        c = compute_correlations(h, dx)
        # Subgradient of max: pick the argmax
        idx = np.argmax(c)
        L = idx - (N - 1)

import numpy as np
from scipy.optimize import minimize
import time

def project_onto_box_sum(x, target_sum):
    n = len(x)
    lo, hi = -20.0, 20.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5_components(h, dx):
    b = 1.0 - h
    return np.correlate(h, b, mode='full') * dx

def evaluate_c5(h):
    N = len(h)
    dx = 2.0 / N
    return compute_c5_components(h, dx).max()

def surrogate_obj_grad(h, dx, tau):
    N = len(h)
    c = compute_c5_components(h, dx)
    m = c.max()
    val = m + tau * np.log(np.sum(np.exp((c - m) / tau)))
    w = np.exp((c - m) / tau)
    w /= w.sum()
    b = 1.0 - h
    grad = np.zeros(N)
    for i in range(2 * N - 1):
        L = i - (N - 1)
        wi = w[i] * dx
        if wi < 1e-15:
            continue
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += wi * b[:N - L]
            grad[:N - L] -= wi * h[L:]
        else:
            k = -L
            grad[:N - k] += wi * b[k:]
            grad[k:] -= wi * h[:N - k]
    return val, grad

def optimize_lbfgsb(h0, N, tau, max_iter=300, penalty_coeff=50.0):
    dx = 2.0 / N

        v, _ = surrogate_obj_grad(h_, dx, tau)

        return v + 0.5 * penalty_coeff * s * s

    def grad(h_):
        _, g = surrogate_obj_grad(h_, dx, tau)
        s = h_.sum() - target
        return g + penalty_coeff * s

    bounds = [(0.0, 1.0)] * N

def optimize_slsqp(h0, N, tau, max_iter=300):
    dx = 2.0 / N
    target = N / 2.0

    def obj(h_):
        v, _ = surrogate_obj_grad(h_, dx, tau)
        return v

    def grad(h_):
        _, g = surrogate_obj_grad(h_, dx, tau)
        return g

    def cons_fun(h_):
        return h_.sum() - target

    def cons_jac(h_):
        return np.ones(N)

    cons = {'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}
    bounds = [(0.0, 1.0)] * N
    res = minimize(obj, h0, jac=grad, method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-12})
    return res.x
