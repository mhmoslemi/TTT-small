import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1-h, mode="full")
    return np.max(corr) * dx

def compute_all_c5(h, dx):
    corr = np.correlate(h, 1-h, mode="full")
    return corr * dx

def project_to_constraint(h, target_sum):
    """Project onto {0<=h<=1, sum(h)=target_sum}"""
    for _ in range(200):
        # Binary search for lambda
        lo, hi = -2.0, 2.0
        for _ in range(100):
            mid = (lo + hi) / 2
            proj = np.clip(h - mid, 0.0, 1.0)
            if np.sum(proj) > target_sum:
                lo = mid
            else:
                hi = mid
        return np.clip(h - (lo+hi)/2, 0.0, 1.0)

def smooth_max_objective(h, dx, alpha):
    corr = compute_all_c5(h, dx)
    m = np.max(corr)
    shifted = alpha * (corr - m)
    return m + np.log(np.sum(np.exp(shifted))) / alpha

def smooth_max_gradient(h, dx, alpha):
    n = len(h)
    corr = compute_all_c5(h, dx)
    m = np.max(corr)
    shifted = alpha * (corr - m)
    weights = np.exp(shifted) / np.sum(np.exp(shifted))
    
    grad = np.zeros(n)
    for k in range(2*n - 1):
        if weights[k] < 1e-12:
            continue

        valid_minus = (j_minus >= 0) & (j_minus < n)
        grad[valid_minus] += weights[k] * (-h[j_minus[valid_minus]])

def compute_c5(h, dx):

def project_to_constraint(h, target_sum):
    """Project onto {0<=h<=1, sum(h)=target_sum} via bisection on lambda."""
    lo, hi = -1.0, 1.0

        mid = 0.5 * (lo + hi)

    m = np.max(corr_vals)

Let me also add a "coordinate descent" phase where I optimize one variable at a time. For each variable h[j], I can find the optimal value (given all others fixed) by minimizing the max over lags. This is a 1D problem (though non-smooth due to the max).

For a fixed j and all other h[i] fixed, the overlap at lag L is:
$c_L = \sum_{i \neq j} h[i](1-h[i-L]) + \text{terms involving } h[j]$

The terms involving h[j] are:
- If 0 ≤ j-L < n: contribution is $h[j] \cdot (1-h[j-L])$ (h[j] as first factor)
- If 0 ≤ j+L < n: contribution is $h[j+L] \cdot (1-h[j])$ (h[j] as second factor, i.e., h[j] = h[(j+L)-L])

So $c_L = A_L + h[j] \cdot B_L$ where:
- $A_L = \sum_{i \neq j, valid} h[i](1-h[i-L])$ (terms not involving h[j])
- $B_L = (1-h[j-L]) \cdot \mathbb{1}[0≤j-L<n] - h[j+L] \cdot \mathbb{1}[0≤j+L<n]$

So $c_L = A_L + B_L \cdot h[j]$

We want to minimize $\max_L (A_L + B_L \cdot h[j])$ over $h[j] \in [0,1]$ (with the constraint that the sum is maintained, but for coordinate descent we can handle this separately).

This is a 1D problem: minimize the maximum of linear functions. The optimal h[j] is where the maximum is minimized, which occurs at an intersection of two of the lines, or at a boundary.

This is essentially a "minimax" in 1D, which can be solved by:
- Computing all the lines $A_L + B_L \cdot x$
- Finding the point x ∈ [0,1] that minimizes the maximum of these lines

This can be done by:
1. For each pair of lines, find their intersection
2. Check which intersection points are in [0,1]
3. Evaluate the max at each candidate point (intersections + boundaries)
4. Return the best

This is O(K^2) where K is the number of lags, but many lines will be identical or nearly so. For n=45, K=89, so K^2 ≈ 7921, which is fine.

Let me implement this coordinate descent:

```python
def coordinate_step(h, j, dx, target_sum):
    """Optimize h[j] given all other coordinates fixed."""
    n = len(h)
    # Compute A_L and B_L for each lag
    # c_L = sum_i h[i](1-h[i-L]) for valid i
    # Terms involving h[j]:
    #   As first factor (i=j): h[j]*(1-h[j-L]) if 0<=j-L<n
    #   As second factor (i-L=j, i=j+L): h[j+L]*(1-h[j]) if 0<=j+L<n
    
    # c_L = A_L + B_L * h[j]
    # where A_L = sum of terms i≠j, and B_L = coefficient of h[j]
    
    h_j = h[j]  # current value
    A = np.zeros(2*n-1)
    B = np.zeros(2*n-1)
    
    idx = np.arange(n)
    for k in range(2*n-1):
        L = k - (n-1)
        # Sum all terms
        jp = idx + L  # i + L for all i... wait
        # c_k = sum_i h[i] * (1-h[i-L]) for valid i
        # Let's compute directly
        total = 0.0
        coeff = 0.0
        for i in range(n):
            ip = i - L
            if 0 <= ip < n:
                if i == j:
                    # h[j]*(1-h[j-L]) = h[j] - h[j]*h[j-L]
                    # Linear in h[j]: (1-h[j-L])*h[j]
                    coeff += (1.0 - h[ip])
                else:
                    total += h[i] * (1.0 - h[ip])
        A[k] = total
        B[k] = coeff
    
    # Now minimize max_k (A[k] + B[k]*x) over x in [0,1]
    # (ignoring the sum constraint for now)
    candidates = [0.0, 1.0]
    nk = 2*n - 1
    for i_k in range(nk):
        for j_k in range(i_k+1, nk):
            # Intersection of A[i]+B[i]*x = A[j]+B[j]*x
            db = B[i_k] - B[j_k]
            if abs(db) < 1e-15:
                continue
            x_int = (A[j_k] - A[i_k]) / db
            if 0.0 <= x_int <= 1.0:
                candidates.append(x_int)
    
    best_x = h_j
    best_val = np.max(A + B * h_j)
    for x in candidates:
        val = np.max(A + B * x)
        if val < best_val:
            best_val = val
            best_x = x
    
    return best_x
```

    A = corr - B * h_val

    # This is a 1D minimax problem
    best_x = h_val
    best_val = float(np.max(corr))

    candidates = [0.0, 1.0, h_val]

    # Find the top few lines
    vals_at_0 = A
    vals_at_1 = A + B
    # The max at x is determined by the upper envelope
    # Check intersections of pairs that could be active

    for i in range(2*n - 1):

```python
import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_constraint(h, target_sum):
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def smooth_grad(h, dx, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    shifted = alpha * (corr - m)
    ew = np.exp(shifted)
    w = ew / np.sum(ew)
    
    grad = np.zeros(n)
    idx = np.arange(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        L = k - (n - 1)
        jm = idx - L
        mask_m = (jm >= 0) & (jm < n)
        grad[mask_m] += wk * (1.0 - h[jm[mask_m]])
        jp = idx + L
        mask_p = (jp >= 0) & (jp < n)
        grad[mask_p] += wk * (-h[jp[mask_p]])
    
    return grad * dx

def optimize_gradient(h_init, dx, target_sum, time_limit, alpha_init=20.0):
    n = len(h_init)
    h = project_to_constraint(h_init.astype(float).copy(), target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    alpha = alpha_init
    lr = 0.01
    stall = 0
    corr_vals = np.correlate(h, 1.0-h, mode="full") * dx
    
    while time.time() < time_limit:
        m = np.max(corr_vals)
        shifted = alpha * (corr_vals - m)
        ew = np.exp(shifted)
        w = ew / np.sum(ew)
        obj = m + np.log(np.sum(ew)) / alpha
        
        grad = np.zeros(n)
        idx = np.arange(n)
        for k in range(2 * n - 1):
            wk = w[k]
            if wk < 1e-14:
                continue
            L = k - (n - 1)
            jm = idx - L
            mask_m = (jm >= 0) & (jm < n)
            grad[mask_m] += wk * (1.0 - h[jm[mask_m]])
            jp = idx + L
            mask_p = (jp >= 0) & (jp < n)
            grad[mask_p] += wk * (-h[jp[mask_p]])
        grad *= dx
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        step = lr
        improved = False
        for _ in range(20):
            h_try = project_to_constraint(h - step * grad, target_sum)
            corr_try = np.correlate(h_try, 1.0 - h_try, mode="full") * dx
            m_try = np.max(corr_try)
            shifted_try = alpha * (corr_try - m_try)
            obj_try = m_try + np.log(np.sum(np.exp(shifted_try))) / alpha
            if obj_try < obj - 1e-15:
                h = h_try
                corr_vals = corr_try
                improved = True
                break
            step *= 0.5

            if stall > 40:

                lr *= 0.5
                stall = 0

        if c5_now < best_c5:

    return best_h, best_c5
