1. We need to find a step function h: [0,2] → [0,1] that minimizes C₅ = max_k ∫ h(x)(1-h(x+k)) dx
2. The constraint is ∫₀² h(x) dx = 1, with 0 ≤ h(x) ≤ 1
3. We're given `initial_h_values` (n=70 samples) as a starting point
4. Current record: C₅ ≤ 0.38085, target: 0.3808

The objective is:
C₅ = max( np.correlate(h, 1-h, mode="full") * dx )

where dx = 2.0/n and sum(h) = n/2.

Strategy:
1. Start from `initial_h_values`
2. Use pair coordinate descent: pick two indices i, j, move h[i] += t, h[j] -= t, minimize the exact objective along this line using golden section search
3. The objective along a pair line is convex (max of convex quadratics), so golden section works
4. Use gradient information for pair selection
5. Try multiple resolutions

def repair_box_sum(h):
    """Enforce 0<=h[i]<=1 and sum(h)=n/2 via water-filling."""
    h = np.asarray(h, dtype=np.float64).copy()
    n = h.size
    target = n / 2.0
    
    # Replace non-finite values
    h[~np.isfinite(h)] = 0.0
    
    # Clip to [0, 1]
    h = np.clip(h, 0.0, 1.0)
    
    # Water-filling to achieve exact target
    for _ in range(200):
        diff = h.sum() - target
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            # Too much mass: reduce entries that are > 0
            mask = h > 1e-15
            if not mask.any():
                break
            count = mask.sum()
            step = min(diff / count, h[mask].min() + 1e-15)
            h[mask] -= step
            h = np.clip(h, 0.0, 1.0)
        else:
            # Too little mass: increase entries that are < 1
            mask = h < 1.0 - 1e-15
            if not mask.any():
                break
            count = mask.sum()
            step = min(-diff / count, (1.0 - h[mask]).min() + 1e-15)
            h[mask] += step
            h = np.clip(h, 0.0, 1.0)
    
    # Final fine adjustment
    diff = h.sum() - target
    if abs(diff) > 1e-12:
        # Find a coordinate to adjust
        if diff > 0:
            mask = h > 1e-12
        else:
            mask = h < 1.0 - 1e-12
        if mask.any():
            idx = np.where(mask)[0][0]
            h[idx] -= diff
            h = np.clip(h, 0.0, 1.0)
    
    return h

def erdos_score(h):
    """Compute C5 = max(np.correlate(h, 1-h, 'full')) * (2/n)."""
    n = h.size
    dx = 2.0 / n
    prof = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(prof) * dx)

def erdos_raw_profile(h):

def pair_line_value(h, i, j, t):

    h[i] = hi
    h[j] = hj

def golden_section_min(f, a, b, max_iter=80, tol=1e-10):
    """Minimize convex f on [a, b] using golden section search."""
    if b - a < tol:
        mid = 0.5 * (a + b)
        return f(mid), mid
    
    invphi = 0.6180339887498949
    c = b - invphi * (b - a)
    d = a + invphi * (b - a)
    fc = f(c)
    fd = f(d)
    
    for _ in range(max_iter):
        if b - a < tol:
            break
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - invphi * (b - a)
            fc = f(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + invphi * (b - a)
            fd = f(d)
    
    # Check endpoints and midpoint
    best_t = 0.5 * (a + b)
    best_val = f(best_t)
    for t in [a, b]:
        v = f(t)
        if v < best_val:
            best_val = v
            best_t = t
    
    return best_val, best_t

def compute_pair_bounds(h, i, j):
    """Feasible range of t for h[i]+t >= 0, h[i]+t <= 1, h[j]-t >= 0, h[j]-t <= 1."""

    hi = min(1.0 - h[i], h[j])
    margin = 1e-12
    lo = max(lo, -1.0)
    hi = min(hi, 1.0)
    return lo + margin, hi - margin

def compute_gradient_matrix_vec(h):
    """Vectorized gradient computation."""
    n = h.size
    num_lags = 2 * n - 1
    G = np.zeros((num_lags, n))
    
    for k_idx in range(num_lags):
        k = k_idx - (n - 1)
        if k >= 0:
            # j-k valid: 0<=j-k<n → k<=j<n+k → j in [k, n-1] (since j<n always, and n+k>n so j<n)
            # So j from k to n-1
            js = np.arange(k, n)
            G[k_idx, js] += (1.0 - h[js - k])  # 1-h[j-k], j-k ranges 0 to n-1-k
            # j+k valid: 0<=j+k<n → j<n-k
            js2 = np.arange(0, n - k)
            G[k_idx, js2] -= h[js2 + k]  # h[j+k], j+k ranges k to n-1
        else:
            a = -k  # positive
            # j-k = j+a valid: 0<=j+a<n → j<n-a
            js = np.arange(0, n - a)
            G[k_idx, js] += (1.0 - h[js + a])  # 1-h[j+a]
            # j+k = j-a valid: 0<=j-a<n → j>=a
            js2 = np.arange(a, n)
            G[k_idx, js2] -= h[js2 - a]  # h[j-a]
    
    return G

def transfer_resolution(h_src, n_dst):
    """Transfer a vector from one resolution to another via interpolation."""
    n_src = h_src.size
    if n_src == 1:
        h_dst = np.full(n_dst, h_src[0])
    else:
        x_src = np.linspace(0.0, 1.0, n_src)

        h_dst = np.interp(x_dst, x_src, h_src)

def repair_box_sum(h):
    """Enforce 0<=h[i]<=1 and sum(h)=n/2 via water-filling."""
    h = np.asarray(h, dtype=np.float64).copy()

    target = n / 2.0

    h = np.clip(h, 0.0, 1.0)

        diff = h.sum() - target

            break
        if diff > 0:

                break

def erdos_score(h):
    """Compute C5 = max(np.correlate(h, 1-h, 'full')) * (2/n)."""
    n = h.size
    dx = 2.0 / n
    prof = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(prof) * dx)


def erdos_raw_max(h):

    prof = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(prof))


def compute_gradient_matrix_vec(h):
    """Gradient of each raw profile component w.r.t. h. Shape (2n-1, n)."""
    n = h.size
    num_lags = 2 * n - 1
    G = np.zeros((num_lags, n))
    for k_idx in range(num_lags):
        k = k_idx - (n - 1)
        if k >= 0:
            js = np.arange(k, n)
            G[k_idx, js] += (1.0 - h[js - k])
            js2 = np.arange(0, n - k)
            G[k_idx, js2] -= h[js2 + k]
        else:

            js = np.arange(0, n - a)
            G[k_idx, js] += (1.0 - h[js + a])
            js2 = np.arange(a, n)
            G[k_idx, js2] -= h[js2 - a]
    return G

def pair_descent_pass(h_cur, cur_raw, max_pairs=20, max_golden_iter=50, rng=None):
    """One pass of pair coordinate descent. Returns (h_new, raw_new, n_accepted)."""
    n = h_cur.size

    raw_work = cur_raw
    accepted = 0
    if rng is None:
        rng = np.random.default_rng(0)

    # Compute gradients for pair selection
    G = compute_gradient_matrix_vec(h_work)
    prof = np.correlate(h_work, 1.0 - h_work, mode="full")
    fmax = np.max(prof)

    # Select active lags
    active_tol = 0.08 * fmax
    active_idx = np.where(prof >= fmax - active_tol)[0]
    if len(active_idx) < 5:
        active_idx = np.argsort(prof)[-min(2 * n, 2 * n - 1):]
    if len(active_idx) > 3 * n:
        active_idx = np.argsort(prof)[-3 * n:]

    G_act = G[active_idx, :]

    # Score all pairs
    pair_list = []
    for i in range(n):
        for j in range(n):
            if i == j:
                continue
            diff = G_act[:, i] - G_act[:, j]
            score = np.max(diff)
            if score < -1e-10:
                pair_list.append((score, i, j))

    pair_list.sort(key=lambda x: x[0])

    # If no useful pairs, add random ones
    if len(pair_list) < max_pairs:
        for _ in range(max_pairs):
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
