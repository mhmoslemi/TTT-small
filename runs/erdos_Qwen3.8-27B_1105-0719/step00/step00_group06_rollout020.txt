def erdos_objective(h):
    """Compute the exact C5 objective value."""
    h = np.asarray(h, dtype=np.float64)
    dx = 2.0 / len(h)
    prof = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(prof) * dx)

def raw_profile(h):

def repair_box_sum(h, n):

            break

            # Need to reduce sum

def golden_section_min(f, a, b, max_iter=50):
    """Golden section minimization of f on [a, b]."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0  # golden ratio ~0.618
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if b - a < 1e-10:
            break
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = f(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = f(d)
    t_best = (a + b) / 2.0
    f_best = f(t_best)
    # Also check endpoints
    fa = f(a)
    fb = f(b)
    if fa < f_best:
        t_best, f_best = a, fa
    if fb < f_best:
        t_best, f_best = b, fb
    return t_best, f_best

    pairs = []
    fmax = float(np.max(profile))
    # Active lags: those within active_frac of max
    active_threshold = fmax - active_frac * max(fmax, 1e-10)
    active_indices = np.where(profile >= active_threshold)[0]

    # profile[k] corresponds to lag s = k - (n-1)

        if s == 0:

    scores = np.zeros((n, n))

        if t_hi - t_lo < 1e-10:
            continue
        pairs.append((i, j))

def build_alternating(n):
    """Build a simple alternating 0/1 pattern with exact sum n/2."""
    h = np.zeros(n)
    # Set every other position to 1
    for i in range(0, n, 2):
        h[i] = 1.0
    # This gives sum = n//2, which is exactly n/2 for even n
    if n % 2 != 0:
        # Adjust: we need sum = n/2 which is not integer
        # Set first n//2 positions to 1, last position to 0.5
        h = np.zeros(n)
        h[:n//2] = 1.0
        h[n//2] = 0.5
    return repair_box_sum(h, n)

def build_cyclic(n, period, fill_fraction, rng=None):
    """Build a cyclic pattern with given period and fill fraction."""
    h = np.zeros(n)
    for i in range(n):
        phase = (i % period) / period
        if phase < fill_fraction:
            h[i] = 1.0
    return repair_box_sum(h, n)

Δ_s = t * (grad_s[i] - grad_s[j]) + t² * Q_s

where Q_s is the number of overlap terms where both i and j participate (can be 0, 1, or 2 depending on the geometry).

Actually, Q_s is at most 1 for a given lag, because for a fixed lag s>0, the term h[r+s]*(1-h[r]) involves exactly one pair of adjacent (in the lag-s sense) positions. For both i and j to appear, we need {i,j} = {r, r+s} for some valid r. There's at most one such r (since r = i-s = j or r = j-s = i, and these are the same condition).

Wait, actually there could be up to 2 cases:
- i = r+s and j = r (requires i-j = s)
- i = r and j = r+s (requires j-i = s)

These are mutually exclusive (can't have both i-j=s and j-i=s unless s=0).

For s=0: the term is h[r]*(1-h[r]). Both i and j appear only if i=j, which we exclude. So Q_0 = 0 always.

For s>0: Q_s = 1 if |i-j| = s, else 0. (More precisely, Q_s = 1 if i-j = s or j-i = s, i.e., |i-j| = s.)

Wait, let me double-check. For s > 0, the sum is over r from 0 to n-s-1 of h[r+s]*(1-h[r]).
- i appears as r+s when r = i-s, valid if 0 <= i-s <= n-s-1, i.e., s <= i <= n-1
- j appears as r when r = j, valid if 0 <= j <= n-s-1
- Both appear in same term when {r, r+s} = {i, j}, which means j = i-s (so r=j, r+s=i) or i = j-s (so r=i, r+s=j)
- First case: j = i-s, i.e., i-j = s
- Second case: i = j-s, i.e., j-i = s

So Q_s = 1 if i-j = s or j-i = s (i.e., |i-j| = s), else 0.

Similarly for s < 0 (let a = -s), the sum is over r from 0 to n-a-1 of h[r]*(1-h[r+a]).
- i appears as r when r=i, valid if 0<=i<=n-a-1
- j appears as r+a when r=j-a, valid if 0<=j-a<=n-a-1, i.e., a<=j<=n-1
- Both in same term: {r, r+a} = {i, j}, so j = i+a (i.e., j-i=a=-s) or i = j+a (i.e., i-j=a=-s)
- So |i-j| = |s|

Therefore: Q_s = 1 if |i-j| = |s| (s≠0), else 0. For s=0, Q_0=0.

This means: profile_s(t) = profile_s + t*(grad_s[i]-grad_s[j]) + t² if |i-j|=|s| and s≠0.

Now, the objective is max_s profile_s. Each profile_s(t) is a quadratic with non-negative leading coefficient (0 or 1), so each is convex. The max of convex functions is convex. 

For the golden section search, I can compute the exact objective at any t efficiently:
- Start with the current profile
- For each lag s: add t*(grad_s[i]-grad_s[j]) + (t² if |i-j|=|s| and s≠0)
- Take the max

This is O(n) per evaluation (since there are 2n-1 lags), much faster than O(n²) for the full correlate.

But I can be even faster by computing the profile change analytically. Let me do that.

For a pair (i,j) with move t:
profile_s(t) = profile_s + t * (g_s_i - g_s_j) + t² * Q_s

where g_s_i is the gradient of profile_s wrt h[i], g_s_j wrt h[j], and Q_s is 0 or 1.

I can precompute the gradients for all lags once per pass (O(n²)), then each evaluation is O(n) (just computing the max over all lags of the updated values).

Actually even simpler: for a given pair (i,j):
- grad_diff[s] = grad_s[i] - grad_s[j] for all s
- Q[s] = 1 if |i-j| == |s| and s != 0, else 0

Then profile_s(t) = profile[s] + t*grad_diff[s] + t²*Q[s]

The max over s of these quadratics. Since Q is at most 1 for at most 2 values of s, and 0 elsewhere, most lags are just linear in t.

Let me implement this. The key computation per golden evaluation:
- new_profile = profile + t * grad_diff + t² * Q_vec
- return max(new_profile)

This is O(n) per evaluation (vectorized), so 50 evaluations × O(n) = 50n per pair, vs 50 × O(n²) for direct. For n=70, that's 3500 vs 245000 operations. Big speedup.

def erdos_objective(h):
    h = np.asarray(h, dtype=np.float64)
    dx = 2.0 / len(h)
    prof = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(prof) * dx)

    return np.correlate(h, 1.0 - h, mode="full")

def repair_box_sum(h, n):
    h = np.array(h, dtype=np.float64)
    h[~np.isfinite(h)] = 0.0
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = np.sum(h) - target
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h > 1e-15
            if not np.any(mask):
                break
            dec = diff / np.sum(mask)
            dec = min(dec, np.min(h[mask]))
            h[mask] -= dec
            h = np.clip(h, 0.0, 1.0)
        else:
            mask = h < 1.0 - 1e-15
            if not np.any(mask):
                break
            inc = (-diff) / np.sum(mask)
            inc = min(inc, 1.0 - np.max(h[mask]))
            h[mask] += inc
            h = np.clip(h, 0.0, 1.0)
    h = np.clip(h, 0.0, 1.0)

    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.argmin(h)
            h[idx] = min(1.0, h[idx] + diff)
        else:
            idx = np.argmax(h)
            h[idx] = max(0.0, h[idx] + diff)
    return h

def compute_grad_diff(h, profile, i, j, n):
    """Compute grad_diff[s] = grad_s[i] - grad_s[j] for all lags s."""
    # profile has length 2n-1, index k corresponds to lag s = k - (n-1)
    grad_i = np.zeros(2*n - 1)
    grad_j = np.zeros(2*n - 1)
    
    for k in range(2*n - 1):
        s = k - (n - 1)
        if s == 0:
            gi = 1.0 - 2.0*h[i]
            gj = 1.0 - 2.0*h[j]
        elif s > 0:
            gi = 0.0
            gj = 0.0
            if s <= i <= n-1:
                gi += (1.0 - h[i - s])
            if 0 <= i <= n - s - 1:
                gi -= h[i + s]
            if s <= j <= n-1:
                gj += (1.0 - h[j - s])
            if 0 <= j <= n - s - 1:
                gj -= h[j + s]
        else:
            a = -s
            gi = 0.0
            gj = 0.0
            if 0 <= i <= n - a - 1:
                gi += (1.0 - h[i + a])
            if a <= i <= n-1:
                gi -= h[i - a]
            if 0 <= j <= n - a - 1:
                gj += (1.0 - h[j + a])
            if a <= j <= n-1:
                gj -= h[j - a]
        grad_i[k] = gi
        grad_j[k] = gj
    
    return grad_i - grad_j

def compute_Q_vec(i, j, n):
    """Compute Q[s] for all lags: 1 if |i-j|==|s| and s!=0, else 0."""
    Q = np.zeros(2*n - 1)
    d = abs(i - j)
    if d > 0:
        # lag s = d: index = d + (n-1)
        idx_pos = d + (n - 1)
        if idx_pos < 2*n - 1:
            Q[idx_pos] = 1.0
        # lag s = -d: index = -d + (n-1)
        idx_neg = -d + (n - 1)
        if idx_neg >= 0:
            Q[idx_neg] = 1.0
    return Q

def pair_line_objective(profile, grad_diff, Q_vec, t):
    """Compute max of profile_s + t*grad_diff_s + t^2*Q_s for all s."""
    new_prof = profile + t * grad_diff + t*t * Q_vec
    return float(np.max(new_prof))

def golden_section_min(func, a, b, max_iter=40, tol=1e-10):
    """Golden section minimization."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = func(c)
    fd = func(d)
    for _ in range(max_iter):
        if abs(b - a) < tol:
            break
        if fc <= fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = func(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = func(d)
    t_mid = (a + b) / 2.0
    f_mid = func(t_mid)
    # Check endpoints
    candidates = [(a, func(a)), (b, func(b)), (t_mid, f_mid)]
    best = min(candidates, key=lambda x: x[1])
    return best[0], best[1]

def select_pairs(h, profile, n, rng, max_pairs=40):
    """Select candidate pairs for exchange."""
    fmax = float(np.max(profile))
    # Use active lags near the max
    active_thresh = fmax - 0.05 * fmax
    active_k = np.where(profile >= active_thresh)[0]
    
    # Compute gradient difference for active lags to score pairs
    # grad_diff for pair (i,j) at lag k = grad_k[i] - grad_j[j]
    # We want pairs where the descent direction is most negative

    # = min_k active_grads[k,i] - max_k active_grads[k,j]  (NO! min of differences ≠ difference of mins)
