def _obj_fn(x):
    return x[-1]

def _eq_fn(x):
    return np.sum(x[:-1]) - _g_state['n'] * 0.5

def _ineq_fn(x):
    h = x[:-1]
    t = x[-1]
    c = np.correlate(h, 1.0 - h, mode="full") * _g_state['dx']
    return t - c[_g_state['active']]

def _eval_c5(h):
    dx = 2.0 / len(h)
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return c.max()

def _slsqp_solve(h0, n, dx, active, maxiter=200, ftol=1e-12, t_offset=0.02):
    _g_state['n'] = n
    _g_state['dx'] = dx
    _g_state['active'] = active
    c5_init = _eval_c5(h0)
    x0 = np.concatenate([h0, [c5_init + t_offset]])
    bnds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    cons = [
        {'type': 'eq', 'fun': _eq_fn},
        {'type': 'ineq', 'fun': _ineq_fn}
    ]
    res = minimize(_obj_fn, x0, method='SLSQP', bounds=bnds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
    if res.x is not None:
        h_out = np.clip(res.x[:-1], 0.0, 1.0)
        c5_out = _eval_c5(h_out)
        if c5_out <= res.x[-1] + 1e-8:
            return h_out, c5_out
        return h0, c5_init
    return h0, c5_init

def _try_n_points(h_source, target_n, dx_target):
    """Interpolate source h to target_n points and re-normalize."""
    src_n = len(h_source)
    x_src = np.linspace(0, 2, src_n, endpoint=False)
    x_tgt = np.linspace(0, 2, target_n, endpoint=False)
    h_interp = np.interp(x_tgt, x_src, h_source, period=0).astype(float)
    h_interp = np.clip(h_interp, 0.0, 1.0)
    s = np.sum(h_interp)
    if s > 1e-10:
        h_interp *= (target_n * 0.5) / s
        h_interp = np.clip(h_interp, 0.0, 1.0)
    return h_interp

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t_start = time.time()

    def _time_ok(min_remaining):
        return (time.time() - t_start) < (budget_s - min_remaining)

    # Get initial values
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float)
    else:
        h_init = np.full(63, 0.5)

    n = len(h_init)
    dx = 2.0 / n
    best_h = h_init.copy()
    best_c5 = _eval_c5(best_h)
    all_indices = np.arange(2 * n - 1)

    # Phase 1: Full SLSQP from initial point
    if _time_ok(60):
        h_new, c5_new = _slsqp_solve(best_h, n, dx, all_indices, maxiter=300, ftol=1e-12)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    # Phase 2: Active set refinement (iterate, growing active set)
    for outer in range(8):
        if not _time_ok(40):
            break
        c = np.correlate(best_h, 1.0 - best_h, mode="full") * dx
        K = min(10 + outer * 3, 2 * n - 1)
        active = np.argsort(c)[-K:]
        h_new, c5_new = _slsqp_solve(best_h, n, dx, active, maxiter=200, ftol=1e-12)
        if c5_new < best_c5 - 1e-10:
            best_h = h_new
            best_c5 = c5_new
        else:
            # Check if solution is still feasible for all lags
            c_check = np.correlate(h_new, 1.0 - h_new, mode="full") * dx
            if c_check.max() <= best_c5 + 1e-10:
                break
            else:
                best_h = h_new
                best_c5 = c_check.max()

    # Phase 3: Random restarts with active set
    n_restarts = 60
    for i in range(n_restarts):
        if not _time_ok(20):
            break

        # Generate perturbed starting point
        if i < 10:
            scale = 0.005 + 0.002 * i
            h_pert = best_h + np.random.randn(n) * scale
        elif i < 25:
            scale = 0.02 + 0.005 * (i - 10)
            h_pert = best_h + np.random.randn(n) * scale
        elif i < 40:
            h_pert = best_h + np.random.randn(n) * 0.08
        elif i < 50:
            h_pert = 0.5 * best_h + 0.5 * np.random.rand(n)
        else:
            h_pert = np.random.rand(n)

        h_pert = np.clip(h_pert, 0.0, 1.0)

        # Active set from perturbed point
        c_pert = np.correlate(h_pert, 1.0 - h_pert, mode="full") * dx
        K = 12
        active = np.argsort(c_pert)[-K:]

        h_new, c5_new = _slsqp_solve(h_pert, n, dx, active, maxiter=150, ftol=1e-10)
        if c5_new < best_c5:
            # Verify against all lags
            c_full = np.correlate(h_new, 1.0 - h_new, mode="full") * dx
            c5_full = c_full.max()
            if c5_full < best_c5:
                best_h = h_new
                best_c5 = c5_full

    # Phase 4: Try different n values
    for target_n in [32, 48, 80, 96, 128]:
        if not _time_ok(30):
            break
        h_try = _try_n_points(best_h, target_n, 2.0 / target_n)
        n_try = target_n
        dx_try = 2.0 / n_try
        c5_try = _eval_c5(h_try)

        if c5_try < best_c5 - 0.001:
            # Promising, optimize at this n
            all_idx_try = np.arange(2 * n_try - 1)
            h_opt, c5_opt = _slsqp_solve(h_try, n_try, dx_try, all_idx_try, maxiter=200, ftol=1e-11)
            if c5_opt < best_c5:
                best_h = h_opt
                best_c5 = c5_opt
                n = n_try
                dx = dx_try

    # Final active set polish
    if _time_ok(20):
        c = np.correlate(best_h, 1.0 - best_h, mode="full") * dx
        K = 25
        active = np.argsort(c)[-K:]
        h_new, c5_new = _slsqp_solve(best_h, len(best_h), 2.0/len(best_h), active, maxiter=100, ftol=1e-13)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    # Final verification
    final_c5 = _eval_c5(best_h)
    n_final = len(best_h)
    return best_h, final_c5, n_final
