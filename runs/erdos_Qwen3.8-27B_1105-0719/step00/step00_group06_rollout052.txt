def coord_descent(h_init, n_pts, deadline, max_iter=2000):
    """Coordinate descent on the smoothed objective."""
    h = h_init.copy()
    c5_cur, _ = exact_objective(h)
    tau = 0.005
    
    for it in range(max_iter):
        if time.time() > deadline - 3:
            break
        tau = max(0.0005, tau * 0.995)
        improved = False
        for j in range(n_pts):
            # Try small moves in coordinate j
            base = h[j]
            for delta in [0.01, -0.01, 0.005, -0.005, 0.02, -0.02]:
                h[j] = np.clip(base + delta, 0.0, 1.0)
                # Adjust sum
                h = project_box_sum(h)
                c5_new, _ = exact_objective(h)
                if c5_new < c5_cur - 1e-13:
                    c5_cur = c5_new
                    improved = True
                else:
                    h[j] = base
                    h = project_box_sum(h)
            # Restore
            h[j] = base
        if not improved:
            break
    return h, c5_cur

def smooth_grad_analytical(h, tau):
    n = len(h)
    dx = 2.0 / n
    raw_profile = np.correlate(h, 1.0 - h, mode='full')
    pmax = np.max(raw_profile)
    scaled = np.clip((raw_profile - pmax) / tau, -50, 50)
    exp_scaled = np.exp(scaled)
    wsum = np.sum(exp_scaled)
    weights = exp_scaled / wsum  # Shape (2n-1,)
    
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        wm = weights[m]
        if wm < 1e-10:
            continue
        k = m - (n - 1)
        # n_valid range: [max(0,-k), min(n-1, n-1-k)]
        lo = max(0, -k)
        hi = min(n - 1, n - 1 - k)
        
        # Contribution 1: (1-h[j-k]) for j where j-k in [lo, hi]
        # i.e., j in [lo+k, hi+k]
        j1_lo = max(0, lo + k)
        j1_hi = min(n - 1, hi + k)
        if j1_lo <= j1_hi:
            idx = np.arange(j1_lo, j1_hi + 1)
            grad[idx] += wm * (1.0 - h[idx - k])
        
        # Contribution 2: -h[j+k] for j where j in [lo, hi] AND j+k in [0, n-1]
        j2_lo = max(lo, -k)  # j >= lo AND j >= -k (from j+k >= 0)
        j2_hi = min(hi, n - 1 - k)  # j <= hi AND j <= n-1-k (from j+k <= n-1)
        j2_lo = max(0, j2_lo)
        j2_hi = min(n - 1, j2_hi)
        if j2_lo <= j2_hi:
            idx = np.arange(j2_lo
