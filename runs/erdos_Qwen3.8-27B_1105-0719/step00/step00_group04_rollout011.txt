def pair_sublevel_feasible(alpha, beta, gamma, lo, hi, T, eps=1e-10):
    """Check if there exists a in [lo,hi] such that all q_k(a) <= T."""
    # Track feasible set as list of intervals
    feasible = [(lo, hi)]
    
    n_lags = len(alpha)
    for k in range(n_lags):
        a_k = alpha[k]
        b_k = beta[k]
        g_k = gamma[k]
        
        new_feasible = []
        for (iv_lo, iv_hi) in feasible:
            if abs(a_k) < eps:
                # Linear: b_k * a + g_k <= T
                if abs(b_k) < eps:
                    # Constant: need g_k <= T
                    if g_k > T + eps:
                        continue  # This interval is entirely infeasible
                    new_feasible.append((iv_lo, iv_hi))
                elif b_k > 0:
                    # a <= (T - g_k) / b_k
                    upper = (T - g_k) / b_k
                    if upper < iv_lo - eps:
                        continue
                    new_feasible.append((iv_lo, min(iv_hi, upper)))
                else:
                    # a >= (T - g_k) / b_k
                    lower = (T - g_k) / b_k
                    if lower > iv_hi + eps:
                        continue
                    new_feasible.append((max(iv_lo, lower), iv_hi))
            elif a_k > 0:
                # Convex: sublevel is [r1, r2] if D >= 0
                D = b_k * b_k - 4.0 * a_k * (g_k - T)
                if D < -eps:
                    continue  # No solution
                D = max(0.0, D)
                sqrt_D = np.sqrt(D)
                r1 = (-b_k - sqrt_D) / (2.0 * a_k)
                r2 = (-b_k + sqrt_D) / (2.0 * a_k)
                # Sublevel is [r1, r2]
                ilo = max(iv_lo, r1)
                ihi = min(iv_hi, r2)
                if ilo <= ihi + eps:
                    new_feasible.append((ilo, ihi))
            else:

                D = b_k * b_k - 4.0 * a_k * (g_k - T)
                if D < -eps:
                    # Always feasible (parabola always below T)
                    new_feasible.append((iv_lo, iv_hi))
                    continue
                D = max(0.0, D)
                sqrt_D = np.sqrt(D)

                if iv_hi >= r2:
                    new_feasible.append((max(iv_lo, r2), iv_hi))
        
        feasible = new_feasible
        if not feasible:
            return False

        mid = (lo + hi) * 0.5
        if np.sum(np.clip(v + mid, 0.0, 1.0)) > target:
            hi = mid
        else:
            lo = mid
    result = np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - result.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(result)
        else:
            order = np.argsort(-result)
        for i in order:
            if diff > 0:
                adj = min(diff, 1.0 - result[i])
            else:
                adj = max(diff, -result[i])
            if abs(adj) < 1e-15:
                continue
            result[i] += adj
            diff -= adj
            if abs(diff) < 1e-14:
                break
    return result

def pair_optimize(h, ip, iq, max_bisect=45):
    """Optimize pair (ip, iq) preserving sum. Returns (new_h, new_score) or (None, None)."""
    n = len(h)
    s = h[ip] + h[iq]
    lo = max(0.0, s - 1.0)
    hi = min(1.0, s)
    if hi - lo < 1e-12:
        return None, None
    
    # Evaluate at anchors for polynomial fit
    h0 = h.copy()
    h0[ip] = 0.0
    h0[iq] = s
    h1 = h.copy()
    h1[ip] = 0.5
    h1[iq] = s - 0.5
    h2 = h.copy()
    h2[ip] = 1.0
    h2[iq] = s - 1.0
    
    p0 = exact_profile(h0)
    p1 = exact_profile(h1)
    p2 = exact_profile(h2)
    
    # Quadratic coefficients: q(a) = alpha*a^2 + beta*a + gamma
    # q(0) = p0 = gamma
    # q(0.5) = p1 = 0.25*alpha + 0.5*beta + gamma
    # q(1) = p2 = alpha + beta + gamma
    gamma = p0
    # alpha + beta = p2 - p0 ... (i)
    # 0.25*alpha + 0.5*beta = p1 - p0 ... (ii)
    # From (i): beta = p2 - p0 - alpha
    # Sub into (ii): 0.25*alpha + 0.5*(p2-p0-alpha) = p1-p0
    # 0.25*alpha + 0.5*p2 - 0.5*p0 - 0.5*alpha = p1 - p0
    # -0.25*alpha = p1 - p0 - 0.5*p2 + 0.5*p0 = p1 - 0.5*p2 - 0.5*p0
    # alpha = -4*(p1 - 0.5*p2 - 0.5*p0) = 2*p2 - 4*p1 + 2*p0
    alpha = 2.0 * p2 - 4.0 * p1 + 2.0 * p0
    beta = p2 - p0 - alpha  # = p2 - p0 - 2*p2 + 4*p1 - 2*p0 = 4*p1 - p2 - 3*p0
    
    # Verify: beta = 4*p1 - p2 - 3*p0
    # Check at a=1: alpha + beta + gamma = (2*p2-4*p1+2*p0) + (4*p1-p2-3*p0) + p0 = p2 - p0 + p0 = p2 ✓
    # Check at a=0: gamma = p0 ✓
    
    # Bisection on T
    # Upper bound: evaluate at a few points in [lo, hi]
    test_points = np.linspace(lo, hi, 10)
    best_upper = float('inf')
    for a in test_points:
        h_t = h.copy()
        h_t[ip] = a
        h_t[iq] = s - a
        best_upper = min(best_upper, exact_score(h_t))
    
    T_hi = best_upper
    T_lo = 0.0
    
    for _ in range(max_bisect):
        T_mid = (T_lo + T_hi) * 0.5
        if _sublevel_feasible(alpha, beta, gamma, lo, hi, T_mid):
            T_hi = T_mid
        else:
            T_lo = T_mid
    
    # Find best a achieving T_hi
    best_a = _find_best_a(alpha, beta, gamma, lo, hi, T_hi)
    
    if best_a is None:
        # Fallback: use the best from grid
        for a in test_points:
            h_t = h.copy()
            h_t[ip] = a
            h_t[iq] = s - a
            sc = exact_score(h_t)
            if sc < best_upper:
                best_upper = sc
                best_a = a
    
    if best_a is None:
        return None, None
    
    h_new = h.copy()
    h_new[ip] = best_a
    h_new[iq] = s - best_a
    new_score = exact_score(h_new)
    return h_new, new_score

def _sublevel_feasible(alpha, beta, gamma, lo, hi, T, eps=1e-12):
    """Check if ∃ a in [lo,hi] with q_k(a) ≤ T for all k."""
    intervals = [(lo, hi)]
    n_lags = len(alpha)
    
    for k in range(n_lags):
        ak = alpha[k]
        bk = beta[k]
        gk = gamma[k]
        new_intervals = []
        
        for (vlo, vhi) in intervals:
            if abs(ak) < eps:
                # Linear or constant
                if abs(bk) < eps:
                    if gk > T + eps:
                        continue
                    new_intervals.append((vlo, vhi))
                elif bk > 0:
                    upper = (T - gk) / bk
                    if upper < vlo - eps:
                        continue
                    new_intervals.append((vlo, min(vhi, upper)))
                else:
                    lower = (T - gk) / bk
                    if lower > vhi + eps:
                        continue
                    new_intervals.append((max(vlo, lower), vhi))
            elif ak > 0:
                # Convex: sublevel [r1, r2]
                D = bk*bk - 4.0*ak*(gk - T)
                if D < -eps:
                    continue
                D = max(0.0, D)
                sqrtD = np.sqrt(D)
                r1 = (-bk - sqrtD) / (2.0 * ak)
                r2 = (-bk + sqrtD) / (2.0 * ak)
                ilo = max(vlo, r1)
                ihi = min(vhi, r2)
                if ilo <= ihi + eps:
                    new_intervals.append((ilo, ihi))
            else:
                # Concave: sublevel (-∞,r1] ∪ [r2,∞)
                D = bk*bk - 4.0*ak*(gk - T)
                if D < -eps:
                    # Always feasible
                    new_intervals.append((vlo, vhi))
                    continue
                D = max(0.0, D)
                sqrtD = np.sqrt(D)
                r1 = (-bk - sqrtD) / (2.0 * ak)
                r2 = (-bk + sqrtD) / (2.0 * ak)
                if r1 > r2:
                    r1, r2 = r2, r1
                # (-∞, r1] ∩ [vlo, vhi]
                if vlo <= r1:
                    new_intervals.append((vlo, min(vhi, r1)))
                # [r2, ∞) ∩ [vlo, vhi]
                if vhi >= r2:
                    new_intervals.append((max(vlo, r2), vhi))
        
        intervals = new_intervals
        if not intervals:
            return False
    
    return len(intervals) > 0

def _find_best_a(alpha, beta, gamma, lo, hi, T, eps=1e-12):
    """Find a in [lo,hi] achieving all q_k(a) ≤ T."""

            elif ak > 0:

                ilo = max(vlo, r1)
                ihi = min(vhi, r2)
                if ilo <= ihi + eps:

    # Pick the midpoint of the first interval
    a = (intervals[0][0] + intervals[0][1]) * 0.5
    a = max(lo, min(hi, a))
    return a

    c_sym = exact_score(h_sym)
    if c_sym < best_c:

    # Main optimization loop

    no_improve_passes = 0

        improved = False
        
        # Choose pairs for this pass
        if pass_count % 3 == 1:
            # Adjacent pairs
            pairs = [(i, i+1) for i in range(n-1)]
        elif pass_count % 3 == 2:
            # Every other
            pairs = [(i, i+2) for i in range(n-2)]
        else:
            # Random pairs
            pairs = [(int(rng.integers(0, n)), int(rng.integers(0, n))) for _ in range(n)]
            pairs = [(a, b) for a, b in pairs if a != b]
        
        for ip, iq in pairs:
            if time.perf_counter() > deadline:
                break
            if ip == iq:
                continue
            
            h_new, c_new = pair_optimize(h, ip, iq)
            if h_new is not None and c_new < best_c - tol:
                h = h_new
                best_c = c_new
                best_h = h_new.copy()
                improved = True
        
        if not improved:
            no_improve_passes += 1
            if no_improve_passes >= 3:
                # Try a different strategy
                no_improve_passes = 0
                # Try random perturbation + re-optimization
                h_pert = h + rng.normal(0, 0.01, n)
                h_pert = project_feasible(h_pert, target)
                c_pert = exact_score(h_pert)
                if c_pert < best_c:
                    h = h_pert
                    best_c = c_pert
                    best_h = h_pert.copy()
                # Re-optimize the perturbed version
                for _ in range(n):
                    if time.perf_counter() > deadline:
                        break
                    ip = int(rng.integers(0, n))
                    iq = int(rng.integers(0, n))
                    if ip == iq:
                        continue
                    h_new, c_new = pair_optimize(h, ip, iq)
                    if h_new is not None and c_new < best_c - tol:
                        h = h_new
                        best_c = c_new
                        best_h = h_new.copy()
    
    # Final verification
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_feasible(best_h, target)
    final_c = exact_score(best_h)
    
    return best_h, final_c, len(best_h)

    if hi - lo < 1e-12:

    best_score = float('inf')
    
    for a in np.linspace(lo, hi, n_grid):
        h_t = h.copy()
        h_t[ip] = a

            best_score = sc
            best_a = a

        # Refine locally
        delta = (hi - lo) / n_grid
        a_lo = max(lo, best_a - delta)
        a_hi = min(hi, best_a + delta)
        invphi = (np.sqrt(5.0) - 1.0) / 2.0
        for _ in range(30):

            if s1 < s2:

                best_score = min(best_score, s1)
                best_a = a1
            else:
                a_lo = a1
                best_score = min(best_score, s2)
                best_a = a2

- A crossing point of a linear and a quadratic function
