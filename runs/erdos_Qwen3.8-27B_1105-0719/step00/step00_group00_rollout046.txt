def evaluate_c5(h):
    """Compute C5 for a given h vector."""
    dx = 2.0 / len(h)
    q = 1.0 - h
    corr = np.correlate(h, q, mode='full')
    return dx * np.max(corr)

def compute_correlations(h):
    """Compute all correlation values."""
    q = 1.0 - h
    return np.correlate(h, q, mode='full')

def smooth_objective(params, beta):
    """Log-sum-exp smooth approximation of max correlation."""
    n = len(params)
    # Reconstruct h from params (first n-1 are free, last is determined)
    h = np.zeros(n)
    h[:n-1] = params
    h[n-1] = n/2.0 - np.sum(params)
    
    corr = compute_correlations(h)
    dx = 2.0 / n
    # Clip for numerical stability
    z = beta * corr * dx
    z_max = np.max(z)
    log_sum = z_max + np.log(np.sum(np.exp(z - z_max)))
    return log_sum / beta

def smooth_gradient(params, beta):
    """Gradient of the smooth objective."""
    n = len(params)
    h = np.zeros(n)
    h[:n-1] = params
    h[n-1] = n/2.0 - np.sum(params)
    
    corr = compute_correlations(h)
    dx = 2.0 / n
    
    z = beta * corr * dx
    z_max = np.max(z)
    weights = np.exp(z - z_max) / np.sum(np.exp(z - z_max))
    
    # Compute gradient of each correlation w.r.t. h
    # corr[k] for k=0,...,2n-2, lag d = k - (n-1)

    #   dcorr/dh[m] = (1-h[m+d']) if m<=n-1-d' else 0, minus h[m-d'] if m>=d' else 0

    grad_h = np.zeros(n)
    
    for k in range(n_corr):

        if d >= 0:

            # d/dh[m] = (1-h[m-d])*[m>=d] - h[m+d]*[m+d<=n-1]

    grad_params = np.zeros(n-1)
    for j in range(n-1):
        grad_params[j] = grad_h[j] - grad_h[n-1]
    
    return grad_params

    """Project y onto {0<=h<=1, sum(h)=n/2}."""
    target = n / 2.0
    # Bisection on threshold t: h_i = clip(y_i - t, 0, 1)
    lo = np.min(y) - 1.0
    hi = np.max(y) + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        h = np.clip(y - mid, 0.0, 1.0)
        s = np.sum(h)
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - (lo+hi)/2.0, 0.0, 1.0)
    # Fix tiny residual
    residual = target - np.sum(h)
    if abs(residual) > 1e-10:
        # Distribute residual to elements not at bounds
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.sum(mask) > 0:
            h[mask] += residual / np.sum(mask)

                for t in ts:
                    h_try = h.copy()
                    h_try[i] += t
                    h_try[j] -= t
                    c5_try = evaluate_c5(h_try)
                    if c5_try < best_val - 1e-13:
                        best_val = c5_try
                        best_t = t
                if best_t != 0.0:
                    h[i] += best_t
                    h[j] -= best_t
                    best_c5 = best_val
                    improved = True
        if not improved:
            break
    h = project_feasible(h, n)
    return h, evaluate_c5(h)

def random_feasible(n, rng):
    h = np.full(n, 0.5)
    h += rng.random(n) * 0.4 - 0.2
    return project_feasible(h, n)

def perturb_feasible(h, rng, scale=0.05):
    n = len(h)
    h_try = h + rng.random(n) * scale - scale / 2.0
    return project_feasible(h_try, n)

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    start_time = time.time()
    rng = np.random.default_rng(seed)
    
    best_h = None
    best_c5 = np.inf
    
    def track(h):
        nonlocal best_h, best_c5
        h = np.clip(h, 0, 1)
        n = len(h)
        h = project_feasible(h, n)
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        return c5

    # Phase 1: Optimize from initial_h_values
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
        n0 = len(h0)
        h0 = project_feasible(h0, n0)
        track(h0)
        
        # Annealing with moderate betas
        betas1 = [5, 10, 20, 50, 100]
        h_a = anneal(h0, betas1, max_iter=300)
        track(h_a)
        
        # Continue with higher betas
        betas2 = [200, 500, 1000]
        h_b = anneal(h_a, betas2, max_iter=400)
        track(h_b)
        
        # Pair polish
        if time.time() - start_time < budget_s * 0.35:
            h_c, _ = pair_polish(h_b, n_sweeps=50, n_grid=15)
            track(h_c)
    
    # Phase 2: Perturbation + re-optimization around best
    if best_h is not None and time.time() - start_time < budget_s * 0.55:
        n_best = len(best_h)
        for trial in range(8):
            if time.time() - start_time > budget_s * 0.55:
                break
            h_pert = perturb_feasible(best_h, rng, scale=0.03)
            h_opt = anneal(h_pert, [50, 100, 300], max_iter=200)
            track(h_opt)
            if time.time() - start_time < budget_s * 0.55:
                h_pol, _ = pair_polish(h_opt, n_sweeps=20, n_grid=10)
                track(h_pol)
    
    # Phase 3: Different resolutions
    if time.time() - start_time < budget_s * 0.75:
        for n_test in [35, 40, 50, 55, 60]:
            if time.time() - start_time > budget_s * 0.8:
                break
            if best_h is not None:
                h_interp = np.interp(
                    np.linspace(0, 2, n_test, endpoint=False),
                    np.linspace(0, 2, len(best_h), endpoint=False),
                    best_h
                )
            else:
                h_interp = np.full(n_test, 0.5)
            h_interp = project_feasible(h_interp, n_test)
            track(h_interp)
            
            h_opt = anneal(h_interp, [10, 50, 200], max_iter=200)
            track(h_opt)
            
            if time.time() - start_time < budget_s * 0.8:
                h_pol, _ = pair_polish(h_opt, n_sweeps=30, n_grid=12)
                track(h_pol)
    
    # Phase 4: Random multi-start
    if time.time() - start_time < budget_s * 0.88:
        for n_test in [45, 50]:
            for trial in range(6):
                if time.time() - start_time > budget_s * 0.9:
                    break
                h_rand = random_feasible(n_test, rng)
                h_opt = anneal(h_rand, [10, 50, 200], max_iter=150)
                track(h_opt)
    
    # Phase 5: Final polish on best
    if best_h is not None and time.time() - start_time < budget_s * 0.95:
        h_final, _ = pair_polish(best_h, n_sweeps=60, n_grid=20)
        track(h_final)
        
        # One more SLSQP pass at very high beta
        h_final2 = slsqp_optimize(h_final, 2000, max_iter=500)
        track(h_final2)
        h_final3, _ = pair_polish(h_final2, n_sweeps=30, n_grid=15)
        track(h_final3)
    
    # Ensure we have a valid result
    if best_h is None:
        best_h = np.full(45, 0.5)
        best_c5 = evaluate_c5(best_h)
    
    # Final check
    final_c5 = evaluate_c5(best_h)
    assert best_c5 <= final_c5 + 1e-10, "Inconsistent best_c5"
    
    return best_h, best_c5, len(best_h)
