def exact_objective(h):
    """Compute C5 = max_k correlate(h, 1-h, full) * dx"""
    n = len(h)
    if n == 0:
        return (float('inf'), np.array([]))
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    return (float(np.max(profile)), profile / dx)  # return raw profile too

def project_box_sum(h, n):
    """Project h onto {0<=x<=1, sum(x)=n/2}"""
    x = np.array(h, dtype=np.float64).copy()
    target = n / 2.0
    # Handle NaN/Inf
    x[~np.isfinite(x)] = 0.0
    # Clip
    x = np.clip(x, 0.0, 1.0)
    # Water-filling to hit target
    # If sum > target, reduce
    for _ in range(100):
        s = x.sum()
        diff = s - target
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            # Need to reduce by diff
            mask = x > 0
            count = mask.sum()
            if count == 0:
                break
            step = min(diff / count, 1.0)
            x[mask] -= min(step, diff / count)
            x = np.clip(x, 0.0, 1.0)
        else:

            mask = x < 1
            count = mask.sum()
            if count == 0:

            step = min(-diff / count, 1.0)
            x[mask] += step
            x = np.clip(x, 0.0, 1.0)

    s = x.sum()
    diff = s - target

        if count > 0:

            x = np.clip(x, 0.0, 1.0)

                    zeros[j] = oi
