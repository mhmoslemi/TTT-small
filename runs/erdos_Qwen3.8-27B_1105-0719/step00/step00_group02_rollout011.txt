import numpy as np
import time

def exact_score(h, n):
    """Compute C5 = max_k corr(h, 1-h, full)[k] * dx"""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def feasible_interval_pair(h, i, j):
    """For direction e_j - e_i, find t range where h+t*u stays in [0,1]^n"""
    # h_i - t >= 0 -> t <= h_i
    # h_i - t <= 1 -> t >= h_i - 1
    # h_j + t >= 0 -> t >= -h_j
    # h_j + t <= 1 -> t <= 1 - h_j
    t_lo = max(0.0, h[i]-1.0, -h[j])
    t_hi = min(h[i], 1.0-h[j], 1.0)
    return t_lo, t_hi

def score_at_t(h, i, j, t):
    """Score when transferring t from i to j"""
    h_new = h.copy()
    h_new[i] -= t
    h_new[j] += t
    n = len(h)
    return exact_score(h_new, n)

```python
def compute_pair_correction(h, i, j):
    """Compute α and β vectors for pair (i,j) transfer.
    Returns alpha (length 2n-1) and beta (length 2n-1) such that
    new_corr[p] = old_corr[p] + alpha[p]*t + beta[p]*t^2
    """
    n = len(h)
    alpha = np.zeros(2*n-1)
    beta = np.zeros(2*n-1)
    
    for p in range(2*n-1):
        ell = p - (n-1)  # lag
        # Affected m values: i, i-ell, j, j-ell
        # For each affected m, compute the exact change in h[m+ell]*(1-h[m])
        
        # m = i: h[m+ell] may or may not be changed, h[m] = h[i] is changed
        # m = i-ell: h[m+ell] = h[i] is changed, h[m] = h[i-ell] may or may not be
        # m = j: similar
        # m = j-ell: similar
        
        affected = set()
        if 0 <= i < n:
            affected.add(i)  # m = i
        if 0 <= i-ell < n:
            affected.add(i-ell)  # m = i-ell (so m+ell = i)
        if 0 <= j < n:
            affected.add(j)  # m = j
        if 0 <= j-ell < n:
            affected.add(j-ell)  # m = j-ell (so m+ell = j)
        
        for m in affected:
            m_pl = m + ell  # m + lag
            if not (0 <= m_pl < n):
                continue
            # Original: h[m+ell] * (1-h[m])
            # New: h'[m+ell] * (1-h'[m])
            # h'[x] = h[x] - t if x==i, h[x] + t if x==j, h[x] otherwise
            
            hm = h[m]  # original h[m]
            hmp = h[m_pl]  # original h[m+ell]
            
            # Determine changes
            dh_m = 0.0   # change in h[m]
            dh_mp = 0.0  # change in h[m+ell]
            if m == i:
                dh_m = -1.0  # coefficient of t
            elif m == j:
                dh_m = 1.0
            if m_pl == i:
                dh_mp = -1.0
            elif m_pl == j:
                dh_mp = 1.0
            
            # New product: (h_mp + t*dh_mp) * (1 - hm - t*dh_m)
            # = h_mp*(1-hm) + t*dh_mp*(1-hm) - t*dh_m*h_mp - t²*dh_mp*dh_m
            # Change: t*(dh_mp*(1-hm) - dh_m*h_mp) - t²*dh_mp*dh_m
            
            alpha[p] += dh_mp*(1-hm) - dh_m*h_mp
            beta[p] += -dh_mp*dh_m
    
    return alpha, beta
```
