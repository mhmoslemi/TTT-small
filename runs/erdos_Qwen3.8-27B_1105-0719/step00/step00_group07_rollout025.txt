import time

def compute_raw(h):
    return np.correlate(h, 1.0 - h, mode="full")

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    raw = compute_raw(h)
    return np.max(raw) * dx

def project_onto_set(y, target):
    """Project y onto {0 <= x <= 1, sum(x) = target}."""
    n = len(y)
    # Bisection on lambda
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        x = np.clip(y - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix floating point residual
    r = target - x.sum()
    if abs(r) > 1e-10:
        if r > 0:
            for i in range(n):
                if r <= 0:
                    break
                slack = 1.0 - x[i]
                add = min(r, slack)
                x[i] += add
                r -= add
        else:
            for i in range(n):
                if r >= 0:
                    break

import numpy as np
import time


def compute_raw(h):
    return np.correlate(h, 1.0 - h, mode="full")


def compute_c5(h):
    n = len(h)
    return float(np.max(compute_raw(h))) * 2.0 / n


def project_onto_set(y, target):
    n = len(y)
    if target < 0:
        target = 0.0
    elif target > n:
        target = float(n)
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y)) + 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        x = np.clip(y - mid, 0.0, 1.0)
        if float(np.sum(x)) > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)
    r = target - float(np.sum(x))
    if r > 1e-12:
        for i in range(n):
            if r <= 1e-14:
                break
            slack = 1.0 - x[i]
            if slack > 0:
                add = min(r, slack)
                x[i] += add
                r -= add
    elif r < -1e-12:
        for i in range(n):
            if r >= -1e-14:
                break
            if x[i] > 0:
                sub = min(-r, x[i])
                x[i] -= sub
                r += sub
    return x


def lse_gradient(h, beta):
    n = len(h)
    raw = compute_raw(h)
    m = float(np.max(raw))
    shifted = beta * (raw - m)
    exp_s = np.exp(shifted)
    se = float(np.sum(exp_s))
    w = exp_s / se
    grad = np.zeros(n)
    for p in range(n):
        g = 0.0
        for k in range(-(n - 1), n):
            wt = w[k + n - 1]
            if wt < 1e-14:
                continue
            gk = 0.0
            a = p - k
            if 0 <= a < n:
                gk += 1.0 - h[a]
            b = p + k
            if 0 <= b < n:
                gk -= h[b]
            g += wt * gk
        grad[p] = g
    return grad


def optimize_pg(h0, start_time, deadline, max_iters=5000):
    h = h0.astype(float).copy()
    n = len(h)
    target = n * 0.5
    cur_c5 = compute_c5(h)
    best_c5 = cur_c5
    best_h = h.copy()
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        progress = min(float(it) / float(max_iters), 1.0)
        beta = 10.0 + 290.0 * progress
        grad = lse_gradient(h, beta)
        grad = grad - np.mean(grad)
        lr = 0.03 * (1.0 - 0.7 * progress) + 0.003
        h_new = project_onto_set(h - lr * grad, target)
        c5_new = compute_c5(h_new)
        h = h_new
        cur_c5 = c5_new
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
    return best_h, best_c5


def optimize_polish(h0, rng, start_time, deadline, max_iters=30000):
    h = h0.astype(float).copy()
    n = len(h)
    target = n * 0.5
    best_c5 = compute_c5(h)
    best_h = h.copy()
    cur_c5 = best_c5
    temp = 0.005
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        temp = max(temp * 0.9997, 0.0003)
        perturb = rng.randn(n) * temp
        h_new = project_onto_set(h + perturb, target)
        c5_new = compute_c5(h_new)
        if c5_new <= cur_c5 or (rng.rand() < np.exp((cur_c5 - c5_new) / temp)):
            h = h_new
            cur_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
    return best_h, best_c5


def binary_sa(n, rng, start_time, deadline, max_iters=80000):
    h = np.zeros(n)
    h[rng.choice(n, n // 2, replace=False)] = 1.0
    cur_c5 = compute_c5(h)
    best_c5 = cur_c5
    best_h = h.copy()
    temp = 0.002
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j or h[i] == h[j]:
            continue
        h[i], h[j] = h[j], h[i]
        c5_new = compute_c5(h)
        if c5_new <= cur_c5 or (rng.rand() < np.exp((cur_c5 - c5_new) / max(temp, 1e-9))):
            cur_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
        else:
            h[i], h[j] = h[j], h[i]
        temp = max(temp * 0.9999, 1e-6)
    return best_h, best_c5


def make_random_init(n, rng):

    return project_onto_set(h, n * 0.5)
