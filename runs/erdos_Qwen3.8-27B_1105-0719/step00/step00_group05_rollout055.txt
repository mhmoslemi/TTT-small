    if delta <= 0:
       order = np.argsort(h)  # smallest first have slack? For positive delta use smallest h -> most room to 1.
       signs = 1.0

    remains=delta

      h[idx] += signs*step
      remains -= step

def finish_sum(h, T):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    delta = T - float(np.sum(h))
    if delta > 1e-13:
        order = np.argsort(h)
        for idx in order:
            room = 1.0 - h[idx]
            if room <= 0.0:
                continue
            step = min(room, delta)
            h[idx] += step
            delta -= step
            if delta <= 1e-14:
                break
    elif delta < -1e-13:
        order = np.argsort(-h)
        for idx in order:
            room = h[idx]
            if room <= 0.0:
                continue
            step = min(room, -delta)
            h[idx] -= step
            delta += step
            if delta >= -1e-14:
                break
    # final correction
    delta2 = T - float(np.sum(h))
    if abs(delta2) > 1e-13:
        if delta2 > 0:
            order = np.argsort(h)
        else:
            order = np.argsort(-h)
        for idx in order:
            val = h[idx] + delta2
            if -1e-15 <= val <= 1.0 + 1e-15:
                h[idx] = val
                break
    h = np.clip(h, 0.0, 1.0)
    return h

class BudgetExceeded(Exception):

def time_left():
    return max(0.0, RUN_DEADLINE - time.time())


def stable_sigmoid(t):
    t = np.asarray(t, dtype=float)
    out = np.empty(t.shape, dtype=float)
    pos = t >= 0
    out[pos] = 1.0 / (1.0 + np.exp(-t[pos]))
    neg = ~pos
    et = np.exp(t[neg])
    out[neg] = et / (1.0 + et)
    return out


def stable_logit(h):
    eps = 1e-10
    h = np.asarray(h, dtype=float)
    h = np.clip(h, eps, 1.0 - eps)
    return np.log(h / (1.0 - h))


def finish_sum(h, T):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    delta = T - float(np.sum(h))

    if delta > 1e-13:
        order = np.argsort(h)
        for idx in order:
            room = 1.0 - h[idx]
            if room <= 0.0:
                continue
            step = min(room, delta)
            h[idx] += step
            delta -= step
            if delta <= 1e-14:
                break
    elif delta < -1e-13:
        order = np.argsort(-h)
        for idx in order:
            room = h[idx]
            if room <= 0.0:
                continue
            step = min(room, -delta)
            h[idx] -= step
            delta += step
            if delta >= -1e-14:
                break

    delta2 = T - float(np.sum(h))
    if abs(delta2) > 1e-13:
        if delta2 > 0:
            order = np.argsort(h)
        else:
            order = np.argsort(-h)
        for idx in order:
            val = h[idx] + delta2
            if -1e-15 <= val <= 1.0 + 1e-15:
                h[idx] = val
                break

    return np.clip(h, 0.0, 1.0)


def project_feasible(u):

    n = u.size
    T = 0.5 * n
    if n == 0:
        return u
    if not np.all(np.isfinite(u)):
        return np.full(n, 0.5)

    v = np.clip(u, -1000.0, 1000.0)
    m = max(1.0, float(np.max(np.abs(v))))
    lo = -m - 10.0
    hi = m + 10.0

    for _ in range(8):
        if float(np.sum(np.clip(v + lo, 0.0, 1.0))) > T:
            lo -= max(1.0, m)
        else:
            break
    for _ in range(8):

            hi += max(1.0, m)
        else:
            break

    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if float(np.sum(np.clip(v + mid, 0.0, 1.0))) > T:
            hi = mid
        else:
            lo = mid

    h = np.clip(v + 0.5 * (lo + hi), 0.0, 1.0)
    return finish_sum(h, T)


def normalize_h(h):
    try:
        h = np.asarray(h, dtype=float)
    except Exception:
        return None
    if h.ndim != 1 or h.size == 0 or not np.all(np.isfinite(h)):
        return None

    T = 0.5 * h.size
    h = np.clip(h, 0.0, 1.0)
    if abs(float(np.sum(h)) - T) > 1e-6:
        h = project_feasible(h)
    else:
        h = finish_sum(h, T)

    if h.min() < -1e-8 or h.max() > 1.0 + 1e-8:
        return None
    if abs(float(np.sum(h)) - T) > 1e-5:
        return None
    return h


def exact_C(h):
    h = np.asarray(h, dtype=float)
    if h.ndim != 1 or h.size == 0 or not np.all(np.isfinite(h)):
        return float("inf")
    if h.min() < -1e-12 or h.max() > 1.0 + 1e-12:
        h = np.clip(h, 0.0, 1.0)
    dx = 2.0 / float(h.size)
    vals = np.correlate(h, 1.0 - h, mode="full")
    c = float(np.max(vals)) * dx
    if not np.isfinite(c):
        return float("inf")
    return c


def record_candidate(h):
    global BEST_H, BEST_C
    h = normalize_h(h)
    if h is None:
        return float("inf")
    c = exact_C(h)
    if np.isfinite(c) and c < BEST_C - 1e-15:
        BEST_C = float(c)
        BEST_H = h.copy()
    return c


def directional_values(h):
    n = h.size
    V = np.empty(2 * n - 1, dtype=float)
    dx = 2.0 / float(n)
    V[0] = dx * float(np.dot(h, 1.0 - h))
    idx = 1
    for m in range(1, n):
        N = n - m
        V[idx] = dx * float(np.dot(h[:N], 1.0 - h[m:]))
        idx += 1
        V[idx] = dx * float(np.dot(h[m:], 1.0 - h[:N]))
        idx += 1
    return V


def gradient_from_weights(h, w):
    n = h.size
    G = np.zeros(n, dtype=float)
    dx = 2.0 / float(n)

    G += (w[0] * dx) * (1.0 - 2.0 * h)

    for m in range(1, n):
        N = n - m
        idx = 1 + 2 * (m - 1)
        wa = w[idx]
        wb = w[idx + 1]

        if wa != 0.0:
            G[:N] += (wa * dx) * (1.0 - h[m:])
            G[m:] -= (wa * dx) * h[:N]
        if wb != 0.0:
            G[:N] -= (wb * dx) * h[m:]
            G[m:] += (wb * dx) * (1.0 - h[:N])

    return G


def smooth_objective_grad_x(h, beta):
    V = directional_values(h)
    if not np.all(np.isfinite(V)):
        return float("inf"), np.zeros(h.size)

    vmax = float(np.max(V))
    shift = beta * (V - vmax)
    e = np.exp(shift)
    se = float(np.sum(e))
    if not np.isfinite(se) or se <= 0.0:
        return float("inf"), np.zeros(h.size)

    w = e / se
    F = vmax + float(np.log(se)) / beta
    G = gradient_from_weights(h, w)
    if not np.all(np.isfinite(G)):
        G = np.nan_to_num(G, nan=0.0, posinf=0.0, neginf=0.0)
    return float(F), G


def active_gradient(h):
    n = h.size
    V = directional_values(h)
    if not np.all(np.isfinite(V)):
        return np.zeros(n)
    vmax = float(np.max(V))
    tol = max(1e-11, 1e-8 * max(1.0, abs(vmax)))
    mask = V >= (vmax - tol)
    k = int(np.count_nonzero(mask))
    if k == 0:
        w = np.zeros(V.size, dtype=float)
        w[0] = 1.0
    else:
        w = np.zeros(V.size, dtype=float)
        w[mask] = 1.0 / float(k)
    return gradient_from_weights(h, w)


def solve_common_lambda(y, T):
    lo = float(np.min(y)) - 50.0
    hi = float(np.max(y)) + 50.0

    for _ in range(8):
        if float(np.sum(stable_sigmoid(y + lo))) > T:
            lo -= 50.0
        else:
            break
    for _ in range(8):
        if float(np.sum(stable_sigmoid(y + hi))) < T:
            hi += 50.0
        else:
            break

    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if float(np.sum(stable_sigmoid(y + mid))) > T:
            hi = mid
        else:
            lo = mid
    return 0.5 * (lo + hi)


def y_to_h(y):
    y = np.asarray(y, dtype=float)
    if y.size == 0 or not np.all(np.isfinite(y)):
        return None
    yc = np.clip(y - float(np.mean(y)), -80.0, 80.0)
    T = 0.5 * y.size
    lam = solve_common_lambda(yc, T)
    h = stable_sigmoid(yc + lam)
    h = np.clip(h, 0.0, 1.0)
    if abs(float(np.sum(h)) - T) > 1e-10:
        h = project_feasible(h)
    else:
        h = finish_sum(h, T)
    return h


def objective_y(y):
    global OBJ_CALL_COUNT

    if time.time() > RUN_DEADLINE - 40.0:
        raise BudgetExceeded()

    y = np.asarray(y, dtype=float)
    if y.size == 0 or not np.all(np.isfinite(y)):
        return 1e6, np.zeros_like(y)

    OBJ_CALL_COUNT += 1
    yc = y - float(np.mean(y))
    yc = np.clip(yc, -80.0, 80.0)
    T = 0.5 * y.size

    lam = solve_common_lambda(yc, T)
    h = stable_sigmoid(yc + lam)
    h = np.clip(h, 0.0, 1.0)
    if abs(float(np.sum(h)) - T) > 1e-9:
        h = project_feasible(h)
    else:
        h = finish_sum(h, T)

    if OBJ_CALL_COUNT % 5 == 0:
        record_candidate(h)

    F, G = smooth_objective_grad_x(h, OBJ_BETA)
    if not np.isfinite(F):
        return 1e6, np.zeros_like(y)

    m = h * (1.0 - h)
    M = float(np.sum(m))
    if M <= 1e-12:
        return float(F), np.zeros_like(y)

    B = float(np.dot(G, m)) / M
    grad = m * (G - B)
    grad -= float(np.mean(grad))
    if not np.all(np.isfinite(grad)):
        grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    return float(F), grad


def linear_sum_eq(h):
    h = np.asarray(h, dtype=float)
    return float(np.sum(h)) - 0.5 * h.size


def linear_sum_jac(h):
    h = np.asarray(h, dtype=float)
    return np.ones_like(h)


def constrained_objective_x(h):
    global SLSQP_CALL_COUNT

    if time.time() > RUN_DEADLINE - 40.0:
        raise BudgetExceeded()

    h = np.asarray(h, dtype=float)
    if h.size == 0 or not np.all(np.isfinite(h)):
        return 1e6
    h = np.clip(h, 0.0, 1.0)

    SLSQP_CALL_COUNT += 1
    if SLSQP_CALL_COUNT % 15 == 0:
        record_candidate(h)

    F, G = smooth_objective_grad_x(h, SLSQP_BETA)
    if not np.isfinite(F):
        return 1e6
    return float(F)


def constrained_grad_x(h):
    h = np.asarray(h, dtype=float)
    if h.size == 0 or not np.all(np.isfinite(h)):
        return np.zeros_like(h)
    h = np.clip(h, 0.0, 1.0)
    F, G = smooth_objective_grad_x(h, SLSQP_BETA)
    if not np.all(np.isfinite(G)):
        G = np.nan_to_num(G, nan=0.0, posinf=0.0, neginf=0.0)
    return G


def pair_cost_global(t):
    t = float(t)
    if t < -1e-12 or t > 1.1:
        return 1e6
    hh = PAIR_H.copy()
    hh[PAIR_A] += t
    hh[PAIR_B] -= t
    if hh[PAIR_A] < -1e-8 or hh[PAIR_A] > 1.0 + 1e-8:
        return 1e6
    if hh[PAIR_B] < -1e-8 or hh[PAIR_B] > 1.0 + 1e-8:
        return 1e6
    return exact_C(hh)


def direction_cost_global(alpha):
    alpha = float(alpha)
    if alpha < -1e-12 or alpha > 1.1:
        return 1e6
    hh = DIR_H + alpha * DIR_P
    if hh.min() < -1e-8 or hh.max() > 1.0 + 1e-8:
        return 1e6
    return exact_C(hh)


def direction_upper(h, p):
    pos = p > 1e-13
    neg = p < -1e-13
    upper = 1e100
    if np.any(pos):
        upper = min(upper, float(np.min((1.0 - h[pos]) / p[pos])))
    if np.any(neg):
        upper = min(upper, float(np.min(-h[neg] / p[neg])))
    if not np.isfinite(upper) or upper < 0.0:
        return 0.0
    return min(upper, 1.0)


def select_pairs(h, G, max_pairs):
    global RUN_RNG
    n = h.size
    if n < 2:
        return []
    if RUN_RNG is None:
        RUN_RNG = np.random.default_rng(12345)

    order = np.argsort(G)
    k = max(3, min(10, n // 5))
    lows = order[:k]
    highs = order[::-1][:k]

    seen = set()
    res = []

    for a in lows:
        ai = int(a)
        for b in highs:
            bi = int(b)
            if ai == bi:
                continue
            if G[ai] > G[bi] + 1e-12:
                continue
            upper = min(1.0 - h[ai], h[bi])
            if upper <= 1e-11:
                continue
            key = (ai, bi)
            if key not in seen:
                seen.add(key)
                res.append(key)

    val_order = np.argsort(h)
    lowv = val_order[:max(3, k)]
    highv = val_order[::-1][:max(3, k)]

    for a in lowv:
        ai = int(a)
        for b in highv:
            bi = int(b)
            if ai == bi:
                continue
            if G[ai] <= G[bi]:
                key = (ai, bi)
            else:
                key = (bi, ai)
            ka, kb = key
            upper = min(1.0 - h[ka], h[kb])
            if upper <= 1e-11:
                continue
            if key not in seen:
                seen.add(key)
                res.append(key)

    for _ in range(max(8, max_pairs // 3)):
        ai = int(RUN_RNG.integers(n))
        bi = int(RUN_RNG.integers(n))
        if ai == bi:
            continue
        if G[ai] > G[bi]:
            ai, bi = bi, ai
        upper = min(1.0 - h[ai], h[bi])
        if upper <= 1e-11:
            continue
        key = (ai, bi)
        if key not in seen:
            seen.add(key)
            res.append(key)

    RUN_RNG.shuffle(res)
    if len(res) > max_pairs:
        res = res[:max_pairs]
    return res


def pair_polish(h, max_pairs=60, time_limit_s=40):
    global PAIR_H, PAIR_A, PAIR_B

    h = normalize_h(h)
    if h is None:
        return None

    n = h.size
    c = exact_C(h)
    start = time.time()
    tested_total = 0

    for _pass in range(4):
        G = active_gradient(h)
        pairs = select_pairs(h, G, max_pairs)
        pass_improved = False

        for a, b in pairs:
            if tested_total >= max_pairs:
                break
            if time.time() - start > time_limit_s:
                break
            if time_left() < FINAL_RESERVE:
                break

            a = int(a)
            b = int(b)
            if a == b:
                continue
            if G[a] > G[b] + 1e-12:
                a, b = b, a

            upper = min(1.0 - h[a], h[b])
            if upper <= 1e-11:
                continue

            tested_total += 1
            PAIR_H = h
            PAIR_A = a
            PAIR_B = b
            best_t = 0.0
            best_c = c

            try:
                res = minimize_scalar(
                    pair_cost_global,
                    bounds=(0.0, float(upper)),
                    method="bounded",
                    options={"xatol": 1e-10, "maxiter": 30},
                )
                if res is not None and np.isfinite(res.fun) and float(res.fun) < best_c - 1e-13:
                    best_t = float(res.x)
                    best_c = float(res.fun)
            except Exception:
                pass

            if best_c >= c - 1e-13:
                for tt in np.linspace(0.0, float(upper), 7):
                    tt = float(tt)
                    hh = h.copy()
                    hh[a] += tt
                    hh[b] -= tt
                    if hh[a] < -1e-8 or hh[a] > 1.0 + 1e-8:
                        continue
                    if hh[b] < -1e-8 or hh[b] > 1.0 + 1e-8:
                        continue
                    cc = exact_C(hh)
                    if cc < best_c - 1e-13:
                        best_c = cc
                        best_t = tt

            if best_c < c - 1e-13 and best_t > 1e-12:
                h[a] += best_t
                h[b] -= best_t
                h = np.clip(h, 0.0, 1.0)
                h = finish_sum(h, 0.5 * n)
                c = exact_C(h)
                record_candidate(h)
                pass_improved = True
                if c <= TARGET_C:
                    return h

        if not pass_improved:
            break

    return h


def active_polish(h, max_iter=3):
    global DIR_H, DIR_P

    h = normalize_h(h)
    if h is None:
        return None

    n = h.size
    c = exact_C(h)

    for _ in range(max_iter):
        if time_left() < FINAL_RESERVE:
            break

        G = active_gradient(h)
        p = float(np.mean(G)) - G
        p = p - float(np.mean(p))
        if not np.all(np.isfinite(p)):
            break
        if float(np.linalg.norm(p)) < 1e-12:
            break

        upper = direction_upper(h, p)
        if upper <= 1e-12:
            break

        DIR_H = h
        DIR_P = p
        best_a = 0.0
        best_c = c

        try:
            res = minimize_scalar(
                direction_cost_global,
                bounds=(0.0, float(upper)),
                method="bounded",
                options={"xatol": 1e-9, "maxiter": 30},
            )
            if res is not None and np.isfinite(res.fun) and float(res.fun) < best_c - 1e-13:
                best_a = float(res.x)
                best_c = float(res.fun)
        except Exception:
            pass

        if best_c >= c - 1e-13:
            for aa in np.linspace(0.0, float(upper), 7):
                aa = float(aa)
                hh = h + aa * p
                if hh.min() < -1e-8 or hh.max() > 1.0 + 1e-8:
                    continue
                cc = exact_C(hh)
                if cc < best_c - 1e-13:
                    best_c = cc
                    best_a = aa

        if best_c < c - 1e-13 and best_a > 1e-12:
            h = h + best_a * p
            h = np.clip(h, 0.0, 1.0)
            h = finish_sum(h, 0.5 * n)
            c = exact_C(h)
            record_candidate(h)
            if c <= TARGET_C:
                break
        else:
            break

    return h


def optimize_from_start(h_start, beta_schedule, max_iter):
    global OBJ_BETA

    h = normalize_h(h_start)
    if h is None:
        return None

    record_candidate(h)
    local_c = exact_C(h)
    best_entry = BEST_C

    for beta in beta_schedule:
        if time_left() < 110:
            break

        y = stable_logit(h)
        y = np.clip(y - float(np.mean(y)), -60.0, 60.0)
        OBJ_BETA = float(beta)

        try:
            res = minimize(
                objective_y,
                y,
                method="BFGS",
                jac=True,
                options={"maxiter": int(max_iter), "gtol": 1e-6, "disp": False},
            )
        except BudgetExceeded:
            break
        except Exception:
            res = None

        if res is not None and hasattr(res, "x") and np.all(np.isfinite(res.x)):
            y2 = np.asarray(res.x, dtype=float)
            y2 = np.clip(y2 - float(np.mean(y2)), -80.0, 80.0)
            h2 = y_to_h(y2)
            if h2 is not None:
                c2 = record_candidate(h2)
                if c2 < local_c - 1e-13:
                    h = h2
                    local_c = c2

        if BEST_C < best_entry - 1e-12 and BEST_H is not None:
            h = BEST_H.copy()
            local_c = float(BEST_C)
            best_entry = local_c

        if time_left() > 100:
            hp = pair_polish(h, max_pairs=30, time_limit_s=25)
            if hp is not None:
                cp = exact_C(hp)
                if cp < local_c - 1e-13:
                    h = hp
                    local_c = cp
                    record_candidate(h)

        if BEST_C < best_entry - 1e-12 and BEST_H is not None:
            h = BEST_H.copy()
            local_c = float(BEST_C)
            best_entry = local_c

        if local_c <= TARGET_C:
            break

    return h


def slsqp_from_h(h_start, beta_schedule, max_iter):
    global SLSQP_BETA

    h = normalize_h(h_start)
    if h is None:
        return None

    record_candidate(h)
    local_c = exact_C(h)
    best_entry = BEST_C

    for beta in beta_schedule:
        if time_left() < 100:
            break

        SLSQP_BETA = float(beta)
        n = h.size
        x0 = h.copy()
        bounds = [(0.0, 1.0) for _ in range(n)]
        cons = [{"type": "eq", "fun": linear_sum_eq, "jac": linear_sum_jac}]

        try:
            res = minimize(
                constrained_objective_x,
                x0,
                method="SLSQP",
                jac=constrained_grad_x,
                bounds=bounds,
                constraints=cons,
                options={"maxiter": int(max_iter), "ftol": 1e-9, "disp": False},
            )
        except BudgetExceeded:
            break
        except Exception:
            res = None

        if res is not None and hasattr(res, "x") and np.all(np.isfinite(res.x)):
            h2 = normalize_h(res.x)
            if h2 is not None:
                c2 = record_candidate(h2)
                if c2 < local_c - 1e-13:
                    h = h2
                    local_c = c2

        if BEST_C < best_entry - 1e-12 and BEST_H is not None:
            h = BEST_H.copy()
            local_c = float(BEST_C)
            best_entry = local_c

        if time_left() > 90:
            hp = pair_polish(h, max_pairs=24, time_limit_s=20)
            if hp is not None:
                cp = exact_C(hp)
                if cp < local_c - 1e-13:
                    h = hp
                    local_c = cp
                    record_candidate(h)

        if local_c <= TARGET_C:
            break

    return h


def build_starts(base_h):
    global RUN_RNG
    if RUN_RNG is None:
        RUN_RNG = np.random.default_rng(7)

    starts = []
    n = base_h.size
    t = (np.arange(n) + 0.5) / float(n)

    starts.append(base_h.copy())
    starts.append(base_h[::-1].copy())

    for s in (1, 2, 3, n // 2):
        starts.append(np.roll(base_h, s).copy())

    wave_specs = [
        (1, 0.05, 0.0),
        (1, 0.10, 0.3),
        (2, 0.05, 0.0),
        (2, 0.10, 0.7),
        (3, 0.08, 0.0),
        (1, 0.07, 1.1),
        (2, 0.07, 2.0),
    ]

    for freq, amp, phase in wave_specs:
        w = np.sin(2.0 * np.pi * freq * t + phase)
        starts.append(project_feasible(base_h + amp * w))
        w2 = np.cos(2.0 * np.pi * freq * t + phase)
        starts.append(project_feasible(base_h + amp * w2))

    for sigma in (0.02, 0.045, 0.08):
        noise = RUN_RNG.normal(0.0, sigma, n)
        starts.append(project_feasible(base_h + noise))

    clean = []
    for h in starts:
        hh = normalize_h(h)
        if hh is not None and np.all(np.isfinite(hh)):
            clean.append(hh)
    return clean


def resize_h(h_old, n_new):
    n_old = h_old.size
    if n_new <= 0:
        return None
    if n_new == n_old:
        return normalize_h(h_old)

    old_x = (np.arange(n_old) + 0.5) / float(n_old)
    new_x = (np.arange(n_new) + 0.5) / float(n_new)
    h_new = np.interp(new_x, old_x, h_old, left=h_old[0], right=h_old[-1])
    return normalize_h(h_new)


def valid_dimensions(n):
    dims = []
    for cand in (n, n - 6, n - 12, n - 18, n + 6):
        if 16 <= cand <= 64 and cand not in dims:
            dims.append(cand)
    return dims


def exact_C_key(h):
    return exact_C(h)


def run(seed=42, budget_s=1100, **kwargs):
    global RUN_DEADLINE, RUN_RNG, BEST_H, BEST_C
    global OBJ_BETA, OBJ_CALL_COUNT, SLSQP_BETA, SLSQP_CALL_COUNT

    t0 = time.time()
    RUN_DEADLINE = t0 + max(45.0, float(budget_s) - 30.0)
    RUN_RNG = np.random.default_rng(seed)

    BEST_H = None
    BEST_C = float("inf")
    OBJ_BETA = 100.0
    OBJ_CALL_COUNT = 0
    SLSQP_BETA = 100.0
    SLSQP_CALL_COUNT = 0

    parent = None
    g = globals()
    init = g.get("initial_h_values", None)

    n_default = 42
    if init is not None:
        try:
            arr = np.asarray(init, dtype=float)
            if arr.ndim != 1:
                arr = arr.ravel()
            if arr.size > 0 and np.all(np.isfinite(arr)):
                parent = normalize_h(arr)
        except Exception:
            parent = None

    if parent is None:
        t = (np.arange(n_default) + 0.5) / float(n_default)
        arr = 0.5 + 0.22 * np.sin(2.0 * np.pi * 2.0 * t) + 0.10 * np.cos(2.0 * np.pi * 5.0 * t)
        parent = normalize_h(arr)

    if parent is None:
        parent = np.full(n_default, 0.5)
        parent = normalize_h(parent)

    if parent.size > 64:
        parent = resize_h(parent, 56)

    record_candidate(parent)

    if BEST_H is None:
        fallback = np.full(parent.size, 0.5)
        record_candidate(fallback)
        parent = normalize_h(BEST_H) if BEST_H is not None else normalize_h(fallback)

    base = BEST_H.copy()
    starts = build_starts(base)
    starts.sort(key=exact_C_key)

    primary_betas = [70.0, 180.0, 480.0]
    max_iter_primary = 140 if base.size <= 48 else 100
    max_start_opt = min(len(starts), 5)

    for i in range(max_start_opt):
        if time_left() < 190:
            break
        optimize_from_start(starts[i], primary_betas, max_iter_primary)
        if BEST_C <= TARGET_C:
            break

    if time_left() > 230:
        besth = normalize_h(BEST_H)
        if besth is not None:
            slsqp_from_h(besth, [180.0, 450.0], 55)
            if BEST_C <= TARGET_C:
                pass

    if time_left() > 190:
        besth = normalize_h(BEST_H)
        if besth is not None:
            for nd in valid_dimensions(besth.size):
                if time_left() < 130:
                    break
                hdim = resize_h(besth, nd)
                if hdim is None:
                    continue
                record_candidate(hdim)
                cd = exact_C(hdim)
                if cd < max(BEST_C + 0.035, 0.45):
                    optimize_from_start(hdim, [150.0, 420.0], 80)
                if BEST_C <= TARGET_C:
                    break

    if time_left() > 170:
        besth = normalize_h(BEST_H)
        if besth is not None:
            for sigma in (0.07, 0.13):
                if time_left() < 120:
                    break
                hrand = normalize_h(besth + RUN_RNG.normal(0.0, sigma, besth.size))
                if hrand is not None:
                    optimize_from_start(hrand, [220.0, 620.0], 80)
                if BEST_C <= TARGET_C:
                    break

    stall = 0
    final_rounds = 0
    while time_left() > FINAL_RESERVE + 30 and final_rounds < 12:
        final_rounds += 1
        hwork = normalize_h(BEST_H)
        if hwork is None:
            break

        cwork = exact_C(hwork)
        hwork = active_polish(hwork, 3)
        hwork = pair_polish(hwork, 60, 32)

        if hwork is not None:
            cfin = exact_C(hwork)
            if cfin < cwork - 1e-12:
                record_candidate(hwork)
                stall = 0
            else:
                stall += 1
                if stall >= 2:
                    hkick = normalize_h(hwork + RUN_RNG.normal(0.0, 0.05, hwork.size))
                    if hkick is not None:
                        optimize_from_start(hkick, [320.0, 800.0], 70)
                if stall >= 5:
                    break

        if BEST_C <= TARGET_C:
            break

    h_final = normalize_h(BEST_H)
    if h_final is None:
        h_final = normalize_h(parent)

    if h_final is not None:
        record_candidate(h_final)
        h_final = normalize_h(BEST_H)

    if h_final is None:
        h_final = normalize_h(parent)

    if h_final is None:
        h_final = np.full(42, 0.5)
        h_final = normalize_h(h_final)
        record_candidate(h_final)
        h_final = normalize_h(BEST_H)

    c_final = exact_C(h_final)
    if not np.isfinite(c_final):
        c_final = 0.5

    return h_final, float(c_final), int(h_final.size)
