import numpy as np
from scipy.optimize import minimize
import time

def project_to_simplex_box(h, target_sum):
    """Project h onto {x : 0 <= x <= 1, sum(x) = target_sum}"""
    n = len(h)
    # Bisection on lambda
    lo = h.min() - 1
    hi = h.max() + 1
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h - mid, 0, 1))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo + hi) / 2, 0, 1)
    return proj

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx, np.argmax(corr)

def grad_of_corr(h, k_full):
    """Gradient of the k_full-th element of np.correlate(h, 1-h, 'full') w.r.t. h"""
    n = len(h)
    lag = k_full - (n - 1)
    grad = np.zeros(n)
    # (1-h[m-lag]) for m-lag in [0, n)
    if lag >= 0:
        # m - lag in [0, n) means m in [lag, n+lag), intersect [0, n) = [lag, n)
        # m + lag in [0, n) means m in [-lag, n-lag), intersect [0, n) = [0, n-lag)
        grad[lag:] += (1 - h[:n-lag])
        grad[:n-lag] -= h[lag:]
    else:
        d = -lag
        # m + d in [0, n) means m in [-d, n-d), intersect [0, n) = [0, n-d)
        # m - d in [0, n) means m in [d, n+d), intersect [0, n) = [d, n)
        grad[:n-d] += (1 - h[d:])
        grad[d:] -= h[:n-d]
    return grad

def projected_gradient_step(h, grad, lr, target_sum):
    h_new = h - lr * grad
    h_new = project_to_simplex_box(h_new, target_sum)
    return h_new

import numpy as np
from scipy.optimize import minimize
import time

def project_polytope(h, target_sum):
    """Project h onto {x: 0<=x<=1, sum(x)=target_sum} via bisection."""
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def c5_of(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full"))) * dx

def grad_corr(h, k_full):
    """Gradient of the k_full-th entry of correlate(h, 1-h, 'full')."""
    n = len(h)
    lag = k_full - (n - 1)
    g = np.zeros(n)
    if lag >= 0:
        g[lag:] += 1.0 - h[:n - lag]
        g[:n - lag] -= h[lag:]
    else:
        d = -lag
        g[:n - d] += 1.0 - h[d:]
        g[d:] -= h[:n - d]
    return g

def pgd_loop(h, deadline, max_iters=10000):
    """Projected gradient descent with smooth-max objective."""
    n = len(h)
    target_sum = n / 2.0
    dx = 2.0 / n
    h = h.astype(float).copy()
    
    lr = 2.0 / (n * (n + 1))
    temp = 100.0
    cool = 0.9998
    best = h.copy()
    best_c5 = c5_of(h)

        if time.time() > deadline:
            break
        
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = float(corr.max()) * dx
        
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best = h.copy()
        
        # Top-k weighted gradient
        top_k = min(15, len(corr))
        idx = np.argsort(corr)[-top_k:]
        T = temp
        wc = np.exp(T * (corr[idx] - corr[idx].max()))
        wc /= wc.sum()
        
        g = np.zeros(n)
        for j in range(top_k):
            g += wc[j] * grad_corr(h, int(idx[j]))
        g *= dx

        c5_new = c5_of(h_new)
        
        if c5_new <= c5 + 1e-14:

            if c5_new < c5 * 0.999:
                lr = min(lr * 1.03, 10.0 / n)
        else:
            lr *= 0.5

import numpy as np
import time

def project_polytope(h, target_sum):
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def c5_of(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full"))) * dx

def grad_corr(h, k_full):
    n = len(h)
    lag = k_full - (n - 1)
    g = np.zeros(n)
    if lag >= 0:
        g[lag:] += 1.0 - h[:n - lag]
        g[:n - lag] -= h[lag:]
    else:
        d = -lag
        g[:n - d] += 1.0 - h[d:]
        g[d:] -= h[:n - d]
    return g
