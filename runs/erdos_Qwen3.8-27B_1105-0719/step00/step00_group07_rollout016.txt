def _sigmoid(x):
    x = np.clip(x, -50, 50)
    out = np.empty_like(x, dtype=float)
    pos = x >= 0
    out[pos] = 1.0 / (1.0 + np.exp(-x[pos]))
    ex = np.exp(x[~pos])
    out[~pos] = ex / (1.0 + ex)
    return out

        if np.sum(_sigmoid(theta + mid)) < target:
            lo = mid
        else:
            hi = mid

    h = _sigmoid(theta + c)
    h = np.clip(h, 0.0, 1.0)

        w[active] = 1.0 / active.sum()

    else:

import numpy as np
from scipy.optimize import minimize
import time

def _sig(x):
    """Vectorized stable sigmoid."""
    x = np.clip(x, -50.0, 50.0)
    out = np.empty_like(x)
    mask = x >= 0
    out[mask] = 1.0 / (1.0 + np.exp(-x[mask]))
    ex = np.exp(x[~mask])
    out[~mask] = ex / (1.0 + ex)
    return out

def _find_c(theta, target):
    """Bisection for c: sum(sigmoid(theta+c)) = target."""
    lo, hi = -50.0, 50.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        if _sig(theta + mid).sum() < target:
            lo = mid
        else:
            hi = mid
    return 0.5 * (lo + hi)

def _mk_h(theta):
    """theta -> feasible h."""
    n = len(theta)
    c = _find_c(theta, 0.5 * n)
    h = _sig(theta + c)
    h = np.clip(h, 0.0, 1.0)
    # Fix sum exactly
    d = 0.5 * n - h.sum()
    if abs(d) > 1e-12:
        sl = np.minimum(h, 1.0 - h)
        ord_ = np.argsort(-sl)
        for i in ord_:
            if abs(d) < 1e-14:
                break
            s = np.sign(d) * min(abs(d), sl[i])
            h[i] += s
            d -= s
    return h

def _to_theta(h):
    """h -> theta."""
    e = 1e-10
    h = np.clip(h, e, 1.0 - e)
    t = np.log(h / (1.0 - h))
    t -= t.mean()
    return t

def _c5(h):
    """Compute C5 bound."""
    n = len(h)
    return (2.0 / n) * np.correlate(h, 1.0 - h, mode='full').max()

def _lse_fg(theta, beta):
    """LSE value and gradient."""
    n = len(theta)
    c = _find_c(theta, 0.5 * n)
    h = _sig(theta + c)
    h = np.clip(h, 0.0, 1.0)
    raw = np.correlate(h, 1.0 - h, mode='full')
    nr = len(raw)
    
    mx = raw.max()
    if beta > 800:
        act = raw >= mx - 1e-12
        w = np.zeros(nr)
        w[act] = 1.0 / act.sum()
        val = mx
    else:
        e = np.exp(beta * (raw - mx))
        w = e / e.sum()
        val = mx + np.log(e.sum()) / beta
    
    gh = np.zeros(n)
    for k in range(nr):
        wk = w[k]
        if wk < 1e-13:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            m = n - lag
            gh[:m] += wk * (1.0 - h[lag:])
            gh[lag:] += wk * (-h[:m])
        else:
            q = -lag
            m = n - q
            gh[q:] += wk * (1.0 - h[:m])
            gh[:m] += wk * (-h[q:])
    
    sh = h * (1.0 - h)
    S = sh.sum()
    if S > 1e-14:
        gt = sh * (gh - np.dot(gh, sh) / S)
    else:
        gt = np.zeros(n)
    return val, gt
