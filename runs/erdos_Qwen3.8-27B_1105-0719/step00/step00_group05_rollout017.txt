import numpy as np
import time

def project_feasible(z, n):
    target = n / 2.0
    # Bisection on lambda for x_i = clip(z_i + lambda, 0, 1)
    lo = -z.max() - 1.0
    hi = -z.min() + 1.0
    
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(z + mid, 0, 1).sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    
    x = np.clip(z + (lo + hi) / 2.0, 0, 1)
    
    # Fix sum to be exactly target
    diff = target - x.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            order = np.argsort(-x)  # closest to 1 first
        else:
            order = np.argsort(x)  # closest to 0 first
        for idx in order:
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                add = min(diff, 1.0 - x[idx])
                if add > 0:
                    x[idx] += add
                    diff -= add
            else:
                sub = min(-diff, x[idx])
                if sub > 0:
                    x[idx] -= sub
                    diff += sub
    return x

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    corr_max = np.max(corr)
    shifted = beta * (corr - corr_max)
    exp_vals = np.exp(shifted)
    total = np.sum(exp_vals)
    p = exp_vals / total  # length 2n-1
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = p[k] * dx
        if w < 1e-16:
            continue
        lag = k - (n - 1)
        if lag == 0:
            grad += w * (1.0 - 2.0 * h)
        elif lag > 0:
            m = lag
            grad[m:] += w * (1.0 - h[:n-m])
            grad[:n-m] -= w * h[m:]
        else:
            m = -lag
            grad[:n-m] += w * (1.0 - h[m:])
            grad[m:] -= w * h[:n-m]
    return grad

def pair_line_search(h, p, q, max_evals=200):
    """1D line search along pair transfer direction"""
    n = len(h)
    eps_lo = max(-h[p], h[q] - 1.0)
    eps_hi = min(1.0 - h[p], h[q])
    if eps_hi - eps_lo < 1e-12:
        return h, evaluate_c5(h)
    
    best_c = evaluate_c5(h)
    best_h = h.copy()
    
    # Grid search
    grid = np.linspace(eps_lo, eps_hi, max_evals)
    h_work = h.copy()
    for eps in grid:
        h_work[p] = h[p] + eps
        h_work[q] = h[q] - eps
        c = evaluate_c5(h_work)
        if c < best_c:
            best_c = c
            best_h = h_work.copy()
    
    return best_h, best_c

    n = len(h)
    h_curr = h.copy()
    c_curr = evaluate_c5(h_curr)
    h_best = h_curr.copy()
    c_best = c_curr

    fail_count = 0

        grad = smooth_grad(h_curr, beta)
        # Project gradient: remove component along all-ones (sum constraint)
        grad_mean = grad.mean()
        grad_pc = grad - grad_mean
        
        gnorm = np.linalg.norm(grad_pc)
        if gnorm < 1e-10:

        for factor in [1.0, 0.5, 0.25, 0.125, 0.0625]:

            c_try = evaluate_c5(h_try)
            if c_try < c_curr - 1e-14:

                c_curr = c_try
                if c_try < c_best:
                    c_best = c_try
                    h_best = h_try.copy()
                accepted = True
                break
        
        if accepted:

            step_size = min(step_size * 1.1, 0.1)
            if c_best <= current_best_c - 1e-6:
                # Good progress, continue
                pass
        else:
            fail_count += 1
            step_size *= 0.7
            if step_size < 1e-6:

    return h_best, c_best

def smooth_gradient(h, beta):
    """Gradient of softmax-max surrogate of the 2n-1 overlap values"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cm = np.max(corr)
    sh = beta * (corr - cm)
    ev = np.exp(sh)
    tv = ev.sum()
    p = ev / tv
    g = np.zeros(n)
    for k in range(2 * n - 1):
        w = p[k] * dx
        if w < 1e-15: continue
        lag = k - (n - 1)
        if lag == 0:
            g += w * (1.0 - 2.0 * h)
        elif lag > 0:
            g[lag:] += w * (1.0 - h[:n - lag])
            g[:n - lag] -= w * h[lag:]
        else:
            m = -lag
            g[:n - m] += w * (1.0 - h[m:])
            g[m:] -= w * h[:n - m]
    return g

def pair_search(h, p, q, n_grid=50):
    """Line search on epsilon transfer from q to p"""
    e_lo = max(-h[p], h[q] - 1.0)
    e_hi = min(1.0 - h[p], h[q])
    if e_hi - e_lo < 1e-13:
        return h, eval_c5(h)
    best_c = eval_c5(h)
    best_h = h.copy()
    hp, hq = h[p], h[q]
    grid = np.linspace(e_lo, e_hi, n_grid)
    for eps in grid:
        h[p] = hp + eps
        h[q] = hq - eps
        c = eval_c5(h)
        if c < best_c - 1e-15:
            best_c = c
            best_h = h.copy()
    h[p] = best_h[p]
    h[q] = best_h[q]
    return best_h, best_c

def resample(h_old, n_new):
    """Resample h from length n_old to n_new via interpolation + projection"""
    n_old_raw = len(h_old)
    if n_old_raw == n_new:
        return project_feasible(h_old.copy(), n_new)
    x_old = (np.arange(n_old_raw) + 0.5) / n_old_raw
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_interp = np.interp(x_new, x_old, h_old)
    return project_feasible(h_interp, n_new)

    cm = np.max(corr)
    sh = beta * (corr - cm)
    ev = np.exp(sh)
    tv = ev.sum()
    p = ev / tv
    g = np.zeros(n)
    for k in range(2 * n - 1):
        w = p[k] * dx
        if w < 1e-15: continue
        lag = k - (n - 1)
        if lag == 0:
            g += w * (1.0 - 2.0 * h)
        elif lag > 0:
            g[lag:] += w * (1.0 - h[:n - lag])
            g[:n - lag] -= w * h[lag:]
        else:
            m = -lag
            g[:n - m] += w * (1.0 - h[m:])
            g[m:] -= w * h[:n - m]
    return g

def smooth_descent(h, beta, max_iter, deadline):
    n = len(h)
    h_c = h.copy()
    c_c = eval_c5(h_c)
    h_b = h_c.copy()
    c_b = c_c
    step = 0.03
    fails = 0
    for _ in range(max_iter):
        if time.time() >= deadline:
            break
        g = smooth_gradient(h_c, beta)
        g -= g.mean()
        gn = np.linalg.norm(g)
        if gn < 1e-12:
            break
        accepted = False
        for f in (1.0, 0.5, 0.25, 0.125, 0.0625):
            h_t = project_feasible(h_c - step * f * g, n)
            c_t = eval_c5(h_t)
            if c_t < c_c - 1e-14:
                h_c = h_t; c_c = c_t
                if c_t < c_b:
                    c_b = c_t; h_b = h_t.copy()
                accepted = True
                break
        if accepted:
            fails = 0
            step = min(step * 1.05, 0.2)
        else:
            fails += 1
            step *= 0.6
            if step < 1e-7 or fails > 40:
                break
    return h_b, c_b

def pair_search(h, p, q, n_grid=40):
    e_lo = max(-h[p], h[q] - 1.0)
    e_hi = min(1.0 - h[p], h[q])
    if e_hi - e_lo < 1e-13:
        return h.copy(), eval_c5(h)
    best_c = eval_c5(h)
    best_h = h.copy()
    hp, hq = h[p], h[q]
    grid = np.linspace(e_lo, e_hi, n_grid)
    h_work = h.copy()
    for eps in grid:
        h_work[p] = hp + eps
        h_work[q] = hq - eps
        c = eval_c5(h_work)
        if c < best_c - 1e-15:
            best_c = c
            best_h = h_work.copy()
    return best_h, best_c
