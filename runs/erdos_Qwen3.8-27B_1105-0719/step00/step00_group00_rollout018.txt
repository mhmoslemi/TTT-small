    h2 = h.copy()

        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)

def _pair_opt(h, i, j):
    """Optimize pair (i,j) keeping h[i]+h[j] fixed. Returns (h, c5)."""
    n = len(h)
    dx = 2.0 / n
    s = h[i] + h[j]
    t_lo = max(0.0, s - 1.0)
    t_hi = min(1.0, s)
    if t_hi - t_lo < 1e-12:
        return h, _c5(h)
    
    ha = h.copy()
    hb = h.copy()
    ha[i] = t_lo
    ha[j] = s - t_lo
    hb[i] = t_hi
    hb[j] = s - t_hi
    
    Ra = _corr(ha)
    Rb = _corr(hb)
    
    slope = (Rb - Ra) / (t_hi - t_lo)
    interp = Ra - slope * t_lo
    
    # Find all pairwise crossings
    m = len(Ra)
    candidates = [t_lo, t_hi]
    for a in range(m):
        for b in range(a + 1, m):
            ds = slope[a] - slope[b]
            if abs(ds) > 1e-12:
                tc = (interp[b] - interp[a]) / ds
                if t_lo < tc < t_hi:
                    candidates.append(tc)
    
    best_t = t_lo
    best_val = np.inf
    for tc in candidates:
        val = np.max(slope * tc + interp) * dx
        if val < best_val:
            best_val = val
            best_t = tc
    
    h[i] = best_t
    h[j] = s - best_t
    return h, best_val

def _local_search(h, max_time=60):
    """Exact pair-local search."""
    n = len(h)
    t0 = time.time()
    c = _c5(h)
    for sweep in range(500):
        if time.time() - t0 > max_time:
            break
        improved = False
        for i in range(n):
            for j in range(i + 1, n):
                h, c_new = _pair_opt(h, i, j)
                if c_new < c - 1e-12:
                    c = c_new
                    improved = True
                # Always keep the improved h (even if not strictly better)
                # Actually, _pair_opt always returns the optimal for that pair
                # given the current h, so we should always keep it if it's better
        if not improved:
            break
    return h, c

Actually, I want to add one more optimization technique: **exact coordinate descent**. For each coordinate i (or pair i,j), compute the exact optimal value by solving a small 1D (or 2D) problem.

For a single coordinate i (with all others fixed), the sum constraint forces h[i] = n/2 - sum_{j≠i} h[j]. So there's no freedom for a single coordinate.

For a pair (i,j), we showed that the problem reduces to 1D, which we handle with `_pair_opt`.

subject to t1 + t2 = s (fixed sum of the three minus the third)
0 <= t1 <= 1, 0 <= t2 <= 1, 0 <= s - t1 - t2 <= 1

This is a 2D linear programming problem (minimize t subject to a_l*t1 + b_l*t2 + c_l <= t for all l, plus the constraints). We can solve this with a small LP.

def _c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full"))) * dx

def _corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def _project(x, n):
    target = n * 0.5

            lo = mid
        else:
