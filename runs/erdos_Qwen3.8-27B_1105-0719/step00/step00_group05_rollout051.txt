    exp_t = np.exp(t[~pos])

    return out

def pair_polish(h, max_pairs=40, max_iter=15, time_limit=None):
    """Local pair-transfer polish."""
    n = len(h)
    h = h.copy()
    C_cur = exact_C(h)
    start_time = time.time()

    beta = 200.0
    # Use the directional values to find most active lags

    v_max = np.max(v)
    # Find lags near maximum
    active_lags = np.argsort(v)[::-1]
    
    # Generate candidate pairs
    pairs = []
    # Pairs based on active lag structure
    for q in np.argsort(v)[::-1][:10]:
        L = q - (n - 1)
        if L == 0:
            # All pairs equally active; pick random
            continue
        M = abs(L)
        for i in range(0, n - M, max(1, (n - M) // 5)):
            if L > 0:
                pairs.append((i, i + L))
            else:
                pairs.append((i + M, i))
    
    # Also add random pairs
    rng = np.random.default_rng(int(time.time()*1000) % 10000)
    for _ in range(max_pairs // 2):
        a = rng.integers(n)
        b = rng.integers(n)
        if a != b:
            pairs.append((a, b))
    
    # Deduplicate
    pairs = list(set(pairs))[:max_pairs * 2]
    
    # Try each pair
    improved = True
    iter_count = 0
    while improved and iter_count < max_iter:
        improved = False
        iter_count += 1
        if time_limit is not None and time.time() - start_time > time_limit:
            break
        for (a, b) in pairs:
            if time_limit is not None and time.time() - start_time > time_limit:
                break
            # Move t from b to a: h[a] += t, h[b] -= t
            # Constraint: 0 <= h[a]+t <= 1, 0 <= h[b]-t <= 1
            lo = max(-h[a], h[b] - 1.0)
            hi = min(1.0 - h[a], h[b])
            if lo >= hi - 1e-12:
                continue
            # Grid search
            n_grid = 15
            t_vals = np.linspace(lo, hi, n_grid)
            best_t = 0.0
            best_C = C_cur
            for t in t_vals:
                h_trial = h.copy()
                h_trial[a] += t
                h_trial[b] -= t
                C_trial = exact_C(h_trial)
                if C_trial < best_C:
                    best_C = C_trial
                    best_t = t
            # Refine around best
            if best_t != 0.0:
                t_range = (hi - lo) / n_grid
                lo_ref = max(lo, best_t - t_range)
                hi_ref = min(hi, best_t + t_range)
                if hi_ref > lo_ref:

                        res = minimize_scalar(lambda t: _pair_obj(h, a, b, t),
                                             bounds=(lo_ref, hi_ref),
                                             method='bounded',

                        if res.success and res.fun < best_C:
                            best_C = res.fun
                            best_t = res.x
                    except Exception:
                        pass
            if best_C < C_cur - 1e-14:
                h[a] += best_t
                h[b] -= best_t
                # Clip for safety
                h = np.clip(h, 0.0, 1.0)
                h = project_feasible(h)
                C_cur = exact_C(h)
                improved = True
                # Update global best
                if _BEST_C is None or C_cur < _BEST_C:
                    global _BEST_H, _BEST_C
                    _BEST_H = h.copy()
                    _BEST_C = C_cur
    
    return h, C_cur

def _pair_obj(h, a, b, t):
    """Objective for pair move: compute C5 after moving t from b to a."""
    h_trial = h.copy()
    h_trial[a] += t
    h_trial[b] -= t
    return exact_C(h_trial)

def exact_C(h):

        return float('inf')

    return float(np.max(cor) * (2.0 / n))


def project_feasible(u):
    u = np.asarray(u, dtype=float).copy()
    n = len(u)
    target = n / 2.0
    lo = -5.0
    hi = 5.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(u + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h = np.clip(u + lam, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            slack = 1.0 - h
            idx = np.argsort(-slack)
            for i in idx:
                if slack[i] <= 1e-15:
                    break
                add = min(residual, slack[i])
                h[i] += add
                residual -= add
                if abs(residual) < 1e-13:
                    break
        else:

            for i in idx:
                if h[i] <= 1e-15:
                    break
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub

    return h


def sigmoid(t):
    t = np.asarray(t, dtype=float)
    out = np.empty_like(t)
    pos = t >= 0
    out[pos] = 1.0 / (1.0 + np.exp(-t[pos]))
    exp_t = np.exp(t[~pos])
    out[~pos] = exp_t / (1.0 + exp_t)
    return out

def compute_grad_F(h, beta):

        mask = v >= v_max - 1e-10

    global _BEST_H, _BEST_C, _BEST_N
    if exact_val < _BEST_C:
        _BEST_C = exact_val
        _BEST_H = h.copy()
        _BEST_N = n
    return F, grad_y


def pair_eval(h, a, b, t):
    h_trial = h.copy()
    h_trial[a] += t
    h_trial[b] -= t
    return exact_C(h_trial)


def pair_polish(h, max_pairs=40, max_iter=20, time_limit=60.0, rng=None):
    n = len(h)
    h = h.copy()
    C_cur = exact_C(h)
    start_time = time.time()
    if rng is None:
        rng = np.random.default_rng(int(time.time() * 1000) % 100000)

    cor = np.correlate(h, 1.0 - h, mode='full')
    v = cor * (2.0 / n)
    sorted_lags = np.argsort(v)[::-1]

    pairs = []
    for q in sorted_lags[:8]:
        L = q - (n - 1)
        if L == 0:
            continue
        M = abs(L)
        step = max(1, (n - M) // 8)
        for i in range(0, n - M, step):
            if L > 0:
                pairs.append((i, i + L))
            else:
                pairs.append((i + M, i))
    for _ in range(max_pairs // 3):
        a = int(rng.integers(n))
        b = int(rng.integers(n))
        if a != b:
            pairs.append((a, b))
    pairs = list(set(pairs))[:max_pairs * 2]

    improved = True
    iter_count = 0
    while improved and iter_count < max_iter:
        improved = False
        iter_count += 1
        if time.time() - start_time > time_limit:
            break
        for (a, b) in pairs:
            if time.time() - start_time > time_limit:
                break
            lo_t = max(-h[a], h[b] - 1.0)
            hi_t = min(1.0 - h[a], h[b])
            if lo_t >= hi_t - 1e-12:
                continue
            n_grid = 21
            t_vals = np.linspace(lo_t, hi_t, n_grid)
            best_t = 0.0
            best_C = C_cur
            for t in t_vals:
                c_t = pair_eval(h, a, b, t)
                if c_t < best_C:
                    best_C = c_t
                    best_t = t
            if best_t != 0.0:
                t_step = (hi_t - lo_t) / n_grid
                lo_ref = max(lo_t, best_t - t_step)
                hi_ref = min(hi_t, best_t + t_step)
                if hi_ref > lo_ref + 1e-14:
                    n_fine = 15
                    t_fine = np.linspace(lo_ref, hi_ref, n_fine)
                    for t in t_fine:
                        c_t = pair_eval(h, a, b, t)
                        if c_t < best_C:
                            best_C = c_t
                            best_t = t
            if best_C < C_cur - 1e-14:
                h[a] += best_t
                h[b] -= best_t
                h = np.clip(h, 0.0, 1.0)
                h = project_feasible(h)
                C_cur = exact_C(h)
                improved = True
                global _BEST_H, _BEST_C
                if C_cur < _BEST_C:
                    _BEST_C = C_cur
                    _BEST_H = h.copy()
    return h, C_cur


def coord_descent(h, time_limit=30.0, rng=None):
    n = len(h)
    h = h.copy()
    C_cur = exact_C(h)
    start_time = time.time()
    if rng is None:
        rng = np.random.default_rng(int(time.time() * 1000) % 100000)
    for _ in range(10):
        if time.time() - start_time > time_limit:
            break
        order = list(rng.permutation(n))
        for i in order:
            if time.time() - start_time > time_limit:
                break
            j = int(rng.integers(n))
            if j == i:
                j = (j + 1) % n
            lo_t = max(-h[i], h[j] - 1.0)
            hi_t = min(1.0 - h[i], h[j])
            if lo_t >= hi_t - 1e-12:
                continue
            n_grid = 15
            t_vals = np.linspace(lo_t, hi_t, n_grid)
            best_t = 0.0
            best_C = C_cur
            for t in t_vals:
                c_t = pair_eval(h, i, j, t)
                if c_t < best_C:
                    best_C = c_t
                    best_t = t
            if best_C < C_cur - 1e-14:
                h[i] += best_t
                h[j] -= best_t
                h = np.clip(h, 0.0, 1.0)
                h = project_feasible(h)
                C_cur = exact_C(h)
                global _BEST_H, _BEST_C
                if C_cur < _BEST_C:
                    _BEST_C = C_cur
                    _BEST_H = h.copy()
    return h, C_cur


def run(seed=42, budget_s=1100, **kwargs):
    global _BEST_H, _BEST_C, _BEST_N, _DEADLINE
    _BEST_H = None
    _BEST_C = float('inf')
    _BEST_N = 0
    _DEADLINE = time.time() + budget_s * 0.95

    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.asarray(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(42, 0.5)

    n0 = len(h0)
    h0 = project_feasible(h0)
    C0 = exact_C(h0)
    _BEST_H = h0.copy()
    _BEST_C = C0
    _BEST_N = n0

    # Generate diverse starting points
    candidates = [h0.copy()]
    candidates.append(project_feasible(h0[::-1]))
    for shift in [1, 2, n0 // 3, n0 // 2]:
        candidates.append(project_feasible(np.roll(h0, shift)))
    for amp in [0.03, 0.06, 0.10, 0.15]:
        for freq in [1, 2, 3, 4]:
            pert = amp * np.sin(2 * np.pi * freq * np.arange(n0) / n0)
            candidates.append(project_feasible(h0 + pert))
    for amp in [0.03, 0.06, 0.10]:
        for freq in [1, 2, 3]:
            pert = amp * np.cos(2 * np.pi * freq * np.arange(n0) / n0)
            candidates.append(project_feasible(h0 + pert))

    for cand in candidates:
        C_cand = exact_C(cand)
        if C_cand < _BEST_C:
            _BEST_C = C_cand
            _BEST_H = cand.copy()
            _BEST_N = n0

    # Build start vector pool for BFGS
    start_vectors = [logit_safe(_BEST_H)]
    start_vectors.append(logit_safe(h0[::-1]))
    start_vectors.append(logit_safe(np.roll(h0, 3)))
    for k in range(1, 5):
        y_struct = 0.3 * k * np.sin(2 * np.pi * k * np.arange(n0) / n0)
        start_vectors.append(y_struct)
    for _ in range(4):
        start_vectors.append(rng.normal(0, 0.5, n0))

    betas = [40.0, 80.0, 160.0, 320.0, 600.0]
    n_sv = len(start_vectors)
    n_beta = len(betas)

    # Multi-round BFGS optimization
    total_rounds = n_sv * n_beta
    round_idx = 0
    while round_idx < total_rounds:
        if time.time() > _DEADLINE - 80:
            break
        s_idx = round_idx % n_sv
        b_idx = (round_idx // n_sv) % n_beta
        y_start = start_vectors[s_idx].copy() + rng.normal(0, 0.005, n0)
        beta = betas[b_idx]
        try:
            maxiter = 80 if n0 <= 45 else 50
            res = minimize(
                smooth_obj_grad, y_start, args=(beta,),
                method='BFGS', jac=True,
                options={'maxiter': maxiter, 'gtol': 1e-7, 'disp': False}
            )
            h_res = feasible_from_logit(res.x)
            h_res = project_feasible(h_res)
            C_res = exact_C(h_res)
            if C_res < _BEST_C - 1e-14:
                _BEST_C = C_res
                _BEST_H = h_res.copy()
                _BEST_N = n0
                start_vectors[s_idx] = logit_safe(_BEST_H)
        except Exception:
            pass
        round_idx += 1

    # Pair polish
    if time.time() < _DEADLINE - 40:
        tl = min(80.0, max(10.0, _DEADLINE - time.time() - 20))
        h_p, C_p = pair_polish(_BEST_H, max_pairs=30, max_iter=15,
                               time_limit=tl, rng=rng)
        if C_p < _BEST_C:
            _BEST_C = C_p
            _BEST_H = h_p.copy()

    # Second BFGS pass from polished result
    if time.time() < _DEADLINE - 50:
        y_start = logit_safe(_BEST_H)
        for beta in [200.0, 500.0, 1000.0]:
            if time.time() > _DEADLINE - 30:
                break
            try:
                res = minimize(
                    smooth_obj_grad, y_start, args=(beta,),
                    method='BFGS', jac=True,
                    options={'maxiter': 60, 'gtol': 1e-8, 'disp': False}
                )
                h_res = feasible_from_logit(res.x)
                h_res = project_feasible(h_res)
                C_res = exact_C(h_res)
                if C_res < _BEST_C - 1e-14:
                    _BEST_C = C_res
                    _BEST_H = h_res.copy()
                    _BEST_N = n0
                    y_start = logit_safe(_BEST_H)
            except Exception:
                pass

    # More pair polish rounds
    for _ in range(3):
        if time.time() > _DEADLINE - 15:
            break
        tl = max(5.0, _DEADLINE - time.time() - 10)
        h_p, C_p = pair_polish(_BEST_H, max_pairs=25, max_iter=10,
                               time_limit=tl, rng=rng)
        if C_p < _BEST_C:
            _BEST_C = C_p
            _BEST_H = h_p.copy()

    # Coordinate descent polish
    if time.time() < _DEADLINE - 10:
        tl = max(5.0, _DEADLINE - time.time() - 5)
        h_cd, C_cd = coord_descent(_BEST_H, time_limit=tl, rng=rng)
        if C_cd < _BEST_C:
            _BEST_C = C_cd
            _BEST_H = h_cd.copy()

    # Final validation
    h_final = _BEST_H
    if h_final is None:
        h_final = np.full(n0, 0.5)
    h_final = project_feasible(h_final)
    C_final = exact_C(h_final)
    n_final = len(h_final)

    # Sanity check
    if C_final > _BEST_C + 1e-6:
        C_final = _BEST_C
        h_final = _BEST_H

    return h_final, C_final, n_final
