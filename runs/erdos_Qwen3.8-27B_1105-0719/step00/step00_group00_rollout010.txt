```python
def thue_morse(n):
    # Generate Thue-Morse sequence of length n, map to {0, 1} with n/2 ones
    # TM: t(i) = popcount(i) % 2
    tm = np.array([bin(i).count('1') % 2 for i in range(n)], dtype=float)
    # tm has roughly n/2 ones; adjust to exact
    num_ones = int(tm.sum())
    target = n // 2
    if num_ones > target:
        # Flip some 1s to 0s
        idx = np.where(tm == 1)[0]
        to_flip = idx[:num_ones - target]
        tm[to_flip] = 0
    elif num_ones < target:
        idx = np.where(tm == 0)[0]
        to_flip = idx[:target - num_ones]
        tm[to_flip] = 1
    return tm
```

- h[i] appears in (1-h[m]) when m = i (if 0 ≤ i ≤ n-1, always true)
  But we also need h[i+k] to be defined, i.e., 0 ≤ i+k ≤ n-1
  Contribution change: h[i+k] * (-δ_i) = -δ_i * h[i+k]

Similarly for h[j] by δ_j.

This allows O(n) update per swap (one for each lag), instead of O(n) for full recomputation. But since np.correlate is already O(n), the constant factor might not matter much.

```python
        G = np.zeros((len(active), n))

            k = s - (n - 1)
            # G[s, j] = dx * d/dh[j] sum_m h[m+k]*(1-h[m])

                    g += (1 - h[j - k])
                if 0 <= j + k < n:
                    g -= h[j + k]
                G[idx, j] = dx * g

        c_obj[-1] = 1.0

        b_ub = np.zeros(len(active))
        for idx in range(len(active)):
            A_ub[idx, :n] = G[idx, :]
            A_ub[idx, n] = -1.0
            b_ub[idx] = -c[active[idx]]
        
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)
        
        bounds = [(-h[i], 1 - h[i]) for i in range(n)] + [(None, None)]
        
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
```

```python
        if 0 <= i + k < n:

        if 0 <= j + k < n:
            d += h[j + k]

def smooth_objective(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    c = dx * corr
    max_c = np.max(c)
    shifted = c - max_c
    exp_vals = np.exp(beta * shifted)
    return max_c + np.log(np.sum(exp_vals)) / beta

def smooth_gradient(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    c = dx * corr
    max_c = np.max(c)
    shifted = c - max_c
    exp_vals = np.exp(beta * shifted)
    weights = exp_vals / np.sum(exp_vals)
    
    grad = np.zeros(n)
    for s in range(2 * n - 1):
        k = s - (n - 1)
        for j in range(n):
            g = 0.0
            if 0 <= j - k < n:
                g += (1.0 - h[j - k])
            if 0 <= j + k < n:
                g -= h[j + k]
            grad[j] += weights[s] * dx * g
    return grad

def smooth_optimize(h, n, beta, time_limit=30, max_iters=200):
    start = time.time()
    h = h.copy()
    c5 = evaluate_c5(h)
    
    x0 = h.copy()
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
                    'jac': lambda x: np.ones(n)}]
    
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        res = minimize(smooth_objective, x0, args=(beta,),
                       jac=smooth_gradient, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': 100, 'ftol': 1e-10})
        if res.success and res.fun < c5:
            h_new = np.clip(res.x, 0, 1)
            # Renormalize
            scale = (n / 2.0) / np.sum(h_new)
            h_new = np.clip(h_new * scale, 0, 1)
            if abs(np.sum(h_new) - n / 2.0) > 1e-6:
                # Fine adjust
                diff = n / 2.0 - np.sum(h_new)
                pos = np.argsort(h_new)
                if diff > 0:
                    for idx in range(len(pos)):
                        room = 1.0 - h_new[pos[idx]]
                        adj = min(room, diff)
                        h_new[pos[idx]] += adj
                        diff -= adj
                        if diff < 1e-10:
                            break
                else:
                    for idx in range(len(pos) - 1, -1, -1):
                        room = h_new[pos[idx]]
                        adj = min(room, -diff)
                        h_new[pos[idx]] -= adj
                        diff += adj
                        if diff > -1e-10:
                            break
            c5_new = evaluate_c5(h_new)
            if c5_new < c5:
                h = h_new
                c5 = c5_new
                x0 = h.copy()
            else:
                break
        else:
            break
    
    return h, c5

def lp_refine(h, n, time_limit=30, max_iters=50):
    start = time.time()
    h = h.copy()
    dx = 2.0 / n
    c5 = evaluate_c5(h)
    
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode="full")
        c = dx * corr
        max_c = np.max(c)
        
        # Active set: lags within 25% of max
        tol = 0.02 * max_c if max_c > 0 else 0.01
        active = np.where(c > max_c - tol)[0]
        if len(active) == 0:
            active = np.where(c == max_c)[0]
        if len(active) == 0:
            break
        
        # Compute gradients for active lags
        G = np.zeros((len(active), n))
        for idx, s in enumerate(active):
            k = s - (n - 1)
            for j in range(n):
                g = 0.0
                if 0 <= j - k < n:
                    g += (1.0 - h[j - k])
                if 0 <= j + k < n:
                    g -= h[j + k]
                G[idx, j] = dx * g
        
        # LP: min t s.t. G@delta - t <= -c[active], sum(delta)=0
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        
        A_ub = np.zeros((len(active), n + 1))
        b_ub = np.zeros(len(active))
        for idx in range(len(active)):
            A_ub[idx, :n] = G[idx, :]
            A_ub[idx, n] = -1.0
            b_ub[idx] = -c[active[idx]]
        
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)
        
        # Tighten bounds for trust region
        trust = 0.5
        bounds = []
        for i in range(n):
            lo = max(-h[i], -trust)
            hi = min(1.0 - h[i], trust)
            if lo > hi:
                lo, hi = 0.0, 0.0
            bounds.append((lo, hi))
        bounds.append((None, None))
        
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        
        if res.status != 0:
            break
        
        delta = res.x[:n]
        if np.max(np.abs(delta)) < 1e-10:
            break
        
        # Line search
        best_alpha = 0.0
        best_c5_trial = c5
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            h_trial = h + alpha * delta
            if h_trial.min() < -1e-9 or h_trial.max() > 1 + 1e-9:
                continue
            h_trial = np.clip(h_trial, 0, 1)
            c5_trial = evaluate_c5(h_trial)
            if c5_trial < best_c5_trial - 1e-12:
                best_c5_trial = c5_trial
                best_alpha = alpha
        
        if best_alpha == 0.0:
            break
        
        h = h + best_alpha * delta
        h = np.clip(h, 0, 1)
        c5 = evaluate_c5(h)
    
    return h, c5

def coordinate_descent(h, n, time_limit=30):
    start = time.time()
    h = h.copy()
    c5 = evaluate_c5(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    dx = 2.0 / n
    max_corr = np.max(corr)
    
    improved = True
    while improved:
        if time.time() - start > time_limit:
            break
        improved = False
        for i in range(n):
            best_val = h[i]
            best_c5_i = c5
            # Try several values
            candidates = [0.0, 1.0, h[i] + 0.1, h[i] - 0.1, h[i] + 0.05, h[i] - 0.05]
            candidates = [max(0.0, min(1.0, v)) for v in candidates]
            candidates = list(set(candidates))
            for val in candidates:
                if abs(val - h[i]) < 1e-10:
                    continue
                # Adjust another coordinate to maintain sum
                diff = val - h[i]
                # Find partner to adjust
                adjust_found = False
                for j in range(n):
                    if j == i:
                        continue
                    adj = h[j] - diff
                    if 0.0 <= adj <= 1.0:
                        h[i] = val
                        h[j] = adj
                        new_c5 = evaluate_c5(h)
                        if new_c5 < best_c5_i - 1e-12:
                            best_c5_i = new_c5
                            best_val = val
                            best_partner = j
                            best_adj = adj
                            adjust_found = True
                        h[i] = h[i] + diff  # restore
                        h[j] = h[j] + diff  # restore
                        break
                if not adjust_found:
                    h[i] = h[i] + diff if i < n else h[i]
            
            if best_c5_i < c5 - 1e-12:
                h[i] = best_val
                h[best_partner] = best_adj
                c5 = best_c5_i
                improved = True
                corr = np.correlate(h, 1.0 - h, mode="full")
    
    return h, c5

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)
    
    best_c5 = float('inf')
    best_h = None
    best_n = 0
    
    def time_left():
        return budget_s - (time.time() - start_time)
    
    # Try different n values and starting points
    n_list = [45, 46, 48, 50, 52, 54, 56, 58, 60, 64, 72, 80]
    
    for n in n_list:
        if time_left() < 45:
            break
        
        max_n_time = min(60, time_left() * 0.6 / len(n_list))
        n_start_time = time.time()
        
        # Starting points
        starts = []
        
        # From initial_h_values if available and n matches
        if n == 45:
            try:
                iv = np.array(initial_h_values, dtype=float)
                if len(iv) == n:
                    starts.append(iv.copy())
            except:
                pass
        
        starts.append(thue_morse_binary(n))
        for _ in range(3):
            starts.append(random_binary(n, rng))
        starts.append(block_pattern(n, rng))
        # Scaled Thue-Morse variant
        tm2 = thue_morse_binary(2 * n)
        # Average pairs
        h2 = np.array([(tm2[2*i] + tm2[2*i+1]) / 2.0 for i in range(n)])
        target = n / 2.0
        # Adjust
        diff = target - h2.sum()
        if abs(diff) > 1e-10:
            for i in np.argsort(h2)[::1 if diff > 0 else -1]:
                adj = min(1.0 - h2[i] if diff > 0 else h2[i], abs(diff))
                h2[i] += adj * np.sign(diff)
                diff -= adj * np.sign(diff)
                if abs(diff) < 1e-9:
                    break
        starts.append(h2)
        
        for si, h0 in enumerate(starts):
            if time.time() - n_start_time > max_n_time:
                break
            if time_left() < 15:
                break
            
            h0 = np.clip(h0, 0, 1)
            # Normalize sum
            s = h0.sum()
            if s > 0:
                h0 = h0 * (n / 2.0) / s
                h0 = np.clip(h0, 0, 1)
            # Fine adjust
            diff = n / 2.0 - h0.sum()
            if abs(diff) > 1e-9:
                if diff > 0:
                    for i in np.argsort(h0):
                        room = 1.0 - h0[i]
                        adj = min(room, diff)
                        h0[i] += adj
                        diff -= adj
                        if diff < 1e-9:
                            break
                else:
                    for i in np.argsort(h0)[::-1]:
                        room = h0[i]
                        adj = min(room, -diff)
                        h0[i] -= adj
                        diff += adj
                        if diff > -1e-9:
                            break
            
            c5 = evaluate_c5(h0)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h0.copy()
                best_n = n
            
            # Local search (binary)
            t_rem = time_left() - 5
            ls_time = min(15, max(3, t_rem * 0.3))
            h_ls, c5_ls = local_search_binary(h0, n, rng, time_limit=ls_time)
            if c5_ls < best_c5:
                best_c5 = c5_ls
                best_h = h_ls.copy()
                best_n = n
            
            # LP refinement
            if time_left() > 20:
                lp_time = min(20, max(5, time_left() * 0.2))
                h_lp, c5_lp = lp_refine(h_ls, n, time_limit=lp_time)
                if c5_lp < best_c5:
                    best_c5 = c5_lp
                    best_h = h_lp.copy()
                    best_n = n
            
            # Smooth optimization
            if time_left() > 15:
                sm_time = min(15, max(3, time_left() * 0.15))
                h_sm, c5_sm = smooth_optimize(h_lp, n, beta=80, time_limit=sm_time)
                if c5_sm < best_c5:
                    best_c5 = c5_sm
                    best_h = h_sm.copy()
                    best_n = n
            
            # Second round local search from LP result
            if time_left() > 10:
                ls2_time = min(10, time_left() * 0.1)
                h_ls2, c5_ls2 = local_search_binary(h_sm, n, rng, time_limit=ls2_time)
                if c5_ls2 < best_c5:
                    best_c5 = c5_ls2
                    best_h = h_ls2.copy()
                    best_n = n
        
        # End of n loop
    
    # Final LP polish on the best solution
    if best_h is not None and time_left() > 10:
        final_time = min(60, time_left() - 5)
        h_final, c5_final = lp_refine(best_h, best_n, time_limit=final_time)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final
    
    if best_h is None:
        # Fallback
        n = 46
        best_h = thue_morse_binary(n)
        best_n = n
        best_c5 = evaluate_c5(best_h)
    
    return best_h, best_c5, best_n
```
