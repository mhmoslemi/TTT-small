import numpy as np
from scipy.optimize import linprog
import time

def _compute_c5(h, n, dx):
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full')
    return float(np.max(corr) * dx)

def _compute_corr(dx_scaled, h, n, dx):
    g = 1.0 - h
    return np.correlate(h, g, mode='full') * dx

def _grad_at_m(h, m, n):
    """Gradient of (dx * corr[m]) w.r.t. h, where m is output index."""
    k = m - (n - 1)
    grad = np.zeros(n)
    # For each j: d/dh[j] of sum_n h[n+k]*(1-h[n]) * dx
    # = dx * [ (1-h[j-k]) if j-k valid - h[j+k] if j+k valid ]
    for j in range(n):
        v = 0.0
        jm = j - k
        if 0 <= jm < n:
            v += (1.0 - h[jm])
        jp = j + k
        if 0 <= jp < n:
            v -= h[jp]
        grad[j] = v
    return grad * dx

def _lp_direction(h, active_indices, c5_vals, n, dx):
    """Solve LP to find direction minimizing linearized active max."""
    m_act = len(active_indices)
    # Variables: [u_0, ..., u_{n-1}, t]
    # Minimize t
    # Subject to:
    #   c5_vals[ai] + grad_ai · u <= t  for each active
    #   sum(u) = 0
    #   -h[i] <= u[i] <= 1-h[i]
    
    A_ub = np.zeros((m_act, n + 1))
    b_ub = np.zeros(m_act)
    
    for idx, m in enumerate(active_indices):
        g = _grad_at_m(h, m, n)
        A_ub[idx, :n] = g
        A_ub[idx, n] = -1.0
        b_ub[idx] = -c5_vals[m]
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    bounds = [(max(-h[i], -1.0), min(1.0 - h[i], 1.0)) for i in range(n)]
    bounds.append((None, None))
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.status != 0:
        return None
    return res.x[:n], res.x[n]

def _line_search(h, direction, c5_current, n, dx, max_alpha=1.0):
    """Backtracking line search along direction."""
    # Find max feasible alpha
    if np.max(np.abs(direction)) < 1e-15:
        return h, c5_current, 0.0
    
    # Box constraint: h + alpha*d in [0,1]
    max_a = max_alpha
    for i in range(n):

            max_a = min(max_a, h[i] / (-direction[i]))
        elif direction[i] > 1e-15:
            max_a = min(max_a, (1.0 - h[i]) / direction[i])
    max_a = max(0.0, min(max_a, max_alpha))
    
    # Backtracking
    alpha = max_a
    for _ in range(40):

        h_trial = np.clip(h_trial, 0.0, 1.0)
        c5_trial = _compute_c5(h_trial, n, dx)
        if c5_trial < c5_current - 1e-12:
            return h_trial, c5_trial, alpha
        alpha *= 0.5
        if alpha < 1e-15:
            break
    return h, c5_current, 0.0

def _pair_optimize(h, i, j, n, dx):

    c = h[i] + h[j]
    if c < 0 or c > 2:

    best_val = h[i]

import numpy as np
from scipy.optimize import linprog
import time

def _c5(h, n, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _corr_all(h, n, dx):
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _grad_m(h, m, n, dx):
    k = m - (n - 1)
    g = np.empty(n)
    for j in range(n):
        v = 0.0
        a = j - k
        if 0 <= a < n:
            v += 1.0 - h[a]
        b = j + k
        if 0 <= b < n:
            v -= h[b]
        g[j] = v * dx
    return g

def _lp_dir(h, act, cvals, n, dx):
    ma = len(act)
    if ma == 0:
        return None, None
    A = np.zeros((ma, n + 1))
    b = np.zeros(ma)
    for i, m in enumerate(act):
        A[i, :n] = _grad_m(h, m, n, dx)
        A[i, n] = -1.0
        b[i] = -cvals[m]
    Ae = np.zeros((1, n + 1))
    Ae[0, :n] = 1.0
    ce = np.zeros(1)
    c = np.zeros(n + 1)
    c[n] = 1.0
    bounds = list(zip(-h, 1.0 - h)) + [(None, None)]
    r = linprog(c, A_ub=A, b_ub=b, A_eq=Ae, b_eq=ce, bounds=bounds, method='highs')
    if r.status == 0:
        return r.x[:n], r.x[n]
    return None, None

def _line(h, d, c0, n, dx):
    if np.max(np.abs(d)) < 1e-14:

        elif d[i] > 1e-14:
            ma = min(ma, (1.0 - h[i]) / d[i])

def _pair(h, i, j, n, dx, ng=80):
    c = h[i] + h[j]
    lo = max(0.0, c - 1.0)
    hi = min(1.0, c)
    if hi - lo < 1e-10:
        return h, _c5(h, n, dx)
    bv = h[i]
    bc = _c5(h, n, dx)
    for t in np.linspace(lo, hi, ng):
        h[i] = t
        h[j] = c - t
        cv = _c5(h, n, dx)
        if cv < bc - 1e-15:
            bc = cv
            bv = t
    h[i] = bv
    h[j] = c - bv
    return h, bc
