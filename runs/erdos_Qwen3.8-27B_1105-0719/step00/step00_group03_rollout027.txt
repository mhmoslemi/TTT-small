    n = len(y)
    lo, hi = -max(np.max(y), 1.0) - 1.0, max(1.0 - np.min(y), 1.0) + 1.0

            break
        hi = hi * 2 + 1
    for _ in range(80):

        s = np.sum(np.clip(y + mid, 0, 1))
        if s > target:
            hi = mid
        else:
            lo = mid
    x = np.clip(y + (lo + hi) / 2, 0, 1)
    # Fix rounding
    diff = target - np.sum(x)
    if abs(diff) > 1e-12:

                diff -= adj
            elif diff < 0 and x[i] > 0.0:

def exact_c5(h):
    """Compute C5 and full scores vector."""
    n = len(h)
    dx = 2.0 / n
    raw = np.correlate(h, 1.0 - h, mode='full')
    scores = raw * dx

def project_feasible(y, target):
    """Project y onto {x in [0,1]^n : sum(x) = target}."""
    y = np.asarray(y, dtype=np.float64)
    n = len(y)
    # Bisection for the shift
    lo = -max(float(np.max(y)), 1.0) - 1.0
    hi = max(1.0 - float(np.min(y)), 1.0) + 1.0
    # Ensure bracket
    for _ in range(30):
        if np.sum(np.clip(y + lo, 0.0, 1.0)) >= target:
            break
        lo *= 2.0 - 1.0
    for _ in range(30):
        if np.sum(np.clip(y + hi, 0.0, 1.0)) <= target:
            break
        hi = hi * 2.0 + 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(y + mid, 0.0, 1.0))
        if s > target:
            hi = mid
        else:
            lo = mid
    x = np.clip(y + 0.5 * (lo + hi), 0.0, 1.0)
    # Fix sum to exact target
    diff = target - np.sum(x)
    if abs(diff) > 1e-12:
        # Distribute diff to interior points
        interior = (x > 1e-10) & (x < 1.0 - 1e-10)
        n_int = np.sum(interior)
        if n_int > 0:
            adj = diff / n_int
            x[interior] += adj
            # Re-clip and fix again
            x = np.clip(x, 0.0, 1.0)
            diff2 = target - np.sum(x)
            if abs(diff2) > 1e-12:
                # Greedy fix
                idx = np.argsort(x - 0.5)
                for i in idx:
                    if diff2 > 0:
                        room = 1.0 - x[i]
                        add = min(diff2, room)
                        x[i] += add
                        diff2 -= add
                    else:
                        room = x[i]
                        sub = min(-diff2, room)
                        x[i] -= sub
                        diff2 += sub
                    if abs(diff2) < 1e-14:
                        break
    return x

def compute_subgradient(h, active_k, dx):
    """Compute subgradient of weighted active lag scores."""
    n = len(h)
    g = np.zeros(n)
    for k in active_k:
        k = int(k)
        if k >= 0:
            if k == 0:
                g += dx * (1.0 - h)
                g -= dx * h
            else:
                g[k:] += dx * (1.0 - h[:n-k])
                g[:n-k] -= dx * h[k:]
        else:
            d = -k
            g[:n-d] += dx * (1.0 - h[d:])
            g[d:] -= dx * h[:n-d]
    return g

def resample_to(h, n_new):
    """Resample h from n_old to n_new using exact area integration."""
    n_old = len(h)
    dx_old = 2.0 / n_old
    dx_new = 2.0 / n_new
    child = np.zeros(n_new)
    # For each new cell, compute integral of h over that cell
    # New cell j covers [j*dx_new, (j+1)*dx_new)
    # Old cell i covers [i*dx_old, (i+1)*dx_old)
    for j in range(n_new):
        # Find overlapping old cells
        j_start = j * dx_new
        j_end = (j + 1) * dx_new
        i_lo = int(np.floor(j_start / dx_old))
        i_hi = int(np.ceil(j_end / dx_old)) - 1
        i_lo = max(0, i_lo)
        i_hi = min(n_old - 1, i_hi)
        total = 0.0
        for i in range(i_lo, i_hi + 1):
            i_start = i * dx_old
            i_end = (i + 1) * dx_old
            overlap = min(j_end, i_end) - max(j_start, i_start)
            if overlap > 0:
                total += h[i] * overlap
        child[j] = total / dx_new
    return child

def compute_subgradient(h, active_k, dx):
    n = len(h)
    g = np.zeros(n)
    for k in active_k:
        k = int(k)
        if k == 0:
            g += dx * (1.0 - 2.0 * h)
        elif k > 0:
            g[k:] += dx * (1.0 - h[:n - k])
            g[:n - k] -= dx * h[k:]
        else:
            d = -k
            g[:n - d] += dx * (1.0 - h[d:])
            g[d:] -= dx * h[:n - d]
    return g

def resample_to(h, n_new):
    n_old = len(h)
    if n_new == n_old:
        return h.copy()
    dx_old = 2.0 / n_old
    dx_new = 2.0 / n_new
    child = np.zeros(n_new)
    for j in range(n_new):
        j_start = j * dx_new
        j_end = (j + 1) * dx_new
        i_lo = max(0, int(np.floor(j_start / dx_old)))
        i_hi = min(n_old - 1, int(np.ceil(j_end / dx_old)) - 1)
        total = 0.0
        for i in range(i_lo, i_hi + 1):
            overlap = min(j_end, (i + 1) * dx_old) - max(j_start, i * dx_old)
            if overlap > 0:
                total += h[i] * overlap
        child[j] = total / dx_new
    return child
