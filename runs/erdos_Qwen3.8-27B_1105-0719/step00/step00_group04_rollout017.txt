def pair_optimize(h, i, j, max_iter=50):
    n = len(h)
    T = h[i] + h[j]
    lo = max(0.0, T - 1.0)
    hi = min(1.0, T)
    if hi - lo < 1e-12:
        return h
    
    def obj(v):
        h_new = h.copy()
        h_new[i] = v
        h_new[j] = T - v
        return evaluate_c5(h_new)
    
    result = minimize_scalar(obj, bounds=(lo, hi), method='bounded', options={'maxiter': max_iter})

        h[j] = T - result.x
    return h

def lp_move(h, n):
    b = 1.0 - h
    dx = 2.0 / n
    n_lags = 2 * n - 1
    
    # Build constraint matrix
    # For lag L (from -(n-1) to n-1), the correlation is:
    # sum_{i: valid} h[i] * b[i+L]
    # In terms of x (new h):
    # sum_{i: valid} x[i] * b[i+L] - t <= 0
    
    A_rows = []
    for k in range(n_lags):
        L = (n - 1) - k  # lag
        row = np.zeros(n + 1)
        if L >= 0:
            for i in range(n - L):
                row[i] = b[i + L]
        else:
            for i in range(-L, n):
                row[i] = b[i + L]
        row[n] = -1.0  # coefficient of t
        A_rows.append(row)
    
    A_ub = np.array(A_rows)
    b_ub = np.zeros(n_lags)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize t
    
    bounds = [(0, 1)] * n + [(None, None)]
    
    try:
        from scipy.optimize import linprog
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                         bounds=bounds, method='highs')

            x_new = project_feasible(x_new, n)

        pass
    return h.copy()

def obj_eval(h, i, j, T, v):
    """Evaluate objective at pair (i,j) with h[i]=v, h[j]=T-v."""
    save_i = h[i]
    save_j = h[j]
    h[i] = v
    h[j] = T - v
    val = np.max(np.correlate(h, 1.0 - h, mode='full'))
    h[i] = save_i
    h[j] = save_j
    return val

def golden_section_min(h, i, j, T, lo, hi, max_iter=100):
    """Minimize obj_eval over v in [lo, hi] using golden section search."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0  # golden ratio
    a = lo
    b = hi
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = obj_eval(h, i, j, T, c)
    fd = obj_eval(h, i, j, T, d)
    for _ in range(max_iter):
        if b - a < 1e-12:
            break
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = obj_eval(h, i, j, T, c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = obj_eval(h, i, j, T, d)
    return 0.5 * (a + b)
