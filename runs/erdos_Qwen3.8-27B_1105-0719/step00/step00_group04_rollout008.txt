```python
def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(corr)

def project_feasible(h):
    """Project h to be feasible: 0 <= h_i <= 1, sum(h) = n/2"""
    n = len(h)
    target = n / 2.0
    h = np.clip(h, 0, 1)
    # Adjust sum to be exactly target
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        # Distribute the difference
        if diff > 0:
            # Need to increase sum
            room = 1.0 - h
            idx = np.argsort(-room)
            for i in idx:
                add = min(diff, room[i])
                h[i] += add
                diff -= add
                if abs(diff) < 1e-12:
                    break
        else:
            # Need to decrease sum
            idx = np.argsort(h)
            for i in idx:
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub
                if abs(diff) < 1e-12:
                    break
    return h

def solve_primal_lp(h_cur, b_cur):
    """Solve LP: minimize t s.t. dx * sum a_i * b_j <= t for all lags, sum(a)=n/2, 0<=a<=1"""
    n = len(h_cur)
    dx = 2.0 / n
    target = n / 2.0
    M = 2 * n - 1
    
    a = cp.Variable(n)
    t = cp.Variable()

            else:
```

```python
            else:
```

  - Derivative w.r.t. h[p]: -h[p-L]

```python
def compute_gradient_matrix(h):
    """Compute G[i, m] = dR_m / dh[i]"""
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    G = np.zeros((n, M))
    
    for m in range(M):
        L = (n - 1) - m
        # Valid first indices: i in [max(0,-L), min(n-1, n-1-L)]
        i_lo = max(0, -L)
        i_hi = min(n - 1, n - 1 - L)
        
        for i in range(i_lo, i_hi + 1):
            # i is first index: contribution (1 - h[i+L])
            G[i, m] += dx * (1.0 - h[i + L])
            # i+L is second index: contribution -h[i]
            G[i + L, m] += -dx * h[i]
    
    return G, dx
```

```python
    n = len(h)
```

        # Log-sum-exp

        weights = np.exp(tau * (corr - max_c))
        weights /= np.sum(weights)
        # dR_m/dh[i] for all i, m
        G = compute_grad_matrix(h_tmp)
        # grad[i] = sum_m weights[m] * G[i, m]
        grad = G @ weights
        return grad
    
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0, 'jac': lambda x: np.ones(n)}]
    bounds = [(0, 1)] * n
    
    result = minimize(objective, h_work, jac=gradient, method='SLSQP',
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': 200, 'ftol': 1e-12})
    
    if result.success:
        h_new = project_feasible(result.x)
        return h_new
    return h_work

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    
    # Get initial values
    h = np.array(initial_h_values, dtype=float).copy()
    h = project_feasible(h)
    n = len(h)
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    def time_remaining():
        return budget_s - (time.time() - start_time)
    
    # Phase 1: Alternating LP
    for it in range(30):
        if time_remaining() < 60:
            break
        
        # Primal LP
        h_p = solve_primal_lp(h)
        if h_p is not None:
            h_p = project_feasible(h_p)
            # Line search
            best_alpha = 0
            best_val = best_c5
            for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 0.8, 1.0]:
                h_try = project_feasible((1-alpha)*h + alpha*h_p)
                val = evaluate_c5(h_try)
                if val < best_val:
                    best_val = val
                    best_alpha = alpha
            if best_alpha > 0:
                h = project_feasible((1-best_alpha)*h + alpha*h_p)
                h = project_feasible(h)
                if best_val < best_c5:
                    best_c5 = best_val
                    best_h = h.copy()
        
        # Dual LP
        b_d = solve_dual_lp(h)
        if b_d is not None:
            h_d = 1.0 - b_d
            h_d = project_feasible(h_d)
            best_alpha = 0
            best_val = best_c5
            for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 0.8, 1.0]:
                h_try = project_feasible((1-alpha)*h + alpha*h_d)
                val = evaluate_c5(h_try)
                if val < best_val:
                    best_val = val
                    best_alpha = alpha
            if best_alpha > 0:
                h = project_feasible((1-best_alpha)*h + alpha*h_d)
                h = project_feasible(h)
                if best_val < best_c5:
                    best_c5 = best_val
                    best_h = h.copy()
    
    # Phase 2: Pair optimization on main h
    if time_remaining() > 60:
        h, c5_val = pair_optimize(h, time_limit=min(120, time_remaining() - 60))
        if c5_val < best_c5:
            best_c5 = c5_val
