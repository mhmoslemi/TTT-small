```python
def optimize_continuous(h0, n, budget_s, start_time):

    target = n / 2.0
    best_h = h.copy()
    best_c5 = compute_c5(h, n)

        grad_proj = grad - grad.mean()
```

```python
    R = np.correlate(h, 1-h, mode='full')
    maxR = R.max()
    shifted = beta * (R - maxR)
    shifted -= shifted.max()
    exp_vals = np.exp(shifted)
    w = exp_vals / exp_vals.sum()  # length 2n-1
    
    grad = np.zeros(n)
    for p in range(n):
        # Sum over all lags k from -(n-1) to n-1
        for k in range(-(n-1), n):
            t = k + n - 1  # index into R and w
            # First term: (1-h[p-k]) if valid
            # valid: 0 <= p-k < n and (p-k) in valid range for lag k
            # For lag k, valid m: max(0,-k) <= m <= min(n-1, n-1-k)
            # First term: m = p-k, valid if max(0,-k) <= p-k <= min(n-1, n-1-k)
            m1 = p - k
            if 0 <= m1 < n and max(0, -k) <= m1 <= min(n-1, n-1-k):

    return grad
```

```python
    R = np.correlate(h, 1-h, mode='full')
    maxR = R.max()

    exp_vals = np.exp(shifted)

        # k >= 0: k from 0 to n-1-p (for second term) or up to p (for first term)
        # First term: k from 0 to p, contributes (1-h[p-k]) * w[k+n-1]
```

def project_feasible(y, target, n):
    """Project y onto {x: 0<=x_i<=1, sum(x)=target}"""
    y = y.copy()
    # Bisection on lambda
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        phi = np.clip(y - mid, 0.0, 1.0).sum()
        if phi > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix residual
    residual = target - x.sum()
    if abs(residual) > 1e-12:
        if residual > 0:
            for i in range(n):
                if x[i] < 1.0:
                    add = min(residual, 1.0 - x[i])
                    x[i] += add
                    residual -= add
                    if abs(residual) < 1e-12:
                        break
        else:
            for i in range(n):
                if x[i] > 0.0:
                    rem = min(-residual, x[i])
                    x[i] -= rem
                    residual += rem
                    if abs(residual) < 1e-12:
                        break
    return x

def compute_c5(h, n):
    """Compute C5 = dx * max(correlate(h, 1-h, mode='full'))"""
    dx = 2.0 / n
    R = np.correlate(h, 1.0 - h, mode='full')
    return dx * R.max()

def compute_lse_grad(h, n, beta):
    """Compute LSE-smoothed objective and gradient"""
    R = np.correlate(h, 1.0 - h, mode='full')
    maxR = R.max()
    shifted = beta * (R - maxR)
    shifted -= shifted.max()
    exp_vals = np.exp(shifted)
    w = exp_vals / exp_vals.sum()
    obj = maxR + np.log(exp_vals.sum()) / beta
    
    n1 = n - 1
    grad = np.zeros(n)
    for p in range(n):
        # Term 1: sum_{j=0}^{p} w[n-1+p-j]*(1-h[j])
        for j in range(p + 1):
            grad[p] += w[n1 + p - j] * (1.0 - h[j])
        # Term 2: -sum_{j=p}^{n-1} w[n-1+j-p]*h[j]
        for j in range(p, n):
            grad[p] -= w[n1 + j - p] * h[j]
        # Term 3: sum_{j=1}^{n-1-p} w[n-1-j]*(1-h[p+j])
        for j in range(1, n - p):
            grad[p] += w[n1 - j] * (1.0 - h[p + j])
        # Term 4: -sum_{j=1}^{p} w[n-1-j]*h[p-j]
        for j in range(1, p + 1):
            grad[p] -= w[n1 - j] * h[p - j]
    return obj, grad, R
