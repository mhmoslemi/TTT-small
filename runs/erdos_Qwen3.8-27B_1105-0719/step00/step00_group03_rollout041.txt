            _alpha_val = alpha
            cons = [{'type': 'eq', 'fun': _csum, 'jac': _cjac}]
            bounds = [(0.0, 1.0)] * n
            maxiter = 250 if n <= 50 else 120

            for pert in [0.0, 0.015, 0.04]:
                if time.time() - t0 > budget_s - 20:
                    break
                if pert > 0.0:
                    x0 = repair_sum(h + rng.normal(0, pert, n), n * 0.5)
                else:
                    x0 = h.copy()
                try:
                    res = minimize(lse_obj_wrap, x0, jac=lse_grad_wrap,
                                   method='SLSQP', bounds=bounds,
                                   constraints=cons,
                                   options={'maxiter': maxiter, 'ftol': 1e-15})
                    if res.x is not None and np.all(np.isfinite(res.x)):
                        h_new = repair_sum(res.x, n * 0.5)
                        c5 = compute_c5(h_new)
                        if c5 < best_c5:
                            best_c5 = c5
                            best_h = h_new.copy()
                        if c5 < compute_c5(h):
                            h = h_new
                except Exception:
                    pass

    # Phase 2b: Try trust-constr on best few
    for h_start in cands_sorted[:3]:
        if time.time() - t0 > budget_s - 60:
            break
        n = len(h_start)
        h = h_start.copy()
        for alpha in [50, 200, 1000]:
            if time.time() - t0 > budget_s - 30:
                break
            _alpha_val = alpha
            from scipy.optimize import LinearConstraint
            A = np.ones((1, n))
            lc = LinearConstraint(A, n * 0.5, n * 0.5)
            bounds = [(0.0, 1.0)] * n
            try:
                res = minimize(lse_obj_wrap, h, jac=lse_grad_wrap,
                               method='trust-constr', bounds=bounds,
                               constraints=[lc],
                               options={'maxiter': 200, 'gtol': 1e-8})
                if res.x is not None and np.all(np.isfinite(res.x)):
                    h_new = repair_sum(res.x, n * 0.5)
                    c5 = compute_c5(h_new)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h_new.copy()
                        h = h_new
            except Exception:
                pass

    # Phase 3: Local pair-perturbation search
    if best_h is not None and time.time() - t0 < budget_s - 20:
        h = best_h.copy()
        n = len(h)
        step = 0.05
        stale = 0
        while time.time() - t0 < budget_s - 10 and stale < 8000:
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue
            sign = 1.0 if rng.random() < 0.5 else -1.0
            d = step * sign
            # Clamp to stay in bounds
            if sign > 0:
                d_max = min(h[i], 1.0 - h[j])
                d = min(d, d_max)
            else:
                d_max = min(h[j], 1.0 - h[i])
                d = max(d, -d_max)
            if abs(d) < 1e-12:
                continue
            h[i] += d
            h[j] -= d
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                stale = 0
            else:
                h[i] -= d
                h[j] += d
                stale += 1
            if stale % 2000 == 0 and stale > 0:
                step *= 0.85

    # Phase 4: Fine polish
    if best_h is not None and time.time() - t0 < budget_s - 5:
        h = best_h.copy()
        n = len(h)
        step = 0.005
        stale = 0
        while time.time() - t0 < budget_s - 2 and stale < 5000:
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue
            sign = 1.0 if rng.random() < 0.5 else -1.0
            d = step * sign
            if sign > 0:
                d_max = min(h[i], 1.0 - h[j])
                d = min(d, d_max)
            else:
                d_max = min(h[j], 1.0 - h[i])
                d = max(d, -d_max)
            if abs(d) < 1e-12:
                continue
            h[i] += d
            h[j] -= d
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                stale = 0
            else:
                h[i] -= d
                h[j] += d
                stale += 1

    # Final feasibility
    if best_h is not None:
        best_h = repair_sum(best_h, len(best_h) * 0.5)
        best_c5 = compute_c5(best_h)
        n_pts = len(best_h)
    else:
        n_pts = 45
        best_h = repair_sum(np.full(n_pts, 0.5), n_pts * 0.5)
        best_c5 = compute_c5(best_h)

    return (best_h, best_c5, int(len(best_h)))
