import numpy as np
import time

def exact_score(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def feasible_interval(h, u):
    t_lo = 0.0
    t_hi = 0.0
    for i in range(len(h)):
        ui = u[i]
        if abs(ui) < 1e-15:
            continue
        if ui > 0:
            v = -h[i] / ui
            if v > t_lo:
                t_lo = v
            v = (1.0 - h[i]) / ui
            if v < t_hi:
                t_hi = v
        else:
            v = (1.0 - h[i]) / ui
            if v > t_lo:
                t_lo = v
            v = -h[i] / ui
            if v < t_hi:
                t_hi = v
    if t_lo > 0.0:
        t_lo = 0.0
    if t_hi < 0.0:
        t_hi = 0.0
    return t_lo, t_hi

def solve_on_ray(h, u, t_lo, t_hi):
    omh = 1.0 - h
    q0 = np.correlate(h, omh, mode="full")
    q1 = np.correlate(u, omh, mode="full") - np.correlate(h, u, mode="full")
    q2 = -np.correlate(u, u, mode="full")
    m = q0.shape[0]
    cand_list = [t_lo, t_hi, 0.0]
    for a in range(m):
        if q2[a] > 1e-10:
            t = -q1[a] / (2.0 * q2[a])
            if t_lo <= t <= t_hi:
                cand_list.append(t)
    for a in range(m):
        qa0, qa1, qa2 = q0[a], q1[a], q2[a]
        for b in range(a + 1, m):
            da = qa2 - q2[b]
            db = qa1 - q1[b]
            dc = qa0 - q0[b]
            if abs(da) < 1e-13:
                if abs(db) > 1e-13:
                    t = -dc / db
                    if t_lo <= t <= t_hi:
                        cand_list.append(t)
            else:
                disc = db * db - 4.0 * da * dc
                if disc >= 0:
                    sd = np.sqrt(disc)
                    t1 = (-db + sd) / (2.0 * da)
                    t2 = (-db - sd) / (2.0 * da)
                    if t_lo <= t1 <= t_hi:
                        cand_list.append(t1)
                    if t_lo <= t2 <= t_hi:
                        cand_list.append(t2)
    cands = np.clip(np.array(cand_list, dtype=np.float64), t_lo, t_hi)
    cands = np.unique(np.round(cands, 10))
    best_t = 0.0
    best_v = np.inf
    for i in range(0, len(cands), 4000):
        ch = cands[i:i + 4000]
        if ch.size == 0:
            continue
        V = q0[None, :] + q1[None, :] * ch[:, None] + q2[None, :] * (ch[:, None] ** 2)
        mv = V.max(axis=1)
        bi = int(np.argmin(mv))
        if mv[bi] < best_v:
            best_v = mv[bi]
            best_t = ch[bi]
    return best_t

def repair(x, mass):
    x = np.clip(x, 0.0, 1.0)
    r = mass - x.sum()
    if r > 1e-11:
        cap = np.maximum(0.0, 1.0 - x)
        tc = cap.sum()
        if tc > 1e-15:
            x = x + r * cap / tc
    elif r < -1e-11:
        aff = np.maximum(0.0, x)
        ta = aff.sum()
        if ta > 1e-15:
            x = x + r * aff / ta
    return np.clip(x, 0.0, 1.0)

def project(x, mass):
    x = np.asarray(x, dtype=np.float64).copy()
    x = np.clip(x, 0.0, 1.0)
    if abs(x.sum() - mass) < 1e-10:
        return repair(x, mass)
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > mass:
            lo = mid
        else:
            hi = mid
    res = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    return repair(res, mass)

def gen_direction(rng, n, idx, pool, cur_h):
    r = rng.random()
    if r < 0.30:
        u = rng.standard_normal(n)
    elif r < 0.50:
        k = int(rng.integers(1, max(2, n // 4)))
        ph = rng.uniform(0, 2 * np.pi)
        if rng.random() < 0.5:
            u = np.cos(2 * np.pi * k * idx / n + ph)
        else:
            u = np.sin(2 * np.pi * k * idx / n + ph)
    elif r < 0.70:
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            j = (i + 1) % n
        u = np.zeros(n)
        u[i] = 1.0
        u[j] = -1.0
    elif r < 0.85:
        k = int(rng.integers(2, min(8, n // 2)))
        p = rng.permutation(n)
        u = np.zeros(n)
        u[p[:k]] = 1.0
        u[p[k:2 * k]] = -1.0
    elif r < 0.95 and len(pool) > 1:
        a = int(rng.integers(0, len(pool)))
        b = int(rng.integers(0, len(pool)))
        u = pool[a].copy() - pool[b].copy() if a != b else pool[0].copy() - cur_h
    else:
        u = rng.standard_normal(n) + 0.3 * np.cos(4 * np.pi * idx / n)
    u -= u.mean()
    return u

def try_direction(cur_h, u, n, mass):
    nu = np.linalg.norm(u)
    if nu < 1e-13:
        return None
    u = u / nu
    t_lo, t_hi = feasible_interval(cur_h, u)
    if t_hi - t_lo < 1e-12:
        return None
    bt = solve_on_ray(cur_h, u, t_lo, t_hi)
    nh = repair(cur_h + bt * u, mass)
    return nh

def optimize_h(h, n, mass, rng, deadline):

    cur_c = best_c

            if nc < best_c - 1e-10:
