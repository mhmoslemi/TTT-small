    """
    lines: list of (D_k, result_old_k) for each k
    Returns the minimum of max_k (result_old_k + delta * D_k) over delta in [delta_lo, delta_hi]
    """
    # Sort by slope D_k
    # The upper envelope of lines is a convex function
    # We need to find where the maximum changes
    
    # For a set of lines y = m_i * x + b_i, the upper envelope
    # can be found by sorting by slope and maintaining a convex chain.
    
    # Actually, for our purpose, we just need to evaluate the max at a few points.
    # The minimum of the max must occur at a breakpoint (where two lines cross) or at a boundary.
    
    # Let's find all breakpoints on the upper envelope.
    # Sort by slope
    lines_sorted = sorted(lines, key=lambda x: x[0])
    
    # Build upper envelope using standard algorithm
    # Each line is (m, b) where y = m*x + b
    envelope = []  # list of (m, b, x_start) where this line is on top from x_start onward
    
    for m, b in lines_sorted:
        if not envelope:
            envelope.append([m, b, -np.inf])
            continue
        # Check if new line is above the last line in envelope
        # Intersection: m*x + b = m_last*x + b_last => x = (b_last - b)/(m - m_last)
        last_m, last_b, _ = envelope[-1]
        if m == last_m:
            if b > last_b:
                envelope[-1] = [m, b, envelope[-1][2]]
            continue
        x_inter = (last_b - b) / (m - last_m)
        # The new line is above for x > x_inter (since m > last_m)
        while len(envelope) > 1 and x_inter <= envelope[-1][2]:
            # Remove last, recompute
            last_m, last_b, _ = envelope[-2]
            x_inter = (last_b - b) / (m - last_m)
            envelope.pop()
        envelope.append([m, b, x_inter])
    
    # Now evaluate at breakpoints within [delta_lo, delta_hi]
    candidates = [delta_lo, delta_hi]
    for m, b, x_start in envelope:
        if delta_lo <= x_start <= delta_hi:
            candidates.append(x_start)
    
    best_val = np.inf
    best_delta = delta_lo
    for x in candidates:
        # Evaluate max at x

        # But the envelope IS the max, so evaluating envelope is sufficient

        val = max(m_line * x + b_line for m_line, b_line in lines)

        b = corr[idx]

    dx = 2.0 / n

    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def compute_D(h, p, q):
    n = len(h)
    m = np.arange(-(n-1), n)
    
    # f(i, m) = (1-h[i+m])*I(0<=i+m<n) - h[i-m]*I(0<=i-m<n)
    def f(i):
        result = np.zeros(2*n-1)
        # (1-h[i+m]) valid when 0 <= i+m < n, i.e., -i <= m < n-i
        mask1 = (m >= -i) & (m < n - i)
        if np.any(mask1):
            result[mask1] += (1.0 - h[i + m[mask1]])
        # h[i-m] valid when 0 <= i-m < n, i.e., i-n < m <= i
        mask2 = (m > i - n) & (m <= i)
        if np.any(mask2):
            result[mask2] -= h[i - m[mask2]]
        return result
    
    return f(p) - f(q)

def find_optimal_tau(corr, D, lo, hi):
    """Find tau in [lo,hi] minimizing max(corr + tau*D)"""
    if hi - lo < 1e-15:
        return lo, np.max(corr + lo * D)
    
    # Find upper envelope vertices
    order = np.argsort(D)
    
    # Build upper envelope
    # Lines: y = D[j]*x + corr[j]
    # Stack of (m, b, x_start)
    env_m = []
    env_b = []
    env_x = []
    
    for idx in order:
        m_val = D[idx]
        b_val = corr[idx]
        
        if not env_m:
            env_m.append(m_val)
            env_b.append(b_val)
            env_x.append(-np.inf)
            continue
        
        if m_val == env_m[-1]:
            if b_val > env_b[-1]:
                env_b[-1] = b_val
            continue
        
        # Intersection with last: m_val*x + b_val = env_m[-1]*x + env_b[-1]
        # x = (env_b[-1] - b_val) / (m_val - env_m[-1])
        x_int = (env_b[-1] - b_val) / (m_val - env_m[-1])
        
        while len(env_m) > 1 and x_int <= env_x[-1]:
            env_m.pop()
            env_b.pop()
            env_x.pop()
            x_int = (env_b[-1] - b_val) / (m_val - env_m[-1])
        
        env_m.append(m_val)
        env_b.append(b_val)
        env_x.append(x_int)
    
    # Candidates: lo, hi, and all env_x in [lo, hi]
    candidates = [lo, hi]
    for x in env_x:
        if lo <= x <= hi:
            candidates.append(x)

    best_val = np.max(corr + lo * D)
    for tau in candidates:
        val = np.max(corr + tau * D)
        if val < best_val:
            best_val = val
            best_tau = tau
    
    return best_tau, best_val

def project_balanced(a, target):
    """Project a to [0,1]^n with sum = target."""
    n = len(a)
    # Bisection on lambda: x = clip(a + lambda, 0, 1), sum(x) = target
    lo = -np.max(a) - 1.0
    hi = 1.0 - np.min(a) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        x = np.clip(a + mid, 0.0, 1.0)
        s = x.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    x = np.clip(a + (lo + hi) / 2.0, 0.0, 1.0)
    
    # Fix residual
    residual = target - x.sum()
    if abs(residual) > 1e-12:
        if residual > 0:
            slack = 1.0 - x
            idx = np.argsort(-slack)
            r = residual
            for i in idx:
                if r <= 1e-13:
                    break
                add = min(r, slack[i])
                x[i] += add
                r -= add

            idx = np.argsort(-x)
            for i in idx:
                if r <= 1e-13:
                    break
                sub = min(r, x[i])
                x[i] -= sub
                r -= sub
    return x

    """Create a random binary-like initial vector."""

    mask2 = (m > p - n) & (m <= p)
    if np.any(mask2):
        fp[mask2] -= h[p - m[mask2]]
    
    # For q:
    fq = np.zeros(2*n-1)
    mask3 = (m >= -q) & (m < n - q)
    if np.any(mask3):
        fq[mask3] += (1.0 - h[q + m[mask3]])
    mask4 = (m > q - n) & (m <= q)
    if np.any(mask4):
        fq[mask4] -= h[q - m[mask4]]
    
    return fp - fq
