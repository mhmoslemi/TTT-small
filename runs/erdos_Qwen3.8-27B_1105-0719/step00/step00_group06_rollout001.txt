def _golden_inner(h, up_idx, down_idx, a, b, max_evals, tol):
    """Inner golden section: minimize f(t) for t in [a, b], h[up_idx]+=t, h[down_idx]-=t."""
    if a >= b:
        return None
    
    phi = (1.0 - np.sqrt(5.0)) / 2.0  # ~ -0.618, so golden ratio conjugate
    gr = (np.sqrt(5.0) - 1.0) / 2.0  # ~ 0.618

    x2 = a + gr * (b - a)
    f1 = pair_raw_value(h, up_idx, down_idx, x1)
    f2 = pair_raw_value(h, up_idx, down_idx, x2)

            b = x2
            x2 = x1
            f2 = f1
            x1 = b - gr * (b - a)
            f1 = pair_raw_value(h, up_idx, down_idx, x1)
        else:
            a = x1
            x1 = x2
            f1 = f2
            x2 = a + gr * (b - a)
            f2 = pair_raw_value(h, up_idx, down_idx, x2)

            best_t = x2

def compute_gradients_active(h, active_lags, n):
    """Compute gradients of active overlap terms w.r.t. h."""
    # G has shape (K, n) where K = number of active lags
    G = np.zeros((len(active_lags), n))
    for k, s in enumerate(active_lags):
        g = np.zeros(n)
        if s == 0:
            g[:] = 1.0 - 2.0 * h
        elif s > 0:
            g[s:] += 1.0 - h[:n-s]
            g[:n-s] -= h[s:]
        else:
            a = -s
            g[:n-a] += 1.0 - h[a:]
            g[a:] -= h[:n-a]
        G[k] = g
    return G

def select_pairs(h, n, rng, max_pairs, active_frac=0.02):
    """Select candidate pairs for exchange based on active lag gradients."""
    prof = np.correlate(h, 1.0 - h, mode="full")
    fmax = np.max(prof)
    
    # Active lags: those within active_frac of max
    active_mask = prof >= fmax - active_frac * fmax
    active_indices = np.where(active_mask)[0]
    
    # Map indices to lags: index k corresponds to lag k - (n-1)
    active_lags = active_indices - (n - 1)
    
    # Compute gradients
    G = compute_gradients_active(h, active_lags, n)
    
    # For each pair (i, j), compute directional score
    # d_i = max_k G[k, i] - G[k, j] (direction: increase i, decrease j)
    # d_j = max_k G[k, j] - G[k, i] (direction: increase j, decrease i)
    # Score = min(d_i, d_j) for best direction
    pairs = []
    for i in range(n):
        for j in range(i + 1, n):
            di = np.max(G[:, i] - G[:, j])
            dj = np.max(G[:, j] - G[:, i])
            # We want to decrease the max, so we want negative directional derivatives
            # If we increase i and decrease j: derivative of c_s w.r.t. t is G[s,i] - G[s,j]
            # We want this to be negative (decreasing c_s) for active s
            # Score is how much we can decrease the max

        if i != j:
            pairs.append((0.0, i, j))

    return pairs[:max_pairs]

def transfer_resolution(h_src, n_dst):

    temp[i] += t
    temp[j] -= t
    np.clip(temp, 0.0, 1.0, out=temp)
    prof = np.correlate(temp, 1.0 - temp, mode="full")
    return float(np.max(prof))


def golden_pair_min(h, i, j, max_evals=40):
    bounds = pair_bounds(h, i, j)
    if bounds is None:
        return None
    a, b = bounds
    if b - a < 1e-12:
        return 0.0
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    x1 = b - gr * (b - a)
    x2 = a + gr * (b - a)
    f1 = pair_raw_eval(h, i, j, x1)
    f2 = pair_raw_eval(h, i, j, x2)
    f0 = pair_raw_eval(h, i, j, 0.0)
    best_t = 0.0
    best_f = f0
    n_evals = 3
    
    for _ in range(max_evals - 3):
        if f1 < f2:
            b = x2
            x2 = x1
            f2 = f1
            x1 = b - gr * (b - a)
            f1 = pair_raw_eval(h, i, j, x1)
        else:
            a = x1
            x1 = x2
            f1 = f2
            x2 = a + gr * (b - a)
            f2 = pair_raw_eval(h, i, j, x2)
        n_evals += 1
        if f1 < best_f:
            best_f = f1
            best_t = x1
        if f2 < best_f:
            best_f = f2
            best_t = x2
        if abs(b - a) < 1e-11:
            break
    return best_t


def compute_active_gradients(h, n):
    prof = np.correlate(h, 1.0 - h, mode="full")
    fmax = np.max(prof)
    if fmax <= 0:
        return None, None
    active_frac = 0.02
    active_mask = prof >= fmax - active_frac * fmax
    active_indices = np.where(active_mask)[0]
    if len(active_indices) == 0:
        active_indices = np.array([n - 1])
    active_lags = active_indices - (n - 1)
    G = np.zeros((len(active_lags), n))
    for k, s in enumerate(active_lags):
        g = np.zeros(n)
        if s == 0:
            g[:] = 1.0 - 2.0 * h
        elif s > 0:
            g[s:] += 1.0 - h[:n - s]
            g[:n - s] -= h[s:]
        else:
            a = -s
            g[:n - a] += 1.0 - h[a:]
            g[a:] -= h[:n - a]
        G[k] = g
    return G, active_indices


def select_pairs(h, n, rng, max_pairs):
    G, _ = compute_active_gradients(h, n)
    pairs = []
    if G is not None:
        for i in range(n):
            for j in range(i + 1, n):
                di = np.max(G[:, i] - G[:, j])
                dj = np.max(G[:, j] - G[:, i])
                score = min(-di, -dj)
                if score > 0:
                    if dj < di:
                        pairs.append((score, i, j))
                    else:
                        pairs.append((score, j, i))
    n_rand = max(4, max_pairs // 3)
    for _ in range(n_rand):
        i, j = rng.integers(0, n, size=2)
        if i != j:
            pairs.append((0.0, int(i), int(j)))
    pairs.sort(key=lambda x: -x[0])
    if len(pairs) > max_pairs:
        pairs = pairs[:max_pairs]
    return pairs


def build_simple_vector(n, pattern="alternating", rng=None):
    if rng is None:
        rng = np.random.default_rng(0)
    if pattern == "alternating":
        h = np.zeros(n)
        h[0::2] = 1.0
    elif pattern == "uniform":
        h = np.full(n, 0.5)
    elif pattern == "block":
        h = np.zeros(n)
        h[:n // 2] = 1.0
    elif pattern == "random":
        h = rng.uniform(0, 1, n)
    else:
        h = np.full(n, 0.5)
    return repair_feasible(h, n)


def transfer_resolution(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.astype(np.float64).copy()
    x_src = np.linspace(0.0, 1.0, n_src)
    x_dst = np.linspace(0.0, 1.0, n_dst)
    h_dst = np.interp(x_dst, x_src, h_src)
    return repair_feasible(h_dst, n_dst)


def run_pair_descent(h, n, rng, max_iter, deadline, max_evals=35):
    cur_h = h.copy()
    cur_score = erdos_objective(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score
    stagnation = 0
    
    for _ in range(max_iter):
        if time.time() > deadline - 3:
            break
        remaining = deadline - time.time()
        if remaining < 2:
            break
        
        if remaining > 100:
            n_pairs = min(30, n // 2)
        elif remaining > 30:
            n_pairs = min(20, n // 3)
        else:
            n_pairs = min(10, n // 4)
        
        pairs = select_pairs(cur_h, n, rng, n_pairs)
        improved = False
        cur_raw = float(np.max(np.correlate(cur_h, 1.0 - cur_h, mode="full")))
        
        for (score, pi, pj) in pairs:
            if time.time() > deadline - 2:
                break
            if score <= 0 and len(pairs) > 8 and rng.random() > 0.15:
                continue
            t_opt = golden_pair_min(cur_h, pi, pj, max_evals=max_evals)
            if t_opt is not None and abs(t_opt) > 1e-10:
                cur_h[pi] += t_opt
                cur_h[pj] -= t_opt
                cur_h = np.clip(cur_h, 0.0, 1.0)
                cur_h = repair_feasible(cur_h, n)
                new_score = erdos_objective(cur_h)
                if new_score < cur_score - 1e-12:
                    cur_score = new_score
                    cur_raw = float(np.max(np.correlate(cur_h, 1.0 - cur_h, mode="full")))
                    improved = True
                    if new_score < best_score - 1e-12:
                        best_score = new_score
                        best_h = cur_h.copy()
                    stagnation = 0
                else:
                    cur_h[pi] -= t_opt
                    cur_h[pj] += t_opt
                    cur_h = np.clip(cur_h, 0.0, 1.0)
                    cur_h = repair_feasible(cur_h, n)
        
        if not improved:
            stagnation += 1
        else:
            stagnation = 0
        
        if stagnation >= 4:
            stagnation = 0
            h_pert = cur_h.copy()
            n_moves = rng.integers(3, 7)
            for _ in range(n_moves):
                i, j = rng.integers(0, n, size=2)
                if i != j:
                    delta = rng.uniform(-0.4, 0.4)
                    h_pert[i] += delta
                    h_pert[j] -= delta
            h_pert = repair_feasible(h_pert, n)
            ps = erdos_objective(h_pert)
            if ps <= cur_score + 0.02:
                cur_h = h_pert
                cur_score = ps
                cur_raw = float(np.max(np.correlate(cur_h, 1.0 - cur_h, mode="full")))
    
    return best_h, best_score


def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    deadline = start_time + float(budget_s)
    rng = np.random.default_rng(seed)
    
    # Get initial values
    try:
        parent = np.asarray(initial_h_values, dtype=np.float64).flatten()
        if len(parent) == 0:
            parent = None
    except (NameError, TypeError, AttributeError):
        parent = None
    
    if parent is not None:
        n0 = len(parent)
        h0 = repair_feasible(parent, n0)
    else:
        n0 = 70
        h0 = build_simple_vector(n0, "alternating")
    
    # Track best globally using a mutable container
    best = {'h': h0.copy(), 'score': erdos_objective(h0), 'n': n0}
    
    def try_store(h_cand):
        n_c = len(h_cand)
        h_cand = repair_feasible(h_cand, n_c)
        score = erdos_objective(h_cand)
        if not np.isfinite(score):
            return False
        if score < best['score'] - 1e-12:
            best['h'] = h_cand.copy()
            best['score'] = score
            best['n'] = n_c
            return True
        elif abs(score - best['score']) < 1e-10 and n_c < best['n']:
            best['h'] = h_cand.copy()
            best['score'] = score
            best['n'] = n_c
            return True
        return False
    
    # Try initial patterns
    for pat in ["alternating", "uniform", "block"]:
        h_try = build_simple_vector(n0, pat)
        try_store(h_try)
    
    # Also try a random start
    h_rand = build_simple_vector(n0, "random", rng)
    try_store(h_rand)
    
    # Main pair descent on best resolution
    cur_h = best['h'][:best['n']].copy() if best['n'] == n0 else transfer_resolution(best['h'], n0)
    cur_h = cur_h.copy()
    cur_n = n0
    cur_score = erdos_objective(cur_h)
    
    remaining = deadline - time.time()
    if remaining > 20:
        r_h, r_score = run_pair_descent(cur_h, cur_n, rng, 200, deadline, max_evals=40)
        if r_score < best['score'] - 1e-12:
            try_store(r_h)
    
    # Try other resolutions
    remaining = deadline - time.time()
    if remaining > 40:
        for n_try in [n0 // 2, n0 + 20, n0 // 3]:
            if n_try < 6 or time.time() > deadline - 15:
                continue
            h_try = transfer_resolution(best['h'], n_try)
            try_store(h_try)
            # Quick descent on this resolution
            rem2 = deadline - time.time()
            if rem2 > 15:
                r_h2, r_s2 = run_pair_descent(h_try, n_try, rng, 80, deadline, max_evals=30)
                if r_s2 < best['score'] - 1e-12:
                    try_store(r_h2)
    
    # Try more diverse starts
    remaining = deadline - time.time()
    if remaining > 30:
        for seed_offset in [1, 7, 13, 42]:
            if time.time() > deadline - 10:
                break
            rng2 = np.random.default_rng(seed + seed_offset)
            h_div = build_simple_vector(n0, "random", rng2)
            # Light descent
            rem3 = deadline - time.time()
            if rem3 > 10:
                r_h3, r_s3 = run_pair_descent(h_div, n0, rng2, 50, deadline, max_evals=30)
                try_store(r_h3)
    
    # Final polish on best
    remaining = deadline - time.time()
    if remaining > 10 and best['n'] >= 10:
        h_final = best['h'].copy()
        r_hf, r_sf = run_pair_descent(h_final, best['n'], rng, 100, deadline, max_evals=50)
        try_store(r_hf)
    
    # Final validation
    best_h = repair_feasible(best['h'], best['n'])
    final_score = erdos_objective(best_h)
    
    # Ensure feasibility
    assert np.all(best_h >= -1e-12) and np.all(best_h <= 1.0 + 1e-12)
    assert abs(np.sum(best_h) - best['n'] / 2.0) < 1e-6
    
    return best_h, final_score, best['n']
