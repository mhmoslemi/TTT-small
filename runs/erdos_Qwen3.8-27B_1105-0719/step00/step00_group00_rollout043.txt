    v = np.asarray(v, dtype=float).copy()
    n = v.size
    if n == 0:
        return v
    target = float(target)
    if target < 0:
        target = 0.0
    if target > n:
        target = float(n)
    v = np.clip(v, 0.0, 1.0)
    s = v.sum()
    if abs(s - target) < 1e-12:
        return v
    lo = -1.5
    hi = 1.5
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        cur = np.sum(np.clip(v - mid, 0.0, 1.0))
        if cur > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - 0.5*(lo+hi), 0.0, 1.0)
    diff = target - x.sum()
    if diff > 0:
        idx = np.argsort(1.0 - x)[::-1]
        for i in idx:
            if diff <= 1e-14: break
            add = min(1.0 - x[i], diff)
            if add > 0:
                x[i] += add; diff -= add
    elif diff < 0:
        idx = np.argsort(x)
        for i in idx:
            if -diff <= 1e-14: break
            sub = min(x[i], -diff)
            if sub > 0:
                x[i] -= sub; diff += sub
    # final exact correction
    diff = target - x.sum()
    if abs(diff) > 0.0:
        if diff > 0:
            i = int(np.argmax(1.0 - x))
        else:
            i = int(np.argmax(x))
        x[i] = float(np.clip(x[i] + diff, 0.0, 1.0))
    return x

def project_sum_box(v, target):
    v = np.asarray(v, dtype=float).copy()
    n = v.size
    if n == 0:
        return v

    target = float(target)
    if target < 0.0:
        target = 0.0
    if target > n:
        target = float(n)

    v = np.clip(v, 0.0, 1.0)
    s = float(v.sum())
    if abs(s - target) < 1e-12:
        return v

    lo = -1.5
    hi = 1.5

        mid = 0.5 * (lo + hi)

        if cur > target:
            lo = mid
        else:
            hi = mid

    x = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - float(x.sum())

    if diff > 0.0:
        idx = np.argsort(1.0 - x)[::-1]

                diff += sub

def objective_from_lags(h):
    n = len(h)
    if n == 0:
        return 0.0
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def grad_lag(h, n, s):
    g = np.zeros(n, dtype=float)
    start = max(0, -s)
    end = min(n, n - s)
    if start >= end:
        return g

    i = np.arange(start, end)
    j = i + s
    g[i] += 1.0 - h[j]
    g[j] -= h[i]
    return g


def soft_grad_wrt_h(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    f = corr * dx
    fmax = float(np.max(f))

    if beta <= 1e-15:
        active = np.where(f >= fmax - 1e-12)[0]
        if active.size == 0:
            active = np.array([int(np.argmax(f))])
        G = np.zeros(n, dtype=float)
        for idx in active:
            s = int((n - 1) - int(idx))
            G += grad_lag(h, n, s)
        return fmax, (G * dx) / max(1, active.size)

    z = (f - fmax) / beta
    z = np.maximum(z, -700.0)
    weights = np.exp(z)
    S = float(weights.sum())
    if S <= 0.0:
        S = 1.0
    probs = weights / S
    fval = fmax - beta * float(np.log(S))

    G = np.zeros(n, dtype=float)

        G += probs[idx] * grad_lag(h, n, s)

                    h_tmp[j] = 0.0
                elif h_tmp[j] > 1.0:
                    h_tmp[j] = 1.0

                f = objective_from_lags(h_tmp)
                eval_count += 1
                if f < cur:
                    cur = f
                    best_d = d

                if eval_count >= max_evals or time_remaining(deadline) < 1.0:
                    break

            if best_d is not None:
                h[i] = float(np.clip(a - best_d, 0.0, 1.0))
                h[j] = float(np.clip(b + best_d, 0.0, 1.0))
                best_f = cur

    h = project_sum_box(h, target)
    return h, objective_from_lags(h)


def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    deadline = start_time + float(budget_s)
    rng = np.random.default_rng(seed)

    init = None
    if "initial_h_values" in globals():
        iv = globals()["initial_h_values"]
        if iv is not None:
            init = np.asarray(iv, dtype=float)

    if init is None:
        n0 = 45
        angles = 2.0 * np.pi * np.arange(n0) / float(n0)
        init = 0.5 + 0.35 * np.sin(angles) + 0.12 * np.cos(2.0 * angles)

    init = project_sum_box(init, len(init) / 2.0)

    best_h = init.copy()
    best_val = objective_from_lags(best_h)

    if time_remaining(deadline) > 5.0:
        opt_h, val = optimize_h(init, deadline, rng)
        if val < best_val:
            best_h = opt_h
            best_val = val

    n0 = len(init)
    for sig in (0.005, 0.03):
        if time_remaining(deadline) < 8.0:
            break
        h_c = project_sum_box(init + rng.normal(0.0, sig, size=n0), n0 / 2.0)
        opt_h, val = optimize_h(h_c, deadline, rng)
        if val < best_val:
            best_h = opt_h
            best_val = val

    if time_remaining(deadline) > 50.0 and n0 < 90:
        rb = random_binary_h(n0, rng)
        opt_h, val = optimize_h(
            rb,
            deadline,
            rng,
            betas=[0.03, 1e-3, 1e-5, 1e-8],
            iters=[25, 35, 45, 60],
        )
        if val < best_val:
            best_h = opt_h
            best_val = val

    base = best_h
    for target_n in (90, 180, 270):
        if time_remaining(deadline) < 40.0:
            break
        if len(base) >= target_n:
            continue
        if target_n > 320:
            break

        h_c = upscale_vector(base, target_n)
        h_c = project_sum_box(h_c, target_n / 2.0)
        opt_h, val = optimize_h(h_c, deadline, rng)

        if val < best_val:
            best_h = opt_h
            best_val = val

        base = opt_h if val < best_val + 1e-3 else best_h

    if time_remaining(deadline) > 15.0:
        h_p = project_sum_box(
            best_h + rng.normal(0.0, 8e-4, size=len(best_h)),
            len(best_h) / 2.0,
        )
        opt_h, val = optimize_h(
            h_p,
            deadline,
            rng,
            betas=[1e-5, 1e-8],
            iters=[40, 50],
        )
        if val < best_val:
            best_h = opt_h
            best_val = val

    if time_remaining(deadline) > 10.0 and len(best_h) <= 140:
        best_h, best_val = pair_line_search(
            best_h,
            len(best_h) / 2.0,
            len(best_h),
            rng,
            deadline=deadline,
            max_evals=50000,
        )

    if best_h is None:
        best_h = init

    best_h = project_sum_box(best_h, len(best_h) / 2.0)
    final_val = objective_from_lags(best_h)

    return best_h, float(final_val + 1e-12), len(best_h)
