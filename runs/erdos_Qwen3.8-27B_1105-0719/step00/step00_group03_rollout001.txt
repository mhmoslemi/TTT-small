import numpy as np
from scipy.optimize import minimize
import time

_sum_target = 0.0
_n_pts = 45

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def lse_value(x, alpha):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(corr)
    s = np.clip(alpha * (corr - m), -500, 500)
    return m + np.logaddexp.reduce(s) / alpha

def lse_gradient(x, alpha):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(corr)
    s = np.clip(alpha * (corr - m), -500, 500)
    w = np.exp(s) / np.sum(np.exp(s))
    grad = np.zeros(n)
    for r in range(2 * n - 1):
        if w[r] < 1e-12:
            continue
        k = r - (n - 1)
        j1s = max(0, k)
        j1e = min(n - 1, k + n - 1)
        if j1s <= j1e:
            grad[j1s:j1e + 1] += w[r] * dx * (1.0 - x[j1s - k:j1e + 1 - k])
        j2s = max(0, -k)
        j2e = min(n - 1, n - 1 - k)
        if j2s <= j2e:
            grad[j2s:j2e + 1] -= w[r] * dx * x[j2s + k:j2e + 1 + k]
    return grad

def project_feasible(x, target):

def pair_swap_search(h, max_rounds=50, budget_start=0, budget_limit=0):
