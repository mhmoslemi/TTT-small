    return grad

    target = n / 2

        if abs(deficit) < 1e-12:

        if deficit > 0:
            # Need to add deficit
            room = 1.0 - h  # available room
            mask = room > 1e-12

            if not mask.any():
                break
            add = deficit / mask.sum()
            h[mask] = np.minimum(h[mask] + add, 1.0)
        else:
            mask = h > 1e-14
            if not mask.any():
                break
            rem = (-deficit) / mask.sum()
            h[mask] = np.maximum(h[mask] - rem, 0.0)
    # Final exact fix on the most "central" entry
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        idx = np.argmin(np.abs(h - 0.5))
        h[idx] = np.clip(h[idx] + diff, 0, 1)

def compute_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

class EpigraphNLP:
    def __init__(self, n, beta=0.0):
        self.n = n
        self.dx = 2.0 / n
        self.beta = beta
        self.M = 2 * n - 1
    
    def objective(self, x):
        h = x[:self.n]
        t = x[self.n]
        if self.beta == 0:
            return t
        return t + self.beta * np.sum(h * (1.0 - h))
    
    def objective_jac(self, x):
        h = x[:self.n]
        n = self.n
        jac = np.zeros(n + 1)
        if self.beta != 0:
            jac[:n] = self.beta * (1.0 - 2.0 * h)
        jac[n] = 1.0
        return jac
    
    def ineq_fun(self, x):
        h = x[:self.n]
        t = x[self.n]
        prof = compute_profile(h)
        return t - prof
    
    def ineq_jac(self, x):
        h = x[:self.n]
        n = self.n
        dx = self.dx
        M = self.M
        J = np.zeros((M, n + 1))
        for m in range(M):
            delta = m - (n - 1)
            if delta >= 0:
                # First factor: grad[0:n-delta] += (1 - h[delta:n])
                J[m, 0:n-delta] += dx * (1.0 - h[delta:n])
                # Second factor: grad[delta:n] += (-h[0:n-delta])
                J[m, delta:n] += dx * (-h[0:n-delta])
            else:
                d = -delta
                # First factor: grad[d:n] += (1 - h[0:n-d])
                J[m, d:n] += dx * (1.0 - h[0:n-d])
                # Second factor: grad[0:n-d] += (-h[d:n])
                J[m, 0:n-d] += dx * (-h[d:n])
            J[m, n] = 1.0
        return J
    
    def eq_fun(self, x):
        h = x[:self.n]
        return np.sum(h) - self.n / 2.0
    
    def eq_jac(self, x):
        n = self.n
        jac = np.zeros(n + 1)
        jac[:n] = 1.0
        return jac

def solve_epigraph(h_start, n, t_start, beta=0.0, maxiter=200, ftol=1e-12):
    nlp = EpigraphNLP(n, beta)
    x0 = np.concatenate([h_start, [t_start]])
    bounds = [(0, 1)] * n + [(0, t_start + 0.5)]
    constraints = [
        {'type': 'eq', 'fun': nlp.eq_fun, 'jac': nlp.eq_jac},
        {'type': 'ineq', 'fun': nlp.ineq_fun, 'jac': nlp.ineq_jac}
    ]
    result = minimize(
        nlp.objective,
        x0,
        jac=nlp.objective_jac,
        bounds=bounds,
        constraints=constraints,
        method='SLSQP',
        options={'maxiter': maxiter, 'ftol': ftol}
    )
    return result

def active_gradient_descent(h, n, top_k=20, max_steps=50, tol=1e-10):
    """Descent direction from average gradient of top-k active lags"""
    best_h = h.copy()
    best_score = exact_score(h)
    dx = 2.0 / n
    M = 2 * n - 1
    
    for step in range(max_steps):
        prof = compute_profile(best_h)
        current_score = np.max(prof)
        if current_score < best_score - 1e-14:
            best_score = current_score
            h = best_h.copy()
        
        # Find top-k lags
        sorted_idx = np.argsort(-prof)
        top_idx = sorted_idx[:top_k]
        
        # Compute weighted average gradient
        weights = np.exp((prof[top_idx] - current_score) / (0.01))  # softmax-like
        weights = weights / weights.sum()
        
        avg_grad = np.zeros(n)
        for w, m in zip(weights, top_idx):
            delta = m - (n - 1)
            grad = np.zeros(n)
            if delta >= 0:
                grad[0:n-delta] += dx * (1.0 - best_h[delta:n])
                grad[delta:n] += dx * (-best_h[0:n-delta])
            else:
                d = -delta
                grad[d:n] += dx * (1.0 - best_h[0:n-d])
                grad[0:n-d] += dx * (-best_h[d:n])
            avg_grad += w * grad
        
        # Project to zero-sum direction
        avg_grad -= np.mean(avg_grad)
        
        # Normalize
        norm = np.linalg.norm(avg_grad)
        if norm < 1e-15:
            break
        avg_grad /= norm
        
        # Line search
        improved = False
        for alpha in [0.05, 0.03, 0.02, 0.01, 0.005, 0.002, 0.001]:
            candidate = best_h + alpha * avg_grad
            candidate = repair_feasible(candidate, n)
            score = exact_score(candidate)
            if score < best_score - 1e-12:
                best_score = score
                best_h = candidate
                improved = True
                break
        
        if not improved:
            # Try negative direction
            best_h = best_h + 0  # stay
            for alpha in [0.05, 0.03, 0.02, 0.01, 0.005]:
                candidate = best_h - alpha * avg_grad
                candidate = repair_feasible(candidate, n)
                score = exact_score(candidate)
                if score < best_score - 1e-12:
                    best_score = score
                    best_h = candidate
                    improved = True
                    break
        
        if not improved:
            break
    
    return best_h, best_score

def interpolate_to_dim(h, new_n):
    """Interpolate a step function to a different number of samples"""
    n = len(h)
    if n == new_n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) * 2.0 / n
    x_new = (np.arange(new_n) + 0.5) * 2.0 / new_n
    # Extend with edge values for interpolation
    x_ext = np.concatenate([[0], x_old, [2]])
    h_ext = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(x_new, x_ext, h_ext)
    h_new = repair_feasible(h_new, new_n)
    return h_new

def coordinate_descent(h, n, max_rounds=20, step_sizes=None):
    if step_sizes is None:
        step_sizes = [0.03, 0.01, 0.005, 0.002, 0.001, 0.0005]
    best_h = h.copy()
    best_score = exact_score(h)
    
    for rnd in range(max_rounds):
        improved = False
        for i in range(n):
            for delta in step_sizes:
                for sign in [1, -1]:
                    new_val = best_h[i] + sign * delta
                    if new_val < 0 or new_val > 1:
                        continue
                    # Check if redistribution would cause issues
                    redistribute = -sign * delta / (n - 1)
                    if np.any(best_h[np.arange(n) != i] + redistribute < -1e-14) or \
                       np.any(best_h[np.arange(n) != i] + redistribute > 1 + 1e-14):
                        continue
                    candidate = best_h.copy()
                    candidate[i] = new_val
                    others = np.arange(n) != i
                    candidate[others] += redistribute
                    score = exact_score(candidate)
                    if score < best_score - 1e-13:
                        best_score = score
                        best_h = candidate
                        improved = True
        if not improved:
            break
    
    return best_h, best_score

    # log-sum-exp for numerical stability
    max_prof = np.max(prof)

def smooth_max_gradient(h, k=100):
    prof = compute_profile(h)
    max_prof = np.max(prof)
    weights = np.exp(k * (prof - max_prof))
    weights /= np.sum(weights)
    # Gradient is weighted average of individual gradients
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2*n - 1):
        delta = m - (n-1)
        if delta >= 0:
            grad[0:n-delta] += weights[m] * dx * (1.0 - h[delta:n])
            grad[delta:n] += weights[m] * dx * (-h[0:n-delta])
        else:
            d = -delta
            grad[d:n] += weights[m] * dx * (1.0 - h[0:n-d])
            grad[0:n-d] += weights[m] * dx * (-h[d:n])
    return grad
