def project_feasible(v, target_sum):
    """Project v onto {x: 0<=x<=1, sum(x)=target_sum}"""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    # Bisection on lambda: find lambda such that sum(clamp(v+lambda, 0, 1)) = target_sum
    lo, hi = -1.0, 1.0  # initial bounds, will expand
    # Actually, v_i + lambda ranges from min(v)+lambda to max(v)+lambda
    # We need sum of clamped values = target_sum
    # g(lambda) = sum(clamp(v_i + lambda, 0, 1)) - target_sum
    # g is non-decreasing in lambda
    def g(lam):
        return np.sum(np.clip(v + lam, 0.0, 1.0)) - target_sum
    
    # Find valid bounds
    while g(lo) > 0:
        lo *= 2.0
        if lo < -1000: break
    while g(hi) < 0:
        hi *= 2.0
        if hi > 1000: break

        if g(mid) > 0:
            hi = mid
        else:
            lo = mid
    
    lam = (lo + hi) / 2.0
    result = np.clip(v + lam, 0.0, 1.0)
    
    # Fix residual
    residual = target_sum - result.sum()
    if abs(residual) > 1e-14:
        # Find indices where we can adjust

            # Need to increase some values

            idx = candidates[np.argmax(candidates)]  # or random
            result[idx] += residual
            result[idx] = np.clip(result[idx], 0.0, 1.0)

    s = h[p] + h[q]
    lo = max(0.0, s - 1.0)

            best_a = a

def slsqp_smoothed(h0, target_sum, T=50.0, maxiter=40):
    """SLSQP with log-sum-exp smoothed max."""
    n = len(h0)

    def objective(x):
        c = np.correlate(x, 1.0 - x, mode="full") * dx
        # Log-sum-exp for smooth max
        c_max = np.max(c)
        if T * c_max < 700:  # avoid overflow
            return (1.0 / T) * np.log(np.sum(np.exp(T * c))) + 0.1 * (np.sum(x) - target_sum)**2
        else:
            return c_max + 0.1 * (np.sum(x) - target_sum)**2
    
    def constraint_eq(x):
        return np.sum(x) - target_sum
    
    constraints = [{'type': 'eq', 'fun': constraint_eq}]
    bounds = [(0.0, 1.0)] * n
    
    try:
        res = minimize(objective, h0, method='SLSQP', bounds=bounds,

            if np.max(x_opt) <= 1.0 + 1e-10 and np.min(x_opt) >= -1e-10:

def feasible_intervals(A, B, C, T, lo, hi):
    """Return list of (l, r) intervals where all constraints are satisfied at threshold T."""
    intervals = [(lo, hi)]
    eps = 1e-14
    
    for k in range(len(A)):
        ak, bk = A[k], B[k]
        d = C[k] - T
        new_intervals = []
        
        if abs(ak) < eps:
            if abs(bk) < eps:
                if d > 0:
                    return []
                # No constraint
                continue
            elif bk > 0:
                bound = -d / bk
                # a <= bound
                for (l, r) in intervals:
                    if l <= bound:
                        new_intervals.append((l, min(r, bound)))
            else:
                bound = -d / bk
                # a >= bound
                for (l, r) in intervals:
                    if r >= bound:
                        new_intervals.append((max(l, bound), r))
        elif ak > 0:
            disc = bk*bk - 4.0*ak*d
            if disc < 0:
                return []
            sd = math.sqrt(disc)
            r1 = (-bk - sd) / (2.0*ak)
            r2 = (-bk + sd) / (2.0*ak)
            # a in [r1, r2]
            for (l, r) in intervals:
                nl = max(l, r1)
                nr = min(r, r2)
                if nl <= nr:
                    new_intervals.append((nl, nr))
        else:
            disc = bk*bk - 4.0*ak*d
            if disc < 0:
                # Always negative, no constraint
                continue

            r1 = (-bk - sd) / (2.0*ak)
            r2 = (-bk + sd) / (2.0*ak)

            lo_r, hi_r = min(r1, r2), max(r1, r2)

            for (l, r) in intervals:

                    new_intervals.append((l, min(r, lo_r)))
                if r >= hi_r:
                    new_intervals.append((max(l, hi_r), r))
        
        intervals = new_intervals
        if not intervals:

import numpy as np
import time
import math

_DEADLINE = 0.0
_best_score = float('inf')

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def project_feasible(v, target_sum):
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    lo, hi = -2.0, 2.0
    for _ in range(40):
        if np.sum(np.clip(v + lo, 0.0, 1.0)) > target_sum: lo *= 2.0
        else: break
    for _ in range(40):
        if np.sum(np.clip(v + hi, 0.0, 1.0)) < target_sum: hi *= 2.0
        else: break
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.sum(np.clip(v + mid, 0.0, 1.0)) > target_sum: hi = mid
        else: lo = mid
    result = np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)
    residual = target_sum - result.sum()
    if abs(residual) > 1e-10:
        if residual > 0:
            idx = np.where(result < 1.0 - 1e-8)[0]
        else:
            idx = np.where(result > 1e-8)[0]
        if len(idx) > 0:
            result[idx[0]] = float(np.clip(result[idx[0]] + residual, 0.0, 1.0))
    return result

    return time.perf_counter() < _DEADLINE

def feasible_intervals(A, B, C, T, lo, hi):
    intervals = [(lo, hi)]
    eps = 1e-14
    for k in range(len(A)):
        ak, bk, ck = A[k], B[k], C[k]
        d = ck - T
        new_iv = []
        if abs(ak) < eps:
            if abs(bk) < eps:
                if d > 0:
                    return []
                continue
            bound = -d / bk
            for (l, r) in intervals:
                if bk > 0:
                    if l <= bound:
                        new_iv.append((l, min(r, bound)))
                else:
                    if r >= bound:
                        new_iv.append((max(l, bound), r))
        elif ak > 0:
            disc = bk*bk - 4.0*ak*d
            if disc < -1e-15:
                return []
            if disc < 0: disc = 0.0
            sd = math.sqrt(disc)
            r1 = (-bk - sd) / (2.0*ak)
            r2 = (-bk + sd) / (2.0*ak)
            for (l, r) in intervals:
                nl = max(l, r1)
                nr = min(r, r2)
                if nl <= nr:
                    new_iv.append((nl, nr))
        else:
            disc = bk*bk - 4.0*ak*d
            if disc < -1e-15:
                continue
            if disc < 0: disc = 0.0
            sd = math.sqrt(disc)
            r1 = (-bk - sd) / (2.0*ak)
            r2 = (-bk + sd) / (2.0*ak)
            lo_r, hi_r = min(r1, r2), max(r1, r2)
            for (l, r) in intervals:
                if l <= lo_r:
                    new_iv.append((l, min(r, lo_r)))
                if r >= hi_r:
                    new_iv.append((max(l, hi_r), r))
        intervals = new_iv
        if not intervals:
            return []
    return intervals

def pair_optimize(h, p, q, current_score):
    s = h[p] + h[q]
    lo = max(0.0, s - 1.0)
    hi = min(1.0, s)
    if hi - lo < 1e-12:
        return False
    
    # Save current values
    h_p_old, h_q_old = h[p], h[q]
    
    # Compute coefficients at 3 points

    dx = 2.0 / n

    c1 = np.zeros(2*n-1)
    c2 = np.zeros(2*n-1)

    c0[:] = np.correlate(h, 1.0 - h, mode="full") * dx
    h[p] = 0.5; h[q] = s - 0.5
    c1[:] = np.correlate(h, 1.0 - h, mode="full") * dx
    h[p] = 1.0; h[q] = s - 1.0
    c2[:] = np.correlate(h, 1.0 - h, mode="full") * dx
    h[p] = h_p_old; h[q] = h_q_old
    
    A = 2.0*(c0 - 2.0*c1 + c2)
    B = 4.0*c1 - 3.0*c0 - c2
    C = c0.copy()
    
    # Upper bound for T
    T_hi = current_score  # current score is achievable
    if T_hi <= 0:
        # Compute a valid upper bound
        h[p] = (lo+hi)*0.5; h[q] = s - (lo+hi)*0.5
        T_hi = exact_score(h)
        h[p] = h_p_old; h[q] = h_q_old
    
    # Lower bound: minimum possible (use a very negative value)
    T_lo = -1.0

    # (It should be, since the current point is in [lo, hi])

        T_mid = (T_lo + T_hi) * 0.5
        iv = feasible_intervals(A, B, C, T_mid, lo, hi)
        if iv:
            T_hi = T_mid
        else:
            T_lo = T_mid

    iv = feasible_intervals(A, B, C, T_hi, lo, hi)
    if not iv:
        return False
    
    # Pick midpoint of the first (or best) interval
    a_star = (iv[0][0] + iv[0][1]) * 0.5
    a_star = float(np.clip(a_star, lo, hi))
    
    h[p] = a_star
    h[q] = s - a_star
    new_score = exact_score(h)
    
    if new_score < current_score - 1e-13:
        return True
    else:

        h[q] = h_q_old

def slsqp_optimize(h0, target_sum, maxiter=40):
    n = len(h0)
    dx = 2.0 / n
    T_sm = 80.0
    
    def objective(x):
        c = np.correlate(x, 1.0 - x, mode="full") * dx
        c_max = np.max(c)
        if T_sm * c_max > 500:
            return c_max
        return (1.0/T_sm) * np.log(np.sum(np.exp(T_sm * c - T_sm * c_max))) + c_max
    
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                   'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    
    try:
        res = minimize(objective, h0.copy(), method='SLSQP', bounds=bounds,
                       constraints=constraints, options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None:
            x_opt = project_feasible(res.x, target_sum)
            if np.all(x_opt >= -1e-10) and np.all(x_opt <= 1.0 + 1e-10):

    return None

    xt = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
