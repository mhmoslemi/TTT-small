def lp_direction(h, current_score):
    """Find a feasible descent direction using linear programming on active constraints."""
    n = len(h)
    dx = 2.0 / n
    prof = np.correlate(h, 1.0 - h, mode="full")
    max_val = np.max(prof)
    
    # Active set: lags where prof[k] is close to max
    threshold = max_val - 0.01  # relative threshold
    active_mask = prof >= threshold
    active_lags = np.where(active_mask)[0].tolist()
    
    if len(active_lags) == 0:
        active_lags = [int(np.argmax(prof))]
    
    # Cap active set size
    if len(active_lags) > 50:
        # Keep top 50
        sorted_lags = sorted(active_lags, key=lambda k: -prof[k])
        active_lags = sorted_lags[:50]
    
    # Compute gradient
    G = compute_gradient(h, active_lags)  # shape (M, n)
    
    # LP: min t s.t. G @ d <= t, sum(d) = 0, |d_i| <= 1
    # Variables: [d_0, ..., d_{n-1}, t]
    # Minimize: [0, ..., 0, 1] (cost for t)
    M = len(active_lags)
    
    c_vec = np.zeros(n + 1)
    c_vec[-1] = 1.0
    
    # A_ub: G @ d - t <= 0  =>  [G | -1] @ [d; t] <= 0
    A_ub = np.hstack([G, -np.ones((M, 1))])
    b_ub = np.zeros(M)
    
    # A_eq: sum(d) = 0  =>  [1, 1, ..., 1, 0] @ [d; t] = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds: d_i in [-1, 1], t free (large bound)
    bounds = [( -1.0, 1.0)] * n + [(-100.0, 100.0)]
    
    try:
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            d = res.x[:n]
            t_val = res.x[-1]
            if t_val < -1e-10:  # descent direction
                return d
    except:

    """Use SLSQP to polish the solution."""
    n = len(h)
    
    def obj(w):
        return exact_score(w)
    
    # Actually, SLSQP needs a smooth objective. The max of correlations is not smooth.
    # Use a soft-max approximation? Or just try it and hope for the best.
    # The exact objective is piecewise smooth, which SLSQP can sometimes handle.
    
    bounds = [(0.0, 1.0)] * n

                    'jac': lambda w: np.ones(n)}]

                       constraints=constraints,

def transfer_resolution(h_src, n_target):
    """Interpolate h from n_src to n_target points."""
    n_src = len(h_src)
    if n_src == n_target:
        return h_src.copy()
    
    # Cell centers
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_tgt = (np.arange(n_target) + 0.5) * (2.0 / n_target)
    
    # Interpolate
    h_tgt = np.interp(x_tgt, x_src, h_src, left=0.0, right=0.0)
    h_tgt = np.clip(h_tgt, 0.0, 1.0)
    
    # Project to correct sum
    target_sum = n_target / 2.0
    h_tgt = project_feasible(h_tgt, target_sum)
    return h_tgt

import numpy as np
from scipy.optimize import minimize, linprog
import time


def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)


def project_feasible(v, target_sum):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v + mid, 0.0, 1.0))
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    result = np.clip(v + lam, 0.0, 1.0)
    diff = target_sum - np.sum(result)
    if abs(diff) > 1e-13:
        mask = (result > 1e-12) & (result < 1.0 - 1e-12)
        idx = np.where(mask)[0]
        if len(idx) > 0:
            result[idx] += diff / len(idx)
            result = np.clip(result, 0.0, 1.0)
    return result


def find_feasible_a(T, alpha, beta_c, gamma, lo, hi):
    intervals = [(lo, hi)]
    n_lags = len(alpha)
    for k in range(n_lags):
        a_k = alpha[k]
        b_k = beta_c[k]
        c_k = gamma[k] - T
        if abs(a_k) < 1e-16:
            if abs(b_k) < 1e-16:
                if c_k > 0:
                    return None
            else:
                bound = -c_k / b_k
                new_iv = []
                for (il, ir) in intervals:
                    if b_k > 0:
                        nir = min(ir, bound)
                        if il <= nir + 1e-14:
                            new_iv.append((il, nir))
                    else:
                        nil = max(il, bound)
                        if nil <= ir + 1e-14:
                            new_iv.append((nil, ir))
                intervals = new_iv
                if not intervals:
                    return None
        else:
            D = b_k * b_k - 4.0 * a_k * c_k
            if D < -1e-12:
                if a_k > 0:
                    return None
                continue
            if D < 0:
                D = 0.0
            sqrt_D = np.sqrt(D)
            r1 = (-b_k - sqrt_D) / (2.0 * a_k)
            r2 = (-b_k + sqrt_D) / (2.0 * a_k)
            if r1 > r2:
                r1, r2 = r2, r1
            new_iv = []
            for (il, ir) in intervals:
                if a_k > 0:
                    nil = max(il, r1)
                    nir = min(ir, r2)
                    if nil <= nir + 1e-14:
                        new_iv.append((nil, nir))
                else:
                    if il <= r1 + 1e-14:
                        nir = min(ir, r1)
                        if il <= nir + 1e-14:
                            new_iv.append((il, nir))
                    if ir >= r2 - 1e-14:
                        nil = max(il, r2)
                        if nil <= ir + 1e-14:
                            new_iv.append((nil, ir))
            intervals = new_iv
            if not intervals:
                return None
    return (intervals[0][0] + intervals[0][1]) / 2.0


def pair_optimize(h, p, q):
    n = len(h)
    s = h[p] + h[q]
    lo = max(0.0, s - 1.0)
    hi = min(1.0, s)
    if hi - lo < 1e-13:
        return None, exact_score(h)

    h0 = h.copy()
    h0[p] = 0.0
    h0[q] = s
    hh = h.copy()
    hh[p] = 0.5
    hh[q] = s - 0.5
    h1 = h.copy()
    h1[p] = 1.0
    h1[q] = s - 1.0

    prof0 = np.correlate(h0, 1.0 - h0, mode="full")
    profh = np.correlate(hh, 1.0 - hh, mode="full")
    prof1 = np.correlate(h1, 1.0 - h1, mode="full")

    gamma = prof0
    alpha = 2.0 * (prof1 - 2.0 * profh + prof0)
    beta_c = prof1 - prof0 - alpha

    a_cur = h[p]
    c_cur = alpha * a_cur * a_cur + beta_c * a_cur + gamma
    T_hi = float(np.max(c_cur))
    T_lo = -1.0

    for _ in range(50):
        T_mid = (T_lo + T_hi) / 2.0
        a_cand = find_feasible_a(T_mid, alpha, beta_c, gamma, lo, hi)
        if a_cand is not None:
            T_hi = T_mid
        else:
            T_lo = T_mid

    T_final = T_hi + 1e-13
    a_star = find_feasible_a(T_final, alpha, beta_c, gamma, lo, hi)
    if a_star is None:
        a_star = a_cur
    a_star = float(np.clip(a_star, lo, hi))

    h_new = h.copy()
    h_new[p] = a_star
    h_new[q] = s - a_star
    new_score = exact_score(h_new)
    return h_new, new_score


def compute_active_gradient(h, active_lags):
    n = len(h)
    eps = 1e-7
    M = len(active_lags)
    G = np.zeros((M, n))
    for i in range(n):
        hp = h.copy()
        hp[i] = min(1.0, h[i] + eps)
        hm = h.copy()
        hm[i] = max(0.0, h[i] - eps)
        pp = np.correlate(hp, 1.0 - hp, mode="full")
        pm = np.correlate(hm, 1.0 - hm, mode="full")
        actual_eps = hp[i] - hm[i]
        if actual_eps < 1e-15:
            continue
        for m, k in enumerate(active_lags):
            G[m, i] = (pp[k] - pm[k]) / actual_eps
    return G


def lp_direction_search(h, current_score):
    n = len(h)
    dx = 2.0 / n
    prof = np.correlate(h, 1.0 - h, mode="full")
    max_val = np.max(prof)

    for margin in [0.005, 0.01, 0.02, 0.05]:
        active_mask = prof >= (max_val - margin * max(1.0, max_val))
        active_lags = np.where(active_mask)[0].tolist()
        if len(active_lags) > 40:
            sorted_lags = sorted(active_lags, key=lambda k: -prof[k])
            active_lags = sorted_lags[:40]
        if not active_lags:
            active_lags = [int(np.argmax(prof))]

        G = compute_active_gradient(h, active_lags)
        M = len(active_lags)

        c_vec = np.zeros(n + 1)
        c_vec[-1] = 1.0
        A_ub = np.hstack([G, -np.ones((M, 1))])
        b_ub = np.zeros(M)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)
        bounds = [(-1.0, 1.0)] * n + [(-1000.0, 1000.0)]

        try:
            res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if res.success and res.x[-1] < -1e-10:
                d = res.x[:n]
                return d
        except:
            pass
    return None


def line_search(h, d, current_score, target_sum):
    n = len(h)
    alpha_max = 1.0
    for i in range(n):
        if d[i] > 1e-15:
            alpha_max = min(alpha_max, (1.0 - h[i]) / d[i])
        elif d[i] < -1e-15:
            alpha_max = min(alpha_max, -h[i] / d[i])
    if alpha_max <= 1e-12:
        return None, current_score
    alpha_max = min(alpha_max, 1.0)

    best_h = None
    best_score = current_score
    n_pts = 16
    alphas = np.linspace(0.01, alpha_max, n_pts)
    for alpha in alphas:
        h_trial = h + alpha * d
        h_trial = project_feasible(h_trial, target_sum)
        score = exact_score(h_trial)
        if score < best_score - 1e-13:
            best_score = score
            best_h = h_trial.copy()
    return best_h, best_score


def slsqp_polish(h, target_sum):
    n = len(h)

    def obj(w):
        return exact_score(w)

    def eq_fun(w):
        return np.sum(w) - target_sum

    def eq_jac(w):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=constraints,
                       options={'maxiter': 60, 'ftol': 1e-10})
        h_new = np.clip(res.x, 0.0, 1.0)
        h_new = project_feasible(h_new, target_sum)
        score = exact_score(h_new)
        return h_new, score
    except:
        return None, exact_score(h)


def transfer_resolution(h_src, n_target):
    n_src = len(h_src)
    if n_src == n_target:
        return h_src.copy()
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_tgt = (np.arange(n_target) + 0.5) * (2.0 / n_target)
    h_tgt = np.interp(x_tgt, x_src, h_src, left=0.0, right=0.0)
    h_tgt = np.clip(h_tgt, 0.0, 1.0)
    target_sum = n_target / 2.0
    h_tgt = project_feasible(h_tgt, target_sum)
    return h_tgt
