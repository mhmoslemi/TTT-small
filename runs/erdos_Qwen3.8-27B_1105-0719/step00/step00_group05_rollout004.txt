Given h = sigmoid(y + lambda), where lambda is chosen so sum(h) = T = n/2.

The smooth objective is F_beta(h) = (1/beta) * log(sum_r exp(beta * V_r(h)))

where V_r are the directional overlap values.

We need dF/dy_j.

Let m_i = h_i(1-h_i) = sigmoid'(y_i + lambda)

The constraint is g(y) = sum_i sigmoid(y_i + lambda(y)) - T = 0

By implicit function theorem:
dg/dy_j = sigmoid'(y_j + lambda) + sum_i sigmoid'(y_i + lambda) * dlambda/dy_j = 0
m_j + M * dlambda/dy_j = 0
dlambda/dy_j = -m_j / M

where M = sum_i m_i

So dh_i/dy_j = sigmoid'(y_i + lambda) * (delta_ij + dlambda/dy_j)
= m_i * (delta_ij - m_j/M)

Therefore:
dF/dy_j = sum_i (dF/dh_i) * dh_i/dy_j
= sum_i G_i * m_i * (delta_ij - m_j/M)
= m_j * G_j - (m_j/M) * sum_i G_i * m_i
= m_j * (G_j - B)

where B = sum_i G_i * m_i / M

This is correct.

Actually, let me think about what np.correlate(h, 1-h, mode="full") gives.

np.correlate(a, b, mode="full") computes the cross-correlation:
c[k] = sum_i a[i] * b[i + k - (len(b)-1)]

For len(a) = len(b) = n:
c[k] = sum_i a[i] * b[i + k - (n-1)] for valid i

Actually, let me just use the definition directly. The maximum of the full correlation between h and 1-h, times dx, is our objective.

Let me verify: np.correlate(h, 1-h, mode='full') has length 2n-1.
The k-th element (0-indexed) is sum_{i} h[i] * (1-h)[i + k - (n-1)]

So for k = 0: sum_{i} h[i] * (1-h)[i - (n-1)] ... this doesn't make sense for small i.

Actually, numpy's correlate definition:
c[k] = sum_{τ} a[τ] * conj(v[τ + k])

where the sum is over valid τ.

For a = h, v = 1-h, both length n:
c[k] = sum_{τ=0}^{n-1} h[τ] * (1-h)[τ + k] where we need 0 <= τ+k <= n-1

So for k = -(n-1) to n-1 (shifted by n-1 to get 0 to 2n-2):
c[k + n - 1] = sum_{τ=max(0,-k)}^{min(n-1, n-1-k)} h[τ] * (1-h)[τ+k]

The max over all these, times dx = 2/n, gives C₅.

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

# Global state
_best_h = None
_best_c = np.inf
_deadline = 0.0
_rng = None

def exact_C(h):
    h = np.asarray(h, dtype=np.float64)
    n = len(h)
    if n == 0:
        return np.inf
    val = np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n)
    if not np.isfinite(val):
        return np.inf
    return val

def project_feasible(h):
    h = np.asarray(h, dtype=np.float64).copy()
    n = len(h)
    T = n / 2.0
    # Water filling: find lambda such that sum(clip(h_i + lambda, 0, 1)) = T
    lo = -max(h.max() - 0.0, 0.0) - 2.0
    hi = min(h.min() + 1.0, 1.0) + 2.0
    # Actually simpler: binary search on lambda
    # sum(clip(h + lambda, 0, 1)) is monotone in lambda
    lo = -h.max() - 1.0
    hi = 1.0 - h.min() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s < T:
            lo = mid
        else:
            hi = mid
    result = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    # Fix sum
    delta = T - np.sum(result)
    if abs(delta) > 1e-15:
        # Distribute delta to coordinates with slack
        if delta > 0:
            # Add to coordinates below 1
            slack = 1.0 - result
            total_slack = slack.sum()
            if total_slack > 0:
                result += delta * slack / total_slack
        else:
            # Remove from coordinates above 0
            room = result.copy()
            total_room = room.sum()
            if total_room > 0:
                result += delta * room / total_room
    result = np.clip(result, 0.0, 1.0)
    return result

def is_feasible(h, tol=1e-8):
    h = np.asarray(h)
    T = len(h) / 2.0
    if h.min() < -tol or h.max() > 1.0 + tol:
        return False
    if abs(h.sum() - T) > tol:
        return False
    return True

def logit_of_h(h, eps=1e-10):
    h = np.clip(h, eps, 1.0 - eps)
    return np.log(h / (1.0 - h))

def sigmoid(x):
    # Stable sigmoid
    result = np.zeros_like(x)
    pos = x >= 0
    result[pos] = 1.0 / (1.0 + np.exp(-x[pos]))
    exp_x = np.exp(x[~pos])
    result[~pos] = exp_x / (1.0 + exp_x)
    return result

def h_from_logit(y):
    y = np.asarray(y, dtype=np.float64)
    n = len(y)
    T = n / 2.0
    # Solve for lambda: sum(sigmoid(y_i + lambda)) = T
    # phi(lambda) = sum(sigmoid(y_i + lambda)) - T is monotone increasing
    # phi(-inf) = -T < 0, phi(+inf) = n - T = T > 0
    y_max = np.max(y)
    y_min = np.min(y)
    lo = y_max - 80.0  # sigmoid(y_i + lo) ≈ 0 for all i
    hi = -y_min + 80.0  # sigmoid(y_i + hi) ≈ 1 for all i
    # Wait, I need phi(lo) < 0 and phi(hi) > 0
    # phi(lo) = sum(sigmoid(y_i + lo)) - T
    # For lo very negative, sigmoid ≈ 0, so phi ≈ -T < 0. Good.
    # For hi very positive, sigmoid ≈ 1, so phi ≈ n - T = T > 0. Good.
    # But I need lo < hi, which means y_max - 80 < -y_min + 80, i.e., y_max + y_min < 160
    # This is almost always true for reasonable y.
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(sigmoid(y + mid))
        if s < T:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    result = sigmoid(y + lam)
    # Fix any numerical issues
    result = np.clip(result, 0.0, 1.0)
    return result

def record_candidate(h):
    global _best_h, _best_c
    h = np.asarray(h, dtype=np.float64)
    if not is_feasible(h, tol=1e-6):
        h = project_feasible(h)
    c = exact_C(h)
    if np.isfinite(c) and c < _best_c:
        _best_c = c
        _best_h = h.copy()

def directional_grad(h, beta):
    """Compute the smooth surrogate value and gradient G = dF/dh."""
    n = len(h)
    dx = 2.0 / n
    T = n / 2.0
    
    # Compute all directional values V
    # V_0 = dx * sum h_i(1-h_i)
    # V_m^+ = dx * sum_{i=0}^{n-m-1} h_i(1-h_{i+m})
    # V_m^- = dx * sum_{i=0}^{n-m-1} h_{i+m}(1-h_i)
    
    # For the smooth objective, we can use the full correlation directly
    # Full correlation has 2n-1 entries
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    
    # Now compute gradient G_j = dF_beta/dh_j
    # F = (1/beta) * log(sum exp(beta * V_r))
    # dF/dV_r = exp(beta * V_r) / sum exp(beta * V_s) = softmax(beta * V)_r
    # dV_r/dh_j as derived above
    
    # For numerical stability
    v_max = np.max(corr)
    weights = np.exp(beta * (corr - v_max))
    w_sum = np.sum(weights)
    if w_sum <= 0 or not np.isfinite(w_sum):
        # Fallback: uniform over max elements
        idx = np.where(np.abs(corr - v_max) < 1e-6)[0]
        weights = np.zeros_like(corr)
        weights[idx] = 1.0 / len(idx)
    else:
        weights /= w_sum
    
    F = v_max + np.log(w_sum) / beta
    
    # Now compute G = dF/dh = sum_r w_r * dV_r/dh
    # The full correlation corr[k] for k=0..2n-2 corresponds to lag = k-(n-1)
    # corr[k] = sum_i h[i] * (1-h)[i + lag] where lag = k - (n-1)
    #         = sum_i h[i] * (1 - h[i + lag])
    
    G = np.zeros(n)
    
    for k in range(2 * n - 1):
        if weights[k] < 1e-15:
            continue
        lag = k - (n - 1)  # lag ranges from -(n-1) to n-1
        if lag == 0:
            # V = dx * sum_i h_i(1-h_i)
            # dV/dh_j = dx * (1 - 2*h_j)
            G += weights[k] * dx * (1.0 - 2.0 * h)
        elif lag > 0:
            # V = dx * sum_{i=0}^{n-lag-1} h_i * (1 - h_{i+lag})
            # dV/dh_j: 
            #   if j < n-lag: term h_j(1-h_{j+lag}) -> dx*(1-h_{j+lag})
            #   if j >= lag: term h_{j-lag}(1-h_j) -> -dx*h_{j-lag}
            m = lag
            if m < n:
                # j in 0..n-m-1: contribution dx*(1-h[j+m])
                G[:n-m] += weights[k] * dx * (1.0 - h[m:])
                # j in m..n-1: contribution -dx*h[j-m]
                G[m:] += weights[k] * dx * (-h[:n-m])
        else:
            # lag < 0, let m = -lag > 0
            # V = dx * sum_{i=0}^{n-m-1} h_{i+m} * (1 - h_i)
            #   = dx * sum_{i=m}^{n-1} h_i * (1 - h_{i-m})
            # dV/dh_j:
            #   if j >= m: term h_j(1-h_{j-m}) -> dx*(1-h_{j-m})
            #   if j < n-m: term h_{j+m}(1-h_j) -> -dx*h_{j+m}
            m = -lag
            if m < n:
                # j in m..n-1: contribution dx*(1-h[j-m])
                G[m:] += weights[k] * dx * (1.0 - h[:n-m])
                # j in 0..n-m-1: contribution -dx*h[j+m]
                G[:n-m] += weights[k] * dx * (-h[m:])
    
    return F, G

def smooth_obj_gamma(y, beta):
    """Compute smooth objective and gradient in y-space."""
    global _deadline
    if time.time() > _deadline:
        raise ValueError("timeout")
    
    h = h_from_logit(y)
    F, G = directional_grad(h, beta)
    
    if not np.isfinite(F):
        return 1e6, np.zeros_like(y)
    
    # Record candidate
    record_candidate(h)
    
    # Chain rule: dF/dy_j = m_j * (G_j - B)
    m = h * (1.0 - h)
    M = np.sum(m)
    if M < 1e-15:
        return F, np.zeros_like(y)
    B = np.dot(G, m) / M
    grad_y = m * (G - B)
    
    if not np.all(np.isfinite(grad_y)):
        grad_y = np.zeros_like(y)
    
    return F, grad_y

def pair_polish(h, max_moves=30, time_limit=None):
    """Local pair transfer optimization."""
    global _best_h, _best_c, _deadline
    if time_limit is None:
        time_limit = _deadline - time.time() - 10.0
    if time_limit <= 0:
        return h
    
    n = len(h)
    T = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    c_cur = exact_C(h)
    
    # Get gradient for pair selection
    beta_tmp = 200.0
    try:
        _, G = directional_grad(h, beta_tmp)
    except:
        G = np.zeros(n)
    
    # Select pairs
    top_idx = np.argsort(-np.abs(G))[:min(16, n)]
    pairs = set()
    for i in range(len(top_idx)):
        for j in range(i + 1, len(top_idx)):
            pairs.add((top_idx[i], top_idx[j]))
        if len(pairs) >= 40:
            break
    # Add some random pairs
    for _ in range(20):
        a = _rng.integers(0, n)
        b = _rng.integers(0, n)
        if a != b:
            pairs.add((a, b))
    
    pairs = list(pairs)[:max(60, 2 * max_moves)]
    _rng.shuffle(pairs)
    
    moves_accepted = 0
    stagnant = 0
    
    for (a, b) in pairs:
        if time.time() > _deadline - 10:
            break
        if moves_accepted >= max_moves:
            break
        
        ha = h[a]
        hb = h[b]
        
        # Valid range for t: h[a]+t in [0,1], h[b]-t in [0,1]
        lo_t = max(-ha, hb - 1.0)
        hi_t = min(1.0 - ha, hb)
        
        if hi_t - lo_t < 1e-12:
            continue
        
        # Grid search
        n_grid = 21
        t_vals = np.linspace(lo_t, hi_t, n_grid)
        best_t = 0.0
        best_c_val = c_cur
        
        for t in t_vals:
            h[a] = ha + t
            h[b] = hb - t
            c_test = exact_C(h)
            if c_test < best_c_val - 1e-14:
                best_c_val = c_test
                best_t = t
            h[a] = ha
            h[b] = hb
        
        if best_t == 0.0 or best_c_val >= c_cur - 1e-14:
            stagnant += 1
            if stagnant >= 3:
                break
            continue
        
        # Local refine around best_t
        idx_best = np.argmin(np.abs(t_vals - best_t))
        left = max(lo_t, t_vals[max(0, idx_best - 1)])
        right = min(hi_t, t_vals[min(n_grid - 1, idx_best + 1)])
        
        def pair_obj(t):
            h[a] = ha + t
            h[b] = hb - t
            val = exact_C(h)
            h[a] = ha
            h[b] = hb
            return val
        
        try:
            res = minimize_scalar(pair_obj, bounds=(left, right), method='bounded',
                                   options={'xatol': 1e-10, 'maxiter': 30})
            if res.success and res.fun < best_c_val:
                best_t = res.x
                best_c_val = res.fun
        except:
            pass
        
        # Accept
        h[a] = ha + best_t
        h[b] = hb - best_t
        h = project_feasible(h)
        c_cur = exact_C(h)
        record_candidate(h)
        moves_accepted += 1
        stagnant = 0
    
    return h

def random_explore(h, n_iter, time_limit):
    """Simple random perturbation with acceptance."""
    global _best_h, _best_c, _deadline
    n = len(h)
    h_cur = h.copy()
    c_cur = exact_C(h_cur)
    
    temperature = 1e-3
    for i in range(n_iter):
        if time.time() > _deadline - 10:
            break
        if time.time() > time_limit:
            break
        
        dy = _rng.standard_normal(n) * temperature
        y = logit_of_h(h_cur) + dy
        h_new = h_from_logit(y)
        h_new = project_feasible(h_new)
        c_new = exact_C(h_new)
        
        if c_new < c_cur:
            h_cur = h_new
            c_cur = c_new
            record_candidate(h_cur)
            temperature = max(temperature * 0.95, 1e-6)
        else:
            # Accept with probability exp(-(c_new-c_cur)/temperature_c)
            delta = c_new - c_cur
            if delta > 0 and delta < 0.01:
                if _rng.random() < np.exp(-delta / (temperature * 0.1)):
                    h_cur = h_new
                    c_cur = c_new
            temperature = max(temperature * 0.9, 1e-6)
    
    return h_cur

def run(seed=42, budget_s=1100, **kwargs):

    # Shifts
    for s in [1, 2]:
        h_sh = np.roll(h0, s)
        h_sh = project_feasible(h_sh)
        record_candidate(h_sh)
    
    # BFGS optimization in logit space
    betas = [50.0, 100.0, 200.0, 400.0]

    x_grid = (np.arange(n) + 0.5) / n

        h_sine = 0.5 + 0.3 * np.sin(2 * np.pi * freq * x_grid)
        h_sine = project_feasible(h_sine)

    # Deduplicate
    seen = []
    unique_starts = []
    for sp in start_points:

            break
        
        # Convert to logit space

                return smooth_obj_gamma(y, beta)

                y0 = np.log(np.clip(h_warm, 1e-8, 1.0 - 1e-8) / (1.0 - np.clip(h_warm, 1e-8, 1.0 - 1e-8)))
