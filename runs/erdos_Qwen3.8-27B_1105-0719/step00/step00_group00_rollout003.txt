        if v3 < best_val:
            best_val = v3
            best_h = h3.copy()

    # Phase 4: Try different initializations at same n
    inits = [
        make_binary_init(n, seed),
        make_alternating_init(n),
        make_block_init(n),
    ]
    for h_init in inits:
        if time_left() < 60:
            break
        h4 = lse_optimize(h_init, n, [5, 20, 50, 200, 500],
                          lr_base=0.05, max_iters=200)
        v4 = evaluate(h4)
        if v4 < best_val:
            best_val = v4
            best_h = h4.copy()
        # Subgradient polish
        if time_left() > 60:
            h4b, v4b = subgrad_optimize(h4, n, n_iter=1000)
            if v4b < best_val:
                best_val = v4b
                best_h = h4b.copy()

    # Phase 5: Upscale to higher resolutions
    for n_new in [60, 90, 120, 180]:
        if time_left() < 80:
            break
        x_old = np.linspace(0, 2, n)
        x_new = np.linspace(0, 2, n_new)
        h_up = np.interp(x_new, x_old, best_h)
        h_up = project(h_up, n_new)
        h5 = lse_optimize(h_up, n_new, [20, 100, 300, 1000],
                          lr_base=0.03, max_iters=150)
        v5 = evaluate(h5)
        if v5 < best_val:
            best_val = v5
            best_h = h5.copy()
            n = n_new
            # Further subgradient polish at new resolution
            if time_left() > 60:
                h5b, v5b = subgrad_optimize(h5, n, n_iter=1500)
                if v5b < best_val:
                    best_val = v5b
                    best_h = h5b.copy()

    # Phase 6: Basin hopping
    if time_left() > 60:
        for restart in range(80):
            if time_left() < 30:
                break
            scale = 0.03 + 0.07 * (restart % 4)
            pert = rng.normal(0, scale, n)
            h_p = project(best_h + pert, n)
            hp = lse_optimize(h_p, n, [50, 200, 500],
                              lr_base=0.04, max_iters=100)
            vp = evaluate(hp)
            if vp < best_val:
                best_val = vp
                best_h = hp.copy()

    # Phase 7: Final combined polish
    if time_left() > 40:
        h7 = lse_optimize(best_h, n, [100, 500, 2000],
                          lr_base=0.02, max_iters=300)
        v7 = evaluate(h7)
        if v7 < best_val:
            best_val = v7
            best_h = h7.copy()

    if time_left() > 30:
        h7b, v7b = subgrad_optimize(best_h, n, n_iter=1500)
        if v7b < best_val:
            best_val = v7b
            best_h = h7b.copy()

    # Phase 8: Final pair sweep
    if time_left() > 20:
        h8, v8 = pair_improve(best_h, n, rng, n_trials=2000)
        if v8 < best_val:
            best_val = v8
            best_h = h8.copy()

    return best_h, best_val, len(best_h)
