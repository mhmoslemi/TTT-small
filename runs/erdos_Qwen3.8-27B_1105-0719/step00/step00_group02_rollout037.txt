def exact_score(h, n):

def feasible_interval(h, u, n):
    """Find [t_lo, t_hi] such that 0 <= h + t*u <= 1 for all elements."""
    t_lo = -np.inf
    t_hi = np.inf
    for i in range(len(u)):
        if u[i] > 1e-14:
            t_lo = max(t_lo, -h[i] / u[i])
            t_hi = min(t_hi, (1.0 - h[i]) / u[i])
        elif u[i] < -1e-14:
            t_lo = max(t_lo, (1.0 - h[i]) / u[i])
            t_hi = min(t_hi, -h[i] / u[i])

    t_lo = max(t_lo, -1.0)  # reasonable bound

    t_hi = max(t_hi, 0.0)
    if t_hi - t_lo < 1e-14:

    return t_lo, t_hi

    t_lo = 0.0
    t_hi = 0.0
    # We want the largest interval containing 0 where 0 <= h + t*u <= 1
    n = len(h)
    for i in range(n):
        ui = u[i]
        if ui > 1e-15:
            # h[i] + t*ui in [0,1] => -h[i]/ui <= t <= (1-h[i])/ui
            lo = -h[i] / ui
            hi = (1.0 - h[i]) / ui
            if lo > t_lo:
                t_lo = lo
            if hi < t_hi:
                t_hi = hi
        elif ui < -1e-15:
            # h[i] + t*ui in [0,1] => (1-h[i])/ui <= t <= -h[i]/ui
            lo = (1.0 - h[i]) / ui
            hi = -h[i] / ui
            if lo > t_lo:
                t_lo = lo
            if hi < t_hi:
                t_hi = hi
    # Ensure 0 is in interval (clamp)
    t_lo = min(t_lo, 0.0)
    t_hi = max(t_hi, 0.0)
    if t_hi - t_lo < 1e-15:
        return None
    return t_lo, t_hi

def line_search(h, u, n_samples=256):
    """Find best t on line h + t*u. Returns (t, h_new) or (None, None)."""
    interval = get_feasible_interval(h, u)
    if interval is None:
        return None, None
    t_lo, t_hi = interval
    
    n = len(h)
    v = 1.0 - h

    q0 = np.correlate(h, v, mode='full')
    q1 = np.correlate(u, v, mode='full') - np.correlate(h, u, mode='full')
    q2 = -np.correlate(u, u, mode='full')
    
    # Dense sampling
    t_vals = np.linspace(t_lo, t_hi, n_samples)

    t_col = t_vals[:, np.newaxis]

    # Refine: finer grid around best
    w = max(2, n_samples // 64)
    lo_idx = max(0, best_idx - w)
    hi_idx = min(n_samples - 1, best_idx + w)
    t_fine = np.linspace(t_vals[lo_idx], t_vals[hi_idx], 128)

    F_fine = np.max(q0[np.newaxis, :] + t_col_fine * q1[np.newaxis, :] + t_col_fine**2 * q2[np.newaxis, :], axis=1)
    best_fine = np.argmin(F_fine)
    t_best = t_fine[best_fine]
    
    h_new = h + t_best * u
    # Tiny repair for floating point
    h_new = np.clip(h_new, 0.0, 1.0)
    # Repair mass
    mass = n / 2.0
    residual = mass - h_new.sum()
    if abs(residual) > 1e-10:
        # Distribute correction
        if residual > 0:
            slack = np.maximum(0, 1.0 - h_new)
            if slack.sum() > 1e-15:
                h_new += residual * slack / slack.sum()
        else:
            afford = np.maximum(0, h_new)
            if afford.sum() > 1e-15:
                h_new += residual * afford / afford.sum()
        h_new = np.clip(h_new, 0.0, 1.0)
