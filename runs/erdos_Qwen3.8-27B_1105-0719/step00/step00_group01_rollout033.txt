    N = len(h)

def project_to_feasible(h, target_sum):
    """Project h onto {0<=x<=1, sum(x)=target_sum}"""
    N = len(h)
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        h_proj = np.clip(h - mid, 0.0, 1.0)
        s = h_proj.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h - (lo + hi) / 2, 0.0, 1.0)
    # Fine-tune
    diff = target_sum - h_proj.sum()
    if abs(diff) > 1e-10:
        # Adjust largest smallest elements
        if diff > 0:
            idx = np.argsort(-h_proj)  # those closest to 1 first... actually those with most room
            idx = np.argsort(1.0 - h_proj)  # most room first
            for i in idx:
                add = min(diff, 1.0 - h_proj[i])
                h_proj[i] += add
                diff -= add
                if diff < 1e-12:
                    break
        else:
            idx = np.argsort(h_proj)  # those closest to 0 first
            for i in idx:
                sub = min(-diff, h_proj[i])
                h_proj[i] -= sub
                diff += sub
                if diff > -1e-12:
                    break
    return h_proj

        return x[-1]

        return g

        g[:N] = 1.0
        return g

        t = x[-1]
        c = np.correlate(h, 1.0 - h, mode="full") * dx

        {'type': 'ineq', 'fun': con_ieq, 'jac': con_ieq_grad}

def subgradient_descent(h0, dx, target_sum, maxiter=5000, alpha0=0.01):

    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)

        c = np.correlate(h, 1.0 - h, mode="full") * dx
        c5 = float(np.max(c))

            best_c5 = c5

        alpha = alpha0 / (1.0 + 0.01 * it)

        h = project_to_feasible(h, target_sum)

            h_new, c5_new = slsqp_solve(h_sub, dx, target_sum, maxiter=300)
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new

    # Phase 3: Random restarts with SLSQP
    n_attempt = 0
    while time.time() - start_time < budget_s * 0.88 and n_attempt < 20:
        scale = 0.08 / (1.0 + 0.3 * n_attempt)
        h_pert = best_h + rng.standard_normal(N) * scale
        h_pert = np.clip(h_pert, 0.0, 1.0)
        h_pert = project_to_feasible(h_pert, target_sum)

        remaining = budget_s * 0.88 - (time.time() - start_time)
        if remaining < 20:
            break
        maxiter = max(100, min(400, int(remaining * 10)))
        h_new, c5_new = slsqp_solve(h_pert, dx, target_sum, maxiter=maxiter)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
        n_attempt += 1

    # Phase 4: Try slightly different n values
    for n_try in [N + 1, N - 1, N + 2, N - 2]:
        if time.time() - start_time > budget_s * 0.93:
            break
        if n_try < 20:
            continue
        dx_try = 2.0 / n_try
        h_try = try_different_n(best_h, n_try, dx, target_sum)
        h_try = project_to_feasible(h_try, n_try / 2.0)
        remaining = budget_s * 0.93 - (time.time() - start_time)
        if remaining < 20:
            continue
        h_new, c5_new = slsqp_solve(h_try, dx_try, n_try / 2.0, maxiter=200)
        c5_new = compute_c5(h_new, dx_try)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            N = n_try
            dx = dx_try
            target_sum = n_try / 2.0

    # Phase 5: Final polish
    if time.time() - start_time < budget_s * 0.97:
        h_new, c5_new = slsqp_solve(best_h, dx, target_sum, maxiter=300)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new

    best_h = np.clip(best_h, 0.0, 1.0)
    final_c5 = compute_c5(best_h, dx)
    return (best_h, final_c5, len(best_h))
