    # === MAIN OPTIMIZATION ===
    best_h = initial_h_values.copy().astype(float)
    best_c5 = float(evaluate_erdos_solution(best_h))
    best_n = len(best_h)

    T = budget_s

    # Phase 1: binary local search
    h, c5 = binary_local_search(best_h, best_n, start_time + T * 0.22, rng)
    if c5 < best_c5:
        best_h, best_c5 = h.copy(), c5

    # Phase 2: continuous refinement
    h, c5 = continuous_refine(best_h, best_n, start_time + T * 0.40, temp=0.004)
    if c5 < best_c5:
        best_h, best_c5 = h.copy(), c5

    # Phase 3: binary local search again from refined
    h, c5 = binary_local_search(best_h, best_n, start_time + T * 0.58, rng)
    if c5 < best_c5:
        best_h, best_c5 = h.copy(), c5

    # Phase 4: try nearby n values
    n_candidates = sorted(set([best_n + 2, best_n - 2, best_n + 4, best_n + 6, best_n + 8]),
                         key=lambda x: abs(x - best_n))
    for n_try in n_candidates:
        if n_try < 12 or n_try > 120:
            continue
        if elapsed() > T * 0.82:
            break
        x_old = np.linspace(0, 2, best_n, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_try = np.interp(x_new, x_old, best_h)
        h_try = project(h_try, n_try)
        dl = min(start_time + T * 0.88, elapsed() + T * 0.04)
        h, c5 = binary_local_search(h_try, n_try, dl, rng)
        if c5 < best_c5:
            best_h, best_c5, best_n = h.copy(), c5, n_try
        if elapsed() > T * 0.82:
            continue
        dl2 = min(start_time + T * 0.90, elapsed() + T * 0.03)
        h, c5 = continuous_refine(best_h, best_n, dl2, temp=0.003)
        if c5 < best_c5:
            best_h, best_c5 = h.copy(), c5
        if elapsed() > T * 0.82:
            continue
        dl3 = min(start_time + T * 0.92, elapsed() + T * 0.03)
        h, c5 = binary_local_search(best_h, best_n, dl3, rng)
        if c5 < best_c5:
            best_h, best_c5 = h.copy(), c5

    # Phase 5: final polish on best solution
    if elapsed() < T * 0.95:
        h, c5 = binary_local_search(best_h, best_n, start_time + T * 0.97, rng)
        if c5 < best_c5:
            best_h, best_c5 = h.copy(), c5

    final_c5 = float(evaluate_erdos_solution(best_h))
    return best_h, min(best_c5, final_c5), best_n
