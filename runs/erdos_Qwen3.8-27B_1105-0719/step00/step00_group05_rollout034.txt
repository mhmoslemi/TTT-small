    start_time = time.time()

import numpy as np
import time

def evaluate_exact(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(z):
    n = len(z)
    target = n / 2.0
    lo = float(-z.max()) - 2.0
    hi = float(1.0 - z.min()) + 2.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = float(np.clip(z + mid, 0.0, 1.0).sum())
        if s < target:
            lo = mid
        else:
            hi = mid
    out = np.clip(z + 0.5*(lo+hi), 0.0, 1.0)
    diff = target - float(out.sum())
    if abs(diff) > 1e-14:
        if diff > 0:
            for i in range(n):
                room = 1.0 - out[i]
                if room >= diff:
                    out[i] += diff
                    diff = 0.0
                    break
                out[i] = 1.0
                diff -= room
        else:
            for i in range(n):
                room = out[i]
                if room >= -diff:
                    out[i] += diff
                    diff = 0.0
                    break
                out[i] = 0.0
                diff += room
    return out

def compute_weights(h, beta):
    n = len(h)
    dvals = np.zeros(2*n - 1)
    dvals[n-1] = float(np.dot(h, 1.0 - h))
    for m in range(1, n):
        dvals[n-1+m] = float(np.dot(h[m:], 1.0 - h[:n-m]))
        dvals[n-1-m] = float(np.dot(h[:n-m], 1.0 - h[m:]))
    dmax = float(dvals.max())
    shifted = np.clip(beta * (dvals - dmax), -500.0, 0.0)
    exps = np.exp(shifted)
    p = exps / float(exps.sum())
    return p

def smooth_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    p = compute_weights(h, beta)
    grad = np.zeros(n)
    grad += p[n-1] * (1.0 - 2.0*h)
    for m in range(1, n):
        wp = p[n-1+m]
        wm = p[n-1-m]
        s = wp + wm
        if s < 1e-15:
            continue
        grad[:n-m] += wm - s * h[m:]
        grad[m:] += wp - s * h[:n-m]
    return dx * grad

def smooth_descent(h0, beta, max_iter, time_limit):
    start = time.time()
    current = h0.copy()
    c_cur = evaluate_exact(current)
    step = 0.05
    no_improve = 0

        g = smooth_grad(current, beta)
        gn = float(np.linalg.norm(g))
        if gn < 1e-10:

        best_c = c_cur
        best_h = current
        for scale in [step, step*0.5, step*0.25, step*2.0, step*4.0]:
            cand = project_feasible(current - scale * g)
            c = evaluate_exact(cand)
            if c < best_c - 1e-14:
                best_c = c
                best_h = cand
        
        if best_c < c_cur - 1e-14:
            current = best_h
            c_cur = best_c
            no_improve = 0
            step = min(step * 1.2, 0.2)
        else:
            no_improve += 1
            step *= 0.5
            if no_improve > 30 or step < 1e-10:
                break
    
    return current, c_cur

def pair_optimize(h, p, q, n_grid=25):
    eps_lo = max(-h[p], h[q] - 1.0)
    eps_hi = min(1.0 - h[p], h[q])
    if eps_hi - eps_lo < 1e-12:

import numpy as np

def evaluate_exact(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_feasible(z):
    n = len(z)
    target = n / 2.0
    lo = float(-z.max()) - 2.0
    hi = float(1.0 - z.min()) + 2.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = float(np.clip(z + mid, 0.0, 1.0).sum())
        if s < target:
            lo = mid
        else:
            hi = mid
    out = np.clip(z + 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - float(out.sum())
    if abs(diff) > 1e-14:
        if diff > 0:
            for i in range(n):
                room = 1.0 - out[i]
                if room >= diff:
                    out[i] += diff
                    break
                out[i] = 1.0
                diff -= room
        else:
            for i in range(n):
                room = out[i]
                if room >= -diff:
                    out[i] += diff
                    break
                out[i] = 0.0
                diff += room
    return out


def compute_weights(h, beta):
    n = len(h)
    dvals = np.zeros(2 * n - 1)
    dvals[n - 1] = float(np.dot(h, 1.0 - h))
    for m in range(1, n):
        dvals[n - 1 + m] = float(np.dot(h[m:], 1.0 - h[:n - m]))
        dvals[n - 1 - m] = float(np.dot(h[:n - m], 1.0 - h[m:]))
    dmax = float(dvals.max())
    shifted = np.clip(beta * (dvals - dmax), -500.0, 0.0)
    exps = np.exp(shifted)
    return exps / float(exps.sum())


def smooth_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    p = compute_weights(h, beta)
    grad = np.zeros(n)
    grad += p[n - 1] * (1.0 - 2.0 * h)
    for m in range(1, n):
        wp = p[n - 1 + m]
        wm = p[n - 1 - m]
        s = wp + wm
        if s < 1e-15:
            continue
        grad[:n - m] += wm - s * h[m:]
        grad[m:] += wp - s * h[:n - m]
    return dx * grad

    start = time.time()
    current = h0.copy()
    c_cur = evaluate_exact(current)
    step = 0.05
    no_improve = 0

    for _ in range(max_iter):
        if time.time() - start > time_limit:
            break
        g = smooth_grad(current, beta)
        gn = float(np.linalg.norm(g))
        if gn < 1e-12:
            break

        best_c = c_cur
        best_h = current
        best_scale = step
        for scale in [step, step * 0.5, step * 0.25, step * 2.0, step * 0.1]:
            cand = project_feasible(current - scale * g)
            c = evaluate_exact(cand)
            if c < best_c - 1e-14:
                best_c = c
                best_h = cand
                best_scale = scale

        if best_c < c_cur - 1e-14:
            current = best_h
            c_cur = best_c
            no_improve = 0
            step = min(best_scale * 1.1, 0.15)
        else:
            no_improve += 1
            step *= 0.6
            if no_improve > 25 or step < 1e-10:
                break

    return current, c_cur
