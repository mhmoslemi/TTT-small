def exact_objective(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair_sum(h, n):
    target = n * 0.5
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    S = target - h.sum()
    if abs(S) < 1e-14:
        return h
    h += S / n
    h = np.clip(h, 0.0, 1.0)
    r = target - h.sum()
    if abs(r) > 1e-13:
        if r > 0:
            i = int(np.argmax(1.0 - h))
            h[i] = min(1.0, h[i] + r)
        else:
            i = int(np.argmax(h))
            h[i] = max(0.0, h[i] + r)
    return h

def golden_pair(h, i, j, n_iter=28):
    n = len(h)
    t_lo = -min(h[i], 1.0 - h[j]) + 1e-10
    t_hi = min(1.0 - h[i], h[j]) - 1e-10

            dv = a + gr * (b - a)
