            r += sub
    return x

import numpy as np
import time

def compute_correlation(h):
    """Compute the full correlation vector."""
    return np.correlate(h, 1.0 - h, mode="full")

def compute_c5(h):
    """Compute C5 value."""
    n = len(h)
    dx = 2.0 / n
    R = compute_correlation(h)
    return np.max(R) * dx

def project_box_hyperplane(y, target):
    """Project y onto {x: 0<=x_i<=1, sum(x)=target}."""
    n = len(y)
    target = float(np.clip(target, 0, n))
    lo = float(y.min()) - 1.0
    hi = float(y.max()) + 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    x = np.clip(y - lam, 0.0, 1.0)
    # Fix residual
    r = target - x.sum()
    if abs(r) > 1e-10:
        if r > 0:
            slack = 1.0 - x
            order = np.argsort(-slack, kind='stable')
            for i in order:
                if r <= 0:
                    break
                add = min(r, slack[i])
                x[i] += add
                r -= add
        else:
            order = np.argsort(x, kind='stable')
            for i in order:
                if r >= 0:
                    break
                sub = min(-r, x[i])
                x[i] -= sub
                r += sub
    return x

def smooth_max(R, beta):
    """Log-sum-exp smoothing of max."""
    m = R.max()
    if beta < 1e-10:
        return m, np.full(len(R), 1.0/len(R))
    exp_b = np.exp(beta * (R - m))
    log_sum = m + np.log(exp_b.sum()) / beta
    w = exp_b / exp_b.sum()
    return log_sum, w

def gradient_smooth(h, w):
    """Gradient of smoothed max w.r.t. h, given weights w."""
    n = len(h)
    lags = np.arange(2*n - 1) - (n - 1)
    
    P = np.arange(n)[:, None]
    L = lags[None, :]
    Q1 = P - L
    Q2 = P + L
    valid1 = (Q1 >= 0) & (Q1 < n)
    valid2 = (Q2 >= 0) & (Q2 < n)
    
    h_q1 = np.where(valid1, h[np.clip(Q1, 0, n-1)], 0.0)
    h_q2 = np.where(valid2, h[np.clip(Q2, 0, n-1)], 0.0)
    
    M = (1.0 - h_q1) * valid1.astype(float) - h_q2 * valid2.astype(float)
    return M.T @ w

def projected_gradient_descent(h_init, target, beta_schedule, n_iters, step_scale):
    """Run projected gradient descent with LSE smoothing."""
    n = len(h_init)
    h = h_init.copy()
    h = project_box_hyperplane(h, target)
    
    best_h = h.copy()
    best_val = compute_c5(h)
    
    for it in range(n_iters):
        beta = beta_schedule(it)
        R = compute_correlation(h)
        _, w = smooth_max(R, beta)
        grad = gradient_smooth(h, w)
        
        # Step size
        step = step_scale / (beta + 1.0)
        
        # Gradient step
        y = h - step * grad
        h = project_box_hyperplane(y, target)
        
        # Track best
        val = compute_c5(h)
        if val < best_val:
            best_val = val
            best_h = h.copy()
    
    return best_h, best_val

import numpy as np
import time
from scipy.optimize import minimize

def compute_r(h):
    return np.correlate(h, 1.0 - h, mode="full")

def compute_c5(h):
    n = len(h)
    return np.max(compute_r(h)) * 2.0 / n

def project_box(y, target):
    n = len(y)
    target = float(np.clip(target, 0.0, float(n)))
    lo = float(y.min()) - 1.0
    hi = float(y.max()) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.clip(y - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - 0.5*(lo+hi), 0.0, 1.0)
    r = target - float(x.sum())
    if r > 1e-12:
        order = np.argsort(1.0 - x)
        for i in order:
            if r <= 0: break
            a = min(r, 1.0 - x[i])
            x[i] += a
            r -= a
    elif r < -1e-12:
        order = np.argsort(x)
        for i in order:
            if r >= 0: break
            a = min(-r, x[i])
            x[i] -= a
            r += a
    return x

def lse_and_weights(R, beta):
    m = R.max()
    if beta < 1e-6:
        return m, np.ones(len(R)) / len(R)
    e = np.exp(beta * (R - m))
    s = e.sum()
    return m + np.log(s) / beta, e / s

def grad_lse(h, w):
    n = len(h)
    lags = np.arange(2*n-1) - (n-1)
    P = np.arange(n)[:, None]
    L = lags[None, :]
    q1 = P - L
    q2 = P + L
    v1 = (q1 >= 0) & (q1 < n)
    v2 = (q2 >= 0) & (q2 < n)
    hq1 = np.where(v1, h[np.clip(q1, 0, n-1)], 0.0)
    hq2 = np.where(v2, h[np.clip(q2, 0, n-1)], 0.0)
    M = (1.0 - hq1) * v1 - hq2 * v2
    return M.T @ w

def optimize_continuous(h_init, budget_s, start_time):
    n = len(h_init)
    target = n / 2.0
    h = project_box(h_init.copy(), target)
    best_h = h.copy()
    best_val = compute_c5(h)
    
    # Projected gradient with decreasing step and increasing beta
    beta0 = 50.0
    beta1 = 500.0
    total_iters = 3000
    base_step = 0.5
    
    for it in range(total_iters):
        if time.time() - start_time > budget_s * 0.4:
            break
        frac = it / total_iters
        beta = beta0 + (beta1 - beta0) * frac
        step = base_step / (1.0 + 0.1 * it)
        
        R = compute_r(h)
        _, w = lse_and_weights(R, beta)
        g = grad_lse(h, w)
        
        # Normalize gradient
        gn = np.linalg.norm(g)
        if gn > 1e-15:
            g = g / gn * min(gn, 1.0)
        
        y = h - step * g
        h = project_box(y, target)
        
        val = compute_c5(h)
        if val < best_val:
            best_val = val
            best_h = h.copy()
    
    return best_h, best_val
