from scipy.optimize import minimize, line_search
import time

def project_feasible(h, n):
    """Project h onto {0 <= h <= 1, sum(h) = n/2}"""
    target = n / 2.0
    # Binary search on shift t such that sum(clip(h - t, 0, 1)) = target
    lo, hi = h.min() - 1, h.max() + 1
    for _ in range(100):
        mid = (lo + hi) / 2
        projected = np.clip(h - mid, 0, 1)
        s = projected.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

def compute_correlations(h):
    """Compute all correlation values R(L) for L = -(n-1), ..., (n-1)"""
    b = 1.0 - h
    corr = np.correlate(h, b, mode='full')
    return corr

def compute_logsumexp(corr, tau):
    """Compute tau * log(sum(exp(corr/tau))) and the weights"""
    M = corr.max()
    shifted = (corr - M) / tau
    exp_shifted = np.exp(shifted)
    log_sum = np.log(exp_shifted.sum())
    value = M + tau * log_sum
    weights = exp_shifted / exp_shifted.sum()
    return value, weights

def compute_gradient(h, weights):

    n = len(h)
    b = 1.0 - h
    # For each lag L, the contribution to the gradient
    # R(L) = sum_i h[i] * (1 - h[i+L])
    # dR(L)/dh[j] = (1-h[j-L]) * 1{0<=j-L<n} - h[j+L] * 1{0<=j+L<n}
    
    # The correlate output index k corresponds to lag L = k - (n-1)
    # So weights[k] corresponds to lag L = k - (n-1)
    
    grad = np.zeros(n)

        w = weights[k]

import numpy as np
from scipy.optimize import minimize
import time

def project_feasible(h, n):
    """Project h onto {0 <= x <= 1, sum(x) = n/2} using bisection on shift."""
    target = n / 2.0
    lo, hi = h.min() - 2.0, h.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if proj.sum() > target:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    proj = np.clip(h - t, 0.0, 1.0)
    # Fix any numerical issues
    diff = target - proj.sum()
    if abs(diff) > 1e-10:
        # Distribute the difference among elements not at bounds
        free = (proj > 1e-10) & (proj < 1 - 1e-10)
        if free.sum() > 0:
            proj[free] += diff / free.sum()
            proj = np.clip(proj, 0.0, 1.0)
    return proj

def evaluate_c5(h):
    """Compute the exact C5 value."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return (corr * dx).max()

def softmax_and_weights(corr, tau):
    """Compute softmax weights and log-sum-exp value."""
    M = corr.max()
    shifted = (corr - M) / tau
    exp_s = np.exp(shifted)
    w = exp_s / exp_s.sum()
    return M + tau * np.log(exp_s.sum()), w

def gradient_surrogate(h, weights):
    """Compute gradient of softmax surrogate w.r.t. h."""
    n = len(h)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        L = k - (n - 1)
        w = weights[k]
        if abs(w) < 1e-14:
            continue
        # dR(L)/dh[j] = (1-h[j-L])*1{valid} - h[j+L]*1{valid}
        js1 = max(0, L)
        je1 = min(n, n + L)
        if js1 < je1:
            grad[js1:je1] += w * (1.0 - h[js1 - L:je1 - L])
        js2 = max(0, -L)
        je2 = min(n, n - L)
        if js2 < je2:
            grad[js2:je2] -= w * h[js2 + L:je2 + L]
    return grad

def optimize_from_start(h0, n, tau_start, tau_end, n_iters, lr, seed=42):
    """Run gradient descent with softmax surrogate from h0."""
    np.random.seed(seed)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    for it in range(n_iters):
        # Anneal temperature
        frac = it / max(1, n_iters - 1)
        tau = tau_start * (tau_end / tau_start) ** frac
        
        # Compute correlations and weights
        corr = np.correlate(h, 1.0 - h, mode='full')
        _, weights = softmax_and_weights(corr, tau)
        
        # Compute gradient
        grad = gradient_surrogate(h, weights)
        
        # Line search with a few steps
        for lr_try in [lr, lr * 0.5, lr * 0.25, lr * 0.1]:
            h_new = h - lr_try * grad
            h_new = project_feasible(h_new, n)
            c5_new = evaluate_c5(h_new)
            if c5_new < best_c5:
                h = h_new
                best_c5 = c5_new
                best_h = h.copy()
                break
    
    return best_h, best_c5

def local_polish(h, n, n_rounds=50, seed=42):
    """Local optimization using pairwise mass transfers."""
    np.random.seed(seed)
    h = h.copy()
    best_c5 = evaluate_c5(h)
    
    for rnd in range(n_rounds):
        improved = False
        # Random pairs
        for _ in range(200):
            i, j = np.random.randint(0, n, 2)
            if i == j:
                continue
            # Try transferring small mass from j to i
            for delta in [0.001, 0.005, 0.01, 0.02, 0.05, -0.001, -0.005, -0.01, -0.02, -0.05]:
                if h[i] + delta < -1e-10 or h[i] + delta > 1 + 1e-10:
                    continue
                if h[j] - delta < -1e-10 or h[j] - delta > 1 + 1e-10:
                    continue
                h[i] += delta
                h[j] -= delta
                c5 = evaluate_c5(h)
                if c5 < best_c5 - 1e-12:
                    best_c5 = c5
                    improved = True
                else:
                    h[i] -= delta
                    h[j] += delta
        if not improved:
            break
    
    return h, best_c5

def make_random_feasible(n, seed=42):
    """Generate a random feasible vector."""
    np.random.seed(seed)
    h = np.random.rand(n)
    return project_feasible(h, n)

def make_block_feasible(n, n_blocks=4, seed=42):
    """Generate a block-structured feasible vector."""
    np.random.seed(seed)
    h = np.concatenate([np.full(n // n_blocks + (1 if i < n % n_blocks else 0), 
                                  np.random.rand()) for i in range(n_blocks)])
    return project_feasible(h[:n], n)

def make_smooth_feasible(n, seed=42):
    """Generate a smooth feasible vector using random Fourier features."""
    np.random.seed(seed)
    x = np.linspace(0, 2, n)
    h = np.zeros(n)
    for freq in [1, 2, 3]:
        h += np.random.rand() * np.sin(np.pi * freq * x + np.random.rand() * 2 * np.pi)
    h = project_feasible(h, n)
    return h

import numpy as np
import time

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = h.min() - 2.0, h.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if proj.sum() > target:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    proj = np.clip(h - t, 0.0, 1.0)
    # Fix residual
    diff = target - proj.sum()
    if abs(diff) > 1e-12:
        free = (proj > 1e-8) & (proj < 1.0 - 1e-8)
        if free.any():
            proj[free] += diff / free.sum()
    return np.clip(proj, 0.0, 1.0)

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float((corr * dx).max())

def softmax_weights(corr, tau):
    M = corr.max()
    s = (corr - M) / tau
    e = np.exp(s)
    return e / e.sum()

def gradient_surrogate(h, weights):
    n = len(h)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        L = k - (n - 1)
        w = weights[k]
        if abs(w) < 1e-14:
            continue
        js1 = max(0, L)
        je1 = min(n, n + L)
        if js1 < je1:
            grad[js1:je1] += w * (1.0 - h[js1 - L:je1 - L])
        js2 = max(0, -L)
        je2 = min(n, n - L)
        if js2 < je2:
            grad[js2:je2] -= w * h[js2 + L:je2 + L]
    return grad

def gradient_descent(h0, n, tau_start, tau_end, n_iters, lr, time_budget, start_time):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    for it in range(n_iters):
        if time.time() - start_time > time_budget:
            break
        frac = it / max(1, n_iters - 1)
        tau = tau_start * (tau_end / tau_start) ** frac
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        weights = softmax_weights(corr, tau)
        grad = gradient_surrogate(h, weights)
        
        accepted = False
        for lr_try in [lr, lr*0.5, lr*0.25, lr*0.1]:
            h_new = project_feasible(h - lr_try * grad, n)
            c5_new = evaluate_c5(h_new)
            if c5_new < best_c5 - 1e-12:
                h = h_new
                best_c5 = c5_new
                best_h = h.copy()
                accepted = True
                break
    
    return best_h, best_c5

def local_polish(h, n, n_rounds, seed, time_budget, start_time):
    rng = np.random.RandomState(seed)
    h = h.copy()
    best_c5 = evaluate_c5(h)
    
    for rnd in range(n_rounds):
        if time.time() - start_time > time_budget:
            break
        improved = False
        for _ in range(300):
            i, j = rng.randint(0, n, 2)
            if i == j:
                continue
            for delta in [0.01, 0.02, 0.05, 0.1, -0.01, -0.02, -0.05, -0.1]:
                if h[i] + delta < -1e-10 or h[i] + delta > 1 + 1e-10:
                    continue
                if h[j] - delta < -1e-10 or h[j] - delta > 1 + 1e-10:
                    continue
                h[i] += delta
                h[j] -= delta
                c5 = evaluate_c5(h)
                if c5 < best_c5 - 1e-12:
                    best_c5 = c5
                    improved = True
                else:
                    h[i] -= delta
                    h[j] += delta
        if not improved:
            break
    return h, best_c5

def make_sinusoidal(n, freq=1.0, amp=1.0, phase=0.0):
    x = np.arange(n)
    h = 0.5 + 0.5 * amp * np.cos(2.0 * np.pi * freq * x / n + phase)
    return project_feasible(h, n)

def make_random_feasible(n, seed):
    rng = np.random.RandomState(seed)
    h = rng.rand(n)
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):

            best_c5 = c5
