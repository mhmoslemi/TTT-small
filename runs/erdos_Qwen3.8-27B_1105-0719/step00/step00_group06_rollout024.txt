def compute_profile(h):

    return compute_profile(h).max()

    n = len(h)
    dx = 2.0 / n
    profile = compute_profile(h)
    pmax = profile.max()

    e = np.exp((profile - pmax) / T)
    w = e / e.sum()
    # Gradient

        np.clip(h, 0.0, 1.0, out=h)

        grad = smooth_grad(h, T)
