def smooth_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode="full")
    c_max = np.max(corr)
    exp_terms = np.exp((corr - c_max) / tau)
    weights = exp_terms / exp_terms.sum()
    
    S = dx * (c_max + tau * np.log(exp_terms.sum()))
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if weights[k] < 1e-14:
            continue
        w = weights[k] * dx
        p = k - (n - 1)
        if p >= 0:
            # j ranges from p to n-1 for first term (j-p from 0 to n-1-p)
            # j ranges from 0 to n-1-p for second term (j+p from p to n-1)
            grad[p:] += w * (1.0 - h[:n-p])
            grad[:n-p] -= w * h[p:]
        else:
            # p < 0, let q = -p > 0
            q = -p
            # First term: j-p = j+q, valid when 0<=j+q<n, i.e., j from 0 to n-1-q
            grad[:n-q] += w * (1.0 - h[q:])
            # Second term: j+p = j-q, valid when 0<=j-q<n, i.e., j from q to n-1
            grad[q:] -= w * h[:n-q]
    return S, grad

import numpy as np

def project_box_sum(x, target_sum):
    """Project x onto {y: 0<=y_i<=1, sum(y)=target_sum} via bisection on lambda."""
    n = len(x)
    x = np.asarray(x, dtype=float).copy()
    lo = x.min() - 1.0
    hi = x.max() + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    result = np.clip(x - lam, 0.0, 1.0)
    residual = target_sum - result.sum()
    if abs(residual) > 1e-13:
        interior = (result > 1e-12) & (result < 1.0 - 1e-12)
        if interior.sum() > 0:
            result[interior] += residual / interior.sum()
        else:
            if residual > 0:
                idx = np.argsort(1.0 - result)
                for i in idx:
                    add = min(residual, 1.0 - result[i])
                    result[i] += add
                    residual -= add
                    if residual <= 1e-15:
                        break
            else:
                idx = np.argsort(result)
                for i in idx:
                    sub = min(-residual, result[i])
                    result[i] -= sub
                    residual += sub
                    if residual >= -1e-15:
                        break
    return np.clip(result, 0.0, 1.0)

    return float(np.max(corr) * dx)

    if max_t <= 1e-15:
        return 0.0

    hi_t = max_t

def pair_descent_pass(h, max_passes_inner=3):
    """One or more passes of pair descent with exact 1D optimization."""
    n = len(h)
    dx = 2.0 / n
    target_sum = n / 2.0
    current_c = exact_score(h)

                    h[i] -= t_star
                    h[j] += t_star
                    new_c = exact_score(h)
                    if new_c < current_c - 1e-14:
                        current_c = new_c
                        improved = True
                        derivs[i] *= 0  # invalidate (will recompute)
                        derivs[j] *= 0
                        # Quick update of derivatives for changed rows
                        derivs = compute_derivatives(h) * dx
                    else:
                        # Revert
                        h[i] += t_star
                        h[j] -= t_star
            # Direction j -> i
            max_t_ji = min(h[j], 1.0 - h[i])
            if max_t_ji > 1e-10:
                V = compute_corr(h) * dx
                delta = (derivs[i] - derivs[j])
                t_star = optimal_transfer(V, delta, max_t_ji)
                if t_star > 1e-10:
                    h[j] -= t_star
                    h[i] += t_star
                    new_c = exact_score(h)
                    if new_c < current_c - 1e-14:
                        current_c = new_c
                        improved = True
                        derivs = compute_derivatives(h) * dx
                    else:
                        h[j] += t_star
                        h[i] -= t_star
    return h, current_c

def smooth_gradient_step(h, tau, target_sum):
    """One step of smooth projected gradient descent. Returns (new_h, accepted)."""
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)  # raw, length 2n-1
    V = corr * dx  # scaled values
    cmax = V.max()
    exp_terms = np.exp((V - cmax) / tau)
    weights = exp_terms / exp_terms.sum()
    
    # Gradient of smooth objective w.r.t. h
    # dS/dh[j] = sum_k w_k * dx * dC_k/dh[j]
    derivs = compute_derivatives(h) * dx  # shape (n, 2n-1)
    grad = (weights[None, :] * derivs).sum(axis=1)  # shape (n,)
    
    # Project gradient onto sum-zero subspace (tangent to constraint)
    grad_tangent = grad - grad.mean()
    
    # Line search
    alpha = 0.05
    cur_c = exact_score(h)
    for _ in range(20):
        cand = h - alpha * grad_tangent
        cand = project_box_sum(cand, target_sum)
        new_c = exact_score(cand)
        if new_c < cur_c - 1e-12:
            return cand, True
        alpha *= 0.5
    return h, False

def coord_descent_pass(h, target_sum):
    """One pass of coordinate descent using exact 1D optimization per coordinate.
    Uses pair-transfer to maintain feasibility: for each j, find best transfer
    to/from a 'pivot' coordinate."""
    n = len(h)
    dx = 2.0 / n
    current_c = exact_score(h)
    # Use the first coordinate as pivot for sum preservation
    # Actually, use a better approach: for each pair (j, pivot), do optimal transfer
    # Instead, do: for each j, try to set h[j] to optimal value by transferring
    # to/from all other coordinates proportionally.
    # Simpler: just use the pair descent which is already exact.
    # For coordinate descent: vary h[j] and compensate by adjusting ONE other coordinate.
    # Best choice: adjust the coordinate with most "room" (closest to 0.5).
    for j in range(n):
        # Find best pivot
        best_pivot = -1
        best_pivot_c = current_c
        for pivot in range(n):
            if pivot == j:
                continue
            # Try transfer from j to pivot
            max_t = min(h[j], 1.0 - h[pivot])
            if max_t > 1e-10:
                V = compute_corr(h) * dx
                delta = -V  # placeholder, compute properly
                # Use exact derivative
                derivs_j = compute_derivatives(h)[j] * dx
                derivs_p = compute_derivatives(h)[pivot] * dx
                # Transfer from j to pivot: h[j]-=t, h[pivot]+=t
                # Change in V_k = t*(derivs_pivot[k] - derivs_j[k])
                delta_jp = derivs_p - derivs_j
                t_star = optimal_transfer(V, delta_jp, max_t)
                if t_star > 1e-10:
                    h[j] -= t_star
                    h[pivot] += t_star
                    c = exact_score(h)
                    if c < best_pivot_c - 1e-14:
                        best_pivot_c = c
                        best_pivot = (pivot, t_star, True)
                    h[j] += t_star
                    h[pivot] -= t_star
            # Try transfer from pivot to j
            max_t2 = min(h[pivot], 1.0 - h[j])
            if max_t2 > 1e-10:
                V = compute_corr(h) * dx
                derivs_j = compute_derivatives(h)[j] * dx
                derivs_p = compute_derivatives(h)[pivot] * dx
                delta_pj = derivs_j - derivs_p
                t_star = optimal_transfer(V, delta_pj, max_t2)
                if t_star > 1e-10:
                    h[pivot] -= t_star
                    h[j] += t_star
                    c = exact_score(h)
                    if c < best_pivot_c - 1e-14:
                        best_pivot_c = c
                        best_pivot = (pivot, t_star, False)
                    h[pivot] += t_star
                    h[j] -= t_star
        if best_pivot is not None:
            h[j] is not None  # noop
            pivot, t_star, from_j = best_pivot
            if from_j:
                h[j] -= t_star
                h[pivot] += t_star
            else:
                h[pivot] -= t_star
                h[j] += t_star
            current_c = best_pivot_c
    return h, current_c

def sa_pair_step(h, current_c, temp, rng, target_sum):
    """One SA step using random pair transfer. Returns (h, new_c, accepted)."""
    n = len(h)
    i = rng.integers(n)
    j = rng.integers(n - 1)
    if j >= i:
        j += 1
    # Determine direction and max transfer
    if rng.random() < 0.5:
        max_t = min(h[i], 1.0 - h[j])
        reverse = False
    else:
        max_t = min(h[j], 1.0 - h[i])
        reverse = True
    if max_t <= 1e-12:
        return h, current_c, False
    # Random fraction
    frac = rng.random() ** 0.5  #偏向 smaller moves
    t = frac * max_t
    if t < 1e-12:
        return h, current_c, False
    if not reverse:
        h[i] -= t
        h[j] += t
    else:
        h[j] -= t
        h[i] += t
    
    new_c = exact_score(h)
    # SA acceptance
    if new_c <= current_c:
        return h, new_c, True
    delta = new_c - current_c
    if delta < temp:
        prob = np.exp(-delta / temp)
        if rng.random() < prob:
            return h, new_c, True
    # Revert
    if not reverse:
        h[i] += t
        h[j] -= t
    else:
        h[j] += t
        h[i] -= t
    return h, current_c, False

def make_fallback(n=47, seed=42):
    """Generate a feasible fallback vector."""
    rng = np.random.default_rng(seed)
    x = np.full(n, 0.5)
    x += 0.15 * np.sin(2 * np.pi * np.arange(n) / n)
    x += 0.05 * np.cos(4 * np.pi * np.arange(n) / n)
    return project_box_sum(x, n / 2.0)

def interpolate_to_resolution(h_src, n_src, n_dst):
    """Interpolate from n_src to n_dst points on [0,2]."""
    x_src = np.linspace(0.0, 2.0, n_src)
    x_dst = np.linspace(0.0, 2.0, n_dst)
    h_dst = np.interp(x_dst, x_src, h_src)
    return project_box_sum(h_dst, n_dst / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    deadline = t0 + budget_s * 0.98  # leave 2% safety margin
    
    # Get parent
    if 'initial_h_values' in globals() and initial_h_values is not None:
        parent = np.array(initial_h_values, dtype=float).copy()
    else:
        parent = make_fallback(47, seed)
    
    n = len(parent)
    target_sum = n / 2.0
    parent = project_box_sum(parent, target_sum)
    
    best_h = parent.copy()
    best_c = exact_score(best_h)
    
    # No improvement tracking
    no_improve_time = t0
    
    # Phase 1: Deterministic pair descent (most effective for polishing)
    h_curr = best_h.copy()
    phase1_budget = min(budget_s * 0.40, 440.0)
    t_phase1_end = t0 + phase1_budget
    
    while time.time() < t_phase1_end:
        h_curr, c_curr = pair_descent_pass(h_curr, max_passes_inner=2)
        if c_curr < best_c - 1e-13:
            best_c = c_curr
            best_h = h_curr.copy()
            no_improve_time = time.time()
        # Check if stuck
        if time.time() - no_improve_time > 60:
            # Try smooth gradient to escape
            h_curr = best_h.copy()
            tau = max(0.005, 0.3 * best_c)
            for _ in range(50):
                if time.time() > t_phase1_end:
                    break
                h_curr, ok = smooth_gradient_step(h_curr, tau, target_sum)
                if not ok:
                    break
            h_curr, c_curr = pair_descent_pass(h_curr, max_passes_inner=2)
            if c_curr < best_c - 1e-13:
                best_c = c_curr
                best_h = h_curr.copy()
                no_improve_time = time.time()
            else:
                h_curr = best_h.copy()
                no_improve_time = time.time()
    
    # Phase 2: Simulated annealing with pair moves
    t_phase2_end = min(t0 + budget_s * 0.70, time.time() + 200)
    h_sa = best_h.copy()
    temp = max(0.001, 0.1 * best_c)
    sa_epoch = 0
    while time.time() < t_phase2_end:
        epoch_best = best_c
        for _ in range(500):
            if time.time() > t_phase2_end:
                break
            h_sa, c_sa, accepted = sa_pair_step(h_sa, best_c, temp, rng, target_sum)
            if accepted and c_sa < best_c - 1e-13:
                best_c = c_sa
                best_h = h_sa.copy()
                no_improve_time = time.time()
            elif accepted:
                pass  # keep h_sa for exploration
            else:
                pass
        if best_c < epoch_best - 1e-13:
            no_improve_time = time.time()
        temp *= 0.85
        if temp < 1e-5:
            temp = 1e-5
        sa_epoch += 1
        if sa_epoch % 10 == 0 and time.time() - no_improve_time > 120:
            # Restart from best
            h_sa = best_h.copy()
            temp = max(0.005, 0.15 * best_c)
            no_improve_time = time.time()
    
    # Phase 3: Final polishing with pair descent
    t_phase3_end = time.time() + min(120, max(0, deadline - time.time() - 5))
    h_final = best_h.copy()
    while time.time() < t_phase3_end:
        h_final, c_final = pair_descent_pass(h_final, max_passes_inner=3)
        if c_final < best_c - 1e-13:
            best_c = c_final
            best_h = h_final.copy()
        else:
            break
    
    # Phase 4: Try other resolutions (if time permits)
    remaining = deadline - time.time()
    if remaining > 60:
        for n_try in [40, 44, 50, 55, 60, 48, 46]:
            if time.time() > deadline - 10:
                break
            n_try = int(n_try)
            if n_try <= 4:
                continue
            h_try = interpolate_to_resolution(best_h, len(best_h), n_try)
            # Quick optimization
            t_sub_end = min(time.time() + remaining * 0.3, deadline - 5)
            while time.time() < t_sub_end:
                h_try, c_try = pair_descent_pass(h_try, max_passes_inner=2)
                if time.time() < t_sub_end - 5:
                    continue
                else:
                    break
            c_try = exact_score(h_try)
            if c_try < best_c - 1e-13:
                best_c = c_try
                best_h = h_try.copy()
    
    # Final validation
    best_h = np.asarray(best_h, dtype=float).copy()
    n_final = len(best_h)
    # Ensure feasibility
    best_h = np.clip(best_h, 0.0, 1.0)
    s = best_h.sum()
    if abs(s - n_final / 2.0) > 1e-10:
        best_h = project_box_sum(best_h, n_final / 2.0)
    
    # Re-evaluate
    final_c = exact_score(best_h)
    # Double-check sum
    if abs(best_h.sum() - n_final / 2.0) > 1e-8:
        best_h = project_box_sum(best_h, n_final / 2.0)
        final_c = exact_score(best_h)
    
    return best_h, final_c, n_final
