    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    # Softmax weights
    max_corr = np.max(corr)
    weights = np.exp(beta * (corr - max_corr))
    probs = weights / weights.sum()
    
    # grad[j] = dx * sum_d probs[d] * df_d/dh[j]
    # f_d = sum_{i in range_d} h[i] * (1 - h[i + d])
    # where d = n-1-k, k is the correlation index

    for d in range(-(n-1), n):
        k = n - 1 - d  # correlation index
        p = probs[k]
        if p < 1e-15:
            continue

        # valid i: max(0,-d) <= i <= min(n-1, n-1-d)
