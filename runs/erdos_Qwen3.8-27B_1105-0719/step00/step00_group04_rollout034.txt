import numpy as np
import time
from scipy.optimize import minimize as _scipy_minimize


def _exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)


def _project(v, target):
    n = len(v)
    lo_l, hi_l = -2.0, 2.0
    for _ in range(80):
        mid = (lo_l + hi_l) * 0.5
        s = np.sum(np.clip(v + mid, 0.0, 1.0))
        if s < target:
            lo_l = mid
        else:
            hi_l = mid
    r = np.clip(v + (lo_l + hi_l) * 0.5, 0.0, 1.0)
    diff = target - np.sum(r)
    if abs(diff) > 1e-10:
        interior = (r > 1e-8) & (r < 1.0 - 1e-8)
        cnt = int(np.sum(interior))
        if cnt > 0:
            r[interior] = np.clip(r[interior] + diff / cnt, 0.0, 1.0)
        diff = target - np.sum(r)
        for i in range(n):
            if abs(diff) < 1e-12:
                break
            if diff > 0.0 and r[i] < 1.0:
                adj = min(diff, 1.0 - r[i])
                r[i] += adj
                diff -= adj
            elif diff < 0.0 and r[i] > 0.0:
                adj = max(diff, -r[i])
                r[i] += adj
                diff -= adj
    return r


def _pair_best_a(h, p, q):
    s = h[p] + h[q]
    lo = max(0.0, s - 1.0)
    hi = min(1.0, s)
    if hi - lo < 1e-12:
        return None
    hp, hq = h[p], h[q]
    h[p] = 0.0
    h[q] = s
    p0 = np.correlate(h, 1.0 - h, mode="full")
    h[p] = 0.5
    h[q] = s - 0.5
    pm = np.correlate(h, 1.0 - h, mode="full")
    h[p] = 1.0
    h[q] = s - 1.0
    p1 = np.correlate(h, 1.0 - h, mode="full")
    h[p], h[q] = hp, hq
    dx = 2.0 / len(h)
    p0 = p0 * dx
    pm = pm * dx
    p1 = p1 * dx
    alpha = 2.0 * (p0 + p1 - 2.0 * pm)
    beta = 4.0 * pm - 3.0 * p0 - p1
    gamma = p0
    n_g = 60
    grid = np.linspace(lo, hi, n_g)
    a2 = grid * grid
    vals = alpha[None, :] * a2[:, None] + beta[None, :] * grid[:, None] + gamma[None, :]
    mv = np.max(vals, axis=1)
    bi = int(np.argmin(mv))
    i0 = max(0, bi - 3)
    i1 = min(n_g - 1, bi + 3)
    fine = np.linspace(grid[i0], grid[i1], 150)
    a2f = fine * fine
    vf = alpha[None, :] * a2f[:, None] + beta[None, :] * fine[:, None] + gamma[None, :]
    mfv = np.max(vf, axis=1)
    fbi = int(np.argmin(mfv))
    return float(fine[fbi])


def _pair_descent(h, seed, t_deadline):
    n = len(h)
    rng = np.random.default_rng(seed)
    c = _exact_score(h)
    all_pairs = [(p, q) for p in range(n) for q in range(p + 1, n)]
    npairs = len(all_pairs)
    for _ in range(2000):
        if time.perf_counter() > t_deadline:
            break
        order = rng.permutation(npairs)
        improved = False
        for idx in order:
            p, q = all_pairs[idx]
            s = h[p] + h[q]
            lo = max(0.0, s - 1.0)
            hi = min(1.0, s)
            if hi - lo < 1e-10:
                continue
            hp, hq = h[p], h[q]
            best_a = _pair_best_a(h, p, q)
            h[p], h[q] = hp, hq
            if best_a is None:
                continue
            h[p] = best_a
            h[q] = s - best_a
            c_new = _exact_score(h)
            if c_new < c - 1e-13:
                c = c_new
                improved = True
            else:
                h[p], h[q] = hp, hq
        if not improved:
            break
    return h, c


def _pen_obj(x, n, target):
    return _exact_score(x) + 50.0 * (np.sum(x) - target) ** 2


def _pen_grad(x, n, target):
    dx_val = 2.0 / n
    prof = np.correlate(x, 1.0 - x, mode="full")
    k = int(np.argmax(prof))
    j = k - (n - 1)
    g = np.zeros(n)
    if j == 0:

            if k2 >= m:
                g[k2] += -x[k2 - m] * dx_val
    g += 100.0 * (np.sum(x) - target) * np.ones(n)
    return g

            c_new = _exact_score(h_new)

    return None

    xs = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    xt = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    h_new = np.interp(xt, xs, h_src, left=h_src[0], right=h_src[-1])
    return _project(h_new, n_dst / 2.0)
