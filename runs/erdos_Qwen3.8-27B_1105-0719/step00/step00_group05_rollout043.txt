    dx = 2.0 / n

def subgradient_c5(h):
    """Compute subgradient of C5 with respect to h."""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode="full")
    max_val = np.max(c)
    # Active lags
    active = c >= max_val - 1e-10
    # Subgradient
    grad = np.zeros(n)
    for k in range(len(c)):
        if not active[k]:

        lag = k - n + 1
        for i in range(n):

            j = i
            # Term 1: h[j] * (1 - h[j+lag])

def compute_autocorr(h):

    n = len(h)
    dx = 2.0 / n
    # Autocorrelation
    R = np.correlate(h, h, mode="full") * dx

import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode="full")
    return float(np.max(c) * dx)

def project_onto_constraint(h, target_sum):
    lo, hi = h.min() - 2.0, h.max() + 2.0
    for _ in range(200):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h - mid, 0, 1))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

def get_G(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def smooth_max_and_grad(h, tau):
    G = get_G(h)
    G_max = G.max()
    G_shift = tau * (G - G_max)
    unnorm = np.exp(G_shift)
    w = unnorm / unnorm.sum()
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        lag = k - (n - 1)
        if lag == 0:
            grad += wk * (1.0 - 2.0 * h) * dx
        elif lag > 0:
            grad[lag:] += wk * (1.0 - h[:n - lag]) * dx
            grad[:n - lag] += wk * (-h[lag:]) * dx
        else:
            m = -lag
            grad[m:] += wk * (1.0 - h[:n - m]) * dx
            grad[:n - m] += wk * (-h[m:]) * dx
    obj = G_max + np.log(unnorm.sum()) / tau
    return obj, grad

def optimize_subgrad(h0, n_iters, tau_start, tau_end, step_init, step_decay,
                     time_limit=None, seed=0):
    n = len(h0)
    target_sum = n / 2.0
    rng = np.random.RandomState(seed)
    h = project_onto_constraint(h0.copy(), target_sum)
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    h_mom = np.zeros(n)
    start_t = time.time()
    for i in range(n_iters):
        if time_limit is not None and (time.time() - start_t) > time_limit:
            break
        frac = i / max(n_iters - 1, 1)
        tau = tau_start * (tau_end / tau_start) ** frac
        obj, grad = smooth_max_and_grad(h, tau)
        step = step_init / (1.0 + step_decay * i)
        h_mom = 0.8 * h_mom - step * grad
        h_new = project_onto_constraint(h + h_mom, target_sum)
        c5 = evaluate_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
        h = h_new
        if i % 1000 == 999:
            noise = rng.normal(0, 0.05, n)
            h_noise = project_onto_constraint(h + noise, target_sum)
            c5_noise = evaluate_c5(h_noise)
            if c5_noise < best_c5:
                best_c5 = c5_noise
                best_h = h_noise.copy()
                h = h_noise
                h_mom *= 0.1
    return best_h, best_c5

def random_h(n, seed):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    h[:n // 2] = 1.0
    rng.shuffle(h)
    h += rng.uniform(-0.2, 0.2, n)
    return project_onto_constraint(h, n / 2.0)

def try_scipy(h0, maxiter=300):
    n = len(h0)
    target_sum = n / 2.0
    tau = 50.0

    def objective(h):
        obj, _ = smooth_max_and_grad(h, tau)
        return obj

    def grad_func(h):
        _, g = smooth_max_and_grad(h, tau)
        return g

    equality = {'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum,
                'jac': lambda h: np.ones(n)}
    bounds = [(0.0, 1.0)] * n
    try:
        result = minimize(objective, h0.copy(), method='SLSQP', jac=grad_func,
                          bounds=bounds, constraints=equality,
                          options={'maxiter': maxiter, 'ftol': 1e-12})
        h_opt = project_onto_constraint(result.x, target_sum)
        if evaluate_c5(h_opt) < evaluate_c5(h0):
            return h_opt
    except Exception:
        pass
    return h0
