```python
def smooth_max_and_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    
    max_c = np.max(corr)
    exps = np.exp(beta * (corr - max_c))
    sum_exp = np.sum(exps)
    p = exps / sum_exp
    Q = max_c + np.log(sum_exp) / beta
    
    g = np.zeros(n)
    
    # d = 0 case (i = n-1)
    g += p[n-1] * (1.0 - 2.0 * h)
    
    # Positive lags d = 1 to n-1 (i = n-1+d)
    for d in range(1, n):
        i = n - 1 + d
        w = p[i]
        if w < 1e-16:
            continue
        # corr[i] = sum_{n=0}^{n-1-d} h[n+d](1-h[n])
        # gradient:
        # g[n+d] += w*(1-h[n]), g[n] -= w*h[n+d] for n=0..n-1-d
        g[d:] += w * (1.0 - h[:-d])  # h[n+d] gets (1-h[n])
        g[:-d] -= w * h[d:]  # h[n] gets -h[n+d]
    
    # Negative lags d = -1 to -(n-1) (i = n-1+d = n-1-d')
    for d in range(1, n):
        i = n - 1 - d
        w = p[i]
        if w < 1e-16:
            continue
        # corr[i] = sum_{n=0}^{n-1-d} h[n](1-h[n+d])
        # gradient:
        # g[n] += w*(1-h[n+d]), g[n+d] -= w*h[n] for n=0..n-1-d
        g[:-d] += w * (1.0 - h[d:])  # h[n] gets (1-h[n+d])
        g[d:] -= w * h[:-d]  # h[n+d] gets -h[n]
    
    g *= dx
    return Q, g
```

```python
def project_feasible(z, n):
    target = n / 2.0
    lo = -1.0 - np.max(z)
    hi = 1.0 - np.min(z)

            lo = mid
        else:
            hi = mid

    # Fix residual
    residual = target - np.sum(h)

        cnt = np.sum(mask)
```

```python
def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(z, n):
    target = n * 0.5
    lo = -1.0 - z.max()
    hi = 1.0 - z.min()
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(z + mid, 0, 1).sum() < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + 0.5*(lo+hi), 0.0, 1.0)
    r = target - h.sum()
    if abs(r) > 1e-13:
        m = (h > 1e-12) & (h < 1.0 - 1e-12)
        if m.any():
            h[m] += r / m.sum()
            h = np.clip(h, 0, 1)

def smooth_max_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    mc = c.max()
    e = np.exp(beta * (c - mc))
    se = e.sum()
    p = e / se
    Q = mc + np.log(se) / beta
```
