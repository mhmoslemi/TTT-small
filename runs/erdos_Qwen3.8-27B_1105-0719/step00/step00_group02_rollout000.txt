def project_to_k(vec, mass):
    """Project vector to the polytope: sum=mass, 0<=x<=1"""
    x = np.clip(vec, 0.0, 1.0)
    # Bisection for lambda such that sum(clip(y - lambda, 0, 1)) = mass
    def f(lam):
        return np.sum(np.clip(vec - lam, 0.0, 1.0)) - mass
    lo, hi = -2.0, 3.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        if f(mid) > 0:
            lo = mid
        else:
            hi = mid
    x = np.clip(vec - (lo + hi) / 2.0, 0.0, 1.0)
    # Repair residual
    residual = mass - np.sum(x)
    for _ in range(5):
        if abs(residual) < 1e-12:
            break
        if residual > 0:
            cap = np.maximum(0.0, 1.0 - x)
            total_cap = np.sum(cap)
            if total_cap > 1e-15:
                x += residual * cap / total_cap
            else:
                break
        else:

            total_aff = np.sum(afford)
            if total_aff > 1e-15:
                x += residual * afford / total_aff
            else:
                break
        residual = mass - np.sum(x)
    return x

def compute_q_coefficients(h, u, n):

    q1 = np.correlate(u, g, mode="full") - np.correlate(h, u, mode="full")
    q2 = -np.correlate(u, u, mode="full")
    return q0, q1, q2

def feasible_interval(h, u):
    """Compute [t_low, t_high] such that 0 <= h + t*u <= 1 for t in interval"""
    eps = 1e-14
    t_low = 0.0
    t_high = 0.0

        if u[i] > 0:

            t_high = min(t_high, (1.0 - h[i]) / u[i])

            t_high = min(t_high, -h[i] / u[i])
            t_low = max(t_low, (1.0 - h[i]) / u[i])

    t_high = max(t_high, 0.0)

            new_h = project_to_k(new_h, mass)
            new_c = exact_score(new_h, n)

            if new_c < current_c - 1e-10:
                current_h = new_h
                current_c = new_c
                improved = True

            if new_c < best_c - 1e-10:
                best_h = new_h.copy()
                best_c = new_c
                stagnation = 0

        if not improved:
            stagnation += 1
        else:
            stagnation = max(0, stagnation - 2)

        # Diversification when stuck
        if stagnation >= 30 and time_left() > 20:
            u = dir_random(n, rng)
            u -= np.mean(u)
            u = u / np.linalg.norm(u)
            t_low, t_high = feasible_interval(current_h, u)
            if t_high - t_low > 1e-8:
                t_mag = 0.15 * min(abs(t_low), abs(t_high))
                t_rand = t_mag * (1.0 if rng.random() > 0.5 else -1.0)
                new_h = current_h + t_rand * u
                new_h = project_to_k(new_h, mass)
                new_c = exact_score(new_h, n)
                if new_c < best_c + 0.005:
                    current_h = new_h
                    current_c = new_c
            stagnation = 0

        # Periodic: try a completely fresh random restart mixed toward best
        if n_iter % 100 == 0 and time_left() > 30:
            u = best_h - current_h
            u -= np.mean(u)
            u_norm = np.linalg.norm(u)
            if u_norm > 1e-10:
                u = u / u_norm
                t_low, t_high = feasible_interval(current_h, u)
                if t_high - t_low > 1e-8:
                    q0, q1, q2 = compute_q(current_h, u)
                    t_best = solve_ray(q0, q1, q2, t_low, t_high, n_rand=10)
                    new_h = current_h + t_best * u
                    new_h = project_to_k(new_h, mass)
                    new_c = exact_score(new_h, n)
                    if new_c < best_c:
                        best_h = new_h.copy()
                        best_c = new_c
                    current_h = new_h
                    current_c = new_c

    # Final evaluation
    best_c = exact_score(best_h, n)
    return best_h, best_c, n
