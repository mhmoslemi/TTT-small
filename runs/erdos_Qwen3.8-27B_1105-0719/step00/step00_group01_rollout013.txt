import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return np.max(corr)

def get_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def smooth_obj(h, temp):
    corr = get_corr(h)
    m = np.max(corr)
    if temp < 1e-12:
        return m
    return m + temp * np.log(np.sum(np.exp((corr - m) / temp)))

def smooth_grad(h, temp):
    n = len(h)
    corr = get_corr(h)
    m = np.max(corr)
    if temp < 1e-12:
        # Subgradient of max
        idx = np.argmax(corr)
        weights = np.zeros(len(corr))
        weights[idx] = 1.0
    else:
        weights = np.exp((corr - m) / temp)
        weights /= weights.sum()
    return weighted_grad(h, weights)

def weighted_grad(h, weights):
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in range(len(weights)):
        w = weights[k]
        if abs(w) < 1e-15:
            continue
        m_lo = max(0, k - n + 1)
        m_hi = min(n - 1, k)
        for m in range(m_lo, m_hi + 1):
            j = m - k + n - 1
            grad[m] += w * dx * (1.0 - h[j])
            grad[j] += w * dx * (-h[m])
    return grad

def project_feasible(h, n):
    """Project to 0<=h<=1, sum(h)=n/2."""
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

    """Simple subgradient descent on exact C5."""
    best = h.copy()
    best_c5 = compute_c5(h)
    step = step0
    for it in range(num_steps):
        corr = get_corr(h)
        # Find top active correlations
        threshold = np.max(corr) - 1e-6
        active = np.where(corr >= threshold)[0]
        if len(active) == 0:
            active = [np.argmax(corr)]
        
        # Average subgradient
        grad = np.zeros(n)
        for k in active:
            m_lo = max(0, k - n + 1)
            m_hi = min(n - 1, k)
            for m in range(m_lo, m_hi + 1):
                j = m - k + n - 1
                grad[m] += dx * (1.0 - h[j])  # need dx
                grad[j] += dx * (-h[m])
        grad /= len(active)
        
        # Project to zero-sum
        grad -= np.mean(grad)
        
        h_new = h - step * grad
        h_new = project_feasible(h_new, n)
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best = h_new.copy()
            best_c5 = c5_new
            h = h_new
        else:
            h = h_new  # still accept for progress
        
        step = max(step * 0.98, step_min)
    
    return best, best_c5

def local_search(h, n, max_rounds=5, time_limit=None, start_time=None):
    """Local search: try moving mass between pairs."""
    best = h.copy()
    best_c5 = compute_c5(h)
    
    for round_num in range(max_rounds):
        if time_limit and start_time and time.time() - start_time > time_limit - 30:
            break
        
        improved = False

                # Try moving mass from i to j
                for delta in [0.01, 0.05, 0.1, 0.2, 0.5]:
                    if h[i] >= delta and h[j] <= 1.0 - delta:
                        h_try = h.copy()
                        h_try[i] -= delta
                        h_try[j] += delta
                        c5_try = compute_c5(h_try)
                        if c5_try < best_c5 - 1e-10:
                            best = h_try.copy()
                            best_c5 = c5_try
                            h = h_try
                            improved = True
                            break
        
        if not improved:
            break
    
    return best, best_c5

    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return np.max(corr)

def get_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def smooth_obj_val(h, temp):
    corr = get_corr(h)
    m = np.max(corr)
    if temp < 1e-12:
        return m
    return m + temp * np.log(np.sum(np.exp((corr - m) / temp)))

def weighted_grad(h, weights):
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in range(len(weights)):
        w = weights[k]
        if abs(w) < 1e-14:
            continue
        m_lo = max(0, k - n + 1)
        m_hi = min(n - 1, k)
        for m in range(m_lo, m_hi + 1):
            j = m - k + n - 1
            grad[m] += w * dx * (1.0 - h[j])
            grad[j] -= w * dx * h[m]
    return grad

def smooth_grad(h, temp):
    corr = get_corr(h)
    m = np.max(corr)
    if temp < 1e-12:
        idx = np.argmax(corr)
        weights = np.zeros(len(corr))
        weights[idx] = 1.0
    else:
        weights = np.exp((corr - m) / temp)
        weights /= weights.sum()
    return weighted_grad(h, weights)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(20):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    # Final exact fix
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        if np.sum(mask) > 0:
            h[mask] += diff / np.sum(mask)
            h = np.clip(h, 0.0, 1.0)
    return h

def slsqp_optimize(h, n, temp, maxiter=150):
    def obj(x):
        return smooth_obj_val(x, temp)
    def jac(x):
        return smooth_grad(x, temp)
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, jac=jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        h_try = project_feasible(res.x, n)
        return h_try
    except Exception:
        return h

def subgradient_refine(h, n, num_steps=2000, step0=0.002):
    best = h.copy()
    best_c5 = compute_c5(h)
    step = step0
    for it in range(num_steps):
        corr = get_corr(h)
        max_val = np.max(corr)
        active = np.where(corr >= max_val - 0.001)[0]
        if len(active) == 0:
            active = np.array([np.argmax(corr)])
        n_active = len(active)
        weights = np.ones(len(corr)) * 0
        weights[active] = 1.0 / n_active
        grad = weighted_grad(h, weights)
        grad -= np.mean(grad)
        h_new = h - step * grad
        h_new = project_feasible(h_new, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-12:
            best = h_new.copy()
            best_c5 = c5_new
            h = h_new
        else:
            h = h_new
        step *= 0.995
        if step < 1e-7:
            break
    return best, best_c5

def local_search(h, n, time_limit, start_time):
    best = h.copy()
    best_c5 = compute_c5(h)
    deltas = [0.01, 0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5]
    rng = np.random.RandomState(12345)
    order = list(range(n))
    rounds = 0
    while time.time() - start_time < time_limit - 30 and rounds < 30:
        rounds += 1
        improved = False
        rng.shuffle(order)
        for i in order:
            for j in order:
                if i == j:
                    continue
                if time.time() - start_time > time_limit - 30:
                    break
                # Try moving from i to j
                for delta in deltas:
                    if best[i] >= delta and best[j] <= 1.0 - delta:
                        h_try = best.copy()
                        h_try[i] -= delta
                        h_try[j] += delta
                        c5_try = compute_c5(h_try)
                        if c5_try < best_c5 - 1e-10:
                            best = h_try
                            best_c5 = c5_try
                            improved = True
                            break
        if not improved:
            break
    return best, best_c5

def binary_local_search(h, n, time_limit, start_time):
    """Try pushing entries toward 0 or 1."""
    best = h.copy()
    best_c5 = compute_c5(h)
    rng = np.random.RandomState(99)
    for round_num in range(20):
        if time.time() - start_time > time_limit - 30:
            break
        improved = False
        for i in range(n):
            if time.time() - start_time > time_limit - 30:
                break
            for target_val in [0.0, 1.0]:
                delta = target_val - best[i]
                if abs(delta) < 1e-6:
                    continue
                # Need to compensate at some j
                h_try = best.copy()
                h_try[i] = target_val
                remaining = -delta
                # Distribute compensation
                candidates = [j for j in range(n) if j != i and
                             (remaining > 0 and h_try[j] < 1.0 - 1e-8) or
                             (remaining < 0 and h_try[j] > 1e-8)]
                if not candidates:
                    continue
                # Spread equally
                per = remaining / len(candidates)
                valid = []
                for j in candidates:
                    if remaining > 0 and h_try[j] + per <= 1.0:
                        valid.append(j)
                    elif remaining < 0 and h_try[j] + per >= 0.0:
                        valid.append(j)
                    else:
                        valid.append(j)
                if valid:
                    per2 = remaining / len(valid)
                    for j in valid:
                        h_try[j] += per2
                    h_try = np.clip(h_try, 0, 1)
                    h_try = project_feasible(h_try, n)
                    c5_try = compute_c5(h_try)
                    if c5_try < best_c5 - 1e-10:
                        best = h_try
                        best_c5 = c5_try
                        improved = True
                        break
        if not improved:
            break
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    best_h = None
    best_c5 = float('inf')
    best_n = 63

    def update_best(h, c5, n):
        nonlocal best_h, best_c5, best_n
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            best_n = n

    # Phase 1: Optimize parent construction (n=63)
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64).copy()
    else:
        h = np.ones(63) * 0.5
    n = len(h)
    h = project_feasible(h, n)
    c5 = compute_c5(h)
    update_best(h, c5, n)

    # SLSQP with decreasing temperature
    for temp in [0.05, 0.02, 0.01, 0.005, 0.002]:
        if time.time() - start_time > budget_s - 180:
            break
        h_try = slsqp_optimize(h, n, temp, maxiter=120)
        c5_try = compute_c5(h_try)
        if c5_try < c5:
            h = h_try
            c5 = c5_try
            update_best(h, c5, n)

    # Subgradient refinement
    h, c5 = subgradient_refine(h, n, num_steps=1500, step0=0.003)
    update_best(h, c5, n)

    # Local search (pair moves)
    h, c5 = local_search(h, n, budget_s - 120, start_time)
    update_best(h, c5, n)

    # Binary pushing
    h, c5 = binary_local_search(h, n, budget_s - 60, start_time)
    update_best(h, c5, n)

    # Phase 2: Try n=64 with interpolated initial condition
    if time.time() - start_time < budget_s - 90:
        n2 = 64
        if best_h is not None:
            # Interpolate from best solution
            x_old = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n2)
            h2 = np.interp(x_new, x_old, best_h)
        else:
            h2 = np.ones(n2) * 0.5
        h2 = project_feasible(h2, n2)
        c5_2 = compute_c5(h2)
        update_best(h2, c5_2, n2)

        for temp in [0.02, 0.01, 0.005]:
            if time.time() - start_time > budget_s - 60:
                break
            h2 = slsqp_optimize(h2, n2, temp, maxiter=100)
            c5_2 = compute_c5(h2)
            if c5_2 < best_c5:
                best_h = h2.copy()
                best_c5 = c5_2
                best_n = n2

        h2, c5_2 = subgradient_refine(h2, n2, num_steps=1000, step0=0.002)
        update_best(h2, c5_2, n2)

        if time.time() - start_time < budget_s - 60:
            h2, c5_2 = local_search(h2, n2, budget_s - 30, start_time)
            update_best(h2, c5_2, n2)

    # Phase 3: Try n=126 (up/sample)
    if time.time() - start_time < budget_s - 60:
        n3 = 126
        if best_h is not None:
            x_old = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n3)
            h3 = np.interp(x_new, x_old, best_h)
        else:
            h3 = np.ones(n3) * 0.5
        h3 = project_feasible(h3, n3)
        c5_3 = compute_c5(h3)

        for temp in [0.02, 0.01]:
            if time.time() - start_time > budget_s - 30:
                break
            h3 = slsqp_optimize(h3, n3, temp, maxiter=80)
            c5_3 = compute_c5(h3)
            if c5_3 < best_c5:
                best_h = h3.copy()
                best_c5 = c5_3
                best_n = n3

        if time.time() - start_time < budget_s - 30:
            h3, c5_3 = subgradient_refine(h3, n3, num_steps=800, step0=0.001)
            update_best(h3, c5_3, n3)

    # Final verification
    if best_h is not None:
        final_c5 = compute_c5(best_h)
        if final_c5 < best_c5:
            best_c5 = final_c5

    return best_h, best_c5, best_n
