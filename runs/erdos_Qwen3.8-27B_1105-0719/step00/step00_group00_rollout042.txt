def project_feasible(h, target=None):
    h = np.array(h, dtype=float, copy=True)
    n = h.size
    if target is None: target = 0.5 * float(n)
    if n == 0: return h
    # If close enough, just clip and small correction
    if np.all(h >= -1e-11) and np.all(h <= 1.0 + 1e-11) and abs(h.sum() - target) <= 1e-9:
        h = np.clip(h, 0.0, 1.0)
        _tiny_sum_fix(h, target)
        return h
    lo = float(np.min(h)) - 1.0 - 1e-8
    hi = float(np.max(h)) + 1e-8
    # ensure bracket
    if np.sum(np.clip(h - lo, 0,1)) < target: lo -= 1.0
    if np.sum(np.clip(h - hi, 0,1)) > target: hi += 1.0
    for _ in range(70):
        mid = 0.5*(lo+hi)
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    out = np.clip(h - 0.5*(lo+hi), 0.0, 1.0)
    _tiny_sum_fix(out,target)
    return out

def _tiny_sum_fix(out, target):
    for _ in range(5):
        diff = target - float(out.sum())
        if abs(diff) < 1e-12: break
        if diff > 0:
            room = 1.0 - out
            idxs = np.where(room > 1e-14)[0]
            if idxs.size == 0: break
            total = float(room[idxs].sum())
            scale = min(1.0, max(0.0, (diff - 1e-13) / total))
            out[idxs] += scale * room[idxs]
        else:
            room = out
            idxs = np.where(room > 1e-14)[0]
            if idxs.size ==0: break
            total = float(room[idxs].sum())
            scale = min(1.0, max(0.0, (-diff - 1e-13) / total))
            out[idxs] -= scale * room[idxs]
    out = np.clip(out,0.0,1.0)
    # final single-point fix
    diff = target - float(out.sum())
    if abs(diff) > 1e-13:
       if diff >0:
          idx = int(np.argmax(1.0-out)); add = min(1.0-out[idx], diff); out[idx]+=add
       else:
          idx = int(np.argmax(out)); add = max(-out[idx], diff); out[idx]+=add
       out=np.clip(out,0,1)

def _tiny_sum_fix(out, target):
    for _ in range(6):
        diff = target - float(out.sum())
        if abs(diff) < 1e-12:
            return out
        if diff > 0.0:
            room = 1.0 - out
            idxs = np.where(room > 1e-14)[0]
            if idxs.size == 0:
                break
            total = float(room[idxs].sum())
            scale = (diff - 1e-13) / total if total > 0.0 else 0.0
            if scale < 0.0: scale = 0.0
            if scale > 1.0: scale = 1.0
            out[idxs] += scale * room[idxs]
        else:
            room = out
            idxs = np.where(room > 1e-14)[0]
            if idxs.size == 0:
                break
            total = float(room[idxs].sum())
            scale = (-diff - 1e-13) / total if total > 0.0 else 0.0
            if scale < 0.0: scale=0.0
            if scale > 1.0: scale=1.0
            out[idxs] -= scale * room[idxs]
    out = np.clip(out, 0.0, 1.0)
    diff = target - float(out.sum())
    if abs(diff) > 1e-13:
        if diff > 0:
            idx = int(np.argmax(1.0 - out))
            add = min(1.0 - out[idx], diff)
            if add > 0: out[idx] += add
        else:
            idx = int(np.argmax(out))
            add = max(-out[idx], diff)
            if add < 0: out[idx] += add
        out = np.clip(out,0.0,1.0)
    return out

import time
import numpy as np
from scipy.optimize import minimize, minimize_scalar

COARSE_K = 33
TOP_COARSE = 50
TOP_REFINE = 12
DEFAULT_STEP_SIZES = np.array([

    0.015625, 0.0078125, 0.00390625, 0.001953125, 0.0009765625

def _tiny_sum_fix(out, target):
    for _ in range(6):
        diff = target - float(out.sum())
        if abs(diff) < 1e-12:
            return out
        if diff > 0.0:
            room = 1.0 - out
            idxs = np.where(room > 1e-14)[0]
            if idxs.size == 0:
                break
            total = float(room[idxs].sum())
            scale = (diff - 1e-13) / total if total > 0.0 else 0.0
            if scale < 0.0:
                scale = 0.0
            if scale > 1.0:
                scale = 1.0
            out[idxs] += scale * room[idxs]
        else:
            room = out
            idxs = np.where(room > 1e-14)[0]
            if idxs.size == 0:
                break
            total = float(room[idxs].sum())
            scale = (-diff - 1e-13) / total if total > 0.0 else 0.0
            if scale < 0.0:
                scale = 0.0
            if scale > 1.0:
                scale = 1.0
            out[idxs] -= scale * room[idxs]

    out = np.clip(out, 0.0, 1.0)
    diff = target - float(out.sum())
    if abs(diff) > 1e-13:
        if diff > 0.0:
            idx = int(np.argmax(1.0 - out))
            add = min(1.0 - out[idx], diff)
            if add > 0.0:
                out[idx] += add
        else:
            idx = int(np.argmax(out))
            add = max(-out[idx], diff)
            if add < 0.0:
                out[idx] += add
        out = np.clip(out, 0.0, 1.0)
    return out


def project_feasible(h, target=None):
    h = np.array(h, dtype=float, copy=True)
    n = h.size
    if target is None:
        target = 0.5 * float(n)
    if n == 0:
        return h

    if np.all(h >= -1e-11) and np.all(h <= 1.0 + 1e-11) and abs(h.sum() - target) <= 1e-9:
        h = np.clip(h, 0.0, 1.0)
        return _tiny_sum_fix(h, target)

    lo = float(np.min(h)) - 1.0 - 1e-8
    hi = float(np.max(h)) + 1e-8

    if np.sum(np.clip(h - lo, 0.0, 1.0)) < target:
        lo -= 1.0
    if np.sum(np.clip(h - hi, 0.0, 1.0)) > target:
        hi += 1.0

    for _ in range(70):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid

    out = np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)
    return _tiny_sum_fix(out, target)


def compute_c(h):
    h = np.asarray(h, dtype=float)
    return np.correlate(h, 1.0 - h, mode="full")


def c5_score(h):
    n = len(h)
    raw = float(np.max(compute_c(h)))
    return raw * (2.0 / float(n))


def lse_value(c, tau, dx):
    if tau <= 0.0:
        return float(np.max(c)) * dx
    f = dx * c
    m = float(np.max(f))
    z = (f - m) / tau
    z = np.clip(z, -700.0, 0.0)
    e = np.exp(z)
    return m + tau * float(np.log(np.sum(e)))


def lse_weights(c, tau, dx):
    if tau <= 0.0:
        w = np.zeros_like(c, dtype=float)
        w[int(np.argmax(c))] = 1.0
        return w
    f = dx * c
    m = float(np.max(f))
    z = (f - m) / tau
    z = np.clip(z, -700.0, 0.0)
    e = np.exp(z)
    return e / float(np.sum(e))


def gradient_raw(h, weights):
    n = len(h)
    g = np.zeros(n, dtype=float)
    if n == 0:
        return g

    w0 = float(weights[n - 1])
    if abs(w0) > 0.0:
        g += w0 * (1.0 - 2.0 * h)

    for s in range(1, n):
        low = h[:-s]
        high = h[s:]

        wp = float(weights[n - 1 + s])
        if abs(wp) > 0.0:
            g[s:] += wp * (1.0 - low)
            g[:-s] -= wp * high

        wn = float(weights[n - 1 - s])
        if abs(wn) > 0.0:
            g[:-s] += wn * (1.0 - high)
            g[s:] -= wn * low

    return g


def sum_constraint(x, target):
    return float(np.sum(x) - target)


def sum_constraint_jac(x, target):
    return np.ones_like(x, dtype=float)


def lse_objective(x, tau, dx):
    return lse_value(compute_c(x), tau, dx)


def lse_jacobian(x, tau, dx):
    c = compute_c(x)
    w = lse_weights(c, tau, dx)
    return dx * gradient_raw(x, w)


def slsqp_optimize(h, tau, maxiter, deadline):
    n = len(h)
    target = 0.5 * float(n)
    dx = 2.0 / float(n)
    h0 = project_feasible(h, target)

    bounds = [(0.0, 1.0)] * n
    cons = [
        {
            "type": "eq",
            "fun": sum_constraint,
            "jac": sum_constraint_jac,
            "args": (target,)
        }
    ]

    try:
        res = minimize(
            lse_objective,
            h0,
            method="SLSQP",
            bounds=bounds,
            constraints=cons,
            jac=lse_jacobian,
            args=(tau, dx),
            options={"maxiter": int(maxiter), "ftol": 1e-12}
        )
        hp = project_feasible(res.x, target)
    except Exception:
        hp = h0

    raw = float(np.max(compute_c(hp)))
    return hp, raw


def projected_lse_optimize(h, tau, max_iters, deadline, step_sizes=None):
    n = len(h)
    target = 0.5 * float(n)
    dx = 2.0 / float(n)
    h = project_feasible(h, target)

    c = compute_c(h)
    raw = float(np.max(c))
    c5 = raw * dx
    best_h = h.copy()
    best_raw = raw
    best_c5 = c5

    if step_sizes is None:
        step_sizes = DEFAULT_STEP_SIZES

    lse = lse_value(c, tau, dx)
    stagnation = 0

    for _ in range(max_iters):
        if time.time() >= deadline:
            break

        weights = lse_weights(c, tau, dx)
        g = gradient_raw(h, weights)
        gn = float(np.linalg.norm(g))
        if gn < 1e-14:
            break

        direction = -g / gn
        accepted = False

        for mv in step_sizes:
            h2 = project_feasible(h + direction * float(mv), target)
            c2 = compute_c(h2)
            raw2 = float(np.max(c2))
            c52 = raw2 * dx
            lse2 = lse_value(c2, tau, dx)

            if c52 < c5 - 1e-12 or (lse2 < lse - 1e-9 and c52 < c5 + 1e-4):
                h = h2
                c = c2
                raw = raw2
                c5 = c52
                lse = lse2
                accepted = True
                break

        if not accepted:
            stagnation += 1
            if stagnation > 20:
                break

        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_raw = raw
            best_h = h.copy()

    return best_h, best_raw


def linear_terms(h):
    n = len(h)
    m = 2 * n - 1
    lin = np.zeros((n, m), dtype=float)
    idx = np.arange(n, dtype=int)

    for r in range(n):
        lin[r, r - idx + n - 1] += 1.0 - h
        lin[r, idx - r + n - 1] -= h

    return lin


def pair_delta_objective(delta, c, A, B):
    t = float(delta)
    return float(np.max(c + A * t + B * t * t))


def pair_local_search(h, raw_score_in, deadline, max_sweeps=1000):
    n = len(h)
    target = 0.5 * float(n)
    m = 2 * n - 1
    h = project_feasible(h, target)

    c = compute_c(h)
    raw = float(np.max(c))
    best_h = h.copy()
    best_raw = raw

    frac = np.linspace(0.0, 1.0, COARSE_K)
    frac2 = frac * frac

    Bmat = np.zeros((n, n, m), dtype=float)
    for i in range(n):
        for j in range(n):
            if i == j:
                continue
            Bmat[i, j, n - 1] = -2.0
            Bmat[i, j, j - i + n - 1] = 1.0
            Bmat[i, j, i - j + n - 1] = 1.0

    tol = 1e-12

    for _ in range(max_sweeps):
        if time.time() >= deadline:
            break

        lin = linear_terms(h)
        top = []
        best_new = raw
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for i in range(n):
            hi = float(h[i])
            if hi <= 1e-12:
                continue
            for j in range(n):
                if i == j:
                    continue
                cap = float(1.0 - h[j])
                if cap <= 1e-12:
                    continue
                dm = hi if hi < cap else cap
                if dm <= 1e-10:
                    continue

                A = lin[j] - lin[i]
                B = Bmat[i, j]
                Acoef = A * dm
                Bcoef = B * (dm * dm)

                vals = np.max(
                    c[None, :] + Acoef[None, :] * frac[:, None] + Bcoef[None, :] * frac2[:, None],
                    axis=1
                )
                k = int(np.argmin(vals))
                v = float(vals[k])
                dval = float(frac[k]) * dm

                if v < best_new - tol:
                    best_new = v
                    best_i = i
                    best_j = j
                    best_delta = dval

                if v < raw + 1.0:
                    if len(top) < TOP_COARSE:
                        top.append((v, i, j, dm, dval))
                    else:
                        max_pos = 0
                        max_val = top[0][0]
                        for t in range(1, len(top)):
                            if top[t][0] > max_val:
                                max_val = top[t][0]
                                max_pos = t

        for a in range(len(top)):

            mv = top[a][0]
            for b in range(a + 1, len(top)):

                    mv = top[b][0]
                    mposa = b
            top[a], top[mposa] = top[mposa], top[a]

        refine = []
        seen = set()
        if best_i >= 0:

        for entry in top:

            width = dm / float(COARSE_K - 1)

            best_d_ref = float(dinit)

                    best_d_ref = dd

        if best_i < 0 or best_new >= raw - tol:
            break

        if best_delta < 0.0:
            best_delta = 0.0
        max_delta = min(float(h[best_i]), float(1.0 - h[best_j]))
        if best_delta > max_delta:

        h[best_i] -= best_delta

def symmetrize(h):
    h = np.asarray(h, dtype=float)
    n = len(h)
    return project_feasible(0.5 * (h + h[::-1]), 0.5 * float(n))


def resample_stairs(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return project_feasible(h_old, 0.5 * float(n_new))
    x_old = (np.arange(n_old) + 0.5) * (2.0 / float(n_old))
    x_new = (np.arange(n_new) + 0.5) * (2.0 / float(n_new))
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, 0.5 * float(n_new))


def make_smooth_candidates(n, rng, random_count=20):
    out = []
    target = 0.5 * float(n)
    x = (np.arange(n) + 0.5) * (2.0 / float(n))

    h = 0.5 * np.ones(n, dtype=float)
    out.append(project_feasible(h, target))

    for freq in (1, 2, 3):
        for phase in (0.0, 0.5 * np.pi):
            h = 0.5 + 0.5 * np.cos(np.pi * freq * x + phase)
            out.append(project_feasible(h, target))

    for _ in range(random_count):
        h = 0.5 * np.ones(n, dtype=float)
        for f in (1, 2, 3, 4):
            amp = rng.uniform(0.05, 0.25) / float(f)
            phase = rng.uniform(0.0, 2.0 * np.pi)
            h += amp * np.cos(np.pi * f * x + phase)
        out.append(project_feasible(h, target))

    for _ in range(3):
        h = (rng.random(n) < 0.5).astype(float)
        h += rng.normal(0.0, 0.05, n)
        out.append(project_feasible(h, target))

    return out


def polish_candidate(h, deadline, max_time, do_slsqp=True, do_pair=True):
    start = time.time()
    lim = min(deadline, start + float(max_time))
    n = len(h)
    target = 0.5 * float(n)
    dx = 2.0 / float(n)

    h = project_feasible(h, target)
    c = compute_c(h)
    raw = float(np.max(c))
    c5 = raw * dx
    best_h = h.copy()
    best_raw = raw
    best_c5 = c5

    if lim <= start:
        return best_h, best_c5

    if do_slsqp and time.time() < lim - 10.0:
        maxiter = 50 if n <= 55 else 35
        for tau in (0.05, 0.02):
            if time.time() >= lim - 10.0:
                break
            h2, raw2 = slsqp_optimize(h, tau, maxiter, lim)
            c52 = raw2 * dx
            if c52 < c5 + 1e-4:
                h = project_feasible(h2, target)
                raw = float(np.max(compute_c(h)))
                c5 = raw * dx
            if c52 < best_c5 - 1e-12:
                best_c5 = c52
                best_raw = raw2
                best_h = np.array(h2, dtype=float, copy=True)

    for tau in (0.04, 0.02, 0.01, 0.005):
        if time.time() >= lim - 15.0:
            break
        sub_deadline = min(lim, time.time() + max(5.0, (lim - time.time()) * 0.4))
        h2, raw2 = projected_lse_optimize(h, tau, 200, sub_deadline)
        h = h2
        raw = float(np.max(compute_c(h)))
        c5 = raw * dx
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_raw = raw
            best_h = h.copy()

    if best_c5 < c5 - 1e-12:
        h = best_h.copy()
        raw = best_raw
        c5 = raw * dx

    if do_pair and time.time() < lim - 5.0:
        h2, raw2 = pair_local_search(h, raw, max(start + 1.0, lim - 2.0))
        h = h2
        raw = raw2
        c5 = raw * dx
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_raw = raw
            best_h = h.copy()

    if time.time() < lim - 10.0:
        h2, raw2 = projected_lse_optimize(h, 0.008, 100, min(lim, time.time() + 30.0))
        h = h2
        raw = float(np.max(compute_c(h)))
        c5 = raw * dx
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_raw = raw
            best_h = h.copy()

    if do_pair and time.time() < lim - 5.0:
        h2, raw2 = pair_local_search(h, raw, max(start + 1.0, lim - 1.0))
        h = h2
        raw = raw2
        c5 = raw * dx
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_raw = raw
            best_h = h.copy()

    hsym = symmetrize(best_h)
    c5s = c5_score(hsym)
    if c5s < best_c5 - 1e-12:
        best_h = hsym
        best_c5 = c5s

    best_h = project_feasible(best_h, 0.5 * float(len(best_h)))
    c5f = c5_score(best_h)
    if c5f < best_c5:
        best_c5 = c5f
        best_h = best_h

    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    if budget_s is None or float(budget_s) <= 0.0:
        total = 1100.0
    else:
        total = float(budget_s)
    deadline = start + max(5.0, total - 3.0)
    rng = np.random.default_rng(seed)

    parent = globals().get("initial_h_values", None)
    if parent is not None:
        parent = np.asarray(parent, dtype=float)
        if parent.size == 0:
            parent = None

    if parent is not None and (parent.size < 8 or parent.size > 150):
        parent = resample_stairs(parent, 45)

    n_main = int(parent.size) if parent is not None else 45

    ns_raw = [n_main, n_main + 10, n_main - 5, n_main + 20, n_main + 40]
    ns = []
    for nn in ns_raw:
        if 15 <= nn <= 97 and nn not in ns:
            ns.append(nn)

    candidates = []
    for nn in ns:
        if parent is not None:
            hp = resample_stairs(parent, nn)
            candidates.append(hp)
            candidates.append(symmetrize(hp))
        rc = 20 if nn == n_main else 8
        for h in make_smooth_candidates(nn, rng, rc):
            candidates.append(h)

    eval_c5 = []
    eval_h = []
    for h in candidates:
        if h is None:
            continue
        h = project_feasible(h, 0.5 * float(len(h)))
        raw = float(np.max(compute_c(h)))
        c5 = raw * (2.0 / float(len(h)))
        if np.isfinite(c5):
            eval_c5.append(float(c5))
            eval_h.append(h)

    if not eval_h:
        h = project_feasible(0.5 * np.ones(45), 22.5)
        eval_c5.append(c5_score(h))
        eval_h.append(h)

    order = list(range(len(eval_c5)))
    for i in range(len(order)):
        min_idx = i
        min_val = eval_c5[order[i]]
        for j in range(i + 1, len(order)):
            if eval_c5[order[j]] < min_val:
                min_val = eval_c5[order[j]]
                min_idx = j
        order[i], order[min_idx] = order[min_idx], order[i]

    best_h = eval_h[order[0]]
    best_c5 = eval_c5[order[0]]

    cap_frac = [0.55, 0.22, 0.10, 0.05]
    for pos in range(min(4, len(order))):
        if time.time() >= deadline - 30.0:
            break
        max_time = max(30.0, total - 30.0) * cap_frac[pos]
        max_time = min(max_time, deadline - time.time() - 30.0)
        if max_time < 12.0:
            break
        h_start = eval_h[order[pos]]
        h2, c52 = polish_candidate(h_start, deadline, max_time, do_slsqp=(pos == 0), do_pair=True)
        if c52 < best_c5 - 1e-12:
            best_c5 = c52
            best_h = h2

    if best_h is not None and time.time() < deadline - 45.0:
        n_base = len(best_h)
        tries = []
        for factor in (1.0, 1.3, 1.6):
            nn = int(round(n_base * factor))
            if 15 <= nn <= 97 and nn != n_base:
                tries.append(nn)
        if n_base > 50 and n_base - 10 >= 15:
            tries.append(n_base - 10)
        if n_base < 97 and n_base + 15 <= 120:
            tries.append(n_base + 15)

        unique_tries = []
        for nn in tries:
            if nn not in unique_tries:
                unique_tries.append(nn)

        for nn in unique_tries[:3]:
            if time.time() >= deadline - 30.0:
                break
            h_try = resample_stairs(best_h, nn)
            c5_try = c5_score(h_try)
            if c5_try <= best_c5 + 0.02:
                max_time = min(90.0, deadline - time.time() - 20.0)
                if max_time < 20.0:
                    break
                h2, c52 = polish_candidate(h_try, deadline, max_time, do_slsqp=False, do_pair=True)
                if c52 < best_c5 - 1e-12:
                    best_c5 = c52
                    best_h = h2

    if best_h is not None and time.time() < deadline - 30.0:
        n_best = len(best_h)
        target = 0.5 * float(n_best)
        scales = [0.08, 0.04, 0.02, 0.01]
        for scale in scales:
            for _rep in range(2):
                if time.time() >= deadline - 25.0:
                    break
                noise = rng.normal(0.0, scale, n_best)
                h_try = project_feasible(best_h + noise, target)
                c5_try = c5_score(h_try)
                if c5_try <= best_c5 + 0.01:
                    max_time = min(45.0, deadline - time.time() - 10.0)
                    if max_time < 12.0:
                        break
                    h2, c52 = polish_candidate(h_try, deadline, max_time, do_slsqp=False, do_pair=True)
                    if c52 < best_c5 - 1e-12:
                        best_c5 = c52
                        best_h = h2

    if best_h is None:
        best_h = project_feasible(0.5 * np.ones(45), 22.5)
        best_c5 = c5_score(best_h)

    hsym = symmetrize(best_h)
    c5s = c5_score(hsym)
    if c5s < best_c5 - 1e-12:
        best_h = hsym
        best_c5 = c5s

    best_h = project_feasible(best_h, 0.5 * float(len(best_h)))
    c5f = c5_score(best_h)
    if c5f < best_c5:
        best_c5 = c5f
        best_h = best_h

    return (best_h, float(best_c5), int(best_h.size))
