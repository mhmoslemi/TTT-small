def exact_C(h):

    n = len(h)
    if n == 0:

    h = np.asarray(h, dtype=np.float64)
    n = len(h)
    if n == 0:
        return False

    h = np.asarray(h, dtype=np.float64).copy()

    target = n / 2.0

    if abs(residual) > 1e-12:

    n = len(h)
    one_minus_h = 1.0 - h
    lags = np.zeros(2*n - 1)

def sigmoid(t):
    # Stable sigmoid
    result = np.zeros_like(t)
    pos = t >= 0
    result[pos] = 1.0 / (1.0 + np.exp(-t[pos]))
    exp_t = np.exp(t[~pos])
    result[~pos] = exp_t / (1.0 + exp_t)
    return result

def h_from_logit(y):

    n = len(y)
    target = n / 2.0
    
    # Find lambda such that sum(sigmoid(y + lambda)) = target
    def sum_sigmoid(lam):
        return np.sum(sigmoid(y + lam))
    
    # Find bounds
    lo = -np.max(y) - 50.0
    hi = 1.0 - np.min(y) + 50.0  # wait this isn't right

    # if lambda -> +inf, all sigmoid -> 1, sum -> n
    # So we need lambda in a range where sum goes from 0 to n
    lo = -np.max(np.abs(y)) - 50.0
    hi = np.max(np.abs(y)) + 50.0

def logit_from_h(h):
    """Convert feasible h to logit vector (without the common shift)."""
    eps = 1e-7
    h_clipped = np.clip(h, eps, 1.0 - eps)
    y = np.log(h_clipped / (1.0 - h_clipped))
    # Remove mean to keep values bounded
    y = y - np.mean(y)
    return y

def smooth_obj_logit(y, beta):
    """Objective in logit space."""
    h, lam = h_from_logit(y)
    val, grad_h = smooth_obj_and_grad(h, beta)
    
    # Chain rule through the constraint
    m = h * (1.0 - h)
    M = np.sum(m)
    if M < 1e-12:
        return val, np.zeros_like(y)
    
    B = np.dot(grad_h, m) / M
    grad_y = m * (grad_h - B)
    
    return val, grad_y

    for l in range(-(n - 1), n):
        if l >= 0:
            lags[l + n - 1] = np.dot(h[:n - l], omh[l:])
        else:
            j = -l
            lags[l + n - 1] = np.dot(h[j:], omh[:n - j])
    return lags

def smooth_obj_grad_h(h, beta):
    n = len(h)
    dx = 2.0 / n
    lags = compute_lag_values(h)
    w = lags * dx
    wmax = np.max(w)
    shifted = beta * (w - wmax)
    shifted -= np.max(shifted)
    exp_s = np.exp(shifted)
    probs = exp_s / np.sum(exp_s)
    val = wmax + np.log(np.sum(exp_s)) / beta
    
    grad_h = np.zeros(n)
    for li in range(2 * n - 1):
        l = li - (n - 1)
        p = probs[li] * dx
        if p < 1e-15:
            continue
        if l >= 0:
            grad_h[:n - l] += p * (1.0 - h[l:])
            grad_h[l:] -= p * h[:n - l]
        else:
            j = -l
            grad_h[j:] += p * (1.0 - h[:n - j])
            grad_h[:n - j] -= p * h[j:]
    return val, grad_h

def h_from_logit(y):
    n = len(y)
    target = n / 2.0
    lo = -float(np.max(np.abs(y))) - 50.0
    hi = float(np.max(np.abs(y))) + 50.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(sigmoid(y + mid))
        if s < target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h = sigmoid(y + lam)
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            idx = np.where(h < 1.0 - 1e-8)[0]
        else:
            idx = np.where(h > 1e-8)[0]
        if len(idx) > 0:
            per = residual / len(idx)
            h[idx] = np.clip(h[idx] + per, 0.0, 1.0)
    return h

def logit_from_h(h):
    eps = 1e-7
    hc = np.clip(h, eps, 1.0 - eps)
    y = np.log(hc / (1.0 - hc))
    return y - np.mean(y)

def smooth_obj_grad_logit(y, beta):
    h = h_from_logit(y)
    val, grad_h = smooth_obj_grad_h(h, beta)
    m = h * (1.0 - h)
    M = np.sum(m)
    if M < 1e-12:
        return val, np.zeros_like(y)
    B = np.dot(grad_h, m) / M
    grad_y = m * (grad_h - B)
    return val, grad_y

def resize_h(h, new_n):
    """Resize h from old_n to new_n samples using interpolation."""
    old_n = len(h)
    if old_n == new_n:
        return h.copy()
    old_x = (np.arange(old_n) + 0.5) / old_n
    new_x = (np.arange(new_n) + 0.5) / new_n
    h_resized = np.interp(new_x, old_x, h)

                    h_try[a] += t

                    if h_try[b] < -1e-10 or h_try[b] > 1.0 + 1e-10:

        h[a] += best_t

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

_G = {'best_h': None, 'best_C': np.inf, 'deadline': 0, 'start_time': 0}


def exact_C(h):
    h = np.asarray(h, dtype=np.float64)
    n = len(h)
    if n == 0:
        return np.inf
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def validate_h(h):
    h = np.asarray(h, dtype=np.float64)
    n = len(h)
    if n == 0:
        return False
    if np.any(h < -1e-8) or np.any(h > 1.0 + 1e-8):
        return False
    if abs(np.sum(h) - n / 2.0) > 1e-6:
        return False
    return True


def project_feasible(h):
    h = np.asarray(h, dtype=np.float64).copy()
    n = len(h)
    target = n / 2.0
    lo = -float(np.max(h)) - 2.0
    hi = 1.0 - float(np.min(h)) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    result = np.clip(h + lam, 0.0, 1.0)
    residual = target - np.sum(result)
    if abs(residual) > 1e-12:
        if residual > 0:
            idx = np.where(result < 1.0 - 1e-10)[0]
        else:
            idx = np.where(result > 1e-10)[0]
        if len(idx) > 0:
            per = residual / len(idx)
            result[idx] = np.clip(result[idx] + per, 0.0, 1.0)
    return result


def sigmoid_f(t):
    t = np.asarray(t, dtype=np.float64)
    result = np.empty_like(t)
    pos = t >= 0
    result[pos] = 1.0 / (1.0 + np.exp(-t[pos]))
    exp_t = np.exp(t[~pos])
    result[~pos] = exp_t / (1.0 + exp_t)
    return result


def compute_lag_values(h):
    n = len(h)
    omh = 1.0 - h
    lags = np.empty(2 * n - 1)
    for l in range(-(n - 1), n):
        if l >= 0:
            lags[l + n - 1] = np.dot(h[l:], omh[:n - l])
        else:
            j = -l
            lags[l + n - 1] = np.dot(h[:n - j], omh[j:])
    return lags


def smooth_obj_grad_h(h, beta):
    n = len(h)
    dx = 2.0 / n
    lags = compute_lag_values(h)
    w = lags * dx
    wmax = np.max(w)
    shifted = beta * (w - wmax)
    shifted -= np.max(shifted)
    exp_s = np.exp(shifted)
    probs = exp_s / np.sum(exp_s)
    val = wmax + np.log(np.sum(exp_s)) / beta

    grad_h = np.zeros(n)
    for li in range(2 * n - 1):
        l = li - (n - 1)
        p = probs[li] * dx
        if p < 1e-15:
            continue
        if l >= 0:
            grad_h[l:] += p * (1.0 - h[:n - l])
            grad_h[:n - l] -= p * h[l:]
        else:
            j = -l
            grad_h[:n - j] += p * (1.0 - h[j:])
            grad_h[j:] -= p * h[:n - j]
    return val, grad_h


def h_from_logit(y):
    n = len(y)
    target = n / 2.0
    yabs = float(np.max(np.abs(y)))
    lo = -yabs - 50.0
    hi = yabs + 50.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(sigmoid_f(y + mid))
        if s < target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h = sigmoid_f(y + lam)
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            idx = np.where(h < 1.0 - 1e-8)[0]
        else:
            idx = np.where(h > 1e-8)[0]
        if len(idx) > 0:
            per = residual / len(idx)
            h[idx] = np.clip(h[idx] + per, 0.0, 1.0)
    return h


def logit_from_h(h):
    eps = 1e-7
    hc = np.clip(h, eps, 1.0 - eps)
    y = np.log(hc / (1.0 - hc))
    return y - np.mean(y)


def smooth_obj_grad_logit(y, beta):
    h = h_from_logit(y)
    val, grad_h = smooth_obj_grad_h(h, beta)
    m = h * (1.0 - h)
    M = np.sum(m)
    if M < 1e-12:
        return val, np.zeros_like(y)
    B = np.dot(grad_h, m) / M
    grad_y = m * (grad_h - B)
    return val, grad_y


def try_record(h):
    h = np.asarray(h, dtype=np.float64)
    if not validate_h(h):
        h = project_feasible(h)
        if not validate_h(h):
            return
    c = exact_C(h)
    if np.isfinite(c) and c < _G['best_C']:
        _G['best_C'] = c
        _G['best_h'] = h.copy()


def resize_h(h, new_n):
    old_n = len(h)
    if old_n == new_n:
        return project_feasible(h)
    old_x = (np.arange(old_n) + 0.5) / old_n
    new_x = (np.arange(new_n) + 0.5) / new_n
    h_resized = np.interp(new_x, old_x, h)
    return project_feasible(h_resized)


def pair_polish(h, max_moves=200, max_time=60.0):

    cur_C = exact_C(h)
    n = len(h)
    t_start = time.time()

            break

        top_coords = np.argsort(np.abs(grad_h))[-10:]
        # Also add some random coords
        extra = rng_local.choice(n, size=min(5, n), replace=False)
        candidate_a = np.unique(np.append(top_coords, extra))

        best_improve = 0.0
        best_a = -1
        best_b = -1
        best_t = 0.0
        best_C_local = cur_C

        for a in candidate_a:
            for b in range(n):
                if a == b:
                    continue
                lo_t = max(-h[a], h[b] - 1.0)

                for t in ts:

                    h_try = h.copy()
                    h_try[a] += t
                    h_try[b] -= t
                    if h_try[a] < -1e-10 or h_try[a] > 1.0 + 1e-10:

                    if h_try[b] < -1e-10 or h_try[b] > 1.0 + 1e-10:

            found = False
            for a in candidate_a[:5]:

                for b in range(n):

        h[best_b] -= best_t
        h = np.clip(h, 0.0, 1.0)

def run_bfgs(h_start, beta, max_iter=150):

        return h_start
    y0 = logit_from_h(h_start)
    try:

        )
        h_end = h_from_logit(res.x)
