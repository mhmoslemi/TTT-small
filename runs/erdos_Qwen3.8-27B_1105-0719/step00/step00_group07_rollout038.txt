import numpy as np
import time

def raw_c5(h):
    """Compute raw max correlation and C5."""
    n = len(h)
    dx = 2.0 / n
    raw = np.correlate(h, 1.0 - h, mode="full")
    raw_max = np.max(raw)
    return float(raw_max * dx), raw, float(raw_max), dx

def gradient_lag(h, i, n):
    """Compute gradient of raw correlation at output index i."""
    grad = np.zeros(n)
    k = i - (n - 1)
    if k >= 0:
        # c = sum_{j=0}^{n-1-k} h[j+k]*(1-h[j])
        grad[k:] += (1.0 - h[:n-k])
        grad[:n-k] += -h[k:]
    else:
        r = -k
        # c = sum_{j=r}^{n-1} h[j-r]*(1-h[j])
        grad[:n-r] += (1.0 - h[r:])
        grad[r:] += -h[:n-r]
    return grad

def project_feasible(x, target_sum):
    """Project x onto {0 <= y <= 1, sum(y) = target_sum}."""
    n = len(x)
    # Bisection on lambda: y = clip(x - lambda, 0, 1)
    lo = x.min() - 1.0
    hi = x.max() + 1.0
    # We want sum(clip(x - lambda, 0, 1)) = target_sum
    # As lambda increases, sum decreases
    
    for _ in range(100):
        mid = (lo + hi) / 2.0
        y = np.clip(x - mid, 0.0, 1.0)
        s = y.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    
    y = np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix rounding to get exact sum
    residual = target_sum - y.sum()
    if abs(residual) > 1e-12:
        # Distribute residual
        if residual > 0:
            # Add to elements with most slack
            slack = 1.0 - y
            order = np.argsort(-slack)
            for idx in order:
                add = min(residual, 1.0 - y[idx])
                y[idx] += add
                residual -= add
                if residual <= 1e-14:
                    break
        else:
            order = np.argsort(y)
            for idx in order:
                sub = min(-residual, y[idx])
                y[idx] -= sub
                residual += sub
                if residual >= -1e-14:
                    break
    return y

def flip_complement(h):
    return 1.0 - h[::-1]

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    start_time = time.perf_counter()
    deadline = start_time + budget_s - 5.0  # 5s safety margin
    rng = np.random.RandomState(seed)
    
    # Get initial values
    if 'initial_h_values' in dir() or True:
        try:

        except:

    else:
        h_best = 0.5 * np.ones(95)

    c5_best, _, raw_max_best, _ = raw_c5(h_best)
    
    # Also try the flip complement
    hc = flip_complement(h_best)
    c5c, _, _, _ = raw_c5(hc)
    if c5c < c5_best:
        h_best = hc
        c5_best = c5c
        raw_max_best = c5c * n_best / 2.0
    
    # Symmetrize
    hs = 0.5 * (h_best + flip_complement(h_best))
    hs = project_feasible(hs, n_best / 2.0)
    c5s, _, raw_max_s, _ = raw_c5(hs)
    if c5s < c5_best:
        h_best = hs.copy()
        c5_best = c5s
        raw_max_best = raw_max_s
    
    # Main optimization loop
    alpha = 0.01
    top_k = 5
    no_improve_count = 0
    max_no_improve = 200
    
    while time.perf_counter() < deadline:
        n = len(h_best)
        dx = 2.0 / n
        target_sum = n / 2.0
        
        # Compute raw correlations
        raw = np.correlate(h_best, 1.0 - h_best, mode="full")
        raw_max = np.max(raw)
        
        # Find top-k active lags
        top_idx = np.argsort(raw)[::-1][:top_k]

        grad = np.zeros(n)
        for idx in top_idx:

        grad /= top_k

        # but let's be safe

        h_new = project_feasible(h_new, target_sum)
        
        # Evaluate

        if c5_new < c5_best - 1e-12:
            h_best = h_new
            c5_best = c5_new
            raw_max_best = raw_max_new
            no_improve_count = 0
            # Increase alpha slightly

def _raw_corr(h):

def _project(x, target):
    n = len(x)
    lo = x.min() - 1.0
    hi = x.max() + 1.0

            hi = mid

        idxs = np.argsort(-(1.0 - y))
        for i in idxs:
            d = min(res, 1.0 - y[i])

            res -= d

    elif res < -1e-10:

    return 1.0 - h[::-1]

    grad = np.zeros(n)
    k = t - (n - 1)
    if k >= 0:
        m = n - k
        grad[k:] += (1.0 - h[:m])
        grad[:m] -= h[k:]
    else:
        r = -k
        m = n - r
        grad[:m] += (1.0 - h[r:])
        grad[r:] -= h[:m]
    return grad
