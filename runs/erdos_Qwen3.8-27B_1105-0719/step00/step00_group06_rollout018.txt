import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    prof = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(prof) * dx)

def raw_profile(h):
    return np.correlate(h, 1.0 - h, mode="full")

def repair_vector(h, n):
    """Repair h to satisfy 0<=h<=1 and sum=h = n/2"""
    h = np.asarray(h, dtype=np.float64).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = h.sum() - target
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h > 1e-15
            if mask.sum() == 0:
                h[:] = target / n
                break
            step = min(diff / mask.sum(), h[mask].min() - 1e-15)
            h[mask] -= step
        else:
            mask = h < 1.0 - 1e-15
            if mask.sum() == 0:
                h[:] = target / n
                break
            step = min(-diff / mask.sum(), (1.0 - 1e-15 - h[mask]).min())
            h[mask] += step
    h = np.clip(h, 0.0, 1.0)
    # Final adjustment
    diff = h.sum() - target
    if abs(diff) > 1e-12:
        # Find a coordinate that can absorb the remaining
        if diff > 0:
            idx = np.argmax(h)
            h[idx] -= min(diff, h[idx])
        else:
            idx = np.argmin(h)
            h[idx] += min(-diff, 1.0 - h[idx])
        h = np.clip(h, 0.0, 1.0)
    return h

def pair_bounds(h, i, j):
    """Feasible range for t where h[i]+=t, h[j]-=t stays in [0,1]"""
    a = max(-h[i], h[j] - 1.0)
    b = min(1.0 - h[i], h[j])
    # Add tiny margin
    a += 1e-15
    b -= 1e-15
    if a >= b:
        return None
    return (a, b)

def pair_objective(h, i, j, t):
    """Evaluate max raw profile at h[i]+t, h[j]-t"""
    hi = h[i] + t
    hj = h[j] - t
    if hi < -1e-12 or hi > 1.0 + 1e-12 or hj < -1e-12 or hj > 1.0 + 1e-12:
        return np.inf
    h2 = h.copy()
    h2[i] = np.clip(hi, 0.0, 1.0)
    h2[j] = np.clip(hj, 0.0, 1.0)
    prof = np.correlate(h2, 1.0 - h2, mode="full")
    return float(np.max(prof))

def golden_min(f, a, b, max_iter=60):
    """Golden section minimization of convex f on [a,b]"""
    if b - a < 1e-12:
        return 0.0, f(a)
    
    golden = (np.sqrt(5.0) - 1.0) / 2.0  # ~0.618
    
    c = b - golden * (b - a)
    d = a + golden * (b - a)
    fc = f(c)
    fd = f(d)
    
    best_t = 0.0
    best_f = min(fc, fd, f(0.0))
    if best_f == f(0.0):
        best_t = 0.0
    
    for _ in range(max_iter):
        if b - a < 1e-11:
            break
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - golden * (b - a)
            if c == a:
                c = a + 1e-15
            fc = f(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + golden * (b - a)
            if d == b:
                d = b - 1e-15
            fd = f(d)
        # Track best
        if fc < best_f:
            best_f = fc
            best_t = c
        if fd < best_f:
            best_f = fd
            best_t = d
    
    return best_t, best_f

def gradient_profile(h, k):
    """
    Gradient of the k-th profile value with respect to h.
    Profile k in np.correlate(h, 1-h, 'full') corresponds to shift.
    For np.correlate(a, v, 'full'), output index i corresponds to:
    sum_j a[j] * v[j + (i - (len(v)-1))]
    Let's verify: for len(a)=len(v)=n, full mode gives 2n-1 outputs.
    Output[i] = sum_j a[j] * v[j - (i - (n-1))] for valid j
    Actually, let me just use the definition:
    np.correlate(a, v)[k] = sum_j a[j+k] * v[j] for the 'valid' part
    In 'full' mode, k ranges from -(n-1) to n-1, and output index = k + (n-1)
    
    So output[i] where i = k + n - 1:
    = sum_j a[j+k] * v[j] where j is valid
    
    Here a=h, v=1-h
    output[k + n - 1] = sum_j h[j+k] * (1-h)[j]
    
    For the gradient with respect to h[m]:
    d/dh[m] sum_j h[j+k] * (1-h)[j]
    
    Case 1: m appears in h[j+k], i.e., j+k=m, j=m-k
    Term: h[m] * (1-h)[m-k], derivative: (1-h)[m-k] = 1 - h[m-k]
    Valid when 0 <= m-k < n, i.e., k <= m < n+k
    
    Case 2: m appears in (1-h)[j], i.e., j=m
    Term: h[m+k] * (1-h)[m], derivative: -h[m+k]
    Valid when 0 <= m+k < n, i.e., -k <= m < n-k
    
    So gradient[m] = (1-h[m-k]) * [0<=m-k<n] - h[m+k] * [0<=m+k<n]
    """
    n = len(h)
    g = np.zeros(n)
    # Case 1: m-k in [0, n-1], i.e., m in [k, n-1+k] but also m in [0, n-1]
    # h[j+k] where j = m-k, valid when 0<=m-k<n => k<=m<n+k, and m in [0,n-1]
    lo1 = max(0, k)
    hi1 = min(n, n + k)  # exclusive
    if lo1 < hi1:
        g[lo1:hi1] += 1.0 - h[lo1-k:hi1-k]
    
    # Case 2: m+k in [0, n-1], i.e., m in [-k, n-1-k] and m in [0, n-1]
    lo2 = max(0, -k)
    hi2 = min(n, n - k)  # exclusive
    if lo2 < hi2:
        g[lo2:hi2] -= h[lo2+k:hi2+k]
    
    return g

def get_active_lags(profile, active_frac=0.05, max_active=40):
    """Get indices of lags that are close to the maximum"""
    max_val = np.max(profile)
    threshold = max_val * (1.0 - active_frac)
    active = np.where(profile >= threshold)[0]
    # Also always include the max
    max_idx = np.argmax(profile)
    if max_idx not in active:
        active = np.append(active, max_idx)
    active = np.sort(active)
    if len(active) > max_active:
        # Keep the largest max_active
        vals = profile[active]
        order = np.argsort(vals)[::-1][:max_active]
        active = np.sort(active[order])
    return active

def select_pairs(h, profile, active_lags, max_pairs=20):
    """Select candidate pairs (i,j) for exchange based on gradients"""
    n = len(h)
    # Compute gradients for active lags
    # For each active lag index in the full profile (which corresponds to shift k = idx - (n-1))
    grads = []
    for lag_idx in active_lags:
        k = lag_idx - (n - 1)
        g = gradient_profile(h, k)
        grads.append(g)
    if not grads:
        return []
    
    G = np.array(grads)  # shape: (num_active, n)
    
    # For pair (i,j) with direction: h[i]+=t, h[j]-=t
    # Directional derivative of profile at lag k is G[k,i] - G[k,j]
    # We want this to be negative (decreasing) for all active k
    # Score: max over active k of (G[k,i] - G[k,j]) -- we want this to be as negative as possible
    
    pairs = []
    for i in range(n):
        for j in range(n):
            if i == j:
                continue
            # Direction: increase i, decrease j
            dir_deriv = G[:, i] - G[:, j]
            score = float(np.max(dir_deriv))  # We want this negative
            pairs.append((score, i, j))
    
    # Sort by score (most negative first)
    pairs.sort(key=lambda x: x[0])
    # Take top pairs with negative score
    result = [(i, j, s) for s, i, j in pairs if s < 0]
    return result[:max_pairs]

def pair_descent(h, profile, active_lags, max_pairs=20, deadline=None):
    """Perform coordinate exchange descent. Returns (h_new, improved)"""
    n = len(h)
    current_raw = float(np.max(profile))
    
    pairs = select_pairs(h, profile, active_lags, max_pairs)
    
    if not pairs:
        # Try random pairs
        rng = np.random.default_rng(42)
        for _ in range(max_pairs):
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j:
                pairs.append((i, j, 0.0))
    
    improved = False
    for i, j, score in pairs:
        if deadline is not None and time.monotonic() > deadline:
            break
        bounds = pair_bounds(h, i, j)
        if bounds is None:
            continue
        a, b = bounds
        
        # Check if feasible
        if a >= b:
            continue
        
        # Golden section search
        best_t = 0.0
        best_val = float(np.max(raw_profile(h)))
        
        # Evaluate at a few points
        test_points = [0.0, (a+b)/2, a*0.3+b*0.7, a*0.7+b*0.3]
        for tp in test_points:
            a_clip = max(tp, a + 1e-14)
            b_clip = min(tp, b - 1e-14)
            if a_clip > b_clip:
                tp2 = (a + b) / 2.0
            else:
                tp2 = tp
            val = pair_objective(h, i, j, tp2)
            if val < best_val - 1e-12:
                best_val = val
                best_t = tp2
        
        # If we found improvement, do a finer search
        if best_val < current_raw - 1e-10:
            # Narrow around best_t
            lo = max(a, best_t - 0.3 * (b - a))
            hi = min(b, best_t + 0.3 * (b - a))
            
            def f(t):
                return pair_objective(h, i, j, t)
            
            t_opt, f_opt = golden_min(f, lo, hi, max_iter=40)
            if f_opt < best_val - 1e-12:
                best_t = t_opt
                best_val = f_opt

        if best_val < current_raw - 1e-10:
            h[i] = np.clip(h[i] + best_t, 0.0, 1.0)
            h[j] = np.clip(h[j] - best_t, 0.0, 1.0)
            # Repair to maintain exact sum
            h = repair_vector(h, n)
            profile = raw_profile(h)
            current_raw = float(np.max(profile))

def random_perturb(h, n, rng, strength=0.1):

    """G is (K, n). Returns (n, n) matrix where score[i,j] = max_k(G[k,i]-G[k,j])"""
    K, n = G.shape
    # G[:, i, None] - G[:, None, j] has shape (K, n, n)

    diff = G[:, :, None] - G[:, None, :]  # shape (K, n, n), diff[k,i,j] = G[k,i]-G[k,j]
    score = np.max(diff, axis=0)  # shape (n, n)
    np.fill_diagonal(score, np.inf)  # disallow i=j
    return score

        pass

    if best_h is None:
        best_h = repair_f(np.ones(70) * 0.5, 70)
        best_score = compute_c5(best_h)

    n_main = len(best_h)
    rng = np.random.default_rng(seed)

    def try_solve(h_start, n_start, time_limit):
        nonlocal best_h, best_score
        h_w = repair_f(h_start.copy(), n_start)
        prof_w = np.correlate(h_w, 1.0 - h_w, mode="full")
        raw_w = float(np.max(prof_w))
        scaled_w = raw_w * 2.0 / n_start
        if scaled_w < best_score - 1e-12:
            best_score = scaled_w
            best_h = h_w.copy()
        stag = 0
        local_best = raw_w
        iter_count = 0
        while time.monotonic() < t0 + time_limit:
            h_w, improved = pair_descent_pass(h_w, prof_w, t0 + time_limit, max_pairs=35)
            if improved:
                prof_w = np.correlate(h_w, 1.0 - h_w, mode="full")
                raw_w = float(np.max(prof_w))
                scaled_w = raw_w * 2.0 / n_start
                stag = 0
                if scaled_w < best_score - 1e-12:
                    best_score = scaled_w
                    best_h = h_w.copy()
                if raw_w < local_best:
                    local_best = raw_w
            else:
                stag += 1
                if stag == 10:
                    h_p = h_w + rng.normal(0, 0.03, n_start)
                    h_p = repair_f(h_p, n_start)
                    prof_p = np.correlate(h_p, 1.0 - h_p, mode="full")
                    raw_p = float(np.max(prof_p))
                    if raw_p < local_best * 1.02:
                        h_w = h_p
                        prof_w = prof_p
                        raw_w = raw_p
                        stag = 0
                elif stag >= 40:
                    break
            iter_count += 1
            if iter_count > 5000:
                break

    # Main optimization on parent resolution
    remaining = deadline - time.monotonic()
    if remaining > 30:
        try_solve(best_h.copy(), n_main, min(deadline - 5, t0 + remaining * 0.7))

    # Try different resolutions
    resolutions = set()
    resolutions.add(n_main)
    if n_main >= 30:
        resolutions.add(n_main - 10)
    if n_main >= 30:
        resolutions.add(n_main + 10)
    if n_main >= 50:
        resolutions.add(n_main // 2)
    resolutions.discard(n_main)
    resolutions = [r for r in sorted(resolutions) if 10 <= r <= 200]

    for res_n in resolutions:
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break
        h_res = interp_to(best_h, res_n)
        alloc = min(remaining * 0.4, 120.0)
        if alloc > 10:
            try_solve(h_res, res_n, min(deadline - 5, t0 + alloc))

    # Try multiple random initializations
    for ptype in range(5):
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        h_init = make_pattern(n_main, seed + ptype * 100, ptype)
        alloc = min(remaining * 0.3, 60.0)
        if alloc > 8:
            try_solve(h_init, n_main, min(deadline - 5, t0 + alloc))

    # Final refinement if time remains
    remaining = deadline - time.monotonic()
    if remaining > 15:
        n_fin = len(best_h)
        h_fin = best_h.copy()
        prof_fin = np.correlate(h_fin, 1.0 - h_fin, mode="full")
        stag = 0
        while time.monotonic() < deadline - 3:
            h_fin, improved = pair_descent_pass(h_fin, prof_fin, deadline - 3, max_pairs=50)
            if improved:
                prof_fin = np.correlate(h_fin, 1.0 - h_fin, mode="full")
                raw_fin = float(np.max(prof_fin))
                scaled_fin = raw_fin * 2.0 / n_fin
                stag = 0
                if scaled_fin < best_score - 1e-12:
                    best_score = scaled_fin
                    best_h = h_fin.copy()
            else:
                stag += 1
                if stag >= 20:
                    break

    best_h = repair_f(best_h, len(best_h))
    final_score = compute_c5(best_h)
    return best_h, final_score, len(best_h)
