import time
from scipy.optimize import linprog

def objective_and_lags(h):
    n = len(h)
    dx = 2.0 / n
    lags = np.correlate(h, 1.0 - h, mode='full') * dx
    return np.max(lags), lags

def project_feasible(z, target_sum):
    n = len(z)
    z = np.asarray(z, dtype=float)
    # Replace nan/inf
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    z = np.clip(z, 0.0, 1.0)
    
    def f(lam):
        return np.sum(np.clip(z - lam, 0.0, 1.0)) - target_sum
    
    lo = -1.0
    hi = 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if f(mid) > 0:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    result = np.clip(z - lam, 0.0, 1.0)
    
    # Fix residual
    diff = target_sum - np.sum(result)
    if abs(diff) > 1e-14:
        if diff > 0:
            mask = result < 1.0
        else:
            mask = result > 0.0
        cnt = mask.sum()
        if cnt > 0:
            result[mask] += diff / cnt
            result = np.clip(result, 0.0, 1.0)
    return result

def soft_max_obj_grad(h, beta, lags=None):
    n = len(h)
    dx = 2.0 / n
    if lags is None:
        _, lags = objective_and_lags(h)
    
    m = np.max(lags)
    shifted = beta * (lags - m)
    shifted = np.clip(shifted, -500, 500)
    exps = np.exp(shifted)
    s = np.sum(exps)
    phi = m + np.log(s) / beta
    w = exps / s
    
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        weight = w[idx]
        if weight < 1e-15:
            continue
        d = idx - (n - 1)
        if d >= 0:
            # Case 1: j = d..n-1, grad[j] += dx*w*(1-h[j-d])
            js = np.arange(d, n)
            if len(js) > 0:
                grad[js] += dx * weight * (1.0 - h[js - d])
            # Case 2: j = 0..n-1-d, grad[j] += dx*w*(-h[j+d])
            js2 = np.arange(0, n - d)
            if len(js2) > 0:
                grad[js2] += dx * weight * (-h[js2 + d])
        else:
            mabs = -d
            # Case 1: j = 0..n-1-mabs, grad[j] += dx*w*(1-h[j+mabs])
            js = np.arange(0, n - mabs)
            if len(js) > 0:
                grad[js] += dx * weight * (1.0 - h[js + mabs])
            # Case 2: j = mabs..n-1, grad[j] += dx*w*(-h[j-mabs])
            js2 = np.arange(mabs, n)
            if len(js2) > 0:
                grad[js2] += dx * weight * (-h[js2 - mabs])
    
    return phi, grad

def projected_gradient_step(h, grad, alpha, target_sum):
    h_new = h - alpha * grad
    return project_feasible(h_new, target_sum)

def pg_descent(h0, target_sum, betas, time_limit, t_start, verbose=False):
    """Run projected gradient descent with soft-max smoothing."""
    best_h = h0.copy()
    best_c5, _ = objective_and_lags(h0)
    current = h0.copy()
    
    for beta in betas:
        alpha = 1.0
        stall_count = 0
        best_phi = soft_max_obj_grad(current, beta)[0]
        
        for it in range(200):
            if time.time() - t_start > time_limit:
                break
            
            phi, grad = soft_max_obj_grad(current, beta)
            
            # Scale gradient
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-10:
                break
            
            # Line search
            improved = False
            for ls in range(20):
                h_new = projected_gradient_step(current, grad, alpha, target_sum)
                phi_new, _ = soft_max_obj_grad(h_new, beta)
                if phi_new < best_phi - 1e-12:
                    current = h_new
                    best_phi = phi_new
                    alpha = min(alpha * 1.1, 5.0)
                    improved = True
                    break
                else:
                    alpha *= 0.5
            
            if not improved:
                stall_count += 1
                alpha *= 0.5
                if alpha < 1e-8:
                    break
            else:
                stall_count = 0
            
            # Check exact objective
            c5_new, _ = objective_and_lags(current)
            if c5_new < best_c5 - 1e-10:
                best_c5 = c5_new
                best_h = current.copy()
        
        # After each beta, update best
        c5_curr, _ = objective_and_lags(current)
        if c5_curr < best_c5:
            best_c5 = c5_curr
            best_h = current.copy()
    
    return best_h, best_c5

    z = z.astype(np.float64).copy()

    np.clip(z, 0.0, 1.0, out=z)

        else:
            m = res > 0.0

def active_lp_phase(h, target, t_limit, best_h, best_c5, top_k=30):
    current = h.copy()
    c5, lags = eval_c5(current)
    if c5 < best_c5 - 1e-12:
        best_c5 = c5; best_h = current.copy()
    radius = 0.05
    stagnation = 0
    for it in range(200):
        if time.time() > t_limit: break
        c5, lags = eval_c5(current)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5; best_h = current.copy()
            stagnation = 0
        else:
            stagnation += 1
        G = grad_all(current)
        # Select active lags
        order = np.argsort(lags)[::-1]
        active = order[:top_k]

        b_vals = []
        for idx in active:

            f_d = lags[idx]
            A_rows.append(np.append(g_d, -1.0))
            b_vals.append(g_d @ current - f_d)
        A_ub = np.array(A_rows)
        b_ub = np.array(b_vals)
        A_eq = np.zeros((1, len(current)+1))

        c_obj = np.zeros(len(current)+1)
        c_obj[-1] = 1.0
        # Trust region bounds

        hi = np.minimum(1.0, current + radius)
        bounds = list(zip(lo, hi)) + [(0.0, 2.0)]
        try:

        except:

        c5_new, _ = eval_c5(x_new)

            radius = min(radius * 1.2, 0.15)
        else:
            radius *= 0.7
            if radius < 1e-5: break
            stagnation += 1
            if stagnation > 30: break
    return current, best_h, best_c5

import numpy as np
import time
from scipy.optimize import linprog

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    lags = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(lags.max()), lags

def project_feasible(z, target_sum):
    z = np.array(z, dtype=np.float64, copy=True)
    z[~np.isfinite(z)] = 0.5
    np.clip(z, 0.0, 1.0, out=z)
    lo, hi = -1.0, 2.0
    for _ in range(70):
        mid = 0.5*(lo+hi)
        if np.sum(np.clip(z-mid, 0.0, 1.0)) > target_sum:
            lo = mid
        else:
            hi = mid
    res = np.clip(z - 0.5*(lo+hi), 0.0, 1.0)
    d = target_sum - res.sum()
    tol = 1e-13
    if d > tol:
        m = res < 1.0
        if m.any():
            res[m] += d / m.sum()
            np.clip(res, 0.0, 1.0, out=res)
    elif d < -tol:
        m = res > 0.0
        if m.any():
            res[m] += d / m.sum()
            np.clip(res, 0.0, 1.0, out=res)
    return res

def lse_grad(h, beta, lags=None):
    n = len(h)
    dx = 2.0 / n
    if lags is None:
        _, lags = eval_c5(h)
    mx = lags.max()
    sh = np.clip(beta * (lags - mx), -350, 350)
    ev = np.exp(sh)
    s = ev.sum()
    w = ev / s
    g = np.zeros(n)
    for k in range(2*n - 1):
        wk = w[k]
        if wk < 1e-13:
            continue
        d = k - (n - 1)
        if d >= 0:
            if d < n:
                g[d:] += dx * wk * (1.0 - h[:n-d])
                g[:n-d] += dx * wk * (-h[d:])
        else:
            ma = -d
            if ma < n:
                g[:n-ma] += dx * wk * (1.0 - h[ma:])
                g[ma:] += dx * wk * (-h[:n-ma])
    phi = mx + np.log(s) / beta
    return phi, g

def all_fgrads(h):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        d = k - (n-1)
        g = G[k]
        if d >= 0:
            if d < n:
                g[d:] += dx * (1.0 - h[:n-d])
                g[:n-d] += dx * (-h[d:])
        else:
            ma = -d
            if ma < n:
                g[:n-ma] += dx * (1.0 - h[ma:])
                g[ma:] += dx * (-h[:n-ma])
    return G

def fallback_init(n, rng):
    q = n // 4
    h = np.empty(n)
    h[:q] = 1.0
    h[q:2*q] = 0.5
    h[2*q:3*q] = 0.0
    h[3*q:] = 0.5
    h += 0.05 * rng.standard_normal(n)
    return project_feasible(h, n / 2.0)

def pg_loop(cur, beta, target, t_limit, best_h, best_c5):
    alpha = 0.3
    stall = 0
    for it in range(500):
        if time.time() > t_limit:
            break
        phi, g = lse_grad(cur, beta)
        gn = np.linalg.norm(g)
        if gn < 1e-11:
            break
        ok = False
        a = alpha
        for _ in range(25):
            ht = project_feasible(cur - a * g, target)
            pt, _ = lse_grad(ht, beta)
            if pt < phi - 1e-14:
                cur = ht
                alpha = min(a * 1.1, 8.0)
                ok = True
                break
            a *= 0.5
        if not ok:
            stall += 1
            alpha *= 0.5
            if stall > 20 or alpha < 1e-9:
                break
        else:
            stall = 0
        c5t, _ = eval_c5(cur)
        if c5t < best_c5 - 1e-12:
            best_c5 = c5t
            best_h = cur.copy()
    return cur, best_h, best_c5

def lp_loop(cur, target, t_limit, best_h, best_c5, top_k=40):
    radius = 0.03
    stale = 0
    for it in range(500):
        if time.time() > t_limit:
            break
        c5, lags = eval_c5(cur)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = cur.copy()
            stale = 0
        else:
            stale += 1
        G = all_fgrads(cur)
        idx = np.argsort(-lags)[:top_k]
        nn = len(cur)
        A_ub = np.empty((len(idx), nn+1))
        b_ub = np.empty(len(idx))
        for i, k in enumerate(idx):
            A_ub[i, :nn] = G[k]
            A_ub[i, nn] = -1.0
            b_ub[i] = G[k] @ cur - lags[k]
        A_eq = np.zeros((1, nn+1))
        A_eq[0, :nn] = 1.0
        b_eq = np.array([target])
        c_obj = np.zeros(nn+1)
        c_obj[nn] = 1.0
        lo_b = np.maximum(0.0, cur - radius)
        hi_b = np.minimum(1.0, cur + radius)
        bnd = list(zip(lo_b, hi_b)) + [(0.0, 1.5)]
        try:
            sol = linprog(c_obj, A_ub=A_ub, b_ub=b_ub,
                          A_eq=A_eq, b_eq=b_eq, bounds=bnd, method='highs')
        except Exception:
            break
        if not sol.success:
            radius *= 0.5
            if radius < 1e-5:
                break
            continue
        xn = project_feasible(sol.x[:nn], target)
        c5n, _ = eval_c5(xn)
        if c5n <= c5 + 1e-10:
            cur = xn
            radius = min(radius * 1.25, 0.12)
        else:
            radius *= 0.6
            stale += 1
            if radius < 1e-5 or stale > 50:
                break
    return cur, best_h, best_c5
