import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h.astype(float), 1.0 - h.astype(float), mode='full')
    return float(dx * np.max(corr))

def lse_grad(h, beta):
    """Compute log-sum-exp objective and gradient."""
    n = len(h)
    counts = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(counts)
    shifted = beta * (counts - m)
    sm = np.max(shifted)
    log_sum = m + (sm + np.log(np.sum(np.exp(shifted - sm)))) / beta
    weights = np.exp(shifted - sm) / np.sum(np.exp(shifted - sm))
    
    grad = np.zeros(n)
    for t in range(len(weights)):
        w = weights[t]
        if w < 1e-12:
            continue
        lag = t - (n - 1)
        # First index contribution: h[p] = h[n+lag], n = p-lag valid
        lo1 = max(0, lag)
        hi1 = min(n - 1, n - 1 + lag)
        if lo1 <= hi1:
            idx = np.arange(lo1, hi1 + 1)
            grad[idx] += w * (1.0 - h[idx - lag])
        # Second index contribution: h[p] in (1-h[n]), n=p, h[p+lag] valid
        lo2 = max(0, -lag)
        hi2 = min(n - 1, n - 1 - lag)
        if lo2 <= hi2:
            idx = np.arange(lo2, hi2 + 1)
            grad[idx] += w * (-h[idx + lag])
    
    return log_sum, grad

def project_feasible(h, target):
    n = len(h)
    h = np.asarray(h, dtype=float)
    lo = -2.0
    hi = 2.0
    for _ in range(80):

        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    r = target - np.sum(x)
    if abs(r) > 1e-9:
        if r > 0:
            slack = 1.0 - x
            order = np.argsort(-slack)
            for i in order:
                add = min(r, slack[i])
                if add <= 0:
                    break
                x[i] += add
                r -= add
                if abs(r) < 1e-12:

        else:

            order = np.argsort(-x)

                sub = min(r, x[i])
                if sub <= 0:
                    break
                x[i] -= sub
                r -= sub
                if abs(r) < 1e-12:
                    break
    return x

def adam_optimize(h0, n, start_time, max_time, target_c5=0.3808):

    best_h = h.copy()

    m_v = np.zeros(n)
    v_v = np.zeros(n)
    t = 0
    lr = 0.05
    b1, b2, eps = 0.9, 0.999, 1e-8
    lse_beta = 5.0
    
    patience = 0
    best_gradient_c5 = best_c5
    
    while time.time() - start_time < start_time + max_time:
        t += 1
        obj, grad = lse_grad(h, lse_beta)
        
        m_v = b1 * m_v + (1 - b1) * grad
        v_v = b2 * v_v + (1 - b2) * grad * grad

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h.astype(float), 1.0 - h.astype(float), mode='full')
    return float(dx * np.max(corr))

def lse_grad(h, beta):
    n = len(h)
    counts = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(counts)
    shifted = beta * (counts - m)
    sm = np.max(shifted)
    exp_shifted = np.exp(shifted - sm)
    log_sum = m + (sm + np.log(np.sum(exp_shifted))) / beta
    weights = exp_shifted / np.sum(exp_shifted)
    
    grad = np.zeros(n)
    for t in range(len(weights)):
        w = weights[t]
        if w < 1e-12:
            continue
        lag = t - (n - 1)
        lo1 = max(0, lag)
        hi1 = min(n, n + lag)
        if lo1 < hi1:
            grad[lo1:hi1] += w * (1.0 - h[lo1 - lag:hi1 - lag])
        lo2 = max(0, -lag)
        hi2 = min(n, n - lag)
        if lo2 < hi2:
            grad[lo2:hi2] += -w * h[lo2 + lag:hi2 + lag]
    
    return log_sum, grad

def project_feasible(h, target):
    n = len(h)
    h = np.asarray(h, dtype=float)
    lo_b = -2.0
    hi_b = 2.0
    for _ in range(80):
        mid = (lo_b + hi_b) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo_b = mid
        else:
            hi_b = mid
    x = np.clip(h - (lo_b + hi_b) / 2.0, 0.0, 1.0)
    r = target - np.sum(x)
    if abs(r) > 1e-10:
        if r > 0:
            slack = 1.0 - x
            order = np.argsort(-slack)
            for i in order:
                add = min(r, slack[i])
                if add <= 0:
                    break
                x[i] += add
                r -= add
                if abs(r) < 1e-12:
                    break
        else:
            r = -r
            order = np.argsort(-x)
            for i in order:
                sub = min(r, x[i])
                if sub <= 0:
                    break
                x[i] -= sub
                r -= sub
                if abs(r) < 1e-12:
                    break
    return x

def adam_optimize(h0, n, start_time, max_time):
    target = n / 2.0
    h = project_feasible(h0.astype(float).copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    m_v = np.zeros(n)
    v_v = np.zeros(n)
    t = 0
    lr = 0.03
    b1, b2, eps = 0.9, 0.999, 1e-8
    lse_beta = 3.0
    
    while time.time() - start_time < max_time:
        t += 1
        obj, grad = lse_grad(h, lse_beta)
        
        m_v = b1 * m_v + (1.0 - b1) * grad
        v_v = b2 * v_v + (1.0 - b2) * grad * grad
        m_hat = m_v / (1.0 - b1 ** t)
        v_hat = v_v / (1.0 - b2 ** t)
        
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_feasible(h - step, target)
        
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
        
        h = h_new
        
        if t % 500 == 0:
            lr = max(0.0005, lr * 0.98)
        lse_beta = min(80.0, 2.0 + 0.003 * t)
    
    return best_h, best_c5

def binary_sa(h_bin, n, start_time, max_time):
    target = n // 2
    h = h_bin.astype(int).copy()
    assert abs(np.sum(h) - target) < 1, f"Sum {np.sum(h)} != {target}"
    
    best_h = h.copy()
    best_max = np.max(np.correlate(h.astype(float), 1.0 - h.astype(float), mode='full'))
    
    rng = np.random.RandomState(12345)
    temperature = 1.0
    ones_idx = np.where(h == 1)[0]
    zeros_idx = np.where(h == 0)[0]
    
    while time.time() - start_time < max_time:
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        
        p = ones_idx[rng.randint(len(ones_idx))]
        q = zeros_idx[rng.randint(len(zeros_idx))]
        
        h[p] = 0
        h[q] = 1
        
        counts = np.correlate(h.astype(float), 1.0 - h.astype(float), mode='full')
        new_max = np.max(counts)
        
        if new_max <= best_max:
            if new_max < best_max:
                best_max = new_max
                best_h = h.copy()
        else:
            if rng.random() < np.exp(-(new_max - best_max) / temperature):
                pass
            else:
                h[p] = 1
                h[q] = 0
        
        ones_idx = np.where(h == 1)[0]
        zeros_idx = np.where(h == 0)[0]
        
        temperature = max(0.05, temperature * 0.99995)
    
    return best_h, float(2.0 / n * best_max)
