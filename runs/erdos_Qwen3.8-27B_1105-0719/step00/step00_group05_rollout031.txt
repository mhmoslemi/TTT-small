import numpy as np
import time

def project_feasible(z, n):
    """Project z onto {h: 0<=h<=1, sum(h)=n/2}"""
    target = n / 2.0
    # Binary search on lambda for clip(z + lambda, 0, 1)
    lo, hi = -z.max() - 1, 1 - z.min() + 1
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.clip(z + mid, 0, 1).sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2, 0, 1)
    # Fix floating point
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        # Distribute correction
        idx = np.arange(n)
        # Adjust from indices with most room
        if diff > 0:
            # Need to increase
            room = 1.0 - h
            order = np.argsort(-room)
        else:
            room = h
            order = np.argsort(room)
        remaining = diff
        for i in order:
            if remaining == 0:
                break
            if diff > 0:
                add = min(1.0 - h[i], remaining)
            else:
                add = max(h[i], remaining)
            h[i] += add
            remaining -= add
    return h

def evaluate_exact(h):
    """Compute C(h) = max(correlate(h, 1-h, 'full')) * 2/n"""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def compute_overlaps(h):
    """Compute all 2n-1 overlap values (without dx factor)"""
    n = len(h)
    values = np.empty(2 * n - 1)
    # k=0 (index n-1 in full output)
    values[n-1] = np.sum(h * (1 - h))
    # k=m for m=1..n-1 (indices n-1+m)
    for m in range(1, n):
        # sum_{i=0}^{n-m-1} h[i+m] * (1-h[i])
        values[n-1+m] = np.sum(h[m:] * (1 - h[:-m]))
        # k=-m (index n-1-m)
        # sum_{i=0}^{n-m-1} h[i] * (1-h[i+m])
        values[n-1-m] = np.sum(h[:-m] * (1 - h[m:]))
    return values

def smooth_max_and_weights(values, beta):
    """Compute smooth max and softmax weights"""
    v_max = np.max(values)
    shifted = beta * (values - v_max)
    exps = np.exp(shifted)
    total = np.sum(exps)
    weights = exps / total
    smooth_val = v_max + np.log(total) / beta
    return smooth_val, weights

def smooth_gradient(h, beta):
    """Compute gradient of smooth objective"""
    n = len(h)
    dx = 2.0 / n
    values = compute_overlaps(h)
    smooth_val, weights = smooth_max_and_weights(values, beta)
    
    # Gradient: g[i] = dx * sum_j weights[j] * (dv_j/dh[i])
    g = np.zeros(n)
    
    # k=0 term (weight at index n-1)
    g += weights[n-1] * (1 - 2 * h)
    
    # k=m terms (indices n-1+m for m=1..n-1)
    for m in range(1, n):
        w_pos = weights[n-1+m]
        w_neg = weights[n-1-m]
        if w_pos < 1e-15 and w_neg < 1e-15:
            continue
        # dv_pos[m]/dh: -h[i+m] for i in [0,n-m), +(1-h[i-m]) for i in [m,n)
        if w_pos > 0:
            g[:n-m] += -w_pos * h[m:]
            g[m:] += w_pos * (1 - h[:n-m])
        # dv_neg[m]/dh: (1-h[i+m]) for i in [0,n-m), -h[i-m] for i in [m,n)
        if w_neg > 0:
            g[:n-m] += w_neg * (1 - h[m:])
            g[m:] += -w_neg * h[:n-m]
    
    g *= dx
    return smooth_val, g

def run_projected_descent(h0, beta_start, max_iters, time_limit):

    n = len(h0)
    h = h0.copy()
    c_best = evaluate_exact(h)
    h_best = h.copy()
    
    start_time = time.time()
    beta = beta_start
    step_size = 0.1

        if g_norm < 1e-12:
            break

        accepted = False
        cos = -g / g_norm  # descent direction
        for scale in [step_size * 2, step_size, step_size * 0.5, step_size * 0.25]:
            cand = h + scale * cos
            cand = project_feasible(cand, n)
            c_cand = evaluate_exact(cand)
            if c_cand < c_best - 1e-14:

                c_best = c_cand

                step_size = min(step_size * 1.2, 1.0)

        if not accepted:

    return h_best, c_best

def interpolate_to(h, n_new):

import numpy as np
import time


def project_feasible(z, n):
    """Project z onto {h: 0<=h_i<=1, sum(h)=n/2}"""
    target = n / 2.0
    lo = -z.max() - 1.0
    hi = 1.0 - z.min() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(z + mid, 0.0, 1.0).sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for i in order:
            if abs(diff) < 1e-15:
                break
            if diff > 0:
                adj = min(1.0 - h[i], diff)
            else:
                adj = max(h[i], diff)
            h[i] += adj
            diff -= adj
    return h


def eval_c5(h):
    """Compute C5 = max(correlate(h, 1-h, 'full')) * 2/n"""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * (2.0 / n)


def compute_all_overlaps(h):
    """Compute all 2n-1 raw overlap values"""
    n = len(h)
    vals = np.empty(2 * n - 1)
    vals[n - 1] = np.dot(h, 1.0 - h)
    for m in range(1, n):
        vals[n - 1 + m] = np.dot(h[m:], 1.0 - h[:n - m])
        vals[n - 1 - m] = np.dot(h[:n - m], 1.0 - h[m:])
    return vals


def smooth_grad(h, beta):
    """Smooth max approximation and its gradient"""
    n = len(h)
    dx = 2.0 / n
    vals = compute_all_overlaps(h)
    v_max = np.max(vals)
    shifted = beta * (vals - v_max)
    exps = np.exp(shifted)
    total = exps.sum()
    p = exps / total
    smooth_val = (v_max + np.log(total) / beta) * dx

    g = np.zeros(n)
    g += p[n - 1] * (1.0 - 2.0 * h)
    for m in range(1, n):
        pm = p[n - 1 + m]
        pn = p[n - 1 - m]
        if pm < 1e-12 and pn < 1e-12:
            continue
        if pm > 1e-12:
            g[:n - m] += -pm * h[m:]
            g[m:] += pm * (1.0 - h[:n - m])
        if pn > 1e-12:
            g[:n - m] += pn * (1.0 - h[m:])
            g[m:] += -pn * h[:n - m]
    g *= dx
    return smooth_val, g


def run_smooth_descent(h0, time_limit, beta0=5.0):
    """Projected gradient descent with smooth max surrogate"""
    n = len(h0)
    h = h0.copy()
    c_best = eval_c5(h)
    h_best = h.copy()
    start = time.time()
    beta = beta0
    step = 0.5
    stall = 0

    while time.time() - start < time_limit:
        sv, g = smooth_grad(h, beta)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        direction = -g / gn
        accepted = False
        for scale in [step, step * 0.5, step * 0.25, step * 0.1, step * 0.02]:
            cand = h + scale * direction
            cand = project_feasible(cand, n)
            c_new = eval_c5(cand)
            if c_new < c_best - 1e-15:
                h = cand
                c_best = c_new
                h_best = h.copy()
                step = min(step * 1.2, 2.0)
                stall = 0
                accepted = True
                break
            elif c_new < sv - 1e-12:
                h = cand
                step = min(step * 1.05, 2.0)
                stall = 0
                accepted = True
                break
        if not accepted:
            step *= 0.7
            stall += 1
            if stall > 30:
                beta *= 2.0
                step = 0.3
                stall = 0
                if beta > 1000:
                    break
    return h_best, c_best

    if best_eps != 0.0:

                h_save = h.copy()
