def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_lags(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx

def compute_subgradient(h, top_indices, n):
    dx = 2.0 / n
    g = np.zeros(n)
    for k in top_indices:
        d = k - (n - 1)
        if d == 0:
            g += (1.0 - 2.0 * h)
        else:
            # (1-h[i+d]) for valid i
            lo = max(0, -d)
            hi = min(n, n - d)
            g[lo:hi] += (1.0 - h[lo+d:hi+d])
            # -h[i-d] for valid i
            lo2 = max(0, d)
            hi2 = min(n, n + d)
            g[lo2:hi2] += (-h[lo2-d:hi2-d])
    g *= dx / max(1, len(top_indices))
    return g

def fw_vertex(g, n):
    order = np.argsort(g)
    v = np.zeros(n)
    target = n / 2.0
    for idx in order:
        if target <= 1e-15:
            break
        v[idx] = min(1.0, target)
        target -= v[idx]
    return v

def line_search_1d(h, v):
    best_gamma = 0.0
    best_val = eval_c5(h)
    for gamma in np.linspace(0, 1, 16):
        h_try = (1 - gamma) * h + gamma * v
        val = eval_c5(h_try)
        if val < best_val - 1e-12:
            best_val = val
            best_gamma = gamma
    if 0 < best_gamma < 1:
        mid = best_gamma
        for gamma in np.linspace(max(0, mid-0.06), min(1, mid+0.06), 12):
            h_try = (1 - gamma) * h + gamma * v
            val = eval_c5(h_try)
            if val < best_val - 1e-12:
                best_val = val
                best_gamma = gamma
    return (1 - best_gamma) * h + best_gamma * v, best_val

def project_feasible(h, n):
    h = np.clip(h, 0, 1)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-13:
        return h
    # Iterative correction
    for _ in range(10):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            order = np.argsort(h)
            for i in order:
                add = min(diff, 1.0 - h[i])
                if add > 0:
                    h[i] += add
                    diff -= add
                if diff <= 1e-13:
                    break
        else:
            order = np.argsort(-h)
            for i in order:
                sub = min(-diff, h[i])
                if sub > 0:
                    h[i] -= sub
                    diff += sub
                if diff >= -1e-13:
                    break
    return np.clip(h, 0, 1)

def make_binary_like(h, n, rng):
    """Create a binary-like version of h."""
    # Round to nearest, then fix sum
    h_b = (h > 0.5).astype(float)
    # For odd n, need one 0.5
    if n % 2 == 1:
        # Find the coordinate closest to 0.5 in original h
        diffs = np.abs(h - 0.5)
        # Set the one closest to 0.5 to 0.5, round rest
        order = np.argsort(diffs)
        h_b = (h > 0.5).astype(float)
        h_b[order[0]] = 0.5
    # Fix: ensure exactly floor(n/2) or ceil(n/2) ones
    target_int = (n - 1) // 2 if n % 2 == 1 else n // 2
    n_ones = int(np.sum(h_b > 0.5))
    if n_ones > target_int:
        # Remove some ones (the ones with smallest original h)
        one_indices = np.where(h_b > 0.5)[0]
        order = np.argsort(h[one_indices])[::-1]
        for i in range(n_ones - target_int):
            h_b[one_indices[order[i]]] = 0
    elif n_ones < target_int:
        zero_indices = np.where(h_b < 0.5)[0]
        order = np.argsort(-h[zero_indices])
        for i in range(target_int - n_ones):
            h_b[zero_indices[order[i]]] = 1
    if n % 2 == 1:
        # Ensure one 0.5
        if not np.any(np.abs(h_b - 0.5) < 0.01):
            # Find best place for 0.5
            # Try all positions
            best_pos = 0
            best_val = float('inf')
            for pos in range(n):
                h_try = h_b.copy()
                h_try[pos] = 0.5
                # Need to adjust: if we set to 0.5, we need to fix sum
                # Actually let's just set one value to 0.5 and adjust
                pass
            # Simpler: just set the one with h closest to 0.5
            pos = np.argmin(np.abs(h - 0.5))
            h_b[pos] = 0.5
            # Fix sum
            target = n / 2.0
            diff = target - np.sum(h_b)
            if abs(diff) > 1e-10:
                # Binary search for a uniform shift
                # Actually just adjust one binary value
                if diff > 0:
                    # Need to add diff somewhere
                    # Change a 0 to diff (if diff <= 1)
                    zeros = np.where(h_b == 0)[0]
                    if len(zeros) > 0:
                        h_b[zeros[0]] = diff
                    # Or change a 1 to 1-diff
                    ones = np.where(h_b == 1)[0]
                else:
                    ones = np.where(h_b == 1)[0]
                    if len(ones) > 0:
                        h_b[ones[0]] = 1 + diff
    return np.clip(h_b, 0, 1)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()
    
    def time_left():
        return budget_s - (time.time() - t0)
    
    # Start from initial
    h = np.array(initial_h_values, dtype=float)
    n = len(h)
    h = project_feasible(h, n)
    
    best_c5 = eval_c5(h)
    best_h = h.copy()
    
    # Optimization loop
    stall = 0
    max_stall = 30
    
    while time_left() > 30:
        r = rng.random()
        
        if r < 0.35:
            # FW step
            lags = compute_lags(h)
            for top_k in [3, 5, 8]:
                if time_left() < 30:
                    break
                top_idx = np.argsort(-lags)[:top_k]
                g = compute_subgradient(h, top_idx, n)
                v = fw_vertex(g, n)
                h_new, val = line_search_1d(h, v)
                if val < best_c5 - 1e-10:
                    h = h_new
                    best_c5 = val
                    best_h = h.copy()
                    stall = 0
                    break
            else:
                stall += 1
                
        elif r < 0.75:
            # Pair transfer
            improved = False
            for _ in range(30):
                if time_left() < 30:
                    break
                i, j = rng.randint(n, 2)
                if i == j:
                    continue
                # Try several step sizes
                for t in [0.02, 0.05, 0.1, 0.2]:
                    # i -> j
                    if h[i] >= t and h[j] + t <= 1.0:
                        h[i] -= t; h[j] += t
                        val = eval_c5(h)
                        if val < best_c5 - 1e-10:
                            best_c5 = val; best_h = h.copy()
                            improved = True; stall = 0
                            break
                        h[i] += t; h[j] -= t
                    # j -> i
                    if h[j] >= t and h[i] + t <= 1.0:
                        h[j] -= t; h[i] += t
                        val = eval_c5(h)
                        if val < best_c5 - 1e-10:
                            best_c5 = val; best_h = h.copy()
                            improved = True; stall = 0
                            break
                        h[j] += t; h[i] -= t
                    if improved:
                        break
                if improved:
                    break
            if not improved:
                stall += 1
                h = best_h.copy()
        
        else:
            # Reset with perturbation
            sigma = 0.1 + 0.1 * min(stall / 10.0, 1.0)
            h = best_h + rng.normal(0, sigma, n)
            h = project_feasible(h, n)
            # Try FW from here
            lags = compute_lags(h)
            top_idx = np.argsort(-lags)[:5]
            g = compute_subgradient(h, top_idx, n)
            v = fw_vertex(g, n)
            h_new, val = line_search_1d(h, v)
            if val < best_c5:
                best_c5 = val
                best_h = h_new.copy()
                h = h_new.copy()
                stall = 0
            else:
                h = best_h.copy()
                stall += 1
        
        if stall > max_stall:
            # Global restart
            # Try binary-like
            h_b = (best_h >= 0.5).astype(float)
            n_ones = int(np.sum(h_b))
            target_ones = n // 2
            if n % 2 == 0:
                if n_ones > target_ones:
                    idx = np.where(h_b == 1)[0]
                    order = np.argsort(best_h[idx])
                    for i in range(n_ones - target_ones):
                        h_b[idx[order[i]]] = 0
                elif n_ones < target_ones:
                    idx = np.where(h_b == 0)[0]
                    order = np.argsort(-best_h[idx])
