def project_feasible(z, n):
    """Project z onto {h in [0,1]^n : sum(h) = n/2}."""
    target = n / 2.0
    z = np.asarray(z, dtype=float)
    # Replace non-finite
    z = np.where(np.isfinite(z), z, 0.5)

    hi = np.max(z) + 1.0

def compute_grad_smooth(h, beta, weights, corr):
    """Compute gradient of log-sum-exp objective w.r.t. h."""
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    for ell in range(2 * n - 1):
        w = weights[ell]
        if w < 1e-14:

        idx_first = m + d
        np.add.at(grad, idx_first, dx * w * (1.0 - h[m]))

def lse_value(h, beta):
    """Compute log-sum-exp value and weights."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    cmax = np.max(corr)
    shifted = np.clip(beta * (corr - cmax), -50, 0.0)

    total = np.sum(exp_vals)
    if total <= 0:

    w = exp_vals / total
    phi = cmax + np.log(total) / beta
    return phi, w, corr

    n = len(h0)
    h = h0.copy()
    phi, w, corr = lse_value(h, beta)
    alpha = alpha0
    best_local_h = h.copy()
    best_local_phi = phi
    staggard = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        g = compute_grad_smooth(h, beta, w, corr)
        # Project gradient to tangent space (remove mean)
        g_tangent = g - np.mean(g)
        # Line search
        accepted = False
        for ls in range(20):
            a = alpha / (2.0 ** ls)
            trial = project_feasible(h - a * g_tangent, n)
            tri_phi, tri_w, tri_corr = lse_value(trial, beta)
            if tri_phi < phi - 1e-12:
                h = trial
                phi, w, corr = tri_phi, tri_w, tri_corr
                if phi < best_local_phi:
                    best_local_phi = best_local_phi  # track
                    best_local_h = h.copy()
                alpha = min(alpha * 1.3, 0.5)

                staggard = 0
                break

    n = len(h)
    c5, corr = obj_corr(h)
    if time.time() > deadline:
        return h, c5

    active_thresh = c5 - 1e-6

        return h, c5

        A.append(g)
        b.append(1.0)  # corr[ell] + g·delta <= c5_new  => g·delta <= c5_new - corr[ell]

    #             |delta| <= radius

    # A @ delta <= t - corr[active_idx]  =>  A @ delta - t <= -corr[active_idx]

    A_eq = np.zeros((1, num_vars))

def random_transfer(h, rng, n):
    """Make a random two-point transfer preserving sum."""
    h_new = h.copy()
    # Find indices with room to go up and down
    up_candidates = np.where(h < 1.0)[0]
    down_candidates = np.where(h > 0.0)[0]
    if len(up_candidates) == 0 or len(down_candidates) == 0:
        return h_new
    i = rng.integers(len(up_candidates))
    j = rng.integers(len(down_candidates))
    i = up_candidates[i]
    j = down_candidates[j]
    if i == j:
        return h_new
    max_delta = min(h[j], 1.0 - h[i])
    if max_delta <= 0:
        return h_new
    delta = rng.uniform(0.001, max_delta)
    h_new[i] += delta
    h_new[j] -= delta
    return h_new

One key insight: the problem has a certain structure. h: [0,2] → [0,1] with integral 1 means the "average" value of h is 1/2 over [0,2]. The overlap integral ∫ h(x)(1-h(x+k)) dx measures how much h and its shifted complement overlap. We want to minimize the maximum of this over all shifts k.

For k=0: ∫ h(x)(1-h(x)) dx = ∫(h - h²) dx. This is maximized when h is 0 or 1 (making it 0) or... wait, h(1-h) ≤ 1/4, so this integral is at most 1/4 * 2 = 1/2.
