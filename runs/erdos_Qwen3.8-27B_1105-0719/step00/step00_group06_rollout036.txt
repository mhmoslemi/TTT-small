import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    c = np.correlate(h, g, mode="full")
    return float(np.max(c)) * dx

def c5_value(h):
    """Same as compute_c5 but returns the raw max (before dx scaling)"""
    g = 1.0 - h
    c = np.correlate(h, g, mode="full")
    return float(np.max(c))

def make_feasible(h, target_sum):
    """Ensure h is in [0,1] and sums to target_sum"""
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s == 0:
        h = np.full(len(h), target_sum / len(h))
    else:
        # Water-filling approach
        # If s > target: need to reduce
        # If s < target: need to increase
        diff = target_sum - s
        if diff > 0:
            # Add to elements that are < 1
            mask = h < 1.0
            add = diff / mask.sum() if mask.sum() > 0 else 0
            h[mask] = np.minimum(1.0, h[mask] + add)
            # If still not enough, distribute remaining
            remaining = target_sum - h.sum()
            if remaining > 1e-12:
                mask2 = h < 1.0
                if mask2.sum() > 0:
                    h[mask2] += remaining / mask2.sum()
                    h[mask2] = np.minimum(1.0, h[mask2])
        else:
            # Reduce from elements that are > 0
            mask = h > 0.0
            sub = -diff / mask.sum() if mask.sum() > 0 else 0
            h[mask] = np.maximum(0.0, h[mask] - sub)
            remaining = h.sum() - target_sum
            if remaining > 1e-12:
                mask2 = h > 0.0
                if mask2.sum() > 0:
                    h[mask2] -= remaining / mask2.sum()
                    h[mask2] = np.maximum(0.0, h[mask2])
    return h

def golden_section_1d(f, a, b, tol=1e-10, max_iter=40):
    """Golden section minimization of f on [a, b]"""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c_ = b - gr * (b - a)
    d_ = a + gr * (b - a)
    fc = f(c_)
    fd = f(d_)
    for _ in range(max_iter):
        if b - a < tol:
            break
        if fc < fd:
            b = d_
            d_ = c_
            fd = fc
            c_ = b - gr * (b - a)
            fc = f(c_)
        else:
            a = c_
            c_ = d_
            fc = fd
            d_ = a + gr * (b - a)
            fd = f(d_)
    return (a + b) / 2.0

    """Line search along h[u]+=s, h[v]-=s. Returns (s_opt, improved_h, c5_val)"""
    s_min = max(-h[u], h[v] - 1.0)
    s_max = min(1.0 - h[u], h[v])

        h[v] -= s

        h[u] -= s
        h[v] += s
        return val

    h[u] = np.clip(h[u], 0.0, 1.0)

        h[v] += s_opt

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    c = np.correlate(h, g, mode="full")
    return float(np.max(c)) * dx

def golden_section_min(f, a, b, max_iter=30):
    """Minimize f on [a,b] using golden section search. f should be unimodal."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1 = f(c1)
    f2 = f(c2)
    for _ in range(max_iter):
        if f1 < f2:
            b = c2
            c2 = c1
            f2 = f1
            c1 = b - gr * (b - a)
            f1 = f(c1)
        else:
            a = c1
            c1 = c2
            f1 = f2
            c2 = a + gr * (b - a)
            f2 = f(c2)
    return (a + b) * 0.5

def pair_update(h, u, v, max_iter=25):
    """Try moving mass between positions u and v. Updates h in place if improved.
       Returns the new C5 value."""
    n = len(h)
    dx = 2.0 / n
    s_lo = max(-h[u], h[v] - 1.0)
    s_hi = min(1.0 - h[u], h[v])
    if s_hi - s_lo < 1e-12:
        return compute_c5(h)
    
    h_u = h[u]
    h_v = h[v]
    cur = compute_c5(h)
    
    def obj(s):
        h[u] = h_u + s
        h[v] = h_v - s
        g = 1.0 - h
        c = np.correlate(h, g, mode="full")
        return float(np.max(c)) * dx
    
    s_best = golden_section_min(obj, s_lo, s_hi, max_iter=max_iter)
    h[u] = h_u + s_best
    h[v] = h_v - s_best
    new_val = compute_c5(h)
    
    if new_val >= cur - 1e-14:
        h[u] = h_u
        h[v] = h_v
        return cur
    return new_val

    elif diff < 0:
        ones = np.where(b == 1)[0]
        if len(ones) >= -diff:
            b[rng.choice(ones, size=-diff, replace=False)] = 0.0
    return b

    """Generate a smooth seed"""
    # Random smooth function via random Fourier coefficients
    x = np.linspace(0, 2, n)
    h = np.ones(n) * 0.5

            h[mask] += step

    return h
