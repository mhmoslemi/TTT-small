    n = len(h)

            order = np.argsort(x)  # smallest first (most slack)

def project_feasible(h, target):
    """Project onto {x: 0≤x≤1, sum(x)=target}"""
    n = len(h)
    # Binary search for shift
    lo, hi = -2.0, 2.0
    for _ in range(100):

        s = np.sum(np.clip(h - mid, 0, 1))
        if s > target:

        else:

    # Fix residual

    if abs(resid) > 1e-12:

            for i in idx:
                add = min(resid, 1.0 - x[i])
                x[i] += add
                resid -= add
                if resid <= 1e-12:
                    break
        else:
            resid = -resid
            idx = np.argsort(-x)
            for i in idx:
                sub = min(resid, x[i])
                x[i] -= sub
                resid -= sub
                if resid <= 1e-12:
                    break
    return x

    """Gradient of R[k] w.r.t. h"""
    n = len(h)
    offset = k - (n - 1)
    g = np.zeros(n)
    # First term: h[n]*(1-h[n+offset]), p=n
    lo1 = max(0, -offset)
    hi1 = min(n, n - offset)
    if lo1 < hi1:
        g[lo1:hi1] += (1 - h[lo1 + offset:hi1 + offset])
    # Second term: -h[n-offset] from 1-h[n+offset] when n+offset=p
    lo2 = max(0, offset)
    hi2 = min(n, n + offset)
    if lo2 < hi2:
        g[lo2:hi2] -= h[lo2 - offset:hi2 - offset]
    return g

def compute_R(h):
    return np.correlate(h, 1.0 - h, mode='full')

def get_C5(h):
    n = len(h)
    return float(np.max(compute_R(h))) * 2.0 / n

def project_feasible(h, target):
    n = len(h)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    resid = target - np.sum(x)
    if abs(resid) > 1e-12:
        if resid > 0:
            for i in np.argsort(x):
                a = min(resid, 1.0 - x[i])
                x[i] += a
                resid -= a
                if resid <= 1e-12:
                    break
        else:
            resid = -resid
            for i in np.argsort(-x):
                a = min(resid, x[i])
                x[i] -= a
                resid -= a
                if resid <= 1e-12:
                    break
    return x

def tm_binary(n):
    h = np.zeros(n)
    for i in range(n):
        x, c = i, 0
        while x:
            c += x & 1
            x >>= 1
        h[i] = 1.0 if c % 2 == 0 else 0.0
    t = n // 2
    s = int(np.sum(h))
    if s < t:
        z = np.where(h == 0)[0]
        h[z[:t - s]] = 1.0
    elif s > t:
        o = np.where(h == 1)[0]

    return h
