```python
    grad = np.zeros(n)
    for s in active:
        grad += dx * gradient_of_lag(h, s)
    grad /= len(active)
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    R = np.correlate(h, g, mode="full")
    return dx * np.max(R)

def project_feasible(v, n):
    """Project v onto {h : 0<=h<=1, sum(h)=n/2}."""
    target = n / 2.0
    # Bisection on tau
    lo, hi = -1.0, 2.0
    for _ in range(100):
        tau = (lo + hi) / 2.0
        h = np.clip(v - tau, 0.0, 1.0)
        s = h.sum()
        if s > target:
            lo = tau
        else:
            hi = tau
    tau = (lo + hi) / 2.0
    return np.clip(v - tau, 0.0, 1.0)

def smoothed_max_gradient(h, epsilon):
    """Gradient of dx * log-sum-exp(R) / n_lags, which approximates the max."""
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    R = dx * np.correlate(h, g, mode="full")
    # Shift for numerical stability
    R_shift = R - np.max(R)
    weights = np.exp(R_shift / epsilon)
    weights /= weights.sum()
    
    grad = np.zeros(n)
    # For each lag s with non-negligible weight, add its gradient
    # To avoid loop over all 2n-1 lags, only use significant ones
    thresh = weights.max() * 0.01
    active = np.where(weights > thresh)[0]
    
    for s in active:
        k = s - (n - 1)
        # grad_lag_k
        i_start = max(0, k)
        i_end = min(n, n + k)
        if i_start < i_end:
            grad[i_start:i_end] += weights[s] * g[i_start - k:i_end - k]
        i_start2 = max(0, -k)
        i_end2 = min(n, n - k)
        if i_start2 < i_end2:
            grad[i_start2:i_end2] -= weights[s] * h[i_start2 + k:i_end2 + k]
    
    return grad

def local_optimize(h, max_iters=1000, epsilon_init=0.01, epsilon_min=0.001):
    """Local optimization using smoothed max."""
    n = len(h)
    best_c5 = evaluate_c5(h)
    best_h = h.copy()
    epsilon = epsilon_init
    step_size = 0.1
    
    for it in range(max_iters):
        grad = smoothed_max_gradient(h, epsilon)
        
        # Project gradient step
        v = h - step_size * grad
        h_new = project_feasible(v, n)
        
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
            # Decrease step size slowly
            step_size *= 0.999
        else:
            step_size *= 0.5
            if step_size < 1e-10:
```

```python
import numpy as np
from scipy.optimize import linprog
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    R = np.correlate(h, g, mode="full")
    return dx * np.max(R)

def project_feasible(v, n):
    target = n / 2.0
    lo, hi = -1.5, 1.5
    for _ in range(80):
        tau = (lo + hi) / 2.0
        h = np.clip(v - tau, 0.0, 1.0)
        if h.sum() > target:
            lo = tau
        else:
            hi = tau
    tau = (lo + hi) / 2.0
    return np.clip(v - tau, 0.0, 1.0)

def all_lag_correlations(h):
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    R = np.correlate(h, g, mode="full")
    return dx * R

def grad_lag(h, s):
    """Gradient of R_s (scaled by dx) with respect to h."""
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    k = s - (n - 1)
    grad = np.zeros(n)
    # First term: g[i-k] for valid i
    i_start = max(0, k)
    i_end = min(n, n + k)
    if i_start < i_end:
        grad[i_start:i_end] += g[i_start - k:i_end - k]
    # Second term: -h[i+k] for valid i
    i_start2 = max(0, -k)
    i_end2 = min(n, n - k)
    if i_start2 < i_end2:
        grad[i_start2:i_end2] -= h[i_start2 + k:i_end2 + k]
    return dx * grad

def lp_step(h, lag_values, n, n_active=10, trust_radius=0.1):
    """Solve the tangent LP to get a step direction."""
    # Find active lags
    max_val = np.max(lag_values)
    tol = 0.01 * max_val + 1e-6
    active = np.where(lag_values >= max_val - tol)[0]
    if len(active) > n_active:
        # Keep the most active
        active = active[np.argsort(lag_values[active])[::-1][:n_active]]
    
    # Build LP: minimize t
    # Variables: mu (n), t (1) -> total n+1
    # Constraints:
    # For each active s: grad_s . mu - t <= -lag_values[s]
    #   i.e., grad_s . mu - t + lag_values[s] <= 0
    # sum(mu) = 0
    # h[i] - trust_radius <= h[i] + mu[i] <= h[i] + trust_radius AND 0 <= h[i]+mu[i] <= 1
    #   i.e., max(0 - h[i], -trust_radius) <= mu[i] <= min(1 - h[i], trust_radius)
    
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize t
    
    A_ub = np.zeros((len(active), n + 1))
    b_ub = np.zeros(len(active))
    for idx, s in enumerate(active):
        A_ub[idx, :n] = grad_lag(h, s)
        A_ub[idx, n] = -1.0
        b_ub[idx] = -lag_values[s]
    
    # Equality: sum(mu) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = [0.0]
    
    # Bounds on mu
    bounds = []
    for i in range(n):
        lo = max(0.0 - h[i], -trust_radius)
        hi = min(1.0 - h[i], trust_radius)
        bounds.append((lo, hi))
    bounds.append((None, None))  # t is free
    # Actually, t should be >= 0 or at least >= some lower bound
    # t >= min(lag_values) - some margin to keep it reasonable
    bounds[n] = (-0.5, None)
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                     bounds=bounds, method='highs')
    
    if result.success:
        mu = result.x[:n]
        t_val = result.x[n]
    else:
        mu = None
    
    return mu, t_val
```

```python
    best_c5 = evaluate_c5(best_h)
    
    # Phase 1: Optimize at original n
    if remaining() > 60:
        alloc = min(remaining() * 0.25, 200.0)
        h_opt, c5_opt = combined_optimize(best_h.copy(), time_budget=alloc, seed=seed)
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt
    
    # Phase 2: Try different n values
    n_list = []
    for n_try in [n0 + 5 if n0 % 2 == 1 else n0 + 10, n0 * 2, n0 * 2 + 10, n0 * 4]:
        if 20 <= n_try <= 720 and n_try not in n_list:
            n_list.append(n_try)
    # Add some fixed sizes
    for n_fix in [50, 90, 150, 240, 360, 480]:
        if n_fix not in n_list and 20 <= n_fix <= 720:
            n_list.append(n_fix)
    n_list.sort()
    
    for n_try in n_list:
        if remaining() < 40:
            break
        alloc = min(remaining() * 0.15, 80.0)
        if alloc < 10:
            alloc = min(remaining() - 20, 30.0)
            if alloc < 5:
                continue
        
        # Initialize
        if n_try == len(best_h):
            h_try = best_h.copy()
        else:
            h_try = interpolated_h(best_h, n_try)
        
        # Optimize
        h_opt, c5_opt = combined_optimize(h_try, time_budget=alloc, seed=seed + n_try)
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt
    
    # Phase 3: Binary constructions for even n
    for n_bin in [50, 90, 150, 240, 360]:
        if remaining() < 30:
            break
        alloc = min(remaining() * 0.1, 50.0)
        if alloc < 8:
            continue
        for attempt in range(3):
            if remaining() < 10:
                break
            h_bin = binary_feasible(n_bin, seed=seed + 1000 + n_bin * 7 + attempt)
            h_opt, c5_opt = combined_optimize(h_bin, time_budget=alloc / 3.0, seed=seed + n_bin)
            if c5_opt < best_c5:
                best_h = h_opt
                best_c5 = c5_opt
    
    # Phase 4: Random restarts at best n
    best_n = len(best_h)
    restart_count = 0
    while remaining() > 20 and restart_count < 50:
        restart_count += 1
        # Alternate between best_n and nearby sizes
        if restart_count % 3 == 0:
            n_restart = best_n + rng.choice([-10, -5, 5, 10, 20])
            n_restart = max(20, min(720, n_restart))
        else:
            n_restart = best_n
        
        h_rand = random_feasible(n_restart, seed=rng.randint(1000000))
        alloc = min(remaining() * 0.05, 20.0)
        if alloc < 5:
            break
        h_opt, c5_opt = combined_optimize(h_rand, time_budget=alloc, seed=rng.randint(1000000))
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt
            best_n = n_restart
    
    # Phase 5: Final polishing
    if remaining() > 10:
        alloc = remaining() - 5
        # Alternate LP and pair transfer
        for _ in range(3):
            if remaining() < 5:
                break
            h_opt, c5_opt = optimize_lp(best_h.copy(), time_budget=alloc * 0.3)
            if c5_opt < best_c5:
                best_h = h_opt
                best_c5 = c5_opt
            if remaining() < 5:
                break
            h_opt, c5_opt = pair_transfer_optimize(best_h.copy(), time_budget=alloc * 0.2, seed=rng.randint(10000))
            if c5_opt < best_c5:
                best_h = h_opt
                best_c5 = c5_opt
    
    # Ensure feasibility
    best_h = project_feasible(best_h, len(best_h))
    best_c5 = evaluate_c5(best_h)
    
    return best_h, best_c5, len(best_h)
```
