import numpy as np
from scipy.optimize import minimize
import time
import math

def exact_objective(h):
    """Compute C5 = max(np.correlate(h, 1-h, mode='full')) * dx"""
    n = len(h)
    if n == 0:
        return 0.0, np.array([0.0])
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(profile)), profile

def project_box_sum(h):
    """Project h to [0,1]^n with sum = n/2 using water-filling."""
    n = len(h)
    target = n / 2.0
    h = np.asarray(h, dtype=np.float64).copy()
    h[~np.isfinite(h)] = 0.0
    h = np.clip(h, 0.0, 1.0)
    
    for _ in range(100):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            # Need to increase sum
            mask = h < 1.0
            cnt = np.sum(mask)
            if cnt == 0:
                break
            step = diff / cnt
            h[mask] = np.minimum(1.0, h[mask] + step)
        else:
            # Need to decrease sum
            mask = h > 0.0
            cnt = np.sum(mask)
            if cnt == 0:
                break
            step = -diff / cnt
            h[mask] = np.maximum(0.0, h[mask] - step)
    
    # Final exact fix
    s = np.sum(h)
    diff = target - s
    if abs(diff) > 1e-14:
        if diff > 0:
            mask = h < 1.0
            idx = np.where(mask)[0]
            if len(idx) > 0:
                h[idx[0]] = min(1.0, h[idx[0]] + diff)
        else:
            mask = h > 0.0
            idx = np.where(mask)[0]
            if len(idx) > 0:
                h[idx[0]] = max(0.0, h[idx[0]] + diff)
    
    return h

def smooth_objective_and_grad(h, tau):
    """Compute log-sum-exp smoothed objective and its gradient."""
    n = len(h)
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full')
    profile_scaled = profile * dx
    
    pmax = np.max(profile_scaled)
    shifted = (profile_scaled - pmax) / tau
    # Clip for numerical stability
    shifted = np.clip(shifted, -500, 0)
    exp_terms = np.exp(shifted)
    log_sum = np.log(np.sum(exp_terms))
    
    value = pmax + tau * log_sum
    
    # Weights
    weights = exp_terms / np.sum(exp_terms)
    
    # Gradient: sum_k w_k * dx * d(profile[k])/dh
    grad = np.zeros(n)
    
    # For output index k, the lag is s = k - (n-1)
    # If s >= 0: profile[k] = sum_{j=0}^{n-1-s} h[j+s]*(1-h[j])
    #   d/dh[i] = (1-h[i-s])*[s<=i<=n-1] - h[i+s]*[0<=i<=n-1-s]
    # If s < 0, let a = -s: profile[k] = sum_{j=0}^{n-1-a} h[j]*(1-h[j+a])
    #   d/dh[i] = (1-h[i+a])*[0<=i<=n-1-a] - h[i-a]*[a<=i<=n-1]
    
    nprof = len(profile)
    for k in range(nprof):
        wk = weights[k] * dx
        if wk < 1e-15:
            continue
        s = k - (n - 1)
        if s >= 0:
            # d/dh[i] = (1-h[i-s])*[s<=i<=n-1] - h[i+s]*[0<=i<=n-1-s]
            for i in range(s, n):
                grad[i] += wk * (1.0 - h[i - s])
            for i in range(0, n - s):
                grad[i] -= wk * h[i + s]
        else:
            a = -s
            # d/dh[i] = (1-h[i+a])*[0<=i<=n-1-a] - h[i-a]*[a<=i<=n-1]
            for i in range(0, n - a):
                grad[i] += wk * (1.0 - h[i + a])
            for i in range(a, n):
                grad[i] -= wk * h[i - a]
    
    return value, grad

def optimize_single(h_start, n, budget_remaining):
    """Run SLSQP optimization on a single starting point."""
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0, 
                    'jac': lambda x: np.ones(n)}]
    
    best_h = project_box_sum(h_start)
    best_val, _ = exact_objective(best_h)
    
    tau_stages = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    
    current_h = best_h.copy()
    
    for tau in tau_stages:
        if budget_remaining() < 5.0:
            break
        
        def obj(x):
            v, _ = smooth_obj_grad(x, tau)
            # Add penalty for constraint violation
            sum_err = abs(np.sum(x) - n/2.0)
            box_viol = np.sum(np.maximum(0, x-1.0)**2) + np.sum(np.maximum(0, -x)**2)
            return v + 1e4 * (sum_err**2 + box_viol)
        
        def obj_grad(x):
            v, g = smooth_obj_grad(x, tau)
            sum_err = np.sum(x) - n/2.0
            g += 2e4 * sum_err * np.ones(n)
            box_hi = np.maximum(0, x - 1.0)
            box_lo = np.maximum(0, -x)
            g += 2e4 * (box_hi - box_lo)
            return g
        
        try:
            res = minimize(obj, current_h, jac=obj_grad, method='SLSQP',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': 200, 'ftol': 1e-12})
            candidate = project_box_sum(res.x)
            cand_val, _ = exact_objective(candidate)
            if cand_val < best_val - 1e-12:
                best_val = cand_val
                best_h = candidate
                current_h = candidate.copy()
        except Exception:
            pass
    
    return best_h, best_val

# Global state for SLSQP callbacks
_g_tau = 0.01
_g_n = 0

def _slsqp_obj(x):
    v, _ = _smooth_h(x, _g_tau)
    n = _g_n
    se = np.sum(x) - n / 2.0
    bv = np.sum(np.maximum(0.0, x - 1.0)**2) + np.sum(np.maximum(0.0, -x)**2)
    return v + 1e4 * (se*se + bv)

    v, g = _smooth_h(x, _g_tau)
    n = _g_n
    se = np.sum(x) - n / 2.0
    g = g + 2e4 * se * np.ones(n)
    g = g + 2e4 * (np.maximum(0.0, x - 1.0) - np.maximum(0.0, -x))
    return g

def _eq_fun(x):
    return np.sum(x) - _g_n / 2.0

    return np.ones(_g_n)

def exact_objective(h):
    n = len(h)
    if n == 0:
        return 0.0
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(profile))

def project_box_sum(h):
    n = len(h)
    target = n / 2.0

def _smooth_h(h, tau):
    n = len(h)
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    pmax = np.max(profile)
    shifted = np.minimum((profile - pmax) / tau, 0.0)
    exp_terms = np.exp(shifted)
    sum_exp = np.sum(exp_terms)
    if sum_exp < 1e-300:
        sum_exp = 1e-300
    weights = exp_terms / sum_exp
    value = pmax + tau * np.log(sum_exp)
    
    grad = np.zeros(n)
    for k in range(len(profile)):
        w = weights[k] * dx
        if w < 1e-16:
            continue
        s = k - (n - 1)
        if s >= 0:
            if s < n:
                grad[s:n] += w * (1.0 - h[0:n-s])
                grad[0:n-s] -= w * h[s:n]
        else:
            a = -s
            if a < n:
                grad[0:n-a] += w * (1.0 - h[a:n])
                grad[a:n] -= w * h[0:n-a]
    return value, grad

def _resample(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return project_box_sum(h_old)
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return project_box_sum(h_new)

def _generate_starts(n, parent_h, rng, max_starts):

    if parent_h is not None:

def _binary_postprocess(h, n, max_rounds=200):
    """Threshold to binary and do swap improvements."""
    if n % 2 != 0:
        return h, exact_objective(h)
    
    b = np.zeros(n)
    target_ones = n // 2
    idx = np.argsort(h)[::-1]
    b[idx[:target_ones]] = 1.0
    b_val = exact_objective(b)
    
    if b_val >= exact_objective(h) + 1e-10:
        return h, exact_objective(h)
    
    best_b = b.copy()
    best_val = b_val
    improved = True
    rounds = 0
    while improved and rounds < max_rounds:
        improved = False
        rounds += 1
        ones = np.where(best_b == 1.0)[0]
        zeros = np.where(best_b == 0.0)[0]
        for i in ones:
            for j in zeros:
                trial = best_b.copy()
                trial[i] = 0.0
                trial[j] = 1.0
                v = exact_objective(trial)
                if v < best_val - 1e-12:
                    best_val = v
                    best_b = trial
                    improved = True
                    break
            if improved:
                break
    return best_b, best_val

def _random_local_search(h, n, n_iter, rng, time_limit):
    """Random pair moves preserving sum."""
    best_h = h.copy()
    best_val = exact_objective(best_h)
    for _ in range(n_iter):
        if time.time() > time_limit:
            break

            continue
        v = exact_objective(trial)
        if v < best_val - 1e-12:
            best_val = v
            best_h = trial
    return best_h, best_val
