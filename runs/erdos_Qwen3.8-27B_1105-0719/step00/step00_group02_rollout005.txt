def project_box_sum(x, s):
    n = len(x)
    x = np.asarray(x, dtype=float)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        y = np.clip(x - mid, 0.0, 1.0)
        if y.sum() > s:
            lo = mid
        else:
            hi = mid
    y = np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)
    diff = s - y.sum()
    if abs(diff) > 1e-10:
        interior = (y > 1e-8) & (y < 1.0 - 1e-8)
        if interior.sum() > 0:
            y[interior] += diff / interior.sum()
            y = np.clip(y, 0.0, 1.0)
        else:
            y += diff / n
            y = np.clip(y, 0.0, 1.0)
    return y

def exact_eval(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * dx)

def compute_lse_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_c = corr.max()
    exps = np.exp((corr - max_c) / tau)
    w = exps / exps.sum()
    grad = np.zeros(n)
    for m in range(n):
        dm = 0.0
        for k in range(2 * n - 1):
            lag = k - (n - 1)
            j = m + lag
            if 0 <= j < n:
                dm += w[k] * (1.0 - h[j])
            j2 = m - lag
            if 0 <= j2 < n:
                dm -= w[k] * h[j2]
        grad[m] = dx * dm
    return grad
