            h[rng.choice(idx, n_ones - n_target, replace=False)] = 0
        else:
            idx = np.where(h == 0)[0]
            h[rng.choice(idx, n_target - n_ones, replace=False)] = 1
    return h

def transfer(h_src, n_dst):
    """Transfer a solution from one dimension to another via resampling."""
    n_src = len(h_src)
    # Create piecewise constant interpolation
    x_src = np.linspace(0, 2, n_src, endpoint=False) + 1.0/n_src
    x_dst = np.linspace(0, 2, n_dst, endpoint=False) + 1.0/n_dst
    h_dst = np.interp(x_dst, x_src, h_src, period=2.0)
    # Actually, simpler: just use linear interpolation
    h_dst = np.interp(x_dst, x_src, h_src)
    # Project and round
    h_dst = project_box_sum(h_dst, n_dst / 2.0)
    return h_dst
