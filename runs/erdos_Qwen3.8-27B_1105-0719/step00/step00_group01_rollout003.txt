import numpy as np
from scipy.optimize import linprog
import time

def objective_exact(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr)), corr

def project_to_feasible(z, target):
    n = len(z)
    z = np.where(np.isfinite(z), z, 0.5)
    lo, hi = z.min() - 1.0, z.max() + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(z - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5*(lo+hi), 0.0, 1.0)
    # Exact sum fix
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        idx = np.where(mask)[0]
        if len(idx) > 0:
            h[idx] += diff / len(idx)
            h = np.clip(h, 0, 1)
    return h

def soft_max_and_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    Fmax = np.max(corr)
    shifted = np.clip(beta * (corr - Fmax), -500.0, 0.0)
    exps = np.exp(shifted)
    sum_exp = np.sum(exps)
    if sum_exp < 1e-300:
        weights = np.zeros(len(corr))
        weights[np.argmax(corr)] = 1.0
    else:
        weights = exps / sum_exp
    phi = Fmax + np.log(max(sum_exp, 1e-300)) / beta
    
    grad = np.zeros(n)
    omh = 1.0 - h
    for ell in range(len(corr)):
        w = weights[ell]
        if w < 1e-10:
            continue
        d = ell - (n - 1)
        if d > 0:
            grad[d:] += dx * w * omh[:n-d]
            grad[:n-d] -= dx * w * h[d:]
        elif d < 0:
            dd = -d
            grad[:n-dd] += dx * w * omh[dd:]
            grad[dd:] -= dx * w * h[:n-dd]
        else:
            grad += dx * w * (1.0 - 2.0 * h)
    return phi, grad

def active_lp(h_cur, c_cur, corr_cur, active_indices, radius, n):
    """Solve trust-region LP to find descent direction"""
    dx = 2.0 / n
    # Build gradient matrix for active lags
    A_rows = []
    b_rows = []
    for ell in active_indices:
        d = ell - (n - 1)
        g = np.zeros(n)
        omh = 1.0 - h_cur
        if d > 0:
            g[d:] += dx * omh[:n-d]
            g[:n-d] -= dx * h_cur[d:]
        elif d < 0:
            dd = -d
            g[:n-dd] += dx * omh[dd:]
            g[dd:] -= dx * h_cur[:n-dd]
        else:
            g = dx * (1.0 - 2.0 * h_cur)
        # Constraint: g · x - t <= -F_ell + g · h_cur
        f_ell = corr_cur[ell]
        rhs = -f_ell + np.dot(g, h_cur)
        row = np.zeros(n + 1)
        row[:n] = g
        row[n] = -1.0
        A_rows.append(row)
        b_rows.append(rhs)
    
    A_ub = np.array(A_rows)
    b_ub = np.array(b_rows)
    
    # Equality: sum(x) = n/2
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    # Objective: minimize t
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    # Bounds: h_cur[i] - radius <= x[i] <= h_cur[i] + radius, intersected with [0,1]
    bounds = [(max(0, h_cur[i] - radius), min(1, h_cur[i] + radius)) for i in range(n)]
    bounds.append((0, 2.0))  # t can be anything non-negative
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n]
    return None

import numpy as np
from scipy.optimize import linprog
import time

def objective_exact(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr)), corr

def project_to_feasible(z, target):
    n = len(z)
    z = np.where(np.isfinite(z), z, 0.5)
    lo, hi = z.min() - 1.0, z.max() + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(z - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        idx = np.where(mask)[0]
        if len(idx) > 0:
            h[idx] += diff / len(idx)
            h = np.clip(h, 0.0, 1.0)
    return h

def lag_gradient(h, d):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    omh = 1.0 - h
    if d > 0:
        g[d:] += omh[:n - d]
        g[:n - d] -= h[d:]
    elif d < 0:
        dd = -d
        g[:n - dd] += omh[dd:]
        g[dd:] -= h[:n - dd]
    else:
        g = 1.0 - 2.0 * h
    return dx * g

def soft_max_and_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    Fmax = np.max(corr)
    shifted = np.clip(beta * (corr - Fmax), -500.0, 0.0)
    exps = np.exp(shifted)
    sum_exp = np.sum(exps)
    if sum_exp < 1e-300:
        weights = np.zeros(len(corr))
        weights[np.argmax(corr)] = 1.0
    else:
        weights = exps / sum_exp
    phi = Fmax + np.log(max(sum_exp, 1e-300)) / beta
    grad = np.zeros(n)
    for ell in range(len(corr)):
        w = weights[ell]
        if w < 1e-10:
            continue
        d = ell - (n - 1)
        g = lag_gradient(h, d)
        grad += w * g
    return phi, grad

def solve_active_lp(h_cur, corr_cur, active_idx, radius, n):
    dx = 2.0 / n
    A_rows = []
    b_rows = []
    for ell in active_idx:
        d = ell - (n - 1)
        g = lag_gradient(h_cur, d)
        f_ell = corr_cur[ell]
        rhs = -f_ell + np.dot(g, h_cur)
        row = np.zeros(n + 1)
        row[:n] = g
        row[n] = -1.0
        A_rows.append(row)
        b_rows.append(rhs)
    A_ub = np.array(A_rows)
    b_ub = np.array(b_rows)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    bounds = [(max(0.0, h_cur[i] - radius), min(1.0, h_cur[i] + radius)) for i in range(n)]
    bounds.append((0.0, 3.0))
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None
