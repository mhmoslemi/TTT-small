from scipy.optimize import minimize
import time

_g_n = 0
_g_beta = 1.0

def _lse_max(x, beta):
    mx = np.max(x)
    return mx + np.log(np.sum(np.exp(beta * (x - mx)))) / beta

def _smooth_obj(z):
    n = _g_n
    beta = _g_beta
    h = np.empty(n)
    h[:-1] = z
    h[-1] = n / 2.0 - np.sum(z)
    pen = 0.0
    if h[-1] < 0:
        pen = 1e4 * h[-1] ** 2
    elif h[-1] > 1:
        pen = 1e4 * (h[-1] - 1) ** 2
    corr = np.correlate(h, 1.0 - h, mode='full')
    return _lse_max(corr, beta) + pen

def compute_c5(h):
    n = len(h)
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = np.min(h) - 1, np.max(h) + 1
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h - mid, 0, 1))
        if s > target:
            lo = mid
        else:
            hi = mid
    h_out = np.clip(h - (lo + hi) / 2, 0, 1)
    diff = target - np.sum(h_out)
    if abs(diff) > 1e-12:
        mask = (h_out > 1e-9) & (h_out < 1 - 1e-9)
        if np.any(mask):
            h_out[mask] += diff / np.sum(mask)
            h_out = np.clip(h_out, 0, 1)
    return h_out

        while improved and time.time() < time_limit:
            improved = False
            for _ in range(500):
                i = rng.integers(0, n)
                j = rng.integers(0, n)

                for sign in (1, -1):

                    h[i] -= sign * eps
                    h[j] += sign * eps
                    c = compute_c5(h)
                    if c < best - 1e-14:
                        best = c
                        improved = True
                        break
                    else:
                        h[i] += sign * eps
                        h[j] -= sign * eps
                if improved:
                    break
    return h, best

                j = int(rng.integers(0, n))
                if i == j:
                    continue
                for sgn in (1, -1):
                    hi = h[i] - sgn * step
                    hj = h[j] + sgn * step
                    if hi < 0 or hi > 1 or hj < 0 or hj > 1:
                        continue
                    h[i] -= sgn * step
                    h[j] += sgn * step
                    c = compute_c5(h)
                    if c < best - 1e-14:
                        best = c
                        improved = True
                        no_improve = 0
                        break
                    else:
                        h[i] += sgn * step
                        h[j] -= sgn * step
                if improved:
                    break
            if not improved:
                no_improve += 150
    return h, best


def _structured_local_search(h, time_limit, seed):
    """Try all pairs systematically when step is large"""
    rng = np.random.default_rng(seed)
    n = len(h)
    best = compute_c5(h)

    for step in [1.0 / n, 2.0 / n, 0.5 / n]:
        for _ in range(50):
            if time.time() > time_limit:
                return h, best
            improved = False
            for i in range(n):
                for j in range(n):
                    if i == j:
                        continue
                    for sgn in (1, -1):
                        hi = h[i] - sgn * step
                        hj = h[j] + sgn * step
                        if hi < 0 or hi > 1 or hj < 0 or hj > 1:
                            continue
                        h[i] -= sgn * step
                        h[j] += sgn * step
                        c = compute_c5(h)
                        if c < best - 1e-14:
                            best = c
                            improved = True
                        else:
                            h[i] += sgn * step
                            h[j] -= sgn * step
            if not improved:
                break
    return h, best


def _golden_ratio_h(n):
    """Generate a balanced binary-like sequence using golden ratio"""
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    h = np.zeros(n)
    for i in range(n):
        h[i] = 1.0 if (int(i * phi) % 2) == 0 else 0.0
    h = project_feasible(h, n)
    return h


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s

    try:
        h0 = np.array(initial_h_values, dtype=float)
    except Exception:
        h0 = None

    if h0 is not None:
        best_h = h0.copy()
        best_c5 = compute_c5(h0)
    else:
        best_h = np.full(45, 0.5)
        best_c5 = compute_c5(best_h)

    n_orig = len(best_h)

    # Phase 1: Scipy optimization on original n with multiple stages
    d1 = t0 + budget_s * 0.35
    best_h, best_c5 = _optimize_scipy(best_h, n_orig, d1,
                                       [3, 5, 8, 12, 20, 35, 60, 100, 200,
                                        400, 800, 1500, 3000, 5000])

    # Random restarts on same n
    for restart in range(4):
        if time.time() > d1:
            break
        rng_r = np.random.default_rng(seed + 1000 * (restart + 1))
        h_rand = rng_r.uniform(0.0, 1.0, n_orig)
        h_rand = project_feasible(h_rand, n_orig)
        h_res, c5_res = _optimize_scipy(h_rand, n_orig, d1,
                                         [20, 100, 500, 2000, 5000])
        if c5_res < best_c5:
            best_c5 = c5_res
            best_h = h_res

    # Try golden ratio pattern
    if time.time() < d1:
        h_gold = _golden_ratio_h(n_orig)
        h_res, c5_res = _optimize_scipy(h_gold, n_orig, d1,
                                         [10, 50, 200, 1000, 3000])
        if c5_res < best_c5:
            best_c5 = c5_res
            best_h = h_res

    # Phase 2: Try other n values
    d2 = t0 + budget_s * 0.6
    for n_try in [44, 46, 50, 40, 36, 54, 48, 32, 60, 64, 42, 48, 52]:
        if time.time() > d2:
            break
        if n_try == n_orig or n_try < 16:
            continue
        x_old = np.linspace(0.0, 2.0, len(best_h), endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, n_try)
        h_res, c5_res = _optimize_scipy(h_interp, n_try, d2,
                                         [10, 50, 200, 800, 2000, 5000])
        if c5_res < best_c5:
            best_c5 = c5_res
            best_h = h_res

    # Phase 3: Structured local search (all pairs)
    d3 = t0 + budget_s * 0.8
    best_h, best_c5 = _structured_local_search(best_h, d3, seed)

    # Phase 4: Random local search (fine-tuning)
    d4 = t0 + budget_s * 0.96
    best_h, best_c5 = _local_search(best_h, d4, seed)

    # Final verification
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
