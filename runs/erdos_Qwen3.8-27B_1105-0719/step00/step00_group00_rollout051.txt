def evaluate(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return dx * np.max(corr)


def project_to_feasible(y, n):
    target = n / 2.0
    lo = float(y.min()) - 1.0
    hi = float(y.max())
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = float(np.clip(y - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)
    residual = target - float(h.sum())
    if abs(residual) > 1e-12:
        eligible = np.where((h > 1e-12) & (h < 1.0 - 1e-12))[0]
        if len(eligible) > 0:
            per = residual / len(eligible)
            for i in eligible:
                if per > 0:
                    add = min(per, 1.0 - h[i])
                else:
                    add = max(per, h[i] - 1.0)
                h[i] += add
