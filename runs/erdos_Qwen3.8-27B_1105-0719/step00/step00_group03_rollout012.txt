def active_weights(scores, beta=10.0):
    max_s = scores.max()
    if max_s <= 1e-15:
        return np.ones(len(scores)) / len(scores)
    # Soft weights: exp(beta * (s / max_s - 1))
    w = np.exp(beta * (scores / max_s - 1.0))
    total = w.sum()
    if total < 1e-15:

        active = scores >= max_s - 1e-10
        if not active.any():
            active = np.argmax(scores)
        w = active.astype(float)
        total = w.sum()
    return w / total

def pair_transfer(h, k, i, amount):
    """Transfer 'amount' from h[i] to h[i+k] (if k>0) or h[i]-k to h[i] (if k<0)"""
    n = len(h)
    if k >= 0:
        src = i
        dst = i + k
    else:
        src = i - k
        dst = i
    if src < 0 or src >= n or dst < 0 or dst >= n:
        return None
    if src == dst:
        return None
    max_pos = min(h[src], 1.0 - h[dst])  # max we can transfer from src to dst
    max_neg = min(h[dst], 1.0 - h[src])  # max we can transfer from dst to src
    if amount >= 0:
        amount = min(amount, max_pos)
        if amount < 1e-12:
            return None
    else:
        amount = max(amount, -max_neg)
        if -amount < 1e-12:
            return None
    new_h = h.copy()
    new_h[src] -= amount
    new_h[dst] += amount
    return new_h

import numpy as np
import time

def exact_eval(h):
    n = len(h)
    dx = 2.0 / n
    scores = np.correlate(h, 1.0 - h, mode='full') * dx
    return scores.max(), scores

def project_feasible(y, target=None):
    n = len(y)
    if target is None:
        target = n / 2.0
    lo = -2.0
    hi = 2.0
    while np.sum(np.clip(y + lo, 0, 1)) > target:
        lo *= 2.0
    while np.sum(np.clip(y + hi, 0, 1)) < target:
        hi = lo + 2.0 * (hi - lo)
        if hi > 100.0:
            break
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(y + mid, 0, 1))
        if s > target:
            hi = mid
        else:
            lo = mid
    x = np.clip(y + (lo + hi) / 2.0, 0, 1)
    residual = target - np.sum(x)
    if abs(residual) > 1e-12:
        interior = np.where((x > 1e-10) & (x < 1.0 - 1e-10))[0]
        if len(interior) > 0:
            x[interior] += residual / len(interior)
            x = np.clip(x, 0, 1)
            residual2 = target - np.sum(x)
            if abs(residual2) > 1e-12:
                idx = np.argmin(np.abs(x - 0.5))
                x[idx] = np.clip(x[idx] + residual2, 0, 1)
    return x

def compute_gradient(h, scores, weights):
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    k_values = np.arange(len(scores)) - (n - 1)
    for k, w in zip(k_values, weights):
        if abs(w) < 1e-14:
            continue
        if k >= 0:
            if n - 1 - k >= 0:
                grad[:n - k] += -w * h[k:]
            if k <= n - 1:
                grad[k:] += w * (1.0 - h[:n - k])
        else:
            d = -k
            grad[d:] += -w * h[:n - d]
            grad[:n - d] += w * (1.0 - h[d:])
    return dx * grad

def softmax_weights(scores, beta=10.0):
    max_s = scores.max()
    if max_s <= 1e-14:
        return np.ones(len(scores)) / len(scores)
    scaled = beta * (scores / max_s - 1.0)
    scaled = np.maximum(scaled, -50.0)
    w = np.exp(scaled)
    total = w.sum()
    if total < 1e-30:
        active = scores >= max_s - 1e-10
        if not active.any():
            idx = np.argmax(scores)
            active[idx] = True
        w = active.astype(float)
        total = w.sum()
    return w / total

def resample(h, n_new):
    n_old = len(h)
    dx_old = 2.0 / n_old
    dx_new = 2.0 / n_new
    new_h = np.zeros(n_new)
    for j in range(n_new):
        a = j * dx_new
        b = (j + 1) * dx_new
        integral = 0.0
        i_start = int(a / dx_old)
        i_end = int(np.ceil(b / dx_old)) - 1
        i_start = max(0, i_start)
        i_end = min(n_old - 1, i_end)
        for i in range(i_start, i_end + 1):
            ia = i * dx_old
            ib = (i + 1) * dx_old
            ov = min(b, ib) - max(a, ia)
            if ov > 0:
                integral += h[i] * ov
        new_h[j] = integral / dx_new
    return project_feasible(new_h, target=n_new / 2.0)

def active_pair_move(h, scores, rng):
    n = len(h)
    n_s = len(scores)
    max_s = scores.max()
    if max_s <= 1e-14:
        return None
    threshold = max_s * (1.0 - 0.05)
    active = np.where(scores >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(scores)])
    r_idx = active[rng.integers(0, len(active))]
    k = r_idx - (n - 1)
    if k >= 0:
        if n - k <= 0:
            return None
        i = int(rng.integers(0, n - k))
        src = i
        dst = i + k
    else:
        d = -k
        if n - d <= 0:
            return None
        i = int(rng.integers(0, n - d))
        src = i + d
        dst = i
    amount = rng.uniform(0.02, 0.25)
    r_dir = rng.random()
    if r_dir < 0.6:
        amt = min(amount, h[src], 1.0 - h[dst])
        if amt < 1e-10:
            return None
        new_h = h.copy()
        new_h[src] -= amt
        new_h[dst] += amt
    else:
        amt = min(amount, h[dst], 1.0 - h[src])
        if amt < 1e-10:
            return None
        new_h = h.copy()
        new_h[dst] -= amt
        new_h[src] += amt
    return new_h

def random_pair(h, rng):
    n = len(h)
    i = int(rng.integers(0, n))
    j = int(rng.integers(0, n))
    if i == j:
        return None
    amount = rng.uniform(0.02, 0.3)
    if rng.random() < 0.5:
        amt = min(amount, h[i], 1.0 - h[j])
        if amt < 1e-10:
            return None
        nh = h.copy()
        nh[i] -= amt
        nh[j] += amt
    else:
        amt = min(amount, h[j], 1.0 - h[i])
        if amt < 1e-10:
            return None
        nh = h.copy()
        nh[j] -= amt
        nh[i] += amt
    return nh

def random_perturb(h, rng):
    n = len(h)
    n_pts = int(rng.integers(1, min(4, n)))
    idx = rng.integers(0, n, size=n_pts)
    amounts = rng.uniform(0.02, 0.15, size=n_pts) * np.where(rng.random(n_pts) < 0.5, 1, -1)
    nh = h.copy()
    for i, a in zip(idx, amounts):
        i = int(i)
        if a > 0:
            amt = min(abs(a), 1.0 - nh[i])
            if amt < 1e-10:
                continue
            nh[i] += amt
            others = np.where(nh > 0.01)[0]
            others = others[np.where(others != i)[0]] if i in others else others
            if len(others) == 0:
                nh[i] -= amt
                continue
            j = int(others[rng.integers(0, len(others))])
            sub = min(amt, nh[j])
            if sub < 1e-10:
                nh[i] -= amt
                continue
            nh[j] -= sub
            nh[i] -= (amt - sub)
            nh[i] = max(nh[i], 0.0)
        else:
            amt = min(abs(a), nh[i])
            if amt < 1e-10:
                continue
            nh[i] -= amt
            others = np.where(nh < 0.99)[0]
            others = others[np.where(others != i)[0]] if i in others else others
            if len(others) == 0:
                nh[i] += amt
                continue
            j = int(others[rng.integers(0, len(others))])
            add = min(amt, 1.0 - nh[j])
            if add < 1e-10:
                nh[i] += amt
                continue
            nh[j] += add
            nh[i] -= (amt - add)
            nh[i] = max(nh[i], 0.0)
    return project_feasible(nh, target=n / 2.0)

def gradient_descent(h, n, rng, max_iters, time_budget):
    dx = 2.0 / n
    best_c, _ = exact_eval(h)
    h_best = h.copy()
    for it in range(max_iters):
        if time_budget() < 20:
            break
        scores_raw = np.correlate(h, 1.0 - h, mode='full')
        scores = scores_raw * dx
        c_cur = scores.max()
        beta = 8.0 + 8.0 * min(1.0, it / 200.0)
        weights = softmax_weights(scores, beta)
        grad = compute_gradient(h, scores, weights)
        gmax = np.max(np.abs(grad))
        if gmax < 1e-12:
            break
        base_step = 0.8 / (1.0 + 0.01 * it)
        steps = [base_step, base_step * 0.5, base_step * 0.25, base_step * 0.1,
                 base_step * 0.05, base_step * 0.02]
        accepted = False
        for s in steps:
            direction = -grad / (gmax + 1e-15)
            new_h = h + s * direction
            new_h = project_feasible(new_h, target=n / 2.0)
            c_new, _ = exact_eval(new_h)
            if c_new < c_cur - 1e-13:
                h = new_h
                c_cur = c_new
                if c_new < best_c:
                    best_c = c_new
                    h_best = h.copy()
                accepted = True
                break
        if not accepted:
            # Try a random nudge in gradient direction with noise
            if it % 50 == 0:
                noise = rng.normal(0, 0.01, size=n)
                direction = -grad / (gmax + 1e-15) + noise
                new_h = h + 0.05 * direction
                new_h = project_feasible(new_h, target=n / 2.0)
                c_new, _ = exact_eval(new_h)
                if c_new < best_c:
                    h = new_h
                    best_c = c_new
                    h_best = h.copy()
                elif c_new < c_cur:
                    h = new_h
    return h_best, best_c

def anneal(h, n, rng, max_iters, time_budget):
    dx = 2.0 / n
    c_cur, scores = exact_eval(h)
    best_c = c_cur
    h_best = h.copy()
    T = max(0.008, 0.03 * c_cur)
    T_min = 0.0003
    cool_rate = 1.0 - 1.0 / max(1000, max_iters // 3)
    for it in range(max_iters):
        if time_budget() < 10:
            break
        T = max(T_min, T * cool_rate)
        r = rng.random()
        if r < 0.45:
            new_h = active_pair_move(h, scores, rng)
        elif r < 0.75:
            new_h = random_pair(h, rng)
        else:
            new_h = random_perturb(h, rng)
        if new_h is None:
            continue
        c_new, scores_new = exact_eval(new_h)
        delta = c_new - c_cur
        if delta < -1e-13:
            h = new_h
            c_cur = c_new
            scores = scores_new
            T = max(T_min, T * 0.95)
            if c_new < best_c:
                best_c = c_new
                h_best = h.copy()
        else:
            if delta / T < 50.0:
                if rng.random() < np.exp(-delta / T):
                    h = new_h
                    c_cur = c_new
                    scores = scores_new
    return h_best, best_c

        if time_left() < 10:
            break
        if nr == n_init:
            h_r = h_init.copy()
        else:
            h_r = resample(h_init, nr)
        c_r, _ = exact_eval(h_r)
        pool[nr] = (h_r, c_r)

    best_c = min(pool[nr][1] for nr in pool)
    best_n = min([nr for nr in pool if pool[nr][1] == best_c])[0]
    best_h = pool[best_n][0].copy()

    # Phase 1: Gradient descent on each resolution
    for nr in sorted(pool.keys()):
        if time_left() < 40:
            break
        h_r, c_r = pool[nr]
        max_iters = min(3000, max(500, int(time_left() * 8)))
        h_opt, c_opt = gradient_descent(h_r, nr, rng, max_iters, time_left)
        pool[nr] = (h_opt, c_opt)
        if c_opt < best_c - 1e-12:
            best_c = c_opt
            best_h = h_opt.copy()
            best_n = nr

    # Phase 2: Annealing on top resolutions
    top_res = sorted(pool.keys(), key=lambda x: pool[x][1])[:4]
    for nr in top_res:
        if time_left() < 30:
            break
        h_r, c_r = pool[nr]
        max_iters = min(30000, max(5000, int(time_left() * 30)))
        h_opt, c_opt = anneal(h_r, nr, rng, max_iters, time_left)
        pool[nr] = (h_opt, c_opt)
        if c_opt < best_c - 1e-12:
            best_c = c_opt
            best_h = h_opt.copy()
            best_n = nr

    # Phase 3: Try re-resampling best solution to nearby resolutions and optimize
    if time_left() > 60:
        nearby = sorted(set([max(20, best_n + d) for d in [-5, -2, 2, 5, 10, 20]
                            if 20 <= best_n + d <= 300]))
        for nr in nearby:
            if time_left() < 20:
                break
            if nr in pool:
                continue
            h_r = resample(best_h, nr)
            c_r, _ = exact_eval(h_r)
            pool[nr] = (h_r, c_r)
            max_iters = min(2000, int(time_left() * 5))
            h_opt, c_opt = gradient_descent(h_r, nr, rng, max_iters, time_left)
            if time_left() > 20:
                max_it2 = min(10000, int(time_left() * 15))
                h_opt2, c_opt2 = anneal(h_opt, nr, rng, max_it2, time_left)
                if c_opt2 < c_opt:
                    h_opt, c_opt = h_opt2, c_opt2
            pool[nr] = (h_opt, c_opt)
            if c_opt < best_c - 1e-12:
                best_c = c_opt
                best_h = h_opt.copy()
                best_n = nr

    # Phase 4: Final fine-tuning with more annealing on best
    if time_left() > 30:
        max_iters = int(time_left() * 40)
        h_final, c_final = anneal(best_h, best_n, rng, max_iters, time_left)
        if c_final < best_c:
            best_c = c_final
            best_h = h_final.copy()

    # Final validation
    c_check, _ = exact_eval(best_h)
    if c_check < best_c:
        best_c = c_check

    return best_h, best_c, best_n
