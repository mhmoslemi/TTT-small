import numpy as np
import time

def project_box_sum(x, s):
    """Euclidean projection onto {0<=x<=1, sum(x)=s}."""
    n = len(x)
    x = np.asarray(x, dtype=float).ravel().copy()
    lo = x.min() - 1.0
    hi = x.max() + 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        y = np.clip(x - mid, 0.0, 1.0)
        if y.sum() > s:
            lo = mid
        else:
            hi = mid
    y = np.clip(x - 0.5*(lo+hi), 0.0, 1.0)
    # Exact sum fixup
    r = s - y.sum()
    if abs(r) > 1e-13:
        if r > 0:
            room = 1.0 - y
            total = room.sum()
            if total > 1e-15:
                y += r * room / total
            else:
                y += r / n
        else:
            total = y.sum()
            if total > 1e-15:
                y += r * y / total
            else:
                y += r / n
    return np.clip(y, 0.0, 1.0)

def eval_c5(h):
    """Exact C5 evaluation."""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def lse_gradient(h, tau):
    """Gradient of LSE smooth objective."""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    c_max = c.max()
    w = np.exp((c - c_max) / tau)
    w /= w.sum()
    grads = np.zeros(n)
    for i in range(n):
        g = 0.0
        for k in range(2*n-1):
            lag = k - (n-1)
            t = 0.0
            j1 = i - lag
            if 0 <= j1 < n:
                t += 1.0 - h[j1]
            j2 = i + lag
            if 0 <= j2 < n:
                t -= h[j2]
            g += w[k] * t
        grads[i] = dx * g
    return grads

    rng = np.random.default_rng(seed)

def pair_polish(h, seed, max_evals=20000):
    """Pair transfer local search."""
    rng = np.random.default_rng(seed)
    n = len(h)
    s = n / 2.0

    rng = np.random.default_rng(seed)
    n = len(h)
    s = n / 2.0
    h = np.asarray(h, dtype=float).copy()
    best_h = h.copy()

    evals = 0

            temp *= 0.97

        if r < 0.75:

            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:

            d_max = min(h[i], 1.0 - h[j])

            if d < 1e-9:
                d = d_max * 0.5
            old_hi, old_hj = h[i], h[j]
            h[i] -= d
            h[j] += d
        else:
            # Projected random step
            old_h = h.copy()
            noise = rng.standard_normal(n) * step_scale
            noise -= np.mean(noise)
            h = project_box_sum(old_h + noise, s)
        
        evals += 1

        delta = c_new - cur_c
        if delta < 0 or (temp > 1e-8 and rng.random() < np.exp(-delta / temp)):

                best_h = h.copy()
        else:

            if r < 0.75:
                h[i] = old_hi
                h[j] = old_hj
            else:
                h = old_h

import numpy as np
import time

def project_box_sum(x, s):
    n = len(x)
    x = np.asarray(x, dtype=float).ravel().copy()
    lo = x.min() - 2.0
    hi = x.max() + 2.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        y = np.clip(x - mid, 0.0, 1.0)
        if y.sum() > s:
            lo = mid
        else:
            hi = mid
    y = np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - y.sum()
    if abs(r) > 1e-13:
        if r > 0:
            room = 1.0 - y
            t = room.sum()
            if t > 1e-15:
                y += r * room / t
            else:
                y += r / n
        else:
            t = y.sum()
            if t > 1e-15:
                y += r * y / t
            else:
                y += r / n
    return np.clip(y, 0.0, 1.0)

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def lse_gradient(h, tau):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    c_max = c.max()
    w = np.exp((c - c_max) / tau)
    w /= w.sum()
    lags = np.arange(-(n - 1), n)
    grads = np.zeros(n)
    for i in range(n):
        g = 0.0
        for k in range(2 * n - 1):
            lag = lags[k]
            t = 0.0
            j1 = i - lag
            if 0 <= j1 < n:
                t += 1.0 - h[j1]
            j2 = i + lag
            if 0 <= j2 < n:
                t -= h[j2]
            g += w[k] * t
        grads[i] = dx * g
    return grads

def smooth_descent(h, budget_s, seed):
    rng = np.random.default_rng(seed)
    n = len(h)
    s = n / 2.0
    h = np.asarray(h, dtype=float).copy()
    best_h = h.copy()
    best_c = eval_c5(h)
    cur_c = best_c
    start = time.perf_counter()
    tau = max(1e-4, 0.03 * cur_c)
    step = 1.0
    no_improve = 0
    max_no_improve = 300

    while time.perf_counter() - start < budget_s and no_improve < max_no_improve:
        if remaining_check(start, budget_s, 2):
            break
        grad = lse_gradient(h, tau)
        grad -= np.mean(grad)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-10:
            no_improve += 10
            tau = max(1e-5, tau * 0.9)
            continue
        accepted = False
        for _ in range(12):
            prop = h - step * grad
            prop = project_box_sum(prop, s)
            c_new = eval_c5(prop)
            if c_new < cur_c - 1e-13:
                h = prop
                cur_c = c_new
                accepted = True
                no_improve = 0
                step = min(step * 1.05, 5.0)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h.copy()
                break
            step *= 0.5
            if step < 1e-7:
                break
        if not accepted:
            no_improve += 1

def remaining_check(start, budget, reserve):
    return (time.perf_counter() - start) > (budget - reserve)

def pair_polish(h, seed, max_evals=30000):
    rng = np.random.default_rng(seed)
    n = len(h)
    s = n / 2.0
    h = np.asarray(h, dtype=float).copy()
    best_c = eval_c5(h)
    evals = 0
    improved = True
    pass_count = 0
    max_passes = 60

    while improved and evals < max_evals and pass_count < max_passes:
        improved = False
        pass_count += 1
        order = rng.permutation(n)
        for i in order:
            if evals >= max_evals:
                break
            for j in order:
                if i == j or evals >= max_evals:
                    continue
                d_max = min(h[i], 1.0 - h[j])
                if d_max < 1e-8:
                    continue
                fracs = [0.5, 0.25, 0.75] if pass_count < 5 else [0.5]

                    d = d_max * frac
                    if d < 1e-9:

                    h[i] -= d
                    h[j] += d
                    evals += 1
                    c = eval_c5(h)
                    if c < best_c - 1e-13:
                        best_c = c
                        improved = True
                    else:
                        h[i] += d
                        h[j] -= d
    return h, best_c

def sa_phase(h, seed, budget_s=60):

            noise = rng.standard_normal(n) * step_scale

        evals += 1
        c_new = eval_c5(h)
        delta = c_new - cur_c

            cur_c = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
        else:
            if r < 0.7:
                h[i] = old_hi
                h[j] = old_hj
            else:
                h = old_h

    return best_h, best_c

def interpolate_to_n(h, n_target):
    n = len(h)
    if n == n_target:
        return h.copy()
    x_old = (np.arange(n) + 0.5) * (2.0 / n)
    x_new = (np.arange(n_target) + 0.5) * (2.0 / n_target)
    h_ext = np.concatenate([h[-1:], h, h[:1]])
    x_ext = np.concatenate([[-1.0/n], x_old, [2.0 + 1.0/n_target]])
    h_new = np.interp(x_new, x_old, h)
    s = n_target / 2.0
    h_new = project_box_sum(h_new, s)
    return h_new
