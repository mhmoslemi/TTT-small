        h5, c5_5 = coordinate_descent(best_h, N, 30, t_stop)
        if c5_5 < best_c5 - 1e-12:
            best_h = h5.copy()
            best_c5 = c5_5

    # Phase 6: SLSQP polish
    if time.time() < t_stop:
        h6, c5_6 = slsqp_polish(best_h, N, beta=2048, t_stop=t_stop)
        if verify_feasible(h6, N) and c5_6 < best_c5 - 1e-12:
            best_h = h6.copy()
            best_c5 = c5_6

    # Phase 7: Fine random + LSE interleaved
    if time.time() < t_stop:
        amps_fine = [0.002, 0.001, 0.0005, 0.0002]
        h7, c5_7 = random_perturb_search(best_h, N, 5000, amps_fine, rng, t_stop)
        if c5_7 < best_c5 - 1e-12:
            best_h = h7.copy()
            best_c5 = c5_7

    # Phase 8: Final LSE + SLSQP
    if time.time() < t_stop:
        h8, c5_8 = lse_descent(best_h, 2048.0, max_iter=200, alpha0=0.01,
                                t_stop=t_stop)
        if verify_feasible(h8, N):
            c5_final, _ = objective_exact(h8)
            if c5_final < best_c5 - 1e-12:
                best_h = h8.copy()
                best_c5 = c5_final

    if time.time() < t_stop:
        h9, c5_9 = slsqp_polish(best_h, N, beta=4096, t_stop=t_stop)
        if verify_feasible(h9, N) and c5_9 < best_c5 - 1e-12:
            best_h = h9.copy()
            best_c5 = c5_9

    # Final validation
    best_h = project_feasible(best_h, N)
    best_h = np.clip(best_h, 0.0, 1.0)
    final_c5, _ = objective_exact(best_h)

    return (best_h, float(final_c5), N)
