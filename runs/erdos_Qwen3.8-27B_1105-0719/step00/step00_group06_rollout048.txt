def exact_objective(h):
    n = len(h)
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(profile)), profile

def project_box_sum(h, target_sum):
    h = np.array(h, dtype=np.float64).copy()
    n = len(h)
    h[~np.isfinite(h)] = 0.0
    h = np.clip(h, 0.0, 1.0)
    # Water-filling to achieve exact sum
    for _ in range(200):
        s = np.sum(h)
        diff = target_sum - s
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
            count = np.sum(mask)
            if count == 0:
                break
            step = diff / count
            step = min(step, 1.0 - h[mask].min()) if step > 0 else step
            h[mask] += step
        else:
            mask = h > 0.0
            count = np.sum(mask)
            if count == 0:
                break
            step = diff / count
            step = max(step, -h[mask].min()) if step < 0 else step
            h[mask] += step
    # Final correction
    s = np.sum(h)
    diff = target_sum - s
    if abs(diff) > 1e-15:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if np.sum(mask) > 0:
            idx = np.where(mask)[0]
            # Distribute among free variables
            remaining = abs(diff)
            for i in range(len(idx)):
                adjust = min(remaining, 1.0 - h[idx[i]] if diff > 0 else h[idx[i]])
                h[idx[i]] += adjust * (1 if diff > 0 else -1)
                remaining -= abs(adjust)
                if remaining < 1e-15:
                    break
    return h

def smooth_objective_and_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    pmax = np.max(profile)
    # Log-sum-exp
    shifted = (profile - pmax) / tau
    shifted = np.clip(shifted, -500, 0)  # prevent overflow
    weights = np.exp(shifted)
    W = np.sum(weights)
    value = pmax + tau * np.log(W)
    
    # Gradient
    grad = np.zeros(n)
    normalized_weights = weights / W
    # s = k - (n-1) for k = 0, 1, ..., 2n-2
    for k in range(2 * n - 1):
        s = k - (n - 1)
        w = normalized_weights[k]
        if w < 1e-15:
            continue
        # d/dh[j] profile[k] = dx * [(1-h[j+s])*I(0<=j+s<n) - h[j-s]*I(0<=j-s<n)]
        # First term: j+s in [0,n-1] => j in [-s, n-1-s] intersect [0,n-1]
        # Second term: j-s in [0,n-1] => j in [s, s+n-1] intersect [0,n-1]
        
        # First term: (1-h[j+s]) for j where 0<=j+s<n
        if s >= 0:
            # j+s in [0,n-1] means j in [0, n-1-s]
            j_range1 = slice(0, n - s)
            # j-s in [0,n-1] means j in [s, n-1]
            j_range2 = slice(s, n)
        else:
            # j+s in [0,n-1] means j in [-s, n-1]
            j_range1 = slice(-s, n)
            # j-s in [0,n-1] means j in [0, n-1+s] = [0, n-1+(-s)-1]... 
            # j-s in [0,n-1] means 0<=j-s<n => s<=j<s+n => since s<0, j in [0, n-1+s]
            j_range2 = slice(0, n + s)  # n+s = n-|s|
        
        grad[j_range1] += w * dx * (1.0 - h[j_range1 + s])
        grad[j_range2] -= w * dx * h[j_range2 - s]
    
    return value, grad

def solver_objective_and_grad(x, tau):
    n = len(x)
    target = n / 2.0
    # Compute smooth objective
    val, grad = smooth_objective_and_grad(x, tau)
    
    # Penalty for constraint violations
    sum_v = np.sum(x) - target
    box_v = np.sum(np.maximum(0, x - 1)**2) + np.sum(np.maximum(0, -x)**2)
    pen_val = 1e4 * (sum_v**2 + box_v)

    pen_grad += 2e4 * np.where(x > 1, x - 1, 0)
    pen_grad += 2e4 * np.where(x < 0, x, 0)
    
    return val + pen_val, grad + pen_grad

def project_box_sum(h, target_sum):
    """Project h onto [0,1]^n with sum = target_sum using water-filling."""
    h = np.array(h, dtype=np.float64).copy()
    n = len(h)
    h[~np.isfinite(h)] = 0.0
    np.clip(h, 0.0, 1.0, out=h)

    for _ in range(500):
        s = np.sum(h)
        diff = target_sum - s
        tol = 1e-14

            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            cnt = np.sum(mask)
            if cnt == 0:

                cnt = np.sum(mask)

    """Log-sum-exp smoothed objective and its gradient."""
    n = len(h)
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx

def solve_slsqp(h0, tau, maxiter=200, ftol=1e-12):

    n = len(h0)
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n

        val, _ = lse_objective_and_grad(x, tau)
        # Add penalty for constraint violations (as safety)
        sv = np.sum(x) - target
        bv = np.sum(np.maximum(0.0, x - 1.0)**2) + np.sum(np.maximum(0.0, -x)**2)
        return val + 1000.0 * (sv*sv + bv)

        sv = np.sum(x) - target

        g += 2000.0 * np.maximum(0.0, x - 1.0)
        g += 2000.0 * np.minimum(0.0, x)
        return g
