import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def project_feasible(y, mass):
    """Project y onto {h: 0<=h_i<=1, sum(h)=mass}"""
    n = len(y)
    # Binary search for the right shift
    lo = y.min() - mass
    hi = y.max()
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > mass:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix sum exactly
    diff = mass - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(-h)  # prefer adding to small values
        else:
            order = np.argsort(h)  # prefer subtracting from large values
        for idx in order:
            if diff > 0:
                room = 1.0 - h[idx]
                add = min(room, diff)
                h[idx] += add
                diff -= add
            else:
                room = h[idx]
                sub = min(room, -diff)
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

        n_pairs = n * (n - 1) // 2
        for _ in range(min(n_pairs, 2000)):

            hi = min(h[i], 1.0 - h[j])  # h[i]-d >= 0, h[j]+d <= 1
