            bounds_alt = [(0.0, 1.0)] * n_alt

            def eq_fun_alt(h):
                return np.sum(h) - n_alt / 2.0

            def slsqp_run_alt(h_start, alpha, maxiter, ftol):
                def obj_alt(h):
                    return smooth_objective(h, dx_alt, alpha)
                try:
                    result = minimize(obj_alt, h_start, method='SLSQP',
                                     bounds=bounds_alt,
                                     constraints=[{'type': 'eq', 'fun': eq_fun_alt,
                                                    'jac': lambda h: np.ones(n_alt)}],
                                     options={'maxiter': maxiter, 'ftol': ftol})
                    h_new = result.x
                    c5_new = compute_c5(h_new, dx_alt)
                    return h_new, c5_new
                except Exception:
                    return h_start, compute_c5(h_start, dx_alt)

            for alpha in [50, 200, 500]:
                if time_left() < 30:
                    break
                h_new, c5_new = slsqp_run_alt(h_alt, alpha, 1000, 1e-13)
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
                    n = n_alt
                    dx = dx_alt
                    bounds = [(0.0, 1.0)] * n
                else:
                    h_alt = h_new  # Continue from this point

    # Phase 6: Final polish with trust-constr
    if time_left() > 50:
        try:
            A = np.ones((1, n))
            lc = LinearConstraint(A, n / 2.0, n / 2.0)
            def obj_tc(h):
                return smooth_objective(h, dx, 5000)
            result = minimize(obj_tc, best_h, method='trust-constr',
                             bounds=bounds, constraints=[lc],
                             options={'maxiter': 300, 'gtol': 1e-14})
            h_new = result.x
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
        except Exception:
            pass

    # Phase 7: One final SLSQP with very high alpha
    if time_left() > 30:
        h_new, c5_new = slsqp_run(best_h, 10000, 3000, 1e-16)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new

    return best_h, best_c5, len(best_h)
