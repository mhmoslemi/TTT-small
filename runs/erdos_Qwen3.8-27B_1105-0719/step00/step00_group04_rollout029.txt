def solve_1d_convex(R, slopes, d_lo, d_hi):
    for _ in range(80):
        mid = (d_lo + d_hi) / 2
        vals = R + mid * slopes
        mx = vals.max()
        active = vals >= mx - 1e-14
        min_s = slopes[active].min()
        max_s = slopes[active].max()
        if min_s <= 0 and max_s >= 0:
            return mid
        elif min_s > 0:
            d_hi = mid
        else:
            d_lo = mid
    return (d_lo + d_hi) / 2

    lags = np.arange(-(n-1), n)

    ip = p - lags
    valid_ip = (ip >= 0) & (ip < n)

        slopes[valid_ip] += 1 - h[ip[valid_ip]]
    
    # -(1-h[q-lag]) term
    iq = q - lags
    valid_iq = (iq >= 0) & (iq < n)
    if valid_iq.any():
        slopes[valid_iq] -= 1 - h[iq[valid_iq]]
    
    # -h[p+lag] term
    jp = p + lags
    valid_jp = (jp >= 0) & (jp < n)
    if valid_jp.any():
        slopes[valid_jp] -= h[jp[valid_jp]]
    
    # +h[q+lag] term
    jq = q + lags
    valid_jq = (jq >= 0) & (jq < n)
    if valid_jq.any():
        slopes[valid_jq] += h[jq[valid_jq]]
    
    return slopes * dx
