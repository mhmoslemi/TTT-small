import numpy as np
from scipy.optimize import minimize
import time

def exact_objective(h):
    n = len(h)
    if n == 0:
        return float('inf'), np.array([])
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    c5 = float(np.max(profile))
    return c5, profile

def project_box_sum(h, target=None):
    n = len(h)
    if target is None:
        target = 0.5 * n
    h = np.asarray(h, dtype=np.float64).copy()
    h[~np.isfinite(h)] = 0.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = h.sum() - target
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h > 0
            if not mask.any():
                break
            step = diff / mask.sum()
            h[mask] = np.clip(h[mask] - step, 0.0, 1.0)
        else:
            mask = h < 1
            if not mask.any():
                break
            step = -diff / mask.sum()
            h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    # Final precise fix
    diff = h.sum() - target
    if abs(diff) > 1e-14:
        if diff > 0:
            mask = h > 0
            idx = np.where(mask)[0]
            for i in idx:
                reduce = min(h[i], diff)
                h[i] -= reduce
                diff -= reduce
                if abs(diff) < 1e-15:
                    break
        else:
            mask = h < 1
            idx = np.where(mask)[0]
            for i in idx:
                add = min(1.0 - h[i], -diff)
                h[i] += add
                diff += add
                if abs(diff) < 1e-15:
                    break
    return h

def smooth_obj_and_grad(x, tau):
    n = len(x)
    dx = 2.0 / n
    # Compute profile
    profile = np.correlate(x, 1.0 - x, mode='full') * dx
    pmax = np.max(profile)
    # Log-sum-exp
    shifted = (profile - pmax) / tau
    # Clip for numerical stability
    shifted = np.clip(shifted, -500, 0)
    weights = np.exp(shifted)
    Z = np.sum(weights)
    value = pmax + tau * np.log(Z)
    # Gradient: weighted sum of individual lag gradients
    grad = np.zeros(n)
    w = weights / Z  # normalized weights
    # profile index k corresponds to lag m = k - (n-1)
    for k in range(len(profile)):
        if w[k] < 1e-15:
            continue
        m = k - (n - 1)
        if m == 0:
            grad += w[k] * dx * (1.0 - 2.0 * x)
        elif m > 0:
            # Phi_m = dx * sum_{j=0}^{n-1-m} h[j+m]*(1-h[j])
            # d/dh_j = dx * [(1-h[j-m]) * 1(m<=j) - h[j+m] * 1(j<=n-1-m)]
            # For j in [m, n-1]: contribution (1-h[j-m])
            # For j in [0, n-1-m]: contribution -h[j+m]
            j_start1 = m
            j_end1 = n  # exclusive
            j_start2 = 0
            j_end2 = n - m  # exclusive
            grad[j_start1:j_end1] += w[k] * dx * (1.0 - x[j_start1-m:j_end1-m])
            grad[j_start2:j_end2] -= w[k] * dx * x[j_start2+m:j_end2+m]
        else:
            a = -m  # a > 0
            # Phi_{-a} = dx * sum_{j=0}^{n-1-a} h[j]*(1-h[j+a])
            # d/dh_k = dx * [(1-h[k+a]) * 1(0<=k<=n-1-a) - h[k-a] * 1(a<=k<=n-1)]
            j_start1 = 0
            j_end1 = n - a  # exclusive
            j_start2 = a
            j_end2 = n
            grad[j_start1:j_end1] += w[k] * dx * (1.0 - x[j_start1+a:j_end1+a])
            grad[j_start2:j_end2] -= w[k] * dx * x[j_start2-a:j_end2-a]
    return value, grad

def slsqp_refine(h0, tau, maxiter=200):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    eq_constraint = {'type': 'eq',
                    'fun': lambda x: np.sum(x) - 0.5 * n,
                    'jac': lambda x: np.ones(n)}
    
    def obj_grad(x):
        val, grad = smooth_obj_and_grad(x, tau)
        # Add penalty for constraint violation (safety)
        sum_err = np.sum(x) - 0.5 * n
        box_viol = np.sum(np.maximum(0, x - 1)**2) + np.sum(np.maximum(0, -x)**2)
        val += 1e6 * (sum_err**2 + box_viol)
        grad += 2e6 * sum_err * np.ones(n)
        grad += 2e6 * (np.maximum(0, x-1) - np.maximum(0, -x))
        return val, grad
    
    res = minimize(obj_grad, h0.copy(), method='SLSQP',
                   jac=True, bounds=bounds, constraints=[eq_constraint],
                   options={'maxiter': maxiter, 'ftol': 1e-12})
    return res.x

def slsqp_multistage(h0, tau_list=None, maxiter_per_tau=200):
    if tau_list is None:
        tau_list = [0.005, 0.002, 0.0008, 0.0003, 0.0001]
    h = h0.copy()
    for tau in tau_list:
        h = slsqp_refine(h, tau, maxiter=maxiter_per_tau)
        h = project_box_sum(h)
    return h

def binary_threshold(h):
    n = len(h)
    if n % 2 != 0:
        return None
    m = n // 2
    idx = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[idx[:m]] = 1.0
    return b

def binary_swap_descent(b, max_rounds=50, max_swaps_per_round=200):
    n = len(b)
    if n % 2 != 0:
        return b
    current_c5, _ = exact_objective(b)
    ones = np.where(b == 1.0)[0].tolist()
    zeros = np.where(b == 0.0)[0].tolist()
    rng = np.random.RandomState(42)
    
    for _ in range(max_rounds):
        improved = False
        attempts = 0
        while attempts < max_swaps_per_round:
            attempts += 1
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            # Swap
            b[i] = 0.0
            b[j] = 1.0
            c5_new, _ = exact_objective(b)
            if c5_new < current_c5 - 1e-12:
                current_c5 = c5_new
                ones.remove(i)
                ones.append(j)
                zeros.remove(j)
                zeros.append(i)
                improved = True
            else:
                b[i] = 1.0
                b[j] = 0.0
        if not improved:
            break
    return b

def binary_swap_descent2(b, max_rounds=100):
    """More systematic swap descent: try all pairs in one pass."""
    n = len(b)
    if n % 2 != 0 or len(b) > 100:
        return b  # too expensive for large n
    current_c5, _ = exact_objective(b)
    ones = np.where(b == 1.0)[0].tolist()
    zeros = np.where(b == 0.0)[0].tolist()
    
    for _ in range(max_rounds):
        improved = False
        for i in ones[:]:
            for j in zeros:
                b[i] = 0.0
                b[j] = 1.0
                c5_new, _ = exact_objective(b)
                if c5_new < current_c5 - 1e-12:
                    current_c5 = c5_new
                    ones.remove(i)
                    ones.append(j)
                    zeros.remove(j)
                    zeros.append(i)
                    improved = True
                    break
                else:
                    b[i] = 1.0
            if improved:
                break
        if not improved:
            break
    return b

def make_starts_from_parent(parent, n, rng, count=6):
    """Generate initial candidates of length n from parent."""
    starts = []
    if n == len(parent):
        starts.append(parent.copy())
    else:
        # Linear interpolation
        x_parent = (np.arange(len(parent)) + 0.5) / len(parent)
        x_new = (np.arange(n) + 0.5) / n
        new_h = np.interp(x_new, x_parent, parent)
        new_h = project_box_sum(new_h)
        starts.append(new_h)
    
    # Add perturbations
    base = starts[0]
    for _ in range(count - 1):
        scale = rng.uniform(0.01, 0.05)
        perturbed = base + rng.normal(0, scale, n)
        perturbed = project_box_sum(perturbed)
        starts.append(perturbed)
    
    return starts

import numpy as np
from scipy.optimize import minimize
import time


def exact_objective(h):
    n = len(h)
    if n == 0:
        return float('inf'), np.array([])
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    c5 = float(np.max(profile))
    return c5, profile


def project_box_sum(h, target=None):
    n = len(h)
    if target is None:
        target = 0.5 * n
    h = np.asarray(h, dtype=np.float64).copy()
    h[~np.isfinite(h)] = 0.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = h.sum() - target
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h > 1e-15
            if not np.any(mask):
                break
            step = diff / np.sum(mask)
            h[mask] = np.maximum(h[mask] - step, 0.0)
        else:
            mask = h < 1.0 - 1e-15
            if not np.any(mask):
                break
            step = -diff / np.sum(mask)
            h[mask] = np.minimum(h[mask] + step, 1.0)
    diff = h.sum() - target
    if abs(diff) > 1e-13:
        if diff > 0:
            for i in range(n):
                if h[i] > 0:
                    red = min(h[i], diff)
                    h[i] -= red
                    diff -= red
                    if abs(diff) < 1e-15:
                        break
        else:
            for i in range(n):
                if h[i] < 1.0:
                    add = min(1.0 - h[i], -diff)
                    h[i] += add
                    diff += add
                    if abs(diff) < 1e-15:
                        break
    return h


def smooth_obj_and_grad(x, tau):
    n = len(x)
    dx = 2.0 / n
    profile = np.correlate(x, 1.0 - x, mode='full') * dx
    pmax = np.max(profile)
    shifted = (profile - pmax) / tau
    shifted = np.clip(shifted, -400.0, 0.0)
    weights = np.exp(shifted)
    Z = np.sum(weights)
    if Z < 1e-300:
        Z = 1e-300
    value = pmax + tau * np.log(Z)
    w = weights / Z
    grad = np.zeros(n)
    for k in range(len(profile)):
        wk = w[k]
        if wk < 1e-14:
            continue
        m = k - (n - 1)
        if m == 0:
            grad += wk * dx * (1.0 - 2.0 * x)
        elif m > 0:
            j1 = m
            j2 = n
            grad[j1:j2] += wk * dx * (1.0 - x[j1 - m:j2 - m])
            j3 = 0
            j4 = n - m
            grad[j3:j4] -= wk * dx * x[j3 + m:j4 + m]
        else:
            a = -m
            j1 = 0
            j2 = n - a
            grad[j1:j2] += wk * dx * (1.0 - x[j1 + a:j2 + a])
            j3 = a
            j4 = n
            grad[j3:j4] -= wk * dx * x[j3 - a:j4 - a]
    return value, grad


def slsqp_single(h0, tau, maxiter=200):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n

    def obj_and_grad(x):
        val, grad = smooth_obj_and_grad(x, tau)
        sum_err = np.sum(x) - 0.5 * n
        if abs(sum_err) > 1e-6:
            val += 1e8 * sum_err * sum_err
            grad += 2e8 * sum_err * np.ones(n)
        box_viol = np.sum(np.maximum(0.0, x - 1.0)) + np.sum(np.maximum(0.0, -x))
        if box_viol > 1e-10:
            val += 1e8 * box_viol
            grad += 1e8 * (np.maximum(0.0, x - 1.0) - np.maximum(0.0, -x))
        return val, grad

    eq = {'type': 'eq', 'fun': lambda x: np.sum(x) - 0.5 * n, 'jac': lambda x: np.ones(n)}

    try:
        res = minimize(obj_and_grad, h0.copy(), method='SLSQP', jac=True,
                       bounds=bounds, constraints=[eq],
                       options={'maxiter': maxiter, 'ftol': 1e-10})
        return res.x
    except Exception:
        return h0.copy()

def binary_threshold(h):
    n = len(h)
    if n % 2 != 0:
        return None
    m = n // 2
    idx = np.argsort(h)[::-1]
    b = np.zeros(n, dtype=np.float64)
    b[idx[:m]] = 1.0
    return b


def binary_swap_descent(b, max_rounds=80, rng=None):
    n = len(b)
    if n % 2 != 0:
        return b
    if rng is None:
        rng = np.random.RandomState(0)
    current_c5 = exact_objective(b)[0]
    ones = np.where(b == 1.0)[0].tolist()
    zeros = np.where(b == 0.0)[0].tolist()
    if len(ones) == 0 or len(zeros) == 0:
        return b

    for _ in range(max_rounds):
        improved = False
        attempts = min(500, len(ones) * len(zeros))
        for _ in range(attempts):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            if i == j:
                continue
            b[i] = 0.0
            b[j] = 1.0
            c5_new = exact_objective(b)[0]
            if c5_new < current_c5 - 1e-13:
                current_c5 = c5_new
                ones.remove(i)
                ones.append(j)
                zeros.remove(j)
                zeros.append(i)
                improved = True
            else:
                b[i] = 1.0
                b[j] = 0.0
        if not improved:
            break
    return b

def make_structured_binary(n, variant=0):
    if n % 2 != 0:
        return None
    m = n // 2
    b = np.zeros(n, dtype=np.float64)
    if variant == 0:
        idx = list(range(0, n, 2))[:m]
    elif variant == 1:
        step = 3
        while np.gcd(step, n) != 1:
            step += 1
        idx = [(j * step) % n for j in range(m)]
    elif variant == 2:
        phi = (1.0 + np.sqrt(5.0)) / 2.0
        idx = sorted(set(int((j * n * phi) % n) for j in range(2 * m)))[:m]
        if len(idx) < m:
            idx = list(range(0, 2 * m))
    elif variant == 3:
        idx = list(range(0, m))
    else:
        idx = list(range(0, n, max(1, n // m)))[:m]
        if len(idx) < m:
            idx = list(range(0, 2 * m, 2))[:m]
    b[sorted(set(idx))[:m]] = 1.0
    if np.sum(b) != m:
        return None
    return b

                if remaining > 60:
                    b3 = binary_swap_systematic(b2, max_rounds=20)
                    c5_b3 = exact_objective(b3)[0]
                    if c5_b3 < best_c5 - 1e-12:
                        best_c5 = c5_b3
                        best_h = b3.copy()
                        best_n = n

            if remaining > 50 and s_i == 0:
                for _ in range(3):
                    h_r = best_h + rng.normal(0, 0.015, n)
                    h_r = project_box_sum(h_r)
                    h_r2 = slsqp_multistage(h_r, tau_list=[0.004, 0.001, 0.0004],
                                            maxiter_per_tau=100)
                    h_r2 = project_box_sum(h_r2)
                    c5_r = exact_objective(h_r2)[0]
                    if c5_r < best_c5 - 1e-12:
                        best_c5 = c5_r
                        best_h = h_r2.copy()
                        best_n = n
                    if n % 2 == 0:
                        br = binary_threshold(h_r2)
                        if br is not None:
                            br2 = binary_swap_descent(br, max_rounds=40, rng=rng)
                            c5_br = exact_objective(br2)[0]
                            if c5_br < best_c5 - 1e-12:
                                best_c5 = c5_br
                                best_h = br2.copy()
                                best_n = n

    best_h = project_box_sum(best_h)
    best_c5 = exact_objective(best_h)[0]
    best_n = len(best_h)

    sum_check = np.sum(best_h) - 0.5 * best_n
    if abs(sum_check) > 1e-9:
        best_h = np.ones(best_n) * 0.5
        best_c5 = exact_objective(best_h)[0]

    return best_h, best_c5, best_n
