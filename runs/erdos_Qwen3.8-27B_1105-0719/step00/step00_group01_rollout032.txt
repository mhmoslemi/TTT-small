    max_val = np.max(corr)
    active = np.where(corr >= max_val - tol)[0]

    grad = np.zeros(n)

            # First term: (1-h[j-s]) for j in [s, n-1]
            grad[s:] += w * (1.0 - h[:n-s])
            # Second term: -h[j+s] for j in [0, n-1-s]
            grad[:n-s] -= w * h[s:]

            # First term: (1-h[j-s]) = (1-h[j+t]) for j in [0, n-1-t]
            grad[:n-t] += w * (1.0 - h[t:])

    n = len(h)

        proj = np.clip(h - mid, 0.0, 1.0)
        if proj.sum() > target_sum:
            lo = mid
        else:

    scaled = tau * corr
    m = np.max(scaled)

import time
from scipy.optimize import minimize

# Global state for passing parameters to scipy
_G_STATE = {}

def project_feasible(h, target_sum):
    """Project h onto {0<=h_i<=1, sum(h)=target_sum} via bisection on shift."""
    lo, hi = -5.0, 5.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0.0, 1.0).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def corr_compute(h):
    """Full cross-correlation of h with 1-h."""
    return np.correlate(h, 1.0 - h, mode="full")

def c5_compute(h):
    """Compute C5 = max_k integral h(x)(1-h(x+k)) dx."""
    n = len(h)
    return float(np.max(corr_compute(h)) * (2.0 / n))

def smooth_max_weights(corr, tau):
    """Returns (smooth_max_value, softmax_weights) for the correlation vector."""
    scaled = tau * corr
    m = np.max(scaled)
    e = np.exp(scaled - m)
    se = e.sum()
    val = (m + np.log(se)) / tau
    w = e / se
    return val, w

def grad_h_from_weights(h, weights):
    """Gradient of weighted combination of correlations w.r.t. h."""
    n = len(h)
    grad = np.zeros(n)
    nw = len(weights)
    for i in range(nw):
        wi = weights[i]
        if wi < 1e-14:
            continue
        s = i - (n - 1)
        if s >= 0:
            grad[s:] += wi * (1.0 - h[:n - s])
            grad[:n - s] -= wi * h[s:]
        else:
            t = -s
            grad[:n - t] += wi * (1.0 - h[t:])
            grad[t:] -= wi * h[:n - t]
    return grad

def _smooth_obj(h, tau, target):
    corr = corr_compute(h)
    val, _ = smooth_max_weights(corr, tau)
    return val

def _smooth_jac(h, tau, target):
    corr = corr_compute(h)
    _, w = smooth_max_weights(corr, tau)
    return grad_h_from_weights(h, w)

def _eq_con_fun(h, tau, target):
    return h.sum() - target

def _eq_con_jac(h, tau, target):

def slsqp_smooth(h0, tau, target, max_iter=500):
    """Run SLSQP on smooth max objective."""
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': _eq_con_fun, 'jac': _eq_con_jac,
                    'args': (tau, target)}]
    try:
        res = minimize(_smooth_obj, h0, args=(tau, target),
                       jac=_smooth_jac, method='SLSQP', bounds=bounds,
                       constraints=constraints, options={'maxiter': max_iter,
                                                         'ftol': 1e-12})
        if res.success and res.x is not None:
            x = project_feasible(np.clip(res.x, 0.0, 1.0), target)
            return x
    except Exception:
        pass
    return h0

def projected_gradient(h, target, tau, max_iter=500, lr_init=0.1):
    """Custom projected gradient with backtracking on smooth max."""
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_val = smooth_max_weights(corr_compute(h), tau)[0]
    lr = lr_init
    for it in range(max_iter):

        obj, w = smooth_max_weights(corr, tau)
        if obj < best_val:

            best_h = h.copy()

def pair_local_search(h, target, n, rng, max_iters=10000):
    """Local search: random pair modifications preserving sum."""
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c5 = float(np.max(corr_compute(h)) * dx)
