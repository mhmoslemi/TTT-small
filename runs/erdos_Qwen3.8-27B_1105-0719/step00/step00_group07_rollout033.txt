import numpy as np

import time

def compute_correlations(h):
    """Compute all correlation values using np.correlate"""
    return np.correlate(h, 1.0 - h, mode="full")

def compute_c5(h, dx):
    """Compute C5 value"""
    corr = compute_correlations(h)
    return np.max(corr) * dx

def project_onto_feasible(v, n):
    """Project v onto {h: 0<=h<=1, sum(h)=n/2} using bisection on shift"""
    target = n / 2.0
    # Find t such that sum(clip(v+t, 0, 1)) = target
    lo, hi = -v.max() - 1, -v.min() + 1
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.clip(v + mid, 0, 1).sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(v + (lo + hi) / 2, 0, 1)
    # Fix rounding
    diff = target - h.sum()
    if abs(diff) > 1e-15:
        # Adjust the element with most slack
        if diff > 0:
            idx = np.argmax(1 - h)
        else:
            idx = np.argmin(h)
        h[idx] += diff
        h[idx] = np.clip(h[idx], 0, 1)
    return h

def lse_objective_and_grad(h, beta):
    """Compute logsumexp of correlations and its gradient"""
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_corr = np.max(corr)
    shifted = corr - max_corr
    exp_shifted = np.exp(beta * shifted)
    denom = np.sum(exp_shifted)
    weights = exp_shifted / denom
    obj = max_corr + np.log(denom) / beta
    
    # Compute gradient
    n = len(h)
    grad = np.zeros(n)
    for L in range(-(n-1), n):
        w = weights[L + n - 1]
        if w < 1e-15:
            continue
        # dR_L/dh for this lag
        if L == 0:
            grad += w * (1 - 2*h)
        elif L > 0:
            # R_L = sum_{i=0}^{n-1-L} h[i]*(1-h[i+L])
            # dR_L/dh[j] = (1-h[j+L]) for j <= n-1-L, plus (-h[j-L]) for j >= L
            valid1 = np.arange(n) <= (n - 1 - L)

            grad[valid1] += w * (1 - h[np.arange(n)[valid1] + L])

    corr = np.correlate(h, 1.0 - h, mode="full")

            grad[Lneg:] += w * (1 - h[:m])

- Pick i (the one to decrease) and j (the one to increase)
- Try delta values
- Check if C5 improved

The key question is: which (i,j) pairs to try? A good heuristic:
- For each i, find the j that minimizes the increase in C5 when we move mass from i to j
- This is expensive (O(n) evaluations per i, O(n²) total per sweep)

For n=95, one sweep with all pairs is 95*94 = 8930 evaluations, each O(n²) ≈ 9025. Total: ~80M operations per sweep. With np.correlate being very fast in C, this should take maybe 1-2 seconds per sweep. We can afford a few hundred sweeps.

But actually, a smarter approach: for each coordinate i, compute the "sensitivity" - how much C5 changes when we slightly increase or decrease h[i]. Then pick the most sensitive coordinates first.

Sensitivity of C5 to h[i]: 
C5 = max_k R_k * dx
∂C5/∂h[i] = max_{k: R_k = max} ∂R_k/∂h[i] * dx

This is just the subgradient at h[i]. We can compute this from the logsumexp gradient with large beta.

OK let me just write the final code. I'll keep it simple and effective:

1. Start from initial_h_values
2. Do logsumexp projected gradient descent (phases with increasing beta)
3. Do coordinate descent (transfer between pairs)
4. Try a few transforms (complement-reverse)
5. Return best

Let me also add: after the gradient descent, try "annealing" - add random noise to h, project, and see if we get a better solution. This helps escape local optima.

    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def project_feasible(v, n):
    target = n / 2.0
    vmin, vmax = float(v.min()), float(v.max())
    lo = -vmax - 0.5
    hi = -vmin + 0.5
    for _ in range(64):
        mid = (lo + hi) * 0.5
        s = np.clip(v + mid, 0.0, 1.0).sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-11:
        if diff > 0:
            idx = int(np.argmin(h))
        else:
            idx = int(np.argmax(h))
        h[idx] = float(np.clip(h[idx] + diff, 0.0, 1.0))
    return h

def lse_grad(h, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = float(np.max(corr))
    ev = np.exp(beta * (corr - mc))
    tot = float(ev.sum())
    w = ev / tot
    grad = np.zeros(n)
    for li in range(2 * n - 1):
        L = li - (n - 1)
        wi = w[li]
        if wi < 1e-13:
            continue
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif L > 0:

            grad[:m] += wi * (1.0 - h[L:])
            grad[L:] += wi * (-h[:m])
        else:
            Ln = -L
            m = n - Ln
            grad[Ln:] += wi * (1.0 - h[:m])
            grad[:m] += wi * (-h[Ln:])
    return grad

                    best_c5 = c5_try
                    best_h = h_try.copy()
                    improved = True
                    break
            if not improved:
