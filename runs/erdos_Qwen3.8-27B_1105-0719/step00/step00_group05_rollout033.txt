    lag = active_idx - (n - 1)

            grad[s:] -= h[:-s]
        else:
            grad[:] = 1.0 - 2.0 * h
        grad *= dx
        h_new = project_box_sum(h - lr * grad, target_sum)
        c5_new = compute_c5(h_new, dx)
        if c5_new < c5:
            h = h_new
            lr = min(lr * 1.02, 0.3)
        else:
            lr = max(lr * 0.85, 0.002)
            if no_improve > 200:
                h = project_box_sum(h + rng.normal(0, 0.04, n), target_sum)
                no_improve = 0
                lr = 0.05
    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    best_c5 = float('inf')
    best_h = None
    best_n = 0

    init_h = None
    try:
        init_h = np.array(initial_h_values, dtype=float)
    except NameError:
        pass

    n_configs = [
        (42, 3),
        (40, 2),
        (44, 2),
        (48, 2),
        (36, 2),
        (50, 2),
        (41, 1),
        (43, 1),
    ]

    for n, n_restarts in n_configs:
        elapsed = time.time() - start_time
        remaining = budget_s - elapsed
        if remaining < 20:
            break

        dx = 2.0 / n
        target_sum = n / 2.0
        time_per = max(15.0, min((remaining - 10) / n_restarts, 90.0))

        for restart in range(n_restarts):
            elapsed = time.time() - start_time
            remaining = budget_s - elapsed
            if remaining < 15:
                break

            if restart == 0 and n == 42 and init_h is not None:
                h0 = init_h.copy()
            elif rng.random() < 0.25:
                h0 = np.zeros(n)
                k_ones = int(round(target_sum))
                idx = rng.choice(n, size=k_ones, replace=False)
                h0[idx] = 1.0
            elif rng.random() < 0.5:
                h0 = np.full(n, 0.5) + rng.normal(0, 0.25, n)
            else:
                h0 = np.full(n, 0.5) + rng.normal(0, 0.1, n)
                h0 = 0.5 + 0.5 * np.sin(2.0 * np.pi * rng.uniform(0.5, 3.0) * np.arange(n) / n)
            h0 = project_box_sum(h0, target_sum)

            # Phase 1: scipy optimization
            t1 = time_per * 0.55
            h_opt, c5_opt = optimize_scipy(h0, dx, target_sum, t1)

            # Phase 2: subgradient polish
            elapsed = time.time() - start_time
            remaining = budget_s - elapsed
            t2 = min(time_per * 0.2, max(5.0, remaining * 0.15))
            if t2 > 3 and c5_opt < 0.42:
                h_opt, c5_opt = projected_subgradient(h_opt, dx, target_sum, rng, t2)

            # Phase 3: pairwise refinement
            elapsed = time.time() - start_time
            remaining = budget_s - elapsed
            t3 = min(time_per * 0.25, max(5.0, remaining * 0.15))
            if t3 > 3 and c5_opt < 0.42:
                h_opt, c5_opt = pairwise_refine(h_opt, dx, target_sum, rng, t3)

            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = n

    # Final short refinement if time remains
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed
    if best_h is not None and remaining > 15:
        dx = 2.0 / best_n
        target_sum = best_n / 2.0
        best_h, best_c5 = pairwise_refine(best_h, dx, target_sum, rng, min(remaining - 5, 60))

    if best_h is None:
        n = 42
        best_h = np.full(n, 0.5)
        best_c5 = compute_c5(best_h, 2.0 / n)
        best_n = n

    return best_h, best_c5, best_n
