import numpy as np

def project_feasible(z, T):
    """Project z onto {x in [0,1]^n : sum(x) = T}"""
    n = len(z)
    # Bisection for lambda such that sum(clip(z+lambda, 0, 1)) = T
    lo = -z.max() - 0.1
    hi = T - z.min() - n * 0.0 + 1.1
    # Better bounds:
    lo = min(z) * -1 - 0.5
    hi = (T - n * 0) / n - min(z) + 0.5  # This isn't right
    
    # Simpler: lo = -max(z) - 1, hi = 1 - min(z) + 1
    lo = -np.max(z) - 1.0
    hi = 1.0 - np.min(z) + 1.0
    
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < T:
            lo = mid
        else:
            hi = mid
    
    x = np.clip(z + 0.5 * (lo + hi), 0.0, 1.0)
    
    # Fix to get exact sum
    diff = T - x.sum()
    if abs(diff) > 1e-13:
        # Sequential correction
        step_dir = 1 if diff > 0 else -1
        remaining = abs(diff)
        for idx in range(n):
            if remaining < 1e-14:
                break
            if step_dir > 0:
                room = 1.0 - x[idx]
                adj = min(remaining, room) if room > 0 else 0
            else:
                room = x[idx]
                adj = min(remaining, room) if room > 0 else 0
            x[idx] += step_dir * adj
            remaining -= adj
    return x

    """Compute all 2n-1 lag overlap values (unscaled)"""
    n = len(h)
    o = 1.0 - h
    vals = np.zeros(2 * n - 1)
    for d in range(n):
        m = n - d
        vals[n - 1 + d] = np.dot(h[:m], o[d:d+m])
        if d > 0:
            vals[n - 1 - d] = np.dot(h[d:d+m], o[:m])
    return vals

def smooth_grad(h, beta):
    """Compute smooth objective value and gradient"""
    n = len(h)
    dx = 2.0 / n
    vals = get_lag_values(h)
    
    # Softmax
    max_v = vals.max()
    exps = np.exp(beta * (vals - max_v))
    p = exps / exps.sum()
    
    obj = dx * (max_v + np.log(exps.sum()) / beta)
    
    grad = np.zeros(n)
    o = 1.0 - h
    for d in range(n):
        m = n - d
        if d == 0:
            grad += dx * p[n-1] * (1.0 - 2.0 * h)
        else:
            # Lag +d
            wp = dx * p[n-1+d]
            grad[:m] += wp * o[d:d+m]
            grad[d:] -= wp * h[:m]
            # Lag -d
            wn = dx * p[n-1-d]
            grad[d:] += wn * o[:m]
            grad[:m] -= wn * h[d:d+m]
    
    return obj, grad

import numpy as np

def project_feasible(z, T):
    n = len(z)
    lo = -np.max(z) - 1.0
    hi = 1.0 - np.min(z) + 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(z + mid, 0.0, 1.0))
        if s < T:
            lo = mid
        else:
            hi = mid
    x = np.clip(z + 0.5 * (lo + hi), 0.0, 1.0)
    diff = T - x.sum()
    if abs(diff) > 1e-13:
        remaining = diff
        for idx in range(2 * n):
            i = idx % n
            if abs(remaining) < 1e-14:
                break
            if remaining > 0:
                room = 1.0 - x[i]
                if room > 0:
                    adj = min(remaining, room)
                    x[i] += adj
                    remaining -= adj
            else:
                room = x[i]
                if room > 0:
                    adj = min(-remaining, room)
                    x[i] -= adj
                    remaining += adj
    return x

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def get_lag_values(h):
    n = len(h)
    o = 1.0 - h
    vals = np.zeros(2 * n - 1)
    for d in range(n):
        m = n - d
        vals[n - 1 + d] = np.dot(h[:m], o[d:d + m])
        if d > 0:
            vals[n - 1 - d] = np.dot(h[d:d + m], o[:m])
    return vals

def smooth_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    vals = get_lag_values(h)
    max_v = vals.max()
    exps = np.exp(np.clip(beta * (vals - max_v), -500, 500))
    total = exps.sum()
    p = exps / total
    obj = dx * (max_v + np.log(total) / beta)
    grad = np.zeros(n)
    o = 1.0 - h
    for d in range(n):
        m = n - d
        if d == 0:
            grad += dx * p[n - 1] * (1.0 - 2.0 * h)
        else:
            wp = dx * p[n - 1 + d]
            grad[:m] += wp * o[d:d + m]
            grad[d:] -= wp * h[:m]
            wn = dx * p[n - 1 - d]
            grad[d:] += wn * o[:m]
            grad[:m] -= wn * h[d:d + m]
    return obj, grad

def optimize_pgd(h, best_c, beta, max_iters, time_fn, alpha_init=0.005):

    h = h.copy()
    c = eval_c5(h)

    best_local_h = h.copy()
    alpha = alpha_init

    for it in range(max_iters):
        if time_fn() < 8.0:
            break
        obj, g = smooth_grad(h, beta)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-12:
            break
        improved = False
        for t in range(30):
            cand = project_feasible(h - alpha * g, T)
            c_cand = eval_c5(cand)
            if c_cand < c - 1e-13:
                h = cand
                c = c_cand
                if c < best_local_c:
                    best_local_c = c
                    best_local_h = h.copy()
                improved = True
                alpha = min(alpha * 1.05, alpha_init * 5)
                break

            if alpha < 1e-11:

        if not improved:

            if alpha < 1e-12:

def pair_optimize(h, p, q, n_grid=30):
    lo = max(-h[p], h[q] - 1.0)
    hi = min(1.0 - h[p], h[q])
    if hi - lo < 1e-12:
        return h.copy(), eval_c5(h)
    n = len(h)
    T = n / 2.0

    best_c = c_curr
    best_eps = 0.0
    grid = np.linspace(lo, hi, n_grid)

        h_new = h.copy()
        h_new[p] += eps
        h_new[q] -= eps
        c = eval_c5(h_new)
        if c < best_c:

            best_eps = eps
    if best_eps == 0.0:
        return h.copy(), c_curr
    idx = int((best_eps - lo) / (hi - lo) * (n_grid - 1))
