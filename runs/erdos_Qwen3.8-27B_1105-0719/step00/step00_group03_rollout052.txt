```python
import numpy as np
from scipy.optimize import minimize
import time

def c5_value(h):
    """Compute exact C5 value for a given h vector."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(corr)

def lse_objective_and_grad(h, alpha):
    """Compute log-sum-exp objective and its gradient."""
    n = len(h)
    dx = 2.0 / n
    q = 1.0 - h
    S = np.correlate(h, q, mode="full") * dx
    F = np.max(S)
    shifted = alpha * (S - F)
    max_shifted = np.max(shifted)
    exp_vals = np.exp(shifted - max_shifted)
    w = exp_vals / np.sum(exp_vals)
    obj = F + (1.0 / alpha) * (max_shifted + np.log(np.sum(exp_vals)))
    
    # Gradient
    term1 = np.convolve(1.0 - h, w, 'full')[n-1 : 2*n-1]
    term2 = np.convolve(h, w[::-1], 'full')[n-1 : 2*n-1]
    grad = dx * (term1 - term2)
    
    return obj, grad

def optimize_h(h0, n, alphas, budget_s, start_time):
    """Optimize h using SLSQP with log-sum-exp smoothing."""
    best_h = h0.copy()
    best_c5 = c5_value(h0)
    
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target, 
                    'jac': lambda x: np.ones(n)}]
    
    current_h = h0.copy()
    
    for alpha in alphas:
        if time.time() - start_time > budget_s - 5:
            break
        
        try:
            result = minimize(
                lse_objective_and_grad,
                current_h,
                args=(alpha,),
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                jac=True,
                options={'maxiter': 200, 'ftol': 1e-10}
            )
            
            if result.success or result.fun < lse_objective_and_grad(current_h, alpha)[0]:
                new_h = np.clip(result.x, 0, 1)
                # Project to exact sum constraint
                diff = target - np.sum(new_h)
                new_h += diff / n
                new_h = np.clip(new_h, 0, 1)
                
                new_c5 = c5_value(new_h)
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = new_h.copy()
                    current_h = new_h.copy()
        except Exception:
            pass
    
    return best_h, best_c5
```

```python
        pc5 = c5_val(parent_h)
        best_h = parent_h.copy()
        best_c5 = pc5
    else:
        n0 = 45
        parent_h = np.full(n0, 0.5)
        best_h = parent_h.copy()
        best_c5 = c5_val(parent_h)
    
    pn = len(parent_h)
    
    def try_update(h, c5):
        nonlocal best_h, best_c5
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    # N values to try, ordered by priority
    n_vals = [pn]
    for cand in [44, 46, 50, 55, 60, 64, 72, 80, 90, 100]:
        if cand != pn:
            n_vals.append(cand)
    
    alphas = [5, 10, 20, 40, 80, 160, 320, 640, 1280]
    
    # Time allocation: spend more time on first few n values
    time_per_n = max(30, (budget_s - 60) / len(n_vals[:5]))
    
    for ni, n in enumerate(n_vals):
        elapsed = time.time() - t0
        remaining = budget_s - elapsed
        if remaining < 30:
            break
        # For later n values, use less time
        alloc = min(time_per_n, remaining - 10)
        if ni > 4:
            alloc = min(alloc, 40)
        
        tgt = n / 2.0
        
        # Starting points
        starts = []
        
        # From parent
        if pn == n:
            starts.append(parent_h.copy())
        else:
            x_old = np.arange(pn, dtype=float)
            x_new = np.linspace(0, pn - 1, n)
            interp = np.interp(x_new, x_old, parent_h)
            starts.append(proj_bs(interp, tgt))
        
        # From best found so far (resampled)
        if best_h is not None and len(best_h) != n:
            x_old2 = np.arange(len(best_h), dtype=float)
            x_new2 = np.linspace(0, len(best_h) - 1, n)
            interp2 = np.interp(x_new2, x_old2, best_h)
            starts.append(proj_bs(interp2, tgt))
        elif best_h is not None and len(best_h) == n:
            starts.append(best_h.copy())
        
        # Random starts
        for _ in range(2):
            rh = rng.random(n)
            starts.append(proj_bs(rh, tgt))
        
        # Structured: golden ratio
        phi = (1.0 + np.sqrt(5.0)) / 2.0
        struct = np.zeros(n)
        for i in range(n):
            struct[i] = 1.0 if ((i * phi) % 1.0) < 0.5 else 0.0
        starts.append(proj_bs(struct, tgt))
        
        # Alternating
        alt = np.zeros(n)
        alt[::2] = 1.0
        starts.append(proj_bs(alt, tgt))
        
        # Optimize from each start
        for h0 in starts:
            if time.time() - t0 > t0 + alloc + elapsed - 10:
                break
            cur, cur_c5 = optimize_at_n(h0, n, alphas, t0, t0 + alloc + elapsed)
            try_update(cur, cur_c5)
            
            # Local search
            if time.time() - t0 < t0 + alloc + elapsed - 5:
                try:
                    ls_h, ls_c5 = local_search(cur, max_rounds=200, rng=rng)
                    try_update(ls_h, ls_c5)
                except Exception:
                    pass
    
    # Final: try local search on the best solution found
    if time.time() - t0 < budget_s - 10 and best_h is not None:
        try:
            ls_h, ls_c5 = local_search(best_h, max_rounds=1000, rng=rng)
            try_update(ls_h, ls_c5)
        except Exception:
            pass
    
    # One more round of SLSQP on the best if time permits
    if time.time() - t0 < budget_s - 20 and best_h is not None:
        n_best = len(best_h)
        _GT = n_best / 2.0
        bounds = [(0.0, 1.0)] * n_best
        cons = [{'type': 'eq', 'fun': _cfun, 'jac': _cjac}]
        try:
            for alpha in [2000, 5000, 10000]:
                if time.time() - t0 > budget_s - 5:
                    break
                res = minimize(lse_fn, best_h, args=(alpha,),
                               method='SLSQP', bounds=bounds,
                               constraints=cons, jac=True,
                               options={'maxiter': 50, 'ftol': 1e-14})
                if res.x is not None and np.all(np.isfinite(res.x)):
                    new_h = proj_bs(res.x, n_best / 2.0)
                    new_c5 = c5_val(new_h)
                    try_update(new_h, new_c5)
        except Exception:
            pass
    
    final_c5 = c5_val(best_h)
    return (best_h, final_c5, len(best_h))
```
