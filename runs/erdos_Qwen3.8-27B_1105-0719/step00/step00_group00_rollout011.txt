import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def smooth_max(corr, beta):
    m = np.max(corr)
    if m < -1e100:
        return 0.0, np.zeros_like(corr)
    shifted = beta * (corr - m)
    exps = np.exp(shifted)
    s = np.sum(exps)
    val = m + np.log(s) / beta
    weights = exps / s
    return val, weights

def objective_and_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    val, weights = smooth_max(corr, beta)
    
    # Gradient
    grad = np.zeros(n)
    # For each lag i in 0..2n-2, k = i - (n-1)
    # dC[i]/dh[j] = (1-h[j+k])*I(0<=j+k<n) - h[j-k]*I(0<=j-k<n)
    # grad[j] = sum_i weights[i] * dC[i]/dh[j]
    
    for i in range(2*n - 1):
        k = i - (n - 1)
        w = weights[i]
        if w < 1e-15:
            continue
        # (1-h[j+k]) for valid j
        if k >= 0:
            # j+k in [0,n-1] means j in [0, n-1-k]
            # h[j-k] for valid j means j-k in [0,n-1] means j in [k, n-1]
            # Term 1: (1-h[j+k]) for j in [0, n-1-k]
            # Term 2: -h[j-k] for j in [max(0,k), n-1]
            j_max1 = n - 1 - k
            if j_max1 >= 0:
                grad[0:j_max1+1] += w * (1.0 - h[k:k+j_max1+1])
            j_min2 = max(0, k)
            if j_min2 < n:
                grad[j_min2:n] -= w * h[0:n-j_min2]
        else:
            # k < 0, let k' = -k > 0
            kp = -k
            # j+k in [0,n-1] means j-kp in [0,n-1] means j in [kp, n-1]
            # h[j-k] = h[j+kp] for valid means j+kp in [0,n-1] means j in [0, n-1-kp]
            j_min1 = kp
            if j_min1 < n:
                grad[j_min1:n] += w * (1.0 - h[0:n-j_min1])
            j_max2 = n - 1 - kp
            if j_max2 >= 0:
                grad[0:j_max2+1] -= w * h[kp:kp+j_max2+1]
    
    return val * dx, grad * dx

def optimize_slsqp(h0, n, beta_schedule, time_limit):
    start_time = time.time()
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n/2.0,
                   'jac': lambda h: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    
    for beta in beta_schedule:
        if time.time() - start_time > time_limit:
            break
        
        res = minimize(
            lambda h: objective_and_grad(h, beta)[0],
            best_h,
            jac=lambda h: objective_and_grad(h, beta)[1],
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 300, 'ftol': 1e-10}
        )
        
        if res.x is not None:
            h_new = res.x
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
    
    return best_h, best_c5

            smooth_obj, best_h, args=(beta,),
            jac=smooth_grad,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': max_iter, 'ftol': 1e-12}

            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = res.x.copy()
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def smooth_max_val_weights(corr, beta):
    m = np.max(corr)
    shifted = beta * (corr - m)
    exps = np.exp(shifted)
    s = np.sum(exps)
    val = m + np.log(s) / beta
    weights = exps / s
    return val, weights


def smooth_obj(h, beta):
    corr = np.correlate(h, 1.0 - h, mode="full")
    val, _ = smooth_max_val_weights(corr, beta)
    return val * (2.0 / len(h))


def smooth_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    _, weights = smooth_max_val_weights(corr, beta)
    grad = np.zeros(n)
    for i in range(2 * n - 1):
        k = i - (n - 1)
        w = weights[i]
        if w < 1e-12:
            continue
        m_lo1 = max(0, k)
        m_hi1 = min(n - 1, n - 1 + k)
        if m_lo1 <= m_hi1:
            grad[m_lo1:m_hi1 + 1] += w * (1.0 - h[m_lo1 - k:m_hi1 + 1 - k])
        m_lo2 = max(0, -k)
        m_hi2 = min(n - 1, n - 1 - k)
        if m_lo2 <= m_hi2:
            grad[m_lo2:m_hi2 + 1] -= w * h[m_lo2 + k:m_hi2 + 1 + k]
    return grad * dx


def eq_con_fun(h):
    return np.sum(h) - len(h) / 2.0


def eq_con_jac(h):
    return np.ones(len(h))


def project_feasible(y, n):
    target = n / 2.0

        if len(eligible) > 0:
            h[eligible] += residual / len(eligible)
            h = np.clip(h, 0.0, 1.0)
    return h

        res = minimize(

            jac=smooth_grad,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': max_iter, 'ftol': 1e-14}
        )
        if res.x is not None:
            c5_new = compute_c5(res.x)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = res.x.copy()
    return best_h, best_c5
