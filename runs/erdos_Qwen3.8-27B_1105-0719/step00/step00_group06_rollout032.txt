def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def do_line_search(h, i, j, n_iter):
    s_lo = max(-h[i], h[j] - 1.0) + 1e-10
    s_hi = min(1.0 - h[i], h[j]) - 1e-10
    if s_lo >= s_hi:
        return compute_c5(h)

    hj0 = h[j]
    f0 = compute_c5(h)
    best_s = 0.0
    best_f = f0

    b = s_hi
    c = b - gr * (b - a)
    d = a + gr * (b - a)

    h[j] = hj0 - c
    fc = compute_c5(h)

def sturmian_init(n, rng):
    alpha = rng.uniform(0.3, 0.7)
    rho = rng.uniform(0, 1)
    seq = np.zeros(n)
    for i in range(n):
        if frac(i * alpha + rho) < alpha:
            seq[i] = 1.0
    # Ensure sum = n/2
    target = n / 2.0
    s = seq.sum()
    if s != target:
        # Flip some bits
        ones = np.where(seq == 1)[0]
        zeros = np.where(seq == 0)[0]
        if s > target:
            to_flip = int(s - target)
            idx = rng.choice(ones, size=min(to_flip, len(ones)), replace=False)
            seq[idx] = 0
        else:
            to_flip = int(target - s)
            idx = rng.choice(zeros, size=min(to_flip, len(zeros)), replace=False)
            seq[idx] = 1
    # Relax to continuous
    eps = 0.1
    return np.where(seq > 0.5, 1.0 - eps, eps)

def frac(x):
    return x - math.floor(x)
