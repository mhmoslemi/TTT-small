import numpy as np
import time

def erdos_objective(h):
    """Compute C5 = max(np.correlate(h, 1-h, 'full')) * dx"""
    n = len(h)
    dx = 2.0 / n
    prof = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(prof) * dx)

def erdos_profile(h):
    """Compute raw profile (before scaling by dx)"""
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full')

def repair_vector(h, n):
    """Repair h to satisfy 0<=h<=1 and sum(h)=n/2"""
    h = np.array(h, dtype=np.float64).flatten()
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            # Need to add mass
            available = h < 1.0 - 1e-12
            count = np.sum(available)
            if count == 0:
                break
            step = min(diff / count, (1.0 - h[available]).min())
            h[available] += step
        else:
            # Need to remove mass
            available = h > 1e-12
            count = np.sum(available)
            if count == 0:
                break
            step = min(-diff / count, h[available].min())
            h[available] -= step
    h = np.clip(h, 0.0, 1.0)
    # Final correction
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        # Distribute to one element
        if diff > 0:
            idx = np.argmin(h)
        else:
            idx = np.argmax(h)
        h[idx] += diff
        h[idx] = np.clip(h[idx], 0.0, 1.0)
        # If still off, distribute
        diff2 = target - np.sum(h)
        if abs(diff2) > 1e-10:
            mask = (h > 0) & (h < 1)
            if np.sum(mask) > 0:
                h[mask] += diff2 / np.sum(mask)
                h = np.clip(h, 0.0, 1.0)
    return h

def build_alternating(n):
    """Build a simple alternating 0-1 vector with sum = n/2"""
    h = np.zeros(n)
    if n % 2 == 0:
        h[::2] = 1.0
    else:
        h[:(n-1)//2] = 1.0  # Place ones in first half
        # Actually for odd n, n/2 is not integer, so we need fractional
        # Let's just use 0.5 everywhere for odd n
        h[:] = 0.5
    return h

def build_block(n, block_size):
    """Build a block pattern"""
    h = np.zeros(n)
    for start in range(0, n, 2 * block_size):
        h[start:start+block_size] = 1.0
    return h

def golden_section_1d(h, i, j, t_lo, t_hi, max_iters=40):
    """Minimize max(correlate) along the line h[i]+=t, h[j]-=t"""
    n = len(h)
    dx = 2.0 / n
    rho = (np.sqrt(5) - 1) / 2  # golden ratio conjugate ~0.618
    
    def f(t):
        h_new = h.copy()
        h_new[i] += t
        h_new[j] -= t
        if h_new[i] < -1e-10 or h_new[i] > 1 + 1e-10 or h_new[j] < -1e-10 or h_new[j] > 1 + 1e-10:
            return np.inf
        prof = np.correlate(h_new, 1.0 - h_new, mode='full')
        return np.max(prof)
    
    a, b = t_lo, t_hi
    c = b - rho * (b - a)
    d = a + rho * (b - a)
    fc = f(c)
    fd = f(d)
    
    for _ in range(max_iters):
        if b - a < 1e-10:
            break
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - rho * (b - a)
            fc = f(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + rho * (b - a)
            fd = f(d)
    
    t_best = (a + b) / 2
    f_best = f(t_best)
    # Also check endpoints
    f0 = f(0.0)
    if f0 < f_best:
        t_best = 0.0
        f_best = f0
    f_lo = f(t_lo)
    if f_lo < f_best:
        t_best = t_lo
        f_best = f_lo
    f_hi = f(t_hi)
    if f_hi < f_best:
        t_best = t_hi
        f_best = f_hi
    
    return t_best, f_best

def compute_active_gradients(h, n, top_k=10):
    """Compute gradients for the active (top) lags"""
    prof = erdos_profile(h)
    n_prof = len(prof)  # 2n-1
    # Find top k active lags
    top_indices = np.argsort(prof)[::-1][:top_k]
    
    # Store gradients: shape (top_k, n)
    grads = np.zeros((len(top_indices), n))
    
    for idx, ki in enumerate(top_indices):
        # ki is index in the profile array
        # lag = ki - (n-1)
        lag = ki - (n - 1)
        g = np.zeros(n)
        if lag >= 0:
            s = lag
            # dPhi_s/dh[p] = (1-h[p-s]) * [0<=p-s<=n-1-s] - h[p+s] * [0<=p<=n-1-s]
            for p in range(n):
                if 0 <= p - s <= n - 1 - s:
                    g[p] += (1.0 - h[p - s])
                if 0 <= p <= n - 1 - s and p + s < n:
                    g[p] -= h[p + s]
        elif lag < 0:
            a = -lag
            for p in range(n):
                if a <= p <= n - 1:
                    g[p] += (1.0 - h[p - a])
                if 0 <= p <= n - 1 - a:
                    g[p] -= h[p + a]
        else:
            g[:] = 1.0 - 2.0 * h
        grads[idx] = g
    
    return prof, top_indices, grads, [prof[ki] for ki in top_indices]

def select_pairs(h, n, grads, n_pairs=20, rng=None):
    """Select promising pairs (i,j) for exchange"""
    if n_pairs <= 0:
        return []
    
    pairs = []
    scores = []
    
    for i in range(n):
        for j in range(n):
            if i == j:
                continue
            # Score: minimum over active lags of (grad[i] - grad[j])
            # We want this to be negative (descent direction)
            if len(grads) > 0:
                diff = grads[:, i] - grads[:, j]
                score = np.min(diff)
            else:
                score = 0.0
            # Only consider if h[i] < 1 and h[j] > 0 (feasible to move)
            if h[i] < 1.0 - 1e-8 and h[j] > 1e-8:
                pairs.append((i, j))
                scores.append(score)
    
    if len(pairs) == 0:
        return []
    
    # Sort by score (most negative first = most promising descent)
    order = np.argsort(scores)
    selected = []
    for idx in order[:min(n_pairs, len(pairs))]:
        selected.append(pairs[idx])
    
    # Add some random pairs for diversity
    if rng is not None and len(pairs) > len(selected):
        remaining = [p for p in pairs if p not in selected]
        extra = min(5, len(remaining))
        if extra > 0:
            idxs = rng.choice(len(remaining), size=extra, replace=False)
            for idx in idxs:
                selected.append(remaining[idx])
    
    return selected

def pair_descent_pass(h, n, deadline, max_pairs_per_pass=15, rng=None, verbose=False):
    """Run one pass of pair descent"""
    curvature_best = erdos_profile(h)
    cur_max_raw = np.max(curvature_best)
    improved = False
    n_accepted = 0
    
    # Compute gradients
    prof, top_idx, grads, active_vals = compute_active_gradients(h, n, top_k=8)
    
    # Select pairs
    pairs = select_pairs(h, n, grads, n_pairs=max_pairs_per_pass, rng=rng)
    
    for (i, j) in pairs:
        if time.monotonic() > deadline - 5.0:
            break
        
        # Feasible range for t
        t_lo = max(-h[i], h[j] - 1.0) + 1e-12
        t_hi = min(1.0 - h[i], h[j]) - 1e-12
        if t_hi <= t_lo:
            continue
        
        # Golden section search
        t_best, f_best = golden_section_1d(h, i, j, t_lo, t_hi, max_iters=35)
        
        # Check if it improves

            h[i] += t_best
            h[j] -= t_best
            h = np.clip(h, 0.0, 1.0)
            # Recompute

            new_max = np.max(new_prof)
            if new_max < cur_max_raw - 1e-10:

                h[i] -= t_best
                h[j] += t_best
