import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def _sigmoid(t):
    """Stable sigmoid."""
    result = np.empty_like(t, dtype=float)
    pos = t >= 0
    result[pos] = 1.0 / (1.0 + np.exp(-t[pos]))
    exp_t = np.exp(t[~pos])
    result[~pos] = exp_t / (1.0 + exp_t)
    return result

def _exact_C(h):
    """Compute exact C5 value."""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def _validate(h, tol=1e-8):
    """Check if h is feasible."""
    n = len(h)
    T = n / 2.0
    if not np.all(np.isfinite(h)):
        return False
    if np.min(h) < -tol or np.max(h) > 1.0 + tol:
        return False
    if abs(np.sum(h) - T) > tol:
        return False
    return True

def _project_feasible(u, n=None):
    """Project u to feasible set [0,1]^n with sum = n/2 using water filling."""
    h = np.array(u, dtype=float)

        n = len(h)

    # Bisection for lambda such that sum clip(h + lambda, 0, 1) = T
    lo = -max(h.max() - 0, 1.0) - 1  # ensure sum <= T
    hi = -min(h.min() - 1, 1.0) + 1   # ensure sum >= T
    
    # Better bounds
    lo = min(h) - 1.0
    hi = max(h) - 1.0 + 2.0  # hmm
    
    # Let's think more carefully
    # f(lambda) = sum clip(h_i + lambda, 0, 1) is monotone increasing in lambda
    # f(-inf) = 0, f(inf) = n
    # We want f(lambda) = n/2
    
    lo = h.min() - 1.5
    hi = h.max() + 0.5
    
    # Verify
    for _ in range(100):
        mid = (lo + hi) / 2.0
        val = np.sum(np.clip(h + mid, 0.0, 1.0))
        if val < T:
            lo = mid
        else:
            hi = mid
    
    lam = (lo + hi) / 2.0
    result = np.clip(h + lam, 0.0, 1.0)
    
    # Fix residual
    delta = T - np.sum(result)
    if abs(delta) > 1e-12:
        # Distribute to coordinates with most slack
        if delta > 0:
            idx = np.argsort(1.0 - result)  # most room first
        else:
            idx = np.argsort(result)  # most value first
        remaining = abs(delta)
        for i in idx:
            if delta > 0:
                add = min(1.0 - result[i], remaining)
                result[i] += add
                remaining -= add
            else:
                sub = min(result[i], -remaining)
                result[i] -= sub
                remaining += sub
            if remaining < 1e-15:
                break
    
    return result

def _logit_from_h(h):
    """Convert h to logit space (inverse sigmoid)."""
    eps = 1e-7
    h_safe = np.clip(h, eps, 1.0 - eps)
    return np.log(h_safe / (1.0 - h_safe))

def _h_from_logit(y):
    """Convert logit space to h, solving for lambda to satisfy sum constraint."""
    n = len(y)
    T = n / 2.0
    
    def phi(lam):
        return np.sum(_sigmoid(y + lam)) - T
    
    # Find bracket
    lo = -np.max(y) - 50.0
    hi = -np.min(y) + 50.0
    
    # Ensure bracket
    if phi(lo) > 0:
        lo -= 100
    if phi(hi) < 0:
        hi += 100

        if phi(mid) < 0:
            lo = mid
        else:
            hi = mid
    
    lam = (lo + hi) / 2.0
    h = _sigmoid(y + lam)
    return h

def _directional_values(h):
    """Compute all directional overlap values (without dx)."""
    n = len(h)
    values = np.empty(2 * n - 1)
    one_minus_h = 1.0 - h
    for k in range(-(n-1), n):
        if k == 0:
            values[k + n - 1] = np.dot(h, one_minus_h)
        elif k > 0:
            values[k + n - 1] = np.dot(h[:n-k], one_minus_h[k:])
        else:
            p = -k
            values[k + n - 1] = np.dot(h[p:], one_minus_h[:n-p])
    return values

def _directional_gradients(h):
    """Compute gradients of all directional values w.r.t. h.
    Returns array of shape (2n-1, n)."""
    n = len(h)
    grads = np.empty((2 * n - 1, n))
    one_minus_h = 1.0 - h
    
    # k = 0
    grads[n-1, :] = 1.0 - 2.0 * h
    
    for k in range(1, n):
        # Positive lag k: v = sum_{i=0}^{n-1-k} h[i]*(1-h[i+k])
        # dv/dh[m] = (1-h[m]) for m < n-k, plus -h[m-k] for m >= k
        idx = k + n - 1
        g = np.zeros(n)
        g[:n-k] += (1.0 - h[:n-k])
        if k < n:
            g[k:] += (-h[:n-k])
        grads[idx, :] = g
        
        # Negative lag -k: v = sum_{i=k}^{n-1} h[i]*(1-h[i-k])
        idx = -k + n - 1
        g = np.zeros(n)
        g[k:] += (1.0 - h[k:])
        g[:n-k] += (-h[k:])
        grads[idx, :] = g
    
    return grads

def _smooth_objective_grad(h, beta):
    """Compute smooth surrogate F_beta and its gradient w.r.t. h."""
    n = len(h)
    dx = 2.0 / n
    values = _directional_values(h) * dx
    v_max = np.max(values)
    
    # Stable softmax
    shifted = beta * (values - v_max)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    
    F = v_max + np.log(np.sum(exp_vals)) / beta
    
    # Gradient: sum_j w[j] * grad_j
    grads = _directional_gradients(h) * dx  # shape (2n-1, n)
    G = np.dot(w, grads)  # shape (n,)
    
    return F, G

def _transformed_objective_grad(y, beta):
    """Objectives in logit space."""
    h = _h_from_logit(y)
    F, G = _smooth_objective_grad(h, beta)
    
    s = h * (1.0 - h)
    S = np.sum(s)
    if S < 1e-10:
        return F, np.zeros_like(y)
    
    B = np.dot(G, s) / S
    grad_y = s * (G - B)
    
    if not np.all(np.isfinite(grad_y)):
        return F, np.zeros_like(y)
    
    return F, grad_y

def _pair_transfer(h, a, b, max_iter=20):
    """Optimize transfer of mass between coordinates a and b."""
    n = len(h)
    dx = 2.0 / n
    C_cur = _exact_C(h)
    
    # Valid range for t: h[a]+t in [0,1], h[b]-t in [0,1]
    t_lo = max(-h[a], h[b] - 1.0)
    t_hi = min(1.0 - h[a], h[b])
    
    if t_hi - t_lo < 1e-12:
        return h, C_cur
    
    def obj(t):
        h_new = h.copy()
        h_new[a] += t
        h_new[b] -= t
        # Quick check
        if h_new[a] < -1e-10 or h_new[a] > 1 + 1e-10 or h_new[b] < -1e-10 or h_new[b] > 1 + 1e-10:
            return 1.0
        return _exact_C(h_new)
    
    # Grid search first
    n_grid = 9
    t_grid = np.linspace(t_lo, t_hi, n_grid)
    obj_grid = np.array([obj(t) for t in t_grid])
    best_idx = np.argmin(obj_grid)
    best_t = t_grid[best_idx]
    best_C = obj_grid[best_idx]
    
    # Refine with bounded search
    if 0 < best_idx < n_grid - 1:
        left = t_grid[best_idx - 1]
        right = t_grid[best_idx + 1]
    elif best_idx == 0:
        left = t_lo
        right = t_grid[1]
    else:
        left = t_grid[n_grid - 2]
        right = t_hi
    
    try:
        res = minimize_scalar(obj, bounds=(left, right), method='bounded',
                             options={'xatol': 1e-10, 'maxiter': max_iter})
        if res.success and res.fun < best_C:
            best_t = res.x
            best_C = res.fun
    except Exception:
        pass
    
    if best_C < C_cur - 1e-14:
        h_new = h.copy()
        h_new[a] += best_t
        h_new[b] -= best_t
        h_new = np.clip(h_new, 0.0, 1.0)
        return h_new, best_C
    
    return h, C_cur

def _gradient_based_pairs(h, beta, n_pairs=30):
    """Suggest pairs based on gradient."""
    n = len(h)
    _, G = _smooth_objective_grad(h, beta)
    
    # Top coordinates by |G|
    top = np.argsort(-np.abs(G))[:min(12, n)]
    
    pairs = set()
    for i in range(len(top)):
        for j in range(i+1, len(top)):
            pairs.add((int(top[i]), int(top[j])))
            if len(pairs) >= n_pairs:
                break
        if len(pairs) >= n_pairs:
            break
    
    # Add random pairs
    rng = np.random.default_rng(12345)
    while len(pairs) < n_pairs:
        a = rng.integers(n)
        b = rng.integers(n)
        if a != b:
            pairs.add((min(int(a), int(b)), max(int(a), int(b))))
    
    return list(pairs)

def _sigmoid(t):
    out = np.empty_like(t, dtype=float)
    pos = t >= 0
    out[pos] = 1.0 / (1.0 + np.exp(-t[pos]))
    e = np.exp(t[~pos])
    out[~pos] = e / (1.0 + e)
    return out

def _exact_C(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _project_feasible(u):
    h = np.asarray(u, dtype=float)
    n = len(h)
    T = n / 2.0
    # Water filling via bisection
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(h + mid, 0.0, 1.0)) < T:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    res = np.clip(h + lam, 0.0, 1.0)
    # Fix any residual
    d = T - res.sum()
    if abs(d) > 1e-14:
        if d > 0:
            idx = np.argsort(1.0 - res)
        else:
            idx = np.argsort(res)
        rem = abs(d)
        for i in idx:
            if d > 0:
                add = min(1.0 - res[i], rem)
                res[i] += add
                rem -= add
            else:
                sub = min(res[i], rem)
                res[i] -= sub
                rem -= sub
            if rem < 1e-15:
                break
    return res

        elif k > 0:

        else:
            p = -k
            vals[j] = np.dot(h[p:], omh[:n-p]) * dx
    return vals

def _dir_grads(h):
    """Gradients of directional values (times dx). Shape (2n-1, n)."""
    n = len(h)
    dx = 2.0 / n
    grads = np.zeros((2 * n - 1, n))
    omh = 1.0 - h
    # k=0
    grads[n-1] = (1.0 - 2.0 * h) * dx
    for k in range(1, n):

        # dv/dh[m] = (1-h[m])*dx for m in [0, n-1-k)
        #           + (-h[m-k])*dx for m in [k, n-1]
        g = np.zeros(n)
        g[:n-k] += (1.0 - h[:n-k]) * dx
        g[k:] += (-h[:n-k]) * dx
        grads[k + n - 1] = g
        # negative lag -k: v = sum_{i=k}^{n-1} h[i]*(1-h[i-k]) * dx
        g = np.zeros(n)
        g[k:] += (1.0 - h[k:]) * dx
        g[:n-k] += (-h[k:]) * dx
        grads[-k + n - 1] = g
    return grads

def _smooth_fg(h, beta):
    """Smooth objective and gradient."""
    vals = _dir_values(h)
    vmax = vals.max()
    shifted = beta * (vals - vmax)
    ev = np.exp(shifted)
    w = ev / ev.sum()
    F = vmax + np.log(ev.sum()) / beta
    G = np.dot(w, _dir_grads(h))
    return F, G

    """Transformed (logit-space) objective and gradient."""
    h = _h_from_logit(y)
    F, G = _smooth_fg(h, beta)
    s = h * (1.0 - h)
    S = s.sum()
    if S < 1e-12:

    B = (G * s).sum() / S
    gy = s * (G - B)

    return F, gy
