def project_feasible(h, n):

    target = n / 2.0
    h = np.clip(h, 0, 1)
    # Adjust to meet sum constraint
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    # Distribute the difference proportionally to the "slack" coordinates
    # Coordinates at 0 can be increased, coordinates at 1 can be decreased
    if diff > 0:

        slack = 1.0 - h  # how much room to increase

        if total_slack < 1e-12:

            return h
        h = h + diff * slack / total_slack
    else:

        slack = h  # how much room to decrease
        total_slack = slack.sum()
        if total_slack < 1e-12:

            h[idx] = min(1.0, h[idx] + diff)

from scipy.optimize import linprog
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def project_feasible(h, n):
    """Project onto {h: 0<=h<=1, sum(h)=n/2}"""
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0, 1)
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        # Need to increase; add to coordinates with most slack
        slack = 1.0 - h
        total = slack.sum()
        if total > 1e-14:
            h += diff * slack / total
        else:
            # All at 1, redistribute
            h[:] = 1.0
            h[target.astype(int)] = target - int(target)  # won't work well for non-integer
            # Actually if all at 1 and sum > target, we need to decrease
            # This case means target > n which is impossible
            pass
    else:
        # Need to decrease; remove from coordinates with most value
        slack = h
        total = slack.sum()
        if total > 1e-14:
            h += diff * slack / total  # diff is negative
        else:
            pass
    h = np.clip(h, 0, 1)
    # Fine tune
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        # Add to or remove from the coordinate with most room
        if diff > 0:
            idx = np.argmax(1.0 - h)
            add = min(diff, 1.0 - h[idx])
            h[idx] += add
        else:
            idx = np.argmax(h)
            rem = min(-diff, h[idx])
            h[idx] -= rem
    return h

def build_A_matrix(n, fixed):
    """Build the (2n-1) x n matrix for the LP.
    A[k, i] = fixed[i + lag_k] if valid, else 0.
    Where lag_k = (n-1) - k for np.correlate convention."""
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        lag = (n-1) - k
        if lag >= 0:
            for i in range(n - lag):
                A[k, i] = fixed[i + lag]
        else:
            for i in range(-lag, n):
                A[k, i] = fixed[i + lag]
    return A

def solve_lp(n, fixed_vec):
    """Solve the LP: min t s.t. A @ x * dx <= t, sum(x) = n/2, 0<=x<=1
    where A is built from fixed_vec.
    Returns the optimal x."""
    dx = 2.0 / n
    A = build_A_matrix(n, fixed_vec)
    
    # Variables: [x_0, ..., x_{n-1}, t]
    # Objective: minimize t -> c = [0,...,0, 1]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Inequality constraints: A @ x * dx <= t -> A @ x - t/dx <= 0
    # In standard form for linprog: A_ub @ vars <= b_ub
    A_ub = np.zeros((2*n-1, n+1))
    A_ub[:, :n] = A * dx
    A_ub[:, -1] = -1.0
    b_ub = np.zeros(2*n-1)
    
    # Equality: sum(x) = n/2
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n/2.0])
    
    # Bounds: 0 <= x_i <= 1, t >= 0
    bounds = [(0, 1)] * n + [(0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    
    if res.status == 0:
        return res.x[:n]
    return None

def alternating_lp(h_init, n, max_iters=200):
    """Alternating LP method."""
    h = project_feasible(h_init.copy(), n)
    best_h = h.copy()
    best_c5 = evaluate_c5(h)

    return best_h, best_c5

def project_box_sum(y, target_sum):
    """Project y onto {h: 0<=h<=1, sum(h)=target_sum}."""
    def sum_at_lam(lam):
        return np.sum(np.clip(y - lam, 0, 1))
    
    lo, hi = y.min() - 1, y.max() + 1
    for _ in range(80):
        mid = (lo + hi) / 2
        if sum_at_lam(mid) > target_sum:
            lo = mid
        else:
            hi = mid

def compute_grad_lag(h, k, n):
    """Gradient of f_k with respect to h."""
    dx = 2.0 / n
    lag = (n - 1) - k
    grad = np.zeros(n)
    if lag >= 0:
        for i in range(n - lag):
            grad[i] += (1.0 - h[i + lag])
            grad[i + lag] += (-h[i])
    else:
        L = -lag
        for i in range(L, n):
            grad[i] += (1.0 - h[i - L])
            grad[i - L] += (-h[i])
    return dx * grad

    """Build (2n-1) x n matrix for LP.
    A[k, i] = fixed[i + lag_k] if i and i+lag_k are both valid."""
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        lag = (n - 1) - k
        if lag >= 0:
            A[k, :n-lag] = fixed[lag:]
        else:
            L = -lag
            A[k, L:] = fixed[:-L]
    return A

def solve_lp_subproblem(n, fixed):
    """Solve min t s.t. A(fixed) @ x * dx <= t, sum(x)=n/2, 0<=x<=1."""
    dx = 2.0 / n
    A = build_lp_matrix(n, fixed)
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    A_ub = np.hstack([A * dx, -np.ones((2*n-1, 1))])
    b_ub = np.zeros(2 * n - 1)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    bounds = [(0, 1)] * n + [(None, None)]

    """Simple coordinate descent: try adjusting each coordinate."""
    h = project_box_sum(h_init.copy(), n / 2.0)
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    target = n / 2.0

        for i in range(n):
            # Try different values for h[i], adjusting others proportionally
            for delta in [0.02, -0.02, 0.01, -0.01, 0.05, -0.05]:
                new_val = h[i] + delta
                if new_val < 0 or new_val > 1:
                    continue
                h_cand = h.copy()
                h_cand[i] = new_val
                # Adjust others to maintain sum
                diff = target - h_cand.sum()
                others = np.arange(n) != i
                if abs(diff) > 1e-10:

                    slack = np.where(others & (diff > 0), 1 - h_cand, 0) + \

                    if total_slack > 1e-10:

                    break

import numpy as np
from scipy.optimize import linprog
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def project_box_sum(y, target_sum):
    """Project y onto {h: 0<=h<=1, sum(h)=target_sum} via bisection on lambda."""
    y = np.asarray(y, dtype=float)
    def sum_at(lam):
        return np.sum(np.clip(y - lam, 0.0, 1.0))
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if sum_at(mid) > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    return np.clip(y - lam, 0.0, 1.0)

def compute_grad_lag(h, k, n):
    """Gradient of f_k(h) = dx * sum h[i]*(1-h[i+lag]) w.r.t. h."""
    dx = 2.0 / n
    lag = (n - 1) - k
    grad = np.zeros(n)
    if lag >= 0:
        for i in range(n - lag):
            grad[i] += (1.0 - h[i + lag])
            grad[i + lag] += (-h[i])
    else:
        L = -lag
        for i in range(L, n):
            grad[i] += (1.0 - h[i - L])
            grad[i - L] += (-h[i])
    return dx * grad

def projected_subgrad(h, n, n_steps=5000, avg_tol=1e-6):
    """Projected subgradient descent with all-max averaging."""
    target = n / 2.0
    h = project_box_sum(h.copy(), target)
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    for step in range(n_steps):
        corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        max_val = np.max(corr)
        mask = corr >= max_val - avg_tol
        idxs = np.where(mask)[0]
        if len(idxs) == 0:
            idxs = np.array([np.argmax(corr)])
        grad = np.zeros(n)
        for k in idxs:
            grad += compute_grad_lag(h, k, n)
        grad /= len(idxs)
        lr = 0.5 / (1.0 + 0.05 * step) ** 0.5
        h_new = h - lr * grad
        h_new = project_box_sum(h_new, target)
        c5_new = evaluate_c5(h_new)
        h = h_new
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
    return best_h, best_c5

def build_lp_matrix(n, fixed):
    """Build (2n-1) x n matrix: A[k,i] = fixed[i+lag_k] if valid, else 0."""
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        lag = (n - 1) - k
        if lag >= 0:
            A[k, :n - lag] = fixed[lag:]
        else:
            L = -lag
            A[k, L:] = fixed[:-L]
    return A

def solve_lp_subproblem(n, fixed):
    """Solve min t s.t. A@x*dx <= t, sum(x)=n/2, 0<=x<=1."""
    dx = 2.0 / n
    A = build_lp_matrix(n, fixed)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.hstack([A * dx, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    bounds = [(0, 1)] * n + [(None, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 5})
        if res.status == 0:
            return res.x[:n]
    except Exception:
        pass
    return None

def alternating_lp(h_init, n, max_rounds=50):
    """Alternating LP: fix one side, optimize the other via LP."""
    target = n / 2.0
    h = project_box_sum(h_init.copy(), target)
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    for r in range(max_rounds):
        improved = False
        # Primal: fix b=1-h, find h'
        b = 1.0 - h
        h_prime = solve_lp_subproblem(n, b)
        if h_prime is not None:
            h_prime = project_box_sum(h_prime, target)
            c5_p = evaluate_c5(h_prime)
            if c5_p < best_c5 - 1e-14:
                best_h, best_c5 = h_prime.copy(), c5_p
                h = h_prime.copy()
                improved = True
        # Dual: fix h, find b', set h'=1-b'
        b_prime = solve_lp_subproblem(n, h)
        if b_prime is not None:
            h_cand = 1.0 - b_prime
            h_cand = project_box_sum(h_cand, target)
            c5_d = evaluate_c5(h_cand)
            if c5_d < best_c5 - 1e-14:
                best_h, best_c5 = h_cand.copy(), c5_d
                h = h_cand.copy()
                improved = True
        if not improved:
            # Try blending
            for alpha in (0.2, 0.4, 0.5, 0.6, 0.8):
                h_blend = (1.0 - alpha) * h + alpha * best_h
                h_blend = project_box_sum(h_blend, target)
                c5_b = evaluate_c5(h_blend)
                if c5_b < best_c5 - 1e-14:
                    best_h, best_c5 = h_blend.copy(), c5_b
                    h = h_blend.copy()
                    improved = True
            if not improved:
                break
    return best_h, best_c5

def coordinate_descent(h_init, n, n_passes=50):
    """Coordinate descent: adjust one coordinate at a time."""
    target = n / 2.0
    h = project_box_sum(h_init.copy(), target)
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    deltas = [0.01, -0.01, 0.02, -0.02, 0.05, -0.05, 0.1, -0.1]
    for p in range(n_passes):
        any_change = False
        for i in range(n):
            for d in deltas:
                nv = h[i] + d
                if nv < -0.01 or nv > 1.01:
                    continue
                h_cand = h.copy()
                h_cand[i] = nv
                h_cand = project_box_sum(h_cand, target)
                c5_cand = evaluate_c5(h_cand)
                if c5_cand < best_c5:
                    best_c5 = c5_cand
                    best_h = h_cand.copy()
                    h = h_cand.copy()
                    any_change = True
                    break
        if not any_change and p > 5:
            break
    return best_h, best_c5

def interpolate_h(h_src, n_src, n_dst):
    """Interpolate h from n_src to n_dst points."""
    x_src = (np.arange(n_src) + 0.5) * 2.0 / n_src
    x_dst = (np.arange(n_dst) + 0.5) * 2.0 / n_dst
    return np.interp(x_dst, x_src, h_src)

def binary_round(h, n):
    """Try to round to binary while maintaining sum=n/2. Return best binary or None."""
    target = n // 2
    if abs(n / 2.0 - target) > 1e-10:
        return None
    # Sort by h value, pick top n/2
    order = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[order[:target]] = 1.0
    return h_bin

def random_perturb(h, n, sigma, rng):
    """Add random perturbation and project."""
    target = n / 2.0
    h_new = h + sigma * rng.standard_normal(n)
    return project_box_sum(h_new, target)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t_start = time.time()

    def time_left():
        return budget_s - (time.time() - t_start)

    # Get initial h
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = np.full(70, 0.5)

    n0 = len(h0)
    h0 = project_box_sum(h0, n0 / 2.0)

    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    best_n = n0

    # Phase 1: Subgradient on initial
    if time_left() > 60:
        h1, c1 = projected_subgrad(h0, n0, n_steps=5000)
        if c1 < best_c5:
            best_h, best_c5, best_n = h1.copy(), c1, n0

    # Phase 2: Alternating LP
    if time_left() > 30:
        h2, c2 = alternating_lp(best_h, best_n, max_rounds=60)
        if c2 < best_c5:
            best_h, best_c5 = h2.copy(), c2

    # Phase 3: More subgradient from improved point
    if time_left() > 60:
        h3, c3 = projected_subgrad(best_h, best_n, n_steps=8000)
        if c3 < best_c5:
            best_h, best_c5 = h3.copy(), c3

    # Phase 4: Coordinate descent polish
    if time_left() > 90:
        h4, c4 = coordinate_descent(best_h, best_n, n_passes=60)
        if c4 < best_c5:
            best_h, best_c5 = h4.copy(), c4

    # Phase 5: Try different n values
    n_list = [50, 60, 70, 80, 90, 100, 120]
    for n_try in n_list:
        if time_left() < 60:
            break
        if n_try == best_n:
            continue
        h_try = interpolate_h(best_h, best_n, n_try)
        h_try = project_box_sum(h_try, n_try / 2.0)
        c5_try = evaluate_c5(h_try)
        # Subtract subgradient
        if time_left() > 50:
            h_try, c5_try = projected_subgrad(h_try, n_try, n_steps=2000)
        # LP
        if time_left() > 30:
            h_try, c5_try = alternating_lp(h_try, n_try, max_rounds=15)
        # Coordinate polish
        if time_left() > 30:
            h_try, c5_try = coordinate_descent(h_try, n_try, n_passes=20)
        if c5_try < best_c5:
            best_h, best_c5, best_n = h_try.copy(), c5_try, n_try

    # Phase 6: Random restarts from current best
    for attempt in range(10):
        if time_left() < 45:
            break
        sigma = 0.05 * (1.0 - attempt * 0.05)
        h_r = random_perturb(best_h, best_n, sigma, rng)
        c5_r = evaluate_c5(h_r)
        if c5_r < best_c5:
            best_h, best_c5 = h_r.copy(), c5_r
        # Quick subgradient from perturbed point
        if time_left() > 30:
            h_r2, c5_r2 = projected_subgrad(h_r, best_n, n_steps=2000)
            if c5_r2 < best_c5:
                best_h, best_c5 = h_r2.copy(), c5_r2
        # Quick LP
        if time_left() > 20:
            h_r3, c5_r3 = alternating_lp(best_h, best_n, max_rounds=10)
            if c5_r3 < best_c5:
                best_h, best_c5 = h_r3.copy(), c5_r3

    # Phase 7: Final coordinate descent
    if time_left() > 60:
        h_f, c_f = coordinate_descent(best_h, best_n, n_passes=80)
        if c_f < best_c5:
            best_h, best_c5 = h_f.copy(), c_f

    # Phase 8: Final subgradient with tight tolerance
    if time_left() > 30:
        h_sf, c_sf = projected_subgrad(best_h, best_n, n_steps=5000, avg_tol=1e-4)
        if c_sf < best_c5:
            best_h, best_c5 = h_sf.copy(), c_sf

    # Final feasibility fix
    best_h = project_box_sum(best_h, best_n / 2.0)
    target = best_n / 2.0
    diff = target - best_h.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            idx = np.argmax(1.0 - best_h)
            best_h = best_h.copy()
            best_h[idx] = min(1.0, best_h[idx] + diff)
        else:
            idx = np.argmax(best_h)
            best_h = best_h.copy()
            best_h[idx] = max(0.0, best_h[idx] + diff)

    best_c5 = evaluate_c5(best_h)
    return best_h, best_c5, best_n
