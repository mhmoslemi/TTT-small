```python
import numpy as np
from scipy.optimize import linprog
import time

def project_to_sum(h, target):
    """Project h to {x: 0<=x<=1, sum(x)=target}."""
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(50):
        residual = target - np.sum(h)
        if abs(residual) < 1e-14:
            break
        if residual > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if mask.sum() == 0:
            # Spread equally
            h[:] = residual / len(h)
            h = np.clip(h, 0, 1)
            continue
        adj = residual / mask.sum()
        h[mask] += adj
        h = np.clip(h, 0, 1)
    return h

def evaluate(h):
    """Compute C5 = max(correlate(h, 1-h, 'full') * dx)."""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return np.max(c) * dx

def get_correlation(h):
    """Get the full correlation array."""
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full')

def compute_gradients(h, n):
    """Compute gradients of all correlation values w.r.t. h.
    Returns array of shape (2n-1, n) where row p is gradient of corr[p]."""
    grads = np.zeros((2*n-1, n))
    h = np.asarray(h, dtype=np.float64)
    for p in range(2*n-1):
        s = p - (n-1)
        if s == 0:
            grads[p, :] = 1.0 - 2.0 * h
        elif s > 0:
            grads[p, :n-s] += (1.0 - h[s:])
            grads[p, s:] -= h[:n-s]
        else:
            sp = -s
            grads[p, sp:] += (1.0 - h[:n-sp])
            grads[p, :n-sp] -= h[sp:]
    return grads

def lp_local_step(h, corr, grads, n, active_lags, move_cost=0.0):
    """Solve LP to find a descent direction.
    Returns: (direction, predicted_improvement) or (None, 0) if no improvement."""
    c0 = np.max(corr)  # current max correlation (unscaled)
    
    # Active lags: those with corr close to max
    # active_lags: list of indices into corr array
    
    if len(active_lags) == 0:
        return None, 0.0
    
    m = len(active_lags)
    # LP variables: u[0..n-1], v[0..n-1], t
    # Total: 2n+1 variables
    
    # c = [0,...,0, move_cost*n (for t)]  → actually minimize t + move_cost * (sum(u) + sum(v))
    c_obj = np.zeros(2*n + 1)
    c_obj[:n] = move_cost
    c_obj[n:2*n] = move_cost
    # t coefficient is 0 (we want to minimize t directly)
    # Actually, objective: minimize t + move_cost * sum(u) + move_cost * sum(v)
    # c = [move_cost]*n + [move_cost]*n + [1]
    c_obj = np.zeros(2*n + 1)
    c_obj[:n] = move_cost
    c_obj[n:2*n] = move_cost
    c_obj[2*n] = 1.0  # coefficient of t
    
    # Inequality constraints: A_ub @ x ≤ b_ub
    # For each active lag p: grads[p] @ (u-v) - t ≤ -corr[p]
    # → [grads[p][0..n-1], -grads[p][0..n-1], -1] · [u, v, t] ≤ -corr[p]
    
    A_ub = np.zeros((m, 2*n + 1))
    b_ub = np.zeros(m)
    for i, p in enumerate(active_lags):
        A_ub[i, :n] = grads[p, :]
        A_ub[i, n:2*n] = -grads[p, :]
        A_ub[i, 2*n] = -1.0
        b_ub[i] = -corr[p]
    
    # Bounds: u[i] ∈ [0, 1-h[i]], v[i] ∈ [0, h[i]], t ∈ (-inf, inf) → [0, inf)
    bounds = []
    for i in range(n):
        bounds.append((0, 1.0 - h[i]))
    for i in range(n):
        bounds.append((0, h[i]))
    bounds.append((0, None))  # t ≥ 0
    
    # Equality: sum(u) - sum(v) = 0
    A_eq = np.zeros((1, 2*n + 1))
    A_eq[0, :n] = 1.0
    A_eq[0, n:2*n] = -1.0
    b_eq = np.zeros(1)
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                      bounds=bounds, method='highs')
    except Exception:
        return None, 0.0
    
    if res.status != 0:
        return None, 0.0
    
    u = res.x[:n]
    v = res.x[n:2*n]
    t_pred = res.x[2*n]
    
    # Check if improvement
    if t_pred >= c0 - 1e-10:
        return None, 0.0
    
    direction = u - v
    return direction, c0 - t_pred

def continuous_optimize(h, n, deadline, move_cost=0.001):
    """Main continuous optimization loop using LP steps."""
    h = h.copy().astype(np.float64)
    target = n / 2.0
    h = project_to_sum(h, target)
    
    best_h = h.copy()
    best_corr = get_correlation(h)
    best_max = np.max(best_corr)
    
    no_improve_count = 0
    
    while time.time() < deadline - 5:
        corr = np.correlate(h, 1.0 - h, mode='full')
        current_max = np.max(corr)
        
        if current_max < best_max - 1e-12:
            best_max = current_max
            best_h = h.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
            
        if no_improve_count > 50:
            break
        
        # Find active lags
        threshold = current_max - max(0.5, 0.01 * current_max)
        active_lags = np.where(corr >= threshold)[0].tolist()
        # Also include the max lag
        max_lag = np.argmax(corr)
        if max_lag not in active_lags:
            active_lags.append(max_lag)
        
        # Compute gradients for active lags
        n_active = len(active_lags)
        grads = np.zeros((n_active, n))
        for i, p in enumerate(active_lags):
            s = p - (n-1)
            if s == 0:
                grads[i, :] = 1.0 - 2.0 * h
            elif s > 0:
                grads[i, :n-s] += (1.0 - h[s:])
                grads[i, s:] -= h[:n-s]
            else:
                sp = -s
                grads[i, sp:] += (1.0 - h[:n-sp])
                grads[i, :n-sp] -= h[sp:]
        
        # Solve LP
        direction, improvement = lp_local_step(h, corr, grads, n, range(n_active), move_cost)
        
        if direction is None:
            # Try with fewer active lags (just the top 3)
            top_lags = np.argsort(corr)[-5:]
            grads_top = np.zeros((len(top
