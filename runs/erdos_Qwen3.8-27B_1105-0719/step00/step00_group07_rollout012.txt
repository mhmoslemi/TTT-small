import time

def compute_R(h):
    """Compute the correlation vector R."""
    return np.correlate(h, 1-h, mode='full')

def compute_c5(h):
    """Compute C5 for a given h."""
    n = len(h)
    dx = 2.0 / n
    R = np.correlate(h, 1-h, mode='full')
    return np.max(R) * dx

def project_box_sum(y, target):
    """Project y onto {0 <= x_i <= 1, sum(x_i) = target}."""
    n = len(y)
    target = np.clip(target, 0, n)
    
    def phi(lam):
        return np.sum(np.clip(y - lam, 0, 1)) - target
    
    # Binary search for lambda
    lo = np.min(y) - 1
    hi = np.max(y) + 1
    for _ in range(100):
        mid = (lo + hi) / 2
        if phi(mid) > 0:
            lo = mid
        else:
            hi = mid
    
    x = np.clip(y - (lo + hi) / 2, 0, 1)
    
    # Fix residual
    residual = target - np.sum(x)
    if residual > 1e-12:
        # Add to entries with slack
        slack = 1 - x
        order = np.argsort(-slack)
        for idx in order:
            add = min(residual, slack[idx])
            x[idx] += add
            residual -= add
            if residual <= 1e-12:
                break
    elif residual < -1e-12:
        # Subtract from entries with mass
        mass = x.copy()
        order = np.argsort(-mass)
        for idx in order:
            sub = min(-residual, mass[idx])
            x[idx] -= sub
            residual += sub
            if residual >= -1e-12:
                break
    
    return x

    start_time = time.time()

        h = initial.copy()
    else:

        positions = rng.permutation(n)
        h[positions[:n//2]] = 1
    
    R = compute_R(h)
    best_R = np.max(R)
    best_h = h.copy()
    
    T = 1.0  # Initial temperature
    cooling = 0.9999
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    step = 0
    while time.time() - start_time < budget_s:
        # Pick random i from ones, j from zeros
        idx_i = rng.integers(len(ones))
        idx_j = rng.integers(len(zeros))
        i = ones[idx_i]
        j = zeros[idx_j]
        
        if i == j:
            continue
        
        # Compute delta for each k
        old_max = best_R  # or np.max(R) for current state
        current_max = np.max(R)
        
        # Apply swap
        h[i] = 0
        h[j] = 1
        
        # Update R
        for k in range(-(n-1), n):
            delta = 0
            q = i - k
            if 0 <= q < n:
                delta -= (1 - h[q])  # h[i] is now 0
            p = i + k
            if 0 <= p < n:
                delta += h[p]  # (1-h[i]) is now 1
            q = j - k
            if 0 <= q < n:
                delta += (1 - h[q])  # h[j] is now 1
            p = j + k
            if 0 <= p < n:
                delta -= h[p]  # (1-h[j]) is now 0
            if k == i - j:
                delta += 1
            if k == j - i:
                delta += 1
            R[k + (n-1)] += delta
        
        new_max = np.max(R)
        delta_cost = new_max - current_max
        
        if delta_cost <= 0 or (T > 0.01 and rng.random() < np.exp(-delta_cost / T)):
            # Accept
            ones[idx_i] = j  # i is no longer 1, j now is
            zeros[idx_j] = i  # j is no longer 0, i now is
            # Update best
            if new_max < best_R:
                best_R = new_max
                best_h = h.copy()
        else:

            h[i] = 1
            h[j] = 0
            for k in range(-(n-1), n):

                q = i - k
                if 0 <= q < n:

                q = j - k

                p = j + k

                    delta -= h[p]
                if k == i - j:
                    delta += 1
                if k == j - i:
                    delta += 1
                R[k + (n-1)] -= delta

        step += 1

    R = np.correlate(h, 1-h, mode='full')
    m = np.max(R)
    shifted = beta * (R - m)
    # Subtract max for numerical stability
    max_shifted = np.max(shifted)
    exp_shifted = np.exp(shifted - max_shifted)
    w = exp_shifted / np.sum(exp_shifted)
    
    grad = np.zeros(n)
    for p in range(n):
        # Valid lags for term 1: 0 ≤ p-lag < n → p-n+1 ≤ lag ≤ p
        lag1_min = max(-(n-1), p - (n-1))
        lag1_max = min(n-1, p)
