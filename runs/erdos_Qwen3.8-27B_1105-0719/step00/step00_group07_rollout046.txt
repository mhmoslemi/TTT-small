import numpy as np
import time

def compute_c5_full(h):
    """Compute C5 and the full correlation vector."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx, corr

def lse_max(corr, beta):
    """Log-sum-exp smoothed maximum."""
    m = corr.max()
    shifted = beta * (corr - m)
    # Numerical stability
    max_shifted = shifted.max()
    return m + (max_shifted + np.log(np.sum(np.exp(shifted - max_shifted)))) / beta

def project_box_sum(y, target):
    """Project y onto {x: 0<=x_i<=1, sum(x)=target}."""
    n = len(y)
    lo = y.min() - 1
    hi = y.max() + 1
    for _ in range(80):
        mid = (lo + hi) / 2
        x = np.clip(y - mid, 0, 1)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) / 2, 0, 1)
    # Fix residual
    r = target - x.sum()
    if abs(r) > 1e-12:
        if r > 0:
            # Add to entries with most slack
            slack = 1.0 - x
            order = np.argsort(-slack)
            for i in order:
                if r <= 1e-12:
                    break
                add = min(r, slack[i])
                x[i] += add
                r -= add
        else:
            # Remove from entries with smallest value
            order = np.argsort(x)
            for i in order:
                if r >= -1e-12:
                    break
                sub = min(-r, x[i])
                x[i] -= sub
                r += sub
    return x

        # = (1-h[p-l]) if p >= l (p is first index) + h[p+l] if p+l < n (p is in (1-h[k]) as k=p, contributing h[k+l]=h[p+l])

            if val < best_val:
                best_val = val
                best = h_new.copy()
                h = h_new
                improved = True
                no_improve = 0
                break
            lr *= 0.5
        if not improved:
            no_improve += 1
            if no_improve > 200:
                # Perturb
                h = project_box_sum(h + rng.randn(n) * 0.03, target)
                no_improve = 0
        
        it += 1
    return best, best_val

def binary_anneal(n, t_start, time_limit, rng=None, init=None):
    if rng is None:
        rng = np.random.RandomState(7)
    s = n // 2
    
    if init is not None and len(init) == n:
        h = (init > 0.5).astype(float).copy()
    else:
        h = None
        try:
            hv = np.array(initial_h_values, dtype=float)
            h_int = np.interp(np.linspace(0, n-1, n), np.linspace(0, len(hv)-1, len(hv)), hv)
            h = (h_int > 0.5).astype(float)
        except:
            pass
        if h is None:
            h = np.zeros(n)
            h[rng.choice(n, s, replace=False)] = 1
    
    # Fix sum
    ones = (h == 1).sum()
    if ones > s:
        idx = np.where(h == 1)[0]
        h[idx[rng.choice(len(idx), ones-s, replace=False)]] = 0
    elif ones < s:
        idx = np.where(h == 0)[0]
        h[idx[rng.choice(len(idx), s-ones, replace=False)]] = 1
    
    corr = np.correlate(h, 1.0 - h, mode='full')
    cur_max = corr.max()
    best_h = h.copy()
    best_max = cur_max
    
    T = 3.0
    T_min = 0.003
    cool_rate = (T_min / T) ** (1.0 / 80000)
    
    while time.time() - t_start < time_limit:
        # Random swap
        pos_1 = np.where(h == 1)[0]
        pos_0 = np.where(h == 0)[0]
        i = pos_1[rng.randint(len(pos_1))]
        j = pos_0[rng.randint(len(pos_0))]
        
        h[i] = 0.0
        h[j] = 1.0
        new_corr = np.correlate(h, 1.0 - h, mode='full')
        new_max = new_corr.max()
        
        accept = new_max < cur_max or (T > T_min and rng.random() < np.exp(-(new_max - cur_max) / T))
        if accept:
            corr = new_corr
            cur_max = new_max
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        
        T = max(T * cool_rate, T_min)
    
    return best_h, best_max * (2.0 / n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    
    # Get starting point

    t1 = t0 + budget_s * 0.35
    h1, c5_1 = local_refine(best_h, t_start=t0, time_limit=t1 - t0, rng=rng)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1
    
    # Phase 2: Multi-start (15% of budget)
    n_starts = 4
    t2 = t0 + budget_s * 0.50
    for k in range(n_starts):
        if time.time() > t2:
            break
        h_pert = best_h + rng.randn(len(best_h)) * 0.04
        h_pert = project_box_sum(h_pert, len(best_h) * 0.5)
        h_try, c5_try = local_refine(h_pert, t_start=time.time(), time_limit=min(20, t2 - time.time()), rng=rng)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try
    
    # Phase 3: Binary annealing (30% of budget)
    t3 = t0 + budget_s * 0.80
    for n_b in [96, 100, 104, 108, 112, 120]:
        if time.time() > t3:
            break
        alloc = min(40, t3 - time.time() - 5)
        if alloc < 15:
            break
        h_init_b = np.interp(np.linspace(0, n_b-1, n_b),
                           np.linspace(0, len(best_h)-1, len(best_h)), best_h)
        h_b, c5_b = binary_anneal(n_b, t_start=time.time(), time_limit=alloc, rng=rng, init=h_init_b)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b
    
    # Phase 4: Final refinement (15% of budget)
    if time.time() < t0 + budget_s - 10:
        h_f = best_h + rng.randn(len(best_h)) * 0.02
        h_f = project_box_sum(h_f, len(best_h) * 0.5)
        h_final, c5_final = local_refine(h_f, t_start=time.time(),
                                         time_limit=budget_s - (time.time() - t0) - 5, rng=rng)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final
    
    # Final: ensure feasibility
    best_h = np.clip(best_h, 0.0, 1.0)
    n = len(best_h)
    best_h = project_box_sum(best_h, n * 0.5)
    best_c5, _ = compute_c5_full(best_h)
    
    return best_h, best_c5, n
