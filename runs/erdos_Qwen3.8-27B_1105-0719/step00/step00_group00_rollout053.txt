    """Compute LSE objective and its gradient."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    
    # Numerically stable softmax
    max_c = np.max(corr)
    shifted = beta * (corr - max_c)
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    weights = exp_shifted / np.sum(exp_shifted)
    
    # Objective value
    obj = (max_c + np.log(np.sum(exp_shifted))) / beta

            second = ii

            grad[second] -= w * h[first]

**Method: Projected gradient / FISTA**

1. Define objective f(h) = LSE(h, β) (or the exact max)
2. Use projected gradient descent: h_{k+1} = Proj_C(h_k - α * ∇f(h_k))

This is simple, doesn't require scipy.optimize, and handles the constraints exactly.

The projection is: given y, find h = clip(y - t, 0, 1) such that sum(h) = T. We solve for t by bisection.

```python
def projected_gradient_step(h, grad, alpha, T):
    """One step of projected gradient descent."""
    y = h - alpha * grad
    return project_to_feasible(y, T)
```

The step size α can be determined by a line search or by using the Lipschitz constant of the gradient.

For the LSE objective, the gradient Lipschitz constant is bounded by β * L where L is related to the Hessian of the correlations. Since each c_p is bilinear in h, the Hessian of c_p is bounded, and the LSE Hessian is bounded by β * (max weight) * (max Hessian of c_p). In practice, we can use a backtracking line search.

        h_best = np.full(45, 0.5)

    n_best = len(h_best)
    T_best = n_best / 2.0
    h_best = project_to_feasible(h_best, T_best)
    score_best = evaluate_c5(h_best)

    # Phase 1: initial pair transfer
    h_best, score_best = pair_transfer_optimize(h_best, time_limit=45)

    # Phase 2: smooth minimax refinement
    h_best, score_best = refine_solution(h_best, n_best, T_best, time_limit=120)

    # Phase 3: try different resolutions
    for n_try in (30, 60, 36, 50):
        if time.time() - t_start > budget_s - 150:
            break
        h_try = resample_h(h_best, n_try)
        T_try = n_try / 2.0
        h_try, s_try = refine_solution(h_try, n_try, T_try, time_limit=60)
        if s_try < score_best:
            h_best = h_try
            score_best = s_try
            n_best = n_try
            T_best = T_try

    # Go back to original n if better there
    if n_best != 45:
        h_back = resample_h(h_best, 45)
        T_back = 22.5
        h_back, s_back = refine_solution(h_back, 45, T_back, time_limit=40)
        if s_back < score_best:
            h_best = h_back
            score_best = s_back
            n_best = 45
            T_best = T_back

    # Phase 4: multi-start basin hopping on best resolution
    while time.time() - t_start < budget_s - 40:
        scale = 0.12
        pert = np.random.randn(n_best) * scale
        h_pert = project_to_feasible(h_best + pert, T_best)
        h_pert, s_pert = pair_transfer_optimize(h_pert, time_limit=10)
        if s_pert < score_best:
            h_best = h_pert
            score_best = s_pert
            h_best, score_best = refine_solution(h_best, n_best, T_best, time_limit=40)
        else:
            # Try smaller perturbation
            scale = 0.04
            pert = np.random.randn(n_best) * scale
            h_pert = project_to_feasible(h_best + pert, T_best)
            h_pert, s_pert = pair_transfer_optimize(h_pert, time_limit=8)
            if s_pert < score_best:
                h_best = h_pert
                score_best = s_pert

    # Final polish
    h_best, score_best = pair_transfer_optimize(h_best, time_limit=20)
    if time.time() - t_start < budget_s - 10:
        h_best, score_best = refine_solution(h_best, n_best, T_best, time_limit=20)

    # Verify feasibility
    h_best = project_to_feasible(h_best, T_best)
    score_best = evaluate_c5(h_best)

    return h_best, score_best, n_best
