import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

_GLOBAL_BEST_H = None
_GLOBAL_BEST_C = np.inf
_GLOBAL_DEADLINE = 0.0
_GLOBAL_N = 42

def exact_C(h):
    n = len(h)
    if n == 0:
        return 1e6
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def validate_h(h):
    if len(h) == 0:
        return False
    if np.any(h < -1e-10) or np.any(h > 1.0 + 1e-10):
        return False
    n = len(h)
    if abs(float(np.sum(h)) - n / 2.0) > 1e-8:
        return False
    return True

def project_feasible(u):
    n = len(u)
    u = np.array(u, dtype=float).copy()
    target = n / 2.0
    lo = -float(np.max(u)) - 3.0
    hi = 1.0 - float(np.min(u)) + 3.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(u + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    result = np.clip(u + 0.5 * (lo + hi), 0.0, 1.0)
    residual = target - np.sum(result)
    if abs(residual) > 1e-12:
        if residual > 0:
            idx = np.where(result < 0.999999)[0]
        else:
            idx = np.where(result > 1e-6)[0]
        if len(idx) > 0:
            per = residual / len(idx)
            result[idx] = np.clip(result[idx] + per, 0.0, 1.0)
        if abs(target - np.sum(result)) > 1e-12 and len(result) > 1:
            result[0] = target - np.sum(result[1:])
            result[0] = np.clip(result[0], 0.0, 1.0)
    return result

def stable_sigmoid(t):
    out = np.empty_like(t, dtype=float)
    pos = t >= 0
    out[pos] = 1.0 / (1.0 + np.exp(-t[pos]))
    et = np.exp(t[~pos])
    out[~pos] = et / (1.0 + et)
    return out

def logit_transform(h):
    eps = 1e-9
    h_c = np.clip(h, eps, 1.0 - eps)
    return np.log(h_c / (1.0 - h_c))

def from_logit(y):
    n = len(y)
    target = n / 2.0
    lo = -float(np.max(y)) - 50.0
    hi = -float(np.min(y)) + 50.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(stable_sigmoid(y + mid))
        if s < target:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    h = stable_sigmoid(y + lam)
    residual = target - np.sum(h)
    if abs(residual) > 1e-10:
        h = project_feasible(h)
    return h

def compute_smoothed_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_c = float(np.max(corr))
    exp_vals = np.exp(beta * (corr - max_c))
    total = float(np.sum(exp_vals))
    if total < 1e-300:
        weights = np.zeros_like(corr)
        weights[np.argmax(corr)] = 1.0
    else:
        weights = exp_vals / total
    F = (max_c + np.log(total) / beta) * dx
    G = np.zeros(n)
    nm1 = n - 1
    for k in range(len(corr)):
        w = weights[k]
        if w < 1e-14:
            continue
        m = k - nm1
        if m >= 0:
            cnt = n - m
            if cnt > 0:
                G[:cnt] += dx * w * (1.0 - h[m:m + cnt])
                G[m:m + cnt] -= dx * w * h[:cnt]
        else:
            mp = -m
            cnt = n - mp
            if cnt > 0:
                G[mp:mp + cnt] += dx * w * (1.0 - h[:cnt])
                G[:cnt] -= dx * w * h[mp:mp + cnt]
    return F, G

def smooth_obj_and_grad_y(y, beta):
    h = from_logit(y)
    n = len(h)
    F, G = compute_smoothed_grad(h, beta)
    if not np.isfinite(F):
        return 1e8, np.zeros(n)
    s = h * (1.0 - h)
    S = float(np.sum(s))
    if S < 1e-15:
        return F, np.zeros(n)
    B = float(np.dot(G, s)) / S
    grad = s * (G - B)
    if not np.all(np.isfinite(grad)):
        grad = np.zeros(n)
    return F, grad

            raise TimeoutError()
        h_chk = from_logit(y_in)

            _GLOBAL_BEST_C = c_chk
            _GLOBAL_BEST_H = h_chk.copy()
        return F, grad

    try:
        res = minimize(obj_wrapper, y, method='BFGS', jac=True,
                       options={'maxiter': max_iter, 'gtol': 5e-6, 'disp': False})
        if hasattr(res, 'x') and res.x is not None and np.all(np.isfinite(res.x)):

    except (TimeoutError, Exception):

    return _GLOBAL_BEST_H.copy() if _GLOBAL_BEST_H is not None else h_start
