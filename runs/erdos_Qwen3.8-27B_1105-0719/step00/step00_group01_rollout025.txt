```python
def _project(h, n):
    """Project h to satisfy sum = n/2 and 0 <= h <= 1."""
    h = np.clip(h, 0, 1)
    target = n / 2
    # Iterative projection
    for _ in range(1000):
        excess = h.sum() - target
        if abs(excess) < 1e-12:
            break
        if excess > 0:
            mask = h > 0
            if not mask.any():
                break
            h[mask] -= excess / mask.sum()
        else:
            mask = h < 1
            if not mask.any():
                break
            h[mask] += (-excess) / mask.sum()
    h = np.clip(h, 0, 1)
    return h
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def _project_feasible(h, n):
    """Project h to the feasible set {0<=h<=1, sum(h)=n/2}."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(10000):
        excess = h.sum() - target
        if abs(excess) < 1e-14:
            break
        if excess > 0:
            mask = h > 1e-15
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] -= excess / cnt
            h = np.clip(h, 0.0, 1.0)
        else:
            mask = h < 1.0 - 1e-15
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] += (-excess) / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def _eval_c5(h):
    """Evaluate C5 for a given h."""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return np.max(c)

def _smooth_max_obj(x, dx, beta):
    """Smooth max of correlation values."""
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(c)
    return m + np.log(np.sum(np.exp(beta * (c - m)))) / beta

def _smooth_max_grad(x, dx, beta):
    """Gradient of smooth max."""
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(c)
    shifted = beta * (c - m)
    exps = np.exp(shifted)
    w = exps / exps.sum()
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        L = max(0, k - (n - 1))
        U = min(n - 1, 2 * n - 2 - k)
        wk = w[k]
        for m_idx in range(n):
            i1 = m_idx - k + n - 1
            if L <= i1 <= U and 0 <= i1 < n:
                grad[m_idx] += wk * (1.0 - x[i1])
            if L <= m_idx <= U:
                i2 = m_idx + k - n + 1
                if 0 <= i2 < n:
                    grad[m_idx] += wk * (-x[i2])
    grad *= dx
    return m + np.log(exps.sum()) / beta, grad

def _optimize_slsqp_smooth(h0, n, dx, beta, maxiter=300):
    """SLSQP optimization of smooth max."""
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        return _smooth_max_obj(x, dx, beta)
    
    def jac(x):
        _, g = _smooth_max_grad(x, dx, beta)
        return g
    
    def eq_con(x):
        return x.sum() - n / 2.0
    
    cons = [{'type': 'eq', 'fun': eq_con}]
    
    res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                   constraints=cons, jac=jac, maxiter=maxiter,
                   options={'ftol': 1e-14})
    return res

def _optimize_slsqp_epigraph(h0, n, dx, maxiter=300):
    """SLSQP epigraph formulation."""
    n_corr = 2 * n - 1
    
    def obj(x):
        return x[-1]
    
    def jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    
    def ineq(x):
        h = x[:-1]
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        return x[-1] - c
    
    def eq_con(x):
        return x[:-1].sum() - n / 2.0
    
    cons = [
        {'type': 'eq', 'fun': eq_con},
        {'type': 'ineq', 'fun': ineq}
    ]
    
    c0 = np.max(np.correlate(h0, 1.0 - h0, mode='full') * dx)
    x0 = np.concatenate([h0, [c0 + 0.01]])
    
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    
    res = minimize(obj, x0, method='SLSQP', bounds=bounds,
                   constraints=cons, jac=jac, maxiter=maxiter,
                   options={'ftol': 1e-14})
    return res

def _random_feasible(n, rng):
```
