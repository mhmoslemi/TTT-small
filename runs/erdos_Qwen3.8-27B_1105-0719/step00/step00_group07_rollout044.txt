            ps, pe = -lag, n - 1
        grad[ps:pe+1] += wk * (1.0 - h[ps+lag:pe+1+lag])
        # Second term: -wk*h[p-lag] for p where p-lag in [0,n-1]
        if lag >= 0:
            ps2, pe2 = lag, n - 1
        else:
            ps2, pe2 = 0, n - 1 + lag
        grad[ps2:pe2+1] -= wk * h[ps2-lag:pe2+1-lag]
    return grad

def optimize_continuous(h0, target, time_limit, seed=42):
    rng = np.random.RandomState(seed)
    n = len(h0)
    start = time.time()
    best = h0.copy()
    best_c5 = compute_c5(best)
    
    # Phase 1: LSE with increasing beta
    for beta in [20, 50, 100, 200, 400, 800]:
        if time.time() - start > time_limit:
            break
        h = best.copy()
        step = 0.05
        for it in range(300):
            if time.time() - start > time_limit:
                break
            g = lse_grad(h, beta)
            obj = lse_value(np.correlate(h, 1-h, mode='full'), beta)
            accepted = False
            for _ in range(50):
                h_new = project_box(h - step * g, target)
                obj_new = lse_value(np.correlate(h_new, 1-h_new, mode='full'), beta)
                if obj_new <= obj * (1 + 1e-10):
                    h = h_new
                    obj = obj_new
                    accepted = True
                    break
                step *= 0.5
            if not accepted:
                step *= 0.5
                if step < 1e-8:
                    break
            else:
                step = min(step * 1.1, 0.5)
        
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best = h.copy()
    
    # Phase 2: Subgradient polishing on the max
    for it in range(500):
        if time.time() - start > time_limit:
            break
        r = np.correlate(best, 1-best, mode='full')
        mx = r.max()
        # Active set: lags where r is close to max
        active = r >= mx - 0.01
        if not active.any():
            active = np.zeros_like(r, dtype=bool)
            active[np.argmax(r)] = True
        # Average subgradient over active lags
        n_lags = np.sum(active)
        g = np.zeros(n)
        for k in np.where(active)[0]:
            lag = k - (n - 1)
            wk = 1.0 / n_lags
            if lag >= 0:
                ps, pe = 0, n - 1 - lag
            else:
                ps, pe = -lag, n - 1
            g[ps:pe+1] += wk * (1.0 - best[ps+lag:pe+1+lag])
            if lag >= 0:
                ps2, pe2 = lag, n - 1
            else:
                ps2, pe2 = 0, n - 1 + lag
            g[ps2:pe2+1] -= wk * best[ps2-lag:pe2+1-lag]
        
        # Step
        best = project_box(best - 0.01 * g, target)
    
    c5 = compute_c5(best)
    if c5 < best_c5:
        best_c5 = c5
    
    return best, best_c5

def compute_c5(h):
    dx = 2.0 / len(h)
    r = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(r) * dx)

def project_box(h, target):
    n = len(h)
    target = max(0.0, min(float(n), target))
    lo, hi = -2.0, 3.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    x = np.clip(h - lam, 0.0, 1.0)
    resid = target - x.sum()
    if resid > 1e-12:
        idx = np.argsort(1.0 - x)
        for i in idx:
            add = min(resid, 1.0 - x[i])
            x[i] += add
            resid -= add
            if resid < 1e-14:
                break
    elif resid < -1e-12:
        idx = np.argsort(x)
        for i in idx:
            sub = min(-resid, x[i])
            x[i] -= sub
            resid += sub
            if resid > -1e-14:
                break
    return x

def lse_value(r, beta):

    s = np.clip(beta * (r - m), -500.0, 500.0)
