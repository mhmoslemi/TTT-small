        if l == 0:
            continue
        # Contribution from i (1->0): -(1-h[i-l] if valid) + h[i+l] if valid
        d_i = 0.0
        if 0 <= i - l < n:
            d_i += (1.0 - h[i - l])
        if 0 <= i + l < n:
            d_i -= h[i + l]
        d_i = -d_i  # because a' - a = -1
        
        # Contribution from j (0->1): +(1-h[j-l] if valid) - h[j+l] if valid
        d_j = 0.0
        if 0 <= j - l < n:
            d_j += (1.0 - h[j - l])
        if 0 <= j + l < n:
            d_j -= h[j + l]
        # a' - a = +1, so delta = +d_j
        
        delta[l + n - 1] = d_i + d_j
    return delta

def sa_binary(h, rng, time_limit=60.0, target=None):
    """Simulated annealing for binary sequence with fixed number of ones."""
    n = len(h)
    if target is None:
        target = n // 2
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    R = compute_R(h)
    best_maxR = np.max(R)
    best_h = h.copy()
    cur_maxR = best_maxR
    start = time.time()
    T = 2.0
    
    while time.time() - start < time_limit:
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        if i == j:
            continue
        delta = binary_swap_delta(h, i, j, R)
        # New max: we need to update R and find new max
        # For efficiency, compute the new max approximately
        R_new = R + delta
        new_maxR = np.max(R_new)
        
        if new_maxR <= cur_maxR or rng.random() < np.exp(-(new_maxR - cur_maxR) / T):
            # Accept
            h[i] = 0
            h[j] = 1
            R = R_new
            cur_maxR = new_maxR
            # Update ones and zeros
            idx_i_in_ones = np.where(ones == i)[0][0]
            ones[idx_i_in_ones] = 0
            ones = ones[ones != 0]
            idx_j_in_zeros = np.where(zeros == j)[0][0]
            zeros[idx_j_in_zeros] = 0
            zeros = zeros[zeros != 0]
            ones = np.append(ones, j)
            zeros = np.append(zeros, i)
            if cur_maxR < best_maxR:
                best_maxR = cur_maxR
                best_h = h.copy()
        else:
            pass
        
        # Cool down
        T = max(T * 0.9997, 0.01)
    
    return best_h, best_maxR

def fast_sa_binary(h, rng, time_limit=60.0):
    """Faster SA using numpy for R update."""
    n = len(h)
    h = h.astype(float).copy()
    R = compute_R(h)
    best_maxR = float(np.max(R))
    best_h = h.copy()
    cur_maxR = best_maxR
    start = time.time()
    T = 3.0
    
    ones_mask = h.astype(int)
    n_ones = int(np.sum(ones_mask))
    
    iters = 0
    while time.time() - start < time_limit:
        # Pick random i (one) and j (zero)
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if ones_mask[i] != 1 or ones_mask[j] != 0 or i == j:
            continue
        
        # Compute delta efficiently
        # For each lag l, delta[l] = -[(1-h[i-l])*I + h[i+l]*I] + [(1-h[j-l])*I - h[j+l]*I]
        delta = np.zeros(2 * n - 1)
        # i contribution (1->0): for each valid lag
        for l in range(-(n - 1), n):
            if l == 0:
                continue
            d = 0.0
            ip = i - l
            if 0 <= ip < n:
                d += (1.0 - h[ip])
            ip2 = i + l
            if 0 <= ip2 < n:
                d -= h[ip2]
            delta[l + n - 1] = -d
            jp = j - l
            if 0 <= jp < n:
                delta[l + n - 1] += (1.0 - h[jp])
            jp2 = j + l
            if 0 <= jp2 < n:
                delta[l + n - 1] -= h[jp2]
        
        R_new = R + delta
        new_maxR = float(np.max(R_new))
        
        if new_maxR <= cur_maxR or rng.random() < np.exp(-(new_maxR - cur_maxR) / T):
            h[i] = 0.0
            h[j] = 1.0
            ones_mask[i] = 0
            ones_mask[j] = 1
            R = R_new
            cur_maxR = new_maxR
            if cur_maxR < best_maxR:
                best_maxR = cur_maxR
                best_h = h.copy()
        
        iters += 1
        if iters % 100 == 0:
            T = max(T * 0.99, 0.05)
    
    return best_h, best_maxR

def greedy_binary(n, rng, iterations=10000):
    """Greedy construction of binary sequence."""
    h = np.zeros(n)
    # Start with random
    h[rng.choice(n, n // 2, replace=False)] = 1.0
    R = compute_R(h)
    cur_maxR = float(np.max(R))
    
    for _ in range(iterations):
        # Find the lag with max R
        max_lag = int(np.argmax(R))
        l = max_lag - (n - 1)
        # Try to reduce R at this lag by swapping
        # For lag l > 0: R_l = sum_{i=l}^{n-1} h[i](1-h[i-l])
        # To reduce, find i where h[i]=1, h[i-l]=0 and swapping helps
        best_swap = None
        best_new_max = cur_maxR
        # Try a few random swaps
        for _ in range(50):
            # Find positions to swap
            candidates_i = np.where(h == 1)[0]
            candidates_j = np.where(h == 0)[0]
            if len(candidates_i) == 0 or len(candidates_j) == 0:
                break
            i = candidates_i[rng.integers(len(candidates_i))]
            j = candidates_j[rng.integers(len(candidates_j))]
            if i == j:
                continue
            # Compute delta
            delta = np.zeros(2 * n - 1)
            for ll in range(-(n - 1), n):
                if ll == 0:
                    continue
                d = 0.0
                ip = i - ll
                if 0 <= ip < n:
                    d += (1.0 - h[ip])
                ip2 = i + ll
                if 0 <= ip2 < n:
                    d -= h[ip2]
                delta[ll + n - 1] = -d
                jp = j - ll
                if 0 <= jp < n:
                    delta[ll + n - 1] += (1.0 - h[jp])
                jp2 = j + ll
                if 0 <= jp2 < n:
                    delta[ll + n - 1] -= h[jp2]
            R_new = R + delta
            new_max = float(np.max(R_new))
            if new_max < best_new_max:
                best_new_max = new_max
                best_swap = (i, j, R_new)
        
        if best_swap is not None and best_new_max < cur_maxR:
            i, j, R = best_swap
            h[i] = 0.0
            h[j] = 1.0
            cur_maxR = best_new_max
    
    return h, cur_maxR

def try_binary_n(n, rng, time_limit=30.0):
    """Try to find a good binary sequence for given n."""
    best_maxR = float('inf')
    best_h = None
    t0 = time.time()
    
    # Multiple random starts
    n_starts = max(1, int(time_limit / 10))
    per_start = time_limit / n_starts
    
    for s in range(n_starts):
        if time.time() - t0 > time_limit:
            break
        h = np.zeros(n)
        h[rng.choice(n, n // 2, replace=False)] = 1.0
        try:
            h_sa, maxR = fast_sa_binary(h, rng, time_limit=per_start)
            c5 = maxR * 2.0 / n
            if maxR < best_maxR:
                best_maxR = maxR
                best_h = h_sa
        except Exception:
            pass
    
    if best_h is not None:
        return best_h, best_maxR * 2.0 / n
    return None, float('inf')

def try_continuous_improve(h, rng, time_limit=60.0):
    """Try to improve a continuous h."""
    n = len(h)
    target = n / 2.0
    # Try different beta and lr combinations
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t0 = time.time()
    
    for beta in [20, 40, 80, 120]:
        if time.time() - t0 > time_limit:
            break
        tl = min(15.0, time_limit - (time.time() - t0))
        if tl <= 0:
            break
        h_i = h.copy()
        h_out, c5_out = optimize_continuous(h_i, rng, time_limit=tl, target=target)
        if c5_out < best_c5:
            best_c5 = c5_out
            best_h = h_out
    
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t_start = time.time()
    
    best_c5 = 0.99
    best_h = None
    best_n = 0
    
    time_budget = budget_s
    
    # Stage 1: Improve initial_h_values if available
    try:
        init = initial_h_values
        n_init = len(init)
        h_init = np.array(init, dtype=float)
        c5_init = compute_c5(h_init)
        if c5_init < best_c5:
            best_c5 = c5_init
            best_h = h_init.copy()
            best_n = n_init
        
        # Try to improve
        elapsed = time.time() - t_start
        remaining = time_budget - elapsed - 200  # reserve 200s for binary
        if remaining > 30:
            h_c, c5_c = try_continuous_improve(h_init, rng, time_limit=remaining)
            if c5_c < best_c5:
                best_c5 = c5_c
                best_h = h_c
                best_n = n_init
    except Exception:
        pass
    
    # Stage 2: Binary optimization at various n
    elapsed = time.time() - t_start
    remaining = time_budget - elapsed - 60  # reserve 60s buffer
    
    # Try even n values, prioritizing those likely to give best C5
    n_candidates = [96, 128, 160, 192, 224, 256]
    # Sort by expected benefit: smaller n first (faster per trial)
    
    for n in n_candidates:
        if time.time() - t_start > time_budget - 60:
            break
        # Time allocation: give more time to promising n
        t_alloc = min(80.0, remaining * 0.3)
        if t_alloc < 10:
            break
        
        try:
            h_b, c5_b = try_binary_n(n, rng, time_limit=t_alloc)
            if h_b is not None and c5_b < best_c5:
                best_c5 = c5_b
                best_h = h_b
                best_n = n
        except Exception:
            pass
    
    # Stage 3: If we have a good binary solution, try to polish it
    elapsed = time.time() - t_start
    remaining = time_budget - elapsed - 10
    if remaining > 20 and best_h is not None and best_n % 2 == 0:
        try:
            h_b = best_h.astype(float)
            h_pol, maxR_pol = fast_sa_binary(h_b, rng, time_limit=remaining)
            c5_pol = maxR_pol * 2.0 / best_n
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol
        except Exception:
            pass
    
    # Stage 4: Try a few more random starts at the best n found
    elapsed = time.time() - t_start
    remaining = time_budget - elapsed - 5
    if remaining > 15 and best_n > 0:
        n_try = best_n
        t_alloc = remaining
        try:
            h_b2, c5_b2 = try_binary_n(n_try, rng, time_limit=t_alloc)
            if h_b2 is not None and c5_b2 < best_c5:
                best_c5 = c5_b2
                best_h = h_b2
                best_n = n_try
        except Exception:
            pass
    
    # Ensure we have a valid solution
    if best_h is None:
        # Fallback: simple construction
        n = 96
        h = np.zeros(n)
        h[::2] = 1.0  # alternating
        best_h = h
        best_c5 = compute_c5(h)
        best_n = n
    
    return (best_h, best_c5, best_n)
