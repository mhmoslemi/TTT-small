import numpy as np
from scipy.optimize import minimize_scalar
import time

def _eval_raw(h):
    """Return max of the full correlation profile (unscaled)."""
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")))

def _make_feasible(h, target_sum):
    """Project h onto [0,1]^n with sum = target_sum."""
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            if mask.any():
                h[mask] += diff / mask.sum()
        else:
            mask = h > 1e-15
            if mask.any():
                h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _pair_step(h, i, j):
    """Optimize shift between h[i] and h[j] in place. Returns new raw value."""
    a = h[i]
    b = h[j]
    s_lo = max(-a, b - 1.0) + 1e-12
    s_hi = min(1.0 - a, b) - 1e-12
    if s_lo >= s_hi:
        return _eval_raw(h)
    if s_hi - s_lo < 1e-7:
        return _eval_raw(h)

    h[i] = a
    h[j] = b

    def obj(s):
        h[i] = a + s
        h[j] = b - s
        return float(np.max(np.correlate(h, 1.0 - h, mode="full")))

    res = minimize_scalar(obj, bounds=(s_lo, s_hi), method='bounded',
                          options={'xatol': 1e-10, 'maxiter': 60})

    h[i] = a
    h[j] = b

    if res.fun < _eval_raw(h) - 1e-12:
        h[i] = a + res.x
        h[j] = b - res.x
        return res.fun
    return _eval_raw(h)

def _optimize(h, n, deadline, rng):

    no_improve = 0

                if time.time() > deadline:

                    break

                    continue
                new_raw = _pair_step(h, i, j)
                if new_raw < current_raw - 1e-11:
                    current_raw = new_raw
                    improved = True
        if improved:
            no_improve = 0
        else:
            no_improve += 1

    final_raw = _eval_raw(h)
    if final_raw < current_raw:
        current_raw = final_raw
    return current_raw

def _try_binary(n, deadline, rng):
    """Try binary local search and continuous relaxation."""
    h_bin = np.zeros(n)
    h_bin[rng.choice(n, size=n // 2, replace=False)] = 1.0
    raw = _eval_raw(h_bin)

    # Binary local search: swap 0 and 1
    no_improve = 0
    while time.time() < deadline and no_improve < 50:
        no_improve += 1
        ones = np.where(h_bin == 1.0)[0]
        zeros = np.where(h_bin == 0.0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h_bin[i], h_bin[j] = h_bin[j], h_bin[i]
        new_raw = _eval_raw(h_bin)

            h_bin[i], h_bin[j] = h_bin[j], h_bin[i]
        else:
            raw = new_raw
            no_improve = 0

        n_opt = len(best_relaxed)
        return best_relaxed, raw_rel, raw_rel * 2.0 / n_opt
    return None, None, None
