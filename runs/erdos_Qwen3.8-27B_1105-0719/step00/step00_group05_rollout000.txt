import numpy as np
from scipy import optimize
import time

def exact_objective(h):
    n = len(h)
    if n == 0:
        return np.inf
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def is_feasible(h, tol=1e-8):
    if len(h) == 0:
        return False
    if not np.all(np.isfinite(h)):
        return False
    if np.min(h) < -tol or np.max(h) > 1 + tol:
        return False
    if abs(np.sum(h) - len(h)/2.0) > tol:
        return False
    return True

def project_feasible(h, tol=1e-12, max_iter=100):
    """Project h onto {0 ≤ x ≤ 1, sum(x) = n/2} using bisection for water-filling."""
    n = len(h)
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    
    # Bisection for water-filling
    lo, hi = -2.0, 2.0
    # Find range
    while np.sum(np.clip(h - lo, 0, 1)) > target and lo > -10:
        lo *= 2
    while np.sum(np.clip(h - hi, 0, 1)) < target and hi < 10:
        hi *= 2
    
    for _ in range(max_iter):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h - mid, 0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
        if hi - lo < 1e-14:
            break
    
    result = np.clip(h - 0.5*(lo+hi), 0.0, 1.0)
    
    # Fix any remaining sum error
    diff = target - np.sum(result)
    if abs(diff) > tol:
        # Distribute the difference
        n_pos = np.sum(result < 1.0)
        n_neg = np.sum(result > 0.0)
        if diff > 0 and n_pos > 0:
            add = diff / n_pos
            mask = result < 1.0
            result[mask] = np.minimum(result[mask] + add, 1.0)
        elif diff < 0 and n_neg > 0:
            sub = -diff / n_neg
            mask = result > 0.0
            result[mask] = np.maximum(result[mask] - sub, 0.0)
    
    return result

def active_constrained_gradient(h, top_k=10, temperature=0.1):
    """
    Compute gradient of weighted sum of top-k overlap constraints.
    Returns (gradient, weights, values)
    """
    n = len(h)
    dx = 2.0 / n
    
    # Compute all overlap values
    corr = np.correlate(h, 1.0 - h, mode='full')
    # corr has length 2n-1, centered at index n-1
    # corr[n-1] is lag 0, corr[n-1+m] is lag m, corr[n-1-m] is lag -m
    
    # Sort to find top constraints
    sorted_idx = np.argsort(corr)[::-1]
    
    # Select top constraints with weights
    k = min(top_k, len(corr))
    top_idx = sorted_idx[:k]
    
    # Temperature-based weighting
    top_vals = corr[top_idx]
    weights = np.exp(-(top_vals - top_vals[0]) / max(temperature, 1e-10))
    weights = weights / weights.sum()
    
    # Build weight array for all lags
    w = np.zeros(len(corr))
    for i, idx in enumerate(top_idx):
        w[idx] = weights[i]
    
    # Compute gradient: for each lag m (indexed as lag = idx - (n-1)),
    # d c_m / d h[j] = (1-h[j-m]) * [j-m in range] - h[j+m] * [j+m in range]
    # where m = idx - (n-1)
    
    grad = np.zeros(n)
    center = n - 1
    for m_idx in range(len(corr)):
        if w[m_idx] == 0:
            continue
        m = m_idx - center  # this is the lag
        wm = w[m_idx]
        # d c_m / d h[j] for j = 0..n-1:
        # Term 1: h[j+m] appears → derivative (1-h[j]) for valid j, j+m
        #   valid when 0 ≤ j ≤ n-1 and 0 ≤ j+m ≤ n-1
        #   → j from max(0, -m) to min(n-1, n-1-m)
        # Term 2: (1-h[j]) appears → derivative -h[j+m] for valid j, j+m
        #   same range
        
        j_start = max(0, -m)
        j_end = min(n-1, n-1-m)
        if j_start > j_end:
            continue
        js = np.arange(j_start, j_end+1)
        # Term 1: (1 - h[js])
        # Term 2: -h[js + m]
        grad[js] += wm * ((1.0 - h[js]) - h[js + m])
    
    return grad, w, corr

def optimize_h(h_init, budget_s=100.0, seed=42, verbose=False):
    """Optimize h to minimize max overlap."""
    t0 = time.time()
    n = len(h_init)
    h = project_feasible(h_init.copy())
    current_c = exact_objective(h)
    best_h = h.copy()
    best_c = current_c
    best_n = n
    
    rng = np.random.RandomState(seed)

    temperature = 0.5

    max_patience = 50

        # Project gradient to zero-sum subspace
        grad = grad - np.mean(grad)
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:

                    current_c = c_new
                    if current_c < best_c:
                        best_c = current_c
                        best_h = h.copy()
                        best_n = n
                    improved = True

            step *= 0.5

                alpha = 0.03
                top_k = min(top_k + 2, 20)

    return best_h, best_c, best_n

def resize_interpolate(h, new_n):
    """Resize h to new_n points using linear interpolation."""
    n_old = len(h)

    x_old = (np.arange(n_old) + 0.5) / n_old

import numpy as np
from scipy import optimize
import time

def exact_objective(h):
    n = len(h)
    if n == 0:
        return np.inf
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def is_feasible(h, tol=1e-7):
    n = len(h)
    if n == 0:
        return False
    if not np.all(np.isfinite(h)):
        return False
    if np.min(h) < -tol or np.max(h) > 1.0 + tol:
        return False
    if abs(np.sum(h) - n / 2.0) > tol:
        return False
    return True

def project_feasible(h):
    n = len(h)
    target = n / 2.0
    h = np.asarray(h, dtype=np.float64).copy()
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        s_lo = np.sum(np.clip(h - lo, 0.0, 1.0))
        if s_lo > target and lo > -5.0:
            lo *= 2.0
            continue
        s_hi = np.sum(np.clip(h - hi, 0.0, 1.0))
        if s_hi < target and hi < 5.0:
            hi *= 2.0
            continue
        break
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
        if hi - lo < 1e-15:
            break
    lam = 0.5 * (lo + hi)
    result = np.clip(h - lam, 0.0, 1.0)
    diff = target - np.sum(result)
    if abs(diff) > 1e-14:
        if diff > 0:
            mask = result < 1.0
            cnt = np.sum(mask)
            if cnt > 0:
                add = min(diff / cnt, 1.0 - result[mask].max())
                result[mask] += add
        else:
            mask = result > 0.0
            cnt = np.sum(mask)
            if cnt > 0:
                sub = min(-diff / cnt, result[mask].min())
                result[mask] -= sub
    return result

def compute_overlaps(h):
    return np.correlate(h, 1.0 - h, mode='full')

def weighted_gradient(h, top_k=8, temperature=0.5):
    n = len(h)
    corr = compute_overlaps(h)
    center = n - 1
    sorted_idx = np.argsort(corr)[::-1]
    k = min(top_k, len(corr))
    top_idx = sorted_idx[:k]
    top_vals = corr[top_idx]
    weights = np.exp(-(top_vals - top_vals[0]) / max(temperature, 1e-10))
    weights = weights / weights.sum()
    w = np.zeros(len(corr))
    for i, idx in enumerate(top_idx):
        w[idx] = weights[i]
    grad = np.zeros(n)
    for p in range(len(corr)):
        if w[p] < 1e-15:
            continue
        k_lag = p - center
        wm = w[p]
        j_start = max(0, -k_lag)
        j_end = min(n - 1, n - 1 - k_lag)
        if j_start > j_end:
            continue
        js = np.arange(j_start, j_end + 1)
        grad[js] += wm * ((1.0 - h[js]) - h[js + k_lag])
    return grad, corr

def optimize_h(h_init, budget_s=60.0, seed=42, top_k_init=8, temp_init=0.8):
    t0 = time.time()
    n = len(h_init)
    h = project_feasible(h_init.copy())
    current_c = exact_objective(h)
    best_h = h.copy()
    best_c = current_c
    alpha = 0.08
    top_k = top_k_init
    temp = temp_init
    patience = 0
    max_patience = 40
    rng = np.random.RandomState(seed)
    iter_count = 0
    while time.time() - t0 < budget_s:
        iter_count += 1
        grad, corr = weighted_gradient(h, top_k=top_k, temperature=temp)
        grad = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-13:
            patience += 5
        else:
            step = alpha / max(gnorm, 1e-10)
            improved = False
            for trial in range(25):
                h_new = project_feasible(h - step * grad)
                if is_feasible(h_new):
                    c_new = exact_objective(h_new)
                    if c_new < current_c - 1e-13:
                        h = h_new
                        current_c = c_new
                        if current_c < best_c:
                            best_c = current_c
                            best_h = h.copy()
                        improved = True
                        break
                step *= 0.5
            if improved:
                patience = 0
                alpha = min(alpha * 1.05, 0.15)
            else:
                patience += 1
                alpha *= 0.75
        if patience > max_patience:
            pert = rng.randn(n) * 0.015
            pert -= np.mean(pert)
            h = project_feasible(h + pert)
            current_c = exact_objective(h)
            patience = 0
            alpha = 0.03
            temp = max(temp * 0.7, 0.1)
            top_k = min(top_k + 1, 20)
        if iter_count % 50 == 0:
            temp = max(temp * 0.95, 0.15)
    return best_h, best_c, best_n if 'best_n' in dir() else len(best_h)

def size_interpolate(h, new_n):
    n_old = len(h)
    if n_old == new_n:
        return h.copy()
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(new_n) + 0.5) / new_n
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new)

def make_binary_pattern(n, seed=0):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    idx = rng.choice(n, n // 2, replace=False)
    h[idx] = 1.0
    return h

def make_smooth_pattern(n, freq=1.0, phase=0.0, amp=0.3):
    x = (np.arange(n) + 0.5) / n
    h = 0.5 + amp * np.sin(2.0 * np.pi * freq * x + phase)
    return project_feasible(h)

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    SAFETY = 3.0

    def remaining():
        return budget_s - (time.time() - t_start) - SAFETY

            best_c = exact_objective(h_p)
            best_n = len(h_p)

    # Phase 1: Optimize parent directly
    if best_h is not None and remaining() > 20:
        t_alloc = min(250.0, remaining() * 0.35)

        if c_o < best_c:
            best_c = c_o
            best_h = h_o
            best_n = len(h_o)

    # Phase 2: Try different top_k and temperature settings on best
    if best_h is not None and remaining() > 20:

            if remaining() < 15:

            t_alloc = min(80.0, remaining() * 0.15)
            if t_alloc < 5:
                break
            h_o, c_o, _ = optimize_h(best_h, budget_s=t_alloc, seed=seed + 1000 + tk, top_k_init=tk, temp_init=tm)
            if c_o < best_c:
                best_c = c_o
                best_h = h_o
                best_n = len(h_o)

    # Phase 3: Try other starting points
    if remaining() > 30:
        n_base = best_n if best_h is not None else 42
        starts = []
        starts.append(make_binary_pattern(n_base, seed=seed))
        starts.append(make_binary_pattern(n_base, seed=seed + 7))
        starts.append(make_smooth_pattern(n_base, freq=1.0, phase=0.0, amp=0.3))
        starts.append(make_smooth_pattern(n_base, freq=2.0, phase=0.5, amp=0.25))
        starts.append(make_smooth_pattern(n_base, freq=3.0, phase=1.0, amp=0.2))
        rng = np.random.RandomState(seed + 555)
        for i in range(3):
            h_rand = make_smooth_pattern(n_base, freq=rng.uniform(0.5, 3.0),
                                         phase=rng.uniform(0, 6.28), amp=rng.uniform(0.2, 0.4))
            starts.append(h_rand)
        for i, s in enumerate(starts):
            if remaining() < 10:
                break
            s = project_feasible(s)
            c_s = exact_objective(s)
            t_alloc = min(50.0, remaining() * 0.12)
            if t_alloc < 5:
                break
            h_o, c_o, _ = optimize_h(s, budget_s=t_alloc, seed=seed + 2000 + i)
            if c_o < best_c:
                best_c = c_o
                best_h = h_o
                best_n = len(h_o)

    # Phase 4: Try different dimensions
    if best_h is not None and remaining() > 25:
        for factor in [1.2, 0.8, 1.4, 0.7]:
            if remaining() < 15:

            new_n = max(20, min(80, int(round(best_n * factor))))
            if new_n == best_n:

            h_resized = size_interpolate(best_h, new_n)
            if not is_feasible(h_resized):

            t_alloc = min(60.0, remaining() * 0.15)

    # Phase 5: Fine-tuning with small steps
    if best_h is not None and remaining() > 10:
        h = best_h.copy()
        cur_c = best_c
