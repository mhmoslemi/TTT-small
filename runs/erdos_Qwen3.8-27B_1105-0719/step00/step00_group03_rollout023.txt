def c5_eval(h):

def project_balanced(h):
    """Project h to satisfy sum(h) = len(h)/2 and 0<=h<=1 using L2 projection"""
    n = len(h)
    target = n / 2.0
    # Bisection on lambda: find lambda such that sum(clip(h + lambda, 0, 1)) = target
    lo, hi = -max(h) - 1.0, 1.0 - min(h) + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        val = np.clip(h + mid, 0, 1)
        s = np.sum(val)
        if s > target:
            hi = mid
        else:
            lo = mid
    h_proj = np.clip(h + (lo + hi) / 2.0, 0, 1)
    # Fix residual
    residual = target - np.sum(h_proj)
    if abs(residual) > 1e-10:
        if residual > 0:
            # Need to add mass
            slacks = 1.0 - h_proj
            order = np.argsort(-slacks)
            for i in order:
                add = min(residual, slacks[i])
                h_proj[i] += add
                residual -= add
                if abs(residual) < 1e-15:
                    break
        else:
            # Need to remove mass
            order = np.argsort(h_proj)
            for i in order:
                rem = min(-residual, h_proj[i])
                h_proj[i] -= rem
                residual += rem
                if abs(residual) < 1e-15:
                    break
    return h_proj

    old_n = len(h)

  Maximum is at k=0: C₅ = 1. That's worse than 0.5 (uniform).

So a block function is bad. We need h to be "spread out."

What about a "comb" pattern? E.g., h = 1 at even positions and 0 at odd positions (in the discrete sense). For n=45 (odd), this doesn't quite work, but let's think about the continuous analog.

If h(x) = 1 for x ∈ [0, ε] ∪ [2ε, 3ε] ∪ [4ε, 5ε] ∪ ... (a periodic comb with period 2ε), then the autocorrelation has peaks at multiples of 2ε. The maximum overlap might be related to the duty cycle.

For a comb with duty cycle d (fraction of time h=1) and period T:
- ∫h over [0,2] = (2/T) * d * T = 2d. We need this to equal 1, so d = 1/2.
- The overlap at lag k: for small |k|, it's approximately d*(1-d) = 1/4 (times the length of the domain minus |k|). For larger |k| that are multiples of T, it's higher.

Actually, this is getting complicated. Let me just rely on the numerical optimization.

Let me write the final, clean version of the code. I'll focus on correctness and efficiency.

One more optimization: in the pair_improve function, instead of copying the whole array for each evaluation, I can modify in-place and restore. But since numpy arrays are small (n=45), the copy overhead is negligible compared to the correlate computation.

Actually, let me benchmark mentally: np.correlate(h, 1-h, 'full') for n=45 involves computing 89 dot products of length up to 45. In numpy, this is a single C call that probably takes 2-5 μs. Copying a 45-element array takes <1 μs. So the copy is about 20% of the correlate cost. Not terrible, but I could save it.

Alternative: precompute the correlation and update it incrementally. When h[p] += tau and h[q] -= tau, the change in correlation at lag L is:
- The correlation at lag L is sum_i h[i] * (1-h[i+L])
- After the change: sum_i h'[i] * (1-h'[i+L])
- The change involves terms where i=p or i+L=p (affecting h) and where i=q or i+L=q (affecting 1-h).

This is doable but complex. Let me skip this optimization and just use the direct evaluation.

import numpy as np
import time

def c5_eval(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def project_balanced(h):
    n = len(h)
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    h_proj = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - h_proj.sum()
    if abs(residual) > 1e-10:
        if residual > 0:
            slacks = 1.0 - h_proj
            for i in np.argsort(-slacks):
                add = min(residual, slacks[i])
                h_proj[i] += add
                residual -= add
                if abs(residual) < 1e-14:
                    break
        else:
            for i in np.argsort(h_proj):
                rem = min(-residual, h_proj[i])
                h_proj[i] -= rem
                residual += rem
                if abs(residual) < 1e-14:
                    break
    return h_proj

def pair_improve(h, p, q, current_c):
    lo = max(-h[p], h[q] - 1.0) + 1e-13
    hi = min(1.0 - h[p], h[q]) - 1e-13
    if lo >= hi:
        return None, current_c
    a, b = lo, hi
    for _ in range(35):
        m1 = a + (b - a) / 3.0
        m2 = b - (b - a) / 3.0
        hc = h.copy()
        hc[p] += m1
        hc[q] -= m1
        c1 = c5_eval(hc)
        hc = h.copy()
        hc[p] += m2
        hc[q] -= m2
        c2 = c5_eval(hc)
        if c1 <= c2:
            b = m2
        else:
            a = m1
    tau = (a + b) / 2.0
    h_new = h.copy()
    h_new[p] = h[p] + tau
    h_new[q] = h[q] - tau
    h_new[p] = np.clip(h_new[p], 0.0, 1.0)
    h_new[q] = np.clip(h_new[q], 0.0, 1.0)
    c_new = c5_eval(h_new)
    if c_new < current_c - 1e-12:
        return h_new, c_new
    return None, current_c

def do_passes(h, c, deadline):
    n = len(h)
    while time.monotonic() < deadline:
        improved = False
        for p in range(n):
            for q in range(n):
                if p == q:
                    continue
                if time.monotonic() > deadline:
                    return h, c
                h_new, c_new = pair_improve(h, p, q, c)
                if h_new is not None:
                    h = h_new
                    c = c_new
                    improved = True
        if not improved:
            break
    return h, c

def interpolate_to(h, new_n):
    old_n = len(h)
    old_dx = 2.0 / old_n
    new_dx = 2.0 / new_n
    h_new = np.zeros(new_n)
    for j in range(new_n):
        x_lo = j * new_dx
        x_hi = (j + 1) * new_dx
        total = 0.0
        for i in range(old_n):
            y_lo = i * old_dx
            y_hi = (i + 1) * old_dx
            ov = min(x_hi, y_hi) - max(x_lo, y_lo)
            if ov > 0:
                total += ov * h[i]
        h_new[j] = total / new_dx
    return project_balanced(h_new)
