def repair_box_sum(h):
    """Ensure 0 <= h[i] <= 1 and sum(h) = n/2"""
    n = len(h)
    target = 0.5 * n
    h = np.clip(np.asarray(h, dtype=np.float64), 0.0, 1.0)
    # Water-filling to achieve exact sum
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            # Need to add mass
            avail = h < 1.0 - 1e-12
            if not avail.any():
                break
            add = diff / avail.sum()
            h[avail] = np.minimum(h[avail] + add, 1.0)
        else:
            # Need to remove mass
            avail = h > 1e-12
            if not avail.any():
                break
            rem = (-diff) / avail.sum()
            h[avail] = np.maximum(h[avail] - rem, 0.0)
    # Final clipping
    h = np.clip(h, 0.0, 1.0)
    return h

def golden_minimize(f, a, b, max_iter=60):
    """Minimize f on [a,b] using golden section search"""
    if b - a < 1e-15:
        return (a+b)/2, f((a+b)/2)
    gr = (np.sqrt(5.0) - 1.0) / 2.0  # golden ratio ~0.618
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = f(c)
    fd = f(d)

            d = c

            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = f(d)
