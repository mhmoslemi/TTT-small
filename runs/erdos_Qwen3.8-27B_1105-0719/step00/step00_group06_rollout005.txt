from scipy.optimize import linprog
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def profile_c5(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def project_feasible(v, n):
    target = n / 2.0
    lo = v.min() - 2.0
    hi = v.max() + 2.0
    for _ in range(200):
        tau = (lo + hi) / 2.0
        h = np.clip(v - tau, 0.0, 1.0)
        s = h.sum()
        if s > target:
            lo = tau
        else:
            hi = tau
    h = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        idx = np.where((h > 0) & (h < 1))[0]
        if len(idx) > 0:
            h[idx] += diff / len(idx)
            h = np.clip(h, 0.0, 1.0)
    return h

def grad_lag(h, k):
    """Gradient of correlation lag k (0-indexed in full output)."""
    n = len(h)
    lag = k - (n - 1)
    g = np.zeros(n)
    if lag >= 0:
        # sum_{i=0}^{n-1-lag} h[i+lag] * (1-h[i])
        # d/dh[j]: +（1-h[j-lag) for j>=lag, -h[j+lag] for j<=n-1-lag
        g[lag:] += (1.0 - h[:n - lag])
        g[:n - lag] -= h[lag:]
    else:
        s = -lag
        # sum_{i=0}^{n-1-s} h[i] * (1-h[i+s])
        # d/dh[j]: +(1-h[j+s]) for j<=n-1-s, -h[j-s] for j>=s
        g[:n - s] += (1.0 - h[s:])
        g[s:] -= h[:n - s]
    return g

def build_grad_matrix(h):
    """Build matrix where row k is gradient of lag k."""
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        G[k] = grad_lag(h, k) * dx
    return G

def lp_descent_step(h, n, active_lags, step_budget=0.2):
    """Find LP descent direction."""
    dx = 2.0 / n
    prof = profile_c5(h)
    G = np.zeros((len(active_lags), n))
    for i, k in enumerate(active_lags):
        G[i] = grad_lag(h, k) * dx
    
    # Variables: d[0..n-1], t
    # Minimize: t
    # Subject to: G @ d - t <= -prof[active_lags]  (i.e., prof[k] + G[k]@d <= t)
    #            sum(d) = 0
    #            -h[j] <= d[j] <= 1-h[j]
    #            sum(|d|) <= step_budget (L1 constraint on step size) - optional
    
    n_vars = n + 1  # d + t
    c = np.zeros(n_vars)
    c[-1] = 1.0  # minimize t
    
    A_ub = np.zeros((len(active_lags), n_vars))
    b_ub = np.zeros(len(active_lags))
    for i in range(len(active_lags)):
        A_ub[i, :n] = G[i]
        A_ub[i, -1] = -1.0
        b_ub[i] = -prof[active_lags[i]]
    
    A_eq = np.zeros((1, n_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    bounds = [(lo, hi) for lo, hi in zip(-h, 1.0 - h)]
    bounds.append((prof.max() - 0.5, None))  # t lower bound
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            d = res.x[:n]
            t = res.x[-1]
            return d, t
    except Exception:
        pass
    return None, None
