    dx = 2.0 / n
    h = h.copy().astype(float)
    g = 1.0 - h
    corr = correlate(h, g, mode='full')
    c5_val = np.max(corr) * dx
    
    for sweep in range(max_sweeps):
        improved = False
        for a in range(n):
            for b in range(a+1, n):  # Only a < b to avoid redundancy
                # Compute correction coefficients
                linear = np.zeros(2*n - 1)
                quad = np.zeros(2*n - 1)

                    # Collect affected (p,q) pairs

                    if 0 <= a-L < n:
                        pairs.append((a, a-L))
                    if 0 <= b-L < n:
                        pairs.append((b, b-L))
                    if 0 <= a+L < n:
                        pairs.append((a+L, a))
                    if 0 <= b+L < n:
                        pairs.append((b+L, b))

                            elif p == b and q == b:

                            elif q == a:
                                linear[t] += -h[p]
                            elif q == b:
                                linear[t] += h[p]

        best_sweep_improvement = 0.0
        
        # Randomize pair order for diversity
        pairs = [(a, b) for a in range(n) for b in range(a+1, n)]
