    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)

        if abs(diff) < 1e-13:

        if diff > 0:

            if np.sum(room) < 1e-15:

from scipy.optimize import linprog
import time


def evaluate(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))


def project_feasible(h, n):
    target = n / 2.0
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(20):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = np.sum(room)
            if s < 1e-15:
                break
            h += diff * room / s
        else:
            mass = h
            s = np.sum(mass)
            if s < 1e-15:
                break
            h += diff * mass / s
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        idx = int(np.argmax(1.0 - h)) if diff > 0 else int(np.argmax(h))
        h[idx] = float(np.clip(h[idx] + diff, 0.0, 1.0))
    return h


def solve_lp_sub(n, fixed_vec, role):
    dx = 2.0 / n
    M = 2 * n - 1
    A_ub = np.zeros((M, n + 1))
    b_ub = np.zeros(M)

    if role == 0:
        # Variable is h_new, fixed is b = 1-h
        # j: k = j-(n-1)
        #   k>=0: sum_{i=0}^{n-1-k} h_new[i+k]*b[i] => coeff of h_new[p] (p=i+k): dx*b[p-k]
        #   k<0, m=-k: sum_{i=m}^{n-1} h_new[i]*b[i-m] => coeff of h_new[p]: dx*b[p-m]
        b = fixed_vec
        for j in range(M):
            k = j - (n - 1)
            if k >= 0:
                for p in range(k, n):
                    A_ub[j, p] = dx * b[p - k]
            else:
                m = -k
                for p in range(m, n):
                    A_ub[j, p] = dx * b[p - m]
            A_ub[j, n] = -1.0
    else:
        # Variable is b_new, fixed is a = h
        # j: k = j-(n-1)
        #   k>=0: sum_{i=0}^{n-1-k} a[i+k]*b_new[i] => coeff of b_new[i]: dx*a[i+k]
        #   k<0, m=-k: sum_{i=m}^{n-1} a[i]*b_new[i-m] => coeff of b_new[j'](j'=i-m): dx*a[j'+m]
        a = fixed_vec
        for j in range(M):
            k = j - (n - 1)
            if k >= 0:
                for i in range(n - k):
                    A_ub[j, i] = dx * a[i + k]
            else:
                m = -k
                for jp in range(n - m):
                    A_ub[j, jp] = dx * a[jp + m]
            A_ub[j, n] = -1.0

    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

            h_pert = project_feasible(h_pert, n)
            b_pert = 1.0 - h_pert
            h_cand = solve_lp_sub(n, b_pert, 0)
            if h_cand is not None:
                h_cand = project_feasible(h_cand, n)
                sc = evaluate(h_cand)
                if sc < best_score - 1e-13:
                    best_h = h_cand
                    best_score = sc
                    improved = True

            if not improved:
                break

    return best_h, best_score


def coordinate_polish(h_start, n, num_passes=3, t_limit=None, t_start=None):
    best_h = project_feasible(h_start.copy(), n)
    best_score = evaluate(best_h)
    steps = [0.05, 0.1, 0.15, 0.2, 0.3, 0.5]
    rng = np.random.RandomState(1)

    for _ in range(num_passes):
        if t_start is not None and time.time() - t_start > t_limit:
            break
        improved = False
        order = rng.permutation(n)
        for p in order:
            if t_start is not None and time.time() - t_start > t_limit:
                break
            for q in range(n):
                if q == p:
                    continue
                for d in steps:
                    # h[p] += d, h[q] -= d
                    if best_h[p] + d <= 1.0 + 1e-12 and best_h[q] - d >= -1e-12:
                        best_h[p] += d
                        best_h[q] -= d
                        best_h = np.clip(best_h, 0.0, 1.0)
                        sc = evaluate(best_h)
                        if sc < best_score - 1e-13:
                            best_score = sc
                            improved = True
                        else:
                            best_h[p] -= d
                            best_h[q] += d
                    # h[p] -= d, h[q] += d
                    elif best_h[p] - d >= -1e-12 and best_h[q] + d <= 1.0 + 1e-12:
                        best_h[p] -= d
                        best_h[q] += d
                        best_h = np.clip(best_h, 0.0, 1.0)
                        sc = evaluate(best_h)
                        if sc < best_score - 1e-13:
                            best_score = sc
                            improved = True
                        else:
                            best_h[p] += d
                            best_h[q] -= d
        if not improved:
            break

    return best_h, best_score


def resample_to(h_src, n_new):
    n_old = len(h_src)
    x_old = (np.arange(n_old) + 0.5) * 2.0 / n_old
    x_new = (np.arange(n_new) + 0.5) * 2.0 / n_new
    h_new = np.interp(x_new, x_old, h_src)
    return project_feasible(h_new, n_new)


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t_start = time.time()
    t_limit = budget_s * 0.95

    # Get initial construction
    try:
        h0 = np.array(initial_h_values, dtype=float)
    except NameError:
        h0 = np.full(70, 0.5)

    n0 = len(h0)
    h0 = project_feasible(h0, n0)

    best_h = h0.copy()
    best_score = evaluate(h0)
    best_n = n0

    # Phase 1: Alternating LP on initial grid
    if time.time() - t_start < t_limit * 0.3:
        h1, s1 = alternate_lp(h0, n0, max_iters=300, t_start=t_start, t_limit=t_limit * 0.3)
        if s1 < best_score:
            best_h, best_score, best_n = h1, s1, n0

    # Phase 2: Coordinate polish on initial grid
    if time.time() - t_start < t_limit * 0.45:
        h2, s2 = coordinate_polish(best_h, best_n, num_passes=3, t_start=t_start, t_limit=t_limit * 0.45)
        if s2 < best_score:
            best_h, best_score = h2, s2

    # Phase 3: Try different grid sizes
    if time.time() - t_start < t_limit * 0.7:
        grid_sizes = sorted(set([best_n - 3, best_n - 2, best_n - 1, best_n + 1,
                                  best_n + 2, best_n + 3, best_n + 5, best_n + 10,
                                  best_n + 20, best_n - 5]))
        for n_try in grid_sizes:
            if n_try < 10 or n_try > 200:
                continue
            if time.time() - t_start > t_limit * 0.85:
                break
            h_try = resample_to(best_h, n_try)
            sc_try = evaluate(h_try)
            h_try2, sc_try2 = alternate_lp(h_try, n_try, max_iters=80,
                                           t_start=t_start, t_limit=t_limit * 0.85)
            sc_try2 = min(sc_try, sc_try2)
            h_best_try = h_try if sc_try < sc_try2 else h_try2
            if sc_try2 < best_score:
                best_h, best_score, best_n = h_best_try, sc_try2, n_try
            if time.time() - t_start < t_limit * 0.85:
                h_try3, sc_try3 = coordinate_polish(h_best_try, n_try, num_passes=2,
                                                    t_start=t_start, t_limit=t_limit * 0.85)
                if sc_try3 < best_score:
                    best_h, best_score, best_n = h_try3, sc_try3, n_try

    # Phase 4: Random restarts
    if time.time() - t_start < t_limit * 0.75:
        rng = np.random.RandomState(seed)
        for restart in range(20):
            if time.time() - t_start > t_limit * 0.92:
                break
            # Perturb current best
            h_r = best_h + rng.randn(best_n) * rng.uniform(0.02, 0.08)
            h_r = project_feasible(h_r, best_n)
            h_r2, sc_r = alternate_lp(h_r, best_n, max_iters=50,
                                      t_start=t_start, t_limit=t_limit * 0.92)
            if sc_r < best_score:
                best_h, best_score = h_r2, sc_r

    # Phase 5: Final polish
    if time.time() - t_start < t_limit * 0.95:
        h_final, sc_final = coordinate_polish(best_h, best_n, num_passes=2,
                                              t_start=t_start, t_limit=t_limit)
        if sc_final < best_score:
            best_h, best_score = h_final, sc_final

    # One more LP round on the best
    if time.time() - t_start < t_limit:
        h_last, sc_last = alternate_lp(best_h, best_n, max_iters=50,
                                       t_start=t_start, t_limit=t_limit)
        if sc_last < best_score:
            best_h, best_score = h_last, sc_last

    return best_h, best_score, best_n
