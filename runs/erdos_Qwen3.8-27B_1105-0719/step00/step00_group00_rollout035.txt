def active_corr_gradient(h, n):
    """Return the gradient of the active (max) correlation w.r.t. h."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_val = corr.max()
    # Active lags
    active = np.where(np.abs(corr - max_val) < 1e-10)[0]
    
    grad = np.zeros(n)
    for s in active:
        d = s - (n - 1)
        j_lo1 = max(0, -d)
        j_hi1 = min(n, n - d)
        if j_lo1 < j_hi1:
            idx = np.arange(j_lo1, j_hi1)
            grad[idx] += (1.0 - h[idx + d])
        j_lo2 = max(0, d)
        j_hi2 = min(n, n + d)
        if j_lo2 < j_hi2:
            idx = np.arange(j_lo2, j_hi2)
            grad[idx] -= h[idx - d]
    
    # Equal weight for all active
    grad /= len(active)
    return grad, max_val

        grad, max_corr = active_corr_gradient(h, n)
        
        # Find best pair (i,j): decrease h in gradient direction
        # We want to move h - t * direction where direction decreases the max
        # Pair transfer: h[i] += t, h[j] -= t
        # Direction: e_i - e_j (i gets more, j gets less)
        # Gradient dot direction = grad[i] - grad[j]
        # We want this to be negative (decreasing the objective)
        # So we want grad[i] < grad[j] (move mass from high grad to low grad)
        
        best_pair = None
        best_dot = 0  # We want most negative grad[i] - grad[j]
        for i in range(n):
            if h[i] >= 1.0 - 1e-10:
                continue
            for j in range(n):
                if i == j or h[j] <= 1e-10:
                    continue
                dot = grad[i] - grad[j]
                if dot < best_dot:
                    best_dot = dot

                    t_test = 0.01 * min(1-h[i], h[j])

                    c5_test = exact_c5(h)

import numpy as np
import time

def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_feasible(y, n):
    target = n / 2.0
    lo = float(y.min()) - 1.5
    hi = float(y.max()) + 1.5
    for _ in range(100):
        mid = (lo + hi) / 2.0
        val = float(np.sum(np.clip(y - mid, 0.0, 1.0)))
        if val > target:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    h = np.clip(y - t, 0.0, 1.0)
    resid = target - float(h.sum())
    if abs(resid) > 1e-13:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = int(mask.sum())
        if cnt > 0:
            h[mask] += resid / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def lse_grad_beta(h, n, beta):
    """Compute LSE smooth max of correlations and its gradient using given beta."""
    corr = compute_corr(h)
    max_c = float(corr.max())
    shifted = beta * (corr - max_c)
    exp_terms = np.exp(shifted)
    total = float(exp_terms.sum())
    if total < 1e-300:
        return max_c, np.zeros(n)
    weights = exp_terms / total
    obj = max_c + np.log(total) / beta

    grad = np.zeros(n)
    for s in range(2 * n - 1):
        w = float(weights[s])
        if w < 1e-13:
            continue
        d = s - (n - 1)
        # First term: (1-h[j-d]) for j in [max(0,d), min(n, n+d))
        lo1 = max(0, d)
        hi1 = min(n, n + d)
        if lo1 < hi1:
            grad[lo1:hi1] += w * (1.0 - h[lo1 - d:hi1 - d])
        # Second term: -h[j+d] for j in [max(0,-d), min(n, n-d))
        lo2 = max(0, -d)
        hi2 = min(n, n - d)
        if lo2 < hi2:
            grad[lo2:hi2] -= w * h[lo2 + d:hi2 + d]
    grad /= beta
    return obj, grad

def lse_optimize(h0, n, time_limit=40):
    best_h = h0.copy()
    best_c5 = exact_c5(best_h)
    start = time.time()
    h = h0.copy()

    for beta in [10, 25, 50, 100, 200, 400, 800, 1600, 3200, 6400]:
        if time.time() - start > time_limit:
            break
        for it in range(2000):
            if time.time() - start > time_limit:
                break
            obj, grad = lse_grad_beta(h, n, beta)
            improved = False
            steps = [0.1, 0.05, 0.01, 0.5, 0.02, 1.0, 0.005, 0.2, 0.001]
            for step in steps:
                h_new = project_to_feasible(h - step * grad, n)
                obj_new, _ = lse_grad_beta(h_new, n, beta)
                if obj_new < obj - 1e-15:
                    h = h_new
                    improved = True
                    break
            if not improved:
                break
            c5_now = exact_c5(h)
            if c5_now < best_c5 - 1e-14:
                best_c5 = c5_now
                best_h = h.copy()
    return best_h, best_c5

def pair_local_search(h0, n, time_limit=60, n_grid=16):
    h = h0.copy()
    best_c5 = exact_c5(h)
    start = time.time()

    while time.time() - start < time_limit:
        best_gain = 1e-14
        best_move = None
        for i in range(n):
            if h[i] >= 1.0 - 1e-9:
                continue
            for j in range(n):
                if i == j or h[j] <= 1e-9:
                    continue
                t_max = min(1.0 - h[i], h[j])
                if t_max < 1e-9:
                    continue
                t_vals = np.linspace(0.0, t_max, n_grid)[1:]
                for t in t_vals:
                    h[i] += t
                    h[j] -= t
                    c5_t = exact_c5(h)
                    h[i] -= t
                    h[j] += t
                    gain = best_c5 - c5_t
                    if gain > best_gain:
                        best_gain = gain
                        best_move = (i, j, t)
        if best_move is not None:
            i, j, t = best_move
            h[i] += t
            h[j] -= t
            best_c5 = exact_c5(h)
        else:
            break
    return h, best_c5

def gradient_local_search(h0, n, time_limit=60):
    """Local search guided by active correlation gradient."""
    h = h0.copy()
    best_c5 = exact_c5(h)
    start = time.time()

    while time.time() - start < time_limit:
        # Get gradient of the max correlation (using very high beta LSE)
        _, grad = lse_grad_beta(h, n, 10000.0)
        # Find best pair: want grad[i] - grad[j] as negative as possible
        # (move mass from high-grad to low-grad)
        best_dot = 0.0
        best_pair = None
        for i in range(n):
            if h[i] >= 1.0 - 1e-9:
                continue
            for j in range(n):
                if i == j or h[j] <= 1e-9:
                    continue
                dot = grad[i] - grad[j]
                if dot < best_dot - 1e-12:
                    best_dot = dot
                    best_pair = (i, j)
        if best_pair is None:
            break
        i, j = best_pair
        t_max = min(1.0 - h[i], h[j])
        # Grid search for optimal t
        best_t = 0.0
        best_val = best_c5
        for t in np.linspace(0.0, t_max, 20)[1:]:
            h[i] += t
            h[j] -= t
            c5_t = exact_c5(h)
            h[i] -= t
            h[j] += t
            if c5_t < best_val:
                best_val = c5_t
                best_t = t
        if best_t > 0:
            h[i] += best_t
            h[j] -= best_t
            best_c5 = best_val
        else:
            break
    return h, best_c5

        if c52 < best_c5:

    while remaining() > 60:
        # Perturb current best
        h_pert = best_h + rng.normal(0, 0.05, n)
        h_pert = project_to_feasible(h_pert, n)

        tl = min(15, remaining() - 40)

            break
        h2, c52 = lse_optimize(h_pert, n, time_limit=tl)

        tl = min(20, remaining() - 30)
        if tl >= 5:
            h3, c53 = gradient_local_search(h2, n, time_limit=tl)
            if c53 < c52:
                h2, c52 = h3, c53

        if c52 < best_c5:
            best_h, best_c5 = h2.copy(), c52

    # Phase 5: Try random restarts at same n
    while remaining() > 45:
        h_rand = rng.uniform(0.1, 0.9, n)
        h_rand = project_to_feasible(h_rand, n)

        tl = min(12, remaining() - 25)
        if tl < 5:
            break
        h2, c52 = lse_optimize(h_rand, n, time_limit=tl)

        tl = min(20, remaining() - 20)
        if tl >= 5:
            h3, c53 = gradient_local_search(h2, n, time_limit=tl)
            if c53 < c52:
                h2, c52 = h3, c53

        tl = min(15, remaining() - 15)
        if tl >= 5:
            h4, c54 = pair_local_search(h2, n, time_limit=tl)
            if c54 < c52:
                h2, c52 = h4, c54

        if c52 < best_c5:
            best_h, best_c5 = h2.copy(), c52

    # Phase 6: Try nearby n values
    for n_try in [n - 1, n + 1, n - 2, n + 2]:
        if n_try < 10 or remaining() < 35:
            continue
        h_rand = rng.uniform(0.1, 0.9, n_try)
        h_rand = project_to_feasible(h_rand, n_try)
        tl = min(12, remaining() - 25)
        if tl < 5:
            continue
        h2, c52 = lse_optimize(h_rand, n_try, time_limit=tl)
        tl = min(15, remaining() - 20)
        if tl >= 5:
            h3, c53 = gradient_local_search(h2, n_try, time_limit=tl)
            if c53 < c52:
                h2, c52 = h3, c53
        if c52 < best_c5:
            best_h, best_c5 = h2.copy(), c52

    # Final verification
    final_c5 = exact_c5(best_h)
    return best_h, final_c5, len(best_h)
