import numpy as np
import time

def get_now():
    return time.perf_counter()

def project_box_sum(y, mass):
    """Project y onto {h: 0 <= h[i] <= 1, sum(h) = mass}"""
    n = len(y)
    y = np.asarray(y, dtype=float)
    # Binary search for threshold t such that sum(clip(y - t, 0, 1)) = mass
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        h = np.clip(y - mid, 0.0, 1.0)
        s = h.sum()
        if s > mass:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix remaining discrepancy
    diff = mass - h.sum()
    if abs(diff) > 1e-12:
        # Distribute remaining to elements not at bounds
        if diff > 0:
            idx = np.argsort(1.0 - h)  # prefer elements with more room
        else:
            idx = np.argsort(h)  # prefer elements with more mass
        for i in idx:
            if abs(diff) < 1e-12:
                break
            if diff > 0:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
            else:
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub
    return h

    """Compute C5 value and all lag values"""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return c.max() * dx, c, dx

def lag_gradient(h, s, n):
    """Compute gradient of f_s(h) = sum_{i: valid} h[i]*(1-h[i+s])
    where s is the shift (can be negative)"""
    grad = np.zeros(n)
    if s >= 0:
        # valid i: 0 <= i < n and 0 <= i+s < n, so 0 <= i < n-s
        for i in range(n - s):
            grad[i] += (1.0 - h[i + s])
            grad[i + s] += (-h[i])
    else:
        # s < 0, let t = -s > 0
        # valid i: 0 <= i < n and 0 <= i - t < n, so t <= i < n
        t = -s
        for i in range(t, n):
            grad[i] += (1.0 - h[i - t])
            grad[i - t] += (-h[i])
    return grad

def compute_subgradient(h, c, top_k=5):
    """Compute subgradient of max of lag values (average of top-k active)"""
    n = len(h)
    # c has length 2n-1, c[k] corresponds to shift s = k - (n-1)
    # Find top-k values of c
    top_indices = np.argsort(c)[-top_k:]
    s_vals = top_indices - (n - 1)
    grad = np.zeros(n)
    for s in s_vals:
        grad += lag_gradient(h, int(s), n)
    grad /= len(s_vals)
    return grad

import time

def get_now():
    return time.perf_counter()

def project_box_sum(y, mass):
    """Project y onto {h: 0 <= h[i] <= 1, sum(h) = mass}"""
    n = len(y)
    y = np.asarray(y, dtype=np.float64)
    lo = y.min() - 2.0

        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > mass:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) * 0.5
    h = np.clip(y - t, 0.0, 1.0)
    # Fix residual
    diff = mass - h.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            room = 1.0 - h
            total_room = room.sum()
            if total_room > 1e-15:
                h += room * (diff / total_room)
        else:
            total = h.sum()
            if total > 1e-15:
                h -= h * (-diff / total)
        # Final clamp and fix
        h = np.clip(h, 0.0, 1.0)
        diff2 = mass - h.sum()
        if abs(diff2) > 1e-15:
            idx = np.arange(n)
            if diff2 > 0:
                for i in idx[np.argsort(1.0 - h)][::-1]:
                    add = min(diff2, 1.0 - h[i])
                    h[i] = min(h[i] + add, 1.0)
                    diff2 -= add
                    if abs(diff2) < 1e-15:
                        break
            else:
                for i in idx[np.argsort(h)]:
                    sub = min(-diff2, h[i])
                    h[i] = max(h[i] - sub, 0.0)
                    diff2 += sub
                    if abs(diff2) < 1e-15:
                        break
    return h

def compute_lags(h):
    """Compute all lag correlation values"""
    c = np.correlate(h, 1.0 - h, mode="full")
    return c

def c5_value(c):
    """Compute C5 from lag values"""
    n = (len(c) + 1) // 2
    dx = 2.0 / n
    return float(c.max() * dx)

def softmax_weights(c, T):
    """Compute softmax weights for the lag values"""
    c_max = c.max()
    exp_c = np.exp(T * (c - c_max))
    return exp_c / exp_c.sum()

def soft_max_grad(h, c, T):
    """Compute gradient of smooth max (1/T)*log(sum exp(T*c_k))"""
    n = len(h)
    w = softmax_weights(c, T)
    # Only process significant weights
    threshold = 1e-8
    grad = np.zeros(n)
    active = w > threshold
    for k in np.where(active)[0]:
        wk = w[k]
        s = k - (n - 1)
        if s >= 0:

                grad[0:n-s] += wk * (1.0 - h[s:n])
                grad[s:n] += wk * (-h[0:n-s])
        else:
            t = -s
            if n - t > 0:
                grad[t:n] += wk * (1.0 - h[0:n-t])
                grad[0:n-t] += wk * (-h[t:n])
    return grad

    current_c5 = c5_value(c)

        return h, best_c5

    best_c5_local = current_c5

        i = rng.integers(0, n)
        j = rng.integers(0, n)

            continue

        # Transfer from i to j: h[i] -= delta, h[j] += delta
        # Feasibility: 0 <= h[i] - delta, h[j] + delta <= 1

        hj = h[j]

            delta = max_delta * frac
            if delta < 1e-12:

            h_try = h.copy()
            h_try[i] -= delta
            h_try[j] += delta

            c5_try = c5_value(c_try)

                h = h_try
                c = c_try
                current_c5 = c5_try
                if c5_try < best_c5_local:
                    best_c5_local = c5_try
                    best_h = h.copy()
                break  # Accept and try next pair
        
    return best_h, min(best_c5_local, best_c5)
