def project_feasible(a):
    """Project a onto {x in [0,1]^n : sum(x) = n/2}."""
    n = len(a)
    target = n / 2.0
    
    # Binary search for lambda
    lo = a.min() - 1.0  # ensures all clipped to 0
    hi = 1.0 - a.max()  # ensures all clipped to 1
    
    for _ in range(200):
        mid = (lo + hi) / 2
        x = np.clip(a + mid, 0, 1)
        s = x.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
        if abs(s - target) < 1e-12:
            break
    
    x = np.clip(a + (lo + hi) / 2, 0, 1)
    
    # Fix residual
    residual = target - x.sum()
    if abs(residual) > 1e-14:
        if residual > 0:
            # Need to add mass: find indices where x < 1
            slack = 1.0 - x
            order = np.argsort(-slack)
            add = residual
            for i in order:
                delta = min(add, slack[i])
                x[i] += delta
                add -= delta
                if add < 1e-15:
                    break
        else:
            # Need to remove mass: find indices where x > 0
            order = np.argsort(x)
            rem = -residual
            for i in order:
                delta = min(rem, x[i])
                x[i] -= delta
                rem -= delta
                if rem < 1e-15:
                    break
    
    return x

def c5_eval(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def project_feasible(a):
    n = len(a)
    target = n / 2.0
    lo = -1.0
    hi = 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        x = np.clip(a + mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    x = np.clip(a + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - x.sum()
    if abs(residual) > 1e-13:
        if residual > 0:
            slack = 1.0 - x
            idx = np.argsort(-slack)
            add = residual
            for i in idx:
                d = min(add, slack[i])
                x[i] += d
                add -= d
                if add < 1e-14:
                    break
        else:
            idx = np.argsort(x)
            rem = -residual
            for i in idx:
                d = min(rem, x[i])
                x[i] -= d
                rem -= d
                if rem < 1e-14:
                    break
    return x

def golden_pair(h, p, q, tau_lo, tau_hi, max_iter=40):
    if tau_hi - tau_lo < 1e-12:
        return 0.0, float('inf')
    n = len(h)
    dx = 2.0 / n
    inv_sqr5_1 = (np.sqrt(5.0) - 1.0) / 2.0  # ~0.618
    a = tau_lo
    b = tau_hi
    c = b - inv_sqr5_1 * (b - a)
    d = a + inv_sqr5_1 * (b - a)
    
    hp = h[p]
    hq = h[q]
    
    h[p] = hp + c
    h[q] = hq - c
    fc = float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))
    
    h[p] = hp + d
    h[q] = hq - d
    fd = float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))
    
    best_tau = c if fc < fd else d
    best_val = fc if fc < fd else fd
    
    for _ in range(max_iter):
        if b - a < 1e-11:
            break
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - inv_sqr5_1 * (b - a)
            h[p] = hp + c
            h[q] = hq - c
            fc = float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))
            if fc < best_val:
                best_val = fc
                best_tau = c
        else:
            a = c
            c = d
            fc = fd
            d = a + inv_sqr5_1 * (b - a)
            h[p] = hp + d
            h[q] = hq - d
            fd = float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))
            if fd < best_val:
                best_val = fd
                best_tau = d
    
    h[p] = hp
    h[q] = hq
    return best_tau, best_val

def pair_descend(h, c_current, max_iter=40, deadline=None):
    n = len(h)
    improved = False
    for p in range(n):
        if deadline is not None and time.monotonic() > deadline:
            break
        for q in range(p + 1, n):
            if deadline is not None and time.monotonic() > deadline:
                break
            tau_lo = max(-h[p], h[q] - 1.0)
            tau_hi = min(1.0 - h[p], h[q])
            if tau_hi - tau_lo < 1e-12:
                continue
            tau, obj = golden_pair(h, p, q, tau_lo, tau_hi, max_iter)
            if obj < c_current - 1e-12:
                h[p] += tau
                h[q] -= tau
                h[p] = np.clip(h[p], 0.0, 1.0)
                h[q] = np.clip(h[q], 0.0, 1.0)
                # Fix sum
                target = n / 2.0
                residual = target - h.sum()
                if abs(residual) > 1e-12:
                    # Distribute among free variables
                    # Simple: add to p and q proportionally
                    slack_p = 1.0 - h[p] if residual > 0 else h[p]
                    slack_q = 1.0 - h[q] if residual > 0 else h[q]
                    total_slack = slack_p + slack_q
                    if total_slack > 1e-15:
                        h[p] += residual * slack_p / total_slack
                        h[q] += residual * slack_q / total_slack
                    h = project_feasible(h)
                c_current = obj
                improved = True
    return h, c_current, improved

def interpolate_h(h_old, n_old, n_new):
    """Interpolate piecewise constant function from n_old to n_new samples."""
    dx_old = 2.0 / n_old
    dx_new = 2.0 / n_new
    h_new = np.zeros(n_new)
    for j in range(n_new):
        # New interval: [j*dx_new, (j+1)*dx_new)
        start = j * dx_new
        end = (j + 1) * dx_new
        # Which old intervals overlap?
        i_start = int(start / dx_old)
        i_end = int((end - 1e-15) / dx_old)
        if i_start >= n_old:
            i_start = n_old - 1
        if i_end >= n_old:
            i_end = n_old - 1
        # Compute weighted average
        weight_sum = 0.0
        val_sum = 0.0
        for i in range(i_start, i_end + 1):
            # Overlap of [i*dx_old, (i+1)*dx_old) and [start, end)
            ov_start = max(i * dx_old, start)
            ov_end = min((i + 1) * dx_old, end)
            if ov_end > ov_start:
                w = ov_end - ov_start
                weight_sum += w
                val_sum += w * h_old[i]
        if weight_sum > 0:
            h_new[j] = val_sum / weight_sum
        else:
            h_new[j] = 0.5  # fallback
    return h_new

            rem = -residual
            for i in idx:
                d = min(rem, x[i])
                if d <= 0:
                    break
                x[i] -= d
                rem -= d
                if rem < 1e-14:
                    break
    return x


def golden_pair(h, p, q, tau_lo, tau_hi, max_iter=35):
    if tau_hi - tau_lo < 1e-12:
        return 0.0, float('inf')
    n = len(h)
    dx = 2.0 / n
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a = tau_lo
    b = tau_hi
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    hp = h[p]
    hq = h[q]
    h[p] = hp + c
    h[q] = hq - c
    fc = float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))
    h[p] = hp + d
    h[q] = hq - d
    fd = float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))
    best_tau = c if fc < fd else d
    best_val = fc if fc < fd else fd
    for _ in range(max_iter):
        if b - a < 1e-11:
            break
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            h[p] = hp + c
            h[q] = hq - c
            fc = float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))
            if fc < best_val:
                best_val = fc
                best_tau = c
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            h[p] = hp + d
            h[q] = hq - d
            fd = float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))
            if fd < best_val:
                best_val = fd
                best_tau = d
    h[p] = hp
    h[q] = hq
    return best_tau, best_val


def pair_descend(h, c_current, max_iter=35, deadline=None):
    n = len(h)
    improved_any = False
    for p in range(n):
        if deadline is not None and time.monotonic() > deadline:
            break
        for q in range(p + 1, n):
            if deadline is not None and time.monotonic() > deadline:
                break
            tau_lo = max(-h[p], h[q] - 1.0)
            tau_hi = min(1.0 - h[p], h[q])
            if tau_hi - tau_lo < 1e-12:
                continue
            tau, obj = golden_pair(h, p, q, tau_lo, tau_hi, max_iter)
            if obj < c_current - 1e-13:
                h[p] += tau
                h[q] -= tau
                h[p] = np.clip(h[p], 0.0, 1.0)
                h[q] = np.clip(h[q], 0.0, 1.0)
                target = n / 2.0
                res = target - h.sum()
                if abs(res) > 1e-10:
                    h += res / n
                    h = np.clip(h, 0.0, 1.0)
                    h = project_feasible(h)
                c_current = obj
                improved_any = True
    return h, c_current, improved_any


def interpolate_h(h_old, n_old, n_new):
    dx_old = 2.0 / n_old
    dx_new = 2.0 / n_new
    h_new = np.zeros(n_new)
    for j in range(n_new):
        start = j * dx_new
        end = (j + 1) * dx_new
        i_start = int(start / dx_old)
        i_end = int((end - 1e-14) / dx_old)
        if i_start >= n_old:
            i_start = n_old - 1
        if i_end >= n_old:
            i_end = n_old - 1
        if i_start < 0:
            i_start = 0
        wsum = 0.0
        vsum = 0.0
        for i in range(i_start, i_end + 1):
            ov_s = max(i * dx_old, start)
            ov_e = min((i + 1) * dx_old, end)
            if ov_e > ov_s:
                w = ov_e - ov_s
                wsum += w
                vsum += w * h_old[i]
        h_new[j] = vsum / wsum if wsum > 0 else 0.5
    return h_new


def random_feasible(n, rng):
    h = rng.uniform(0, 1, size=n)
    return project_feasible(h)


def binary_feasible(n, rng):
    target = int(n / 2)
    h = np.zeros(n)
    h[rng.choice(n, size=target, replace=False)] = 1.0
    return h


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + budget_s

    try:
        h_init = np.array(initial_h_values, dtype=np.float64).copy()
    except Exception:
        h_init = np.full(45, 0.5)

    n_best = len(h_init)
    c_best = c5_eval(h_best_init := h_init)
    h_best = h_init.copy()

    h_work = h_init.copy()
    c_work = c_best
    max_iter = 35
    no_improve = 0
    max_no_improve = 8
    perturbation_magnitude = 0.08
    pass_count = 0

    while time.monotonic() < deadline - 10.0:
        t_pass_start = time.monotonic()
        h_work, c_work, improved = pair_descend(h_work, c_work, max_iter, deadline - 2.0)
        pass_count += 1

        if c_work < c_best - 1e-11:
            c_best = c_work
            h_best = h_work.copy()
            n_best = len(h_best)
            no_improve = 0
            if max_iter < 45:
                max_iter += 1
        else:
            no_improve += 1
            max_iter = max(20, max_iter - 1)

        if no_improve >= max_no_improve:
            roll = rng.random()
            if roll < 0.3:
                mag = perturbation_magnitude * (0.5 + rng.random())
                h_work = h_best + rng.uniform(-mag, mag, size=n_best)
                h_work = project_feasible(h_work)
            elif roll < 0.5:
                h_work = binary_feasible(n_best, rng)
            elif roll < 0.7:
                h_work = random_feasible(n_best, rng)
            else:
                h_sorted = h_best.copy()
                h_sorted.sort()
                shift = int(rng.integers(1, n_best))
                h_work = np.roll(h_sorted, shift)
                h_work = project_feasible(h_work)
            c_work = c5_eval(h_work)
            no_improve = 0
            perturbation_magnitude = max(0.02, perturbation_magnitude * 0.9)

        if pass_count % 50 == 0 and time.monotonic() < deadline - 60.0:
            n_try = n_best + int(rng.choice([-10, -5, 5, 10, 15, 20]))
            if 20 <= n_try <= 200 and n_try != n_best:
                h_try = interpolate_h(h_best, n_best, n_try)
                h_try = project_feasible(h_try)
                c_try = c5_eval(h_try)
                for _ in range(3):
                    h_try, c_try, _ = pair_descend(h_try, c_try, 25, deadline - 5.0)
                if c_try < c_best - 1e-6:
                    c_best = c_try
                    h_best = h_try.copy()
                    n_best = n_try
                    h_work = h_best.copy()
                    c_work = c_best
                    no_improve = 0

    # Final refinement on best
    h_final = h_best.copy()
    c_final = c_best
    for _ in range(5):
        if time.monotonic() > deadline - 2.0:
            break
        h_final, c_final, _ = pair_descend(h_final, c_final, 40, deadline - 1.0)
    if c_final < c_best:
        c_best = c_final
        h_best = h_final

    # Validate
    h_best = np.clip(h_best, 0.0, 1.0)
    h_best = project_feasible(h_best)
    c_best = c5_eval(h_best)
    n_best = len(h_best)

    return h_best, c_best, n_best
