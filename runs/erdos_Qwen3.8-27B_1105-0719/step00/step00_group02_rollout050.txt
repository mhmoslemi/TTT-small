import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_box_sum(y, s):
    """Euclidean projection onto {x: 0 <= x_i <= 1, sum(x) = s}"""
    lo, hi = -2.0, 3.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        total = np.sum(np.clip(y - mid, 0.0, 1.0))
        if total > s:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    x = np.clip(y - lam, 0.0, 1.0)
    res = s - x.sum()
    if abs(res) > 1e-14:
        if res > 0:
            slack = 1.0 - x
            ss = slack.sum()
            if ss > 1e-14:
                x += res * slack / ss
        else:
            ss = x.sum()
            if ss > 1e-14:
                x += res * x / ss
        x = np.clip(x, 0.0, 1.0)
    return x

def compute_lse_gradient(h, tau):
    """Gradient of LSE(h)/tau with respect to h, where LSE = tau*log(sum(exp(R/tau)))"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    R = corr * dx
    R_max = np.max(R)
    exp_vals = np.exp((R - R_max) / tau)
    w = exp_vals / exp_vals.sum()
    
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        p = idx - (n - 1)
        wi = w[idx]
        if wi < 1e-12:
            continue
        if p >= 0:
            k_start = 0
            k_end = n - p
        else:
            k_start = -p
            k_end = n - p
        for k in range(k_start, k_end):
            grad[k] += wi * (1.0 - h[k + p])
            grad[k + p] -= wi * h[k]
    return dx * grad

def compute_topk_gradient(h, k):
    """Gradient using only top-k lags with uniform weights"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    R = corr * dx
    # Get indices of top-k lags
    top_idx = np.argsort(R)[-k:]
    w = np.zeros(2 * n - 1)
    w[top_idx] = 1.0 / k
    
    grad = np.zeros(n)
    for idx in top_idx:
        p = idx - (n - 1)
        if p >= 0:
            k_start = 0
            k_end = n - p
        else:
            k_start = -p
            k_end = n - p
        for i in range(k_start, k_end):
            grad[i] += (1.0 / k) * (1.0 - h[i + p])
            grad[i + p] -= (1.0 / k) * h[i]
    return dx * grad

    """Smooth gradient descent with adaptive temperature"""
    n = len(h)
    s = n / 2.0
    start_time = time.time()
    current = h.copy()
    current_c = exact_score(current)
    best = current.copy()
    best_c = current_c
    
    if initial_tau is None:
        initial_tau = 0.02
    
    tau = initial_tau
    step = 0.05
    stall_count = 0
    
    while time.time() - start_time < time_limit:
        # Compute gradient
        grad = compute_lse_gradient(current, tau)
        
        # Remove mean to stay in sum=0 subspace approximately
        gradcentered = grad - grad.mean()
        
        # Try several step sizes
        best_candidate = None
        best_candidate_c = current_c
        for alpha in [step, step*0.5, step*0.1, step*2]:
            candidate = current + alpha * gradcentered
            projected = project_box_sum(candidate, s)
            c = exact_score(projected)
            if c < best_candidate_c - 1e-12:
                best_candidate_c = c
                best_candidate = projected
        
        if best_candidate is not None:
            current = best_candidate
            current_c = best_candidate_c
            step = min(step * 1.5, 0.5)
            stall_count = 0
            if current_c < best_c:
                best_c = current_c
                best = current.copy()
        else:
            stall_count += 1
            step *= 0.5
            tau *= 0.8
            if step < 1e-6 or tau < 1e-5:
                break
    
    return best, best_c

def pair_transfer_search(h, time_limit, rng):

    n = len(h)

    best = current.copy()
    best_c = current_c

    for i in range(n):

    eval_count = 0
    passes = 0
    
    while time.time() - start_time < time_limit:
        passes += 1
        # Shuffle pairs
        order = rng.permutation(len(pairs))

            if (eval_count % 200 == 0) and (time.time() - start_time > time_limit * 0.95):
                break
            i, j = pairs[idx]
            
            # Try transferring mass from i to j and j to i
            for (donor, recipient) in [(i, j), (j, i)]:

                if cap < 1e-10:
                    continue
                
                # Try several fractions of cap
                for frac in [1.0, 0.7, 0.5, 0.3, 0.15, 0.05]:
                    delta = frac * cap
                    if delta < 1e-10:
                        continue
                    candidate = current.copy()
                    candidate[donor] -= delta
                    candidate[recipient] += delta
                    c = exact_score(candidate)
                    eval_count += 1
                    if c < current_c - 1e-12:
                        current = candidate
                        current_c = c
                        improved = True
                        if current_c < best_c:
                            best_c = current_c
                            best = current.copy()
                        break  # Move to next pair
        
        if not improved:
            break
    
    return best, best_c

def sa_search(h, time_limit, rng):
    """Simulated annealing with pair moves"""
    n = len(h)
    s = n / 2.0
    start_time = time.time()
    current = h.copy()
    current_c = exact_score(current)
    best = current.copy()
    best_c = current_c
    
    temp = 0.01
    delta_scale = 0.1
    accept = 0
    total = 0
    
    while time.time() - start_time < time_limit:
        # Random pair
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        
        # Random delta
        max_d = min(current[i], 1.0 - current[j]) if current[i] > current[j] else 0
        max_d_rev = min(current[j], 1.0 - current[i])
        
        if rng.random() < 0.5:
            if max_d < 1e-10:
                continue
            delta = rng.random() * max_d * delta_scale
            candidate = current.copy()
            candidate[i] -= delta
            candidate[j] += delta
        else:
            if max_d_rev < 1e-10:
                continue
            delta = rng.random() * max_d_rev * delta_scale
            candidate = current.copy()
            candidate[j] -= delta
            candidate[i] += delta
        
        c = exact_score(candidate)
        total += 1
        
        if c <= current_c:
            current = candidate
            current_c = c
            accept += 1
            if c < best_c:
                best_c = c
                best = current.copy()
        else:
            diff = c - current_c
            if rng.random() < np.exp(-diff / max(temp, 1e-8)):
                current = candidate
                current_c = c
                accept += 1
        
        # Adapt temperature
        if total % 500 == 0:
            rate = accept / max(total - 500, 1)
            if rate < 0.3:
                delta_scale *= 0.7
            elif rate > 0.7:
                delta_scale *= 1.3
            temp *= 0.98
            accept = 0
    
    return best, best_c

    best_c = exact_score(best_h)

    return best_h.copy(), best_c, n
