        c5_cand = _max_overlap(h_cand)
        return h_cand, c5_cand
    return None, None

def run(seed=42, budget_s=1100, **kwargs):
    global _g_n, _g_dx, _g_target_sum, _g_n_lags
    
    start_time = time.time()
    
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = np.ones(63) * 0.5
    
    n = len(h0)
    _g_n = n
    _g_dx = 2.0 / n
    _g_target_sum = n / 2.0
    _g_n_lags = 2 * n - 1
    
    h0 = np.clip(h0, 0.0, 1.0)
    h0 = _fix_sum(h0, n)
    
    best_h = h0.copy()
    best_c5 = _max_overlap(h0)
    
    def elapsed():
        return time.time() - start_time
    
    # Phase 1: SLSQP with reduced constraints (faster)
    if elapsed() < budget_s * 0.3:
        h_c, c5_c = _try_slsqp(best_h, top_k=30, maxiter=300, ftol=1e-12)
        if h_c is not None and c5_c < best_c5:
            best_h = h_c
            best_c5 = c5_c
    
    # Phase 2: L-BFGS-B with smooth max
    for tau, pen in [(20.0, 100.0), (50.0, 200.0), (100.0, 500.0)]:
        if elapsed() > budget_s * 0.6:
            break
        h_c, c5_c = _try_lbfgsb(best_h, tau=tau, penalty=pen, maxiter=3000)
        if h_c is not None and c5_c < best_c5:
            best_h = h_c
            best_c5 = c5_c
    
    # Phase 3: SLSQP full refinement
    if elapsed() < budget_s * 0.7:
        h_c, c5_c = _try_slsqp(best_h, top_k=None, maxiter=400, ftol=1e-13)
        if h_c is not None and c5_c < best_c5:
            best_h = h_c
            best_c5 = c5_c
    
    # Phase 4: More L-BFGS-B with higher tau
    for tau, pen in [(150.0, 500.0), (300.0, 1000.0)]:
        if elapsed() > budget_s * 0.8:
            break
        h_c, c5_c = _try_lbfgsb(best_h, tau=tau, penalty=pen, maxiter=5000)
        if h_c is not None and c5_c < best_c5:
            best_h = h_c
            best_c5 = c5_c
    
    # Phase 5: Random restarts
    rng = np.random.RandomState(seed)
    n_restarts = 10
    for i in range(n_restarts):
        if elapsed() > budget_s * 0.93:
            break
        
        if i == 0:
            h_start = best_h.copy()
        else:
            scale = 0.02 + 0.04 * (i % 4)
            h_start = best_h + rng.randn(n) * scale
            h_start = np.clip(h_start, 0.0, 1.0)
            h_start = _fix_sum(h_start, n)
        
        # L-BFGS-B quick pass
        h_c, c5_c = _try_lbfgsb(h_start, tau=50.0, penalty=200.0, maxiter=1500)
        if h_c is not None and c5_c < best_c5:
            best_h = h_c
            best_c5 = c5_c
            # SLSQP refinement if improved
            if elapsed() < budget_s * 0.93:
                h_r, c5_r = _try_slsqp(best_h, top_k=30, maxiter=200, ftol=1e-12)
                if h_r is not None and c5_r < best_c5:
                    best_h = h_r
                    best_c5 = c5_r
    
    # Phase 6: Final SLSQP polish
    if elapsed() < budget_s * 0.97:
        h_c, c5_c = _try_slsqp(best_h, top_k=None, maxiter=300, ftol=1e-14)
        if h_c is not None and c5_c < best_c5:
            best_h = h_c
            best_c5 = c5_c
    
    # Final verification
    try:
        val = evaluate_erdos_solution(best_h)
        if val is not None and val < best_c5:
            best_c5 = val
    except Exception:
        pass
    
    return best_h, best_c5, n
