def exact_C(h):
    n = len(h)
    if n == 0:
        return float('inf')
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(u):
    u = np.array(u, dtype=float)
    n = len(u)
    T = n / 2.0
    lo = -1.0 - np.max(u) - 1.0
    hi = 1.0 - np.min(u) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(u + mid, 0.0, 1.0)) - T > 0:
            hi = mid
        else:
            lo = mid
    h = np.clip(u + (lo + hi) / 2.0, 0.0, 1.0)
    residual = T - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if residual > 0:
                amt = min(residual, 1.0 - h[idx])
                h[idx] += amt
                residual -= amt
            else:
                amt = min(-residual, h[idx])
                h[idx] -= amt
                residual += amt
            if abs(residual) < 1e-14:
                break
    return h

def stable_sigmoid(t):
    out = np.empty_like(t, dtype=float)
    pos = t >= 0
    out[pos] = 1.0 / (1.0 + np.exp(-t[pos]))
    et = np.exp(t[~pos])
    out[~pos] = et / (1.0 + et)
    return out

def feasible_from_logit(y):
    n = len(y)
    T = n / 2.0
    lo = -np.max(y) - 80.0
    hi = -np.min(y) + 80.0

        else:

def logit_from_h(h, eps=1e-7):

def smooth_obj_grad(y, beta):
    h = feasible_from_logit(y)
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    v = corr * dx
    vmax = np.max(v)
    if not np.isfinite(vmax):
        return 1e6, np.zeros_like(y)
    ev = np.exp(beta * (v - vmax))
    Z = np.sum(ev)
    if not np.isfinite(Z) or Z <= 0:
        w = np.zeros(len(v))
        w[np.argmax(v)] = 1.0
    else:
        w = ev / Z
    F = vmax + np.log(max(Z, 1e-300)) / beta
    G = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-30:
            continue
        lag = k - (n - 1)
        if lag == 0:
            G += wk * dx * (1.0 - 2.0 * h)
        elif lag > 0:
            m = lag
            G[m:] += wk * dx * (1.0 - h[:n - m])
            G[:n - m] -= wk * dx * h[m:]
        else:
            m = -lag
            G[:n - m] += wk * dx * (1.0 - h[m:])
            G[m:] -= wk * dx * h[:n - m]
