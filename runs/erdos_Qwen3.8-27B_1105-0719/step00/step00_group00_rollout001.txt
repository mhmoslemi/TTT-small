                upd(h_pg2, n)

    # Final refinement on best solution
    if best_h is not None and not done():
        n = best_n
        dx = 2.0 / n
        ts = n / 2.0
        h_ref = best_h + rng.normal(0, 0.03, n)
        h_ref = project_to_constraint(h_ref, ts)
        cons = [{'type': 'eq', 'fun': lambda x: x.sum() - ts,
                 'jac': lambda x: np.ones(n)}]
        try:
            res = minimize(c5_value, h_ref, args=(dx,), method='SLSQP',
                           bounds=[(0.0, 1.0)] * n, constraints=cons,
                           options={'maxiter': 500, 'ftol': 1e-14})
            upd(project_to_constraint(res.x, ts), n)
        except Exception:
            pass
        h_pg_final, _ = sorted_projected_gradient(best_h, dx, ts, n, 3000, 50.0, done)
        upd(h_pg_final, n)

    return best_h, best_c5, best_n
