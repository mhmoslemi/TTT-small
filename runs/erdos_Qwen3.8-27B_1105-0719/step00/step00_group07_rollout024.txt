import numpy as np
from scipy.optimize import minimize, brentq
import time

def _sigmoid(x):
    x = np.asarray(x, dtype=float)
    x = np.clip(x, -500, 500)
    out = np.empty_like(x)
    pos = x >= 0
    out[pos] = 1.0 / (1.0 + np.exp(-x[pos]))
    ex = np.exp(x[~pos])
    out[~pos] = ex / (1.0 + ex)
    return out

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def solve_c(theta, target):
    def f(c):
        return float(np.sum(_sigmoid(theta + c)) - target)
    mid = -float(np.mean(theta))
    lo, hi = mid - 40.0, mid + 40.0
    for _ in range(8):
        if f(lo) * f(hi) < 0:
            break
        lo -= 40.0
        hi += 40.0
    else:
        return mid
    try:
        return brentq(f, lo, hi, xtol=1e-13, maxiter=200)
    except Exception:
        return mid

def theta_to_h(theta, target):
    c = solve_c(theta, target)
    return _sigmoid(theta + c)

def h_to_theta(h):
    eps = 1e-12
    h = np.clip(h, eps, 1.0 - eps)
    return np.log(h / (1.0 - h))

def lse_pgrad(h, beta):
    n = len(h)
    R = np.correlate(h, 1.0 - h, mode='full')
    Rmax = float(np.max(R))
    shifted = np.maximum(beta * (R - Rmax), -700.0)
    exp_R = np.exp(shifted)
    Z = float(np.sum(exp_R))
    if Z < 1e-300:
        w = np.zeros_like(R)
        w[np.argmax(R)] = 1.0
    else:
        w = exp_R / Z
    lse_val = Rmax + np.log(Z) / beta
    grad = np.zeros(n)
    for t in range(2 * n - 1):
        wt = w[t]
        if wt < 1e-15:
            continue
        k = t - (n - 1)
        wl = wt / beta
        if k >= 0:
            m = n - k
            if m > 0:
                grad[k:n] += wl * (1.0 - h[:m])
                grad[:m] -= wl * h[k:n]
        else:
            q = -k
            m = n - q
            if m > 0:
                grad[:m] += wl * (1.0 - h[q:q+m])
                grad[q:n] -= wl * h[:m]
    return lse_val, grad

def lse_theta_obj_grad(theta, target, beta):
    h = theta_to_h(theta, target)
    val, pgrad = lse_pgrad(h, beta)
    s = h * (1.0 - h)
    S = float(np.sum(s))
    if S < 1e-15:
        return val, np.zeros_like(theta)
    mean_term = float(np.dot(s, pgrad)) / S
    theta_grad = s * (pgrad - mean_term)
    return val, theta_grad

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - float(np.sum(h))
        if abs(diff) < 1e-13:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if not np.any(free):
            break
        n_free = int(np.sum(free))
        adj = diff / n_free
        h[free] += adj
        h = np.clip(h, 0.0, 1.0)
    diff = target_sum - float(np.sum(h))
    if abs(diff) > 1e-14:
        for i in range(len(h)):
            if diff > 0 and h[i] < 1.0:
                amt = min(diff, 1.0 - h[i])
                h[i] += amt
                diff -= amt
            elif diff < 0 and h[i] > 0.0:
                amt = min(-diff, h[i])
                h[i] -= amt
                diff += amt
            if abs(diff) < 1e-14:
                break
    return h

def interpolate_h(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_src = (np.arange(n_src) + 0.5) / n_src * 2.0
    x_dst = (np.arange(n_dst) + 0.5) / n_dst * 2.0
    return np.interp(x_dst, x_src, h_src)

def coordinate_descent_exact(h, target, max_iters=5000, timeout=30.0, t0=None, seed=42):
    if t0 is None:
        t0 = time.time()
    n = len(h)
    h = h.astype(float).copy()
    best_c5 = eval_c5(h)
    rng = np.random.default_rng(seed)
    sizes = np.array([0.4, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005])
    for it in range(max_iters):
        if time.time() - t0 > timeout:
            break
        j = int(rng.integers(n))
        k = int(rng.integers(n))
        if j == k:
            k = (j + 1) % n
        improved = False
        for delta in sizes:
            if h[k] >= delta and h[j] + delta <= 1.0:
                h[j] += delta
                h[k] -= delta
                c5_new = eval_c5(h)
                if c5_new < best_c5 - 1e-14:
                    best_c5 = c5_new
                    improved = True
                    break
                h[j] -= delta
                h[k] += delta
            if h[j] >= delta and h[k] + delta <= 1.0:
                h[j] -= delta
                h[k] += delta
                c5_new = eval_c5(h)
                if c5_new < best_c5 - 1e-14:
                    best_c5 = c5_new
                    improved = True
                    break
                h[j] += delta
                h[k] -= delta
    return h, best_c5

def active_set_descent(h, target, timeout=30.0, t0=None, seed=99):
    if t0 is None:
        t0 = time.time()
    n = len(h)
    h = h.astype(float).copy()
    best_c5 = eval_c5(h)
    rng = np.random.default_rng(seed)
    sizes = [0.3, 0.15, 0.08, 0.04, 0.02, 0.01, 0.005, 0.002]
    stagnation = 0
    max_stagnation = 5000
    for it in range(max_stagnation):
        if time.time() - t0 > timeout:
            break
        R = np.correlate(h, 1.0 - h, mode='full')
        Rmax = float(np.max(R))
        active = R > Rmax - max(1e-8, 0.01 * Rmax)
        n_active = int(np.sum(active))
        if n_active == 0:
            break
        # Compute weighted gradient of active rows
        w = R[active] / Rmax
        grad = np.zeros(n)
        idx_active = np.where(active)[0]
        for idx in idx_active:
            k = idx - (n - 1)
            wt = w[idx]
            if k >= 0:
                m = n - k
                if m > 0:
                    grad[k:n] += wt * (1.0 - h[:m])
                    grad[:m] -= wt * h[k:n]
            else:
                q = -k
                m = n - q
                if m > 0:
                    grad[:m] += wt * (1.0 - h[q:q+m])
                    grad[q:n] -= wt * h[:m]
        # Project to zero-sum direction
        grad = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad_dir = -grad / gnorm
        # Try to move along this direction (negate since we want to decrease)
        improved = False
        for scale in sizes:
            h_trial = h + scale * grad_dir
            h_trial = project_feasible(h_trial, target)
            c5_trial = eval_c5(h_trial)
            if c5_trial < best_c5 - 1e-13:
                h = h_trial
                best_c5 = c5_trial
                improved = True
                break
        if not improved:
            # Random perturbation to escape
            perturb = rng.normal(0, 0.05, n)
            h_trial = project_feasible(h + perturb, target)
            c5_trial = eval_c5(h_trial)
            if c5_trial < best_c5 + 0.001:
                h = h_trial
                if c5_trial < best_c5:
                    best_c5 = c5_trial
            stagnation += 1
        else:
            stagnation = 0
        if stagnation > 3000:
            break
    return h, best_c5

def lse_optimize(h, target, betas, maxiter_per_beta=150, timeout=60.0, t0=None):
    if t0 is None:
        t0 = time.time()
    n = len(h)
    theta = h_to_theta(h)
    best_h = h.copy()
    best_c5 = eval_c5(h)
    for beta in betas:
        if time.time() - t0 > timeout:
            break
        try:
            result = minimize(
                lse_theta_obj_grad, theta,
                args=(target, beta),
                method='BFGS', jac=True,
                options={'maxiter': maxiter_per_beta, 'gtol': 1e-9}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                theta = result.x
                h_new = theta_to_h(theta, target)
                h_new = project_feasible(h_new, target)
                c5_new = eval_c5(h_new)
                if c5_new < best_c5 - 1e-14:
                    best_c5 = c5_new
                    best_h = h_new.copy()
        except Exception:
            pass
    return best_h, best_c5
