```python
import numpy as np
import time

def compute_objective(h, dx):
    c = np.correlate(h, 1-h, mode="full") * dx
    return np.max(c), c

def compute_subgradient(h, dx, active_lags):
    n = len(h)
    sg = np.zeros(n)
    for k in active_lags:
        # k is the lag (can be negative)
        # ∂c[k]/∂h[j] = (1-h[j-k])*[0≤j-k<n] - h[j+k]*[0≤j+k<n]
        # Term 1: for each j, (1-h[j-k]) if 0 ≤ j-k < n
        #   j-k in [0, n-1] means j in [k, k+n-1], intersected with [0, n-1]
        # Term 2: for each j, -h[j+k] if 0 ≤ j+k < n
        #   j+k in [0, n-1] means j in [-k, -k+n-1], intersected with [0, n-1]
        
        for j in range(n):
            contrib = 0.0
            if 0 <= j - k < n:
                contrib += (1.0 - h[j - k])
            if 0 <= j + k < n:
                contrib -= h[j + k]
            sg[j] += dx * contrib
    return sg / len(active_lags)
```

```python
def project_feasible(h, target_sum):
    n = len(h)
    # Find λ such that sum(clip(h - λ, 0, 1)) = target_sum
    lo, hi = h.min() - target_sum/n - 1, h.max() + target_sum/n + 1
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h - mid, 0, 1))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo+hi)/2, 0, 1)
```

```python
def sg_at_lag(h, dx, k):
    n = len(h)
    sg = np.zeros(n)
    # term1: (1-h[j-k]) for valid j-k
    # j-k in [0, n-1] => j in [k, k+n-1] intersect [0, n-1]
    lo1 = max(0, k)

        # j-k ranges from lo1-k to hi1-1-k
        idx = np.arange(lo1, hi1)
        src = idx - k
        sg[lo1:hi1] += (1.0 - h[src])
    # term2: -h[j+k] for valid j+k

        idx = np.arange(lo2, hi2)
        src = idx + k
        sg[lo2:hi2] -= h[src]
```

```python
import numpy as np
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def obj_val(h, dx):
    return float(np.max(compute_corr(h))) * dx

def lse_objective(h, dx, eps):
    corr = compute_corr(h) * dx
    m = np.max(corr)
    s = np.sum(np.exp((corr - m) / eps))
    return m + eps * np.log(s)

def lse_gradient(h, dx, eps):
    n = len(h)
    corr = compute_corr(h) * dx
    exps = np.exp((corr - np.max(corr)) / eps)
    w = exps / np.sum(exps)
    grad = np.zeros(n)
    for pos in range(2 * n - 1):
        k = pos - (n - 1)
        wi = w[pos] * dx
        lo = max(0, k)
        hi = min(n, n + k)
        if lo < hi:
            grad[lo:hi] += wi * (1.0 - h[np.arange(lo, hi) - k])
        lo2 = max(0, -k)
        hi2 = min(n, n - k)
        if lo2 < hi2:
            grad[lo2:hi2] -= wi * h[np.arange(lo2, hi2) + k]
    return grad

def proj(h, target):
    lo, hi = -1.5, 1.5

        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

    bh = h.copy()
    bc = obj_val(h, dx)

    step = 0.02
    mom = 0.85
    v = np.zeros(n)
    eps_idx = 0
    stall = 0
    
    while time.time() - t0 < t_limit:
        eps = eps_candidates[eps_idx]
        obj_lse = lse_objective(h, dx, eps)
        grad = lse_gradient(h, dx, eps)
        
        v = mom * v - step * grad
        nh = proj(h + v, target)

            v = np.zeros(n)
            continue

        if real_obj < bc:
            bc = real_obj
            bh = nh.copy()
            h = nh
            stall = 0
            step = min(step * 1.05, 0.05)
        else:
            stall += 1
            if stall > 50:
                step *= 0.7
                v = np.zeros(n)
                if step < 1e-6:
                    eps_idx = min(eps_idx + 1, len(eps_candidates) - 1)
                    step = 0.02
                    stall = 0
                    if eps_idx >= len(eps_candidates) - 1 and step < 1e-6:
                        break
            h = proj(h, target)  # reset to last feasible
    
    return bh, bc

def pair_refine(h0, dx, target, t_limit=30.0):
    n = len(h0)
    t0 = time.time()
    h = h0.astype(float).copy()
    bh = h.copy()

    cnt = int(target)
    h[idx[:cnt]] = 1.0
    frac = target - cnt
    if 0 < frac < 1:

    return proj(h, target)
```
