from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

    """Compute C5 value directly"""
    return compute_c5(h)

def project_feasible(h):
    """Project h to satisfy sum(h) = n/2 and 0 <= h <= 1"""
    n = len(h)
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        # Find coordinates that can absorb the adjustment
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if not np.any(mask):
            break
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    # Final adjustment on the largest slack coordinate
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        if diff > 0:
            idx = np.argmax(1.0 - h)
        else:
            idx = np.argmin(h)
        h[idx] += diff
        h[idx] = np.clip(h[idx], 0.0, 1.0)
    return h

    x = theta + c
    # Stable sigmoid
    h = np.empty_like(x)
    pos = x >= 0
    h[pos] = 1.0 / (1.0 + np.exp(-x[pos]))
    ex = np.exp(x[~pos])
    h[~pos] = ex / (1.0 + ex)
    return h

    """Compute raw correlation values (unscaled)"""
    return np.correlate(h, 1.0 - h, mode="full")

def lse_value_and_raw_weights(raw, beta):
    """Compute LSE value and softmax weights over raw values"""
    M = np.max(raw)
    shifted = beta * (raw - M)
    # Clip for numerical stability
    shifted = np.clip(shifted, -700, 700)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    lse_value = M + np.log(np.sum(exp_vals)) / beta
    return lse_value, w

    Compute dR_t/dh where t is the correlation index (0 to 2n-2)
    and t corresponds to lag l = t - (n-1)
    """

        v += w[t] * row_gradient_h(t, h)
    
    # Project: grad_theta_j = s_j * (v_j - s_j * sum(s*v) / Q)
    Sv = np.dot(s, v)
    grad = s * (v - s * Sv / Q)
    
    return lse_val, grad

    try:

        if not np.all(np.isfinite(grad)):
            return val, np.zeros_like(theta)
        return val, grad
    except Exception:
        return 1e6, np.zeros_like(theta)

    M = np.max(raw)
    shifted = np.clip(beta * (raw - M), -700, 700)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    lse_val = M + np.log(np.sum(exp_vals)) / beta
    
    # Gradient computation
    s = h * (1.0 - h)
    Q = np.sum(s)
    if Q < 1e-15:
        return lse_val, np.zeros(n)

    num_rows = len(raw)

        if wt < 1e-12:

            v[:m] += wt * (1.0 - h[l:])
            v[l:] += wt * (-h[:m])

            q = -l
            m = n - q
            v[q:] += wt * (1.0 - h[:m])
            v[:m] += wt * (-h[q:])
    
    # Projected gradient
    Sv = np.dot(s, v)
    grad = s * (v - s * (Sv / Q))
    
    return lse_val, grad

from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h):
    n = len(h)
    target = n / 2.0
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not np.any(mask):
            break
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        idx = int(np.argmax(1.0 - h)) if diff > 0 else int(np.argmin(h))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)

def sigmoid(x):
    out = np.empty_like(x, dtype=float)
    pos = x >= 0
    out[pos] = 1.0 / (1.0 + np.exp(-x[pos]))
    ex = np.exp(x[~pos])
    out[~pos] = ex / (1.0 + ex)
    return out

    c = 0.0
    for _ in range(80):
        h = sigmoid(theta + c)
        s = h * (1.0 - h)
        f = np.sum(h) - target
        fp = np.sum(s)
        if fp < 1e-30:
            break
        step = f / fp
        c_new = c - step
        if abs(c_new - c) > 100:
            c_new = c + np.sign(-step) * min(abs(step), 10)
        c = c_new
        if abs(step) < 1e-15:

    h = sigmoid(theta + c)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:

        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)

def theta_from_h(h):
    eps = 1e-10
    hc = np.clip(h, eps, 1.0 - eps)
    theta = np.log(hc / (1.0 - hc))
    return theta - np.mean(theta)

def lse_objective_grad(theta, beta):
    n = len(theta)
    h = h_from_theta(theta)
    raw = np.correlate(h, 1.0 - h, mode="full")
    M = np.max(raw)
    shifted = np.clip(beta * (raw - M), -700.0, 700.0)
    exp_v = np.exp(shifted)
    total = np.sum(exp_v)
    w = exp_v / total
    lse_val = M + np.log(total) / beta
    s = h * (1.0 - h)
    Q = np.sum(s)
    if Q < 1e-15:
        return lse_val, np.zeros(n)
    v = np.zeros(n)
    for t in range(len(raw)):
        wt = w[t]
        if wt < 1e-12:
            continue
        l = t - (n - 1)
        if l >= 0:
            m = n - l
            v[:m] += wt * (1.0 - h[l:])
            v[l:] += -wt * h[:m]
        else:
            q = -l
            m = n - q
            v[q:] += wt * (1.0 - h[:m])
            v[:m] += -wt * h[q:]
    Sv = np.dot(s, v)
    grad = s * (v - s * (Sv / Q))
    return lse_val, grad

def _min_fg(theta, beta):
    val, grad = lse_objective_grad(theta, beta)
    if not np.isfinite(val):
        return 1e6, np.zeros_like(theta)
    if not np.all(np.isfinite(grad)):
        return val, np.zeros_like(theta)
    return val, grad

def smooth_optimize(h_init, time_fn, reserve_s):
    theta = theta_from_h(h_init)
    n = len(theta)
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)
    betas = [5.0, 15.0, 40.0, 100.0, 250.0, 600.0]
    if n <= 50:
        betas = [5.0, 15.0, 40.0, 100.0, 250.0, 600.0, 1500.0]
    for beta in betas:

            break
        try:
            res = minimize(_min_fg, x0=theta, args=(beta,), jac=True,
                          method='L-BFGS-B', options={'maxiter': 200, 'gtol': 1e-5})
            if res.x is not None and np.all(np.isfinite(res.x)):

                    best_h = h_new
                    best_c5 = c5_new
                theta = np.clip(res.x, -25, 25)
        except Exception:
            pass
    return best_h, best_c5

def coord_descent(h, time_fn, reserve_s):
    n = len(h)
    current_c5 = compute_c5(h)
    offsets = [n//2, n//3, 1, max(1, n//4), max(1, 2*n//5)]
    offsets = list(set(o % n for o in offsets if o > 0))
    deltas = [0.3, 0.15, 0.08, 0.04, 0.02, 0.01, 0.005, 0.002]
    no_improve_count = 0
    while time_fn() > reserve_s:
        improved = False
        for offset in offsets:
            for j in range(n):
                k = (j + offset) % n
                if k == j:
                    continue
                for delta in deltas:
                    for sgn in (1.0, -1.0):
                        d = sgn * delta
                        a = h[j] + d
                        b = h[k] - d
                        if a < 0.0 or a > 1.0 or b < 0.0 or b > 1.0:
                            continue
                        h[j] = a
                        h[k] = b
                        c5 = compute_c5(h)
                        if c5 < current_c5 - 1e-14:
                            current_c5 = c5
                            improved = True
                            no_improve_count = 0
                            break
                        else:
                            h[j] = a - d  # restore
                            h[k] = b + d
                    if improved:
                        break
                if time_fn() < reserve_s:
                    return h, current_c5
        if not improved:
            no_improve_count += 1
            if no_improve_count >= 3:
                deltas = [d * 0.5 for d in deltas]
                no_improve_count = 0
                if deltas[0] < 1e-7:
                    break
    return h, current_c5

def interpolate_h(h_src, new_n):

    if new_n == n_src:
        return h_src.copy()
    x_src = np.linspace(0, 2, n_src)
    x_new = np.linspace(0, 2, new_n)
    h_new = np.interp(x_new, x_src, h_src)
    return project_feasible(h_new)

def project_feasible(h):
