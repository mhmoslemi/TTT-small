import numpy as np
import time
import math
from scipy.optimize import minimize

def exact_objective(h):
    n = len(h)
    if n == 0:
        return float('inf'), np.array([])
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    c5 = float(np.max(profile))
    return c5, profile

def project_box_sum(h, target=None):
    h = np.array(h, dtype=np.float64).copy()
    n = len(h)
    if target is None:
        target = n / 2.0
    # Handle non-finite
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    # Clip to [0,1]
    h = np.clip(h, 0.0, 1.0)
    # Water-filling to achieve exact sum
    for _ in range(200):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            # Need to add
            mask = h < 1.0
            cnt = np.sum(mask)
            if cnt == 0:
                break
            step = min(diff / cnt, 1.0 - h[mask])
            if step <= 0:
                # Distribute to those closest to 1
                for idx in np.where(mask)[0]:
                    add = min(diff, 1.0 - h[idx])
                    h[idx] += add
                    diff -= add
                    if abs(diff) < 1e-14:
                        break
            else:
                h[mask] += step
        else:
            # Need to subtract
            mask = h > 0.0
            cnt = np.sum(mask)
            if cnt == 0:
                break
            step = min(-diff / cnt, h[mask])
            if step <= 0:
                for idx in np.where(mask)[0]:
                    sub = min(-diff, h[idx])
                    h[idx] -= sub
                    diff += sub
                    if abs(diff) < 1e-14:
                        break
            else:
                h[mask] -= step
    # Final tiny correction
    residual = target - np.sum(h)
    if abs(residual) > 0:
        # Distribute to any element with room
        if residual > 0:
            for i in range(n):
                if h[i] < 1.0:
                    add = min(residual, 1.0 - h[i])
                    h[i] += add
                    residual -= add
                    if abs(residual) < 1e-15:
                        break
        else:
            for i in range(n):
                if h[i] > 0.0:
                    sub = min(-residual, h[i])
                    h[i] -= sub
                    residual += sub
                    if abs(residual) < 1e-15:
                        break
    return h

def compute_gradient_smooth(h, profile, tau):
    """Compute gradient of log-sum-exp smoothed objective."""
    n = len(h)
    dx = 2.0 / n
    pmax = np.max(profile)
    # Weights
    shifted = (profile - pmax) / tau
    # Clip for numerical stability
    shifted = np.clip(shifted, -100, 0)
    weights = np.exp(shifted)
    wsum = np.sum(weights)
    if wsum < 1e-300:
        # Degenerate: only max matters
        w = np.zeros_like(profile)
        w[np.argmax(profile)] = 1.0
    else:
        w = weights / wsum
    
    # Compute gradient
    grad = np.zeros(n, dtype=np.float64)
    # For each lag m (0 to 2n-2), compute contribution
    # Lag s = m - (n-1)
    for m in range(2 * n - 1):
        wm = w[m]
        if wm < 1e-15:
            continue
        s = m - (n - 1)
        if s >= 0:
            # P(s) = dx * sum_{i=0}^{n-1-s} h[i+s]*(1-h[i])
            # dP/dh[j] = dx * [(1-h[j-s]) * 1_{s<=j<=n-1} - h[j+s] * 1_{0<=j<=n-1-s}]
            if s == 0:
                grad += wm * dx * (1.0 - 2.0 * h)
            else:
                # First term: j from s to n-1
                for j in range(s, n):
                    grad[j] += wm * dx * (1.0 - h[j - s])
                # Second term: j from 0 to n-1-s
                for j in range(0, n - s):
                    grad[j] -= wm * dx * h[j + s]
        else:
            a = -s  # a > 0
            # P(-a) = dx * sum_{i=a}^{n-1} h[i]*(1-h[i-a])
            # dP/dh[j] = dx * [(1-h[j-a]) * 1_{a<=j<=n-1} - h[j+a] * 1_{0<=j<=n-1-a}]
            for j in range(a, n):
                grad[j] += wm * dx * (1.0 - h[j - a])
            for j in range(0, n - a):
                grad[j] -= wm * dx * h[j + a]
    return grad

  Second term: grad[j] -= wm * dx * h[j+a] for j in [0, n-1-a]

```python
def compute_gradient_smooth(h, profile, tau):
    n = len(h)
    dx = 2.0 / n
    pmax = np.max(profile)
    shifted = np.clip((profile - pmax) / tau, -100, 0)
    weights = np.exp(shifted)
    wsum = np.sum(weights)
    if wsum < 1e-300:
        w = np.zeros(2*n-1)
        w[np.argmax(profile)] = 1.0
    else:
        w = weights / wsum
    
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        wm = w[m]
        if wm < 1e-12:
            continue
        s = m - (n - 1)
        if s >= 0:
            grad[s:] += wm * dx * (1.0 - h[:n-s])
            grad[:n-s] -= wm * dx * h[s:]
        else:
            a = -s
            grad[a:] += wm * dx * (1.0 - h[:n-a])
            grad[:n-a] -= wm * dx * h[a:]
    return grad
```

def slsqp_optimize(h0, n, tau_list, deadline):
    h = h0.copy()
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0, 'jac': lambda x: np.ones(n)}]
    
    for tau in tau_list:
        if time.time() > deadline - 5:
            break
        
        def func(x):
            prof = np.correlate(x, 1.0 - x, mode='full') * (2.0/n)
            pmax = np.max(prof)
            shifted = np.clip((prof - pmax) / tau, -100, 0)
            w = np.exp(shifted)
            val = pmax + tau * np.log(np.sum(w))
            # Penalty for constraint violation
            sum_err = np.sum(x) - n/2.0
            box_viol = np.sum(np.maximum(0, x-1)**2) + np.sum(np.maximum(0, -x)**2)
            return val + 1e6 * sum_err**2 + 1e6 * box_viol
        
        def jac(x):
            prof = np.correlate(x, 1.0 - x, mode='full') * (2.0/n)
            g = compute_gradient_smooth(x, prof, tau)
            # Add penalty gradient
            g += 2e6 * (np.sum(x) - n/2.0) * np.ones(n)
            g += 2e6 * (np.maximum(0, x-1) - np.maximum(0, -x))
            return g
        
        try:
            res = minimize(func, h, method='SLSQP', bounds=bounds, constraints=cons,
                          jac=jac, options={'maxiter': 200, 'ftol': 1e-12, 'disp': False})
            h_new = res.x
            h_new = project_box_sum(h_new)
            c5_new, _ = exact_objective(h_new)
            c5_old, _ = exact_objective(h)
            if c5_new < c5_old - 1e-12:
                h = h_new
        except Exception:
            pass
    
    return h

```python
import numpy as np
import time
import math
from scipy.optimize import minimize

def exact_objective(h):
    n = len(h)
    if n == 0:
        return float('inf'), np.array([])
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    c5 = float(np.max(profile))
    return c5, profile

def project_box_sum(h, target=None):
    h = np.array(h, dtype=np.float64).copy()
    n = len(h)
    if target is None:
        target = n / 2.0
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = np.sum(mask)
            if cnt == 0:
                break
            step = diff / cnt
            h[mask] = np.minimum(h[mask] + step, 1.0)
        else:
            mask = h > 0.0
            cnt = np.sum(mask)
            if cnt == 0:
                break
            step = (-diff) / cnt
            h[mask] = np.maximum(h[mask] - step, 0.0)
    # Final correction
    residual = target - np.sum(h)
    if abs(residual) > 1e-15:
        if residual > 0:
            for i in range(n):
                add = min(residual, 1.0 - h[i])
                h[i] += add
                residual -= add
                if abs(residual) < 1e-15:
                    break
        else:
            for i in range(n):
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
                if abs(residual) < 1e-15:
                    break
    return h

def compute_smooth_gradient(h, profile, tau):
    n = len(h)
    dx = 2.0 / n
    pmax = float(np.max(profile))
    shifted = np.clip((profile - pmax) / tau, -50.0, 0.0)
    weights = np.exp(shifted)
    wsum = np.sum(weights)
    if wsum < 1e-300:
        w = np.zeros(2 * n - 1)
        w[int(np.argmax(profile))] = 1.0
    else:
        w = weights / wsum
    
    grad = np.zeros(n, dtype=np.float64)
    for m in range(2 * n - 1):
        wm = w[m]
        if wm < 1e-14:
            continue
        s = m - (n - 1)
        if s >= 0:
            g = wm * dx
            grad[s:] += g * (1.0 - h[:n - s])
            grad[:n - s] -= g * h[s:]
        else:
            a = -s
            g = wm * dx
            grad[a:] += g * (1.0 - h[:n - a])
            grad[:n - a] -= g * h[a:]
    return grad

def slsqp_optimize(h0, n, tau_list, deadline):
    h = h0.copy()
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    
    current_c5, _ = exact_objective(h)
    
    for tau in tau_list:
        if time.time() > deadline - 3:
            break
        
        try:
            def func(x):
                prof = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
                pmax = float(np.max(prof))
                sh = np.clip((prof - pmax) / tau, -50.0, 0.0)
                val = pmax + tau * math.log(float(np.sum(np.exp(sh))))
                return val
            
            def jac(x):
                prof = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
                return compute_smooth_gradient(x, prof, tau)
            
            res = minimize(func, h, method='SLSQP', bounds=bounds,
                          constraints=cons, jac=jac,
                          options={'maxiter': 150, 'ftol': 1e-10, 'disp': False})
            
            h_new = project_box_sum(res.x)
            c5_new, _ = exact_objective(h_new)
            if c5_new < current_c5 - 1e-12:
                h = h_new
                current_c5 = c5_new
        except Exception:
            pass
    
    return h

def binary_greedy_swap(h, deadline, max_rounds=20):
    n = len(h)
    if n % 2 != 0:
        return h, exact_objective(h)[0]
    
    M = n // 2
    idx = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[idx[:M]] = 1.0
    
    current = b.copy()
    current_c5, _ = exact_objective(current)
    best = current.copy()
    best_c5 = current_c5
    
    for rnd in range(max_rounds):
        if time.time() > deadline - 2:
            break
        improved = False
        ones = np.where(current == 1.0)[0]
        zeros = np.where(current == 0.0)[0]
        
        for i in ones:
            if time.time() > deadline - 2:
                break
            for j in zeros:
                current[i] = 0.0
                current[j] = 1.0
                c5_try, _ = exact_objective(current)
                if c5_try < current_c5 - 1e-13:
                    current_c5 = c5_try
                    improved = True
                else:
                    current[i] = 1.0
                    current[j] = 0.0
        if current_c5 < best_c5 - 1e-12:
            best = current.copy()
            best_c5 = current_c5
        if not improved:
            break
    
    return best, best_c5

def binary_simulated_annealing(h_bin, deadline, rng, n_iter=30000):
    n = len(h_bin)
    M = n // 2
    idx = np.argsort(h_bin)[::-1]
    current = np.zeros(n)
    current[idx[:M]] = 1.0
    
    current_c5, _ = exact_objective(current)
    best = current.copy()
    best_c5 = current_c5
    
    ones = list(np.where(current == 1.0)[0])
    zeros = list(np.where(current == 0.0)[0])
    
    T = 0.1
    cool_rate = 0.9997
    pos_ones = {v: i for i, v in enumerate(ones)}
    pos_zeros = {v: i for i, v in enumerate(zeros)}
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        a = ones[rng.randint(len(ones))]
        b = zeros[rng.randint(len(zeros))]
        
        current[a] = 0.0
        current[b] = 1.0
        new_c5, _ = exact_objective(current)
        delta = new_c5 - current_c5
        
        if delta <= 0 or (T > 1e-6 and rng.random() < math.exp(-delta / T)):
            current_c5 = new_c5
            # Update lists
            pa = pos_ones[a]
            pb = pos_zeros[b]
            ones[pa] = b
            zeros[pb] = a
            pos_ones[b] = pa
            pos_zeros[a] = pb
            del pos_ones[a]
            del pos_zeros[b]
            if current_c5 < best_c5 - 1e-13:
                best_c5 = current_c5
                best = current.copy()
        else:
            current[a] = 1.0
            current[b] = 0.0
        
        if it % 500 == 499:
            T *= cool_rate
    
    return best, best_c5

def random_continuous_descent(h, deadline, rng, n_steps=20000):
    n = len(h)
    current = h.copy()
    current_c5, _ = exact_objective(current)
    
    scale = 0.1
    for step in range(n_steps):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        delta = rng.uniform(-scale, scale)
        trial_i = current[i] + delta
        trial_j = current[j] - delta
        if trial_i < 0 or trial_i > 1 or trial_j < 0 or trial_j > 1:
            continue
        current[i] = trial_i
        current[j] = trial_j
        c5_try, _ = exact_objective(current)
        if c5_try < current_c5 - 1e-13:
            current_c5 = c5_try
            scale = max(0.01, scale * 0.9995)
        else:
            current[i] = trial_i - delta
            current[j] = trial_j + delta
    
    return current
