import numpy as np
import time

def eval_c5_fn(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_fn(v, target_sum, n):
    lo = v.min() - target_sum - 1.0
    hi = v.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        h = np.clip(v - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_smooth_fn(h, tau, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = corr.max()
    exp_c = np.exp((corr - max_c) / tau)
    w = exp_c / exp_c.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        L = k - (n - 1)
        wk = w[k]
        if wk < 1e-14:
            continue
        if L >= 0:
            grad[:n-L] += wk * (1.0 - h[L:n])
            grad[L:] -= wk * h[:n-L]
        else:
            Lneg = -L
            grad[Lneg:] += wk * (1.0 - h[:n-Lneg])
            grad[:n-Lneg] -= wk * h[Lneg:]
    return grad

def grad_variance_fn(h, n, mu):
    corr = np.correlate(h, 1.0 - h, mode='full')
    deviations = corr - mu
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        L = k - (n - 1)
        d = deviations[k]
        if abs(d) < 1e-14:
            continue
        if L >= 0:
            grad[:n-L] += 2.0 * d * (1.0 - h[L:n])
            grad[L:] -= 2.0 * d * h[:n-L]
        else:
            Lneg = -L
            grad[Lneg:] += 2.0 * d * (1.0 - h[:n-Lneg])
            grad[:n-Lneg] -= 2.0 * d * h[Lneg:]

cos²(x) = (1+cos(2x))/2
sum = (1/2)*sum(1) + (1/2)*sum cos(4πi/n)
= n/4 + (1/2)*sum_{i=0}^{n/2-1} cos(4πi/n)

sum_{i=0}^{n/2-1} cos(4πi/n) = Re(sum_{i=0}^{n/2-1} e^{4πi/n})
= Re(e^{2πi(n/2-1)/n} * (1-e^{4πi(n/2)/n}) / (1-e^{4πi/n}))
= Re(e^{2πi(n/2-1)/n} * (1-e^{2πi}) / (1-e^{4πi/n}))
= Re(e^{2πi(n/2-1)/n} * 0 / (...)) = 0

Since e^{2πi} = 1, the numerator is 0. So the sum is 0 (for n/2 being an integer, which it is when n is even).

So sum cos²(2πi/n) for i=0..n/2-1 = n/4.

Therefore:
g(n/2) = n/2 * 0.25 + 2a * 0 + a² * n/4 = n/8 + a²n/4

C5 contribution: (n/8 + a²n/4) * 2/n = 1/4 + a²/2

And for L=0:
g(0) = n/4 - a²n/2
C5 contribution: (n/4 - a²n/2) * 2/n = 1/2 - a²

So C5(a) = max(1/2 - a², 1/4 + a²/2)

Setting equal: 1/2 - a² = 1/4 + a²/2 → 1/4 = 3a²/2 → a² = 1/6 → a = 1/√6

At this point: C5 = 1/2 - 1/6 = 1/3 ≈ 0.3333

So the minimum C5 using this family is 1/3 ≈ 0.333, achieved at a = 1/√6 ≈ 0.408.

This is well below the current record of 0.38085! So just by using this simple cosine initialization, we should be able to beat the record.

But we need to verify numerically, because the analysis has approximations (the "≈ 0" for the cosine sums).
