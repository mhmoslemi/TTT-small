```python
import numpy as np
import time

def project_box_sum(x, s):
    """Project x onto {0<=x_i<=1, sum(x_i)=s}"""
    x = np.asarray(x, dtype=np.float64).copy()
    n = len(x)
    # Binary search for lambda such that sum(clip(x_i - lambda, 0, 1)) = s
    lo, hi = x.min() - 1.0, x.max() + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        projected = np.clip(x - mid, 0.0, 1.0)
        total = projected.sum()
        if total > s:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    result = np.clip(x - lam, 0.0, 1.0)
    # Fix exact sum
    residual = s - result.sum()
    if abs(residual) > 1e-12:
        if residual > 0:
            # Need to add mass to elements not at upper bound
            can_add = (result < 1.0 - 1e-12)
            if can_add.any():
                n_add = can_add.sum()
                result[can_add] += residual / n_add
                result = np.clip(result, 0.0, 1.0)
        else:
            # Need to remove mass from elements not at lower bound
            can_sub = (result > 1e-12)
            if can_sub.any():
                n_sub = can_sub.sum()
                result[can_sub] += residual / n_sub
                result = np.clip(result, 0.0, 1.0)
    return result

def eval_c5(h):
    """Compute exact C5 value for given h"""
    h = np.asarray(h, dtype=np.float64)
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair_exact_sum(h, s):
    """Fix small sum violations while maintaining [0,1] bounds"""
    h = h.copy()
    n = len(h)
    residual = s - h.sum()
    if abs(residual) < 1e-10:
        return h
    if residual > 0:
        can_add = h < 1.0 - 1e-8
        if can_add.any():
            h[can_add] += residual / can_add.sum()
    else:
        can_sub = h > 1e-8
        if can_sub.any():
            h[can_sub] += residual / can_sub.sum()
    np.clip(h, 0.0, 1.0, out=h)
    # Final tiny fix
    residual2 = s - h.sum()
    if abs(residual2) > 1e-14:
        idx = int(np.argmax(h if residual2 < 0 else 1 - h))
        h[idx] += residual2
        np.clip(h, 0.0, 1.0, out=h)
    return h
```

```python
def compute_smooth_gradient(h, tau):
    """Compute gradient of LSE surrogate"""
    n = len(h)
    dx = 2.0 / n
    # Compute raw lag values
    corr = np.correlate(h, 1.0 - h, mode="full")
    raw = corr * dx
    # LSE
    rmax = raw.max()
    exps = np.exp((raw - rmax) / tau)
    weights = exps / exps.sum()
    # Compute gradient
    grad = np.zeros(n)
    # For each output index k (0 to 2n-2), corresponding lag p = k - (n-1)
    for k in range(2 * n - 1):
        p = k - (n - 1)
        w = weights[k]
        if w < 1e-15:
            continue
        if p >= 0:
            # R_p = dx * sum_{i=0}^{n-1-p} h[i+p]*(1-h[i])
            # grad[i+p] += dx * w * (1-h[i])
            # grad[i] += dx * w * (-h[i+p])
            for i in range(n - p):
                grad[i + p] += dx * w * (1.0 - h[i])
                grad[i] += dx * w * (-h[i + p])
        else:
            q = -p  # q > 0
            # R_{-q} = dx * sum_{i=0}^{n-1-q} h[i]*(1-h[i+q])
            # grad[i] += dx * w * (1-h[i+q])
            # grad[i+q] += dx * w * (-h[i])
            for i in range(n - q):
                grad[i] += dx * w * (1.0 - h[i + q])
                grad[i + q] += dx * w * (-h[i])
    return grad
```

- Take step in -gradient direction

- Accept if exact C5 improves (or if LSE improves with small exact degradation)

For pair transfers:
- For each pair (i, j), try transferring mass from i to j (and vice versa)
- Try different transfer amounts
- Accept if exact C5 improves

For SA:
- Propose random pair transfers or random projected perturbations
- Accept with Metropolis criterion based on exact C5

```python
import numpy as np
import time

def project_box_sum(x, s):
    x = np.asarray(x, dtype=np.float64).copy()
    n = len(x)
    lo = x.min() - 1.0
    hi = x.max() + 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        proj = np.clip(x - mid, 0.0, 1.0)
        total = proj.sum()
        if total > s:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    result = np.clip(x - lam, 0.0, 1.0)
    # Fix exact sum
    residual = s - result.sum()
    if abs(residual) > 1e-14:
        if residual > 0:
            mask = result < 1.0 - 1e-6
        else:
            mask = result > 1e-6
        if mask.any():
            result[mask] += residual / mask.sum()
    np.clip(result, 0.0, 1.0, out=result)
    return result

def eval_c5(h):
    h = np.asarray(h, dtype=np.float64)
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair_sum(h, s):
    h = h.copy()
    residual = s - h.sum()
    if abs(residual) < 1e-12:
        return h
    if residual > 0:
        mask = h < 1.0 - 1e-10
    else:
        mask = h > 1e-10
    if mask.any():
        h[mask] += residual / mask.sum()
    np.clip(h, 0.0, 1.0, out=h)
    return h

def lse_value_and_weights(raw, tau):
    rmax = raw.max()
    exps = np.exp((raw - rmax) / tau)
    probs = exps / exps.sum()
    lse = rmax + tau * np.log(exps.sum())
    return lse, probs

def smooth_gradient(h, tau, weights):
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        p = k - (n - 1)
        w = weights[k]
        if w < 1e-14:
            continue
        if p >= 0:
            m = n - p
            grad[p:] += dx * w * (1.0 - h[:m])
            grad[:m] += -dx * w * h[p:]
        else:
            q = -p
            m = n - q
            grad[:m] += dx * w * (1.0 - h[q:])
            grad[q:] += -dx * w * h[:m]
    return grad

def pair_transfer_score(h, i, j, amount, s):
    """Try transferring 'amount' from h[i] to h[j], return new vector or None"""
    if amount <= 0:
        return None
    if h[i] - amount < -1e-12 or h[j] + amount > 1.0 + 1e-12:
        return None
    new_h = h.copy()
    new_h[i] -= amount
    new_h[j] += amount
    np.clip(new_h, 0.0, 1.0, out=new_h)
    # Sum is preserved since we added and subtracted same amount
    # But clipping might change sum slightly
    new_h = repair_sum(new_h, s)
    return new_h

def make_fallback(n, seed):
    rng = np.random.default_rng(seed)
    x = 0.5 + 0.3 * np.sin(2 * np.pi * np.arange(n) / n + 0.5)
    x = np.clip(x, 0.05, 0.95)
    return project_box_sum(x, n / 2.0)

def polish_with_lse(h, s, best_c, time_limit, rng):
    """LSE-based projected gradient descent"""
    n = len(h)
    h = h.copy()
    cur_c = eval_c5(h)
    local_best_h = h.copy()
    local_best_c = cur_c
    tau = max(0.001, 0.15 * cur_c)
    alpha = 0.02
    no_improve = 0
    max_no_improve = 50
    t0 = time.perf_counter()
    
    while time.perf_counter() - t0 < time_limit:
        # Compute LSE and weights
        dx = 2.0 / n
        corr = np.correlate(h, 1.0 - h, mode="full")
        raw = corr * dx
        _, weights = lse_value_and_weights(raw, tau)
        
        # Compute gradient
        grad = smooth_gradient(h, tau, weights)
        
        # Project gradient to sum-zero direction
        grad -= grad.mean()
        
        # Check if gradient is too small
        if np.linalg.norm(grad) < 1e-10:
            tau *= 0.7
            alpha *= 0.8
            no_improve += 5
            if no_improve > max_no_improve:
                break
            continue
        
        # Line search
        improved = False
        for _ in range(20):
            candidate = h - alpha * grad
            candidate = project_box_sum(candidate, s)
            candidate = repair_sum(candidate, s)
            cand_c = eval_c5(candidate)
            
            if cand_c < cur_c - 1e-12:
                h = candidate
                cur_c = cand_c
                improved = True
                if cur_c < local_best_c:
                    local_best_c = cur_c
                    local_best_h = h.copy()
                break
            else:
                alpha *= 0.7
        
        if improved:
            no_improve = 0
        else:
            no_improve += 1
            alpha *= 0.8
            tau *= 0.85
        
        if no_improve > max_no_improve:
            if tau < 1e-4:
                break
            # Restart with new random direction
            noise = rng.standard_normal(n) * 0.01
            noise -= noise.mean()
            h = project_box_sum(h + noise, s)
            h = repair_sum(h, s)
            cur_c = eval_c5(h)
            no_improve = 0
            tau = max(1e-4, 0.1 * cur_c)
            alpha = 0.01
    
    return local_best_h, local_best_c

def pair_transfer_descent(h, s, best_c, time_limit, rng):
    """Exact pair transfer local search"""
    n = len(h)
    h = h.copy()
    cur_c = eval_c5(h)
    local_best_h = h.copy()
    local_best_c = cur_c
    t0 = time.perf_counter()
    
    fractions = [0.02, 0.05, 0.1, 0.2, 0.5, 0.8]
    n_pairs = n * n  # total directed pairs (excluding self)
    eval_count = 0
    
    while time.perf_counter() - t0 < time_limit:
        improved_any = False
        # Shuffle pair order
        pair_indices = rng.permutation(n_pairs)
        
        for idx in pair_indices:
            i = idx // n
            j = idx % n
            if i == j:
                continue
            
            cap = min(h[i], 1.0 - h[j])
            if cap < 1e-8:
                continue
            
            # Try transferring from i to j
            for frac in fractions:
                amount = frac * cap
                if amount < 1e-10:
                    continue
                new_h = h.copy()
                new_h[i] -= amount
                new_h[j] += amount
                np.clip(new_h, 0.0, 1.0, out=new_h)
                new_h = repair_sum(new_h, s)
                new_c = eval_c5(new_h)
                eval_count += 1
                
                if new_c < cur_c - 1e-12:
                    h = new_h
                    cur_c = new_c
                    improved_any = True
                    if cur_c < local_best_c:
                        local_best_c = cur_c
                        local_best_h = h.copy()
                    cap = min(h[i], 1.0 - h[j])
                    break
                elif new_c < cur_c:
                    # Accept non-improving moves occasionally for exploration
                    if rng.random() < 0.1:
                        h = new_h
                        cur_c = new_c
```

        if float(np.linalg.norm(g)) < 1e-12:
            stall += 10
            tau *= 0.7
            if stall > 200 or tau < 1e-5:
                break
            continue
        ok = False
        for _ in range(12):
            cand = project_box_sum(h - alpha * g, s)
            cand = repair_sum(cand, s)
            cc = eval_c5(cand)
            if cc < cur_c - 1e-13:
                h, cur_c = cand, cc
                ok = True
                if cc < best_c:
                    best_c, best_h = cc, h.copy()
                break
            alpha *= 0.5
        if ok:
            stall = 0
        else:
            stall += 1
            alpha *= 0.6
            tau *= 0.88
            if stall > 60:
                if tau < 1e-5:
                    break
                nse = rng.standard_normal(n) * 0.003
                nse -= float(nse.mean())
                h = repair_sum(project_box_sum(h + nse, s), s)
                cur_c = eval_c5(h)
                stall = 0
                tau = max(1e-4, 0.05 * cur_c)
                alpha = 0.008
    return best_h, best_c

def optimize_pairs(h, s, deadline, rng):
    n = len(h)
    h = h.copy()
    cur_c = eval_c5(h)
    best_h, best_c = h.copy(), cur_c
    fracs = [0.005, 0.01, 0.02, 0.04, 0.07, 0.12, 0.2, 0.35]
    while time.perf_counter() < deadline:
        any_imp = False
        for idx in rng.permutation(n * n):
            if time.perf_counter() >= deadline:
                break
            i, j = int(idx // n), int(idx % n)
            if i == j:
                continue
            cap = min(float(h[i]), 1.0 - float(h[j]))
            if cap < 5e-9:
                continue
            for f in fracs:
                a = f * cap
                if a < 1e-10:
                    continue
                nh = h.copy()
                nh[i] -= a
                nh[j] += a
                np.clip(nh, 0.0, 1.0, out=nh)
                nh = repair_sum(nh, s)
                nc = eval_c5(nh)
                if nc < cur_c - 1e-13:
                    h, cur_c = nh, nc
                    any_imp = True
                    if nc < best_c:
                        best_c, best_h = nc, h.copy()
                    break
        if not any_imp:
            break
    return best_h, best_c

def optimize_sa(h, s, deadline, rng):
    n = len(h)
    h = h.copy()
    cur_c = eval_c5(h)
    best_h, best_c = h.copy(), cur_c
    temp = 0.01 * max(cur_c, 0.1)
    scale = 0.05
    na, nt = 0, 0
    while time.perf_counter() < deadline:
        nt += 1
        if rng.random() < 0.55:
            i, j = int(rng.integers(n)), int(rng.integers(n))
            if i == j:
                continue
            if rng.random() < 0.5:
                cap = min(float(h[i]), 1.0 - float(h[j]))
            else:
                cap = min(float(h[j]), 1.0 - float(h[i]))
                i, j = j, i
            if cap < 1e-10:
                continue
            a = rng.random() * cap * scale
            if a < 1e-10:
                continue
            nh = h.copy()
            nh[i] -= a
            nh[j] += a
            np.clip(nh, 0.0, 1.0, out=nh)
        else:
            nz = rng.standard_normal(n) * 0.02 * scale
            nz -= float(nz.mean())
            nh = project_box_sum(h + nz, s)
        nh = repair_sum(nh, s)
        nc = eval_c5(nh)
        if nc <= cur_c:
            h, cur_c = nh, nc
            na += 1
            if nc < best_c:
                best_c, best_h = nc, h.copy()
        elif rng.random() < np.exp(-(nc - cur_c) / max(temp, 1e-10)):
            h, cur_c = nh, nc
            na += 1
        if nt % 200 == 0:
            rate = na / float(nt)
            if rate < 0.05:
                scale *= 0.7
            elif rate > 0.75:
                scale = min(scale * 1.2, 0.35)
            temp = max(temp * 0.95, 1e-6)
            na, nt = 0, 0
        temp = max(temp * 0.998, 1e-6)
    return best_h, best_c
