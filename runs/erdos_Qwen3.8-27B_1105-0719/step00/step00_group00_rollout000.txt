import numpy as np
from scipy import optimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return np.max(corr), corr

def project_h(y, n):
    """Project y onto {h: 0<=h<=1, sum(h)=n/2} via bisection on Lagrange multiplier."""
    target = n / 2.0
    lo, hi = -y.max() - 0.5, 1.0 - y.min() + 0.5
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h = np.clip(y - mid, 0.0, 1.0)
        s = np.sum(h)
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    # Tweak to fix residual
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        nf = np.sum(free)
        if nf > 0:
            h[free] += diff / nf
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(corr, alpha):
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -500, 500)
    return m + np.log(np.sum(np.exp(shifted)) + 1e-300) / alpha

def subgrad_sum(h, n, dx):
    """Compute subgradient of C5 = max_k corr[k] using active set."""
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    c5 = np.max(corr)
    tol = 1e-8
    active = np.where(corr >= c5 - tol)[0]
    grad = np.zeros(n)
    for k in active:
        # Analytical gradient of corr[k] w.r.t. h
        # corr[k] = sum_i h[i] * (1-h[i+k-(n-1)]) * dx over valid indices
        # Let offset = k - (n-1)
        offset = k - (n - 1)
        if offset >= 0:
            # corr[k] = sum_{i=0}^{n-1-offset} h[i] * (1-h[i+offset]) * dx
            # d/dh[j] = dx * (1-h[j+offset]) for j=0..n-1-offset
            #         - dx * h[j-offset] for j=offset..n-1
            nterms = n - offset
            grad[:nterms] += dx * (1.0 - h[offset:offset + nterms])
            grad[offset:] += -dx * h[:nterms]
        else:
            # offset < 0, let o = -offset > 0
            o = -offset
            # corr[k] = sum_{i=o}^{n-1} h[i] * (1-h[i-o]) * dx
            # d/dh[j] = dx * (1-h[j-o]) for j=o..n-1
            #         - dx * h[j+o] for j=0..n-1-o
            nterms = n - o
            grad[o:] += dx * (1.0 - h[:nterms])
            grad[:nterms] += -dx * h[o:o + nterms]
    return grad / len(active)

def local_search_pairs(h, n, dx, target_sum, max_iters=20000, time_limit=60.0):
    """Coordinate pair local search: transfer mass between pairs."""
    start = time.time()
    best_c5, _ = evaluate_c5(h)
    
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        i = it % n
        j = (it // n) % n
        if i == j:
            continue
        # Try transferring mass from i to j (and vice versa)
        # h[i] decreases by t, h[j] increases by t (keeps sum constant)
        # t in (-h[i], 1-h[j]) for i->j, or (-h[j], 1-h[i]) for j->i
        best_t = 0.0
        best_obj = best_c5
        for t_sign in [1.0, -1.0]:
            if t_sign == 1.0:
                # h[i] -= t, h[j] += t, t in (0, min(h[i], 1-h[j]))
                t_lo, t_hi = 0.0, min(h[i], 1.0 - h[j])
            else:
                # h[i] += t, h[j] -= t, t in (0, min(1-h[i], h[j]))
                t_lo, t_hi = 0.0, min(1.0 - h[i], h[j])
            if t_hi <= 1e-10:
                continue
            # Optimize over t in [t_lo, t_hi]
            # Use golden section or just try a few points
            ts = np.linspace(t_lo, t_hi, 8)
            for t in ts:
                h[i] -= t_sign * t
                h[j] += t_sign * t
                c5_try, _ = evaluate_c5(h)
                if c5_try < best_obj:
                    best_obj = c5_try
                    best_t = t_sign * t
                else:
                    h[i] += t_sign * t
                    h[j] -= t_sign * t
            # Undo if best_t was 0
            if best_t == 0.0:
                h[i] += t_sign * t
                h[j] -= t_sign * t
            else:
                # Reapply best
                h[i] += t_sign * t
                h[j] -= t_sign * t
                # Reapply best_t
                bt_abs = abs(best_t)
                bt_sign = np.sign(best_t)
                h[i] -= bt_sign * bt_abs
                h[j] += bt_sign * bt_abs
        
        if best_obj < best_c5:
            best_c5 = best_obj
    
    return h, best_c5
