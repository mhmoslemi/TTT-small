import numpy as np
import time

def project_sum(h, target):
    """Project h onto {0<=h<=1, sum(h)=target}"""
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    # Iterative water-filling
    for _ in range(100):
        if diff > 0:
            # Need to add mass; add to elements below 1
            mask = h < 1.0 - 1e-12
            if not mask.any():
                break
            per = diff / mask.sum()
            h[mask] = np.minimum(h[mask] + per, 1.0)
        else:
            # Need to remove mass; remove from elements above 0
            mask = h > 1e-12
            if not mask.any():
                break
            per = diff / mask.sum()
            h[mask] = np.maximum(h[mask] + per, 0.0)
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
    return h

def c5_eval(h, dx):
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def golden_search_1d(h, u, v, s_min, s_max, dx, target_sum):
    """
    Golden section search along h[u]+s, h[v]-s
    Returns (best_s, best_f) or (None, None)
    """
    phi = (np.sqrt(5.0) - 1.0) / 2.0  # golden ratio conjugate
    a, b = s_min, s_max
    c = b - phi * (b - a)
    d = a + phi * (b - a)
    
    # Evaluate endpoints and interior points
    h_orig_u = h[u]
    h_orig_v = h[v]
    
    def f_at(s):
        h[u] = h_orig_u + s
        h[v] = h_orig_v - s
        val = c5_eval(h, dx)
        return val
    
    fc = f_at(c)
    fd = f_at(d)
    
    for _ in range(40):
        if b - a < 1e-10:
            break
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - phi * (b - a)
            fc = f_at(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + phi * (b - a)
            fd = f_at(d)
    
    # Restore
    h[u] = h_orig_u
    h[v] = h_orig_v
    
    best_s = (a + b) / 2.0
    h[u] = h_orig_u + best_s
    h[v] = h_orig_v - best_s
    best_f = c5_eval(h, dx)

    h[v] = h_orig_v

    best_s = np.clip(best_s, s_min, s_max)
    
    return best_s, best_f

        if not improved:

    n = len(h)
    h = _project(h, n / 2.0)
    
    best_h = h.copy()
    best_c5 = _c5s(h)
    
    # Phase 1: Optimize at initial n
    d1 = t0 + min(250, budget_s * 0.25)
    h1, raw1, bh1 = _cd(h, n, d1, seed)
    c5_1 = raw1 * 2.0 / n
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = bh1
    
    # Phase 2: Double resolution
    rem = budget_s - (time.time() - t0)
    if rem > 150:
        n2 = n * 2
        h2 = _interp(best_h, n2)
        d2 = t0 + budget_s * 0.55
        if d2 > time.time() + 60:
            h2, raw2, bh2 = _cd(h2, n2, d2, seed + 1)
            c5_2 = raw2 * 2.0 / n2
            if c5_2 < best_c5:
                best_c5 = c5_2
