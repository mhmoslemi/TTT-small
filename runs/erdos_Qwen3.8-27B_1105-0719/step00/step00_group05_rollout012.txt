def validate_h(h):

def project_feasible(u):
    """Water-filling projection to {0<=x<=1, sum(x)=n/2}"""
    u = np.asarray(u, dtype=np.float64)
    n = len(u)
    T = n / 2.0

def sigmoid(t):
    # Stable sigmoid
    out = np.zeros_like(t, dtype=np.float64)
    pos = t >= 0
    out[pos] = 1.0 / (1.0 + np.exp(-t[pos]))
    exp_t = np.exp(t[~pos])
    out[~pos] = exp_t / (1.0 + exp_t)
    return out

def smooth_objective_and_grad_raw(h, beta):
    """
    Compute F_beta(h) and its gradient G = dF/dh.
    Returns (F, G) where G is gradient w.r.t. h components.
    """
    n = len(h)
    dx = 2.0 / n
    one_minus_h = 1.0 - h
    
    # Compute directional values
    v = np.empty(2*n - 1)
    v[0] = dx * np.sum(h * (1.0 - h))
    idx = 1
    for l in range(1, n):
        v[idx] = dx * np.sum(h[:n-l] * one_minus_h[l:])
        v[idx+1] = dx * np.sum(h[l:] * one_minus_h[:n-l])
        idx += 2

    v_max = np.max(v)
    exp_v = np.exp(beta * (v - v_max))
    w = exp_v / np.sum(exp_v)
    
    F = v_max + np.log(np.sum(exp_v)) / beta
    
    # Gradient
    G = np.zeros(n)
    # v[0] = dx * sum h_i(1-h_i), dv/dh_i = dx*(1-2h_i)
    G += w[0] * dx * (1.0 - 2.0 * h)
    
    for l in range(1, n):
        wp = w[2*l-1]  # positive lag weight
        wn = w[2*l]    # negative lag weight
        
        # v_pos = dx * sum_{i=0}^{n-1-l} h_i(1-h_{i+l})
        # dv/dh_i = dx*(1-h_{i+l}) for i < n-l
        # dv/dh_{i+l} = dx*(-h_i) for i < n-l

        G[l:]   += wp * dx * (-h[:n-l])

        # dv/dh_i = dx*(-h_{i+l}) for i < n-l
        G[l:]   += wn * dx * (1.0 - h[:n-l])
        G[:n-l] += wn * dx * (-h[l:])
    
    return F, G

def exact_C(h):
    n = len(h)
    if n < 2:

    corr = np.correlate(h, 1.0 - h, mode="full")

def project_feasible(u):

    out = np.empty_like(t, dtype=np.float64)

    out[pos] = 1.0 / (1.0 + np.exp(-t[pos]))
    e = np.exp(t[~pos])
    out[~pos] = e / (1.0 + e)
    return out

def smooth_obj_grad_raw(h, beta):

    dx = 2.0 / n

    dims = [d for d in dims if 12 <= d <= 70]
    
    # BFGS phase
    betas = [40.0, 80.0, 150.0, 300.0, 500.0]
    bfgs_count = 0
    max_bfgs = 50
    
    for dim in dims:
        if time.time() > _DEADLINE - 150:
            break
        if bfgs_count >= max_bfgs:
            break
        
        # Get starts at this dimension
        dim_starts = []
        for hc in pool[:6]:
            if len(hc) == dim:
                dim_starts.append(hc.copy())
            else:
                dim_starts.append(resize_h(hc, dim))
        # Add some random
        for _ in range(2):
            dim_starts.append(project_feasible(0.5 + 0.15 * rng.standard_normal(dim)))
        
        for beta in betas:
            if time.time() > _DEADLINE - 100:
                break
            if bfgs_count >= max_bfgs:
                break
            for hs in dim_starts[:4]:
                if time.time() > _DEADLINE - 60:
                    break
                if bfgs_count >= max_bfgs:
                    break
                mi = 120 if dim <= 50 else 70
                he = bfgs_run(hs, beta, mi)
                bfgs_count += 1
                if he is not None:
                    he = project_feasible(he)
                    record_candidate(he)
                    c_e = exact_C(he)
                    if c_e < _BEST_C + 5e-5 and time.time() < _DEADLINE - 80:
                        hp, cp = pair_polish(he, budget=min(15, _DEADLINE - time.time() - 30))
    
    # Final polish on best
    if _BEST_H is not None and time.time() < _DEADLINE - 10:
        budget_rem = min(80, _DEADLINE - time.time() - 5)
        if budget_rem > 5:
            hf, cf = pair_polish(_BEST_H, budget=budget_rem)
    
    # Return
    if _BEST_H is not None:
        h_out = project_feasible(_BEST_H)
        c_out = exact_C(h_out)
        return h_out, c_out, len(h_out)
    else:
        h_out = project_feasible(h_parent)
        c_out = exact_C(h_out)
        return h_out, c_out, len(h_out)
