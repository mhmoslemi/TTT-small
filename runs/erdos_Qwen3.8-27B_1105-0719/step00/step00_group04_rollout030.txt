import numpy as np
import time

def project(h, target_sum):
    """Project h onto {0<=x<=1, sum(x)=target_sum} via bisection on Lagrange multiplier."""
    n = len(h)
    # f(lam) = sum(clip(h-lam, 0, 1)) - target_sum is decreasing in lam
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    
    def f(lam):
        return np.clip(h - lam, 0.0, 1.0).sum() - target_sum
    
    flo = f(lo)
    fhi = f(hi)
    if flo <= 0:
        return np.clip(h, 0.0, 1.0)
    if fhi >= 0:
        return np.clip(h, 0.0, 1.0)
    
    for _ in range(80):
        mid = (lo + hi) * 0.5
        fm = f(mid)
        if fm > 0:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    return np.clip(h - lam, 0.0, 1.0)

def compute_C5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

def lse_value_and_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    F = np.correlate(h, 1.0 - h, mode='full') * dx
    m = F.max()
    exp_F = np.exp((F - m) / tau)
    s = exp_F.sum()
    w = exp_F / s
    lse_val = m + tau * np.log(s)
    
    grad = np.zeros(n)
    for L in range(-(n-1), n):
        wL = w[n-1-L]
        if wL < 1e-14:
            continue
        # Term 1: (1-h[j+L]) for 0<=j<n and 0<=j+L<n
        j_lo = max(0, -L)
        j_hi = min(n, n - L)
        if j_lo < j_hi:
            grad[j_lo:j_hi] += wL * (1.0 - h[j_lo+L:j_hi+L])
        # Term 2: -h[j-L] for 0<=j<n and 0<=j-L<n
        j_lo2 = max(0, L)
        j_hi2 = min(n, n + L)
        if j_lo2 < j_hi2:
            grad[j_lo2:j_hi2] -= wL * h[j_lo2-L:j_hi2-L]
    
    grad *= dx
    return lse_val, grad

def optimize_n(n, h_init, budget_s, start_time):
    """Optimize for given n, starting from h_init. Returns best (h, C5)."""
    target_sum = n / 2.0
    h = project(h_init.copy(), target_sum)
    best_C5 = compute_C5(h)
    best_h = h.copy()
    
    tau = 0.01  # initial temperature
    step = 0.1
    no_improve = 0
    
    while time.time() - start_time < budget_s:
        # Compute LSE and gradient
        lse_val, grad = lse_value_and_grad(h, tau)
        
        # Try a step
        for trial in range(20):
            h_new = project(h - step * grad, target_sum)
            lse_new, _ = lse_value_and_grad(h_new, tau)
            if lse_new < lse_val:
                break
            step *= 0.7
        else:
            # Can't improve, decrease tau and reset step
            tau = max(tau * 0.95, 1e-5)
            step = 0.05
            no_improve += 1
            if no_improve > 200:
                break
            continue
        
        h = h_new
        no_improve = 0
        
        # Evaluate exact C5 periodically
        if no_improve == 0 and (h == h).all():  # always true, just a marker
            c5 = compute_C5(h)
            if c5 < best_C5:
                best_C5 = c5
                best_h = h.copy()
        
        # Decrease tau adaptively
        tau = max(tau * 0.999, 1e-5)
        step = min(step * 1.01, 0.5)
    
    return best_h, best_C5

def project(h, target_sum):
    """Project h onto {0<=x<=1, sum(x)=target_sum}."""
    n = len(h)
    lo, hi = -2.0, 3.0
    # Ensure bracketing
    if np.clip(h - lo, 0, 1).sum() <= target_sum:
        return np.clip(h - lo, 0, 1)
    if np.clip(h - hi, 0, 1).sum() >= target_sum:
        return np.clip(h - hi, 0, 1)
    for _ in range(64):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5*(lo+hi), 0, 1)
