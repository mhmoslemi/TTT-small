from scipy.optimize import minimize, NonlinearConstraint

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n

def smooth_max_and_grad(h, beta):
    n = len(h)
    q = 1.0 - h
    corr = np.correlate(h, q, mode="full")
    nc = len(corr)
    c = corr - corr.max()
    exp_c = np.exp(c * beta)
    weights = exp_c / exp_c.sum()
    smooth_val = corr.max() + np.log(exp_c.sum()) / beta
    
    # Gradient: for each lag k, lag_k = k - (n-1)
    # dcorr[k]/dh[m] = (1-h[m-lag_k])*I(0<=m-lag_k<n) - h[m+lag_k]*I(0<=m+lag_k<n)
    grad = np.zeros(n)
    for k in range(nc):
        lag = k - (n - 1)
        wk = weights[k]
        if wk < 1e-15:
            continue
        for m in range(n):
            if 0 <= m - lag < n:
                grad[m] += wk * (1.0 - h[m - lag])
            if 0 <= m + lag < n:
                grad[m] -= wk * h[m + lag]
    return smooth_val, grad
