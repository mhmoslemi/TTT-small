def project_to_feasible(x, n):
    """Project x to the feasible set: 0 <= x[i] <= 1, sum(x) = n/2"""
    target = n / 2.0
    x = np.clip(x, 0.0, 1.0)
    # Adjust sum using bisection
    diff = target - np.sum(x)
    if abs(diff) < 1e-10:
        return x
    # If diff > 0, need to increase some entries
    # If diff < 0, need to decrease some entries
    # Use a simple approach: distribute among entries with room
    if diff > 0:
        # Add to entries that are not at 1
        room = 1.0 - x
        total_room = np.sum(room)
        if total_room > 0:
            x += diff * room / total_room
    else:
        # Subtract from entries that are not at 0
        room = x
        total_room = np.sum(room)
        if total_room > 0:
            x += diff * room / total_room
    # Clip again and fine-tune
    x = np.clip(x, 0.0, 1.0)
    # Exact correction
    residual = target - np.sum(x)
    if abs(residual) > 1e-12:
        if residual > 0:

                residual -= add

import numpy as np
from scipy.optimize import linprog

def project_to_feasible(x, n):
    target = n / 2.0
    x = np.asarray(x, dtype=float).copy()
    x = np.clip(x, 0.0, 1.0)
    diff = target - np.sum(x)
    if abs(diff) > 1e-12:
        if diff > 0:
            room = 1.0 - x
            total = np.sum(room)
            if total > 1e-12:
                x += diff * room / total
        else:
            room = x.copy()
            total = np.sum(room)
            if total > 1e-12:
                x += diff * room / total
        x = np.clip(x, 0.0, 1.0)
        # Fine correction
        residual = target - np.sum(x)
        if abs(residual) > 1e-13:
            if residual > 0:
                order = np.argsort(1.0 - x)[::-1]
                for i in order:
                    add = min(residual, 1.0 - x[i])
                    if add <= 0: break
                    x[i] += add
                    residual -= add
                    if residual <= 1e-13: break
            else:
                order = np.argsort(x)[::-1]
                for i in order:
                    sub = min(-residual, x[i])
                    if sub <= 0: break
                    x[i] -= sub
                    residual += sub
                    if residual >= -1e-13: break
    return x

def compute_raw(h):
    return np.correlate(h, 1.0 - h, mode='full')

def score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_raw(h)) * dx)

def gradient_for_shift(h, d):
    n = len(h)
    g = np.zeros(n)
    if d == 0:
        g[:] = 1.0 - 2.0 * h
    elif d > 0:
        g[:n-d] += (1.0 - h[d:])
        g[d:] += -h[:n-d]
    else:
        r = -d
        g[r:] += (1.0 - h[:n-r])
        g[:n-r] += -h[r:]
    return g

    return (n - 1) - t

    c = np.zeros(n + 1)
    c[n] = 1.0
    
    A_rows = []
    b_rows = []

        row[n] = -1.0

    b_ub = np.array(b_rows)

def project_to_feasible(x, n):
    target = n / 2.0
    x = np.asarray(x, dtype=float).copy()
    x = np.clip(x, 0.0, 1.0)
    diff = target - np.sum(x)
    if abs(diff) > 1e-12:
        if diff > 0:
            room = 1.0 - x
            total = np.sum(room)
            if total > 1e-12:
                x += diff * room / total
        else:
            room = x.copy()
            total = np.sum(room)
            if total > 1e-12:
                x += diff * room / total
        x = np.clip(x, 0.0, 1.0)
        residual = target - np.sum(x)
        if abs(residual) > 1e-13:
            if residual > 0:
                order = np.argsort(1.0 - x)[::-1]
                for i in order:
                    add = min(residual, 1.0 - x[i])
                    if add <= 1e-15:
                        break
                    x[i] += add
                    residual -= add
                    if residual <= 1e-13:
                        break
            else:

                    if sub <= 1e-15:
                        break
                    x[i] -= sub
                    residual += sub
                    if residual >= -1e-13:
                        break
    return x

        s_flip = score(h_flip)
        if s_flip < best_score:
            best_h = h_flip
            best_score = s_flip
        t2 = min(remaining * 0.25, 200)
        if t2 > 10:
            h2, s2 = local_optimize_lp(h_flip, t2, rng)
            if s2 < best_score:
                best_h = h2
                best_score = s2
    elapsed = time.perf_counter() - start_time
    remaining = budget_s - elapsed - 15
    # Phase 3: Subgradient polish on current best
    if remaining > 15:
        t3 = min(remaining * 0.3, 200)
        h3, s3 = local_optimize_subgrad(best_h, t3, rng)
        if s3 < best_score:
            best_h = h3
            best_score = s3
    elapsed = time.perf_counter() - start_time
    remaining = budget_s - elapsed - 15
    # Phase 4: Random perturbations + local LP
    if remaining > 15:
        n_attempts = min(6, int(remaining // 30) + 1)
        for _ in range(n_attempts):
            elapsed = time.perf_counter() - start_time
            remaining = budget_s - elapsed - 10
            if remaining < 15:
                break
            pert_size = rng.uniform(0.003, 0.02)
            pert = rng.normal(0, pert_size, n)
            pert -= np.mean(pert)
            h_pert = project_to_feasible(best_h + pert, n)
            t4 = min(remaining * 0.35, 150)
            if t4 > 5:
                h4, s4 = local_optimize_lp(h_pert, t4, rng)
                if s4 < best_score:
                    best_h = h4
                    best_score = s4
    elapsed = time.perf_counter() - start_time
    remaining = budget_s - elapsed - 10
    # Phase 5: Final polish
    if remaining > 5:
        h5, s5 = local_optimize_subgrad(best_h, remaining, rng)
        if s5 < best_score:
            best_h = h5
            best_score = s5
    # Final validation
    best_h = project_to_feasible(best_h, n)
    final_score = score(best_h)
    return (best_h, float(final_score), n)
