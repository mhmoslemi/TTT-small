import numpy as np
import time

def eval_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_box_sum(x, n):
    s = n / 2.0
    x = np.array(x, dtype=float).ravel().copy()
    lo, hi = -2.0, 3.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        c = np.clip(x - mid, 0.0, 1.0)
        if c.sum() > s:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    res = np.clip(x - lam, 0.0, 1.0)
    r = s - res.sum()
    if abs(r) > 1e-12:
        if r > 0:
            slack = 1.0 - res
            ss = slack.sum()
            if ss > 1e-15:
                res += r * slack / ss
        else:
            ms = res.sum()
            if ms > 1e-15:
                res += r * res / ms
    return np.clip(res, 0.0, 1.0)

def compute_soft_grad(h, n, tau):
    dx = 2.0 / n
    raw = np.zeros(2 * n - 1)
    for k in range(2 * n - 1):
        p = k - (n - 1)
        if p >= 0:
            m = n - p
            raw[k] = dx * np.sum(h[p:p+m] * (1.0 - h[0:m]))
        else:
            q = -p
            m = n - q
            raw[k] = dx * np.sum(h[0:m] * (1.0 - h[q:q+m]))
    rmax = raw.max()
    exps = np.exp((raw - rmax) / tau)
    w = exps / exps.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        wk = w[k] * dx
        p = k - (n - 1)
        if p >= 0:
            m = n - p
            grad[p:n] += wk * (1.0 - h[0:m])
            grad[0:m] += wk * (-h[p:p+m])
        else:
            q = -p
            m = n - q
            grad[0:m] += wk * (1.0 - h[q:q+m])
            grad[q:n] += wk * (-h[0:m])
    return raw, w, grad

def smooth_grad_descent(h, n, c_start, time_limit, rng):

    best_c = c_start
    tau = max(0.008, 0.06 * c_start)
    alpha = 0.015
    no_improve = 0
    t0 = time.perf_counter()
    
    while time.perf_counter() - t0 < time_limit:
        if tau < 1e-6:
            break
        raw, w, grad = compute_soft_grad(h, n, tau)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-10:
            tau *= 0.7
            continue
        # Project to tangent (sum-zero)
        grad_tan = grad - np.mean(grad)
        direction = -grad_tan
        dn = np.linalg.norm(direction)
        if dn < 1e-12:
            break
        direction /= dn
        improved = False
        a = alpha

                break
            elif cc < c_start + 0.5 * tau:

                improved = True
                no_improve = 0
                break

        if not improved:

            no_improve += 1
            if no_improve > 20:

                no_improve = 0

def pair_descent(h, n, c_start, time_limit, rng):
    best_h = h.copy()
    best_c = c_start
    c_cur = c_start
    t0 = time.perf_counter()
    n_pairs = n * (n - 1) // 2
    fracs = [0.5, 0.25, 0.12, 0.06, 0.03, 0.015, 0.008]
    passes = 0
    
    while time.perf_counter() - t0 < time_limit and passes < 200:
        passes += 1
        improved_this_pass = False
        order = rng.integers(0, n_pairs, size=n_pairs, dtype=np.int64)
        for pi in range(n_pairs):
            if pi % 500 == 0 and passes > 5:
                if time.perf_counter() - t0 > time_limit:
                    return best_h, best_c
            idx = order[pi]
            i = int(idx // (n - 1))
            j = int(idx % (n - 1)) + i + 1
            for (a, b) in ((i, j), (j, i)):
                cap = min(h[a], 1.0 - h[b])
                if cap < 1e-10:
                    continue
                for fr in fracs:
                    d = cap * fr
                    if d < 1e-11:
                        continue
                    hb = h.copy()
                    hb[a] -= d
                    hb[b] += d
                    cc = eval_score(hb)
                    if cc < c_cur - 1e-12:
                        h = hb
                        c_cur = cc
                        if cc < best_c:
                            best_h = h.copy()
                            best_c = cc
                        improved_this_pass = True
                        break
        if not improved_this_pass:
            break
    return best_h, best_c

def sa_search(h, n, c_start, time_limit, rng):
    best_h = h.copy()
    best_c = c_start
    c_cur = c_start
    temp = 0.004
    scale = 0.08
    t0 = time.perf_counter()
    
    while time.perf_counter() - t0 < time_limit and temp > 5e-7:
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        if rng.random() < 0.5:
            cap = min(h[i], 1.0 - h[j])
            if cap < 1e-10:
                continue
            d = min(rng.uniform(0.02, 1.0) * cap * scale, cap)
            if d < 1e-10:
                d = cap * 0.1
            d = min(d, cap)
            hb = h.copy()
            hb[i] -= d
            hb[j] += d
        else:
            cap = min(h[j], 1.0 - h[i])
            if cap < 1e-10:
                continue
            d = min(rng.uniform(0.02, 1.0) * cap * scale, cap)
            if d < 1e-10:
                d = cap * 0.1
            d = min(d, cap)
            hb = h.copy()
            hb[j] -= d
            hb[i] += d
        cc = eval_score(hb)
        delta = cc - c_cur
        if delta < 0 or rng.random() < np.exp(-abs(delta) / temp):
            h = hb
            c_cur = cc
            if cc < best_c:
                best_h = h.copy()
                best_c = cc
        temp *= 0.997
        if rng.random() < 0.05:
            scale *= 0.92
    return best_h, best_c
