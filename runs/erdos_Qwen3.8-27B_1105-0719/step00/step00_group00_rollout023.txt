        a = 0.0
        js = j - s  # index where h[j] appears as second factor
        if 0 <= js < n:
            a += (1.0 - h[js])
        jp = j + s  # index where h[j] appears as first factor (1 - h[j])
        if 0 <= jp < n:
            a += (-h[jp])
        slopes[k] = a

    intercepts = corr - slopes * h[j]

    # => x = (intercepts[k2] - intercepts[k1]) / (slopes[k1] - slopes[k2])

        if obj < best_val:

            best_x = x

    return best_val

def project_feasible(h, n):
    """Project onto {0<=h<=1, sum(h)=n/2}."""
    target = n / 2.0

        diff = target - np.sum(h)
        if abs(diff) < 1e-12:

            step = min(step, max_step)

def binary_swap_delta(h, n, i, j):
    """Compute delta for each correlation element when swapping h[i]:1->0, h[j]:0->1.
    Returns array of deltas (length 2n-1)."""
    m = 2 * n - 1
    deltas = np.zeros(m)
    for k in range(m):
        s = k - (n - 1)
        d = 0
        # h[i] was 1, now 0:
        #   As first factor (m=i): lose h[i]*(1-h[i+s]) = 1*(1-h[i+s]) = (1-h[i+s]) if i+s valid
        #   As second factor (m=i-s): lose h[i-s]*(1-h[i]) = h[i-s]*0 = 0, gain h[i-s]*(1-0) = h[i-s] if i-s valid
        # h[j] was 0, now 1:
        #   As first factor (m=j): gain h[j]*(1-h[j+s]) = 1*(1-h[j+s]) = (1-h[j+s]) if j+s valid
        #   As second factor (m=j-s): gain h[j-s]*(1-h[j]) = h[j-s]*1 = h[j-s], lose h[j-s]*(1-0) = h[j-s]
        #   Wait: old: h[j-s]*(1-h[j]) = h[j-s]*1 = h[j-s]. New: h[j-s]*(1-1) = 0. Delta: -h[j-s]
        
        # Let me just compute old and new for each affected m:
        affected = set()
        for mv in [i, i-s, j, j-s]:
            if 0 <= mv < n and 0 <= mv+s < n:
                affected.add(mv)
        
        for mv in affected:
            # Old: h[mv] * (1-h[mv+s])
            old_h_m = 1.0 if mv == i else (0.0 if mv == j else h[mv])
            old_h_ms = 1.0 if (mv+s) == i else (0.0 if (mv+s) == j else h[mv+s])
            old_val = old_h_m * (1.0 - old_h_ms)
            
            # New: h_new[mv] * (1-h_new[mv+s])
            new_h_m = 0.0 if mv == i else (1.0 if mv == j else h[mv])
            new_h_ms = 0.0 if (mv+s) == i else (1.0 if (mv+s) == j else h[mv+s])
            new_val = new_h_m * (1.0 - new_h_ms)
            
            d += new_val - old_val
        
        deltas[k] = d
    return deltas

For 484 swaps, that's ~6ms per full evaluation of all swaps. We can do ~160 such evaluations per second. Over 100 seconds, that's 16,000 sweeps. More than enough.

But actually, I don't need to evaluate ALL swaps. I can use a steepest-descent approach: find the single best swap, apply it, repeat. This is O(n^3) per improvement step (n^2 swaps × O(n) delta computation). For n=44, that's ~85K operations per step. Very fast.

Let me implement the steepest-descent binary local search:

```python
def _binary_ls(h, n, time_limit=60, seed=0):
    h = h.astype(float).copy()
    ones = list(np.where(h == 1.0)[0])
    zeros = list(np.where(h == 0.0)[0])
    dx = 2.0 / n
    m = 2 * n - 1
    
    corr = np.correlate(h, 1.0 - h, mode="full")
    best_val = float(np.max(corr) * dx)
    current_max = float(np.max(corr))
    
    start = time.time()
    rng = np.random.RandomState(seed)
    stuck = 0
    
    while time.time() - start < time_limit:
        # Find best swap
        best_delta_max = 0  # We want to minimize the max, so look for most negative change in max
        best_i, best_j = -1, -1
        best_new_max = current_max
        
        # Compute deltas for all swaps
        for i in ones:
            for j in zeros:
                # Compute delta for this swap
                d = np.zeros(m)
                for k in range(m):
                    s = k - (n - 1)
                    dk = 0
                    # Affected m values
                    for mv in (i, i-s, j, j-s):
                        if 0 <= mv < n and 0 <= mv + s < n:
                            # Old
                            hm = h[mv]
                            hms = h[mv + s]
                            # If mv or mv+s is i or j, adjust
                            # Actually h already has the old values, so just use them
                            old = hm * (1.0 - hms)
                            # New
                            hm_new = 0.0 if mv == i else (1.0 if mv == j else hm)
                            hms_new = 0.0 if (mv+s) == i else (1.0 if (mv+s) == j else hms)
                            new = hm_new * (1.0 - hms_new)
                            dk += new - old
                    d[k] = dk
                
                new_corr = corr + d
                new_max = float(np.max(new_corr))
                if new_max < best_new_max - 1e-12:
                    best_new_max = new_max
                    best_i, best_j = i, j
        
        if best_i >= 0:
            # Apply the swap
            h[best_i] = 0.0
            h[best_j] = 1.0
            ones.remove(best_i)
            zeros.remove(best_j)
            ones.append(best_j)
            zeros.append(best_i)
            corr = np.correlate(h, 1.0 - h, mode="full")
            current_max = float(np.max(corr))
            best_val = max(best_val, current_max * dx)  # Wait, we want min
            best_val = current_max * dx
            corr = corr  # keep updated
            stuck = 0
        else:
            stuck += 1
            if stuck > 20:
                # Random perturbation
                if len(ones) > 0 and len(zeros) > 0:
                    for _ in range(rng.randint(1, 5)):
                        i = ones[rng.randint(len(ones))]
                        j = zeros[rng.randint(len(zeros))]
                        h[i] = 0.0
                        h[j] = 1.0
                        ones.remove(i)
                        zeros.remove(j)
                        ones.append(j)
                        zeros.append(i)
                    corr = np.correlate(h, 1.0 - h, mode="full")
                    current_max = float(np.max(corr))
                    best_val = min(best_val, current_max * dx)
                    corr = corr
                if stuck > 200:
                    break
    
    return h, best_val
```

Wait, this is O(n^2 * n) per sweep = O(n^3) for the inner loop (n^2 swaps, each O(n) for delta computation). For n=44: 44^2 * 44 ≈ 85K per sweep. In Python, each "operation" in the inner loop is maybe 100ns, so 85K * 100ns = 8.5ms per sweep. Over 60 seconds, that's ~7000 sweeps. Should be plenty.

But the pure Python triple nested loop might be slow. Let me try to vectorize the delta computation.

For a given swap (i, j), the delta at shift s is:
δ(s) = Σ_{mv ∈ {i, i-s, j, j-s} ∩ valid} [h_new[mv](1-h_new[mv+s]) - h[mv](1-h[mv+s])]

Since h is binary, h[mv](1-h[mv+s]) is 1 if h[mv]=1 and h[mv+s]=0, else 0.

So δ(s) counts how many of the affected positions change from (1,0) to not-(1,0) [decrease] or from not-(1,0) to (1,0) [increase].

Let me think of a vectorized approach:

For a fixed i (h[i]=1 going to 0):
- As first factor: position i, shift s. Affected when 0 ≤ i+s < n. Change: -(1-h[i+s])
- As second factor: position i-s, shift s. Affected when 0 ≤ i-s < n and 0 ≤ i-s+s = i < n (always true). Change: +h[i-s]

For a fixed j (h[j]=0 going to 1):
- As first factor: position j, shift s. Affected when 0 ≤ j+s < n. Change: +(1-h[j+s])
- As second factor: position j-s, shift s. Affected when 0 ≤ j-s < n. Change: -h[j-s]

But we need to handle overlaps (when i = j-s, etc.). For the binary case:

If i + s = j (i.e., the "first factor" position for i is the "first factor" position for j... no, it's that h[i+s] = h[j]):
- The "first factor at i" term uses h[i+s] = h[j] = 0 (old). Change: -(1-0) = -1.
- The "first factor at j" term uses h[j+s]. No interaction.
- The "second factor at i-s" term: if i-s = j, uses h[i-s] = h[j] = 0. Change: +0 = 0.
- The "second factor at j-s = i" term: uses h[j
