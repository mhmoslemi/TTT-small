```python
    shifted = alpha * (s - max_s)
    shifted = np.clip(shifted, -50, 50)  # for numerical stability (we only need relative weights)
    exp_shifted = np.exp(shifted)
    w = exp_shifted / np.sum(exp_shifted)
    
    grad = np.zeros(n)
    for r in range(2 * n - 1):
        if w[r] < 1e-15:
            continue
        k = r - (n - 1)
        j = np.arange(n)
        j_minus_k = j - k
        j_plus_k = j + k
        
        mask1 = (j_minus_k >= 0) & (j_minus_k < n)
        mask2 = (j_plus_k >= 0) & (j_plus_k < n)
        
        term = np.zeros(n)
        term[mask1] += dx
        # term2: -dx * h[j+k] for valid j+k
        if np.any(mask2):
            term[mask2] -= dx * h[j_plus_k[mask2]]
        # term3: -dx * h[j-k] for valid j-k
        if np.any(mask1):
            term[mask1] -= dx * h[j_minus_k[mask1]]
        
        grad += w[r] * term
    
    return grad
```

```python
def compute_c5(h, dx):
    s = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(s))

def smooth_objective(h, dx, alpha):
    s = np.correlate(h, 1.0 - h, mode="full") * dx
    max_s = np.max(s)
    shifted = alpha * (s - max_s)
    shifted = np.clip(shifted, -500, 500)
    return float(max_s + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_objective_grad(h, dx, alpha):
    n = len(h)
    s = np.correlate(h, 1.0 - h, mode="full") * dx
    max_s = np.max(s)
    shifted = alpha * (s - max_s)
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    w = exp_shifted / np.sum(exp_shifted)
    
    grad = np.zeros(n)
    for r in range(2 * n - 1):
        if w[r] < 1e-12:
            continue
        k = r - (n - 1)
        j = np.arange(n)
        j_mk = j - k
        j_pk = j + k
        
        mask_mk = (j_mk >= 0) & (j_mk < n)
        mask_pk = (j_pk >= 0) & (j_pk < n)
        
        term = np.zeros(n)
        term[mask_mk] += dx
        if np.any(mask_pk):
            term[mask_pk] -= dx * h[j_pk[mask_pk]]
        if np.any(mask_mk):
            term[mask_mk] -= dx * h[j_mk[mask_mk]]
        
        grad += w[r] * term
    
    return grad

def project_to_feasible(h, n, target):
    if n == 0:
        return h
    # Bisection for t such that sum(clip(h - t, 0, 1)) = target
    lo = np.min(h) - 1.0
    hi = np.max(h) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    return np.clip(h - t, 0.0, 1.0)

    max_s = np.max(s)
    shifted = alpha * (s - max_s)
    shifted = np.clip(shifted, -500, 500)
    return max_s + np.log(np.sum(np.exp(shifted))) / alpha

def resample_step(h_old, n_old, n_new):
    dx_old = 2.0 / n_old
    dx_new = 2.0 / n_new
    h_new = np.zeros(n_new)
    for j in range(n_new):
        start = j * dx_new
        end = (j + 1) * dx_new
        old_starts = np.arange(n_old) * dx_old
        old_ends = old_starts + dx_old
        overlaps = np.minimum(end, old_ends) - np.maximum(start, old_starts)
        overlaps = np.maximum(overlaps, 0)
        total = np.sum(overlaps)
        if total > 1e-15:
            h_new[j] = np.sum(h_old * overlaps) / total
        else:
            h_new[j] = h_old[np.clip(int(start / dx_old), 0, n_old - 1)]
    return h_new

def optimize_slsqp(h_init, n, dx, target, alpha, deadline, max_iter=300):
    bounds = [(0.0, 1.0)] * n
    constraint = [{'type': 'eq',
                   'fun': lambda x: np.sum(x) - target,
                   'jac': lambda x: np.ones(n)}]
    
    def fun(x):
        return smooth_objective(x, dx, alpha)
    
    def grad(x):
        return smooth_objective_grad(x, dx, alpha)
    
    try:
        result = minimize(fun, h_init, jac=grad, method='SLSQP',
                          bounds=bounds, constraints=constraint,
                          options={'maxiter': max_iter, 'ftol': 1e-14})
        h_out = np.clip(result.x, 0, 1)
        h_out = project_to_feasible(h_out, n, target)
        c5 = compute_c5(h_out, dx)
        return h_out, c5
    except Exception:
        return h_init.copy(), compute_c5(h_init, dx)

def optimize_projected_gradient(h_init, n, dx, target, alpha, deadline, max_iter=2000):

        else:

            lr *= 0.7
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    s = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(s))

def smooth_obj(h, dx, alpha):
    s = np.correlate(h, 1.0 - h, mode="full") * dx
    max_s = np.max(s)
    shifted = alpha * (s - max_s)
    shifted = np.clip(shifted, -500, 500)
    return float(max_s + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_obj_grad(h, dx, alpha):
    n = len(h)
    s = np.correlate(h, 1.0 - h, mode="full") * dx
    max_s = np.max(s)
    shifted = alpha * (s - max_s)
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    total = np.sum(exp_shifted)
    w = exp_shifted / total
    
    grad = np.zeros(n)
    j = np.arange(n)
    for r in range(2 * n - 1):
        if w[r] < 1e-14:
            continue
        k = r - (n - 1)
        j_mk = j - k
        j_pk = j + k
        mask_mk = (j_mk >= 0) & (j_mk < n)
        mask_pk = (j_pk >= 0) & (j_pk < n)
        term = np.zeros(n)
        if np.any(mask_mk):
            term[mask_mk] += (1.0 - h[j_mk[mask_mk]])
        if np.any(mask_pk):
            term[mask_pk] -= h[j_pk[mask_pk]]
        grad += w[r] * dx * term
    return grad

def project_feasible(h, n, target):
    lo = float(np.min(h)) - 1.0
    hi = float(np.max(h)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    t = 0.5 * (lo + hi)
    return np.clip(h - t, 0.0, 1.0)

def resample_step(h_old, n_old, n_new):
    dx_old = 2.0 / n_old
    dx_new = 2.0 / n_new
    h_new = np.zeros(n_new)
    for j in range(n_new):
        start = j * dx_new
        end = (j + 1) * dx_new
        old_idx = (np.arange(n_old) * dx_old)
        old_ends = old_idx + dx_old
        overlaps = np.maximum(0.0, np.minimum(end, old_ends) - np.maximum(start, old_idx))
        total = np.sum(overlaps)
        if total > 1e-15:
            h_new[j] = np.dot(h_old, overlaps) / total
        else:
            idx = int(np.clip(start / dx_old, 0, n_old - 1))
            h_new[j] = h_old[idx]
    return h_new

def optimize_slsqp(h_init, n, dx, target, alpha, deadline, max_iter=200):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    
    def fun(x):
        return smooth_obj(x, dx, alpha)
    
    def grad(x):
        return smooth_obj_grad(x, dx, alpha)
    
    try:
        res = minimize(fun, h_init, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-14})
        h_out = project_feasible(np.clip(res.x, 0, 1), n, target)
        return h_out, compute_c5(h_out, dx)
    except Exception:
        return h_init.copy(), compute_c5(h_init, dx)

def optimize_pg(h_init, n, dx, target, alpha, deadline, max_iter=3000):
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    lr = 1.0 / (alpha * n * dx + 1.0) * 5.0
    lr = min(lr, 0.5)
    stall = 0
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        grad = smooth_obj_grad(h, dx, alpha)
        
        # Backtracking line search
        step_lr = lr
        improved = False
        for _bt in range(20):
            h_new = project_feasible(h - step_lr * grad, n, target)
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5 - 1e-15:
                h = h_new
                best_h = h_new.copy()
                best_c5 = c5_new
                stall = 0
                improved = True
                break
            step_lr *= 0.5
        
        if not improved:
            lr *= 0.5
            stall += 1
            if stall > 80:
                break
    
    return best_h, best_c5
```
