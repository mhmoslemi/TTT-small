    h[i] = oi
    h[j] = oj
    return v

def golden_min_pair(h, i, j, max_iter=40):
    """Golden section on pair (i,j). Returns (best_raw, best_t)."""
    lo = max(-h[i], h[j] - 1.0) + 1e-12
    hi = min(1.0 - h[i], h[j]) - 1e-12
    if lo >= hi:
        return raw_max(h), 0.0
    
    phi = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = lo, hi
    x1 = b - phi * (b - a)
    x2 = a + phi * (b - a)
    f1 = eval_pair(h, i, j, x1)
    f2 = eval_pair(h, i, j, x2)
    
    for _ in range(max_iter):
        if b - a < 1e-13:
            break
        if f1 <= f2:
            b = x2; x2 = x1; f2 = f1
            x1 = b - phi * (b - a); f1 = eval_pair(h, i, j, x1)
        else:
            a = x1; x1 = x2; f1 = f2
            x2 = a + phi * (b - a); f2 = eval_pair(h, i, j, x2)
    
    t_mid = 0.5 * (a + b)
    f_mid = eval_pair(h, i, j, t_mid)
    f_zero = eval_pair(h, i, j, 0.0)
    
    if f_zero < f_mid:
        return f_zero, 0.0
    return f_mid, t_mid

def repair(h):
    n = len(h)
    h = np.array(h, dtype=np.float64).ravel().copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = h.sum() - target
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h > 1e-15
            cnt = mask.sum()
            if cnt == 0:
                break
            step = min(diff / cnt, h[mask].min() * 0.999)
            if step <= 1e-15:
                break
            h[mask] -= step
        else:
            mask = h < 1.0 - 1e-15
            cnt = mask.sum()
            if cnt == 0:
                break
            step = min(-diff / cnt, (1.0 - h[mask]).min() * 0.999)
            if step <= 1e-15:
                break
            h[mask] += step
    diff = h.sum() - target
    if abs(diff) > 1e-13:
        if diff > 0:
            idx = np.argmax(h)
            h[idx] = max(0.0, h[idx] - diff)
        else:
            idx = np.argmin(h)
            h[idx] = min(1.0, h[idx] - diff)
    return np.clip(h, 0.0, 1.0)
