def make_valid_h(arr, n):
    """Ensure h is valid: in [0,1], sum = n/2."""
    h = np.array(arr, dtype=np.float64).flatten()

        # Resample or pad
        if len(h) > n:
            h = h[:n]
        else:
            h = np.pad(h, (0, n - len(h)), mode='edge')

        else:
            mask = h > 1e-10
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
            h = np.clip(h, 0.0, 1.0)
    return h

    # Solve for coefficients
    Xmat = np.array([[x0*x0, x0, 1.0],
                     [x1*x1, x1, 1.0],
                     [x2*x2, x2, 1.0]])
    Ymat = np.array([Y0, Y1, Y2])  # shape (3, 2n-1)
    V = np.linalg.solve(Xmat, Ymat)  # shape (3, 2n-1)
    a_c, b_c, c_c = V[0], V[1], V[2]
    
    # Find optimal x
    x_best, f_best = find_min_x(a_c, b_c, c_c, low, high)
    
    # Current f value
    f_cur = float(np.max(a_c * x_cur * x_cur + b_c * x_cur + c_c))
    
    if f_best < f_cur - 1e-10:
        return (x_best, S - x_best)
    return None
