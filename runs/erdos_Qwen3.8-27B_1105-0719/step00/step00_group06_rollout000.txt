import numpy as np

def exact_objective(h):
    """Compute C5 = max(np.correlate(h, 1-h, mode='full') * dx)"""
    n = len(h)
    if n == 0:
        return float('inf')
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(profile))

def project_box_sum(h, target=None):
    """Project h onto {0<=h<=1, sum(h)=target} using water-filling"""
    n = len(h)
    x = np.array(h, dtype=np.float64).copy()
    if target is None:
        target = 0.5 * n
    # Handle non-finite
    x[~np.isfinite(x)] = 0.0
    # Iterative water-filling
    for _ in range(100):
        s = np.sum(x)
        if abs(s - target) < 1e-14:
            break
        if s > target:
            deficit = s - target
            # Subtract from elements that are > 0
            pos = x > 1e-15
            if not np.any(pos):
                break
            step_min = np.min(x[pos])
            if step_min >= deficit / np.sum(pos):
                x[pos] -= deficit / np.sum(pos)
            else:
                x[pos] -= step_min
                x[x < 1e-15] = 0.0
        else:
            deficit = target - s
            # Add to elements that are < 1
            neg = x < 1.0 - 1e-15
            if not np.any(neg):
                break
            step_min = 1.0 - np.max(x[neg])
            if step_min >= deficit / np.sum(neg):
                x[neg] += deficit / np.sum(neg)
            else:
                x[neg] += step_min
                x[x > 1.0 - 1e-15] = 1.0
    # Final clip
    x = np.clip(x, 0.0, 1.0)
    # Final exact adjustment
    s = np.sum(x)
    diff = target - s
    if abs(diff) > 1e-15:
        # Distribute remaining
        if diff > 0:
            idx = np.argsort(x)[:-int(np.ceil(np.abs(diff)))] if np.sum(x < 1) > 0 else np.arange(n)
            # Just add small amount to spread

                room = 1.0 - x[i]
                add = min(room, diff)
                x[i] += add
                diff -= add
                if abs(diff) < 1e-15:

        else:

                x[i] -= sub
                diff += sub

def compute_phi_and_grad(h, tau):
    """
    Compute the log-sum-exp smoothed objective and its gradient.
    Returns (psi, grad_psi)
    """
    n = len(h)
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    m = len(profile)  # 2n-1
    
    max_p = np.max(profile)
    shifted = (profile - max_p) / tau
    # Clip for numerical stability
    shifted = np.clip(shifted, -500, 0)
    exp_sum = np.sum(np.exp(shifted))
    psi = max_p + tau * math.log(exp_sum)
    
    # Weights
    w = np.exp(shifted) / exp_sum
    
    # Gradient computation
    grad = np.zeros(n)
    for k in range(m):
        if w[k] < 1e-15:
            continue
        # k in [0, 2n-2], corresponds to lag = k - (n-1)
        lag = k - (n - 1)
        if lag >= 0:
            # c[lag] = sum_{m=0}^{n-1-lag} h[m+lag] * (1-h[m])
            # d/dh[j] = (1-h[j-lag]) * [0 <= j-lag <= n-1-lag] - h[j+lag] * [0 <= j <= n-1-lag]
            # First term: j-lag in [0, n-1-lag] => j in [lag, n-1]
            # Second term: j in [0, n-1-lag]
            j_start1 = lag
            j_end1 = n - 1
            # (1-h[j-lag]) for j in [lag, n-1]
            grad[j_start1:j_end1+1] += w[k] * dx * (1.0 - h[j_start1-lag:j_end1+1-lag])
            j_start2 = 0
            j_end2 = n - 1 - lag
            if lag == 0:
                # Special case: c[0] = sum h[m](1-h[m]), d/dh[j] = 1-2h[j]
                # Already handled above: first term gives (1-h[j]), second gives -h[j]
                # So total = 1 - 2h[j] which is correct
                pass
            else:
                grad[j_start2:j_end2+1] += w[k] * dx * (-h[j_start2+lag:j_end2+1+lag])
        else:
            a = -lag  # a > 0
            # c[-a] = sum_{m=0}^{n-1-a} h[m] * (1-h[m+a])
            # d/dh[j] = (1-h[j+a]) * [0 <= j <= n-1-a] - h[j-a] * [a <= j <= n-1]
            j_start2 = 0
            j_end2 = n - 1 - a

            j_start1 = a
            j_end1 = n - 1
            grad[j_start1:j_end1+1] += w[k] * dx * (-h[j_start1-a:j_end1+1-a])
    
    return psi, grad

def smooth_objective_and_grad(h, tau):
    """SLSQP objective: smoothed C5 + penalty for constraint violation"""
    n = len(h)
    target = 0.5 * n
    
    # Check finite
    if not np.all(np.isfinite(h)):
        return 1e10, np.zeros(n)
    
    psi, grad = compute_phi_and_grad(h, tau)
    
    # Penalty for sum constraint
    sum_err = np.sum(h) - target
    psi += 1e6 * sum_err ** 2
    grad += 2e6 * sum_err * np.ones(n)
    
    # Penalty for box constraint
    # (SLSQP handles bounds, but add small penalty for safety)
    over = np.maximum(h - 1.0, 0.0)
    under = np.maximum(-h, 0.0)
    psi += 1e6 * (np.sum(over**2) + np.sum(under**2))
    grad += 2e6 * over - 2e6 * under
    
    return psi, grad

def run_slsqp(h0, tau_schedule, maxiter_per_stage=150):
    """Run SLSQP with decreasing tau schedule"""
    from scipy.optimize import minimize
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - 0.5*n, 'jac': lambda x: np.ones(n)}]
    
    current = np.array(h0, dtype=np.float64)
    
    for tau in tau_schedule:
        count = [0]
        def func(x):
            count[0] += 1
            val, g = smooth_objective_and_grad(x, tau)
            return val
        def jfunc(x):
            count[0] += 1  # Actually this might double count; let's use wrapped
            val, g = smooth_objective_and_grad(x, tau)
            return g
        
        options = {'maxiter': maxiter_per_stage, 'ftol': 1e-12, 'disp': False}
        
        try:
            res = minimize(func, current, method='SLSQP', jac=jfunc,
                          bounds=bounds, constraints=cons, options=options)

                c5_current = exact_objective(current)
                if c5 < c5_current - 1e-12:

        except Exception:
            pass

def binary_threshold(h, n):
    """Convert to binary with exactly n/2 ones"""
    if n % 2 != 0:
        return None
    m = n // 2
    idx = np.argsort(-h)[:m]
    b = np.zeros(n)
    b[idx] = 1.0
    return b

def binary_swap_search(h, max_rounds=50):
    """Improve binary vector via swap moves"""
    n = len(h)
    if n % 2 != 0:
        return h.copy()
    
    b = h.copy()
    ones = np.where(b > 0.5)[0]
    zeros = np.where(b < 0.5)[0]
    
    current_c = exact_objective(b)
    
    for round_idx in range(max_rounds):
        improved = False
        # Try all pairs (one, zero)
        for i in ones:
            for j in zeros:
                b[i] = 0.0
                b[j] = 1.0
                c = exact_objective(b)
                if c < current_c - 1e-12:
                    current_c = c
                    ones = np.where(b > 0.5)[0]
                    zeros = np.where(b < 0.5)[0]
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
    
    return b

def make_initial_candidates(n, parent_h, rng):

    candidates = []
    target = 0.5 * n
    
    if parent_h is not None:

        else:

            f = interp1d(x_old, parent_h, kind='linear', fill_value='extrapolate')
