def project_to_feasible(g, n):
    """Project g onto {0 <= h_i <= 1, sum h_i = n/2}"""
    target = n / 2.0
    
    # Binary search on lambda
    lo, hi = -1.0, 2.0  # lambda range
    for _ in range(60):
        mid = (lo + hi) / 2.0
        h = np.clip(g - mid, 0.0, 1.0)
        s = h.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix any floating point issues
    diff = target - h.sum()
    if abs(diff) < 1e-10:
        return h
    # Adjust
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    return h

def evaluate_c5(h):

    F = np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(g, n):

    target = n / 2.0
    # Binary search on shift
    lo = -2.0
    hi = 2.0

        h = np.clip(g - mid, 0.0, 1.0)
        s = h.sum()
        if s > target:
