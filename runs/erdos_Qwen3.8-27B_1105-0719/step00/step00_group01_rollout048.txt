```python
def objective_exact(h):
    """Compute C5 = max_k integral h(x)(1-h(x+k)) dx"""
    n = len(h)
    dx = 2.0 / n
    F = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(F), F

def project_feasible(z, target_sum):
    """Project z onto {x in [0,1]^n : sum(x) = target_sum}"""
    n = len(z)
    z = np.clip(np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0), 0.0, 1.0)
    
    # Bisection for lambda: sum clip(z_i - lambda, 0, 1) = target_sum
    lo = z.min() - 1.0
    hi = z.max() + 1.0
    
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(z - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    
    result = np.clip(z - 0.5*(lo+hi), 0.0, 1.0)
    
    # Exact repair
    residual = target_sum - result.sum()
    if abs(residual) > 1e-14:
        if residual > 0:
            # Need to add residual; add to entries with room
            idx = np.where(result < 1.0 - 1e-12)[0]
        else:
            # Need to subtract; subtract from entries > 0
            idx = np.where(result > 1e-12)[0]
        if len(idx) > 0:
            per = residual / len(idx)
            result[idx] = np.clip(result[idx] + per, 0.0, 1.0)
    
    return result

def soft_max_and_grad(h, beta, F=None):
    """Compute soft-max objective and its gradient"""
    n = len(h)
    if F is None:
        _, F = objective_exact(h)
    
    Fmax = np.max(F)
    shifted = beta * (F - Fmax)
    # Clip to avoid underflow
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    log_sum = np.log(np.sum(exp_vals) + 1e-300)
    phi = Fmax + log_sum / beta
    
    # Weights
    w = exp_vals / (np.sum(exp_vals) + 1e-300)
    
    # Gradient of soft-max = weighted average of gradients of F_l
    dx = 2.0 / n
    grad = np.zeros(n)
    
    # For each offset d, compute dF_d/dh and accumulate
    # d ranges from -(n-1) to (n-1)
    # F index: l = d + (n-1)
    for d in range(-(n-1), n):
        l = d + (n - 1)
        wl = w[l]
        if wl < 1e-14:
            continue
        
        if d > 0:
            # dF_d/dh_i = dx * [1_{d<=i<=n-1}*(1-h_{i-d}) - 1_{0<=i<=n-1-d}*h_{i+d}]
            # First term: i from d to n-1, contribution dx*(1-h_{i-d})
            idx1 = np.arange(d, n)
            grad[idx1] += dx * wl * (1.0 - h[idx1 - d])
            # Second term: i from 0 to n-1-d, contribution -dx*h_{i+d}
            idx2 = np.arange(0, n - d)
            grad[idx2] -= dx * wl * h[idx2 + d]
        elif d < 0:
            ad = -d
            # dF_d/dh_i = dx * [1_{0<=i<=n-1-ad}*(1-h_{i+ad}) - 1_{ad<=i<=n-1}*h_{i-ad}]
            # First term: i from 0 to n-1-ad, contribution dx*(1-h_{i+ad})
            idx1 = np.arange(0, n - ad)
            grad[idx1] += dx * wl * (1.0 - h[idx1 + ad])
            # Second term: i from ad to n-1, contribution -dx*h_{i-ad}
            idx2 = np.arange(ad, n)
            grad[idx2] -= dx * wl * h[idx2 - ad]
        else:
            # d = 0: dF_0/dh_i = dx*(1-2*h_i)
            grad += dx * wl * (1.0 - 2.0 * h)
    
    return phi, grad, w

def two_point_transfer(h, rng, max_delta=None):
    """Perform a random transfer move maintaining sum constraint"""
    n = len(h)
    i, j = rng.choice(n, size=2, replace=False)
    
    room_up = 1.0 - h[j]  # how much we can add to j
    room_down = h[i]      # how much we can remove from i
    
    max_d = min(room_up, room_down)
    if max_d <= 0:
        return h.copy()
    
    if max_delta is None:
        max_delta = 0.05
    delta = rng.uniform(1e-5, min(max_delta, max_d))
    
    new_h = h.copy()
    new_h[i] -= delta
    new_h[j] += delta
    return new_h

def random_perturb(h, rng, amplitude):
    """Add small random perturbation and project back"""
    noise = rng.standard_normal(len(h)) * amplitude
    return project_feasible(h + noise, len(h) / 2.0)
```

  - If d >= 0: i in [d, N-1], access h[i-d] (indices 0 to N-1-d)

```python
def compute_grad_weighted(h, w, dx):
    """Compute weighted sum of gradients of F_k"""
    n = len(h)
    grad = np.zeros(n)
    
    for d in range(-(n-1), n):
        k = d + (n - 1)
        wk = w[k]
        if wk < 1e-14:
            continue
        
        if d >= 0:
            # First term: i in [0, n-1-d], contribution (1-h[i+d])
            if n - 1 - d >= 0:
                grad[:n-d] += dx * wk * (1.0 - h[d:])
            # Second term: i in [d, n-1], contribution -h[i-d]
            if n - 1 - d >= 0:
                grad[d:] -= dx * wk * h[:n-d]
        else:
            ad = -d
            # First term: i in [ad, n-1], contribution (1-h[i-ad]) = (1-h[i+d])
            grad[ad:] += dx * wk * (1.0 - h[:n-ad])
            # Second term: i in [0, n-1-ad], contribution -h[i+ad] = -h[i-d]
            grad[:n-ad] -= dx * wk * h[ad:]
    
    return grad
```

```python
def objective_exact(h):
    n = len(h)
    dx = 2.0 / n
    F = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(F), F


def project_feasible(z, target_sum):
    z = np.nan_to_num(np.asarray(z, dtype=float), nan=0.5, posinf=1.0, neginf=0.0)
    z = np.clip(z, 0.0, 1.0)
    lo = z.min() - 1.0
    hi = z.max() + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(z - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    result = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    residual = target_sum - result.sum()
    if abs(residual) > 1e-13:
        if residual > 0:
            idx = np.where(result < 1.0 - 1e-12)[0]
        else:
            idx = np.where(result > 1e-12)[0]
        if len(idx) > 0:
            per = residual / len(idx)
            result[idx] = np.clip(result[idx] + per, 0.0, 1.0)
    return result


def compute_grad_weighted(h, w, dx):
    n = len(h)
    grad = np.zeros(n)
    for d in range(-(n - 1), n):
        k = d + (n - 1)
        wk = w[k]
        if wk < 1e-14:
            continue

            m = n - d

                grad[:m] += dx * wk * (1.0 - h[d:])

            ad = -d
            m = n - ad
            if m > 0:
                grad[ad:] += dx * wk * (1.0 - h[:m])
                grad[:m] -= dx * wk * h[ad:]
    return grad


def soft_max_obj_grad(h, beta):
    n = len(h)
    dx = 2.0 / n

    Fmax = np.max(F)
    shifted = beta * (F - Fmax)
    shifted = np.clip(shifted, -500, 0)
    exp_vals = np.exp(shifted)
    s = exp_vals.sum()
    if s < 1e-300:
        return Fmax, np.zeros(n), np.zeros_like(F)
    w = exp_vals / s
    phi = Fmax + np.log(s) / beta
    grad = compute_grad_weighted(h, w, dx)
    return phi, grad, w


def two_point_transfer(h, rng, max_delta):
    n = len(h)
    i, j = rng.choice(n, size=2, replace=False)
    room_up = 1.0 - h[j]
    room_down = h[i]
    max_d = min(room_up, room_down)
    if max_d <= 1e-12:
        return h.copy()
    delta = rng.uniform(1e-5, min(max_delta, max_d))
    new_h = h.copy()
    new_h[i] -= delta
    new_h[j] += delta
    return new_h


def random_perturb(h, rng, amplitude):
    noise = rng.standard_normal(len(h)) * amplitude
    return project_feasible(h + noise, len(h) / 2.0)


def active_lp_solve(h, F, radius, n, max_active=20):
    dx = 2.0 / n
    c5_cur = np.max(F)
    thresh = c5_cur - 0.003
    active_mask = F >= thresh
    active_indices = np.where(active_mask)[0]
    if len(active_indices) == 0:
        active_indices = np.argsort(F)[-5:]
    if len(active_indices) > max_active:

    N_act = len(active_indices)
    A_ub = np.zeros((N_act, n + 1))
    b_ub = np.zeros(N_act)
    
    for row, l in enumerate(active_indices):
        d = l - (n - 1)
        g = np.zeros(n)
        if d >= 0:
            m = n - d
```
