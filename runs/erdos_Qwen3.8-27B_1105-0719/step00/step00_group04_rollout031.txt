        diff = target - np.sum(h)

    n = len(h_cur)
    dx = 2.0 / n
    v = 1.0 - h_cur

        # For lag d, a[j] has coefficient v[j-d]*dx if 0 <= j-d <= n-1

        A_ub[k, n] = -1.0  # coefficient of t

from scipy.optimize import linprog
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def project_feasible(h):
    n = len(h)
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        # Add diff/n to all, clip, repeat
        mask = (h > 0) & (h < 1)
        if not np.any(mask):
            break
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def build_lp_matrix(n, fixed):
    """Build constraint matrix for the LP.
    fixed: the fixed vector (either v=1-h for primal, or h for dual)
    Returns A_ub, b_ub, A_eq, b_eq, c_obj, bounds
    """
    dx = 2.0 / n
    n_lags = 2 * n - 1
    A_ub = np.zeros((n_lags, n + 1))
    
    # For lag d (d = k - (n-1)), coefficient of variable a[j] is fixed[j-d]*dx if valid
    for k in range(n_lags):
        d = k - (n - 1)
        # j ranges over valid indices where 0 <= j-d < n
        j_start = max(0, d)
        j_end = min(n, n + d)  # exclusive
        for j in range(j_start, j_end):
            m = j - d
            A_ub[k, j] = fixed[m] * dx
        A_ub[k, n] = -1.0
    
    b_ub = np.zeros(n_lags)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    bounds = [(0, 1)] * n + [(0, None)]
    
    return A_ub, b_ub, A_eq, b_eq, c_obj, bounds

def solve_lp(n, fixed_vec):
    """Solve the LP: min t s.t. lag sums <= t, sum = n/2, 0<=x<=1"""
    A_ub, b_ub, A_eq, b_eq, c_obj, bounds = build_lp_matrix(n, fixed_vec)
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            x = res.x[:n]
            x = project_feasible(x)
            return x
    except:
        pass
    return None

    """Fix v=1-h, optimize h_new"""
    n = len(h)
    v = 1.0 - h
    h_new = solve_lp(n, v)
    return h_new

def try_dual_lp(h):
    """Fix h, optimize b, return h_new = 1-b"""
    n = len(h)
    b = solve_lp(n, h)
    if b is not None:
        h_new = 1.0 - b
        h_new = project_feasible(h_new)
        return h_new
    return None

def dimension_transfer(h, n_target):
    """Transfer h to a different grid size"""
    n_src = len(h)

    src_x = (np.arange(n_src) + 0.5) * 2.0 / n_src

    tgt_x = (np.arange(n_target) + 0.5) * 2.0 / n_target
    # Interpolate
    h_target = np.interp(tgt_x, src_x, h, left=h[0], right=h[-1])

import numpy as np
from scipy.optimize import linprog
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def project_feasible(h):
    n = len(h)
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(1000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-15) & (h < 1.0 - 1e-15)
        if not np.any(mask):
            # Try all elements
            mask = np.ones(n, dtype=bool)
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def lp_solve(n, fixed_vec):
    dx = 2.0 / n
    n_lags = 2 * n - 1
    A_ub = np.zeros((n_lags, n + 1))
    
    for k in range(n_lags):
        d = k - (n - 1)
        j_start = max(0, d)
        j_end = min(n, n + d)
        for j in range(j_start, j_end):
            A_ub[k, j] = fixed_vec[j - d] * dx
        A_ub[k, n] = -1.0
    
    b_ub = np.zeros(n_lags)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            x = project_feasible(res.x[:n])
            return x
    except Exception:
        pass
    return None

def try_primal(h):
    n = len(h)
    return lp_solve(n, 1.0 - h)

def try_dual(h):
    n = len(h)
    b = lp_solve(n, h)
    if b is not None:
        return project_feasible(1.0 - b)
    return None

def dim_transfer(h, n_tgt):
    n_src = len(h)
    src_x = (np.arange(n_src) + 0.5) * 2.0 / n_src
    tgt_x = (np.arange(n_tgt) + 0.5) * 2.0 / n_tgt
    h_new = np.interp(tgt_x, src_x, h, left=h[0], right=h[-1])
    return project_feasible(h_new)

def pair_optimize(h, rounds=2, n_try=30):

    dx = 2.0 / n
    score = evaluate_c5(h)
    v = 1.0 - h
    lag_sums = np.correlate(h, v, mode="full") * dx
    
    for _ in range(rounds):
        improved = False
        for p in range(n):
            for q in range(p + 1, n):
                s = h[p] + h[q]
                lo = max(0.0, s - 1.0)
                hi = min(1.0, s)
                if hi - lo < 1e-10:
                    continue
                # Compute C
                C = np.zeros(2 * n - 1)
                for d_idx in range(2 * n - 1):
                    d = d_idx - (n - 1)
                    cv = 0.0
                    jm = p - d
                    if 0 <= jm < n: cv += v[jm]
                    jp = p + d
                    if 0 <= jp < n: cv -= h[jp]
                    jm = q - d
                    if 0 <= jm < n: cv -= v[jm]
                    jp = q + d
                    if 0 <= jp < n: cv += h[jp]
                    C[d_idx] = cv * dx

                best_a = h[p]

import numpy as np
from scipy.optimize import linprog
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def project_feasible(h):
    n = len(h)
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(1000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-15) & (h < 1.0 - 1e-15)
        cnt = np.sum(mask)
        if cnt == 0:
            mask = np.ones(n, dtype=bool)
            cnt = n
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def lp_solve(n, fixed):
    dx = 2.0 / n
    n_lags = 2 * n - 1
    A_ub = np.zeros((n_lags, n + 1))
    for k in range(n_lags):
        d = k - (n - 1)
        j_lo = max(0, d)
        j_hi = min(n, n + d)
        for j in range(j_lo, j_hi):
            A_ub[k, j] = fixed[j - d] * dx
        A_ub[k, n] = -1.0
    b_ub = np.zeros(n_lags)
    A_eq = np.array([[1.0] * n + [0.0]])
    b_eq = np.array([n / 2.0])
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return project_feasible(res.x[:n])
    except Exception:
        pass
    return None

    n = len(h)
    sx = (np.arange(n) + 0.5) * 2.0 / n
    tx = (np.arange(n_tgt) + 0.5) * 2.0 / n_tgt
    return project_feasible(np.interp(tx, sx, h))

    n = len(h)
    dx = 2.0 / n
    score = evaluate_c5(h)
    v = 1.0 - h
    ls = np.correlate(h, v, mode="full") * dx
    for _ in range(rounds):
        imp = False
        for p in range(n):
            for q in range(p + 1, n):
                s = h[p] + h[q]
                lo = max(0.0, s - 1.0)
                hi = min(1.0, s)
                if hi - lo < 1e-10:
                    continue
                C = np.zeros(2 * n - 1)
                for di in range(2 * n - 1):
                    d = di - (n - 1)
                    cv = 0.0
                    jm = p - d
                    if 0 <= jm < n: cv += v[jm]
                    jp = p + d
                    if 0 <= jp < n: cv -= h[jp]
                    jm = q - d
                    if 0 <= jm < n: cv -= v[jm]
                    jp = q + d
                    if 0 <= jp < n: cv += h[jp]
                    C[di] = cv * dx
                al = np.linspace(lo, hi, n_try)
                nl = ls[None, :] + (al - h[p])[:, None] * C[None, :]
                bv = np.max(nl, axis=1)
                bi = np.argmin(bv)
                if bv[bi] < score - 1e-12:

                    h[q] = s - al[bi]
                    h = np.clip(h, 0, 1)
                    v = 1.0 - h
                    ls = np.correlate(h, v, mode="full") * dx
                    score = float(np.max(ls))
                    imp = True
        if not imp:
            break
    return h, score

                        h_lp2 = lp_solve(nt, 1.0 - h_best)
                        if h_lp2 is not None:
                            for lam in [1.0, 0.5, 0.3]:
                                h_c2 = project_feasible((1.0 - lam) * h_best + lam * h_lp2)
                                s_c2 = evaluate_c5(h_c2)
                                if s_c2 < score_best - 1e-10:
                                    h_best, score_best = h_c2, s_c2
                                    if s_c2 < best_score:
                                        best_h, best_score, best_n = h_best.copy(), s_c2, nt
                                    break
        
        # Random perturbation every 25 iterations
        if it % 25 == 0:
            for noise_scale in [0.1, 0.05]:
                h_r = h_best + rng.randn(n_best) * noise_scale
                h_r = project_feasible(h_r)
                s_r = evaluate_c5(h_r)
                if s_r < score_best - 1e-10:
                    h_best, score_best = h_r.copy(), s_r
                    if s_r < best_score:
                        best_h, best_score, best_n = h_best.copy(), s_r, n_best
                    break
        
        # Check if we should reset to best and try different n
        if it % 30 == 0 and score_best > best_score + 1e-6:
            h_best = best_h.copy()
            score_best = best_score
            n_best = best_n
            # Try a random different dimension
            nt = target_ns[rng.randint(len(target_ns))]
            if nt != n_best:
                h_best = dim_transfer(h_best, nt)
                score_best = evaluate_c5(h_best)
                n_best = nt
    
    # Final polish
    if time.time() - t0 < budget_s - 5:
        h_f, s_f = pair_opt(best_h.copy(), rounds=2, n_try=50)
        if s_f < best_score:
            best_h, best_score = h_f, s_f
        # Try LP one more time on the best
        if time.time() - t0 < budget_s - 3:
            h_lp = lp_solve(best_n, 1.0 - best_h)
            if h_lp is not None:
                for lam in [1.0, 0.5, 0.3]:
                    h_c = project_feasible((1.0 - lam) * best_h + lam * h_lp)
                    s_c = evaluate_c5(h_c)
                    if s_c < best_score:
                        best_h, best_score = h_c, s_c
                        break
    
    best_h = project_feasible(best_h)
    best_score = evaluate_c5(best_h)
    return (best_h, best_score, len(best_h))
