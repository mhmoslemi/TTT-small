def local_search_binary(h_init, max_iters=100000, time_limit=300):
    """Local search for binary h"""
    n = len(h_init)
    h = np.zeros(n)
    # Make h binary with sum = n/2
    h[h_init >= 0.5] = 1.0
    # Adjust to get exactly n/2 ones
    n_ones = int(np.sum(h))
    target_ones = n // 2
    if n_ones > target_ones:
        # Remove excess ones
        one_indices = np.where(h == 1)[0]
        np.random.shuffle(one_indices)
        for idx in one_indices[:n_ones - target_ones]:
            h[idx] = 0
    elif n_ones < target_ones:
        # Add needed ones
        zero_indices = np.where(h == 0)[0]
        np.random.shuffle(zero_indices)
        for idx in zero_indices[:target_ones - n_ones]:
            h[idx] = 1
    
    corr = compute_corr(h)
    best_c5 = np.max(corr) * (2.0/n)
    
    one_positions = set(np.where(h == 1)[0].tolist())
    zero_positions = set(np.where(h == 0)[0].tolist())
    
    start_time = time.time()
    stalled = 0
    
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        # Random swap
        if not one_positions or not zero_positions:
            break
        i = np.random.choice(list(one_positions))
        j = np.random.choice(list(zero_positions))
        
        # Try the swap
        h[i], h[j] = h[j], h[i]
        new_corr = compute_corr(h)
        new_c5 = np.max(new_corr) * (2.0/n)
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            one_positions.remove(i)
            one_positions.add(j)
            zero_positions.remove(j)
            zero_positions.add(i)
            corr = new_corr
            stalled = 0
        else:
            # Revert
            h[i], h[j] = h[j], h[i]
            stalled += 1
    
    return h, best_c5

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def evaluate_c5(h):

    return np.max(compute_corr(h)) * (2.0 / n)

def make_binary(h, seed=42):
