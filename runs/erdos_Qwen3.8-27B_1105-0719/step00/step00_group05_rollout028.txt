            s_diff = np.sum(h_new) - np.sum(h)
            if abs(s_diff) > 1e-10:
                j = np.random.randint(0, n)
                while j == i:
                    j = np.random.randint(0, n)
                h_new[j] = np.clip(h[j] - s_diff, 0, 1)
            
            c5_new = compute_c5(h_new)
            c5_old = compute_c5(h)
            
            if c5_new < c5_old or np.random.random() < np.exp(-(c5_new - c5_old) / T):
                h = h_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
    
    return best_h, best_c5, len(best_h)

import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def _make_obj(n, beta):
    def obj(h):
        dx = 2.0 / n
        corr = np.correlate(h, 1.0 - h, mode="full") * dx
        m = np.max(corr)
        shifted = np.clip((corr - m) * beta, -100.0, 0.0)
        val = m + np.log(np.sum(np.exp(shifted)) + 1e-30) / beta
        s = np.sum(h) - n / 2.0
        val += 200.0 * s * s
        return val
    return obj

def _optimize(h_start, n, beta, maxiter, method):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda h, n=n: np.sum(h) - n / 2.0,
             'jac': lambda h, n=n: np.ones(n)}]
    obj = _make_obj(n, beta)
    res = minimize(obj, h_start, method=method, bounds=bounds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10})
    h_opt = np.clip(res.x, 0.0, 1.0)
    s = np.sum(h_opt)
    if s > 1e-12:
        h_opt *= (n / 2.0) / s
        h_opt = np.clip(h_opt, 0.0, 1.0)
    return h_opt

def _fix_sum(h, n):
    s = np.sum(h)
    if s > 1e-12:
        h = h * (n / 2.0) / s
    return np.clip(h, 0.0, 1.0)
