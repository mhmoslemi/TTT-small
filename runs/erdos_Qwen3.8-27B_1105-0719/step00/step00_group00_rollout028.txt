def project_onto_feasible(h, target_sum):
    h = np.array(h, dtype=float)
    h = np.clip(h, 0, 1)

        return h

        if proj.sum() > target_sum:
            lo = mid
        else:
            hi = mid

import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def project_onto_feasible(h, target_sum):
    h = np.array(h, dtype=float)
    h = np.clip(h, 0, 1)
    lo, hi = -1.0, 1.0
    for _ in range(100):
        mid = (lo + hi) / 2
        proj = np.clip(h + mid, 0, 1)
        if proj.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) / 2, 0, 1)

def subgradient_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    c5_raw = np.max(corr)
    active = np.where(corr >= c5_raw - 1e-7)[0]
    k_vals = np.arange(n)
    grad = np.zeros(n)
    for s in active:
        lag = s - (n - 1)
        idx1 = k_vals - lag
        idx2 = k_vals + lag
        mask1 = (idx1 >= 0) & (idx1 < n)
        mask2 = (idx2 >= 0) & (idx2 < n)
        term1 = np.zeros(n)
        term1[mask1] = 1.0 - h[idx1[mask1]]
        term2 = np.zeros(n)
        term2[mask2] = -h[idx2[mask2]]
        grad += term1 + term2
    grad *= dx / len(active)
    return grad

def solve_active_lp(h, delta_max):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    c5_raw = np.max(corr)
    tol = 1e-4
    active = np.where(corr >= c5_raw - tol)[0]
    if len(active) == 0:
        return None
    k_vals = np.arange(n)
    A_rows = []
    b_rows = []
    for s in active:
        lag = s - (n - 1)
        idx1 = k_vals - lag
        idx2 = k_vals + lag
        mask1 = (idx1 >= 0) & (idx1 < n)
        mask2 = (idx2 >= 0) & (idx2 < n)
        t1 = np.zeros(n)
        t1[mask1] = 1.0 - h[idx1[mask1]]
        t2 = np.zeros(n)
        t2[mask2] = -h[idx2[mask2]]
        g = dx * (t1 + t2)
        b_s = dx * corr[s]
        row = np.zeros(n + 1)
        row[:n] = g
        row[n] = -1.0
        A_rows.append(row)
        b_rows.append(-b_s)
    A_ub = np.array(A_rows)
    b_ub = np.array(b_rows)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    lower = np.zeros(n + 1)
    upper = np.zeros(n + 1)
    for i in range(n):
        lower[i] = -min(h[i], delta_max)
        upper[i] = min(1.0 - h[i], delta_max)
    lower[n] = -1.0
    upper[n] = 1.0
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=list(zip(lower, upper)), method='highs')
        if res.status == 0:
            return res.x[:n]
    except:
        pass
    return None
