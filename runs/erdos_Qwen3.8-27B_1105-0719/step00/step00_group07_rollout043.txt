from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def h_from_a(a, n):
    m = n // 2
    h = np.empty(n)
    h[:m] = a
    if n % 2 == 1:
        h[m] = 0.5
        h[m+1:] = 1.0 - a[::-1]
    else:
        h[m:] = 1.0 - a[::-1]
    return h

def a_from_h(h):
    n = len(h)
    m = n // 2
    return h[:m].copy()

def smooth_objective_and_grad(a, n, beta):
    h = h_from_a(a, n)
    # Compute correlation values
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Smooth objective
    max_corr = np.max(corr)
    shifted = beta * (corr - max_corr)
    weights = np.exp(shifted)
    weights /= np.sum(weights)
    F = (max_corr + np.log(np.sum(np.exp(shifted)))) / beta
    # Compute gradient of each corr[k] w.r.t. h
    # Then reduce to gradient w.r.t. a
    grad_h = np.zeros_like(h)
    for k in range(2 * n - 1):
        s = (n - 1) - k  # shift
        w = weights[k]
        if w < 1e-15:
            continue
        if s >= 0:
            # S[k] = sum_{i=0}^{n-1-s} h[i]*(1-h[i+s])
            upper = n - 1 - s
            for i in range(upper + 1):
                grad_h[i] += w * (1.0 - h[i + s])
                grad_h[i + s] += w * (-h[i])
        else:
            q = -s
            # S[k] = sum_{i=q}^{n-1} h[i]*(1-h[i-q])
            for i in range(q, n):
                grad_h[i] += w * (1.0 - h[i - q])
                grad_h[i - q] += w * (-h[i])
    # Reduce to gradient w.r.t. a
    m = n // 2
    grad_a = grad_h[:m] - grad_h[n-1-m::-1]
    # Wait, need to be careful with indexing
    # h[n-1-i] corresponds to a[i] for i = 0, ..., m-1
    # So ∂F/∂a[i] = ∂F/∂h[i] - ∂F/∂h[n-1-i]

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def h_from_a(a, n):
    m = n // 2
    h = np.empty(n)
    h[:m] = a
    if n % 2 == 1:
        h[m] = 0.5
        h[m+1:] = 1.0 - a[::-1]
    else:
        h[m:] = 1.0 - a[::-1]
    return h


def smooth_obj_and_grad(a, n, beta):
    h = h_from_a(a, n)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    shifted = beta * (corr - max_c)
    exp_s = np.exp(shifted)
    sum_exp = np.sum(exp_s)
    if sum_exp == 0:
        return 0.0, np.zeros_like(a)
    weights = exp_s / sum_exp
    F = (max_c + np.log(sum_exp)) / beta
    grad_h = np.zeros(n)
    for k in range(2 * n - 1):
        s = (n - 1) - k
        w = weights[k]
        if w < 1e-15:
            continue
        if s >= 0:
            grad_h[:n - s] += w * (1.0 - h[s:])
            grad_h[s:] += w * (-h[:n - s])
        else:
            q = -s
            grad_h[q:] += w * (1.0 - h[:-q])
            grad_h[:-q] += w * (-h[q:])
    m2 = n // 2
    grad_a = grad_h[:m2] - grad_h[n - 1::-1][:m2]
    dx = 2.0 / n

            best_val = compute_c5(h_full)
            for alpha in [0.1, 0.05, 0.02, 0.01, 0.005]:
                h_trial = project_full(h_full - alpha * grad_tan, target)
                c5_trial = compute_c5(h_trial)
                if c5_trial < best_val - 1e-14:
                    best_val = c5_trial
                    best_alpha = alpha
                    h_full = h_trial
                    break
            if best_alpha == 0.0:
                break
        c5_full = compute_c5(h_full)
        if
</br>c5_full < best_c5:
            best_c5 = c5_full
            best_h = h_full.copy()

    # Phase 5: Final coordinate descent polish (if time)
    if remaining() > 20:
        h_sym2 = 0.5 * (best_h + (1.0 - best_h[::-1]))
        h_sym2 = np.clip(h_sym2, 0.0, 1.0)
        a2 = h_sym2[:m].copy()
        max_sw = 40 if remaining() > 60 else 15
        a_final, c5_final = coordinate_descent(a2, n, max_sweeps=max_sw, t0=t0, time_limit=budget_s - 10)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_from_a(a_final, n)

    # Final validation
    final_h = np.clip(best_h, 0.0, 1.0)
    # Ensure mass constraint
    mass_diff = target - np.sum(final_h)
    if abs(mass_diff) > 1e-10:
        final_h = project_full(final_h, target)
    final_c5 = compute_c5(final_h)

    return (final_h, final_c5, len(final_h))
