            shifted = np.clip(beta * (corr - m), -500, 500)
            exp_s = np.exp(shifted)
            w_sum = exp_s.sum()
            if w_sum == 0:
                break
            weights = exp_s / w_sum
            
            grad = np.zeros(n)
            for t in range(2 * n - 1):

                w = weights[t]
                if w < 1e-8:

                p_hi = min(n, l + n)
                if p_lo < p_hi:
                    idx = np.arange(p_lo, p_hi)
                    grad[idx] += w * (1.0 - h[idx - l])
                p_lo2 = max(0, -l)
                p_hi2 = min(n, n - l)
                if p_lo2 < p_hi2:
                    idx = np.arange(p_lo2, p_hi2)
                    grad[idx] -= w * h[idx + l]

            else:
                stall += 1
                lr *= 0.8

                    break

import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def max_corr_value(h):
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")))

def project_box_hyperplane(y, target):
    n = len(y)
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    x = np.clip(y - lam, 0.0, 1.0)
    r = target - x.sum()
    if abs(r) > 1e-14:
        if r > 0:
            order = np.argsort(1.0 - x)
            for i in order:
                add = min(r, 1.0 - x[i])
                if add <= 0:
                    break
                x[i] += add
                r -= add
                if r <= 1e-15:
                    break
        else:
            order = np.argsort(x)
            for i in order:
                sub = min(-r, x[i])
                if sub <= 0:
                    break
                x[i] -= sub
                r += sub
                if r >= -1e-15:
                    break
    return x
