where w_k = exp(alpha * s[k]) / sum_j exp(alpha * s[j])

def _upsample(x, new_n):
    """Upsample a signal using linear interpolation."""
    old_n = len(x)
    old_x = np.linspace(0, 2, old_n, endpoint=False)

def _downsample(x, new_n):
    """Downsample by averaging blocks."""
    old_n = len(x)
    block_size = old_n // new_n
    remainder = old_n % new_n
    result = []
    idx = 0
    for i in range(new_n):
        if i < remainder:
            end = idx + block_size + 1
        else:
            end = idx + block_size
        result.append(np.mean(x[idx:end]))
        idx = end
    result = np.array(result)
    target = new_n / 2.0
    result = result + (target - np.sum(result)) / new_n
    result = np.clip(result, 0.0, 1.0)

_g_alpha = 100.0

def _correlation_values(x):
    """Compute dx * correlate(x, 1-x, full)."""
    n = len(x)
    dx = 2.0 / n
    return np.correlate(x, 1.0 - x, mode="full") * dx

def _compute_c5(x):
    """Compute the C5 value (max correlation)."""
    return float(np.max(_correlation_values(x)))

def _loss_grad(x, alpha):
    """Compute log-sum-exp smoothed max and its gradient."""
    n = len(x)
    dx = 2.0 / n
    s = np.correlate(x, 1.0 - x, mode="full") * dx

    s_max = np.max(s)
    shifted = np.clip(alpha * (s - s_max), -500.0, 500.0)
    exp_s = np.exp(shifted)
    sum_exp = np.sum(exp_s)
    log_sum = np.log(sum_exp)
    loss = s_max + log_sum / alpha

    weights = exp_s / sum_exp

    # Gradient: ds[r]/dx[m] where k_r = r - (n-1)
    # = dx * [ (1-x[m-k_r]) if 0<=m-k_r<n  -  x[m+k_r] if 0<=m+k_r<n ]
    grad = np.zeros(n)
    for r in range(2 * n - 1):
        k_r = r - (n - 1)
        wr = weights[r]
        if wr < 1e-15:
            continue
        # Term 1: (1-x[m-k_r]) for m in [max(0,k_r), min(n, n+k_r))
        m_lo = max(0, k_r)
        m_hi = min(n, n + k_r)
        if m_lo < m_hi:
            idx = np.arange(m_lo, m_hi) - k_r
            grad[m_lo:m_hi] += wr * dx * (1.0 - x[idx])
        # Term 2: -x[m+k_r] for m in [max(0,-k_r), min(n, n-k_r))
        m_lo2 = max(0, -k_r)
        m_hi2 = min(n, n - k_r)
        if m_lo2 < m_hi2:
            idx2 = np.arange(m_lo2, m_hi2) + k_r
            grad[m_lo2:m_hi2] -= wr * dx * x[idx2]

    return loss, grad

    return _loss_grad(x, _g_alpha)[0]

    return _loss_grad(x, _g_alpha)[1]

    x = np.clip(x, 0.0, 1.0)

        # Try uniform adjustment
        x_adj = x - step
        if np.all(x_adj >= 0.0) and np.all(x_adj <= 1.0):
            x = x_adj
        else:

                else:

                    amt = min(-d, 1.0 - x[idx], 1e-4)
                    x[idx] += amt
    return x

def _upsample(x, new_n):
    """Upsample via linear interpolation, fix sum."""
    old_n = len(x)
    old_pts = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_pts = np.linspace(0.0, 2.0, new_n, endpoint=False)
    new_vals = np.interp(new_pts, old_pts, x)
    target = new_n / 2.0
    new_vals = new_vals + (target - np.sum(new_vals)) / new_n
    return np.clip(new_vals, 0.0, 1.0)

def _downsample(x, new_n):
    """Downsample by block averaging, fix sum."""
    old_n = len(x)
    if old_n < new_n:
        return _upsample(x, new_n)
    block = old_n // new_n
    rem = old_n % new_n
    result = np.zeros(new_n)
    idx = 0
    for i in range(new_n):
        size = block + (1 if i < rem else 0)
        result[i] = np.mean(x[idx:idx + size])
        idx += size
    target = new_n / 2.0
    result = result + (target - np.sum(result)) / new_n
    return np.clip(result, 0.0, 1.0)
