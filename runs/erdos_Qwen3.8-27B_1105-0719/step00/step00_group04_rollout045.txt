def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))

def repair_feasible(vec, target_sum):
    n = len(vec)
    x = np.array(vec, dtype=np.float64)
    x = np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0)
    x = np.clip(x, 0.0, 1.0)
    s = x.sum()
    rem = target_sum - s
    if abs(rem) < 1e-14:
        return x
    if rem > 0:
        room = 1.0 - x
        order = np.argsort(-room)
        for idx in order:
            add = min(room[idx], rem)
            x[idx] += add
            rem -= add
            if rem <= 1e-15:
                break
    else:
        rem = -rem
        order = np.argsort(x)
        for idx in order:
            sub = min(x[idx], rem)
            x[idx] -= sub
            rem -= sub
            if rem <= 1e-15:
                break
    return x

def compute_profile(h):

def interpolate_to_n(h, n_new):
    n = len(h)
    if n == n_new:
        return h.copy()

    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    # Pad with edge values for interpolation
    x_ext = np.concatenate([[x_old[0] - 1.0/n], x_old, [x_old[-1] + 1.0/n]])
    h_ext = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(x_new, x_ext, h_ext)
    h_new = np.clip(h_new, 0.0, 1.0)
    return h_new

    m = profile.max()
    if m == 0:
        return 0.0
    shifted = beta * (profile - m)
    return m + (1.0/beta) * np.log(np.sum(np.exp(shifted)))

    profile = profile_raw * dx
    m = profile.max()

    weights = np.exp(shifted)
    weights /= weights.sum()
    
    grad = np.zeros(n)

        w = weights[m_idx]
        if w < 1e-10:

            grad[idx] += w * dx * (1.0 - h[idx + delta])
            grad[idx + delta] += w * dx * (-h[idx])
        else:

            grad[jdx + d] += w * dx * (1.0 - h[jdx])
            grad[jdx] += w * dx * (-h[jdx + d])
    return grad
