import numpy as np
from scipy.optimize import minimize
import time

def compute_overlaps(h, dx):
    return np.correlate(h, 1.0 - h, mode='full') * dx

def c5_value(h, dx):
    return float(np.max(compute_overlaps(h, dx)))

def smooth_max_and_grad(h, dx, n, tau):
    ov = compute_overlaps(h, dx)
    M = float(np.max(ov))
    scaled = (ov - M) / tau
    s_max = float(np.max(scaled))
    exponents = np.exp(scaled - s_max)
    exponents_sum = float(exponents.sum())
    p = exponents / exponents_sum
    val = M + tau * (s_max + np.log(exponents_sum))
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = float(p[k])
        if pk < 1e-13:
            continue
        lag = k - (n - 1)
        if lag == 0:
            grad += pk * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:] += pk * (1.0 - h[:n - lag])
            if n - 1 - lag >= 0:
                grad[:n - lag] -= pk * h[lag:]
        else:
            m = -lag
            if n - 1 - m >= 0:
                grad[:n - m] += pk * (1.0 - h[m:])
            grad[m:] -= pk * h[:n - m]
    return val, grad

def project_sum_box(x, target):

        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)

        return np.ones(self.n)
