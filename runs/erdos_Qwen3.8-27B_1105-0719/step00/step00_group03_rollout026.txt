def compute_scores(h, dx):
    return dx * np.correlate(h, 1.0 - h, mode="full")

def smooth_max_obj_grad(h, alpha, dx):
    n = len(h)
    scores = compute_scores(h, dx)
    max_s = np.max(scores)
    z = alpha * (scores - max_s)
    z = np.minimum(z, 0.0)
    exp_z = np.exp(z)
    sum_exp = np.sum(exp_z)
    obj = max_s + (1.0 / alpha) * np.log(sum_exp)
    w = exp_z / sum_exp
    n_scores = len(scores)
    grad = np.zeros(n)
    for k in range(n_scores):
        wk = w[k]
        if wk < 1e-14:
            continue
        j_min1 = max(0, k - n + 1)
        j_max1 = min(n - 1, k)
        if j_min1 <= j_max1:
            L = np.arange(j_min1, j_max1 + 1) - k + n - 1
            grad[j_min1:j_max1 + 1] += wk * dx * (1.0 - h[L])
        j_min2 = max(0, n - 1 - k)
        j_max2 = min(n - 1, 2 * n - 2 - k)
        if j_min2 <= j_max2:
            R = np.arange(j_min2, j_max2 + 1) + k - n + 1
            grad[j_min2:j_max2 + 1] -= wk * dx * h[R]
    return obj, grad
