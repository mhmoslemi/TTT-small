        remaining = budget_s - 30 - elapsed

    # Phase 5: Final local search
    elapsed = time.time() - start_time
    remaining = budget_s - 3 - elapsed
    if remaining > 3:
        target_final = n / 2.0
        best_h, best_c5 = _local_search(best_h, target_final, time_limit=remaining)

    # Final feasibility projection
    best_h = project_feasible(best_h, n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n
