            #   - wm_pos * [h[k+m] if k<n-m else 0] * dx

    grad_y = np.clip(grad_y, -1e6, 1e6)

def project_feasible(h):

    target = n / 2.0

    for _ in range(80):

    lam = (lo + hi) / 2.0
    return _sigmoid(y + lam)


def smooth_obj_grad_in_h(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    v = corr * dx
    v_max = np.max(v)
    shifted = beta * (v - v_max)
    shifted = np.clip(shifted, -80, 80)
    e = np.exp(shifted)
    e_sum = np.sum(e)
    F = v_max + np.log(e_sum) / beta
    w = e / e_sum
    G = np.zeros(n)
    G += w[n - 1] * dx * (1.0 - 2.0 * h)
    for m in range(1, n):
        wm_pos = w[n - 1 + m]
        wm_neg = w[n - 1 - m]
        if wm_pos < 1e-15 and wm_neg < 1e-15:
            continue
        for k in range(m, n):
            G[k] += dx * (wm_pos * (1.0 - h[k - m]) - wm_neg * h[k - m])
        for k in range(0, n - m):
            G[k] += dx * (wm_neg * (1.0 - h[k + m]) - wm_pos * h[k + m])
    return F, G


def smooth_obj_grad_y(y, beta):
    h = feasible_from_logit(y)
    F, G = smooth_obj_grad_in_h(h, beta)
    m = h * (1.0 - h)
    M = np.sum(m)
    if M < 1e-12:
        return F, np.zeros_like(y)
    B = np.dot(G, m) / M
    grad_y = m * (G - B)
    grad_y = np.clip(grad_y, -1e6, 1e6)
    if not np.all(np.isfinite(grad_y)):
        grad_y = np.zeros_like(y)
    return F, grad_y


def _bfgs_objective(y, beta):
    return smooth_obj_grad_y(y, beta)


def _bfgs_objective2(y, beta):
    return smooth_obj_grad_y(y, beta)


def _pair_obj_scalar(t, h, a, b):
    h_test = h.copy()
    h_test[a] += t
    h_test[b] -= t
    h_test = np.clip(h_test, 0.0, 1.0)
    return exact_C(h_test)


def pair_polish(h, deadline, min_reserve=30.0, max_rounds=50, seed=42):
    n = len(h)
    h = h.copy()
    c = exact_C(h)
    no_improve = 0
    rng = np.random.default_rng(seed)

    for rnd in range(max_rounds):
        if time.time() > deadline - min_reserve:
            break
        if no_improve >= 8:
            break

        # Get subgradient-based importance
        F, G = smooth_obj_grad_in_h(h, 500)
        abs_G = np.abs(G)

        # Build candidate pairs
        candidates = set()
        top_k = min(10, n)
        top_idx = np.argsort(-abs_G)[:top_k]
        for a in top_idx:
            for b in range(n):
                if b != a:
                    candidates.add((min(a, b), max(a, b)))
        for _ in range(30):
            a = int(rng.integers(n))
            b = int(rng.integers(n))
            if a != b:
                candidates.add((min(a, b), max(a, b)))

        # Also try adjacent pairs
        for i in range(n - 1):
            candidates.add((i, i + 1))

        cand_list = list(candidates)
        rng.shuffle(cand_list)
        cand_list = cand_list[:50]

        improved_this_round = False
        for (a, b) in cand_list:
            if time.time() > deadline - min_reserve:
                break

            t_lo = max(-h[a], h[b] - 1.0)
            t_hi = min(1.0 - h[a], h[b])
            if t_hi - t_lo < 1e-12:
                continue

            # Grid search
            n_grid = 21
            t_grid = np.linspace(t_lo, t_hi, n_grid)
            best_t = 0.0
            best_c_t = c

            for t_val in t_grid:
                if abs(t_val) < 1e-14:
                    continue
                h_test = h.copy()
                h_test[a] += t_val
                h_test[b] -= t_val
                h_test = np.clip(h_test, 0.0, 1.0)
                c_test = exact_C(h_test)
                if c_test < best_c_t - 1e-13:
                    best_c_t = c_test
                    best_t = t_val

            # Refine with golden section
            if best_t != 0.0:
                margin = (t_hi - t_lo) / n_grid
                lo_r = max(best_t - margin, t_lo)
                hi_r = min(best_t + margin, t_hi)
                if hi_r - lo_r > 1e-10:
                    try:
                        res = minimize_scalar(
                            _pair_obj_scalar, args=(h, a, b),
                            bounds=(lo_r, hi_r), method='bounded',
                            options={'xatol': 1e-12, 'maxiter': 40}
                        )
                        if res.fun < best_c_t - 1e-13:
                            best_c_t = float(res.fun)
                            best_t = float(res.x)
                    except Exception:
                        pass

            if best_c_t < c - 1e-13:
                h[a] += best_t
                h[b] -= best_t
                h = np.clip(h, 0.0, 1.0)
                c = best_c_t
                improved_this_round = True
                no_improve = 0
                break

        if not improved_this_round:
            no_improve += 1

    return h, c


def try_other_dimensions(h_best, c_best, n_best, deadline, min_reserve=60):
    """Try a few nearby dimensions by interpolation."""
    rng = np.random.default_rng(123)
    for n_try in [n_best - 4, n_best + 4, n_best + 8]:
        if n_try < 16 or n_try > 80:
            continue
        if time.time() > deadline - min_reserve:
            break

        # Interpolate
        x_old = (np.arange(n_best) + 0.5) / n_best
        x_new = (np.arange(n_try) + 0.5) / n_try
        h_interp = np.interp(x_new, x_old, h_best)
        h_interp = project_feasible(h_interp)
        c_interp = exact_C(h_interp)

        if c_interp < c_best:
            h_best = h_interp
            c_best = c_interp
            n_best = n_try

        # Quick BFGS on this dimension
        if time.time() > deadline - min_reserve:
            break
        y_try = logit_from_h(h_best if n_try == len(h_best) else project_feasible(h_interp))
        y_try = logit_from_h(np.interp(x_new, x_old, h_best))
        y_try = logit_from_h(project_feasible(np.interp(x_new, x_old, h_best)))

        for beta in [100, 300]:
            if time.time() > deadline - min_reserve:
                break
            try:
                res = minimize(_bfgs_objective, y_try, args=(beta,),
                               method='BFGS', jac=True,
                               options={'maxiter': 80, 'gtol': 1e-6})
                if res.x is not None and np.all(np.isfinite(res.x)):
                    h_c = feasible_from_logit(res.x)
                    h_c = project_feasible(h_c)
                    c_c = exact_C(h_c)
                    if c_c < c_best:
                        c_best = c_c
                        h_best = h_c
                        n_best = n_try
            except Exception:
                pass

    return h_best, c_best, n_best
