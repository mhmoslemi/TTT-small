                int2 = np.where((result > 1e-10) & (result < 1.0 - 1e-10))[0]
                if len(int2) > 0:
                    result[int2[0]] += residual2
                    result = np.clip(result, 0.0, 1.0)
    return result

def optimize_pair(h, p, q, n_grid=80):
    n = len(h)
    s = h[p] + h[q]
    lo = max(0.0, s - 1.0)
    hi = min(1.0, s)
    if hi - lo < 1e-13:
        return None, None

    h0 = h.copy(); h0[p] = 0.0; h0[q] = s
    h05 = h.copy(); h05[p] = 0.5; h05[q] = s - 0.5
    h1 = h.copy(); h1[p] = 1.0; h1[q] = s - 1

    dx = 2.0 / n
    f0 = np.correlate(h0, 1.0 - h0, mode="full") * dx
    f05 = np.correlate(h05, 1.0 - h05, mode="full") * dx
    f1 = np.correlate(h1, 1.0 - h1, mode="full") * dx

    A = 2.0 * f0 - 4.0 * f05 + 2.0 * f1
    B = -3.0 * f0 + 4.0 * f05 - f1.0
    C = f0

    a_grid = np.linspace(lo, hi, n_grid)
    scores = np.empty(n_grid)
    for i in range(n_grid):
        a = a_grid[i]
        scores[i] = np.max(A * a * a + B * a + C)

    best_idx = int(np.argmin(scores))
    if best_idx > 0:
        gl = a_grid[best_idx - 1]
    else:
        gl = lo
    if best_idx < n_grid - 1:
        gr = a_grid[best_idx + 1]
    else:
        gr = hi

    gr_inv = (np.sqrt(5.0) - 1.0) / 2.0
    x1 = gl + (1.0 - gr_inv) * (gr - gl)
    x2 = gl + gr_inv * (gr - gl)
    f1_m = np.max(A * x1 * x1 + B * x1 + C)
    f2_m = np.max(A * x2 * x2 + B * x2 + C)

    for _ in range(35):
        if f1_m < f2_m:
            gr = x2; x2 = x1; f2_m = f1_m
            x1 = gl + (1.0 - gr_inv) * (gr - gl)
            f1_m = np.max(A * x1 * x1 + B * x1 + C)
        else:
            gl = x1; x1 = x2; f1_m = f2_m
            x2 = gl + gr_inv * (gr - gl)
            f2_m = np.max(A * x2 * x2 + B * x2 + C)

    a_opt = (gl + gr) / 2.0
    candidates = [lo, hi, a_opt, (lo + hi) / 2.0]
    for k in range(len(A)):
        if abs(A[k]) > 1e-14:
            v = -B[k] / (2.0 * A[k])
            if lo <= v <= hi:
                candidates.append(v)
    for k in range(len(A)):
        for l2 in range(k + 1, len(A)):
            if abs(A[k]) > 1e-14 and abs(A[l2]) > 1e-14:
                dA = A[k] - A[l2]
                dB = B[k] - B[l2]
                dC = C[k] - C[l2]
                disc = dB * dB - 4.0 * dA * dC
                if disc >= 0:
                    sq = np.sqrt(disc)
                    r1 = (-dB - sq) / (2.0 * dA)
                    r2 = (-dB + sq) / (2.0 * dA)
                    if lo <= r1 <= hi:
                        candidates.append(r1)
                    if lo <= r2 <= hi:
                        candidates.append(r2)

    best_a = None
    best_sc = float('inf')
    for a_c in candidates:
        a_c = float(np.clip(a_c, lo, hi))
        h_new = h.copy()
        h_new[p] = a_c
        h_new[q] = s - a_c
        if np.min(h_new) < -1e-10 or np.max(h_new) > 1.0 + 1e-10:
            continue
        h_new = np.clip(h_new, 0.0, 1.0)
        sc = exact_score(h_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_c

    if best_a is None:
        return None, None
    h_new = h.copy()
    h_new[p] = best_a
    h_new[q] = s - best_a
    h_new = np.clip(h_new, 0.0, 1.0)
    return h_new, best_sc

def pair_descent(h, rng, time_limit, start_time, max_passes=200, n_grid=60):
    n = len(h)
    best_h = h.copy()
    best_score = exact_score(best_h)
    pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
    no_improve_passes = 0

    for pidx in range(max_passes):
        if time.perf_counter() - start_time > time_limit:
            break
        rng.shuffle(pairs)
        improved = False
        n_tried = 0
        for p, q in pairs:
            if time.perf_counter() - start_time > time_limit:
                break
            n_tried += 1
            h_new, sc_new = optimize_pair(best_h, p, q, n_grid=n_grid)
            if h_new is not None and sc_new < best_score - 1e-12:
                best_h = h_new
                best_score = sc_new
                improved = True

        if not improved:
            no_improve_passes += 1
            if no_improve_passes >= 3:
                break
        else:
            no_improve_passes = 0
            if pidx > 0 and pidx % 20 == 0:
                n_grid = max(40, n_grid - 5)

    return best_h, best_score

def gradient_direction(h, top_k=10):
    n = len(h)
    prof = exact_profile(h)
    C = prof.max()
    thresh = C - max(1e-8, 0.001 * C)
    active = np.where(prof >= thresh)[0]
    if len(active) > top_k:
        order = np.argsort(prof[active])[::-1][:top_k]
        active = active[order]
    if len(active) == 0:
        active = np.array([int(np.argmax(prof))])

    n_lags = len(prof)
    grad = np.zeros(n)
    shift_range = np.arange(-(n - 1), n)

    for k_idx in active:
        shift = k_idx - (n - 1)
        for i in range(n):
            g = 0.0
            j2 = i - shift
            if 0 <= j2 < n:
                g += (1.0 - h[j2])
            j1 = i + shift
            if 0 <= j1 < n:
                g += (-h[j1])
            grad[i] += g

    dx = 2.0 / n
    grad *= dx
    return grad, active

def lp_direction_search(h, rng, time_limit, start_time, max_solves=10):
    n = len(h)
    best_h = h.copy()
    best_score = exact_score(best_h)

    for attempt in range(max_solves):
        if time.perf_counter() - start_time > time_limit:
            break
        grad, active = gradient_direction(best_h, top_k=15)
        if grad is None or np.all(np.abs(grad) < 1e-15):
            break

        # LP: minimize t s.t. grad_k . d + c_k <= t, sum(d)=0, |d_i|<=1
        # Variables: d (n), t (1) -> total n+1
        # Constraints: for each active k: grad_k . d - t <= 0
        #              sum(d) = 0
        #              -1 <= d_i <= 1

        m = len(active)
        A_ub = np.zeros((m + n, n + 1))
        b_ub = np.zeros(m + n)

        for i_act in range(m):
            # grad[active[i_act]] . d - t <= 0
            # But we need per-lag gradients, not the averaged one
            # Let me recompute per-lag gradient
            pass

        # Simpler: use the averaged gradient as a descent direction
        # Project onto sum=0
        grad_mean = grad - np.mean(grad)
        norm_d = np.linalg.norm(grad_mean)
        if norm_d < 1e-15:
            break
        d = -grad_mean / norm_d

        # Line search
        alphas = np.array([0.5, 0.3, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002])
        found = False
        for alpha in alphas:
            h_try = best_h + alpha * d
            h_try = project_feasible(h_try)
            sc = exact_score(h_try)
            if sc < best_score - 1e-12:
                best_h = h_try
                best_score = sc
                found = True
                break

        if not found:
            break

    return best_h, best_score

def try_resolution_transfer(h_src, n_tgt, rng, time_limit, start_time):
    if n_tgt <= 0 or n_tgt > 200:
        return None, None
    n_src = len(h_src)
    if n_src == n_tgt:
        return h_src.copy(), exact_score(h_src)

    xs_src = (np.arange(n_src) + 0.5) / n_src * 2.0
    xs_tgt = (np.arange(n_tgt) + 0.5) / n_tgt * 2.0

    # Add boundary points for interpolation
    x_ext = np.concatenate([[-0.1], xs_src, [2.1]])
    y_ext = np.concatenate([[0.0], h_src, [0.0]])
    h_tgt = np.interp(xs_tgt, x_ext, y_ext)
    h_tgt = project_feasible(h_tgt)
    sc = exact_score(h_tgt)
    if not np.isfinite(sc):
        return None, None
    return h_tgt, sc

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.perf_counter()
    hard_deadline = start_time + budget_s - 2.0
    work_deadline = start_time + budget_s * 0.88

    rng = np.random.default_rng(seed)

    # Get initial solution
    best_h = None
    best_score = float('inf')
    best_n = 0

    try:
        h_init = np.asarray(initial_h_values, dtype=np.float64).copy()
        if len(h_init) > 0 and np.all(np.isfinite(h_init)):
            h_init = project_feasible(h_init)
            sc = exact_score(h_init)
            if np.isfinite(sc):
                best_h = h_init.copy()
                best_score = sc
                best_n = len(h_init)
    except Exception:
        pass

    if best_h is None:
        n_fb = 70
        best_h = np.full(n_fb, 0.5)
        best_score = exact_score(best_h)
        best_n = n_fb

    target = 0.3808

    # Try symmetry
    if len(best_h) > 2:
        h_sym = 0.5 * (best_h + best_h[::-1])
        h_sym = project_feasible(h_sym)
        sc_sym = exact_score(h_sym)
        if sc_sym < best_score - 1e-12:
            best_h = h_sym
            best_score = sc_sym

    # Main pair descent
    n_base = len(best_h)
    time_for_pairs = min(work_deadline - time.perf_counter(), budget_s * 0.5)
    if time_for_pairs > 5:
        h_opt, sc_opt = pair_descent(best_h, rng, time_for_pairs, start_time, max_passes=300, n_grid=80)
        if sc_opt < best_score - 1e-12:
            best_h = h_opt
            best_score = sc_opt

    # Second round with different grid
    elapsed = time.perf_counter() - start_time
    if elapsed < budget_s * 0.7 and best_score > target + 1e-6:
        time_for_pairs2 = min(work_deadline - time.perf_counter(), budget_s * 0.15)
        if time_for_pairs2 > 3:
            h_opt2, sc_opt2 = pair_descent(best_h, rng, time_for_pairs2, start_time, max_passes=100, n_grid=100)
            if sc_opt2 < best_score - 1e-12:
                best_h = h_opt2
                best_score = sc_opt2

    # LP direction search
    elapsed = time.perf_counter() - start_time
    if elapsed < budget_s * 0.8 and best_score > target + 1e-6:
        time_lp = min(work_deadline - time.perf_counter(), 30.0)
        if time_lp > 2:
            h_lp, sc_lp = lp_direction_search(best_h, rng, time_lp, start_time, max_solves=8)
            if sc_lp < best_score - 1e-12:
                best_h = h_lp
                best_score = sc_lp

    # Second pair descent if improved
    elapsed = time.perf_counter() - start_time
    if elapsed < budget_s * 0.85 and best_score > target + 1e-6:
        time_p3 = min(work_deadline - time.perf_counter(), 30.0)
        if time_p3 > 3:
            h_opt3, sc_opt3 = pair_descent(best_h, rng, time_p3, start_time, max_passes=80, n_grid=80)
            if sc_opt3 < best_score - 1e-12:
                best_h = h_opt3
                best_score = sc_opt3

    # Try resolution transfers
    elapsed = time.perf_counter() - start_time
    if elapsed < budget_s * 0.9:
        n_src = len(best_h)
        candidates_n = []
        if n_src % 2 == 0:
            candidates_n.extend([n_src - 4, n_src + 4, n_src - 8, n_src + 8])
        else:
            candidates_n.extend([n_src + 1, n_src - 1, n_src + 3, n_src - 3])
        candidates_n = [cn for cn in candidates_n if 30 <= cn <= 150 and cn != n_src]

        for cn in candidates_n[:3]:
            if time.perf_counter() - start_time > work_deadline:
                break
            h_tr, sc_tr = try_resolution_transfer(best_h, cn, rng, work_deadline, start_time)
            if h_tr is not None and sc_tr < best_score:
                # Quick local optimization
                t_rem = work_deadline - time.perf_counter()
                if t_rem > 5:
                    h_tr, sc_tr = pair_descent(h_tr, rng, min(t_rem, 20.0), start_time, max_passes=50, n_grid=60)
                if sc_tr < best_score:
                    best_h = h_tr
                    best_score = sc_tr
                    best_n = cn

    # SLSQP attempt if time remains
    elapsed = time.perf_counter() - start_time
    if elapsed < budget_s * 0.93 and best_score > target + 1e-6:
        n_sl = len(best_h)
        if n_sl <= 100:
            def slsqp_obj(x):
                x = np.clip(x, 0.0, 1.0)
                return exact_score(x)

            def slsqp_constraint(x):
                return np.sum(x) - n_sl / 2.0

            cons = [{'type': 'eq', 'fun': slsqp_constraint}]
            bounds = [(0.0, 1.0)] * n_sl

            try:
                res = minimize(slsqp_obj, best_h.copy(), method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 50, 'ftol': 1e-10})
                if res.success or res.fun < best_score:
                    h_sl = project_feasible(res.x)
                    sc_sl = exact_score(h_sl)
                    if sc_sl < best_score - 1e-12:
                        best_h = h_sl
                        best_score = sc_sl
            except Exception:
                pass

    # Final validation
    best_h = np.asarray(best_h, dtype=np.float64)
    best_h = np.clip(best_h, 0.0, 1.0)
    # Ensure sum constraint
    target_sum = len(best_h) / 2.0
    if abs(np.sum(best_h) - target_sum) > 1e-10:
        best_h = project_feasible(best_h)

    if not np.all(np.isfinite(best_h)):
        best_h = np.full(70, 0.5)
        best_score = exact_score(best_h)

    final_score = exact_score(best_h)

    return (best_h, float(final_score), int(len(best_h)))
