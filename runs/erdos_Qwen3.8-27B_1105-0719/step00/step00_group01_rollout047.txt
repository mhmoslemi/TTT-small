def project_to_simplex_box(z, target_sum):
    """Project z onto {x : 0 <= x_i <= 1, sum(x) = target_sum}."""
    n = len(z)
    lo = -2.0
    hi = 2.0

        if s > target_sum:

        else:

    if abs(resid) > 1e-14:

        else:

            result = np.clip(result, 0.0, 1.0)
    return result

    """Compute soft-max of F and its gradient."""
    F_arr = compute_F(h)
    F_max = np.max(F_arr)
    n = len(h)
    dx = 2.0 / n
    
    # Soft-max
    shifted = beta * (F_arr - F_max)
    shifted = np.clip(shifted, -50.0, 0.0)
    exp_vals = np.exp(shifted)
    logsum = np.log(exp_vals.sum() + 1e-300)
    Phi = F_max + logsum / beta
    
    # Weights
    weights = exp_vals / (exp_vals.sum() + 1e-300)

    grad = np.zeros(n)
    for k in range(len(F_arr)):
        w = weights[k]
        if w < 1e-15:

        grad[m_idx] -= dx * w * h[first_idx]
    
    return Phi, grad

        else:

def project_box_sum(z, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5

            lo = mid
        else:
