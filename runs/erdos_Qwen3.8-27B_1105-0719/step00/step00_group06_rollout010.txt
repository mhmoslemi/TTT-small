def project_to_feasible(h, n):
    """Project h to [0,1]^n with sum = n/2"""
    h = np.clip(h, 0, 1)
    target = n / 2.0
    # Iterative projection
    for _ in range(100):
        diff = h.sum() - target
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            # need to decrease, reduce elements that are > 0
            mask = h > 0
            count = mask.sum()
            if count == 0:
                break
            step = min(diff / count, h[mask].min())
            h[mask] -= step
        else:
            # need to increase, increase elements that are < 1
            mask = h < 1
            count = mask.sum()
            if count == 0:
                break
            step = min(-diff / count, (1 - h[mask]).min())
            h[mask] += step
    h = np.clip(h, 0, 1)
    # Final exact adjustment
    diff = h.sum() - target
    if abs(diff) > 1e-14:
        # Distribute among non-bound elements
        mask = (h > 1e-10) & (h < 1 - 1e-10)
        if mask.sum() > 0:
            h[mask] -= diff / mask.sum()
            h = np.clip(h, 0, 1)
    return h

        # D(-k) = sum h[i+k] * (1 - h[i]) for k = 1..n-1

        # profile has 2n-1 entries, lag goes from -(n-1) to +(n-1)
        # profile[n-1] is lag 0
        return profile.max()
    
    current_max = max_disagreement(h_bin)
    best_max = current_max
    best_h = h_bin.copy()
    
    for it in range(max_iters):
        if timeout is not None and time.time() - start_time > timeout:
            break
        
        # Find positions of 1s and 0s
        ones = np.where(h_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]
        
        # Try random swaps

        improved = False

                if i == j:

                h_bin[j] = 1

                    best_swap = (i, j)
                    if new_max < current_max - 1:  # Early termination for big improvement
                        break
            if best_swap and best_swap_val < current_max - 1:

            h_bin[j] = 1
            current_max = best_swap_val

        else:

Now let me think about the continuous optimization.

The LP-based approach:
At current point h (feasible), compute:
- Profile values: P_s = dx * profile_internal[s] for each lag s
- Gradients: G_s for each active lag s

LP:
minimize t
subject to:
  For each active lag s: G_s · x + (P_s - G_s · h) ≤ t  [linearization]
  Actually: P_s(h) + G_s · (x - h) ≤ t
  Which is: G_s · x ≤ t + h·G_s - P_s(h)
  
  0 ≤ x ≤ 1
  sum(x) = n/2

Variables: x[0..n-1], t (total n+1 variables)
Constraints: |active| inequality + 2n bound + 1 equality

For |active| = 20 (say), that's 20 + 2n + 1 = 20 + 140 + 1 = 161 constraints, 71 variables. This is a small LP that HiGHS can solve in milliseconds.

The issue is: the linearization overestimates the bilinear function (since for h ∈ [0,1], h*(1-h) ≤ h*(1-h_current) + ∂/∂h * (h-h_current) is not quite right... actually, h*(1-h) is concave, so the linearization at h_current is an UPPER bound on the true function for h near h_current but not globally).

Wait: f(h_i, h_{i+k}) = h_i * (1 - h_{i+k})
This is bilinear, not concave or convex globally. But on the hyperrectangle [0,1]², it's:
- Linear in h_i (for fixed h_{i+k}): f = h_i * (1 - h_{i+k}), which is linear in h_i
- Linear in h_{i+k} (for fixed h_i): f = h_i * (1 - h_{i+k}) = h_i - h_i * h_{i+k}, which is linear in h_{i+k}
- But jointly, it's bilinear.

The sum F_s = ∑ h_i * (1-h_{i+k}) is also bilinear in the full vector h.

For the LP approach, we use the first-order Taylor approximation:
F_s(x) ≈ F_s(h) + ∇F_s(h) · (x - h)

This is an affine function of x, so the LP is valid. The issue is that the approximation might not be tight, so we need to verify the actual objective after moving.

The standard approach is:
1. Solve LP to get x_LP
2. Set x_new = h + α * (x_LP - h) for α ∈ (0, 1]
3. Check if F(x_new) < F(h). If yes, accept. If not, reduce α and retry.
4. If α → 0 and no improvement, add the current linearization as a "cut" and resolve.

But for our purposes, we can be simpler:
1. Solve LP
2. Check actual objective of x_LP (projected to feasible)
3. If improved, accept. Otherwise, try x_new = h + 0.5*(x_LP - h), then h + 0.25*(x_LP - h), etc.
4. If none work, move to next strategy.

import numpy as np
import time
import sys

def evaluate_c5(h):
    """Compute C5 = max of np.correlate(h, 1-h, 'full') * dx"""
    n = len(h)
    dx = 2.0 / n
    profile = np.correlate(h.astype(np.float64), 1.0 - h.astype(np.float64), mode='full')
    return profile.max() * dx

def get_profile(h):
    """Get the full profile scaled by dx"""
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h.astype(np.float64), 1.0 - h.astype(np.float64), mode='full') * dx

def project_feasible(h, n):
    """Project h to [0,1]^n with sum = n/2 exactly"""
    h = np.array(h, dtype=np.float64).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = h.sum() - target
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h > 0
            cnt = mask.sum()
            if cnt == 0: break
            step = min(diff / cnt, h[mask].min() - 1e-15)
            if step <= 0: break
            h[mask] -= step
        else:
            mask = h < 1
            cnt = mask.sum()
            if cnt == 0: break
            step = min(-diff / cnt, 1.0 - h[mask].max() - 1e-15)
            if step <= 0: break
            h[mask] += step
    h = np.clip(h, 0.0, 1.0)
    # Final correction
    diff = h.sum() - target
    if abs(diff) > 1e-14:
        # Distribute among interior points
        mask = (h > 1e-8) & (h < 1.0 - 1e-8)
        if mask.sum() > 0:
            h[mask] -= diff / mask.sum()
        else:
            # All at bounds, need to move some
            if diff > 0:
                idx = np.argmax(h)
            else:
                idx = np.argmin(h)
            h[idx] -= diff / 1  # This might not work perfectly
    h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, n):
    """Convert h to a binary vector with exactly n/2 ones"""
    # Rank by value, set top n/2 to 1
    order = np.argsort(-h)
    b = np.zeros(n)
    b[order[:n//2]] = 1.0
    return b

def max_sw_search(h_bin, max_iters=5000, timeout_s=60, time_start=None, first_improve_after=500):
    """Binary local search using swap moves"""
    n = len(h_bin)
    h = h_bin.astype(np.float64).copy()
    if time_start is None:
        time_start = time.time()
    
    def profile_max(h):
        p = np.correlate(h, 1.0 - h, mode='full')
        return p.max()
    
    current_max = profile_max(h)
    best_max = current_max
    best_h = h.copy()
    no_improve_count = 0
    
    for it in range(max_iters):
        if time.time() - time_start > timeout_s:
            break
        
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        improved = False
        best_swap = None
        best_swap_val = current_max
        check_all = (it < first_improve_after)
        
        # Try swaps
        n_ones = len(ones)
        n_zeros = len(zeros)
        
        for ii in range(n_ones):
            i = ones[ii]
            for jj in range(n_zeros):
                j = zeros[jj]
                if i == j:
                    continue
                # Perform swap
                h[i] = 0.0
                h[j] = 1.0
                new_max = profile_max(h)
                # Undo swap
                h[i] = 1.0
                h[j] = 0.0
                
                if new_max < best_swap_val:
                    best_swap_val = new_max
                    best_swap = (i, j)
                    if not check_all:
                        break  # First improvement
            if not check_all and best_swap is not None:
                break
        
        if best_swap is not None:
            i, j = best_swap
            h[i] = 0.0
            h[j] = 1.0
            current_max = best_swap_val
            no_improve_count = 0
            if current_max < best_max:
                best_max = current_max
                best_h = h.copy()
        else:
            no_improve_count += 1
            # Try random perturbation to escape local optimum
            if no_improve_count >= 5:
                i = ones[np.random.randint(n_ones)]
                j = zeros[np.random.randint(n_zeros)]
                h[i] = 0.0
                h[j] = 1.0
                current_max = profile_max(h)
                no_improve_count = 0
            if no_improve_count > 20:
                break
        
        if best_max <= 10:  # Good enough for n=70?
            break
    
    return best_h, best_max * (2.0 / n)

def continuous_lp_refine(h, timeout_s=30, time_start=None):
    """LP-based continuous refinement"""
    from scipy.optimize import linprog
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    if time_start is None:
        time_start = time.time()
    
    h = project_feasible(h.copy(), n)
    current_c5 = evaluate_c5(h)
    best_c5 = current_c5
    best_h = h.copy()
    
    for iteration in range(200):
        if time.time() - time_start > timeout_s:
            break
        
        # Compute profile and gradients
        profile_raw = np.correlate(h, 1.0 - h, mode='full')
        profile = profile_raw * dx
        current_max = profile.max()
        
        # Select active lags (top K)
        K = min(2 * n - 1, 30)
        sorted_idx = np.argsort(-profile)
        active = sorted_idx[:K]
        
        # Build gradient matrix (K x n) for active lags
        # profile_raw has 2n-1 entries, indexed from -(n-1) to (n-1)
        # active[k] is the index into profile_raw
        # lag = active[k] - (n-1)
        
        G = np.zeros((K, n))
        for k_idx in range(K):
            lag = active[k_idx] - (n - 1)  # actual lag
            if lag == 0:
                # F0 = sum h[i]*(1-h[i]), grad = 1-2h[i]
                G[k_idx, :] = (1.0 - 2.0 * h) * dx
            elif lag > 0:
                # D(-lag) = sum_{n=0}^{n-1-lag} h[n+lag]*(1-h[n])
                # d/dh[i]: (1-h[i-lag]) for i >= lag, and -h[i+lag] for i + lag < n
                for i in range(n):
                    grad = 0.0
                    if i >= lag and (i - lag) < n:
                        grad += (1.0 - h[i - lag])
                    if (i + lag) < n:
                        grad += -h[i + lag]
                    G[k_idx, i] = grad * dx
            else:
                # lag < 0, let m = -lag > 0
                m = -lag
                # D(m) = sum_{i=0}^{n-1-m} h[i]*(1-h[i+m])
                # d/dh[i]: (1-h[i+m]) for i < n-m, and -h[i-m] for i >= m
                for i in range(n):
                    grad = 0.0
                    if i < n - m:
                        grad += (1.0 - h[i + m])
                    if i >= m:
                        grad += -h[i - m]
                    G[k_idx, i] = grad * dx
        
        # Solve LP: minimize t
        # s.t. G[k] @ x - t <= b[k] for each active k
        #      0 <= x_i <= 1
        #      sum(x) = n/2
        #
        # where b[k] = profile[k] - G[k] @ h + t... 
        # Actually: F_active_k(h) + G_active_k @ (x - h) <= t
        # => G_active_k @ x <= t + G_active_k @ h - F_active_k(h)
        # => G_active_k @ x - t <= G_active_k @ h - F_active_k(h)
        
        b = G @ h - profile[active]
        
        # Variables: x[0..n-1], t
        # c = [0, ..., 0, 1] (minimize t)
        c = np.zeros(n + 1)
        c[n] = 1.0
        
        # A_ub: G[k] @ x - t <= b[k]
        A_ub = np.zeros((K, n + 1))
        A_ub[:, :n] = G
        A_ub[:, n] = -1.0
        b_ub = b
        
        # A_eq: sum(x) = n/2
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([target])
        
        # Bounds: 0 <= x_i <= 1, t >= 0
        bounds = [(0, 1)] * n + [(0, None)]
        
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
        except:
            break
        
        if not res.success:
            break
        
        x_lp = res.x[:n]
        x_lp = np.clip(x_lp, 0, 1)
        x_lp = project_feasible(x_lp, n)
        
        # Try different step sizes
        accepted = False
        for alpha in [1.0, 0.5, 0.25, 0.1, 0.05]:
            x_try = h + alpha * (x_lp - h)
            x_try = project_feasible(x_try, n)
            c5_try = evaluate_c5(x_try)
            if c5_try < current_c5 - 1e-10:

                current_c5 = c5_try
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = x_try.copy()
                accepted = True
                break
        
        if not accepted:
            # Try a small random perturbation
            # Break after a few failures
            break
    
    return best_h, best_c5
