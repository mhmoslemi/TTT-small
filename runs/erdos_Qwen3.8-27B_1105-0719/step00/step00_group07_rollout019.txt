Let me think about the gradient computation more carefully.

For np.correlate(h, 1-h, mode="full"), the output has length 2n-1.
The element at index t corresponds to:
- For t >= n-1: sum over i from 0 to t-(n-1) of h[n-1+i-(t-(n-1))] * (1-h[i])... 

Actually, let me be more careful. np.correlate(a, v, mode="full")[k] = sum_i a[i+k] * v[i] for valid i.

So np.correlate(h, 1-h, mode="full")[k] = sum_i h[i+k] * (1-h[i]) for valid i (where both indices are in [0, n-1]).

For k >= 0: i ranges from 0 to n-1-k, so sum_{i=0}^{n-1-k} h[i+k]*(1-h[i])
For k < 0: let k' = -k, i ranges from k' to n-1, so sum_{i=k'}^{n-1} h[i-k']*(1-h[i])

The output index in "full" mode: for arrays of length n, the output has length 2n-1, and index n-1 corresponds to lag 0.

So output[t] where t = n-1 + k gives the correlation at lag k.

The gradient of output[t] with respect to h[p]:
- output[t] = sum_i h[i+k]*(1-h[i]) where k = t - (n-1)
- For a specific p:
  - If p is used as the "first index" (i+k = p, so i = p-k): contributes (1-h[p-k]) if p-k is valid
  - If p is used as the "second index" (i = p): contributes -h[p+k] if p+k is valid

So ∂output[t]/∂h[p] = (1-h[p-k]) * indicator(0 <= p-k < n) - h[p+k] * indicator(0 <= p+k < n)

where k = t - (n-1).

For the log-sum-exp smoothing:
F = log(sum_t exp(beta * (output[t] - max_t output[t]))) / beta + max_t output[t]

∂F/∂h[p] = sum_t w_t * ∂output[t]/∂h[p]
where w_t = exp(beta * (output[t] - max)) / sum_u exp(beta * (output[u] - max))

This gradient computation is O(n^2) per step which is fine for n=95.

def project_box_hyperplane(y, target):
    """Project y onto {0 <= x <= 1, sum(x) = target}"""
    n = len(y)
    # Binary search for lambda
    lo, hi = y.min() - 1.0, y.max() + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        x = np.clip(y - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    x = np.clip(y - lam, 0.0, 1.0)
    # Fix residual
    residual = target - x.sum()
    if abs(residual) > 1e-12:
        if residual > 0:
            # Add to entries with slack
            order = np.argsort(1.0 - x)  # most slack first
            for i in order:
                add = min(residual, 1.0 - x[i])
                x[i] += add
                residual -= add
                if abs(residual) < 1e-14:
                    break
        else:
            # Remove from entries with mass
            order = np.argsort(x)  # least first
            for i in order:
                rem = min(-residual, x[i])
                x[i] -= rem
                residual += rem
                if abs(residual) < 1e-14:
                    break
    return x

def smooth_objective(h, beta):
    """Compute log-sum-exp smoothed objective and its gradient"""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    shifted = beta * (corr - m)
    # Numerical stability
    shifted = np.clip(shifted, -700, 700)
    exps = np.exp(shifted)
    logsumexp = m + np.log(exps.sum()) / beta
    w = exps / exps.sum()
    
    # Gradient: dw[p] = sum_t w[t] * dcorr[t]/dh[p]
    # corr[t] = sum_i h[i+k]*(1-h[i]) where k = t - (n-1)
    # dcorr[t]/dh[p] = (1-h[p-k])*I(0<=p-k<n) - h[p+k]*I(0<=p+k<n)
    
    grad = np.zeros(n)
    for t in range(len(corr)):
        k = t - (n - 1)
        wt = w[t]
        if wt < 1e-15:
            continue
        # p acts as first index: i + k = p => i = p - k, valid if 0 <= p-k < n
        # contributes (1-h[p-k])
        # p acts as second index: i = p, contributes -h[p+k] if 0 <= p+k < n
        for p in range(n):
            g = 0.0
            if 0 <= p - k < n:
                g += (1.0 - h[p - k])
            if 0 <= p + k < n:
                g -= h[p + k]
            grad[p] += wt * g
    return logsumexp, grad

def compute_grad(corr, h, w, n):

    for t in range(len(corr)):
        if w[t] < 1e-15:
            continue
        k = t - (n - 1)
        wt = w[t]
        # First index: p in range such that 0 <= p-k < n
        p_lo = max(0, k) if k > 0 else 0
        p_hi = min(n, n + k) if k < 0 else n
        # Actually: 0 <= p-k < n => k <= p < n+k
        # So p ranges from max(0,k) to min(n-1, n-1+k), inclusive
        p_start = max(0, k)
        p_end = min(n-1, n-1+k)
        if p_start <= p_end:
            idx = np.arange(p_start, p_end+1)
            grad[idx] += wt * (1.0 - h[idx - k])
        
        # Second index: p in range such that 0 <= p+k < n
        # 0 <= p+k < n => -k <= p < n-k
        p_start2 = max(0, -k)
        p_end2 = min(n-1, n-1-k)
        if p_start2 <= p_end2:
            idx2 = np.arange(p_start2, p_end2+1)
            grad[idx2] -= wt * h[idx2 + k]
    return grad

def optimize_binary(h0, max_time, start_time, rng):
    """Simulated annealing on binary sequences"""
    n = len(h0)
    if n % 2 != 0:
        return None, None
    
    # Round to binary
    h = (h0 >= 0.5).astype(float)
    # Adjust to get exactly n/2 ones
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    if len(ones) > n // 2:
        # Remove excess ones
        excess = len(ones) - n // 2
        for i in range(excess):
            h[ones[i]] = 0
    elif len(ones) < n // 2:
        deficit = n // 2 - len(ones)
        for i in range(deficit):
            h[zeros[i]] = 1
    
    corr = compute_corr(h)
    current_max = np.max(corr)
    best_max = current_max
    best_h = h.copy()
    
    T = 0.5
    T_min = 0.01
    cooling = 0.999
    
    while time.time() - start_time < max_time and T > T_min:
        # Pick a random swap
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        
        # Try swap
        h[i], h[j] = h[j], h[i]
        new_corr = compute_corr(h)
        new_max = np.max(new_corr)
        
        delta = new_max - current_max
        if delta < 0 or rng.random() < np.exp(-delta / T):
            current_max = new_max
            if current_max < best_max:
                best_max = current_max
                best_h = h.copy()
        else:
            h[i], h[j] = h[j], h[i]
        
        T *= cooling
    
    return best_h, (2.0 / n) * best_max

def project_box_hyperplane(y, target):
    """Project y onto {x: 0<=x_i<=1, sum(x)=target}"""
    n = len(y)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    x = np.clip(y - lam, 0.0, 1.0)
    # Fix floating point residual
    resid = target - x.sum()
    if abs(resid) > 1e-12:
        if resid > 0:
            order = np.argsort(-(1.0 - x))
            for i in order:
                if resid <= 1e-14:
                    break
                add = min(resid, 1.0 - x[i])

import numpy as np
import time

def project_box_hyperplane(y, target):
    n = len(y)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    x = np.clip(y - lam, 0.0, 1.0)
    resid = target - x.sum()
    if abs(resid) > 1e-12:
        if resid > 0:
            order = np.argsort(-(1.0 - x))
            for i in order:
                if resid <= 1e-14:
                    break
                add = min(resid, 1.0 - x[i])
                x[i] += add
                resid -= add
        else:
            order = np.argsort(x)
            for i in order:
                if -resid <= 1e-14:
                    break
                rem = min(-resid, x[i])
                x[i] -= rem
                resid += rem
    return x

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def c5_value(h):
    n = len(h)
    return (2.0 / n) * np.max(compute_corr(h))

def smooth_grad(h, beta):
    n = len(h)
    corr = compute_corr(h)
    m = corr.max()
    shifted = np.clip(beta * (corr - m), -700.0, 700.0)
    exps = np.exp(shifted)
    total = exps.sum()
    w = exps / total
    value = m + np.log(total) / beta
    grad = np.zeros(n)
    for t in range(len(corr)):
        if w[t] < 1e-12:
            continue
        k = t - (n - 1)
        wt = w[t]
        ps = max(0, k)
        pe = min(n - 1, n - 1 + k)
        if ps <= pe:
            idx = np.arange(ps, pe + 1)
            grad[idx] += wt * (1.0 - h[idx - k])
        ps2 = max(0, -k)
        pe2 = min(n - 1, n - 1 - k)
        if ps2 <= pe2:
            idx2 = np.arange(ps2, pe2 + 1)
            grad[idx2] -= wt * h[idx2 + k]
    return value, grad, corr

def optimize_cont(h0, deadline):
    n = len(h0)
    target = n / 2.0
    h = project_box_hyperplane(h0.copy(), target)
    v = np.zeros(n)
    mu = 0.85
    best_h = h.copy()
    best_val = c5_value(h)
    beta = 40.0
    lr = 0.008
    stalled = 0
    while time.time() < deadline:
        _, grad, _ = smooth_grad(h, beta)
        v = mu * v - lr * grad
        h_new = project_box_hyperplane(h + v, target)
        new_val = c5_value(h_new)
        if new_val < best_val:
            best_val = new_val
            best_h = h_new.copy()
            h = h_new
            stalled = 0
        else:
            h = h_new
            stalled += 1
            if stalled % 80 == 0:
                lr *= 0.85
                v *= 0.3
            if stalled > 400:
                v = np.zeros(n)
                h = project_box_hyperplane(h + 0.015 * np.random.randn(n), target)
                lr = 0.008
                beta = max(25.0, beta * 0.75)
                stalled = 0
            if lr < 1e-9:
                break
    return best_h, best_val

def optimize_binary_sa(h0, deadline, rng):
    n = len(h0)
    if n % 2 != 0:
        return None, None
    h = (h0 >= 0.5).astype(float).copy()
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    if len(ones) > n // 2:
        idx = rng.choice(ones, len(ones) - n // 2, replace=False)
        h[idx] = 0
    elif len(ones) < n // 2:
        idx = rng.choice(zeros, n // 2 - len(ones), replace=False)
        h[idx] = 1
    cur_max = compute_corr(h).max()
    best_max = cur_max
    best_h = h.copy()
    T = 0.8
    cool = 0.9997
    min_T = 0.03
    while time.time() < deadline and T > min_T:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        h[i], h[j] = h[j], h[i]
        new_max = compute_corr(h).max()
        delta = new_max - cur_max
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur_max = new_max
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
        else:
            h[i], h[j] = h[j], h[i]
        T *= cool
    return best_h, (2.0 / n) * best_max

def make_periodic(n, period):
    h = np.zeros(n)
    per = np.zeros(period)
    per[::2] = 1.0
    for i in range(n):
        h[i] = per[i % period]
    return h

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)
    best_h = None
    best_val = float('inf')
    best_n = 0

    def _update(h, val):
        nonlocal best_h, best_val, best_n
        if h is not None and val < best_val:
            best_h = h.copy()
            best_val = val
            best_n = len(h)

    # Phase 1: optimize initial_h_values
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()

        dl = start + budget_s * 0.35
        h_opt, val = optimize_cont(h0, dl)
        _update(h_opt, val)

    # Phase 2: try even n with periodic starts and binary SA
    if time.time() < start + budget_s * 0.82:
        n_list = [96, 128, 160, 200]
        remain = start + budget_s * 0.82 - time.time()
        per_n = max(remain / len(n_list), 20.0)
        for n in n_list:
            if time.time() >= start + budget_s * 0.82 - 5:
                break
            dl = min(time.time() + per_n, start + budget_s * 0.82)
            # Try a few periodic starts
            cands = []
            for period in [4, 6, 8]:
                if n % period == 0:
                    cands.append(make_periodic(n, period))
            if not cands:
                h_init = np.zeros(n)
                h_init[::2] = 1.0
                cands.append(h_init)
            for h_init in cands:
                if time.time() >= dl - 5:
                    break
                sub_dl = min(dl, time.time() + max(per_n * 0.3, 10))
                h_c, val_c = optimize_cont(h_init, sub_dl)
                _update(h_c, val_c)
                # Binary SA
                bin_dl = min(dl, time.time() + max(per_n * 0.25, 8))

        h_pol, val_pol = optimize_cont(best_h, dl)

        n = 96
        best_h = np.zeros(n)
        best_h[::2] = 1.0
        best_val = c5_value(best_h)
        best_n = n

    return best_h, best_val, best_n
