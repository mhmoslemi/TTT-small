import numpy as np
from scipy import optimize
import time

def compute_c5_array(h):
    """Compute all lag values c_k * dx."""
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx

def get_max_lag(c5_array):
    """Get the lag index (0 to 2n-2) with maximum value."""
    return np.argmax(c5_array)

def lag_from_index(idx, n):
    """Convert correlation index to lag value."""
    return idx - (n - 1)

def compute_gradient_for_lag(h, k):
    """Compute gradient of c_k * dx w.r.t. h."""
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    
    if k >= 0:
        for j in range(k, n):
            grad[j] += (1.0 - h[j - k])
        for j in range(n - k):
            grad[j] -= h[j + k]
    else:
        m = -k
        for j in range(n - m):
            grad[j] += (1.0 - h[j + m])
        for j in range(m, n):
            grad[j] -= h[j - m]
    
    return grad * dx

    """Project h onto {h: sum(h) = target_sum, 0 <= h <= 1}."""
    # Bisection on offset
    lo = -h.max() + 1e-10
    hi = 1.0 - h.min() - 1e-10
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h_new = np.clip(h + mid, 0.0, 1.0)
        s = h_new.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h_new = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    # Fine-tune
    diff = target_sum - h_new.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            # Add to the smallest elements
            order = np.argsort(h_new)
            for idx in order:
                add = min(diff, 1.0 - h_new[idx])
                h_new[idx] += add
                diff -= add
                if diff <= 1e-12:
                    break
        else:
            order = np.argsort(h_new)[::-1]
            for idx in order:
                sub = min(-diff, h_new[idx])
                h_new[idx] -= sub
                diff += sub
                if diff >= -1e-12:

    best_streak = 0
    
    while time.time() - start < time_limit:

        c5_val = np.max(c5_arr)
        idx_max = np.argmax(c5_arr)
        k = idx_max - (n - 1)

            best_c5 = c5_val
            best_h = h.copy()
            best_streak = 0
        else:
            best_streak += 1

        grad = compute_gradient_for_lag(h, k)
        h = h - lr * grad
        h = project_onto_constraint(h, target_sum)
        
        # Decay learning rate
        lr *= 0.999
        
        # Periodic perturbation to escape local minima
        if best_streak > 200:

            h_pert = project_onto_constraint(h_pert, target_sum)
            c5_pert = np.max(compute_c5_array(h_pert))

                lr = 0.1
                best_streak = 0
            else:

                lr = 0.05

                best_h = h_ls.copy()
                h = h_ls.copy()
                lr = 0.05
                best_streak = 0
    
    return best_h, best_c5

                    h = h_new  # continue from new point
                    improved = True
                    break
            if improved:

        if not improved:
            break
    
    return best_h

    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx

def grad_for_lag(h, k):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    if k >= 0:
        for j in range(k, n):
            g[j] += 1.0 - h[j-k]
        for j in range(n-k):
            g[j] -= h[j+k]
    else:
        m = -k
        for j in range(n-m):
            g[j] += 1.0 - h[j+m]
        for j in range(m, n):
            g[j] -= h[j-m]
    return g * dx

def project_constraint(h, s):
    lo = -h.max() - 0.01
    hi = 1.0 - h.min() + 0.01
    for _ in range(70):
        mid = (lo + hi) * 0.5
        hn = np.clip(h + mid, 0.0, 1.0)
        if hn.sum() > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + (lo+hi)*0.5, 0.0, 1.0)
    d = s - h.sum()
    if abs(d) > 1e-12:
        if d > 0:
            for i in np.argsort(h):
                a = min(d, 1.0-h[i])
                h[i] += a
                d -= a
                if d <= 1e-12: break
        else:
            for i in np.argsort(h)[::-1]:
                a = min(-d, h[i])
                h[i] -= a
                d += a
                if d >= -1e-12: break
    return h

    n = len(h)
    best = h.copy()
    best_c5 = np.max(compute_c5_array(h))
    for _ in range(n_iter):

        c5_arr = compute_c5_array(best)

            # g[j] > 0 means increasing h[j] increases the objective
            # g[j] < 0 means increasing h[j] decreases the objective
            # So we want to move mass from positions with g > 0 to positions with g < 0
            pos = np.where(g > 0.01)[0]
            neg = np.where(g < -0.01)[0]

But the Fourier perspective: if h has Fourier coefficients ĥ(ξ), then:
A(k) = ∫ h(x)h(x+k) dx = ∫ ĥ(ξ)² e^{iξk} dξ (in the continuous case)

So we want to choose ĥ(ξ) to make A(k) as constant as possible (or at least have a large minimum).

import numpy as np
from scipy import correlate
import time

def compute_c5_array(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx

def grad_for_lag(h, k):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    if k >= 0:
        for j in range(k, n):
            g[j] += 1.0 - h[j - k]
        for j in range(n - k):
            g[j] -= h[j + k]
    else:
        m = -k
        for j in range(n - m):
            g[j] += 1.0 - h[j + m]
        for j in range(m, n):
            g[j] -= h[j - m]
    return g * dx

def project_constraint(h, s):
    lo = -h.max() - 0.01
    hi = 1.0 - h.min() + 0.01
    for _ in range(70):
        mid = (lo + hi) * 0.5
        hn = np.clip(h + mid, 0.0, 1.0)
        if hn.sum() > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)
    d = s - h.sum()
    if abs(d) > 1e-12:
        if d > 0:
            for i in np.argsort(h):
                a = min(d, 1.0 - h[i])
                h[i] += a
                d -= a
                if d <= 1e-12:
                    break
        else:
            for i in np.argsort(h)[::-1]:
                a = min(-d, h[i])
                h[i] -= a
                d += a
                if d >= -1e-12:
                    break
    return h

def local_search(h, s, n_iter=100):
    n = len(h)
    best = h.copy()
    best_c5 = np.max(compute_c5_array(best))
    for _ in range(n_iter):
        improved = False
        c5_arr = compute_c5_array(best)
        top_idx = np.argsort(c5_arr)[::-1][:4]
        for tl in top_idx:
            k = tl - (n - 1)
            g = grad_for_lag(best, k)
            pos = np.where(g > 0.005)[0][:15]
            neg = np.where(g < -0.005)[0][:15]
            found = False
            for i in pos:
                for j in neg:
                    delta = 0.06
                    mx = min(best[i], 1.0 - best[j])
                    if mx < delta:
                        delta = mx * 0.6
                    if delta < 1e-5:
                        continue
                    hn = best.copy()
                    hn[i] -= delta
                    hn[j] += delta
                    c5_new = np.max(compute_c5_array(hn))
                    if c5_new < best_c5 - 1e-13:
                        best = hn
                        best_c5 = c5_new
                        found = True
                        improved = True
                        break
                if found:
                    break
            if improved:
                break
        if not improved:
            break
    return best, best_c5

def optimize_candidate(h_init, time_limit, rng):
    n = len(h_init)
    s = n / 2.0
    h = project_constraint(h_init.copy(), s)
    best_h = h.copy()
    best_c5 = np.max(compute_c5_array(h))
    start = time.time()
    lr = 0.06
    momentum = np.zeros(n)
    mu = 0.85
    no_improve = 0
    alpha = 15.0

    while time.time() - start < time_limit:
        c5_arr = compute_c5_array(h)
        c5_val = np.max(c5_arr)
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        c5_shift = c5_arr - np.max(c5_arr)
        w = np.exp(alpha * c5_shift)
        w /= w.sum()

        g = np.zeros(n)
        for idx in range(len(w)):
            if w[idx] < 0.003:
                continue
            k = idx - (n - 1)
            g += w[idx] * grad_for_lag(h, k)

        momentum = mu * momentum - lr * g
        h = h + momentum
        h = project_constraint(h, s)
        lr *= 0.9997

        if no_improve > 80:
            h_ls, c5_ls = local_search(h, s, n_iter=40)
            if c5_ls < best_c5 - 1e-13:
                best_c5 = c5_ls
                best_h = h_ls.copy()
                h = h_ls.copy()
                lr = 0.04
                momentum *= 0.3
                no_improve = 0
            else:
                h = best_h + rng.standard_normal(n) * 0.06
                h = project_constraint(h, s)
                lr = 0.03
                momentum = np.zeros(n)
                no_improve = 0

        if no_improve > 400:
            h = best_h + rng.standard_normal(n) * 0.15
            h = project_constraint(h, s)
            lr = 0.05
            momentum = np.zeros(n)
            no_improve = 0

    # Final local search polish
    h_ls, c5_ls = local_search(best_h, s, n_iter=150)
    if c5_ls < best_c5:
        best_h = h_ls
        best_c5 = c5_ls
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()

    best_h = None
    best_c5 = float('inf')

    h0 = None
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)

    trials = []
    if h0 is not None:
        trials.append(h0.copy())
        # Interpolated versions
        for factor in [2, 4]:
            n_new = len(h0) * factor
            x_old = np.linspace(0, 2, len(h0), endpoint=False)
            x_new = np.linspace(0, 2, n_new, endpoint=False)
            h_interp = np.interp(x_new, x_old, h0, period=2.0)
            h_interp = project_constraint(h_interp, n_new / 2.0)
            trials.append(h_interp)

    # Random near-binary starts
    ns_to_try = [44, 45, 46, 48, 50, 52, 56, 60, 64, 72, 80, 96, 100, 120]
    existing_ns = set(len(t) for t in trials)
    for n in ns_to_try:
        if n in existing_ns:
            continue
        h_rand = np.zeros(n)
        ones = n // 2
        rem = n / 2.0 - ones
        idx = rng.permutation(n)
        h_rand[idx[:ones]] = 1.0
        if rem > 1e-10:
            h_rand[idx[ones]] = rem
        trials.append(h_rand)
        # Second random start for smaller n
        if n <= 64:
            h_rand2 = np.zeros(n)
            idx2 = rng.permutation(n)
            h_rand2[idx2[:ones]] = 1.0
            if rem > 1e-10:
                h_rand2[idx2[ones]] = rem
            trials.append(h_rand2)

    # Prioritize: initial_h_values first, then by n (smaller preferred)
    trials.sort(key=lambda x: (0 if x is trials[0] or (h0 is not None and len(x) == len(h0)) else 1, len(x)))

    n_trials = len(trials)
    time_per_trial = (budget_s * 0.82) / n_trials

    for trial in trials:
        elapsed = time.time() - t0
        if elapsed > budget_s * 0.92:
            break
        remaining = min(time_per_trial, budget_s - elapsed - 5.0)
        if remaining < 2.0:
            break
        h_opt, c5 = optimize_candidate(trial, remaining, rng)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt.copy()

    # Final polish
    remaining = budget_s - (time.time() - t0) - 1.0
    if remaining > 1.0 and best_h is not None:
        h_ls, c5_ls = local_search(best_h, len(best_h) / 2.0, n_iter=300)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls

    return best_h, best_c5, len(best_h)
