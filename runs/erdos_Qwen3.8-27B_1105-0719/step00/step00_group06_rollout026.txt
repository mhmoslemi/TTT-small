import numpy as np

import time

def exact_objective(h):
    """Compute C5 = max(correlate(h, 1-h, 'full') * dx)"""
    n = len(h)
    dx = 2.0 / n
    profile = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(profile)), profile

def project_box_sum(h, n):
    """Project h to [0,1]^n with sum = n/2 using water-filling."""
    h = np.array(h, dtype=np.float64).copy()
    target = n / 2.0
    # Clip to [0, 1]
    h = np.clip(h, 0.0, 1.0)
    # Iterative water-filling
    for _ in range(100):
        s = h.sum()
        diff = s - target
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            # Too much, reduce from positive entries
            mask = h > 0
            count = mask.sum()
            if count == 0:
                break
            step = diff / count
            h[mask] = np.maximum(h[mask] - step, 0.0)
        else:
            # Too little, add to entries < 1
            mask = h < 1.0
            count = mask.sum()
            if count == 0:
                break
            step = -diff / count
            h[mask] = np.minimum(h[mask] + step, 1.0)
    # Final fix: distribute remaining exactly
    residual = target - h.sum()
    if abs(residual) > 1e-15:
        if residual > 0:
            mask = h < 1.0
            if mask.sum() > 0:
                h[mask] += residual / mask.sum()
                h = np.minimum(h, 1.0)
        else:
            mask = h > 0.0
            if mask.sum() > 0:
                h[mask] += residual / mask.sum()
                h = np.maximum(h, 0.0)
    return h

def get_profile(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def lse_value_and_grad(h, tau):
    """Compute log-sum-exp smoothed objective and gradient."""
    n = len(h)
    dx = 2.0 / n
    profile = get_profile(h)
    # LSE
    max_p = np.max(profile)
    if tau < 1e-15:
        # Just use the max
        # Find all indices achieving max
        active = np.where(np.abs(profile - max_p) < 1e-12)[0]
        weights = np.zeros(len(profile))
        weights[active] = 1.0 / len(active)
    else:
        shifted = (profile - max_p) / tau
        # Avoid overflow
        shifted = np.minimum(shifted, 500.0)
        exp_shifted = np.exp(shifted)
        sum_exp = np.sum(exp_shifted)
        weights = exp_shifted / sum_exp
    value = max_p + tau * np.log(np.sum(np.exp(np.minimum((profile - max_p) / tau, 500.0))))
    
    # Gradient: weighted sum of individual lag gradients
    grad = np.zeros(n)
    # profile[m] corresponds to lag tau_m = m - (n-1)
    for m in range(len(profile)):
        if weights[m] < 1e-14:
            continue
        lag = m - (n - 1)
        w = weights[m]
        if lag >= 0:
            # F = dx * sum_{j=0}^{n-1-lag} h[j+lag](1-h[j])
            # dF/dh[i] = dx*(1-h[i-lag]) if lag <= i <= n-1
            #          - dx*h[i+lag] if 0 <= i <= n-1-lag
            if lag == 0:
                grad += w * dx * (1.0 - 2.0 * h)
            else:
                # First term: (1-h[i-lag]) for i in [lag, n-1]
                for i in range(lag, n):
                    grad[i] += w * dx * (1.0 - h[i - lag])
                # Second term: -h[i+lag] for i in [0, n-1-lag]
                for i in range(0, n - lag):
                    grad[i] -= w * dx * h[i + lag]
        else:
            a = -lag
            # F = dx * sum_{i=0}^{n-1-a} h[i](1-h[i+a])
            # dF/dh[i] = dx*(1-h[i+a]) if 0 <= i <= n-1-a
            #          - dx*h[i-a] if a <= i <= n-1
            for i in range(0, n - a):
                grad[i] += w * dx * (1.0 - h[i + a])
            for i in range(a, n):
                grad[i] -= w * dx * h[i - a]
    return value, grad

def slsqp_stage(h, n, tau, maxiter=200):
    """Run one SLSQP stage with given tau."""
    def obj(x):
        val, _ = lse_value_and_grad(x, tau)
        # Add penalty for constraint violation
        sum_err = x.sum() - n / 2.0
        box_viol = np.sum(np.maximum(0, x - 1)**2) + np.sum(np.maximum(0, -x)**2)
        return val + 1e6 * (sum_err**2 + box_viol)
    
    def jac(x):
        _, g = lse_value_and_grad(x, tau)
        sum_err = x.sum() - n / 2.0
        g += 2e6 * sum_err * np.ones(n)
        box_viol_grad = 2 * np.maximum(0, x - 1) - 2 * np.maximum(0, -x)
        g += 1e6 * box_viol_grad
        return g
    
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0, 'jac': lambda x: np.ones(n)}]
    
    res = minimize(obj, h, method='SLSQP', jac=jac, bounds=bounds,
                   constraints=constraints, options={'maxiter': maxiter, 'ftol': 1e-12})
    return res

        tau_list = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    best_h = h.copy()
    best_c5, _ = exact_objective(h)
    for tau in tau_list:
        if time.time() > t_limit - 5:
            break
        _g_n = n
        _g_tau = tau
        _g_target = n / 2.0
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': _sp_eq, 'jac': _sp_eq_jac}]
        try:
            res = minimize(_sp_obj, h, method='SLSQP', jac=_sp_jac,
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 400, 'ftol': 1e-12})
            x_opt = np.array(res.x, dtype=np.float64)
            x_opt = project_box_sum(x_opt, n)
            c5_t, _ = exact_objective(x_opt)
            if c5_t < best_c5 - 1e-12:
                best_c5 = c5_t
                best_h = x_opt.copy()
                h = x_opt.copy()
        except Exception:
            pass
    return best_h, best_c5

def binary_threshold(h, n):
    if n % 2 != 0:
        return None
    k = n // 2
    idx = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[idx[:k]] = 1.0
    return b

def binary_swap_search(h, n, max_rounds=50, t_limit=None):
    if n % 2 != 0:
        return h
    best_c5, _ = exact_objective(h)
    for _ in range(max_rounds):
        if t_limit is not None and time.time() > t_limit - 3:
            break
        improved = False
        best_val = best_c5
        best_swap = None
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        for i in ones:
            for j in zeros:
                trial = h.copy()
                trial[i] = 0.0
                trial[j] = 1.0
                c5, _ = exact_objective(trial)
                if c5 < best_val - 1e-14:
                    best_val = c5
                    best_swap = (i, j)
        if best_swap is not None:
            i, j = best_swap
            h[i] = 0.0
            h[j] = 1.0
            best_c5 = best_val
            improved = True
        if not improved:
            break
    return h

def subgradient_polish(h, n, t_limit, n_iter=20000):
    best_h = h.copy()
    best_c5, _ = exact_objective(h)
    step = 0.02
    dx = 2.0 / n
    for _ in range(n_iter):
        if time.time() > t_limit - 2:
            break
        profile = get_profile(h)
        max_p = np.max(profile)
        active = np.where(profile >= max_p - 1e-5)[0]
        sg = np.zeros(n)
        for m in active:
            lag = m - (n - 1)
            if lag == 0:
                sg += dx * (1.0 - 2.0 * h)
            elif lag > 0:
                sg[lag:] += dx * (1.0 - h[:n - lag])
                sg[:n - lag] -= dx * h[lag:]
            else:
                a = -lag
                sg[:n - a] += dx * (1.0 - h[a:])
                sg[a:] -= dx * h[:n - a]
        sg /= len(active)
        h_new = h - step * sg
        h_new = project_box_sum(h_new, n)
        c5, _ = exact_objective(h_new)
        if c5 < best_c5 - 1e-13:
            best_c5 = c5
            best_h = h_new.copy()
            h = h_new
        else:
            step *= 0.97
            h = best_h.copy()
        if step < 1e-8:
            break
    return best_h, best_c5

def transfer_resolution(h, n_from, n_to):
    if n_from == n_to:
        return h.copy()
    x_old = (np.arange(n_from) + 0.5) / n_from * 2.0
    x_new = (np.arange(n_to) + 0.5) / n_to * 2.0
    new_h = np.interp(x_new, x_old, h)
    return project_box_sum(new_h, n_to)

def coordinate_descent(h, n, t_limit, n_sweeps=20):
    best_h = h.copy()
    best_c5, _ = exact_objective(h)
    for sweep in range(n_sweeps):
        if time.time() > t_limit - 3:
            break
        improved_any = False
        for i in range(n):
            if time.time() > t_limit - 3:
                break
            for trial_val in [0.0, 0.25, 0.5, 0.75, 1.0]:
                if abs(trial_val - h[i]) < 1e-6:
                    continue
                trial = h.copy()
                # Adjust to maintain sum
                old_sum = h.sum()
                new_sum = old_sum - h[i] + trial_val
                if new_sum > n / 2.0:
                    # Reduce elsewhere
                    others = np.arange(n) != i
                    slack = np.maximum(0, h[others] - 0.0)
                    total_excess = new_sum - n / 2.0
                    if slack.sum() < total_excess:
                        continue
                    # Distribute reduction
                    h_try = trial.copy()
                    h_try[others] = project_box_sum(h_try[others], int(others.sum()))
                    h_try[i] = trial_val
                    # Re-fix sum
                    h_try = project_box_sum(h_try, n)
                    if abs(h_try[i] - trial_val) > 0.1:
                        continue
                elif new_sum < n / 2.0:
                    h_try = trial.copy()
                    h_try = project_box_sum(h_try, n)
                    if abs(h_try[i] - trial_val) > 0.1:
                        continue
                else:
                    h_try = trial.copy()
                c5, _ = exact_objective(h_try)
                if c5 < best_c5 - 1e-12:
                    best_c5 = c5
                    best_h = h_try.copy()
                    h = h_try
                    improved_any = True
                    break
        if not improved_any:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    global _g_n, _g_tau, _g_target
    t_start = time.time()
    t_limit = t_start + budget_s
    rng = np.random.RandomState(seed)

    # Get initial
    init = globals().get('initial_h_values', None)
    if init is not None:
        try:
            init = np.asarray(init, dtype=np.float64)
        except Exception:
            init = None
    if init is None:
        init = np.full(70, 0.5)
    n_init = len(init)

    best_h = project_box_sum(init.copy(), n_init)
    best_c5, _ = exact_objective(best_h)

    # Evaluate with external evaluator if available
    eval_fn = globals().get('evaluate_erdos_solution', None)

    def update_best(h_cand, n_cand):
        nonlocal best_h, best_c5
        h_cand = project_box_sum(h_cand, n_cand)
        c5, _ = exact_objective(h_cand)
        if c5 < best_c5 - 1e-13:
            best_h = h_cand.copy()
            best_c5 = c5
        return c5

    # Phase 1: SLSQP from parent (main refinement)
    if time.time() < t_limit - 60:
        h1, c1 = run_slsqp(best_h.copy(), n_init, t_limit - 30,
                           tau_list=[0.01, 0.005, 0.003, 0.002, 0.001, 0.0005, 0.0003, 0.0001, 0.00005])
        update_best(h1, n_init)

    # Phase 2: Binary threshold + swap
    if time.time() < t_limit - 60 and n_init % 2 == 0:
        b = binary_threshold(best_h, n_init)
        if b is not None:
            b = binary_swap_search(b, n_init, max_rounds=40, t_limit=t_limit)
            c_b = update_best(b, n_init)
            # SLSQP from binary
            if time.time() < t_limit - 30:
                h2, _ = run_slsqp(b.copy(), n_init, t_limit - 15,
                                   tau_list=[0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001])
                update_best(h2, n_init)

    # Phase 3: Subgradient polish
    if time.time() < t_limit - 60:
        h3, _ = subgradient_polish(best_h.copy(), n_init, t_limit - 30, n_iter=30000)
        update_best(h3, n_init)

    # Phase 4: Random perturbations + SLSQP
    scales = [0.15, 0.1, 0.05, 0.03, 0.01]
    for trial in range(20):
        if time.time() > t_limit - 40:
            break
        sc = scales[trial % len(scales)]
        noise = rng.normal(0, sc, n_init)
        h_p = project_box_sum(best_h + noise, n_init)
        h4, _ = run_slsqp(h_p, n_init, t_limit - 10,
                          tau_list=[0.005, 0.002, 0.001, 0.0005])
        update_best(h4, n_init)
        # Binary from perturbed result
        if n_init % 2 == 0 and time.time() < t_limit - 20:
            b2 = binary_threshold(h4, n_init)
            if b2 is not None:
                b2 = binary_swap_search(b2, n_init, max_rounds=20, t_limit=t_limit)
                update_best(b2, n_init)

    # Phase 5: Try different resolutions
    n_init_for_trans = n_init
    other_ns = [40, 50, 60, 80, 90]
    for n_try in other_ns:
        if time.time() > t_limit - 30:
            break
        h_t = transfer_resolution(best_h, n_init_for_trans, n_try)
        h5, _ = run_slsqp(h_t, n_try, t_limit - 10,
                          tau_list=[0.005, 0.002, 0.001, 0.0005])
        update_best(h5, n_try)
        if n_try % 2 == 0 and time.time() < t_limit - 15:
            b3 = binary_threshold(h5, n_try)
            if b3 is not None:
                b3 = binary_swap_search(b3, n_try, max_rounds=20, t_limit=t_limit)
                update_best(b3, n_try)

    # Phase 6: Final SLSQP refinement with very small tau
    if time.time() < t_limit - 10:
        h6, _ = run_slsqp(best_h.copy(), len(best_h), t_limit - 2,
                          tau_list=[0.0001, 0.00005, 0.00002, 0.00001])
        update_best(h6, len(best_h))

    # Phase 7: Final subgradient polish
    if time.time() < t_limit - 3:
        h7, _ = subgradient_polish(best_h.copy(), len(best_h), t_limit - 1, n_iter=10000)
        update_best(h7, len(best_h))

    # Final validation
    n_final = len(best_h)
    final_h = project_box_sum(best_h, n_final)
    final_c5, _ = exact_objective(final_h)

    # Double check with external evaluator
    if eval_fn is not None:
        try:
            ext_val = eval_fn(final_h)
            if isinstance(ext_val, (float, int)) and ext_val < final_c5:
                final_c5 = float(ext_val)
        except Exception:
            pass

    return (final_h, final_c5, n_final)
