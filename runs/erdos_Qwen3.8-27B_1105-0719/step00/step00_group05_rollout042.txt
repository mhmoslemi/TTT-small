import numpy as np
import time

def compute_c5_values(h):
    """Return (c5, max_corr, n) for a given h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def compute_correlation(h):
    """Return the full correlation vector h * (1-h)."""
    return np.correlate(h, 1.0 - h, mode="full")

def initial_binary(n):
    """Create initial binary solution with n/2 ones."""
    h = np.zeros(n, dtype=float)
    h[:n//2] = 1.0
    return h

def sa_binary(h, n, n_iter, T0, T1, rng):
    """Simulated annealing for binary h."""
    h = h.copy().astype(int)
    dx = 2.0 / n
    current = np.max(np.correlate(h.astype(float), 1.0 - h.astype(float), mode="full")) * dx
    best_h = h.copy()
    best_val = current
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    for it in range(n_iter):
        T = T0 * (T1 / T0) ** ((it + 1) / n_iter)
        
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        
        h[i] = 0
        h[j] = 1
        
        hf = h.astype(float)
        new_val = np.max(np.correlate(hf, 1.0 - hf, mode="full")) * dx
        delta = new_val - current
        
        if delta <= 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            current = new_val
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
        else:
            h[i] = 1
            h[j] = 0
    
    return best_h.astype(float), best_val

def continuous_refine(h, n, n_steps, step_size):
    """Simple projected subgradient refinement for continuous h."""
    h = h.copy().astype(float)
    n = len(h)
    dx = 2.0 / n
    current_c5 = np.max(np.correlate(h, 1.0 - h, mode="full")) * dx
    
    for step in range(n_steps):
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_val = np.max(corr)
        active = np.where(corr == max_val)[0]

        # For each active shift k, the gradient wrt h[i] is:
        # (1 - h[i+k']) - h[i-k'] where k' is the signed shift
        # Center of full corr is at index n-1
        grad = np.zeros(n)
        for k_idx in active:
            k = k_idx - (n - 1)  # signed shift
            # h[i] appears as h[i]*(1-h[i+k]) -> gradient (1-h[i+k])
            # h[i] appears as h[i-k]*(1-h[i]) -> gradient -h[i-k]
            # First term: for valid i, i+k in [0,n-1]
            if k >= 0:
                # i ranges from 0 to n-1-k
                idx1 = np.arange(n - k)
                grad[idx1] += (1.0 - h[idx1 + k])
                # Second term: i-k in [0,n-1], i in [k, n-1]
                idx2 = np.arange(k, n)
                grad[idx2] -= h[idx2 - k]
            else:
                kk = -k
                # i ranges from kk to n-1
                idx1 = np.arange(kk, n)
                grad[idx1] += (1.0 - h[idx1 + k])  # h[i+k] = h[i-kk]
                # Second term: i-k = i+kk in [0,n-1], i in [0, n-1-kk]
                idx2 = np.arange(n - kk)
                grad[idx2] -= h[idx2 + kk]
        
        # Normalize
        grad = grad / len(active)
        
        # Projected gradient step
        h_new = h - step_size * grad * max_val
        h_new = np.clip(h_new, 0, 1)
        
        # Project onto sum constraint
        target_sum = n / 2.0
        current_sum = np.sum(h_new)
        if current_sum != target_sum:
            h_new = h_new + (target_sum - current_sum) / n
            h_new = np.clip(h_new, 0, 1)
            # If clipping changed the sum, do a simple watermarking
            diff = target_sum - np.sum(h_new)
            if abs(diff) > 1e-10:
                order = np.argsort(h_new)
                if diff > 0:
                    for idx in order:
                        add = min(diff, 1.0 - h_new[idx])
                        h_new[idx] += add
                        diff -= add
                        if abs(diff) < 1e-12:

                else:
                    for idx in reversed(order):
                        rem = min(-diff, h_new[idx])
                        h_new[idx] -= rem
                        diff += rem
                        if abs(diff) < 1e-12:

        new_c5 = np.max(np.correlate(h_new, 1.0 - h_new, mode="full")) * dx
        if new_c5 <= current_c5:
            h = h_new
            current_c5 = new_c5
        
        # Decrease step size
        step_size *= 0.99
    
    return h, current_c5

def project_to_simplex(h, target_sum):
    """Project h onto {x: 0<=x[i]<=1, sum(x)=target_sum}."""
    h = np.clip(h, 0.0, 1.0)

    else:
        order = np.argsort(h)  # least value first
    remaining = diff
    for idx in order:
        if diff > 0:
            add = min(remaining, 1.0 - h[idx])
        else:
            add = max(remaining, -h[idx])
        h[idx] += add
        remaining -= add
        if abs(remaining) < 1e-12:
            break
    return h

def coordinate_descent(h, n_sweeps=200, time_limit=120, enforce_sum=True):
    n = len(h)
    dx = 2.0 / n
    h = h.copy().astype(float)
    target_sum = n / 2.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    start = time.time()
    for sweep in range(n_sweeps):
        if (time.time() - start) > time_limit:
            break
        improved = False
        for m in range(n):
            # Compute a and b vectors
            old_val = h[m]
            h[m] = 0.0
            corr_0 = np.correlate(h, 1.0 - h, mode="full")
            h[m] = 1.0
            corr_1 = np.correlate(h, 1.0 - h, mode="full")
            h[m] = old_val
            
            a = corr_1 - corr_0
            b = corr_0
            
            # Minimize max_k (a[k]*t + b[k]) over t in [0,1]
            candidates = [0.0, 1.0]
            n_corr = len(a)
            for i_k in range(n_corr):
                for j_k in range(i_k + 1, n_corr):
                    da = a[i_k] - a[j_k]
                    if abs(da) > 1e-12:
                        t = (b[j_k] - b[i_k]) / da
                        if -0.01 <= t <= 1.01:
                            candidates.append(np.clip(t, 0.0, 1.0))
            
            best_t = 0.0
            best_val_t = float('inf')
            for t in candidates:
                val = float(np.max(a * t + b))
                if val < best_val_t:
                    best_val_t = val
                    best_t = t
            
            h[m] = best_t
        
        # Project to maintain sum constraint
        if enforce_sum:
            h = project_to_simplex(h, target_sum)
        
        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-10:
            best_c5 = c5
            best_h = h.copy()
            improved = True
        elif not improved and sweep > 10:
            break
    
    return best_h, best_c5

def sa_binary(n, h_init, time_limit, T0=0.5, T1=0.001, rng=None):
    if rng is None:
        rng = np.random.default_rng()
    
    h = h_init.copy().astype(float)
    dx = 2.0 / n
    best_h = h.copy()
    best_val = compute_c5(h)
    current_val = best_val
    
    ones = list(np.where(h_init >= 0.5)[0])
    zeros = list(np.where(h_init < 0.5)[0])
    
    start = time.time()
    it = 0
    max_iter = 10000000  # Safety limit
    
    while it < max_iter:
        if (time.time() - start) > time_limit:
            break
        
        T = T0 * (T1 / T0) ** (it / max(1, int(time_limit * 100000)))
        
        i_idx = int(rng.integers(len(ones)))
        j_idx = int(rng.integers(len(zeros)))
        i = ones[i_idx]
        j = zeros[j_idx]
        
        h[i] = 0.0
        h[j] = 1.0
        ones[i_idx] = j
        zeros[j_idx] = i
        
        corr = np.correlate(h, 1.0 - h, mode="full")
        new_val = float(np.max(corr)) * dx
        delta = new_val - current_val
        
        if delta <= 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            current_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
            ones[i_idx] = i
            zeros[j_idx] = j
        
        it += 1
    
    return best_h, best_val

    """Find t in [0,1] minimizing max_k (a[k]*t + b[k])."""
    n = len(a)
    # Compute all intersection points
    # a[i]*t + b[i] = a[j]*t + b[j] => t = (b[j]-b[i])/(a[i]-a[j])
    da = a[:, None] - a[None, :]  # n x n
    db = b[None, :] - b[:, None]  # n x n
    # t[i,j] = db[i,j] / da[i,j] where da[i,j] != 0
    with np.errstate(divide='ignore', invalid='ignore'):
        t_matrix = db / da
    t_matrix = np.where(np.abs(da) > 1e-14, t_matrix, np.nan)
    
    # Collect valid t values in [0,1]
    valid = (t_matrix >= 0) & (t_matrix <= 1) & ~np.isnan(t_matrix)
    candidates = np.unique(t_matrix[valid])
    candidates = np.concatenate([[0.0, 1.0], candidates])
    
    # Evaluate f at each candidate
    # f(t) = max_k(a_k * t + b_k)
    best_t = 0.0
    best_v = float('inf')
    for t in candidates:
        v = float(np.max(a * t + b))
        if v < best_v:
            best_v = v
            best_t = t
    return best_t
