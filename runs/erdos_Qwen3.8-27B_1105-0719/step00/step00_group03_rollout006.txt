def project_to_feasible(h, target):
    """Project h to [0,1]^n with sum = target using bisection."""
    h = h.copy()
    # Find lambda such that sum(clip(h + lambda, 0, 1)) = target
    lo, hi = -max(h.max(), 1.0) - 1, 1.0 - h.min() + 1
    for _ in range(100):

        s = np.sum(np.clip(h + mid, 0, 1))
        if s > target:
            lo = mid
        else:
            hi = mid

    residual = target - np.sum(x)

                    break
                x[i] -= rem_now
                rem -= rem_now
