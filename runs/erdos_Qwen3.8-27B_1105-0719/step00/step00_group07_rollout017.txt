def _project_mass(x, target):
    n = len(x)
    target = float(target)
    # Bisection on lambda: u_i = clip(x_i - lambda, 0, 1)
    lo = float(np.min(x)) - 1.0
    hi = float(np.max(x)) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        u = np.clip(x - mid, 0.0, 1.0)
        s = np.sum(u)
        if s > target:
            lo = mid
        else:
            hi = mid
    u = np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(u)
    if abs(residual) < 1e-12:
        return u
    # Sequential correction
    if residual > 0:
        idx = np.argsort(1.0 - u)  # most slack first
    else:
        idx = np.argsort(u)  # most room to decrease first
    remaining = residual
    for i in idx:
        if abs(remaining) < 1e-14:
            break
        if remaining > 0:
            delta = min(remaining, 1.0 - u[i])
            u[i] += delta
            remaining -= delta
        else:
            delta = max(remaining, -u[i])
            u[i] += delta
            remaining -= delta
    return u

def _flip_complement(x):
    return 1.0 - x[::-1]

def _gradient_row(h, idx):
    n = len(h)
    d = (n - 1) - idx

    else:

    target = n / 2.0
    # Variables: [s, x_0, ..., x_{n-1}]

    # Constraints:
    #   s >= raw_base_j + g_j^T (x - h)  for j in active
    #   0 <= x_i <= 1 (enforced by trust region bounds)
    #   max(0, h_i - rho) <= x_i <= min(1, h_i + rho)
    #   sum(x) = n/2

    c[0] = 1.0  # minimize s
    
    A_ub = []
    b_ub = []
    
    for j in active_rows:
        g = _gradient_row(h, j)
        base_val = raw[j]
        # s >= base_val + g^T(x - h)
        # -s + g^T x <= -base_val + g^T h
        row = np.zeros(n + 1)
        row[0] = -1.0
        row[1:] = g
        rhs = -base_val + np.dot(g, h)
        A_ub.append(row)
        b_ub.append(rhs)
    
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    A_eq = np.zeros((1, n + 1))

    b_eq = np.array([target])

    for i in range(n):

        hi = min(1.0, h[i] + rho)
        bounds.append((lo, hi))

            # Ensure feasibility
            x_new = np.clip(x_new, 0.0, 1.0)
            x_new = _project_mass(x_new, target)

    return None

def _line_search(base, direction, best_score):

        if s < best_s - 1e-12:

def _random_perturb(h, n_perturb, scale):
    rng = np.random.default_rng()
    n = len(h)
    idx_p = rng.choice(n, size=n_perturb, replace=False)
    idx_q = rng.choice(n, size=n_perturb, replace=False)

        delta = min(abs(delta), room_p, room_q) * (1 if delta > 0 else -1)
        h[p] += delta
        h[q] -= delta
    return _project_mass(h, n / 2.0)

import numpy as np
import time
from scipy.optimize import linprog

def _score(h):
    n = len(h)
    dx = 2.0 / n
    raw = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(raw) * dx), raw

def _project(x, target):
    n = len(x)
    lo = float(np.min(x)) - 1.0
    hi = float(np.max(x)) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    u = np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)
    r = target - np.sum(u)
    if r > 0:
        idx = np.argsort(1.0 - u)
    else:
        idx = np.argsort(u)
    rem = r
    for i in idx:
        if abs(rem) < 1e-14:
            break
        if rem > 0:
            d = min(rem, 1.0 - u[i])
        else:
            d = max(rem, -u[i])
        u[i] += d
        rem -= d
    return u

def _fc(x):
    return 1.0 - x[::-1]

def _grem(h, idx):
    n = len(h)
    d = (n - 1) - idx
    g = np.zeros(n)
    if d == 0:
        g[:] = 1.0 - 2.0 * h
    elif d > 0:
        g[:n - d] += 1.0 - h[d:]
        g[d:] += -h[:n - d]
    else:
        r = -d
        g[r:] += 1.0 - h[:n - r]
        g[:n - r] += -h[r:]
    return g

def _ls(base, direction, best_s):
    n = len(base)
    for a in [1.0, 0.6, 0.4, 0.25, 0.15, 0.08, 0.04]:
        cand = base + a * direction
        if cand.min() < -0.05 or cand.max() > 1.05:
            continue
        cand = np.clip(cand, 0.0, 1.0)
        cand = _project(cand, n / 2.0)
        s, _ = _score(cand)
        if s < best_s - 1e-12:
            return cand, s
    return None, best_s

def _grad_pair_move(h, raw, active, rng):
    n = len(h)
    k_star = active[0]
    g_star = _grem(h, k_star)
    neg_c = np.where(g_star < -0.005)[0]
    pos_c = np.where(g_star > 0.005)[0]
    if len(neg_c) == 0 or len(pos_c) == 0:
        return None
    best_cand = None
    best_cs = np.inf
    for _ in range(8):
        p = neg_c[rng.integers(0, len(neg_c))]
        q = pos_c[rng.integers(0, len(pos_c))]
        for dv in [0.02, 0.05, 0.08]:
            c = h.copy()
            c[p] += dv
            c[q] -= dv
            if 0.0 <= c[p] <= 1.0 and 0.0 <= c[q] <= 1.0:
                c = _project(c, n / 2.0)
                cs, _ = _score(c)
                if cs < best_cs:
                    best_cs = cs
                    best_cand = c
    if best_cand is not None and best_cs < 1e6:
        return best_cand, best_cs
    return None
