def repair_feasible(vec, n):
    x = np.clip(np.asarray(vec, dtype=np.float64), 0.0, 1.0).copy()
    target = n / 2.0
    s = x.sum()
    diff = target - s
    if abs(diff) < 1e-13:
        return x
    if diff > 0:
        mask = x < 1.0 - 1e-9
        if mask.sum() == 0:
            return x
        x[mask] += diff / mask.sum()
        x[mask] = np.minimum(x[mask], 1.0)
    else:
        mask = x > 1e-9
        if mask.sum() == 0:
            return x
        x[mask] += diff / mask.sum()  # diff is negative
        x[mask] = np.maximum(x[mask], 0.0)
    # Final exact fix
    residual = target - x.sum()
    if abs(residual) > 1e-10:
        if residual > 0:
            idx = np.argmin(x)
            if x[idx] + residual <= 1.0:
                x[idx] += residual
        else:
            idx = np.argmax(x)
            if x[idx] + residual >= 0.0:
                x[idx] += residual
    return x

    n = len(vec)
    if n == new_n:
        return vec.copy()

    x_tgt = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    x_ext = np.concatenate([[0.0], x_src, [2.0]])
    h_ext = np.concatenate([[vec[0]], vec, [vec[-1]]])
    h_new = np.interp(x_tgt, x_ext, h_ext)

def compute_gradient_matrix(h, dx, n):
    M = 2 * n - 1
    G = np.zeros((M, n))
    for k in range(M):
        delta = k - (n - 1)
        if delta >= 0:
            if delta < n:
                G[k, delta:] += dx * (1.0 - h[:n - delta])
            G[k, :n - delta] += dx * (-h[delta:])
        else:
            d = -delta
            G[k, :n - d] += dx * (1.0 - h[d:])
            G[k, d:] += dx * (-h[:n - d])
    return G
