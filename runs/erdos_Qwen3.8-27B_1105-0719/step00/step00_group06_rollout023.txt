        g = np.zeros(n)
        if s == 0:

        elif s > 0:
            g[s:] += (1.0 - h[:n-s]) * dx
            g[:n-s] -= h[s:] * dx

def _project(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)

        if abs(diff) < 1e-14:

        else:

        if cnt == 0:

        # Water-filling: distribute evenly among available
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    return h

        grad = G[active].mean(axis=0)
        return float(max_val), grad

    log_sum = np.log(exp_vals.sum())
    obj = max_val + log_sum / beta
    
    weights = exp_vals / exp_vals.sum()
    G = _gradients_all(h, n)
    grad = weights @ G

def _eval(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def _gradients_all(h, n):
    dx = 2.0 / n
    n_corr = 2 * n - 1
    G = np.zeros((n_corr, n))
    for k in range(n_corr):
        s = k - (n - 1)
        if s == 0:
            G[k] = (1.0 - 2.0 * h) * dx
        elif s > 0:
            G[k, s:] += (1.0 - h[:n-s]) * dx
            G[k, :n-s] -= h[s:] * dx
        else:
            s2 = -s
            G[k, :n-s2] += (1.0 - h[s2:]) * dx
            G[k, s2:] -= h[:n-s2] * dx
    return G

def _project(h, n):
    target = n / 2.0
    h = np.clip(h.astype(np.float64).copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] = np.clip(h[mask] + diff / cnt, 0.0, 1.0)
    return h

def _smooth_eval(h, n, beta):
    corr = np.correlate(h, 1.0 - h, mode="full") * (2.0/n)
    max_c = corr.max()
    shifted = beta * (corr - max_c)
    # Avoid overflow
    if shifted.max() > 500:
        return float(max_c)
    return float(max_c + np.log(np.sum(np.exp(shifted))) / beta)

def _smooth_grad(h, n, beta):
    corr = np.correlate(h, 1.0 - h, mode="full") * (2.0/n)
    max_c = corr.max()
