def sigmoid(t):
    """Numerically stable sigmoid."""
    t = np.asarray(t, dtype=np.float64)
    out = np.empty_like(t)
    pos = t >= 0
    out[pos] = 1.0 / (1.0 + np.exp(-t[pos]))
    exp_t = np.exp(t[~pos])
    out[~pos] = exp_t / (1.0 + exp_t)
    return out

def interpolate(h, new_n):
    """Interpolate h from its current length to new_n."""
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = (np.arange(old_n) + 0.5) / old_n
    new_x = (np.arange(new_n) + 0.5) / new_n
    h_new = np.interp(new_x, old_x, h)
    h_new = project_feasible(h_new)
    return h_new
