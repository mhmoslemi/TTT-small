def project_box_sum(y, target):
    n = len(y)
    target = max(0.0, min(float(n), target))
    lo = float(y.min()) - 1.0
    hi = float(y.max()) + 1.0
    for _ in range(70):

            lo = mid
        else:
            hi = mid

import numpy as np
import time

def compute_corrs(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode="full")

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corrs(h)) * dx)

def project_box_sum(y, target):
    n = len(y)
    target = max(0.0, min(float(n), float(target)))
    lo = float(y.min()) - 1.0
    hi = float(y.max()) + 1.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = float(np.clip(y - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    x = np.clip(y - lam, 0.0, 1.0)
    r = target - float(x.sum())
    if abs(r) > 1e-14:
        if r > 0:
            order = np.argsort(-(x))  # most room (1-x) first
        else:
            order = np.argsort(x)  # least value first
        for i in order:
            if abs(r) < 1e-15:
                break
            if r > 0:
                add = min(r, 1.0 - x[i])
                x[i] += add
                r -= add
            else:
                sub = min(-r, x[i])
                x[i] -= sub
                r += sub
    return x

def gradient_lse(h, weights):
    """Gradient of LSE(corr) with respect to h.
    corr has length 2n-1, weights has length 2n-1.
    """
    n = len(h)
    grad = np.zeros(n)
    for p in range(n):
        g = 0.0
        for idx in range(2 * n - 1):
            w = weights[idx]
            if w < 1e-12:
                continue
            k = idx - (n - 1)
            # Term 1: (1-h[p-k]) if 0 <= p-k < n
            j1 = p - k
            if 0 <= j1 < n:
                g += w * (1.0 - h[j1])
            # Term 2: -h[p+k] if 0 <= p+k < n
            j2 = p + k
            if 0 <= j2 < n:
                g -= w * h[j2]
        grad[p] = g
    return grad

def optimize_continuous(h0, time_budget):
    """Projected gradient descent with LSE smoothing."""
    h = h0.copy().astype(float)
    n = len(h)
    target = n / 2.0
    h = project_box_sum(h, target)
    
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    t_start = time.time()
    beta = 20.0
    lr = 0.005
    best_time = 0.0
    
    iteration = 0
    while time.time() - t_start < time_budget:
        corr = compute_corrs(h)
        m = corr.max()
        # Compute weights
        shifted = np.clip(beta * (corr - m), -500, 0)
        exp_s = np.exp(shifted)
        exp_sum = exp_s.sum()
        if exp_sum < 1e-300:
            weights = np.zeros_like(corr)
            weights[np.argmax(corr)] = 1.0
        else:
            weights = exp_s / exp_sum
        
        grad = gradient_lse(h, weights)
        
        # Gradient step with projection
        h_new = h - lr * grad
        h_new = project_box_sum(h_new, target)
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-10:
            best_c5 = c5_new
            best_h = h_new.copy()
        
        h = h_new
        iteration += 1
        
        # Adaptive parameters
        if iteration % 100 == 0:
            # Slowly increase beta and decrease lr
            beta *= 1.2
            lr *= 0.95
            if lr < 1e-6:
                lr = 0.001
                beta = 20.0  # Reset for a "restart"
        
        # Periodic restart
        if iteration % 500 == 0:
            lr = 0.005 * (1.0 + 0.5 * np.random.random())
            beta = 10.0 + 50.0 * np.random.random()
    
    return best_h, best_c5

def binary_local_search(h, max_iterations=10000):
    """Local search for binary sequence. h is int array of 0s and 1s."""
    n = len(h)
    h = h.copy()
    corr = compute_corrs(h.astype(float))
    max_val = corr.max()
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    for it in range(max_iterations):
        best_improvement = -np.inf
        best_pair = None
        best_new_corr = None
        
        for a_pos in ones:
            for b_pos in zeros:
                if a_pos == b_pos:
                    continue
                # Compute change in corr
                a = a_pos
                b = b_pos
                delta = np.zeros(2 * n - 1)
                
                # Term 1: -(1-h[a-k]) for k where 0<=a-k<n
                k_min1 = max(-(n-1), a - (n-1))  # a-k >= 0 => k <= a
                k_max1 = min(n-1, a)  # a-k <= n-1 => k >= a-(n-1)
                # index = k + (n-1)
                for k in range(k_min1, k_max1 + 1):
                    j = a - k
                    delta[k + (n-1)] += -(1 - h[j])
                
                # Term 2: h[a+k] for k where 0<=a+k<n
                k_min2 = max(-(n-1), -a)
                k_max2 = min(n-1, n-1-a)
                for k in range(k_min2, k_max2 + 1):
                    j = a + k
                    delta[k + (n-1)] += h[j]
                
                # Term 3: (1-h[b-k]) for k where 0<=b-k<n
                k_min3 = max(-(n-1), b - (n-1))
                k_max3 = min(n-1, b)
                for k in range(k_min3, k_max3 + 1):
                    j = b - k
                    delta[k + (n-1)] += (1 - h[j])
                
                # Term 4: -h[b+k] for k where 0<=b+k<n
                k_min4 = max(-(n-1), -b)
                k_max4 = min(n-1, n-1-b)
                for k in range(k_min4, k_max4 + 1):
                    j = b + k
                    delta[k + (n-1)] += -h[j]
                
                new_corr = corr + delta
                new_max = new_corr.max()
                improvement = max_val - new_max
                
                if improvement > best_improvement:
                    best_improvement = improvement
                    best_pair = (a, b)
                    best_new_corr = new_corr
        
        if best_improvement > 1e-10 and best_pair is not None:
            a, b = best_pair
            h[a] = 0
            h[b] = 1
            corr = best_new_corr
            max_val = corr.max()
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
        else:
            break
    
    return h
