def repair_feasibility(h, n, tol=1e-10):

            h = np.clip(h, 0.0, 1.0)
    return h

def exact_score(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair_feasibility(h, n):
    mass = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(5):
        residual = mass - np.sum(h)
        if abs(residual) < 1e-12:
            break
        if residual > 0:
            slack = np.maximum(0.0, 1.0 - h)
            total = np.sum(slack)
            if total < 1e-12:
                break
            h += residual * slack / total
        else:
            afford = np.maximum(0.0, h)
            total = np.sum(afford)
            if total < 1e-12:
                break
            h += residual * afford / total  # residual is negative
        h = np.clip(h, 0.0, 1.0)
    return h

                        candidates.add(round(t_int, 12))
            else:
                disc = dq1*dq1 - 4.0*dq2*dq0
                if disc >= 0:
                    sd = np.sqrt(disc)
                    t1 = (-dq1 + sd) / (2.0*dq2)
                    t2 = (-dq1 - sd) / (2.0*dq2)
                    if t_low <= t1 <= t_high:
                        candidates.add(round(t1, 12))
                    if t_low <= t2 <= t_high:
                        candidates.add(round(t2, 12))
    
    candidates = np.array(list(candidates))
    candidates = np.clip(candidates, t_low, t_high)
    
    # Evaluate
    F_cand = q0[None,:] + candidates[:,None]*q1[None,:] + (candidates[:,None])**2 * q2[None,:]
    F_cand_max = np.max(F_cand, axis=1)
    best_ci = np.argmin(F_cand_max)
    best_t = float(candidates[best_ci])
    
    h_new = h + best_t * u
    h_new = repair_feasibility(h_new, n)
    return best_t, h_new

def subgradient_direction(h, n, rng):
    """Compute subgradient-based direction to reduce max lags"""
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = np.max(corr)
    # Active lags: within 5% of max
    threshold = max_val - 0.02 * max(1.0, abs(max_val))
    active = np.where(corr >= threshold)[0]
    
    grad = np.zeros(n)
    for i in active:
        lo = max(0, n-1-i)
        hi = min(n-1, 2*n-2-i)
        if lo > hi:
            continue
        j_range = np.arange(lo, hi+1)
        p1 = j_range + i - n + 1  # shifted indices
        grad[p1] += (1.0 - h[j_range])
        grad[j_range] += -h[p1]
    
    grad /= max(1, len(active))
    # Direction is negative subgradient
    u = -grad
    u -= np.mean(u)
    norm = np.linalg.norm(u)
    if norm < 1e-10:
        return make_random_direction(n, rng)
    return u / norm

def make_random_direction(n, rng):
    u = rng.standard_normal(n)
    u -= np.mean(u)
    norm = np.linalg.norm(u)
    if norm > 0:
        u /= norm
    return u

def make_smooth_direction(n, rng):
    x = np.arange(n, dtype=float)
    k = rng.integers(1, max(2, n//3))
    phase = rng.uniform(0, 2*np.pi)
    u = np.sin(2*np.pi*k*x/n + phase)
    u -= np.mean(u)
    norm = np.linalg.norm(u)
    if norm > 0:
        u /= norm
    return u
