import numpy as np
import time


def _score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)


def _pair_obj(h, i, j, t):
    hi = h[i] + t
    hj = h[j] - t
    if hi < -1e-12 or hi > 1.0 + 1e-12 or hj < -1e-12 or hj > 1.0 + 1e-12:
        return float('inf')
    h[i] = hi
    h[j] = hj
    n = len(h)
    dx = 2.0 / n
    result = float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)
    h[i] = hi - t
    h[j] = hj + t
    return result


def _golden_min(h, i, j, lo, hi, max_iter=50):
    """Golden section search for min of pair_obj(h, i, j, t) on [lo, hi]."""
    if hi - lo < 1e-14:
        return 0.0, _pair_obj(h, i, j, 0.0)
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = _pair_obj(h, i, j, c)
    fd = _pair_obj(h, i, j, d)
    
    for _ in range(max_iter):
        if abs(hi - lo) < 1e-12:
            break
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = _pair_obj(h, i, j, c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = _pair_obj(h, i, j, d)
    
    x = (lo + hi) / 2.0
    return x, _pair_obj(h, i, j, x)


def _compute_gradients(h, active_shifts):
    """Compute gradients for given shifts. Returns array of shape (len(active_shifts), n)."""
    n = len(h)
    grads = np.zeros((len(active_shifts), n))
    for idx, s in enumerate(active_shifts):
        p = abs(s)
        if p == 0:
            grads[idx, :] = 1.0 - 2.0 * h
        elif p < n:
            grads[idx, :n-p] += (1.0 - h[p:])
            grads[idx, p:] -= h[:n-p]
    return grads


def _repair(h):
    """Ensure feasibility: 0<=h<=1, sum(h)=n/2."""
    n = len(h)
    h = np.clip(h.astype(np.float64), 0.0, 1.0).copy()
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-10:
        return h
    for _ in range(200):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            avail = np.where(h < 1.0 - 1e-12)[0]
            if len(avail) == 0:
                break
            step = min(diff / len(avail), (1.0 - h[avail].max()) / 2.0)
            if step <= 0:
                break
            h[avail] += step
        else:
            avail = np.where(h > 1e-12)[0]
            if len(avail) == 0:
                break
            step = min(-diff / len(avail), h[avail].min() / 2.0)
            if step <= 0:
                break
            h[avail] -= step
    h = np.clip(h, 0.0, 1.0)
    # Final adjustment
    diff = target - h.sum()
    if abs(diff) > 1e-10:

        if mask.sum() > 0:
            per = diff / mask.sum()
            h[mask] = np.clip(h[mask] + per, 0, 1)

    n = len(h)
