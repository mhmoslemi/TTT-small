import numpy as np
from scipy.optimize import minimize
import time

def exact_objective(h):
    n = len(h)
    if n == 0:
        return float('inf')
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def is_feasible(h, tol=1e-9):
    n = len(h)
    if n == 0:
        return False
    if not np.all(np.isfinite(h)):
        return False
    if np.min(h) < -1e-10 or np.max(h) > 1.0 + 1e-10:
        return False
    if abs(np.sum(h) - n / 2.0) > tol:
        return False
    return True

def project_feasible(h_raw):
    """Project h_raw to feasible set: 0<=h<=1, sum(h)=n/2"""
    n = len(h_raw)
    target = n / 2.0
    h = np.clip(h_raw, 0.0, 1.0)
    # Water fill to achieve exact sum
    # Use bisection on lambda: sum(clip(h_raw + lambda, 0, 1)) = target
    lo = -max(h_raw) - 1.0
    hi = 1.0 - min(h_raw) + 1.0
    
    def f(lam):
        return np.sum(np.clip(h_raw + lam, 0.0, 1.0)) - target
    
    flo, fhi = f(lo), f(hi)
    if flo > 0:
        # All already above target sum, need to decrease
        # This shouldn't happen if h_raw is reasonable
        h = np.ones(n)
        excess = np.sum(h) - target
        # Reduce from largest
        idx = np.argsort(h)[::-1]
        for i in idx:
            dec = min(excess, h[i])
            h[i] -= dec
            excess -= dec
            if excess < 1e-15:
                break
        return h
    if fhi < 0:
        # All already below target sum
        h = np.zeros(n)
        deficit = target - np.sum(h)
        idx = np.argsort(h_raw)  # fill smallest first? No, fill to reach target
        # Actually just set all to 1 until we reach target
        idx = np.argsort(-h_raw)  # preferentially fill where h_raw is large
        for i in idx:
            inc = min(deficit, 1.0 - h[i])
            h[i] += inc
            deficit -= inc
            if deficit < 1e-15:
                break
        return h
    
    # Bisection
    for _ in range(80):
        mid = (lo + hi) / 2.0
        fm = f(mid)
        if abs(fm) < 1e-12:
            break
        if fm > 0:
            hi = mid
        else:
            lo = mid
    
    lam = (lo + hi) / 2.0
    h = np.clip(h_raw + lam, 0.0, 1.0)
    
    # Fix residual
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            idx = np.argsort(1.0 - h)  # most headroom first
        else:
            idx = np.argsort(h)  # smallest first
        for i in idx:
            if residual > 0:
                add = min(residual, 1.0 - h[i])
                h[i] += add
                residual -= add
            else:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-14:
                break
    
    return h

def directional_values(h):
    """Compute all directional overlap values. Returns array of length 2n-1."""
    n = len(h)
    # V_0
    v0 = np.sum(h * (1.0 - h))
    # V_m^+ for m=1..n-1
    vals = [v0]
    for m in range(1, n):
        vp = np.sum(h[:n-m] * (1.0 - h[m:]))
        vn = np.sum(h[m:] * (1.0 - h[:n-m]))
        vals.append(vp)
        vals.append(vn)
    return np.array(vals)

def build_weights(vals, K=None, decay=0.15):
    """Build weights for top K directional values."""
    n = len(h)  # Not available here, pass n separately? No, vals has length 2n-1
    L = len(vals)
    if K is None:
        n = (L + 1) // 2
        K = max(4, min(18, n // 3 + 4))
    
    K = min(K, L)
    
    # Sort descending
    sorted_idx = np.argsort(-vals, kind='stable')
    
    # Find tie threshold
    thresh = vals[sorted_idx[min(K-1, L-1)]]
    tie_tol = max(1e-6, 0.01 * max(1.0, vals[0] - thresh))
    
    # Include ties
    active_mask = vals >= thresh - tie_tol
    active_count = int(np.sum(active_mask))
    active_count = min(active_count, L)
    
    # Build weights
    weights = np.zeros(L)
    active_indices = np.where(active_mask)[0]
    # Sort active by value descending
    active_sorted = active_indices[np.argsort(-vals[active_indices], kind='stable')]
    
    for rank, idx in enumerate(active_sorted):
        weights[idx] = np.exp(-decay * rank)
    
    total = np.sum(weights)
    if total > 1e-15:
        weights /= total
    else:
        weights = np.ones(L) / L
    
    return weights

def weighted_gradient(h, weights):
    """Compute gradient of weighted sum of directional values."""
    n = len(h)
    g = np.zeros(n)
    
    # V_0 = sum h_i(1-h_i), gradient: 1-2h_i
    p0 = weights[0]
    g += p0 * (1.0 - 2.0 * h)
    
    # V_m^+ = sum_{i=0}^{n-m-1} h_i (1-h_{i+m}), weight = weights[2m-1]
    # V_m^- = sum_{i=0}^{n-m-1} h_{i+m} (1-h_i), weight = weights[2m]
    for m in range(1, n):
        wp = weights[2*m - 1]
        wn = weights[2*m]
        
        if wp > 1e-15:
            # dV_m^+/dh_j:
            # For j in 0..n-m-1: += (1 - h_{j+m})
            # For j in m..n-1: += -h_{j-m}
            g[:n-m] += wp * (1.0 - h[m:])
            g[m:] += -wp * h[:n-m]
        
        if wn > 1e-15:
            # dV_m^-/dh_j:
            # For j in m..n-1: += (1 - h_{j-m})
            # For j in 0..n-m-1: += -h_{j+m}
            g[m:] += wn * (1.0 - h[:n-m])
            g[:n-m] += -wn * h[m:]
    
    return g

def build_hessian(h, weights, edge_scale=1.0, rho=None):
    """Build Hessian approximation: graph Laplacian + rho*I"""
    n = len(h)
    M = np.zeros((n, n))
    
    # V_0 contribution: second derivative of h_i(1-h_i) is -2 (negative!)
    # For the quadratic model we want PD, so use +2 instead (or just rely on rho*I)
    # Actually for the trust region model, M should be PSD. The true Hessian of V_0 has
    # diagonal entries -2*p0 which is negative. So we can't use the true Hessian.
    # Instead, use a graph Laplacian structure which is PSD.
    
    # Add Laplacian edges for active directions
    for m in range(1, n):
        wp = weights[2*m - 1]
        wn = weights[2*m]
        w = edge_scale * (wp + wn)
        if w > 1e-15:
            for i in range(n - m):
                j = i + m
                M[i, i] += w
                M[j, j] += w
                M[i, j] -= w
                M[j, i] -= w
    
    # Add V_0 self-term (use positive for PD)
    p0 = weights[0]
    if p0 > 1e-15:
        M[np.diag_indices(n)] += p0 * 2.0  # Positive for PD
    
    # Add identity for conditioning
    if rho is None:
        g = weighted_gradient(h, weights)
        gnorm = np.max(np.abs(g))
        rho = max(1e-6, gnorm / (2 * 0.03 + 1e-6))
    M[np.diag_indices(n)] += rho
    
    return M

def solve_subproblem(h, g, M, trust):
    """Solve the trust region subproblem using SLSQP."""
    n = len(h)
    
    # Objective: g^T d + 0.5 d^T M d
    def obj(d):
        return g @ d + 0.5 * d @ M @ d
    
    def grad_obj(d):
        return g + M @ d
    
    # Constraints: sum(d) = 0
    # Bounds: max(-h_i, -trust) <= d_i <= min(1-h_i, trust)
    lo = np.maximum(-h, -trust)
    hi = np.minimum(1.0 - h, trust)
    
    # Ensure lo <= hi
    hi = np.maximum(hi, lo)
    
    bounds = list(zip(lo, hi))
    
    # Equality constraint: sum(d) = 0
    cons = ({'type': 'eq', 'fun': lambda d: np.sum(d), 'jac': lambda d: np.ones(n)})
    
    # Start from 0
    d0 = np.zeros(n)
    
    try:
        res = minimize(obj, d0, jac=grad_obj, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': 50, 'ftol': 1e-10})
        if res.success or (hasattr(res, 'fun') and np.isfinite(res.fun)):
            d = res.x
            # Verify sum constraint
            d -= np.mean(d)  # Project to sum=0
            # Clip to bounds
            d = np.clip(d, lo, hi)
            d -= np.mean(d)  # Re-project
            return d
        else:
            return None
    except Exception:
        return None

def fallback_direction(g, h, trust):
    """Fallback: use -gradient projected to feasible direction."""
    d = -g
    d -= np.mean(d)
    scale = np.max(np.abs(d))
    if scale > 1e-15:
        d = d / scale * trust * 0.5
    d = np.clip(d, np.maximum(-h, -trust), np.minimum(1.0 - h, trust))
    d -= np.mean(d)
    return d

def run_sqp(h_start, time_limit, params=None, seed=42):
    """Run SQP local optimization from h_start."""
    if params is None:
        params = {}
    
    K = params.get('K', 10)
    decay = params.get('decay', 0.15)
    edge_scale = params.get('edge_scale', 1.0)
    init_trust = params.get('init_trust', 0.04)
    trust_floor = params.get('trust_floor', 0.001)
    trust_growth = params.get('trust_growth', 1.3)
    trust_shrink = params.get('trust_shrink', 0.5)
    
    h = h_start.copy().astype(np.float64)
    c_current = exact_objective(h)
    c_best_local = c_current
    h_best_local = h.copy()
    
    trust = init_trust
    n_fails = 0
    max_fails = 10
    
    rng = np.random.RandomState(seed)
    
    while time.time() < time_limit:
        n = len(h)
        
        # Compute directional values and weights
        vals = directional_values(h)
        weights = build_weights(vals, K=K, decay=decay)
        
        # Compute gradient
        g = weighted_gradient(h, weights)
        gnorm = np.linalg.norm(g)
        
        if gnorm < 1e-10:
            # Try random perturbation
            h_new = h + rng.randn(n) * 0.01
            h_new = project_feasible(h_new)
            c_new = exact_objective(h_new)
            if c_new < c_current:
                h = h_new
                c_current = c_new
                if c_new < c_best_local:
                    c_best_local = c_new
                    h_best_local = h.copy()
            n_fails += 1
            if n_fails > max_fails:
                break
            continue
        
        # Build Hessian and solve
        M = build_hessian(h, weights, edge_scale=edge_scale)
        d = solve_subproblem(h, g, M, trust)
        
        if d is None or np.max(np.abs(d)) < 1e-12:
            # Fallback direction
            d = fallback_direction(g, h, trust)
        
        if np.max(np.abs(d)) < 1e-12:
            n_fails += 1
            trust *= trust_shrink
            if trust < trust_floor:
                break
            continue
        
        # Line search
        alphas = [1.0, 0.7, 0.5, 0.3, 0.15, 0.07, 0.03]
        accepted = False
        
        for alpha in alphas:
            h_try = h + alpha * d
            h_try = project_feasible(h_try)
            c_try = exact_objective(h_try)
            
            if c_try < c_current - 1e-12:
                h = h_try
                c_current = c_try
                accepted = True
                n_fails = 0
                trust = min(trust * trust_growth, 0.2)
                
                if c_try < c_best_local:
                    c_best_local = c_try
                    h_best_local = h.copy()
                break
            elif c_try < c_current + 1e-4 and not accepted:
                # Allow slight worsening for exploration
                pass
        
        if not accepted:
            n_fails += 1
            trust *= trust_shrink
            if trust < trust_floor:
                # Try random escape
                h_new = h + rng.randn(n) * 0.015
                h_new = project_feasible(h_new)
                c_new = exact_objective(h_new)
                if c_new < c_best_local:
                    h = h_new
                    c_current = c_new
                    c_best_local = c_new
                    h_best_local = h.copy()
                if n_fails > max_fails:
                    break
            elif n_fails > max_fails:
                break
    
    return h_best_local, c_best_local

def resize_interpolate(h, new_n):
    """Interpolate h to new length new_n."""
    n0 = len(h)
    if n0 == new_n:
        return h.copy()
    x0 = (np.arange(n0) + 0.5) / n0
    x1 = (np.arange(new_n) + 0.5) / new_n
    h_new = np.interp(x1, x0, h)
    return project_feasible(h_new)

def generate_variants(h, rng):
    """Generate variants of h for search."""
    variants = [h.copy()]
    
    # Reverse
    variants.append(h[::-1].copy())
    
    # Circular shifts
    for s in [1, 2]:
        variants.append(np.roll(h, s).copy())
    
    # Random perturbation
    for sigma in [0.01, 0.02, 0.03]:
        vp = h + rng.randn(len(h)) * sigma
        variants.append(project_feasible(vp))
    
    # Quantize to 0/1 with threshold variations
    for thresh in [0.4, 0.5, 0.6]:
        vq = np.where(h > thresh, 1.0, 0.0)
        vq = project_feasible(vq)
        variants.append(vq)
    
    return variants

def make_fallback_h(n):
    """Create a trivially feasible h of length n."""
    target = n / 2.0
    h = np.zeros(n)
    flo = int(np.floor(target))
    frac = target - flo
    for i in range(flo):
        h[i] = 1.0
    if frac > 1e-15:
        h[flo] = frac
    return h

def directional_values(h):
    n = len(h)
    v0 = np.sum(h * (1.0 - h))
    vals = [v0]
    for m in range(1, n):
        vp = np.sum(h[:n - m] * (1.0 - h[m:]))
        vn = np.sum(h[m:] * (1.0 - h[:n - m]))
        vals.append(vp)
        vals.append(vn)
    return np.array(vals)


def build_weights(vals, K, decay):
    L = len(vals)
    K = min(K, L)
    sorted_idx = np.argsort(-vals, kind='stable')
    thresh = vals[sorted_idx[K - 1]]
    tie_tol = max(1e-6, 0.05 * max(1.0, vals[0] - thresh))
    active_mask = vals >= (thresh - tie_tol)
    active_indices = np.where(active_mask)[0]
    active_sorted = active_indices[np.argsort(-vals[active_indices], kind='stable')]
    weights = np.zeros(L)
    for rank, idx in enumerate(active_sorted):
        weights[idx] = np.exp(-decay * rank)
    total = np.sum(weights)
    if total > 1e-15:
        weights /= total
    else:
        weights = np.ones(L) / L
    return weights

        wn = weights[2 * m]
        if wp > 1e-15:
            g[:n - m] += wp * (1.0 - h[m:])
            g[m:] += -wp * h[:n - m]
        if wn > 1e-15:
            g[m:] += wn * (1.0 - h[:n - m])
            g[:n - m] += -wn * h[m:]
    return g


def build_hessian(h, weights, edge_scale):
    n = len(h)
    M = np.zeros((n, n))
    p0 = weights[0]
    if p0 > 1e-15:
        M[np.diag_indices(n)] += 2.0 * p0
    for m in range(1, n):
        wp = weights[2 * m - 1]
        wn = weights[2 * m]
        w = edge_scale * (wp + wn)
        if w > 1e-15:
            for i in range(n - m):
                j = i + m
                M[i, i] += w
                M[j, j] += w
                M[i, j] -= w
                M[j, i] -= w
    g = weighted_gradient(h, weights)
    gnorm = max(np.max(np.abs(g)), 1e-10)
    rho = gnorm * 0.5
    M[np.diag_indices(n)] += rho
    return M


def _qp_obj(d, g, M):
    return float(g @ d + 0.5 * d @ M @ d)


def _qp_grad(d, g, M):
    return g + M @ d


def _eq_fun(d):
    return np.array([np.sum(d)])


def _eq_jac(d):
    return np.ones((1, len(d)))


def solve_subproblem(h, g, M, trust):
    n = len(h)
    lo = np.maximum(-h, -trust)
    hi = np.minimum(1.0 - h, trust)
    hi = np.maximum(hi, lo)
    bounds = list(zip(lo, hi))
    cons = {'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}
    d0 = np.zeros(n)
    try:
        res = minimize(_qp_obj, d0, args=(g, M), jac=lambda d: _qp_grad(d, g, M),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': 40, 'ftol': 1e-12})
        if hasattr(res, 'x') and np.all(np.isfinite(res.x)):
            d = res.x.copy()
            d -= np.mean(d)
            d = np.clip(d, lo, hi)
            d -= np.mean(d)
            return d
    except Exception:
        pass
    return None


def fallback_direction(g, h, trust):
    d = -g
    d -= np.mean(d)
    scale = np.max(np.abs(d))
    if scale > 1e-12:
        d = d / scale * trust * 0.5
    lo = np.maximum(-h, -trust)
    hi = np.minimum(1.0 - h, trust)
    d = np.clip(d, lo, hi)
    d -= np.mean(d)
    return d


def run_sqp(h_start, t_deadline, K, decay, edge_scale, init_trust, seed):
    h = h_start.copy().astype(np.float64)
    c_current = exact_objective(h)
    c_best = c_current
    h_best = h.copy()
    trust = init_trust
    n_fails = 0
    rng = np.random.RandomState(seed)
    n = len(h)

    while time.time() < t_deadline:
        vals = directional_values(h)
        weights = build_weights(vals, K, decay)
        g = weighted_gradient(h, weights)
        gnorm = np.linalg.norm(g)

        if gnorm < 1e-10:
            h_new = project_feasible(h + rng.randn(n) * 0.01)
            c_new = exact_objective(h_new)
            if c_new < c_current - 1e-12:
                h, c_current = h_new, c_new
                if c_new < c_best:
                    c_best, h_best = c_new, h.copy()
            n_fails += 1
            if n_fails > 12:
                break
            continue

        M = build_hessian(h, weights, edge_scale)
        d = solve_subproblem(h, g, M, trust)
        if d is None or np.max(np.abs(d)) < 1e-12:
            d = fallback_direction(g, h, trust)
        if np.max(np.abs(d)) < 1e-12:
            n_fails += 1
            trust *= 0.5
            if trust < 0.0005:
                break
            continue

        alphas = [1.0, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05]
        accepted = False
        for alpha in alphas:
            h_try = project_feasible(h + alpha * d)
            c_try = exact_objective(h_try)
            if c_try < c_current - 1e-12:
                h, c_current = h_try, c_try
                accepted = True
                n_fails = 0
                trust = min(trust * 1.3, 0.2)
                if c_try < c_best:
                    c_best, h_best = c_try, h.copy()
                break

        if not accepted:
            n_fails += 1
            trust *= 0.6
            if trust < 0.0005:
                h_esc = project_feasible(h + rng.randn(n) * 0.02)
                c_esc = exact_objective(h_esc)
                if c_esc < c_best:
                    h, c_current = h_esc, c_esc
                    c_best, h_best = c_esc, h.copy()
                if n_fails > 12:
                    break

    return h_best, c_best


def resize_interpolate(h, new_n):
    n0 = len(h)
    if n0 == new_n:
        return h.copy()
    x0 = (np.arange(n0) + 0.5) / n0
    x1 = (np.arange(new_n) + 0.5) / new_n
    h_new = np.interp(x1, x0, h)
    return project_feasible(h_new)


def make_fallback_h(n):
    target = n / 2.0
    h = np.zeros(n)
    flo = int(np.floor(target))
    frac = target - flo
    for i in range(flo):
        h[i] = 1.0
    if frac > 1e-14:
        h[flo] = frac
    return h


def update_best(best_h, best_c, h_cand):
    h_cand = project_feasible(h_cand)
    if not is_feasible(h_cand):
        return best_h, best_c
    c_cand = exact_objective(h_cand)
    if c_cand < best_c - 1e-12:
        return h_cand.copy(), c_cand
    if abs(c_cand - best_c) <= 1e-12 and len(h_cand) < len(best_h):
        return h_cand.copy(), c_cand
    return best_h, best_c
